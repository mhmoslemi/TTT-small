def subgradient_polish(a, max_iters, deadline, lb=1.45):
    """Subgradient descent with Polyak step size."""
    a = np.maximum(a.astype(np.float64), 1e-10)
    a = a / a.sum()
    n = len(a)
    best_a = a.copy()
    best_M = fast_conv(a).max()
    running_sum = np.zeros(n)
    count = 0

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        b = fast_conv(a)
        M = float(b.max())

        if M < best_M * (1 - 1e-9):
            best_M = M
            best_a = a.copy()

        # Polyak step size: (f(x) - f*) / ||g||^2
        gap = M - lb
        if gap < 1e-10:
            gap = 1e-8

        # Compute subgradient (average over active lags)
        tol = 1e-6 * M
        active = np.where(b >= M - tol)[0]
        grad = np.zeros(n)
        for k in active:
            lo = max(0, k - n + 1)
            hi = min(n - 1, k)
            grad[lo:hi + 1] += 2.0 * a[k - hi:k - lo + 1][::-1]
        grad /= len(active)

        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-15:
            break

        # Polyak step
        step = gap / (gnorm * gnorm)
        step = min(step, 1.0 / (it + 1))  # cap for safety

        a_new = a - step * grad
        a_new = np.maximum(a_new, 0)
        s = a_new.sum()
        if s < 1e-10:
            a_new = a.copy()
        else:
            a_new = a_new / s

        running_sum += a_new
        count += 1
        a = a_new

    # Check averaged solution
    if count > 20:
        a_avg = running_sum / count
        a_avg = np.maximum(a_avg, 0)
        s = a_avg.sum()
        if s > 1e-10:
            a_avg = a_avg / s
            M_avg = fast_conv(a_avg).max()
            if M_avg < best_M:
                best_a = a_avg

    return best_a


def coordinate_descent(a, max_sweeps, deadline, symmetric=False):
    """
    Efficient coordinate descent exploiting convolution structure.
    Changing a[i] by delta changes conv[k] by 2*delta*a[k-i] (for valid k-i).
    We find optimal delta by 1D minimization.
    """
    a = np.maximum(a.astype(np.float64), 0)
    s = a.sum()
    if s < 1e-10:
        return a
    a = a / s
    n = len(a)
    best_a = a.copy()
    best_M = fast_conv(a).max()

    if symmetric:
        indices = list(range(n // 2 + 1))
    else:
        indices = list(range(n))

    # Shuffle for better convergence
    rng = np.random.default_rng(12345)

    for sweep in range(max_sweeps):
        if time.monotonic() > deadline:
            break

        rng.shuffle(indices)

        for i in indices:
            b = fast_conv(a)
            M = b.max()

            # Influence vector: conv change per unit increase in a[i]
            # influence[k] = 2*a[k-i] if 0<=k-i<n, else 0
            k_lo = i
            k_hi = i + n - 1
            influence = np.zeros(2 * n - 1)
            influence[k_lo:k_hi + 1] = 2.0 * a[0:n]

            # For symmetric case, also include influence from a[n-1-i]
            if symmetric and n - 1 - i != i:
                j = n - 1 - i
                k_lo2 = j
                k_hi2 = j + n - 1
                influence[k_lo2:k_hi2 + 1] += 2.0 * a[0:n]

            # We want to find delta minimizing max(b + delta * influence)
            # subject to a[i] + delta >= 0 (and a[n-1-i] + delta >= 0 if symmetric)
            # Since influence >= 0, the max is non-decreasing in delta
            # So optimal delta <= 0 (decrease a[i]) unless a[i] = 0

            if a[i] < 1e-10:
                # Try small increase
                for d in [0.002, 0.005, 0.01, 0.02]:
                    a_trial = a.copy()
                    a_trial[i] = d
                    if symmetric and n - 1 - i != i:
                        a_trial[n - 1 - i] = d
                    a_trial = np.maximum(a_trial, 0)
                    st = a_trial.sum()
                    if st < 1e-10:
                        continue
                    a_trial = a_trial / st
                    M_trial = fast_conv(a_trial).max()
                    if M_trial < M:
                        a = a_trial
                        break
                continue

            # Find max negative delta: b[k] + delta*influence[k] <= M for all k
            # delta <= (M - b[k]) / influence[k] for all k with influence[k] > 0
            pos = influence > 1e-12
            if not np.any(pos):
                continue

            ratios = (M - b[pos]) / influence[pos]
            max_neg_delta = np.min(ratios)

            # Constraint: can't go below 0
            if symmetric and n - 1 - i != i:
                max_dec = (a[i] + a[n - 1 - i]) / 2.0
            else:
                max_dec = a[i]

            if max_neg_delta < -1e-12:
                delta = max(max_neg_delta, -max_dec * 0.8)
            else:
                # Can't decrease; try not changing
                delta = 0.0

            if abs(delta) > 1e-12:
                a[i] += delta
                if symmetric and n - 1 - i != i:
                    a[n - 1 - i] += delta
                a = np.maximum(a, 0)
                st = a.sum()
                if st > 1e-10:
                    a = a / st

            M_new = fast_conv(a).max()
            if M_new < best_M * (1 - 1e-10):
                best_M = M_new
                best_a = a.copy()

    return best_a


def fourier_smooth(a, alpha=0.1):
    """
    Apply smoothing in Fourier domain: dampen high-frequency components
    of the signal to reduce convolution peaks.
    """
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    freqs = np.fft.rfftfreq(size)
    # Apply low-pass: keep DC, dampen high freqs
    mask = np.exp(-alpha * (freqs * size) ** 2)
    mask[0] = 1.0
    fa_smooth = fa * mask
    a_smooth = np.fft.irfft(fa_smooth, size)[:n]
    a_smooth = np.maximum(a_smooth, 0)
    s = a_smooth.sum()
    if s > 1e-10:
        a_smooth = a_smooth / s
    return a_smooth


def resample_seq(src, dest_n):
    """Resample 1D sequence to new length via linear interpolation."""
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)


class ScoreTracker:
    def __init__(self):
        self.best = None
        self.best_score = float('inf')
        self.t_start = time.monotonic()

    def update(self, a):
        a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
        if a.size == 0 or a.sum() < 0.01:
            return
        score = float(evaluate_sequence(a.tolist()))
        if np.isfinite(score) and score < self.best_score:
            self.best = a.copy()
            self.best_score = score
            elapsed = time.monotonic() - self.t_start
            print(f"  ** J={score:.8f} n={a.size} t={elapsed:.0f}s", flush=True)

    def report(self, label=""):
        if self.best is not None:
            print(f"  {label} best: J={self.best_score:.8f} n={self.best.size}", flush=True)


def generate_inits(n, rng, parent=None):
    """Generate diverse initial sequences."""
    inits = []
    t = np.linspace(0, 1, n, endpoint=False)

    if parent is not None:
        p = resample_seq(parent, n)
        inits.append(p.copy())
        inits.append(np.maximum(p + rng.normal(0, 0.02, n), 0.01))
        inits.append(np.maximum(p + rng.normal(0, 0.08, n), 0.01))
        # Symmetrize
        p_sym = (p + p[::-1]) / 2.0
        inits.append(np.maximum(p_sym, 0.01))

    inits.append(np.ones(n))
    inits.append(rng.uniform(0.5, 1.0, n))

    for freq in [1, 2, 3, 5]:
        inits.append(np.maximum(1.0 + 0.3 * np.cos(2 * np.pi * freq * t), 0.01))

    # Gaussian
    inits.append(np.exp(-2.0 * (t - 0.5) ** 2) + 0.2)
    # Raised cosine
    inits.append(0.5 + 0.5 * np.cos(2 * np.pi * t))
    inits.append(0.3 + 0.7 * np.cos(2 * np.pi * t) ** 2)
    # Power law
    inits.append(np.maximum(1.0 + 0.3 * t ** 2, 0.01))
    # Bump pair
    inits.append(np.exp(-8*(t-0.3)**2) + np.exp(-8*(t-0.7)**2) + 0.1)

    return inits


def propose_candidate(seed=42, budget_s=1500, **kwargs):
    """
    Multi-phase search:
    Phase 1: LP Frank-Wolfe with exact line search (primary)
    Phase 2: Coordinate descent polish
    Phase 3: Subgradient with Polyak step
    Phase 4: Multi-length exploration
    Phase 5: Final intensive LP
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + float(budget_s) - 20.0
    tracker = ScoreTracker()

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        if p.ndim == 1 and p.size > 0 and np.all(np.isfinite(p)):
            parent = np.maximum(p, 0.0)
            tracker.update(parent)
            print(f"Parent: n={parent.size}, J={tracker.best_score:.8f}", flush=True)
    except (NameError, AttributeError):
        pass

    if parent is None:
        parent = np.ones(256)
        tracker.update(parent)

    current_best = parent.copy()
    print(f"\n{'='*60}", flush=True)
    print(f"START: J={tracker.best_score:.8f} n={current_best.size}", flush=True)
    print(f"{'='*60}\n", flush=True)

    # === PHASE 1: Intensive LP Frank-Wolfe on the best length ===
    p1_deadline = min(time.monotonic() + 400, deadline - 200)
    print(f"PHASE 1: LP Frank-Wolfe (deadline in {p1_deadline - time.monotonic():.0f}s)", flush=True)

    n_main = current_best.size
    a = current_best.copy()
    a = a / a.sum()

    # Run LP Frank-Wolfe with exact line search
    a = frank_wolfe_lp(a, max_iters=2000, deadline=p1_deadline, use_ls=True)
    tracker.update(a)

    # Run again from the result (second pass often helps)
    if time.monotonic() < deadline - 150:
        p1b_deadline = min(time.monotonic() + 200, deadline - 100)
        a2 = frank_wolfe_lp(a, max_iters=1000, deadline=p1b_deadline, use_ls=False)
        tracker.update(a2)
        # Also try with line search
        a3 = frank_wolfe_lp(a2, max_iters=500, deadline=min(time.monotonic() + 100, deadline - 50), use_ls=True)
        tracker.update(a3)

    print(f"  Phase 1 done: J={tracker.best_score:.8f}", flush=True)

    # === PHASE 2: Coordinate Descent Polish ===
    p2_deadline = min(time.monotonic() + 200, deadline - 80)
    print(f"\nPHASE 2: Coordinate Descent (deadline in {p2_deadline - time.monotonic():.0f}s)", flush=True)

    if tracker.best is not None:
        a_cd = coordinate_descent(tracker.best, max_sweeps=50, deadline=p2_deadline, symmetric=False)
        tracker.update(a_cd)

        a_cd_sym = coordinate_descent(tracker.best, max_sweeps=30,
                                      deadline=min(time.monotonic() + 100, deadline - 50),
                                      symmetric=True)
        tracker.update(a_cd_sym)

    print(f"  Phase 2 done: J={tracker.best_score:.8f}", flush=True)

    # === PHASE 3: Subgradient with Polyak ===
    p3_deadline = min(time.monotonic() + 100, deadline - 50)
    print(f"\nPHASE 3: Subgradient Polish (deadline in {p3_deadline - time.monotonic():.0f}s)", flush=True)

    if tracker.best is not None:
        # Use a lower bound on the optimal J/n ratio
        # J >= 2 (for uniform), better bound: use current best as upper, 1 as lower
        # For Polyak: use lb on M (i.e., M_opt), which is J_opt / (2n)
        # J_opt for n=512 is about 1.5027/2*512... no
        # J = 2*n*M/s^2, so M = J*s^2/(2*n). With s=1: M = J/(2*n)
        # J_opt ~ 1.50, n=512: M_opt ~ 1.50/(2*512) ~ 0.00146
        # Current M: J_current/(2*n)
        J_cur = tracker.best_score
        M_cur = J_cur / (2.0 * tracker.best.size)
        M_lb = 1.45 / (2.0 * tracker.best.size)  # optimistic lower bound

        a_sub = subgradient_polish(tracker.best, max_iters=5000, deadline=p3_deadline, lb=M_lb)
        tracker.update(a_sub)

    print(f"  Phase 3 done: J={tracker.best_score:.8f}", flush=True)

    # === PHASE 4: Multi-length exploration ===
    p4_deadline = min(time.monotonic() + 200, deadline - 30)
    print(f"\nPHASE 4: Multi-length Exploration (deadline in {p4_deadline - time.monotonic():.0f}s)", flush=True)

    if tracker.best is not None:
        base_n = tracker.best.size
        # Try nearby lengths
        for n_try in [base_n - 32, base_n, base_n + 32, base_n + 64]:
            if n_try < 64 or n_try > 2048 or time.monotonic() > p4_deadline:
                continue
            a_try = resample_seq(tracker.best, n_try)
            a_try = np.maximum(a_try, 1e-6)
            a_try = a_try / a_try.sum()

            # Quick LP Frank-Wolfe
            sub_deadline = min(time.monotonic() + 60, p4_deadline)
            a_try = frank_wolfe_lp(a_try, max_iters=300, deadline=sub_deadline, use_ls=True)
            tracker.update(a_try)

    print(f"  Phase 4 done: J={tracker.best_score:.8f}", flush=True)

    # === PHASE 5: Final intensive LP pass ===
    p5_deadline = deadline
    print(f"\nPHASE 5: Final LP Polish (deadline in {p5_deadline - time.monotonic():.0f}s)", flush=True)

    if tracker.best is not None and time.monotonic() < p5_deadline - 5:
        a_final = frank_wolfe_lp(tracker.best, max_iters=5000, deadline=p5_deadline - 5, use_ls=True)
        tracker.update(a_final)

        # Coordinate descent on the final result
        if time.monotonic() < p5_deadline - 2:
            a_cd_final = coordinate_descent(a_final, max_sweeps=20, deadline=p5_deadline - 2, symmetric=False)
            tracker.update(a_cd_final)

    print(f"\n{'='*60}", flush=True)
    print(f"FINAL: J={tracker.best_score:.8f} n={tracker.best.size} "
          f"time={time.monotonic()-tracker.t_start:.0f}s", flush=True)
    print(f"{'='*60}", flush=True)

    if tracker.best is None:
        tracker.best = np.ones(256)
        tracker.best_score = compute_J(tracker.best)

    return [float(x) for x in tracker.best]
