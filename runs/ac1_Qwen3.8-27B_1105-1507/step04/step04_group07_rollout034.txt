class Adam:
    def __init__(self, params, lr=0.001, beta1=0.9, beta2=0.999, eps=1e-8):
        self.m = [np.zeros_like(p) for p in params]
        self.v = [np.zeros_like(p) for p in params]
        self.t = 0
        self.lr = lr
        self.beta1 = beta1
        self.beta2 = beta2
        self.eps = eps
    
    def step(self, params, grads):
        self.t += 1
        for i, (p, g) in enumerate(zip(params, grads)):
            self.m[i] = self.beta1 * self.m[i] + (1 - self.beta1) * g
            self.v[i] = self.beta2 * self.v[i] + (1 - self.beta2) * g**2
            m_hat = self.m[i] / (1 - self.beta1**self.t)
            v_hat = self.v[i] / (1 - self.beta2**self.t)
            params[i] -= self.lr * m_hat / (np.sqrt(v_hat) + self.eps)

"""
Minimizes J(a) = 2*n*max(convolve(a,a))/(sum(a))^2 using:
1. Lightweight LP initialization
2. Adam LSE optimization with adaptive tau and spectral perturbations
3. Multi-n search with diverse initializations
4. Subgradient + coordinate descent polishing

import numpy as np
from scipy.optimize import linprog
from scipy.sparse import csr_matrix
import time

_spar_cache = {}

def get_sparsity(n):
    if n not in _spar_cache:
        rows = []
        cols = []
        for k in range(2 * n - 1):
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            rows.extend([k] * (je - js + 1))
            cols.extend(range(js, je + 1))
        _spar_cache[n] = (np.array(rows), np.array(cols), len(rows))
    return _spar_cache[n]

def fast_conv(a):
    n = len(a)
    s = 1
    while s < 2 * n - 1:
        s <<= 1
    fa = np.fft.rfft(a, s)
    return np.fft.irfft(fa * fa, s)[:2 * n - 1]

def compute_J(a):
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))

def lse_and_grad(a, tau):
    """Compute LSE of convolution and its gradient w.r.t. a."""
    n = len(a)
    s_fft = 1
    while s_fft < 2 * n - 1:
        s_fft <<= 1
    
    fa = np.fft.rfft(a, s_fft)
    b = np.fft.irfft(fa * fa, s_fft)[:2 * n - 1]
    
    b_max = b.max()
    shifted = tau * (b - b_max)
    # Use log-sum-exp for numerical stability
    max_shifted = shifted.max()
    lse = b_max + (max_shifted + np.log(np.exp(shifted - max_shifted).sum())) / tau
    
    exp_b = np.exp(shifted - max_shifted)
    p = exp_b / exp_b.sum()
    
    # Gradient: grad[j] = tau * sum_{m} a[m] * p[m+j]
    # This is tau * (a cross-correlate p)[:n]
    s_corr = 1
    while s_corr < 2 * n - 1 + n - 1:
        s_corr <<= 1
    fa_full = np.fft.rfft(a, s_corr)
    fp = np.fft.rfft(p, s_corr)
    corr = np.fft.irfft(np.conj(fa_full) * fp, s_corr)
    grad = tau * corr[:n]
    
    return lse, grad, b

def lp_best_response(a, M, time_limit=5):
    n = len(a)
    rows, cols, nnz = get_sparsity(n)
    vals = np.empty(nnz)
    for i in range(nnz):
        k = rows[i]
        j = cols[i]
        vals[i] = a[k - j]
    A = csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                options={'presolve': True, 'time_limit': time_limit})
    if r.success and r.x is not None:
        g = np.maximum(r.x, 0.0)
        if g.sum() > 1e-10:
            return g / g.sum()
    return None

def spectral_perturb(a, rng, strength=0.3):
    """Perturb the spectrum of a to escape local minima."""
    n = len(a)
    # Compute DFT
    A = np.fft.rfft(a)
    
    # Modify the magnitude with multiplicative noise
    log_mag = np.log(np.abs(A) + 1e-10)
    noise = rng.normal(0, strength, log_mag.shape)
    # Attenuate noise at high frequencies
    freqs = np.arange(len(log_mag)) / len(log_mag)
    attenuation = np.exp(-3 * freqs)
    log_mag += noise * attenuation
    
    # Reconstruct
    phase = np.angle(A)
    A_new = np.exp(log_mag) * np.exp(1j * phase)
    
    # Make full spectrum (conjugate symmetric)
    A_full = np.zeros(n, dtype=complex)
    half = n // 2 + 1
    A_full[:half] = A_new
    for k in range(1, n - n // 2):
        A_full[k] = np.conj(A_full[n - k])
    
    a_new = np.fft.ifft(A_full).real
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s < 1e-10:
        return a
    return a_new / s

def adam_lse_optimize(a, deadline, tracker, max_tau=200.0, initial_tau=10.0):
    """Adam optimization with LSE-smoothed objective."""
    n = len(a)
    a = a.copy().astype(np.float64)
    a = np.maximum(a, 1e-8)
    a /= a.sum()
    
    current_J = compute_J(a)
    tracker.update(a)
    
    # Adam state
    m = np.zeros(n)
    v = np.zeros(n)
    t = 0
    lr = 0.01 / np.sqrt(n)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    
    tau = initial_tau
    stuck_count = 0
    best_local_J = current_J
    
    while time.monotonic() < deadline:
        t += 1
        
        lse_val, grad, b = lse_and_grad(a, tau)
        
        # Adam update
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * grad ** 2
        m_hat = m / (1 - beta1 ** t)
        v_hat = v / (1 - beta2 ** t)
        
        a_new = a - lr * m_hat / (np.sqrt(v_hat) + eps)
        a_new = np.maximum(a_new, 1e-8)
        a_new /= a_new.sum()
        
        J_new = compute_J(a_new)
        
        if J_new < current_J - 1e-14:
            a = a_new
            current_J = J_new
            stuck_count = 0
            if current_J < best_local_J:
                best_local_J = current_J
                tracker.update(a)
            # Increase tau slowly
            tau = min(max_tau, tau * 1.005)
            # Slightly increase learning rate
            lr = min(0.1 / np.sqrt(n), lr * 1.001)
        else:
            stuck_count += 1
            lr *= 0.95
            tau = max(5.0, tau * 0.99)
            
            if stuck_count > 100:
                # Try spectral perturbation
                a = spectral_perturb(a, np.random.default_rng(t), strength=0.2)
                a /= a.sum()
                current_J = compute_J(a)
                m = np.zeros(n)
                v = np.zeros(n)
                t = 0
                lr = 0.01 / np.sqrt(n)
                stuck_count = 0
                tau = initial_tau
    
    tracker.update(a)
    return a

def exact_subgradient(a, deadline, tracker, max_iters=5000):
    """Subgradient descent on exact max objective."""
    n = len(a)
    a = a.copy().astype(np.float64)
    a = np.maximum(a, 1e-8)
    a /= a.sum()
    current_J = compute_J(a)
    tracker.update(a)
    
    lr = 0.3 / n
    rng = np.random.default_rng(123)
    
    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        
        b = fast_conv(a)
        M = b.max()
        
        # Find active set
        active = np.where(b >= M - 1e-8)[0]
        
        # Subgradient
        grad = np.zeros(n)
        for k in active:
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            for j in range(js, je + 1):
                grad[j] += a[k - j]
        grad /= len(active)
        
        # Direction: minimize LSE ~ max, so go opposite to subgradient
        # But we're minimizing max(b), so gradient of max(b) is the subgradient
        # We want to DECREASE max(b), so: a -= lr * grad
        a_new = a - lr * grad
        a_new = np.maximum(a_new, 1e-8)
        a_new /= a_new.sum()
        
        J_new = compute_J(a_new)
        if J_new < current_J - 1e-15:
            a = a_new
            current_J = J_new
            lr = min(lr * 1.02, 5.0 / n)
        else:
            lr *= 0.8
        
        if it % 100 == 0:
            tracker.update(a)
    
    tracker.update(a)
    return a

def coordinate_descent(a, deadline, tracker, max_steps=5000):
    """Random coordinate descent."""
    n = len(a)
    a = a.copy().astype(np.float64)
    a = np.maximum(a, 1e-8)
    a /= a.sum()
    current_J = compute_J(a)
    tracker.update(a)
    rng = np.random.default_rng(42)
    
    for step in range(max_steps):
        if time.monotonic() > deadline:
            break
        
        j = rng.integers(0, n)
        best_trial = None
        best_J = current_J
        
        for scale in [0.5, 1.5, 0.7, 1.3, 0.3, 2.0]:
            a_trial = a.copy()
            a_trial[j] *= scale
            a_trial = np.maximum(a_trial, 1e-8)
            ss = a_trial.sum()
            if ss < 1e-10:
                continue
            a_trial /= ss
            J_trial = compute_J(a_trial)
            if J_trial < best_J:
                best_J = J_trial
                best_trial = a_trial
        
        if best_trial is not None:
            a = best_trial
            current_J = best_J
            if step % 200 == 0:
                tracker.update(a)
    
    tracker.update(a)
    return a

def flat_spectrum_init(n, rng):
    """Generate sequence with approximately flat power spectrum."""
    half = n // 2 + 1
    angles = rng.uniform(0, 2 * np.pi, half)
    A = np.zeros(n, dtype=complex)
    A[:half] = np.exp(1j * angles[:half])
    for k in range(1, n - n // 2):
        A[k] = np.conj(A[n - k])
    a = np.fft.ifft(A).real
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-10:
        return np.ones(n) / n
    return a / s

def generate_inits(n, rng, parent=None):

b[k] = sum_{j} a[j] * a[k-j] (where the sum is over valid j, i.e., 0 <= j < n and 0 <= k-j < n)

db[k]/da[j] = a[k-j] (if 0 <= k-j < n)

So the gradient of LSE = sum_k p[k] * b[k] (approximately, with the tau factor) is:
grad[j] = tau * sum_k p[k] * a[k-j] for valid (k,j) pairs

Let m = k-j, so k = m+j:
grad[j] = tau * sum_m p[m+j] * a[m]

This is the cross-correlation of a and p at lag j.

Using FFT: cross-correlation(a, p) = IFFT(conj(FFT(a)) * FFT(p))

import numpy as np
from scipy.optimize import linprog
from scipy.sparse import csr_matrix
import time

_spar_cache = {}

def get_sparsity(n):
    if n not in _spar_cache:
        rows = []
        cols = []
        for k in range(2 * n - 1):
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            rows.extend([k] * (je - js + 1))
            cols.extend(range(js, je + 1))
        _spar_cache[n] = (np.array(rows), np.array(cols), len(rows))
    return _spar_cache[n]

def fast_conv(a):
    n = len(a)
    s = 1
    while s < 2 * n - 1:
        s <<= 1
    fa = np.fft.rfft(a, s)
    return np.fft.irfft(fa * fa, s)[:2 * n - 1]

def compute_J(a):
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))

def lse_and_grad(a, tau):
    """Compute LSE of convolution and its gradient w.r.t. a."""
    n = len(a)
    s_fft = 1
    while s_fft < 2 * n - 1:
        s_fft <<= 1

    fa = np.fft.rfft(a, s_fft)
    b = np.fft.irfft(fa * fa, s_fft)[:2 * n - 1]

    b_max = b.max()
    shifted = tau * (b - b_max)
    max_shifted = shifted.max()
    exp_shifted = np.exp(shifted - max_shifted)
    lse = b_max + (max_shifted + np.log(exp_shifted.sum())) / tau

    p = exp_shifted / exp_shifted.sum()

    # Gradient: grad[j] = tau * sum_m a[m] * p[m+j]
    s_corr = 1
    while s_corr < 3 * n - 2:
        s_corr <<= 1
    fa_full = np.fft.rfft(a, s_corr)
    fp = np.fft.rfft(p, s_corr)
    corr = np.fft.irfft(np.conj(fa_full) * fp, s_corr)
    grad = tau * corr[:n]

    return lse, grad, b

def lp_best_response(a, M, time_limit=5):
    n = len(a)
    rows, cols, nnz = get_sparsity(n)
    vals = np.empty(nnz)
    for i in range(nnz):
        k = rows[i]
        j = cols[i]
        vals[i] = a[k - j]
    A = csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                options={'presolve': True, 'time_limit': time_limit})
    if r.success and r.x is not None:
        g = np.maximum(r.x, 0.0)
        if g.sum() > 1e-10:
            return g / g.sum()
    return None

def spectral_perturb(a, rng, strength=0.3):
    """Perturb the log-magnitude spectrum to escape local minima."""
    n = len(a)
    A = np.fft.rfft(a)
    mag = np.abs(A)
    phase = np.angle(A)

    log_mag = np.log(mag + 1e-12)
    noise = rng.normal(0, strength, log_mag.shape)
    freqs = np.arange(len(log_mag)) / max(1, len(log_mag))
    attenuation = np.exp(-2.0 * freqs)
    log_mag += noise * attenuation

    A_new = np.exp(log_mag) * np.exp(1j * phase)

    A_full = np.zeros(n, dtype=complex)
    half = n // 2 + 1
    A_full[:half] = A_new
    for k in range(1, n - n // 2):
        A_full[k] = np.conj(A_full[n - k])

    a_new = np.fft.ifft(A_full).real
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s < 1e-10:
        return a
    return a_new / s

def adam_lse_optimize(a, deadline, tracker, max_tau=200.0, initial_tau=10.0, perturb_interval=200):
    """Adam optimization with LSE-smoothed objective and spectral perturbations."""
    n = len(a)
    a = a.copy().astype(np.float64)
    a = np.maximum(a, 1e-8)
    a /= a.sum()

    current_J = compute_J(a)
    tracker.update(a)

    m = np.zeros(n)
    v = np.zeros(n)
    t = 0
    lr = 0.005 / np.sqrt(n)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8

    tau = initial_tau
    stuck_count = 0
    best_local_J = current_J
    rng = np.random.default_rng(int(time.time()) % 100000)

    while time.monotonic() < deadline:
        t += 1

        lse_val, grad, b = lse_and_grad(a, tau)

        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * grad ** 2
        m_hat = m / (1 - beta1 ** t)
        v_hat = v / (1 - beta2 ** t)

        a_new = a - lr * m_hat / (np.sqrt(v_hat) + eps)
        a_new = np.maximum(a_new, 1e-8)
        a_new /= a_new.sum()

        J_new = compute_J(a_new)

        if J_new < current_J - 1e-14:
            a = a_new
            current_J = J_new
            stuck_count = 0
            if current_J < best_local_J:
                best_local_J = current_J
                tracker.update(a)
            tau = min(max_tau, tau * 1.003)
            lr = min(0.05 / np.sqrt(n), lr * 1.002)
        else:
            stuck_count += 1
            lr *= 0.97
            tau = max(5.0, tau * 0.995)

            if stuck_count > perturb_interval:
                a = spectral_perturb(a, rng, strength=0.15 + 0.15 * min(1.0, stuck_count / (4.0 * perturb_interval)))
                a = np.maximum(a, 1e-8)
                a /= a.sum()
                current_J = compute_J(a)
                m = np.zeros(n)
                v = np.zeros(n)
                t = 0

                stuck_count = 0
                tau = max(initial_tau, best_local_J * n * 0.01)
                tracker.update(a)

    return a

def exact_subgradient(a, deadline, tracker, max_iters=10000):

def coordinate_descent(a, deadline, tracker, max_steps=10000):

    n = len(a)

def flat_spectrum_init(n, rng):
    """Generate sequence with approximately flat power spectrum."""
    half = n // 2 + 1
    angles = rng.uniform(0, 2 * np.pi, half)
    A = np.zeros(n, dtype=complex)
    A[:half] = np.exp(1j * angles[:half])
    for k in range(1, n - n // 2):
        A[k] = np.conj(A[n - k])
    a = np.fft.ifft(A).real
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-10:
        return np.ones(n) / n
    return a / s

def resample_to_length(src, dest_n):
    """Resample a sequence to a different length."""
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size, endpoint=False)
    t_new = np.linspace(0, 1, dest_n, endpoint=False)
    return np.maximum(np.interp(t_new, t_old, src), 0.001)

def generate_inits(n, rng, parent=None):

    if parent is not None and parent.size > 0:

        inits.append(p_resampled / p_resampled.sum())
        p_rev_resampled = resample_to_length(parent[::-1], n)
        inits.append(p_rev_resampled / p_rev_resampled.sum())

    inits.append(np.ones(n) / n)
    inits.append(rng.uniform(0.1, 1.0, n))
    inits.append(flat_spectrum_init(n, rng))
    inits.append(flat_spectrum_init(n, rng))

                a = exact_subgradient(a, time.monotonic() + sg_budget, tracker)
                print(f"  Subgradient: J={compute_J(a):.8f}", flush=True)

    # === Phase 4: Final polish (remaining time) ===
    print(f"\n{'='*50} Phase 4: Final Polish {'='*50}", flush=True)

    if tracker.best is not None:
        a = tracker.best.copy().astype(np.float64)

        # LP refinement passes
        for i in range(5):
            if time.monotonic() > deadline - 20:
                break
            a_norm = a / a.sum()
            b = fast_conv(a_norm)
            M = b.max()
            tighten = [0.995, 0.997, 0.998, 0.999, 0.9995][i]
            g = lp_best_response(a_norm, M * tighten, time_limit=8)
            if g is not None:
                for t_mix in [0.1, 0.3, 0.5, 0.7]:
                    a_try = (1 - t_mix) * a_norm + t_mix * g
                    a_try = np.maximum(a_try, 1e-8)
                    a_try /= a_try.sum()
                    tracker.update(a_try)
            if time.monotonic() > deadline - 40:
                break

        # Final coordinate descent
        if time.monotonic() < deadline - 30:
            cd_budget = min(60.0, deadline - time.monotonic() - 15)
            if cd_budget > 10:
                a = coordinate_descent(tracker.best, time.monotonic() + cd_budget, tracker)
                print(f"  CoordDescent: J={compute_J(a):.8f}", flush=True)

        # Very final Adam
        if time.monotonic() < deadline - 10:
            a = adam_lse_optimize(tracker.best, deadline - 5, tracker,
                                  max_tau=1000.0, initial_tau=200.0, perturb_interval=50)

    # Fallback
    if tracker.best is None:
        a = np.ones(256) / 256
        tracker.best = a
        tracker.best_score = compute_J(a)
        tracker.update(a)

    print(f"\n{'='*50} FINAL {'='*50}", flush=True)
    print(f"J={tracker.best_score:.8f} n={tracker.best.size} time={time.monotonic()-tracker.t_start:.1f}s", flush=True)
    return [float(x) for x in tracker.best]
