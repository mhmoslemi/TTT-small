def build_lp_matrix(n, a):
    """Build sparse constraint matrix for the LP problem."""
    rows = []
    cols = []
    vals = []
    for k in range(2*n - 1):
        j_min = max(0, k - n + 1)
        j_max = min(n - 1, k)
        for j in range(j_min, j_max + 1):
            rows.append(k)
            cols.append(j)
            vals.append(a[k - j])

"""
Minimizes J(a) = 2*n*max(conv(a,a))/(sum(a))^2 using:
1. LP mirror descent with fast precomputed-convolution line search
2. Coordinate descent polishing (exact 1D optimization per coordinate)
3. Simulated annealing with random coordinate perturbations for global exploration
4. Multi-length exploration and diverse initializations
Key improvements over previous: faster line search via bilinear convolution decomposition,
coordinate descent as complement to LP, and simulated annealing for escaping local optima.

import numpy as np
from scipy.optimize import linprog, minimize_scalar
from scipy.sparse import csr_matrix
import time


def fast_conv(a):
    """Full self-convolution via FFT."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]


def cross_conv(a, b):
    """Full cross-convolution a*b via FFT."""
    n = max(len(a), len(b))
    size = 1
    while size < len(a) + len(b) - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    return np.fft.irfft(fa * fb, size)[:len(a) + len(b) - 1]


def compute_J(a):
    """Compute J(a) = 2*n*max(conv(a,a))/(sum(a))^2."""
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))


def build_lp_matrix(n, a):
    """Build sparse constraint matrix for LP: A[k,j] = a[k-j] if valid."""
    rows = np.empty((2 * n - 1) * n, dtype=np.int32)
    cols = np.empty((2 * n - 1) * n, dtype=np.int32)
    vals = np.empty((2 * n - 1) * n, dtype=np.float64)
    idx = 0
    for k in range(2 * n - 1):
        j_min = max(0, k - n + 1)
        j_max = min(n - 1, k)
        for j in range(j_min, j_max + 1):
            rows[idx] = k
            cols[idx] = j
            vals[idx] = a[k - j]
            idx += 1
    rows = rows[:idx]
    cols = cols[:idx]
    vals = vals[:idx]
    A = csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))
    return A


def lp_best_response(a, M):
    """LP: max sum(g) s.t. (a*g)[k] <= M for all k, g >= 0."""
    n = len(a)
    A = build_lp_matrix(n, a)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    try:
        r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                    options={'time_limit': 120, 'presolve': True})
        if r.success and r.x is not None:
            g = np.maximum(r.x, 0)
            if g.sum() > 1e-15:
                return g
    except Exception:
        pass
    return None


def line_search_bilinear(b_aa, b_ag, b_gg, n):
    """Find t in [0,1] minimizing 2n*max((1-t)^2*b_aa + 2t(1-t)*b_ag + t^2*b_gg)."""
    def obj(t):
        b = (1 - t) ** 2 * b_aa + 2 * t * (1 - t) * b_ag + t * t * b_gg
        return b.max()

    result = minimize_scalar(obj, bounds=(0.0, 1.0), method='bounded',
                             options={'xatol': 1e-8})
    return float(result.x)


def lp_mirror_descent(a, max_iters, deadline, use_cd_every=40):
    """
    LP mirror descent with fast line search and periodic coordinate descent.
    Returns (best_a, best_J).
    """
    a = np.maximum(a.astype(np.float64), 1e-10)
    a = a / a.sum()
    n = len(a)

    best_a = a.copy()
    best_J = compute_J(a)
    no_improve = 0

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        b_aa = fast_conv(a)
        M = float(b_aa.max())

        g = lp_best_response(a, M)
        if g is None:
            break
        g_sum = g.sum()
        if g_sum < 1e-15:
            break
        g = g / g_sum

        # Fast line search using precomputed convolutions
        b_ag = cross_conv(a, g)
        b_gg = fast_conv(g)

        t_opt = line_search_bilinear(b_aa, b_ag, b_gg, n)

        a_new = np.maximum((1 - t_opt) * a + t_opt * g, 0)
        s = a_new.sum()
        if s < 1e-10:
            break
        a_new = a_new / s

        J_new = compute_J(a_new)

        if J_new < best_J - 1e-13:
            best_a = a_new.copy()
            best_J = J_new
            a = a_new
            no_improve = 0
        else:
            no_improve += 1

        # Periodic coordinate descent pass
        if use_cd_every > 0 and it % use_cd_every == use_cd_every - 1:
            a_cd = coordinate_descent_pass(a, deadline, max_passes=2)
            J_cd = compute_J(a_cd)
            if J_cd < best_J - 1e-13:
                best_J = J_cd
                best_a = a_cd.copy()
            a = a_cd
            no_improve = 0

        if no_improve > 60:
            # Perturb and restart
            a = a + np.random.normal(0, 0.02, n)
            a = np.maximum(a, 0.001)
            a = a / a.sum()
            no_improve = 0

    return best_a, best_J


def coordinate_descent_pass(a, deadline, max_passes=5):
    """
    Coordinate descent: for each coordinate, try optimal value.
    Uses the formula: b_new[k] = b[k] + 2*delta*a[k-i] (k in range, k!=2i)

    """
    a = np.maximum(a.astype(np.float64), 0)
    s = a.sum()
    if s < 0.01:
        return a

    n = len(a)

    best_M = fast_conv(a).max()

    for p in range(max_passes):
        if time.monotonic() > deadline:
            break

        b = fast_conv(a)
        S = a.sum()
        improved = False

        for i in order:
            if time.monotonic() > deadline:
                break

            ai = a[i]
            if ai < 1e-12:
                candidates = [0.01, 0.05, 0.1, 0.2]
            else:
                candidates = [0.0, ai * 0.3, ai * 0.5, ai * 0.75, ai * 1.3, ai * 1.6, ai * 2.0]

            best_new = ai

                if abs(delta) < 1e-14:
                    continue
                S_new = S + delta
                if S_new < 0.01:
                    continue

                                      b[k_hi + 1:].max() if k_hi < 2 * n - 2 else 0)
                elif k_lo == 0:
                    outside_max = b[k_hi + 1:].max() if k_hi < 2 * n - 2 else 0

                    outside_max = b[:k_lo].max()

                affected = []

                    else:
                        new_val = b[k] + 2 * delta * a[j]
                    affected.append(new_val)

                inside_max = max(affected) if affected else 0
                new_max = max(outside_max, inside_max)
                val = 2 * n * new_max / (S_new * S_new)

                if val < best_val - 1e-12:
                    best_val = val
                    best_new = cand

            if best_new != ai:
                a[i] = best_new
                improved = True
                # Update b and S incrementally
                delta = best_new - ai
                S = S + delta
                k_lo = max(0, i)
                k_hi = min(i + n - 1, 2 * n - 2)
                for k in range(k_lo, k_hi + 1):
                    j = k - i
                    if j == i:
                        b[k] += 2 * delta * a[i] + delta * delta
                    else:
                        b[k] += 2 * delta * a[j]
                # Renormalize to keep sum=1
                if S > 0.01:
                    a = a / S
                    b = b / (S * S) * S  # Actually b scales as (a/S)^2 * S^2...
                    # Wait, if I normalize a -> a/S, then b -> b/S^2
                    # But I want to track b = conv(a_normalized, a_normalized)
                    # Let me just recompute b periodically
                    pass

        # Recompute b to avoid drift
        b = fast_conv(a)
        s = a.sum()
        if s > 0.01:
            a = a / s
            b = fast_conv(a)
            M = b.max()
            if M < best_M:
                best_M = M
                best_a = a.copy()

        if not improved:
            break

    return best_a


def coordinate_descent_fast(a, deadline, max_passes=20):
    """
    Faster coordinate descent: recompute convolution each pass,
    for each coordinate try a few candidate values.
    """
    a = np.maximum(a.astype(np.float64), 0)
    s = a.sum()
    if s < 0.01:
        return a
    a = a / s
    n = len(a)

    best_a = a.copy()
    best_J = compute_J(a)

    for p in range(max_passes):
        if time.monotonic() > deadline:
            break

        order = np.random.permutation(n)
        improved = False

        for i in order:
            if time.monotonic() > deadline:
                break

            ai = a[i]
            if ai < 1e-12:
                candidates = [0.02, 0.08, 0.15]
            else:
                candidates = [0.0, ai * 0.5, ai * 0.75, ai * 1.25, ai * 1.5, ai * 2.0]

            best_i = ai
            best_score = compute_J(a)

            for cand in candidates:
                if abs(cand - ai) < 1e-14:
                    continue
                a[i] = cand
                a_clip = np.maximum(a, 0)
                s_check = a_clip.sum()
                if s_check < 0.01:
                    a[i] = ai
                    continue
                score = compute_J(a_clip / s_check)
                if score < best_score:
                    best_score = score
                    best_i = cand

            if best_i != ai:
                a[i] = best_i
                improved = True

        a = np.maximum(a, 0)
        s = a.sum()
        if s > 0.01:
            a = a / s
            J = compute_J(a)
            if J < best_J:
                best_J = J
                best_a = a.copy()

        if not improved:
            break

    return best_a


def simulated_annealing(a, deadline, max_iters=5000, T_start=0.05, T_end=0.001):
    """
    Simulated annealing: random coordinate perturbations with acceptance criterion.
    """
    a = np.maximum(a.astype(np.float64), 0)
    a = a / a.sum()
    n = len(a)

    best_a = a.copy()
    best_J = compute_J(a)
    current_J = best_J

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        frac = it / max(1, max_iters)
        T = T_start * (T_end / T_start) ** frac

        # Pick random coordinate
        i = np.random.randint(n)
        old_val = a[i]

        # Generate new value
        if old_val < 0.05:
            new_val = np.random.exponential(0.1)
        else:
            new_val = np.random.normal(old_val, 0.15 * old_val + 0.02)
        new_val = max(0.0, new_val)

        a[i] = new_val
        a_new = np.maximum(a, 0)
        s = a_new.sum()
        if s < 0.01:
            a[i] = old_val
            continue
        a_new = a_new / s
        new_J = compute_J(a_new)

        # Acceptance criterion
        delta = new_J - current_J
        if delta < 0 or np.random.random() < np.exp(-delta / max(T, 1e-10)):
            a = a_new
            current_J = new_J
            if new_J < best_J - 1e-12:
                best_J = new_J
                best_a = a_new.copy()
        else:
            a[i] = old_val

    return best_a


def generate_spectral_init(n, rng):
    """Generate initialization based on eigenstructure."""
    # Create a random symmetric PSD matrix and take eigenvector
    size = min(n, 200)
    A = rng.normal(0, 1, (size, size))
    A = A @ A.T / size
    eigenvalues, eigenvectors = np.linalg.eigh(A)
    # Take the largest eigenvector
    idx = np.argmax(eigenvalues)
    v = np.abs(eigenvectors[:, idx])
    # Resample to length n
    t_old = np.linspace(0, 1, size)
    t_new = np.linspace(0, 1, n)
    result = np.interp(t_new, t_old, v)
    result = np.maximum(result, 0.01)
    return result / result.sum()


def generate_diverse_inits(n, rng):
    """Generate diverse initializations."""
    inits = []
    t = np.linspace(0, 1, n, endpoint=False)

    # Uniform

    sm = np.convolve(rng.uniform(0, 1, n), kernel, mode='same')
    sm = np.maximum(sm + 0.1, 0.01)
    inits.append(sm / sm.sum())

    # Cosine window
    cw = 0.5 + 0.5 * np.cos(2 * np.pi * t)
    inits.append(cw / cw.sum())

    # Bell
    bell = np.exp(-4 * (t - 0.5) ** 2) + 0.05
    inits.append(bell / bell.sum())

    # Double bump
    dbl = np.exp(-8 * (t - 0.3) ** 2) + np.exp(-8 * (t - 0.7) ** 2) + 0.03
    inits.append(dbl / dbl.sum())

    # Frequency modulated
    for freq in [1, 2, 3, 5]:
        fm = 1.0 + 0.3 * np.cos(2 * np.pi * freq * t)
        fm = np.maximum(fm, 0.01)
        inits.append(fm / fm.sum())

    # Spectral
    inits.append(generate_spectral_init(n, rng))

    # Power law
    for alpha in [0.5, 1.0, 2.0]:
        pl = (t + 0.01) ** alpha
        inits.append(pl / pl.sum())

    # Zigzag
    zz = 0.5 + 0.3 * np.abs(2 * (t * 4 % 1) - 1)
    inits.append(zz / zz.sum())

    return inits


def resample_seq(src, dest_n):
    """Resample 1D sequence to new length via linear interpolation."""
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)


def propose_candidate(seed=42, budget_s=1500, **kwargs):
    """
    Multi-phase search for minimizing J(a) = 2n*max(conv(a,a))/(sum(a))^2.
    Phase 1: LP mirror descent from parent (main workhorse)
    Phase 2: Coordinate descent polishing
    Phase 3: LP mirror descent from diverse initializations
    Phase 4: Simulated annealing
    Phase 5: Multi-length exploration
    Phase 6: Final polishing
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + float(budget_s) - 20.0
    t_start = time.monotonic()

    best_seq = None
    best_score = float('inf')

    def update_best(a):
        nonlocal best_seq, best_score
        a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
        if a.size == 0 or a.sum() < 0.01:
            return
        score = float(evaluate_sequence(a.tolist()))
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_seq = a.copy()
            elapsed = time.monotonic() - t_start
            print(f"  ** J={score:.8f} n={a.size} t={elapsed:.0f}s", flush=True)

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        if p.ndim == 1 and p.size > 0 and np.all(np.isfinite(p)):
            parent = np.maximum(p, 0.0)
            update_best(parent)
            print(f"Parent loaded: n={parent.size}, J={best_score:.6f}", flush=True)
    except (NameError, AttributeError):
        pass

    if parent is None:
        parent = np.ones(256) / 256
        update_best(parent)

    # === PHASE 1: LP mirror descent from parent ===
    n_parent = parent.size
    p1_deadline = t_start + 450
    print(f"\n{'='*60}", flush=True)
    print(f"PHASE 1: LP Mirror Descent (n={n_parent}, budget=450s)", flush=True)
    print(f"{'='*60}", flush=True)

    a = parent.copy() / parent.sum()
    a_result, J_result = lp_mirror_descent(a, 5000, min(p1_deadline, deadline - 200),
                                            use_cd_every=50)
    update_best(a_result)
    print(f"  Phase 1 done: J={J_result:.8f}", flush=True)

    # === PHASE 2: Coordinate descent polishing ===
    p2_deadline = min(t_start + 650, deadline - 300)
    print(f"\n{'='*60}", flush=True)
    print(f"PHASE 2: Coordinate Descent Polishing (budget=200s)", flush=True)
    print(f"{'='*60}", flush=True)

    if best_seq is not None:
        a_cd = coordinate_descent_fast(best_seq, p2_deadline, max_passes=30)
        update_best(a_cd)
        # Also try on the LP result
        a_cd2 = coordinate_descent_fast(a_result, min(p2_deadline, deadline - 300), max_passes=20)
        update_best(a_cd2)
        print(f"  Phase 2 done: best J={best_score:.8f}", flush=True)

    # === PHASE 3: LP mirror descent from diverse initializations ===
    p3_deadline = min(t_start + 900, deadline - 300)
    print(f"\n{'='*60}", flush=True)
    print(f"PHASE 3: Diverse LP Mirror Descent (budget=250s)", flush=True)
    print(f"{'='*60}", flush=True)

    if best_seq is not None:
        n_best = best_seq.size
        inits = generate_diverse_inits(n_best, rng)
        # Also add perturbed best
        inits.append(np.maximum(best_seq + rng.normal(0, 0.05, n_best), 0.01))
        inits[-1] = inits[-1] / inits[-1].sum()
        inits.append(np.maximum(best_seq + rng.normal(0, 0.1, n_best), 0.01))
        inits[-1] = inits[-1] / inits[-1].sum()

        # Sort by initial J, try best ones first
        init_scores = [(compute_J(init), init) for init in inits]
        init_scores.sort(key=lambda x: x[0])

        for score_init, init in init_scores[:8]:
            if time.monotonic() > p3_deadline:
                break
            phase_dl = min(time.monotonic() + 30, p3_deadline)
            a_r, J_r = lp_mirror_descent(init, 500, phase_dl, use_cd_every=30)
            update_best(a_r)
            # Quick CD on result
            a_cd = coordinate_descent_fast(a_r, min(phase_dl + 10, p3_deadline), max_passes=5)
            update_best(a_cd)

        print(f"  Phase 3 done: best J={best_score:.8f}", flush=True)

    # === PHASE 4: Simulated annealing ===
    p4_deadline = min(t_start + 1150, deadline - 250)
    print(f"\n{'='*60}", flush=True)
    print(f"PHASE 4: Simulated Annealing (budget=250s)", flush=True)
    print(f"{'='*60}", flush=True)

    if best_seq is not None:
        n_s = best_seq.size
        # Run SA from best
        a_sa = simulated_annealing(best_seq, p4_deadline, max_iters=10000,
                                    T_start=0.03, T_end=0.002)
        update_best(a_sa)
        # Polish SA result
        a_cd = coordinate_descent_fast(a_sa, min(p4_deadline + 30, deadline - 200), max_passes=10)
        update_best(a_cd)
        print(f"  Phase 4 done: best J={best_score:.8f}", flush=True)

    # === PHASE 5: Multi-length exploration ===
    p5_deadline = min(t_start + 1350, deadline - 150)
    print(f"\n{'='*60}", flush=True)
    print(f"PHASE 5: Multi-length Exploration (budget=200s)", flush=True)
    print(f"{'='*60}", flush=True)

    if best_seq is not None:
        n_base = best_seq.size
        target_lengths = set()
        for mult in [0.75, 1.25, 1.5, 2.0]:
            target_lengths.add(int(n_base * mult))
        # Also try some fixed lengths
        for L in [384, 448, 512, 640, 768, 1024]:
            target_lengths.add(L)

        target_lengths = sorted([L for L in target_lengths if 128 <= L <= 2048])

        for tgt_n in target_lengths:
            if time.monotonic() > p5_deadline:
                break
            if tgt_n == n_base:
                continue

            a_ext = resample_seq(best_seq, tgt_n)
            a_ext = np.maximum(a_ext, 1e-6)
            a_ext = a_ext / a_ext.sum()

            # Quick LP + CD
            phase_dl = min(time.monotonic() + 25, p5_deadline)
            a_r, J_r = lp_mirror_descent(a_ext, 200, phase_dl, use_cd_every=30)
            update_best(a_r)
            a_cd = coordinate_descent_fast(a_r, min(phase_dl + 10, p5_deadline), max_passes=5)
            update_best(a_cd)

        print(f"  Phase 5 done: best J={best_score:.8f}", flush=True)

    # === PHASE 6: Final polishing ===
    p6_deadline = deadline
    print(f"\n{'='*60}", flush=True)
    print(f"PHASE 6: Final Polishing", flush=True)
    print(f"{'='*60}", flush=True)

    if best_seq is not None:
        # One more LP pass
        n_f = best_seq.size
        a_f, J_f = lp_mirror_descent(best_seq, 200, min(t_start + 1500 - 30, deadline),
                                      use_cd_every=20)
        update_best(a_f)
        # Final CD
        a_cd = coordinate_descent_fast(a_f, deadline, max_passes=10)
        update_best(a_cd)

    # Fallback
    if best_seq is None:
        best_seq = np.ones(128) / 128
        best_score = compute_J(best_seq)
        update_best(best_seq)

    print(f"\n{'='*60}", flush=True)
    print(f"FINAL: J={best_score:.8f} n={best_seq.size} time={time.monotonic()-t_start:.0f}s", flush=True)
    print(f"{'='*60}", flush=True)

    return [float(x) for x in best_seq]
