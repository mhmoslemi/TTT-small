# Precomputed sparsity cache
_spar_cache = {}

def get_sparsity(n):
    if n not in _spar_cache:
        rows = []
        cols = []
        for k in range(2 * n - 1):
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            rows.extend([k] * (je - js + 1))
            cols.extend(range(js, je + 1))
        _spar_cache[n] = (np.array(rows, dtype=np.int32), np.array(cols, dtype=np.int32))

def fast_conv(a):
    n = len(a)
    s = 1
    while s < 2 * n - 1:
        s = s << 1
    fa = np.fft.rfft(a, s)
    return np.fft.irfft(fa * fa, s)[:2 * n - 1]

def compute_J(a):
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))

def lp_best_response(a, M, n):

    nnz = len(rows)

        j = cols[i]
        vals[i] = a[k - j]
    A = csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                options={'presolve': True, 'time_limit': 5})
    if r.success and r.x is not None:

        if g.sum() > 1e-10:
            return g
    return None

def mirror_descent(a, max_iters, deadline, tracker):
    """Mirror descent with multiplicative weights update on the simplex."""
    a = np.maximum(a.astype(np.float64), 1e-10)
    a = a / a.sum()

    current_J = compute_J(a)

    for it in range(max_iters):

            break

        M = float(b.max())
        active = np.where(b >= M - 1e-12)[0]
        
        # Compute subgradient of max convolution
        grad = np.zeros(n)
        for k in active:
            js = max(0, k - n + 1)
            je = min(n - 1, k)

                grad[j] += a[k - j]
        grad = grad / len(active)
        
        # Full gradient of J (with sum=1 normalization)
        # J = 2n*M (since sum=1)

        # But we need to account for the constraint sum(a)=1

        # For mirror descent on simplex: a_j *= exp(-eta * (grad_j - mean(grad)))

        effective_grad = grad - np.dot(grad, a)

        a_new = a * np.exp(log_update)

def lse_gradient_descent(a, max_iters, deadline, tracker):

    n = len(a)
    current_J = compute_J(a)

    lr = 1.0 / (n * tau)
    momentum = 0.9
    v = np.zeros(n)

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        b = fast_conv(a)
        b_max = b.max()
        
        # Use shifted exponents for numerical stability
        exp_b = np.exp(tau * (b - b_max))
        p = exp_b / exp_b.sum()

        # Gradient of LSE smooth approximation
        # d(sum_k exp(tau*b_k))/da_j = tau * sum_k p_k * db_k/da_j
        # = tau * sum_k p_k * a[k-j] (for valid k-j)

        fp = np.fft.rfft(p, s)

        # cross-correlation: sum_k p[k] * a[k-j] = ifft(fft(p) * conj(fft(a)))[:n]
        corr = np.fft.irfft(fp * np.conj(fa), s)

        a_new = a - lr * v

            v *= 0.5

                break

def flat_spectrum_init(n, rng):
    half = n // 2 + 1
    angles = rng.uniform(0, 2 * np.pi, half)
    A = np.zeros(n, dtype=complex)
    A[:half] = np.exp(1j * angles[:half])
    for k in range(1, n - n // 2):
        A[k] = np.conj(A[n - k])
    a = np.fft.ifft(A).real
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-10:
        return np.ones(n) / n
    return a / s

def resample_seq(src, dest_n):
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)

def simulated_annealing(a, deadline, tracker, n_restarts=30):

    tracker.update(a)
    temp = 0.1

    for r in range(n_restarts):
        if time.monotonic() > deadline:
            break
        a_pert = a + rng.normal(0, temp, n)
        a_pert = np.maximum(a_pert, 1e-6)
        a_pert = a_pert / a_pert.sum()

        # Short LP refinement
        a_refined = run_lp_opt(a_pert, max(3, 15 - r // 4), deadline, tracker, tighten=0.01)

        J = compute_J(a_refined)
        delta = J - current_J
        if delta < 0 or rng.random() < np.exp(-delta / max(temp, 1e-10)):
            a = a_refined
            current_J = J
        temp *= 0.93
    tracker.update(a)
    return a

import numpy as np

_spar_cache = {}

def get_sparsity(n):
    if n not in _spar_cache:
        rows = []
        cols = []
        for k in range(2 * n - 1):
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            rows.extend([k] * (je - js + 1))
            cols.extend(range(js, je + 1))

    while s < 2 * n - 1:

    fa = np.fft.rfft(a, s)
    return np.fft.irfft(fa * fa, s)[:2 * n - 1]

                continue
            a_trial = a_trial / ss
            J_trial = compute_J(a_trial)
            if J_trial < best_J - 1e-15:
                best_J = J_trial
                best_a = a_trial

        if best_a is None:
            for t in [0.05, 0.02, 0.01]:
                a_trial = (1.0 - t) * a + t * g
                a_trial = np.maximum(a_trial, 0.0)
                a_trial = a_trial / a_trial.sum()
                J_trial = compute_J(a_trial)
                if J_trial < best_J - 1e-15:
                    best_J = J_trial
                    best_a = a_trial
                    break

        if best_a is None:
            break

        a = best_a
        current_J = best_J
        if it % 5 == 0:
            tracker.update(a)

    tracker.update(a)
    return a

def mirror_descent(a, max_iters, deadline, tracker):
    """Mirror descent with multiplicative weights on the simplex."""
    a = np.maximum(a.astype(np.float64), 1e-10)
    a = a / a.sum()
    n = len(a)
    current_J = compute_J(a)
    tracker.update(a)

    eta = 2.0 / np.sqrt(n)
    stall = 0

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        b = fast_conv(a)
        M = float(b.max())
        tol = max(1e-12, 1e-10 * M)
        active = np.where(b >= M - tol)[0]

        grad = np.zeros(n)
        for k in active:
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            for j in range(js, je + 1):
                grad[j] += a[k - j]
        grad = grad / len(active)

        # Project gradient onto tangent space of simplex
        effective_grad = grad - np.dot(grad, a)

        # Multiplicative weights update
        a_new = a * np.exp(-eta * effective_grad)
        a_new = np.maximum(a_new, 1e-12)
        a_new = a_new / a_new.sum()

        J_new = compute_J(a_new)
        if J_new < current_J - 1e-15:
            a = a_new
            current_J = J_new
            stall = 0
            if it % 30 == 0:
                eta = min(eta * 1.01, 20.0 / np.sqrt(n))
        else:
            eta *= 0.85
            stall += 1
            if stall > 40:
                break

        if it % 20 == 0:
            tracker.update(a)

    tracker.update(a)
    return a

def lse_gradient_descent(a, max_iters, deadline, tracker):
    """LSE smooth approximation with gradient descent."""
    a = np.maximum(a.astype(np.float64), 1e-6)
    a = a / a.sum()
    n = len(a)
    current_J = compute_J(a)
    tracker.update(a)

    tau = 30.0
    lr = 1.0 / (n * tau)
    momentum = 0.9
    v = np.zeros(n)
    stall = 0

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        b = fast_conv(a)
        b_max = b.max()
        exp_b = np.exp(tau * (b - b_max))
        exp_sum = exp_b.sum()
        if exp_sum < 1e-300:
            break
        p = exp_b / exp_sum

        s = 1
        while s < 2 * n:
            s = s << 1
        fp = np.fft.rfft(p, s)
        fa = np.fft.rfft(a, s)
        corr = np.fft.irfft(fp * np.conj(fa), s)
        grad = tau * corr[:n]

        v = momentum * v + (1.0 - momentum) * grad
        a_new = a - lr * v
        a_new = np.maximum(a_new, 1e-6)
        a_new = a_new / a_new.sum()
        J_new = compute_J(a_new)

        if J_new < current_J - 1e-15:
            a = a_new
            current_J = J_new
            stall = 0
            if it % 50 == 0:
                tau = min(200.0, tau * 1.08)
            lr = 1.0 / (n * tau)
        else:
            lr *= 0.5
            v *= 0.3
            stall += 1
            if stall > 60 or lr < 1e-18:
                break

        if it % 50 == 0:
            tracker.update(a)

    tracker.update(a)
    return a

def subgradient_descent(a, max_iters, deadline, tracker):
    """Subgradient descent with adaptive step size."""
    a = np.maximum(a.astype(np.float64), 1e-8)
    a = a / a.sum()
    n = len(a)
    current_J = compute_J(a)
    tracker.update(a)

    lr = 1.0 / n
    stall = 0

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        b = fast_conv(a)
        M = float(b.max())
        tol = max(1e-12, 1e-10 * M)
        active = np.where(b >= M - tol)[0]

        grad = np.zeros(n)
        for k in active:
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            for j in range(js, je + 1):
                grad[j] += a[k - j]
        grad = grad / len(active)

        direction = grad - 2.0 * M * a
        a_new = a - lr * direction
        a_new = np.maximum(a_new, 1e-8)
        a_new = a_new / a_new.sum()
        J_new = compute_J(a_new)

        if J_new < current_J - 1e-15:
            a = a_new
            current_J = J_new
            lr = min(lr * 1.03, 5.0 / n)
            stall = 0
        else:
            lr *= 0.7
            stall += 1
            if stall > 60:
                break

        if it % 30 == 0:
            tracker.update(a)

    tracker.update(a)
    return a

def flat_spectrum_init(n, rng):
    half = n // 2 + 1
    angles = rng.uniform(0, 2 * np.pi, half)
    A = np.zeros(n, dtype=complex)
    A[:half] = np.exp(1j * angles[:half])
    for k in range(1, n - n // 2):
        A[k] = np.conj(A[n - k])
    a = np.fft.ifft(A).real
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-10:
        return np.ones(n) / n
    return a / s

def resample_seq(src, dest_n):
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)

def generate_inits(n, rng, parent):
    inits = []
    if parent is not None and parent.size > 0:
        inits.append(resample_seq(parent, n))
        inits.append(resample_seq(parent[::-1], n))

    inits.append(np.ones(n) / n)
    inits.append(rng.uniform(0.1, 1.0, n))
    inits.append(flat_spectrum_init(n, rng))
    inits.append(flat_spectrum_init(n, rng))

    t = np.linspace(0, 1, n, endpoint=False)

    for freq in [2, 3, 5, 7, 11, 13]:
        inits.append(1.0 + 0.3 * np.cos(2 * np.pi * freq * t + rng.uniform(0, 6.28)))

    inits.append(1.0 + 0.3 * np.cos(2 * np.pi * 3.7 * t) + 0.2 * np.cos(2 * np.pi * 7.1 * t))
    inits.append(0.5 + 0.5 * np.cos(np.pi * t) ** 2)

    for cx in [0.25, 0.5, 0.75]:
        inits.append(0.5 * np.exp(-8 * (t - cx) ** 2) + 0.5 * np.exp(-8 * (t - (1 - cx)) ** 2) + 0.05)

    inits.append(1.0 + 0.2 * np.sin(2 * np.pi * t) + 0.15 * np.sin(4 * np.pi * t)
                + 0.1 * np.sin(7 * np.pi * t) + 0.08 * np.sin(11 * np.pi * t))

    inits.append((t + 0.1) ** 0.5)
    inits.append((1.0 - t + 0.1) ** 0.5)

    sparse = np.zeros(n)
    n_spikes = max(5, n // 12)
    positions = rng.choice(n, size=n_spikes, replace=False)
    sparse[positions] = 1.0
    kernel_size = max(3, n // 25)
    kernel = np.ones(kernel_size) / kernel_size
    inits.append(np.convolve(sparse, kernel, mode='same') + 0.05)

    n_coeff = min(n // 2 + 1, 15)
    coeffs = rng.normal(0, 1, n_coeff)
    sig = np.zeros(n)
    for i in range(n_coeff):
        sig += coeffs[i] * np.cos(2 * np.pi * i * t / n + rng.uniform(0, 6.28))
    inits.append(np.abs(sig) + 0.5)

    for x in inits:
        x = np.maximum(x, 0.001).astype(np.float64)
        x = x / x.sum()
    return inits

class ScoreTracker:
    def __init__(self):
        self.best = None
        self.best_score = float('inf')
        self.t_start = time.monotonic()

    def update(self, a):
        a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
        if a.size == 0 or a.sum() < 0.01:
            return
        score = float(evaluate_sequence(a.tolist()))
        if np.isfinite(score) and score < self.best_score:
            self.best = a.copy()
            self.best_score = score
            elapsed = time.monotonic() - self.t_start
            print(f"  * J={score:.8f} n={a.size} t={elapsed:.1f}s", flush=True)

    def report(self, label=""):
        if self.best is not None:
            print(f"  {label} best: J={self.best_score:.8f} n={self.best.size}", flush=True)

def simulated_annealing(a, deadline, tracker, n_restarts=25):
    rng = np.random.default_rng(777)
    n = a.size
    a = a.copy().astype(np.float64)
    current_J = compute_J(a)
    tracker.update(a)
    temp = 0.08

    for r in range(n_restarts):
        if time.monotonic() > deadline:
            break
        a_pert = a + rng.normal(0, temp, n)
        a_pert = np.maximum(a_pert, 1e-6)
        a_pert = a_pert / a_pert.sum()

        # Short min-peak LP refinement
        a_refined = run_lp_minpeak_opt(a_pert, max(3, 12 - r // 3), deadline, tracker, time_limit=3)

        J = compute_J(a_refined)
        delta = J - current_J
        if delta < 0 or rng.random() < np.exp(-delta / max(temp, 1e-10)):
            a = a_refined
            current_J = J
        temp *= 0.93

    tracker.update(a)
    return a

def coordinate_descent(a, max_steps, deadline, tracker):
    a = a.copy().astype(np.float64)
    a = a / a.sum()
    n = len(a)
    current_J = compute_J(a)
    tracker.update(a)
    rng = np.random.default_rng(42)

    for step in range(max_steps):
        if time.monotonic() > deadline:
            break
        j = rng.integers(0, n)
        improved = False
        for scale in [0.7, 1.3, 0.5, 1.5, 0.85, 1.15]:
            a_trial = a.copy()
            a_trial[j] *= scale
            a_trial = np.maximum(a_trial, 1e-8)
            ss = a_trial.sum()
            if ss < 1e-10:
                continue
            a_trial = a_trial / ss
            J_trial = compute_J(a_trial)
            if J_trial < current_J - 1e-14:
                a = a_trial
                current_J = J_trial
                improved = True
                break

        if not improved and step % 100 == 0 and step > 0:
            j2 = rng.integers(0, n)
            for s1, s2 in [(0.8, 1.2), (1.2, 0.8), (0.7, 1.3), (1.3, 0.7)]:
                a_trial = a.copy()
                a_trial[j] *= s1
                a_trial[j2] *= s2
                a_trial = np.maximum(a_trial, 1e-8)
                a_trial = a_trial / a_trial.sum()
                J_trial = compute_J(a_trial)
                if J_trial < current_J - 1e-14:
                    a = a_trial
                    current_J = J_trial
                    improved = True
                    break

        if step % 100 == 0:
            tracker.update(a)

    tracker.update(a)
    return a

def propose_candidate(seed=42, budget_s=1500, **kwargs):
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + float(budget_s) - 30.0
    tracker = ScoreTracker()

    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        if p.ndim == 1 and p.size > 0 and np.all(np.isfinite(p)):
            parent = np.maximum(p, 0.0)
            tracker.update(parent)
            print(f"Parent: n={parent.size}, J={tracker.best_score:.8f}", flush=True)
    except (NameError, AttributeError):
        pass

    # Phase 1: Multi-length search with min-peak LP
    lengths = [512, 384, 256]
    remaining = deadline - time.monotonic() - 250
    time_per_length = remaining / len(lengths)

    for n in lengths:
        t_phase_start = time.monotonic()
        t_phase_end = min(deadline - 250, t_phase_start + time_per_length)
        if time.monotonic() >= t_phase_end - 30:
            continue
        print(f"\n{'='*50} Phase1 n={n} budget={time_per_length:.0f}s {'='*50}", flush=True)

        inits = generate_inits(n, rng, parent)
        best_for_n = None
        best_J_for_n = float('inf')
        n_inits = len(inits)
        init_budget = max(5.0, (t_phase_end - time.monotonic() - 20) / n_inits)

        for idx, a0 in enumerate(inits):
            if time.monotonic() > t_phase_end - 15:
                break

            # Min-peak LP (primary method)
            n_iters = max(5, int(30.0 / max(1, n / 256)))
            tl = max(2, int(10.0 / (n / 256)))
            a = run_lp_minpeak_opt(a0, n_iters, time.monotonic() + init_budget * 0.5, tracker, time_limit=tl)

            # Mirror descent
            if time.monotonic() < t_phase_end - 15:
                a = mirror_descent(a, max(20, int(15000 / n)), t_phase_end - 15, tracker)

            # LSE polish
            if time.monotonic() < t_phase_end - 8:
                a = lse_gradient_descent(a, max(50, int(15000 / n)), t_phase_end - 8, tracker)

            J = compute_J(a)
            if J < best_J_for_n:
                best_J_for_n = J
                best_for_n = a.copy()

        if best_for_n is not None:
            tracker.update(best_for_n)
            if parent is None or best_J_for_n < compute_J(resample_seq(parent, best_for_n.size)):
                parent = best_for_n
            tracker.report(f"Phase1 n={n} ({time.monotonic()-t_phase_start:.1f}s): J={best_J_for_n:.8f}")

    # Phase 2: Deep optimization on best
    if tracker.best is not None and time.monotonic() < deadline - 200:
        t0 = time.monotonic()
        n = tracker.best.size
        print(f"\n{'='*50} Phase2 DeepOpt n={n} {'='*50}", flush=True)

        a = tracker.best.copy().astype(np.float64)

        # Extended min-peak LP
        a = run_lp_minpeak_opt(a, 60, deadline - 200, tracker, time_limit=5)

        # Mirror descent
        if time.monotonic() < deadline - 150:
            a = mirror_descent(a, max(200, int(40000 / n)), deadline - 150, tracker)

        # More min-peak LP
        if time.monotonic() < deadline - 100:
            a = run_lp_minpeak_opt(a, 40, deadline - 100, tracker, time_limit=5)

        # LSE
        if time.monotonic() < deadline - 60:
            a = lse_gradient_descent(a, max(300, int(40000 / n)), deadline - 60, tracker)

        # Subgradient
        if time.monotonic() < deadline - 40:
            a = subgradient_descent(a, max(200, int(20000 / n)), deadline - 40, tracker)

        # Final LP pass
        if time.monotonic() < deadline - 20:
            a = run_lp_minpeak_opt(a, 30, deadline - 20, tracker, time_limit=5)

        tracker.report(f"Phase2 ({time.monotonic()-t0:.1f}s)")

    # Phase 3: Also try max-sum LP on the best (alternative formulation)
    if tracker.best is not None and time.monotonic() < deadline - 80:
        t0 = time.monotonic()
        print(f"\n{'='*50} Phase3 MaxSumLP {'='*50}", flush=True)
        a = tracker.best.copy().astype(np.float64)
        for tight in [0.005, 0.01, 0.015, 0.02]:
            if time.monotonic() > deadline - 80:
                break
            a = run_lp_maxsum_opt(a, 15, deadline - 80, tracker, tighten=tight, time_limit=4)
        if time.monotonic() < deadline - 40:
            a = mirror_descent(a, 100, deadline - 40, tracker)
        tracker.report(f"Phase3 ({time.monotonic()-t0:.1f}s)")

    # Phase 4: Simulated annealing
    if tracker.best is not None and time.monotonic() < deadline - 60:
        t0 = time.monotonic()
        print(f"\n{'='*50} Phase4 SA {'='*50}", flush=True)
        a = simulated_annealing(tracker.best, deadline - 60, tracker, n_restarts=20)
        if time.monotonic() < deadline - 20:
            a = run_lp_minpeak_opt(a, 15, deadline - 20, tracker, time_limit=4)
        tracker.report(f"Phase4 ({time.monotonic()-t0:.1f}s)")

    # Phase 5: Final polish
    if tracker.best is not None and time.monotonic() < deadline - 10:
        t0 = time.monotonic()
        print(f"\n{'='*50} Phase5 Polish {'='*50}", flush=True)
        a = coordinate_descent(tracker.best, 1500, deadline - 8, tracker)
        if time.monotonic() < deadline - 3:
            a = run_lp_minpeak_opt(a, 8, deadline - 1, tracker, time_limit=3)
        if time.monotonic() < deadline - 0.5:
            a = mirror_descent(a, 100, deadline - 0.3, tracker)
        tracker.report(f"Phase5 ({time.monotonic()-t0:.1f}s)")

    # Fallback
    if tracker.best is None:
        tracker.best = np.ones(256)
        tracker.best_score = compute_J(tracker.best)
        tracker.update(tracker.best)

    print(f"\n{'='*50} FINAL {'='*50}", flush=True)
    print(f"J={tracker.best_score:.8f} n={tracker.best.size} time={time.monotonic()-tracker.t_start:.1f}s", flush=True)
    return [float(x) for x in tracker.best]
