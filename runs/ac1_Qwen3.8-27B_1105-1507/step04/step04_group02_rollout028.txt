    n = len(a)
    size = 1
    while size < 2*n - 1:

"""
Optimized step-function coefficient search minimizing 2*n*max(conv(a,a))/(sum(a))^2.
Uses FFT-based subgradient with Nesterov momentum, adaptive active-set tracking,
alternating LP refinement, multi-scale exploration, and stochastic perturbation escapes.
Key improvements: O(n log n) gradient computation via FFT convolution, tighter active-set
selection with adaptive epsilon, and more aggressive multi-start with diverse initializations.
"""

import numpy as np
import time
from scipy.optimize import linprog
from scipy.sparse import coo_matrix, csr_matrix
from scipy.interpolate import interp1d


def compute_conv(a):
    """Compute linear convolution a*a via FFT."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    conv = np.fft.irfft(fa * fa, size)
    return conv[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    S = float(np.sum(a))
    if S < 1e-10:
        return np.inf
    conv = compute_conv(a)
    M = float(np.max(conv))
    return 2.0 * n * M / (S * S)


def compute_grad_fft(a, active_mask):
    """Compute subgradient of max(conv) using FFT. active_mask is boolean array of length 2n-1."""
    n = len(a)
    c_flipped = active_mask[::-1].astype(np.float64)
    size = 1
    while size < 3 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    fc = np.fft.rfft(c_flipped, size)
    conv = np.fft.irfft(fa * fc, size)
    grad = conv[n - 1:2 * n - 1][::-1].copy()
    count = active_mask.sum()
    if count > 0:
        grad /= count
    return grad


def subgradient_descent(a, max_iters, time_budget, epsilon=1e-6, top_k=None,
                        base_step=0.1, momentum=0.8, restart_every=200):
    """Nesterov-style subgradient descent with adaptive step size."""
    n = len(a)
    a = a.copy().astype(np.float64)
    S = np.sum(a)
    if S < 1e-10:
        return a
    a = a / S

    best_a = a.copy()
    best_obj = compute_obj(a)
    v = np.zeros(n)
    start = time.monotonic()
    stall = 0
    step = base_step

    for it in range(max_iters):
        if time.monotonic() - start > time_budget:
            break

        conv = compute_conv(a)
        M = float(np.max(conv))

        # Find active set
        if top_k is not None and top_k < len(conv):
            active_indices = np.argsort(conv)[-top_k:]
            active_mask = np.zeros(2 * n - 1, dtype=bool)
            active_mask[active_indices] = True
        else:
            threshold = M * (1.0 - epsilon)
            active_mask = conv >= threshold

        # Compute gradient
        grad = compute_grad_fft(a, active_mask)

        # Center gradient to preserve sum
        grad_centered = grad - np.mean(grad)

        # Nesterov momentum
        v = momentum * v + (1.0 - momentum) * grad_centered

        # Adaptive step: decrease over time, reset on restart
        eff_step = step / (1.0 + 0.005 * it)

        # Update
        a_new = a - eff_step * v
        a_new = np.maximum(a_new, 0.0)
        S_new = np.sum(a_new)
        if S_new < 1e-10:
            step *= 0.5
            v = np.zeros(n)
            continue
        a_new = a_new / S_new

        obj = compute_obj(a_new)
        if obj < best_obj - 1e-10:
            best_obj = obj
            best_a = a_new.copy()
            a = a_new
            stall = 0
        else:
            stall += 1
            if stall % 50 == 0:
                step *= 0.8
            if stall > restart_every:
                # Stochastic restart
                rng = np.random.default_rng(hash((n, it)) & 0xFFFFFFFF)
                a = a * np.exp(rng.normal(0, 0.08, n))
                a = np.maximum(a, 0)
                S = np.sum(a)
                if S < 1e-10:
                    a = np.ones(n) / n
                    S = 1.0
                a = a / S
                v = np.zeros(n)
                step = max(base_step * 0.5, 0.02)
                stall = 0

        # Periodic momentum reset
        if it > 0 and it % restart_every == 0:
            v *= 0.5

    return best_a


def solve_lp_maxsum(a, M, time_limit=5):
    """Solve LP: max sum(g) s.t. conv(a,g) <= M, g >= 0."""
    n = len(a)
    rows = []
    cols = []
    vals = []
    for j in range(n):
        start = j
        end = min(j + n, 2 * n - 1)
        length = end - start
        rows.extend(range(start, end))
        cols.extend([j] * length)
        vals.extend(list(a[:length]))
    A = coo_matrix((vals, (rows, cols)), shape=(2 * n - 1, n)).tocsr()
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': time_limit})
        if res.status == 0:
            return np.maximum(res.x, 0)
    except Exception:
        pass
    return None


def alternating_lp(a, max_iters=30, time_budget=25.0):
    """Alternating LP with geometric line search."""
    n = len(a)
    a = a.copy().astype(np.float64)
    S = np.sum(a)
    if S < 1e-10:
        return a
    a = a / S

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    stall = 0

    for it in range(max_iters):
        if time.monotonic() - start > time_budget:
            break

        conv = compute_conv(a)
        M = float(np.max(conv))
        M_try = M * (1.0 + 0.005 * (it % 4 - 1.5))

        g = solve_lp_maxsum(a, M_try, time_limit=4)
        if g is None:
            break

        S_g = np.sum(g)
        if S_g < 1e-10:
            break
        g = g / S_g

        best_t = 0.0
        best_obj_ls = best_obj
        for t in [0.05, 0.1, 0.15, 0.2, 0.3, 0.5, 0.7, 1.0]:
            a_try = (1.0 - t) * a + t * g
            a_try = np.maximum(a_try, 0)
            S_try = np.sum(a_try)
            if S_try < 1e-10:
                continue
            a_try = a_try / S_try
            obj_try = compute_obj(a_try)
            if obj_try < best_obj_ls:
                best_obj_ls = obj_try
                best_t = t

        if best_t > 0.0:
            a = (1.0 - best_t) * a + best_t * g
            a = np.maximum(a, 0)
            a = a / np.sum(a)
            best_obj = best_obj_ls
            best_a = a.copy()
            stall = 0
        else:
            stall += 1
            if stall > 10:
                rng = np.random.default_rng(it * 7919 + 123)
                a = a * np.exp(rng.normal(0, 0.1, n))
                a = np.maximum(a, 0)
                a = a / np.sum(a)
                stall = 0

    return best_a


def cosine_init(n, n_modes=10, seed=0):
    """Generate cosine-series initialization."""
    rng = np.random.default_rng(seed)
    k = np.arange(n)
    a = np.ones(n)
    for m in range(1, n_modes + 1):
        c = rng.uniform(-0.5, 0.5) / m
        phase = rng.uniform(0, 2 * np.pi)
        a += c * np.cos(2 * np.pi * m * k / n + phase)
    a = np.maximum(a, 0)
    S = np.sum(a)
    if S < 1e-10:
        return np.ones(n) / n
    return a / S


def random_smooth_init(n, seed=0):
    """Generate random smooth initialization via low-pass filtering."""
    rng = np.random.default_rng(seed)
    a = rng.uniform(0.5, 1.5, n)
    # Apply simple moving average for smoothness
    window = max(3, n // 20)
    kernel = np.ones(window) / window
    a = np.convolve(a, kernel, mode='same')
    a = np.maximum(a, 0)
    S = np.sum(a)
    if S < 1e-10:
        return np.ones(n) / n
    return a / S


def geometric_init(n, ratio=0.98, seed=0):
    """Generate geometric sequence initialization."""
    k = np.arange(n)
    center = (n - 1) / 2.0
    a = np.exp(-ratio * np.abs(k - center) / (n * 0.3))
    rng = np.random.default_rng(seed)
    a += rng.uniform(-0.1, 0.1, n)
    a = np.maximum(a, 0)
    S = np.sum(a)
    if S < 1e-10:
        return np.ones(n) / n
    return a / S

                        best_seq = a_p.copy()
                        print(f"  Perturb: {best_score:.6f} ** NEW BEST **")

    # Final validation
    if best_seq is None:
        best_seq = np.ones(300)
        best_score = compute_obj(best_seq)

    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    val = evaluate_sequence(result)
    if val < np.inf and val < best_score:
        best_score = val
    elif val == np.inf:
        result = [1.0] * 300
        best_score = evaluate_sequence(result)

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_score:.6f}, time={elapsed:.1f}s ===")

    return result
