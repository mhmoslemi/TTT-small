3. Adam optimizer on LSE-smoothed objective with simplex projection

import numpy as np
from scipy.optimize import linprog
from scipy.sparse import csr_matrix
import time

_spar_cache = {}

def get_sparsity(n):
    if n not in _spar_cache:
        rows = []
        cols = []
        for k in range(2 * n - 1):
            js = max(0, k - (n - 1))
            je = min(n - 1, k)
            for j in range(js, je + 1):
                rows.append(k)
                cols.append(j)
        _spar_cache[n] = (np.array(rows), np.array(cols), len(rows))
    return _spar_cache[n]

def fast_conv(a):
    """Fast linear convolution using FFT."""
    n = len(a)
    s = 1
    while s < 2 * n - 1:
        s <<= 1
    fa = np.fft.rfft(a, s)
    return np.fft.irfft(fa * fa, s)[:2 * n - 1]

def compute_J(a):
    """Compute J = 2n * max(a*a) / (sum a)^2, assuming sum(a) > 0."""
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-10:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))

def normalize(a):
    """Normalize to sum = 1, non-negative."""
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-10:
        return np.ones(len(a)) / len(a)
    return a / s

    """Solve LP: max sum(g) s.t. (a*g)_k <= M for all k, g >= 0."""
    n = len(a)

        idx = k - j
        if 0 <= idx < n:
            vals[i] = a[idx]
        else:
            vals[i] = 0.0
    A = csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    try:
        r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                    options={'presolve': True, 'time_limit': time_limit})
        if r.success and r.x is not None:
            g = np.maximum(r.x, 0.0)
            if g.sum() > 1e-10:
                return g
    except Exception:
        pass
    return None

def adam_lse(a, max_iters, deadline, tracker, tau0=20.0, lr0=None, beta1=0.9, beta2=0.999):
    """Adam optimizer on LSE-smoothed objective with simplex projection."""
    a = normalize(a.astype(np.float64))
    n = len(a)
    current_J = compute_J(a)
    tracker.update(a)
    
    if lr0 is None:
        lr0 = 1.0 / np.sqrt(n)
    
    tau = tau0
    lr = lr0
    m = np.zeros(n)
    v = np.zeros(n)
    t = 0
    
    while t < max_iters and time.monotonic() < deadline:
        t += 1
        b = fast_conv(a)
        b_max = b.max()
        # Softmax with temperature
        exp_b = np.exp(tau * (b - b_max))
        p = exp_b / exp_b.sum()
        
        # Gradient of LSE: 2 * cross_correlation(p, a)
        # cross_correlation(p, a)[j] = sum_k p[k] * a[k-j] (with valid indices)
        # This is the linear cross-correlation
        s = 1
        while s < 3 * n:
            s <<= 1
        fp = np.fft.rfft(p, s)
        fa = np.fft.rfft(a, s)
        corr = np.fft.irfft(fp * np.conj(fa), s)
        grad = 2.0 * corr[:n]
        
        # Adam update
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * grad * grad
        m_hat = m / (1 - beta1**t)
        v_hat = v / (1 - beta2**t)
        a_new = a - lr * m_hat / (np.sqrt(v_hat) + 1e-8)
        
        # Project onto simplex (sum=1, a>=0)
        a_new = np.maximum(a_new, 0.0)
        s_sum = a_new.sum()
        if s_sum < 1e-10:
            a_new = a.copy()
        else:
            a_new = a_new / s_sum
        
        J_new = compute_J(a_new)
        if J_new < current_J - 1e-12:
            a = a_new
            current_J = J_new
            # Slightly increase tau for more accuracy
            if t % 100 == 0:
                tau = min(500.0, tau * 1.02)
        else:
            lr *= 0.8
        
        if t % 200 == 0:
            tracker.update(a)
    
    tracker.update(a)
    return a

def frank_wolfe(a, max_iters, deadline, tracker, tau=50.0):
    """Frank-Wolfe (conditional gradient) method on LSE-smoothed objective."""
    a = normalize(a.astype(np.float64))
    n = len(a)
    current_J = compute_J(a)
    tracker.update(a)
    
    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        
        b = fast_conv(a)
        b_max = b.max()
        exp_b = np.exp(tau * (b - b_max))
        p = exp_b / exp_b.sum()
        
        # Gradient
        s = 1
        while s < 3 * n:
            s <<= 1
        fp = np.fft.rfft(p, s)
        fa = np.fft.rfft(a, s)
        corr = np.fft.irfft(fp * np.conj(fa), s)
        grad = 2.0 * corr[:n]
        
        # FW step: s = e_{argmin grad}
        j_min = np.argmin(grad)
        s_vec = np.zeros(n)
        s_vec[j_min] = 1.0
        
        # Step size: gamma = 2/(it+2) (standard FW)
        gamma = 2.0 / (it + 2)
        a_new = (1 - gamma) * a + gamma * s_vec
        
        J_new = compute_J(a_new)
        if J_new < current_J - 1e-14:
            a = a_new
            current_J = J_new
        else:
            # Try smaller step
            gamma = 1.0 / (it + 2)
            a_new = (1 - gamma) * a + gamma * s_vec
            J_new = compute_J(a_new)
            if J_new < current_J - 1e-14:
                a = a_new
                current_J = J_new
        
        if it % 100 == 0:
            tracker.update(a)
    
    tracker.update(a)
    return a

def lp_refine(a, max_iters, deadline, tracker, tighten_start=0.5, tighten_end=0.95):
    """LP best-response refinement with adaptive tightening."""
    a = normalize(a.astype(np.float64))
    n = len(a)
    current_J = compute_J(a)
    tracker.update(a)
    
    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        
        b = fast_conv(a)
        M = float(b.max())
        
        # Adaptive tightening: start loose, get tighter
        progress = it / max(1, max_iters - 1)
        tighten = tighten_start + (tighten_end - tighten_start) * progress
        M_tight = M * (1.0 - tighten)
        
        g = lp_best_response(a, M_tight, time_limit=5)
        if g is None:
            g = lp_best_response(a, M * (1 - tighten * 0.5), time_limit=5)
        if g is None:
            g = lp_best_response(a, M, time_limit=5)
        if g is None:
            break
        
        g = normalize(g)
        
        # Line search for mixing parameter
        best_J = current_J
        best_alpha = 0.0
        for alpha in [0.1, 0.2, 0.3, 0.5, 0.7, 0.9, 1.0]:
            a_trial = normalize((1 - alpha) * a + alpha * g)
            J_trial = compute_J(a_trial)
            if J_trial < best_J:
                best_J = J_trial
                best_alpha = alpha
        
        if best_alpha == 0.0:
            break
        
        a = normalize((1 - best_alpha) * a + best_alpha * g)
        current_J = best_J
        
        if it % 5 == 0:
            tracker.update(a)
    
    tracker.update(a)
    return a

def generate_inits(n, rng, parent=None):
    """Generate diverse initializations."""
    inits = []
    t = np.linspace(0, 1, n, endpoint=False)
    
    # Uniform
    inits.append(np.ones(n) * 1.0)
    
    # Linear ramps
    inits.append(t + 0.01)
    inits.append(1.0 - t + 0.01)
    
    # Quadratic
    inits.append(t**2 + 0.01)
    inits.append((1-t)**2 + 0.01)
    
    # Exponential decay
    for lam in [0.5, 1.0, 2.0, 4.0, 8.0]:
        inits.append(np.exp(-lam * t))
        inits.append(np.exp(-lam * (1 - t)))
    
    # Gaussian (various positions)
    for mu in [0.2, 0.3, 0.5, 0.7, 0.8]:
        for sigma in [0.05, 0.1, 0.2, 0.3]:
            inits.append(np.exp(-((t - mu)**2) / (2 * sigma**2)) + 0.01)
    
    # Sine windows
    for freq in [0.5, 1.0, 1.5, 2.0, 3.0]:
        inits.append(0.5 + 0.5 * np.sin(np.pi * freq * t + 0.5))
    
    # Power law
    for alpha in [0.5, 1.0, 1.5, 2.0, 3.0]:
        inits.append((t + 0.01)**(-alpha))
        inits.append((1 - t + 0.01)**(-alpha))
    
    # Cosine raised
    inits.append(np.cos(np.pi * t / 2)**2 + 0.01)
    inits.append(np.sin(np.pi * t / 2)**2 + 0.01)
    
    # Flat spectrum (random phases)
    for _ in range(3):
        half = n // 2 + 1
        angles = rng.uniform(0, 2 * np.pi, half)
        A = np.zeros(n, dtype=complex)
        A[:half] = np.exp(1j * angles[:half])
        for k in range(1, n - n // 2):
            A[k] = np.conj(A[n - k])
        a_init = np.fft.ifft(A).real
        inits.append(np.maximum(a_init, 0.0) + 0.01)
    
    # Random
    inits.append(rng.uniform(0.1, 1.0, n))
    
    # From parent
    if parent is not None and parent.size > 0:
        t_old = np.linspace(0, 1, parent.size)
        t_new = np.linspace(0, 1, n, endpoint=False)
        inits.append(np.maximum(np.interp(t_new, t_old, parent), 0.01))
        inits.append(np.maximum(np.interp(t_new, t_old, parent[::-1]), 0.01))
    
    # Normalize
    result = []
    for x in inits:
        x = np.maximum(np.asarray(x, dtype=np.float64), 0.001)
        x = x / x.sum()
        result.append(x)
    return result

import numpy as np

_spar_cache = {}

def get_sparsity(n):
    """Build sparse matrix structure for convolution constraint."""
    if n not in _spar_cache:
        rows = []
        cols = []
        for k in range(2 * n - 1):
            js = max(0, k - (n - 1))
            je = min(n - 1, k)
            for j in range(js, je + 1):
                rows.append(k)
                cols.append(j)
        _spar_cache[n] = (np.array(rows, dtype=np.int32), np.array(cols, dtype=np.int32), len(rows))
    return _spar_cache[n]

def fast_conv(a):
    """Fast linear self-convolution via FFT."""
    n = len(a)
    s = 1
    while s < 2 * n - 1:
        s <<= 1
    fa = np.fft.rfft(a, s)
    return np.fft.irfft(fa * fa, s)[:2 * n - 1]

def compute_J(a):
    """Compute J = 2n * max(a*a) / (sum a)^2."""
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-10:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))

def normalize(a):
    """Normalize to non-negative, sum=1."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-10:
        return np.ones(a.size) / a.size
    return a / s

def lp_best_response(a, M, time_limit=8):
    """Solve: max sum(g) s.t. (a*g)_k <= M for all k, g >= 0."""
    n = len(a)
    rows, cols, nnz = get_sparsity(n)
    vals = np.empty(nnz, dtype=np.float64)
    for i in range(nnz):
        k = int(rows[i])
        j = int(cols[i])
        idx = k - j
        if 0 <= idx < n:
            vals[i] = a[idx]
        else:
            vals[i] = 0.0
    A = csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    try:
        r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                    options={'presolve': True, 'time_limit': time_limit})
        if r.success and r.x is not None:
            g = np.maximum(r.x, 0.0)
            if g.sum() > 1e-10:
                return g
    except Exception:
        pass
    return None

def adam_lse(a, max_iters, deadline, tracker, tau0=30.0, lr0=None):
    """Adam optimizer on LSE-smoothed objective."""
    a = normalize(a)
    n = len(a)
    current_J = compute_J(a)
    tracker.update(a)
    
    if lr0 is None:
        lr0 = 2.0 / np.sqrt(n)
    
    tau = tau0
    lr = lr0
    m = np.zeros(n)
    v = np.zeros(n)
    t = 0
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    
    s_fft = 1
    while s_fft < 3 * n:
        s_fft <<= 1
    
    while t < max_iters and time.monotonic() < deadline:
        t += 1
        b = fast_conv(a)
        b_max = b.max()
        
        if b_max < 1e-30:
            break
        
        exp_b = np.exp(np.clip(tau * (b - b_max), -500, 0))
        p = exp_b / exp_b.sum()
        
        # Gradient: 2 * cross_correlation(p, a)
        fp = np.fft.rfft(p, s_fft)
        fa = np.fft.rfft(a, s_fft)
        corr = np.fft.irfft(fp * np.conj(fa), s_fft)
        grad = 2.0 * corr[:n]
        
        # Adam update
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * grad * grad
        m_hat = m / (1 - beta1**t)
        v_hat = v / (1 - beta2**t)
        
        a_new = a - lr * m_hat / (np.sqrt(v_hat) + eps)
        a_new = np.maximum(a_new, 0.0)
        s_sum = a_new.sum()
        if s_sum < 1e-10:
            a_new = a.copy()
        else:
            a_new = a_new / s_sum
        
        J_new = compute_J(a_new)
        if J_new < current_J - 1e-14:
            a = a_new
            current_J = J_new
            if t % 50 == 0 and tau < 300:
                tau = min(300.0, tau * 1.05)
                lr = lr * 0.98
        else:
            lr *= 0.7
            if lr < 1e-10:
                break
        
        if t % 300 == 0:
            tracker.update(a)
    
    tracker.update(a)
    return a

def frank_wolfe(a, max_iters, deadline, tracker, tau=80.0):
    """Frank-Wolfe conditional gradient method."""
    a = normalize(a)
    n = len(a)
    current_J = compute_J(a)
    tracker.update(a)
    
    s_fft = 1
    while s_fft < 3 * n:
        s_fft <<= 1
    
    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        
        b = fast_conv(a)
        b_max = b.max()
        if b_max < 1e-30:
            break
        exp_b = np.exp(np.clip(tau * (b - b_max), -500, 0))
        p = exp_b / exp_b.sum()
        
        fp = np.fft.rfft(p, s_fft)
        fa = np.fft.rfft(a, s_fft)
        corr = np.fft.irfft(fp * np.conj(fa), s_fft)
        grad = 2.0 * corr[:n]
        
        # FW: move toward e_{argmin grad}
        j_min = int(np.argmin(grad))
        gamma = 2.0 / (it + 2)
        a_new = (1.0 - gamma) * a
        a_new[j_min] += gamma
        a_new = np.maximum(a_new, 0.0)
        a_new /= a_new.sum()
        
        J_new = compute_J(a_new)
        if J_new < current_J - 1e-14:
            a = a_new
            current_J = J_new
        
        if it % 200 == 0:
            tracker.update(a)
    
    tracker.update(a)
    return a

def generate_inits(n, rng, parent=None):

    # Quadratic / cubic

    inits.append((1.0 - t)**2 + 0.01)
    inits.append(t**3 + 0.01)
    
    # Sine/cosine windows
    for freq in [0.5, 1.0, 1.5, 2.0]:

        inits.append((t + 0.02)**(-alpha))

    for mu in [0.2, 0.4, 0.6, 0.8]:
        inits.append(np.exp(-((t - mu) / 0.15)**2) + 0.02)
    
    # Flat spectrum (random phase)
    for _ in range(4):
        half = n // 2 + 1
        angles = rng.uniform(0, 2 * np.pi, half)
        A = np.zeros(n, dtype=complex)
        A[:half] = np.exp(1j * angles[:half])
        for k in range(1, n - n // 2):
            A[k] = np.conj(A[n - k])
        a_init = np.fft.ifft(A).real
        inits.append(np.maximum(a_init, 0.0) + 0.01)
    
    # Weighted flat spectrum (emphasize low frequencies)
    for _ in range(2):
        half = n // 2 + 1
        angles = rng.uniform(0, 2 * np.pi, half)
        mags = 1.0 / (np.arange(half) + 1) ** 0.5
        A = np.zeros(n, dtype=complex)

        for k in range(1, n - n // 2):

        a_init = np.fft.ifft(A).real
        inits.append(np.maximum(a_init, 0.0) + 0.01)
    
    # Random
    inits.append(rng.uniform(0.1, 1.0, n))
    
    # From parent (resampled)
    if parent is not None and parent.size > 1:
        t_old = np.linspace(0, 1, parent.size)
        inits.append(np.maximum(np.interp(t, t_old, parent), 0.01))
        inits.append(np.maximum(np.interp(t, t_old, parent[::-1]), 0.01))
        # Add noise to parent
        noisy = parent.copy()
        t_p = np.linspace(0, 1, len(noisy))
        base = np.interp(t, t_p, noisy)
        inits.append(np.maximum(base * (1 + 0.1 * rng.standard_normal(n)), 0.01))
    
    # Normalize all
    result = []
    for x in inits:
        x = np.maximum(np.asarray(x, dtype=np.float64), 0.001)
        s = x.sum()
        if s > 1e-10:
            result.append(x / s)

                    g = lp_best_response(a_large, M * 0.95, time_limit=lp_tl)
                if g is None:
                    break
                g = normalize(g)
                a_trial = normalize(0.5 * a_large + 0.5 * g)
                if compute_J(a_trial) < compute_J(a_large):
                    a_large = a_trial
                else:
                    a_trial2 = normalize(0.8 * a_large + 0.2 * g)
                    if compute_J(a_trial2) < compute_J(a_large):
                        a_large = a_trial2
                    else:
                        break
            
            # Adam polish
            if time.monotonic() < deadline - 30:
                a_large = adam_lse(a_large, 5000, deadline - 30, tracker, tau0=30.0)
            
            tracker.update(a_large)
            J_large = compute_J(a_large)
            print(f"  n={n_large}: J={J_large:.8f}", flush=True)
    
    # Final verification
    tracker.update(tracker.best)
    print(f"\n{'='*60}", flush=True)
    print(f"FINAL: J={tracker.best_score:.8f} n={tracker.best.size} "
          f"time={time.monotonic()-tracker.t_start:.1f}s", flush=True)
    print(f"{'='*60}", flush=True)
    
    if tracker.best is None:
        tracker.best = np.ones(256)
        tracker.best_score = compute_J(tracker.best)
    
    return [float(x) for x in tracker.best]
