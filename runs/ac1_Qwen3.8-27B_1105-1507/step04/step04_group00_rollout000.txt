import numpy as np
from scipy.optimize import linprog, minimize
from scipy.sparse import csr_matrix
import time


def fast_conv(a):
    """Full self-convolution via FFT."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]


def compute_J(a):
    """Compute J(a) = 2*n*max(conv(a,a))/(sum(a))^2."""
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))


def smooth_max_gradient(a, tau):
    """Compute gradient of smooth max of conv(a,a) w.r.t. a."""
    n = len(a)
    b = fast_conv(a)
    b_max = b.max()
    exp_b = np.exp(tau * (b - b_max))
    p = exp_b / exp_b.sum()
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        pk = p[k]
        if pk < 1e-13:
            continue
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        grad[lo:hi + 1] += 2.0 * pk * a[k - hi:k - lo + 1][::-1]
    return grad, p, b


def subgradient_step(a, tol_frac=1e-8):
    """Compute exact subgradient of max(conv(a,a))."""
    n = len(a)
    b = fast_conv(a)
    M = b.max()
    tol = tol_frac * M if M > 0 else 1e-10
    active = np.where(b >= M - tol)[0]
    grad = np.zeros(n)
    for k in active:
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        grad[lo:hi + 1] += 2.0 * a[k - hi:k - lo + 1][::-1]
    gnorm = np.linalg.norm(grad)
    if gnorm > 1e-15:
        grad /= gnorm
    return grad, active, b


def run_smooth_gradient(a, max_iters, deadline, initial_tau=2.0, final_tau=80.0,
                        lr_init=0.3, lr_min=0.01, momentum=True):
    """Smooth gradient descent with adaptive tau and learning rate."""
    a = np.maximum(a.astype(np.float64), 1e-8)
    a = a / a.sum()
    n = len(a)
    best_a = a.copy()
    best_M = fast_conv(a).max()
    v = np.zeros(n)
    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        frac = it / max(1, max_iters)
        tau = initial_tau + (final_tau - initial_tau) * (frac ** 0.5)
        lr = lr_init * (lr_min / lr_init) ** frac
        grad, p, b = smooth_max_gradient(a, tau)
        M = b.max()
        if M < best_M * (1 - 1e-8):
            best_M = M
            best_a = a.copy()
        beta = 0.9 if momentum else 0.0
        v = beta * v + lr * grad
        a_new = a - v
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s < 1e-10:
            a_new = a.copy()
        else:
            a_new = a_new / s
        a = a_new
    if fast_conv(best_a).max() < fast_conv(a).max():
        return best_a
    return a


def run_subgradient(a, max_iters, deadline, lr_init=0.2, averaging=True):
    """Subgradient descent with exact max."""
    a = np.maximum(a.astype(np.float64), 1e-8)
    a = a / a.sum()
    n = len(a)
    best_a = a.copy()
    best_M = fast_conv(a).max()
    running_sum = np.zeros(n)
    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        lr = lr_init / (it + 1) ** 0.5
        grad, active, b = subgradient_step(a)
        M = b.max()
        if M < best_M * (1 - 1e-8):
            best_M = M
            best_a = a.copy()
        a_new = a - lr * grad
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s < 1e-10:
            a_new = a.copy()
        else:
            a_new = a_new / s
        if averaging:
            running_sum += a_new
        a = a_new
    if averaging and max_iters > 10:
        a_avg = running_sum / max_iters
        a_avg = np.maximum(a_avg, 0.0)
        s = a_avg.sum()
        if s > 1e-10:
            a_avg /= s
            M_avg = fast_conv(a_avg).max()
            if M_avg < best_M:
                best_a = a_avg
    return best_a


def build_A_sparse(n, a):
    """Build sparse constraint matrix for LP."""
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        j_min = max(0, k - n + 1)
        j_max = min(n - 1, k)
        for j in range(j_min, j_max + 1):
            rows.append(k)
            cols.append(j)
            vals.append(a[k - j])
    A = csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))
    return A


def lp_best_response(a, M):
    """LP: max sum(g) s.t. (a*g)[k] <= M for all k, g >= 0."""
    n = len(a)
    A = build_A_sparse(n, a)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    try:
        r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                    options={'time_limit': 60})
        if r.success and r.x is not None:
            g = np.maximum(r.x, 0)
            if g.sum() > 1e-15:
                return g
    except Exception:
        pass
    return None


def run_lp_mirror_descent(a, max_iters, deadline, verbose=False):
    """LP mirror descent with adaptive step size and dual-gap monitoring."""
    a = np.maximum(a.astype(np.float64), 1e-8)
    a = a / a.sum()
    current_M = fast_conv(a).max()
    best_a = a.copy()
    best_M = current_M
    no_improve = 0
    best_t = 0.3

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        b = fast_conv(a)
        M = float(b.max())
        g = lp_best_response(a, M)
        if g is None:
            break
        g = np.maximum(g, 0.0)
        g_sum = g.sum()
        if g_sum < 1e-10:
            break
        g /= g_sum

        # Compute dual gap: how different is g from a?
        gap = np.linalg.norm(g - a)

        # Adaptive base step: larger when gap is large, smaller when converging
        progress = it / max(1, max_iters)
        base_t = max(0.03, best_t * (1.0 - 0.5 * progress))

        improved = False
        for t in [base_t, base_t * 0.6, base_t * 0.3, 0.05, 0.02, 0.01]:
            a_trial = np.maximum((1 - t) * a + t * g, 0)
            s = a_trial.sum()
            if s < 1e-10:
                continue
            a_trial /= s
            M_trial = fast_conv(a_trial).max()
            if M_trial < best_M - 1e-14:
                a = a_trial
                best_M = M_trial
                best_a = a.copy()
                improved = True
                if t > 0.05:
                    best_t = max(0.03, min(0.6, t * 1.1))
                break

        if not improved:
            no_improve += 1
            best_t *= 0.7
            if no_improve > 30:
                break
        else:
            no_improve = 0

        if verbose and it % 20 == 0:
            print(f"    LP iter {it}: M={M:.10f}, best_M={best_M:.10f}, gap={gap:.4f}", flush=True)

    return best_a


def convolution_flattening_step(a, n_steps=5, deadline=None):
    """
    Targeted perturbation that identifies peak convolution coefficients
    and creates moves to reduce them specifically.
    """
    a = np.maximum(a.astype(np.float64), 1e-8)
    a = a / a.sum()
    n = len(a)
    best_a = a.copy()
    best_M = fast_conv(a).max()

    for step in range(n_steps):
        if deadline is not None and time.monotonic() > deadline:
            break

        b = fast_conv(a)
        M = b.max()
        # Find peak positions
        peak_thresh = M * 0.95
        peak_pos = np.where(b >= peak_thresh)[0]

        # For each peak position k, the gradient contribution points toward
        # reducing a[i]*a[k-i]. We create a "repulsion" from peak positions.
        repulsion = np.zeros(n)
        for k in peak_pos:
            lo = max(0, k - n + 1)
            hi = min(n - 1, k)
            # Weight by how much above threshold
            w = (b[k] - peak_thresh) / (M - peak_thresh + 1e-15)
            repulsion[lo:hi + 1] += w * a[k - hi:k - lo + 1][::-1]

        # Apply repulsion and renormalize
        lr = 0.05 * (1.0 - step / n_steps)
        a_new = a - lr * repulsion
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s < 1e-10:
            continue
        a_new /= s
        M_new = fast_conv(a_new).max()
        if M_new < best_M:
            best_M = M_new
            best_a = a_new.copy()
            a = a_new
        else:
            break

    return best_a


def resample_seq(src, dest_n):
    """Resample 1D sequence to new length via linear interpolation."""
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)


def generate_initializations(n, rng, parent=None):
    """Generate diverse initial sequences of length n."""
    inits = []
    t_arr = np.linspace(0, 1, n, endpoint=False)

    if parent is not None:
        inits.append(resample_seq(parent, n))
        p = resample_seq(parent, n)
        inits.append(np.maximum(p + rng.normal(0, 0.02, n), 0.01))
        inits.append(np.maximum(p + rng.normal(0, 0.08, n), 0.01))
        # Structured perturbation: modulate by cosine
        for freq in [2, 3, 5]:
            mod = 1.0 + 0.1 * np.cos(2 * np.pi * freq * t_arr)
            inits.append(np.maximum(p * mod, 0.01))

    inits.append(np.ones(n))
    inits.append(rng.uniform(0.5, 1.0, n))

    # Various frequency structures designed to flatten autocorrelation
    for freq in [1, 2, 3, 4, 5, 7]:
        inits.append(np.maximum(1.0 + 0.3 * np.cos(2 * np.pi * freq * t_arr), 0.01))
        inits.append(np.maximum(1.0 + 0.4 * np.sin(2 * np.pi * freq * t_arr), 0.01))

    # Optimal-ish shapes: raised cosine, watermelon, etc.
    inits.append(0.5 + 0.5 * np.cos(np.pi * t_arr))  # raised cosine
    inits.append(np.sqrt(np.maximum(1.0 - (2 * t_arr - 1) ** 2, 0.01)) + 0.05)  # semicircle
    inits.append(np.maximum(1.0 - 0.3 * (2 * t_arr - 1) ** 2, 0.01))  # parabola

    # Chirp
    inits.append(np.maximum(1.0 + 0.3 * np.cos(2 * np.pi * 3 * t_arr ** 2), 0.01))

    # Gaussian bumps
    inits.append(np.exp(-3.0 * (t_arr - 0.5) ** 2) + 0.1)
    inits.append(np.exp(-6.0 * (t_arr - 0.3) ** 2) + np.exp(-6.0 * (t_arr - 0.7) ** 2) + 0.05)
    inits.append(0.2 + np.exp(-10 * (t_arr - 0.2) ** 2) + np.exp(-10 * (t_arr - 0.8) ** 2))

    # Multi-peak
    n_peaks = rng.integers(3, 8)
    centers = rng.uniform(0.1, 0.9, n_peaks)
    widths = rng.uniform(0.05, 0.15, n_peaks)
    multi = np.zeros(n)
    for c, w in zip(centers, widths):
        multi += np.exp(-((t_arr - c) / w) ** 2)
    inits.append(multi + 0.05)

    # Smooth random
    ksize = max(3, n // 20)
    kernel = np.ones(ksize) / ksize
    smooth = np.convolve(rng.uniform(0, 1, n), kernel, mode='same')
    inits.append(np.maximum(smooth + 0.1, 0.01))

    # Block structure
    n_blocks = rng.integers(4, max(5, n // 30))
    block_vals = rng.uniform(0.3, 1.0, n_blocks)
    block_seq = np.repeat(block_vals, n // n_blocks + 1)[:n]
    inits.append(block_seq)

    # Power asymmetric
    inits.append(np.maximum(0.3 + 0.7 * t_arr ** 2, 0.01))
    inits.append(np.maximum(0.3 + 0.7 * (1 - t_arr) ** 2, 0.01))

    # Log-sine shape (good for flat autocorrelation)
    inits.append(np.maximum(1.0 + 0.2 * np.log(1 + 10 * np.sin(np.pi * t_arr) ** 2), 0.01))

    return inits

def _lbb_objective(x):
    """Objective for L-BFGS-B: smooth max proxy."""
    ax = np.maximum(x, 0.0)
    s = ax.sum()
    if s < 1e-8:
        return 1e10
    ax = ax / s
    b = fast_conv(ax)
    tau = 200.0
    b_shifted = b - b.max()
    return float(np.log(np.sum(np.exp(tau * b_shifted))) / tau)


def _lbb_objective_grad(x):
    """Gradient for L-BFGS-B smooth objective."""
    ax = np.maximum(x, 0.0)
    s = ax.sum()
    if s < 1e-8:
        return np.zeros_like(x)
    ax = ax / s
    n = len(ax)
    b = fast_conv(ax)
    tau = 200.0
    b_max = b.max()
    exp_b = np.exp(tau * (b - b_max))
    p = exp_b / exp_b.sum()
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        pk = p[k]
        if pk < 1e-12:
            continue
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        grad[lo:hi + 1] += 2.0 * pk * ax[k - hi:k - lo + 1][::-1]
    return grad

    fine_a = fine_a / fine_a.sum()

    # Quick smooth gradient at fine scale
    iters = max(50, min(300, 50000 // fine_n))
    fine_a = run_smooth_gradient(fine_a, iters, deadline,
                                  initial_tau=3.0, final_tau=50.0,
                                  lr_init=0.15, lr_min=0.02)
    return fine_a


def propose_candidate(seed=42, budget_s=1500, **kwargs):
    """
    Multi-phase hybrid search with convolution-aware targeting:
    Phase 1: Systematic multi-length exploration (smooth gradient)
    Phase 2: Convolution flattening + subgradient on top candidates
    Phase 3: Heavy LP mirror descent with adaptive scheduling
    Phase 4: Targeted peak-reduction mutations + LP
    Phase 5: L-BFGS-B polish + final LP sweep
    Phase 6: Length tuning (try n±small adjustments)
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + float(budget_s) - 20.0
    tracker = ScoreTracker()

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        if p.ndim == 1 and p.size > 0 and np.all(np.isfinite(p)):
            parent = np.maximum(p, 0.0)
            tracker.update(parent)
            print(f"Parent loaded: n={parent.size}, J={tracker.best_score:.6f}", flush=True)
    except (NameError, AttributeError):
        pass

    if parent is None:
        parent = np.ones(256)
        tracker.update(parent)

    current_best = parent.copy()

    # === PHASE 1: Multi-length exploration ===
    p1_start = time.monotonic()
    total_time_p1 = 300.0
    print(f"\n{'='*60}", flush=True)
    print(f"PHASE 1: Multi-Length Exploration (budget={total_time_p1:.0f}s)", flush=True)
    print(f"{'='*60}", flush=True)

    # Choose lengths strategically: include parent length and neighbors
    parent_n = parent.size if parent is not None else 256
    base_lengths = [128, 192, 256, 320, 384, 448, 512, 576, 640, 768, 896, 1024]
    # Add parent length and nearby
    for dn in [-64, -32, 0, 32, 64]:
        candidate = parent_n + dn
        if 64 <= candidate <= 2048:
            base_lengths.append(candidate)
    lengths = sorted(set(base_lengths))
    lengths = [n for n in lengths if 64 <= n <= 2048]

    # Track candidates: list of (score, sequence, n)
    candidates = []

    for n in lengths:
        if time.monotonic() > deadline - 250:
            break

        # Time allocation: shorter gets less, but ensure minimum
        weight = 1.0 / (n ** 0.5)
        total_weight = sum(1.0 / (nn ** 0.5) for nn in lengths)
        time_for_n = total_time_p1 * weight / total_weight
        time_for_n = max(15, min(60, time_for_n))
        phase_deadline = min(time.monotonic() + time_for_n, deadline - 200)

        if time.monotonic() >= phase_deadline:
            continue

        t0 = time.monotonic()
        # Limit initializations based on available time
        n_inits = min(8, max(3, int(time_for_n / 8)))
        inits = generate_initializations(n, rng, current_best)[:n_inits]
        best_for_n = None
        best_M_for_n = float('inf')

        for idx, a0 in enumerate(inits):
            if time.monotonic() > phase_deadline:
                break
            a0 = np.maximum(a0.astype(np.float64), 1e-6)
            a0 = a0 / a0.sum()

            iters = max(30, int(3000 / max(1, n // 64)))
            a = run_smooth_gradient(a0, iters, phase_deadline,
                                     initial_tau=2.0, final_tau=50.0,
                                     lr_init=0.25, lr_min=0.02)

            M_val = fast_conv(a).max()
            if M_val < best_M_for_n:
                best_M_for_n = M_val
                best_for_n = a.copy()

        if best_for_n is not None:
            J_val = compute_J(best_for_n)
            candidates.append((J_val, best_for_n, n))
            print(f"  n={n:4d}: J={J_val:.6f} ({time.monotonic()-t0:.1f}s)", flush=True)
            if J_val < compute_J(current_best):
                current_best = best_for_n.copy()
            tracker.update(best_for_n)

    # Sort candidates by score
    candidates.sort(key=lambda x: x[0])
    tracker.report("Phase 1 complete")
    print(f"Phase 1 total time: {time.monotonic()-p1_start:.1f}s", flush=True)

    # === PHASE 2: Convolution flattening + subgradient on top candidates ===
    p2_start = time.monotonic()
    print(f"\n{'='*60}", flush=True)
    print(f"PHASE 2: Convolution Flattening + Subgradient (budget=~120s)", flush=True)
    print(f"{'='*60}", flush=True)

    top_k = min(5, len(candidates))
    for i in range(top_k):
        if time.monotonic() > deadline - 200:
            break
        _, a_cand, n_cand = candidates[i]
        t0 = time.monotonic()

        # Convolution flattening
        a_flat = convolution_flattening_step(a_cand, n_steps=10,
                                              deadline=min(time.monotonic() + 10, deadline - 200))
        tracker.update(a_flat)

        # Subgradient polish
        a_sub = run_subgradient(a_flat, 2000, min(time.monotonic() + 30, deadline - 200),
                                lr_init=0.12, averaging=True)
        tracker.update(a_sub)

        print(f"  Candidate {i} (n={n_cand}): {time.monotonic()-t0:.1f}s", flush=True)

    tracker.report("Phase 2 complete")
    print(f"Phase 2 total time: {time.monotonic()-p2_start:.1f}s", flush=True)

    # === PHASE 3: Heavy LP mirror descent ===
    p3_start = time.monotonic()
    print(f"\n{'='*60}", flush=True)
    print(f"PHASE 3: LP Mirror Descent (budget=~300s)", flush=True)
    print(f"{'='*60}", flush=True)

    if tracker.best is not None:
        n_best = tracker.best.size
        est_lp_time = 0.005 * (n_best / 128.0) ** 2
        max_iters = max(50, min(500, int(280 / max(0.01, est_lp_time))))
        print(f"  n={n_best}, est LP time={est_lp_time*1000:.0f}ms, max_iters={max_iters}", flush=True)

        a = run_lp_mirror_descent(tracker.best, max_iters,
                                  min(time.monotonic() + 200, deadline - 150),
                                  verbose=True)
        tracker.update(a)

        # Second LP pass
        if time.monotonic() < deadline - 100:
            a2 = run_lp_mirror_descent(a, max_iters // 2, deadline - 100, verbose=True)
            tracker.update(a2)

        # Third pass if improved
        if time.monotonic() < deadline - 80:
            a3 = run_lp_mirror_descent(a2, max_iters // 3, deadline - 80)
            tracker.update(a3)

    tracker.report("Phase 3 complete")
    print(f"Phase 3 total time: {time.monotonic()-p3_start:.1f}s", flush=True)

    # === PHASE 4: Targeted peak-reduction mutations ===
    p4_start = time.monotonic()
    print(f"\n{'='*60}", flush=True)
    print(f"PHASE 4: Targeted Mutations + LP (budget=~150s)", flush=True)
    print(f"{'='*60}", flush=True)

    if tracker.best is not None and time.monotonic() < deadline - 60:
        n_best = tracker.best.size
        est_lp_time = 0.005 * (n_best / 128.0) ** 2

        # Analyze the convolution structure
        b = fast_conv(tracker.best)
        M = b.max()
        b_mean = b.mean()

        # Strategy: create mutations that specifically reduce peak positions
        n_mutations = min(10, int((deadline - 60 - time.monotonic()) / max(1.0, est_lp_time * 3)))

        for m in range(n_mutations):
            if time.monotonic() > deadline - 40:
                break

            a = tracker.best.copy().astype(np.float64)

            if m % 3 == 0:
                # Small random perturbation
                a += rng.normal(0, 0.02, n_best)
            elif m % 3 == 1:
                # Targeted: reduce mass at positions contributing to peaks
                b_local = fast_conv(a)
                peak_pos = np.where(b_local >= b_local.max() * 0.9)[0]
                reduce_mask = np.zeros(n_best)
                for k in peak_pos:
                    lo = max(0, k - n_best + 1)
                    hi = min(n_best - 1, k)
                    reduce_mask[lo:hi+1] += 1
                a -= 0.01 * reduce_mask * a
            else:
                # Frequency-domain perturbation: modify spectrum
                size = 1
                while size < 2 * n_best - 1:
                    size <<= 1
                fa = np.fft.rfft(a, size)
                # Slightly flatten the magnitude spectrum
                mag = np.abs(fa)
                phase = np.angle(fa)
                mag_new = mag ** 0.95  # compress dynamic range
                fa_new = mag_new * np.exp(1j * phase)
                a = np.fft.irfft(fa_new, size)[:n_best]

            a = np.maximum(a, 1e-6)
            a = a / a.sum()

            # Quick smooth gradient
            a = run_smooth_gradient(a, 30, min(time.monotonic() + 10, deadline - 40),
                                     initial_tau=5, final_tau=40,
                                     lr_init=0.1, lr_min=0.02)
            tracker.update(a)

            # Short LP
            remaining = deadline - 40 - time.monotonic()
            if remaining > est_lp_time * 2:
                a = run_lp_mirror_descent(a, max(3, int(remaining / (est_lp_time * 3))),
                                          min(time.monotonic() + est_lp_time * 2, deadline - 40))
                tracker.update(a)

    tracker.report("Phase 4 complete")
    print(f"Phase 4 total time: {time.monotonic()-p4_start:.1f}s", flush=True)

    # === PHASE 5: L-BFGS-B polish ===
    p5_start = time.monotonic()
    print(f"\n{'='*60}", flush=True)
    print(f"PHASE 5: L-BFGS-B Polish (budget=~80s)", flush=True)
    print(f"{'='*60}", flush=True)

    if tracker.best is not None and time.monotonic() < deadline - 30:
        n_best = tracker.best.size
        bounds_list = [(0.0, None)] * n_best
        try:
            res = minimize(_lbb_objective, tracker.best.astype(np.float64),
                           jac=_lbb_objective_grad,
                           method='L-BFGS-B', bounds=bounds_list,
                           options={'maxiter': 500, 'max_fun': 100000, 'ftol': 1e-20,
                                    'gtol': 1e-12, 'maxls': 50})
            if res.fun < 1e10:
                tracker.update(res.x)
        except Exception as e:
            print(f"  L-BFGS-B error: {e}", flush=True)

    tracker.report("Phase 5 complete")
    print(f"Phase 5 total time: {time.monotonic()-p5_start:.1f}s", flush=True)

    # === PHASE 6: Length tuning ===
    p6_start = time.monotonic()
    print(f"\n{'='*60}", flush=True)
    print(f"PHASE 6: Length Tuning (remaining time)", flush=True)
    print(f"{'='*60}", flush=True)

    if tracker.best is not None and time.monotonic() < deadline - 15:
        base_n = tracker.best.size
        # Try neighboring lengths
        for delta in [-32, -16, 16, 32, -64, 64]:
            if time.monotonic() > deadline - 15:
                break
            new_n = base_n + delta
            if new_n < 64 or new_n > 3072:
                continue
            a_ext = resample_seq(tracker.best, new_n)
            a_ext = np.maximum(a_ext, 1e-6)
            a_ext = a_ext / a_ext.sum()
            a_ext = run_smooth_gradient(a_ext, 80, deadline - 15,
                                         initial_tau=3, final_tau=40,
                                         lr_init=0.12, lr_min=0.03)
            tracker.update(a_ext)

    # === FINAL: One more LP pass ===
    if tracker.best is not None and time.monotonic() < deadline - 2:
        n_best = tracker.best.size
        est_lp_time = 0.005 * (n_best / 128.0) ** 2
        max_final_iters = max(5, int((deadline - time.monotonic() - 2) / max(0.01, est_lp_time)))
        a = run_lp_mirror_descent(tracker.best, max_final_iters, deadline)
        tracker.update(a)

    # Fallback
    if tracker.best is None:
        tracker.best = np.ones(128)
        tracker.best_score = compute_J(tracker.best)

    print(f"\n{'='*60}", flush=True)
    print(f"FINAL: J={tracker.best_score:.8f} n={tracker.best.size} "
          f"time={time.monotonic()-tracker.t_start:.0f}s", flush=True)
    print(f"{'='*60}", flush=True)

    return [float(x) for x in tracker.best]
