    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]

    fb = np.fft.rfft(b, size)

    return float(2 * len(a) * b.max() / (s * s))


def coord_best_response(a, M, n_passes=40, seed=0):
    """Fast coordinate descent: find g>=0 maximizing sum(g) s.t. (a*g)_k <= M."""
    n = len(a)
    g = np.zeros(n)
    bg = np.zeros(2 * n - 1)
    rng = np.random.default_rng(seed)
    order = np.arange(n)

    for p in range(n_passes):
        rng.shuffle(order)
        improved = False
        for j in order:

            if L <= 0:

            a_s = a[:L]
            mask = a_s > 1e-15
            if not np.any(mask):
                continue
            base = bg[j:j + L] - g[j] * a_s
            safe_a = np.where(mask, a_s, 1.0)
            limits = (M - base) / safe_a
            limits = np.where(mask, limits, np.inf)
            g_new = max(0.0, np.min(limits))

                delta = g_new - g[j]
                g[j] = g_new
                bg[j:j + L] += delta * a_s
                improved = True
        if not improved:
            break
    return g


def best_response_lp(a, M, time_limit=5):
    """LP-based best response using scipy linprog (fallback/verification)."""
    from scipy.optimize import linprog
    from scipy.sparse import csr_matrix
    n = len(a)
    n_cons = 2 * n - 1
    rows = []
    cols = []
    for k in range(n_cons):
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        rows.extend([k] * (je - js + 1))
        cols.extend(range(js, je + 1))
    vals = np.array([a[k - j] for k, j in zip(rows, cols)])
    A = csr_matrix((vals, (np.array(rows), np.array(cols))), shape=(n_cons, n))
    c = -np.ones(n)
    b_ub = np.full(n_cons, M)
    bounds = [(0.0, None)] * n
    r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                options={'time_limit': time_limit, 'presolve': True})
    if r.success and r.x is not None:
        return np.maximum(r.x, 0.0)
    return None


def full_iteration(a, tighten, rng_seed):
    """One full iteration: best response + line search. Returns (a_new, J_new, improved)."""
    n = len(a)
    b_aa = fast_conv(a)
    M = b_aa.max()
    M_tight = M * (1.0 - tighten)

    g = coord_best_response(a, M_tight, n_passes=40, seed=rng_seed)
    S = g.sum()
    if S <= 1.0 + 1e-8:
        return a, compute_J(a), False

    g_norm = g / S
    c_ag = cross_conv(a, g_norm)
    d_gg = fast_conv(g_norm)

    def obj(t):
        val = (1 - t) ** 2 * b_aa + 2 * t * (1 - t) * c_ag + t * t * d_gg
        return val.max()

    result = minimize_scalar(obj, bounds=(0.001, 0.999), method='bounded',
                             options={'xatol': 1e-10})
    t_opt = max(0.01, result.x)

    a_new = (1 - t_opt) * a + t_opt * g_norm
    a_new = np.maximum(a_new, 0.0)
    s_new = a_new.sum()
    if s_new < 1e-10:
        return a, compute_J(a), False
    a_new /= s_new
    J_new = compute_J(a_new)
    return a_new, J_new, True


def optimize_sequence(a0, deadline, tracker, max_iters=5000, tighten_start=0.1,
                      tighten_end=0.001, min_iter_gap=200):
    """Main optimization loop with adaptive tightening."""
    a = np.maximum(a0.astype(np.float64), 0.0)
    a /= a.sum()
    n = len(a)
    current_J = compute_J(a)
    tracker.update(a)
    rng = np.random.default_rng(12345)

    no_improve = 0
    it = 0
    while it < max_iters and time.monotonic() < deadline:
        progress = np.min(1.0, it / max_iters)
        tighten = tighten_start * (tighten_end / tighten_start) ** progress

        a, J_new, improved = full_iteration(a, tighten, seed=rng.integers(0, 2**31))
        it += 1

        if improved and J_new < current_J - 1e-14:
            current_J = J_new
            no_improve = 0
            if it % 10 == 0:
                tracker.update(a)
        else:
            no_improve += 1
            if no_improve >= min_iter_gap:
                # Try LP as escape
                b_aa = fast_conv(a)
                M = b_aa.max()
                g_lp = best_response_lp(a, M * (1 - tighten * 0.5), time_limit=8)
                if g_lp is not None and g_lp.sum() > 1.0 + 1e-8:
                    g_norm = g_lp / g_lp.sum()
                    c_ag = cross_conv(a, g_norm)
                    d_gg = fast_conv(g_norm)

                    def obj2(t):
                        val = (1 - t) ** 2 * b_aa + 2 * t * (1 - t) * c_ag + t * t * d_gg
                        return val.max()

                    res = minimize_scalar(obj2, bounds=(0.001, 0.999), method='bounded')
                    a = np.maximum((1 - res.x) * a + res.x * g_norm, 0.0)
                    a /= a.sum()
                    J_new = compute_J(a)
                    if J_new < current_J:
                        current_J = J_new
                        no_improve = 0
                        tracker.update(a)
                    else:
                        no_improve += 1
                else:
                    if no_improve >= min_iter_gap * 3:
                        break
        if it % 50 == 0:
            tracker.update(a)

    tracker.update(a)
    return a


def flat_spectrum_init(n, rng):
    half = n // 2 + 1
    angles = rng.uniform(0, 2 * np.pi, half)
    A = np.zeros(n, dtype=complex)
    A[:half] = np.exp(1j * angles[:half])
    for k in range(1, n - n // 2):
        A[k] = np.conj(A[n - k])
    a = np.fft.ifft(A).real
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-10:
        return np.ones(n) / n
    return a / s

def generate_inits(n, rng, parent):
    inits = []
    t = np.linspace(0, 1, n, endpoint=False)

    if parent is not None and parent.size > 0:
        inits.append(upsample(parent, n))
        inits.append(upsample(parent[::-1], n))

    inits.append(np.ones(n) / n)
    inits.append(flat_spectrum_init(n, rng))
    inits.append(flat_spectrum_init(n, rng))

    # Cosine series
    for n_cos in [3, 5, 7]:
        sig = np.ones(n)
        for i in range(n_cos):
            freq = rng.integers(1, n // 2 + 1)
            phase = rng.uniform(0, 2 * np.pi)
            amp = rng.uniform(0.1, 0.4)
            sig += amp * np.cos(2 * np.pi * freq * t + phase)
        inits.append(np.maximum(sig, 0.01))

    # Gaussian bumps
    n_bumps = rng.integers(2, 5)
    sig = np.ones(n) * 0.3
    for _ in range(n_bumps):
        center = rng.uniform(0.1, 0.9)
        width = rng.uniform(0.05, 0.2)
        sig += np.exp(-((t - center) / width) ** 2)
    inits.append(sig)

    # Random smoothed
    raw = rng.uniform(0.1, 1.0, n)
    kernel_size = max(3, n // 20)
    kernel = np.ones(kernel_size) / kernel_size
    inits.append(np.convolve(raw, kernel, mode='same'))

    # Power law
    p = rng.uniform(-0.5, 1.5)
    inits.append((t + 0.1) ** p)

    # Sine wave
    freq = rng.uniform(1, 8)
    inits.append(1.0 + 0.3 * np.sin(2 * np.pi * freq * t + rng.uniform(0, 6.28)))

    for x in inits:
        x = np.maximum(x, 0.001).astype(np.float64)
        x /= x.sum()
    return inits


def local_polish(a, deadline, tracker):
    """Local search: try small perturbations and coordinate changes."""
    a = a.copy().astype(np.float64)
    a /= a.sum()
    n = len(a)
    current_J = compute_J(a)
    tracker.update(a)
    rng = np.random.default_rng(777)

    for step in range(20000):
        if time.monotonic() > deadline:
            break
        mode = rng.integers(0, 3)
        if mode == 0:
            # Random small perturbation
            a_pert = a + rng.normal(0, 0.02, n)
            a_pert = np.maximum(a_pert, 0.0)
            a_pert /= a_pert.sum()
        elif mode == 1:
            # Coordinate scale
            j = rng.integers(0, n)
            a_pert = a.copy()
            a_pert[j] *= rng.uniform(0.5, 1.5)
            a_pert = np.maximum(a_pert, 0.0)
            a_pert /= a_pert.sum()
        else:
            # Pair swap
            i = rng.integers(0, n)
            j = rng.integers(0, n)
            if i == j:
                continue
            a_pert = a.copy()
            a_pert[i], a_pert[j] = a_pert[j], a_pert[i]
            a_pert /= a_pert.sum()

        J_pert = compute_J(a_pert)
        if J_pert < current_J - 1e-12:
            a = a_pert
            current_J = J_pert
            if step % 100 == 0:
                tracker.update(a)

    tracker.update(a)
    return a


def coordinate_polish(a, deadline, tracker):
    """Cyclic coordinate descent on the original objective."""
    a = a.copy().astype(np.float64)
    a /= a.sum()
    n = len(a)
    current_J = compute_J(a)
    tracker.update(a)

    scales = [0.5, 0.7, 0.85, 1.15, 1.3, 1.5, 0.3, 2.0]
    n_scales = len(scales)

    for epoch in range(200):
        if time.monotonic() > deadline:
            break
        improved = False
        for j in range(n):
            if time.monotonic() > deadline:
                break
            for sc in scales:
                a_trial = a.copy()
                a_trial[j] *= sc
                if a_trial[j] < 0:
                    continue
                a_trial /= a_trial.sum()
                J_trial = compute_J(a_trial)
                if J_trial < current_J - 1e-12:
                    a = a_trial
                    current_J = J_trial
                    improved = True
                    break
        if not improved:
            break
        if epoch % 10 == 0:
            tracker.update(a)

    tracker.update(a)
    return a


def propose_candidate(seed=42, budget_s=1500, **kwargs):
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + float(budget_s) - 15.0
    tracker = Tracker()

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        if p.ndim == 1 and p.size > 0 and np.all(np.isfinite(p)):
            parent = np.maximum(p, 0.0)
            tracker.update(parent)
            print(f"Parent: n={parent.size}, J={tracker.best_score:.8f}", flush=True)
    except (NameError, AttributeError):
        pass

    # === Phase 1: Multi-length exploration ===
    lengths = [256, 384, 512, 768]
    n_inits_per = 4
    phase1_budget = 400.0
    per_init_time = phase1_budget / (len(lengths) * n_inits_per)

    best_overall = None
    best_overall_J = float('inf')

    for n in lengths:
        if time.monotonic() > deadline - 300:
            break
        print(f"\n{'='*40} Phase1 n={n} {'='*40}", flush=True)
        t0 = time.monotonic()
        inits = generate_inits(n, rng, parent)
        inits = inits[:n_inits_per]

        best_n = None
        best_n_J = float('inf')

        for idx, a0 in enumerate(inits):
            if time.monotonic() > deadline - 300:
                break
            t_alloc = min(60.0, deadline - 300 - time.monotonic())
            if t_alloc < 10:
                break
            a = optimize_sequence(a0, time.monotonic() + t_alloc, tracker,
                                  max_iters=200, tighten_start=0.08, tighten_end=0.005,
                                  min_iter_gap=100)
            J = compute_J(a)
            if J < best_n_J:
                best_n_J = J
                best_n = a.copy()
            if J < best_overall_J:
                best_overall_J = J
                best_overall = a.copy()

        if best_n is not None:
            tracker.report(f"Phase1 n={n}: {best_n_J:.8f} ({time.monotonic()-t0:.1f}s)")

    # === Phase 2: Focused optimization on best found ===
    if best_overall is not None and time.monotonic() < deadline - 100:
        print(f"\n{'='*40} Phase2 Focus {'='*40}", flush=True)
        t0 = time.monotonic()
        n = best_overall.size

        # Adaptive tightening: start aggressive, end gentle
        tighten_schedule = [0.05, 0.03, 0.02, 0.01, 0.005, 0.002]
        a = best_overall.copy()
        for tight in tighten_schedule:
            if time.monotonic() > deadline - 100:
                break
            remaining = deadline - 100 - time.monotonic()
            n_iters = int(remaining / 0.5)
            n_iters = max(50, min(n_iters, 3000))
            a = optimize_sequence(a, time.monotonic() + remaining * 0.8, tracker,
                                  max_iters=n_iters, tighten_start=tight,
                                  tighten_end=max(0.0005, tight * 0.1),
                                  min_iter_gap=300)

        tracker.report(f"Phase2: {compute_J(a):.8f} ({time.monotonic()-t0:.1f}s)")

    # === Phase 3: Multigrid upscale ===
    if tracker.best is not None and time.monotonic() < deadline - 80:
        print(f"\n{'='*40} Phase3 Multigrid {'='*40}", flush=True)
        t0 = time.monotonic()
        a_cur = tracker.best.copy()
        n_cur = a_cur.size

        for target_n in [n_cur * 2, n_cur * 4]:
            if target_n > 2048 or time.monotonic() > deadline - 60:
                break
            a_up = upsample(a_cur, target_n)
            a_up /= a_up.sum()
            J_up = compute_J(a_up)
            print(f"  Upsampled to n={target_n}: J={J_up:.8f}", flush=True)

            remaining = deadline - 60 - time.monotonic()
            if remaining < 30:
                break
            a_up = optimize_sequence(a_up, time.monotonic() + remaining * 0.7, tracker,
                                     max_iters=int(remaining * 0.7 / 1.0),
                                     tighten_start=0.03, tighten_end=0.001,
                                     min_iter_gap=200)
            J_final = compute_J(a_up)
            print(f"  Optimized n={target_n}: J={J_final:.8f}", flush=True)
            if J_final < tracker.best_score:
                a_cur = a_up
            else:
                break

        tracker.report(f"Phase3: ({time.monotonic()-t0:.1f}s)")

    # === Phase 4: Local polish ===
    if tracker.best is not None and time.monotonic() < deadline - 20:
        print(f"\n{'='*40} Phase4 Polish {'='*40}", flush=True)
        t0 = time.monotonic()
        a = tracker.best.copy()

        # Local random search
        a = local_polish(a, min(deadline - 10, time.monotonic() + 80), tracker)

        # Short LP-based refinement
        if time.monotonic() < deadline - 10:
            a = optimize_sequence(a, deadline - 5, tracker, max_iters=500,
                                  tighten_start=0.005, tighten_end=0.0005,
                                  min_iter_gap=500)

        # Coordinate polish
        if time.monotonic() < deadline - 5:
            a = coordinate_polish(a, deadline - 2, tracker)

        tracker.report(f"Phase4: ({time.monotonic()-t0:.1f}s)")

    # Final
    if tracker.best is None:
        tracker.best = np.ones(256)
        tracker.best_score = compute_J(tracker.best)
        tracker.update(tracker.best)

    print(f"\n{'='*40} FINAL {'='*40}", flush=True)
    print(f"J={tracker.best_score:.8f} n={tracker.best.size} time={time.monotonic()-tracker.t0:.1f}s", flush=True)
    return [float(x) for x in tracker.best]
