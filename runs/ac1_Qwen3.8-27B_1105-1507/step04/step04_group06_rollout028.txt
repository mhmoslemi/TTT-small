```python
Minimizes J(a) = 2*n*max(conv(a,a))/(sum(a))^2 using exact coordinate descent.

Includes multi-length exploration, periodic perturbations, and L-BFGS warm-start.

import numpy as np
from scipy.optimize import minimize
import time

def fast_conv(a):
    n = len(a)
    s = 1
    while s < 2 * n - 1:
        s <<= 1
    fa = np.fft.rfft(a, s)
    return np.fft.irfft(fa * fa, s)[:2 * n - 1]

def compute_J(a):
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))
```

```python
"""
Minimizes J(a) = 2*n*max(conv(a,a))/(sum(a))^2 using exact coordinate descent
with golden-section search, combined with L-BFGS warm-start, multi-length
exploration, and adaptive perturbations.

Key innovation: O(n) per-evaluation coordinate descent exploiting the structure
that changing one coordinate only affects n convolution values linearly (plus
one quadratic term), enabling fast exact 1D minimization.
"""

import numpy as np
from scipy.optimize import minimize
import time

_GR = (np.sqrt(5.0) - 1.0) / 2.0


def fast_conv(a):
    """Full self-convolution via FFT."""
    n = len(a)
    s = 1
    while s < 2 * n - 1:
        s <<= 1
    fa = np.fft.rfft(a, s)
    return np.fft.irfft(fa * fa, s)[:2 * n - 1]


def compute_J(a):
    """Compute J(a) = 2*n*max(conv(a,a))/(sum(a))^2."""
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))


def smooth_F_and_grad(a, beta):
    """Smooth approximation of J(a) using log-sum-exp. Returns (value, gradient)."""
    n = len(a)
    b = fast_conv(a)
    b_max = float(b.max())
    shifted = beta * (b - b_max)
    shifted = np.clip(shifted, -500, 500)
    exp_shifted = np.exp(shifted)
    Z = float(exp_shifted.sum())
    if Z < 1e-300:
        p = np.zeros(2 * n - 1)
        p[np.argmax(b)] = 1.0
    else:
        p = exp_shifted / Z
    g_val = (b_max + np.log(max(Z, 1e-300))) / beta

    # Gradient of g_val = 2 * convolve(a, p) (by symmetry of convolution)
    # d/d a_j [max_k (a*a)_k] approx 2 * (a * p)_j
    S = np.convolve(a, p[::-1], mode='full')
    S = S[n - 1:2 * n - 1]
    grad_g = 2.0 * S

    s = float(a.sum())
    F_val = 2.0 * n * g_val / (s * s)
    grad_F = 2.0 * n * (s * grad_g - 2.0 * g_val * np.ones(n)) / (s ** 3)
    return F_val, grad_F


def lbfgs_optimize(a0, beta_schedule, deadline):
    """L-BFGS-B with beta annealing for fast basin finding."""
    a = np.maximum(a0.astype(np.float64), 1e-8)
    a = a / a.sum()

    for beta, max_iters in beta_schedule:
        if time.monotonic() > deadline:
            break

        def func(x):
            x2 = np.maximum(x, 0.0)
            f, _ = smooth_F_and_grad(x2, beta)
            return f if np.isfinite(f) else 1e10

        def jac(x):
            x2 = np.maximum(x, 0.0)
            _, g = smooth_F_and_grad(x2, beta)
            g = np.nan_to_num(g, nan=0.0, posinf=0.0, neginf=0.0)
            g[x2 < 1e-12] = 0.0
            return g

        try:
            res = minimize(func, a.copy(), jac=jac, method='L-BFGS-B',
                           bounds=[(0.0, None)] * len(a),
                           options={'maxiter': max_iters, 'max_fun': max_iters * 30,
                                    'gtol': 1e-9, 'ftol': 1e-15})
            if res.fun < 1e10:
                new_a = np.maximum(res.x, 0.0)
                if new_a.sum() > 1e-6:
                    a = new_a / new_a.sum()
        except Exception:
            pass
    return a


def coordinate_descent(a, max_passes, num_evals, deadline, direction=1):
    """
    Run coordinate descent for up to max_passes full sweeps.
    Returns (a, current_J).
    """
    n = len(a)
    a = a.astype(np.float64).copy()
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-10:
        return a, float('inf')
    a = a / s  # normalize to sum = 1

    full_conv = fast_conv(a)
    current_J = 2.0 * n * float(full_conv.max())  # since sum=1, J = 2n*max(conv)

    base_aff = np.empty(n, dtype=np.float64)
    two_a = np.empty(n, dtype=np.float64)

    indices = list(range(n))
    if direction == -1:
        indices = indices[::-1]

    prev_J = current_J

    for p in range(max_passes):
        if time.monotonic() > deadline:
            break

        S = a.sum()
        pass_improved = False

        for i in indices:
            if time.monotonic() > deadline:
                break

            old_ai = a[i]
            S_other = S - old_ai

            # Max of unaffected convolution values
            max_unaff = 0.0
            if i > 0:
                max_unaff = max(max_unaff, float(np.max(full_conv[:i])))
            if i + n < 2 * n - 1:
                max_unaff = max(max_unaff, float(np.max(full_conv[i + n:])))

            # Base for affected indices: full_conv[i:i+n] minus a[i]'s contribution
            np.copyto(base_aff, full_conv[i:i + n])
            np.multiply(a, 2.0 * old_ai, out=two_a)
            base_aff -= two_a
            # Correct quadratic term at position i (k=2i):
            # We subtracted 2*old_ai*a[i] = 2*old_ai^2, but actual contribution is old_ai^2
            base_aff[i] += old_ai * old_ai

            # Precompute 2*a for evaluation
            np.multiply(a, 2.0, out=two_a)
            ai_val = a[i]

            # Local function for evaluation
            def _eval_x(x, ba=base_aff, ta=two_a, mx=max_unaff, so=S_other,
                        nn=n, pv=ai_val):
                aff = ba + x * ta
                aff[i] += x * x - 2.0 * x * pv
                max_aff = np.max(aff)
                max_conv = max(mx, max_aff)
                denom = so + x
                if denom < 1e-15:
                    return 1e10
                return 2.0 * nn * max_conv / (denom * denom)

            # Golden section search on [0, S]
            lo = 0.0
            hi = S
            t1 = hi - _GR * (hi - lo)
            t2 = lo + _GR * (hi - lo)
            J1 = _eval_x(t1)
            J2 = _eval_x(t2)

            best_x = old_ai
            best_Jx = _eval_x(old_ai)

            for _ in range(num_evals):
                if J1 < J2:
                    hi = t2
                    t2 = t1
                    J2 = J1
                    t1 = hi - _GR * (hi - lo)
                    J1 = _eval_x(t1)
                else:
                    lo = t1
                    t1 = t2
                    J1 = J2
                    t2 = lo + _GR * (hi - lo)
                    J2 = _eval_x(t2)

                # Track best at mid
                mid = (lo + hi) * 0.5
                Jm = _eval_x(mid)
                if Jm < best_Jx:
                    best_Jx = Jm
                    best_x = mid

            # Check boundaries
            J0 = _eval_x(0.0)
            if J0 < best_Jx:
                best_x = 0.0
                best_Jx = J0
            Jhi = _eval_x(hi)
            if Jhi < best_Jx:
                best_x = hi
                best_Jx = Jhi

            # Update if improved
            if best_Jx < current_J - 1e-14 * current_J:
                delta = best_x - old_ai
                a[i] = best_x
                # Update full_conv incrementally
                full_conv[i:i + n] += 2.0 * delta * a
                full_conv[2 * i] -= delta * delta
                S += delta
                current_J = best_Jx
                pass_improved = True

        # Renormalize at end of pass
        S = a.sum()
        if S > 1e-10:
            a = a / S
            full_conv = fast_conv(a)
            current_J = 2.0 * n * float(full_conv.max())

        if not pass_improved:
            # No improvement this pass - check if truly converged
            if abs(current_J - prev_J) < 1e-12 * prev_J:
                break
        prev_J = current_J

    return a, current_J


def resample_seq(src, dest_n):
    """Resample 1D sequence to new length via linear interpolation."""
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    if src.size == 0:
        return np.ones(dest_n)
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    result = np.interp(t_new, t_old, src)
    return np.maximum(result, 0.0)


def generate_inits(n, rng, parent):

    inits = []
    t_arr = np.linspace(0, 1, n, endpoint=False)

    if parent is not None and parent.size > 0:

        p = resample_seq(parent, n)
        p = p + rng.normal(0, 0.05, n) * p
        inits.append(np.maximum(p, 0.01))
        p2 = resample_seq(parent, n)
        p2 = p2 + rng.normal(0, 0.12, n) * p2
        inits.append(np.maximum(p2, 0.01))

    for freq in [2, 3, 4, 5, 7, 11]:
        inits.append(1.0 + 0.35 * np.cos(2.0 * np.pi * freq * t_arr))

    inits.append(np.exp(-4.0 * (t_arr - 0.5) ** 2) + 0.05)
    inits.append(np.abs(np.sin(np.pi * t_arr)) + 0.15)
    inits.append(1.0 + 0.3 * np.cos(4.0 * np.pi * t_arr) + 0.1 * np.cos(8.0 * np.pi * t_arr))
    inits.append((t_arr + 0.1) ** 0.5)
    inits.append(np.exp(-1.5 * t_arr) + 0.2)
    inits.append(1.0 - 0.3 * np.abs(2.0 * (t_arr % 0.6) - 0.5))
    # Periodic with irrational frequency
    inits.append(1.0 + 0.25 * np.sin(2.0 * np.pi * 3.7 * t_arr) + 0.15 * np.sin(2.0 * np.pi * 6.1 * t_arr))
    # Bump + oscillation
    inits.append(np.exp(-3.0 * (t_arr - 0.5) ** 2) * (1.0 + 0.3 * np.cos(6.0 * np.pi * t_arr)) + 0.1)

    return inits

            a_work, _ = coordinate_descent(a_work, max_p2, 40, deadline, direction=-1)
            tracker.update(a_work)

        # Try a few nearby lengths
        if time.monotonic() < deadline:
            n_best = tracker.best.size
            for n_try in [n_best - 32, n_best + 32, n_best - 64, n_best + 64]:
                if n_try < 128 or n_try > 4096 or time.monotonic() > deadline:
                    continue
                a_res = resample_seq(tracker.best, n_try)
                a_res = a_res / max(a_res.sum(), 1e-10)
                max_p = int(min(30.0, deadline - time.monotonic()) * 12)
                a_res, _ = coordinate_descent(a_res, max_p, 35, deadline, direction=1)
                tracker.update(a_res)

        tracker.report(f"Phase 4: {time.monotonic()-t0:.1f}s")

    # Fallback
    if tracker.best is None:
        tracker.best = np.ones(512)
        tracker.best_score = compute_J(tracker.best)
        print(f"  Fallback: J={tracker.best_score:.8f}", flush=True)

    print(f"\n{'='*50} FINAL RESULT {'='*50}", flush=True)
    print(f"J={tracker.best_score:.8f} n={tracker.best.size} "
          f"time={time.monotonic()-tracker.t_start:.1f}s", flush=True)

    result = np.clip(tracker.best, 0.0, 1000.0)
    if result.sum() < 0.01:
        result = np.ones(128)

    return [float(x) for x in result]
```
