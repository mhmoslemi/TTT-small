```python
def proj_simplex(v):
    """Project v onto {x >= 0, sum(x) = 1}."""
    n = len(v)
    # Sort in decreasing order
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.where(u * np.arange(1, n+1) > (cssv - 1))[0][-1]
    tau = (cssv[rho] - 1) / (rho + 1)
    return np.maximum(v - tau, 0)
```

This is O(n log n) due to the sort.

    a = np.maximum(a, 1e-10)
    a = a / a.sum()
    best_a = a.copy()
    best_M = fast_conv(a).max()
    
    y = a.copy()  # momentum variable
    t_k = 1.0
    
    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        
        # Compute gradient at y
        b = fast_conv(y)
        M = b.max()
        # Active set (within tolerance)
        tol = 1e-10 * M
        active = b >= M - tol
        
        # Subgradient
        n = len(a)
        grad = np.zeros(n)
        for k in np.where(active)[0]:
            lo = max(0, k - n + 1)
            hi = min(n - 1, k)
            for i in range(lo, hi + 1):
                grad[i] += 2.0 * y[k - i]

        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-15:

        beta = (t_k - 1) / t_new

        y_new = proj_simplex(y_new)

            best_a = a_new.copy()
        
        a = a_new
        y = y_new
        t_k = t_new
