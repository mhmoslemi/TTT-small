    """Solve the LP exploiting symmetry. Returns (optimal_g, lp_value)."""
    n = len(a)
    # Build the lower-triangular matrix A where A[k,j] = a[k-j] for k>=j>=0
    # Constraint: sum_j A[k,j] * g[j] <= M for k=0..n-1
    # Maximize: sum_j g[j]
    
    from scipy.sparse import csr_matrix
    rows = []
    cols = []
    vals = []
    for k in range(n):
        for j in range(k + 1):  # j from 0 to k (lower triangular)
            rows.append(k)
            cols.append(j)
            vals.append(a[k - j])
    
    A = csr_matrix((vals, (rows, cols)), shape=(n, n))

    bounds = [(0, None)] * n

                     options={'presolve': True})

        g = np.maximum(result.x, 0)
        return g, -result.fun  # lp_value = sum(g)
    return None, None

import numpy as np
from scipy.optimize import linprog
from scipy.sparse import csr_matrix
import time


def fast_conv(a):
    """Full self-convolution via FFT."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]


def compute_J(a):
    """Compute J(a) = 2*n*max(conv(a,a))/(sum(a))^2."""
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))


class ConvLPSolver:
    """Pre-allocated LP solver for the convolution best-response problem."""
    def __init__(self, n):
        self.n = n
        n_rows = 2 * n - 1
        # Build the sparse structure: A[k,j] = a[k-j] for valid (k,j)
        rows_list = []
        cols_list = []
        k_list = []
        j_list = []
        for k in range(n_rows):
            j_min = max(0, k - n + 1)
            j_max = min(n - 1, k)
            for j in range(j_min, j_max + 1):
                rows_list.append(k)
                cols_list.append(j)
                k_list.append(k)
                j_list.append(j)

        self.col_idx = np.array(cols_list, dtype=np.int32)

        self.bounds = [(0.0, None)] * n
        self.data = np.empty(self.nnz, dtype=np.float64)

    n = len(a)
    b = fast_conv(a)
    b_max = b.max()
    exp_b = np.exp(tau * (b - b_max))
    p = exp_b / exp_b.sum()
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        pk = p[k]
        if pk < 1e-14:
            continue
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        grad[lo:hi + 1] += 2.0 * pk * a[k - hi:k - lo + 1][::-1]
    return grad, b


def run_gradient_polish(a, deadline, max_iters=500):
    """Quick smooth gradient descent to polish the initial point."""
    a = np.maximum(a.astype(np.float64), 1e-10)
    a = a / a.sum()
    n = len(a)
    best_a = a.copy()
    best_M = fast_conv(a).max()
    v = np.zeros(n)

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        frac = it / max(1, max_iters)
        tau = 5.0 + 60.0 * frac
        lr = 0.15 * (0.1 / 0.15) ** frac
        grad, b = smooth_gradient_step(a, tau)
        M = b.max()
        if M < best_M * (1 - 1e-9):
            best_M = M
            best_a = a.copy()
        v = 0.85 * v + lr * grad
        a_new = np.maximum(a - v, 0.0)
        s = a_new.sum()
        if s < 1e-10:
            a_new = a.copy()
        else:
            a_new = a_new / s
        a = a_new

    return best_a


def lp_mirror_descent(a, deadline, C=1.0, t_max=0.4, perturb_interval=2000,
                      noise_scale=0.005, verbose=True):
    """
    Aggressive LP mirror descent with:
    - Pre-allocated LP solver
    - Adaptive step size (C/sqrt(k)) with backtracking
    - Periodic perturbation to escape plateaus
    - Momentum-averaged best responses
    """
    a = np.maximum(a.astype(np.float64), 1e-12)
    a = a / a.sum()
    n = len(a)
    solver = ConvLPSolver(n)

    best_a = a.copy()
    best_M = fast_conv(a).max()
    best_score = 2.0 * n * best_M

    # Momentum buffer for averaged best responses
    g_avg = a.copy()
    g_avg_weight = 0.0
    avg_window = 50  # exponential moving average window

    no_improve_count = 0
    gap_history = []

    for it in range(1000000):
        if time.monotonic() > deadline:
            break

        # Compute current max
        b = fast_conv(a)
        M = float(b.max())

        # Periodic perturbation to escape plateaus
        if it > 0 and it % perturb_interval == 0 and no_improve_count > 100:
            a = a + np.random.default_rng(it).normal(0, noise_scale, n)
            a = np.maximum(a, 1e-8)
            a = a / a.sum()
            noise_scale *= 0.8
            no_improve_count = 0
            if verbose:
                print(f"    [LP-iter {it}] Perturb! noise={noise_scale:.5f}", flush=True)
            continue

        # Solve LP for best response
        g, lp_val = solver.solve(a, M)
        if g is None:
            # Fallback: use reciprocal heuristic
            g = 1.0 / (a + 0.01)
            g = np.maximum(g, 0)
            g = g / g.sum()
            lp_val = g.sum()
            a_new = (1 - 0.01) * a + 0.01 * g
            a = np.maximum(a_new, 0) / np.maximum(a_new, 0).sum()
            continue

        # Normalize g to sum 1
        g_sum = g.sum()
        if g_sum < 1e-15:
            break
        g_norm = g / g_sum

        # Update momentum buffer
        alpha = 1.0 / avg_window
        g_avg = (1 - alpha) * g_avg + alpha * g_norm
        g_avg_weight += 1

        # Choose step size
        t = min(t_max, C / np.sqrt(it + 1))
        # Also try the averaged direction
        g_use = 0.7 * g_norm + 0.3 * (g_avg / g_avg.sum()) if g_avg.sum() > 1e-10 else g_norm

        # Line search: try several step sizes
        best_t = t
        best_trial_M = M
        for t_trial in [t, t * 0.5, t * 2.0, t * 0.25, 0.05, 0.1, 0.2]:
            t_trial = min(t_max, max(0.001, t_trial))
            a_trial = (1 - t_trial) * a + t_trial * g_use
            a_trial = np.maximum(a_trial, 0.0)
            s = a_trial.sum()
            if s < 1e-10:
                continue
            a_trial = a_trial / s
            M_trial = fast_conv(a_trial).max()
            if M_trial < best_trial_M:
                best_trial_M = M_trial
                best_t = t_trial

        # Update
        a_new = (1 - best_t) * a + best_t * g_use
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s < 1e-10:
            break
        a = a_new / s

        # Check improvement
        M_new = fast_conv(a).max()
        if M_new < best_M * (1 - 1e-10):
            best_M = M_new
            best_a = a.copy()
            best_score = 2.0 * n * best_M
            no_improve_count = 0
            if verbose and (it % 1000 == 0 or it < 20):
                gap = lp_val - 1.0
                print(f"    [LP-iter {it}] M={best_M:.10f} J={best_score:.8f} "
                      f"gap={gap:.6f} t={best_t:.4f}", flush=True)
        else:
            no_improve_count += 1

        # Track gap for diagnostics
        if it < 50 or it % 5000 == 0:
            gap_history.append(lp_val - 1.0)

        # Convergence check: if gap is very small and no improvement for a while
        if len(gap_history) > 10:
            recent_gap = gap_history[-1]
            if recent_gap < 1e-8 and no_improve_count > 500:
                if verbose:
                    print(f"    [LP-iter {it}] Converged (gap={recent_gap:.2e})", flush=True)
                break

    return best_a, best_score


def resample_seq(src, dest_n):
    """Resample 1D sequence to new length via linear interpolation."""
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)


def propose_candidate(seed=42, budget_s=1500, **kwargs):
    """
    Multi-phase search optimized for LP mirror descent:
    Phase 1: Quick gradient polish (20s)
    Phase 2: Aggressive LP mirror descent at parent length (1100s)
    Phase 3: Multi-length exploration - upsample and optimize (280s)
    Phase 4: Final polish (80s)
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + float(budget_s) - 10.0
    t_start = time.monotonic()

    # Track best
    best_global = None
    best_global_score = float('inf')

    def update_best(a, label=""):
        nonlocal best_global, best_global_score
        if a is None or a.size == 0:
            return
        a_clean = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
        if a_clean.sum() < 0.01:
            return
        try:
            score = float(evaluate_sequence(a_clean.tolist()))
        except Exception:
            return
        if np.isfinite(score) and score < best_global_score:
            best_global = a_clean.copy()
            best_global_score = score
            elapsed = time.monotonic() - t_start
            print(f"  *** NEW BEST: J={score:.8f} n={a_clean.size} "
                  f"t={elapsed:.0f}s {label}", flush=True)

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        if p.ndim == 1 and p.size > 0 and np.all(np.isfinite(p)):
            parent = np.maximum(p, 0.0)
            update_best(parent, "parent")
            print(f"Parent loaded: n={parent.size}, J={best_global_score:.8f}", flush=True)
    except (NameError, AttributeError):
        pass

    if parent is None:
        parent = np.ones(512)
        update_best(parent, "default")

    n_parent = parent.size
    print(f"\n{'='*60}", flush=True)
    print(f"Starting: n={n_parent}, J={best_global_score:.8f}", flush=True)
    print(f"Budget: {budget_s}s", flush=True)
    print(f"{'='*60}\n", flush=True)

    # === Phase 1: Quick gradient polish ===
    p1_deadline = time.monotonic() + 20.0
    print("Phase 1: Gradient Polish", flush=True)
    polished = run_gradient_polish(parent, p1_deadline, max_iters=300)
    update_best(polished, "phase1")
    work_a = polished.copy()

    # === Phase 2: Main LP mirror descent ===
    p2_deadline = time.monotonic() + (budget_s * 0.72)
    print(f"\nPhase 2: LP Mirror Descent (n={n_parent}) "
          f"(~{p2_deadline - time.monotonic():.0f}s)", flush=True)

    a_opt, score_opt = lp_mirror_descent(work_a, p2_deadline, C=1.0, t_max=0.3,
                                         perturb_interval=3000, noise_scale=0.008,
                                         verbose=True)
    update_best(a_opt, "phase2")
    work_a = a_opt.copy()

    # === Phase 3: Multi-length exploration ===
    p3_deadline = time.monotonic() + (budget_s * 0.88)
    print(f"\nPhase 3: Multi-Length Exploration "
          f"(~{p3_deadline - time.monotonic():.0f}s)", flush=True)

    target_lengths = []
    for mult in [1.5, 2, 2.5]:
        nl = int(n_parent * mult)
        nl = (nl // 64) * 64  # Round to multiple of 64
        if nl != n_parent and nl <= 2048:
            target_lengths.append(nl)

    # Also try slightly smaller
    if n_parent > 256:
        target_lengths.insert(0, n_parent // 2)

    # Remove duplicates, sort
    target_lengths = sorted(set(target_lengths))

    per_length_time = max(30, (p3_deadline - time.monotonic()) / max(1, len(target_lengths)))

    for nl in target_lengths:
        if time.monotonic() > p3_deadline:
            break
        print(f"\n  Trying n={nl} (time budget: {per_length_time:.0f}s)", flush=True)

        # Upsample/downsample best solution
        a_init = resample_seq(work_a, nl)
        a_init = np.maximum(a_init, 1e-8)
        a_init = a_init / a_init.sum()

        # Quick gradient polish at new length
        p_sub = min(time.monotonic() + per_length_time * 0.15, p3_deadline)
        a_init = run_gradient_polish(a_init, p_sub, max_iters=100)

        # LP mirror descent at new length
        p_sub2 = min(time.monotonic() + per_length_time * 0.85, p3_deadline)
        a_new, s_new = lp_mirror_descent(a_init, p_sub2, C=0.8, t_max=0.25,
                                         perturb_interval=2000, noise_scale=0.006,
                                         verbose=False)
        update_best(a_new, f"n={nl}")

        # Also try from scratch with different init
        if time.monotonic() < p3_deadline - 30:
            a_rand = np.ones(nl) + rng.normal(0, 0.1, nl)
            a_rand = np.maximum(a_rand, 0.01)
            a_rand = a_rand / a_rand.sum()
            a_rand = run_gradient_polish(a_rand, min(time.monotonic() + 10, p3_deadline - 20),
                                         max_iters=50)
            a_new2, s_new2 = lp_mirror_descent(a_rand, p3_deadline - 20, C=0.8,
                                               t_max=0.25, perturb_interval=2000,
                                               noise_scale=0.01, verbose=False)
            update_best(a_new2, f"n={nl}_rand")

    # === Phase 4: Final polish on best found ===
    print(f"\nPhase 4: Final Polish (~{deadline - time.monotonic():.0f}s)", flush=True)
    if best_global is not None and time.monotonic() < deadline - 5:
        a_final = lp_mirror_descent(best_global, deadline - 2, C=0.5, t_max=0.15,
                                    perturb_interval=5000, noise_scale=0.003,
                                    verbose=True)
        update_best(a_final, "final")

    # Fallback
    if best_global is None:
        best_global = np.ones(128)
        best_global_score = compute_J(best_global)

    print(f"\n{'='*60}", flush=True)
    print(f"FINAL: J={best_global_score:.8f} n={best_global.size} "
          f"time={time.monotonic()-t_start:.0f}s", flush=True)
    print(f"{'='*60}", flush=True)

    return [float(x) for x in best_global]
