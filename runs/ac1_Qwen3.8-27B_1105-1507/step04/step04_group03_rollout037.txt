Soft Mirror Descent optimizer for minimizing J(a) = 2*n*max(conv(a,a))/sum(a)^2.
Uses softmax-smoothed gradient with multiplicative (mirror) updates,
adaptive step size, and L-BFGS-B polishing.

import numpy as np
from scipy.optimize import minimize
import time

def fast_conv(a):
    """Full self-convolution via FFT."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]

def compute_g(a, w):
    """Compute g[m] = 2 * sum_j a[j] * w[m+j] via FFT."""
    n = len(a)
    # w has length 2n-1
    N = 1
    while N < 3 * n - 1:
        N <<= 1
    W = np.fft.rfft(w, N)
    A = np.fft.rfft(a, N)
    g_full = np.fft.irfft(W * np.conj(A), N)
    return 2.0 * g_full[:n]

def compute_J(a):
    """Compute J(a) = 2*n*max(conv(a,a))/(sum(a))^2."""
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))

def softmax(x, tau):
    """Numerically stable softmax with temperature."""
    x_shifted = (x - x.max()) / tau
    e = np.exp(x_shifted)
    return e / e.sum()

Uses softmax-smoothed gradient with multiplicative mirror updates, adaptive step size,
multi-init strategy, length adaptation, and L-BFGS-B final polishing.
Key insight: for non-negative a, the subgradient of max_k conv(a,a)_k is a shifted copy of a,

import numpy as np
from scipy.optimize import minimize
import time


def fast_conv(a):
    """Full self-convolution via FFT."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]


def compute_g_fft(a, w):
    """Compute g[m] = 2 * sum_j a[j] * w[m+j] via FFT. w has length 2n-1, a has length n."""
    n = len(a)
    N = 1
    while N < 3 * n - 1:
        N <<= 1
    W = np.fft.rfft(w, N)
    A = np.fft.rfft(a, N)
    g_full = np.fft.irfft(W * np.conj(A), N)
    return 2.0 * g_full[:n]


def compute_J(a):
    """Compute J(a) = 2*n*max(conv(a,a))/(sum(a))^2."""
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))


def stable_softmax(x, tau):
    """Numerically stable softmax with temperature."""
    x_shifted = (x - x.max()) / tau
    e = np.exp(x_shifted)
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(x) / len(x)
    return e / s

        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()

def lbb_smoothed(a_init, tau, max_iter, deadline):
    """L-BFGS-B on the smoothed objective (log-sum-exp of convolution)."""
    n = len(a_init)
    a0 = a_init.astype(np.float64).copy()

    def obj(x):
        ax = np.maximum(x, 0.0)
        s = ax.sum()
        if s < 1e-8:
            return 1e10
        ax = ax / s
        b = fast_conv(ax)
        b_max = b.max()
        lse = b_max + tau * np.log(np.sum(np.exp((b - b_max) / tau)))
        return float(2 * n * lse)

    try:
        res = minimize(obj, a0, method='L-BFGS-B',
                       bounds=[(0.0, None)] * n,
                       options={'maxiter': max_iter, 'max_fun': 50000,
                                'ftol': 1e-20, 'gtol': 1e-15,
                                'ftol_cb': 1e-10})
        if res.fun < 1e10:
            ax = np.maximum(res.x, 0.0)
            s = ax.sum()
            if s > 1e-8:
                ax = ax / s
                return ax
    except Exception:
        pass
    return None


def resample_seq(src, dest_n):
    """Resample 1D sequence to new length."""
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)


def generate_inits(n, rng, parent=None):

    inits = []
    t = np.arange(n, dtype=np.float64) / n

    if parent is not None:

        inits.append(p.copy())
        inits.append(np.maximum(p + rng.normal(0, 0.01, n), 0.001))
        inits.append(np.maximum(p + rng.normal(0, 0.03, n), 0.001))

    # Structured inits
    inits.append(np.ones(n))
    inits.append(0.5 * (1 + np.cos(2 * np.pi * t)))
    inits.append(0.5 * (1 - np.cos(2 * np.pi * t)))
    inits.append(np.maximum(0.5 + 0.3 * np.cos(2 * np.pi * 2 * t), 0.01))
    inits.append(np.maximum(0.5 + 0.3 * np.sin(2 * np.pi * t), 0.01))
    inits.append(np.exp(-3 * (t - 0.5) ** 2) + 0.05)
    inits.append(np.exp(-6 * (t - 0.3) ** 2) + np.exp(-6 * (t - 0.7) ** 2) + 0.03)
    inits.append(np.maximum(0.5 + 0.2 * np.cos(2 * np.pi * 3 * t) + 0.1 * np.cos(2 * np.pi * 5 * t), 0.01))

    # Smooth random
    k = max(5, n // 20)
    kernel = np.ones(k) / k
    smooth = np.convolve(rng.uniform(0, 1, n), kernel, mode='same')
    inits.append(np.maximum(smooth + 0.1, 0.01))

    result = []
    for a in inits:
        a = np.maximum(a, 0.001)
        s = a.sum()
        if s > 1e-6:
            result.append(a / s)
    return result

        print(f"{'='*60}", flush=True)

        p3_dl = min(deadline - 250, time.monotonic() + 200)
        a3, J3 = smooth_mirror_descent(current_best, p3_dl, max_iters=50000,
                                       verbose=True, label="SM3")
        tracker.update(a3)
        print(f"  Phase 3 done: J={J3:.8f} ({time.monotonic()-t0:.1f}s)", flush=True)

        if J3 < current_best_J:
            current_best = a3.copy()
            current_best_J = J3

    # === Phase 4: Multi-init exploration ===
    if time.monotonic() < deadline - 200:
        t0 = time.monotonic()
        print(f"\n{'='*60}", flush=True)
        print(f"Phase 4: Multi-init (n={base_n})", flush=True)
        print(f"{'='*60}", flush=True)

        inits = generate_inits(base_n, rng, parent=current_best)
        # Remove the first one (which is the parent itself, already optimized)
        inits = inits[1:]

        per_init_time = max(20, min(60, (deadline - 200 - time.monotonic()) / max(1, len(inits))))

        for i, a_init in enumerate(inits):
            if time.monotonic() > deadline - 180:
                break
            a_init = a_init / a_init.sum()
            J_init = compute_J(a_init)
            if J_init > current_best_J + 0.05:
                continue  # Skip if too far from best
            dl = time.monotonic() + per_init_time
            a_try, J_try = smooth_mirror_descent(a_init, dl, max_iters=10000,
                                                 verbose=False, label=f"in{i}")
            tracker.update(a_try)
            if J_try < current_best_J:
                current_best = a_try.copy()
                current_best_J = J_try
                print(f"  Init {i}: J={J_try:.8f} (improved!) ({time.monotonic()-t0:.1f}s)", flush=True)

        print(f"  Phase 4 done ({time.monotonic()-t0:.1f}s)", flush=True)

    # === Phase 5: Length adaptation ===
    if time.monotonic() < deadline - 150:
        t0 = time.monotonic()
        print(f"\n{'='*60}", flush=True)
        print(f"Phase 5: Length adaptation (base n={current_best.size})", flush=True)
        print(f"{'='*60}", flush=True)

        base_n_cur = current_best.size
        n_candidates = sorted(set([
            base_n_cur - 32, base_n_cur + 32,
            base_n_cur - 64, base_n_cur + 64,
            base_n_cur + 128,
        ]))
        n_candidates = [n for n in n_candidates if 128 <= n <= 2048 and n != base_n_cur]

        per_n_time = max(30, min(80, (deadline - 150 - time.monotonic()) / max(1, len(n_candidates))))

        for n_try in n_candidates:
            if time.monotonic() > deadline - 120:
                break
            a_try = resample_seq(current_best, n_try)
            a_try = np.maximum(a_try, 1e-10)
            a_try = a_try / a_try.sum()
            dl = time.monotonic() + per_n_time
            a_try, J_try = smooth_mirror_descent(a_try, dl, max_iters=20000,
                                                 verbose=False, label=f"n{n_try}")
            tracker.update(a_try)
            print(f"  n={n_try}: J={J_try:.8f}", flush=True)

            if J_try < current_best_J:
                current_best = a_try.copy()
                current_best_J = J_try

        print(f"  Phase 5 done ({time.monotonic()-t0:.1f}s)", flush=True)

    # === Phase 6: Intensive soft mirror descent on best ===
    if time.monotonic() < deadline - 80:
        t0 = time.monotonic()
        print(f"\n{'='*60}", flush=True)
        print(f"Phase 6: Intensive refinement (n={current_best.size})", flush=True)
        print(f"{'='*60}", flush=True)

        p6_dl = min(deadline - 60, time.monotonic() + 150)
        a6, J6 = smooth_mirror_descent(current_best, p6_dl, max_iters=100000,
                                       verbose=True, label="INT")
        tracker.update(a6)
        print(f"  Phase 6 done: J={J6:.8f} ({time.monotonic()-t0:.1f}s)", flush=True)

        if J6 < current_best_J:
            current_best = a6.copy()
            current_best_J = J6

    # === Phase 7: L-BFGS-B polishing ===
    if time.monotonic() < deadline - 10 and tracker.best is not None:
        t0 = time.monotonic()
        print(f"\n{'='*60}", flush=True)
        print(f"Phase 7: L-BFGS-B polishing (n={tracker.best.size})", flush=True)
        print(f"{'='*60}", flush=True)

        # Try different temperatures
        for tau_frac in [0.01, 0.005]:
            if time.monotonic() > deadline - 10:
                break
            c = fast_conv(tracker.best / tracker.best.sum())
            tau = c.max() * tau_frac
            a_lbb = lbb_smoothed(tracker.best / tracker.best.sum(), tau,
                                 max_iter=200, deadline=deadline - 10)
            if a_lbb is not None:
                tracker.update(a_lbb)

        # Try on current_best too
        c = fast_conv(current_best / current_best.sum())
        tau = c.max() * 0.01
        a_lbb2 = lbb_smoothed(current_best / current_best.sum(), tau,
                              max_iter=200, deadline=deadline - 5)
        if a_lbb2 is not None:
            tracker.update(a_lbb2)

        print(f"  Phase 7 done ({time.monotonic()-t0:.1f}s)", flush=True)

    # === Phase 8: Final fast mirror descent squeeze ===
    if time.monotonic() < deadline - 3 and tracker.best is not None:
        t0 = time.monotonic()
        a_final, J_final = fast_mirror_descent(
            tracker.best / tracker.best.sum(),
            deadline - 1, max_iters=50000)
        tracker.update(a_final)
        print(f"  Final squeeze: J={J_final:.8f} ({time.monotonic()-t0:.1f}s)", flush=True)

    # Final decision
    if tracker.best is not None:
        final_a = tracker.best
    else:
        final_a = current_best

    print(f"\n{'='*60}", flush=True)
    print(f"FINAL: J={tracker.best_score:.8f} n={final_a.size} "
          f"time={time.monotonic()-tracker.t_start:.0f}s", flush=True)
    print(f"{'='*60}", flush=True)

    return [float(x) for x in final_a]
