def fft_conv(a, b):
    """Compute full convolution using FFT."""
    n = len(a)
    m = len(b)
    size = 1
    while size < n + m - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    conv = np.fft.irfft(fa * fb, size)
    return conv[:n + m - 1]

"""
Improved step function optimizer using soft-max gradient descent,
LP refinement, and multi-scale exploration.
Key improvements over previous:
- Direct soft-max gradient (no approximation)
- Adaptive tau for progressive sharpening
- Efficient FFT-based gradient computation
- Better LP refinement with warm starts
- Systematic exploration of n values
"""

import numpy as np
from scipy.optimize import linprog
import time


def fft_conv_full(a, b):
    """Compute full linear convolution using FFT."""
    n = len(a)
    m = len(b)
    size = 1
    while size < n + m - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    conv = np.fft.irfft(fa * fb, size)
    return conv[:n + m - 1]


def compute_obj(a):
    """Compute 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    sum_a = float(np.sum(a))
    if sum_a < 0.01:
        return np.inf
    conv = fft_conv_full(a, a)
    max_b = float(np.max(conv))
    return 2.0 * n * max_b / (sum_a * sum_a)


def softmax_obj_grad(a, tau):
    """
    Compute objective (soft-approximation of max conv) and gradient.
    f(a) = (1/tau) * log(sum_k exp(tau * conv(a,a)[k]))
    Returns (f, grad) where grad = df/da.
    """
    n = len(a)
    conv = fft_conv_full(a, a)
    max_conv = float(np.max(conv))
    exp_vals = np.exp(tau * (conv - max_conv))
    w = exp_vals / np.sum(exp_vals)
    
    # Objective (up to constant  max_conv which doesn't affect gradient)
    f = max_conv + np.log(np.sum(exp_vals)) / tau
    
    # Gradient: grad[j] = 2 * sum_{m=0}^{n-1} a[m] * w[m+j]
    # = 2 * corr[:n][::-1] where corr = np.correlate(w, a, 'full')
    corr = np.correlate(w, a, mode='full')
    grad = 2.0 * corr[:n][::-1].copy()
    
    return f, grad, w, conv


def softmax_gradient_descent(a, time_budget=60.0, tau_start=10.0, tau_end=200.0, verbose=False):
    """
    Optimize a using soft-max gradient descent with adaptive tau.
    Uses multiplicative updates to maintain non-negativity.
    """
    n = len(a)
    a = a.copy().astype(np.float64)
    s = np.sum(a)
    if s < 1e-10:
        return a
    a = a / s  # Normalize to sum=1
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    
    step = 0.5 / (tau_start * np.max(a) * n)  # Initial step size
    stall = 0
    patience = 100
    
    while time.monotonic() - start < time_budget:
        # Progressively increase tau
        elapsed_frac = (time.monotonic() - start) / time_budget
        tau = tau_start * (tau_end / tau_start) ** elapsed_frac
        
        f, grad, w, conv = softmax_obj_grad(a, tau)
        
        # Renormalize gradient (we're on the simplex, so subtract mean)
        grad_centered = grad - np.mean(grad)
        
        # Multiplicative update
        log_update = -step * grad_centered
        log_update = np.clip(log_update, -3, 3)
        a_new = a * np.exp(log_update)
        a_new = np.maximum(a_new, 1e-15)
        s_new = np.sum(a_new)
        if s_new > 1e-10:
            a_new = a_new / s_new
        
        obj = compute_obj(a_new)
        if obj < best_obj - 1e-15:
            best_obj = obj
            best_a = a_new.copy()
            a = a_new
            stall = 0
            # Slightly increase step if improving
            step = min(step * 1.01, 2.0 / (tau * np.max(a) * n))
        else:
            a = a_new  # Still move (helps escape flat regions)
            stall += 1
            if stall > patience:
                step *= 0.7
                if step < 1e-6:

                a = a * np.exp(rng.normal(0, 0.02, n))
                a = np.maximum(a, 1e-15)
                s = np.sum(a)
                a = a / s
                stall = 0

        if verbose and stall % 50 == 0:

    return best_a


def build_toeplitz(a):
    """Build matrix A where A @ g = conv(a, g) (full, length 2n-1)."""
    n = len(a)
    A = np.zeros((2 * n - 1, n), dtype=np.float64)
    for j in range(n):
        k_start = j
        k_end = min(j + n, 2 * n - 1)
        A[k_start:k_end, j] = a[:k_end - k_start]
    return A


def lp_maxsum(a, M, time_limit=10):
    """Solve LP: max sum(g) s.t. conv(a,g)[k] <= M, g >= 0."""
    n = len(a)
    A = build_toeplitz(a)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds,
                      method='highs', options={'time_limit': time_limit})
        if res.status == 0:
            g = np.maximum(res.x, 0)
            return g, float(np.sum(g))
    except Exception:
        pass
    return None, None


def lp_minpeak(a, time_limit=10):
    """Solve LP: min M s.t. conv(a,g)[k] <= M, sum(g)=1, g>=0."""
    n = len(a)
    A = build_toeplitz(a)
    c = np.zeros(n + 1)
    c[-1] = 1.0  # minimize M
    A_ub = np.zeros((2 * n - 1, n + 1))
    A_ub[:, :n] = A
    A_ub[:, n] = -1.0
    b_ub = np.zeros(2 * n - 1)
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])
    bounds = [(0.0, None)] * n + [(0.0, None)]
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs', options={'time_limit': time_limit})
        if res.status == 0:
            g = np.maximum(res.x[:n], 0)
            s = np.sum(g)
            if s < 1e-10:
                return None, None
            g = g / s
            return g, float(res.x[n])
    except Exception:
        pass
    return None, None


def lp_refine(a, time_budget=60.0, verbose=False):
    """Refine a using alternating LP method."""
    n = len(a)
    a = a.copy().astype(np.float64)
    s = np.sum(a)
    if s < 1e-10:
        return a
    a = a / s
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    stall = 0
    
    it = 0
    while time.monotonic() - start < time_budget - 5:
        it += 1
        conv = fft_conv_full(a, a)
        M = float(np.max(conv))
        
        # Solve LP: max sum(g) s.t. conv(a,g) <= M, g >= 0
        g, sum_g = lp_maxsum(a, M, time_limit=min(15, time_budget - (time.monotonic() - start)))
        if g is None or sum_g < 1e-10:
            break
        
        g_norm = g / sum_g
        
        # Line search: blend a and g_norm
        best_t = 0.0
        best_obj_ls = best_obj
        for t in [0.1, 0.2, 0.3, 0.5, 0.7, 1.0]:
            a_try = (1.0 - t) * a + t * g_norm
            a_try = np.maximum(a_try, 0)
            s_try = np.sum(a_try)
            if s_try < 1e-10:
                continue
            a_try = a_try / s_try
            obj_try = compute_obj(a_try)
            if obj_try < best_obj_ls:
                best_obj_ls = obj_try
                best_t = t
        
        if best_t > 0 and best_obj_ls < best_obj - 1e-14:
            a_try = (1.0 - best_t) * a + best_t * g_norm
            a_try = np.maximum(a_try, 0)
            s_try = np.sum(a_try)
            if s_try > 1e-10:
                a_try = a_try / s_try
            a = a_try
            best_obj = best_obj_ls
            best_a = a.copy()
            stall = 0
            if verbose:
                print(f"    LP iter {it}: obj={best_obj:.8f}, t={best_t}")
        else:
            stall += 1
            if stall > 15:
                break
            # Try smaller blend
            a = 0.95 * a + 0.05 * g_norm
            a = np.maximum(a, 0)
            s = np.sum(a)
            if s > 1e-10:
                a = a / s
    
    return best_a

def resample(a, new_n):
    """Resample sequence to new length."""
    n = len(a)
    if n == new_n:
        return a.copy()
    old_k = np.arange(n)
    new_k = np.linspace(0, n - 1, new_n)
    a_new = np.interp(new_k, old_k, a)
    a_new = np.maximum(a_new, 0)
    s = np.sum(a_new)
    if s > 1e-10:
        a_new = a_new / s
    return a_new


def optimize_pipeline(a, time_budget=60.0, verbose=False):
    """Run full optimization pipeline on a sequence."""
    n = len(a)
    a = a.copy().astype(np.float64)

        a = np.ones(n)

    start = time.monotonic()

    # Phase 1: Soft-max gradient descent
    remaining = time_budget - (time.monotonic() - start)
    if remaining > 10:
        sg_time = min(remaining * 0.3, 30.0)

    if remaining > 15:

    # Phase 3: More SGD with higher tau
    remaining = time_budget - (time.monotonic() - start)

        sg_time = min(remaining * 0.25, 20.0)

def diverse_init(n, seed):
    """Generate diverse initializations."""
    rng = np.random.default_rng(seed)
    k = np.arange(n, dtype=np.float64)
    choice = rng.integers(0, 8)
    
    if choice == 0:
        # Linear ramp + noise
        a = 1.0 + 0.5 * k / n
        a += 0.1 * rng.standard_normal(n)
    elif choice == 1:
        # Cosine with random frequency
        f = rng.uniform(0.5, 2.0)
        phase = rng.uniform(0, 2 * np.pi)
        a = 1.0 + 0.6 * np.cos(2 * np.pi * f * k / n + phase)
    elif choice == 2:
        # Two-sided: high at ends, low in middle (or vice versa)
        a = 1.0 + 0.5 * np.cos(2 * np.pi * k / n)
        if rng.random() < 0.5:
            a = a[::-1].copy()
    elif choice == 3:
        # Gaussian bump
        center = n * rng.uniform(0.2, 0.8)
        width = n * rng.uniform(0.1, 0.4)
        a = 1.0 + 0.8 * np.exp(-0.5 * ((k - center) / width) ** 2)
    elif choice == 4:
        # Sigmoid transition
        midpoint = n * rng.uniform(0.3, 0.7)
        sharpness = rng.uniform(3.0, 10.0)
        a = 1.0 + 0.5 / (1.0 + np.exp(-sharpness * (k - midpoint) / n))
    elif choice == 5:
        # Multiple cosine terms
        a = np.ones(n)
        for _ in range(rng.integers(1, 4)):
            f = rng.uniform(0.5, 5.0)
            amp = rng.uniform(0.1, 0.4)
            a += amp * np.cos(2 * np.pi * f * k / n + rng.uniform(0, 2 * np.pi))
    elif choice == 6:
        # Power law
        alpha = rng.uniform(0.3, 1.5)
        a = (k + 1) ** alpha
        if rng.random() < 0.5:
            a = a[::-1].copy()
    else:
        # Random smooth
        a = rng.random(max(5, n // 15))
        a = np.interp(k, np.arange(len(a)), a)
        a = 0.5 + a / np.max(a)
    
    a = np.maximum(a, 0.01)
    s = np.sum(a)
    return a / s
