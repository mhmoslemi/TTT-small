import numpy as np

# Cache for sparsity patterns
_spar_cache = {}

def get_sparsity(n):
    if n not in _spar_cache:
        rows = []
        cols = []
        for k in range(2 * n - 1):
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            for j in range(js, je + 1):
                rows.append(k)
                cols.append(j)
        _spar_cache[n] = (np.array(rows, dtype=np.int32), 
                          np.array(cols, dtype=np.int32), len(rows))
    return _spar_cache[n]

def fast_conv(a):
    """FFT-based linear convolution of a with itself."""
    n = len(a)
    s = 1
    while s < 2 * n - 1:
        s <<= 1
    fa = np.fft.rfft(a, s)
    return np.fft.irfft(fa * fa, s)[:2 * n - 1]

def compute_J(a):
    """Compute the objective J(a) = 2*n*max(conv(a,a))/(sum(a))^2."""
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))

def lp_best_response(a, M):
    """Solve LP: max sum(g) s.t. (a*g)_k <= M for all k, g >= 0."""
    n = len(a)
    rows, cols, nnz = get_sparsity(n)
    vals = np.empty(nnz)
    for i in range(nnz):
        k = rows[i]
        j = cols[i]
        vals[i] = a[k - j]
    A = csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                options={'presolve': True, 'time_limit': 5})
    if r.success and r.x is not None:
        g = np.maximum(r.x, 0.0)
        if g.sum() > 1e-12:
            return g
    return None

def resample_seq(src, dest_n):
    """Resample a sequence to a new length using linear interpolation."""
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)

def generate_inits(n, rng, parent):
    """Generate diverse initial sequences."""
    inits = []
    
    if parent is not None and parent.size > 0:
        inits.append(resample_seq(parent, n))
        inits.append(resample_seq(parent[::-1], n))
    
    inits.append(np.ones(n) / n)
    
    # Cosine variations
    t = np.linspace(0, 1, n, endpoint=False)
    for freq in [2, 3, 5, 7, 11]:
        inits.append(1.0 + 0.3 * np.cos(2 * np.pi * freq * t))
    inits.append(1.0 + 0.2 * np.cos(2 * np.pi * 3.7 * t) + 0.15 * np.cos(2 * np.pi * 7.3 * t))
    
    # Gaussian-like
    inits.append(np.exp(-4 * (t - 0.5) ** 2) + 0.1)
    
    # Flat spectrum (random phases)
    for _ in range(3):
        half = n // 2 + 1
        angles = rng.uniform(0, 2 * np.pi, half)
        A = np.zeros(n, dtype=complex)
        A[:half] = np.exp(1j * angles[:half])

        for k in range(2 * n - 1):
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            for j in range(js, je + 1):
                rows.append(k)
                cols.append(j)
        _spar_cache[n] = (np.array(rows, dtype=np.int32),
                          np.array(cols, dtype=np.int32), len(rows))
    return _spar_cache[n]

def fast_conv(a):
    n = len(a)
    s = 1
    while s < 2 * n - 1:
        s <<= 1
    fa = np.fft.rfft(a, s)
    return np.fft.irfft(fa * fa, s)[:2 * n - 1]

def compute_J(a):
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))

def lp_best_response(a, M):
    n = len(a)
    rows, cols, nnz = get_sparsity(n)
    vals = np.empty(nnz)
    for i in range(nnz):
        k = rows[i]
        j = cols[i]
        vals[i] = a[k - j]
    A = csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                options={'presolve': True, 'time_limit': 5})
    if r.success and r.x is not None:
        g = np.maximum(r.x, 0.0)
        if g.sum() > 1e-12:
            return g
    return None

def run_lp_iteration(a, deadline, tracker, max_iters=200):
    a = np.maximum(a.astype(np.float64), 1e-8)
    a /= a.sum()
    n = len(a)
    current_J = compute_J(a)
    tracker.update(a)
    no_improve = 0

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        b = fast_conv(a)
        M_cur = float(b.max())
        best_g = None
        fracs = [0.82, 0.85, 0.88, 0.90, 0.92, 0.94, 0.96, 0.98, 1.0]
        for frac in fracs:
            M_try = M_cur * frac
            g = lp_best_response(a, M_try)
            if g is not None:
                best_g = g
                break
        if best_g is None:
            best_g = lp_best_response(a, M_cur * 1.001)
            if best_g is None:
                break
        best_g = np.maximum(best_g, 0.0)
        gs = best_g.sum()
        if gs < 1e-10:
            break
        best_g /= gs
        best_a = None
        best_J_local = current_J
        t_list = [1.0, 0.8, 0.6, 0.4, 0.2]
        for t in t_list:
            a_cand = (1.0 - t) * a + t * best_g
            a_cand = np.maximum(a_cand, 0.0)
            ss = a_cand.sum()
            if ss < 1e-10:
                continue
            a_cand /= ss
            J_cand = compute_J(a_cand)
            if J_cand < best_J_local - 1e-15:
                best_J_local = J_cand
                best_a = a_cand
        if best_a is not None:
            a = best_a
            current_J = best_J_local
            no_improve = 0
            if it % 5 == 0:
                tracker.update(a)
        else:
            no_improve += 1
            if no_improve > 15:
                break
    tracker.update(a)
    return a

def lse_gradient_descent(a, deadline, tracker, max_iters=2000, tau_init=30.0):
    a = np.maximum(a.astype(np.float64), 1e-6)
    a /= a.sum()
    n = len(a)
    current_J = compute_J(a)
    tracker.update(a)
    tau = tau_init
    lr = 1.0 / (n * 0.5)
    momentum = 0.85
    v = np.zeros(n)
    no_improve = 0

    s_fft = 1
    while s_fft < 3 * n:
        s_fft <<= 1

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        b = fast_conv(a)
        b_max = b.max()
        exp_b = np.exp(tau * (b - b_max))
        p = exp_b / exp_b.sum()
        fp = np.fft.rfft(p, s_fft)
        fa = np.fft.rfft(a, s_fft)
        grad = np.fft.irfft(fp * np.conj(fa), s_fft)[:n]
        v = momentum * v + (1.0 - momentum) * grad
        a_new = a - lr * v
        a_new = np.maximum(a_new, 1e-6)
        a_new /= a_new.sum()
        J_new = compute_J(a_new)
        if J_new < current_J - 1e-15:
            a = a_new
            current_J = J_new
            no_improve = 0
            tau = min(500.0, tau * 1.003)
            lr = min(lr * 1.01, 2.0)
            if it % 30 == 0:
                tracker.update(a)
        else:
            no_improve += 1
            lr *= 0.6
            v *= 0.3
            if no_improve > 40 or lr < 1e-20:
                break
    tracker.update(a)
    return a

def fourier_polish(a, deadline, tracker, max_iters=300):
    a = np.maximum(a.astype(np.float64), 1e-8)
    a /= a.sum()
    n = len(a)
    current_J = compute_J(a)
    tracker.update(a)
    rng = np.random.default_rng(54321)
    lr_base = 0.05
    no_improve = 0
    t = np.arange(n)

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        best_step = None
        best_J_trial = current_J
        n_trials = 15
        for _ in range(n_trials):
            n_frequencies = rng.integers(1, 4)
            amp = lr_base * rng.uniform(0.5, 2.0)
            pert = np.zeros(n)
            for _ in range(n_frequencies):
                freq = rng.integers(1, max(2, n // 4))
                phase = rng.uniform(0, 2 * np.pi)
                pert += amp * np.sin(2 * np.pi * freq * t / n + phase)
            a_trial = a + pert
            a_trial = np.maximum(a_trial, 1e-8)
            a_trial /= a_trial.sum()
            J_trial = compute_J(a_trial)
            if J_trial < best_J_trial - 1e-15:
                best_J_trial = J_trial
                best_step = a_trial
        if best_step is not None:
            a = best_step
            current_J = best_J_trial
            no_improve = 0
            lr_base = min(lr_base * 1.03, 0.3)
            if it % 20 == 0:
                tracker.update(a)
        else:
            no_improve += 1
            lr_base *= 0.85
            if no_improve > 25:
                break
    tracker.update(a)
    return a

def peak_shave(a, deadline, tracker, max_iters=200):
    """Targeted mass transfer from peak-contributing positions."""
    a = np.maximum(a.astype(np.float64), 1e-8)
    a /= a.sum()
    n = len(a)
    current_J = compute_J(a)
    tracker.update(a)
    rng = np.random.default_rng(9999)

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        b = fast_conv(a)
        k_star = int(np.argmax(b))
        peak_val = b[k_star]
        contrib = np.zeros(n)
        js = max(0, k_star - n + 1)
        je = min(n - 1, k_star)
        for j in range(js, je + 1):
            contrib[j] = a[j] * a[k_star - j]
        total_contrib = contrib.sum()
        if total_contrib < 1e-15:
            break
        n_moves = rng.integers(2, 6)
        a_trial = a.copy()
        moved = 0
        for _ in range(n_moves):
            hot_idx = np.argsort(contrib)[-max(3, n // 20):]
            cold_mask = contrib < np.median(contrib)
            cold_idx = np.where(cold_mask)[0]
            if len(cold_idx) == 0:
                break
            src = hot_idx[rng.integers(len(hot_idx))]
            dst = cold_idx[rng.integers(len(cold_idx))]
            frac = rng.uniform(0.02, 0.15)
            transfer = a_trial[src] * frac
            a_trial[src] -= transfer
            a_trial[dst] += transfer
            moved += 1
        if moved == 0:
            break
        a_trial = np.maximum(a_trial, 1e-8)
        a_trial /= a_trial.sum()
        J_trial = compute_J(a_trial)
        if J_trial < current_J - 1e-15:
            a = a_trial
            current_J = J_trial
            if it % 30 == 0:
                tracker.update(a)
        else:
            if it % 50 == 0:
                break
    tracker.update(a)
    return a

def resample_seq(src, dest_n):
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)

def generate_inits(n, rng, parent):
    inits = []
    if parent is not None and parent.size > 0:
        inits.append(resample_seq(parent, n))
        inits.append(resample_seq(parent[::-1], n))
    inits.append(np.ones(n) / n)
    t = np.linspace(0, 1, n, endpoint=False)
    for freq in [2, 3, 5, 7, 11, 13]:
        inits.append(1.0 + 0.25 * np.cos(2 * np.pi * freq * t + rng.uniform(0, 6.28)))
    inits.append(1.0 + 0.15 * np.cos(2 * np.pi * 3.3 * t) + 0.1 * np.cos(2 * np.pi * 6.7 * t))
    inits.append(np.exp(-5 * (t - 0.5) ** 2) + 0.05)
    for _ in range(2):
        half = n // 2 + 1
        angles = rng.uniform(0, 2 * np.pi, half)
        amplitudes = rng.uniform(0.5, 1.0, half)
        A = np.zeros(n, dtype=complex)
        A[:half] = amplitudes * np.exp(1j * angles)
        for k in range(1, n - n // 2):
            A[k] = np.conj(A[n - k])
        a = np.fft.ifft(A).real
        inits.append(np.maximum(a, 0.05))
    inits.append(rng.uniform(0.3, 1.0, n))
    inits.append(0.5 + 0.5 * np.cos(np.pi * t) ** 2)
    for x in inits:
        x = np.maximum(x, 0.001).astype(np.float64)
        x /= x.sum()
    return inits

class ScoreTracker:
    def __init__(self):
        self.best = None
        self.best_score = float('inf')
        self.t_start = time.monotonic()

    def update(self, a):
        a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
        if a.size == 0 or a.sum() < 0.01:
            return
        score = float(evaluate_sequence(a.tolist()))
        if np.isfinite(score) and score < self.best_score:
            self.best = a.copy()
            self.best_score = score
            elapsed = time.monotonic() - self.t_start
            print(f"  * J={score:.8f} n={a.size} t={elapsed:.1f}s", flush=True)

    def report(self, label=""):
        if self.best is not None:
            print(f"  {label} best: J={self.best_score:.8f} n={self.best.size}", flush=True)

def simulated_annealing(a, deadline, tracker, n_restarts=50):
    rng = np.random.default_rng(31415)
    n = a.size
    a = a.copy().astype(np.float64)
    current_J = compute_J(a)
    tracker.update(a)
    temp = 0.08

    for r in range(n_restarts):
        if time.monotonic() > deadline:
            break
        if r % 10 == 0:
            perturb_scale = temp
        else:
            perturb_scale = temp * 0.5
        a_pert = a + rng.normal(0, perturb_scale, n)
        a_pert = np.maximum(a_pert, 1e-6)
        a_pert /= a_pert.sum()
        a_refined = run_lp_iteration(a_pert, deadline, tracker, max_iters=max(3, 20 - r // 4))
        J = compute_J(a_refined)
        delta = J - current_J
        if delta < 0 or rng.random() < np.exp(-delta / max(temp, 1e-10)):
            a = a_refined
            current_J = J
        temp *= 0.92
    tracker.update(a)
    return a

def coordinate_descent_polish(a, deadline, tracker, max_steps=8000):
    a = np.maximum(a.astype(np.float64), 1e-8)
    a /= a.sum()
    n = len(a)
    current_J = compute_J(a)
    tracker.update(a)
    rng = np.random.default_rng(27182)

    for step in range(max_steps):
        if time.monotonic() > deadline:
            break
        j = rng.integers(0, n)
        improved = False
        for scale in [0.6, 1.4, 0.8, 1.2, 0.4, 1.8, 0.7, 1.3]:
            a_trial = a.copy()
            a_trial[j] *= scale
            a_trial = np.maximum(a_trial, 1e-8)
            ss = a_trial.sum()
            if ss < 1e-10:
                continue
            a_trial /= ss
            J_trial = compute_J(a_trial)
            if J_trial < current_J - 1e-14:
                a = a_trial
                current_J = J_trial
                improved = True
                break
        if step % 1000 == 0:
            tracker.update(a)
    tracker.update(a)
    return a

def propose_candidate(seed=42, budget_s=1500, **kwargs):
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + float(budget_s) - 25.0
    tracker = ScoreTracker()

    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        if p.ndim == 1 and p.size > 0 and np.all(np.isfinite(p)):
            parent = np.maximum(p, 0.0)
            tracker.update(parent)
            print(f"Parent loaded: n={parent.size}, J={tracker.best_score:.8f}", flush=True)
    except (NameError, AttributeError):
        parent = None
        print("No parent, starting from scratch.", flush=True)

    if parent is None:
        parent = np.ones(512) / 512
        tracker.update(parent)

    # Phase 1: Multi-length LP
    print(f"\n{'='*60}", flush=True)
    print("Phase 1: Multi-length LP optimization", flush=True)
    print(f"{'='*60}", flush=True)

    lengths = [384, 448, 512, 576, 640]
    remaining = deadline - time.monotonic() - 350
    time_per = remaining / len(lengths)

    for n in lengths:
        if time.monotonic() > deadline - 350:
            break
        t_phase_start = time.monotonic()
        t_phase_end = min(deadline - 250, t_phase_start + time_per)
        print(f"\n--- n={n} (budget {time_per:.0f}s) ---", flush=True)

        inits = generate_inits(n, rng, parent)
        best_for_n = None
        best_J_for_n = float('inf')

        n_inits = len(inits)
        init_budget = max(5.0, (t_phase_end - time.monotonic()) / n_inits)

        for idx, a0 in enumerate(inits):
            if time.monotonic() > t_phase_end - 20:
                break
            a = a0.copy()
            a = run_lp_iteration(a, t_phase_end - 15, tracker, max_iters=150)
            if time.monotonic() < t_phase_end - 10:
                a = lse_gradient_descent(a, t_phase_end - 5, tracker, max_iters=300)
            J = compute_J(a)
            if J < best_J_for_n - 1e-15:
                best_J_for_n = J
                best_for_n = a.copy()
                tracker.update(a)

        if best_for_n is not None:
            tracker.update(best_for_n)
            print(f"  n={n} best: J={best_J_for_n:.8f} t={time.monotonic()-t_phase_start:.1f}s", flush=True)

    # Phase 2: Focused deep optimization
    if tracker.best is not None and time.monotonic() < deadline - 200:
        print(f"\n{'='*60}", flush=True)
        print(f"Phase 2: Deep optimization (n={tracker.best.size}, J={tracker.best_score:.8f})", flush=True)
        print(f"{'='*60}", flush=True)

        a = tracker.best.copy().astype(np.float64)

        # Extended LP with tighter M
        a = run_lp_iteration(a, deadline - 150, tracker, max_iters=400)

        # LSE polish
        if time.monotonic() < deadline - 100:
            a = lse_gradient_descent(a, deadline - 100, tracker, max_iters=2000)

        # Peak shaving
        if time.monotonic() < deadline - 60:
            a = peak_shave(a, deadline - 60, tracker, max_iters=100)

        # Fourier polish
        if time.monotonic() < deadline - 30:
            a = fourier_polish(a, deadline - 30, tracker, max_iters=200)

        tracker.report("Phase 2")

    # Phase 3: Simulated annealing for escape
    if tracker.best is not None and time.monotonic() < deadline - 80:
        print(f"\n{'='*60}", flush=True)
        print(f"Phase 3: Simulated annealing (n={tracker.best.size})", flush=True)
        print(f"{'='*60}", flush=True)

        a = simulated_annealing(tracker.best, deadline - 60, tracker, n_restarts=40)

        if time.monotonic() < deadline - 25:
            a = run_lp_iteration(a, deadline - 20, tracker, max_iters=50)
        if time.monotonic() < deadline - 10:
            a = lse_gradient_descent(a, deadline - 5, tracker, max_iters=500)
        tracker.report("Phase 3")

    # Phase 4: Final polish
    if tracker.best is not None and time.monotonic() < deadline - 10:
        print(f"\n{'='*60}", flush=True)
        print("Phase 4: Final polish", flush=True)
        print(f"{'='*60}", flush=True)

        a = coordinate_descent_polish(tracker.best, deadline - 8, tracker, max_steps=6000)
        if time.monotonic() < deadline - 3:
            a = run_lp_iteration(a, deadline - 2, tracker, max_iters=30)
        if time.monotonic() < deadline - 1:
            a = lse_gradient_descent(a, deadline - 0.5, tracker, max_iters=200)
        tracker.report("Phase 4")

    if tracker.best is None:
        tracker.best = np.ones(256)
        tracker.best_score = compute_J(tracker.best)
        tracker.update(tracker.best)

    print(f"\n{'='*60}", flush=True)
    print(f"FINAL: J={tracker.best_score:.8f} n={tracker.best.size} "
          f"time={time.monotonic()-tracker.t_start:.1f}s", flush=True)
    print(f"{'='*60}", flush=True)

    return [float(x) for x in tracker.best]
