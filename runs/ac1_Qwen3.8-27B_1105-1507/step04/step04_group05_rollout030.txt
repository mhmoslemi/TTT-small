```python
def solve_for_active_set(K, n):
    """Solve A_K * a = c*1, sum(a)=1, a>=0 for given active set K."""
    # Build A_K: A[i,j] = 1 if (i+j) in K
    K_set = set(K)
    A = np.zeros((n, n))
    for i in range(n):
        for j in range(n):
            if (i + j) in K_set:
                A[i, j] = 1.0
    
    # System: [A, -1; 1^T, 0] [a; c] = [0; 1]
    M = np.zeros((n+1, n+1))
    M[:n, :n] = A
    M[:n, n] = -1.0
    M[n, :n] = 1.0
    rhs = np.zeros(n+1)
    rhs[n] = 1.0
    
    try:
        x = np.linalg.solve(M, rhs)
    except np.linalg.LinAlgError:
        x = np.linalg.lstsq(M, rhs, rcond=None)[0]
    
    a = x[:n]
    c = x[n]
    
    if np.all(a >= -1e-10):
        a = np.maximum(a, 0)
        a /= a.sum()
        return a
    return None
```

            K = {n - 1}
            if 2*(n-1) - (n-1) != n-1:
                K.add(2*(n-1) - (n-1))
        
        # Try linear system first
        a_new = solve_active_set_system(K, n)
        
        if a_new is not None:
            c_new = fast_conv(a_new)
            c_new_max = c_new.max()
            J_new = 2.0 * n * c_new_max  # since sum(a_new) = 1
            if J_new < current_J - 1e-12:
                a = a_new
                current_J = J_new
                if tracker:
                    tracker.update(a)
                if verbose and it % 5 == 0:
                    print(f"    AS iter {it}: |K|={len(K)}, J={current_J:.8f}", flush=True)
            else:
                # Linear system solution doesn't improve; try QP
                a_qp = solve_active_set_qp(K, n, a)
                if a_qp is not None:
                    c_qp = fast_conv(a_qp)
                    J_qp = 2.0 * n * c_qp.max()
                    if J_qp < current_J - 1e-12:
                        a = a_qp
                        current_J = J_qp
                        if tracker:
                            tracker.update(a)
        else:
            # Linear system has negative solution; try QP
            a_qp = solve_active_set_qp(K, n, a)
            if a_qp is not None:
                c_qp = fast_conv(a_qp)
                J_qp = 2.0 * n * c_qp.max()
                if J_qp < current_J - 1e-12:
                    a = a_qp
                    current_J = J_qp
                    if tracker:
                        tracker.update(a)
        
        # Check convergence
        c_now = fast_conv(a)
        c_now_max = c_now.max()
        if abs(c_now_max - c_max) < 1e-10:
            break
    
    if tracker:
        tracker.update(a)
    return a, current_J


def gradient_J(a):
    """Compute gradient of J(a) = 2n*max(c)/S^2 where c=conv(a,a)."""
    n = len(a)
    S = a.sum()
    c = fast_conv(a)
    c_max = c.max()
    
    # Subgradient: average over active positions
    tol = 1e-6 * c_max
    active = np.where(c >= c_max - tol)[0]
    
    grad = np.zeros(n)
    for k in active:
        for i in range(n):
            j = k - i
            if 0 <= j < n:
                grad[i] += 2.0 * a[j]
    grad /= len(active)
    
    # dJ/da[i] = (2n/S^2) * dM/da[i] - (4n*M/S^3) * 1
    grad = (2.0 * n / (S * S)) * grad - (4.0 * n * c_max / (S ** 3)) * np.ones(n)
    return grad


def slsqp_polish(a_init, deadline, tracker, max_iter=200):
    """Polish using SLSQP on the normalized problem: min 2n*max(c) s.t. sum(a)=1, a>=0."""
    a = np.maximum(a_init.astype(np.float64), 0)
    a /= a.sum()
    n = len(a)
    current_J = compute_J(a)
    if tracker:
        tracker.update(a)
    
    def obj(a):
        c = fast_conv(a)
        return 2.0 * n * c.max()
    
    def grad(a):
        n_local = len(a)
        S = a.sum()
        c = fast_conv(a)
        c_max = c.max()
        tol = 1e-7 * c_max
        active = np.where(c >= c_max - tol)[0]
        g = np.zeros(n_local)
        for k in active:
            for i in range(n_local):
                j = k - i
                if 0 <= j < n_local:
                    g[i] += 2.0 * a[j]
        g /= len(active)
        return g
    
    constraints = [{'type': 'eq', 'fun': lambda a: np.sum(a) - 1.0,
                   'jac': lambda a: np.ones(n)}]
    bounds = [(0.0, None)] * n
    
    result = minimize(obj, a, jac=grad, method='SLSQP',
                      bounds=bounds, constraints=constraints,
                      options={'maxiter': max_iter, 'ftol': 1e-14})
    
    if result.success:
        a_res = np.maximum(result.x, 0)
        s = a_res.sum()
        if s > 1e-10:
            a_res /= s
            J_res = compute_J(a_res)
            if tracker:
                tracker.update(a_res)
            return a_res, J_res
    
    return a, current_J


def coordinate_descent_1d_fast(a, i, b, S):
    """Find optimal a[i] given others fixed, using golden section on 1D function."""
    n = len(a)
    ai = a[i]
    
    def f(x):
        new_S = S - ai + x
        if new_S < 1e-10:
            return 1e10
        # Only need to check positions k where i participates
        max_c = b.max()  # Start with current max for positions not involving i
        for k in range(max(0, i), min(2 * n - 1, i + n)):
            ki = k - i
            if 0 <= ki < n:
                if k == 2 * i:
                    ck = b[k] - ai * ai + x * x
                else:
                    ck = b[k] - 2.0 * ai * a[ki] + 2.0 * x * a[ki]
            else:
                ck = b[k]
            if ck > max_c:
                max_c = ck
        return 2.0 * n * max_c / (new_S * new_S)
    
    # Golden section search
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    lo = 0.0
    hi = max(3.0 * ai + 0.01, S / n * 5.0, 0.1)
    
    c_pt = hi - gr * (hi - lo)
    d_pt = lo + gr * (hi - lo)
    fc = f(c_pt)
    fd = f(d_pt)
    
    for _ in range(30):
        if fc < fd:
            hi = d_pt
            d_pt = c_pt
            fd = fc
            c_pt = hi - gr * (hi - lo)
            fc = f(c_pt)
        else:
            lo = c_pt
            c_pt = d_pt
            fc = fd
            d_pt = lo + gr * (hi - lo)
            fd = f(d_pt)
    
    return max(0.0, (lo + hi) / 2.0)


def coordinate_descent_pass(a, b=None, S=None):
    """One pass of coordinate descent with incremental convolution update."""
    a = np.maximum(a.astype(np.float64), 0)
    S = a.sum()
    if S < 1e-10:
        return a
    n = len(a)
    
    if b is None:
        b = fast_conv(a)
    else:
        b = b.copy()
    
    indices = list(range(n))
    np.random.shuffle(indices)
    
    for i in indices:
        new_val = coordinate_descent_1d_fast(a, i, b, S)
        old_val = a[i]
        if abs(new_val - old_val) > 1e-12:
            a[i] = new_val
            S = S - old_val + new_val
            # Update convolution
            for k in range(max(0, i), min(2 * n - 1, i + n)):
                ki = k - i
                if 0 <= ki < n:
                    if k == 2 * i:
                        b[k] = b[k] - old_val * old_val + new_val * new_val
                    else:
                        b[k] = b[k] - 2.0 * old_val * a[ki] + 2.0 * new_val * a[ki]
    
    # Verify
    b_check = fast_conv(a)
    return a, b_check


def gradient_descent(a, deadline, tracker, lr=None, max_iter=500):
    """Projected gradient descent with backtracking line search."""
    a = np.maximum(a_init if (a_init := a) is None else a, 0).astype(np.float64)
    a /= a.sum()
    n = len(a)
    current_J = compute_J(a)
    if tracker:
        tracker.update(a)
    
    if lr is None:
        lr = 0.01
    
    for it in range(max_iter):
        if time.monotonic() > deadline:
            break
        
        g = gradient_J(a)
        g_norm = np.linalg.norm(g)
        if g_norm < 1e-12:
            break
        
        # Backtracking line search
        step = lr
        improved = False
        for _ in range(20):
            a_trial = a - step * g
            a_trial = np.maximum(a_trial, 0)
            s = a_trial.sum()
            if s < 1e-10:
                step *= 0.5
                continue
            a_trial /= s
            J_trial = compute_J(a_trial)
            if J_trial < current_J:
                a = a_trial
                current_J = J_trial
                improved = True
                if tracker:
                    tracker.update(a)
                break
            step *= 0.5
        
        if not improved:
            lr *= 0.5
            if lr < 1e-8:
                break
    
    if tracker:
        tracker.update(a)
    return a, current_J


class ScoreTracker:
    def __init__(self):
        self.best = None
        self.best_score = float('inf')
        self.t_start = time.monotonic()
    
    def update(self, a):
        a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
        if a.size == 0 or a.sum() < 0.01:
            return
        score = float(evaluate_sequence(a.tolist()))
        if np.isfinite(score) and score < self.best_score:
            self.best = a.copy()
            self.best_score = score
            elapsed = time.monotonic() - self.t_start
            print(f"  * NEW BEST: J={score:.8f} n={a.size} t={elapsed:.0f}s", flush=True)
    
    def report(self, label=""):
        if self.best is not None:
            print(f"  {label} best: J={self.best_score:.8f} n={self.best.size} "
                  f"t={time.monotonic()-self.t_start:.0f}s", flush=True)


def resample_seq(src, dest_n):
    """Resample 1D sequence to new length."""
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    if src.size == 1:
        return np.full(dest_n, src[0])
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)


def multi_k_exploration(a_base, n, deadline, tracker, n_trials=20):
    """Try multiple different active sets K and pick the best."""
    best_a = a_base.copy()
    best_J = compute_J(a_base)
    tracker.update(best_a)
    
    c = fast_conv(a_base)
    c_max = c.max()
    all_positions = set(range(2 * n - 1))
    
    # Generate various K candidates
    K_candidates = []
    
    # Standard active set
    for tol in [0.0001, 0.001, 0.005, 0.01, 0.02, 0.05]:
        K = get_active_set(c, tol, make_symmetric=True, n=n)
        if 2 <= len(K) <= 2 * n - 3:
            K_candidates.append(K)
    
    # Center-biased sets
    center = n - 1
    for radius in [1, 2, 3, 5, 8, 12, 20]:
        K = set(range(max(0, center - radius), min(2 * n - 2, center + radius) + 1))
        K_candidates.append(K)
    
    # Random symmetric sets
    rng = np.random.default_rng(123)
    for _ in range(n_trials):
        size = rng.integers(10, min(2 * n - 1, 100))
        K = set(rng.choice(2 * n - 1, size=size, replace=False).tolist())
        # Make symmetric
        K_sym = set()
        for k in K:
            K_sym.add(k)
            K_sym.add(2 * (n - 1) - k)
        K_candidates.append(K_sym)
    
    # Remove duplicates (by frozenset)
    seen = set()
    unique_Ks = []
    for K in K_candidates:
        key = frozenset(K)
        if key not in seen and 2 <= len(K) <= 2 * n - 3:
            seen.add(key)
            unique_Ks.append(K)
    
    for K in unique_Ks:
        if time.monotonic() > deadline:
            break
        
        a_sol = solve_active_set_system(K, n)
        if a_sol is not None:
            J_sol = compute_J(a_sol)
            if J_sol < best_J - 1e-10:
                best_a = a_sol
                best_J = J_sol
                tracker.update(best_a)
                print(f"    K size {len(K)}: J={J_sol:.8f}", flush=True)
    
    return best_a, best_J
