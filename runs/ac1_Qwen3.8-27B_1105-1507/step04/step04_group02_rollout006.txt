```python
import numpy as np
import time
from scipy.sparse import coo_matrix
from scipy.optimize import linprog

def compute_conv(a):
    """FFT-based linear convolution."""
    n = len(a)
    size = 1
    while size < 2*n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    conv = np.fft.irfft(fa * fa, size)
    return conv[:2*n-1]

def compute_obj(a):
    """Compute the objective value."""
    n = len(a)
    S = np.sum(a)
    if S < 1e-10:
        return np.inf
    conv = compute_conv(a)
    M = np.max(conv)
    return 2.0 * n * M / (S*S)

def build_toeplitz(a):
    """Build the Toeplitz matrix T such that (T*g)[k] = (a*g)[k]."""
    n = len(a)
    rows = []
    cols = []
    vals = []
    for j in range(n):
        for k in range(j, min(j+n, 2*n-1)):
            i = k - j
            rows.append(k)
            cols.append(j)
            vals.append(a[i])
    A = coo_matrix((vals, (rows, cols)), shape=(2*n-1, n)).tocsr()
    return A

def solve_lp(a, M, time_limit=10):
    """Solve LP: max sum(g) s.t. (a*g)[k] <= M, g >= 0."""
    n = len(a)
    A = build_toeplitz(a)
    c = -np.ones(n)
    b_ub = np.full(2*n-1, M)
    bounds = [(0, None)] * n
    try:
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': time_limit})
        if res.status == 0:
            g = np.maximum(res.x, 0)
            if np.sum(g) > 1e-10:
                return g
    except:
        pass
    return None

def smoothed_gradient(a, tau=1.0, topk=None):
    """Compute smoothed gradient using log-sum-exp or top-K."""
    n = len(a)
    conv = compute_conv(a)
    
    if topk is not None and topk < len(conv):
        # Top-K approach
        k_idx = np.argsort(conv)[-topk:]
        weights = np.zeros(len(conv))
        weights[k_idx] = 1.0 / topk
    else:
        # Log-sum-exp approach
        max_c = np.max(conv)
        exp_vals = np.exp((conv - max_c) / tau)
        weights = exp_vals / np.sum(exp_vals)
    
    grad = np.zeros(n)
    for k in range(len(conv)):
        if weights[k] < 1e-10:
            continue
        for i in range(n):
            j = k - i
            if 0 <= j < n:
                coeff = 2.0 if i != j else 1.0
                grad[i] += weights[k] * coeff * a[j]
    
    return grad

def check_conv(a, M, tol=1e-6):
```

```python
def cyclic_cdd_v2(a, max_rounds=200, time_budget=60):
    """Cyclic coordinate descent with O(n) update per coordinate."""
    n = len(a)
    a = a.copy()
    S = np.sum(a)
    if S < 1e-10:
        return a
    a = a / S
    S = 1.0
    
    conv = compute_conv(a)
    current_max = np.max(conv)
    best_obj = 2.0 * n * current_max
    best_a = a.copy()
    start = time.monotonic()
    
    # Scales to try
    scales = [0.2, 0.5, 0.7, 0.8, 1.0, 1.25, 1.5, 2.0, 5.0, 0.0]
    
    for round_num in range(max_rounds):
        if time.monotonic() - start > time_budget:
            break
        
        round_improved = False
        for i in range(n):
            if time.monotonic() - start > time_budget:
                break
            
            a_old = a[i]
            if S - a_old < 1e-10:
                continue  # Can't reduce this further
            
            best_t = a_old
            best_local = best_obj
            best_conv = conv
            
            for s in scales:
                if s == 0.0:
                    t = 0.0
                else:
                    t = a_old * s
                    if t > S:  # can't exceed total sum
                        continue
            
                if t == a_old:
                    continue
                
                S_new = S + (t - a_old)
                if S_new < 1e-10:
                    continue
                
                # Update convolution
                delta = t - a_old
                new_conv = conv.copy()  # This is O(n) copy, acceptable
                
                # Affected positions: k in [i, i+n-1]
                k_start = i
                k_end = min(i + n - 1, 2*n - 2)
                
                for k in range(k_start, k_end + 1):
                    j = k - i  # the other index
                    if 0 <= j < n:
                        if j == i:
                            # k = 2i: change is t^2 - a_old^2 = (t-a_old)(t+a_old)
                            new_conv[k] += delta * (t + a_old)
                        else:
                            # change is 2*delta*a[j]
                            new_conv[k] += 2.0 * delta * a[j]
                
                new_max = max(np.max(new_conv[:k_start]), np.max(new_conv[k_start:k_end+1]), np.max(new_conv[k_end+1:]) if k_end+1 < len(new_conv) else 0)
                
                new_obj = 2.0 * n * new_max / (S_new * S_new)
                
                if new_obj < best_local:
                    best_local = new_obj
                    best_t = t
                    best_conv = new_conv
        
            if best_t != a_old:
                a[i] = best_t
                S = S + (best_t - a_old)
                conv = best_conv
                current_max = np.max(conv)
                obj = 2.0 * n * current_max / (S*S)
                if obj < best_obj:
                    best_obj = obj
                    best_a = a.copy()
                    round_improved = True
        
        if not round_improved:
            break
    
    return best_a / np.sum(best_a) if np.sum(best_a) > 1e-10 else best_a
```

This is O(n * |scales| * n) = O(n^2 * |scales|) per round, which for n=500 and |scales|=10 gives about 2.5 million operations per round. With 200 rounds, that's 500 million operations. This might be too slow in Python.

Let me vectorize the inner loop. Instead of looping over k for each candidate t, I can precompute the "pattern" of how the convolution changes:

For coordinate i, the change in conv[k] when a[i] changes by delta is:
Δconv[k] = 2 * delta * a[k-i] for k in [i, i+n-1], k ≠ 2i
Δconv[2i] = delta * (t + a_old) (special case)

Let me precompute the vector `v` where v[k] = 2*a[k-i] for k in [i, i+n-1] (with v[2i] = 0, handled separately).

Then for a candidate t with delta = t - a_old:
new_conv = conv + delta * v + delta*(t+a_old) * e_{2i} (where e_{2i} is the unit vector)

The max of new_conv can be computed as:
max(max(conv[k] for k not in [i, i+n-1]), max(conv[k] + delta*v[k] for k in [i, i+n-1], k ≠ 2i), conv[2i] + delta*(t+a_old))

The first max is precomputed (it's the max of the "unaffected" positions).

This makes each candidate evaluation O(n) for the max computation, and with |scales| candidates per coordinate and n coordinates, each round is O(n^2 * |scales|).

```python
def cyclic_cdd_v3(a, max_rounds=200, time_budget=120):
    n = len(a)
    a = a.copy()
    S = np.sum(a)
    if S < 1e-10:
        return a
    a = a / S
    S = 1.0
    
    conv = compute_conv(a)
    best_obj = 2.0 * n * np.max(conv)
    best_a = a.copy()
    start = time.monotonic()
    
    scales = np.array([0.0, 0.25, 0.5, 0.7, 0.8, 1.25, 1.5, 2.0])
    
    for round_num in range(max_rounds):
        if time.monotonic() - start > time_budget:
            break
        
        round_improved = False
        for i in range(n):
            if time.monotonic() - start > time_budget:
                break
            
            a_old = a[i]
            
            # Precompute the "pattern" vector
            # v[k] = 2*a[k-i] for k in [i, i+n-1] where k-i != i
            # Special: v[2i] will be handled with (t + a_old) factor
            
            k_start = i
            k_end = min(i + n - 1, 2*n - 2)
            
            # Build the pattern for this i
            j_vals = np.arange(k_start, k_end + 1) - i  # = 0, 1, ..., min(n-1, n-1-i)
            # v for general k (k ≠ 2i): 2 * a[k-i]
            v = 2.0 * a[j_vals]
            # Position of 2i in the range (if it exists)
            if 2*i >= k_start and 2*i <= k_end:
                idx_2i = 2*i - k_start
                v[idx_2i] = 0.0  # Handle separately
            
            # Unaffected max (positions outside [k_start, k_end])
            if k_start > 0:
                left_max = np.max(conv[:k_start])
            else:
                left_max = -np.inf
            if k_end + 1 < len(conv):
                right_max = np.max(conv[k_end+1:])
            else:
                right_max = -np.inf
            unaffected_max = max(left_max, right_max)
            
            # Current conv values in the affected range
            conv_affected = conv[k_start:k_end+1].copy()
            
            best_t = a_old
            best_local = 2.0 * n * max(np.max(conv), unaffected_max) / (S*S)
            
            S_other = S - a_old  # sum of all other elements
            
            for s in scales:
                t = a_old * s if s != 0 else 0.0
                if t == a_old:
                    continue
                delta = t - a_old
                S_new = S_other + t
                if S_new < 1e-10:
                    continue
                
                # New conv in affected range
                new_affected = conv_affected + delta * v
                
                # Handle the 2i position specially
                if 2*i >= k_start and 2*i <= k_end:
                    idx_2i = 2*i - k_start
                    # Original value at 2i: includes a_old^2
                    # New value: should include t^2
                    # Change: t^2 - a_old^2 = (t-a_old)(t+a_old) = delta*(t+a_old)
                    new_affected[idx_2i] += delta * (t + a_old)
                
                new_max = max(unaffected_max, np.max(new_affected))
                new_obj = 2.0 * n * new_max / (S_new * S_new)
                
                if new_obj < best_local:
                    best_local = new_obj
                    best_t = t
        
            if best_t != a_old:
                delta = best_t - a_old
                a[i] = best_t
                S = S_other + best_t
                conv[k_start:k_end+1] = conv_affected + delta * v
                if 2*i >= k_start and 2*i <= k_end:
                    idx_2i = 2*i - k_start
                    conv[2*i] += delta * (best_t + a_old)
                obj = 2.0 * n * np.max(conv) / (S*S)
                if obj < best_obj:
                    best_obj = obj
                    best_a = a.copy()
                    round_improved = True
        
        if not round_improved:
            break
    
    result = best_a / np.sum(best_a) if np.sum(best_a) > 1e-10 else best_a
    return result
```
