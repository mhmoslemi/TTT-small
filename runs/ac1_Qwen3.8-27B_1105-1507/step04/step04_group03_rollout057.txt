Improved optimizer for minimizing J(a) = 2*n*max(conv(a,a))/sum(a)^2.
Key improvements over previous version:

2. Warm-starting for the LP
3. Better step size strategy (geometric line search)
4. Multi-length sweep with intelligent resampling
5. "Chasing" phase that focuses on top peak positions
6. Final L-BFGS-B polish

import numpy as np
from scipy.optimize import linprog, minimize
from scipy.sparse import csr_matrix
import time

def fast_conv(a):
    n = len(a)
    size = 1
    while size < 2*n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2*n-1]

def compute_J(a):
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s*s))

def build_sparse_A(n, a, active_indices=None):
    rows = []
    cols = []
    vals = []
    if active_indices is None:
        active_indices = range(2*n - 1)
    for idx, k in enumerate(active_indices):
        j_min = max(0, k - n + 1)
        j_max = min(n-1, k)
        for j in range(j_min, j_max + 1):
            rows.append(idx)
            cols.append(j)
            vals.append(a[k-j])
    return csr_matrix((vals, (rows, cols)), shape=(len(active_indices), n))

def lp_best_response(a, M, active_tol=0.05, deadline_check=None):
    n = len(a)
    b = fast_conv(a)
    # Select active constraints: those where b[k] > M*(1-active_tol)
    threshold = M * (1 - active_tol)
    active = np.where(b > threshold)[0]
    if len(active) == 0:
        active = np.array([np.argmax(b)])
    
    A = build_sparse_A(n, a, active)
    c_obj = -np.ones(n)
    b_ub = np.full(len(active), M)
    bounds = [(0, None)] * n
    
    try:
        r = linprog(c_obj, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                    options={'presolve': True, 'time_limit': 5})
        if r.success:
            return np.maximum(r.x, 0)
    except Exception as e:
        pass
    return None

        n_active_max = max(10, int(len(b) * active_frac_cur))
        if len(active) > n_active_max:
            active = np.argsort(b)[-n_active_max:]
        elif len(active) < 3:
            active = np.argsort(b)[-max(5, len(b) // 10):]

        g = solve_lp(a, M, active)
        if g is None:
            no_improve += 5
            if no_improve > 60:
                active_frac_cur = max(0.05, active_frac_cur * 0.7)
                no_improve = 0
                if active_frac_cur < 0.05:
                    break
            continue

        g = np.maximum(g, 0.0)
        g_sum = g.sum()
        if g_sum < 1e-12:
            break
        g = g / g_sum

        # Line search
        best_trial_J = current_J
        best_trial_a = None
        t = 0.4 if not use_mult else 0.25

        for _ in range(35):
            if use_mult:
                ratio = g / np.maximum(a, 1e-12)
                a_trial = a * np.power(ratio, t)
            else:
                a_trial = (1.0 - t) * a + t * g
            a_trial = np.maximum(a_trial, 0.0)
            s = a_trial.sum()
            if s < 1e-12:
                t *= 0.5
                continue
            a_trial = a_trial / s
            J_trial = compute_J(a_trial)
            if J_trial < best_trial_J:
                best_trial_J = J_trial
                best_trial_a = a_trial.copy()
                break
            t *= 0.55
            if t < 0.001:
                break

        if best_trial_a is not None:
            a = best_trial_a
            current_J = best_trial_J
            if current_J < best_J - 1e-13:
                best_J = current_J
                best_a = a.copy()
                no_improve = 0
            else:
                no_improve += 1
        else:
            no_improve += 1

        if no_improve > 100:
            active_frac_cur = max(0.03, active_frac_cur * 0.75)
            no_improve = 0
            if active_frac_cur < 0.03:
                break

        if it % 300 == 0:
            print(f"      [{label}] it={it} J={best_J:.8f} frac={active_frac_cur:.3f}", flush=True)

    return best_a, best_J


def dual_guided_descent(a_init, deadline, label=""):
    a = np.maximum(a_init.astype(np.float64), 1e-12)
    a = a / a.sum()
    n = len(a)
    current_J = compute_J(a)
    best_J = current_J
    best_a = a.copy()
    it = 0
    momentum = np.zeros_like(a)
    beta = 0.85

    while time.monotonic() < deadline:
        it += 1
        b = fast_conv(a)
        M = float(b.max())
        tol = max(1e-6 * M, 1e-14)
        active = np.where(b >= M - tol)[0]

        # Solve dual to get weights y
        # Dual: min M*sum(y) s.t. A^T y >= 1, y >= 0
        # Instead, use a simple weighting: y[k] proportional to (b[k] - mean(b))^+
        b_clipped = np.clip(b - 0.8 * M, 0, None)
        y = b_clipped / max(b_clipped.sum(), 1e-15)

        # Compute dual gradient
        grad = np.zeros_like(a)
        for k in active:
            w = y[k]
            if w < 1e-8:
                continue
            lo = max(0, k - n + 1)
            hi = min(n - 1, k)
            for j in range(lo, hi + 1):
                grad[j] += 2.0 * w * a[k - j]

        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-15:
            break

        t_step = 1.5 / (gnorm * np.sqrt(it + 1.0))
        momentum = beta * momentum + (1.0 - beta) * grad
        a_new = a - t_step * momentum
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s < 1e-12:
            break
        a_new = a_new / s

        J_new = compute_J(a_new)
        if J_new < best_J:
            best_J = J_new
            best_a = a_new.copy()
        a = a_new

        if it % 200 == 0:
            print(f"      [{label}] it={it} J={best_J:.8f}", flush=True)

    return best_a, best_J


def lbb_objective(x):
    ax = np.maximum(x, 0.0)
    s = ax.sum()
    if s < 1e-8:
        return 1e10
    ax = ax / s
    b = fast_conv(ax)
    return float(b.max())


def lbb_polish(a, max_fun=50000):
    n = len(a)
    try:
        res = minimize(lbb_objective, a.astype(np.float64),
                       method='L-BFGS-B', bounds=[(0.0, None)] * n,
                       options={'maxiter': 300, 'max_fun': max_fun, 'ftol': 1e-16,
                                'gtol': 1e-12})
        if res.fun < 1e10:
            return np.maximum(res.x, 0.0)
    except Exception:
        pass
    return a.copy()


def perturb_seq(a, rng, ptype):
    n = len(a)
    t = np.arange(n, dtype=np.float64) / n
    if ptype == 0:
        s = rng.integers(1, max(2, n // 12))
        ap = np.roll(a, int(s))
    elif ptype == 1:
        freq = rng.uniform(1.0, 6.0)
        amp = rng.uniform(0.03, 0.08)
        ap = a * (1.0 + amp * np.cos(2.0 * np.pi * freq * t + rng.uniform(0, 2 * np.pi)))
    elif ptype == 2:
        k = int(rng.integers(3, max(4, n // 8)))
        kernel = np.ones(k) / k
        ap = np.convolve(a, kernel, mode='same')
    elif ptype == 3:
        alpha = rng.uniform(0.4, 0.8)
        edge = max(1, int(n * (1.0 - alpha) / 2))
        window = np.ones(n)
        for i in range(edge):
            frac = (i + 1) / (edge + 1)
            window[i] = frac
            window[n - 1 - i] = frac
        ap = a * window
    else:
        ap = a + rng.normal(0, 0.02, n)
    ap = np.maximum(ap, 0.001)
    s = ap.sum()
    if s > 1e-8:
        ap = ap / s
    return ap


class Tracker:
    def __init__(self):
        self.best = None
        self.best_score = float('inf')
        self.t_start = time.monotonic()

    def update(self, a):
        a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
        if a.size == 0 or a.sum() < 0.01:
            return
        score = float(evaluate_sequence(a.tolist()))
        if np.isfinite(score) and score < self.best_score:
            self.best_score = score
            self.best = a.copy()
            elapsed = time.monotonic() - self.t_start
            print(f"  [NEW BEST] J={score:.8f} n={a.size} t={elapsed:.0f}s", flush=True)

    def report(self):
        if self.best is not None:
            print(f"  Best: J={self.best_score:.8f} n={self.best.size}", flush=True)


def propose_candidate(seed=42, budget_s=1500, **kwargs):
    rng = np.random.default_rng(seed)
    tracker = Tracker()
    t_start = time.monotonic()
    deadline = t_start + float(budget_s) - 35.0

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        if p.ndim == 1 and p.size > 0 and np.all(np.isfinite(p)):
            parent = np.maximum(p, 0.0)
            tracker.update(parent)
            print(f"Parent: n={parent.size}, J={tracker.best_score:.6f}", flush=True)
    except (NameError, AttributeError):
        pass

    if parent is None:
        parent = np.ones(512)
        tracker.update(parent)

    # === Phase 1: LP mirror descent (linear) on parent ===
    print(f"\n{'='*50}", flush=True)
    print(f"Phase 1: LP linear (n={parent.size})", flush=True)
    print(f"{'='*50}", flush=True)
    dl1 = min(deadline - 900, time.monotonic() + 250)
    a1, J1 = lp_mirror_descent(parent, dl1, use_mult=False, active_frac=0.4, label="P1")
    tracker.update(a1)
    print(f"  P1 done: J={J1:.8f}", flush=True)

    # === Phase 2: LP multiplicative ===
    if time.monotonic() < deadline - 700:
        print(f"\nPhase 2: LP multiplicative", flush=True)
        start2 = tracker.best if tracker.best is not None else a1
        dl2 = min(deadline - 700, time.monotonic() + 180)
        a2, J2 = lp_mirror_descent(start2, dl2, use_mult=True, active_frac=0.3, label="P2")
        tracker.update(a2)
        print(f"  P2 done: J={J2:.8f}", flush=True)

    # === Phase 3: Multi-length exploration ===
    if time.monotonic() < deadline - 450:
        print(f"\nPhase 3: Multi-length", flush=True)
        base = tracker.best if tracker.best is not None else a1
        base_n = base.size

        candidate_lengths = set()
        for n_try in [256, 320, 384, 448, 512, 576, 640, 768, 896, 1024, 1280, 1536, 2048, 2560]:
            if 128 <= n_try <= 3000 and n_try != base_n:
                candidate_lengths.add(n_try)
        # Add near parent
        for delta in [-128, -64, 64, 128, 256]:
            n_try = base_n + delta
            if 128 <= n_try <= 3000:
                candidate_lengths.add(n_try)

        lengths_to_try = sorted(candidate_lengths)[:10]

        remaining = deadline - 450 - time.monotonic()
        per_len = max(25.0, remaining / max(1, len(lengths_to_try)) * 0.65)

        for n_try in lengths_to_try:
            if time.monotonic() > deadline - 430:
                break
            a_try = resample_seq(base, n_try)
            a_try = np.maximum(a_try, 1e-10)
            a_try = a_try / a_try.sum()
            J_init = compute_J(a_try)
            dl_try = time.monotonic() + per_len
            a_try, J_try = lp_mirror_descent(a_try, dl_try, use_mult=False,
                                             active_frac=0.3, label=f"ML-n{n_try}")
            tracker.update(a_try)
            print(f"  n={n_try}: J={J_try:.8f} (init={J_init:.6f})", flush=True)

    # === Phase 4: Long LP on best ===
    if time.monotonic() < deadline - 180:
        print(f"\nPhase 4: Long LP", flush=True)
        start4 = tracker.best if tracker.best is not None else a1
        dl4 = min(deadline - 130, time.monotonic() + 280)
        a4, J4 = lp_mirror_descent(start4, dl4, use_mult=False, active_frac=0.25, label="P4")
        tracker.update(a4)
        print(f"  P4 done: J={J4:.8f}", flush=True)

        if time.monotonic() < deadline - 40:
            dl4b = min(deadline - 25, time.monotonic() + 100)
            a4b, J4b = lp_mirror_descent(tracker.best, dl4b, use_mult=True,
                                         active_frac=0.15, label="P4b")
            tracker.update(a4b)
            print(f"  P4b done: J={J4b:.8f}", flush=True)

    # === Phase 5: Perturbation + LP ===
    if time.monotonic() < deadline - 80:
        print(f"\nPhase 5: Perturbed LP", flush=True)
        base5 = tracker.best if tracker.best is not None else a1
        n_pert = 4
        per_pert = max(10.0, (deadline - 80 - time.monotonic()) / n_pert * 0.6)
        for i in range(n_pert):
            if time.monotonic() > deadline - 75:
                break
            a_p = perturb_seq(base5, rng, i % 5)
            dl_p = time.monotonic() + per_pert
            a_p, J_p = lp_mirror_descent(a_p, dl_p, use_mult=False,
                                         active_frac=0.3, label=f"PT{i}")
            tracker.update(a_p)
            if J_p < tracker.best_score:
                print(f"  Pert {i}: J={J_p:.8f}", flush=True)

    # === Phase 6: Dual-guided descent ===
    if time.monotonic() < deadline - 30:
        print(f"\nPhase 6: Dual-guided descent", flush=True)
        start6 = tracker.best if tracker.best is not None else a1
        dl6 = min(deadline - 15, time.monotonic() + 50)
        a6, J6 = dual_guided_descent(start6, dl6, label="P6")
        tracker.update(a6)
        print(f"  P6 done: J={J6:.8f}", flush=True)

    # === Phase 7: L-BFGS-B final polish ===
    if time.monotonic() < deadline - 3 and tracker.best is not None:
        print(f"\nPhase 7: L-BFGS-B", flush=True)
        remaining_fun = int((deadline - time.monotonic() - 2.0) * 40)
        max_fun = max(5000, min(80000, remaining_fun))
        a7 = lbb_polish(tracker.best, max_fun=max_fun)
        tracker.update(a7)
        print(f"  P7 done: J={tracker.best_score:.8f}", flush=True)

    # Final
    if tracker.best is not None:
        print(f"\n{'='*50}", flush=True)
        print(f"FINAL: J={tracker.best_score:.8f} n={tracker.best.size} "
              f"time={time.monotonic()-t_start:.0f}s", flush=True)
        print(f"{'='*50}", flush=True)
        return [float(x) for x in tracker.best]
    else:
        return [1.0] * 512
