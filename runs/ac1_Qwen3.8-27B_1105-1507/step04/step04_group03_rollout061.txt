def fast_conv(a):
    """Full self-convolution via FFT."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]

def compute_J(a):
    """Compute J(a) = 2*n*max(conv(a,a))/(sum(a))^2."""
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))

def build_A_sparse(n, a, active_indices=None):
    """Build sparse matrix A with optional row subset."""
    if active_indices is None:
        active_indices = range(2 * n - 1)
    
    rows = []
    cols = []
    vals = []
    for k in active_indices:
        k = int(k)
        j_min = max(0, k - n + 1)
        j_max = min(n - 1, k)
        for j in range(j_min, j_max + 1):
            rows.append(k)
            cols.append(j)
            vals.append(a[k - j])
    return csr_matrix((vals, (rows, cols)), shape=(len(active_indices), n))

def lp_best_response_active(a, M, active_set, history=None):

def nesterov_subgradient(a, max_iters, deadline):
    """Nesterov-accelerated subgradient descent."""
    a = np.maximum(a.astype(np.float64), 1e-10)
    a = a / a.sum()
    n = len(a)
    
    best_M = fast_conv(a).max()
    best_a = a.copy()
    
    y = a.copy()  # Nesterov look-ahead point
    v = np.zeros_like(a)  # velocity
    beta = 0.9
    step_base = 1.0
    
    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        
        # Compute subgradient at y
        b_y = fast_conv(y)
        M = b_y.max()
        tol = 1e-10 * M
        active = np.where(b_y >= M - tol)[0]
        
        grad = np.zeros_like(y)
        for k in active:
            lo = max(0, k - n + 1)
            hi = min(n - 1, k)
            for i in range(lo, hi + 1):
                grad[i] += 2.0 * y[k - i]
        
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-15:
            break
        
        # Adaptive step size
        t = step_base / (gnorm * (it + 1) ** 0.6)
        
        # Nesterov update
        v = beta * v + (1 - beta) * grad / gnorm
        y_new = a - t * v
        y_new = np.maximum(y_new, 0.0)
        s = y_new.sum()
        if s < 1e-10:
            break
        y_new = y_new / s
        
        # Nesterov look-ahead
        t_k = (1 + np.sqrt(1 + 4 * it * it)) / 2
        y = a + (t_k - 1) / (t_k + 1) * (y_new - a)
        y = np.maximum(y, 0)
        y = y / y.sum()
        
        a = y_new
        new_M = fast_conv(a).max()
        if new_M < best_M:
            best_M = new_M
            best_a = a.copy()
        
        # Gradually decrease step size
        step_base *= 0.999
    
    return best_a

def fourier_params_to_seq(params, n, n_harm):
    """Convert Fourier parameters to a non-negative sequence."""
    t = np.arange(n, dtype=np.float64) / n
    a = np.full(n, params[0], dtype=np.float64)
    for k in range(1, n_harm + 1):
        a += params[2 * k - 1] * np.cos(2.0 * np.pi * k * t)
        a += params[2 * k] * np.sin(2.0 * np.pi * k * t)
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-8:
        a = np.ones(n) / n
    else:
        a = a / s
    return a

def seq_to_fourier_params(a, n_harm):
    """Extract Fourier parameters from a sequence."""
    n = len(a)
    a_norm = a / a.sum()
    t = np.arange(n, dtype=np.float64) / n
    params = np.zeros(2 * n_harm + 1)
    params[0] = np.mean(a_norm)
    for k in range(1, n_harm + 1):
        params[2 * k - 1] = 2.0 * np.mean(a_norm * np.cos(2 * np.pi * k * t))
        params[2 * k] = 2.0 * np.mean(a_norm * np.sin(2 * np.pi * k * t))
    return params

def resample_seq(src, dest_n):
    """Resample 1D sequence to new length."""
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    src = np.maximum(src, 0)
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)

"""
Refined hybrid optimizer for minimizing J(a) = 2*n*max(conv(a,a))/sum(a)^2.
Uses active-set LP mirror descent, Nesterov subgradient, multi-length exploration,
Fourier-domain search, and gradient-based polishing.

import numpy as np
from scipy.optimize import linprog, minimize
from scipy.sparse import csr_matrix
import time


def fast_conv(a):
    """Full self-convolution via FFT."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]


def compute_J(a):
    """Compute J(a) = 2*n*max(conv(a,a))/(sum(a))^2."""
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))


def build_A_sparse(n, a, row_indices=None):
    """Build sparse matrix A where A[k,j] = a[k-j] for valid j, subset of rows."""
    if row_indices is None:
        row_indices = np.arange(2 * n - 1)
    rows = []
    cols = []
    vals = []
    for idx, k in enumerate(row_indices):
        k = int(k)
        j_min = max(0, k - n + 1)
        j_max = min(n - 1, k)
        for j in range(j_min, j_max + 1):
            rows.append(idx)
            cols.append(j)
            vals.append(a[k - j])
    return csr_matrix((vals, (rows, cols)), shape=(len(row_indices), n))


def solve_lp_full(n, a, M):
    """Solve full LP: max sum(g) s.t. A@g <= M, g >= 0."""
    A = build_A_sparse(n, a)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    try:
        r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
        if r.success and r.x is not None:
            return np.maximum(r.x, 0), r
    except Exception:
        pass
    return None, None


def solve_lp_active(n, a, M, active_indices):

    A = build_A_sparse(n, a, active_indices)
    c = -np.ones(n)
    b_ub = np.full(len(active_indices), M)
    bounds = [(0, None)] * n
    try:
        r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
        if r.success and r.x is not None:
            return np.maximum(r.x, 0), r
    except Exception:
        pass
    return None, None

            if r is not None and hasattr(r, 'ineqlin') and r.ineqlin is not None:
                duals = np.abs(r.ineqlin)
                new_active = active_set[duals > 1e-5]
                # Always keep near-active constraints
                near_active = np.where(b >= M * 0.8)[0]
                active_set = np.union1d(new_active, near_active)
                if len(active_set) < 10:
                    active_set = np.where(b >= M * 0.5)[0]
        else:
            g, r = solve_lp_full(n, a, M)

        if g is None:
            break

        g_sum = g.sum()
        if g_sum < 1e-10:
            break
        g = g / g_sum

        # Line search
        best_trial_J = current_J
        best_trial_a = None
        t = step_size

        for _ in range(20):
            a_trial = (1.0 - t) * a + t * g
            a_trial = np.maximum(a_trial, 0.0)
            s = a_trial.sum()
            if s < 1e-10:
                t *= 0.5
                continue
            a_trial = a_trial / s
            J_trial = compute_J(a_trial)
            if J_trial < best_trial_J - 1e-14:
                best_trial_J = J_trial
                best_trial_a = a_trial.copy()
                break
            t *= 0.6
            if t < 0.001:
                break

        if best_trial_a is not None:
            improvement = current_J - best_trial_J
            if improvement > 0.0005:
                step_size = min(0.45, step_size * 1.08)
            elif improvement > 0.00005:
                step_size = min(0.45, step_size * 1.01)
            else:
                step_size = max(0.04, step_size * 0.92)

            a = best_trial_a
            current_J = best_trial_J
            if current_J < best_J:
                best_J = current_J
                best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1
            step_size = max(0.04, step_size * 0.93)
            if no_improve > 60:
                break

    return best_a, best_J


def nesterov_subgradient(a, max_iters, deadline):
    """Nesterov-accelerated subgradient descent for max-convolution."""
    a = np.maximum(a.astype(np.float64), 1e-10)
    a = a / a.sum()
    n = len(a)

    best_M = fast_conv(a).max()
    best_a = a.copy()

    v = np.zeros_like(a)
    beta = 0.92
    step_base = 2.0

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        b = fast_conv(a)
        M = b.max()
        tol = 1e-9 * M
        active = np.where(b >= M - tol)[0]

        grad = np.zeros_like(a)
        for k in active:
            k = int(k)
            lo = max(0, k - n + 1)
            hi = min(n - 1, k)
            for j in range(lo, hi + 1):
                grad[j] += 2.0 * a[k - j]

        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-14:
            break

        t = step_base / (gnorm * (it + 1) ** 0.55)
        v = beta * v + (1 - beta) * grad / gnorm
        a_new = a - t * v
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s < 1e-10:
            break
        a_new = a_new / s

        a = a_new
        new_M = fast_conv(a).max()
        if new_M < best_M:
            best_M = new_M
            best_a = a.copy()

        step_base *= 0.9995

    return best_a


def fourier_to_seq(params, n, n_harm):
    """Convert Fourier parameters to a non-negative normalized sequence."""
    t = np.arange(n, dtype=np.float64) / n
    a = np.full(n, params[0], dtype=np.float64)
    for k in range(1, min(n_harm + 1, n // 2 + 1)):
        if 2 * k - 1 < len(params):
            a += params[2 * k - 1] * np.cos(2.0 * np.pi * k * t)
        if 2 * k < len(params):
            a += params[2 * k] * np.sin(2.0 * np.pi * k * t)
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-8:
        return np.ones(n) / n
    return a / s


def seq_to_fourier(a, n_harm):
    """Extract Fourier parameters from a normalized sequence."""
    n = len(a)
    a_norm = a / a.sum()
    t = np.arange(n, dtype=np.float64) / n
    params = np.zeros(2 * n_harm + 1)
    params[0] = np.mean(a_norm)
    for k in range(1, n_harm + 1):
        params[2 * k - 1] = 2.0 * np.mean(a_norm * np.cos(2.0 * np.pi * k * t))
        params[2 * k] = 2.0 * np.mean(a_norm * np.sin(2.0 * np.pi * k * t))
    return params


def resample_seq(src, dest_n):
    """Resample 1D sequence to new length."""
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    src = np.maximum(src, 0)
    if src.sum() < 1e-15:
        return np.ones(dest_n) / dest_n
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    result = np.maximum(np.interp(t_new, t_old, src), 0.0)
    s = result.sum()
    if s < 1e-15:
        return np.ones(dest_n) / dest_n
    return result / s


def generate_diverse_inits(n, rng, parent=None):
    """Generate diverse initial sequences for exploration."""
    inits = []
    t = np.arange(n, dtype=np.float64) / n

    if parent is not None:
        p = resample_seq(parent, n)
        inits.append(p)
        noisy1 = np.maximum(p + rng.normal(0, 0.015, n), 0.01)
        inits.append(noisy1 / noisy1.sum())
        noisy2 = np.maximum(p + rng.normal(0, 0.04, n), 0.01)
        inits.append(noisy2 / noisy2.sum())

    inits.append(np.ones(n) / n)
    inits.append(np.maximum(0.5 + 0.5 * np.cos(2 * np.pi * t), 0.01))
    inits.append(np.maximum(0.5 - 0.4 * np.cos(2 * np.pi * t), 0.01))
    for f in [2, 3, 5, 7, 11]:
        inits.append(np.maximum(1.0 + 0.25 * np.cos(2 * np.pi * f * t), 0.01))
    inits.append(np.exp(-5 * (t - 0.5) ** 2) + 0.15)
    inits.append(np.exp(-8 * (t - 0.3) ** 2) + np.exp(-8 * (t - 0.7) ** 2) + 0.08)

    # Smooth random
    k = max(5, n // 12)
    kernel = np.ones(k) / k
    smooth = np.convolve(rng.uniform(0, 1, n), kernel, mode='same')
    inits.append(np.maximum(smooth + 0.2, 0.01))

    # Random Fourier
    n_harm = min(6, n // 8)
    params = np.concatenate([np.array([1.0]), rng.normal(0, 0.12, 2 * n_harm)])
    inits.append(fourier_to_seq(params, n, n_harm))

    result = []
    for a in inits:
        a = np.maximum(a, 0.001)
        s = a.sum()
        if s > 1e-6:
            result.append(a / s)
    return result


def max_conv_obj(x):
    """Objective for L-BFGS-B: max of self-convolution (scale invariant)."""
    ax = np.maximum(x, 0.0)
    s = ax.sum()
    if s < 1e-8:
        return 1e10
    ax = ax / s
    b = fast_conv(ax)
    return float(b.max())


def structured_perturb(a, rng, ptype):
    """Apply structured perturbation to sequence."""
    n = len(a)
    t = np.arange(n, dtype=np.float64) / n

    if ptype == 'shift':
        s = int(rng.integers(1, max(2, n // 12)))
        a_p = np.roll(a, s)
    elif ptype == 'modulate':
        freq = rng.uniform(1.0, 6.0)
        amp = rng.uniform(0.03, 0.12)
        phase = rng.uniform(0, 2 * np.pi)
        a_p = a * (1 + amp * np.cos(2 * np.pi * freq * t + phase))
    elif ptype == 'smooth':
        k = int(rng.integers(5, max(6, n // 15)))
        kernel = np.ones(k) / k
        a_p = np.convolve(a, kernel, mode='same')
    elif ptype == 'taper':
        alpha = rng.uniform(0.6, 0.9)
        window = np.ones(n)
        edge = max(1, int(n * (1 - alpha) / 2))
        for i in range(edge):
            frac = (i + 1) / (edge + 1)
            window[i] = frac
            window[n - 1 - i] = frac
        a_p = a * window
    else:
        a_p = a + rng.normal(0, 0.02, n)

    a_p = np.maximum(a_p, 0.001)
    s = a_p.sum()
    if s > 1e-8:
        a_p = a_p / s
    return a_p


def propose_candidate(seed=42, budget_s=1500, **kwargs):
    """
    Hybrid optimizer: active-set LP mirror descent, Nesterov subgradient,
    multi-length exploration, Fourier search, structured perturbations,
    and L-BFGS-B polishing.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + float(budget_s) - 30.0
    t_start = time.monotonic()

    global_best_a = None
    global_best_score = float('inf')

    def track(a):
        nonlocal global_best_a, global_best_score
        a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
        if a.size == 0 or a.sum() < 0.01:
            return
        score = float(evaluate_sequence(a.tolist()))
        if np.isfinite(score) and score < global_best_score:
            global_best_a = a.copy()
            global_best_score = score
            elapsed = time.monotonic() - t_start
            print(f"  [NEW BEST] J={score:.8f} n={a.size} t={elapsed:.0f}s", flush=True)

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        if p.ndim == 1 and p.size > 0 and np.all(np.isfinite(p)):
            parent = np.maximum(p, 0.0)
            if parent.sum() > 1e-10:
                parent = parent / parent.sum()
            track(parent)
            print(f"Parent: n={parent.size}, J={global_best_score:.6f}", flush=True)
    except (NameError, AttributeError):
        pass

    if parent is None:
        parent = np.ones(512) / 512
        track(parent)

    current_best = parent.copy()
    current_best_J = compute_J(current_best)

    # === Phase 1: LP on parent ===
    print(f"\n{'='*50}", flush=True)
    print(f"Phase 1: LP mirror descent (n={current_best.size})", flush=True)
    print(f"{'='*50}", flush=True)

    phase1_dl = min(deadline - 700, time.monotonic() + 100)
    a1, J1 = lp_mirror_descent_v2(current_best.copy(), 500, phase1_dl, use_active=True)
    track(a1)
    print(f"  J={J1:.8f} ({time.monotonic()-t_start:.1f}s)", flush=True)
    if J1 < current_best_J:
        current_best = a1.copy()
        current_best_J = J1

    # === Phase 2: Multi-length exploration ===
    if time.monotonic() < deadline - 500:
        print(f"\n{'='*50}", flush=True)
        print(f"Phase 2: Multi-length exploration", flush=True)
        print(f"{'='*50}", flush=True)

        base_n = current_best.size
        lengths = set()
        lengths.add(base_n)
        for nice_n in [192, 256, 320, 384, 448, 512, 576, 640, 704, 768, 832, 896, 1024]:
            if abs(nice_n - base_n) <= base_n * 0.5:
                lengths.add(nice_n)

        lengths = sorted([l for l in lengths if 128 <= l <= 1200 and l != base_n])

        time_budget = deadline - 450 - time.monotonic()
        per_length = max(15, min(50, time_budget / max(1, len(lengths))))

        for n_try in lengths:
            if time.monotonic() > deadline - 420:
                break
            a_try = resample_seq(current_best, n_try)
            a_try = np.maximum(a_try, 1e-10)
            a_try = a_try / a_try.sum()
            dl = time.monotonic() + per_length
            a_try, J_try = lp_mirror_descent_v2(a_try, 300, dl, use_active=True)
            track(a_try)
            print(f"  n={n_try}: J={J_try:.8f} (t={time.monotonic()-t_start:.0f}s)", flush=True)
            if J_try < current_best_J:
                current_best = a_try.copy()
                current_best_J = J_try

    # === Phase 3: Diverse initializations + LP ===
    if time.monotonic() < deadline - 300:
        print(f"\n{'='*50}", flush=True)
        print(f"Phase 3: Diverse inits (n={current_best.size})", flush=True)
        print(f"{'='*50}", flush=True)

        n_div = current_best.size
        inits = generate_diverse_inits(n_div, rng, parent)

        time_budget = deadline - 250 - time.monotonic()
        per_init = max(8, min(30, time_budget / max(1, len(inits))))

        for i, a_init in enumerate(inits):
            if time.monotonic() > deadline - 230:
                break
            dl = time.monotonic() + per_init
            a_res, J_res = lp_mirror_descent_v2(a_init.copy(), 200, dl, use_active=True)
            track(a_res)
            if J_res < current_best_J:
                current_best = a_res.copy()
                current_best_J = J_res
                print(f"  init[{i}]: J={J_res:.8f} (t={time.monotonic()-t_start:.0f}s)", flush=True)

    # === Phase 4: Nesterov subgradient ===
    if time.monotonic() < deadline - 180:
        print(f"\n{'='*50}", flush=True)
        print(f"Phase 4: Nesterov subgradient (n={current_best.size})", flush=True)
        print(f"{'='*50}", flush=True)

        dl = min(deadline - 130, time.monotonic() + 60)
        a_sub = nesterov_subgradient(current_best, 3000, dl)
        track(a_sub)
        J_sub = compute_J(a_sub)
        if J_sub < current_best_J:
            current_best = a_sub.copy()
            current_best_J = J_sub
        print(f"  J={J_sub:.8f} (t={time.monotonic()-t_start:.0f}s)", flush=True)

    # === Phase 5: Structured perturbations ===
    if time.monotonic() < deadline - 120:
        print(f"\n{'='*50}", flush=True)
        print(f"Phase 5: Structured perturb + LP", flush=True)
        print(f"{'='*50}", flush=True)

        perturb_types = ['shift', 'modulate', 'smooth', 'taper', 'modulate', 'shift', 'noise', 'taper']
        time_budget = deadline - 80 - time.monotonic()
        per_pert = max(8, min(25, time_budget / len(perturb_types)))

        for ptype in perturb_types:
            if time.monotonic() > deadline - 70:
                break
            a_pert = structured_perturb(current_best, rng, ptype)
            dl = time.monotonic() + per_pert
            a_pert, J_pert = lp_mirror_descent_v2(a_pert, 150, dl, use_active=True)
            track(a_pert)
            if J_pert < current_best_J:
                current_best = a_pert.copy()
                current_best_J = J_pert
                print(f"  [{ptype}]: J={J_pert:.8f}", flush=True)

    # === Phase 6: Fourier exploration ===
    if time.monotonic() < deadline - 80:
        print(f"\n{'='*50}", flush=True)
        print(f"Phase 6: Fourier exploration (n={current_best.size})", flush=True)
        print(f"{'='*50}", flush=True)

        n_four = current_best.size
        n_harm = min(8, n_four // 10)

        for restart in range(4):
            if time.monotonic() > deadline - 50:
                break
            if restart == 0:
                params = seq_to_fourier(current_best, n_harm)
            else:
                params = seq_to_fourier(current_best, n_harm) + rng.normal(0, 0.04, 2 * n_harm + 1)
            params[0] = max(params[0], 0.3)
            a_four = fourier_to_seq(params, n_four, n_harm)

            dl = min(deadline - 40, time.monotonic() + 25)
            a_four, J_four = lp_mirror_descent_v2(a_four, 200, dl, use_active=True)
            track(a_four)
            if J_four < current_best_J:
                current_best = a_four.copy()
                current_best_J = J_four
            print(f"  Fourier r{restart}: J={J_four:.8f}", flush=True)

    # === Phase 7: Long LP refinement ===
    if time.monotonic() < deadline - 30:
        print(f"\n{'='*50}", flush=True)
        print(f"Phase 7: Long LP (n={current_best.size})", flush=True)
        print(f"{'='*50}", flush=True)

        a_long = current_best.copy()
        a_long = np.maximum(a_long, 1e-10)
        a_long = a_long / a_long.sum()
        a_long, J_long = lp_mirror_descent_v2(a_long, 20000, deadline - 15, use_active=True)
        track(a_long)
        if J_long < current_best_J:
            current_best = a_long.copy()
            current_best_J = J_long
        print(f"  J={J_long:.8f} (t={time.monotonic()-t_start:.0f}s)", flush=True)

    # === Phase 8: L-BFGS-B polish ===
    if time.monotonic() < deadline - 5 and global_best_a is not None:
        print(f"\n{'='*50}", flush=True)
        print(f"Phase 8: L-BFGS-B polish (n={global_best_a.size})", flush=True)
        print(f"{'='*50}", flush=True)

        try:
            x0 = global_best_a.astype(np.float64).copy()
            res = minimize(max_conv_obj, x0,
                           method='L-BFGS-B',
                           bounds=[(0.0, None)] * len(x0),
                           options={'maxiter': 60, 'max_fun': 15000, 'ftol': 1e-16})
            if res.fun < 1e10:
                track(res.x)
                J_res = compute_J(res.x)
                if J_res < current_best_J:
                    current_best = res.x.copy()
                    current_best_J = J_res
        except Exception as e:
            print(f"  L-BFGS-B error: {e}", flush=True)
        print(f"  Done (t={time.monotonic()-t_start:.0f}s)", flush=True)

    # Final
    if global_best_a is not None:
        final_a = global_best_a
        final_score = global_best_score
    else:
        final_a = current_best
        final_score = compute_J(current_best)

    print(f"\n{'='*50}", flush=True)
    print(f"FINAL: J={final_score:.8f} n={final_a.size} time={time.monotonic()-t_start:.0f}s", flush=True)
    print(f"{'='*50}", flush=True)

    return [float(x) for x in final_a]
