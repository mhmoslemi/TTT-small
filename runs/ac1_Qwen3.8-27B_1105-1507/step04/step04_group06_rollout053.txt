def fast_conv(a):
    """Full self-convolution via FFT, O(n log n)."""
    n = len(a)
    s = 1
    while s < 2 * n - 1:
        s <<= 1
    fa = np.fft.rfft(a, s)
    return np.fft.irfft(fa * fa, s)[:2 * n - 1]


def compute_J(a):
    """Compute J(a) = 2*n*max(conv(a,a))/(sum(a))^2."""
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))


def build_A_for_k(a, k_indices):
    """Build sparse matrix rows for selected k values of (a*g) constraints."""
    n = len(a)
    rows = []
    cols = []
    for row_idx, k in enumerate(k_indices):
        jl = max(0, k - n + 1)
        jh = min(n - 1, k)
        for j in range(jl, jh + 1):
            rows.append(row_idx)
            cols.append(j)
    if len(rows) == 0:
        return None
    n_constraints = len(k_indices)
    # vals[i] = a[k - j] for the (row, col) pair
    vs = []
    for row_idx, k in enumerate(k_indices):
        jl = max(0, k - n + 1)
        jh = min(n - 1, k)
        for j in range(jl, jh + 1):
            vs.append(a[k - j])
    vs = np.array(vs)
    A = csr_matrix((vs, (rows, cols)), shape=(n_constraints, n))
    return A

def smooth_max_and_grad(conv_val, beta):
    """Smooth max of conv_val via log-sum-exp, return value and gradient weights."""
    b_max = float(conv_val.max())
    shifted = beta * (conv_val - b_max)
    shifted = np.clip(shifted, -500, 500)
    exp_shifted = np.exp(shifted)
    Z = float(exp_shifted.sum())
    if Z < 1e-300:
        p = np.zeros(len(conv_val))
        p[np.argmax(conv_val)] = 1.0
    else:
        p = exp_shifted / Z
    g_val = (b_max + np.log(max(Z, 1e-300))) / beta
    return g_val, p


def F_and_grad(a, beta):

    n = len(a)
    s = float(a.sum())
    if s < 1e-10:
        return 1e10, np.zeros(n)
    conv_a = fast_conv(a)
    g_val, p = smooth_max_and_grad(conv_a, beta)
    F_val = 2.0 * n * g_val / (s * s)

    # d/d a_j [sum_k p_k * conv(a,a)_k] = sum_k p_k * d conv_k / d a_j
    # d conv_k / d a_j = sum_i: i+j=k (with bounds) a_i = (a * p_reversed)[j] essentially
    # conv_k = sum_i a[i]*a[k-i], d conv_k / d a[j] = sum_{i: i+j=k, valid} a[i]
    # = (a flipped and convolved with p) at position j
    
    # p has length 2n-1, index k corresponds to conv_k
    # d/d a[j] sum_k p[k] * conv_k = sum_k p[k] * sum_{i: i+j=k} a[i]
    # = sum_k p[k] * a[k-j] (where 0<=k-j<n and 0<=k<2n-1)
    # = sum_{i: 0<=i<n} a[i] * p[i+j] (where 0<=i+j<2n-1)
    # This is the cross-correlation of a with p
    
    # More efficiently: pad a and convolve
    grad_raw = np.convolve(a, p, mode='full')

    # Let me use a different approach
    # d/d a[j] of conv_k = a[k-j] if 0 <= k-j < n, else 0
    # So grad[j] = sum_{k: max(0,j) <= k <= min(2n-2, j+n-1)} p[k] * a[k-j]
    # = sum_{i: 0<=i<n} p[i+j] * a[i] (where 0<=i+j<2n-1)

    # Let's compute it as: for each j, sum over valid i of a[i]*p[i+j]

def lbfgs_optimize(a0, deadline, n_step=1024):

def resample_seq(src, dest_n):
    """Resample 1D sequence to new length via linear interpolation."""
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    result = np.interp(t_new, t_old, src)
    return np.maximum(result, 0.0)

                best_a = a.copy()
                tracker.update(a)
            stall_count = 0
        else:
            # Try smaller steps
            improved = False
            for t_try in [0.05, 0.02, 0.01, 0.005]:
                a_try = np.maximum((1.0 - t_try) * a + t_try * g, 0.0)
                s_try = a_try.sum()
                if s_try < 1e-10:
                    continue
                a_try = a_try / s_try
                J_try = compute_J(a_try)
                if J_try < current_J - 1e-14:
                    a = a_try
                    current_J = J_try
                    no_improve_count = 0
                    if J_try < best_J:
                        best_J = J_try
                        best_a = a.copy()
                        tracker.update(a)
                    improved = True
                    break
            if not improved:
                no_improve_count += 1
                if no_improve_count >= 60:
                    break
        
        # Periodic reporting
        if iter_count % 100 == 0:
            elapsed = time.monotonic() - tracker.t_start
            print(f"    LP iter {iter_count}: J={current_J:.8f}, best={best_J:.8f}, t={elapsed:.0f}s", flush=True)
    
    return best_a, best_J


def generate_initializations(n, rng, parent=None):
    """Generate diverse initial sequences of length n."""
    inits = []
    t_arr = np.linspace(0, 1, n, endpoint=False)
    
    if parent is not None and parent.size > 0:
        inits.append(resample_seq(parent, n))
        p = resample_seq(parent, n)
        p = p + rng.normal(0, 0.03, n) * p
        inits.append(np.maximum(p, 0.01))
        p2 = p + rng.normal(0, 0.1, n) * p
        inits.append(np.maximum(p2, 0.01))
    
    # Uniform
    inits.append(np.ones(n))
    
    # Cosine families with various frequencies
    for freq in [1, 2, 3, 4, 5, 6, 7, 8]:
        inits.append(1.0 + 0.3 * np.cos(2.0 * np.pi * freq * t_arr))
    
    # Two-component
    inits.append(1.0 + 0.25 * np.cos(2.0 * np.pi * 3 * t_arr) + 0.15 * np.cos(2.0 * np.pi * 7 * t_arr))
    
    # Gaussian bump
    inits.append(np.exp(-3.0 * (t_arr - 0.5) ** 2) + 0.1)
    
    # Power law
    inits.append((t_arr + 0.05) ** 0.7 + 0.05)
    
    # Exponential decay
    inits.append(np.exp(-2.0 * t_arr) + 0.1)
    
    # Triangular
    inits.append(1.0 - 0.3 * np.abs(2.0 * ((t_arr + 0.25) % 0.5) - 0.5))
    
    # Random smooth
    noise = rng.normal(0, 1, n)
    kernel_size = max(3, n // 32)
    kernel = np.ones(kernel_size) / kernel_size
    smooth_noise = np.convolve(noise, kernel, mode='same')
    inits.append(1.0 + 0.2 * smooth_noise / (np.max(np.abs(smooth_noise)) + 1e-10))
    
    for init in inits:
        init = np.maximum(init, 0.01)
        init = init / init.sum()
    
    return inits


def propose_candidate(seed=42, budget_s=1500, **kwargs):
    """
    Multi-phase optimization for minimizing J(a) = 2n*max(conv(a,a))/(sum(a))^2.
    
    Phase 1: L-BFGS-B with smooth objective (fast basin finding)
    Phase 2: LP-based optimization with relaxed M and top-k targeting (main)
    Phase 3: Multi-restart with diverse perturbations
    Phase 4: Final LP polish with small steps
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + float(budget_s) - 30.0
    tracker = ScoreTracker()
    
    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        if p.ndim == 1 and p.size > 0 and np.all(np.isfinite(p)):
            parent = np.maximum(p, 0.0)
            tracker.update(parent)
            print(f"Parent loaded: n={parent.size}, J={tracker.best_score:.8f}", flush=True)
    except (NameError, AttributeError):
        pass
    
    if parent is None:
        parent = np.ones(1024)
        tracker.update(parent)
        print(f"Using default parent: n=1024, J={tracker.best_score:.8f}", flush=True)
    
    # === Phase 1: L-BFGS-B (fast, ~80s) ===
    t0 = time.monotonic()
    phase1_budget = 80.0
    deadline_p1 = time.monotonic() + phase1_budget
    print(f"\n{'=' * 50} Phase 1: L-BFGS-B (budget {phase1_budget:.0f}s) {'=' * 50}", flush=True)
    
    n_main = parent.size if parent.size in [512, 1024] else 1024
    if n_main not in [256, 512, 1024]:
        n_main = 1024
    
    inits = generate_initializations(n_main, rng, parent)
    max_inits = 5
    
    for idx, a0 in enumerate(inits[:max_inits]):
        if time.monotonic() > deadline_p1:
            break
        t_start = time.monotonic()
        a_opt = lbfgs_optimize(a0, deadline_p1)
        if a_opt is not None and a_opt.sum() > 0.01:
            a_opt = a_opt / a_opt.sum()
            tracker.update(a_opt)
        print(f"  L-BFGS-B init {idx}: {time.monotonic()-t_start:.1f}s", flush=True)
    
    tracker.report("Phase 1 complete")
    
    # === Phase 2: Main LP optimization (~800s) ===
    if tracker.best is not None and time.monotonic() < deadline:
        t0 = time.monotonic()
        n_best = tracker.best.size
        phase2_budget = min(800.0, deadline - time.monotonic())
        deadline_p2 = time.monotonic() + phase2_budget
        print(f"\n{'=' * 50} Phase 2: LP Main n={n_best} (budget {phase2_budget:.0f}s) {'=' * 50}", flush=True)
        
        # Primary run
        a_opt, J_opt = lp_main_phase(tracker.best, min(time.monotonic() + phase2_budget * 0.6, deadline_p2),
                                       tracker, n_line_evals=22, use_topk=True, topk=5)
        tracker.update(a_opt)
        
        # Continue with full constraints (more expensive but better)
        remaining_p2 = deadline_p2 - time.monotonic()
        if remaining_p2 > 30:
            a_opt2, J_opt2 = lp_main_phase(tracker.best, time.monotonic() + remaining_p2 * 0.7,
                                            tracker, n_line_evals=20, use_topk=False, topk=5)
            tracker.update(a_opt2)
        
        # Try different lengths
        for n_try in [512, 1024, 2048]:
            if time.monotonic() > deadline_p2 or n_try == n_best:
                continue
            a_res = resample_seq(tracker.best, n_try)
            a_res = a_res / max(a_res.sum(), 1e-10)
            budget_here = min(120.0, deadline_p2 - time.monotonic())
            if budget_here < 20:
                break
            a_res, _ = lp_main_phase(a_res, time.monotonic() + budget_here,
                                      tracker, n_line_evals=18, use_topk=True, topk=3)
            tracker.update(a_res)
        
        tracker.report(f"Phase 2: {time.monotonic()-t0:.1f}s")
    
    # === Phase 3: Multi-restart (~250s) ===
    if tracker.best is not None and time.monotonic() < deadline:
        t0 = time.monotonic()
        phase3_budget = min(250.0, deadline - time.monotonic())
        deadline_p3 = time.monotonic() + phase3_budget
        print(f"\n{'=' * 50} Phase 3: Restarts (budget {phase3_budget:.0f}s) {'=' * 50}", flush=True)
        
        strategies = [
            ('gaussian', 0.05),
            ('gaussian', 0.02),
            ('gaussian', 0.10),
            ('smooth', 0.08),
            ('spike', 5),
            ('gaussian', 0.01),
            ('uniform', 0.08),
            ('smooth', 0.04),
            ('gaussian', 0.03),
            ('spike', 10),
        ]
        
        n_restarts = min(len(strategies) * 2, int(phase3_budget / 20))
        attempt = 0
        for i in range(n_restarts):
            if time.monotonic() > deadline_p3:
                break
            pert_type, param = strategies[i % len(strategies)]
            attempt += 1
            a = tracker.best.copy().astype(np.float64)
            n = len(a)
            
            if pert_type == 'gaussian':
                a = a + rng.normal(0, param, n) * a
            elif pert_type == 'uniform':
                a = a + rng.uniform(-param, param, n) * a
            elif pert_type == 'spike':
                k = min(param, n // 4)
                idx = rng.choice(n, size=k, replace=False)
                a[idx] *= rng.uniform(0.3, 3.0, k)
            elif pert_type == 'smooth':
                noise = rng.normal(0, param, n)
                kernel_size = max(3, n // 64)
                kernel = np.ones(kernel_size) / kernel_size
                noise = np.convolve(noise, kernel, mode='same')
                a = a + noise * a
            
            a = np.maximum(a, 1e-5)
            a = a / a.sum()
            
            budget_here = min(25.0, deadline_p3 - time.monotonic())
            if budget_here < 5:
                break
            a_opt, _ = lp_main_phase(a, time.monotonic() + budget_here,
                                      tracker, n_line_evals=15, use_topk=True, topk=3)
            tracker.update(a_opt)
        
        tracker.report(f"Phase 3: {time.monotonic()-t0:.1f}s, {attempt} restarts")
    
    # === Phase 4: Final polish (~200s) ===
    if tracker.best is not None and time.monotonic() < deadline:
        t0 = time.monotonic()
        remaining = min(200.0, deadline - time.monotonic())
        deadline_p4 = time.monotonic() + remaining
        print(f"\n{'=' * 50} Phase 4: Final Polish (budget {remaining:.0f}s) {'=' * 50}", flush=True)
        
        # Run LP with more careful line search
        a_opt, _ = lp_main_phase(tracker.best, deadline_p4,
                                  tracker, n_line_evals=25, use_topk=True, topk=8)
        tracker.update(a_opt)
        
        # Second pass
        if time.monotonic() < deadline_p4:
            a_opt2, _ = lp_main_phase(tracker.best, deadline_p4,
                                       tracker, n_line_evals=25, use_topk=False, topk=5)
            tracker.update(a_opt2)
        
        # Final L-BFGS-B polish
        if time.monotonic() < deadline_p4:
            a_lbfgs = lbfgs_optimize(tracker.best, deadline_p4)
            tracker.update(a_lbfgs)
        
        tracker.report(f"Phase 4: {time.monotonic()-t0:.1f}s")
    
    # === Phase 5: Try adjacent lengths (remaining time) ===
    if tracker.best is not None and time.monotonic() < deadline:
        t0 = time.monotonic()
        n_best = tracker.best.size
        print(f"\n{'=' * 50} Phase 5: Length Exploration {'=' * 50}", flush=True)
        
        for n_try in [n_best - 64, n_best - 32, n_best + 32, n_best + 64,
                       384, 512, 640, 768, 1024, 1280, 1536]:
            if n_try < 64 or n_try > 2048 or time.monotonic() > deadline:
                continue
            if abs(n_try - n_best) < 16:
                continue
            budget_here = min(30.0, deadline - time.monotonic())
            if budget_here < 10:
                break
            a_res = resample_seq(tracker.best, n_try)
            a_res = a_res / max(a_res.sum(), 1e-10)
            a_opt, _ = lp_main_phase(a_res, time.monotonic() + budget_here,
                                      tracker, n_line_evals=15, use_topk=True, topk=3)
            tracker.update(a_opt)
        
        tracker.report(f"Phase 5: {time.monotonic()-t0:.1f}s")
    
    # Fallback
    if tracker.best is None:
        tracker.best = np.ones(512)
        tracker.best_score = compute_J(tracker.best)
        print(f"  Using fallback: J={tracker.best_score:.8f}", flush=True)
    
    print(f"\n{'=' * 50} FINAL RESULT {'=' * 50}", flush=True)
    print(f"J={tracker.best_score:.8f} n={tracker.best.size} "
          f"time={time.monotonic()-tracker.t_start:.1f}s", flush=True)
    
    result = np.clip(tracker.best, 0.0, 1000.0)
    if result.sum() < 0.01:
        result = np.ones(128)
    
    return [float(x) for x in result]
