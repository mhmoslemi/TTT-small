"""
Hybrid multiplicative gradient descent + LP alternating optimization for minimizing
2*n*max(conv(a,a))/(sum(a))^2. Uses multiplicative (mirror descent) updates with
adaptive temperature annealing, combined with LP-based direction finding and
perturbation restarts. Explores multiple sequence lengths and uses cross-seeding.

import numpy as np
from scipy.optimize import linprog
from scipy.fft import rfft, irfft
import time


def compute_conv(a):
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    conv = irfft(fa * fa, size)
    return conv[:2 * n - 1]


def compute_obj(a):
    n = len(a)
    if n == 0:
        return np.inf
    conv = compute_conv(a)
    max_b = float(np.max(conv))
    sum_a = float(np.sum(a))
    if sum_a < 0.01:
        return np.inf
    return 2.0 * n * max_b / (sum_a * sum_a)


def fft_convolve(a, b):
    n, m = len(a), len(b)
    size = 1
    while size < n + m - 1:
        size <<= 1
    fa = rfft(a, size)
    fb = rfft(b, size)
    return irfft(fa * fb, size)[:n + m - 1]


def softmax(z):
    z = z - np.max(z)
    e = np.exp(z)
    return e / np.sum(e)


def build_toeplitz(a):
    n = len(a)
    A = np.zeros((2 * n - 1, n), dtype=np.float64)
    for j in range(n):
        k_start = j
        k_end = min(j + n, 2 * n - 1)
        A[k_start:k_end, j] = a[:k_end - k_start]
    return A


def solve_lp_direction(a, M, time_limit=5):
    """Solve LP: max sum(g) s.t. (a*g)[k] <= M for all k, g >= 0. Returns normalized g."""
    n = len(a)
    if n > 1500:
        return None
    A = build_toeplitz(a)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': time_limit})
        if res.status == 0:
            g = np.maximum(res.x, 0)
            s = float(np.sum(g))
            if s > 1e-10:
                return g / s
    except Exception:
        pass
    return None


def solve_lp_minpeak(a, time_limit=5):
    """Solve LP: min M s.t. (a*g)[k] <= M for all k, sum(g)=1, g>=0."""
    n = len(a)
    if n > 1500:
        return None, None
    A = build_toeplitz(a)
    c = np.zeros(n + 1)
    c[-1] = 1.0
    A_ub = np.zeros((2 * n - 1, n + 1))
    A_ub[:, :n] = A
    A_ub[:, n] = -1.0
    b_ub = np.zeros(2 * n - 1)
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])
    bounds = [(0.0, None)] * n + [(0.0, None)]
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': time_limit})
        if res.status == 0:
            g = np.maximum(res.x[:n], 0)
            s = np.sum(g)
            if s > 1e-10:
                g = g / s
                M_opt = float(res.x[n])
                return g, M_opt
    except Exception:
        pass
    return None, None


def multiplicative_grad_descent(a, tau_schedule, eta_schedule, max_iters, time_budget,
                                use_lp=True, lp_interval=50):
    """
    Multiplicative gradient descent with soft-max objective.
    a is on the simplex (sum=1, a>=0).
    tau_schedule: list or array of tau values to anneal through
    eta_schedule: list or array of step sizes
    """
    n = len(a)
    a = a.copy().astype(np.float64)
    s = np.sum(a)
    if s < 1e-10:
        a = np.ones(n) / n
    else:
        a = a / s
    a = np.maximum(a, 1e-30)

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    n_tau = len(tau_schedule)
    n_eta = len(eta_schedule)
    iters_per_tau = max(1, max_iters // n_tau)

    for ti in range(n_tau):
        tau = tau_schedule[ti]
        for it in range(iters_per_tau):
            if time.monotonic() - start > time_budget:
                return best_a, best_obj

            eta = eta_schedule[min(it // max(1, n_eta / n_tau), n_eta - 1)] if n_eta > 1 else eta_schedule[0]

            conv = compute_conv(a)
            max_conv = float(np.max(conv))
            shifted = np.clip((conv - max_conv) / tau, -500.0, 500.0)
            exp_conv = np.exp(shifted)
            Z = float(np.sum(exp_conv))
            if Z < 1e-300:
                idx = int(np.argmax(conv))
                w = np.zeros(len(conv))
                w[idx] = 1.0
            else:
                w = exp_conv / Z

            w_rev = w[::-1]
            conv_fw = fft_convolve(a, w_rev)
            grad_a = 2.0 * conv_fw[n - 1:2 * n - 1][::-1]

            mean_grad = float(np.mean(grad_a))
            centered = grad_a - mean_grad

            log_update = -eta * centered
            log_update = np.clip(log_update, -50.0, 50.0)
            a = a * np.exp(log_update)
            a = np.maximum(a, 1e-30)
            s = np.sum(a)
            a = a / s

            if it % 50 == 0 or it == iters_per_tau - 1:
                obj = compute_obj(a)
                if obj < best_obj:
                    best_obj = obj
                    best_a = a.copy()

            if use_lp and n <= 800 and it % lp_interval == lp_interval - 1:
                M = float(np.max(compute_conv(a)))
                g = solve_lp_direction(a, M, time_limit=3)
                if g is not None:
                    for t_blend in [0.3, 0.5, 0.7, 1.0]:
                        a_try = (1.0 - t_blend) * a + t_blend * g
                        a_try = np.maximum(a_try, 1e-30)
                        s_try = np.sum(a_try)
                        if s_try > 1e-10:
                            a_try = a_try / s_try
                            obj_try = compute_obj(a_try)
                            if obj_try < best_obj:
                                best_obj = obj_try
                                best_a = a_try.copy()
                                a = a_try.copy()

    obj = compute_obj(a)
    if obj < best_obj:
        best_obj = obj
        best_a = a.copy()

    return best_a, best_obj


def lbfgs_polish(a, tau_list, max_iter=200, time_budget=30.0):
    """L-BFGS-B polish with softmax parametrization."""
    from scipy.optimize import minimize

    n = len(a)
    a = a.copy().astype(np.float64)
    s = np.sum(a)
    if s < 1e-10:
        a = np.ones(n) / n
    else:
        a = a / s
    a = np.maximum(a, 1e-30)
    z = np.log(a)

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    for tau in tau_list:
        if time.monotonic() - start > time_budget:
            break

        def func_and_grad(z_val):
            a_s = softmax(z_val)
            conv = compute_conv(a_s)
            cmax = float(np.max(conv))
            shifted = np.clip((conv - cmax) / tau, -500.0, 500.0)
            exp_conv = np.exp(shifted)
            Z = float(np.sum(exp_conv))
            soft_max_val = cmax + tau * np.log(Z)
            w = exp_conv / Z

            w_rev = w[::-1]
            conv_fw = fft_convolve(a_s, w_rev)
            grad_a = 2.0 * conv_fw[n - 1:2 * n - 1][::-1]
            mean_grad = float(np.dot(a_s, grad_a))
            grad_z = a_s * (grad_a - mean_grad)
            return soft_max_val, grad_z

        try:
            result = minimize(func_and_grad, z, jac=True, method='L-BFGS-B',
                              options={'maxiter': max_iter, 'ftol': 1e-15, 'gtol': 1e-10})
            z_new = result.x
            a_new = softmax(z_new)
            a_new = np.maximum(a_new, 1e-30)
            s_new = np.sum(a_new)
            if s_new > 1e-10:
                a_new = a_new / s_new
                obj = compute_obj(a_new)
                if obj < best_obj:
                    best_obj = obj
                    best_a = a_new.copy()
                    z = z_new
        except Exception:
            pass

    return best_a, best_obj


def alternating_lp_optimize(a, max_iters=40, time_budget=30.0):
    """Alternating LP with line search."""
    n = len(a)
    if n > 800:
        return a
    a = a.copy().astype(np.float64)
    s = np.sum(a)
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    stall = 0

    for it in range(max_iters):
        if time.monotonic() - start > time_budget:
            break

        conv = compute_conv(a)
        M = float(np.max(conv))

        g = solve_lp_direction(a, M, time_limit=5)
        if g is None:
            break

        improved = False
        for t in [0.1, 0.2, 0.3, 0.5, 0.7, 1.0]:
            a_try = (1.0 - t) * a + t * g
            a_try = np.maximum(a_try, 1e-30)
            s_try = np.sum(a_try)
            if s_try < 1e-10:
                continue
            a_try = a_try / s_try
            obj_try = compute_obj(a_try)
            if obj_try < best_obj - 1e-12:
                best_obj = obj_try
                best_a = a_try.copy()
                a = a_try.copy()
                improved = True
                break

        if improved:
            stall = 0
        else:
            stall += 1
            if stall > 10:
                rng = np.random.default_rng(it * 1009 + 777)
                a = a * np.exp(rng.normal(0, 0.05, n))
                a = np.maximum(a, 1e-30)
                a = a / np.sum(a)
                stall = 0

    return best_a


def resample_sequence(a, new_n):
    n = len(a)
    if n == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    return np.interp(x_new, x_old, a)

                        a_try = (1 - t) * a_cand + t * g_lp
                        a_try = np.maximum(a_try, 1e-30)
                        s_try = np.sum(a_try)
                        if s_try > 1e-10:
                            a_try = a_try / s_try
                            obj_try = compute_obj(a_try)
                            if obj_try < best_score:
                                best_score = obj_try
                                best_seq = a_try.copy()
                                a_cand = a_try.copy()
                                print(f"    r{round_num} MinLP t={t}: {best_score:.6f}")

    # Phase 3: Intensive final polish on global best
    if best_seq is not None and time.monotonic() < deadline - 20:
        remaining = deadline - time.monotonic()
        n_best = len(best_seq)
        print(f"\n=== FINAL POLISH: n={n_best}, {remaining:.0f}s ===")

        a_cur = best_seq.copy()
        for final_round in range(6):
            if time.monotonic() > deadline - 10:
                break
            remaining = deadline - time.monotonic()
            if remaining < 15:
                break

            # Ultra-fine multiplicative GD
            tau_ultra = np.geomspace(0.05 / n_best, 0.005 / n_best, 3)
            a_new, obj_new = multiplicative_grad_descent(
                a_cur, tau_ultra, [0.05, 0.02, 0.01],
                max_iters=3000, time_budget=min(remaining * 0.2, 15.0),
                use_lp=False
            )
            if obj_new < best_score:
                best_score = obj_new
                best_seq = a_new.copy()
                a_cur = a_new.copy()
                print(f"  Final MG {final_round}: {best_score:.6f}")

            # L-BFGS-B ultra-fine
            remaining = deadline - time.monotonic()
            if remaining > 15:
                a_new, obj_new = lbfgs_polish(
                    a_cur, [0.05 / n_best, 0.01 / n_best],
                    max_iter=500, time_budget=min(remaining * 0.2, 15.0)
                )
                if obj_new < best_score:
                    best_score = obj_new
                    best_seq = a_new.copy()
                    a_cur = a_new.copy()
                    print(f"  Final LBFGS {final_round}: {best_score:.6f}")

            # LP
            remaining = deadline - time.monotonic()
            if remaining > 20 and n_best <= 700:
                a_new = alternating_lp_optimize(
                    a_cur, max_iters=25, time_budget=min(remaining * 0.25, 20.0)
                )
                obj_new = compute_obj(a_new)
                if obj_new < best_score:
                    best_score = obj_new
                    best_seq = a_new.copy()
                    a_cur = a_new.copy()
                    print(f"  Final LP {final_round}: {best_score:.6f}")

            # Perturbation + re-optimize
            remaining = deadline - time.monotonic()
            if remaining > 20:
                rng = np.random.default_rng(9999 + final_round)
                for ptry in range(3):
                    if time.monotonic() > deadline - 15:
                        break
                    scale = rng.uniform(0.01, 0.06)
                    a_pert = a_cur * np.exp(rng.normal(0, scale, n_best))
                    a_pert = np.maximum(a_pert, 1e-30)
                    a_pert = a_pert / np.sum(a_pert)
                    a_pert, obj_pert = multiplicative_grad_descent(
                        a_pert, np.geomspace(0.1 / n_best, 0.01 / n_best, 3),
                        [0.1, 0.05, 0.02], max_iters=1000,
                        time_budget=min(remaining * 0.08, 8.0), use_lp=False
                    )
                    if obj_pert < best_score:
                        best_score = obj_pert
                        best_seq = a_pert.copy()
                        a_cur = a_pert.copy()
                        print(f"  Final Perturb {ptry}: {best_score:.6f}")

    # Final validation
    if best_seq is None:
        best_seq = np.ones(200)
        best_score = compute_obj(best_seq)

    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    val = evaluate_sequence(result)
    if val < np.inf:
        best_score = val
    else:
        result = [1.0] * 100
        best_score = 2.0

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_score:.6f}, time={elapsed:.1f}s ===")

    return result
