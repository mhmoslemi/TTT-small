```python
from scipy.optimize import linprog, minimize, minimize_scalar
from scipy.sparse import csr_matrix
import time

def fast_conv(a):
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]

def compute_J(a):
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))

def build_A_sparse(n, a):
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        j_min = max(0, k - n + 1)
        j_max = min(n - 1, k)
        for j in range(j_min, j_max + 1):
            rows.append(k)
            cols.append(j)
            vals.append(a[k - j])
    return csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))

def lp_best_response(a, M):
    n = len(a)
    A = build_A_sparse(n, a)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    try:
        r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
        if r.success and r.x is not None:
            g = np.maximum(r.x, 0)
            if g.sum() > 1e-15:
                return g
    except Exception:
        pass
    return None
```

```python
def optimize_coord_fast(a, b, j, S):
    """Find optimal value for coordinate j, given precomputed autocorrelation b."""
    n = len(a)
    a_j = a[j]
    
    # Precompute affected positions
    # Position k is affected if j in [0,n-1] (always true) and k-j in [0,n-1]
    # i.e., k in [j, j+n-1]
    k_min = j  # k-j = 0
    k_max = j + n - 1  # k-j = n-1
    
    # Also need k in [0, 2n-2]
    k_min = max(0, k_min)
    k_max = min(2*n-2, k_max)
    
    # B_fixed: max over positions NOT in [k_min, k_max]
    if k_min > 0:
        left_max = b[:k_min].max() if k_min > 0 else -np.inf
    else:
        left_max = -np.inf
    if k_max < 2*n-2:
        right_max = b[k_max+1:].max() if k_max+1 < 2*n-1 else -np.inf
    else:
        right_max = -np.inf
    B_fixed = max(left_max, right_max)
    
    # Values at affected positions
    ks = np.arange(k_min, k_max+1)
    # For each k, the coefficient a_{k-j}
    k_minus_j = ks - j  # these are in [0, n-1]
    coeffs = a[k_minus_j]
    
    # The quadratic term at k=2j
    has_quad = (2*j >= 0 and 2*j <= 2*n-2)
    b_2j = b[2*j] if has_quad else 0
    
    # Function to evaluate new max autocorrelation for given t
    def new_max_auto(t):
        # Affine terms: b_k + 2*(t-a_j)*a_{k-j} for k in affected, k != 2j
        mask = ks != 2*j
        if mask.any():
            affine_max = (b[ks[mask]] + 2*(t-a_j)*coeffs[mask]).max()
        else:
            affine_max = -np.inf
        # Quadratic term
        quad_val = b_2j + t*t - a_j*a_j if has_quad else -np.inf
        return max(B_fixed, affine_max, quad_val)
    
    # Objective: new_max_auto(t) / (S - a_j + t)^2
    def obj(t):
        S_new = S - a_j + t
        if S_new < 1e-10:
            return 1e10
        return new_max_auto(t) / (S_new * S_new)
    
    # Coarse search
    max_val = max(a_j * 3, S / n * 2, S / n)
    grid = np.linspace(0, max_val, 20)
    vals = [obj(t) for t in grid]
    best_idx = np.argmin(vals)
    best_t = grid[best_idx]
    best_val = vals[best_idx]
    
    # Fine search
    if best_idx > 0 and best_idx < len(grid)-1:
        lo = grid[best_idx-1]
        hi = grid[best_idx+1]
    else:
        lo = max(0, best_t - max_val/10)
        hi = best_t + max_val/10
    
    fine_grid = np.linspace(lo, hi, 10)
    for t in fine_grid:
        v = obj(t)
        if v < best_val:
            best_val = v
            best_t = t
    
    return best_t
```

```python
                b[ks[aff_mask]] += 2.0 * delta * a[ks[aff_mask] - j]

            # Quadratic update at k = 2j
            if k_lo <= two_j <= k_hi:
                b[two_j] += t_new * t_new - a[j] * a[j]

            a[j] = t_new
            s += delta

        # Recompute b periodically to avoid drift
        if (sweep + 1) % 10 == 0:
            b = fast_conv(a).copy()
            s = float(a.sum())

        new_J = float(2 * n * np.max(fast_conv(a)) / (s * s))
        if new_J < best_J - 1e-12:
            best_J = new_J
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1
            if no_improve > 30:
                break

        current_J = new_J

    return best_a, best_J


def resample_seq(src, dest_n):
    """Resample 1D sequence to new length."""
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)


def generate_inits(n, rng, parent=None):
    """Generate diverse initial sequences of length n."""
    inits = []
    t = np.arange(n, dtype=np.float64) / n

    if parent is not None:
        p = resample_seq(parent, n)
        inits.append(p.copy())
        inits.append(np.maximum(p + rng.normal(0, 0.03, n), 0.01))
        inits.append(np.maximum(p * rng.uniform(0.8, 1.2, n), 0.01))

    inits.append(np.ones(n))
    inits.append(0.5 * (1 + np.cos(2 * np.pi * t)))
    inits.append(np.maximum(0.42 - 0.5 * np.cos(2 * np.pi * t) + 0.08 * np.cos(4 * np.pi * t), 0.01))
    inits.append(np.exp(-4 * (t - 0.5) ** 2) + 0.1)
    inits.append(np.exp(-8 * (t - 0.3) ** 2) + np.exp(-8 * (t - 0.7) ** 2) + 0.05)
    for f in [1, 2, 3, 5, 7, 11, 13]:
        inits.append(np.maximum(1.0 + 0.3 * np.cos(2 * np.pi * f * t), 0.01))
    inits.append(np.maximum(1.0 - 0.4 * np.abs(2 * t - 1), 0.05))
    inits.append(np.maximum(1.0 + 0.2 * np.cos(2 * np.pi * t) + 0.15 * np.cos(4 * np.pi * t + 0.5), 0.01))

    # Smooth random
    k = max(3, n // 16)
    kernel = np.ones(k) / k
    smooth = np.convolve(rng.uniform(0, 1, n), kernel, mode='same')
    inits.append(np.maximum(smooth + 0.2, 0.01))

    # Binary-like smoothed
    binary = rng.choice([0.1, 1.0], size=n)
    k2 = max(3, n // 8)
    kernel2 = np.ones(k2) / k2
    inits.append(np.maximum(np.convolve(binary, kernel2, mode='same'), 0.01))

    result = []
    for a in inits:
        a = np.maximum(a, 0.001)
        s = a.sum()
        if s > 1e-6:
            result.append(a / s)
    return result


def structured_perturb(a, rng):
    """Apply a random structured perturbation."""
    n = len(a)
    t = np.arange(n, dtype=np.float64) / n
    ptype = rng.integers(0, 5)

    if ptype == 0:  # shift
        s = int(rng.integers(1, max(2, n // 10)))
        a_p = np.roll(a, s)
    elif ptype == 1:  # cosine modulation
        freq = rng.uniform(1, 8)
        amp = rng.uniform(0.02, 0.15)
        a_p = a * (1 + amp * np.cos(2 * np.pi * freq * t + rng.uniform(0, 2 * np.pi)))
    elif ptype == 2:  # smooth
        k = int(rng.integers(3, max(4, n // 10)))
        kernel = np.ones(k) / k
        a_p = np.convolve(a, kernel, mode='same')
    elif ptype == 3:  # taper
        alpha = rng.uniform(0.5, 0.9)
        edge = max(1, int(n * (1 - alpha) / 2))
        window = np.ones(n)
        for i in range(edge):
            frac = i / edge
            window[i] = frac
            window[n - 1 - i] = frac
        a_p = a * window
    else:  # noise
        a_p = a + rng.normal(0, 0.04, n)

    a_p = np.maximum(a_p, 0.001)
    s = a_p.sum()
    if s > 1e-8:
        a_p = a_p / s
    return a_p


def lbb_objective(x):
    """Objective for L-BFGS-B: max of self-convolution (normalized)."""
    ax = np.maximum(x, 0.0)
    s = ax.sum()
    if s < 1e-8:
        return 1e10
    ax = ax / s
    b = fast_conv(ax)
    return float(b.max())


class ScoreTracker:
    def __init__(self):
        self.best = None
        self.best_score = float('inf')
        self.t_start = time.monotonic()

    def update(self, a):
        a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
        if a.size == 0 or a.sum() < 0.01:
            return
        score = float(evaluate_sequence(a.tolist()))
        if np.isfinite(score) and score < self.best_score:
            self.best = a.copy()
            self.best_score = score
            elapsed = time.monotonic() - self.t_start
            print(f"  [NEW BEST] J={score:.8f} n={a.size} t={elapsed:.0f}s", flush=True)

    def report(self, label=""):
        if self.best is not None:
            print(f"  {label} best: J={self.best_score:.8f} n={self.best.size}", flush=True)


def propose_candidate(seed=42, budget_s=1500, **kwargs):
    """
    Hybrid optimizer: fast coordinate descent with incremental updates,
    LP mirror descent, multi-length search, structured perturbations.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + float(budget_s) - 20.0
    tracker = ScoreTracker()

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        if p.ndim == 1 and p.size > 0 and np.all(np.isfinite(p)):
            parent = np.maximum(p, 0.0)
            tracker.update(parent)
            print(f"Parent loaded: n={parent.size}, J={tracker.best_score:.6f}", flush=True)
    except (NameError, AttributeError):
        pass

    if parent is None:
        parent = np.ones(640)
        tracker.update(parent)
        print(f"Default parent: n=640, J={tracker.best_score:.6f}", flush=True)

    # === Phase 1: Multi-length initialization ===
    print(f"\n{'='*50}", flush=True)
    print(f"Phase 1: Multi-length initialization", flush=True)
    print(f"{'='*50}", flush=True)

    t0 = time.monotonic()
    best_a = None
    best_J = float('inf')

    base_n = parent.size
    lengths = sorted(set([
        max(256, base_n - 128),
        base_n,
        base_n + 64,
        base_n + 128,
        base_n + 256,
        min(2048, base_n * 2),
    ]))
    lengths = [n for n in lengths if 128 <= n <= 2048]

    per_len_time = max(5, 60 / len(lengths))

    for n_try in lengths:
        if time.monotonic() > deadline - 800:
            break
        inits = generate_inits(n_try, rng, parent)
        for a_init in inits:
            if time.monotonic() > deadline - 800:
                break
            a_cd, J_cd = coord_sweep(a_init, rng, n_sweeps=15,
                                      deadline=time.monotonic() + per_len_time / len(inits))
            if J_cd < best_J:
                best_J = J_cd
                best_a = a_cd.copy()
                tracker.update(a_cd)

    print(f"Phase 1 done: best J={best_J:.8f}, n={best_a.size} ({time.monotonic()-t0:.0f}s)", flush=True)

    # === Phase 2: Deep coordinate descent ===
    print(f"\n{'='*50}", flush=True)
    print(f"Phase 2: Deep coordinate descent (n={best_a.size})", flush=True)
    print(f"{'='*50}", flush=True)

    t0 = time.monotonic()
    phase2_dl = min(deadline - 500, time.monotonic() + 350)

    # Run in blocks to allow periodic perturbations
    while time.monotonic() < phase2_dl:
        block_dl = min(phase2_dl, time.monotonic() + 30)
        best_a, J_new = coord_sweep(best_a, rng, n_sweeps=1000, deadline=block_dl)
        if J_new < best_J:
            best_J = J_new
            tracker.update(best_a)

        # Occasionally perturb to escape local minima
        if rng.random() < 0.3 and time.monotonic() < phase2_dl - 30:
            a_pert = structured_perturb(best_a, rng)
            a_pert, J_pert = coord_sweep(a_pert, rng, n_sweeps=30,
                                          deadline=time.monotonic() + 20)
            if J_pert < best_J:
                best_J = J_pert
                best_a = a_pert
                tracker.update(best_a)

    print(f"Phase 2 done: J={best_J:.8f} ({time.monotonic()-t0:.0f}s)", flush=True)

    # === Phase 3: LP mirror descent ===
    print(f"\n{'='*50}", flush=True)
    print(f"Phase 3: LP mirror descent (n={best_a.size})", flush=True)
    print(f"{'='*50}", flush=True)

    t0 = time.monotonic()
    phase3_dl = min(deadline - 300, time.monotonic() + 200)

    a_lp = best_a.copy()
    a_lp = np.maximum(a_lp, 1e-10)
    a_lp = a_lp / a_lp.sum()
    a_lp, J_lp = lp_mirror_descent(a_lp, 500, phase3_dl, use_multiplicative=False)
    tracker.update(a_lp)

    if J_lp < best_J:
        best_J = J_lp
        best_a = a_lp
    print(f"  LP linear: J={J_lp:.8f} ({time.monotonic()-t0:.0f}s)", flush=True)

    # Try multiplicative
    if time.monotonic() < deadline - 250:
        t0b = time.monotonic()
        a_lp2 = best_a.copy()
        a_lp2 = np.maximum(a_lp2, 1e-10)
        a_lp2 = a_lp2 / a_lp2.sum()
        a_lp2, J_lp2 = lp_mirror_descent(a_lp2, 300,
                                          min(deadline - 250, time.monotonic() + 100),
                                          use_multiplicative=True)
        tracker.update(a_lp2)
        if J_lp2 < best_J:
            best_J = J_lp2
            best_a = a_lp2
        print(f"  LP mult: J={J_lp2:.8f} ({time.monotonic()-t0b:.0f}s)", flush=True)

    # === Phase 4: Alternate coord + LP ===
    print(f"\n{'='*50}", flush=True)
    print(f"Phase 4: Alternate coord+LP (n={best_a.size})", flush=True)
    print(f"{'='*50}", flush=True)

    t0 = time.monotonic()
    phase4_dl = min(deadline - 150, time.monotonic() + 250)

    for round_i in range(50):
        if time.monotonic() > phase4_dl:
            break

        # Coordinate descent phase
        cd_dl = min(phase4_dl - 30, time.monotonic() + 30)
        best_a, J_cd = coord_sweep(best_a, rng, n_sweeps=200, deadline=cd_dl)
        if J_cd < best_J:
            best_J = J_cd
            tracker.update(best_a)

        # LP phase
        lp_dl = min(phase4_dl, time.monotonic() + 30)
        a_tmp = best_a.copy()
        a_tmp = np.maximum(a_tmp, 1e-10)
        a_tmp = a_tmp / a_tmp.sum()
        a_tmp, J_lp = lp_mirror_descent(a_tmp, 30, lp_dl, use_multiplicative=False)
        if J_lp < best_J:
            best_J = J_lp
            best_a = a_tmp
            tracker.update(best_a)

    print(f"Phase 4 done: J={best_J:.8f} ({time.monotonic()-t0:.0f}s)", flush=True)

    # === Phase 5: Multi-length refinement ===
    print(f"\n{'='*50}", flush=True)
    print(f"Phase 5: Multi-length refinement", flush=True)
    print(f"{'='*50}", flush=True)

    t0 = time.monotonic()
    phase5_dl = min(deadline - 100, time.monotonic() + 150)

    base_n = best_a.size
    nearby_lengths = sorted(set([
        max(128, base_n - 64),
        base_n + 64,
        base_n + 128,
        max(128, base_n - 128),
    ]))
    nearby_lengths = [n for n in nearby_lengths if 128 <= n <= 2048 and n != base_n]

    per_len_time = max(10, (phase5_dl - time.monotonic()) / max(1, len(nearby_lengths)))

    for n_try in nearby_lengths:
        if time.monotonic() > phase5_dl:
            break
        a_try = resample_seq(best_a, n_try)
        a_try = np.maximum(a_try, 1e-8)
        a_try = a_try / a_try.sum()
        a_try, J_try = coord_sweep(a_try, rng, n_sweeps=200,
                                    deadline=time.monotonic() + per_len_time)
        tracker.update(a_try)
        print(f"  n={n_try}: J={J_try:.8f}", flush=True)
        if J_try < best_J:
            best_J = J_try
            best_a = a_try

    print(f"Phase 5 done ({time.monotonic()-t0:.0f}s)", flush=True)

    # === Phase 6: Final long coordinate descent ===
    print(f"\n{'='*50}", flush=True)
    print(f"Phase 6: Final coordinate descent (n={best_a.size})", flush=True)
    print(f"{'='*50}", flush=True)

    t0 = time.monotonic()
    final_dl = min(deadline - 10, time.monotonic() + 120)
    best_a, J_final = coord_sweep(best_a, rng, n_sweeps=5000, deadline=final_dl)
    tracker.update(best_a)
    if J_final < best_J:
        best_J = J_final
        best_a = a_cd if False else best_a  # best_a already updated
    print(f"  Final CD: J={J_final:.8f} ({time.monotonic()-t0:.0f}s)", flush=True)

    # === Phase 7: L-BFGS-B polish ===
    if time.monotonic() < deadline and tracker.best is not None:
        t0 = time.monotonic()
        print(f"\nPhase 7: L-BFGS-B polish", flush=True)
        try:
            x0 = tracker.best.astype(np.float64).copy()
            res = minimize(lbb_objective, x0, method='L-BFGS-B',
                           bounds=[(0.0, None)] * len(x0),
                           options={'maxiter': 50, 'max_fun': 20000, 'ftol': 1e-16})
            if res.fun < 1e10:
                tracker.update(res.x)
                J_lbb = compute_J(res.x)
                if J_lbb < best_J:
                    best_J = J_lbb
                    best_a = res.x
                print(f"  L-BFGS-B: J={J_lbb:.8f}", flush=True)
        except Exception as e:
            print(f"  L-BFGS-B error: {e}", flush=True)
        print(f"  ({time.monotonic()-t0:.1f}s)", flush=True)

    # Final
    if tracker.best is not None:
        final_a = tracker.best
    else:
        final_a = best_a

    print(f"\n{'='*50}", flush=True)
    print(f"FINAL: J={tracker.best_score:.8f} n={final_a.size} time={time.monotonic()-tracker.t_start:.0f}s", flush=True)
    print(f"{'='*50}", flush=True)

    return [float(x) for x in final_a]
```
