Optimized step-function coefficient search using alternating LP as the primary optimizer,
with subgradient polishing, dual-guided perturbation, and multi-scale exploration.
Key improvements: LP-centric optimization, smarter active sets, better time allocation.
"""

import numpy as np
from scipy.optimize import linprog
from scipy.sparse import coo_matrix
import time

def compute_conv(a):
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    conv = np.fft.irfft(fa * fa, size)
    return conv[:2 * n - 1]

def compute_obj(a):
    n = len(a)
    S = float(np.sum(a))
    if S < 1e-10:
        return np.inf
    conv = compute_conv(a)
    M = float(np.max(conv))
    return 2.0 * n * M / (S * S)

def build_toeplitz(a):
    n = len(a)
    rows = []
    cols = []
    vals = []
    for j in range(n):
        start = j
        end = min(j + n, 2 * n - 1)
        length = end - start
        rows.extend(range(start, end))
        cols.extend([j] * length)
        vals.extend(list(a[:length]))
    A = coo_matrix((vals, (rows, cols)), shape=(2 * n - 1, n)).tocsr()
    return A

def solve_lp(a, M, time_limit=10):
    n = len(a)
    A = build_toeplitz(a)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': time_limit})
        if res.status == 0:
            return np.maximum(res.x, 0)
    except Exception:
        pass
    return None

def subgradient_step(a, n_active='all', topk=50):
    """One subgradient step with given active set size."""
    n = len(a)
    conv = compute_conv(a)
    M = float(np.max(conv))
    
    if n_active == 'all':
        # Use all positions within 1% of max
        threshold = M * 0.99
        active = np.where(conv >= threshold)[0]
        if len(active) == 0:
            active = np.argmax(conv)
            active = [active]
    else:
        active = np.argsort(conv)[-topk:]
    
    # Compute gradient
    grad = np.zeros(n)
    for k in active:
        # d(conv[k])/da[i] = 2*a[k-i] if 0 <= k-i < n
        j_min = max(0, k - (n - 1))
        j_max = min(n - 1, k)
        if j_min <= j_max:
            indices = np.arange(j_min, j_max + 1)
            a_vals = a[k - indices]  # a[k-i] for i in indices
            grad[indices] += 2.0 * a_vals
    
    grad /= len(active)
    return grad, M

    v = np.zeros(n)

def cosine_init(n, n_modes=10, seed=0):
    rng = np.random.default_rng(seed)
    k = np.arange(n)
    a = np.ones(n)
    for m in range(1, n_modes + 1):
        c = rng.uniform(-0.8, 0.8) / (m * 0.8)
        phase = rng.uniform(0, 2 * np.pi)
        a += c * np.cos(2 * np.pi * m * k / n + phase)
    a = np.maximum(a, 0)
    S = np.sum(a)
    if S < 1e-10:
        return np.ones(n) / n
    return a / S

def smooth_random_init(n, seed=0, smoothness=0.3):
    """Random init smoothed by convolution with a Gaussian kernel."""
    rng = np.random.default_rng(seed)
    a = rng.exponential(1.0, n)
    # Smooth
    kernel_size = max(3, int(n * smoothness))
    kernel = np.exp(-0.5 * (np.arange(kernel_size) - kernel_size//2)**2 / (kernel_size/4)**2)
    kernel = kernel / kernel.sum()
    a = np.convolve(a, kernel, mode='same')
    a = np.maximum(a, 0)
    a = a / a.sum()
    return a

"""
Optimized step-function coefficient search using alternating LP as the primary optimizer,
with FFT-based subgradient polishing, dual-guided perturbation, adaptive M-relaxation,
and multi-scale n exploration. Key improvements over previous: LP-centric design,
vectorized subgradient via cross-correlation, smarter stall handling, and better
time allocation strategy.

import numpy as np
from scipy.optimize import linprog
from scipy.sparse import coo_matrix
import time


def compute_conv(a):
    """Compute linear convolution a*a via FFT."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    conv = np.fft.irfft(fa * fa, size)
    return conv[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    S = float(np.sum(a))
    if S < 1e-10:
        return np.inf
    conv = compute_conv(a)
    M = float(np.max(conv))
    return 2.0 * n * M / (S * S)


def build_toeplitz(a):
    """Build sparse Toeplitz matrix for conv(a, g)."""
    n = len(a)
    rows = []
    cols = []
    vals = []
    for j in range(n):
        start = j
        end = min(j + n, 2 * n - 1)
        length = end - start
        rows.extend(range(start, end))
        cols.extend([j] * length)
        vals.extend(list(a[:length]))
    A = coo_matrix((vals, (rows, cols)), shape=(2 * n - 1, n)).tocsr()
    return A


def solve_lp(a, M, time_limit=10):
    """Solve LP: max sum(g) s.t. conv(a,g) <= M, g >= 0."""
    n = len(a)
    A = build_toeplitz(a)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': time_limit})
        if res.status == 0:
            return np.maximum(res.x, 0)
    except Exception:
        pass
    return None


def compute_subgrad_fft(a, active_mask):
    """Compute subgradient using FFT-based cross-correlation.
    active_mask: boolean/float array of length 2n-1 marking active positions.
    Returns gradient of length n (approximate subgradient of max(conv)).
    """
    n = len(a)
    conv_len = 2 * n - 1
    # Need linear convolution of a_rev (length n) with active_mask (length conv_len)
    # Result length: n + conv_len - 1 = 3n - 2
    size = 1
    while size < 3 * n - 1:
        size <<= 1
    a_rev = a[::-1].copy()
    fa = np.fft.rfft(a_rev, size)
    fm = np.fft.rfft(active_mask.astype(np.float64), size)
    conv_full = np.fft.irfft(fa * fm, size)
    # c[i] = conv_full[i + n - 1] for i = 0..n-1
    c = conv_full[n - 1:2 * n - 1]
    return 2.0 * c


def subgradient_descent(a, n_iters, time_budget, step_size=0.1, momentum=0.85, topk=50):
    """Momentum subgradient descent with adaptive active set."""
    n = len(a)
    a = a.copy().astype(np.float64)
    S = np.sum(a)
    if S < 1e-10:
        return a
    a = a / S

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    v = np.zeros(n)

    for _ in range(n_iters):
        if time.monotonic() - start > time_budget:
            break

        conv = compute_conv(a)
        M = float(np.max(conv))

        # Use all positions within 0.5% of max as active set
        threshold = M * 0.995
        active_indices = np.where(conv >= threshold)[0]
        if len(active_indices) > topk:
            # Use top-k by value
            active_indices = np.argsort(conv)[-topk:]
        elif len(active_indices) == 0:
            active_indices = np.array([np.argmax(conv)])

        active_mask = np.zeros(2 * n - 1)
        active_mask[active_indices] = 1.0 / len(active_indices)

        grad = compute_subgrad_fft(a, active_mask)
        grad_centered = grad - np.mean(grad)

        v = momentum * v + (1.0 - momentum) * grad_centered
        a_new = a - step_size * v
        a_new = np.maximum(a_new, 0)
        S_new = np.sum(a_new)
        if S_new < 1e-10:
            break
        a_new = a_new / S_new

        obj = compute_obj(a_new)
        if obj < best_obj:
            best_obj = obj
            best_a = a_new.copy()
            a = a_new

    return best_a


def alternating_lp_refine(a, time_budget, verbose=False, seed=0):
    """Main LP-based refinement with adaptive perturbation."""
    n = len(a)
    a = a.copy().astype(np.float64)
    S = np.sum(a)
    if S < 1e-10:
        return a, np.inf
    a = a / S

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    stall_count = 0
    lp_iter = 0
    m_relax = 1.0  # M relaxation factor

    while time.monotonic() - start < time_budget - 3:
        remaining = time.monotonic() - start
        rem_elapsed = time_budget - remaining
        if rem_elapsed > time_budget - 3:
            break

       (lp_iter)
        lp_iter += 1

        conv = compute_conv(a)
        M = float(np.max(conv))

        # Try with slightly relaxed M for diversity
        M_eff = M * m_relax

        g = solve_lp(a, M_eff, time_limit=min((time_budget - remaining) * 0.4, 12))
        if g is None:
            break

        S_g = np.sum(g)
        if S_g < 1e-10:
            break

        g_norm = g / S_g

        # Line search with geometric + linear
        best_t = 0.0
        best_obj_ls = best_obj
        ts = np.concatenate([
            np.linspace(0.01, 0.3, 8),
            np.linspace(0.3, 0.7, 5),
            np.linspace(0.7, 1.0, 3)
        ])
        for t in ts:
            a_try = (1.0 - t) * a + t * g_norm
            a_try = np.maximum(a_try, 0)
            S_try = np.sum(a_try)
            if S_try < 1e-10:
                continue
            a_try = a_try / S_try
            obj_try = compute_obj(a_try)
            if obj_try < best_obj_ls - 1e-12:
                best_obj_ls = obj_try
                best_t = t

        if best_t > 0:
            a = (1.0 - best_t) * a + best_t * g_norm
            a = np.maximum(a, 0)
            a = a / np.sum(a)
            best_obj = best_obj_ls
            best_a = a.copy()
            stall_count = 0
            # Reset M relaxation if we made progress
            m_relax = max(0.99, m_relax - 0.005)
            if verbose:
                print(f"    LP {lp_iter}: obj={best_obj:.6f}, t={best_t:.3f}, Sg={S_g:.4f}")
        else:
            stall_count += 1
            if verbose:
                print(f"    LP {lp_iter}: no improvement, stall={stall_count}")

            if stall_count >= 4:
                # Aggressive perturbation guided by LP solution structure
                rng = np.random.default_rng(seed + lp_iter * 7919 + 1)
                free_mask = g > 1e-8
                scale = rng.uniform(0.08, 0.2)
                pert = np.zeros(n)
                n_free = max(1, free_mask.sum())
                n_bound = max(1, n - n_free)
                pert[free_mask] = rng.normal(0, scale, n_free)
                pert[~free_mask] = rng.normal(0, scale * 0.2, n_bound)
                a = a * np.exp(pert)
                a = np.maximum(a, 0)
                S = np.sum(a)
                if S < 1e-10:
                    a = np.ones(n) / n
                else:
                    a = a / S
                stall_count = 0
                m_relax = rng.uniform(1.0, 1.05)
                # Quick subgradient after perturbation
                a = subgradient_descent(a, 300, 4.0, step_size=0.08, topk=40)
                obj_after = compute_obj(a)
                if obj_after < best_obj:
                    best_obj = obj_after
                    best_a = a.copy()
            elif stall_count == 2:
                # Medium perturbation
                rng = np.random.default_rng(seed + lp_iter * 4441 + 3)
                a_pert = a * np.exp(rng.normal(0, 0.06, n))
                a_pert = np.maximum(a_pert, 0)
                a_pert = a_pert / np.sum(a_pert)
                a = a_pert
                m_relax = 1.0 + 0.01 * stall_count
            elif stall_count == 3:
                # Try direct subgradient push
                a = subgradient_descent(a, 500, 5.0, step_size=0.12, topk=60)
                obj_check = compute_obj(a)
                if obj_check < best_obj:
                    best_obj = obj_check
                    best_a = a.copy()

    # Final subgradient polish
    elapsed = time.monotonic() - start
    if elapsed < time_budget - 5:
        a_sg = subgradient_descent(best_a, 1000, time_budget - elapsed, step_size=0.04, topk=50)
        obj_sg = compute_obj(a_sg)
        if obj_sg < best_obj:
            best_obj = obj_sg
            best_a = a_sg.copy()

    return best_a, best_obj


def cosine_init(n, n_modes=10, seed=0):
    """Cosine series initialization."""
    rng = np.random.default_rng(seed)
    k = np.arange(n)
    a = np.ones(n)
    for m in range(1, n_modes + 1):
        c = rng.uniform(-0.8, 0.8) / (m * 0.7)
        phase = rng.uniform(0, 2 * np.pi)
        a += c * np.cos(2.0 * np.pi * m * k / n + phase)
    a = np.maximum(a, 0)
    S = np.sum(a)
    if S < 1e-10:
        return np.ones(n) / n
    return a / S


def smooth_random_init(n, seed=0, smooth_frac=0.3):
    """Random exponential init smoothed with Gaussian kernel."""
    rng = np.random.default_rng(seed)
    a = rng.exponential(1.0, n)
    kernel_size = max(3, int(n * smooth_frac) | 1)
    t = np.arange(kernel_size) - kernel_size // 2
    kernel = np.exp(-0.5 * (t / (kernel_size / 3.0)) ** 2)
    kernel = kernel / kernel.sum()
    a = np.convolve(a, kernel, mode='same')
    a = np.maximum(a, 0)
    S = np.sum(a)
    if S < 1e-10:
        return np.ones(n) / n
    return a / S


def spectral_flat_init(n, seed=0):
    """Initialize toward flat power spectrum (low auto-correlation)."""
    rng = np.random.default_rng(seed)
    # Start with random phase in frequency domain, take real part
    nfft = n
    spectrum = np.ones(nfft // 2 + 1)
    # Add small random perturbation to make it non-uniform
    for m in range(1, nfft // 2 + 1):
        spectrum[m] = 1.0 + rng.uniform(-0.3, 0.3)
    # Construct time-domain signal
    a = np.fft.irfft(spectrum, nfft)
    # Make non-negative
    a = a - np.min(a)
    a = np.maximum(a, 0)
    S = np.sum(a)
    if S < 1e-10:
        return np.ones(n) / n
    return a / S


def try_parent():
    """Try to load parent sequence from global variable."""
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64)
        parent = np.clip(parent, 0, 1000.0)
        if parent.size > 0 and np.isfinite(parent).all():
            obj = compute_obj(parent)
            print(f"Parent (n={parent.size}): obj={obj:.6f}")
            return parent.copy(), obj
    except Exception:
        pass
    return None, np.inf


def propose_candidate(seed=42, budget_s=1500):
    """
    Main search: alternating LP with FFT subgradient, dual-guided perturbation,
    adaptive M-relaxation, and multi-n exploration.
    """
    start_time = time.monotonic()
    deadline = start_time + float(budget_s) - 25.0

    best_seq = None
    best_score = np.inf

    # Load parent
    parent, parent_obj = try_parent()
    if parent is not None:
        best_score = parent_obj
        best_seq = parent.copy()

    n_parent = len(best_seq) if best_seq is not None else 300
    print(f"Starting: best={best_score:.6f} (n={n_parent})")

    # === Phase 1: Diverse initialization ===
    print("\n=== Phase 1: Diverse candidates ===")
    n_values = sorted(set([
        max(100, n_parent - 100), max(100, n_parent - 50),
        n_parent,
        n_parent + 50, n_parent + 100, n_parent + 200,
        min(1200, n_parent + 350)
    ]))
    n_values = [nn for nn in n_values if 100 <= nn <= 1500]

    candidates = []

    for nn in n_values:
        if time.monotonic() > deadline - 60:
            break
        t_remaining = deadline - time.monotonic()
        if t_remaining < 15:
            break

        inits = []
        for s in range(2):
            inits.append(cosine_init(nn, n_modes=8, seed=seed + nn * 13 + s))
            inits.append(smooth_random_init(nn, seed=seed + nn * 29 + s, smooth_frac=0.2 + 0.15 * s))
        inits.append(spectral_flat_init(nn, seed=seed + nn * 41))

        for i, a_init in enumerate(inits):
            if time.monotonic() > deadline - 60:
                break
            a_quick = subgradient_descent(a_init, 200, 2.5, step_size=0.1, topk=30)
            obj = compute_obj(a_quick)
            candidates.append((a_quick, obj))
            if obj < best_score:
                best_score = obj
                best_seq = a_quick.copy()
                print(f"  n={nn} init[{i}]: {obj:.6f} ** NEW BEST **")

    candidates.sort(key=lambda x: x[1])
    if candidates:
        print(f"  Top: obj={candidates[0][1]:.6f} (n={len(candidates[0][0])})")

    # === Phase 2: LP refinement on top candidates ===
    print("\n=== Phase 2: LP refinement ===")
    n_refine = min(4, len(candidates))
    for idx in range(n_refine):
        if time.monotonic() > deadline - 30:
            break
        t_remaining = deadline - time.monotonic()
        if t_remaining < 25:
            break

        a_cand, obj_cand = candidates[idx]
        if obj_cand > best_score + 0.03:
            break

        nn = len(a_cand)
        weight = 1.0 / (idx + 1)
        alloc = min(t_remaining * 0.3 * weight, 80.0)
        print(f"  Cand {idx} (n={nn}, obj={obj_cand:.6f}), budget={alloc:.0f}s")

        a_ref, obj_ref = alternating_lp_refine(a_cand, alloc, verbose=(idx == 0), seed=seed + idx * 1000)

        if obj_ref < best_score:
            best_score = obj_ref
            best_seq = a_ref.copy()
            print(f"  ** NEW BEST: n={nn}, obj={best_score:.6f} **")

    # === Phase 3: Intensive LP on current best ===
    print("\n=== Phase 3: Intensive LP on best ===")
    a_cur = best_seq.copy()

    for rnd in range(12):
        if time.monotonic() > deadline - 20:
            break
        t_remaining = deadline - time.monotonic()
        if t_remaining < 15:
            break

        alloc = min(t_remaining * 0.25, 60.0)
        a_new, obj_new = alternating_lp_refine(a_cur, alloc, verbose=(rnd == 0), seed=seed + rnd * 5000)

        if obj_new < best_score - 1e-9:
            best_score = obj_new
            best_seq = a_new.copy()
            a_cur = a_new.copy()
            print(f"  R{rnd}: {best_score:.6f} ** NEW BEST **")
        else:
            print(f"  R{rnd}: {obj_new:.6f} (best={best_score:.6f})")
            # Multiple perturbation attempts
            rng = np.random.default_rng(seed + rnd * 31337 + 7)
            for ps in [0.05, 0.12, 0.25]:
                if time.monotonic() > deadline - 15:
                    break
                a_p = a_cur * np.exp(rng.normal(0, ps, len(a_cur)))
                a_p = np.maximum(a_p, 0)
                S_p = np.sum(a_p)
                if S_p < 1e-10:
                    continue
                a_p = a_p / S_p
                alloc_p = min(t_remaining * 0.12, 25.0)
                a_pp, obj_pp = alternating_lp_refine(a_p, alloc_p, verbose=False, seed=seed + rnd * 10000 + int(ps * 100))
                if obj_pp < best_score:
                    best_score = obj_pp
                    best_seq = a_pp.copy()
                    a_cur = a_pp.copy()
                    print(f"  R{rnd} pert({ps}): {best_score:.6f} ** NEW BEST **")
                    break

    # === Phase 4: Multi-n exploration ===
    print("\n=== Phase 4: Multi-n exploration ===")
    if time.monotonic() < deadline - 25:
        n_base = len(best_seq)
        n_tries = sorted(set([
            max(100, n_base - 80), max(100, n_base - 40),
            n_base + 40, n_base + 80, n_base + 150,
            max(100, n_base + 250), max(100, n_base + 400)
        ]))
        n_tries = [nn for nn in n_tries if 100 <= nn <= 2000 and nn != n_base]

        for nn in n_tries:
            if time.monotonic() > deadline - 12:
                break
            t_remaining = deadline - time.monotonic()
            if t_remaining < 12:
                break

            # Interpolate to new length
            x_old = np.linspace(0, 1, n_base)
            x_new = np.linspace(0, 1, nn)
            a_interp = np.interp(x_new, x_old, best_seq)
            a_interp = np.maximum(a_interp, 0)
            S_i = np.sum(a_interp)
            if S_i > 1e-10:
                a_interp = a_interp / S_i

            alloc = min(t_remaining * 0.3, 35.0)
            a_new, obj_new = alternating_lp_refine(a_interp, alloc, verbose=False, seed=seed + nn)
            if obj_new < best_score:
                best_score = obj_new
                best_seq = a_new.copy()
                print(f"  n={nn}: {best_score:.6f} ** NEW BEST **")
            else:
                print(f"  n={nn}: {obj_new:.6f}")

    # === Phase 5: Final polish ===
    print("\n=== Phase 5: Final polish ===")
    a_cur = best_seq.copy()
    n_final = len(a_cur)

    if time.monotonic() < deadline - 8:
        t_remaining = deadline - time.monotonic()

        # Extended subgradient with decreasing step
        a_sg = subgradient_descent(a_cur, int(t_remaining * 25), t_remaining * 0.4,
                                   step_size=0.03, topk=60)
        obj_sg = compute_obj(a_sg)
        if obj_sg < best_score:
            best_score = obj_sg
            best_seq = a_sg.copy()
            a_cur = a_sg.copy()
            print(f"  SG: {best_score:.6f} ** NEW BEST **")

        # Final LP if reasonable size
        t_remaining = deadline - time.monotonic()
        if t_remaining > 8 and n_final <= 700:
            a_lp, obj_lp = alternating_lp_refine(a_cur, min(t_remaining * 0.6, 30.0),
                                                 verbose=False, seed=99999)
            if obj_lp < best_score:
                best_score = obj_lp
                best_seq = a_lp.copy()
                print(f"  Final LP: {best_score:.6f} ** NEW BEST **")

        # One last perturbation attempt
        t_remaining = deadline - time.monotonic()
        if t_remaining > 10:
            rng = np.random.default_rng(123456)
            a_p = a_cur * np.exp(rng.normal(0, 0.04, n_final))
            a_p = np.maximum(a_p, 0)
            a_p = a_p / np.sum(a_p)
            a_p = subgradient_descent(a_p, 2000, t_remaining * 0.5, step_size=0.02, topk=80)
            obj_p = compute_obj(a_p)
            if obj_p < best_score:
                best_score = obj_p
                best_seq = a_p.copy()
                print(f"  Final pert: {best_score:.6f} ** NEW BEST **")

    # Final validation
    if best_seq is None:
        best_seq = np.ones(300) / 300
        best_score = compute_obj(best_seq)

    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    val = evaluate_sequence(result)
    if val < np.inf and val < best_score:
        best_score = val
    elif val == np.inf:
        result = [1.0] * 300
        best_score = 2.0

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_score:.6f}, time={elapsed:.1f}s ===")

    return result
