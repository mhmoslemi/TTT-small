"""
Intensive peak-reduction optimization for minimizing 2*n*max(conv(a,a))/(sum(a))^2.
Uses aggressive LP alternating, soft-max L-BFGS-B with adaptive temperature,
spectral equalization, and peak-directed subgradient methods.

import numpy as np
from scipy.optimize import minimize, linprog
from scipy.fft import rfft, irfft
import time

def compute_conv(a):
    n = len(a)
    size = 1
    while size < 2*n - 1:
        size <<= 1
    fa = rfft(a, size)
    conv = irfft(fa * fa, size)
    return conv[:2*n-1]

def compute_obj(a):
    n = len(a)
    if n == 0:
        return np.inf
    conv = compute_conv(a)
    max_b = float(np.max(conv))
    sum_a = float(np.sum(a))
    if sum_a < 0.01:
        return np.inf
    return 2.0 * n * max_b / (sum_a * sum_a)

def fft_convolve(a, b):
    n, m = len(a), len(b)
    size = 1
    while size < n + m - 1:
        size <<= 1
    fa = rfft(a, size)
    fb = rfft(b, size)
    return irfft(fa * fb, size)[:n+m-1]

def softmax(z):
    z = z - np.max(z)
    e = np.exp(z)
    return e / np.sum(e)

def soft_max_obj_and_grad_z(z, tau):
    n = len(z)
    a = softmax(z)
    conv = compute_conv(a)
    cmax = float(np.max(conv))
    shifted = np.clip((conv - cmax) / tau, -500, 500)
    exp_conv = np.exp(shifted)
    Z = float(np.sum(exp_conv))
    soft_max_val = cmax + tau * np.log(Z)
    w = exp_conv / Z
    w_rev = w[::-1]
    conv_fw = fft_convolve(a, w_rev)
    f1 = conv_fw[n-1:2*n-1][::-1]
    grad_a = 2.0 * f1
    mean_grad = float(np.dot(a, grad_a))
    grad_z = a * (grad_a - mean_grad)
    return soft_max_val, grad_z

def build_toeplitz(a):
    n = len(a)
    A = np.zeros((2*n-1, n))
    for j in range(n):
        k_start = j
        k_end = min(j + n, 2*n - 1)
        A[k_start:k_end, j] = a[:k_end - k_start]
    return A

def solve_lp_g(a, M, time_limit=20):
    n = len(a)
    A = build_toeplitz(a)
    c = -np.ones(n)
    b_ub = np.full(2*n-1, M)
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': time_limit})
        if res.status == 0:
            g = np.maximum(res.x, 0)
            s = float(np.sum(g))
            if s > 1e-10:
                return g/s, s
    except Exception:
        pass
    return None, None

def alternating_lp(a, max_iters=100, time_budget=60.0):
    n = len(a)
    a = a.copy().astype(np.float64)
    s = np.sum(a)
    if s < 1e-10:
        return a
    a = a / s
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    stall_count = 0
    
    for it in range(max_iters):
        if time.monotonic() - start > time_budget:
            break
        
        conv = compute_conv(a)
        M = float(np.max(conv))
        
        g, _ = solve_lp_g(a, M, time_limit=10)
        if g is None:
            break
        
        # Adaptive line search
        best_t_obj = best_obj
        best_a_new = best_a.copy()
        
        # Try multiple step sizes with golden-section refinement
        t_candidates = [0.05, 0.1, 0.2, 0.3, 0.5, 0.7, 0.85, 1.0]
        for t in t_candidates:
            a_try = (1.0 - t) * a + t * g
            a_try = np.maximum(a_try, 0)
            s_try = np.sum(a_try)
            if s_try < 1e-10:
                continue
            a_try = a_try / s_try
            obj_try = compute_obj(a_try)
            if obj_try < best_t_obj:
                best_t_obj = obj_try
                best_a_new = a_try.copy()
        
        if best_t_obj < best_obj - 1e-10:
            best_obj = best_t_obj
            best_a = best_a_new.copy()
            a = best_a_new
            stall_count = 0
        else:
            stall_count += 1
            if stall_count > 15:
                # Perturb to escape local minimum
                rng = np.random.default_rng(it * 7919 + 12345)
                a = a * np.exp(rng.normal(0, 0.05, n))
                a = np.maximum(a, 0)
                s = np.sum(a)
                if s > 1e-10:
                    a = a / s
                stall_count = 0
    
    return best_a

def subgradient_optimize(a, max_iters=2000, time_budget=20.0, tau_start=5.0, tau_end=50.0):
    n = len(a)
    a = a.copy().astype(np.float64)
    s = np.sum(a)
    if s < 1e-10:
        a = np.ones(n) / n
    else:
        a = a / s
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    step = 0.05
    stall = 0
    
    for it in range(max_iters):
        if time.monotonic() - start > time_budget:
            break
        
        frac = it / max(1, max_iters)
        tau = tau_start + (tau_end - tau_start) * frac
        
        conv = compute_conv(a)
        max_conv = float(np.max(conv))
        shifted = np.clip(tau * (conv - max_conv), -300, 300)
        weights = np.exp(shifted)
        w_sum = np.sum(weights)
        if w_sum < 1e-300:
            idx = int(np.argmax(conv))
            weights = np.zeros(len(conv))
            weights[idx] = 1.0
        else:
            weights /= w_sum
        
        w_rev = weights[::-1]
        conv_fw = fft_convolve(a, w_rev)
        grad_a = 2.0 * conv_fw[n-1:2*n-1][::-1]
        
        grad_centered = grad_a - np.mean(grad_a)
        grad_norm = np.linalg.norm(grad_centered)
        if grad_norm < 1e-15:
            break
        
        a_new = a - step * grad_centered / grad_norm
        a_new = np.maximum(a_new, 0)
        s_new = np.sum(a_new)
        if s_new > 1e-10:
            a_new = a_new / s_new
        else:
            a_new = a.copy()
        
        a = a_new
        obj = compute_obj(a)
        
        if obj < best_obj - 1e-12:
            best_obj = obj
            best_a = a.copy()
            stall = 0
        else:
            stall += 1
            if stall > 80:
                step *= 0.85
                if step < 1e-10:
                    break
        
        if it > 50:
            step *= 0.999
    
    return best_a

def optimize_lfbs(a_init, tau_list, max_iter=200, time_budget=30.0):
    n = len(a_init)
    a = a_init.copy()
    s = np.sum(a)
    if s < 1e-10:
        a = np.ones(n) / n
    else:
        a = a / s
    a = np.maximum(a, 1e-30)
    z = np.log(a)
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    
    for tau in tau_list:
        if time.monotonic() - start > time_budget:
            break
        
        def func_and_grad(z_val):
            return soft_max_obj_and_grad_z(z_val, tau)
        
        try:
            result = minimize(func_and_grad, z, jac=True, method='L-BFGS-B',
                              options={'maxiter': max_iter, 'ftol': 1e-15, 'gtol': 1e-9})
            z_new = result.x
        except Exception:
            continue
        
        a_new = softmax(z_new)
        a_new = np.maximum(a_new, 0)
        s_new = np.sum(a_new)
        if s_new < 1e-10:
            continue
        a_new = a_new / s_new
        obj = compute_obj(a_new)
        
        if obj < best_obj:
            best_obj = obj
            best_a = a_new.copy()
            z = z_new
    
    return best_a, best_obj

def spectral_equalize(a, n_iters=50, time_budget=10.0):
    """Try to flatten |A(f)|^2 in the frequency domain."""
    n = len(a)
    a = a.copy().astype(np.float64)
    s = np.sum(a)
    if s < 1e-10:
        return a
    a = a / s
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    
    for it in range(n_iters):
        if time.monotonic() - start > time_budget:
            break
        
        A = rfft(a)
        power = A.real**2 + A.imag**2
        
        # We want power to be flat. Compute the "excess" power.
        target = np.mean(power)
        # Scale: reduce where power > target, increase where power < target
        # But we need to work in the time domain to maintain non-negativity
        
        # Alternative: compute the gradient of max(conv) w.r.t. a using spectral info
        conv = compute_conv(a)
        max_conv = float(np.max(conv))
        
        # Find peak positions
        peak_threshold = max_conv * 0.95
        peaks = conv[conv >= peak_threshold]
        
        if len(peaks) == 0:
            break
        
        # For each peak position k, the gradient contribution is:
        # d conv[k] / d a[i] = a[k-i] (if valid)
        # So to reduce the peak at k, we should decrease a[i] where a[k-i] is large
        
        # Simplified: use the subgradient of the max
        peak_idx = np.argmax(conv)
        # d conv[peak_idx] / d a[i] = a[peak_idx - i] for valid i
        grad = np.zeros(n)
        for i in range(n):
            j = peak_idx - i
            if 0 <= j < n:
                grad[i] = a[j]
        
        # Normalize and take a step
        grad = grad - np.mean(grad)
        grad_norm = np.linalg.norm(grad)
        if grad_norm < 1e-15:
            break
        
        step = 0.01
        a_new = a - step * grad / grad_norm
        a_new = np.maximum(a_new, 0)
        s_new = np.sum(a_new)
        if s_new > 1e-10:
            a_new = a_new / s_new
        
        obj_new = compute_obj(a_new)
        if obj_new < best_obj:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new
        else:
            step *= 0.8
            if step < 1e-10:
                break
    
    return best_a

"""
Intensive peak-reduction optimization for minimizing 2*n*max(conv(a,a))/(sum(a))^2.
Uses aggressive LP alternating with adaptive step sizes, soft-max L-BFGS-B with
geometric tau annealing, peak-directed subgradient, and multi-phase polishing.

import numpy as np
from scipy.optimize import minimize, linprog
from scipy.fft import rfft, irfft
import time


def compute_conv(a):
    """Compute linear convolution a*a using FFT. Returns array of length 2n-1."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    conv = irfft(fa * fa, size)
    return conv[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    conv = compute_conv(a)
    max_b = float(np.max(conv))
    sum_a = float(np.sum(a))
    if sum_a < 0.01:
        return np.inf
    return 2.0 * n * max_b / (sum_a * sum_a)


def fft_convolve(a, b):
    """Linear convolution of a and b using FFT."""
    n, m = len(a), len(b)
    size = 1
    while size < n + m - 1:
        size <<= 1
    fa = rfft(a, size)
    fb = rfft(b, size)
    return irfft(fa * fb, size)[:n + m - 1]


def softmax_fn(z):
    """Numerically stable softmax."""
    z = z - np.max(z)
    e = np.exp(z)
    return e / np.sum(e)


def soft_max_obj_and_grad_z(z, tau):
    """
    Compute soft-max of conv(softmax(z), softmax(z)) and its gradient w.r.t. z.
    Returns (value, gradient).
    """
    n = len(z)
    a = softmax_fn(z)
    conv = compute_conv(a)
    cmax = float(np.max(conv))
    shifted = np.clip((conv - cmax) / tau, -500.0, 500.0)
    exp_conv = np.exp(shifted)
    Z = float(np.sum(exp_conv))
    soft_max_val = cmax + tau * np.log(Z)
    w = exp_conv / Z

    w_rev = w[::-1]
    conv_fw = fft_convolve(a, w_rev)
    f1 = conv_fw[n - 1:2 * n - 1][::-1]
    grad_a = 2.0 * f1

    mean_grad = float(np.dot(a, grad_a))
    grad_z = a * (grad_a - mean_grad)
    return soft_max_val, grad_z


def _lfbs_fun(z, tau):
    """Wrapper for scipy minimize with args."""
    return soft_max_obj_and_grad_z(z, tau)


def optimize_lfbs(a_init, tau_list, max_iter=200, time_budget=30.0):

def build_toeplitz(a):
    """Build Toeplitz matrix where (A@g)[k] = (conv(a,g))[k]."""
    n = len(a)
    A = np.zeros((2 * n - 1, n), dtype=np.float64)
    for j in range(n):
        k_start = j
        k_end = min(j + n, 2 * n - 1)
        A[k_start:k_end, j] = a[:k_end - k_start]
    return A


def solve_lp_g(a, M, time_limit=15):
    """Solve LP: max sum(g) s.t. (a*g)[k] <= M for all k, g >= 0. Returns normalized g."""
    n = len(a)
    A = build_toeplitz(a)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': time_limit})
        if res.status == 0:
            g = np.maximum(res.x, 0)
            s = float(np.sum(g))
            if s > 1e-10:
                return g / s
    except Exception:
        pass
    return None


def solve_lp_minpeak(a, time_limit=15):
    """Solve LP: min M s.t. (a*g)[k] <= M, sum(g)=1, g>=0. Returns (g, M_opt)."""
    n = len(a)
    A = build_toeplitz(a)
    c = np.zeros(n + 1)
    c[-1] = 1.0
    A_ub = np.zeros((2 * n - 1, n + 1))
    A_ub[:, :n] = A
    A_ub[:, n] = -1.0
    b_ub = np.zeros(2 * n - 1)
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])
    bounds = [(0.0, None)] * n + [(0.0, None)]
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': time_limit})
        if res.status == 0:
            g = np.maximum(res.x[:n], 0)
            s = np.sum(g)
            if s > 1e-10:
                g = g / s
                M_opt = float(res.x[n])
                return g, M_opt
    except Exception:
        pass
    return None, None

def subgradient_optimize(a, max_iters=2000, time_budget=20.0, tau_start=5.0, tau_end=50.0):
    """Smoothed subgradient optimization with normalized steps."""
    n = len(a)
    a = a.copy().astype(np.float64)
    s = np.sum(a)

        a = a / s

    best_obj = compute_obj(a)

    step = 0.03
    stall = 0

    for it in range(max_iters):
        if time.monotonic() - start > time_budget:

        tau = tau_start + (tau_end - tau_start) * frac

        conv = compute_conv(a)
        max_conv = float(np.max(conv))
        shifted = np.clip(tau * (conv - max_conv), -300, 300)
        weights = np.exp(shifted)
        w_sum = np.sum(weights)
        if w_sum < 1e-300:
            idx = int(np.argmax(conv))
            weights = np.zeros(len(conv))
            weights[idx] = 1.0
        else:
            weights /= w_sum

        w_rev = weights[::-1]
        conv_fw = fft_convolve(a, w_rev)
        grad_a = 2.0 * conv_fw[n - 1:2 * n - 1][::-1]

        grad_centered = grad_a - np.mean(grad_a)
        grad_norm = np.linalg.norm(grad_centered)
        if grad_norm < 1e-15:
            break

        a_new = a - step * grad_centered / grad_norm
        a_new = np.maximum(a_new, 0)

        obj = compute_obj(a)

        if obj < best_obj - 1e-12:
            best_obj = obj
            best_a = a.copy()
            stall = 0
        else:
            stall += 1
            if stall > 60:
                step *= 0.9
                if step < 1e-8:

def peak_directed_optimize(a, max_iters=500, time_budget=15.0):
    """Focus optimization on reducing the top peaks in the convolution."""
    n = len(a)
    a = a.copy().astype(np.float64)

        return a

        if n_peaks == 0:

def resample_sequence(a, new_n):
    """Resample sequence to new length via interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return a_new

    lp_budget = min(remaining * 0.25, 250.0)
    a = alternating_lp(best_seq, max_iters=200, time_budget=lp_budget)
    obj = compute_obj(a)
    if obj < best_score:
        best_score = obj
        best_seq = a.copy()
        print(f"  After LP: {best_score:.8f}")
    else:
        a = best_seq.copy()

    # === Phase 2: L-BFGS-B with geometric tau ===
    remaining = deadline - time.monotonic()
    print(f"\n--- Phase 2: L-BFGS-B ({remaining:.0f}s) ---")
    n_cur = len(a)
    tau_list = [5.0/n_cur, 2.0/n_cur, 1.0/n_cur, 0.5/n_cur,
                0.25/n_cur, 0.125/n_cur, 0.0625/n_cur,
                0.03125/n_cur, 0.015625/n_cur, 0.0078125/n_cur]
    lfbs_budget = min(remaining * 0.2, 150.0)
    a, obj = optimize_lfbs(a, tau_list, max_iter=300, time_budget=lfbs_budget)
    if obj < best_score:
        best_score = obj
        best_seq = a.copy()
        print(f"  After L-BFGS: {best_score:.8f}")
    else:
        a = best_seq.copy()

    # === Phase 3: Multi-round polishing ===
    remaining = deadline - time.monotonic()
    print(f"\n--- Phase 3: Polishing ({remaining:.0f}s) ---")
    polish_budget = min(remaining * 0.35, 300.0)
    a, obj = polish_sequence(a, time_budget=polish_budget, verbose=True)
    if obj < best_score:
        best_score = obj
        best_seq = a.copy()
        print(f"  After polish: {best_score:.8f}")
    else:
        a = best_seq.copy()

    # === Phase 4: Try nearby n values ===
    remaining = deadline - time.monotonic()
    print(f"\n--- Phase 4: Nearby n ({remaining:.0f}s) ---")
    if remaining > 80:
        n_tries = sorted(set([n_best - 20, n_best - 10, n_best + 10, n_best + 20,
                              n_best - 50, n_best + 50, n_best + 100]))
        n_tries = [n for n in n_tries if 200 <= n <= 1200]
        for n_try in n_tries:
            if time.monotonic() > deadline - 40:
                break
            remaining = deadline - time.monotonic()
            if remaining < 20:
                break

            a_try = resample_sequence(best_seq, n_try)
            a_try = np.maximum(a_try, 0)
            s_try = np.sum(a_try)
            if s_try < 1e-10:
                continue
            a_try = a_try / s_try

            # Quick L-BFGS refine
            tau_quick = [1.0/n_try, 0.5/n_try, 0.2/n_try, 0.1/n_try]
            a_try, obj_try = optimize_lfbs(a_try, tau_quick, max_iter=100,
                                           time_budget=min(25, remaining * 0.15))
            # Quick LP
            remaining_sub = deadline - time.monotonic()
            if remaining_sub > 30:
                a_try = alternating_lp(a_try, max_iters=30, time_budget=min(20, remaining_sub * 0.3))
                obj_try = compute_obj(a_try)

            if obj_try < best_score:
                best_score = obj_try
                best_seq = a_try.copy()
                print(f"  n={n_try}: {best_score:.8f} ** UPDATE **")
            else:
                print(f"  n={n_try}: {obj_try:.8f}")

    # === Phase 5: Final intensive optimization ===
    remaining = deadline - time.monotonic()
    print(f"\n--- Phase 5: Final ({remaining:.0f}s) ---")
    if remaining > 30:
        a = best_seq.copy()
        for round_idx in range(6):
            if time.monotonic() > deadline - 15:
                break
            remaining = deadline - time.monotonic()
            if remaining < 20:
                break

            # LP alternating
            a_lp = alternating_lp(a, max_iters=40, time_budget=min(remaining * 0.25, 30.0))
            obj_lp = compute_obj(a_lp)
            if obj_lp < best_score:
                best_score = obj_lp
                best_seq = a_lp.copy()
                a = a_lp.copy()
                print(f"  Final LP {round_idx}: {best_score:.8f}")

            # L-BFGS with very small tau
            remaining = deadline - time.monotonic()
            if remaining > 20:
                n_cur = len(a)
                tau_very_small = [0.05/n_cur, 0.025/n_cur, 0.0125/n_cur]
                a_lb, obj_lb = optimize_lfbs(a, tau_very_small, max_iter=200,
                                             time_budget=min(remaining * 0.2, 18.0))
                if obj_lb < best_score:
                    best_score = obj_lb
                    best_seq = a_lb.copy()
                    a = a_lb.copy()
                    print(f"  Final L-BFGS {round_idx}: {best_score:.8f}")

            # Peak-directed
            remaining = deadline - time.monotonic()
            if remaining > 15:
                a_pk = peak_directed_optimize(a, max_iters=int(remaining * 5),
                                              time_budget=min(remaining * 0.15, 12.0))
                obj_pk = compute_obj(a_pk)
                if obj_pk < best_score:
                    best_score = obj_pk
                    best_seq = a_pk.copy()
                    a = a_pk.copy()
                    print(f"  Final Peak {round_idx}: {best_score:.8f}")

            # Subgradient
            remaining = deadline - time.monotonic()
            if remaining > 12:
                a_sg = subgradient_optimize(a, max_iters=int(remaining * 6),
                                            time_budget=min(remaining * 0.15, 12.0),
                                            tau_start=3.0, tau_end=20.0)
                obj_sg = compute_obj(a_sg)
                if obj_sg < best_score:
                    best_score = obj_sg
                    best_seq = a_sg.copy()
                    a = a_sg.copy()
                    print(f"  Final SG {round_idx}: {best_score:.8f}")

    # Final validation
    if best_seq is None:
        best_seq = np.ones(200)
        best_score = compute_obj(best_seq)

    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    val = evaluate_sequence(result)
    if val < np.inf:
        best_score = val
    else:
        result = [1.0] * 100
        best_score = 2.0

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_score:.8f}, time={elapsed:.1f}s ===")

    return result
