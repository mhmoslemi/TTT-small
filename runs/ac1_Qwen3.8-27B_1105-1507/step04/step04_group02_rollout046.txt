"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 for non-negative sequences.
Uses: weighted full subgradient with backtracking line search, Nesterov momentum
with adaptive restart, periodic LP refinement (HiGHS), two-scale optimization,
targeted symmetry-breaking perturbations, and multi-start exploration.

import numpy as np
from scipy.optimize import linprog
from scipy.sparse import coo_matrix
from scipy.interpolate import interp1d
import time


def compute_conv(a):
    """Compute full linear convolution a*a via FFT."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    conv = np.fft.irfft(fa * fa, size)
    return conv[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    S = float(np.sum(a))
    if S < 1e-10:
        return np.inf
    conv = compute_conv(a)
    M = float(np.max(conv))
    return float(2.0 * n * M / (S * S))


def normalize(a):
    """Normalize to sum=1, clip negatives."""
    a = np.maximum(a, 0.0)
    S = np.sum(a)
    if S < 1e-15:
        return np.ones(len(a)) / len(a)
    return a / S


def weighted_subgradient(a, conv, threshold=0.3):
    """
    Compute weighted subgradient of max(conv) using all peaks above threshold.
    grad[j] = 2 * sum_{k active} conv[k] * a[k-j] / (sum active weights)
    """
    n = len(a)
    M = float(np.max(conv))
    if M < 1e-15:
        return np.zeros(n)

    active_mask = conv > threshold * M
    w = conv * active_mask
    total_w = float(np.sum(w))
    if total_w < 1e-15:
        return np.zeros(n)

    # grad[j] = 2/total_w * sum_{k=0}^{2n-2} w[k] * a[k-j] (valid indices only)
    # For j in [0, n-1]: grad[j] = 2/total_w * sum_{m=0}^{n-1} w[j+m] * a[m] (clipped)
    grad = np.zeros(n)
    # Use FFT for efficiency
    size = 1
    while size < 3 * n:
        size <<= 1
    W = np.fft.rfft(w, size)
    A = np.fft.rfft(a, size)
    # We want cross-correlation: S[j] = sum_m w[j+m] * a[m]
    # This equals (w * a_flipped) convolved appropriately
    # Actually: S[j] = sum_m a[m] * w[m+j] = cross-correlation of a and w at lag j
    # Using FFT: cross_corr = ifft(conj(A) * W) gives sum_m a[m] * w[m+k] for circular
    # For our linear case with proper zero-padding, the first n entries work
    cross = np.fft.irfft(np.conj(A) * W, size)
    # cross[k] = sum_m a[m] * w[m+k] (circular, but with enough padding it's linear)
    # We want S[j] = sum_m a[m] * w[m+j] = cross[j]
    grad = 2.0 * cross[:n] / total_w
    
    # Verify with direct computation for small n (sanity check in debug)
    return grad


def subgradient_with_momentum(a, time_budget, threshold=0.3, momentum=0.7, base_lr=0.1):
    """Subgradient descent with Nesterov momentum and backtracking line search."""
    n = len(a)
    a = normalize(a.copy())
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    
    v = np.zeros(n)
    mu = momentum
    lr = base_lr
    stall_count = 0
    max_stall = 80
    
    obj, conv = compute_obj_and_conv_safe(a)
    
    while time.monotonic() - start < time_budget:
        # Compute subgradient
        grad = weighted_subgradient(a, conv, threshold)
        # Center gradient to preserve sum
        grad = grad - np.mean(grad)
        
        # Nesterov momentum
        v = mu * v + grad
        v_centered = v - np.mean(v)
        
        # Backtracking line search
        alpha = lr
        improved = False
        a_prev = a.copy()
        
        for _ in range(15):
            a_try = a - alpha * v_centered
            a_try = np.maximum(a_try, 0)
            S_try = np.sum(a_try)
            if S_try < 1e-10:
                alpha *= 0.5
                continue
            a_try = a_try / S_try
            obj_try = compute_obj(a_try)
            
            if obj_try < best_obj - 1e-10:
                a = a_try
                best_a = a.copy()
                best_obj = obj_try
                improved = True
                stall_count = 0
                conv = compute_conv(a)
                break
            else:
                alpha *= 0.5
        
        if not improved:
            stall_count += 1
            if stall_count >= max_stall:
                # Restart momentum after stall
                v = np.zeros(n)
                mu = max(0.5, mu * 0.8)
                lr *= 0.7
                stall_count = 0
                if lr < 1e-6:
                    break
        
        # Check time
        if time.monotonic() - start > time_budget * 0.8:
            # Try a few more iterations with smaller steps
            lr *= 0.5
            if lr < 1e-7:
                break
    
    return best_a, best_obj


def compute_obj_and_conv_safe(a):
    """Safe computation of objective and convolution."""
    n = len(a)
    S = float(np.sum(a))
    if S < 1e-10:
        return np.inf, np.zeros(2 * n - 1)
    conv = compute_conv(a)
    M = float(np.max(conv))
    return float(2.0 * n * M / (S * S)), conv


def build_toeplitz(a):
    """Build sparse matrix T such that T @ g = conv(a, g)."""
    n = len(a)
    rows = []
    cols = []
    vals = []
    for j in range(n):
        start = j
        end = min(j + n, 2 * n - 1)
        length = end - start
        rows.extend(range(start, end))
        cols.extend([j] * length)
        vals.extend(a[:length].tolist())
    A = coo_matrix((vals, (rows, cols)), shape=(2 * n - 1, n)).tocsr()
    return A


def solve_lp(a, M, time_limit=10):
    """Solve LP: max sum(g) s.t. conv(a,g) <= M, g >= 0."""
    n = len(a)
    A = build_toeplitz(a)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': time_limit,
                               'dual_feasibility_tolerance': 1e-9})
        if res.status == 0:
            g = np.maximum(res.x, 0)
            if np.sum(g) > 1e-10:
                return g
    except Exception:
        pass
    return None


def alternating_lp_refine(a, num_iters=10, time_budget=60.0):
    """Alternating LP: fix a, solve for g, mix, repeat."""
    a = normalize(a.copy())
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    
    for it in range(num_iters):
        if time.monotonic() - start > time_budget:
            break
        
        conv = compute_conv(a)
        M = float(np.max(conv))
        
        # Try slightly relaxed M for diversity
        M_relaxed = M * 1.02
        g = solve_lp(a, M_relaxed, time_limit=min(8, (time_budget - (time.monotonic() - start)) * 0.5))
        if g is None:
            break
        
        g = normalize(g)
        
        # Line search
        best_t = 0.0
        best_obj_ls = best_obj
        for t in [0.1, 0.2, 0.3, 0.5, 0.7, 1.0]:
            a_try = normalize((1 - t) * a + t * g)
            obj_try = compute_obj(a_try)
            if obj_try < best_obj_ls:
                best_obj_ls = obj_try
                best_t = t
        
        if best_t > 0:
            a = normalize((1 - best_t) * a + best_t * g)
            best_a = a.copy()
            best_obj = best_obj_ls
        else:
            # Nudge towards g
            a = normalize(0.9 * a + 0.1 * g)
    
    return best_a, best_obj


def cosine_init(n, n_modes=10, seed=0):
    """Cosine series initialization."""
    rng = np.random.default_rng(seed)
    k = np.arange(n, dtype=np.float64)
    a = np.ones(n)
    for m in range(1, n_modes + 1):
        c = rng.uniform(-0.8, 0.8) / (m ** 0.7)
        phase = rng.uniform(0, 2 * np.pi)
        a += c * np.cos(2 * np.pi * m * k / n + phase)
    return normalize(a)


def edge_tapered_init(n, edge_width=50, seed=0):
    """Initialization with tapered edges to reduce endpoint effects."""
    rng = np.random.default_rng(seed)
    k = np.arange(n, dtype=np.float64)
    a = np.ones(n)
    # Taper edges
    for i in range(n):
        dist = min(i, n - 1 - i)
        if dist < edge_width:
            a[i] *= 0.3 + 0.7 * (dist / edge_width)
    # Add some structure
    for m in range(1, 6):
        c = rng.uniform(-0.4, 0.4) / m
        a += c * np.cos(2 * np.pi * m * k / n)
    return normalize(a)


def two_scale_optimize(a, time_budget):
    """Downsample, optimize, upsample, refine."""
    n = len(a)
    n_coarse = max(50, n // 2)
    
    # Downsample by averaging pairs
    a_coarse = np.zeros(n_coarse)
    for i in range(n_coarse):
        start = i * (n // n_coarse)
        end = start + (n // n_coarse)
        a_coarse[i] = np.mean(a[start:end])
    a_coarse = normalize(a_coarse)
    
    # Optimize coarse
    a_coarse, obj_coarse = subgradient_with_momentum(
        a_coarse, time_budget=time_budget * 0.3, threshold=0.4, base_lr=0.15)
    
    # Upsample
    x_old = np.linspace(0, 1, n_coarse)
    x_new = np.linspace(0, 1, n)
    f_interp = interp1d(x_old, a_coarse, kind='cubic', fill_value='extrapolate')
    a_up = np.maximum(f_interp(x_new), 0)
    a_up = normalize(a_up)
    
    # Refine at full resolution
    a_ref, obj_ref = subgradient_with_momentum(
        a_up, time_budget=time_budget * 0.5, threshold=0.3, base_lr=0.08)
    
    return a_ref, obj_ref

    elif mode == 1:

        width = n // 5
        center_mask = np.zeros(n)

    elif mode == 2:

        if obj_cand > best_score + 0.1:
            continue
        
        n_cand = len(a_cand)
        t_alloc = min(150.0, (deadline - time.monotonic()) * 0.3)
        print(f"  Refining candidate {idx} (n={n_cand}, obj={obj_cand:.6f}), budget={t_alloc:.0f}s")
        
        a_ref, obj_ref = refine_pipeline(a_cand, time_budget=t_alloc, use_lp=True)
        if obj_ref < best_score:
            best_score = obj_ref
            best_seq = a_ref.copy()
            print(f"    ** NEW BEST: {best_score:.6f} **")
        else:
            print(f"    Best after refine: {obj_ref:.6f} (global best: {best_score:.6f})")
    
    # === Phase 3: Two-scale optimization (next 200s) ===
    print("\n=== Phase 3: Two-scale optimization ===")
    if time.monotonic() < deadline - 150:
        t_alloc = min(120.0, (deadline - time.monotonic()) * 0.5)
        print(f"  Two-scale (budget={t_alloc:.0f}s)")
        a_2s, obj_2s = two_scale_optimize(best_seq, time_budget=t_alloc)
        if obj_2s < best_score:
            best_score = obj_2s
            best_seq = a_2s.copy()
            print(f"    ** NEW BEST: {best_score:.6f} **")
        else:
            print(f"    Two-scale result: {obj_2s:.6f}")
    
    # === Phase 4: Intensive perturbation search (next 300s) ===
    print("\n=== Phase 4: Intensive perturbation search ===")
    rng = np.random.default_rng(seed + 7777)
    pert_round = 0
    
    while time.monotonic() < deadline - 100:
        pert_round += 1
        remaining = deadline - time.monotonic()
        if remaining < 30:
            break
        
        # Try multiple perturbations
        for mode in range(5):
            if time.monotonic() > deadline - 80:
                break
            
            a_pert = targeted_perturb(best_seq, rng, mode=mode)
            t_pert = min(20.0, remaining * 0.1)
            a_p, obj_p = subgradient_with_momentum(
                a_pert, time_budget=t_pert, threshold=0.3, base_lr=0.08)
            
            if obj_p < best_score:
                best_score = obj_p
                best_seq = a_p.copy()
                print(f"  Pert round {pert_round}, mode {mode}: ** NEW BEST: {best_score:.6f} **")
            elif pert_round % 5 == 0:
                print(f"  Pert round {pert_round}, mode {mode}: {obj_p:.6f}")
    
    # === Phase 5: Final LP + subgradient polish (last 100s) ===
    print("\n=== Phase 5: Final polish ===")
    if time.monotonic() < deadline - 10:
        t_final = min(60.0, deadline - time.monotonic() - 5)
        n_best = len(best_seq)
        
        # LP polish
        if n_best <= 600:
            a_lp, obj_lp = alternating_lp_refine(
                best_seq, num_iters=15, time_budget=t_final * 0.5)
            if obj_lp < best_score:
                best_score = obj_lp
                best_seq = a_lp.copy()
                print(f"  LP polish: ** NEW BEST: {best_score:.6f} **")
        
        # Final subgradient
        if time.monotonic() < deadline - 5:
            t_sg = max(5.0, deadline - time.monotonic() - 3)
            a_final, obj_final = subgradient_with_momentum(
                best_seq, time_budget=t_sg, threshold=0.2, base_lr=0.03)
            if obj_final < best_score:
                best_score = obj_final
                best_seq = a_final.copy()
                print(f"  Final SG: ** NEW BEST: {best_score:.6f} **")
    
    # === Final validation ===
    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    val = evaluate_sequence(result)
    if val < np.inf:
        if val < best_score:
            best_score = val
    else:
        print("WARNING: evaluate_sequence returned inf, trying fallback")
        result = [1.0] * 300
        best_score = evaluate_sequence(result)
    
    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_score:.6f}, time={elapsed:.1f}s ===")
    
    return result
