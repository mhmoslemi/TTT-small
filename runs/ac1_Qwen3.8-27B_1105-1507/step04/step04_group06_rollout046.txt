```
def hot_spot_step(a, p, step_size):
    # p is the probability distribution over lags (from softmax of conv)
    n = len(a)
    # Compute "blame" for each index
    # blame[j] = sum_k p[k] * a[k-j] (this is exactly h[j] from before!)
    r = np.convolve(a, p[::-1], mode='full')
    blame = r[(2*n-2) - np.arange(n)]
    
    # Reduce high-blame indices, increase low-blame indices
    # Use a multiplicative update: a_new[j] = a[j] * exp(-step_size * (blame[j] - mean(blame)))
    a_new = a * np.exp(-step_size * (blame - blame.mean()))
    a_new = np.maximum(a_new, 1e-10)
    a_new = a_new / a_new.sum()
    return a_new
```

```python
Minimizes J(a) = 2n*max(conv(a,a))/(sum(a))^2 using:
1. FFT-based gradient computation for fast smooth objectives
2. Mirror descent with multiplicative updates (no LP needed)
3. Adaptive step size with backtracking
4. L-BFGS-B with adaptive beta for basin finding

import numpy as np
from scipy.optimize import minimize
import time


def fast_conv(a):
    """Full self-convolution via FFT."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]


def compute_J(a):
    """Compute J(a) = 2*n*max(conv(a,a))/(sum(a))^2."""
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))


def compute_J_normalized(a):
    """Compute J for normalized a (sum=1). Returns max conv value."""
    # Assumes sum(a) = 1 already
    b = fast_conv(a)
    return float(2 * len(a) * b.max())


def smooth_max_and_probs(a, beta):
    """
    Compute smooth max of conv(a,a) and the probability distribution.
    Returns (smooth_value, prob_vector, exact_max).
    """
    b = fast_conv(a)
    b_max = float(b.max())
    exponents = beta * (b - b_max)
    exponents = np.clip(exponents, -500, 500)
    exp_vals = np.exp(exponents)
    Z = float(exp_vals.sum())
    if Z < 1e-300:
        p = np.zeros_like(b)
        p[np.argmax(b)] = 1.0
    else:
        p = exp_vals / Z
    smooth_val = b_max + np.log(max(Z, 1e-300)) / beta
    return smooth_val, p, b_max


def gradient_smooth_conv(a, p):
    """
    Compute gradient of sum_k p[k]*conv(a,a)[k] w.r.t. a.
    Uses FFT-based convolution.
    """
    n = len(a)
    # We need h[j] = sum_k p[k] * a[k-j] for j = 0,...,n-1
    # h[j] = r[2n-2-j] where r = conv(a, p[::-1])
    r = np.convolve(a, p[::-1], mode='full')
    indices = (2 * n - 2) - np.arange(n)
    return 2.0 * r[indices]


def objective_and_grad(a, beta):
    """
    Compute F(a) = 2n * smooth_max_conv(a) / (sum(a))^2 and its gradient.
    """
    n = len(a)
    s = float(a.sum())
    if s < 1e-10:
        return 1e10, np.zeros(n)
    
    smooth_val, p, _ = smooth_max_and_probs(a, beta)
    F_val = 2.0 * n * smooth_val / (s * s)
    
    grad_conv = gradient_smooth_conv(a, p)
    # d/d a[j] of F = 2n * (s * grad_conv[j] - 2 * smooth_val) / s^3
    grad_F = 2.0 * n * (s * grad_conv - 2.0 * smooth_val) / (s ** 3)
    
    return F_val, grad_F


def mirror_descent_step(a, step_size):
    """
    One mirror descent step: compute blame, apply multiplicative update.
    """
    n = len(a)
    b = fast_conv(a)
    b_max = float(b.max())
    
    # Use a moderate beta for the probability distribution
    beta = 20.0
    exponents = beta * (b - b_max)
    exponents = np.clip(exponents, -500, 500)
    exp_vals = np.exp(exponents)
    Z = float(exp_vals.sum())
    if Z < 1e-300:
        p = np.zeros_like(b)
        p[np.argmax(b)] = 1.0
    else:
        p = exp_vals / Z
    
    # Compute blame
    blame = gradient_smooth_conv(a, p) / 2.0  # This is h[j] = sum_k p[k]*a[k-j]
    
    # Multiplicative update
    mean_blame = float(blame.mean())
    a_new = a * np.exp(-step_size * (blame - mean_blame))
    a_new = np.maximum(a_new, 1e-15)
    a_new = a_new / a_new.sum()
    return a_new


def mirror_descent_loop(a, deadline, tracker, max_iters=100000, 
                        initial_step=0.5, min_step=0.01, 
                        patience=200, beta_for_probs=20.0):
    """
    Mirror descent with adaptive step size.
    """
    a = a.copy()
    a = np.maximum(a, 1e-10)
    a = a / a.sum()
    
    current_J = compute_J(a)
    tracker.update(a)
    
    step_size = initial_step
    best_J = current_J
    best_a = a.copy()
    stall = 0
    
    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        
        # Mirror descent step
        b = fast_conv(a)
        b_max = float(b.max())
        
        exponents = beta_for_probs * (b - b_max)
        exponents = np.clip(exponents, -500, 500)
        exp_vals = np.exp(exponents)
        Z = float(exp_vals.sum())
        if Z < 1e-300:
            p = np.zeros_like(b)
            p[np.argmax(b)] = 1.0
        else:
            p = exp_vals / Z
        
        blame = gradient_smooth_conv(a, p) / 2.0
        mean_blame = float(blame.mean())
        
        a_new = a * np.exp(-step_size * (blame - mean_blame))
        a_new = np.maximum(a_new, 1e-15)
        a_new = a_new / a_new.sum()
        
        J_new = compute_J(a_new)
        
        if J_new < current_J:
            a = a_new
            current_J = J_new
            stall = 0
            if J_new < best_J:
                best_J = J_new
                best_a = a.copy()
                tracker.update(a)
            # Increase step size slightly if improving
            step_size = min(step_size * 1.05, 5.0)
        else:
            stall += 1
            step_size = max(step_size * 0.8, min_step)
            if stall >= patience:
                break
    
    return best_a

    """L-BFGS-B optimization with beta annealing."""

def _lbfgs_single_beta(a, beta, max_iters, deadline):
    """Single L-BFGS-B run at fixed beta."""
    n = len(a)
    
    # Use a closure-free approach
    # We'll use a class-like approach with global state... 
    # Actually, let's just use the minimize function directly
    
    def func(x):
        x2 = np.maximum(x, 1e-10)
        f, _ = objective_and_grad(x2, beta)
        return f if np.isfinite(f) else 1e10
    
    def jac(x):
        x2 = np.maximum(x, 1e-10)
        _, g = objective_and_grad(x2, beta)
        g = np.nan_to_num(g, nan=0.0, posinf=0.0, neginf=0.0)
        return g
    
    # Use bounded L-BFGS-B
    bounds = [(1e-10, None)] * n
    
    try:
        time_left = deadline - time.monotonic()
        if time_left < 5:
            return a
        res = minimize(func, a.copy(), jac=jac, method='L-BFGS-B',
                       bounds=bounds,
                       options={'maxiter': max_iters, 'max_fun': max_iters * 50,
                                'gtol': 1e-7, 'ftol': 1e-13})
        if res.fun < 1e10:
            new_a = np.maximum(res.x, 1e-8)
            if new_a.sum() > 0.01:
                return new_a / new_a.sum()
    except Exception:
        pass
    
    return a


def resample_seq(src, dest_n):
    """Resample 1D sequence to new length."""
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    result = np.interp(t_new, t_old, src)
    return np.maximum(result, 0.0)
```
