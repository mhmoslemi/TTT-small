import numpy as np
import time
from scipy.optimize import linprog

    HAS_CVXPY = False

# Precomputed sparsity cache
_spar_cache = {}

def get_sparsity(n):
    if n not in _spar_cache:
        rows = []
        cols = []
        for k in range(2 * n - 1):
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            rows.extend([k] * (je - js + 1))
            cols.extend(range(js, je + 1))
        _spar_cache[n] = (np.array(rows, dtype=np.int32), np.array(cols, dtype=np.int32), len(rows))
    return _spar_cache[n]

def fast_conv(a):
    n = len(a)
    s = 1
    while s < 2 * n - 1:
        s <<= 1
    fa = np.fft.rfft(a, s)
    return np.fft.irfft(fa * fa, s)[:2 * n - 1]

def compute_J(a):
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))

def cross_correlation_fft(p, a):
    """Compute cross[j] = sum_m p[m+j]*a[m] for j=0..n-1 using FFT."""
    n = len(a)
    s = 1
    while s < 2 * n:
        s <<= 1
    fp = np.fft.rfft(p, s)
    fa = np.fft.rfft(a, s)
    result = np.fft.irfft(fp * np.conj(fa), s)
    return result[:n]

def lp_best_response_cvxpy(a, M, deadline):
    """Solve LP using CVXPY: max sum(g) s.t. A@g <= M, g >= 0."""
    n = len(a)
    g_var = cp.Variable(n, nonneg=True)
    # Build the constraint matrix as a CVXPY expression
    # A[k, j] = a[k-j] if 0 <= k-j < n
    # Use the fact that A is a convolution matrix
    # A @ g = convolve_restricted(a, g)
    # We'll build it explicitly for clarity
    rows, cols, nnz = get_sparsity(n)
    vals = np.empty(nnz)
    for i in range(nnz):
        k = rows[i]
        j = cols[i]
        vals[i] = a[k - j]
    A = csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))
    constraints = [A @ g_var <= M]
    objective = cp.Maximize(cp.sum(g_var))
    prob = cp.Problem(objective, constraints)
    try:
        prob.solve(solver=cp.ECOS, max_time=max(1, deadline - time.monotonic()) if deadline > time.monotonic() else 10)
        if g_var.value is not None:
            return np.maximum(g_var.value, 0.0)
    except Exception:
        pass
    return None

def lp_best_response_scipy(a, M, deadline):
    """Solve LP using scipy linprog."""
    n = len(a)
    rows, cols, nnz = get_sparsity(n)
    vals = np.empty(nnz)
    for i in range(nnz):
        k = rows[i]
        j = cols[i]
        vals[i] = a[k - j]
    A = csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    remaining = deadline - time.monotonic() if deadline else 10
    tl = max(1.0, min(15.0, remaining))
    r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                options={'presolve': True, 'time_limit': tl})
    if r.success and r.x is not None:
        return np.maximum(r.x, 0.0)
    return None

def lp_best_response(a, M, deadline):
    if HAS_CVXPY:
        g = lp_best_response_cvxpy(a, M, deadline)
        if g is not None and g.sum() > 1e-10:
            return g
    return lp_best_response_scipy(a, M, deadline)

def run_lp_opt(a, max_iters, deadline, tracker, tighten=0.01):
    a = np.maximum(a.astype(np.float64), 1e-10)
    a /= a.sum()
    n = len(a)
    current_J = compute_J(a)
    tracker.update(a)

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        b = fast_conv(a)
        M = float(b.max())
        # Adaptive tightening: start aggressive, relax if no progress
        M_tight = M * (1.0 - tighten)
        g = lp_best_response(a, M_tight, deadline)
        if g is None or g.sum() < 1e-10:
            g = lp_best_response(a, M, deadline)
        if g is None or g.sum() < 1e-10:
            break
        g = np.maximum(g, 0.0)
        gs = g.sum()
        if gs < 1e-10:
            break
        g /= gs

        best_a = None
        best_J = current_J
        progress = it / max(1, max_iters)
        base_t = max(0.05, 0.4 * (1.0 - progress))
        for t in [base_t, base_t * 1.5, base_t * 0.5, 0.7, 0.3, 0.1]:
            if t > 1.0:
                continue
            a_trial = (1.0 - t) * a + t * g
            a_trial = np.maximum(a_trial, 1e-10)
            ss = a_trial.sum()
            if ss < 1e-10:
                continue
            a_trial /= ss
            J_trial = compute_J(a_trial)
            if J_trial < best_J - 1e-15:
                best_J = J_trial
                best_a = a_trial
        if best_a is None:
            break
        a = best_a
        current_J = best_J
        if it % 5 == 0:
            tracker.update(a)
    tracker.update(a)
    return a

def lse_adam_descent(a, max_iters, deadline, tracker):
    """LSE gradient descent with Adam optimizer."""
    a = np.maximum(a.astype(np.float64), 1e-8)
    a /= a.sum()
    n = len(a)
    current_J = compute_J(a)
    tracker.update(a)

    tau = 20.0
    lr0 = 0.01
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8

    m = np.zeros(n)
    v = np.zeros(n)
    t = 0
    no_improve = 0

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        b = fast_conv(a)
        b_max = b.max()
        exp_b = np.exp(tau * (b - b_max))
        p = exp_b / exp_b.sum()

        # Gradient of LSE: grad[j] = 2 * sum_k p[k] * a[k-j] * I(0<=k-j<n)
        # = 2 * cross_correlation(p, a)[j]
        grad = 2.0 * cross_correlation_fft(p, a)

        # Also include the -4nM*a/s^3 term for full J gradient (s=1)
        M = b_max
        grad_full = grad - 4.0 * M * n * a / tau  # scaled appropriately

        t += 1
        m = beta1 * m + (1 - beta1) * grad_full
        v = beta2 * v + (1 - beta2) * grad_full ** 2
        m_hat = m / (1 - beta1 ** t)
        v_hat = v / (1 - beta2 ** t)

        a_new = a - lr0 * m_hat / (np.sqrt(v_hat) + eps)
        a_new = np.maximum(a_new, 1e-8)
        a_new /= a_new.sum()
        J_new = compute_J(a_new)

        if J_new < current_J - 1e-16:
            a = a_new
            current_J = J_new
            no_improve = 0
            # Slowly increase tau
            tau = min(200.0, tau * 1.005)
        else:
            no_improve += 1
            if no_improve > 50:
                lr0 *= 0.5
                tau = max(5.0, tau * 0.8)
                no_improve = 0
                if lr0 < 1e-8:
                    break
        if it % 200 == 0:
            tracker.update(a)
    tracker.update(a)
    return a

def subgradient_nesterov(a, max_iters, deadline, tracker):
    """Subgradient descent with Nesterov acceleration on exact max."""
    a = np.maximum(a.astype(np.float64), 1e-8)
    a /= a.sum()
    n = len(a)
    current_J = compute_J(a)
    tracker.update(a)

    lr0 = 1.0 / n
    beta = 0.9
    y = a.copy()
    v = np.zeros(n)
    rng = np.random.default_rng(123)

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        b = fast_conv(y)
        M = float(b.max())
        active = np.where(b >= M - 1e-9 * max(M, 1e-10))[0]

        # Subgradient of max: 2 * cross_correlation(active_indicator, a) / len(active)
        active_vec = np.zeros(2 * n - 1)
        active_vec[active] = 1.0 / len(active)
        grad_m = 2.0 * cross_correlation_fft(active_vec, y)

        # Full gradient of J (s=1): 2n*grad_M - 4nM*a
        grad = 2.0 * n * grad_m - 4.0 * n * M * y

        lr = lr0 / (1 + it * 0.01)
        v = beta * v + grad
        y_new = y - lr * v
        a_new = (1 - (1 + np.sqrt(1 + 4 * it)) // (2 * (it + 1))) * a + ((1 + np.sqrt(1 + 4 * it)) // (2 * (it + 1))) * y_new
        # Simplified Nesterov: just use standard momentum
        a_new = y_new
        a_new = np.maximum(a_new, 1e-8)
        a_new /= a_new.sum()
        J_new = compute_J(a_new)

        if J_new < current_J:
            a = a_new
            current_J = J_new
        y = a

        if it % 100 == 0:
            tracker.update(a)
    tracker.update(a)
    return a

def flat_spectrum_init(n, rng):
    """Generate sequence with approximately flat power spectrum."""
    half = n // 2 + 1
    angles = rng.uniform(0, 2 * np.pi, half)
    A = np.zeros(n, dtype=complex)
    A[:half] = np.exp(1j * angles[:half])
    for k in range(1, n - n // 2):
        A[k] = np.conj(A[n - k])
    a = np.fft.ifft(A).real
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-10:
        return np.ones(n) / n
    return a / s

def resample_seq(src, dest_n):
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)

def generate_inits(n, rng, parent):
    inits = []
    if parent is not None and parent.size > 0:
        inits.append(resample_seq(parent, n))
        inits.append(resample_seq(parent[::-1], n))

    inits.append(np.ones(n) / n)
    inits.append(rng.uniform(0.1, 1.0, n))
    inits.append(flat_spectrum_init(n, rng))
    inits.append(flat_spectrum_init(n, rng))
    inits.append(flat_spectrum_init(n, rng))

    t = np.linspace(0, 1, n, endpoint=False)
    for freq in [2, 3, 5, 7, 11, 13]:
        inits.append(1.0 + 0.4 * np.cos(2 * np.pi * freq * t + rng.uniform(0, 6.28)))
    inits.append(1.0 + 0.3 * np.cos(2 * np.pi * 3.7 * t) + 0.2 * np.cos(2 * np.pi * 7.1 * t))
    inits.append(0.5 + 0.5 * np.cos(np.pi * t) ** 2)
    inits.append(0.3 + 0.7 * (1.0 - np.abs(2 * t - 1.0)) ** 0.5)
    inits.append(0.5 * np.exp(-6 * (t - 0.3) ** 2) + 0.5 * np.exp(-6 * (t - 0.7) ** 2) + 0.05)

    # Multi-frequency
    inits.append(1.0 + 0.25 * np.sin(2 * np.pi * t) + 0.2 * np.sin(4 * np.pi * t)
                + 0.15 * np.sin(7 * np.pi * t) + 0.1 * np.sin(11 * np.pi * t))

    # Power-law
    inits.append((t + 0.1) ** rng.uniform(-1.0, 1.0))
    inits.append((1.0 - t + 0.1) ** rng.uniform(-1.0, 1.0))

    # Sparse + smooth
    sparse = np.zeros(n)
    n_spikes = max(5, n // 15)
    positions = rng.choice(n, size=n_spikes, replace=False)
    sparse[positions] = 1.0
    kernel_size = max(3, n // 30)
    kernel = np.ones(kernel_size) / kernel_size
    inits.append(np.convolve(sparse, kernel, mode='same') + 0.05)

    # Random Fourier with limited bandwidth
    n_coeff = min(n // 2 + 1, 16)
    coeffs = rng.normal(0, 1, n_coeff)
    sig = np.zeros(n)
    for i in range(n_coeff):
        sig += coeffs[i] * np.cos(2 * np.pi * i * t / n + rng.uniform(0, 6.28))
    inits.append(np.abs(sig) + 0.5)

    # "Chebyshev-like" polynomials
    x = 2 * t - 1
    for deg in [4, 6, 8]:
        cheb = np.polynomial.chebyshev.chebval(x, np.ones(deg + 1) * (-1) ** np.arange(deg + 1))
        inits.append(np.abs(cheb) + 0.3)

    for x in inits:
        x = np.maximum(np.asarray(x, dtype=np.float64), 0.001)
        x /= x.sum()
    return inits

class ScoreTracker:
    def __init__(self):
        self.best = None
        self.best_score = float('inf')
        self.t_start = time.monotonic()

    def update(self, a):
        a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
        if a.size == 0 or a.sum() < 0.01:
            return
        score = float(evaluate_sequence(a.tolist()))
        if np.isfinite(score) and score < self.best_score:
            self.best = a.copy()
            self.best_score = score
            elapsed = time.monotonic() - self.t_start
            print(f"  * J={score:.8f} n={a.size} t={elapsed:.1f}s", flush=True)

    def report(self, label=""):
        if self.best is not None:
            print(f"  {label} best: J={self.best_score:.8f} n={self.best.size}", flush=True)

def coordinate_descent(a, max_steps, deadline, tracker):
    """Random coordinate descent with multiple step sizes."""
    a = a.copy().astype(np.float64)
    a /= a.sum()
    n = len(a)
    current_J = compute_J(a)
    tracker.update(a)
    rng = np.random.default_rng(42)

    for step in range(max_steps):
        if time.monotonic() > deadline:
            break
        # Try a few random coordinates
        for _ in range(3):
            if time.monotonic() > deadline:
                break
            j = rng.integers(0, n)
            for scale in [0.7, 1.3, 0.5, 1.5, 0.85, 1.15]:
                a_trial = a.copy()
                a_trial[j] *= scale
                a_trial = np.maximum(a_trial, 1e-10)
                ss = a_trial.sum()
                if ss < 1e-10:
                    continue
                a_trial /= ss
                J_trial = compute_J(a_trial)
                if J_trial < current_J - 1e-15:
                    a = a_trial
                    current_J = J_trial
                    break
        if step % 200 == 0:
            tracker.update(a)
    tracker.update(a)
    return a

def pair_descent(a, max_steps, deadline, tracker):
    """Try pairwise coordinate updates for better mixing."""
    a = a.copy().astype(np.float64)
    a /= a.sum()
    n = len(a)
    current_J = compute_J(a)
    tracker.update(a)
    rng = np.random.default_rng(99)

    for step in range(max_steps):
        if time.monotonic() > deadline:
            break
        j = rng.integers(0, n)
        k = rng.integers(0, n)
        if j == k:
            continue
        for ratio in [0.5, 1.5, 0.3, 2.0, 0.7, 1.3]:
            a_trial = a.copy()
            total = a_trial[j] + a_trial[k]
            a_trial[j] = total * ratio / (1 + ratio)
            a_trial[k] = total * 1 / (1 + ratio)
            a_trial = np.maximum(a_trial, 1e-10)
            ss = a_trial.sum()
            if ss < 1e-10:
                continue
            a_trial /= ss
            J_trial = compute_J(a_trial)
            if J_trial < current_J - 1e-15:
                a = a_trial
                current_J = J_trial
                break
        if step % 100 == 0:
            tracker.update(a)
    tracker.update(a)
    return a

def simulated_annealing(a, deadline, tracker, n_restarts=30, t0=0.12):
    rng = np.random.default_rng(777)
    n = a.size
    a = a.copy().astype(np.float64)
    current_J = compute_J(a)
    tracker.update(a)
    temp = t0

    for r in range(n_restarts):
        if time.monotonic() > deadline:
            break
        a_pert = a + rng.normal(0, temp, n)
        a_pert = np.maximum(a_pert, 1e-6)
        a_pert /= a_pert.sum()

        # Short LP refinement
        n_lp = max(3, 15 - r // 4)
        a_refined = run_lp_opt(a_pert, n_lp, deadline, tracker, tighten=0.02)

        J = compute_J(a_refined)
        delta = J - current_J
        if delta < 0 or rng.random() < np.exp(-delta / max(temp, 1e-10)):
            a = a_refined
            current_J = J
        temp *= 0.93
    tracker.update(a)
    return a

def propose_candidate(seed=42, budget_s=1500, **kwargs):
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + float(budget_s) - 25.0
    tracker = ScoreTracker()

    print(f"CVXPY available: {HAS_CVXPY}", flush=True)

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        if p.ndim == 1 and p.size > 0 and np.all(np.isfinite(p)):
            parent = np.maximum(p, 0.0)
            tracker.update(parent)
            print(f"Parent: n={parent.size}, J={tracker.best_score:.8f}", flush=True)
    except (NameError, AttributeError):
        pass

    # === Phase 1: Multi-length LP optimization ===
    lengths = [512, 384, 448]
    if parent is not None:
        lengths.insert(0, parent.size)
    # Deduplicate and sort
    lengths = sorted(set(lengths))

    total_p1_time = min(700.0, (deadline - time.monotonic()) * 0.55)
    time_per_length = total_p1_time / len(lengths)

    for n in lengths:
        t_phase_start = time.monotonic()
        t_phase_end = min(deadline - 200, t_phase_start + time_per_length)
        if time.monotonic() >= t_phase_end - 30:
            continue
        print(f"\n{'='*50} Phase1 n={n} budget={time_per_length:.0f}s {'='*50}", flush=True)

        inits = generate_inits(n, rng, parent)
        best_for_n = None
        best_J_for_n = float('inf')
        n_inits = min(len(inits), 8)  # Limit number of inits to manage time

        init_budget = max(5.0, (t_phase_end - time.monotonic()) / n_inits)

        for idx in range(n_inits):
            if time.monotonic() > t_phase_end - 20:
                break
            a0 = inits[idx]
            print(f"  Init {idx}/{n_inits} (budget {init_budget:.1f}s)", flush=True)

            # LP phase - adaptive iterations based on time
            n_lp_iters = max(8, int(45.0 / max(1, n / 256)))
            a = run_lp_opt(a0, n_lp_iters, t_phase_end - 60, tracker, tighten=0.01)

            # LSE Adam phase
            if time.monotonic() < t_phase_end - 40:
                n_lse = max(500, int(20000 / n))
                a = lse_adam_descent(a, n_lse, t_phase_end - 40, tracker)

            # Subgradient Nesterov
            if time.monotonic() < t_phase_end - 20:
                a = subgradient_nesterov(a, max(200, int(10000 / n)), t_phase_end - 20, tracker)

            J = compute_J(a)
            if J < best_J_for_n:
                best_J_for_n = J
                best_for_n = a.copy()

        if best_for_n is not None:
            tracker.update(best_for_n)
            if parent is None or best_J_for_n < compute_J(resample_seq(parent, best_for_n.size)):
                parent = best_for_n
            tracker.report(f"Phase1 n={n} ({time.monotonic()-t_phase_start:.1f}s): J={best_J_for_n:.8f}")

    # === Phase 2: Focused optimization on best length ===
    if tracker.best is not None and time.monotonic() < deadline - 150:
        t0 = time.monotonic()
        n = tracker.best.size
        print(f"\n{'='*50} Phase2 Focus n={n} {'='*50}", flush=True)

        a = tracker.best.copy().astype(np.float64)

        # Multiple LP passes with increasing tightening
        tighten_sched = [0.005, 0.008, 0.01, 0.015, 0.02, 0.025, 0.03, 0.04]
        for tight in tighten_sched:
            if time.monotonic() > deadline - 100:
                break
            a = run_lp_opt(a, 20, deadline - 100, tracker, tighten=tight)

        # LSE Adam (extended)
        if time.monotonic() < deadline - 80:
            a = lse_adam_descent(a, 5000, deadline - 80, tracker)

        # Subgradient
        if time.monotonic() < deadline - 50:
            a = subgradient_nesterov(a, 3000, deadline - 50, tracker)

        tracker.report(f"Phase2 ({time.monotonic()-t0:.1f}s)")

    # === Phase 3: Simulated annealing ===
    if tracker.best is not None and time.monotonic() < deadline - 80:
        t0 = time.monotonic()
        n = tracker.best.size
        print(f"\n{'='*50} Phase3 SA n={n} {'='*50}", flush=True)
        a = simulated_annealing(tracker.best, deadline - 60, tracker, n_restarts=30, t0=0.1)
        # Polish after SA
        if time.monotonic() < deadline - 40:
            a = run_lp_opt(a, 15, deadline - 40, tracker, tighten=0.01)
            if time.monotonic() < deadline - 20:
                a = lse_adam_descent(a, 2000, deadline - 20, tracker)
        tracker.report(f"Phase3 ({time.monotonic()-t0:.1f}s)")

    # === Phase 4: Coordinate/pair descent polish ===
    if tracker.best is not None and time.monotonic() < deadline - 15:
        t0 = time.monotonic()
        print(f"\n{'='*50} Phase4 CoordDescent {'='*50}", flush=True)
        a = coordinate_descent(tracker.best, 5000, min(deadline - 10, time.monotonic() + 40), tracker)
        a = pair_descent(a, 3000, min(deadline - 8, time.monotonic() + 30), tracker)

        # Final LP + LSE
        if time.monotonic() < deadline - 5:
            a = run_lp_opt(a, 10, deadline - 5, tracker, tighten=0.005)
            if time.monotonic() < deadline - 2:
                a = lse_adam_descent(a, 1000, deadline - 2, tracker)
        tracker.report(f"Phase4 ({time.monotonic()-t0:.1f}s)")

    # === Phase 5: Try a different length for diversity ===
    if time.monotonic() < deadline - 30:
        t0 = time.monotonic()
        print(f"\n{'='*50} Phase5 Alt Length {'='*50}", flush=True)
        alt_n = 640 if (tracker.best is None or tracker.best.size != 640) else 320
        inits = generate_inits(alt_n, rng, tracker.best)
        for idx in range(min(4, len(inits))):
            if time.monotonic() > deadline - 10:
                break
            a0 = inits[idx]
            a = run_lp_opt(a0, 10, deadline - 20, tracker, tighten=0.01)
            if time.monotonic() < deadline - 10:
                a = lse_adam_descent(a, 1000, deadline - 10, tracker)
        tracker.report(f"Phase5 n={alt_n} ({time.monotonic()-t0:.1f}s)")

    # Fallback
    if tracker.best is None:
        tracker.best = np.ones(256)
        tracker.best_score = compute_J(tracker.best)
        tracker.update(tracker.best)

    print(f"\n{'='*50} FINAL {'='*50}", flush=True)
    print(f"J={tracker.best_score:.8f} n={tracker.best.size} time={time.monotonic()-tracker.t_start:.1f}s", flush=True)
    return [float(x) for x in tracker.best]
