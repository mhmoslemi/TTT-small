import numpy as np
from scipy.optimize import linprog
from scipy.sparse import csr_matrix
import time

_spar_cache = {}

def get_conv_matrix_sparse(n):
    """Get sparse representation of the convolution constraint matrix."""
    if n not in _spar_cache:
        rows = []
        cols = []
        for k in range(2 * n - 1):
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            rows.extend([k] * (je - js + 1))
            cols.extend(range(js, je + 1))
        _spar_cache[n] = (np.array(rows, dtype=np.int32), np.array(cols, dtype=np.int32), len(rows))
    return _spar_cache[n]


def fast_conv(a):
    """Full linear convolution via FFT."""
    n = len(a)
    s = 1
    while s < 2 * n - 1:
        s <<= 1
    fa = np.fft.rfft(a, s)
    return np.fft.irfft(fa * fa, s)[:2 * n - 1]


def compute_J(a):
    """Compute objective J(a) = 2n*max(conv(a,a))/(sum(a))^2."""
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))


def lp_best_response(a, M, time_limit=5.0):
    """
    Solve LP: max sum(g) s.t. (a*g)_k <= M for all k, g >= 0.
    Returns normalized g or None if infeasible.
    """
    n = len(a)
    rows, cols, nnz = get_conv_matrix_sparse(n)
    vals = np.empty(nnz)
    for i in range(nnz):
        k = rows[i]
        j = cols[i]
        idx = k - j
        if 0 <= idx < n:
            vals[i] = a[idx]
        else:
            vals[i] = 0.0
    A = csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    try:
        r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                    options={'presolve': True, 'time_limit': time_limit})
        if r.success and r.x is not None:
            g = np.maximum(r.x, 0.0)
            gs = g.sum()
            if gs > 1e-10:
                return g / gs
        elif r.status == 2:  # infeasible
            return None
    except Exception:
        pass
    return None

        best_J = current_J

        base_t = max(0.1, 0.6 * (1.0 - 0.5 * progress))

    a /= a.sum()
    n = len(a)
    current_J = compute_J(a)
    tracker.update(a)

    tau = 30.0
    lr = 2.0 / (n * tau)
    momentum = 0.8
    v = np.zeros(n)
    patience = 0
    max_patience = 200

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        b = fast_conv(a)
        b_max = b.max()
        # Softmax weights
        shifted = tau * (b - b_max)
        # Clip for numerical stability
        shifted = np.clip(shifted, -50, 0)
        exp_b = np.exp(shifted)
        p = exp_b / exp_b.sum()

        # Gradient of LSE: 2 * (p * a circular cross-correlation)[:n]
        # (a*a)_k = sum_j a_j a_{k-j}, d/db_k / da_j = a_{k-j}
        # grad_a_j = 2 * sum_k p_k * a_{k-j} (where valid)
        s_fft = 1
        while s_fft < 3 * n:
            s_fft <<= 1
        fp = np.fft.rfft(p, s_fft)
        fa = np.fft.rfft(a, s_fft)
        # We need sum_k p_k * a_{k-j} for j=0..n-1
        # This is the cross-correlation of p and a, evaluated at lags 0..n-1
        corr = np.fft.irfft(fp * fa, s_fft)
        # corr[m] = sum_k p_k * a_{k-m} (circular), we want the linear part
        grad = 2.0 * corr[:n]

        v = momentum * v + (1.0 - momentum) * grad
        a_new = a - lr * v
        a_new = np.maximum(a_new, 1e-8)
        a_new /= a_new.sum()
        J_new = compute_J(a_new)

        if J_new < current_J - 1e-15:
            a = a_new
            current_J = J_new
            patience = 0
            tau = min(500.0, tau * 1.02)
            lr = 2.0 / (n * tau)
            momentum = min(0.95, momentum * 1.001)
        else:
            patience += 1
            lr *= 0.7
            v *= 0.5
            if patience > max_patience:
                break
            if lr < 1e-18:
                break
        if it % 50 == 0:
            tracker.update(a)

    tracker.update(a)
    return a


def subgradient_averaging(a, deadline, tracker, max_iters=10000):
    """Subgradient descent with Polyak step and iterate averaging."""
    a = np.maximum(a.astype(np.float64), 1e-10)
    a /= a.sum()
    n = len(a)
    current_J = compute_J(a)
    tracker.update(a)

    # Polyak lower bound: sum of all (a*a)_k = (sum a)^2 = 1, so max >= 1/(2n-1)
    f_lower = 1.0 / (2 * n - 1)
    f_star_est = f_lower
    a_avg = a.copy()
    a_best = a.copy()
    J_best = current_J

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        b = fast_conv(a)
        M = float(b.max())
        # Active set
        eps = 1e-8 * max(M, 1e-12)
        active = np.where(b >= M - eps)[0]

        # Compute subgradient
        grad = np.zeros(n)
        for k in active:
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            for j in range(js, je + 1):
                idx = k - j
                if 0 <= idx < n:
                    grad[j] += a[idx]
        grad /= len(active)

        # Add negative gradient of -2*M*sum(a) term (for the normalized objective)
        # dJ/da_j (with sum=1) involves: 2n*(grad_j - 2*M*a_j) / (sum_a)^2
        # Direction to decrease J: grad - 2*M*a (simplified)
        direction = grad - 2.0 * M * a

        # Polyak step: lr = (f(a) - f*) / ||grad||^2
        f_current = M  # since sum(a)=1, f = max(a*a) = M
        gap = f_current - f_star_est
        gnorm2 = np.dot(direction, direction)
        if gnorm2 < 1e-20:
            break
        lr = gap / gnorm2
        lr = min(lr, 1.0 / np.sqrt(it + 1))  # cap for stability

        a_new = a - lr * direction
        # Project onto {x >= 0, sum(x) = 1}
        a_new = np.maximum(a_new, 0.0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            a_new = np.ones(n) / n
        else:
            a_new /= s_new

        a_avg = a_avg * (it / (it + 1)) + a_new * (1.0 / (it + 1))

        J_new = compute_J(a_new)
        if J_new < J_best - 1e-12:
            a_best = a_new.copy()
            J_best = J_new
            f_star_est = f_new_val = J_new / (2 * n)  # max(a*a) for sum=1
            tracker.update(a_new)
        a = a_new
        current_J = J_new

        # Update lower bound estimate occasionally
        if it % 100 == 0:
            J_avg = compute_J(a_avg)
            if J_avg < J_best:
                a_best = a_avg.copy()
                J_best = J_avg
                tracker.update(a_avg)

    # Compare a_best with a_avg
    J_avg_final = compute_J(a_avg)
    if J_avg_final < J_best:
        a_best = a_avg
        J_best = J_avg_final
    tracker.update(a_best)
    return a_best


def mirror_descent(a, deadline, tracker, max_iters=5000):
    """Multiplicative weight (mirror) descent for the nonnegativity constraint."""
    a = np.maximum(a.astype(np.float64), 1e-8)
    a /= a.sum()
    n = len(a)
    current_J = compute_J(a)
    tracker.update(a)

    eta = 5.0 / n
    patience = 0
    max_patience = 300

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        b = fast_conv(a)
        M = float(b.max())
        eps = 1e-8 * max(M, 1e-12)
        active = np.where(b >= M - eps)[0]

        grad = np.zeros(n)
        for k in active:
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            for j in range(js, je + 1):
                idx = k - j
                if 0 <= idx < n:
                    grad[j] += a[idx]
        grad /= len(active)

        # Mirror descent update: a_j *= exp(-eta * grad_j)
        update = np.exp(-eta * grad / max(grad.max(), 1e-15))
        a_new = a * update
        a_new = np.maximum(a_new, 1e-10)
        a_new /= a_new.sum()

        J_new = compute_J(a_new)
        if J_new < current_J - 1e-14:
            a = a_new
            current_J = J_new
            patience = 0
            eta = min(eta * 1.01, 50.0 / n)
        else:
            patience += 1
            eta *= 0.85
            if patience > max_patience:
                break

        if it % 100 == 0:
            tracker.update(a)

    tracker.update(a)
    return a


def flat_spectrum_init(n, rng):
    """Generate sequence with approximately flat power spectrum (nonnegative)."""
    half = n // 2 + 1
    angles = rng.uniform(0, 2 * np.pi, half)
    A = np.zeros(n, dtype=complex)
    A[:half] = np.exp(1j * angles[:half])
    for k in range(1, n - n // 2):
        A[k] = np.conj(A[n - k])
    a = np.fft.ifft(A).real
    # Make nonnegative by taking absolute value or rectifying
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-10:
        return np.ones(n) / n
    return a / s


def diverse_init(n, rng, parent=None):
    """Generate diverse initial sequences."""
    inits = []
    t = np.linspace(0, 1, n, endpoint=False)

    if parent is not None:
        p = np.asarray(parent, dtype=np.float64)
        if p.size != n:
            t_old = np.linspace(0, 1, p.size)
            p = np.interp(t, t_old, p)
        p = np.maximum(p, 0.0)
        ps = p.sum()
        if ps > 0:
            inits.append(p / ps)
            inits.append(p[::-1] / ps)  # reversed

    inits.append(np.ones(n) / n)

    # Cosine-modulated
    for freq, amp in [(1, 0.3), (2, 0.25), (3, 0.2), (5, 0.15), (7, 0.1)]:
        sig = 1.0 + amp * np.cos(2 * np.pi * freq * t + rng.uniform(0, 6.28))
        inits.append(np.maximum(sig, 0.01) / np.maximum(sig, 0.01).sum())

    # Multi-frequency
    sig = 1.0
    for i, (f, a) in enumerate([(1, 0.2), (2.3, 0.15), (4.1, 0.1), (7.7, 0.08)]):
        sig += a * np.cos(2 * np.pi * f * t + rng.uniform(0, 6.28))
    inits.append(np.maximum(sig, 0.01) / np.maximum(sig, 0.01).sum())

    # Flat spectrum
    inits.append(flat_spectrum_init(n, rng))
    inits.append(flat_spectrum_init(n, rng))

    # Gaussian bumps
    for n_bumps in [2, 3, 5]:
        centers = rng.uniform(0.1, 0.9, n_bumps)
        widths = rng.uniform(0.05, 0.2, n_bumps)
        sig = np.ones(n) * 0.1
        for c, w in zip(centers, widths):
            sig += np.exp(-0.5 * ((t - c) / w) ** 2)
        inits.append(sig / sig.sum())

    # Power law
    alpha = rng.uniform(-0.5, 1.5)
    sig = (t + 0.05) ** alpha
    inits.append(sig / sig.sum())

    # Random (smoothed)
    raw = rng.random(n)
    kernel = np.ones(max(3, n // 50)) / max(3, n // 50)
    sig = np.convolve(raw, kernel, mode='same')
    inits.append(np.maximum(sig, 0.01) / np.maximum(sig, 0.01).sum())

    # Interval-based (block structure)
    n_blocks = rng.integers(3, 10)
    edges = np.sort(rng.uniform(0, 1, n_blocks - 1))
    edges = np.insert(np.insert(edges, 0, 0), -1, 1)
    sig = np.zeros(n)
    for i in range(n_blocks):
        mask = (t >= edges[i]) & (t < edges[i + 1])
        sig[mask] = rng.uniform(0.3, 1.0)
    inits.append(np.maximum(sig, 0.01) / np.maximum(sig, 0.01).sum())

    return inits


class ScoreTracker:
    def __init__(self):
        self.best = None
        self.best_score = float('inf')
        self.t_start = time.monotonic()

    def update(self, a):
        try:
            a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
            if a.size == 0 or a.sum() < 0.01:
                return
            score = float(evaluate_sequence(a.tolist()))
            if np.isfinite(score) and score < self.best_score:
                self.best = a.copy()
                self.best_score = score
                elapsed = time.monotonic() - self.t_start
                print(f"  [NEW BEST] J={score:.8f} n={a.size} t={elapsed:.1f}s", flush=True)
        except Exception:
            pass

    def report(self, label=""):
        if self.best is not None:
            print(f"  {label} Best: J={self.best_score:.8f} n={self.best.size}", flush=True)


def coordinate_descent(a, deadline, tracker, max_steps=5000):
    """Coordinate descent with random coordinate selection."""
    a = np.maximum(a.astype(np.float64), 1e-10)
    a /= a.sum()
    n = len(a)
    current_J = compute_J(a)
    tracker.update(a)
    rng = np.random.default_rng(123)

    scales = [0.3, 0.5, 0.7, 1.3, 1.5, 2.0, 0.1, 3.0]

    steps_since_improve = 0
    max_no_improve = 500

    for step in range(max_steps):
        if time.monotonic() > deadline:
            break
        if steps_since_improve > max_no_improve:
            break

        j = rng.integers(0, n)
        improved = False
        # Try multiple scales
        for scale in scales:
            a_trial = a.copy()
            a_trial[j] *= scale
            a_trial = np.maximum(a_trial, 1e-10)
            ss = a_trial.sum()
            if ss < 1e-10:
                continue
            a_trial /= ss
            J_trial = compute_J(a_trial)
            if J_trial < current_J - 1e-13:
                a = a_trial
                current_J = J_trial
                improved = True
                break

        if improved:
            steps_since_improve = 0
            if step % 100 == 0:
                tracker.update(a)
        else:
            steps_since_improve += 1

    tracker.update(a)
    return a


def simulated_annealing(a, deadline, tracker, n_restarts=30):
    """SA with LP re-optimization at each perturbation."""
    rng = np.random.default_rng(42)
    n = a.size
    a = a.copy().astype(np.float64)
    current_J = compute_J(a)
    tracker.update(a)

    for r in range(n_restarts):
        if time.monotonic() > deadline:
            break
        temp = 0.1 * (1.0 - r / n_restarts) + 0.01
        a_pert = a + rng.normal(0, temp, n)
        a_pert = np.maximum(a_pert, 1e-8)
        a_pert /= a_pert.sum()

        # Short LP refinement
        remaining = deadline - time.monotonic()
        if remaining < 10:
            break
        iters = max(5, min(20, int(remaining / 10)))
        a_refined = run_lp_optimize(a_pert, time.monotonic() + min(30, remaining * 0.3), tracker,
                                     max_iters=iters, tighten_start=0.01, tighten_end=0.05)
        J_refined = compute_J(a_refined)

        delta = J_refined - current_J
        accept = delta < 0 or rng.random() < np.exp(-abs(delta) / max(0.001, 0.05 / (r + 1)))
        if accept:
            a = a_refined
            current_J = J_refined

    tracker.update(a)
    return a


def propose_candidate(seed=42, budget_s=1500, **kwargs):
    """Main search function: multi-phase optimization to minimize J(a)."""
    rng = np.random.default_rng(seed)
    t_start = time.monotonic()
    deadline = t_start + float(budget_s) - 15.0  # safety margin
    tracker = ScoreTracker()

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        if p.ndim == 1 and p.size > 0 and np.all(np.isfinite(p)):
            parent = np.maximum(p, 0.0)
            tracker.update(parent)
            print(f"Loaded parent: n={parent.size}, J={tracker.best_score:.8f}", flush=True)
    except (NameError, AttributeError):
        pass

    # === Phase 1: Multi-length LP exploration (first ~400s) ===
    phase1_end = t_start + 400.0
    print(f"\n{'='*60}", flush=True)
    print(f"PHASE 1: Multi-length LP exploration (until t={phase1_end - t_start:.0f}s)", flush=True)
    print(f"{'='*60}", flush=True)

    lengths = [256, 320, 384, 448, 512, 640, 768]
    # Filter to reasonable lengths based on time
    time_per_length = max(30.0, (phase1_end - time.monotonic()) / len(lengths))
    best_overall = None
    best_overall_J = float('inf')
    best_length = None

    for n in lengths:
        t_phase_start = time.monotonic()
        t_phase_end = min(phase1_end, t_phase_start + time_per_length)
        if time.monotonic() >= t_phase_end - 10:
            continue

        print(f"\n  --- n={n} (budget {time_per_length:.0f}s) ---", flush=True)
        inits = diverse_init(n, rng, parent)
        n_inits = len(inits)
        init_budget = max(5.0, (t_phase_end - time.monotonic()) / n_inits)
        best_for_n = None
        best_J_for_n = float('inf')

        for idx, a0 in enumerate(inits):
            if time.monotonic() > t_phase_end - 5:
                break
            t_init_end = min(t_phase_end - 5, time.monotonic() + init_budget)

            # LP phase
            n_lp = max(8, int(init_budget / 3.0))
            a = run_lp_optimize(a0, t_init_end, tracker, max_iters=n_lp,
                               tighten_start=0.01, tighten_end=0.10)
            J = compute_J(a)
            if J < best_J_for_n:
                best_J_for_n = J
                best_for_n = a.copy()

        if best_for_n is not None:
            tracker.update(best_for_n)
            if best_J_for_n < best_overall_J:
                best_overall_J = best_J_for_n
                best_overall = best_for_n.copy()
                best_length = n
            print(f"    n={n} best J={best_J_for_n:.8f} (T={time.monotonic()-t_phase_start:.1f}s)", flush=True)

    if best_overall is None and parent is not None:
        best_overall = parent.copy()
        best_overall_J = tracker.best_score
        best_length = parent.size
    elif best_overall is None:
        best_overall = np.ones(256)
        best_overall_J = compute_J(best_overall)
        best_length = 256
        tracker.update(best_overall)

    tracker.report("Phase 1")
    print(f"  Best so far: J={tracker.best_score:.8f} n={tracker.best.size}", flush=True)

    # === Phase 2: Focused optimization on best length (~400s) ===
    phase2_end = min(deadline, t_start + 800.0)
    print(f"\n{'='*60}", flush=True)
    print(f"PHASE 2: Focused optimization n={best_length} (until t={phase2_end - t_start:.0f}s)", flush=True)
    print(f"{'='*60}", flush=True)

    a = best_overall.copy().astype(np.float64)
    n = a.size

    # 2a: Aggressive LP with progressive tightening
    tighten_steps = [0.005, 0.01, 0.02, 0.03, 0.05, 0.07, 0.10, 0.12, 0.15]
    remaining = phase2_end - time.monotonic()
    per_tighten = remaining / len(tighten_steps)
    for tight in tighten_steps:
        if time.monotonic() > phase2_end:
            break
        t_sub_end = min(phase2_end, time.monotonic() + per_tighten)
        a = run_lp_optimize(a, t_sub_end, tracker, max_iters=50,
                           tighten_start=tight * 0.5, tighten_end=tight * 1.5)

    # 2b: LSE gradient descent
    if time.monotonic() < phase2_end - 30:
        t_lse_end = min(phase2_end, time.monotonic() + max(30, (phase2_end - time.monotonic()) * 0.25))
        a = lse_gradient_descent(a, t_lse_end, tracker, max_iters=10000)

    # 2c: Subgradient with averaging
    if time.monotonic() < phase2_end - 20:
        t_sg_end = min(phase2_end, time.monotonic() + max(20, (phase2_end - time.monotonic()) * 0.25))
        a_sg = subgradient_averaging(a, t_sg_end, tracker, max_iters=5000)
        J_sg = compute_J(a_sg)
        J_cur = compute_J(a)
        if J_sg < J_cur:
            a = a_sg

    # 2d: Mirror descent
    if time.monotonic() < phase2_end - 15:
        t_md_end = min(phase2_end, time.monotonic() + max(15, (phase2_end - time.monotonic()) * 0.2))
        a_md = mirror_descent(a, t_md_end, tracker, max_iters=5000)
        J_md = compute_J(a_md)
        J_cur = compute_J(a)
        if J_md < J_cur:
            a = a_md

    tracker.update(a)
    a = tracker.best.copy() if tracker.best is not None else a
    tracker.report("Phase 2")

    # === Phase 3: Simulated Annealing escape (~300s) ===
    phase3_end = min(deadline, t_start + 1100.0)
    if time.monotonic() < phase3_end:
        print(f"\n{'='*60}", flush=True)
        print(f"PHASE 3: SA escape (until t={phase3_end - t_start:.0f}s)", flush=True)
        print(f"{'='*60}", flush=True)
        a = simulated_annealing(a, phase3_end, tracker, n_restarts=25)
        # Polish after SA
        if time.monotonic() < phase3_end - 10:
            a = run_lp_optimize(a, phase3_end, tracker, max_iters=30,
                               tighten_start=0.01, tighten_end=0.05)
            if time.monotonic() < phase3_end - 5:
                a = lse_gradient_descent(a, phase3_end, tracker, max_iters=2000)
        tracker.report("Phase 3")
        a = tracker.best.copy() if tracker.best is not None else a

    # === Phase 4: Try a second best length if time remains (~150s) ===
    phase4_end = min(deadline, t_start + 1250.0)
    if time.monotonic() < phase4_end and best_overall is not None:
        # Try another length that might be better
        alt_lengths = [l for l in [384, 512, 640, 896] if l != best_length]
        if alt_lengths:
            alt_n = alt_lengths[0]
            t_alt_end = min(phase4_end, time.monotonic() + 80)
            if t_alt_end > time.monotonic() + 20:
                print(f"\n  Trying alt length n={alt_n}...", flush=True)
                parent_resampled = None
                if parent is not None:
                    t_old = np.linspace(0, 1, parent.size)
                    parent_resampled = np.interp(np.linspace(0, 1, alt_n), t_old, np.maximum(parent, 0))
                inits = diverse_init(alt_n, rng, parent_resampled)
                for a0 in inits[:4]:
                    if time.monotonic() > t_alt_end - 5:
                        break
                    a_alt = run_lp_optimize(a0, t_alt_end, tracker, max_iters=20,
                                           tighten_start=0.02, tighten_end=0.10)
                    J_alt = compute_J(a_alt)
                    if J_alt < tracker.best_score:
                        tracker.update(a_alt)
                        a = a_alt.copy()
                tracker.report("Phase 4 (alt length)")

    # === Phase 5: Final polishing (last ~130s) ===
    if tracker.best is not None and time.monotonic() < deadline:
        print(f"\n{'='*60}", flush=True)
        print(f"PHASE 5: Final polishing (until deadline)", flush=True)
        print(f"{'='*60}", flush=True)
        a = tracker.best.copy().astype(np.float64)

        # Fine LP
        t_lp_end = min(deadline, time.monotonic() + 40)
        a = run_lp_optimize(a, t_lp_end, tracker, max_iters=40,
                           tighten_start=0.003, tighten_end=0.02)

        # Fine LSE
        if time.monotonic() < deadline - 5:
            t_lse_end = min(deadline - 2, time.monotonic() + max(10, (deadline - time.monotonic()) * 0.4))
            a = lse_gradient_descent(a, t_lse_end, tracker, max_iters=5000)

        # Coordinate descent
        if time.monotonic() < deadline - 3:
            t_cd_end = min(deadline - 1, time.monotonic() + max(5, (deadline - time.monotonic()) * 0.4))
            a = coordinate_descent(a, t_cd_end, tracker, max_steps=3000)

        # Final LP + LSE
        if time.monotonic() < deadline - 2:
            a = run_lp_optimize(a, deadline - 1, tracker, max_iters=10,
                               tighten_start=0.002, tighten_end=0.01)
            if time.monotonic() < deadline - 0.5:
                a = lse_gradient_descent(a, deadline - 0.1, tracker, max_iters=500)

    tracker.report("Final")

    # Fallback
    if tracker.best is None:
        tracker.best = np.ones(256)
        tracker.best_score = compute_J(tracker.best)
        tracker.update(tracker.best)

    elapsed = time.monotonic() - t_start
    print(f"\n{'='*60}", flush=True)
    print(f"FINAL: J={tracker.best_score:.8f} n={tracker.best.size} time={elapsed:.1f}s", flush=True)
    print(f"{'='*60}", flush=True)

    return [float(x) for x in tracker.best]
