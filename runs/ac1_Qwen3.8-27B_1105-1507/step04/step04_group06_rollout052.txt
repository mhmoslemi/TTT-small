"""
Minimizes J(a) = 2*n*max(convolve(a,a))/(sum(a))² using:
1. FISTA (accelerated projected gradient) with log-sum-exp smoothing
2. LP mirror descent with reduced active-set constraints
3. Alternating FISTA+LP for refined basin finding

5. Symmetrization and active-set exploitation
6. Adaptive beta annealing and backtracking line search

import numpy as np
from scipy.optimize import linprog, minimize
from scipy.sparse import lil_matrix, csr_matrix
import time


def fast_conv(a):
    """Full self-convolution via FFT, O(n log n)."""
    n = len(a)
    s = 1
    while s < 2 * n - 1:
        s <<= 1
    fa = np.fft.rfft(a, s)
    return np.fft.irfft(fa * fa, s)[:2 * n - 1]


def compute_J(a):
    """Compute J(a) = 2*n*max(conv(a,a))/(sum(a))^2."""
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))


def smooth_max_and_grad(a, beta):
    """
    Compute log-sum-exp smooth approximation of max(conv(a,a)) and gradient.
    Returns (smooth_value, gradient).
    Gradient is with respect to a (unconstrained).
    """
    n = len(a)
    b = fast_conv(a)
    b_max = float(b.max())
    shifted = beta * (b - b_max)
    shifted = np.clip(shifted, -500.0, 500.0)
    exp_shifted = np.exp(shifted)
    Z = float(exp_shifted.sum())
    if Z < 1e-300:
        p = np.zeros(2 * n - 1)
        p[np.argmax(b)] = 1.0
    else:
        p = exp_shifted / Z
    S_val = (b_max + np.log(max(Z, 1e-300))) / beta

    # Gradient: grad[i] = 2 * sum_{m=0}^{n-1} p[m+i] * a[m]
    # = 2 * (a_rev * p)[i + n - 1]
    a_rev = a[::-1]
    s = 1
    while s < len(a_rev) + len(p) - 1:
        s <<= 1
    A_r = np.fft.rfft(a_rev, s)
    P = np.fft.rfft(p, s)
    conv_full = np.fft.irfft(A_r * P, s)
    grad = 2.0 * conv_full[n - 1:2 * n - 1]
    return S_val, grad


def proj_simplex(v):
    """Euclidean projection onto the simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - 1))[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def proj_simplex_sum(v, target_sum):
    """Projection onto {x >= 0, sum(x) = target_sum}."""
    return target_sum * proj_simplex(v / target_sum)


def fista_smooth(a0, beta, max_iters, deadline, tracker=None, momentum_switch=200):
    """
    FISTA with projected gradient on simplex {a >= 0, sum(a) = 1}.
    Minimizes smooth_max(conv(a,a)) with log-sum-exp at temperature beta.
    """
    n = len(a0)
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-10:
        a = np.ones(n) / n
    else:
        a = a / s
    x = a.copy()
    z = a.copy()
    t = 1.0
    best_J = compute_J(a)
    best_a = a.copy()
    
    if tracker is not None:
        tracker.update(a)
    
    # Backtracking step size initialization
    step = 1.0 / (4.0 * beta * n)
    
    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        
        f_z, grad_z = smooth_max_and_grad(z, beta)
        
        # FISTA momentum (switch to plain GD after momentum_switch)
        if it < momentum_switch:
            t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t * t))
            y = z + ((t - 1.0) / t_new) * (z - x)
            y = np.maximum(y, 0.0)
            ys = y.sum()
            if ys < 1e-10:
                y = z.copy()
            else:
                y = y / ys
        else:
            y = z.copy()
            t_new = 1.0
        
        # Gradient step with backtracking
        f_y, _ = smooth_max_and_grad(y, beta)
        for _bt in range(20):
            x_new = y - step * grad_z
            x_new = proj_simplex(x_new)
            f_new, _ = smooth_max_and_grad(x_new, beta)
            if f_new <= f_y + 0.5 * step * step * (grad_z * grad_z).sum():
                break
            step *= 0.5
            x_new = y - step * grad_z
            x_new = proj_simplex(x_new)
            f_new, _ = smooth_max_and_grad(x_new, beta)
        else:
            x_new = z.copy()
            f_new = f_z
        
        # Update
        x = z.copy()
        z = x_new
        t = t_new
        
        # Track best
        J_cur = f_new * 2 * n  # scale back to J (since sum=1)
        # Actually compute exact J for tracking
        J_exact = compute_J(z)
        if J_exact < best_J:
            best_J = J_exact
            best_a = z.copy()
            if tracker is not None:
                tracker.update(z)
    
    return best_a, best_J


def fista_adaptive(a0, deadline, tracker=None, betas=None, iters_per_beta=None):
    """FISTA with adaptive beta annealing."""
    if betas is None:
        betas = [1.0, 3.0, 10.0, 30.0, 100.0, 300.0]
    if iters_per_beta is None:
        iters_per_beta = [300, 300, 300, 250, 200, 150]
    
    n = len(a0)
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-10:
        a = np.ones(n) / n
    else:
        a = a / s
    
    best_J = compute_J(a)
    best_a = a.copy()
    
    for i, (beta, n_iters) in enumerate(zip(betas, iters_per_beta)):
        if time.monotonic() > deadline:
            break
        remaining = deadline - time.monotonic()
        if remaining < 5:
            break
        a, J_cur = fista_smooth(a, beta, n_iters, deadline, tracker=tracker,
                                 momentum_switch=max(100, n_iters // 3))
        if J_cur < best_J:
            best_J = J_cur
            best_a = a.copy()
        # Symmetrize
        a_sym = 0.5 * (a + a[::-1])
        a_sym = a_sym / a_sym.sum()
        J_sym = compute_J(a_sym)
        if J_sym < best_J:
            best_J = J_sym
            best_a = a_sym.copy()
            a = a_sym.copy()
    
    return best_a, best_J


def build_A_sparse(a, active_idx=None):
    """Build sparse constraint matrix for LP. Use only active_idx constraints if given."""
    n = len(a)
    if active_idx is None:
        active_idx = np.arange(2 * n - 1)
    
    rows = []
    cols = []
    vals = []
    for row_i, k in enumerate(active_idx):
        jl = max(0, k - n + 1)
        jh = min(n - 1, k)
        for j in range(jl, jh + 1):
            rows.append(row_i)
            cols.append(j)
            vals.append(a[j])
    
    if len(rows) == 0:
        return None
    A = csr_matrix((vals, (np.array(rows, dtype=np.int32), np.array(cols, dtype=np.int32))),
                   shape=(len(active_idx), n))
    return A


def lp_best_response(a, M, top_k=None):
    """
    LP: max sum(g) s.t. (a*g)_k <= M for all k (or top-k), g >= 0.
    Returns normalized g or None.
    """
    n = len(a)
    b = fast_conv(a)
    
    if top_k is not None and top_k < 2 * n - 1:
        idx = np.argsort(b)[::-1][:top_k]
    else:
        idx = np.arange(2 * n - 1)
    
    A = build_A_sparse(a, idx)
    if A is None:
        return None
    
    c = -np.ones(n)
    b_ub = np.full(len(idx), M)
    bounds = [(0.0, None)] * n
    
    try:
        r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                    options={'presolve': True, 'time_limit': 30})
    except Exception:
        return None
    
    if r.success and r.x is not None:
        g = np.maximum(r.x, 0.0)
        g_sum = g.sum()
        if g_sum > 1e-10:
            return g / g_sum
    return None


def line_search(a, g, n_evals=15):
    """
    Golden section + parabolic line search for t in [0,1].
    Minimizes J((1-t)*a + t*g).
    Returns (best_a, best_J).
    """
    n = len(a)
    best_t = 0.0
    best_J = compute_J(a)
    
    # Evaluate endpoints
    J1 = compute_J(g)
    if J1 < best_J:
        best_J = J1
        best_t = 1.0
    
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    lo = 0.0
    hi = 1.0
    f1 = best_J if best_t < 0.5 else J1
    f2 = J1 if best_t < 0.5 else best_J
    
    # If best is at endpoint, narrow search
    if best_t > 0.5:
        lo = 0.0
        hi = 0.5
    else:
        lo = 0.5
        hi = 1.0
    
    # Golden section
    for _ in range(n_evals):
        t1 = hi - gr * (hi - lo)
        t2 = lo + gr * (hi - lo)
        a1 = (1.0 - t1) * a + t1 * g
        a2 = (1.0 - t2) * a + t2 * g
        a1 = np.maximum(a1, 0.0)
        a2 = np.maximum(a2, 0.0)
        J_a1 = compute_J(a1)
        J_a2 = compute_J(a2)
        
        if J_a1 < best_J:
            best_J = J_a1
            best_t = t1
        if J_a2 < best_J:
            best_J = J_a2
            best_t = t2
        
        if J_a1 < J_a2:
            hi = t2
        else:
            lo = t1
    
    a_best = (1.0 - best_t) * a + best_t * g
    a_best = np.maximum(a_best, 0.0)
    s = a_best.sum()
    if s > 1e-10:
        a_best = a_best / s
    return a_best, best_J


def lp_mirror_descent(a, max_iters, deadline, tracker=None, n_line_evals=15,
                      stall_patience=40, top_k=200):
    """
    LP-based mirror descent with line search and reduced constraints.
    """
    a = np.maximum(a.astype(np.float64), 1e-6)
    a = a / a.sum()
    current_J = compute_J(a)
    best_J = current_J
    best_a = a.copy()
    
    if tracker is not None:
        tracker.update(a)
    
    stall_count = 0
    
    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        
        b = fast_conv(a)
        M = float(b.max())
        
        g = lp_best_response(a, M, top_k=top_k)
        if g is None:
            break
        
        a_new, J_new = line_search(a, g, n_evals=n_line_evals)
        
        if J_new < current_J - 1e-12:
            a = a_new
            current_J = J_new
            stall_count = 0
            if J_new < best_J:
                best_J = J_new
                best_a = a.copy()
                if tracker is not None:
                    tracker.update(a)
        else:
            # Try smaller steps
            found = False
            for t_try in [0.01, 0.005, 0.02, 0.03]:
                a_try = np.maximum((1.0 - t_try) * a + t_try * g, 0.0)
                a_try = a_try / a_try.sum()
                J_try = compute_J(a_try)
                if J_try < current_J - 1e-13:
                    a = a_try
                    current_J = J_try
                    stall_count = 0
                    if J_try < best_J:
                        best_J = J_try
                        best_a = a.copy()
                        if tracker is not None:
                            tracker.update(a)
                    found = True
                    break
            if not found:
                stall_count += 1
                if stall_count >= stall_patience:
                    break
    
    return best_a, best_J


def alternating_fista_lp(a, deadline, tracker=None, 
                          fista_iters=50, lp_iters=5,
                          fista_betas=[10.0, 50.0, 200.0],
                          top_lp_k=150):
    """Alternate between FISTA and LP mirror descent."""
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0.0)
    a = a / a.sum()
    best_J = compute_J(a)
    best_a = a.copy()
    
    if tracker is not None:
        tracker.update(a)
    
    beta_idx = 0
    
    while time.monotonic() < deadline:
        # FISTA phase
        if time.monotonic() > deadline:
            break
        beta = fista_betas[min(beta_idx, len(fista_betas) - 1)]
        fista_deadline = min(deadline, time.monotonic() + 60.0)
        a, J_f = fista_smooth(a, beta, fista_iters, fista_deadline, tracker=tracker,
                               momentum_switch=fista_iters // 2)
        if J_f < best_J:
            best_J = J_f
            best_a = a.copy()
        
        # LP phase
        if time.monotonic() > deadline:
            break
        lp_deadline = min(deadline, time.monotonic() + 40.0)
        a, J_l = lp_mirror_descent(a, lp_iters, lp_deadline, tracker=tracker,
                                    n_line_evals=12, stall_patience=10,
                                    top_k=top_lp_k)
        if J_l < best_J:
            best_J = J_l
            best_a = a.copy()
        
        # Increase beta for next FISTA phase
        if beta_idx < len(fista_betas) - 1:
            beta_idx += 1
        top_lp_k = min(top_lp_k + 20, 400)
    
    return best_a, best_J


def resample_seq(src, dest_n):
    """Resample 1D sequence to new length via linear interpolation."""
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    result = np.interp(t_new, t_old, src)
    return np.maximum(result, 0.0)


def generate_initializations(n, rng, parent=None):
    """Generate diverse initial sequences of length n."""
    inits = []
    t_arr = np.linspace(0, 1, n, endpoint=False)
    
    if parent is not None and parent.size > 0:
        p = resample_seq(parent, n)
        if p.sum() > 1e-10:
            p = p / p.sum()
        inits.append(p)
        # Slightly perturbed parent
        p2 = p + rng.normal(0, 0.03, n) * p
        p2 = np.maximum(p2, 0.001)
        p2 = p2 / p2.sum()
        inits.append(p2)
    
    # Uniform
    inits.append(np.ones(n) / n)
    
    # Cosine modulations
    for freq in [1, 2, 3, 4]:
        for amp in [0.2, 0.4, 0.6]:
            c = 1.0 + amp * np.cos(2.0 * np.pi * freq * t_arr)
            c = np.maximum(c, 0.01)
            inits.append(c / c.sum())
    
    # Raised cosine window
    rc = 0.5 * (1 + np.cos(2 * np.pi * t_arr))
    inits.append(rc / rc.sum())
    
    # Gaussian bump
    gb = np.exp(-6.0 * (t_arr - 0.5) ** 2) + 0.1
    inits.append(gb / gb.sum())
    
    # Inverse Gaussian (U-shape)
    ig = np.exp(3.0 * (t_arr - 0.5) ** 2) + 0.3
    inits.append(ig / ig.sum())
    
    # Power law
    pl = (t_arr + 0.05) ** 0.5
    inits.append(pl / pl.sum())
    
    # Sine absolute
    sa = np.abs(np.sin(np.pi * t_arr)) + 0.2
    inits.append(sa / sa.sum())
    
    # Bessel-like
    bl = np.exp(-2.0 * t_arr) * (1.0 + 0.5 * np.cos(4.0 * np.pi * t_arr))
    bl = np.maximum(bl, 0.01)
    inits.append(bl / bl.sum())
    
    return inits


class ScoreTracker:
    """Tracks the best sequence found so far."""
    
    def __init__(self):
        self.best = None
        self.best_score = float('inf')
        self.t_start = time.monotonic()
    
    def update(self, a):
        a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
        if a.size == 0 or a.sum() < 0.01:
            return
        score = float(evaluate_sequence(a.tolist()))
        if np.isfinite(score) and score < self.best_score:
            self.best = a.copy()
            self.best_score = score
            elapsed = time.monotonic() - self.t_start
            print(f"  * NEW BEST: J={score:.8f} n={a.size} t={elapsed:.1f}s", flush=True)
    
    def report(self, label=""):
        if self.best is not None:
            print(f"  {label} best: J={self.best_score:.8f} n={self.best.size}", flush=True)


def propose_candidate(seed=42, budget_s=1500, **kwargs):
    """
    Multi-phase optimization for minimizing J(a) = 2n*max(conv(a,a))/(sum(a))^2.
    
    Phase 1: FISTA with smooth objective at multiple lengths (fast basin finding)
    Phase 2: LP mirror descent with reduced constraints (main refinement)
    Phase 3: Alternating FISTA+LP for escape from local minima
    Phase 4: Final LP polish with full constraints
    Phase 5: Length exploration and perturbation restarts
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + float(budget_s) - 20.0
    tracker = ScoreTracker()
    
    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        if p.ndim == 1 and p.size > 0 and np.all(np.isfinite(p)):
            parent = np.maximum(p, 0.0)
            tracker.update(parent)
            print(f"Parent loaded: n={parent.size}, J={tracker.best_score:.8f}", flush=True)
    except (NameError, AttributeError):
        pass
    
    if parent is None:
        parent = np.ones(1024)
        tracker.update(parent)
        print(f"Using default parent: n=1024, J={tracker.best_score:.8f}", flush=True)
    
    # === Phase 1: FISTA with smooth objective (fast basin finding) ===
    t0 = time.monotonic()
    phase1_budget = 250.0
    deadline_p1 = time.monotonic() + phase1_budget
    print(f"\n{'='*50} Phase 1: FISTA (budget {phase1_budget:.0f}s) {'='*50}", flush=True)
    
    lengths_p1 = [512, 1024, 2048]
    
    for n in lengths_p1:
        if time.monotonic() > deadline_p1:
            break
        t_start = time.monotonic()
        print(f"  FISTA n={n}...", flush=True)
        
        inits = generate_initializations(n, rng, parent)
        # Try top 5 initializations
        max_inits = min(5, len(inits))
        
        for idx in range(max_inits):
            if time.monotonic() > deadline_p1:
                break
            a0 = inits[idx].copy()
            a_opt, J_opt = fista_adaptive(
                a0, time.monotonic() + 50.0, tracker=tracker,
                betas=[1.0, 5.0, 20.0, 100.0],
                iters_per_beta=[100, 100, 100, 80]
            )
        
        print(f"    n={n} done in {time.monotonic()-t_start:.1f}s", flush=True)
    
    tracker.report("Phase 1 complete")
    
    # === Phase 2: LP mirror descent (main refinement) ===
    if tracker.best is not None and time.monotonic() < deadline:
        t0 = time.monotonic()
        n_best = tracker.best.size
        phase2_budget = 550.0
        deadline_p2 = min(deadline, time.monotonic() + phase2_budget)
        print(f"\n{'='*50} Phase 2: LP MD n={n_best} (budget {phase2_budget:.0f}s) {'='*50}", flush=True)
        
        remaining = deadline_p2 - time.monotonic()
        max_iters = int(remaining / 0.15)  # Estimate ~150ms per LP iteration
        max_iters = min(max_iters, 4000)
        
        a_opt, J_opt = lp_mirror_descent(tracker.best, max_iters, deadline_p2,
                                          tracker=tracker, n_line_evals=16,
                                          stall_patience=50, top_k=250)
        
        # Try at other lengths
        for n_try in [512, 1024, 2048]:
            if time.monotonic() > deadline_p2:
                break
            if n_try == n_best:
                continue
            a_res = resample_seq(tracker.best, n_try)
            if a_res.sum() < 1e-10:
                continue
            a_res = a_res / a_res.sum()
            budget_here = min(150.0, deadline_p2 - time.monotonic())
            if budget_here < 10:
                continue
            max_iters_here = int(budget_here / 0.2)
            max_iters_here = min(max_iters_here, 2000)
            a_res, _ = lp_mirror_descent(a_res, max_iters_here, deadline_p2,
                                          tracker=tracker, n_line_evals=14,
                                          stall_patience=30, top_k=200)
        
        tracker.report(f"Phase 2: {time.monotonic()-t0:.1f}s")
    
    # === Phase 3: Alternating FISTA + LP ===
    if tracker.best is not None and time.monotonic() < deadline:
        t0 = time.monotonic()
        n_best = tracker.best.size
        phase3_budget = 350.0
        deadline_p3 = min(deadline, time.monotonic() + phase3_budget)
        print(f"\n{'='*50} Phase 3: Alternating FISTA+LP (budget {phase3_budget:.0f}s) {'='*50}", flush=True)

        # Restart with perturbation
        if time.monotonic() < deadline_p3 - 30:
            a_pert = tracker.best + rng.normal(0, 0.05, n_best) * tracker.best
            a_pert = np.maximum(a_pert, 0.001)
            a_pert = a_pert / a_pert.sum()
            a_opt2, J_opt2 = alternating_fista_lp(
                a_pert, deadline_p3, tracker=tracker,

                top_lp_k=150
            )
        
        # Another restart with different perturbation
        if time.monotonic() < deadline_p3 - 30:
            a_pert2 = tracker.best.copy()
            # Spike perturbation
            k_spikes = max(8, n_best // 64)
            idx_spikes = rng.choice(n_best, size=k_spikes, replace=False)
            a_pert2[idx_spikes] *= rng.uniform(0.2, 3.0, k_spikes)
            a_pert2 = np.maximum(a_pert2, 0.001)
            a_pert2 = a_pert2 / a_pert2.sum()
            a_opt3, J_opt3 = alternating_fista_lp(
                a_pert2, deadline_p3, tracker=tracker,

    if tracker.best is not None and time.monotonic() < deadline:
        t0 = time.monotonic()
        n_best = tracker.best.size
        remaining = deadline - time.monotonic()
        phase4_budget = min(200.0, remaining)
        deadline_p4 = time.monotonic() + phase4_budget
        print(f"\n{'='*50} Phase 4: Final Polish n={n_best} (budget {phase4_budget:.0f}s) {'='*50}", flush=True)
        
        # Extended LP with more constraints
        max_iters = int(phase4_budget / 0.12)
        max_iters = min(max_iters, 3000)
        a_opt, J_opt = lp_mirror_descent(tracker.best, max_iters, deadline_p4,
                                          tracker=tracker, n_line_evals=20,
                                          stall_patience=60, top_k=400)

            max_iters2 = min(max_iters2, 2000)
            a_opt2, _ = lp_mirror_descent(tracker.best, max_iters2, deadline_p4,
                                           tracker=tracker, n_line_evals=18,
                                           stall_patience=50, top_k=300)

    # === Phase 5: Length exploration ===
    if tracker.best is not None and time.monotonic() < deadline:
        t0 = time.monotonic()
        n_best = tracker.best.size
        remaining = deadline - time.monotonic()
        if remaining > 40:
            print(f"\n{'='*50} Phase 5: Length Exploration (budget {remaining:.0f}s) {'='*50}", flush=True)
            
            candidates = []
            for delta in [-128, -64, -32, 32, 64, 128, 256]:
                n_try = n_best + delta
                if 256 <= n_try <= 4096:
                    candidates.append(n_try)
            # Also try standard lengths
            for std in [384, 512, 640, 768, 896, 1024, 1280, 1536, 2048]:
                if abs(std - n_best) > 64 and std not in candidates:
                    candidates.append(std)
            
            for n_try in candidates[:8]:
                if time.monotonic() > deadline:
                    break
                a_res = resample_seq(tracker.best, n_try)
                if a_res.sum() < 1e-10:
                    continue
                a_res = a_res / a_res.sum()
                
                # Quick FISTA then LP
                a_res, _ = fista_adaptive(a_res, time.monotonic() + 20.0, tracker=tracker,
                                           betas=[20.0, 100.0],
                                           iters_per_beta=[50, 50])
                
                budget_here = min(40.0, deadline - time.monotonic())
                if budget_here > 5:
                    max_iters = int(budget_here / 0.15)
                    max_iters = min(max_iters, 500)
                    a_res, _ = lp_mirror_descent(a_res, max_iters, deadline,
                                                  tracker=tracker, n_line_evals=12,
                                                  stall_patience=20, top_k=150)

    # Fallback
    if tracker.best is None:
        tracker.best = np.ones(1024) / 1024.0
        tracker.best_score = compute_J(tracker.best)
        print(f"  Using fallback: J={tracker.best_score:.8f}", flush=True)
    
    print(f"\n{'='*50} FINAL RESULT {'='*50}", flush=True)
    print(f"J={tracker.best_score:.8f} n={tracker.best.size} "
          f"time={time.monotonic()-tracker.t_start:.1f}s", flush=True)
    
    # Ensure output is valid
    result = np.clip(tracker.best, 0.0, 1000.0)
    if result.sum() < 0.01:
        result = np.ones(128)
    
    return [float(x) for x in result]
