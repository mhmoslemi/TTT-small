import numpy as np
from scipy.optimize import linprog
import time

def fast_conv(a):
    """Full self-convolution via FFT."""
    n = len(a)
    size = 1
    while size < 2*n-1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa*fa, size)[:2*n-1]

def compute_J(a):
    """Compute J(a) = 2*n*max(conv(a,a))/(sum(a))^2."""
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2*len(a)*b.max()/(s*s))

def smooth_max_grad(a, tau):
    """Gradient of smooth max of conv(a,a)."""
    n = len(a)
    b = fast_conv(a)
    b_max = b.max()
    exp_b = np.exp(tau*(b - b_max))
    p = exp_b / exp_b.sum()
    grad = np.zeros(n)
    for k in range(2*n-1):
        pk = p[k]
        if pk < 1e-14:
            continue
        lo = max(0, k-n+1)
        hi = min(n-1, k)
        grad[lo:hi+1] += 2.0*pk*a[k-hi:k-lo+1][::-1]
    return grad, b

def smooth_gradient_descent(a, max_iters, deadline, tau0=2.0, tau1=60.0, lr0=0.3, lr1=0.01):
    """Smooth gradient descent with adaptive parameters."""
    a = np.maximum(a.astype(np.float64), 1e-8)
    a = a / a.sum()
    n = len(a)
    best_a = a.copy()
    best_M = fast_conv(a).max()
    v = np.zeros(n)
    
    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        frac = it / max(1, max_iters)
        tau = tau0 + (tau1-tau0)*(frac**0.5)
        lr = lr0*(lr1/lr0)**frac
        grad, b = smooth_max_grad(a, tau)
        M = b.max()
        if M < best_M*(1-1e-9):
            best_M = M
            best_a = a.copy()
        beta = 0.9
        v = beta*v + lr*grad
        a_new = np.maximum(a - v, 0.0)
        s = a_new.sum()
        if s < 1e-10:
            a_new = a.copy()
        else:
            a_new = a_new/s
        a = a_new
    return best_a

def build_lp_matrix(n, a):
    """Build sparse matrix for LP constraint (a*g)[k] <= M."""
    rows = []
    cols = []
    vals = []
    for k in range(2*n-1):

        j_hi = min(n-1, k)
        for j in range(j_lo, j_hi+1):
            rows.append(k)
            cols.append(j)
            vals.append(a[k-j])
    from scipy.sparse import csr_matrix
    return csr_matrix((vals, (rows, cols)), shape=(2*n-1, n))

def lp_best_response(a, M):
    """Solve LP: max sum(g) s.t. (a*g)[k] <= M, g >= 0."""
    n = len(a)
    A = build_lp_matrix(n, a)
    c = -np.ones(n)
    b_ub = np.full(2*n-1, M)
    bounds = [(0, None)]*n
    try:

                    options={'time_limit': 30})
        if r.success and r.x is not None:
            g = np.maximum(r.x, 0)
            if g.sum() > 1e-15:
                return g
    except Exception:
        pass
    return None

def lp_mirror_descent(a, max_iters, deadline):
    """LP mirror descent with adaptive M and line search."""
    a = np.maximum(a.astype(np.float64), 1e-8)
    a = a / a.sum()
    best_a = a.copy()
    best_M = fast_conv(a).max()
    no_improve = 0

        b = fast_conv(a)

            g_try = lp_best_response(a, M_try)
            if g_try is not None and (g is None or g_try.sum() > g.sum()):

            if time.monotonic() > deadline:

        if g is None:

        g = g / g.sum()

        base_t = max(0.05, 0.5*(1 - 0.5*it/max(1, max_iters)))

            if s < 1e-10:

            a_trial = a_trial / s
            M_trial = fast_conv(a_trial).max()
            if M_trial < best_M - 1e-14:

def coordinate_descent(a, max_iters, deadline):

    a = a / a.sum()
    n = len(a)

    best_M = fast_conv(a).max()

        # Random order of coordinates
        order = np.random.permutation(n)
        for i in order:

            best_val = a[i]
            best_local_M = best_M  # Use global best_M for comparison

            candidates = [best_val, best_val*0.5, best_val*1.5, 0.0, best_val*0.1, best_val*2.0]
            for c in candidates:

                M_new = fast_conv(a).max()

                    best_M = M_new

"""
Minimizes J(a) = 2*n*max(conv(a,a))/(sum(a))^2 using a multi-phase approach:
1. Multi-length smooth gradient descent for broad exploration
2. Aggressive LP mirror descent with adaptive M for precise local optimization
3. Fast coordinate descent for final polish
4. Structural diversity via multiple initializations and length exploration
Key improvement: significantly more LP iterations with adaptive constraint level,
plus coordinate descent that exploits incremental conv structure.

import numpy as np
from scipy.optimize import linprog
from scipy.sparse import csr_matrix
import time


def fast_conv(a):
    """Full self-convolution via FFT."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]


def compute_J(a):
    """Compute J(a) = 2*n*max(conv(a,a))/(sum(a))^2."""
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))


def smooth_max_grad(a, tau):
    """Gradient of (1/tau)*log(sum exp(tau*conv_k)) w.r.t. a."""
    n = len(a)
    b = fast_conv(a)
    b_max = b.max()
    exp_b = np.exp(tau * (b - b_max))
    p = exp_b / exp_b.sum()
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        pk = p[k]
        if pk < 1e-14:
            continue
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        grad[lo:hi + 1] += 2.0 * pk * a[k - hi:k - lo + 1][::-1]
    return grad, b


def smooth_grad_descent(a, max_iters, deadline, tau0=2.0, tau1=60.0, lr0=0.3, lr1=0.01, momentum=True):
    """Smooth gradient descent with adaptive tau and learning rate."""
    a = np.maximum(a.astype(np.float64), 1e-8)
    a = a / a.sum()
    n = len(a)
    best_a = a.copy()
    best_M = fast_conv(a).max()
    v = np.zeros(n)

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        frac = it / max(1, max_iters)
        tau = tau0 + (tau1 - tau0) * (frac ** 0.5)
        lr = lr0 * (lr1 / lr0) ** frac
        grad, b = smooth_max_grad(a, tau)
        M = b.max()
        if M < best_M * (1 - 1e-9):
            best_M = M
            best_a = a.copy()
        beta = 0.9 if momentum else 0.0
        v = beta * v + lr * grad
        a_new = np.maximum(a - v, 0.0)
        s = a_new.sum()
        if s < 1e-10:
            a_new = a.copy()
        else:
            a_new = a_new / s
        a = a_new
    return best_a


def build_lp_matrix(n, a):
    """Build sparse constraint matrix for LP: A[k,j] = a[k-j] if valid."""
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        j_lo = max(0, k - n + 1)
        j_hi = min(n - 1, k)
        for j in range(j_lo, j_hi + 1):
            rows.append(k)
            cols.append(j)
            vals.append(a[k - j])
    return csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))


def lp_best_response(a, M, time_limit=20):
    """Solve LP: max sum(g) s.t. (a*g)[k] <= M for all k, g >= 0."""
    n = len(a)
    try:
        A = build_lp_matrix(n, a)
        c = -np.ones(n)
        b_ub = np.full(2 * n - 1, M)
        bounds = [(0, None)] * n
        r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                    options={'time_limit': time_limit})
        if r.success and r.x is not None:
            g = np.maximum(r.x, 0)
            if g.sum() > 1e-15:
                return g
    except Exception:
        pass
    return None


def lp_mirror_descent(a, max_iters, deadline, aggressive=True):
    """LP mirror descent with adaptive M levels and multi-step line search."""
    a = np.maximum(a.astype(np.float64), 1e-8)
    a = a / a.sum()
    best_a = a.copy()
    best_M = fast_conv(a).max()
    no_improve = 0
    n = len(a)
    # Estimate time per LP call
    t0 = time.monotonic()
    g_test = lp_best_response(a, best_M, time_limit=5)
    lp_time = time.monotonic() - t0
    if g_test is None:
        return best_a

    m_multipliers = [1.0, 0.9, 0.8, 1.1] if aggressive else [1.0, 0.95]
    num_m = len(m_multipliers)

    for it in range(max_iters):
        elapsed = time.monotonic()
        remaining = deadline - elapsed
        est_time_per_iter = lp_time * num_m + 0.05
        if remaining < est_time_per_iter * 3:
            break

        b = fast_conv(a)
        M = float(b.max())
        if M <= best_M * (1 + 1e-10):
            M = best_M

        # Try multiple M levels, keep the best response
        best_g = None
        best_g_sum = -1
        for mf in m_multipliers:
            if time.monotonic() > deadline - 5:
                break
            g_try = lp_best_response(a, M * mf, time_limit=15)
            if g_try is not None:
                gs = g_try.sum()
                if gs > best_g_sum:
                    best_g = g_try
                    best_g_sum = gs

        if best_g is None:
            break
        g = best_g / best_g.sum()

        # Line search with more candidates
        improved = False
        decay = max(0.03, 0.6 * (1 - 0.7 * it / max(1, max_iters)))
        t_candidates = [decay, decay * 0.7, decay * 0.5, decay * 0.3,
                        decay * 0.15, 0.05, 0.02, 0.01]
        for t in t_candidates:
            if time.monotonic() > deadline - 2:
                break
            a_trial = np.maximum((1 - t) * a + t * g, 0)
            s = a_trial.sum()
            if s < 1e-10:
                continue
            a_trial = a_trial / s
            M_trial = fast_conv(a_trial).max()
            if M_trial < best_M - 1e-14:
                a = a_trial
                best_M = M_trial
                best_a = a.copy()
                improved = True
                break

        if not improved:
            no_improve += 1
            if no_improve > 40:
                break
        else:
            no_improve = 0

        if (it + 1) % 50 == 0:
            J_cur = 2 * n * best_M / (a.sum() ** 2)
            print(f"    LP iter {it+1}: M={best_M:.10f} J={J_cur:.8f} no_improve={no_improve}", flush=True)

    return best_a


def coordinate_descent(a, max_sweeps, deadline, rng=None):
    """Coordinate descent: try multiple values per coordinate, keep best."""
    if rng is None:
        rng = np.random.default_rng(123)
    a = np.maximum(a.astype(np.float64), 1e-8)
    a = a / a.sum()
    n = len(a)
    best_a = a.copy()
    best_J = 2.0 * n * fast_conv(a).max() / (a.sum() ** 2)
    mults = [0.0, 0.2, 0.5, 0.7, 1.3, 1.7, 2.5]

    for sweep in range(max_sweeps):
        if time.monotonic() > deadline:
            break
        order = rng.permutation(n)
        any_improved = False
        for i in order:
            if time.monotonic() > deadline:
                break
            old_val = a[i]
            best_local_J = best_J
            best_local_val = old_val
            s_base = a.sum()
            for m in mults:
                new_val = old_val * m
                a[i] = new_val
                s_new = s_base + (new_val - old_val)
                if s_new < 0.01:
                    continue
                M_new = fast_conv(a).max()
                J_new = 2.0 * n * M_new / (s_new * s_new)
                if J_new < best_local_J - 1e-12:
                    best_local_J = J_new
                    best_local_val = new_val
            a[i] = best_local_val
            if best_local_val != old_val:
                any_improved = True
                s = a.sum()
                if s > 0:
                    a = a / s
                J_now = 2.0 * n * fast_conv(a).max() / (a.sum() ** 2)
                if J_now < best_J:
                    best_J = J_now
                    best_a = a.copy()
                    if (sweep + 1) % 10 == 0:
                        print(f"    Coord sweep {sweep+1}: J={best_J:.8f}", flush=True)
        if not any_improved:
            break
    return best_a


def resample_seq(src, dest_n):
    """Resample 1D sequence to new length via linear interpolation."""
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)


def generate_diverse_inits(n, parent, rng):
    """Generate diverse initializations of length n."""
    inits = []
    t = np.linspace(0, 1, n, endpoint=False)

    if parent is not None:
        inits.append(resample_seq(parent, n) + 0.01)
        p = resample_seq(parent, n)
        inits.append(np.maximum(p + rng.normal(0, 0.05, n), 0.01))
        inits.append(np.maximum(p + rng.normal(0, 0.15, n), 0.01))
        inits.append(np.maximum(p + rng.normal(0, 0.4, n), 0.01))
        # Symmetric version
        sym = 0.5 * (p + p[::-1])
        inits.append(np.maximum(sym + 0.01, 0.01))
        # Antisymmetric perturbation
        anti = np.zeros(n)
        anti[:n//2] = rng.normal(0, 0.1, n//2)
        anti[n//2:] = -anti[:n//2][::-1]
        inits.append(np.maximum(p + anti, 0.01))

    # Basic shapes
    inits.append(np.ones(n) + 0.01)
    inits.append(rng.uniform(0.3, 1.0, n) + 0.01)

    # Cosine windows (reduce boundary effects)
    for alpha in [0.5, 1.0, 2.0, 3.0]:
        inits.append(np.maximum(0.5 * (1 + np.cos(2 * np.pi * t * alpha)), 0.01))

    # Raised cosine with different centering
    inits.append(np.maximum(0.3 + 0.7 * np.cos(np.pi * (2 * t - 1)) ** 2, 0.01))
    inits.append(np.maximum(0.2 + 0.8 * np.exp(-4 * (t - 0.5) ** 2), 0.01))

    # Multi-frequency
    for f1, f2 in [(1, 2), (1, 3), (2, 5), (3, 7)]:
        sig = 0.5 + 0.3 * np.cos(2 * np.pi * f1 * t) + 0.2 * np.cos(2 * np.pi * f2 * t)
        inits.append(np.maximum(sig, 0.01))

    # Block structure
    for nblocks in [4, 8, 16]:
        bv = rng.uniform(0.2, 1.0, nblocks)
        inits.append(np.repeat(bv, n // nblocks + 1)[:n] + 0.01)

    # Smooth random
    ksize = max(5, n // 20)
    kernel = np.ones(ksize) / ksize
    smooth = np.convolve(rng.uniform(0, 1, n), kernel, mode='same')
    inits.append(np.maximum(smooth + 0.1, 0.01))

    # Power law
    inits.append(np.maximum(0.2 + 0.8 * t ** 3, 0.01))
    inits.append(np.maximum(0.2 + 0.8 * (1 - t) ** 3, 0.01))

    # Bimodal
    inits.append(np.maximum(np.exp(-20*(t-0.25)**2) + np.exp(-20*(t-0.75)**2) + 0.05, 0.01))

    # Triangular
    inits.append(np.maximum(1.0 - np.abs(2*t - 1), 0.01))

    return inits

    except (NameError, AttributeError):
        pass

    if parent is None:
        parent = np.ones(256)
        tracker.update(parent)

    # === PHASE 1: Smooth gradient descent across multiple lengths ===
    p1_deadline = time.monotonic() + 250.0
    print(f"\nPHASE 1: Smooth Gradient (budget=250s)", flush=True)

    lengths = [384, 512, 640, 768]
    if parent is not None:
        lengths.insert(0, parent.size)

    candidates = {}  # n -> best_a for that n

    for n in lengths:
        if time.monotonic() > p1_deadline - 20:
            break
        t0 = time.monotonic()
        phase_dl = min(time.monotonic() + max(30, min(60, 200.0 / max(1, n // 128))), p1_deadline - 10)

        inits = generate_diverse_inits(n, parent, rng)
        best_for_n = None
        best_M_for_n = float('inf')

        for a0 in inits:
            if time.monotonic() > phase_dl:
                break
            a0 = np.maximum(a0, 1e-6)
            a0 = a0 / a0.sum()
            iters = max(100, int(3000 / max(1, n // 128)))
            a = smooth_grad_descent(a0, iters, phase_dl, tau0=2.0, tau1=50.0, lr0=0.25, lr1=0.02)
            M_val = fast_conv(a).max()
            if M_val < best_M_for_n:
                best_M_for_n = M_val
                best_for_n = a.copy()

        if best_for_n is not None:
            J_val = compute_J(best_for_n)
            print(f"  n={n:4d}: M={best_M_for_n:.10f} J={J_val:.8f} ({time.monotonic()-t0:.1f}s)", flush=True)
            candidates[n] = best_for_n.copy()
            tracker.update(best_for_n)

    tracker.report("Phase 1 done")

    # === PHASE 2: LP Mirror Descent (main optimization) ===
    # Allocate most time here
    p2_budget = 900.0
    p2_start = time.monotonic()
    print(f"\nPHASE 2: LP Mirror Descent (budget={p2_budget:.0f}s)", flush=True)

    # Sort candidates by J, start with best
    sorted_candidates = sorted(candidates.items(), key=lambda x: compute_J(x[1]))

    # Also include the parent if not already in candidates
    if parent is not None and parent.size not in candidates:
        sorted_candidates.append((parent.size, parent.copy()))

    # Allocate LP time to top candidates (more time for better ones)
    top_k = min(4, len(sorted_candidates))
    time_weights = [1.0, 0.6, 0.4, 0.3][:top_k]
    total_w = sum(time_weights)

    for idx in range(top_k):
        if time.monotonic() > p2_start + p2_budget - 10:
            break
        n_c, a_c = sorted_candidates[idx]
        alloc_time = p2_budget * time_weights[idx] / total_w
        alloc_time = max(60, min(400, alloc_time))
        lp_deadline = min(time.monotonic() + alloc_time, p2_start + p2_budget - 10)
        if time.monotonic() >= lp_deadline:
            continue

        print(f"  LP on n={n_c}, budget={alloc_time:.0f}s", flush=True)
        a_opt = lp_mirror_descent(a_c, 10000, lp_deadline, aggressive=True)
        tracker.update(a_opt)

        # If LP converged, try a second pass from slightly perturbed start
        if time.monotonic() < p2_start + p2_budget - 30:
            a_pert = a_opt.copy() + rng.normal(0, 0.02, a_opt.size)
            a_pert = np.maximum(a_pert, 1e-6)
            a_pert = a_pert / a_pert.sum()
            short_dl = min(time.monotonic() + 60, p2_start + p2_budget - 10)
            a_opt2 = lp_mirror_descent(a_pert, 5000, short_dl, aggressive=False)
            tracker.update(a_opt2)

    tracker.report("Phase 2 done")
    print(f"Phase 2 total: {time.monotonic()-p2_start:.1f}s", flush=True)

    # === PHASE 3: Coordinate Descent Polish ===
    p3_deadline = min(time.monotonic() + 180, deadline - 20)
    print(f"\nPHASE 3: Coordinate Descent (deadline in {p3_deadline - time.monotonic():.0f}s)", flush=True)

    if tracker.best is not None:
        a_cd = coordinate_descent(tracker.best, max_sweeps=200, deadline=p3_deadline, rng=rng)
        tracker.update(a_cd)

        # Try coordinate descent on a slightly perturbed version too
        if time.monotonic() < p3_deadline - 20:
            a_p = tracker.best + rng.normal(0, 0.03, tracker.best.size)
            a_p = np.maximum(a_p, 1e-6)
            a_p = a_p / a_p.sum()
            a_cd2 = coordinate_descent(a_p, max_sweeps=50, deadline=p3_deadline - 5, rng=rng)
            tracker.update(a_cd2)

    tracker.report("Phase 3 done")

    # === PHASE 4: Final LP passes + length extension ===
    print(f"\nPHASE 4: Final Polish (until t={deadline - time.monotonic():.0f}s remaining)", flush=True)

    if tracker.best is not None and time.monotonic() < deadline - 10:
        # One more LP pass on the absolute best
        a_final = lp_mirror_descent(tracker.best, 2000, deadline - 5, aggressive=True)
        tracker.update(a_final)

        # Try extending to 2x length
        if time.monotonic() < deadline - 30:
            n_ext = tracker.best.size * 2
            if n_ext <= 2048:
                a_ext = resample_seq(tracker.best, n_ext)
                a_ext = np.maximum(a_ext, 1e-6)
                a_ext = a_ext / a_ext.sum()
                a_ext = smooth_grad_descent(a_ext, 200, deadline - 15, tau0=3, tau1=30, lr0=0.15, lr1=0.03)
                tracker.update(a_ext)

    # Final LP if time remains
    if tracker.best is not None and time.monotonic() < deadline - 5:
        a = lp_mirror_descent(tracker.best, 500, deadline, aggressive=False)
        tracker.update(a)

    tracker.report("FINAL")

    # Fallback
    if tracker.best is None:
        tracker.best = np.ones(128)
        tracker.best_score = compute_J(tracker.best)

    print(f"\n{'='*60}", flush=True)
    print(f"FINAL: J={tracker.best_score:.8f} n={tracker.best.size} time={time.monotonic()-tracker.t_start:.0f}s", flush=True)
    print(f"{'='*60}", flush=True)

    return [float(x) for x in tracker.best]
