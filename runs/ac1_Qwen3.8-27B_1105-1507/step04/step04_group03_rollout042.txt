Wait, if i ≠ k-i, then a_i appears in two terms: a_i * a_{k-i} (as the first factor) and a_{k-i} * a_i = a_i * a_{k-i} (as the second factor, in the term j=k-i). So ∂c_k/∂a_i = 2*a_{k-i} if i ≠ k-i.

If i = k-i (i.e., k = 2i), then a_i appears only once: a_i * a_i = a_i^2, so ∂c_k/∂a_i = 2*a_i.

So in all cases: ∂c_k/∂a_i = 2*a_{k-i} (when k-i is valid, i.e., max(0,k-n+1) <= i <= min(n-1,k)).

This is nice! The subgradient of the objective (when the subgradient is a single active k) is simply 2*a reversed and shifted.

For the general case (multiple active k's), the subgradient is a convex combination of these.

This means the subgradient computation is very cheap! We can do many subgradient steps.

**Subgradient method with Nesterov acceleration:**
1. Start with a_0
2. At each step, compute b = a*a, find M = max(b), find active set
3. Compute subgradient g = (1/|active|) * sum_{k in active} 2 * shift(a, k)
4. Update: a_{t+1} = projection(a_t - eta_t * g) (clip to [0, ∞) and normalize)
5. Use Nesterov's scheme for accelerated convergence

The step size eta_t = 1/sqrt(t) (standard) or eta_t = C/(t+1) with Polyak's rule if we know M*.

Since we don't know M*, we can use a heuristic: eta_t = initial_eta / (1 + decay * t).

This subgradient method is much faster per iteration than the LP approach (no LP solve needed), so we can do many more iterations. But it converges more slowly asymptotically.

**Hybrid approach**: Use subgradient for fast exploration, then LP for precision.

```python
    """Precompute (rows, cols) structure of the convolution matrix."""
    rows = []
    cols = []
    for k in range(2*n - 1):

        j_max = min(n - 1, k)
        rows.extend([k] * (j_max - j_min + 1))
        cols.extend(range(j_min, j_max + 1))
```

```python
"""
Minimize J(a) = 2*n*max(conv(a,a))/sum(a)^2 using:
1. U-shaped and edge-concentrated initializations
2. LP mirror descent with adaptive step sizes (vectorized matrix construction)
3. Variance-minimization L-BFGS-B polishing
4. Subgradient descent with momentum
5. Multi-length exploration
Key insight: optimal sequences should be U-shaped (large at edges, small in middle)
since edge convolution terms have fewer contributions and must be balanced with
the many-term middle contributions.
"""

import numpy as np
from scipy.optimize import linprog, minimize
from scipy.sparse import csr_matrix
import time

_A_cache = {}

def fast_conv(a):
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]

def compute_J(a):
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))

def get_A_structure(n):
    if n in _A_cache:
        return _A_cache[n]
    k_vals = np.arange(2 * n - 1)[:, None]
    j_vals = np.arange(n)[None, :]
    d = k_vals - j_vals
    valid = (d >= 0) & (d < n)
    rows, cols = np.where(valid)
    d_vals = d[rows, cols].astype(np.int32)
    _A_cache[n] = (rows, cols, d_vals)
    return (rows, cols, d_vals)

def build_A_fast(n, a):
    rows, cols, d_vals = get_A_structure(n)
    vals = a[d_vals]
    return csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))

def lp_solve(a, M):
    n = len(a)
    A = build_A_fast(n, a)
    c_obj = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    try:
        r = linprog(c_obj, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
        if r.success and r.x is not None:
            g = np.maximum(r.x, 0)
            gs = g.sum()
            if gs > 1e-15:
                return g / gs
    except Exception:
        pass
    return None

def lp_mirror_descent(a, deadline, verbose=False):
    a = np.maximum(a.astype(np.float64), 1e-10)
    a = a / a.sum()
    best_J = compute_J(a)
    best_a = a.copy()
    t = 0.25
    no_improve = 0
    it = 0

    while time.monotonic() < deadline:
        it += 1
        b = fast_conv(a)
        M = float(b.max())
        g = lp_solve(a, M)
        if g is None:
            break

        best_trial_J = current_J if 'current_J' in dir() else best_J
        current_J = best_J
        best_trial_a = None
        best_ts = t

        step_candidates = [t, t * 0.8, t * 0.6, t * 0.4, t * 0.2, 0.05]
        for ts in step_candidates:
            ts = min(max(ts, 0.01), 0.95)
            a_trial = (1.0 - ts) * a + ts * g
            a_trial = np.maximum(a_trial, 0)
            s = a_trial.sum()
            if s < 1e-10:
                continue
            a_trial = a_trial / s
            J_trial = compute_J(a_trial)
            if J_trial < best_trial_J - 1e-15:
                best_trial_J = J_trial
                best_trial_a = a_trial.copy()
                best_ts = ts

        if best_trial_a is not None:
            a = best_trial_a
            current_J = best_trial_J
            if current_J < best_J - 1e-15:
                best_J = current_J
                best_a = a.copy()
                no_improve = 0
                if verbose and it % 200 == 0:
                    print(f"      iter {it}: J={best_J:.8f} t={best_ts:.3f}", flush=True)
            else:
                no_improve += 1
            t = min(0.85, t * 1.03)
        else:
            no_improve += 1
            t = max(0.01, t * 0.75)
            if no_improve > 60:
                break

    return best_a, best_J

def subgradient_polish(a, deadline):
    a = np.maximum(a, 1e-10).astype(np.float64)
    a = a / a.sum()
    best_J = compute_J(a)
    best_a = a.copy()
    momentum = np.zeros_like(a)
    n = len(a)
    it = 0

    while time.monotonic() < deadline:
        it += 1
        b = fast_conv(a)
        M = b.max()
        tol = 1e-9 * M
        active = np.where(b >= M - tol)[0]

        grad = np.zeros(n)
        for k in active:
            i_lo = max(0, k - n + 1)
            i_hi = min(n - 1, k)
            segment = a[k - i_hi:k - i_lo + 1]
            grad[i_lo:i_hi + 1] += 2.0 * segment[::-1]

        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-15:
            break

        eta = 0.08 / np.sqrt(it + 1.0)
        momentum = 0.82 * momentum + 0.18 * grad / gnorm
        a_new = a - eta * momentum
        a_new = np.maximum(a_new, 0)
        s = a_new.sum()
        if s < 1e-10:
            break
        a_new = a_new / s

        J_new = compute_J(a_new)
        if J_new < best_J:
            best_J = J_new
            best_a = a_new.copy()
        a = a_new

    return best_a, best_J

def variance_objective(x):
    ax = np.maximum(x, 0)
    s = ax.sum()
    if s < 1e-8:
        return 1e10
    ax = ax / s
    b = fast_conv(ax)
    n = len(ax)
    mu = 1.0 / (2 * n - 1)
    return float(np.sum((b - mu) ** 2))

def variance_gradient(x):
    ax = np.maximum(x, 1e-15)
    s = ax.sum()
    ax = ax / s
    n = len(ax)
    b = fast_conv(ax)
    mu = 1.0 / (2 * n - 1)
    w = b - mu
    size = 1
    while size < 2 * n:
        size <<= 1
    grad = 4.0 * np.real(np.fft.ifft(np.conj(np.fft.fft(w, size)) * np.fft.fft(ax, size)))[:n]
    return grad

def variance_lbb(a, deadline):
    a = np.maximum(a, 1e-10).astype(np.float64)
    a = a / a.sum()
    best_J = compute_J(a)
    best_a = a.copy()

    def obj(x):
        ax = np.maximum(x, 0)
        s = ax.sum()
        if s < 1e-8:
            return 1e10
        ax = ax / s
        b = fast_conv(ax)
        n = len(ax)
        mu = 1.0 / (2 * n - 1)
        return float(np.sum((b - mu) ** 2))

    try:
        res = minimize(obj, a.copy(), method='L-BFGS-B',
                       bounds=[(0, None)] * len(a),
                       options={'maxiter': 300, 'max_fun': 100000, 'ftol': 1e-18, 'gtol': 1e-14})
        a_res = np.maximum(res.x, 0)
        J_res = compute_J(a_res)
        if J_res < best_J:
            best_J = J_res
            best_a = a_res.copy()
    except Exception:
        pass

    return best_a, best_J

def lbb_maxobj(x):
    ax = np.maximum(x, 0)
    s = ax.sum()
    if s < 1e-8:
        return 1e10
    ax = ax / s
    b = fast_conv(ax)
    return float(b.max())

def make_u_shape(n, gamma=0.9):
    t = np.arange(n, dtype=np.float64) / n
    a = 1.0 + gamma * np.cos(2.0 * np.pi * t)
    return np.maximum(a, 0.01)

def make_u_shape2(n, alpha=2.0):
    k = np.arange(n, dtype=np.float64)
    center = (n - 1) / 2.0
    x = np.abs(k - center) / center
    a = 1.0 + alpha * x ** 2
    return a

def make_bimodal(n, width=0.15):
    t = np.arange(n, dtype=np.float64) / n
    a = 0.05 + np.exp(-((t - 0.05) / width) ** 2) + np.exp(-((t - 0.95) / width) ** 2)
    return a

def make_random_smooth(n, rng, kernel_size=12):
    raw = rng.uniform(0, 1, n)
    kernel = np.ones(kernel_size) / kernel_size
    a = np.convolve(raw, kernel, mode='same')
    return np.maximum(a + 0.05, 0.01)

def make_power_edge(n, p=0.7):
    k = np.arange(n, dtype=np.float64)
    center = (n - 1) / 2.0
    x = np.abs(k - center) / center
    a = 1.0 + p * (x / (1.0 - x + 1e-6)) ** 0.5
    return a

def resample_seq(src, dest_n):
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    return np.maximum(np.interp(t_new, t_old, src), 0.01)

def propose_candidate(budget_s=1500):
    """
    Hybrid optimizer: U-shaped inits + LP mirror descent + variance L-BFGS-B
    + subgradient polish + multi-length exploration.
    """
    rng = np.random.default_rng(777)
    deadline = time.monotonic() + float(budget_s) - 20.0
    best_a = None
    best_J = float('inf')

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        if p.ndim == 1 and p.size > 0 and np.all(np.isfinite(p)) and p.min() >= 0:
            parent = np.maximum(p, 0)
            J_p = compute_J(parent)
            print(f"Parent: n={parent.size}, J={J_p:.8f}", flush=True)
            if J_p < best_J:
                best_J = J_p
                best_a = parent.copy()
    except (NameError, AttributeError):
        pass

    n_base = parent.size if parent is not None else 640

    # Generate diverse initializations
    inits = []
    inits.append(('u0.7', make_u_shape(n_base, 0.7)))
    inits.append(('u0.85', make_u_shape(n_base, 0.85)))
    inits.append(('u0.95', make_u_shape(n_base, 0.95)))
    inits.append(('u1.0', make_u_shape2(n_base, 1.5)))
    inits.append(('u2.0', make_u_shape2(n_base, 3.0)))
    inits.append(('u3.0', make_u_shape2(n_base, 5.0)))
    inits.append(('bimod', make_bimodal(n_base, 0.12)))
    inits.append(('bimod2', make_bimodal(n_base, 0.2)))
    inits.append(('edge', make_power_edge(n_base, 1.0)))
    inits.append(('edge2', make_power_edge(n_base, 2.0)))
    inits.append(('rand1', make_random_smooth(n_base, rng, 16)))
    inits.append(('rand2', make_random_smooth(n_base, rng, 8)))
    inits.append(('const', np.ones(n_base)))

    if parent is not None:
        inits.append(('parent', parent.copy()))
        inits.append(('pert1', np.maximum(parent + rng.normal(0, 0.02, n_base), 0.01)))
        inits.append(('pert2', np.maximum(parent + rng.normal(0, 0.05, n_base), 0.01)))
        # Reshape parent to be more U-shaped
        t = np.arange(n_base) / n_base
        u_target = 1.0 + 0.9 * np.cos(2.0 * np.pi * t)
        u_target = np.maximum(u_target, 0.01)
        u_target = u_target / u_target.sum()
        p_norm = np.maximum(parent, 0.01)
        p_norm = p_norm / p_norm.sum()
        inits.append(('blend_u', 0.5 * p_norm + 0.5 * u_target))
        inits.append(('blend_u2', 0.3 * p_norm + 0.7 * u_target))

    # Normalize all
    inits_norm = []
    for name, a in inits:
        a = np.maximum(a, 0.01)
        s = a.sum()
        if s > 1e-8:
            inits_norm.append((name, a / s))

    # Phase 1: Quick screening
    print(f"\n{'='*55}", flush=True)
    print(f"Phase 1: Quick LP screening (n={n_base}, {len(inits_norm)} inits)", flush=True)
    print(f"{'='*55}", flush=True)
    t0 = time.monotonic()

    screen_results = []
    screen_time = min(25.0, (deadline - t0) / max(1, len(inits_norm)) * 0.4)
    for name, init in inits_norm:
        if time.monotonic() > deadline - 400:
            break
        a_scr, J_scr = lp_mirror_descent(init, time.monotonic() + screen_time)
        screen_results.append((J_scr, name, a_scr))
        print(f"  {name:10s}: J={J_scr:.8f}", flush=True)
        if J_scr < best_J:
            best_J = J_scr
            best_a = a_scr.copy()

    print(f"Screening done in {time.monotonic()-t0:.1f}s", flush=True)
    screen_results.sort(key=lambda x: x[0])

    # Phase 2: Long LP on top candidates
    print(f"\n{'='*55}", flush=True)
    print(f"Phase 2: Long LP refinement (top 3)", flush=True)
    print(f"{'='*55}", flush=True)
    t0 = time.monotonic()

    top_k = min(3, len(screen_results))
    remaining = deadline - 150 - time.monotonic()
    per_budget = max(30.0, remaining / top_k * 0.7)

    for i in range(top_k):
        if time.monotonic() > deadline - 130:
            break
        J_c, name_c, a_c = screen_results[i]
        dl = time.monotonic() + min(per_budget, deadline - 130 - time.monotonic() - i * 0)
        print(f"  [{name_c}] start J={J_c:.8f}, budget={min(per_budget, deadline-130-time.monotonic()):.0f}s", flush=True)
        a_ref, J_ref = lp_mirror_descent(a_c, dl, verbose=True)
        print(f"  [{name_c}] end J={J_ref:.8f}", flush=True)
        if J_ref < best_J:
            best_J = J_ref
            best_a = a_ref.copy()

    print(f"Long LP done in {time.monotonic()-t0:.1f}s", flush=True)

    # Phase 3: Variance-based L-BFGS-B
    if time.monotonic() < deadline - 80 and best_a is not None:
        print(f"\n{'='*55}", flush=True)
        print(f"Phase 3: Variance L-BFGS-B", flush=True)
        print(f"{'='*55}", flush=True)
        t0 = time.monotonic()
        a_var, J_var = variance_lbb(best_a, time.monotonic() + min(50, deadline - time.monotonic() - 60))
        print(f"  Variance L-BFGS-B: J={J_var:.8f} ({time.monotonic()-t0:.1f}s)", flush=True)
        if J_var < best_J:
            best_J = J_var
            best_a = a_var.copy()

    # Phase 4: LP from variance result (if improved)
    if time.monotonic() < deadline - 50 and best_a is not None:
        print(f"\n{'='*55}", flush=True)
        print(f"Phase 4: LP polish after variance", flush=True)
        print(f"{'='*55}", flush=True)
        t0 = time.monotonic()
        a_lp2, J_lp2 = lp_mirror_descent(best_a, time.monotonic() + min(40, deadline - time.monotonic() - 30), verbose=True)
        print(f"  LP polish: J={J_lp2:.8f} ({time.monotonic()-t0:.1f}s)", flush=True)
        if J_lp2 < best_J:
            best_J = J_lp2
            best_a = a_lp2.copy()

    # Phase 5: Subgradient polish
    if time.monotonic() < deadline - 25 and best_a is not None:
        print(f"\n{'='*55}", flush=True)
        print(f"Phase 5: Subgradient polish", flush=True)
        print(f"{'='*55}", flush=True)
        t0 = time.monotonic()
        a_sub, J_sub = subgradient_polish(best_a, time.monotonic() + min(20, deadline - time.monotonic() - 10))
        print(f"  Subgradient: J={J_sub:.8f} ({time.monotonic()-t0:.1f}s)", flush=True)
        if J_sub < best_J:
            best_J = J_sub
            best_a = a_sub.copy()

    # Phase 6: Final L-BFGS-B on max objective
    if time.monotonic() < deadline - 5 and best_a is not None:
        print(f"\n{'='*55}", flush=True)
        print(f"Phase 6: Final L-BFGS-B (max objective)", flush=True)
        print(f"{'='*55}", flush=True)
        t0 = time.monotonic()
        try:
            res = minimize(lbb_maxobj, best_a.copy(), method='L-BFGS-B',
                          bounds=[(0, None)] * len(best_a),
                          options={'maxiter': 200, 'max_fun': 50000, 'ftol': 1e-18})
            a_fin = np.maximum(res.x, 0)
            J_fin = compute_J(a_fin)
            print(f"  Final L-BFGS-B: J={J_fin:.8f} ({time.monotonic()-t0:.1f}s)", flush=True)
            if J_fin < best_J:
                best_J = J_fin
                best_a = a_fin.copy()
        except Exception as e:
            print(f"  Error: {e}", flush=True)

    # Phase 7: Multi-length exploration
    if time.monotonic() < deadline - 10 and best_a is not None:
        print(f"\n{'='*55}", flush=True)
        print(f"Phase 7: Multi-length exploration", flush=True)
        print(f"{'='*55}", flush=True)
        base_n = best_a.size
        candidates_n = []
        for delta in [-32, 32, 64, -64, 128]:
            nn = base_n + delta
            if 64 <= nn <= 2048 and nn != base_n:
                candidates_n.append(nn)
        # Also try doubling
        if base_n * 2 <= 1500:
            candidates_n.append(base_n * 2)

        seen = set()
        for nn in candidates_n:
            if time.monotonic() > deadline - 5:
                break
            if nn in seen:
                continue
            seen.add(nn)
            a_try = resample_seq(best_a, nn)
            a_try = a_try / a_try.sum()
            a_try, J_try = lp_mirror_descent(a_try, time.monotonic() + 10.0)
            print(f"  n={nn}: J={J_try:.8f}", flush=True)
            if J_try < best_J:
                best_J = J_try
                best_a = a_try.copy()

    # Final
    print(f"\n{'='*55}", flush=True)
    if best_a is not None:
        final_score = evaluate_sequence(best_a.tolist())
        print(f"FINAL: J={final_score:.8f} (internal: {best_J:.8f}) n={best_a.size}", flush=True)
        print(f"Total time: {time.monotonic() - (deadline - budget_s + 20):.1f}s", flush=True)
        print(f"{'='*55}", flush=True)
        return [float(x) for x in best_a]
    else:
        print("FALLBACK: returning constant sequence", flush=True)
        return [1.0 / 640] * 640
```
