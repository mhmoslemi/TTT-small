    grad = 4.0 * n * cross[:n] / (s * s) - 2.0 * F_val / s
    return F_val, grad


def nesterov_smoothed(a0, deadline, betas=None, max_iters_per_beta=100):
    """Nesterov-accelerated gradient on smoothed objective."""
    if betas is None:
        betas = [0.5, 1.0, 2.0, 5.0, 10.0, 20.0, 40.0, 80.0, 160.0]

    n = len(a0)
    a = np.maximum(a0.astype(np.float64), 0)
    a = a / a.sum()

    best_a = a.copy()
    best_J = compute_J(a)

    for beta in betas:
        if time.monotonic() > deadline:
            break

        # Estimate step size
        conv = fast_conv(a)
        L = 4.0 * n * beta * max(float(conv.max()), 1e-6) * 2.0 + 1.0
        lr = 1.0 / L

        x = a.copy()
        t_k = 1.0
        y = a.copy()

        for it in range(max_iters_per_beta):
            if time.monotonic() > deadline:
                break

            F_y, grad_y = smoothed_obj_grad(y, beta)
            if not np.isfinite(F_y):
                break

            # Gradient step
            x_new = y - lr * grad_y

            # Project to simplex
            x_new = np.maximum(x_new, 0)
            s = x_new.sum()
            if s < 1e-10:
                x_new = np.ones(n) / n
            else:
                x_new = x_new / s

            # Nesterov update
            t_new = (1.0 + np.sqrt(1.0 + 4.0 * t_k * t_k)) / 2.0
            momentum = (t_k - 1.0) / t_new
            y = x_new + momentum * (x_new - x)
            t_k = t_new
            x = x_new
            a = x.copy()

            # Track
            if it % 20 == 0:
                J = compute_J(a)
                if J < best_J:
                    best_J = J
                    best_a = a.copy()

        J = compute_J(a)
        if J < best_J:
            best_J = J
            best_a = a.copy()

    return best_a, best_J


def resample_seq(src, dest_n):
    """Resample 1D sequence to new length via linear interpolation."""
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    result = np.interp(t_new, t_old, src)
    return np.maximum(result, 0.0)


def generate_initializations(n, rng):
    """Generate diverse initial sequences of length n (all on simplex)."""
    inits = []
    t = np.linspace(0, 1, n, endpoint=False)

    # Uniform
    inits.append(np.ones(n) / n)

    # Triangular (increasing)
    tri = np.arange(1, n + 1, dtype=np.float64)
    inits.append(tri / tri.sum())

    # Triangular (decreasing)
    tri_d = tri[::-1].copy()
    inits.append(tri_d / tri_d.sum())

    # U-shaped (high at edges)
    u_shape = 1.0 + 2.0 * (2.0 * t - 1.0) ** 2
    inits.append(u_shape / u_shape.sum())

    # Cosine various frequencies
    for freq in [1, 2, 3, 4, 5, 7, 10]:
        c = 1.0 + 0.4 * np.cos(2.0 * np.pi * freq * t)
        c = np.maximum(c, 0.01)
        inits.append(c / c.sum())

    # Gaussian centered
    g = np.exp(-8.0 * (t - 0.5) ** 2) + 0.05
    inits.append(g / g.sum())

    # Two bumps
    two = np.exp(-20.0 * (t - 0.25) ** 2) + np.exp(-20.0 * (t - 0.75) ** 2) + 0.03
    inits.append(two / two.sum())

    # Power law
    pl = (t + 0.05) ** 1.5
    inits.append(pl / pl.sum())

    # Random smooth
    noise = rng.normal(0, 1, n)
    kernel = np.exp(-0.5 * (np.arange(-5, 6) / 2.0) ** 2)
    kernel /= kernel.sum()
    smooth_noise = np.convolve(noise, kernel, mode='same')
    rn = np.abs(smooth_noise) + 0.1
    inits.append(rn / rn.sum())

    # Random
    r = rng.uniform(0.5, 1.0, n)
    inits.append(r / r.sum())

    # Zipf-like
    z = 1.0 / (np.arange(1, n + 1) ** 0.7)
    inits.append(z / z.sum())

    # Symmetric triangle
    mid = n // 2
    sy = np.array([min(i + 1, n - i) for i in range(n)], dtype=np.float64)
    inits.append(sy / sy.sum())

    return inits


class Tracker:
    def __init__(self):
        self.best = None
        self.best_score = float('inf')
        self.t_start = time.monotonic()

    def update(self, a):
        a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
        if a.size == 0 or a.sum() < 0.01:
            return
        score = evaluate_sequence(a.tolist())
        if np.isfinite(score) and score < self.best_score:
            self.best = a.copy()
            self.best_score = score
            elapsed = time.monotonic() - self.t_start
            print(f"  * NEW BEST: J={score:.8f} n={a.size} t={elapsed:.1f}s", flush=True)

    def report(self, label=""):
        if self.best is not None:
            print(f"  {label} best: J={self.best_score:.8f} n={self.best.size}", flush=True)


def propose_candidate(seed=42, budget_s=1500, **kwargs):
    """
    Multi-phase optimization:
    Phase 1: Diverse initialization + quick screening
    Phase 2: Nesterov smoothed gradient for basin finding
    Phase 3: Coordinate descent for precise local optimization
    Phase 4: Multi-restart with perturbations
    Phase 5: Final coordinate descent polish + length exploration
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + float(budget_s) - 15.0
    tracker = Tracker()

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        if p.ndim == 1 and p.size > 0 and np.all(np.isfinite(p)):
            parent = np.maximum(p, 0.0)
            ps = parent.sum()
            if ps > 0.01:
                parent = parent / ps
            tracker.update(np.clip(parent, 0, 1000))
            print(f"Parent loaded: n={parent.size}, J={tracker.best_score:.8f}", flush=True)
    except (NameError, AttributeError):
        pass

    if parent is None:
        parent = np.ones(512) / 512.0
        tracker.update(np.ones(512))
        print(f"Default parent: n=512, J={tracker.best_score:.8f}", flush=True)

    # === Phase 1: Initialization and screening (200s) ===
    phase1_deadline = time.monotonic() + 150.0
    print(f"\n{'='*50} Phase 1: Initialization (n={parent.size}) {'='*50}", flush=True)

    n_base = parent.size
    init_list = generate_initializations(n_base, rng)

    # Also include parent
    init_list.append(parent.copy())

    # Quick coordinate descent on each init
    best_a = None
    best_J = float('inf')

    for idx, a0 in enumerate(init_list):
        if time.monotonic() > phase1_deadline:
            break
        a0 = np.maximum(a0, 0)
        a0_sum = a0.sum()
        if a0_sum < 1e-10:
            continue
        a0 = a0 / a0_sum

        # Quick CD: 5 passes
        a_opt = coord_descent_passes(a0, 5, phase1_deadline, track_best=False)
        J = compute_J(a_opt)
        if J < best_J:
            best_J = J
            best_a = a_opt.copy()
        tracker.update(a_opt)
        if idx < 5:
            print(f"  init[{idx}]: J={J:.6f}", flush=True)

    if best_a is None:
        best_a = parent.copy()
        best_J = compute_J(best_a)

    print(f"  Phase 1 best: J={best_J:.6f}", flush=True)

    # === Phase 2: Nesterov smoothed gradient (400s) ===
    phase2_deadline = time.monotonic() + 350.0
    print(f"\n{'='*50} Phase 2: Nesterov Smoothed Gradient {'='*50}", flush=True)

    # Run from best init
    a_nest, J_nest = nesterov_smoothed(best_a, phase2_deadline,
                                        betas=[0.5, 1.0, 2.0, 5.0, 10.0, 20.0, 40.0])
    tracker.update(a_nest)
    print(f"  Nesterov result: J={compute_J(a_nest):.6f}", flush=True)

    # Also try from a few other inits
    for idx in [2, 4, 7]:
        if time.monotonic() > phase2_deadline:
            break
        if idx < len(init_list):
            a_try, J_try = nesterov_smoothed(init_list[idx], phase2_deadline,
                                              betas=[1.0, 5.0, 20.0])
            tracker.update(a_try)

    # === Phase 3: Coordinate descent refinement (400s) ===
    phase3_deadline = time.monotonic() + 400.0
    print(f"\n{'='*50} Phase 3: Coordinate Descent Refinement {'='*50}", flush=True)

    # Use the best found so far
    a_curr = tracker.best.copy() if tracker.best is not None else best_a
    a_curr = np.maximum(a_curr, 0)
    a_curr = a_curr / a_curr.sum()

    # Long coordinate descent run
    n_passes_est = int((phase3_deadline - time.monotonic()) / 0.25)
    n_passes_est = max(50, min(n_passes_est, 3000))
    a_curr = coord_descent_passes(a_curr, n_passes_est, phase3_deadline, tracker=tracker)
    print(f"  CD passes done, J={compute_J(a_curr):.6f}", flush=True)

    # Alternate: CD + Nesterov
    if time.monotonic() < phase3_deadline:
        for round in range(3):
            if time.monotonic() > phase3_deadline:
                break
            a_curr = coord_descent_passes(a_curr, 50, phase3_deadline, tracker=tracker)
            a_curr, _ = nesterov_smoothed(a_curr, phase3_deadline,
                                           betas=[5.0, 20.0, 50.0], max_iters_per_beta=50)
            a_curr = np.maximum(a_curr, 0)
            a_curr = a_curr / a_curr.sum()
            tracker.update(a_curr)

    # === Phase 4: Multi-restart (250s) ===
    phase4_deadline = time.monotonic() + 250.0
    print(f"\n{'='*50} Phase 4: Multi-Restart {'='*50}", flush=True)

    n_restart = len(tracker.best) if tracker.best is not None else n_base
    base = tracker.best.copy() if tracker.best is not None else a_curr
    base = np.maximum(base, 0)
    base = base / base.sum()

    restart_strategies = [
        ('gauss_small', 0.03),
        ('gauss_med', 0.08),
        ('gauss_large', 0.15),
        ('uniform', 0.1),
        ('spike', 5),
        ('spike_large', 15),
        ('gauss_small', 0.02),
        ('local_block', 0.05),
    ]

    attempt = 0
    for pert_type, param in restart_strategies:
        for trial in range(4):
            if time.monotonic() > phase4_deadline:
                break
            attempt += 1
            a = base.copy()

            if pert_type == 'gauss_small' or pert_type == 'gauss_med' or pert_type == 'gauss_large':
                a = a + rng.normal(0, param, n_restart) * a
            elif pert_type == 'uniform':
                a = a + rng.uniform(-param, param, n_restart) * a
            elif pert_type == 'spike':
                k = min(param, n_restart // 4)
                idx_s = rng.choice(n_restart, size=k, replace=False)
                a[idx_s] *= rng.uniform(0.2, 4.0, k)
            elif pert_type == 'spike_large':
                k = min(param, n_restart // 4)
                idx_s = rng.choice(n_restart, size=k, replace=False)
                a[idx_s] *= rng.uniform(0.1, 5.0, k)
            elif pert_type == 'local_block':
                # Perturb a random contiguous block
                block_len = max(4, n_restart // 16)
                start = rng.integers(0, n_restart - block_len)
                a[start:start + block_len] += rng.normal(0, param, block_len)

            a = np.maximum(a, 0)
            s = a.sum()
            if s < 1e-10:
                continue
            a = a / s

            # Quick CD to refine
            budget_here = min(40.0, phase4_deadline - time.monotonic())
            n_p = max(10, int(budget_here / 0.3))
            a = coord_descent_passes(a, n_p, phase4_deadline, tracker=tracker)

    print(f"  {attempt} restarts done", flush=True)

    # === Phase 5: Final polish + length exploration (remaining) ===
    remaining = deadline - time.monotonic()
    if remaining > 20:
        print(f"\n{'='*50} Phase 5: Final Polish ({remaining:.0f}s remaining) {'='*50}", flush=True)

        # Long CD on best
        a_final = tracker.best.copy() if tracker.best is not None else a_curr
        a_final = np.maximum(a_final, 0)
        a_final = a_final / a_final.sum()

        n_p = max(50, int(remaining * 0.5 / 0.25))
        a_final = coord_descent_passes(a_final, n_p, deadline, tracker=tracker)
        print(f"  Final CD: J={compute_J(a_final):.6f}", flush=True)

        # Try a few nearby lengths
        if time.monotonic() < deadline - 30:
            n_best = len(tracker.best) if tracker.best is not None else n_base
            for n_try in [n_best - 64, n_best + 64, n_best - 128, n_best + 128, 768, 896, 1152, 1280]:
                if n_try < 128 or n_try > 2048:
                    continue
                if time.monotonic() > deadline - 10:
                    break
                a_res = resample_seq(tracker.best, n_try)
                a_res = np.maximum(a_res, 0)
                s = a_res.sum()
                if s < 1e-10:
                    continue
                a_res = a_res / s

                budget_here = min(30.0, deadline - time.monotonic() - 5.0)
                if budget_here < 10:
                    break
                n_p = max(5, int(budget_here / 0.3))
                a_res = coord_descent_passes(a_res, n_p, deadline - 5.0, tracker=tracker)

    # Ensure output is valid
    if tracker.best is None:
        tracker.best = np.ones(512)
        tracker.best_score = compute_J(tracker.best)

    result = np.clip(tracker.best, 0.0, 1000.0)
    if result.sum() < 0.01:
        result = np.ones(128)

    print(f"\n{'='*50} FINAL RESULT {'='*50}", flush=True)
    print(f"J={tracker.best_score:.8f} n={tracker.best.size} "
          f"time={time.monotonic()-Tracker.t_start:.1f}s", flush=True)

    return [float(x) for x in result]
