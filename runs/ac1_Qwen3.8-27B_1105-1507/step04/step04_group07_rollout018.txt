Minimizes J(a) = 2*n*max(convolve(a,a))/(sum(a))^2 using:
1. Aggressive LP best-response iteration (primary driver)
2. LSE (log-sum-exp) smoothed gradient descent for polishing

4. Simulated annealing with LP refinement for escape from local minima
5. Adaptive constraint tightening based on LP dual information

import numpy as np
from scipy.optimize import linprog
from scipy.sparse import csr_matrix
import time

# ===== Precomputed sparsity cache =====
_spar_cache = {}

def get_sparsity(n):
    """Get sparse structure of the convolution constraint matrix."""
    if n not in _spar_cache:
        rows = []
        cols = []
        for k in range(2 * n - 1):
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            rows.extend([k] * (je - js + 1))
            cols.extend(range(js, je + 1))
        _spar_cache[n] = (np.array(rows), np.array(cols), len(rows))
    return _spar_cache[n]

def fast_conv(a):
    """Fast linear convolution via FFT."""
    n = len(a)
    s = 1
    while s < 2 * n - 1:
        s <<= 1
    fa = np.fft.rfft(a, s)

"""
Minimizes J(a) = 2*n*max(conv(a,a))/sum(a)^2 using aggressive LP best-response
iteration, LSE-smoothed gradient descent, multi-length search, and simulated
annealing. Key improvements over previous: faster LP calls, correct LSE gradient,
progressive tightening schedule, and better time allocation.

import numpy as np
from scipy.optimize import linprog
from scipy.sparse import csr_matrix
import time

_spar_cache = {}

def get_sparsity(n):
    if n not in _spar_cache:
        rows = []
        cols = []
        for k in range(2 * n - 1):
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            rows.extend([k] * (je - js + 1))
            cols.extend(range(js, je + 1))
        _spar_cache[n] = (np.array(rows), np.array(cols), len(rows))
    return _spar_cache[n]

def fast_conv(a):
    n = len(a)
    s = 1
    while s < 2 * n - 1:
        s <<= 1
    fa = np.fft.rfft(a, s)
    return np.fft.irfft(fa * fa, s)[:2 * n - 1]

def compute_J(a):
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))

def lp_best_response(a, M, time_limit=2.0):
    n = len(a)
    rows, cols, nnz = get_sparsity(n)
    vals = np.empty(nnz)
    for i in range(nnz):
        k = rows[i]
        j = cols[i]
        vals[i] = a[k - j]
    A = csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                options={'presolve': True, 'time_limit': time_limit})
    if r.success and r.x is not None:
        g = np.maximum(r.x, 0.0)
        if g.sum() > 1e-10:
            return g
    elif r.status == 3:  # time limit
        # Try to use the best solution found
        if r.x is not None:
            g = np.maximum(r.x, 0.0)
            if g.sum() > 1e-10:
                return g
    return None

def run_lp_opt(a, max_iters, deadline, tracker, tighten_start=0.02, tighten_max=0.15):
    a = np.maximum(a.astype(np.float64), 1e-8)
    a = a / a.sum()
    n = len(a)
    current_J = compute_J(a)
    tracker.update(a)
    tighten = tighten_start
    
    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        b = fast_conv(a)
        M = float(b.max())
        M_tight = M * (1.0 - tighten)
        g = lp_best_response(a, M_tight, time_limit=2.0)
        if g is None:
            # Try without tightening
            g = lp_best_response(a, M, time_limit=2.0)
            if g is None:
                tighten *= 0.5
                continue
            tighten = max(0.005, tighten * 0.5)
        
        g = np.maximum(g, 0.0)
        gs = g.sum()
        if gs < 1e-10:
            break
        g = g / gs
        
        # Try multiple mixing ratios
        best_a = None
        best_J = current_J
        progress = it / max(1, max_iters)
        base_t = max(0.1, 0.6 * (1.0 - progress * 0.5))
        for t in [base_t, base_t * 0.5, base_t * 1.5, 0.9, 0.3, 0.7]:
            if t > 1.0 or t <= 0:
                continue
            a_trial = (1.0 - t) * a + t * g
            a_trial = np.maximum(a_trial, 0.0)
            ss = a_trial.sum()
            if ss < 1e-10:
                continue
            a_trial = a_trial / ss
            J_trial = compute_J(a_trial)
            if J_trial < best_J - 1e-14:
                best_J = J_trial
                best_a = a_trial.copy()
        
        if best_a is None:
            # No improvement; try smaller tighten
            tighten *= 0.7
            if tighten < 0.003:
                break
            continue
        
        a = best_a
        current_J = best_J
        # Increase tighten gradually
        tighten = min(tighten_max, tighten * 1.05)
        if it % 10 == 0:
            tracker.update(a)
    tracker.update(a)
    return a

def lse_gradient_descent(a, max_iters, deadline, tracker):
    a = np.maximum(a.astype(np.float64), 1e-6)
    a = a / a.sum()
    n = len(a)
    current_J = compute_J(a)
    tracker.update(a)
    tau = 30.0
    lr = 2.0 / n
    momentum = 0.8
    v = np.zeros(n)
    stall_count = 0
    
    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        b = fast_conv(a)
        b_max = float(b.max())
        exp_b = np.exp(tau * (b - b_max))
        p = exp_b / exp_b.sum()
        
        # Compute gradient using convolution
        flip_p = p[::-1]
        conv1 = np.convolve(flip_p, a, mode='full')
        S1 = conv1[n-1 : 2*n-1][::-1]
        S2 = np.convolve(p, a, mode='full')[:n]
        dLSE = S1 + S2
        lse_val = b_max + np.log(exp_b.sum()) / tau
        grad = 2.0 * n * (dLSE - 2.0 * lse_val * a)
        
        v = momentum * v + (1.0 - momentum) * grad
        a_new = a - lr * v
        a_new = np.maximum(a_new, 1e-6)
        a_new = a_new / a_new.sum()
        J_new = compute_J(a_new)
        
        if J_new < current_J - 1e-15:
            a = a_new
            current_J = J_new
            stall_count = 0
            tau = min(500.0, tau * 1.02)
        else:
            stall_count += 1
            lr *= 0.7
            v *= 0.5
            if stall_count > 50:
                break
    
    tracker.update(a)
    return a

def flat_spectrum_init(n, rng):
    half = n // 2 + 1
    angles = rng.uniform(0, 2 * np.pi, half)
    A = np.zeros(n, dtype=complex)
    A[:half] = np.exp(1j * angles[:half])
    for k in range(1, n - n // 2):
        A[k] = np.conj(A[n - k])
    a = np.fft.ifft(A).real
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-10:
        return np.ones(n) / n
    return a / s

def resample_seq(src, dest_n):
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)

def generate_inits(n, rng, parent):
    inits = []
    if parent is not None and parent.size > 0:
        inits.append(resample_seq(parent, n))
        inits.append(resample_seq(parent[::-1], n))
    
    inits.append(np.ones(n) / n)
    inits.append(rng.uniform(0.1, 1.0, n) / n)
    inits.append(flat_spectrum_init(n, rng))
    inits.append(flat_spectrum_init(n, rng))
    
    t = np.linspace(0, 1, n, endpoint=False)
    for freq in [2, 3, 5, 7, 11]:
        inits.append(1.0 + 0.4 * np.cos(2 * np.pi * freq * t + rng.uniform(0, 6.28)))
    inits.append(0.5 + 0.5 * np.cos(np.pi * t) ** 2)
    inits.append(1.0 + 0.3 * np.cos(2 * np.pi * 3.7 * t) + 0.2 * np.cos(2 * np.pi * 7.1 * t))
    
    # Multi-frequency
    inits.append(1.0 + 0.25 * np.sin(2 * np.pi * t) + 0.2 * np.sin(4 * np.pi * t)
                + 0.15 * np.sin(7 * np.pi * t) + 0.1 * np.sin(11 * np.pi * t))
    
    for x in inits:
        x = np.maximum(x, 0.001).astype(np.float64)
        x = x / x.sum()
    return inits

def simulated_annealing(a, deadline, tracker, n_restarts=30, t0=0.08):
    rng = np.random.default_rng(777)
    n = a.size
    a = a.copy().astype(np.float64)
    a = a / a.sum()
    current_J = compute_J(a)
    tracker.update(a)
    temp = t0
    
    for r in range(n_restarts):
        if time.monotonic() > deadline:
            break
        scale = max(1e-6, temp / a.max())
        a_pert = a + rng.normal(0, 1, n) * scale
        a_pert = np.maximum(a_pert, 1e-6)
        a_pert = a_pert / a_pert.sum()
        
        # Short LP refinement
        a_refined = run_lp_opt(a_pert, max(5, 15 - r // 8), deadline, tracker, 
                               tighten_start=0.02, tighten_max=0.10)
        
        J = compute_J(a_refined)
        delta = J - current_J
        if delta < 0 or rng.random() < np.exp(-delta / max(temp, 1e-10)):
            a = a_refined
            current_J = J
        temp *= 0.93
    tracker.update(a)
    return a

def coordinate_descent(a, max_steps, deadline, tracker):
    a = a.copy().astype(np.float64)
    a = a / a.sum()
    n = len(a)
    current_J = compute_J(a)
    tracker.update(a)
    rng = np.random.default_rng(42)
    best_j = np.argsort(a)[:max(20, n//10)]
    
    for step in range(max_steps):
        if time.monotonic() > deadline:
            break
        if step % 100 == 0:
            # Re-prioritize: focus on larger elements
            best_j = np.argsort(a)[-max(20, n//8):]
        j = best_j[rng.integers(0, len(best_j))]
        improved = False
        for scale in [0.7, 1.3, 0.5, 1.5, 0.9, 1.1]:
            a_trial = a.copy()
            a_trial[j] *= scale
            a_trial = np.maximum(a_trial, 1e-8)
            ss = a_trial.sum()
            if ss < 1e-10:
                continue
            a_trial = a_trial / ss
            J_trial = compute_J(a_trial)
            if J_trial < current_J - 1e-14:
                a = a_trial
                current_J = J_trial
                improved = True
                break
        if not improved and step % 500 == 0:
            # Try random coordinate
            j = rng.integers(0, n)
            for scale in [0.8, 1.2]:
                a_trial = a.copy()
                a_trial[j] *= scale
                a_trial = np.maximum(a_trial, 1e-8)
                a_trial = a_trial / a_trial.sum()
                J_trial = compute_J(a_trial)
                if J_trial < current_J - 1e-14:

                    current_J = J_trial
