```python
    """Project v onto the probability simplex {x : sum(x)=1, x>=0}."""
```

```python
        a = a_new
```

```python
def solve_qp_fista(w, n, max_iter=5000):
    """Solve min a^T M a s.t. sum(a)=1, a>=0 using FISTA."""
    # Estimate Lipschitz constant L of grad(f) = 2*M*a
    # L = 2 * lambda_max(M)
    # Use a few power iterations
    v = np.ones(n) / n
    for _ in range(10):
        v = hankel_matvec(w, v)
        v = v / np.linalg.norm(v)
    L = 2.0 * float(np.dot(hankel_matvec(w, v), v) / np.dot(v, v)) + 1e-10
    
    a = np.ones(n) / n
    y = a.copy()
    t = 1.0
    for it in range(max_iter):
        grad = 2.0 * hankel_matvec(w, y)
        a_new = project_simplex(y - grad / L)
        t_new = (1 + np.sqrt(1 + 4*t*t)) / 2
        y = a_new + ((t - 1) / t_new) * (a_new - a)
        a = a_new
        t = t_new
    return a
```

```python
"""
Hybrid QP-FISTA + L-BFGS-B + LP alternating optimization for minimizing
2*n*max(conv(a,a))/(sum(a))^2. Key new idea: solve the non-smooth max problem
via homotopy continuation using FISTA on the simplex to solve exact QP subproblems
(min a^T M_w a on simplex, where M_w is a Hankel matrix from soft-max weights).
Combined with structured initializations, subgradient, L-BFGS-B, and LP polishing.
"""

import numpy as np
from scipy.optimize import minimize, linprog
from scipy.fft import rfft, irfft
import time


def compute_conv(a):
    """Compute linear convolution a*a using FFT. Returns array of length 2n-1."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    conv = irfft(fa * fa, size)
    return conv[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    conv = compute_conv(a)
    max_b = float(np.max(conv))
    sum_a = float(np.sum(a))
    if sum_a < 0.01:
        return np.inf
    return 2.0 * n * max_b / (sum_a * sum_a)


def fft_convolve(a, b):
    """Linear convolution of a and b using FFT."""
    n, m = len(a), len(b)
    size = 1
    while size < n + m - 1:
        size <<= 1
    fa = rfft(a, size)
    fb = rfft(b, size)
    return irfft(fa * fb, size)[:n + m - 1]


def hankel_matvec(w, a):
    """Compute (M a)[i] = sum_j w[i+j] * a[j] where M[i,j] = w[i+j].
    Uses FFT for O(n log n) complexity."""
    n = len(a)
    a_flip = a[::-1]
    conv_aw = fft_convolve(a_flip, w)
    return conv_aw[n - 1:2 * n - 1]


def project_simplex(v):
    """Euclidean projection onto the probability simplex {x: sum(x)=1, x>=0}."""
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not np.any(cond):
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0)


def estimate_lipschitz(w, a_init, iters=15):
    """Estimate Lipschitz constant L = 2*lambda_max(M) using power iteration."""
    n = len(a_init)
    v = a_init.copy()
    v = v - np.mean(v)
    nv = np.linalg.norm(v)
    if nv < 1e-15:
        v = np.random.default_rng(0).random(n)
        nv = np.linalg.norm(v)
    v = v / nv
    for _ in range(iters):
        v_new = hankel_matvec(w, v)
        v_new = v_new - np.mean(v_new)
        nv_new = np.linalg.norm(v_new)
        if nv_new < 1e-15:
            v = np.random.default_rng(1).random(n)
            nv = np.linalg.norm(v)
            v = v / nv
            continue
        v = v_new / nv_new
        nv = nv_new
    # Rayleigh quotient
    Mv = hankel_matvec(w, v)
    rayleigh = float(np.dot(v, Mv) / (np.dot(v, v) + 1e-30))
    L = 2.0 * max(rayleigh, 0.1) + 0.1
    # Also use Gershgorin bound as upper bound
    max_row_sum = float(np.max(np.array([np.sum(w[i:i+n]) for i in range(n)])))
    L = max(L, 2.0 * max_row_sum * 0.5)
    return L + 1e-6


def solve_qp_fista(w, n, max_iter=3000, L=None):
    """Solve min a^T M_w a s.t. sum(a)=1, a>=0 using FISTA.
    M_w[i,j] = w[i+j]."""
    a0 = np.ones(n) / n
    if L is None:
        L = estimate_lipschitz(w, a0)
    
    a = a0.copy()
    y = a0.copy()
    t = 1.0
    prev_obj = np.inf
    for it in range(max_iter):
        grad = 2.0 * hankel_matvec(w, y)
        a_new = project_simplex(y - grad / L)
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t * t)) / 2.0
        y = a_new + ((t - 1.0) / t_new) * (a_new - a)
        a = a_new
        t = t_new
        if it % 500 == 0 and it > 0:
            conv = compute_conv(a)
            obj = float(np.max(conv))
            if abs(prev_obj - obj) < 1e-12:
                break
            prev_obj = obj
    return a


def solve_qp_slsqp(w, n, a_init=None, maxiter=500):
    """Solve min a^T M_w a s.t. sum(a)=1, a>=0 using SLSQP."""
    if a_init is None:
        a_init = np.ones(n) / n
    
    def obj_and_grad(a):
        Ma = hankel_matvec(w, a)
        val = float(np.dot(a, Ma))
        grad = 2.0 * Ma
        return val, grad
    
    constraints = [{'type': 'eq', 'fun': lambda a: np.sum(a) - 1.0,
                    'jac': lambda a: np.ones(n)}]
    bounds = [(0.0, None)] * n
    
    try:
        res = minimize(obj_and_grad, a_init, jac=True, method='SLSQP',
                       bounds=bounds, constraints=constraints,
                       options={'maxiter': maxiter, 'ftol': 1e-14})
        if res.success:
            a = np.maximum(res.x, 0)
            s = np.sum(a)
            if s > 1e-10:
                return a / s
    except Exception:
        pass
    return None


def qp_fista_optimize(a, tau_list, time_budget=30.0):
    """Apply QP-FISTA with tau annealing. Returns best a found."""
    n = len(a)
    a = a.copy().astype(np.float64)
    s = np.sum(a)
    if s < 1e-10:
        a = np.ones(n) / n
    else:
        a = a / s
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    
    for tau in tau_list:
        if time.monotonic() - start > time_budget - 2:
            break
        
        conv = compute_conv(a)
        cmax = float(np.max(conv))
        shifted = np.clip((conv - cmax) / tau, -300, 300)
        exp_conv = np.exp(shifted)
        w = exp_conv / np.sum(exp_conv)
        
        # Solve QP with FISTA
        a_new = solve_qp_fista(w, n, max_iter=3000)
        s_new = np.sum(a_new)
        if s_new < 1e-10:
            continue
        a_new = a_new / s_new
        obj_new = compute_obj(a_new)
        
        # Line search on blend
        for t_blend in [1.0, 0.8, 0.6, 0.4, 0.2]:
            a_try = (1.0 - t_blend) * best_a + t_blend * a_new
            a_try = np.maximum(a_try, 0)
            s_try = np.sum(a_try)
            if s_try < 1e-10:
                continue
            a_try = a_try / s_try
            obj_try = compute_obj(a_try)
            if obj_try < best_obj:
                best_obj = obj_try
                best_a = a_try.copy()
        
        a = best_a.copy()
    
    return best_a, best_obj


def softmax(z):
    """Numerically stable softmax."""
    z = z - np.max(z)
    e = np.exp(z)
    return e / np.sum(e)


def soft_max_obj_and_grad_z(z, tau):
    """Compute soft-max of conv(softmax(z), softmax(z)) and grad w.r.t. z."""
    n = len(z)
    a = softmax(z)
    conv = compute_conv(a)
    cmax = float(np.max(conv))
    shifted = np.clip((conv - cmax) / tau, -500.0, 500.0)
    exp_conv = np.exp(shifted)
    Z = float(np.sum(exp_conv))
    soft_max_val = cmax + tau * np.log(Z)
    w = exp_conv / Z
    
    w_rev = w[::-1]
    conv_fw = fft_convolve(a, w_rev)
    f1 = conv_fw[n - 1:2 * n - 1][::-1]
    grad_a = 2.0 * f1
    mean_grad = float(np.dot(a, grad_a))
    grad_z = a * (grad_a - mean_grad)
    
    return soft_max_val, grad_z


def optimize_lbfgs(a_init, tau_list, max_iter=300, time_budget=30.0):
    """L-BFGS-B with soft-max parametrization and tau annealing."""
    n = len(a_init)
    a = a_init.copy()
    s = np.sum(a)
    if s < 1e-10:
        a = np.ones(n) / n
    else:
        a = a / s
    a = np.maximum(a, 1e-30)
    z = np.log(a)
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    
    for tau in tau_list:
        if time.monotonic() - start > time_budget:
            break
        
        def func_and_grad(z_val):
            return soft_max_obj_and_grad_z(z_val, tau)
        
        try:
            result = minimize(func_and_grad, z, jac=True, method='L-BFGS-B',
                              options={'maxiter': max_iter, 'ftol': 1e-14, 'gtol': 1e-8})
            z_new = result.x
        except Exception:
            continue
        
        a_new = softmax(z_new)
        a_new = np.maximum(a_new, 0)
        s_new = np.sum(a_new)
        if s_new < 1e-10:
            continue
        a_new = a_new / s_new
        obj = compute_obj(a_new)
        
        if obj < best_obj:
            best_obj = obj
            best_a = a_new.copy()
            z = z_new
    
    return best_a, best_obj


def build_toeplitz(a):
    """Build Toeplitz matrix where (A@g)[k] = (conv(a,g))[k]."""
    n = len(a)
    A = np.zeros((2 * n - 1, n), dtype=np.float64)
    for j in range(n):
        k_end = min(j + n, 2 * n - 1)
        A[j:k_end, j] = a[:k_end - j]
    return A


def solve_lp_g(a, M, time_limit=20):
    """Solve LP: max sum(g) s.t. (a*g)[k] <= M for all k, g >= 0."""
    n = len(a)
    A = build_toeplitz(a)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': time_limit})
        if res.status == 0:
            g = np.maximum(res.x, 0)
            s = float(np.sum(g))
            if s > 1e-10:
                return g / s, s
    except Exception:
        pass
    return None, None


def alternating_lp(a, max_iters=30, time_budget=40.0):
    """Alternating LP with line search."""
    n = len(a)
    a = a.copy().astype(np.float64)
    s = np.sum(a)
    if s < 1e-10:
        return a
    a = a / s
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    stall_count = 0
    
    for it in range(max_iters):
        if time.monotonic() - start > time_budget:
            break
        
        conv = compute_conv(a)
        M = float(np.max(conv))
        
        g, _ = solve_lp_g(a, M, time_limit=15)
        if g is None:
            break
        
        best_t_obj = best_obj
        best_a_new = best_a.copy()
        for t in [0.1, 0.2, 0.3, 0.5, 0.7, 1.0]:
            a_try = (1.0 - t) * a + t * g
            a_try = np.maximum(a_try, 0)
            s_try = np.sum(a_try)
            if s_try < 1e-10:
                continue
            a_try = a_try / s_try
            obj_try = compute_obj(a_try)
            if obj_try < best_t_obj:
                best_t_obj = obj_try
                best_a_new = a_try.copy()
        
        if best_t_obj < best_obj - 1e-10:
            best_obj = best_t_obj
            best_a = best_a_new.copy()
            a = best_a_new
            stall_count = 0
        else:
            stall_count += 1
            if stall_count > 15:
                rng = np.random.default_rng(it * 7919 + 12345)
                a = a * np.exp(rng.normal(0, 0.1, n))
                a = np.maximum(a, 0)
                s = np.sum(a)
                if s > 1e-10:
                    a = a / s
                stall_count = 0
            else:
                a = 0.9 * a + 0.1 * g
                a = np.maximum(a, 0)
                s = np.sum(a)
                if s > 1e-10:
                    a = a / s
    
    return best_a


def subgradient_optimize(a, max_iters=2000, time_budget=15.0, tau_start=5.0, tau_end=50.0):
    """Smoothed subgradient optimization."""
    n = len(a)
    a = a.copy().astype(np.float64)
    s = np.sum(a)
    if s < 1e-10:
        a = np.ones(n) / n
    else:
        a = a / s
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    step = 0.1
    stall = 0
    
    for it in range(max_iters):
        if time.monotonic() - start > time_budget:
            break
        
        frac = it / max(1, max_iters)
        tau = tau_start + (tau_end - tau_start) * frac
        
        conv = compute_conv(a)
        max_conv = float(np.max(conv))
        shifted = np.clip(tau * (conv - max_conv), -300, 300)
        weights = np.exp(shifted)
        w_sum = np.sum(weights)
        if w_sum < 1e-300:
            idx = int(np.argmax(conv))
            weights = np.zeros(len(conv))
            weights[idx] = 1.0
        else:
            weights /= w_sum
        
        w_rev = weights[::-1]
        conv_fw = fft_convolve(a, w_rev)
        grad_a = 2.0 * conv_fw[n - 1:2 * n - 1][::-1]
        
        grad_centered = grad_a - np.mean(grad_a)
        a_new = a - step * grad_centered
        a_new = np.maximum(a_new, 0)
        s_new = np.sum(a_new)
        if s_new > 1e-10:
            a_new = a_new / s_new
        else:
            a_new = a.copy()
        
        a = a_new
        obj = compute_obj(a)
        
        if obj < best_obj - 1e-12:
            best_obj = obj
            best_a = a.copy()
            stall = 0
        else:
            stall += 1
            if stall > 80:
                step *= 0.85
                if step < 1e-10:
                    break
                rng = np.random.default_rng(it * 31 + 7)
                a = a + rng.normal(0, 0.01, n) * step
                a = np.maximum(a, 0)
                s = np.sum(a)
                if s > 1e-10:
                    a = a / s
                stall = 0
        
        if it > 50:
            step *= 0.997
    
    return best_a


def structured_init(n, idx):
    """Generate structured initializations with good autocorrelation properties."""
    k = np.arange(n, dtype=np.float64)
    k_norm = (k - (n - 1) / 2.0) / ((n - 1) / 2.0)
    rng = np.random.default_rng(idx * 104729 + 7)
    
    if idx % 7 == 0:
        # Hann window
        a = 0.5 * (1 + np.cos(np.pi * k_norm))
    elif idx % 7 == 1:
        # Flat-top window (designed for minimal sidelobes)
        a1 = 1.0
        a2 = 1.93 * rng.uniform(0.8, 1.2)
        a3 = 1.29 * rng.uniform(0.8, 1.2)
        a4 = 0.388
        a5 = 0.028
        a = (a1 - a2 * np.cos(2 * np.pi * k_norm / 2) + a3 * np.cos(4 * np.pi * k_norm / 2)
             - a4 * np.cos(6 * np.pi * k_norm / 2) + a5 * np.cos(8 * np.pi * k_norm / 2))
    elif idx % 7 == 2:
        # Raised cosine with adjustable parameter
        beta = rng.uniform(1.0, 3.0)
        a = np.exp(-beta * k_norm ** 2)
    elif idx % 7 == 3:
        # Sinc-like (prolate spheroidal approximation)
        W = rng.uniform(0.3, 0.7)
        a = np.sinc(W * k_norm)  # sin(pi*W*x)/(pi*W*x)
        a = np.abs(a)
    elif idx % 7 == 4:
        # Cosine with random frequency
        f = rng.uniform(0.5, 2.5)
        phase = rng.uniform(0, np.pi)
        a = 1.0 + 0.5 * np.cos(2 * np.pi * f * (k / (n - 1) - 0.5) + phase)
        a = a * (0.5 * (1 + np.cos(np.pi * k_norm)))  # taper
    elif idx % 7 == 5:
        # Two-lobe structure
        f1 = rng.uniform(0.5, 1.5)
        f2 = rng.uniform(2.0, 5.0)
        a = 1.0 + 0.3 * np.cos(2 * np.pi * f1 * k_norm) + 0.2 * np.cos(2 * np.pi * f2 * k_norm)
        a = a * (0.5 * (1 + np.cos(np.pi * k_norm)))
    else:
        # Gaussian with random width
        sigma = rng.uniform(0.2, 0.8)
        a = np.exp(-0.5 * (k_norm / sigma) ** 2)
    
    a = np.maximum(a, 0.001)
    s = np.sum(a)
    return a / s


def random_init(n, seed):
    """Generate diverse random initializations."""
    rng = np.random.default_rng(seed)
    k = np.arange(n, dtype=np.float64)
    k_norm = (k - (n - 1) / 2.0) / ((n - 1) / 2.0)
    choice = rng.integers(0, 10)
    
    if choice == 0:
        f = rng.uniform(0.3, 3.0)
        phase = rng.uniform(0, 2 * np.pi)
        a = 1.0 + 0.7 * np.cos(2 * np.pi * f * k / n + phase)
        a = a * (0.5 * (1 + np.cos(np.pi * k_norm)))
    elif choice == 1:
        a = 1.0 + 0.4 * np.cos(2 * np.pi * rng.uniform(0.3, 2.0) * k / n) + \
            0.3 * np.cos(2 * np.pi * rng.uniform(2.0, 6.0) * k / n)
        a = a * (0.5 * (1 + np.cos(np.pi * k_norm)))
    elif choice == 2:
        a = np.exp(-0.5 * (k_norm / rng.uniform(0.3, 0.7)) ** 2)
    elif choice == 3:
        a = rng.random(n)
        win = max(3, n // 15)
        kernel = np.ones(win) / win
        a = np.convolve(a, kernel, mode='same')
    elif choice == 4:
        a = np.ones(n)
        for _ in range(rng.integers(1, 4)):
            f = rng.uniform(0.5, 5.0)
            amp = rng.uniform(0.1, 0.4)
            a += amp * np.cos(2 * np.pi * f * k / n + rng.uniform(0, 2 * np.pi))
        a = a * (0.5 * (1 + np.cos(np.pi * k_norm)))
    elif choice == 5:
        f = rng.uniform(0.3, 2.0)
        a = 1.0 + 0.5 * np.sin(2 * np.pi * f * k / n)
        a = a * (0.5 * (1 + np.cos(np.pi * k_norm)))
    elif choice == 6:
        nblocks = rng.integers(2, 6)
        a = np.ones(n)
        for b in range(nblocks):
            start = b * n // nblocks
            end = (b + 1) * n // nblocks
            a[start:end] = rng.uniform(0.3, 1.5)
    elif choice == 7:
        # Chirp
        f0 = rng.uniform(0.5, 2.0)
        f1 = rng.uniform(2.0, 6.0)
        a = 1.0 + 0.5 * np.cos(2 * np.pi * (f0 * k / n + (f1 - f0) * k**2 / (2 * n**2)))
        a = a * (0.5 * (1 + np.cos(np.pi * k_norm)))
    elif choice == 8:
        a = np.sinc(rng.uniform(0.3, 0.7) * k_norm)
        a = np.abs(a)
    else:
        a = rng.random(n) + 0.1
        a = a * (0.5 * (1 + np.cos(np.pi * k_norm)))
    
    a = np.maximum(a, 0.001)
    s = np.sum(a)
    return a / s


def resample_sequence(a, new_n):
    """Resample sequence to new length via interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    return np.interp(x_new, x_old, a)


def try_parent():
    """Try to load parent sequence."""
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64)
        parent = np.clip(parent, 0, 1000.0)
        if parent.size > 0 and np.isfinite(parent).all():
            obj = compute_obj(parent)
            print(f"Parent (n={parent.size}): obj={obj:.6f}")
            return parent.copy(), obj
    except Exception:
        pass
    return None, np.inf


def polish_sequence(a, time_budget, verbose=False):
    """Intensive polishing combining all methods."""
    n = len(a)
    a = a.copy().astype(np.float64)
    start = time.monotonic()
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    round_num = 0
    
    while time.monotonic() - start < time_budget - 3:
        round_num += 1
        remaining = time_budget - (time.monotonic() - start)
        if remaining < 5:
            break
        
        # QP-FISTA (primary new method)
        if remaining > 15:
            tau_list = [1.0 / n, 0.5 / n, 0.2 / n, 0.1 / n, 0.05 / n]
            a_new, obj_new = qp_fista_optimize(best_a, tau_list,
                                               time_budget=min(remaining * 0.3, 20.0))
            if obj_new < best_obj:
                best_obj = obj_new
                best_a = a_new.copy()
                if verbose:
                    print(f"    Polish r{round_num} QP-FISTA: {best_obj:.6f}")
        
        # L-BFGS-B
        remaining = time_budget - (time.monotonic() - start)
        if remaining > 12:
            tau_lb = [0.3 / n, 0.1 / n, 0.05 / n]
            a_new, obj_new = optimize_lbfgs(best_a, tau_lb,
                                            max_iter=150, time_budget=min(remaining * 0.2, 15.0))
            if obj_new < best_obj:
                best_obj = obj_new
                best_a = a_new.copy()
                if verbose:
                    print(f"    Polish r{round_num} L-BFGS: {best_obj:.6f}")
        
        # Subgradient
        remaining = time_budget - (time.monotonic() - start)
        if remaining > 10:
            a_new = subgradient_optimize(best_a, max_iters=int(remaining * 8),
                                         time_budget=min(remaining * 0.15, 12.0),
                                         tau_start=10.0, tau_end=40.0)
            obj = compute_obj(a_new)
            if obj < best_obj:
                best_obj = obj
                best_a = a_new.copy()
                if verbose:
                    print(f"    Polish r{round_num} SG: {best_obj:.6f}")
        
        # LP
        remaining = time_budget - (time.monotonic() - start)
        if n <= 800 and remaining > 20:
            a_new = alternating_lp(best_a, max_iters=20,
                                   time_budget=min(remaining * 0.3, 25.0))
            obj = compute_obj(a_new)
            if obj < best_obj:
                best_obj = obj
                best_a = a_new.copy()
                if verbose:
                    print(f"    Polish r{round_num} LP: {best_obj:.6f}")
        
        # SLSQP on QP (final polish)
        remaining = time_budget - (time.monotonic() - start)
        if remaining > 15:
            conv = compute_conv(best_a)
            cmax = float(np.max(conv))
            tau_final = 0.05 / n
            shifted = np.clip((conv - cmax) / tau_final, -300, 300)
            exp_conv = np.exp(shifted)
            w = exp_conv / np.sum(exp_conv)
            a_slq = solve_qp_slsqp(w, n, a_init=best_a, maxiter=200)
            if a_slq is not None:
                obj_slq = compute_obj(a_slq)
                if obj_slq < best_obj:
                    best_obj = obj_slq
                    best_a = a_slq.copy()
                    if verbose:
                        print(f"    Polish r{round_num} SLSQP: {best_obj:.6f}")
    
    return best_a, best_obj


def propose_candidate(seed=42, budget_s=1500):
    """
    Main search: QP-FISTA + L-BFGS-B + LP alternating + subgradient.
    Uses homotopy continuation with soft-max weights for the QP subproblems.
    """
    start_time = time.monotonic()
    deadline = start_time + float(budget_s) - 20.0
    
    best_seq = None
    best_score = np.inf
    
    # Load parent
    parent, parent_obj = try_parent()
    if parent is not None and parent_obj < best_score:
        best_score = parent_obj
        best_seq = parent.copy()
    
    # N values to explore
    n_values = [400, 500, 600, 700, 800, 900, 1000]
    if best_seq is not None:
        np_ = len(best_seq)
        extra = [np_, max(100, np_ - 100), min(1200, np_ + 100), min(1200, np_ + 200)]
        n_values = sorted(set(n_values + extra))
    
    print(f"Exploring n values: {n_values}")
    print(f"Initial best: {best_score:.6f}")
    
    candidates = []
    
    # Phase 1: Exploration
    for n in n_values:
        if time.monotonic() > deadline - 120:
            break
        remaining = deadline - time.monotonic()
        if remaining < 10:
            break
        
        # Time allocation
        if n <= 500:
            alloc = min(40.0, remaining * 0.1)
        elif n <= 800:
            alloc = min(55.0, remaining * 0.12)
        else:
            alloc = min(65.0, remaining * 0.13)
        alloc = max(15.0, alloc)
        
        print(f"\nN={n}: {remaining:.0f}s remaining, alloc={alloc:.0f}s")
        
        n_inits = max(2, int(alloc / 18))
        n_inits = min(n_inits, 5)
        
        for init_idx in range(n_inits):
            if time.monotonic() > deadline - 120:
                break
            remaining_init = deadline - time.monotonic()
            if remaining_init < 12:
                break
            
            # Choose initialization
            if init_idx == 0 and best_seq is not None:
                a = resample_sequence(best_seq, n)
            elif init_idx == 1:
                a = structured_init(n, seed + n)
            elif init_idx == 2:
                a = structured_init(n, seed + n + 1000)
            else:
                a = random_init(n, seed + n * 104729 + init_idx * 79193)
            
            # Quick subgradient shaping
            sg_time = min(5.0, (alloc / n_inits) * 0.15)
            a = subgradient_optimize(a, max_iters=int(sg_time * 30),
                                      time_budget=sg_time,
                                      tau_start=3.0, tau_end=20.0)
            
            # QP-FISTA (primary method)
            remaining_init = deadline - time.monotonic()
            if remaining_init > 15:
                tau_list = [3.0 / n, 1.0 / n, 0.5 / n, 0.2 / n, 0.1 / n, 0.05 / n]
                a, obj = qp_fista_optimize(a, tau_list,
                                           time_budget=min((alloc / n_inits) * 0.35, 20.0))
            else:
                obj = compute_obj(a)
            
            # L-BFGS-B
            remaining_init = deadline - time.monotonic()
            if remaining_init > 12:
                tau_lb = [1.0 / n, 0.5 / n, 0.2 / n, 0.1 / n, 0.05 / n]
                a, obj = optimize_lbfgs(a, tau_lb,
                                        max_iter=200, time_budget=min((alloc / n_inits) * 0.3, 18.0))
            
            # Subgradient polish
            remaining_init = deadline - time.monotonic()
            if remaining_init > 8:
                a = subgradient_optimize(a, max_iters=int(remaining_init * 10),
                                          time_budget=min(remaining_init * 0.1, 8.0),
                                          tau_start=8.0, tau_end=30.0)
                obj = compute_obj(a)
            
            # LP (for smaller n)
            remaining_init = deadline - time.monotonic()
            if remaining_init > 15 and n <= 800:
                a = alternating_lp(a, max_iters=30,
                                   time_budget=min(remaining_init * 0.3, 20.0))
                obj = compute_obj(a)
            
            print(f"  init{init_idx}: n={n}, obj={obj:.6f}")
            
            if obj < best_score:
                best_score = obj
                best_seq = a.copy()
                print(f"  ** UPDATE: n={n}, obj={best_score:.6f} **")
            
            candidates.append((a.copy(), obj, n))
    
    # Phase 2: Polishing top candidates
    print(f"\n=== POLISHING PHASE ===")
    print(f"Best before polish: {best_score:.6f}")
    
    candidates.sort(key=lambda x: x[1])
    n_polish = min(4, len(candidates))
    
    for ci in range(n_polish):
        if time.monotonic() > deadline - 50:
            break
        remaining = deadline - time.monotonic()
        if remaining < 20:
            break
        
        a_cand = candidates[ci][0]
        n_cand = candidates[ci][2]
        print(f"  Polishing candidate {ci} (n={n_cand}, obj={candidates[ci][1]:.6f}), {remaining:.0f}s left")
        
        polish_budget = min(remaining * 0.35, 100.0)
        a_pol, obj_pol = polish_sequence(a_cand, time_budget=polish_budget, verbose=True)
        
        if obj_pol < best_score:
            best_score = obj_pol
            best_seq = a_pol.copy()
            print(f"  ** UPDATE after polish: obj={best_score:.6f} **")
        else:
            # Perturbation attempts
            remaining = deadline - time.monotonic()
            if remaining > 15:
                rng = np.random.default_rng(ci * 555 + 999)
                for ptry in range(3):
                    if time.monotonic() > deadline - 20:
                        break
                    scale = rng.uniform(0.02, 0.1)
                    a_pert = a_cand * np.exp(rng.normal(0, scale, n_cand))
                    a_pert = np.maximum(a_pert, 0)
                    s = np.sum(a_pert)
                    if s < 1e-10:
                        continue
                    a_pert = a_pert / s
                    # Quick QP-FISTA on perturbed
                    tau_pert = [1.0 / n_cand, 0.3 / n_cand, 0.1 / n_cand]
                    a_pert, obj_pert = qp_fista_optimize(a_pert, tau_pert, time_budget=6.0)
                    if obj_pert < best_score:
                        best_score = obj_pert
                        best_seq = a_pert.copy()
                        print(f"  ** UPDATE (perturb {ptry}): obj={best_score:.6f} **")
    
    # Phase 3: Final intensive polish on global best
    if best_seq is not None and time.monotonic() < deadline - 15:
        remaining = deadline - time.monotonic()
        n_best = len(best_seq)
        print(f"\n=== FINAL POLISH: n={n_best}, {remaining:.0f}s ===")
        
        a_cur = best_seq.copy()
        for final_round in range(5):
            if time.monotonic() > deadline - 8:
                break
            remaining = deadline - time.monotonic()
            if remaining < 12:
                break
            
            # QP-FISTA with very small tau
            tau_vsmall = [0.2 / n_best, 0.1 / n_best, 0.05 / n_best, 0.02 / n_best]
            a_new, obj_new = qp_fista_optimize(a_cur, tau_vsmall,
                                               time_budget=min(remaining * 0.3, 20.0))
            if obj_new < best_score:
                best_score = obj_new
                best_seq = a_new.copy()
                a_cur = a_new.copy()
                print(f"  Final QP {final_round}: {best_score:.6f}")
            
            # L-BFGS-B
            remaining = deadline - time.monotonic()
            if remaining > 12:
                a_new, obj_new = optimize_lbfgs(a_cur, [0.1 / n_best, 0.05 / n_best],
                                                max_iter=200, time_budget=min(remaining * 0.2, 12.0))
                if obj_new < best_score:
                    best_score = obj_new
                    best_seq = a_new.copy()
                    a_cur = a_new.copy()
                    print(f"  Final L-BFGS {final_round}: {best_score:.6f}")
            
            # SLSQP QP
            remaining = deadline - time.monotonic()
            if remaining > 15:
                conv = compute_conv(a_cur)
                cmax = float(np.max(conv))
                tau_sf = 0.03 / n_best
                shifted = np.clip((conv - cmax) / tau_sf, -300, 300)
                exp_conv = np.exp(shifted)
                w = exp_conv / np.sum(exp_conv)
                a_slq = solve_qp_slsqp(w, n_best, a_init=a_cur, maxiter=300)
                if a_slq is not None:
                    obj_slq = compute_obj(a_slq)
                    if obj_slq < best_score:
                        best_score = obj_slq
                        best_seq = a_slq.copy()
                        a_cur = a_slq.copy()
                        print(f"  Final SLSQP {final_round}: {best_score:.6f}")
            
            # Subgradient micro-polish
            remaining = deadline - time.monotonic()
            if remaining > 8:
                a_new = subgradient_optimize(a_cur, max_iters=int(remaining * 5),
                                              time_budget=min(remaining * 0.15, 8.0),
                                              tau_start=15.0, tau_end=60.0)
                obj_new = compute_obj(a_new)
                if obj_new < best_score:
                    best_score = obj_new
                    best_seq = a_new.copy()
                    a_cur = a_new.copy()
                    print(f"  Final SG {final_round}: {best_score:.6f}")
    
    # Final validation
    if best_seq is None:
        best_seq = np.ones(200)
        best_score = compute_obj(best_seq)
    
    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    val = evaluate_sequence(result)
    if val < np.inf:
        best_score = val
    else:
        result = [1.0] * 100
        best_score = 2.0
    
    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_score:.6f}, time={elapsed:.1f}s ===")
    
    return result
```
