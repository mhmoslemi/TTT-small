"""
Optimized step-function coefficient search using adaptive active-set LP alternating
method with aggressive M-tightening and multi-scale exploration.
"""

import numpy as np
from scipy.optimize import linprog
from scipy.sparse import coo_matrix
from scipy.interpolate import interp1d
import time


def compute_conv(a):
    """Compute linear convolution a*a via FFT."""
    n = len(a)
    size = 2 * n - 1
    fa = np.fft.rfft(a, size)
    conv = np.fft.irfft(fa * fa, size)
    return conv


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    S = float(np.sum(a))
    if S < 1e-10:
        return np.inf
    conv = compute_conv(a)
    M = float(np.max(conv))
    return 2.0 * n * M / (S * S)


def solve_reduced_lp(a, active_idx, M, time_limit=5):
    """
    Solve LP: max sum(g) s.t. (a*g)[k] <= M for k in active_idx, g >= 0.
    Only uses active constraints for speed.
    """
    n = len(a)
    rows = []
    cols = []
    vals = []
    
    for k in active_idx:
        for i in range(n):
            j = k - i
            if 0 <= j < n:
                rows.append(k)
                cols.append(i)
                vals.append(a[j])
    
    if not rows:
        return None
    
    A = coo_matrix((vals, (rows, cols)), shape=(2 * n - 1, n)).tocsr()
    # Only keep active rows
    active_set = set(active_idx)
    row_mask = np.array([r in active_set for r in range(2 * n - 1)])
    A_active = A[row_mask, :]
    
    c = -np.ones(n)
    b_ub = np.full(len(active_idx), M)
    bounds = [(0.0, None)] * n
    
    try:
        res = linprog(c, A_ub=A_active, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': time_limit,
                               'dual_feasibility_tolerance': 1e-9,
                               'primal_feasibility_tolerance': 1e-9})
        if res.status == 0:
            return np.maximum(res.x, 0)
    except Exception:
        pass
    return None

def momentum_subgradient(a, max_iters, time_budget, topk=8):
    """Momentum subgradient descent with adaptive step size."""
    n = len(a)
    a = a.copy().astype(np.float64)
    S = np.sum(a)
    if S < 1e-10:
        return a
    a = a / S
    
    v = np.zeros(n)
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    base_step = 0.15
    momentum = 0.8
    stall = 0
    
    for it in range(max_iters):
        if time.monotonic() - start > time_budget:
            break
        
        conv = compute_conv(a)
        M = float(np.max(conv))
        
        # Find top-K active positions
        if topk >= len(conv):
            active = np.arange(len(conv))
        else:
            active = np.argsort(conv)[-topk:]
        
        # Gradient of max(conv) w.r.t. a
        # d(conv[k])/da[i] = a[k-i] + a[i+k]... 
        # Actually (a*a)[k] = sum_i a[i]*a[k-i], so d/d a[i] = a[k-i] (if k-i in range)
        # For the max, gradient is average over active set
        grad = np.zeros(n)
        for k in active:
            for i in range(n):
                j = k - i
                if 0 <= j < n:
                    grad[i] += a[j]
        grad /= len(active)
        
        # Center gradient (preserve sum)
        grad_centered = grad - np.mean(grad)
        
        v = momentum * v + (1.0 - momentum) * grad_centered
        step = base_step / (1.0 + 0.005 * it)
        
        a_new = a - step * v
        a_new = np.maximum(a_new, 0)
        S_new = np.sum(a_new)
        if S_new < 1e-10:
            break
        a_new = a_new / S_new
        
        obj = compute_obj(a_new)
        if obj < best_obj - 1e-10:
            best_obj = obj
            best_a = a_new.copy()
            a = a_new
            stall = 0
        else:
            stall += 1
            if stall > 50:
                rng = np.random.default_rng(it * 31 + 7)
                a = a * np.exp(rng.normal(0, 0.06, n))
                a = np.maximum(a, 0)
                a = a / np.sum(a)
                v = np.zeros(n)
                stall = 0
                base_step *= 0.7
                if base_step < 1e-8:

def cosine_init(n, n_modes=10, seed=0):
    """Generate a smooth cosine-series initialization."""
    rng = np.random.default_rng(seed)
    k = np.arange(n)
    a = np.ones(n)
    for m in range(1, n_modes + 1):
        c = rng.uniform(-0.5, 0.5) / m
        phase = rng.uniform(0, 2 * np.pi)
        a += c * np.cos(2 * np.pi * m * k / n + phase)
    a = np.maximum(a, 0)
    S = np.sum(a)
    if S < 1e-10:
        return np.ones(n) / n
    return a / S

"""
Optimized step-function coefficient search using adaptive active-set LP alternating
with fast reduced-constraint SOLVER, multiplicative updates, and momentum subgradient.
Key improvements: top-K active set for much faster LP, M-tightening schedule,
symmetric alternating, and multi-scale perturbation exploration.

import numpy as np
from scipy.optimize import linprog
from scipy.sparse import coo_matrix
from scipy.interpolate import interp1d
import time


def compute_conv(a):
    """Compute linear convolution a*a via FFT."""
    n = len(a)
    size = 2 * n - 1
    fa = np.fft.rfft(a, size)
    conv = np.fft.irfft(fa * fa, size)
    return conv


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    S = float(np.sum(a))
    if S < 1e-10:
        return np.inf
    conv = compute_conv(a)
    M = float(np.max(conv))
    return 2.0 * n * M / (S * S)


def solve_lp_active(a, active_idx, M, time_limit=3):
    """
    Solve LP: max sum(g) s.t. (a*g)[k] <= M for k in active_idx, g >= 0.
    Uses only active constraints for speed.
    """
    n = len(a)
    rows = []
    cols = []
    vals = []

    for k in active_idx:
        for i in range(n):
            j = k - i
            if 0 <= j < n:
                rows.append(k)
                cols.append(i)
                vals.append(a[j])

    if not rows:
        return None

    shape_rows = 2 * n - 1
    A_full = coo_matrix((vals, (rows, cols)), shape=(shape_rows, n)).tocsr()

    # Extract only active rows
    active_arr = np.array(active_idx)
    A_active = A_full[active_arr, :].tocsr()

    c = -np.ones(n)
    b_ub = np.full(len(active_idx), M)
    bounds = [(0.0, None)] * n

    try:
        res = linprog(c, A_ub=A_active, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': time_limit})
        if res.status == 0:
            return np.maximum(res.x, 0)
    except Exception:
        pass
    return None


def multiplicative_update(a, time_budget, n_max_active=50, eta_init=0.3):
    """
    Multiplicative (exponential) update to reduce convolution peaks.
    g_new[i] proportional to g[i] * exp(-eta * penalty_i)
    where penalty_i measures how much position i contributes to the peak.
    """
    n = len(a)
    a = a.copy().astype(np.float64)
    S = np.sum(a)
    if S < 1e-10:
        return a
    a = a / S

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    eta = eta_init

    for it in range(500):
        if time.monotonic() - start > time_budget:
            break

        conv = compute_conv(a)
        M = float(np.max(conv))

        # Active set: top positions
        n_active = min(n_max_active, len(conv))
        active = np.argsort(conv)[-n_active:]

        # Compute penalty for each position i
        # penalty[i] = sum over active k of (a[i]*a[k-i]) * weight
        penalty = np.zeros(n)
        for k in active:
            weight = conv[k] / M  # Normalize
            for i in range(n):
                j = k - i
                if 0 <= j < n:
                    penalty[i] += weight * a[j]

        # Multiplicative update
        a_new = a * np.exp(-eta * penalty / max(1e-10, np.max(penalty)))
        a_new = np.maximum(a_new, 1e-15)
        S_new = np.sum(a_new)
        a_new = a_new / S_new

        obj = compute_obj(a_new)
        if obj < best_obj:
            best_obj = obj
            best_a = a_new.copy()
            a = a_new
        else:
            eta *= 0.85
            if eta < 1e-6:
                break

    start = time.monotonic()
    base_step = 0.12
    momentum = 0.82
    stall = 0

    for it in range(2000):
        if time.monotonic() - start > time_budget:
            break

        conv = compute_conv(a)
        M = float(np.max(conv))

        # Top-K active
        active = np.argsort(conv)[-topk:]

        # Gradient: d(max conv)/da[m] = (1/|A|) * sum_{k in A} 2*a[k-m] (if valid)
        grad = np.zeros(n)
        for k in active:
            for m in range(n):
                j = k - m
                if 0 <= j < n:
                    grad[m] += 2.0 * a[j]
        grad /= len(active)

        # Center (preserve sum)
        grad_c = grad - np.mean(grad)

        v = momentum * v + (1.0 - momentum) * grad_c
        step = base_step / (1.0 + 0.003 * it)

        a_new = a - step * v
        a_new = np.maximum(a_new, 0)
        S_new = np.sum(a_new)
        if S_new < 1e-10:
            break
        a_new = a_new / S_new

        obj = compute_obj(a_new)
        if obj < best_obj - 1e-12:
            best_obj = obj
            best_a = a_new.copy()
            a = a_new
            stall = 0
        else:
            stall += 1
            if stall > 80:
                rng = np.random.default_rng(it * 31 + 7)
                a = a * np.exp(rng.normal(0, 0.05, n))
                a = np.maximum(a, 0)
                a = a / np.sum(a)
                v = np.zeros(n)
                stall = 0
                base_step *= 0.8
                if base_step < 1e-9:
                    break

    return best_a


def cosine_init(n, n_modes=10, seed=0):
    """Cosine-series initialization."""
    rng = np.random.default_rng(seed)
    k = np.arange(n)
    a = np.ones(n)
    for m in range(1, n_modes + 1):
        c = rng.uniform(-0.4, 0.4) / m
        phase = rng.uniform(0, 2 * np.pi)
        a += c * np.cos(2 * np.pi * m * k / n + phase)
    a = np.maximum(a, 0)
    S = np.sum(a)
    if S < 1e-10:
        return np.ones(n) / n
    return a / S


def try_parent():
    """Load parent sequence."""
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64)
        parent = np.clip(parent, 0, 1000.0)
        if parent.size > 0 and np.isfinite(parent).all():
            obj = compute_obj(parent)
            print(f"Parent (n={parent.size}): obj={obj:.6f}")
            return parent.copy(), obj
    except Exception:
        pass
    return None, np.inf


def refine_combined(a, time_budget, use_lp=True):
    """Combined refinement: LP + multiplicative + subgradient in cycles."""
    n = len(a)
    a = a.copy().astype(np.float64)
    S = np.sum(a)
    if S < 1e-10:
        return a, np.inf
    a = a / S

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    cycle = 0

    while time.monotonic() - start < time_budget - 3:
        remaining = time_budget - (time.monotonic() - start)
        if remaining < 8:
            break
        cycle += 1

        # LP phase (fast, top-K)
        if use_lp and remaining > 15 and n <= 700:
            lp_t = min(remaining * 0.35, 35.0)
            a_lp = fast_lp_alternation(best_a, time_budget=lp_t, top_k=35, verbose=False)
            obj_lp = compute_obj(a_lp)
            if obj_lp < best_obj:
                best_obj = obj_lp
                best_a = a_lp.copy()
                if cycle % 3 == 0:
                    print(f"      cycle {cycle} LP: {best_obj:.6f}")

        # Multiplicative update
        remaining = time_budget - (time.monotonic() - start)
        if remaining > 10:
            mu_t = min(remaining * 0.2, 12.0)
            a_mu = multiplicative_update(best_a, time_budget=mu_t, n_max_active=40)
            obj_mu = compute_obj(a_mu)
            if obj_mu < best_obj:
                best_obj = obj_mu
                best_a = a_mu.copy()
                if cycle % 3 == 0:
                    print(f"      cycle {cycle} MU: {best_obj:.6f}")

        # Subgradient
        remaining = time_budget - (time.monotonic() - start)
        if remaining > 8:
            sg_t = min(remaining * 0.25, 15.0)
            a_sg = momentum_subgradient(best_a, time_budget=sg_t, topk=5)
            obj_sg = compute_obj(a_sg)
            if obj_sg < best_obj:
                best_obj = obj_sg
                best_a = a_sg.copy()
                if cycle % 3 == 0:
                    print(f"      cycle {cycle} SG: {best_obj:.6f}")

        # Perturbation escape
        remaining = time_budget - (time.monotonic() - start)
        if remaining > 12:
            rng = np.random.default_rng(cycle * 991 + 17)
            a_p = best_a * np.exp(rng.normal(0, 0.04, n))
            a_p = np.maximum(a_p, 0)
            S_p = np.sum(a_p)
            if S_p > 1e-10:
                a_p = a_p / S_p
                # Quick subgradient on perturbed
                a_p = momentum_subgradient(a_p, time_budget=min(4.0, remaining * 0.15), topk=5)
                obj_p = compute_obj(a_p)
                if obj_p < best_obj:
                    best_obj = obj_p
                    best_a = a_p.copy()
                    print(f"      cycle {cycle} pert: {best_obj:.6f}")

    return best_a, best_obj


def propose_candidate(seed=42, budget_s=1500):
    """
    Main search: fast LP alternation (top-K active set) + multiplicative updates
    + momentum subgradient, with multi-n exploration and perturbation escapes.
    """
    start_time = time.monotonic()
    deadline = start_time + float(budget_s) - 15.0

    best_seq = None
    best_score = np.inf

    parent, parent_obj = try_parent()
    if parent is not None:
        best_score = parent_obj
        best_seq = parent.copy() / np.sum(parent)
        n_parent = len(parent)
    else:
        n_parent = 500
        best_seq = np.ones(n_parent) / n_parent
        best_score = compute_obj(best_seq)

    print(f"Starting: n={n_parent}, score={best_score:.6f}")

    # === Phase 1: Multi-n quick exploration ===
    print("\n=== Phase 1: Multi-n exploration ===")
    n_values = sorted(set([
        max(150, n_parent - 150),
        max(100, n_parent - 75),
        n_parent,
        n_parent + 75,
        n_parent + 150,
        min(1200, n_parent + 300)
    ]))
    n_values = [n for n in n_values if 100 <= n <= 1500]

    candidates = []
    for n in n_values:
        if time.monotonic() > deadline - 120:
            break
        remaining = deadline - time.monotonic()
        if remaining < 15:
            break

        # Interpolate to new length
        x_old = np.linspace(0, 1, len(best_seq))
        x_new = np.linspace(0, 1, n)
        try:
            f_interp = interp1d(x_old, best_seq, kind='cubic',
                                fill_value=(best_seq[0], best_seq[-1]),
                                bounds_error=False)
            a_init = np.maximum(f_interp(x_new), 0)
        except Exception:
            a_init = cosine_init(n, n_modes=8, seed=seed + n)

        S_i = np.sum(a_init)
        if S_i < 1e-10:
            a_init = np.ones(n) / n
        else:
            a_init = a_init / S_i

        # Quick improvement
        a_quick = momentum_subgradient(a_init, time_budget=3.0, topk=5)
        obj_quick = compute_obj(a_quick)
        candidates.append((a_quick, obj_quick))

        if obj_quick < best_score:
            best_score = obj_quick
            best_seq = a_quick.copy()
            print(f"  n={n}: {obj_quick:.6f} ** NEW BEST **")
        else:
            print(f"  n={n}: {obj_quick:.6f}")

    candidates.sort(key=lambda x: x[1])

    # === Phase 2: Main refinement on top candidates ===
    print("\n=== Phase 2: Main refinement ===")
    n_refine = min(3, len(candidates))
    for idx in range(n_refine):
        if time.monotonic() > deadline - 80:
            break
        remaining = deadline - time.monotonic()
        if remaining < 40:
            break

        a_cand, obj_cand = candidates[idx]
        if obj_cand > best_score + 0.02:
            break

        n_cand = len(a_cand)
        alloc = min(remaining * 0.3, 70.0)
        print(f"  Refining candidate {idx} (n={n_cand}, obj={obj_cand:.6f}), alloc={alloc:.0f}s")

        a_ref, obj_ref = refine_combined(a_cand, time_budget=alloc, use_lp=True)
        if obj_ref < best_score:
            best_score = obj_ref
            best_seq = a_ref.copy()
            print(f"    ** NEW BEST: n={n_cand}, obj={best_score:.6f} **")

    # === Phase 3: Intensive iterative refinement ===
    print("\n=== Phase 3: Intensive refinement ===")
    a_cur = best_seq.copy()
    n_best = len(a_cur)

    for rnd in range(12):
        if time.monotonic() > deadline - 50:
            break
        remaining = deadline - time.monotonic()
        if remaining < 30:
            break

        alloc = min(remaining * 0.25, 50.0)
        a_new, obj_new = refine_combined(a_cur, time_budget=alloc, use_lp=True)

        if obj_new < best_score:
            best_score = obj_new
            best_seq = a_new.copy()
            a_cur = a_new.copy()
            print(f"  R{rnd}: {best_score:.6f} ** NEW BEST **")
        else:
            print(f"  R{rnd}: {obj_new:.6f} (best={best_score:.6f})")

            # Try perturbation
            rng = np.random.default_rng(rnd * 2749 + 55)
            for p_scale in [0.08, 0.15, 0.25]:
                if time.monotonic() > deadline - 30:
                    break
                a_p = a_cur * np.exp(rng.normal(0, p_scale, n_best))
                a_p = np.maximum(a_p, 0)
                S_p = np.sum(a_p)
                if S_p < 1e-10:
                    continue
                a_p = a_p / S_p
                a_p, obj_p = refine_combined(a_p, time_budget=min(15.0, remaining * 0.12),
                                             use_lp=True)
                if obj_p < best_score:
                    best_score = obj_p
                    best_seq = a_p.copy()
                    a_cur = a_p.copy()
                    print(f"  R{rnd} pert({p_scale}): {best_score:.6f} ** NEW BEST **")
                    break

    # === Phase 4: Final polish ===
    print("\n=== Phase 4: Final polish ===")
    a_cur = best_seq.copy()
    remaining = deadline - time.monotonic()

    if remaining > 10:
        # Long subgradient with small step
        a_fp = momentum_subgradient(a_cur, time_budget=remaining * 0.4, topk=4)
        obj_fp = compute_obj(a_fp)
        if obj_fp < best_score:
            best_score = obj_fp
            best_seq = a_fp.copy()
            print(f"  Final SG: {best_score:.6f} ** NEW BEST **")

    if remaining > 8:
        # Final multiplicative
        remaining2 = deadline - time.monotonic()
        a_fm = multiplicative_update(best_seq, time_budget=remaining2 * 0.5, n_max_active=30)
        obj_fm = compute_obj(a_fm)
        if obj_fm < best_score:
            best_score = obj_fm
            best_seq = a_fm.copy()
            print(f"  Final MU: {best_score:.6f} ** NEW BEST **")

    # === Validation ===
    if best_seq is None:
        best_seq = np.ones(500) / 500
        best_score = compute_obj(best_seq)

    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    val = evaluate_sequence(result)
    if val < np.inf and val < best_score:
        best_score = val
    elif val == np.inf:
        result = [1.0] * 500
        best_score = 2.0

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_score:.6f}, time={elapsed:.1f}s ===")

    return result
