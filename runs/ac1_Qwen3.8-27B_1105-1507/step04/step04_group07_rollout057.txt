import numpy as np
from scipy.optimize import linprog
from scipy.sparse import csr_matrix
from scipy.signal import fftconvolve
import time

# ============ Utility Functions ============

def fast_conv(a):
    """Compute linear self-convolution using FFT."""
    n = len(a)
    s = 1
    while s < 2 * n - 1:
        s <<= 1
    fa = np.fft.rfft(a, s)
    return np.fft.irfft(fa * fa, s)[:2 * n - 1]

def compute_J(a):
    """Compute the objective J(a) = 2*n*max(conv(a,a))/sum(a)^2."""
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))

def normalize(a):
    """Normalize to sum=1, ensuring non-negativity."""
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(len(a)) / len(a)
    return a / s

# ============ LP Best Response ============

_spar_cache = {}

def get_sparsity(n):
    """Get sparse matrix structure for convolution constraints."""
    if n not in _spar_cache:
        rows = []
        cols = []
        for k in range(2 * n - 1):
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            rows.extend([k] * (je - js + 1))
            cols.extend(range(js, je + 1))
        _spar_cache[n] = (np.array(rows), np.array(cols))
    return _spar_cache[n]

def build_A(n, a):
    """Build the sparse constraint matrix for the LP."""
    rows, cols = get_sparsity(n)
    vals = a[rows - cols]  # a[k-j] for constraint k, variable j
    return csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))

def lp_best_response(a, M, time_limit=15):
    """Solve LP: max sum(g) s.t. (a*g)[k] <= M, g >= 0."""
    n = len(a)
    A = build_A(n, a)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                options={'presolve': True, 'time_limit': time_limit})
    if r.success and r.x is not None:
        g = np.maximum(r.x, 0.0)
        if g.sum() > 1e-10:
            return g, r
    return None, None

def lp_min_peak(a, time_limit=15):
    """Solve LP: min M s.t. (a*g)[k] <= M, sum(g)=1, g >= 0."""
    n = len(a)
    # Variables: g[0..n-1], M
    # Constraints: (a*g)[k] - M <= 0 for all k, sum(g) = 1, g >= 0, M >= 0
    rows, cols = get_sparsity(n)
    # For each constraint k: sum_j a[k-j]*g[j] - M <= 0
    # A matrix: [conv_matrix, -e] where e is the column for M
    n_cols = n + 1  # g + M
    A_rows = []
    A_cols = []
    A_vals = []
    b_ub = []
    for k in range(2 * n - 1):
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        for j in range(js, je + 1):
            A_rows.append(k)
            A_cols.append(j)
            A_vals.append(a[k - j])
        A_rows.append(k)
        A_cols.append(n)  # M column
        A_vals.append(-1.0)
        b_ub.append(0.0)
    
    A_ub = csr_matrix((A_vals, (A_rows, A_cols)), shape=(2 * n - 1, n_cols))
    
    # Equality: sum(g) = 1
    A_eq = np.zeros((1, n_cols))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])
    
    c = np.zeros(n_cols)
    c[n] = 1.0  # minimize M
    
    bounds = [(0.0, None)] * n + [(0.0, None)]
    
    r = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                bounds=bounds, method='highs', options={'presolve': True, 'time_limit': time_limit})
    if r.success and r.x is not None:
        return r.x[:n], r.x[n], r
    return None, None, None

# ============ Initializations ============

def zadoff_chu_init(n, rng):
    """Generate sequence based on Zadoff-Chu-like chirp."""
    m = int(rng.integers(1, n))
    if m < 1:
        m = 1
    t = np.arange(n)
    # Zadoff-Chu: exp(-j*pi*m*k*(k+1)/N)
    if (m * n) % 2 == 0:
        m += 1
    phase = -np.pi * m * t * (t + 1) / n
    A = np.exp(1j * phase)
    # Make it non-negative by taking real part of IFFT
    a = np.fft.ifft(A).real
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-10:
        # Try absolute value
        a = np.abs(np.fft.ifft(A).real)
        s = a.sum()
    if s < 1e-10:
        return np.ones(n) / n
    return a / s

def chirp_init(n, rng):
    """Linear chirp with random parameters."""
    t = np.arange(n) / n
    f0 = rng.uniform(0, 5)
    f1 = rng.uniform(5, 20)
    phase = 2 * np.pi * (f0 * t + 0.5 * (f1 - f0) * t**2)
    a = np.abs(np.exp(1j * phase) + rng.normal(0, 0.3, n))
    a = np.maximum(a, 0.01)
    return a / a.sum()

def random_fourier_init(n, rng, n_coeff=None):
    """Random Fourier series with limited bandwidth."""
    if n_coeff is None:
        n_coeff = int(rng.integers(max(3, n // 10), max(5, n // 5)))
    n_coeff = min(n_coeff, n // 2)
    t = np.linspace(0, 1, n, endpoint=False)
    a = np.ones(n) * 0.5
    for k in range(n_coeff):
        freq = rng.integers(1, n)
        amp = rng.uniform(0.1, 0.5)
        phase = rng.uniform(0, 2 * np.pi)
        a += amp * np.cos(2 * np.pi * freq * t + phase)
    a = np.maximum(a, 0.01)
    return a / a.sum()

def gaussian_mix_init(n, rng):
    """Mixture of Gaussian peaks."""
    n_peaks = int(rng.integers(2, 8))
    a = np.zeros(n)
    for _ in range(n_peaks):
        mu = rng.uniform(0, n)
        sigma = rng.uniform(n * 0.05, n * 0.3)
        a += np.exp(-0.5 * ((np.arange(n) - mu) / sigma) ** 2)
    a += 0.05
    return a / a.sum()

def flat_spectrum_init(n, rng):
    """Sequence with approximately flat power spectrum."""
    half = n // 2 + 1
    angles = rng.uniform(0, 2 * np.pi, half)
    # Slightly non-uniform amplitudes
    amps = 1.0 + rng.uniform(-0.2, 0.2, half)
    A = np.zeros(n, dtype=complex)
    A[:half] = amps[:half] * np.exp(1j * angles[:half])
    for k in range(1, n - n // 2):
        A[k] = np.conj(A[n - k])
    a = np.fft.ifft(A).real
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-10:
        return np.ones(n) / n
    return a / s

def resample_seq(src, dest_n):
    """Resample sequence to new length."""
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)

def generate_inits(n, rng, parent):
    """Generate diverse initializations."""
    inits = []
    
    if parent is not None and parent.size > 0:
        inits.append(resample_seq(parent, n))
        inits.append(resample_seq(parent[::-1], n))
        # Perturbed parent
        p = resample_seq(parent, n).copy()
        p += rng.normal(0, 0.05, n)
        inits.append(np.maximum(p, 0.01))
    
    inits.append(np.ones(n))
    inits.append(rng.uniform(0.2, 1.0, n))
    
    # Spectral inits
    for _ in range(3):
        inits.append(flat_spectrum_init(n, rng))
    
    # Chirp
    for _ in range(2):
        inits.append(chirp_init(n, rng))
    
    # Zadoff-Chu
    inits.append(zadoff_chu_init(n, rng))
    
    # Random Fourier
    for _ in range(3):
        inits.append(random_fourier_init(n, rng))
    
    # Gaussian mix
    for _ in range(2):
        inits.append(gaussian_mix_init(n, rng))
    
    # Structured
    t = np.linspace(0, 1, n, endpoint=False)
    inits.append(1.0 + 0.3 * np.cos(2 * np.pi * 3 * t))
    inits.append(1.0 + 0.2 * np.cos(2 * np.pi * 5 * t) + 0.1 * np.sin(2 * np.pi * 8 * t))
    inits.append(0.5 + 0.5 * np.cos(np.pi * t) ** 2)
    inits.append(0.3 + 0.7 * np.abs(np.sin(np.pi * t)) ** 0.5)
    
    # Power law
    inits.append((t + 0.05) ** rng.uniform(-0.5, 1.5))
    inits.append((1.0 - t + 0.05) ** rng.uniform(-0.5, 1.5))
    
    # Sparse + smooth
    sparse = np.zeros(n)
    n_spikes = max(5, n // 20)
    positions = rng.choice(n, size=n_spikes, replace=False)
    sparse[positions] = 1.0
    kernel_size = max(3, n // 25)
    kernel = np.exp(-0.5 * ((np.arange(kernel_size) - kernel_size // 2) / (kernel_size // 4)) ** 2)
    kernel /= kernel.sum()
    inits.append(np.convolve(sparse, kernel, mode='same') + 0.05)
    
    # Periodic variants
    for period in [8, 13, 21]:
        if period < n:
            pattern = rng.uniform(0.5, 1.0, period)
            inits.append(np.tile(pattern, n // period + 1)[:n])
    
    # Clean up
    cleaned = []
    for x in inits:
        x = np.asarray(x, dtype=np.float64)
        x = np.maximum(x, 0.001)
        x /= x.sum()
        cleaned.append(x)
    return cleaned

# ============ LP Optimization ============

def run_lp_opt(a, max_iters, deadline, tracker, tighten_start=0.05, tighten_end=0.005):
    """LP-based optimization with adaptive tightening."""
    a = np.maximum(a.astype(np.float64), 1e-8)
    a = normalize(a)
    n = len(a)
    current_J = compute_J(a)
    tracker.update(a)
    
    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        
        b = fast_conv(a)
        M = float(b.max())
        
        # Adaptive tightening: start loose, get tighter
        progress = it / max(1, max_iters)
        tighten = tighten_start * (1 - progress) + tighten_end * progress
        
        M_tight = M * (1.0 - tighten)
        g, lp_result = lp_best_response(a, M_tight, time_limit=10)
        
        if g is None:
            g, lp_result = lp_best_response(a, M, time_limit=10)
        
        if g is None:
            break
        
        g = np.maximum(g, 0.0)
        gs = g.sum()
        if gs < 1e-10:
            break
        g = g / gs
        
        # Adaptive mixing: determine step based on similarity
        cosine_sim = np.dot(a, g) / (np.linalg.norm(a) * np.linalg.norm(g) + 1e-15)
        base_t = 0.3 + 0.4 * (1.0 - cosine_sim)  # More different = larger step
        base_t = max(0.1, min(0.9, base_t))
        
        best_a = None
        best_J = current_J
        candidates_t = [base_t, base_t * 1.5, base_t * 0.5, 0.7, 0.3, 0.9]
        for t in candidates_t:
            if t >= 1.0:
                continue
            a_trial = (1.0 - t) * a + t * g
            a_trial = np.maximum(a_trial, 0.0)
            ss = a_trial.sum()
            if ss < 1e-10:
                continue
            a_trial /= ss
            J_trial = compute_J(a_trial)
            if J_trial < best_J - 1e-14:
                best_J = J_trial
                best_a = a_trial
        
        if best_a is None:
            # Try min-peak LP as alternative
            g2, M2, _ = lp_min_peak(a, time_limit=10)
            if g2 is not None:
                g2 = normalize(g2)
                a_trial = 0.5 * a + 0.5 * g2
                a_trial = normalize(a_trial)
                J_trial = compute_J(a_trial)
                if J_trial < best_J - 1e-14:
                    best_J = J_trial
                    best_a = a_trial
        
        if best_a is None:
            # Try smaller perturbations
            for t in [0.1, 0.05, 0.02]:
                a_trial = (1.0 - t) * a + t * g
                a_trial = normalize(np.maximum(a_trial, 0.0))
                J_trial = compute_J(a_trial)
                if J_trial < best_J - 1e-14:
                    best_J = J_trial
                    best_a = a_trial
                    break
        
        if best_a is None:
            break
        
        a = best_a
        current_J = best_J
        if it % 5 == 0:
            tracker.update(a)
    
    tracker.update(a)
    return a

# ============ Adam LSE Gradient Descent ============

def adam_lse_descent(a, max_iters, deadline, tracker):
    """Adam-optimized LSE smoothed gradient descent."""
    a = np.maximum(a.astype(np.float64), 1e-6)
    a = normalize(a)
    n = len(a)
    current_J = compute_J(a)
    tracker.update(a)
    
    # Adam state
    m_v = np.zeros(n)
    v_v = np.zeros(n)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    t_adam = 0
    lr = 1.0 / (n * 20)
    
    # LSE temperature schedule
    tau0 = 20.0
    tau_max = 200.0
    
    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        
        b = fast_conv(a)
        b_max = b.max()
        
        # Adaptive temperature
        tau = min(tau_max, tau0 * (1 + it / 100))
        
        # Stable LSE
        exp_b = np.exp(tau * (b - b_max))
        p = exp_b / exp_b.sum()
        
        # Gradient of LSE: 2 * cross-correlation(p, a)[:n]
        s = 1
        while s < 3 * n:
            s <<= 1
        fp = np.fft.rfft(p, s)
        fa = np.fft.rfft(a, s)
        corr = np.fft.irfft(fp * np.conj(fa), s)
        grad = 2.0 * corr[:n]
        
        # Adam update
        t_adam += 1
        m_v = beta1 * m_v + (1 - beta1) * grad
        v_v = beta2 * v_v + (1 - beta2) * grad ** 2
        m_hat = m_v / (1 - beta1 ** t_adam)
        v_hat = v_v / (1 - beta2 ** t_adam)
        
        a_new = a - lr * m_hat / (np.sqrt(v_hat) + eps)
        a_new = np.maximum(a_new, 1e-6)
        a_new = normalize(a_new)
        
        J_new = compute_J(a_new)
        if J_new < current_J - 1e-15:
            a = a_new

def subgradient_descent(a, max_iters, deadline, tracker):
    """Subgradient descent with averaged smoothing."""
    a = np.maximum(a.astype(np.float64), 1e-8)
    a = normalize(a)
    n = len(a)
    current_J = compute_J(a)
    tracker.update(a)
    
    lr = 0.3 / n
    smooth_grad = np.zeros(n)
    alpha = 0.7  # smoothing factor
    
    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        
        b = fast_conv(a)
        M = float(b.max())
        
        # Find active set (peaks near maximum)
        threshold = M - M * 0.01  # 1% below max
        active = np.where(b >= threshold)[0]
        
        # Compute subgradient
        grad = np.zeros(n)
        for k in active:
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            for j in range(js, je + 1):
                grad[j] += a[k - j]
        grad /= len(active)
        
        # Smooth the gradient
        smooth_grad = alpha * smooth_grad + (1 - alpha) * grad
        direction = smooth_grad - 2.0 * M * a
        
        a_new = a - lr * direction
        a_new = np.maximum(a_new, 1e-8)
        a_new = normalize(a_new)
        
        J_new = compute_J(a_new)
        if J_new < current_J - 1e-15:
            a = a_new
            current_J = J_new
            lr = min(lr * 1.01, 3.0 / n)
        else:
            lr *= 0.7
        
        if it % 30 == 0:
            tracker.update(a)
    
    tracker.update(a)
    return a

# ============ Coordinate Descent ============

def coordinate_descent(a, max_steps, deadline, tracker):
    """Cyclic coordinate descent with optimal line search."""
    a = a.copy().astype(np.float64)
    a = normalize(a)
    n = len(a)
    current_J = compute_J(a)
    tracker.update(a)
    
    for step in range(max_steps):
        if time.monotonic() > deadline:
            break
        
        improved = False
        for j in range(n):
            if time.monotonic() > deadline:
                    break
            
            old_val = a[j]
            if old_val < 1e-10:
                continue
            
            # Try different scales
            for scale in [0.5, 0.7, 1.3, 1.5, 0.3, 2.0]:
                a_trial = a.copy()
                a_trial[j] = old_val * scale
                a_trial = np.maximum(a_trial, 0.0)
                ss = a_trial.sum()

                J_trial = compute_J(a_trial)

                    current_J = J_trial
                    improved = True
                    break

    return a

def pairwise_descent(a, max_steps, deadline, tracker, rng):

# ============ Score Tracker ============

class ScoreTracker:
    def __init__(self):
        self.best = None
        self.best_score = float('inf')
        self.t_start = time.monotonic()
    
    def update(self, a):
        if a is None:
            return
        a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)

# ============ Simulated Annealing ============

def simulated_annealing(a, deadline, tracker, n_restarts=80, t0=0.12):
    """Simulated annealing with LP polishing."""
    rng = np.random.default_rng(123)
    n = a.size
    a = a.copy().astype(np.float64)
    a = normalize(a)
    current_J = compute_J(a)
    tracker.update(a)
    temp = t0
    
    for r in range(n_restarts):
        if time.monotonic() > deadline:
            break
        
        # Perturb
        sigma = temp * a.std() + 1e-4
        a_pert = a + rng.normal(0, sigma, n)
        a_pert = np.maximum(a_pert, 1e-6)
        a_pert = normalize(a_pert)
        
        # Short LP refinement
        a_refined = run_lp_opt(a_pert, max(3, 15 - r // 10), deadline, tracker,
                                tighten_start=0.03, tighten_end=0.01)
        
        J = compute_J(a_refined)
        delta = J - current_J
        if delta < 0 or rng.random() < np.exp(-delta / max(temp, 1e-10)):
            a = a_refined
            current_J = J
            tracker.update(a)
        temp *= 0.96
    
    return a
