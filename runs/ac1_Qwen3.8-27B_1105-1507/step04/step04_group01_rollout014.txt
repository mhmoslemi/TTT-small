Intensive LP-based optimization with dual-guided updates, coordinate descent,
and simulated annealing for minimizing 2*n*max(conv(a,a))/(sum(a))^2.
Focuses on n~600 starting from the parent construction.
"""

import numpy as np
from scipy.optimize import linprog, minimize
from scipy.fft import rfft, irfft
import time


def compute_conv(a):
    n = len(a)
    size = 1
    while size < 2*n-1:
        size <<= 1
    fa = rfft(a, size)
    conv = irfft(fa*fa, size)
    return conv[:2*n-1]


def compute_obj(a):
    n = len(a)
    conv = compute_conv(a)
    max_b = float(np.max(conv))
    sum_a = float(np.sum(a))
    if sum_a < 0.01:
        return np.inf
    return 2.0*n*max_b/(sum_a*sum_a)


def fft_convolve(a, b):
    n, m = len(a), len(b)
    size = 1
    while size < n+m-1:
        size <<= 1
    fa = rfft(a, size)
    fb = rfft(b, size)
    return irfft(fa*fb, size)[:n+m-1]


def build_toeplitz(a):
    n = len(a)
    rows = 2*n-1
    A = np.zeros((rows, n))
    for j in range(n):
        A[j:j+n, j] = a
    return A


def solve_lp_max_sum(a, M, time_limit=10):
    """LP: max sum(g) s.t. (a*g)[k] <= M, g >= 0"""
    n = len(a)
    A = build_toeplitz(a)
    c = -np.ones(n)
    b_ub = np.full(2*n-1, M)
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': time_limit})
        if res.status == 0:
            g = np.maximum(res.x, 0)
            s = float(np.sum(g))
            if s > 1e-10:
                return g/s, s, res
    except Exception:
        pass
    return None, None, None


def solve_lp_min_peak(a, time_limit=10):
    """LP: min M s.t. (a*g)[k] <= M, sum(g)=1, g>=0"""
    n = len(a)
    A = build_toeplitz(a)
    c = np.zeros(n+1)
    c[-1] = 1.0
    A_ub = np.zeros((2*n-1, n+1))
    A_ub[:, :n] = A
    A_ub[:, n] = -1.0
    b_ub = np.zeros(2*n-1)
    A_eq = np.zeros((1, n+1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])
    bounds = [(0.0, None)] * (n+1)
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': time_limit})
        if res.status == 0:
            g = np.maximum(res.x[:n], 0)
            M_opt = float(res.x[n])
            return g, M_opt
    except Exception:
        pass
    return None, None


def lp_step(a, time_limit=10):
    """One LP step: solve LP, find best blend direction."""
    n = len(a)
    s = np.sum(a)
    if s < 1e-10:
        return a
    a_norm = a / s
    
    conv = compute_conv(a_norm)
    M = float(np.max(conv))

    if g is None:

    # Try several blend factors
    best_a = a.copy()
    best_obj = compute_obj(a)

def coordinate_descent(a, max_iters=10000, time_budget=30.0):
    """Coordinate descent: optimize one element at a time."""
    n = len(a)
    a = a.copy().astype(np.float64)
    best_obj = compute_obj(a)
    start = time.monotonic()
    rng = np.random.default_rng(42)
    
    it = 0
    while it < max_iters and time.monotonic() - start < time_budget:
        it += 1
        i = rng.integers(0, n)
        a_old = a[i]
        
        # Try several multiplicative perturbations
        best_val = a_old
        best_local_obj = best_obj
        
        for scale in [0.5, 0.7, 0.85, 1.0, 1.15, 1.3, 1.5, 2.0, 0.0]:

    a_obj = compute_obj(a)
    return a


def softmax(z):
    z = z - np.max(z)
    e = np.exp(z)
    return e / np.sum(e)


def smooth_objective(a, tau):
    """Soft-max smoothed objective."""
    conv = compute_conv(a)
    cmax = float(np.max(conv))
    if tau < 1e-10:
        return cmax
    shifted = (conv - cmax) / tau
    shifted = np.clip(shifted, -500, 500)
    exp_c = np.exp(shifted)
    Z = np.sum(exp_c)
    return cmax + tau * np.log(Z)


def smooth_grad(a, tau):
    """Gradient of smooth objective w.r.t. a."""
    conv = compute_conv(a)
    cmax = float(np.max(conv))
    shifted = (conv - cmax) / tau
    shifted = np.clip(shifted, -500, 500)
    exp_c = np.exp(shifted)
    w = exp_c / np.sum(exp_c)
    
    # grad_a[i] = 2 * sum_j w[i+j] * a[j]
    w_rev = w[::-1]
    conv_fw = fft_convolve(a, w_rev)
    grad_a = 2.0 * conv_fw[len(a)-1:2*len(a)-1][::-1]
    
    return grad_a


def lbfgs_smooth(a_init, taus, time_budget=30.0):

    n = len(a_init)
    a = a_init.copy()

    a = a / s
    a = np.maximum(a, 1e-15)
    z = np.log(a)
    
    best_a = a.copy()
    best_obj = compute_obj(a)

            break

"""
Intensive LP-driven optimization for minimizing 2*n*max(conv(a,a))/(sum(a))^2.
Uses alternating LP with dual-guided line search, coordinated with smooth
L-BFGS-B relaxation and coordinate descent for fine-tuning. Focuses on
n~600 starting from the parent construction. Key improvements over prior:
(1) much more LP iterations with adaptive line search, (2) min-peak LP

import numpy as np
from scipy.optimize import linprog, minimize
from scipy.fft import rfft, irfft
import time


def compute_conv(a):
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    conv = irfft(fa * fa, size)
    return conv[:2 * n - 1]


def compute_obj(a):
    n = len(a)
    if n == 0:
        return np.inf
    conv = compute_conv(a)
    max_b = float(np.max(conv))
    sum_a = float(np.sum(a))
    if sum_a < 0.01:
        return np.inf
    return 2.0 * n * max_b / (sum_a * sum_a)


def fft_convolve(a, b):
    n, m = len(a), len(b)
    size = 1
    while size < n + m - 1:
        size <<= 1
    fa = rfft(a, size)
    fb = rfft(b, size)
    return irfft(fa * fb, size)[:n + m - 1]


def build_toeplitz(a):
    n = len(a)
    rows = 2 * n - 1
    A = np.zeros((rows, n))
    for j in range(n):
        A[j:j + n, j] = a
    return A


def build_toeplitz_sym(a_half, n):
    """Build constraint matrix for symmetric LP. a_half has length n//2+1 (for even n, n//2)."""
    half_n = len(a_half)
    # Full a is determined by a_half: a[i] = a_half[min(i, n-1-i)]
    # For even n: a_half has length n//2, a[i] = a_half[i] for i < n//2, a_half[n-1-i] for i >= n//2
    full_a = np.zeros(n)
    for i in range(n):
        full_a[i] = a_half[min(i, n - 1 - i)]
    return build_toeplitz(full_a)


def solve_lp_max_sum(a, M, time_limit=10):
    n = len(a)
    A = build_toeplitz(a)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': time_limit})
        if res.status == 0:
            g = np.maximum(res.x, 0)
            s = float(np.sum(g))
            if s > 1e-10:
                return g / s, s
    except Exception:
        pass
    return None, None


def solve_lp_min_peak(a, time_limit=10):
    n = len(a)
    A = build_toeplitz(a)
    c = np.zeros(n + 1)
    c[-1] = 1.0
    A_ub = np.zeros((2 * n - 1, n + 1))
    A_ub[:, :n] = A
    A_ub[:, n] = -1.0
    b_ub = np.zeros(2 * n - 1)
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])
    bounds = [(0.0, None)] * (n + 1)
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': time_limit})
        if res.status == 0:
            g = np.maximum(res.x[:n], 0)
            M_opt = float(res.x[n])

    except Exception:
        pass

        a_try = a_try / s

            a_try = a_try / s
            if abs(compute_obj(a_try) - best_local) < 1e-12:
                t_best = t
                break
        # Fine search around t_best
        for dt in [-0.03, -0.02, -0.01, 0.01, 0.02, 0.03]:
            t2 = t_best + dt
            if t2 <= 0 or t2 > 1:
                continue
            a_try = (1.0 - t2) * a_norm + t2 * dir_norm
            a_try = np.maximum(a_try, 0)
            s = np.sum(a_try)
            if s < 1e-10:
                continue
            a_try = a_try / s
            obj = compute_obj(a_try)
            if obj < best_local:
                best_local = obj
                best_a = a_try.copy()

    return best_a, best_local


def softmax_fn(z):
    z = z - np.max(z)
    e = np.exp(z)
    return e / np.sum(e)


def smooth_obj_and_grad_z(z, tau):
    n = len(z)
    a = softmax_fn(z)
    conv = compute_conv(a)
    cmax = float(np.max(conv))
    shifted = np.clip((conv - cmax) / tau, -500.0, 500.0)
    exp_c = np.exp(shifted)
    Z = float(np.sum(exp_c))
    soft_val = cmax + tau * np.log(Z)
    w = exp_c / Z
    w_rev = w[::-1]
    conv_fw = fft_convolve(a, w_rev)
    grad_a = 2.0 * conv_fw[n - 1:2 * n - 1][::-1]
    mean_g = float(np.dot(a, grad_a))
    grad_z = a * (grad_a - mean_g)
    return soft_val, grad_z


def lbfgs_smooth(a_init, taus, time_budget=30.0, max_iter=200):
    n = len(a_init)
    a = a_init.copy()
    s = np.sum(a)
    if s < 1e-10:
        a = np.ones(n) / n
    else:
        a = a / s
    a = np.maximum(a, 1e-15)
    z = np.log(a)

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    for tau in taus:
        if time.monotonic() - start > time_budget:
            break

        def func_grad(z_val):
            return smooth_obj_and_grad_z(z_val, tau)

        try:
            res = minimize(func_grad, z, jac=True, method='L-BFGS-B',
                           options={'maxiter': max_iter, 'ftol': 1e-15, 'gtol': 1e-8})
            z = res.x
            a_new = softmax_fn(z)
            a_new = np.maximum(a_new, 0)
            s_new = np.sum(a_new)
            if s_new > 1e-10:
                a_new = a_new / s_new
                obj_new = compute_obj(a_new)
                if obj_new < best_obj:
                    best_obj = obj_new
                    best_a = a_new.copy()
        except Exception:
            pass

    return best_a


def coordinate_descent(a_init, time_budget=30.0, n_scales=7):
    n = len(a_init)
    a = a_init.copy().astype(np.float64)
    s = np.sum(a)
    a = a / s
    best_obj = compute_obj(a)
    best_a = a.copy()
    start = time.monotonic()
    rng = np.random.default_rng(777)

    scales = [0.3, 0.5, 0.7, 0.85, 1.15, 1.3, 1.7, 3.0, 0.0]
    scales = scales[:n_scales + 1] if n_scales + 1 <= len(scales) else scales

    while time.monotonic() - start < time_budget:
        # Cycle through a subset of coordinates
        for _ in range(min(n, 10)):
            if time.monotonic() - start > time_budget:
                break
            i = rng.integers(0, n)
            a_old = a[i]
            best_local = best_obj
            best_val = a_old

            for sc in scales:
                a[i] = a_old * sc
                obj = compute_obj(a)
                if obj < best_local:
                    best_local = obj
                    best_val = a[i]

            a[i] = best_val
            if best_local < best_obj - 1e-14:
                best_obj = best_local
                best_a = a.copy()

    return best_a


def alternating_lp_intensive(a, time_budget=60.0, verbose=False):
    """Run alternating LP with adaptive strategy."""
    n = len(a)
    a = a.copy().astype(np.float64)
    s = np.sum(a)
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    no_improve = 0
    use_min_peak = False
    iteration = 0

    while time.monotonic() - start < time_budget:
        iteration += 1
        remaining = time_budget - (time.monotonic() - start)
        if remaining < 2:
            break

        # Alternate between LP formulations
        if use_min_peak and n <= 600:
            g, M_opt = solve_lp_min_peak(a, time_limit=8)
        else:
            conv = compute_conv(a)
            M = float(np.max(conv))
            g, _ = solve_lp_max_sum(a, M, time_limit=8)
        use_min_peak = not use_min_peak

        if g is None:
            # Perturb and retry
            rng = np.random.default_rng(iteration * 31 + 7)
            a = a * np.exp(rng.normal(0, 0.04, n))
            a = np.maximum(a, 0)
            s = np.sum(a)
            if s > 1e-10:
                a = a / s
            continue

        a_new, obj_new = line_search(a, g, best_obj)

        if obj_new < best_obj - 1e-13:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new
            no_improve = 0
            if verbose and iteration % 20 == 0:
                print(f"    LP iter {iteration}: obj={best_obj:.8f}, remaining={remaining:.0f}s")
        else:
            no_improve += 1
            if no_improve >= 10:
                # Stronger perturbation
                rng = np.random.default_rng(iteration * 7919 + 13)
                scale = 0.03 + 0.02 * (no_improve // 10)
                scale = min(scale, 0.15)
                a = a * np.exp(rng.normal(0, scale, n))
                a = np.maximum(a, 0)
                s = np.sum(a)
                if s > 1e-10:
                    a = a / s
                no_improve = 0
            else:
                # Small step towards g even without improvement (momentum)
                a = 0.97 * a + 0.03 * g
                a = np.maximum(a, 0)
                s = np.sum(a)
                if s > 1e-10:
                    a = a / s

    return best_a


def try_parent():
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64)
        parent = np.clip(parent, 0, 1000.0)
        if parent.size > 0 and np.isfinite(parent).all():
            obj = compute_obj(parent)
            print(f"Parent (n={parent.size}): obj={obj:.8f}")
            return parent.copy(), obj
    except Exception:
        pass
    return None, np.inf


def resample_seq(a, new_n):
    n = len(a)
    if n == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    return np.interp(x_new, x_old, a)


def propose_candidate(seed=42, budget_s=1500):
    """
    Main search: intensive LP-driven optimization with smooth relaxation.
    Phase 1: Quick n exploration (60s)
    Phase 2: Intensive alternating LP (1200s)
    Phase 3: Final polish with L-BFGS + coordinate descent (150s)
    """
    start_time = time.monotonic()
    deadline = start_time + budget_s - 25.0

    best_seq = None
    best_score = np.inf

    # Load parent
    parent, parent_obj = try_parent()
    if parent is not None:
        best_score = parent_obj
        best_seq = parent.copy()

    if best_seq is None:
        n = 600
        best_seq = np.ones(n) / n
        best_score = compute_obj(best_seq)
        print(f"Random init: n={n}, score={best_score:.8f}")

    n_best = len(best_seq)
    print(f"Starting: n={n_best}, score={best_score:.8f}")

    # Phase 1: Quick exploration (60s)
    print("\n=== Phase 1: Quick n exploration ===")
    phase1_end = start_time + 60.0

    n_tries = [n_best]
    for dn in [-15, -10, -5, 5, 10, 15, 20, -20]:
        nt = n_best + dn
        if 200 <= nt <= 900:
            n_tries.append(nt)
    n_tries = sorted(set(n_tries))

    for nt in n_tries:
        if time.monotonic() > phase1_end:
            break
        if nt == n_best:
            continue
        a_try = resample_seq(best_seq, nt)
        a_try = lbfgs_smooth(a_try, [2.0/nt, 0.5/nt], time_budget=5.0, max_iter=100)
        obj_try = compute_obj(a_try)
        print(f"  n={nt}: {obj_try:.8f}")
        if obj_try < best_score:
            best_score = obj_try
            best_seq = a_try.copy()
            n_best = nt
            print(f"  ** New best: n={nt}, {best_score:.8f}")

    a_work = best_seq.copy()

    # Phase 2: Intensive LP (majority of time)
    print(f"\n=== Phase 2: Intensive LP (n={n_best}, score={best_score:.8f}) ===")
    phase2_end = start_time + budget_s - 200.0
    n_w = len(a_work)

    # Run LP in chunks with periodic smooth refresh
    chunk_size = 90.0
    while time.monotonic() < phase2_end:
        remaining = phase2_end - time.monotonic()
        if remaining < 15:
            break

        t_chunk = min(chunk_size, remaining)
        a_work = alternating_lp_intensive(a_work, time_budget=t_chunk, verbose=True)
        obj_work = compute_obj(a_work)

        if obj_work < best_score - 1e-12:
            best_score = obj_work
            best_seq = a_work.copy()
            print(f"  ** Improvement: {best_score:.8f} ({phase2_end - time.monotonic():.0f}s left)")
        else:
            # Periodic smooth refresh to escape LP local minima
            remaining = phase2_end - time.monotonic()
            if remaining > 30:
                n_w = len(a_work)
                tau_list = [1.0/n_w, 0.3/n_w, 0.1/n_w]
                a_smooth = lbfgs_smooth(a_work, tau_list, time_budget=min(15, remaining*0.2), max_iter=150)
                obj_smooth = compute_obj(a_smooth)
                if obj_smooth < best_score:
                    best_score = obj_smooth
                    best_seq = a_smooth.copy()
                    a_work = a_smooth.copy()
                    print(f"  ** Smooth improvement: {best_score:.8f}")
                else:
                    # Perturb
                    rng = np.random.default_rng(int(time.monotonic()*1000) % (2**31))
                    a_work = a_work * np.exp(rng.normal(0, 0.06, n_w))
                    a_work = np.maximum(a_work, 0)
                    s = np.sum(a_work)
                    if s > 1e-10:
                        a_work = a_work / s

    # Phase 3: Final polish
    print(f"\n=== Phase 3: Final polish (score={best_score:.8f}) ===")

    a_work = best_seq.copy()
    n_f = len(a_work)

    # 3a: L-BFGS with tiny tau
    if time.monotonic() < deadline - 50:
        rem = deadline - time.monotonic() - 50
        taus = [0.3/n_f, 0.1/n_f, 0.05/n_f, 0.02/n_f, 0.01/n_f]
        a_work = lbfgs_smooth(a_work, taus, time_budget=min(50, rem*0.4), max_iter=300)
        obj = compute_obj(a_work)
        if obj < best_score:
            best_score = obj
            best_seq = a_work.copy()
            print(f"  L-BFGS: {best_score:.8f}")

    # 3b: Coordinate descent
    if time.monotonic() < deadline - 35:
        rem = deadline - time.monotonic() - 35
        a_work = coordinate_descent(best_seq, time_budget=min(35, rem*0.5), n_scales=7)
        obj = compute_obj(a_work)
        if obj < best_score:
            best_score = obj
            best_seq = a_work.copy()
            print(f"  Coord: {best_score:.8f}")

    # 3c: Final LP bursts
    if time.monotonic() < deadline - 20:
        rem = deadline - time.monotonic() - 20
        for burst in range(3):
            if time.monotonic() > deadline - 15:
                break
            a_work = alternating_lp_intensive(best_seq, time_budget=min(15, rem*0.3), verbose=False)
            obj = compute_obj(a_work)
            if obj < best_score:
                best_score = obj
                best_seq = a_work.copy()
                print(f"  Final LP {burst}: {best_score:.8f}")
            rem = deadline - time.monotonic() - 15

    # 3d: Very short coordinate descent
    if time.monotonic() < deadline - 5:
        a_work = coordinate_descent(best_seq, time_budget=min(10, deadline - time.monotonic() - 5), n_scales=5)
        obj = compute_obj(a_work)
        if obj < best_score:
            best_score = obj
            best_seq = a_work.copy()
            print(f"  Final coord: {best_score:.8f}")

    # Validate
    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    val = evaluate_sequence(result)
    if np.isfinite(val):
        best_score = val
    else:
        result = [1.0] * 100
        best_score = 2.0

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_score:.8f}, time={elapsed:.1f}s ===")

    return result
