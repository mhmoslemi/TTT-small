def lp_cutting_plane(a, max_iters, deadline, verbose=True):

    LP cutting-plane method to minimize max_k b[k] s.t. a>=0, sum(a)=1.

            best_a = a.copy()

        #      sum a'[i] = 1

        c[n] = 1.0  # minimize M_new

        cols = []
        vals = []

                cols.append(i)
                vals.append(2.0 * a[k-i])

            rows.append(k)
            cols.append(n)

def fast_conv(a):
    """Full self-convolution via FFT."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]


def compute_J(a):
    """Compute J(a) = 2*n*max(conv(a,a))/(sum(a))^2."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0)
    s = a.sum()
    if s < 0.01:

    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))


def smooth_max_gradient(a, tau):
    """Compute gradient of smooth max proxy. Returns (gradient, b)."""
    n = len(a)
    b = fast_conv(a)
    b_max = b.max()
    exp_b = np.exp(tau * (b - b_max))

        grad[lo:hi + 1] += 2.0 * pk * a[k - hi:k - lo + 1][::-1]

    return grad, b

            best_M = M
            best_a = a.copy()

        beta = 0.9 if momentum else 0.0
        v = beta * v + lr * grad
        a_new = a - v
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s < 1e-10:
            a_new = a.copy()
        else:
            a_new = a_new / s
        a = a_new

    M_final = fast_conv(a).max()
    return best_a if best_M < M_final else a


def lp_cutting_plane(a, max_iters, deadline, verbose=True):
    """
    LP cutting-plane: minimize M s.t. linearized b[k](a') <= M, a'>=0, sum(a')=1.
    Uses backtracking when the full LP step doesn't improve the true objective.
    """
    a = np.maximum(a.astype(np.float64), 0.0)
    a = a / a.sum()
    n = len(a)
    best_a = a.copy()
    best_M = fast_conv(a).max()
    best_cJ = 2.0 * n * best_M  # since sum=1, J = 2n*max_b
    no_improve = 0
    stuck_count = 0

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        b = fast_conv(a)
        M = float(b.max())

        if M < best_M * (1 - 1e-10):
            best_M = M
            best_a = a.copy()
            best_cJ = 2.0 * n * M
            no_improve = 0
        else:
            no_improve += 1

        if no_improve > 50 and it > 20:
            # Add small perturbation to escape
            rng = np.random.default_rng(it * 7919)
            a = a + rng.normal(0, 0.01, n)
            a = np.maximum(a, 0.001)
            a /= a.sum()
            no_improve = 0
            stuck_count += 1
            if stuck_count > 10:
                break

        # Build LP matrix
        rows = []
        cols = []
        vals = []
        b_ub = np.empty(2 * n - 1)

        for k in range(2 * n - 1):
            b_ub[k] = b[k]
            j_min = max(0, k - n + 1)
            j_max = min(n - 1, k)
            for i in range(j_min, j_max + 1):
                rows.append(k)
                cols.append(i)
                vals.append(2.0 * a[k - i])
            rows.append(k)
            cols.append(n)
            vals.append(-1.0)

        A_ub = csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n + 1))

        c_obj = np.zeros(n + 1)
        c_obj[n] = 1.0

        A_eq = np.zeros((1, n + 1))
        A_eq[0, :n] = 1.0
        b_eq = np.array([1.0])

        bounds = [(0, None)] * (n + 1)

        try:
            r = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                        bounds=bounds, method='highs',
                        options={'presolve': True})
        except Exception:
            break

        if not r.success or r.x is None:
            break

        a_new = np.maximum(r.x[:n], 0.0)
        s = a_new.sum()
        if s < 1e-10:
            break
        a_new /= s

        M_new = float(fast_conv(a_new).max())

        if M_new < M * (1 - 1e-10):
            a = a_new
        else:
            # Backtracking
            improved = False
            for t in [0.5, 0.25, 0.125, 0.0625, 0.03125]:
                a_trial = (1.0 - t) * a + t * a_new
                a_trial = np.maximum(a_trial, 0.0)
                st = a_trial.sum()
                if st < 1e-10:
                    continue
                a_trial /= st
                M_trial = float(fast_conv(a_trial).max())
                if M_trial < M * (1 - 1e-12):
                    a = a_trial
                    improved = True
                    break
            if not improved:
                if verbose and it % 50 == 0:
                    print(f"    LP iter {it}: M={M:.10f}, stuck", flush=True)
                no_improve += 10

        if verbose and it % 100 == 0:
            print(f"    LP iter {it}: M={M:.10f}, best={best_M:.10f}", flush=True)

    # Final check
    M_best = float(fast_conv(best_a).max())
    if M_best < best_M:
        best_M = M_best
    return best_a, best_M


def lp_best_response_polish(a, max_iters, deadline, verbose=True):
    """
    Original state-of-the-art: max sum(g) s.t. conv(a,g) <= M, g >= 0.
    Then interpolate a toward g.
    """
    a = np.maximum(a.astype(np.float64), 0.0)
    a = a / a.sum()
    n = len(a)
    best_a = a.copy()
    best_M = float(fast_conv(a).max())
    no_improve = 0

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        b = fast_conv(a)
        M = float(b.max())

        if M < best_M * (1 - 1e-10):
            best_M = M
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        if no_improve > 30:
            break

        # Build LP: max sum(g) s.t. A_k @ g <= M for all k, g >= 0
        rows = []
        cols = []
        vals = []
        for k in range(2 * n - 1):
            j_min = max(0, k - n + 1)
            j_max = min(n - 1, k)
            for i in range(j_min, j_max + 1):
                rows.append(k)
                cols.append(i)
                vals.append(a[k - i])

        A_mat = csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))
        c_obj = -np.ones(n)
        b_ub = np.full(2 * n - 1, M)
        bounds = [(0, None)] * n

        try:
            r = linprog(c_obj, A_ub=A_mat, b_ub=b_ub, bounds=bounds,
                        method='highs', options={'presolve': True})
        except Exception:
            break

        if not r.success or r.x is None:
            break

        g = np.maximum(r.x, 0.0)
        g_sum = g.sum()
        if g_sum < 1e-10:
            break
        g /= g_sum

        # Line search
        improved = False
        base_t = max(0.05, 0.5 * (1.0 - 0.5 * it / max(1, max_iters)))
        for t in [base_t, base_t * 0.5, base_t * 0.25, 0.02, 0.01]:
            a_trial = np.maximum((1.0 - t) * a + t * g, 0.0)
            st = a_trial.sum()
            if st < 1e-10:
                continue
            a_trial /= st
            M_trial = float(fast_conv(a_trial).max())
            if M_trial < best_M - 1e-14:
                a = a_trial
                best_M = M_trial
                best_a = a.copy()
                improved = True
                break

        if not improved:
            no_improve += 5

        if verbose and it % 50 == 0:
            print(f"    BR iter {it}: M={M:.10f}, best={best_M:.10f}", flush=True)

    return best_a, best_M


def generate_initializations(n, rng, parent=None):
    """Generate diverse initial sequences of length n."""
    inits = []
    t_arr = np.linspace(0, 1, n, endpoint=False)

    if parent is not None:
        from scipy.interpolate import interp1d
        p = np.array(parent, dtype=np.float64)
        if p.size != n:
            x_old = np.linspace(0, 1, p.size)
            x_new = np.linspace(0, 1, n)
            p = np.interp(x_new, x_old, p)
        p = np.maximum(p, 0.0)
        inits.append(p.copy())
        inits.append(np.maximum(p + rng.normal(0, 0.02, n), 0.01))
        inits.append(np.maximum(p + rng.normal(0, 0.08, n), 0.01))

    inits.append(np.ones(n))
    inits.append(rng.uniform(0.5, 1.0, n))

    for freq in [1, 2, 3, 5, 7, 11]:
        inits.append(np.maximum(1.0 + 0.3 * np.cos(2 * np.pi * freq * t_arr), 0.01))
        inits.append(np.maximum(1.0 + 0.4 * np.sin(2 * np.pi * freq * t_arr), 0.01))

    inits.append(np.maximum(1.0 + 0.3 * np.cos(2 * np.pi * 2 * t_arr**2), 0.01))
    inits.append(np.exp(-4.0 * (t_arr - 0.5)**2) + 0.1)
    inits.append(np.exp(-8.0 * (t_arr - 0.3)**2) + np.exp(-8.0 * (t_arr - 0.7)**2) + 0.05)
    inits.append(np.maximum(1.0 - 0.4 * np.abs(2 * t_arr - 1), 0.05))

    ksize = max(3, n // 30)
    kernel = np.ones(ksize) / ksize
    smooth = np.convolve(rng.uniform(0, 1, n), kernel, mode='same')
    inits.append(np.maximum(smooth + 0.1, 0.01))

    # Two-tone
    inits.append(np.where(np.arange(n) % 2 == 0, 1.0, 0.5))
    inits.append(np.where(np.arange(n) % 3 == 0, 0.3, 1.0))

    # Ramp
    inits.append(np.maximum(0.3 + 0.7 * t_arr, 0.01))
    inits.append(np.maximum(0.3 + 0.7 * (1 - t_arr), 0.01))

    # Mixed frequency
    inits.append(np.maximum(1.0 + 0.2 * np.cos(4 * np.pi * t_arr) +
                            0.2 * np.cos(7 * np.pi * t_arr) +
                            0.1 * np.sin(3 * np.pi * t_arr), 0.01))

    for ini in inits:
        ini = np.maximum(ini.astype(np.float64), 0.001)
        ini = ini / ini.sum()
    return inits


def score_with_evaluate(a):
    """Evaluate using the provided evaluate_sequence function."""
    a_clipped = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
    if a_clipped.size == 0 or a_clipped.sum() < 0.01:
        return float('inf')
    return float(evaluate_sequence(a_clipped.tolist()))


class ScoreTracker:
    """Tracks the best sequence found so far."""

    def __init__(self):
        self.best = None
        self.best_score = float('inf')
        self.t_start = time.monotonic()

    def update(self, a, label=""):
        a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
        if a.size == 0:
            return
        score = score_with_evaluate(a)
        if np.isfinite(score) and score < self.best_score:
            self.best = a.copy()
            self.best_score = score
            elapsed = time.monotonic() - self.t_start
            print(f"  ** NEW BEST: J={score:.8f} n={a.size} t={elapsed:.0f}s {label}",
                  flush=True)
        return score

    def report(self, label=""):
        if self.best is not None:
            print(f"  {label} best: J={self.best_score:.8f} n={self.best.size}", flush=True)


def resample_seq(src, dest_n):
    """Resample 1D sequence to new length via linear interpolation."""
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)


def propose_candidate(seed=42, budget_s=1500, **kwargs):
    """
    Multi-phase search using LP cutting-plane as the main driver,
    with LP best-response for polishing and smooth gradient for exploration.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + float(budget_s) - 20.0
    tracker = ScoreTracker()

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        if p.ndim == 1 and p.size > 0 and np.all(np.isfinite(p)):
            parent = np.maximum(p, 0.0)
            if parent.sum() > 0.01:
                tracker.update(parent, "(parent)")
                print(f"Parent loaded: n={parent.size}, J={tracker.best_score:.6f}", flush=True)
    except (NameError, AttributeError):
        pass

    if parent is None:
        parent = np.ones(256)
        tracker.update(parent, "(default)")

    print(f"\n{'='*60}", flush=True)
    print(f"TOTAL BUDGET: {budget_s}s, deadline in {deadline - time.monotonic():.0f}s", flush=True)
    print(f"{'='*60}", flush=True)

    # === PHASE 1: Quick smooth gradient initialization (100s) ===
    p1_deadline = time.monotonic() + 100.0
    print(f"\n--- PHASE 1: Smooth Gradient Init (100s) ---", flush=True)

    n_parent = parent.size
    # Try a few lengths around the parent
    init_lengths = sorted(set([128, 192, 256, 384, n_parent, 640, 768]))
    init_lengths = [n for n in init_lengths if 64 <= n <= 1024]

    best_smooth = None
    best_smooth_score = float('inf')

    for n in init_lengths:
        if time.monotonic() > p1_deadline - 10:
            break
        t0 = time.monotonic()
        inits = generate_initializations(n, rng, parent if n == n_parent else None)

        # Quick smooth run on best few inits
        for a0 in inits[:min(6, len(inits))]:
            if time.monotonic() > p1_deadline - 5:
                break
            iters = max(30, min(200, int(8000 / n)))
            a = run_smooth_gradient(a0, iters, p1_deadline - 5,
                                    initial_tau=3.0, final_tau=50.0,
                                    lr_init=0.2, lr_min=0.03)
            score = compute_J(a)
            if score < best_smooth_score:
                best_smooth_score = score
                best_smooth = a.copy()

        elapsed = time.monotonic() - t0
        print(f"  n={n:4d}: best J_smooth={best_smooth_score:.6f} ({elapsed:.1f}s)", flush=True)

    if best_smooth is not None:
        tracker.update(best_smooth, "(smooth)")
    tracker.report("After Phase 1")

    # === PHASE 2: LP Cutting-Plane on best (main optimization, 700s) ===
    if tracker.best is not None:
        p2_deadline = time.monotonic() + 700.0
        print(f"\n--- PHASE 2: LP Cutting-Plane (700s) ---", flush=True)
        print(f"  Starting n={tracker.best.size}, J={tracker.best_score:.6f}", flush=True)

        # Run LP cutting plane on the current best
        a_opt, M_opt = lp_cutting_plane(tracker.best, 5000, p2_deadline, verbose=True)
        tracker.update(a_opt, "(LP-cp)")

        # If we have time, run on a resampled version at a different length
        if time.monotonic() < p2_deadline - 60:
            n2 = tracker.best.size
            trial_n = n2 + 64 if n2 % 128 != 0 else n2 + 128
            if trial_n <= 1536:
                a_trial = resample_seq(a_opt, trial_n)
                a_trial = np.maximum(a_trial, 1e-6)
                a_trial /= a_trial.sum()
                a_t2, M_t2 = lp_cutting_plane(a_trial, 2000, min(time.monotonic() + 120, p2_deadline - 30),
                                              verbose=False)
                tracker.update(a_t2, f"(LP-cp n={trial_n})")

    tracker.report("After Phase 2")

    # === PHASE 3: LP Best-Response polishing (250s) ===
    if tracker.best is not None and time.monotonic() < deadline - 50:
        p3_deadline = time.monotonic() + 250.0
        print(f"\n--- PHASE 3: LP Best-Response Polish (250s) ---", flush=True)
        print(f"  Starting n={tracker.best.size}, J={tracker.best_score:.6f}", flush=True)

        a_br, M_br = lp_best_response_polish(tracker.best, 500, p3_deadline, verbose=True)
        tracker.update(a_br, "(BR)")

        # Another round if improved
        if time.monotonic() < p3_deadline - 30:
            a_br2, M_br2 = lp_best_response_polish(a_br, 200, p3_deadline - 30, verbose=False)
            tracker.update(a_br2, "(BR2)")

    tracker.report("After Phase 3")

    # === PHASE 4: Explore other lengths (150s) ===
    if tracker.best is not None and time.monotonic() < deadline - 30:
        p4_deadline = time.monotonic() + 150.0
        print(f"\n--- PHASE 4: Multi-Length Exploration (150s) ---", flush=True)

        base_n = tracker.best.size
        # Try nearby lengths
        for trial_n in [base_n - 64, base_n + 64, base_n - 128, base_n + 128]:
            if trial_n < 64 or trial_n > 2048:
                continue
            if time.monotonic() > p4_deadline - 20:
                break
            a_ext = resample_seq(tracker.best, trial_n)
            a_ext = np.maximum(a_ext, 1e-6)
            a_ext /= a_ext.sum()

            # Quick smooth + short LP
            a_ext = run_smooth_gradient(a_ext, 100, p4_deadline - 20,
                                        initial_tau=5, final_tau=40,
                                        lr_init=0.15, lr_min=0.03)
            tracker.update(a_ext, f"(explore n={trial_n})")

            if time.monotonic() < p4_deadline - 10:
                a_lp, _ = lp_cutting_plane(a_ext, 200, p4_deadline - 10, verbose=False)
                tracker.update(a_lp, f"(LP n={trial_n})")

    tracker.report("After Phase 4")

    # === PHASE 5: Final intensive LP on best (remaining time) ===
    if tracker.best is not None and time.monotonic() < deadline - 5:
        remaining = deadline - time.monotonic()
        print(f"\n--- PHASE 5: Final LP Polish ({remaining:.0f}s remaining) ---", flush=True)

        # Restart LP from current best with fresh start
        a_final, _ = lp_cutting_plane(tracker.best, 5000, deadline - 5, verbose=False)
        tracker.update(a_final, "(final-LP)")

        # One more BR pass
        if time.monotonic() < deadline - 10:
            a_final2, _ = lp_best_response_polish(a_final, 100, deadline - 5, verbose=False)
            tracker.update(a_final2, "(final-BR)")

    tracker.report("After Phase 5")

    # === PHASE 6: Random restarts if time remains ===
    if tracker.best is not None and time.monotonic() < deadline - 10:
        remaining = deadline - time.monotonic()
        print(f"\n--- PHASE 6: Random Restarts ({remaining:.0f}s) ---", flush=True)

        n_best = tracker.best.size
        for r in range(20):
            if time.monotonic() > deadline - 10:
                break
            noise_scale = 0.03 * (1.0 - 0.03 * r)
            a_r = tracker.best.copy()
            a_r = a_r + rng.normal(0, noise_scale, n_best)
            a_r = np.maximum(a_r, 0.001)
            a_r /= a_r.sum()

            # Quick smooth
            a_r = run_smooth_gradient(a_r, 50, deadline - 10,
                                      initial_tau=5, final_tau=40,
                                      lr_init=0.1, lr_min=0.03)
            score_r = tracker.update(a_r, f"(restart {r})")

    tracker.report("FINAL")

    # Fallback
    if tracker.best is None:
        tracker.best = np.ones(128)
        tracker.best_score = compute_J(tracker.best)

    print(f"\n{'='*60}", flush=True)
    print(f"FINAL ANSWER: J={tracker.best_score:.8f} n={tracker.best.size} "
          f"time={time.monotonic()-tracker.t_start:.0f}s", flush=True)
    print(f"{'='*60}", flush=True)

    return [float(x) for x in tracker.best]
