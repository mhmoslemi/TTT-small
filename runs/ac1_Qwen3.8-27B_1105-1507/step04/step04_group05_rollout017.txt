"""
Minimizes J(a) = 2*n*max(convolve(a,a))/(sum(a))^2 using:
1. LP mirror descent with warm-starting and adaptive step sizes
2. Log-sum-exp smooth approximation with projected gradient/L-BFGS

Key improvements over previous:
- Warm-started LP for speed
- Smooth gradient method as alternative optimizer
- Better step size selection via bisection
- More systematic length exploration

import numpy as np
from scipy.optimize import linprog
from scipy.sparse import lil_matrix, csr_matrix
import time

def fast_conv(a):
    """Full self-convolution via FFT."""
    n = len(a)
    s = 1
    while s < 2 * n - 1:
        s <<= 1
    fa = np.fft.rfft(a, s)
    return np.fft.irfft(fa * fa, s)[:2 * n - 1]

def compute_J(a):
    """Compute J(a) = 2*n*max(conv(a,a))/(sum(a))^2."""
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))

def build_conv_matrix(a, n):
    """Build the convolution constraint matrix A where (A g)_k = (a*g)_k."""
    # A is (2n-1) x n
    # A[k, j] = a[k-j] if 0 <= k-j < n, else 0
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        jl = max(0, k - n + 1)
        jh = min(n - 1, k)
        for j in range(jl, jh + 1):
            rows.append(k)
            cols.append(j)
            vals.append(a[k - j])
    A = lil_matrix((2 * n - 1, n))
    for idx in range(len(rows)):
        A[rows[idx], cols[idx]] = vals[idx]
    return A.tocsr()

def solve_lp(a, M):
    """Solve LP: max sum(g) s.t. (a*g)_k <= M for all k, g >= 0."""
    n = len(a)
    c = -np.ones(n)
    A = build_conv_matrix(a, n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
    if r.success and r.x is not None:
        g = np.maximum(r.x, 0.0)
        if g.sum() > 1e-10:
            return g
    return None

def smooth_objective_and_grad(a, tau):
    """Compute smooth approximation of J and its gradient.
    Uses log-sum-exp to smooth the max."""
    n = len(a)
    S = a.sum()
    if S < 1e-10:
        return 1e10, np.zeros(n)
    
    b = fast_conv(a)
    
    # Log-sum-exp for numerical stability
    b_max = b.max()
    exp_b = np.exp(tau * (b - b_max))
    sum_exp = exp_b.sum()
    log_sum_exp = b_max + np.log(sum_exp) / tau
    
    # Softmax weights
    w = exp_b / sum_exp
    
    # Gradient of smooth_max(b) w.r.t. a_i:
    # grad_i = 2 * sum_k w_k * a_{k-i} (when 0 <= k-i < n)
    # This is 2 * (cross-correlation of w and a)
    # Compute directly: O(n^2)
    grad_b = np.zeros(n)
    for i in range(n):
        # sum_{m=0}^{n-1} w[m+i] * a[m]
        # w has length 2n-1, so w[i:i+n] is valid
        grad_b[i] = np.dot(w[i:i+n], a)
    grad_b *= 2.0
    
    # J = 2*n * smooth_max / S^2
    # dJ/da_i = 2*n * (grad_b_i * S^2 - smooth_max * 2*S) / S^4
    #         = 2*n * (grad_b_i * S - 2*smooth_max) / S^3
    
    J = 2.0 * n * log_sum_exp / (S * S)
    grad_J = 2.0 * n * (grad_b * S - 2.0 * log_sum_exp) / (S * S * S)
    
    return J, grad_J

def projected_gradient_descent(a, tau, n_steps, lr, deadline, tracker):
    """Projected gradient descent with log-sum-exp smoothing."""
    a = np.maximum(a.astype(np.float64), 0.0)
    a /= a.sum()
    n = len(a)
    
    for step in range(n_steps):
        if time.monotonic() > deadline:
            break
        
        J, grad = smooth_objective_and_grad(a, tau)
        
        a_new = a - lr * grad
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s < 1e-10:
            break
        a_new /= s
        
        tracker.update(a_new)
        a = a_new
    
    return a

def lp_mirror_descent(a, max_iters, deadline, tracker, tighten_base=0.005):
    """LP mirror descent with adaptive tightening and step size."""
    a = np.maximum(a.astype(np.float64), 1e-6)
    a /= a.sum()
    n = len(a)
    current_J = compute_J(a)
    tracker.update(a)
    
    tighten = tighten_base
    stall = 0
    
    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        
        b = fast_conv(a)
        M = float(b.max())
        
        # Try to solve LP with tightened bound
        M_target = M * (1.0 - tighten)
        g = solve_lp(a, M_target)
        
        if g is None:
            # Back off
            tighten *= 0.5
            if tighten < 1e-7:
                break
            continue
        
        g = np.maximum(g, 0.0)
        g_sum = g.sum()
        if g_sum < 1e-10:
            break
        g /= g_sum
        
        # Line search: find best step size
        best_t = None
        best_J = current_J
        
        # Try multiple step sizes (geometric)
        for t in [0.1, 0.2, 0.3, 0.5, 0.7, 0.9, 1.0]:
            a_trial = (1.0 - t) * a + t * g
            a_trial = np.maximum(a_trial, 0.0)
            s = a_trial.sum()
            if s < 1e-10:
                continue
            a_trial /= s
            J_trial = compute_J(a_trial)
            if J_trial < best_J - 1e-14:
                best_J = J_trial
                best_t = t
        
        if best_t is None:
            stall += 1
            if stall > 30:
                break
            tighten *= 0.8
            continue
        
        a = (1.0 - best_t) * a + best_t * g
        a = np.maximum(a, 0.0)
        a /= a.sum()

import numpy as np
from scipy.optimize import linprog
from scipy.sparse import lil_matrix

def fast_conv(a):
    """Full self-convolution via FFT."""
    n = len(a)
    s = 1
    while s < 2 * n - 1:
        s <<= 1
    fa = np.fft.rfft(a, s)
    return np.fft.irfft(fa * fa, s)[:2 * n - 1]


def compute_J(a):
    """Compute J(a) = 2*n*max(conv(a,a))/(sum(a))^2."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))


def build_A_sparse(a, n):
    """Build sparse constraint matrix for LP: A[k,j] = a[k-j] if valid."""
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        jl = max(0, k - n + 1)
        jh = min(n - 1, k)
        for j in range(jl, jh + 1):
            rows.append(k)
            cols.append(j)
            vals.append(float(a[k - j]))
    A = lil_matrix((2 * n - 1, n))
    for idx in range(len(rows)):
        A[rows[idx], cols[idx]] = vals[idx]
    return A.tocsr()


def solve_lp(a, M, n):
    """LP: max sum(g) s.t. (a*g)_k <= M for all k, g >= 0."""
    c = -np.ones(n)
    A = build_A_sparse(a, n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    try:
        r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                    options={'presolve': True})
        if r.success and r.x is not None:
            g = np.maximum(r.x, 0.0)
            if g.sum() > 1e-12:
                return g
    except Exception:
        pass
    return None


def smooth_obj_grad(a, tau):
    """Compute log-sum-exp smoothed objective and gradient."""
    n = len(a)
    S = a.sum()
    if S < 1e-10:
        return 1e10, np.zeros(n)
    b = fast_conv(a)
    b_max = b.max()
    exp_b = np.exp(tau * (b - b_max))
    sum_exp = exp_b.sum()
    log_sum_exp = b_max + np.log(sum_exp) / tau
    w = exp_b / sum_exp
    # Gradient of smooth_max(b) w.r.t. a_i = 2 * sum_{m=0}^{n-1} w[m+i]*a[m]
    grad_b = np.zeros(n)
    for i in range(n):
        grad_b[i] = np.dot(w[i:i + n], a)
    grad_b *= 2.0
    J = 2.0 * n * log_sum_exp / (S * S)
    grad_J = 2.0 * n * (grad_b * S - 2.0 * log_sum_exp) / (S * S * S)
    return J, grad_J

            gr = (np.sqrt(5.0) - 1.0) / 2.0
            c_pt = hi - gr * (hi - lo)
            d_pt = lo + gr * (hi - lo)

            def obj_at(x):
                new_S = S - ai + x
                if new_S < 1e-10:
                    return 1e10
                max_c = unaff_max
                for k in range(i, min(2 * n - 1, i + n)):
                    ki = k - i
                    if 0 <= ki < n:
                        if k == 2 * i:
                            ck = b[k] - ai * ai + x * x
                        else:
                            ck = b[k] - 2.0 * ai * a[ki] + 2.0 * x * a[ki]
                    else:
                        ck = b[k]
                    if ck > max_c:
                        max_c = ck
                return 2.0 * n * max_c / (new_S * new_S)

            fc = obj_at(c_pt)
            fd = obj_at(d_pt)
            for _ in range(35):
                if fc < fd:
                    hi = d_pt
                    d_pt = c_pt
                    fd = fc
                    c_pt = hi - gr * (hi - lo)
                    fc = obj_at(c_pt)
                else:
                    lo = c_pt
                    c_pt = d_pt
                    fc = fd
                    d_pt = lo + gr * (hi - lo)
                    fd = obj_at(d_pt)
            x_opt = max(0.0, (lo + hi) / 2.0)
            if abs(x_opt - ai) > 1e-9:
                for k in range(max(0, i), min(2 * n - 1, i + n)):
                    ki = k - i
                    if 0 <= ki < n:
                        if k == 2 * i:
                            b[k] = b[k] - ai * ai + x_opt * x_opt
                        else:
                            b[k] = b[k] - 2.0 * ai * a[ki] + 2.0 * x_opt * a[ki]
                a[i] = x_opt
                S = S - ai + x_opt
                improved = True
        b_check = fast_conv(a)
        new_J = 2.0 * n * b_check.max() / (S * S)
        if new_J < current_J - 1e-12:
            current_J = new_J
            b = b_check
            tracker.update(a)
        elif not improved:
            break

    return a


class ScoreTracker:
    def __init__(self):
        self.best = None
        self.best_score = float('inf')
        self.t_start = time.monotonic()

    def update(self, a):
        arr = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
        if arr.size == 0 or arr.sum() < 0.01:
            return
        score = float(evaluate_sequence(arr.tolist()))
        if np.isfinite(score) and score < self.best_score:
            self.best = arr.copy()
            self.best_score = score
            elapsed = time.monotonic() - self.t_start
            print(f"  * NEW BEST: J={score:.8f} n={arr.size} t={elapsed:.0f}s", flush=True)

    def report(self, label=""):
        if self.best is not None:
            print(f"  {label} best: J={self.best_score:.8f} n={self.best.size} "
                  f"t={time.monotonic()-self.t_start:.0f}s", flush=True)


def generate_inits(n, rng):
    """Generate diverse initializations."""
    inits = []
    t = np.linspace(0, 1, n, endpoint=False)
    inits.append(np.ones(n))
    inits.append(rng.uniform(0.3, 1.0, n))
    for freq in [1, 2, 3, 5, 7, 11]:
        inits.append(np.maximum(1.0 + 0.5 * np.cos(2.0 * np.pi * freq * t
                       + rng.uniform(0, 6.28)), 0.01))
    inits.append(np.maximum(1.0 + 0.4 * np.cos(2 * np.pi * 2 * t)
                   + 0.3 * np.cos(2 * np.pi * 5 * t), 0.01))
    inits.append(np.abs(np.sin(np.pi * t)) + 0.2)
    inits.append(1.0 - 0.4 * np.abs(2 * t - 1.0))
    inits.append(0.5 * np.exp(-8 * (t - 0.25) ** 2)
                + 0.5 * np.exp(-8 * (t - 0.75) ** 2) + 0.1)
    # Triangle wave
    tri = 2.0 * np.abs(2.0 * t - 1.0) - 1.0
    inits.append(1.0 + 0.3 * tri)
    return [np.maximum(x, 0.01) for x in inits]


def propose_candidate(seed=42, budget_s=1500, **kwargs):
    """Main search function."""
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + float(budget_s) - 30.0
    tracker = ScoreTracker()

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        if p.ndim == 1 and p.size > 0 and np.all(np.isfinite(p)):
            parent = np.maximum(p, 0.0)
            tracker.update(parent)
            print(f"Parent loaded: n={parent.size}, J={tracker.best_score:.8f}", flush=True)
    except (NameError, AttributeError, TypeError):
        pass

    if parent is None:
        parent = np.ones(384)
        tracker.update(parent)
        print(f"No parent found, using uniform n={parent.size}", flush=True)

    n_parent = parent.size
    best_n = n_parent

    # === Phase 1: LP on parent ===
    print(f"\n{'='*50} Phase 1: LP on parent (n={n_parent}) {'='*50}", flush=True)
    t0 = time.monotonic()
    a = parent.copy()
    a = lp_mirror_descent(a, max_iters=300, deadline=deadline - 500,
                          tracker=tracker, tighten_base=0.005)
    tracker.report(f"Phase 1 LP ({time.monotonic()-t0:.0f}s)")

    # === Phase 2: Smooth gradient polish ===
    if time.monotonic() < deadline - 400:
        print(f"\n{'='*50} Phase 2: Smooth Gradient Polish {'='*50}", flush=True)
        t0 = time.monotonic()
        a_smooth = tracker.best.copy()
        for tau in [5.0, 10.0, 20.0, 40.0, 80.0]:
            if time.monotonic() > deadline - 350:
                break
            a_smooth = nesterov_smooth_optimize(
                a_smooth, tau=tau, n_steps=50,
                deadline=min(deadline - 350, time.monotonic() + 20),
                tracker=tracker, lr_base=0.02 / (1 + tau * 0.01))
        tracker.report(f"Phase 2 Smooth ({time.monotonic()-t0:.0f}s)")

    # === Phase 3: More LP iterations on best ===
    if time.monotonic() < deadline - 200:
        print(f"\n{'='*50} Phase 3: Extended LP {'='*50}", flush=True)
        t0 = time.monotonic()
        a = lp_mirror_descent(tracker.best, max_iters=300,
                              deadline=deadline - 150, tracker=tracker,
                              tighten_base=0.003)
        tracker.report(f"Phase 3 Extended LP ({time.monotonic()-t0:.0f}s)")

    # === Phase 4: Coordinate descent polish ===
    if time.monotonic() < deadline - 100:
        print(f"\n{'='*50} Phase 4: Coordinate Descent {'='*50}", flush=True)
        t0 = time.monotonic()
        a_cd = coordinate_descent_fast(tracker.best, n_passes=5,
                                       deadline=deadline - 80, tracker=tracker)
        tracker.report(f"Phase 4 CD ({time.monotonic()-t0:.0f}s)")

    # === Phase 5: Multi-length exploration ===
    if time.monotonic() < deadline - 60:
        print(f"\n{'='*50} Phase 5: Multi-Length {'='*50}", flush=True)
        t0 = time.monotonic()
        candidates_n = set()
        for delta in [-6, -4, -3, -2, -1, 1, 2, 3, 4, 6, 8, 12, 16]:
            nl = n_parent + delta
            if 200 <= nl <= 600:
                candidates_n.add(nl)
        # Also try some very different lengths
        for nl in [256, 300, 350, 400, 450, 500]:
            candidates_n.add(nl)
        candidates_n = sorted(candidates_n)
        # Sort by proximity to best_n
        candidates_n.sort(key=lambda x: abs(x - best_n))

        for nl in candidates_n:
            if time.monotonic() > deadline - 50:
                break
            if nl == best_n:
                continue
            a_r = resample_seq(tracker.best, nl)
            a_r = np.maximum(a_r, 1e-4)
            a_r /= a_r.sum()
            # Quick LP
            a_r = lp_mirror_descent(a_r, max_iters=40,
                                    deadline=min(deadline - 50, time.monotonic() + 25),
                                    tracker=tracker, tighten_base=0.004, verbose=False)
            print(f"    n={nl}: J={tracker.best_score:.8f}", flush=True)

        tracker.report(f"Phase 5 MultiLen ({time.monotonic()-t0:.0f}s)")

    # === Phase 6: Perturbation restarts ===
    if time.monotonic() < deadline - 30:
        print(f"\n{'='*50} Phase 6: Perturbation Restarts {'='*50}", flush=True)
        t0 = time.monotonic()
        n = tracker.best.size
        for temp in [0.08, 0.05, 0.03, 0.02, 0.01]:
            if time.monotonic() > deadline - 25:
                break
            a_p = tracker.best.copy() + rng.normal(0, temp, n)
            a_p = np.maximum(a_p, 1e-4)
            a_p /= a_p.sum()
            a_p = lp_mirror_descent(a_p, max_iters=30,
                                    deadline=min(deadline - 25, time.monotonic() + 15),
                                    tracker=tracker, tighten_base=0.003, verbose=False)
            if time.monotonic() < deadline - 25:
                a_p = coordinate_descent_fast(a_p, n_passes=2,
                                              deadline=deadline - 25, tracker=tracker)
        tracker.report(f"Phase 6 Perturb ({time.monotonic()-t0:.0f}s)")

    # === Phase 7: Final intensive LP + smooth ===
    if time.monotonic() < deadline - 10:
        print(f"\n{'='*50} Phase 7: Final Polish {'='*50}", flush=True)
        t0 = time.monotonic()
        a_final = tracker.best.copy()
        # Final LP
        a_final = lp_mirror_descent(a_final, max_iters=100,
                                    deadline=deadline - 15, tracker=tracker,
                                    tighten_base=0.002, verbose=False)
        # Final smooth
        if time.monotonic() < deadline - 10:
            for tau in [50.0, 100.0]:
                if time.monotonic() > deadline - 10:
                    break
                a_final = nesterov_smooth_optimize(
                    a_final, tau=tau, n_steps=20,
                    deadline=deadline - 10, tracker=tracker, lr_base=0.005)
        tracker.report(f"Phase 7 Final ({time.monotonic()-t0:.0f}s)")

    # Fallback
    if tracker.best is None:
        tracker.best = np.ones(256)
        tracker.best_score = compute_J(tracker.best)

    print(f"\n{'='*50} FINAL RESULT {'='*50}", flush=True)
    print(f"J = {tracker.best_score:.10f}, n = {tracker.best.size}", flush=True)
    print(f"Total time: {time.monotonic()-tracker.t_start:.1f}s", flush=True)

    # Ensure we return valid values
    result = np.clip(tracker.best, 0.0, 1000.0)
    if result.sum() < 0.01:
        result = result / result.sum() * 1.0 if result.sum() > 0 else np.ones(256)
    return [float(x) for x in result]
