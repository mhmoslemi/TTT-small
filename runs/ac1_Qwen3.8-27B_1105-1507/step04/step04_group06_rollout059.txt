def fast_conv(a):
    n = len(a)
    s = 1
    while s < 2 * n - 1:
        s <<= 1
    fa = np.fft.rfft(a, s)
    return np.fft.irfft(fa * fa, s)[:2 * n - 1]

def compute_J(a):

    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))

def smooth_max_and_grad(b, beta):
    """Log-sum-exp approximation of max(b) and its gradient w.r.t. b."""
    b_max = b.max()
    shifted = beta * (b - b_max)
    shifted = np.clip(shifted, -100, 100)
    exp_shifted = np.exp(shifted)
    Z = exp_shifted.sum()
    p = exp_shifted / Z
    val = b_max + np.log(Z) / beta
    return val, p

def lbfgs_smooth(a0, beta_schedule, deadline):
    """Run L-BFGS-B with a sequence of beta values."""
    a = np.maximum(a0.astype(np.float64), 1e-10)
    n = a.size
    
    for beta, max_iters in beta_schedule:
        if time.monotonic() > deadline:
            break
        
        def func(x):
            x2 = np.maximum(x, 0.0)
            f, _ = F_and_grad(x2, beta)
            return f if np.isfinite(f) else 1e10
        
        def jac(x):
            x2 = np.maximum(x, 0.0)
            _, g = F_and_grad(x2, beta)
            g = np.nan_to_num(g, nan=0.0, posinf=0.0, neginf=0.0)
            g[x2 < 1e-12] = 0.0
            return g
        
        bounds = [(0.0, None)] * n
        try:
            res = minimize(func, a.copy(), jac=jac, method='L-BFGS-B',
                           bounds=bounds,
                           options={'maxiter': max_iters, 'gtol': 1e-10, 'ftol': 1e-15})
            if res.fun < 1e10:
                new_a = np.maximum(res.x, 1e-12)
                if new_a.sum() > 1e-6:
                    a = new_a
        except Exception:
            pass
    
    return a

def coordinate_descent_polish(a, deadline, max_rounds=100):
    """Coordinate descent: for each position, find optimal value given others fixed."""
    a = np.maximum(a.astype(np.float64), 0.0)
    n = a.size
    s = a.sum()
    if s < 1e-10:
        return a
    a = a / s
    
    best_J = compute_J(a)
    
    for rnd in range(max_rounds):
        if time.monotonic() > deadline:
            break
        improved = False
        for i in range(n):
            if time.monotonic() > deadline:
                break
            # Try a few values for a[i]
            old_val = a[i]
            if old_val < 1e-8:
                continue
            # The optimal a[i] given others: we want to minimize max(C)
            # C_k depends on a[i] as: C_k = (other terms) + 2*a[i]*a[k-i] + a[i]^2 (if k=2i)
            # For simplicity, try a few candidates
            for scale in [0.5, 0.7, 0.85, 1.15, 1.3, 1.5]:
                a[i] = old_val * scale
                s_new = a.sum()
                if s_new < 0.01:
                    a[i] = old_val
                    continue
                a_temp = a / s_new
                J_try = compute_J(a_temp)
                if J_try < best_J - 1e-12:
                    a = a_temp
                    best_J = J_try
                    improved = True
                    break
            else:
                a[i] = old_val
                # Re-normalize
                a = a / a.sum()
        
        if not improved:
            # Try adding small values to zero positions
            zero_idx = np.where(a < 1e-8)[0]
            if len(zero_idx) > 0:
                for idx in zero_idx[:20]:
                    a[idx] = a.max() * 0.1
                    a = a / a.sum()
                    J_try = compute_J(a)
                    if J_try < best_J:
                        best_J = J_try
                        improved = True
                    else:
                        a[idx] = 0.0
                        a = a / a.sum()
            if not improved:
                break
    
    return a

def fourier_init(n, rng):
    """Generate a non-negative sequence with approximately flat magnitude spectrum."""
    # Target flat spectrum
    freqs = np.fft.rfftfreq(n)
    # Random phases
    phases = rng.uniform(0, 2*np.pi, len(freqs))
    # Flat magnitude (with DC component to ensure non-negativity)
    mags = np.ones(len(freqs))
    mags[0] = n * 0.5  # Large DC component for non-negativity
    spectrum = mags * np.exp(1j * phases)
    a = np.fft.irfft(spectrum, n)
    a = np.maximum(a, 0.0)
    if a.sum() < 1e-6:
        a = np.ones(n)
    return a / a.sum()

def resample_seq(src, dest_n):
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    result = np.interp(t_new, t_old, src)
    return np.maximum(result, 0.0)

def fast_conv(a):

    a = a / s
    best_J = compute_J(a)
    
    multipliers = [0.3, 0.5, 0.7, 0.85, 1.15, 1.3, 1.5, 2.0]
    
    for rnd in range(max_rounds):
        if time.monotonic() > deadline:
            break
        improved = False
        for i in range(n):
            if time.monotonic() > deadline:
                break
            old_val = a[i]
            if old_val < 1e-9:
                # Try adding a small value
                a[i] = a.max() * 0.05
                a = a / a.sum()
                J_try = compute_J(a)
                if J_try < best_J - 1e-13:
                    best_J = J_try
                    improved = True
                else:
                    a[i] = 0.0
                    a = a / a.sum()
                continue
            
            best_mult = 1.0
            best_local_J = best_J
            
            for mult in multipliers:
                a[i] = old_val * mult
                a_sum = a.sum()
                if a_sum < 1e-10:
                    a[i] = old_val
                    continue
                a_temp = a / a_sum
                J_try = compute_J(a_temp)
                if J_try < best_local_J:
                    best_local_J = J_try
                    best_mult = mult
            
            if best_mult != 1.0:
                a[i] = old_val * best_mult
                a = a / a.sum()
                best_J = best_local_J
                improved = True
            else:
                a[i] = old_val
        
        if not improved:
            break
    
    return a


def fourier_init(n, rng):
    """Generate non-negative sequence with approximately flat magnitude spectrum."""
    phases = rng.uniform(0, 2 * np.pi, n // 2 + 1)
    mags = np.ones(n // 2 + 1)
    mags[0] = np.sqrt(n) * 0.8  # Large DC for non-negativity
    # Taper high frequencies for smoothness
    for f in range(len(mags)):
        mags[f] *= np.exp(-0.5 * (f / (n // 4)) ** 2)
    mags[0] = np.sqrt(n)  # Restore DC
    spectrum = mags * np.exp(1j * phases)
    a = np.fft.irfft(spectrum, n)
    a = np.maximum(a, 0.0)
    if a.sum() < 1e-6:
        a = np.ones(n)
    return a / a.sum()


def generate_structured_init(n, rng, parent=None):
    """Generate diverse structured initializations."""
    inits = []
    t = np.linspace(0, 1, n, endpoint=False)
    
    if parent is not None:
        p = _resample(parent, n)
        inits.append(p)
        p2 = p + rng.normal(0, 0.05, n) * p
        inits.append(np.maximum(p2, 0.01))
    
    inits.append(np.ones(n))
    inits.append(fourier_init(n, rng))
    
    # Multi-frequency cosine
    for freq in [3, 5, 7, 11]:
        inits.append(1.0 + 0.3 * np.cos(2 * np.pi * freq * t + rng.uniform(0, 2 * np.pi)))
    
    # Gaussian mixture
    n_peaks = rng.integers(2, 6)
    centers = np.sort(rng.uniform(0, 1, n_peaks))
    widths = rng.uniform(0.02, 0.1, n_peaks)
    heights = rng.uniform(0.3, 1.0, n_peaks)
    a = np.zeros(n)
    for c, w, h in zip(centers, widths, heights):
        a += h * np.exp(-0.5 * ((t - c) / w) ** 2)
    a += 0.05
    inits.append(a)
    
    # Exponential decay from random point
    center = rng.uniform(0.2, 0.8)
    decay = rng.uniform(3, 10)
    a = np.exp(-decay * np.abs(t - center)) + 0.1
    inits.append(a)
    
    # Watertight: ensure all are valid
    valid_inits = []
    for a in inits:
        a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
        if a.sum() > 0.01:
            valid_inits.append(a / a.sum())
    
    return valid_inits if valid_inits else [np.ones(n) / n]


def _resample(src, dest_n):
    """Resample 1D sequence to new length."""
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    result = np.interp(t_new, t_old, src)
    return np.maximum(result, 0.0)


# Global deadline for subgradient
deadline_global = float('inf')


class Tracker:
    def __init__(self):
        self.best = None
        self.best_score = float('inf')
        self.t_start = time.monotonic()
    
    def update(self, a):
        global deadline_global
        a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
        if a.size == 0 or a.sum() < 0.01:
            return
        score = float(evaluate_sequence(a.tolist()))
        if np.isfinite(score) and score < self.best_score:
            self.best = a.copy()
            self.best_score = score
            elapsed = time.monotonic() - self.t_start
            print(f"  * NEW BEST: J={score:.8f} n={a.size} t={elapsed:.1f}s", flush=True)
    
    def report(self, msg=""):
        if self.best is not None:
            print(f"  {msg} Best: J={self.best_score:.8f} n={self.best.size}", flush=True)


def propose_candidate(seed=42, budget_s=1500, **kwargs):
    """
    Multi-phase optimization for minimizing J(a) = 2n*max(conv(a,a))/(sum(a))^2.
    
    Phase 1: Multi-scale L-BFGS-B (128->256->512->1024)
    Phase 2: Aggressive high-beta L-BFGS-B at target length
    Phase 3: Active-set nudging + subgradient steps
    Phase 4: Multi-restart with diverse perturbations
    Phase 5: Final coordinate descent polish
    """
    global deadline_global
    
    rng = np.random.default_rng(seed)
    total_deadline = time.monotonic() + float(budget_s) - 10.0
    deadline_global = total_deadline
    tracker = Tracker()
    n_target = 1024
    
    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        if p.ndim == 1 and p.size > 0 and np.all(np.isfinite(p)):
            parent = np.maximum(p, 0.0)
            tracker.update(parent)
            print(f"Parent loaded: n={parent.size}, J={tracker.best_score:.8f}", flush=True)
    except (NameError, AttributeError):
        pass
    
    if parent is None:
        parent = np.ones(n_target)
        tracker.update(parent)
        print(f"Default parent: n={n_target}, J={tracker.best_score:.8f}", flush=True)
    
    # === Phase 1: Multi-scale L-BFGS-B ===
    p1_deadline = time.monotonic() + 250.0
    print(f"\n{'='*50}", flush=True)
    print(f"Phase 1: Multi-scale L-BFGS-B (budget 250s)", flush=True)
    print(f"{'='*50}", flush=True)
    
    scales = [128, 256, 512, n_target]
    current_best_n = None
    
    for n_scale in scales:
        if time.monotonic() > p1_deadline:
            break
        t0 = time.monotonic()
        print(f"  n={n_scale}...", flush=True)
        
        inits = generate_structured_init(n_scale, rng, parent)
        max_inits = min(len(inits), 4)
        
        for a0 in inits[:max_inits]:
            if time.monotonic() > p1_deadline:
                break
            # Moderate beta schedule for basin finding
            beta_sched = [(1.0, 40), (5.0, 50), (20.0, 60), (50.0, 80)]
            a_opt = lbfgs_smooth(a0, beta_sched, p1_deadline)
            if a_opt.sum() > 0.01:
                tracker.update(a_opt)
        
        # Keep track of best at this scale for upscaling
        if tracker.best is not None:
            current_best_n = tracker.best.size
        
        print(f"    n={n_scale} done in {time.monotonic()-t0:.1f}s", flush=True)
    
    tracker.report("Phase 1 complete")
    
    # === Phase 2: Aggressive L-BFGS-B at target length ===
    p2_deadline = time.monotonic() + 450.0
    print(f"\n{'='*50}", flush=True)
    print(f"Phase 2: High-beta L-BFGS-B n={n_target} (budget 450s)", flush=True)
    print(f"{'='*50}", flush=True)
    
    if tracker.best is not None:
        a_base = _resample(tracker.best, n_target)
        a_base = np.maximum(a_base, 1e-8)
        a_base = a_base / a_base.sum()
    else:
        a_base = np.ones(n_target)
    
    # High beta schedule for precise optimization
    beta_sched_high = [(100.0, 100), (300.0, 150), (500.0, 200), (1000.0, 250)]
    a_opt = lbfgs_smooth(a_base, beta_sched_high, p2_deadline)
    tracker.update(a_opt)
    
    # Second pass with even higher beta
    if time.monotonic() < p2_deadline:
        beta_sched_vhigh = [(500.0, 150), (1000.0, 250), (2000.0, 300)]
        a_opt2 = lbfgs_smooth(a_opt, beta_sched_vhigh, p2_deadline)
        tracker.update(a_opt2)
    
    # Subgradient refinement
    if time.monotonic() < p2_deadline:
        a_sg = subgradient_step(tracker.best, n_steps=2000, eta_scale=0.5)
        tracker.update(a_sg)
        
        # Re-optimize after subgradient
        if time.monotonic() < p2_deadline:
            beta_sched_after = [(200.0, 100), (500.0, 150)]
            a_opt3 = lbfgs_smooth(a_sg, beta_sched_after, p2_deadline)
            tracker.update(a_opt3)
    
    tracker.report("Phase 2 complete")
    
    # === Phase 3: Active-set nudging ===
    p3_deadline = time.monotonic() + 200.0
    print(f"\n{'='*50}", flush=True)
    print(f"Phase 3: Active-set nudging (budget 200s)", flush=True)
    print(f"{'='*50}", flush=True)
    
    if tracker.best is not None:
        a_nudged = active_set_nudge(tracker.best, p3_deadline, n_rounds=20)
        tracker.update(a_nudged)
        
        if time.monotonic() < p3_deadline:
            beta_sched_nudge = [(200.0, 100), (500.0, 150), (1000.0, 200)]
            a_opt4 = lbfgs_smooth(a_nudged, beta_sched_nudge, p3_deadline)
            tracker.update(a_opt4)
        
        # More subgradient steps
        if time.monotonic() < p3_deadline:
            a_sg2 = subgradient_step(tracker.best, n_steps=5000, eta_scale=0.3)
            tracker.update(a_sg2)
            if time.monotonic() < p3_deadline:
                a_opt5 = lbfgs_smooth(a_sg2, [(500.0, 150)], p3_deadline)
                tracker.update(a_opt5)
    
    tracker.report("Phase 3 complete")
    
    # === Phase 4: Multi-restart ===
    p4_deadline = total_deadline
    remaining = p4_deadline - time.monotonic()
    print(f"\n{'='*50}", flush=True)
    print(f"Phase 4: Multi-restart (budget {remaining:.0f}s)", flush=True)
    print(f"{'='*50}", flush=True)
    
    if tracker.best is not None and remaining > 30:
        base = tracker.best.copy()
        n = base.size
        
        restart_configs = [
            ('gauss', 0.03),
            ('gauss', 0.08),
            ('gauss', 0.15),
            ('fourier_blend', 0.5),
            ('gauss', 0.02),
            ('spike', 16),
            ('gauss', 0.05),
            ('gauss', 0.2),
            ('fourier_blend', 0.3),
            ('gauss', 0.01),
            ('spike', 8),
            ('gauss', 0.1),
        ]
        
        for rtype, rparam in restart_configs:
            if time.monotonic() > p4_deadline:
                break
            
            a = base.copy()
            if rtype == 'gauss':
                a = a + rng.normal(0, rparam, n) * a
                a = np.maximum(a, 0.0)
            elif rtype == 'fourier_blend':
                a_f = fourier_init(n, rng)
                a = rparam * a_f + (1 - rparam) * base
                a = np.maximum(a, 0.0)
            elif rtype == 'spike':
                k = min(rparam, n // 8)
                idx = rng.choice(n, size=k, replace=False)
                a[idx] *= rng.uniform(0.3, 3.0, k)
            
            a_sum = a.sum()
            if a_sum < 0.01:
                continue
            a = a / a_sum
            tracker.update(a)
            
            # Short optimization
            if time.monotonic() < p4_deadline:
                budget = min(35.0, (p4_deadline - time.monotonic()) * 0.25)
                sub_dl = time.monotonic() + budget
                beta_s = [(100.0, 60), (300.0, 80), (500.0, 100)]
                a_opt = lbfgs_smooth(a, beta_s, sub_dl)
                tracker.update(a_opt)
        
        tracker.report("Phase 4 complete")
    
    # === Phase 5: Final polish ===
    if tracker.best is not None and time.monotonic() < total_deadline:
        print(f"\n{'='*50}", flush=True)
        print(f"Phase 5: Final polish", flush=True)
        print(f"{'='*50}", flush=True)
        
        a_final = tracker.best.copy()
        
        # High-beta L-BFGS
        if time.monotonic() < total_deadline:
            beta_final = [(500.0, 200), (1000.0, 300)]
            a_final = lbfgs_smooth(a_final, beta_final, total_deadline)
            tracker.update(a_final)
        
        # Coordinate descent
        if time.monotonic() < total_deadline:
            a_cd = coordinate_descent_polish(tracker.best, total_deadline, max_rounds=20)
            tracker.update(a_cd)
        
        # One more L-BFGS if time
        if time.monotonic() < total_deadline - 5:
            beta_last = [(1000.0, 200)]
            a_last = lbfgs_smooth(tracker.best, beta_last, total_deadline)
            tracker.update(a_last)
        
        tracker.report("Phase 5 complete")
    
    # Fallback
    if tracker.best is None:
        tracker.best = np.ones(n_target)
        tracker.best_score = compute_J(tracker.best)
        print(f"  Fallback: J={tracker.best_score:.8f}", flush=True)
    
    print(f"\n{'='*50}", flush=True)
    print(f"FINAL RESULT: J={tracker.best_score:.8f} n={tracker.best.size}", flush=True)
    print(f"{'='*50}", flush=True)
    
    result = np.clip(tracker.best, 0.0, 1000.0)
    if result.sum() < 0.01:
        result = np.ones(128)
    
    return [float(x) for x in result]
