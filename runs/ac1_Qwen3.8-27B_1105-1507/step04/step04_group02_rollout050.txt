```python
import numpy as np
from scipy.optimize import linprog
from scipy.sparse import coo_matrix
import time

def compute_conv(a):
    """Compute full linear convolution a*a via FFT."""
    n = len(a)
    size = 1
    while size < 2*n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    conv = np.fft.irfft(fa * fa, size)
    return conv[:2*n - 1]

def compute_obj(a):
    """Compute 2*n*max_conv/sum^2."""
    n = len(a)
    S = np.sum(a)
    if S < 1e-10:
        return np.inf
    conv = compute_conv(a)
    M = np.max(conv)
    return 2.0 * n * M / (S*S)

def build_toeplitz(a):
    """Build sparse matrix T such that T @ g = conv(a, g)."""
    n = len(a)
    rows, cols, vals = [], [], []
    for j in range(n):
        for k in range(2*n - 1):
            i = k - j
            if 0 <= i < n:
                rows.append(k)
                cols.append(j)
                vals.append(a[i])
    return coo_matrix((vals, (rows, cols)), shape=(2*n-1, n)).tocsr()
```

```python
def compute_conv(a):
    """Compute full linear convolution a*a via FFT."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    conv = np.fft.irfft(fa * fa, size)
    return conv[:2 * n - 1]

    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    S = float(np.sum(a))

    M = float(np.max(conv))

    """Build sparse Toeplitz matrix T such that T @ g = convolve(a, g)."""
    n = len(a)
    rows = []
    cols = []
    vals = []
    for j in range(n):
        i_min = max(0, j - (n - 1))
        i_max = min(n - 1, j + (n - 1))
        for i in range(i_min, i_max + 1):
            k = j + i
            rows.append(k)
            cols.append(j)
            vals.append(a[i])
    return coo_matrix((vals, (rows, cols)), shape=(2 * n - 1, n)).tocsr()

            if stall > 8:
                # Simulated annealing perturbation
                rng = np.random.default_rng(it * 7919 + n)
                scale = min(0.1, 0.02 + 0.01 * stall)
                a = a * np.exp(rng.normal(0, scale, n))
                a = np.maximum(a, 0)
                S_p = np.sum(a)
                if S_p < 1e-10:
                    a = np.ones(n) / n
                else:
                    a = a / S_p
                warm_g = None
                stall = 0
                m_offset = 0.0
            elif warm_g is not None:
                a = 0.95 * a + 0.05 * warm_g
                a = np.maximum(a, 0)
                a = a / np.sum(a)

    return best_a, best_obj

    while time.monotonic() - start < time_budget:
        pass_count += 1
        improved = False

        # Random order for diversity
        rng = np.random.default_rng(pass_count * 1000 + n)
        order = rng.permutation(n)

        for i in order:
            if time.monotonic() - start > time_budget:
                break

            x_old = a[i]
            # Compute E: conv with a[i] terms removed
            conv = compute_conv(a)
            E = conv.copy()
            # Remove a[i] contributions
            for k in range(2 * n - 1):
                j = k - i
                if 0 <= j < n:
                    if k == 2 * i:
                        E[k] -= x_old * x_old
                    else:
                        E[k] -= 2.0 * x_old * a[j]

            # Golden section search for optimal x >= 0
            # conv[k](x) = E[k] + 2*x*a[k-i] (linear mostly) or E[k] + x^2

            lo = 0.0
            hi = max(3.0 * x_old + 0.5 / n, 2.0 / n)

            def objective(x):
                mx = 0.0
                for k in range(2 * n - 1):
                    j = k - i
                    if 0 <= j < n:
                        if k == 2 * i:
                            val = E[k] + x * x
                        else:
                            val = E[k] + 2.0 * x * a[j]
                        if val > mx:
                            mx = val
                return mx

            c_pt = hi - gr * (hi - lo)
            d_pt = lo + gr * (hi - lo)
            fc = objective(c_pt)
            fd = objective(d_pt)
            for _ in range(60):
                if hi - lo < 1e-14:
                    break
                if fc < fd:
                    hi = d_pt
                    d_pt = c_pt
                    fd = fc
                    c_pt = hi - gr * (hi - lo)
                    fc = objective(c_pt)
                else:
                    lo = c_pt
                    c_pt = d_pt
                    fc = fd
                    d_pt = lo + gr * (hi - lo)
                    fd = objective(d_pt)

            x_new = (lo + hi) / 2.0
            a[i] = x_new
            S_new = np.sum(a)
            a = a / S_new

            obj_new = compute_obj(a)
            if obj_new < best_obj:
                best_obj = obj_new
                best_a = a.copy()
                improved = True

        if verbose and pass_count % 10 == 0:
            print(f"    CD pass={pass_count}: obj={best_obj:.6f}")

        if not improved and pass_count > 3:
            # Perturb to escape
            rng = np.random.default_rng(pass_count * 32749 + 7)
            a = a * np.exp(rng.normal(0, 0.03, n))
            a = np.maximum(a, 0)
            S_p = np.sum(a)
            if S_p > 1e-10:
                a = a / S_p

    return best_a, best_obj


def subgradient_polish(a, time_budget, topk=8):
    """Fast subgradient descent for final polishing."""
    n = len(a)
    a = a.copy().astype(np.float64)
    S = np.sum(a)
    a = a / S

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    v = np.zeros(n)
    momentum = 0.8
    step = 0.08

    while time.monotonic() - start < time_budget:
        conv = compute_conv(a)
        M = float(np.max(conv))
        active = np.argsort(conv)[-topk:]

        grad = np.zeros(n)
        for k in active:
            j_min = max(0, k - (n - 1))
            j_max = min(n - 1, k)
            for j in range(j_min, j_max + 1):
                grad[j] += 2.0 * a[k - j]
        grad /= len(active)

        # Center to preserve sum
        grad_centered = grad - np.mean(grad)
        v = momentum * v + (1.0 - momentum) * grad_centered

        a_new = a - step * v
        a_new = np.maximum(a_new, 0)
        S_new = np.sum(a_new)
        if S_new < 1e-10:
            break
        a_new = a_new / S_new

        obj_new = compute_obj(a_new)
        if obj_new < best_obj:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new
        else:
            step *= 0.85
            if step < 1e-10:
                break

    return best_a, best_obj


def cosine_init(n, n_modes=10, seed=0):
    """Cosine series initialization."""
    rng = np.random.default_rng(seed)
    k = np.arange(n)
    a = np.ones(n)
    for m in range(1, n_modes + 1):
        c = rng.uniform(-0.5, 0.5) / m
        phase = rng.uniform(0, 2 * np.pi)
        a += c * np.cos(2 * np.pi * m * k / n + phase)
    a = np.maximum(a, 1e-10)
    S = np.sum(a)
    return a / S
```
