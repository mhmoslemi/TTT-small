"""
Peak-targeting LP + coordinate descent + adaptive subgradient for minimizing
2n*max(conv(a,a))/(sum(a))^2. Key improvements: tighter peak constraints in LP,
fast coordinate descent polish, and simulated annealing for escape from local minima.
"""

import numpy as np
from scipy.optimize import linprog
from scipy.fft import rfft, irfft
import time


def compute_conv(a):
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    conv = irfft(fa * fa, size)
    return conv[:2 * n - 1]


def compute_obj(a):
    n = len(a)
    if n == 0:
        return np.inf
    conv = compute_conv(a)
    max_b = float(np.max(conv))
    sum_a = float(np.sum(a))
    if sum_a < 0.01:
        return np.inf
    return 2.0 * n * max_b / (sum_a * sum_a)


def fft_convolve(a, b):
    n, m = len(a), len(b)
    size = 1
    while size < n + m - 1:
        size <<= 1
    fa = rfft(a, size)
    fb = rfft(b, size)
    return irfft(fa * fb, size)[:n + m - 1]


def build_conv_matrix(a):
    """Build matrix A such that A @ g = conv(a, g). A has shape (2n-1, n)."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        A[j:j + n, j] = a
    return A


def lp_solve(a, M, peak_frac, tight_factor, time_limit):
    """
    LP: max sum(g) s.t. conv(a,g)[k] <= M * tight_factor for peaks, 
                              conv(a,g)[k] <= M for others, g >= 0
    """
    n = len(a)
    A = build_conv_matrix(a)
    
    conv_aa = compute_conv(a)
    if peak_frac > 0:
        n_peaks = max(1, int(peak_frac * (2 * n - 1)))
        threshold = np.partition(conv_aa, (2*n-1) - n_peaks)[-n_peaks]
        b_ub = np.full(2 * n - 1, M)
        b_ub[conv_aa >= threshold] = M * tight_factor
    else:
        b_ub = np.full(2 * n - 1, M)
    
    c = -np.ones(n)
    bounds = [(0.0, None)] * n
    
    try:
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                     options={'presolve': True, 'time_limit': time_limit})
        if res.status == 0:
            g = np.maximum(res.x, 0)
            s = float(np.sum(g))
            if s > 1e-10:
                return g / s
    except Exception:
        pass
    return None


def lp_minpeak(a, time_limit=15):
    """LP: min M s.t. conv(a,g)[k] <= M, sum(g)=1, g>=0."""
    n = len(a)
    A = build_conv_matrix(a)
    
    c = np.zeros(n + 1)
    c[-1] = 1.0
    A_ub = np.zeros((2 * n - 1, n + 1))
    A_ub[:, :n] = A
    A_ub[:, n] = -1.0
    b_ub = np.zeros(2 * n - 1)
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])
    bounds = [(0.0, None)] * n + [(0.0, None)]
    
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                     bounds=bounds, method='highs',
                     options={'presolve': True, 'time_limit': time_limit})
        if res.status == 0:
            g = np.maximum(res.x[:n], 0)
            return g, float(res.x[n])
    except Exception:
        pass
    return None, None

def subgradient_step(a, tau, rng):
    """One subgradient step with temperature tau."""
    n = len(a)
    conv = compute_conv(a)
    max_conv = float(np.max(conv))
    
    # Soft-max weights over convolution
    shifted = (conv - max_conv) / tau
    shifted = np.clip(shifted, -500, 500)
    weights = np.exp(shifted)
    w_sum = np.sum(weights)
    if w_sum < 1e-300:
        idx = int(np.argmax(conv))
        weights = np.zeros(len(conv))
        weights[idx] = 1.0
    else:
        weights /= w_sum
    
    # Gradient: grad_a[i] = 2 * sum_j weights[i+j] * a[j]
    # = 2 * conv(a, weights_reversed)[2n-2-i]
    w_rev = weights[::-1]
    conv_fw = fft_convolve(a, w_rev)
    grad_a = 2.0 * conv_fw[n - 1:2 * n - 1][::-1]
    
    return grad_a, conv, max_conv


def subgradient_optimize(a, time_budget=20.0, tau_start=1.0, tau_end=50.0, learning_rate=0.1):
    """Subgradient optimization with annealing."""
    n = len(a)
    a = a.copy().astype(np.float64)
    s = np.sum(a)
    if s < 1e-10:
        return a
    a = a / s
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    rng = np.random.default_rng(12345)
    step = learning_rate
    it = 0
    max_iters = int(time_budget * 100)
    
    for it in range(max_iters):
        if time.monotonic() - start > time_budget:
            break
        
        frac = it / max(1, max_iters)
        tau = tau_start + (tau_end - tau_start) * frac ** 0.5
        step = learning_rate * (1 - 0.5 * frac)
        
        grad_a, conv, max_conv = subgradient_step(a, tau, rng)
        
        # Normalize gradient
        grad_norm = np.linalg.norm(grad_a)
        if grad_norm > 1e-10:
            grad_a = grad_a / grad_norm
        
        # Projected gradient step (stay non-negative, maintain sum)
        a_new = a - step * grad_a
        a_new = np.maximum(a_new, 0)
        s_new = np.sum(a_new)
        if s_new > 1e-10:
            a_new = a_new / s_new
        else:
            a_new = a.copy()
        
        a = a_new
        obj = compute_obj(a)
        
        if obj < best_obj - 1e-12:
            best_obj = obj
            best_a = a.copy()
        elif it % 100 == 0 and it > 0:
            # Occasionally reset to best
            if obj > best_obj * 1.01:
                a = best_a.copy()
    
    return best_a


def alternating_lp_peaktarget(a, max_iters=50, time_budget=60.0, peak_fracs=None, tight_factors=None):

def random_init(n, seed):
    """Diverse random initialization."""
    rng = np.random.default_rng(seed)
    k = np.arange(n)
    choice = rng.integers(0, 12)
    
    if choice == 0:
        f = rng.uniform(0.5, 4.0)
        phase = rng.uniform(0, 2 * np.pi)
        a = 1.0 + 0.8 * np.cos(2 * np.pi * f * k / n + phase)
    elif choice == 1:
        f1 = rng.uniform(0.3, 2.0)
        f2 = rng.uniform(2.0, 6.0)
        a = 1.0 + 0.4 * np.cos(2 * np.pi * f1 * k / n) + 0.3 * np.cos(2 * np.pi * f2 * k / n)
    elif choice == 2:
        center = n * rng.uniform(0.3, 0.7)
        alpha = rng.uniform(0.5, 3.0)
        a = 1.0 + 0.5 * np.exp(-alpha * np.abs(k - center) / (n * 0.3))
    elif choice == 3:
        a = rng.random(n)
        win = max(3, n // 20)
        kernel = np.ones(win) / win
        a = np.convolve(a, kernel, mode='same')
    elif choice == 4:
        a = np.ones(n)
        for _ in range(rng.integers(1, 5)):
            f = rng.uniform(0.5, 5.0)
            amp = rng.uniform(0.1, 0.4)
            a += amp * np.cos(2 * np.pi * f * k / n + rng.uniform(0, 2 * np.pi))
    elif choice == 5:
        f = rng.uniform(0.3, 2.0)
        a = 1.0 + 0.5 * np.sin(2 * np.pi * f * k / n)
    elif choice == 6:
        nblocks = rng.integers(2, 7)
        a = np.ones(n)
        for b in range(nblocks):
            start = b * n // nblocks
            end = (b + 1) * n // nblocks
            a[start:end] = rng.uniform(0.3, 1.5)
    elif choice == 7:
        f0 = rng.uniform(0.5, 2.0)
        f1 = rng.uniform(2.0, 6.0)
        a = 1.0 + 0.5 * np.cos(2 * np.pi * (f0 * k / n + (f1 - f0) * k**2 / (2 * n**2)))
    elif choice == 8:
        center = n * rng.uniform(0.2, 0.8)
        width = n * rng.uniform(0.1, 0.4)
        a = 1.0 + 0.6 * np.exp(-0.5 * ((k - center) / width) ** 2)
    elif choice == 9:
        # Binary-like (Barker-inspired)
        a = rng.choice([0.3, 1.0], size=n)
        win = max(3, n // 15)
        kernel = np.ones(win) / win
        a = np.convolve(a, kernel, mode='same')
    elif choice == 10:

    else:
        a = rng.random(n) + 0.1
    
    a = np.maximum(a, 0.01)
    s = np.sum(a)
    return a / s


def resample_sequence(a, new_n):
    n = len(a)
    if n == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return a_new


def softmax(z):
    z = z - np.max(z)
    e = np.exp(z)
    return e / np.sum(e)


def soft_max_obj_and_grad(z, tau):
    n = len(z)
    a = softmax(z)
    conv = compute_conv(a)
    cmax = float(np.max(conv))
    shifted = np.clip((conv - cmax) / tau, -500.0, 500.0)
    exp_conv = np.exp(shifted)
    Z = float(np.sum(exp_conv))
    soft_max_val = cmax + tau * np.log(Z)
    w = exp_conv / Z
    
    w_rev = w[::-1]
    conv_fw = fft_convolve(a, w_rev)
    f1 = conv_fw[n - 1:2 * n - 1][::-1]
    grad_a = 2.0 * f1
    
    mean_grad = float(np.dot(a, grad_a))
    grad_z = a * (grad_a - mean_grad)
    
    return soft_max_val, grad_z


def optimize_lfbs(a_init, tau_list, max_iter=200, time_budget=30.0):
    from scipy.optimize import minimize as sp_minimize
    
    n = len(a_init)
    a = a_init.copy()
    s = np.sum(a)
    if s < 1e-10:
        a = np.ones(n) / n
    else:
        a = a / s
    a = np.maximum(a, 1e-30)
    z = np.log(a)
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    
    for tau in tau_list:
        if time.monotonic() - start > time_budget:
            break
        
        def func_and_grad(z_val):
            return soft_max_obj_and_grad(z_val, tau)
        
        try:
            result = sp_minimize(func_and_grad, z, jac=True, method='L-BFGS-B',
                                options={'maxiter': max_iter, 'ftol': 1e-14, 'gtol': 1e-10,
                                         'maxfun': 500})
            z_new = result.x
        except Exception:
            continue
        
        a_new = softmax(z_new)
        a_new = np.maximum(a_new, 0)
        s_new = np.sum(a_new)
        if s_new < 1e-10:
            continue
        a_new = a_new / s_new
        obj = compute_obj(a_new)
        
        if obj < best_obj:
            best_obj = obj
            best_a = a_new.copy()
            z = z_new
    
    return best_a, best_obj
