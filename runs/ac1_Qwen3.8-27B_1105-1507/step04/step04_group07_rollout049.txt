import numpy as np
from scipy.optimize import linprog
from scipy.sparse import csr_matrix

# Precomputed sparsity cache
_spar_cache = {}

def get_sparsity(n):
    if n not in _spar_cache:
        rows = []
        cols = []
        for k in range(2 * n - 1):
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            rows.extend([k] * (je - js + 1))
            cols.extend(range(js, je + 1))
        _spar_cache[n] = (np.array(rows, dtype=np.int32), np.array(cols, dtype=np.int32), len(rows))
    return _spar_cache[n]

def fast_conv(a):
    n = len(a)
    s = 1
    while s < 2 * n - 1:
        s <<= 1
    fa = np.fft.rfft(a, s)
    return np.fft.irfft(fa * fa, s)[:2 * n - 1]

def compute_J(a):
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))

def lp_best_response(a, M, time_limit=5):
    n = len(a)
    rows, cols, nnz = get_sparsity(n)
    vals = np.empty(nnz)
    for i in range(nnz):
        k = rows[i]
        j = cols[i]
        vals[i] = a[k - j]
    A = csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                options={'presolve': True, 'time_limit': time_limit})
    if r.success and r.x is not None:
        g = np.maximum(r.x, 0.0)
        if g.sum() > 1e-10:
            return g
    return None

def lse_adam(a, max_iters, deadline, tracker):
    """LSE gradient descent with Adam optimizer."""
    a = np.maximum(a.astype(np.float64), 1e-6)
    a /= a.sum()
    n = len(a)
    current_J = compute_J(a)
    tracker.update(a)

    tau = 30.0
    lr = 0.5 / n
    beta1, beta2, eps = 0.9, 0.999, 1e-8
    m = np.zeros(n)
    v = np.zeros(n)
    t_step = 0
    stalled = 0

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        b = fast_conv(a)
        b_max = b.max()
        exp_b = np.exp(tau * (b - b_max))
        exp_sum = exp_b.sum()
        if exp_sum < 1e-300:
            break
        p = exp_b / exp_sum

        # Gradient of LSE wrt a: 2 * cross_correlation(p, a)[:n]
        s = 1
        while s < 3 * n:
            s <<= 1
        fp = np.fft.rfft(p, s)
        fa = np.fft.rfft(a, s)
        # cross-correlation: ifft(conj(fp) * fa) gives sum p[k]*a[k+j] type thing
        # Actually we need: grad[j] = 2 * sum_k p[k] * (a[k-j] + a[j-k] if valid)
        # This equals 2 * (p * a_rev_conv) but let's use the direct formula:
        # d(a*a)[k]/da[j] = a[k-j] if 0<=k-j<n else 0, plus a[j-k] if 0<=j-k<n
        # sum_k p[k] * a[k-j] = (p_conv_a)[j] where p_conv_a = ifft(fft(p)*conj(fft(a)))
        # Actually: (p * a)[k] (cross-correlation) = sum_j p[j]*a[j+k]... 
        # Let me use: grad[j] = sum_k p[k] * (a[k-j] + a[j-k] for valid indices)
        # = sum_k p[k]*a[k-j] (k-j in [0,n)) + sum_k p[k]*a[j-k] (j-k in [0,n))
        # First term: let m = k-j, m in [0,n): sum_m p[m+j]*a[m] = (p shifted) * a correlation
        # Second term: let m = j-k, m in [0,n): sum_m p[j-m]*a[m] = similar
        # Together: 2 * sum_m p[j+m]*a[m] (approximately, for valid ranges)
        # This is the cross-correlation of p and a evaluated at lag j.
        corr = np.fft.irfft(np.conj(fp) * fa, s)[:n]
        grad = 2.0 * corr

        # Adam update
        t_step += 1
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * grad ** 2
        m_hat = m / (1 - beta1 ** t_step)
        v_hat = v / (1 - beta2 ** t_step)
        a_new = a - lr * m_hat / (np.sqrt(v_hat) + eps)
        a_new = np.maximum(a_new, 1e-6)
        ss = a_new.sum()
        if ss < 1e-10:
            a_new = a.copy()
        else:
            a_new /= ss

        J_new = compute_J(a_new)
        if J_new < current_J - 1e-15:
            a = a_new
            current_J = J_new
            stalled = 0
            tau = min(500.0, tau * 1.005)
            lr = lr * 1.002
        else:
            stalled += 1
            lr = lr * 0.8
            if stalled > 50:
                break
        if it % 50 == 0:
            tracker.update(a)
    tracker.update(a)
    return a

def peak_shave(a, max_iters, deadline, tracker):
    """Direct peak shaving: reduce components that contribute most to the peak."""
    a = np.maximum(a.astype(np.float64), 1e-6)
    a /= a.sum()
    n = len(a)
    current_J = compute_J(a)
    tracker.update(a)
    rng = np.random.default_rng(123)

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        b = fast_conv(a)
        k_max = int(np.argmax(b))

        # Find top contributing indices to b[k_max]
        js = max(0, k_max - n + 1)
        je = min(n - 1, k_max)
        contributions = a[js:je + 1] * a[k_max - js:k_max - je]
        # Component j contributes a[j]*a[k_max-j] to b[k_max]
        # To reduce b[k_max], reduce the largest contributors
        # But we want to increase other components to maintain the sum

        # Pick a component to reduce (one with high contribution)
        if contributions.max() > 0:
            top_contrib = np.argsort(contributions)[-min(3, len(contributions)):][::-1]
        else:
            top_contrib = [0]

        # Pick a component to increase (one with low contribution to peak and low value)
        all_contribs = np.zeros(n)
        all_contribs[js:je + 1] = contributions
        # Also consider contribution to other high-value lags
        top_lags = np.argsort(b)[-5:]
        for kl in top_lags:
            js2 = max(0, kl - n + 1)
            je2 = min(n - 1, kl)
            all_contribs[js2:je2 + 1] += a[js2:je2 + 1] * a[kl - js2:kl - je2]

        low_contrib = np.argsort(all_contribs)[:5]

        best_J = current_J
        best_a = None

        for ci in top_contrib:
            j_reduce = js + int(ci)
            for ei in low_contrib:
                j_increase = int(ei)
                if j_increase == j_reduce:
                    continue
                for ratio in [0.5, 0.7, 0.3, 0.8]:
                    a_trial = a.copy()
                    delta = a_trial[j_reduce] * (1 - ratio)
                    a_trial[j_reduce] *= ratio
                    a_trial[j_increase] += delta
                    a_trial = np.maximum(a_trial, 1e-8)
                    ss = a_trial.sum()
                    if ss < 1e-10:
                        continue
                    a_trial /= ss
                    J_trial = compute_J(a_trial)
                    if J_trial < best_J - 1e-14:
                        best_J = J_trial
                        best_a = a_trial.copy()

        if best_a is not None:
            a = best_a
            current_J = best_J
        if it % 20 == 0:
            tracker.update(a)
    tracker.update(a)
    return a

def flat_spectrum_init(n, rng, bandwidth=None):
    """Generate sequence with approximately flat power spectrum."""
    half = n // 2 + 1
    if bandwidth is None:
        bandwidth = half
    bandwidth = min(bandwidth, half)
    angles = rng.uniform(0, 2 * np.pi, half)
    amplitudes = np.zeros(half)
    amplitudes[:bandwidth] = 1.0
    # Add some random variation in amplitude
    amplitudes[:bandwidth] *= (1.0 + 0.3 * rng.normal(0, 1, bandwidth))
    amplitudes[:bandwidth] = np.maximum(amplitudes[:bandwidth], 0.1)

    A = np.zeros(n, dtype=complex)
    A[:half] = amplitudes * np.exp(1j * angles[:half])
    for k in range(1, min(n - half + 1, half)):
        A[n - k] = np.conj(A[k])
    a = np.fft.ifft(A).real
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-10:
        return np.ones(n) / n
    return a / s

def resample_seq(src, dest_n):
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    return np.maximum(np.interp(t_new, t_old, src), 1e-6)

def generate_inits(n, rng, parent=None):
    """Generate diverse initializations."""
    inits = []

    if parent is not None and parent.size > 0:
        inits.append(resample_seq(parent, n))
        inits.append(resample_seq(parent[::-1], n))

    inits.append(np.ones(n) / n)

    # Flat spectrum with different bandwidths
    for bw_frac in [0.3, 0.5, 0.7, 1.0]:
        inits.append(flat_spectrum_init(n, rng, bandwidth=int(n // 2 + 1 * bw_frac)))

    # Cosine modulations
    t = np.linspace(0, 1, n, endpoint=False)
    for freq in [2, 3, 5, 8, 13]:
        inits.append(1.0 + 0.4 * np.cos(2 * np.pi * freq * t + rng.uniform(0, 6.28)))

    # Multi-frequency
    inits.append(1.0 + 0.2 * np.sin(2 * np.pi * t) + 0.15 * np.sin(4 * np.pi * t)
                + 0.1 * np.sin(7 * np.pi * t) + 0.05 * np.sin(11 * np.pi * t))

    # Triangle / tent
    inits.append(1.0 - np.abs(2 * t - 1.0))
    inits.append(0.5 + 0.5 * (1.0 - np.abs(2 * t - 1.0)) ** 2)

    # Gaussian bumps
    for center in [0.3, 0.7]:
        inits.append(np.exp(-10 * (t - center) ** 2) + 0.1)

    # Sparse smoothed
    sparse = np.zeros(n)
    n_spikes = max(5, n // 20)
    positions = rng.choice(n, size=min(n_spikes, n), replace=False)
    sparse[positions] = 1.0
    kernel_size = max(3, n // 40)
    kernel = np.ones(kernel_size) / kernel_size
    inits.append(np.convolve(sparse, kernel, mode='same') + 0.05)

    # Power law
    inits.append((t + 0.05) ** rng.uniform(-1.0, 1.0))

    # Random (smoothed)
    raw = rng.random(n)
    kernel = np.ones(5) / 5
    inits.append(np.convolve(raw, kernel, mode='same') + 0.1)

    for x in inits:
        x = np.maximum(np.asarray(x, dtype=np.float64), 0.001)
        x /= x.sum()
    return inits

        if gen % 5 == 0:
            elapsed = time.monotonic() - t_start
            print(f"    Pop gen {gen}: best_J={best_J:.8f} t={elapsed:.1f}s", flush=True)

    return best_a

def propose_candidate(seed=42, budget_s=1500, **kwargs):
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + float(budget_s) - 30.0
    tracker = ScoreTracker()

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        if p.ndim == 1 and p.size > 0 and np.all(np.isfinite(p)):
            parent = np.maximum(p, 0.0)
            if parent.sum() > 0.01:
                parent = parent / parent.sum() * np.sqrt(2 * len(parent))
                tracker.update(parent)
                print(f"Parent loaded: n={parent.size}, J={tracker.best_score:.8f}", flush=True)
    except (NameError, AttributeError):
        pass

    # === Phase 1: Multi-length exploration ===
    print(f"\n{'='*60}", flush=True)
    print(f"PHASE 1: Multi-length exploration", flush=True)
    print(f"{'='*60}", flush=True)

    lengths = [256, 384, 512, 768]
    # If we have time, add larger lengths
    if budget_s > 1000:
        lengths.append(1024)

    phase1_budget = (deadline - time.monotonic()) * 0.45
    phase1_deadline = time.monotonic() + phase1_budget

    best_overall = None
    best_overall_J = tracker.best_score if tracker.best is not None else float('inf')

    for n in lengths:
        if time.monotonic() > phase1_deadline - 30:
            break
        t_start = time.monotonic()
        t_alloc = min(phase1_deadline - time.monotonic(), 120.0)
        print(f"\n  --- n={n}, alloc_time={t_alloc:.0f}s ---", flush=True)

        inits = generate_inits(n, rng, parent)
        n_inits = min(len(inits), max(3, int(60.0 / max(1, t_alloc / 15))))

        best_for_n = None
        best_J_for_n = float('inf')

        for idx in range(n_inits):
            if time.monotonic() > phase1_deadline - 20:
                break
            a0 = inits[idx]

            # LP optimization
            n_lp = max(10, min(30, int(30.0 / max(1, n / 256))))
            t_sub = min(time.monotonic() + 25.0, phase1_deadline - 40)
            a = lp_optimize(a0, n_lp, t_sub, tracker, tighten_start=0.0, tighten_end=0.025)

            # LSE polish
            if time.monotonic() < phase1_deadline - 30:
                a = lse_adam(a, min(1000, int(20000 / n)), phase1_deadline - 30, tracker)

            J = compute_J(a)
            if J < best_J_for_n:
                best_J_for_n = J
                best_for_n = a.copy()

        if best_for_n is not None:
            tracker.update(best_for_n)
            if best_J_for_n < best_overall_J:
                best_overall_J = best_J_for_n
                best_overall = best_for_n.copy()
            print(f"  n={n} done: J={best_J_for_n:.8f} (time={time.monotonic()-t_start:.1f}s)", flush=True)

    # === Phase 2: Population search on best length ===
    if time.monotonic() < deadline - 200:
        print(f"\n{'='*60}", flush=True)
        print(f"PHASE 2: Population search", flush=True)
        print(f"{'='*60}", flush=True)

        pop_n = best_overall.size if best_overall is not None else (parent.size if parent is not None else 512)
        t_pop = min(deadline - 200, 300.0)
        t_start = time.monotonic()
        a_pop = population_search(pop_n, n_pop=8, deadline=time.monotonic() + t_pop, tracker=tracker, parent=best_overall)
        J_pop = compute_J(a_pop)
        if J_pop < best_overall_J:
            best_overall_J = J_pop
            best_overall = a_pop.copy()
        print(f"  Population search: J={J_pop:.8f} (time={time.monotonic()-t_start:.1f}s)", flush=True)

    # === Phase 3: Focused deep optimization ===
    if best_overall is not None and time.monotonic() < deadline - 100:
        print(f"\n{'='*60}", flush=True)
        print(f"PHASE 3: Deep optimization on best (n={best_overall.size})", flush=True)
        print(f"{'='*60}", flush=True)

        a = best_overall.copy()
        n = a.size
        t_start = time.monotonic()

        # Multiple LP passes with increasing tightening
        for tighten in [0.005, 0.01, 0.015, 0.02, 0.025, 0.03, 0.035, 0.04]:
            if time.monotonic() > deadline - 80:
                break
            a = lp_optimize(a, 20, deadline - 80, tracker, tighten_start=tighten * 0.5, tighten_end=tighten)

        # LSE with high tau
        if time.monotonic() < deadline - 60:
            a = lse_adam(a, min(5000, int(100000 / n)), deadline - 60, tracker)

        # Peak shaving
        if time.monotonic() < deadline - 40:
            a = peak_shave(a, 500, deadline - 40, tracker)

        J_final = compute_J(a)
        if J_final < best_overall_J:
            best_overall_J = J_final
            best_overall = a.copy()
        print(f"  Deep opt: J={J_final:.8f} (time={time.monotonic()-t_start:.1f}s)", flush=True)

    # === Phase 4: Final polishing ===
    if best_overall is not None and time.monotonic() < deadline - 20:
        print(f"\n{'='*60}", flush=True)
        print(f"PHASE 4: Final polishing", flush=True)
        print(f"{'='*60}", flush=True)

        a = best_overall.copy()

        # Short LP
        if time.monotonic() < deadline - 15:
            a = lp_optimize(a, 10, deadline - 15, tracker, tighten_start=0.01, tighten_end=0.02)

        # LSE
        if time.monotonic() < deadline - 10:
            a = lse_adam(a, 500, deadline - 10, tracker)

        # Coordinate refinement
        if time.monotonic() < deadline - 5:
            a = coordinate_refine(a, 2000, deadline - 5, tracker)

        # One more LP
        if time.monotonic() < deadline - 2:
            a = lp_optimize(a, 5, deadline - 2, tracker, tighten_start=0.005, tighten_end=0.01)

        J_final = compute_J(a)
        if J_final < best_overall_J:
            best_overall_J = J_final
            best_overall = a.copy()
        print(f"  Final: J={J_final:.8f}", flush=True)

    # Fallback
    if tracker.best is None:
        tracker.best = np.ones(256) / 256
        tracker.best_score = compute_J(tracker.best)
        tracker.update(tracker.best)

    print(f"\n{'='*60}", flush=True)
    print(f"FINAL: J={tracker.best_score:.8f} n={tracker.best.size} time={time.monotonic()-tracker.t_start:.1f}s", flush=True)
    print(f"{'='*60}", flush=True)

    return [float(x) for x in tracker.best]
