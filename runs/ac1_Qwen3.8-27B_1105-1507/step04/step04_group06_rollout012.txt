"""
Minimizes J(a) = 2*n*max(convolve(a,a))/(sum(a))² using:
1. Efficient subgradient computation via FFT
2. LP mirror descent with quadratic line search
3. Subgradient projected gradient as alternative

import numpy as np
from scipy.optimize import linprog
from scipy.sparse import csr_matrix
import time

# Cache for LP constraint structures
_A_cache = {}

def fast_conv(a):
    n = len(a)
    s = 1
    while s < 2 * n - 1:
        s <<= 1
    fa = np.fft.rfft(a, s)
    return np.fft.irfft(fa * fa, s)[:2 * n - 1]

def compute_J(a):
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))

def get_A_structure(n):
    if n not in _A_cache:
        rows = []
        cols = []
        for k in range(2 * n - 1):
            jl = max(0, k - n + 1)
            jh = min(n - 1, k)
            for j in range(jl, jh + 1):
                rows.append(k)
                cols.append(j)
        _A_cache[n] = (np.array(rows, dtype=np.int32), np.array(cols, dtype=np.int32))
    return _A_cache[n]

def build_A_sparse(a):
    n = len(a)
    rows_arr, cols_arr = get_A_structure(n)
    vals = a[rows_arr - cols_arr]
    return csr_matrix((vals, (rows_arr, cols_arr)), shape=(2 * n - 1, n))

def subgrad_of_max_conv(a):
    """
    Compute subgradient of max_k (a*a)_k.
    Returns (max_val, subgradient).
    """
    n = len(a)
    b = fast_conv(a)
    b_max = float(b.max())
    # Find all indices achieving max (within tolerance)
    tol = 1e-10 * b_max
    idx = np.where(b >= b_max - tol)[0]
    p = np.zeros(2 * n - 1)
    p[idx] = 1.0 / len(idx)
    # grad[j] = 2 * sum_{m=0}^{n-1} a[m] * p[j+m]
    # = 2 * (a * p_rev)[2n-2-j]
    p_rev = p[::-1]
    # Compute full convolution of a and p_rev
    s = 1
    while s < n + (2 * n - 1) - 1:
        s <<= 1
    fa = np.fft.rfft(a, s)
    fp_rev = np.fft.rfft(p_rev, s)
    conv = np.fft.irfft(fa * fp_rev, s)
    # conv has length n + (2n-1) - 1 = 3n - 2
    # grad[j] = 2 * conv[2n-2-j]
    grad = 2.0 * conv[2 * n - 2 - np.arange(n)][::-1]
    # Wait, let me recheck: grad[j] = 2 * (a * p_rev)[2n-2-j]
    # (a*p_rev)[k] = conv[k]
    # grad[j] = 2 * conv[2n-2-j]
    grad = 2.0 * conv[2 * n - 2 - np.arange(n)]
    return b_max, grad

def proj_simplex(v):
    """Project v onto the simplex {x >= 0, sum x = 1}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.where(u * np.arange(1, n + 1) > (cssv - 1))[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0)

def grad_J(a):
    """
    Compute (sub)gradient of J(a) = 2n*max(a*a)/(sum a)^2.
    Assumes a >= 0, sum(a) = 1.
    """
    n = len(a)
    b_max, grad_bmax = subgrad_of_max_conv(a)
    s = a.sum()  # should be 1
    # J = 2n * g / s^2
    # dJ/da = 2n * (s * dg/da - 2*g) / s^3
    grad_J = 2.0 * n * (s * grad_bmax - 2.0 * b_max) / (s ** 3)
    return grad_J

def lp_best_response(a, M):
    """Solve LP: max sum(g) s.t. (a*g)_k <= M, g >= 0."""
    n = len(a)
    c = -np.ones(n)
    A = build_A_sparse(a)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    try:
        r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                    options={'presolve': True, 'time_limit': 30})
        if r.success and r.x is not None:
            g = np.maximum(r.x, 0.0)
            g_sum = g.sum()
            if g_sum > 1e-10:
                return g / g_sum
    except Exception:
        pass
    return None

def line_search_quad(a, g, current_J, max_evals=20):
    """
    Line search with quadratic interpolation.
    Find t in [0,1] minimizing J((1-t)*a + t*g).
    """
    best_t = 0.0
    best_J = current_J
    best_a = a.copy()
    
    # Evaluate at t=0 and t=1
    J1 = compute_J(g)
    if J1 < best_J:
        best_J = J1
        best_t = 1.0
        best_a = g.copy()
    
    if best_t == 1.0:
        return best_a, best_J, best_t
    
    # Try t=0.5
    a_half = 0.5 * a + 0.5 * g
    J_half = compute_J(a_half)
    if J_half < best_J:
        best_J = J_half
        best_t = 0.5
        best_a = a_half.copy()
    
    # Now we have 3 points: (0, current_J), (0.5, J_half), (1, J1)
    # Use quadratic interpolation
    points = [(0.0, current_J), (0.5, J_half), (1.0, J1)]
    
    for _ in range(max_evals - 3):
        if len(points) < 3:
            # Pick a new point
            t_new = best_t + 0.1
            if t_new > 0.95:
                t_new = best_t - 0.1
            if t_new < 0.05:
                t_new = 0.5
            a_new = (1 - t_new) * a + t_new * g
            J_new = compute_J(a_new)
            points.append((t_new, J_new))
            continue
        
        # Fit quadratic to 3 closest points
        points.sort(key=lambda p: abs(p[0] - best_t))
        t0, f0 = points[0]
        t1, f1 = points[1]
        t2, f2 = points[2]
        
        if t1 == t0 or t2 == t1 or t2 == t0:
            t_new = max(0.01, min(0.99, best_t + 0.05))
        else:
            # Quadratic interpolation
            a_q = (f0 * (t1-t2) / ((t0-t1)*(t0-t2)) +
                   f1 * (t0-t2) / ((t1-t0)*(t1-t2)) +
                   f2 * (t0-t1) / ((t2-t0)*(t2-t1)))
            b_q = (f0 * (t1-t0) / ((t0-t1)*(t0-t2)) +
                   f1 * (t2-t0) / ((t1-t0)*(t1-t2)) +
                   f2 * (t2-t1) / ((t2-t0)*(t2-t1)))
            if abs(b_q) > 1e-15:
                t_new = -a_q / (2 * b_q)
            else:
                t_new = (t0 + t1 + t2) / 3.0
        
        t_new = max(0.0, min(1.0, t_new))
        
        a_new = (1 - t_new) * a + t_new * g
        J_new = compute_J(a_new)
        
        if J_new < best_J:
            best_J = J_new
            best_t = t_new
            best_a = a_new.copy()
        
        points.append((t_new, J_new))
        # Keep only the 5 closest points to best_t
        points.sort(key=lambda p: abs(p[0] - best_t))
        points = points[:5]
    
    return best_a, best_J, best_t

def lp_mirror_descent(a, deadline, tracker, max_time=600, n_line_evals=18):
    """LP mirror descent with quadratic line search."""
    a = np.maximum(a.astype(np.float64), 1e-8)
    a = a / a.sum()
    current_J = compute_J(a)
    stall_count = 0
    best_J = current_J
    best_a = a.copy()
    
    start_time = time.monotonic()
    
    while time.monotonic() < min(deadline, start_time + max_time):
        b = fast_conv(a)
        M = float(b.max())
        
        g = lp_best_response(a, M)
        if g is None:
            break
        
        a_new, J_new, t_used = line_search_quad(a, g, current_J, max_evals=n_line_evals)
        
        if J_new < current_J - 1e-14:
            a = a_new
            current_J = J_new
            stall_count = 0
            if J_new < best_J:
                best_J = J_new
                best_a = a.copy()
                tracker.update(a)
        else:
            # Try smaller steps
            improved = False
            for t_try in [0.01, 0.02, 0.05, 0.005]:
                a_try = (1 - t_try) * a + t_try * g
                a_try = np.maximum(a_try, 0)
                s = a_try.sum()
                if s < 1e-10:
                    continue
                a_try = a_try / s
                J_try = compute_J(a_try)
                if J_try < current_J - 1e-14:
                    a = a_try
                    current_J = J_try
                    stall_count = 0
                    if J_try < best_J:
                        best_J = J_try
                        best_a = a.copy()
                        tracker.update(a)
                    improved = True
                    break
            if not improved:
                stall_count += 1
                if stall_count >= 30:
                    break
    
    tracker.update(best_a)
    return best_a

def subgrad_method(a, deadline, tracker, max_time=100):
    """
    Subgradient projected gradient method with adaptive step size.
    """
    a = np.maximum(a.astype(np.float64), 1e-8)
    a = a / a.sum()
    current_J = compute_J(a)
    best_J = current_J
    best_a = a.copy()
    
    eta = 0.1  # initial step size
    start_time = time.monotonic()
    n = len(a)
    
    while time.monotonic() < min(deadline, start_time + max_time):
        grad = grad_J(a)
        # Projected subgradient step
        a_new = a - eta * grad
        a_new = proj_simplex(a_new)
        
        J_new = compute_J(a_new)
        
        if J_new < current_J - 1e-14:
            a = a_new
            current_J = J_new
            eta = min(eta * 1.05, 1.0)  # increase step size slightly
            if J_new < best_J:
                best_J = J_new
                best_a = a.copy()
                tracker.update(a)
        else:
            # Try smaller step
            for factor in [0.5, 0.25, 0.1]:
                a_try = proj_simplex(a - eta * factor * grad)
                J_try = compute_J(a_try)
                if J_try < current_J - 1e-14:
                    a = a_try
                    current_J = J_try
                    eta = eta * factor
                    if J_try < best_J:
                        best_J = J_try
                        best_a = a.copy()
                        tracker.update(a)
                    break
            else:
                eta *= 0.5
                if eta < 1e-6:
                    break
    
    tracker.update(best_a)
    return best_a

def coordinate_descent(a, deadline, tracker, max_time=60):
    """
    Coordinate descent: optimize one coordinate at a time.
    """
    a = np.maximum(a.astype(np.float64), 1e-10)
    n = len(a)
    current_J = compute_J(a)
    best_J = current_J
    best_a = a.copy()
    
    start_time = time.monotonic()
    # Fix sum = 1 by normalizing
    s = a.sum()
    a = a / s
    
    # For each coordinate, the optimal value (given others fixed) can be found
    # by solving a quadratic (since J is quadratic in each coordinate when others are fixed)
    # But the max makes it piecewise quadratic.
    # Simpler: for each coordinate, do a 1D line search.
    
    order = np.random.default_rng(42).permutation(n)
    it = 0
    while time.monotonic() < min(deadline, start_time + max_time):
        it += 1
        j = order[it % n]
        a_j = a[j]
        a[j] = 0
        
        # Binary search for best a[j]
        lo, hi = 0.0, max(2 * a_j, 0.1)
        best_val = a_j
        best_J_j = compute_J(np.maximum(a, 0))
        
        for _ in range(12):  # 12 iterations of bisection
            mid = (lo + hi) / 2
            a[j] = mid
            # Renormalize: we want sum to stay = 1
            # Actually, let's not renormalize during the search, just find best a[j]
            # Then renormalize at the end
            J_mid = compute_J(np.maximum(a, 0))
            if J_mid < best_J_j:
                best_J_j = J_mid
                best_val = mid
                # Try to go higher
                lo = mid
            else:
                hi = mid
        
        a[j] = best_val
        J_new = compute_J(np.maximum(a, 0))
        if J_new < current_J - 1e-14:
            current_J = J_new
            if J_new < best_J:
                best_J = J_new
                best_a = a.copy()
                tracker.update(a)
        elif it % 100 == 0 and it > 100:
            # No improvement in a while, stop
            if it % 500 == 0:
                break
    
    tracker.update(best_a)
    return best_a

def resample_seq(src, dest_n):
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    result = np.interp(t_new, t_old, src)
    return np.maximum(result, 0.0)

class ScoreTracker:
    def __init__(self):
        self.best = None
        self.best_score = float('inf')
        self.t_start = time.monotonic()
    
    def update(self, a):
        a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
        if a.size == 0 or a.sum() < 0.01:
            return
        score = float(evaluate_sequence(a.tolist()))
        if np.isfinite(score) and score < self.best_score:
            self.best = a.copy()
            self.best_score = score
            elapsed = time.monotonic() - self.t_start
            print(f"  * NEW BEST: J={score:.8f} n={a.size} t={elapsed:.1f}s", flush=True)
    
    def report(self):
        if self.best is not None:
            print(f"  best: J={self.best_score:.8f} n={self.best.size}", flush=True)
