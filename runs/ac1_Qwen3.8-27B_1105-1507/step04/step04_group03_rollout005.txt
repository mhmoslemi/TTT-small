def fast_conv(a):
    """Full self-convolution via FFT."""
    n = len(a)
    if n == 0:
        return np.array([0.0])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]


def compute_J(a):
    """Compute J(a) = 2*n*max(conv(a,a))/(sum(a))^2."""
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))


def build_A_sparse(n, a):
    """Build sparse matrix A where A[k,j] = a[k-j] for valid indices."""
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        j_min = max(0, k - n + 1)
        j_max = min(n - 1, k)
        for j in range(j_min, j_max + 1):
            rows.append(k)
            cols.append(j)
            vals.append(a[k - j])
    return csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))


def lp_best_response(a, M):

    except Exception:
        pass
    return None, None

def resample_seq(src, dest_n):
    """Resample 1D sequence to new length."""
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)

    n = len(a)
    a = np.maximum(a, 1e-10)
    fa = np.fft.rfft(a)
    magnitudes = np.abs(fa)
    phases = np.angle(fa)

    # Add random phase perturbation (preserves magnitudes)
    phase_pert = rng.normal(0, strength * 0.5, len(fa))
    new_fa = magnitudes * np.exp(1j * (phases + phase_pert))

    # Add small magnitude perturbation
    mag_pert = 1.0 + rng.normal(0, strength * 0.3, len(fa))
    new_fa = new_fa * mag_pert

    a_new = np.fft.irfft(new_fa, n)
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s < 1e-8:
        return a.copy()
    return a_new / s


def structured_perturb(a, rng, perturb_type, strength=1.0):
    """Apply a structured perturbation."""
    n = len(a)
    t = np.arange(n, dtype=np.float64) / n

    if perturb_type == 'shift':
        s = int(rng.integers(1, max(2, n // 8)))
        a_p = np.roll(a, s)
    elif perturb_type == 'modulate':
        freq = rng.uniform(0.5, 6.0)
        amp = strength * rng.uniform(0.05, 0.15)
        a_p = a * (1 + amp * np.cos(2 * np.pi * freq * t + rng.uniform(0, 2 * np.pi)))
    elif perturb_type == 'smooth':
        k = int(rng.integers(3, max(4, n // 12)))
        kernel = np.ones(k) / k
        a_p = np.convolve(a, kernel, mode='same')
    elif perturb_type == 'taper':
        alpha = rng.uniform(0.4, 0.8)
        edge = max(1, int(n * (1 - alpha) / 2))
        window = np.ones(n)
        for i in range(edge):
            frac = (i + 1) / (edge + 1)
            window[i] = frac
            window[n - 1 - i] = frac
        a_p = a * window
    elif perturb_type == 'bimodal':
        # Create a bimodal perturbation
        center = rng.uniform(0.2, 0.8)
        width = rng.uniform(0.05, 0.2)
        bump = np.exp(-0.5 * ((t - center) / width) ** 2)
        a_p = a * (1 + strength * 0.1 * bump)
    elif perturb_type == 'spectral':
        return spectral_perturb(a, rng, strength)
    else:
        sigma = strength * 0.03
        a_p = a + rng.normal(0, sigma, n)

    a_p = np.maximum(a_p, 0.001)
    s = a_p.sum()
    if s > 1e-8:
        a_p = a_p / s
    return a_p


def generate_diverse_inits(n, rng, parent=None):
    """Generate diverse initial sequences from various parametric families."""
    inits = []
    t = np.arange(n, dtype=np.float64) / n

    if parent is not None:
        p = resample_seq(parent, n)
        inits.append(p)
        inits.append(np.maximum(p + rng.normal(0, 0.01, n), 0.005))
        inits.append(np.maximum(p + rng.normal(0, 0.03, n), 0.005))

    # Uniform
    inits.append(np.ones(n) / n)

    # Cosine window (various parameters)
    for freq in [1, 2, 3, 4, 5]:
        for amp in [0.3, 0.5, 0.7]:
            a = np.maximum(1.0 - amp * np.cos(2 * np.pi * freq * t), 0.01)
            inits.append(a / a.sum())

    # Asymmetric shapes
    for skew in [0.3, 0.5, 0.7]:
        a = np.maximum(np.exp(-skew * (t - 0.5) ** 2 / (0.5 - skew * 0.1)), 0.01)
        inits.append(a / a.sum())

    # Bimodal
    for c in [0.25, 0.3, 0.4]:
        a = np.exp(-8 * (t - c) ** 2) + np.exp(-8 * (t - (1 - c)) ** 2) + 0.05
        inits.append(a / a.sum())

    # Random smooth
    k = max(3, n // 16)
    kernel = np.ones(k) / k
    for _ in range(3):
        smooth = np.convolve(rng.uniform(0, 1, n), kernel, mode='same')
        inits.append(np.maximum(smooth + 0.1, 0.01))

    # Sinc-like
    for beta in [2, 4, 6]:
        a = np.maximum(np.sinc(beta * (t - 0.5)), 0.001)
        inits.append(a / a.sum())

    result = []
    for a in inits:
        a = np.maximum(a, 0.001)
        s = a.sum()
        if s > 1e-6:
            result.append(a / s)
    return result


def subgradient_polish(a, max_iters, deadline):
    """Subgradient descent with momentum and backtracking."""
    a = np.maximum(a.astype(np.float64), 1e-10)
    a = a / a.sum()
    best_M = fast_conv(a).max()
    best_a = a.copy()
    momentum = np.zeros_like(a)
    beta = 0.8
    n = len(a)

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        b = fast_conv(a)
        M = b.max()
        tol = 1e-10 * M
        active = np.where(b >= M - tol)[0]

        grad = np.zeros_like(a)
        for k in active:
            lo = max(0, k - n + 1)
            hi = min(n - 1, k)
            for i in range(lo, hi + 1):
                grad[i] += 2.0 * a[k - i]

        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-15:
            break

        # Backtracking line search
        t = 2.0 / (gnorm * np.sqrt(it + 1.0))
        best_t = 0
        best_val = M
        for _ in range(10):
            momentum = beta * momentum + (1 - beta) * grad / gnorm
            a_new = a - t * momentum
            a_new = np.maximum(a_new, 0.0)
            s = a_new.sum()
            if s < 1e-10:
                t *= 0.5
                continue
            a_new = a_new / s
            new_M = fast_conv(a_new).max()
            if new_M < best_val:
                best_val = new_M
                best_t = t
                best_a_new = a_new.copy()
            else:
                t *= 0.5
                if t < 1e-8:
                    break

        if best_t > 0:
            a = best_a_new
            if best_val < best_M:
                best_M = best_val
                best_a = a.copy()

    return best_a


def lbb_objective(x):
    """Objective for L-BFGS-B: max of self-convolution (scale invariant)."""
    ax = np.maximum(x, 0.0)
    s = ax.sum()
    if s < 1e-8:
        return 1e10
    ax = ax / s
    b = fast_conv(ax)
    return float(b.max())


def evaluate_and_track(a, tracker):
    """Evaluate a sequence and update tracker if it's better."""
    try:
        a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
        if a.size == 0 or a.sum() < 0.01:
            return None
        score = float(evaluate_sequence(a.tolist()))
        if np.isfinite(score) and score < tracker['best_score']:
            tracker['best'] = a.copy()
            tracker['best_score'] = score
            elapsed = time.monotonic() - tracker['t_start']
            print(f"  [NEW BEST] J={score:.8f} n={a.size} t={elapsed:.0f}s", flush=True)
    except Exception:
        pass
    return tracker['best_score']


def propose_candidate(seed=42, budget_s=1500):
    """
    Hybrid optimizer combining extended LP mirror descent, systematic length
    exploration, adaptive perturbation schedules, and spectral diversity.
    """
    rng = np.random.default_rng(seed)
    t_start = time.monotonic()
    deadline = t_start + float(budget_s) - 30.0
    tracker = {'best': None, 'best_score': float('inf'), 't_start': t_start}

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        if p.ndim == 1 and p.size > 0 and np.all(np.isfinite(p)):
            parent = np.maximum(p, 0.0)
            evaluate_and_track(parent, tracker)
            print(f"Parent: n={parent.size}, J={tracker['best_score']:.8f}", flush=True)
    except (NameError, AttributeError):
        pass

    if parent is None:
        parent = np.ones(256)
        evaluate_and_track(parent, tracker)
        print("No parent found, using uniform", flush=True)

    parent_n = parent.size

    # === Phase 1: LP refinement on parent ===
    print(f"\n{'='*60}", flush=True)
    print(f"Phase 1: LP linear on parent (n={parent_n})", flush=True)
    print(f"{'='*60}", flush=True)

    phase1_budget = min(200, (deadline - t_start) * 0.15)
    a1 = parent.copy()
    a1 = np.maximum(a1, 1e-10)
    a1 = a1 / a1.sum()
    a1, J1 = lp_mirror_descent(a1, 5000, t_start + phase1_budget, use_multiplicative=False)
    evaluate_and_track(a1, tracker)
    print(f"  J={J1:.8f} ({time.monotonic()-t_start:.1f}s elapsed)", flush=True)

    current_best = tracker['best'].copy() if tracker['best'] is not None else a1
    current_best_J = compute_J(current_best)

    # Try multiplicative mode
    if time.monotonic() < deadline - (budget_s - 30) * 0.85:
        mult_budget = min(150, (deadline - time.monotonic()) * 0.1)
        a1m = current_best.copy()
        a1m = np.maximum(a1m, 1e-10)
        a1m = a1m / a1m.sum()
        a1m, J1m = lp_mirror_descent(a1m, 5000, time.monotonic() + mult_budget, use_multiplicative=True)
        evaluate_and_track(a1m, tracker)
        print(f"  Multiplicative: J={J1m:.8f}", flush=True)
        if J1m < current_best_J:
            current_best = a1m.copy()
            current_best_J = J1m

    # === Phase 2: Systematic length exploration ===
    print(f"\n{'='*60}", flush=True)
    print(f"Phase 2: Length exploration (base n={parent_n})", flush=True)
    print(f"{'='*60}", flush=True)

    # Generate candidate lengths: around parent_n, plus some fixed good ones
    base_n = parent_n
    length_candidates = sorted(set([
        max(128, base_n - 128), max(128, base_n - 64),
        base_n + 64, base_n + 128,
        base_n // 2 if base_n >= 256 else base_n,
        base_n * 2 if base_n * 2 <= 2048 else base_n + 128,
        512, 768, 1024, 1280
    ]))
    length_candidates = [n for n in length_candidates if 128 <= n <= 2048 and n != base_n]

    # Sort by proximity to base_n
    length_candidates.sort(key=lambda x: abs(x - base_n))
    length_candidates = length_candidates[:6]  # Try at most 6 lengths

    phase2_budget = min(300, (deadline - time.monotonic()) * 0.2)
    if phase2_budget > 60:
        per_length_time = phase2_budget / len(length_candidates) if length_candidates else phase2_budget

        for n_try in length_candidates:
            if time.monotonic() > deadline - 200:
                break
            t0 = time.monotonic()
            a_try = resample_seq(current_best, n_try)
            a_try = np.maximum(a_try, 1e-10)
            a_try = a_try / a_try.sum()

            remaining = min(per_length_time, deadline - time.monotonic() - 200)
            if remaining < 10:
                continue

            a_try, J_try = lp_mirror_descent(a_try, 3000, time.monotonic() + remaining, use_multiplicative=False)
            evaluate_and_track(a_try, tracker)
            print(f"  n={n_try}: J={J_try:.8f} ({time.monotonic()-t0:.1f}s)", flush=True)

            if J_try < current_best_J:
                current_best = a_try.copy()
                current_best_J = J_try
    else:
        print(f"  Skipping (insufficient budget: {phase2_budget:.0f}s)", flush=True)

    # === Phase 3: Diverse initialization + LP ===
    print(f"\n{'='*60}", flush=True)
    print(f"Phase 3: Diverse inits at n={current_best.size}", flush=True)
    print(f"{'='*60}", flush=True)

    n_search = current_best.size
    inits = generate_diverse_inits(n_search, rng, parent=current_best)
    # Also try at a nearby length
    n_alt = n_search + 64 if n_search + 64 <= 2048 else n_search - 64
    if n_alt >= 128:
        inits_alt = generate_diverse_inits(n_alt, rng, parent=current_best)
        inits.extend(inits_alt[:3])

    phase3_budget = min(300, (deadline - time.monotonic()) * 0.2)
    if phase3_budget > 30:
        per_init_time = max(10, phase3_budget / len(inits))
        init_results = []

        for i, a_init in enumerate(inits):
            if time.monotonic() > deadline - 200:
                break
            t0 = time.monotonic()
            a_init = np.maximum(a_init, 1e-10)
            a_init = a_init / a_init.sum()

            remaining = min(per_init_time, deadline - time.monotonic() - 200)
            if remaining < 5:
                continue

            a_res, J_res = lp_mirror_descent(a_init, 2000, time.monotonic() + remaining, use_multiplicative=False)
            evaluate_and_track(a_res, tracker)
            init_results.append((J_res, a_res))
            print(f"  Init {i} (n={a_res.size}): J={J_res:.8f}", flush=True)

            if J_res < current_best_J:
                current_best = a_res.copy()
                current_best_J = J_res

    # === Phase 4: Adaptive perturbation + LP (annealing) ===
    print(f"\n{'='*60}", flush=True)
    print(f"Phase 4: Adaptive perturbation + LP", flush=True)
    print(f"{'='*60}", flush=True)

    perturbation_types = ['shift', 'modulate', 'smooth', 'taper', 'bimodal',
                          'spectral', 'shift', 'modulate', 'smooth', 'spectral',
                          'bimodal', 'shift', 'modulate', 'smooth', 'spectral',
                          'taper', 'shift', 'bimodal', 'modulate', 'smooth']

    n_pert = len(perturbation_types)
    phase4_budget = min(400, max(100, deadline - time.monotonic() - 200))
    per_pert_time = phase4_budget / n_pert

    for i, ptype in enumerate(perturbation_types):
        if time.monotonic() > deadline - 180:
            break

        # Annealing: strength decreases over iterations
        progress = i / n_pert
        strength = 1.0 * (1.0 - 0.7 * progress) + 0.3 * rng.uniform(0.5, 1.5)

        t0 = time.monotonic()
        a_pert = structured_perturb(current_best, rng, ptype, strength)
        a_pert = np.maximum(a_pert, 1e-10)
        a_pert = a_pert / a_pert.sum()

        remaining = min(per_pert_time, deadline - time.monotonic() - 180)
        if remaining < 5:
            continue

        a_pert, J_pert = lp_mirror_descent(a_pert, 2000, time.monotonic() + remaining, use_multiplicative=False)
        evaluate_and_track(a_pert, tracker)

        if J_pert < current_best_J:
            current_best = a_pert.copy()
            current_best_J = J_pert
            print(f"  [{i}] {ptype}: J={J_pert:.8f} (NEW BEST)", flush=True)
        elif i % 3 == 0:
            print(f"  [{i}] {ptype}: J={J_pert:.8f}", flush=True)

    # === Phase 5: More aggressive exploration with different lengths ===
    print(f"\n{'='*60}", flush=True)
    print(f"Phase 5: Cross-length perturbation", flush=True)
    print(f"{'='*60}", flush=True)

    if time.monotonic() < deadline - 150:
        # Try resampling to nearby lengths and perturbing
        nearby_lengths = []
        base = current_best.size
        for delta in [-96, -48, 48, 96, 192]:
            nl = base + delta
            if 128 <= nl <= 2048:
                nearby_lengths.append(nl)

        phase5_budget = min(200, deadline - time.monotonic() - 150)
        if phase5_budget > 30 and nearby_lengths:
            per_len_time = phase5_budget / len(nearby_lengths)
            for nl in nearby_lengths:
                if time.monotonic() > deadline - 140:
                    break
                a_nl = resample_seq(current_best, nl)
                a_nl = structured_perturb(a_nl, rng, 'spectral', 0.8)
                a_nl = np.maximum(a_nl, 1e-10)
                a_nl = a_nl / a_nl.sum()

                remaining = min(per_len_time, deadline - time.monotonic() - 140)
                if remaining < 5:
                    continue
                a_nl, J_nl = lp_mirror_descent(a_nl, 2000, time.monotonic() + remaining, use_multiplicative=False)
                evaluate_and_track(a_nl, tracker)
                if J_nl < current_best_J:
                    current_best = a_nl.copy()
                    current_best_J = J_nl
                    print(f"  n={nl}: J={J_nl:.8f} (NEW BEST)", flush=True)
                else:
                    print(f"  n={nl}: J={J_nl:.8f}", flush=True)

    # === Phase 6: Extended LP on global best ===
    print(f"\n{'='*60}", flush=True)
    print(f"Phase 6: Extended LP on best (n={current_best.size})", flush=True)
    print(f"{'='*60}", flush=True)

    if time.monotonic() < deadline - 50:
        a_ext = current_best.copy()
        a_ext = np.maximum(a_ext, 1e-10)
        a_ext = a_ext / a_ext.sum()
        a_ext, J_ext = lp_mirror_descent(a_ext, 100000, deadline - 40, use_multiplicative=False)
        evaluate_and_track(a_ext, tracker)
        print(f"  Linear LP: J={J_ext:.8f}", flush=True)

        if J_ext < current_best_J:
            current_best = a_ext.copy()
            current_best_J = J_ext

        if time.monotonic() < deadline - 30:
            a_ext2 = current_best.copy()
            a_ext2 = np.maximum(a_ext2, 1e-10)
            a_ext2 = a_ext2 / a_ext2.sum()
            a_ext2, J_ext2 = lp_mirror_descent(a_ext2, 50000, deadline - 20, use_multiplicative=True)
            evaluate_and_track(a_ext2, tracker)
            if J_ext2 < current_best_J:
                current_best = a_ext2.copy()
                current_best_J = J_ext2
                print(f"  Mult LP: J={J_ext2:.8f} (NEW BEST)", flush=True)

    # === Phase 7: Subgradient polish ===
    if time.monotonic() < deadline - 10:
        print(f"\nPhase 7: Subgradient polish", flush=True)
        a_sub = subgradient_polish(current_best, 500, min(deadline - 5, time.monotonic() + 30))
        evaluate_and_track(a_sub, tracker)
        J_sub = compute_J(a_sub)
        print(f"  Subgradient: J={J_sub:.8f}", flush=True)
        if J_sub < current_best_J:
            current_best = a_sub.copy()
            current_best_J = J_sub

    # === Phase 8: L-BFGS-B final refinement ===
    if time.monotonic() < deadline and tracker['best'] is not None:
        print(f"\nPhase 8: L-BFGS-B", flush=True)
        n_final = tracker['best'].size
        try:
            res = minimize(lbb_objective, tracker['best'].astype(np.float64),
                           method='L-BFGS-B', bounds=[(0.0, None)] * n_final,
                           options={'maxiter': 50, 'max_fun': 20000, 'ftol': 1e-16})
            if res.fun < 1e10:
                evaluate_and_track(res.x, tracker)
        except Exception as e:
            print(f"  L-BFGS-B error: {e}", flush=True)

    # Final
    if tracker['best'] is not None:
        final_a = tracker['best']
    else:
        final_a = current_best

    print(f"\n{'='*60}", flush=True)
    print(f"FINAL: J={tracker['best_score']:.8f} n={final_a.size}", flush=True)
    print(f"Total time: {time.monotonic()-t_start:.0f}s", flush=True)
    print(f"{'='*60}", flush=True)

    return [float(x) for x in final_a]
