For the coordinate descent, here's the key insight: if we fix all of a except a[i], then (a*a)[k] is linear in a[i] (for k where a[i] appears). Specifically:

- For k = i: (a*a)[i] includes a[i]*a[0] = t*a[0]
- For k = 2i: (a*a)[2i] includes a[i]*a[i] = t^2
- For k = i + j (j != i): (a*a)[k] includes a[i]*a[j] + a[j]*a[i] = 2t*a[j]
- etc.

So (a*a)[k] is at most quadratic in t (the quadratic term only appears when j = k-j, i.e., k = 2j, and both terms are a[i]).

Actually, let me be more careful. (a*a)[k] = sum_j a[j]*a[k-j]. The terms involving a[i] = t are:
- j = i: a[i]*a[k-i] = t * a[k-i] (if 0 <= k-i < n)
- j = k-i: a[k-i]*a[i] = a[k-i] * t (if 0 <= k-i < n and k-i != i)

If k-i = i (i.e., k = 2i), these are the same term: a[i]*a[i] = t^2.
Otherwise, they're two separate terms: 2t*a[k-i].

So (a*a)[k] as a function of t:
- If k = 2i: ... + t^2 + const
- If k != 2i: ... + 2*a[k-i]*t + const (if k-i is valid and k-i != i)
- If k-i is not in [0, n-1]: no dependence on t

We want to minimize max_k (a*a)[k] over t >= 0. This is a 1D problem that we can solve by bisection or by finding the intersection points.

This is actually a nice approach! The 1D subproblem is:
min_t max_k [c_k + d_k * t + e_k * t^2] where e_k is 0 or 1 (at most one k has e_k = 1, namely k = 2i).

The max of a set of linear and at most one quadratic function is convex in t, so we can find the minimum by ternary search or by finding where the active constraints intersect.

Actually, for the coordinate descent, since we're minimizing max of affine (and one quadratic) functions, the optimal t is where two or more "active" constraints intersect at their minimum. We can find this by:
1. Evaluating the current max at t = t_old

3. Among the active constraints, finding the t that minimizes the max of just those
4. This gives a new t, and we iterate

This is essentially a 1D min-max problem, solvable in O(m log m) where m is the number of active constraints.

Minimizes J(a) = 2n*max(conv(a,a))/(sum(a))^2 using a hybrid approach:

2. LP mirror descent with Polyak averaging and adaptive step sizes

4. Structured restarts and final refinement

import numpy as np
from scipy.optimize import linprog, minimize_scalar

def fast_conv(a):
    """Full self-convolution via FFT."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]


def compute_J(a):
    """Compute J(a) = 2*n*max(conv(a,a))/(sum(a))^2."""
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))


def resample_seq(src, dest_n):
    """Resample 1D sequence to new length via linear interpolation."""
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)


def smooth_grad_descent(a, max_iters, deadline, tau_start=5.0, tau_end=100.0,
                        lr=0.12, momentum=0.82):
    """Smooth gradient descent using log-sum-exp approximation of max."""
    a = np.maximum(a.astype(np.float64), 1e-10)
    a = a / a.sum()
    n = len(a)
    best_a = a.copy()
    best_M = fast_conv(a).max()
    v = np.zeros(n)

        if time.monotonic() > deadline:
            break
        frac = it / max(1, max_iters)
        tau = tau_start + (tau_end - tau_start) * (frac ** 0.7)
        l = lr * (1.0 - 0.4 * frac)

        b = fast_conv(a)
        b_max = b.max()
        exp_b = np.exp(tau * (b - b_max))

def build_sparse_conv(a, n):
    """Build sparse matrix A where A[k,j] = a[k-j] for valid (k,j)."""
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        j_min = max(0, k - n + 1)
        j_max = min(n - 1, k)
        for j in range(j_min, j_max + 1):
            rows.append(k)
            cols.append(j)
            vals.append(a[k - j])
    return csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))


def lp_direction(a, M, time_limit=20):
    """Solve LP: max sum(g) s.t. A@g <= M, g >= 0. Returns normalized g or None."""
    n = len(a)
    A = build_sparse_conv(a, n)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    try:
        r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                    options={'time_limit': time_limit})
        if r.success and r.x is not None:
            g = np.maximum(r.x, 0)
            s = g.sum()
            if s > 1e-12:
                return g / s
    except Exception:
        pass
    return None

    a = np.maximum(a.astype(np.float64), 1e-10)

    no_improve = 0

        M = float(b.max())
        g = lp_direction(a, M, time_limit=max(5, int(25 - 0.05 * it)))
        if g is None:

        # Adaptive step size: larger when direction is very different
        diff = np.abs(a - g).sum() / n
        base_t = t_init * (t_min / t_init) ** (it / max(1, max_iters))
        t = min(0.8, base_t * (1.0 + 2.0 * diff))

def coord_objective(t, b_base, lin, has_quad, quad_k):
    """1D objective for coordinate descent: max_k [b_base[k] + lin[k]*t + quad]."""
    b = b_base + lin * t
    m = b.max()
    if has_quad:
        q_val = b[quad_k] + t * t
        m = max(m, q_val)
    return float(m)


def coordinate_descent(a, max_rounds, deadline):
    """1D coordinate descent: for each coordinate, solve exact min-max via golden section."""
    a = np.maximum(a.astype(np.float64), 0.0).copy()
    a = a / a.sum()
    n = len(a)
    best_M = fast_conv(a).max()

    ks = np.arange(2 * n - 1)

    for rnd in range(max_rounds):
        if time.monotonic() > deadline:
            break

        any_improved = False

        for i in range(n):
            if time.monotonic() > deadline:
                break

            t_old = a[i]
            a[i] = 0.0
            b_base = fast_conv(a)

            # Linear coefficients: for each k, coefficient of t in b(t)[k]
            dis = ks - i
            valid = (dis >= 0) & (dis < n) & (dis != i)
            lin = np.zeros(2 * n - 1)
            lin[valid] = 2.0 * a[dis[valid]]

            has_quad = (2 * i <= 2 * n - 2)
            quad_k = 2 * i

            # Upper bound for t: don't go too far above current
            t_hi = max(t_old * 3.0, 0.1)

            res = minimize_scalar(coord_objective, bounds=(0.0, t_hi),
                                  args=(b_base, lin, has_quad, quad_k),
                                  method='bounded',
                                  options={'xatol': 1e-9, 'maxiter': 40})
            t_opt = res.x
            a[i] = t_opt
            any_improved = True

        s = a.sum()
        if s < 1e-10:
            break
        a /= s
        M = fast_conv(a).max()
        if M < best_M:
            best_M = M
        elif not any_improved:
            break

    return a

"""
Minimizes J(a) = 2n*max(conv(a,a))/(sum(a))^2 using a hybrid approach:
1. Smooth log-sum-exp gradient descent with momentum (fast multi-length exploration)
2. LP mirror descent with Polyak averaging and adaptive step sizes
3. 1D coordinate descent min-max polish (exact per-coordinate optimization)
4. Structured restarts and final refinement
Key innovations: coordinate descent solving exact 1D min-max subproblems,
adaptive LP step sizes based on direction distance, and diverse structured
initializations to escape local minima.
"""

import numpy as np
from scipy.optimize import linprog, minimize_scalar
from scipy.sparse import csr_matrix
import time


def fast_conv(a):
    """Full self-convolution via FFT."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]


def compute_J(a):
    """Compute J(a) = 2*n*max(conv(a,a))/(sum(a))^2."""
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))


def resample_seq(src, dest_n):
    """Resample 1D sequence to new length via linear interpolation."""
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)


def smooth_grad_descent(a, max_iters, deadline, tau_start=5.0, tau_end=100.0,
                        lr=0.12, momentum=0.82):
    """Smooth gradient descent using log-sum-exp approximation of max."""
    a = np.maximum(a.astype(np.float64), 1e-10)
    a = a / a.sum()
    n = len(a)
    best_a = a.copy()
    best_M = fast_conv(a).max()
    v = np.zeros(n)

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        frac = it / max(1, max_iters)
        tau = tau_start + (tau_end - tau_start) * (frac ** 0.7)
        l = lr * (1.0 - 0.4 * frac)

        b = fast_conv(a)
        b_max = b.max()
        exp_b = np.exp(tau * (b - b_max))
        p = exp_b / exp_b.sum()

        grad = np.zeros(n)
        for k in range(2 * n - 1):
            pk = p[k]
            if pk < 1e-14:
                continue
            lo = max(0, k - n + 1)
            hi = min(n - 1, k)
            grad[lo:hi + 1] += 2.0 * pk * a[k - hi:k - lo + 1][::-1]

        v = momentum * v + l * grad
        a_new = a - v
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s < 1e-10:
            a_new = a.copy()
        else:
            a_new = a_new / s
        a = a_new

        M_new = fast_conv(a).max()
        if M_new < best_M * (1 - 1e-10):
            best_M = M_new
            best_a = a.copy()

    return best_a


def build_sparse_conv(a, n):
    """Build sparse matrix A where A[k,j] = a[k-j] for valid (k,j)."""
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        j_min = max(0, k - n + 1)
        j_max = min(n - 1, k)
        for j in range(j_min, j_max + 1):
            rows.append(k)
            cols.append(j)
            vals.append(a[k - j])
    return csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))
