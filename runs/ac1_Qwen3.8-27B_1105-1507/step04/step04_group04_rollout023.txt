"""
Step function optimization using peak-shaving subgradient, alternating LP,
and simulated annealing local search. Targets improving below 1.5027.
"""

import numpy as np
from scipy.optimize import linprog
import time

try:
    import cvxpy as cp
    HAS_CVXPY = True
except ImportError:
    HAS_CVXPY = False


def compute_conv_ff(a):
    """Fast convolution using FFT. Returns array of length 2n-1."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    conv = np.fft.irfft(fa * fa, size)
    return conv[:2 * n - 1]


def compute_obj(a):
    """Compute objective value."""
    n = len(a)
    if n == 0:
        return np.inf
    conv = compute_conv_ff(a)
    max_b = float(np.max(conv))
    sum_a = float(np.sum(a))
    if sum_a < 0.01:
        return np.inf
    return 2.0 * n * max_b / (sum_a * sum_a)


def build_toeplitz(a):
    """Build the matrix A such that A@g = conv(a,g)."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        lo = j
        hi = min(j + n, 2 * n - 1)
        A[lo:hi, j] = a[:hi - lo]
    return A


def solve_lp_direction(a, M, time_limit=10):
    """Solve LP: max sum(g) s.t. conv(a,g) <= M, g >= 0.
    Returns the optimal g (normalized to sum=1) or None."""
    n = len(a)
    if HAS_CVXPY:
        try:
            g_var = cp.Variable(n, nonneg=True)
            A = build_toeplitz(a)
            constraints = [A @ g_var <= M]
            prob = cp.Problem(cp.Maximize(cp.sum(g_var)), constraints)
            for solver in [cp.MOSEK, cp.CLARABEL, cp.SCS]:
                try:
                    prob.solve(solver=solver, time_limit=int(time_limit))
                    break
                except Exception:
                    continue
            if g_var.value is not None:
                g_val = np.maximum(g_var.value, 0)
                s = float(np.sum(g_val))
                if s > 1e-10:
                    return g_val / s
        except Exception:
            pass
    
    A = build_toeplitz(a)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': time_limit})
        if res.status == 0:
            g = np.maximum(res.x, 0)
            s = float(np.sum(g))
            if s > 1e-10:
                return g / s
    except Exception:
        pass
    return None


def peak_shaving_step(a, tau, step, num_peaks=5):
    """Subgradient step focused on the top convolution entries."""
    n = len(a)
    conv = compute_conv_ff(a)
    max_conv = np.max(conv)
    
    # Find top entries
    top_indices = np.argsort(conv)[::-1][:num_peaks]
    
    # Compute weighted gradient (emphasize peaks)
    weights = np.zeros(len(conv))
    for idx in top_indices:
        weights[idx] = (conv[idx] - max_conv + max_conv) ** 2  # emphasize peaks
    
    # Actually, use exponential weighting
    shifted = tau * (conv - max_conv)
    shifted = np.clip(shifted, -100, 0)
    weights = np.exp(shifted)
    w_sum = np.sum(weights)
    if w_sum < 1e-300:
        weights = np.zeros(len(conv))
        weights[top_indices[0]] = 1.0
    else:
        weights /= w_sum
    
    # Gradient of weighted convolution
    # d/d(a[i]) conv[k] = 2*a[k-i] (if valid)
    # For weighted: grad[i] = 2 * sum_k weights[k] * a[k-i] (if valid)
    # This is 2 * (weights * a)[reversed correlation]
    grad = 2.0 * np.correlate(weights, a, mode='full')
    grad = grad[n - 1:n - 1 + n]
    
    # Center the gradient to preserve the sum
    # (Since we normalize after anyway, this is optional but helps)
    grad_centered = grad - np.mean(grad)
    
    a_new = a * np.exp(-step * grad_centered)
    a_new = np.maximum(a_new, 1e-15)
    s = np.sum(a_new)
    if s > 1e-10:
        a_new = a_new / s
    return a_new

    s = np.sum(a)
    if s < 1e-10:
        return a
    a = a / s
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    
    step = 0.05
    tau = 20.0
    stall = 0
    patience = 50
    
    if verbose:
        print(f"  Peak-shaving start: obj={best_obj:.8f}")
    
    it = 0
    while time.monotonic() - start < time_budget:
        it += 1
        a = peak_shaving_step(a, tau, step)
        obj = compute_obj(a)
        
        if obj < best_obj - 1e-14:
            best_obj = obj
            best_a = a.copy()
            stall = 0
        else:
            stall += 1
            if stall > patience:
                # Reduce step, increase tau, add noise
                step = max(step * 0.7, 0.002)

                if step < 0.003:
                    break
                rng = np.random.default_rng(it)
                a = a * np.exp(rng.normal(0, 0.02, n))
                a = np.maximum(a, 1e-10)
                s = np.sum(a)
                a = a / s
                stall = 0
                patience = max(patience // 2, 20)
        
        if it > 100:
            step *= 0.998
            tau *= 1.003
    
    if verbose:
        print(f"  Peak-shaving end: obj={best_obj:.8f}")
    return best_a


def alternating_lp_optimize(a, time_budget=60.0, verbose=False):
    """Alternating LP with better line search."""
    n = len(a)
    a = a.copy().astype(np.float64)
    s = np.sum(a)
    if s < 1e-10:
        return a
    a = a / s
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    stall = 0
    
    if verbose:
        print(f"  Alt-LP start: obj={best_obj:.8f}")
    
    it = 0
    while time.monotonic() - start < time_budget - 5:
        it += 1
        conv = compute_conv_ff(a)
        M = float(np.max(conv))
        
        g = solve_lp_direction(a, M, time_limit=8)
        if g is None:
            break
        
        # Line search: find best t in [0, 1]
        best_t = 0.0
        best_initer = best_obj
        for t in [0.1, 0.2, 0.3, 0.5, 0.7, 0.9, 1.0]:
            a_try = (1 - t) * a + t * g
            a_try = np.maximum(a_try, 0)
            s_try = np.sum(a_try)
            if s_try < 1e-10:
                continue
            a_try = a_try / s_try
            obj_try = compute_obj(a_try)
            if obj_try < best_initer:
                best_initer = obj_try
                best_t = t
        
        if best_t > 0 and best_initer < best_obj - 1e-14:
            a = (1 - best_t) * best_t * 0 + (1 - best_t) * a  # oops
            a = (1.0 - best_t) * a + best_t * g
            a = np.maximum(a, 0)
            s = np.sum(a)
            a = a / s
            best_obj = best_initer
            best_a = a.copy()
            stall = 0
        else:
            stall += 1
            if stall > 20:
                break
            # Small step towards g
            a = 0.95 * a + 0.05 * g
            a = np.maximum(a, 0)
            s = np.sum(a)
            a = a / s
    
    if verbose:
        print(f"  Alt-LP end: obj={best_obj:.8f}")
    return best_a


def local_search_simulated(a, time_budget=120.0, verbose=False):
    """Local search with simulated annealing acceptance."""
    n = len(a)
    a = a.copy().astype(np.float64)
    s = np.sum(a)
    if s < 1e-10:
        return a
    a = a / s
    
    best_a = a.copy()

    current_obj = best_obj
    current_a = a.copy()
    
    start = time.monotonic()
    temp = 0.005  # initial temperature
    attempt = 0
    
    if verbose:
        print(f"  Local search start: obj={best_obj:.8f}")
    
    while time.monotonic() - start < time_budget - 5:
        attempt += 1
        remaining = time_budget - (time.monotonic() - start)
        if remaining < 5:
            break
        
        # Generate perturbation
        rng = np.random.default_rng(attempt * 7919)
        
        # Different perturbation strategies
        strategy = attempt % 4
        if strategy == 0:
            # Global multiplicative noise
            scale = temp * 2
            a_try = current_a * np.exp(rng.normal(0, scale, n))
        elif strategy == 1:
            # Local multiplicative noise (affect a subset)
            scale = temp * 3
            mask = rng.random(n) < 0.3
            a_try = current_a.copy()
            a_try[mask] *= np.exp(rng.normal(0, scale, mask.sum()))
        elif strategy == 2:
            # Additive noise on a subset
            scale = temp * np.mean(current_a)
            mask = rng.random(n) < 0.2
            a_try = current_a.copy()
            a_try[mask] += rng.normal(0, scale, mask.sum())
        else:
            # Exchange: reduce high values, increase low values
            sorted_idx = np.argsort(current_a)
            n_swap = max(1, n // 10)
            low_idx = sorted_idx[:n_swap]
            high_idx = sorted_idx[-n_swap:]
            a_try = current_a.copy()
            transfer = temp * np.mean(current_a[high_idx])
            a_try[high_idx] -= transfer
            a_try[low_idx] += transfer
        
        a_try = np.maximum(a_try, 1e-10)
        s_try = np.sum(a_try)
        if s_try < 1e-10:
            continue
        a_try = a_try / s_try
        
        # Quick optimization of the perturbed sequence
        obj_try = compute_obj(a_try)
        
        if obj_try < current_obj:
            # Always accept improvements
            current_a = a_try.copy()
            current_obj = obj_try
            if obj_try < best_obj:
                best_obj = obj_try
                best_a = a_try.copy()
                if verbose and attempt % 100 == 0:
                    print(f"    Attempt {attempt}: new best {best_obj:.8f}")
        elif np.log(rng.random()) > (obj_try - current_obj) / temp:
            # Accept with probability exp(-delta/T)
            current_a = a_try.copy()
            current_obj = obj_try
        
        # Cool down
        temp *= 0.9995
    
    if verbose:
        print(f"  Local search end: best={best_obj:.8f}")
    return best_a


def resample_sequence(a, new_n):
    """Resample to new length using spline interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    old_k = np.arange(n, dtype=np.float64)
    new_k = np.linspace(0, n - 1, new_n)
    a_new = np.interp(new_k, old_k, a)
    a_new = np.maximum(a_new, 0)
    s = np.sum(a_new)
    if s > 1e-10:
        a_new = a_new / s
    return a_new

def compute_conv_ff(a):
    """Fast convolution using FFT."""
    n = len(a)
    size = 1
    while size < 2 * n:
        size <<= 1
    fa = np.fft.rfft(a, size)
    conv = np.fft.irfft(fa * fa, size)
    return conv[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    conv = compute_conv_ff(a)
    max_b = float(np.max(conv))
    sum_a = float(np.sum(a))
    if sum_a < 0.01:
        return np.inf
    return 2.0 * n * max_b / (sum_a * sum_a)


def compute_conv_and_grad(a):
    """Compute convolution and its gradient w.r.t. a.
    Returns (conv, grad) where grad[i] = d/d(a[i]) max_k conv[k]
    using the active set of peak entries."""
    n = len(a)
    conv = compute_conv_ff(a)
    max_conv = np.max(conv)
    
    # Active set: entries within 1% of max
    active = conv > max_conv * 0.99
    
    # Weighted by proximity to peak
    if np.sum(active) > 1:
        weights = np.zeros_like(conv)
        weights[active] = (conv[active] - max_conv * 0.99) / (max_conv - max_conv * 0.99 + 1e-30)
    else:
        weights = np.zeros_like(conv)
        weights[np.argmax(conv)] = 1.0
    
    # Gradient: d/d(a[i]) conv[k] = 2*a[k-i] if valid
    # Sum over active k with weights
    grad = 2.0 * np.correlate(weights, a, mode='full')
    grad = grad[n - 1:n - 1 + n]
    
    return conv, grad, active, max_conv


def build_toeplitz(a):
    """Build matrix A where A@g = conv(a,g)."""
    n = len(a)
    A = np.zeros((2 * n - 1, n), dtype=np.float64)
    for j in range(n):
        lo = j
        hi = min(j + n, 2 * n - 1)
        A[lo:hi, j] = a[:hi - lo]
    return A


def solve_lp_direction(a, M, time_limit=10):
    """Solve LP: max sum(g) s.t. conv(a,g) <= M, g >= 0.
    Returns normalized g (sum=1) or None."""
    n = len(a)

            prob = cp.Problem(cp.Maximize(cp.sum(g_var)), constraints)
            for solver in [cp.MOSEK, cp.CLARABEL, cp.SCS]:
                try:
                    prob.solve(solver=solver, time_limit=int(time_limit))
                    break
                except Exception:
                    continue
            if g_var.value is not None:
                g_val = np.maximum(g_var.value, 0)

                a_opt, obj = optimize_pipeline(a_resamp, time_budget=alloc)
                if obj < best_score:
                    best_score = obj
                    best_seq = a_opt.copy()
                    print(f"  n={n_val}: {best_score:.8f} ** IMPROVED **")
    
    # Phase 5: Intensive fine local search
    if best_seq is not None:
        remaining = deadline - time.monotonic()
        if remaining > 15:
            print(f"\n=== Phase 5: Intensive local search ({remaining:.0f}s) ===")
            p5_budget = min(remaining * 0.4, 300.0)
            a_opt = fine_local_search(best_seq, time_budget=p5_budget, verbose=True)
            obj = compute_obj(a_opt)
            if obj < best_score:
                best_score = obj
                best_seq = a_opt.copy()
                print(f"  Phase 5: {best_score:.8f}")
    
    # Phase 6: Final polish round
    if best_seq is not None:
        remaining = deadline - time.monotonic()
        if remaining > 20:
            print(f"\n=== Phase 6: Final polish ({remaining:.0f}s) ===")
            p6_budget = min(remaining * 0.5, remaining - 8.0)
            a_opt, obj = optimize_pipeline(best_seq, time_budget=p6_budget, verbose=True)
            if obj < best_score:
                best_score = obj
                best_seq = a_opt.copy()
                print(f"  Phase 6: {best_score:.8f}")
    
    # Final validation
    if best_seq is None:
        best_seq = np.ones(100)
    
    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    final_val = evaluate_sequence(result)
    if final_val < best_score:
        best_score = final_val
    
    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_score:.8f}, time={elapsed:.1f}s ===")
    
    return result
