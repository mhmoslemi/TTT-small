import numpy as np
from scipy.optimize import linprog
from scipy.sparse import lil_matrix
import time


def fast_conv(a):
    """Full self-convolution via FFT."""
    n = len(a)
    s = 1
    while s < 2 * n - 1:
        s <<= 1
    fa = np.fft.rfft(a, s)
    return np.fft.irfft(fa * fa, s)[:2 * n - 1]


def compute_J(a):
    """Compute J(a) = 2*n*max(conv(a,a))/(sum(a))^2."""
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))


def build_A(a):
    """Build dense constraint matrix for LP: A[k,j] = a[k-j] if valid."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for k in range(2 * n - 1):
        jl = max(0, k - n + 1)
        jh = min(n - 1, k)
        if jl <= jh:
            A[k, jl:jh+1] = a[k - jh : k - jl + 1][::-1]
    return A


def solve_lp(a, M_target):
    """Solve LP: max sum(g) s.t. (a*g)_k <= M_target, g >= 0.
    Returns (g_primal, y_dual) or (None, None)."""
    n = len(a)
    c = -np.ones(n)
    A = build_A(a)
    b_ub = np.full(2 * n - 1, M_target)
    bounds = [(0.0, None)] * n
    r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
    if r.success and r.x is not None:
        g = np.maximum(r.x, 0.0)
        y = np.zeros(2 * n - 1)
        if r.ineqlin is not None:
            y[:2*n-1] = np.maximum(r.ineqlin[:2*n-1], 0.0)
        return g, y
    return None, None


def compute_subgradient(a, y):
    """Compute subgradient of max_k (a*a)_k using dual weights y.
    sg[j] = sum_k y[k] * d(a*a)_k / da[j]"""
    n = len(a)
    sg = np.zeros(n)
    for k in range(2 * n - 1):
        if y[k] < 1e-15:
            continue
        yl = y[k]
        for j in range(max(0, k - n + 1), min(n, k + 1)):
            ki = k - j
            if 0 <= ki < n:
                if k == 2 * j:
                    sg[j] += yl * 2.0 * a[j]
                else:
                    sg[j] += yl * a[ki]
    return sg


def mirror_descent_step(a, sg, eta):
    """Multiplicative (mirror descent) update."""
    a_new = a * np.exp(-eta * sg)
    s = a_new.sum()
    if s < 1e-15 or not np.isfinite(s):
        return a
    return a_new / s


def linear_mix(a, g, t):
    """Linear mixing update."""
    g_norm = g / g.sum() if g.sum() > 1e-15 else a
    a_new = (1.0 - t) * a + t * g_norm
    return np.maximum(a_new, 0.0)


def backtracking_line_search(a, direction_fn, current_J, max_iters=30):
    """Find step size via backtracking. direction_fn(t) returns new a."""
    best_a = a
    best_J = current_J
    t = 1.0
    for _ in range(max_iters):
        a_trial = direction_fn(t)
        a_trial = np.maximum(a_trial, 0.0)
        s = a_trial.sum()
        if s < 1e-10:
            t *= 0.5

        a_trial = a_trial / s
        J_trial = compute_J(a_trial)
        if J_trial < best_J - 1e-14:

            best_a = a_trial
            # Try to increase step

            t = min(1.0, t * 1.5)
        else:
            t *= 0.5
            if t < 1e-6:

def coordinate_descent_pass(a, max_iter_per_coord=60):
    """One pass of coordinate descent with golden section search."""
    a = np.maximum(a.astype(np.float64), 0.0)
    S = a.sum()
    if S < 1e-10:
        return a, float('inf')
    n = len(a)
    b = fast_conv(a).copy()
    current_J = 2.0 * n * b.max() / (S * S)

    order = np.arange(n)
    np.random.shuffle(order)

    for i in order:
        ai = a[i]

        def f_of_x(x):
            """J when a[i] = x."""
            new_S = S - ai + x
            if new_S < 1e-10:
                return 1e10
            # Compute new max convolution
            max_c = 0.0
            # Only b[k] for k in [i, min(i+n-1, 2n-2)] changes
            k_lo = i
            k_hi = min(i + n - 1, 2 * n - 2)
            for k in range(k_lo, k_hi + 1):
                ki = k - i
                if 0 <= ki < n:
                    if k == 2 * i:
                        ck = b[k] - ai * ai + x * x
                    else:
                        ck = b[k] - 2.0 * ai * a[ki] + 2.0 * x * a[ki]
                else:
                    ck = b[k]
                if ck > max_c:
                    max_c = ck
            # Also check unchanged convolution values
            for k in range(2 * n - 1):
                if k < k_lo or k > k_hi:
                    if b[k] > max_c:
                        max_c = b[k]
            return 2.0 * n * max_c / (new_S * new_S)

        # Golden section search
        gr = (np.sqrt(5.0) - 1.0) / 2.0
        lo = 0.0
        hi = max(3.0 * ai, 2.0 * S / n, 0.01)
        c_pt = hi - gr * (hi - lo)
        d_pt = lo + gr * (hi - lo)
        fc = f_of_x(c_pt)
        fd = f_of_x(d_pt)

        for _ in range(max_iter_per_coord):
            if fc < fd:
                hi = d_pt
                d_pt = c_pt
                fd = fc
                c_pt = hi - gr * (hi - lo)
                fc = f_of_x(c_pt)
            else:
                lo = c_pt
                c_pt = d_pt
                fc = fd
                d_pt = lo + gr * (hi - lo)
                fd = f_of_x(d_pt)

        x_opt = (lo + hi) / 2.0
        x_opt = max(0.0, x_opt)

        if abs(x_opt - ai) > 1e-12:
            old_val = ai
            a[i] = x_opt
            S = S - old_val + x_opt
            # Update b incrementally
            k_lo = i
            k_hi = min(i + n - 1, 2 * n - 2)
            for k in range(k_lo, k_hi + 1):
                ki = k - i
                if 0 <= ki < n:
                    if k == 2 * i:
                        b[k] = b[k] - old_val * old_val + x_opt * x_opt
                    else:
                        b[k] = b[k] - 2.0 * old_val * a[ki] + 2.0 * x_opt * a[ki]

    # Verify
    b_check = fast_conv(a)
    new_J = 2.0 * n * b_check.max() / (S * S)
    return a, new_J


def resample_seq(src, dest_n):
    """Resample 1D sequence to new length via linear interpolation."""
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)

            if J_trial < best_trial_J - 1e-14:
                best_trial_J = J_trial
                best_trial_a = a_trial.copy()

        # Strategy 2: Mirror descent with various eta
        for e in [eta * 0.5, eta, eta * 2.0]:
            a_trial = mirror_descent_step(a, sg, e)
            s = a_trial.sum()
            if s < 1e-10:
                continue
            a_trial /= s
            J_trial = compute_J(a_trial)
            if J_trial < best_trial_J - 1e-14:
                best_trial_J = J_trial
                best_trial_a = a_trial.copy()

        # Strategy 3: Backtracking from best so far
        if best_trial_a is not None:
            a = best_trial_a
            current_J = best_trial_J
            stall_count = 0
            if current_J < best_ever_J:
                best_ever_J = current_J
                best_ever_a = a.copy()
                tracker.update(a)
            # Adapt tightening upward (be more aggressive)
            tighten = min(0.08, tighten * 1.02)
        else:
            stall_count += 1
            if stall_count > 30:
                # Try more aggressive tightening
                tighten = min(0.15, tighten * 1.5)
                if stall_count > 60:
                    break
            else:
                tighten = max(0.001, tighten * 0.7)
                # Occasionally try a random perturbation
                if stall_count % 10 == 0:
                    a_pert = a + np.random.normal(0, 0.01, n)
                    a_pert = np.maximum(a_pert, 0.001)
                    a_pert /= a_pert.sum()
                    J_pert = compute_J(a_pert)
                    if J_pert < current_J:
                        a = a_pert
                        current_J = J_pert

        if it % 100 == 0:
            elapsed = time.monotonic() - tracker.t_start
            print(f"    LP-MD iter {it}: J={current_J:.8f} tighten={tighten:.4f} eta={eta:.4f} t={elapsed:.0f}s", flush=True)

    tracker.update(best_ever_a)
    return best_ever_a, best_ever_J


def run_coordinate_descent(a, deadline, tracker, n_passes=20):
    """Run multiple passes of coordinate descent."""
    a = np.maximum(a.astype(np.float64), 0.0)
    a /= a.sum()
    current_J = compute_J(a)
    tracker.update(a)

    no_improve = 0
    for p in range(n_passes):
        if time.monotonic() > deadline:
            break
        t0 = time.monotonic()
        a, new_J = coordinate_descent_pass(a)
        elapsed_pass = time.monotonic() - t0
        if new_J < current_J - 1e-12:
            current_J = new_J
            no_improve = 0
            tracker.update(a)
            if p % 5 == 0:
                print(f"    CD pass {p}: J={current_J:.8f} ({elapsed_pass:.2f}s)", flush=True)
        else:
            no_improve += 1
            if no_improve >= 5:
                break

    tracker.update(a)
    return a, current_J


def generate_inits(n, rng):
    """Generate diverse initial sequences."""
    inits = []
    t = np.linspace(0, 1, n, endpoint=False)
    inits.append(np.ones(n))
    inits.append(rng.uniform(0.3, 1.0, n))
    for freq in [2, 3, 5, 7, 11]:
        inits.append(1.0 + 0.3 * np.cos(2.0 * np.pi * freq * t + rng.uniform(0, 6.28)))
    inits.append(np.abs(np.sin(np.pi * t)) + 0.2)
    inits.append(0.5 + 0.5 * np.cos(2 * np.pi * t) ** 2)
    inits.append(1.0 - 0.4 * np.abs(2 * t - 1.0))
    # Pseudo-random with good autocorrelation properties
    inits.append(0.5 + 0.5 * np.sign(np.sin(2 * np.pi * rng.random() * t ** 1.5)))
    return [np.maximum(x, 0.01) for x in inits]

            parent = np.maximum(p, 0.0)
            tracker.update(parent)
            print(f"Parent loaded: n={parent.size}, J={tracker.best_score:.8f}", flush=True)
    except (NameError, AttributeError):
        pass

    if parent is None:
        parent = np.ones(384)
        tracker.update(parent)
        print(f"No parent, using uniform n={parent.size}", flush=True)

    n_parent = parent.size

    print(f"\n{'='*50}", flush=True)
    print(f"Phase 1: LP Mirror Descent (n={n_parent})", flush=True)
    print(f"{'='*50}", flush=True)
    t0 = time.monotonic()

    a_best_p1 = parent.copy()
    a_best_p1 = a_best_p1 / a_best_p1.sum()
    phase1_deadline = time.monotonic() + min(600, (deadline - time.monotonic()) * 0.4)
    a_result, J_result = run_lp_mirror_descent(a_best_p1, phase1_deadline, tracker)
    print(f"Phase 1 done: J={J_result:.8f} time={time.monotonic()-t0:.1f}s", flush=True)

    # === Phase 2: Coordinate Descent Polish ===
    if time.monotonic() < deadline - 200:

        cd_deadline = min(deadline - 150, time.monotonic() + 200)

    if time.monotonic() < deadline - 150:
        print(f"\n{'='*50}", flush=True)
        print(f"Phase 3: Alternating LP + CD", flush=True)
        print(f"{'='*50}", flush=True)
        t0 = time.monotonic()
        a_alt = tracker.best.copy()
        a_alt = a_alt / a_alt.sum()
        n = len(a_alt)

        for outer in range(10):
            if time.monotonic() > deadline - 100:
                break
            # Short LP
            lp_deadline = time.monotonic() + min(30, (deadline - 100 - time.monotonic()) * 0.4)
            a_alt, _ = run_lp_mirror_descent(a_alt, lp_deadline, tracker, max_iters=200)
            # Short CD
            if time.monotonic() < deadline - 100:
                a_alt, _ = coordinate_descent_pass(a_alt)
                a_alt = a_alt / a_alt.sum()
                tracker.update(a_alt)
            if outer % 3 == 0:
                print(f"    Alt iter {outer}: J={tracker.best_score:.8f}", flush=True)

        print(f"Phase 3 done: time={time.monotonic()-t0:.1f}s", flush=True)

    # === Phase 4: Multi-length exploration ===
    if time.monotonic() < deadline - 100:

        for nl in [256, 384, 512, 640, 512, 320, 448]:

    if time.monotonic() < deadline - 40:

            lp_dl = time.monotonic() + min(15, max(5, (deadline - 30 - time.monotonic()) * 0.2))

        # Final LP with tight parameters
        a_final = tracker.best.copy()
        a_final = a_final / a_final.sum()
        a_final, _ = run_lp_mirror_descent(a_final, deadline - 20, tracker, max_iters=500)

            a_final /= a_final.sum()

        tracker.best_score = compute_J(tracker.best)

    print(f"Total time: {time.monotonic()-tracker.t_start:.1f}s", flush=True)
    print(f"{'='*50}", flush=True)

    return [float(x) for x in tracker.best]
