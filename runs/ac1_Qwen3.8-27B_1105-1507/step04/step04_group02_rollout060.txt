"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. LP-based alternating (primary tool)
2. LSE-smoothed gradient descent (for better local optimization)
3. Multi-start with diverse initializations

import numpy as np
from scipy.optimize import linprog, minimize
from scipy.sparse import coo_matrix, csr_matrix
import time

def compute_conv(a):
    n = len(a)
    size = 1
    while size < 2*n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    conv = np.fft.irfft(fa * fa, size)
    return conv[:2*n-1]

def compute_obj(a):
    n = len(a)
    S = np.sum(a)
    if S < 1e-10:
        return np.inf
    conv = compute_conv(a)
    return 2.0 * n * np.max(conv) / (S*S)

def build_toeplitz_sparse(a):
    """Sparse matrix T such that T @ g = conv(a, g)."""
    n = len(a)
    rows = []
    cols = []
    vals = []
    for j in range(n):
        # Column j of T corresponds to g[j]
        # conv(a,g)[k] includes a[k-j]*g[j] for valid k-j
        # So T[k, j] = a[k-j] if 0 <= k-j <= n-1
        for k in range(j, j+n):
            if k < 2*n - 1:
                rows.append(k)
                cols.append(j)
                vals.append(a[k-j])
    return coo_matrix((vals, (rows, cols)), shape=(2*n-1, n)).tocsr()

def solve_lp(a, M, time_limit=10):
    """max sum(g) s.t. conv(a,g) <= M, g >= 0. Returns g or None."""
    n = len(a)
    T = build_toeplitz_sparse(a)
    c = -np.ones(n)
    b_ub = np.full(2*n-1, M)
    bounds = [(0, None)] * n
    try:
        res = linprog(c, A_ub=T, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': time_limit, 'mip_rel_gap': 0.1})
        if res.status == 0:
            return np.maximum(res.x, 0)
    except Exception:
        pass
    return None

def lse_subgradient(a, tau=50, topk=20):
    """Compute smooth subgradient using log-sum-exp approximation."""
    n = len(a)
    conv = compute_conv(a)
    # Weighted gradient using softmax over convolution values
    shifted = tau * (conv - np.max(conv))
    weights = np.exp(shifted)
    weights /= weights.sum()
    
    # Gradient: sum_k weights[k] * 2*a[k-j] for valid indices
    grad = np.zeros(n)
    # Efficient computation using convolution
    # grad[j] = 2 * sum_k weights[k] * a[k-j] (for 0<=k-j<=n-1)
    # This is 2 * convolve(weights, a_reversed)[j + n - 1]... let me think
    
    # Actually: grad[j] = 2 * sum_{k: 0<=k-j<=n-1} weights[k] * a[k-j]
    # = 2 * sum_{m=0}^{n-1} weights[j+m] * a[m] (where m = k-j)
    # = 2 * cross_correlation(weights, a)[j]
    
    # Using FFT: cross_correlation(w, a)[j] = sum_m w[j+m] * a[m]
    # = ifft(fft(w) * conj(fft(a)))... hmm, let me just use the direct formula
    
    # Pad and convolve
    size = 1
    while size < 2*n:
        size <<= 1
    fw = np.fft.rfft(weights, size)
    fa = np.fft.rfft(a, size)
    # conv[j] = sum_m weights[j+m] * a[m] for m in [0, n-1], j+m in [0, 2n-2]
    # This is like a cross-correlation
    # Let's compute: h = ifft(fw * conj(fa)) gives cross-correlation
    # h[lag] = sum_m w[m] * a[m+lag] (or similar convention)
    
    # Let me just do it directly for clarity (n up to ~1000, this is fine)
    # grad[j] = 2 * sum_{m=0}^{n-1} weights[j+m] * a[m] (if j+m < 2n-1)
    # For speed, use the fact that weights is of length 2n-1
    for j in range(n):
        # sum over m where 0<=m<n and 0<=j+m<2n-1
        m_start = 0
        m_end = min(n, 2*n-1-j)
        if m_end > m_start:
            grad[j] = 2.0 * np.dot(weights[j:j+m_end], a[:m_end])
    
    return grad, np.max(conv)

def lse_smooth_opt(a, max_iters=500, time_budget=30.0, tau=50.0):
    """Optimize using LSE-smoothed gradient with adaptive step size."""
    n = len(a)
    a = a.copy().astype(np.float64)
    S = np.sum(a)
    a = a / S
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    
    # Adaptive tau: start large (sharp), decrease over time (smoother)
    tau_cur = tau
    step = 0.1
    momentum = 0.8
    v = np.zeros(n)
    
    for it in range(max_iters):
        if time.monotonic() - start > time_budget:
            break
        
        grad, M = lse_subgradient(a, tau=tau_cur)
        # Center to maintain sum
        grad = grad - np.mean(grad)
        
        v = momentum * v + (1-momentum) * grad
        a_new = a - step * v
        a_new = np.maximum(a_new, 0)
        S_new = np.sum(a_new)
        if S_new < 1e-10:
            break
        a_new = a_new / S_new
        
        obj = compute_obj(a_new)
        if obj < best_obj:
            best_obj = obj
            best_a = a_new.copy()
            a = a_new
            # Slightly increase step if improving
            step = min(step * 1.02, 0.3)
        else:
            step *= 0.9
            if step < 1e-6:
                break
        
        # Slowly decrease tau (make smoother)
        tau_cur *= 0.995
    
    return best_a, best_obj

def lp_alternating(a, max_iters=50, time_budget=60.0):
    """LP-based alternating optimization."""
    n = len(a)
    a = a.copy().astype(np.float64)
    S = np.sum(a)

        conv = compute_conv(a)
        M = np.max(conv)

                    break
                continue
        
        S_g = np.sum(g)
        if S_g < 1e-10:
            stall += 1
            if stall > 5:
                break
            continue
        g = g / S_g
        
        # Line search

        best_obj_ls = best_obj

def cosine_init(n, n_modes=10, seed=0):
    rng = np.random.default_rng(seed)
    k = np.arange(n)
    a = np.ones(n)
    for m in range(1, n_modes+1):
        c = rng.uniform(-0.5, 0.5) / m
        phase = rng.uniform(0, 2*np.pi)
        a += c * np.cos(2*np.pi*m*k/n + phase)
    a = np.maximum(a, 0)
    S = np.sum(a)
    if S < 1e-10:
        return np.ones(n)/n
    return a/S

def team_init(n, seed=0):
    """Try to use TEA (Two Enormous Astronomical numbers) based construction."""
    # Start with random and refine
    rng = np.random.default_rng(seed)
    a = rng.uniform(0.5, 1.5, n)
    return a / np.sum(a)

def polyphase_init(n, m=4, seed=0):
    """Polyphase construction for low autocorrelation."""
    # Divide into m phases of length n//m
    phase_len = n // m
    a = np.zeros(n)
    rng = np.random.default_rng(seed)
    for p in range(m):
        # Random phase with values in {0, 1, ..., m-1} scaled
        phase_vals = rng.integers(0, m, phase_len)
        a[p*phase_len:(p+1)*phase_len] = phase_vals / (m-1) if m > 1 else 1.0
    # Fill remainder
    if m*phase_len < n:
        a[m*phase_len:] = rng.uniform(0, 1, n - m*phase_len)
    a = np.maximum(a, 0)
    S = np.sum(a)
    return a/S

Let me also think about another approach: **Direct QP formulation**

We want to minimize `M` subject to:
- `a^T T_k a <= M` for all k (quadratic constraints)
- `sum(a) = 1`
- `a >= 0`

This is a QCQP. We can try to solve it using:
- Sequential Quadratic Programming: linearize the quadratic constraints at the current point, solve the resulting QP, update.
- Penalty method: minimize `M + penalty * max(0, max_k (a^T T_k a - M))^2` iteratively.

Actually, a simpler approach: at each iteration, we know which constraints are active (the lags where `(a*a)[k] ≈ M`). We can solve a QP:
- Minimize `sum_k mu_k * (a^T T_k a)` (weighted sum of active quadratic forms)
- Subject to `sum(a) = 1`, `a >= 0`

where `mu_k` are the KKT multipliers (obtained from the LP dual).

This QP is: `min a^T (sum_k mu_k T_k) a s.t. sum(a)=1, a>=0`.

The matrix `sum_k mu_k T_k` is sparse (each `T_k` has entries on one diagonal). We can solve this QP efficiently.

Let me code this up:

```python
def solve_qp(a, M, mu_weights, time_limit=5):
    """Solve QP: min a^T Q a s.t. sum(a)=1, a>=0 where Q = sum_k mu_k * T_k."""
    n = len(a)
    # Build Q = sum_k mu_k * T_k
    # T_k[i,j] = 1 if |i-j| relates to k... 
    # Actually T_k is the matrix such that a^T T_k a = (a*a)[k]
    # (a*a)[k] = sum_i a[i]*a[k-i] = sum_i sum_j a[i]*a[j]*delta(i+j, k) [with bounds]
    # So T_k[i,j] = 1 if i+j = k (and 0<=i<n, 0<=j<n), else 0
    # (This gives a^T T_k a = sum_{i,j: i+j=k} a[i]*a[j] = (a*a)[k])
    
    # Build Q as sparse
    rows = []
    cols = []
    vals = []
    for k in range(2*n-1):
        if mu_weights[k] > 1e-10:
            for i in range(max(0, k-n+1), min(k+1, n)):
                j = k - i
                rows.append(i)
                cols.append(j)
                vals.append(mu_weights[k])
    
    from scipy.sparse import coo_matrix
    Q = coo_matrix((vals, (rows, cols)), shape=(n, n)).tocsr()
    
    # Solve QP: min a^T Q a s.t. 1^T a = 1, a >= 0
    # Using scipy.optimize.minimize with SLSQP
    def obj(x):
        return x @ Q @ x
    def grad(x):
        return 2 * Q @ x
    
    from scipy.optimize import minimize
    res = minimize(obj, a, jac=grad, method='SLSQP',
                   bounds=[(0, None)]*n,
                   constraints=[{'type': 'eq', 'fun': lambda x: np.sum(x) - 1}],
                   options={'maxiter': 100, 'ftol': 1e-12})
    
    if res.success:
        return np.maximum(res.x, 0) / np.sum(np.maximum(res.x, 0))
    return None
```

and intensive final polishing. Key improvement over previous: exploits LP dual

import numpy as np
from scipy.optimize import linprog, minimize
from scipy.sparse import coo_matrix, csr_matrix
from scipy.interpolate import interp1d
import time


def compute_conv(a):
    """Compute linear convolution a*a via FFT."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    conv = np.fft.irfft(fa * fa, size)
    return conv[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    S = float(np.sum(a))
    if S < 1e-10:
        return np.inf
    conv = compute_conv(a)
    M = float(np.max(conv))
    return 2.0 * n * M / (S * S)


def build_toeplitz_sparse(a):
    """Build sparse Toeplitz matrix T such that T @ g = conv(a, g)."""
    n = len(a)
    rows = []
    cols = []
    vals = []
    for j in range(n):
        for k in range(j, min(j + n, 2 * n - 1)):
            rows.append(k)
            cols.append(j)
            vals.append(a[k - j])
    return coo_matrix((vals, (rows, cols)), shape=(2 * n - 1, n)).tocsr()


def solve_lp_with_dual(a, M, time_limit=10):
    """Solve LP: max sum(g) s.t. conv(a,g) <= M, g >= 0. Returns (g, duals) or (None, None)."""
    n = len(a)
    T = build_toeplitz_sparse(a)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c, A_ub=T, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': time_limit})
        if res.status == 0:
            g = np.maximum(res.x, 0)
            duals = None
            try:
                if hasattr(res.ineqlin, 'marginals'):
                    duals = np.maximum(res.ineqlin.marginals, 0)
                elif hasattr(res, 'lambdas_lin'):
                    duals = np.maximum(res.lambdas_lin, 0)
            except Exception:
                pass
            return g, duals
    except Exception:
        pass
    return None, None


def dual_weighted_grad(a, duals, conv):
    """Compute gradient using LP dual variables as weights."""
    n = len(a)
    total_d = np.sum(duals)
    if total_d < 1e-15:
        # Fallback: uniform weights on top-K
        k_active = np.argsort(conv)[-20:]
        w = np.zeros(2 * n - 1)
        w[k_active] = 1.0 / len(k_active)
    else:
        w = duals / total_d

    # grad[j] = 2 * sum_k w[k] * a[k-j] for valid k-j
    # Efficient: this is a cross-correlation
    size = 1
    while size < 2 * n:
        size <<= 1
    fw = np.fft.rfft(w, size)
    fa = np.fft.rfft(a, size)
    # cross_corr[lag] = sum_m w[m+lag] * a[m]
    # = ifft(fw * conj(fa))[lag]... let me verify
    # Actually: ifft(fw * conj(fa))[l] = sum_m w[m] * conj(a[m-l])*... 
    # Let me just use the direct approach for correctness
    # grad[j] = 2 * sum_{m=0}^{n-1} w[j+m] * a[m] (where j+m < 2n-1)
    grad = np.zeros(n)
    # Use convolution: grad = 2 * cross_correlate(w, a)
    # cross_correlate(w, a)[j] = sum_m w[j+m] * a[m]
    # This equals convolve(w_reversed, a)[j + len(w) - 1 - (len(a)-1)]... 
    # Let me just do it with FFT properly
    # cross_correlate(w, a) = ifft(conj(fft(a)) * fft(w))... 
    # Standard: cross_correlate(w,a)[lag] = sum_m w[m] * a[m+lag]
    # = ifft(fft(w) * conj(fft(a)))[lag + n_a - 1]... 
    # Let me just compute directly for clarity (n <= 2000, this is O(n^2) = 4M ops, fine)
    for j in range(n):
        idx = np.arange(0, min(n, 2 * n - 1 - j))
        grad[j] = 2.0 * np.dot(w[j:j + len(idx)], a[idx])
    return grad


def lp_dual_optimize(a, max_iters=100, time_budget=100.0):

    a = a.copy().astype(np.float64)

    if S < 1e-10:
        return a, np.inf
    a = a / S

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    stall = 0

    for it in range(max_iters):
        if time.monotonic() - start > time_budget:
            break
        remaining = time_budget - (time.monotonic() - start)
        if remaining < 2:
            break

        conv = compute_conv(a)
        M = float(np.max(conv))

        # Adaptive M: try current M, slightly larger, slightly smaller
        best_g = None
        best_duals = None
        best_S_g = 0

        for M_try in M_candidates:
            if time.monotonic() - start > time_budget:
                break
            g, duals = solve_lp_with_dual(a, M_try, time_limit=min(6, remaining * 0.3))
            if g is not None:
                S_g = np.sum(g)
                if S_g > best_S_g:
                    best_S_g = S_g
                    best_g = g.copy()
                    best_duals = duals

        if best_g is None or best_S_g < 1e-10:
            stall += 1
            if stall > 8:
                break
            # Perturb
            rng = np.random.default_rng(it * 7919 + 13)
            a = a * np.exp(rng.normal(0, 0.08, n))
            a = np.maximum(a, 0)
            a = a / np.sum(a)
            stall = 0
            continue

        g = best_g / best_S_g

        # Line search
        best_t = 0.0
        best_obj_ls = best_obj
        t_candidates = [0.1, 0.2, 0.3, 0.5, 0.7, 0.9, 1.0]
        for t in t_candidates:
            a_try = (1.0 - t) * a + t * g
            a_try = np.maximum(a_try, 0)
            S_try = np.sum(a_try)
            if S_try < 1e-10:
                continue
            a_try = a_try / S_try
            obj_try = compute_obj(a_try)
            if obj_try < best_obj_ls:
                best_obj_ls = obj_try
                best_t = t

        if best_t > 0.0:
            a = (1.0 - best_t) * a + best_t * g
            a = np.maximum(a, 0)
            a = a / np.sum(a)
            best_obj = best_obj_ls
            best_a = a.copy()
            stall = 0
        else:
            # Use dual-weighted gradient step
            stall += 1
            if best_duals is not None and stall <= 4:

                grad = grad - np.mean(grad)
                # Adaptive step size
                step = 0.05 / (1.0 + 0.05 * stall)
                a_new = a - step * grad
                a_new = np.maximum(a_new, 0)
                S_new = np.sum(a_new)
                if S_new > 1e-10:
                    a_new = a_new / S_new
                    obj_new = compute_obj(a_new)
                    if obj_new < best_obj:
                        best_obj = obj_new
                        best_a = a_new.copy()
                        a = a_new
                        stall = 0
                    else:
                        a = 0.95 * a + 0.05 * g
                        a = np.maximum(a, 0)
                        a = a / np.sum(a)
                else:
                    stall = 5
            else:
                if stall > 6:
                    # Perturb
                    rng = np.random.default_rng(it * 4213 + 7)
                    a = a * np.exp(rng.normal(0, 0.12, n))
                    a = np.maximum(a, 0)
                    a = a / np.sum(a)
                    stall = 0
                else:
                    a = 0.95 * a + 0.05 * g
                    a = np.maximum(a, 0)
                    a = a / np.sum(a)

    return best_a, best_obj


def subgradient_polish(a, max_iters=1000, time_budget=20.0):
    """Fast subgradient descent with momentum for quick local improvement."""
    n = len(a)
    a = a.copy().astype(np.float64)
    S = np.sum(a)
    if S < 1e-10:
        return a, np.inf
    a = a / S

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    v = np.zeros(n)
    momentum = 0.85
    step = 0.1

    for it in range(max_iters):
        if time.monotonic() - start > time_budget:
            break

        conv = compute_conv(a)
        M = float(np.max(conv))

        # Top-K active set
        K = min(15, len(conv))
        active = np.argsort(conv)[-K:]

        grad = np.zeros(n)
        weights = conv[active] - conv[active[0]] + 1e-10
        weights = np.exp(weights / (TEMP if (TEMP := 0.1) else 0.1))
        weights /= weights.sum()

        for idx, k in enumerate(active):
            w = weights[idx]
            j_start = max(0, k - (n - 1))
            j_end = min(n, k + 1)
            for j in range(j_start, j_end):
                m = k - j
                if 0 <= m < n:
                    grad[j] += 2.0 * w * a[m]

        grad = grad - np.mean(grad)
        v = momentum * v + (1.0 - momentum) * grad
        a_new = a - step * v
        a_new = np.maximum(a_new, 0)
        S_new = np.sum(a_new)
        if S_new < 1e-10:
            break
        a_new = a_new / S_new

        obj = compute_obj(a_new)
        if obj < best_obj:
            best_obj = obj
            best_a = a_new.copy()
            a = a_new
            step = min(step * 1.01, 0.2)
        else:
            step *= 0.92
            if step < 1e-7:
                break

    return best_a, best_obj


def cosine_init(n, n_modes=10, seed=0):
    """Cosine series initialization."""
    rng = np.random.default_rng(seed)
    k = np.arange(n)
    a = np.ones(n)
    for m in range(1, n_modes + 1):
        c = rng.uniform(-0.5, 0.5) / m
        phase = rng.uniform(0, 2 * np.pi)
        a += c * np.cos(2 * np.pi * m * k / n + phase)
    a = np.maximum(a, 0)
    S = np.sum(a)
    if S < 1e-10:
        return np.ones(n) / n
    return a / S


def smooth_noise_init(n, sigma=0.3, seed=0):

    # Generate smooth random function via Fourier features

        phase = rng.uniform(0, 2 * np.pi)
        a += amp * np.cos(2 * np.pi * freq * k / n + phase)

    S = np.sum(a)

def interp_resample(a, n_new):
    """Resample sequence to new length via interpolation."""
    n_old = len(a)
    if n_old == n_new:
        return a.copy()
    x_old = np.linspace(0, 1, n_old)
    x_new = np.linspace(0, 1, n_new)
    try:
        f = interp1d(x_old, a, kind='cubic', fill_value='extrapolate')
        a_new = np.maximum(f(x_new), 0)
    except Exception:
        a_new = np.interp(x_new, x_old, a)
        a_new = np.maximum(a_new, 0)
    S = np.sum(a_new)
    if S < 1e-10:
        return np.ones(n_new) / n_new
    return a_new / S

    # === Phase 2: Multi-start exploration ===
    print("\n=== Phase 2: Multi-start exploration ===")
    n_options = sorted(set([
        max(100, n_parent - 150), max(100, n_parent - 75),
        n_parent, n_parent + 50, n_parent + 100,
        n_parent + 150, n_parent + 250, n_parent + 400
    ]))
    n_options = [n for n in n_options if 100 <= n <= 3000]

    candidates = []
    for n in n_options:
        if time.monotonic() > deadline - 90:
            break
        remaining = deadline - time.monotonic()
        if remaining < 20:
            break

        inits = []
        # Cosine inits
        inits.append(cosine_init(n, n_modes=8, seed=seed + n))
        inits.append(cosine_init(n, n_modes=14, seed=seed + n + 10000))
        # Smooth noise
        inits.append(smooth_noise_init(n, sigma=0.3, seed=seed + n + 20000))
        inits.append(smooth_noise_init(n, sigma=0.5, seed=seed + n + 30000))
        # Interpolated best
        if best_seq is not None and len(best_seq) != n:
            inits.append(interp_resample(best_seq, n))
        # Uniform
        inits.append(np.ones(n) / n)

        for ii, a_init in enumerate(inits):
            if time.monotonic() > deadline - 30:
                break
            obj_init = compute_obj(a_init)
            candidates.append((a_init, obj_init, n, ii))
            if obj_init < best_score - 1e-9:
                best_score = obj_init
                best_seq = a_init.copy()
                print(f"  n={n} init{ii}: {obj_init:.6f} ** NEW BEST **")

    # === Phase 3: Refine top candidates ===
    print("\n=== Phase 3: Refine top candidates ===")
    candidates.sort(key=lambda x: x[1])
    # Always include current best at top
    candidates.insert(0, (best_seq, best_score, len(best_seq), -1))

    n_refine = min(5, len(candidates))
    for idx in range(n_refine):
        if time.monotonic() > deadline - 60:
            break
        remaining = deadline - time.monotonic()
        if remaining < 25:
            break

        a_cand, obj_cand, n_cand, _ = candidates[idx]
        if idx > 0 and obj_cand > best_score + 0.03:
            break

        alloc = min(remaining * 0.25, 50.0)
        print(f"  Refine cand {idx} (n={n_cand}, obj={obj_cand:.6f}, alloc={alloc:.0f}s)")

        a_ref, obj_ref = refine_full(a_cand, time_budget=alloc)
        if obj_ref < best_score:
            best_score = obj_ref
            best_seq = a_ref.copy()
            print(f"    ** NEW BEST: n={n_cand}, obj={best_score:.6f} **")

    # === Phase 4: Intensive refinement of best ===
    print(f"\n=== Phase 4: Intensive refinement (current best={best_score:.6f}) ===")
    a_cur = best_seq.copy()
    n_best = len(a_cur)

    for rnd in range(30):
        if time.monotonic() > deadline - 20:
            break
        remaining = deadline - time.monotonic()
        if remaining < 10:
            break

        alloc = min(remaining * 0.15, 35.0)

        # Alternate LP and SG
        if rnd % 3 == 0:
            a_new, obj_new = lp_dual_optimize(a_cur, max_iters=50, time_budget=alloc)
        elif rnd % 3 == 1:
            a_new, obj_new = subgradient_polish(a_cur, max_iters=2000, time_budget=alloc)
            # Follow up with LP
            rem2 = deadline - time.monotonic()
            if rem2 > 10:
                a_new2, obj_new2 = lp_dual_optimize(a_new, max_iters=30, time_budget=min(rem2 * 0.3, 20.0))
                if obj_new2 < obj_new:
                    a_new, obj_new = a_new2, obj_new2
        else:
            # Perturbation + LP
            rng = np.random.default_rng(rnd * 777 + 111)
            scale = min(0.05 + 0.02 * (rnd % 5), 0.15)
            a_p = a_cur * np.exp(rng.normal(0, scale, n_best))
            a_p = np.maximum(a_p, 0)
            S_p = np.sum(a_p)
            if S_p > 1e-10:
                a_p = a_p / S_p
                a_new, obj_new = lp_dual_optimize(a_p, max_iters=30, time_budget=alloc)
            else:
                a_new, obj_new = a_cur, compute_obj(a_cur)

        if obj_new < best_score - 1e-9:
            best_score = obj_new
            best_seq = a_new.copy()
            a_cur = a_new.copy()
            print(f"  R{rnd}: {best_score:.6f} ** NEW BEST **")
        else:
            if rnd % 5 == 4:
                print(f"  R{rnd}: best={best_score:.6f} (got {obj_new:.6f})")
            # Try stronger perturbation
            remaining = deadline - time.monotonic()
            if remaining > 15:
                rng = np.random.default_rng(rnd * 555 + 999)
                for ps in [0.2, 0.4, 0.6]:
                    if time.monotonic() > deadline - 10:
                        break
                    a_p2 = a_cur * np.exp(rng.normal(0, ps, n_best))
                    a_p2 = np.maximum(a_p2, 0)
                    S_p2 = np.sum(a_p2)
                    if S_p2 < 1e-10:
                        continue
                    a_p2 = a_p2 / S_p2
                    a_p2, obj_p2 = lp_dual_optimize(a_p2, max_iters=20, time_budget=min(remaining * 0.08, 12.0))
                    if obj_p2 < best_score:
                        best_score = obj_p2
                        best_seq = a_p2.copy()
                        a_cur = a_p2.copy()
                        print(f"  R{rnd} pert({ps}): {best_score:.6f} ** NEW BEST **")
                        break

    # === Phase 5: Try different n for the best solution ===
    print(f"\n=== Phase 5: N-exploration (best={best_score:.6f}) ===")
    if time.monotonic() < deadline - 30:
        n_base = len(best_seq)
        n_tries = sorted(set([
            max(100, n_base - 100), max(100, n_base - 50),
            n_base + 50, n_base + 100, n_base + 200,
            n_base + 350, n_base + 500
        ]))
        n_tries = [n for n in n_tries if 100 <= n <= 3000 and n != n_base]

        for n_try in n_tries:
            if time.monotonic() > deadline - 15:
                break
            remaining = deadline - time.monotonic()
            if remaining < 15:
                break

            a_interp = interp_resample(best_seq, n_try)
            alloc = min(remaining * 0.2, 25.0)
            a_new, obj_new = refine_full(a_interp, time_budget=alloc)
            if obj_new < best_score:
                best_score = obj_new
                best_seq = a_new.copy()
                print(f"  n={n_try}: {best_score:.6f} ** NEW BEST **")
            else:
                print(f"  n={n_try}: {obj_new:.6f}")

    # === Phase 6: Final intensive polish ===
    print(f"\n=== Phase 6: Final polish (best={best_score:.6f}) ===")
    a_cur = best_seq.copy()
    if time.monotonic() < deadline - 5:
        remaining = deadline - time.monotonic()

        # Long LP run
        a_final, obj_final = lp_dual_optimize(a_cur, max_iters=100, time_budget=min(remaining * 0.4, 80.0))
        if obj_final < best_score:
            best_score = obj_final
            best_seq = a_final.copy()
            print(f"  Final LP: {best_score:.6f} ** NEW BEST **")

        # SG polish
        remaining = deadline - time.monotonic()
        if remaining > 8:
            a_sg, obj_sg = subgradient_polish(best_seq, max_iters=5000, time_budget=min(remaining * 0.4, 30.0))
            if obj_sg < best_score:
                best_score = obj_sg
                best_seq = a_sg.copy()
                print(f"  Final SG: {best_score:.6f} ** NEW BEST **")

        # One more LP
        remaining = deadline - time.monotonic()
        if remaining > 5:
            a_f2, obj_f2 = lp_dual_optimize(best_seq, max_iters=50, time_budget=min(remaining * 0.6, 30.0))
            if obj_f2 < best_score:
                best_score = obj_f2
                best_seq = a_f2.copy()
                print(f"  Final LP2: {best_score:.6f} ** NEW BEST **")

    # Final validation
    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    val = evaluate_sequence(result)
    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, evaluate={val:.6f} (internal={best_score:.6f}), time={elapsed:.1f}s ===")

    if val == np.inf or not np.isfinite(val):
        print("WARNING: evaluate returned inf, using fallback")
        result = [1.0] * 300
        val = evaluate_sequence(result)

    return result
