    return 2.0 * n * float(np.max(c)) / (S * S)


def proj_simplex(v):
    """Project v onto {x >= 0, sum(x) = 1}."""
    n = len(v)
    u = np.sort(v)[::-1]
    cv = np.cumsum(u) - 1.0
    cond = u - cv / np.arange(1, n + 1) > 0
    if not np.any(cond):
        return np.ones(n) / n
    rho = np.max(np.where(cond)[0])
    theta = cv[rho] / (rho + 1)
    return np.maximum(v - theta, 0)


def subgradient_active(a, c, threshold_frac=0.01):
    """Compute subgradient using all constraints within threshold of max."""
    n = len(a)
    M = np.max(c)
    active = c >= M * (1 - threshold_frac)
    if not np.any(active):
        active = c == M
    
    grad = np.zeros(n)
    for k in np.where(active)[0]:
        # c_k = sum_j a[j]*a[k-j]
        # d(c_k)/d(a_i) = a[k-i] + a[k-i] (if k-i != i) or 2*a[i] (if k-i == i)
        # More precisely: d(c_k)/d(a_i) = a[k-i] if 0<=k-i<n, plus a[k-i] from the other term
        # Actually for convolution: c_k = sum_j a[j]*a[k-j]
        # d(c_k)/d(a_i) = a[k-i] (from j=i term) + a[k-i] (from k-j=i term, i.e., j=k-i)
        # = 2*a[k-i] if 0 <= k-i < n and k-i != i
        # = 2*a[i] if k-i == i (i.e., k = 2i)
        # = a[k-i] if only one of j=i or j=k-i is valid
        
        # Simpler: d(c_k)/d(a_i) = a[k-i] if 0<=k-i<n (from j=i)
        #                            + a[i] if 0<=i<n and k-i==i... no
        # Let me just use: d(c_k)/d(a_i) = a[max(0, min(n-1, k-i))] if 0 <= k-i < n else 0
        # Wait, that's not right either.
        
        # For c_k = sum_{j=max(0,k-n+1)}^{min(n-1,k)} a[j]*a[k-j]:
        # d(c_k)/d(a_i) = a[k-i] (if i is in the valid range for j) + a[k-i] (if k-i is in the valid range for j, i.e., if i is in the valid range for k-j)
        # = 2*a[k-i] if max(0,k-n+1) <= i <= min(n-1,k) AND max(0,k-n+1) <= k-i <= min(n-1,k)
        # = a[k-i] if only one condition holds
        # = 0 if neither holds
        
        # For i in [0, n-1] and k in [0, 2n-2]:
        # j = i is valid iff max(0, k-n+1) <= i <= min(n-1, k), which is always true for 0<=i<n, 0<=k<2n-1
        # (since k-n+1 <= k and 0 <= n-1)
        # Actually: max(0, k-n+1) <= i requires i >= 0 (always true) and i >= k-n+1
        # min(n-1, k) >= i requires i <= n-1 (always true) and i <= k
        # So j=i is valid iff k-n+1 <= i <= k, i.e., i in [max(0,k-n+1), min(n-1,k)]
        
        # Similarly, j = k-i is valid iff k-i in [max(0,k-n+1), min(n-1,k)]
        # = k - min(n-1,k) <= i <= k - max(0,k-n+1)
        # = max(0, k-(n-1)) <= i <= min(n-1, k)
        # Hmm, this is getting complicated. Let me just use the simple formula:
        # d(c_k)/d(a_i) = a[k-i] if 0 <= k-i < n else 0
        # This counts the j=i term.
        # Plus: d(c_k)/d(a_i) += a[i] if 0 <= k-i < n and k-i != i... no.
        
        # Actually, the correct formula is:
        # c_k = sum_j a[j] * a[k-j]
        # d(c_k)/d(a_i) = a[k-i] (from the term where j=i) + sum over terms where k-j=i, i.e., j=k-i: a[k-i] * 1
        # Wait, the second term: when j = k-i, the term is a[k-i]*a[i], and d/d(a_i) of this is a[k-i].
        # So if k-i is a valid j (i.e., 0 <= k-i < n and k-(k-i) = i is also a valid index in the range), then we get another a[k-i].
        
        # Let me just be very explicit:
        # c_k = sum_j a[j]*a[k-j] where j ranges over {0,...,n-1} intersected with {k-(n-1),...,k}
        # For a specific i:
        #   Term 1: j=i, contributes a[i]*a[k-i] if 0<=k-i<n. Derivative wrt a[i]: a[k-i]
        #   Term 2: k-j=i, i.e., j=k-i. Contributes a[k-i]*a[i] if 0<=k-i<n. Derivative wrt a[i]: a[k-i]
        #   If i == k-i (i.e., k=2i): Terms 1 and 2 are the same term (a[i]*a[i]), so derivative is 2*a[i]
        #   If i != k-i: derivative is a[k-i] + a[k-i] = 2*a[k-i] (if both are valid)
        #   If only one is valid: derivative is a[k-i] or a[i]
        
        # For 0 <= i < n and 0 <= k < 2n-1:
        #   k-i is in [-(n-1), 2n-2-i]
        #   k-i >= 0 iff k >= i
        #   k-i < n iff k < i+n
        #   So 0 <= k-i < n iff i <= k < i+n
        
        # The valid range for j in the convolution sum is [max(0, k-n+1), min(n-1, k)]
        # j = i is valid iff max(0,k-n+1) <= i <= min(n-1,k)
        #   Since 0 <= i <= n-1 always, this simplifies to: k-n+1 <= i <= k
        #   i.e., i >= max(0, k-n+1) and i <= min(n-1, k) [the second is automatic if i <= n-1 and k >= 0]
        #   Actually: i <= k (needed) and i >= k-n+1 (needed)
        #   So: max(0, k-n+1) <= i <= min(n-1, k)
        
        # j = k-i is valid iff max(0,k-n+1) <= k-i <= min(n-1,k)
        #   k-i >= 0 iff k >= i (same as above)
        #   k-i <= n-1 iff k-i < n iff i > k-n iff i >= k-n+1 (same as above)
        #   k-i >= 0 iff i <= k (same)
        #   k-i <= n-1: k-i <= n-1 iff i >= k-(n-1)
        #   Also k-i >= 0: i <= k
        #   So: k-(n-1) <= i <= k, intersected with [0, n-1]
        #   = max(0, k-n+1) <= i <= min(n-1, k)
        
        # So both j=i and j=k-i are valid for the same range of i!
        # Therefore: d(c_k)/d(a_i) = 2*a[k-i] if max(0,k-n+1) <= i <= min(n-1,k) and i != k-i
        #                               = 2*a[i] if i = k-i (i.e., k=2i) [same formula since a[k-i]=a[i]]
        #                               = 0 otherwise
        
        # Great, so the formula is simply:
        # d(c_k)/d(a_i) = 2*a[k-i] if 0 <= k-i < n else 0
        # (which automatically handles the k=2i case correctly)
        
        j = k - i
        if 0 <= j < n:
            grad[i] += 2.0 * a[j]
    
    # Normalize by number of active constraints
    n_active = np.sum(active)
    if n_active > 0:
        grad /= n_active
    
    return grad


def bb_step(a_prev, a_curr, grad_prev, grad_curr):
    """Barzilai-Borwein step size."""
    s = a_curr - a_prev
    y = grad_curr - grad_prev
    ss = np.dot(s, s)
    sy = np.dot(s, y)
    if ss < 1e-15:
        return 0.01
    if y > 1e-10:
        return ss / sy
    elif y < -1e-10:
        return -ss / sy
    return 0.01


def projected_subgradient(a, max_iters=5000, time_budget=60.0, verbose=False, seed=42):
    """Projected subgradient descent with BB step size."""
    n = len(a)
    a = a.copy().astype(np.float64)
    a = proj_simplex(a)
    
    start = time.monotonic()
    best_a = a.copy()
    best_obj = compute_obj(a)
    rng = np.random.default_rng(seed)
    
    a_prev = a.copy()
    grad_prev = np.zeros(n)
    step = 0.01
    stall = 0
    stall_limit = 200
    
    for it in range(max_iters):
        if time.monotonic() - start > time_budget:
            break
        
        c = compute_conv(a)
        M = float(np.max(c))
        obj = 2.0 * n * M  # since sum(a)=1, obj = 2n*M
        
        if obj < best_obj - 1e-12:
            best_obj = obj
            best_a = a.copy()
            stall = 0
        else:
            stall += 1
        
        if stall > stall_limit:
            # Perturb
            scale = rng.uniform(0.05, 0.15)
            a = a * np.exp(rng.normal(0, scale, n))
            a = proj_simplex(a)
            stall = 0
            step = max(step * 0.5, 0.005)
            continue
        
        # Compute subgradient
        threshold = 0.005
        grad = subgradient_active(a, c, threshold_frac=threshold)
        
        # BB step
        if it > 0:
            step_bb = bb_step(a_prev, a, grad_prev, grad)
            if step_bb > 0 and np.isfinite(step_bb):
                step = 0.1 * step_bb + 0.9 * step  # smooth
                step = min(step, 0.1)
        
        a_prev = a.copy()
        grad_prev = grad.copy()
        
        # Update
        a_new = a - step * grad
        a = proj_simplex(a_new)
    
    return best_a, best_obj


def solve_lp_polish(a, M, time_limit=10):
    """Solve LP: max sum(g) s.t. conv(a,g) <= M, g >= 0."""
    n = len(a)
    rows = []
    cols = []
    vals = []
    for j in range(n):
        k_lo = j
        k_hi = min(j + n - 1, 2 * n - 2)
        rows.extend(range(k_lo, k_hi + 1))
        cols.extend([j] * (k_hi - k_lo + 1))
        vals.extend(list(a[:k_hi - k_lo + 1]))
    
    A = coo_matrix((vals, (rows, cols)), shape=(2 * n - 1, n)).tocsr()
    c_obj = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c_obj, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': time_limit})
        if res.status == 0:
            g = np.maximum(res.x, 0)
            S = np.sum(g)
            if S > 1e-10:
                return g / S
    except Exception:
        pass
    return None


def alternating_lp(a, max_iters=30, time_budget=30.0):
    """Alternating LP method."""
    n = len(a)
    a = proj_simplex(a)
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    stall = 0
    
    for it in range(max_iters):
        if time.monotonic() - start > time_budget:
            break
        
        c = compute_conv(a)
        M = float(np.max(c))
        
        g = solve_lp_polish(a, M, time_limit=8)
        if g is None:
            break
        
        # Line search
        best_t = 0.0
        best_obj_ls = best_obj
        for t in [0.1, 0.2, 0.3, 0.5, 0.7, 1.0]:
            a_try = proj_simplex((1 - t) * a + t * g)
            obj_try = compute_obj(a_try)
            if obj_try < best_obj_ls:
                best_obj_ls = obj_try
                best_t = t
        
        if best_t > 0:
            a = proj_simplex((1 - best_t) * a + best_t * g)
            best_obj = best_obj_ls
            best_a = a.copy()
            stall = 0
        else:
            stall += 1
            if stall > 10:
                rng = np.random.default_rng(it * 104729 + 1)
                a = a * np.exp(rng.normal(0, 0.08, n))
                a = proj_simplex(a)
                stall = 0
    
    return best_a, best_obj


def cosine_init(n, n_modes=8, seed=0):
    rng = np.random.default_rng(seed)
    k = np.arange(n)
    a = np.ones(n)
    for m in range(1, n_modes + 1):
        c = rng.uniform(-0.5, 0.5) / m
        phase = rng.uniform(0, 2 * np.pi)
        a += c * np.cos(2 * np.pi * m * k / n + phase)
    a = np.maximum(a, 1e-10)
    return proj_simplex(a)


def raised_cosine_init(n, alpha=1.0):
    k = np.arange(n)
    x = 2.0 * k / (n - 1) - 1.0
    a = 0.5 * (1.0 + np.cos(np.pi * x)) ** alpha
    a = np.maximum(a, 1e-10)
    return proj_simplex(a)


def gaussian_init(n, sigma_frac=0.3):
    k = np.arange(n)
    center = n / 2.0
    sigma = sigma_frac * n
    a = np.exp(-0.5 * ((k - center) / sigma) ** 2)
    return proj_simplex(a)


def random_init(n, seed=0):
    rng = np.random.default_rng(seed)
    a = rng.dirichlet(np.ones(n))
    return a


def refine_sequence(a, time_budget, verbose=False):
    """Full refinement pipeline."""
    n = len(a)
    a = proj_simplex(a.copy())
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    round_num = 0
    
    while time.monotonic() - start < time_budget - 3:
        round_num += 1
        remaining = time_budget - (time.monotonic() - start)
        if remaining < 5:
            break
        
        # Subgradient phase
        sg_t = min(remaining * 0.35, 30.0)
        a_sg, obj_sg = projected_subgradient(best_a, max_iters=int(sg_t * 30),
                                              time_budget=sg_t, seed=round_num)
        if obj_sg < best_obj:
            best_obj = obj_sg
            best_a = a_sg.copy()
            if verbose:
                print(f"    R{round_num} PGD: {best_obj:.6f}")
        
        # LP phase
        remaining = time_budget - (time.monotonic() - start)
        if remaining > 15 and n <= 600:
            lp_t = min(remaining * 0.3, 30.0)
            a_lp, obj_lp = alternating_lp(best_a, max_iters=20, time_budget=lp_t)
            if obj_lp < best_obj:
                best_obj = obj_lp
                best_a = a_lp.copy()
                if verbose:
                    print(f"    R{round_num} LP: {best_obj:.6f}")
        
        # Perturbation
        remaining = time_budget - (time.monotonic() - start)
        if remaining > 10:
            rng = np.random.default_rng(round_num * 7919)
            for scale in [0.05, 0.1, 0.2]:
                if time.monotonic() - start > time_budget - 5:
                    break
                a_p = best_a * np.exp(rng.normal(0, scale, n))
                a_p = proj_simplex(a_p)
                a_p, obj_p = projected_subgradient(a_p, max_iters=1000, time_budget=2.0,
                                                    seed=round_num * 100 + int(scale*100))
                if obj_p < best_obj:
                    best_obj = obj_p
                    best_a = a_p.copy()
                    if verbose:
                        print(f"    R{round_num} Pert({scale}): {best_obj:.6f}")
    
    return best_a, best_obj

Let me optimize the subgradient computation using numpy:

def subgradient_fast(a, c, threshold=0.01):
    """Vectorized subgradient computation."""
    n = len(a)
    M = np.max(c)
    active = c >= M * (1 - threshold)
    active_idx = np
