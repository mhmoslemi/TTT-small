import numpy as np
from scipy.optimize import linprog
from scipy.sparse import csr_matrix
import time

def fast_conv(a):
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]

def compute_J(a):
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))

def build_conv_matrix(n, a):
    """Build sparse matrix A where (A @ g)[k] = (a*g)[k]"""
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        j_min = max(0, k - n + 1)
        j_max = min(n - 1, k)
        for j in range(j_min, j_max + 1):
            rows.append(k)
            cols.append(j)
            vals.append(a[k - j])
    return csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))

def solve_lp(a, M):
    """Solve: max sum(g) s.t. (a*g)[k] <= M, g >= 0"""
    n = len(a)
    A = build_conv_matrix(n, a)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    try:
        r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                    options={'time_limit': 10, 'presolve': True})
        if r.success and r.x is not None:
            return np.maximum(r.x, 0)
    except Exception:
        pass
    return None

def lp_mirror_descent(a, deadline, max_iters=100000, verbose=True):
    """Aggressive LP mirror descent with adaptive step size and restarts."""
    a = np.maximum(a.astype(np.float64), 1e-10)
    a = a / a.sum()
    n = len(a)
    
    best_a = a.copy()
    best_J = compute_J(a)
    no_improve = 0
    restart_count = 0
    
    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        
        b = fast_conv(a)
        M = float(b.max())
        
        g = solve_lp(a, M)
        if g is None or g.sum() < 1e-10:
            break
        
        g = g / g.sum()
        
        # Adaptive step size with backtracking
        improved = False
        # Try a range of step sizes
        base_t = 0.5 if it < 100 else 0.3
        for t in [base_t, base_t*0.5, base_t*0.25, 0.1, 0.05, 0.02, 0.01]:
            a_trial = (1 - t) * a + t * g
            a_trial = np.maximum(a_trial, 0)
            s = a_trial.sum()
            if s < 1e-10:
                continue
            a_trial = a_trial / s
            J_trial = compute_J(a_trial)
            if J_trial < best_J - 1e-12:
                a = a_trial
                best_J = J_trial
                best_a = a.copy()
                improved = True

import numpy as np
from scipy.optimize import linprog
from scipy.sparse import csr_matrix
from scipy.signal import fftconvolve

def fast_conv(a):
    """Full self-convolution via FFT."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]


def fast_cross_conv(a, b):
    """Cross-convolution of a and b via FFT."""
    n = len(a)
    m = len(b)
    size = 1
    while size < n + m - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    return np.fft.irfft(fa * fb, size)[:n + m - 1]


def compute_J(a):
    """Compute J(a) = 2*n*max(conv(a,a))/(sum(a))^2."""
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))


def build_lp_matrix_full(n, a):
    """Build full sparse convolution matrix A where (A@g)[k] = (a*g)[k]."""
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        j_min = max(0, k - n + 1)
        j_max = min(n - 1, k)
        for j in range(j_min, j_max + 1):
            rows.append(k)
            cols.append(j)
            vals.append(a[k - j])
    return csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))

def smooth_max_gradient(a, tau):
    """Compute gradient of smooth max approximation."""
    n = len(a)
    b = fast_conv(a)
    b_max = b.max()
    exp_b = np.exp(tau * (b - b_max))
    p = exp_b / exp_b.sum()
    
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        pk = p[k]
        if pk < 1e-14:
            continue
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        seg = a[k - hi: k - lo + 1][::-1]
        grad[lo: hi + 1] += 2.0 * pk * seg
    return grad, p, b

        s = a_new.sum()
        if s > 1e-10:
            a = a_new / s
    return best_a


def make_palindrome(a):
    """Enforce palindrome symmetry: a[i] = a[n-1-i]."""
    n = len(a)
    half = (n + 1) // 2
    a_new = np.zeros(n)
    for i in range(half):
        val = (a[i] + a[n - 1 - i]) / 2.0
        a_new[i] = val
        a_new[n - 1 - i] = val
    return a_new


def resample_seq(src, dest_n):
    """Resample 1D sequence to new length."""
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)


class ScoreTracker:
    def __init__(self):
        self.best = None
        self.best_score = float('inf')
        self.t_start = time.monotonic()

    def update(self, a, label=""):
        if a is None or a.size == 0:
            return
        a = np.clip(a.astype(np.float64), 0.0, 1000.0)
        if a.sum() < 0.01:
            return
        score = compute_J(a)
        if score < self.best_score:
            self.best = a.copy()
            self.best_score = score
            elapsed = time.monotonic() - self.t_start
            msg = f"  ** J={score:.8f} n={a.size} t={elapsed:.0f}s"
            if label:
                msg += f" [{label}]"
            print(msg, flush=True)

    def report(self, label=""):
        if self.best is not None:
            print(f"  {label} best: J={self.best_score:.8f} n={self.best.size}", flush=True)


def lp_mirror_descent_aggressive(a, deadline, verbose=True, use_reduced_lp=True,
                                  top_frac=0.6, max_no_improve=300):
    """
    Aggressive LP mirror descent with:
    - Adaptive step size (geometric backtracking)
    - Reduced LP for speed
    - Periodic structured perturbation
    - Dual-gap based convergence detection
    """
    a = np.maximum(a.astype(np.float64), 1e-10)
    a = a / a.sum()
    n = len(a)
    
    best_a = a.copy()
    best_J = compute_J(a)
    no_improve = 0
    restart_count = 0
    lp_time_total = 0.0
    it = 0
    
    while time.monotonic() < deadline:
        it += 1
        t0 = time.monotonic()
        
        b = fast_conv(a)
        M = float(b.max())
        M_min_possible = a.sum() ** 2 / (2 * n - 1)
        dual_gap = M - M_min_possible
        
        # Solve LP
        if use_reduced_lp:
            g = solve_lp_reduced(a, M, top_frac=top_frac)
        else:
            g = solve_lp_full(a, M)
        
        lp_time_total += time.monotonic() - t0
        
        if g is None or g.sum() < 1e-10:
            break
        
        g = g / g.sum()
        lp_obj = g.sum()  # after normalization this is 1, but before it was > 1
        
        # Check if we're close to fixed point
        diff = np.max(np.abs(g - a))
        
        # Line search with multiple step sizes
        improved = False
        # Adaptive base step: larger when far from fixed point
        base_t = min(0.8, max(0.1, diff * 2.0))
        t_candidates = [base_t, base_t * 0.5, base_t * 0.25,
                        base_t * 0.125, 0.05, 0.02, 0.01]
        
        current_J = compute_J(a)
        
        for t in t_candidates:
            a_trial = (1.0 - t) * a + t * g
            a_trial = np.maximum(a_trial, 0.0)
            s = a_trial.sum()
            if s < 1e-10:
                continue
            a_trial = a_trial / s
            J_trial = compute_J(a_trial)
            
            if J_trial < best_J - 1e-13:
                a = a_trial
                best_J = J_trial
                best_a = a.copy()
                improved = True
                no_improve = 0
                break
            elif J_trial < current_J - 1e-13:
                a = a_trial
                current_J = J_trial
                no_improve = 0
                break
        
        if not improved:
            no_improve += 1
        
        # Periodic perturbation to escape local minima
        if no_improve > max_no_improve:
            restart_count += 1
            # Structured perturbation: mix with a random smooth sequence
            rng = np.random.default_rng(it * 7919 + restart_count)
            scale = 0.15 / (1.0 + 0.3 * restart_count)
            
            # Generate a smooth random perturbation
            freqs = rng.integers(1, 5, size=3)
            phases = rng.uniform(0, 2 * np.pi, size=3)
            amplitudes = rng.uniform(0.3, 1.0, size=3)
            t_arr = np.linspace(0, 1, n, endpoint=False)
            pert = np.zeros(n)
            for f, ph, amp in zip(freqs, phases, amplitudes):
                pert += amp * np.cos(2 * np.pi * f * t_arr + ph)
            
            a = a + scale * pert
            a = np.maximum(a, 1e-6)
            a = a / a.sum()
            no_improve = 0
            
            if verbose and restart_count % 5 == 0:
                print(f"    Restart #{restart_count} at iter {it}, best_J={best_J:.8f}", flush=True)
        
        # Very slow convergence: try even more aggressive perturbation
        if no_improve > max_no_improve * 3:
            restart_count += 1
            rng = np.random.default_rng(it * 104729 + restart_count)
            scale = 0.3 / (1.0 + 0.2 * restart_count)
            a = a * (1.0 + scale * rng.normal(0, 1, n))
            a = np.maximum(a, 1e-6)
            a = a / a.sum()
            no_improve = 0
        
        if verbose and it % 2000 == 0:
            avg_lp_time = lp_time_total / it * 1000
            print(f"    iter={it} J={best_J:.8f} M={M:.6f} diff={diff:.4f} "
                  f"lp_time={avg_lp_time:.1f}ms restarts={restart_count}", flush=True)
        
        if it > 500000:
            break
    
    return best_a, best_J


def generate_diverse_inits(n, rng, parent=None):
    """Generate diverse initial sequences."""
    inits = []
    t_arr = np.linspace(0, 1, n, endpoint=False)
    
    if parent is not None:
        p = resample_seq(parent, n)
        inits.append(p.copy())
        inits.append(np.maximum(p + rng.normal(0, 0.05, n), 0.01))
        inits.append(np.maximum(p + rng.normal(0, 0.15, n), 0.01))
        # Palindrome version
        inits.append(make_palindrome(p))
    
    inits.append(np.ones(n))
    
    # Cosine bumps (concentrated in center)
    inits.append(np.maximum(0.5 + 0.5 * np.cos(np.pi * (2 * t_arr - 1)), 0.01))
    inits.append(np.maximum(0.3 + 0.7 * np.exp(-4 * (t_arr - 0.5) ** 2), 0.01))
    
    # Multi-frequency
    for f in [2, 3, 5]:
        inits.append(np.maximum(1.0 + 0.3 * np.cos(2 * np.pi * f * t_arr), 0.01))
    
    # Edge-concentrated
    inits.append(np.maximum(np.exp(-3 * t_arr ** 2) + np.exp(-3 * (t_arr - 1) ** 2) + 0.1, 0.01))
    
    # Block structure
    n_blocks = max(4, n // 64)
    block_vals = rng.uniform(0.3, 1.0, n_blocks)
    inits.append(np.repeat(block_vals, n // n_blocks + 1)[:n])
    
    # Smooth random
    ksize = max(3, n // 32)
    kernel = np.ones(ksize) / ksize
    smooth = np.convolve(rng.uniform(0, 1, n), kernel, mode='same')
    inits.append(np.maximum(smooth + 0.1, 0.01))
    
    return inits


def propose_candidate(seed=42, budget_s=1500, **kwargs):
    """
    Multi-phase search for optimal step function heights:
    Phase 1: Diverse initialization with quick smooth gradient (150s)
    Phase 2: Aggressive LP mirror descent with restarts (1100s)
    Phase 3: Multi-length exploration (150s)
    Phase 4: Final polishing (remaining time)
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + float(budget_s) - 20.0
    tracker = ScoreTracker()
    
    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        if p.ndim == 1 and p.size > 0 and np.all(np.isfinite(p)):
            parent = np.maximum(p, 0.0)
            tracker.update(parent, "parent")
            print(f"Parent: n={parent.size}, J={tracker.best_score:.8f}", flush=True)
    except (NameError, AttributeError):
        pass
    
    if parent is None:
        parent = np.ones(256)
        tracker.update(parent, "default")
    
    # === PHASE 1: Diverse initialization ===
    print(f"\n{'=' * 60}", flush=True)
    print("PHASE 1: Diverse Initialization + Quick Gradient", flush=True)
    print(f"{'=' * 60}", flush=True)
    
    p1_deadline = time.monotonic() + 150.0
    n_main = 512
    
    candidates = []
    inits = generate_diverse_inits(n_main, rng, parent)
    
    for idx, a0 in enumerate(inits):
        if time.monotonic() > p1_deadline:
            break
        a0 = np.maximum(a0, 1e-6).astype(np.float64)
        a0 = a0 / a0.sum()
        
        # Quick smooth gradient
        a_opt = run_smooth_gradient(a0, 80, p1_deadline, tau=15.0, lr=0.15)
        J_val = compute_J(a_opt)
        candidates.append((J_val, a_opt))
        print(f"  init {idx:2d}: J={J_val:.8f}", flush=True)
    
    # Sort by quality
    candidates.sort(key=lambda x: x[0])
    
    # Update tracker with best candidates
    for J_val, a_opt in candidates[:3]:
        tracker.update(a_opt, "phase1")
    
    print(f"Phase 1 best: J={candidates[0][0]:.8f}", flush=True)
    
    # === PHASE 2: Aggressive LP Mirror Descent ===
    print(f"\n{'=' * 60}", flush=True)
    print("PHASE 2: Aggressive LP Mirror Descent", flush=True)
    print(f"{'=' * 60}", flush=True)
    
    p2_deadline = time.monotonic() + 1050.0
    p2_start = time.monotonic()
    
    # Try top 3 candidates with LP mirror descent
    # Allocate time: best gets most, others get less
    time_alloc = [0.5, 0.3, 0.2]
    
    top_candidates = candidates[:min(3, len(candidates))]
    
    for ci, (J_init, a_init) in enumerate(top_candidates):
        if time.monotonic() > p2_deadline:
            break
        
        alloc_time = (p2_deadline - p2_start) * time_alloc[ci] if ci < len(time_alloc) else 60.0
        alloc_time = max(60.0, alloc_time)
        phase_dl = min(time.monotonic() + alloc_time, p2_deadline)
        
        print(f"\n  Candidate {ci} (init J={J_init:.8f}), time_budget={alloc_time:.0f}s", flush=True)
        
        a_best, J_best = lp_mirror_descent_aggressive(
            a_init, phase_dl, verbose=True,
            use_reduced_lp=True, top_frac=0.6, max_no_improve=200
        )
        tracker.update(a_best, f"lp_cand{ci}")
    
    # === PHASE 3: Multi-length exploration ===
    print(f"\n{'=' * 60}", flush=True)
    print("PHASE 3: Multi-length Exploration", flush=True)
    print(f"{'=' * 60}", flush=True)
    
    p3_deadline = min(time.monotonic() + 150.0, deadline - 80.0)
    
    if tracker.best is not None and time.monotonic() < p3_deadline:
        base_n = tracker.best.size
        # Try nearby lengths
        for n_try in [base_n, base_n + 64, base_n - 64, base_n * 2]:
            if time.monotonic() > p3_deadline:
                break
            if n_try < 64 or n_try > 2048:
                continue
            if n_try == base_n:
                continue
            
            a_ext = resample_seq(tracker.best, n_try)
            a_ext = np.maximum(a_ext, 1e-6)
            a_ext = a_ext / a_ext.sum()
            
            # Quick gradient + short LP
            a_ext = run_smooth_gradient(a_ext, 50, min(time.monotonic() + 30, p3_deadline - 20),
                                         tau=10, lr=0.12)
            tracker.update(a_ext, f"n={n_try}_grad")
            
            if time.monotonic() < p3_deadline - 10:
                a_ext2, _ = lp_mirror_descent_aggressive(
                    a_ext, min(time.monotonic() + 40, p3_deadline - 5),
                    verbose=False, use_reduced_lp=True, top_frac=0.7, max_no_improve=100
                )
                tracker.update(a_ext2, f"n={n_try}_lp")
        
        # Also try palindrome-constrained version
        if time.monotonic() < p3_deadline - 30:
            a_pal = make_palindrome(tracker.best)
            a_pal = np.maximum(a_pal, 1e-6)
            a_pal = a_pal / a_pal.sum()
            a_pal, _ = lp_mirror_descent_aggressive(
                a_pal, min(time.monotonic() + 50, p3_deadline - 10),
                verbose=False, use_reduced_lp=True, top_frac=0.7, max_no_improve=150
            )
            tracker.update(a_pal, "palindrome")
    
    tracker.report("Phase 3 complete")
    
    # === PHASE 4: Final polishing ===
    print(f"\n{'=' * 60}", flush=True)
    print("PHASE 4: Final Polishing", flush=True)
    print(f"{'=' * 60}", f"Remaining: {deadline - time.monotonic():.0f}s")
    
    if tracker.best is not None and time.monotonic() < deadline - 5:
        # Full LP (not reduced) for final precision
        a_polish, _ = lp_mirror_descent_aggressive(
            tracker.best, deadline - 5, verbose=True,
            use_reduced_lp=False, top_frac=1.0, max_no_improve=500
        )
        tracker.update(a_polish, "polish_full")
        
        # One more reduced LP pass
        if time.monotonic() < deadline - 5:
            a_polish2, _ = lp_mirror_descent_aggressive(
                tracker.best, deadline - 5, verbose=False,
                use_reduced_lp=True, top_frac=0.5, max_no_improve=300
            )
            tracker.update(a_polish2, "polish_red")
    
    # Fallback
    if tracker.best is None:
        tracker.best = np.ones(128)
        tracker.best_score = compute_J(tracker.best)
    
    print(f"\n{'=' * 60}", flush=True)
    print(f"FINAL: J={tracker.best_score:.8f} n={tracker.best.size} "
          f"time={time.monotonic() - tracker.t_start:.0f}s", flush=True)
    print(f"{'=' * 60}", flush=True)
    
    return [float(x) for x in tracker.best]
