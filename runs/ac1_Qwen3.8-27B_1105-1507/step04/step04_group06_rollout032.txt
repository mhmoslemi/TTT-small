import numpy as np
from scipy.optimize import linprog, minimize

def fast_conv(a):
    """Full self-convolution via FFT."""
    n = len(a)
    s = 1
    while s < 2 * n - 1:
        s <<= 1
    fa = np.fft.rfft(a, s)
    return np.fft.irfft(fa * fa, s)[:2 * n - 1]


def compute_J(a):
    """Compute J(a) = 2*n*max(conv(a,a))/(sum(a))^2."""
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))


def conv_matrix_rows(a):
    """Build the (rows, cols) structure for convolution matrix A where (A*g)[k] = (a*g)[k]."""
    n = len(a)
    rows = []
    cols = []
    for k in range(2 * n - 1):
        jl = max(0, k - n + 1)
        jh = min(n - 1, k)
        for j in range(jl, jh + 1):
            rows.append(k)
            cols.append(j)
    return np.array(rows, dtype=np.int32), np.array(cols, dtype=np.int32)


_lp_structure_cache = {}


def build_conv_matrix(a):
    """Build sparse convolution matrix."""
    from scipy.sparse import csr_matrix
    n = len(a)
    if n not in _lp_structure_cache:
        _lp_structure_cache[n] = conv_matrix_rows(a)
    rows, cols = _lp_structure_cache[n]
    vals = a[rows - cols]
    return csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))


def lp_best_response(a, M, warm_start=None):
    """LP: max sum(g) s.t. (a*g)[k] <= M for all k, g >= 0."""
    n = len(a)
    c = -np.ones(n)
    A = build_conv_matrix(a)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    options = {'presolve': True}
    if warm_start is not None:
        options['fordson'] = True
    r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs', options=options)
    if r.success and r.x is not None:
        g = np.maximum(r.x, 0.0)
        g_sum = g.sum()
        if g_sum > 1e-10:
            return g / g_sum
    return None

    # For Toeplitz: T[i,j] = first_row[j-i] if j>=i, first_col[i-j] if i>j
    # Here T[i,j] = p[i+j], so first_col[i] = p[i], first_row[j] = p[j]
    # Check: T[i,j] for i>j should be first_col[i-j] = p[i-j]. But T[i,j] = p[i+j]. Mismatch!
    # So this is NOT a standard Toeplitz matrix. It's a "Hankel-adjacent" structure.
    
    # T[i,j] = p[i+j]. This is actually a Hankel matrix (constant along anti-diagonals)
    # Hankel matvec can also be done in O(n log n) via FFT
    # (H*a)[i] = sum_j H[i,j]*a[j] = sum_j p[i+j]*a[j]
    # = sum_j p[i+j]*a[j]
    # Let b = a reversed: b[j] = a[n-1-j]
    # (H*a)[i] = sum_j p[i+j]*a[j] = sum_j p[i+j]*b[n-1-j]
    # = sum_k p[i+n-1-k]*b[k] (substituting k=n-1-j)
    # = (p_segment * b)[i+n-1] where p_segment = p reversed... 
    # Let pr[m] = p[2n-2-m]
    # (pr * b)[m] = sum_k pr[k]*b[m-k] = sum_k p[2n-2-k]*b[m-k]
    # Set m = i+n-1: sum_k p[2n-2-k]*b[i+n-1-k]
    # Let j = 2n-2-k, so k = 2n-2-j: sum_j p[j]*b[i+n-1-(2n-2-j)] = sum_j p[j]*b[i+n-1-2n+2+j]
    # = sum_j p[j]*b[i-j+1] ... not clean
    
    # I'll just use the O(n^2) direct method. For n=1024, it's ~1M multiply-adds which numpy handles in ~1ms
    grad_g = np.zeros(n)
    for i in range(n):
        idx = slice(i, min(i + n, 2 * n - 1))
        j_idx = slice(0, min(n, 2 * n - 1 - i))
        grad_g[i] = 2.0 * np.dot(p[idx], a[j_idx])
    
    grad_F = 2.0 * n * (s * grad_g - 2.0 * g_val) / (s ** 3)
    grad_F = np.nan_to_num(grad_F, nan=0.0, posinf=0.0, neginf=0.0)
    return F_val, grad_F


def fista_smooth(a0, beta, max_iters, deadline, tol=1e-10):
    """FISTA on the smoothed objective with projection onto a>=0."""
    n = len(a0)
    a = np.maximum(a0.copy(), 0.0)
    a = a / max(a.sum(), 1e-10)
    y = a.copy()
    t_k = 1.0
    prev_F = compute_J(a)
    
    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        
        F_y, grad_y = F_and_grad_fft(y, beta)
        
        # Step size: use Lipschitz estimate or fixed
        # For now use a conservative step
        step = 1.0 / max(10.0, np.abs(grad_y).max() + 1e-10)
        
        x_new = y - step * grad_y
        x_new = np.maximum(x_new, 0.0)
        s = x_new.sum()
        if s < 1e-10:
            x_new = a.copy()
        else:
            x_new = x_new / s
        
        F_new = compute_J(x_new)
        
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t_k * t_k)) / 2.0
        y = x_new + ((t_k - 1.0) / t_new) * (x_new - a)
        y = np.maximum(y, 0.0)
        y = y / max(y.sum(), 1e-10)
        
        a = x_new
        t_k = t_new
        
        if abs(prev_F - F_new) < tol:
            break
        prev_F = F_new
    
    return a


def lbfgs_optimize_improved(a0, deadline, beta_start=5.0, beta_end=500.0, n_betas=8):
    """L-BFGS-B with adaptive beta annealing."""
    a = np.maximum(a0.astype(np.float64), 1e-8)
    a = a / a.sum()
    n = len(a)
    
    betas = np.geomspace(beta_start, beta_end, n_betas)
    for beta in betas:
        if time.monotonic() > deadline:
            break
        
        def func(x):
            x2 = np.maximum(x, 0.0)
            s = x2.sum()
            if s < 1e-10:
                return 1e10
            f, _ = F_and_grad_fft(x2, beta)
            return f if np.isfinite(f) else 1e10
        
        def jac(x):
            x2 = np.maximum(x, 0.0)
            _, g = F_and_grad_fft(x2, beta)
            g = np.nan_to_num(g, nan=0.0, posinf=0.0, neginf=0.0)
            g[x2 < 1e-12] = 0.0
            return g
        
        bounds = [(0.0, None)] * n
        max_iter = 80
        try:
            res = minimize(func, a.copy(), jac=jac, method='L-BFGS-B',
                           bounds=bounds,
                           options={'maxiter': max_iter, 'max_fun': max_iter * 30,
                                    'gtol': 1e-9, 'ftol': 1e-15})
            if res.fun < 1e10 and np.all(np.isfinite(res.x)):
                new_a = np.maximum(res.x, 1e-10)
                if new_a.sum() > 1e-6:
                    a = new_a / new_a.sum()
        except Exception:
            pass
    
    return a


def resample_seq(src, dest_n):
    """Resample 1D sequence to new length via linear interpolation."""
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    if src.size == 1:
        return np.full(dest_n, src[0])
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    result = np.interp(t_new, t_old, src)
    return np.maximum(result, 0.0)


def generate_diverse_inits(n, rng, parent=None):
    """Generate diverse initial sequences with emphasis on anti-symmetric structures."""
    inits = []
    t_arr = np.linspace(0, 1, n, endpoint=False)
    t_mid = np.linspace(0, 1, n)
    
    if parent is not None and parent.size > 0:
        inits.append(resample_seq(parent, n))
        p = resample_seq(parent, n)
        p = p + rng.normal(0, 0.03, n) * p
        inits.append(np.maximum(p, 0.01))
        # Anti-symmetric perturbation of parent
        p_as = p * (1.0 + 0.1 * np.cos(np.pi * n * t_mid))
        inits.append(np.maximum(p_as, 0.01))
    
    # Uniform
    inits.append(np.ones(n))
    
    # Anti-symmetric cosines (key family!)
    for alpha in [0.3, 0.5, 0.7, 0.9]:
        a = 1.0 + alpha * np.cos(np.pi * n * t_mid)
        inits.append(np.maximum(a, 0.01))
    
    # Windowed anti-symmetric
    window = np.sin(np.pi * t_mid) ** 0.5
    for alpha in [0.5, 0.8]:
        a = (1.0 + alpha * np.cos(np.pi * n * t_mid)) * window + 0.05
        inits.append(np.maximum(a, 0.01))
    
    # Multi-frequency anti-symmetric
    a = 1.0 + 0.5 * np.cos(np.pi * n * t_mid) + 0.2 * np.cos(3 * np.pi * n * t_mid)
    inits.append(np.maximum(a, 0.01))
    
    # Gaussian bump (symmetric - for comparison)
    inits.append(np.exp(-4.0 * (t_mid - 0.5) ** 2) + 0.1)
    
    # Asymmetric linear ramp
    inits.append(1.0 + 0.5 * (2.0 * t_mid - 1.0))
    
    # Power asymmetric
    inits.append(0.3 + 0.7 * t_mid ** 2)
    
    # Random smooth
    freqs = rng.uniform(1, 10, 5)
    amps = rng.uniform(0.1, 0.4, 5)
    phases = rng.uniform(0, 2 * np.pi, 5)
    a = 1.0
    for f, amp, ph in zip(freqs, amps, phases):
        a += amp * np.cos(2 * np.pi * f * t_mid + ph)
    inits.append(np.maximum(a, 0.05))
    
    return inits


def lp_mirror_descent_improved(a, max_iters, deadline, tracker=None, prev_lp_sol=None):
    """Improved LP mirror descent with warm-starting and better line search."""
    a = np.maximum(a.astype(np.float64), 1e-8)
    a = a / a.sum()
    current_J = compute_J(a)
    if tracker is not None:
        tracker.update(a)
    
    best_a = a.copy()
    best_J = current_J
    stall_count = 0
    g_prev = None
    
    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        
        b = fast_conv(a)
        M = float(b.max())
        
        # Solve LP
        g = lp_best_response(a, M)
        if g is None:
            # Fallback: use subgradient direction
            k_max = int(np.argmax(b))
            grad = np.zeros(len(a))
            n = len(a)
            for i in range(n):
                ki = k_max - i
                if 0 <= ki < n:
                    grad[i] += 2.0 * a[ki]
            g = np.maximum(grad, 0.0)
            if g.sum() > 1e-10:
                g = g / g.sum()
            else:
                break
        
        # Multi-point line search
        best_t = 0.0
        best_J_ls = current_J
        best_a_ls = a.copy()
        
        # Try several t values adaptively
        t_candidates = [0.01, 0.02, 0.05, 0.1, 0.2, 0.3, 0.5, 0.7, 1.0]
        
        # First quick pass
        for t in t_candidates:
            a_try = (1.0 - t) * a + t * g
            a_try = np.maximum(a_try, 0.0)
            s = a_try.sum()
            if s < 1e-10:
                continue
            a_try = a_try / s
            J_try = compute_J(a_try)
            if J_try < best_J_ls:
                best_J_ls = J_try
                best_t = t
                best_a_ls = a_try.copy()
        
        # Refine around best t
        if best_t > 0:
            lo = max(0.0, best_t - 0.05)
            hi = min(1.0, best_t + 0.05)
            gr = (np.sqrt(5.0) - 1.0) / 2.0
            for _ in range(12):
                t1 = hi - gr * (hi - lo)
                t2 = lo + gr * (hi - lo)
                a1 = np.maximum((1.0 - t1) * a + t1 * g, 0.0)
                a2 = np.maximum((1.0 - t2) * a + t2 * g, 0.0)
                a1 = a1 / max(a1.sum(), 1e-10)
                a2 = a2 / max(a2.sum(), 1e-10)
                J1 = compute_J(a1)
                J2 = compute_J(a2)
                if J1 < J2:
                    hi = t2
                else:
                    lo = t1
                tm = (lo + hi) / 2.0
                am = np.maximum((1.0 - tm) * a + tm * g, 0.0)
                am = am / max(am.sum(), 1e-10)
                Jm = compute_J(am)
                if Jm < best_J_ls:
                    best_J_ls = Jm
                    best_t = tm
                    best_a_ls = am.copy()
        
        if best_J_ls < current_J - 1e-13:
            a = best_a_ls
            current_J = best_J_ls
            stall_count = 0
            if current_J < best_J:
                best_J = current_J
                best_a = a.copy()
                if tracker is not None:
                    tracker.update(a)
        else:
            stall_count += 1
            if stall_count >= 50:
                break
        
        g_prev = g
    
    if tracker is not None:
        tracker.update(best_a)
    return best_a


class ScoreTracker:
    """Tracks the best sequence found so far."""
    
    def __init__(self):
        self.best = None
        self.best_score = float('inf')
        self.t_start = time.monotonic()
    
    def update(self, a):
        a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
        if a.size == 0 or a.sum() < 0.01:
            return
        score = float(evaluate_sequence(a.tolist()))
        if np.isfinite(score) and score < self.best_score:
            self.best = a.copy()
            self.best_score = score
            elapsed = time.monotonic() - self.t_start
            print(f"  * NEW BEST: J={score:.8f} n={a.size} t={elapsed:.1f}s", flush=True)
    
    def report(self, label=""):
        if self.best is not None:
            print(f"  {label} best: J={self.best_score:.8f} n={self.best.size}", flush=True)


def subgradient_step(a, n_steps=50, deadline=None, lr_init=0.01):
    """Simple subgradient method on J(a) as a fallback/exploration tool."""
    a = np.maximum(a.astype(np.float64), 1e-6)
    a = a / a.sum()
    best_a = a.copy()
    best_J = compute_J(a)
    lr = lr_init
    
    for step in range(n_steps):
        if deadline is not None and time.monotonic() > deadline:
            break
        
        b = fast_conv(a)
        k_max = int(np.argmax(b))
        n = len(a)
        
        # Subgradient of max(conv(a,a)) at a
        grad = np.zeros(n)
        for i in range(n):
            ki = k_max - i
            if 0 <= ki < n:
                grad[i] = 2.0 * a[ki]
        
        # Also account for the (sum(a))^{-2} term
        s = a.sum()
        grad_J = grad / s - 2.0 * a / (s * s) * (2.0 * n * b[k_max] / s)
        
        # Gradient step with projection
        a_new = a - lr * grad_J
        a_new = np.maximum(a_new, 0.0)
        a_new = a_new / max(a_new.sum(), 1e-10)
        
        J_new = compute_J(a_new)
        if J_new < best_J:
            best_J = J_new
            best_a = a_new.copy()
        a = a_new
        lr *= 0.995
    
    return best_a

                max_iters = min(max_iters, 800)
                a_res = lp_mirror_descent_improved(a_res, max_iters, deadline, tracker=tracker)
            
            tracker.report(f"Phase 5: {time.monotonic()-t0:.1f}s")
    
    # Fallback
    if tracker.best is None:
        tracker.best = np.ones(1024)
        tracker.best_score = compute_J(tracker.best)
        print(f"  Fallback: J={tracker.best_score:.8f}", flush=True)
    
    print(f"\n{'='*50} FINAL RESULT {'='*50}", flush=True)
    print(f"J={tracker.best_score:.8f} n={tracker.best.size} "
          f"time={time.monotonic()-tracker.t_start:.1f}s", flush=True)
    
    result = np.clip(tracker.best, 0.0, 1000.0)
    if result.sum() < 0.01:
        result = np.ones(128)
    
    return [float(x) for x in result]
