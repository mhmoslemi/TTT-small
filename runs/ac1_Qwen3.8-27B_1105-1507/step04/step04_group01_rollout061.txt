"""
Optimized search for minimizing 2*n*max(conv(a,a))/(sum(a))^2 using
tilted parametric initialization, soft-max L-BFGS-B, LP alternating,
and targeted spectral shaping. Key insight: non-symmetric (tilted)
sequences allow the convolution center to drop below 1/n, reducing
the objective below 2.

import numpy as np
from scipy.optimize import minimize, linprog
from scipy.fft import rfft, irfft
import time


def compute_conv(a):
    """Compute linear convolution a*a using FFT."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    conv = irfft(fa * fa, size)
    return conv[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    conv = compute_conv(a)
    max_b = float(np.max(conv))
    sum_a = float(np.sum(a))
    if sum_a < 0.01:
        return np.inf
    return 2.0 * n * max_b / (sum_a * sum_a)


def fft_convolve(a, b):
    """Linear convolution of a and b using FFT."""
    n, m = len(a), len(b)
    size = 1
    while size < n + m - 1:
        size <<= 1
    fa = rfft(a, size)
    fb = rfft(b, size)
    return irfft(fa * fb, size)[:n + m - 1]


def softmax(z):
    """Numerically stable softmax."""
    z = z - np.max(z)
    e = np.exp(z)
    return e / np.sum(e)


def soft_max_obj_and_grad_z(z, tau):
    """Compute soft-max of conv(softmax(z), softmax(z)) and gradient w.r.t. z."""
    n = len(z)
    a = softmax(z)
    conv = compute_conv(a)
    cmax = float(np.max(conv))
    shifted = np.clip((conv - cmax) / tau, -500.0, 500.0)
    exp_conv = np.exp(shifted)
    Z = float(np.sum(exp_conv))
    soft_max_val = cmax + tau * np.log(Z)
    w = exp_conv / Z
    w_rev = w[::-1]
    conv_fw = fft_convolve(a, w_rev)
    f1 = conv_fw[n - 1:2 * n - 1][::-1]
    grad_a = 2.0 * f1
    mean_grad = float(np.dot(a, grad_a))
    grad_z = a * (grad_a - mean_grad)
    return soft_max_val, grad_z


def optimize_lfbs(a_init, tau_list, max_iter=300, time_budget=30.0):
    """Optimize using L-BFGS-B with soft-max parametrization and tau annealing."""
    n = len(a_init)
    a = a_init.copy()
    s = np.sum(a)
    if s < 1e-10:
        a = np.ones(n) / n
    else:
        a = a / s
    a = np.maximum(a, 1e-30)
    z = np.log(a)
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    for tau in tau_list:
        if time.monotonic() - start > time_budget:
            break

        def func_and_grad(z_val):
            return soft_max_obj_and_grad_z(z_val, tau)

        try:
            result = minimize(func_and_grad, z, jac=True, method='L-BFGS-B',
                              options={'maxiter': max_iter, 'ftol': 1e-14, 'gtol': 1e-9})
            z_new = result.x
        except Exception:
            continue

        a_new = softmax(z_new)
        a_new = np.maximum(a_new, 0)
        s_new = np.sum(a_new)
        if s_new < 1e-10:
            continue
        a_new = a_new / s_new
        obj = compute_obj(a_new)
        if obj < best_obj:
            best_obj = obj
            best_a = a_new.copy()
            z = z_new
    return best_a, best_obj


def build_toeplitz(a):
    """Build Toeplitz matrix for convolution with a."""
    n = len(a)
    A = np.zeros((2 * n - 1, n), dtype=np.float64)
    for j in range(n):
        k_start = j
        k_end = min(j + n, 2 * n - 1)
        A[k_start:k_end, j] = a[:k_end - k_start]
    return A


def solve_lp_g(a, M, time_limit=20):
    """Solve LP: max sum(g) s.t. (a*g)[k] <= M for all k, g >= 0."""
    n = len(a)
    A = build_toeplitz(a)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': time_limit})
        if res.status == 0:
            g = np.maximum(res.x, 0)
            s = float(np.sum(g))
            if s > 1e-10:
                return g / s, s
    except Exception:
        pass
    return None, None


def solve_lp_minpeak(a, time_limit=20):
    """Solve LP: min M s.t. (a*g)[k] <= M, sum(g)=1, g>=0."""
    n = len(a)
    A = build_toeplitz(a)
    c = np.zeros(n + 1)
    c[-1] = 1.0
    A_ub = np.zeros((2 * n - 1, n + 1))
    A_ub[:, :n] = A
    A_ub[:, n] = -1.0
    b_ub = np.zeros(2 * n - 1)
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])
    bounds = [(0.0, None)] * n + [(0.0, None)]
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': time_limit})
        if res.status == 0:
            g = np.maximum(res.x[:n], 0)
            s = np.sum(g)
            if s > 1e-10:
                g = g / s
                M_opt = float(res.x[n])
                return g, M_opt
    except Exception:
        pass
    return None, None


def alternating_lp(a, max_iters=50, time_budget=40.0):
    """Alternating LP with line search."""
    n = len(a)
    a = a.copy().astype(np.float64)
    s = np.sum(a)
    if s < 1e-10:
        return a
    a = a / s
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    stall_count = 0
    patience = 25

    for it in range(max_iters):
        if time.monotonic() - start > time_budget:
            break
        conv = compute_conv(a)
        M = float(np.max(conv))
        g, _ = solve_lp_g(a, M, time_limit=15)
        if g is None:
            break
        best_t_obj = best_obj
        best_a_new = best_a.copy()
        for t in [0.05, 0.1, 0.2, 0.3, 0.5, 0.7, 1.0]:
            a_try = (1.0 - t) * a + t * g
            a_try = np.maximum(a_try, 0)
            s_try = np.sum(a_try)
            if s_try < 1e-10:

            a_try = a_try / s_try
            obj_try = compute_obj(a_try)
            if obj_try < best_t_obj:
                best_t_obj = obj_try
                best_a_new = a_try.copy()
        if best_t_obj < best_obj - 1e-12:
            best_obj = best_t_obj
            best_a = best_a_new.copy()
            a = best_a_new
            stall_count = 0
        else:
            stall_count += 1
            if stall_count > patience:
                rng = np.random.default_rng(it * 7919 + 12345)
                a = a * np.exp(rng.normal(0, 0.05, n))
                a = np.maximum(a, 0)
                s = np.sum(a)
                if s > 1e-10:
                    a = a / s
                stall_count = 0
                patience = 15
            else:
                a = 0.95 * a + 0.05 * g
                a = np.maximum(a, 0)
                s = np.sum(a)
                if s > 1e-10:

def subgradient_optimize(a, max_iters=3000, time_budget=20.0, tau_start=3.0, tau_end=50.0):
    """Smoothed subgradient optimization."""
    n = len(a)
    a = a.copy().astype(np.float64)
    s = np.sum(a)
    if s < 1e-10:
        a = np.ones(n) / n
    else:
        a = a / s
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    step = 0.1
    stall = 0

    for it in range(max_iters):
        if time.monotonic() - start > time_budget:
            break
        frac = it / max(1, max_iters)
        tau = tau_start + (tau_end - tau_start) * frac
        conv = compute_conv(a)
        max_conv = float(np.max(conv))
        shifted = np.clip(tau * (conv - max_conv), -300, 300)
        weights = np.exp(shifted)
        w_sum = np.sum(weights)
        if w_sum < 1e-300:
            idx = int(np.argmax(conv))
            weights = np.zeros(len(conv))
            weights[idx] = 1.0
        else:
            weights /= w_sum
        w_rev = weights[::-1]
        conv_fw = fft_convolve(a, w_rev)
        grad_a = 2.0 * conv_fw[n - 1:2 * n - 1][::-1]
        grad_centered = grad_a - np.mean(grad_a)
        a_new = a - step * grad_centered
        a_new = np.maximum(a_new, 0)
        s_new = np.sum(a_new)
        if s_new > 1e-10:
            a_new = a_new / s_new
        else:
            a_new = a.copy()
        a = a_new
        obj = compute_obj(a)
        if obj < best_obj - 1e-13:
            best_obj = obj
            best_a = a.copy()
            stall = 0
        else:
            stall += 1
            if stall > 80:
                step *= 0.85
                if step < 1e-12:
                    break
                rng = np.random.default_rng(it * 31 + 7)
                a = a + rng.normal(0, 0.01, n) * step
                a = np.maximum(a, 0)
                s = np.sum(a)
                if s > 1e-10:
                    a = a / s
                stall = 0
        if it > 100:
            step *= 0.999
    return best_a


def tilted_init(n, alpha, gamma, smooth=0.0):
    """Generate a tilted sequence. alpha controls linear tilt, gamma controls curvature."""
    k = np.arange(n, dtype=np.float64)
    t = k / max(n - 1, 1)
    a = 1.0 + alpha * (2.0 * t - 1.0) + gamma * (t - 0.5) ** 2
    if smooth > 0:
        # Apply a small Gaussian smoothing
        from scipy.ndimage import gaussian_filter1d
        a = gaussian_filter1d(a, smooth)
    a = np.maximum(a, 0.01)
    s = np.sum(a)
    return a / s


def cos_tilted_init(n, alpha, beta):
    """Cosine-based tilted initialization."""
    k = np.arange(n, dtype=np.float64)
    t = k / max(n - 1, 1)
    a = 1.0 + alpha * np.cos(np.pi * t) + beta * np.cos(2 * np.pi * t)
    a = np.maximum(a, 0.01)
    s = np.sum(a)
    return a / s


def poly_init(n, coeffs):
    """Polynomial initialization: a_i = sum_j coeffs[j] * (i/n)^j."""
    k = np.arange(n, dtype=np.float64)
    t = k / max(n - 1, 1)
    a = np.ones(n)
    for j, c in enumerate(coeffs):
        a += c * t ** j
    a = np.maximum(a, 0.01)
    s = np.sum(a)
    return a / s


def resample_sequence(a, new_n):
    """Resample sequence to new length via interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return a_new


def try_parent():
    """Try to load parent sequence."""
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64)
        parent = np.clip(parent, 0, 1000.0)
        if parent.size > 0 and np.isfinite(parent).all():
            obj = compute_obj(parent)
            print(f"Parent (n={parent.size}): obj={obj:.6f}")
            return parent.copy(), obj
    except Exception:
        pass
    return None, np.inf


def active_set_lp(a, M, time_limit=15):
    """More targeted LP using only active constraints (near-max convolution positions)."""
    n = len(a)
    conv = compute_conv(a)
    # Use constraints where conv is close to M (top 10% or threshold)
    threshold = M * 0.85
    active = np.where(conv >= threshold)[0]
    if len(active) == 0:
        active = np.argsort(conv)[-20:]
    A_full = build_toeplitz(a)
    A = A_full[active, :]
    c = -np.ones(n)
    b_ub = np.full(len(active), M)
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': time_limit})
        if res.status == 0:
            g = np.maximum(res.x, 0)
            s = float(np.sum(g))
            if s > 1e-10:
                g_norm = g / s
                # Verify full constraint
                conv_check = fft_convolve(a, g_norm)
                if np.max(conv_check) <= M * 1.001:
                    return g_norm, s
                # Scale down if violated
                violation = np.max(conv_check) / M
                if violation > 1:
                    g_norm = g_norm / violation
                    s = float(np.sum(g_norm))
                    if s > 1e-10:
                        return g_norm / s, s
    except Exception:
        pass
    return None, None


def intensive_lp_polish(a, time_budget=60.0):
    """Intensive LP-based polishing with active set refinement."""
    n = len(a)
    a = a.copy().astype(np.float64)
    s = np.sum(a)
    if s < 1e-10:
        return a
    a = a / s
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    stall = 0

    for it in range(200):
        if time.monotonic() - start > time_budget:
            break
        conv = compute_conv(a)
        M = float(np.max(conv))
        # Try active set LP first (faster)
        g, _ = active_set_lp(a, M, time_limit=8)
        if g is None:
            g, _ = solve_lp_g(a, M, time_limit=10)
        if g is None:
            break
        # Line search with more steps
        best_t = best_obj
        best_g = best_a.copy()
        for t in [0.01, 0.03, 0.05, 0.1, 0.2, 0.3, 0.5, 0.7, 1.0]:
            a_try = (1.0 - t) * a + t * g
            a_try = np.maximum(a_try, 0)
            s_try = np.sum(a_try)
            if s_try < 1e-10:
                continue
            a_try = a_try / s_try
            obj_try = compute_obj(a_try)
            if obj_try < best_t:
                best_t = obj_try
                best_g = a_try.copy()
        if best_t < best_obj - 1e-13:
            best_obj = best_t
            best_a = best_g.copy()
            a = best_g
            stall = 0
        else:
            stall += 1
            if stall > 30:
                rng = np.random.default_rng(it * 104729 + 42)
                a = a * np.exp(rng.normal(0, 0.03, n))
                a = np.maximum(a, 0)
                s = np.sum(a)
                if s > 1e-10:
                    a = a / s
                stall = 0
    return best_a


def propose_candidate(seed=42, budget_s=1500):
    """
    Main search combining tilted parametric initialization, soft-max L-BFGS-B,
    active-set LP polishing, and subgradient methods. Focuses on n in [400, 900]
    with intensive polishing of top candidates.
    """
    start_time = time.monotonic()
    deadline = start_time + float(budget_s) - 30.0

    best_seq = None
    best_score = np.inf

    # Load parent
    parent, parent_obj = try_parent()
    if parent is not None and parent_obj < best_score:
        best_score = parent_obj
        best_seq = parent.copy()

    print(f"Initial best: {best_score:.6f}")

    # N values to explore - focused range
    n_values = [400, 450, 500, 550, 600, 650, 700, 750, 800]
    if best_seq is not None:
        np_ = len(best_seq)
        extra = [np_, max(100, np_ - 50), min(1000, np_ + 50), min(1000, np_ + 100)]
        n_values = sorted(set(n_values + extra))

    print(f"N values: {n_values}")

    candidates = []  # (a, obj, n)

    # Phase 1: Explore different n and initializations
    for n in n_values:
        if time.monotonic() > deadline - 180:
            break
        remaining = deadline - time.monotonic()
        if remaining < 15:
            break

        # Time allocation
        if n <= 500:
            alloc = min(40.0, remaining * 0.1)
        elif n <= 700:
            alloc = min(55.0, remaining * 0.12)
        else:
            alloc = min(65.0, remaining * 0.13)
        alloc = max(15.0, alloc)

        print(f"\nN={n}: {remaining:.0f}s remaining, alloc={alloc:.0f}s")

        # Multiple initializations per n
        n_inits = max(3, int(alloc / 12))
        n_inits = min(n_inits, 6)

        for init_idx in range(n_inits):
            if time.monotonic() > deadline - 180:
                break
            remaining_init = deadline - time.monotonic()
            if remaining_init < 10:
                break

            # Different initialization strategies
            if init_idx == 0 and best_seq is not None:
                # Resample parent
                a = resample_sequence(best_seq, n)
            elif init_idx == 1:
                # Linear tilt (left-heavy: larger values at start)
                a = tilted_init(n, alpha=0.3, gamma=0.0)
            elif init_idx == 2:
                # Linear tilt (right-heavy)
                a = tilted_init(n, alpha=-0.3, gamma=0.0)
            elif init_idx == 3:
                # Cosine tilt
                a = cos_tilted_init(n, alpha=0.2, beta=0.1)
            elif init_idx == 4:
                # Polynomial (quadratic decrease)
                a = poly_init(n, coeffs=[-0.5, 0.3])
            else:
                # Random tilted
                rng = np.random.default_rng(seed + n * 104729 + init_idx * 79193)
                alpha = rng.uniform(-0.5, 0.5)
                gamma = rng.uniform(-0.3, 0.3)
                a = tilted_init(n, alpha=alpha, gamma=gamma)

            # Quick subgradient shaping
            sg_time = min(6.0, (alloc / n_inits) * 0.15)
            a = subgradient_optimize(a, max_iters=int(sg_time * 50),
                                      time_budget=sg_time,
                                      tau_start=2.0, tau_end=15.0)

            # L-BFGS-B with tau annealing
            remaining_init = deadline - time.monotonic()
            if remaining_init > 10:
                lfbs_time = min((alloc / n_inits) * 0.35, 20.0)
                tau_list = [3.0/n, 1.5/n, 0.7/n, 0.3/n, 0.15/n, 0.07/n]
                a, obj = optimize_lfbs(a, tau_list, max_iter=250, time_budget=lfbs_time)
            else:
                obj = compute_obj(a)

            # LP polishing
            remaining_init = deadline - time.monotonic()
            if remaining_init > 8 and n <= 800:
                a = intensive_lp_polish(a, time_budget=min(remaining_init * 0.3, 20.0))
                obj = compute_obj(a)

            print(f"  init{init_idx}: n={n}, obj={obj:.6f}")

            if obj < best_score:
                best_score = obj
                best_seq = a.copy()
                print(f"  ** UPDATE: n={n}, obj={best_score:.6f} **")

            candidates.append((a.copy(), obj, n))

    # Phase 2: Intensive polishing of top candidates
    print(f"\n=== POLISHING PHASE ===")
    print(f"Best before polish: {best_score:.6f}")

    candidates.sort(key=lambda x: x[1])
    n_polish = min(4, len(candidates))

    for ci in range(n_polish):
        if time.monotonic() > deadline - 80:
            break
        remaining = deadline - time.monotonic()
        if remaining < 25:
            break

        a_cand = candidates[ci][0]
        n_cand = candidates[ci][2]
        print(f"  Polishing candidate {ci} (n={n_cand}, obj={candidates[ci][1]:.6f}), {remaining:.0f}s left")

        polish_budget = min(remaining * 0.35, 70.0)

        # Intensive LP
        a_pol = intensive_lp_polish(a_cand, time_budget=polish_budget * 0.4)
        obj_pol = compute_obj(a_pol)
        if obj_pol < best_score:
            best_score = obj_pol
            best_seq = a_pol.copy()
            print(f"  ** LP polish: obj={best_score:.6f} **")
        a_cur = a_pol

        # L-BFGS-B with very small tau
        remaining = deadline - time.monotonic()
        if remaining > 20:
            tau_vsmall = 0.05 / n_cand
            a_new, obj_new = optimize_lfbs(a_cur, [0.3/n_cand, 0.1/n_cand, tau_vsmall],
                                           max_iter=400, time_budget=min(remaining * 0.25, 20.0))
            if obj_new < best_score:
                best_score = obj_new
                best_seq = a_new.copy()
                a_cur = a_new.copy()
                print(f"  ** L-BFGS polish: obj={best_score:.6f} **")

        # More LP
        remaining = deadline - time.monotonic()
        if remaining > 15:
            a_new = intensive_lp_polish(a_cur, time_budget=min(remaining * 0.3, 25.0))
            obj_new = compute_obj(a_new)
            if obj_new < best_score:
                best_score = obj_new
                best_seq = a_new.copy()
                a_cur = a_new.copy()
                print(f"  ** LP polish 2: obj={best_score:.6f} **")

        # Subgradient final touch
        remaining = deadline - time.monotonic()
        if remaining > 10:
            a_new = subgradient_optimize(a_cur, max_iters=int(remaining * 10),
                                          time_budget=min(remaining * 0.2, 12.0),
                                          tau_start=15.0, tau_end=60.0)
            obj_new = compute_obj(a_new)
            if obj_new < best_score:
                best_score = obj_new
                best_seq = a_new.copy()
                print(f"  ** SG polish: obj={best_score:.6f} **")

    # Phase 3: Final intensive polish on global best
    if best_seq is not None and time.monotonic() < deadline - 20:
        remaining = deadline - time.monotonic()
        n_best = len(best_seq)
        print(f"\n=== FINAL INTENSIVE POLISH: n={n_best}, {remaining:.0f}s ===")

        a_cur = best_seq.copy()
        for rnd in range(6):
            if time.monotonic() > deadline - 10:
                break
            remaining = deadline - time.monotonic()
            if remaining < 10:
                break

            # LP
            a_new = intensive_lp_polish(a_cur, time_budget=min(remaining * 0.3, 25.0))
            obj_new = compute_obj(a_new)
            if obj_new < best_score - 1e-14:
                best_score = obj_new
                best_seq = a_new.copy()
                a_cur = a_new.copy()
                print(f"  Final LP {rnd}: {best_score:.6f}")

            # Tiny L-BFGS
            remaining = deadline - time.monotonic()
            if remaining > 8:
                tau_tiny = 0.02 / n_best
                a_new2, obj_new2 = optimize_lfbs(a_cur, [0.1/n_best, tau_tiny],
                                                  max_iter=300, time_budget=min(remaining * 0.2, 12.0))
                if obj_new2 < best_score - 1e-14:
                    best_score = obj_new2
                    best_seq = a_new2.copy()
                    a_cur = a_new2.copy()
                    print(f"  Final L-BFGS {rnd}: {best_score:.6f}")

            # Perturbation + re-optimize
            remaining = deadline - time.monotonic()
            if remaining > 10 and rnd >= 2:
                rng = np.random.default_rng(rnd * 99999 + 7)
                a_pert = a_cur * np.exp(rng.normal(0, 0.02, n_best))
                a_pert = np.maximum(a_pert, 0)
                s_pert = np.sum(a_pert)
                if s_pert > 1e-10:
                    a_pert = a_pert / s_pert
                    a_pert = intensive_lp_polish(a_pert, time_budget=min(remaining * 0.25, 15.0))
                    obj_pert = compute_obj(a_pert)
                    if obj_pert < best_score - 1e-14:
                        best_score = obj_pert
                        best_seq = a_pert.copy()
                        a_cur = a_pert.copy()
                        print(f"  Final perturb {rnd}: {best_score:.6f}")

    # Phase 4: Try slightly different n values for final squeeze
    if time.monotonic() < deadline - 30:
        remaining = deadline - time.monotonic()
        n_best = len(best_seq) if best_seq is not None else 600
        for n_try in [n_best - 10, n_best + 10, n_best - 20, n_best + 20]:
            if time.monotonic() > deadline - 10:
                break
            if n_try < 100 or n_try > 1000:
                continue
            remaining = deadline - time.monotonic()
            if remaining < 15:
                break
            a_try = resample_sequence(best_seq, n_try)
            a_try = intensive_lp_polish(a_try, time_budget=min(remaining * 0.4, 20.0))
            obj_try = compute_obj(a_try)
            if obj_try < best_score:
                best_score = obj_try
                best_seq = a_try.copy()
                print(f"  N-squeeze n={n_try}: {best_score:.6f}")

    # Final validation
    if best_seq is None:
        best_seq = np.ones(200)
        best_score = compute_obj(best_seq)

    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    val = evaluate_sequence(result)
    if val < np.inf and val < best_score:
        best_score = val
    elif val >= np.inf:
        result = [1.0] * 100
        best_score = evaluate_sequence(result)

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_score:.6f}, time={elapsed:.1f}s ===")
    return result
