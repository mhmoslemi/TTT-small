import numpy as np
from scipy.optimize import linprog, minimize, minimize_scalar
from scipy.sparse import csr_matrix

def fast_conv(a):
    """Full self-convolution via FFT."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]


def compute_J(a):
    """Compute J(a) = 2*n*max(conv(a,a))/(sum(a))^2."""
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))


def build_A_sparse(n, a):
    """Build sparse Toeplitz matrix A where A[k,j] = a[k-j] for valid j."""
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        j_min = max(0, k - n + 1)
        j_max = min(n - 1, k)
        for j in range(j_min, j_max + 1):
            rows.append(k)
            cols.append(j)
            vals.append(a[k - j])
    return csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))


def lp_best_response(a, M):
    """Solve LP: max sum(g) s.t. A(a)@g <= M*1, g >= 0."""
    n = len(a)
    A = build_A_sparse(n, a)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    try:
        r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
        if r.success and r.x is not None:
            g = np.maximum(r.x, 0)
            if g.sum() > 1e-15:
                return g
    except Exception:
        pass
    return None


def qp_best_response(a, M, reg=0.01):

        return -np.sum(g) + reg * np.dot(g, g)

    return None


def spectral_shape(a, gamma=0.5, n_iters=50, deadline=None):
    """
    Iterative spectral shaping: flatten the power spectrum |A|^2
    to reduce the peak of the autocorrelation.
    """
    a = np.maximum(a.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-10:
        return a
    a = a / s
    best_a = a.copy()
    best_J = compute_J(a)
    n = len(a)
    
    for it in range(n_iters):
        if deadline is not None and time.monotonic() > deadline:
            break
        
        A = np.fft.rfft(a, 2 * n)
        P = np.abs(A) ** 2
        
        # Compress the power spectrum towards flat
        P_mean = P.mean()
        if P_mean < 1e-30:
            break
        
        # Apply amplitude compression: P_new = P_mean * (1 + gamma * (P/P_mean - 1))^alpha
        # where alpha < 1 compresses dynamic range
        ratio = P / P_mean
        # Soft clipping: compress high values, boost low values
        P_new = P_mean * np.power(np.maximum(ratio, 1e-10), 1.0 - gamma)
        
        # Recover with new amplitude, preserve phase
        phase = A / np.maximum(np.abs(A), 1e-30)
        A_new = phase * np.sqrt(P_new)
        
        a_new = np.fft.irfft(A_new, 2 * n)
        a_new = a_new[:n]
        a_new = np.maximum(a_new, 0.0)
        s_new = a_new.sum()
        if s_new < 1e-10:
            break
        a_new = a_new / s_new
        
        J_new = compute_J(a_new)
        if J_new < best_J:
            best_J = J_new
            best_a = a_new.copy()
        a = a_new
        
        # Anneal: reduce gamma over time for convergence
        gamma = max(0.1, gamma * 0.97)
    
    return best_a


def spectral_shape_v2(a, deadline=None):
    """
    Alternative spectral shaping: percentile-based power spectrum clipping.
    More aggressive for peak reduction.
    """
    a = np.maximum(a.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-10:
        return a
    a = a / s
    best_a = a.copy()
    best_J = compute_J(a)
    n = len(a)
    
    for it in range(200):
        if deadline is not None and time.monotonic() > deadline:
            break
        
        A = np.fft.rfft(a, 2 * n)
        P = np.abs(A) ** 2
        phase = A / np.maximum(np.abs(A), 1e-30)
        
        # Percentile clipping: cap the top p% of power
        # and boost the bottom to maintain total power
        total_power = P.sum()
        p_high = 0.95 - 0.3 * it / 200.0  # start at 95%, go to 65%
        p_low = 0.05 + 0.3 * it / 200.0
        
        threshold_high = np.percentile(P, p_high * 100)
        P_clipped = np.minimum(P, threshold_high)
        
        # Redistribute: scale up the low-power bins
        low_mask = P < np.percentile(P, p_low * 100)
        if low_mask.sum() > 0:
            excess = P.sum() - P_clipped.sum()
            P_clipped[low_mask] += excess / low_mask.sum()
        
        # Ensure non-negative
        P_clipped = np.maximum(P_clipped, 1e-20)
        # Renormalize to preserve total power
        P_clipped = P_clipped * (total_power / P_clipped.sum())
        
        A_new = phase * np.sqrt(P_clipped)
        a_new = np.fft.irfft(A_new, 2 * n)[:n]
        a_new = np.maximum(a_new, 0.0)
        s_new = a_new.sum()
        if s_new < 1e-10:
            break
        a_new = a_new / s_new
        
        J_new = compute_J(a_new)
        if J_new < best_J:
            best_J = J_new
            best_a = a_new.copy()
        a = a_new
    
    return best_a


def lp_mirror_descent(a, max_iters, deadline, use_multiplicative=False, lp_budget=0.05):
    """LP mirror descent with adaptive line search."""
    a = np.maximum(a.astype(np.float64), 1e-10)
    a = a / a.sum()
    current_J = compute_J(a)
    best_J = current_J
    best_a = a.copy()
    no_improve = 0
    n = len(a)
    
    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        if time.monotonic() + 2 * lp_budget > deadline:
            break
        
        b = fast_conv(a)
        M = float(b.max())
        
        t_lp_start = time.monotonic()
        g = lp_best_response(a, M)
        t_lp_end = time.monotonic()
        
        if g is None:
            break
        g = np.maximum(g, 0.0)
        g_sum = g.sum()
        if g_sum < 1e-10:
            break
        g = g / g_sum
        
        # Adaptive step size
        base_t = 0.3 * np.exp(-0.2 * it / 50.0) + 0.05
        t = min(1.0, base_t)
        
        best_trial_J = current_J
        best_trial_a = None
        
        for ls in range(20):
            if use_multiplicative:
                ratio = g / np.maximum(a, 1e-15)
                a_trial = a * np.power(ratio, t)
            else:
                a_trial = (1.0 - t) * a + t * g
            a_trial = np.maximum(a_trial, 0.0)
            s = a_trial.sum()
            if s < 1e-10:
                t *= 0.5
                continue
            a_trial = a_trial / s
            J_trial = compute_J(a_trial)
            if J_trial < best_trial_J - 1e-14:
                best_trial_J = J_trial
                best_trial_a = a_trial.copy()
                break
            t *= 0.6
            if t < 0.001:
                break
        
        if best_trial_a is not None:
            a = best_trial_a
            current_J = best_trial_J
            if current_J < best_J:
                best_J = current_J
                best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1
            if no_improve > 30:
                break
    
    return best_a, best_J


def subgradient_polish(a, max_iters=2000, deadline=None):
    """Subgradient descent with adaptive step size and momentum."""
    a = np.maximum(a.astype(np.float64), 1e-10)
    a = a / a.sum()
    best_M = fast_conv(a).max()
    best_a = a.copy()
    best_J = compute_J(a)
    momentum = np.zeros_like(a)
    beta = 0.9
    n = len(a)
    
    for it in range(max_iters):
        if deadline is not None and time.monotonic() > deadline:
            break
        
        b = fast_conv(a)
        M = float(b.max())
        tol = max(1e-12 * M, 1e-15)
        active = np.where(b >= M - tol)[0]
        
        grad = np.zeros_like(a)
        for k in active:
            lo = max(0, k - n + 1)
            hi = min(n - 1, k)
            for i in range(lo, hi + 1):
                grad[i] += 2.0 * a[k - i]
            # Fix diagonal terms (k = 2i case)
            if 0 <= k < 2 * n - 1:
                mid = k // 2
                if 2 * mid == k and 0 <= mid < n:
                    grad[mid] -= a[mid]  # subtract one copy since we double-counted
        
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-15:
            break
        
        # Polyak-like step size
        t = 1.0 / (gnorm * (1 + 0.5 * np.sqrt(it + 1)))
        momentum = beta * momentum + (1 - beta) * grad / gnorm
        a_new = a - t * momentum
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s < 1e-10:
            break
        a_new = a_new / s
        
        new_M = fast_conv(a_new).max()
        J_new = compute_J(a_new)
        if J_new < best_J:
            best_J = J_new
            best_M = new_M
            best_a = a_new.copy()
        a = a_new
    
    return best_a, best_J


def resample_seq(src, dest_n):
    """Resample 1D sequence to new length using linear interpolation."""
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)


def fourier_to_seq(params, n, n_harm):
    """Convert Fourier parameters to a non-negative sequence (normalized)."""
    t = np.arange(n, dtype=np.float64) / n
    a = np.full(n, params[0], dtype=np.float64)
    for k in range(1, n_harm + 1):
        a += params[2 * k - 1] * np.cos(2.0 * np.pi * k * t)
        a += params[2 * k] * np.sin(2.0 * np.pi * k * t)
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-8:
        a = np.ones(n) / n
    else:
        a = a / s
    return a


def seq_to_fourier(a, n_harm):
    """Extract Fourier parameters from a sequence."""
    n = len(a)
    a_norm = a / a.sum()
    t = np.arange(n, dtype=np.float64) / n
    params = np.zeros(2 * n_harm + 1)
    params[0] = np.mean(a_norm)
    for k in range(1, n_harm + 1):
        params[2 * k - 1] = 2.0 * np.mean(a_norm * np.cos(2.0 * np.pi * k * t))
        params[2 * k] = 2.0 * np.mean(a_norm * np.sin(2.0 * np.pi * k * t))
    return params


def cmaes_simple(params0, sigma0, n, n_harm, max_gens, deadline, tracker):
    """Simplified CMA-ES on Fourier parameters."""
    dim = 2 * n_harm + 1
    lam = max(8, 4 + int(3 * np.log(dim)))
    mu = lam // 2
    
    weights = np.log(mu + 0.5) - np.log(np.arange(1, mu + 1))
    weights = weights / weights.sum()
    mu_eff = 1.0 / np.sum(weights ** 2)
    
    c_sigma = (mu_eff + 2) / (dim + mu_eff + 5)
    c_c = (4 + mu_eff / dim) / (dim + 4 + 2 * mu_eff / dim)
    d_sigma = 1 + 2 * max(0, np.sqrt((mu_eff - 1) / (dim + 1) - 1) - 1) + c_sigma
    c_1 = 2 / ((dim + 1.3) ** 2 + mu_eff)
    c_mu = min(1 - c_1, 2 * (mu_eff - 2 + 1.0 / mu_eff) / ((dim + 2) ** 2 + mu_eff))
    chi_N = np.sqrt(dim) * (1 - 1 / (4 * dim) + 1 / (21 * dim ** 2))
    
    C = np.eye(dim)
    Cs = np.zeros((dim, dim))
    p_sigma = np.zeros(dim)
    p_c = np.zeros(dim)
    x = params0.copy()
    sigma = sigma0
    
    best_x = params0.copy()
    best_J = compute_J(fourier_to_seq(params0, n, n_harm))
    tracker_update_best = best_J
    
    for gen in range(max_gens):
        if time.monotonic() > deadline:
            break
        
        offspring = []
        offspring_J = []
        for i in range(lam):
            z = np.random.standard_normal(dim)
            y = (np.linalg.cholesky(C + 1e-10*np.eye(dim)) @ z)
            x_trial = x + sigma * y
            a_trial = fourier_to_seq(x_trial, n, n_harm)
            J_trial = compute_J(a_trial)
            offspring.append(x_trial)
            offspring_J.append(J_trial)
        
        idx = np.argsort(offspring_J)
        
        x_new = np.zeros(dim)
        for i in range(mu):
            x_new += weights[i] * offspring[idx[i]]
        
        if sigma > 1e-30:
            diff = (x_new - x) / (sigma * chi_N)
        else:
            diff = np.zeros(dim)
        
        p_sigma = (1 - c_sigma) * p_sigma + np.sqrt(c_sigma * (2 - c_sigma) * mu_eff) * diff
        sigma *= np.exp(c_sigma / d_sigma * (np.linalg.norm(p_sigma) / chi_N - 1))
        sigma = max(sigma, 1e-12)
        sigma = min(sigma, 0.5)
        
        h_sigma = 1.0
        if np.linalg.norm(p_sigma) / chi_N > (1.4 + c_c) * np.sqrt((1 + c_sigma) / (1 - c_sigma)):
            h_sigma = 0.0
        p_c = (1 - c_c) * p_c + h_sigma * np.sqrt(c_c * (2 - c_c) * mu_eff) * (x_new - x) / max(sigma, 1e-30)
        
        rank_1 = np.outer(p_c, p_c)
        rank_mu = np.zeros((dim, dim))
        for i in range(mu):
            d = offspring[idx[i]] - x
            rank_mu += weights[i] * np.outer(d, d)
        
        Cs = (1 - c_1 - c_mu) * Cs + c_1 * (rank_1 + (1 - h_sigma) * h_sigma * np.outer(p_c, p_c)) + c_mu * rank_mu
        C = (1 - c_1 - c_mu) * C + c_1 * (rank_1 + (1 - h_sigma) * h_sigma * np.outer(p_c, p_c)) + c_mu * rank_mu
        C = (C + C.T) / 2
        x = x_new
        
        if offspring_J[idx[0]] < best_J:
            best_J = offspring_J[idx[0]]
            best_x = offspring[idx[0]].copy()
            a_best = fourier_to_seq(best_x, n, n_harm)
            tracker_update_best = best_J
            score = float(evaluate_sequence(a_best.tolist()))
            if np.isfinite(score) and score < tracker_best_score[0]:
                tracker_best_score[0] = score
                tracker_best_seq[0] = a_best.copy()
                print(f"    [CMA-ES] NEW BEST: J={score:.8f} gen={gen}", flush=True)
    
    a_final = fourier_to_seq(best_x, n, n_harm)
    return a_final, best_J

def lbb_objective(x):
    """Objective for L-BFGS-B: max of self-convolution (scale invariant)."""
    ax = np.maximum(x, 0.0)
    s = ax.sum()
    if s < 1e-8:
        return 1e10
    ax = ax / s
    b = fast_conv(ax)
    return float(b.max())


def generate_diverse_inits(n, rng, parent=None):
    """Generate diverse initial sequences of length n."""
    inits = []
    t = np.arange(n, dtype=np.float64) / n
    
    if parent is not None:
        p = resample_seq(parent, n)
        inits.append(p)
        for noise_scale in [0.01, 0.03, 0.05, 0.1]:
            inits.append(np.maximum(p + rng.normal(0, noise_scale, n), 0.001))
    
    # Various structured inits
    inits.append(np.ones(n))
    inits.append(0.5 * (1 + np.cos(2 * np.pi * t)))
    inits.append(0.5 * (1 - np.cos(2 * np.pi * t)))
    inits.append(np.maximum(0.3 + 0.5 * np.cos(2 * np.pi * t) + 0.1 * np.cos(4 * np.pi * t), 0.01))
    inits.append(np.exp(-4 * (t - 0.5) ** 2) + 0.05)
    inits.append(np.exp(-8 * (t - 0.3) ** 2) + np.exp(-8 * (t - 0.7) ** 2) + 0.02)
    inits.append(1.0 / (1 + 10 * np.exp(-30 * (t - 0.5))) + 0.05)  # sigmoid
    inits.append(1.0 / (1 + 10 * np.exp(30 * (t - 0.5))) + 0.05)
    
    for f in [2, 3, 4, 5, 6, 8, 12]:
        inits.append(np.maximum(1.0 + 0.3 * np.cos(2 * np.pi * f * t + rng.uniform(0, 2*np.pi)), 0.01))
    
    # Beta-distribution shaped
    for alpha, beta_val in [(2, 2), (3, 3), (1.5, 1.5), (4, 4), (0.5, 0.5)]:
        from scipy.stats import betarv
        inits.append(betarv(alpha, beta_val, size=n, random_state=rng))
    
    # Smooth random
    k = max(3, n // 8)
    kernel = np.ones(k) / k
    smooth = np.convolve(rng.uniform(0, 1, n), kernel, mode='same')
    inits.append(np.maximum(smooth + 0.1, 0.01))
    
    # Boundary-concentrated (U-shape)
    inits.append(1.0 / (0.1 + t * (1 - t) * 20))
    
    result = []
    for a in inits:
        a = np.maximum(np.asarray(a, dtype=np.float64), 0.001)
        s = a.sum()
        if s > 1e-6:
            result.append(a / s)
    return result

            a_cma_lp = a_cma_lp / a_cma_lp.sum()
            a_cma_lp, J_cma_lp = lp_mirror_descent(a_cma_lp, 300, min(deadline - 80, time.monotonic() + 100))
            tracker_update(a_cma_lp)
            print(f"  LP from CMA-ES: J={J_cma_lp:.8f}", flush=True)
    
    # === Phase 5: Subgradient polishing ===
    if time.monotonic() < deadline - 60:
        print(f"\n--- Phase 5: Subgradient polish ---", flush=True)
        t0 = time.monotonic()
        
        a_sub = tracker_best_seq[0].copy() if tracker_best_seq[0] is not None else current_best
        a_sub, J_sub = subgradient_polish(a_sub, 5000, min(deadline - 40, t0 + 120))
        tracker_update(a_sub)
        print(f"  Subgradient: J={J_sub:.8f} ({time.monotonic()-t0:.1f}s)", flush=True)
    
    # === Phase 6: Final LP refinement (long) ===
    if time.monotonic() < deadline - 30:
        print(f"\n--- Phase 6: Final long LP ---", flush=True)
        t0 = time.monotonic()
        
        a_final_lp = tracker_best_seq[0].copy() if tracker_best_seq[0] is not None else current_best
        a_final_lp = np.maximum(a_final_lp, 1e-10)
        a_final_lp = a_final_lp / a_final_lp.sum()
        
        # Linear mode
        a_final_lp, J_final1 = lp_mirror_descent(a_final_lp, 10000, min(deadline - 20, t0 + (deadline - 30 - t0) * 0.7))
        tracker_update(a_final_lp)
        print(f"  LP linear: J={J_final1:.8f}", flush=True)
        
        # Multiplicative mode on the result
        if time.monotonic() < deadline - 15:
            a_final_lp2 = tracker_best_seq[0].copy()
            a_final_lp2 = np.maximum(a_final_lp2, 1e-10)
            a_final_lp2 = a_final_lp2 / a_final_lp2.sum()
            a_final_lp2, J_final2 = lp_mirror_descent(a_final_lp2, 5000, deadline - 5, use_multiplicative=True)
            tracker_update(a_final_lp2)
            print(f"  LP mult: J={J_final2:.8f}", flush=True)
    
    # === Phase 7: L-BFGS-B final refinement ===
    if time.monotonic() < deadline - 5 and tracker_best_seq[0] is not None:
        print(f"\n--- Phase 7: L-BFGS-B ---", flush=True)
        t0 = time.monotonic()
        n = tracker_best_seq[0].size
        try:
            x0 = tracker_best_seq[0].astype(np.float64)
            res = minimize(lbb_objective, x0, method='L-BFGS-B',
                           bounds=[(0.0, None)] * n,
                           options={'maxiter': 50, 'max_fun': min(20000, int((deadline - time.monotonic()) * 500)),
                                    'ftol': 1e-16})
            if res.fun < 1e10:
                tracker_update(res.x)
                print(f"  L-BFGS-B: J={compute_J(res.x):.8f}", flush=True)
        except Exception as e:
            print(f"  L-BFGS-B error: {e}", flush=True)
        print(f"  ({time.monotonic()-t0:.1f}s)", flush=True)
    
    # Final
    if tracker_best_seq[0] is not None:
        final_a = tracker_best_seq[0]
        print(f"\n{'='*60}", flush=True)
        print(f"FINAL: J={tracker_best_score[0]:.8f} n={final_a.size} time={time.monotonic()-t_start_global:.0f}s", flush=True)
        print(f"{'='*60}", flush=True)
        return [float(x) for x in final_a]
    else:
        return [float(x) for x in current_best]
