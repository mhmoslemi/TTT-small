def fast_conv(a):
    n = len(a)
    s = 1
    while s < 2 * n - 1:
        s <<= 1
    fa = np.fft.rfft(a, s)
    return np.fft.irfft(fa * fa, s)[:2 * n - 1]

    s = a.sum()

    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))

def get_structure(n):
    if n not in _structure_cache:
        rows = []
        cols = []
        for k in range(2 * n - 1):
            jl = max(0, k - n + 1)
            jh = min(n - 1, k)
            for j in range(jl, jh + 1):
                rows.append(k)
                cols.append(j)
        _structure_cache[n] = (np.array(rows, dtype=np.int32),
                                np.array(cols, dtype=np.int32))
    return _structure_cache[n]

def build_A(a):
    n = len(a)
    rows, cols = get_structure(n)
    vals = a[rows - cols]
    return csr_matrix((vals, (rows, cols)), shape=(2*n-1, n))

    bounds = [(0.0, None)] * n

                options={'presolve': True, 'dual_feasibility_tolerance': 1e-10})

        g = np.maximum(r.x, 0.0)
        gs = g.sum()
