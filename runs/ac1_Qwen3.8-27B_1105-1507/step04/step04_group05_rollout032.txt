Minimizes J(a) = 2*n*max(convolve(a,a))/(sum(a))^2 using:
1. LP best-response with multiplicative weight (mirror descent) updates
2. Coordinate descent with analytical 1D subproblem solution
3. Log-sum-exp smoothed gradient descent (L-BFGS refinement)

5. Perturbation-based restarts for escaping local minima
Key innovation: combines mirror descent geometry with LSE gradient refinement,
exploiting the convexity of the underlying problem.

import numpy as np
from scipy.optimize import linprog
from scipy.sparse import lil_matrix
import time


def fast_conv(a):
    """Full self-convolution via FFT."""
    n = len(a)
    if n == 0:
        return np.array([])
    s = 1
    while s < 2 * n - 1:
        s <<= 1
    fa = np.fft.rfft(a, s)
    return np.fft.irfft(fa * fa, s)[:2 * n - 1]


def compute_J(a):
    """Compute J(a) = 2*n*max(conv(a,a))/(sum(a))^2."""
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))


def build_A_sparse(a):
    """Build sparse matrix A where (A @ g)[k] = (a*g)[k] (convolution)."""
    n = len(a)
    m = 2 * n - 1
    A = lil_matrix((m, n))
    for k in range(m):
        jl = max(0, k - n + 1)
        jh = min(n - 1, k)
        for j in range(jl, jh + 1):
            A[k, j] = a[k - j]
    return A.tocsr()


def lp_best_response(a, M):
    """LP: max sum(g) s.t. (a*g)_k <= M for all k, g >= 0."""
    n = len(a)
    c = -np.ones(n)
    A = build_A_sparse(a)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
    if r.success and r.x is not None:
        g = np.maximum(r.x, 0.0)
        if g.sum() > 1e-10:
            return g
    return None


def lse_obj_grad(a, tau):
    """
    Compute the log-sum-exp smoothed objective and its gradient.
    J_smooth = 2n * LSE(tau*b) / S^2
    Returns (J, grad) where grad is dJ/da.
    """
    n = len(a)
    b = fast_conv(a)
    S = a.sum()
    if S < 1e-10:
        return 1e10, np.zeros(n)

    # Numerically stable log-sum-exp
    tb = tau * b
    tb_max = tb.max()
    exp_tb = np.exp(tb - tb_max)
    sum_exp = exp_tb.sum()
    lse_val = (tb_max + np.log(sum_exp)) / tau

    # Softmax weights p_k = exp(tau*b_k) / sum(exp(tau*b))
    p = exp_tb / sum_exp

    # Gradient of LSE w.r.t. a_i:
    # dLSE/da_i = sum_k p_k * db_k/da_i = sum_k p_k * 2*a[k-i] * 1{0<=k-i<n}
    grad_lse = np.zeros(n)
    for i in range(n):
        j_max = min(n - 1, 2 * n - 2 - i)
        if j_max >= 0:
            grad_lse[i] = 2.0 * np.dot(a[:j_max + 1], p[i:i + j_max + 1])

    # J = 2n * lse / S^2
    J = 2.0 * n * lse_val / (S * S)

    # dJ/da_i = 2n/S^2 * (grad_lse[i] - 2*lse/S)
    grad_J = (2.0 * n / (S * S)) * (grad_lse - 2.0 * lse_val / S)

    return J, grad_J


def project_simplex(v):
    """Project vector v onto the probability simplex {x>=0, sum(x)=1}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.where(u * np.arange(1, n + 1) > (cssv - 1))[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0)

        # Update: try multiplicative weight and averaging
        progress = min(1.0, it / max(1, max_iters))
        base_t = 0.3 * (1.0 - 0.5 * progress) + 0.05

        candidates = []
        # Averaging
        for t in [base_t, base_t * 0.5, base_t * 1.5]:
            a_trial = (1.0 - t) * a + t * g
            a_trial = np.maximum(a_trial, 0.0)
            if a_trial.sum() > 1e-10:
                a_trial = a_trial / a_trial.sum()
                candidates.append(a_trial)
        # Multiplicative weight
        if use_mw:
            for t in [base_t * mw_power, base_t * mw_power * 0.5]:
                safe_a = np.maximum(a, 1e-10)
                safe_g = np.maximum(g, 1e-10)
                a_trial = safe_a ** (1.0 - t) * safe_g ** t
                a_trial = np.maximum(a_trial, 0.0)
                if a_trial.sum() > 1e-10:
                    a_trial = a_trial / a_trial.sum()
                    candidates.append(a_trial)

        best_trial = None
        best_trial_J = current_J
        for a_trial in candidates:
            J_trial = compute_J(a_trial)
            if J_trial < best_trial_J - 1e-12:
                best_trial_J = J_trial
                best_trial = a_trial

        if best_trial is None:
            stall += 1
            if stall > 20:
                break
            tighten = max(0.001, tighten * 0.8)
            continue

        a = best_trial
        current_J = best_trial_J
        stall = 0

        if current_J < best_J:
            best_J = current_J
            best_a = a.copy()

        # Adapt tightening
        tighten = min(0.02, max(0.001, tighten * 0.995))

        if it % 50 == 0:
            print(f"    MD-LP iter {it}: J={current_J:.8f} t={time.monotonic()-t0_lp:.1f}s", flush=True)

    return best_a, best_J


def coordinate_descent_exact(a, max_passes=10, deadline=None):
    """
    Coordinate descent with near-exact 1D minimization.
    For each coordinate, finds the optimal value by checking:
    - The analytical minimum for the currently active constraint
    - Boundary at 0
    - A few candidate points from golden section
    """
    a = np.maximum(a.astype(np.float64), 0.0)
    S = a.sum()
    if S < 1e-10:
        return a
    a = a / S
    n = len(a)
    b = fast_conv(a)
    current_J = 2.0 * n * b.max() / (S * S)
    best_a = a.copy()
    best_J = current_J

    for p in range(max_passes):
        if deadline is not None and time.monotonic() > deadline:
            break

        S = a.sum()
        b = fast_conv(a)
        bmax = b.max()
        improved = False
        indices = np.arange(n)
        np.random.shuffle(indices)

        for i in indices:
            old_val = a[i]
            S_prime = S - old_val

            # Find the best x for coordinate i
            # c_k(x) for k in [i, min(i+n-1, 2n-2)]:
            #   if k == 2*i: x^2 + (b[2i] - old_val^2)
            #   else: 2*a[k-i]*x + (b[k] - 2*old_val*a[k-i])
            # For k outside: b[k] (constant)

            # Precompute the constant parts
            const_max = 0.0
            # Constants from k outside [i, i+n-1]
            for k in range(2 * n - 1):
                if k < i or k > i + n - 1:
                    if b[k] > const_max:
                        const_max = b[k]

            # For each k in [i, min(i+n-1, 2n-2)], we have a linear (or quadratic) function
            # We want to minimize f(x) = 2n * max(const_max, max_k c_k(x)) / (S' + x)^2

            # Strategy: try the analytical optimum for the currently dominant constraint
            # plus a small golden section search

            # Find which constraint is most binding at x = old_val
            active_k = -1
            active_val = const_max
            for k in range(max(0, i), min(2 * n, i + n)):
                if k == 2 * i:
                    val = old_val * old_val
                elif 0 <= k - i < n:
                    val = b[k]
                else:
                    val = 0
                if val > active_val:
                    active_val = val
                    active_k = k

            # Candidate 1: x = 0
            # Candidate 2: analytical for active constraint
            # Candidate 3: golden section

            candidates = [0.0, old_val]

            if active_k >= 0 and active_k != 2 * i:
                # Linear: c_k(x) = 2*a[k-i]*x + (b[k] - 2*old_val*a[k-i])
                alpha = 2.0 * a[k - i] if 0 <= k - i < n else 0.0
                # Wait, k is active_k
                alpha = 2.0 * a[active_k - i]
                beta = b[active_k] - 2.0 * old_val * a[active_k - i]
                if alpha > 1e-10:
                    # Minimize (alpha*x + beta)/(S'+x)^2
                    # f'(x) = 0 => alpha*S' - alpha*x - 2*beta = 0 => x = S' - 2*beta/alpha
                    x_cand = S_prime - 2.0 * beta / alpha
                    if x_cand > 0:
                        # Check if this is valid (i.e., this constraint is still active at x_cand)
                        candidates.append(min(x_cand, 3.0 * old_val + S_prime / n))
                elif beta < const_max:
                    # If the linear part is below const_max, the binding constraint is const_max
                    # Minimize const_max/(S'+x)^2 => always decreasing => x as large as possible
                    # But other constraints might kick in
                    candidates.append(S_prime / n * 2.0)

            # Always add a few more candidates
            hi_bound = max(3.0 * old_val, S_prime / n * 3.0, 0.01)
            candidates.append(hi_bound * 0.5)
            candidates.append(hi_bound)

            # Evaluate all candidates
            best_x = old_val
            best_f = 2.0 * n * bmax / (S * S)

            for x in candidates:
                x = max(0.0, x)
                new_S = S_prime + x
                if new_S < 1e-10:
                    continue
                # Compute max convolution
                max_c = const_max
                for k in range(max(0, i), min(2 * n, i + n)):
                    if k == 2 * i:
                        ck = x * x
                    elif 0 <= k - i < n:
                        ck = b[k] - 2.0 * old_val * a[k - i] + 2.0 * x * a[k - i]
                    else:
                        ck = b[k]
                    if ck > max_c:
                        max_c = ck
                f_val = 2.0 * n * max_c / (new_S * new_S)
                if f_val < best_f - 1e-14:
                    best_f = f_val
                    best_x = x

            # Also do a quick golden section around best_x
            if best_x > 0:
                lo = max(0.0, best_x * 0.5)
                hi = best_x * 1.5 + 0.01
                gr = (np.sqrt(5.0) - 1.0) / 2.0
                c_pt = hi - gr * (hi - lo)
                d_pt = lo + gr * (hi - lo)

                def eval_x(x):
                    new_S = S_prime + x
                    if new_S < 1e-10:
                        return 1e10
                    max_c = const_max
                    for k in range(max(0, i), min(2 * n, i + n)):
                        if k == 2 * i:
                            ck = x * x
                        elif 0 <= k - i < n:
                            ck = b[k] - 2.0 * old_val * a[k - i] + 2.0 * x * a[k - i]
                        else:
                            ck = b[k]
                        if ck > max_c:
                            max_c = ck
                    return 2.0 * n * max_c / (new_S * new_S)

                fc = eval_x(c_pt)
                fd = eval_x(d_pt)
                for _ in range(15):
                    if fc < fd:
                        hi = d_pt
                        d_pt = c_pt
                        fd = fc
                        c_pt = hi - gr * (hi - lo)
                        fc = eval_x(c_pt)
                    else:
                        lo = c_pt
                        c_pt = d_pt
                        fc = fd
                        d_pt = lo + gr * (hi - lo)
                        fd = eval_x(d_pt)
                x_gs = (lo + hi) / 2.0
                f_gs = eval_x(x_gs)
                if f_gs < best_f - 1e-14:
                    best_f = f_gs
                    best_x = x_gs

            if best_x != old_val:
                a[i] = best_x
                S = S_prime + best_x
                improved = True

        # Recompute full J after pass
        b = fast_conv(a)
        S = a.sum()
        new_J = 2.0 * n * b.max() / (S * S)
        if new_J < best_J:
            best_J = new_J
            best_a = a.copy()
        if not improved:
            break

    return best_a


def lse_gradient_descent(a, deadline, tau_start=50.0, tau_end=500.0, n_iters=200):
    """
    Gradient descent on the LSE-smoothed objective with adaptive step size.
    Uses the analytic gradient and projects onto the simplex.
    """
    a = np.maximum(a.astype(np.float64), 0.0)
    a = a / a.sum()
    n = len(a)
    current_J = compute_J(a)
    best_a = a.copy()
    best_J = current_J

    step_size = 1.0 / np.sqrt(n)
    patience = 30
    stall_count = 0

    for it in range(n_iters):
        if time.monotonic() > deadline:
            break

        # Anneal tau
        progress = it / max(1, n_iters)
        tau = tau_start * (tau_end / tau_start) ** progress

        J_smooth, grad = lse_obj_grad(a, tau)

        # Gradient step with projection onto simplex
        a_trial = a - step_size * grad
        a_trial = project_simplex(a_trial)

        # Evaluate true (non-smooth) objective
        J_true = compute_J(a_trial)

        if J_true < current_J - 1e-12:
            current_J = J_true
            a = a_trial
            step_size = min(step_size * 1.1, 2.0)
            stall_count = 0
            if J_true < best_J:
                best_J = J_true
                best_a = a_trial.copy()
        else:
            stall_count += 1
            step_size *= 0.8

            # Try a smaller step
            a_trial2 = a - step_size * 0.5 * grad
            a_trial2 = project_simplex(a_trial2)
            J_trial2 = compute_J(a_trial2)
            if J_trial2 < current_J - 1e-12:
                current_J = J_trial2
                a = a_trial2
                stall_count = 0
                if J_trial2 < best_J:
                    best_J = J_trial2
                    best_a = a_trial2.copy()

        if stall_count > patience:
            break

        if it % 30 == 0:
            print(f"    LSE-GD iter {it}: J={current_J:.8f} tau={tau:.1f} step={step_size:.4f}", flush=True)

    return best_a, best_J


def lse_lbfgs(a, deadline, max_iter=100):
    """
    Use scipy L-BFGS-B on the LSE-smoothed objective.
    """
    from scipy.optimize import minimize as sp_minimize

    a = np.maximum(a.astype(np.float64), 0.0)
    a = a / a.sum()
    n = len(a)

    def wrapper(z):
        # z is the parameter vector, a = softmax(z) to ensure positivity and sum=1
        z_max = z.max()
        exp_z = np.exp(z - z_max)
        a_local = exp_z / exp_z.sum()
        J, grad_z = _lse_obj_grad_softmax(a_local, z, 100.0)
        return J, grad_z

    def _lse_obj_grad_softmax(a_local, z, tau):
        # Compute J and dJ/dz where a = softmax(z)
        n_l = len(a_local)
        b = fast_conv(a_local)
        S = a_local.sum()  # Should be 1

        tb = tau * b
        tb_max = tb.max()
        exp_tb = np.exp(tb - tb_max)
        sum_exp = exp_tb.sum()
        lse_val = (tb_max + np.log(sum_exp)) / tau
        p = exp_tb / sum_exp

        # dLSE/da_i
        grad_lse_a = np.zeros(n_l)
        for i in range(n_l):
            j_max = min(n_l - 1, 2 * n_l - 2 - i)
            if j_max >= 0:
                grad_lse_a[i] = 2.0 * np.dot(a_local[:j_max + 1], p[i:i + j_max + 1])

        J = 2.0 * n_l * lse_val / (S * S)
        grad_J_da = (2.0 * n_l / (S * S)) * (grad_lse_a - 2.0 * lse_val / S)

        # Chain rule: da_i/dz_j = a_i*(delta_ij - a_j)
        # dJ/dz_j = sum_i grad_J_da_i * a_i * (delta_ij - a_j)
        #         = a_j * grad_J_da_j - a_j * sum_i grad_J_da_i * a_i
        grad_dot = np.dot(grad_J_da, a_local)
        grad_J_dz = a_local * (grad_J_da - grad_dot)

        return J, grad_J_dz

    # Convert to softmax parameters
    z0 = np.log(np.maximum(a, 1e-10))

    try:
        result = sp_minimize(
            fun=lambda z: _lse_obj_grad_softmax(_softmax(z), z, 100.0)[0],
            x0=z0,
            jac=lambda z: _lse_obj_grad_softmax(_softmax(z), z, 100.0)[1],
            method='L-BFGS-B',
            options={'maxiter': max_iter, 'maxfun': 500,
                     'iprint': -1,
                     'gtol': 1e-8}
        )
        if result.success or result.fun < compute_J(a):
            a_new = _softmax(result.x)
            J_new = compute_J(a_new)
            if J_new < compute_J(a):
                return a_new, J_new
    except Exception:
        pass

    return a, compute_J(a)


def _softmax(z):
    z_max = z.max()
    exp_z = np.exp(z - z_max)
    return exp_z / exp_z.sum()


def resample_seq(src, dest_n):
    """Resample 1D sequence to new length via linear interpolation."""
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)


def generate_structured_init(n, rng):
    """Generate a structured initial sequence."""
    t = np.linspace(0, 1, n, endpoint=False)
    choices = [
        np.ones(n),
        1.0 + 0.3 * np.cos(2 * np.pi * 2 * t),
        1.0 + 0.3 * np.cos(2 * np.pi * 3 * t),
        1.0 + 0.2 * np.cos(2 * np.pi * 5 * t) + 0.2 * np.cos(2 * np.pi * 7 * t),
        np.abs(np.sin(np.pi * t)) + 0.2,
        0.5 + 0.5 * np.cos(2 * np.pi * t) ** 2,
        np.exp(-4 * (t - 0.5) ** 2) + 0.1,
        0.5 + 0.3 * np.cos(2 * np.pi * 4 * t) + 0.2 * np.cos(2 * np.pi * 11 * t),
        # Flat-top-ish
        np.where(np.abs(t - 0.5) < 0.3, 1.0, 0.3),
        # Two peaks
        np.exp(-20 * (t - 0.25) ** 2) + np.exp(-20 * (t - 0.75) ** 2) + 0.05,
        # Sinusoidal with growing frequency
        1.0 + 0.3 * np.sin(2 * np.pi * (2 + 3 * t) * t),
    ]
    idx = rng.integers(0, len(choices))
    a = choices[idx].copy()
    a = np.maximum(a, 0.0)
    if a.sum() < 1e-6:
        a = np.ones(n)
    a = a / a.sum()
    return a


class Tracker:
    def __init__(self):
        self.best = None
        self.best_score = float('inf')
        self.t_start = time.monotonic()

    def update(self, a):
        a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
        if a.size == 0 or a.sum() < 0.01:
            return
        score = float(evaluate_sequence(a.tolist()))
        if np.isfinite(score) and score < self.best_score:
            self.best = a.copy()
            self.best_score = score
            elapsed = time.monotonic() - self.t_start
            print(f"  *** NEW BEST: J={score:.8f} n={a.size} t={elapsed:.0f}s", flush=True)

    def report(self, label=""):
        if self.best is not None:
            print(f"  {label}: J={self.best_score:.8f} n={self.best.size} "
                  f"t={time.monotonic()-self.t_start:.0f}s", flush=True)


def optimize_single(a, deadline, tracker, phase_label=""):
    """Run a full optimization pipeline on a single sequence."""
    a = np.maximum(a.astype(np.float64), 0.0)
    if a.sum() < 1e-10:
        return a
    a = a / a.sum()
    n = len(a)

    current_J = compute_J(a)
    tracker.update(a)

    # Step 1: LP mirror descent
    t_rem = deadline - time.monotonic()
    if t_rem > 30:
        lp_deadline = time.monotonic() + min(t_rem * 0.4, 200)
        a, J_lp = mirror_descent_lp(a, 300, lp_deadline)
        if J_lp < current_J:
            current_J = J_lp
        tracker.update(a)
        print(f"    {phase_label} LP: J={current_J:.8f}", flush=True)

    # Step 2: Coordinate descent
    t_rem = deadline - time.monotonic()
    if t_rem > 20:
        cd_deadline = time.monotonic() + min(t_rem * 0.3, 100)
        a = coordinate_descent_exact(a, max_passes=5, deadline=cd_deadline)
        a = a / a.sum()
        current_J = compute_J(a)
        tracker.update(a)
        print(f"    {phase_label} CD: J={current_J:.8f}", flush=True)

    # Step 3: LSE gradient descent
    t_rem = deadline - time.monotonic()
    if t_rem > 20:
        lse_deadline = time.monotonic() + min(t_rem * 0.3, 100)
        a, J_lse = lse_gradient_descent(a, lse_deadline, n_iters=100)
        if J_lse < current_J:
            current_J = J_lse
        a = a / a.sum()
        tracker.update(a)
        print(f"    {phase_label} LSE-GD: J={current_J:.8f}", flush=True)

    # Step 4: Another round of LP + CD
    t_rem = deadline - time.monotonic()
    if t_rem > 30:
        lp_deadline = time.monotonic() + min(t_rem * 0.4, 80)
        a, J_lp2 = mirror_descent_lp(a, 100, lp_deadline, use_mw=True, mw_power=0.3)
        if J_lp2 < current_J:
            current_J = J_lp2
        a = a / a.sum()
        cd_deadline = time.monotonic() + min(t_rem * 0.2, 40)
        a = coordinate_descent_exact(a, max_passes=3, deadline=cd_deadline)
        a = a / a.sum()
        current_J = compute_J(a)
        tracker.update(a)
        print(f"    {phase_label} Final: J={current_J:.8f}", flush=True)

    return a


def propose_candidate(seed=42, budget_s=1500, **kwargs):
    """
    Main search: multi-phase optimization combining LP mirror descent,
    coordinate descent, and LSE gradient methods.
    """
    global t0_lp
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + float(budget_s) - 30.0
    t0_lp = time.monotonic()
    tracker = Tracker()

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        if p.ndim == 1 and p.size > 0 and np.all(np.isfinite(p)):
            parent = np.maximum(p, 0.0)
            if parent.sum() > 0.01:
                tracker.update(parent)
                print(f"Parent: n={parent.size}, J={tracker.best_score:.8f}", flush=True)
    except (NameError, AttributeError):
        pass

    if parent is None:
        parent = np.ones(300)
        tracker.update(parent)
        print(f"No parent, using uniform n=300", flush=True)

    n_parent = parent.size

    # === Phase 1: Intensive optimization on parent ===
    print(f"\n{'='*60}", flush=True)
    print(f"Phase 1: Optimize parent (n={n_parent})", flush=True)
    print(f"{'='*60}", flush=True)

    phase1_deadline = time.monotonic() + min(350, (deadline - time.monotonic()) * 0.3)
    a_best = optimize_single(parent.copy(), phase1_deadline, tracker, "P1")

    # === Phase 2: Try perturbed versions of parent ===
    print(f"\n{'='*60}", flush=True)
    print(f"Phase 2: Perturbation restarts", flush=True)
    print(f"{'='*60}", flush=True)

    phase2_deadline = time.monotonic() + min(250, (deadline - time.monotonic()) * 0.25)
    temps = [0.03, 0.05, 0.02, 0.08, 0.01]
    for temp in temps:
        if time.monotonic() > phase2_deadline:
            break
        a_pert = tracker.best.copy() + rng.normal(0, temp, tracker.best.size)
        a_pert = np.maximum(a_pert, 0.0)
        if a_pert.sum() > 1e-6:
            a_pert = a_pert / a_pert.sum()
        sub_deadline = min(phase2_deadline, time.monotonic() + 40)
        a_pert = optimize_single(a_pert, sub_deadline, tracker, f"P2(t={temp})")

    # === Phase 3: Multi-length exploration ===
    print(f"\n{'='*60}", flush=True)
    print(f"Phase 3: Multi-length exploration", flush=True)
    print(f"{'='*60}", flush=True)

    phase3_deadline = time.monotonic() + min(300, (deadline - time.monotonic()) * 0.3)

    # Try lengths around the parent and some diverse ones
    candidate_lengths = sorted(set([
        n_parent - 10, n_parent - 5, n_parent + 5, n_parent + 10,
        n_parent + 20, n_parent - 20,
        200, 256, 300, 400, 500, 384, 420,
    ]))
    candidate_lengths = [l for l in candidate_lengths if 100 <= l <= 800]

    for nl in candidate_lengths:
        if time.monotonic() > phase3_deadline:
            break
        if nl == n_parent:
            continue

        # Resample best to new length
        a_res = resample_seq(tracker.best, nl)
        a_res = np.maximum(a_res, 0.0)
        if a_res.sum() > 1e-6:
            a_res = a_res / a_res.sum()
        else:
            a_res = generate_structured_init(nl, rng)

        # Also try a fresh structured init
        a_fresh = generate_structured_init(nl, rng)

        # Optimize both, keep the better
        sub_deadline = min(phase3_deadline, time.monotonic() + 45)
        a1 = optimize_single(a_res, sub_deadline, tracker, f"P3(n={nl},res)")
        if time.monotonic() < phase3_deadline - 20:
            a2 = optimize_single(a_fresh, min(phase3_deadline, time.monotonic() + 30),
                                 tracker, f"P3(n={nl},fresh)")

    # === Phase 4: L-BFGS refinement on best ===
    print(f"\n{'='*60}", flush=True)
    print(f"Phase 4: L-BFGS refinement", flush=True)
    print(f"{'='*60}", flush=True)

    if time.monotonic() < deadline - 60:
        t_lse = time.monotonic()
        a_lse, J_lse = lse_lbfgs(tracker.best.copy(), min(deadline - 60, time.monotonic() + 120),
                                  max_iter=80)
        a_lse = a_lse / a_lse.sum()
        tracker.update(a_lse)
        print(f"  L-BFGS: J={compute_J(a_lse):.8f} ({time.monotonic()-t_lse:.1f}s)", flush=True)

    # === Phase 5: Final intensive polish ===
    print(f"\n{'='*60}", flush=True)
    print(f"Phase 5: Final polish", flush=True)
    print(f"{'='*60}", flush=True)

    if time.monotonic() < deadline - 10:
        a_final = optimize_single(tracker.best.copy(), deadline - 5, tracker, "P5")

    # Final check
    if tracker.best is None:
        tracker.best = np.ones(256)
        tracker.best_score = compute_J(tracker.best)

    # Final evaluation
    final_score = float(evaluate_sequence([float(x) for x in tracker.best]))
    print(f"\n{'='*60}", flush=True)
    print(f"FINAL: J={final_score:.10f}, n={tracker.best.size}", flush=True)
    print(f"Total time: {time.monotonic()-tracker.t_start:.1f}s", flush=True)
    print(f"{'='*60}", flush=True)

    return [float(x) for x in tracker.best]
