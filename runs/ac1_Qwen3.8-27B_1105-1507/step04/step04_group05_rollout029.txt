import numpy as np
from scipy.optimize import linprog
from scipy.sparse import lil_matrix
import time


def fast_conv(a):
    """Full self-convolution via FFT."""
    n = len(a)
    s = 1
    while s < 2 * n - 1:
        s <<= 1
    fa = np.fft.rfft(a, s)
    return np.fft.irfft(fa * fa, s)[:2 * n - 1]


def compute_J(a):
    """Compute J(a) = 2*n*max(conv(a,a))/(sum(a))^2."""
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))


def build_A_sparse(a):
    """Build sparse constraint matrix for LP: A[k,j] = a[k-j] if valid."""
    n = len(a)
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        jl = max(0, k - n + 1)
        jh = min(n - 1, k)
        for j in range(jl, jh + 1):
            rows.append(k)
            cols.append(j)
            vals.append(a[k - j])
    A = lil_matrix((2 * n - 1, n))
    for idx in range(len(rows)):
        A[rows[idx], cols[idx]] = vals[idx]
    return A.tocsr()


def lp_best_response(a, M):
    """LP: max sum(g) s.t. (a*g)_k <= M for all k, g >= 0."""
    n = len(a)
    c = -np.ones(n)
    A = build_A_sparse(a)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
    if r.success and r.x is not None:
        g = np.maximum(r.x, 0.0)
        if g.sum() > 1e-10:
            return g
    return None

    a = np.maximum(a.astype(np.float64).copy(), 0.0)
    S = a.sum()
    if S < 1e-10:
        return a
    n = len(a)
    b = fast_conv(a)
    
    indices = np.arange(n)
    np.random.shuffle(indices)
    
    for i in indices:
        ai = a[i]
        if ai < 1e-15:
            # For zero coordinates, just try a small value
            pass
        
        k_lo = max(0, i)
        k_hi = min(2 * n - 2, i + n - 1)
        k_vals = np.arange(k_lo, k_hi + 1)
        ki_vals = k_vals - i
        a_ki = a[ki_vals]
        
        # Precompute for vectorized evaluation
        P = b[k_vals].astype(np.float64).copy() - 2.0 * ai * a_ki
        Q = 2.0 * a_ki.copy()
        is_2i = (k_vals == 2 * i)
        if is_2i.any():
            P[is_2i] = b[k_vals[is_2i]] - ai * ai
            Q[is_2i] = 0.0
        
        # Unaffected max
        all_mask = np.ones(2 * n - 1, dtype=bool)

        if all_mask.any():
            unaffected_max = np.max(b[all_mask])
        else:
            unaffected_max = 0.0
        
        def obj_1d(x):
            new_S = S - ai + x
            if new_S < 1e-10:
                return 1e10
            vals = P + Q * x
            if is_2i.any():
                idx2i = np.where(is_2i)[0][0]
                vals[idx2i] = b[k_vals[idx2i]] - ai * ai + x * x
            max_c = max(np.max(vals), unaffected_max)
            return 2.0 * n * max_c / (new_S * new_S)
        
        # Golden section search
        gr = (np.sqrt(5.0) - 1.0) / 2.0
        lo = 0.0
        hi = max(3.0 * ai + 0.1, S / n * 2.0, 0.01)
        
        c_pt = hi - gr * (hi - lo)
        d_pt = lo + gr * (hi - lo)
        fc = obj_1d(c_pt)
        fd = obj_1d(d_pt)
        
        for _ in range(35):
            if fc < fd:
                hi = d_pt
                d_pt = c_pt
                fd = fc
                c_pt = hi - gr * (hi - lo)
                fc = obj_1d(c_pt)
            else:
                lo = c_pt
                c_pt = d_pt
                fc = fd
                d_pt = lo + gr * (hi - lo)
                fd = obj_1d(d_pt)
        
        x_opt = max(0.0, (lo + hi) / 2.0)
        
        if abs(x_opt - ai) > 1e-12:
            delta = x_opt - ai
            # Update b incrementally
            b[k_lo:k_hi + 1] += 2.0 * delta * a_ki
            if is_2i.any():
                b[2 * i] += delta * delta
            a[i] = x_opt
            S += delta
    
    return a


def coordinate_descent_polish(a, deadline, tracker, min_time_per_pass=5.0):
    """Multiple passes of coordinate descent until deadline."""
    a = np.maximum(a.astype(np.float64).copy(), 0.0)
    a /= a.sum()
    current_J = compute_J(a)
    tracker.update(a)
    
    stall = 0
    while time.monotonic() < deadline:
        t0 = time.monotonic()
        a = coordinate_descent_pass_fast(a)
        a = np.maximum(a, 0.0)
        a /= a.sum()
        new_J = compute_J(a)
        
        if new_J < current_J - 1e-12:
            current_J = new_J
            tracker.update(a)
            stall = 0
        else:
            stall += 1
            if stall > 5:
                break
            if stall > 10:
                break
        
        if time.monotonic() - t0 < min_time_per_pass and stall == 0:
            # Continue
            pass
        elif stall >= 3:
            break
    
    return a


def analytic_gradient(a):
    """Compute subgradient of J(a) (normalized, sum=1)."""
    n = len(a)
    S = a.sum()
    if S < 1e-10:
        return np.zeros(n)
    b = fast_conv(a)
    b_max = b.max()
    
    # Find all active k's (within tolerance)
    tol = 1e-8 * b_max
    active = np.where(b >= b_max - tol)[0]
    
    # Subgradient: average over active set
    grad = np.zeros(n)
    for k in active:
        # db_k/da[j] = 2*a[k-j] if 0 <= k-j < n
        valid = np.arange(n)
        kj = k - valid
        mask = (kj >= 0) & (kj < n)
        grad += 2.0 * a[kj[mask]]
    
    grad /= len(active)
    
    # dJ/da[j] = 2n * (grad_b[j] * S - 2 * b_max) / S^3
    # where grad_b[j] = db_max/da[j]
    j = S  # sum
    grad_J = 2.0 * n * (grad * j - 2.0 * b_max) / (j ** 3)
    
    return grad_J


def gradient_descent_step(a, step_size):
    """One gradient descent step with projection to simplex."""
    a = np.maximum(a, 0.0)
    S = a.sum()
    if S < 1e-10:
        return a
    a_norm = a / S
    
    grad = analytic_gradient(a_norm)
    a_new = a_norm - step_size * grad
    a_new = np.maximum(a_new, 0.0)
    if a_new.sum() < 1e-10:
        return a
    return a_new


def gradient_descent_phase(a, deadline, tracker):
    """Phase of gradient descent with adaptive step size."""
    a = np.maximum(a.astype(np.float64).copy(), 0.0)
    a /= a.sum()
    current_J = compute_J(a)
    tracker.update(a)
    
    step_size = 0.1
    stall = 0
    
    while time.monotonic() < deadline:
        a_new = gradient_descent_step(a, step_size)
        a_new /= a_new.sum()
        new_J = compute_J(a_new)
        
        if new_J < current_J - 1e-12:
            a = a_new
            current_J = new_J
            tracker.update(a)
            stall = 0
            step_size = min(step_size * 1.05, 0.5)
        else:
            stall += 1
            step_size *= 0.7
            if stall > 15 or step_size < 1e-6:
                break
    
    return a


def run_lp_iterations_improved(a, max_iters, deadline, tracker):
    """Improved LP mirror descent."""
    a = np.maximum(a.astype(np.float64).copy(), 0.001)
    a /= a.sum()
    n = len(a)
    current_J = compute_J(a)
    tracker.update(a)
    
    tighten = 0.005
    stall_count = 0
    
    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        
        b = fast_conv(a)
        M = float(b.max())
        M_target = M * (1.0 - tighten)
        
        g = lp_best_response(a, M_target)
        if g is None:
            M_target = M * (1.0 - tighten * 0.3)
            g = lp_best_response(a, M_target)
        if g is None:
            tighten *= 0.5
            if tighten < 1e-7:
                tighten = 0.005
            continue
        
        g = np.maximum(g, 0.0)
        g_sum = g.sum()
        if g_sum < 1e-10:
            break
        g /= g_sum
        
        # Try multiple step sizes
        progress = min(1.0, it / max(1, max_iters))
        base_t = max(0.02, 0.4 * (1.0 - 0.5 * progress))
        candidates_t = [base_t, base_t * 0.5, base_t * 1.5, base_t * 0.3, base_t * 2.0]
        
        best_a = None
        best_J = current_J
        
        for t in candidates_t:
            a_trial = (1.0 - t) * a + t * g
            a_trial = np.maximum(a_trial, 0.0)
            s = a_trial.sum()
            if s < 1e-10:
                continue
            a_trial /= s
            J_trial = compute_J(a_trial)
            if J_trial < best_J - 1e-14:
                best_J = J_trial
                best_a = a_trial
        
        if best_a is None:
            stall_count += 1
            if stall_count > 30:
                # Try with much looser tightening
                tighten = max(0.001, tighten * 0.5)
                if tighten < 5e-5:
                    break
            else:
                tighten *= 0.7
            continue
        
        a = best_a
        current_J = best_J
        stall_count = 0
        tighten = min(0.03, tighten * 1.02)
        
        if it % 25 == 0:
            tracker.update(a)
    
    tracker.update(a)
    return a


def variance_flattening_step(a, step_size=0.01):
    """Step to reduce variance of convolution (auxiliary objective)."""
    a = np.maximum(a, 0.0)
    S = a.sum()
    if S < 1e-10:
        return a
    a_norm = a / S
    n = len(a)
    b = fast_conv(a_norm)
    
    # Gradient of var(b) = E[b^2] - E[b]^2 w.r.t. a
    # d(b_k)/da_j = 2*a[k-j] (as before)
    # d(var(b))/da_j = 2 * mean(b) * (mean(db/da_j) - mean(db/da_j))... 
    # Actually let's just use the gradient of sum(b_k^2) / (2n-1)^2 as proxy
    # d(sum b_k^2)/da_j = 2 * sum_k b_k * db_k/da_j
    
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        bk = b[k]
        if bk < 1e-15:
            continue
        # db_k/da_j = 2*a[k-j] if valid
        kj = k - np.arange(n)
        mask = (kj >= 0) & (kj < n)
        grad += 2.0 * bk * 2.0 * a[kj[mask]]
    
    # Normalize gradient
    gnorm = np.linalg.norm(grad)
    if gnorm > 1e-15:
        grad /= gnorm
    
    a_new = a_norm - step_size * grad
    a_new = np.maximum(a_new, 0.0)
    if a_new.sum() < 1e-10:
        return a
    return a_new / a_new.sum()


def resample_seq(src, dest_n):
    """Resample 1D sequence to new length via interpolation."""
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)


class ScoreTracker:
    def __init__(self):
        self.best = None
        self.best_score = float('inf')
        self.t_start = time.monotonic()
    
    def update(self, a):
        a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
        if a.size == 0 or a.sum() < 0.01:
            return
        score = float(evaluate_sequence(a.tolist()))
        if np.isfinite(score) and score < self.best_score:
            self.best = a.copy()
            self.best_score = score
            elapsed = time.monotonic() - self.t_start
            print(f"  * NEW BEST: J={score:.8f} n={a.size} t={elapsed:.0f}s", flush=True)
    
    def report(self, label=""):
        if self.best is not None:
            print(f"  {label} best: J={self.best_score:.8f} n={self.best.size} t={time.monotonic()-self.t_start:.0f}s", flush=True)


def generate_structured_inits(n, rng):
    """Generate diverse structured initial sequences."""
    inits = []
    t = np.linspace(0, 1, n, endpoint=False)
    
    # Uniform
    inits.append(np.ones(n))
    
    # U-shape (high at ends)
    inits.append(1.0 + 0.5 * (2 * t - 1.0) ** 2)
    
    # Cosine windows
    for beta in [1.0, 2.0, 3.0, 5.0]:
        inits.append(0.5 * (1 + np.cos(np.pi * (2 * t - 1) ** beta)))
    
    # Multi-frequency
    for freq in [2, 3, 4, 5, 7]:
        phase = rng.uniform(0, 2 * np.pi)
        inits.append(1.0 + 0.3 * np.cos(2 * np.pi * freq * t + phase))
    
    # Gaussian mixtures
    for _ in range(3):
        centers = rng.uniform(0.1, 0.9, 2)
        widths = rng.uniform(0.05, 0.2, 2)
        sig = sum(np.exp(-0.5 * ((t - c) / w) ** 2) for c, w in zip(centers, widths))
        inits.append(0.3 + sig)
    
    # Symmetric U-shape variants
    inits.append(0.5 + 0.5 * (2 * t - 1.0) ** 4)
    inits.append(0.5 + 0.5 * np.abs(2 * t - 1.0))
    
    # Fejer kernel-like (sinc^2)
    x = (t - 0.5) * n
    with np.errstate(invalid='ignore'):
        sinc_x = np.sinc(x / n)
    inits.append(np.maximum(sinc_x ** 2, 0.01))
    
    return inits


def run_cd_lp_alternating(a, deadline, tracker):
    """Alternate between coordinate descent and LP steps."""
    a = np.maximum(a.astype(np.float64).copy(), 0.001)
    a /= a.sum()
    n = len(a)
    current_J = compute_J(a)
    tracker.update(a)
    
    it = 0
    while time.monotonic() < deadline:
        remaining = deadline - time.monotonic()
        if remaining < 10:
            break
        
        # CD phase
        cd_deadline = min(deadline, time.monotonic() + min(remaining * 0.5, 120))
        a = coordinate_descent_pass_fast(a)
        a = np.maximum(a, 0.0)
        a /= a.sum()
        new_J = compute_J(a)
        if new_J < current_J - 1e-13:
            current_J = new_J
            tracker.update(a)
        
        # LP phase (shorter)
        lp_deadline = min(deadline, time.monotonic() + min(remaining * 0.3, 30))
        if time.monotonic() < lp_deadline:
            a = run_lp_iterations_improved(a, max_iters=5, deadline=lp_deadline, tracker=tracker)
        
        it += 1
        if it % 10 == 0:
            print(f"    Alt iter {it}: J={compute_J(a):.8f}", flush=True)
    
    return a


def fine_tune(a, deadline, tracker):
    """Very fine coordinate descent with small steps for final polish."""
    a = np.maximum(a.astype(np.float64).copy(), 0.0)
    a /= a.sum()
    n = len(a)
    current_J = compute_J(a)
    tracker.update(a)
    
    b = fast_conv(a)
    S = a.sum()
    
    # Focus on coordinates near the "active" convolution peaks
    b_max = b.max()
    active_k = np.where(b >= b_max - 1e-4 * b_max)[0]
    
    # Coordinates that affect active k's
    active_coords = set()
    for k in active_k:
        for j in range(max(0, k - n + 1), min(n, k + 1)):
            active_coords.add(j)
    
    # Do multiple passes focusing on active coords first
    for sweep in range(20):
        if time.monotonic() > deadline:
            break
        
        # Active coords first
        coords = sorted(active_coords) + [j for j in range(n) if j not in active_coords]
        np.random.shuffle(coords)
        # Put active coords first half
        active_list = [c for c in coords if c in active_coords]
        other_list = [c for c in coords if c not in active_coords]
        final_order = active_list + other_list
        
        b = fast_conv(a)
        S = a.sum()
        
        for i in final_order:
            if time.monotonic() > deadline:
                break
            ai = a[i]
            
            k_lo = max(0, i)
            k_hi = min(2 * n - 2, i + n - 1)
            k_vals = np.arange(k_lo, k_hi + 1)
            ki_vals = k_vals - i
            a_ki = a[ki_vals]
            
            P = b[k_vals].astype(np.float64).copy() - 2.0 * ai * a_ki
            Q = 2.0 * a_ki.copy()
            is_2i = (k_vals == 2 * i)
            if is_2i.any():
                P[is_2i] = b[k_vals[is_2i]] - ai * ai
                Q[is_2i] = 0.0
            
            all_mask = np.ones(2 * n - 1, dtype=bool)
            all_mask[k_lo:k_hi + 1] = False
            unaffected_max = np.max(b[all_mask]) if all_mask.any() else 0.0
            
            def obj_1d(x):
                new_S = S - ai + x
                if new_S < 1e-10:
                    return 1e10
                vals = P + Q * x
                if is_2i.any():
                    idx2i = np.where(is_2i)[0][0]
                    vals[idx2i] = b[k_vals[idx2i]] - ai * ai + x * x
                max_c = max(np.max(vals), unaffected_max)
                return 2.0 * n * max_c / (new_S * new_S)
            
            # More iterations for fine tuning
            gr = (np.sqrt(5.0) - 1.0) / 2.0
            lo = 0.0
            hi = max(2.0 * ai + 0.01, S / n * 1.5, 0.005)
            
            c_pt = hi - gr * (hi - lo)
            d_pt = lo + gr * (hi - lo)
            fc = obj_1d(c_pt)
            fd = obj_1d(d_pt)
            
            for _ in range(45):
                if fc < fd:
                    hi = d_pt
                    d_pt = c_pt
                    fd = fc
                    c_pt = hi - gr * (hi - lo)
                    fc = obj_1d(c_pt)
                else:
                    lo = c_pt
                    c_pt = d_pt
                    fc = fd
                    d_pt = lo + gr * (hi - lo)
                    fd = obj_1d(d_pt)
            
            x_opt = max(0.0, (lo + hi) / 2.0)
            
            if abs(x_opt - ai) > 1e-14:
                delta = x_opt - ai
                b[k_lo:k_hi + 1] += 2.0 * delta * a_ki
                if is_2i.any():
                    b[2 * i] += delta * delta
                a[i] = x_opt
                S += delta
        
        a = np.maximum(a, 0.0)
        a /= a.sum()
        new_J = compute_J(a)
        if new_J < current_J - 1e-14:
            current_J = new_J
            tracker.update(a)
    
    return a


def propose_candidate(seed=42, budget_s=1500, **kwargs):
    """
    Main search function combining fast CD, LP, gradient descent, and fine-tuning.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + float(budget_s) - 30.0
    tracker = ScoreTracker()
    
    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        if p.ndim == 1 and p.size > 0 and np.all(np.isfinite(p)):
            parent = np.maximum(p, 0.0)
            tracker.update(parent)
            print(f"Parent loaded: n={parent.size}, J={tracker.best_score:.8f}", flush=True)
    except (NameError, AttributeError):
        pass
    
    if parent is None:
        parent = np.ones(384)
        tracker.update(parent)
        print(f"No parent, using uniform n={parent.size}", flush=True)
    
    n_parent = parent.size
    
    # === Phase 1: Quick LP initialization (2 min) ===
    print(f"\n{'='*50} Phase 1: LP Init (n={n_parent}) {'='*50}", flush=True)
    t0 = time.monotonic()
    p1_deadline = min(deadline, time.monotonic() + 120)
    a = run_lp_iterations_improved(parent.copy(), max_iters=30, deadline=p1_deadline, tracker=tracker)
    print(f"  Phase 1 done: J={compute_J(a):.8f} ({time.monotonic()-t0:.1f}s)", flush=True)
    
    # === Phase 2: Intensive coordinate descent (8 min) ===
    if time.monotonic() < deadline - 600:
        print(f"\n{'='*50} Phase 2: Intensive CD {'='*50}", flush=True)
        t0 = time.monotonic()
        p2_deadline = min(deadline, time.monotonic() + 480)
        a = coordinate_descent_polish(a, p2_deadline, tracker)
        print(f"  Phase 2 done: J={compute_J(a):.8f} ({time.monotonic()-t0:.1f}s)", flush=True)
    
    # === Phase 3: LP + CD alternating (6 min) ===
    if time.monotonic() < deadline - 420:
        print(f"\n{'='*50} Phase 3: LP+CD Alt {'='*50}", flush=True)
        t0 = time.monotonic()
        p3_deadline = min(deadline, time.monotonic() + 360)
        a = run_cd_lp_alternating(a, p3_deadline, tracker)
        print(f"  Phase 3 done: J={compute_J(a):.8f} ({time.monotonic()-t0:.1f}s)", flush=True)
    
    # === Phase 4: Gradient descent (3 min) ===
    if time.monotonic() < deadline - 300:
        print(f"\n{'='*50} Phase 4: Gradient Descent {'='*50}", flush=True)
        t0 = time.monotonic()
        p4_deadline = min(deadline, time.monotonic() + 180)
        a = gradient_descent_phase(a, p4_deadline, tracker)
        # Follow up with CD
        a = coordinate_descent_pass_fast(a)
        a = np.maximum(a, 0.0) / np.sum(np.maximum(a, 0.0))
        tracker.update(a)
        print(f"  Phase 4 done: J={compute_J(a):.8f} ({time.monotonic()-t0:.1f}s)", flush=True)
    
    # === Phase 5: Fine tuning (5 min) ===
    if time.monotonic() < deadline - 240:
        print(f"\n{'='*50} Phase 5: Fine Tuning {'='*50}", flush=True)
        t0 = time.monotonic()
        p5_deadline = min(deadline, time.monotonic() + 300)
        best_for_finetune = tracker.best.copy() if tracker.best is not None else a
        a = fine_tune(best_for_finetune, p5_deadline, tracker)
        print(f"  Phase 5 done: J={compute_J(a):.8f} ({time.monotonic()-t0:.1f}s)", flush=True)
    
    # === Phase 6: Try nearby lengths (4 min) ===
    if time.monotonic() < deadline - 180:
        print(f"\n{'='*50} Phase 6: Length Exploration {'='*50}", flush=True)
        t0 = time.monotonic()
        p6_deadline = min(deadline, time.monotonic() + 240)
        
        nearby = [n_parent - 4, n_parent - 2, n_parent + 2, n_parent + 4,

        seen = set()
        uniq = []
        for l in nearby:
            if l not in seen:
                seen.add(l)
                uniq.append(l)
        
        for nl in uniq:
            if time.monotonic() > p6_deadline:

            a_r = resample_seq(tracker.best, nl)

        # Best LP
        a_best = tracker.best.copy() if tracker.best is not None else a
        a_best = run_lp_iterations_improved(a_best, max_iters=20, deadline=min(p7_deadline, time.monotonic() + 60), tracker=tracker)
        
        # Best CD
        if time.monotonic() < p7_deadline:
            a_best = fine_tune(a_best, p7_deadline, tracker)
        
        print(f"  Phase 7 done: J={compute_J(a_best):.8f} ({time.monotonic()-t0:.1f}s)", flush=True)
    
    # === Phase 8: Final quick CD passes ===
    if time.monotonic() < deadline - 10:
        a_final = tracker.best.copy() if tracker.best is not None else np.ones(256)
        a_final = coordinate_descent_pass_fast(a_final)
        a_final = np.maximum(a_final, 0.0)
        a_final /= a_final.sum()
        tracker.update(a_final)
    
    # Fallback
    if tracker.best is None:
        tracker.best = np.ones(256)
        tracker.best_score = compute_J(tracker.best)
    
    print(f"\n{'='*50} FINAL RESULT {'='*50}", flush=True)
    print(f"J = {tracker.best_score:.10f}, n = {tracker.best.size}", flush=True)
    print(f"Total time: {time.monotonic()-tracker.t_start:.1f}s", flush=True)
    
    return [float(x) for x in tracker.best]
