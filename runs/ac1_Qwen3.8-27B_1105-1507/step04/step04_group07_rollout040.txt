"""
Minimizes J(a) = 2*n*max(convolve(a,a))/(sum(a))^2 using:
1. Spectral Gerchberg-Saxton iterations for spectrum flattening (new)
2. Extended LP best response with adaptive mixing schedule (improved)

import numpy as np
from scipy.optimize import linprog
from scipy.sparse import csr_matrix
import time

_spar_cache = {}

def get_sparsity(n):
    if n not in _spar_cache:
        rows = []
        cols = []
        for k in range(2 * n - 1):
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            rows.extend([k] * (je - js + 1))
            cols.extend(range(js, je + 1))
        _spar_cache[n] = (np.array(rows, dtype=np.int32),
                          np.array(cols, dtype=np.int32), len(rows))
    return _spar_cache[n]

def fast_conv(a):
    n = len(a)
    s = 1
    while s < 2 * n - 1:
        s <<= 1
    fa = np.fft.rfft(a, s)
    return np.fft.irfft(fa * fa, s)[:2 * n - 1]

def compute_J(a):
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))

def spectral_shape(a, alpha, edge_decay=0.0):
    n = len(a)
    F = np.fft.fft(a)
    power = np.abs(F) ** 2
    mean_power = power.mean()
    new_power = (1.0 - alpha) * power + alpha * mean_power
    new_power = np.maximum(new_power, 1e-20)
    phase = np.angle(F)
    new_F = np.sqrt(new_power) * np.exp(1j * phase)
    new_a = np.fft.ifft(new_F).real
    if edge_decay > 0:
        w = np.ones(n)
        ew = max(1, n // 25)
        for i in range(ew):
            t = 1.0 - i / ew
            w[i] = 1.0 - edge_decay * t
            w[n - 1 - i] = 1.0 - edge_decay * t
        new_a = new_a * w
    new_a = np.maximum(new_a, 0.0)
    s = new_a.sum()
    if s < 1e-10:
        return a.copy()
    return new_a / s

def lp_best_response(a, M, time_limit=8):
    n = len(a)
    rows, cols, nnz = get_sparsity(n)
    vals = np.empty(nnz)
    for i in range(nnz):
        k = rows[i]
        j = cols[i]
        vals[i] = a[k - j]
    A = csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                options={'presolve': True, 'time_limit': time_limit})
    if r.success and r.x is not None:
        g = np.maximum(r.x, 0.0)
        gs = g.sum()
        if gs > 1e-10:
            return g / gs
    return None

def run_lp_extended(a, max_iters, deadline, tracker, tighten=0.0):

    a = a / a.sum()
    n = len(a)

    tracker.update(a)
    no_improve_count = 0

        else:
            g = lp_best_response(a, M, time_limit=8)

            break

        t_base = max(0.03, 0.9 - 0.6 * progress)
        candidates = [t_base, min(1.0, t_base * 1.3), t_base * 0.6, 0.95, 0.7, 0.4, 0.2]
        best_a = None
        best_J = current_J

            if t > 1.0:

            a_trial = (1.0 - t) * a + t * g
            a_trial = np.maximum(a_trial, 0.0)
            ss = a_trial.sum()
            if ss < 1e-10:
                continue

def nesterov_lse(a, max_iters, deadline, tracker):
    a = np.maximum(a.astype(np.float64), 1e-6)
    a = a / a.sum()
    n = len(a)
    current_J = compute_J(a)
    tracker.update(a)
    tau = 25.0
    lr = 0.8 / np.sqrt(n)
    beta = 0.85
    a_prev = a.copy()
    stalled = 0

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        y = a + beta * (a - a_prev)
        y = np.maximum(y, 1e-8)
        y = y / y.sum()
        b = fast_conv(y)
        b_max = b.max()
        exp_b = np.exp(tau * (b - b_max))
        p = exp_b / exp_b.sum()
        s_size = 1
        while s_size < 3 * n:
            s_size <<= 1
        fp = np.fft.rfft(p, s_size)
        fy = np.fft.rfft(y, s_size)
        corr = np.fft.irfft(fp * np.conj(fy), s_size)
        grad = 2.0 * corr[:n]
        a_new = y - lr * grad
        a_new = np.maximum(a_new, 1e-8)
        a_new = a_new / a_new.sum()
        J_new = compute_J(a_new)
        if J_new < current_J - 1e-15:
            a_prev = a.copy()
            a = a_new
            current_J = J_new
            stalled = 0
            if it % 15 == 0:
                tau = min(400.0, tau * 1.04)
                lr = min(lr * 1.015, 3.0 / n)
            beta = min(0.96, beta + 0.0008)
        else:
            stalled += 1
            lr *= 0.82
            if stalled > 25:
                tau = max(8.0, tau * 0.85)
                lr = 0.8 / np.sqrt(n)
                beta = 0.85
                a_prev = a.copy()
                stalled = 0
                if tau <= 8.5:
                    break
        if it % 60 == 0:
            tracker.update(a)
    tracker.update(a)
    return a

def freq_annealing(a, deadline, tracker, n_iter=400):
    a = a.copy().astype(np.float64)
    a = a / a.sum()
    n = len(a)
    current_J = compute_J(a)
    tracker.update(a)
    rng = np.random.default_rng(2024)
    temp = 0.08
    best_a = a.copy()
    best_J = current_J
    accept_count = 0

    for it in range(n_iter):
        if time.monotonic() > deadline:
            break
        F = np.fft.fft(a)
        noise_phase = rng.normal(0, temp * 0.5, n)
        noise_mag = rng.normal(0, temp, n)
        new_F = F * np.exp(1j * noise_phase) * (1.0 + noise_mag)
        for k in range(1, n - n // 2):
            new_F[k] = np.conj(new_F[n - k])
        new_F[0] = np.abs(new_F[0])
        new_a = np.fft.ifft(new_F).real
        new_a = np.maximum(new_a, 0.0)
        s = new_a.sum()
        if s < 1e-10:
            temp *= 0.97
            continue
        new_a = new_a / s
        J_new = compute_J(new_a)
        delta = J_new - current_J
        if delta < 0 or (temp > 1e-6 and rng.random() < np.exp(-delta / temp)):
            a = new_a
            current_J = J_new
            accept_count += 1
            if current_J < best_J:
                best_a = a.copy()
                best_J = current_J
                tracker.update(a)
        temp *= 0.997
    return best_a

def resample_seq(src, dest_n):
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)

def generate_inits(n, rng, parent):
    inits = []
    if parent is not None and parent.size > 0:
        inits.append(resample_seq(parent, n))
        inits.append(resample_seq(parent[::-1], n))
    inits.append(np.ones(n) / n)
    t = np.linspace(0, 1, n)
    # Asymmetric profiles
    inits.append((1.0 + t) / (2.0 * n))
    inits.append((1.0 - t) / (2.0 * n))
    inits.append((1.0 + t ** 2) / (n * (4.0 / 3.0)))
    inits.append((1.0 + t ** 3) / (n * 1.25))
    inits.append((1.0 + 2.0 * t) / (3.0 * n))
    # Off-center Gaussians
    for mu in [0.25, 0.35, 0.65, 0.75]:
        inits.append(np.exp(-10.0 * (t - mu) ** 2) + 0.01)
    # Cosine-based
    for f, amp in [(1, 0.4), (2, 0.3), (3, 0.35), (5, 0.25)]:
        inits.append(0.5 + amp * np.cos(2.0 * np.pi * f * t + 0.7))
    # Edge-windowed
    edge_w = 0.05 + 0.95 * np.sin(np.pi * t) ** 2
    inits.append(edge_w / edge_w.sum())
    # Two-hump asymmetric
    inits.append(0.4 * np.exp(-12.0 * (t - 0.25) ** 2) + 0.6 * np.exp(-8.0 * (t - 0.7) ** 2) + 0.02)
    inits.append(0.6 * np.exp(-8.0 * (t - 0.3) ** 2) + 0.4 * np.exp(-12.0 * (t - 0.75) ** 2) + 0.02)
    # Random smooth
    for _ in range(3):
        raw = rng.uniform(0, 1, n)
        ksize = max(3, n // 25)
        kernel = np.ones(ksize) / ksize
        inits.append(np.convolve(raw, kernel, mode='same') + 0.05)
    # Flat-spectrum inits
    for _ in range(3):
        half = n // 2 + 1
        angles = rng.uniform(0, 2 * np.pi, half)
        A = np.zeros(n, dtype=complex)
        A[:half] = np.exp(1j * angles[:half])
        for k in range(1, n - n // 2):
            A[k] = np.conj(A[n - k])
        aa = np.fft.ifft(A).real
        aa = np.maximum(aa, 0.0)
        ss = aa.sum()
        if ss > 1e-10:
            inits.append(aa / ss)
        else:
            inits.append(np.ones(n) / n)
    # Random Fourier low-freq
    n_coeff = min(n // 2 + 1, 10)
    coeffs = rng.normal(0, 1, n_coeff)
    sig = np.zeros(n)
    for i in range(n_coeff):
        sig += coeffs[i] * np.cos(2.0 * np.pi * i * t / n + rng.uniform(0, 6.28))
    inits.append(np.abs(sig) + 0.4)
    # Power-law asymmetric
    inits.append((t + 0.05) ** 0.7)
    inits.append((1.0 - t + 0.05) ** 0.7)
    for x in inits:
        x = np.maximum(x, 0.001).astype(np.float64)
        x = x / x.sum()
    return inits

class ScoreTracker:
    def __init__(self):
        self.best = None
        self.best_score = float('inf')
        self.t_start = time.monotonic()

    def update(self, a):
        a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
        if a.size == 0 or a.sum() < 0.01:
            return
        score = float(evaluate_sequence(a.tolist()))
        if np.isfinite(score) and score < self.best_score:
            self.best = a.copy()
            self.best_score = score
            elapsed = time.monotonic() - self.t_start
            print(f"  * J={score:.8f} n={a.size} t={elapsed:.1f}s", flush=True)

    def report(self, label=""):
        if self.best is not None:
            print(f"  {label} best: J={self.best_score:.8f} n={self.best.size}", flush=True)

def spectral_descent(a, max_iters, deadline, tracker, alpha_start=0.3, alpha_end=0.05):
    """Iterative spectral shaping with greedy acceptance."""
    a = a.copy().astype(np.float64)
    a = a / a.sum()
    current_J = compute_J(a)
    tracker.update(a)
    n = a.size

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        progress = it / max(1, max_iters)
        alpha = alpha_start + (alpha_end - alpha_start) * progress
        a_new = spectral_shape(a, alpha, edge_decay=0.15)
        J_new = compute_J(a_new)
        if J_new < current_J - 1e-12:
            a = a_new
            current_J = J_new
        else:
            # Try smaller alpha
            a_new2 = spectral_shape(a, alpha * 0.3, edge_decay=0.1)
            J_new2 = compute_J(a_new2)
            if J_new2 < current_J - 1e-12:
                a = a_new2
                current_J = J_new2
            elif it > max_iters * 0.8:
                break
        if it % 10 == 0:
            tracker.update(a)
    tracker.update(a)
    return a

def coordinate_polish(a, deadline, tracker, max_steps=2000):
    """Random coordinate scaling for final polish."""
    a = a.copy().astype(np.float64)
    a = a / a.sum()
    n = len(a)
    current_J = compute_J(a)
    tracker.update(a)
    rng = np.random.default_rng(99)

    for step in range(max_steps):
        if time.monotonic() > deadline:
            break
        j = rng.integers(0, n)
        improved = False
        for scale in [0.7, 1.3, 0.5, 1.5, 0.85, 1.15]:
            a_trial = a.copy()
            a_trial[j] *= scale
            a_trial = np.maximum(a_trial, 1e-8)
            ss = a_trial.sum()
            if ss < 1e-10:
                continue
            a_trial = a_trial / ss
            J_trial = compute_J(a_trial)
            if J_trial < current_J - 1e-13:
                a = a_trial
                current_J = J_trial
                improved = True
                break
        if step % 200 == 0:
            tracker.update(a)
    tracker.update(a)
    return a

def propose_candidate(seed=42, budget_s=1500, **kwargs):
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + float(budget_s) - 15.0
    tracker = ScoreTracker()

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        if p.ndim == 1 and p.size > 0 and np.all(np.isfinite(p)):
            parent = np.maximum(p, 0.0)
            tracker.update(parent)
            print(f"Parent: n={parent.size}, J={tracker.best_score:.8f}", flush=True)
    except (NameError, AttributeError):
        pass

    if parent is None:
        parent = np.ones(512) / 512
        tracker.update(parent)
        print(f"No parent, using uniform n=512", flush=True)

    # === Phase 1: Spectral descent from parent ===
    n = parent.size
    print(f"\n{'='*50} Phase1: Spectral descent n={n} {'='*50}", flush=True)
    t0 = time.monotonic()
    a1 = spectral_descent(parent, 50, deadline - 800, tracker)
    J1 = compute_J(a1)
    print(f"  Spectral descent: J={J1:.8f} ({time.monotonic()-t0:.1f}s)", flush=True)

    # === Phase 2: Extended LP from best of spectral ===
    a_best = a1 if J1 < compute_J(parent) else parent.copy()
    print(f"\n{'='*50} Phase2: Extended LP n={n} {'='*50}", flush=True)
    t0 = time.monotonic()
    a2 = run_lp_extended(a_best, 300, deadline - 500, tracker, tighten=0.005)
    J2 = compute_J(a2)
    print(f"  LP: J={J2:.8f} ({time.monotonic()-t0:.1f}s)", flush=True)

    # === Phase 3: Nesterov LSE polish ===
    print(f"\n{'='*50} Phase3: Nesterov LSE n={n} {'='*50}", flush=True)
    t0 = time.monotonic()
    a3 = nesterov_lse(a2, 5000, deadline - 300, tracker)
    J3 = compute_J(a3)
    print(f"  Nesterov: J={J3:.8f} ({time.monotonic()-t0:.1f}s)", flush=True)

    # === Phase 4: Multi-init exploration ===
    n_explore = tracker.best.size if tracker.best is not None else n
    print(f"\n{'='*50} Phase4: Multi-init n={n_explore} {'='*50}", flush=True)
    t0 = time.monotonic()
    inits = generate_inits(n_explore, rng, tracker.best)
    n_inits = len(inits)
    remaining = deadline - time.monotonic() - 200
    per_init = max(8.0, remaining / max(1, n_inits))

    for idx, a0 in enumerate(inits):
        if time.monotonic() > deadline - 200:
            break
        # Quick spectral + LP
        a_trial = spectral_shape(a0, 0.15, edge_decay=0.1)
        a_trial = run_lp_extended(a_trial, min(40, int(per_init / 0.04)), deadline - 200, tracker, tighten=0.003)
        if time.monotonic() < deadline - 200:
            a_trial = nesterov_lse(a_trial, min(400, int(per_init)), deadline - 200, tracker)
        if (idx + 1) % 8 == 0:
            tracker.report(f"Phase4 [{idx+1}/{n_inits}]")

    print(f"  Multi-init done ({time.monotonic()-t0:.1f}s)", flush=True)
    tracker.report("Phase4")

    # === Phase 5: Focused deep LP on current best ===
    if tracker.best is not None and time.monotonic() < deadline - 80:
        print(f"\n{'='*50} Phase5: Deep LP on best {'='*50}", flush=True)
        t0 = time.monotonic()
        a5 = tracker.best.copy().astype(np.float64)
        a5 = a5 / a5.sum()
        # Multiple LP passes
        for tight in [0.0, 0.008, 0.015, 0.0, 0.008]:
            if time.monotonic() > deadline - 80:
                break
            a5 = run_lp_extended(a5, 60, deadline - 80, tracker, tighten=tight)
        # Nesterov
        if time.monotonic() < deadline - 40:
            a5 = nesterov_lse(a5, 2000, deadline - 40, tracker)
        print(f"  Deep LP done ({time.monotonic()-t0:.1f}s)", flush=True)
        tracker.report("Phase5")

    # === Phase 6: Frequency annealing ===
    if tracker.best is not None and time.monotonic() < deadline - 30:
        print(f"\n{'='*50} Phase6: Freq Annealing {'='*50}", flush=True)
        t0 = time.monotonic()
        a6 = freq_annealing(tracker.best, deadline - 30, tracker, n_iter=600)
        if time.monotonic() < deadline - 10:
            a6 = run_lp_extended(a6, 25, deadline - 10, tracker, tighten=0.003)
        print(f"  Freq annealing done ({time.monotonic()-t0:.1f}s)", flush=True)
        tracker.report("Phase6")

    # === Phase 7: Coordinate polish ===
    if tracker.best is not None and time.monotonic() < deadline - 5:
        print(f"\n{'='*50} Phase7: Coordinate polish {'='*50}", flush=True)
        t0 = time.monotonic()
        a7 = coordinate_polish(tracker.best, deadline - 5, tracker, max_steps=3000)
        if time.monotonic() < deadline - 2:
            a7 = run_lp_extended(a7, 15, deadline - 2, tracker, tighten=0.002)
        print(f"  Coord polish done ({time.monotonic()-t0:.1f}s)", flush=True)
        tracker.report("Phase7")

    # === Phase 8: Alternative lengths ===
    for alt_n in [384, 640]:
        if time.monotonic() > deadline - 30:
            break
        if tracker.best is not None and alt_n == tracker.best.size:
            continue
        print(f"\n{'='*50} Phase8: Try n={alt_n} {'='*50}", flush=True)
        t0 = time.monotonic()
        a0 = resample_seq(tracker.best, alt_n) if tracker.best is not None else np.ones(alt_n) / alt_n
        a0 = spectral_descent(a0, 20, deadline - 30, tracker)
        a0 = run_lp_extended(a0, 30, deadline - 30, tracker, tighten=0.003)
        if time.monotonic() < deadline - 10:
            a0 = nesterov_lse(a0, 500, deadline - 10, tracker)
        print(f"  n={alt_n} done ({time.monotonic()-t0:.1f}s)", flush=True)
        tracker.report(f"Alt n={alt_n}")

    # Fallback
    if tracker.best is None:
        tracker.best = np.ones(256)
        tracker.best_score = compute_J(tracker.best)
        tracker.update(tracker.best)

    print(f"\n{'='*50} FINAL {'='*50}", flush=True)
    print(f"J={tracker.best_score:.8f} n={tracker.best.size} time={time.monotonic()-tracker.t_start:.1f}s", flush=True)
    return [float(x) for x in tracker.best]
