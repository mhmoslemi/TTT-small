    S = a.sum()
    n = len(a)
    conv = fast_conv(a)
    current_J = 2.0 * n * conv.max() / (S*S)
    tracker.update(a)
    
    for it in range(n_iters):
        if time.monotonic() > deadline:
            break

        order = np.random.permutation(n)
        
        for i in order:
            old = a[i]
            
            # Affected lag range
            k_lo = i
            k_hi = min(i + n - 1, 2*n - 2)

            if k_lo > 0 or k_hi < 2*n - 2:

                mask[k_lo:k_hi+1] = False

            offsets = k_range - i
            coeffs = 2.0 * a[offsets]
            base_vals = conv[k_lo:k_hi+1].copy()
            
            # Handle k=2i correction
            idx_2i = 2*i - k_lo
            has_2i = (2*i >= k_lo) and (2*i <= k_hi)
            
            def eval_x(x):
                new_S = S - old + x
                if new_S < 1e-15:
                    return 1e10
                new_vals = base_vals + 2.0*(x - old)*coeffs
                if has_2i:
                    new_vals[idx_2i] = (conv[2*i] - old*old + x*x)
                max_b = max(U, new_vals.max())
                return 2.0 * n * max_b / (new_S * new_S)
            
            # Golden section
            gr = (np.sqrt(5.0) - 1.0) / 2.0
            lo = 0.0
            hi = max(3.0*old, 3.0*S/n, 0.01)
            
            c = hi - gr*(hi - lo)
            d = lo + gr*(hi - lo)
            fc = eval_x(c)
            fd = eval_x(d)
            
            for _ in range(40):
                if fc < fd:
                    hi = d
                    d = c
                    fd = fc
                    c = hi - gr*(hi - lo)
                    fc = eval_x(c)
                else:
                    lo = c
                    c = d
                    fc = fd
                    d = lo + gr*(hi - lo)
                    fd = eval_x(d)
            
            x_opt = max(0.0, (lo+hi)/2.0)
            
            if abs(x_opt - old) > 1e-12:
                a[i] = x_opt
                S = S - old + x_opt
                # Update conv incrementally
                for k in range(k_lo, k_hi+1):
                    off = k - i
                    if k == 2*i:
                        conv[k] = conv[k] - old*old + x_opt*x_opt
                    else:
                        conv[k] = conv[k] + 2.0*(x_opt - old)*a[off]
        
        # Recompute conv to avoid drift
        conv = fast_conv(a)
        new_J = 2.0 * n * conv.max() / (S*S)
        if new_J < current_J - 1e-14:
            current_J = new_J
            tracker.update(a)
    
    return a, current_J

The subgradient of J(a) at a (ignoring the max, using the active set):
- Let K* be the set of lags where conv[k] is near max
- grad_max_b[i] = sum_{k in K*} (a[k-i] + a[i-k]) / |K*| (with appropriate bounds)
- grad_J = 2*n*grad_max_b/S^2 - 4*n*max_b*a/S^3
- Projected gradient step: a_new = max(0, a - eta * grad_J)

With Nesterov momentum:
- z = a + (t-1)/(t+2) * (a - a_prev)
- a_new = max(0, z - eta * grad_J(z))
- a_prev = a

"""
Minimizes J(a) = 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Fast vectorized coordinate descent with golden section search
2. LP-based best response (mirror descent / game theory)
3. Nesterov-accelerated projected subgradient descent
4. Multi-length and multi-init exploration with adaptive allocation
Key: vectorized numpy operations for O(n^2) per CD pass (fast in practice).
"""

import numpy as np
from scipy.optimize import linprog
import time

def fast_conv(a):
    """Full self-convolution via FFT."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    A = np.fft.rfft(a, size)
    return np.fft.irfft(A * A, size)[:2 * n - 1]

def compute_J(a):
    """Compute J(a) = 2*n*max(conv(a,a))/(sum(a))^2."""
    a = np.maximum(a, 0.0)
    S = a.sum()
    if S < 0.01:
        return float('inf')
    n = len(a)
    b = fast_conv(a)
    return float(2.0 * n * b.max() / (S * S))

def cd_pass_vectorized(a, n_golden=30):
    """One full pass of vectorized coordinate descent.
    Returns updated a and new J."""
    a = np.maximum(a.copy(), 0.0)
    S = a.sum()
    n = len(a)
    if S < 1e-15:
        return a, float('inf')
    
    conv = fast_conv(a)
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    
    for i in range(n):
        old = a[i]
        
        # Affected lag range
        k_lo = i
        k_hi = min(i + n - 1, 2 * n - 2)
        
        # Unaffected max (precompute once per pass would be better, but this is fine)
        if k_lo > 0 or k_hi < 2 * n - 2:
            U = np.maximum(
                conv[:k_lo].max() if k_lo > 0 else 0.0,
                conv[k_hi+1:].max() if k_hi < 2*n-2 else 0.0
            )
        else:
            U = 0.0
        
        # Precompute affected data
        offsets = np.arange(k_lo - i, k_hi - i + 1)  # = arange(0, k_hi-k_lo+1)
        coeffs = 2.0 * a[offsets]
        base_vals = conv[k_lo:k_hi+1].copy()
        
        idx_2i = (2 * i) - k_lo
        has_2i = (0 <= idx_2i < (k_hi - k_lo + 1))
        
        # Golden section search
        hi = max(3.0 * old, 3.0 * S / n, 0.01)
        if hi < 1e-10:
            hi = 1.0
        
        lo = 0.0
        c = hi - gr * (hi - lo)
        d = lo + gr * (hi - lo)
        
        # Inline evaluation for speed
        new_S_c = S - old + c
        new_vals_c = base_vals + 2.0 * (c - old) * coeffs
        if has_2i:
            new_vals_c[idx_2i] = conv[2*i] - old*old + c*c
        fc = 2.0 * n * max(U, new_vals_c.max()) / (new_S_c * new_S_c)
        
        new_S_d = S - old + d
        new_vals_d = base_vals + 2.0 * (d - old) * coeffs
        if has_2i:
            new_vals_d[idx_2i] = conv[2*i] - old*old + d*d
        fd = 2.0 * n * max(U, new_vals_d.max()) / (new_S_d * new_S_d)
        
        for _ in range(n_golden):
            if fc < fd:
                hi = d
                d = c
                fd = fc
                c = hi - gr * (hi - lo)
                new_S_c = S - old + c
                new_vals_c = base_vals + 2.0 * (c - old) * coeffs
                if has_2i:
                    new_vals_c[idx_2i] = conv[2*i] - old*old + c*c
                fc = 2.0 * n * max(U, new_vals_c.max()) / (new_S_c * new_S_c)
            else:
                lo = c
                c = d
                fc = fd
                d = lo + gr * (hi - lo)
                new_S_d = S - old + d
                new_vals_d = base_vals + 2.0 * (d - old) * coeffs
                if has_2i:
                    new_vals_d[idx_2i] = conv[2*i] - old*old + d*d
                fd = 2.0 * n * max(U, new_vals_d.max()) / (new_S_d * new_S_d)
        
        x_opt = max(0.0, (lo + hi) / 2.0)
        
        if abs(x_opt - old) > 1e-12:
            # Update conv incrementally (vectorized)
            conv[k_lo:k_hi+1] += 2.0 * (x_opt - old) * a[offsets]
            if has_2i:
                # Correct the k=2i entry
                conv[2*i] = conv[2*i] - 2.0 * (x_opt - old) * old + (x_opt*x_opt - old*old)
            a[i] = x_opt
            S = S - old + x_opt
    
    # Verify
    conv_check = fast_conv(a)
    J = 2.0 * n * conv_check.max() / (S * S)
    return a, J

def lp_best_response(a, M):
    """Solve LP: max sum(g) s.t. (a*g)_k <= M for all k, g >= 0."""
    n = len(a)
    c = -np.ones(n)
    
    # Build sparse constraint matrix
    from scipy.sparse import lil_matrix
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        jl = max(0, k - n + 1)
        jh = min(n - 1, k)
        for j in range(jl, jh + 1):
            rows.append(k)
            cols.append(j)
            vals.append(a[k - j])
    
    A = lil_matrix((2 * n - 1, n))
    for idx in range(len(rows)):
        A[rows[idx], cols[idx]] = vals[idx]
    A = A.tocsr()
    
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
    if r.success and r.x is not None:
        g = np.maximum(r.x, 0.0)
        if g.sum() > 1e-10:
            return g
    return None

def nesterov_subgradient(a, n_iters, deadline, tracker):
    """Nesterov-accelerated projected subgradient descent."""
    a = np.maximum(a.copy(), 0.0)
    S = a.sum()
    n = len(a)
    
    # Normalize
    a = a / S
    
    a_prev = a.copy()
    t = 1.0
    eta0 = 1.0 / np.sqrt(2.0 * n)
    best_J = compute_J(a)
    tracker.update(a)
    
    for it in range(n_iters):
        if time.monotonic() > deadline:
            break
        
        # Nesterov momentum
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t * t)) / 2.0
        z = a + (t - 1.0) / t_new * (a - a_prev)
        z = np.maximum(z, 0.0)
        t = t_new
        
        # Compute subgradient at z
        conv_z = fast_conv(z)
        max_b = conv_z.max()
        tol = 1e-4 * max_b
        active = np.where(conv_z >= max_b - tol)[0]
        
        # Subgradient of max(conv) w.r.t. z
        grad = np.zeros(n)
        for k in active:
            jl = max(0, k - n + 1)
            jh = min(n - 1, k)
            for j in range(jl, jh + 1):
                grad[j] += z[k - j]
        grad /= len(active)
        
        # Full gradient of J (with S=1 normalization)
        # J = 2n * max_b / S^2, dJ/da = 2n * grad_max_b / S^2 - 4n * max_b / S^3 * 1
        S_z = z.sum()
        grad_J = 2.0 * n * grad / (S_z * S_z) - 4.0 * n * max_b / (S_z**3)
        
        # Step
        eta = eta0 / t
        a_new = z - eta * grad_J
        a_new = np.maximum(a_new, 0.0)
        
        a_prev = a
        a = a_new
        
        if a.sum() < 1e-10:
            a = np.ones(n) / n
            continue
        
        # Periodically evaluate
        if it % 10 == 0:
            J = compute_J(a)
            if J < best_J:
                best_J = J
                tracker.update(a)
    
    tracker.update(a)
    return a

def resample_seq(src, dest_n):
    """Resample 1D sequence to new length."""
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)

"""
Minimizes J(a) = 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Fast vectorized coordinate descent with golden section search
2. LP-based best response (mirror descent / game theory)
3. Simulated annealing for escaping local minima
4. Nesterov-accelerated projected subgradient
5. Multi-length exploration and parametric search
Key: fully vectorized numpy ops for O(n) per golden-section eval.

import numpy as np
from scipy.optimize import linprog, minimize
from scipy.sparse import lil_matrix
import time


def fast_conv(a):
    """Full self-convolution via FFT."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    A = np.fft.rfft(a, size)
    return np.fft.irfft(A * A, size), size


def compute_J(a):
    """Compute J(a) = 2*n*max(conv(a,a))/(sum(a))^2."""
    a = np.maximum(a, 0.0)
    S = a.sum()
    if S < 0.01:
        return float('inf')
    n = len(a)
    b, _ = fast_conv(a)
    return float(2.0 * n * b.max() / (S * S))


def cd_single_pass(a, n_golden=30, random_order=True):
    """One full pass of vectorized coordinate descent.
    Returns updated a and new J."""
    a = np.maximum(a.copy(), 0.0)
    S = a.sum()
    n = len(a)
    if S < 1e-15:
        return a, float('inf')

    conv = fast_conv(a)[0]
    gr = (np.sqrt(5.0) - 1.0) / 2.0

    order = np.random.permutation(n) if random_order else np.arange(n)

    for i in order:
        old = a[i]

        k_lo = i
        k_hi = min(i + n - 1, 2 * n - 2)
        k_len = k_hi - k_lo + 1

        # Unaffected max
        if k_lo > 0 or k_hi < 2 * n - 2:
            U1 = conv[:k_lo].max() if k_lo > 0 else 0.0
            U2 = conv[k_hi + 1:].max() if k_hi < 2 * n - 2 else 0.0
            U = max(U1, U2)
        else:
            U = 0.0

        # Precompute affected data
        offsets = np.arange(0, k_len)
        coeffs = 2.0 * a[offsets]
        base_vals = conv[k_lo:k_hi + 1].copy()

        idx_2i = (2 * i) - k_lo
        has_2i = (0 <= idx_2i < k_len)
        conv_2i = conv[2 * i] if has_2i else 0.0

        # Golden section search
        hi = max(3.0 * old, 3.0 * S / n, 0.001)
        lo = 0.0

        c_pt = hi - gr * (hi - lo)
        d_pt = lo + gr * (hi - lo)

        # Evaluate at c
        ns = S - old + c_pt
        nv = base_vals + 2.0 * (c_pt - old) * coeffs
        if has_2i:
            nv[idx_2i] = conv_2i - old * old + c_pt * c_pt
        fc = 2.0 * n * max(U, nv.max()) / (ns * ns)

        # Evaluate at d
        ns = S - old + d_pt
        nv = base_vals + 2.0 * (d_pt - old) * coeffs
        if has_2i:
            nv[idx_2i] = conv_2i - old * old + d_pt * d_pt
        fd = 2.0 * n * max(U, nv.max()) / (ns * ns)

        for _ in range(n_golden):
            if fc < fd:
                hi = d_pt
                d_pt = c_pt
                fd = fc
                c_pt = hi - gr * (hi - lo)
                ns = S - old + c_pt
                nv = base_vals + 2.0 * (c_pt - old) * coeffs
                if has_2i:
                    nv[idx_2i] = conv_2i - old * old + c_pt * c_pt
                fc = 2.0 * n * max(U, nv.max()) / (ns * ns)
            else:
                lo = c_pt
                c_pt = d_pt
                fc = fd
                d_pt = lo + gr * (hi - lo)
                ns = S - old + d_pt
                nv = base_vals + 2.0 * (d_pt - old) * coeffs
                if has_2i:
                    nv[idx_2i] = conv_2i - old * old + d_pt * d_pt
                fd = 2.0 * n * max(U, nv.max()) / (ns * ns)

        x_opt = max(0.0, (lo + hi) / 2.0)

        if abs(x_opt - old) > 1e-14:
            # Update conv incrementally
            conv[k_lo:k_hi + 1] += 2.0 * (x_opt - old) * a[offsets]
            if has_2i:
                conv[2 * i] = conv_2i - old * old + x_opt * x_opt
            a[i] = x_opt
            S = S - old + x_opt

    # Verify with full conv
    conv_check = fast_conv(a)[0]
    J = 2.0 * n * conv_check.max() / (S * S)
    return a, J


def lp_best_response(a, M):
    """Solve LP: max sum(g) s.t. (a*g)_k <= M for all k, g >= 0."""
    n = len(a)
    c = -np.ones(n)

    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        jl = max(0, k - n + 1)
        jh = min(n - 1, k)
        for j in range(jl, jh + 1):
            rows.append(k)
            cols.append(j)
            vals.append(a[k - j])

    A = lil_matrix((2 * n - 1, n))
    for idx in range(len(rows)):
        A[rows[idx], cols[idx]] = vals[idx]
    A = A.tocsr()

    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
    if r.success and r.x is not None:
        g = np.maximum(r.x, 0.0)
        if g.sum() > 1e-10:
            return g
    return None


def run_lp_phase(a, deadline, tracker, max_iters=100):
    """LP-based mirror descent."""
    a = np.maximum(a.copy(), 0.0)
    a = a / a.sum()
    n = len(a)
    current_J = compute_J(a)
    tracker.update(a)

    tighten = 0.004
    stall = 0

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        conv = fast_conv(a)[0]
        M = float(conv.max())
        M_target = M * (1.0 - tighten)

        try:
            g = lp_best_response(a, M_target)
        except Exception:
            g = None

        if g is None:
            M_target = M * (1.0 - tighten * 0.5)
            try:
                g = lp_best_response(a, M_target)
            except Exception:
                g = None

        if g is None:
            tighten *= 0.5
            if tighten < 1e-7:
                break
            stall += 1
            if stall > 15:
                tighten = 0.004
                stall = 0
            continue

        g = g / g.sum()
        best_a = a
        best_J = current_J

        for t in [0.1, 0.3, 0.5, 0.7, 0.9]:
            a_trial = (1 - t) * a + t * g
            a_trial = np.maximum(a_trial, 0.0)
            s = a_trial.sum()
            if s < 1e-10:
                continue
            a_trial = a_trial / s
            J_trial = compute_J(a_trial)
            if J_trial < best_J - 1e-14:
                best_J = J_trial
                best_a = a_trial

        if best_a is a:
            stall += 1
            if stall > 20:
                break
            tighten *= 0.8
        else:
            a = best_a
            current_J = best_J
            stall = 0
            tighten = min(0.01, tighten * 1.05)
            tracker.update(a)

        if it % 20 == 0:
            print(f"    LP iter {it}: J={current_J:.8f}", flush=True)

    tracker.update(a)
    return a


def simulated_annealing(a, deadline, tracker, n_trials=2000):
    """Simulated annealing: perturb random coords, accept/reject."""
    a = np.maximum(a.copy(), 0.0)
    a = a / a.sum()
    n = len(a)
    conv = fast_conv(a)[0]
    current_J = compute_J(a)
    tracker.update(a)

    T_start = 0.01
    T_end = 0.0001
    n_perturb = max(1, n // 20)

    for trial in range(n_trials):
        if time.monotonic() > deadline:
            break

        progress = trial / max(1, n_trials)
        T = T_start * (T_end / T_start) ** progress

        # Perturb random coordinates
        a_new = a.copy()
        idxs = np.random.choice(n, n_perturb, replace=False)
        for idx in idxs:
            scale = np.exp(np.random.normal(0, 0.3))
            a_new[idx] = max(0.0, a_new[idx] * scale)

        if a_new.sum() < 1e-10:
            continue

        J_new = compute_J(a_new)
        delta = J_new - current_J

        if delta < 0 or np.random.random() < np.exp(-delta / T):
            a = a_new
            current_J = J_new
            tracker.update(a)

        if trial % 200 == 0:
            print(f"    SA trial {trial}: J={current_J:.8f}, T={T:.6f}", flush=True)

    tracker.update(a)
    return a


def nesterov_subgrad(a, deadline, tracker, max_iters=200):
    """Nesterov-accelerated projected subgradient descent."""
    a = np.maximum(a.copy(), 0.0)
    a = a / a.sum()
    n = len(a)

    a_prev = a.copy()
    t = 1.0
    eta0 = 2.0 / np.sqrt(n)
    best_J = compute_J(a)
    tracker.update(a)

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t * t)) / 2.0
        momentum = (t - 1.0) / t_new
        z = a + momentum * (a - a_prev)
        z = np.maximum(z, 0.0)
        z = z / z.sum()
        t = t_new

        conv_z = fast_conv(z)[0]
        max_b = float(conv_z.max())
        tol = 1e-3 * max_b
        active = np.where(conv_z >= max_b - tol)[0]

        # Subgradient of max(conv) w.r.t. z
        grad = np.zeros(n)
        for k in active:
            jl = max(0, k - n + 1)
            jh = min(n - 1, k)
            for j in range(jl, jh + 1):
                grad[j] += z[k - j]
        grad /= len(active)

        S_z = z.sum()
        grad_J = 2.0 * n * grad / (S_z * S_z) - 4.0 * n * max_b / (S_z ** 3)

        eta = eta0 / t
        a_new = z - eta * grad_J
        a_new = np.maximum(a_new, 0.0)
        if a_new.sum() < 1e-10:
            a_new = np.ones(n) / n
        a_new = a_new / a_new.sum()

        a_prev = a
        a = a_new

        if it % 10 == 0:
            J = compute_J(a)
            if J < best_J:
                best_J = J
            tracker.update(a)

    tracker.update(a)
    return a

def resample_seq(src, dest_n):
    """Resample 1D sequence to new length via linear interpolation."""
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)

        x0 = np.random.normal(0, 0.05, n_harmonics)

        def obj(params, t=t, n=n, n_harmonics=n_harmonics):
            log_a = np.zeros(n)
            for k in range(1, n_harmonics + 1):
                log_a += params[k - 1] * np.cos(2.0 * np.pi * k * t)
            a = np.exp(log_a)
            a = np.maximum(a, 0.0)
            s = a.sum()
            if s < 1e-10:
                return 1e6
            a = a / s
            return compute_J(a)

        try:
            res = minimize(obj, x0, method='Nelder-Mead',
                           options={'maxiter': 500, 'xatol': 1e-8, 'fatol': 1e-10})
            if res.fun < best_val:
                best_val = res.fun
                best_params = res.x.copy()
        except Exception:
            pass

    if best_params is not None:
        log_a = np.zeros(n)
        for k in range(1, n_harmonics + 1):
            log_a += best_params[k - 1] * np.cos(2.0 * np.pi * k * t)
        a = np.exp(log_a)
        a = np.maximum(a, 0.0)
        a = a / a.sum()

        a, J = cd_single_pass(a, n_golden=30)
        n_passes += 1
        if n_passes % 5 == 0:
            tracker.update(a)
            print(f"  CD pass {n_passes}: J={J:.8f} ({time.monotonic() - t0:.1f}s)", flush=True)

    tracker.report("Phase 1 CD")
    print(f"  Total passes: {n_passes} in {time.monotonic() - t0:.1f}s", flush=True)

    # === Phase 2: LP best response ===
    if time.monotonic() < deadline - 500:
        print(f"\n{'='*50}", flush=True)
        print(f"Phase 2: LP best response", flush=True)
        print(f"{'='*50}", flush=True)
        t0 = time.monotonic()
        a_lp = run_lp_phase(tracker.best.copy(), deadline - 500, tracker, max_iters=80)
        tracker.report("Phase 2 LP")
        print(f"  Time: {time.monotonic() - t0:.1f}s", flush=True)

    # === Phase 3: More CD on best ===
    if time.monotonic() < deadline - 400:
        print(f"\n{'='*50}", flush=True)
        print(f"Phase 3: More CD", flush=True)
        print(f"{'='*50}", flush=True)
        t0 = time.monotonic()
        a = tracker.best.copy()
        a = a / a.sum()
        n_passes = 0
        while time.monotonic() < deadline - 400:
            a, J = cd_single_pass(a, n_golden=30)
            n_passes += 1
            if n_passes % 5 == 0:
                tracker.update(a)
        tracker.update(a)
        tracker.report("Phase 3 CD")
        print(f"  Passes: {n_passes} in {time.monotonic() - t0:.1f}s", flush=True)

    # === Phase 4: Simulated annealing ===
    if time.monotonic() < deadline - 250:
        print(f"\n{'='*50}", flush=True)
        print(f"Phase 4: Simulated Annealing", flush=True)
        print(f"{'='*50}", flush=True)
        t0 = time.monotonic()
        n_sa = max(500, int((deadline - 250 - time.monotonic()) / 0.01))
        a = simulated_annealing(tracker.best, deadline - 250, tracker, n_trials=n_sa)
        tracker.report("Phase 4 SA")
        print(f"  Time: {time.monotonic() - t0:.1f}s", flush=True)

    # === Phase 5: Nesterov subgradient ===
    if time.monotonic() < deadline - 200:
        print(f"\n{'='*50}", flush=True)
        print(f"Phase 5: Nesterov Subgradient", flush=True)
        print(f"{'='*50}", flush=True)
        t0 = time.monotonic()
        n_it = max(50, int((deadline - 200 - time.monotonic()) / 0.05))
        a = nesterov_subgrad(tracker.best, deadline - 200, tracker, max_iters=n_it)
        tracker.report("Phase 5 Nesterov")
        print(f"  Time: {time.monotonic() - t0:.1f}s", flush=True)

    # === Phase 6: Multi-length exploration ===
    if time.monotonic() < deadline - 150:
        print(f"\n{'='*50}", flush=True)
        print(f"Phase 6: Multi-length", flush=True)
        print(f"{'='*50}", flush=True)
        t0 = time.monotonic()

        lengths_to_try = []
        for delta in [-8, -4, -2, 2, 4, 6, 8, 12, 16]:
            nl = n_parent + delta
            if 150 <= nl <= 1000:
                lengths_to_try.append(nl)
        # Also try some non-local lengths
        for factor in [0.8, 0.9, 1.1, 1.2]:
            nl = int(n_parent * factor)
            if 150 <= nl <= 1000 and nl not in lengths_to_try:
                lengths_to_try.append(nl)

        for nl in lengths_to_try:
            if time.monotonic() > deadline - 150:
                break

            a_res = resample_seq(tracker.best, nl)
            a_res = np.maximum(a_res, 1e-6)
            a_res = a_res / a_res.sum()

            # Quick CD
            n_cd = max(2, int(min(10, (deadline - 150 - time.monotonic()) / 0.2)))
            for _ in range(n_cd):
                if time.monotonic() > deadline - 150:
                    break
                a_res, _ = cd_single_pass(a_res, n_golden=20)

            tracker.update(a_res)
            print(f"  n={nl}: J={tracker.best_score:.8f}", flush=True)

        tracker.report("Phase 6 Multi-length")
        print(f"  Time: {time.monotonic() - t0:.1f}s", flush=True)

    # === Phase 7: Parametric search ===
    if time.monotonic() < deadline - 100:
        print(f"\n{'='*50}", flush=True)
        print(f"Phase 7: Parametric search", flush=True)
        print(f"{'='*50}", flush=True)
        t0 = time.monotonic()
        parametric_search(tracker, deadline - 100, n_parent, n_harmonics=6)
        # Polish with CD
        if time.monotonic() < deadline - 100:
            a = tracker.best.copy()
            a = a / a.sum()
            for _ in range(3):
                if time.monotonic() > deadline - 100:
                    break
                a, _ = cd_single_pass(a, n_golden=25)
            tracker.update(a)
        tracker.report("Phase 7 Parametric")
        print(f"  Time: {time.monotonic() - t0:.1f}s", flush=True)

    # === Phase 8: Final CD polish ===
    if time.monotonic() < deadline - 10:
        print(f"\n{'='*50}", flush=True)
        print(f"Phase 8: Final CD polish", flush=True)
        print(f"{'='*50}", flush=True)
        t0 = time.monotonic()
        a = tracker.best.copy()
        a = a / a.sum()
        while time.monotonic() < deadline - 10:
            a, J = cd_single_pass(a, n_golden=35)
            tracker.update(a)
        tracker.report("Phase 8 Final")
        print(f"  Time: {time.monotonic() - t0:.1f}s", flush=True)

    # Fallback
    if tracker.best is None:
        tracker.best = np.ones(256)
        tracker.best_score = compute_J(tracker.best)

    print(f"\n{'='*50}", flush=True)
    print(f"FINAL RESULT", flush=True)
    print(f"{'='*50}", flush=True)
    print(f"J = {tracker.best_score:.10f}, n = {tracker.best.size}", flush=True)
    print(f"Total time: {time.monotonic() - tracker.t_start:.1f}s", flush=True)

    return [float(x) for x in tracker.best]
