import numpy as np
from scipy.optimize import linprog
from scipy.sparse import lil_matrix
import time


def fast_conv(a):
    """Full self-convolution via FFT."""
    n = len(a)
    s = 1 << (2 * n - 1 - 1).bit_length() if n > 1 else 1
    fa = np.fft.rfft(a, s)
    return np.fft.irfft(fa * fa, s)[:2 * n - 1]


def compute_J(a):
    """Compute J(a) = 2*n*max(conv(a,a))/(sum(a))^2."""
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))


def smoothed_gradient(a, T):

    S = a.sum()

    W = np.fft.rfft(w, N)
    A = np.fft.rfft(a, N)
    g = np.fft.irfft(W * np.conj(A), N)[:n]
    
    # Gradient: dJ/da_i = 4*n/S^2 * (T*g_i - L/S)
    grad = 4.0 * n / (S * S) * (T * g - L / S)
    
    # Smoothed objective value
    J_smooth = 2.0 * n * L / (S * S)
    
    return grad, J_smooth, b, w


def build_A_sparse(a):
    """Build sparse constraint matrix for LP: (a*g)_k <= M for all k."""
    n = len(a)
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        jl = max(0, k - n + 1)
        jh = min(n - 1, k)
        for j in range(jl, jh + 1):
            rows.append(k)
            cols.append(j)
            vals.append(a[k - j])
    A = lil_matrix((2 * n - 1, n))
    for idx in range(len(rows)):
        A[rows[idx], cols[idx]] = vals[idx]
    return A.tocsr()


def lp_best_response(a, M):
    """LP: max sum(g) s.t. (a*g)_k <= M for all k, g >= 0."""
    n = len(a)
    c = -np.ones(n)
    A = build_A_sparse(a)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    try:
        r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                    options={'presolve': True, 'time_limit': 30})
        if r.success and r.x is not None:
            g = np.maximum(r.x, 0.0)
            if g.sum() > 1e-10:
                return g
    except Exception:
        pass
    return None


class AdamOptimizer:
    def __init__(self, params, lr=0.001, beta1=0.9, beta2=0.999, eps=1e-8):
        self.params = params.copy()
        self.lr = lr
        self.beta1 = beta1
        self.beta2 = beta2
        self.eps = eps
        self.m = np.zeros_like(self.params)
        self.v = np.zeros_like(self.params)
        self.t = 0
    
    def step(self, grad):
        self.t += 1
        self.m = self.beta1 * self.m + (1 - self.beta1) * grad
        self.v = self.beta2 * self.v + (1 - self.beta2) * (grad * grad)
        m_hat = self.m / (1 - self.beta1 ** self.t)
        v_hat = self.v / (1 - self.beta2 ** self.t)
        self.params = self.params - self.lr * m_hat / (np.sqrt(v_hat) + self.eps)
        self.params = np.maximum(self.params, 0.0)
        return self.params.copy()


def smoothed_gradient_descent(a, n_iters, deadline, tracker, T_start=10.0, T_end=0.1, 
                                 lr_start=0.01, lr_end=0.001, use_adam=True):
    """ smoothed gradient descent with adaptive temperature and learning rate."""
    a = np.maximum(a.astype(np.float64), 0.001)
    a = a / a.sum()
    n = len(a)
    current_J = compute_J(a)
    tracker.update(a)
    
    # Temperature schedule
    T_schedule = np.geomspace(T_start, T_end, n_iters)
    # Learning rate schedule
    lr_schedule = np.linspace(lr_start, lr_end, n_iters)
    
    if use_adam:
        adam = AdamOptimizer(a, lr=lr_start)
    
    best_local_J = current_J
    best_local_a = a.copy()
    patience = 0
    
    for it in range(n_iters):
        if time.monotonic() > deadline:
            break
        
        T = T_schedule[it]
        
        if use_adam:
            adam.lr = lr_schedule[it]
        else:
            lr = lr_schedule[it]
        
        grad, J_smooth, b, w = smoothed_gradient(a, T)
        
        if use_adam:
            a_new = adam.step(grad)
        else:
            a_new = a - lr * grad
            a_new = np.maximum(a_new, 0.0)
        
        # Renormalize periodically
        if a_new.sum() < 0.01:
            a_new = a_new + 0.001
        if it % 10 == 0:
            a_new = a_new / a_new.sum()
        
        # Check actual objective
        if it % 5 == 0:
            J_actual = compute_J(a_new)
            if J_actual < best_local_J - 1e-10:
                best_local_J = J_actual
                best_local_a = a_new.copy()
                patience = 0
                if it % 20 == 0:
                    tracker.update(a_new)
                    print(f"    SGD iter {it}: T={T:.3f}, J_smooth={J_smooth:.8f}, J={J_actual:.8f}", flush=True)
            else:
                patience += 1
        
        a = a_new
        
        # Early stopping if stuck
        if patience > 50:
            a = best_local_a.copy()
            # Try lower learning rate
            if use_adam:
                adam = AdamOptimizer(a, lr=adam.lr * 0.5)
            else:
                lr *= 0.5
            patience = 0
    
    tracker.update(best_local_a)
    return best_local_a


def fast_coordinate_descent_pass(a, max_iters=3):
    """Fast coordinate descent using FFT for convolution update."""
    a = np.maximum(a.astype(np.float64), 0.0)
    S = a.sum()
    if S < 1e-10:
        return a
    n = len(a)
    current_J = compute_J(a)
    
    for it in range(max_iters):
        b = fast_conv(a)
        improved = False
        
        # Process coordinates in random order
        indices = np.random.permutation(n)
        
        for i in indices:
            ai = a[i]
            if ai < 1e-12:
                continue
            
            # Try a range of values for a[i]
            # The optimal value should be near the current value
            # Use golden section search on a small interval
            gr = (np.sqrt(5.0) - 1.0) / 2.0
            
            lo = max(0.0, ai * 0.5)
            hi = ai * 1.5 + 0.01 * S / n
            
            # Evaluate at a few points
            best_val = ai
            best_J = current_J
            
            for frac in [0.5, 0.7, 0.8, 0.9, 1.0, 1.1, 1.2, 1.3, 1.5]:
                trial = ai * frac
                new_S = S - ai + trial
                if new_S < 1e-10:
                    continue
                a[i] = trial
                b_new = fast_conv(a)
                J_trial = 2.0 * n * b_new.max() / (new_S * new_S)
                if J_trial < best_J - 1e-12:
                    best_J = J_trial
                    best_val = trial
                    b = b_new
                    S = new_S
                a[i] = trial
            a[i] = best_val
            S = S - ai + best_val
            if best_val != ai:
                improved = True
                if best_J < current_J:
                    current_J = best_J
        
        if not improved:
            break
    
    return a


def run_lp_mirror(a, max_iters, deadline, tracker, tighten=0.005):
    """LP mirror descent with adaptive tightening."""
    a = np.maximum(a.astype(np.float64), 0.001)
    a = a / a.sum()
    current_J = compute_J(a)
    tracker.update(a)
    
    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        
        b = fast_conv(a)
        M = float(b.max())
        M_target = M * (1.0 - tighten)
        
        g = lp_best_response(a, M_target)
        if g is None:
            M_target = M * (1.0 - tighten * 0.3)
            g = lp_best_response(a, M_target)
        if g is None:
            break
        
        g = np.maximum(g, 0.0)
        g_sum = g.sum()
        if g_sum < 1e-10:
            break
        g = g / g_sum
        
        # Try multiple step sizes
        best_a = a
        best_J = current_J
        for t in [0.3, 0.5, 0.7, 0.1, 0.2]:
            a_trial = (1.0 - t) * a + t * g
            a_trial = np.maximum(a_trial, 0.0)
            if a_trial.sum() < 1e-10:
                continue
            a_trial = a_trial / a_trial.sum()
            J_trial = compute_J(a_trial)
            if J_trial < best_J - 1e-12:
                best_J = J_trial
                best_a = a_trial
        
        if best_a is a:
            break
        
        a = best_a
        current_J = best_J
        tracker.update(a)
        
        if it % 5 == 0:
            print(f"    LP iter {it}: J={current_J:.8f}", flush=True)
    
    return a


def resample_seq(src, dest_n):
    """Resample 1D sequence to new length."""
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)


class ScoreTracker:
    def __init__(self):
        self.best = None
        self.best_score = float('inf')
        self.t_start = time.monotonic()
    
    def update(self, a):
        a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
        if a.size == 0 or a.sum() < 0.01:
            return
        score = float(evaluate_sequence(a.tolist()))
        if np.isfinite(score) and score < self.best_score:
            self.best = a.copy()
            self.best_score = score
            elapsed = time.monotonic() - self.t_start
            print(f"  * NEW BEST: J={score:.8f} n={a.size} t={elapsed:.0f}s", flush=True)
    
    def report(self, label=""):
        if self.best is not None:
            print(f"  {label} best: J={self.best_score:.8f} n={self.best.size} t={time.monotonic()-self.t_start:.0f}s", flush=True)
