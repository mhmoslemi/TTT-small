from scipy.sparse import lil_matrix

def fast_conv(a):
    n = len(a)
    s = 1
    while s < 2*n-1:
        s <<= 1
    fa = np.fft.rfft(a, s)
    return np.fft.irfft(fa*fa, s)[:2*n-1]

def compute_J(a):
    a = np.maximum(a, 0)
    S = a.sum()
    if S < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2*len(a)*b.max()/(S*S))

def project_simplex(v):
    """Project v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if v.sum() < 1e-15:
        return np.ones(n)/n
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n+1) > (cssv - 1))[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0)

def subgradient_direction(a, active_lags):
    """Compute the subgradient of max(b) at a, using active lags."""
    n = len(a)
    grad = np.zeros(n)

        for i in range(n):
            ki = k - i
            if 0 <= ki < n:
                grad[i] += 2.0 * a[ki]
    return grad

"""
Minimizes J(a) = 2*n*max(convolve(a,a))/(sum(a))^2 using:
1. Fast exact subgradient descent with adaptive step sizes
2. Efficient coordinate descent exploiting 1D structure
3. LP best-response for large steps
4. Multi-length exploration
5. Simulated annealing restarts
Key innovation: exploiting the exact gradient formula ∂b[k]/∂a[i] = 2a[k-i]
for O(n) gradient computation, enabling many more iterations than LP-based methods.
"""

import numpy as np
from scipy.optimize import linprog

def fast_conv(a):
    """Full self-convolution via FFT."""
    n = len(a)
    s = 1
    while s < 2 * n - 1:
        s <<= 1
    fa = np.fft.rfft(a, s)
    return np.fft.irfft(fa * fa, s)[:2 * n - 1]

def compute_J(a):
    """Compute J(a) = 2*n*max(conv(a,a))/(sum(a))^2."""
    a = np.maximum(a, 0)
    S = a.sum()
    if S < 0.01:
        return float('inf')
    n = len(a)
    b = fast_conv(a)
    return float(2 * n * b.max() / (S * S))

def project_simplex(v):
    """Project v onto probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    rho = np.nonzero(cond)[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho = rho[-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0)

def subgradient_of_max_b(a, b, active_frac=0.1):
    """Compute subgradient of max(b) w.r.t. a using active lags.
    
    ∂b[k]/∂a[i] = 2*a[k-i] if 0 <= k-i < n, else 0.
    """
    n = len(a)
    b_max = b.max()
    tol = active_frac * b_max
    active = np.where(b >= b_max - tol)[0]
    
    grad = np.zeros(n)
    for k in active:
        # For each i, grad[i] += 2*a[k-i] if 0 <= k-i < n
        i_min = max(0, k - n + 1)
        i_max = min(n - 1, k)
        if i_min <= i_max:
            ki = k - np.arange(i_min, i_max + 1)
            grad[i_min:i_max+1] += 2.0 * a[ki]
    
    return grad, active

def subgradient_step(a, eta):
    """One subgradient step with projection onto simplex."""
    n = len(a)
    b = fast_conv(a)
    b_max = b.max()
    S = a.sum()
    
    grad, active = subgradient_of_max_b(a, b, active_frac=0.15)
    
    # Full subgradient of J (normalized so S=1)
    # J = 2n * max_b / S^2, with S=1: J = 2n * max_b
    # dJ/da = 2n * grad_max_b - 4n*max_b * 1/S^2 * (dS/da) ... 
    # Actually with S=1 constraint, we just minimize max_b
    # So direction is -grad_max_b
    # But we also need to maintain S=1, which the projection handles.
    
    # Actually, let's minimize J directly.
    # J = 2n * max_b / S^2
    # dJ/da[i] = 2n * (d(max_b)/da[i]) / S^2 - 2n * max_b * 2 * (dS/da[i]) / S^3
    # = 2n * grad[i] / S^2 - 4n * max_b / S^3
    # (since dS/da[i] = 1 for all i)
    
    J_grad = 2.0 * n * grad / (S * S) - 4.0 * n * b_max / (S * S * S)
    
    a_new = a - eta * J_grad
    a_new = np.maximum(a_new, 0)
    S_new = a_new.sum()
    if S_new < 1e-15:
        return a, b_max
    a_new = a_new / S_new
    return a_new, b_max

def backtracking_line_search(a, direction, J_current, max_iter=50):
    """Find step size using backtracking line search."""
    eta = 1.0
    for _ in range(max_iter):
        a_trial = a + eta * direction
        a_trial = np.maximum(a_trial, 0)
        S = a_trial.sum()
        if S < 1e-15:
            eta *= 0.5
            continue
        a_trial = a_trial / S
        J_trial = compute_J(a_trial)
        if J_trial < J_current:
            return a_trial, eta
        eta *= 0.5
    return a, 0.0

def coordinate_descent_1d_fast(a, i, b, S):

    Uses the fact that only O(n) b values change."""
    n = len(a)
    old = a[i]

    k_lo = i
    k_hi = min(i + n - 1, 2 * n - 2)
    
    # Max b outside affected range
    if k_lo == 0 and k_hi == 2*n-2:
        b_max_outside = 0.0  # No outside
    else:
        mask = np.ones(2*n-1, dtype=bool)
        mask[k_lo:k_hi+1] = False
        if mask.any():
            b_max_outside = b[mask].max()
        else:
            b_max_outside = 0.0
    
    # For affected indices, compute b[k] as function of x = a[i]
    # b'[k] = b[k] - old*a[k-i] + x*a[k-i] (for k != 2i)
    # b'[2i] = b[2i] - old^2 + x^2 (if 2i in range)
    
    ks = np.arange(k_lo, k_hi + 1)
    kis = ks - i  # these are all in [0, n-1] by construction

    base = b[k_lo:k_hi+1].copy()

        base[idx_2i] = base[idx_2i] - old*old  # remove old^2, will add x^2
        coeffs[idx_2i] = 0.0  # handle separately

            fd = fc
            c_pt = hi - gr * (hi - lo)

            lo = c_pt
            c_pt = d_pt
            fc = fd
            d_pt = lo + gr * (hi - lo)
            fd = eval_J(d_pt)

            v = eval_J(test_x)
            if v < best_val:
                best_val = v
                best_x = test_x
    
    return max(0.0, best_x)

2. Thorough coordinate descent with efficient 1D golden-section search
3. LP best-response for periodic large steps

Key: exact gradient formula partial_b[k]/partial_a[i] = 2*a[k-i] enables
extremely fast gradient steps (O(n) per step vs O(n^2) for finite differences).

import numpy as np
from scipy.optimize import linprog
from scipy.sparse import lil_matrix, csr_matrix
import time


def fast_conv(a):
    """Full self-convolution via FFT."""
    n = len(a)
    s = 1
    while s < 2 * n - 1:
        s <<= 1
    fa = np.fft.rfft(a, s)
    return np.fft.irfft(fa * fa, s)[:2 * n - 1]


def compute_J(a):
    """Compute J(a) = 2*n*max(conv(a,a))/(sum(a))^2."""
    a = np.maximum(a, 0.0)
    S = a.sum()
    if S < 0.01:
        return float('inf')
    n = len(a)
    b = fast_conv(a)
    return float(2 * n * b.max() / (S * S))


def project_simplex(v):
    """Project v onto probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    rho = np.nonzero(cond)[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho = rho[-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0)


def compute_subgradient(a, b, tol_frac=0.1):
    """Compute subgradient of max(b) w.r.t. a at the given point.
    
    Uses all lags k where b[k] >= (1-tol_frac) * max(b).
    Returns (gradient vector, number of active lags).
    """
    n = len(a)
    b_max = b.max()
    if b_max <= 0:
        return np.zeros(n), 0
    tol = tol_frac * b_max
    active = np.where(b >= b_max - tol)[0]
    
    grad = np.zeros(n)
    for k in active:
        i_min = max(0, k - n + 1)
        i_max = min(n - 1, k)
        if i_min <= i_max:
            ki = k - np.arange(i_min, i_max + 1)
            grad[i_min:i_max + 1] += 2.0 * a[ki]
    
    return grad, len(active)


def subgradient_descent(a, deadline, tracker, n_iters=None, use_momentum=True):
    """Fast subgradient descent with adaptive parameters."""
    a = np.maximum(a.astype(np.float64), 0.0)
    S = a.sum()
    if S < 1e-15:
        return a
    a = a / S  # Normalize to simplex
    n = len(a)
    
    if n_iters is None:
        n_iters = 100000
    
    # Momentum
    v = np.zeros(n)
    momentum = 0.9
    # Adaptive step size (decaying)
    step_base = 2.0 / (n ** 0.5)
    best_J = compute_J(a)
    tracker.update(a)
    
    patience = 0
    best_a = a.copy()
    
    for it in range(n_iters):
        if time.monotonic() > deadline:
            break
        
        b = fast_conv(a)
        b_max = b.max()
        
        # Adaptive tolerance: start loose, tighten over time
        progress = min(1.0, it / (n_iters * 0.5))
        tol_frac = 0.3 * (1 - progress) + 0.02 * progress
        
        grad, n_active = compute_subgradient(a, b, tol_frac)
        
        # Normalize gradient
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-15:
            break
        grad = grad / gnorm
        
        # Step size: decaying
        step = step_base / (1.0 + 0.01 * it)
        
        if use_momentum:
            v = momentum * v - step * grad
            a_new = a + v
        else:
            a_new = a - step * grad
        
        # Project onto simplex
        a_new = project_simplex(a_new)
        
        J_new = compute_J(a_new)
        
        if J_new < best_J:
            best_J = J_new
            best_a = a_new.copy()
            patience = 0
            tracker.update(a_new)
        else:
            patience += 1
        
        a = a_new
        
        if patience > 5000:
            # Reset momentum and reduce step
            v = np.zeros(n)
            step_base *= 0.9
            patience = 0
            momentum = min(momentum, 0.8)
        
        if it % 10000 == 0:
            print(f"      SG iter {it}: J={J_new:.8f}, active={n_active}, step={step:.6f}", flush=True)
    
    # Return best found
    if compute_J(best_a) < compute_J(a):
        return best_a
    return a


def coordinate_descent_1d(a, i, b, S):
    """Find optimal value for coordinate i using golden section search."""
    n = len(a)
    old = a[i]
    
    k_lo = i
    k_hi = min(i + n - 1, 2 * n - 2)
    
    # Max b outside affected range
    if k_lo > 0 or k_hi < 2 * n - 2:
        mask = np.ones(2 * n - 1, dtype=bool)
        mask[k_lo:k_hi + 1] = False
        b_max_outside = float(b[mask].max()) if mask.any() else 0.0
    else:
        b_max_outside = 0.0
    
    ks = np.arange(k_lo, k_hi + 1)
    kis = ks - i
    
    base = b[k_lo:k_hi + 1].copy()
    coeffs = 2.0 * a[kis].copy()
    
    has_2i = (k_lo <= 2 * i <= k_hi)
    if has_2i:
        idx_2i = 2 * i - k_lo
        base[idx_2i] -= old * old
        coeffs[idx_2i] = 0.0
    
    def eval_J(x):
        new_S = S - old + x
        if new_S < 1e-15:
            return 1e10
        b_aff = base + (x - old) * coeffs
        if has_2i:
            b_aff = b_aff.copy()
            b_aff[idx_2i] += x * x
        max_b = max(b_max_outside, float(b_aff.max()))
        return 2.0 * n * max_b / (new_S * new_S)
    
    # Golden section search
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    lo = 0.0
    hi = max(3.0 * old + 0.01, 5.0 * S / n, 0.01)
    
    c_pt = hi - gr * (hi - lo)
    d_pt = lo + gr * (hi - lo)
    fc = eval_J(c_pt)
    fd = eval_J(d_pt)
    
    for _ in range(45):
        if fc < fd:
            hi = d_pt
            d_pt = c_pt
            fd = fc
            c_pt = hi - gr * (hi - lo)
            fc = eval_J(c_pt)
        else:
            lo = c_pt
            c_pt = d_pt
            fc = fd
            d_pt = lo + gr * (hi - lo)
            fd = eval_J(d_pt)
    
    x_opt = (lo + hi) / 2.0
    best_x = x_opt
    best_val = eval_J(x_opt)
    
    # Check a few additional points
    for tx in [0.0, old, x_opt * 0.5, x_opt * 1.5, 2.0 * S / n]:
        if tx >= 0:
            v = eval_J(tx)
            if v < best_val:
                best_val = v
                best_x = tx
    
    return max(0.0, best_x)
