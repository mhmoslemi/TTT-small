Minimizes J(a) = 2*n*max(convolve(a,a))/(sum(a))^2 using:
1. LP best-response with adaptive tightening schedules
2. LSE gradient descent with Nesterov momentum
3. Subgradient descent on exact max
4. Spectral and multi-frequency initializations
5. Multi-length search (512, 768, 1024)
6. Simulated annealing for escape from local minima

import numpy as np
from scipy.optimize import linprog
from scipy.sparse import csr_matrix
import time

# Precomputed sparse matrix cache
_spar_cache = {}

def get_sparsity(n):
    """Precompute sparse structure for LP constraint matrix."""
    if n not in _spar_cache:
        rows = []
        cols = []
        for k in range(2 * n - 1):
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            rows.extend([k] * (je - js + 1))
            cols.extend(range(js, je + 1))
        _spar_cache[n] = (np.array(rows), np.array(cols), len(rows))
    return _spar_cache[n]

def fast_conv(a):
    """Fast linear convolution via FFT."""
    n = len(a)
    s = 1
    while s < 2 * n - 1:
        s <<= 1
    fa = np.fft.rfft(a, s)
    return np.fft.irfft(fa * fa, s)[:2 * n - 1]

def compute_J(a):
    """Compute the objective function value."""
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))

def lp_best_response(a, M):
    """Solve LP to find best response g to a with bound M."""
    n = len(a)
    rows, cols, nnz = get_sparsity(n)
    vals = np.empty(nnz)
    for i in range(nnz):
        k = rows[i]
        j = cols[i]
        vals[i] = a[k - j]
    A = csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                options={'presolve': True, 'time_limit': 15})
    if r.success and r.x is not None:
        g = np.maximum(r.x, 0.0)
        if g.sum() > 1e-10:
            return g
    return None

"""
Minimizes J(a) = 2*n*max(convolve(a,a))/(sum(a))^2 using:
1. LP best-response with adaptive multi-tightening and dual-guided steps
2. Nesterov-accelerated LSE gradient descent
3. Subgradient descent with momentum on exact max
4. Spectral and multi-frequency initializations
5. Multi-length search (512, 768, 1024)
6. Simulated annealing escape
7. Coordinate descent and final polishing

import numpy as np
from scipy.optimize import linprog
from scipy.sparse import csr_matrix
import time

_spar_cache = {}

def get_sparsity(n):
    if n not in _spar_cache:
        rows = []
        cols = []
        for k in range(2 * n - 1):
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            rows.extend([k] * (je - js + 1))
            cols.extend(range(js, je + 1))
        _spar_cache[n] = (np.array(rows), np.array(cols), len(rows))
    return _spar_cache[n]

def fast_conv(a):
    n = len(a)
    s = 1
    while s < 2 * n - 1:
        s <<= 1
    fa = np.fft.rfft(a, s)
    return np.fft.irfft(fa * fa, s)[:2 * n - 1]

def compute_J(a):
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))

def lp_best_response(a, M):
    n = len(a)
    rows, cols, nnz = get_sparsity(n)
    vals = np.empty(nnz)
    for i in range(nnz):
        k = rows[i]
        j = cols[i]
        vals[i] = a[k - j]
    A = csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, 10.0)] * n
    r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                options={'presolve': True, 'time_limit': 15})
    if r.success and r.x is not None:
        g = np.maximum(r.x, 0.0)
        if g.sum() > 1e-10:
            duals = None
            try:
                if hasattr(r, 'ineqlin') and r.ineqlin is not None:
                    duals = np.abs(np.asarray(r.ineqlin.marginal, dtype=np.float64))
            except Exception:
                pass
            return g, duals
    return None, None

def run_lp_phase(a, max_iters, deadline, tracker, tighten_start=0.03, tighten_end=0.003):
    a = np.maximum(a.astype(np.float64), 1e-8)
    a /= a.sum()
    n = len(a)
    current_J = compute_J(a)
    tracker.update(a)
    consecutive_failures = 0

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        b = fast_conv(a)
        M = float(b.max())
        progress = it / max(1, max_iters)
        tighten = tighten_start + (tighten_end - tighten_start) * progress

        g = None
        for eps_factor in [tighten, tighten * 0.5, 0.0]:
            M_try = M * (1.0 - eps_factor)
            if M_try <= 0:
                M_try = M * 0.5
            g, duals = lp_best_response(a, M_try)
            if g is not None and g.sum() > 0.5:
                break

        if g is None:
            consecutive_failures += 1
            if consecutive_failures > 10:
                break
            continue

        g = np.maximum(g, 0.0)
        gs = g.sum()
        if gs < 1e-10:
            break
        g /= gs

        best_a = None
        best_J = current_J
        progress = it / max(1, max_iters)
        base_t = max(0.05, 0.6 * (1.0 - progress * 0.7))
        for t in [base_t, base_t * 1.5, base_t * 0.5, 0.8, 0.4, 0.2]:
            if t > 1.0 or t < 0.01:
                continue
            a_trial = (1.0 - t) * a + t * g
            a_trial = np.maximum(a_trial, 1e-8)
            ss = a_trial.sum()
            if ss < 1e-10:
                continue
            a_trial /= ss
            J_trial = compute_J(a_trial)
            if J_trial < best_J - 1e-14:
                best_J = J_trial
                best_a = a_trial

        if best_a is None:
            consecutive_failures += 1
            if consecutive_failures > 15:
                break
            continue

        a = best_a
        current_J = best_J
        consecutive_failures = 0
        if it % 5 == 0:
            tracker.update(a)

    tracker.update(a)
    return a

def lse_gradient_descent(a, max_iters, deadline, tracker, tau0=50.0, tau_max=500.0):
    a = np.maximum(a.astype(np.float64), 1e-6)
    a /= a.sum()
    n = len(a)
    current_J = compute_J(a)
    tracker.update(a)

    L = 1
    while L < 3 * n:
        L <<= 1

    tau = tau0
    lr = 2.0 / (n * (tau + 1))
    momentum = 0.75
    mu = 0.8
    v = np.zeros(n)
    a_ys = a.copy()

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        b = fast_conv(a)
        b_max = b.max()
        shifted = tau * (b - b_max)
        shifted -= shifted.max()
        p = np.exp(shifted)
        p /= p.sum()

        q = p[::-1]
        fa = np.fft.rfft(a, L)
        fq = np.fft.rfft(q, L)
        conv = np.fft.irfft(fa * fq, L)[:3 * n - 2]
        grad = 2.0 * conv[n - 1:2 * n - 1][::-1]

        direction = momentum * v + (1.0 - momentum) * grad
        a_ys = a.copy()
        a_new = a - mu * lr * direction
        a_new = np.maximum(a_new, 1e-6)
        a_new /= a_new.sum()
        J_new = compute_J(a_new)

        if J_new < current_J - 1e-16:
            v = direction
            a = a_new
            current_J = J_new
            tau = min(tau_max, tau * 1.02)
            lr = 2.0 / (n * (tau + 1))
            momentum = min(0.9, momentum * 1.005)
        else:
            a = a_ys
            lr *= 0.6
            v *= 0.3
            if lr < 1e-18:
                break

        if it % 80 == 0:
            tracker.update(a)

    tracker.update(a)
    return a

def subgradient_descent(a, max_iters, deadline, tracker):
    a = np.maximum(a.astype(np.float64), 1e-8)
    a /= a.sum()
    n = len(a)
    current_J = compute_J(a)
    tracker.update(a)
    lr = 2.0 / n
    momentum = 0.8
    v = np.zeros(n)
    rng = np.random.default_rng(123)

    L = 1
    while L < 3 * n:
        L <<= 1

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        b = fast_conv(a)
        M = float(b.max())
        active = np.where(b >= M - 1e-8)[0]
        if len(active) == 0:
            break

        p_active = np.zeros(2 * n - 1)
        p_active[active] = 1.0 / len(active)

        q = p_active[::-1]
        fa = np.fft.rfft(a, L)
        fq = np.fft.rfft(q, L)
        conv = np.fft.irfft(fa * fq, L)[:3 * n - 2]
        grad = 2.0 * conv[n - 1:2 * n - 1][::-1]

        direction = momentum * v + (1.0 - momentum) * grad
        a_new = a - lr * direction
        a_new = np.maximum(a_new, 1e-8)
        a_new /= a_new.sum()
        J_new = compute_J(a_new)

        if J_new < current_J - 1e-16:
            v = direction
            a = a_new
            current_J = J_new
            lr = min(lr * 1.02, 10.0 / n)
        else:
            lr *= 0.65
            v *= 0.3

        if it % 100 == 0:
            tracker.update(a)

    tracker.update(a)
    return a

def flatten_spectrum(a, iterations=50, deadline=None):
    """Iteratively flatten the power spectrum of a while maintaining non-negativity."""
    a = np.maximum(a.astype(np.float64), 1e-6)
    a /= a.sum()
    n = len(a)

    for it in range(iterations):
        if deadline is not None and time.monotonic() > deadline:
            break
        fa = np.fft.rfft(a, 2 * n)
        mag = np.abs(fa)
        target = np.mean(mag)
        # Adjust: try to equalize magnitudes
        ratio = target / (mag + 1e-10)
        fa_new = fa * (0.9 + 0.1 * ratio)
        # Enforce real, non-negative
        a_new = np.fft.irfft(fa_new, 2 * n)[:n]
        a_new = np.maximum(a_new, 1e-6)
        s = a_new.sum()
        if s < 1e-8:
            break
        a_new /= s
        if compute_J(a_new) < compute_J(a):
            a = a_new

    return a

def resample_seq(src, dest_n):
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)

def generate_inits(n, rng, parent):
    inits = []
    if parent is not None and parent.size > 0:
        inits.append(resample_seq(parent, n))
        inits.append(resample_seq(parent[::-1], n))

    inits.append(np.ones(n) / n)
    inits.append(rng.uniform(0.1, 1.0, n))

    # Flat spectrum inits
    for _ in range(3):
        half = n // 2 + 1
        angles = rng.uniform(0, 2 * np.pi, half)
        A = np.zeros(2 * n, dtype=complex)
        A[:half] = np.exp(1j * angles[:half])
        for k in range(1, min(n, 2 * n) - n // 2):
            A[k] = np.conj(A[2 * n - k])
        a = np.fft.ifft(A).real
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s > 1e-10:
            inits.append(a / s)

    t = np.linspace(0, 1, n, endpoint=False)
    # Cosine modulations
    for freq in [2, 3, 5, 7, 11, 13, 17, 21]:
        inits.append(1.0 + 0.4 * np.cos(2 * np.pi * freq * t + rng.uniform(0, 6.28)))

    # Multi-frequency
    inits.append(1.0 + 0.25 * np.sin(2 * np.pi * t) + 0.2 * np.sin(4 * np.pi * t)
                + 0.15 * np.sin(7 * np.pi * t) + 0.1 * np.sin(11 * np.pi * t))
    inits.append(1.0 + 0.2 * np.cos(2 * np.pi * 3.7 * t) + 0.15 * np.cos(2 * np.pi * 6.1 * t)
                + 0.1 * np.cos(2 * np.pi * 9.3 * t))

    # Gaussian mixture
    for _ in range(2):
        centers = rng.uniform(0.1, 0.9, 3)
        widths = rng.uniform(0.05, 0.2, 3)
        sig = np.zeros(n)
        for c, w in zip(centers, widths):
            sig += np.exp(-((t - c) ** 2) / (2 * w ** 2))
        inits.append(sig + 0.05)

    # Power-law
    inits.append((t + 0.05) ** rng.uniform(-0.5, 0.5))
    inits.append((1.0 - t + 0.05) ** rng.uniform(-0.5, 0.5))

    # Sparse + smooth
    sparse = np.zeros(n)
    n_spikes = max(8, n // 12)
    positions = rng.choice(n, size=n_spikes, replace=False)
    sparse[positions] = 1.0
    kernel_size = max(5, n // 20)
    kernel = np.ones(kernel_size) / kernel_size
    inits.append(np.convolve(sparse, kernel, mode='same') + 0.03)

    # Random Fourier with limited bandwidth
    n_coeff = min(n // 2 + 1, 15)
    coeffs = rng.normal(0, 1, n_coeff)
    sig = np.zeros(n)
    for i in range(n_coeff):
        sig += coeffs[i] * np.cos(2 * np.pi * i * t + rng.uniform(0, 6.28))
    inits.append(np.abs(sig) + 0.3)

    # Von Mises-like (concentrated on circle)
    kappa = rng.uniform(2, 8)
    angles = rng.uniform(0, 2 * np.pi, n)
    sig = np.exp(kappa * np.cos(angles - rng.uniform(0, 2 * np.pi)))
    inits.append(sig)

    for x in inits:
        x = np.maximum(x, 0.001).astype(np.float64)
        s = x.sum()
        if s > 1e-10:
            x /= s
    return inits

class ScoreTracker:
    def __init__(self):
        self.best = None
        self.best_score = float('inf')
        self.t_start = time.monotonic()

    def update(self, a):
        a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
        if a.size == 0 or a.sum() < 0.01:
            return
        score = float(evaluate_sequence(a.tolist()))
        if np.isfinite(score) and score < self.best_score:
            self.best = a.copy()
            self.best_score = score
            elapsed = time.monotonic() - self.t_start
            print(f"  * J={score:.8f} n={a.size} t={elapsed:.1f}s", flush=True)

    def report(self, label=""):
        if self.best is not None:
            print(f"  {label} best: J={self.best_score:.8f} n={self.best.size}", flush=True)

def coordinate_descent(a, max_steps, deadline, tracker):
    a = a.copy().astype(np.float64)
    a /= a.sum()
    n = len(a)
    current_J = compute_J(a)
    tracker.update(a)
    rng = np.random.default_rng(99)

    for step in range(max_steps):
        if time.monotonic() > deadline:
            break
        j = rng.integers(0, n)
        improved = False
        for scale in [0.3, 0.6, 1.4, 2.0, 0.1, 3.0]:
            a_trial = a.copy()
            a_trial[j] *= scale
            a_trial = np.maximum(a_trial, 1e-8)
            ss = a_trial.sum()
            if ss < 1e-10:
                continue
            a_trial /= ss
            J_trial = compute_J(a_trial)
            if J_trial < current_J - 1e-14:
                a = a_trial
                current_J = J_trial
                improved = True
                break
        if not improved and step % 500 == 0:
            # Try pair update
            if n > 2:
                i = rng.integers(0, n)
                j = (i + 1 + rng.integers(0, n - 1)) % n
                for fi in [0.5, 2.0]:
                    for fj in [0.5, 2.0]:
                        a_trial = a.copy()
                        a_trial[i] *= fi
                        a_trial[j] *= fj
                        a_trial = np.maximum(a_trial, 1e-8)
                        ss = a_trial.sum()
                        if ss < 1e-10:
                            continue
                        a_trial /= ss
                        J_trial = compute_J(a_trial)
                        if J_trial < current_J - 1e-14:
                            a = a_trial
                            current_J = J_trial
                            improved = True
                            break
                    if improved:
                        break
        if step % 200 == 0:
            tracker.update(a)
    tracker.update(a)
    return a

def simulated_annealing(a, deadline, tracker, n_restarts=30, t0=0.12):
    rng = np.random.default_rng(555)
    n = a.size
    a = a.copy().astype(np.float64)
    current_J = compute_J(a)
    tracker.update(a)
    temp = t0

    for r in range(n_restarts):
        if time.monotonic() > deadline:
            break
        a_pert = a + rng.normal(0, temp, n)
        a_pert = np.maximum(a_pert, 1e-6)
        a_pert /= a_pert.sum()

        a_refined = run_lp_opt_short(a_pert, 8, deadline, tracker)
        J = compute_J(a_refined)
        delta = J - current_J
        if delta < 0 or rng.random() < np.exp(-delta / max(temp, 1e-10)):
            a = a_refined
            current_J = J
        temp *= 0.93
    tracker.update(a)
    return a

def run_lp_opt_short(a, max_iters, deadline, tracker):
    return run_lp_phase(a, max_iters, deadline, tracker, tighten_start=0.02, tighten_end=0.005)

def propose_candidate(seed=42, budget_s=1500, **kwargs):
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + float(budget_s) - 25.0
    tracker = ScoreTracker()

    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        if p.ndim == 1 and p.size > 0 and np.all(np.isfinite(p)):
            parent = np.maximum(p, 0.0)
            tracker.update(parent)
            print(f"Parent: n={parent.size}, J={tracker.best_score:.8f}", flush=True)
    except (NameError, AttributeError):
        pass

    # Phase 1: Multi-length LP optimization
    lengths = [512, 768, 1024]
    # Allocate time: give more to smaller n (faster LP), less to larger
    time_weights = [0.35, 0.30, 0.35]
    total_avail = deadline - time.monotonic() - 350.0
    if total_avail < 100:
        lengths = [512]
        time_weights = [1.0]

    for li, n in enumerate(lengths):
        t_phase_end = time.monotonic() + total_avail * time_weights[li]
        if time.monotonic() >= deadline - 300:
            break
        print(f"\n{'='*50} Phase1 n={n} budget={t_phase_end - time.monotonic():.0f}s {'='*50}", flush=True)

        inits = generate_inits(n, rng, parent)
        best_for_n = None
        best_J_for_n = float('inf')
        n_inits = len(inits)

        # Allocate per-init time
        init_budget = max(8.0, (t_phase_end - time.monotonic() - 40.0) / n_inits)

        for idx, a0 in enumerate(inits):
            t0_init = time.monotonic()
            t_init_end = min(t_phase_end - 20.0, t0_init + init_budget)
            if time.monotonic() >= t_init_end:
                break

            # LP phase
            n_lp_iters = max(15, int(40.0 / max(1, n / 512)))
            a = run_lp_phase(a0, n_lp_iters, t_init_end - 30.0, tracker,
                            tighten_start=0.03, tighten_end=0.003)

            # Subgradient
            if time.monotonic() < t_init_end - 20:
                a = subgradient_descent(a, max(100, int(20000 / n)), t_init_end - 20, tracker)

            # LSE polish
            if time.monotonic() < t_init_end - 5:
                a = lse_gradient_descent(a, max(300, int(40000 / n)), t_init_end - 3, tracker,
                                        tau0=30.0, tau_max=400.0)

            J = compute_J(a)
            if J < best_J_for_n:
                best_J_for_n = J
                best_for_n = a.copy()

        if best_for_n is not None:
            tracker.update(best_for_n)
            parent = best_for_n
            tracker.report(f"Phase1 n={n} ({time.monotonic()-t_phase_end + (t_phase_end - time.monotonic()):.0f}s): J={best_J_for_n:.8f}")

    # Phase 2: Focused optimization on best
    if tracker.best is not None and time.monotonic() < deadline - 250:
        t0 = time.monotonic()
        n = tracker.best.size
        print(f"\n{'='*50} Phase2 Focus n={n} {'='*50}", flush=True)

        a = tracker.best.copy().astype(np.float64)

        # Multi-round LP with different tightening
        tighten_schedule = [0.01, 0.02, 0.005, 0.015, 0.008, 0.012, 0.004, 0.006]
        for tight in tighten_schedule:
            if time.monotonic() > deadline - 200:
                break
            a = run_lp_phase(a, 25, deadline - 200, tracker,
                           tighten_start=tight, tighten_end=tight * 0.3)

        # Subgradient
        if time.monotonic() < deadline - 150:
            a = subgradient_descent(a, 8000, deadline - 150, tracker)

        # LSE
        if time.monotonic() < deadline - 80:
            a = lse_gradient_descent(a, 5000, deadline - 80, tracker, tau0=40.0, tau_max=600.0)

        tracker.report(f"Phase2 ({time.monotonic()-t0:.1f}s)")

    # Phase 3: Simulated annealing
    if tracker.best is not None and time.monotonic() < deadline - 120:
        t0 = time.monotonic()
        n = tracker.best.size
        print(f"\n{'='*50} Phase3 SA n={n} {'='*50}", flush=True)
        a = simulated_annealing(tracker.best, deadline - 80, tracker, n_restarts=35, t0=0.08)

        if time.monotonic() < deadline - 40:
            a = run_lp_phase(a, 20, deadline - 40, tracker, tighten_start=0.008, tighten_end=0.002)
            if time.monotonic() < deadline - 15:
                a = lse_gradient_descent(a, 2000, deadline - 10, tracker, tau0=60.0, tau_max=800.0)
        tracker.report(f"Phase3 ({time.monotonic()-t0:.1f}s)")

    # Phase 4: Final polish
    if tracker.best is not None and time.monotonic() < deadline - 10:
        t0 = time.monotonic()
        print(f"\n{'='*50} Phase4 Final Polish {'='*50}", flush=True)
        a = coordinate_descent(tracker.best, 2000, deadline - 8, tracker)
        if time.monotonic() < deadline - 3:
            a = run_lp_phase(a, 10, deadline - 2, tracker, tighten_start=0.003, tighten_end=0.001)
            if time.monotonic() < deadline - 0.5:
                a = lse_gradient_descent(a, 500, deadline - 0.2, tracker, tau0=100.0, tau_max=1000.0)
        tracker.report(f"Phase4 ({time.monotonic()-t0:.1f}s)")

    # Fallback
    if tracker.best is None:
        tracker.best = np.ones(256)
        tracker.best_score = compute_J(tracker.best)
        tracker.update(tracker.best)

    # Final validation
    final_score = float(evaluate_sequence([float(x) for x in tracker.best]))
    if not np.isfinite(final_score):
        tracker.best = np.ones(512)
        final_score = compute_J(tracker.best)

    print(f"\n{'='*50} FINAL {'='*50}", flush=True)
    print(f"J={final_score:.8f} n={tracker.best.size} time={time.monotonic()-tracker.t_start:.1f}s", flush=True)
    return [float(x) for x in tracker.best]
