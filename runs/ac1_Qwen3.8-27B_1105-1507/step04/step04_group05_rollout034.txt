def fast_conv(a):
    """Full self-convolution via FFT."""
    n = len(a)
    if n == 0:
        return np.array([0.0])
    s = 1
    while s < 2 * n - 1:
        s <<= 1
    fa = np.fft.rfft(a, s)
    return np.fft.irfft(fa * fa, s)[:2 * n - 1]


def compute_J(a):
    """Compute J(a) = 2*n*max(conv(a,a))/(sum(a))^2."""
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))

    # Let me rebuild properly

def lp_best_response(a, M):
    """LP: max sum(g) s.t. (a*g)_k <= M for all k, g >= 0."""
    n = len(a)
    c = -np.ones(n)
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        jl = max(0, k - n + 1)
        jh = min(n - 1, k)
        for j in range(jl, jh + 1):
            # (a*g)_k = sum_j a[j] * g[k-j]
            # coefficient of g[j] in constraint k is a[k-j]
            ki = k - j
            if 0 <= ki < n:
                rows.append(k)
                cols.append(j)
                vals.append(a[ki])
    A = coo_matrix((vals, (rows, cols)), shape=(2 * n - 1, n)).tocsr()
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
    if r.success and r.x is not None:
        g = np.maximum(r.x, 0.0)
        if g.sum() > 1e-10:
            return g
    return None


def mirror_descent_step(a, b, S, eta, active_tol=1e-4):
    """One mirror descent step. a: current (on simplex), b: convolution, S: sum(a)."""
    n = len(a)
    b_max = b.max()
    if b_max < 1e-15:
        return a.copy()
    # Active set
    active = np.where(b >= b_max * (1.0 - active_tol))[0]
    if len(active) == 0:
        active = np.array([int(np.argmax(b))])
    
    # Subgradient: g_i = (1/|K|) * sum_{k in K} 2*a[k-i]
    g = np.zeros(n)
    for k in active:
        jl = max(0, k - n + 1)
        jh = min(n - 1, k)
        for j in range(jl, jh + 1):
            ki = k - j
            if 0 <= ki < n:
                g[j] += a[ki]
    g /= len(active)
    
    # Mirror descent: v_i = a_i * exp(-eta * g_i) / Z
    # Clip for numerical stability
    log_ratio = -eta * g
    log_ratio = np.clip(log_ratio, -50, 50)
    v = a * np.exp(log_ratio)
    Z = v.sum()
    if Z < 1e-15:
        return a.copy()
    v /= Z
    return v


def vectorized_cd_pass(a, n_samples=15):
    """Fast vectorized coordinate descent pass. Returns (new_a, new_J)."""
    a = np.maximum(a.astype(np.float64), 0.0).copy()
    n = len(a)
    S = a.sum()
    if S < 1e-10:
        return a, float('inf')
    b = fast_conv(a)
    current_J = 2.0 * n * b.max() / (S * S)
    
    indices = list(range(n))
    np.random.shuffle(indices)
    
    for i in indices:
        S_prime = S - a[i]
        if S_prime < 1e-10:
            continue
        
        # Determine range of k that change
        k_lo = i
        k_hi = min(2 * n - 2, i + n - 1)
        
        # Outside max (k not in [i, i+n-1])
        if k_lo > 0 or k_hi < 2 * n - 2:
            if k_lo == 0 and k_hi == 2 * n - 2:
                C_out = 0.0
            else:
                mask = np.ones(2*n-1, dtype=bool)
                mask[k_lo:k_hi+1] = False
                C_out = b[mask].max() if mask.any() else 0.0
        else:
            C_out = 0.0
        
        # For k in [i, i+n-1], k != 2i:
        # c_new[k] = b[k] + 2*(x - a[i]) * a[k-i]
        # For k = 2i (if in range): c_new[2i] = b[2i] + x^2 - a[i]^2
        
        # Build arrays for vectorized evaluation
        k_range = np.arange(k_lo, k_hi + 1)
        ki_range = k_range - i  # = 0 to n-1 (or subset)
        valid = (ki_range >= 0) & (ki_range < n)
        k_valid = k_range[valid]
        ki_valid = ki_range[valid]
        
        # Separate k=2i case
        two_i = 2 * i
        if k_lo <= two_i <= k_hi:
            mask_linear = k_valid != two_i
        else:
            mask_linear = np.ones(len(k_valid), dtype=bool)
        
        k_lin = k_valid[mask_linear]
        ki_lin = ki_valid[mask_linear]
        b_lin = b[k_lin]
        a_ki = a[ki_lin]
        
        # Sample x values
        x_max = max(3.0 * a[i], S_prime / max(1, n - 1), 0.01)
        x_max = min(x_max, 10.0)
        x_candidates = np.linspace(0.0, x_max, n_samples)
        x_candidates = np.append(x_candidates, a[i])
        
        best_x = a[i]
        best_val = float('inf')
        
        for x in x_candidates:
            # Compute max convolution
            max_c = C_out
            # Linear part
            if len(k_lin) > 0:
                c_lin = b_lin + 2.0 * (x - a[i]) * a_ki
                max_c = max(max_c, c_lin.max())
            # Quadratic part (k = 2i)
            if k_lo <= two_i <= k_hi:
                c_quad = b[two_i] + x * x - a[i] * a[i]
                max_c = max(max_c, c_quad)
            
            new_S = S_prime + x
            if new_S < 1e-10:
                continue
            val = max_c / (new_S * new_S)
            if val < best_val:
                best_val = val
                best_x = x
        
        # Refine around best_x with a few more samples
        if best_x > 0 and best_x < x_max:
            delta = x_max / n_samples
            x_fine = np.linspace(max(0, best_x - 2*delta), best_x + 2*delta, 8)
            for x in x_fine:
                if x < 0:
                    continue
                max_c = C_out
                if len(k_lin) > 0:
                    c_lin = b_lin + 2.0 * (x - a[i]) * a_ki
                    max_c = max(max_c, c_lin.max())
                if k_lo <= two_i <= k_hi:
                    c_quad = b[two_i] + x * x - a[i] * a[i]
                    max_c = max(max_c, c_quad)
                new_S = S_prime + x
                if new_S < 1e-10:
                    continue
                val = max_c / (new_S * new_S)
                if val < best_val:
                    best_val = val
                    best_x = x
        
        # Update
        old_val = a[i]
        if abs(best_x - old_val) > 1e-12:
            a[i] = best_x
            S = S_prime + best_x
            # Update convolution
            for k in range(k_lo, k_hi + 1):
                ki = k - i
                if 0 <= ki < n:
                    if k == 2 * i:
                        b[k] = b[k] - old_val * old_val + best_x * best_x
                    else:
                        b[k] = b[k] - 2.0 * old_val * a[ki] + 2.0 * best_x * a[ki]
    
    # Verify
    b_check = fast_conv(a)
    S = a.sum()
    new_J = 2.0 * n * b_check.max() / (S * S) if S > 1e-10 else float('inf')
    return a, new_J


def run_mirror_descent(a, n_steps, deadline, tracker):
    """Run mirror descent iterations."""
    a = np.maximum(a.astype(np.float64), 0.001).copy()
    a /= a.sum()
    n = len(a)
    current_J = compute_J(a)
    tracker.update(a)
    
    b = fast_conv(a)
    S = 1.0  # normalized
    
    eta = 0.5
    stall = 0
    
    for step in range(n_steps):
        if time.monotonic() > deadline:
            break
        
        v = mirror_descent_step(a, b, S, eta)
        
        # Line search
        best_t = 0.0
        best_J = current_J
        for t in [0.1, 0.2, 0.3, 0.5, 0.7, 0.9]:
            a_trial = (1 - t) * a + t * v
            a_trial = np.maximum(a_trial, 0.0)
            s = a_trial.sum()
            if s < 1e-10:
                continue
            a_trial /= s
            J_trial = compute_J(a_trial)
            if J_trial < best_J - 1e-12:
                best_J = J_trial
                best_t = t
        
        if best_t == 0.0:
            stall += 1
            eta *= 0.7
            if stall > 30:
                break
        else:
            a = (1 - best_t) * a + best_t * v
            a = np.maximum(a, 0.0)
            a /= a.sum()
            current_J = best_J
            stall = 0
            b = fast_conv(a)
            tracker.update(a)
        
        # Adapt eta
        if step % 50 == 0 and step > 0:
            J_now = compute_J(a)
            if J_now < current_J * 0.99:
                eta = min(eta * 1.2, 5.0)
            else:
                eta = max(eta * 0.8, 0.05)
            current_J = J_now
    
    return a


def run_lp_iterations(a, max_iters, deadline, tracker, tighten_base=0.003):
    """LP-based best response iterations."""
    a = np.maximum(a.astype(np.float64), 0.001).copy()
    a /= a.sum()
    n = len(a)
    current_J = compute_J(a)
    tracker.update(a)
    
    tighten = tighten_base
    stall_count = 0
    
    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        
        b = fast_conv(a)
        M = float(b.max())
        
        # Try to tighten
        M_target = M * (1.0 - tighten)
        g = lp_best_response(a, M_target)
        
        if g is None:
            # Relax
            M_target = M * (1.0 - tighten * 0.3)
            g = lp_best_response(a, M_target)
        
        if g is None:
            tighten *= 0.5
            if tighten < 1e-7:
                break
            continue
        
        g = np.maximum(g, 0.0)
        g /= g.sum()
        
        # Line search along direction g
        best_t = 0.0
        best_J = current_J
        progress = min(1.0, it / max(1, max_iters))
        base_t = max(0.05, 0.6 * (1.0 - 0.5 * progress))
        for t in [base_t, base_t*0.5, base_t*1.5, base_t*0.3, 0.8, 0.5, 0.3]:
            if t <= 0 or t > 1:
                continue
            a_trial = (1.0 - t) * a + t * g
            a_trial = np.maximum(a_trial, 0.0)
            s = a_trial.sum()
            if s < 1e-10:
                continue
            a_trial /= s
            J_trial = compute_J(a_trial)
            if J_trial < best_J - 1e-13:
                best_J = J_trial
                best_t = t
        
        if best_t == 0.0:
            stall_count += 1
            tighten *= 0.6
            if stall_count > 30:
                break
            continue
        
        a = (1.0 - best_t) * a + best_t * g
        a = np.maximum(a, 0.0)
        a /= a.sum()
        current_J = best_J
        stall_count = 0
        
        # Adapt tighten
        if it % 10 == 0:
            tighten = min(0.05, tighten * 1.1)
        if stall_count > 5:
            tighten = max(0.001, tighten * 0.7)
        
        tracker.update(a)
    
    return a


def resample_seq(src, dest_n):
    """Resample 1D sequence to new length."""
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)


class ScoreTracker:
    def __init__(self):
        self.best = None
        self.best_score = float('inf')
        self.t_start = time.monotonic()
    
    def update(self, a):
        a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
        if a.size == 0 or a.sum() < 0.01:
            return
        score = float(evaluate_sequence(a.tolist()))
        if np.isfinite(score) and score < self.best_score:
            self.best = a.copy()
            self.best_score = score
            elapsed = time.monotonic() - self.t_start
            print(f"  * NEW BEST: J={score:.8f} n={a.size} t={elapsed:.0f}s", flush=True)
    
    def report(self, label=""):
        if self.best is not None:
            print(f"  {label} best: J={self.best_score:.8f} n={self.best.size} t={time.monotonic()-self.t_start:.0f}s", flush=True)


def optimize_single(a, deadline, tracker, phase_name=""):
    """Run full optimization pipeline on a single sequence."""
    a = np.maximum(a.astype(np.float64), 0.001).copy()
    a /= a.sum()
    n = len(a)
    
    t0 = time.monotonic()
    J0 = compute_J(a)
    tracker.update(a)
    print(f"  {phase_name}: n={n}, J_start={J0:.8f}", flush=True)
    
    remaining = deadline - time.monotonic()
    
    # Phase A: Mirror descent (fast)
    md_budget = min(remaining * 0.3, 60)
    if md_budget > 5:
        a = run_mirror_descent(a, 500, t0 + md_budget, tracker)
    
    # Phase B: LP iterations
    remaining = deadline - time.monotonic()
    lp_budget = min(remaining * 0.4, 120)
    if lp_budget > 10:
        n_lp = max(10, int(lp_budget / 0.1))
        a = run_lp_iterations(a, min(n_lp, 200), t0 + md_budget + lp_budget, tracker,
                             tighten_base=0.003)
    
    # Phase C: Coordinate descent polish
    remaining = deadline - time.monotonic()
    if remaining > 10:
        cd_budget = min(remaining * 0.6, 90)
        cd_deadline = time.monotonic() + cd_budget
        J_prev = compute_J(a)
        for p in range(20):
            if time.monotonic() > cd_deadline:
                break
            a, J_new = vectorized_cd_pass(a)
            a /= a.sum()
            J_new = compute_J(a)
            tracker.update(a)
            if J_new >= J_prev - 1e-10:
                if p > 3:
                    break
            J_prev = J_new
    
    # Phase D: Final LP + CD
    remaining = deadline - time.monotonic()
    if remaining > 15:
        a = run_lp_iterations(a, 20, time.monotonic() + min(remaining*0.5, 30), tracker,
                             tighten_base=0.001)
        if time.monotonic() < deadline - 5:
            a, _ = vectorized_cd_pass(a)
            a /= a.sum()
            tracker.update(a)
    
    elapsed = time.monotonic() - t0
    print(f"  {phase_name}: J_end={compute_J(a):.8f}, time={elapsed:.1f}s", flush=True)
    return a


def propose_candidate(seed=42, budget_s=1500, **kwargs):
    """
    Main search function combining fast CD, mirror descent, LP,
    and multi-length exploration.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + float(budget_s) - 20.0
    tracker = ScoreTracker()
    
    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        if p.ndim == 1 and p.size > 0 and np.all(np.isfinite(p)):
            parent = np.maximum(p, 0.0)
            if parent.sum() > 0.01:
                tracker.update(parent)
                print(f"Parent loaded: n={parent.size}, J={tracker.best_score:.8f}", flush=True)
    except (NameError, AttributeError):
        pass
    
    if parent is None:
        parent = np.ones(384)
        tracker.update(parent)
        print(f"No parent, using uniform n={parent.size}", flush=True)
    
    n_parent = parent.size
    
    # === Phase 1: Intensive optimization on parent ===
    print(f"\n{'='*50} Phase 1: Optimize parent (n={n_parent}) {'='*50}", flush=True)
    t0 = time.monotonic()
    p1_deadline = deadline - 600
    a_best = optimize_single(parent.copy(), p1_deadline, tracker, phase_name="Parent")
    
    # === Phase 2: Try different lengths ===
    print(f"\n{'='*50} Phase 2: Length exploration {'='*50}", flush=True)
    t0 = time.monotonic()
    
    # Generate candidate lengths
    candidate_lengths = []
    for delta in [-10, -8, -6, -4, -2, 0, 2, 4, 6, 8, 10, 12, 16, 20, 30, 50]:
        nl = n_parent + delta
        if 100 <= nl <= 800:
            candidate_lengths.append(nl)
    
    # Also try some very different lengths
    for nl in [128, 192, 256, 320, 448, 512, 640]:
        if nl not in candidate_lengths and 100 <= nl <= 800:
            candidate_lengths.append(nl)
    
    candidate_lengths = sorted(set(candidate_lengths))
    
    # Prioritize lengths close to parent
    candidate_lengths.sort(key=lambda x: abs(x - n_parent))
    
    p2_deadline = deadline - 200
    per_length_time = max(10, (p2_deadline - time.monotonic()) / max(1, len(candidate_lengths)))
    
    for nl in candidate_lengths:
        if time.monotonic() > p2_deadline:
            break
        
        a_resampled = resample_seq(tracker.best, nl)
        a_resampled = np.maximum(a_resampled, 0.001)
        a_resampled /= a_resampled.sum()
        
        # Quick optimization with limited budget
        budget = min(per_length_time, 40)
        if budget < 5:
            # Very quick: just 1 CD pass
            a_resampled, _ = vectorized_cd_pass(a_resampled)
            a_resampled /= a_resampled.sum()
            tracker.update(a_resampled)
        else:
            optimize_single(a_resampled, time.monotonic() + budget, tracker, phase_name=f"n={nl}")
    
    # === Phase 3: Spectral initialization attempts ===
    print(f"\n{'='*50} Phase 3: Spectral inits {'='*50}", flush=True)
    t0 = time.monotonic()
    
    if time.monotonic() < deadline - 100:
        n_spec = tracker.best.size
        # Try: a[j] = |eigenvector components| for the top eigenvector of avg Q
        # Avg Q: the matrix that averages over all Q_k weighted by the "importance"
        # Simplified: use a random projection approach
        for attempt in range(3):
            if time.monotonic() > deadline - 80:
                break
            # Initialize with a structured pattern
            t = np.linspace(0, 1, n_spec, endpoint=False)
            if attempt == 0:
                a_init = 1.0 + 0.3 * np.cos(2*np.pi*3*t + rng.uniform(0, 6)) + 0.2*np.cos(2*np.pi*7*t)
            elif attempt == 1:
                a_init = np.exp(-0.5*((t-0.5)*4)**2) + 0.3
            else:
                a_init = np.abs(np.sin(2*np.pi*2*t)) * 0.5 + 0.5
            
            a_init = np.maximum(a_init, 0.01)
            a_init /= a_init.sum()
            
            budget = min(30, deadline - time.monotonic() - 50)
            if budget > 5:
                a_trial = run_mirror_descent(a_init, 100, time.monotonic() + budget*0.5, tracker)
                if time.monotonic() < deadline - 50:
                    a_trial, _ = vectorized_cd_pass(a_trial)
                    a_trial /= a_trial.sum()
                    tracker.update(a_trial)
    
    # === Phase 4: Perturbation restarts on best ===
    print(f"\n{'='*50} Phase 4: Perturbation restarts {'='*50}", flush=True)
    t0 = time.monotonic()
    
    if time.monotonic() < deadline - 40:
        n = tracker.best.size
        for temp in [0.1, 0.05, 0.03, 0.02, 0.01]:
            if time.monotonic() > deadline - 30:
                break
            a_pert = tracker.best + rng.normal(0, temp, n)
            a_pert = np.maximum(a_pert, 0.01)
            a_pert /= a_pert.sum()
            
            budget = min(20, deadline - time.monotonic() - 20)
            if budget > 5:
                a_pert = run_mirror_descent(a_pert, 50, time.monotonic() + budget*0.4, tracker)
                if time.monotonic() < deadline - 15:
                    a_pert, _ = vectorized_cd_pass(a_pert)
                    a_pert /= a_pert.sum()
                    tracker.update(a_pert)
    
    # === Phase 5: Final intensive polish ===
    print(f"\n{'='*50} Phase 5: Final polish {'='*50}", flush=True)
    t0 = time.monotonic()
    
    if time.monotonic() < deadline - 5:
        remaining = deadline - time.monotonic() - 5
        # Intensive CD
        a_final = tracker.best.copy()
        for p in range(30):
            if time.monotonic() > deadline - 5:
                break
            a_final, J_new = vectorized_cd_pass(a_final)
            a_final /= a_final.sum()
            J_new = compute_J(a_final)
            tracker.update(a_final)
            if p > 5:
                # Check convergence
                J_check = compute_J(a_final)
                if J_check >= J_new * (1 - 1e-10):
                    pass
        
        # Final LP
        if time.monotonic() < deadline - 10:
            a_final = run_lp_iterations(a_final, 30, deadline - 5, tracker, tighten_base=0.001)
            if time.monotonic() < deadline - 3:
                a_final, _ = vectorized_cd_pass(a_final)
                a_final /= a_final.sum()
                tracker.update(a_final)
    
    # === Phase 6: Try to shave with very small CD passes ===
    if time.monotonic() < deadline - 2:
        a_final = tracker.best.copy()
        for _ in range(10):
            if time.monotonic() > deadline - 2:
                break
            a_final, _ = vectorized_cd_pass(a_final, n_samples=10)
            a_final /= a_final.sum()
            tracker.update(a_final)
    
    # Fallback
    if tracker.best is None:
        tracker.best = np.ones(256)
        tracker.best_score = compute_J(tracker.best)
    
    print(f"\n{'='*50} FINAL RESULT {'='*50}", flush=True)
    print(f"J = {tracker.best_score:.10f}, n = {tracker.best.size}", flush=True)
    print(f"Total time: {time.monotonic()-tracker.t_start:.1f}s", flush=True)
    
    return [float(x) for x in tracker.best]
