def fast_conv(a):

    while s < 2 * n - 1:
        s <<= 1
    fa = np.fft.rfft(a, s)
    return np.fft.irfft(fa * fa, s)[:2 * n - 1]

import numpy as np

_A_structure_cache = {}

def fast_conv(a):
    n = len(a)
    s = 1
    while s < 2 * n - 1:
        s <<= 1
    fa = np.fft.rfft(a, s)
    return np.fft.irfft(fa * fa, s)[:2 * n - 1]

def compute_J(a):
    a = np.maximum(a, 0)
    s = float(a.sum())
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))

def get_A_structure(n):
    if n not in _A_structure_cache:
        rows = []
        cols = []
        for k in range(2 * n - 1):
            jl = max(0, k - n + 1)
            jh = min(n - 1, k)
            for j in range(jl, jh + 1):
                rows.append(k)
                cols.append(j)

def build_A_sparse(a):
    n = len(a)
    rows_arr, cols_arr = get_A_structure(n)
    vals = a[rows_arr - cols_arr]
    return csr_matrix((vals, (rows_arr, cols_arr)), shape=(2 * n - 1, n))

def lp_solve(a, M, method='highs'):
    """Solve LP: max sum(g) s.t. (a*g)_k <= M, g >= 0. Returns g/sum(g) or None."""
    n = len(a)
    c = -np.ones(n)
    A = build_A_sparse(a)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method=method)
    if r.success and r.x is not None:
        g = np.maximum(r.x, 0.0)
        g_sum = g.sum()
        if g_sum > 1e-10:
            return g / g_sum, r
    return None, None

def backtracking_search(a, g, J_cur, alpha0=1.0, decay=0.7, c1=1e-4, max_iter=20):
    """Simple backtracking line search. Returns (a_new, J_new)."""
    alpha = alpha0
    for _ in range(max_iter):
        a_try = (1 - alpha) * a + alpha * g
        a_try = np.maximum(a_try, 0)
        s = a_try.sum()
        if s < 1e-10:
            alpha *= decay
            continue
        a_try = a_try / s
        J_try = compute_J(a_try)
        if J_try < J_cur - c1 * alpha * (J_cur - 0):  # Simple: just want improvement
            return a_try, J_try
        alpha *= decay
    return a, J_cur

def smoothed_max_and_grad(a, beta):
    """Compute (1/beta)log(sum exp(beta*b_k)) and its gradient."""
    n = len(a)
    b = fast_conv(a)
    b_max = float(b.max())
    shifted = beta * (b - b_max)
    shifted = np.clip(shifted, -500, 0)
    exp_s = np.exp(shifted)
    Z = float(exp_s.sum())
    p = exp_s / Z
    smooth_val = b_max + np.log(Z) / beta
    
    # Gradient: d(smooth_val)/da_i = sum_k p_k * 2*a_{k-i} * [0<=k-i<n]
    # = 2 * sum_j a_j * p_{i+j}

def lbfgs_smooth(a0, deadline, betas=[(5, 50), (20, 50), (100, 80), (500, 100)]):

    a = np.maximum(a0.copy(), 1e-8)
    n = len(a)

            break

            x2 = np.maximum(x, 0)

            x2 = np.maximum(x, 0)
            _, g = J_and_grad(x2, beta)
            g = np.nan_to_num(g)
            return g

                         bounds=[(0, None)]*n,

            if res.fun < 1e10:

            pass
