import numpy as np
from scipy.optimize import linprog
from scipy.sparse import csr_matrix
import time


def fast_conv(a):
    """Full self-convolution via FFT."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]


def compute_J(a):
    """Compute J(a) = 2*n*max(conv(a,a))/(sum(a))^2."""
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))


def build_A_sparse(n, a):
    """Build sparse constraint matrix for LP: A[k,j] = a[k-j] if valid."""
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        j_min = max(0, k - n + 1)
        j_max = min(n - 1, k)
        for j in range(j_min, j_max + 1):
            rows.append(k)
            cols.append(j)
            vals.append(a[k - j])
    A = csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))
    return A


def solve_lp(a, M, time_limit=5.0):
    """LP: max sum(g) s.t. (a*g)[k] <= M for all k, g >= 0."""
    n = len(a)
    A = build_A_sparse(n, a)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    try:
        r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                    options={'time_limit': time_limit, 'presolve': True})
        if r.success and r.x is not None:
            g = np.maximum(r.x, 0)
            if g.sum() > 1e-15:
                return g
    except Exception:
        pass
    return None


def lp_mirror_descent_enhanced(a0, max_iters, deadline, shrink_factor=1.0,
                                min_shrink=0.95, verbose=False):
    """
    Enhanced LP mirror descent with adaptive M-shrinking.
    
    At each iteration:
    1. Compute M = max conv(a,a)
    2. Try to solve LP with constraint conv <= M * shrink_factor
    3. If infeasible, use M directly
    4. Blend with backtracking line search
    """
    a = np.maximum(a0.astype(np.float64), 1e-10)
    a = a / a.sum()
    n = len(a)
    
    best_a = a.copy()
    best_M = float(fast_conv(a).max())
    no_improve_count = 0
    current_shrink = shrink_factor
    
    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        
        b = fast_conv(a)
        M = float(b.max())
        
        if M < best_M * (1 - 1e-10):
            best_M = M
            best_a = a.copy()
            no_improve_count = 0
        else:
            no_improve_count += 1
        
        if no_improve_count > 100:
            break
        
        # Try to shrink M for the LP constraint
        M_target = M * current_shrink
        g = solve_lp(a, M_target, time_limit=3.0)
        
        if g is None:
            # Fallback: use M directly
            g = solve_lp(a, M, time_limit=3.0)
            if g is None:
                # Reduce shrink factor (be more conservative)
                current_shrink = max(min_shrink, current_shrink + 0.01)
                continue
        
        g = g / g.sum()
        
        # Backtracking line search
        improved = False
        t = 0.5
        for _ in range(10):
            a_trial = (1 - t) * a + t * g
            a_trial = np.maximum(a_trial, 0.0)
            s = a_trial.sum()
            if s < 1e-10:
                t *= 0.5
                continue
            a_trial = a_trial / s
            M_trial = float(fast_conv(a_trial).max())
            
            if M_trial < M - 1e-14:
                a = a_trial
                if M_trial < best_M:
                    best_M = M_trial
                    best_a = a.copy()
                improved = True
                # Try to increase shrink for next iteration
                current_shrink = max(min_shrink, current_shrink - 0.005)
                break
            t *= 0.5
        
        if not improved:
            # Try smaller shrink
            current_shrink = min(1.0, current_shrink + 0.02)
            if current_shrink >= 1.0 and no_improve_count > 50:
                break

    if verbose:
        print(f"    LP-MD done: M={best_M:.10f} after {it+1} iters", flush=True)
    
    return best_a


def coordinate_descent_polish(a0, max_sweeps, deadline):
    """
    Coordinate descent: for each coordinate i, find the optimal value
    that minimizes the max convolution coefficient.
    """
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-10:
        return a
    a = a / s
    n = len(a)
    
    best_a = a.copy()
    best_M = float(fast_conv(a).max())
    
    for sweep in range(max_sweeps):
        if time.monotonic() > deadline:
            break
        
        improved_this_sweep = False
        # Cycle through coordinates (randomized order for diversity)
        order = np.random.permutation(n)
        
        for i in order:
            if time.monotonic() > deadline:
                break
            
            old_val = a[i]
            if old_val < 1e-15 and fast_conv(a).max() < best_M * 0.99:
                continue
            
            # Binary search for optimal value of a[i]
            # We want to minimize max(conv(a,a)) after changing a[i]
            # conv changes only at positions involving i:
            # conv[k] changes by 2*a[i]*a[k-i] (for the cross terms)
            # More precisely: new_conv[k] = old_conv[k] + 2*delta*a[k-i] - delta^2*(1 if k-i==i else 0)
            # where delta = new_val - old_val
            
            # Try a few candidate values
            candidates = []
            if old_val > 1e-10:
                candidates.append(old_val * 0.5)
                candidates.append(old_val * 0.75)
                candidates.append(old_val * 1.25)
                candidates.append(old_val * 1.5)
                candidates.append(0.0)
            else:
                base_sum = a.sum() - old_val
                candidates.append(base_sum / n * 0.5)
                candidates.append(base_sum / n)
                candidates.append(base_sum / n * 2)
            
            best_val = old_val
            best_M_local = fast_conv(a).max()
            
            for val in candidates:
                val = max(0.0, val)
                a[i] = val
                s_new = a.sum()
                if s_new < 1e-10:
                    a[i] = old_val
                    continue
                # We don't renormalize during CD (work in the original scale)
                M_local = float(fast_conv(a).max()) / (s_new * s_new)
                if M_local < best_M_local - 1e-14:
                    best_M_local = M_local
                    best_val = val
                a[i] = val  # restore for next candidate
            
            a[i] = best_val
            s_new = a.sum()
            if s_new > 1e-10:
                M_check = float(fast_conv(a).max()) / (s_new * s_new)
                if M_check < best_M / (2 * n):  # Compare normalized
                    pass
                # Actually let's just compare the raw max after normalization
                a_norm = a / a.sum()
                M_norm = float(fast_conv(a_norm).max())
                if M_norm < best_M - 1e-12:
                    best_M = M_norm
                    best_a = a.copy()
                    improved_this_sweep = True
        
        if not improved_this_sweep:
            # One more sweep with finer search
            for i in range(n):
                if time.monotonic() > deadline:
                    break
                old_val = a[i]
                # Fine binary search around current value
                lo = max(0.0, old_val - 0.1 * (1 + old_val))
                hi = old_val + 0.1 * (1 + old_val)
                
                for _ in range(5):
                    mid = (lo + hi) / 2
                    a[i] = mid
                    a_norm = a / a.sum()
                    M_mid = float(fast_conv(a_norm).max())
                    a[i] = (lo + mid) / 2
                    a_norm2 = a / a.sum()
                    M_lo = float(fast_conv(a_norm2).max())
                    a[i] = (mid + hi) / 2
                    a_norm3 = a / a.sum()
                    M_hi = float(fast_conv(a_norm3).max())
                    
                    if M_mid <= M_lo and M_mid <= M_hi:
                        a[i] = mid
                        break
                    elif M_lo < M_hi:
                        hi = mid
                    else:
                        lo = mid
                else:
                    a[i] = (lo + hi) / 2
            
            break
    
    a_final = np.maximum(best_a, 0.0)
    s = a_final.sum()
    if s > 1e-10:
        a_final = a_final / s
    return a_final


def smooth_gradient_step(a, tau, lr):
    """One step of smooth gradient descent on the log-sum-exp proxy."""
    n = len(a)
    b = fast_conv(a)
    b_max = b.max()
    exp_b = np.exp(tau * (b - b_max))
    p = exp_b / exp_b.sum()
    
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        pk = p[k]
        if pk < 1e-14:
            continue
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        grad[lo:hi + 1] += 2.0 * pk * a[k - hi:k - lo + 1][::-1]
    
    a_new = a - lr * grad
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s < 1e-10:
        return a
    return a_new / s


def quick_smooth_optimize(a0, n_iters, deadline, tau_start=5.0, tau_end=60.0,
                           lr=0.15):
    """Quick smooth gradient descent."""
    a = np.maximum(a0.astype(np.float64), 1e-8)
    a = a / a.sum()
    best_a = a.copy()
    best_M = float(fast_conv(a).max())
    
    for it in range(n_iters):
        if time.monotonic() > deadline:
            break
        frac = it / max(1, n_iters)
        tau = tau_start + (tau_end - tau_start) * frac
        lr_t = lr * (1 - 0.5 * frac)
        a = smooth_gradient_step(a, tau, lr_t)
        M = float(fast_conv(a).max())
        if M < best_M:
            best_M = M
            best_a = a.copy()
    
    return best_a


def resample_seq(src, dest_n):
    """Resample 1D sequence to new length via linear interpolation."""
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)


def generate_diverse_inits(n, rng, parent=None):
    """Generate diverse initial sequences."""
    inits = []
    t = np.linspace(0, 1, n, endpoint=False)
    
    if parent is not None:
        p = resample_seq(parent, n)
        inits.append(p)
        inits.append(np.maximum(p + rng.normal(0, 0.02, n), 0.01))
        inits.append(np.maximum(p + rng.normal(0, 0.08, n), 0.01))
        # Reversed
        inits.append(p[::-1])
    
    # Uniform
    inits.append(np.ones(n))
    # Random
    inits.append(rng.uniform(0.3, 1.0, n))
    # Cosine waves
    for f in [1, 2, 3, 5]:
        inits.append(np.maximum(1.0 + 0.4 * np.cos(2 * np.pi * f * t), 0.01))
        inits.append(np.maximum(1.0 + 0.3 * np.sin(2 * np.pi * f * t + 0.5), 0.01))
    # Gaussian
    inits.append(np.exp(-4.0 * (t - 0.5) ** 2) + 0.05)
    # Two bumps
    inits.append(np.exp(-8 * (t - 0.3) ** 2) + np.exp(-8 * (t - 0.7) ** 2) + 0.05)
    # Linear ramp
    inits.append(0.3 + 0.7 * t)
    inits.append(0.3 + 0.7 * (1 - t))
    # Quadratic
    inits.append(np.maximum(0.2 + 0.8 * (1 - (2*t - 1)**2), 0.01))
    # Smooth random
    k = max(3, n // 30)
    kern = np.ones(k) / k
    sr = np.convolve(rng.uniform(0, 1, n), kern, mode='same')
    inits.append(np.maximum(sr + 0.1, 0.01))
    # Block
    nb = max(3, n // 50)
    bv = rng.uniform(0.3, 1.0, nb)
    inits.append(np.repeat(bv, n // nb + 1)[:n])
    
    # Normalize all
    result = []
    for x in inits:
        x = np.maximum(np.asarray(x, dtype=np.float64), 1e-6)
        x = x / x.sum()
        result.append(x)
    return result


class Tracker:
    def __init__(self):
        self.best = None
        self.best_score = float('inf')
        self.t0 = time.monotonic()
    
    def update(self, a):
        a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
        if a.size == 0 or a.sum() < 0.01:
            return
        score = float(evaluate_sequence(a.tolist()))
        if np.isfinite(score) and score < self.best_score:
            self.best = a.copy()
            self.best_score = score
            elapsed = time.monotonic() - self.t0
            print(f"  ** NEW BEST: J={score:.8f} n={a.size} t={elapsed:.0f}s", flush=True)
    
    def report(self, label=""):
        if self.best is not None:
            print(f"  {label} best: J={self.best_score:.8f} n={self.best.size}", flush=True)


def propose_candidate(seed=42, budget_s=1500, **kwargs):
    """
    Multi-phase search with LP mirror descent as primary optimizer.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + float(budget_s) - 20.0
    tracker = Tracker()
    
    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        if p.ndim == 1 and p.size > 0 and np.all(np.isfinite(p)):
            parent = np.maximum(p, 0.0)
            tracker.update(parent)
            print(f"Parent: n={parent.size}, J={tracker.best_score:.6f}", flush=True)
    except (NameError, AttributeError):
        pass
    
    if parent is None:
        parent = np.ones(256)
        tracker.update(parent)
    
    current_best = parent.copy()
    
    # === PHASE 1: Multi-length exploration with quick smooth optimization ===
    print(f"\n{'='*60}", flush=True)
    print("PHASE 1: Multi-length smooth exploration", flush=True)
    print(f"{'='*60}", flush=True)
    
    p1_start = time.monotonic()
    p1_budget = 300.0
    
    lengths = [256, 320, 384, 448, 512, 576, 640, 768, 896, 1024]
    lengths = [n for n in lengths if 128 <= n <= 1200]
    
    candidates = []  # List of (a, J_score)
    
    for n in lengths:
        if time.monotonic() > deadline - 400:
            break
        time_per_n = min(40.0, p1_budget * (512.0 / n) / len(lengths))
        phase_dl = min(time.monotonic() + time_per_n, deadline - 350)
        
        if time.monotonic() >= phase_dl:
            continue
        
        inits = generate_diverse_inits(n, rng, current_best)
        best_for_n = None
        best_M_for_n = float('inf')
        
        for a0 in inits:
            if time.monotonic() > phase_dl:
                break
            a = quick_smooth_optimize(a0, 80, phase_dl, tau_start=3, tau_end=40, lr=0.12)
            M_val = float(fast_conv(a).max())
            if M_val < best_M_for_n:
                best_M_for_n = M_val
                best_for_n = a.copy()
        
        if best_for_n is not None:
            J_val = compute_J(best_for_n)
            candidates.append((best_for_n, J_val))
            print(f"  n={n:4d}: J={J_val:.6f} ({time.monotonic()-p1_start:.0f}s)", flush=True)
            tracker.update(best_for_n)
            if J_val < compute_J(current_best):
                current_best = best_for_n.copy()
    
    print(f"Phase 1 time: {time.monotonic()-p1_start:.0f}s", flush=True)
    
    # === PHASE 2: LP Mirror Descent on top candidates ===
    print(f"\n{'='*60}", flush=True)
    print("PHASE 2: LP Mirror Descent (primary optimizer)", flush=True)
    print(f"{'='*60}", flush=True)
    
    p2_start = time.monotonic()
    
    # Sort candidates by score
    candidates.sort(key=lambda x: x[1])
    
    # Also add the parent at its original length
    candidates.append((parent.copy(), compute_J(parent)))
    candidates.sort(key=lambda x: x[1])
    
    # Run LP on top 5 candidates (or fewer if time is tight)
    max_lp_runs = 5
    lp_time_per_run = 300.0
    
    for idx, (a_cand, J_cand) in enumerate(candidates[:max_lp_runs]):
        if time.monotonic() > deadline - 150:
            break
        
        n_cand = a_cand.size
        # Estimate time per LP iteration
        est_iter_time = 0.005 * (n_cand / 128) ** 2
        max_iters = max(20, int(lp_time_per_run / max(0.01, est_iter_time)))
        
        print(f"  LP run {idx+1}: n={n_cand}, initial J={J_cand:.6f}, max_iters={max_iters}", flush=True)
        
        a_opt = lp_mirror_descent_enhanced(a_cand, max_iters, min(time.monotonic() + lp_time_per_run, deadline - 100))
        J_opt = compute_J(a_opt)
        print(f"  LP run {idx+1} result: J={J_opt:.6f}", flush=True)
        tracker.update(a_opt)
        
        if J_opt < compute_J(current_best):
            current_best = a_opt.copy()
    
    tracker.report("Phase 2")
    print(f"Phase 2 time: {time.monotonic()-p2_start:.0f}s", flush=True)
    
    # === PHASE 3: Coordinate Descent Polishing ===
    print(f"\n{'='*60}", flush=True)
    print("PHASE 3: Coordinate Descent Polishing", flush=True)
    print(f"{'='*60}", flush=True)
    
    p3_start = time.monotonic()
    
    if tracker.best is not None:
        # Run CD on the best
        a_cd = coordinate_descent_polish(tracker.best, max_sweeps=30,
                                          deadline=min(time.monotonic() + 120, deadline - 80))
        tracker.update(a_cd)
        
        # Also try on current_best if different
        if not np.allclose(tracker.best, current_best, atol=1e-6):
            a_cd2 = coordinate_descent_polish(current_best, max_sweeps=20,
                                               deadline=min(time.monotonic() + 60, deadline - 60))
            tracker.update(a_cd2)
    
    tracker.report("Phase 3")
    print(f"Phase 3 time: {time.monotonic()-p3_start:.0f}s", flush=True)
    
    # === PHASE 4: SA-style perturbation + re-optimize ===
    print(f"\n{'='*60}", flush=True)
    print("PHASE 4: Perturbation + Re-optimize", flush=True)
    print(f"{'='*60}", flush=True)
    
    p4_start = time.monotonic()
    
    if tracker.best is not None:
        n_best = tracker.best.size
        temperature = 0.03
        n_restarts = 30
        
        for r in range(n_restarts):
            if time.monotonic() > deadline - 40:
                break
            
            a = tracker.best.copy().astype(np.float64)
            noise = rng.normal(0, temperature * 0.05, n_best)
            a = a + noise
            a = np.maximum(a, 0.0)
            s = a.sum()
            if s < 1e-10:
                continue
            a = a / s
            
            # Quick smooth + short LP
            a = quick_smooth_optimize(a, 30, deadline - 40, tau_start=5, tau_end=30, lr=0.1)
            tracker.update(a)
            
            # Short LP if time allows
            time_left = deadline - 40 - time.monotonic()
            if time_left > 10:
                est_iter_time = 0.005 * (n_best / 128) ** 2
                max_it = max(5, min(30, int(time_left / max(0.01, est_iter_time))))
                a = lp_mirror_descent_enhanced(a, max_it, deadline - 40)
                tracker.update(a)
            
            temperature *= 0.93
    
    tracker.report("Phase 4")
    print(f"Phase 4 time: {time.monotonic()-p4_start:.0f}s", flush=True)
    
    # === PHASE 5: Length extension + final LP ===
    print(f"\n{'='*60}", flush=True)
    print("PHASE 5: Length Extension + Final Polish", flush=True)
    print(f"{'='*60}", flush=True)
    
    p5_start = time.monotonic()
    
    if tracker.best is not None:
        base_n = tracker.best.size
        
        # Try extending to 2x length
        if time.monotonic() < deadline - 60:
            new_n = min(base_n * 2, 1500)
            if new_n > base_n:
                a_ext = resample_seq(tracker.best, new_n)
                a_ext = np.maximum(a_ext, 1e-6)
                a_ext = a_ext / a_ext.sum()
                a_ext = quick_smooth_optimize(a_ext, 50, deadline - 30,
                                              tau_start=3, tau_end=25, lr=0.1)
                tracker.update(a_ext)
                
                if time.monotonic() < deadline - 20:
                    a_ext = lp_mirror_descent_enhanced(a_ext, 20, deadline - 15)
                    tracker.update(a_ext)
        
        # Final LP polish on the absolute best
        if time.monotonic() < deadline - 5:
            time_left = deadline - 5 - time.monotonic()
            n_final = tracker.best.size
            est_iter_time = 0.005 * (n_final / 128) ** 2
            max_it = max(10, int(time_left / max(0.01, est_iter_time)))
            a_final = lp_mirror_descent_enhanced(tracker.best, max_it, deadline - 2)
            tracker.update(a_final)
    
    tracker.report("Phase 5")
    print(f"Phase 5 time: {time.monotonic()-p5_start:.0f}s", flush=True)
    
    # Fallback
    if tracker.best is None:
        tracker.best = np.ones(256)
        tracker.best_score = compute_J(tracker.best)
    
    print(f"\n{'='*60}", flush=True)
    print(f"FINAL: J={tracker.best_score:.8f} n={tracker.best.size} "
          f"time={time.monotonic()-tracker.t0:.0f}s", flush=True)
    print(f"{'='*60}", flush=True)
    
    return [float(x) for x in tracker.best]
