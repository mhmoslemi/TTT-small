    n = len(v)
    mu = np.zeros(n)
    total_pos = 0.0

Core idea: alternate between computing softmax weights on convolution values

import numpy as np
from scipy.optimize import minimize
from scipy.fft import rfft, irfft
import time

def compute_conv(a):
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    conv = irfft(fa * fa, size)
    return conv[:2 * n - 1]

def compute_obj(a):
    n = len(a)
    if n == 0:
        return np.inf
    conv = compute_conv(a)
    max_b = float(np.max(conv))
    sum_a = float(np.sum(a))
    if sum_a < 0.01:
        return np.inf
    return 2.0 * n * max_b / (sum_a * sum_a)

def softmax_weights(conv, tau):
    """Compute softmax weights from convolution values."""
    cmax = np.max(conv)
    shifted = (conv - cmax) / tau
    w = np.exp(np.clip(shifted, -100, 100))
    s = np.sum(w)
    if s < 1e-300:
        idx = np.argmax(conv)
        w = np.zeros(len(conv))
        w[idx] = 1.0
    else:
        w /= s
    return w

def proj_simplex(v):
    """Project v onto the probability simplex."""
    n = len(v)
    v_sorted = np.sort(v)[::-1]
    cssv = np.cumsum(v_sorted)
    rho = np.nonzero(v_sorted * np.arange(1, n + 1) > (cssv - 1))[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho = rho[-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0)

"""
Iterative QP reweighting + LP alternation + L-BFGS-B for minimizing
2*n*max(conv(a,a))/sum(a)^2.

Core new idea: Uses softmax-reweighted QP (min a^T W a s.t. simplex)
as the primary optimization driver, where W encodes the convolution
weights. This provides a convex subproblem at each iteration that is
more focused on actual peaks than log-sum-exp smoothing. Combined with
LP alternation for global progress and L-BFGS-B for local refinement.

import numpy as np
from scipy.optimize import minimize, linprog
from scipy.fft import rfft, irfft
import time


def compute_conv(a):
    """Compute linear convolution a*a using FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 1:
        return np.array([a[0] * a[0]])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    conv = irfft(fa * fa, size)
    return conv[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    conv = compute_conv(a)
    max_b = float(np.max(conv))
    sum_a = float(np.sum(a))
    if sum_a < 0.01:
        return np.inf
    return 2.0 * n * max_b / (sum_a * sum_a)


def fft_convolve(a, b):
    """Linear convolution of a and b using FFT."""
    n, m = len(a), len(b)
    size = 1
    while size < n + m - 1:
        size <<= 1
    fa = rfft(a, size)
    fb = rfft(b, size)
    return irfft(fa * fb, size)[:n + m - 1]


def softmax_weights(conv, tau):
    """Compute softmax weights from convolution values."""
    cmax = float(np.max(conv))
    shifted = (conv - cmax) / tau
    w = np.exp(np.clip(shifted, -500, 500))
    s = np.sum(w)
    if s < 1e-300:
        idx = int(np.argmax(conv))
        w = np.zeros(len(conv))
        w[idx] = 1.0
    else:
        w /= s
    return w


def proj_simplex(v):
    """Project v onto the probability simplex {x: x>=0, sum(x)=1}."""
    n = len(v)
    v_sorted = np.sort(v)[::-1]
    cssv = np.cumsum(v_sorted)
    idx = np.arange(1, n + 1)
    rho = np.nonzero(v_sorted * idx > (cssv - 1))[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho = rho[-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0)


def matvec_w(a, w):
    """Compute W @ a where W[i,j] = w[i+j]. Uses FFT for efficiency."""
    n = len(a)
    ar = a[::-1]
    size = 1
    while size < n + len(w) - 1:
        size <<= 1
    fa = rfft(ar, size)
    fw = rfft(w, size)
    conv = irfft(fa * fw, size)
    return conv[n - 1:2 * n - 1]


def grad_max_conv(a, w):
    """Compute gradient of weighted max: 2 * sum_k w[k] * a[k-i]."""
    n = len(a)
    w_rev = w[::-1]
    conv_fw = fft_convolve(a, w_rev)
    return 2.0 * conv_fw[n - 1:2 * n - 1][::-1]


def quad_form(a, w):
    """Compute a^T W a where W[i,j] = w[i+j]."""
    return float(np.dot(a, matvec_w(a, w)))


def solve_qp_projected(a, w, max_iter=200):
    """
    Solve min a^T W a s.t. sum(a)=1, a>=0 using projected gradient
    with backtracking line search.
    """
    n = len(a)
    a = a.copy().astype(np.float64)
    s = np.sum(a)
    if s < 1e-10:
        a = np.ones(n) / n
    else:
        a = a / s

    # Initial step size based on Lipschitz constant estimate
    # ||W||_2 <= sum(w) * n (rough upper bound)
    step = 1.0 / (n * np.sum(w) * 2.0)
    step = max(step, 1e-6)
    step = min(step, 0.1)

    best_a = a.copy()
    best_qobj = quad_form(a, w)

    for it in range(max_iter):
        grad = 2.0 * matvec_w(a, w)
        a_new = a - step * grad
        a_new = proj_simplex(a_new)
        qobj = quad_form(a_new, w)

        if qobj < best_qobj:
            best_qobj = qobj
            best_a = a_new.copy()

        # Adaptive step
        if qobj < best_qobj * 0.999:
            step = min(step * 1.08, 0.2 / n)
        else:
            step *= 0.5
            if step < 1e-12:
                break
            a_new = a - step * grad
            a_new = proj_simplex(a_new)
            if quad_form(a_new, w) >= best_qobj:

        a = a_new if qobj <= best_qobj else a

def build_toeplitz(a):
    """Build Toeplitz matrix A where (A@g)[k] = (conv(a,g))[k]."""
    n = len(a)
    A = np.zeros((2 * n - 1, n), dtype=np.float64)
    for j in range(n):
        k_end = min(j + n, 2 * n - 1)
        A[j:k_end, j] = a[:k_end - j]
    return A


def solve_lp_g(a, M, time_limit=20):
    """Solve LP: max sum(g) s.t. (a*g)[k] <= M for all k, g >= 0."""
    n = len(a)
    A = build_toeplitz(a)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': time_limit})
        if res.status == 0:
            g = np.maximum(res.x, 0)
            s = float(np.sum(g))
            if s > 1e-10:
                return g / s, s
    except Exception:
        pass
    return None, None


def solve_lp_minpeak(a, time_limit=20):

    c = np.zeros(n + 1)

    A_ub[:, n] = -1.0

    A_eq = np.zeros((1, n + 1))

def lbfgs_optimize(a, tau_list, max_iter=200, time_budget=20.0):
    """L-BFGS-B with log-sum-exp smoothing and softmax parametrization."""
    n = len(a)
    a = a.copy().astype(np.float64)
    s = np.sum(a)
    if s < 1e-10:
        a = np.ones(n) / n
    else:
        a = a / s
    a = np.maximum(a, 1e-30)
    z = np.log(a)

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    for tau in tau_list:
        if time.monotonic() - start > time_budget:
            break

        def func_and_grad(z_val):
            n_local = len(z_val)
            z_val = z_val - np.max(z_val)
            e = np.exp(z_val)
            a_local = e / np.sum(e)

            conv = compute_conv(a_local)
            cmax = float(np.max(conv))
            shifted = np.clip((conv - cmax) / tau, -500, 500)
            exp_conv = np.exp(shifted)
            Z = float(np.sum(exp_conv))
            soft_max_val = cmax + tau * np.log(Z)
            w = exp_conv / Z

            w_rev = w[::-1]
            conv_fw = fft_convolve(a_local, w_rev)
            grad_a = 2.0 * conv_fw[n_local - 1:2 * n_local - 1][::-1]

            mean_grad = float(np.dot(a_local, grad_a))
            grad_z = a_local * (grad_a - mean_grad)

            return soft_max_val, grad_z

        try:
            result = minimize(func_and_grad, z, jac=True, method='L-BFGS-B',
                              options={'maxiter': max_iter, 'ftol': 1e-14, 'gtol': 1e-8})
            z_new = result.x
            a_new = np.exp(z_new - np.max(z_new))
            a_new = a_new / np.sum(a_new)
            a_new = np.maximum(a_new, 0)
            obj = compute_obj(a_new)
            if obj < best_obj:
                best_obj = obj
                best_a = a_new.copy()
                z = z_new
        except Exception:
            continue

    return best_a, best_obj

        tau = tau_start + (tau_end - tau_start) * frac

        conv = compute_conv(a)
        w = softmax_weights(conv, tau)
        grad = grad_max_conv(a, w)
        grad_centered = grad - np.mean(grad)

        a_new = a - step * grad_centered
        a_new = np.maximum(a_new, 1e-15)
        s_new = np.sum(a_new)
        if s_new < 1e-10:
            a_new = a.copy()
        else:
            a_new = a_new / s_new

        obj = compute_obj(a_new)

        if obj < best_obj - 1e-12:
            best_obj = obj
            best_a = a_new.copy()
            a = a_new
            stall = 0
        else:
            stall += 1
            if stall > 80:
                step *= 0.7
                if step < 1e-10:
                    break
                rng = np.random.default_rng(it * 31 + 7)
                a = a + rng.normal(0, 0.02, n) * step
                a = np.maximum(a, 1e-15)
                a = a / np.sum(a)
                stall = 0

        if it > 100:
            step *= 0.997

    return best_a, best_obj


def generate_init(n, seed, type_idx=None):
    """Generate diverse random initializations."""
    rng = np.random.default_rng(seed)
    k = np.arange(n)
    if type_idx is None:
        type_idx = rng.integers(0, 12)
    type_idx = type_idx % 12

    if type_idx == 0:
        f = rng.uniform(0.5, 4.0)
        phase = rng.uniform(0, 2 * np.pi)
        a = 1.0 + 0.7 * np.cos(2 * np.pi * f * k / n + phase)
    elif type_idx == 1:
        f1 = rng.uniform(0.3, 2.0)
        f2 = rng.uniform(2.0, 8.0)
        a = 1.0 + 0.3 * np.cos(2 * np.pi * f1 * k / n) + 0.3 * np.cos(2 * np.pi * f2 * k / n)
    elif type_idx == 2:
        # U-shape
        center = n / 2.0
        a = 1.0 + 0.8 * ((k - center) / center) ** 2
    elif type_idx == 3:
        # W-shape
        a = 1.0 + 0.5 * np.cos(4 * np.pi * k / n) + 0.3 * np.cos(8 * np.pi * k / n)
    elif type_idx == 4:
        a = rng.random(n)
        win = max(3, n // 15)
        kernel = np.ones(win) / win
        a = np.convolve(a, kernel, mode='same')
    elif type_idx == 5:
        # Multi-frequency
        a = np.ones(n)
        for _ in range(rng.integers(2, 6)):
            f = rng.uniform(0.5, 6.0)
            amp = rng.uniform(0.1, 0.4)
            a += amp * np.cos(2 * np.pi * f * k / n + rng.uniform(0, 2 * np.pi))
    elif type_idx == 6:
        # Bimodal
        c1 = n * rng.uniform(0.15, 0.35)
        c2 = n - c1
        w1 = n * rng.uniform(0.05, 0.15)
        w2 = w1
        a = np.exp(-0.5 * ((k - c1) / w1) ** 2) + np.exp(-0.5 * ((k - c2) / w2) ** 2) + 0.3
    elif type_idx == 7:
        # Chirp
        f0 = rng.uniform(1.0, 3.0)
        f1 = rng.uniform(3.0, 8.0)
        a = 1.0 + 0.5 * np.cos(2 * np.pi * (f0 * k / n + (f1 - f0) * k ** 2 / (2 * n ** 2)))
    elif type_idx == 8:
        # Inverted U
        center = n / 2.0
        a = 1.5 - 0.8 * ((k - center) / center) ** 2
    elif type_idx == 9:
        # Random with spectral shaping
        a = rng.standard_normal(n)
        A = rfft(a)
        # Flatten spectrum
        mag = np.abs(A)
        target = np.ones_like(mag)
        A_new = A / (mag + 1e-10) * target
        a = irfft(A_new, n * 2)[:n]
        a = np.abs(a) + 0.1
    elif type_idx == 10:
        # Gaussian mixture
        comps = rng.integers(2, 5)
        a = np.zeros(n)
        for _ in range(comps):
            c = n * rng.uniform(0.1, 0.9)
            w = n * rng.uniform(0.05, 0.2)
            amp = rng.uniform(0.5, 1.5)
            a += amp * np.exp(-0.5 * ((k - c) / w) ** 2)
        a += 0.2
    else:
        # Random smooth
        a = rng.random(n)
        win = max(5, n // 10)
        kernel = np.ones(win) / win
        a = np.convolve(a, kernel, mode='same') + 0.1

    a = np.maximum(a, 0.01)
    s = np.sum(a)
    return a / s


def resample_sequence(a, new_n):
    """Resample sequence to new length via interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0.001)


def try_parent():
    """Try to load parent sequence."""
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64)
        parent = np.clip(parent, 0, 1000.0)
        if parent.size > 0 and np.isfinite(parent).all():
            obj = compute_obj(parent)
            print(f"Parent (n={parent.size}): obj={obj:.6f}")
            return parent.copy(), obj
    except Exception:
        pass
    return None, np.inf


def polish_sequence(a, time_budget, verbose=False):
    """Intensive polishing of a single sequence using multiple methods."""
    n = len(a)
    a = a.copy().astype(np.float64)
    start = time.monotonic()

    best_a = a.copy()
    best_obj = compute_obj(a)
    round_num = 0

    while time.monotonic() - start < time_budget - 3:
        round_num += 1
        remaining = time_budget - (time.monotonic() - start)
        if remaining < 5:
            break

        # QP reweighting
        if remaining > 20:
            tau_q = 0.5 / n
            a_new, obj_new = qp_reweighting_optimize(
                best_a, [tau_q * 4, tau_q * 2, tau_q],
                max_iter_per_tau=40, time_budget=min(remaining * 0.3, 25.0))
            if obj_new < best_obj:
                best_obj = obj_new
                best_a = a_new.copy()
                if verbose:
                    print(f"    Polish r{round_num} QP: {best_obj:.6f}")

        # LP step
        remaining = time_budget - (time.monotonic() - start)
        if n <= 800 and remaining > 20:
            a_new, obj_new = alternating_lp(best_a, max_iters=25,
                                           time_budget=min(remaining * 0.25, 30.0))
            if obj_new < best_obj:
                best_obj = obj_new
                best_a = a_new.copy()
                if verbose:
                    print(f"    Polish r{round_num} LP: {best_obj:.6f}")

        # L-BFGS-B
        remaining = time_budget - (time.monotonic() - start)
        if remaining > 15:
            tau_sm = 0.2 / n
            a_new, obj_new = lbfgs_optimize(best_a, [tau_sm * 3, tau_sm, tau_sm * 0.3],
                                           max_iter=150, time_budget=min(remaining * 0.2, 15.0))
            if obj_new < best_obj:
                best_obj = obj_new
                best_a = a_new.copy()
                if verbose:
                    print(f"    Polish r{round_num} LBFGS: {best_obj:.6f}")

        # Subgradient
        remaining = time_budget - (time.monotonic() - start)
        if remaining > 10:
            a_new = subgradient_optimize(best_a, max_iters=int(remaining * 8),
                                         time_budget=min(remaining * 0.15, 12.0),
                                         tau_start=5.0, tau_end=25.0)
            obj_new = compute_obj(a_new)
            if obj_new < best_obj:
                best_obj = obj_new
                best_a = a_new.copy()
                if verbose:
                    print(f"    Polish r{round_num} SG: {best_obj:.6f}")

        # Min-peak LP
        remaining = time_budget - (time.monotonic() - start)
        if n <= 600 and remaining > 15:
            g_lp, M_opt = solve_lp_minpeak(best_a, time_limit=10)
            if g_lp is not None:
                for t in [0.3, 0.5, 0.7, 1.0]:
                    a_try = (1 - t) * best_a + t * g_lp
                    a_try = np.maximum(a_try, 0)
                    s_try = np.sum(a_try)
                    if s_try < 1e-10:
                        continue
                    a_try = a_try / s_try
                    obj_try = compute_obj(a_try)
                    if obj_try < best_obj:
                        best_obj = obj_try
                        best_a = a_try.copy()
                        if verbose:
                            print(f"    Polish r{round_num} MinPeak t={t}: {best_obj:.6f}")

    return best_a, best_obj


def propose_candidate(seed=42, budget_s=1500):
    """
    Main search function. Uses QP reweighting as primary optimization,
    combined with LP alternation and L-BFGS-B for polishing.
    Explores multiple sequence lengths and diverse initializations.
    """
    start_time = time.monotonic()
    deadline = start_time + float(budget_s) - 30.0

    best_seq = None
    best_score = np.inf

    # Load parent
    parent, parent_obj = try_parent()
    if parent is not None and parent_obj < best_score:
        best_score = parent_obj
        best_seq = parent.copy()

    # N values to explore - focus around parent length
    n_values = [300, 400, 500, 550, 600, 650, 700, 800, 900, 1000]
    if best_seq is not None:
        np_ = len(best_seq)
        extra = [np_, max(200, np_ - 100), min(1200, np_ + 50),
                 min(1200, np_ + 100), min(1400, np_ + 200)]
        n_values = sorted(set(n_values + extra))

    print(f"Exploring n values: {n_values}")
    print(f"Initial best: {best_score:.6f}")
    print(f"Deadline in {deadline - start_time:.0f}s")

    candidates = []

    # Phase 1: Exploration with QP reweighting
    for n in n_values:
        if time.monotonic() > deadline - 120:
            break
        remaining = deadline - time.monotonic()
        if remaining < 15:
            break

        # Time allocation
        if n <= 400:
            alloc = min(40.0, remaining * 0.12)
        elif n <= 700:
            alloc = min(60.0, remaining * 0.15)
        elif n <= 1000:
            alloc = min(70.0, remaining * 0.18)
        else:
            alloc = min(50.0, remaining * 0.15)
        alloc = max(15.0, alloc)

        print(f"\nN={n}: {remaining:.0f}s remaining, alloc={alloc:.0f}s")

        n_inits = max(2, int(alloc / 18))
        n_inits = min(n_inits, 5)

        for init_idx in range(n_inits):
            if time.monotonic() > deadline - 120:
                break
            remaining_init = deadline - time.monotonic()
            if remaining_init < 12:
                break

            # Generate initialization
            if init_idx == 0 and best_seq is not None:
                a = resample_sequence(best_seq, n)
            elif init_idx == 1 and best_seq is not None:
                # Perturbed parent
                a = resample_sequence(best_seq, n)
                rng = np.random.default_rng(seed + n + 1)
                a = a * np.exp(rng.normal(0, 0.1, n))
                a = np.maximum(a, 0.001)
                a = a / np.sum(a)
            else:
                a = generate_init(n, seed + n * 104729 + init_idx * 79193,
                                  type_idx=init_idx + n % 7)

            obj_init = compute_obj(a)
            print(f"  init{init_idx}: n={n}, obj_init={obj_init:.6f}")

            # Step 1: QP reweighting (main driver)
            qp_time = min((alloc / n_inits) * 0.35, 25.0)
            if remaining_init > 20:
                tau_base = 1.0 / n
                tau_list = [tau_base * 10, tau_base * 5, tau_base * 2, tau_base, tau_base * 0.5]
                a, obj = qp_reweighting_optimize(a, tau_list,
                                                 max_iter_per_tau=60,
                                                 time_budget=qp_time)
            else:
                obj = compute_obj(a)

            # Step 2: Subgradient shaping
            remaining_init = deadline - time.monotonic()
            if remaining_init > 10:
                sg_time = min((alloc / n_inits) * 0.2, 10.0)
                a = subgradient_optimize(a, max_iters=int(sg_time * 20),
                                         time_budget=sg_time,
                                         tau_start=3.0, tau_end=15.0)
                obj = compute_obj(a)

            # Step 3: L-BFGS-B refinement
            remaining_init = deadline - time.monotonic()
            if remaining_init > 15:
                lbfgs_time = min((alloc / n_inits) * 0.2, 15.0)
                tau_lb = 0.5 / n
                a, obj = lbfgs_optimize(a, [tau_lb * 3, tau_lb, tau_lb * 0.4],
                                        max_iter=200, time_budget=lbfgs_time)

            # Step 4: LP polishing
            remaining_init = deadline - time.monotonic()
            if remaining_init > 20 and n <= 800:
                lp_time = min((alloc / n_inits) * 0.25, 25.0)
                a, obj = alternating_lp(a, max_iters=30, time_budget=lp_time)

            # Step 5: Final QP pass
            remaining_init = deadline - time.monotonic()
            if remaining_init > 10:
                tau_final = 0.3 / n
                a, obj = qp_reweighting_optimize(a, [tau_final * 2, tau_final, tau_final * 0.5],
                                                 max_iter_per_tau=40,
                                                 time_budget=min(remaining_init * 0.3, 10.0))

            print(f"  init{init_idx} final: n={n}, obj={obj:.6f}")

            if obj < best_score:
                best_score = obj
                best_seq = a.copy()
                print(f"  ** UPDATE: n={n}, obj={best_score:.6f} **")

            candidates.append((a.copy(), obj, n))

    # Phase 2: Polishing top candidates
    print(f"\n=== POLISHING PHASE ===")
    print(f"Best before polish: {best_score:.6f}")
    print(f"Time remaining: {deadline - time.monotonic():.0f}s")

    candidates.sort(key=lambda x: x[1])

    n_polish = min(4, len(candidates))
    for ci in range(n_polish):
        if time.monotonic() > deadline - 50:
            break
        remaining = deadline - time.monotonic()
        if remaining < 20:
            break

        a_cand = candidates[ci][0]
        n_cand = candidates[ci][2]
        print(f"  Polishing candidate {ci} (n={n_cand}, obj={candidates[ci][1]:.6f}), {remaining:.0f}s left")

        polish_budget = min(remaining * 0.35, 60.0)
        a_pol, obj_pol = polish_sequence(a_cand, time_budget=polish_budget, verbose=True)

        if obj_pol < best_score:
            best_score = obj_pol
            best_seq = a_pol.copy()
            print(f"  ** UPDATE after polish: obj={best_score:.6f} **")
        else:
            # Try perturbation restarts
            remaining = deadline - time.monotonic()
            if remaining > 15:
                rng = np.random.default_rng(ci * 555 + 999)
                for ptry in range(3):
                    if time.monotonic() > deadline - 30:
                        break
                    scale = rng.uniform(0.03, 0.1)
                    a_pert = a_cand * np.exp(rng.normal(0, scale, n_cand))
                    a_pert = np.maximum(a_pert, 0.001)
                    a_pert = a_pert / np.sum(a_pert)

                    # Quick QP + SG
                    tau_p = 0.5 / n_cand
                    a_pert, obj_pert = qp_reweighting_optimize(
                        a_pert, [tau_p * 3, tau_p], max_iter_per_tau=30, time_budget=5.0)
                    a_pert = subgradient_optimize(a_pert, max_iters=300, time_budget=4.0,
                                                   tau_start=5.0, tau_end=20.0)
                    obj_pert = compute_obj(a_pert)

                    if obj_pert < best_score:
                        best_score = obj_pert
                        best_seq = a_pert.copy()
                        print(f"  ** UPDATE (perturb {ptry}): obj={best_score:.6f} **")

    # Phase 3: Final intensive polish
    if best_seq is not None and time.monotonic() < deadline - 15:
        remaining = deadline - time.monotonic()
        n_best = len(best_seq)
        print(f"\n=== FINAL POLISH: n={n_best}, {remaining:.0f}s ===")

        a_cur = best_seq.copy()
        for fr in range(5):
            if time.monotonic() > deadline - 10:
                break
            remaining = deadline - time.monotonic()
            if remaining < 10:
                break

            # QP with tiny tau
            tau_tiny = 0.1 / n_best
            a_new, obj_new = qp_reweighting_optimize(
                a_cur, [tau_tiny * 3, tau_tiny, tau_tiny * 0.4],
                max_iter_per_tau=50, time_budget=min(remaining * 0.2, 15.0))
            if obj_new < best_score:
                best_score = obj_new
                best_seq = a_new.copy()
                a_cur = a_new.copy()
                print(f"  Final QP {fr}: {best_score:.6f}")

            # LP
            remaining = deadline - time.monotonic()
            if remaining > 15 and n_best <= 800:
                a_new, obj_new = alternating_lp(a_cur, max_iters=25,
                                               time_budget=min(remaining * 0.25, 20.0))
                if obj_new < best_score:
                    best_score = obj_new
                    best_seq = a_new.copy()
                    a_cur = a_new.copy()
                    print(f"  Final LP {fr}: {best_score:.6f}")

            # L-BFGS-B
            remaining = deadline - time.monotonic()
            if remaining > 10:
                tau_v = 0.05 / n_best
                a_new, obj_new = lbfgs_optimize(a_cur, [tau_v * 5, tau_v, tau_v * 0.3],
                                               max_iter=300, time_budget=min(remaining * 0.2, 15.0))
                if obj_new < best_score:
                    best_score = obj_new
                    best_seq = a_new.copy()
                    a_cur = a_new.copy()
                    print(f"  Final LBFGS {fr}: {best_score:.6f}")

            # Subgradient
            remaining = deadline - time.monotonic()
            if remaining > 8:
                a_new = subgradient_optimize(a_cur, max_iters=int(remaining * 10),
                                             time_budget=min(remaining * 0.15, 10.0),
                                             tau_start=10.0, tau_end=40.0)
                obj_new = compute_obj(a_new)
                if obj_new < best_score:
                    best_score = obj_new
                    best_seq = a_new.copy()
                    a_cur = a_new.copy()
                    print(f"  Final SG {fr}: {best_score:.6f}")

    # Fallback
    if best_seq is None:
        best_seq = np.ones(200)
        best_score = compute_obj(best_seq)

    # Final validation
    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    val = evaluate_sequence(result)
    if val < np.inf:
        best_score = val
    else:
        result = [1.0] * 100
        best_score = 2.0

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_score:.6f}, time={elapsed:.1f}s ===")

    return result
