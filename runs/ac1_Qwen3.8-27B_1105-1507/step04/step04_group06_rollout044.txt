```python
    """Reduce peaks in convolution by redistributing mass."""
    n = len(a)
    b = fast_conv(a)
    
    # Find top m peaks
    top_indices = np.argsort(b)[-m_peaks:]
    
    # For each peak, compute which a's are responsible
    # The gradient of b_k w.r.t. a_j is a_{k-j} (if valid)
    # So the "responsibility" of a_j for peak k is a_j * a_{k-j}
    
    # Compute a weighted "pressure" on each a_j
    pressure = np.zeros(n)
    for k in top_indices:
        for j in range(n):
            kj = k - j
            if 0 <= kj < n:
                pressure[j] += a[j] * a[kj]
    
    # Reduce high-pressure a's and increase low-pressure a's
    # Use a multiplicative update
    mean_pressure = pressure.mean()
    ratio = 1.0 + alpha * (mean_pressure - pressure) / (mean_pressure + 1e-10)
    a_new = a * ratio
    a_new = np.maximum(a_new, 0)
    a_new = a_new / a_new.sum()
    return a_new
```

```python
def backtracking_line_search(a, g, J0, grad_J, max_backtrack=10):
    """
    Backtracking line search for direction g from point a.
    J0 = J(a), grad_J = gradient of J at a.
    Returns (best_a, best_J, best_t).
    """
    direction = g - a
    directional_deriv = np.dot(grad_J, direction)
    
    if directional_deriv >= 0:
        # Direction is not a descent direction, try smaller step
        return a, J0, 0.0
    
    t = 1.0
    c1 = 1e-4
    rho = 0.5
    
    for _ in range(max_backtrack):
        a_try = (1 - t) * a + t * g
        a_try = np.maximum(a_try, 0)
        if a_try.sum() < 0.01:
            t *= rho
            continue
        J_try = compute_J(a_try)
        
        if J_try <= J0 + c1 * t * directional_deriv:
            return a_try, J_try, t
        
        t *= rho
    
    # If we get here, try the best we found
    return a, J0, 0.0
```

```python
"""
Minimizes J(a) = 2*n*max(convolve(a,a))/(sum(a))² using:
1. Fast LP mirror descent with adaptive line search

3. Multi-length exploration with resampling

import numpy as np
from scipy.optimize import minimize, linprog
from scipy.sparse import lil_matrix, csr_matrix
import time

_A_structure_cache = {}


def fast_conv(a):
    """Full self-convolution via FFT."""
    n = len(a)
    s = 1
    while s < 2 * n - 1:
        s <<= 1
    fa = np.fft.rfft(a, s)
    return np.fft.irfft(fa * fa, s)[:2 * n - 1]


def compute_J(a):
    """Compute J(a) = 2*n*max(conv(a,a))/(sum(a))^2."""
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))


def get_A_structure(n):
    """Get or pre-compute the (row, col) arrays for the constraint matrix."""
    if n not in _A_structure_cache:
        rows = []
        cols = []
        for k in range(2 * n - 1):
            jl = max(0, k - n + 1)
            jh = min(n - 1, k)
            for j in range(jl, jh + 1):
                rows.append(k)
                cols.append(j)
        _A_structure_cache[n] = (np.array(rows, dtype=np.int32),
                                  np.array(cols, dtype=np.int32))
    return _A_structure_cache[n]


def build_A_sparse_fast(a):
    """Build sparse constraint matrix using pre-computed structure."""
    n = len(a)
    rows_arr, cols_arr = get_A_structure(n)
    vals = a[rows_arr - cols_arr]
    return csr_matrix((vals, (rows_arr, cols_arr)), shape=(2 * n - 1, n))


def lp_best_response(a, M):
    """LP: max sum(g) s.t. (a*g)_k <= M for all k, g >= 0. Returns normalized g or None."""
    n = len(a)
    c = -np.ones(n)
    A = build_A_sparse_fast(a)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    try:
        r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                    options={'presolve': True, 'time_limit': 5.0})
        if r.success and r.x is not None:
            g = np.maximum(r.x, 0.0)
            g_sum = g.sum()
            if g_sum > 1e-10:
                return g / g_sum
    except Exception:
        pass
    return None


def adaptive_line_search(a, g, J0, max_tries=8):
    """
    Adaptive line search: try a set of step sizes, return the best.
    Uses a set that adapts based on what worked before.
    """
    best_a = a
    best_J = J0
    best_t = 0.0

    # Try a range of step sizes, larger first (full step often works)
    t_values = [1.0, 0.7, 0.5, 0.3, 0.15, 0.75, 0.4, 0.1]

    for t in t_values:
        a_try = (1.0 - t) * a + t * g
        a_try = np.maximum(a_try, 0.0)
        s = a_try.sum()
        if s < 0.01:
            continue
        a_try = a_try / s
        J_try = compute_J(a_try)
        if J_try < best_J:
            best_J = J_try
            best_a = a_try
            best_t = t
            if t >= 0.9:
                break  # Full step worked, no need to try smaller

    return best_a, best_J, best_t


def flattening_direction(a, n_peaks=10, beta=2.0):
    """
    Compute a "flattening" direction using multiplicative weights.
    Reduces mass where it contributes most to peaks.
    """
    n = len(a)
    b = fast_conv(a)
    b_max = b.max()

    # Compute soft "pressure" on each a_j based on its contribution to peaks
    # Use log-sum-exp weighted approach
    shifted = beta * (b - b_max)
    shifted = np.clip(shifted, -500, 500)
    w = np.exp(shifted)
    w = w / w.sum()

    # The "responsibility" of a_j is sum_k w_k * a_{k-j}
    # This is a cross-correlation of w with a
    S = np.convolve(w, a, mode='full')
    pressure = S[n - 1:2 * n - 1]  # length n

    # Direction: reduce high-pressure, keep low-pressure
    # g_j = a_j * (mean_pressure / pressure_j)^alpha
    mean_p = pressure.mean() + 1e-15
    alpha = 0.5
    g = a * (mean_p / (pressure + 1e-15)) ** alpha
    g = np.maximum(g, 0)
    g_sum = g.sum()
    if g_sum < 1e-10:
        return a.copy()
    return g / g_sum


def smoothed_obj_and_grad(a, beta):
    """
    Compute smooth approximation of J(a) and its gradient.
    Uses log-sum-exp to smooth the max in convolution.
    """
    n = len(a)
    b = fast_conv(a)
    b_max = float(b.max())
    shifted = beta * (b - b_max)
    shifted = np.clip(shifted, -500, 500)
    exp_shifted = np.exp(shifted)
    Z = float(exp_shifted.sum())
    if Z < 1e-300:
        p = np.zeros(2 * n - 1)
        p[np.argmax(b)] = 1.0
    else:
        p = exp_shifted / Z

    g_val = (b_max + np.log(max(Z, 1e-300))) / beta

    # Gradient of g_val w.r.t. a: 2 * (a cross-correlated with p)
    S = np.convolve(a, p[::-1], mode='full')
    grad_g = 2.0 * S[n - 1:2 * n - 1]

    s = float(a.sum())
    if s < 1e-10:
        return 1e10, np.zeros(n)

    F_val = 2.0 * n * g_val / (s * s)
    grad_F = 2.0 * n * (s * grad_g - 2.0 * g_val) / (s ** 3)

    return F_val, grad_F


def lbfgs_optimize(a0, betas, deadline):
    """L-BFGS-B optimization with beta annealing schedule."""
    a = np.maximum(a0.astype(np.float64), 1e-8)
    n = len(a)

    for beta, max_iters in betas:
        if time.monotonic() > deadline:
            break

        def func_wrapper(x):
            x2 = np.maximum(x, 0.0)
            f, _ = smoothed_obj_and_grad(x2, beta)
            if not np.isfinite(f):
                return 1e10
            return f

        def jac_wrapper(x):
            x2 = np.maximum(x, 0.0)
            _, g = smoothed_obj_and_grad(x2, beta)
            g = np.nan_to_num(g, nan=0.0, posinf=0.0, neginf=0.0)
            g[x2 < 1e-12] = 0.0
            return g

        bounds = [(0.0, None)] * n
        try:
            res = minimize(func_wrapper, a.copy(), jac=jac_wrapper,
                           method='L-BFGS-B', bounds=bounds,
                           options={'maxiter': max_iters, 'max_fun': max_iters * 30,
                                    'gtol': 1e-9, 'ftol': 1e-15})
            if res.fun < 1e10 and np.isfinite(res.fun):
                new_a = np.maximum(res.x, 1e-10)
                if new_a.sum() > 0.01:
                    a = new_a
        except Exception:
            pass

    return a


def resample_seq(src, dest_n):
    """Resample 1D sequence to new length via linear interpolation."""
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    result = np.interp(t_new, t_old, src)
    return np.maximum(result, 0.0)


def make_palindromic(a):
    """Make a sequence palindromic by averaging a_j with a_{n-1-j}."""
    n = len(a)
    a_pal = 0.5 * (a + a[::-1])
    return np.maximum(a_pal, 0)


def lp_mirror_descent(a, max_iters, deadline, tracker=None, use_flattening_every=200):
    """
    LP-based mirror descent with adaptive line search.
    Periodically uses flattening direction to escape plateaus.
    """
    a = np.maximum(a.astype(np.float64), 1e-6)
    a = a / a.sum()
    current_J = compute_J(a)
    if tracker is not None:
        tracker.update(a)

    stall_count = 0
    best_J_so_far = current_J
    best_a_so_far = a.copy()
    consec_no_improve = 0

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        b = fast_conv(a)
        M = float(b.max())

        # Choose direction: LP best response or flattening
        if it % use_flattening_every == use_flattening_every - 1 and consec_no_improve > 50:
            g = flattening_direction(a)
        else:
            g = lp_best_response(a, M)
            if g is None:
                g = flattening_direction(a)

        # Line search
        a_new, J_new, t_used = adaptive_line_search(a, g, current_J)

        if J_new < current_J - 1e-14:
            a = a_new
            current_J = J_new
            consec_no_improve = 0
            if J_new < best_J_so_far:
                best_J_so_far = J_new
                best_a_so_far = a.copy()
                if tracker is not None:
                    tracker.update(a)
        else:
            consec_no_improve += 1
            # Try smaller explicit steps
            found = False
            for t_try in [0.05, 0.02, 0.01]:
                a_try = (1.0 - t_try) * a + t_try * g
                a_try = np.maximum(a_try, 0.0)
                if a_try.sum() < 0.01:
                    continue
                a_try = a_try / a_try.sum()
                J_try = compute_J(a_try)
                if J_try < current_J - 1e-15:
                    a = a_try
                    current_J = J_try
                    consec_no_improve = 0
                    if J_try < best_J_so_far:
                        best_J_so_far = J_try
                        best_a_so_far = a.copy()
                        if tracker is not None:
                            tracker.update(a)
                    found = True
                    break
            if not found and consec_no_improve >= 80:
                break

        if tracker is not None and it % 200 == 0:
            tracker.update(a)

    if tracker is not None:
        tracker.update(best_a_so_far)
    return best_a_so_far


def generate_initializations(n, rng, parent=None):
    """Generate diverse initial sequences of length n."""
    inits = []
    t_arr = np.linspace(0, 1, n, endpoint=False)

    if parent is not None and parent.size > 0:
        inits.append(resample_seq(parent, n))
        p = resample_seq(parent, n)
        inits.append(np.maximum(p + rng.normal(0, 0.05, n) * p, 0.01))

    # Uniform
    inits.append(np.ones(n))

    # Sine hump (optimal continuous approximation)
    inits.append(np.sin(np.pi * t_arr) + 0.01)

    # Sine squared
    inits.append(np.sin(np.pi * t_arr) ** 2 + 0.01)

    # Raised cosine (Hanning window)
    inits.append(0.5 * (1 + np.cos(2 * np.pi * t_arr)) + 0.01)

    # Triangular
    inits.append(1.0 - np.abs(2.0 * t_arr - 1.0))

    # Cosine families
    for freq in [2, 3, 4, 5, 7, 11]:
        inits.append(np.maximum(1.0 + 0.3 * np.cos(2.0 * np.pi * freq * t_arr), 0.01))

    # Gaussian bump
    inits.append(np.exp(-4.0 * (t_arr - 0.5) ** 2) + 0.05)

    # Two-peak
    inits.append(1.0 + 0.3 * np.cos(4.0 * np.pi * t_arr) + 0.1 * np.cos(8.0 * np.pi * t_arr))

    # Power law
    inits.append((t_arr + 0.1) ** 0.5)

    # Exponential decay
    inits.append(np.exp(-1.5 * t_arr) + 0.2)

    # Bell-shaped (optimal for flat correlation)
    inits.append(np.sqrt(np.maximum(np.sin(np.pi * t_arr), 0)) + 0.01)

    return inits


class ScoreTracker:
    """Tracks the best sequence found so far."""

    def __init__(self):
        self.best = None
        self.best_score = float('inf')
        self.t_start = time.monotonic()

    def update(self, a):
        a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
        if a.size == 0 or a.sum() < 0.01:
            return
        score = float(evaluate_sequence(a.tolist()))
        if np.isfinite(score) and score < self.best_score:
            self.best = a.copy()
            self.best_score = score
            elapsed = time.monotonic() - self.t_start
            print(f"  * NEW BEST: J={score:.8f} n={a.size} t={elapsed:.1f}s", flush=True)

    def report(self, label=""):
        if self.best is not None:
            print(f"  {label} best: J={self.best_score:.8f} n={self.best.size}", flush=True)


def residual_based_perturbation(a, rng, strength=0.1):
    """
    Perturb a based on the residual (where convolution is still peaked).
    Reduce mass at positions contributing to peaks, add to flat regions.
    """
    n = len(a)
    b = fast_conv(a)
    b_mean = b.mean()
    b_max = b.max()

    # Compute "excess" at each position
    excess = np.maximum(b - b_mean, 0)
    excess = excess / (excess.max() + 1e-15)

    # For each a_j, compute how much it contributes to excess
    # Contribution of a_j to b_k is a_j * a_{k-j}
    S = np.convolve(excess, a, mode='full')
    contrib = S[n - 1:2 * n - 1]
    contrib = contrib / (contrib.max() + 1e-15)

    # Reduce high-contribution, increase low-contribution
    noise = rng.normal(0, strength, n)
    a_new = a * (1.0 - strength * contrib) + rng.uniform(0, strength * 0.1, n)
    a_new = a_new + noise * a
    a_new = np.maximum(a_new, 1e-4)
    return a_new / a_new.sum()


def propose_candidate(seed=42, budget_s=1500, **kwargs):
    """
    Multi-phase optimization for minimizing J(a) = 2n*max(conv(a,a))/(sum(a))^2.

    Phase 1: L-BFGS-B with smooth objective at multiple lengths
    Phase 2: LP mirror descent with adaptive line search (main refinement)
    Phase 3: Multi-restart with residual-based and structured perturbations
    Phase 4: Final LP polish with extended iterations
    Phase 5: Length exploration around best
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + float(budget_s) - 25.0
    tracker = ScoreTracker()

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        if p.ndim == 1 and p.size > 0 and np.all(np.isfinite(p)):
            parent = np.maximum(p, 0.0)
            tracker.update(parent)
            print(f"Parent loaded: n={parent.size}, J={tracker.best_score:.8f}", flush=True)
    except (NameError, AttributeError):
        pass

    if parent is None:
        parent = np.ones(512)
        tracker.update(parent)
        print(f"Using default parent: n=512, J={tracker.best_score:.8f}", flush=True)

    # === Phase 1: L-BFGS-B with smooth objective ===
    t0 = time.monotonic()
    phase1_budget = 120.0
    deadline_p1 = time.monotonic() + phase1_budget
    print(f"\n{'=' * 50} Phase 1: L-BFGS-B (budget {phase1_budget:.0f}s) {'=' * 50}", flush=True)

    lengths_p1 = [512, 1024]
    beta_schedule_p1 = [(2.0, 30), (10.0, 40), (50.0, 50), (200.0, 60)]

    for n in lengths_p1:
        if time.monotonic() > deadline_p1:
            break
        t_start = time.monotonic()
        print(f"  L-BFGS-B n={n}...", flush=True)

        inits = generate_initializations(n, rng, parent)
        max_inits = min(len(inits), 8)

        for idx, a0 in enumerate(inits[:max_inits]):
            if time.monotonic() > deadline_p1:
                break
            a0 = np.maximum(a0.astype(np.float64), 0.01)
            a0 = a0 / a0.sum()
            a_opt = lbfgs_optimize(a0, beta_schedule_p1, deadline_p1)
            if a_opt is not None and a_opt.sum() > 0.01:
                tracker.update(a_opt)
            # Also try palindromic version
            if a_opt is not None:
                a_pal = make_palindromic(a_opt)
                if a_pal.sum() > 0.01:
                    a_pal = a_pal / a_pal.sum()
                    tracker.update(a_pal)

        print(f"    n={n} done in {time.monotonic()-t_start:.1f}s", flush=True)

    tracker.report("Phase 1 complete")

    # === Phase 2: LP mirror descent (MAIN PHASE) ===
    if tracker.best is not None and time.monotonic() < deadline:
        t0 = time.monotonic()
        n_best = tracker.best.size
        phase2_budget = 700.0
        deadline_p2 = min(deadline, time.monotonic() + phase2_budget)
        print(f"\n{'=' * 50} Phase 2: LP MD n={n_best} (budget {phase2_budget:.0f}s) {'=' * 50}", flush=True)

        # Primary run at best length
        remaining = deadline_p2 - time.monotonic()
        max_iters = int(remaining * 150)
        max_iters = min(max_iters, 20000)
        a = lp_mirror_descent(tracker.best, max_iters, deadline_p2, tracker=tracker,
                              use_flattening_every=300)

        # Try at other lengths
        for n_try in [512, 768, 1024, 1536, 2048]:
            if time.monotonic() > deadline_p2:
                break
            if n_try == n_best:
                continue
            a_res = resample_seq(tracker.best, n_try)
            a_res = a_res / max(a_res.sum(), 1e-10)
            budget_here = min(150.0, deadline_p2 - time.monotonic())
            if budget_here < 10:
                break
            max_iters_here = int(budget_here * 100)
            max_iters_here = min(max_iters_here, 10000)
            a_res = lp_mirror_descent(a_res, max_iters_here, deadline_p2, tracker=tracker,
                                      use_flattening_every=200)

        tracker.report(f"Phase 2: {time.monotonic()-t0:.1f}s")

    # === Phase 3: Multi-restart with diverse perturbations ===
    if tracker.best is not None and time.monotonic() < deadline:
        t0 = time.monotonic()
        n_best = tracker.best.size
        phase3_budget = 300.0
        deadline_p3 = min(deadline, time.monotonic() + phase3_budget)
        print(f"\n{'=' * 50} Phase 3: Restarts (budget {phase3_budget:.0f}s) {'=' * 50}", flush=True)

        strategies = []
        # Residual-based perturbations (informed)
        for strength in [0.05, 0.1, 0.2, 0.3]:
            strategies.append(('residual', strength))
        # Gaussian perturbations
        for sigma in [0.03, 0.08, 0.15, 0.25]:
            strategies.append(('gaussian', sigma))
        # Structured: palindromic
        strategies.append(('palindromic', 1.0))
        # Binary-like rounding
        strategies.append(('binary', 1.0))
        # Resample to different length
        for n_alt in [768, 1536, 2048]:
            strategies.append(('resample', n_alt))

        attempt = 0
        for pert_type, param in strategies:
            for trial in range(4):
                if time.monotonic() > deadline_p3:
                    break
                attempt += 1
                a = tracker.best.copy().astype(np.float64)
                n = len(a)

                if pert_type == 'gaussian':
                    a = a + rng.normal(0, param, n) * a
                elif pert_type == 'residual':
                    a = residual_based_perturbation(a, rng, strength=param)
                elif pert_type == 'palindromic':
                    a = make_palindromic(a + rng.normal(0, 0.02, n) * a)
                elif pert_type == 'binary':
                    # Round to create binary-like structure
                    threshold = a.mean()
                    a_bin = np.where(a > threshold, 1.0, 0.1)
                    a = 0.5 * a + 0.5 * a_bin * a.sum() / a_bin.sum()
                elif pert_type == 'resample':
                    a = resample_seq(a, param)
                    n = len(a)

                a = np.maximum(a, 1e-4)
                a = a / a.sum()

                # Quick check
                J_pert = compute_J(a)
                if J_pert < tracker.best_score:
                    tracker.update(a)

                # Run LP refinement
                budget_here = min(45.0, deadline_p3 - time.monotonic())
                if budget_here < 3:
                    break
                max_iters = int(budget_here * 100)
                max_iters = min(max_iters, 5000)
                a = lp_mirror_descent(a, max_iters, deadline_p3, tracker=tracker,
                                      use_flattening_every=200)

        tracker.report(f"Phase 3: {time.monotonic()-t0:.1f}s, {attempt} attempts")

    # === Phase 4: Final LP polish ===
    if tracker.best is not None and time.monotonic() < deadline:
        t0 = time.monotonic()
        n_best = tracker.best.size
        remaining = deadline - time.monotonic()
        phase4_budget = min(250.0, remaining)
        deadline_p4 = time.monotonic() + phase4_budget
        print(f"\n{'=' * 50} Phase 4: Final Polish n={n_best} (budget {phase4_budget:.0f}s) {'=' * 50}", flush=True)

        # Long run
        max_iters = int(phase4_budget * 120)
        max_iters = min(max_iters, 15000)
        a = lp_mirror_descent(tracker.best, max_iters, deadline_p4, tracker=tracker,
                              use_flattening_every=500)

        # Second pass
        if tracker.best is not None and time.monotonic() < deadline_p4:
            remaining2 = deadline_p4 - time.monotonic()
            if remaining2 > 20:
                max_iters2 = int(remaining2 * 120)
                max_iters2 = min(max_iters2, 8000)
                a = lp_mirror_descent(tracker.best, max_iters2, deadline_p4, tracker=tracker,
                                      use_flattening_every=400)

        tracker.report(f"Phase 4: {time.monotonic()-t0:.1f}s")

    # === Phase 5: Try adjacent lengths ===
    if tracker.best is not None and time.monotonic() < deadline:
        t0 = time.monotonic()
        n_best = tracker.best.size
        remaining = deadline - time.monotonic()
        if remaining > 30:
            print(f"\n{'=' * 50} Phase 5: Length Exploration (budget {remaining:.0f}s) {'=' * 50}", flush=True)

            candidate_lengths = [n_best - 64, n_best - 32, n_best + 32, n_best + 64,
                                  n_best + 128, n_best - 128,
                                  896, 1152, 1280, 1408, 1792]
            seen = set()
            for n_try in candidate_lengths:
                if n_try < 128 or n_try > 3000 or time.monotonic() > deadline:
                    continue
                if abs(n_try - n_best) < 16 or n_try in seen:
                    continue
                seen.add(n_try)
                a_res = resample_seq(tracker.best, n_try)
                a_res = a_res / max(a_res.sum(), 1e-10)
                budget_here = min(30.0, deadline - time.monotonic())
                if budget_here < 3:
                    break
                max_iters = int(budget_here * 100)
                max_iters = min(max_iters, 3000)
                a_res = lp_mirror_descent(a_res, max_iters, deadline, tracker=tracker,
                                          use_flattening_every=150)

            tracker.report(f"Phase 5: {time.monotonic()-t0:.1f}s")

    # Fallback
    if tracker.best is None:
        tracker.best = np.ones(512)
        tracker.best_score = float(evaluate_sequence(tracker.best.tolist()))
        print(f"  Using fallback: J={tracker.best_score:.8f}", flush=True)

    print(f"\n{'=' * 50} FINAL RESULT {'=' * 50}", flush=True)
    print(f"J={tracker.best_score:.8f} n={tracker.best.size} "
          f"time={time.monotonic()-tracker.t_start:.1f}s", flush=True)

    # Ensure output is valid
    result = np.clip(tracker.best, 0.0, 1000.0)
    if result.sum() < 0.01:
        result = np.ones(128)

    return [float(x) for x in result]
```
