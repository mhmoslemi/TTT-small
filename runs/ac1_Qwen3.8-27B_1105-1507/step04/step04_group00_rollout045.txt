"""
Hybrid optimization using:
1. Adam + Nesterov on peak-aware smooth max loss (main workhorse)
2. Multi-length exploration with diverse inits
3. Peak-targeting coordinate descent
4. Short LP mirror descent polish

import numpy as np
from scipy.optimize import linprog
from scipy.sparse import csr_matrix
import time

def fast_conv(a):
    n = len(a)
    size = 1
    while size < 2*n-1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa*fa, size)[:2*n-1]

def compute_J(a):
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2*len(a)*b.max()/(s*s))

def smooth_loss_and_grad(a, tau):
    """Compute log-sum-exp loss and its gradient."""
    n = len(a)
    b = fast_conv(a)
    b_max = b.max()
    exp_b = np.exp(tau*(b - b_max))
    p = exp_b / exp_b.sum()
    loss = b_max + np.log(exp_b.sum())/tau
    
    grad = np.zeros(n)
    for k in range(2*n-1):
        pk = p[k]
        if pk < 1e-14:
            continue
        lo = max(0, k-n+1)
        hi = min(n-1, k)
        grad[lo:hi+1] += 2.0*pk*a[k-hi:k-lo+1][::-1]
    return loss, grad, b

    m = np.zeros(n)
    v = np.zeros(n)
    best_a = a.copy()
    best_M = fast_conv(a).max()
    
    for t in range(max_iters):
        if time.monotonic() > deadline:

"""
Hybrid optimizer for minimizing J(a) = 2*n*max(conv(a,a))/(sum(a))^2.
Key innovations over previous approach:
1. Adam optimizer with per-coordinate adaptive learning rates (vs plain SGD)
2. Two-stage smooth-to-sharp tau annealing
3. Peak-targeting coordinate descent with smart coordinate selection
4. Structured "anti-peak" perturbations for restarts
5. Odd-length exploration
6. Efficient time allocation focusing on the most promising lengths

import numpy as np
from scipy.optimize import linprog
from scipy.sparse import csr_matrix
import time


def fast_conv(a):
    """Full self-convolution via FFT."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]


def compute_J(a):
    """Compute J(a) = 2*n*max(conv(a,a))/(sum(a))^2."""
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))


def smooth_loss_grad(a, tau):
    """Compute log-sum-exp smooth max and gradient."""
    n = len(a)
    b = fast_conv(a)
    b_max = b.max()
    exp_b = np.exp(tau * (b - b_max))
    exp_sum = exp_b.sum()
    p = exp_b / exp_sum
    loss = b_max + np.log(exp_sum) / tau

    grad = np.zeros(n)
    for k in range(2 * n - 1):
        pk = p[k]
        if pk < 1e-14:
            continue
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        grad[lo:hi + 1] += 2.0 * pk * a[k - hi:k - lo + 1][::-1]
    return loss, grad, b, p


def exact_subgradient(a, tol_frac=1e-5):
    """Compute exact subgradient of max(conv(a,a))."""
    n = len(a)
    b = fast_conv(a)
    M = b.max()
    tol = tol_frac * M if M > 0 else 1e-10
    active = np.where(b >= M - tol)[0]

    grad = np.zeros(n)
    for k in active:
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        grad[lo:hi + 1] += 2.0 * a[k - hi:k - lo + 1][::-1]
    return grad, active, b, M


def adam_smooth_max(a, max_iters, deadline, tau_start=10.0, tau_end=500.0,
                    lr=0.05, beta1=0.9, beta2=0.999, eps=1e-8,
                    nesterov=True, restart_threshold=500):
    """
    Adam optimizer on log-sum-exp smooth max.
    Supports Nesterov lookahead and adaptive restarts.
    """
    a = np.maximum(a.astype(np.float64), 1e-10)
    a = a / a.sum()
    n = len(a)
    m = np.zeros(n)
    v = np.zeros(n)
    best_a = a.copy()
    best_M = fast_conv(a).max()
    no_improve = 0
    t = 0

    while t < max_iters and time.monotonic() < deadline:
        frac = min(1.0, t / max(1, max_iters))
        # Faster tau ramp: stay smooth longer, then sharpen quickly
        if frac < 0.6:
            tau = tau_start + (50.0 - tau_start) * (frac / 0.6)
        else:
            tau = 50.0 + (tau_end - 50.0) * ((frac - 0.6) / 0.4) ** 2

        # Learning rate schedule: high initial, sublinear decay
        lr_t = lr / (1 + 0.05 * t)

        # Nesterov lookahead
        if nesterov and t > 0:
            m_hat_curr = m / (1 - beta1 ** t)
            a_look = a - lr_t * m_hat_curr
            a_look = np.maximum(a_look, 0)
            sl = a_look.sum()
            if sl > 1e-10:
                a_look = a_look / sl
        else:
            a_look = a

        _, grad, b, p = smooth_loss_grad(a_look, tau)
        M = b.max()

        if M < best_M * (1 - 1e-10):
            best_M = M
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        # Adam update
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * grad ** 2
        m_hat = m / (1 - beta1 ** (t + 1))
        v_hat = v / (1 - beta2 ** (t + 1))

        a = a - lr_t * m_hat / (np.sqrt(v_hat) + eps)
        a = np.maximum(a, 0)
        s = a.sum()
        if s < 1e-10:
            a = best_a.copy()
        else:
            a = a / s

        t += 1

        # Adaptive restart: if stuck, perturb toward reducing peaks
        if no_improve > restart_threshold and time.monotonic() < deadline - 30:
            b = fast_conv(a)
            M = b.max()
            active = np.where(b >= M * 0.99)[0]
            # Create perturbation that opposes the active peaks
            pert = np.zeros(n)
            for k in active:
                lo = max(0, k - n + 1)
                hi = min(n - 1, k)
                pert[lo:hi + 1] += a[k - hi:k - lo + 1][::-1]
            # Subtract a small fraction of the perturbation
            a = a - 0.02 * pert
            a = np.maximum(a, 1e-8)
            a = a / a.sum()
            m = np.zeros(n)
            v = np.zeros(n)
            no_improve = 0

    return best_a


def subgradient_adam(a, max_iters, deadline, lr=0.02, beta1=0.9, beta2=0.999):
    """Adam with exact subgradient of max."""
    a = np.maximum(a.astype(np.float64), 1e-10)
    a = a / a.sum()

        lr_t = lr / (1 + 0.02 * t)
        m = beta1 * m + (1 - beta1) * grad

        m_hat = m / (1 - beta1 ** (t + 1))

    return best_a


def coordinate_descent_peak(a, max_iters, deadline):
    """
    Coordinate descent targeting the peak of the convolution.
    Cycles through coordinates, taking the best small step for each.
    """
    a = np.maximum(a.astype(np.float64), 1e-10)
    a = a / a.sum()
    n = len(a)
    best_a = a.copy()
    best_M = fast_conv(a).max()
    rng = np.random.default_rng(12345)

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        b = fast_conv(a)
        M = b.max()
        peak_k = np.argmax(b)

        # Pick a coordinate that contributes to the peak
        # The peak at k involves a_i for i in [max(0,k-n+1), min(n-1,k)]
        lo = max(0, peak_k - n + 1)
        hi = min(n - 1, peak_k)
        candidates = list(range(lo, hi + 1))
        if not candidates:
            break

        # Also consider random coordinates for exploration
        idx = rng.choice(candidates + [rng.integers(0, n)])

        # Try several step sizes for this coordinate
        best_trial = a.copy()
        best_trial_M = M
        current_val = a[idx]

        for delta_frac in [-0.3, -0.15, -0.05, 0.05, 0.15, 0.3]:
            trial = a.copy()
            trial[idx] = max(0, current_val * (1 + delta_frac))
            s = trial.sum()
            if s < 1e-10:
                continue
            trial = trial / s
            tr_b = fast_conv(trial)
            tr_M = tr_b.max()
            if tr_M < best_trial_M:
                best_trial_M = tr_M
                best_trial = trial

        if best_trial_M < M * (1 - 1e-12):
            a = best_trial
            if a.max() < best_M * (1 - 1e-10):
                best_M = a.max()
                best_a = a.copy()

    return best_a


def build_A_sparse(n, a):
    """Build sparse constraint matrix for LP."""
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        j_min = max(0, k - n + 1)
        j_max = min(n - 1, k)
        for j in range(j_min, j_max + 1):
            rows.append(k)
            cols.append(j)
            vals.append(a[k - j])
    return csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))


def lp_best_response(a, M, time_limit=10):
    """LP: max sum(g) s.t. (a*g)[k] <= M for all k, g >= 0."""
    n = len(a)
    A = build_A_sparse(n, a)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    try:
        r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                    options={'time_limit': time_limit})
        if r.success and r.x is not None:
            g = np.maximum(r.x, 0)
            if g.sum() > 1e-15:
                return g
    except Exception:
        pass
    return None


def lp_mirror_descent(a, max_iters, deadline, line_search_fracs=None):
    """LP mirror descent with geometric line search."""
    a = np.maximum(a.astype(np.float64), 1e-10)
    a = a / a.sum()
    best_a = a.copy()
    best_M = fast_conv(a).max()
    no_improve = 0

    if line_search_fracs is None:
        line_search_fracs = [0.5, 0.3, 0.15, 0.08, 0.04, 0.02]

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        b = fast_conv(a)
        M = float(b.max())
        g = lp_best_response(a, M, time_limit=8)
        if g is None:
            break
        g = np.maximum(g, 0.0)
        g_sum = g.sum()
        if g_sum < 1e-10:
            break
        g = g / g_sum

        improved = False
        for t in line_search_fracs:
            a_trial = np.maximum((1 - t) * a + t * g, 0)
            s = a_trial.sum()
            if s < 1e-10:
                continue
            a_trial = a_trial / s
            M_trial = fast_conv(a_trial).max()
            if M_trial < best_M - 1e-14:
                a = a_trial
                best_M = M_trial
                best_a = a.copy()
                improved = True
                break

        if not improved:
            no_improve += 1
            if no_improve > 15:
                break
        else:
            no_improve = 0

    return best_a


def resample_seq(src, dest_n):
    """Resample 1D sequence to new length via linear interpolation."""
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)


def generate_inits(n, rng, parent=None):
    """Generate diverse initial sequences."""
    inits = []
    t = np.linspace(0, 1, n, endpoint=False)

    if parent is not None:
        p = resample_seq(parent, n)
        inits.append(p.copy())
        inits.append(np.maximum(p + rng.normal(0, 0.02, n), 0.01))
        inits.append(np.maximum(p + rng.normal(0, 0.08, n), 0.01))
        # Structured anti-peak perturbation
        b = fast_conv(p)
        M = b.max()
        pert = np.zeros(n)
        for k in np.where(b >= M * 0.95)[0]:
            lo = max(0, k - n + 1)
            hi = min(n - 1, k)
            pert[lo:hi + 1] += p[k - hi:k - lo + 1][::-1]
        inits.append(np.maximum(p - 0.05 * pert / (pert.max() + 1e-10), 0.01))

    # Uniform
    inits.append(np.ones(n) / n)

    # Cosine family
    for freq in [1, 2, 3, 5]:
        for amp in [0.2, 0.4, 0.6]:
            inits.append(np.maximum(1.0 + amp * np.cos(2 * np.pi * freq * t), 0.01))
            inits.append(np.maximum(1.0 + amp * np.sin(2 * np.pi * freq * t), 0.01))

    # Gaussian bumps
    for center in [0.3, 0.5, 0.7]:
        for width in [0.1, 0.2, 0.35]:
            inits.append(np.exp(-((t - center) / width) ** 2) + 0.05)

    # Double bump
    inits.append(np.exp(-8 * (t - 0.25) ** 2) + np.exp(-8 * (t - 0.75) ** 2) + 0.05)
    inits.append(np.exp(-12 * (t - 0.2) ** 2) + np.exp(-12 * (t - 0.5) ** 2) + np.exp(-12 * (t - 0.8) ** 2) + 0.03)

    # Smooth random
    ksize = max(3, n // 20)
    kernel = np.ones(ksize) / ksize
    smooth = np.convolve(rng.uniform(0, 1, n), kernel, mode='same')
    inits.append(np.maximum(smooth, 0.01))

    # Power law
    inits.append(np.maximum(t ** 0.5 + 0.3, 0.01))
    inits.append(np.maximum((1 - t) ** 0.5 + 0.3, 0.01))
    inits.append(np.maximum(0.5 + 0.5 * np.abs(np.sin(np.pi * t)), 0.01))

    # Multi-frequency
    inits.append(np.maximum(1.0 + 0.15 * np.cos(4 * np.pi * t) + 0.15 * np.cos(7 * np.pi * t)
                            + 0.1 * np.cos(11 * np.pi * t), 0.01))

    # Chirp
    inits.append(np.maximum(1.0 + 0.3 * np.cos(2 * np.pi * (3 + 5 * t) * t), 0.01))

    return inits

        a2 = np.maximum(a2, 0.01)
        a2 = a2 / a2.sum()
        a2 = subgradient_adam(a2, 2000, p2_deadline, lr=0.01, beta1=0.9, beta2=0.999)
        tracker.update(a2)

    tracker.report("Phase 2")

    # === PHASE 3: Coordinate descent peak targeting ===
    p3_deadline = time.monotonic() + (deadline - time.monotonic()) * 0.18
    print(f"\n{'='*60}", flush=True)
    print(f"PHASE 3: Coordinate Descent (until {p3_deadline - time.monotonic():.0f}s)", flush=True)
    print(f"{'='*60}", flush=True)

    a = tracker.best.copy()
    n_best = a.size
    # Coordinate descent is O(n^2) per iter for n coords, so limit iterations
    est_cd_time = 0.005 * (n_best / 128) ** 2
    max_cd_iters = max(1000, int((p3_deadline - time.monotonic()) / max(0.001, est_cd_time)))
    a = coordinate_descent_peak(a, max_cd_iters, p3_deadline)
    tracker.update(a)

    tracker.report("Phase 3")

    # === PHASE 4: LP Mirror Descent ===
    p4_deadline = time.monotonic() + (deadline - time.monotonic()) * 0.15
    print(f"\n{'='*60}", flush=True)
    print(f"PHASE 4: LP Mirror Descent (until {p4_deadline - time.monotonic():.0f}s)", flush=True)
    print(f"{'='*60}", flush=True)

    a = tracker.best.copy()
    n_best = a.size
    est_lp_time = 0.005 * (n_best / 128) ** 2
    max_lp_iters = max(20, int((p4_deadline - time.monotonic()) / max(0.01, est_lp_time)))
    print(f"  n={n_best}, est LP={est_lp_time*1000:.0f}ms, max_iters={max_lp_iters}", flush=True)

    a = lp_mirror_descent(a, max_lp_iters, p4_deadline)
    tracker.update(a)

    # Second LP pass if improved
    if time.monotonic() < p4_deadline - 5:
        a2 = lp_mirror_descent(tracker.best, max(10, max_lp_iters // 3), p4_deadline)
        tracker.update(a2)

    tracker.report("Phase 4")

    # === PHASE 5: Final adaptive polish ===
    print(f"\n{'='*60}", flush=True)
    print(f"PHASE 5: Final Polish (remaining time)", flush=True)
    print(f"{'='*60}", flush=True)

    # Try Adam one more time from the best with very high tau
    if tracker.best is not None and time.monotonic() < deadline - 5:
        a = tracker.best.copy()
        a = adam_smooth_max(a, 2000, deadline - 3,
                            tau_start=200.0, tau_end=1000.0,
                            lr=0.01, beta1=0.95, beta2=0.999,
                            nesterov=False, restart_threshold=200)
        tracker.update(a)

        # Short coordinate descent
        if time.monotonic() < deadline - 2:
            a = tracker.best.copy()
            est_cd_time = 0.005 * (a.size / 128) ** 2
            cd_iters = max(500, int((deadline - 3 - time.monotonic()) / max(0.001, est_cd_time)))
            a = coordinate_descent_peak(a, cd_iters, deadline - 1)
            tracker.update(a)

    # Try one more length: resample best to a nearby length
    if tracker.best is not None and time.monotonic() < deadline - 10:
        base_n = tracker.best.size
        for new_n in [base_n - 16, base_n + 16, base_n - 8, base_n + 8]:
            if new_n <= 0 or time.monotonic() > deadline - 5:
                continue
            a_ext = resample_seq(tracker.best, new_n)
            a_ext = np.maximum(a_ext, 1e-6)
            a_ext = a_ext / a_ext.sum()
            a_ext = adam_smooth_max(a_ext, 300, deadline - 3,
                                    tau_start=50, tau_end=400,
                                    lr=0.02, beta1=0.9, beta2=0.999,
                                    nesterov=True, restart_threshold=100)
            tracker.update(a_ext)

    # Fallback
    if tracker.best is None:
        tracker.best = np.ones(128)
        tracker.best_score = compute_J(tracker.best)

    print(f"\n{'='*60}", flush=True)
    print(f"FINAL: J={tracker.best_score:.8f} n={tracker.best.size} "
          f"time={time.monotonic()-tracker.t0:.0f}s", flush=True)
    print(f"{'='*60}", flush=True)

    return [float(x) for x in tracker.best]
