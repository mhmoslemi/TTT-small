Actually, let me reconsider the whole approach. The key challenge is:
1. The objective max(conv(a,a)) is non-smooth
2. The feasible set {a >= 0, sum(a) = 1} is a simplex

A cleaner approach might be:

**Minimize J(a) = 2n * max(conv(a,a)) / (sum(a))^2 directly**

Using the log-sum-exp (LSE) approximation for the max:
max_k b_k ≈ (1/tau) * log(sum_k exp(tau * b_k))

as tau → ∞. This gives a smooth objective that we can optimize with standard gradient methods.

The gradient of LSE w.r.t. a_j:
d/d a_j [log(sum_k exp(tau * b_k))] = (1/sum_k exp(tau*b_k)) * sum_k exp(tau*b_k) * tau * (db_k/da_j)
= tau * sum_k p_k * 2*a[k-j] / (valid)
where p_k = exp(tau*b_k) / sum_m exp(tau*b_m)

This is a weighted average of the individual gradients, weighted by the softmax of the convolution values. This is very smooth and easy to compute!

And the full gradient of J = 2n * LSE / (sum a)^2:
dJ/da_j = 2n * (dLSE/da_j * (sum a)^2 - LSE * 2*(sum a)*a_j) / (sum a)^4
= 2n * (dLSE/da_j * (sum a) - 2*LSE*a_j) / (sum a)^3

If we normalize sum(a) = 1: dJ/da_j = 2n * (dLSE/da_j - 2*LSE*a_j)

But wait, if we fix sum(a) = 1, we should project the gradient onto the tangent space of the simplex. The projection of a vector v onto the tangent space of {x: sum x = 1} is v - mean(v) (to maintain the sum constraint).

Actually, for the simplex {a >= 0, sum(a) = 1}, the projected gradient is:
1. Compute gradient g
2. a_new = a - alpha * g
3. Project a_new onto the simplex (clip negatives, renormalize)

"""
Minimizes J(a) = 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Symmetric LSE (Log-Sum-Exp) gradient descent with adaptive tau
2. FFT-based fast convolution and gradient computation
3. LP best-response polishing
4. Multi-length search (256 to 1024)
5. Tapered symmetric initializations
6. Active-set SLSQP polishing
7. Simulated annealing for global exploration

import numpy as np
from scipy.optimize import linprog, minimize
from scipy.sparse import csr_matrix
import time


def fast_conv(a):
    n = len(a)
    s = 1
    while s < 2 * n - 1:
        s <<= 1
    fa = np.fft.rfft(a, s)
    return np.fft.irfft(fa * fa, s)[:2 * n - 1]


def compute_J(a):
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))


def symmetrize(a):
    return (a + a[::-1]) * 0.5


def project_simplex(v):
    """Project onto {x >= 0, sum(x) = 1}."""
    v = np.maximum(v, 0.0)
    s = v.sum()
    if s < 1e-15:
        return np.ones(len(v)) / len(v)
    return v / s


def lse_gradient(a, b, tau):
    """Compute gradient of LSE approximation of max(b) w.r.t. a.
    Returns grad where grad[j] = 2 * sum_k p[k] * a[k-j] (valid indices)."""
    n = len(a)
    b_max = b.max()
    exp_b = np.exp(tau * (b - b_max))
    p = exp_b / exp_b.sum()
    lse_val = b_max + np.log(exp_b.sum()) / tau

    # Compute g[j] = 2 * sum_k p[k] * a[k-j] using FFT
    # g[j] = 2 * conv(a, p_rev)[2n-2-j]
    p_rev = p[::-1]
    s = 1
    while s < 3 * n:
        s <<= 1
    fa = np.fft.rfft(a, s)
    fp = np.fft.rfft(p_rev, s)
    conv_ap = np.fft.irfft(fa * fp, s)[:2 * n - 1]
    # g[j] = 2 * conv_ap[2n-2-j]
    grad = 2.0 * conv_ap[::-1][:n]
    return grad, lse_val, p


def symmetric_lse_descent(a, max_iters, deadline, tracker, tau_start=30.0, tau_max=500.0):
    """LSE gradient descent with symmetry enforcement and adaptive tau."""
    a = np.maximum(a.astype(np.float64), 1e-10)
    a = symmetrize(a)
    a = project_simplex(a)
    n = len(a)
    current_J = compute_J(a)
    tracker.update(a)

    tau = tau_start
    lr = 1.0 / (tau * n)
    momentum = 0.85
    v = np.zeros(n)
    stall_count = 0

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        b = fast_conv(a)
        grad, lse_val, p = lse_gradient(a, b, tau)
        grad = symmetrize(grad)

        # Project gradient onto tangent space of sum(a)=1
        grad = grad - grad.mean()

        v = momentum * v + (1.0 - momentum) * grad
        a_new = a - lr * v
        a_new = symmetrize(a_new)
        a_new = project_simplex(a_new)

        new_J = compute_J(a_new)

        if new_J < current_J - 1e-14:
            a = a_new
            current_J = new_J
            stall_count = 0
            # Adapt step size and tau
            lr = min(lr * 1.03, 5.0 / (tau * n))
            tau = min(tau * 1.01, tau_max)
        else:
            stall_count += 1
            lr *= 0.7
            v *= 0.5
            if stall_count > 30:
                # Perturb to escape
                rng = np.random.default_rng(it * 7919 + 13)
                a = a + rng.normal(0, 0.01 / n, n)
                a = symmetrize(a)
                a = project_simplex(a)
                v = np.zeros(n)
                lr = 1.0 / (tau * n)
                stall_count = 0

        if it % 30 == 0:
            tracker.update(a)

    tracker.update(a)
    return a


def get_sparsity(n):
    """Get sparse structure for convolution constraint matrix."""
    rows_list = []
    cols_list = []
    for k in range(2 * n - 1):
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        for j in range(js, je + 1):
            rows_list.append(k)
            cols_list.append(j)
    return np.array(rows_list), np.array(cols_list)


def lp_best_response(a, M, time_limit=5):
    """LP: maximize sum(b) subject to conv(a,b) <= M, b >= 0."""
    n = len(a)
    rows_list = []
    cols_list = []
    vals_list = []
    for k in range(2 * n - 1):
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        for j in range(js, je + 1):
            rows_list.append(k)
            cols_list.append(j)
            vals_list.append(a[k - j])

    A = csr_matrix((vals_list, (rows_list, cols_list)), shape=(2 * n - 1, n))
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n

    try:
        r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                    options={'presolve': True, 'time_limit': time_limit})
        if r.success and r.x is not None:
            g = np.maximum(r.x, 0.0)
            gs = g.sum()
            if gs > 1e-10:
                return g / gs
    except Exception:
        pass
    return None


def run_lp_opt(a, max_iters, deadline, tracker, tighten=0.01):
    """Iterative LP best-response optimization."""
    a = np.maximum(a.astype(np.float64), 1e-10)
    a = symmetrize(a)
    a = project_simplex(a)
    n = len(a)
    current_J = compute_J(a)
    tracker.update(a)

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        b = fast_conv(a)
        M = float(b.max())
        M_tight = M * (1.0 - tighten)
        g = lp_best_response(a, M_tight, time_limit=8)
        if g is None:
            g = lp_best_response(a, M, time_limit=8)
        if g is None:
            break
        g = symmetrize(g)
        g = project_simplex(g)

        best_a = None
        best_J = current_J
        for t in [0.3, 0.5, 0.7, 0.9, 0.15]:
            a_trial = (1.0 - t) * a + t * g
            a_trial = symmetrize(a_trial)
            a_trial = project_simplex(a_trial)
            J_trial = compute_J(a_trial)
            if J_trial < best_J - 1e-14:
                best_J = J_trial
                best_a = a_trial

        if best_a is None:
            break
        a = best_a
        current_J = best_J
        tracker.update(a)

    tracker.update(a)
    return a


def tapered_init(n, rng, kind=0):
    """Generate symmetric tapered initializations."""
    t = np.arange(n) / max(n - 1, 1)
    if kind == 0:
        # Cosine taper (Hamming-like)
        a = 0.4 + 0.6 * np.cos(np.pi * (2 * t - 1))
    elif kind == 1:
        # Raised cosine squared
        a = (0.5 + 0.5 * np.cos(2 * np.pi * t)) ** 0.5
        a = np.maximum(a, 1e-6)
    elif kind == 2:
        # Sinc-like (main lobe)
        x = (t - 0.5) * 4
        a = np.sinc(x) ** 2
        a = np.maximum(a, 1e-6)
    elif kind == 3:
        # Gaussian
        a = np.exp(-8 * (t - 0.5) ** 2) + 0.05
    elif kind == 4:
        # Optimal edge taper: very small at edges
        edge = max(2, n // 32)
        a = np.ones(n)
        for i in range(edge):
            a[i] = 0.01 + 0.99 * (i + 1) / edge
            a[n - 1 - i] = a[i]
    elif kind == 5:
        # Multi-optimized cosine
        a = 0.34 + 0.5 * np.cos(np.pi * (2 * t - 1)) + 0.16 * np.cos(2 * np.pi * (2 * t - 1))
        a = np.maximum(a, 0.01)
    else:
        # Random symmetric
        half = n // 2 + 1
        a_half = rng.uniform(0.1, 1.0, half)
        a = np.zeros(n)
        a[:half] = a_half
        a[half:] = a_half[:n - half][::-1]

    a = symmetrize(a)
    a = np.maximum(a, 1e-6)
    a /= a.sum()
    return a


def resample_symmetric(src, dest_n):
    """Resample a symmetric sequence to a new length, preserving symmetry."""
    src = np.asarray(src, dtype=np.float64)
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    resampled = np.interp(t_new, t_old, src)
    resampled = np.maximum(resampled, 0.0)
    resampled = symmetrize(resampled)
    resampled = project_simplex(resampled)
    return resampled


def coordinate_descent_sym(a, max_steps, deadline, tracker):
    """Coordinate descent on symmetric pairs."""
    a = np.maximum(a.astype(np.float64), 1e-10)
    a = symmetrize(a)
    a = project_simplex(a)
    n = len(a)
    current_J = compute_J(a)
    tracker.update(a)
    rng = np.random.default_rng(12345)
    half = n // 2 + 1
    stagnation = 0

    for step in range(max_steps):
        if time.monotonic() > deadline:
            break
        j = rng.integers(0, half)
        best_improve = 0
        best_scale = 1.0
        for scale in [0.7, 0.85, 1.15, 1.3, 0.5, 1.5, 0.9, 1.1]:
            a_trial = a.copy()
            a_trial[j] *= scale
            if n - 1 - j != j:
                a_trial[n - 1 - j] *= scale
            a_trial = symmetrize(a_trial)
            a_trial = project_simplex(a_trial)
            J_trial = compute_J(a_trial)
            improve = current_J - J_trial
            if improve > best_improve:
                best_improve = improve
                best_scale = scale
                a_trial_best = a_trial.copy()

        if best_improve > 1e-12:
            a = a_trial_best
            current_J -= best_improve
            stagnation = 0
        else:
            stagnation += 1
            if stagnation > 200:
                # Perturb
                j = rng.integers(0, half)
                a[j] *= rng.uniform(0.5, 2.0)
                if n - 1 - j != j:
                    a[n - 1 - j] = a[j]
                a = symmetrize(a)
                a = project_simplex(a)
                current_J = compute_J(a)
                stagnation = 0

        if step % 200 == 0:
            tracker.update(a)

    tracker.update(a)
    return a


def slsqp_polish(a, max_time=60):
    """Polish using SLSQP on active constraints."""
    a = np.maximum(a.astype(np.float64), 1e-8)
    a = symmetrize(a)
    a = project_simplex(a)
    n = len(a)
    b = fast_conv(a)
    M = b.max()

    # Find active constraints (within 5% of max)
    active = np.where(b >= M * 0.95)[0]

    def objective(x):
        a_local = np.maximum(x[:n], 0)
        s = a_local.sum()
        if s < 1e-10:
            return 1e10
        a_local = a_local / s
        a_local = symmetrize(a_local)
        conv = fast_conv(a_local)
        return conv.max()

    def grad_objective(x):
        a_local = np.maximum(x[:n], 1e-10)
        s = a_local.sum()
        a_local = a_local / s
        a_local = symmetrize(a_local)
        b_local = fast_conv(a_local)
        M_local = b_local.max()
        active_local = np.where(b_local >= M_local * 0.99)[0]
        grad = np.zeros(n)
        for k in active_local:
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            m_vals = k - np.arange(js, je + 1)
            grad[js:je + 1] += 2.0 * a_local[m_vals]
        grad /= len(active_local)
        grad = symmetrize(grad)
        grad = grad - grad.mean()
        return np.concatenate([grad, [0.0]])

    constraints = []
    for k in active[:20]:  # Limit to 20 most active constraints
        def make_constraint(kk):
            def constraint(x):
                a_local = np.maximum(x[:n], 0)
                s = a_local.sum()
                if s < 1e-10:
                    return 1e10
                a_local = a_local / s
                a_local = symmetrize(a_local)
                conv = fast_conv(a_local)
                return M - conv[kk]
            return constraint
        constraints.append({'type': 'ineq', 'fun': make_constraint(k)})

    # Sum constraint
    def sum_constraint(x):
        return x[:n].sum() - 1.0
    constraints.append({'type': 'eq', 'fun': sum_constraint})

    bounds = [(1e-8, None)] * n + [(0, None)]
    x0 = np.concatenate([a, [M]])

    try:
        result = minimize(objective, x0, method='SLSQP', jac=grad_objective,
                          constraints=constraints, bounds=bounds,
                          options={'maxiter': 200, 'ftol': 1e-12})
        if result.success:
            a_new = np.maximum(result.x[:n], 0)
            a_new = symmetrize(a_new)
            a_new = project_simplex(a_new)
            return a_new
    except Exception:
        pass
    return a


class ScoreTracker:
    def __init__(self):
        self.best = None
        self.best_score = float('inf')
        self.t_start = time.monotonic()

    def update(self, a):
        if a is None:
            return
        a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
        if a.size == 0 or a.sum() < 0.01:
            return
        score = float(evaluate_sequence(a.tolist()))
        if np.isfinite(score) and score < self.best_score:
            self.best = a.copy()
            self.best_score = score
            elapsed = time.monotonic() - self.t_start
            print(f"  * J={score:.8f} n={a.size} t={elapsed:.1f}s", flush=True)

    def report(self, label=""):
        if self.best is not None:
            print(f"  {label} best: J={self.best_score:.8f} n={self.best.size}", flush=True)


def simulated_annealing_sym(a, deadline, tracker, n_restarts=30):
    """Simulated annealing with symmetric perturbations."""
    a = a.copy().astype(np.float64)
    n = len(a)
    current_J = compute_J(a)
    tracker.update(a)
    rng = np.random.default_rng(42)
    temp = 0.1
    half = n // 2 + 1

    for r in range(n_restarts):
        if time.monotonic() > deadline:
            break

        a_pert = a.copy()
        # Perturb only the free (half) variables
        for _ in range(rng.integers(1, n // 10 + 1)):
            j = rng.integers(0, half)
            a_pert[j] += rng.normal(0, temp)
            if n - 1 - j != j:
                a_pert[n - 1 - j] = a_pert[j]
        a_pert = symmetrize(a_pert)
        a_pert = project_simplex(a_pert)

        # Short LSE refinement
        a_refined = symmetric_lse_descent(a_pert, 100, min(deadline, time.monotonic() + 20), tracker,
                                          tau_start=50, tau_max=200)
        J = compute_J(a_refined)
        delta = J - current_J
        if delta < 0 or rng.random() < np.exp(-max(delta, 0) / max(temp, 1e-10)):
            a = a_refined
            current_J = J
        temp *= 0.93

    tracker.update(a)
    return a

        t0 = time.monotonic()
        n = tracker.best.size
        print(f"\n{'='*50} Phase2 Focus n={n} {'='*50}", flush=True)

        a = tracker.best.copy().astype(np.float64)

        # Aggressive LSE with increasing tau
        for tau_s in [50, 100, 200, 400]:
            if time.monotonic() > deadline - 80:
                break
            a = symmetric_lse_descent(a, 300, deadline - 80, tracker,
                                      tau_start=tau_s, tau_max=tau_s * 3)

        # LP polishing with tightening schedule
        for tight in [0.005, 0.01, 0.015, 0.02]:
            if time.monotonic() > deadline - 60:
                break
            a = run_lp_opt(a, 15, deadline - 60, tracker, tighten=tight)

        # Coordinate descent
        if time.monotonic() < deadline - 40:
            a = coordinate_descent_sym(a, 5000, deadline - 40, tracker)

        tracker.report(f"Phase2 ({time.monotonic() - t0:.1f}s)")

    # === Phase 3: SLSQP polish ===
    if tracker.best is not None and time.monotonic() < deadline - 30:
        print(f"\n{'='*50} Phase3 SLSQP {'='*50}", flush=True)
        a_sl = slsqp_polish(tracker.best, max_time=45)
        if a_sl is not None:
            J_sl = compute_J(a_sl)
            if J_sl < tracker.best_score:
                tracker.update(a_sl)
                print(f"  SLSQP improved to J={J_sl:.8f}", flush=True)

    # === Phase 4: Final coordinate descent + LSE ===
    if tracker.best is not None and time.monotonic() < deadline - 10:
        print(f"\n{'='*50} Phase4 Final Polish {'='*50}", flush=True)
        a = tracker.best.copy()
        a = coordinate_descent_sym(a, 3000, deadline - 15, tracker)
        if time.monotonic() < deadline - 5:
            a = symmetric_lse_descent(a, 100, deadline - 5, tracker,
                                      tau_start=200, tau_max=500)
        if time.monotonic() < deadline - 2:
            a = coordinate_descent_sym(a, 1000, deadline - 2, tracker)

    # === Phase 5: Try different n from best ===
    if tracker.best is not None and time.monotonic() < deadline - 20:
        print(f"\n{'='*50} Phase5 Resample Explore {'='*50}", flush=True)
        best_n = tracker.best.size
        for n_try in [best_n * 2, best_n + 128, best_n + 64]:
            if n_try > 2048 or time.monotonic() > deadline - 20:
                break
            a_res = resample_symmetric(tracker.best, n_try)
            a_res = symmetric_lse_descent(a_res, 200, deadline - 10, tracker,
                                          tau_start=100, tau_max=400)
            if time.monotonic() < deadline - 5:
                a_res = coordinate_descent_sym(a_res, 2000, deadline - 5, tracker)

    # Fallback
    if tracker.best is None:
        a_fb = tapered_init(256, rng, kind=0)
        tracker.update(a_fb)
        tracker.best = a_fb
        tracker.best_score = compute_J(a_fb)

    print(f"\n{'='*50} FINAL {'='*50}", flush=True)
    print(f"J={tracker.best_score:.8f} n={tracker.best.size} "
          f"time={time.monotonic() - tracker.t_start:.1f}s", flush=True)

    return [float(x) for x in tracker.best]
