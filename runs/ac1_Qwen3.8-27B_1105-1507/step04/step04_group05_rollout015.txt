The smoothed gradient is computed via FFT in O(n log n), enabling many fast iterations.

import numpy as np
from scipy.optimize import linprog
from scipy.sparse import lil_matrix
import time


def fast_conv(a):
    """Full self-convolution via FFT."""
    n = len(a)
    s = 1
    while s < 2 * n - 1:
        s <<= 1
    fa = np.fft.rfft(a, s)
    return np.fft.irfft(fa * fa, s)[:2 * n - 1]


def compute_J_local(a):
    """Compute J(a) = 2*n*max(conv(a,a))/(sum(a))^2 locally (assumes a >= 0)."""
    S = a.sum()
    if S < 1e-10:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (S * S))


def smooth_objective_and_grad(a, beta):
    """Compute smoothed objective and its gradient.
    f(a) = (2n/beta) * log(sum_k exp(beta * (a*a)_k))
    grad_i = (4n/beta) * sum_j W_{i+j} * a[j] where W_k = exp(beta*b_k)/sum(exp(beta*b))
    """
    n = len(a)
    b = fast_conv(a)
    b_shifted = b - b.max()
    exp_b = np.exp(beta * b_shifted)
    E = exp_b.sum()
    W = exp_b / E  # normalized, length 2n-1
    obj = (2.0 * n / beta) * (np.log(E) + beta * b.max())
    # Gradient: grad_i = (4n/beta) * sum_j W[i+j] * a[j] for i=0..n-1
    cross = np.correlate(W, a, mode='valid')
    grad = (4.0 * n / beta) * cross
    return obj, grad


def adam_step(a, grad, m, v, t, lr, b1=0.9, b2=0.999, eps=1e-8):
    """One Adam update step. Returns (a_new, m_new, v_new)."""
    m = b1 * m + (1.0 - b1) * grad
    v = b2 * v + (1.0 - b2) * grad * grad
    m_hat = m / (1.0 - b1 ** t)
    v_hat = v / (1.0 - b2 ** t)
    a_new = a - lr * m_hat / (np.sqrt(v_hat) + eps)
    return a_new, m, v


def project_simplex(a):
    """Project onto the probability simplex {x: sum(x)=1, x>=0}."""
    n = len(a)
    u = np.sort(a)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - 1))[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(a - theta, 0)

def build_A_sparse(a):
    """Build sparse constraint matrix for LP: conv(a,g)_k <= M for all k."""
    n = len(a)
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        jl = max(0, k - n + 1)
        jh = min(n - 1, k)
        if jl <= jh:
            cnt = jh - jl + 1
            rows.extend([k] * cnt)
            cols.extend(range(jl, jh + 1))
            # conv(a,g)_k = sum_j a[j]*g[k-j], so coefficient of g[j] is a[k-j]
            for j in range(jl, jh + 1):
                vals.append(a[k - j])
    A = lil_matrix((2 * n - 1, n))
    for idx in range(len(rows)):
        A[rows[idx], cols[idx]] = vals[idx]
    return A.tocsr()


def lp_best_response(a, M):
    """LP: max sum(g) s.t. conv(a,g)_k <= M for all k, g >= 0."""
    n = len(a)
    c = -np.ones(n)
    A = build_A_sparse(a)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    try:
        r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
        if r.success and r.x is not None:
            g = np.maximum(r.x, 0.0)
            if g.sum() > 1e-10:
                return g
    except Exception:
        pass
    return None

def coord_descent_1d(a, i, b, S):

    coeffs = []

        if new_S < 1e-10:
            return 1e10

            d_pt = c_pt
            fd = fc
            c_pt = hi - gr * (hi - lo)
            fc = j_of_x(c_pt)
        else:
            lo = c_pt
            c_pt = d_pt
            fc = fd
            d_pt = lo + gr * (hi - lo)
            fd = j_of_x(d_pt)
    
    x_opt = (lo + hi) / 2.0
    return max(0.0, x_opt)


def coord_descent_pass(a, rng=None):
    """One full pass of coordinate descent with incremental updates."""
    a = np.maximum(a.astype(np.float64), 0.0)
    S = a.sum()
    if S < 1e-10:
        return a, float('inf')
    n = len(a)
    b = fast_conv(a)
    current_J = 2.0 * n * b.max() / (S * S)
    
    indices = list(range(n))
    if rng is not None:
        rng.shuffle(indices)
    
    for i in indices:
        new_val = coord_descent_1d(a, i, b, S)
        if abs(new_val - a[i]) > 1e-12:
            old_val = a[i]
            a[i] = new_val
            S = S - old_val + new_val
            # Update convolution
            k_start = max(0, i)
            k_end = min(2 * n - 1, i + n - 1)
            for k in range(k_start, k_end + 1):
                ki = k - i
                if 0 <= ki < n:
                    if k == 2 * i:
                        b[k] = b[k] - old_val * old_val + new_val * new_val
                    else:
                        b[k] = b[k] - 2.0 * old_val * a[ki] + 2.0 * new_val * a[ki]
    
    # Verify and return
    b_check = fast_conv(a)
    new_J = 2.0 * n * b_check.max() / (S * S)
    return a, new_J


def cd_phase(a, deadline, tracker, max_passes=50):
    """Phase 3: Coordinate descent iterations."""
    a = np.maximum(a.astype(np.float64), 0.0)
    a = a / a.sum()
    rng = np.random.default_rng(12345)
    current_J = compute_J_local(a)
    tracker.update(a)
    
    no_improve = 0
    for p in range(max_passes):
        if time.monotonic() > deadline:
            break
        a, new_J = coord_descent_pass(a, rng)
        a = a / a.sum()
        new_J = compute_J_local(a)
        if new_J < current_J - 1e-12:
            current_J = new_J
            no_improve = 0
            tracker.update(a)
        else:
            no_improve += 1
            if no_improve > 5:
                break
    
    tracker.update(a)
    return a


def resample_seq(src, dest_n):
    """Resample 1D sequence to new length via linear interpolation."""
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)


class ScoreTracker:
    def __init__(self):
        self.best = None
        self.best_score = float('inf')
        self.t_start = time.monotonic()
    
    def update(self, a):
        a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
        if a.size == 0 or a.sum() < 0.01:
            return
        score = float(evaluate_sequence(a.tolist()))
        if np.isfinite(score) and score < self.best_score:
            self.best = a.copy()
            self.best_score = score
            elapsed = time.monotonic() - self.t_start
            print(f"  * NEW BEST: J={score:.8f} n={a.size} t={elapsed:.0f}s", flush=True)
    
    def report(self, label=""):
        if self.best is not None:
            print(f"  {label} best: J={self.best_score:.8f} n={self.best.size} t={time.monotonic()-self.t_start:.0f}s", flush=True)


def structured_init(n, rng):
    """Generate structured initial sequences."""
    t = np.linspace(0, 1, n, endpoint=False)
    candidates = []
    
    # Cosine window
    candidates.append(0.5 + 0.5 * np.cos(2 * np.pi * t))
    # Raised cosine
    candidates.append(0.5 - 0.5 * np.cos(2 * np.pi * t))
    # Two peaks
    candidates.append(0.3 + 0.7 * np.abs(np.cos(np.pi * t)))
    # Gaussian-like center
    candidates.append(np.exp(-2.0 * (t - 0.5) ** 2) + 0.1)
    # Uniform
    candidates.append(np.ones(n))
    # Slight modulation
    candidates.append(1.0 + 0.3 * np.cos(2 * np.pi * 3 * t))
    # Random
    candidates.append(rng.uniform(0.5, 1.5, n))
    
    return candidates


def propose_candidate(seed=42, budget_s=1500, **kwargs):

    # Try to load parent

        p = np.asarray(height_sequence_1, dtype=np.float64)
        if p.ndim == 1 and p.size > 0 and np.all(np.isfinite(p)):
            parent = np.maximum(p, 0.0)
            if parent.sum() > 0.01:
                tracker.update(parent)
                print(f"Parent loaded: n={parent.size}, J={tracker.best_score:.8f}", flush=True)
            else:
                parent = None
    except (NameError, AttributeError):
        pass
    
    if parent is None:
        parent = np.ones(386)
        tracker.update(parent)
        print(f"No parent, using uniform n=386, J={tracker.best_score:.8f}", flush=True)
    
    n_parent = parent.size

    print(f"Phase 1: Smoothed Adam (n={n_parent})", flush=True)
    print(f"{'='*50}", flush=True)
    t0 = time.monotonic()
    
    # Split phase 1 into sub-phases with increasing beta
    p1_budget = min(500.0, (deadline - time.monotonic()) * 0.35)

    # === Phase 4: Alternating LP + CD + Adam cycles ===
    if time.monotonic() < deadline - 100:

        print(f"Phase 4: Alternating Refinement", flush=True)
        print(f"{'='*50}", flush=True)
        t0 = time.monotonic()
        
        cycle_deadline = min(deadline - 100, time.monotonic() + 200)

            if tracker.best_score > 1.503:

                if time.monotonic() < deadline - 15:

    return [float(x) for x in tracker.best]
