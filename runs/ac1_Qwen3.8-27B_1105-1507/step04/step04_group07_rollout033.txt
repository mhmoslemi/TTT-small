"""
Minimizes J(a) = 2*n*max(convolve(a,a))/(sum(a))^2 using:
1. LP-based alternating optimization with adaptive tightening
2. LSE (log-sum-exp) smooth approximation gradient descent
3. Diversified spectral initialization
4. Multi-length search with parent seeding
5. Hybrid refinement combining LP and gradient methods
"""

import numpy as np
from scipy.optimize import linprog, minimize
from scipy.sparse import csr_matrix
import time

# Cache for sparsity patterns
_spar_cache = {}

def get_sparsity(n):
    """Get sparse matrix structure for convolution constraints."""
    if n not in _spar_cache:
        rows = []
        cols = []
        for k in range(2 * n - 1):
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            rows.extend([k] * (je - js + 1))
            cols.extend(range(js, je + 1))
        _spar_cache[n] = (np.array(rows, dtype=np.int32), np.array(cols, dtype=np.int32), len(rows))
    return _spar_cache[n]

def fast_conv(a):
    """Fast self-convolution using FFT."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]

def compute_J(a):
    """Compute the objective function J(a)."""
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))

def lp_best_response(a, M, time_limit=5):
    """Solve LP: max sum(b) s.t. (a*b)[k] <= M, b >= 0."""
    n = len(a)
    rows, cols, nnz = get_sparsity(n)
    vals = np.empty(nnz)
    for i in range(nnz):
        k = rows[i]
        j = cols[i]
        idx = k - j
        if 0 <= idx < n:
            vals[i] = a[idx]
        else:
            vals[i] = 0.0
    A = csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                options={'presolve': True, 'time_limit': time_limit})
    if r.success and r.x is not None:
        g = np.maximum(r.x, 0.0)
        if g.sum() > 1e-10:
            return g
    return None

def lse_logsumexp(b, tau):
    """Compute log-sum-exp of tau*b."""
    m = b.max()
    return m + np.log(np.sum(np.exp(tau * (b - m)))) / tau

def lse_gradient(a, tau):
    """Compute gradient of LSE approximation of max(a*a)."""
    n = len(a)
    b = fast_conv(a)
    b_max = b.max()
    exp_b = np.exp(tau * (b - b_max))
    p = exp_b / exp_b.sum()
    # grad_j = 2 * sum_k p[k] * a[k-j] (for valid k-j)
    # This is 2 * (p * a)[j] in the circular sense, but we need linear
    # (p_a)[j] = sum_k p[k] * a[k-j] for 0 <= k-j < n
    # = sum_j' p[j+j'] * a[j'] where j' = k-j
    # This is a cross-correlation
    size = 1
    while size < 3 * n:
        size <<= 1
    fp = np.fft.rfft(p, size)
    fa = np.fft.rfft(a, size)
    # cross-corr: we want sum_k p[k] * a[k-j]
    # = ifft(conj(FFT(p)) * FFT(a)) ... no
    # Actually: (p * a_flipped)[j] = sum_k p[k] * a_flipped[j-k] = sum_k p[k] * a[k-j]
    # So it's the convolution of p with a_flipped, evaluated at j
    a_flipped = a[::-1]
    faf = np.fft.rfft(a_flipped, size)
    corr = np.fft.irfft(fp * faf, size)
    grad = 2.0 * corr[:n]
    return grad

def lse_descent(a, max_iters, deadline, tracker, tau_start=20.0, tau_end=200.0):
    """LSE-based gradient descent for the smooth approximation."""
    a = np.maximum(a.astype(np.float64), 1e-8)
    a = a / a.sum()
    n = len(a)
    current_J = compute_J(a)
    tracker.update(a)
    
    tau = tau_start
    lr = 0.5 / n
    momentum = 0.9
    v = np.zeros(n)
    
    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        b = fast_conv(a)
        b_max = b.max()
        exp_b = np.exp(tau * (b - b_max))
        p = exp_b / exp_b.sum()
        
        # Gradient of LSE w.r.t. a
        a_flipped = a[::-1]
        size = 1
        while size < 3 * n:
            size <<= 1
        fp = np.fft.rfft(p, size)
        faf = np.fft.rfft(a_flipped, size)
        corr = np.fft.irfft(fp * faf, size)
        grad = 2.0 * corr[:n]
        
        # Add gradient of the normalization penalty (keep sum = 1)
        # Actually we'll just project after each step
        
        v = momentum * v + (1.0 - momentum) * grad
        a_new = a - lr * v
        a_new = np.maximum(a_new, 1e-8)
        a_new = a_new / a_new.sum()
        
        J_new = compute_J(a_new)
        if J_new < current_J * (1 - 1e-10):
            a = a_new
            current_J = J_new
            tau = min(tau_end, tau * 1.02)
            lr = min(lr * 1.01, 2.0 / n)
        else:
            lr *= 0.7
            v *= 0.5
            if lr < 1e-18:
                break
        if it % 50 == 0:
            tracker.update(a)
    
    tracker.update(a)
    return a

def lp_optimize(a, max_iters, deadline, tracker, tighten=0.005):
    """LP-based alternating optimization."""
    a = np.maximum(a.astype(np.float64), 1e-8)
    a = a / a.sum()
    n = len(a)
    current_J = compute_J(a)
    tracker.update(a)
    
    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        b = fast_conv(a)
        M = float(b.max())
        
        # Try tightening
        M_tight = M * (1.0 - tighten)
        g = lp_best_response(a, M_tight, time_limit=3)
        if g is None:
            g = lp_best_response(a, M, time_limit=3)
        if g is None:
            break
        g = np.maximum(g, 0.0)
        gs = g.sum()
        if gs < 1e-10:
            break
        g = g / gs
        
        # Try multiple mixing ratios
        best_a = None
        best_J = current_J
        progress = it / max(1, max_iters)
        base_t = 0.3 * (1.0 - 0.5 * progress)
        for t in [base_t, base_t * 1.5, base_t * 0.5, 0.7, 0.2]:
            if t > 1.0 or t <= 0:
                continue
            a_trial = (1.0 - t) * a + t * g
            a_trial = np.maximum(a_trial, 0.0)
            ss = a_trial.sum()
            if ss < 1e-10:
                continue
            a_trial = a_trial / ss
            J_trial = compute_J(a_trial)
            if J_trial < best_J - 1e-12:
                best_J = J_trial
                best_a = a_trial
        if best_a is None:
            # Try smaller tightening
            break
        a = best_a
        current_J = best_J
        if it % 5 == 0:
            tracker.update(a)
    
    tracker.update(a)
    return a

def generate_inits(n, rng, parent=None):
    """Generate diverse initial sequences."""
    inits = []
    t = np.linspace(0, 1, n, endpoint=False)
    
    # Parent resampled
    if parent is not None and parent.size > 0:
        p = parent.copy()
        if p.size != n:
            t_old = np.linspace(0, 1, p.size)
            p = np.interp(t, t_old, p)
        p = np.maximum(p, 0.001)
        p = p / p.sum()
        inits.append(p)
        inits.append(p[::-1].copy())
    
    # Uniform
    inits.append(np.ones(n) / n)
    
    # Sine window
    a = np.sin(np.pi * (np.arange(n) + 0.5) / n)
    inits.append(np.maximum(a, 0.01) / np.maximum(a, 0.01).sum())
    
    # Cosine window (Hann)
    a = 0.5 * (1 + np.cos(2 * np.pi * t))
    inits.append(np.maximum(a, 0.01) / np.maximum(a, 0.01).sum())
    
    # Gaussian
    center = n / 2
    for sigma in [n/4, n/6, n/8]:
        a = np.exp(-0.5 * ((np.arange(n) - center) / sigma) ** 2)
        inits.append(a / a.sum())
    
    # Flat spectrum (random phases)
    for _ in range(3):
        half = n // 2 + 1
        angles = rng.uniform(0, 2 * np.pi, half)
        A = np.zeros(n, dtype=complex)
        A[:half] = np.exp(1j * angles[:half])
        for k in range(1, n - n // 2):
            A[k] = np.conj(A[n - k])
        a = np.fft.ifft(A).real
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s > 1e-10:
            inits.append(a / s)
        else:

def subgradient_step(a, n_active=None, epsilon=None):
    """Compute subgradient of max(a*a) and take a step."""
    n = len(a)
    b = fast_conv(a)
    M = b.max()
    if epsilon is None:
        epsilon = 1e-8 * M
    if n_active is None:
        n_active = 5
    active = np.where(b >= M - epsilon)[0]
    if len(active) > n_active:
        # Use top n_active
        idx = np.argsort(b)[::-1][:n_active]
        active = idx
    grad = np.zeros(n)
    for k in active:
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        for j in range(js, je + 1):
            grad[j] += a[k - j]
    grad /= len(active)
    return grad, M

def subgradient_descent(a, max_iters, deadline, tracker):
    """Subgradient descent with variable step size."""
    a = np.maximum(a.astype(np.float64), 1e-8)
    a = a / a.sum()
    n = len(a)
    current_J = compute_J(a)
    tracker.update(a)
    
    # Polyak step size with running average of best
    best_J = current_J
    lr = 1.0 / n
    rng = np.random.default_rng(123)
    
    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        grad, M = subgradient_step(a)
        
        # Direction: reduce max by moving against gradient, but also toward uniform
        # to maintain feasibility
        direction = grad - M * a  # This pushes toward reducing the max component
        
        a_new = a - lr * direction
        a_new = np.maximum(a_new, 1e-8)
        a_new = a_new / a_new.sum()
        
        J_new = compute_J(a_new)
        if J_new < current_J:
            a = a_new
            current_J = J_new
            best_J = min(best_J, J_new)
            lr = min(lr * 1.05, 5.0 / n)
        else:
            lr *= 0.8
            if lr < 1e-18:
                break
        if it % 20 == 0:
            tracker.update(a)
    
    tracker.update(a)
    return a

def spectral_shaping(a, deadline, tracker):
    """Try to flatten the power spectrum of a while maintaining a >= 0."""
    a = np.maximum(a.astype(np.float64), 1e-8)
    a = a / a.sum()
    n = len(a)
    current_J = compute_J(a)
    tracker.update(a)
    
    # Current power spectrum
    fa = np.fft.rfft(a)
    power = np.abs(fa) ** 2
    
    # Target: flat power spectrum (but maintaining DC)
    # We'll iteratively adjust the spectrum
    lr = 0.1
    rng = np.random.default_rng(456)
    
    for it in range(200):
        if time.monotonic() > deadline:
            break
        fa = np.fft.rfft(a)
        power = np.abs(fa) ** 2
        p_max = power.max()
        p_mean = power[1:].mean() if n > 1 else power[0]
        
        # Target power: DC stays, others toward mean
        target = power.copy()
        target[1:] = 0.5 * power[1:] + 0.5 * p_mean
        
        # Adjust: new_fa = fa * (target/power)^0.5 (for non-DC)
        ratio = np.ones_like(fa)
        mask = power > 1e-20
        ratio[1:][mask[1:]] = (target[1:][mask[1:]] / power[1:][mask[1:]]) ** 0.5
        ratio[1:] = np.clip(ratio[1:], 0.5, 2.0)
        
        new_fa = fa * (1 - lr + lr * ratio)
        # Ensure Hermitian symmetry
        new_a = np.fft.irfft(new_fa, n)
        new_a = np.maximum(new_a, 1e-8)
        new_a = new_a / new_a.sum()
        
        J_new = compute_J(new_a)
        if J_new < current_J:

                break

    return a

def hybrid_optimize(a, deadline, tracker):
    """Alternate between LP, LSE gradient, and subgradient methods."""
    a = np.maximum(a.astype(np.float64), 1e-8)
    a = a / a.sum()
    n = len(a)
    current_J = compute_J(a)
    tracker.update(a)
    
    phase = 'lp'
    phase_count = 0
    
    while time.monotonic() < deadline:
        phase_time = min(60, deadline - time.monotonic())
        
        if phase == 'lp':
            a = lp_optimize(a, max_iters=20, deadline=time.monotonic() + phase_time,
                          tracker=tracker, tighten=0.005)
            phase = 'lse'
        elif phase == 'lse':
            a = lse_descent(a, max_iters=200, deadline=time.monotonic() + phase_time,
                          tracker=tracker, tau_start=30, tau_end=300)
            phase = 'subgrad'
        elif phase == 'subgrad':
            a = subgradient_descent(a, max_iters=100, deadline=time.monotonic() + phase_time,
                                   tracker=tracker)
            phase = 'lp'
        
        J_new = compute_J(a)
        if J_new >= current_J * (1 + 1e-6):
            # No progress, try different phase
            if phase == 'lp':
                phase = 'lse'
            elif phase == 'lse':
                phase = 'subgrad'
            else:
                phase = 'lp'
            current_J = J_new
            # Add small perturbation to escape
            rng = np.random.default_rng(int(time.monotonic() * 1000) % 10000)

            a = np.maximum(a, 1e-8)

def coordinate_refinement(a, deadline, tracker):
    """Try coordinate-wise adjustments."""
    a = np.maximum(a.astype(np.float64), 1e-8)
    a = a / a.sum()
    n = len(a)
    current_J = compute_J(a)
    tracker.update(a)
    rng = np.random.default_rng(789)
    
    for step in range(5000):
        if time.monotonic() > deadline:
            break
        j = rng.integers(0, n)
        improved = False
        for scale in [0.5, 1.5, 0.2, 2.0, 0.7, 1.3]:
            a_trial = a.copy()
            a_trial[j] *= scale
            a_trial = np.maximum(a_trial, 1e-8)
            ss = a_trial.sum()
            if ss < 1e-10:
                continue
            a_trial = a_trial / ss
            J_trial = compute_J(a_trial)
            if J_trial < current_J - 1e-12:
                a = a_trial
                current_J = J_trial
                improved = True
                break
        if not improved and step % 100 == 0:
            # Try pair updates
            j1 = rng.integers(0, n)
            j2 = rng.integers(0, n)
            if j1 != j2:
                for delta in [-0.3, -0.1, 0.1, 0.3]:
                    a_trial = a.copy()
                    moved = a_trial[j1] * abs(delta)
                    a_trial[j1] -= moved
                    a_trial[j2] += moved
                    a_trial = np.maximum(a_trial, 1e-8)
                    ss = a_trial.sum()
                    if ss < 1e-10:
                        continue
                    a_trial = a_trial / ss
                    J_trial = compute_J(a_trial)
                    if J_trial < current_J - 1e-12:
                        a = a_trial
                        current_J = J_trial
                        improved = True
                        break
        if step % 200 == 0:
            tracker.update(a)
    
    tracker.update(a)
    return a

"""
Minimizes J(a) = 2*n*max(convolve(a,a))/(sum(a))^2 using:
1. LP-based alternating optimization with adaptive tightening
2. LSE (log-sum-exp) smooth gradient descent
3. Spectral shaping to flatten power spectrum
4. Diverse initialization (windows, flat-spectrum, multi-frequency)
5. Multi-length search with parent seeding
6. Hybrid refinement and simulated annealing escape

import numpy as np
from scipy.optimize import linprog
from scipy.sparse import csr_matrix
import time

_spar_cache = {}

def get_sparsity(n):
    if n not in _spar_cache:
        rows = []
        cols = []
        for k in range(2 * n - 1):
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            rows.extend([k] * (je - js + 1))
            cols.extend(range(js, je + 1))
        _spar_cache[n] = (np.array(rows, dtype=np.int32), np.array(cols, dtype=np.int32), len(rows))
    return _spar_cache[n]

def fast_conv(a):
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]

def compute_J(a):
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))

def lp_best_response(a, M, time_limit=4):
    n = len(a)
    rows, cols, nnz = get_sparsity(n)
    vals = np.empty(nnz)
    for i in range(nnz):
        k = rows[i]
        j = cols[i]
        idx = k - j
        if 0 <= idx < n:
            vals[i] = a[idx]
        else:
            vals[i] = 0.0
    A = csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                options={'presolve': True, 'time_limit': time_limit})
    if r.success and r.x is not None:
        g = np.maximum(r.x, 0.0)
        if g.sum() > 1e-10:
            return g
    return None

def lp_optimize(a, max_iters, deadline, tracker, tighten=0.005):
    a = np.maximum(a.astype(np.float64), 1e-8)
    a = a / a.sum()
    n = len(a)
    current_J = compute_J(a)
    tracker.update(a)

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        b = fast_conv(a)
        M = float(b.max())
        M_tight = M * (1.0 - tighten)
        g = lp_best_response(a, M_tight, time_limit=3)
        if g is None:
            g = lp_best_response(a, M, time_limit=3)
        if g is None:
            break
        g = np.maximum(g, 0.0)
        gs = g.sum()
        if gs < 1e-10:
            break
        g = g / gs

        best_a = None
        best_J = current_J
        progress = it / max(1, max_iters)
        base_t = 0.4 * (1.0 - 0.5 * progress)
        for t in [base_t, base_t * 1.5, base_t * 0.5, 0.8, 0.25]:
            if t > 1.0 or t <= 0:
                continue
            a_trial = (1.0 - t) * a + t * g
            a_trial = np.maximum(a_trial, 0.0)
            ss = a_trial.sum()
            if ss < 1e-10:
                continue
            a_trial = a_trial / ss
            J_trial = compute_J(a_trial)
            if J_trial < best_J - 1e-12:
                best_J = J_trial
                best_a = a_trial
        if best_a is None:
            break
        a = best_a
        current_J = best_J
        if it % 5 == 0:
            tracker.update(a)

    tracker.update(a)
    return a

def lse_gradient_vector(a, tau):
    n = len(a)
    b = fast_conv(a)
    b_max = b.max()
    exp_b = np.exp(tau * (b - b_max))
    p = exp_b / exp_b.sum()
    a_flipped = a[::-1]
    size = 1
    while size < n + len(p) - 1:
        size <<= 1
    faf = np.fft.rfft(a_flipped, size)
    fp = np.fft.rfft(p, size)
    conv = np.fft.irfft(faf * fp, size)
    grad = conv[n - 1: 2 * n - 1]
    return 2.0 * grad

def lse_descent(a, max_iters, deadline, tracker, tau_start=20.0, tau_end=300.0):
    a = np.maximum(a.astype(np.float64), 1e-8)
    a = a / a.sum()
    n = len(a)
    current_J = compute_J(a)
    tracker.update(a)

    tau = tau_start
    lr = 0.3 / n
    momentum = 0.85
    v = np.zeros(n)

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        grad = lse_gradient_vector(a, tau)
        v = momentum * v + (1.0 - momentum) * grad
        a_new = a - lr * v
        a_new = np.maximum(a_new, 1e-8)
        a_new = a_new / a_new.sum()

        J_new = compute_J(a_new)
        if J_new < current_J * (1 - 1e-12):
            a = a_new
            current_J = J_new
            tau = min(tau_end, tau * 1.03)
            lr = min(lr * 1.02, 3.0 / n)
        else:
            lr *= 0.6
            v *= 0.3
            if lr < 1e-18:
                break
        if it % 30 == 0:
            tracker.update(a)

    tracker.update(a)
    return a

def subgradient_descent(a, max_iters, deadline, tracker):
    a = np.maximum(a.astype(np.float64), 1e-8)
    a = a / a.sum()
    n = len(a)
    current_J = compute_J(a)
    tracker.update(a)

    lr = 0.5 / n
    best_J = current_J

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        b = fast_conv(a)
        M = b.max()
        epsilon = 1e-6 * M
        active = np.where(b >= M - epsilon)[0]
        if len(active) > 10:
            active = np.argsort(b)[::-1][:10]

        grad = np.zeros(n)
        for k in active:
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            for j in range(js, je + 1):
                grad[j] += a[k - j]
        grad /= len(active)

        direction = grad - M * a
        a_new = a - lr * direction
        a_new = np.maximum(a_new, 1e-8)
        a_new = a_new / a_new.sum()

        J_new = compute_J(a_new)
        if J_new < current_J:
            a = a_new
            current_J = J_new
            best_J = min(best_J, J_new)
            lr = min(lr * 1.03, 4.0 / n)
        else:
            lr *= 0.75
            if lr < 1e-18:
                break
        if it % 20 == 0:
            tracker.update(a)

    tracker.update(a)
    return a

def spectral_shaping(a, deadline, tracker):
    a = np.maximum(a.astype(np.float64), 1e-8)
    a = a / a.sum()
    n = len(a)
    current_J = compute_J(a)
    tracker.update(a)

    lr = 0.05
    for it in range(300):
        if time.monotonic() > deadline:
            break
        fa = np.fft.rfft(a)
        power = np.abs(fa) ** 2
        if n <= 1:
            break
        p_mean_nonzero = power[1:].mean()
        target = power.copy()
        target[1:] = 0.5 * power[1:] + 0.5 * p_mean_nonzero

        ratio = np.ones(len(fa))
        mask = power > 1e-20
        ratio_nonzero = np.ones(power[1:].shape)
        mask1 = power[1:] > 1e-20
        ratio_nonzero[mask1] = (target[1:][mask1] / power[1:][mask1]) ** 0.5
        ratio[1:] = np.clip(ratio_nonzero, 0.3, 3.0)

        new_fa = fa * (1.0 - lr + lr * ratio)
        new_a = np.fft.irfft(new_fa, n)
        new_a = np.maximum(new_a, 1e-8)
        new_a = new_a / new_a.sum()

        J_new = compute_J(new_a)
        if J_new < current_J * (1 - 1e-12):
            a = new_a
            current_J = J_new
            lr = min(lr * 1.05, 0.3)
        else:
            lr *= 0.7
            if lr < 1e-12:
                break
        if it % 50 == 0:
            tracker.update(a)

    tracker.update(a)
    return a

def generate_inits(n, rng, parent=None):
    inits = []
    t = np.linspace(0, 1, n, endpoint=False)

    if parent is not None and parent.size > 0:
        p = parent.copy()
        if p.size != n:
            t_old = np.linspace(0, 1, p.size)
            p = np.interp(t, t_old, p)
        p = np.maximum(p, 0.001)
        p = p / p.sum()
        inits.append(p)
        inits.append(p[::-1].copy())

    inits.append(np.ones(n) / n)

    # Sine window
    a = np.sin(np.pi * (np.arange(n) + 0.5) / n)
    inits.append(np.maximum(a, 0.01) / np.maximum(a, 0.01).sum())

    # Hann window
    a = 0.5 * (1.0 + np.cos(2.0 * np.pi * t))
    inits.append(np.maximum(a, 0.01) / np.maximum(a, 0.01).sum())

    # Blackman window
    a = 0.42 + 0.5 * np.cos(2.0 * np.pi * t) + 0.08 * np.cos(4.0 * np.pi * t)
    inits.append(np.maximum(a, 0.01) / np.maximum(a, 0.01).sum())

    # Gaussians
    center = (n - 1) / 2.0
    for sigma_frac in [0.2, 0.3, 0.4]:
        sigma = sigma_frac * n
        a = np.exp(-0.5 * ((np.arange(n) - center) / sigma) ** 2)
        inits.append(a / a.sum())

    # Flat spectrum (random phases)
    for _ in range(4):
        half = n // 2 + 1
        angles = rng.uniform(0, 2.0 * np.pi, half)
        A = np.zeros(n, dtype=complex)
        A[:half] = np.exp(1j * angles[:half])
        for k in range(1, n - n // 2):
            A[k] = np.conj(A[n - k])
        a = np.fft.ifft(A).real
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s > 1e-10:
            inits.append(a / s)
        else:
            inits.append(np.ones(n) / n)

    # Shaped power spectrum
    for _ in range(3):
        half = n // 2 + 1
        freqs = np.arange(half)
        shape_type = rng.integers(0, 3)
        if shape_type == 0:
            amp = np.ones(half)
        elif shape_type == 1:
            amp = 0.5 * (1.0 + np.cos(np.pi * freqs / max(1, half - 1)))
        else:
            amp = 1.0 / (1.0 + freqs / max(1, half * 0.4))
        phases = rng.uniform(0, 2.0 * np.pi, half)
        A = amp * np.exp(1j * phases)
        Ac = np.zeros(n, dtype=complex)
        Ac[:half] = A
        for k in range(1, n - n // 2):
            Ac[k] = np.conj(Ac[n - k])
        a = np.fft.ifft(Ac).real
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s > 1e-10:
            inits.append(a / s)
        else:
            inits.append(np.ones(n) / n)

    # Multi-frequency
    n_coeff = min(6, n // 4)
    coeffs = rng.uniform(0.1, 0.4, n_coeff)
    freqs_sel = rng.permutation(n_coeff)[:min(4, n_coeff)] * 2 + 1
    a = np.ones(n)
    for i in range(min(4, n_coeff)):
        a += coeffs[i] * np.cos(2.0 * np.pi * freqs_sel[i] * t + rng.uniform(0, 2.0 * np.pi))
    inits.append(np.maximum(a, 0.01) / np.maximum(a, 0.01).sum())

    # Two-peak Gaussian
    a = np.exp(-6.0 * (t - 0.3) ** 2) + np.exp(-6.0 * (t - 0.7) ** 2) + 0.05
    inits.append(a / a.sum())

    # Raised cosine powers
    for p in [2, 3, 5]:
        a = np.maximum(0.5 + 0.5 * np.cos(2.0 * np.pi * t), 0.0) ** p
        inits.append(np.maximum(a, 0.001) / np.maximum(a, 0.001).sum())

    # Downward triangle
    a = 1.0 - 2.0 * np.abs(t - 0.5)
    inits.append(np.maximum(a, 0.01) / np.maximum(a, 0.01).sum())

    # Sparse smoothed
    n_spikes = max(4, n // 20)
    positions = rng.choice(n, size=n_spikes, replace=False)
    sparse = np.zeros(n)
    sparse[positions] = 1.0
    kernel_size = max(3, n // 20)
    kernel = np.hanning(kernel_size) if kernel_size >= 3 else np.ones(kernel_size)
    kernel = kernel / kernel.sum()
    a = np.convolve(sparse, kernel, mode='same')
    inits.append(np.maximum(a, 0.005) / np.maximum(a, 0.005).sum())

    result = []
    for x in inits:
        x = np.maximum(x, 0.001).astype(np.float64)
        x = x / x.sum()
        result.append(x)
    return result

                break
            t_init_end = min(t_end - 8, time.monotonic() + init_budget)

            a = lp_optimize(a0, max_iters=12, deadline=t_init_end - 4,
                           tracker=tracker, tighten=0.01)
            if time.monotonic() < t_init_end - 3:
                a = lse_descent(a, max_iters=60, deadline=t_init_end - 3,
                               tracker=tracker, tau_start=20, tau_end=150)

            J = compute_J(a)
            if J < best_J_n:
                best_J_n = J
                best_for_n = a.copy()

        if best_for_n is not None:
            tracker.update(best_for_n)
            print(f"  n={n}: J={best_J_n:.8f} ({time.monotonic()-t_start:.1f}s)", flush=True)

    # Phase 2: Focused optimization
    if tracker.best is not None and time.monotonic() < deadline - 200:
        t0 = time.monotonic()
        n = tracker.best.size
        print(f"\n{'='*60}", flush=True)
        print(f"Phase 2: Focused optimization n={n}", flush=True)
        print(f"{'='*60}", flush=True)

        a = tracker.best.copy().astype(np.float64)

        for tight in [0.02, 0.015, 0.01, 0.007, 0.005, 0.003, 0.002, 0.001]:
            if time.monotonic() > deadline - 200:
                break
            a = lp_optimize(a, max_iters=15, deadline=deadline - 200,
                           tracker=tracker, tighten=tight)

        if time.monotonic() < deadline - 150:
            a = lse_descent(a, max_iters=400, deadline=deadline - 150,
                           tracker=tracker, tau_start=50, tau_end=500)

        if time.monotonic() < deadline - 120:
            a = spectral_shaping(a, deadline=deadline - 120, tracker=tracker)

        if time.monotonic() < deadline - 100:
            a = subgradient_descent(a, max_iters=200, deadline=deadline - 100,
                                   tracker=tracker)

        tracker.update(a)
        print(f"  Phase 2: J={tracker.best_score:.8f} ({time.monotonic()-t0:.1f}s)", flush=True)

    # Phase 3: Simulated annealing escape
    if tracker.best is not None and time.monotonic() < deadline - 150:
        t0 = time.monotonic()
        n = tracker.best.size
        print(f"\n{'='*60}", flush=True)
        print(f"Phase 3: SA escape n={n}", flush=True)
        print(f"{'='*60}", flush=True)

        rng_sa = np.random.default_rng(321)
        temp = 0.05
        a_best = tracker.best.copy()
        J_best = compute_J(a_best)

        for restart in range(30):
            if time.monotonic() > deadline - 150:
                break
            a_pert = a_best + rng_sa.normal(0, temp, n)
            a_pert = np.maximum(a_pert, 1e-6)
            a_pert = a_pert / a_pert.sum()

            a_opt = lp_optimize(a_pert, max_iters=8, deadline=deadline - 150,
                               tracker=tracker, tighten=0.01)
            if time.monotonic() < deadline - 150:
                a_opt = lse_descent(a_opt, max_iters=40, deadline=deadline - 150,
                                   tracker=tracker, tau_start=30, tau_end=200)

            J_opt = compute_J(a_opt)
            delta = J_opt - J_best
            if delta < 0 or rng_sa.random() < np.exp(-delta / max(temp, 1e-10)):
                a_best = a_opt
                J_best = J_opt
            temp *= 0.93

        tracker.update(a_best)
        print(f"  Phase 3: J={tracker.best_score:.8f} ({time.monotonic()-t0:.1f}s)", flush=True)

    # Phase 4: Final refinement
    if tracker.best is not None and time.monotonic() < deadline - 30:
        t0 = time.monotonic()
        print(f"\n{'='*60}", flush=True)
        print(f"Phase 4: Final refinement", flush=True)
        print(f"{'='*60}", flush=True)

        a = tracker.best.copy().astype(np.float64)
        n = a.size

        for tight in [0.001, 0.0005, 0.0003]:
            if time.monotonic() > deadline - 30:
                break
            a = lp_optimize(a, max_iters=12, deadline=deadline - 30,
                           tracker=tracker, tighten=tight)

        if time.monotonic() < deadline - 20:
            a = lse_descent(a, max_iters=300, deadline=deadline - 20,
                           tracker=tracker, tau_start=100, tau_end=1000)

        if time.monotonic() < deadline - 15:
            a = spectral_shaping(a, deadline=deadline - 15, tracker=tracker)

        if time.monotonic() < deadline - 8:
            a = coordinate_refinement(a, max_steps=1500, deadline=deadline - 5,
                                     tracker=tracker)

        if time.monotonic() < deadline - 3:
            a = lp_optimize(a, max_iters=8, deadline=deadline - 2,
                           tracker=tracker, tighten=0.001)

        # Try symmetrization
        a_sym = (a + a[::-1]) / 2.0
        a_sym = np.maximum(a_sym, 1e-8)
        a_sym = a_sym / a_sym.sum()
        if compute_J(a_sym) < compute_J(a):
            a = a_sym

        tracker.update(a)
        tracker.update(a_sym)
        print(f"  Phase 4: J={tracker.best_score:.8f} ({time.monotonic()-t0:.1f}s)", flush=True)

    # Phase 5: Alternative lengths
    if time.monotonic() < deadline - 60 and tracker.best is not None:
        t0 = time.monotonic()
        print(f"\n{'='*60}", flush=True)
        print(f"Phase 5: Alternative lengths", flush=True)
        print(f"{'='*60}", flush=True)

        best_n = tracker.best.size
        alt_lengths = [int(best_n * 0.75), int(best_n * 1.25), int(best_n * 1.5)]
        for n in alt_lengths:
            if time.monotonic() > deadline - 60:
                break
            if n < 64 or n > 2048:
                continue
            t_end = min(deadline - 60, time.monotonic() + 30)
            inits = generate_inits(n, rng, tracker.best)
            for a0 in inits[:3]:
                if time.monotonic() > t_end:
                    break
                a = lp_optimize(a0, max_iters=8, deadline=t_end - 3,
                               tracker=tracker, tighten=0.01)
                if time.monotonic() < t_end - 2:
                    a = lse_descent(a, max_iters=30, deadline=t_end - 2,
                                   tracker=tracker, tau_start=20, tau_end=100)
        print(f"  Phase 5: J={tracker.best_score:.8f} ({time.monotonic()-t0:.1f}s)", flush=True)

    if tracker.best is None:
        tracker.best = np.ones(256) / 256
        tracker.update(tracker.best)

    print(f"\n{'='*60}", flush=True)
    print(f"FINAL: J={tracker.best_score:.8f} n={tracker.best.size} "
          f"time={time.monotonic()-tracker.t_start:.1f}s", flush=True)
    print(f"{'='*60}", flush=True)

    return [float(x) for x in tracker.best]
