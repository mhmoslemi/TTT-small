```python
def project_simplex(v):
    """Project v onto the probability simplex {a : sum(a)=1, a>=0}."""
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(1, n+1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.ones(n) / n
    i = np.where(cond)[0][-1] + 1
    theta = cssv[i-1] / i
    return np.maximum(v - theta, 0)
```

```python
def fast_conv(a, b=None):
    """Compute conv(a, b) or conv(a, a) using FFT."""
    if b is None:
        b = a
    n = len(a)
    m = len(b)
    s = 1
    while s < n + m - 1:
        s <<= 1
    fa = np.fft.rfft(a, s)
    fb = np.fft.rfft(b, s)
    return np.fft.irfft(fa * fb, s)[:n + m - 1]

def cross_corr(y, a, n_out):
```

```python
import numpy as np
from scipy.optimize import linprog
from scipy.sparse import csr_matrix

# ========== FFT-based convolution utilities ==========

def fast_conv(a, b=None):
    """Compute full linear convolution using FFT."""
    if b is None:
        b = a
    n = len(a)
    m = len(b)
    s = 1
    while s < n + m - 1:
        s <<= 1
    fa = np.fft.rfft(a, s)
    fb = np.fft.rfft(b, s)
    return np.fft.irfft(fa * fb, s)[:n + m - 1]

def cross_corr(y, a):
    """Compute cross-correlation: (C_a^T y)[:len(a)] where C_a*x = conv(a,x).
    Returns vector of length len(a)."""
    n = len(a)
    s = 1
    while s < len(y) + n - 1:
        s <<= 1
    Y = np.fft.rfft(y, s)
    A = np.fft.rfft(a, s)
    # Ifft of Y * conj(A) gives cross-correlation
    cc = np.fft.irfft(Y * np.conj(A), s)
    # The cross-correlation at lag j is cc[j]
    # (C_a^T y)[j] = sum_k y[k] * a[k-j] = cross_corr(y, a)[j]
    return cc[:n]

# ========== Objective function ==========

def compute_J(a):
    """Compute J(a) = 2*n*max(conv(a,a))/(sum(a))^2"""
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))

# ========== Projection onto simplex ==========

def project_simplex(v):
    """Project v onto {a : sum(a)=1, a>=0}."""
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.ones(n) / n
    i = np.where(cond)[0][-1] + 1
    theta = cssv[i - 1] / i
    return np.maximum(v - theta, 0.0)

# ========== LSE gradient with Adam ==========

def lse_gradient(a, tau):
    """Compute gradient of LSE(tau * conv(a,a)) w.r.t. a. Returns (gradient, lse_value)."""
    n = len(a)
    c = fast_conv(a)
    c_shifted = c - c.max()
    exp_c = np.exp(tau * c_shifted)
    p = exp_c / exp_c.sum()
    lse_val = float(c.max() + np.log(exp_c.sum()) / tau)

    # grad[m] = (2/tau) * sum_k p[k] * 2*a[k-m] * [0<=k-m<n]
    # = (4/tau) * cross_correlation(p, a)[m]
    grad = (4.0 / tau) * cross_corr(p, a)
    return grad, lse_val

def lse_adam_descent(a, max_iters, deadline, tracker, tau0=30.0, tau_max=500.0):
    """LSE gradient descent with Adam optimizer."""
    a = np.maximum(a.astype(np.float64), 1e-8)
    a = a / a.sum()
    n = len(a)
    current_J = compute_J(a)
    tracker.update(a)

    tau = tau0
    lr = 0.1 / np.sqrt(n)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    m_hat = np.zeros(n)
    v_hat = np.zeros(n)
    t = 0
    stall_count = 0

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        t += 1
        grad, lse_val = lse_gradient(a, tau)

        # Full gradient of J = 2n * LSE / (sum(a))^2
        # dJ/da[m] = 2n * (grad[m]/(sum(a))^2 - 2*LSE*a[m]/(sum(a))^3)
        # With sum(a)=1: = 2n * (grad[m] - 2*LSE*a[m])
        s = a.sum()
        full_grad = 2.0 * n * (grad / (s * s) - 2.0 * lse_val * a / (s * s * s))

        # Adam update
        m_hat = beta1 * m_hat + (1 - beta1) * full_grad
        v_hat = beta2 * v_hat + (1 - beta2) * (full_grad ** 2)
        m_hat_corr = m_hat / (1 - beta1 ** t)
        v_hat_corr = v_hat / (1 - beta2 ** t)
        a_new = a - lr * m_hat_corr / (np.sqrt(v_hat_corr) + eps)
        a_new = np.maximum(a_new, 1e-8)
        a_new = a_new / a_new.sum()

        J_new = compute_J(a_new)
        if J_new < current_J - 1e-14:
            a = a_new
            current_J = J_new
            stall_count = 0
            tau = min(tau_max, tau * 1.02)
            lr = min(lr * 1.01, 1.0 / np.sqrt(n))
        else:
            a_new = a - 0.5 * lr * m_hat_corr / (np.sqrt(v_hat_corr) + eps)
            a_new = np.maximum(a_new, 1e-8)
            a_new = a_new / a_new.sum()
            J_new2 = compute_J(a_new)
            if J_new2 < current_J - 1e-14:
                a = a_new
                current_J = J_new2
                stall_count = 0
            else:
                stall_count += 1
                tau = max(tau0, tau * 0.95)
                lr *= 0.95
                if stall_count > 200:
                    break
        if it % 200 == 0:
            tracker.update(a)
    tracker.update(a)
    return a

# ========== QP-based quadratic approximation ==========

def qp_iteration(a, tighten=0.02, n_inner=100, deadline=None, tracker=None):
    """
    QP iteration: minimize ||2*conv(a_old, x) - d||^2 + mu*||x - a_old||^2
    s.t. sum(x)=1, x>=0
    using projected gradient.
    """
    a = np.maximum(a.astype(np.float64), 1e-8)
    a = a / a.sum()
    n = len(a)
    current_J = compute_J(a)
    if tracker:
        tracker.update(a)

    mu = 1.0
    lr_qp = 1.0 / (8.0 * n + 2.0 * mu + 1.0)

    for outer in range(5):
        if deadline and time.monotonic() > deadline:
            break

        a_old = a.copy()
        c_old = fast_conv(a_old)
        M_current = c_old.max()
        M_target = M_current * (1.0 - tighten)

        # Target: flat at M_target
        target = np.full(2 * n - 1, M_target)
        d = c_old + target  # We want 2*conv(a_old, x) ≈ d, i.e., conv(a_old,x) ≈ d/2

        # QP: min ||2*C*a - d||^2 + mu*||a - a_old||^2
        # grad = 8*C^T*(C*a - d/2) + 2*mu*(a - a_old)
        # where C*x = conv(a_old, x)
        # C^T*y = cross_corr(y, a_old)

        best_a = a_old.copy()
        best_J = current_J
        best_obj = float('inf')

        for inner in range(n_inner):
            if deadline and time.monotonic() > deadline:
                break

            # Compute C*a = conv(a_old, a)
            Ca = fast_conv(a_old, a)
            # Residual: C*a - d/2
            residual = Ca - d / 2.0
            # C^T * residual
            CT_r = cross_corr(residual, a_old)
            # Full gradient: 8*CT_r + 2*mu*(a - a_old)
            grad = 8.0 * CT_r + 2.0 * mu * (a - a_old)

            # Projected gradient step
            a_new = a - lr_qp * grad
            a_new = project_simplex(a_new)

            # Check objective
            Ca_new = fast_conv(a_old, a_new)
            obj = np.sum((2.0 * Ca_new - d) ** 2) + mu * np.sum((a_new - a_old) ** 2)

            if obj < best_obj:
                best_obj = obj
                best_a = a_new.copy()

            # Backtracking if not improving
            if obj > best_obj:
                lr_qp *= 0.5
                a = a + lr_qp * grad  # undo step
                a = project_simplex(a)
            else:
                a = a_new
                lr_qp = min(lr_qp * 1.05, 2.0 / (8.0 * n + 2.0 * mu + 1.0))

        J_a = compute_J(best_a)
        if J_a < best_J - 1e-14:
            a = best_a
            current_J = J_a
            if tracker:
                tracker.update(a)
        else:
            break

    return a

_spar_cache = {}

def get_sparsity(n):
    if n not in _spar_cache:
        rows = []
        cols = []
        for k in range(2 * n - 1):
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            rows.extend([k] * (je - js + 1))
            cols.extend(range(js, je + 1))
        _spar_cache[n] = (np.array(rows), np.array(cols), len(rows))
    return _spar_cache[n]

def lp_best_response(a, M):

    rows, cols, nnz = get_sparsity(n)
    # A[k,j] = a[k-j] if 0<=k-j<n

    for i in range(nnz):
        k = rows[i]
        j = cols[i]
        vals[i] = a[k - j]
    A = csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                options={'presolve': True, 'time_limit': 15})
    if r.success and r.x is not None:
        g = np.maximum(r.x, 0.0)
        if g.sum() > 1e-10:
            return g
    return None

def run_lp_opt(a, max_iters, deadline, tracker, tighten=0.01):
    a = np.maximum(a.astype(np.float64), 1e-8)
    a = a / a.sum()
    n = len(a)
    current_J = compute_J(a)
    tracker.update(a)

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        b = fast_conv(a)
        M = float(b.max())
        M_tight = M * (1.0 - tighten)
        g = lp_best_response(a, M_tight)
        if g is None:
            g = lp_best_response(a, M)
        if g is None:
            break
        g = np.maximum(g, 0.0)
        gs = g.sum()
        if gs < 1e-10:
            break
        g = g / gs

        best_a = None
        best_J = current_J
        progress = it / max(1, max_iters)
        base_t = max(0.05, 0.5 * (1.0 - progress))
        for t in [base_t, base_t * 1.5, base_t * 0.5, 0.8, 0.4, 0.95]:
            if t > 1.0:
                continue
            a_trial = (1.0 - t) * a + t * g
            a_trial = np.maximum(a_trial, 0.0)
            ss = a_trial.sum()
            if ss < 1e-10:
                continue
            a_trial = a_trial / ss
            J_trial = compute_J(a_trial)
            if J_trial < best_J - 1e-14:
                best_J = J_trial
                best_a = a_trial
        if best_a is None:
            break
        a = best_a
        current_J = best_J
        if it % 5 == 0:
            tracker.update(a)
    tracker.update(a)
    return a

# ========== Subgradient descent ==========

def subgradient_descent(a, max_iters, deadline, tracker):
    a = np.maximum(a.astype(np.float64), 1e-8)
    a = a / a.sum()
    n = len(a)
    current_J = compute_J(a)
    tracker.update(a)
    lr = 0.5 / n
    rng = np.random.default_rng(12345)

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        b = fast_conv(a)
        M = float(b.max())
        active = np.where(b >= M - 1e-10 * M)[0]
        # grad[j] = (1/|active|) * sum_{k in active} 2*a[k-j] * [0<=k-j<n]
        grad = np.zeros(n)
        for k in active:
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            grad[js:je+1] += 2.0 * a[k - js:k - js + (je - js + 1)]
        grad /= len(active)

        direction = grad - 2.0 * M * a
        a_new = a - lr * direction
        a_new = np.maximum(a_new, 1e-8)
        a_new = a_new / a_new.sum()
        J_new = compute_J(a_new)
        if J_new < current_J - 1e-15:
            a = a_new
            current_J = J_new
            lr = min(lr * 1.02, 5.0 / n)
        else:
            lr *= 0.7
        if it % 100 == 0:
            tracker.update(a)
    tracker.update(a)
    return a

# ========== Initialization ==========

def resample_seq(src, dest_n):
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)

def flat_spectrum_init(n, rng):
    """Generate sequence with approximately flat power spectrum, non-negative."""
    half = n // 2 + 1
    angles = rng.uniform(0, 2 * np.pi, half)
    A = np.zeros(n, dtype=complex)
    A[:half] = np.exp(1j * angles[:half])
    for k in range(1, n - n // 2):
        A[k] = np.conj(A[n - k])
    a = np.fft.ifft(A).real
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-10:
        return np.ones(n) / n
    return a / s

def cosine_init(n, rng, n_freq=5):
    """Initialize with sum of cosines (guaranteed non-negative if amplitudes small)."""
    t = np.linspace(0, 1, n, endpoint=False)
    a = np.ones(n)
    for _ in range(n_freq):
        freq = rng.uniform(1, 15)
        phase = rng.uniform(0, 2 * np.pi)
        amp = rng.uniform(0.05, 0.3)
        a += amp * np.cos(2 * np.pi * freq * t + phase)
    a = np.maximum(a, 0.01)
    return a / a.sum()

def generate_inits(n, rng, parent):
    inits = []
    if parent is not None and parent.size > 0:
        inits.append(resample_seq(parent, n))
        inits.append(resample_seq(parent[::-1], n))

    inits.append(np.ones(n) / n)
    inits.append(cosine_init(n, rng))
    inits.append(cosine_init(n, rng, n_freq=8))
    inits.append(flat_spectrum_init(n, rng))
    inits.append(flat_spectrum_init(n, rng))

    t = np.linspace(0, 1, n, endpoint=False)
    for freq in [2, 3, 5, 7, 11]:
        inits.append(1.0 + 0.3 * np.cos(2 * np.pi * freq * t + rng.uniform(0, 6.28)))
    inits.append(0.5 + 0.5 * np.cos(np.pi * t) ** 2)
    inits.append(np.exp(-4 * (t - 0.5) ** 2) + 0.1)

    # Anti-symmetric perturbation (still non-negative)
    base = np.ones(n) / n
    pert = np.sin(2 * np.pi * rng.uniform(2, 10) * t) * 0.3
    inits.append(np.maximum(base + pert, 0.01))

    for x in inits:
        x = np.maximum(x, 0.001).astype(np.float64)
        x = x / x.sum()
    return inits

# ========== Main optimization ==========

def optimize_sequence(a0, deadline, tracker, n_min=500, n_max=5000):
    """Run a full optimization pipeline on a single sequence."""
    a = np.maximum(a0.astype(np.float64), 1e-8)
    a = a / a.sum()
    n = a.size

    # Step 1: QP iteration
    t_qp_end = min(deadline - 60, time.monotonic() + (deadline - time.monotonic()) * 0.3)
    if time.monotonic() < t_qp_end:
        a = qp_iteration(a, tighten=0.015, n_inner=80, deadline=t_qp_end, tracker=tracker)

    # Step 2: LSE Adam
    t_lse_end = min(deadline - 60, time.monotonic() + (deadline - time.monotonic()) * 0.3)
    if time.monotonic() < t_lse_end:
        a = lse_adam_descent(a, n_min, t_lse_end, tracker)

    # Step 3: Subgradient
    t_sg_end = min(deadline - 60, time.monotonic() + (deadline - time.monotonic()) * 0.2)
    if time.monotonic() < t_sg_end:
        a = subgradient_descent(a, max(200, n_min // 2), t_sg_end, tracker)

    # Step 4: LP polishing
    if time.monotonic() < deadline - 30:
        a = run_lp_opt(a, 15, deadline - 30, tracker, tighten=0.008)

    # Step 5: Final LSE polish
    if time.monotonic() < deadline - 10:
        a = lse_adam_descent(a, 1000, deadline - 10, tracker, tau0=50.0)

    return a

# ========== Main entry point ==========

def propose_candidate(seed=42, budget_s=1500, **kwargs):
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + float(budget_s) - 15.0
    tracker = ScoreTracker()

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        if p.ndim == 1 and p.size > 0 and np.all(np.isfinite(p)):
            parent = np.maximum(p, 0.0)
            tracker.update(parent)
            print(f"Parent loaded: n={parent.size}, J={tracker.best_score:.8f}", flush=True)
    except (NameError, AttributeError):
        pass

    # === Phase 1: Optimize parent at its length ===
    if parent is not None and time.monotonic() < deadline - 300:
        n_parent = parent.size
        t0 = time.monotonic()
        print(f"\n{'='*50} Phase 1: Optimize parent n={n_parent} {'='*50}", flush=True)
        phase1_deadline = min(deadline - 250, t0 + 200)
        a_opt = optimize_sequence(parent.copy(), phase1_deadline, tracker)
        tracker.update(a_opt)
        tracker.report(f"Phase1 ({time.monotonic()-t0:.1f}s)")
        if tracker.best is not None and tracker.best.size == n_parent:
            parent = tracker.best.copy()

    # === Phase 2: Try multiple lengths with diverse inits ===
    if time.monotonic() < deadline - 150:
        lengths = [256, 320, 384, 448, 512, 576, 640]
        if parent is not None:
            lengths = [parent.size] + [l for l in lengths if l != parent.size]

        t0 = time.monotonic()
        print(f"\n{'='*50} Phase 2: Multi-length search {'='*50}", flush=True)
        time_per = (deadline - time.monotonic() - 200) / len(lengths)

        for n in lengths:
            if time.monotonic() > deadline - 100:
                break
            if time.monotonic() - t0 > (deadline - time.monotonic() - 100) * 0.6:
                break

            t_start = time.monotonic()
            t_end = min(deadline - 80, t_start + max(30.0, time_per * 1.5))
            print(f"  n={n} (budget {t_end - t_start:.0f}s)", flush=True)

            inits = generate_inits(n, rng, parent)
            best_for_n = None
            best_J_for_n = float('inf')

            init_budget = max(8.0, (t_end - time.monotonic()) / len(inits))

            for idx, a0 in enumerate(inits[:6]):  # Limit to 6 inits per length
                if time.monotonic() > t_end - 15:
                    break
                t_init_end = min(t_end - 10, time.monotonic() + init_budget)
                if time.monotonic() > t_init_end:
                    continue

                # Quick optimization: QP + LSE
                a = a0.copy()
                if time.monotonic() < t_init_end - 5:
                    a = qp_iteration(a, tighten=0.02, n_inner=40, deadline=t_init_end - 5, tracker=tracker)
                if time.monotonic() < t_init_end:
                    a = lse_adam_descent(a, 500, t_init_end, tracker)

                J = compute_J(a)
                if J < best_J_for_n:
                    best_J_for_n = J
                    best_for_n = a.copy()

            if best_for_n is not None:
                tracker.update(best_for_n)
                if parent is None or best_J_for_n < (compute_J(resample_seq(parent, best_for_n.size)) if parent is not None else float('inf')):
                    if best_J_for_n < tracker.best_score + 0.01:
                        parent = best_for_n.copy()

        tracker.report(f"Phase2 ({time.monotonic()-t0:.1f}s)")

    # === Phase 3: Focused optimization on best found ===
    if tracker.best is not None and time.monotonic() < deadline - 100:
        t0 = time.monotonic()
        n_best = tracker.best.size
        print(f"\n{'='*50} Phase 3: Focused n={n_best} {'='*50}", flush=True)

        a = tracker.best.copy()
        phase3_deadline = min(deadline - 50, t0 + 150)

        # Multiple QP passes with increasing tightening
        for tighten in [0.01, 0.015, 0.02, 0.025]:
            if time.monotonic() > phase3_deadline - 20:
                break
            a = qp_iteration(a, tighten=tighten, n_inner=60, deadline=phase3_deadline - 20, tracker=tracker)

        # LSE Adam
        if time.monotonic() < phase3_deadline - 15:
            a = lse_adam_descent(a, 2000, phase3_deadline - 15, tracker, tau0=40.0)

        # LP
        if time.monotonic() < phase3_deadline - 5:
            a = run_lp_opt(a, 10, phase3_deadline - 5, tracker, tighten=0.005)

        # Final LSE
        if time.monotonic() < phase3_deadline:
            a = lse_adam_descent(a, 1000, phase3_deadline, tracker, tau0=80.0)

        tracker.update(a)
        tracker.report(f"Phase3 ({time.monotonic()-t0:.1f}s)")

    # === Phase 4: Random restarts ===
    if tracker.best is not None and time.monotonic() < deadline - 60:
        t0 = time.monotonic()
        n_best = tracker.best.size
        print(f"\n{'='*50} Phase 4: Restarts n={n_best} {'='*50}", flush=True)
        rng2 = np.random.default_rng(999)
        temp = 0.08
        a = tracker.best.copy()

        for r in range(30):
            if time.monotonic() > deadline - 30:
                break
            a_pert = a + rng2.normal(0, temp, n_best)
            a_pert = np.maximum(a_pert, 1e-6)
            a_pert = a_pert / a_pert.sum()

            t_short_end = min(deadline - 25, time.monotonic() + 25)
            a_ref = qp_iteration(a_pert, tighten=0.02, n_inner=30, deadline=t_short_end - 5, tracker=tracker)
            if time.monotonic() < t_short_end:
                a_ref = lse_adam_descent(a_ref, 300, t_short_end, tracker)

            J_ref = compute_J(a_ref)
            J_cur = compute_J(a)
            if J_ref < J_cur or rng2.random() < np.exp(-(J_ref - J_cur) / max(temp, 1e-8)):
                a = a_ref
            temp *= 0.93

        # Polish
        if time.monotonic() < deadline - 20:
            a = qp_iteration(a, tighten=0.01, n_inner=50, deadline=deadline - 15, tracker=tracker)
            if time.monotonic() < deadline - 10:
                a = lse_adam_descent(a, 500, deadline - 10, tracker)
            if time.monotonic() < deadline - 5:
                a = run_lp_opt(a, 8, deadline - 5, tracker, tighten=0.005)

        tracker.update(a)
        tracker.report(f"Phase4 ({time.monotonic()-t0:.1f}s)")

    # === Phase 5: Final coordinate descent polish ===
    if tracker.best is not None and time.monotonic() < deadline - 15:
        t0 = time.monotonic()
        print(f"\n{'='*50} Phase 5: Final polish {'='*50}", flush=True)
        a = tracker.best.copy()
        rng3 = np.random.default_rng(555)
        n = a.size
        current_J = compute_J(a)

        for step in range(2000):
            if time.monotonic() > deadline - 10:
                break
            j = rng3.integers(0, n)
            for scale in [0.7, 1.3, 0.5, 1.5, 0.85, 1.15]:
                a_trial = a.copy()
                a_trial[j] *= scale
                a_trial = np.maximum(a_trial, 1e-8)
                ss = a_trial.sum()
                if ss < 1e-10:
                    continue
                a_trial = a_trial / ss
                J_trial = compute_J(a_trial)
                if J_trial < current_J - 1e-14:
                    a = a_trial
                    current_J = J_trial
                    break
            if step % 200 == 0:
                tracker.update(a)

        tracker.update(a)
        tracker.report(f"Phase5 ({time.monotonic()-t0:.1f}s)")

    # Fallback
    if tracker.best is None:
        tracker.best = np.ones(256)
        tracker.update(tracker.best)

    print(f"\n{'='*50} FINAL {'='*50}", flush=True)
    print(f"J={tracker.best_score:.8f} n={tracker.best.size} time={time.monotonic()-tracker.t_start:.1f}s", flush=True)
    return [float(x) for x in tracker.best]
```
