    return np.fft.irfft(fa * fa, s)[:2 * n - 1]

Minimizes J(a) = 2*n*max(convolve(a,a))/(sum(a))^2 using:
1. Adam subgradient descent with top-k active constraint weighting
2. Exact coordinate descent via golden section search (vectorized)
3. LP best-response with adaptive line search
4. Multi-length exploration with structured initializations
5. Simulated annealing restarts for escaping local minima
Key improvements: Adam adaptive per-coordinate learning rates,

import numpy as np
from scipy.optimize import linprog
from scipy.sparse import lil_matrix
import time


def fast_conv(a):
    """Full self-convolution via FFT."""
    n = len(a)
    s = 1
    while s < 2 * n - 1:
        s <<= 1
    fa = np.fft.rfft(a, s)
    return np.fft.irfft(fa * fa, s)[:2 * n - 1]


def compute_J(a):
    """Compute J(a) = 2*n*max(conv(a,a))/(sum(a))^2."""
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))


def compute_subgradient(b, a, k_active=15, temperature=0.01):
    """Compute weighted subgradient of max(b) using top-k active constraints.
    Uses softmax-like weighting for smoothness."""
    n = len(a)
    b_max = b.max()
    # Find top-k active
    k = min(k_active, len(b))
    top_indices = np.argsort(b)[-k:]
    top_vals = b[top_indices]
    # Softmax weights (smooth approximation to uniform on active set)
    weights = np.exp((top_vals - b_max) / max(temperature * b_max, 1e-10))
    weights /= weights.sum()
    
    grad = np.zeros(n)
    for w, kk in zip(weights, top_indices):
        kk = int(kk)
        i_low = max(0, kk - n + 1)
        i_high = min(n - 1, kk)
        if i_low <= i_high:
            # grad[i] += 2 * w * a[kk - i] for i in [i_low, i_high]
            # a[kk - i] for i in [i_low, i_high] is a[kk-i_high : kk-i_low+1] reversed
            idx = kk - np.arange(i_low, i_high + 1)
            grad[i_low:i_high + 1] += 2.0 * w * a[idx]
    
    return grad, b_max


def adam_subgradient_descent(a, max_iters, deadline, tracker, k_active=15, lr=0.05):
    """Adam subgradient descent on the normalized objective."""
    a = np.maximum(a.astype(np.float64), 0.0)
    S = a.sum()
    if S < 1e-10:
        return a
    a = a / S
    n = len(a)
    
    beta1 = 0.8
    beta2 = 0.999
    eps = 1e-8
    m = np.zeros(n)
    v = np.zeros(n)
    t = 0
    current_J = compute_J(a)
    tracker.update(a)
    
    best_J = current_J
    no_improve = 0
    
    while t < max_iters and time.monotonic() < deadline:
        t += 1
        b = fast_conv(a)
        b_max = b.max()
        S = a.sum()
        J = 2.0 * n * b_max / (S * S)
        
        # Compute subgradient of max(b)
        grad_max, b_max_check = compute_subgradient(b, a, k_active=k_active)
        
        # Full subgradient of J = 2n * max_b / S^2
        # dJ/da_i = 2n/S^2 * d(max_b)/da_i - 2n*max_b/S^3 * 2 * dS/da_i
        #         = 2n/S^2 * grad_max[i] - 4n*b_max/S^2
        grad = (2.0 * n / (S * S)) * grad_max - (4.0 * n * b_max / (S * S)) * np.ones(n)
        
        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        
        # Adaptive learning rate: decrease when no improvement
        current_lr = lr / (1.0 + 0.01 * no_improve)
        
        a_new = a - current_lr * m_hat / (np.sqrt(v_hat) + eps)
        a_new = np.maximum(a_new, 0.0)
        new_S = a_new.sum()
        if new_S < 1e-10:
            a_new = a
        else:
            a_new = a_new / new_S
        
        new_J = compute_J(a_new)
        
        if new_J < J - 1e-12:
            a = a_new
            no_improve = 0
            if new_J < best_J:
                best_J = new_J
                tracker.update(a)
        else:
            # Try half step
            a_half = a - 0.5 * current_lr * m_hat / (np.sqrt(v_hat) + eps)
            a_half = np.maximum(a_half, 0.0)
            half_S = a_half.sum()
            if half_S > 1e-10:
                a_half = a_half / half_S
                half_J = compute_J(a_half)
                if half_J < J - 1e-12:
                    a = a_half
                    no_improve = 0
                    if half_J < best_J:
                        best_J = half_J
                        tracker.update(a)
                else:
                    no_improve += 1
            else:
                no_improve += 1
        
        # Periodic reporting
        if t % 200 == 0:
            print(f"    Adam iter {t}: J={J:.8f}, best={best_J:.8f}, no_imp={no_improve}", flush=True)
        
        # Early stop if stuck
        if no_improve > 500:
            break
    
    tracker.update(a)
    return a


def coordinate_descent_single(a, i, b, S, n):

    k_hi = i + n - 1

    # b_new[k] = b[k] + 2*(x-ai)*a[k-i] for k != 2i

    #                            max_{k in [k_lo,k_hi], k!=2i} (b[k] + 2*(x-ai)*a[k-i]),
    #                            b[2i] + (x-ai)*(x+ai) )

    else:

            intercepts = np.array([0.0])

        intercepts = np.array([0.0])

            b_2i_new = -np.inf

    # Golden section search
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    lo = 0.0

    hi = max(hi_candidates)
    # Make sure hi is large enough
    if objective(hi) > objective(hi * 2):
        hi = hi * 2
    
    c_pt = hi - gr * (hi - lo)
    d_pt = lo + gr * (hi - lo)
    fc = objective(c_pt)
    fd = objective(d_pt)
    
    for _ in range(50):
        if fc < fd:
            hi = d_pt
            d_pt = c_pt
            fd = fc
            c_pt = hi - gr * (hi - lo)
            fc = objective(c_pt)
        else:
            lo = c_pt
            c_pt = d_pt
            fc = fd
            d_pt = lo + gr * (hi - lo)
            fd = objective(d_pt)
    
    x_opt = (lo + hi) / 2.0
    return max(0.0, x_opt)


def coordinate_descent_pass(a, b, S, n, order=None):
    """One full pass of coordinate descent. Returns updated a, b, S."""
    if order is None:
        order = list(range(n))
    
    for i in order:
        new_val = coordinate_descent_single(a, i, b, S, n)
        if abs(new_val - a[i]) > 1e-12:
            old_val = a[i]
            a[i] = new_val
            S = S - old_val + new_val
            # Update b incrementally
            for k in range(max(0, i), min(2 * n - 1, i + n)):
                ki = k - i
                if 0 <= ki < n:
                    if k == 2 * i:
                        b[k] = b[k] - old_val * old_val + new_val * new_val
                    else:
                        b[k] = b[k] - 2.0 * old_val * a[ki] + 2.0 * new_val * a[ki]
    
    return a, b, S


def build_and_solve_lp(a, M):

    n = len(a)

    # Build constraint matrix: A[k, j] = a[k-j] if valid
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        jl = max(0, k - n + 1)
        jh = min(n - 1, k)
        for j in range(jl, jh + 1):
            ki = k - j
            if 0 <= ki < n:
                rows.append(k)
                cols.append(j)
                vals.append(a[ki])
    
    if len(rows) == 0:
        return None
    
    A = lil_matrix((2 * n - 1, n))
    for idx in range(len(rows)):
        A[rows[idx], cols[idx]] = vals[idx]
    A = A.tocsr()
    
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    
    r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                options={'presolve': True, 'time_limit': 30})
    if r.success and r.x is not None:
        g = np.maximum(r.x, 0.0)
        if g.sum() > 1e-10:
            return g
    return None


def lp_best_response_step(a, tracker, deadline, tighten=0.005, max_lp_iters=20):
    """LP mirror descent step with line search."""
    a = np.maximum(a.astype(np.float64), 0.001)
    a = a / a.sum()
    n = len(a)
    current_J = compute_J(a)
    
    for it in range(max_lp_iters):
        if time.monotonic() > deadline:
            break
        
        b = fast_conv(a)
        M = float(b.max())
        M_target = M * (1.0 - tighten)
        
        g = build_and_solve_lp(a, M_target)
        if g is None:
            M_target = M * (1.0 - tighten * 0.5)
            g = build_and_solve_lp(a, M_target)
        if g is None:
            break
        
        g = np.maximum(g, 0.0)
        g_sum = g.sum()
        if g_sum < 1e-10:
            break
        g = g / g_sum
        
        # Line search over step sizes
        best_a = None
        best_J = current_J
        t_candidates = [0.3, 0.5, 0.7, 1.0, 0.15, 0.85]
        
        for t in t_candidates:
            a_trial = (1.0 - t) * a + t * g
            a_trial = np.maximum(a_trial, 0.0)
            s = a_trial.sum()
            if s < 1e-10:
                continue
            a_trial = a_trial / s
            J_trial = compute_J(a_trial)
            if J_trial < best_J - 1e-14:
                best_J = J_trial
                best_a = a_trial
        
        if best_a is None:
            break
        
        a = best_a
        current_J = best_J
        tracker.update(a)
    
    return a


def resample_seq(src, dest_n):
    """Resample 1D sequence to new length via linear interpolation."""
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)


class ScoreTracker:
    def __init__(self):
        self.best = None
        self.best_score = float('inf')
        self.t_start = time.monotonic()
    
    def update(self, a):
        a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
        if a.size == 0 or a.sum() < 0.01:
            return
        score = float(evaluate_sequence(a.tolist()))
        if np.isfinite(score) and score < self.best_score:
            self.best = a.copy()
            self.best_score = score
            elapsed = time.monotonic() - self.t_start
            print(f"  * NEW BEST: J={score:.8f} n={a.size} t={elapsed:.0f}s", flush=True)
    
    def report(self, label=""):
        if self.best is not None:
            print(f"  {label} best: J={self.best_score:.8f} n={self.best.size} t={time.monotonic()-self.t_start:.0f}s", flush=True)


def generate_structured_inits(n, rng):
    """Generate diverse structured initial sequences."""
    inits = []
    t = np.linspace(0, 1, n, endpoint=False)
    
    inits.append(np.ones(n))
    inits.append(0.5 + 0.5 * np.cos(2 * np.pi * t))
    inits.append(0.7 + 0.3 * np.cos(4 * np.pi * t))
    inits.append(0.6 + 0.2 * np.cos(2 * np.pi * t) + 0.2 * np.cos(6 * np.pi * t))
    
    # Symmetric patterns
    half = n // 2
    sym = np.concatenate([t[:half], t[half:][::-1]])
    inits.append(0.5 + 0.5 * sym)
    inits.append(1.0 - 0.3 * np.abs(2 * t - 1.0))
    
    # Multi-peak
    for npeaks in [3, 5, 7]:
        peaks = np.zeros(n)
        for p in range(npeaks):
            center = (p + 0.5) / npeaks
            peaks += np.exp(-50 * (t - center) ** 2)
        inits.append(0.3 + 0.7 * peaks / peaks.max())
    
    # Random smooth
    x = rng.normal(0, 1, n)
    # Smooth via cumulative sum of random walk
    x_smooth = np.cumsum(x) - np.linspace(0, np.cumsum(x)[-1], n)
    x_smooth = 0.5 + 0.5 * x_smooth / (np.abs(x_smooth).max() + 1e-10)
    inits.append(np.maximum(x_smooth, 0.1))
    
    return inits


def intensive_coordinate_descent(a, deadline, tracker, max_passes=50):
    """Run many passes of coordinate descent until deadline or convergence."""
    a = np.maximum(a.astype(np.float64), 0.0)
    S = a.sum()
    if S < 1e-10:
        return a
    a = a / S
    S = 1.0
    n = len(a)
    b = fast_conv(a)
    current_J = compute_J(a)
    tracker.update(a)
    
    rng = np.random.default_rng(12345)
    stall_count = 0
    
    for p in range(max_passes):
        if time.monotonic() > deadline:
            break
        
        order = list(range(n))
        rng.shuffle(order)
        
        a, b, S = coordinate_descent_pass(a, b, S, n, order)
        
        # Verify
        b = fast_conv(a)
        new_S = a.sum()
        new_J = 2.0 * n * b.max() / (new_S * new_S)
        
        if new_J < current_J - 1e-10:
            current_J = new_J
            stall_count = 0
            tracker.update(a)
        else:
            stall_count += 1
            if stall_count > 5:
                break
        
        if p % 10 == 0:
            print(f"    CD pass {p}: J={current_J:.8f}", flush=True)
    
    return a


def propose_candidate(seed=42, budget_s=1500, **kwargs):
    """
    Main search function combining Adam subgradient, coordinate descent,
    LP best-response, and multi-length exploration.
    """
    rng = np.random.default_rng(seed)
    deadline_abs = time.monotonic() + float(budget_s) - 20.0
    tracker = ScoreTracker()
    
    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        if p.ndim == 1 and p.size > 0 and np.all(np.isfinite(p)):
            parent = np.maximum(p, 0.0)
            tracker.update(parent)
            print(f"Parent loaded: n={parent.size}, J={tracker.best_score:.8f}", flush=True)
    except (NameError, AttributeError):
        pass
    
    if parent is None:
        parent = np.ones(384)
        tracker.update(parent)
        print(f"No parent found, using uniform n={parent.size}", flush=True)
    
    n_parent = parent.size
    
    # === Phase 1: Coordinate descent from parent ===
    print(f"\n{'='*60} Phase 1: CD from parent (n={n_parent}) {'='*60}", flush=True)
    t0 = time.monotonic()
    deadline_p1 = min(deadline_abs - 600, t0 + 200)
    a_best = intensive_coordinate_descent(parent, deadline_p1, tracker, max_passes=30)
    tracker.report(f"Phase 1 CD: {time.monotonic()-t0:.1f}s")
    
    # === Phase 2: Adam subgradient for exploration ===
    if time.monotonic() < deadline_abs - 400:
        print(f"\n{'='*60} Phase 2: Adam Subgradient {'='*60}", flush=True)
        t0 = time.monotonic()
        deadline_p2 = min(deadline_abs - 300, t0 + 300)
        a_best = adam_subgradient_descent(a_best, max_iters=5000, deadline=deadline_p2,
                                           tracker=tracker, k_active=20, lr=0.03)
        tracker.report(f"Phase 2 Adam: {time.monotonic()-t0:.1f}s")
    
    # === Phase 3: Coordinate descent again ===
    if time.monotonic() < deadline_abs - 300:
        print(f"\n{'='*60} Phase 3: CD after Adam {'='*60}", flush=True)
        t0 = time.monotonic()
        deadline_p3 = min(deadline_abs - 250, t0 + 150)
        a_best = intensive_coordinate_descent(a_best, deadline_p3, tracker, max_passes=30)
        tracker.report(f"Phase 3 CD: {time.monotonic()-t0:.1f}s")
    
    # === Phase 4: LP best-response ===
    if time.monotonic() < deadline_abs - 200:
        print(f"\n{'='*60} Phase 4: LP Best Response {'='*60}", flush=True)
        t0 = time.monotonic()
        deadline_p4 = min(deadline_abs - 150, t0 + 200)
        a_best = lp_best_response_step(a_best, tracker, deadline_p4,
                                        tighten=0.005, max_lp_iters=15)
        # Follow up with CD
        if time.monotonic() < deadline_abs - 130:
            a_best = intensive_coordinate_descent(a_best, deadline_abs - 130, tracker, max_passes=20)
        tracker.report(f"Phase 4 LP: {time.monotonic()-t0:.1f}s")
    
    # === Phase 5: Multi-length exploration ===
    if time.monotonic() < deadline_abs - 100:
        print(f"\n{'='*60} Phase 5: Multi-Length Exploration {'='*60}", flush=True)
        t0 = time.monotonic()
        
        best_n = tracker.best.size if tracker.best is not None else n_parent
        # Try lengths around the best
        lengths_to_try = []
        for delta in [-20, -10, -5, -2, 2, 5, 10, 20, 30, -30]:
            nl = best_n + delta
            if 100 <= nl <= 800 and nl not in lengths_to_try:
                lengths_to_try.append(nl)
        
        per_length_budget = min(20, max(5, (deadline_abs - 100 - time.monotonic()) / max(1, len(lengths_to_try))))
        
        for nl in lengths_to_try:
            if time.monotonic() > deadline_abs - 80:
                break
            
            t_start = time.monotonic()
            a_resampled = resample_seq(tracker.best, nl)
            a_resampled = np.maximum(a_resampled, 0.001)
            a_resampled = a_resampled / a_resampled.sum()
            
            # Quick Adam
            if time.monotonic() - t_start < per_length_budget * 0.5:
                a_trial = adam_subgradient_descent(a_resampled, max_iters=500,
                                                    deadline=t_start + per_length_budget * 0.5,
                                                    tracker=tracker, k_active=15, lr=0.03)
            else:
                a_trial = a_resampled
            
            # Quick CD
            if time.monotonic() - t_start < per_length_budget:
                a_trial = intensive_coordinate_descent(a_trial, min(t_start + per_length_budget,
                                                                     deadline_abs - 70),
                                                        tracker, max_passes=10)
                tracker.update(a_trial)
            
            print(f"    Length {nl}: J={tracker.best_score:.8f}", flush=True)
        
        tracker.report(f"Phase 5 Lengths: {time.monotonic()-t0:.1f}s")
    
    # === Phase 6: Restart with perturbations on best ===
    if time.monotonic() < deadline_abs - 50:
        print(f"\n{'='*60} Phase 6: Perturbation Restarts {'='*60}", flush=True)
        t0 = time.monotonic()
        
        best_a = tracker.best.copy()
        n = best_a.size
        
        for temp in [0.03, 0.015, 0.005]:
            if time.monotonic() > deadline_abs - 40:
                break
            
            # Multiple random restarts
            for trial in range(3):
                if time.monotonic() > deadline_abs - 35:
                    break
                
                a_pert = best_a + rng.normal(0, temp, n)
                a_pert = np.maximum(a_pert, 0.001)
                a_pert = a_pert / a_pert.sum()
                
                # Short Adam
                a_pert = adam_subgradient_descent(a_pert, max_iters=300,
                                                    deadline=min(time.monotonic() + 15, deadline_abs - 30),
                                                    tracker=tracker, k_active=15, lr=0.02)
                # Short CD
                if time.monotonic() < deadline_abs - 30:
                    a_pert = intensive_coordinate_descent(a_pert, deadline_abs - 30, tracker, max_passes=8)
        
        tracker.report(f"Phase 6 Restarts: {time.monotonic()-t0:.1f}s")
    
    # === Phase 7: Final intensive polish ===
    if time.monotonic() < deadline_abs - 10:
        print(f"\n{'='*60} Phase 7: Final Polish {'='*60}", flush=True)
        t0 = time.monotonic()
        
        # Final CD
        a_final = intensive_coordinate_descent(tracker.best, deadline_abs - 10, tracker, max_passes=20)
        tracker.update(a_final)
        
        # Try one more LP if time
        if time.monotonic() < deadline_abs - 5:
            a_final = lp_best_response_step(a_final, tracker, deadline_abs - 5,
                                             tighten=0.002, max_lp_iters=5)
            tracker.update(a_final)
        
        tracker.report(f"Phase 7 Final: {time.monotonic()-t0:.1f}s")
    
    # Fallback
    if tracker.best is None:
        tracker.best = np.ones(256)
        tracker.best_score = compute_J(tracker.best)
    
    print(f"\n{'='*60} FINAL RESULT {'='*60}", flush=True)
    print(f"J = {tracker.best_score:.10f}, n = {tracker.best.size}", flush=True)
    print(f"Total time: {time.monotonic()-tracker.t_start:.1f}s", flush=True)
    
    return [float(x) for x in tracker.best]
