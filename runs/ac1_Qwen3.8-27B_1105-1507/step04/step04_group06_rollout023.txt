# Cache for convolution plans
_fft_cache = {}

def get_fft_size(n):
    """Get the next power of 2 >= 2n-1."""
    s = 1
    while s < 2 * n - 1:
        s <<= 1
    return s

def fast_conv(a):
    """Full linear self-convolution via FFT."""
    n = len(a)
    s = 1
    while s < 2 * n - 1:
        s <<= 1
    fa = np.fft.rfft(a, s)
    return np.fft.irfft(fa * fa, s)[:2 * n - 1]

def compute_J(a):
    """Compute J(a) = 2*n*max(conv(a,a))/(sum(a))^2."""
    a = np.asarray(a, dtype=np.float64)
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))

def normalize(a):
    """Normalize to sum = 1."""
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(len(a)) / len(a)
    return a / s

def project_simplex(v, target_sum=1.0):
    """Project v onto {x >= 0, sum(x) = target_sum}."""
    n = len(v)
    # We want to find alpha such that sum(max(v - alpha, 0)) = target_sum
    # Use bisection
    v = np.asarray(v, dtype=np.float64)
    lo = v.min() - 1.0
    hi = v.max() + 1.0
    for _ in range(60):
        mid = (lo + hi) / 2
        s = np.maximum(v - mid, 0).sum()
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    result = np.maximum(v - (lo + hi) / 2, 0)
    # Fix rounding
    s = result.sum()
    if s > 0 and abs(s - target_sum) > 1e-10:
        result = result * target_sum / s
    return result

def subgradient_step(a, eta):
    """One subgradient step for minimizing max(conv(a,a)) with sum(a)=1."""
    n = len(a)
    b = fast_conv(a)
    k_star = int(np.argmax(b))
    # Gradient of b_{k*} w.r.t. a
    # grad[j] = 2 * a[k* - j] if 0 <= k* - j < n, else 0
    j = np.arange(n)
    idx = k_star - j
    valid = (idx >= 0) & (idx < n)
    grad = np.zeros(n)
    grad[valid] = 2.0 * a[idx[valid]]
    # Handle the diagonal term (when k* = 2*j)
    if k_star % 2 == 0:
        j_diag = k_star // 2
        if 0 <= j_diag < n:
            grad[j_diag] -= a[j_diag]  # Only one contribution, not two
    # Update
    a_new = a - eta * grad
    a_new = project_simplex(a_new, 1.0)
    return a_new, k_star

    avg_start = n_steps // 4  # Start averaging after first quarter

def get_A_sparse(n, a):
    """Build the sparse constraint matrix for the LP."""
    # A has shape (2n-1, n)
    # A[k, j] = a[|k-j|] if |k-j| < n... 
    # Actually A[k,j] = a[k-j] if 0 <= k-j < n (for the convolution structure)
    # (a*g)_k = sum_j g_j * a[k-j] (when 0 <= k-j < n)
    # So A[k,j] = a[k-j] if 0 <= k-j < n, else 0
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        jl = max(0, k - n + 1)
        jh = min(n - 1, k)
        for j in range(jl, jh + 1):
            rows.append(k)
            cols.append(j)
            vals.append(a[k - j])
    return csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))

def lp_best_response(a, M):
    """Solve LP: max sum(g) s.t. (a*g)_k <= M for all k, g >= 0.
    Returns (g, y) where g is the primal solution and y is the dual solution."""
    n = len(a)
    c = -np.ones(n)
    A = get_A_sparse(n, a)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
    if r.success and r.x is not None:
        g = np.maximum(r.x, 0.0)
        g_sum = g.sum()
        if g_sum > 1e-10:
            return g / g_sum, r.ineqlin
    return None, None

def line_search(a, g, n_evals=20):
    """Golden section + parabolic refinement for best t in [0,1]."""
    best_t = 0.0
    best_J = compute_J(a)
    
    a1 = g.copy()
    J1 = compute_J(a1)
    if J1 < best_J:
        best_J = J1
        best_t = 1.0
    
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    lo = 0.0
    hi = 1.0
    f_lo = best_J  # J at t=lo (recomputed below)
    f_hi = J1  # J at t=hi
    
    # Actually let's just do golden section properly
    lo = 0.0
    hi = 1.0
    t1 = hi - gr * (hi - lo)
    t2 = lo + gr * (hi - lo)
    f1 = compute_J((1-t1)*a + t1*g)
    f2 = compute_J((1-t2)*a + t2*g)
    
    for _ in range(n_evals - 2):
        if f1 < f2:
            hi = t2
            t2 = t1
            f2 = f1
            t1 = hi - gr * (hi - lo)
            f1 = compute_J((1-t1)*a + t1*g)
        else:
            lo = t1
            t1 = t2
            f1 = f2
            t2 = lo + gr * (hi - lo)
            f2 = compute_J((1-t2)*a + t2*g)
        
        t_mid = (lo + hi) / 2
        f_mid = compute_J((1-t_mid)*a + t_mid*g)
        if f_mid < best_J:
            best_J = f_mid
            best_t = t_mid
    
    # Parabolic refinement
    t_opt = (lo + hi) / 2
    f_opt = compute_J((1-t_opt)*a + t_opt*g)
    if f_opt < best_J:
        best_J = f_opt
        best_t = t_opt
    
    return (1-best_t)*a + best_t*g, best_J

def lp_mirror_descent(a, max_iters, deadline, tracker=None, n_line_evals=20,
                      stall_patience=50, use_pressure=False):
    """LP-based mirror descent with adaptive line search."""
    a = normalize(a.astype(np.float64).copy())
    current_J = compute_J(a)
    best_a = a.copy()
    best_J = current_J
    stall_count = 0
    
    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        
        b = fast_conv(a)
        M = float(b.max())
        
        # Optional: apply pressure (use slightly smaller M)
        M_eff = M * (1.0 - 0.02 * min(it/100, 1.0)) if use_pressure else M
        
        g, dual_y = lp_best_response(a, M_eff)
        if g is None:
            break
        
        a_new, J_new = line_search(a, g, n_evals=n_line_evals)
        a_new = normalize(a_new)
        J_new = compute_J(a_new)
        
        if J_new < current_J - 1e-12:
            a = a_new
            current_J = J_new
            stall_count = 0
            if J_new < best_J:
                best_J = J_new
                best_a = a.copy()
                if tracker is not None:
                    tracker.update(a)
        else:
            # Try smaller steps
            improved = False
            for t in [0.02, 0.01, 0.005, 0.05]:
                a_try = normalize((1-t)*a + t*g)
                J_try = compute_J(a_try)
                if J_try < current_J - 1e-14:
                    a = a_try
                    current_J = J_try
                    stall_count = 0
                    improved = True
                    if J_try < best_J:
                        best_J = J_try
                        best_a = a.copy()
                        if tracker is not None:
                            tracker.update(a)
                    break
            if not improved:
                stall_count += 1
                if stall_count >= stall_patience:
                    break
        
        if tracker is not None and it % 100 == 0:
            tracker.update(a)
    
    if tracker is not None:
        tracker.update(best_a)
    return best_a

def smooth_objective_and_grad(a, beta):
    """Compute smooth approximation of max(conv(a,a)) and gradient."""
    n = len(a)
    b = fast_conv(a)
    b_max = float(b.max())
    shifted = beta * (b - b_max)
    shifted = np.clip(shifted, -500, 500)
    exp_shifted = np.exp(shifted)
    Z = float(exp_shifted.sum())
    if Z < 1e-300:
        p = np.zeros(2*n-1)
        p[np.argmax(b)] = 1.0
    else:
        p = exp_shifted / Z
    
    val = (b_max + np.log(max(Z, 1e-300))) / beta
    
    # Gradient of val w.r.t. a:
    # val = (1/beta) * log(sum_k exp(beta * b_k))
    # d(val)/d(a_j) = (1/sum_k exp(beta*b_k)) * sum_k beta * exp(beta*b_k) * db_k/d(a_j)
    # = sum_k p_k * db_k/d(a_j)
    # db_k/d(a_j) = 2*a[k-j] (as before)
    # So grad = sum_k p_k * 2 * shift(a, k)
    # = 2 * conv(p_reversed, a) ... let me think
    
    # Actually: grad_j = sum_k p_k * 2 * a[k-j] (with bounds)
    # = 2 * sum_k p_k * a[k-j]
    # This is 2 * (p * a) with appropriate indexing
    # = 2 * convolve(p, a)[j] ... not quite
    
    # Let me compute it directly using the correlation:
    # grad_j = 2 * sum_{k: 0<=k-j<n} p_k * a[k-j]
    # Let m = k-j, so k = j+m:
    # grad_j = 2 * sum_{m: 0<=j+m<2n-1, 0<=m<n} p_{j+m} * a[m]
    # = 2 * sum_{m=0}^{min(n-1, 2n-2-j)} p_{j+m} * a[m]
    
    # This is a cross-correlation: grad = 2 * corr(p, a)
    # In numpy: np.correlate(p, a, mode='full')[n-1:2n-1]
    # Or: grad[j] = 2 * np.dot(p[j:2n-1], a[:min(n, 2n-1-j)])
    
    # Efficient computation using FFT:
    # grad = 2 * ifft(fft(p) * conj(fft(a))) ... this gives circular correlation
    # For linear: pad and use FFT
    
    # Simpler: use the fact that grad_j = 2 * (p_crosscorrelated_with_a)[j]
    # Let's compute via: for each k, add p_k * 2 * shift(a, k) to grad
    # This is: grad = 2 * sum_k p_k * T_k * a where T_k is the shift-reverse operator
    
    # Most efficient: use FFT
    # The operation is: grad[j] = 2 * sum_{m} p[j+m] * a[m] for valid indices
    # This is the cross-correlation of p and a: (p ⊛ a)[j] = sum_m p[j+m] * a[m]
    # = flip of convolution: (flip(p) * a)[n-1-j] ... 
    
    # Let me just use direct computation for simplicity (n up to ~2048, this is fine)
    # grad[j] = 2 * sum_{m=0}^{n-1} p[j+m] * a[m] if j+m < 2n-1
    
    # Actually, for n up to 2048, I'll use the FFT approach:
    s = 1
    while s < 2*n-1:
        s <<= 1
    P = np.fft.rfft(p, s)
    A = np.fft.rfft(a, s)
    # Cross-correlation: ifft(conj(P) * A) gives sum_k p[k] * a[k+j] (circular)
    # We want sum_k p[k+j] * a[k] = sum_m p[m] * a[m-j] ... 
    
    # Let me just do it the straightforward way with the shift:
    # grad[j] = 2 * sum_{k=max(j,0)}^{min(2n-2, j+n-1)} p[k] * a[k-j]
    # = 2 * (for valid k, p[k]*a[k-j])
    
    # This is: convolve a with reversed p, at the right offset
    # conv(a, p)[j] = sum_m a[m] * p[j-m] = sum_m a[m] * p[j-m]
    # I want sum_k p[k] * a[k-j] = sum_m p[m+j] * a[m] (letting m=k-j)
    # = conv(p, a)[j] where conv(p,a)[j] = sum_m p[m] * a[j-m]... 
    
    # OK let me just compute it directly using a loop over the nonzero p entries
    # Or better, use the matrix form:
    # grad = 2 * M^T * p where M[k,j] = a[k-j] (the same A matrix from the LP!)
    # So grad = 2 * A^T * p
    
    # This is a sparse matrix-vector product!
    rows = []
    cols = []
    for k in range(2*n-1):
        jl = max(0, k - n + 1)
        jh = min(n-1, k)
        for j in range(jl, jh+1):
            rows.append(j)
            cols.append(k)
    # grad[j] = 2 * sum_k A[k,j] * p[k] = 2 * sum_k a[k-j] * p[k] (when valid)
    
    # Let me just use np for this:
    grad = np.zeros(n)
    for j in range(n):
        # grad[j] = 2 * sum_k p[k] * a[k-j] for valid k
        k_min = j  # so that k-j >= 0
        k_max = min(2*n-2, j+n-1)  # so that k-j < n
        if k_min <= k_max:
            grad[j] = 2.0 * np.dot(p[k_min:k_max+1], a[0:k_max-k_min+1])
    
    return val, grad

# Actually the loop above is O(n²) which is too slow for n=1024. Let me use FFT.
# grad[j] = 2 * sum_{m=0}^{n-1} a[m] * p[j+m] (for j+m < 2n-1)
# This is the cross-correlation: (a ⊛ p)[j] where (a⊛p)[j] = sum_m a[m]*p[j+m]
# = ifft(fft(a) * conj(fft(p))) in the circular case
# For linear: pad both to length >= 2n-1

def smooth_grad_fft(a, p):
    """Compute grad[j] = 2 * sum_m a[m] * p[j+m] (linear correlation)."""
    n = len(a)
    s = 1
    while s < 2*n-1:
        s <<= 1
    A = np.fft.rfft(a, s)
    P = np.fft.rfft(p, s)
    # Circular cross-correlation: ifft(A * conj(P)) gives sum_k a[k]*p[k+j] (circular)
    # We want the linear version
    corr = np.fft.irfft(A * np.conj(P), s)[:2*n-1]
    # corr[j] = sum_{k=0}^{s-1-j} a[k]*p[k+j] (approximately, for the linear part)
    # Actually for zero-padded: corr[j] = sum_{k: k+k+j < 2n-1} a[k]*p[k+j] = sum_k a[k]*p[k+j]
    # which is exactly what we want!
    return 2.0 * corr[:n]

def lbfgs_optimize(a0, deadline, beta_schedule=None):
    """L-BFGS-B optimization with log-sum-exp smoothing and beta annealing."""
    if beta_schedule is None:
        beta_schedule = [(2.0, 30), (10.0, 40), (50.0, 50), (200.0, 60)]
    
    a = np.maximum(a0.astype(np.float64), 1e-10)
    a = a / a.sum()
    
    for beta, max_iters in beta_schedule:
        if time.monotonic() > deadline:
            break
        
        n = len(a)
        
        def func(x):
            x = np.maximum(x, 0)
            s = x.sum()
            if s < 1e-10:
                return 1e10
            b = fast_conv(x)
            b_max = float(b.max())
            shifted = beta * (b - b_max)
            shifted = np.clip(shifted, -500, 500)
            Z = np.exp(shifted).sum()
            smooth_max = (b_max + np.log(max(Z, 1e-300))) / beta
            return 2.0 * n * smooth_max / (s * s)
        
        def jac(x):
            x = np.maximum(x, 0)
            s = x.sum()
            if s < 1e-10:
                return np.zeros(n)
            b = fast_conv(x)
            b_max = float(b.max())
            shifted = beta * (b - b_max)
            shifted = np.clip(shifted, -500, 500)
            exp_shifted = np.exp(shifted)
            Z = exp_shifted.sum()
            if Z < 1e-300:
                p = np.zeros(2*n-1)
                p[np.argmax(b)] = 1.0
            else:
                p = exp_shifted / Z
            smooth_max = (b_max + np.log(max(Z, 1e-300))) / beta
            grad_smooth = smooth_grad_fft(x, p)
            # d/dx [2n * smooth_max / s²] = 2n * (s * grad_smooth - 2*smooth_max) / s³
            grad = 2.0 * n * (s * grad_smooth - 2.0 * smooth_max * np.ones(n)) / (s ** 3)
            grad = np.nan_to_num(grad, nan=0.0, posinf=0.0, neginf=0.0)
            grad[x < 1e-12] = 0.0
            return grad
        
        bounds = [(0.0, None)] * n
        try:
            res = minimize(func, a.copy(), jac=jac, method='L-BFGS-B',
                          bounds=bounds,
                          options={'maxiter': max_iters, 'max_fun': max_iters * 30,
                                   'gtol': 1e-8, 'ftol': 1e-12})
            if res.fun < 1e10:
                new_a = np.maximum(res.x, 1e-10)
                if new_a.sum() > 1e-6:
                    a = new_a / new_a.sum()
        except Exception:
            pass
    
    return a

def resample_seq(src, dest_n):
    """Resample 1D sequence to new length."""
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    result = np.interp(t_new, t_old, src)
    return np.maximum(result, 0.0)
