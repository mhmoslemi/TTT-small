"""
Minimizes J(a) = 2*n*max(convolve(a,a))/(sum(a))^2 using:
1. LP-based alternating optimization with proper 1D line search
2. Symmetric sequence optimization (reduces problem size)
3. Multi-length exploration
4. Stochastic perturbations for escaping local minima
5. Efficient coordinate descent with incremental convolution updates
"""

import numpy as np
from scipy.optimize import linprog
import time

def fast_conv(a):
    """Full self-convolution via FFT."""
    n = len(a)
    s = 1
    while s < 2 * n - 1:
        s <<= 1
    fa = np.fft.rfft(a, s)
    return np.fft.irfft(fa * fa, s)[:2 * n - 1]

def compute_J(a):
    """Compute J(a) = 2*n*max(conv(a,a))/(sum(a))^2."""
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))

def build_A_for_lp(a):
    """Build the constraint matrix for the LP.
    A[k,j] = a[k-j] if 0 <= k-j < n, else 0
    """
    n = len(a)
    # A is (2n-1) x n
    A = np.zeros((2 * n - 1, n))
    for k in range(2 * n - 1):
        jl = max(0, k - n + 1)
        jh = min(n - 1, k)
        for j in range(jl, jh + 1):
            A[k, j] = a[k - j]
    return A

def lp_best_response(a, M):
    """LP: max sum(g) s.t. (a*g)_k <= M for all k, g >= 0."""
    n = len(a)
    c = -np.ones(n)
    A = build_A_for_lp(a)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
    if r.success and r.x is not None:
        g = np.maximum(r.x, 0.0)
        if g.sum() > 1e-10:
            return g
    return None

def line_search_mix(a, g, max_evals=30):
    """Find optimal t in (1-t)*a + t*g using golden section search.
    Returns the best sequence found."""
    a = np.maximum(a, 0)
    g = np.maximum(g, 0)
    a_sum = a.sum()
    g_sum = g.sum()
    
    def obj(t):
        x = (1 - t) * a + t * g
        x = np.maximum(x, 0)
        s = x.sum()
        if s < 1e-10:
            return 1e10
        x_norm = x / s
        b = fast_conv(x_norm)
        return 2.0 * len(x_norm) * b.max()
    
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    lo = 0.0
    hi = 1.0
    
    c_pt = hi - gr * (hi - lo)
    d_pt = lo + gr * (hi - lo)
    fc = obj(c_pt)
    fd = obj(d_pt)
    
    for _ in range(max_evals):
        if fc < fd:
            hi = d_pt
            d_pt = c_pt
            fd = fc
            c_pt = hi - gr * (hi - lo)
            fc = obj(c_pt)
        else:
            lo = c_pt
            c_pt = d_pt
            fc = fd
            d_pt = lo + gr * (hi - lo)
            fd = obj(d_pt)
    
    t_opt = (lo + hi) / 2.0
    x = (1 - t_opt) * a + t_opt * g
    x = np.maximum(x, 0)
    s = x.sum()
    if s < 1e-10:
        return a
    return x / s

        g = np.maximum(g, 0.0)

            # If 0 <= i-(k-i) < n i.e. 0 <= 2i-k < n: a[i]*a[2i-k] is a term
            # But if k-i = i (i.e., k = 2i), then a[i]*a[i] = a[i]^2 appears once
            # So b_new[k] = b[k] + (x-ai)*(a[k-i] if 0<=k-i<n else 0) + (x-ai)*(a[2i-k] if 0<=2i-k<n else 0) - (x-ai)*([1 if k-i==i and 0<=k-i<n else 0] + [1 if 2i-k==i and 0<=2i-k<n else 0]) + ... 
            
            # This is getting too complex for a clean formula. Let me use a simpler approach:
            # Just evaluate J for a few candidate values of x.
            
            S = a_sum
            candidates_x = [0.0, ai * 0.5, ai * 0.75, ai, ai * 1.25, ai * 1.5, 2 * ai, S / n * 2]
            candidates_x = [max(0, min(1000, c)) for c in candidates_x]
            
            best_x = ai
            best_j = current_J
            
            for x in candidates_x:
                if abs(x - ai) < 1e-8:
                    continue
                a_test = a.copy()
                a_test[i] = x
                s_test = S - ai + x
                if s_test < 1e-10:
                    continue
                a_test = a_test / s_test
                # Efficient conv update... actually for n=386, just do FFT
                b_test = fast_conv(a_test)
                j_test = 2.0 * n * b_test.max() / (s_test * s_test)  # Wait, a_test is already normalized
                # No: J = 2*n*max(conv(a_test,a_test)) / sum(a_test)^2
                # But a_test is already normalized to sum 1, so:
                j_test = 2.0 * n * fast_conv(a_test).max()
                if j_test < best_j:
                    best_j = j_test
                    best_x = x
            
            if best_x != ai:
                a[i] = best_x
                a_sum = S - ai + best_x
                a = a / a_sum
                current_J = best_j
                improved = True
        
        if not improved:
            break
    
    return a

class ScoreTracker:
    def __init__(self):
        self.best = None
        self.best_score = float('inf')
        self.t_start = time.monotonic()
    
    def update(self, a):
        a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
        if a.size == 0 or a.sum() < 0.01:
            return
        score = float(evaluate_sequence(a.tolist()))
        if np.isfinite(score) and score < self.best_score:
            self.best = a.copy()
            self.best_score = score
            elapsed = time.monotonic() - self.t_start
            print(f"  * NEW BEST: J={score:.8f} n={a.size} t={elapsed:.0f}s", flush=True)

        if self.best is not None:
            print(f"  {label} best: J={self.best_score:.8f} n={self.best.size} t={time.monotonic()-self.t_start:.0f}s", flush=True)

def resample_seq(src, dest_n):
    """Resample 1D sequence to new length via linear interpolation."""
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)

def generate_inits(n, rng):
    """Generate diverse initial sequences of length n."""
    inits = []
    t = np.linspace(0, 1, n, endpoint=False)
    
    inits.append(np.ones(n) / n)
    inits.append(rng.uniform(0.5, 1.5, n))
    
    for freq in [2, 3, 5, 7]:
        inits.append(np.maximum(1.0 + 0.3 * np.cos(2.0 * np.pi * freq * t), 0.1))
    
    inits.append(1.0 + 0.2 * np.cos(2*np.pi*3*t) + 0.15 * np.cos(2*np.pi*7*t))
    inits.append(np.abs(np.sin(np.pi * t)) + 0.2)
    inits.append(0.5 + 0.5 * np.cos(2 * np.pi * t) ** 2)
    inits.append(1.0 - 0.4 * np.abs(2 * t - 1.0))
    
    # Symmetric tent
    inits.append(np.abs(1 - 2*t) * 0.5 + 0.5)
    # Symmetric bell
    inits.append(np.exp(-4 * (t - 0.5)**2 / (0.5*0.5)))
    
    for init in inits:
        init = np.maximum(init, 0.01)
        init = init / init.sum()
        inits[-1] = init
    
    return inits
