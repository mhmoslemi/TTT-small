import numpy as np
from scipy.optimize import minimize, linprog
from scipy.fft import rfft, irfft

def compute_conv(a):
    """Compute linear convolution a*a using FFT. Returns array of length 2n-1."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    conv = irfft(fa * fa, size)
    return conv[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    conv = compute_conv(a)
    max_b = float(np.max(conv))
    sum_a = float(np.sum(a))
    if sum_a < 0.01:
        return np.inf
    return 2.0 * n * max_b / (sum_a * sum_a)


def fft_convolve(a, b):
    """Linear convolution of a and b using FFT."""
    n, m = len(a), len(b)
    size = 1
    while size < n + m - 1:
        size <<= 1
    fa = rfft(a, size)
    fb = rfft(b, size)
    return irfft(fa * fb, size)[:n + m - 1]


def softmax(z):
    """Numerically stable softmax."""
    z = z - np.max(z)
    e = np.exp(z)
    s = np.sum(e)
    if s < 1e-300:
        return np.ones(len(z)) / len(z)
    return e / s


def power_mean_obj_grad(z, p):
    """
    Compute power-mean objective (sum conv^p)^(1/p) and gradient w.r.t. z
    for a = softmax(z). The full objective is 2*n*PM (since sum(softmax)=1).
    """
    n = len(z)
    a = softmax(z)
    conv = compute_conv(a)
    conv_pos = np.maximum(conv, 1e-30)

    pm_val = float(np.sum(conv_pos ** p)) ** (1.0 / p)

    # Weights: w[k] = conv[k]^(p-1) / PM^(p-1)
    w = conv_pos ** (p - 1) / pm_val ** (p - 1)

    # cross[j] = sum_k w[k] * a[k-j] (with a[m]=0 outside [0,n-1])
    # = sum_{m=0}^{n-1} a[m] * w[m+j]
    # Use FFT for cross-correlation
    size_c = 1
    while size_c < 3 * n:
        size_c <<= 1
    fa_c = rfft(a, size_c)
    fw_c = rfft(w, size_c)
    # cross[j] = sum_m a[m]*w[m+j] = IFFT(conj(FFT(a)) * FFT(w))[j] (with proper shift)
    # Actually: IFFT(conj(F)*G)[j] = sum_m a[m] * g[m-j] (circular)
    # We want sum_m a[m]*w[m+j], which is IFFT(conj(F)*G)[-j] or IFFT(F*conj(G))[j]
    # Let's use: cross[j] = IFFT(FFT(w) * conj(FFT(a)))[size_c - j] ... this is getting confusing
    # Let me just do it directly for n <= 1500
    if n <= 1500:
        cross = np.zeros(n)
        for j in range(n):
            m = np.arange(n)
            idx = m + j
            valid = idx < len(w)
            cross[j] = np.dot(a[m[valid]], w[idx[valid]])
    else:
        # FFT approach: cross[j] = sum_m a[m]*w[m+j]
        # Pad a to length size_c, pad w to length size_c
        # IFFT(FFT(w) * conj(FFT(a))) gives h[k] = sum_m w[m] * a[m-k] (circular)
        # So h[k] = sum_m w[m] * a[(m-k) mod size_c]
        # We want cross[j] = sum_m a[m]*w[m+j] = sum_m w[m+j]*a[m]
        # Let t = m+j: cross[j] = sum_t w[t]*a[t-j]
        # This is the cross-correlation of a and w at lag j
        # IFFT(conj(FFT(a)) * FFT(w))[j] = sum_m a[m]*w[m+j] for linear case with padding
        corr = irfft(np.conj(fa_c) * fw_c, size_c)
        cross = corr[:n]

    grad_z = 4.0 * n * a * (cross - pm_val)

    # Scale by 2*n for full objective
    full_val = 2.0 * n * pm_val
    full_grad = 4.0 * n * a * (cross - pm_val)

    return full_val, full_grad


def logsumexp_obj_grad(z, tau):
    """Log-sum-exp smoothed max of convolution, times 2*n. Plus gradient w.r.t. z."""
    n = len(z)
    a = softmax(z)
    conv = compute_conv(a)
    cmax = float(np.max(conv))
    shifted = np.clip((conv - cmax) / tau, -300.0, 300.0)
    exp_conv = np.exp(shifted)
    Z = float(np.sum(exp_conv))
    lse_val = cmax + tau * np.log(Z)
    w = exp_conv / Z

    # cross[j] = sum_k w[k] * a[k-j]
    if n <= 1500:
        cross = np.zeros(n)
        for j in range(n):
            m = np.arange(n)
            idx = m + j
            valid = idx < len(w)
            cross[j] = np.dot(a[m[valid]], w[idx[valid]])
    else:
        size_c = 1
        while size_c < 3 * n:
            size_c <<= 1
        fa_c = rfft(a, size_c)
        fw_c = rfft(w, size_c)
        corr = irfft(np.conj(fa_c) * fw_c, size_c)
        cross = corr[:n]

    # Objective: 2*n*LSE
    # d(2n*LSE)/dz[j] = 2*n * 2*a[j] * (cross[j] - LSE) = 4*n*a[j]*(cross[j] - LSE)
    full_val = 2.0 * n * lse_val
    full_grad = 4.0 * n * a * (cross - lse_val)

    return full_val, full_grad


def optimize_lfbs_powermean(a_init, p_list, max_iter=200, time_budget=30.0):
    """Optimize using L-BFGS-B with power-mean parametrization and p annealing."""
    n = len(a_init)
    a = a_init.copy()
    s = np.sum(a)
    if s < 1e-10:
        a = np.ones(n) / n
    else:
        a = a / s
    a = np.maximum(a, 1e-30)
    z = np.log(a)

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    for p in p_list:
        if time.monotonic() - start > time_budget:
            break

        def func_and_grad(z_val):
            return power_mean_obj_grad(z_val, p)

        try:
            result = minimize(func_and_grad, z, jac=True, method='L-BFGS-B',
                              options={'maxiter': max_iter, 'ftol': 1e-15, 'gtol': 1e-10})
            z_new = result.x
        except Exception:
            continue

        a_new = softmax(z_new)
        a_new = np.maximum(a_new, 0)
        s_new = np.sum(a_new)
        if s_new < 1e-10:
            continue
        a_new = a_new / s_new
        obj = compute_obj(a_new)

        if obj < best_obj:
            best_obj = obj
            best_a = a_new.copy()
            z = z_new

    return best_a, best_obj

def build_toeplitz(a):
    """Build Toeplitz matrix where (A@g)[k] = (conv(a,g))[k]."""
    n = len(a)
    A = np.zeros((2 * n - 1, n), dtype=np.float64)
    for j in range(n):
        k_start = j
        k_end = min(j + n, 2 * n - 1)
        A[k_start:k_end, j] = a[:k_end - k_start]
    return A


def solve_lp_g(a, M, time_limit=15):
    """Solve LP: max sum(g) s.t. (a*g)[k] <= M for all k, g >= 0.
    Returns (g_normalized, sum_g, dual_variables)."""
    n = len(a)
    A = build_toeplitz(a)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': time_limit})
        if res.status == 0:
            g = np.maximum(res.x, 0)
            s = float(np.sum(g))
            if s > 1e-10:
                duals = res.uneqfun.values[0] if res.uneqfun is not None else None
                return g / s, s, duals
    except Exception:
        pass
    return None, None, None


def solve_lp_g_with_dual(a, M, time_limit=15):
    """Solve LP and extract dual variables for subgradient estimation."""
    n = len(a)
    A = build_toeplitz(a)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': time_limit})
        if res.status == 0:
            g = np.maximum(res.x, 0)
            s = float(np.sum(g))
            if s > 1e-10:
                # Dual variables from res
                duals = None
                if hasattr(res, 'ineqlin') and res.ineqlin is not None:
                    duals = np.array(res.ineqlin['marginals'])
                elif hasattr(res, 'uneqfun') and res.uneqfun is not None:
                    duals = np.array(res.uneqfun['Values'])
                return g / s, s, duals
    except Exception:
        pass
    return None, None, None


def alternating_lp_golden(a, max_iters=50, time_budget=40.0):
    """Alternating LP with golden-section line search."""
    n = len(a)
    a = a.copy().astype(np.float64)
    s = np.sum(a)
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    stall_count = 0
    patience = 15

    # Golden section constants
    gr = (np.sqrt(5.0) - 1.0) / 2.0  # ~0.618

    for it in range(max_iters):
        if time.monotonic() - start > time_budget:
            break
        remaining = time_budget - (time.monotonic() - start)
        if remaining < 5:
            break

        conv = compute_conv(a)
        M = float(np.max(conv))

        lp_time = min(max(3.0, remaining * 0.15), 12.0)
        g, _, _ = solve_lp_g(a, M, time_limit=lp_time)
        if g is None:
            break

        # Golden section line search on t in [0, 1]
        def obj_at_t(t):
            a_try = (1.0 - t) * a + t * g
            a_try = np.maximum(a_try, 0)
            s_try = np.sum(a_try)
            if s_try < 1e-10:
                return np.inf
            a_try = a_try / s_try
            return compute_obj(a_try)

        # Golden section search
        a_gs = 0.0
        b_gs = 1.0
        c_gs = b_gs - gr * (b_gs - a_gs)
        d_gs = a_gs + gr * (b_gs - a_gs)
        fc = obj_at_t(c_gs)
        fd = obj_at_t(d_gs)

        for _ in range(8):
            if fc < fd:
                b_gs = d_gs
                d_gs = c_gs
                fd = fc
                c_gs = b_gs - gr * (b_gs - a_gs)
                fc = obj_at_t(c_gs)
            else:
                a_gs = c_gs
                c_gs = d_gs
                fc = fd
                d_gs = a_gs + gr * (b_gs - a_gs)
                fd = obj_at_t(d_gs)

        t_best = (a_gs + b_gs) / 2.0
        a_try = (1.0 - t_best) * a + t_best * g
        a_try = np.maximum(a_try, 0)
        s_try = np.sum(a_try)
        if s_try < 1e-10:
            break
        a_try = a_try / s_try
        obj_try = compute_obj(a_try)

        # Also check t=1
        obj_t1 = compute_obj(np.maximum(g, 0) / np.sum(np.maximum(g, 0))) if np.sum(g) > 1e-10 else np.inf

        if obj_try < best_obj - 1e-12:
            best_obj = obj_try
            best_a = a_try.copy()
            a = a_try
            stall_count = 0
        elif obj_t1 < best_obj - 1e-12:
            best_obj = obj_t1
            best_a = g.copy()
            a = g.copy()
            stall_count = 0
        else:
            stall_count += 1
            if stall_count > patience:
                rng = np.random.default_rng(it * 104729 + 7)
                scale = 0.05 + 0.05 * (stall_count - patience) / 10.0
                a = a * np.exp(rng.normal(0, min(scale, 0.15), n))
                a = np.maximum(a, 0)
                s_new = np.sum(a)
                if s_new > 1e-10:
                    a = a / s_new
                best_obj = compute_obj(a)
                best_a = a.copy()
                stall_count = 0
                patience = 8
            else:
                t_small = 0.1
                a_try2 = (1.0 - t_small) * a + t_small * g
                a_try2 = np.maximum(a_try2, 0)
                s_try2 = np.sum(a_try2)
                if s_try2 > 1e-10:
                    a = a_try2 / s_try2

    return best_a


def subgradient_optimize_v2(a, max_iters=2000, time_budget=20.0, p_start=10.0, p_end=50.0):
    """
    Power-mean subgradient optimization with adaptive step size.
    Uses the power-mean gradient for smooth subgradient estimation.
    """
    n = len(a)
    a = a.copy().astype(np.float64)
    s = np.sum(a)
    if s < 1e-10:
        a = np.ones(n) / n
    else:
        a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    # Polyak-style step size
    obj_upper = best_obj + 0.1
    step = 0.05
    stall = 0
    min_step = 1e-6

    for it in range(max_iters):
        if time.monotonic() - start > time_budget:
            break

        frac = it / max(1, max_iters)
        p_cur = p_start + (p_end - p_start) * frac

        conv = compute_conv(a)
        conv_pos = np.maximum(conv, 1e-30)
        pm_val = float(np.sum(conv_pos ** p_cur)) ** (1.0 / p_cur)
        w = conv_pos ** (p_cur - 1) / pm_val ** (p_cur - 1)

        # Compute gradient direction
        if n <= 1000:
            cross = np.zeros(n)
            for j in range(n):
                m = np.arange(n)
                idx = m + j
                valid = idx < len(w)
                cross[j] = np.dot(a[m[valid]], w[idx[valid]])
        else:
            size_c = 1
            while size_c < 3 * n:
                size_c <<= 1
            fa_c = rfft(a, size_c)
            fw_c = rfft(w, size_c)
            corr = irfft(np.conj(fa_c) * fw_c, size_c)
            cross = corr[:n]

        # Gradient of 2n*PM: grad_i = 4*n*a[i]*(cross[i] - PM)
        grad = 4.0 * n * a * (cross - pm_val)

        # Center the gradient (since we're on the simplex, the direction should be tangent)
        grad_centered = grad - np.sum(a * grad)

        # Adaptive step size
        grad_norm = np.sqrt(np.sum(a * grad_centered * grad_centered))
        if grad_norm < 1e-15:
            break

        # Mirror descent step in log space
        z = np.log(np.maximum(a, 1e-30))
        z_new = z - step * grad_centered / (0.5 * (a + 1e-30))
        a_new = softmax(z_new)
        a_new = np.maximum(a_new, 0)
        s_new = np.sum(a_new)
        if s_new > 1e-10:
            a_new = a_new / s_new
        else:
            a_new = a.copy()

        a = a_new
        obj = compute_obj(a)

        if obj < best_obj - 1e-12:
            best_obj = obj
            best_a = a.copy()
            stall = 0
            # Update Polyak bound
            obj_upper = max(best_obj + 0.01, 0.5 * obj_upper + 0.5 * best_obj)
        else:
            stall += 1
            if stall > 80:
                step *= 0.85
                if step < min_step:
                    break
                # Perturb to escape
                rng = np.random.default_rng(it * 31 + 7)
                a = a * np.exp(rng.normal(0, 0.03, n))
                a = np.maximum(a, 0)
                s = np.sum(a)
                if s > 1e-10:
                    a = a / s
                stall = 0

        # Gradual step decrease
        if it > 200:
            step *= 0.999

    return best_a


def random_init(n, seed):
    """Generate diverse random initializations."""
    rng = np.random.default_rng(seed)
    k = np.arange(n)
    choice = rng.integers(0, 12)

    if choice == 0:
        f = rng.uniform(0.3, 3.0)
        phase = rng.uniform(0, 2 * np.pi)
        a = 1.0 + 0.7 * np.cos(2 * np.pi * f * k / n + phase)
    elif choice == 1:
        f1 = rng.uniform(0.2, 1.5)
        f2 = rng.uniform(2.0, 5.0)
        a = 1.0 + 0.3 * np.cos(2 * np.pi * f1 * k / n) + 0.3 * np.cos(2 * np.pi * f2 * k / n)
    elif choice == 2:
        center = n * rng.uniform(0.2, 0.8)
        alpha = rng.uniform(0.5, 4.0)
        a = 1.0 + 0.4 * np.exp(-alpha * np.abs(k - center) / (n * 0.3))
    elif choice == 3:
        a = rng.random(n)
        win = max(3, n // 15)
        kernel = np.ones(win) / win
        a = np.convolve(a, kernel, mode='same')
    elif choice == 4:
        a = np.ones(n)
        for _ in range(rng.integers(1, 6)):
            f = rng.uniform(0.3, 5.0)
            amp = rng.uniform(0.1, 0.5)
            a += amp * np.cos(2 * np.pi * f * k / n + rng.uniform(0, 2 * np.pi))
    elif choice == 5:
        f = rng.uniform(0.3, 2.0)
        a = 1.0 + 0.4 * np.sin(2 * np.pi * f * k / n)
    elif choice == 6:
        nblocks = rng.integers(2, 8)
        a = np.ones(n)
        for b in range(nblocks):
            start = b * n // nblocks
            end = (b + 1) * n // nblocks
            a[start:end] = rng.uniform(0.2, 1.5)
    elif choice == 7:
        f0 = rng.uniform(0.5, 2.0)
        f1 = rng.uniform(2.0, 6.0)
        a = 1.0 + 0.4 * np.cos(2 * np.pi * (f0 * k / n + (f1 - f0) * k**2 / (2 * n**2)))
    elif choice == 8:
        center = n * rng.uniform(0.2, 0.8)
        width = n * rng.uniform(0.1, 0.4)
        a = 1.0 + 0.5 * np.exp(-0.5 * ((k - center) / width) ** 2)
    elif choice == 9:
        # Multi-peak
        npeaks = rng.integers(2, 5)
        a = np.ones(n)
        for _ in range(npeaks):
            c = rng.uniform(0, n)
            w = rng.uniform(0.05, 0.2) * n
            a += 0.3 * np.exp(-0.5 * ((k - c) / w) ** 2)
    elif choice == 10:
        # Sawtooth
        period = rng.uniform(n * 0.1, n * 0.4)
        a = 1.0 + 0.3 * np.sinc(k / (period / 2))
    else:
        a = 0.5 + 0.5 * rng.random(n)

    a = np.maximum(a, 0.01)
    s = np.sum(a)
    return a / s


def resample_sequence(a, new_n):
    """Resample sequence to new length via interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return a_new


def try_parent():
    """Try to load parent sequence."""
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64)
        parent = np.clip(parent, 0, 1000.0)
        if parent.size > 0 and np.isfinite(parent).all():
            obj = compute_obj(parent)
            print(f"Parent (n={parent.size}): obj={obj:.6f}")
            return parent.copy(), obj
    except Exception:
        pass
    return None, np.inf


def polish_sequence(a, time_budget, verbose=False):
    """Intensive polishing of a single sequence with multiple methods."""
    n = len(a)
    a = a.copy().astype(np.float64)
    start = time.monotonic()

    best_a = a.copy()
    best_obj = compute_obj(a)
    round_num = 0

    while time.monotonic() - start < time_budget - 3:
        round_num += 1
        remaining = time_budget - (time.monotonic() - start)
        if remaining < 8:
            break

        improved = False

        # LP alternating with golden section
        if n <= 800 and remaining > 20:
            a_new = alternating_lp_golden(best_a, max_iters=40, time_budget=min(remaining * 0.3, 45.0))
            obj = compute_obj(a_new)
            if obj < best_obj - 1e-12:
                best_obj = obj
                best_a = a_new.copy()
                improved = True
                if verbose:
                    print(f"    Polish r{round_num} LP: {best_obj:.7f}")

        # L-BFGS-B with power mean (high p)
        remaining = time_budget - (time.monotonic() - start)
        if remaining > 15:
            p_list = [50.0, 100.0, 200.0]
            a_new, obj_new = optimize_lfbs_powermean(best_a, p_list,
                                                     max_iter=150, time_budget=min(remaining * 0.2, 20.0))
            if obj_new < best_obj - 1e-12:
                best_obj = obj_new
                best_a = a_new.copy()
                improved = True
                if verbose:
                    print(f"    Polish r{round_num} PM-LBFGS: {best_obj:.7f}")

        # L-BFGS-B with LSE (small tau)
        remaining = time_budget - (time.monotonic() - start)
        if remaining > 12:
            tau_list = [0.5 / n, 0.2 / n, 0.1 / n, 0.05 / n]
            a_new, obj_new = optimize_lfbs_lse(best_a, tau_list,
                                               max_iter=150, time_budget=min(remaining * 0.2, 18.0))
            if obj_new < best_obj - 1e-12:
                best_obj = obj_new
                best_a = a_new.copy()
                improved = True
                if verbose:
                    print(f"    Polish r{round_num} LSE-LBFGS: {best_obj:.7f}")

        # Subgradient with power mean
        remaining = time_budget - (time.monotonic() - start)
        if remaining > 10:
            a_new = subgradient_optimize_v2(best_a, max_iters=int(remaining * 8),
                                            time_budget=min(remaining * 0.2, 15.0),
                                            p_start=20.0, p_end=80.0)
            obj = compute_obj(a_new)
            if obj < best_obj - 1e-12:
                best_obj = obj
                best_a = a_new.copy()
                improved = True
                if verbose:
                    print(f"    Polish r{round_num} SG: {best_obj:.7f}")

        if not improved:
            # Try a small perturbation
            rng = np.random.default_rng(round_num * 2654435761)
            a_pert = best_a * np.exp(rng.normal(0, 0.02, n))
            a_pert = np.maximum(a_pert, 0)
            s = np.sum(a_pert)
            if s > 1e-10:
                a_pert = a_pert / s
                obj_pert = compute_obj(a_pert)
                if obj_pert < best_obj:
                    best_obj = obj_pert
                    best_a = a_pert
                else:
                    # No improvement, break if we've been stuck
                    if round_num > 10:
                        break

    return best_a, best_obj


def propose_candidate(seed=42, budget_s=1500):
    """
    Main search: explores multiple sequence lengths using power-mean and LSE
    smoothed optimization, LP alternating with golden-section line search,
    and active-set aware subgradient methods.
    """
    start_time = time.monotonic()
    deadline = start_time + float(budget_s) - 30.0

    best_seq = None
    best_score = np.inf

    # Load parent
    parent, parent_obj = try_parent()
    if parent is not None and parent_obj < best_score:
        best_score = parent_obj
        best_seq = parent.copy()

    # N values to explore
    np_ = len(best_seq) if best_seq is not None else 600
    n_values = [500, 550, 575, 600, 625, 650, 700, 750]
    # Add parent length and neighbors
    extra = [np_, max(100, np_ - 25), np_ + 25, np_ + 50]
    n_values = sorted(set(n_values + [x for x in extra if 100 <= x <= 900]))

    print(f"Exploring n values: {n_values}")
    print(f"Initial best: {best_score:.7f}")

    candidates = []

    # Phase 1: Exploration
    for n in n_values:
        if time.monotonic() > deadline - 180:
            break
        remaining = deadline - time.monotonic()
        if remaining < 15:
            break

        # Time allocation
        if n <= 550:
            alloc = min(40.0, remaining * 0.1)
        elif n <= 650:
            alloc = min(55.0, remaining * 0.12)
        else:
            alloc = min(65.0, remaining * 0.13)
        alloc = max(15.0, alloc)

        print(f"\nN={n}: {remaining:.0f}s remaining, alloc={alloc:.0f}s")

        n_inits = max(2, int(alloc / 20))
        n_inits = min(n_inits, 4)

        for init_idx in range(n_inits):
            if time.monotonic() > deadline - 180:
                break
            remaining_init = deadline - time.monotonic()
            if remaining_init < 15:
                break

            if init_idx == 0 and best_seq is not None:
                a = resample_sequence(best_seq, n)
            else:
                a = random_init(n, seed + n * 104729 + init_idx * 79193)

            # Step 1: Quick subgradient shaping
            sg_time = min(6.0, (alloc / n_inits) * 0.15)
            a = subgradient_optimize_v2(a, max_iters=int(sg_time * 30),
                                         time_budget=sg_time,
                                         p_start=5.0, p_end=30.0)

            # Step 2: L-BFGS-B with power mean
            remaining_init = deadline - time.monotonic()
            if remaining_init > 20:
                p_list = [10.0, 20.0, 50.0, 100.0]
                lfbs_time = min((alloc / n_inits) * 0.35, 30.0)
                a, obj = optimize_lfbs_powermean(a, p_list,
                                                  max_iter=200, time_budget=lfbs_time)
            else:
                obj = compute_obj(a)

            # Step 3: L-BFGS-B with LSE (small tau)
            remaining_init = deadline - time.monotonic()
            if remaining_init > 15:
                tau_list = [1.0 / n, 0.5 / n, 0.2 / n, 0.1 / n]
                a, obj = optimize_lfbs_lse(a, tau_list,
                                            max_iter=150, time_budget=min(remaining_init * 0.2, 15.0))

            # Step 4: LP alternating
            remaining_init = deadline - time.monotonic()
            if remaining_init > 20 and n <= 750:
                a = alternating_lp_golden(a, max_iters=35,
                                           time_budget=min(remaining_init * 0.35, 35.0))
                obj = compute_obj(a)

            # Step 5: Final subgradient
            remaining_init = deadline - time.monotonic()
            if remaining_init > 8:
                a = subgradient_optimize_v2(a, max_iters=int(remaining_init * 10),
                                             time_budget=min(remaining_init * 0.15, 10.0),
                                             p_start=30.0, p_end=100.0)
                obj = compute_obj(a)

            print(f"  init{init_idx}: n={n}, obj={obj:.7f}")

            if obj < best_score:
                best_score = obj
                best_seq = a.copy()
                print(f"  ** UPDATE: n={n}, obj={best_score:.7f} **")

            candidates.append((a.copy(), obj, n))

    # Phase 2: Polishing
    print(f"\n=== POLISHING PHASE ===")
    print(f"Best before polish: {best_score:.7f}")

    candidates.sort(key=lambda x: x[1])

    n_polish = min(4, len(candidates))
    for ci in range(n_polish):
        if time.monotonic() > deadline - 80:
            break
        remaining = deadline - time.monotonic()
        if remaining < 25:
            break

        a_cand = candidates[ci][0]
        n_cand = candidates[ci][2]
        print(f"  Polishing candidate {ci} (n={n_cand}, obj={candidates[ci][1]:.7f}), {remaining:.0f}s left")

        polish_budget = min(remaining * 0.35, 100.0)
        a_pol, obj_pol = polish_sequence(a_cand, time_budget=polish_budget, verbose=True)

        if obj_pol < best_score:
            best_score = obj_pol
            best_seq = a_pol.copy()
            print(f"  ** UPDATE after polish: obj={best_score:.7f} **")

    # Phase 3: Perturbation-based restarts on best
    if best_seq is not None and time.monotonic() < deadline - 40:
        n_best = len(best_seq)
        print(f"\n=== PERTURBATION RESTARTS: n={n_best} ===")

        a_cur = best_seq.copy()
        for ptry in range(8):
            if time.monotonic() > deadline - 30:
                break
            remaining = deadline - time.monotonic()
            if remaining < 20:
                break

            rng = np.random.default_rng(ptry * 999983 + 42)
            scale = rng.uniform(0.02, 0.10)

            # Structured perturbation: target the active lags
            conv = compute_conv(a_cur)
            M = np.max(conv)
            active = conv > 0.95 * M
            n_active = np.sum(active)

            if n_active > 0 and n_active < len(conv):
                # Perturb more at positions that affect active lags
                pert = rng.normal(0, 1, n_best)
                # Emphasize positions that contribute to active lags
                emphasis = np.zeros(n_best)
                for k in range(len(active)):
                    if active[k]:
                        for j in range(max(0, k - n_best + 1), min(n_best, k + 1)):
                            emphasis[j] += 1
                emphasis = emphasis / (np.max(emphasis) + 1e-10)
                pert = pert * (1.0 + emphasis)
                a_pert = a_cur * np.exp(scale * pert)
            else:
                a_pert = a_cur * np.exp(scale * rng.normal(0, 1, n_best))

            a_pert = np.maximum(a_pert, 0)
            s = np.sum(a_pert)
            if s < 1e-10:
                continue
            a_pert = a_pert / s

            # Quick optimization
            a_pert = subgradient_optimize_v2(a_pert, max_iters=500, time_budget=4.0,
                                              p_start=20.0, p_end=60.0)
            if n_best <= 700:
                a_pert = alternating_lp_golden(a_pert, max_iters=20, time_budget=10.0)

            obj_pert = compute_obj(a_pert)
            if obj_pert < best_score:
                best_score = obj_pert
                best_seq = a_pert.copy()
                a_cur = a_pert.copy()
                print(f"  Perturb {ptry}: obj={best_score:.7f} ** UPDATE **")
            else:
                print(f"  Perturb {ptry}: obj={obj_pert:.7f}")

    # Phase 4: Final intensive polish
    if best_seq is not None and time.monotonic() < deadline - 15:
        remaining = deadline - time.monotonic()
        n_best = len(best_seq)
        print(f"\n=== FINAL POLISH: n={n_best}, {remaining:.0f}s ===")

        a_cur = best_seq.copy()
        for final_round in range(5):
            if time.monotonic() > deadline - 10:
                break
            remaining = deadline - time.monotonic()
            if remaining < 10:
                break

            # L-BFGS-B with very fine tau
            tau_vsmall = [0.3 / n_best, 0.1 / n_best, 0.05 / n_best, 0.02 / n_best]
            a_new, obj_new = optimize_lfbs_lse(a_cur, tau_vsmall,
                                               max_iter=300, time_budget=min(remaining * 0.2, 20.0))
            if obj_new < best_score:
                best_score = obj_new
                best_seq = a_new.copy()
                a_cur = a_new.copy()
                print(f"  Final LSE-LBFGS {final_round}: {best_score:.7f}")

            # Power mean with very high p
            remaining = deadline - time.monotonic()
            if remaining > 15:
                a_new, obj_new = optimize_lfbs_powermean(a_cur, [100.0, 200.0, 500.0],
                                                         max_iter=200, time_budget=min(remaining * 0.2, 15.0))
                if obj_new < best_score:
                    best_score = obj_new
                    best_seq = a_new.copy()
                    a_cur = a_new.copy()
                    print(f"  Final PM-LBFGS {final_round}: {best_score:.7f}")

            # LP
            remaining = deadline - time.monotonic()
            if remaining > 20 and n_best <= 750:
                a_new = alternating_lp_golden(a_cur, max_iters=40,
                                               time_budget=min(remaining * 0.3, 30.0))
                obj_new = compute_obj(a_new)
                if obj_new < best_score:
                    best_score = obj_new
                    best_seq = a_new.copy()
                    a_cur = a_new.copy()
                    print(f"  Final LP {final_round}: {best_score:.7f}")

            # Subgradient
            remaining = deadline - time.monotonic()
            if remaining > 10:
                a_new = subgradient_optimize_v2(a_cur, max_iters=int(remaining * 10),
                                                time_budget=min(remaining * 0.2, 12.0),
                                                p_start=50.0, p_end=200.0)
                obj_new = compute_obj(a_new)
                if obj_new < best_score:
                    best_score = obj_new
                    best_seq = a_new.copy()
                    a_cur = a_new.copy()
                    print(f"  Final SG {final_round}: {best_score:.7f}")

    # Final validation
    if best_seq is None:
        best_seq = np.ones(200)
        best_score = compute_obj(best_seq)

    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    val = evaluate_sequence(result)
    if val < np.inf:
        best_score = val
    else:
        result = [1.0] * 100
        best_score = 2.0

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_score:.7f}, time={elapsed:.1f}s ===")

    return result
