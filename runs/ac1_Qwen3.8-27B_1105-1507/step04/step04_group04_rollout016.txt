import numpy as np
import time
from scipy.optimize import linprog

def compute_conv(a):
    """Compute conv(a,a) using FFT."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    conv = np.fft.irfft(fa * fa, size)
    return conv[:2 * n - 1]

def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    conv = compute_conv(a)
    max_b = float(np.max(conv))
    sum_a = float(np.sum(a))
    if sum_a < 0.01:
        return np.inf
    return 2.0 * n * max_b / (sum_a * sum_a)

def softmax_weights(conv, tau):
    """Compute softmax weights over convolution values."""
    shifted = tau * (conv - np.max(conv))
    shifted = np.clip(shifted, -500, 500)
    w = np.exp(shifted)
    s = np.sum(w)
    if s < 1e-300:
        idx = int(np.argmax(conv))
        w = np.zeros(len(conv))
        w[idx] = 1.0
    else:
        w /= s
    return w

def smooth_gradient(a, tau):
    """Compute gradient of log-sum-exp smoothed objective."""
    n = len(a)
    conv = compute_conv(a)

def build_toeplitz(a):
    """Build Toeplitz matrix for conv(a, g)."""
    n = len(a)
    A = np.zeros((2 * n - 1, n), dtype=np.float64)
    for j in range(n):
        k_start = j
        k_end = min(j + n, 2 * n - 1)
        A[k_start:k_end, j] = a[:k_end - k_start]
    return A

def solve_lp_maxsum(a, M, time_limit=10):
    """Solve LP: max sum(g) s.t. conv(a,g)[k] <= M, g >= 0."""
    n = len(a)
    A = build_toeplitz(a)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': time_limit})
        if res.status == 0:
            g = np.maximum(res.x, 0)
            return g, float(np.sum(g))
    except Exception:
        pass
    return None, None

def alternating_lp(a, time_budget=30.0, max_iters=50):
    """Alternating LP method."""

        return a

    best_obj = compute_obj(a)

    stall = 0

        if time.monotonic() - start > time_budget - 2:
            break
        
        conv = compute_conv(a)
        M = float(np.max(conv))
        
        g, sum_g = solve_lp_maxsum(a, M, time_limit=5)
        if g is None or sum_g < 1e-10:
            break
        
        g_norm = g / sum_g
        
        # Line search
        best_t = 0.0
        best_obj_ls = best_obj
        for t in [0.1, 0.3, 0.5, 0.7, 1.0]:
            a_new = (1.0 - t) * a + t * g_norm
            a_new = np.maximum(a_new, 0)
            s_new = np.sum(a_new)
            if s_new < 1e-10:

            a_new = a_new / s_new
            obj_new = compute_obj(a_new)

                best_obj_ls = obj_new
                best_t = t

def compute_conv(a):

def compute_obj(a):

        progress = min(it / max(total_iters, 1), 1.0)
        cos_factor = 0.5 * (1 + np.cos(np.pi * progress))
        tau = tau_start + (tau_end - tau_start) * progress
        step = step_end + (step_start - step_end) * cos_factor

        # Nesterov extrapolation in log-space
        log_y = log_a + beta * (log_a - log_a_prev)
        a_y = np.exp(np.clip(log_y, -50, 50))
        a_y = np.maximum(a_y, 1e-15)
        a_y = a_y / np.sum(a_y)

        # Gradient at extrapolated point
        conv = compute_conv(a_y)
        w = softmax_weights(conv, tau)
        c = np.correlate(w, a_y, mode='full')
        grad = 2.0 * c[n - 1 : n - 1 + n]
        grad_centered = grad - np.mean(grad)

        # Multiplicative update
        log_update = -step * grad_centered
        log_update = np.clip(log_update, -3, 3)
        a_new = a * np.exp(log_update)
        a_new = np.maximum(a_new, 1e-15)
        a_new = a_new / np.sum(a_new)

        log_a_prev = log_a
        a = a_new
        log_a = np.log(np.maximum(a, 1e-30))

        obj = compute_obj(a)
        if obj < best_obj - 1e-14:
            best_obj = obj
            best_a = a.copy()

    return best_a


def build_toeplitz(a):
    """Build Toeplitz matrix where (A@g)[k] = conv(a,g)[k]."""
    n = len(a)
    A = np.zeros((2 * n - 1, n), dtype=np.float64)
    for j in range(n):
        k_start = j
        k_end = min(j + n, 2 * n - 1)
        A[k_start:k_end, j] = a[:k_end - k_start]
    return A


def solve_lp_maxsum(a, M, time_limit=10):
    """Solve LP: max sum(g) s.t. conv(a,g)[k] <= M for all k, g >= 0."""
    n = len(a)
    if HAS_CVXPY:
        try:
            g_var = cp.Variable(n, nonneg=True)
            A = build_toeplitz(a)
            constraints = [A @ g_var <= M]
            prob = cp.Problem(cp.Maximize(cp.sum(g_var)), constraints)
            for solver in [cp.CLARABEL, cp.SCS, cp.ECOS]:
                try:
                    prob.solve(solver=solver, time_limit=int(time_limit))
                    break
                except Exception:
                    continue
            if g_var.value is not None:
                g_val = np.maximum(g_var.value, 0)
                s = float(np.sum(g_val))
                if s > 1e-10:
                    return g_val, s
        except Exception:
            pass

    A = build_toeplitz(a)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': time_limit})
        if res.status == 0:
            g = np.maximum(res.x, 0)
            s = float(np.sum(g))
            if s > 1e-10:
                return g, s
    except Exception:
        pass
    return None, None


def solve_lp_minpeak(a, time_limit=10):
    """Solve LP: min M s.t. conv(a,g)[k] <= M, sum(g)=1, g>=0."""
    n = len(a)
    if HAS_CVXPY:
        try:
            g_var = cp.Variable(n, nonneg=True)
            M_var = cp.Variable()
            A = build_toeplitz(a)
            constraints = [A @ g_var <= M_var, cp.sum(g_var) == 1]
            prob = cp.Problem(cp.Minimize(M_var), constraints)
            for solver in [cp.CLARABEL, cp.SCS]:
                try:
                    prob.solve(solver=solver, time_limit=int(time_limit))
                    break
                except Exception:
                    continue
            if g_var.value is not None and M_var.value is not None:
                g_val = np.maximum(g_var.value, 0)
                s = np.sum(g_val)
                if s > 1e-10:
                    return g_val / s, float(M_var.value)
        except Exception:
            pass

    A = build_toeplitz(a)
    c = np.zeros(n + 1)
    c[-1] = 1.0
    A_ub = np.zeros((2 * n - 1, n + 1))
    A_ub[:, :n] = A
    A_ub[:, n] = -1.0
    b_ub = np.zeros(2 * n - 1)
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])
    bounds = [(0.0, None)] * n + [(0.0, None)]
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': time_limit})
        if res.status == 0:
            g = np.maximum(res.x[:n], 0)
            s = np.sum(g)
            if s > 1e-10:
                return g / s, float(res.x[n])
    except Exception:
        pass
    return None, None


def alternating_lp(a, time_budget=30.0, max_iters=40):
    """Alternating LP with line search (max-sum formulation)."""
    n = len(a)
    a = a.copy().astype(np.float64)
    s = np.sum(a)
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    stall = 0

    for it in range(max_iters):
        if time.monotonic() - start > time_budget - 2:
            break

        conv = compute_conv(a)
        M = float(np.max(conv))

        g, sum_g = solve_lp_maxsum(a, M, time_limit=6)
        if g is None or sum_g < 1e-10:
            break

        g_norm = g / sum_g

        # Line search
        best_t = 0.0
        best_obj_ls = best_obj
        for t in [0.1, 0.3, 0.5, 0.7, 1.0]:
            a_new = (1.0 - t) * a + t * g_norm
            a_new = np.maximum(a_new, 0)
            s_new = np.sum(a_new)
            if s_new < 1e-10:
                continue
            a_new = a_new / s_new
            obj_new = compute_obj(a_new)
            if obj_new < best_obj_ls - 1e-14:
                best_obj_ls = obj_new
                best_t = t

        if best_t > 0:
            a_new = (1.0 - best_t) * a + best_t * g_norm
            a_new = np.maximum(a_new, 0)
            a_new = a_new / np.sum(a_new)
            best_obj = best_obj_ls
            best_a = a_new.copy()
            a = a_new
            stall = 0
        else:
            stall += 1
            if stall > 12:
                break
            a = 0.9 * a + 0.1 * g_norm
            a = np.maximum(a, 0)
            a = a / np.sum(a)

    return best_a

def fourier_perturbation(a, rng, num_modes=3, amplitude=0.03):
    """Apply structured Fourier perturbation."""
    n = len(a)
    k = np.arange(n)
    delta = np.zeros(n)
    for _ in range(num_modes):
        freq = rng.uniform(0.5, n / 2)
        phase = rng.uniform(0, 2 * np.pi)
        amp = amplitude * rng.uniform(0.5, 1.5)
        delta += amp * np.cos(2 * np.pi * freq * k / n + phase)
    a_new = a * np.exp(delta)
    a_new = np.maximum(a_new, 1e-10)
    a_new = a_new / np.sum(a_new)
    return a_new


def coordinate_descent_polish(a, time_budget=20.0):
    """Coordinate descent: optimize each coordinate in turn."""
    n = len(a)
    a = a.copy().astype(np.float64)
    s = np.sum(a)
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    # Precompute the convolution structure for efficient updates
    # Changing a[j] affects conv[k] for k in [j, j+n-1]
    # conv[k] = sum_i a[i]*a[k-i], so d(conv[k])/d(a[j]) = a[k-j]

    iterations = 0
    while time.monotonic() - start < time_budget:
        iterations += 1
        improved = False
        conv = compute_conv(a)

        for j in range(n):
            if time.monotonic() - start > time_budget - 1:
                break

            # Try a few values for a[j]
            current_val = a[j]
            # The gradient with respect to a[j] (ignoring normalization):
            # 2 * sum_k w[k] * a[k-j] where w is peaked at max conv
            # For a simple approach, try scaling a[j]
            candidates = [current_val * 0.5, current_val * 0.8, current_val * 1.2,
                          current_val * 1.5, current_val * 0.1]
            candidates = [max(0, min(1000, c)) for c in candidates]

            best_candidate = current_val
            # Compute current max
            current_max = np.max(conv)

            for cand in candidates:
                if cand == current_val:
                    continue
                a_test = a.copy()
                a_test[j] = cand
                s_test = np.sum(a_test)
                if s_test < 1e-10:
                    continue
                # Quickly compute new max conv (only affected entries change)
                # conv[k] changes for k in [j, j+n-1]
                # New conv[k] = old conv[k] - old_val * a[k-j] + new_val * a[k-j]
                # = old conv[k] + (new_val - old_val) * a[k-j]
                diff = cand - current_val
                for k in range(j, min(j + n, 2 * n - 1)):
                    idx = k - j
                    if 0 <= idx < n:
                        conv[k] += diff * a[idx]
                new_max = np.max(conv)
                if new_max < current_max - 1e-12:
                    best_candidate = cand
                    current_max = new_max
                else:
                    # Revert
                    for k in range(j, min(j + n, 2 * n - 1)):
                        idx = k - j
                        if 0 <= idx < n:
                            conv[k] -= diff * a[idx]

            if best_candidate != current_val:
                a[j] = best_candidate
                improved = True
                obj = compute_obj(a)
                if obj < best_obj:
                    best_obj = obj
                    best_a = a.copy()
                # Re-normalize occasionally
                if iterations % n == 0:
                    a = a / np.sum(a)
                    conv = compute_conv(a)

        if not improved:
            break

    return best_a


def local_search(a, time_budget=60.0, verbose=False):
    """Local search with Fourier perturbations and quick re-optimization."""
    n = len(a)
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    attempt = 0

    if verbose:
        print(f"    Local search start: {best_obj:.8f}")

    while time.monotonic() - start < time_budget - 5:
        attempt += 1
        remaining = time_budget - (time.monotonic() - start)
        if remaining < 8:
            break

        rng = np.random.default_rng(attempt * 7919 + 13 + seed)

        # Try different perturbation strategies
        strategies = [
            ('fourier_small', 0.01, 3),
            ('fourier_med', 0.03, 3),
            ('fourier_large', 0.06, 5),
            ('gaussian', 0.02, 0),
        ]

        for strat_name, amp, n_modes in strategies:
            if time.monotonic() - start > time_budget - 3:
                break

            if strat_name == 'gaussian':
                a_pert = best_a * np.exp(rng.normal(0, amp, n))
                a_pert = np.maximum(a_pert, 1e-10)
                a_pert = a_pert / np.sum(a_pert)
            else:
                a_pert = fourier_perturbation(best_a, rng, num_modes=n_modes, amplitude=amp)

            # Quick optimization
            sub_time = min(8.0, remaining * 0.25)
            a_opt = multiplicative_subgradient(a_pert, time_budget=sub_time,
                                               tau=30, step=0.04)
            obj = compute_obj(a_opt)
            if obj < best_obj - 1e-14:
                best_obj = obj
                best_a = a_opt.copy()
                if verbose:
                    print(f"    LS improvement: {best_obj:.8f} (attempt {attempt}, {strat_name})")

    return best_a, best_obj


def optimize_single(a, time_budget=60.0, verbose=False):
    """Full optimization pipeline for a single sequence."""
    n = len(a)
    a = a.copy().astype(np.float64)
    s = np.sum(a)
    if s < 1e-10:
        a = np.ones(n) / n
        s = 1
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    if verbose:
        print(f"  Initial: {best_obj:.8f}")

    # Phase 1: L-BFGS-B on smoothed objective
    remaining = time_budget - (time.monotonic() - start)
    if remaining > 10:
        t1 = min(15.0, remaining * 0.25)
        a_new = lbfgs_optimize(best_a, time_budget=t1, tau=30)
        obj = compute_obj(a_new)
        if obj < best_obj - 1e-14:
            best_obj = obj
            best_a = a_new.copy()
            if verbose:
                print(f"  After L-BFGS-B: {best_obj:.8f}")

    # Phase 2: Nesterov multiplicative subgradient
    remaining = time_budget - (time.monotonic() - start)
    if remaining > 8:
        t2 = min(12.0, remaining * 0.25)
        a_new = nesterov_subgradient(best_a, time_budget=t2,
                                     tau_start=5.0, tau_end=80.0,
                                     step_start=0.12, step_end=0.008,
                                     beta=0.85)
        obj = compute_obj(a_new)
        if obj < best_obj - 1e-14:
            best_obj = obj
            best_a = a_new.copy()
            if verbose:
                print(f"  After Nesterov: {best_obj:.8f}")

    # Phase 3: Alternating LP
    remaining = time_budget - (time.monotonic() - start)
    if remaining > 12:
        t3 = min(20.0, remaining * 0.35)
        a_new = alternating_lp(best_a, time_budget=t3)
        obj = compute_obj(a_new)
        if obj < best_obj - 1e-14:
            best_obj = obj
            best_a = a_new.copy()
            if verbose:
                print(f"  After LP: {best_obj:.8f}")

    # Phase 4: Min-peak LP
    remaining = time_budget - (time.monotonic() - start)
    if remaining > 10:
        t4 = min(15.0, remaining * 0.25)
        a_new = alternating_lp_minpeak(best_a, time_budget=t4)
        obj = compute_obj(a_new)
        if obj < best_obj - 1e-14:
            best_obj = obj
            best_a = a_new.copy()
            if verbose:
                print(f"  After MinPeak LP: {best_obj:.8f}")

    # Phase 5: Fine-tuning subgradient
    remaining = time_budget - (time.monotonic() - start)
    if remaining > 5:
        t5 = min(8.0, remaining * 0.3)
        a_new = multiplicative_subgradient(best_a, time_budget=t5,
                                           tau=80, step=0.015)
        obj = compute_obj(a_new)
        if obj < best_obj - 1e-14:
            best_obj = obj
            best_a = a_new.copy()
            if verbose:
                print(f"  After fine-tune: {best_obj:.8f}")

    # Phase 6: Coordinate descent polish
    remaining = time_budget - (time.monotonic() - start)
    if remaining > 5:
        t6 = min(10.0, remaining * 0.4)
        a_new = coordinate_descent_polish(best_a, time_budget=t6)
        obj = compute_obj(a_new)
        if obj < best_obj - 1e-14:
            best_obj = obj
            best_a = a_new.copy()
            if verbose:
                print(f"  After coord descent: {best_obj:.8f}")

    return best_a, best_obj


def resample_sequence(a, new_n):
    """Resample a sequence to a new length."""
    n = len(a)
    if n == new_n:
        return a.copy()
    old_k = np.linspace(0, 1, n)
    new_k = np.linspace(0, 1, new_n)
    a_new = np.interp(new_k, old_k, a)
    a_new = np.maximum(a_new, 0)
    s = np.sum(a_new)
    if s > 1e-10:
        a_new = a_new / s
    return a_new


def random_init(n, seed):
    """Generate diverse random initialization."""
    rng = np.random.default_rng(seed)
    k = np.arange(n)
    choice = rng.integers(0, 8)

    if choice == 0:
        f = rng.uniform(0.5, 3.0)
        phase = rng.uniform(0, 2 * np.pi)
        a = 1.0 + 0.8 * np.cos(2 * np.pi * f * k / n + phase)
    elif choice == 1:
        f1, f2 = rng.uniform(0.3, 2.0), rng.uniform(2.0, 5.0)
        a = 1.0 + 0.5 * np.cos(2 * np.pi * f1 * k / n) + 0.3 * np.cos(2 * np.pi * f2 * k / n)
    elif choice == 2:
        a = np.ones(n)
        for _ in range(rng.integers(2, 5)):
            f = rng.uniform(0.3, 6.0)
            amp = rng.uniform(0.1, 0.4)
            a += amp * np.cos(2 * np.pi * f * k / n + rng.uniform(0, 2 * np.pi))
    elif choice == 3:
        a = rng.random(n)
        win = max(3, n // 15)
        kernel = np.ones(win) / win
        a = np.convolve(a, kernel, mode='same')
    elif choice == 4:
        a = np.ones(n)
        for scale in range(4):
            f = rng.uniform(0.3 * (2 ** scale), 2.0 * (2 ** scale))
            amp = 0.25 / (scale + 1)
            a += amp * np.cos(2 * np.pi * f * k / n + rng.uniform(0, 2 * np.pi))
    elif choice == 5:
        center = n * rng.uniform(0.3, 0.7)
        width = n * rng.uniform(0.1, 0.4)
        a = 1.0 + 0.6 * np.exp(-0.5 * ((k - center) / width) ** 2)
    elif choice == 6:
        # Multi-frequency with random amplitudes
        a = np.ones(n)
        for _ in range(5):
            f = rng.uniform(0.2, 8.0)
            amp = rng.uniform(0.05, 0.2)
            a += amp * np.sin(2 * np.pi * f * k / n + rng.uniform(0, 2 * np.pi))
    else:
        a = np.ones(n) * rng.uniform(0.5, 2.0)
        a += 0.2 * rng.normal(0, 1, n)

    a = np.maximum(a, 0.001)
    s = np.sum(a)
    if s < 1e-10:
        a = np.ones(n)
        s = n
    return a / s


def try_parent():
    """Try to load and validate parent sequence."""
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0, 1000.0)
        if p.size > 0 and np.isfinite(p).all():
            obj = compute_obj(p)
            if obj < np.inf:
                print(f"Parent loaded: n={p.size}, obj={obj:.8f}")
                return p.copy(), obj
    except Exception:
        pass
    return None, np.inf


def propose_candidate(seed=42, budget_s=1500):
    """
    Step function optimization using multi-scale smoothed optimization:
    - L-BFGS-B on log-sum-exp objective with increasing tau
    - Nesterov multiplicative subgradient with cosine annealing
    - Alternating LP (both max-sum and min-peak formulations)
    - Coordinate descent polish
    - Fourier-mode structured local search
    - Systematic n-value exploration around the parent
    """
    global seed
    start_time = time.monotonic()
    deadline = start_time + float(budget_s) - 15.0

    best_seq = None
    best_score = np.inf

    # Load parent
    parent, parent_obj = try_parent()
    if parent is not None:
        best_score = parent_obj
        best_seq = parent.copy()
        n_parent = len(parent)
    else:
        n_parent = 300
        best_seq = np.ones(n_parent) / n_parent
        best_score = compute_obj(best_seq)
        print(f"No parent found, starting from ones(n={n_parent}): {best_score:.6f}")

    print(f"Initial best: {best_score:.8f} (n={n_parent})")
    print(f"Time budget: {budget_s}s, deadline in {budget_s - 15:.0f}s")

    # Phase 1: Deep optimization of the best initial sequence
    print("\n=== Phase 1: Deep optimization of initial sequence ===")
    remaining = deadline - time.monotonic()
    if remaining > 40:
        p1_budget = min(remaining * 0.3, 250.0)
        print(f"  Budget: {p1_budget:.0f}s")
        a_opt, obj = optimize_single(best_seq, time_budget=p1_budget, verbose=True)
        if obj < best_score:
            best_score = obj
            best_seq = a_opt.copy()
            print(f"  Phase 1 result: {best_score:.8f}")

    # Phase 2: Intensive local search
    if best_seq is not None:
        remaining = deadline - time.monotonic()
        if remaining > 30:
            print(f"\n=== Phase 2: Intensive local search ({remaining:.0f}s) ===")
            p2_budget = min(remaining * 0.2, 150.0)
            a_opt, obj = local_search(best_seq, time_budget=p2_budget, verbose=True)
            if obj < best_score:
                best_score = obj
                best_seq = a_opt.copy()
                print(f"  Phase 2 result: {best_score:.8f}")

    # Phase 3: Try nearby n values
    print("\n=== Phase 3: Nearby n values ===")
    n_candidates = sorted(set([
        n_parent, n_parent - 5, n_parent + 5,
        n_parent - 10, n_parent + 10, n_parent - 20, n_parent + 20,
        n_parent - 30, n_parent + 30, n_parent - 40, n_parent + 40,
        270, 280, 310, 320, 330, 340, 350, 360, 370, 380, 390, 400
    ]))
    n_candidates = [n for n in n_candidates if 200 <= n <= 600 and n != n_parent]

    remaining = deadline - time.monotonic()
    if remaining > 40:
        per_n = max(15.0, min(60.0, remaining / max(len(n_candidates), 1) * 1.3))
        for n_val in n_candidates:
            if time.monotonic() > deadline - 25:
                break
            remaining = deadline - time.monotonic()
            if remaining < 20:
                break
            alloc = min(per_n, remaining * 0.4, 70.0)
            alloc = max(alloc, 12.0)

            # Initialize from resampled parent
            if parent is not None:
                a_init = resample_sequence(parent, n_val)
            else:
                a_init = random_init(n_val, seed + n_val * 104729)

            print(f"  n={n_val} (alloc={alloc:.0f}s, remaining={remaining:.0f}s)")
            a_opt, obj = optimize_single(a_init, time_budget=alloc,
                                          verbose=(n_val == n_parent + 10))
            if obj < best_score:
                best_score = obj
                best_seq = a_opt.copy()
                print(f"  ** UPDATE: n={n_val}, obj={best_score:.8f} **")

    # Phase 4: Final deep optimization
    if best_seq is not None:
        remaining = deadline - time.monotonic()
        if remaining > 30:
            print(f"\n=== Phase 4: Final deep optimization ({remaining:.0f}s) ===")
            p4_budget = min(remaining * 0.5, remaining - 10.0)
            a_opt, obj = optimize_single(best_seq, time_budget=p4_budget, verbose=True)
            if obj < best_score:
                best_score = obj
                best_seq = a_opt.copy()
                print(f"  Phase 4 result: {best_score:.8f}")

            # Local search again
            remaining = deadline - time.monotonic()
            if remaining > 20:
                p4b_budget = min(remaining * 0.5, remaining - 5.0)
                a_opt, obj = local_search(best_seq, time_budget=p4b_budget, verbose=True)
                if obj < best_score:
                    best_score = obj
                    best_seq = a_opt.copy()
                    print(f"  Phase 4 LS: {best_score:.8f}")

    # Phase 5: One more shot at different n if time allows
    if best_seq is not None:
        remaining = deadline - time.monotonic()
        if remaining > 50:
            print(f"\n=== Phase 5: Extended n search ({remaining:.0f}s) ===")
            extra_ns = [310, 320, 330, 340, 350, 360, 380, 400, 420, 450]
            extra_ns = [n for n in extra_ns if n != len(best_seq)]
            for n_val in extra_ns[:5]:
                if time.monotonic() > deadline - 20:
                    break
                remaining = deadline - time.monotonic()
                if remaining < 25:
                    break
                alloc = min(35.0, remaining * 0.4)
                a_init = resample_sequence(best_seq, n_val) if best_seq is not None else random_init(n_val, seed + n_val)
                a_opt, obj = optimize_single(a_init, time_budget=alloc, verbose=False)
                if obj < best_score:
                    best_score = obj
                    best_seq = a_opt.copy()
                    print(f"  ** UPDATE: n={n_val}, obj={best_score:.8f} **")

    # Final validation
    if best_seq is None:
        best_seq = np.ones(100)
        best_score = compute_obj(best_seq)

    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    val = evaluate_sequence(result)
    final_score = min(best_score, val)
    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={final_score:.8f}, time={elapsed:.1f}s ===")

    return result
