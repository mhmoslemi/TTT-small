**New idea: Use the dual of the LP more directly**

The LP is: max 1^T g s.t. A g <= M*1, g >= 0
Dual: min M * 1^T y s.t. A^T y >= 1, y >= 0

The dual has 2n-1 variables (y) and n constraints. For n=640, that's 1279 variables and 640 constraints.

The complementary slackness conditions give us:
- g_j > 0 implies (A^T y)_j = 1
- y_k > 0 implies (A g)_k = M

So at the optimum, the active set of constraints (where (A g)_k = M) and the active set of variables (where g_j > 0) are complementary.

We could try an iterative scheme:
1. Start with a guess for the active set S (which k's have (A g)_k = M)
2. Solve the reduced problem with only those constraints
3. Update the active set
4. Repeat

This is essentially the active set method for LPs, which might be faster than a general LP solver.

**Even simpler: Use the KKT conditions directly**

At the optimum of the mirror descent, we have:
- g is the best response: max 1^T g s.t. A g <= M*1, g >= 0
- a is a convex combination of the previous a and g

At a fixed point, a = g (the best response of a to itself).

So we're looking for a fixed point of: a = best_response(a, M(a))
where M(a) = max_k (a*a)[k].

This is: find a >= 0 such that:
- a = argmax 1^T g s.t. A(a) g <= M(a)*1, g >= 0

By KKT: there exist y >= 0 (dual variables) such that:
- A(a)^T y = 1 (for all j where a_j > 0)
- A(a)^T y >= 1 (for all j where a_j = 0)
- (A(a) a)_k = M(a) for all k where y_k > 0
- (A(a) a)_k <= M(a) for all k where y_k = 0

This gives us a system of equations that we could try to solve iteratively.

"""
Optimized step function search using LP mirror descent with multiple initializations,
adaptive step sizes, multi-length exploration, coordinate descent polishing, and
final gradient refinement. Designed to minimize J(a) = 2n*max(conv(a,a))/(sum(a))^2.

import numpy as np
from scipy.optimize import linprog, minimize
from scipy.sparse import csr_matrix
import time


def fast_conv(a):
    """Full self-convolution via FFT."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]


def compute_J(a):
    """Compute J(a) = 2*n*max(conv(a,a))/(sum(a))^2."""
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))


def build_conv_structure(n):
    """Precompute sparse structure for convolution matrix."""
    row_indices = []
    col_indices = []
    offsets = []
    for k in range(2 * n - 1):
        j_min = max(0, k - n + 1)
        j_max = min(n - 1, k)
        count = j_max - j_min + 1
        row_indices.extend([k] * count)
        col_indices.extend(range(j_min, j_max + 1))
        offsets.extend([k - j for j in range(j_min, j_max + 1)])
    return np.array(row_indices), np.array(col_indices), np.array(offsets)


def make_sparse_conv(n, a, structure):
    """Build sparse convolution matrix with current a values."""
    row_indices, col_indices, offsets = structure
    vals = a[offsets]
    return csr_matrix((vals, (row_indices, col_indices)), shape=(2 * n - 1, n))


def lp_mirror_descent(a, structure, max_iters=5000, deadline=None, verbose=False):
    """LP mirror descent with adaptive step size."""
    n = len(a)
    a = np.maximum(a.astype(np.float64), 1e-10)
    a = a / a.sum()
    current_J = compute_J(a)
    best_J = current_J
    best_a = a.copy()

    c = -np.ones(n)
    bounds = [(0, None)] * n
    no_improve = 0
    step = 0.4
    min_step = 0.02

    for it in range(max_iters):
        if deadline is not None and time.monotonic() > deadline:
            break

        b = fast_conv(a)
        M = float(b.max())

        A = make_sparse_conv(n, a, structure)
        b_ub = np.full(2 * n - 1, M)

        try:
            r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                        options={'presolve': True, 'time_limit': 5.0})
            if not r.success or r.x is None:
                if verbose:
                    print(f"  LP failed at iter {it}: {r.message}")
                break
            g = np.maximum(r.x, 0)
        except Exception:
            break

        g_sum = g.sum()
        if g_sum < 1e-10:
            break
        g = g / g_sum

        # Check if g is very similar to a (convergence)
        if np.linalg.norm(g - a, 1) < 1e-8:
            if verbose:
                print(f"  Converged at iter {it}")
            break

        # Line search with geometric backtracking
        found = False
        trial_step = min(step, 0.8)
        for ls in range(30):
            a_trial = (1.0 - trial_step) * a + trial_step * g
            a_trial = np.maximum(a_trial, 0.0)

            if s < 1e-10:

                continue

            J_trial = compute_J(a_trial)

                current_J = J_trial

def coordinate_descent_polish(a, max_sweeps=50, deadline=None):
    """Coordinate descent: optimize each element in turn."""
    n = len(a)
    a = np.maximum(a.astype(np.float64), 1e-10)
    a = a / a.sum()
    best_J = compute_J(a)
    best_a = a.copy()
    S = a.sum()

    for sweep in range(max_sweeps):
        if deadline is not None and time.monotonic() > deadline:
            break

        improved = False
        for j in range(n):
            old_val = a[j]
            S_prime = S - old_val

            # Compute b_k = (a*a)[k] - a[j]*a[k-j] for all k
            # (a*a)[k] = sum_i a[i]*a[k-i]
            # Terms with i=j: a[j]*a[k-j]
            # Terms with k-i=j i.e. i=k-j: a[k-j]*a[j] (same if k-j==j)
            # If k-j != j: two terms, each a[j]*a[k-j], total contribution 2*a[j]*a[k-j]
            # If k-j == j (k==2j): one term a[j]^2, contribution a[j]^2

            # Compute full conv and subtract
            b_full = fast_conv(a)
            c_k = np.zeros(2 * n - 1)
            # a[j] contributes to (a*a)[k] via a[j]*a[k-j] and a[k-j]*a[j]
            for k in range(2 * n - 1):
                m = k - j
                if 0 <= m < n:
                    if m == j:  # k = 2j
                        c_k[k] = a[j] ** 2
                    else:
                        c_k[k] = 2.0 * a[j] * a[m]

            b_base = b_full - c_k

            # Now optimize a[j]: J(a_j) = 2n * max_k(b_base[k] + a_j * a[k-j]) / (S_prime + a_j)^2
            # Let c_k[j] = a[k-j] (the coefficient of a_j in (a*a)[k])
            # (a*a)[k] = b_base[k] + a_j * a[k-j] (for k-j in [0,n-1])
            #           = b_base[k] (for k-j not in [0,n-1])

            # The max is over k of (b_base[k] + a_j * d[k]) where d[k] = a[k-j] if valid else 0
            d = np.zeros(2 * n - 1)
            for k in range(2 * n - 1):
                m = k - j
                if 0 <= m < n:
                    d[k] = a[m]

            # Find optimal a_j by ternary search or grid
            # J(a_j) = 2n * max_k(b_base[k] + a_j * d[k]) / (S_prime + a_j)^2
            # This is a somewhat complex 1D function. Use golden section search.
            lo = 0.0
            hi = 3.0 * old_val if old_val > 0 else 3.0 * S / n

            if hi < lo:
                hi = lo + 1e-3

            # Golden section
            gr = (np.sqrt(5.0) - 1.0) / 2.0
            c1 = hi - gr * (hi - lo)
            c2 = lo + gr * (hi - lo)

            def obj_cv(x):
                conv_val = b_base + x * d
                mx = conv_val.max()
                ss = S_prime + x
                if ss < 1e-10:
                    return 1e10
                return mx / (ss * ss)

            f1 = obj_cv(c1)
            f2 = obj_cv(c2)
            for _ in range(40):
                if f1 < f2:
                    hi = c2
                    c2 = c1
                    f2 = f1
                    c1 = hi - gr * (hi - lo)
                    f1 = obj_cv(c1)
                else:
                    lo = c1
                    c1 = c2
                    f1 = f2
                    c2 = lo + gr * (hi - lo)
                    f2 = obj_cv(c2)

            best_x = (lo + hi) / 2.0
            a[j] = best_x
            S = S_prime + best_x
            improved = True

        J_now = compute_J(a)
        if J_now < best_J:
            best_J = J_now
            best_a = a.copy()

        if not improved:
            break

    return best_a


def window_hann(n):
    """Hann window."""
    t = np.arange(n) / n
    return 0.5 * (1 + np.cos(2 * np.pi * t))


def window_blackman(n):
    """Blackman window."""
    t = np.arange(n) / n
    return 0.42 - 0.5 * np.cos(2 * np.pi * t) + 0.08 * np.cos(4 * np.pi * t)


def window_gaussian(n, sigma_frac=0.25):
    """Gaussian window."""
    t = (np.arange(n) - n / 2.0) / n
    sigma = sigma_frac
    return np.exp(-0.5 * (t / sigma) ** 2)


def window_flat_top(n):
    """Approximate flat-top window."""
    t = np.arange(n) / n
    return (0.21557895 - 0.41663158 * np.cos(2 * np.pi * t)
            + 0.277263158 * np.cos(4 * np.pi * t)
            - 0.083578947 * np.cos(6 * np.pi * t)
            + 0.006947368 * np.cos(8 * np.pi * t))


def window_kaiser_like(n, beta=3.0):
    """Approximate Kaiser using a smooth bump."""
    t = np.arange(n) / n
    # Use a polynomial approximation
    x = 2 * t - 1
    return np.maximum(1 - beta * x ** 2 + (beta - 1) * x ** 4, 0)


def generate_initializations(n, parent=None, rng=None):
    """Generate diverse initial sequences of length n."""
    inits = []
    t = np.arange(n, dtype=np.float64) / n

    # Standard windows
    inits.append(window_hann(n))
    inits.append(window_blackman(n))
    inits.append(window_gaussian(n, 0.2))
    inits.append(window_gaussian(n, 0.3))
    inits.append(window_gaussian(n, 0.4))
    inits.append(window_flat_top(n))
    inits.append(window_kaiser_like(n, 2.0))
    inits.append(window_kaiser_like(n, 4.0))
    inits.append(window_kaiser_like(n, 6.0))

    # Raised cosine with different exponents
    for p in [1.5, 2.0, 2.5, 3.0, 4.0]:
        base = 0.5 * (1 + np.cos(2 * np.pi * t))
        inits.append(np.maximum(base ** p, 1e-6))

    # Power-law taper
    for alpha in [0.5, 1.0, 1.5, 2.0]:
        x = np.abs(2 * t - 1)
        inits.append(np.maximum((1 - x) ** alpha, 1e-6))

    # Bell shapes
    for mu in [0.3, 0.5, 0.7]:
        for sig in [0.1, 0.15, 0.2, 0.25]:
            inits.append(np.exp(-0.5 * ((t - mu) / sig) ** 2) + 1e-4)

    # Bimodal
    inits.append(np.exp(-8 * (t - 0.3) ** 2) + np.exp(-8 * (t - 0.7) ** 2) + 0.05)
    inits.append(np.exp(-4 * (t - 0.25) ** 2) + np.exp(-4 * (t - 0.75) ** 2) + 0.1)

    # Parent (resampled)
    if parent is not None:
        from scipy.interpolate import interp1d
        t_old = np.linspace(0, 1, parent.size)
        t_new = np.linspace(0, 1, n)
        f = interp1d(t_old, parent, kind='cubic', fill_value='extrapolate')
        p_res = np.maximum(f(t_new), 1e-6)
        inits.append(p_res)

        if rng is not None:
            inits.append(np.maximum(p_res + rng.normal(0, 0.02, n), 1e-6))

    # Random smooth
    if rng is not None:
        k = max(3, n // 16)
        kernel = np.ones(k) / k
        smooth = np.convolve(rng.uniform(0, 1, n), kernel, mode='same')
        inits.append(np.maximum(smooth + 0.1, 1e-6))

    # Normalize all
    result = []
    for a in inits:
        a = np.asarray(a, dtype=np.float64)
        a = np.maximum(a, 1e-6)
        s = a.sum()
        if s > 1e-6:
            result.append(a / s)
    return result


def resample_seq(src, dest_n):
    """Resample 1D sequence to new length."""
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    return np.maximum(np.interp(t_new, t_old, src), 1e-10)


def structured_perturb(a, rng, ptype):
    """Apply a structured perturbation."""
    n = len(a)
    t = np.arange(n, dtype=np.float64) / n

    if ptype == 'shift':
        s = int(rng.integers(1, max(2, n // 8)))
        a_p = np.roll(a, s)
    elif ptype == 'cosine':
        freq = rng.uniform(1, 10)
        amp = rng.uniform(0.02, 0.15)
        phase = rng.uniform(0, 2 * np.pi)
        a_p = a * (1 + amp * np.cos(2 * np.pi * freq * t + phase))
    elif ptype == 'smooth':
        k = int(rng.integers(3, max(4, n // 8)))
        kernel = np.ones(k) / k
        a_p = np.convolve(a, kernel, mode='same')
    elif ptype == 'taper':
        alpha = rng.uniform(0.6, 0.95)
        window = np.ones(n)
        edge = max(1, int(n * (1 - alpha) / 2))
        for i in range(edge):
            frac = i / max(1, edge)
            window[i] = frac
            window[n - 1 - i] = frac
        a_p = a * window
    elif ptype == 'amplitude_mod':
        a_p = a + rng.normal(0, 0.02, n)
    else:
        a_p = a.copy()

    a_p = np.maximum(a_p, 1e-6)
    s = a_p.sum()
    if s > 1e-10:
        a_p = a_p / s
    return a_p


def lbb_objective(x, n):
    """Objective for L-BFGS-B: max of self-convolution (scale invariant)."""
    ax = np.maximum(x, 0.0)
    s = ax.sum()
    if s < 1e-8:
        return 1e10
    b = fast_conv(ax)
    return float(b.max() / (s * s) * n)


def propose_candidate(seed=42, budget_s=1500, **kwargs):
    """
    Hybrid optimizer: diverse initializations + LP mirror descent (linear & multiplicative),
    coordinate descent polish, multi-length search, perturbation-based diversification,
    and L-BFGS-B final refinement.
    """
    rng = np.random.default_rng(seed)
    t_start = time.monotonic()
    deadline = t_start + float(budget_s) - 10.0

    best_score = float('inf')
    best_seq = None

    def try_update(a):
        nonlocal best_score, best_seq
        a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
        if a.size == 0 or a.sum() < 0.01:
            return
        score = float(evaluate_sequence(a.tolist()))
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_seq = a.copy()
            elapsed = time.monotonic() - t_start
            print(f"  [NEW BEST] J={score:.8f} n={a.size} t={elapsed:.1f}s", flush=True)

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        if p.ndim == 1 and p.size > 0 and np.all(np.isfinite(p)):
            parent = np.maximum(p, 0.0)
            try_update(parent)
            print(f"Parent loaded: n={parent.size}, J={best_score:.6f}", flush=True)
    except (NameError, AttributeError):
        pass

    if parent is None:
        parent = window_hann(512)
        try_update(parent)

    parent_n = parent.size
    current_best = parent.copy()
    current_best_n = parent_n

    # === Phase 1: Multi-initialization quick screen ===
    print(f"\n{'='*60}", flush=True)
    print(f"Phase 1: Multi-init screen (parent n={parent_n})", flush=True)
    print(f"{'='*60}", flush=True)

    # Try several lengths
    lengths = []
    for n_try in [256, 384, 512, 640, 768, 896, 1024, 1280]:
        if n_try != parent_n:
            lengths.append(n_try)
    lengths.append(parent_n)
    lengths = sorted(set(lengths))

    # Time budget per length for quick screen
    screen_time = min(30.0, (deadline - time.monotonic() - 600.0) / max(1, len(lengths)))

    for n_try in lengths:
        if time.monotonic() > deadline - 500:
            break
        struct = build_conv_structure(n_try)
        inits = generate_initializations(n_try, parent, rng)

        # Quick-evaluate initializations (just compute J, no optimization)
        best_init_J = float('inf')
        best_init = None
        for a_init in inits:
            J_init = compute_J(a_init)
            if J_init < best_init_J:
                best_init_J = J_init
                best_init = a_init.copy()

        if best_init is not None:
            # Do a short LP mirror descent
            dl = time.monotonic() + max(5.0, screen_time)
            a_opt, J_opt = lp_mirror_descent(best_init, struct, max_iters=200, deadline=dl)
            try_update(a_opt)
            if J_opt < compute_J(current_best if current_best.size == n_try else resample_seq(current_best, n_try)):
                current_best = a_opt.copy()
                current_best_n = n_try
            print(f"  n={n_try}: best J={J_opt:.8f}", flush=True)

    # === Phase 2: Deep LP mirror descent on best ===
    print(f"\n{'='*60}", flush=True)
    print(f"Phase 2: Deep LP (n={current_best_n})", flush=True)
    print(f"{'='*60}", flush=True)

    n_cur = current_best_n
    struct = build_conv_structure(n_cur)
    a_work = np.maximum(current_best.copy(), 1e-10)
    a_work = a_work / a_work.sum()

    # Linear mode
    dl2 = min(deadline - 300, time.monotonic() + 400)
    t0 = time.monotonic()
    a_lin, J_lin = lp_mirror_descent(a_work, struct, max_iters=10000, deadline=dl2, verbose=True)
    try_update(a_lin)
    print(f"  Linear LP: J={J_lin:.8f} ({time.monotonic()-t0:.1f}s)", flush=True)
    if J_lin < compute_J(current_best):
        current_best = a_lin.copy()
        current_best_n = a_lin.size

    # Multiplicative mode from the result
    if time.monotonic() < deadline - 250:
        t0 = time.monotonic()
        a_work2 = np.maximum(a_lin.copy(), 1e-10)
        a_work2 = a_work2 / a_work2.sum()
        dl3 = min(deadline - 200, time.monotonic() + 200)
        a_mult, J_mult = lp_mirror_descent_multiplicative(a_work2, struct, max_iters=5000, deadline=dl3)
        try_update(a_mult)
        print(f"  Multiplicative LP: J={J_mult:.8f} ({time.monotonic()-t0:.1f}s)", flush=True)
        if J_mult < compute_J(current_best):
            current_best = a_mult.copy()
            current_best_n = a_mult.size

    # === Phase 3: Perturbation and re-optimization ===
    print(f"\n{'='*60}", flush=True)
    print(f"Phase 3: Perturbation diversification (n={current_best_n})", flush=True)
    print(f"{'='*60}", flush=True)

    n_cur = current_best_n
    struct = build_conv_structure(n_cur)
    perturb_types = ['shift', 'cosine', 'smooth', 'taper', 'amplitude_mod',
                     'shift', 'cosine', 'smooth', 'taper', 'amplitude_mod']

    remaining = deadline - time.monotonic() - 200
    per_perturb = max(10.0, remaining / len(perturb_types) / 1.5)

    for i, ptype in enumerate(perturb_types):
        if time.monotonic() > deadline - 180:
            break
        a_pert = structured_perturb(current_best, rng, ptype)
        dl_p = time.monotonic() + per_perturb
        a_opt, J_opt = lp_mirror_descent(a_pert, struct, max_iters=500, deadline=dl_p)
        try_update(a_opt)
        if J_opt < compute_J(current_best):
            current_best = a_opt.copy()
            current_best_n = a_opt.size
            print(f"  [{ptype}] improved: J={J_opt:.8f}", flush=True)

    # === Phase 4: Multi-length refinement ===
    print(f"\n{'='*60}", flush=True)
    print(f"Phase 4: Multi-length refinement (base n={current_best_n})", flush=True)
    print(f"{'='*60}", flush=True)

    n_cur = current_best_n
    nearby_lengths = []
    for mult in [0.75, 1.0, 1.25, 1.5, 2.0]:
        n_try = int(round(n_cur * mult / 64)) * 64
        n_try = max(128, min(2048, n_try))
        if n_try not in nearby_lengths and n_try != n_cur:
            nearby_lengths.append(n_try)
    nearby_lengths.sort()

    for n_try in nearby_lengths:
        if time.monotonic() > deadline - 120:
            break
        a_res = resample_seq(current_best, n_try)
        struct_n = build_conv_structure(n_try)
        per_len = max(15.0, min(60.0, (deadline - 120 - time.monotonic()) / max(1, len(nearby_lengths))))
        a_opt, J_opt = lp_mirror_descent(a_res, struct_n, max_iters=1000, deadline=time.monotonic() + per_len)
        try_update(a_opt)
        print(f"  n={n_try}: J={J_opt:.8f}", flush=True)
        if J_opt < compute_J(current_best):
            current_best = a_opt.copy()
            current_best_n = n_try

    # === Phase 5: Long LP on the best ===
    print(f"\n{'='*60}", flush=True)
    print(f"Phase 5: Long LP refinement (n={current_best_n})", flush=True)
    print(f"{'='*60}", flush=True)

    n_cur = current_best_n
    struct = build_conv_structure(n_cur)
    a_work = np.maximum(current_best.copy(), 1e-10)
    a_work = a_work / a_work.sum()

    t0 = time.monotonic()
    a_long, J_long = lp_mirror_descent(a_work, struct, max_iters=20000, deadline=deadline - 40)
    try_update(a_long)
    print(f"  Long LP: J={J_long:.8f} ({time.monotonic()-t0:.1f}s)", flush=True)
    if J_long < compute_J(current_best):
        current_best = a_long.copy()
        current_best_n = a_long.size

    # === Phase 6: Coordinate descent polish ===
    if time.monotonic() < deadline - 20:
        print(f"\n{'='*60}", flush=True)
        print(f"Phase 6: Coordinate descent polish (n={current_best_n})", flush=True)
        print(f"{'='*60}", flush=True)
        t0 = time.monotonic()
        a_cd = coordinate_descent_polish(current_best, max_sweeps=20, deadline=deadline - 15)
        try_update(a_cd)
        J_cd = compute_J(a_cd)
        print(f"  Coord descent: J={J_cd:.8f} ({time.monotonic()-t0:.1f}s)", flush=True)
        if J_cd < compute_J(current_best):
            current_best = a_cd.copy()
            current_best_n = a_cd.size

    # === Phase 7: L-BFGS-B final ===
    if time.monotonic() < deadline - 5 and best_seq is not None:
        print(f"\n{'='*60}", flush=True)
        print(f"Phase 7: L-BFGS-B final polish (n={best_seq.size})", flush=True)
        print(f"{'='*60}", flush=True)
        t0 = time.monotonic()
        n_lb = best_seq.size
        try:
            res = minimize(lbb_objective, best_seq.astype(np.float64),
                           args=(n_lb,),
                           method='L-BFGS-B', bounds=[(0.0, None)] * n_lb,
                           options={'maxiter': 200, 'max_fun': 50000, 'ftol': 1e-16, 'gtol': 1e-14})
            if res.fun < 1e10:
                try_update(res.x)
                J_lb = compute_J(res.x)
                print(f"  L-BFGS-B: J={J_lb:.8f} ({time.monotonic()-t0:.1f}s)", flush=True)
        except Exception as e:
            print(f"  L-BFGS-B error: {e}", flush=True)

    # Final
    print(f"\n{'='*60}", flush=True)
    print(f"FINAL: J={best_score:.8f} n={best_seq.size if best_seq is not None else 'N/A'} "
          f"time={time.monotonic()-t_start:.1f}s", flush=True)
    print(f"{'='*60}", flush=True)

    if best_seq is not None:
        return [float(x) for x in best_seq]
    return [float(x) for x in current_best]
