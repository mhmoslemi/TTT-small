            size <<= 1
        fp = np.fft.rfft(p, size)
        fa = np.fft.rfft(a, size)
        corr = np.fft.irfft(fp * np.conj(fa), size)
        grad = 2.0 * corr[:n]
        
        v = momentum * v + (1.0 - momentum) * grad
        a_new = a - lr * v
        a_new = np.maximum(a_new, 1e-6)
        ss = a_new.sum()
        if ss < 1e-6:
            a_new = a
            lr *= 0.5
            continue
        a_new = a_new / ss
        J_new = compute_J(a_new)
        
        if J_new < current_J - 1e-15:
            a = a_new
            current_J = J_new
            stuck_count = 0
            tau = min(tau_max, tau * 1.03)
            lr = 1.0 / (n * tau) * 0.5
        else:
            stuck_count += 1
            lr *= 0.7
            v *= 0.5
            if stuck_count > 50:
                tau = max(10.0, tau * 0.8)
                lr = 1.0 / (n * tau) * 0.5
                stuck_count = 0
            if lr < 1e-18:
                break
        if it % 200 == 0:
            tracker.update(a)
    
    tracker.update(a)
    return a

def resample_seq(src, dest_n):
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    t_old = np.linspace(0.0, 1.0, src.size)
    t_new = np.linspace(0.0, 1.0, dest_n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)

def generate_inits(n, rng, parent=None):
    inits = []
    t = np.linspace(0.0, 1.0, n, endpoint=False)
    
    if parent is not None and parent.size > 0:
        inits.append(resample_seq(parent, n))
        inits.append(resample_seq(parent[::-1], n))
    
    inits.append(np.ones(n) / n)
    inits.append(rng.uniform(0.1, 1.0, n))
    
    # Flat spectrum
    for _ in range(2):
        half = n // 2 + 1
        angles = rng.uniform(0, 2 * np.pi, half)
        A = np.zeros(n, dtype=complex)
        A[:half] = np.exp(1j * angles[:half])
        for k in range(1, n - n // 2):
            A[k] = np.conj(A[n - k])
        a_flat = np.fft.ifft(A).real
        a_flat = np.maximum(a_flat, 0.0)
        s = a_flat.sum()
        if s > 1e-10:
            inits.append(a_flat / s)
        else:
            inits.append(np.ones(n) / n)
    
    # Cosine modulations
    for freq in [2, 3, 5, 7, 11, 13]:
        inits.append(1.0 + 0.4 * np.cos(2 * np.pi * freq * t + rng.uniform(0, 6.28)))
    
    # Multi-frequency
    inits.append(1.0 + 0.25 * np.sin(2 * np.pi * t) + 0.2 * np.sin(4 * np.pi * t)
                + 0.15 * np.sin(7 * np.pi * t) + 0.1 * np.sin(11 * np.pi * t))
    
    # Gaussian bumps
    inits.append(0.5 * np.exp(-6 * (t - 0.3) ** 2) + 0.5 * np.exp(-6 * (t - 0.7) ** 2) + 0.05)
    
    # Power law
    inits.append((t + 0.1) ** rng.uniform(-1.0, 1.0))
    
    # Smoothed spikes
    sparse = np.zeros(n)
    n_spikes = max(5, n // 12)
    positions = rng.choice(n, size=n_spikes, replace=False)
    sparse[positions] = 1.0
    kernel_size = max(3, n // 20)
    kernel = np.ones(kernel_size) / kernel_size
    inits.append(np.convolve(sparse, kernel, mode='same') + 0.05)
    
    # Limited bandwidth Fourier
    n_coeff = min(n // 2 + 1, 15)
    coeffs = rng.normal(0, 1, n_coeff)
    sig = np.zeros(n)
    for i in range(n_coeff):
        sig += coeffs[i] * np.cos(2 * np.pi * i * t / n + rng.uniform(0, 6.28))
    inits.append(np.abs(sig) + 0.5)
    
    for x in inits:
        x = np.maximum(x, 0.001).astype(np.float64)
        x = x / x.sum()
    
    return inits

class Tracker:
    def __init__(self):
        self.best = None
        self.best_score = float('inf')
        self.t_start = time.monotonic()
        self.count = 0
    
    def update(self, a):
        a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
        if a.size == 0:
            return
        s = a.sum()
        if s < 0.01:
            return
        self.count += 1
        score = compute_J(a)
        if np.isfinite(score) and score < self.best_score:
            self.best_score = score
            self.best = a.copy()
            elapsed = time.monotonic() - self.t_start
            if self.count % 5 == 1 or score < 1.51:
                print(f"  * J={score:.8f} n={a.size} t={elapsed:.1f}s eval#{self.count}", flush=True)
    
    def report(self, label=""):
        if self.best is not None:
            print(f"  {label} best: J={self.best_score:.8f} n={self.best.size}", flush=True)
