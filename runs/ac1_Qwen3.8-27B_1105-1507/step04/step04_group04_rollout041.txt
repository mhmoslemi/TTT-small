"""
Search for non-negative step-function heights minimizing
    2*n*max(conv(a,a)) / (sum(a))^2
by combining:
- FFT-based objective evaluation,
- smoothed multiplicative subgradient descent,
- exact pair-coordinate descent in the simplex,
- a linear-programming direction fixing one factor of the convolution,
- multi-start / length exploration around the parent construction.
"""

import time
import numpy as np

try:
    from scipy.optimize import linprog
    HAS_LINPROG = True
except Exception:
    HAS_LINPROG = False


# --------------------------------------------------------------------
# Basic utilities
# --------------------------------------------------------------------

def normalize_sequence(x, S):
    """Clip to nonnegative finite numbers and rescale to sum S."""
    x = np.asarray(x, dtype=np.float64).ravel()
    if x.size == 0:
        x = np.array([1.0], dtype=np.float64)
    x = np.where(np.isfinite(x), x, 0.0)
    x = np.maximum(x, 0.0)
    s = float(np.sum(x))
    if s <= 1e-300:
        x = np.ones_like(x, dtype=np.float64)
        s = float(x.size)
    return x * (float(S) / s)


def convolution_fft(a, sym=True):
    """Full convolution a*a using FFT, optionally symmetrized."""
    a = np.asarray(a, dtype=np.float64)
    n = int(a.size)
    m = 2 * n - 1
    if n == 0:
        return np.array([], dtype=np.float64)
    size = 1
    while size < m:
        size <<= 1
    fa = np.fft.rfft(a, size)
    c = np.fft.irfft(fa * fa, size)
    c = c[:m].copy()
    if sym:
        c = 0.5 * (c + c[::-1])
    return c


def internal_score_state(a, c=None):
    """Scale-invariant objective used internally."""
    a = np.asarray(a, dtype=np.float64)
    n = int(a.size)
    if n == 0:
        return float("inf")
    if c is None:
        c = convolution_fft(a)
    s = float(np.sum(a))
    if s <= 1e-15:
        return float("inf")
    return float(2.0 * n * float(np.max(c)) / (s * s))


def make_state(x, S):
    """Return (a, convolution, max-convolution) at sum S."""
    a = normalize_sequence(x, S)
    c = convolution_fft(a)
    m = float(np.max(c))
    return a, c, m


def softmax_weights(c, beta):
    """Softmax weights over convolution peaks."""
    c = np.asarray(c, dtype=np.float64)
    maxc = float(np.max(c))
    u = beta * (c - maxc)
    u = np.clip(u, -50.0, 50.0)
    w = np.exp(u)
    sw = float(np.sum(w))
    if sw <= 1e-300 or not np.isfinite(sw):
        w = np.zeros_like(c, dtype=np.float64)
        w[int(np.argmax(c))] = 1.0
    else:
        w = w / sw
    return w, maxc


def compute_G(a, w):
    """
    G_i = sum_t w_t * a_{t-i} over valid t.
    This is half of the gradient of sum_t w_t (a*a)_t.
    """
    a = np.asarray(a, dtype=np.float64)
    w = np.asarray(w, dtype=np.float64)
    n = int(a.size)
    m = 2 * n - 1
    total = m + n - 1
    size = 1
    while size < total:
        size <<= 1
    ar = a[::-1]
    fw = np.fft.rfft(w, size)
    fa = np.fft.rfft(ar, size)
    h = np.fft.irfft(fw * fa, size)
    return h[n - 1:n - 1 + n].copy()

    M_in = float(M_in)

    if time_budget <= 1.0:
        return a_in.copy(), c_in.copy(), M_in

    n = int(a_in.size)
    S = float(np.sum(a_in))
    if S <= 1e-300:

        a_in = a_in * (S / max(1e-300, float(np.sum(a_in))))

    a = a_in.copy()
    c = c_in.copy()
    M = M_in
    best_a = a.copy()
    best_c = c.copy()
    best_M = M

    rng = np.random.default_rng(seed)
    beta = float(beta_start)
    eta = float(eta_start)
    stall = 0

            a2 = a2 * (S / s2)
            c2 = convolution_fft(a2)
            M2 = float(np.max(c2))

def make_pair_deltas(ai, aj):
    """Candidate deltas for a_i <- a_i + d, a_j <- a_j - d."""
    ai = float(ai)
    aj = float(aj)
    low = -ai
    high = aj
    span = high - low
    if span <= 1e-15 * max(1.0, ai, aj):
        return np.array([0.0], dtype=np.float64)

    eps = 1e-13 * max(1.0, ai, aj)
    inner_lo = low + eps
    inner_hi = high - eps
    if inner_lo >= inner_hi:
        return np.array([0.0], dtype=np.float64)

    minm = min(ai, aj)
    candidates = [0.0]
    factors = (1e-3, 3e-3, 1e-2, 3e-2, 0.1, 0.3, 0.6, 0.9)
    for f in factors:
        d = f * minm
        candidates.append(d)
        candidates.append(-d)

    for q in (0.02, 0.08, 0.2, 0.5, 0.8, 0.95, 0.99):
        candidates.append(low + q * span)

    arr = np.asarray(candidates, dtype=np.float64)
    arr = arr[np.isfinite(arr)]
    arr = np.clip(arr, inner_lo, inner_hi)
    arr = np.unique(arr)
    arr = np.concatenate((np.array([0.0], dtype=np.float64), arr))
    arr = np.unique(arr)
    return arr

                bf = int(np.argmin(tf))
                if float(tf[bf]) < best_trial:
                    best_trial = float(tf[bf])
                    best_d = float(fine[bf])

        if best_trial < M - 1e-15:
            a[i] += best_d
            a[j] -= best_d
            c = c + L * best_d
            for idx, qv in q_list:
                c[idx] += qv * best_d * best_d

            if a[i] < 0.0 and a[i] > -1e-10:
                a[i] = 0.0
            if a[j] < 0.0 and a[j] > -1e-10:
                a[j] = 0.0
            if a[i] < 0.0:
                a[i] = 0.0
            if a[j] < 0.0:
                a[j] = 0.0

            s = float(np.sum(a))
            if s > 1e-300 and abs(s - S) > 1e-12 * max(1.0, S):
                factor = S / s
                a *= factor
                c *= factor * factor

            c = 0.5 * (c + c[::-1])
            M = float(np.max(c))
            accepted += 1
            if M < best_M - 1e-15:
                best_M = M
                best_a = a.copy()
                best_c = c.copy()
            stall = 0
        else:
            stall += 1
            if stall > 100:
                sigma = 0.02 + 0.03 * ((stall // 100) % 4)
                a = best_a * np.exp(rng.normal(0.0, sigma, n))
                a = np.maximum(a, 1e-300)
                a = a * (S / float(np.sum(a)))
                c = convolution_fft(a)
                M = float(np.max(c))
                if M < best_M - 1e-15:
                    best_M = M
                    best_a = a.copy()
                    best_c = c.copy()
                stall = 0
                beta = min(120.0, beta * 1.4)

        it += 1

    return best_a, best_c, best_M


# --------------------------------------------------------------------
# LP direction
# --------------------------------------------------------------------

def solve_lp_minpeak(a, S, M, time_limit=5.0):
    """
    Solve:
        min tau
        s.t. a*g <= tau,
             sum(g) = S,
             g >= 0, tau >= 0.
    Returns a direction g.
    """
    if not HAS_LINPROG:
        return None, None
    a = np.asarray(a, dtype=np.float64)
    n = int(a.size)
    if n == 0 or n > 800:
        return None, None

    m = 2 * n - 1
    A = np.zeros((m, n), dtype=np.float64)
    for j in range(n):
        A[j:j + n, j] = a

    c_obj = np.zeros(n + 1, dtype=np.float64)
    c_obj[-1] = 1.0

    A_ub = np.zeros((m, n + 1), dtype=np.float64)
    A_ub[:, :n] = A
    A_ub[:, n] = -1.0
    b_ub = np.zeros(m, dtype=np.float64)

    A_eq = np.zeros((1, n + 1), dtype=np.float64)
    A_eq[0, :n] = 1.0
    b_eq = np.array([float(S)], dtype=np.float64)

    bounds = [(0.0, None)] * n + [(0.0, None)]
    try:
        res = linprog(
            c_obj,
            A_ub=A_ub,
            b_ub=b_ub,
            A_eq=A_eq,
            b_eq=b_eq,
            bounds=bounds,
            method="highs",
            options={"presolve": True, "time_limit": float(max(1.0, min(time_limit, 8.0)))}
        )
    except Exception:
        return None, None

    if res is None or res.status != 0 or res.x is None:
        return None, None

    g = np.asarray(res.x[:n], dtype=np.float64)
    g = np.maximum(g, 0.0)
    tau = float(res.x[n])
    s = float(np.sum(g))
    if s <= 1e-12:
        return None, None
    g = g * (float(S) / s)
    return g, tau


def lp_blend_improve(a_in, c_in, M_in, time_budget=30.0, seed=1):
    """Use LP directions and simple blending."""
    start = time.monotonic()
    a_in = np.asarray(a_in, dtype=np.float64).copy()
    c_in = np.asarray(c_in, dtype=np.float64).copy()
    M_in = float(M_in)

    if time_budget <= 6.0 or a_in.size > 800:
        return a_in.copy(), c_in.copy(), M_in

    n = int(a_in.size)
    S = float(np.sum(a_in))
    best_a = a_in.copy()
    best_c = c_in.copy()
    best_M = M_in
    rng = np.random.default_rng(seed)
    end = start + float(time_budget) - 4.0
    attempts = 0

    while time.monotonic() < end and attempts < 3:
        attempts += 1
        g, _ = solve_lp_minpeak(best_a, S, float(np.max(best_c)), time_limit=4.0)
        if g is None:
            break

        g = normalize_sequence(g, S)
        improved = False

        for t in (0.2, 0.5, 0.8, 1.0):
            a2 = (1.0 - float(t)) * best_a + float(t) * g
            a2 = normalize_sequence(a2, S)
            c2 = convolution_fft(a2)
            M2 = float(np.max(c2))
            if M2 < best_M - 1e-15:
                best_A = a2
                best_C = c2
                best_M = M2
                best_a = best_A
                best_c = best_C
                improved = True

        if time.monotonic() < end:
            gp = g * np.exp(rng.normal(0.0, 0.05, n))
            gp = normalize_sequence(gp, S)
            for t in (0.3, 0.7):
                a2 = (1.0 - float(t)) * best_a + float(t) * gp
                a2 = normalize_sequence(a2, S)
                c2 = convolution_fft(a2)
                M2 = float(np.max(c2))
                if M2 < best_M - 1e-15:
                    best_A = a2
                    best_C = c2
                    best_M = M2
                    best_a = best_A
                    best_c = best_C
                    improved = True

        if not improved and attempts >= 2:
            break

    return best_a, best_c, best_M

def resample_sequence(a, new_n, S):
    """Resample a 1-periodic step-height array to length new_n."""
    a = np.asarray(a, dtype=np.float64).ravel()
    new_n = int(max(2, new_n))
    old_n = int(a.size)
    if old_n == 0:
        return normalize_sequence(np.ones(new_n, dtype=np.float64), S)
    if old_n == new_n:
        return normalize_sequence(a, S)

    x_old = np.linspace(0.0, 1.0, old_n, endpoint=False) + 0.5 / old_n
    x_new = np.linspace(0.0, 1.0, new_n, endpoint=False) + 0.5 / new_n
    y = np.interp(x_new, x_old, a)
    return normalize_sequence(y, S)


def random_init(n, seed):
    """Diverse positive random initializations."""
    rng = np.random.default_rng(seed)
    n = int(max(2, n))
    k = np.arange(n, dtype=np.float64)
    choice = int(rng.integers(0, 9))

    if choice == 0:
        f = float(rng.uniform(0.5, 4.0))
        ph = float(rng.uniform(0.0, 2.0 * np.pi))
        a = 1.0 + 0.75 * np.cos(2.0 * np.pi * f * k / n + ph)
    elif choice == 1:
        f1 = float(rng.uniform(0.3, 2.0))
        f2 = float(rng.uniform(2.0, 6.0))
        ph = float(rng.uniform(0.0, 2.0 * np.pi))
        a = 1.0 + 0.45 * np.cos(2.0 * np.pi * f1 * k / n) + 0.30 * np.cos(2.0 * np.pi * f2 * k / n + ph)
    elif choice == 2:
        center = n * float(rng.uniform(0.3, 0.7))
        alpha = float(rng.uniform(1.0, 4.0))
        a = 1.0 + 0.6 * np.exp(-alpha * np.abs(k - center) / max(1.0, n * 0.3))
    elif choice == 3:
        a = rng.random(n)
        win = max(3, n // 18)
        kernel = np.ones(win, dtype=np.float64) / float(win)
        a = np.convolve(a, kernel, mode="same")
    elif choice == 4:
        a = np.ones(n, dtype=np.float64)
        for _ in range(int(rng.integers(1, 4))):
            f = float(rng.uniform(0.5, 5.0))
            amp = float(rng.uniform(0.1, 0.4))
            ph = float(rng.uniform(0.0, 2.0 * np.pi))
            a += amp * np.cos(2.0 * np.pi * f * k / n + ph)
    elif choice == 5:
        a = np.ones(n, dtype=np.float64)
        for scale in range(3):
            f = float(rng.uniform(0.3 * (2 ** scale), 1.5 * (2 ** scale)))
            amp = 0.30 / float(scale + 1)
            ph = float(rng.uniform(0.0, 2.0 * np.pi))
            a += amp * np.cos(2.0 * np.pi * f * k / n + ph)
    elif choice == 6:
        center = n * float(rng.uniform(0.2, 0.8))
        width = n * float(rng.uniform(0.1, 0.5))
        a = 1.0 + 0.7 * np.exp(-0.5 * ((k - center) / max(1.0, width)) ** 2)
    elif choice == 7:
        a = rng.random(max(5, n // 10))
        a = np.interp(k, np.arange(a.size), a)
    else:
        a = np.ones(n, dtype=np.float64)
        for _ in range(int(rng.integers(1, 3))):
            f = float(rng.uniform(0.5, 3.0))
            amp = float(rng.uniform(0.2, 0.5))
            ph = float(rng.uniform(0.0, 2.0 * np.pi))
            a += amp * np.sin(2.0 * np.pi * f * k / n + ph)

    a = np.maximum(a, 0.0)
    s = float(np.sum(a))
    if s <= 1e-300:
        a = np.ones(n, dtype=np.float64)
        s = float(n)
    return a / s


def perturb_log(a, sigma, seed):
    """Multiplicative log-normal perturbation."""
    a = np.asarray(a, dtype=np.float64).copy()
    n = int(a.size)
    S = float(np.sum(a))
    if S <= 1e-300:
        S = float(np.sqrt(2.0 * n))
        a = a * (S / max(1e-300, float(np.sum(a))))
    rng = np.random.default_rng(seed)
    a2 = a * np.exp(rng.normal(0.0, float(sigma), n))
    return normalize_sequence(a2, S)

# Full local optimizer

def full_local_optimize(a_in, time_budget, seed=1, verbose=False):
    """Run a multi-stage local optimizer on a fixed length sequence."""
    a_in = np.asarray(a_in, dtype=np.float64).ravel()
    n = int(a_in.size)
    if n <= 0:
        a_in = np.ones(330, dtype=np.float64)
        n = 330

    S = float(np.sqrt(2.0 * n))
    a, c, M = make_state(a_in, S)
    best_a = a.copy()
    best_c = c.copy()
    best_M = M

    start = time.monotonic()
    end = start + max(1.0, float(time_budget)) - 1.0

    if verbose:
        print(f"      local start n={n} M={best_M:.9f} budget={float(time_budget):.1f}")

    # For tiny budgets, only do pair descent.
    if time_budget < 35.0:
        b = max(1.0, time_budget - 0.5)
        a2, c2, M2 = pair_descent(a, c, M, b, beta=30.0, seed=seed)
        if M2 < best_M - 1e-15:
            best_M = M2
            best_a = a2.copy()
            best_c = c2.copy()
        if verbose:
            print(f"      local end   n={n} M={best_M:.9f}")
        return best_a, best_M

    # Stage 1: smoothed multiplicative descent.
    b = min(0.18 * float(time_budget), 60.0)
    rem = end - time.monotonic()
    if rem > max(3.0, b):
        a1, c1, M1 = smoothed_opt(a, c, M, min(b, rem - 0.5),
                                  beta_start=3.0, beta_end=30.0,
                                  eta_start=0.40, seed=seed + 11)
        if M1 < best_M - 1e-15:
            best_M = M1
            best_a = a1.copy()
            best_c = c1.copy()

    a, c, M = best_a, best_c, best_M

    # Stage 2: pair descent.
    b = min(0.35 * float(time_budget), 220.0)
    rem = end - time.monotonic()
    if rem > max(3.0, b):
        a2, c2, M2 = pair_descent(a, c, M, min(b, rem - 0.5),
                                  beta=20.0, seed=seed + 22)
        if M2 < best_M - 1e-15:
            best_M = M2
            best_a = a2.copy()
            best_c = c2.copy()

    a, c, M = best_a, best_c, best_M

    # Stage 3: LP direction for moderate lengths.
    if n <= 700:
        b = min(0.14 * float(time_budget), 45.0)
        rem = end - time.monotonic()
        if rem > max(7.0, b):
            a3, c3, M3 = lp_blend_improve(a, c, M, min(b, rem - 0.5), seed=seed + 33)
            if M3 < best_M - 1e-15:
                best_M = M3
                best_a = a3.copy()
                best_c = c3.copy()

            b2 = min(0.08 * float(time_budget), 25.0)
            rem = end - time.monotonic()
            if rem > max(3.0, b2):
                a4, c4, M4 = pair_descent(a3, c3, M3, min(b2, rem - 0.5),
                                          beta=35.0, seed=seed + 44)
                if M4 < best_M - 1e-15:
                    best_M = M4
                    best_a = a4.copy()
                    best_c = c4.copy()

    a, c, M = best_a, best_c, best_M

    # Stage 4: high-behavior smoothed descent.
    b = min(0.12 * float(time_budget), 30.0)
    rem = end - time.monotonic()
    if rem > max(3.0, b):
        a5, c5, M5 = smoothed_opt(a, c, M, min(b, rem - 0.5),
                                  beta_start=12.0, beta_end=120.0,
                                  eta_start=0.18, seed=seed + 55)
        if M5 < best_M - 1e-15:
            best_M = M5
            best_a = a5.copy()
            best_c = c5.copy()

    a, c, M = best_a, best_c, best_M

    # Final pair polish.
    b = max(0.0, end - time.monotonic() - 0.3)
    if b > 2.0:
        a6, c6, M6 = pair_descent(a, c, M, b, beta=50.0, seed=seed + 66)
        if M6 < best_M - 1e-15:
            best_M = M6
            best_a = a6.copy()
            best_c = c6.copy()

    if verbose:
        print(f"      local end   n={n} M={best_M:.9f}")

    return best_a, best_M

def safe_eval(a):
    """Evaluate using the external evaluator if available; fallback to internal."""
    a = np.asarray(a, dtype=np.float64).ravel()
    try:
        lst = [float(max(0.0, min(1000.0, float(x)))) for x in a]
        val = float(evaluate_sequence(lst))
        if np.isfinite(val):
            return val
    except Exception:
        pass
    return internal_score_state(a)

def propose_candidate(seed=42, budget_s=1500):
    """
    Main candidate construction under a time budget.
    Returns a Python list of non-negative floats.
    """
    start_time = time.monotonic()
    total_budget = float(budget_s)
    deadline = start_time + total_budget - 25.0

    # ----------------------------------------------------------------
    # Load parent if available.
    # ----------------------------------------------------------------
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if p.size > 0 and np.all(np.isfinite(p)):
            parent = p
    except Exception:
        parent = None

    if parent is None:
        n_parent = 330
        parent = np.ones(n_parent, dtype=np.float64)
    else:
        n_parent = int(parent.size)

    S_parent = float(np.sqrt(2.0 * n_parent))
    best_a = normalize_sequence(parent, S_parent)
    best_M = internal_score_state(best_a)
    best_true = safe_eval(best_a)
    if not np.isfinite(best_true):
        best_true = best_M

    print(f"Parent length: {n_parent}")
    print(f"Initial internal score: {best_M:.9f}")
    print(f"Initial true score:     {best_true:.9f}")

    # ----------------------------------------------------------------
    # Build a priority list of lengths to try.
    # ----------------------------------------------------------------
    exp_assign = {}

    def add_exp(n2, budget):
        n2 = int(n2)
        if n2 < 80 or n2 > 1500:
            return
        if n2 not in exp_assign:
            exp_assign[n2] = float(budget)

    add_exp(n_parent, min(320.0, max(140.0, total_budget * 0.25)))

    for delta in (-30, -20, -10, 10, 20, 30, -60, 60, -90, 90):
        add_exp(n_parent + delta, 55.0 if abs(delta) <= 30 else 40.0)

    for n2 in (250, 300, 320, 350, 400, 450, 500, 600, 800, 1000):
        add_exp(n2, 42.0)

    # Priority order: parent first, then nearby, then others.
    exp_order = [n_parent]
    seen = set()
    exp_order_seen = {n_parent}

    def insert_near(n2):
        if n2 in exp_order_seen or n2 not in exp_assign:
            return
        if 80 <= n2 <= 1500:
            exp_order.append(n2)
            exp_order_seen.add(n2)

    for delta in (-10, -20, -30, 10, 20, 30, -60, 60, -90, 90):
        insert_near(n_parent + delta)

    for n2 in exp_assign.keys():
        if n2 not in exp_order_seen:
            exp_order.append(n2)
            exp_order_seen.add(n2)

    # ----------------------------------------------------------------
    # Run fixed-length experiments.
    # ----------------------------------------------------------------
    attempted = 0
    for n_exp in exp_order:
        now = time.monotonic()
        if now > deadline - 40.0:
            break
        remaining = deadline - now - 25.0
        if remaining < 22.0:
            break

        n_exp = int(n_exp)
        base_budget = exp_assign.get(n_exp, 35.0)
        alloc = min(base_budget, remaining)
        if alloc < 20.0:
            continue

        S_exp = float(np.sqrt(2.0 * n_exp))
        if n_exp == n_parent and best_a.size == n_exp:
            seq0 = best_a.copy()
        elif parent is not None and parent.size > 0:
            seq0 = resample_sequence(parent, n_exp, S_exp)
        else:
            seq0 = random_init(n_exp, seed + 1000 + 7919 * n_exp)

        verbose = (n_exp == n_parent and attempted == 0)
        a_opt, M_opt = full_local_optimize(seq0, alloc, seed=seed + 104729 * n_exp + attempted,
                                           verbose=verbose)
        true_opt = safe_eval(a_opt)
        if not np.isfinite(true_opt):
            true_opt = M_opt

        attempted += 1
        elapsed = time.monotonic() - start_time
        print(f"  experiment n={n_exp:<4d} internal={M_opt:.9f} true={true_opt:.9f} elapsed={elapsed:7.1f}s")

        if true_opt < best_true - 1e-12:
            best_true = true_opt
            best_M = M_opt
            best_a = np.asarray(a_opt, dtype=np.float64).copy()
            print(f"  ** UPDATE best true={best_true:.9f} at n={n_exp} **")

    # ----------------------------------------------------------------
    # Multi-start polishing of the current best length.
    # ----------------------------------------------------------------
    attempt = 0
    no_improve = 0

    while time.monotonic() < deadline - 40.0:
        attempt += 1
        remaining = deadline - time.monotonic() - 25.0
        if remaining < 25.0:
            break

        n_best = int(best_a.size)
        S_best = float(np.sqrt(2.0 * n_best))

        # Different perturbation scales.
        q = attempt % 7
        if q == 0:
            sigma = 0.015
        elif q == 1:
            sigma = 0.030
        elif q == 2:
            sigma = 0.055
        elif q == 3:
            sigma = 0.090
        elif q == 4:
            sigma = 0.140
        elif q == 5:
            sigma = 0.220
        else:
            sigma = 0.040

        if q == 6:
            rnd = random_init(n_best, seed + 999983 + attempt * 104729)
            rnd = normalize_sequence(rnd, S_best)
            seq0 = 0.65 * best_a + 0.35 * rnd
            seq0 = normalize_sequence(seq0, S_best)
        else:
            seq0 = perturb_log(best_a, sigma, seed + 999983 + attempt * 104729)

        alloc = min(48.0, remaining * 0.38)
        alloc = max(20.0, alloc)
        if alloc > remaining - 5.0:
            alloc = max(5.0, remaining - 5.0)
        if alloc < 18.0:
            break

        a_opt, M_opt = full_local_optimize(seq0, alloc, seed=seed + 999983 + attempt * 99991,
                                           verbose=False)
        true_opt = safe_eval(a_opt)
        if not np.isfinite(true_opt):
            true_opt = M_opt

        if true_opt < best_true - 1e-12:
            best_true = true_opt
            best_M = M_opt
            best_a = np.asarray(a_opt, dtype=np.float64).copy()
            no_improve = 0
            print(f"  multi-start {attempt}: UPDATE true={best_true:.9f} n={n_best}")
        else:
            no_improve += 1
            if attempt % 5 == 0:
                print(f"  multi-start {attempt}: try={true_opt:.9f} best={best_true:.9f} n={n_best}")

        # Occasionally do direct long pair-polish on the current best.
        if attempt % 8 == 0:
            remaining2 = deadline - time.monotonic() - 25.0
            if remaining2 > 25.0:
                a_pol, c_pol, M_pol = make_state(best_a, float(np.sqrt(2.0 * best_a.size)))
                pol_time = min(70.0, remaining2 - 10.0)
                a_pol2, c_pol2, M_pol2 = pair_descent(a_pol, c_pol, M_pol, pol_time,
                                                      beta=60.0,
                                                      seed=seed + 55555 + attempt)
                true_pol = safe_eval(a_pol2)
                if np.isfinite(true_pol) and true_pol < best_true - 1e-12:
                    best_true = true_pol
                    best_M = M_pol2
                    best_a = np.asarray(a_pol2, dtype=np.float64).copy()
                    print(f"  direct polish: UPDATE true={best_true:.9f} n={best_a.size}")

    # ----------------------------------------------------------------
    # Final polish with whatever remains.
    # ----------------------------------------------------------------
    remaining = deadline - time.monotonic() - 5.0
    if remaining > 5.0:
        a_fin, c_fin, M_fin = make_state(best_a, float(np.sqrt(2.0 * best_a.size)))
        a_fin2, c_fin2, M_fin2 = pair_descent(a_fin, c_fin, M_fin, remaining,
                                              beta=70.0,
                                              seed=seed + 1234567)
        true_fin = safe_eval(a_fin2)
        if np.isfinite(true_fin) and true_fin < best_true:
            best_true = true_fin
            best_M = M_fin2
            best_a = np.asarray(a_fin2, dtype=np.float64).copy()

    # Final validation and cleanup.
    result = [float(max(0.0, min(1000.0, float(x)))) for x in best_a]
    try:
        final_val = float(evaluate_sequence(result))
        if np.isfinite(final_val):
            best_true = final_val
    except Exception:
        pass

    elapsed = time.monotonic() - start_time
    print(f"FINAL: n={len(result)} score={best_true:.9f} elapsed={elapsed:.2f}s")
    return result
