def fourier_init(n, harmonics, rng):
    """Generate a random Fourier-based initialization."""
    t = np.arange(n, dtype=np.float64) / n
    a = np.full(n, 1.0)
    for k in range(1, harmonics + 1):
        a += rng.uniform(-0.3, 0.3) * np.cos(2 * np.pi * k * t + rng.uniform(0, 2 * np.pi))
        a += rng.uniform(-0.3, 0.3) * np.sin(2 * np.pi * k * t)
    a = np.maximum(a, 0.01)
    return a / a.sum()


def gaussian_init(n, rng):
    """Gaussian-shaped initialization."""
    t = np.arange(n, dtype=np.float64) / n
    center = rng.uniform(0.3, 0.7)
    width = rng.uniform(0.1, 0.3)
    a = np.exp(-((t - center) ** 2) / (2 * width ** 2)) + 0.05
    return a / a.sum()


def cosine_init(n):
    """Cosine-shaped initialization (optimal for n=2 case, good warm start)."""
    t = np.arange(n, dtype=np.float64) / n
    a = 0.5 * (1 + np.cos(2 * np.pi * t))
    a = np.maximum(a, 0.001)
    return a / a.sum()


def watermelon_init(n):
    """Watermelon shape: high in middle."""
    t = np.arange(n, dtype=np.float64) / n
    a = 1.0 - 0.5 * (2 * t - 1) ** 2
    a = np.maximum(a, 0.01)
    return a / a.sum()


def resample_seq(src, dest_n):
    """Resample 1D sequence to new length."""
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)


def subgradient_with_momentum(a, deadline, max_iters=2000):
    """Subgradient descent with heavy ball momentum on the max-convolution."""
    a = np.maximum(a.astype(np.float64), 1e-10)
    a = a / a.sum()
    n = len(a)
    best_M = fast_conv(a).max()
    best_a = a.copy()
    v = np.zeros_like(a)
    beta = 0.9
    lr = 0.5
    
    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        
        b = fast_conv(a)
        M = b.max()
        
        # Find active set (lags close to max)
        tol = 1e-6 * M
        active = b >= (M - tol)
        active_idx = np.where(active)[0]
        
        if len(active_idx) == 0:
            break
        
        # Subgradient of max_k (a*a)[k] w.r.t. a
        # d/d a[i] of (a*a)[k] = a[k-i] + a[k-i] = 2*a[k-i] (if i != k-i)
        # or a[k-i] + a[i] = 2*a[i] (if i == k-i, i.e., k = 2i)
        grad = np.zeros(n)
        for k in active_idx:
            for i in range(n):
                j = k - i
                if 0 <= j < n:
                    grad[i] += a[j]
        
        # Normalize
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-15:
            break
        
        # Momentum update
        v = beta * v + (1 - beta) * grad / gnorm
        a_new = a - lr / np.sqrt(it + 1) * v
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s < 1e-10:
            break
        a_new = a_new / s
        
        new_M = fast_conv(a_new).max()
        if new_M < best_M:
            best_M = new_M
            best_a = a_new.copy()
        a = a_new
    
    return best_a


def lbfgs_polish(a, deadline):
    """Polish using L-BFGS-B on the max-conviction objective."""
    n = len(a)
    
    def objective(x):
        ax = np.maximum(x, 0.0)
        s = ax.sum()
        if s < 1e-8:
            return 1e10
        ax = ax / s
        b = fast_conv(ax)
        return float(b.max())
    
    try:
        res = minimize(objective, a.astype(np.float64),
                       method='L-BFGS-B', bounds=[(0.0, None)] * n,
                       options={'maxiter': 50, 'max_fun': 10000, 'ftol': 1e-18, 'gtol': 1e-12})
        if res.fun < 1e10:
            return np.maximum(res.x, 0.0)
    except Exception:
        pass
    return a

Improved optimizer for minimizing J(a) = 2*n*max(conv(a,a))/sum(a)^2.
Uses fast vectorized LP mirror descent with exponentiation mechanism,

import numpy as np
from scipy.optimize import linprog, minimize
from scipy.sparse import csr_matrix
import time


def fast_conv(a):
    """Full self-convolution via FFT."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]


def compute_J(a):
    """Compute J(a) = 2*n*max(conv(a,a))/(sum(a))^2."""
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))


def build_A_structure(n):
    """Pre-compute the row/col/index structure for the convolution matrix (depends only on n)."""
    ks = np.arange(2 * n - 1)
    j_starts = np.maximum(0, ks - n + 1)
    j_ends = np.minimum(n - 1, ks)
    counts = j_ends - j_starts + 1
    rows = np.repeat(ks, counts)
    cols = np.concatenate([np.arange(s, e + 1) for s, e in zip(j_starts, j_ends)])
    idx = rows - cols
    return rows, cols, idx


def make_A(n, a, structure):
    """Create sparse convolution matrix using pre-computed structure."""
    rows, cols, idx = structure
    vals = a[idx]
    return csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))


def lp_solve(a, M, structure):
    """Solve LP: max sum(g) s.t. A(a)*g <= M, g >= 0. Returns g or None."""
    n = len(a)
    A = make_A(n, a, structure)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    try:
        r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
        if r.success and r.x is not None:
            g = np.maximum(r.x, 0)
            if g.sum() > 1e-15:
                return g
    except Exception:
        pass
    return None


def lp_mirror_descent(a, deadline, structure, use_mult=True, max_iters=20000):
    """
    LP mirror descent with adaptive step size.
    Uses exponentiation (multiplicative) or linear (convex combination) updates.
    """
    a = np.maximum(a.astype(np.float64), 1e-12)
    a = a / a.sum()
    n = len(a)
    best_J = compute_J(a)
    best_a = a.copy()
    no_improve = 0
    stagnation_count = 0

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        b = fast_conv(a)
        M = float(b.max())

        g = lp_solve(a, M, structure)
        if g is None:
            break
        g_sum = g.sum()
        if g_sum < 1e-12:
            break
        g = g / g_sum

        # Adaptive step size: start larger, decay slowly
        # Use a schedule that allows more exploration early and refinement late
        if it < 50:
            base_t = 0.4
        elif it < 200:
            base_t = 0.3 - 0.15 * (it - 50) / 150.0
        else:
            base_t = 0.15 * np.exp(-0.02 * (it - 200) / 100.0) + 0.02
        t = min(1.0, base_t)

        best_trial_J = best_J  # Only accept if better than best
        best_trial_a = None

        for ls in range(40):
            if use_mult:
                ratio = g / np.maximum(a, 1e-15)
                a_trial = a * np.power(ratio, t)
            else:
                a_trial = (1.0 - t) * a + t * g

            a_trial = np.maximum(a_trial, 0.0)
            s_trial = a_trial.sum()
            if s_trial < 1e-10:
                t *= 0.5
                continue
            a_trial = a_trial / s_trial
            J_trial = compute_J(a_trial)

            if J_trial < best_trial_J - 1e-15:
                best_trial_J = J_trial
                best_trial_a = a_trial.copy()
                break
            t *= 0.65

            current_J = best_trial_J
            if current_J < best_J - 1e-13:

                best_a = a.copy()
                no_improve = 0
                stagnation_count = 0
            else:

                stagnation_count += 1
        else:
            no_improve += 1
            stagnation_count += 1

        if no_improve > 200:

def subgradient_polish(a, deadline, max_iters=5000):
    """Subgradient descent with momentum on max-convolution."""
    a = np.maximum(a.astype(np.float64), 1e-12)
    a = a / a.sum()
    n = len(a)
    best_M = fast_conv(a).max()
    best_a = a.copy()
    v = np.zeros_like(a)
    beta = 0.9
    lr = 1.0

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        b = fast_conv(a)
        M = b.max()

        # Active set: lags within tolerance of max
        tol = max(1e-8 * M, 1e-12)
        active_idx = np.where(b >= M - tol)[0]

        if len(active_idx) == 0:
            break

        # Compute subgradient
        grad = np.zeros(n)
        for k in active_idx:
            # d(conv[k])/d(a[i]) = a[k-i] if 0 <= k-i < n
            lo = max(0, k - n + 1)
            hi = min(n - 1, k)
            for i in range(lo, hi + 1):
                grad[i] += a[k - i]

        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-15:
            break

        v = beta * v + (1 - beta) * grad / gnorm
        step = lr / (1.0 + 0.1 * np.sqrt(it + 1.0))
        a_new = a - step * v
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s < 1e-10:
            break
        a_new = a_new / s

        new_M = fast_conv(a_new).max()
        if new_M < best_M:
            best_M = new_M
            best_a = a_new.copy()
        a = a_new

    return best_a


def fourier_obj(params, n, n_harm):
    """Objective for Fourier parameter optimization: max of self-conv (scale-invariant)."""
    t = np.arange(n, dtype=np.float64) / n
    a = np.full(n, params[0], dtype=np.float64)
    for k in range(1, n_harm + 1):
        a += params[2 * k - 1] * np.cos(2.0 * np.pi * k * t)
        a += params[2 * k] * np.sin(2.0 * np.pi * k * t)
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-8:
        return 1e10
    a = a / s
    b = fast_conv(a)
    return float(b.max())


def fourier_to_seq(params, n, n_harm):
    """Convert Fourier params to non-negative normalized sequence."""
    t = np.arange(n, dtype=np.float64) / n
    a = np.full(n, params[0], dtype=np.float64)
    for k in range(1, n_harm + 1):
        a += params[2 * k - 1] * np.cos(2.0 * np.pi * k * t)
        a += params[2 * k] * np.sin(2.0 * np.pi * k * t)
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-8:
        a = np.ones(n) / n
    else:
        a = a / s
    return a


def seq_to_fourier(a, n_harm):
    """Extract Fourier parameters from a normalized sequence."""
    n = len(a)
    a_norm = a / a.sum()
    t = np.arange(n, dtype=np.float64) / n
    params = np.zeros(2 * n_harm + 1)
    params[0] = np.mean(a_norm)
    for k in range(1, n_harm + 1):
        params[2 * k - 1] = 2.0 * np.mean(a_norm * np.cos(2.0 * np.pi * k * t))
        params[2 * k] = 2.0 * np.mean(a_norm * np.sin(2.0 * np.pi * k * t))
    return params


def resample_seq(src, dest_n):
    """Resample 1D sequence to new length."""
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)
