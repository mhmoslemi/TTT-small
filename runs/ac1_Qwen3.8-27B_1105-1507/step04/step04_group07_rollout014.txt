```python
Minimizes J(a) = 2*n*max(convolve(a,a))/(sum(a))^2 using:
1. LP best-response with precomputed sparse matrices
2. Adam-optimized LSE gradient descent
3. Multi-length multi-init search
4. Simulated annealing with LP refinement
5. Coordinate descent polish
"""

import numpy as np
from scipy.optimize import linprog
from scipy.sparse import csr_matrix
import time

# Precomputed sparsity cache
_spar_cache = {}

def get_sparsity(n):
    if n not in _spar_cache:
        rows = []
        cols = []
        for k in range(2 * n - 1):
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            rows.extend([k] * (je - js + 1))
            cols.extend(range(js, je + 1))
        _spar_cache[n] = (np.array(rows, dtype=np.int32), np.array(cols, dtype=np.int32), len(rows))
    return _spar_cache[n]

def fast_conv(a):
    n = len(a)
    s = 1
    while s < 2 * n - 1:
        s <<= 1
    fa = np.fft.rfft(a, s)
    return np.fft.irfft(fa * fa, s)[:2 * n - 1]

def compute_J(a):
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))

def lp_best_response(a, M, time_limit=5):
    n = len(a)
    rows, cols, nnz = get_sparsity(n)
    # Build the values for the constraint matrix
    # A[k,j] = a[k-j] if 0 <= k-j < n
    vals = np.empty(nnz)
    for i in range(nnz):
        k = rows[i]
        j = cols[i]
        vals[i] = a[k - j]
    A = csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                options={'presolve': True, 'time_limit': time_limit})
    if r.success and r.x is not None:
        g = np.maximum(r.x, 0.0)
        if g.sum() > 1e-10:
            return g
    return None
```

```python
def lse_adam(a, max_iters, deadline, tracker, tau_init=50.0):
    a = np.maximum(a.astype(np.float64), 1e-8)
    a /= a.sum()
    n = len(a)
    current_J = compute_J(a)
    tracker.update(a)
    
    tau = tau_init
    lr = 0.1
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    
    m = np.zeros(n)  # first moment
    v = np.zeros(n)  # second moment
    t = 0
    
    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        
        t += 1
        b = fast_conv(a)
        b_max = b.max()
        exp_b = np.exp(tau * (b - b_max))
        S_b = exp_b.sum()
        p = exp_b / S_b
        
        # Gradient of LSE: grad_j = 2 * sum_k p[k] * (db_k/da_j)
        # db_k/da_j = a[k-j] if 0 <= k-j < n
        # So grad_j = 2 * sum_{k: 0<=k-j<n} p[k] * a[k-j]
        # Let l = k-j, then k = j+l, sum over valid l
        # grad_j = 2 * sum_{l=0}^{n-1} p[j+l] * a[l] (where j+l < 2n-1)
        # This is a cross-correlation!
        
        # Using FFT: cross-correlation of p and a
        s = 1
        while s < 2 * n - 1:
            s <<= 1
        fp = np.fft.rfft(p, s)
        fa = np.fft.rfft(a, s)
        # Cross-correlation: ifft(conj(fp) * fa) or ifft(fp * conj(fa)) depending on convention
        # We want: sum_k p[k] * a[k-j] for each j
        # This is the cross-correlation (p ⊛ a)[j] = sum_k p[k] * a[k-j]
        # In FFT: ifft(conj(FFT(p)) * FFT(a))[j]
        corr = np.fft.irfft(np.conj(fp) * fa, s)
        grad_lse = 2.0 * corr[:n]
        
        # Full gradient of J = 2n * LSE / S^2 where S = sum(a) = 1 (normalized)
        # dJ/da_j = 2n * grad_lse_j / S^2 - 4n * LSE_val * a_j / S^3
        # With S = 1: dJ/da_j = 2n * grad_lse_j - 4n * LSE_val * a_j
        # LSE_val = log(S_b) + tau * b_max (but we only need the ratio, so use the smoothed max)
        smoothed_max = b_max + np.log(S_b) / tau
        grad = 2.0 * n * grad_lse / 1.0 - 4.0 * n * smoothed_max * a
        
        # Adam update
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad ** 2)
        m_hat = m / (1 - beta1 ** t)
        v_hat = v / (1 - beta2 ** t)
        
        a_new = a - lr * m_hat / (np.sqrt(v_hat) + eps)
        a_new = np.maximum(a_new, 1e-8)
        a_new /= a_new.sum()
        
        J_new = compute_J(a_new)
        if J_new < current_J:
            a = a_new
            current_J = J_new
            # Increase tau slightly for sharper approximation
            tau = min(500.0, tau * 1.02)
        else:
            lr *= 0.8
        
        if it % 50 == 0:
            tracker.update(a)
    
    tracker.update(a)
    return a
```

```python
"""
Minimizes J(a) = 2*n*max(convolve(a,a))/(sum(a))^2 using:
1. Vectorized LP best-response with precomputed sparse index arrays
2. Adam-optimized LSE gradient descent with adaptive temperature
3. Multi-length multi-init search (256-1024)
4. Simulated annealing with LP refinement
5. Coordinate descent + final polishing
"""

import numpy as np
from scipy.optimize import linprog
from scipy.sparse import csr_matrix
import time

_spar_cache = {}

def get_sparsity(n):
    if n not in _spar_cache:
        ks = []
        js = []
        for k in range(2 * n - 1):
            j_start = max(0, k - n + 1)
            j_end = min(n - 1, k)
            ks.extend([k] * (j_end - j_start + 1))
            js.extend(range(j_start, j_end + 1))
        ks = np.array(ks, dtype=np.int32)
        js = np.array(js, dtype=np.int32)
        idx_diff = (ks.astype(np.int64) - js.astype(np.int64))
        _spar_cache[n] = (ks, js, idx_diff, len(ks))
    return _spar_cache[n]

def fast_conv(a):
    n = len(a)
    s = 1
    while s < 2 * n - 1:
        s <<= 1
    fa = np.fft.rfft(a, s)
    return np.fft.irfft(fa * fa, s)[:2 * n - 1]

def compute_J(a):
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))

def lp_best_response(a, M, time_limit=5):
    n = len(a)
    ks, js, idx_diff, nnz = get_sparsity(n)
    vals = a[idx_diff]
    A = csr_matrix((vals, (ks, js)), shape=(2 * n - 1, n))
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                options={'presolve': True, 'time_limit': time_limit})
    if r.success and r.x is not None:
        g = np.maximum(r.x, 0.0)
        if g.sum() > 1e-10:
            return g
    return None

def lp_optimize(a, max_iters, deadline, tracker, tighten=0.01):
    a = np.maximum(a.astype(np.float64), 1e-8)
    a /= a.sum()
    n = len(a)
    current_J = compute_J(a)
    tracker.update(a)

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        b = fast_conv(a)
        M = float(b.max())
        M_tight = M * (1.0 - tighten)
        g = lp_best_response(a, M_tight, time_limit=8)
        if g is None:
            g = lp_best_response(a, M * 1.001, time_limit=8)
        if g is None:
            break
        g = np.maximum(g, 0.0)
        gs = g.sum()
        if gs < 1e-10:
            break
        g /= gs

        best_a = None
        best_J = current_J
        progress = it / max(1, max_iters)
        base_t = max(0.05, 0.5 * (1.0 - progress))
        for t in [base_t, base_t * 1.5, base_t * 0.5, 0.8, 0.4, 0.15]:
            if t > 1.0:
                continue
            a_trial = (1.0 - t) * a + t * g
            a_trial = np.maximum(a_trial, 0.0)
            ss = a_trial.sum()
            if ss < 1e-10:
                continue
            a_trial /= ss
            J_trial = compute_J(a_trial)
            if J_trial < best_J - 1e-14:
                best_J = J_trial
                best_a = a_trial
        if best_a is None:
            break
        a = best_a
        current_J = best_J
        if it % 5 == 0:
            tracker.update(a)
    tracker.update(a)
    return a

def lse_adam_descent(a, max_iters, deadline, tracker, tau_init=80.0):
    a = np.maximum(a.astype(np.float64), 1e-8)
    a /= a.sum()
    n = len(a)
    current_J = compute_J(a)
    tracker.update(a)

    tau = tau_init
    lr = 0.05
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    m = np.zeros(n)
    v = np.zeros(n)
    t_count = 0
    stale = 0

    s = 1
    while s < 3 * n:
        s <<= 1

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        t_count += 1
        b = fast_conv(a)
        b_max = b.max()
        exp_b = np.exp(tau * (b - b_max))
        S_b = exp_b.sum()
        p = exp_b / S_b
        smoothed_max = b_max + np.log(S_b) / tau

        fp = np.fft.rfft(p, s)
        fa = np.fft.rfft(a, s)
        corr = np.fft.irfft(fp * np.conj(fa), s)
        grad_lse = 2.0 * corr[:n]

        grad = 2.0 * n * grad_lse - 4.0 * n * smoothed_max * a

        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad * grad)
        m_hat = m / (1 - beta1 ** t_count)
        v_hat = v / (1 - beta2 ** t_count)

        a_new = a - lr * m_hat / (np.sqrt(v_hat) + eps)
        a_new = np.maximum(a_new, 1e-8)
        a_new /= a_new.sum()

        J_new = compute_J(a_new)
        if J_new < current_J - 1e-15:
            a = a_new
            current_J = J_new
            stale = 0
            tau = min(500.0, tau * 1.01)
            lr = min(lr * 1.02, 0.5)
        else:
            stale += 1
            lr *= 0.7
            if stale > 200:
                break

        if it % 100 == 0:
            tracker.update(a)
    tracker.update(a)
    return a

def subgradient_step(a, max_iters, deadline, tracker):
    a = np.maximum(a.astype(np.float64), 1e-8)
    a /= a.sum()
    n = len(a)
    current_J = compute_J(a)
    tracker.update(a)
    lr = 0.3 / np.sqrt(n)
    rng = np.random.default_rng(12345)

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        b = fast_conv(a)
        M = float(b.max())
        active = np.where(b >= M - 1e-8)[0]

        grad = np.zeros(n)
        for k in active:
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            for j in range(js, je + 1):
                grad[j] += a[k - j]
        grad /= len(active)

        direction = grad - 2.0 * M * a
        a_new = a - lr * direction
        a_new = np.maximum(a_new, 1e-8)
        a_new /= a_new.sum()
        J_new = compute_J(a_new)
        if J_new < current_J - 1e-15:
            a = a_new
            current_J = J_new
            lr = min(lr * 1.01, 2.0 / np.sqrt(n))
        else:
            lr *= 0.75
        if it % 30 == 0:
            tracker.update(a)
    tracker.update(a)
    return a

def coordinate_descent(a, max_steps, deadline, tracker):
    a = a.copy().astype(np.float64)
    a /= a.sum()
    n = len(a)
    current_J = compute_J(a)
    tracker.update(a)
    rng = np.random.default_rng(42)

    for step in range(max_steps):
        if time.monotonic() > deadline:
            break
        j = rng.integers(0, n)
        improved = False
        for scale in [0.5, 1.5, 0.2, 2.0, 0.7, 1.3]:
            a_trial = a.copy()
            a_trial[j] *= scale
            a_trial = np.maximum(a_trial, 1e-10)
            ss = a_trial.sum()
            if ss < 1e-10:
                continue
            a_trial /= ss
            J_trial = compute_J(a_trial)
            if J_trial < current_J - 1e-14:
                a = a_trial
                current_J = J_trial
                improved = True
                break
        if step % 200 == 0:
            tracker.update(a)
    tracker.update(a)
    return a

def generate_inits(n, rng, parent):
    inits = []
    if parent is not None and parent.size > 0:
        t_old = np.linspace(0, 1, parent.size)
        t_new = np.linspace(0, 1, n)
        inits.append(np.maximum(np.interp(t_new, t_old, parent), 0.0))
        inits.append(np.maximum(np.interp(t_new, t_old, parent[::-1]), 0.0))

    inits.append(np.ones(n) / n)
    inits.append(rng.uniform(0.1, 1.0, n))

    # Flat spectrum
    for _ in range(3):
        half = n // 2 + 1
        angles = rng.uniform(0, 2 * np.pi, half)
        A = np.zeros(n, dtype=complex)
        A[:half] = np.exp(1j * angles[:half])
        for k in range(1, n - n // 2):
            A[k] = np.conj(A[n - k])
        a = np.fft.ifft(A).real
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s > 1e-10:
            inits.append(a / s)

    t = np.linspace(0, 1, n, endpoint=False)
    for freq in [2, 3, 5, 7, 11, 13, 17]:
        inits.append(1.0 + 0.4 * np.cos(2 * np.pi * freq * t + rng.uniform(0, 6.28)))
    inits.append(1.0 + 0.3 * np.cos(2 * np.pi * 3.7 * t) + 0.2 * np.cos(2 * np.pi * 7.1 * t))
    inits.append(0.5 + 0.5 * np.cos(np.pi * t) ** 2)
    inits.append(0.3 + 0.7 * (1.0 - np.abs(2 * t - 1.0)) ** 0.5)
    inits.append(0.5 * np.exp(-6 * (t - 0.3) ** 2) + 0.5 * np.exp(-6 * (t - 0.7) ** 2) + 0.05)
    inits.append(1.0 + 0.25 * np.sin(2 * np.pi * t) + 0.2 * np.sin(4 * np.pi * t)
                + 0.15 * np.sin(7 * np.pi * t) + 0.1 * np.sin(11 * np.pi * t))

    # Symmetric inits
    half_n = n // 2
    for _ in range(2):
        left = rng.uniform(0.3, 1.0, half_n)
        if n % 2 == 0:
            sym = np.concatenate([left, left[::-1]])
        else:
            sym = np.concatenate([left, [rng.uniform(0.3, 1.0)], left[::-1]])
        inits.append(sym)

    # Power-law
    inits.append((t + 0.05) ** rng.uniform(-1.0, 1.0))
    inits.append((1.0 - t + 0.05) ** rng.uniform(-1.0, 1.0))

    # Sparse + smooth
    sparse = np.zeros(n)
    n_spikes = max(5, n // 15)
    positions = rng.choice(n, size=n_spikes, replace=False)
    sparse[positions] = 1.0
    kernel_size = max(3, n // 25)
    kernel = np.ones(kernel_size) / kernel_size
    inits.append(np.convolve(sparse, kernel, mode='same') + 0.03)

    # Multi-frequency random
    n_coeff = min(n // 2 + 1, 15)
    coeffs = rng.normal(0, 1, n_coeff)
    sig = np.zeros(n)
    for i in range(n_coeff):
        sig += coeffs[i] * np.cos(2 * np.pi * i * t + rng.uniform(0, 6.28))
    inits.append(np.abs(sig) + 0.5)

    # Bump functions
    for _ in range(2):
        c1, c2 = sorted(rng.uniform(0.1, 0.9, 2))
        w = rng.uniform(0.1, 0.3)
        inits.append(np.exp(-((t - c1) / w) ** 2) + np.exp(-((t - c2) / w) ** 2) + 0.05)

    out = []
    for x in inits:
        x = np.maximum(np.asarray(x, dtype=np.float64), 0.001)
        x /= x.sum()
        out.append(x)
    return out

def resample_seq(src, dest_n):
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)

class ScoreTracker:
    def __init__(self):
        self.best = None
        self.best_score = float('inf')
        self.t_start = time.monotonic()

    def update(self, a):
        a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
        if a.size == 0 or a.sum() < 0.01:
            return
        if np.any(np.isnan(a)) or np.any(np.isinf(a)):
            return
        score = float(evaluate_sequence(a.tolist()))
        if np.isfinite(score) and score < self.best_score:
            self.best = a.copy()
            self.best_score = score
            elapsed = time.monotonic() - self.t_start
            print(f"  * J={score:.8f} n={a.size} t={elapsed:.1f}s", flush=True)

    def report(self, label=""):
        if self.best is not None:
            print(f"  {label} best: J={self.best_score:.8f} n={self.best.size}", flush=True)

def simulated_annealing(a, deadline, tracker, n_restarts=30, t0=0.12):
    rng = np.random.default_rng(777)
    n = a.size
    a = a.copy().astype(np.float64)
    current_J = compute_J(a)
    tracker.update(a)
    temp = t0

    for r in range(n_restarts):
        if time.monotonic() > deadline:
            break
        a_pert = a + rng.normal(0, temp, n)
        a_pert = np.maximum(a_pert, 1e-6)
        a_pert /= a_pert.sum()

        a_refined = lp_optimize(a_pert, max(3, 15 - r // 5), deadline, tracker, tighten=0.02)
        J = compute_J(a_refined)
        delta = J - current_J
        if delta < 0 or (temp > 1e-10 and rng.random() < np.exp(-delta / temp)):
            a = a_refined
            current_J = J
        temp *= 0.93
    tracker.update(a)
    return a

def propose_candidate(seed=42, budget_s=1500):
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + float(budget_s) - 15.0
    tracker = ScoreTracker()

    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        if p.ndim == 1 and p.size > 0 and np.all(np.isfinite(p)):
            parent = np.maximum(p, 0.0)
            tracker.update(parent)
            print(f"Parent: n={parent.size}, J={tracker.best_score:.8f}", flush=True)
    except (NameError, AttributeError):
        pass

    # === Phase 1: Multi-length LP optimization ===
    print(f"\n{'='*50} Phase 1: Multi-length LP {'='*50}", flush=True)
    lengths = [256, 320, 384, 512, 640]
    time_per_length = (deadline - time.monotonic() - 400) / len(lengths)

    for n in lengths:
        t_phase_start = time.monotonic()
        t_phase_end = min(deadline - 350, t_phase_start + time_per_length)
        if time.monotonic() >= t_phase_end - 30:
            continue
        print(f"\n  Phase1 n={n} budget={t_phase_end - time.monotonic():.1f}s", flush=True)

        inits = generate_inits(n, rng, parent)
        best_for_n = None
        best_J_for_n = float('inf')
        n_inits = len(inits)
        init_budget = max(8.0, (t_phase_end - time.monotonic()) / n_inits)

        for idx, a0 in enumerate(inits):
            if time.monotonic() > t_phase_end - 20:
                break
            a = a0.copy()
            # LP iterations
            n_lp_iters = max(8, int(50.0 / max(1, n / 256)))
            a = lp_optimize(a, n_lp_iters, t_phase_end - 40, tracker, tighten=0.01)

            # Subgradient
            if time.monotonic() < t_phase_end - 30:
                a = subgradient_step(a, max(30, int(5000 / n)), t_phase_end - 30, tracker)

            # LSE Adam
            if time.monotonic() < t_phase_end - 15:
                a = lse_adam_descent(a, max(200, int(5000 / n)), t_phase_end - 10, tracker)

            J = compute_J(a)
            if J < best_J_for_n:
                best_J_for_n = J
                best_for_n = a.copy()

        if best_for_n is not None:
            tracker.update(best_for_n)
            if parent is None:
                parent = best_for_n.copy()
            else:
                parent_resampled = resample_seq(parent, best_for_n.size)
                if best_J_for_n < compute_J(parent_resampled):
                    parent = best_for_n.copy()
            print(f"  Phase1 n={n} done: J={best_J_for_n:.8f} ({time.monotonic()-t_phase_start:.1f}s)", flush=True)

    # === Phase 2: Focused optimization on best ===
    if tracker.best is not None and time.monotonic() < deadline - 200:
        t0 = time.monotonic()
        print(f"\n{'='*50} Phase 2: Focus {'='*50}", flush=True)
        a = tracker.best.copy().astype(np.float64)

        # Multiple LP passes with varying tightening
        for tight in [0.005, 0.01, 0.015, 0.02, 0.025, 0.03, 0.04]:
            if time.monotonic() > deadline - 180:
                break
            a = lp_optimize(a, 25, deadline - 180, tracker, tighten=tight)

        # Subgradient
        if time.monotonic() < deadline - 120:
            a = subgradient_step(a, 3000, deadline - 120, tracker)

        # LSE Adam
        if time.monotonic() < deadline - 80:
            a = lse_adam_descent(a, 3000, deadline - 80, tracker)

        tracker.report(f"Phase2 ({time.monotonic()-t0:.1f}s)")

    # === Phase 3: Simulated annealing ===
    if tracker.best is not None and time.monotonic() < deadline - 120:
        t0 = time.monotonic()
        print(f"\n{'='*50} Phase 3: SA {'='*50}", flush=True)
        a = simulated_annealing(tracker.best, deadline - 100, tracker, n_restarts=35, t0=0.1)
        if time.monotonic() < deadline - 60:
            a = lp_optimize(a, 20, deadline - 60, tracker, tighten=0.01)
        if time.monotonic() < deadline - 30:
            a = lse_adam_descent(a, 1000, deadline - 30, tracker)
        tracker.report(f"Phase3 ({time.monotonic()-t0:.1f}s)")

    # === Phase 4: Coordinate descent + final polish ===
    if tracker.best is not None and time.monotonic() < deadline - 20:
        t0 = time.monotonic()
        print(f"\n{'='*50} Phase 4: Polish {'='*50}", flush=True)
        a = coordinate_descent(tracker.best, 2000, deadline - 15, tracker)
        if time.monotonic() < deadline - 8:
            a = lp_optimize(a, 10, deadline - 5, tracker, tighten=0.005)
        if time.monotonic() < deadline - 2:
            a = lse_adam_descent(a, 500, deadline - 1, tracker)
        tracker.report(f"Phase4 ({time.monotonic()-t0:.1f}s)")

    # Fallback
    if tracker.best is None:
        tracker.best = np.ones(256, dtype=np.float64)
        tracker.best_score = compute_J(tracker.best)
        tracker.update(tracker.best)

    print(f"\n{'='*50} FINAL {'='*50}", flush=True)
    print(f"J={tracker.best_score:.8f} n={tracker.best.size} time={time.monotonic()-tracker.t_start:.1f}s", flush=True)
    return [float(x) for x in tracker.best]
```
