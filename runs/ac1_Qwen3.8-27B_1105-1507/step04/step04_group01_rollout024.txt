    return A


def solve_lp_g(a, M):
    """Solve LP: max sum(g) s.t. (a*g)[k] <= M for all k, g >= 0."""
    n = len(a)

            g = np.maximum(res.x, 0)
            s = float(np.sum(g))

        pass
    return None

- Linear functions: `base_k + 2*a[k-i]*x` for k ≠ 2i
- One quadratic function: `base_{2i} + x^2` (if 2i is in the affected range)

The max of these divided by `(s0+x)^2`... The minimum of this max can occur at:
1. x = 0 (boundary)
2. A point where `base_k + delta_k(x) = max_unaffected` for some k (the affected max equals the unaffected max)
3. A point where two affected functions cross: `base_k1 + delta_k1(x) = base_k2 + delta_k2(x)`
4. A point where the derivative of a single affected function / `(s0+x)^2` is zero

For case 3 (two linear functions cross):
`base_k1 + c1*x = base_k2 + c2*x`
`x = (base_k2 - base_k1) / (c1 - c2)` (if c1 ≠ c2)

This gives O(n^2) intersection points, which is too many to check explicitly.

"""
Exact coordinate descent + LP alternating optimization for minimizing
2*n*max(conv(a,a))/(sum(a))^2. Uses ternary search for exact 1D coordinate
optimization, combined with LP-based best response for escaping local minima.
"""

import numpy as np
from scipy.optimize import linprog
from scipy.fft import rfft, irfft
import time


def compute_conv(a):
    """Compute linear convolution a*a via FFT."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    conv = irfft(fa * fa, size)
    return conv[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    conv = compute_conv(a)
    max_b = float(np.max(conv))
    sum_a = float(np.sum(a))
    if sum_a < 0.01:
        return np.inf
    return 2.0 * n * max_b / (sum_a * sum_a)


def fft_convolve(a, b):
    """Linear convolution of a and b using FFT."""
    n, m = len(a), len(b)
    size = 1
    while size < n + m - 1:
        size <<= 1
    fa = rfft(a, size)
    fb = rfft(b, size)
    return irfft(fa * fb, size)[:n + m - 1]


def build_toeplitz(a):
    """Build matrix A where (A@g)[k] = (convolve(a,g))[k]."""
    n = len(a)
    A = np.zeros((2 * n - 1, n), dtype=np.float64)
    for j in range(n):
        k_start = j
        k_end = min(j + n, 2 * n - 1)
        A[k_start:k_end, j] = a[:k_end - k_start]
    return A


def solve_lp_best_response(a, M, time_limit=20):
    """Solve LP: max sum(g) s.t. (a*g)[k] <= M for all k, g >= 0.
    Returns normalized g (sum=1) or None."""
    n = len(a)
    A = build_toeplitz(a)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': time_limit})
        if res.status == 0:
            g = np.maximum(res.x, 0)
            s = float(np.sum(g))
            if s > 1e-10:
                return g / s
    except Exception:
        pass
    return None


def cd_one_pass(a, rng):
    """
    One full pass of coordinate descent using ternary search.
    Processes coordinates in random order.
    Returns updated a.
    """
    n = len(a)
    a = a.copy().astype(np.float64)
    conv = compute_conv(a)
    s = float(np.sum(a))
    a = a / s
    conv = conv / (s * s)
    
    order = rng.permutation(n)
    
    for i in order:
        x_old = a[i]
        s0 = 1.0 - x_old
        
        if s0 < 1e-14:
            continue
        
        # Affected k range
        k_lo = i
        k_hi = min(i + n - 1, 2 * n - 2)
        
        # Base values (conv without a[i] contribution)
        k_range = np.arange(k_lo, k_hi + 1)
        j_vals = k_range - i  # j = k - i
        is_quad = (j_vals == i)  # True where k = 2i
        
        # Subtract a[i] contribution
        base_aff = conv[k_lo:k_hi + 1].copy()
        # For non-quad positions: subtract 2 * x_old * a[j]
        lin_mask = ~is_quad
        if np.any(lin_mask):
            base_aff[lin_mask] -= 2.0 * x_old * a[k_range[lin_mask] - i]
        # For quad position: subtract x_old^2
        if np.any(is_quad):
            quad_pos = np.where(is_quad)[0]
            for qp in quad_pos:
                base_aff[qp] -= x_old * x_old
        
        # Max of unaffected
        max_unaffected = 0.0
        if k_lo > 0:
            max_unaffected = max(max_unaffected, float(np.max(conv[:k_lo])))
        if k_hi < 2 * n - 2:
            max_unaffected = max(max_unaffected, float(np.max(conv[k_hi + 1:])))
        
        # Coefficients for ternary search
        c_aff = np.zeros(len(k_range))
        c_aff[lin_mask] = 2.0 * a[k_range[lin_mask] - i]
        
        # Ternary search for optimal x
        lo = 0.0
        hi = max(x_old * 4.0, s0 * 2.0, 4.0 / n)
        
        def f_of_x(x):
            # Compute max_conv
            delta = c_aff * x
            if np.any(is_quad):
                qp = np.where(is_quad)[0]
                for q in qp:
                    delta[q] = x * x
            aff_max = float(np.max(base_aff + delta))
            max_conv = max(max_unaffected, aff_max)
            denom = (s0 + x) * (s0 + x)
            if denom < 1e-28:
                return 1e18
            return 2.0 * n * max_conv / denom
        
        # Ternary search
        for _ in range(50):
            m1 = lo + (hi - lo) / 3.0
            m2 = hi - (hi - lo) / 3.0
            f1 = f_of_x(m1)
            f2 = f_of_x(m2)
            if f1 < f
                hi = m2
            else:
                lo = m1
        
        x_new = (lo + hi) / 2.0
        x_new = max(0.0, x_new)
        
        # Update conv
        delta = c_aff * x_new
        if np.any(is_quad):
            qp = np.where(is_quad)[0]
            for q in qp:
                delta[q] = x_new * x_new
        conv[k_lo:k_hi + 1] = base_aff + delta
        
        a[i] = x_new
    
    # Renormalize
    s_new = float(np.sum(a))
    if s_new > 1e-10:
        a = a / s_new
        conv = conv / (s_new * s_new)
    
    return a, conv

def resample(a, new_n):
    """Resample sequence to new length."""
    n = len(a)
    if n == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    return np.interp(x_new, x_old, a)
