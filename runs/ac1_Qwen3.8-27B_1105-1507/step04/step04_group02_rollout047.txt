```python
def project_simplex(v, s=1.0):
    """Project v onto the simplex {x >= 0, sum(x) = s}."""
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n+1) > (cssv - s))[0][-1]
    theta = (cssv[rho] - s) / (rho + 1.0)
    return np.maximum(v - theta, 0)
```

import numpy as np
from scipy.optimize import linprog
from scipy.sparse import coo_matrix, csr_matrix
from scipy.interpolate import interp1d
import time

def compute_conv(a):
    n = len(a)
    size = 1
    while size < 2*n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2*n-1]

def compute_obj(a):
    n = len(a)
    S = float(np.sum(a))
    if S < 1e-10:
        return np.inf
    conv = compute_conv(a)
    M = float(np.max(conv))
    return 2.0 * n * M / (S*S)

def project_simplex(v, s=1.0):
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n+1) > (cssv - s)
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - s) / (rho + 1.0)
    return np.maximum(v - theta, 0)

def get_active_set(a, frac=0.1):
    """Get the top fraction of convolution positions as active set."""
    conv = compute_conv(a)
    M = np.max(conv)
    threshold = M * (1 - frac)
    active = np.nonzero(conv >= threshold)[0]
    return active, M, conv

def subgradient(a, active_set):
    """Compute subgradient at a for given active set."""
    n = len(a)
    grad = np.zeros(n)
    for k in active_set:
        lo = max(0, k - n + 1)
        hi = min(k, n - 1)
        if lo > hi:
            continue
        j_lo = k - hi
        j_hi = k - lo
        grad[lo:hi+1] += 2.0 * a[j_lo:j_hi+1][::-1]
    grad /= len(active_set)
    return grad
