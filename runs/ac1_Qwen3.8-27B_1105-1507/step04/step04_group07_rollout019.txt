import numpy as np
from scipy.optimize import linprog
from scipy.sparse import csr_matrix

# Cache for sparse matrix structure
_spar_cache = {}

def get_sparsity(n):
    """Get sparse matrix structure for convolution constraint."""
    if n not in _spar_cache:
        rows = []
        cols = []
        for k in range(2 * n - 1):
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            rows.extend([k] * (je - js + 1))
            cols.extend(range(js, je + 1))
        _spar_cache[n] = (np.array(rows, dtype=np.int32), 
                         np.array(cols, dtype=np.int32), 
                         np.array(rows, dtype=np.int32),
                         len(rows))
    return _spar_cache[n]

def fast_conv(a):
    """Fast convolution using FFT."""
    n = len(a)
    s = 1
    while s < 2 * n - 1:
        s <<= 1
    fa = np.fft.rfft(a, s)
    return np.fft.irfft(fa * fa, s)[:2 * n - 1]

def compute_J(a):
    """Compute the objective value."""
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))

def build_A(a, n):
    """Build the sparse matrix A for the LP: A @ g = conv(a, g)."""
    rows, cols, rows_dup, nnz = get_sparsity(n)
    # For row k, col j: A[k,j] = a[k-j] if 0 <= k-j < n
    vals = np.empty(nnz)
    for i in range(nnz):
        k = rows[i]
        j = cols[i]
        idx = k - j
        vals[i] = a[idx]
    return csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))

def lp_best_response(a, M, x0=None, time_limit=5):
    """Solve LP: max sum(g) s.t. conv(a,g) <= M, g >= 0."""
    n = len(a)
    A = build_A(a, n)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    opts = {'presolve': True, 'time_limit': time_limit}
    r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs', options=opts)
    if r.success and r.x is not None:
        g = np.maximum(r.x, 0.0)
        if g.sum() > 1e-10:
            return g
    return None

def lp_best_response_adaptive(a, M, max_tries=5, time_limit=5):
    """Try decreasing M until LP is feasible."""
    for try_i in range(max_tries):
        M_try = M * (1.0 - 0.01 * (try_i + 1))
        if M_try < M * 0.9:
            break
        g = lp_best_response(a, M_try, time_limit=time_limit)
        if g is not None:
            return g, M_try
    # Fallback: use M itself
    g = lp_best_response(a, M, time_limit=time_limit)
    if g is not None:
        return g, M
    return None, M

def subgradient_descent(a, max_iters, deadline, tracker):
    """Subgradient descent with momentum on the exact max."""
    a = np.maximum(a.astype(np.float64), 1e-10)
    a = a / a.sum()
    n = len(a)
    current_J = compute_J(a)
    tracker.update(a)
    
    lr = 1.0 / n
    momentum = 0.8
    v = np.zeros(n)
    rng = np.random.default_rng(12345)
    stall_count = 0
    
    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        
        b = fast_conv(a)
        M = float(b.max())
        
        # Find active set (indices where b is close to max)
        tol = 1e-8 * M
        active = np.where(b >= M - tol)[0]
        
        # Compute subgradient
        grad = np.zeros(n)
        for k in active:
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            for j in range(js, je + 1):
                grad[j] += a[k - j]
        grad /= len(active)
        
        # Full gradient direction for J (after normalization s=1):
        # dJ/da_j ∝ 2n * grad_j - 4n*M*a_j
        # Simplified direction: grad - 2*M*a
        direction = grad - 2.0 * M * a
        
        # Momentum update
        v = momentum * v + direction
        a_new = a - lr * v
        a_new = np.maximum(a_new, 1e-10)
        a_new = a_new / a_new.sum()
        
        J_new = compute_J(a_new)
        if J_new < current_J - 1e-16:
            a = a_new
            current_J = J_new
            # Adaptive learning rate
            if stall_count > 0:
                lr = min(lr * 1.05, 5.0 / n)
            stall_count = 0
        else:
            lr *= 0.8
            v *= 0.5
            stall_count += 1
            if stall_count > 30:
                break
        
        if it % 100 == 0:
            tracker.update(a)
    
    tracker.update(a)
    return a

def lse_descent(a, max_iters, deadline, tracker):
    """Log-sum-exp smoothing gradient descent."""
    a = np.maximum(a.astype(np.float64), 1e-8)
    a = a / a.sum()
    n = len(a)
    current_J = compute_J(a)
    tracker.update(a)
    
    tau = 20.0
    lr = 1.0 / (n * max(1, np.log(n)))
    momentum = 0.9
    v = np.zeros(n)
    stall_count = 0
    
    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        
        b = fast_conv(a)
        b_max = b.max()
        
        # LSE gradient
        exp_b = np.exp(tau * (b - b_max))
        p = exp_b / exp_b.sum()
        
        # grad of LSE w.r.t. a: 2 * (cross-correlation of p and a)
        # (p * a) in some sense - actually it's the gradient of sum_k p_k * b_k
        # d(sum_k p_k * b_k)/da_j = 2 * sum_k p_k * a_{k-j} (for the self-conv part)
        # But p depends on a too. For simplicity, treat p as fixed (one step).
        s = 1
        while s < 3 * n:
            s <<= 1
        fp = np.fft.rfft(p, s)
        fa = np.fft.rfft(a, s)
        # We need: for each j, sum_k p_k * a_{k-j}
        # This is a cross-correlation
        corr = np.fft.irfft(fp * np.conj(fa), s)
        grad = 2.0 * corr[:n]
        
        v = momentum * v + (1.0 - momentum) * grad
        a_new = a - lr * v
        a_new = np.maximum(a_new, 1e-8)
        a_new = a_new / a_new.sum()
        
        J_new = compute_J(a_new)
        if J_new < current_J - 1e-16:
            a = a_new
            current_J = J_new
            tau = min(200.0, tau * 1.02)
            lr = 1.0 / (n * max(1, np.log(tau)))
            stall_count = 0
        else:
            lr *= 0.7
            v *= 0.5
            tau = max(5.0, tau * 0.95)
            stall_count += 1
            if stall_count > 50:
                break
        
        if it % 200 == 0:
            tracker.update(a)
    
    tracker.update(a)
    return a

def flat_spectrum_init(n, rng):
    """Generate sequence with approximately flat power spectrum."""
    half = n // 2 + 1
    angles = rng.uniform(0, 2 * np.pi, half)
    A = np.zeros(n, dtype=complex)
    A[:half] = np.exp(1j * angles[:half])
    for k in range(1, n - n // 2):
        A[k] = np.conj(A[n - k])
    a = np.fft.ifft(A).real
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-10:
        return np.ones(n) / n
    return a / s

def barker_like_init(n, rng):
    """Initialize based on Barker sequence concept (low autocorrelation)."""
    # Start with random ±1 sequence, then smooth
    signs = rng.choice([-1, 1], n)
    # Smooth with a window
    w = max(3, n // 32)
    kernel = np.hanning(w)
    kernel /= kernel.sum()
    a = np.convolve(signs.astype(float) + 1.5, kernel, mode='same')
    a = np.maximum(a, 0.01)
    return a / a.sum()

def random_phase_init(n, rng, bandwidth=None):

    bandwidth = min(bandwidth, n // 2 + 1)
    A = np.zeros(n, dtype=complex)

        else:
            A[k] = 0

    a = np.maximum(a, 0.0)
    s = a.sum()

        return np.ones(n) / n
    return a / s

"""
Minimizes J(a) = 2*n*max(convolve(a,a))/(sum(a))^2 using:
1. LP best-response with adaptive tightening and feasibility tracking
2. Exact subgradient descent with momentum and adaptive step size
3. LSE (log-sum-exp) smooth optimization with ramp-up
4. Multi-scale exploration (small n → large n)
5. Diverse spectral and structural initializations
6. Randomized restarts to escape local minima

import numpy as np
from scipy.optimize import linprog
from scipy.sparse import csr_matrix
import time

# Cache for sparse matrix structure
_spar_cache = {}

def get_sparsity(n):
    if n not in _spar_cache:
        rows = []
        cols = []
        for k in range(2 * n - 1):
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            rows.extend([k] * (je - js + 1))
            cols.extend(range(js, je + 1))
        _spar_cache[n] = (np.array(rows, dtype=np.int32),
                          np.array(cols, dtype=np.int32), len(rows))
    return _spar_cache[n]

def fast_conv(a):
    n = len(a)
    s = 1
    while s < 2 * n - 1:
        s <<= 1
    fa = np.fft.rfft(a, s)
    return np.fft.irfft(fa * fa, s)[:2 * n - 1]

def compute_J(a):
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))

def build_A(a, n):
    rows, cols, nnz = get_sparsity(n)
    vals = np.empty(nnz, dtype=np.float64)
    for i in range(nnz):
        k = rows[i]
        j = cols[i]
        vals[i] = a[k - j]
    return csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))

def lp_solve(a, M, time_limit=4):
    n = len(a)
    A = build_A(a, n)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    opts = {'presolve': True, 'time_limit': time_limit}
    r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs', options=opts)
    if r.success and r.x is not None:
        g = np.maximum(r.x, 0.0)
        if g.sum() > 1e-10:
            return g
    return None

def run_lp_opt(a, max_iters, deadline, tracker, min_tight=0.002, max_tight=0.04):

            M_try = M * (1.0 - tight_frac)
            if M_try <= 0:
                continue
            g = lp_solve(a, M_try, time_limit=3)
            if g is not None and np.any(g > 0):
                break
            g = None
        
        if g is None:
            g = lp_solve(a, M * 1.005, time_limit=5)
            if g is None:

        gs = g.sum()
        if gs < 1e-10:

        g = g / gs

        best_J = current_J

def subgradient_descent(a, max_iters, deadline, tracker):
    a = np.maximum(a.astype(np.float64), 1e-10)
    a = a / a.sum()
    n = len(a)
    current_J = compute_J(a)
    tracker.update(a)
    
    lr = 2.0 / n
    momentum = 0.85
    v = np.zeros(n)
    stall = 0
    
    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        b = fast_conv(a)
        M = float(b.max())
        
        # Active set
        tol = max(1e-10, 1e-7 * M)
        active = np.where(b >= M - tol)[0]
        
        # Subgradient of max(b)
        grad = np.zeros(n)
        for k in active:
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            for j in range(js, je + 1):
                grad[j] += a[k - j]
        grad /= len(active)
        
        # Direction: minimize J ≈ 2n*max_b - 4n*max_b*sum(a)/sum(a)^2 * a
        # With sum(a)=1: direction = grad - 2*M*a
        direction = grad - 2.0 * M * a
        
        v = momentum * v + direction
        a_new = a - lr * v
        a_new = np.maximum(a_new, 1e-10)
        ss = a_new.sum()
        if ss < 1e-10:
            break
        a_new = a_new / ss
        
        J_new = compute_J(a_new)
        if J_new < current_J - 1e-16:
            a = a_new
            current_J = J_new
            stall = 0
            lr = min(lr * 1.02, 10.0 / n)
        else:
            lr *= 0.85
            v *= 0.4
            stall += 1
            if stall > 40:
                break
        
        if it % 200 == 0:
            tracker.update(a)
    
    tracker.update(a)
    return a

def lse_descent(a, max_iters, deadline, tracker):
    a = np.maximum(a.astype(np.float64), 1e-8)
    a = a / a.sum()
    n = len(a)
    current_J = compute_J(a)
    tracker.update(a)
    
    tau = 15.0
    lr = 2.0 / (n * max(1.0, np.log2(n)))
    momentum = 0.9
    v = np.zeros(n)
    stall = 0
    
    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        b = fast_conv(a)
        b_max = b.max()
        
        exp_b = np.exp(tau * (b - b_max))
        p = exp_b / exp_b.sum()
        
        # Gradient of sum_k p_k * b_k w.r.t. a (treating p as fixed)
        s = 1
        while s < 3 * n:
            s <<= 1
        fp = np.fft.rfft(p, s)
        fa = np.fft.rfft(a, s)
        corr = np.fft.irfft(fp * np.conj(fa), s)
        grad = 2.0 * corr[:n]
        
        v = momentum * v + (1.0 - momentum) * grad
        a_new = a - lr * v
        a_new = np.maximum(a_new, 1e-8)
        ss = a_new.sum()
        if ss < 1e-10:
            break
        a_new = a_new / ss
        
        J_new = compute_J(a_new)
        if J_new < current_J - 1e-16:
            a = a_new
            current_J = J_new
            stall = 0
            tau = min(300.0, tau * 1.03)
            lr = 2.0 / (n * max(1.0, np.log2(tau)))
        else:
            lr *= 0.75
            v *= 0.4
            tau = max(5.0, tau * 0.97)
            stall += 1
            if stall > 60:
                break
        
        if it % 300 == 0:
            tracker.update(a)
    
    tracker.update(a)
    return a

def flat_spectrum_init(n, rng):
    half = n // 2 + 1
    angles = rng.uniform(0, 2 * np.pi, half)
    A = np.zeros(n, dtype=complex)
    A[:half] = np.exp(1j * angles[:half])
    for k in range(1, n - n // 2):
        A[k] = np.conj(A[n - k])
    a = np.fft.ifft(A).real
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-10:
        return np.ones(n) / n
    return a / s

def barker_like_init(n, rng):
    signs = rng.choice([-1, 1], n)
    w = max(3, n // 24)
    kernel = np.hanning(w)
    kernel = kernel / max(1e-10, kernel.sum())
    a = np.convolve(signs.astype(float) + 1.8, kernel, mode='same')
    a = np.maximum(a, 0.01)
    return a / a.sum()

def random_phase_init(n, rng, bandwidth=None):
    if bandwidth is None:
        bandwidth = n // 2 + 1
    bandwidth = min(bandwidth, n // 2 + 1)
    A = np.zeros(n, dtype=complex)
    phases = rng.uniform(0, 2 * np.pi, bandwidth)
    A[:bandwidth] = np.exp(1j * phases[:bandwidth])
    for k in range(1, min(n - n // 2, bandwidth)):
        A[k] = np.conj(A[n - k])
    a = np.fft.ifft(A).real
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-10:
        return np.ones(n) / n
    return a / s

def generate_inits(n, rng, parent=None):
    inits = []
    
    if parent is not None and parent.size > 0:
        t_old = np.linspace(0, 1, parent.size)
        t_new = np.linspace(0, 1, n)
        inits.append(np.maximum(np.interp(t_new, t_old, parent), 0.001))
        inits.append(np.maximum(np.interp(t_new, t_old, parent[::-1]), 0.001))
    
    inits.append(np.ones(n))
    inits.append(rng.uniform(0.1, 1.0, n))
    inits.append(flat_spectrum_init(n, rng))
    inits.append(flat_spectrum_init(n, rng))
    inits.append(random_phase_init(n, rng, bandwidth=max(4, n // 8)))
    inits.append(random_phase_init(n, rng, bandwidth=max(4, n // 4)))
    inits.append(barker_like_init(n, rng))
    inits.append(barker_like_init(n, rng))
    
    t = np.linspace(0, 1, n, endpoint=False)
    for freq in [2, 3, 5, 7, 11]:
        inits.append(np.maximum(1.0 + 0.4 * np.cos(2 * np.pi * freq * t + rng.uniform(0, 6.28)), 0.01))
    
    inits.append(np.maximum(1.0 + 0.25 * np.sin(2 * np.pi * t) + 0.2 * np.sin(5 * np.pi * t)
                     + 0.15 * np.sin(9 * np.pi * t) + 0.1 * np.sin(13 * np.pi * t), 0.01))
    
    inits.append(np.maximum(0.4 * np.exp(-10 * (t - 0.25)**2) + 0.4 * np.exp(-10 * (t - 0.75)**2)
                     + 0.3 * np.exp(-6 * (t - 0.5)**2) + 0.05, 0.01))
    inits.append(np.maximum(1.0 - 0.7 * np.abs(2*t - 1.0)**1.5, 0.01))
    inits.append(np.maximum((t + 0.05) ** 0.5 + 0.2, 0.01))
    inits.append(np.maximum((1.0 - t + 0.05) ** 0.5 + 0.2, 0.01))
    
    # Sparse + smooth
    sparse = np.zeros(n)
    n_spikes = max(10, n // 12)
    positions = rng.choice(n, size=n_spikes, replace=False)
    sparse[positions] = 1.0
    kernel_size = max(5, n // 16)
    kernel = np.hanning(kernel_size)
    kernel = kernel / max(1e-10, kernel.sum())
    inits.append(np.maximum(np.convolve(sparse, kernel, mode='same'), 0.01))
    
    # Multi-bump
    n_bumps = rng.integers(3, 7)
    centers = np.sort(rng.uniform(0.1, 0.9, n_bumps))
    widths = rng.uniform(0.03, 0.1, n_bumps)
    sig = np.zeros(n)
    for c, w in zip(centers, widths):
        sig += np.exp(-((t - c) / w)**2)
    inits.append(np.maximum(sig + 0.1, 0.01))
    
    result = []
    for x in inits:
        x = np.maximum(x, 0.001).astype(np.float64)
        x = x / x.sum()
        result.append(x)
    return result

def resample(src, dest_n):
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    return np.maximum(np.interp(t_new, t_old, src), 0.001)

class ScoreTracker:
    def __init__(self):
        self.best = None
        self.best_score = float('inf')
        self.t_start = time.monotonic()
    
    def update(self, a):
        a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
        if a.size == 0 or a.sum() < 0.01:
            return
        score = float(evaluate_sequence(a.tolist()))
        if np.isfinite(score) and score < self.best_score:
            self.best = a.copy()
            self.best_score = score
            elapsed = time.monotonic() - self.t_start
            print(f"  * NEW BEST: J={score:.8f} n={a.size} t={elapsed:.1f}s", flush=True)
    
    def report(self, label=""):
        if self.best is not None:
            print(f"  {label} current best: J={self.best_score:.8f} n={self.best.size} t={time.monotonic()-self.t_start:.1f}s", flush=True)
