"""
Optimization for minimizing 2*n*max(conv(a,a))/(sum(a))^2 using heavy LP-based
alternating optimization, direct peak shaving, and softmax L-BFGS-B refinement.
Key innovation: aggressive use of the convex LP subproblem (maximize sum(g) s.t.
conv(a,g) <= M, g>=0) combined with targeted peak reduction and smart restarts.
"""

import numpy as np
from scipy.optimize import minimize, linprog
from scipy.fft import rfft, irfft
import time


def compute_conv(a):
    """Compute linear convolution a*a using FFT. Returns array of length 2n-1."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    conv = irfft(fa * fa, size)
    return conv[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    conv = compute_conv(a)
    max_b = float(np.max(conv))
    sum_a = float(np.sum(a))
    if sum_a < 0.01:
        return np.inf
    return 2.0 * n * max_b / (sum_a * sum_a)


def normalize(a):
    """Normalize to sum=1, ensure non-negative."""
    a = np.maximum(a, 0)
    s = np.sum(a)
    if s < 1e-15:
        return np.ones(len(a)) / len(a)
    return a / s


def build_toeplitz(a):
    """Build matrix A where (A@g)[k] = sum_j a[max(0,k-j):k-j+n overlap] * g[j]."""
    n = len(a)
    A = np.zeros((2 * n - 1, n), dtype=np.float64)
    for j in range(n):
        # g[j] contributes to conv[k] for k = j, j+1, ..., j+n-1
        # with coefficient a[k-j]
        for k in range(j, min(j + n, 2 * n - 1)):
            A[k, j] = a[k - j]
    return A


def build_toeplitz_fast(a):
    """More efficient Toeplitz construction."""
    n = len(a)
    A = np.zeros((2 * n - 1, n), dtype=np.float64)
    for j in range(n):
        A[j:j+n, j] = a[:n]
    return A


def solve_lp_g(a, M, time_limit=30):
    """
    Solve LP: maximize sum(g) subject to (a*g)[k] <= M for all k, g >= 0.
    Returns (g_normalized, sum_g) or (None, None).
    """
    n = len(a)
    A = build_toeplitz_fast(a)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': time_limit,
                               'dual_feasibility_tolerance': 1e-9,
                               'primal_feasibility_tolerance': 1e-9})
        if res.status == 0:
            g = np.maximum(res.x, 0)
            s = float(np.sum(g))
            if s > 1e-10:
                return g, s
    except Exception:
        pass
    return None, None


def lp_alternating_step(a, M_override=None, time_limit=20):
    """
    One step of LP alternating. Returns (blended_a, improved, sum_g).
    """
    n = len(a)
    conv = compute_conv(a)
    if M_override is None:
        M = float(np.max(conv))
    else:
        M = M_override
    sum_a = float(np.sum(a))

    g, sum_g = solve_lp_g(a, M, time_limit=time_limit)
    if g is None:
        return a, False, 0.0

    # If sum_g > sum_a, we have an improvement direction
    if sum_g <= sum_a * (1 + 1e-10):
        return a, False, sum_g

    # Blend: find best t
    a_norm = normalize(a)
    g_norm = normalize(g)
    best_obj = compute_obj(a_norm)
    best_t = 0.0
    best_a = a_norm.copy()

    for t in [0.1, 0.2, 0.3, 0.5, 0.7, 1.0]:
        a_try = (1.0 - t) * a_norm + t * g_norm
        a_try = normalize(a_try)
        obj_try = compute_obj(a_try)
        if obj_try < best_obj:
            best_obj = obj_try
            best_t = t
            best_a = a_try.copy()

    return best_a, best_t > 0, sum_g


def peak_shave(a, num_peaks=5, alpha=0.1):
    """
    Direct peak shaving: reduce values of a that contribute most to the peak.
    """
    n = len(a)
    a = normalize(a.copy())
    conv = compute_conv(a)

    # Find top peaks
    peak_indices = np.argsort(conv)[::-1][:num_peaks]

    a_new = a.copy()
    for k_star in peak_indices:
        # Contribution of a[i] to conv[k_star] is a[k_star - i] for valid i
        c = np.zeros(n)
        for i in range(n):
            j = k_star - i
            if 0 <= j < n:
                c[i] = a[j]
        # Reduce a where contribution is high, increase where low
        c_centered = c - np.mean(c)
        a_new = a_new - alpha * c_centered / (np.max(np.abs(c_centered)) + 1e-10)

    return normalize(a_new)


def softmax(z):
    """Numerically stable softmax."""
    z = z - np.max(z)
    e = np.exp(z)
    return e / np.sum(e)


def soft_max_obj_grad(z, tau):
    """
    Compute soft-max of conv(softmax(z), softmax(z)) and gradient w.r.t. z.
    """
    n = len(z)
    a = softmax(z)
    conv = compute_conv(a)
    cmax = float(np.max(conv))
    shifted = np.clip((conv - cmax) / tau, -500.0, 500.0)
    exp_conv = np.exp(shifted)
    Z = float(np.sum(exp_conv))
    soft_max_val = cmax + tau * np.log(Z)
    w = exp_conv / Z

    # grad_a[i] = 2 * sum_{j: i+j valid} w[i+j] * a[j]
    # Use convolution: grad_a[i] = 2 * (w_rev * a)[n-1+i] where w_rev = w[::-1]
    w_rev = w[::-1]
    size = 1
    while size < n + len(w) - 1:
        size <<= 1
    fa = rfft(a, size)
    fw = rfft(w_rev, size)
    conv_fw = irfft(fa * fw, size)
    # conv_fw[k] = sum_j a[j] * w_rev[k-j] = sum_j a[j] * w[len(w)-1-k+j]
    # We want: grad_a[i] = 2 * sum_j w[i+j] * a[j]
    # w_rev[m] = w[len(w)-1-m], so w[i+j] = w_rev[len(w)-1-i-j]
    # conv_fw[k] = sum_j a[j] * w_rev[k-j], set k-j = len(w)-1-i-j => k = len(w)-1-i
    # So grad_a[i] = 2 * conv_fw[len(w)-1-i]
    # len(w) = 2n-1, so len(w)-1 = 2n-2
    grad_a = 2.0 * conv_fw[2*n-2-n+1:2*n-2+1][::-1]
    # Actually let me be more careful:
    # conv_fw[m] = sum_{j=0}^{n-1} a[j] * w_rev[m-j]
    # w_rev[q] = w[2n-2-q]
    # conv_fw[m] = sum_j a[j] * w[2n-2-m+j]
    # We want grad_a[i] = 2 * sum_j w[i+j] * a[j]
    # So set 2n-2-m = i => m = 2n-2-i
    grad_a = 2.0 * np.array([conv_fw[2*n-2-i] for i in range(n)])
    # More efficient:
    grad_a = 2.0 * conv_fw[2*n-2:2*n-2-n:-1]

    mean_grad = float(np.dot(a, grad_a))
    grad_z = a * (grad_a - mean_grad)

    return soft_max_val, grad_z


def lbfgs_refine(a, time_budget=15.0, max_iter=200):
    """L-BFGS-B refinement using softmax parametrization."""
    n = len(a)
    a = normalize(a)
    a = np.maximum(a, 1e-30)
    z = np.log(a)

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    tau_list = [5.0/n, 2.0/n, 1.0/n, 0.5/n, 0.2/n]

    for tau in tau_list:
        if time.monotonic() - start > time_budget:
            break

        def func_and_grad(z_val):
            return soft_max_obj_grad(z_val, tau)

        try:
            result = minimize(func_and_grad, z, jac=True, method='L-BFGS-B',
                              options={'maxiter': max_iter, 'ftol': 1e-13})
            z_new = result.x
        except Exception:
            continue

        a_new = softmax(z_new)
        a_new = normalize(a_new)
        obj = compute_obj(a_new)

        if obj < best_obj:
            best_obj = obj
            best_a = a_new.copy()
            z = z_new

    return best_a, best_obj


def random_init(n, seed):
    """Generate diverse random initializations."""
    rng = np.random.default_rng(seed)
    k = np.arange(n)
    choice = rng.integers(0, 8)

    if choice == 0:
        f = rng.uniform(0.5, 4.0)
        phase = rng.uniform(0, 2 * np.pi)
        a = 1.0 + 0.7 * np.cos(2 * np.pi * f * k / n + phase)
    elif choice == 1:
        f1 = rng.uniform(0.3, 2.0)
        f2 = rng.uniform(2.0, 6.0)
        a = 1.0 + 0.3 * np.cos(2 * np.pi * f1 * k / n) + 0.3 * np.cos(2 * np.pi * f2 * k / n)
    elif choice == 2:
        a = rng.random(n)
        win = max(3, n // 30)
        kernel = np.ones(win) / win
        a = np.convolve(a, kernel, mode='same')
    elif choice == 3:
        a = np.ones(n)
        for _ in range(rng.integers(1, 4)):
            f = rng.uniform(0.5, 5.0)
            amp = rng.uniform(0.1, 0.5)
            a += amp * np.cos(2 * np.pi * f * k / n + rng.uniform(0, 2 * np.pi))
    elif choice == 4:
        # Two-tone
        f1 = rng.uniform(1.0, 3.0)
        f2 = rng.uniform(4.0, 8.0)
        a = 1.0 + 0.4 * np.cos(2 * np.pi * f1 * k / n) + 0.2 * np.cos(2 * np.pi * f2 * k / n)
    elif choice == 5:
        # Block structure
        nblocks = rng.integers(2, 8)
        a = np.ones(n)
        for b in range(nblocks):
            start = b * n // nblocks
            end = (b + 1) * n // nblocks
            a[start:end] = rng.uniform(0.3, 1.5)
    elif choice == 6:
        # Triangular
        center = n // 2
        a = 1.0 - 0.5 * np.abs(k - center) / (n / 2)
    else:
        a = 1.0 + 0.3 * rng.normal(0, 1, n)

    a = np.maximum(a, 0.01)
    return normalize(a)


def resample_sequence(a, new_n):
    """Resample sequence to new length via interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return normalize(a_new)


def try_parent():
    """Try to load parent sequence."""
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64)
        parent = np.clip(parent, 0, 1000.0)
        if parent.size > 0 and np.isfinite(parent).all():
            obj = compute_obj(parent)
            print(f"Parent (n={parent.size}): obj={obj:.6f}")
            return normalize(parent), obj
    except Exception:
        pass
    return None, np.inf


def intensive_lp_optimize(a, time_budget=60.0, max_iters=200):
    """
    Intensive LP-based optimization with peak shaving restarts.
    This is the main workhorse.
    """
    n = len(a)
    a = normalize(a.copy())
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    stall_count = 0

    print(f"  LP opt start: obj={best_obj:.6f}, n={n}")

    for it in range(max_iters):
        if time.monotonic() - start > time_budget - 2:
            break

        remaining = time_budget - (time.monotonic() - start)
        if remaining < 5:
            break

        # LP step
        a_new, improved, sum_g = lp_alternating_step(a, time_limit=min(15, remaining - 3))

        if improved:
            best_obj = compute_obj(a_new)
            if best_obj < best_obj:  # always true by construction
                pass
            obj_new = compute_obj(a_new)
            if obj_new < best_obj:
                best_obj = obj_new
                best_a = a_new.copy()
                a = a_new
                stall_count = 0
            else:
                # Try smaller blend
                a = a  # keep
                stall_count += 1
        else:
            stall_count += 1

        # If stalled, try peak shaving
        if stall_count >= 3:
            stall_count = 0
            a_shaved = peak_shave(a, num_peaks=3, alpha=0.05)
            obj_shaved = compute_obj(a_shaved)
            if obj_shaved < best_obj:
                best_obj = obj_shaved
                best_a = a_shaved.copy()
                a = a_shaved
            else:
                # Random perturbation
                rng = np.random.default_rng(it * 7919 + 1)
                scale = rng.uniform(0.02, 0.08)
                a_pert = a * np.exp(rng.normal(0, scale, n))
                a = normalize(a_pert)

        # Every 10 iterations, try a different peak shaving
        if it % 10 == 5:
            a_shaved = peak_shave(a, num_peaks=min(10, n // 10), alpha=0.08)
            obj_shaved = compute_obj(a_shaved)
            if obj_shaved < best_obj:
                best_obj = obj_shaved
                best_a = a_shaved.copy()
                a = a_shaved

    print(f"  LP opt done: best_obj={best_obj:.6f}")
    return best_a, best_obj


def hybrid_optimize(a, time_budget=30.0):
    """Combined LP + L-BFGS + peak shaving optimization."""
    n = len(a)
    a = normalize(a.copy())
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    round_num = 0

    while time.monotonic() - start < time_budget - 5:
        round_num += 1
        remaining = time_budget - (time.monotonic() - start)
        if remaining < 8:
            break

        # LP step (most powerful)
        if remaining > 20:
            a_new, improved, _ = lp_alternating_step(a, time_limit=min(15, remaining - 10))
            obj_new = compute_obj(a_new)
            if obj_new < best_obj:
                best_obj = obj_new
                best_a = a_new.copy()
                a = a_new
                continue

        # L-BFGS refinement
        if remaining > 15:
            a_new, obj_new = lbfgs_refine(a, time_budget=min(10, remaining - 8), max_iter=100)
            if obj_new < best_obj:
                best_obj = obj_new
                best_a = a_new.copy()
                a = a_new
                continue

        # Peak shaving
        if remaining > 5:
            a_shaved = peak_shave(a, num_peaks=5, alpha=0.05)
            obj_shaved = compute_obj(a_shaved)
            if obj_shaved < best_obj:
                best_obj = obj_shaved
                best_a = a_shaved.copy()
                a = a_shaved

        # If nothing worked, perturb
        rng = np.random.default_rng(round_num * 31 + 7)
        scale = rng.uniform(0.02, 0.1)
        a_pert = a * np.exp(rng.normal(0, scale, n))
        a = normalize(a_pert)

    return best_a, best_obj


def propose_candidate(seed=42, budget_s=1500):
    """
    Main search function. Uses aggressive LP-based optimization combined with
    peak shaving, L-BFGS-B refinement, and smart restarts to minimize
    2*n*max(conv(a,a))/(sum(a))^2.
    """
    start_time = time.monotonic()
    deadline = start_time + float(budget_s) - 30.0

    best_seq = None
    best_score = np.inf

    # Load parent
    parent, parent_obj = try_parent()
    if parent is not None and parent_obj < best_score:
        best_score = parent_obj
        best_seq = parent.copy()

    # N values to explore (centered around parent if available)
    if best_seq is not None:
        np_ = len(best_seq)
        n_values = sorted(set([np_] + list(range(max(100, np_ - 100), min(900, np_ + 100), 25))))
    else:
        n_values = [150, 200, 300, 400, 500, 600, 700]

    # Filter to reasonable range
    n_values = [n for n in n_values if 80 <= n <= 1000]

    print(f"N values to explore: {n_values}")
    print(f"Initial best: {best_score:.6f}")
    print(f"Time budget: {budget_s}s, deadline in {deadline - start_time:.0f}s")

    # Phase 1: Quick exploration
    print(f"\n{'='*60}")
    print(f"PHASE 1: Quick Exploration")
    print(f"{'='*60}")

    candidates = []  # (a, obj, n)

    for n in n_values:
        if time.monotonic() > deadline - 200:
            break
        remaining = deadline - time.monotonic()
        if remaining < 15:
            break

        # Time per n value
        alloc = min(25.0, remaining * 0.08)
        alloc = max(8.0, alloc)

        n_inits = max(1, min(3, int(alloc / 8)))

        for init_idx in range(n_inits):
            if time.monotonic() > deadline - 200:
                break
            if deadline - time.monotonic() < 8:
                break

            # Generate initial sequence
            if init_idx == 0 and best_seq is not None:
                a = resample_sequence(best_seq, n)
            else:
                a = random_init(n, seed + n * 104729 + init_idx * 79193)

            # Quick hybrid optimization
            t_phase = min(alloc / n_inits - 2, 15.0)
            a, obj = hybrid_optimize(a, time_budget=max(5.0, t_phase))

            print(f"  n={n}, init{init_idx}: obj={obj:.6f}")

            if obj < best_score:
                best_score = obj
                best_seq = a.copy()
                print(f"  ** UPDATE: n={n}, obj={best_score:.6f} **")

            candidates.append((a.copy(), obj, n))

    # Phase 2: Intensive LP optimization on top candidates
    print(f"\n{'='*60}")
    print(f"PHASE 2: Intensive LP Optimization")
    print(f"{'='*60}")

    # Sort candidates by objective
    candidates.sort(key=lambda x: x[1])

    # Take top candidates + parent (if not already there)
    top_cands = candidates[:5]

    # Ensure parent is included
    if best_seq is not None:
        parent_obj_cur = compute_obj(best_seq)
        top_cands.append((best_seq.copy(), parent_obj_cur, len(best_seq)))

    # Deduplicate by (n, approximate obj)
    seen = set()
    unique_cands = []
    for a_c, obj_c, n_c in top_cands:
        key = (n_c, round(obj_c, 4))
        if key not in seen:
            seen.add(key)
            unique_cands.append((a_c, obj_c, n_c))
    top_cands = unique_cands

    for ci, (a_cand, obj_cand, n_cand) in enumerate(top_cands):
        if time.monotonic() > deadline - 100:
            break
        remaining = deadline - time.monotonic()
        if remaining < 30:
            break

        # Allocate time: more for better candidates
        time_alloc = min(remaining * 0.3, 120.0)
        time_alloc = max(20.0, time_alloc)

        print(f"\n  Candidate {ci}: n={n_cand}, obj={obj_cand:.6f}, time={time_alloc:.0f}s")

        # Intensive LP optimization
        a_new, obj_new = intensive_lp_optimize(a_cand, time_budget=time_alloc)

        if obj_new < best_score:
            best_score = obj_new
            best_seq = a_new.copy()
            print(f"  ** UPDATE: obj={best_score:.6f} **")
        else:
            # Try perturbation from best
            rng = np.random.default_rng(ci * 1337 + 42)
            for ptry in range(2):
                if time.monotonic() > deadline - 80:
                    break
                scale = rng.uniform(0.03, 0.1)
                a_pert = best_seq * np.exp(rng.normal(0, scale, len(best_seq)))
                a_pert = normalize(a_pert)
                a_pert, obj_pert = hybrid_optimize(a_pert, time_budget=min(10, (deadline - time.monotonic()) * 0.1))
                if obj_pert < best_score:
                    best_score = obj_pert
                    best_seq = a_pert.copy()
                    print(f"  ** UPDATE (pert {ptry}): obj={best_score:.6f} **")

    # Phase 3: Final intensive polishing on the global best
    print(f"\n{'='*60}")
    print(f"PHASE 3: Final Polishing")
    print(f"{'='*60}")

    if best_seq is not None and time.monotonic() < deadline - 20:
        n_best = len(best_seq)
        remaining = deadline - time.monotonic()
        print(f"\n  Final polish: n={n_best}, obj={best_score:.6f}, {remaining:.0f}s remain")

        a_cur = best_seq.copy()

        for round_num in range(10):
            if time.monotonic() > deadline - 15:
                break
            remaining = deadline - time.monotonic()
            if remaining < 20:
                break

            sub_budget = min(remaining * 0.3, 60.0)
            sub_budget = max(10.0, sub_budget)

            # Alternating: LP, peak shave, L-BFGS
            method = round_num % 3

            if method == 0:
                # LP
                a_new, improved, _ = lp_alternating_step(a_cur, time_limit=min(15, sub_budget - 5))
                obj_new = compute_obj(a_new)
                if obj_new < best_score:
                    best_score = obj_new
                    best_seq = a_new.copy()
                    a_cur = a_new
                    print(f"  Round {round_num} LP: {best_score:.6f}")
                else:
                    # Try peak shave + LP
                    a_shaved = peak_shave(a_cur, num_peaks=5, alpha=0.03)
                    a_shaved, improved2, _ = lp_alternating_step(a_shaved, time_limit=min(10, sub_budget - 5))
                    obj_shaved = compute_obj(a_shaved)
                    if obj_shaved < best_score:
                        best_score = obj_shaved
                        best_seq = a_shaved.copy()
                        a_cur = a_shaved
                        print(f"  Round {round_num} PeakShave+LP: {best_score:.6f}")

            elif method == 1:
                # L-BFGS
                a_new, obj_new = lbfgs_refine(a_cur, time_budget=min(12, sub_budget - 5), max_iter=200)
                if obj_new < best_score:
                    best_score = obj_new
                    best_seq = a_new.copy()
                    a_cur = a_new
                    print(f"  Round {round_num} L-BFGS: {best_score:.6f}")

            else:
                # Peak shaving + LP
                a_shaved = peak_shave(a_cur, num_peaks=min(10, n_best // 20), alpha=0.04)
                obj_shaved = compute_obj(a_shaved)
                if obj_shaved < best_score:
                    best_score = obj_shaved
                    best_seq = a_shaved.copy()
                    a_cur = a_shaved
                    print(f"  Round {round_num} PeakShave: {best_score:.6f}")

                # Follow with LP
                if time.monotonic() < deadline - 15:
                    a_new, improved, _ = lp_alternating_step(a_cur, time_limit=min(10, max(5, deadline - time.monotonic() - 10)))
                    obj_new = compute_obj(a_new)
                    if obj_new < best_score:
                        best_score = obj_new
                        best_seq = a_new.copy()
                        a_cur = a_new
                        print(f"  Round {round_num} PS+LP: {best_score:.6f}")

    # Final validation and return
    print(f"\n{'='*60}")
    print(f"FINAL RESULTS")
    print(f"{'='*60}")

    if best_seq is None:
        best_seq = np.ones(200)
        best_score = compute_obj(best_seq)

    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    val = evaluate_sequence(result)
    if val < np.inf:
        final_score = val
    else:
        result = [1.0] * 100
        final_score = 2.0

    elapsed = time.monotonic() - start_time
    print(f"  n={len(result)}, score={final_score:.6f}, time={elapsed:.1f}s")
    print(f"  Target: 1.5027")
    print(f"  Difference: {final_score - 1.5027:.6f}")

    return result
