"""

Key innovations over previous approach:

- Focused gradient using only top-K active lags for faster convergence

import numpy as np
from scipy.optimize import linprog
import time


def compute_conv_fft(a):
    """Compute full convolution of a with itself using FFT."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    conv = np.fft.irfft(fa * fa, size)
    return conv[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    conv = compute_conv_fft(a)
    max_b = float(np.max(conv))
    sum_a = float(np.sum(a))
    if sum_a < 0.01:
        return np.inf
    return 2.0 * n * max_b / (sum_a * sum_a)


def build_toeplitz(a):
    """Build Toeplitz matrix where (A@g)[k] = (a*g)[k] for k=0..2n-2."""
    n = len(a)
    A = np.zeros((2 * n - 1, n), dtype=np.float64)
    for j in range(n):
        k_start = j
        k_end = min(j + n, 2 * n - 1)
        A[k_start:k_end, j] = a[:k_end - k_start]
    return A


def solve_lp_minpeak(a, warm_start=None, time_limit=5):
    """
    Solve LP: min M s.t. (a*g)[k] <= M for all k, sum(g)=1, g>=0.
    Returns (g, M, dual_y) where dual_y are shadow prices on lag constraints.
    """
    n = len(a)
    A = build_toeplitz(a)
    num_constraints = 2 * n - 1

    c = np.zeros(n + 1)
    c[-1] = 1.0

    A_ub = np.zeros((num_constraints, n + 1))
    A_ub[:, :n] = A
    A_ub[:, n] = -1.0
    b_ub = np.zeros(num_constraints)

    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])

    bounds = [(0.0, None)] * n + [(0.0, None)]

    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': time_limit})
        if res.status == 0:
            g = np.maximum(res.x[:n], 0)
            s = np.sum(g)
            if s < 1e-10:
                return None, None, None
            g = g / s
            M_val = float(res.x[n])
            # Extract dual variables (shadow prices)
            dual_y = None
            if hasattr(res, 'ineqlin') and res.ineqlin is not None:
                dual_y = np.maximum(res.ineqlin.marginals, 0) if hasattr(res.ineqlin, 'marginals') else None
            if dual_y is not None and hasattr(dual_y, 'marginals'):
                dual_y = np.array(dual_y.marginals)
                dual_y = np.maximum(dual_y, 0)
            return g, M_val, dual_y
    except Exception:
        pass
    return None, None, None


def solve_lp_maxsum(a, M, time_limit=4):
    """Solve LP: max sum(g) s.t. (a*g)[k] <= M for all k, g >= 0."""
    n = len(a)
    A = build_toeplitz(a)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': time_limit})
        if res.status == 0:
            g = np.maximum(res.x, 0)
            return g, float(np.sum(g))
    except Exception:
        pass
    return None, None


def alternating_lp(a, time_budget=120.0, verbose=False):
    """
    Alternating LP optimization with geometric blending and adaptive steps.
    """
    n = len(a)

    s = np.sum(a)
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    stall = 0
    patience = 40
    it = 0

    while time.monotonic() - start < time_budget:
        it += 1
        remaining = time_budget - (time.monotonic() - start)
        if remaining < 3:

        g, M_opt, dual_y = solve_lp_minpeak(a, time_limit=min(5, remaining - 2))
        if g is None or M_opt is None:
            break

        # Geometric blend in log-space
        log_a = np.log(np.maximum(a, 1e-15))
        log_g = np.log(np.maximum(g, 1e-15))

        # Adaptive step: larger when making progress, smaller when stalling
        base_t = 0.5 if stall < 10 else 0.3 if stall < 20 else 0.15

        best_t = 0.0
        best_obj_ls = best_obj
        for t in [base_t, base_t * 0.5, base_t * 1.5, 0.8, 1.0]:
            if t <= 0 or t > 1:

            log_new = (1 - t) * log_a + t * log_g

            log_new = (1 - best_t) * log_a + best_t * log_g
            a_new = np.exp(log_new)
            a_new = np.maximum(a_new, 1e-15)
            s_new = np.sum(a_new)

            best_a = a_new.copy()
            a = a_new
            stall = 0

def focused_subgradient(a, time_budget=60.0, top_k=5, verbose=False):
    """
    Focused subgradient using only top-K active lags.
    More targeted than full softmax approach.
    """
    n = len(a)
    a = a.copy().astype(np.float64)
    s = np.sum(a)
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    # Adaptive parameters
    step = 0.03
    temperature = 1.0
    stall = 0
    it = 0

    while time.monotonic() - start < time_budget:
        it += 1
        if it > 100000:
            break

        conv = compute_conv_fft(a)
        S = float(np.sum(a))
        M = float(np.max(conv))

        # Get top-K active lags
        if top_k >= len(conv):
            top_indices = np.arange(len(conv))
        else:
            top_indices = np.argsort(conv)[-top_k:]

        # Compute focused gradient
        grad = np.zeros(n)
        for k in top_indices:
            m_low = max(0, int(k) - n + 1)
            m_high = min(n - 1, int(k))
            if m_low <= m_high:
                valid_m = np.arange(m_low, m_high + 1)
                valid_idx = int(k) - valid_m
                valid_idx = np.clip(valid_idx, 0, n - 1)
                grad[valid_m] += a[valid_idx]

        # Normalize gradient
        grad_mean = np.mean(grad)
        grad_centered = grad - grad_mean

        # Multiplicative update
        log_update = -step * temperature * grad_centered
        log_update = np.clip(log_update, -3, 3)
        a_new = a * np.exp(log_update)
        a_new = np.maximum(a_new, 1e-15)
        s_new = np.sum(a_new)
        if s_new > 1e-10:
            a_new = a_new / s_new
        a = a_new

        obj = compute_obj(a)
        if obj < best_obj - 1e-14:
            best_obj = obj
            best_a = a.copy()
            stall = 0
        else:
            stall += 1
            if stall % 50 == 0:
                step *= 0.9
                temperature *= 1.05
            if stall > 500:
                break

        # Cooldown
        if it % 200 == 0:
            step *= 0.95
            temperature *= 1.02

    return best_a

def local_search(a, time_budget=120.0, verbose=False):

            a_pert = best_a * np.exp(rng.normal(0, 0.02, n))

            best_a = a_opt.copy()
            if verbose:
                print(f"    LS attempt {attempt} (strat={strategy}): {best_obj:.8f}")

    return best_a


def optimize_brief(a, time_budget=30.0):
    """Quick full optimization for a single sequence."""
    n = len(a)
    a = a.copy().astype(np.float64)
    s = np.sum(a)
    if s < 1e-10:
        a = np.ones(n)
        s = n
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    # Quick LP
    remaining = time_budget - (time.monotonic() - start)
    if remaining > 10:
        a_new = alternating_lp(best_a, time_budget=min(15.0, remaining * 0.5))
        obj = compute_obj(a_new)
        if obj < best_obj:
            best_obj = obj
            best_a = a_new.copy()

    # Quick subgradient
    remaining = time_budget - (time.monotonic() - start)
    if remaining > 5:
        a_new = focused_subgradient(best_a, time_budget=min(10.0, remaining * 0.6), top_k=3)
        obj = compute_obj(a_new)
        if obj < best_obj:
            best_obj = obj
            best_a = a_new.copy()

    return best_a, best_obj


def resample_sequence(a, new_n):
    """Resample a sequence to a new length."""
    n = len(a)
    if n == new_n:
        return a.copy()
    old_k = np.linspace(0.0, 1.0, n)
    new_k = np.linspace(0.0, 1.0, new_n)
    a_new = np.interp(new_k, old_k, a)
    a_new = np.maximum(a_new, 1e-10)
    s = np.sum(a_new)
    if s > 1e-10:
        a_new = a_new / s
    return a_new


def try_parent():
    """Try to load parent sequence."""
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64)
        parent = np.clip(parent, 0, 1000.0)
        if parent.size > 0 and np.isfinite(parent).all():
            obj = compute_obj(parent)
            print(f"Parent (n={parent.size}): obj={obj:.8f}")
            return parent.copy(), obj
    except NameError:
        pass
    except Exception as e:
        print(f"Failed to load parent: {e}")
    return None, np.inf


def propose_candidate(seed=42, budget_s=1500):
    """
    Main search function. Uses alternating LP, focused subgradient, and
    local search to minimize the step function objective.
    """
    start_time = time.monotonic()
    deadline = start_time + float(budget_s) - 15.0

    best_seq = None
    best_score = np.inf

    # Load parent
    parent, parent_obj = try_parent()
    if parent is not None:
        best_score = parent_obj
        best_seq = parent.copy()
        n_parent = len(parent)
    else:
        n_parent = 330
        best_seq = np.ones(n_parent) / n_parent
        best_score = compute_obj(best_seq)
        print(f"No parent, using uniform n={n_parent}: obj={best_score:.8f}")

    print(f"Initial best: obj={best_score:.8f} (n={len(best_seq)})")
    print(f"Time budget: {budget_s}s, deadline in {deadline - start_time:.0f}s")

    # ============ Phase 1: Alternating LP on parent (main workhorse) ============
    remaining = deadline - time.monotonic()
    if remaining > 50:
        print(f"\n=== Phase 1: Alternating LP ({min(remaining*0.3, 300):.0f}s) ===")
        p1_budget = min(remaining * 0.3, 300.0)
        a_opt = alternating_lp(best_seq, time_budget=p1_budget, verbose=True)
        obj = compute_obj(a_opt)
        if obj < best_score:
            best_score = obj
            best_seq = a_opt.copy()
            print(f"  After Phase 1 LP: {best_score:.8f}")
        else:
            print(f"  Phase 1 LP: no improvement ({compute_obj(a_opt):.8f})")

    # ============ Phase 2: Focused subgradient ============
    remaining = deadline - time.monotonic()
    if remaining > 30:
        print(f"\n=== Phase 2: Focused subgradient ({min(remaining*0.15, 100):.0f}s) ===")
        p2_budget = min(remaining * 0.15, 100.0)
        a_opt = focused_subgradient(best_seq, time_budget=p2_budget, top_k=5)
        obj = compute_obj(a_opt)
        if obj < best_score:
            best_score = obj
            best_seq = a_opt.copy()
            print(f"  After Phase 2: {best_score:.8f}")
        else:
            print(f"  Phase 2: no improvement")

    # ============ Phase 3: Full softmax subgradient ============
    remaining = deadline - time.monotonic()
    if remaining > 30:
        print(f"\n=== Phase 3: Full softmax SG ({min(remaining*0.12, 80):.0f}s) ===")
        p3_budget = min(remaining * 0.12, 80.0)
        a_opt = full_softmax_subgradient(best_seq, time_budget=p3_budget, tau=25.0, step=0.04)
        obj = compute_obj(a_opt)
        if obj < best_score:
            best_score = obj
            best_seq = a_opt.copy()
            print(f"  After Phase 3: {best_score:.8f}")
        else:
            print(f"  Phase 3: no improvement")

    # ============ Phase 4: Local search ============
    remaining = deadline - time.monotonic()
    if remaining > 50:
        print(f"\n=== Phase 4: Local search ({min(remaining*0.2, 200):.0f}s) ===")
        p4_budget = min(remaining * 0.2, 200.0)
        a_opt = local_search(best_seq, time_budget=p4_budget, verbose=True)
        obj = compute_obj(a_opt)
        if obj < best_score:
            best_score = obj
            best_seq = a_opt.copy()
            print(f"  After Phase 4: {best_score:.8f}")
        else:
            print(f"  Phase 4: no improvement")

    # ============ Phase 5: Second round of LP + subgradient ============
    remaining = deadline - time.monotonic()
    if remaining > 40:
        print(f"\n=== Phase 5: Second round LP ({min(remaining*0.2, 150):.0f}s) ===")
        p5_budget = min(remaining * 0.2, 150.0)
        a_opt = alternating_lp(best_seq, time_budget=p5_budget, verbose=True)
        obj = compute_obj(a_opt)
        if obj < best_score:
            best_score = obj
            best_seq = a_opt.copy()
            print(f"  After Phase 5 LP: {best_score:.8f}")

        remaining = deadline - time.monotonic()
        if remaining > 20:
            p5b_budget = min(remaining * 0.15, 60.0)
            a_opt = focused_subgradient(best_seq, time_budget=p5b_budget, top_k=3)
            obj = compute_obj(a_opt)
            if obj < best_score:
                best_score = obj
                best_seq = a_opt.copy()
                print(f"  After Phase 5 SG: {best_score:.8f}")

    # ============ Phase 6: Try nearby n values ============
    remaining = deadline - time.monotonic()
    if remaining > 60:
        print(f"\n=== Phase 6: Nearby n values ({remaining:.0f}s) ===")
        n_candidates = [n_parent - 5, n_parent - 3, n_parent - 1,
                        n_parent + 1, n_parent + 3, n_parent + 5,
                        n_parent + 10, n_parent - 10]
        n_candidates = [n for n in n_candidates if 100 <= n <= 2000 and n != n_parent]
        n_candidates = sorted(set(n_candidates))

        per_n_time = min(30.0, (remaining - 20) / max(len(n_candidates), 1))

        for n_val in n_candidates:
            if time.monotonic() > deadline - 30:
                break
            remaining = deadline - time.monotonic()
            if remaining < 20:
                break

            a_try = resample_sequence(best_seq, n_val)
            a_opt, obj = optimize_brief(a_try, time_budget=max(5.0, per_n_time))
            if obj < best_score:
                best_score = obj
                best_seq = a_opt.copy()
                print(f"  ** n={n_val}: obj={best_score:.8f} **")
            else:
                print(f"  n={n_val}: obj={obj:.8f} (no improvement)")

    # ============ Phase 7: Final intensive polish ============
    remaining = deadline - time.monotonic()
    if remaining > 60:
        print(f"\n=== Phase 7: Final polish ({remaining:.0f}s) ===")

        # Multiple rounds of short LP + subgradient
        rounds = min(5, int(remaining // 30))
        for rnd in range(rounds):
            if time.monotonic() > deadline - 20:
                break
            sub_budget = min(25.0, (deadline - time.monotonic()) - 15)
            if sub_budget < 10:
                break

            # LP
            a_opt = alternating_lp(best_seq, time_budget=sub_budget * 0.5)
            obj = compute_obj(a_opt)
            if obj < best_score:
                best_score = obj
                best_seq = a_opt.copy()

            # Subgradient
            remaining = deadline - time.monotonic()
            if remaining > 15:
                a_opt = focused_subgradient(best_seq, time_budget=min(15.0, remaining - 10), top_k=3)
                obj = compute_obj(a_opt)
                if obj < best_score:
                    best_score = obj
                    best_seq = a_opt.copy()

            # Quick local search
            remaining = deadline - time.monotonic()
            if remaining > 20:
                a_opt = local_search(best_seq, time_budget=min(20.0, remaining - 10))
                obj = compute_obj(a_opt)
                if obj < best_score:
                    best_score = obj
                    best_seq = a_opt.copy()

            print(f"  Round {rnd + 1}: obj={best_score:.8f}")

    # ============ Final: Very fine subgradient ============
    remaining = deadline - time.monotonic()
    if remaining > 10:
        print(f"\n=== Final fine-tune ({remaining:.0f}s) ===")
        a_opt = focused_subgradient(best_seq, time_budget=remaining - 5, top_k=2)
        obj = compute_obj(a_opt)
        if obj < best_score:
            best_score = obj
            best_seq = a_opt.copy()
        print(f"  Final: obj={best_score:.8f}")

    # Validation and return
    best_seq = np.clip(best_seq, 0.0, 1000.0)
    best_seq = np.maximum(best_seq, 0.0)
    result = [float(x) for x in best_seq]

    # Validate with the actual evaluation function
    try:
        eval_val = evaluate_sequence(result)
        if np.isfinite(eval_val) and eval_val < best_score:
            best_score = eval_val
        print(f"\n=== FINAL: n={len(best_seq)}, our_obj={compute_obj(best_seq):.8f}, "
              f"eval={eval_val:.8f}, time={time.monotonic()-start_time:.1f}s ===")
    except Exception as e:
        print(f"\n=== FINAL: n={len(best_seq)}, obj={best_score:.8f}, "
              f"eval error: {e}, time={time.monotonic()-start_time:.1f}s ===")

    return result
