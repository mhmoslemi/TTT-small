import numpy as np
from scipy.optimize import linprog
from scipy.sparse import csr_matrix
import time

# Cache for convolution sparsity
_spar_cache = {}

def get_sparsity(n):
    if n not in _spar_cache:
        rows = []
        cols = []
        for k in range(2 * n - 1):
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            for j in range(js, je + 1):
                rows.append(k)
                cols.append(j)
        _spar_cache[n] = (np.array(rows), np.array(cols))
    return _spar_cache[n]

def fast_conv(a):
    n = len(a)
    s = 1
    while s < 2 * n - 1:
        s <<= 1
    fa = np.fft.rfft(a, s)
    return np.fft.irfft(fa * fa, s)[:2 * n - 1]

def compute_J(a):
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))

def lp_best_response(a, M, max_time=5):
    n = len(a)
    rows, cols = get_sparsity(n)
    nnz = len(rows)
    vals = np.empty(nnz)
    for i in range(nnz):
        k = rows[i]
        j = cols[i]
        vals[i] = a[k - j]
    A = csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                options={'presolve': True, 'time_limit': max_time})
    if r.success and r.x is not None:
        g = np.maximum(r.x, 0.0)
        if g.sum() > 1e-10:
            return g / g.sum()
    return None

def adaptive_lp_opt(a, deadline, tracker, max_iters=100):

        best_J = current_J

            a /= a.sum()
            current_J = best_J

    tracker.update(a)
    return a

def spectral_init(n, rng):
    """Initialize with approximately flat autocorrelation using spectral methods."""
    # Target: (a*a)[k] ≈ constant for all k
    # In frequency domain: |A(f)|^2 should make the inverse DFT of |A|^2 flat
    # |A(f)|^2 = DFT(a*a) (by convolution theorem, up to normalization)
    # So we want DFT(a*a) to be as flat as possible
    # DFT(a*a) = |DFT(a)|^2 (by the convolution-multiplication duality)
    # Wait: DFT(a*a) = DFT(a) * DFT(a) (circular convolution)
    # For linear convolution, it's more complex, but approximately:
    # We want |A(f)|^2 to be flat → A(f) has constant magnitude
    # This is a constant-magnitude sequence in frequency domain
    
    # Generate a constant-magnitude spectrum with random phases
    half = n // 2 + 1
    phases = rng.uniform(0, 2 * np.pi, half)
    A = np.zeros(n, dtype=complex)
    A[:half] = np.exp(1j * phases[:half])
    # Enforce conjugate symmetry for real signal
    for k in range(1, n - n // 2 if n % 2 == 0 else n - n // 2):
        A[k] = np.conj(A[n - k])
    # Fix DC and Nyquist
    A[0] = 1.0 + 0j
    if n % 2 == 0:
        A[n // 2] = 1.0 + 0j
    
    a = np.fft.ifft(A).real
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-10:
        return np.ones(n) / n
    return a / s

def tapered_init(n, rng, alpha):
    """Tapered (raised cosine or similar) initialization."""
    t = np.linspace(0, 1, n)
    a = (1.0 + np.cos(2 * np.pi * t)) / 2.0  # Raised cosine
    a = a ** alpha + 0.01
    a = np.maximum(a, 0.001)
    a /= a.sum()
    return a

def multi_freq_init(n, rng):
    """Multi-frequency cosine initialization."""
    t = np.linspace(0, 1, n)
    n_freq = rng.integers(3, 8)
    freqs = rng.uniform(1, 10, n_freq).astype(int)
    phases = rng.uniform(0, 2 * np.pi, n_freq)
    amps = rng.uniform(0.1, 0.4, n_freq)
    a = np.ones(n)
    for f, p, amp in zip(freqs, phases, amps):
        a += amp * np.cos(2 * np.pi * f * t + p)
    a = np.maximum(a, 0.001)
    a /= a.sum()
    return a

def resample_seq(src, dest_n):
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)

    if parent is not None and parent.size > 0:

    inits.append(tapered_init(n, rng, 1.0))
    inits.append(tapered_init(n, rng, 2.0))
    inits.append(tapered_init(n, rng, 0.5))
    inits.append(spectral_init(n, rng))
    inits.append(spectral_init(n, rng))
    inits.append(multi_freq_init(n, rng))
    inits.append(multi_freq_init(n, rng))
    
    # Gaussian-like
    t = np.linspace(-3, 3, n)
    inits.append(np.exp(-t**2 / 2) + 0.01)
    
    # Two-sided exponential
    t = np.linspace(0, 1, n)
    inits.append(np.exp(-4 * (t - 0.5)**2) + 0.05)
    inits.append(np.exp(-2 * np.abs(t - 0.5)) + 0.05)
    
    # Concave
    inits.append(np.sin(np.pi * t)**0.5 + 0.01)
    
    for x in inits:
        x = np.maximum(x, 0.001).astype(np.float64)
        x /= x.sum()
    return inits

def block_optimize(a, block_size, max_rounds, deadline, tracker):
    """Optimize blocks of the sequence while keeping others fixed."""
    a = a.copy().astype(np.float64)
    a /= a.sum()
    n = len(a)
    current_J = compute_J(a)
    tracker.update(a)
    
    n_blocks = max(1, n // block_size)
    
    for rnd in range(max_rounds):
        if time.monotonic() > deadline:
            break
        for blk in range(n_blocks):
            start = blk * block_size
            end = min(start + block_size, n)
            if end - start < 2:
                continue
            
            # Try to optimize this block
            best_block = a[start:end].copy()
            best_J = current_J
            
            # Simplex search within the block
            for _ in range(10):
                # Random perturbation of the block
                pert = best_block + np.random.standard_normal(end - start) * 0.1
                pert = np.maximum(pert, 1e-4)
                
                a_trial = a.copy()
                a_trial[start:end] = pert
                ss = a_trial.sum()
                a_trial /= ss
                J_trial = compute_J(a_trial)
                
                if J_trial < best_J:
                    best_J = J_trial
                    best_block = pert
            
            if best_J < current_J - 1e-14:
                a[start:end] = best_block
                current_J = best_J
        
        if rnd % 5 == 0:
            tracker.update(a)
    
    tracker.update(a)
    return a

import numpy as np
from scipy.optimize import linprog
from scipy.sparse import csr_matrix
import time

_spar_cache = {}

def get_sparsity(n):
    if n not in _spar_cache:
        rows = []
        cols = []
        for k in range(2 * n - 1):
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            for j in range(js, je + 1):
                rows.append(k)
                cols.append(j)
        _spar_cache[n] = (np.array(rows, dtype=np.int32), np.array(cols, dtype=np.int32))
    return _spar_cache[n]

def fast_conv(a):
    n = len(a)
    s = 1
    while s < 2 * n - 1:
        s <<= 1
    fa = np.fft.rfft(a, s)
    return np.fft.irfft(fa * fa, s)[:2 * n - 1]

def compute_J(a):
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))

def compute_grad_lse(p, a):
    """Compute gradient: grad[j] = sum_k p[k]*a[k-j] using FFT cross-correlation."""
    n = len(a)
    s = 1
    while s < 3 * n:
        s <<= 1
    P = np.fft.rfft(p, s)
    A = np.fft.rfft(a, s)
    corr = np.fft.irfft(P * np.conj(A), s)
    return corr[:n]

def lp_min_max_response(a, time_limit=5):
    """Find g minimizing max_k (a*g)[k] subject to sum(g)=1, g>=0."""
    n = len(a)
    rows, cols = get_sparsity(n)
    nnz = len(rows)
    n_rows = 2 * n - 1
    
    vals_g = np.array([a[rows[i] - cols[i]] for i in range(nnz)], dtype=np.float64)
    
    A_ub_data = np.concatenate([vals_g, np.full(n_rows, -1.0)])
    A_ub_rows = np.concatenate([rows, np.arange(n_rows, dtype=np.int32)])
    A_ub_cols = np.concatenate([cols, np.zeros(n_rows, dtype=np.int32)])
    
    A_ub = csr_matrix((A_ub_data, (A_ub_rows, A_ub_cols)), shape=(n_rows, n + 1))
    
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])
    
    c = np.zeros(n + 1)
    c[-1] = 1.0
    bounds = [(0, None)] * n + [(0, None)]
    
    r = linprog(c, A_ub=A_ub, b_ub=np.zeros(n_rows),
                A_eq=A_eq, b_eq=b_eq, bounds=bounds,
                method='highs', options={'presolve': True, 'time_limit': time_limit})
    
    if r.success and r.x is not None:
        g = np.maximum(r.x[:n], 0.0)
        M_opt = float(r.x[n])
        gs = g.sum()
        if gs > 1e-10:
            g /= gs
            return g, M_opt
    return None, None

def subgradient_descent(a, max_iters, deadline, tracker):
    """Subgradient descent on max(a*a) with FFT gradient."""
    a = np.maximum(a.astype(np.float64), 1e-8)
    a /= a.sum()
    n = len(a)
    current_J = compute_J(a)
    tracker.update(a)
    lr = 1.0 / n
    patience = 0
    
    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        b = fast_conv(a)
        M = float(b.max())
        active = b >= M - 1e-10 * M
        p = active.astype(np.float64) / active.sum()
        
        grad = compute_grad_lse(p, a)
        direction = grad - np.mean(grad)
        a_new = a - lr * direction
        a_new = np.maximum(a_new, 1e-8)
        a_new /= a_new.sum()
        
        J_new = compute_J(a_new)
        if J_new < current_J - 1e-15:
            a = a_new
            current_J = J_new
            patience = 0
            lr = min(lr * 1.03, 5.0 / n)
        else:
            lr *= 0.7
            patience += 1
            if patience > 40:
                break
        if it % 30 == 0:
            tracker.update(a)
    
    tracker.update(a)
    return a

def lse_gradient_descent(a, max_iters, deadline, tracker):
    """LSE-smoothed gradient descent with momentum."""
    a = np.maximum(a.astype(np.float64), 1e-6)
    a /= a.sum()
    n = len(a)
    current_J = compute_J(a)
    tracker.update(a)
    
    tau = 80.0
    lr = 0.3 / n
    momentum = 0.85
    v = np.zeros(n)
    patience = 0
    
    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        b = fast_conv(a)
        b_shifted = b - b.max()
        exp_b = np.exp(tau * b_shifted)
        p = exp_b / exp_b.sum()
        
        grad = compute_grad_lse(p, a)
        v = momentum * v + (1.0 - momentum) * grad
        a_new = a - lr * v
        a_new = np.maximum(a_new, 1e-6)
        a_new /= a_new.sum()
        
        J_new = compute_J(a_new)
        if J_new < current_J - 1e-15:
            a = a_new
            current_J = J_new
            patience = 0
            tau = min(400.0, tau * 1.015)
            lr = min(lr * 1.02, 3.0 / n)
        else:
            lr *= 0.5
            v *= 0.3
            patience += 1
            if patience > 60:
                break
        if it % 40 == 0:
            tracker.update(a)
    
    tracker.update(a)
    return a

def spectral_init(n, rng):
    """Flat-magnitude spectrum initialization."""
    half = n // 2 + 1
    phases = rng.uniform(0, 2 * np.pi, half)
    A = np.zeros(n, dtype=complex)
    A[:half] = np.exp(1j * phases[:half])
    for k in range(1, n - n // 2):
        A[k] = np.conj(A[n - k])
    A[0] = 1.0 + 0j
    if n % 2 == 0:
        A[n // 2] = 1.0 + 0j
    a = np.fft.ifft(A).real
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-10:
        return np.ones(n) / n
    return a / s

def tapered_init(n, rng, alpha):
    t = np.linspace(0, 1, n)
    a = (1.0 + np.cos(2 * np.pi * t)) / 2.0
    a = a ** alpha + 0.005
    a = np.maximum(a, 0.001)
    a /= a.sum()
    return a

def multi_freq_init(n, rng):
    t = np.linspace(0, 1, n)
    n_freq = rng.integers(3, 8)
    freqs = rng.uniform(1, 12, n_freq).astype(int)
    phases = rng.uniform(0, 2 * np.pi, n_freq)
    amps = rng.uniform(0.1, 0.35, n_freq)
    a = np.ones(n)
    for f, p, amp in zip(freqs, phases, amps):
        a += amp * np.cos(2 * np.pi * f * t + p)
    a = np.maximum(a, 0.001)
    a /= a.sum()
    return a

def gaussian_init(n, rng):
    t = np.linspace(-3, 3, n)
    a = np.exp(-t ** 2 / (2 * rng.uniform(0.5, 1.5))) + 0.01
    a /= a.sum()
    return a

def resample_seq(src, dest_n):
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)

            a = adaptive_lp_opt(a0, min(t_phase_end - 10, time.monotonic() + init_time),
                               tracker, max_iters=25)
            
            if time.monotonic() < t_phase_end - 5:
                a = subgradient_descent(a, 100, t_phase_end - 3, tracker)
            
            J = compute_J(a)
            if J < best_J_for_n:
                best_J_for_n = J
                best_for_n = a.copy()
        
        if best_for_n is not None:
            tracker.update(best_for_n)
            if parent is None or best_J_for_n < compute_J(resample_seq(parent, best_for_n.size)):
                parent = best_for_n
            tracker.report(f"Phase1 n={n} ({time.monotonic()-t_phase_start:.1f}s): J={best_J_for_n:.8f}")
    
    # === Phase 2: Intensive optimization on best candidate ===
    if tracker.best is not None and time.monotonic() < deadline - 200:
        t0 = time.monotonic()
        n = tracker.best.size
        print(f"\n{'='*40} Phase2 Intensive n={n} {'='*40}", flush=True)
        
        a = tracker.best.copy().astype(np.float64)
        
        # Multiple LP passes with varying time limits
        for _ in range(8):
            if time.monotonic() > deadline - 150:
                break
            a = adaptive_lp_opt(a, deadline - 150, tracker, max_iters=40)
        
        # LSE gradient
        if time.monotonic() < deadline - 100:
            a = lse_gradient_descent(a, 3000, deadline - 100, tracker)
        
        # Subgradient
        if time.monotonic() < deadline - 60:
            a = subgradient_descent(a, 2000, deadline - 60, tracker)
        
        # Another LP pass
        if time.monotonic() < deadline - 30:
            a = adaptive_lp_opt(a, deadline - 25, tracker, max_iters=30)
        
        tracker.report(f"Phase2 ({time.monotonic()-t0:.1f}s)")
    
    # === Phase 3: SA escape ===
    if tracker.best is not None and time.monotonic() < deadline - 100:
        t0 = time.monotonic()
        print(f"\n{'='*40} Phase3 SA {'='*40}", flush=True)
        a = simulated_annealing_escape(tracker.best, deadline - 40, tracker, n_restarts=20)
        
        if time.monotonic() < deadline - 15:
            a = adaptive_lp_opt(a, deadline - 10, tracker, max_iters=15)
            if time.monotonic() < deadline - 5:
                a = lse_gradient_descent(a, 500, deadline - 2, tracker)
        
        tracker.report(f"Phase3 ({time.monotonic()-t0:.1f}s)")
    
    # === Phase 4: Coordinate + pairwise polish ===
    if tracker.best is not None and time.monotonic() < deadline - 15:
        t0 = time.monotonic()
        print(f"\n{'='*40} Phase4 Polish {'='*40}", flush=True)
        a = coordinate_descent(tracker.best, 3000, deadline - 8, tracker)
        if time.monotonic() < deadline - 4:
            a = pairwise_swap(a, 1000, deadline - 2, tracker)
        if time.monotonic() < deadline - 1:
            a = adaptive_lp_opt(a, deadline - 0.5, tracker, max_iters=8)
        
        tracker.report(f"Phase4 ({time.monotonic()-t0:.1f}s)")
    
    # Fallback
    if tracker.best is None:
        tracker.best = np.ones(256)
        tracker.best_score = compute_J(tracker.best)
        tracker.update(tracker.best)
    
    print(f"\n{'='*40} FINAL {'='*40}", flush=True)
    print(f"J={tracker.best_score:.8f} n={tracker.best.size} time={time.monotonic()-tracker.t_start:.1f}s", flush=True)
    return [float(x) for x in tracker.best]
