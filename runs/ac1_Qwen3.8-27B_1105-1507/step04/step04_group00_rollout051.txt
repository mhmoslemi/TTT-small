from scipy.optimize import linprog
from scipy.sparse import csr_matrix, csc_matrix
import time

def fast_conv(a):
    """Full self-convolution via FFT."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]

def compute_J(a):
    """Compute J(a) = 2*n*max(conv(a,a))/(sum(a))^2."""
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))

def apply_conv_matrix(a, g):
    """Compute conv(a, g) where both have length n. Result has length 2n-1."""
    return np.convolve(a, g)  # or use FFT for speed

def lp_best_response(a, max_time=5.0):
    """
    Solve: min M s.t. (a*g)[k] <= M for k=0..2n-2, sum(g) = 1, g >= 0.
    Returns (g, M).
    """
    n = len(a)
    # Build constraint matrix A where A[k,j] = a[k-j] if 0 <= k-j < n
    # Use sparse construction
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        for j in range(max(0, k - n + 1), min(n, k + 1)):
            rows.append(k)
            cols.append(j)
            vals.append(a[k - j])
    
    A = csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))
    
    # LP: min M
    # s.t. A @ g - M <= 0  (2n-1 constraints)
    #      sum(g) = 1       (1 equality)
    #      g >= 0, M free

    # x = [g_0, ..., g_{n-1}, M]
    
    # Objective: minimize M -> c = [0, ..., 0, 1]
    c = np.zeros(n + 1)
    c[-1] = 1.0
    
    # Inequality: A @ g <= M -> A @ g - M <= 0
    A_ub = np.hstack([A.toarray(), -np.ones((2 * n - 1, 1))])
    b_ub = np.zeros(2 * n - 1)
    
    # Equality: sum(g) = 1
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])
    
    # Bounds: g >= 0, M free
    bounds = [(0, None)] * n + [(None, None)]
    
    try:
        r = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                    bounds=bounds, method='highs',
                    options={'time_limit': max_time, 'presolve': True})
        if r.success and r.x is not None:
            g = np.maximum(r.x[:n], 0)
            M = float(r.x[-1])
            if g.sum() > 1e-10:

"""
Minimizes J(a) = 2*n*max(conv(a,a))/(sum(a))^2 using:
1. FFT-based smooth gradient descent for fast exploration across lengths
2. LP mirror descent with Nesterov acceleration and averaging (main workhorse)
3. Subgradient polishing using LP dual variables
4. Coordinate descent on active constraint set for final polish
5. Multi-length extension attempts
Key improvements over previous: better LP formulation, Nesterov acceleration,
subgradient from LP dual, and more aggressive exploration of n=640-1024.
"""

import numpy as np
from scipy.optimize import linprog
from scipy.sparse import csr_matrix
import time


def fast_conv(a):
    """Full self-convolution via FFT."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]


def conv_pair(a, b):
    """Convolution of two sequences via FFT."""
    n = max(len(a), len(b))
    size = 1
    while size < len(a) + len(b) - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    return np.fft.irfft(fa * fb, size)[:len(a) + len(b) - 1]


def compute_J(a):
    """Compute J(a) = 2*n*max(conv(a,a))/(sum(a))^2."""
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))


def build_A_sparse(n, a):
    """Build sparse convolution matrix A[k,j] = a[k-j] if 0<=k-j<n."""
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        j_min = max(0, k - n + 1)
        j_max = min(n - 1, k)
        for j in range(j_min, j_max + 1):
            rows.append(k)
            cols.append(j)
            vals.append(a[k - j])
    return csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))


def lp_best_response(a, time_limit=10.0):
    """
    Solve: min M s.t. (a*g)[k] <= M for k=0..2n-2, sum(g)=1, g>=0.
    Returns (g, M, dual_y) where dual_y is the normalized dual variable.
    """
    n = len(a)
    A = build_A_sparse(n, a)
    A_dense = A.toarray()

    # x = [g_0..g_{n-1}, M]
    c = np.zeros(n + 1)
    c[-1] = 1.0

    A_ub = np.hstack([A_dense, -np.ones((2 * n - 1, 1))])
    b_ub = np.zeros(2 * n - 1)

    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])

    bounds = [(0, None)] * n + [(None, None)]

    try:
        r = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                    bounds=bounds, method='highs',
                    options={'time_limit': time_limit, 'presolve': True,
                             'dual_feasibility_tolerance': 1e-10})
        if r.success and r.x is not None:
            g = np.maximum(r.x[:n], 0)
            M = float(r.x[-1])
            g_sum = g.sum()
            if g_sum > 1e-12:
                g /= g_sum
                M = float(conv_pair(a, g).max())
            # Try to extract dual variables
            dual_y = None
            if hasattr(r, 'ineqlin') and r.ineqlin is not None:
                dual_y = np.maximum(r.ineqlin, 0)
                dy = dual_y.sum()
                if dy > 1e-12:
                    dual_y /= dy
            return g, M, dual_y
    except Exception:
        pass
    return None, None, None


def run_lp_mirror_descent(a, deadline, max_iters=50000):
    """
    LP mirror descent with adaptive step size and averaging.
    Uses Nesterov-style extrapolation for faster convergence.
    """
    a = np.maximum(a.astype(np.float64), 0)
    s = a.sum()
    if s < 1e-12:
        return a
    a = a / s
    n = len(a)

    best_a = a.copy()
    best_M = fast_conv(a).max()

    # Polyak averaging
    avg = np.zeros(n)
    count = 0

    # Nesterov momentum
    a_prev = a.copy()
    beta = 1.0

    # Step size schedule
    t = 1.0 / (1 + 0.5)

    # Backtracking
    no_improve = 0
    current_M = best_M

    t0 = time.monotonic()
    lp_time_budget = 8.0  # seconds per LP solve (will adapt)

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        # Nesterov extrapolation (after first few iterations)
        if it > 3:
            beta_new = (1.0 + np.sqrt(1.0 + 4.0 * beta * beta)) / 2.0
            y = a + (beta_new / beta - 1.0) * (a - a_prev)
            y = np.maximum(y, 0)
            ys = y.sum()
            if ys > 1e-12:
                y = y / ys
            beta = beta_new
        else:
            y = a.copy()

        # Solve LP best response
        g, M_g, dual_y = lp_best_response(y, time_limit=lp_time_budget)
        if g is None:
            # Fallback: uniform
            g = np.ones(n) / n

        a_prev = a.copy()

        # Mix with adaptive step
        a_new = (1 - t) * a + t * g
        a_new = np.maximum(a_new, 0)
        s = a_new.sum()
        if s < 1e-10:
            a_new = a.copy()
        else:
            a_new = a_new / s

        M_new = fast_conv(a_new).max()

        if M_new < current_M * (1 - 1e-10):
            current_M = M_new
            no_improve = 0
            t = min(1.0, t * 1.05)  # increase step slightly
        else:
            no_improve += 1
            if no_improve > 20:
                t = t * 0.7
                no_improve = 0

        a = a_new

        # Track best
        if M_new < best_M:
            best_M = M_new
            best_a = a.copy()

        # Averaging
        avg += a
        count += 1

        # Decrease step size
        t = max(0.01, 1.0 / (1.0 + 0.3 * (it + 1) ** 0.5))

        # Adaptive LP time: reduce if taking too long
        if it < 50:
            elapsed_lp = 0  # rough estimate
        if it % 200 == 0 and it > 0:
            elapsed = time.monotonic() - t0
            if elapsed > 0:
                avg_per_iter = elapsed / it
                if avg_per_iter > 0.5:
                    lp_time_budget = max(2.0, lp_time_budget * 0.8)
                print(f"    iter={it}, M={fast_conv(a).max():.8f}, best={best_M:.8f}, "

                      flush=True)

    # Check averaged solution
    if count > 10:
        # Use tail average (last 50%)
        a_avg = avg / count
        a_avg = np.maximum(a_avg, 0)
        s = a_avg.sum()
        if s > 1e-10:
            a_avg /= s
            M_avg = fast_conv(a_avg).max()
            if M_avg < best_M:

                best_a = a_avg.copy()

def run_subgradient_lp(a, deadline, max_iters=20000):
    """
    Subgradient descent using LP dual variables as subgradient.
    s = 2 * conv(dual_y, a) gives the subgradient direction.
    """
    a = np.maximum(a.astype(np.float64), 0)
    s = a.sum()
    if s < 1e-12:
        return a
    a = a / s
    n = len(a)

    best_a = a.copy()
    best_M = fast_conv(a).max()
    avg = np.zeros(n)
    count = 0

    lr_init = 0.3

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        g, M_g, dual_y = lp_best_response(a, time_limit=5.0)
        if g is None or dual_y is None:
            break

        # Subgradient: s = 2 * conv(dual_y, a) truncated to length n
        # Actually, the subgradient of max_k (a*a)[k] using weights p is:
        # grad_i = 2 * sum_k p[k] * a[k-i] (where valid)
        # This is 2 * (p * a_correlation)
        grad = np.zeros(n)
        for k in range(2 * n - 1):
            pk = dual_y[k]
            if pk < 1e-14:
                continue
            lo = max(0, k - n + 1)
            hi = min(n - 1, k)
            # grad[lo:hi+1] += 2 * pk * a[k-hi : k-lo+1] reversed
            grad[lo:hi + 1] += 2.0 * pk * a[k - hi:k - lo + 1][::-1]

        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-15:
            break

        lr = lr_init / (1.0 + 0.1 * it) ** 0.5

        a_new = a - lr * grad
        a_new = np.maximum(a_new, 0)
        s = a_new.sum()
        if s < 1e-10:
            a_new = a.copy()
        else:
            a_new = a_new / s

        M_new = fast_conv(a_new).max()

        if M_new < best_M:
            best_M = M_new
            best_a = a_new.copy()

        avg += a_new
        count += 1
        a = a_new

        if it % 500 == 0:
            print(f"    SG iter={it}, M={M_new:.8f}, best={best_M:.8f}, lr={lr:.4f}",
                  flush=True)

    # Check average
    if count > 10:
        a_avg = avg / count
        a_avg = np.maximum(a_avg, 0)
        s = a_avg.sum()
        if s > 1e-10:
            a_avg /= s
            M_avg = fast_conv(a_avg).max()
            if M_avg < best_M:
                best_M = M_avg
                best_a = a_avg.copy()

    return best_a, best_M

    # cross_corr[i] = sum_j p[j+i] * a[j] ... hmm
    # Let me just use the loop but skip small p values
    grad = np.zeros(n)
    active = np.where(p > 1e-12)[0]
    for k in active:
        pk = p[k]
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        grad[lo:hi + 1] += 2.0 * pk * a[k - hi:k - lo + 1][::-1]

    return grad, p, b


def run_smooth_gradient(a, max_iters, deadline, tau_init=3.0, tau_final=100.0,
                        lr_init=0.3, lr_final=0.01, use_momentum=True):
    """Smooth gradient descent with adaptive parameters."""
    a = np.maximum(a.astype(np.float64), 1e-10)
    a = a / a.sum()
    n = len(a)

    best_a = a.copy()
    best_M = fast_conv(a).max()
    v = np.zeros(n)

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        frac = min(1.0, it / max(1, max_iters))
        tau = tau_init * (tau_final / tau_init) ** (frac ** 0.5)
        lr = lr_init * (lr_final / lr_init) ** frac

        grad, p, b = smooth_max_gradient_vec(a, tau)
        M = b.max()

        if M < best_M * (1 - 1e-9):
            best_M = M
            best_a = a.copy()

        if use_momentum:
            v = 0.9 * v + lr * grad
            update = v
        else:
            update = lr * grad

        a_new = a - update
        a_new = np.maximum(a_new, 0)
        s = a_new.sum()
        if s < 1e-10:
            a_new = a.copy()
        else:
            a_new = a_new / s
        a = a_new

    if fast_conv(best_a).max() <= fast_conv(a).max():
        return best_a
    return a


def resample_seq(src, dest_n):
    """Resample 1D sequence to new length via linear interpolation."""
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)


def generate_initializations(n, rng, parent=None):
    """Generate diverse initial sequences of length n."""
    inits = []
    t_arr = np.linspace(0, 1, n, endpoint=False)

    if parent is not None:
        p = resample_seq(parent, n)
        inits.append(p.copy())
        inits.append(np.maximum(p + rng.normal(0, 0.02, n), 0.01))
        inits.append(np.maximum(p + rng.normal(0, 0.05, n), 0.01))
        inits.append(np.maximum(p + rng.normal(0, 0.15, n), 0.01))

    inits.append(np.ones(n))
    inits.append(rng.uniform(0.5, 1.0, n))

    # Frequency structures
    for freq in [1, 2, 3, 5, 7, 11]:
        inits.append(np.maximum(1.0 + 0.3 * np.cos(2 * np.pi * freq * t_arr), 0.01))
        inits.append(np.maximum(1.0 + 0.4 * np.sin(2 * np.pi * freq * t_arr + 0.5), 0.01))

    # Chirp
    inits.append(np.maximum(1.0 + 0.3 * np.cos(2 * np.pi * 5 * t_arr ** 2), 0.01))

    # Gaussian bumps
    inits.append(np.exp(-3.0 * (t_arr - 0.5) ** 2) + 0.1)
    inits.append(np.exp(-6.0 * (t_arr - 0.3) ** 2) + np.exp(-6.0 * (t_arr - 0.7) ** 2) + 0.05)
    inits.append(np.exp(-10 * (t_arr - 0.2) ** 2) + np.exp(-10 * (t_arr - 0.8) ** 2) + 0.1)

    # Triangle
    inits.append(np.maximum(1.0 - 0.4 * np.abs(2 * t_arr - 1), 0.05))
    inits.append(0.3 + 0.7 * np.abs(2 * t_arr - 1))

    # Smooth random
    ksize = max(3, n // 20)
    kernel = np.ones(ksize) / ksize
    smooth = np.convolve(rng.uniform(0, 1, n), kernel, mode='same')
    inits.append(np.maximum(smooth + 0.1, 0.01))

    # Block structure
    n_blocks = rng.integers(4, max(5, n // 30))
    block_vals = rng.uniform(0.3, 1.0, n_blocks)
    block_seq = np.repeat(block_vals, n // n_blocks + 1)[:n]
    inits.append(block_seq)

    # Multi-frequency
    inits.append(np.maximum(1.0 + 0.2 * np.cos(4 * np.pi * t_arr) +
                            0.15 * np.cos(7 * np.pi * t_arr) +
                            0.1 * np.sin(11 * np.pi * t_arr), 0.01))

    # Power-law
    inits.append(np.maximum(0.3 + 0.7 * t_arr ** 0.5, 0.01))
    inits.append(np.maximum(0.3 + 0.7 * (1 - t_arr) ** 0.5, 0.01))

    # Alternate high/low
    alt = np.ones(n)
    alt[::2] = 0.4
    inits.append(alt)
    alt2 = np.ones(n)
    alt2[::3] = 0.3
    inits.append(alt2)

    return inits


class ScoreTracker:
    def __init__(self):
        self.best = None
        self.best_score = float('inf')
        self.t_start = time.monotonic()

    def update(self, a):
        a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
        if a.size == 0 or a.sum() < 0.01:
            return
        score = float(evaluate_sequence(a.tolist()))
        if np.isfinite(score) and score < self.best_score:
            self.best = a.copy()
            self.best_score = score
            elapsed = time.monotonic() - self.t_start
            print(f"  ** NEW BEST: J={score:.8f} n={a.size} t={elapsed:.0f}s", flush=True)

    def report(self, label=""):
        if self.best is not None:
            print(f"  {label} best: J={self.best_score:.8f} n={self.best.size}", flush=True)


def coordinate_descent_polish(a, deadline, max_sweeps=50):
    """
    Coordinate descent: for each coordinate, find the best value
    that reduces the max convolution.
    """
    a = np.maximum(a.astype(np.float64), 0).copy()
    a = a / a.sum()
    n = len(a)
    best_M = fast_conv(a).max()

    for sweep in range(max_sweeps):
        if time.monotonic() > deadline:
            break

        improved = False
        for i in range(n):
            # Try a few perturbations of a[i]
            old_val = a[i]
            best_local = old_val
            best_local_M = fast_conv(a).max()

            # Try: increase, decrease, zero
            candidates = [old_val * 0.5, old_val * 1.5, old_val * 0.8,
                          old_val * 1.2, old_val * 0.3, 0.0]
            # Also try values based on neighbors
            if i > 0:
                candidates.append(a[i - 1] * 0.9)
            if i < n - 1:
                candidates.append(a[i + 1] * 0.9)

            for c in candidates:
                if c < 0 or c > 2.0:
                    continue
                a[i] = c
                s = a.sum()
                if s < 1e-10:
                    a[i] = best_local
                    continue
                b = fast_conv(a)
                m = b.max()
                # We want to minimize max(b) / s^2 * 2n, but since 2n is fixed,
                # minimize max(b) / s^2
                val = m / (s * s)
                if val < best_local_M / (a.sum() ** 2) - 1e-15:
                    best_local = c
                    best_local_M = m

            a[i] = best_local

        # Renormalize
        s = a.sum()
        if s > 1e-10:
            a /= s
            M = fast_conv(a).max()
            if M < best_M - 1e-12:
                best_M = M
                improved = True

        if not improved and sweep > 5:
            break

        if sweep % 5 == 0:
            print(f"    CD sweep={sweep}, M={fast_conv(a).max():.8f}", flush=True)

    return a, best_M


def propose_candidate(seed=42, budget_s=1500, **kwargs):
    """
    Multi-phase search:
    Phase 1: Fast gradient descent across lengths (200s)
    Phase 2: LP mirror descent with Nesterov (700s)
    Phase 3: Subgradient with LP dual (200s)
    Phase 4: Coordinate descent polish (150s)
    Phase 5: Multi-start + extension (remaining)
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + float(budget_s) - 10.0
    tracker = ScoreTracker()

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        if p.ndim == 1 and p.size > 0 and np.all(np.isfinite(p)):
            parent = np.maximum(p, 0.0)
            tracker.update(parent)
            print(f"Parent loaded: n={parent.size}, J={tracker.best_score:.6f}", flush=True)
    except (NameError, AttributeError):
        pass

    if parent is None:
        parent = np.ones(512)
        tracker.update(parent)

    current_best = parent.copy()

    # === PHASE 1: Fast smooth gradient across lengths ===
    p1_deadline = time.monotonic() + 200.0
    print(f"\n{'='*60}", flush=True)
    print(f"PHASE 1: Smooth Gradient Descent (budget=200s)", flush=True)
    print(f"{'='*60}", flush=True)

    lengths = [256, 384, 512, 640, 768, 1024]
    for n in lengths:
        if time.monotonic() > p1_deadline - 20:
            break

        inits = generate_initializations(n, rng, current_best)
        best_for_n = None
        best_M_for_n = float('inf')

        # Run gradient on subset of inits (time-limited)
        inits_per_n = max(3, min(8, int(30 / max(1, n / 256))))
        for idx, a0 in enumerate(inits[:inits_per_n]):
            if time.monotonic() > p1_deadline - 20:
                break
            a0 = np.maximum(a0.astype(np.float64), 1e-6)
            a0 = a0 / a0.sum()

            iters = max(80, int(3000 / max(1, n // 128)))
            a = run_smooth_gradient(a0, iters, min(time.monotonic() + 25, p1_deadline - 20),
                                    tau_init=3.0, tau_final=80.0,
                                    lr_init=0.25, lr_final=0.02)

            M_val = fast_conv(a).max()
            if M_val < best_M_for_n:
                best_M_for_n = M_val
                best_for_n = a.copy()

        if best_for_n is not None:
            J_val = compute_J(best_for_n)
            print(f"  n={n:4d}: J={J_val:.6f} ({time.monotonic()-tracker.t_start:.0f}s)",
                  flush=True)
            if J_val < compute_J(current_best):
                current_best = best_for_n.copy()
            tracker.update(best_for_n)

    tracker.report("Phase 1 done")

    # === PHASE 2: LP Mirror Descent (main workhorse) ===
    p2_deadline = time.monotonic() + 700.0
    if time.monotonic() > deadline - 100:
        p2_deadline = deadline - 50

    print(f"\n{'='*60}", flush=True)
    print(f"PHASE 2: LP Mirror Descent (budget={p2_deadline - time.monotonic():.0f}s)", flush=True)
    print(f"{'='*60}", flush=True)

    # Run on the best sequence found so far
    if tracker.best is not None:
        n_best = tracker.best.size
        print(f"  Starting LP mirror descent on n={n_best}", flush=True)
        a, M = run_lp_mirror_descent(tracker.best, p2_deadline, max_iters=100000)
        tracker.update(a)

        # If time remains, do another pass
        if time.monotonic() < p2_deadline - 30 and tracker.best is not None:
            print(f"  Second LP pass...", flush=True)
            a, M = run_lp_mirror_descent(tracker.best, p2_deadline, max_iters=50000)
            tracker.update(a)

    tracker.report("Phase 2 done")

    # === PHASE 3: Subgradient with LP dual ===
    p3_deadline = min(time.monotonic() + 200.0, deadline - 80)
    print(f"\n{'='*60}", flush=True)
    print(f"PHASE 3: Subgradient with LP Dual (budget={max(0, p3_deadline - time.monotonic()):.0f}s)",
          flush=True)
    print(f"{'='*60}", flush=True)

    if tracker.best is not None and time.monotonic() < p3_deadline:
        a, M = run_subgradient_lp(tracker.best, p3_deadline, max_iters=50000)
        tracker.update(a)

    tracker.report("Phase 3 done")

    # === PHASE 4: Coordinate Descent Polish ===
    p4_deadline = min(time.monotonic() + 120.0, deadline - 30)
    print(f"\n{'='*60}", flush=True)
    print(f"PHASE 4: Coordinate Descent Polish (budget={max(0, p4_deadline - time.monotonic()):.0f}s)",
          flush=True)
    print(f"{'='*60}", flush=True)

    if tracker.best is not None and time.monotonic() < p4_deadline:
        n = tracker.best.size
        if n <= 768:  # CD is O(n^2) per sweep
            a, M = coordinate_descent_polish(tracker.best, p4_deadline, max_sweeps=30)
            tracker.update(a)

    tracker.report("Phase 4 done")

    # === PHASE 5: Multi-start + length extension ===
    print(f"\n{'='*60}", flush=True)
    print(f"PHASE 5: Multi-start + Extension (remaining time)", flush=True)
    print(f"{'='*60}", flush=True)

    if tracker.best is not None:
        # Try different lengths by resampling
        base_n = tracker.best.size
        try_lengths = set()
        if base_n <= 512:
            try_lengths.update([base_n * 2, base_n + 128])
        if base_n <= 768:
            try_lengths.update([base_n + 128, base_n + 256])
        try_lengths = [n for n in try_lengths if 256 <= n <= 1536]

        for new_n in sorted(try_lengths):
            if time.monotonic() > deadline - 30:
                break
            a_ext = resample_seq(tracker.best, new_n)
            a_ext = np.maximum(a_ext, 1e-6)
            a_ext = a_ext / a_ext.sum()

            # Quick gradient + short LP
            a_ext = run_smooth_gradient(a_ext, 100, min(time.monotonic() + 30, deadline - 30),
                                        tau_init=3, tau_final=50, lr_init=0.2, lr_final=0.03)
            tracker.update(a_ext)

            if time.monotonic() < deadline - 40:
                a_ext, _ = run_lp_mirror_descent(a_ext, deadline - 30, max_iters=5000)
                tracker.update(a_ext)

        # Noise restarts on best
        if tracker.best is not None:
            n_best = tracker.best.size
            for r in range(10):
                if time.monotonic() > deadline - 20:
                    break
                noise_scale = 0.03 * (1.0 + 0.5 * r)
                a = tracker.best + rng.normal(0, noise_scale * 0.02, n_best)
                a = np.maximum(a, 1e-4)
                a = a / a.sum()

                a = run_smooth_gradient(a, 50, min(time.monotonic() + 20, deadline - 20),
                                        tau_init=5, tau_final=60, lr_init=0.15, lr_final=0.03)
                tracker.update(a)

                if time.monotonic() < deadline - 30:
                    a, _ = run_lp_mirror_descent(a, deadline - 20, max_iters=2000)
                    tracker.update(a)

    tracker.report("Phase 5 done")

    # === Final: one more LP pass ===
    if tracker.best is not None and time.monotonic() < deadline - 5:
        a, _ = run_lp_mirror_descent(tracker.best, deadline, max_iters=10000)
        tracker.update(a)

    # Fallback
    if tracker.best is None:
        tracker.best = np.ones(128)
        tracker.best_score = compute_J(tracker.best)

    print(f"\n{'='*60}", flush=True)
    print(f"FINAL: J={tracker.best_score:.8f} n={tracker.best.size} "
          f"time={time.monotonic()-tracker.t_start:.0f}s", flush=True)
    print(f"{'='*60}", flush=True)

    return [float(x) for x in tracker.best]
