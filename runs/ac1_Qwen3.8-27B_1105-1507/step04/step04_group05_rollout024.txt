import numpy as np
from scipy.optimize import linprog
from scipy.sparse import lil_matrix, csr_matrix
import time

def fast_conv(a):
    n = len(a)
    s = 1
    while s < 2 * n - 1:
        s <<= 1
    fa = np.fft.rfft(a, s)
    return np.fft.irfft(fa * fa, s)[:2 * n - 1]

def compute_J(a):
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))

def build_conv_matrix(a):
    """Build sparse matrix A where A @ g = convolve(a, g) (full)."""
    n = len(a)
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        jl = max(0, k - n + 1)
        jh = min(n - 1, k)
        for j in range(jl, jh + 1):
            rows.append(k)
            cols.append(j)
            vals.append(a[k - j])
    A = lil_matrix((2 * n - 1, n))
    A[np.array(rows), np.array(cols)] = np.array(vals)
    return A.tocsr()

def lp_solve(a, M, prev_x=None):
    """Solve LP: max sum(g) s.t. conv(a,g) <= M, g >= 0."""
    n = len(a)
    c = -np.ones(n)
    A = build_conv_matrix(a)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    opts = {'presolve': True}
    if prev_x is not None:
        # Can't easily warm-start linprog, but we can try
        pass
    r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs', options=opts)
    if r.success and r.x is not None:
        return np.maximum(r.x, 0.0)
    return None

def coordinate_descent_pass(a, n_iter_max=50):
    """One pass of coordinate descent with golden section search."""
    a = a.astype(np.float64).copy()
    n = len(a)
    S = a.sum()
    if S < 1e-10:
        return a
    b = fast_conv(a)
    current_J = 2.0 * n * b.max() / (S * S)
    
    indices = list(range(n))
    np.random.shuffle(indices)
    
    for i in indices:
        ai = a[i]
        
        def obj(x):
            new_S = S - ai + x
            if new_S < 1e-10:
                return 1e10
            # Only need to check convolution coefficients that involve a[i]
            # These are b[k] for k in [i, i+n-1]
            max_c = b.max()  # Start with current max for positions not involving i
            for k in range(max(0, i), min(2*n-1, i+n)):
                ki = k - i
                if 0 <= ki < n:
                    if k == 2 * i:
                        ck = b[k] - ai*ai + x*x
                    else:
                        ck = b[k] - 2.0*ai*a[ki] + 2.0*x*a[ki]
                    if ck > max_c:
                        max_c = ck
            return 2.0 * n * max_c / (new_S * new_S)
        
        # Golden section search
        gr = (np.sqrt(5.0) - 1.0) / 2.0
        lo = 0.0
        hi = max(3.0 * ai, 3.0 * S / n, 0.01)
        
        # Check if current point is good enough
        f_lo = obj(lo)
        f_hi = obj(hi)
        f_mid = obj(ai)
        
        if f_hi < f_mid:
            # Optimum is beyond ai
            lo = ai
            f_lo = f_mid
        else:
            hi = ai
            f_hi = f_mid
        
        c_pt = hi - gr * (hi - lo)
        d_pt = lo + gr * (hi - lo)
        fc = obj(c_pt)
        fd = obj(d_pt)
        
        for _ in range(30):
            if fc < fd:
                hi, d_pt, fd = d_pt, c_pt, fc
                c_pt = hi - gr * (hi - lo)
                fc = obj(c_pt)
            else:
                lo, c_pt, fc = c_pt, d_pt, fd
                d_pt = lo + gr * (hi - lo)
                fd = obj(d_pt)
        
        x_opt = max(0.0, (lo + hi) / 2.0)
        
        if abs(x_opt - ai) > 1e-12:
            a[i] = x_opt
            S = S - ai + x_opt
            # Update convolution
            for k in range(max(0, i), min(2*n-1, i+n)):
                ki = k - i
                if 0 <= ki < n:
                    if k == 2 * i:
                        b[k] = b[k] - ai*ai + x_opt*x_opt
                    else:
                        b[k] = b[k] - 2.0*ai*a[ki] + 2.0*x_opt*a[ki]
    
    # Verify
    b_refresh = fast_conv(a)
    new_J = 2.0 * n * b_refresh.max() / (S * S)
    return a, new_J

def active_set_leveling(a, n_levels=5, n_iters=20):
    """Try to equalize the top n_levels convolution peaks."""
    a = a.astype(np.float64).copy()
    n = len(a)
    S = a.sum()
    
    for _ in range(n_iters):
        b = fast_conv(a)
        # Find top peaks
        top_idx = np.argsort(b)[-n_levels:]
        target = b[top_idx].mean()
        
        # For each top peak, reduce it by adjusting contributing a[j] values
        # Simple approach: for each peak at position k, reduce a[j] for j that contribute to b[k]
        changed = False
        for k in top_idx:
            if b[k] <= target:
                continue
            # Find contributing indices
            jl = max(0, k - n + 1)
            jh = min(n - 1, k)
            excess = b[k] - target
            # Reduce a[j] proportionally to a[j]*a[k-j]
            contribs = np.array([a[j] * a[k-j] for j in range(jl, jh+1)])
            total_contrib = contribs.sum()
            if total_contrib < 1e-15:
                continue
            # Reduce each a[j] by a small amount
            reduction = min(0.1, excess / (2 * total_contrib + 1e-15))
            for j in range(jl, jh+1):
                a[j] = max(0, a[j] - reduction * a[k-j])
            changed = True
        
        if not changed:
            break
    
    a = np.maximum(a, 0)
    if a.sum() > 0:
        a /= a.sum()
    return a

def generate_initializations(n, rng):
    """Generate diverse initial sequences."""
    inits = []
    t = np.linspace(0, 1, n, endpoint=False)
    
    # Uniform
    inits.append(np.ones(n))
    
    # Random
    inits.append(rng.uniform(0.5, 1.0, n))
    
    # Cosine with various frequencies
    for freq in [1, 2, 3, 4, 5, 7, 11]:
        inits.append(0.7 + 0.3 * np.cos(2 * np.pi * freq * t))
    
    # Sum of cosines
    inits.append(0.5 + 0.3*np.cos(2*np.pi*2*t) + 0.2*np.cos(2*np.pi*5*t))
    inits.append(0.5 + 0.25*np.cos(2*np.pi*3*t) + 0.25*np.cos(2*np.pi*7*t))
    
    # Exponential decay from center
    center = n // 2
    inits.append(np.exp(-0.5 * ((np.arange(n) - center) / (n/4))**2) + 0.1)
    
    # Two peaks
    inits.append(0.5*np.exp(-20*(t-0.3)**2) + 0.5*np.exp(-20*(t-0.7)**2) + 0.1)
    
    # Triangle
    inits.append(1.0 - 0.5*np.abs(2*t - 1.0))
    
    # B-spline like (convolution of rectangles)
    rect = np.zeros(n)
    rect[:n//3] = 1.0
    inits.append(fast_conv(rect, rect)[:n] + 0.1)
    
    # Sawtooth
    inits.append(0.5 + 0.4 * (2 * (t - np.floor(t + 0.5))))
    
    # Random smooth (Gaussian process like)
    freqs = rng.uniform(0, 10, 5)
    phases = rng.uniform(0, 2*np.pi, 5)
    amps = rng.uniform(0.1, 0.3, 5)
    sig = 0.5 + sum(a * np.cos(2*np.pi*f*t + p) for a, f, p in zip(amps, freqs, phases))
    inits.append(np.maximum(sig, 0.1))
    
    return inits

def lp_mirror_descent(a, deadline, tracker, tighten=0.005, max_iters=500):
    """LP mirror descent with adaptive tightening."""
    a = np.maximum(a.astype(np.float64), 1e-6)
    a = a / a.sum()
    n = len(a)
    current_J = compute_J(a)
    tracker.update(a)
    
    stall = 0
    best_J = current_J
    prev_g = None
    
    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        
        b = fast_conv(a)
        M = b.max()
        
        # Try progressively tighter constraints
        found = False
        for tight_factor in [tighten, tighten*2, tighten*3, tighten*5, tighten*10]:
            M_target = M * (1.0 - tight_factor)
            if M_target < 0:
                M_target = M * 0.5
            g = lp_solve(a, M_target, prev_g)
            if g is not None and g.sum() > 1e-10:
                found = True
                break
        
        if not found:
            stall += 1
            if stall > 30:
                break
            continue
        
        g = g / g.sum()
        prev_g = g
        
        # Try multiple step sizes
        best_a = a
        best_J_trial = current_J
        step_candidates = [0.3, 0.5, 0.7, 0.8, 0.9, 1.0]
        
        for t in step_candidates:
            a_trial = (1 - t) * a + t * g
            a_trial = np.maximum(a_trial, 1e-8)
            a_trial = a_trial / a_trial.sum()
            J_trial = compute_J(a_trial)
            if J_trial < best_J_trial:
                best_J_trial = J_trial
                best_a = a_trial
        
        if best_J_trial < current_J - 1e-12:
            a = best_a
            current_J = best_J_trial
            stall = 0
            if current_J < best_J:
                best_J = current_J
            if it % 20 == 0:
                tracker.update(a)
        else:
            stall += 1
            tighten = max(0.001, tighten * 0.8)
    
    tracker.update(a)
    return a, current_J

def gradient_descent(a, deadline, tracker, lr=None):
    """Subgradient descent on J(a)."""
    a = np.maximum(a.astype(np.float64), 1e-6)
    a = a / a.sum()
    n = len(a)
    current_J = compute_J(a)
    tracker.update(a)
    
    if lr is None:
        lr = 0.01
    
    for it in range(10000):
        if time.monotonic() > deadline:
            break
        
        b = fast_conv(a)
        b_max = b.max()
        S = a.sum()
        
        # Subgradient of max_b / S^2 w.r.t. a
        # d(max_b)/da[j] = sum over active k of a[k-j] (where k-j is valid)
        tol = 1e-4 * b_max
        active = np.where(b >= b_max - tol)[0]
        
        grad_max = np.zeros(n)
        for k in active:
            jl = max(0, k - n + 1)
            jh = min(n - 1, k)
            for j in range(jl, jh + 1):
                grad_max[j] += a[k - j]
        
        # J = 2n * max_b / S^2
        # dJ/da[j] = 2n * (grad_max[j] * S^2 - max_b * 2*S) / S^4
        #         = 2n * (grad_max[j] * S - 2*max_b) / S^3
        grad = (grad_max * S - 2.0 * b_max)
        
        # Projected gradient step
        a_new = a - lr * grad
        a_new = np.maximum(a_new, 0)
        if a_new.sum() < 1e-10:
            break
        a_new = a_new / a_new.sum()
        
        J_new = compute_J(a_new)
        if J_new < current_J:
            a = a_new
            current_J = J_new
            if it % 100 == 0:
                tracker.update(a)
        else:
            lr *= 0.9
            if lr < 1e-8:
                break
    
    tracker.update(a)
    return a, current_J

"""
Minimizes J(a) = 2*n*max(convolve(a,a))/(sum(a))^2 using:
1. LP mirror descent with adaptive M-tightening
2. Soft-max gradient descent (smooth approximation of max)
3. Vectorized coordinate descent with golden section search
4. Active-set leveling (equalize top peaks)
5. Multi-length exploration and perturbation restarts
6. Final intensive polish

import numpy as np
from scipy.optimize import linprog
from scipy.sparse import lil_matrix
import time


def fast_conv(a):
    """Full self-convolution via FFT."""
    n = len(a)
    s = 1
    while s < 2 * n - 1:
        s <<= 1
    fa = np.fft.rfft(a, s)
    return np.fft.irfft(fa * fa, s)[:2 * n - 1]


def compute_J(a):
    """Compute J(a) = 2*n*max(conv(a,a))/(sum(a))^2."""
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))


def build_conv_matrix(a):
    """Build sparse matrix A where (A @ g)[k] = sum_j a[k-j]*g[j] (full conv)."""
    n = len(a)
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        jl = max(0, k - n + 1)
        jh = min(n - 1, k)
        for j in range(jl, jh + 1):
            rows.append(k)
            cols.append(j)
            vals.append(a[k - j])
    if not rows:
        return lil_matrix((2 * n - 1, n))
    r_arr = np.array(rows)
    c_arr = np.array(cols)
    v_arr = np.array(vals)
    A = lil_matrix((2 * n - 1, n))
    A[r_arr, c_arr] = v_arr
    return A.tocsr()


def lp_solve(a, M):
    """Solve LP: max sum(g) s.t. conv(a,g) <= M elementwise, g >= 0."""
    n = len(a)
    c = -np.ones(n)
    A = build_conv_matrix(a)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                options={'presolve': True, 'time_limit': 10})
    if r.success and r.x is not None:
        return np.maximum(r.x, 0.0)
    return None


def soft_max_gradient(w, a, T):
    """Compute gradient of log-sum-exp(conv(a,a)/T) w.r.t. a.
    Returns grad[j] = sum_k (w[k]/Z) * a[k-j] for valid indices."""
    n = len(a)
    b = fast_conv(a)
    b_shifted = b - b.max()
    weights = np.exp(b_shifted / T)
    Z = weights.sum()
    w_norm = weights / Z
    
    grad = np.zeros(n)
    for j in range(n):
        k_start = j
        k_end = min(j + n, 2 * n - 1)
        if k_start < k_end:
            grad[j] = np.dot(w_norm[k_start:k_end], a[:k_end - k_start])
    return grad, b.max()


def soft_max_descent(a, deadline, tracker, T_init=None, lr_init=0.1, max_iters=5000):
    """Minimize soft-max approximation of max(conv(a,a)) via gradient descent."""
    a = np.maximum(a.astype(np.float64), 1e-6)
    a = a / a.sum()
    n = len(a)
    current_J = compute_J(a)
    tracker.update(a)
    
    b = fast_conv(a)
    if T_init is None:
        T = b.max() * 0.15
    else:
        T = T_init
    lr = lr_init
    
    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        
        grad, b_max = soft_max_gradient(None, a, T)
        
        # Normalize gradient for stable updates
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-15:
            break
        grad = grad / gnorm
        
        a_new = a - lr * grad
        a_new = np.maximum(a_new, 1e-8)
        sn = a_new.sum()
        if sn < 1e-10:
            break
        a_new = a_new / sn
        
        J_new = compute_J(a_new)
        if J_new < current_J - 1e-14:
            a = a_new
            current_J = J_new
            T = max(T * 0.995, b_max * 0.02)
            if it % 100 == 0:
                tracker.update(a)
        else:
            lr *= 0.95
            if lr < 1e-10:
                break
    
    tracker.update(a)
    return a, current_J


def lp_mirror_descent(a, deadline, tracker, tighten=0.004, max_iters=300):
    """LP mirror descent with adaptive tightening."""
    a = np.maximum(a.astype(np.float64), 1e-6)
    a = a / a.sum()
    n = len(a)
    current_J = compute_J(a)
    tracker.update(a)
    
    stall = 0
    
    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        
        b = fast_conv(a)
        M = float(b.max())
        
        found = False
        for tight_factor in [tighten, tighten * 2, tighten * 4, tighten * 8, tighten * 16]:
            M_target = M * (1.0 - tight_factor)
            if M_target < M * 0.3:
                M_target = M * 0.3
            g = lp_solve(a, M_target)
            if g is not None and g.sum() > 1e-8:
                found = True
                break
        
        if not found:
            stall += 1
            if stall > 25:
                break
            tighten = max(0.001, tighten * 0.7)
            continue
        
        g = g / g.sum()
        
        best_a = a
        best_J_trial = current_J
        step_candidates = [0.3, 0.5, 0.6, 0.7, 0.8, 0.9, 1.0]
        
        for t in step_candidates:
            a_trial = (1.0 - t) * a + t * g
            a_trial = np.maximum(a_trial, 1e-8)
            sn = a_trial.sum()
            if sn < 1e-10:

            a = best_a
            current_J = best_J_trial
            stall = 0
            if it % 15 == 0:
                tracker.update(a)
        else:
            stall += 1
            tighten = max(0.0005, tighten * 0.8)
    
    tracker.update(a)
    return a, current_J


def coordinate_descent_pass(a, rng=None):
    """One full pass of coordinate descent with golden section search."""
    a = a.astype(np.float64).copy()
    n = len(a)
    S = a.sum()
    if S < 1e-10:
        return a, compute_J(a)
    b = fast_conv(a)
    
    if rng is None:
        rng = np.random.default_rng()
    
    indices = list(range(n))
    rng.shuffle(indices)
    
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    b_max_exclude = b.max()
    
    for i in indices:
        ai = a[i]
        
        def obj(x):
            new_S = S - ai + x
            if new_S < 1e-10:
                return 1e10
            max_c = b_max_exclude
            for k in range(max(0, i), min(2 * n - 1, i + n)):
                ki = k - i
                if 0 <= ki < n:
                    if k == 2 * i:
                        ck = b[k] - ai * ai + x * x
                    else:
                        ck = b[k] - 2.0 * ai * a[ki] + 2.0 * x * a[ki]
                    if ck > max_c:
                        max_c = ck
            return 2.0 * n * max_c / (new_S * new_S)
        
        # Determine search bounds
        lo = 0.0
        hi = max(3.0 * ai + 0.01, 3.0 * S / max(n, 1))
        
        f_current = obj(ai)
        f_lo = obj(lo)
        f_hi = obj(hi)
        
        if f_current <= f_lo and f_current <= f_hi:
            continue
        
        if f_hi < f_lo:
            lo = ai
            f_lo = f_current
        else:
            hi = ai
            f_hi = f_current
        
        c_pt = hi - gr * (hi - lo)
        d_pt = lo + gr * (hi - lo)
        fc = obj(c_pt)
        fd = obj(d_pt)
        
        for _ in range(25):
            if fc < fd:
                hi, d_pt, fd = d_pt, c_pt, fc
                c_pt = hi - gr * (hi - lo)
                fc = obj(c_pt)
            else:
                lo, c_pt, fc = c_pt, d_pt, fd
                d_pt = lo + gr * (hi - lo)
                fd = obj(d_pt)
        
        x_opt = max(0.0, (lo + hi) / 2.0)
        
        if abs(x_opt - ai) > 1e-12:
            a[i] = x_opt
            S = S - ai + x_opt
            for k in range(max(0, i), min(2 * n - 1, i + n)):
                ki = k - i
                if 0 <= ki < n:
                    if k == 2 * i:
                        b[k] = b[k] - ai * ai + x_opt * x_opt
                    else:
                        b[k] = b[k] - 2.0 * ai * a[ki] + 2.0 * x_opt * a[ki]
            b_max_exclude = b.max()
    
    b_refresh = fast_conv(a)
    new_J = 2.0 * n * b_refresh.max() / (S * S)
    return a, new_J


def active_set_leveling(a, n_levels=5, n_iters=15):
    """Equalize the top n_levels convolution peaks by reducing contributing elements."""
    a = a.astype(np.float64).copy()
    n = len(a)
    
    for it in range(n_iters):
        b = fast_conv(a)
        b_max = b.max()
        top_idx = np.argsort(b)[-n_levels:]
        target = b[top_idx].mean()
        
        changed = False
        for k in top_idx:
            if b[k] <= target + 1e-10:
                continue
            jl = max(0, k - n + 1)
            jh = min(n - 1, k)
            excess = b[k] - target
            if excess < 1e-12:
                continue
            
            contribs = []
            for j in range(jl, jh + 1):
                contribs.append(a[j] * a[k - j])
            contribs = np.array(contribs)
            total = contribs.sum()
            if total < 1e-15:
                continue
            
            reduction = min(0.15, excess / (2.0 * total + 1e-15))
            for idx, j in enumerate(range(jl, jh + 1)):
                new_val = a[j] - reduction * a[k - j]
                if new_val < 0:
                    new_val = 0.0
                a[j] = new_val
            changed = True
        
        if not changed:
            break
        a = np.maximum(a, 0)
        if a.sum() < 1e-10:
            break
    
    if a.sum() > 0:
        a = a / a.sum()
    return a


def resample_seq(src, dest_n):
    """Resample 1D sequence to new length via linear interpolation."""
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    if src.size == 1:
        return np.full(dest_n, src[0])
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)


def generate_diverse_inits(n, rng):
    """Generate diverse initial sequences of length n."""
    inits = []
    t = np.linspace(0, 1, n, endpoint=False)
    
    inits.append(np.ones(n))
    inits.append(rng.uniform(0.4, 1.0, n))
    
    for freq in [1, 2, 3, 4, 5, 7, 8, 11, 13]:
        inits.append(0.6 + 0.4 * np.cos(2 * np.pi * freq * t + rng.uniform(0, 6.28)))
    
    inits.append(0.4 + 0.3 * np.cos(2 * np.pi * 2 * t) + 0.3 * np.cos(2 * np.pi * 5 * t))
    inits.append(0.3 + 0.25 * np.cos(2 * np.pi * 3 * t) + 0.25 * np.cos(2 * np.pi * 7 * t) + 0.2 * np.cos(2 * np.pi * 11 * t))
    
    center = n / 2.0
    inits.append(np.exp(-2.0 * ((np.arange(n) - center) / (n / 3.0)) ** 2) + 0.1)
    
    inits.append(0.4 * np.exp(-15 * (t - 0.25) ** 2) + 0.4 * np.exp(-15 * (t - 0.75) ** 2) + 0.15)
    inits.append(0.3 * np.exp(-10 * (t - 0.2) ** 2) + 0.3 * np.exp(-10 * (t - 0.5) ** 2) + 0.3 * np.exp(-10 * (t - 0.8) ** 2) + 0.1)
    
    inits.append(1.0 - 0.5 * np.abs(2 * t - 1.0))
    
    rect = np.zeros(n)
    rect[:n // 4] = 1.0
    conv_rect = fast_conv(rect, rect)[:n]
    inits.append(conv_rect / conv_rect.max() * 0.8 + 0.1)
    
    # Random smooth
    n_freqs = 6
    freqs = rng.uniform(1, 15, n_freqs)
    phases = rng.uniform(0, 2 * np.pi, n_freqs)
    amps = rng.uniform(0.05, 0.2, n_freqs)
    sig = 0.4
    for f, p, am in zip(freqs, phases, amps):
        sig = sig + am * np.cos(2 * np.pi * f * t + p)
    inits.append(np.maximum(sig, 0.05))
    
    return inits


class ScoreTracker:
    def __init__(self):
        self.best = None
        self.best_score = float('inf')
        self.t_start = time.monotonic()
    
    def update(self, a):
        a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
        if a.size == 0 or a.sum() < 0.01:
            return
        score = float(evaluate_sequence(a.tolist()))
        if np.isfinite(score) and score < self.best_score:
            self.best = a.copy()
            self.best_score = score
            elapsed = time.monotonic() - self.t_start
            print(f"  * NEW BEST: J={score:.8f} n={a.size} t={elapsed:.0f}s", flush=True)
    
    def report(self, label=""):
        if self.best is not None:
            print(f"  {label} best: J={self.best_score:.8f} n={self.best.size} t={time.monotonic()-self.t_start:.0f}s", flush=True)


def full_optimize(a, deadline, tracker, n_lp=80, n_cd=3):
    """Alternate LP and coordinate descent until deadline."""
    a = np.maximum(a.astype(np.float64), 1e-6)
    a = a / a.sum()
    current_J = compute_J(a)
    tracker.update(a)
    rng = np.random.default_rng(123)
    
    it = 0
    while time.monotonic() < deadline:
        # LP phase
        lp_deadline = min(deadline, time.monotonic() + 15)
        a, J_lp = lp_mirror_descent(a, lp_deadline, tracker, tighten=0.003 / (1 + 0.1 * it), max_iters=20)
        
        if time.monotonic() >= deadline:
            break
        
        # CD phase
        cd_deadline = min(deadline, time.monotonic() + 10)
        while time.monotonic() < cd_deadline:
            a, J_cd = coordinate_descent_pass(a, rng)
            a = np.maximum(a, 0)
            if a.sum() > 0:
                a = a / a.sum()
            tracker.update(a)
        
        it += 1
        if it % 5 == 0:
            J = compute_J(a)
            print(f"    FullOpt iter {it}: J={J:.8f}", flush=True)
    
    tracker.update(a)
    return a


def propose_candidate(seed=42, budget_s=1500, **kwargs):
    """
    Main search function combining LP mirror descent, soft-max gradient descent,
    coordinate descent, active-set leveling, and multi-length exploration.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + float(budget_s) - 15.0
    tracker = ScoreTracker()
    
    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        if p.ndim == 1 and p.size > 0 and np.all(np.isfinite(p)):
            parent = np.maximum(p, 0.0)
            if parent.sum() > 0.01:
                tracker.update(parent)
                print(f"Parent loaded: n={parent.size}, J={tracker.best_score:.8f}", flush=True)
                parent = parent / parent.sum()
    except (NameError, AttributeError):
        pass
    
    if parent is None:
        parent = np.ones(384) / 384.0
        tracker.update(parent)
        print(f"No parent found, using uniform n={parent.size}", flush=True)
    
    n_parent = parent.size
    
    # === Phase 1: LP mirror descent on parent ===
    print(f"\n{'='*60}", flush=True)
    print(f"Phase 1: LP mirror descent (n={n_parent})", flush=True)
    print(f"{'='*60}", flush=True)
    t0 = time.monotonic()
    
    p1_deadline = min(deadline, time.monotonic() + 350)
    a1, J1 = lp_mirror_descent(parent.copy(), p1_deadline, tracker, tighten=0.005, max_iters=250)
    print(f"Phase 1: J={J1:.8f} in {time.monotonic()-t0:.1f}s", flush=True)
    
    # === Phase 2: Soft-max gradient descent ===
    if time.monotonic() < deadline - 250:
        print(f"\n{'='*60}", flush=True)
        print(f"Phase 2: Soft-max gradient descent", flush=True)
        print(f"{'='*60}", flush=True)
        t0 = time.monotonic()
        
        p2_deadline = min(deadline, time.monotonic() + 150)
        a2, J2 = soft_max_descent(a1.copy(), p2_deadline, tracker, T_init=None, lr_init=0.08)
        print(f"Phase 2: J={J2:.8f} in {time.monotonic()-t0:.1f}s", flush=True)
    
    # === Phase 3: Coordinate descent polish ===
    if time.monotonic() < deadline - 200:
        print(f"\n{'='*60}", flush=True)
        print(f"Phase 3: Coordinate descent polish", flush=True)
        print(f"{'='*60}", flush=True)
        t0 = time.monotonic()
        
        a3 = tracker.best.copy()
        a3 = a3 / a3.sum()
        for p in range(8):
            if time.monotonic() > deadline - 200:
                break
            a3, J_cd = coordinate_descent_pass(a3, rng)
            a3 = np.maximum(a3, 0)
            a3 = a3 / a3.sum()
            tracker.update(a3)
            print(f"  Pass {p}: J={J_cd:.8f}", flush=True)
        
        print(f"Phase 3 done in {time.monotonic()-t0:.1f}s", flush=True)
    
    # === Phase 4: Active set leveling + re-optimize ===
    if time.monotonic() < deadline - 150:
        print(f"\n{'='*60}", flush=True)
        print(f"Phase 4: Active set leveling", flush=True)
        print(f"{'='*60}", flush=True)
        t0 = time.monotonic()
        
        for n_lev in [3, 5, 8]:
            if time.monotonic() > deadline - 140:
                break
            a_lev = active_set_leveling(tracker.best.copy(), n_levels=n_lev, n_iters=20)
            tracker.update(a_lev)
            
            if time.monotonic() < deadline - 130:
                a_lev, _ = lp_mirror_descent(a_lev, min(deadline, time.monotonic() + 40), tracker,
                                             tighten=0.002, max_iters=30)
        
        print(f"Phase 4 done in {time.monotonic()-t0:.1f}s", flush=True)
    
    # === Phase 5: Full alternating optimization ===
    if time.monotonic() < deadline - 120:
        print(f"\n{'='*60}", flush=True)
        print(f"Phase 5: Full alternating LP+CD", flush=True)
        print(f"{'='*60}", flush=True)
        t0 = time.monotonic()
        
        p5_deadline = min(deadline, time.monotonic() + 100)
        a5 = full_optimize(tracker.best.copy(), p5_deadline, tracker)
        
        print(f"Phase 5 done in {time.monotonic()-t0:.1f}s", flush=True)
    
    # === Phase 6: Multi-length exploration ===
    if time.monotonic() < deadline - 80:
        print(f"\n{'='*60}", flush=True)
        print(f"Phase 6: Multi-length exploration", flush=True)
        print(f"{'='*60}", flush=True)
        t0 = time.monotonic()
        
        best_a = tracker.best
        deltas = [-6, -4, -2, 2, 4, 6, 8, 10, -10, 12, -8]
        
        for dl in deltas:
            nl = n_parent + dl
            if nl < 200 or nl > 800:
                continue
            if time.monotonic() > deadline - 50:
                break
            
            a_res = resample_seq(best_a, nl)
            a_res = np.maximum(a_res, 0.01)
            a_res = a_res / a_res.sum()
            
            a_res, _ = lp_mirror_descent(a_res, min(deadline, time.monotonic() + 25), tracker,
                                         tighten=0.003, max_iters=30)
            if time.monotonic() < deadline - 50:
                a_res, _ = coordinate_descent_pass(a_res, rng)
                a_res = np.maximum(a_res, 0)
                a_res = a_res / max(a_res.sum(), 1e-10)
                tracker.update(a_res)
            
            print(f"  n={nl}: best J={tracker.best_score:.8f}", flush=True)
        
        print(f"Phase 6 done in {time.monotonic()-t0:.1f}s", flush=True)
    
    # === Phase 7: Diverse initializations at best length ===
    if time.monotonic() < deadline - 60:
        print(f"\n{'='*60}", flush=True)
        print(f"Phase 7: Diverse initializations", flush=True)
        print(f"{'='*60}", flush=True)
        t0 = time.monotonic()
        
        n_best = tracker.best.size
        inits = generate_diverse_inits(n_best, rng)
        
        for idx, init in enumerate(inits):
            if time.monotonic() > deadline - 40:
                break
            init = np.maximum(init, 0.01)
            init = init / init.sum()
            
            # Quick LP
            a_init, _ = lp_mirror_descent(init, min(deadline, time.monotonic() + 15), tracker,
                                          tighten=0.003, max_iters=15)
        
        print(f"Phase 7 done in {time.monotonic()-t0:.1f}s", flush=True)
    
    # === Phase 8: Perturbation restarts ===
    if time.monotonic() < deadline - 40:
        print(f"\n{'='*60}", flush=True)
        print(f"Phase 8: Perturbation restarts", flush=True)
        print(f"{'='*60}", flush=True)
        t0 = time.monotonic()
        
        n = tracker.best.size
        for temp in [0.04, 0.025, 0.015, 0.008, 0.004]:
            if time.monotonic() > deadline - 25:
                break
            a_pert = tracker.best + rng.normal(0, temp, n)
            a_pert = np.maximum(a_pert, 0.001)
            a_pert = a_pert / a_pert.sum()
            
            a_pert, _ = lp_mirror_descent(a_pert, min(deadline, time.monotonic() + 18), tracker,
                                          tighten=0.002, max_iters=25)
            if time.monotonic() < deadline - 25:
                a_pert, _ = coordinate_descent_pass(a_pert, rng)
                a_pert = np.maximum(a_pert, 0)
                a_pert = a_pert / max(a_pert.sum(), 1e-10)
                tracker.update(a_pert)
        
        print(f"Phase 8 done in {time.monotonic()-t0:.1f}s", flush=True)
    
    # === Phase 9: Final intensive polish ===
    if time.monotonic() < deadline - 10:
        print(f"\n{'='*60}", flush=True)
        print(f"Phase 9: Final polish", flush=True)
        print(f"{'='*60}", flush=True)
        t0 = time.monotonic()
        
        a_final = tracker.best.copy()
        a_final = a_final / a_final.sum()
        
        # Very tight LP
        for tighten in [0.001, 0.0005, 0.0003]:
            if time.monotonic() > deadline - 5:
                break
            b = fast_conv(a_final)
            M = b.max()
            M_target = M * (1.0 - tighten)
            g = lp_solve(a_final, M_target)
            if g is None or g.sum() < 1e-10:
                continue
            g = g / g.sum()
            for t in [0.5, 0.7, 0.8, 0.9, 1.0]:
                a_t = (1 - t) * a_final + t * g
                a_t = np.maximum(a_t, 1e-8)
                a_t = a_t / a_t.sum()
                J_t = compute_J(a_t)
                J_cur = compute_J(a_final)
                if J_t < J_cur:
                    a_final = a_t
                    tracker.update(a_final)
        
        # Final CD
        if time.monotonic() < deadline - 3:
            a_final, J_final = coordinate_descent_pass(a_final, rng)
            a_final = np.maximum(a_final, 0)
            a_final = a_final / max(a_final.sum(), 1e-10)
            tracker.update(a_final)
            print(f"  Final CD: J={J_final:.8f}", flush=True)
        
        # One more leveling
        if time.monotonic() < deadline - 2:
            a_lev = active_set_leveling(tracker.best.copy(), n_levels=4, n_iters=10)
            tracker.update(a_lev)
        
        print(f"Phase 9 done in {time.monotonic()-t0:.1f}s", flush=True)
    
    # Fallback
    if tracker.best is None:
        tracker.best = np.ones(256) / 256.0
        tracker.best_score = compute_J(tracker.best)
    
    print(f"\n{'='*60}", flush=True)
    print(f"FINAL RESULT: J = {tracker.best_score:.10f}, n = {tracker.best.size}", flush=True)
    print(f"Total time: {time.monotonic()-tracker.t_start:.1f}s", flush=True)
    print(f"{'='*60}", flush=True)
    
    return [float(x) for x in tracker.best]
