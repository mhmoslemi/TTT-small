def build_toeplitz_sparse(a):
    """Build sparse Toeplitz matrix T such that T @ g = conv(a, g)."""
    n = len(a)
    from scipy.sparse import lil_matrix
    T = lil_matrix((2 * n - 1, n))
    for j in range(n):
        for i in range(n):
            T[i + j, j] = a[i]
    return T.tocsr()


def solve_lp_maxsum(a, M, time_limit=5):
    """Solve LP: max sum(g) s.t. conv(a,g) <= M, g >= 0."""
    n = len(a)
    T = build_toeplitz_sparse(a)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c, A_ub=T, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': time_limit})
        if res.status == 0:
            return np.maximum(res.x, 0)
    except Exception:
        pass
    return None


def alternating_lp(a, max_iters=60, time_budget=40.0):
    """Alternating LP with adaptive M and line search."""
    n = len(a)
    a = a.copy().astype(np.float64)
    S = np.sum(a)
    if S < 1e-10:
        return a, np.inf
    a = a / S
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    stall = 0
    
    for it in range(max_iters):
        if time.monotonic() - start > time_budget:
            break
        
        conv = compute_conv(a)
        M = float(np.max(conv))
        
        g = solve_lp_maxsum(a, M, time_limit=4)
        if g is None:
            break
        
        S_g = np.sum(g)
        if S_g < 1e-10:
            break
        g = g / S_g
        
        # Line search
        best_t = 0.0
        best_obj_ls = best_obj
        for t in [0.05, 0.1, 0.15, 0.25, 0.4, 0.6, 0.8, 1.0]:
            a_try = (1.0 - t) * a + t * g
            a_try = np.maximum(a_try, 0)
            S_try = np.sum(a_try)
            if S_try < 1e-10:
                continue
            a_try = a_try / S_try
            obj_try = compute_obj(a_try)
            if obj_try < best_obj_ls:
                best_obj_ls = obj_try
                best_t = t
        
        if best_t > 0.0:
            a = (1.0 - best_t) * a + best_t * g
            a = np.maximum(a, 0)
            a = a / np.sum(a)
            best_obj = best_obj_ls
            best_a = a.copy()
            stall = 0
        else:
            stall += 1
            if stall > 10:
                rng = np.random.default_rng(it * 7919 + 123)
                a = a * np.exp(rng.normal(0, 0.06, n))
                a = np.maximum(a, 0)
                a = a / np.sum(a)
                stall = 0
            else:
                # Small blend to avoid getting stuck
                a = 0.95 * a + 0.05 * g
                a = np.maximum(a, 0)
                a = a / np.sum(a)
    
    return best_a, best_obj


def cosine_init(n, n_modes=10, seed=0):
    """Generate a smooth cosine-series initialization."""
    rng = np.random.default_rng(seed)
    k = np.arange(n)
    a = np.ones(n)
    for m in range(1, n_modes + 1):
        c = rng.uniform(-0.5, 0.5) / m
        phase = rng.uniform(0, 2 * np.pi)
        a += c * np.cos(2 * np.pi * m * k / n + phase)
    a = np.maximum(a, 0)
    S = np.sum(a)
    if S < 1e-10:
        return np.ones(n) / n
    return a / S


def raised_cosine_init(n, alpha=1.0):
    """Raised cosine shape."""
    k = np.arange(n)
    x = 2.0 * k / (n - 1) - 1.0
    a = 0.5 * (1.0 + np.cos(np.pi * x)) ** alpha
    a = np.maximum(a, 1e-10)
    S = np.sum(a)
    return a / S


def gaussian_init(n, sigma_frac=0.3):
    """Gaussian bump initialization."""
    k = np.arange(n)
    center = n / 2.0
    sigma = sigma_frac * n
    a = np.exp(-0.5 * ((k - center) / sigma) ** 2)
    S = np.sum(a)
    return a / S


def random_init(n, seed=0, smoothness=0.3):
    """Random initialization with Gaussian smoothing."""
    rng = np.random.default_rng(seed)
    a = rng.random(n) + 0.1
    # Apply Gaussian smoothing
    kernel = np.exp(-0.5 * (np.arange(-10, 11) / (smoothness * 5)) ** 2)
    a = np.convolve(a, kernel / kernel.sum(), mode='same')
    S = np.sum(a)
    return a / S

"""
Optimized step-function coefficient search minimizing 2*n*max(conv(a,a))/(sum(a))^2.
Key improvements over previous: exact active-set subgradient (all near-max positions),
backtracking line search, flat-convolution theoretical initialization, more aggressive
LP alternation, adaptive perturbation escaping, and coarse-to-fine tolerance refinement.

import numpy as np
from scipy.optimize import linprog
from scipy.interpolate import interp1d
import time


def compute_conv(a):
    """Compute linear convolution a*a via FFT."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    conv = np.fft.irfft(fa * fa, size)
    return conv[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    S = float(np.sum(a))
    if S < 1e-10:
        return np.inf
    conv = compute_conv(a)
    M = float(np.max(conv))
    return 2.0 * n * M / (S * S)


def get_active_set(conv, rel_tol=0.001):
    """Get all positions where conv is within rel_tol (relative) of max."""
    M = float(np.max(conv))
    if M <= 0:
        return np.array([0]), M
    active = np.where(conv >= M * (1.0 - rel_tol))[0]
    return active, M


def exact_subgradient(a, rel_tol=0.001):
    """Compute exact subgradient using all near-max positions (vectorized inner loop)."""
    n = len(a)
    conv = compute_conv(a)
    M = float(np.max(conv))
    active = get_active_set(conv, rel_tol)[0]

    grad = np.zeros(n)
    for k in active:
        i_start = max(0, k - n + 1)
        i_end = min(k, n - 1)
        if i_start <= i_end:
            j = k - np.arange(i_start, i_end + 1)
            grad[i_start:i_end + 1] += 2.0 * a[j]

    grad /= len(active)
    return grad, M, conv, active


def backtracking_subgrad(a, max_iters=5000, time_budget=30.0, rel_tol=0.001):
    """Subgradient descent with exact active set and backtracking line search."""
    n = len(a)
    a = a.copy().astype(np.float64)
    S = np.sum(a)
    if S < 1e-10:
        return a, np.inf
    a = a / S

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    base_step = 0.1
    momentum = 0.82
    v = np.zeros(n)
    stall = 0

    for it in range(max_iters):
        if time.monotonic() - start > time_budget:
            break

        grad, M, conv, active = exact_subgradient(a, rel_tol=rel_tol)
        grad_centered = grad - np.mean(grad)

        v = momentum * v + (1.0 - momentum) * grad_centered

        # Backtracking line search
        alpha = base_step
        improved = False
        for _ in range(15):
            a_new = a - alpha * v
            a_new = np.maximum(a_new, 0)
            S_new = np.sum(a_new)
            if S_new < 1e-10:
                alpha *= 0.5
                continue
            a_new = a_new / S_new
            obj_new = compute_obj(a_new)
            if obj_new < best_obj - 1e-14:
                best_obj = obj_new
                best_a = a_new.copy()
                a = a_new.copy()
                improved = True
                break
            alpha *= 0.6

        if improved:
            stall = 0
        else:
            stall += 1
            if stall > 25:
                rng = np.random.default_rng(it * 13 + 7)
                scale = 0.03 + 0.04 * min(stall / 50.0, 1.0)
                a = a * np.exp(rng.normal(0, scale, n))
                a = np.maximum(a, 0)
                S = np.sum(a)
                if S < 1e-10:
                    a = np.ones(n) / n
                else:
                    a = a / S
                v = np.zeros(n)
                stall = 0
                base_step *= 0.8

    return best_a, best_obj


def build_toeplitz_sparse(a):
    """Build sparse matrix T such that T @ g = conv(a, g)."""
    n = len(a)
    from scipy.sparse import lil_matrix
    T = lil_matrix((2 * n - 1, n))
    for j in range(n):
        for i in range(n):
            T[i + j, j] = a[i]
    return T.tocsr()


def solve_lp_maxsum(a, M, time_limit=5):
    """Solve LP: max sum(g) s.t. conv(a,g) <= M, g >= 0."""
    n = len(a)
    T = build_toeplitz_sparse(a)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c, A_ub=T, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': time_limit})
        if res.status == 0:
            return np.maximum(res.x, 0)
    except Exception:
        pass
    return None


def alternating_lp(a, max_iters=60, time_budget=40.0):
    """Alternating LP with adaptive M and line search."""
    n = len(a)
    a = a.copy().astype(np.float64)
    S = np.sum(a)
    if S < 1e-10:
        return a, np.inf
    a = a / S

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    stall = 0

    for it in range(max_iters):
        if time.monotonic() - start > time_budget:
            break

        conv = compute_conv(a)
        M = float(np.max(conv))

        # Try slightly perturbed M for diversity
        M_offsets = [0.0, 0.005, -0.005]
        g = None
        for mo in M_offsets:
            if stall > 5:
                M_try = M * (1.0 + mo)
            else:
                M_try = M
            g = solve_lp_maxsum(a, M_try, time_limit=4)
            if g is not None and np.sum(g) > 1e-10:
                break

        if g is None or np.sum(g) < 1e-10:
            if stall > 10:
                break
            stall += 1
            continue

        S_g = np.sum(g)
        g = g / S_g

        # Line search
        best_t = 0.0
        best_obj_ls = best_obj
        for t in [0.05, 0.1, 0.15, 0.25, 0.4, 0.6, 0.8, 1.0]:
            a_try = (1.0 - t) * a + t * g
            a_try = np.maximum(a_try, 0)
            S_try = np.sum(a_try)
            if S_try < 1e-10:
                continue
            a_try = a_try / S_try
            obj_try = compute_obj(a_try)
            if obj_try < best_obj_ls:
                best_obj_ls = obj_try
                best_t = t

        if best_t > 0.0:
            a = (1.0 - best_t) * a + best_t * g
            a = np.maximum(a, 0)
            a = a / np.sum(a)
            best_obj = best_obj_ls
            best_a = a.copy()
            stall = 0

            stall += 1

                stall = 0
            else:
                a = 0.95 * a + 0.05 * g
                a = np.maximum(a, 0)
                a = a / np.sum(a)

    return best_a, best_obj


def flat_conv_init(n):
    """Initialize from the theoretical flat-convolution solution via DFT."""
    size = 1
    while size < 2 * n - 1:
        size <<= 1

    c_flat = np.zeros(size)
    c_flat[:2 * n - 1] = 1.0 / (2 * n - 1)
    C_flat = np.fft.rfft(c_flat, size)

    # A_target = sqrt(C_flat) - principal square root
    A_target = np.sqrt(C_flat + 0j)

    a_target = np.fft.irfft(A_target, size)[:n]
    a_target = np.maximum(a_target, 0)
    S = np.sum(a_target)
    if S < 1e-10:
        return np.ones(n) / n
    return a_target / S


def cosine_init(n, n_modes=10, seed=0):
    """Generate a smooth cosine-series initialization."""
    rng = np.random.default_rng(seed)
    k = np.arange(n)
    a = np.ones(n)
    for m in range(1, n_modes + 1):
        c = rng.uniform(-0.5, 0.5) / m
        phase = rng.uniform(0, 2 * np.pi)
        a += c * np.cos(2 * np.pi * m * k / n + phase)
    a = np.maximum(a, 0)
    S = np.sum(a)
    if S < 1e-10:
        return np.ones(n) / n
    return a / S


def raised_cosine_init(n, alpha=1.0):
    """Raised cosine shape."""
    k = np.arange(n)
    x = 2.0 * k / (n - 1) - 1.0
    a = 0.5 * (1.0 + np.cos(np.pi * x)) ** alpha
    a = np.maximum(a, 1e-10)
    S = np.sum(a)
    return a / S


def gaussian_init(n, sigma_frac=0.3):
    """Gaussian bump initialization."""
    k = np.arange(n)
    center = n / 2.0
    sigma = sigma_frac * n
    a = np.exp(-0.5 * ((k - center) / sigma) ** 2)
    S = np.sum(a)
    return a / S


def random_smooth_init(n, seed=0, smoothness=0.4):
    """Random initialization with Gaussian smoothing."""
    rng = np.random.default_rng(seed)
    a = rng.random(n) + 0.1
    kernel_size = int(10 * smoothness) + 1
    if kernel_size % 2 == 0:
        kernel_size += 1
    kernel = np.exp(-0.5 * (np.arange(-kernel_size, kernel_size + 1) / (smoothness * 3)) ** 2)
    a = np.convolve(a, kernel / kernel.sum(), mode='same')
    S = np.sum(a)
    return a / S

                a_p = np.maximum(a_p, 0)
                S_p = np.sum(a_p)
                if S_p > 1e-10:
                    a_p = a_p / S_p
                    a_p, obj_p = backtracking_subgrad(a_p, max_iters=1000,
                                                       time_budget=min(remaining * 0.2, 10.0),
                                                       rel_tol=0.0001)
                    if obj_p < best_score:
                        best_score = obj_p
                        best_seq = a_p.copy()
                        print(f"  Perturb: {best_score:.6f} ** NEW BEST **")

    # Final validation
    if best_seq is None:
        best_seq = np.ones(300) / 300
        best_score = compute_obj(best_seq)

    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    val = evaluate_sequence(result)
    if val < np.inf and val < best_score:
        best_score = val
    elif val == np.inf:
        result = [1.0] * 300
        best_score = 2.0

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_score:.6f}, time={elapsed:.1f}s ===")

    return result
