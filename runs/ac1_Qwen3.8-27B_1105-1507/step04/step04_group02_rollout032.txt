def gs_step(a, N=None):
    n = len(a)
    if N is None:
        N = 1  # next power of 2 >= 2n-1
        while N < 2*n - 1:
            N *= 2
    a_pad = np.zeros(N)
    a_pad[:n] = a
    A = np.fft.fft(a_pad)
    # Target magnitude: constant = sqrt(mean(|A|^2))
    target_mag = np.sqrt(np.mean(np.abs(A)**2))
    # New A with same phase, target magnitude
    A_new = target_mag * A / np.abs(A)  # careful with zeros
    a_new = np.fft.ifft(A_new).real
    a_new = a_new[:n]
    a_new = np.maximum(a_new, 0)
    S = np.sum(a_new)
    if S < 1e-10:
        return a
    return a_new / S
```

def project_to_simplex(v, s=1.0):
    """Project v onto {x >= 0, sum(x) = s}."""
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.where(u * np.arange(1, n+1) > (cssv - s))[0][-1]
    theta = (cssv[rho] - s) / (rho + 1.0)
    return np.maximum(v - theta, 0)
```

Now, let me also think about a **direct peak-shaving** approach:

**Peak-shaving**:
1. Compute R = conv(a,a)
2. Find k* = argmax R
3. R[k*] = sum_i a[i] * a[k*-i]
4. The "excess" is R[k*] - mean(R)
5. To reduce R[k*], we can decrease a[i] for the i that contribute most to R[k*]
6. Specifically, the contribution of a[i] to R[k*] is a[i] * a[k*-i]
7. If we decrease a[i] by δ, R[k*] decreases by approximately 2 * δ * a[k*-i] (for the symmetric pair)
8. To maintain sum(a) = 1, we increase a[j] for j that don't contribute to R[k*]

This is a greedy approach that directly targets the peak. It's similar to the "iterative cancellation" method used in multi-user detection.

def peak_shave(a, n_steps=100):
    n = len(a)
    a = a.copy()
    S = np.sum(a)
    a = a / S
    
    for step in range(n_steps):
        conv = compute_conv(a)
        k_star = np.argmax(conv)
        excess = conv[k_star] - np.mean(conv)
        if excess < 1e-10:
            break
        
        # Find elements that contribute most to the peak
        contrib = np.zeros(n)
        for i in range(n):
            j = k_star - i
            if 0 <= j < n:
                contrib[i] = a[i] * a[j]
        
        # Total contribution
        total = np.sum(contrib)
        if total < 1e-15:
            break
        
        # Fraction to reduce from each element
        # We want to reduce R[k*] by 'excess' while maintaining sum(a) = 1
        # Reducing a[i] by delta[i] changes R[k*] by -2*sum(delta[i]*a[k*-i]) (approximately)
        # And changes sum(a) by -sum(delta[i])
        
        # Simple approach: reduce the top contributors proportionally
        top_k = min(10, n)
        top_idx = np.argsort(contrib)[-top_k:]
        
        # Reduce a at top_idx, increase a at the rest
        reduce_total = excess / (2 * np.mean(a[k_star - top_idx : k_star - top_idx + 1] if ... ))
        # This is getting complicated. Let me use a simpler approach.
        
        # Multiply a by exp(-alpha * contrib) to reduce high contributors
        alpha = 0.1
        a = a * np.exp(-alpha * contrib / np.max(contrib + 1e-15))
        a = np.maximum(a, 0)
        S = np.sum(a)
        if S < 1e-10:
