import numpy as np
from scipy.optimize import linprog
from scipy.sparse import csr_matrix
import time


def fast_conv(a):
    """Full self-convolution via FFT."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]


def compute_J(a):
    """Compute J(a) = 2*n*max(conv(a,a))/(sum(a))^2."""
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))


def smooth_max_and_grad(a, tau):
    """Compute smooth max approximation and its gradient."""
    n = len(a)
    b = fast_conv(a)
    b_max = b.max()
    exp_b = np.exp(tau * (b - b_max))
    p = exp_b / exp_b.sum()

    grad = np.zeros(n)
    for k in range(2 * n - 1):
        pk = p[k]
        if pk < 1e-14:
            continue
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        grad[lo:hi + 1] += 2.0 * pk * a[k - hi:k - lo + 1][::-1]

    return grad, p, b, float((b_max + np.log(exp_b.sum()) / tau))


def solve_primal_lp(a, M, constraint_indices=None):
    """
    Solve the primal LP: max sum(g) s.t. (a*g)[k] <= M for k in constraint_indices, g >= 0.
    Returns g or None.
    """
    n = len(a)
    if constraint_indices is None:
        constraint_indices = range(2 * n - 1)
    else:
        constraint_indices = list(constraint_indices)

    num_cons = len(constraint_indices)
    if num_cons == 0:
        return np.ones(n)

    rows = []
    cols = []
    vals = []
    for idx, k in enumerate(constraint_indices):
        j_min = max(0, k - n + 1)
        j_max = min(n - 1, k)
        for j in range(j_min, j_max + 1):
            rows.append(idx)
            cols.append(j)
            vals.append(a[k - j])

    if len(rows) == 0:
        return np.ones(n) / n

    A = csr_matrix((vals, (rows, cols)), shape=(num_cons, n))

    c = -np.ones(n)
    b_ub = np.full(num_cons, M)
    bounds = [(0, None)] * n

    try:
        r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                    options={'presolve': True, 'dual_feasibility_tolerance': 1e-9})
        if r.success and r.x is not None:
            g = np.maximum(r.x, 0)
            if g.sum() > 1e-15:
                return g
    except Exception:
        pass
    return None


def run_mirror_descent(a, deadline, max_iters=50000, beta=0.7,
                       adaptive_constraints=True, verbose=False):

    Nesterov-accelerated LP mirror descent.
    Minimizes max_k (a*a)[k] subject to sum(a)=1, a>=0.
    Uses entropy mirror (multiplicative updates) projected to simplex.
    """
    a = np.maximum(a.astype(np.float64), 1e-12)
    a = a / a.sum()
    n = len(a)

    best_a = a.copy()
    best_M = fast_conv(a).max()
    v = np.zeros(n)  # momentum
    step = 0.5

    # Nesterov counter
    t_nest = 1.0
    a_prev = a.copy()

    stall_count = 0
    last_improve = 0.0

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        b = fast_conv(a)
        M = float(b.max())

        if M < best_M - 1e-12:
            best_M = M
            best_a = a.copy()
            last_improve = time.monotonic()
            stall_count = 0
            if verbose and it % 100 == 0:
                print(f"    MD iter {it}: M={M:.10f}", flush=True)
        else:
            stall_count += 1

        # If stagnant for too long, reduce step
        if stall_count > 500:
            step *= 0.95
            if step < 0.01:
                break

        # Determine constraint set
        if adaptive_constraints and it > 100:
            tol = 0.05 * M
            active = np.where(b >= M - tol)[0]
            if len(active) < (2 * n - 1) // 2:
                constraint_indices = active
            else:
                constraint_indices = None
        else:
            constraint_indices = None

        # Solve LP
        g = solve_primal_lp(a, M, constraint_indices)
        if g is None:
            # Fallback: use uniform
            g = np.ones(n) / n
        g = g / g.sum()

        # Nesterov acceleration
        t_new = (1 + np.sqrt(1 + 4 * t_nest ** 2)) / 2

        # Momentum in function space (simpler than Bregman momentum)
        v = beta * v + (g - a)
        a_new = a + step * v

        # Project to simplex (sum=1, a>=0)
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s < 1e-15:
            a_new = a.copy()
        else:
            a_new = a_new / s

        M_new = fast_conv(a_new).max()

            # Backtrack

                a_trial = a + step * shrunk * v
                a_trial = np.maximum(a_trial, 0.0)

                    continue
                a_trial /= st
                M_trial = fast_conv(a_trial).max()

            if M_new < M - 1e-8:
                step = min(step * 1.02, 1.0)

        a_prev = a.copy()
        a = a_new
        t_nest = t_new

        # Convergence
        if best_M < 1e-4:
            break
        if time.monotonic() > deadline - 5:
            break

    return best_a

def resample_seq(src, dest_n):
    """Resample 1D sequence to new length via linear interpolation."""
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)

        if n_try in tried_lengths and n_try == best_n:
            continue

        a_try = resample_seq(main_a, n_try)
        a_try = np.maximum(a_try, 1e-6)
        a_try /= a_try.sum()

        # Quick smooth gradient
        a_try = run_smooth_gradient(a_try, 50, min(time.monotonic() + 20, p4_deadline),
                                    tau=40, lr=0.1, gamma=0.85)

        # Short mirror descent
        remaining = p4_deadline - time.monotonic()
        if remaining > 30:
            a_try = run_mirror_descent(a_try, time.monotonic() + min(120, remaining * 0.7),
                                       max_iters=10000, beta=0.75)
        # Polish
        if time.monotonic() < p4_deadline - 5:
            a_try = coordinate_polish(a_try, min(time.monotonic() + 20, p4_deadline),
                                      n_trials_per_coord=2)

        J_try = compute_J(a_try)
        print(f"  n={n_try:4d}: J={J_try:.8f}", flush=True)
        tracker.update(a_try)

        if J_try < compute_J(main_a) - 0.001:
            main_a = a_try.copy()

    tracker.report("Phase 4")

    # === PHASE 5: Final intensive polish ===
    print(f"\n{'='*60}", flush=True)
    print("PHASE 5: Final Intensive Polish", flush=True)
    print(f"{'='*60}", flush=True)

    if time.monotonic() < deadline - 20:
        # One more round of mirror descent on the best
        final_a = run_mirror_descent(main_a, deadline - 20,
                                     max_iters=20000, beta=0.8)
        tracker.update(final_a)
        main_a = final_a

        # Final coordinate polish
        if time.monotonic() < deadline - 5:
            final_a = coordinate_polish(main_a, deadline - 3, n_trials_per_coord=5)
            tracker.update(final_a)
            main_a = final_a

    tracker.report("FINAL")
    print(f"\n  Final: J={tracker.best_score:.8f} n={tracker.best.size} "
          f"time={time.monotonic()-tracker.t0:.0f}s", flush=True)

    # Fallback
    if tracker.best is None:
        tracker.best = np.ones(128)
        tracker.best_score = compute_J(tracker.best)

    return [float(x) for x in tracker.best]
