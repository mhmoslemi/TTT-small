def fast_conv(a):
    """Full self-convolution via FFT."""
    n = len(a)
    s = 1
    while s < 2 * n - 1:
        s <<= 1
    fa = np.fft.rfft(a, s)
    return np.fft.irfft(fa * fa, s)[:2 * n - 1]

def compute_J(a):
    """Compute J(a) = 2*n*max(conv(a,a))/(sum(a))^2."""
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))

    """Build dense constraint matrix for LP (works for moderate n)."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for k in range(2 * n - 1):
        jl = max(0, k - n + 1)
        jh = min(n - 1, k)
        for j in range(jl, jh + 1):
            A[k, j] = a[k - j]
    return A

def lp_best_response(a, M):
    """LP: max sum(g) s.t. (a*g)_k <= M for all k, g >= 0."""
    n = len(a)
    c = -np.ones(n)

    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
    if r.success and r.x is not None:
        g = np.maximum(r.x, 0.0)
        if g.sum() > 1e-10:
            return g
    return None

Wait, one more thought: the LP formulation. Let me double-check.

We want to find g >= 0 that maximizes sum(g) subject to (a * g)[k] <= M for all k.

Here (a * g)[k] = sum_j a[j] * g[k-j] where the sum is over j such that both j and k-j are in [0, n-1].

So the constraint matrix A has:
A[k, j] = a[k-j] if 0 <= k-j <= n-1, else 0

And the constraint is A @ g <= M * ones(2n-1).

This is correct. The LP finds the "best" direction g to move in, given that we want to keep the convolution with a bounded by M.

The intuition: if a is the "current" sequence and we want to move toward g, the constraint (a*g)[k] <= M ensures that the "cross-correlation" between a and g is bounded. This is related to the idea that if we replace a by g, the new self-convolution (g*g) will be smaller (by some argument involving the AM-GM inequality or similar).

Actually, the rigorous argument is:
(a*g)[k] <= M for all k
implies that the "energy" of the cross-convolution is bounded.
When we take a step a <- (1-t)*a + t*g, the new self-convolution is:
((1-t)*a + t*g) * ((1-t)*a + t*g) = (1-t)^2*(a*a) + 2t(1-t)*(a*g) + t^2*(g*g)

We know a*a has max M (by definition), a*g has max <= M (by LP constraint).
If g*g also has max <= M (which is not guaranteed by the LP), then the new max would be <= M.
But g*g might exceed M. However, for small t, the t^2*(g*g) term is small, so the new max should be close to M but slightly less.

The LP doesn't directly constrain g*g, but empirically this approach works well because the LP tends to find g that is "spread out" (to maximize sum(g) subject to the constraints), which means g*g has a lower peak.

"""
Minimizes J(a) = 2*n*max(convolve(a,a))/(sum(a))^2 using:
1. LP mirror descent with adaptive tightening and multi-step-size selection
2. Efficient coordinate descent with incremental convolution updates
3. Multi-length exploration via resampling
4. Structured initializations (cosine modulated, uniform, random)
5. Alternating LP/CD polishing with adaptive scheduling

import numpy as np
from scipy.optimize import linprog
from scipy.sparse import lil_matrix, csr_matrix
import time


def fast_conv(a):
    """Full self-convolution via FFT."""
    n = len(a)
    s = 1
    while s < 2 * n - 1:
        s <<= 1
    fa = np.fft.rfft(a, s)
    return np.fft.irfft(fa * fa, s)[:2 * n - 1]


def compute_J(a):
    """Compute J(a) = 2*n*max(conv(a,a))/(sum(a))^2."""
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))


def build_A_dense(a):
    """Build dense constraint matrix for LP: A[k,j] = a[k-j] if valid."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for k in range(2 * n - 1):
        jl = max(0, k - n + 1)
        jh = min(n - 1, k)
        if jl <= jh:
            A[k, jl:jh + 1] = a[k - jh:k - jl + 1][::-1]
    return A


def lp_best_response(a, M):
    """LP: max sum(g) s.t. (a*g)_k <= M for all k, g >= 0."""
    n = len(a)
    c = -np.ones(n)
    A = build_A_dense(a)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                options={'presolve': True, 'time_limit': 30})
    if r.success and r.x is not None:
        g = np.maximum(r.x, 0.0)
        if g.sum() > 1e-10:
            return g
    return None


def cd_optimize_coordinate(a, i, b, S):
    """Find optimal value for coordinate i using 1D golden section search."""
    n = len(a)
    x_old = a[i]

    # Affected convolution positions: k in [i, i+n-1] intersect [0, 2n-2]
    k_min = i
    k_max = min(i + n - 1, 2 * n - 2)

    # Unaffected max
    unaffected_max = 0.0
    for k in range(2 * n - 1):
        if k < k_min or k > k_max:
            if b[k] > unaffected_max:
                unaffected_max = b[k]

    # Precompute base values for affected positions
    base = np.zeros(k_max - k_min + 1)
    coeff = np.zeros(k_max - k_min + 1)
    is_quad = np.zeros(k_max - k_min + 1, dtype=bool)
    for idx, k in enumerate(range(k_min, k_max + 1)):
        ki = k - i
        if 0 <= ki < n:
            if k == 2 * i:
                base[idx] = b[k] - x_old * x_old
                coeff[idx] = 0.0
                is_quad[idx] = True
            else:
                base[idx] = b[k] - 2.0 * x_old * a[ki]
                coeff[idx] = 2.0 * a[ki]
                is_quad[idx] = False

    def J_of_x(x):
        new_S = S - x_old + x
        if new_S < 1e-10:
            return 1e10
        max_c = unaffected_max
        for idx in range(k_max - k_min + 1):
            if is_quad[idx]:
                ck = base[idx] + x * x
            else:
                ck = base[idx] + coeff[idx] * x
            if ck > max_c:
                max_c = ck
        return 2.0 * n * max_c / (new_S * new_S)

    # Golden section search
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    lo = 0.0
    hi = max(3.0 * x_old if x_old > 0 else 0.2, S / n * 2.0, 0.1)

    c_pt = hi - gr * (hi - lo)
    d_pt = lo + gr * (hi - lo)
    fc = J_of_x(c_pt)
    fd = J_of_x(d_pt)

    for _ in range(25):
        if fc < fd:
            hi = d_pt
            d_pt = c_pt
            fd = fc
            c_pt = hi - gr * (hi - lo)
            fc = J_of_x(c_pt)
        else:
            lo = c_pt
            c_pt = d_pt
            fc = fd
            d_pt = lo + gr * (hi - lo)
            fd = J_of_x(d_pt)

    x_opt = (lo + hi) / 2.0
    return max(0.0, x_opt)


def cd_pass(a, max_passes_check=None):
    """One full pass of coordinate descent with incremental convolution."""
    a = np.maximum(a.astype(np.float64), 0.0)
    n = len(a)
    S = a.sum()
    if S < 1e-10:
        return a, float('inf')
    b = fast_conv(a).copy()
    current_J = 2.0 * n * b.max() / (S * S)

    indices = list(range(n))
    np.random.shuffle(indices)

    for i in indices:
        new_val = cd_optimize_coordinate(a, i, b, S)
        if abs(new_val - a[i]) > 1e-12:
            old_val = a[i]
            d = new_val - old_val
            a[i] = new_val
            S = S + d
            # Update convolution incrementally
            k_min = i
            k_max = min(i + n - 1, 2 * n - 2)
            for k in range(k_min, k_max + 1):
                ki = k - i
                if 0 <= ki < n:
                    if k == 2 * i:
                        b[k] = b[k] - old_val * old_val + new_val * new_val
                    else:
                        b[k] = b[k] + 2.0 * d * a[ki]

    # Verify
    b_check = fast_conv(a)
    new_J = 2.0 * n * b_check.max() / (S * S)
    return a, new_J


def resample_seq(src, dest_n):
    """Resample 1D sequence to new length via linear interpolation."""
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)


def lp_mirror_descent(a, max_iters, deadline, tracker, tighten_base=0.005):
    """LP mirror descent with adaptive tightening and step size selection."""
    a = np.maximum(a.astype(np.float64), 1e-6)
    a = a / a.sum()
    n = len(a)
    current_J = compute_J(a)
    tracker.update(a)

    tighten = tighten_base
    stall = 0

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        b = fast_conv(a)
        M = float(b.max())

        # Try progressively looser constraints if LP fails
        g = None
        for frac in [0.001, 0.003, 0.005, 0.01, 0.02, 0.05]:
            M_target = M * (1.0 - frac)
            if M_target < 1e-10:
                continue
            g = lp_best_response(a, M_target)
            if g is not None:
                break

        if g is None:
            stall += 1
            if stall > 15:
                break
            tighten = max(0.001, tighten * 0.7)
            continue

        g = np.maximum(g, 0.0)
        g_sum = g.sum()
        if g_sum < 1e-10:
            break
        g = g / g_sum

        # Try multiple step sizes
        progress = min(1.0, it / max(1, max_iters))
        base_t = max(0.02, 0.4 * (1.0 - 0.5 * progress))
        t_candidates = [base_t, base_t * 0.5, base_t * 1.5, base_t * 0.3, 0.8, 0.6, 0.4]
        t_candidates = [t for t in t_candidates if 0 < t < 1.0]

        best_a = None
        best_J = current_J
        for t in t_candidates:
            a_trial = (1.0 - t) * a + t * g
            a_trial = np.maximum(a_trial, 0.0)
            s = a_trial.sum()
            if s < 1e-10:
                continue
            a_trial = a_trial / s
            J_trial = compute_J(a_trial)
            if J_trial < best_J - 1e-14:
                best_J = J_trial
                best_a = a_trial

        if best_a is None:
            stall += 1
            if stall > 25:
                break
            tighten = max(0.0005, tighten * 0.7)
            continue

        a = best_a
        current_J = best_J
        stall = 0

        # Adapt tightening
        if it % 30 == 0:
            tighten = min(0.02, tighten * 1.05)

        if it % 20 == 0:
            tracker.update(a)

    tracker.update(a)
    return a

def optimize_sequence(a, deadline, tracker, n_lp=50, n_cd=3):
    """Alternate LP and CD to optimize a sequence."""
    a = np.maximum(a.astype(np.float64), 1e-6)
    a = a / a.sum()
    tracker.update(a)

    while time.monotonic() < deadline:
        # LP phase
        remaining = deadline - time.monotonic()
        if remaining < 10:
            break
        lp_budget = min(remaining * 0.7, 15.0)
        n_lp_iters = max(5, int(20 * lp_budget / 15.0))
        a = lp_mirror_descent(a, n_lp_iters, time.monotonic() + lp_budget, tracker,
                              tighten_base=0.004)

        if time.monotonic() > deadline - 5:
            break

        # CD phase
        cd_budget = deadline - time.monotonic()
        if cd_budget > 5:
            for _ in range(min(n_cd, int(cd_budget / 5))):
                if time.monotonic() > deadline - 3:
                    break
                a, new_J = cd_pass(a)
                a = a / a.sum()
                tracker.update(a)

    return a
