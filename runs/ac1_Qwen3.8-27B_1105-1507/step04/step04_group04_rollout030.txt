"""
Step function optimization using targeted peak deflation, LP-based alternating,
and structured multi-n search. Key innovations over previous approach:
- Direct peak-deflation multiplicative updates targeting top-K peaks simultaneously
- Tighter LP constraints with adaptive M
- Structured initializations (two-level, ramped) in addition to random
- Multi-n sweep with resampling between scales
- Fine-grained local search with adaptive perturbation
"""

import numpy as np
import time
from scipy.optimize import linprog

try:
    import cvxpy as cp
    HAS_CVXPY = True
except ImportError:
    HAS_CVXPY = False


def compute_conv(a):
    """Compute full convolution of a with itself using FFT."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    conv = np.fft.irfft(fa * fa, size)
    return conv[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    conv = compute_conv(a)
    max_b = float(np.max(conv))
    sum_a = float(np.sum(a))
    if sum_a < 0.01:
        return np.inf
    return 2.0 * n * max_b / (sum_a * sum_a)


def build_toeplitz_conv(a):
    """Build matrix A where (A @ g)[k] = (a * g)[k] for k=0..2n_g-2+n_a-1.
    Here a has length n and g has length m. Result has length n+m-1.
    For a*g where both have same length n: result length 2n-1.
    """
    n_a = len(a)
    # We'll build for g of same length as a
    n = n_a
    A = np.zeros((2 * n - 1, n), dtype=np.float64)
    for j in range(n):
        # a * g: coefficient of x^(j+0), x^(j+1), ..., x^(j+n-1)
        # conv[k] = sum_i a[i]*g[k-i]
        # For fixed j (g[j]), it contributes to conv[k] where k = i+j for i=0..n-1
        # So conv[j+i] += a[i]*g[j], i.e., A[j+i, j] = a[i] for i=0..n-1
        for i in range(n):
            A[j + i, j] = a[i]
    return A


def solve_lp_maxsum(a, M, time_limit=5):
    """Solve LP: max sum(g) s.t. (a*g)[k] <= M for all k, g >= 0.
    a and g have the same length n.
    """
    n = len(a)
    if HAS_CVXPY:
        try:
            g_var = cp.Variable(n, nonneg=True)
            A_mat = build_toeplitz_conv(a)
            constraints = [A_mat @ g_var <= M]
            prob = cp.Problem(cp.Maximize(cp.sum(g_var)), constraints)
            for solver in [cp.MOSEK, cp.CLARABEL, cp.SCS]:
                try:
                    prob.solve(solver=solver, time_limit=int(time_limit))
                    if g_var.value is not None:
                        g_val = np.maximum(g_var.value, 0)
                        return g_val, float(np.sum(g_val))
                except Exception:
                    continue
        except Exception:
            pass

    A_mat = build_toeplitz_conv(a)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c, A_ub=A_mat, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': time_limit})
        if res.status == 0:
            g = np.maximum(res.x, 0)
            return g, float(np.sum(g))
    except Exception:
        pass
    return None, None


def peak_deflation_step(a, eta=0.1, num_peaks=5, floor=1e-10):
    """One peak-deflation multiplicative step.
    Directly targets the top peaks of conv(a,a) by reducing contributing elements.
    """
    n = len(a)
    conv = compute_conv(a)
    max_conv = np.max(conv)
    if max_conv < 1e-15:
        return a.copy()

    # Find top K peak positions
    # Use a threshold to identify "significant" peaks
    threshold = max_conv * 0.8
    peak_positions = np.where(conv >= threshold)[0]
    if len(peak_positions) > num_peaks:
        # Take only the top ones
        peak_vals = conv[peak_positions]
        top_idx = np.argsort(peak_vals)[-num_peaks:]
        peak_positions = peak_positions[top_idx]

    # For each peak at position k, compute partner strength for each a[j]
    # a[j] contributes a[j]*a[k-j] to c[k]
    partner_strength = np.zeros(n, dtype=np.float64)
    for k in peak_positions:
        # For each j, the partner is a[k-j] if 0 <= k-j < n
        valid_j = np.arange(n)
        partner_idx = k - valid_j  # k - j for each j
        valid_mask = (partner_idx >= 0) & (partner_idx < n)
        for j in valid_j[valid_mask]:
            p_idx = k - j
            if p_idx != j:
                partner_strength[j] += a[p_idx]
            else:
                partner_strength[j] += a[j] * 0.5  # self-pair contribution

    # Normalize partner strength
    max_ps = np.max(partner_strength)
    if max_ps < 1e-15:
        return a.copy()
    partner_strength = partner_strength / max_ps

    # Multiplicative update
    a_new = a * np.exp(-eta * partner_strength)
    a_new = np.maximum(a_new, floor)

    # Renormalize
    s = np.sum(a_new)
    if s > 1e-10:
        a_new = a_new / s
    return a_new


def peak_deflation_optimize(a, time_budget=30.0, eta_init=0.15, num_peaks=5):
    """Optimize using peak deflation with adaptive parameters."""
    n = len(a)
    a = a.copy().astype(np.float64)
    s = np.sum(a)
    if s < 1e-10:
        return np.ones(n) / n
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    eta = eta_init
    stall = 0
    patience = 50

    while time.monotonic() - start < time_budget:
        a = peak_deflation_step(a, eta=eta, num_peaks=num_peaks)
        obj = compute_obj(a)

        if obj < best_obj - 1e-14:
            best_obj = obj
            best_a = a.copy()
            stall = 0
        else:
            stall += 1
            if stall > patience:
                eta *= 0.7
                if eta < 0.005:
                    break
                # Perturb to escape
                rng = np.random.default_rng(int(stall * 7 + n))
                a_pert = a * np.exp(rng.normal(0, 0.05, n))
                a_pert = np.maximum(a_pert, 1e-10)
                a_pert = a_pert / np.sum(a_pert)
                a = a_pert
                stall = 0
                patience = max(patience // 2, 20)
                num_peaks = max(3, num_peaks - 1)

    return best_a


def alternating_lp_tight(a, M_factor=0.98, time_budget=40.0):
    """Alternating LP with tighter constraint (M_factor * current_peak)."""
    n = len(a)
    a = a.copy().astype(np.float64)
    s = np.sum(a)
    if s < 1e-10:
        return np.ones(n) / n
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    stall = 0

    while time.monotonic() - start < time_budget - 3:
        conv = compute_conv(a)
        M_current = float(np.max(conv))
        M_target = M_current * M_factor

        g, sum_g = solve_lp_maxsum(a, M_target, time_limit=4)
        if g is None or sum_g < 1e-10:
            # Try with slightly looser constraint
            g, sum_g = solve_lp_maxsum(a, M_current, time_limit=4)
            if g is None or sum_g < 1e-10:
                break

        g_norm = g / sum_g

        # Line search
        best_t = 0.0
        best_obj_ls = best_obj
        for t in [0.1, 0.2, 0.3, 0.5, 0.7, 1.0]:
            a_new = (1.0 - t) * a + t * g_norm
            a_new = np.maximum(a_new, 0)
            s_new = np.sum(a_new)
            if s_new < 1e-10:
                continue
            a_new = a_new / s_new
            obj_new = compute_obj(a_new)
            if obj_new < best_obj_ls:
                best_obj_ls = obj_new
                best_t = t

        if best_t > 0 and best_obj_ls < best_obj - 1e-13:
            a_new = (1.0 - best_t) * a + best_t * g_norm
            a_new = np.maximum(a_new, 0)
            a_new = a_new / np.sum(a_new)
            best_obj = best_obj_ls
            best_a = a_new.copy()
            a = a_new
            stall = 0
        else:
            stall += 1
            if stall > 15:
                break
            # Gentle update
            a = 0.9 * a + 0.1 * g_norm
            a = np.maximum(a, 0)
            a = a / np.sum(a)

def structured_init(n, seed, mode=0):
    """Generate structured initializations."""
    rng = np.random.default_rng(seed)
    k = np.arange(n, dtype=np.float64)

    if mode == 0:
        # Two-level: low then high
        split = n * rng.uniform(0.3, 0.7)
        a = np.zeros(n)
        a[:int(split)] = rng.uniform(0.3, 0.8)
        a[int(split):] = rng.uniform(1.0, 2.0)
    elif mode == 1:
        # Ramped with noise
        a = 1.0 + rng.uniform(0.5, 2.0) * (k / n)
        a += rng.normal(0, 0.1, n)
    elif mode == 2:
        # Cosine-based (always non-negative)
        f = rng.uniform(0.5, 3.0)
        a = 1.0 + rng.uniform(0.3, 0.7) * np.cos(2 * np.pi * f * k / n + rng.uniform(0, 2 * np.pi))
    elif mode == 3:
        # Gaussian bump
        center = n * rng.uniform(0.3, 0.7)
        width = n * rng.uniform(0.1, 0.4)
        a = 1.0 + 0.5 * np.exp(-0.5 * ((k - center) / width) ** 2)
    elif mode == 4:
        # Two bumps
        c1 = n * rng.uniform(0.2, 0.4)
        c2 = n * rng.uniform(0.6, 0.8)
        w = n * rng.uniform(0.05, 0.2)
        a = 1.0 + 0.4 * np.exp(-0.5 * ((k - c1) / w) ** 2) + 0.4 * np.exp(-0.5 * ((k - c2) / w) ** 2)
    elif mode == 5:
        # Sine squared (guaranteed non-negative, periodic)
        f = rng.uniform(1.0, 5.0)
        a = np.sin(np.pi * f * k / n + rng.uniform(0, np.pi)) ** 2
        a = a / np.mean(a)  # normalize mean to 1
    elif mode == 6:
        # Random walk (smooth)
        a = np.cumsum(rng.normal(0, 1, n))
        a = a - a.min() + 0.5
    elif mode == 7:
        # Alternating blocks
        block_size = max(2, n // rng.integers(5, 20))
        a = np.ones(n)
        for i in range(0, n, block_size):
            a[i:i+block_size] *= rng.uniform(0.5, 1.5)
    else:
        # Random smooth
        a = rng.random(max(3, n // 20))
        a = np.interp(k, np.arange(len(a)), a)
        a = a / max(a.max(), 1e-10)

    a = np.maximum(a, 0)
    s = np.sum(a)
    if s < 1e-10:
        a = np.ones(n) / n
    else:
        a = a / s
    return a


def resample_sequence(a, new_n):
    """Resample a sequence to a new length using interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    old_k = np.arange(n, dtype=np.float64)
    new_k = np.linspace(0, n - 1, new_n)
    a_new = np.interp(new_k, old_k, a)
    a_new = np.maximum(a_new, 0)
    s = np.sum(a_new)
    if s > 1e-10:
        a_new = a_new / s
    return a_new

        obj = compute_obj(a_cand)

        update_best(a_new, "PeakDefl")
        a = best_a

    # Phase 2: LP alternating tight (25% of budget)

    if remaining > 15:
        t2 = min(remaining * 0.3, 30.0)
        a_new = alternating_lp_tight(a, M_factor=0.98, time_budget=t2)
        update_best(a_new, "LP-tight")
        a = best_a

    # Phase 3: LP alternating deflect (20% of budget)
    remaining = time_budget - (time.monotonic() - start)
    if remaining > 15:
        t3 = min(remaining * 0.25, 25.0)
        a_new = alternating_lp_deflect(a, time_budget=t3)
        update_best(a_new, "LP-deflect")
        a = best_a

    # Phase 4: More peak deflation with smaller steps (15% of budget)
    remaining = time_budget - (time.monotonic() - start)
    if remaining > 10:
        t4 = min(remaining * 0.2, 15.0)
        a_new = peak_deflation_optimize(a, time_budget=t4, eta_init=0.06, num_peaks=3)
        update_best(a_new, "PeakDefl2")

    # Phase 6: Final LP polish
    remaining = time_budget - (time.monotonic() - start)
    if remaining > 10:
        t6 = min(remaining * 0.4, 20.0)
        a_new = alternating_lp_tight(a, M_factor=0.99, time_budget=t6)
        update_best(a_new, "LP-final")

    return best_a


def optimize_for_n(n, seed, time_budget, parent_seq=None, verbose=False):
    """Optimize for a specific n with multiple initializations."""
    start = time.monotonic()
    best_a = None
    best_obj = np.inf

    # Number of inits based on budget
    n_inits = max(2, min(6, int(time_budget / 30)))

    for init_idx in range(n_inits):
        if time.monotonic() - start > time_budget - 20:
            break
        remaining = time_budget - (time.monotonic() - start)
        if remaining < 30:
            break

        if init_idx == 0 and parent_seq is not None:
            a = resample_sequence(parent_seq, n)
        else:
            mode = (seed + init_idx) % 9
            a = structured_init(n, seed + init_idx * 9973, mode=mode)

        sub_budget = min(remaining * 0.6, 50.0)
        a_opt = optimize_pipeline(a, time_budget=sub_budget, verbose=(verbose and init_idx == 0))
        obj = compute_obj(a_opt)

        if obj < best_obj:
            best_obj = obj
            best_a = a_opt.copy()

    if best_a is None:
        best_a = np.ones(n) / n
        best_obj = compute_obj(best_a)

    return best_a, best_obj


def intensive_refine(a, time_budget=100.0, verbose=False):
    """Intensive refinement using multiple small perturbations + re-optimization."""
    n = len(a)
    a = a.copy().astype(np.float64)
    s = np.sum(a)
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    if verbose:
        print(f"  Intensive refine start: {best_obj:.8f}")

    attempt = 0
    while time.monotonic() - start < time_budget - 15:
        attempt += 1
        remaining = time_budget - (time.monotonic() - start)
        if remaining < 20:
            break

        rng = np.random.default_rng(attempt * 31337 + n)

        # Different perturbation strategies
        strategy = attempt % 4
        if strategy == 0:
            # Gaussian multiplicative
            scale = 0.03 + 0.04 * (attempt % 3) / 3.0
            a_pert = best_a * np.exp(rng.normal(0, scale, n))
        elif strategy == 1:
            # Burst: perturb a random contiguous block
            block_size = max(2, n // (10 + attempt % 5))
            start_pos = rng.integers(0, n - block_size)
            a_pert = best_a.copy()
            a_pert[start_pos:start_pos + block_size] *= np.exp(rng.normal(0, 0.05, block_size))
        elif strategy == 2:
            # Swap pairs
            a_pert = best_a.copy()
            num_swaps = max(1, n // 30)
            for _ in range(num_swaps):
                i, j = rng.choice(n, size=2, replace=False)
                a_pert[i], a_pert[j] = best_a[j], best_a[i]
        else:
            # Mirror perturbation
            a_pert = best_a.copy()
            num_pts = max(2, n // 20)
            indices = rng.choice(n, size=num_pts, replace=False)
            factors = rng.uniform(0.9, 1.1, size=num_pts)
            a_pert[indices] *= factors
            # Mirror
            a_pert[n - 1 - indices] *= factors

        a_pert = np.maximum(a_pert, 1e-12)
        s_pert = np.sum(a_pert)
        if s_pert < 1e-10:
            continue
        a_pert = a_pert / s_pert

        # Quick optimization
        sub_time = min(15.0, remaining * 0.25)
        a_opt = optimize_pipeline(a_pert, time_budget=sub_time, verbose=False)
        obj = compute_obj(a_opt)

        if obj < best_obj:
            best_obj = obj
            best_a = a_opt.copy()
            if verbose:
                print(f"    Refine attempt {attempt}: {best_obj:.8f}")

    return best_a


def propose_candidate(seed=42, budget_s=1500):
    """
    Main search: combines peak deflation, LP alternating, structured inits,
    multi-n sweep, and intensive refinement.
    """
    start_time = time.monotonic()
    deadline = start_time + float(budget_s) - 35.0

    best_seq = None
    best_score = np.inf

    # Load parent
    parent = None
    n_parent = 330
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64)
        parent = np.clip(parent, 0, 1000.0)
        if parent.size > 0 and np.isfinite(parent).all():
            parent_obj = compute_obj(parent)
            print(f"Parent (n={parent.size}): obj={parent_obj:.8f}")
            if parent_obj < best_score:
                best_score = parent_obj
                best_seq = parent.copy()
                n_parent = len(parent)
    except Exception as e:
        print(f"Could not load parent: {e}")

    if best_seq is None:
        # Fallback: create a decent initial sequence
        n_parent = 330
        best_seq = structured_init(n_parent, seed, mode=0)
        best_score = compute_obj(best_seq)
        print(f"Fallback init (n={n_parent}): obj={best_score:.8f}")

    print(f"Initial best: {best_score:.8f} (n={n_parent})")
    print(f"Time budget: {budget_s}s, deadline in {deadline - time.monotonic():.0f}s")

    # === Phase 1: Intensive optimization of parent ===
    print("\n=== Phase 1: Intensive optimization of parent ===")
    remaining = deadline - time.monotonic()
    p1_budget = min(remaining * 0.30, 300.0)

    a_opt = optimize_pipeline(best_seq, time_budget=p1_budget, verbose=True)
    obj = compute_obj(a_opt)
    if obj < best_score:
        best_score = obj
        best_seq = a_opt.copy()
        print(f"  After pipeline: {best_score:.8f}")

    # Intensive refinement
    remaining = deadline - time.monotonic()
    if remaining > 50:
        p1b_budget = min(remaining * 0.20, 150.0)
        a_opt = intensive_refine(best_seq, time_budget=p1b_budget, verbose=True)
        obj = compute_obj(a_opt)
        if obj < best_score:
            best_score = obj
            best_seq = a_opt.copy()
            print(f"  After intensive refine: {best_score:.8f}")

    # === Phase 2: Sweep nearby n values ===
    print("\n=== Phase 2: n sweep ===")
    # Focus on n values near the parent
    n_candidates = []
    for delta in range(-50, 51, 10):
        n_val = n_parent + delta
        if 150 <= n_val <= 600:
            n_candidates.append(n_val)
    # Add some specific values
    for nv in [200, 250, 400, 500]:
        if nv not in n_candidates:
            n_candidates.append(nv)
    n_candidates = sorted(set(n_candidates))

    remaining = deadline - time.monotonic()
    if remaining > 60:
        # Allocate time proportional to proximity to parent
        per_n_base = remaining / len(n_candidates)
        for n_val in n_candidates:
            if time.monotonic() > deadline - 40:
                break
            remaining = deadline - time.monotonic()
            if remaining < 30:
                break

            # More time for n close to parent
            proximity = max(0.5, 1.0 - abs(n_val - n_parent) / 200.0)
            alloc = min(per_n_base * proximity * 1.5, remaining * 0.25, 60.0)
            alloc = max(alloc, 20.0)

            if abs(n_val - n_parent) <= 20:
                print(f"\n  n={n_val} (proximity={proximity:.2f}, alloc={alloc:.0f}s)")
            else:
                print(f"\n  n={n_val} (alloc={alloc:.0f}s)")

            a_opt, obj = optimize_for_n(n_val, seed=seed + n_val * 7919,
                                         time_budget=alloc,
                                         parent_seq=parent,
                                         verbose=(abs(n_val - n_parent) <= 10))
            if obj < best_score:
                best_score = obj
                best_seq = a_opt.copy()
                print(f"  ** UPDATE: n={n_val}, obj={best_score:.8f} **")

    # === Phase 3: Second pass on best n with more time ===
    if best_seq is not None:
        remaining = deadline - time.monotonic()
        if remaining > 40:
            print(f"\n=== Phase 3: Second pass (n={len(best_seq)}, {remaining:.0f}s) ===")
            p3_budget = min(remaining * 0.4, 120.0)
            a_opt = optimize_pipeline(best_seq, time_budget=p3_budget, verbose=True)
            obj = compute_obj(a_opt)
            if obj < best_score:
                best_score = obj
                best_seq = a_opt.copy()
                print(f"  After 2nd pipeline: {best_score:.8f}")

            remaining = deadline - time.monotonic()
            if remaining > 30:
                p3b_budget = min(remaining * 0.35, 80.0)
                a_opt = intensive_refine(best_seq, time_budget=p3b_budget, verbose=True)
                obj = compute_obj(a_opt)
                if obj < best_score:
                    best_score = obj
                    best_seq = a_opt.copy()
                    print(f"  After 2nd refine: {best_score:.8f}")

    # === Phase 4: Try a few completely different approaches ===
    if best_seq is not None:
        remaining = deadline - time.monotonic()
        if remaining > 50:
            print(f"\n=== Phase 4: Alternative approaches ({remaining:.0f}s) ===")

            # Try very different n with structured inits
            alt_ns = [n_parent - 30, n_parent + 30, int(n_parent * 0.75), int(n_parent * 1.3)]
            alt_ns = [n for n in alt_ns if 100 <= n <= 800 and n != len(best_seq)]

            for n_val in alt_ns[:3]:
                if time.monotonic() > deadline - 30:
                    break
                remaining = deadline - time.monotonic()
                if remaining < 40:
                    break
                alloc = min(remaining * 0.3, 60.0)

                # Use different structured inits
                for mode in range(4):
                    if time.monotonic() > deadline - 20:
                        break
                    a_init = structured_init(n_val, seed + n_val * 104729 + mode * 31, mode=mode)
                    sub_alloc = min(alloc / 4, remaining * 0.2, 30.0)
                    if sub_alloc < 15:
                        continue
                    a_opt = optimize_pipeline(a_init, time_budget=sub_alloc, verbose=False)
                    obj = compute_obj(a_opt)
                    if obj < best_score:
                        best_score = obj
                        best_seq = a_opt.copy()
                        print(f"  ** UPDATE (alt): n={n_val}, mode={mode}, obj={best_score:.8f} **")

    # === Phase 5: Final polish ===
    if best_seq is not None:
        remaining = deadline - time.monotonic()
        if remaining > 20:
            print(f"\n=== Phase 5: Final polish ({remaining:.0f}s) ===")
            p5_budget = min(remaining * 0.5, remaining - 10.0)

            # Peak deflation with very small steps
            a_new = peak_deflation_optimize(best_seq, time_budget=p5_budget * 0.5,
                                             eta_init=0.03, num_peaks=4)
            obj = compute_obj(a_new)
            if obj < best_score:
                best_score = obj
                best_seq = a_new.copy()
                print(f"  Peak deflation final: {best_score:.8f}")

            remaining = deadline - time.monotonic()
            if remaining > 15:
                a_new = fine_grained_search(best_seq, time_budget=min(remaining * 0.6, 30.0))
                obj = compute_obj(a_new)
                if obj < best_score:
                    best_score = obj
                    best_seq = a_new.copy()
                    print(f"  Fine search final: {best_score:.8f}")

    # Final validation
    if best_seq is None:
        best_seq = np.ones(100) / 100
        best_score = compute_obj(best_seq)

    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    try:
        val = evaluate_sequence(result)
        if val < best_score:
            best_score = val
    except Exception:
        pass

    elapsed = time.monotonic() - start_time
    print(f"\n{'='*60}")
    print(f"FINAL: n={len(best_seq)}, score={best_score:.8f}, time={elapsed:.1f}s")
    print(f"{'='*60}")

    return result
