import numpy as np

# Cache for sparse matrix structures
_A_cache = {}


def fast_conv(a):
    """Full self-convolution via FFT."""
    n = len(a)
    s = 1
    while s < 2 * n - 1:
        s <<= 1
    fa = np.fft.rfft(a, s)
    return np.fft.irfft(fa * fa, s)[:2 * n - 1]


def compute_J(a):
    """Compute J(a) = 2*n*max(conv(a,a))/(sum(a))^2."""
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))


def get_A_structure(n):
    """Get or pre-compute the (row, col) arrays for the constraint matrix."""
    if n not in _A_cache:
        rows = []
        cols = []
        for k in range(2 * n - 1):
            jl = max(0, k - n + 1)
            jh = min(n - 1, k)
            for j in range(jl, jh + 1):
                rows.append(k)
                cols.append(j)
        _A_cache[n] = (np.array(rows, dtype=np.int32),
                        np.array(cols, dtype=np.int32))
    return _A_cache[n]


def build_A_sparse(a):
    """Build sparse constraint matrix."""
    n = len(a)
    rows_arr, cols_arr = get_A_structure(n)
    vals = a[rows_arr - cols_arr]
    return csr_matrix((vals, (rows_arr, cols_arr)), shape=(2 * n - 1, n))


def lp_best_response(a, M):
    """LP: max sum(g) s.t. (a*g)_k <= M for all k, g >= 0."""
    n = len(a)
    c = -np.ones(n)
    A = build_A_sparse(a)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
    if r.success and r.x is not None:
        g = np.maximum(r.x, 0.0)
        g_sum = g.sum()
        if g_sum > 1e-10:
            return g / g_sum
    return None

    # Actually, let's just do it directly with the convolution structure

    #   a[k-j] * p[k]      (when k-j == j, i.e., k == 2j)
    # For simplicity (and since the diagonal is a small correction), use:
    # grad[j] = 2 * sum_{k: 0<=k-j<n} p[k] * a[k-j]
    # = 2 * (a convolved with p, evaluated appropriately)
    
    # Let's compute S[j] = sum_k p[k] * a[k-j] for j = 0, ..., n-1
    # = sum_m p[j+m] * a[m] where m ranges over 0 to n-1 and j+m ranges over 0 to 2n-2

    # = (a * p_reversed_full)[n-1+j] where p is zero-padded to length 2n-1 and reversed
    
    # Most efficient: use FFT
    # S[j] = sum_m a[m] * p[j+m]
    # Let q = p reversed (length 2n-1): q[t] = p[2n-2-t]
    # Then S[j] = sum_m a[m] * p[j+m] = sum_m a[m] * q[2n-2-j-m]
    # = (a * q)[n-1 + j]  ... hmm this is getting complicated
    
    # Let's just use a direct approach with the structure:
    # grad[j] = 2 * sum_{valid k} p[k] * a[k-j]
    # where k-j must be in [0, n-1], so k must be in [j, j+n-1] ∩ [0, 2n-2]
    
    if n <= 2048:
        # Direct computation (O(n^2) but fine for n <= 2048)
        grad = np.zeros(n)
        for k in range(2 * n - 1):
            if p[k] < 1e-15:
                continue

            jh = min(n - 1, k)
            for j in range(jl, jh + 1):
                k_j = k - j
                if 0 <= k_j < n:
                    grad[j] += 2.0 * p[k] * a[k_j]

        # = 2 * sum_m a[m] * p[m+j]  (letting m = k-j)
        # This is like a "running dot product"
        # Use the fact that this is the (n-1-j)-th coefficient of a(z) * p(1/z)
        # Or just: pad a to length s, pad p to length s, compute via FFT
        
        s = 1
        while s < 3 * n:
            s <<= 1

        # = cross-correlation
        # cross_corr = ifft(conj(fft(a)) * fft(p)) ... but we need the right indexing
        # Actually: (a and p cross-correlation at lag j) = sum_m a[m+j] * conj(p[m])
        # That's not quite what we want.
        # sum_m a[m] * p[m+j] = (a * reverse(p))[n-1+j] if lengths align properly
        # Let's just use the direct method for now, or a smarter FFT
        
        # Alternative: use the conjugate FFT trick
        # sum_m a[m] * p[m+j] = real part of ifft(fft(a) * conj(fft(p))) at appropriate index
        # This is getting complicated. Let's just do the O(n^2) loop but vectorized:

            pk = p[k]
            if pk < 1e-15:
                continue

            jh = min(n - 1, k)

            # = a[k-jh : k-jl+1] reversed
            seg = a[k - jh: k - jl + 1][::-1]

    return g_val, grad, k_max


def F_smoothed(a, beta):
    """Compute smoothed J(a) = 2n * smooth_max(conv(a,a)) / (sum(a))^2 and gradient."""
    n = len(a)
    s = float(a.sum())
    if s < 1e-10:
        return 1e10, np.zeros(n)
    g_val, grad_g, _ = smooth_max_and_subgrad(a, beta)
    F_val = 2.0 * n * g_val / (s * s)
    # d/da [2n * g / s^2] = 2n * (s * grad_g - 2*g * ones) / s^3
    grad_F = (2.0 * n * (s * grad_g - 2.0 * g_val)) / (s ** 3)
    return F_val, grad_F


def project_simplex(v):
    """Project vector onto the probability simplex (sum=1, all >= 0)."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - 1))[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0)


def nesterov_smoothed_descent(a0, deadline, beta_schedule, max_iters_per_beta=None):
    """
    Nesterov accelerated gradient descent on the smoothed objective,
    projected onto the simplex.
    """
    a = np.maximum(a0.astype(np.float64), 1e-8)
    a = a / a.sum()
    n = len(a)
    
    best_a = a.copy()
    best_J = compute_J(a)
    
    beta_list = beta_schedule
    
    for beta in beta_list:
        if time.monotonic() > deadline:
            break
        
        # Adaptive learning rate
        # The Lipschitz constant of the gradient is roughly 2*n*beta*max(a) / s^2 * something
        # Use a simple heuristic
        lr = 1.0 / (beta * n * 0.1)
        lr = min(lr, 1.0)
        
        momentum = 0.9
        v = np.zeros(n)
        y = a.copy()
        t_count = 1.0
        
        iters = 0
        max_it = max_iters_per_beta if max_iters_per_beta else 200
        
        for it in range(max_it):
            if time.monotonic() > deadline:
                break
            
            f_val, grad = F_smoothed(y, beta)
            if not np.isfinite(f_val):
                break
            
            # Nesterov update
            t_new = (1.0 + np.sqrt(1.0 + 4.0 * t_count ** 2)) / 2.0
            beta_mom = (t_new - 1.0) / t_new
            
            y_new = a - lr * grad
            y = (1.0 - beta_mom) * y_new + beta_mom * (a + (a - (a - lr * grad)))
            # Actually standard Nesterov:
            # y = a + (t-1)/t * (a - a_prev)
            # a_new = project(y - lr * grad(y))
            
            a_prev = a.copy()
            y = a + (t_count - 1.0) / t_new * (a - a_prev) if t_count > 1 else a
            t_count = t_new
            
            f_y, grad_y = F_smoothed(np.maximum(y, 1e-10), beta)
            a_new = y - lr * grad_y
            a_new = project_simplex(a_new)
            a = a_new
            
            # Occasionally check actual (unsmoothed) J
            if it % 50 == 0:
                J_actual = compute_J(a)
                if J_actual < best_J:
                    best_J = J_actual
                    best_a = a.copy()
            
            iters += 1
        
        # Check after each beta
        J_now = compute_J(a)
        if J_now < best_J:
            best_J = J_now
            best_a = a.copy()
    
    return best_a, best_J


def line_search_golden(a, g, n_evals=15):
    """Golden section search for best t in [0,1] minimizing J((1-t)*a + t*g)."""
    best_t = 0.0
    best_J = compute_J(a)
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    lo = 0.0
    hi = 1.0
    
    a_1 = g.copy()
    J_at_1 = compute_J(a_1)
    if J_at_1 < best_J:
        best_J = J_at_1
        best_t = 1.0
    
    for _ in range(n_evals):
        t1 = hi - gr * (hi - lo)
        t2 = lo + gr * (hi - lo)
        a1 = (1.0 - t1) * a + t1 * g
        a2 = (1.0 - t2) * a + t2 * g
        J1 = compute_J(a1)
        J2 = compute_J(a2)
        if J1 < J2:
            hi = t2
        else:
            lo = t1
    
    return (1.0 - best_t) * a + best_t * g, best_J, best_t


def lp_mirror_descent(a, max_iters, deadline, tracker=None, n_line_evals=15, stall_patience=30):
    """LP-based mirror descent with golden-section line search."""
    a = np.maximum(a.astype(np.float64), 1e-6)
    a = a / a.sum()
    current_J = compute_J(a)
    if tracker is not None:
        tracker.update(a)
    
    stall_count = 0
    best_J_so_far = current_J
    best_a_so_far = a.copy()
    
    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        
        b = fast_conv(a)
        M = float(b.max())
        
        g = lp_best_response(a, M)
        if g is None:
            break
        
        a_new, J_new, t_used = line_search_golden(a, g, n_evals=n_line_evals)
        
        if J_new < current_J - 1e-12:
            a = a_new
            current_J = J_new
            stall_count = 0
            if J_new < best_J_so_far:
                best_J_so_far = J_new
                best_a_so_far = a.copy()
                if tracker is not None:
                    tracker.update(a)
        else:
            # Try smaller steps
            improved = False
            for t_try in [0.01, 0.005, 0.02, 0.05, 0.001]:
                a_try = (1.0 - t_try) * a + t_try * g
                a_try = np.maximum(a_try, 0.0)
                s = a_try.sum()
                if s < 1e-10:
                    continue
                a_try = a_try / s
                J_try = compute_J(a_try)
                if J_try < current_J - 1e-14:
                    a = a_try
                    current_J = J_try
                    stall_count = 0
                    improved = True
                    if J_try < best_J_so_far:
                        best_J_so_far = J_try
                        best_a_so_far = a.copy()
                        if tracker is not None:
                            tracker.update(a)
                    break
            if not improved:
                stall_count += 1
                if stall_count >= stall_patience:
                    break
        
        if tracker is not None and it % 100 == 0:
            tracker.update(a)
    
    if tracker is not None:
        tracker.update(best_a_so_far)
    return best_a_so_far


def surgical_optimize(a, deadline, tracker=None, max_rounds=200):
    """
    Surgical local optimization: identify peak lags in convolution,
    then perform targeted mass transfers to reduce them.
    """
    a = np.maximum(a.astype(np.float64), 1e-10)
    a = a / a.sum()
    n = len(a)
    current_J = compute_J(a)
    
    if tracker is not None:
        tracker.update(a)
    
    best_a = a.copy()
    best_J = current_J
    
    for rnd in range(max_rounds):
        if time.monotonic() > deadline:
            break
        
        b = fast_conv(a)
        k_max = int(np.argmax(b))
        M = b[k_max]
        
        # Find the top contributing pairs to lag k_max
        # (conv(a,a))_{k_max} = sum_j a[j] * a[k_max - j]
        jl = max(0, k_max - n + 1)
        jh = min(n - 1, k_max)
        contributions = np.array([a[j] * a[k_max - j] for j in range(jl, jh + 1)])
        js = np.arange(jl, jh + 1)
        
        if len(contributions) == 0:
            break
        
        # Find the pair with the largest contribution
        idx_max = int(np.argmax(contributions))
        j_star = js[idx_max]
        j_partner = k_max - j_star
        
        # Strategy: transfer mass from the larger of a[j_star], a[j_partner]
        # to a random position that doesn't contribute to lag k_max
        if a[j_star] >= a[j_partner]:
            src = j_star
            partner = j_partner
        else:
            src = j_partner
            partner = j_star
        
        # Find candidate destination positions (those that don't affect lag k_max)
        # Position p affects lag k_max if p + (k_max - p) = k_max, which is always true
        # Wait, every position p in [0, n-1] potentially affects lag k_max
        # if k_max - p is also in [0, n-1]
        # So positions p where k_max - p is NOT in [0, n-1] are "safe"
        # That is: p < k_max - (n-1) or p > k_max
        safe_positions = []
        for p in range(n):
            if 0 <= k_max - p < n and p != src and p != partner:
                continue
            if p == src or p == partner:
                continue
            safe_positions.append(p)
        
        if len(safe_positions) == 0:

            # Transfer within the set, trying to equalize
            break

        # and pick the one that gives the most improvement
        best_dest = -1
        best_amount = 0.0
        best_J_try = current_J

                dest_candidates = safe_positions

                a_try = a.copy()
                a_try[src] -= amount
                a_try[dest] += amount
                a_try = np.maximum(a_try, 0.0)
                s = a_try.sum()
                if s < 1e-10:
                    continue
                a_try = a_try / s
                J_try = compute_J(a_try)
                if J_try < best_J_try:
                    best_J_try = J_try
                    best_dest = int(dest)
                    best_amount = amount
        
        if best_dest >= 0 and best_J_try < current_J - 1e-12:
            a[src] -= best_amount
            a[best_dest] += best_amount

            a = a / a.sum()

                    current_J = J_try

                            best_a = a.copy()
                            if tracker is not None:
                                tracker.update(a)
                        break
    
    if tracker is not None:
        tracker.update(best_a)
    return best_a


def multi_lag_deflation(a, deadline, tracker=None, max_rounds=100):
    """
    Deflate multiple lags simultaneously by identifying the top m peaks
    and performing collective adjustments.
    """
    a = np.maximum(a.astype(np.float64), 1e-10)
    a = a / a.sum()
    n = len(a)
    current_J = compute_J(a)
    best_a = a.copy()
    best_J = current_J
    top_m = min(5, n)
    
    for rnd in range(max_rounds):
        if time.monotonic() > deadline:
            break
        
        b = fast_conv(a)
        # Get top m peaks
        top_indices = np.argsort(b)[::-1][:top_m]
        
        # For each top peak, find a "safe" direction to push mass
        improved = False
        for k_peak in top_indices:
            if time.monotonic() > deadline:
                break
            
            jl = max(0, k_peak - n + 1)
            jh = min(n - 1, k_peak)
            if jl > jh:
                continue
            
            # Find the position with the largest a value that contributes to this peak
            # and move it to a position that doesn't contribute
            contrib = a[jl:jh+1] * a[k_peak - jh: k_peak - jl + 1][::-1]
            j_local = int(np.argmax(contrib))
            j_src = jl + j_local
            j_part = k_peak - j_src
            
            # Find safe destinations
            safe = []
            for p in range(n):
                if p == j_src or p == j_part:
                    continue
                if 0 <= k_peak - p < n:
                    continue
                safe.append(p)
            
            if len(safe) == 0:
                continue
            
            # Move a small fraction
            for frac in [0.02, 0.05, 0.1]:
                amt = a[j_src] * frac
                dest = safe[np.random.randint(len(safe))]
                a_try = a.copy()
                a_try[j_src] -= amt
                a_try[dest] += amt
                a_try = np.maximum(a_try, 0.0)
                s = a_try.sum()
                if s < 1e-10:
                    continue
                a_try = a_try / s
                J_try = compute_J(a_try)
                if J_try < current_J - 1e-13:
                    a = a_try
                    current_J = J_try
                    improved = True
                    if current_J < best_J:
                        best_J = current_J
                        best_a = a.copy()
                        if tracker is not None:
                            tracker.update(a)
                    break
        
        if not improved and rnd > 20:
            break
    
    if tracker is not None:
        tracker.update(best_a)
    return best_a


def resample_seq(src, dest_n):
    """Resample 1D sequence to new length via linear interpolation."""
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    result = np.interp(t_new, t_old, src)
    return np.maximum(result, 0.0)


def generate_initializations(n, rng, parent=None):
    """Generate diverse initial sequences of length n."""
    inits = []
    t_arr = np.linspace(0, 1, n, endpoint=False)
    
    if parent is not None and parent.size > 0:
        inits.append(resample_seq(parent, n))
        p = resample_seq(parent, n)
        p = p + rng.normal(0, 0.03, n) * p
        inits.append(np.maximum(p, 0.01))
    
    inits.append(np.ones(n))
    inits.append(rng.uniform(0.3, 1.0, n))
    
    # Cosine families with various frequencies
    for freq in [2, 3, 4, 5, 7, 8, 12]:
        inits.append(1.0 + 0.4 * np.cos(2.0 * np.pi * freq * t_arr))
    
    # Ensure non-negativity
    inits = [np.maximum(x, 0.01) for x in inits]
    
    # Additional structured inits
    inits.append(np.exp(-3.0 * (t_arr - 0.5) ** 2) + 0.1)
    inits.append(np.abs(np.sin(np.pi * t_arr)) + 0.2)
    inits.append(1.0 + 0.3 * np.cos(4.0 * np.pi * t_arr) * np.cos(2.0 * np.pi * t_arr))
    
    return inits


class ScoreTracker:
    """Tracks the best sequence found so far."""
    
    def __init__(self):
        self.best = None
        self.best_score = float('inf')
        self.t_start = time.monotonic()
    
    def update(self, a):
        a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
        if a.size == 0 or a.sum() < 0.01:
            return
        score = float(evaluate_sequence(a.tolist()))
        if np.isfinite(score) and score < self.best_score:
            self.best = a.copy()
            self.best_score = score
            elapsed = time.monotonic() - self.t_start
            print(f"  * NEW BEST: J={score:.8f} n={a.size} t={elapsed:.1f}s", flush=True)
    
    def report(self, label=""):
        if self.best is not None:
            print(f"  {label} best: J={self.best_score:.8f} n={self.best.size}", flush=True)


def propose_candidate(seed=42, budget_s=1500, **kwargs):
    """
    Multi-phase optimization combining:
    1. Nesterov smoothed gradient descent (fast broad search)
    2. LP mirror descent (precise refinement)
    3. Surgical local optimization (targeted peak reduction)
    4. Multi-lag deflation (collective peak management)
    5. Length exploration
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + float(budget_s) - 15.0
    tracker = ScoreTracker()
    
    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        if p.ndim == 1 and p.size > 0 and np.all(np.isfinite(p)):
            parent = np.maximum(p, 0.0)
            tracker.update(parent)
            print(f"Parent loaded: n={parent.size}, J={tracker.best_score:.8f}", flush=True)
    except (NameError, AttributeError):
        pass
    
    if parent is None:
        parent = np.ones(512)
        tracker.update(parent)
        print(f"Using default parent: n=512, J={tracker.best_score:.8f}", flush=True)
    
    # === Phase 1: Nesterov smoothed gradient descent (broad search) ===
    t0 = time.monotonic()
    phase1_budget = 150.0
    deadline_p1 = time.monotonic() + phase1_budget
    print(f"\n{'='*50} Phase 1: Nesterov GD (budget {phase1_budget:.0f}s) {'='*50}", flush=True)
    
    lengths_p1 = [256, 512, 1024]
    beta_schedule_p1 = [1.0, 3.0, 10.0, 30.0, 100.0]
    
    for n in lengths_p1:
        if time.monotonic() > deadline_p1:
            break
        t_start = time.monotonic()
        print(f"  Nesterov n={n}...", flush=True)
        
        inits = generate_initializations(n, rng, parent)
        max_inits = min(len(inits), 5)
        
        for idx, a0 in enumerate(inits[:max_inits]):
            if time.monotonic() > deadline_p1:
                break
            a0 = np.maximum(a0.astype(np.float64), 0.01)
            a0 = a0 / a0.sum()
            a_opt, J_opt = nesterov_smoothed_descent(
                a0, deadline_p1, beta_schedule_p1, max_iters_per_beta=150
            )
            if a_opt is not None and a_opt.sum() > 0.01:
                tracker.update(a_opt)
        
        print(f"    n={n} done in {time.monotonic()-t_start:.1f}s", flush=True)
    
    tracker.report("Phase 1 complete")
    
    # === Phase 2: LP mirror descent (main refinement) ===
    if tracker.best is not None and time.monotonic() < deadline:
        t0 = time.monotonic()
        n_best = tracker.best.size
        phase2_budget = min(600.0, deadline - time.monotonic())
        deadline_p2 = time.monotonic() + phase2_budget
        print(f"\n{'='*50} Phase 2: LP MD n={n_best} (budget {phase2_budget:.0f}s) {'='*50}", flush=True)
        
        remaining = deadline_p2 - time.monotonic()
        max_iters = int(remaining * 50)
        max_iters = min(max_iters, 5000)
        a = lp_mirror_descent(tracker.best, max_iters, deadline_p2, tracker=tracker,
                             n_line_evals=15, stall_patience=35)
        
        # Try at other lengths
        for n_try in [512, 1024, 2048]:
            if time.monotonic() > deadline_p2 or n_try == n_best:
                continue
            a_res = resample_seq(tracker.best, n_try)
            a_res = a_res / max(a_res.sum(), 1e-10)
            budget_here = min(150.0, deadline_p2 - time.monotonic())
            max_iters_here = int(budget_here * 35)
            max_iters_here = min(max_iters_here, 2000)
            a_res = lp_mirror_descent(a_res, max_iters_here, deadline_p2, tracker=tracker,
                                      n_line_evals=12, stall_patience=25)
        
        tracker.report(f"Phase 2: {time.monotonic()-t0:.1f}s")
    
    # === Phase 3: Surgical optimization ===
    if tracker.best is not None and time.monotonic() < deadline:
        t0 = time.monotonic()
        n_best = tracker.best.size
        phase3_budget = min(250.0, deadline - time.monotonic())
        deadline_p3 = time.monotonic() + phase3_budget
        print(f"\n{'='*50} Phase 3: Surgical Opt n={n_best} (budget {phase3_budget:.0f}s) {'='*50}", flush=True)
        
        # Run surgical optimization on the best
        a_surg = surgical_optimize(tracker.best, deadline_p3, tracker=tracker, max_rounds=500)
        
        # Then run multi-lag deflation
        if time.monotonic() < deadline_p3:
            a_defl = multi_lag_deflation(tracker.best, deadline_p3, tracker=tracker, max_rounds=200)
        
        # Alternate between surgical and LP for a while
        if time.monotonic() < deadline_p3 and tracker.best is not None:
            remaining_p3 = deadline_p3 - time.monotonic()
            max_iters_lp = int(remaining_p3 * 40)
            max_iters_lp = min(max_iters_lp, 1500)
            a = lp_mirror_descent(tracker.best, max_iters_lp, deadline_p3, tracker=tracker,
                                  n_line_evals=15, stall_patience=25)
        
        tracker.report(f"Phase 3: {time.monotonic()-t0:.1f}s")
    
    # === Phase 4: Multi-restart with diverse perturbations ===
    if tracker.best is not None and time.monotonic() < deadline:
        t0 = time.monotonic()
        n_best = tracker.best.size
        phase4_budget = min(200.0, deadline - time.monotonic())
        deadline_p4 = time.monotonic() + phase4_budget
        print(f"\n{'='*50} Phase 4: Restarts (budget {phase4_budget:.0f}s) {'='*50}", flush=True)
        
        strategies = [
            ('gaussian', 0.05),
            ('gaussian', 0.1),
            ('smoothing', 0.03),
            ('spike', 5),
            ('gaussian', 0.02),
            ('uniform', 0.08),
        ]
        
        attempt = 0
        for pert_type, param in strategies:
            for trial in range(4):
                if time.monotonic() > deadline_p4:
                    break
                attempt += 1
                a = tracker.best.copy().astype(np.float64)
                n = len(a)
                
                if pert_type == 'gaussian':
                    a = a + rng.normal(0, param, n) * a
                elif pert_type == 'uniform':
                    a = a + rng.uniform(-param, param, n) * a
                elif pert_type == 'spike':
                    k = min(param, n // 8)
                    idx = rng.choice(n, size=k, replace=False)
                    a[idx] *= rng.uniform(0.5, 2.0, k)
                elif pert_type == 'smoothing':
                    kernel_size = max(3, int(n * param))
                    kernel = np.ones(kernel_size) / kernel_size
                    a = np.convolve(a, kernel, mode='same')
                
                a = np.maximum(a, 1e-4)
                a = a / a.sum()
                
                J_pert = compute_J(a)
                if J_pert < tracker.best_score:
                    tracker.update(a)
                
                # Quick LP refinement
                budget_here = min(40.0, deadline_p4 - time.monotonic())
                max_iters = int(budget_here * 40)
                max_iters = min(max_iters, 1000)
                a = lp_mirror_descent(a, max_iters, deadline_p4, tracker=tracker,
                                      n_line_evals=12, stall_patience=20)
        
        tracker.report(f"Phase 4: {time.monotonic()-t0:.1f}s, {attempt} attempts")
    
    # === Phase 5: Final polish ===
    if tracker.best is not None and time.monotonic() < deadline:
        t0 = time.monotonic()
        n_best = tracker.best.size
        remaining = deadline - time.monotonic()
        phase5_budget = min(150.0, remaining)
        deadline_p5 = time.monotonic() + phase5_budget
        print(f"\n{'='*50} Phase 5: Final Polish (budget {phase5_budget:.0f}s) {'='*50}", flush=True)
        
        # Long LP run with more precise line search
        max_iters = int(phase5_budget * 45)
        max_iters = min(max_iters, 3000)
        a = lp_mirror_descent(tracker.best, max_iters, deadline_p5, tracker=tracker,
                             n_line_evals=20, stall_patience=40)
        
        # Surgical finish
        if time.monotonic() < deadline_p5:
            surgical_optimize(tracker.best, deadline_p5, tracker=tracker, max_rounds=200)
        
        # One more LP pass
        if tracker.best is not None and time.monotonic() < deadline_p5:
            remaining2 = deadline_p5 - time.monotonic()
            if remaining2 > 20:
                max_iters2 = int(remaining2 * 40)
                max_iters2 = min(max_iters2, 1500)
                lp_mirror_descent(tracker.best, max_iters2, deadline_p5, tracker=tracker,
                                  n_line_evals=18, stall_patience=30)
        
        tracker.report(f"Phase 5: {time.monotonic()-t0:.1f}s")
    
    # === Phase 6: Try adjacent lengths if time remains ===
    if tracker.best is not None and time.monotonic() < deadline:
        t0 = time.monotonic()
        n_best = tracker.best.size
        remaining = deadline - time.monotonic()
        if remaining > 30:
            print(f"\n{'='*50} Phase 6: Length Exploration (budget {remaining:.0f}s) {'='*50}", flush=True)
            
            candidates = [n_best - 64, n_best + 64, n_best // 2, n_best * 2]
            candidates = [c for c in candidates if 64 <= c <= 4096 and c != n_best]
            
            for n_try in candidates:
                if time.monotonic() > deadline:
                    break
                a_res = resample_seq(tracker.best, n_try)
                a_res = a_res / max(a_res.sum(), 1e-10)
                budget_here = min(30.0, deadline - time.monotonic())
                max_iters = int(budget_here * 35)
                max_iters = min(max_iters, 800)
                a_res = lp_mirror_descent(a_res, max_iters, deadline, tracker=tracker,
                                          n_line_evals=12, stall_patience=15)
                surgical_optimize(tracker.best, min(deadline, time.monotonic() + 10),
                                  tracker=tracker, max_rounds=50)
            
            tracker.report(f"Phase 6: {time.monotonic()-t0:.1f}s")
    
    # Fallback
    if tracker.best is None:
        tracker.best = np.ones(512)
        tracker.best_score = compute_J(tracker.best)
        print(f"  Using fallback: J={tracker.best_score:.8f}", flush=True)
    
    print(f"\n{'='*50} FINAL RESULT {'='*50}", flush=True)
    print(f"J={tracker.best_score:.8f} n={tracker.best.size} "
          f"time={time.monotonic()-tracker.t_start:.1f}s", flush=True)
    
    # Ensure output is valid
    result = np.clip(tracker.best, 0.0, 1000.0)
    if result.sum() < 0.01:
        result = np.ones(128)
    
    return [float(x) for x in result]
