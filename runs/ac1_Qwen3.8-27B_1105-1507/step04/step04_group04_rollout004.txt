def compute_conv(a):
    """Compute full conv(a,a) using FFT. Returns array of length 2n-1."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    conv = np.fft.irfft(fa * fa, size)
    return conv[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2. Assumes a >= 0."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 1e-10:
        return np.inf
    # Scale invariant: normalize to sum=1
    a_norm = a / s
    conv = compute_conv(a_norm)
    max_b = float(np.max(conv))
    return 2.0 * n * max_b


def compute_subgradient(a):
    """
    Compute the subgradient of max(conv(a,a)) with respect to a.
    At the argmax location k*, the subgradient[i] = a[k*-i] if 0 <= k*-i < n, else 0.
    Returns (grad, k_star) where grad has the same length as a.

    n = len(a)
    conv = compute_conv(a)
    k_star = int(np.argmax(conv))
    # subgrad[i] = sum over j of a[j] * 1[i+j == k_star] = a[k_star - i] if valid
    grad = np.zeros(n)
    i_min = max(0, k_star - n + 1)
    i_max = min(n - 1, k_star)
    for i in range(i_min, i_max + 1):
        j = k_star - i
        if 0 <= j < n:
            grad[i] += a[j]
    return grad, k_star, float(conv[k_star])


def exact_subgradient_step(a, eta, k_star=None):

    s_new = np.sum(a_new)

    return a_new / s_new, k_star

def build_toeplitz(a):
    """Build Toeplitz matrix A where (A@g)[k] = (a*g)[k] for k=0..2n-2."""
    n = len(a)
    A = np.zeros((2 * n - 1, n), dtype=np.float64)
    for j in range(n):
        k_start = j
        k_end = min(j + n, 2 * n - 1)
        A[k_start:k_end, j] = a[:k_end - k_start]
    return A


def solve_lp_direction(a, M, time_limit=5):
    """
    Solve LP: max sum(g) s.t. (a*g)[k] <= M for all k, g >= 0.
    Returns (g_normalized, success) where g_normalized has sum=1.
    """
    n = len(a)
    if HAS_CVXPY:
        try:
            g_var = cp.Variable(n, nonneg=True)
            A = build_toeplitz(a)
            constraints = [A @ g_var <= M]
            prob = cp.Problem(cp.Maximize(cp.sum(g_var)), constraints)
            for solver in [cp.MOSEK, cp.CLARABEL, cp.SCS, cp.ECOS]:
                try:
                    prob.solve(solver=solver, time_limit=int(time_limit), verbose=False)
                    if g_var.value is not None:
                        g_val = np.maximum(g_var.value, 0)
                        s = np.sum(g_val)
                        if s > 1e-10:
                            return g_val / s, True
                except Exception:
                    continue
        except Exception:
            pass
    
    A = build_toeplitz(a)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': time_limit})
        if res.status == 0:
            g = np.maximum(res.x, 0)
            s = np.sum(g)
            if s > 1e-10:
                return g / s, True
    except Exception:
        pass
    return None, False

def resample_to(a, new_n):
    """Resample sequence to new length."""
    n = len(a)
    if n == new_n:
        return a.copy()
    old_k = np.linspace(0, 1, n)
    new_k = np.linspace(0, 1, new_n)
    a_new = np.interp(new_k, old_k, a)
    a_new = np.maximum(a_new, 0)
    s = np.sum(a_new)
    if s > 1e-10:
        a_new = a_new / s
    return a_new

    rng = np.random.default_rng(999)
    
    if verbose:
        print(f"    Polish start: obj={best_obj:.8f}")
    
    attempt = 0
    while time.monotonic() - start < time_budget - 10:
        attempt += 1
        remaining = time_budget - (time.monotonic() - start)
        if remaining < 15:
            break
        
        scale = 0.01 + 0.03 * (attempt % 3) / 3.0
        
        # Generate perturbation
        a_pert = best_a * np.exp(rng.normal(0, scale, n))
        a_pert = np.maximum(a_pert, 1e-10)
        a_pert = a_pert / np.sum(a_pert)
        
        # Quick optimize
        sub_time = min(12.0, remaining * 0.4)
        a_opt, obj = optimize_single(a_pert, time_budget=sub_time, verbose=False)
        
        if obj < best_obj - 1e-12:
            best_obj = obj
            best_a = a_opt.copy()
            if verbose:
                print(f"    Polish attempt {attempt}: {best_obj:.8f}")
    
    return best_a, best_obj


def propose_candidate(seed=42, budget_s=1500):
    """
    Main search: multi-n exploration with Nesterov SG, LP, coordinate descent,
    and intensive polishing. Focuses on n values around the parent length.
    """
    start_time = time.monotonic()
    deadline = start_time + float(budget_s) - 30.0
    
    best_seq = None
    best_score = np.inf
    
    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0, 1000.0)
        if p.size > 0 and np.isfinite(p).all():
            parent = p.copy()
            s = np.sum(parent)
            if s > 0.01:
                parent = parent / s  # normalize
                obj_p = compute_obj(parent)
                if obj_p < best_score:
                    best_score = obj_p
                    best_seq = parent.copy()
                print(f"Parent (n={parent.size}): obj={obj_p:.6f}")
    except Exception:
        pass
    
    if parent is None:
        parent = np.ones(300) / 300
        best_score = compute_obj(parent)
        best_seq = parent.copy()
    
    n_parent = len(best_seq) if best_seq is not None else 300
    print(f"Initial best: {best_score:.6f} (n={n_parent})")
    print(f"Time budget: {budget_s}s, deadline in {deadline - start_time:.0f}s")
    
    # Phase 1: Intensive optimization of parent
    print("\n=== Phase 1: Optimize parent ===")
    remaining = deadline - time.monotonic()
    p1_budget = min(remaining * 0.30, 300.0)
    a_opt, obj = optimize_single(best_seq, time_budget=p1_budget, verbose=True)
    if obj < best_score:
        best_score = obj
        best_seq = a_opt.copy()
        print(f"  After optimize_single: {best_score:.6f}")
    
    # Intensive polish
    remaining = deadline - time.monotonic()
    if remaining > 30:
        p1b_budget = min(remaining * 0.15, 150.0)
        a_opt, obj = intensive_polish(best_seq, time_budget=p1b_budget, verbose=True)
        if obj < best_score:
            best_score = obj
            best_seq = a_opt.copy()
            print(f"  After polish: {best_score:.6f}")
    
    # Phase 2: Try nearby n values
    print("\n=== Phase 2: Nearby n values ===")
    deltas = [0, -10, 10, -20, 20, -30, 30, -50, 50, -40, 40, -60, 60]
    n_candidates = []
    for d in deltas:
        nv = n_parent + d
        if 150 <= nv <= 1200:
            n_candidates.append(nv)
    
    # Sort by proximity to parent
    n_candidates = sorted(set(n_candidates), key=lambda x: abs(x - n_parent))
    
    remaining = deadline - time.monotonic()
    if remaining > 30:
        # Allocate time based on proximity
        for n_val in n_candidates:
            if time.monotonic() > deadline - 40:
                break
            remaining = deadline - time.monotonic()
            if remaining < 25:
                break
            if n_val == n_parent:
                continue  # Already done
            
            # Time allocation: closer n values get more time
            proximity = max(0.3, 1.0 - abs(n_val - n_parent) / 80.0)
            alloc = min(remaining * 0.25 * proximity, 80.0)
            alloc = max(alloc, 15.0)
            
            print(f"\n  n={n_val} (remaining={remaining:.0f}s, alloc={alloc:.0f}s)")
            
            # Use resampled parent as starting point
            a_start = resample_to(best_seq, n_val)
            a_opt, obj = optimize_single(a_start, time_budget=alloc, verbose=False)
            
            if obj < best_score:
                best_score = obj
                best_seq = a_opt.copy()
                print(f"  ** UPDATE: n={n_val}, obj={best_score:.6f} **")
            
            # Try one random init too
            remaining = deadline - time.monotonic()
            if remaining > 20 and alloc > 20:
                a_rand = random_init_n(n_val, seed + n_val * 104729)
                a_opt2, obj2 = optimize_single(a_rand, time_budget=min(20.0, remaining * 0.2), verbose=False)
                if obj2 < best_score:
                    best_score = obj2
                    best_seq = a_opt2.copy()
                    print(f"  ** UPDATE (random): n={n_val}, obj={best_score:.6f} **")
    
    # Phase 3: Final intensive polish on best
    if best_seq is not None:
        remaining = deadline - time.monotonic()
        if remaining > 30:
            print(f"\n=== Phase 3: Final polish ({remaining:.0f}s) ===")
            p3_budget = min(remaining * 0.4, remaining - 15.0)
            a_opt, obj = intensive_polish(best_seq, time_budget=p3_budget, verbose=True)
            if obj < best_score:
                best_score = obj
                best_seq = a_opt.copy()
                print(f"  Final polish: {best_score:.6f}")
            
            # One more full optimization round
            remaining = deadline - time.monotonic()
            if remaining > 20:
                p3b_budget = min(remaining * 0.35, remaining - 10.0)
                a_opt, obj = optimize_single(best_seq, time_budget=p3b_budget, verbose=True)
                if obj < best_score:
                    best_score = obj
                    best_seq = a_opt.copy()
                    print(f"  Final full opt: {best_score:.6f}")
    
    # Phase 4: Try a few far n values if time permits
    if best_seq is not None:
        remaining = deadline - time.monotonic()
        if remaining > 40:
            print(f"\n=== Phase 4: Far n values ({remaining:.0f}s) ===")
            far_ns = [200, 250, 500, 600, 400]
            for n_val in far_ns:
                if time.monotonic() > deadline - 20:
                    break
                if abs(n_val - len(best_seq)) < 30:
                    continue
                remaining = deadline - time.monotonic()
                if remaining < 25:
                    break
                alloc = min(30.0, remaining * 0.25)
                a_opt, obj = optimize_single(resample_to(best_seq, n_val), 
                                             time_budget=alloc, verbose=False)
                if obj < best_score:
                    best_score = obj
                    best_seq = a_opt.copy()
                    print(f"  ** UPDATE: n={n_val}, obj={best_score:.6f} **")
    
    # Final validation
    if best_seq is None:
        best_seq = np.ones(100) / 100
    
    # Ensure all values are valid
    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    
    # Verify with evaluate_sequence
    try:
        val = evaluate_sequence(result)
        if np.isfinite(val) and val < best_score:
            best_score = val
            print(f"evaluate_sequence gives: {best_score:.6f}")
    except Exception:
        pass
    
    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_score:.6f}, time={elapsed:.1f}s ===")
    
    return result
