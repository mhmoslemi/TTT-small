**Key insight**: The LP best-response is the most powerful tool here. The question is how to use it most effectively.

**My strategy:**

**Important optimization**: The LP is the bottleneck. For n=512, the LP has n=512 variables and 2n-1=1023 constraints. This is not too large for Modern HiGHS solver. The key is to:
- Use warm starts
- Presolve
- Set appropriate tolerances
- Limit time per LP call

Let me also think about a **dual approach**: Instead of solving the LP directly, I can use the dual variables to get gradient information.

The primal LP is:
- Maximize `1^T g`
- Subject to `A g <= M * 1`, `g >= 0`

where `A` is the `(2n-1) x n` convolution matrix.

The dual is:
- Minimize `M * 1^T y`
- Subject to `A^T y >= 1`, `y >= 0`

The optimal dual variables `y` tell us which constraints are "most important". The support of `y` gives the active set.

I can use the dual variables to construct a "gradient" for updating `a`:
- The shadow price of tightening `M` is `y^T * 1` (the optimal value).
- If I increase `M` slightly, the optimal value increases by `y^T * 1 * dM`.

This suggests a **penalty method**: Instead of hard constraints, use a smooth penalty on the violation:
- Minimize `max_k (a*a)_k + lambda * max(0, max_k (a*a)_k - M)^2`

Or better, use the **Lagrangian dual**:
- For the inner problem (given `a`), the optimal `g` satisfies `convolve(a, g) <= M`.
- The "dual gradient" tells us how to update `M` (or equivalently, how to update `a`).

Actually, I think the cleanest approach is:

1. **Alternating LP + Line Search**:
   - Given `a`, compute `b = convolve(a,a)`, `M = max(b)`.
   - Solve LP to get best response `g` with bound `M*(1-eps)`.
   - Do line search along `a + t*(g - a)` (projected onto simplex) to minimize J.
   - Repeat.

2. **Smoothed approach**:
   - Replace `max` with LSE (Log-Sum-Exp) for a differentiable objective.
   - Use proximal gradient / FISTA with the simplex projection.

"""
Minimizes J(a) = 2*n*max(convolve(a,a))/(sum(a))^2 using:
- LP best-response with warm starts and adaptive tightening
- Multiplicative weight (mirror descent) updates
- LSE-smoothed gradient descent with backtracking
- Multi-length search with diverse initializations
- Simulated annealing for escape from local minima
"""

import numpy as np
from scipy.optimize import linprog
from scipy.sparse import csr_matrix
import time

# Cache for sparsity pattern
_spar_cache = {}

def get_sparsity(n):
    if n not in _spar_cache:
        rows = []
        cols = []
        for k in range(2 * n - 1):
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            for j in range(js, je + 1):
                rows.append(k)
                cols.append(j)
        _spar_cache[n] = (np.array(rows), np.array(cols))
    return _spar_cache[n]

def fast_conv(a):
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]

def compute_J(a):
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))

def lp_solve(a, M, time_limit=5.0):
    """Solve LP: max sum(g) s.t. convolve(a,g) <= M, g >= 0"""
    n = len(a)
    rows, cols = get_sparsity(n)
    nnz = len(rows)
    vals = np.empty(nnz)
    for i in range(nnz):
        k = rows[i]
        j = cols[i]
        vals[i] = a[k - j]
    A = csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                options={'presolve': True, 'time_limit': time_limit})
    if r.success and r.x is not None:
        return np.maximum(r.x, 0.0)
    return None

def lp_best_response_iteration(a, max_iters, deadline, tracker, base_tighten=0.01):
    """Main LP-based optimization loop."""
    a = np.maximum(a.astype(np.float64), 1e-10)
    a /= a.sum()
    n = len(a)
    current_J = compute_J(a)
    tracker.update(a)
    best_a = a.copy()
    
    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        b = fast_conv(a)
        M = float(b.max())
        
        # Adaptive tightening: start tight, relax if LP fails
        tighten = base_tighten * (1.0 + 0.5 * np.sin(it * 0.1))
        M_eff = M * (1.0 - max(0.001, tighten))
        
        g = lp_solve(a, M_eff, time_limit=3.0)
        if g is None:
            g = lp_solve(a, M, time_limit=3.0)
        if g is None:
            break
        
        gs = g.sum()
        if gs < 1e-10:
            break
        g /= gs
        
        # Line search
        best_t = 0.0
        best_J = current_J
        progress = it / max(1, max_iters)
        t_candidates = [max(0.1, 0.5*(1-progress)), 0.3, 0.5, 0.7, 0.1, 0.9]
        for t in t_candidates:
            a_trial = (1.0 - t) * a + t * g
            a_trial = np.maximum(a_trial, 0.0)
            ss = a_trial.sum()
            if ss < 1e-10:
                continue
            a_trial /= ss
            J_trial = compute_J(a_trial)
            if J_trial < best_J - 1e-14:
                best_J = J_trial
                best_t = t
        
        if best_t == 0.0:
            # No improvement, try smaller steps
            for t in [0.05, 0.02, 0.01]:
                a_trial = (1.0 - t) * a + t * g
                a_trial = np.maximum(a_trial, 0.0)
                ss = a_trial.sum()
                if ss < 1e-10:
                    continue
                a_trial /= ss
                J_trial = compute_J(a_trial)
                if J_trial < best_J - 1e-14:
                    best_J = J_trial
                    best_t = t
                    break
            if best_t == 0.0:
                break
        
        a = (1.0 - best_t) * a + best_t * g
        a = np.maximum(a, 0.0)
        a /= a.sum()
        current_J = best_J
        best_a = a.copy()
        tracker.update(a)
    
    return best_a

def mirror_descent(a, max_iters, deadline, tracker):
    """Multiplicative weight / mirror descent on the simplex."""
    a = np.maximum(a.astype(np.float64), 1e-10)
    a /= a.sum()
    n = len(a)
    current_J = compute_J(a)
    tracker.update(a)
    best_a = a.copy()
    best_J = current_J
    
    eta = 0.5 / np.sqrt(n)
    patience = 0
    max_patience = 300
    
    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        
        b = fast_conv(a)
        k_max = int(np.argmax(b))
        M = b[k_max]
        
        # Gradient of (a*a)_{k_max} w.r.t. a
        grad = np.zeros(n)
        for m in range(n):
            idx = k_max - m
            if 0 <= idx < n:
                grad[m] = 2.0 * a[idx]
        
        # Also consider the effect on J: J = 2n * max(b) / s^2, s = 1
        # dJ/da_m = 2n * grad_m (since s=1, d(m*b)/da_m = grad_m, and d(s^-2)/da_m = -2/s^3 * 1 = 0 when s=1)
        # Actually: J = 2n * M / s^2. dJ/da_m = 2n * (dM/da_m * s^2 - M * 2s * ds/da_m) / s^4
        # = 2n * (grad_m * s - 2*M*a_m/s) / s^2  (with s = sum(a) = 1)
        # = 2n * (grad_m - 2*M*a_m)  (when s=1)
        
        full_grad = grad - 2.0 * M * a
        
        # Mirror descent: a_m <- a_m * exp(-eta * full_grad_m)
        # Clip for numerical stability
        exponent = -eta * full_grad
        exponent = np.clip(exponent, -50, 50)
        a_new = a * np.exp(exponent)
        a_new = np.maximum(a_new, 1e-15)
        ss = a_new.sum()
        if ss < 1e-10:
            break
        a_new /= ss
        
        J_new = compute_J(a_new)
        if J_new < best_J - 1e-14:
            best_J = J_new
            best_a = a_new.copy()
            a = a_new
            current_J = J_new
            patience = 0
        else:
            patience += 1
            a = a_new  # still move, but track best
            if patience % 30 == 0:
                eta *= 0.85
            if patience > max_patience:
                break
        
        if it % 100 == 0:
            tracker.update(a)
    
    a = best_a
    tracker.update(a)
    return a

def lse_polish(a, max_iters, deadline, tracker):
    """Polish using LSE-smoothed gradient with backtracking line search."""
    a = np.maximum(a.astype(np.float64), 1e-8)
    a /= a.sum()
    n = len(a)
    current_J = compute_J(a)
    tracker.update(a)
    best_a = a.copy()
    best_J = current_J
    
    tau = 100.0  # LSE temperature (higher = closer to max)
    lr = 0.1
    momentum = 0.9
    v = np.zeros(n)
    
    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        
        b = fast_conv(a)
        b_max = b.max()
        exp_b = np.exp(tau * (b - b_max))
        p = exp_b / exp_b.sum()
        
        # Gradient of LSE: sum_k p_k * grad(b_k)
        # grad(b_k)_m = 2*a[k-m] if 0<=k-m<n
        # But this is O(n^2). Use FFT:
        # The gradient is 2 * cross-correlation of p and a (truncated to n)
        size = 1
        while size < 3 * n:
            size <<= 1
        fp = np.fft.rfft(p, size)
        fa = np.fft.rfft(a, size)
        # We need: grad_m = sum_k p_k * 2 * a[k-m] (for valid k-m)
        # This is 2 * (p * a)_m where * is cross-correlation
        # cross_corr[p, a]_m = sum_k p_k * a_{k-m}
        # In FFT: IFFT(FFT(p) * conj(FFT(a))) gives cross-correlation
        corr = np.fft.irfft(fp * np.conj(fa), size)
        grad = 2.0 * corr[:n]
        
        # Add the -4n*M*a/s^2 term (for J = 2n*max(b)/s^2, s=1)
        # dJ/da = 2n*grad_LSE - 4n*max(b)*a (approximately)
        # But since we're using LSE as proxy for max, just use the LSE gradient
        full_grad = grad - 2.0 * b_max * a
        
        # Momentum update
        v = momentum * v + (1.0 - momentum) * full_grad
        
        # Backtracking line search
        a_trial = a - lr * v
        a_trial = np.maximum(a_trial, 1e-8)
        a_trial /= a_trial.sum()
        J_trial = compute_J(a_trial)
        
        count = 0
        while J_trial > current_J - 1e-14 and count < 10:
            lr *= 0.5
            a_trial = a - lr * v
            a_trial = np.maximum(a_trial, 1e-8)
            a_trial /= a_trial.sum()
            J_trial = compute_J(a_trial)
            count += 1
        
        if J_trial < current_J - 1e-14:
            a = a_trial
            current_J = J_trial
            if current_J < best_J:
                best_J = current_J
                best_a = a.copy()
            lr = min(lr * 1.05, 1.0)
            tau = min(500.0, tau * 1.005)
        else:
            lr = max(lr * 0.5, 1e-8)
            if lr < 1e-10:
                break
        
        if it % 50 == 0:
            tracker.update(a)
    
    a = best_a
    tracker.update(a)
    return a

def resample_seq(src, dest_n):
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)

def generate_inits(n, rng, parent):
    """Generate diverse initial sequences."""
    inits = []
    
    if parent is not None and parent.size > 0:
        # Resampled parent and variants
        p_res = resample_seq(parent, n)
        inits.append(p_res)
        inits.append(resample_seq(parent[::-1], n))
        # Perturbed parent
        for scale in [0.5, 1.0, 2.0]:
            pert = p_res * scale + rng.uniform(0, 0.1, n)
            inits.append(pert)
    
    # Uniform
    inits.append(np.ones(n) / n)
    
    # Random
    inits.append(rng.uniform(0.1, 1.0, n))
    
    # Flat spectrum (random phases)
    half = n // 2 + 1
    angles = rng.uniform(0, 2 * np.pi, half)
    A = np.zeros(n, dtype=complex)
    A[:half] = np.exp(1j * angles[:half])
    for k in range(1, n - n // 2):
        A[k] = np.conj(A[n - k])
    a_flat = np.fft.ifft(A).real
    a_flat = np.maximum(a_flat, 0.0)
    if a_flat.sum() > 1e-10:
        inits.append(a_flat / a_flat.sum())
    
    # Cosine-based
    t = np.linspace(0, 1, n, endpoint=False)
    for freq in [2, 3, 5, 7, 11]:
        inits.append(1.0 + 0.3 * np.cos(2 * np.pi * freq * t + rng.uniform(0, 6.28)))
    
    # Gaussian bump
    center = rng.uniform(0.3, 0.7)
    width = rng.uniform(0.1, 0.3)
    inits.append(np.exp(-((t - center) / width) ** 2) + 0.05)
    
    # Two bumps
    c1, c2 = 0.3, 0.7
    inits.append(np.exp(-((t - c1) / 0.1) ** 2) + np.exp(-((t - c2) / 0.1) ** 2) + 0.03)
    
    # Power law
    inits.append((t + 0.05) ** rng.uniform(-1.5, 1.5))
    inits.append((1.0 - t + 0.05) ** rng.uniform(-1.5, 1.5))
    
    # Sparse smoothed
    n_spikes = max(5, n // 10)
    positions = rng.choice(n, size=n_spikes, replace=False)
    sparse = np.zeros(n)
    sparse[positions] = 1.0
    kernel_size = max(5, n // 20)
    kernel = np.ones(kernel_size) / kernel_size
    inits.append(np.convolve(sparse, kernel, mode='same') + 0.02)
    
    # Triangle wave
    inits.append(1.0 - 2.0 * np.abs(t - 0.5))
    
    # Sine wave
    inits.append(0.5 + 0.5 * np.sin(2 * np.pi * t))
    
    # Multi-sine
    inits.append(1.0 + 0.2 * np.sin(2*np.pi*t) + 0.15 * np.sin(4*np.pi*t) +
               0.1 * np.sin(6*np.pi*t) + 0.05 * np.sin(8*np.pi*t))
    
    # Normalize all
    result = []
    for x in inits:
        x = np.maximum(x, 0.001).astype(np.float64)
        s = x.sum()
        if s > 1e-10:
            result.append(x / s)
    return result

class ScoreTracker:
    def __init__(self):
        self.best = None
        self.best_score = float('inf')
        self.t_start = time.monotonic()
    
    def update(self, a):
        a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
        if a.size == 0 or a.sum() < 0.01:
            return
        score = float(evaluate_sequence(a.tolist()))
        if np.isfinite(score) and score < self.best_score:
            self.best = a.copy()
            self.best_score = score
            elapsed = time.monotonic() - self.t_start
            print(f"  * J={score:.8f} n={a.size} t={elapsed:.1f}s", flush=True)
    
    def report(self, label=""):
        if self.best is not None:
            print(f"  {label} best: J={self.best_score:.8f} n={self.best.size}", flush=True)

def simulated_annealing(a, deadline, tracker, n_restarts=30, t0=0.05):
    """Simulated annealing with LP refinement."""
    rng = np.random.default_rng(777)
    n = a.size
    a = a.copy().astype(np.float64)
    current_J = compute_J(a)
    tracker.update(a)
    best_a = a.copy()
    best_J = current_J
    temp = t0
    
    for r in range(n_restarts):
        if time.monotonic() > deadline:
            break
        a_pert = a + rng.normal(0, temp, n)
        a_pert = np.maximum(a_pert, 1e-8)
        a_pert /= a_pert.sum()
        
        # Quick LP refinement
        a_refined = lp_best_response_iteration(a_pert, max(3, 15 - r // 4), 
                                                deadline, tracker, base_tighten=0.01)
        J = compute_J(a_refined)
        delta = J - current_J
        if delta < 0 or rng.random() < np.exp(-max(delta, 0) / max(temp, 1e-10)):
            a = a_refined
            current_J = J
            if J < best_J:
                best_J = J
                best_a = a.copy()
                tracker.update(a)
        temp *= 0.93
    
    tracker.update(best_a)
    return best_a

_spar_cache = {}

def get_sparsity(n):
    if n not in _spar_cache:
        rows = []
        cols = []
        for k in range(2 * n - 1):
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            for j in range(js, je + 1):
                rows.append(k)
                cols.append(j)
        _spar_cache[n] = (np.array(rows, dtype=np.int32), np.array(cols, dtype=np.int32))
    return _spar_cache[n]

def fast_conv(a):
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]

def compute_J(a):

    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))

def lp_solve(a, M, time_limit=4.0):

    nnz = len(rows)
    vals = np.empty(nnz)
    for i in range(nnz):
        k = rows[i]
        j = cols[i]
        vals[i] = a[k - j]
    A = csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                options={'presolve': True, 'time_limit': time_limit})
    if r.success and r.x is not None:
        return np.maximum(r.x, 0.0)
    return None

def resample_seq(src, dest_n):
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:

    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)

            if time.monotonic() > t_end - 15:
                break
            t_init_end = min(t_end - 10, time.monotonic() + time_per_init)

            a = a0.copy()
            n_lp = max(5, int(25.0 / max(1, n / 256)))
            a = lp_optimize(a, n_lp, t_init_end - 25, tracker, base_tighten=0.01)
            if time.monotonic() < t_init_end - 15:
                a = mirror_descent(a, max(200, int(8000 / n)), t_init_end - 15, tracker)
            if time.monotonic() < t_init_end - 5:
                a = lse_polish(a, max(100, int(3000 / n)), t_init_end - 5, tracker)

            J = compute_J(a)
            if J < best_J_for_n:
                best_J_for_n = J
                best_for_n = a.copy()

        if best_for_n is not None:
            tracker.update(best_for_n)
            print(f"  n={n} best init: J={best_J_for_n:.8f}", flush=True)

    # Phase 3: Intensive optimization on global best
    if tracker.best is not None and time.monotonic() < deadline - 100:
        t0 = time.monotonic()
        n = tracker.best.size
        print(f"\n{'='*50} Phase 3: Intensive n={n} {'='*50}", flush=True)

        a = tracker.best.copy()
        for tighten in [0.005, 0.008, 0.012, 0.015, 0.01, 0.005]:
            if time.monotonic() > deadline - 80:
                break
            a = lp_optimize(a, 25, deadline - 80, tracker, base_tighten=tighten)

        if time.monotonic() < deadline - 60:
            a = mirror_descent(a, 15000, deadline - 60, tracker)
        if time.monotonic() < deadline - 30:
            a = lse_polish(a, 5000, deadline - 30, tracker)

        print(f"Phase 3 done: J={tracker.best_score:.8f} t={time.monotonic()-t0:.1f}s", flush=True)

    # Phase 4: Simulated annealing + final polish
    if tracker.best is not None and time.monotonic() < deadline - 50:
        t0 = time.monotonic()
        print(f"\n{'='*50} Phase 4: SA n={tracker.best.size} {'='*50}", flush=True)
        a = simulated_annealing(tracker.best, deadline - 25, tracker, n_restarts=20, t0=0.03)
        if time.monotonic() < deadline - 10:
            a = lp_optimize(a, 15, deadline - 10, tracker, base_tighten=0.005)
        if time.monotonic() < deadline - 3:
            a = lse_polish(a, 2000, deadline - 3, tracker)

        print(f"Phase 4 done: J={tracker.best_score:.8f} t={time.monotonic()-t0:.1f}s", flush=True)

    if tracker.best is None:
        tracker.best = np.ones(256)
        tracker.best_score = compute_J(tracker.best)
        tracker.update(tracker.best)

    print(f"\n{'='*50} FINAL {'='*50}", flush=True)
    print(f"J={tracker.best_score:.8f} n={tracker.best.size} time={time.monotonic()-tracker.t_start:.1f}s", flush=True)
    return [float(x) for x in tracker.best]
