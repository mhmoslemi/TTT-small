def evaluate_sequence(sequence: list[float]) -> float:
    """
    Evaluates a sequence of coefficients with enhanced security checks.
    Returns np.inf if the input is invalid.
    """
    # --- Security Checks ---

    # Verify that the input is a list
    if not isinstance(sequence, list):
        return np.inf

    # Reject empty lists
    if not sequence:
        return np.inf

    # Check each element in the list for validity
    for x in sequence:

        # any other non-integer/non-float types (like strings or complex numbers).
        if isinstance(x, bool) or not isinstance(x, (int, float)):
            return np.inf

        # Reject Not-a-Number (NaN) and infinity values.
        if np.isnan(x) or np.isinf(x):
            return np.inf

    # Convert all elements to float for consistency
    sequence = [float(x) for x in sequence]

    # Protect against negative numbers
    sequence = [max(0, x) for x in sequence]

    sequence = [min(1000.0, x) for x in sequence]

    b_sequence = np.convolve(sequence, sequence)
    max_b = max(b_sequence)
    sum_a = np.sum(sequence)

    # Protect against the case where the sum is too close to zero
    if sum_a < 0.01:
        return np.inf

def fast_conv(a):
    """Full self-convolution via FFT, O(n log n)."""
    n = len(a)
    s = 1
    while s < 2 * n - 1:
        s <<= 1
    fa = np.fft.rfft(a, s)
    return np.fft.irfft(fa * fa, s)[:2 * n - 1]


def compute_J(a):
    """Compute J(a) = 2*n*max(conv(a,a))/(sum(a))^2."""
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))

def line_search_golden(a, g, n_evals=18):
    """
    Golden section search for best t in [0,1] minimizing J((1-t)*a + t*g).
    Returns (best_a, best_J, best_t).
    """
    best_t = 0.0
    best_J = compute_J(a)
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    lo = 0.0
    hi = 1.0
    # Pre-evaluate at t=0 and t=1
    J_at_0 = best_J
    a_1 = g.copy()
    J_at_1 = compute_J(a_1)
    if J_at_1 < J_at_0:
        best_J = J_at_1
        best_t = 1.0

    for _ in range(n_evals):
        t1 = hi - gr * (hi - lo)
        t2 = lo + gr * (hi - lo)
        a1 = (1.0 - t1) * a + t1 * g
        a2 = (1.0 - t2) * a + t2 * g
        J1 = compute_J(a1)
        J2 = compute_J(a2)
        if J1 < J2:
            hi = t2
        else:
            lo = t1
        # Track best

        a_mid = (1.0 - t_mid) * a + t_mid * g
        J_mid = compute_J(a_mid)
        if J_mid < best_J:
            best_J = J_mid
            best_t = t_mid

    return (1.0 - best_t) * a + best_t * g, best_J, best_t

def smooth_conv_max_and_grad(a, beta):
    """
    Compute smooth approximation of max(conv(a,a)) and its gradient.
    Returns (smooth_max_value, gradient_array).
    """
    n = len(a)
    b = fast_conv(a)
    b_max = float(b.max())
    shifted = beta * (b - b_max)
    shifted = np.clip(shifted, -500, 500)
    exp_shifted = np.exp(shifted)
    Z = float(exp_shifted.sum())
    if Z < 1e-300:
        p = np.zeros(2 * n - 1)
        p[np.argmax(b)] = 1.0
    else:
        p = exp_shifted / Z
    g_val = (b_max + np.log(max(Z, 1e-300))) / beta

    if n < 2:
        grad = np.array([2.0 * a[0] * p.sum()])
    else:
        S = np.convolve(a, p[::-1], mode='full')
        S = S[n - 1:2 * n - 1][::-1]
        grad = 2.0 * S
    return g_val, grad


def F_and_grad(a, beta):
    """Compute F(a) = 2n*smooth_max(conv(a,a))/(sum(a))^2 and gradient."""
    n = len(a)
    s = float(a.sum())
    if s < 1e-10:
        return 1e10, np.zeros(n)
    g_val, grad_g = smooth_conv_max_and_grad(a, beta)
    F_val = 2.0 * n * g_val / (s * s)
    grad_F = 2.0 * n * (s * grad_g - 2.0 * g_val) / (s ** 3)
    return F_val, grad_F


def lbfgs_optimize(a0, betas, deadline):
    """L-BFGS-B optimization with beta annealing."""
    a = np.maximum(a0.astype(np.float64), 1e-8)
    n = len(a)

    for beta, max_iters in betas:
        if time.monotonic() > deadline:
            break

        def func(x):
            x2 = np.maximum(x, 0.0)
            f, _ = F_and_grad(x2, beta)
            if not np.isfinite(f):
                return 1e10
            return f

        def jac(x):
            x2 = np.maximum(x, 0.0)
            _, g = F_and_grad(x2, beta)
            g = np.nan_to_num(g, nan=0.0, posinf=0.0, neginf=0.0)
            g[x2 < 1e-12] = 0.0
            return g

        bounds = [(0.0, None)] * n
        try:
            res = minimize(func, a.copy(), jac=jac, method='L-BFGS-B',
                           bounds=bounds,
                           options={'maxiter': max_iters, 'max_fun': max_iters * 20,
                                    'gtol': 1e-8, 'ftol': 1e-14})
            if res.fun < 1e10:
                new_a = np.maximum(res.x, 1e-10)
                if new_a.sum() > 1e-6:
                    a = new_a
        except Exception:
            pass

    return a


def resample_seq(src, dest_n):
    """Resample 1D sequence to new length via linear interpolation."""
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    result = np.interp(t_new, t_old, src)
    return np.maximum(result, 0.0)


def generate_initializations(n, rng, parent=None):

    inits.append(np.abs(np.sin(np.pi * t_arr)) + 0.15)

    # Triangular wave
    inits.append(1.0 - 0.3 * np.abs(2.0 * (t_arr % 0.5) - 0.5))

class ScoreTracker:

    def __init__(self):
        self.best = None
        self.best_score = float('inf')
        self.t_start = time.monotonic()

    def update(self, a):
        a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
        if a.size == 0 or a.sum() < 0.01:
            return
        score = float(evaluate_sequence(a.tolist()))
        if np.isfinite(score) and score < self.best_score:
            self.best = a.copy()
            self.best_score = score
            elapsed = time.monotonic() - self.t_start
            print(f"  * NEW BEST: J={score:.8f} n={a.size} t={elapsed:.1f}s", flush=True)

    def report(self, label=""):
        if self.best is not None:
            print(f"  {label} best: J={self.best_score:.8f} n={self.best.size}", flush=True)
