def compute_conv(a):

    # Golden section search for x >= 0
    # Bracket: try around current value
    golden_ratio = (np.sqrt(5) - 1) / 2  # ~0.618
    
    # Initial bracket
    x0 = max(0.0, old_val * 0.5)
    x3 = old_val * 1.5 + 0.1
    x1 = x0 + golden_ratio * (x3 - x0)
    x2 = x3 - golden_ratio * (x3 - x0)
    
    f1 = obj_of_x(x1)
    f2 = obj_of_x(x2)
    
    # Expand bracket if needed
    iters = 0
    while f1 > obj_of_x(x0) and iters < 5:
        x3 = x0
        x0 = x0 - (x3 - x0)
        x1 = x0 + golden_ratio * (x3 - x0)
        x2 = x3 - golden_ratio * (x3 - x0)
        f1 = obj_of_x(x1)
        f2 = obj_of_x(x2)
        iters += 1
    while f2 > obj_of_x(x3) and iters < 10:
        x0 = x3
        x3 = x3 + (x3 - x0) * 2
        x1 = x0 + golden_ratio * (x3 - x0)
        x2 = x3 - golden_ratio * (x3 - x0)
        f1 = obj_of_x(x1)
        f2 = obj_of_x(x2)
        iters += 1
    
    # Golden section search
    for _ in range(60):
        if x3 - x0 < 1e-12 * max(1.0, abs(old_val)):
            break
        if f1 < f2:
            x3 = x2
            x2 = x1
            f2 = f1
            x1 = x0 + golden_ratio * (x3 - x0)
            f1 = obj_of_x(x1)
        else:
            x0 = x1
            x1 = x2
            f1 = f2
            x2 = x3 - golden_ratio * (x3 - x0)
            f2 = obj_of_x(x2)
    
    x_opt = (x0 + x3) / 2
    x_opt = max(0.0, x_opt)
    f_opt = obj_of_x(x_opt)
    
    # Also try x = 0 and x = old_val
    f_zero = obj_of_x(0.0)
    f_old = obj_of_x(old_val)
    
    if f_zero < f_opt:
        x_opt = 0.0
        f_opt = f_zero
    if f_old < f_opt:
        x_opt = old_val
        f_opt = f_old
    
    a[i] = x_opt
    return x_opt, f_opt


def cd_optimize(a, time_budget=30.0, verbose=False):
    """Coordinate descent optimization with time budget."""
    n = len(a)
    a = a.copy().astype(np.float64)
    best_obj = compute_obj(a)
    best_a = a.copy()
    start = time.monotonic()
    rng = np.random.default_rng(42)
    
    sweep_count = 0
    stall = 0
    best_stall_obj = best_obj
    
    while time.monotonic() - start < time_budget - 2:
        # Random coordinate order
        order = rng.permutation(n)
        sweep_start = time.monotonic()
        
        for i in order:
            if time.monotonic() - sweep_start > 2.0:
                break
            a[i], _ = _cd_step_1d(a, i)
        
        sweep_count += 1
        obj = compute_obj(a)
        
        if obj < best_obj - 1e-12:
            best_obj = obj
            best_a = a.copy()
            stall = 0
        else:
            stall += 1
            if stall > 20:
                # Try different random order or perturb
                a = best_a.copy()
                pert = 1.0 + 0.01 * rng.normal(0, 1, n)
                a = a * pert
                a = np.maximum(a, 0)
                s = np.sum(a)
                if s > 1e-10:
                    a = a / s
                stall = 0
        
        if verbose and sweep_count % 100 == 0:
            elapsed = time.monotonic() - start
            print(f"    CD sweep {sweep_count}: obj={best_obj:.8f} ({elapsed:.1f}s)")
    
    return best_a, best_obj


def alternating_lp(a, max_iters=30, time_budget=20.0):
    """Alternating LP with line search."""
    n = len(a)
    if n > 800:
        return a, compute_obj(a)
    
    a = a.copy().astype(np.float64)
    s = np.sum(a)
    if s < 1e-10:
        return a, np.inf
    a = a / s
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    stall = 0

    for it in range(max_iters):
        if time.monotonic() - start > time_budget - 2:
            break

        conv = compute_conv(a)
        M = float(np.max(conv))

        g, sum_g = solve_lp_maxsum(a, M, time_limit=3)
        if g is None or sum_g < 1e-10:
            break

        g_norm = g / sum_g

        best_t = 0.0
        best_obj_ls = best_obj
        for t in [0.1, 0.2, 0.3, 0.5, 0.7, 1.0]:
            a_new = (1.0 - t) * a + t * g_norm
            a_new = np.maximum(a_new, 0)
            s_new = np.sum(a_new)
            if s_new < 1e-10:
                continue
            a_new = a_new / s_new
            obj_new = compute_obj(a_new)
            if obj_new < best_obj_ls:
                best_obj_ls = obj_new
                best_t = t

        if best_t > 0 and best_obj_ls < best_obj - 1e-12:
            a_new = (1.0 - best_t) * a + best_t * g_norm
            a_new = np.maximum(a_new, 0)
            s_new = np.sum(a_new)
            if s_new > 1e-10:
                a_new = a_new / s_new
            best_obj = best_obj_ls
            best_a = a_new.copy()
            a = a_new
            stall = 0
        else:
            stall += 1
            if stall > 15:
                break

    return best_a, best_obj


def smooth_gradient_optimize(a, time_budget=15.0, tau_init=0.01, tau_final=0.001):
    """Optimize using smooth (logsumexp) approximation with gradient descent."""
    n = len(a)
    a = a.copy().astype(np.float64)
    s = np.sum(a)
    if s < 1e-10:
        a = np.ones(n)
        s = n
    a = a / s
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    
    # Work in log-space for non-negativity
    log_a = np.log(np.maximum(a, 1e-15))
    
    step = 0.01
    total_steps = max(100, int(time_budget * 50))
    stall = 0
    
    for it in range(total_steps):
        if time.monotonic() - start > time_budget - 1:
            break
        
        # Adaptive tau
        tau = tau_init * (tau_final / tau_init) ** (it / total_steps)
        
        a_curr = np.exp(log_a)
        a_curr = a_curr / np.sum(a_curr)
        
        conv = compute_conv(a_curr)
        max_conv = np.max(conv)
        
        # Gradient of logsumexp(conv/tau) w.r.t. a
        shifted = (conv - max_conv) / tau
        weights = np.exp(shifted)
        w_sum = np.sum(weights)
        if w_sum < 1e-300:
            break
        weights = weights / w_sum
        
        # d(logsumexp)/d(a_i) = (1/tau) * (weights @ a) ... 
        # Actually d(conv_k)/d(a_i) = a_{k-i} + a_{k-i} (symmetric) except when k=2i
        # More precisely: d(conv_k)/d(a_i) = a_{k-i} (from j=i term) + a_{k-i} (from j=k-i term)
        # if k-i != i, else a_{k-i} = a_i (just one term, but it's a_i^2 so derivative is 2*a_i)
        # Actually d/d(a_i) [sum_j a_j a_{k-j}] = a_{k-i} + a_{k-i} if k != 2i, = 2*a_i if k=2i
        # Wait no. d/d(a_i) [a_j * a_{k-j}] = delta_{j,i} * a_{k-j} + a_j * delta_{k-j,i}
        # = delta_{j,i} * a_{k-i} + delta_{j,k-i} * a_j
        # Summing over j: a_{k-i} + a_{k-i} if k-i != i, or a_i + a_i = 2*a_i if k-i = i
        # But this only holds if k-i is in range.
        
        # So grad_i = (1/tau) * sum_k weights_k * d(conv_k)/d(a_i)
        # = (1/tau) * sum_k weights_k * (a_{k-i} + a_{k-i} [k!=2i] or 2*a_i [k=2i])
        # = (1/tau) * sum_k weights_k * 2*a_{k-i} (with appropriate range checks)
        
        # This is essentially (1/tau) * 2 * (correlate(weights, a)) at position i
        # But with the correct "full" correlation
        
        c = np.correlate(weights, a_curr, mode='full')
        grad = 2.0 * c[n - 1:n - 1 + n] / tau
        
        # Gradient of the full objective: 2n * LSE(conv/tau) / S^2
        # d/d(a_i) = 2n * [grad_i * S^2 - LSE(conv/tau) * 2*S] / S^4
        # = 2n * [grad_i * S - 2*LSE*S / S^2 * S] ... let me just use the chain rule
        # f(a) = 2n * g(a) / S(a)^2 where g(a) = LSE(conv(a)/tau)
        # df/d(a_i) = 2n * [dg/d(a_i) * S^2 - g * 2*S * 1] / S^4
        # = 2n * [dg/d(a_i) * S - 2*g] / S^3
        
        g_val = max_conv + tau * np.log(w_sum)  # LSE value (shifted)
        S = np.sum(a_curr)
        
        full_grad = 2.0 * n * (grad * S - 2.0 * g_val) / (S ** 3)
        
        # Center gradient to preserve normalization
        full_grad_centered = full_grad - np.mean(full_grad)
        
        # Gradient step in log-space
        log_a = log_a - step * full_grad_centered * a_curr
        log_a = np.clip(log_a, -30, 30)
        
        a_curr = np.exp(log_a)
        a_curr = a_curr / np.sum(a_curr)
        
        obj = compute_obj(a_curr)
        if obj < best_obj - 1e-12:
            best_obj = obj
            best_a = a_curr.copy()
            stall = 0
        else:
            stall += 1
            if stall > 50:
                step *= 0.7
                if step < 0.001:
                    break
                stall = 0
        
        # Decay step
        step *= 0.999
    
    return best_a, best_obj


def multiplicative_subgradient(a, time_budget=10.0, tau=20.0, step=0.05):
    """Multiplicative subgradient method."""
    n = len(a)
    a = a.copy().astype(np.float64)
    s = np.sum(a)
    if s < 1e-10:
        return a, np.inf
    a = a / s
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    stall = 0
    
    for it in range(int(time_budget * 40)):
        if time.monotonic() - start > time_budget - 1:
            break
        
        conv = compute_conv(a)
        max_conv = np.max(conv)
        shifted = tau * (conv - max_conv)
        shifted = np.clip(shifted, -500, 500)
        weights = np.exp(shifted)
        w_sum = np.sum(weights)
        if w_sum < 1e-300:
            break
        weights /= w_sum
        
        c = np.correlate(weights, a, mode='full')
        grad = 2.0 * c[n - 1:n - 1 + n]
        
        a_new = a * np.exp(-step * grad)
        a_new = np.maximum(a_new, 1e-15)
        s_new = np.sum(a_new)
        if s_new > 1e-10:
            a_new = a_new / s_new
        a = a_new
        
        obj = compute_obj(a)
        if obj < best_obj - 1e-12:
            best_obj = obj
            best_a = a.copy()
            stall = 0
        else:
            stall += 1
            if stall > 60:
                step *= 0.8
                if step < 0.005:
                    break
                # Perturb
                rng = np.random.default_rng(it * 7 + 3)
                a = a * np.exp(rng.normal(0, 0.05, n))
                a = np.maximum(a, 1e-10)
                a = a / np.sum(a)
                stall = 0
        
        step *= 0.999
        tau *= 1.003
    
    return best_a, best_obj


def random_init(n, seed):
    """Generate diverse random initializations."""
    rng = np.random.default_rng(seed)
    k = np.arange(n)
    choice = rng.integers(0, 8)
    
    if choice == 0:
        f = rng.uniform(0.5, 4.0)
        phase = rng.uniform(0, 2 * np.pi)
        a = 1.0 + 0.8 * np.cos(2 * np.pi * f * k / n + phase)
    elif choice == 1:
        f1 = rng.uniform(0.3, 2.0)
        f2 = rng.uniform(2.0, 6.0)
        a = 1.0 + 0.4 * np.cos(2 * np.pi * f1 * k / n) + 0.3 * np.cos(2 * np.pi * f2 * k / n)
    elif choice == 2:
        center = n * rng.uniform(0.3, 0.7)
        alpha = rng.uniform(0.5, 3.0)
        a = 1.0 + 0.5 * np.exp(-alpha * np.abs(k - center) / (n * 0.3))
    elif choice == 3:
        a = rng.random(n)
        win = max(3, n // 20)
        kernel = np.ones(win) / win
        a = np.convolve(a, kernel, mode='same')
    elif choice == 4:
        a = np.ones(n)
        for _ in range(rng.integers(1, 4)):
            f = rng.uniform(0.5, 5.0)
            amp = rng.uniform(0.1, 0.4)
            a += amp * np.cos(2 * np.pi * f * k / n + rng.uniform(0, 2 * np.pi))
    elif choice == 5:
        a = np.ones(n)
        for scale in range(3):
            f = rng.uniform(0.3 * (2 ** scale), 1.5 * (2 ** scale))
            amp = 0.3 / (scale + 1)
            a += amp * np.cos(2 * np.pi * f * k / n + rng.uniform(0, 2 * np.pi))
    elif choice == 6:
        # Multi-periodic
        n_periods = rng.integers(2, 5)
        a = np.ones(n)
        for p in range(n_periods):
            f = rng.uniform(0.5, 3.0) * (p + 1)
            a += 0.3 / (p + 1) * np.sin(2 * np.pi * f * k / n + rng.uniform(0, 2 * np.pi))
    else:
        # Smooth random
        a = rng.random(max(5, n // 10))
        a = np.interp(k, np.arange(len(a)), a)
    
    a = np.maximum(a, 0)
    s = np.sum(a)
    if s < 1e-10:
        a = np.ones(n)
    return a / np.sum(a)


def resample_sequence(a, new_n):
    """Resample a sequence to a new length using interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy() / np.sum(a)
    old_k = np.arange(n)
    new_k = np.linspace(0, n - 1, new_n)
    a_new = np.interp(new_k, old_k, a)
    a_new = np.maximum(a_new, 0)
    s = np.sum(a_new)
    if s > 1e-10:
        a_new = a_new / s
    return a_new


def full_pipeline(a, time_budget=60.0):
    """Run full optimization pipeline on a single sequence."""
    n = len(a)
    a = a.copy().astype(np.float64)
    s = np.sum(a)
    if s < 1e-10:
        a = np.ones(n)
    a = a / s
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    
    # Phase 1: Coordinate descent (fast)
    remaining = time_budget - (time.monotonic() - start)
    if remaining > 5:
        cd_time = min(remaining * 0.4, 25.0)
        a_new, obj = cd_optimize(best_a, time_budget=cd_time)
        if obj < best_obj:
            best_obj = obj
            best_a = a_new.copy()
    
    # Phase 2: LP-based (if n small enough)
    remaining = time_budget - (time.monotonic() - start)
    if remaining > 10 and n <= 500:
        lp_time = min(remaining * 0.25, 15.0)
        a_new, obj = alternating_lp(best_a, max_iters=20, time_budget=lp_time)
        if obj < best_obj:
            best_obj = obj
            best_a = a_new.copy()
    
    # Phase 3: More coordinate descent
    remaining = time_budget - (time.monotonic() - start)
    if remaining > 5:
        cd_time = min(remaining * 0.3, 20.0)
        a_new, obj = cd_optimize(best_a, time_budget=cd_time)
        if obj < best_obj:
            best_obj = obj
            best_a = a_new.copy()
    
    # Phase 4: Smooth gradient polish
    remaining = time_budget - (time.monotonic() - start)
    if remaining > 5:
        sg_time = min(remaining * 0.2, 10.0)
        a_new, obj = smooth_gradient_optimize(best_a, time_budget=sg_time)
        if obj < best_obj:
            best_obj = obj
            best_a = a_new.copy()
    
    # Phase 5: Final coordinate descent
    remaining = time_budget - (time.monotonic() - start)
    if remaining > 3:
        cd_time = min(remaining * 0.4, 10.0)
        a_new, obj = cd_optimize(best_a, time_budget=cd_time)
        if obj < best_obj:
            best_obj = obj
            best_a = a_new.copy()
    
    return best_a, best_obj


def local_search_around(a, time_budget=30.0):
    """Local search with perturbations and re-optimization."""
    n = len(a)
    a = a.copy().astype(np.float64)
    s = np.sum(a)
    if s < 1e-10:
        return a, np.inf
    a = a / s
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    rng = np.random.default_rng(12345)
    
    attempt = 0
    while time.monotonic() - start < time_budget - 5:
        attempt += 1
        remaining = time_budget - (time.monotonic() - start)
        if remaining < 8:
            break
        
        # Various perturbation strategies
        strategy = attempt % 4
        if strategy == 0:
            # Multiplicative noise
            scale = 0.02 + 0.04 * (attempt % 3) / 3.0
            a_pert = best_a * np.exp(rng.normal(0, scale, n))
        elif strategy == 1:
            # Additive noise
            a_pert = best_a + rng.normal(0, 0.01, n)
        elif strategy == 2:
            # Swap two random coordinates
            i, j = rng.choice(n, 2, replace=False)
            a_pert = best_a.copy()
            a_pert[i], a_pert[j] = a_pert[j], a_pert[i]
        else:
            # Partial re-randomization
            n_rand = max(5, n // 10)
            mask = rng.random(n) < n_rand / n
            a_pert = best_a * (~mask) + random_init(n, attempt * 999)[mask]
        
        a_pert = np.maximum(a_pert, 0)
        s_pert = np.sum(a_pert)
        if s_pert < 1e-10:
            continue
        a_pert = a_pert / s_pert
        
        # Quick optimization
        sub_time = min(8.0, remaining * 0.3)
        a_opt, obj = full_pipeline(a_pert, time_budget=sub_time)
        
        if obj < best_obj:
            best_obj = obj
            best_a = a_opt.copy()
    
    return best_a, best_obj


def propose_candidate(seed=42, budget_s=1500):
    """
    Main search: Uses coordinate descent with exact 1D solves as the primary
    optimization, combined with LP-based alternating and smooth gradient methods.
    Focuses on n around 300-400 with intensive local search.
    """
    start_time = time.monotonic()
    deadline = start_time + float(budget_s) - 20.0
    
    best_seq = None
    best_score = np.inf
    
    # Try to load parent
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64)
        parent = np.clip(parent, 0, 1000.0)
        if parent.size > 0 and np.isfinite(parent).all():
            obj = compute_obj(parent)
            parent_obj_eval = evaluate_sequence([float(x) for x in parent])
            if parent_obj_eval < np.inf:
                obj = parent_obj_eval
            if obj < best_score:
                best_score = obj
                best_seq = parent.copy()
                print(f"Loaded parent: n={parent.size}, score={best_score:.6f}")
    except Exception:
        pass
    
    if best_seq is None:
        # Fallback initialization
        n_init = 330
        best_seq = random_init(n_init, seed)
        best_score = compute_obj(best_seq)
        print(f"Generated initial: n={n_init}, score={best_score:.6f}")
    
    n_parent = len(best_seq)
    print(f"Time budget: {budget_s}s, deadline in {deadline - start_time:.0f}s")
    
    # === Phase 1: Intensive optimization of parent ===
    print(f"\n=== Phase 1: Optimize parent (n={n_parent}) ===")
    remaining = deadline - time.monotonic()
    p1_budget = min(remaining * 0.35, 350.0)
    a_opt, obj = full_pipeline(best_seq, time_budget=p1_budget)
    if obj < best_score:
        best_score = obj
        best_seq = a_opt.copy()
        print(f"  After pipeline: {best_score:.8f}")
    
    # Additional CD sweeps
    remaining = deadline - time.monotonic()
    if remaining > 20:
        cd_budget = min(remaining * 0.15, 80.0)
        a_opt, obj = cd_optimize(best_seq, time_budget=cd_budget, verbose=True)
        if obj < best_score:
            best_score = obj
            best_seq = a_opt.copy()
            print(f"  After extra CD: {best_score:.8f}")
    
    # Local search
    remaining = deadline - time.monotonic()
    if remaining > 20:
        ls_budget = min(remaining * 0.15, 80.0)
        a_opt, obj = local_search_around(best_seq, time_budget=ls_budget)
        if obj < best_score:
            best_score = obj
            best_seq = a_opt.copy()
            print(f"  After local search: {best_score:.8f}")
    
    # === Phase 2: Try nearby n values ===
    print(f"\n=== Phase 2: Explore nearby n ===")
    n_values = sorted(set([
        n_parent, n_parent-10, n_parent+10, n_parent-20, n_parent+20,
        n_parent-30, n_parent+30, n_parent-50, n_parent+50,
        n_parent-70, n_parent+70, n_parent+100
    ]))
    n_values = [nv for nv in n_values if 150 <= nv <= 800]
    
    remaining = deadline - time.monotonic()
    if remaining > 30:
        per_n = remaining / max(len(n_values), 1) * 0.7
        for n_val in n_values:
            if time.monotonic() > deadline - 25:
                break
            if n_val == n_parent:
                continue  # Already optimized
            remaining = deadline - time.monotonic()
            if remaining < 20:
                break
            alloc = min(per_n, remaining * 0.2, 60.0)
            alloc = max(alloc, 12.0)
            
            # Start from resampled parent or random
            a_init = resample_sequence(best_seq, n_val)
            a_opt, obj = full_pipeline(a_init, time_budget=alloc)
            
            if obj < best_score:
                best_score = obj
                best_seq = a_opt.copy()
                print(f"  ** n={n_val}: {best_score:.8f} (improved!)")
            else:
                print(f"  n={n_val}: {obj:.8f} (no improvement, best={best_score:.8f})")
    
    # === Phase 3: Intensive local search on best ===
    print(f"\n=== Phase 3: Intensive local search ===")
    remaining = deadline - time.monotonic()
    if remaining > 20:
        ls_budget = min(remaining * 0.3, remaining - 10)
        a_opt, obj = local_search_around(best_seq, time_budget=ls_budget)
        if obj < best_score:
            best_score = obj
            best_seq = a_opt.copy()
            print(f"  After intensive LS: {best_score:.8f}")
        
        # More CD
        remaining = deadline - time.monotonic()
        if remaining > 15:
            cd_budget = min(remaining * 0.25, remaining - 10)
            a_opt, obj = cd_optimize(best_seq, time_budget=cd_budget, verbose=True)
            if obj < best_score:
                best_score = obj
                best_seq = a_opt.copy()
                print(f"  After final CD: {best_score:.8f}")
    
    # === Phase 4: One more round on best structure ===
    remaining = deadline - time.monotonic()
    if remaining > 30:
        print(f"\n=== Phase 4: Final optimization round ===")
        full_budget = min(remaining * 0.4, remaining - 10)
        a_opt, obj = full_pipeline(best_seq, time_budget=full_budget)
        if obj < best_score:
            best_score = obj
            best_seq = a_opt.copy()
            print(f"  After final pipeline: {best_score:.8f}")
        
        remaining = deadline - time.monotonic()
        if remaining > 10:
            cd_budget = min(remaining * 0.3, remaining - 5)
            a_opt, obj = cd_optimize(best_seq, time_budget=cd_budget)
            if obj < best_score:
                best_score = obj
                best_seq = a_opt.copy()
                print(f"  Final CD: {best_score:.8f}")
    
    # Final validation
    if best_seq is None:
        best_seq = np.ones(300) / 300
        best_score = compute_obj(best_seq)
    
    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    val = evaluate_sequence(result)
    if val < best_score:
        best_score = val
    
    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_score:.8f}, time={elapsed:.1f}s ===")
    
    return result
