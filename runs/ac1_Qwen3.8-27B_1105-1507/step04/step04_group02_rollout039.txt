```python
def solve_blend_lp(a, t, time_limit=10):
    """
    Solve LP to find g that minimizes max_k conv((1-t)a+tg, (1-t)a+tg)[k]
    (ignoring the t^2*conv(g,g) term as approximation).
    
    min z
    s.t. (1-t)^2 * conv(a,a)[k] + 2t(1-t) * conv(a,g)[k] <= z  for all k
         sum(g) = 1
         g >= 0
    """
    n = len(a)
    C_a = compute_conv(a)
    A_toeplitz = build_toeplitz_sparse(a)  # A * g = conv(a, g)
    
    # Variables: g[0..n-1], z
    # Objective: minimize z
    c = np.zeros(n + 1)
    c[-1] = 1.0  # minimize z
    
    # Constraints: (1-t)^2 * C_a[k] + 2t(1-t) * (A@g)[k] - z <= 0 for all k
    # i.e., (A@g)[k] - z/(2t(1-t)) <= -C_a[k] * (1-t)^2 / (2t(1-t))
    
    # Better: (1-t)^2 * C_a[k] + 2t(1-t) * (A@g)[k] - z <= 0
    # Rearranged: 2t(1-t) * (A@g)[k] - z <= -(1-t)^2 * C_a[k]
    
    n_conv = 2 * n - 1
    # Constraint matrix: for each k, the row is [2t(1-t) * A[k,:], -1]
    A_ub = np.zeros((n_conv, n + 1))
    A_ub[:, :n] = 2 * t * (1 - t) * A_toeplitz.toarray()  # This is expensive for large n
    A_ub[:, n] = -1.0
    b_ub = -((1 - t) ** 2) * C_a
    
    # Equality: sum(g) = 1
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])
    
    bounds = [(0, None)] * n + [(None, None)]  # g >= 0, z free
    
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq, 
                  bounds=bounds, method='highs', options={'time_limit': time_limit})
    if res.status == 0:
        g = np.maximum(res.x[:n], 0)
        S = np.sum(g)
        if S > 1e-10:
            g = g / S
        return g, res.x[-1]  # g and the optimal z
    return None, np.inf
```

Given `a` with `sum(a) = 1`, current max `M = max_k conv(a,a)[k]`.

We want to find `g` (sum=1, g>=0) and `t` in [0,1] to minimize:
`max_k [(1-t)^2 * conv(a,a)[k] + 2t(1-t) * conv(a,g)[k] + t^2 * conv(g,g)[k]]`

For the LP approximation (dropping the `t^2 * conv(g,g)` term):
`min z`
`s.t. (1-t)^2 * C_a[k] + 2t(1-t) * (A_k @ g) <= z  for k=0,...,2n-2`
`     sum(g) = 1`
`     g >= 0`

where `A_k @ g = conv(a,g)[k]`.

The constraint can be rewritten as:
`2t(1-t) * (A_k @ g) - z <= -(1-t)^2 * C_a[k]`

In matrix form:
`[2t(1-t) * A_matrix, -1] @ [g; z] <= -(1-t)^2 * C_a`

where `A_matrix` is the (2n-1) x n Toeplitz matrix such that `A_matrix @ g = conv(a,g)`.

Plus: `sum(g) = 1` (equality), `g >= 0`, `z` free.

This is a standard LP with (2n-1) + 1 equality constraints and n+1 variables. Very tractable.

The optimal `z*` gives us an upper bound on the new maximum. We can then:
- Set `a_new = (1-t)*a + t*g`
- Check the actual objective
- If it improved, accept; otherwise, try smaller t or different g.

We can also try multiple values of t (e.g., 0.1, 0.2, 0.3, 0.5) and pick the best.

Now, for the `t^2 * conv(g,g)` term that we dropped: at the optimal g, this term might be significant. We can bound it: since `g >= 0` and `sum(g) = 1`, we have `conv(g,g)[k] <= (sum g)^2 / (2n-1) * (2n-1) = 1`... actually that's not a useful bound.

A better bound: `conv(g,g)[k] = sum_j g[j]*g[k-j] <= max_j g[j] * sum_j g[j] = max_j g[j]`. So if the optimal g is spread out (small max), this term is small.

In practice, the LP will tend to produce a spread-out g (since we're maximizing the sum subject to constraints), so the dropped term should be manageable.

```python
"""

import numpy as np

def compute_conv(a):
    """Compute linear convolution a*a via FFT."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    conv = np.fft.irfft(fa * fa, size)
    return conv[:2 * n - 1]

def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    S = float(np.sum(a))
    if S < 1e-10:
        return np.inf
    conv = compute_conv(a)
    M = float(np.max(conv))
    return 2.0 * n * M / (S * S)

def build_toeplitz_array(a):
    """Build dense Toeplitz matrix for conv(a, g): A @ g = conv(a, g)."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        start = j
        end = min(j + n, 2 * n - 1)
        A[start:end, j] = a[:end - start]
    return A

def solve_blend_lp(a, t, time_limit=10):
    """
    Solve LP to find g minimizing predicted max after blending (1-t)*a + t*g.
    Ignores t^2*conv(g,g) term.
    
    min z
    s.t. (1-t)^2 * conv(a,a)[k] + 2t(1-t) * conv(a,g)[k] <= z  for all k
         sum(g) = 1
         g >= 0
    """
    n = len(a)
    C_a = compute_conv(a)
    A_mat = build_toeplitz_array(a)
    
    # Constraint: 2t(1-t) * (A_mat @ g) - z <= -(1-t)^2 * C_a
    n_conv = 2 * n - 1
    coeff_g = 2.0 * t * (1.0 - t)
    rhs = -((1.0 - t) ** 2) * C_a
    
    # Full constraint matrix: [coeff_g * A_mat | -1] @ [g; z] <= rhs
    A_ub = np.hstack([coeff_g * A_mat, -np.ones((n_conv, 1))])
    
    # Objective: minimize z (last variable)
    c = np.zeros(n + 1)
    c[-1] = 1.0
    
    # Equality: sum(g) = 1
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])
    
    bounds = [(0, None)] * n + [(None, None)]
    
    try:
        res = linprog(c, A_ub=A_ub, b_ub=rhs, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs',
                      options={'time_limit': time_limit, 'presolve': True})
        if res.status == 0:
            g = np.maximum(res.x[:n], 0)
            S = np.sum(g)
            if S > 1e-10:
                g = g / S
            return g, float(res.x[-1])
    except Exception:
        pass
    return None, np.inf

def solve_sum_lp(a, M, time_limit=10):
    """Original LP: max sum(g) s.t. conv(a,g) <= M, g >= 0."""
    n = len(a)
    A_mat = build_toeplitz_array(a)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    try:
        res = linprog(c, A_ub=A_mat, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'time_limit': time_limit, 'presolve': True})
        if res.status == 0:
            g = np.maximum(res.x, 0)
            S = np.sum(g)
            if S > 1e-10:
                g = g / S
            return g
    except Exception:
        pass
    return None

def subgradient_step(a, topk=10):
    """Compute subgradient of max(conv(a,a)) and return step direction."""
    n = len(a)
    conv = compute_conv(a)
    M = float(np.max(conv))
    
    # Find top-K active positions
    if topk >= len(conv):
        active = np.arange(len(conv))
    else:
        active = np.argsort(conv)[-topk:]
    
    # Gradient
    grad = np.zeros(n)
    for k in active:
        for i in range(n):
            j = k - i
            if 0 <= j < n:
                grad[i] += 2.0 * a[j]
    grad /= len(active)
    
    return grad, M, conv

def momentum_subgrad(a, max_iters=2000, time_budget=30.0, topk=8, momentum=0.85):
    """Momentum subgradient descent with adaptive step size."""
    n = len(a)
    a = a.copy().astype(np.float64)
    S = np.sum(a)

        return a
    a = a / S

            break
        
        grad, M, conv = subgradient_step(a, topk)

def log_param_subgrad(a, max_iters=1000, time_budget=20.0):

def cosine_init(n, n_modes=8, seed=0):
    """Generate a smooth cosine-series initialization."""
    rng = np.random.default_rng(seed)
    k = np.arange(n)
    a = np.ones(n)
    for m in range(1, n_modes + 1):
        c = rng.uniform(-0.5, 0.5) / m
        phase = rng.uniform(0, 2 * np.pi)
        a += c * np.cos(2 * np.pi * m * k / n + phase)
    a = np.maximum(a, 0)
    S = np.sum(a)
    if S < 1e-10:
        return np.ones(n) / n
    return a / S

def symmetric_init(n, seed=0):
    """Generate a symmetric (palindromic) initialization."""
    half = (n + 1) // 2
    k = np.arange(half)
    a_half = np.ones(half)
    rng = np.random.default_rng(seed)
    for m in range(1, 6):
        c = rng.uniform(-0.4, 0.4) / m
        a_half += c * np.cos(2 * np.pi * m * k / half)
    a_half = np.maximum(a_half, 0)
    # Make full symmetric sequence
    a = np.zeros(n)
    a[:half] = a_half
    a[half:] = a_half[:n-half][::-1]
    S = np.sum(a)
    if S < 1e-10:
        return np.ones(n) / n
    return a / S

def raised_cosine_init(n, alpha=1.5):
    """Raised cosine shape."""

    x = 2.0 * k / (n - 1) - 1.0
    a = 0.5 * (1.0 + np.cos(np.pi * x)) ** alpha
    a = np.maximum(a, 1e-10)
    S = np.sum(a)
    return a / S
```

```python
def compute_conv(a):
    """Compute linear convolution a*a via FFT."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    conv = np.fft.irfft(fa * fa, size)
    return conv[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    S = float(np.sum(a))
    if S < 1e-10:
        return np.inf
    conv = compute_conv(a)
    M = float(np.max(conv))
    return 2.0 * n * M / (S * S)


def build_toeplitz_array(a):
    """Build dense Toeplitz matrix A such that A @ g = conv(a, g)."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        start = j
        end = min(j + n, 2 * n - 1)
        A[start:end, j] = a[:end - start]
    return A


def solve_blend_lp(a, t, time_limit=10):
    """
    Solve LP: min z s.t. (1-t)^2*conv(a,a)[k] + 2t(1-t)*conv(a,g)[k] <= z, sum(g)=1, g>=0.
    Returns (g, z) or (None, inf).
    """
    n = len(a)
    C_a = compute_conv(a)
    A_mat = build_toeplitz_array(a)

    n_conv = 2 * n - 1
    coeff_g = 2.0 * t * (1.0 - t)
    rhs = -((1.0 - t) ** 2) * C_a

    A_ub = np.hstack([coeff_g * A_mat, -np.ones((n_conv, 1))])

    c = np.zeros(n + 1)
    c[-1] = 1.0

    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])

    bounds = [(0, None)] * n + [(None, None)]

    try:
        res = linprog(c, A_ub=A_ub, b_ub=rhs, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs',
                      options={'time_limit': time_limit, 'presolve': True})
        if res.status == 0:
            g = np.maximum(res.x[:n], 0.0)
            S = np.sum(g)
            if S > 1e-10:
                g = g / S
            return g, float(res.x[-1])
    except Exception:
        pass
    return None, np.inf

    n = len(a)
    conv = compute_conv(a)
    M = float(np.max(conv))

    if topk >= len(conv):

    else:
        active = np.argsort(conv)[-topk:]

        hi = min(k, n - 1)
        for i in range(lo, hi + 1):
            grad[i] += 2.0 * a[k - i]

    return grad, M, conv

def cosine_init(n, n_modes=8, seed=0):

    rng = np.random.default_rng(seed)
    k = np.arange(n)
    a = np.ones(n)
    for m in range(1, n_modes + 1):
        c = rng.uniform(-0.5, 0.5) / m

def symmetric_init(n, seed=0):
    """Palindromic (symmetric) initialization."""
    half = (n + 1) // 2
    k = np.arange(half)
    a_half = np.ones(half)
    rng = np.random.default_rng(seed)
    for m in range(1, 6):

    a_half = np.maximum(a_half, 0.0)
    a = np.zeros(n)
    a[:half] = a_half
    a[half:] = a_half[:n - half][::-1]
    S = np.sum(a)
    if S < 1e-10:
        return np.ones(n) / n
    return a / S

                best_obj = obj_new
                best_a = a_new.copy()

        # Secondary: sum-LP
        remaining = time_budget - (time.monotonic() - start)
        if remaining > 12.0:
            alloc = min(remaining * 0.2, 20.0)
            a_new = alternating_sum_lp(best_a, max_iters=20, time_budget=alloc)
            obj_new = compute_obj(a_new)
            if obj_new < best_obj:
                best_obj = obj_new
                best_a = a_new.copy()

        # Tertiary: subgradient
        remaining = time_budget - (time.monotonic() - start)
        if remaining > 8.0:
            alloc = min(remaining * 0.15, 12.0)
            a_new = momentum_subgrad(best_a, max_iters=800, time_budget=alloc, topk=8)
            obj_new = compute_obj(a_new)
            if obj_new < best_obj:
                best_obj = obj_new
                best_a = a_new.copy()

        # Quaternary: log-param
        remaining = time_budget - (time.monotonic() - start)
        if remaining > 8.0:
            alloc = min(remaining * 0.1, 8.0)
            a_new = log_param_opt(best_a, max_iters=400, time_budget=alloc)
            obj_new = compute_obj(a_new)
            if obj_new < best_obj:
                best_obj = obj_new
                best_a = a_new.copy()

        # Perturbation escape
        remaining = time_budget - (time.monotonic() - start)
        if remaining > 10.0:
            rng = np.random.default_rng(int(time.monotonic() * 10000) % (2**31))
            for scale in [0.04, 0.08]:
                if time.monotonic() - start > time_budget - 5.0:
                    break
                a_p = best_a * np.exp(rng.normal(0, scale, n))
                a_p = np.maximum(a_p, 0.0)
                S_p = np.sum(a_p)
                if S_p > 1e-10:
                    a_p = a_p / S_p
                    a_p = momentum_subgrad(a_p, max_iters=200, time_budget=min(4.0, remaining * 0.15), topk=6)
                    obj_p = compute_obj(a_p)
                    if obj_p < best_obj:
                        best_obj = obj_p
                        best_a = a_p.copy()

    return best_a, best_obj


def propose_candidate(seed=42, budget_s=1500):
    """
    Main search function. Uses blend-LP, log-parameterization, symmetric exploitation,
    and adaptive multi-start to minimize 2*n*max(conv(a,a))/(sum(a))^2.
    """
    start_time = time.monotonic()
    deadline = start_time + float(budget_s) - 20.0

    best_seq = None
    best_score = np.inf

    # Load parent
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64)
        parent = np.clip(parent, 0, 1000.0)
        if parent.size > 0 and np.isfinite(parent).all():
            best_seq = parent.copy()
            best_score = compute_obj(best_seq)
            n_parent = len(best_seq)
            print(f"Parent: n={n_parent}, obj={best_score:.6f}")
    except Exception:
        n_parent = 500
        best_seq = np.ones(n_parent) / n_parent
        best_score = compute_obj(best_seq)
        print(f"No parent. Init: n={n_parent}, obj={best_score:.6f}")

    # === Phase 1: Diverse initialization ===
    print("\n=== Phase 1: Diverse initializations ===")
    n_list = sorted(set([
        n_parent,
        max(200, n_parent - 150),
        min(1000, n_parent + 150),
        max(300, int(n_parent * 1.3)),
    ]))
    n_list = [nn for nn in n_list if 150 <= nn <= 1200]

    candidates = []
    for n in n_list:
        if time.monotonic() > deadline - 120:
            break
        inits = [
            cosine_init(n, n_modes=10, seed=seed + n),
            raised_cosine_init(n, 1.5),
            raised_cosine_init(n, 2.5),
            gaussian_init(n, 0.25),
            symmetric_init(n, seed=seed + n * 7),
        ]
        for idx, a_init in enumerate(inits):
            if time.monotonic() > deadline - 60:
                break
            obj_init = compute_obj(a_init)
            a_quick = momentum_subgrad(a_init, max_iters=200, time_budget=2.5, topk=6)
            obj_quick = compute_obj(a_quick)
            best_local = a_quick if obj_quick < obj_init else a_init
            best_local_obj = min(obj_quick, obj_init)
            candidates.append((best_local, best_local_obj))
            if best_local_obj < best_score:
                best_score = best_local_obj
                best_seq = best_local.copy()
                print(f"  init n={n} type={idx}: {best_score:.6f} ** NEW BEST **")

    # === Phase 2: Intensive refinement of best ===
    print(f"\n=== Phase 2: Intensive refinement (best={best_score:.6f}) ===")
    a_cur = best_seq.copy()

    for round_num in range(40):
        if time.monotonic() > deadline - 40:
            break
        remaining = deadline - time.monotonic()
        if remaining < 25:
            break

        alloc = min(remaining * 0.18, 45.0)
        a_new, obj_new = refine_sequence(a_cur, time_budget=alloc)

        if obj_new < best_score:
            best_score = obj_new
            best_seq = a_new.copy()
            a_cur = a_new.copy()
            print(f"  R{round_num}: {best_score:.6f} ** NEW BEST **")
        else:
            if round_num % 3 == 0:
                print(f"  R{round_num}: {obj_new:.6f} (best={best_score:.6f})")
            # Perturbation at stalls
            if remaining > 30:
                rng = np.random.default_rng(round_num * 777 + 333 + n_parent)
                for scale in [0.06, 0.12, 0.2]:
                    if time.monotonic() > deadline - 20:
                        break
                    a_p = a_cur * np.exp(rng.normal(0, scale, len(a_cur)))
                    a_p = np.maximum(a_p, 0.0)
                    S_p = np.sum(a_p)
                    if S_p > 1e-10:
                        a_p = a_p / S_p
                        a_p, obj_p = refine_sequence(a_p, time_budget=min(15.0, remaining * 0.12))
                        if obj_p < best_score:
                            best_score = obj_p
                            best_seq = a_p.copy()
                            a_cur = a_p.copy()
                            print(f"  R{round_num} pert({scale}): {best_score:.6f} ** NEW BEST **")
                            break

    # === Phase 3: Length exploration ===
    print(f"\n=== Phase 3: Length exploration (best={best_score:.6f}) ===")
    if time.monotonic() < deadline - 50:
        n_base = len(best_seq)
        n_tries = sorted(set([
            max(150, n_base - 100),
            n_base + 50,
            n_base + 100,
            n_base + 200,
            max(200, int(n_base * 1.4)),
        ]))
        n_tries = [nn for nn in n_tries if 150 <= nn <= 2000]

        for n_try in n_tries:
            if time.monotonic() > deadline - 25:
                break
            remaining = deadline - time.monotonic()
            if remaining < 25:
                break

            x_old = np.linspace(0, 1, n_base)
            x_new = np.linspace(0, 1, n_try)
            try:
                f_interp = interp1d(x_old, best_seq, kind='cubic', fill_value='extrapolate')
                a_interp = np.maximum(f_interp(x_new), 0.0)
            except Exception:
                a_interp = np.ones(n_try) / n_try

            S_i = np.sum(a_interp)
            if S_i > 1e-10:
                a_interp = a_interp / S_i

            alloc = min(remaining * 0.3, 35.0)
            a_new, obj_new = refine_sequence(a_interp, time_budget=alloc)
            if obj_new < best_score:
                best_score = obj_new
                best_seq = a_new.copy()
                print(f"  n={n_try}: {best_score:.6f} ** NEW BEST **")
            else:
                print(f"  n={n_try}: {obj_new:.6f}")

    # === Phase 4: Final intensive polish ===
    print(f"\n=== Phase 4: Final polish (best={best_score:.6f}) ===")
    a_cur = best_seq.copy()
    remaining = deadline - time.monotonic()

    if remaining > 20:
        # Extended blend-LP
        alloc = min(remaining * 0.4, 50.0)
        a_new = alternating_blend(a_cur, max_iters=40, time_budget=alloc,
                                   t_values=[0.05, 0.1, 0.15, 0.2, 0.3, 0.5])
        obj_new = compute_obj(a_new)
        if obj_new < best_score:
            best_score = obj_new
            best_seq = a_new.copy()
            a_cur = a_new.copy()
            print(f"  Final blend-LP: {best_score:.6f} ** NEW BEST **")

        # Extended subgradient
        remaining = deadline - time.monotonic()
        if remaining > 15:
            a_new = momentum_subgrad(a_cur, max_iters=5000, time_budget=remaining * 0.25, topk=10)
            obj_new = compute_obj(a_new)
            if obj_new < best_score:
                best_score = obj_new
                best_seq = a_new.copy()
                print(f"  Final SG: {best_score:.6f} ** NEW BEST **")

        # Final perturbation rounds
        remaining = deadline - time.monotonic()
        if remaining > 12:
            rng = np.random.default_rng(424242)
            for _ in range(4):
                if time.monotonic() > deadline - 8:
                    break
                a_p = a_cur * np.exp(rng.normal(0, 0.04, len(a_cur)))
                a_p = np.maximum(a_p, 0.0)
                S_p = np.sum(a_p)
                if S_p > 1e-10:
                    a_p = a_p / S_p
                    a_p = momentum_subgrad(a_p, max_iters=1500, time_budget=min(remaining * 0.12, 10.0), topk=10)
                    obj_p = compute_obj(a_p)
                    if obj_p < best_score:
                        best_score = obj_p
                        best_seq = a_p.copy()
                        a_cur = a_p.copy()
                        print(f"  Perturb: {best_score:.6f} ** NEW BEST **")

    # Final validation
    if best_seq is None:
        best_seq = np.ones(300) / 300
        best_score = compute_obj(best_seq)

    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    val = evaluate_sequence(result)
    if val < np.inf:
        if val < best_score:
            best_score = val
    else:
        result = [1.0] * 300
        best_score = 2.0

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_score:.6f}, time={elapsed:.1f}s ===")

    return result
```
