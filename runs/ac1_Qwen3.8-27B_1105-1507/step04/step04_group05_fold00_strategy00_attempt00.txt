        final_evaluation = evaluate_sequence(best_candidate)
        # If not found a solution under 1.5, restart from intersecting frequencies
        if final_evaluation > 1.5075:
            print("Full optimization exceeded budget, early termination.", file=sys.stderr)
            return best_candidate, final_evaluation

    # Print with flexibility for multiple formats
    print(f"Final candidate J={final_evaluation}, sequence length = {len(best_candidate)}")
    return best_candidate, final_evaluation
