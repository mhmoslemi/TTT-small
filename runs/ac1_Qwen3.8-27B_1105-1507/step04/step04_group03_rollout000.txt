from scipy.sparse import csr_matrix
import time

def fast_conv(a):
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]

def compute_J(a):
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))

def build_A_sparse(n, a):
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        j_min = max(0, k - n + 1)
        j_max = min(n - 1, k)
        for j in range(j_min, j_max + 1):
            rows.append(k)
            cols.append(j)
            vals.append(a[k - j])
    return csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))

def lp_best_response(a, M):
    n = len(a)
    A = build_A_sparse(n, a)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    try:
        r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
        if r.success and r.x is not None:
            g = np.maximum(r.x, 0)
            if g.sum() > 1e-15:
                return g
    except Exception:
        pass
    return None

def lp_mirror_descent(a, max_iters, deadline, use_mult=False, patience=60):
    a = np.maximum(a.astype(np.float64), 1e-10)
    a = a / a.sum()
    current_J = compute_J(a)
    best_J = current_J
    best_a = a.copy()
    no_improve = 0

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        b = fast_conv(a)
        M = float(b.max())
        g = lp_best_response(a, M)
        if g is None:
            break
        g = np.maximum(g, 0.0)
        g_sum = g.sum()
        if g_sum < 1e-10:
            break
        g = g / g_sum

        # Gap-adaptive step size
        gap = np.linalg.norm(g - a) / (np.linalg.norm(a) + 1e-15)
        base_t = min(1.0, 0.3 * gap + 0.08)

        best_trial_J = current_J
        best_trial_a = None
        t = base_t

        for ls in range(30):
            if use_mult:
                ratio = g / np.maximum(a, 1e-15)
                a_trial = a * np.power(ratio, t)
            else:
                a_trial = (1.0 - t) * a + t * g
            a_trial = np.maximum(a_trial, 0.0)
            s = a_trial.sum()
            if s < 1e-10:
                t *= 0.5
                continue
            a_trial = a_trial / s
            J_trial = compute_J(a_trial)
            if J_trial < best_trial_J - 1e-15:
                best_trial_J = J_trial
                best_trial_a = a_trial.copy()
                break
            t *= 0.6
            if t < 0.001:
                break

        if best_trial_a is not None:
            a = best_trial_a
            current_J = best_trial_J
            if current_J < best_J:
                best_J = current_J
                best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1
            if no_improve > patience:
                break

    return best_a, best_J

def softmax_objective(x, tau):
    ax = np.maximum(x, 0.0)
    s = ax.sum()
    if s < 1e-8:
        return 1e10
    ax = ax / s
    b = fast_conv(ax)
    # Soft-max approximation
    scaled = tau * (b - b.max())
    logsumexp_val = b.max() + np.log(np.sum(np.exp(scaled))) / tau
    return float(logsumexp_val)

def softmax_lbb(a, deadline, tau0=50.0, tau_min=200.0, n_stages=5):
    a = np.maximum(a.astype(np.float64), 1e-10)
    a = a / a.sum()
    best_a = a.copy()
    best_M = fast_conv(a).max()
    
    taus = np.geomspace(tau0, tau_min, n_stages)
    
    for tau in taus:
        if time.monotonic() > deadline:
            break
        
        def obj(x):
            return softmax_objective(x, tau)
        
        def grad(x):
            ax = np.maximum(x, 0.0)
            s = ax.sum()
            if s < 1e-8:
                return np.zeros_like(x)
            ax = ax / s
            b = fast_conv(ax)
            scaled = tau * (b - b.max())
            weights = np.exp(scaled) / np.sum(np.exp(scaled))
            n = len(ax)
            g = np.zeros_like(ax)
            for k in range(2 * n - 1):
                if weights[k] < 1e-15:
                    continue
                lo = max(0, k - n + 1)
                hi = min(n - 1, k)
                for i in range(lo, hi + 1):
                    g[i] += 2.0 * weights[k] * ax[k - i]
            # Subtract mean for normalization
            g = g - np.mean(g)
            return g / s
        
        try:
            res = minimize(obj, a, jac=grad, method='L-BFGS-B',
                          bounds=[(0.0, None)] * len(a),
                          options={'maxiter': 200, 'max_fun': 5000, 'ftol': 1e-14})
            if res.fun < 1e10:
                a_trial = np.maximum(res.x, 0.0)
                if a_trial.sum() > 1e-8:
                    a_trial = a_trial / a_trial.sum()
                    M_trial = fast_conv(a_trial).max()
                    if M_trial < best_M:
                        best_M = M_trial
                        best_a = a_trial.copy()
                        a = a_trial
        except Exception:
            pass
    
    return best_a

def resample_seq(src, dest_n):
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)

def random_perturb(a, rng, scale=0.1):
    n = len(a)
    t = np.arange(n, dtype=np.float64) / n
    ptype = rng.integers(0, 5)
    if ptype == 0:
        a_p = a + rng.normal(0, scale / np.sqrt(n), n)
    elif ptype == 1:
        s = rng.integers(1, max(2, n // 8))
        a_p = np.roll(a, s)
    elif ptype == 2:
        freq = rng.uniform(0.5, 5)
        amp = rng.uniform(0.02, scale)
        a_p = a * (1 + amp * np.cos(2 * np.pi * freq * t + rng.uniform(0, 2 * np.pi)))
    elif ptype == 3:
        k = rng.integers(3, max(4, n // 8))
        kernel = np.ones(k) / k
        a_p = np.convolve(a, kernel, mode='same')
    else:
        # Block perturbation
        block_size = rng.integers(4, max(5, n // 16))
        start = rng.integers(0, n - block_size)
        a_p = a.copy()
        a_p[start:start+block_size] *= rng.uniform(0.5, 1.5)
    
    a_p = np.maximum(a_p, 1e-6)
    s = a_p.sum()
    if s > 1e-8:
        a_p = a_p / s
    return a_p
