import numpy as np

def fast_conv(a):
    """Full self-convolution via FFT."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]


def compute_J(a):
    """Compute J(a) = 2*n*max(conv(a,a))/(sum(a))^2."""
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))


def smooth_max_and_grad(a, tau):
    """Compute smooth max approximation and its gradient."""
    n = len(a)
    b = fast_conv(a)
    b_max = b.max()
    exp_b = np.exp(tau * (b - b_max))
    p = exp_b / exp_b.sum()
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        pk = p[k]
        if pk < 1e-14:
            continue
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        grad[lo:hi + 1] += 2.0 * pk * a[k - hi:k - lo + 1][::-1]
    return grad, p, b, b_max


def subgradient_direction(a, tol_frac=1e-7):
    """Compute subgradient of max(conv(a,a)) with tolerance-based active set."""
    n = len(a)
    b = fast_conv(a)
    M = b.max()
    tol = tol_frac * M if M > 0 else 1e-10
    active = np.where(b >= M - tol)[0]
    grad = np.zeros(n)
    for k in active:
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        grad[lo:hi + 1] += a[k - hi:k - lo + 1][::-1]
    gnorm = np.linalg.norm(grad)
    if gnorm > 1e-15:
        grad /= gnorm
    return grad, active, b, M


def build_convolution_matrix(n, a):
    """Build sparse matrix A where A[k,j] = a[k-j] if valid."""
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        j_min = max(0, k - n + 1)
        j_max = min(n - 1, k)
        if j_max >= j_min:
            js = np.arange(j_min, j_max + 1)
            rows.extend([k] * len(js))
            cols.extend(js)
            vals.extend(a[k - js])
    A = csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))
    return A


def solve_lp_best_response(a, M):
    """LP: max sum(g) s.t. (a*g)[k] <= M, g >= 0. Returns g or None."""
    n = len(a)
    A = build_convolution_matrix(n, a)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    try:
        r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                    options={'time_limit': 30, 'presolve': True})
        if r.success and r.x is not None:
            g = np.maximum(r.x, 0)
            if g.sum() > 1e-15:
                return g, r
    except Exception:
        pass
    return None, None


def solve_lp_dual_info(a, M):
    """Solve LP and extract dual variables for weighted subgradient."""
    result = solve_lp_best_response(a, M)
    if result[0] is None:
        return None, None
    g, r = result
    # Extract dual variables from HiGHS solution
    try:
        if hasattr(r, 'ineqlin') and r.ineqlin is not None:
            y = getattr(r.ineqlin, 'duals', None)
            if y is None and hasattr(r, 'duals'):
                y = r.duals
            if y is not None:
                y = np.maximum(np.asarray(y), 0)
                return g, y
    except Exception:
        pass
    return g, None

    n = len(a)
    best_a = a.copy()
    best_M = fast_conv(a).max()
    v = np.zeros(n)

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        frac = it / max(1, max_iters)
        tau = initial_tau + (final_tau - initial_tau) * (frac ** 0.6)

            best_a = a.copy()

        a_new = a - v
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s < 1e-10:
            a_new = a.copy()
        else:
            a_new = a_new / s
        a = a_new

    return a

def structured_sa(a, deadline, rng, max_iters=5000):
    """Simulated annealing with structured moves."""
    a = np.maximum(a.astype(np.float64), 1e-10)
    a = a / a.sum()
    n = len(a)
    best_a = a.copy()
    best_M = float(fast_conv(a).max())
    current_M = best_M
    temperature = 0.05

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        move_type = rng.integers(0, 4)
        a_trial = a.copy()

        if move_type == 0:
            # Swap two elements
            i, j = rng.integers(0, n, size=2)
            if i != j:
                a_trial[i], a_trial[j] = a_trial[j], a_trial[i]
        elif move_type == 1:
            # Shift mass from one element to another
            i = rng.integers(0, n)
            j = rng.integers(0, n)
            if i != j:
                delta = a[i] * rng.uniform(0.01, 0.2)
                a_trial[i] -= delta
                a_trial[j] += delta
        elif move_type == 2:
            # Smooth a pair of neighboring elements
            i = rng.integers(0, n - 1)
            avg = (a_trial[i] + a_trial[i + 1]) / 2
            a_trial[i] = avg
            a_trial[i + 1] = avg
        else:
            # Block rescale: pick a block and rescale
            blen = rng.integers(3, max(4, n // 10))
            start = rng.integers(0, n - blen + 1)
            block_sum = a_trial[start:start + blen].sum()
            if block_sum > 1e-10:
                new_vals = rng.dirichlet(np.ones(blen) * 0.5) * block_sum
                a_trial[start:start + blen] = new_vals

        a_trial = np.maximum(a_trial, 0)
        s = a_trial.sum()
        if s < 1e-10:
            continue
        a_trial /= s
        M_trial = float(fast_conv(a_trial).max())

        # Metropolis acceptance
        delta_M = M_trial - current_M
        if delta_M < 0 or rng.random() < np.exp(-abs(delta_M) / max(temperature, 1e-10)):
            a = a_trial
            current_M = M_trial
            if M_trial < best_M:
                best_M = M_trial
                best_a = a_trial.copy()

        # Cool down
        temperature *= 0.9995
        if temperature < 0.001:
            temperature = 0.001

    return best_a


def resample_seq(src, dest_n):
    """Resample 1D sequence to new length."""
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)


def generate_diverse_inits(n, rng, parent=None):

    block_seq = np.repeat(block_vals, n // n_blocks + 1)[:n]
    inits.append(np.maximum(block_seq, 0.01))

    # Fourier random (band-limited)
    spectrum = np.zeros(n // 2 + 1)
    max_band = rng.integers(2, max(3, n // 8))
    spectrum[:max_band + 1] = rng.uniform(0.5, 1.0, max_band + 1)
    x = np.fft.irfft(spectrum * np.exp(1j * rng.uniform(0, 2 * np.pi, max_band + 1)), n)
    inits.append(np.maximum(x - x.min() + 0.1, 0.01))

    # Normalize all
    result = []
    for x in inits:
        x = np.maximum(x, 0)
        s = x.sum()
        if s > 1e-10:
            result.append(x / s)
    return result


def run_optimization_pipeline(a0, deadline, n):
    """Run a full optimization pipeline on a starting sequence."""
    a = np.maximum(a0.astype(np.float64), 1e-8)
    a = a / a.sum()
    best = a.copy()
    best_M = float(fast_conv(a).max())

    def update_best(candidate):
        nonlocal best, best_M
        M_cand = float(fast_conv(candidate).max())
        if M_cand < best_M:
            best = candidate.copy()
            best_M = M_cand

    # Step 1: Quick smooth gradient
    t1_deadline = min(time.monotonic() + 30, deadline - 60)
    a = run_smooth_gradient(a, 200, t1_deadline, initial_tau=3, final_tau=50,
                           lr_init=0.2, lr_min=0.03)
    update_best(a)

    # Step 2: LP mirror descent
    t2_deadline = min(time.monotonic() + 120, deadline - 30)
    a2 = run_lp_mirror_descent_improved(a, 100, t2_deadline)
    update_best(a2)

    # Step 3: Peak shaving
    t3_deadline = min(time.monotonic() + 60, deadline - 15)
    a3 = peak_shaving_descent(a2, 500, t3_deadline)
    update_best(a3)

    # Step 4: SA polish
    if time.monotonic() < deadline - 5:
        rng = np.random.default_rng(hash(tuple(np.round(a2, 6))) % (2**31))
        a4 = structured_sa(a3, deadline, rng, max_iters=3000)
        update_best(a4)

    return best, best_M


def propose_candidate(seed=42, budget_s=1500, **kwargs):
    """
    Multi-phase hybrid search with improved LP mirror descent,
    peak-shaving coordinate descent, and structured SA.
    """
    rng = np.random.default_rng(seed)
    t_start = time.monotonic()
    deadline = t_start + float(budget_s) - 20.0

    best_seq = None
    best_score = float('inf')

    def update_tracker(a):
        nonlocal best_seq, best_score
        a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
        if a.size == 0 or a.sum() < 0.01:
            return
        score = float(evaluate_sequence(a.tolist()))
        if np.isfinite(score) and score < best_score:
            best_seq = a.copy()
            best_score = score
            elapsed = time.monotonic() - t_start
            print(f"  ** NEW BEST: J={score:.8f} n={a.size} t={elapsed:.0f}s", flush=True)

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        if p.ndim == 1 and p.size > 0 and np.all(np.isfinite(p)):
            parent = np.maximum(p, 0.0)
            update_tracker(parent)
            print(f"Parent: n={parent.size}, J={best_score:.6f}", flush=True)
    except (NameError, AttributeError):
        pass

    if parent is None:
        parent = np.ones(256)
        update_tracker(parent)

    # ============ PHASE 1: Multi-length smooth gradient exploration (300s) ============
    p1_deadline = min(t_start + 250, deadline - 400)
    print(f"\n{'='*60}", flush=True)
    print(f"PHASE 1: Multi-length smooth gradient (deadline={p1_deadline - time.monotonic():.0f}s)", flush=True)
    print(f"{'='*60}", flush=True)

    lengths_to_try = [256, 384, 512, 640, 768, 1024]
    if parent is not None:
        # Ensure parent length is included
        if parent.size not in lengths_to_try:
            lengths_to_try.append(parent.size)
        lengths_to_try.sort()

    for n in lengths_to_try:
        if time.monotonic() > p1_deadline:
            break
        # Time budget proportional to 1/n (shorter = more time per unit)
        time_budget = min(40, max(15, 200.0 * (512.0 / n) / len(lengths_to_try)))
        phase_dl = min(time.monotonic() + time_budget, p1_deadline)

        inits = generate_diverse_inits(n, rng, parent)
        best_n = None
        best_n_M = float('inf')

        for idx, a0 in enumerate(inits[:8]):  # Limit number of inits
            if time.monotonic() > phase_dl:
                break
            a0 = np.maximum(a0, 1e-6)
            a0 = a0 / a0.sum()
            iters = max(30, int(500 * (512 / n)))
            a = run_smooth_gradient(a0, iters, phase_dl,
                                    initial_tau=2, final_tau=40,
                                    lr_init=0.2, lr_min=0.02)
            M_val = float(fast_conv(a).max())
            if M_val < best_n_M:
                best_n_M = M_val
                best_n = a.copy()

        if best_n is not None:
            J_val = 2 * n * best_n_M / (best_n.sum() ** 2)
            print(f"  n={n:4d}: J={J_val:.6f} ({time.monotonic() - (p1_deadline - time_budget):.0f}s)", flush=True)
            update_tracker(best_n)
            if best_seq is not None and best_seq.size == n:
                # Save for later refinement
                pass

    print(f"Phase 1 done. Best: J={best_score:.6f}", flush=True)

    # ============ PHASE 2: LP Mirror Descent on best (350s) ============
    p2_deadline = min(t_start + 600, deadline - 200)
    print(f"\n{'='*60}", flush=True)
    print(f"PHASE 2: LP Mirror Descent (budget ~{p2_deadline - time.monotonic():.0f}s)", flush=True)
    print(f"{'='*60}", flush=True)

    if best_seq is not None:
        a = best_seq.copy()
        # Run LP mirror descent with generous budget
        a_opt = run_lp_mirror_descent_improved(a, 200, p2_deadline, use_momentum=True)
        update_tracker(a_opt)

        # Second pass starting from the result
        if time.monotonic() < p2_deadline - 30:
            a_opt2 = run_lp_mirror_descent_improved(a_opt, 100, p2_deadline, use_momentum=True)
            update_tracker(a_opt2)

    print(f"Phase 2 done. Best: J={best_score:.6f}", flush=True)

    # ============ PHASE 3: Peak shaving + SA (150s) ============
    p3_deadline = min(t_start + 750, deadline - 100)
    print(f"\n{'='*60}", flush=True)
    print(f"PHASE 3: Peak Shaving + SA (budget ~{p3_deadline - time.monotonic():.0f}s)", flush=True)
    print(f"{'='*60}", flush=True)

    if best_seq is not None:
        a = best_seq.copy()
        a_ps = peak_shaving_descent(a, 2000, min(time.monotonic() + 60, p3_deadline - 30))
        update_tracker(a_ps)

        if time.monotonic() < p3_deadline - 20:
            rng_sa = np.random.default_rng(seed + 100)
            a_sa = structured_sa(a_ps, p3_deadline, rng_sa, max_iters=10000)
            update_tracker(a_sa)

    print(f"Phase 3 done. Best: J={best_score:.6f}", flush=True)

    # ============ PHASE 4: Multi-start pipeline on different lengths (200s) ============
    p4_deadline = min(t_start + 950, deadline - 50)
    print(f"\n{'='*60}", flush=True)
    print(f"PHASE 4: Multi-start pipeline (budget ~{p4_deadline - time.monotonic():.0f}s)", flush=True)
    print(f"{'='*60}", flush=True)

    if best_seq is not None:
        # Try refining at different lengths
        base_n = best_seq.size
        try_lengths = [base_n // 2, base_n, base_n * 2]
        try_lengths = [n for n in try_lengths if 64 <= n <= 2048]
        # Remove duplicates
        seen = set()
        unique_lengths = []
        for n in try_lengths:
            if n not in seen:
                seen.add(n)
                unique_lengths.append(n)

        for n in unique_lengths:
            if time.monotonic() > p4_deadline - 30:
                break
            a0 = resample_seq(best_seq, n)
            a0 = np.maximum(a0, 1e-6)
            a0 = a0 / a0.sum()

            # Also try a perturbed version
            a0_pert = np.maximum(a0 + rng.normal(0, 0.03, n), 0.01)
            a0_pert = a0_pert / a0_pert.sum()

            for a_init in [a0, a0_pert]:
                if time.monotonic() > p4_deadline - 20:
                    break
                a_opt, M_opt = run_optimization_pipeline(a_init, min(time.monotonic() + 80, p4_deadline - 10), n)
                update_tracker(a_opt)

    print(f"Phase 4 done. Best: J={best_score:.6f}", flush=True)

    # ============ PHASE 5: Final intensive LP on best (remaining time) ============
    print(f"\n{'='*60}", flush=True)
    print(f"PHASE 5: Final intensive refinement", flush=True)
    print(f"{'='*60}", flush=True)

    if best_seq is not None and time.monotonic() < deadline - 10:
        # Multiple short LP passes with different initializations
        for attempt in range(5):
            if time.monotonic() > deadline - 10:
                break
            a = best_seq.copy()
            if attempt > 0:
                a = a + rng.normal(0, 0.01 * (1 - attempt * 0.15), best_seq.size)
                a = np.maximum(a, 0)
                a = a / a.sum()

            # Short smooth gradient then LP
            a = run_smooth_gradient(a, 50, min(time.monotonic() + 15, deadline - 15),
                                   initial_tau=10, final_tau=80, lr_init=0.05, lr_min=0.01)
            a = run_lp_mirror_descent_improved(a, 30, min(time.monotonic() + 30, deadline - 10),
                                              use_momentum=True)
            update_tracker(a)

    # ============ PHASE 6: One final LP pass with maximum time ============
    if best_seq is not None and time.monotonic() < deadline - 5:
        a = best_seq.copy()
        a = run_lp_mirror_descent_improved(a, 50, deadline, use_momentum=True)
        update_tracker(a)

    # Fallback
    if best_seq is None:
        best_seq = np.ones(128)
        best_score = compute_J(best_seq)

    print(f"\n{'='*60}", flush=True)
    print(f"FINAL: J={best_score:.8f} n={best_seq.size} time={time.monotonic()-t_start:.0f}s", flush=True)
    print(f"{'='*60}", flush=True)

    return [float(x) for x in best_seq]
