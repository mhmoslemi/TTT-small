```python
def solve_qp_fista(lam, n, max_iters=100):
    """Solve min 0.5 a^T H(λ) a s.t. a in simplex using FISTA."""
    a = np.ones(n) / n
    a_prev = a.copy()
    t = 1.0
    # Lipschitz constant of H(λ): ||H||_2 <= sum_k λ_k * ||A_k||_2
    # ||A_k||_2 <= max eigenvalue of the k-th convolution matrix
    # For simplicity, use ||H|| <= sum_k λ_k * (max over i of sum_j |H[i,j]|) 
    # Actually, ||H||_2 <= max_i sum_j H[i,j] (for PSD) = max row sum
    # H[i,j] = lam[i+j], so row i sum = sum_j lam[i+j] for j=0..n-1
    # = sum_{m=i}^{i+n-1} lam[m]
    # Max over i of this sum
    
    row_sums = np.array([sum(lam[i:i+n]) for i in range(n)])
    L = row_sums.max()  # Upper bound on ||H||_2
    if L < 1e-15:
        return np.ones(n) / n
    
    lr = 1.0 / L
    
    for it in range(max_iters):
        grad = hankel_matvec(lam, a)  # H @ a
        a_new = project_simplex(a - lr * grad)
        
        # FISTA update
        t_new = (1 + np.sqrt(1 + 4 * t * t)) / 2
        a_prev = a_new + ((t - 1) / t_new) * (a_new - a)
        a = a_new
        t = t_new
    
    return a
```

```python
1. Fictitious Play with FISTA QP solver (exploits Hankel structure via FFT)

import numpy as np
from scipy.optimize import linprog, minimize
from scipy.sparse import csr_matrix
import time


def fast_conv(a):
    """Full self-convolution via FFT."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]


def compute_J(a):
    """Compute J(a) = 2*n*max(conv(a,a))/(sum(a))^2."""
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))


def hankel_matvec(lam, a):
    """Compute H(a) where H[i,j] = lam[i+j]. Uses FFT."""
    n = len(a)
    a_flipped = a[::-1]
    size = 1
    while size < n + len(lam):
        size <<= 1
    fa = np.fft.rfft(a_flipped, size)
    fl = np.fft.rfft(lam, size)
    conv = np.fft.irfft(fa * fl, size)
    return conv[n - 1:2 * n - 1]


def project_simplex(v):
    """Project v onto probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.where(u * np.arange(1, n + 1) > (cssv - 1))[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho = rho[-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def solve_qp_fista(lam, n, max_iters=80, warm_start=None):
    """Solve min 0.5 a^T H(lam) a s.t. a in simplex using FISTA."""
    if warm_start is not None:
        a = warm_start.copy()
        a = project_simplex(a)
    else:
        a = np.ones(n) / n
    a_prev = a.copy()
    t = 1.0

    # Lipschitz constant upper bound
    if len(lam) >= n:
        row_sums = np.array([lam[i:i + n].sum() for i in range(n)])
        L = row_sums.max()
    else:
        L = lam.sum() + 1e-10
    if L < 1e-15:
        return np.ones(n) / n
    lr = 1.0 / L

    for it in range(max_iters):
        grad = hankel_matvec(lam, a)
        a_new = project_simplex(a - lr * grad)
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t * t)) / 2.0
        a_prev = a_new + ((t - 1.0) / t_new) * (a_new - a)
        a = a_new
        t = t_new

    return a


def fictitious_play(a_init, n, max_iters=2000, deadline=None, 
                    eta=1.0, qpd_iters_start=30, qpd_iters_end=100,
                    optimistic=True, verbose=True):
    """
    Fictitious play / dual averaging for the game:
    min_a max_k (a*a)[k] s.t. a in simplex.
    """
    a = np.maximum(a_init, 0.0)
    s = a.sum()
    if s < 1e-10:
        a = np.ones(n) / n
    else:
        a = a / s

    # Initialize lambda from current convolution
    b = fast_conv(a)
    b = np.maximum(b, 1e-15)
    lam = b / b.sum()

    best_a = a.copy()
    best_obj = b.max()
    b_prev = b.copy()
    a_warm = a.copy()

    for it in range(max_iters):
        if deadline is not None and time.monotonic() > deadline:
            break

        # Adaptive QP precision
        frac = it / max(1, max_iters)
        qpd_iters = int(qpd_iters_start + (qpd_iters_end - qpd_iters_start) * frac)

        # Solve QP (best response)
        a = solve_qp_fista(lam, n, max_iters=qpd_iters, warm_start=a_warm)
        a_warm = a.copy()

        # Compute new convolution
        b = fast_conv(a)
        obj = b.max()
        if obj < best_obj * (1.0 - 1e-10):
            best_obj = obj
            best_a = a.copy()

        # Update lambda
        if optimistic and it > 0:
            b_signal = 2.0 * b - b_prev
        else:
            b_signal = b.copy()
        b_prev = b.copy()

        # Exponential weights update
        b_signal = np.maximum(b_signal, 1e-15)
        lam_new = lam * np.power(b_signal / b_signal.max(), eta)
        lam_new = np.maximum(lam_new, 1e-15)
        lam = lam_new / lam_new.sum()

        if verbose and it % 500 == 0:
            elapsed = ""
            if deadline is not None:
                elapsed = f" remaining={deadline - time.monotonic():.0f}s"
            print(f"    FP iter {it}: obj={obj:.10f}, best={best_obj:.10f}{elapsed}", flush=True)

    return best_a


def run_lp_mirror_descent(a, max_iters, deadline, verbose=True):
    """LP mirror descent with Nesterov acceleration and backtracking line search."""
    a = np.maximum(a.astype(np.float64), 1e-10)
    a = a / a.sum()
    n = len(a)

    best_a = a.copy()
    best_M = fast_conv(a).max()
    y = a.copy()  # Nesterov virtual position
    beta = 0.0
    no_improve = 0

    # Time tracking
    t_start = time.monotonic()

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        # Nesterov step
        a_curr = y + beta * (y - a) if beta > 0 else y
        a_curr = np.maximum(a_curr, 0.0)
        s = a_curr.sum()
        if s < 1e-10:
            a_curr = a.copy()
        else:
            a_curr = a_curr / s

        # Solve LP: max sum(g) s.t. (a_curr * g)[k] <= M for all k, g >= 0
        b_curr = fast_conv(a_curr)
        M = float(b_curr.max())

        # Build constraint matrix
        rows = []
        cols = []
        vals = []
        for k in range(2 * n - 1):
            j_min = max(0, k - n + 1)
            j_max = min(n - 1, k)
            for j in range(j_min, j_max + 1):
                rows.append(k)
                cols.append(j)
                vals.append(a_curr[k - j])
        A = csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))
        c = -np.ones(n)
        b_ub = np.full(2 * n - 1, M)
        bounds = [(0, None)] * n

        t_lp_start = time.monotonic()
        try:
            r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                        options={'time_limit': 30, 'presolve': True})
            if r.success and r.x is not None:
                g = np.maximum(r.x, 0)
            else:
                break
        except Exception:
            break
        t_lp = time.monotonic() - t_lp_start

        g_sum = g.sum()
        if g_sum < 1e-10:
            break
        g = g / g_sum

        # Backtracking line search
        improved = False
        base_t = 0.3
        for attempt in range(8):
            t = base_t * (0.7 ** attempt)
            a_trial = np.maximum((1.0 - t) * a + t * g, 0.0)
            s = a_trial.sum()
            if s < 1e-10:
                continue
            a_trial = a_trial / s
            M_trial = fast_conv(a_trial).max()
            if M_trial < best_M - 1e-14:
                # Nesterov momentum update
                beta_new = (1.0 + np.sqrt(1.0 + 4.0 * beta * beta)) / 2.0
                y = a_trial
                a = a_trial + ((beta - 1.0) / beta_new) * (a_trial - a)
                a = np.maximum(a, 0.0)
                s2 = a.sum()
                if s2 < 1e-10:
                    a = a_trial.copy()
                else:
                    a = a / s2
                beta = beta_new

                best_M = M_trial
                best_a = a_trial.copy()
                improved = True
                no_improve = 0
                break
            elif M_trial < M * (1.0 + 1e-6):
                # Accept even without improvement to best (to escape plateaus)
                if no_improve > 30:
                    a = a_trial
                    no_improve += 1
                    break
                else:
                    break
        if not improved:
            no_improve += 1
            if no_improve > 50:
                # Reset momentum
                beta = 0.0
                y = a.copy()
                no_improve = 0

        if verbose and it % 20 == 0:
            elapsed = time.monotonic() - t_start
            print(f"    LP-MD iter {it}: M={best_M:.10f}, lp_time={t_lp*1000:.0f}ms, "
                  f"elapsed={elapsed:.0f}s", flush=True)

    return best_a


def min_norm_subgradient(a, tol_frac=1e-6):
    """Compute minimum norm subgradient of max(conv(a,a))."""
    n = len(a)
    b = fast_conv(a)
    M = b.max()
    tol = tol_frac * M
    active = np.where(b >= M - tol)[0]

    if len(active) == 0:
        active = np.array([np.argmax(b)])
    if len(active) == 1:
        k = active[0]
        grad = np.zeros(n)
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        grad[lo:hi + 1] = 2.0 * a[k - hi:k - lo + 1][::-1]
        return grad, active

    # Build gradient vectors
    grads = []
    for k in active:
        g = np.zeros(n)
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        g[lo:hi + 1] = 2.0 * a[k - hi:k - lo + 1][::-1]
        grads.append(g)

    # G[k,l] = <grad_k, grad_l>
    G = np.zeros((len(active), len(active)))
    for i in range(len(active)):
        for j in range(i, len(active)):
            G[i, j] = np.dot(grads[i], grads[j])
            G[j, i] = G[i, j]

    # Solve: min 0.5 alpha^T G alpha s.t. sum(alpha) = 1, alpha >= 0
    m = len(active)
    # First try without bound (ignore alpha >= 0)
    ones = np.ones(m)
    # KKT: G @ alpha + nu * ones = 0, ones @ alpha = 1
    KKT = np.zeros((m + 1, m + 1))
    KKT[:m, :m] = G
    KKT[:m, m] = ones
    KKT[m, :m] = ones
    rhs = np.zeros(m + 1)
    rhs[m] = 1.0
    try:
        sol = np.linalg.solve(KKT, rhs)
        alpha = sol[:m]
        if np.all(alpha >= -1e-12):
            alpha = np.maximum(alpha, 0)
            alpha = alpha / alpha.sum()
            g_min = sum(alpha[i] * grads[i] for i in range(m))
            return g_min, active
    except np.linalg.LinAlgError:
        pass

    # Fallback: convex optimization
    def obj(alpha):
        return 0.5 * alpha @ G @ alpha
    def grad_obj(alpha):
        return G @ alpha
    constraints = [{'type': 'eq', 'fun': _simplex_eq, 'jac': _simplex_eq_jac}]
    bounds_list = [(0, None)] * m
    alpha0 = np.ones(m) / m
    res = minimize(obj, alpha0, jac=grad_obj, constraints=constraints,
                   bounds=bounds_list, method='SLSQP', options={'maxiter': 200})
    if res.success:
        alpha = np.maximum(res.x, 0)
        alpha = alpha / alpha.sum()
    else:
        alpha = np.ones(m) / m
    g_min = sum(alpha[i] * grads[i] for i in range(m))
    return g_min, active


def _simplex_eq(x):
    return x.sum() - 1.0


def _simplex_eq_jac(x):
    return np.ones(len(x))


def run_min_norm_subgradient(a, max_iters, deadline, lr_init=0.3, verbose=True):
    """Subgradient descent using minimum norm subgradient."""
    a = np.maximum(a.astype(np.float64), 1e-10)
    a = a / a.sum()
    n = len(a)

    best_a = a.copy()
    best_M = fast_conv(a).max()

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        lr = lr_init / (1.0 + 0.1 * it) ** 0.7
        grad, active = min_norm_subgradient(a)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-15:
            break
        grad = grad / gnorm

        b = fast_conv(a)
        M = b.max()
        if M < best_M * (1.0 - 1e-9):
            best_M = M
            best_a = a.copy()

        a_new = a - lr * grad
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s < 1e-10:
            a_new = a.copy()
        else:
            a_new = a_new / s
        a = a_new

        if verbose and it % 200 == 0:
            print(f"    Subgrad iter {it}: M={best_M:.10f}", flush=True)

    return best_a


def resample_seq(src, dest_n):
    """Resample 1D sequence to new length via linear interpolation."""
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)


def fourier_init(n, rng, max_harmonics=8):
    """Generate a Fourier-based initialization."""
    t = np.arange(n) / n
    coeffs = []
    for k in range(1, max_harmonics + 1):
        c = rng.normal(0, 0.3 / np.sqrt(k))
        d = rng.normal(0, 0.3 / np.sqrt(k))
        coeffs.append((c, d))
    a = np.ones(n) * (0.5 + rng.uniform(0, 0.3))
    for k, (c, d) in enumerate(coeffs):
        a += c * np.cos(2 * np.pi * (k + 1) * t) + d * np.sin(2 * np.pi * (k + 1) * t)
    return np.maximum(a, 0.01)


def run_smooth_gradient(a, max_iters, deadline, initial_tau=3.0, final_tau=60.0,
                        lr_init=0.3, lr_min=0.01):
    """Smooth gradient descent with adaptive tau (log-sum-exp smoothing)."""
    a = np.maximum(a.astype(np.float64), 1e-10)
    a = a / a.sum()
    n = len(a)

    best_a = a.copy()
    best_M = fast_conv(a).max()
    v = np.zeros(n)

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        frac = it / max(1, max_iters)
        tau = initial_tau + (final_tau - initial_tau) * (frac ** 0.5)
        lr = lr_init * (lr_min / lr_init) ** frac

        b = fast_conv(a)
        b_max = b.max()
        exp_b = np.exp(tau * (b - b_max))
        p = exp_b / exp_b.sum()

        grad = np.zeros(n)
        for k in range(2 * n - 1):
            pk = p[k]
            if pk < 1e-13:
                continue
            lo = max(0, k - n + 1)
            hi = min(n - 1, k)
            grad[lo:hi + 1] += 2.0 * pk * a[k - hi:k - lo + 1][::-1]

        M = b.max()
        if M < best_M * (1.0 - 1e-9):
            best_M = M
            best_a = a.copy()

        v = 0.9 * v + lr * grad
        a_new = np.maximum(a - v, 0.0)
        s = a_new.sum()
        if s < 1e-10:
            a_new = a.copy()
        else:
            a_new = a_new / s
        a = a_new

    return best_a


class ScoreTracker:
    """Tracks the best sequence found so far."""

    def __init__(self):
        self.best = None
        self.best_score = float('inf')
        self.t_start = time.monotonic()

    def update(self, a):
        a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
        if a.size == 0 or a.sum() < 0.01:
            return
        score = float(evaluate_sequence(a.tolist()))
        if np.isfinite(score) and score < self.best_score:
            self.best = a.copy()
            self.best_score = score
            elapsed = time.monotonic() - self.t_start
            print(f"  ** NEW BEST: J={score:.10f} n={a.size} t={elapsed:.0f}s", flush=True)

    def report(self, label=""):
        if self.best is not None:
            print(f"  {label} best: J={self.best_score:.10f} n={self.best.size}", flush=True)


def propose_candidate(seed=42, budget_s=1500, **kwargs):
    """
    Multi-phase hybrid search combining:
    1. Fictitious Play with FISTA QP (exploits game structure)
    2. Nesterov LP Mirror Descent 
    3. Min-norm subgradient polishing
    4. Smooth gradient + multi-length exploration
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + float(budget_s) - 20.0
    tracker = ScoreTracker()

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        if p.ndim == 1 and p.size > 0 and np.all(np.isfinite(p)):
            parent = np.maximum(p, 0.0)
            tracker.update(parent)
            print(f"Parent loaded: n={parent.size}, J={tracker.best_score:.8f}", flush=True)
    except (NameError, AttributeError):
        pass

    if parent is None:
        parent = np.ones(256)
        tracker.update(parent)

    current_best = parent.copy()
    n_best = current_best.size

    # === PHASE 1: Fictitious Play (new approach) ===
    p1_deadline = min(time.monotonic() + 350, deadline - 400)
    print(f"\n{'='*60}", flush=True)
    print(f"PHASE 1: Fictitious Play (deadline={p1_deadline - time.monotonic():.0f}s)", flush=True)
    print(f"{'='*60}", flush=True)

    # Try multiple lengths
    fp_lengths = [n_best]
    # Add nearby lengths
    for mult in [0.75, 1.25, 1.5, 2.0]:
        fn = int(n_best * mult)
        if 128 <= fn <= 2048 and fn not in fp_lengths:
            fp_lengths.append(fn)
    # Add some standard lengths
    for fl in [256, 384, 512, 640, 768, 1024]:
        if 128 <= fl <= 2048 and fl not in fp_lengths:
            fp_lengths.append(fl)

    fp_lengths = sorted(set(fp_lengths))
    print(f"  FP lengths: {fp_lengths}", flush=True)

    for n in fp_lengths:
        if time.monotonic() > p1_deadline - 30:
            break

        # Generate initializations for this length
        inits = []
        if parent is not None:
            inits.append(resample_seq(parent, n))
        inits.append(fourier_init(n, rng))
        inits.append(np.ones(n))
        inits.append(0.5 + 0.5 * np.cos(2 * np.pi * np.arange(n) / n * 3))
        inits.append(fourier_init(n, rng, max_harmonics=5))

        best_fp = None
        best_fp_M = float('inf')
        sub_deadline = min(time.monotonic() + 60, p1_deadline - 20)

        for a0 in inits:
            if time.monotonic() > sub_deadline:
                break
            a0 = np.maximum(a0.astype(np.float64), 1e-8)
            a0 = a0 / a0.sum()

            a_fp = fictitious_play(a0, n, max_iters=3000, deadline=sub_deadline,
                                   eta=1.5, qpd_iters_start=20, qpd_iters_end=60,
                                   optimistic=True, verbose=False)
            M_val = fast_conv(a_fp).max()
            if M_val < best_fp_M:
                best_fp_M = M_val
                best_fp = a_fp

        if best_fp is not None:
            J_val = compute_J(best_fp)
            print(f"  FP n={n:4d}: M={best_fp_M:.10f} J={J_val:.8f}", flush=True)
            tracker.update(best_fp)
            if J_val < compute_J(current_best):
                current_best = best_fp.copy()
                n_best = n

    tracker.report("Phase 1 (FP) complete")

    # === PHASE 2: LP Mirror Descent with Nesterov ===
    p2_deadline = min(time.monotonic() + 350, deadline - 200)
    print(f"\n{'='*60}", flush=True)
    print(f"PHASE 2: LP Mirror Descent (deadline={p2_deadline - time.monotonic():.0f}s)", flush=True)
    print(f"{'='*60}", flush=True)

    if tracker.best is not None:
        a = run_lp_mirror_descent(tracker.best, max_iters=200, deadline=p2_deadline, verbose=True)
        tracker.update(a)

        # Second pass if improved
        if time.monotonic() < p2_deadline - 30:
            a2 = run_lp_mirror_descent(a, max_iters=100, deadline=p2_deadline, verbose=False)
            tracker.update(a2)

    # Also try on current_best if different
    if tracker.best is not None and not np.allclose(tracker.best, current_best):
        if time.monotonic() < deadline - 100:
            a3 = run_lp_mirror_descent(current_best, max_iters=50, deadline=deadline - 100, verbose=False)
            tracker.update(a3)

    tracker.report("Phase 2 (LP-MD) complete")

    # === PHASE 3: Min-Norm Subgradient Polishing ===
    p3_deadline = min(time.monotonic() + 150, deadline - 100)
    print(f"\n{'='*60}", flush=True)
    print(f"PHASE 3: Min-Norm Subgradient (deadline={p3_deadline - time.monotonic():.0f}s)", flush=True)
    print(f"{'='*60}", flush=True)

    if tracker.best is not None:
        a = run_min_norm_subgradient(tracker.best, max_iters=5000, deadline=p3_deadline,
                                     lr_init=0.2, verbose=True)
        tracker.update(a)

    tracker.report("Phase 3 (Subgrad) complete")

    # === PHASE 4: Smooth Gradient + Restarts ===
    p4_deadline = min(time.monotonic() + 200, deadline - 50)
    print(f"\n{'='*60}", flush=True)
    print(f"PHASE 4: Smooth Gradient Restarts (deadline={p4_deadline - time.monotonic():.0f}s)", flush=True)
    print(f"{'='*60}", flush=True)

    if tracker.best is not None:
        n_best = tracker.best.size
        # Multiple noisy restarts
        for r in range(15):
            if time.monotonic() > p4_deadline - 10:
                break
            noise_scale = 0.02 * (1.0 + 0.5 * r / 15.0)
            a = tracker.best.copy() + rng.normal(0, noise_scale, n_best)
            a = np.maximum(a, 0.001)
            a = a / a.sum()

            a = run_smooth_gradient(a, max_iters=100, deadline=p4_deadline - 10,
                                    initial_tau=3, final_tau=40, lr_init=0.15, lr_min=0.02)
            tracker.update(a)

            # Quick LP polish
            if time.monotonic() < p4_deadline - 20:
                a_lp = run_lp_mirror_descent(a, max_iters=20, deadline=p4_deadline - 10, verbose=False)
                tracker.update(a_lp)

    tracker.report("Phase 4 (Restarts) complete")

    # === PHASE 5: Final LP polish on best ===
    if tracker.best is not None and time.monotonic() < deadline - 5:
        p5_deadline = deadline
        print(f"\n{'='*60}", flush=True)
        print(f"PHASE 5: Final LP Polish", flush=True)
        print(f"{'='*60}", flush=True)
        a = run_lp_mirror_descent(tracker.best, max_iters=100, deadline=p5_deadline, verbose=True)
        tracker.update(a)

    # Fallback
    if tracker.best is None:
        tracker.best = np.ones(128)
        tracker.best_score = compute_J(tracker.best)

    print(f"\n{'='*60}", flush=True)
    print(f"FINAL: J={tracker.best_score:.10f} n={tracker.best.size} "
          f"time={time.monotonic()-tracker.t_start:.0f}s", flush=True)
    print(f"{'='*60}", flush=True)

    return [float(x) for x in tracker.best]
```
