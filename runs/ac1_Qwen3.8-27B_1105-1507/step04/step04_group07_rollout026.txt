"""
Minimizes J(a) = 2*n*max(convolve(a,a))/(sum(a))^2 using:
1. KKT-based equalization: solve linear system from KKT conditions
2. Adaptive peak set identification
3. Support reduction for negative entries
4. Multi-start with diverse initializations
5. LSE gradient polishing
"""

import numpy as np
import time

def fast_conv(a):
    n = len(a)
    s = 1
    while s < 2 * n - 1:
        s <<= 1
    fa = np.fft.rfft(a, s)
    return np.fft.irfft(fa * fa, s)[:2 * n - 1]

def compute_J(a):
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))

def solve_kkt_system(n, K, S):
    """
    Solve the KKT linear system:
    For m in S: sum_{l in S, m+l in K} p_l = c
    sum_{l in S} p_l = 1
    
    Parameters:
    - n: length of sequence
    - K: set/list of active convolution indices (0 to 2n-2)
    - S: set/list of support indices (0 to n-1)
    
    Returns:
    - p: solution vector (length n, zeros outside S)
    - success: boolean
    """
    S = sorted(S)
    K_set = set(K)
    pS_idx = np.array(S)
    m = len(S)
    
    # Build matrix A (m x m) where A[i,j] = 1 if S[i] + S[j] in K
    A = np.zeros((m, m))
    for i, mi in enumerate(S):
        for j, lj in enumerate(S):
            if (mi + lj) in K_set:
                A[i, j] = 1.0
    
    # System: [A | -1] [p_S]   [0]
    #         [1^T | 0] [c]  = [1]
    M = np.zeros((m + 1, m + 1))
    M[:m, :m] = A
    M[:m, m] = -1.0
    M[m, :m] = 1.0
    M[m, m] = 0.0
    
    rhs = np.zeros(m + 1)
    rhs[m] = 1.0
    
    try:
        sol = np.linalg.solve(M, rhs)
    except np.linalg.LinAlgError:
        try:
            sol = np.linalg.lstsq(M, rhs, rcond=None)[0]
        except:
            return None, False
    
    p_S = sol[:m]
    c = sol[m]
    
    # Check validity
    if np.any(p_S < -1e-10):
        return None, False
    
    p = np.zeros(n)
    p[pS_idx] = np.maximum(p_S, 0)
    p /= p.sum()  # Renormalize
    
    return p, True

def kkt_optimize(a, max_iters=200, deadline=None, tracker=None):
    """
    KKT-based optimization.
    """
    a = np.maximum(a.astype(np.float64), 0.0)
    a /= a.sum()
    n = len(a)
    
    if tracker:
        tracker.update(a)
    
    for it in range(max_iters):
        if deadline and time.monotonic() > deadline:
            break
        
        b = fast_conv(a)
        b_max = b.max()
        
        # Find active peak set K (indices where b is close to max)
        # Try different tolerances
        best_result = None
        best_J = compute_J(a)
        
        for tol in [0.0, 0.01, 0.05, 0.1, 0.2, 0.3, 0.5]:
            K = list(np.where(b >= b_max * (1 - tol))[0])
            if len(K) < 2:
                K = list(np.argsort(b)[-max(2, n//10) :][::-1])
            
            # Support: where a > threshold
            S = list(np.where(a > 1e-10)[0])
            
            p, success = solve_kkt_system(n, K, S)
            if not success or p is None:
                continue
            
            # Check if peak is actually at K
            b_new = fast_conv(p)
            b_new_max = b_new.max()
            K_actual = set(np.where(b_new >= b_new_max * 0.99)[0])
            
            # Check consistency: actual peaks should be subset of or overlap with K
            overlap = len(K_actual & set(K))
            if overlap >= max(1, len(K) // 4):  # At least 25% overlap
                J_new = 2 * n * b_new_max / p.sum()**2
                if J_new < best_J:
                    best_J = J_new
                    best_result = p.copy()
        
        if best_result is not None:
            a = best_result
            if tracker:
                tracker.update(a)
        else:
            break
    
    if tracker:
        tracker.update(a)
    return a

def lse_gradient_descent(a, max_iters, deadline, tracker=None, tau_init=50.0):
    """LSE-smoothed gradient descent with momentum."""
    a = np.maximum(a.astype(np.float64), 1e-8)
    a /= a.sum()
    n = len(a)
    current_J = compute_J(a)
    
    if tracker:
        tracker.update(a)
    
    tau = tau_init
    lr = 2.0 / (n * tau)
    momentum = 0.8
    v = np.zeros(n)
    
    for it in range(max_iters):
        if deadline and time.monotonic() > deadline:
            break
        
        b = fast_conv(a)
        b_max = b.max()
        exp_b = np.exp(tau * (b - b_max))
        p_weights = exp_b / exp_b.sum()
        
        # Gradient of LSE w.r.t. a[m]:
        # d/da_m [sum_k w_k * (a*a)(k)] = sum_k w_k * 2 * a_{k-m} * 1{valid}
        # = 2 * (a * w_rev)(m) where w_rev is the reversed/separated weights
        # More precisely: sum_k w_k * (a_{k-m} + a_{m} * 1{...})
        # = 2 * sum_{k: valid} w_k * a_{k-m}
        # This is a cross-correlation of w with a
        
        # Compute: grad[m] = 2 * sum_{l=0}^{n-1} w[m+l] * a[l] (with w zero-padded)
        # Using FFT:
        s = 1
        while s < 3 * n:
            s <<= 1
        fp = np.fft.rfft(p_weights, s)
        fa = np.fft.rfft(a, s)
        corr = np.fft.irfft(fp * np.conj(fa), s)
        grad = 2.0 * corr[:n]
        
        # Also subtract the "cost" of increasing a (to maintain sum=1)
        # The Lagrangian gradient is grad - lambda * 1, where lambda = sum(grad * a)
        # For simplicity, just center the gradient
        grad_centered = grad - grad.mean()
        
        v = momentum * v + (1.0 - momentum) * grad_centered
        a_new = a - lr * v
        a_new = np.maximum(a_new, 1e-8)
        a_new /= a_new.sum()
        
        J_new = compute_J(a_new)
        if J_new < current_J - 1e-15:
            a = a_new
            current_J = J_new
            tau = min(500.0, tau * 1.02)
            lr = 2.0 / (n * tau)
        else:
            lr *= 0.7
            v *= 0.5
            if lr < 1e-18:
                break
        
        if tracker and it % 50 == 0:
            tracker.update(a)
    
    if tracker:
        tracker.update(a)
    return a

def lp_best_response(a, M, time_limit=5):
    """LP: maximize sum(g) s.t. a*g <= M, g >= 0."""
    from scipy.optimize import linprog
    from scipy.sparse import csr_matrix
    
    n = len(a)
    # Build constraint matrix: (a*g)(k) = sum_j a[j]*g[k-j] <= M
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        for j in range(js, je + 1):
            rows.append(k)
            cols.append(j)
            vals.append(a[k - j])
    
    A = csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                options={'presolve': True, 'time_limit': time_limit})
    if r.success and r.x is not None:
        g = np.maximum(r.x, 0.0)
        if g.sum() > 1e-10:
            return g / g.sum()
    return None

def run_lp_opt(a, max_iters, deadline, tracker=None, tighten=0.01):
    """LP-based best response iteration."""
    a = np.maximum(a.astype(np.float64), 1e-8)
    a /= a.sum()
    n = len(a)
    current_J = compute_J(a)
    
    if tracker:
        tracker.update(a)
    
    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        
        b = fast_conv(a)
        M = float(b.max())
        M_tight = M * (1.0 - tighten)
        
        g = lp_best_response(a, M_tight, time_limit=3)
        if g is None:
            g = lp_best_response(a, M, time_limit=3)
        if g is None:
            break
        
        # Line search
        best_a = None
        best_J = current_J
        for t in [0.3, 0.5, 0.7, 0.9, 1.0]:
            a_trial = (1.0 - t) * a + t * g
            a_trial = np.maximum(a_trial, 0.0)
            ss = a_trial.sum()
            if ss < 1e-10:
                continue
            a_trial /= ss
            J_trial = compute_J(a_trial)
            if J_trial < best_J - 1e-14:
                best_J = J_trial
                best_a = a_trial
        
        if best_a is None:
            break
        a = best_a
        current_J = best_J
        if tracker and it % 5 == 0:
            tracker.update(a)
    
    if tracker:
        tracker.update(a)
    return a
