Step function optimization using binomial-coefficient initialization and analytical
subgradient descent with LP refinement. Key insight: the sequence a_k = binom(2k,k)/4^k
has a flat-top linear convolution (first n values all equal), giving an initial
objective of ~pi/2. We then refine using subgradient descent with the analytical

import numpy as np
import time

def compute_conv(a):
    """Compute linear convolution conv(a,a) using FFT. Returns array of length 2n-1."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a.astype(np.float64), size)
    conv = np.fft.irfft(fa * fa, size)
    return conv[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 1e-10:
        return np.inf
    conv = compute_conv(a)
    max_b = float(np.max(conv))
    return 2.0 * n * max_b / (s * s)

    log_a = np.zeros(n)
    # log(binom(2k,k)/4^k) = lgamma(2k+1) - 2*lgamma(k+1) - k*log(4)
    from math import lgamma, log
    for i in range(n):
        if i == 0:
            log_a[i] = 0.0
        else:
            log_a[i] = lgamma(2*i+1) - 2*lgamma(i+1) - i*log(4)
    a = np.exp(log_a)
    s = np.sum(a)
    if s < 1e-10:
        a = np.ones(n)
        s = n
    return a / s


def compute_subgradient(a, top_k=10):
    """
    Compute the (approximate) subgradient of max(conv(a,a)) with respect to a.
    Uses top-k blending for robustness when multiple positions are near-maximum.
    Returns gradient vector of length n.
    """
    n = len(a)
    conv = compute_conv(a)
    M = np.max(conv)

    # Find positions near the maximum
    threshold = M - 1e-10 * M
    near_max = np.where(conv >= threshold)[0]

    if len(near_max) == 0:
        near_max = np.array([int(np.argmax(conv))])

    # Use top-k or all near-max (whichever is smaller)
    if len(near_max) > top_k:
        # Take the top-k by value
        conv_at_max = conv[near_max]
        order = np.argsort(-conv_at_max)[:top_k]
        near_max = near_max[order]

    # Compute subgradient as average over near-max positions
    grad = np.zeros(n)
    for k_star in near_max:
        k_star = int(k_star)
        # dc_k/da_j = 2 * a_{k-j} if 0 <= k-j < n, else 0
        idx = k_star - np.arange(n)
        valid = (idx >= 0) & (idx < n)
        grad[valid] += 2.0 * a[idx[valid]]

    grad /= len(near_max)
    return grad


def projected_subgradient(a, top_k=10):
    """
    Compute the projected subgradient of the objective (removing the scale-invariant component).
    """
    n = len(a)
    S = float(np.sum(a))
    if S < 1e-10:
        return np.zeros(n)

    # Gradient of max_c w.r.t. a
    grad_c = compute_subgradient(a, top_k=top_k)

    # Full gradient of F = 2n * max_c / S^2
    # dF/da_j = 2n * grad_c_j / S^2 - 4n * max_c / S^3
    # = (2n/S^2) * (grad_c_j - 2*max_c / S)
    # Since we only care about direction, use:
    # P_j = grad_c_j - mean(grad_c)  (projected onto tangent of sum=const)

    # Project: subtract mean to remove scale component
    P = grad_c - np.mean(grad_c)
    return P


def multiplicative_step(a, P, eta):
    """Apply multiplicative update: a_j <- a_j * exp(-eta * P_j), then renormalize."""
    n = len(a)
    update = -eta * P
    update = np.clip(update, -10, 10)
    a_new = a * np.exp(update)
    a_new = np.maximum(a_new, 1e-15)
    s = np.sum(a_new)
    if s > 1e-10:
        a_new = a_new / s
    return a_new


def subgradient_optimize(a, time_budget=60.0, top_k=10, eta_init=1.0):
    """Main subgradient optimization loop with adaptive step size."""
    n = len(a)
    a = a.copy().astype(np.float64)
    s = np.sum(a)
    if s < 1e-10:
        a = np.ones(n) / n
        s = 1.0
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    eta = eta_init / n  # Scale step size with n
    stall = 0
    patience = 100

    max_iters = int(time_budget * 200)
    for it in range(max_iters):
        if time.monotonic() - start > time_budget:
            break

        P = projected_subgradient(a, top_k=top_k)
        a_new = multiplicative_step(a, P, eta)
        obj_new = compute_obj(a_new)

        if obj_new < best_obj - 1e-12:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new
            stall = 0
        else:
            stall += 1
            if stall > patience:
                eta *= 0.7
                if eta < 1e-8:
                    break
                # Add noise to escape local minimum
                rng = np.random.default_rng(it * 7919 + 3)
                a = a * np.exp(rng.normal(0, 0.05, n))
                a = np.maximum(a, 1e-12)
                a = a / np.sum(a)
                stall = 0
                patience = max(patience // 2, 30)
                top_k = min(top_k + 1, 20)
        # Slowly decrease eta and increase top_k
        if it % 100 == 0 and it > 0:
            eta *= 0.95
            top_k = min(top_k + 1, 30)

    return best_a, best_obj


def build_toeplitz(a):
    """Build matrix A where (A @ g)[k] = (a*g)[k] for k=0..2n-2."""
    n = len(a)
    A = np.zeros((2 * n - 1, n), dtype=np.float64)
    for j in range(n):
        k_start = j
        k_end = min(j + n, 2 * n - 1)
        A[k_start:k_end, j] = a[:k_end - k_start]
    return A


def solve_lp_minpeak(a, time_limit=10):
    """Solve LP: min M s.t. (a*g)[k] <= M for all k, sum(g)=1, g>=0."""
    n = len(a)
    A = build_toeplitz(a)

    # Formulation: min M
    # Variables: g_0,...,g_{n-1}, M
    # Constraints: A@g - M <= 0 (2n-1 constraints)
    #              sum(g) = 1
    #              g >= 0, M >= 0

    c = np.zeros(n + 1)
    c[-1] = 1.0

    A_ub = np.zeros((2 * n - 1, n + 1))
    A_ub[:, :n] = A
    A_ub[:, n] = -1.0
    b_ub = np.zeros(2 * n - 1)

    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])

    bounds = [(0.0, None)] * n + [(None, None)]

    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': time_limit})
        if res.status == 0:
            g = np.maximum(res.x[:n], 0)
            s = np.sum(g)
            if s < 1e-10:
                return None, None
            g = g / s
            return g, float(res.x[n])
    except Exception:
        pass

    # Try cvxpy
    if HAS_CVXPY:
        try:
            g_var = cp.Variable(n, nonneg=True)
            M_var = cp.Variable()
            constraints = [A @ g_var <= M_var, cp.sum(g_var) == 1]
            prob = cp.Problem(cp.Minimize(M_var), constraints)
            for solver in [cp.CLARABEL, cp.SCS, cp.ECOS]:
                try:
                    prob.solve(solver=solver, time_limit=int(time_limit))
                    break
                except Exception:
                    continue
            if g_var.value is not None:
                g = np.maximum(g_var.value, 0)
                s = np.sum(g)
                if s > 1e-10:
                    return g / s, float(M_var.value)
        except Exception:
            pass

    return None, None


def alternating_lp_optimize(a, time_budget=60.0):
    """Alternating LP optimization: alternate between fixing a and optimizing g."""
    n = len(a)
    a = a.copy().astype(np.float64)
    s = np.sum(a)
    if s < 1e-10:
        a = np.ones(n) / n
        s = 1.0
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    stall = 0

    for it in range(200):
        if time.monotonic() - start > time_budget - 5:
            break

        g, M_opt = solve_lp_minpeak(a, time_limit=8)
        if g is None:
            break

        # Line search on the convex combination
        best_t = 0.0
        best_obj_ls = best_obj
        for t in [0.1, 0.3, 0.5, 0.7, 1.0]:
            a_try = (1.0 - t) * a + t * g
            a_try = np.maximum(a_try, 0)
            s_try = np.sum(a_try)
            if s_try < 1e-10:
                continue
            a_try = a_try / s_try
            obj_try = compute_obj(a_try)
            if obj_try < best_obj_ls:
                best_obj_ls = obj_try
                best_t = t

        if best_t > 0 and best_obj_ls < best_obj - 1e-12:
            a = (1.0 - best_t) * a + best_t * g
            a = np.maximum(a, 0)
            a = a / np.sum(a)
            best_obj = best_obj_ls
            best_a = a.copy()
            stall = 0
        else:
            stall += 1
            if stall > 15:
                break
            # Small step anyway to make progress
            a = 0.95 * a + 0.05 * g
            a = np.maximum(a, 0)
            a = a / np.sum(a)

    return best_a, best_obj


def full_optimize(a, time_budget=120.0):
    """Full optimization pipeline: subgradient -> LP -> subgradient -> fine tune."""
    n = len(a)
    a = a.copy().astype(np.float64)
    s = np.sum(a)
    if s < 1e-10:
        a = np.ones(n) / n
        s = 1.0
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    # Phase 1: Subgradient (fast)
    remaining = time_budget - (time.monotonic() - start)
    if remaining > 10:
        sg_time = min(30.0, remaining * 0.3)
        a_new, obj_new = subgradient_optimize(a, time_budget=sg_time, top_k=10, eta_init=2.0)
        if obj_new < best_obj:
            best_obj = obj_new
            best_a = a_new.copy()
        a = a_new  # Continue from here even if not better (might help LP)

    # Phase 2: LP refinement
    remaining = time_budget - (time.monotonic() - start)
    if remaining > 15:
        lp_time = min(40.0, remaining * 0.35)
        a_new, obj_new = alternating_lp_optimize(best_a, time_budget=lp_time)
        if obj_new < best_obj:
            best_obj = obj_new
            best_a = a_new.copy()

    # Phase 3: More subgradient from current best
    remaining = time_budget - (time.monotonic() - start)
    if remaining > 10:
        sg_time = min(30.0, remaining * 0.3)
        a_new, obj_new = subgradient_optimize(best_a, time_budget=sg_time, top_k=15, eta_init=0.5)
        if obj_new < best_obj:
            best_obj = obj_new
            best_a = a_new.copy()

    # Phase 4: Final LP polish
    remaining = time_budget - (time.monotonic() - start)
    if remaining > 15:
        lp_time = min(20.0, remaining * 0.4)
        a_new, obj_new = alternating_lp_optimize(best_a, time_budget=lp_time)
        if obj_new < best_obj:
            best_obj = obj_new
            best_a = a_new.copy()

    return best_a, best_obj


def local_search(a, time_budget=60.0):
    """Local search: perturb and re-optimize."""
    n = len(a)
    a = a.copy().astype(np.float64)
    s = np.sum(a)
    if s < 1e-10:
        a = np.ones(n) / n
        s = 1.0
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    attempt = 0

    while time.monotonic() - start < time_budget - 10:
        attempt += 1
        remaining = time_budget - (time.monotonic() - start)
        if remaining < 15:
            break

        rng = np.random.default_rng(attempt * 31337 + 42)

        # Different perturbation strategies
        strategy = attempt % 4
        if strategy == 0:
            # Multiplicative noise
            a_pert = best_a * np.exp(rng.normal(0, 0.05, n))
        elif strategy == 1:
            # Perturb a random subset
            mask = rng.random(n) < 0.3
            a_pert = best_a.copy()
            a_pert[mask] *= np.exp(rng.normal(0, 0.1, mask.sum()))
        elif strategy == 2:
            # Smooth perturbation
            noise = rng.normal(0, 0.03, n)
            win = max(3, n // 20)
            kernel = np.ones(win) / win
            noise = np.convolve(noise, kernel, mode='same')
            a_pert = best_a * np.exp(noise)
        else:
            # Targeted: perturb near the peak
            conv = compute_conv(best_a)
            peak_idx = int(np.argmax(conv))
            # Perturb elements that contribute to the peak
            center = peak_idx // 2
            width = max(5, n // 10)
            positions = np.arange(max(0, center - width), min(n, center + width + 1))
            a_pert = best_a.copy()
            a_pert[positions] *= np.exp(rng.normal(0, 0.08, len(positions)))

        a_pert = np.maximum(a_pert, 1e-12)
        a_pert = a_pert / np.sum(a_pert)

        # Quick optimization
        sub_time = min(15.0, remaining * 0.4)
        a_opt, obj_opt = full_optimize(a_pert, time_budget=sub_time)

        if obj_opt < best_obj:
            best_obj = obj_opt
            best_a = a_opt.copy()

    return best_a, best_obj


def tapered_binomial_init(n, alpha=1.0):
    """Binomial init with tapering at the end to reduce edge effects."""
    k = np.arange(n, dtype=np.float64)
    from math import lgamma, log
    log_a = np.zeros(n)
    for i in range(n):
        if i == 0:
            log_a[i] = 0.0
        else:
            log_a[i] = lgamma(2*i+1) - 2*lgamma(i+1) - i*log(4)
    a = np.exp(log_a)
    # Apply taper to last 20% of the sequence
    taper_start = int(0.8 * n)
    if taper_start < n:
        for i in range(taper_start, n):
            frac = (i - taper_start) / (n - 1 - taper_start + 1)
            a[i] *= (1.0 - alpha * frac * frac)
    a = np.maximum(a, 0)
    s = np.sum(a)
    if s < 1e-10:
        a = np.ones(n) / n
    else:
        a = a / s
    return a


def propose_candidate(seed=42, budget_s=1500):
    """

    a flat-top convolution) and refines with analytical subgradient descent
    and LP-based optimization. Explores multiple sequence lengths.
    """
    start_time = time.monotonic()
    deadline = start_time + float(budget_s) - 30.0

    best_seq = None
    best_score = np.inf

    # Try to load parent
    parent = None
    parent_n = 330
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64)
        parent = np.clip(parent, 0, 1000.0)
        if parent.size > 0 and np.isfinite(parent).all():
            parent_score = compute_obj(parent)
            print(f"Parent (n={parent.size}): obj={parent_score:.6f}")
            if parent_score < best_score:
                best_score = parent_score
                best_seq = parent.copy()
            parent_n = len(parent)
    except Exception:
        pass

    if best_seq is None:
        # Start with binomial init
        n_init = 330
        best_seq = binomial_init(n_init)
        best_score = compute_obj(best_seq)
        print(f"Binomial init (n={n_init}): obj={best_score:.6f}")
        parent_n = n_init

    print(f"Initial best: {best_score:.6f} (n={len(best_seq)})")

    # Phase 1: Intensive optimization of the best starting point
    print("\n=== Phase 1: Optimize best starting point ===")
    remaining = deadline - time.monotonic()
    p1_budget = min(remaining * 0.30, 300.0)
    a_opt, obj_opt = full_optimize(best_seq, time_budget=p1_budget)
    if obj_opt < best_score:
        best_score = obj_opt
        best_seq = a_opt.copy()
        print(f"  After full_optimize: {best_score:.6f}")

    # Phase 1b: Local search
    remaining = deadline - time.monotonic()
    if remaining > 30:
        ls_budget = min(remaining * 0.15, 150.0)
        a_opt, obj_opt = local_search(best_seq, time_budget=ls_budget)
        if obj_opt < best_score:
            best_score = obj_opt
            best_seq = a_opt.copy()
            print(f"  After local search: {best_score:.6f}")

    # Phase 2: Try binomial initials with different n and tapers
    print("\n=== Phase 2: Explore n values with binomial init ===")
    n_candidates = []
    # Near the parent
    for delta in range(-20, 21, 5):
        n_val = parent_n + delta
        if 150 <= n_val <= 800:
            n_candidates.append(n_val)
    # Some other values
    for n_val in [200, 250, 400, 500]:
        if n_val not in n_candidates:
            n_candidates.append(n_val)

    # Sort by proximity to parent_n (most likely to be good)
    n_candidates = sorted(set(n_candidates), key=lambda x: abs(x - parent_n))

    remaining = deadline - time.monotonic()
    if remaining > 30:
        # Allocate time: more time for n values close to current best
        for n_val in n_candidates:
            remaining = deadline - time.monotonic()
            if remaining < 25:
                break
            # Skip if we already have good results at this n
            if len(best_seq) == n_val and best_score < 1.505:
                continue

            proximity_factor = 1.0 / (1 + abs(n_val - parent_n) / 50.0)
            alloc = min(remaining * 0.3 * proximity_factor, 80.0)
            alloc = max(alloc, 15.0)

            if time.monotonic() > deadline - 20:
                break

            # Try standard binomial init
            a_init = binomial_init(n_val)
            obj_init = compute_obj(a_init)

            if obj_init < best_score - 0.001:

                a_opt, obj_opt = full_optimize(a_init, time_budget=alloc)
                if obj_opt < best_score:

                    best_seq = a_opt.copy()
                    print(f"  ** UPDATE: n={n_val}, obj={best_score:.6f} **")
            elif obj_init < best_score + 0.01:
                # Close to best, worth trying
                a_opt, obj_opt = full_optimize(a_init, time_budget=min(alloc, 40.0))
                if obj_opt < best_score:
                    best_score = obj_opt
                    best_seq = a_opt.copy()
                    print(f"  ** UPDATE: n={n_val}, obj={best_score:.6f} **")

                    if obj_taper < best_score + 0.005:

                        if obj_opt < best_score:

                            best_seq = a_opt.copy()
                            print(f"  ** UPDATE (tapered alpha={alpha}): n={n_val}, obj={best_score:.6f} **")

        remaining = deadline - time.monotonic()
        if remaining > 20:

    # Phase 5: Try a few very different n values if time allows
    if best_seq is not None:

        if remaining > 40:

            for n_val in [280, 350, 450, 550]:
                if time.monotonic() > deadline - 20:

                if abs(n_val - len(best_seq)) < 10:
                    continue
                a_init = binomial_init(n_val)
                obj_init = compute_obj(a_init)
                if obj_init < best_score + 0.02:
                    alloc = min(remaining * 0.25, 50.0)
                    a_opt, obj_opt = full_optimize(a_init, time_budget=alloc)
                    if obj_opt < best_score:
                        best_score = obj_opt
                        best_seq = a_opt.copy()
                        print(f"  ** UPDATE: n={n_val}, obj={best_score:.6f} **")

    # Final: validate and return
    if best_seq is None:

    best_seq = np.clip(best_seq, 0.0, 1000.0)

    print(f"\n=== FINAL: n={len(best_seq)}, score={best_score:.6f}, time={elapsed:.1f}s ===")

    return result
