"""
Minimizes J(a) = 2*n*max(convolve(a,a))/(sum(a))^2 using exact coordinate descent
with golden-section line search, combined with smooth gradient exploration and
multi-length resampling.
"""

import numpy as np
import time


def fast_conv(a):
    """Full self-convolution via FFT."""
    n = len(a)
    if n == 1:
        return np.array([a[0] * a[0]])
    s = 1
    while s < 2 * n - 1:
        s <<= 1
    fa = np.fft.rfft(a, s)
    return np.fft.irfft(fa * fa, s)[:2 * n - 1]


def compute_J(a):
    """Compute J(a) = 2*n*max(conv(a,a))/(sum(a))^2."""
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))


def eval_J_coord(i, x, a, b, S, n):
    """Evaluate J if a[i] were set to x (given current a, b, S)."""
    old_x = a[i]
    delta = x - old_x
    if abs(delta) < 1e-15:
        max_b = b.max()
        return 2.0 * n * max_b / (S * S)
    
    lo = i
    hi = min(i + n - 1, 2 * n - 2)
    
    # Affected positions: k in [i, i+n-1]
    indices = np.arange(lo, hi + 1)
    contrib = a[indices - i]  # a[k-i] for k in affected range
    b_aff = b[indices].copy() + 2.0 * delta * contrib
    
    # Fix the k=2i position (quadratic term)
    if 2 * i >= lo and 2 * i <= hi:
        b_aff[2 * i - lo] = b[2 * i] + delta * (x + old_x)
    
    max_aff = b_aff.max()
    
    # Unaffected max
    max_unaff = -np.inf
    if lo > 0:
        max_unaff = max(max_unaff, b[:lo].max())
    if hi < 2 * n - 2:
        max_unaff = max(max_unaff, b[hi + 1:].max())
    
    max_b = max(max_aff, max_unaff)
    sum_new = S + delta
    if sum_new < 1e-10:
        return 1e10
    return 2.0 * n * max_b / (sum_new * sum_new)


def golden_section_eval(f, lo, hi, n_iter=40):
    """Golden section minimization of f on [lo, hi]. Returns optimal x."""
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    a_pt = lo
    b_pt = hi
    c_pt = b_pt - gr * (b_pt - a_pt)
    d_pt = a_pt + gr * (b_pt - a_pt)
    fc = f(c_pt)
    fd = f(d_pt)
    
    best_x = c_pt if fc < fd else d_pt
    best_f = min(fc, fd)
    
    for _ in range(n_iter):
        if fc < fd:
            b_pt = d_pt
            d_pt = c_pt
            fd = fc
            c_pt = b_pt - gr * (b_pt - a_pt)
            fc = f(c_pt)
        else:
            a_pt = c_pt
            c_pt = d_pt
            fc = fd
            d_pt = a_pt + gr * (b_pt - a_pt)
            fd = f(d_pt)
        
        cur_best = c_pt if fc < fd else d_pt
        cur_best_f = min(fc, fd)
        if cur_best_f < best_f:
            best_f = cur_best_f
            best_x = cur_best
    
    return best_x


def coordinate_descent_pass(a, b, S, n, order, deadline, n_gs=35, stall_threshold=1e-12):
    """One pass of coordinate descent. Returns updated a, b, S, and number of changes."""
    changes = 0
    total_improvement = 0.0
    
    for i in order:
        if time.monotonic() > deadline:
            break
        
        old_x = a[i]
        
        # Determine search range
        upper = max(3.0 * old_x if old_x > 0 else 0.1, 2.0 * np.max(a), 0.05)
        # But don't go too far
        upper = min(upper, 5.0 * max(np.mean(a), 1e-6) + 0.5)
        
        if old_x < 1e-8:
            # Singleton: try a few values
            candidates = [0.0, 0.001, 0.01, np.mean(a) * 0.5, np.mean(a)]
            best_x = old_x
            best_j = eval_J_coord(i, old_x, a, b, S, n)
            for x_c in candidates:
                j_c = eval_J_coord(i, x_c, a, b, S, n)
                if j_c < best_j:
                    best_j = j_c
                    best_x = x_c
        else:
            # Golden section search on [0, upper]
            # First do a coarse grid to bracket
            n_grid = 12
            grid = np.linspace(0.0, upper, n_grid)
            best_grid_x = old_x
            best_grid_j = 2.0 * n * b.max() / (S * S)
            for x_g in grid:
                j_g = eval_J_coord(i, x_g, a, b, S, n)
                if j_g < best_grid_j:
                    best_grid_j = j_g
                    best_grid_x = x_g
            
            # Refine with golden section around the best grid point
            idx = int(np.argmin([eval_J_coord(i, x_g, a, b, S, n) for x_g in grid]))
            # Actually this is wasteful, let's just use the best grid point
            # and do fine golden section in a small neighborhood
            if 0 < best_grid_x < upper:
                half_step = upper / n_grid
                lo_gs = max(0.0, best_grid_x - half_step)
                hi_gs = min(upper, best_grid_x + half_step)
                best_x = golden_section_eval(
                    lambda x: eval_J_coord(i, x, a, b, S, n),
                    lo_gs, hi_gs, n_iter=n_gs
                )
            else:
                best_x = best_grid_x
        
        delta = best_x - old_x
        if abs(delta) > max(1e-12, 1e-10 * old_x):
            # Apply update
            a[i] = best_x
            # Update b
            lo = i
            hi = min(i + n - 1, 2 * n - 2)
            if 2 * i <= 2 * n - 2:
                indices = np.arange(lo, hi + 1)
                mask = np.ones(len(indices), dtype=bool)
                two_i_pos = 2 * i - lo
                mask[two_i_pos] = False
                b[indices[mask]] += 2.0 * delta * a[indices[mask] - i]
                b[2 * i] += 2.0 * old_x * delta + delta * delta
            else:
                indices = np.arange(lo, hi + 1)
                b[indices] += 2.0 * delta * a[indices - i]
            S += delta
            changes += 1
            total_improvement += delta
    
    return a, b, S, changes, total_improvement


def resample_seq(src, dest_n):
    """Resample 1D sequence to new length via linear interpolation."""
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    result = np.interp(t_new, t_old, src)
    return np.maximum(result, 0.0)


def smooth_gradient_step(a, beta=5.0, lr=0.5):
    """One step of smooth multiplicative gradient descent."""
    n = len(a)
    b = fast_conv(a)
    b_max = b.max()
    shifted = beta * (b - b_max)
    shifted = np.clip(shifted, -500, 500)
    exp_shifted = np.exp(shifted)
    Z = exp_shifted.sum()
    p = exp_shifted / Z
    
    # Gradient of smooth max w.r.t. a: 2 * (a * p_flip)
    # where p_flip is p reversed (because convolution)
    p_flip = p[::-1]
    # grad[i] = 2 * sum_k p[k] * a[k-i] (where valid)
    # This is 2 * (p_flip * a)[i] in the right range
    # Actually: d/d a_i max_b ≈ 2 * sum_k p[k] * (a[k-i] + a[i-k]) ...
    # For self-convolution: d b_k / d a_i = a[k-i] + a[i-k] (but for self-conv it's 2*a[k-i] when k≠2i, and 2*a[i] when k=2i)
    # More precisely: d b_k / d a_i = a[k-i] (if 0<=k-i<n) + a[k-i] (if 0<=k-i<n, same term for self-conv)
    # Wait, b_k = sum_j a_j * a_{k-j}. d b_k / d a_i = a_{k-i} (from j=i term) + a_{k-i} (from k-j=i, i.e., j=k-i term)
    # = 2*a_{k-i} if k-i is valid, minus a[i] if k-i = i (i.e., k=2i, to avoid double counting)
    # Actually: d b_k / d a_i = a[k-i] + (a[k-i] if k-i != i else 0) 
    # No wait. b_k = sum_j a_j * a_{k-j}. The terms involving a_i are:
    # j = i: a_i * a_{k-i}
    # k-j = i, i.e., j = k-i: a_{k-i} * a_i
    # If i != k-i (i.e., k != 2i): both terms exist, giving 2*a_i*... no wait, we're differentiating w.r.t. a_i.
    # d/d a_i [a_i * a_{k-i}] = a_{k-i}
    # d/d a_i [a_{k-i} * a_i] = a_{k-i}
    # So d b_k / d a_i = 2 * a_{k-i} (if k-i is valid and k != 2i)
    # If k = 2i: the term is a_i * a_i, so d/d a_i = 2 * a_i
    # But that's the same formula: 2 * a_{k-i} = 2 * a_{2i-i} = 2 * a_i. ✓
    
    # So d b_k / d a_i = 2 * a_{k-i} (if 0 <= k-i < n, i.e., i <= k <= i+n-1)
    # d/d a_i [sum_k p_k * b_k] = sum_{k: i<=k<=i+n-1} p_k * 2 * a_{k-i}
    # = 2 * sum_{j: 0<=j<n, i<=i+j<=i+n-1} p_{i+j} * a_j  (substituting j = k-i)
    # = 2 * sum_{j=0}^{n-1} p[i+j] * a_j  (where p is indexed from 0 to 2n-2)
    
    # This is a cross-correlation! grad[i] = 2 * sum_j p[i+j] * a_j
    # = 2 * (a * p_reversed_correlation)[i]
    
    # Let me compute it via convolution:
    # grad[i] = 2 * sum_{j=0}^{n-1} a[j] * p[i+j]
    # This is the cross-correlation of a with p, evaluated at lag i.
    # = (a * p_flipped)[i + n - 1] where p_flipped = p[::-1]
    # Let's verify: (a * p_flipped)[m] = sum_j a[j] * p_flipped[m-j] = sum_j a[j] * p[2n-2-(m-j)]
    # We want sum_j a[j] * p[i+j]. Set 2n-2-(m-j) = i+j => m-j = 2n-2-i-j => m = 2n-2-i
    # So grad[i] = 2 * (a * p_flipped)[2n-2-i]
    
    p_flipped = p[::-1]
    cross = np.convolve(a, p_flipped, mode='full')
    grad = 2.0 * cross[2*n-2 : n-1:-1]  # cross[2n-2-i] for i=0,...,n-1
    # Actually: cross has length n + (2n-1) - 1 = 3n-1
    # cross[m] for m = 2n-2, 2n-3, ..., n-1 correspond to i = 0, 1, ..., n-1
    # cross[2n-2-i] for i in 0..n-1: indices are 2n-2, 2n-3, ..., n-1
    # So grad[i] = 2 * cross[2n-2-i]
    grad = 2.0 * cross[n-1:2*n-1][::-1]
    
    # Normalization factor
    S = a.sum()
    # J = 2n * max_b / S^2
    # dJ/d a_i = 2n * (S * grad_b[i] - 2 * max_b * 1) / S^3  (approximately, using smooth max)
    # For multiplicative updates, we use: a_i <- a_i * exp(-lr * grad_normalized[i])
    # where grad_normalized accounts for the ratio
    
    # Simpler: use the subgradient of max_b
    # a_i <- a_i * exp(-lr * 2 * (p * a_correlation)[i] / (S^2 / (2n)))
    # = a_i * exp(-lr * 2n * 2 * correl[i] / S^2)
    
    # Even simpler: just use the gradient of the smooth objective
    # F = 2n * g_smooth / S^2
    # dF/d a_i = 2n * (S * grad_smooth[i] - 2 * g_smooth) / S^3
    
    g_smooth = b_max + np.log(Z) / beta
    grad_F = 2.0 * n * (S * grad - 2.0 * g_smooth) / (S ** 3)
    
    # Multiplicative update
    step = np.exp(-lr * grad_F * S ** 2 / (2.0 * n))
    step = np.clip(step, 0.1, 10.0)
    a_new = a * step
    a_new = np.maximum(a_new, 0.0)
    
    return a_new

"""
Minimizes J(a) = 2*n*max(convolve(a,a))/(sum(a))^2 using exact coordinate descent
with adaptive golden-section search, smooth gradient exploration, and multi-length
resampling. The coordinate descent is the primary workhorse (much faster than LP),
with periodic exploration phases to escape local minima.

import numpy as np
import time


def fast_conv(a):
    """Full self-convolution via FFT, O(n log n)."""
    n = len(a)
    if n == 1:
        return np.array([a[0] * a[0]])
    s = 1
    while s < 2 * n - 1:
        s <<= 1
    fa = np.fft.rfft(a, s)
    return np.fft.irfft(fa * fa, s)[:2 * n - 1]


def compute_J(a):
    """Compute J(a) = 2*n*max(conv(a,a))/(sum(a))^2."""
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))


def eval_J_coord(i, x, a, b, S, n):
    """Evaluate J if a[i] were set to x, given current a, b, S. O(n)."""
    old_x = a[i]
    delta = x - old_x
    if abs(delta) < 1e-15:
        max_b = b.max()
        return 2.0 * n * max_b / (S * S)

    lo = i
    hi = min(i + n - 1, 2 * n - 2)

    indices = np.arange(lo, hi + 1)
    contrib = a[indices - i]
    b_aff = b[indices] + 2.0 * delta * contrib

    if 2 * i >= lo and 2 * i <= hi:
        b_aff[2 * i - lo] = b[2 * i] + delta * (x + old_x)

    max_aff = b_aff.max()

    max_unaff = -np.inf
    if lo > 0:
        max_unaff = max(max_unaff, b[:lo].max())
    if hi < 2 * n - 2:
        max_unaff = max(max_unaff, b[hi + 1:].max())

    max_b = max(max_aff, max_unaff)
    sum_new = S + delta
    if sum_new < 1e-10:
        return 1e10
    return 2.0 * n * max_b / (sum_new * sum_new)


def golden_section_1d(f, lo, hi, n_iter=30):
    """Golden section minimization. Returns optimal x in [lo, hi]."""
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    a_pt = lo
    b_pt = hi
    c_pt = b_pt - gr * (b_pt - a_pt)
    d_pt = a_pt + gr * (b_pt - a_pt)
    fc = f(c_pt)
    fd = f(d_pt)
    best_x = c_pt if fc < fd else d_pt
    best_f = min(fc, fd)

    for _ in range(n_iter):
        if fc < fd:
            b_pt = d_pt
            d_pt = c_pt
            fd = fc
            c_pt = b_pt - gr * (b_pt - a_pt)
            fc = f(c_pt)
        else:
            a_pt = c_pt
            c_pt = d_pt
            fc = fd
            d_pt = a_pt + gr * (b_pt - a_pt)
            fd = f(d_pt)
        cur_best_f = min(fc, fd)
        if cur_best_f < best_f:
            best_f = cur_best_f
            best_x = c_pt if fc < fd else d_pt

    return best_x


def coord_optimize(i, a, b, S, n, range_width, n_gs_iter=28):

def apply_coord_update(i, old_x, new_x, a, b, S, n):
    """Apply coordinate update in-place. Returns new S."""
    delta = new_x - old_x
    if abs(delta) < 1e-14:
        return S
    a[i] = new_x
    lo = i
    hi = min(i + n - 1, 2 * n - 2)
    indices = np.arange(lo, hi + 1)
    if 2 * i <= 2 * n - 2:
        mask = np.ones(len(indices), dtype=bool)
        mask[2 * i - lo] = False
        b[indices[mask]] += 2.0 * delta * a[indices[mask] - i]
        b[2 * i] += 2.0 * old_x * delta + delta * delta
    else:
        b[indices] += 2.0 * delta * a[indices - i]
    return S + delta


def resample_seq(src, dest_n):
    """Resample 1D sequence to new length via linear interpolation."""
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    result = np.interp(t_new, t_old, src)
    return np.maximum(result, 0.0)

                if abs(bx - ox) > 1e-13:
                    S = apply_coord_update(i, ox, bx, a, b, S, n)
            j_final = 2.0 * n * b.max() / (S * S)
            if j_final < best_score:
                best_score = j_final
                best_a = a.copy()
                len_best[n] = a.copy()
                print(f"    Final polish: J={best_score:.8f}", flush=True)

    # Final validation
    final_score = float(evaluate_sequence(best_a.tolist()))
    if np.isfinite(final_score) and final_score < best_score:
        best_score = final_score

    print(f"\n{'='*50}", flush=True)
    print(f"FINAL: J={best_score:.8f} n={best_a.size} "
          f"time={time.monotonic()-t_start:.1f}s", flush=True)
    print(f"{'='*50}", flush=True)

    result = np.clip(best_a, 0.0, 1000.0)
    if result.sum() < 0.01:
        result = np.ones(128)
    return [float(x) for x in result]
