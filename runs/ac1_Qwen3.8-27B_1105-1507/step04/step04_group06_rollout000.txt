"""

import numpy as np
from scipy.optimize import linprog, minimize
from scipy.sparse import csr_matrix
import time

# Cache for LP matrix structure
_structure_cache = {}


def fast_conv(a):
    """Full self-convolution via FFT."""
    n = len(a)
    s = 1
    while s < 2 * n - 1:
        s <<= 1
    fa = np.fft.rfft(a, s)
    return np.fft.irfft(fa * fa, s)[:2 * n - 1]


def compute_J(a):
    """Compute J(a) = 2*n*max(conv(a,a))/(sum(a))^2."""
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))


def get_A_structure(n):
    """Get or pre-compute the (row, col) arrays for the constraint matrix."""
    if n not in _structure_cache:
        rows = []
        cols = []
        for k in range(2 * n - 1):
            jl = max(0, k - n + 1)
            jh = min(n - 1, k)
            for j in range(jl, jh + 1):
                rows.append(k)
                cols.append(j)
        _structure_cache[n] = (np.array(rows, dtype=np.int32),
                               np.array(cols, dtype=np.int32))
    return _structure_cache[n]


def build_A_sparse(a):
    """Build sparse constraint matrix A where (A @ g)_k = (a*g)_k."""
    n = len(a)
    rows_arr, cols_arr = get_A_structure(n)
    vals = a[rows_arr - cols_arr]
    return csr_matrix((vals, (rows_arr, cols_arr)), shape=(2 * n - 1, n))


def smooth_max_and_grad(a, beta):
    """
    Compute smooth approximation of max(conv(a,a)) and its gradient w.r.t. a.
    Uses log-sum-exp trick.
    """
    n = len(a)
    b = fast_conv(a)
    b_max = float(b.max())

    shifted = np.clip(shifted, -500, 0)
    exp_shifted = np.exp(shifted)
    Z = float(exp_shifted.sum())
    if Z < 1e-300:
        p = np.zeros(2 * n - 1)
        p[np.argmax(b)] = 1.0
    else:
        p = exp_shifted / Z
    g_val = b_max + np.log(max(Z, 1e-300)) / beta

    # Gradient: d(max c)/da_i = sum_k p_k * dc_k/da_i = sum_k p_k * 2*a_{k-i}

    # = 2 * sum_{m=0}^{n-1} p_{i+m} * a_m (where i+m < 2n-1)
    # = 2 * (a * p_flipped_partial)[i]

    # grad_i = 2 * sum_m a_m * p_{i+m}

    # Actually, let's use: for each i, grad_i = 2 * sum_{m=0}^{n-1} a_m * p[m+i] (if m+i < 2n-1)
    # This is 2 * (a flipped cross-correlated with p)
    
    # Using FFT: grad = 2 * ifft(fft(a) * fft(p) conjugated)... 
    # Actually simpler: grad_i = 2 * sum_m a_m * p_{m+i}
    # = 2 * (a_rev * p)_i where a_rev is a reversed...

    # p has length 2n-1, a has length n
    # We want for each i in [0, n-1]: sum_{m=0}^{n-1} a_m * p[m+i]
    # This is like a cross-correlation.
    
    # Using np.correlate: np.correlate(p, a, 'valid')... no that's not quite right.
    # Let's use: flip a to get a_flip[m] = a[n-1-m]
    # Then sum_m a_m * p[m+i] = sum_m a_flip[n-1-m] * p[m+i]
    # Let j = n-1-m: sum_j a_flip[j] * p[n-1-j+i] = (a_flip * p)[n-1+i]

    grad = 2.0 * full_conv[n-1:n-1+n]

def F_smooth_and_grad(a, beta):
    """Compute F(a) = 2n*smooth_max(conv(a,a))/(sum(a))^2 and its gradient."""
    n = len(a)
    s = float(a.sum())
    if s < 1e-10:
        return 1e10, np.zeros(n)
    g_val, grad_g = smooth_max_and_grad(a, beta)
    F_val = 2.0 * n * g_val / (s * s)
    # dF/da_i = 2n * (s * grad_g_i - 2 * g_val * 1) / s^3
    #         = 2n * (s * grad_g_i - 2*g_val) / s^3
    grad_F = 2.0 * n * (s * grad_g - 2.0 * g_val) / (s ** 3)
    return F_val, grad_F


def adam_smooth_optimize(a0, betas, deadline, max_iters_per_beta=None):
    """
    Adam optimization with log-sum-exp smoothing and beta annealing.
    """
    a = np.maximum(a0.astype(np.float64), 1e-6)
    a = a / a.sum()
    n = len(a)
    
    if max_iters_per_beta is None:
        max_iters_per_beta = 200
    
    for beta in betas:
        if time.monotonic() > deadline:
            break
        
        # Adam state
        m = np.zeros(n)
        v = np.zeros(n)
        t = 0
        lr = 0.1 / (n ** 0.5)  # Scale learning rate with n
        
        for it in range(max_iters_per_beta):
            if time.monotonic() > deadline:
                break
            
            t += 1
            F_val, grad = F_smooth_and_grad(a, beta)
            if not np.isfinite(F_val):
                break
            
            # Adam update
            lr_t = lr * (0.5 + 0.5 * t / min(t, 50))  # Warmup
            m = 0.9 * m + 0.1 * grad
            v = 0.999 * v + 0.001 * grad ** 2
            m_hat = m / (1 - 0.9 ** t)
            v_hat = v / (1 - 0.999 ** t)
            a_new = a - lr_t * m_hat / (np.sqrt(v_hat) + 1e-8)
            
            # Project onto {x >= 0, sum(x) = 1}
            a_new = np.maximum(a_new, 0)
            s = a_new.sum()
            if s < 1e-10:
                a_new = a.copy()
            else:
                a_new = a_new / s
            
            a = a_new
        
    return a


def lp_mirror_descent_v2(a, max_iters, deadline, tracker=None, n_line_evals=20,
                         stall_patience=30):
    """
    Improved LP mirror descent with:
    - Parabolic line search refinement
    - Better warm-starting
    """
    a = np.maximum(a.astype(np.float64), 1e-8)
    a = a / a.sum()
    n = len(a)
    current_J = compute_J(a)
    if tracker is not None:
        tracker.update(a)

    stall_count = 0
    best_J_so_far = current_J
    best_a_so_far = a.copy()
    
    # Pre-build structure
    rows_arr, cols_arr = get_A_structure(n)
    
    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        b = fast_conv(a)
        M = float(b.max())

        # Solve LP: max sum(g) s.t. (a*g)_k <= M, g >= 0
        c_vec = -np.ones(n)
        A = csr_matrix((a[rows_arr - cols_arr], (rows_arr, cols_arr)), shape=(2 * n - 1, n))
        b_ub = np.full(2 * n - 1, M)
        bounds = [(0.0, None)] * n
        r = linprog(c_vec, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                    options={'presolve': True, 'time_limit': 5.0})
        
        if not (r.success and r.x is not None):
            break

        g = np.maximum(r.x, 0.0)
        g_sum = g.sum()
        if g_sum < 1e-10:
            break
        g = g / g_sum

        # Line search: golden section + parabolic refinement
        best_t = 0.0
        best_J = current_J
        
        # Evaluate at t=0, 0.5, 1.0
        points = [0.0, 0.3, 0.5, 0.7, 1.0]
        J_vals = []
        for t in points:
            a_t = (1.0 - t) * a + t * g
            a_t = np.maximum(a_t, 0)
            s_t = a_t.sum()
            if s_t < 1e-10:
                J_vals.append(1e10)
            else:
                a_t = a_t / s_t
                J_vals.append(compute_J(a_t))
        
        # Find best bracket
        best_idx = int(np.argmin(J_vals))
        best_t = points[best_idx]
        best_J = J_vals[best_idx]
        
        # Golden section in the best bracket
        if best_idx > 0:
            lo = points[best_idx - 1]
        else:
            lo = 0.0
        if best_idx < len(points) - 1:
            hi = points[best_idx + 1]
        else:
            hi = 1.0
        
        gr = (np.sqrt(5.0) - 1.0) / 2.0
        for _ in range(min(n_line_evals, 12)):
            if time.monotonic() > deadline:
                break
            t1 = hi - gr * (hi - lo)
            t2 = lo + gr * (hi - lo)
            a1 = np.maximum((1.0 - t1) * a + t1 * g, 0)
            a2 = np.maximum((1.0 - t2) * a + t2 * g, 0)
            s1, s2 = a1.sum(), a2.sum()
            J1 = compute_J(a1) if s1 > 1e-10 else 1e10
            J2 = compute_J(a2) if s2 > 1e-10 else 1e10
            if J1 < J2:
                hi = t2
            else:
                lo = t1
            # Track best
            t_mid = (lo + hi) / 2.0
            a_m = np.maximum((1.0 - t_mid) * a + t_mid * g, 0)
            s_m = a_m.sum()
            if s_m > 1e-10:
                a_m = a_m / s_m
                J_m = compute_J(a_m)
                if J_m < best_J:
                    best_J = J_m
                    best_t = t_mid

        a_new = np.maximum((1.0 - best_t) * a + best_t * g, 0)
        s_new = a_new.sum()
        if s_new > 1e-10:
            a_new = a_new / s_new
        else:
            a_new = a

        if best_J < current_J - 1e-13:
            a = a_new
            current_J = best_J
            stall_count = 0
            if best_J < best_J_so_far:
                best_J_so_far = best_J
                best_a_so_far = a.copy()
                if tracker is not None:
                    tracker.update(a)
        else:
            stall_count += 1
            if stall_count >= stall_patience:
                break

    if tracker is not None:
        tracker.update(best_a_so_far)
    return best_a_so_far


def subgradient_flatten(a, deadline, tracker=None, max_iters=2000):
    """
    Subgradient method that targets flattening the convolution.
    Uses M_target slightly below current max for aggressive flattening.
    """
    a = np.maximum(a.astype(np.float64), 1e-6)
    a = a / a.sum()
    n = len(a)
    current_J = compute_J(a)
    best_J = current_J
    best_a = a.copy()
    
    lr = 1.0 / (n ** 0.7)
    momentum = 0.9
    velocity = np.zeros(n)
    
    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        
        b = fast_conv(a)
        M = float(b.max())
        
        # Subgradient of max(c) w.r.t. a:
        # For active set (where c_k == M): grad_i = 2 * sum_{k active, 0<=k-i<n} a_{k-i}
        # Use softmax weighting for a smoothed version
        beta_sub = 10.0
        shifted = beta_sub * (b - M)
        shifted = np.clip(shifted, -500, 0)
        w = np.exp(shifted)
        w_sum = w.sum()
        if w_sum < 1e-10:
            w = np.zeros(2*n-1)
            w[np.argmax(b)] = 1.0
        else:
            w = w / w_sum
        
        # grad_i = 2 * sum_m a_m * w[m+i] (for 0 <= m+i < 2n-1)
        a_flip = a[::-1]
        full_conv = np.convolve(a_flip, w, mode='full')
        grad = 2.0 * full_conv[n-1:n-1+n]
        
        # Gradient of (sum a)^{-2}: d/da_i (sum a)^{-2} = -2*(sum a)^{-3}
        # So grad of J = 2n * [grad_max / S^2 - 2*M_target/S^3]
        # But we're working with normalized a (S=1), so:
        # grad J_i = 2n * grad_max_i (since S = 1, dJ/da_i = 2n * dmax/da_i)
        # Actually with S=1: J = 2n * max_c, dJ/da_i = 2n * dmax_c/da_i = 2n * grad_i
        # But we also need to maintain the constraint sum(a) = 1.
        # Projected subgradient: a_new = project(a - lr * grad)
        
        velocity = momentum * velocity + (1 - momentum) * grad
        a_new = a - lr * velocity
        a_new = np.maximum(a_new, 0)
        s_new = a_new.sum()
        if s_new < 1e-10:
            a_new = a.copy()
        else:
            a_new = a_new / s_new
        
        J_new = compute_J(a_new)
        if J_new < best_J:
            best_J = J_new
            best_a = a_new.copy()
            if tracker is not None:
                tracker.update(a_new)
        
        a = a_new
        # Decay learning rate
        lr *= 0.999
        
        if it % 100 == 0:
            # Periodic re-normalization and check
            if compute_J(a) > current_J * 1.5:
                # Revert to best
                a = best_a.copy()
    
    if tracker is not None:
        tracker.update(best_a)
    return best_a
