def fast_conv(a):
    """Full self-convolution via FFT."""
    n = len(a)
    if n == 1:

    a = np.maximum(a, 0)

    """Build sparse (2n-1) x n matrix A where A[k,j] = a[k-j] if 0<=k-j<n."""

        j_max = min(n - 1, k)
        count = j_max - j_min + 1
        rows[idx:idx + count] = k

    v = np.zeros(n)

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        frac = it / max(1, max_iters)
        tau = 3.0 + 80.0 * (frac ** 0.6)
        lr = 0.3 * (0.015 / 0.3) ** (frac ** 0.4)

        grad, p, b = smooth_gradient_step(a, tau)
        M = b.max()
        if M < best_M * (1e-10):
            best_M = M
            best_a = a.copy()

        v = 0.92 * v + lr * grad
        a_new = a - v
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s < 1e-10:
            a_new = a.copy()
        else:
            a_new = a_new / s
        a = a_new

    return best_a


def run_lp_ftrl(a, max_iters, deadline, verbose=True):
    """
    LP best-response iteration with FTRL-style averaging.
    At each step:
      1. Solve LP to get best response g to current a
      2. Update running sum of g's
      3. Set a = running_sum / count (the average)
    This gives O(1/sqrt(T)) convergence to the game value.
    """
    a = np.maximum(a.astype(np.float64), 1e-8)
    a = a / a.sum()
    n = len(a)
    best_a = a.copy()
    best_J = compute_J(a)
    running_sum = np.zeros(n)
    count = 0

    # Initial M from current a
    M = fast_conv(a).max()
    sum_a = a.sum()

    if verbose:
        print(f"  LP-FTRL start: n={n}, M={M:.8f}, J={best_J:.6f}", flush=True)

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        # Solve LP for best response
        g, success, lp_time = solve_best_response_lp(a, M, time_limit=30)
        if not success or g is None:
            break

        g_sum = g.sum()
        count += 1
        running_sum += g

        # The average is the FTRL iterate
        a_avg = running_sum / count
        a_avg = np.maximum(a_avg, 0.0)
        s_avg = a_avg.sum()
        if s_avg < 1e-10:
            a_avg = a.copy()
        else:
            a_avg = a_avg / s_avg

        # Evaluate the average
        J_avg = compute_J(a_avg)
        if J_avg < best_J:
            best_J = J_avg
            best_a = a_avg.copy()

        # Also check g itself (scaled)
        if g_sum > 1e-10:
            g_norm = g / g_sum
            J_g = compute_J(g_norm)
            if J_g < best_J:
                best_J = J_g
                best_a = g_norm.copy()

        # Update a: use the average for stability
        a = a_avg.copy()

        # Update M: recompute from current a
        M = fast_conv(a).max()

        # Track gap for monitoring
        gap = a.sum() - g_sum / (s_avg / a.sum() if s_avg > 0 else 1)

        if verbose and (it % 10 == 0 or it < 5):
            print(f"    it={it:4d} M={M:.8f} J_avg={J_avg:.6f} J_g={J_g:.6f} "
                  f"best={best_J:.6f} gap={abs(a.sum()-g_sum):.6f} "
                  f"lp_t={lp_time:.2f}s", flush=True)

        # Early stopping if gap is very small
        if abs(a.sum() - g_sum) < 1e-6 * a.sum() and it > 20:
            if verbose:
                print(f"    Converged at it={it}, gap={abs(a.sum()-g_sum):.2e}", flush=True)
            # Do a few more iterations for polish
            for extra in range(5):
                if time.monotonic() > deadline:
                    break
                g2, s2, _ = solve_best_response_lp(a, M, time_limit=20)
                if not s2:
                    break
                a_trial = (a + g2 / g2.sum()) / 2
                a_trial = np.maximum(a_trial, 0)
                s = a_trial.sum()
                if s > 1e-10:
                    a_trial /= s
                    J_t = compute_J(a_trial)
                    if J_t < best_J:
                        best_J = J_t
                        best_a = a_trial.copy()
                a = a_trial
                M = fast_conv(a).max()
            break

    return best_a, best_J


def run_subgradient_polish(a, max_iters, deadline):
    """Subgradient descent with Polyak step for final polish."""
    a = np.maximum(a.astype(np.float64), 1e-8)
    a = a / a.sum()
    n = len(a)
    best_a = a.copy()
    best_M = fast_conv(a).max()
    best_J = compute_J(a)
    running_sum = np.zeros(n)

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        b = fast_conv(a)
        M = b.max()
        tol = 1e-8 * M
        active = np.where(b >= M - tol)[0]

        grad = np.zeros(n)
        for k in active:
            lo = max(0, k - n + 1)
            hi = min(n - 1, k)
            grad[lo:hi + 1] += 2.0 * a[k - hi:k - lo + 1][::-1]

        gn = np.linalg.norm(grad)
        if gn < 1e-15:
            break
        lr = 0.1 / (1 + it) ** 0.5
        a_new = a - lr * grad / gn
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s < 1e-10:
            a_new = a.copy()
        else:
            a_new = a_new / s

        running_sum += a_new
        a = a_new

        J_cur = compute_J(a)
        if J_cur < best_J:
            best_J = J_cur
            best_a = a.copy()

    # Check averaged
    a_avg = running_sum / max(1, min(it + 1, max_iters))
    a_avg = np.maximum(a_avg, 0)
    s = a_avg.sum()
    if s > 1e-10:
        a_avg /= s
        J_avg = compute_J(a_avg)
        if J_avg < best_J:
            best_J = J_avg
            best_a = a_avg

    return best_a, best_J


def coordinate_descent_polish(a, max_sweeps, deadline):
    """
    Coordinate descent: for each j, find the optimal a[j] given others fixed.
    We minimize M = max(a⊛a) subject to sum(a)=1, a>=0.
    For coordinate j, the constraint (a⊛a)[k] <= M gives an upper bound on a[j].
    We try to increase a[j] to use all budget (which helps balance the autocorrelation).
    """
    a = np.maximum(a.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-10:
        return a
    a = a / s
    n = len(a)

    best_a = a.copy()
    best_M = fast_conv(a).max()
    best_J = compute_J(a)

    for sweep in range(max_sweeps):
        if time.monotonic() > deadline:
            break

        b = fast_conv(a)
        M = b.max()

        # For each coordinate, compute the max allowed value
        # given the constraint (a⊛a)[k] <= M for all k
        new_a = a.copy()
        for j in range(n):
            # Find upper bound on a[j] from each constraint row k
            # (a⊛a)[k] = sum_i a[i]*a[k-i]
            # The terms involving a[j]:
            #   i=j: a[j]*a[k-j] → contributes a[k-j] * a[j]
            #   i=k-j (if k-j != j): a[k-j]*a[j] → contributes a[k-j] * a[j]
            # So: (a⊛a)[k] = a[j] * (a[k-j] + a[j] if k!=2j else a[j]) + rest_k
            # where rest_k = sum of terms not involving a[j]
            # Upper bound: a[j] <= (M - rest_k) / coeff_k
            # where coeff_k = a[k-j] + (a[j] if k-j != j else a[j])... 
            # This is getting circular. Let me just do:
            # DB = (a⊛a) with a[j] set to 0, i.e., baseline
            # Then (a⊛a)[k] with new a[j] = t:
            #   = DB[k] + t * a[k-j] + t * a[j_old] (if k-j != j)
            #   = DB[k] + 2*t*a[j_old] (if k = 2j... wait no)
            #   
            # Actually: if we change a[j] from old to new = t:
            # (a⊛a)[k] changes by: t*a[k-j] - old*a[k-j] + (if k-j!=j: t*a[k-j] - old*a[k-j])
            # = (t - old) * a[k-j] + (if k-j != j and k-j valid: (t-old)*a[j]... 
            # This is getting too complicated. Let me use a simpler approach.
            
            # Simple: try increasing a[j] slightly and check if M increases
            # If yes, the increase caused a constraint violation.
            # Use binary search for the max allowed a[j].
            
            old_val = a[j]
            if old_val < 1e-12:
                continue
            
            # Set a[j] = 0 to get baseline
            a[j] = 0.0
            b_base = fast_conv(a)
            
            # Now find max t such that b_base[k] + t * coeff[k] <= M for all k
            # where coeff[k] = a[k-j] (from i=j term) + a[j] (from i=k-j term, if valid and != j)
            # But a[j] is now 0 in a_base, so:
            # coeff[k] = a_base[k-j] + (a_base[k-j] if k-j != j else 0)
            # Wait, a_base has a[j]=0, so:
            # (a_new⊛a_new)[k] = b_base[k] + t*a_base[k-j] + (t*a_base[k-j] if k-j != j else t*0)
            # Hmm, this isn't quite right either.
            # 
            # Let me just compute directly:
            # a_new = a_base but a_new[j] = t
            # (a_new⊛a_new)[k] = sum_i a_new[i]*a_new[k-i]
            # = b_base[k] + t*a_base[k-j] + t*a_base[k-j] (if k-j valid and k-j != j)
            # Wait no. b_base already has a[j]=0. Adding a[j]=t:
            # The new terms are: t * a_base[k-j] (from i=j) + a_base[j] * t (from i=k-j, but a_base[j]=0)
            # So: (a_new⊛a_new)[k] = b_base[k] + t * a_base[k-j]
            # (where a_base[k-j] is 0 if k-j is out of range or k-j == j)
            
            # Wait, that's not right. If k-j == j (i.e., k=2j), then:
            # The term is t * t = t^2, not t * a_base[k-j] = t * 0.
            # So for k = 2j: (a_new⊛a_new)[2j] = b_base[2j] + t^2
            # For k != 2j: (a_new⊛a_new)[k] = b_base[k] + t * a_base[k-j] (if k-j in [0,n))
            
            # But we also need the term from i = k-j:
            # (a_new⊛a_new)[k] includes a_new[k-j]*a_new[j] = a_base[k-j]*t (if k-j != j)
            # So total: b_base[k] + t*a_base[k-j] + t*a_base[k-j] = b_base[k] + 2*t*a_base[k-j]
            # Wait no, b_base[k] = sum_i a_base[i]*a_base[k-i] which does NOT include a[j] since a_base[j]=0.
            # When we set a_new[j] = t, the new terms in the sum are:
            #   i=j: a_new[j]*a_new[k-j] = t * a_base[k-j] (if k-j in [0,n) and k-j != j)
            #   i=k-j: a_new[k-j]*a_new[j] = a_base[k-j] * t (same thing if k-j != j)
            # So if k-j != j and both valid: added term = 2*t*a_base[k-j]
            # If k-j == j (k=2j): added term = t*t = t^2
            # If k-j not valid: added term = 0 (only i=j term matters, but a_base[k-j]=0)
            #   Actually if k-j not in [0,n), then i=j is still valid, and a_new[k-j] doesn't exist
            #   So the term a_new[j]*a_new[k-j] doesn't exist. Similarly for i=k-j.
            #   So added term = 0.
            
            # This is getting too complex for an inline computation. Let me just do:
            # Binary search on t in [0, 2*old_val] to find max t with max(b_new) <= M_current
            # where b_new = fast_conv(a_base with a[j]=t)
            
            # Too slow per coordinate. Let me skip this and use a different approach.
            a[j] = old_val  # restore
        
        # Instead of per-coordinate, do a global approach:
        # Find the peak of b, and reduce the coordinates that contribute most to it.
        b = fast_conv(a)
        M = b.max()
        peak_idx = np.argmax(b)
        
        # The peak at index k involves a[i]*a[k-i] for valid i
        # Reduce the coordinates with largest contributions
        lo = max(0, peak_idx - n + 1)
        hi = min(n - 1, peak_idx)
        contributions = np.abs(a[lo:hi + 1]) * np.abs(a[peak_idx - hi:peak_idx - lo + 1][::-1])
        
        # Reduce the top contributors slightly
        top_k = min(5, len(contributions))
        top_indices = np.argsort(contributions)[-top_k:]
        for idx in top_indices:
            j = lo + idx
            a[j] *= 0.999  # small reduction
        
        # Renormalize
        s = a.sum()
        if s > 1e-10:
            a = a / s
        
        J_cur = compute_J(a)
        if J_cur < best_J:
            best_J = J_cur
            best_a = a.copy()
    
    return best_a, best_J


def resample_seq(src, dest_n):
    """Resample 1D sequence to new length via linear interpolation."""
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)


def generate_inits(n, rng, parent=None):
    """Generate diverse initial sequences of length n."""
    inits = []
    t = np.linspace(0, 1, n, endpoint=False)

    if parent is not None:
        p = resample_seq(parent, n)
        inits.append(p)
        inits.append(np.maximum(p + rng.normal(0, 0.02, n), 0.01))
        inits.append(np.maximum(p + rng.normal(0, 0.05, n), 0.01))
        inits.append(np.maximum(p + rng.normal(0, 0.1, n), 0.01))

    inits.append(np.ones(n))
    inits.append(0.5 + 0.5 * np.cos(2 * np.pi * t))
    inits.append(0.5 + 0.5 * np.cos(4 * np.pi * t))
    inits.append(0.5 + 0.3 * np.cos(2 * np.pi * t) + 0.3 * np.cos(6 * np.pi * t))
    inits.append(np.maximum(0.8 + 0.4 * np.cos(2 * np.pi * 3 * t), 0.05))
    inits.append(np.exp(-3 * (t - 0.5) ** 2) + 0.1)
    inits.append(0.2 + 0.8 * np.abs(np.sin(np.pi * t)))
    inits.append(np.maximum(1.0 + 0.3 * np.cos(4 * np.pi * t) + 0.2 * np.cos(10 * np.pi * t), 0.05))
    inits.append(0.3 + 0.7 * np.abs(2 * t - 1))
    inits.append(np.maximum(0.5 + 0.4 * np.sin(2 * np.pi * 2 * t + np.pi / 4), 0.05))

    # Random smooth
    k = max(3, n // 30)
    kernel = np.ones(k) / k
    smooth = np.convolve(rng.uniform(0.3, 1.0, n), kernel, mode='same')
    inits.append(np.maximum(smooth, 0.01))

    return [np.maximum(x.astype(np.float64), 0.01) for x in inits]


def propose_candidate(seed=42, budget_s=1500, **kwargs):
    """
    Multi-phase search:
    Phase 1: Smooth gradient descent on multiple lengths (fast exploration)
    Phase 2: LP best-response with FTRL averaging (main optimization)
    Phase 3: Subgradient + coordinate polish
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + float(budget_s) - 20.0
    t_start = time.monotonic()

    best_sequence = None
    best_score = float('inf')

    def try_update(a):
        nonlocal best_sequence, best_score
        if a is None:
            return
        a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
        if a.size == 0 or a.sum() < 0.01:
            return
        score = float(evaluate_sequence(a.tolist()))
        if np.isfinite(score) and score < best_score:
            best_sequence = a.copy()
            best_score = score
            elapsed = time.monotonic() - t_start
            print(f"  ** NEW BEST: J={score:.8f} n={a.size} t={elapsed:.0f}s", flush=True)

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        if p.ndim == 1 and p.size > 0 and np.all(np.isfinite(p)):
            parent = np.maximum(p, 0.0)
            try_update(parent)
            print(f"Parent loaded: n={parent.size}, J={best_score:.6f}", flush=True)
    except (NameError, AttributeError):
        pass

    if parent is None:
        parent = np.ones(256)
        try_update(parent)

    # === PHASE 1: Smooth gradient descent across lengths ===
    p1_deadline = min(t_start + 250, deadline - 600)
    print(f"\n{'='*60}", flush=True)
    print(f"PHASE 1: Smooth Gradient Descent (deadline={p1_deadline - t_start:.0f}s)", flush=True)
    print(f"{'='*60}", flush=True)

    lengths = [384, 512, 768, 1024, 1536]
    n_best_p1 = parent.size
    a_best_p1 = parent.copy()

    for n in lengths:
        if time.monotonic() > p1_deadline:
            break
        phase_dl = min(time.monotonic() + max(20, 180 * 512 / n), p1_deadline)
        if time.monotonic() >= phase_dl:
            continue

        inits = generate_inits(n, rng, parent)
        best_M = float('inf')
        best_a_n = None

        for a0 in inits:
            if time.monotonic() > phase_dl:
                break
            a0 = np.maximum(a0, 1e-6)
            a0 = a0 / a0.sum()
            iters = max(80, int(3000 * 512 / n))
            a = run_smooth_descent(a0, iters, phase_dl)
            M = fast_conv(a).max()
            if M < best_M:
                best_M = M
                best_a_n = a.copy()

        if best_a_n is not None:
            J = compute_J(best_a_n)
            print(f"  n={n:4d}: M={best_M:.8f} J={J:.6f}", flush=True)
            try_update(best_a_n)
            if J < compute_J(a_best_p1):
                a_best_p1 = best_a_n.copy()
                n_best_p1 = n

    print(f"Phase 1 best: n={n_best_p1}, J={compute_J(a_best_p1):.6f}", flush=True)

    # === PHASE 2: LP best-response with FTRL averaging ===
    print(f"\n{'='*60}", flush=True)
    print(f"PHASE 2: LP FTRL Iteration (main phase)", flush=True)
    print(f"{'='*60}", flush=True)

    p2_deadline = deadline - 150

    # Try the best from phase 1
    a_lp = a_best_p1.copy()
    n_lp = len(a_lp)
    est_lp_time = 0.005 * (n_lp / 128) ** 2
    max_lp_iters = max(20, int((p2_deadline - time.monotonic()) / max(0.01, est_lp_time * 0.7)))
    print(f"  Starting LP: n={n_lp}, est_time_per_LP={est_lp_time*1000:.0f}ms, "
          f"max_iters~{max_lp_iters}", flush=True)

    a_result, J_result = run_lp_ftrl(a_lp, max_lp_iters, p2_deadline, verbose=True)
    try_update(a_result)

    # If time remains, try a different length
    remaining = p2_deadline - time.monotonic()
    if remaining > 120 and best_sequence is not None:
        # Try extending or changing length
        alt_n = best_sequence.size * 2 if best_sequence.size <= 768 else best_sequence.size // 2
        alt_n = max(384, min(2048, alt_n))
        if alt_n != best_sequence.size:
            print(f"\n  Trying alternate length n={alt_n}...", flush=True)
            a_alt = resample_seq(best_sequence, alt_n)
            a_alt = np.maximum(a_alt, 1e-6)
            a_alt = a_alt / a_alt.sum()
            # Quick gradient
            a_alt = run_smooth_descent(a_alt, 200, time.monotonic() + remaining * 0.3)
            try_update(a_alt)
            # LP on alternate
            if time.monotonic() < p2_deadline - 30:
                a_alt_res, J_alt = run_lp_ftrl(a_alt, 50, p2_deadline, verbose=False)
                try_update(a_alt_res)

    # === PHASE 3: Final polish ===
    print(f"\n{'='*60}", flush=True)
    print(f"PHASE 3: Final Polish", flush=True)
    print(f"{'='*60}", flush=True)

    if best_sequence is not None and time.monotonic() < deadline - 10:
        # Subgradient polish
        a_sg, J_sg = run_subgradient_polish(best_sequence, 3000, time.monotonic() + 60)
        try_update(a_sg)

        # Another LP round on the polished result
        if time.monotonic() < deadline - 20:
            a_lp2, J_lp2 = run_lp_ftrl(best_sequence, 30, deadline - 5, verbose=False)
            try_update(a_lp2)

        # Coordinate descent
        if time.monotonic() < deadline - 10:
            a_cd, J_cd = coordinate_descent_polish(best_sequence, 200, deadline - 5)
            try_update(a_cd)

    # Fallback
    if best_sequence is None:
        best_sequence = parent
        best_score = compute_J(best_sequence)
        try_update(best_sequence)

    elapsed = time.monotonic() - t_start
    print(f"\n{'='*60}", flush=True)
    print(f"FINAL: J={best_score:.8f} n={best_sequence.size} time={elapsed:.0f}s", flush=True)
    print(f"{'='*60}", flush=True)

    return [float(x) for x in best_sequence]
