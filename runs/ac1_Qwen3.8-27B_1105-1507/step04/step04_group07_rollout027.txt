    """grad[j] = sum_m p[m] * w[j+m] for valid indices."""
    n = len(p)
    L = len(w)  # = 2n-1
    S = 1
    while S < n + L - 1:
        S <<= 1
    fp = np.fft.rfft(p, S)
    fw_rev = np.fft.rfft(w[::-1], S)
    conv = np.fft.irfft(fp * fw_rev, S)

    # j=0: conv[2n-2], j=1: conv[2n-3], ..., j=n-1: conv[n-1]
    return conv[n-1:2*n-1][::-1].copy()

3. A **KKT-based approach**: At the optimum, the gradient of the objective is orthogonal to the feasible set. For the LSE approximation, this gives us a system of equations we can solve.

Let me think about this more carefully. The objective (with sum(p)=1 normalization) is:
minimize f(p) = max_k c[k] = max_k (p*p)[k]

At the optimum, the KKT conditions are:
- For active constraints (c[k] = M): ∂c[k]/∂p[j] = λ + μ_j (where λ is the Lagrange multiplier for sum(p)=1, μ_j ≥ 0 for p_j ≥ 0)
- For inactive constraints (c[k] < M): no condition
- Complementary slackness: μ_j * p_j = 0

So for j with p_j > 0: sum_{k active} α_k * ∂c[k]/∂p[j] = λ
where α_k ≥ 0 are weights on the active constraints.

This is a system that can be solved iteratively.

**Algorithm: KKT iteration**

1. Start with p (normalized, non-negative).
2. Compute c = conv(p, p), find M = max(c), active set A = {k : c[k] ≈ M}.
3. Solve for (α, λ): find α_k ≥ 0 for k in A and λ such that for all j with p_j > 0:
   sum_{k in A} α_k * ∂c[k]/∂p[j] = λ
   This is a linear system in (α, λ).
4. Update p in a direction that reduces the active peaks:
   direction[j] = -sum_{k in A} α_k * ∂c[k]/∂p[j] + λ (for j with p_j > 0)
   This direction should reduce all active peaks simultaneously.
5. Take a step in this direction, project to simplex.
6. Repeat.

The key advantage: this direction is "balanced" across all active peaks, unlike a simple gradient descent which might fix one peak while worsening another.

The linear system in step 3:
For j in support(p): sum_{k in A} α_k * p[k-j] (valid) = λ

This is a system with |A| + 1 unknowns (α_k for k in A, plus λ) and |support(p)| equations. If |A| << |support(p)|, we can solve the least-squares problem.

Actually, a simpler approach: just use the average gradient over active peaks:
direction[j] = (1/|A|) * sum_{k in A} ∂c[k]/∂p[j]

And take a negative gradient step. This is what the subgradient method does, but with a better step size.

For a more sophisticated approach, use the **Dantzig-Wolfe** style: find the combination of active constraints that gives the steepest descent.

import numpy as np
import time

def fast_conv(a):
    n = len(a)
    s = 1
    while s < 2*n - 1:
        s <<= 1
    fa = np.fft.rfft(a, s)
    return np.fft.irfft(fa * fa, s)[:2*n-1]

def compute_J(a):
    a = np.clip(a, 0, 1000)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))

def lse_grad_fft(p, w):
    """Compute grad[j] = sum_m p[m]*w[j+m] using FFT."""
    n = len(p)
    L = len(w)
    S = 1
    while S < n + L - 1:
        S <<= 1
    fp = np.fft.rfft(p, S)
    fw = np.fft.rfft(w[::-1], S)
    conv = np.fft.irfft(fp * fw, S)
    return conv[n-1:2*n-1][::-1].copy()

def lse_objective_and_grad(p, beta):
    """Compute LSE(beta * c) and its gradient."""
    n = len(p)
    c = fast_conv(p)
    max_c = c.max()
    exp_c = np.exp(beta * (c - max_c))
    Z = exp_c.sum()
    LSE = max_c + np.log(Z) / beta
    w = beta * exp_c / Z
    grad = lse_grad_fft(p, w)
    return LSE, grad, c

def simplex_project(v):
    """Project v onto the simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.where(u * np.arange(1, n+1) > (cssv - 1))[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1)
    return np.maximum(v - theta, 0)

def lse_descent(p, max_iters, deadline, tracker, beta_start=5, beta_end=200, momentum=0.7):
    """LSE-smoothed gradient descent with momentum and line search."""
    n = len(p)
    p = p.copy()
    p = simplex_project(p)
    current_J = compute_J(p)
    tracker.update(p)
    
    v = np.zeros(n)  # momentum
    beta = beta_start
    
    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        
        # Schedule beta
        progress = it / max(1, max_iters)
        beta = beta_start * (beta_end / beta_start) ** progress
        
        LSE, grad, c = lse_objective_and_grad(p, beta)
        
        # Momentum
        v = momentum * v + (1 - momentum) * grad
        
        # Line search
        best_p = p
        best_J = current_J
        lr_base = 0.1 / (n * np.linalg.norm(v))
        for scale in [1.0, 2.0, 0.5, 4.0, 0.25]:
            lr = lr_base * scale
            p_trial = p - lr * v
            p_trial = simplex_project(p_trial)
            J_trial = compute_J(p_trial)
            if J_trial < best_J - 1e-15:
                best_J = J_trial
                best_p = p_trial
        
        if np.array_equal(best_p, p):
            # No improvement, try smaller step or break
            break
        
        p = best_p
        current_J = best_J
        
        if it % 20 == 0:
            tracker.update(p)
    
    tracker.update(p)
    return p

def spectral_perturb(p, rng, strength=0.1):
    """Perturb p in the spectral domain."""
    n = len(p)
    S = 1
    while S < 2*n:
        S <<= 1
    P = np.fft.rfft(p, S)
    # Random phase perturbation
    phases = rng.normal(0, strength, P.size)
    P *= np.exp(1j * phases)
    p_new = np.fft.irfft(P, S)[:n]
    p_new = np.maximum(p_new, 0)
    if p_new.sum() < 1e-10:
        return p.copy()
    return p_new / p_new.sum()

    n = len(p)
