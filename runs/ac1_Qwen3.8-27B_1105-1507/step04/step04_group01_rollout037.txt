import numpy as np

def compute_conv(a):
    """Compute linear convolution a*a using FFT."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    conv = irfft(fa * fa, size)
    return conv[:2 * n - 1]

def compute_obj(a):
    """Compute 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    conv = compute_conv(a)
    max_b = float(np.max(conv))
    sum_a = float(np.sum(a))
    if sum_a < 0.01:
        return np.inf
    return 2.0 * n * max_b / (sum_a * sum_a)

def fft_convolve(a, b):
    """Linear convolution via FFT."""
    n, m = len(a), len(b)
    size = 1
    while size < n + m - 1:
        size <<= 1
    fa = rfft(a, size)
    fb = rfft(b, size)
    return irfft(fa * fb, size)[:n + m - 1]

def softmax(z):
    z = z - np.max(z)
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300:
        return np.ones(len(z)) / len(z)
    return e / s

def build_toeplitz(a):
    """Build matrix A where (A @ g)[k] = conv(a, g)[k]."""
    n = len(a)
    A = np.zeros((2*n-1, n))
    for j in range(n):
        A[j:j+n, j] = a[:n]
    return A

def lp_max_sum(a, M, time_limit=15):
    """LP: max sum(g) s.t. conv(a,g) <= M, g >= 0."""
    n = len(a)
    A = build_toeplitz(a)
    c = -np.ones(n)
    b_ub = np.full(2*n-1, M)
    bounds = [(0, None)] * n
    try:
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                     options={'presolve': True, 'time_limit': time_limit})
        if res.status == 0:
            g = np.maximum(res.x, 0)
            s = g.sum()
            if s > 1e-10:
                return g / s
    except Exception:
        pass
    return None

            lr *= 0.5
            continue
        a_new = a_new / s_new
        
        obj_new = compute_obj(a_new)
        if obj_new < best_obj:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new
            # Slowly decrease learning rate
            lr = max(0.01, lr * 0.999)

def random_init(n, seed):
    """Generate diverse random initializations."""
    rng = np.random.default_rng(seed)
    k = np.arange(n)
    choice = rng.integers(0, 8)
    
    if choice == 0:
        # Cosine
        f = rng.uniform(0.3, 3.0)
        phase = rng.uniform(0, 2*np.pi)
        a = 1.0 + 0.7 * np.cos(2*np.pi*f*k/n + phase)
    elif choice == 1:
        # Two cosines
        f1 = rng.uniform(0.2, 2.0)
        f2 = rng.uniform(2.0, 8.0)
        a = 1.0 + 0.4*np.cos(2*np.pi*f1*k/n) + 0.3*np.cos(2*np.pi*f2*k/n + rng.uniform(0, 2*np.pi))
    elif choice == 2:
        # Block structure
        nblocks = rng.integers(2, 6)
        a = np.zeros(n)
        for b in range(nblocks):
            start = b * n // nblocks
            end = (b+1) * n // nblocks
            a[start:end] = rng.uniform(0.2, 1.5)
    elif choice == 3:
        # Triangular / tent
        center = n * rng.uniform(0.3, 0.7)
        a = 1.0 - 0.6 * np.abs(k - center) / (n * 0.4)
    elif choice == 4:
        # Random smoothed
        a = rng.random(n)
        win = max(3, n // 30)
        kernel = np.ones(win) / win
        a = np.convolve(a, kernel, mode='same')
    elif choice == 5:
        # Chirp
        f0 = rng.uniform(0.3, 1.5)
        f1 = rng.uniform(2.0, 5.0)
        a = 1.0 + 0.5*np.cos(2*np.pi*(f0*k/n + (f1-f0)*k**2/(2*n**2)))
    elif choice == 6:
        # Edge-concentrated
        a = np.ones(n)
        edge_size = n // 5
        a[:edge_size] = rng.uniform(0.3, 1.0)
        a[-edge_size:] = rng.uniform(0.3, 1.0)
    else:
        # Pure random
        a = rng.random(n) + 0.1
    
    a = np.maximum(a, 0.01)
    return a / a.sum()

def resample(a, new_n):
    """Resample to new length."""
    if len(a) == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, len(a))
    x_new = np.linspace(0, 1, new_n)
    return np.interp(x_new, x_old, a)

Actually, I think the most impactful change I can make is to improve the LP alternating method. Let me think about why it might be stalling:

The LP alternating works as follows:
1. Given a, solve: max sum(g) s.t. conv(a,g) <= M, g >= 0, where M = max(conv(a,a))
2. Blend: a_new = (1-t)*a + t*g
3. Repeat

The issue is that the LP solution g might be very different from a, and a small blend (1-t)*a + t*g might not improve the objective much. A larger t might overshoot.

Alternative: instead of a fixed M, use M = max(conv(a,a)) * (1 + epsilon) for a small epsilon > 0. This gives more "room" in the LP and might find a better g.

Or: use the "min peak" LP formulation:
min M s.t. conv(a,g) <= M, sum(g) = 1, g >= 0

This directly finds the best g that minimizes the peak of conv(a,g). The resulting M tells us how much we can reduce the peak.

Let me add both formulations and use whichever gives a better result.

"""
Hybrid optimization for minimizing 2*n*max(conv(a,a))/(sum(a))^2 using:
- Softmax-parametrized L-BFGS-B with adaptive tau annealing
- LP alternating (both max-sum and min-peak formulations)
- Direct projected gradient with adaptive step
- Multi-scale length exploration
- Convolution-aware perturbation for local minimum escape
- Intensive final polishing with fine-tau L-BFGS
"""

import numpy as np
from scipy.optimize import minimize, linprog
from scipy.fft import rfft, irfft
import time


def compute_conv(a):
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    conv = irfft(fa * fa, size)
    return conv[:2 * n - 1]


def compute_obj(a):
    n = len(a)
    if n == 0:
        return np.inf
    conv = compute_conv(a)
    max_b = float(np.max(conv))
    sum_a = float(np.sum(a))
    if sum_a < 0.01:
        return np.inf
    return 2.0 * n * max_b / (sum_a * sum_a)


def fft_convolve(a, b):
    n, m = len(a), len(b)
    size = 1
    while size < n + m - 1:
        size <<= 1
    fa = rfft(a, size)
    fb = rfft(b, size)
    return irfft(fa * fb, size)[:n + m - 1]


def softmax(z):
    z = z - np.max(z)
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300:
        return np.ones(len(z)) / len(z)
    return e / s


def soft_max_grad_z(z, tau):
    n = len(z)
    a = softmax(z)
    conv = compute_conv(a)
    cmax = float(np.max(conv))
    shifted = np.clip((conv - cmax) / tau, -500.0, 500.0)
    exp_c = np.exp(shifted)
    Z = float(exp_c.sum())
    soft_val = cmax + tau * np.log(Z)
    w = exp_c / Z
    w_rev = w[::-1]
    conv_aw = fft_convolve(a, w_rev)
    grad_a = 2.0 * conv_aw[n - 1:2 * n - 1][::-1]
    mean_g = float(a @ grad_a)
    grad_z = a * (grad_a - mean_g)
    return soft_val, grad_z


def lfbs_optimize(a_init, tau_list, max_iter=200, time_budget=20.0):
    n = len(a_init)
    s = a_init.sum()
    if s < 1e-10:
        a = np.ones(n) / n
    else:
        a = a_init / s
    a = np.maximum(a, 1e-30)
    z = np.log(a)

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    for tau in tau_list:
        if time.monotonic() - start > time_budget:
            break

        def func_grad(z_val):
            return soft_max_grad_z(z_val, tau)

        try:
            res = minimize(func_grad, z, jac=True, method='L-BFGS-B',
                           options={'maxiter': max_iter, 'ftol': 1e-14,
                                    'gtol': 1e-9, 'maxls': 40})
            z = res.x
        except Exception:
            continue

        a_new = softmax(z)
        a_new = np.maximum(a_new, 0)
        s_new = a_new.sum()
        if s_new < 1e-10:
            continue
        a_new = a_new / s_new
        obj = compute_obj(a_new)

        if obj < best_obj:
            best_obj = obj
            best_a = a_new.copy()

    return best_a, best_obj


def build_toeplitz(a):
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        A[j:j + n, j] = a
    return A


def lp_max_sum(a, M, time_limit=15):
    n = len(a)
    A = build_toeplitz(a)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    try:
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': time_limit})
        if res.status == 0:
            g = np.maximum(res.x, 0)
            s = g.sum()
            if s > 1e-10:
                return g / s
    except Exception:
        pass
    return None


def lp_min_peak(a, time_limit=15):
    n = len(a)
    A = build_toeplitz(a)
    c = np.zeros(n + 1)
    c[-1] = 1.0
    A_ext = np.zeros((2 * n - 1, n + 1))
    A_ext[:, :n] = A
    A_ext[:, n] = -1.0
    b_ub = np.zeros(2 * n - 1)
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])
    bounds = [(0, None)] * n + [(0, None)]
    try:
        res = linprog(c, A_ub=A_ext, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': time_limit})
        if res.status == 0:
            g = np.maximum(res.x[:n], 0)
            s = g.sum()
            if s > 1e-10:
                return g / s, float(res.x[n])
    except Exception:
        pass
    return None, None

def projected_grad(a, n_iters=500, time_budget=15.0):

        return a

    best_a = a.copy()

        conv = compute_conv(a)
        cmax = float(conv.max())
        tau = max(0.3 / n, 0.3 * cmax / (n * 0.1))

            lr *= 0.5
            continue
        a_new = a_new / s_new

        obj_new = compute_obj(a_new)
        if obj_new < best_obj - 1e-14:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new
            lr = min(0.1, lr * 1.01)
            stall = 0
        else:
            stall += 1
            lr *= 0.9
            if stall > 80 or lr < 1e-7:
                break
        if it % 50 == 0:
            lr *= 0.95

    return best_a


def random_init(n, seed):
    rng = np.random.default_rng(seed)
    k = np.arange(n)
    choice = rng.integers(0, 10)

    if choice == 0:
        f = rng.uniform(0.3, 3.0)
        phase = rng.uniform(0, 2 * np.pi)
        a = 1.0 + 0.7 * np.cos(2 * np.pi * f * k / n + phase)
    elif choice == 1:
        f1 = rng.uniform(0.2, 2.0)
        f2 = rng.uniform(2.0, 8.0)
        a = 1.0 + 0.4 * np.cos(2 * np.pi * f1 * k / n) + 0.3 * np.cos(2 * np.pi * f2 * k / n + rng.uniform(0, 2 * np.pi))
    elif choice == 2:
        nblocks = rng.integers(2, 6)
        a = np.zeros(n)
        for b in range(nblocks):
            st = b * n // nblocks
            en = (b + 1) * n // nblocks
            a[st:en] = rng.uniform(0.2, 1.5)
    elif choice == 3:
        center = n * rng.uniform(0.3, 0.7)
        a = 1.0 - 0.6 * np.abs(k - center) / (n * 0.4)
    elif choice == 4:
        a = rng.random(n)
        win = max(3, n // 30)
        kernel = np.ones(win) / win
        a = np.convolve(a, kernel, mode='same')
    elif choice == 5:
        f0 = rng.uniform(0.3, 1.5)
        f1 = rng.uniform(2.0, 5.0)
        a = 1.0 + 0.5 * np.cos(2 * np.pi * (f0 * k / n + (f1 - f0) * k**2 / (2 * n**2)))
    elif choice == 6:
        a = np.ones(n)
        edge = n // 5
        a[:edge] = rng.uniform(0.3, 1.0)
        a[-edge:] = rng.uniform(0.3, 1.0)
    elif choice == 7:
        a = np.ones(n)
        for _ in range(rng.integers(1, 4)):
            f = rng.uniform(0.5, 4.0)
            amp = rng.uniform(0.1, 0.5)
            a += amp * np.cos(2 * np.pi * f * k / n + rng.uniform(0, 2 * np.pi))
    elif choice == 8:
        center = n * rng.uniform(0.2, 0.8)
        width = n * rng.uniform(0.1, 0.4)
        a = 1.0 + 0.6 * np.exp(-0.5 * ((k - center) / width) ** 2)
    else:
        a = rng.random(n) + 0.1

    a = np.maximum(a, 0.01)
    return a / a.sum()


def resample(a, new_n):
    if len(a) == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, len(a))
    x_new = np.linspace(0, 1, new_n)
    return np.interp(x_new, x_old, a)


def try_parent():
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64)
        parent = np.clip(parent, 0, 1000.0)
        if parent.size > 0 and np.isfinite(parent).all():
            obj = compute_obj(parent)
            print(f"Parent (n={parent.size}): obj={obj:.6f}")
            return parent.copy(), obj
    except Exception:
        pass
    return None, np.inf


def polish_sequence(a, time_budget):
    n = len(a)
    a = a.copy()
    start = time.monotonic()
    best_a = a.copy()
    best_obj = compute_obj(a)
    rnd = 0

    while time.monotonic() - start < time_budget - 2:
        rnd += 1
        rem = time_budget - (time.monotonic() - start)
        if rem < 5:
            break

        if rem > 10:
            tau_list = [1.5 / n, 0.7 / n, 0.3 / n, 0.15 / n, 0.07 / n]
            a_new, obj_new = lfbs_optimize(best_a, tau_list, max_iter=200,
                                           time_budget=min(rem * 0.18, 12.0))
            if obj_new < best_obj:
                best_obj = obj_new
                best_a = a_new.copy()
                print(f"    [r{rnd}] L-BFGS: {best_obj:.6f}")

        rem = time_budget - (time.monotonic() - start)
        if rem > 12 and n <= 800:
            a_new = alternating_lp(best_a, max_iters=40, time_budget=min(rem * 0.3, 25.0))
            obj_new = compute_obj(a_new)
            if obj_new < best_obj:
                best_obj = obj_new
                best_a = a_new.copy()
                print(f"    [r{rnd}] LP: {best_obj:.6f}")

        rem = time_budget - (time.monotonic() - start)
        if rem > 10 and n <= 700:
            a_new = lp_min_peak_blend(best_a, time_budget=min(rem * 0.25, 15.0))
            obj_new = compute_obj(a_new)
            if obj_new < best_obj:
                best_obj = obj_new
                best_a = a_new.copy()
                print(f"    [r{rnd}] LPMinPeak: {best_obj:.6f}")

        rem = time_budget - (time.monotonic() - start)
        if rem > 6:
            a_new = projected_grad(best_a, n_iters=int(rem * 12), time_budget=min(rem * 0.2, 10.0))
            obj_new = compute_obj(a_new)
            if obj_new < best_obj:
                best_obj = obj_new
                best_a = a_new.copy()
                print(f"    [r{rnd}] ProjGrad: {best_obj:.6f}")

    return best_a, best_obj


def propose_candidate(seed=42, budget_s=1500):
    """
    Main search: multi-scale exploration with L-BFGS-B (softmax), LP alternating,
    min-peak LP blending, and projected gradient. Final intensive polishing.
    """
    start_time = time.monotonic()
    deadline = start_time + budget_s - 30.0

    best_seq = None
    best_score = np.inf

    parent, parent_obj = try_parent()
    if parent is not None:
        best_seq = parent.copy()
        best_score = parent_obj

    print(f"Initial best: {best_score:.6f}")

    n_parent = len(best_seq) if best_seq is not None else 600
    n_values = sorted(set([
        n_parent,
        max(150, n_parent - 100),
        min(900, n_parent + 100),
        max(150, n_parent - 200),
        min(900, n_parent + 200),
        300, 400, 500, 700, 800
    ]))
    n_values = [n for n in n_values if 150 <= n <= 900]

    print(f"Lengths to explore: {n_values}")

    candidates = []
    total_explore = (deadline - start_time) * 0.45
    per_n = total_explore / len(n_values)

    for n in n_values:
        if time.monotonic() > deadline - 90:
            break
        rem = deadline - time.monotonic()
        if rem < 20:
            break

        alloc = max(18.0, min(per_n * 1.6, rem * 0.35))
        print(f"\n--- n={n}, alloc={alloc:.0f}s ---")

        n_inits = max(1, min(4, int(alloc / 14)))

        for ii in range(n_inits):
            if time.monotonic() > deadline - 90:
                break

            if ii == 0 and best_seq is not None:
                a = resample(best_seq, n)
            else:
                a = random_init(n, seed + n * 104729 + ii * 7919)

            a = projected_grad(a, n_iters=400, time_budget=min(6.0, alloc * 0.12))

            tau_list = [6.0/n, 3.0/n, 1.5/n, 0.7/n, 0.3/n, 0.15/n, 0.07/n]
            a, obj = lfbs_optimize(a, tau_list, max_iter=250, time_budget=min(alloc * 0.4, 25.0))

            a = projected_grad(a, n_iters=600, time_budget=min(alloc * 0.12, 8.0))
            obj = compute_obj(a)

            if n <= 750 and time.monotonic() < deadline - 90:
                a = alternating_lp(a, max_iters=35, time_budget=min(alloc * 0.25, 20.0))
                obj = compute_obj(a)

            if n <= 650 and time.monotonic() < deadline - 90:
                a = lp_min_peak_blend(a, time_budget=min(alloc * 0.15, 12.0))
                obj = compute_obj(a)

            print(f"  init{ii}: n={n}, obj={obj:.6f}")

            if obj < best_score:
                best_score = obj
                best_seq = a.copy()
                print(f"  ** NEW BEST: n={n}, obj={best_score:.6f} **")

            candidates.append((a.copy(), obj, n))

    candidates.sort(key=lambda x: x[1])

    print(f"\n=== PHASE 2: POLISHING (best={best_score:.6f}) ===")

    for ci in range(min(5, len(candidates))):
        if time.monotonic() > deadline - 40:
            break
        rem = deadline - time.monotonic()
        if rem < 25:
            break

        a_c = candidates[ci][0]
        n_c = candidates[ci][2]
        print(f"  Polish {ci} (n={n_c}, obj={candidates[ci][1]:.6f}), rem={rem:.0f}s")

        p_budget = min(rem * 0.3, 55.0)
        a_p, obj_p = polish_sequence(a_c, p_budget)

        if obj_p < best_score:
            best_score = obj_p
            best_seq = a_p.copy()
            print(f"  ** NEW BEST: {best_score:.6f} **")

    print(f"\n=== PHASE 3: FINAL POLISH (best={best_score:.6f}) ===")

    if best_seq is not None and time.monotonic() < deadline - 15:
        rem = deadline - time.monotonic()
        n_best = len(best_seq)
        print(f"  n={n_best}, rem={rem:.0f}s")

        a_cur = best_seq.copy()
        for fr in range(8):
            if time.monotonic() > deadline - 8:
                break
            rem = deadline - time.monotonic()
            if rem < 8:
                break

            tau_fine = [0.4/n_best, 0.2/n_best, 0.1/n_best, 0.05/n_best, 0.02/n_best]
            a_new, obj_new = lfbs_optimize(a_cur, tau_fine, max_iter=350,
                                           time_budget=min(rem * 0.18, 12.0))
            if obj_new < best_score:
                best_score = obj_new
                best_seq = a_new.copy()
                a_cur = a_new.copy()
                print(f"  L-BFGS {fr}: {best_score:.6f}")

            rem = deadline - time.monotonic()
            if rem > 12 and n_best <= 800:
                a_new = alternating_lp(a_cur, max_iters=60, time_budget=min(rem * 0.25, 25.0))
                obj_new = compute_obj(a_new)
                if obj_new < best_score:
                    best_score = obj_new
                    best_seq = a_new.copy()
                    a_cur = a_new.copy()
                    print(f"  LP {fr}: {best_score:.6f}")

            rem = deadline - time.monotonic()
            if rem > 10 and n_best <= 700:
                a_new = lp_min_peak_blend(a_cur, time_budget=min(rem * 0.2, 12.0))
                obj_new = compute_obj(a_new)
                if obj_new < best_score:
                    best_score = obj_new
                    best_seq = a_new.copy()
                    a_cur = a_new.copy()
                    print(f"  LPMinPeak {fr}: {best_score:.6f}")

            rem = deadline - time.monotonic()
            if rem > 6:
                a_new = projected_grad(a_cur, n_iters=int(rem * 10), time_budget=min(rem * 0.15, 8.0))
                obj_new = compute_obj(a_new)
                if obj_new < best_score:
                    best_score = obj_new
                    best_seq = a_new.copy()
                    a_cur = a_new.copy()
                    print(f"  ProjGrad {fr}: {best_score:.6f}")

    if best_seq is None:
        best_seq = np.ones(200)
        best_score = compute_obj(best_seq)

    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    val = evaluate_sequence(result)
    if val < np.inf:
        best_score = val
    else:
        result = [1.0] * 100
        best_score = 2.0

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_score:.6f}, time={elapsed:.1f}s ===")

    return result
