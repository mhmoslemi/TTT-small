def find_active_set(a, g, M, tol=1e-8):

    active = np.where(conv >= M - tol)[0]

    expanded = set()

        for dk in range(-2, 3):

            if 0 <= kk < 2 * n - 1:

def backtracking_line_search(a, g, J_current, max_iter=10):
    """Backtracking line search. Returns (best_a, best_J, best_t)."""
    best_t = 0.0
    best_J = J_current
    best_a = a.copy()
    t = 0.3  # Start with a reasonable step
    for _ in range(max_iter):
        a_try = (1.0 - t) * a + t * g
        a_try = np.maximum(a_try, 0.0)
        s = a_try.sum()
        if s < 1e-10:
            t *= 0.5
            continue
        a_try = a_try / s
        J_try = compute_J(a_try)
        if J_try < best_J:
            best_J = J_try
            best_t = t
            best_a = a_try.copy()
        t *= 0.5
    return best_a, best_J, best_t


def golden_line_search(a, g, J_current, n_evals=15):

    best_t = 0.0
    best_J = J_current
    best_a = a.copy()

    lo = 0.0
    hi = 1.0

    a_1 = g.copy()

    if J_1 < best_J:
        best_J = J_1
        best_t = 1.0
        best_a = a_1.copy()

        t1 = hi - gr * (hi - lo)
        t2 = lo + gr * (hi - lo)
        a1 = np.maximum((1.0 - t1) * a + t1 * g, 0.0)
        a2 = np.maximum((1.0 - t2) * a + t2 * g, 0.0)
        if a1.sum() < 1e-10:
            J1 = float('inf')
        else:
            J1 = compute_J(a1 / a1.sum())
        if a2.sum() < 1e-10:
            J2 = float('inf')
        else:
            J2 = compute_J(a2 / a2.sum())
        if J1 < J2:
            hi = t2
        else:
            lo = t1

    a_mid = np.maximum((1.0 - t_mid) * a + t_mid * g, 0.0)

            best_J = J_mid

def fast_conv(a):
    """Full self-convolution via FFT."""
    n = len(a)
    s = 1
    while s < 2 * n - 1:
        s <<= 1
    fa = np.fft.rfft(a, s)
    return np.fft.irfft(fa * fa, s)[:2 * n - 1]


def compute_J(a):
    """Compute J(a) = 2*n*max(conv(a,a))/(sum(a))^2."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))


def build_A_sparse(a, constraint_indices=None):
    """Build sparse convolution constraint matrix A where (A @ x)_k = (a*x)_k."""
    n = len(a)
    if constraint_indices is None:
        constraint_indices = range(2 * n - 1)
    rows = []
    cols = []
    for k in constraint_indices:
        jl = max(0, k - n + 1)
        jh = min(n - 1, k)
        for j in range(jl, jh + 1):
            rows.append(k)
            cols.append(j)
    if not rows:
        return None
    row_arr = np.array(rows, dtype=np.int32)
    col_arr = np.array(cols, dtype=np.int32)
    vals = a[row_arr - col_arr]
    return csr_matrix((vals, (row_arr, col_arr)), shape=(2 * n - 1, n))


def solve_lp(a, M, constraint_indices=None):
    """Solve LP: max sum(g) s.t. (a*g)_k <= M for specified k, g >= 0."""
    n = len(a)
    A = build_A_sparse(a, constraint_indices)
    if A is None:
        return None
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)

    try:

    except Exception:
        return None
    if r.success and r.x is not None:
        g = np.maximum(r.x, 0.0)
        if g.sum() > 1e-10:
            return g / g.sum()
    return None

def F_smooth(a, beta):
    """F(a) = 2n * smooth_max(conv(a,a)) / (sum(a))^2"""
    n = len(a)
    s = float(a.sum())

def gradient_descent_smooth(a0, beta_schedule, deadline, lr=0.5):
    """Gradient descent on smooth objective."""
    a = np.maximum(np.asarray(a0, dtype=np.float64), 1e-8)
    a = a / a.sum()
    
    for beta, n_iters in beta_schedule:
        if time.monotonic() > deadline:
            break
        current_F, current_grad = F_smooth(a, beta)
        for it in range(n_iters):
            if time.monotonic() > deadline:
                break
            # Gradient step
            a_new = a - lr * current_grad
            a_new = np.maximum(a_new, 1e-10)
            a_new = a_new / a_new.sum()
            F_new, grad_new = F_smooth(a_new, beta)
            if F_new < current_F:
                a = a_new
                current_F = F_new
                current_grad = grad_new
            else:
                # Halve learning rate
                lr *= 0.5
                if lr < 1e-8:
                    break
    return a


def resample_seq(src, dest_n):
    """Resample 1D sequence to new length via linear interpolation."""
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    result = np.interp(t_new, t_old, src)
    return np.maximum(result, 0.0)

        a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
        if a.size == 0 or a.sum() < 0.01:
            return
        score = float(evaluate_sequence(a.tolist()))
        if np.isfinite(score) and score < self.best_score:
            self.best = a.copy()
            self.best_score = score
            elapsed = time.monotonic() - self.t_start
            print(f"  * NEW BEST: J={score:.8f} n={a.size} t={elapsed:.1f}s", flush=True)
    
    def report(self, label=""):
        if self.best is not None:
            print(f"  {label} best: J={self.best_score:.8f} n={self.best.size}", flush=True)


def structured_perturb(a, rng, pert_type, param):
    """Apply a structured perturbation to a."""
    n = len(a)
    a_new = a.copy()
    
    if pert_type == 'gaussian':
        a_new = a_new + rng.normal(0, param, n) * a_new
    elif pert_type == 'uniform':
        a_new = a_new + rng.uniform(-param, param, n) * a_new
    elif pert_type == 'freq':
        # Add random sinusoid
        freq = rng.uniform(1, 8)
        phase = rng.uniform(0, 2 * np.pi)
        amp = param
        a_new = a_new + amp * np.sin(2 * np.pi * freq * np.arange(n) / n + phase) * a_new
    elif pert_type == 'block':
        # Modify a random block
        block_size = max(4, n // param)
        start = rng.integers(0, n - block_size)
        a_new[start:start + block_size] *= rng.uniform(0.3, 2.0, block_size)
    elif pert_type == 'symmetrize':
        # Blend with symmetric version
        sym = (a_new + a_new[::-1]) / 2.0
        a_new = (1 - param) * a_new + param * sym
    elif pert_type == 'mirror':
        # Blend with reversed
        a_new = (1 - param) * a_new + param * a_new[::-1]
    elif pert_type == 'smooth_noise':
        noise = rng.normal(0, param, n)
        kernel_size = max(3, n // 64)
        kernel = np.ones(kernel_size) / kernel_size
        noise = np.convolve(noise, kernel, mode='same')
        a_new = a_new + noise * a_new
    elif pert_type == 'spike':
        k = min(param, n // 4)
        idx = rng.choice(n, size=k, replace=False)
        a_new[idx] *= rng.uniform(0.3, 3.0, k)
    
    return np.maximum(a_new, 1e-4)
