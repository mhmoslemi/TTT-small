def coord_descent_pass(a, max_passes=1):

    S = a.sum()
    if S < 1e-10:

    n = len(a)

    indices = list(range(n))
    np.random.shuffle(indices)
    
    improved = False
    for i in indices:
        new_val = coord_descent_single(a, b, S, i)
        if abs(new_val - a[i]) > 1e-10:
            old_val = a[i]
            a[i] = new_val
            S = S - old_val + new_val
            improved = True
            # Update b incrementally
            k_lo = i
            k_hi = min(i + n - 1, 2 * n - 2)
            for k in range(k_lo, k_hi + 1):

                        b[k] = b[k] - 2.0 * old_val * a[ki] + 2.0 * new_val * a[ki]

    b_check = fast_conv(a)
    new_J = 2.0 * n * b_check.max() / (S * S)
    return a, new_J


def subgradient_step(a, step_size, top_k=5):
    """One subgradient step on J (using top-k active constraints)."""
    a = np.maximum(a, 0.0)
    S = a.sum()
    n = len(a)
    b = fast_conv(a)
    b_max = b.max()
    
    # Find top-k active constraints
    active = np.argsort(b)[-top_k:]
    
    # Subgradient of max_b w.r.t. a (averaged over active set)
    grad = np.zeros(n)
    for k in active:
        jl = max(0, k - n + 1)
        jh = min(n - 1, k)
        for j in range(jl, jh + 1):
            grad[j] += a[k - j]
    grad /= len(active)
    
    # Full gradient of J = 2n * b_max / S^2
    # dJ/da_i = 2n * (grad[i] * S^2 - b_max * 2S) / S^4
    #         = 2n * (grad[i] * S - 2*b_max) / S^3
    grad_J = 2.0 * n * (grad * S - 2.0 * b_max) / (S * S * S)
    
    a_new = a - step_size * grad_J
    a_new = np.maximum(a_new, 0.0)
    if a_new.sum() < 1e-10:
        return a
    return a_new


def resample_seq(src, dest_n):
    """Resample 1D sequence to new length via linear interpolation."""
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)

import numpy as np

def fast_conv(a):
    """Full self-convolution via FFT."""
    n = len(a)
    s = 1
    while s < 2 * n - 1:
        s <<= 1
    fa = np.fft.rfft(a, s)
    return np.fft.irfft(fa * fa, s)[:2 * n - 1]


def compute_J(a):
    """Compute J(a) = 2*n*max(conv(a,a))/(sum(a))^2."""
    a = np.maximum(a, 0)
    S = a.sum()
    if S < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (S * S))


def build_A_for_lp(a):
    """Build sparse matrix A such that (A @ g)_k = (a * g)_k."""
    n = len(a)
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        jl = max(0, k - n + 1)
        jh = min(n - 1, k)
        for j in range(jl, jh + 1):
            rows.append(k)
            cols.append(j)
            vals.append(a[k - j])
    A = csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))
    return A


def lp_best_response(a, M):
    """LP: max sum(g) s.t. (a*g)_k <= M for all k, g >= 0."""
    n = len(a)
    c = -np.ones(n)
    A = build_A_for_lp(a)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    try:

                    options={'time_limit': 30})
        if r.success and r.x is not None:
            g = np.maximum(r.x, 0.0)
            if g.sum() > 1e-10:
                return g
    except Exception:
        pass
    return None

    x_opt = (lo + hi) / 2.0
    return max(0.0, x_opt)


def coord_descent_pass(a, random_order=True):
    """One full pass of vectorized coordinate descent."""
    a = np.maximum(a.astype(np.float64), 0.0)
    S = a.sum()
    if S < 1e-10:
        return a, float('inf')
    n = len(a)
    b = fast_conv(a)
    current_J = 2.0 * n * b.max() / (S * S)

    indices = list(range(n))
    if random_order:
        np.random.shuffle(indices)

    for i in indices:
        new_val = coord_descent_single_vec(a, b, S, i)
        if abs(new_val - a[i]) > 1e-12:
            old_val = a[i]
            a[i] = new_val
            S = S - old_val + new_val
            # Update b incrementally
            k_lo = i
            k_hi = min(i + n - 1, 2 * n - 2)
            ks = np.arange(k_lo, k_hi + 1)
            for k in ks:
                if k == 2 * i:
                    b[k] = b[k] - old_val * old_val + new_val * new_val
                else:
                    ki = k - i
                    if 0 <= ki < n:
                        b[k] = b[k] - 2.0 * old_val * a[ki] + 2.0 * new_val * a[ki]

    # Verify with full convolution
    b_check = fast_conv(a)
    new_J = 2.0 * n * b_check.max() / (S * S)
    return a, new_J


def logsumexp_grad(a, temperature=0.01):
    """Compute smoothed gradient using log-sum-exp approximation to max."""
    a = np.maximum(a, 0.0)
    S = a.sum()
    n = len(a)
    b = fast_conv(a)
    b_max = b.max()

    # Soft max weights
    b_shifted = (b - b_max) / temperature
    weights = np.exp(b_shifted)
    weights /= weights.sum()

    # Gradient of logsumexp(b/tau) w.r.t. a[i]
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        w = weights[k]
        if w < 1e-10:
            continue
        jl = max(0, k - n + 1)
        jh = min(n - 1, k)
        for j in range(jl, jh + 1):
            grad[j] += w * a[k - j]

    # Full gradient of J_soft = 2n * logsumexp(b/tau) / S^2
    # dJ/da_i = 2n * (grad[i]/tau * S^2 - logsumexp * 2S) / S^4
    #         = 2n * (grad[i]/tau * S - 2*logsumexp_val) / S^3
    lse_val = b_max + temperature * np.log(weights.sum())  # approx
    lse_val = float(np.log(np.sum(np.exp(b / temperature))) * temperature)  # exact

    grad_J = 2.0 * n * (grad / temperature * S - 2.0 * lse_val) / (S * S * S)
    return grad_J, lse_val


def subgradient_direction(a, top_k=10, temperature=0.005):
    """Compute subgradient direction using top-k + logsumexp blend."""
    a = np.maximum(a, 0.0)
    S = a.sum()
    n = len(a)
    b = fast_conv(a)

    # Top-k active
    top_idx = np.argsort(b)[-top_k:]
    grad_top = np.zeros(n)
    for k in top_idx:
        jl = max(0, k - n + 1)
        jh = min(n - 1, k)
        for j in range(jl, jh + 1):
            grad_top[j] += a[k - j]
    grad_top /= len(top_idx)

    # Logsumexp gradient
    b_max = b.max()
    b_shifted = (b - b_max) / temperature
    weights = np.exp(b_shifted)
    weights /= weights.sum()
    grad_lse = np.zeros(n)
    for k in range(2 * n - 1):
        w = weights[k]
        if w < 1e-8:
            continue
        jl = max(0, k - n + 1)
        jh = min(n - 1, k)
        for j in range(jl, jh + 1):
            grad_lse[j] += w * a[k - j]

    # Blend
    alpha = 0.5
    grad = alpha * grad_top + (1 - alpha) * grad_lse

    # Full subgradient of J
    b_max_val = b.max()
    grad_J = 2.0 * n * (grad * S - 2.0 * b_max_val) / (S * S * S)
    return grad_J


def run_coord_descent(a, n_passes, deadline, tracker):
    """Run multiple coordinate descent passes."""
    a = np.maximum(a.astype(np.float64), 0.0)
    S = a.sum()
    if S < 1e-10:
        return a
    a = a / S * np.sqrt(2 * len(a))  # normalize to sum = sqrt(2n)
    current_J = compute_J(a)
    tracker.update(a)

    stall = 0
    for p in range(n_passes):
        if time.monotonic() > deadline:
            break
        t0 = time.monotonic()
        a, new_J = coord_descent_pass(a)
        elapsed = time.monotonic() - t0

        if new_J < current_J - 1e-12:
            improvement = current_J - new_J
            current_J = new_J
            stall = 0
            tracker.update(a)
            if p % 10 == 0 or improvement > 1e-5:
                print(f"    CD pass {p}: J={current_J:.8f} (impr={improvement:.2e}, {elapsed:.2f}s)", flush=True)
        else:
            stall += 1
            if stall > 5:
                break

    tracker.update(a)
    return a


def run_subgradient_cd(a, n_subgrad, cd_passes_between, deadline, tracker):
    """Alternate subgradient steps with coordinate descent."""
    a = np.maximum(a.astype(np.float64), 0.001)
    S = a.sum()
    n = len(a)
    current_J = compute_J(a)
    tracker.update(a)

    # Adaptive step size
    step_size = S / n * 0.1
    best_J = current_J
    best_a = a.copy()
    stall = 0

    for sg_iter in range(n_subgrad):
        if time.monotonic() > deadline:
            break

        # Subgradient step
        grad = subgradient_direction(a, top_k=8, temperature=0.005)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-15:
            break

        # Try multiple step sizes
        best_trial = None
        best_trial_J = current_J
        candidates = [step_size, step_size * 0.5, step_size * 2.0, step_size * 0.25]
        for alpha in candidates:
            a_trial = a - alpha * grad
            a_trial = np.maximum(a_trial, 0.0)
            if a_trial.sum() < 1e-10:
                continue
            a_trial = a_trial / a_trial.sum() * S  # keep same sum
            J_trial = compute_J(a_trial)
            if J_trial < best_trial_J:
                best_trial_J = J_trial
                best_trial = a_trial.copy()

        if best_trial is not None:
            a = best_trial
            if current_J - best_trial_J > 1e-8:
                step_size = min(step_size * 1.2, S / n)
                stall = 0
            else:
                step_size *= 0.8
            current_J = best_trial_J
        else:
            step_size *= 0.7
            stall += 1

        # Coordinate descent polish
        if time.monotonic() < deadline:
            for _ in range(min(cd_passes_between, 3)):
                if time.monotonic() > deadline:
                    break
                a, new_J = coord_descent_pass(a)
                if new_J < current_J:
                    current_J = new_J
            tracker.update(a)

        if stall > 15:
            break

    return a


def resample_seq(src, dest_n):
    """Resample 1D sequence to new length via linear interpolation."""
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)


class ScoreTracker:
    def __init__(self):
        self.best = None
        self.best_score = float('inf')
        self.t_start = time.monotonic()

    def update(self, a):
        a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
        if a.size == 0 or a.sum() < 0.01:
            return
        score = float(evaluate_sequence(a.tolist()))
        if np.isfinite(score) and score < self.best_score:
            self.best = a.copy()
            self.best_score = score
            elapsed = time.monotonic() - self.t_start
            print(f"  * NEW BEST: J={score:.8f} n={a.size} t={elapsed:.0f}s", flush=True)

    def report(self, label=""):
        if self.best is not None:
            print(f"  {label} best: J={self.best_score:.8f} n={self.best.size} t={time.monotonic()-self.t_start:.0f}s", flush=True)


def generate_structured_inits(n, rng):
    """Generate diverse structured initializations."""
    inits = []
    t = np.linspace(0, 1, n, endpoint=False)

    # Uniform
    inits.append(np.ones(n))

    # Linear ramp (biased)
    inits.append(np.linspace(0.5, 1.5, n))

    # Cosine modulations at various frequencies
    for freq in [1, 2, 3, 5, 7, 11]:
        inits.append(np.maximum(1.0 + 0.4 * np.cos(2 * np.pi * freq * t), 0.01))

    # Quadratic phase (chirp-like)
    inits.append(np.maximum(1.0 + 0.3 * np.cos(np.pi * t * t * 4), 0.01))

    # Two-tone
    inits.append(np.maximum(1.0 + 0.3 * np.cos(2 * np.pi * 3 * t) + 0.2 * np.cos(2 * np.pi * 7 * t + 0.5), 0.01))

    # Gaussian bump
    inits.append(0.5 + 0.5 * np.exp(-4 * (t - 0.5) ** 2))

    # Two peaks
    inits.append(0.3 + 0.7 * np.exp(-20 * (t - 0.3) ** 2) + 0.7 * np.exp(-20 * (t - 0.7) ** 2))

    # Random (smooth)
    coeffs = rng.normal(0, 0.1, 4)
    inits.append(np.maximum(1.0 + sum(c * np.cos(2 * np.pi * (k + 1) * t) for k, c in enumerate(coeffs)), 0.01))

    return inits


def run_lp_phase(a, n_iters, deadline, tracker):
    """Run LP best response iterations."""
    a = np.maximum(a.astype(np.float64), 0.001)
    a = a / a.sum()
    n = len(a)
    current_J = compute_J(a)
    tracker.update(a)

    for it in range(n_iters):
        if time.monotonic() > deadline:
            break

        b = fast_conv(a)
        M = float(b.max())

        # Try tightening
        for factor in [0.995, 0.99, 0.98, 0.97]:
            g = lp_best_response(a, M * factor)
            if g is not None:
                break

        if g is None:
            g = lp_best_response(a, M * 1.001)
            if g is None:
                continue

        g = np.maximum(g, 0.0)
        g = g / g.sum()

        # Try multiple blend factors
        best_a = None
        best_J = current_J
        for t in [0.3, 0.5, 0.7, 0.8, 0.9]:
            a_trial = (1.0 - t) * a + t * g
            a_trial = np.maximum(a_trial, 0.0)
            s = a_trial.sum()
            if s < 1e-10:
                continue
            a_trial = a_trial / s
            J_trial = compute_J(a_trial)
            if J_trial < best_J:
                best_J = J_trial
                best_a = a_trial.copy()

        if best_a is not None:
            a = best_a
            current_J = best_J
            tracker.update(a)
            if it % 5 == 0:
                print(f"    LP iter {it}: J={current_J:.8f}", flush=True)
        else:
            break

    return a


def propose_candidate(seed=42, budget_s=1500, **kwargs):
    """
    Main search function: intensive coordinate descent + subgradient + LP.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + float(budget_s) - 20.0
    tracker = ScoreTracker()

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        if p.ndim == 1 and p.size > 0 and np.all(np.isfinite(p)):
            parent = np.maximum(p, 0.0)
            tracker.update(parent)
            print(f"Parent loaded: n={parent.size}, J={tracker.best_score:.8f}", flush=True)
    except (NameError, AttributeError):
        pass

    if parent is None:
        parent = np.ones(386)
        tracker.update(parent)
        print(f"No parent, using uniform n=386", flush=True)

    n_parent = parent.size
    a_best = parent.copy()

    # === Phase 1: Intensive coordinate descent on parent ===
    print(f"\n{'='*50} Phase 1: CD on parent (n={n_parent}) {'='*50}", flush=True)
    t0 = time.monotonic()
    budget1 = 400
    a_best = run_coord_descent(a_best, n_passes=500, deadline=time.monotonic() + budget1, tracker=tracker)
    tracker.report(f"Phase 1 ({time.monotonic()-t0:.1f}s)")

    # === Phase 2: Subgradient + CD alternation ===
    if time.monotonic() < deadline - 200:
        print(f"\n{'='*50} Phase 2: Subgradient + CD {'='*50}", flush=True)
        t0 = time.monotonic()
        budget2 = 200
        a_best = run_subgradient_cd(a_best, n_subgrad=100, cd_passes_between=2,
                                     deadline=time.monotonic() + budget2, tracker=tracker)
        tracker.report(f"Phase 2 ({time.monotonic()-t0:.1f}s)")

    # === Phase 3: More CD passes ===
    if time.monotonic() < deadline - 150:
        print(f"\n{'='*50} Phase 3: More CD {'='*50}", flush=True)
        t0 = time.monotonic()
        a_best = run_coord_descent(a_best, n_passes=300, deadline=time.monotonic() + 150, tracker=tracker)
        tracker.report(f"Phase 3 ({time.monotonic()-t0:.1f}s)")

    # === Phase 4: LP best response ===
    if time.monotonic() < deadline - 150:
        print(f"\n{'='*50} Phase 4: LP Best Response {'='*50}", flush=True)
        t0 = time.monotonic()
        a_lp = run_lp_phase(tracker.best, n_iters=20, deadline=time.monotonic() + 100, tracker=tracker)
        # Polish with CD
        if time.monotonic() < deadline - 100:
            a_lp = run_coord_descent(a_lp, n_passes=100, deadline=time.monotonic() + 50, tracker=tracker)
        tracker.report(f"Phase 4 ({time.monotonic()-t0:.1f}s)")

    # === Phase 5: Try nearby lengths ===
    if time.monotonic() < deadline - 100:
        print(f"\n{'='*50} Phase 5: Nearby Lengths {'='*50}", flush=True)
        t0 = time.monotonic()

        nearby = [n_parent - 2, n_parent + 2, n_parent + 4, n_parent - 4,
                  n_parent + 6, n_parent + 8, n_parent - 6, n_parent + 10]
        nearby = [l for l in nearby if 200 <= l <= 600]
        seen = set()
        unique = []
        for l in nearby:
            if l not in seen:
                seen.add(l)
                unique.append(l)

        for nl in unique[:5]:  # Try up to 5 nearby lengths
            if time.monotonic() > deadline - 60:
                break
            a_res = resample_seq(tracker.best, nl)
            a_res = np.maximum(a_res, 0.001)
            a_res = a_res / a_res.sum() * np.sqrt(2 * nl)
            a_res = run_coord_descent(a_res, n_passes=50, deadline=time.monotonic() + 40, tracker=tracker)
            print(f"    Length {nl}: best so far J={tracker.best_score:.8f}", flush=True)

        tracker.report(f"Phase 5 ({time.monotonic()-t0:.1f}s)")

    # === Phase 6: Perturbation restarts ===
    if time.monotonic() < deadline - 50:
        print(f"\n{'='*50} Phase 6: Perturbation Restarts {'='*50}", flush=True)
        t0 = time.monotonic()
        n = tracker.best.size

        for temp in [0.03, 0.02, 0.015, 0.01, 0.005]:
            if time.monotonic() > deadline - 30:
                break
            a_pert = tracker.best + rng.normal(0, temp, n)
            a_pert = np.maximum(a_pert, 0.001)
            a_pert = a_pert / a_pert.sum() * np.sqrt(2 * n)
            a_pert = run_coord_descent(a_pert, n_passes=50, deadline=time.monotonic() + 30, tracker=tracker)

        # Also try structured inits
        for init in generate_structured_inits(n, rng)[:3]:
            if time.monotonic() > deadline - 20:
                break
            init = init / init.sum() * np.sqrt(2 * n)
            init = run_coord_descent(init, n_passes=30, deadline=time.monotonic() + 20, tracker=tracker)

        tracker.report(f"Phase 6 ({time.monotonic()-t0:.1f}s)")

    # === Phase 7: Final intensive CD on best ===
    if time.monotonic() < deadline - 5:
        print(f"\n{'='*50} Phase 7: Final Polish {'='*50}", flush=True)
        t0 = time.monotonic()
        a_final = run_coord_descent(tracker.best, n_passes=200, deadline=deadline - 2, tracker=tracker)
        tracker.report(f"Phase 7 ({time.monotonic()-t0:.1f}s)")

    # Fallback
    if tracker.best is None:
        tracker.best = np.ones(256)
        tracker.best_score = compute_J(tracker.best)

    print(f"\n{'='*50} FINAL RESULT {'='*50}", flush=True)
    print(f"J = {tracker.best_score:.10f}, n = {tracker.best.size}", flush=True)
    print(f"Total time: {time.monotonic()-tracker.t_start:.1f}s", flush=True)

    return [float(x) for x in tracker.best]
