"""
Minimizes J(a) = 2*n*max(conv(a,a))/(sum(a))^2 using a hybrid approach:
1. Iterative Fourier Transform (IFT) / spectral flatness approach
2. LP mirror descent for precise refinement
3. Smooth gradient descent with log-sum-exp

import numpy as np
from scipy.optimize import linprog, minimize
from scipy.sparse import csr_matrix
import time

def fast_conv(a):
    """Full self-convolution via FFT."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]

def compute_J(a):
    """Compute J(a) = 2*n*max(conv(a,a))/(sum(a))^2."""
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))

def ilt_optimize(a, max_iters=1000, hard=True, oversample=1, deadline=None):
    """
    Iterative Fourier Transform (Gerchberg-Saxton variant) for spectral flatness.
    Flattens the magnitude spectrum to minimize peak autocorrelation.
    """
    a = np.maximum(a.astype(np.float64), 0)
    s = a.sum()
    if s < 1e-10:
        return a
    a = a / s
    n = len(a)
    
    # FFT size
    fft_size = max(2 * n - 1, n * oversample)
    fft_size = 1 << (fft_size - 1).bit_length()  # next power of 2
    
    best_a = a.copy()
    best_M = fast_conv(a).max()
    
    for it in range(max_iters):
        if deadline is not None and time.monotonic() > deadline:
            break
        
        # Forward FFT
        A = np.fft.fft(a, fft_size)
        
        # Flatten magnitude
        mag = np.abs(A)
        target_mag = np.mean(mag)
        
        # Build new spectrum with flattened magnitude, same phase
        phase = np.angle(A)
        A_new = target_mag * np.exp(1j * phase)
        
        # Inverse FFT
        a_new = np.fft.ifft(A_new).real[:n]
        
        # Project to non-negative
        if hard:
            a_new = np.maximum(a_new, 0)
        else:
            a_new = a_new + np.min(a_new)
        
        # Normalize
        s_new = a_new.sum()
        if s_new < 1e-10:
            break
        a_new = a_new / s_new
        
        # Check improvement
        M_new = fast_conv(a_new).max()
        if M_new < best_M:
            best_M = M_new
            best_a = a_new.copy()
        
        a = a_new
    
    return best_a

def ilt_soft(a, max_iters=1000, start_alpha=1.0, end_alpha=0.1, deadline=None):
    """
    Soft IFT: instead of perfectly flat, use a soft flatness target.
    alpha controls how much we flatten (1=full flatten, 0=keep original).
    """
    a = np.maximum(a.astype(np.float64), 0)
    s = a.sum()
    if s < 1e-10:
        return a
    a = a / s
    n = len(a)
    
    fft_size = max(2 * n - 1, 512)
    fft_size = 1 << (fft_size - 1).bit_length()
    
    best_a = a.copy()
    best_M = fast_conv(a).max()
    
    for it in range(max_iters):
        if deadline is not None and time.monotonic() > deadline:
            break
        
        alpha = start_alpha + (end_alpha - start_alpha) * (it / max_iters)
        
        A = np.fft.fft(a, fft_size)
        mag = np.abs(A)
        phase = np.angle(A)
        
        # Soft flatness: blend between original and flat
        target_mag = (1 - alpha) * mag + alpha * np.mean(mag)
        
        A_new = target_mag * np.exp(1j * phase)
        a_new = np.fft.ifft(A_new).real[:n]
        
        # Project
        a_new = np.maximum(a_new, 0)
        s_new = a_new.sum()
        if s_new < 1e-10:
            break
        a_new = a_new / s_new
        
        M_new = fast_conv(a_new).max()
        if M_new < best_M:
            best_M = M_new
            best_a = a_new.copy()
        
        a = a_new
    
    return best_a

def smooth_max_gradient(a, tau):
    """Compute gradient of smooth max of conv(a,a)."""
    n = len(a)
    b = fast_conv(a)
    b_max = b.max()
    exp_b = np.exp(tau * (b - b_max))
    p = exp_b / exp_b.sum()
    
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        pk = p[k]
        if pk < 1e-13:
            continue
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        grad[lo:hi + 1] += 2.0 * pk * a[k - hi:k - lo + 1][::-1]
    
    return grad, p, b

def run_smooth_gradient(a, max_iters, deadline, initial_tau=2.0, final_tau=80.0,
                        lr_init=0.3, lr_min=0.01, momentum=True):
    """Smooth gradient descent with adaptive tau."""
    a = np.maximum(a.astype(np.float64), 1e-8)
    a = a / a.sum()
    n = len(a)
    
    best_a = a.copy()
    best_M = fast_conv(a).max()
    v = np.zeros(n)
    
    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        
        frac = it / max(1, max_iters)
        tau = initial_tau + (final_tau - initial_tau) * (frac ** 0.5)
        lr = lr_init * (lr_min / lr_init) ** frac
        
        grad, p, b = smooth_max_gradient(a, tau)
        M = b.max()
        
        if M < best_M * (1 - 1e-8):
            best_M = M
            best_a = a.copy()
        
        beta = 0.9 if momentum else 0.0
        v = beta * v + lr * grad
        a_new = a - v
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s < 1e-10:
            a_new = a.copy()
        else:
            a_new = a_new / s
        a = a_new
    
    return best_a

def build_A_sparse(n, a):
    """Build sparse constraint matrix for LP."""
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        j_min = max(0, k - n + 1)
        j_max = min(n - 1, k)
        for j in range(j_min, j_max + 1):
            rows.append(k)
            cols.append(j)
            vals.append(a[k - j])
    A = csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))
    return A

def lp_best_response(a, M):
    """LP: max sum(g) s.t. (a*g)[k] <= M for all k, g >= 0."""
    n = len(a)
    A = build_A_sparse(n, a)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    try:
        r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                    options={'time_limit': 60})
        if r.success and r.x is not None:
            g = np.maximum(r.x, 0)
            if g.sum() > 1e-15:
                return g
    except Exception:
        pass
    return None

def run_lp_mirror_descent(a, max_iters, deadline):
    """LP mirror descent with geometric line search."""
    a = np.maximum(a.astype(np.float64), 1e-8)
    a = a / a.sum()
    best_a = a.copy()
    best_M = fast_conv(a).max()
    no_improve = 0
    
    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        
        b = fast_conv(a)
        M = float(b.max())
        g = lp_best_response(a, M)
        if g is None:
            break
        g = np.maximum(g, 0.0)
        g_sum = g.sum()
        if g_sum < 1e-10:
            break
        g /= g_sum
        
        improved = False
        base_t = max(0.05, 0.6 * (1 - 0.3 * it / max(1, max_iters)))
        for t in [base_t, base_t * 0.5, base_t * 0.25, 0.02, 0.01, 0.005]:
            a_trial = np.maximum((1 - t) * a + t * g, 0)
            s = a_trial.sum()
            if s < 1e-10:
                continue
            a_trial /= s
            M_trial = fast_conv(a_trial).max()
            if M_trial < best_M - 1e-14:
                a = a_trial
                best_M = M_trial
                best_a = a.copy()
                improved = True
                break
        
        if not improved:
            no_improve += 1
            if no_improve > 30:
                break
        else:
            no_improve = 0
    
    return best_a

def resample_seq(src, dest_n):
    """Resample 1D sequence to new length."""
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)

def generate_initializations(n, rng, parent=None):

    # U-shape (high at boundaries)

    # W-shape (high at 0, n/2, and n-1)

    # Cosine modulation
    for freq in [1, 2, 3, 5]:
        inits.append(np.maximum(1.0 + 0.3 * np.cos(2 * np.pi * freq * t_arr), 0.01))
        inits.append(np.maximum(1.0 + 0.5 * np.cos(2 * np.pi * freq * t_arr), 0.01))
    
    # Gaussian bumps
    inits.append(np.exp(-4.0 * (t_arr - 0) ** 2) + np.exp(-4.0 * (t_arr - 1) ** 2) + 0.1)
    inits.append(np.exp(-8.0 * (t_arr - 0.25) ** 2) + np.exp(-8.0 * (t_arr - 0.75) ** 2) + 0.05)
    
    # Triangle
    inits.append(np.maximum(1.0 - 0.5 * np.abs(2 * t_arr - 1), 0.05))
    
    # Smooth random
    ksize = max(3, n // 20)
    kernel = np.ones(ksize) / ksize
    smooth = np.convolve(rng.uniform(0, 1, n), kernel, mode='same')
    inits.append(np.maximum(smooth + 0.1, 0.01))
    
    # High-frequency noise (smoothed)
    ksize2 = max(2, n // 50)
    kernel2 = np.ones(ksize2) / ksize2
    noisy = np.convolve(rng.uniform(0.3, 1.0, n), kernel2, mode='same')
    inits.append(np.maximum(noisy, 0.01))
    
    return inits

"""
Minimizes J(a) = 2*n*max(conv(a,a))/(sum(a))^2 using a hybrid approach:
1. Iterative Fourier Transform (IFT) / Gerchberg-Saxton for spectral flatness
2. LP mirror descent (best-response dynamics) for precise refinement
3. Smooth gradient descent with adaptive log-sum-exp
4. Fourier-domain structured perturbations for escaping local minima
5. Multi-length exploration with diverse initializations

Key innovation: IFT directly targets spectral flatness (flattening |FFT(a)|),
providing structurally different starting points than the LP approach alone,
enabling escape from suboptimal saddle points of the best-response game.
"""

import numpy as np
from scipy.optimize import linprog
from scipy.sparse import csr_matrix
import time


def fast_conv(a):
    """Full self-convolution via FFT."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]


def compute_J(a):
    """Compute J(a) = 2*n*max(conv(a,a))/(sum(a))^2."""
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))


def ilt_optimize(a, max_iters=1000, deadline=None, soft_alpha=0.0):
    """
    Iterative Fourier Transform (Gerchberg-Saxton variant) for spectral flatness.
    Flattens the magnitude spectrum to minimize peak autocorrelation.
    soft_alpha > 0 means we don't fully flatten (interpolate).
    """
    a = np.maximum(a.astype(np.float64), 0)
    s = a.sum()
    if s < 1e-10:
        return a
    a = a / s
    n = len(a)

    fft_size = max(2 * n - 1, 256)
    fft_size = 1 << (fft_size - 1).bit_length()

    best_a = a.copy()
    best_M = fast_conv(a).max()

    for it in range(max_iters):
        if deadline is not None and time.monotonic() > deadline:
            break

        A = np.fft.fft(a, fft_size)
        mag = np.abs(A)
        phase = np.angle(A)

        # Target magnitude: blend between flat and original
        flat_mag = np.mean(mag)
        if soft_alpha > 0:
            # Adaptive: start with more flattening, reduce over iterations
            alpha = soft_alpha * max(0.1, 1.0 - 0.3 * it / max_iters)
            target_mag = (1 - alpha) * mag + alpha * flat_mag
        else:
            target_mag = flat_mag * np.ones_like(mag)

        A_new = target_mag * np.exp(1j * phase)
        a_new = np.fft.ifft(A_new).real[:n]

        # Project to non-negative
        a_new = np.maximum(a_new, 0)
        s_new = a_new.sum()
        if s_new < 1e-10:
            break
        a_new = a_new / s_new

        M_new = fast_conv(a_new).max()
        if M_new < best_M:
            best_M = M_new
            best_a = a_new.copy()

        a = a_new

    return best_a


def ilt_multi_scale(a, max_iters=800, deadline=None):

    """

def smooth_max_gradient(a, tau):
    """Compute gradient of smooth max of conv(a,a)."""
    n = len(a)
    b = fast_conv(a)
    b_max = b.max()
    exp_b = np.exp(tau * (b - b_max))
    p = exp_b / exp_b.sum()

    grad = np.zeros(n)
    for k in range(2 * n - 1):
        pk = p[k]
        if pk < 1e-13:
            continue
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        grad[lo:hi + 1] += 2.0 * pk * a[k - hi:k - lo + 1][::-1]

    return grad, p, b


def run_smooth_gradient(a, max_iters, deadline, initial_tau=5.0, final_tau=100.0,
                        lr_init=0.2, lr_min=0.01, momentum=True):
    """Smooth gradient descent with adaptive tau and momentum."""
    a = np.maximum(a.astype(np.float64), 1e-8)
    a = a / a.sum()
    n = len(a)

    best_a = a.copy()
    best_M = fast_conv(a).max()
    v = np.zeros(n)

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        frac = it / max(1, max_iters)
        tau = initial_tau + (final_tau - initial_tau) * (frac ** 0.5)
        lr = lr_init * (lr_min / lr_init) ** frac

        grad, p, b = smooth_max_gradient(a, tau)
        M = b.max()

        if M < best_M * (1 - 1e-8):
            best_M = M
            best_a = a.copy()

        beta = 0.9 if momentum else 0.0
        v = beta * v + lr * grad
        a_new = a - v
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s < 1e-10:
            a_new = a.copy()
        else:
            a_new = a_new / s
        a = a_new

    return best_a


def build_A_sparse(n, a):
    """Build sparse constraint matrix for LP: A[k,j] = a[k-j] if valid."""
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        j_min = max(0, k - n + 1)
        j_max = min(n - 1, k)
        for j in range(j_min, j_max + 1):
            rows.append(k)
            cols.append(j)
            vals.append(a[k - j])
    A = csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))
    return A


def lp_best_response(a, M, time_limit=60):
    """LP: max sum(g) s.t. (a*g)[k] <= M for all k, g >= 0."""
    n = len(a)
    A = build_A_sparse(n, a)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    try:
        r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                    options={'time_limit': time_limit})
        if r.success and r.x is not None:
            g = np.maximum(r.x, 0)
            if g.sum() > 1e-15:
                return g
    except Exception:
        pass
    return None


def run_lp_mirror_descent(a, max_iters, deadline, m_scale=1.0):
    """
    LP mirror descent. m_scale allows using a scaled M for the constraint,
    which can help the LP find a more interior point.
    """
    a = np.maximum(a.astype(np.float64), 1e-8)
    a = a / a.sum()
    best_a = a.copy()
    best_M = fast_conv(a).max()
    no_improve = 0

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        b = fast_conv(a)
        M = float(b.max()) * m_scale
        g = lp_best_response(a, M)
        if g is None:
            break
        g = np.maximum(g, 0.0)
        g_sum = g.sum()
        if g_sum < 1e-10:
            break
        g /= g_sum

        improved = False
        base_t = max(0.05, 0.5 * (1.0 - 0.5 * it / max(1, max_iters)))
        for t in [base_t, base_t * 0.6, base_t * 0.3, 0.05, 0.02, 0.01, 0.005]:
            a_trial = np.maximum((1 - t) * a + t * g, 0)
            s = a_trial.sum()
            if s < 1e-10:
                continue
            a_trial /= s
            M_trial = fast_conv(a_trial).max()
            if M_trial < best_M - 1e-14:
                a = a_trial
                best_M = M_trial
                best_a = a.copy()
                improved = True
                break

        if not improved:
            no_improve += 1
            if no_improve > 25:
                break
        else:
            no_improve = 0

    return best_a


def fourier_perturb(a, rng, scale=0.05):
    """Perturb in Fourier domain: modify magnitude and/or phase."""
    n = len(a)
    fft_size = max(2 * n, 256)
    fft_size = 1 << (fft_size - 1).bit_length()

    A = np.fft.fft(a, fft_size)
    mag = np.abs(A)
    phase = np.angle(A)

    # Perturb magnitude
    mag_pert = mag * (1 + rng.normal(0, scale, len(mag)))
    mag_pert = np.maximum(mag_pert, 0)

    # Slightly perturb phase (only for non-DC, non-Nyquist)
    phase_pert = phase + rng.normal(0, scale * 0.5, len(phase))
    phase_pert[0] = 0  # Keep DC real
    if fft_size % 2 == 0:
        phase_pert[fft_size // 2] = 0

    A_new = mag_pert * np.exp(1j * phase_pert)
    a_new = np.fft.ifft(A_new).real[:n]
    return np.maximum(a_new, 0)


def resample_seq(src, dest_n):
    """Resample 1D sequence to new length via linear interpolation."""
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)

    # Gaussian edge concentration
    inits.append(np.exp(-3.0 * (t_arr) ** 2) + np.exp(-3.0 * (t_arr - 1) ** 2) + 0.1)
    inits.append(np.exp(-6.0 * (t_arr) ** 2) + np.exp(-6.0 * (t_arr - 1) ** 2) + 0.05)

    # Smooth random
    ksize = max(3, n // 25)
    kernel = np.ones(ksize) / ksize
    smooth = np.convolve(rng.uniform(0, 1, n), kernel, mode='same')
    inits.append(np.maximum(smooth + 0.1, 0.01))

    # Block structure
    n_blocks = rng.integers(4, max(5, n // 30))
    block_vals = rng.uniform(0.3, 1.0, n_blocks)
    block_seq = np.repeat(block_vals, n // n_blocks + 1)[:n]
    inits.append(block_seq)

    # Sawtooth
    inits.append(np.maximum(0.5 + 0.4 * (2 * (t_arr * 2 % 1) - 1), 0.01))

    return inits


class ScoreTracker:
    def __init__(self):
        self.best = None
        self.best_score = float('inf')
        self.t_start = time.monotonic()

    def update(self, a):
        a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
        if a.size == 0 or a.sum() < 0.01:
            return
        score = float(evaluate_sequence(a.tolist()))
        if np.isfinite(score) and score < self.best_score:
            self.best = a.copy()
            self.best_score = score
            elapsed = time.monotonic() - self.t_start
            print(f"  ** J={score:.8f} n={a.size} t={elapsed:.0f}s", flush=True)

    def report(self, label=""):
        if self.best is not None:
            print(f"  {label} best: J={self.best_score:.8f} n={self.best.size}", flush=True)


def propose_candidate(seed=42, budget_s=1500, **kwargs):
    """
    Multi-phase hybrid search combining IFT spectral flatness with LP mirror descent:
    Phase 1: IFT (Gerchberg-Saxton) multi-scale on multiple lengths
    Phase 2: LP mirror descent for precise saddle-point convergence
    Phase 3: Smooth gradient descent for fine-tuning
    Phase 4: Fourier-domain perturbations + re-optimization (escape local minima)
    Phase 5: Final polish and length extension
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + float(budget_s) - 20.0
    tracker = ScoreTracker()

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        if p.ndim == 1 and p.size > 0 and np.all(np.isfinite(p)):
            parent = np.maximum(p, 0.0)
            tracker.update(parent)
            print(f"Parent loaded: n={parent.size}, J={tracker.best_score:.6f}", flush=True)
    except (NameError, AttributeError):
        pass

    if parent is None:
        parent = np.ones(256)
        tracker.update(parent)

    current_best = parent.copy()

    # === PHASE 1: IFT Spectral Flatness ===
    print(f"\n{'='*60}", flush=True)
    print(f"PHASE 1: IFT Spectral Flatness (multi-length, multi-init)", flush=True)
    print(f"{'='*60}", flush=True)

    p1_start = time.monotonic()
    p1_budget = 350.0

    # Determine lengths to try
    if parent is not None and parent.size > 0:
        n_par = parent.size
        lengths = sorted(set([
            max(64, n_par // 2),
            max(64, n_par // 3),
            n_par,
            min(2048, n_par * 2),
            min(2048, n_par * 3),
            128, 256, 384, 512, 768, 1024
        ]))
    else:
        lengths = [128, 256, 384, 512, 768, 1024]
    lengths = [n for n in lengths if 64 <= n <= 2048]

    for n in lengths:
        if time.monotonic() > deadline - 250:
            break
        if time.monotonic() - p1_start > p1_budget:
            break

        time_per_n = min(50.0, p1_budget / max(1, len(lengths)))
        phase_deadline = min(time.monotonic() + time_per_n, deadline - 200)

        if time.monotonic() >= phase_deadline:
            continue

        t0 = time.monotonic()
        inits = generate_initializations(n, rng, current_best)
        best_for_n = None
        best_M_for_n = float('inf')

        for idx, a0 in enumerate(inits):
            if time.monotonic() > phase_deadline - 5:
                break

            a0 = np.maximum(a0.astype(np.float64), 1e-6)
            a0 = a0 / a0.sum()

            # Hard IFT
            a_ilt = ilt_optimize(a0, max_iters=400, deadline=phase_deadline - 3)
            M_val = fast_conv(a_ilt).max()
            if M_val < best_M_for_n:
                best_M_for_n = M_val
                best_for_n = a_ilt.copy()

            # Multi-scale IFT (different trajectory)
            if time.monotonic() < phase_deadline - 10:
                a_ms = ilt_multi_scale(a0, max_iters=400, deadline=phase_deadline - 3)
                M_ms = fast_conv(a_ms).max()
                if M_ms < best_M_for_n:
                    best_M_for_n = M_ms
                    best_for_n = a_ms.copy()

        if best_for_n is not None:
            J_val = compute_J(best_for_n)
            print(f"  n={n:4d}: M={best_M_for_n:.8f} J={J_val:.6f} "
                  f"({time.monotonic()-t0:.1f}s)", flush=True)
            if J_val < compute_J(current_best):
                current_best = best_for_n.copy()
            tracker.update(best_for_n)

    tracker.report("Phase 1 complete")
    print(f"Phase 1 time: {time.monotonic() - p1_start:.1f}s", flush=True)

    # === PHASE 2: LP Mirror Descent ===
    print(f"\n{'='*60}", flush=True)
    print(f"PHASE 2: LP Mirror Descent", flush=True)
    print(f"{'='*60}", flush=True)

    p2_start = time.monotonic()
    p2_budget = 300.0

    if tracker.best is not None:
        n_best = tracker.best.size
        est_lp_time = 0.01 * (n_best / 128) ** 2
        max_iters = max(20, int(p2_budget / max(0.02, est_lp_time)))
        print(f"  n={n_best}, est LP time={est_lp_time*1000:.0f}ms, max_iters={max_iters}", flush=True)

        a = run_lp_mirror_descent(tracker.best, max_iters,
                                   min(time.monotonic() + p2_budget, deadline - 120))
        tracker.update(a)

        # Second LP pass with slightly relaxed constraint
        if time.monotonic() < deadline - 100:
            a2 = run_lp_mirror_descent(a, min(30, max_iters // 3),
                                       min(time.monotonic() + p2_budget * 0.4, deadline - 100),
                                       m_scale=1.001)
            tracker.update(a2)

    tracker.report("Phase 2 complete")
    print(f"Phase 2 time: {time.monotonic() - p2_start:.1f}s", flush=True)

    # === PHASE 3: Smooth Gradient Descent ===
    print(f"\n{'='*60}", flush=True)
    print(f"PHASE 3: Smooth Gradient Descent", flush=True)
    print(f"{'='*60}", flush=True)

    p3_start = time.monotonic()
    p3_budget = 150.0

    if tracker.best is not None:
        # Multiple runs with different tau schedules
        for tau_start, tau_end, lr_s in [(5, 80, 0.2), (2, 200, 0.1), (10, 150, 0.15)]:
            if time.monotonic() > deadline - 80:
                break
            a = run_smooth_gradient(tracker.best, 4000,
                                     min(time.monotonic() + p3_budget / 3, deadline - 80),
                                     initial_tau=tau_start, final_tau=tau_end,
                                     lr_init=lr_s, lr_min=0.008)
            tracker.update(a)

    tracker.report("Phase 3 complete")
    print(f"Phase 3 time: {time.monotonic() - p3_start:.1f}s", flush=True)

    # === PHASE 4: Fourier-Domain Perturbations + Re-optimize ===
    print(f"\n{'='*60}", flush=True)
    print(f"PHASE 4: Fourier Perturbations + Re-optimize", flush=True)
    print(f"{'='*60}", flush=True)

    p4_start = time.monotonic()
    p4_budget = 200.0

    if tracker.best is not None:
        n_best = tracker.best.size
        temperature = 0.15

        for r in range(40):
            if time.monotonic() > deadline - 40:
                break
            if time.monotonic() - p4_start > p4_budget:
                break

            a = tracker.best.copy().astype(np.float64)
            strategy = r % 4

            if strategy == 0:
                # Fourier magnitude perturbation
                a = fourier_perturb(a, rng, scale=temperature * 0.1)
            elif strategy == 1:
                # Gaussian spatial perturbation
                a = a + rng.normal(0, temperature * 0.04, n_best)
            elif strategy == 2:
                # Block perturbation
                blk = rng.integers(10, max(11, n_best // 8))
                start = rng.integers(0, n_best - blk)
                a[start:start + blk] += rng.normal(0, temperature * 0.08, blk)
            else:
                # Edge-weighted perturbation
                weights = np.abs(2 * np.linspace(0, 1, n_best) - 1)
                a = a + rng.normal(0, temperature * 0.05, n_best) * weights

            a = np.maximum(a, 0.001)
            a = a / a.sum()

            # Quick IFT to recover spectral flatness
            a = ilt_optimize(a, max_iters=80, deadline=min(time.monotonic() + 8, deadline - 40))
            tracker.update(a)

            # Short LP polish
            if time.monotonic() < deadline - 50:
                a = run_lp_mirror_descent(a, 8, min(time.monotonic() + 15, deadline - 50))
                tracker.update(a)

            temperature *= 0.96

    tracker.report("Phase 4 complete")
    print(f"Phase 4 time: {time.monotonic() - p4_start:.1f}s", flush=True)

    # === PHASE 5: IFT on best + final LP + length extension ===
    print(f"\n{'='*60}", flush=True)
    print(f"PHASE 5: Final Polish", flush=True)
    print(f"{'='*60}", flush=True)

    p5_start = time.monotonic()

    if tracker.best is not None and time.monotonic() < deadline - 10:
        # IFT on current best for one more pass
        a = ilt_multi_scale(tracker.best, max_iters=300, deadline=min(time.monotonic() + 40, deadline - 30))
        tracker.update(a)

        # Final LP
        if time.monotonic() < deadline - 15:
            a = run_lp_mirror_descent(tracker.best, 20, deadline)
            tracker.update(a)

        # Try 2x length extension
        if time.monotonic() < deadline - 10:
            new_n = tracker.best.size * 2
            if new_n <= 3072:
                a_ext = resample_seq(tracker.best, new_n)
                a_ext = np.maximum(a_ext, 1e-6)
                a_ext = a_ext / a_ext.sum()
                a_ext = ilt_optimize(a_ext, max_iters=200, deadline=deadline)
                tracker.update(a_ext)

    # Fallback
    if tracker.best is None:
        tracker.best = np.ones(128)
        tracker.best_score = compute_J(tracker.best)

    print(f"\n{'='*60}", flush=True)
    print(f"FINAL: J={tracker.best_score:.8f} n={tracker.best.size} "
          f"time={time.monotonic() - tracker.t_start:.0f}s", flush=True)
    print(f"{'='*60}", flush=True)

    return [float(x) for x in tracker.best]
