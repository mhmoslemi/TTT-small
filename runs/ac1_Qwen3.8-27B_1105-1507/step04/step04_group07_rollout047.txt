            cols.extend(range(js, je + 1))
        _spar_cache[n] = (np.array(rows), np.array(cols), len(rows))
    return _spar_cache[n]

def fast_conv(a):
    n = len(a)
    s = 1
    while s < 2 * n - 1:
        s <<= 1
    fa = np.fft.rfft(a, s)
    return np.fft.irfft(fa * fa, s)[:2 * n - 1]

def compute_J(a):
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))

def smooth_grad(a, beta):
    """Compute smoothed gradient of max(a*a) using softmax weighting."""
    n = len(a)
    b = fast_conv(a)
    b_shifted = b - b.max()
    p = np.exp(beta * b_shifted)
    p_sum = p.sum()
    if p_sum < 1e-300:
        p = np.zeros(2 * n - 1)
        p[int(np.argmax(b))] = 1.0
    else:
        p /= p_sum
    # grad[j] = sum_k p[k] * a[k-j] = (p cross-corr a)[j]
    size = 1
    while size < 3 * n:
        size <<= 1
    fp = np.fft.fft(p, size)
    fa = np.fft.fft(a, size)
    corr = np.fft.ifft(fp * np.conj(fa)).real
    return corr[:n]

def lp_solve(a, M, time_limit=10):
    """Solve LP: max 1^T g s.t. A g <= M, g >= 0 where (Ag)_k = (a*g)_k."""
    n = len(a)
    rows, cols, nnz = get_sparsity(n)
    vals = np.empty(nnz)
    for i in range(nnz):
        k = rows[i]
        j = cols[i]
        vals[i] = a[k - j]
    A = csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                options={'presolve': True, 'time_limit': time_limit})
    if r.success and r.x is not None:
        g = np.maximum(r.x, 0.0)
        if g.sum() > 1e-10:
            return g
    return None

def run_lp_opt(a, max_iters, deadline, tracker, tighten_start=0.005, tighten_end=0.04):
    """LP best-response iteration with adaptive tightening."""
    a = np.maximum(a.astype(np.float64), 1e-10)
    a /= a.sum()
    n = len(a)
    current_J = compute_J(a)
    tracker.update(a)

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        b = fast_conv(a)
        M = float(b.max())
        # Adaptive tightening: start loose, get tighter
        progress = it / max(1, max_iters)
        tighten = tighten_start + (tighten_end - tighten_start) * progress
        M_try = M * (1.0 - tighten)
        g = lp_solve(a, M_try, time_limit=min(15, 40 - 30 * progress))
        if g is None:
            g = lp_solve(a, M * 0.99, time_limit=10)
        if g is None:
            break
        g = np.maximum(g, 0.0)
        gs = g.sum()
        if gs < 1e-10:
            break
        g /= gs

        # Line search: try multiple step sizes
        best_a = None
        best_J = current_J
        progress = it / max(1, max_iters)
        base_t = max(0.1, 0.8 * (1.0 - progress))
        for t in [base_t, base_t * 0.5, base_t * 1.5, 0.3, 0.6, 0.9, 0.1]:
            if t <= 0 or t > 1.0:
                continue
            a_trial = (1.0 - t) * a + t * g
            a_trial = np.maximum(a_trial, 1e-10)
            ss = a_trial.sum()
            if ss < 1e-10:
                continue
            a_trial /= ss
            J_trial = compute_J(a_trial)
            if J_trial < best_J - 1e-14:
                best_J = J_trial
                best_a = a_trial
        if best_a is None:
            # Try pure mirror descent step as fallback
            break
        a = best_a
        current_J = best_J
        if it % 5 == 0:
            tracker.update(a)
    tracker.update(a)
    return a

def mirror_descent(a, max_iters, deadline, tracker, beta_start=20.0, beta_end=200.0):
    """Mirror descent with multiplicative updates on the simplex."""
    a = np.maximum(a.astype(np.float64), 1e-10)
    a /= a.sum()
    n = len(a)
    current_J = compute_J(a)
    tracker.update(a)

    eta = 0.5 / np.sqrt(n)
    rng = np.random.default_rng(123)

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        # Adaptive beta
        progress = it / max(1, max_iters)
        beta = beta_start + (beta_end - beta_start) * progress

        grad = smooth_grad(a, beta)
        # Center the gradient (remove mean for simplex)
        grad = grad - grad.mean()

        # Multiplicative update: a_j <- a_j * exp(-eta * grad_j)
        scale = np.exp(-eta * grad)
        a_new = a * scale
        a_new = np.maximum(a_new, 1e-15)
        a_new /= a_new.sum()

        J_new = compute_J(a_new)
        if J_new < current_J - 1e-12:
            a = a_new
            current_J = J_new
            eta = min(eta * 1.05, 5.0 / np.sqrt(n))
        else:
            eta *= 0.7
            # Try smaller step
            scale = np.exp(-eta * grad)
            a_new = a * scale
            a_new = np.maximum(a_new, 1e-15)
            a_new /= a_new.sum()
            J_new = compute_J(a_new)
            if J_new < current_J - 1e-12:
                a = a_new
                current_J = J_new
            else:
                if eta < 1e-10:
                    break

        if it % 20 == 0:
            tracker.update(a)
    tracker.update(a)
    return a

def lse_descent(a, max_iters, deadline, tracker):
    """LSE (Log-Sum-Exp) smoothed gradient descent with momentum."""
    a = np.maximum(a.astype(np.float64), 1e-8)
    a /= a.sum()
    n = len(a)
    current_J = compute_J(a)
    tracker.update(a)

    tau = 30.0
    lr = 1.0 / (n * tau)
    momentum = 0.8
    v = np.zeros(n)

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        b = fast_conv(a)
        b_max = b.max()
        exp_b = np.exp(tau * (b - b_max))
        p = exp_b / exp_b.sum()

        # Gradient via cross-correlation
        size = 1
        while size < 3 * n:
            size <<= 1
        fp = np.fft.fft(p, size)
        fa = np.fft.fft(a, size)
        corr = np.fft.ifft(fp * np.conj(fa)).real
        grad = 2.0 * corr[:n]

        v = momentum * v + (1.0 - momentum) * grad
        a_new = a - lr * v
        a_new = np.maximum(a_new, 1e-8)
        a_new /= a_new.sum()
        J_new = compute_J(a_new)
        if J_new < current_J - 1e-12:
            a = a_new
            current_J = J_new
            tau = min(500.0, tau * 1.02)
            lr = 1.0 / (n * tau)
        else:
            lr *= 0.6
            v *= 0.5
            if lr < 1e-18:
                break
        if it % 50 == 0:
            tracker.update(a)
    tracker.update(a)
    return a

def flat_spectrum_init(n, rng):
    """Generate sequence with approximately flat power spectrum."""
    half = n // 2 + 1
    angles = rng.uniform(0, 2 * np.pi, half)
    A = np.zeros(n, dtype=complex)
    A[:half] = np.exp(1j * angles[:half])
    for k in range(1, n - n // 2):
        A[k] = np.conj(A[n - k])
    a = np.fft.ifft(A).real
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-10:
        return np.ones(n) / n
    return a / s

def resample_seq(src, dest_n):
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)

def generate_inits(n, rng, parent):
    inits = []
    if parent is not None and parent.size > 0:
        inits.append(resample_seq(parent, n))
        inits.append(resample_seq(parent[::-1], n))

    inits.append(np.ones(n) / n)
    inits.append(rng.uniform(0.1, 1.0, n))
    inits.append(flat_spectrum_init(n, rng))
    inits.append(flat_spectrum_init(n, rng))

    t = np.linspace(0, 1, n, endpoint=False)
    for freq in [2, 3, 5, 7, 11]:
        inits.append(1.0 + 0.3 * np.cos(2 * np.pi * freq * t + rng.uniform(0, 6.28)))

    inits.append(1.0 + 0.2 * np.cos(2 * np.pi * 3.7 * t) + 0.15 * np.cos(2 * np.pi * 7.3 * t))
    inits.append(0.5 + 0.5 * np.cos(np.pi * t) ** 2)

    # Multi-frequency
    inits.append(1.0 + 0.2 * np.sin(2 * np.pi * t) + 0.15 * np.sin(4 * np.pi * t)
                + 0.1 * np.sin(7 * np.pi * t))

    # Random Fourier with limited bandwidth
    n_coeff = min(n // 2 + 1, 15)
    coeffs = rng.normal(0, 1, n_coeff)
    sig = np.zeros(n)
    for i in range(n_coeff):
        sig += coeffs[i] * np.cos(2 * np.pi * i * t / n + rng.uniform(0, 6.28))
    inits.append(np.abs(sig) + 0.3)

    # Sparse + smooth
    sparse = np.zeros(n)
    n_spikes = max(10, n // 10)
    positions = rng.choice(n, size=min(n_spikes, n), replace=False)
    sparse[positions] = 1.0
    kernel_size = max(5, n // 25)
    kernel = np.ones(kernel_size) / kernel_size
    inits.append(np.convolve(sparse, kernel, mode='same') + 0.03)

    for x in inits:
        x = np.maximum(np.asarray(x, dtype=np.float64), 0.001)
        s = x.sum()
        if s > 0:
            x /= s
    return inits

class ScoreTracker:
    def __init__(self):
        self.best = None
        self.best_score = float('inf')
        self.t_start = time.monotonic()

    def update(self, a):
        a_clip = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
        if a_clip.size == 0 or a_clip.sum() < 0.01:
            return
        score = float(evaluate_sequence(a_clip.tolist()))
        if np.isfinite(score) and score < self.best_score:
            self.best = a_clip.copy()
            self.best_score = score
            elapsed = time.monotonic() - self.t_start
            print(f"  * J={score:.8f} n={a_clip.size} t={elapsed:.1f}s", flush=True)

    def report(self, label=""):
        if self.best is not None:
            print(f"  {label} best: J={self.best_score:.8f} n={self.best.size}", flush=True)

def perturb_and_refine(a, deadline, tracker, rng, n_perturbs=20):
    """Perturb the best solution and re-optimize to escape local optima."""
    n = a.size
    a_best = a.copy()
    J_best = compute_J(a_best)

    for i in range(n_perturbs):
        if time.monotonic() > deadline:
            break
        temp = 0.08 * (1.0 - i / n_perturbs)
        a_pert = a_best + rng.normal(0, temp, n)
        a_pert = np.maximum(a_pert, 1e-8)
        a_pert /= a_pert.sum()

        # Quick LP refinement
        a_ref = run_lp_opt(a_pert, 8, deadline, tracker, tighten_start=0.01, tighten_end=0.02)

        # Quick mirror descent
        if time.monotonic() < deadline:
            a_ref = mirror_descent(a_ref, 100, deadline, tracker)

        J_ref = compute_J(a_ref)
        if J_ref < J_best:
            a_best = a_ref
            J_best = J_ref
            tracker.update(a_best)
    return a_best

def coordinate_polish(a, max_steps, deadline, tracker):
    """Coordinate descent with random coordinate selection."""
    a = a.copy().astype(np.float64)
    a /= a.sum()
    n = len(a)
    current_J = compute_J(a)
    tracker.update(a)
    rng = np.random.default_rng(42)

    for step in range(max_steps):
        if time.monotonic() > deadline:
            break
        j = rng.integers(0, n)
        improved = False
        for scale in [0.7, 1.3, 0.5, 1.5, 0.85, 1.15]:
            a_trial = a.copy()
            a_trial[j] *= scale
            a_trial = np.maximum(a_trial, 1e-10)
            ss = a_trial.sum()
            if ss < 1e-10:
                continue
            a_trial /= ss
            J_trial = compute_J(a_trial)
            if J_trial < current_J - 1e-13:
                a = a_trial
                current_J = J_trial
                improved = True
                break
        if not improved and step % 200 == 0:
            tracker.update(a)
    tracker.update(a)
    return a

def propose_candidate(seed=42, budget_s=1500, **kwargs):
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + float(budget_s) - 30.0
    tracker = ScoreTracker()

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        if p.ndim == 1 and p.size > 0 and np.all(np.isfinite(p)):
            parent = np.maximum(p, 0.0)
            tracker.update(parent)
            print(f"Parent: n={parent.size}, J={tracker.best_score:.8f}", flush=True)
    except (NameError, AttributeError):
        pass

    # === Phase 1: Multi-length exploration ===
    lengths = [384, 512, 640, 768]
    if parent is not None:
        lengths.insert(0, parent.size)
    # Remove duplicates while preserving order
    seen = set()
    unique_lengths = []
    for L in lengths:
        if L not in seen:
            seen.add(L)
            unique_lengths.append(L)
    lengths = unique_lengths

    time_per_length = (deadline - time.monotonic() - 400) / len(lengths)
    best_overall = None
    best_J_overall = float('inf')
    best_len = None

    for n in lengths:
        t_phase_start = time.monotonic()
        t_phase_end = min(deadline - 350, t_phase_start + time_per_length + 30)
        if time.monotonic() >= t_phase_end - 30:
            continue
        print(f"\n{'='*40} Phase1 n={n} {'='*40}", flush=True)

        inits = generate_inits(n, rng, parent)
        best_for_n = None
        best_J_for_n = float('inf')
        n_inits = len(inits)

        # Limit inits based on available time
        max_inits = max(2, min(n_inits, int((t_phase_end - time.monotonic()) / 15)))

        for idx in range(min(max_inits, n_inits)):
            if time.monotonic() > t_phase_end - 15:
                break
            a0 = inits[idx]
            # LP phase
            n_lp = max(8, int(40.0 / max(1, n / 384)))
            a = run_lp_opt(a0, n_lp, t_phase_end - 30, tracker, tighten_start=0.005, tighten_end=0.03)
            # Mirror descent
            if time.monotonic() < t_phase_end - 20:
                a = mirror_descent(a, max(50, int(20000 / n)), t_phase_end - 20, tracker)

            J = compute_J(a)
            if J < best_J_for_n:
                best_J_for_n = J
                best_for_n = a.copy()

        if best_for_n is not None:
            tracker.update(best_for_n)
            if best_J_for_n < best_J_overall:
                best_J_overall = best_J_for_n
                best_overall = best_for_n.copy()
                best_len = n
            tracker.report(f"Phase1 n={n} ({time.monotonic()-t_phase_start:.1f}s): J={best_J_for_n:.8f}")

    # === Phase 2: Focused optimization on best ===
    if best_overall is not None and time.monotonic() < deadline - 200:
        t0 = time.monotonic()
        n = best_overall.size
        print(f"\n{'='*40} Phase2 Focus n={n} {'='*40}", flush=True)

        a = best_overall.copy().astype(np.float64)
        # Multiple LP passes with increasing tightening
        tighten_sched = [0.01, 0.015, 0.02, 0.025, 0.03, 0.035, 0.04, 0.045]
        for tight in tighten_sched:
            if time.monotonic() > deadline - 200:
                break
            a = run_lp_opt(a, 20, deadline - 200, tracker, tighten_start=tight * 0.5, tighten_end=tight)

        # Mirror descent
        if time.monotonic() < deadline - 150:
            a = mirror_descent(a, 2000, deadline - 150, tracker, beta_start=30, beta_end=300)

        # LSE polish
        if time.monotonic() < deadline - 100:
            a = lse_descent(a, 3000, deadline - 100, tracker)

        tracker.report(f"Phase2 ({time.monotonic()-t0:.1f}s): J={compute_J(a):.8f}")

    # === Phase 3: Perturbation to escape local optima ===
    if tracker.best is not None and time.monotonic() < deadline - 100:
        t0 = time.monotonic()
        n = tracker.best.size
        print(f"\n{'='*40} Phase3 Perturb n={n} {'='*40}", flush=True)
        a = perturb_and_refine(tracker.best, deadline - 80, tracker, rng, n_perturbs=30)
        # Polish after perturbation
        if time.monotonic() < deadline - 40:
            a = run_lp_opt(a, 15, deadline - 30, tracker, tighten_start=0.01, tighten_end=0.03)
            if time.monotonic() < deadline - 20:
                a = mirror_descent(a, 500, deadline - 20, tracker)
        tracker.report(f"Phase3 ({time.monotonic()-t0:.1f}s)")

    # === Phase 4: Final polishing ===
    if tracker.best is not None and time.monotonic() < deadline - 15:
        t0 = time.monotonic()
        print(f"\n{'='*40} Phase4 Polish {'='*40}", flush=True)
        a = coordinate_polish(tracker.best, 2000, deadline - 10, tracker)
        if time.monotonic() < deadline - 5:
            a = run_lp_opt(a, 10, deadline - 3, tracker, tighten_start=0.005, tighten_end=0.015)
            if time.monotonic() < deadline - 2:
                a = lse_descent(a, 1000, deadline - 1, tracker)
        tracker.report(f"Phase4 ({time.monotonic()-t0:.1f}s)")

    # === Phase 5: Try a different length if time remains ===
    if time.monotonic() < deadline - 60:
        t0 = time.monotonic()
        alt_n = 640 if best_len != 640 else 896
        print(f"\n{'='*40} Phase5 Alt n={alt_n} {'='*40}", flush=True)
        inits = generate_inits(alt_n, rng, tracker.best)
        for a0 in inits[:4]:
            if time.monotonic() > deadline - 20:
                break
            a = run_lp_opt(a0, 15, deadline - 10, tracker, tighten_start=0.01, tighten_end=0.03)
            a = mirror_descent(a, 200, deadline - 5, tracker)
        tracker.report(f"Phase5 ({time.monotonic()-t0:.1f}s)")

    # Fallback
    if tracker.best is None:
        tracker.best = np.ones(256) / 256
        tracker.best_score = compute_J(tracker.best)
        tracker.update(tracker.best)

    print(f"\n{'='*40} FINAL {'='*40}", flush=True)
    print(f"J={tracker.best_score:.8f} n={tracker.best.size} time={time.monotonic()-tracker.t_start:.1f}s", flush=True)
    return [float(x) for x in tracker.best]
