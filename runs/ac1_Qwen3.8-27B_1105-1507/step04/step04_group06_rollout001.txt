def fast_conv(a):
    """Full self-convolution via FFT."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]

def compute_J(a):
    """Compute J(a) = 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    s = a.sum()
    if s < 1e-10:
        return float('inf')
    b = fast_conv(a)
    return float(2 * n * b.max() / (s * s))

def compute_J_and_conv(a):
    """Compute J and the full convolution."""
    n = len(a)
    s = a.sum()
    b = fast_conv(a)
    J = float(2 * n * b.max() / (s * s))
    return J, b

def resample_seq(src, dest_n):
    """Resample 1D sequence to new length via linear interpolation."""
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)

def build_A_matrix(a):
    """Build the convolution constraint matrix A where (A @ g)_k = (a*g)_k."""
    n = len(a)
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):

        jh = min(n - 1, k)

            rows.append(k)
            cols.append(j)
            vals.append(a[k - j])
    return csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))

def lp_best_response_v2(a, M, top_k=10, eps=0.05):
    """
    LP best response focusing on top-k active constraints.
    max sum(g) s.t. (a*g)_k <= M for k in active set, g >= 0
    """
    n = len(a)
    # Find active constraints (near-max)
    b = fast_conv(a)
    b_max = float(b.max())
    threshold = b_max * (1.0 - eps)
    active_mask = b >= threshold
    
    # If too many active, take top_k
    if active_mask.sum() > top_k:
        top_idx = np.argsort(b)[-top_k:]
        active_mask = np.zeros(2 * n - 1, dtype=bool)
        active_mask[top_idx] = True
    
    # Build sparse matrix for active constraints only
    active_rows = np.where(active_mask)[0]
    if len(active_rows) == 0:
        active_rows = np.array([np.argmax(b)])
    
    # Build A submatrix
    A_full = build_A_matrix(a)
    A_sub = A_full[active_rows, :]
    
    c = -np.ones(n)
    b_ub = np.full(len(active_rows), M)
    bounds = [(0.0, None)] * n
    
    r = linprog(c, A_ub=A_sub, b_ub=b_ub, bounds=bounds, method='highs')
    if r.success and r.x is not None:
        g = np.maximum(r.x, 0.0)
        s = g.sum()
        if s > 1e-10:
            return g / s, r
    return None, None

def dual_guided_direction(a, M, top_k=5):
    """
    Use dual variables from LP to construct a descent direction.
    The dual gives weights on each constraint; high-weight constraints
    are the binding ones we should focus on reducing.
    """
    n = len(a)
    g, r = lp_full_best_response(a, M)
    if g is None:
        return None
    
    # Get dual values (shadow prices)
    if r is not None and r.ineqlin is not None:
        dual = np.abs(r.ineqlin)
    else:
        dual = np.zeros(2 * n - 1)
    
    # Construct descent direction from dual-weighted subgradients
    # subgradient of (a*a)_k w.r.t. a is: for each j, a_{k-j} (where valid)
    # This is: convolve(a, delta_k) = shift of a
    # More precisely: d(a*a)/da at position k gives vector where element j = a_{k-j} (if 0<=k-j<n)
    
    # The descent direction is: -sum_k dual_k * d(a*a)_k / da
    # d(a*a)_k / da[j] = a_{k-j} if 0 <= k-j < n, else 0
    # So direction[j] = -sum_k dual_k * a_{k-j} = -convolve(dual, a)[j + n - 1]... 
    
    # Actually let me think more carefully
    # (a*a)_k = sum_j a_j * a_{k-j}
    # d(a*a)_k / da_j = a_{k-j} (if 0 <= k-j < n)
    # So the gradient of max_k (a*a)_k w.r.t. a is:
    # grad_j = sum_{k in argmax} a_{k-j}
    
    # With dual weights:
    # direction_j = sum_k dual_k * a_{k-j}
    # This is (dual * a_reversed) convolved... let me just compute directly
    
    # direction[j] = sum_k dual[k] * a[k-j] for valid indices
    # = sum_k dual[k] * a[k-j] where 0 <= k-j < n
    # = sum_{j} dual[j+m] * a[m] for m = k-j
    # This is cross-correlation of dual and a
    
    # Simpler: use FFT
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    
    # direction[j] = sum_k dual[k] * a[k-j] = (dual convolve a)[j + (n-1)]... 
    # Actually: if we define c[m] = sum_j dual[j] * a[m-j], then c = convolve(dual, a)
    # And direction[j] = sum_k dual[k] * a[k-j] = c[j] where c = convolve(dual, a)
    # Wait: c[m] = sum_j dual[j] * a[m-j]
    # direction[j] = sum_k dual[k] * a[k-j]
    # Let m = k-j, then k = m+j: direction[j] = sum_m dual[m+j] * a[m]
    # This is cross-correlation: direction = correlate(dual, a)
    
    # Using FFT: correlate(x, y) = ifft(conj(fft(x)) * fft(y))
    fd = np.fft.rfft(dual, size)
    fa = np.fft.rfft(a, size)
    direction = np.fft.irfft(fd * fa, size)
    # Take the first n elements (or appropriate slice)
    # correlate(dual, a)[j] = sum_m dual[m+j] * a[m]
    # The full correlate has length 2n-1, we want indices 0 to n-1
    direction = direction[:n]
    direction = -np.abs(direction)  # descent direction
    
    return g, direction


def smooth_max_and_grad(a, beta):
    """Compute smooth max of conv(a,a) and its gradient w.r.t. a."""
    n = len(a)
    b = fast_conv(a)
    b_max = float(b.max())
    shifted = beta * (b - b_max)
    shifted = np.clip(shifted, -500, 500)
    exp_shifted = np.exp(shifted)
    Z = float(exp_shifted.sum())
    if Z < 1e-300:
        p = np.zeros(2 * n - 1)
        p[np.argmax(b)] = 1.0
    else:
        p = exp_shifted / Z
    
    g_val = (b_max + np.log(max(Z, 1e-300))) / beta
    
    # Gradient: d/d a_j of (a*a)_k = a_{k-j} (if valid)
    # So grad_j = sum_k p[k] * a_{k-j} = correlate(p, a)[j]
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fp = np.fft.rfft(p, size)
    fa = np.fft.rfft(a, size)
    grad = np.fft.irfft(fp * fa, size)[:n]
    grad = 2.0 * grad  # factor of 2 from self-convolution symmetry
    
    return g_val, grad


def F_and_grad(a, beta):
    """Compute smooth objective and gradient."""
    n = len(a)
    s = float(a.sum())
    if s < 1e-10:
        return 1e10, np.zeros(n)
    g_val, grad_g = smooth_max_and_grad(a, beta)
    F_val = 2.0 * n * g_val / (s * s)
    # d/da [2n * G / s^2] = 2n * (s * grad_G - 2 * G) / s^3
    grad_F = 2.0 * n * (s * grad_g - 2.0 * g_val) / (s ** 3)
    return F_val, grad_F


def lbfgs_smooth(a0, beta_schedule, deadline):
    """L-BFGS-B with beta annealing."""
    a = np.maximum(a0.astype(np.float64), 1e-8)
    n = len(a)
    
    for beta, max_iters in beta_schedule:
        if time.monotonic() > deadline:
            break
        
        def func(x):
            x2 = np.maximum(x, 0.0)
            f, _ = F_and_grad(x2, beta)
            return f if np.isfinite(f) else 1e10
        
        def jac(x):
            x2 = np.maximum(x, 0.0)
            _, g = F_and_grad(x2, beta)
            g = np.nan_to_num(g, nan=0.0, posinf=0.0, neginf=0.0)
            g[x2 < 1e-12] = 0.0
            return g
        
        bounds = [(0.0, None)] * n
        try:
            res = minimize(func, a.copy(), jac=jac, method='L-BFGS-B',
                           bounds=bounds,
                           options={'maxiter': max_iters, 'max_fun': max_iters * 30,
                                    'gtol': 1e-10, 'ftol': 1e-14, 'maxls': 50})
            if res.fun < 1e10:
                new_a = np.maximum(res.x, 0.0)
                if new_a.sum() > 1e-6:
                    a = new_a
        except Exception:
            pass
    
    return a

def peak_flattening_step(a, n_steps=3, deadline=None):
    """
    Directly target the top peaks of the convolution.
    For each peak, compute the direction that reduces it most, then combine.
    """
    a = a.copy()
    n = len(a)
    
    for step in range(n_steps):
        if deadline is not None and time.monotonic() > deadline:
            break
        
        b = fast_conv(a)
        b_max = float(b.max())
        
        # Find top peaks
        top_idx = np.argsort(b)[-5:]
        
        # For each peak, compute subgradient
        direction = np.zeros(n)
        for pk in top_idx:
            weight = b[pk] - b_max * 0.99  # weight by how close to max
            weight = max(weight, 0.1)
            for j in range(n):
                k_minus_j = pk - j
                if 0 <= k_minus_j < n:
                    direction[j] += weight * a[k_minus_j]
        
        direction *= 2.0  # factor from self-conv
        direction = -direction  # descent
        
        # Normalize and take step
        dir_norm = np.linalg.norm(direction)
        if dir_norm < 1e-10:
            break
        direction = direction / dir_norm
        
        # Line search
        best_t = 0.0
        best_J = compute_J(a)
        for t_try in [0.001, 0.003, 0.01, 0.03, 0.1]:
            a_try = a + t_try * direction * a.sum() / n
            a_try = np.maximum(a_try, 0)
            s = a_try.sum()
            if s < 1e-10:
                continue
            a_try = a_try / s
            J_try = compute_J(a_try)
            if J_try < best_J:
                best_J = J_try
                best_t = t_try
        
        if best_t > 0:
            a = a + best_t * direction * a.sum() / n
            a = np.maximum(a, 0)
            a = a / a.sum()
        else:
            break
    
    return a


def coordinate_pushdown(a, n_rounds=5, deadline=None):
    """
    For each coordinate, find the optimal value that minimizes the max convolution.
    This is a block coordinate method.
    """
    a = a.copy()
    n = len(a)
    
    for rnd in range(n_rounds):
        if deadline is not None and time.monotonic() > deadline:
            break
        
        # Shuffle order for diversity
        for j in range(n):
            if deadline is not None and time.monotonic() > deadline and (j % 100 == 0):
                break

def generate_diverse_inits(n, rng, parent=None):
    """Generate diverse initial sequences."""
    inits = []
    t = np.linspace(0, 1, n, endpoint=False)
    
    if parent is not None and parent.size > 0:
        inits.append(resample_seq(parent, n))
        p = resample_seq(parent, n)
        inits.append(np.maximum(p + rng.normal(0, 0.05, n), 0.01))
        inits.append(np.maximum(p + rng.normal(0, 0.1, n), 0.01))
    
    # Uniform
    inits.append(np.ones(n))
    
    # Random smooth
    inits.append(np.maximum(rng.uniform(0.5, 1.5, n), 0.1))
    
    # Cosine
    for freq in [1, 2, 3, 4, 5, 7, 11]:
        inits.append(1.0 + 0.3 * np.cos(2 * np.pi * freq * t))
    
    # Multisine
    inits.append(1.0 + 0.2 * np.cos(2 * np.pi * 3 * t) + 0.1 * np.cos(2 * np.pi * 7 * t))
    
    # Gaussian
    inits.append(np.exp(-3 * (t - 0.5) ** 2) + 0.1)
    
    # Absolute sine
    inits.append(np.abs(np.sin(np.pi * t)) + 0.2)
    
    # Power law
    inits.append((t + 0.05) ** 0.5)
    
    # Wigner semicircle
    inits.append(np.sqrt(np.maximum(1 - 4 * (t - 0.5) ** 2, 0)) + 0.1)
    
    # Ramanujan-like (not really, but structured)
    inits.append(1.0 + 0.15 * np.cos(2 * np.pi * 2 * t) - 0.1 * np.cos(2 * np.pi * 5 * t))
    
    return inits
