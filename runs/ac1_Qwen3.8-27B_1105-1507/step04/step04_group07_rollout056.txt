"""
Minimizes J(a) = 2*n*max(convolve(a,a))/(sum(a))^2 using:
1. LP best-response with adaptive tightening and warm-start
2. Mirror descent (multiplicative updates) with exact subgradient

import numpy as np
from scipy.optimize import linprog
from scipy.sparse import csr_matrix, coo_matrix
import time

# Cache for sparse matrix construction
_spar_cache = {}

def get_conv_matrix(n):
    """Build sparse representation of the convolution constraint matrix."""
    if n not in _spar_cache:
        rows = []
        cols = []
        for k in range(2 * n - 1):
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            rows.extend([k] * (je - js + 1))
            cols.extend(range(js, je + 1))
        nz = len(rows)
        _spar_cache[n] = (np.array(rows, dtype=np.int32), np.array(cols, dtype=np.int32), nz, 2*n-1)
    return _spar_cache[n]

def fast_conv(a):
    """Fast auto-convolution via FFT."""
    n = len(a)
    s = 1
    while s < 2 * n - 1:
        s <<= 1
    fa = np.fft.rfft(a, s)
    return np.fft.irfft(fa * fa, s)[:2 * n - 1]

def compute_J(a):
    """Compute the objective value."""
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))

def lp_best_response(a, M, prev_x=None):
    """Solve the LP best-response problem: max sum(b) s.t. a*b <= M, b >= 0."""
    n = len(a)
    rows, cols, nz, n_cons = get_conv_matrix(n)
    vals = np.empty(nz)
    for i in range(nz):
        k = rows[i]
        j = cols[i]
        vals[i] = a[k - j]
    A = csr_matrix((vals, (rows, cols)), shape=(n_cons, n))
    c = -np.ones(n)
    b_ub = np.full(n_cons, M)
    bounds = [(0.0, None)] * n
    options = {'presolve': True, 'time_limit': 5}
    if prev_x is not None:
        # Use warm start if available
        pass  # HiGHS doesn't directly support warm start in scipy interface
    r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs', options=options)
    if r.success and r.x is not None:
        g = np.maximum(r.x, 0.0)
        if g.sum() > 1e-10:
            return g
    return None

def run_lp_opt(a, max_iters, deadline, tracker, tighten=0.01):
    """Run LP-based optimization loop."""
    a = np.maximum(a.astype(np.float64), 1e-8)
    a /= a.sum()
    n = len(a)
    current_J = compute_J(a)
    tracker.update(a)

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        b = fast_conv(a)
        M = float(b.max())
        M_tight = M * (1.0 - tighten)
        g = lp_best_response(a, M_tight)
        if g is None:
            g = lp_best_response(a, M)
        if g is None:
            break
        g = np.maximum(g, 0.0)
        gs = g.sum()
        if gs < 1e-10:
            break
        g /= gs

        best_a = None
        best_J = current_J
        progress = it / max(1, max_iters)
        base_t = max(0.1, 0.6 * (1.0 - progress))
        for t in [base_t, base_t * 1.5, base_t * 0.7, 0.8, 0.4, 0.2]:
            if t > 1.0:
                continue
            a_trial = (1.0 - t) * a + t * g
            a_trial = np.maximum(a_trial, 0.0)
            ss = a_trial.sum()
            if ss < 1e-10:
                continue
            a_trial /= ss
            J_trial = compute_J(a_trial)
            if J_trial < best_J - 1e-14:
                best_J = J_trial
                best_a = a_trial
        if best_a is None:
            break
        a = best_a
        current_J = best_J
        if it % 5 == 0:
            tracker.update(a)
    tracker.update(a)
    return a

def mirror_descent(a, max_iters, deadline, tracker):
    """Mirror descent (multiplicative updates) for the auto-convolution minimization."""
    a = np.maximum(a.astype(np.float64), 1e-8)
    a /= a.sum()
    n = len(a)
    current_J = compute_J(a)
    tracker.update(a)

    eta = 1.0 / max(10, n // 10)
    best_J = current_J
    stall = 0

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        b = fast_conv(a)
        M = float(b.max())
        # Find active set (lags near maximum)
        threshold = M - 1e-8 * max(1.0, M)
        active = np.where(b >= threshold)[0]

        # Compute subgradient: for each j, grad_j = 2/|A| * sum_{k in A, 0<=k-j<n} a[k-j]
        grad = np.zeros(n)
        for k in active:
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            for j in range(js, je + 1):
                grad[j] += a[k - j]
        grad *= 2.0 / len(active)

        # Full subgradient of J (with sum=1 normalization):
        # dJ/da_j = 2n * (grad_j / 1 - 2*M*a_j / 1) = 2n*(grad_j - 2M*a_j)
        # For mirror descent, we use the subgradient of the objective
        obj_grad = grad - 2.0 * M * a

        # Mirror descent update: a_j <- a_j * exp(-eta * obj_grad_j)
        # but project to simplex
        logits = -eta * obj_grad
        # Clip for numerical stability
        logits_max = logits.max()
        exp_logits = np.exp(logits - logits_max)
        a_new = a * exp_logits
        s = a_new.sum()
        if s < 1e-20:
            break
        a_new /= s
        a_new = np.maximum(a_new, 1e-10)
        a_new /= a_new.sum()

        J_new = compute_J(a_new)
        if J_new < current_J - 1e-15:
            a = a_new
            current_J = J_new
            if current_J < best_J:
                best_J = current_J
                tracker.update(a)
            stall = 0
            eta = min(eta * 1.02, 5.0)
        else:
            stall += 1
            eta *= 0.8
            if stall > 50:
                break
    tracker.update(a)
    return a

def active_set_opt(a, max_iters, deadline, tracker, top_k=5):
    """Optimize focusing on the top-k maximum lags."""
    a = np.maximum(a.astype(np.float64), 1e-8)
    a /= a.sum()
    n = len(a)
    current_J = compute_J(a)
    tracker.update(a)

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        b = fast_conv(a)
        # Get top-k lags
        top_indices = np.argsort(b)[-top_k:]
        M = b[top_indices[0]]  # minimum of top-k (conservative)

        # Use only top-k constraints for LP
        rows = []
        cols = []
        for k in top_indices:
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            rows.extend([0] * (je - js + 1))
            cols.extend(range(js, je + 1))
        n_cons = len(top_indices)
        if len(rows) == 0:
            break
        A = csr_matrix((np.ones(len(rows)), (np.repeat(np.arange(n_cons), np.array([min(n-1, k) - max(0, k-n+1) + 1 for k in top_indices])), np.array(cols))), shape=(n_cons, n))
        
        # Actually let me rebuild this properly
        # For each constraint k, the row in A is the vector a[k-j] for valid j
        A_vals = []
        A_rows = []
        A_cols = []
        for idx, k in enumerate(top_indices):
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            for j in range(js, je + 1):
                A_vals.append(a[k - j])
                A_rows.append(idx)
                A_cols.append(j)
        if len(A_vals) == 0:
            break
        A_mat = coo_matrix((A_vals, (A_rows, A_cols)), shape=(n_cons, n)).tocsr()
        
        g, _ = _lp_solve(A_mat, np.full(n_cons, M), n)
        if g is None:
            break
        g /= g.sum()

        # Mix
        best_a = None
        best_J = current_J
        for t in [0.3, 0.5, 0.7, 1.0]:
            a_trial = (1.0 - t) * a + t * g
            a_trial = np.maximum(a_trial, 0.0)
            a_trial /= a_trial.sum()
            J_trial = compute_J(a_trial)
            if J_trial < best_J - 1e-14:
                best_J = J_trial
                best_a = a_trial
        if best_a is None:
            break
        a = best_a
        current_J = best_J
        tracker.update(a)
    tracker.update(a)
    return a

def _lp_solve(A_mat, b_ub, n):
    """Solve LP: max 1^T x s.t. A x <= b_ub, x >= 0"""
    c = -np.ones(n)
    bounds = [(0.0, None)] * n
    r = linprog(c, A_ub=A_mat, b_ub=b_ub, bounds=bounds, method='highs',
                options={'presolve': True, 'time_limit': 3})
    if r.success and r.x is not None:
        x = np.maximum(r.x, 0.0)
        if x.sum() > 1e-10:
            return x, r
    return None, None

def generate_inits(n, rng, parent=None):
    """Generate diverse initial sequences."""
    inits = []
    
    if parent is not None and parent.size > 0:
        # Resample parent
        t_old = np.linspace(0, 1, parent.size)
        t_new = np.linspace(0, 1, n)
        inits.append(np.maximum(np.interp(t_new, t_old, parent), 0.01))
        inits.append(np.maximum(np.interp(t_new, t_old, parent[::-1]), 0.01))

    t = np.linspace(0, 1, n, endpoint=False)

    # Uniform
    inits.append(np.ones(n))

    # Flat spectrum (random phases)
    for trial in range(3):
        half = n // 2 + 1
        angles = rng.uniform(0, 2 * np.pi, half)
        A = np.zeros(n, dtype=complex)
        A[:half] = np.exp(1j * angles[:half])
        for k in range(1, n - n // 2):
            A[k] = np.conj(A[n - k])
        a = np.fft.ifft(A).real
        inits.append(np.maximum(a, 0.0) + 0.05)

    # Multi-frequency cosine
    for norm_seed in range(4):
        nfreq = rng.integers(3, 8)
        a = np.ones(n)
        for fi in range(nfreq):
            freq = rng.uniform(0.5, 10)
            phase = rng.uniform(0, 2 * np.pi)
            amp = rng.uniform(0.1, 0.4)
            a += amp * np.cos(2 * np.pi * freq * t + phase)
        inits.append(np.maximum(a, 0.05))

    # Smooth bump functions
    for _ in range(3):
        centers = rng.uniform(0.1, 0.9, size=rng.integers(2, 5))
        widths = rng.uniform(0.05, 0.2, size=len(centers))
        a = np.ones(n) * 0.3
        for c, w in zip(centers, widths):
            a += np.exp(-((t - c) / w) ** 2)
        inits.append(a)

    # Power law
    inits.append((t + 0.05) ** rng.uniform(0.5, 2.0))
    inits.append((1.0 - t + 0.05) ** rng.uniform(0.5, 2.0))
    inits.append(np.abs(t - 0.5) ** rng.uniform(0.3, 1.5) + 0.3)

    # Random smooth (spline-like)
    for _ in range(2):
        k = rng.integers(5, 15)
        knots = np.sort(rng.uniform(0, 1, k))
        vals = rng.uniform(0.2, 1.0, k)
        vals = np.concatenate([[vals[0]], vals, [vals[-1]]])
        knots = np.concatenate([[0], knots, [1]])
        a = np.interp(t, knots, vals)
        inits.append(a)

    # Sinusoidal with envelope
    for _ in range(2):
        freq = rng.uniform(1, 5)
        env = 1.0 + 0.5 * np.cos(2 * np.pi * rng.uniform(0.5, 2) * t)
        inits.append(np.maximum(0.5 + 0.5 * env * np.sin(2 * np.pi * freq * t + rng.uniform(0, 6.28)), 0.05))

    # Normalization
    for i in range(len(inits)):
        inits[i] = np.maximum(inits[i], 0.01).astype(np.float64)
        inits[i] /= inits[i].sum()
    
    return inits

"""
Minimizes J(a) = 2*n*max(convolve(a,a))/(sum(a))^2 using:
1. Adaptive active-set LP best-response (fast, uses only binding constraints)
2. Mirror descent (multiplicative updates) with exact subgradient
3. Multi-length search (384-1024) with diverse structured initializations
4. Focused polishing with progressive tightening
5. Diversification through perturbation and resampling

import numpy as np
from scipy.optimize import linprog
from scipy.sparse import csr_matrix, coo_matrix
import time

_spar_cache = {}

def fast_conv(a):
    n = len(a)
    s = 1
    while s < 2 * n - 1:
        s <<= 1
    fa = np.fft.rfft(a, s)
    return np.fft.irfft(fa * fa, s)[:2 * n - 1]

def compute_J(a):
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))

def _lp_solve(c, A_mat, b_ub, n_vars, time_limit=3):
    bounds = [(0.0, None)] * n_vars
    r = linprog(c, A_ub=A_mat, b_ub=b_ub, bounds=bounds, method='highs',
                options={'presolve': True, 'time_limit': time_limit})
    if r.success and r.x is not None:
        x = np.maximum(r.x, 0.0)
        if x.sum() > 1e-10:
            return x
    return None

def adaptive_lp_response(a, M, frac=0.95, max_active=80):
    """LP best-response using only near-active constraints for speed."""
    n = len(a)
    b = fast_conv(a)
    threshold = b.max() * frac
    active_k = np.where(b >= threshold)[0]
    if len(active_k) == 0:
        active_k = np.array([np.argmax(b)])
    if len(active_k) > max_active:
        idx = np.argsort(b[active_k])[::-1][:max_active]
        active_k = active_k[idx]

    A_vals = []
    A_rows = []
    A_cols = []
    for idx2, k in enumerate(active_k):
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        for j in range(js, je + 1):
            A_vals.append(a[k - j])
            A_rows.append(idx2)
            A_cols.append(j)

    if len(A_vals) == 0:
        return None

    A_mat = coo_matrix((A_vals, (A_rows, A_cols)), shape=(len(active_k), n)).tocsr()
    c = -np.ones(n)
    b_ub = np.full(len(active_k), M)
    g = _lp_solve(c, A_mat, b_ub, n, time_limit=2)
    return g

def full_lp_response(a, M):
    """Full LP best-response with all constraints."""
    n = len(a)
    A_vals = []
    A_rows = []
    A_cols = []
    for k in range(2 * n - 1):
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        for j in range(js, je + 1):
            A_vals.append(a[k - j])
            A_rows.append(k)
            A_cols.append(j)
    A_mat = coo_matrix((A_vals, (A_rows, A_cols)), shape=(2 * n - 1, n)).tocsr()
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    g = _lp_solve(c, A_mat, b_ub, n, time_limit=5)
    return g

def run_lp_opt(a, max_iters, deadline, tracker, tighten=0.01):
    a = np.maximum(a.astype(np.float64), 1e-8)
    a /= a.sum()
    n = len(a)
    current_J = compute_J(a)
    tracker.update(a)

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        b = fast_conv(a)
        M = float(b.max())
        M_tight = M * (1.0 - tighten)

        g = adaptive_lp_response(a, M_tight, frac=0.93)
        if g is None:
            g = adaptive_lp_response(a, M, frac=0.95)
        if g is None:
            g = full_lp_response(a, M)
        if g is None:
            break

        g = np.maximum(g, 0.0)
        gs = g.sum()
        if gs < 1e-10:
            break
        g /= gs

        best_a = None
        best_J = current_J
        progress = it / max(1, max_iters)
        base_t = max(0.1, 0.5 * (1.0 - progress))
        for t in [base_t, base_t * 1.5, base_t * 0.6, 0.7, 0.4, 0.2]:
            if t > 1.0:
                continue
            a_trial = (1.0 - t) * a + t * g
            a_trial = np.maximum(a_trial, 0.0)
            ss = a_trial.sum()
            if ss < 1e-10:

        a = best_a

        if it % 5 == 0:

def mirror_descent(a, max_iters, deadline, tracker, use_ema=True):
    """Mirror descent with multiplicative updates on the simplex."""
    a = np.maximum(a.astype(np.float64), 1e-8)
    a /= a.sum()
    n = len(a)
    current_J = compute_J(a)
    tracker.update(a)

    eta = 0.5 / max(5, n // 50)
    stall = 0
    ema_grad = None
    alpha_ema = 0.7

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        b = fast_conv(a)
        M = float(b.max())
        threshold = M - 1e-9 * max(1.0, M)
        active = np.where(b >= threshold)[0]

        grad = np.zeros(n)
        for k in active:
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            for j in range(js, je + 1):
                grad[j] += a[k - j]
        grad *= 2.0 / len(active)

        obj_grad = grad - 2.0 * M * a

        if use_ema:
            if ema_grad is None:
                ema_grad = obj_grad.copy()
            else:
                ema_grad = alpha_ema * ema_grad + (1 - alpha_ema) * obj_grad
            step_grad = ema_grad
        else:
            step_grad = obj_grad

        logits = -eta * step_grad
        logits_max = logits.max()
        if logits_max > 500:
            logits_max = 500
        exp_logits = np.exp(logits - logits_max)
        a_new = a * exp_logits
        s = a_new.sum()
        if s < 1e-20 or not np.isfinite(s):
            break
        a_new /= s
        a_new = np.maximum(a_new, 1e-10)
        a_new /= a_new.sum()

        J_new = compute_J(a_new)
        if J_new < current_J - 1e-15:
            a = a_new
            current_J = J_new
            stall = 0
            eta = min(eta * 1.01, 10.0)
            if it % 20 == 0:
                tracker.update(a)
        else:
            stall += 1
            eta *= 0.85
            if use_ema:
                ema_grad *= alpha_ema
            if stall > 80:
                break
    tracker.update(a)
    return a

def lse_mirror_descent(a, max_iters, deadline, tracker):
    """LSE smooth approximation with mirror descent."""
    a = np.maximum(a.astype(np.float64), 1e-6)
    a /= a.sum()
    n = len(a)
    current_J = compute_J(a)
    tracker.update(a)

    tau = 30.0
    lr = 0.5 / (n * tau)
    v = np.zeros(n)
    momentum = 0.8
    stall = 0

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        b = fast_conv(a)
        b_max = b.max()
        # Clip for numerical stability
        scaled = tau * (b - b_max)
        scaled = np.clip(scaled, -500, 0)
        exp_b = np.exp(scaled)
        p = exp_b / exp_b.sum()

        # Gradient of LSE w.r.t. a_j = 2 * sum_k p_k * a[k-j] (if 0<=k-j<n)
        # Use FFT: grad = 2 * ifft(fft(p) * conj(fft(a)))[:n]
        s = 1
        while s < 3 * n:
            s <<= 1
        fp = np.fft.rfft(p, s)
        fa = np.fft.rfft(a, s)
        corr = np.fft.irfft(fp * np.conj(fa), s)
        grad_lse = 2.0 * corr[:n]

        # Full gradient of J (normalized sum=1):
        # dJ/da_j = 2n * (grad_lse_j - 2*b_max*a_j) / s^2, s=1
        obj_grad = grad_lse - 2.0 * b_max * a

        v = momentum * v + (1.0 - momentum) * obj_grad
        a_new = a - lr * v
        a_new = np.maximum(a_new, 1e-6)
        a_new /= a_new.sum()

        J_new = compute_J(a_new)
        if J_new < current_J - 1e-15:
            a = a_new
            current_J = J_new
            stall = 0
            tau = min(tau * 1.02, 200.0)
            lr = 0.5 / (n * tau)
            if it % 50 == 0:
                tracker.update(a)
        else:
            stall += 1
            lr *= 0.5
            v *= 0.5
            if stall > 100 or lr < 1e-18:
                break
    tracker.update(a)
    return a

def generate_inits(n, rng, parent=None):

    t = np.linspace(0, 1, n, endpoint=False)

    if parent is not None and parent.size > 0:

        inits.append(np.maximum(np.interp(t, t_old, parent[::-1]), 0.01))
        # Add slight perturbation to parent
        p_res = np.maximum(np.interp(t, t_old, parent), 0.01)
        p_res = p_res + rng.normal(0, 0.03, n)
        inits.append(np.maximum(p_res, 0.01))

    # Uniform
    inits.append(np.ones(n))

    # Flat spectrum (random phases) - multiple trials
    for _ in range(4):
        half = n // 2 + 1
        angles = rng.uniform(0, 2 * np.pi, half)
        amps = rng.uniform(0.5, 1.0, half)

            A[k] = np.conj(A[n - k])
        a = np.fft.ifft(A).real
        inits.append(np.maximum(a, 0.0) + 0.05)

    # Multi-frequency with random structure
    for _ in range(4):

        a = np.ones(n)

            phase = rng.uniform(0, 2 * np.pi)
            amp = rng.uniform(0.05, 0.35)
            a += amp * np.cos(2 * np.pi * freq * t + phase)
        inits.append(np.maximum(a, 0.05))

    # Smooth splines

        knots = np.sort(rng.uniform(0, 1, k))

        knots = np.concatenate([[0.0], knots, [1.0]])
        a = np.interp(t, knots, vals)

    # Bump functions
    for _ in range(3):
        nb = rng.integers(2, 5)
        centers = rng.uniform(0.1, 0.9, nb)
        widths = rng.uniform(0.05, 0.15, nb)
        a = np.ones(n) * 0.2
        for c, w in zip(centers, widths):
            a += np.exp(-((t - c) / w) ** 2)
        inits.append(a)

    # Power laws and asymmetric
    inits.append((t + 0.05) ** rng.uniform(0.5, 2.5))
    inits.append((1.0 - t + 0.05) ** rng.uniform(0.5, 2.5))
    inits.append(np.abs(t - 0.5) ** rng.uniform(0.3, 1.5) + 0.2)

    # Sinusoidal with envelope
    for _ in range(2):
        freq = rng.uniform(1, 6)
        env_freq = rng.uniform(0.3, 1.5)
        env = 1.0 + 0.5 * np.cos(2 * np.pi * env_freq * t)
        inits.append(np.maximum(0.4 + 0.6 * env * np.abs(np.sin(2 * np.pi * freq * t + rng.uniform(0, 6.28))), 0.05))

    # Random Fourier with limited bandwidth
    for _ in range(2):
        n_coeff = rng.integers(4, min(12, n // 4))
        a = np.ones(n)
        for i in range(n_coeff):
            freq = rng.integers(1, n // 2)
            amp = rng.uniform(0.05, 0.3)
            phase = rng.uniform(0, 2 * np.pi)
            a += amp * np.cos(2 * np.pi * freq * t / n + phase)
        inits.append(np.maximum(a, 0.05))

    # Sparse + smoothing
    for _ in range(2):
        n_spikes = max(3, n // 20)
        positions = rng.choice(n, size=n_spikes, replace=False)
        sparse = np.zeros(n)
        sparse[positions] = 1.0
        kernel_size = max(3, n // 25)
        kernel = np.exp(-((np.arange(kernel_size) - kernel_size // 2) / (kernel_size / 4)) ** 2)
        kernel /= kernel.sum()
        smooth = np.convolve(sparse, kernel, mode='same')
        inits.append(smooth + 0.1)

    for i in range(len(inits)):
        inits[i] = np.maximum(inits[i], 0.01).astype(np.float64)
        inits[i] /= inits[i].sum()

    return inits

class ScoreTracker:
    def __init__(self):
        self.best = None
        self.best_score = float('inf')
        self.t_start = time.monotonic()

    def update(self, a):
        a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
        if a.size == 0 or a.sum() < 0.01:
            return
        score = float(evaluate_sequence(a.tolist()))
        if np.isfinite(score) and score < self.best_score:
            self.best = a.copy()
            self.best_score = score
            elapsed = time.monotonic() - self.t_start
            print(f"  * J={score:.8f} n={a.size} t={elapsed:.1f}s", flush=True)

    def report(self, label=""):
        if self.best is not None:
            print(f"  {label} best: J={self.best_score:.8f} n={self.best.size}", flush=True)

def resample_to(a_src, n_dest):
    t_old = np.linspace(0, 1, a_src.size)
    t_new = np.linspace(0, 1, n_dest)
    return np.maximum(np.interp(t_new, t_old, a_src), 0.01)

def propose_candidate(seed=42, budget_s=1500, **kwargs):
    rng = np.random.default_rng(seed)
    total_budget = float(budget_s)
    deadline = time.monotonic() + total_budget - 25.0
    tracker = ScoreTracker()

    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        if p.ndim == 1 and p.size > 0 and np.all(np.isfinite(p)):
            parent = np.maximum(p, 0.0)
            tracker.update(parent)
            print(f"Parent loaded: n={parent.size}, J={tracker.best_score:.8f}", flush=True)
    except (NameError, AttributeError):
        pass

    # === Phase 1: Multi-length exploration (55% time) ===
    phase1_budget = 0.55 * total_budget
    phase1_end = tracker.t_start + phase1_budget

    lengths = [384, 448, 512, 640, 768, 1024]
    if parent is not None:
        pn = parent.size
        weights = [1.0 / (1.0 + abs(n - pn) / 100.0) for n in lengths]
    else:
        weights = [1.0, 1.2, 1.5, 1.2, 0.8, 0.6]
    tw = sum(weights)
    weights = [w / tw for w in weights]

    for n, w in zip(lengths, weights):
        alloc = w * phase1_budget
        t_start = time.monotonic()
        t_end = min(t_start + alloc, phase1_end - 5)
        if time.monotonic() >= t_end - 3:
            continue

        print(f"\n{'='*40} Phase1: n={n} alloc={alloc:.0f}s {'='*40}", flush=True)

        inits = generate_inits(n, rng, parent)
        n_inits = len(inits)
        per_init = max(4.0, (t_end - time.monotonic()) / n_inits)

        best_for_n = None
        best_J_for_n = float('inf')

        for idx, a0 in enumerate(inits):
            if time.monotonic() > t_end - 3:
                break
            t0 = time.monotonic()
            t_init_end = min(t0 + per_init, t_end - 3)

            # LP phase
            n_lp = max(5, int(40.0 / max(1.0, n / 256.0)))
            a = run_lp_opt(a0, n_lp, t_init_end - 5, tracker, tighten=0.02)

            # Mirror descent
            if time.monotonic() < t_init_end - 2:
                a = mirror_descent(a, max(30, int(5000.0 / n)), t_init_end - 1, tracker)

            J = compute_J(a)
            if J < best_J_for_n:
                best_J_for_n = J
                best_for_n = a.copy()

            if (idx + 1) % 5 == 0 or idx == n_inits - 1:
                print(f"    [{idx+1}/{n_inits}] best J={best_J_for_n:.6f} (t={time.monotonic()-t_start:.0f}s)", flush=True)

        if best_for_n is not None:
            tracker.update(best_for_n)
            print(f"  Phase1 n={n}: J={best_J_for_n:.6f} in {time.monotonic()-t_start:.0f}s", flush=True)

    # === Phase 2: Focused polishing (30% time) ===
    if tracker.best is not None and time.monotonic() < deadline - 50:
        t0 = time.monotonic()
        n = tracker.best.size
        phase2_end = min(deadline - 50, t0 + 0.30 * total_budget)
        print(f"\n{'='*40} Phase2: Focus n={n} {'='*40}", flush=True)

        a = tracker.best.copy().astype(np.float64)

        # Progressive tightening with LP
        tighten_sched = [0.04, 0.03, 0.02, 0.015, 0.01, 0.008, 0.005]
        for tight in tighten_sched:
            if time.monotonic() > phase2_end - 20:
                break
            a = run_lp_opt(a, 15, phase2_end - 20, tracker, tighten=tight)

        # Mirror descent for local refinement
        if time.monotonic() < phase2_end - 15:
            a = mirror_descent(a, 3000, phase2_end - 15, tracker)

        # LSE mirror descent for smooth polishing
        if time.monotonic() < phase2_end - 10:
            a = lse_mirror_descent(a, 2000, phase2_end - 5, tracker)

        # Final tight LP
        if time.monotonic() < phase2_end - 3:
            a = run_lp_opt(a, 10, phase2_end - 2, tracker, tighten=0.003)
            if time.monotonic() < phase2_end - 1:
                a = mirror_descent(a, 1000, phase2_end - 0.5, tracker)

        tracker.report(f"Phase2 ({time.monotonic()-t0:.0f}s)")

    # === Phase 3: Diversification (15% time) ===
    if tracker.best is not None and time.monotonic() < deadline - 20:
        t0 = time.monotonic()
        phase3_end = min(deadline - 10, t0 + 0.15 * total_budget)
        print(f"\n{'='*40} Phase3: Diversify {'='*40}", flush=True)

        parent_a = tracker.best.copy()
        pn = parent_a.size

        # Try nearby lengths
        for n_try in sorted(set([max(64, pn // 2), pn, min(2048, pn * 2)])):
            if time.monotonic() > phase3_end - 10:
                break
            a0 = resample_to(parent_a, n_try)
            a0 = a0 + rng.normal(0, 0.02, n_try)
            a0 = np.maximum(a0, 0.01)
            a0 /= a0.sum()

            a = run_lp_opt(a0, 12, phase3_end - 10, tracker, tighten=0.015)
            if time.monotonic() < phase3_end - 5:
                a = mirror_descent(a, 1000, phase3_end - 5, tracker)
            tracker.update(a)
            print(f"    n={n_try}: J={tracker.best_score:.6f}", flush=True)

        # Perturbation restarts
        if time.monotonic() < phase3_end - 10:
            n = parent_a.size
            for trial in range(8):
                if time.monotonic() > phase3_end - 10:
                    break
                noise_scale = 0.03 + 0.02 * trial
                a_pert = parent_a + rng.normal(0, noise_scale, n)
                a_pert = np.maximum(a_pert, 0.01)
                a_pert /= a_pert.sum()

                a = run_lp_opt(a_pert, 8, phase3_end - 10, tracker, tighten=0.015)
                if time.monotonic() < phase3_end - 5:
                    a = mirror_descent(a, 500, phase3_end - 5, tracker)
                tracker.update(a)

        # Best-frequency search: try weighted combinations
        if time.monotonic() < phase3_end - 5:
            n = parent_a.size
            t = np.linspace(0, 1, n, endpoint=False)
            for freq in [1.2, 2.7, 3.3, 5.1, 7.8]:
                if time.monotonic() > phase3_end - 5:
                    break
                a_mod = parent_a * (1.0 + 0.1 * np.cos(2 * np.pi * freq * t))
                a_mod = np.maximum(a_mod, 0.01)
                a_mod /= a_mod.sum()
                a = run_lp_opt(a_mod, 8, phase3_end - 5, tracker, tighten=0.01)
                tracker.update(a)

        tracker.report(f"Phase3 ({time.monotonic()-t0:.0f}s)")

    # === Final polish on absolute best ===
    if tracker.best is not None and time.monotonic() < deadline - 3:
        a_final = tracker.best.copy().astype(np.float64)
        a_final = mirror_descent(a_final, 500, deadline - 2, tracker)
        tracker.report("Final")

    if tracker.best is None:
        a_fallback = np.ones(256)
        tracker.best = a_fallback
        tracker.best_score = compute_J(a_fallback)
        tracker.update(tracker.best)

    print(f"\n{'='*40} FINAL RESULT {'='*40}", flush=True)
    print(f"J = {tracker.best_score:.8f}", flush=True)
    print(f"n = {tracker.best.size}", flush=True)
    print(f"Total time = {time.monotonic() - tracker.t_start:.1f}s", flush=True)
    return [float(x) for x in tracker.best]
