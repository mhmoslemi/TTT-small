_A_cache = {}

_cols_cache = {}

def fast_conv(a):
    n = len(a)
    size = 1
    while size < 2*n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2*n-1]

def compute_J(a):
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s*s))

def get_conv_matrix_info(n):
    """Precompute row/col structure for the convolution constraint matrix."""
    if n in _A_cache:
        return _A_cache[n]
    # For constraint k (0 to 2n-2), the variables involved are j where 0<=k-j<=n-1
    # i.e., j from max(0, k-n+1) to min(n-1, k)
    rows = []
    cols = []
    for k in range(2*n-1):
        jl = max(0, k - n + 1)
        jh = min(n-1, k)
        for j in range(jl, jh+1):
            rows.append(k)
            cols.append(j)
    rows = np.array(rows, dtype=np.int32)
    cols = np.array(cols, dtype=np.int32)
    _A_cache[n] = (rows, cols)
    return (rows, cols)

def build_conv_matrix(a):
    """Build the sparse matrix A where (A @ a)_k = (a*a)_k for k=0..2n-2."""
    n = len(a)
    rows, cols = get_conv_matrix_info(n)
    # For constraint k, coefficient of a_j is a_{k-j}
    vals = a[rows - cols]
    return csr_matrix((vals, (rows, cols)), shape=(2*n-1, n))

def lp_direction(a, M):
    """Solve LP: max sum(g) s.t. (a*g)_k <= M for all k, g >= 0. Return g/sum(g)."""
    n = len(a)
    c = -np.ones(n)
    A = build_conv_matrix(a)
    b_ub = np.full(2*n-1, M)
    bounds = [(0, None)] * n
    try:
        r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                   options={'presolve': True, 'dual_feasibility_tolerance': 1e-9})
        if r.success and r.x is not None:
            g = np.maximum(r.x, 0)
            gs = g.sum()
            if gs > 1e-12:
                return g / gs
    except Exception:
        pass
    return None

def F_smooth(a, beta):

    s = a.sum()
    if s < 1e-10:
        return 1e10, np.zeros(n)
    gval, grad_g = smooth_grad(a, beta)
    F = 2.0 * n * gval / (s*s)
    grad_F = 2.0 * n * (s * grad_g - 2.0 * gval) / (s**3)
    return F, grad_F

def adam_smooth_optimize(a0, beta_schedule, deadline, tracker):

def coordinate_refine(a, deadline, tracker):
    """Coordinate-wise line search for final refinement."""
    a = np.maximum(a.astype(np.float64), 0)
    a = a / a.sum()
    n = len(a)
    current_J = compute_J(a)
    if tracker:
        tracker.update(a)
    
    improved = True
    while improved and time.monotonic() < deadline:
        improved = False
        for i in range(n):
            if time.monotonic() > deadline:
                break
            # Try changing a[i]
            orig = a[i]
            if orig < 1e-15:
                # Try adding a small amount
                for delta in [0.001, 0.005, 0.01] * a.max():
                    a[i] = delta
                    a = a / a.sum()
                    J = compute_J(a)
                    if J < current_J - 1e-14:
                        current_J = J
                        improved = True
                        if tracker:
                            tracker.update(a)
                        break
                else:
                    a[i] = 0
                    a = a / a.sum()
                    J = compute_J(a)
                    if J < current_J - 1e-14:
                        current_J = J
                        improved = True
                        if tracker:
                            tracker.update(a)
                    a[i] = orig
                    continue
            
            # Try scaling a[i]
            for scale in [0.8, 0.9, 1.1, 1.2, 0.7, 1.3]:
                a[i] = orig * scale
                a = a / a.sum()
                J = compute_J(a)
                if J < current_J - 1e-14:
                    current_J = J
                    improved = True
                    if tracker:
                        tracker.update(a)
                    # Continue trying to improve
                    orig = a[i]
                else:
                    a[i] = orig
                    a = a / a.sum()

class ScoreTracker:
    def __init__(self):
        self.best = None
        self.best_score = float('inf')
        self.t_start = time.monotonic()
    
    def update(self, a):
        a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
        if a.size == 0 or a.sum() < 0.01:
            return
        score = float(evaluate_sequence(a.tolist()))
        if np.isfinite(score) and score < self.best_score:
            self.best = a.copy()
            self.best_score = score
            elapsed = time.monotonic() - self.t_start
            print(f"  * NEW BEST: J={score:.8f} n={a.size} t={elapsed:.1f}s", flush=True)
    
    def report(self, label=""):
        if self.best is not None:
            print(f"  {label} best: J={self.best_score:.8f} n={self.best.size}", flush=True)

def resample_seq(src, dest_n):
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    result = np.interp(t_new, t_old, src)
    return np.maximum(result, 0.0)

def generate_initializations(n, rng, parent=None):
    inits = []
    t = np.linspace(0, 1, n, endpoint=False)
    
    if parent is not None and parent.size > 0:
        p = resample_seq(parent, n)
        inits.append(p)
        # Perturbed parent
        p2 = p + rng.normal(0, 0.03, n) * p
        inits.append(np.maximum(p2, 0.01))
    
    # Uniform
    inits.append(np.ones(n))
    
    # Cosine-based (related to Barker sequence structure)
    for freq in [1, 2, 3, 4, 5]:
        inits.append(1.0 + 0.3 * np.cos(2*np.pi*freq*t))
    
    # Random smooth
    noise = rng.normal(0, 1, n)
    kernel = np.exp(-0.5 * (np.linspace(-3, 3, 16))**2)
    kernel /= kernel.sum()
    smooth = np.convolve(noise, kernel, mode='same')
    inits.append(np.abs(smooth) + 0.1)
    
    # m-sequence like (for prime n)
    # Skip for now, just use structured random
    
    # Piecewise constant
    n_blocks = rng.integers(4, 12)
    block_size = max(1, n // n_blocks)
    blocks = rng.uniform(0.3, 1.0, n_blocks)
    seq = np.repeat(blocks, block_size)[:n]
    if len(seq) < n:
        seq = np.concatenate([seq, rng.uniform(0.3, 1.0, n - len(seq))])
    inits.append(seq)
    
    # Exponential decay with oscillation
    inits.append(np.exp(-2*t) * (1 + 0.5*np.cos(2*np.pi*3*t)))
    
    return [np.maximum(np.asarray(x, dtype=np.float64), 0.01) for x in inits]

output[k] = sum_n in1[n + k - (M-1)] * in2[n]  where M = len(in2)

So output[k] = sum_n in1[n + k - (M-1)] * in2[n]

For in1=p (length 3), in2=a (length 2), M-1=1:
output[0] = sum_n p[n-1] * a[n] = p[-1]*a[0] + p[0]*a[1] = 0 + 1*5 = 5
output[1] = sum_n p[n] * a[n] = p[0]*a[0] + p[1]*a[1] = 1*4 + 2*5 = 14
output[2] = sum_n p[n+1] * a[n] = p[1]*a[0] + p[2]*a[1] = 2*4 + 3*5 = 23
output[3] = sum_n p[n+2] * a[n] = p[2]*a[0] = 3*4 = 12

So np.correlate(p, a, 'full') = [5, 14, 23, 12]

And what I want is:
out[i] = sum_m p[m+i] * a[m]
out[0] = p[0]*a[0] + p[1]*a[1] = 1*4 + 2*5 = 14
out[1] = p[1]*a[0] + p[2]*a[1] = 2*4 + 3*5 = 23

So out = np.correlate(p, a, 'full')[1:1+n] = output[1:3] = [14, 23] ✓

The offset is (len(a)-1) = 1 in this case.

General formula: out[i] = np.correlate(p, a, 'full')[(len(a)-1) + i] = np.correlate(p, a, 'full')[n-1 : n-1+n]

Since len(a) = n, the offset is n-1.

So: grad = 2 * np.correlate(p, a, 'full')[n-1 : 2*n-1]

For the FFT computation:
np.correlate(p, a, 'full') = ifft(FFT(p, N) * conj(FFT(a, N)))[:len(p)+len(a)-1]

where N >= len(p) + len(a) - 1 = (2n-1) + n - 1 = 3n-2

So:
N = next_power_of_2(3*n-2)
P = np.fft.rfft(p, N)
A = np.fft.rfft(a, N)
xcorr = np.fft.irfft(P * np.conj(A), N)[:3*n-2]
grad = 2 * xcorr[n-1 : 2*n-1]
