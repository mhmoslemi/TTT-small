def fast_conv(a):
    """Fast linear convolution via FFT."""
    n = len(a)
    s = 1
    while s < 2 * n - 1:
        s <<= 1
    fa = np.fft.rfft(a, s)
    return np.fft.irfft(fa * fa, s)[:2 * n - 1]


def compute_J(a):
    """Compute objective J = 2n*max(conv(a,a))/(sum(a))^2."""
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))


def subgradient_of_max_conv(a, tol_frac=1e-8):
    """
    Compute subgradient of max(conv(a,a)) w.r.t. a.
    Returns (grad, active_indices) where grad[j] = (2/|S|) * sum_{k in S} a[k-j] (valid).
    """
    n = len(a)
    b = fast_conv(a)
    M = b.max()
    active = np.where(b >= M - tol_frac * M)[0]
    grad = np.zeros(n)
    for k in active:
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        for j in range(js, je + 1):
            grad[j] += a[k - j]
    grad *= 2.0 / len(active)
    return grad, active, M


def multiplicative_weights_step(a, eta=None, tol_frac=1e-8):
    """
    One step of multiplicative weights (KL mirror descent).
    a_new[j] = a[j] * exp(-eta * g[j]), normalized.
    """
    n = len(a)
    grad, active, M = subgradient_of_max_conv(a, tol_frac)
    if eta is None:
        gmax = grad.max()
        eta = 1.0 / (gmax + 1e-12)
    log_a = np.log(np.maximum(a, 1e-30))
    log_a_new = log_a - eta * grad
    # Shift for numerical stability
    log_a_new -= log_a_new.max()
    a_new = np.exp(log_a_new)
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s < 1e-20:
        return a.copy()
    a_new /= s
    return a_new

_spar_cache = {}

def get_sparsity(n):
    if n not in _spar_cache:
        rows = []
        cols = []
        for k in range(2 * n - 1):
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            for j in range(js, je + 1):
                rows.append(k)
                cols.append(j)
        _spar_cache[n] = (np.array(rows), np.array(cols), len(rows))
    return _spar_cache[n]


def lp_best_response(a, M, time_limit=15):
    """
    Solve: maximize sum(b) s.t. conv(a,b) <= M, b >= 0.
    Returns optimal b (normalized) or None.
    """
    n = len(a)
    rows, cols, nnz = get_sparsity(n)
    vals = np.empty(nnz)
    for i in range(nnz):
        k = rows[i]
        j = cols[i]
        vals[i] = a[k - j]
    A = csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    try:
        r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                    options={'presolve': True, 'time_limit': time_limit,
                             'dual_feasibility_tolerance': 1e-7,
                             'primal_feasibility_tolerance': 1e-7})
        if r.success and r.x is not None:
            g = np.maximum(r.x, 0.0)
            if g.sum() > 1e-10:
                return g
    except Exception:
        pass
    return None


def lp_best_response_dual(a, M, time_limit=15):
    """
    Solve dual: minimize M*sum(lam) s.t. A^T lam >= 1, lam >= 0.
    Recover primal from dual solution if possible.
    """
    n = len(a)
    rows, cols, nnz = get_sparsity(n)
    vals = np.empty(nnz)
    for i in range(nnz):
        k = rows[i]
        j = cols[i]
        vals[i] = a[k - j]
    A = csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))
    # Dual: min M * sum(lam) s.t. A^T lam >= 1, lam >= 0
    c_dual = np.full(2 * n - 1, M)
    A_eq_dual = A.T  # shape (n, 2n-1)
    # A^T lam >= 1  =>  -A^T lam <= -1
    r = linprog(c_dual, A_ub=-A_eq_dual, b_ub=-np.ones(n),
                bounds=[(0.0, None)] * (2 * n - 1), method='highs',
                options={'presolve': True, 'time_limit': time_limit})
    if r.success and r.x is not None:
        # Recover primal: b[j] > 0 where (A^T lam)[j] = 1
        # This gives the support but not the values. Use primal for values.
        pass
    return None

def run_lp_opt(a, max_iters, deadline, tracker, tighten=0.01, time_per_lp=10):
    """Run LP best-response optimization loop."""
    a = np.maximum(a.astype(np.float64), 1e-10)
    a /= a.sum()
    n = len(a)
    current_J = compute_J(a)
    tracker.update(a)

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        b = fast_conv(a)
        M = float(b.max())

        # Adaptive tightening: start loose, get tighter
        progress = it / max(1, max_iters)
        adaptive_tight = tighten * (0.5 + progress)
        M_tight = M * (1.0 - adaptive_tight)

        g = lp_best_response(a, M_tight, time_limit=time_per_lp)
        if g is None:
            g = lp_best_response(a, M, time_limit=time_per_lp)
        if g is None:
            break

        g = np.maximum(g, 0.0)
        gs = g.sum()
        if gs < 1e-10:
            break
        g /= gs

        # Backtracking line search
        best_a = None
        best_J = current_J
        t_candidates = [0.8, 0.5, 0.3, 0.15, 0.1, 0.05]
        for t in t_candidates:
            a_trial = (1.0 - t) * a + t * g
            a_trial = np.maximum(a_trial, 0.0)
            ss = a_trial.sum()
            if ss < 1e-10:
                continue
            a_trial /= ss
            J_trial = compute_J(a_trial)
            if J_trial < best_J - 1e-14:
                best_J = J_trial
                best_a = a_trial
                break  # Take first improving step (greedy)

        if best_a is None:
            # Try multiplicative weights as fallback
            a_mw = multiplicative_weights_step(a)
            J_mw = compute_J(a_mw)
            if J_mw < current_J - 1e-14:
                a = a_mw
                current_J = J_mw
            else:
                break
        else:
            a = best_a
            current_J = best_J

        if it % 5 == 0:
            tracker.update(a)

    tracker.update(a)
    return a

def subgradient_descent(a, max_iters, deadline, tracker):
    """Subgradient descent with Polyak step size."""
    a = np.maximum(a.astype(np.float64), 1e-10)
    a /= a.sum()
    n = len(a)
    current_J = compute_J(a)
    tracker.update(a)

    # J = 2*n*max(conv(a,a)) for sum(a)=1
    # Optimal value lower bound: 2n/(2n-1)
    J_lower = 2.0 * n / (2 * n - 1)
    gap = current_J - J_lower

    lr = 2.0 / gap if gap > 0 else 0.1
    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        grad, active, M = subgradient_of_max_conv(a)

        # Subgradient of J = 2n * max(conv) is 2n * grad (since sum=1)
        # But we need to account for the sum constraint projection
        # Direction: move in -grad direction
        a_new = a - lr * grad
        a_new = np.maximum(a_new, 1e-10)
        a_new /= a_new.sum()

        J_new = compute_J(a_new)
        if J_new < current_J - 1e-15:
            a = a_new
            current_J = J_new
            gap = max(gap * 0.99, J_new - J_lower, 1e-6)
            lr = min(2.0 / gap, 5.0 / n)
        else:
            lr *= 0.7
            if lr < 1e-12:
                break
        if it % 30 == 0:
            tracker.update(a)
    tracker.update(a)
    return a

def lse_gradient_descent(a, max_iters, deadline, tracker, tau_start=30.0, tau_end=300.0):
    """LSE-smoothed gradient descent with Adam optimizer and staged tau."""
    a = np.maximum(a.astype(np.float64), 1e-6)
    a /= a.sum()
    n = len(a)
    current_J = compute_J(a)
    tracker.update(a)

    tau = tau_start
    beta1, beta2, eps = 0.8, 0.999, 1e-8
    m = np.zeros(n)
    v = np.zeros(n)
    t_adam = 0
    lr = 0.5

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        # Staged tau increase
        progress = it / max(1, max_iters)
        tau = tau_start * (tau_end / tau_start) ** progress

        b = fast_conv(a)
        b_max = b.max()
        # Softmax distribution (stable)
        scaled = tau * (b - b_max)
        exp_b = np.exp(scaled)
        p = exp_b / exp_b.sum()

        # Gradient of LSE w.r.t. a: 2 * correlation(p, a)[:n]
        # corr[k] = sum_j p[j] * a[k-j] (valid range)
        s = 1
        while s < 3 * n:
            s <<= 1
        fp = np.fft.rfft(p, s)
        fa = np.fft.rfft(a, s)
        # Cross-correlation: ifft(conj(FP) * FA)
        corr = np.fft.irfft(np.conj(fp) * fa, s)
        grad = 2.0 * corr[:n]

        # Adam update
        t_adam += 1
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * grad ** 2
        m_hat = m / (1 - beta1 ** t_adam)
        v_hat = v / (1 - beta2 ** t_adam)
        a_new = a - lr * m_hat / (np.sqrt(v_hat) + eps)
        a_new = np.maximum(a_new, 1e-6)
        a_new /= a_new.sum()

        J_new = compute_J(a_new)
        if J_new < current_J - 1e-15:
            a = a_new
            current_J = J_new
            lr = min(lr * 1.01, 2.0)
        else:
            lr *= 0.5
            m *= 0.5
            v *= 0.5
            if lr < 1e-14:
                break
        if it % 50 == 0:
            tracker.update(a)
    tracker.update(a)
    return a

def combined_optimize(a, deadline, tracker, n_lp=30, n_mw=50, n_lse=200,

def resample_seq(src, dest_n):
    """Resample a sequence to a different length."""
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    if src.size == 0:
        return np.ones(dest_n) / dest_n
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)


def generate_inits(n, rng, parent=None):
    """Generate diverse initializations."""
    inits = []
    t = np.linspace(0, 1, n, endpoint=False)

    # From parent
    if parent is not None and parent.size > 0:
        inits.append(resample_seq(parent, n))
        inits.append(resample_seq(parent[::-1], n))

    # Uniform
    inits.append(np.ones(n))

    # Random
    inits.append(rng.uniform(0.1, 1.0, n))

    # Raised cosine
    inits.append(0.5 + 0.5 * np.cos(np.pi * t))

    # Raised cosine squared
    inits.append(0.5 + 0.5 * np.cos(np.pi * t) ** 2)

    # Gaussian bump
    for center, width in [(0.5, 0.3), (0.5, 0.5), (0.5, 0.2), (0.3, 0.3), (0.7, 0.3)]:
        inits.append(np.exp(-((t - center) / width) ** 2))

    # Multi-frequency smooth
    for freqs, amps in [
        ([1, 2, 3], [1.0, 0.4, 0.2]),
        ([2, 3, 5], [1.0, 0.5, 0.3]),
        ([1, 3, 5, 7], [1.0, 0.5, 0.3, 0.15]),
    ]:
        sig = np.zeros(n)
        for f, amp in zip(freqs, amps):
            sig += amp * np.cos(2 * np.pi * f * t + rng.uniform(0, 6.28))
        inits.append(1.0 + 0.5 * sig)

    # Power-law
    inits.append((t + 0.05) ** 0.5)
    inits.append((1 - t + 0.05) ** 0.5)
    inits.append(1.0 / (t + 0.1 + (1 - t + 0.1)))

    # Two bumps
    inits.append(0.8 * np.exp(-8 * (t - 0.3) ** 2) + 0.8 * np.exp(-8 * (t - 0.7) ** 2) + 0.05)

    # Flat spectrum (random phases)
    half = n // 2 + 1
    angles = rng.uniform(0, 2 * np.pi, half)
    A = np.zeros(n, dtype=complex)
    A[:half] = np.exp(1j * angles[:half])
    for k in range(1, n - n // 2):
        A[k] = np.conj(A[n - k])
    a = np.fft.ifft(A).real
    inits.append(np.maximum(a, 0.0) + 0.1)

    # Weighted flat spectrum
    angles2 = rng.uniform(0, 2 * np.pi, half)
    mags = np.ones(half)
    mags[1:] = 0.7
    A2 = np.zeros(n, dtype=complex)
    A2[:half] = mags * np.exp(1j * angles2[:half])
    for k in range(1, n - n // 2):
        A2[k] = np.conj(A2[n - k])
    a2 = np.fft.ifft(A2).real
    inits.append(np.maximum(a2, 0.0) + 0.1)

    # Normalize all
    result = []
    for x in inits:
        x = np.maximum(x, 0.001).astype(np.float64)
        s = x.sum()
        if s > 0:
            x /= s
        result.append(x)
    return result

def coordinate_descent(a, max_steps, deadline, tracker):

    a = a.copy().astype(np.float64)
    a = np.maximum(a, 1e-10)

    for step in range(max_steps):
        if time.monotonic() > deadline:
            break
        j = rng.integers(0, n)
        improved = False
        for scale in scales:
            a_trial = a.copy()
            a_trial[j] *= scale

            ss = a_trial.sum()

            a_trial /= ss
            J_trial = compute_J(a_trial)
            if J_trial < current_J - 1e-14:
                a = a_trial
                current_J = J_trial
                improved = True
                break
        if step % 200 == 0:
            tracker.update(a)
    tracker.update(a)
    return a

# Main Search Function

def propose_candidate(seed=42, budget_s=1500, **kwargs):
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + float(budget_s) - 30.0
    tracker = ScoreTracker()

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        if p.ndim == 1 and p.size > 0 and np.all(np.isfinite(p)):
            parent = np.maximum(p, 0.0)
            tracker.update(parent)
            print(f"Parent loaded: n={parent.size}, J={tracker.best_score:.8f}", flush=True)
    except (NameError, AttributeError):
        pass

    # === Phase 1: Explore multiple n values ===
    # Allocate ~50% of time to exploration
    explore_deadline = time.monotonic() + 0.5 * float(budget_s)

    # Try n values, budget time per n based on LP cost
    n_values = [320, 384, 448, 512, 576, 640, 768]

    # If parent exists, prioritize its length
    if parent is not None:
        n_parent = parent.size
        if n_parent not in n_values:
            n_values.insert(0, n_parent)
        else:
            n_values.remove(n_parent)
            n_values.insert(0, n_parent)

    best_per_n = {}

    for n in n_values:
        if time.monotonic() > explore_deadline:
            break
        # Estimate time per LP for this n (rough: ~n^2 / 50000 seconds)
        est_lp_time = max(2.0, n * n / 50000.0)
        # Budget for this n
        remaining = max(0, explore_deadline - time.monotonic())
        n_remaining = len(n_values) - n_values.index(n)
        time_for_n = remaining / n_remaining
        if time_for_n < 30:
            break

        n_lp_iters = max(5, int(60.0 / est_lp_time))
        t0_n = time.monotonic()
        print(f"\n--- Phase1: n={n}, est_lp_time={est_lp_time:.1f}s, budget={time_for_n:.0f}s ---", flush=True)

        inits = generate_inits(n, rng, parent)
        n_inits = len(inits)
        init_budget = time_for_n / n_inits
        best_J_n = float('inf')
        best_a_n = None

        for idx, a0 in enumerate(inits):
            if time.monotonic() > explore_deadline:
                break
            t_init = time.monotonic()

            # Quick LP optimization
            a = run_lp_opt(a0, n_lp_iters, explore_deadline, tracker,
                          tighten=0.015, time_per_lp=est_lp_time)

            # Quick MW
            if time.monotonic() < explore_deadline - 10:
                for _ in range(20):
                    if time.monotonic() > explore_deadline - 10:
                        break
                    a_mw = multiplicative_weights_step(a)
                    J_mw = compute_J(a_mw)
                    if J_mw < compute_J(a) - 1e-14:
                        a = a_mw
                    else:
                        break

            # Quick LSE
            if time.monotonic() < explore_deadline - 10:
                a = lse_gradient_descent(a, 100, explore_deadline - 10, tracker,
                                         tau_start=20, tau_end=80)

            J = compute_J(a)
            if J < best_J_n:
                best_J_n = J
                best_a_n = a.copy()

            elapsed_init = time.monotonic() - t_init
            if idx % 3 == 0:
                print(f"    init {idx}/{n_inits}: J={J:.6f} ({elapsed_init:.1f}s)", flush=True)

        if best_a_n is not None:
            best_per_n[n] = (best_a_n, best_J_n)
            print(f"  n={n} best: J={best_J_n:.8f} ({time.monotonic()-t0_n:.1f}s)", flush=True)

    # === Phase 2: Focused optimization on top candidates ===
    # Sort by best J
    sorted_ns = sorted(best_per_n.keys(), key=lambda x: best_per_n[x][1])
    top_ns = sorted_ns[:3] if len(sorted_ns) >= 3 else sorted_ns

    focus_deadline = time.monotonic() + 0.35 * float(budget_s)

    for n in top_ns:
        if time.monotonic() > focus_deadline:
            break
        a = best_per_n[n][0].copy()
        tracking_J = best_per_n[n][1]
        t0 = time.monotonic()
        print(f"\n--- Phase2: Focus n={n}, starting J={tracking_J:.8f} ---", flush=True)

        est_lp_time = max(2.0, n * n / 50000.0)

        # Progressive tightening LP
        tighten_schedule = [0.005, 0.01, 0.015, 0.02, 0.025, 0.03, 0.04]
        for tight in tighten_schedule:
            if time.monotonic() > focus_deadline - 30:
                break
            n_iters = max(3, int(40.0 / est_lp_time))
            a = run_lp_opt(a, n_iters, focus_deadline - 30, tracker,
                          tighten=tight, time_per_lp=est_lp_time)

        # Extended MW
        if time.monotonic() < focus_deadline - 20:
            for _ in range(100):
                if time.monotonic() > focus_deadline - 20:
                    break
                a_mw = multiplicative_weights_step(a)
                J_mw = compute_J(a_mw)
                if J_mw < compute_J(a) - 1e-15:
                    a = a_mw
                    tracker.update(a)
                else:
                    break

        # Extended LSE with higher tau
        if time.monotonic() < focus_deadline - 10:
            a = lse_gradient_descent(a, 500, focus_deadline - 10, tracker,
                                     tau_start=50, tau_end=400)

        print(f"  Phase2 n={n} done: J={compute_J(a):.8f} ({time.monotonic()-t0:.1f}s)", flush=True)

    # === Phase 3: Intensive polish on global best ===
    if tracker.best is not None and time.monotonic() < deadline - 10:
        t0 = time.monotonic()
        n = tracker.best.size
        print(f"\n--- Phase3: Polish n={n}, J={tracker.best_score:.8f} ---", flush=True)
        a = tracker.best.copy()
        est_lp_time = max(2.0, n * n / 50000.0)

        # Final LP with very tight parameters
        for tight in [0.003, 0.005, 0.008, 0.01]:
            if time.monotonic() > deadline - 30:
                break
            n_iters = max(2, int(20.0 / est_lp_time))
            a = run_lp_opt(a, n_iters, deadline - 30, tracker,
                          tighten=tight, time_per_lp=est_lp_time)

        # Intensive LSE
        if time.monotonic() < deadline - 10:
            a = lse_gradient_descent(a, 1000, deadline - 10, tracker,
                                     tau_start=100, tau_end=500)

        # Coordinate descent
        if time.monotonic() < deadline - 5:
            a = coordinate_descent(a, 2000, deadline - 5, tracker)

        # Final LSE
        if time.monotonic() < deadline - 1:
            a = lse_gradient_descent(a, 200, deadline - 1, tracker,
                                     tau_start=200, tau_end=500)

        print(f"  Phase3 done: J={compute_J(a):.8f} ({time.monotonic()-t0:.1f}s)", flush=True)

    # Fallback
    if tracker.best is None:
        # Generate a reasonable fallback
        n = 512
        t = np.linspace(0, 1, n, endpoint=False)
        a = 0.5 + 0.5 * np.cos(np.pi * t)
        a = np.maximum(a, 0.01)
        a /= a.sum()
        tracker.best = a
        tracker.best_score = compute_J(a)
        tracker.update(tracker.best)

    print(f"\n{'='*60}", flush=True)
    print(f"FINAL: J={tracker.best_score:.8f} n={tracker.best_n} time={time.monotonic()-tracker.t_start:.1f}s", flush=True)
    print(f"{'='*60}", flush=True)
    return [float(x) for x in tracker.best]
