```python
    m = np.zeros(n)  # first moment
    v = np.zeros(n)  # second moment
    t = 0
    for it in range(max_iters):
        t += 1
        # Compute gradient
        grad = compute_smooth_gradient(a, tau)
        # Adam update
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * grad**2
        m_hat = m / (1 - beta1**t)
        v_hat = v / (1 - beta2**t)
        a = a - lr * m_hat / (np.sqrt(v_hat) + eps)
```

```python
def fast_conv(a):
    """Full self-convolution via FFT."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]


def compute_J(a):
    """Compute J(a) = 2*n*max(conv(a,a))/(sum(a))^2."""
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))

    n = len(a)

        hi = min(n - 1, k)
        grad[lo:hi+1] += 2.0 * pk * a[k-hi:k-lo+1][::-1]
    return grad, b
```

```python
        if S_new < 1e-10:
            return 1e10
        b_new_affected = b_affected + c * delta
        if has_quad:
            b_new_affected[2*i - k_start] += delta * delta
        new_max = max(b_new_affected.max(), b_unaffected_max)
        return 2 * n * new_max / (S_new * S_new)
    
    # Minimize obj(delta) for delta >= -a_i
    # Use bounded scalar minimization
    lo = -a_i
    hi = max(2 * a_i, 0.1)
    
    result = minimize_scalar(obj, bounds=(lo, hi), method='bounded', 
                            options={'xatol': 1e-12})
    return result.x
```

This is O(n) per evaluation (the b_new_affected computation) and minimize_scalar does about 30-50 evaluations. So per coordinate: O(n * 50) = O(25600) for n=512. Per sweep: O(n^2 * 50) = O(13M). With 100 sweeps: 1.3B. In Python with numpy, this should take about 30-60 seconds per 100 sweeps. That's fine.
