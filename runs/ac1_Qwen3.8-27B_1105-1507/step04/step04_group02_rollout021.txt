"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using smooth-max L-BFGS optimization,
alternating LP refinement, and coordinate descent polishing.
Key innovation: FFT-based gradient computation for the smooth (log-sum-exp) 
approximation of the max-convolution objective, enabling efficient quasi-Newton 
optimization that outperforms subgradient methods.
"""

import numpy as np
from scipy.optimize import minimize, linprog, minimize_scalar
from scipy.sparse import coo_matrix
import time

# Global state for smooth max objective (to avoid closures)
_g_tau = 100.0
_g_n = 0

def compute_conv(a):
    """Full linear convolution of a with itself via FFT."""
    n = len(a)
    size = 2 * n - 1
    # Use next power of 2 for efficiency
    size_pow2 = 1
    while size_pow2 < 2 * n:
        size_pow2 <<= 1
    fa = np.fft.rfft(a, size_pow2)
    conv = np.fft.irfft(fa * fa, size_pow2)
    return conv[:2 * n - 1]

def compute_obj(a):
    """Compute 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    S = float(np.sum(a))
    if S < 1e-10:
        return np.inf
    conv = compute_conv(a)
    M = float(np.max(conv))
    return 2.0 * n * M / (S * S)

def smooth_max_obj(a):
    """Smooth max of conv(a,a) using log-sum-exp with global tau."""
    tau = _g_tau
    conv = compute_conv(a)
    max_c = np.max(conv)
    shifted = tau * (conv - max_c)
    # Numerical stability: clip shifted values
    shifted = np.clip(shifted, -500, 500)
    return max_c + np.log(np.sum(np.exp(shifted))) / tau

def smooth_max_grad(a):
    """Gradient of smooth max of conv(a,a) via FFT."""
    tau = _g_tau
    n = len(a)
    conv = compute_conv(a)
    max_c = np.max(conv)
    shifted = tau * (conv - max_c)
    shifted = np.clip(shifted, -500, 500)
    exp_vals = np.exp(shifted)
    p = exp_vals / np.sum(exp_vals)
    
    # Compute gradient: grad[j] = sum_k p[k] * (a[k-j] + a[j-k]) with bounds
    # = t1[j] + t2[j]
    # where t1[j] = sum_m a[m] * p[m+j] (cross-correlation)
    # and t2[j] = sum_m a[m] * p[j-m] (linear conv, first n elements)
    
    # FFT size
    N = 1
    while N < 3 * n:
        N <<= 1
    
    fa = np.fft.rfft(a, N)
    fp = np.fft.rfft(p, N)
    
    # t2 = linear conv(a, p)[0:n]
    t2 = np.fft.irfft(fa * fp, N)[:n]
    
    # t1[j] = conv(a, p_rev)[2n-2-j] for j=0..n-1
    # = conv(a, p_rev)[n-1 : 2n-1] reversed
    p_rev = p[::-1]
    fp_rev = np.fft.rfft(p_rev, N)
    conv_apr = np.fft.irfft(fa * fp_rev, N)
    t1 = conv_apr[n-1:2*n-1][::-1]
    
    return t1 + t2

def smooth_max_optimize(a_init, time_budget, verbose=False):

def cosine_init(n, n_modes=8, seed=0):
    """Cosine series initialization."""
    rng = np.random.default_rng(seed)
    k = np.arange(n)
    a = np.ones(n)
    for m in range(1, n_modes + 1):
        c = rng.uniform(-0.5, 0.5) / m
        phase = rng.uniform(0, 2 * np.pi)
        a += c * np.cos(2 * np.pi * m * k / n + phase)
    a = np.maximum(a, 0)
    S = np.sum(a)
    if S < 1e-10:
        return np.ones(n) / n
    return a / S

def gaussian_init(n, sigma_frac=0.3):
    """Gaussian bump."""
    k = np.arange(n)
    center = n / 2.0
    sigma = sigma_frac * n
    a = np.exp(-0.5 * ((k - center) / sigma) ** 2)
    return a / np.sum(a)

def raised_cosine_init(n, alpha=1.0):
    """Raised cosine."""
    k = np.arange(n)
    x = 2.0 * k / (n - 1) - 1.0
    a = 0.5 * (1.0 + np.cos(np.pi * x)) ** alpha
    a = np.maximum(a, 1e-10)
    return a / np.sum(a)

def random_smooth_init(n, seed=0):
    """Random smooth initialization via low-pass filtered noise."""
    rng = np.random.default_rng(seed)
    a = rng.random(n)
    # Apply a simple smoothing (moving average)
    kernel = np.ones(5) / 5
    a = np.convolve(a, kernel, mode='same')
    a = np.maximum(a, 0)
    S = np.sum(a)
    if S < 1e-10:
        return np.ones(n) / n
    return a / S

def build_toeplitz_sparse(a):
    """Build sparse matrix for conv(a, g) where g is the variable."""
    n = len(a)
    rows = []
    cols = []
    vals = []
    for j in range(n):
        start = j
        end = min(j + n, 2 * n - 1)
        length = end - start
        rows.extend(range(start, end))
        cols.extend([j] * length)
        vals.extend(list(a[:length]))
    A = coo_matrix((vals, (rows, cols)), shape=(2 * n - 1, n)).tocsr()
    return A

def solve_lp_maxsum(a, M, time_limit=10):
    """Solve: max sum(g) s.t. conv(a,g) <= M, g >= 0."""
    n = len(a)
    A = build_toeplitz_sparse(a)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': time_limit})
        if res.status == 0:
            return np.maximum(res.x, 0)
    except Exception:
        pass
    return None

def alternating_lp(a_init, max_iters=50, time_budget=60.0, verbose=False):
    """Alternating LP refinement."""
    n = len(a_init)
    a = a_init.copy().astype(np.float64)
    S = np.sum(a)
    if S < 1e-10:
        return a
    a = a / S
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    stall = 0
    
    for it in range(max_iters):
        if time.monotonic() - start > time_budget - 3:
            break
        
        conv = compute_conv(a)
        M = float(np.max(conv))
        
        g = solve_lp_maxsum(a, M, time_limit=max(2, time_budget / 10))
        if g is None:
            break
        
        S_g = np.sum(g)
        if S_g < 1e-10:
            break
        g = g / S_g
        
        # Line search
        best_t = 0.0
        best_obj_ls = best_obj
        for t in [0.1, 0.2, 0.3, 0.5, 0.7, 0.9, 1.0]:
            a_try = np.maximum((1.0 - t) * a + t * g, 0)
            S_try = np.sum(a_try)
            if S_try < 1e-10:
                continue
            a_try = a_try / S_try
            obj_try = compute_obj(a_try)
            if obj_try < best_obj_ls:
                best_obj_ls = obj_try
                best_t = t
        
        if best_t > 0:
            a = np.maximum((1.0 - best_t) * a + best_t * g, 0)
            a = a / np.sum(a)
            best_obj = best_obj_ls
            best_a = a.copy()
            stall = 0
            if verbose:
                print(f"    LP it={it}: obj={best_obj:.8f}, t={best_t}")
        else:
            stall += 1
            if stall > 10:
                break
    
    return best_a

def coordinate_descent(a_init, max_sweeps=5, time_budget=120.0, verbose=False):
    """Coordinate descent with 1D optimization per coordinate."""
    n = len(a_init)
    a = a_init.copy().astype(np.float64)
    S = np.sum(a)
    a = a / S
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    
    rng = np.random.default_rng(42)
    
    for sweep in range(max_sweeps):
        if time.monotonic() - start > time_budget - 10:
            break
        
        # Random order
        order = rng.permutation(n)
        
        for i in order:
            if time.monotonic() - start > time_budget - 5:
