import numpy as np
from scipy.optimize import linprog, minimize
from scipy.sparse import lil_matrix
import time

def fast_conv(a):
    n = len(a)
    s = 1
    while s < 2*n - 1:
        s <<= 1
    fa = np.fft.rfft(a, s)
    return np.fft.irfft(fa * fa, s)[:2*n - 1]

def compute_J(a):
    a = np.maximum(a, 0)
    S = a.sum()
    if S < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (S * S))

def build_A_sparse(a):
    n = len(a)
    rows = []
    cols = []
    vals = []
    for k in range(2*n - 1):
        jl = max(0, k - n + 1)
        jh = min(n - 1, k)
        for j in range(jl, jh + 1):
            rows.append(k)
            cols.append(j)
            vals.append(a[k - j])
    A = lil_matrix((2*n - 1, n))
    for idx in range(len(rows)):
        A[rows[idx], cols[idx]] = vals[idx]
    return A.tocsr()

def lp_best_response(a, M):
    n = len(a)
    c = -np.ones(n)
    A = build_A_sparse(a)
    b_ub = np.full(2*n - 1, M)
    bounds = [(0.0, None)] * n
    r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs', 
                options={'presolve': True, 'time_limit': 10})
    if r.success and r.x is not None:
        g = np.maximum(r.x, 0.0)
        if g.sum() > 1e-10:
            return g
    return None

def multiplicative_weights(a, g, t):
    """Geometric interpolation: a_new ∝ a * (g/mean(g))^t"""
    g_norm = g / g.sum() * len(a)  # normalize to have same mean as uniform
    # Actually, let's normalize g to have the same sum as a
    g_scaled = g * (a.sum() / g.sum())
    a_new = a * (g_scaled / a) ** t  # This is a^(1-t) * g_scaled^t
    a_new = np.maximum(a_new, 0.0)
    if a_new.sum() < 1e-15:

    return a_new / a_new.sum() * a.sum()  # preserve sum

def softmax_objective_and_grad(a, beta):

    b = fast_conv(a)
    # Shift for numerical stability

    exp_b = np.exp(beta * b_shifted)
    Z = exp_b.sum()
    p = exp_b / Z

    # Gradient
    n = len(a)

    for k in range(2*n - 1):

        jl = max(0, k - n + 1)

                    grad[j] += p[k] * a[j]
                else:

    return f_val, grad

def fast_conv(a):
    """Full self-convolution via FFT."""
    n = len(a)
    if n == 0:
        return np.array([0.0])
    s = 1
    while s < 2 * n - 1:
        s <<= 1
    fa = np.fft.rfft(a, s)
    return np.fft.irfft(fa * fa, s)[:2 * n - 1]


def compute_J(a):
    """Compute J(a) = 2*n*max(conv(a,a))/(sum(a))^2."""
    a = np.maximum(a, 0.0)
    S = a.sum()
    if S < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (S * S))


def build_A_sparse(a):
    """Build sparse constraint matrix for LP: (a*g)_k for all k."""
    n = len(a)
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        jl = max(0, k - n + 1)
        jh = min(n - 1, k)
        for j in range(jl, jh + 1):
            rows.append(k)
            cols.append(j)
            vals.append(a[k - j])
    A = lil_matrix((2 * n - 1, n))
    for idx in range(len(rows)):
        A[rows[idx], cols[idx]] = vals[idx]
    return A.tocsr()


def lp_best_response(a, M):
    """LP: max sum(g) s.t. (a*g)_k <= M for all k, g >= 0."""
    n = len(a)

                    options={'time_limit': 15, 'presolve': True})

            g = np.maximum(r.x, 0.0)

    except Exception:
        pass
    return None


def multiplicative_update(a, g, t):

    if g_bar < 1e-15:

    ratio = np.maximum(ratio, 1e-30)
    log_a = np.log(np.maximum(a, 1e-30))
    log_g = np.log(ratio)
    log_new = (1.0 - t) * log_a + t * log_g

    log_new -= log_new.max()
    a_new = np.exp(log_new)
    S = a_new.sum()
    if S < 1e-15:

    return a_new / S * a.sum()

def project_simplex(v):
    """Project v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if v.sum() < 1e-15:
        return np.ones(n) / n
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.ones(n) / n
    k = np.max(ind[cond])
    theta = cssv[k - 1] / k
    return np.maximum(v - theta, 0.0)

def cd_step(a, i, b, S):
    """Find optimal value for coordinate i given others fixed."""
    n = len(a)
    ai = a[i]
    
    def f_at(x):
        new_S = S - ai + x
        if new_S < 1e-10:
            return 1e10
        max_c = 0.0
        # Only need to check k in [i, i+n-1]
        for k in range(max(0, i), min(2 * n - 1, i + n)):
            ki = k - i
            if 0 <= ki < n:
                if 2 * i == k:
                    ck = b[k] - ai * ai + x * x
                else:
                    ck = b[k] - 2.0 * ai * a[ki] + 2.0 * x * a[ki]
            else:
                ck = b[k]
            if ck > max_c:
                max_c = ck
        # Also check unchanged k's
        for k in range(2 * n - 1):
            if k < i or k >= i + n:
                if b[k] > max_c:
                    max_c = b[k]
        return 2.0 * n * max_c / (new_S * new_S)
    
    # Golden section search
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    lo = 0.0
    hi = max(3.0 * ai + 0.01, S / n * 3.0, 0.1)
    
    c_pt = hi - gr * (hi - lo)
    d_pt = lo + gr * (hi - lo)
    fc = f_at(c_pt)
    fd = f_at(d_pt)
    
    for _ in range(30):
        if fc < fd:
            hi = d_pt
            d_pt = c_pt
            fd = fc
            c_pt = hi - gr * (hi - lo)
            fc = f_at(c_pt)
        else:
            lo = c_pt
            c_pt = d_pt
            fc = fd
            d_pt = lo + gr * (hi - lo)
            fd = f_at(d_pt)
    
    return max(0.0, (lo + hi) / 2.0)


def coordinate_descent_pass(a, b=None, S=None):
    """One full pass of coordinate descent."""
    a = np.maximum(a.astype(np.float64), 0.0)
    S = a.sum()

    a = a / S

def resample_seq(src, dest_n):
    """Resample 1D sequence to new length via linear interpolation."""
    src = np.asarray(src, dtype=np.float64)
    if src.size == 0:
        return np.ones(dest_n) / dest_n
    if src.size == dest_n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    return np.maximum(np.interp(t_new, t_old, src), 0.001)
