def multiplicative_weights_descent(a, max_iters, deadline, eta0=0.5):

def nesterov_subgradient(a, max_iters, deadline):
    a = np.maximum(a.astype(np.float64), 1e-12)
    a = a / a.sum()
    n = len(a)

    best_a = a.copy()
    best_M = fast_conv(a).max()
    best_J = 2 * n * best_M

    y = a.copy()
    t_k = 1.0
    no_improve = 0

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        c = fast_conv(y)
        M = c.max()
        k_star = int(np.argmax(c))

        grad = np.zeros(n)
        for i in range(n):
            idx = k_star - i
            if 0 <= idx < n:
                grad[i] = 2.0 * y[idx]

        grad = grad - grad.mean()
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-15:
            break

        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_k * t_k))
        alpha = (t_k - 1.0) / t_new
        step = 1.0 / (gnorm * np.sqrt(t_new + 1.0))
        step = min(step, 0.3)

        x_new = y - step * grad
        x_new = np.maximum(x_new, 0)
        s_new = x_new.sum()
        if s_new < 1e-15:
            break
        x_new = x_new / s_new

        y = x_new + alpha * (x_new - y)
        y = np.maximum(y, 0)
        y = y / y.sum()
        t_k = t_new

        new_M = fast_conv(y).max()
        new_J = 2 * n * new_M
        if new_J < best_J:
            best_J = new_J
            best_a = y.copy()
            no_improve = 0
        else:
            no_improve += 1
            if no_improve > 200:
                break

def resample_seq(src, dest_n):
    src = np.asarray(src, dtype=np.float64)

    t_new = np.linspace(0, 1, dest_n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)

        score_tracker_update(tracker, a_lp)
        if compute_J(a_lp) < compute_J(best_overall):
            best_overall = a_lp.copy()
        print(f"  J={compute_J(best_overall):.8f} ({time.monotonic()-t0:.1f}s)", flush=True)

    # === Phase 4: LP mirror descent (multiplicative) ===
    if time.monotonic() < deadline - 400:
        t0 = time.monotonic()
        print(f"\nPhase 4: LP multiplicative (n={best_overall.size})", flush=True)
        dl = min(deadline - 300, time.monotonic() + 150)
        a_lp2, J_lp2 = lp_mirror_descent(best_overall, 200, dl, use_mult=True)
        score_tracker_update(tracker, a_lp2)
        if compute_J(a_lp2) < compute_J(best_overall):
            best_overall = a_lp2.copy()
        print(f"  J={compute_J(best_overall):.8f} ({time.monotonic()-t0:.1f}s)", flush=True)

    # === Phase 5: Another round of multiplicative weights (tuned) ===
    if time.monotonic() < deadline - 300:
        t0 = time.monotonic()
        print(f"\nPhase 5: MW tuned (n={best_overall.size})", flush=True)
        dl = min(deadline - 200, time.monotonic() + 150)
        a_mw2, J_mw2 = multiplicative_weights_descent(best_overall, 5000, dl, eta0=0.1)
        score_tracker_update(tracker, a_mw2)
        if compute_J(a_mw2) < compute_J(best_overall):
            best_overall = a_mw2.copy()
        print(f"  J={compute_J(best_overall):.8f} ({time.monotonic()-t0:.1f}s)", flush=True)

    # === Phase 6: Multi-length exploration ===
    if time.monotonic() < deadline - 250:
        t0 = time.monotonic()
        print(f"\nPhase 6: Multi-length exploration", flush=True)
        base_n = best_overall.size
        lengths = [base_n, base_n + 32, base_n - 32, base_n + 64, max(128, base_n - 64)]
        lengths = sorted(set([n for n in lengths if 64 <= n <= 4096]))

        for n_try in lengths:
            if time.monotonic() > deadline - 200:
                break
            a_try = resample_seq(best_overall, n_try)
            a_try = np.maximum(a_try, 1e-10)
            a_try = a_try / a_try.sum()

            # Quick MW + LP
            dl1 = min(time.monotonic() + 30, deadline - 180)
            a_try, _ = multiplicative_weights_descent(a_try, 500, dl1, eta0=0.3)
            score_tracker_update(tracker, a_try)

            if time.monotonic() < deadline - 160:
                dl2 = min(time.monotonic() + 20, deadline - 140)
                a_try, _ = lp_mirror_descent(a_try, 50, dl2, use_mult=False)
                score_tracker_update(tracker, a_try)

            if tracker[1] is not None:
                J_try = compute_J(tracker[1])
                J_best = compute_J(best_overall)
                if J_try < J_best:
                    best_overall = tracker[1].copy()
                    print(f"  n={n_try}: J={J_try:.8f} (NEW BEST)", flush=True)
                else:
                    print(f"  n={n_try}: J={compute_J(a_try):.8f}", flush=True)

        print(f"Phase 6 done ({time.monotonic()-t0:.1f}s)", flush=True)

    # === Phase 7: Random restarts with diverse inits ===
    if time.monotonic() < deadline - 150:
        t0 = time.monotonic()
        print(f"\nPhase 7: Random restarts", flush=True)
        n_base = best_overall.size
        n_inits = 8
        per_init_time = (deadline - 120 - time.monotonic()) / max(1, n_inits)
        per_init_time = max(5, min(30, per_init_time))

        inits = generate_diverse_inits(n_base, rng, parent=best_overall)
        rng.shuffle(inits)

        for i, a_init in enumerate(inits[:n_inits]):
            if time.monotonic() > deadline - 120:
                break
            a_init = np.maximum(a_init, 1e-10)
            a_init = a_init / a_init.sum()

            dl1 = time.monotonic() + per_init_time * 0.4
            a_init, _ = multiplicative_weights_descent(a_init, 500, dl1, eta0=0.5)

            if time.monotonic() < deadline - 110:
                dl2 = time.monotonic() + per_init_time * 0.4
                a_init, _ = lp_mirror_descent(a_init, 30, dl2, use_mult=False)

            score_tracker_update(tracker, a_init)
            J_try = compute_J(a_init)
            J_best = tracker[0] if tracker[1] is not None else float('inf')
            print(f"  restart {i+1}: J={J_try:.8f}", flush=True)

        print(f"Phase 7 done ({time.monotonic()-t0:.1f}s)", flush=True)

    # === Phase 8: Long LP refinement on best ===
    if time.monotonic() < deadline - 50:
        t0 = time.monotonic()
        print(f"\nPhase 8: Long LP refinement (n={best_overall.size})", flush=True)
        a_long = best_overall.copy()
        a_long = np.maximum(a_long, 1e-10)
        a_long = a_long / a_long.sum()
        a_long, J_long = lp_mirror_descent(a_long, 5000, deadline - 40, use_mult=False)
        score_tracker_update(tracker, a_long)
        if compute_J(a_long) < compute_J(best_overall):
            best_overall = a_long.copy()
        print(f"  J={compute_J(best_overall):.8f} ({time.monotonic()-t0:.1f}s)", flush=True)

    # === Phase 9: Final MW + Nesterov polish ===
    if time.monotonic() < deadline - 15:
        t0 = time.monotonic()
        print(f"\nPhase 9: Final polish", flush=True)
        a_polish = best_overall.copy()
        dl1 = min(time.monotonic() + 20, deadline - 10)
        a_polish, _ = multiplicative_weights_descent(a_polish, 2000, dl1, eta0=0.05)
        score_tracker_update(tracker, a_polish)
        if compute_J(a_polish) < compute_J(best_overall):
            best_overall = a_polish.copy()

        if time.monotonic() < deadline - 5:
            a_polish2 = best_overall.copy()
            a_polish2, _ = nesterov_subgradient(a_polish2, 2000, deadline - 3)
            score_tracker_update(tracker, a_polish2)
            if compute_J(a_polish2) < compute_J(best_overall):
                best_overall = a_polish2.copy()
        print(f"  Final J={compute_J(best_overall):.8f} ({time.monotonic()-t0:.1f}s)", flush=True)

    # === Final: Use tracker's best (validated by evaluate_sequence) ===
    if tracker[1] is not None:
        final_a = tracker[1]
    else:
        final_a = best_overall

    final_score = float(evaluate_sequence(final_a.tolist()))
    print(f"\n{'='*60}", flush=True)
    print(f"FINAL: J={final_score:.8f} n={final_a.size} time={time.monotonic()-t_start:.0f}s", flush=True)
    print(f"{'='*60}", flush=True)

    return [float(x) for x in final_a]
