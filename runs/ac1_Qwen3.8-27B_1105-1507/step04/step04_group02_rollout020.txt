        tau = tau_start + (tau_end - tau_start) * min(1.0, it / (max_iters * 0.4))
        grad, M, conv, p = softmax_subgradient(a, tau, topk=topk)
        support = a > 1e-12
        if not np.any(support):
            break
        g_proj = grad.copy()
        g_proj[~support] = 0.0
        g_proj[support] -= np.mean(g_proj[support])
        if g_prev is not None and it > 2:
            alpha_bb = bb_step_size(a_prev, a, g_prev, g_proj)
            alpha = 0.7 * alpha + 0.3 * alpha_bb
        v = momentum * v + (1.0 - momentum) * g_proj
        v[~support] = 0.0
        v[support] -= np.mean(v[support])
        a_prev = a.copy()
        g_prev = g_proj.copy()
        found = False
        for ls in range(25):
            a_try = a - alpha * v
            a_try = np.maximum(a_try, 0.0)
            S_try = np.sum(a_try)
            if S_try < 1e-10:
                break
            a_try = a_try / S_try
            obj_try = compute_obj(a_try)
            if obj_try < best_obj:
                best_obj = obj_try
                best_a = a_try.copy()
                a = a_try
                found = True
                break
            alpha *= 0.5
        if found:
            stall = 0
            alpha = min(alpha * 1.2, 0.5)
        else:
            stall += 1
            alpha *= 0.7
            if stall > 80:
                rng = np.random.default_rng(seed + it * 7)
                a = a * np.exp(rng.normal(0, 0.08, n))
                a = np.maximum(a, 0.0)
                S = np.sum(a)
                if S < 1e-10:
                    break
                a = a / S
                v = np.zeros(n)
                alpha = 0.05
                stall = 0
            if alpha < 1e-9:
                break
    return best_a


def build_toeplitz(a):
    n = len(a)
    rows = []
    cols = []
    vals = []
    for j in range(n):
        start = j
        end = min(j + n, 2 * n - 1)
        length = end - start
        rows.extend(range(start, end))
        cols.extend([j] * length)
        vals.extend(list(a[:length]))
    from scipy.sparse import coo_matrix
    A = coo_matrix((vals, (rows, cols)), shape=(2 * n - 1, n)).tocsr()
    return A


def lp_direction(a, M_bound, time_limit=10):
    n = len(a)
    A = build_toeplitz(a)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M_bound)
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': time_limit})
        if res.status == 0:
            return np.maximum(res.x, 0.0)
    except Exception:
        pass
    return None


def lp_refine(a, time_budget, max_rounds=50):
    n = len(a)
    a = a.astype(np.float64).copy()
    S = np.sum(a)
    if S < 1e-10:
        return a
    a = a / S
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    stall = 0

    for rnd in range(max_rounds):
        if time.monotonic() - start > time_budget:
            break
        conv = compute_conv(a)
        M = float(np.max(conv))
        g = lp_direction(a, M * 0.999, time_limit=min(8, (time_budget - (time.monotonic() - start)) * 0.3))
        if g is None:
            break
        S_g = np.sum(g)
        if S_g < 1e-10:
            break
        g = g / S_g
        best_t = 0.0
        best_ls = best_obj
        for t in [0.05, 0.1, 0.15, 0.25, 0.4, 0.6, 0.8]:
            a_try = (1.0 - t) * a + t * g
            a_try = np.maximum(a_try, 0.0)
            S_try = np.sum(a_try)
            if S_try < 1e-10:
                continue
            a_try = a_try / S_try
            obj_try = compute_obj(a_try)
            if obj_try < best_ls:
                best_ls = obj_try
                best_t = t
        if best_t > 0.0:
            a = (1.0 - best_t) * a + best_t * g
            a = np.maximum(a, 0.0)
            a = a / np.sum(a)
            best_obj = best_ls
            best_a = a.copy()
            stall = 0
        else:
            stall += 1
            if stall > 10:
                rng = np.random.default_rng(rnd * 104729 + 7)
                a = a * np.exp(rng.normal(0, 0.1, n))
                a = np.maximum(a, 0.0)
                a = a / np.sum(a)
                stall = 0
            else:
                a = 0.95 * a + 0.05 * g
                a = np.maximum(a, 0.0)
                a = a / np.sum(a)
    return best_a


def raised_cosine_init(n, alpha=2.0):
    k = np.arange(n, dtype=np.float64)
    x = 2.0 * k / (n - 1) - 1.0
    a = (0.5 * (1.0 + np.cos(np.pi * x))) ** alpha
    a = np.maximum(a, 1e-10)
    return a / np.sum(a)


def gaussian_init(n, sigma_frac=0.3):
    k = np.arange(n, dtype=np.float64)
    center = (n - 1) / 2.0
    sigma = sigma_frac * n
    a = np.exp(-0.5 * ((k - center) / sigma) ** 2)
    return a / np.sum(a)


def cosine_mix_init(n, n_modes=10, seed=0):
    rng = np.random.default_rng(seed)
    k = np.arange(n, dtype=np.float64)
    a = np.ones(n)
    for m in range(1, n_modes + 1):
        c = rng.uniform(-0.5, 0.5) / m
        phase = rng.uniform(0, 2.0 * np.pi)
        a += c * np.cos(2.0 * np.pi * m * k / n + phase)
    a = np.maximum(a, 1e-10)
    return a / np.sum(a)


def beta_init(n, alpha=2.0, beta=3.0):
    k = np.arange(n, dtype=np.float64)
    x = k / (n - 1)
    a = x ** (alpha - 1) * (1.0 - x) ** (beta - 1)
    a = np.maximum(a, 1e-10)
    return a / np.sum(a)


def fourier_perturb(a, rng, strength=0.05):
    n = len(a)
    fa = np.fft.rfft(a)
    phase_pert = rng.normal(0, strength, len(fa))
    mag_pert = np.exp(rng.normal(0, strength * 0.5, len(fa)))
    fa_new = fa * mag_pert * np.exp(1j * phase_pert)
    a_new = np.fft.irfft(fa_new, n)
    a_new = np.maximum(a_new, 0.0)
    S = np.sum(a_new)
    if S < 1e-10:
        return a
    return a_new / S


def refine_pipeline(a, time_budget, use_lp=True, seed=0):
    n = len(a)
    a = a.astype(np.float64).copy()
    S = np.sum(a)
    if S < 1e-10:
        return a, np.inf
    a = a / S
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    round_num = 0

    while time.monotonic() - start < time_budget - 3:
        round_num += 1
        remaining = time_budget - (time.monotonic() - start)
        if remaining < 5:
            break
        sg_budget = min(remaining * 0.4, 40.0)
        a_sg = smoothed_descent(best_a, time_budget=sg_budget, tau_start=5, tau_end=200, topk=50, seed=seed + round_num)
        obj_sg = compute_obj(a_sg)
        if obj_sg < best_obj:
            best_obj = obj_sg
            best_a = a_sg.copy()
        remaining = time_budget - (time.monotonic() - start)
        if remaining > 10 and use_lp and n <= 1200:
            lp_budget = min(remaining * 0.3, 30.0)
            a_lp = lp_refine(best_a, time_budget=lp_budget, max_rounds=30)
            obj_lp = compute_obj(a_lp)
            if obj_lp < best_obj:
                best_obj = obj_lp
                best_a = a_lp.copy()
        remaining = time_budget - (time.monotonic() - start)
        if remaining > 8:
            rng = np.random.default_rng(seed + round_num * 31 + 7)
            for pert_i in range(2):
                if time.monotonic() - start > time_budget - 4:
                    break
                a_pert = fourier_perturb(best_a, rng, strength=0.06)
                a_pert = smoothed_descent(a_pert, time_budget=min(5.0, remaining * 0.15), tau_start=10, tau_end=100, topk=30, seed=seed + 100 + pert_i)
                obj_pert = compute_obj(a_pert)
                if obj_pert < best_obj:
                    best_obj = obj_pert
                    best_a = a_pert.copy()
    return best_a, best_obj


def propose_candidate(budget_s=1500):
    start_time = time.monotonic()
    deadline = start_time + budget_s - 15.0

    best_seq = None
    best_score = np.inf

    try:
        parent = np.array(height_sequence_1, dtype=np.float64)
        parent = np.clip(parent, 0, 1000.0)
        if parent.size > 0 and np.isfinite(parent).all():
            best_seq = parent.copy()
            best_score = compute_obj(parent)
            print(f"Parent: n={parent.size}, obj={best_score:.6f}")
    except Exception:
        pass

    if best_seq is None:
        best_seq = raised_cosine_init(500, alpha=2.0)
        best_score = compute_obj(best_seq)
        print(f"Default init: obj={best_score:.6f}")

    n_base = len(best_seq)
    print(f"Starting optimization with n={n_base}, obj={best_score:.6f}")

    n_list = sorted(set([
        max(200, n_base - 100),
        n_base,
        n_base + 100,
        n_base + 200,
        n_base + 400,
        min(3000, n_base + 800),
    ]))
    n_list = [n for n in n_list if 150 <= n <= 3000]

    print("\n=== Phase 1: Multi-start exploration ===")
    candidate_pool = []
    for n_try in n_list:
        if time.monotonic() > deadline - 60:
            break
        remaining = deadline - time.monotonic()
        if remaining < 30:
            break
        inits = [
            raised_cosine_init(n_try, alpha=1.5),
            raised_cosine_init(n_try, alpha=2.5),
            raised_cosine_init(n_try, alpha=4.0),
            gaussian_init(n_try, sigma_frac=0.25),
            gaussian_init(n_try, sigma_frac=0.35),
            beta_init(n_try, 2.0, 3.0),
            beta_init(n_try, 3.0, 2.0),
            cosine_mix_init(n_try, n_modes=8, seed=n_try + 1),
            cosine_mix_init(n_try, n_modes=12, seed=n_try + 2),
        ]
        if n_try == n_base and best_seq is not None:
            inits.append(best_seq.copy())
        for i, a_init in enumerate(inits):
            if time.monotonic() > deadline - 30:
                break
            a_quick = smoothed_descent(a_init, time_budget=3.0, tau_start=5, tau_end=50, topk=30, seed=n_try * 100 + i)
            obj_quick = compute_obj(a_quick)
            candidate_pool.append((obj_quick, n_try, a_quick, i))
            if obj_quick < best_score:
                best_score = obj_quick
                best_seq = a_quick.copy()
                print(f"  n={n_try} init{i}: {obj_quick:.6f} ** NEW BEST **")

    candidate_pool.sort(key=lambda x: x[0])
    print(f"\nTop 5 candidates: {[f'{c[0]:.5f}' for c in candidate_pool[:5]]}")

    print("\n=== Phase 2: Refine top candidates ===")
    for idx in range(min(6, len(candidate_pool))):
        if time.monotonic() > deadline - 60:
            break
        remaining = deadline - time.monotonic()
        if remaining < 30:
            break
        obj_c, n_c, a_c, _ = candidate_pool[idx]
        if obj_c > best_score + 0.03 and idx > 0:
            break
        alloc = min(remaining * 0.2, 80.0)
        print(f"  Refining n={n_c} (obj={obj_c:.6f}), budget={alloc:.0f}s")
        a_ref, obj_ref = refine_pipeline(a_c, time_budget=alloc, use_lp=(n_c <= 1000), seed=idx * 1000)
        if obj_ref < best_score:
            best_score = obj_ref
            best_seq = a_ref.copy()
            print(f"    ** NEW BEST: n={n_c}, obj={obj_ref:.6f} **")

    print(f"\n=== Phase 3: Intensive refinement (obj={best_score:.6f}) ===")
    a_cur = best_seq.copy()
    n_cur = len(a_cur)

    for rnd in range(20):
        if time.monotonic() > deadline - 40:
            break
        remaining = deadline - time.monotonic()
        if remaining < 20:
            break
        alloc = min(remaining * 0.25, 60.0)
        a_new, obj_new = refine_pipeline(a_cur, time_budget=alloc, use_lp=(n_cur <= 1000), seed=10000 + rnd)
        if obj_new < best_score:
            best_score = obj_new
            best_seq = a_new.copy()
            a_cur = a_new.copy()
            print(f"  R{rnd}: {best_score:.6f} ** NEW BEST **")
        else:
            print(f"  R{rnd}: {obj_new:.6f} (best={best_score:.6f})")
            if rnd % 3 == 2:
                rng = np.random.default_rng(50000 + rnd)
                a_pert = fourier_perturb(a_cur, rng, strength=0.1)
                a_pert = smoothed_descent(a_pert, time_budget=min(10.0, remaining * 0.2), tau_start=5, tau_end=100, topk=40, seed=50000 + rnd)
                obj_pert = compute_obj(a_pert)
                if obj_pert < best_score:
                    best_score = obj_pert
                    best_seq = a_pert.copy()
                    a_cur = a_pert.copy()
                    print(f"  R{rnd} pert: {best_score:.6f} ** NEW BEST **")

    print(f"\n=== Phase 4: Multi-n transfer (obj={best_score:.6f}) ===")
    if time.monotonic() < deadline - 40:
        n_best = len(best_seq)
        from scipy.interpolate import interp1d
        for delta in [100, 200, 300, 500, -100, -200]:
            if time.monotonic() > deadline - 20:
                break
            n_try = n_best + delta
            if n_try < 150 or n_try > 3000:
                continue
            x_old = np.linspace(0, 1, n_best)
            x_new = np.linspace(0, 1, n_try)
            try:
                f_interp = interp1d(x_old, best_seq, kind='cubic', fill_value='extrapolate')
                a_interp = np.maximum(f_interp(x_new), 0.0)
            except Exception:
                a_interp = raised_cosine_init(n_try, 2.0)
            S_i = np.sum(a_interp)
            if S_i > 1e-10:
                a_interp = a_interp / S_i
            remaining = deadline - time.monotonic()
            alloc = min(remaining * 0.25, 40.0)
            a_new, obj_new = refine_pipeline(a_interp, time_budget=alloc, use_lp=(n_try <= 1000), seed=90000 + n_try)
            if obj_new < best_score:
                best_score = obj_new
                best_seq = a_new.copy()
                print(f"  n={n_try}: {obj_new:.6f} ** NEW BEST **")
            else:
                print(f"  n={n_try}: {obj_new:.6f}")

    print(f"\n=== Phase 5: Final polish ===")
    if time.monotonic() < deadline - 10:
        remaining = deadline - time.monotonic()
        n_fin = len(best_seq)
        a_sg = smoothed_descent(best_seq, time_budget=remaining * 0.4, tau_start=20, tau_end=500, topk=80, seed=999999)
        obj_sg = compute_obj(a_sg)
        if obj_sg < best_score:
            best_score = obj_sg
            best_seq = a_sg.copy()
            print(f"  SG polish: {best_score:.6f} ** NEW BEST **")
        remaining = deadline - time.monotonic()
        if remaining > 15 and n_fin <= 800:
            a_lp = lp_refine(best_seq, time_budget=remaining * 0.4, max_rounds=40)
            obj_lp = compute_obj(a_lp)
            if obj_lp < best_score:
                best_score = obj_lp
                best_seq = a_lp.copy()
                print(f"  LP polish: {best_score:.6f} ** NEW BEST **")
        remaining = deadline - time.monotonic()
        if remaining > 10:
            rng = np.random.default_rng(424242)
            for pi in range(3):
                if time.monotonic() > deadline - 5:
                    break
                a_p = best_seq * np.exp(rng.normal(0, 0.03, n_fin))
                a_p = np.maximum(a_p, 0.0)
                a_p = a_p / np.sum(a_p)
                a_p = smoothed_descent(a_p, time_budget=min(8.0, remaining * 0.25), tau_start=50, tau_end=300, topk=80, seed=424242 + pi)
                obj_p = compute_obj(a_p)
                if obj_p < best_score:
                    best_score = obj_p
                    best_seq = a_p.copy()
                    print(f"  Final pert{pi}: {best_score:.6f} ** NEW BEST **")

    if best_seq is None:
        best_seq = raised_cosine_init(500, 2.0)
        best_score = compute_obj(best_seq)

    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    val = evaluate_sequence(result)
    if val < np.inf:
        print(f"\n=== FINAL: n={len(result)}, score={val:.6f}, time={time.monotonic()-start_time:.1f}s ===")
    else:
        result = [1.0] * 300
        print(f"\nValidation failed, using fallback. time={time.monotonic()-start_time:.1f}s")

    return result
