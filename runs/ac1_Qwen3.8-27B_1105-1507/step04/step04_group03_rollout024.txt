import numpy as np

def fast_conv(a):
    """Full self-convolution via FFT."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]


def compute_J(a):
    """Compute J(a) = 2*n*max(conv(a,a))/(sum(a))^2."""
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))


def build_A_rows(n, a, lags):
    """Build sparse rows for given lags."""
    rows, cols, vals = [], [], []

        for j in range(lo, hi + 1):

            cols.append(j)
            vals.append(a[k - j])

                    a = a_new
        except Exception:
            pass

    return best_a


def subgradient_polish(a, max_iters, deadline):
    """Subgradient descent with Gauss-Southwell and momentum."""
    a = np.maximum(a.astype(np.float64), 1e-10)
    a = a / a.sum()
    n = len(a)
    best_M = fast_conv(a).max()
    best_a = a.copy()
    momentum = np.zeros_like(a)
    beta = 0.8

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        b = fast_conv(a)
        M = b.max()
        # Gauss-Southwell: pick the largest subgradient norm
        # Subgradient at lag k: grad_i = 2*a[k-i]
        # We use the top few lags
        tol = 1e-10 * M
        active = np.where(b >= M - max(tol, 0.05 * M))[0]

        grad = np.zeros_like(a)
        for k in active:
            lo = max(0, k - n + 1)
            hi = min(n - 1, k)
            for i in range(lo, hi + 1):
                grad[i] += 2.0 * a[k - i]

        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-15:
            break

        t = 2.0 / (gnorm * (it + 1.0 ))
        momentum = beta * momentum + (1 - beta) * grad / gnorm
        a_new = a - t * momentum
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s < 1e-10:
            break
        a_new = a_new / s

        new_M = fast_conv(a_new).max()
        if new_M < best_M:
            best_M = new_M
            best_a = a_new.copy()
        a = a_new

    return best_a


def resample_seq(src, dest_n):
    """Resample 1D sequence to new length."""
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size, endpoint=False)
    t_new = np.linspace(0, 1, dest_n, endpoint=False)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)


def generate_diverse_inits(n, rng, parent=None):
    """Generate diverse initial sequences of length n."""
    inits = []
    t = np.arange(n, dtype=np.float64) / n

    if parent is not None:
        p = resample_seq(parent, n)
        inits.append(p)
        inits.append(np.maximum(p + rng.normal(0, 0.03, n), 0.01))
        inits.append(np.maximum(p + rng.normal(0, 0.08, n), 0.01))
        # Shifted versions
        for s in [1, n // 16, n // 8]:
            inits.append(np.roll(p, s))

    # Standard initializations
    inits.append(np.ones(n) / n)
    inits.append(0.5 * (1 + np.cos(2 * np.pi * t)) + 0.01)
    inits.append(np.maximum(0.3 - 0.4 * np.cos(2 * np.pi * t) + 0.1 * np.cos(4 * np.pi * t), 0.01))
    inits.append(np.exp(-3 * (t - 0.5) ** 2) + 0.1)
    for f in [1, 2, 3, 5, 7, 13]:
        inits.append(np.maximum(1.0 + 0.4 * np.cos(2 * np.pi * f * t + rng.uniform(0, 2 * np.pi)), 0.01))
    # Random smooth
    kernel_size = max(3, n // 16)
    kernel = np.ones(kernel_size) / kernel_size
    smooth = np.convolve(rng.uniform(0, 1, n), kernel, mode='same')
    inits.append(np.maximum(smooth + 0.1, 0.01))
    # Raised cosine
    inits.append(0.5 * (1 + np.cos(np.pi * (2 * t - 1))) + 0.05)
    # Kronecker-like pattern
    half = n // 2
    pat = np.ones(half)
    pat[half // 2] = 0.5
    inits.append(np.tile(pat, 2) / n)

    result = []
    for a in inits:
        a = np.maximum(np.asarray(a, dtype=np.float64), 0.001)
        s = a.sum()
        if s > 1e-6:
            result.append(a / s)
    return result


def structured_perturb(a, rng, ptype):
    """Apply structured perturbation."""
    n = len(a)
    t = np.arange(n, dtype=np.float64) / n
    if ptype == 'shift':
        s = int(rng.integers(1, max(2, n // 12)))
        a_p = np.roll(a, s)
    elif ptype == 'cosine':
        freq = rng.uniform(0.5, 10)
        amp = rng.uniform(0.02, 0.15)
        a_p = a * (1 + amp * np.cos(2 * np.pi * freq * t + rng.uniform(0, 2 * np.pi)))
    elif ptype == 'smooth':
        k = int(rng.integers(3, max(4, n // 8)))
        kernel = np.ones(k) / k
        a_p = np.convolve(a, kernel, mode='same')
    elif ptype == 'taper':
        alpha = rng.uniform(0.4, 0.85)
        window = np.ones(n)
        edge = max(1, int(n * (1 - alpha) / 2))
        for i in range(edge):
            frac = i / edge
            window[i] = frac
            window[n - 1 - i] = frac
        a_p = a * window
    elif ptype == 'noise':
        a_p = a + rng.normal(0, rng.uniform(0.01, 0.08), n)
    else:
        a_p = a + rng.normal(0, 0.05, n)
    a_p = np.maximum(a_p, 0.001)
    s = a_p.sum()
    if s > 1e-8:
        a_p = a_p / s
    return a_p


class ScoreTracker:
    def __init__(self):
        self.best = None
        self.best_score = float('inf')
        self.t_start = time.monotonic()

    def update(self, a):
        a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
        if a.size == 0 or a.sum() < 0.01:
            return
        try:
            score = float(evaluate_sequence(a.tolist()))
        except Exception:
            return
        if np.isfinite(score) and score < self.best_score:
            self.best = a.copy()
            self.best_score = score
            elapsed = time.monotonic() - self.t_start
            print(f"  [NEW BEST] J={score:.8f} n={a.size} t={elapsed:.0f}s", flush=True)

    def report(self, label=""):
        if self.best is not None:
            print(f"  {label} best: J={self.best_score:.8f} n={self.best.size}", flush=True)


def propose_candidate(seed=42, budget_s=1500):
    """
    Hybrid optimizer for minimizing J(a) = 2*n*max(conv(a,a))/sum(a)^2.
    Uses top-k LP mirror descent, smooth max L-BFGS-B, subgradient polish,
    multi-length exploration, and structured perturbations.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + float(budget_s) - 15.0
    tracker = ScoreTracker()

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        if p.ndim == 1 and p.size > 0 and np.all(np.isfinite(p)):
            parent = np.maximum(p, 0.0)
            tracker.update(parent)
            print(f"Parent: n={parent.size}, J={tracker.best_score:.6f}", flush=True)
    except (NameError, AttributeError, Exception):
        pass

    if parent is None:
        parent = np.ones(640) / 640
        tracker.update(parent)

    best_overall = parent.copy()
    best_overall_J = compute_J(parent)

    # === Phase 1: Quick LP on parent ===
    t0 = time.monotonic()
    print(f"\n{'='*60}", flush=True)
    print(f"Phase 1: LP mirror descent on parent (n={parent.size})", flush=True)
    print(f"{'='*60}", flush=True)

    phase1_dl = min(deadline - 600, time.monotonic() + 120)
    a1, J1 = mirror_descent(parent, 500, phase1_dl, topk=60, mode='linear')
    tracker.update(a1)
    print(f"  J={J1:.8f} ({time.monotonic()-t0:.1f}s)", flush=True)
    if J1 < best_overall_J:
        best_overall = a1.copy()
        best_overall_J = J1

    # Geometric mode
    if time.monotonic() < deadline - 550:
        t0 = time.monotonic()
        a1g, J1g = mirror_descent(best_overall, 300, min(deadline - 550, time.monotonic() + 80),
                                  topk=60, mode='geometric')
        tracker.update(a1g)
        print(f"  Geometric: J={J1g:.8f} ({time.monotonic()-t0:.1f}s)", flush=True)
        if J1g < best_overall_J:
            best_overall = a1g.copy()
            best_overall_J = J1g

    # === Phase 2: Multi-length exploration ===
    if time.monotonic() < deadline - 400:
        t0 = time.monotonic()
        print(f"\n{'='*60}", flush=True)
        print(f"Phase 2: Multi-length exploration", flush=True)
        print(f"{'='*60}", flush=True)

        base_n = best_overall.size
        # Try a range of lengths
        candidate_ns = []
        for n_try in [384, 448, 512, 576, 640, 704, 768, 832, 896, 960, 1024, 1280, 1536, 2048]:
            if n_try != base_n:
                candidate_ns.append(n_try)
        # Sort by proximity to base_n for efficiency
        candidate_ns.sort(key=lambda x: abs(x - base_n))

        per_n_time = max(15, min(40, (deadline - 400 - time.monotonic()) / (len(candidate_ns) + 1) / 2))

        for n_try in candidate_ns[:10]:  # Try at most 10 different lengths
            if time.monotonic() > deadline - 200:
                break
            a_try = resample_seq(best_overall, n_try)
            a_try = np.maximum(a_try, 1e-10)
            a_try = a_try / a_try.sum()

            sub_dl = min(time.monotonic() + per_n_time, deadline - 200)
            a_try, J_try = mirror_descent(a_try, 200, sub_dl, topk=50, mode='linear')
            tracker.update(a_try)
            print(f"  n={n_try}: J={J_try:.8f} ({time.monotonic()-t0:.1f}s elapsed)", flush=True)

            if J_try < best_overall_J:
                best_overall = a_try.copy()
                best_overall_J = J_try

        print(f"Phase 2 done ({time.monotonic()-t0:.1f}s)", flush=True)

    # === Phase 3: Deep LP refinement on best ===
    if time.monotonic() < deadline - 200:
        t0 = time.monotonic()
        print(f"\n{'='*60}", flush=True)
        print(f"Phase 3: Deep LP (n={best_overall.size}, topk=100)", flush=True)
        print(f"{'='*60}", flush=True)

        a3 = best_overall.copy()
        a3 = np.maximum(a3, 1e-10)
        a3 = a3 / a3.sum()
        a3, J3 = mirror_descent(a3, 5000, min(deadline - 150, time.monotonic() + 250),
                                topk=100, mode='linear')
        tracker.update(a3)
        print(f"  J={J3:.8f} ({time.monotonic()-t0:.1f}s)", flush=True)
        if J3 < best_overall_J:
            best_overall = a3.copy()
            best_overall_J = J3

        # Geometric mode deep
        if time.monotonic() < deadline - 120:
            a3g = best_overall.copy()
            a3g = np.maximum(a3g, 1e-10)
            a3g = a3g / a3g.sum()
            a3g, J3g = mirror_descent(a3g, 2000, min(deadline - 100, time.monotonic() + 100),
                                      topk=100, mode='geometric')
            tracker.update(a3g)
            print(f"  Geometric: J={J3g:.8f}", flush=True)
            if J3g < best_overall_J:
                best_overall = a3g.copy()
                best_overall_J = J3g

    # === Phase 4: Structured perturbations + LP ===
    if time.monotonic() < deadline - 150:
        t0 = time.monotonic()
        print(f"\n{'='*60}", flush=True)
        print(f"Phase 4: Perturbations + LP (n={best_overall.size})", flush=True)
        print(f"{'='*60}", flush=True)

        perturb_types = ['shift', 'cosine', 'smooth', 'taper', 'noise',
                         'shift', 'cosine', 'smooth', 'taper', 'noise']
        per_pert = max(8, min(30, (deadline - 150 - time.monotonic()) / len(perturb_types) / 2))

        for i, ptype in enumerate(perturb_types):
            if time.monotonic() > deadline - 130:
                break
            a_pert = structured_perturb(best_overall, rng, ptype)
            sub_dl = time.monotonic() + per_pert
            a_pert, J_pert = mirror_descent(a_pert, 100, sub_dl, topk=50, mode='linear')
            tracker.update(a_pert)
            if J_pert < best_overall_J:
                best_overall = a_pert.copy()
                best_overall_J = J_pert
                print(f"  [{ptype}] J={J_pert:.8f}", flush=True)

        print(f"Phase 4 done ({time.monotonic()-t0:.1f}s)", flush=True)

    # === Phase 5: Smooth max L-BFGS-B polish ===
    if time.monotonic() < deadline - 60:
        t0 = time.monotonic()
        print(f"\n{'='*60}", flush=True)
        print(f"Phase 5: Smooth max L-BFGS-B (n={best_overall.size})", flush=True)
        print(f"{'='*60}", flush=True)

        a5 = smooth_max_lbb(best_overall, min(deadline - 40, time.monotonic() + 80),
                            T_start=0.05, T_end=0.005, n_anneal=5)
        J5 = compute_J(a5)
        tracker.update(a5)
        print(f"  J={J5:.8f} ({time.monotonic()-t0:.1f}s)", flush=True)
        if J5 < best_overall_J:
            best_overall = a5.copy()
            best_overall_J = J5

    # === Phase 6: Subgradient polish ===
    if time.monotonic() < deadline - 30:
        t0 = time.monotonic()
        print(f"\n{'='*60}", flush=True)
        print(f"Phase 6: Subgradient polish (n={best_overall.size})", flush=True)
        print(f"{'='*60}", flush=True)

        a6 = subgradient_polish(best_overall, 1000, min(deadline - 20, time.monotonic() + 40))
        J6 = compute_J(a6)
        tracker.update(a6)
        print(f"  J={J6:.8f} ({time.monotonic()-t0:.1f}s)", flush=True)
        if J6 < best_overall_J:
            best_overall = a6.copy()
            best_overall_J = J6

    # === Phase 7: Final LP refinement ===
    if time.monotonic() < deadline - 10:
        t0 = time.monotonic()
        print(f"\n{'='*60}", flush=True)
        print(f"Phase 7: Final LP (n={best_overall.size})", flush=True)
        print(f"{'='*60}", flush=True)

        a7 = best_overall.copy()
        a7 = np.maximum(a7, 1e-10)
        a7 = a7 / a7.sum()
        a7, J7 = mirror_descent(a7, 3000, deadline - 5, topk=120, mode='linear')
        tracker.update(a7)
        print(f"  J={J7:.8f} ({time.monotonic()-t0:.1f}s)", flush=True)
        if J7 < best_overall_J:
            best_overall = a7.copy()
            best_overall_J = J7

    # === Phase 8: Try a few more diverse starts if time allows ===
    if time.monotonic() < deadline - 30:
        t0 = time.monotonic()
        print(f"\n{'='*60}", flush=True)
        print(f"Phase 8: Diverse restart", flush=True)
        print(f"{'='*60}", flush=True)

        n_best = best_overall.size
        inits = generate_diverse_inits(n_best, rng, parent=best_overall)
        per_init = max(5, min(20, (deadline - 30 - time.monotonic()) / len(inits) / 2))

        for i, a_init in enumerate(inits[:6]):
            if time.monotonic() > deadline - 20:
                break
            a_init = np.maximum(a_init, 1e-10)
            a_init = a_init / a_init.sum()
            a_init, J_init = mirror_descent(a_init, 80, time.monotonic() + per_init,
                                            topk=50, mode='linear')
            tracker.update(a_init)
            if J_init < best_overall_J:
                best_overall = a_init.copy()
                best_overall_J = J_init
                print(f"  init[{i}]: J={J_init:.8f}", flush=True)

        print(f"Phase 8 done ({time.monotonic()-t0:.1f}s)", flush=True)

    # Final decision
    if tracker.best is not None:
        final_a = tracker.best
    else:
        final_a = best_overall

    print(f"\n{'='*60}", flush=True)
    print(f"FINAL: J={tracker.best_score:.8f} n={final_a.size} "
          f"time={time.monotonic()-tracker.t_start:.0f}s", flush=True)
    print(f"{'='*60}", flush=True)

    # Ensure valid output
    final_a = np.clip(final_a.astype(np.float64), 0.0, 1000.0)
    if final_a.sum() < 0.01:
        final_a = np.ones(640) / 640

    return [float(x) for x in final_a]
