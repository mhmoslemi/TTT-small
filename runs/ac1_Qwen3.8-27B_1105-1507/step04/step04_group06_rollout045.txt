def fast_conv(a):

    while s < 2 * n - 1:

    fa = np.fft.rfft(a, s)
    return np.fft.irfft(fa * fa, s)[:2 * n - 1]

def compute_J(a):

    # But in practice, we can use backtracking.

        b = fast_conv(a)
        M = float(b.max())
        
        # Find active lags (within tolerance of max)
        tol_active = max(1e-10, 0.01 * M)
        active_lags = np.where(b >= M - tol_active)[0]

            active_lags = np.array([np.argmax(b)])

        grad_M = compute_grad_active(a, active_lags)

        # J(a - alpha * grad_J) should decrease

        best_step_alpha = alpha
        best_step_J = current_J
        best_step_a = a.copy()
        
        for trial in range(20):

            a_trial = np.maximum(a_trial, 0.0)
            s_trial = a_trial.sum()
            if s_trial < 1e-10:
                best_step_alpha /= 2
                continue
            a_trial = a_trial / s_trial
            J_trial = compute_J(a_trial)
            if J_trial < current_J - 1e-12:
                best_step_J = J_trial
                best_step_a = a_trial.copy()
                break
            best_step_alpha /= 2.0

            current_J = best_step_J
            alpha = min(alpha * 1.5, 1.0)  # Increase step size if successful

"""
Minimizes J(a) = 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Active-set projected subgradient descent with momentum (fast per-iteration)
2. FFT-based fast convolution for O(n log n) cost per step
3. Adaptive step size with backtracking line search
4. Multi-length exploration and cross-pollination
5. Diverse restarts with structured perturbations (cosine, spike, smooth noise)
6. Hybrid: occasional LP steps when gradient method stalls

import numpy as np
import time
from scipy.optimize import linprog
from scipy.sparse import csr_matrix

# Cache for sparse matrix structures
_A_struct_cache = {}


def fast_conv(a):
    """Full self-convolution via FFT, O(n log n)."""
    n = len(a)
    s = 1
    while s < 2 * n - 1:
        s <<= 1
    fa = np.fft.rfft(a, s)
    return np.fft.irfft(fa * fa, s)[:2 * n - 1]


def compute_J(a):
    """Compute J(a) = 2*n*max(conv(a,a))/(sum(a))^2."""
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))


def compute_grad_active(a, active_lags):
    """
    Compute gradient of mean of (a*a)_k over active lags, w.r.t. a.
    grad[i] = (2/|S|) * sum_{k in S} a[k-i] for valid k-i in [0, n-1].
    """
    n = len(a)
    m = len(active_lags)
    if m == 0:
        return np.zeros(n)

    grad = np.zeros(n)
    for k in active_lags:
        k = int(k)
        i_lo = max(0, k - (n - 1))
        i_hi = min(n - 1, k)
        if i_lo <= i_hi:
            # grad[i] gets += a[k-i] for i in [i_lo, i_hi]
            # a[k-i] for i from i_lo to i_hi: indices k-i_lo down to k-i_hi
            grad[i_lo:i_hi + 1] += a[k - i_lo:k - i_hi - 1:-1]
    return (2.0 / m) * grad


def get_A_structure(n):
    """Get or pre-compute the (row, col) arrays for the constraint matrix."""
    if n not in _A_struct_cache:
        rows = []
        cols = []
        for k in range(2 * n - 1):
            jl = max(0, k - n + 1)
            jh = min(n - 1, k)
            for j in range(jl, jh + 1):
                rows.append(k)
                cols.append(j)
        _A_struct_cache[n] = (np.array(rows, dtype=np.int32),
                              np.array(cols, dtype=np.int32))
    return _A_struct_cache[n]


def lp_best_response(a, M):
    """LP: max sum(g) s.t. (a*g)_k <= M for all k, g >= 0. Returns normalized g or None."""
    n = len(a)
    c = -np.ones(n)
    rows_arr, cols_arr = get_A_structure(n)
    vals = a[rows_arr - cols_arr]
    A = csr_matrix((vals, (rows_arr, cols_arr)), shape=(2 * n - 1, n))
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    try:
        r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                    options={'time_limit': 5})
        if r.success and r.x is not None:
            g = np.maximum(r.x, 0.0)
            g_sum = g.sum()
            if g_sum > 1e-10:
                return g / g_sum
    except Exception:
        pass
    return None


def gradient_optimize(a, deadline, max_iters=500000, momentum_start=0.9,
                      momentum_end=0.99, verbose=False, use_lp_repair=False,
                      lp_repair_every=2000):
    """
    Active-set projected subgradient descent with momentum.
    Works on the simplex (sum(a)=1, a>=0).
    Minimizes 2n * max(conv(a,a)) equivalently.
    """
    a = np.maximum(a.astype(np.float64), 1e-12)
    a = a / a.sum()
    n = len(a)
    current_J = compute_J(a)
    best_a = a.copy()
    best_J = current_J

    momentum = momentum_start
    vel = np.zeros(n)
    alpha = 1.0  # Step size
    stall_count = 0
    it = 0

    while it < max_iters and time.monotonic() < deadline:
        # Compute gradient
        b = fast_conv(a)
        M = float(b.max())

        # Active set: lags within 1% of max (or exact argmax for tightness)
        tol_active = max(1e-12, 0.005 * M)
        active_mask = b >= M - tol_active
        active_lags = np.where(active_mask)[0]
        if len(active_lags) == 0:
            active_lags = np.array([int(np.argmax(b))])

        # Gradient of objective on simplex:
        # We minimize f(a) = 2n * M / s^2 with s = sum(a) = 1 (since we normalize)
        # So f(a) = 2n * M, and grad_f = 2n * grad_M
        # But we need to project onto the simplex tangent.
        # Projected gradient: subtract the mean to maintain sum=0.
        grad_M = compute_grad_active(a, active_lags)
        grad = grad_M - np.mean(grad_M)  # Project onto sum=0 hyperplane

        # Momentum update
        momentum = momentum_start + (momentum_end - momentum_start) * min(1.0, it / 10000)
        vel = momentum * vel + (1 - momentum) * grad
        direction = -vel  # Descent direction

        # Backtracking line search
        a_new = None
        J_new = current_J
        for trial in range(15):
            candidate = a + alpha * direction
            candidate = np.maximum(candidate, 0.0)
            s_cand = candidate.sum()
            if s_cand < 1e-10:
                alpha *= 0.5
                continue
            candidate = candidate / s_cand
            J_cand = compute_J(candidate)
            if J_cand < current_J - 1e-14:
                a_new = candidate
                J_new = J_cand
                break
            alpha *= 0.5

        if a_new is not None and J_new < current_J - 1e-14:
            a = a_new
            current_J = J_new
            stall_count = 0
            # Increase step size cautiously
            alpha = min(alpha * 1.3, 2.0 / n)
            if current_J < best_J:
                best_J = current_J
                best_a = a.copy()
                if verbose and it % 1000 == 0:
                    print(f"    it={it}, J={best_J:.8f}, alpha={alpha:.6f}, "
                          f"|S|={len(active_lags)}", flush=True)
        else:
            stall_count += 1
            alpha *= 0.7

        # LP repair when stalled
        if use_lp_repair and stall_count > lp_repair_every:
            stall_count = 0
            alpha = 1.0 / n
            g = lp_best_response(a, M)
            if g is not None:
                # Blend with LP direction
                for t in [0.3, 0.5, 0.7, 1.0]:
                    candidate = (1 - t) * a + t * g
                    candidate = np.maximum(candidate, 0.0)
                    s_c = candidate.sum()
                    if s_c < 1e-10:
                        continue
                    candidate = candidate / s_c
                    J_c = compute_J(candidate)
                    if J_c < current_J:
                        a = candidate
                        current_J = J_c
                        if J_c < best_J:
                            best_J = J_c
                            best_a = a.copy()
                        break

        it += 1

    return best_a, best_J


def resample_seq(src, dest_n):
    """Resample 1D sequence to new length via linear interpolation."""
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    result = np.interp(t_new, t_old, src)
    return np.maximum(result, 0.0)


def generate_initializations(n, rng, parent=None):
    """Generate diverse initial sequences of length n."""
    inits = []
    t = np.linspace(0, 1, n, endpoint=False)

    if parent is not None and parent.size > 0:
        p = resample_seq(parent, n)
        p = p / max(p.sum(), 1e-10)
        inits.append(p.copy())
        inits.append(np.maximum(p + rng.normal(0, 0.05, n) * p, 0.01) / max(
            np.maximum(p + rng.normal(0, 0.05, n) * p, 0.01).sum(), 1e-10))

    # Uniform
    inits.append(np.ones(n) / n)

    # Cosine families with various frequencies
    for freq in [1, 2, 3, 4, 5, 7, 8]:
        c = 1.0 + 0.3 * np.cos(2.0 * np.pi * freq * t)
        inits.append(c / c.sum())

    # Two-tone cosine
    for f1, f2 in [(2, 5), (3, 7), (1, 4)]:
        c = 1.0 + 0.25 * np.cos(2 * np.pi * f1 * t) + 0.15 * np.cos(2 * np.pi * f2 * t)
        inits.append(np.maximum(c, 0.1) / np.maximum(c, 0.1).sum())

    # Gaussian bump
    g = np.exp(-6.0 * (t - 0.5) ** 2) + 0.1
    inits.append(g / g.sum())

    # Sine absolute
    s = np.abs(np.sin(np.pi * t)) + 0.2
    inits.append(s / s.sum())

    # Random smooth
    r = rng.uniform(0.5, 1.0, n)
    kernel = np.ones(7) / 7
    r = np.convolve(r, kernel, mode='same')
    inits.append(r / r.sum())

    # Triangular
    tri = 1.0 - 0.4 * np.abs(2.0 * (t % 0.5) - 0.5)
    inits.append(tri / tri.sum())

    # Power law
    p = (t + 0.2) ** 0.7 + 0.3
    inits.append(p / p.sum())

    return inits

def perturb_sequence(a, rng, pert_type, param):
    """Apply a perturbation to sequence a."""
    n = len(a)
    a_p = a.copy().astype(np.float64)

    if pert_type == 'gaussian':
        a_p = a_p + rng.normal(0, param, n) * a_p
    elif pert_type == 'uniform':
        a_p = a_p + rng.uniform(-param, param, n) * a_p
    elif pert_type == 'spike':
        k = min(max(2, param), n // 4)
        idx = rng.choice(n, size=k, replace=False)
        a_p[idx] *= rng.uniform(0.3, 3.0, k)
    elif pert_type == 'smooth_noise':
        noise = rng.normal(0, param, n)
        kernel_size = max(3, n // 64)
        kernel = np.ones(kernel_size) / kernel_size
        noise = np.convolve(noise, kernel, mode='same')
        a_p = a_p + noise * a_p
    elif pert_type == 'block_swap':
        block_size = max(4, n // 32)
        b1 = rng.integers(0, n - 2 * block_size)
        b2 = rng.integers(0, n - 2 * block_size)
        if abs(b1 - b2) > block_size:
            a_p[b1:b1 + block_size], a_p[b2:b2 + block_size] = \
                a_p[b2:b2 + block_size].copy(), a_p[b1:b1 + block_size].copy()
    elif pert_type == 'mirror':
        # Mirror a random portion
        block_size = max(8, n // 8)
        start = rng.integers(0, n - block_size)
        a_p[start:start + block_size] = a_p[start + block_size - 1:start - 1:-1]

    a_p = np.maximum(a_p, 1e-6)
    a_p = a_p / a_p.sum()
    return a_p

            candidates = sorted(set(c for c in candidates if 64 <= c <= 2048
                                    and abs(c - n_best) >= 16))

            for n_try in candidates[:12]:
                if time.monotonic() > deadline:
                    break
                a_res = resample_seq(tracker.best, n_try)
                a_res = a_res / max(a_res.sum(), 1e-10)
                budget_here = min(30.0, deadline - time.monotonic())
                if budget_here < 5:
                    break
                a_opt, J_opt = gradient_optimize(
                    a_res, time.monotonic() + budget_here,
                    max_iters=100000,
                    momentum_start=0.9, momentum_end=0.99,
                    use_lp_repair=False
                )
                tracker.update(a_opt)

            tracker.report(f"Phase 5: {time.monotonic()-t0:.1f}s")

    # Final: one more polish with all remaining time
    if tracker.best is not None and time.monotonic() < deadline:
        remaining = deadline - time.monotonic()
        if remaining > 10:
            print(f"\n  Final polish with {remaining:.0f}s remaining...", flush=True)
            a = tracker.best.copy().astype(np.float64)
            a = a / a.sum()
            a_opt, J_opt = gradient_optimize(
                a, deadline,
                max_iters=10000000,
                momentum_start=0.98, momentum_end=0.999,
                use_lp_repair=True,
                lp_repair_every=5000
            )
            tracker.update(a_opt)

    # Fallback
    if tracker.best is None:
        tracker.best = np.ones(512)
        tracker.best_score = compute_J(tracker.best)
        print(f"  Using fallback: J={tracker.best_score:.8f}", flush=True)

    print(f"\n{'=' * 60}", flush=True)
    print(f"FINAL RESULT: J={tracker.best_score:.8f} n={tracker.best.size} "
          f"total_time={time.monotonic()-tracker.t_start:.1f}s", flush=True)
    print(f"{'=' * 60}", flush=True)

    # Ensure valid output
    result = np.clip(tracker.best, 0.0, 1000.0)
    if result.sum() < 0.01:
        result = np.ones(128)

    return [float(x) for x in result]
