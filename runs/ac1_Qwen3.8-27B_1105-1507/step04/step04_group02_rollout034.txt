"""
Optimized step-function coefficient search minimizing 2*n*max(conv(a,a))/(sum(a))^2.
Key improvements over previous approach:
1. FFT-based O(n log n) gradient computation for soft-max subgradient
2. Nesterov-accelerated momentum with backtracking line search
3. Adaptive temperature annealing for soft-max
4. Coordinate descent polishing for fine-tuning
5. Alternating LP with dual warm-starts
6. Non-symmetric perturbation strategies
7. Multi-n interpolation to exploit the 2n scaling factor
"""

import numpy as np
from scipy.optimize import linprog
from scipy.interpolate import interp1d
import time
from scipy.sparse import coo_matrix


def compute_conv(a):
    """Compute linear convolution a*a via FFT."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    conv = np.fft.irfft(fa * fa, size)
    return conv[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    S = float(np.sum(a))
    if S < 1e-10:
        return np.inf
    conv = compute_conv(a)
    M = float(np.max(conv))
    return 2.0 * n * M / (S * S)


def compute_softmax_grad(a, temperature):
    """
    Compute gradient of soft-max approximation to max(conv(a,a)).
    Uses FFT for O(n log n) computation.
    Returns (gradient, max_conv, conv).
    """
    n = len(a)
    conv = compute_conv(a)
    M = float(np.max(conv))
    
    # Compute soft-max weights with temperature
    conv_scaled = conv / temperature
    conv_scaled -= np.max(conv_scaled)  # numerical stability
    p = np.exp(conv_scaled)
    p_sum = np.sum(p)
    if p_sum < 1e-300:
        p = np.zeros_like(conv)
        p[np.argmax(conv)] = 1.0
    else:
        p /= p_sum
    
    # Compute gradient: grad[j] = sum_k p[k]*(a[k-j] + a[j]) * I(0<=k-j<n)
    # = first_part[j] + a[j] * window_sum[j]
    # where first_part[j] = sum_m p[m+j]*a[m] (cross-correlation)
    # and window_sum[j] = sum_{k=j}^{j+n-1} p[k]
    
    L = 1
    while L < 3 * n:
        L <<= 1
    
    # Cross-correlation: first_part[j] = sum_m p[m] * a[m+j] for valid m
    # Using: ifft(conj(fft(a)) * fft(p))[:n]
    A = np.fft.rfft(a, L)
    P = np.fft.rfft(p, L)
    first_part = np.fft.irfft(np.conj(A) * P, L)[:n]
    
    # Window sum: sliding sum of p of width n
    cum_p = np.concatenate([[0.0], np.cumsum(p)])
    window_sum = cum_p[n:n + n] - cum_p[:n]
    
    grad = first_part + a * window_sum
    return grad, M, conv


def nesterov_softmax(a, time_budget, T_start=1.0, T_end=0.01, max_iters=5000):
    """Nesterov-accelerated soft-max subgradient descent with backtracking line search."""
    n = len(a)
    a = a.copy().astype(np.float64)
    S = np.sum(a)
    if S < 1e-10:
        return a
    a = a / S
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    
    momentum = 0.9
    step = 0.1
    T = T_start
    stall = 0
    v = np.zeros(n)
    
    for it in range(max_iters):
        if time.monotonic() - start > time_budget:
            break
        
        # Anneal temperature
        progress = min(1.0, it / max_iters)
        T = T_start * (T_end / T_start) ** progress
        
        # Nesterov lookahead
        y = a + momentum * v
        
        # Compute gradient at y
        grad_y, M_y, _ = compute_softmax_grad(y, T)
        # Center gradient (preserve sum)
        grad_y = grad_y - np.mean(grad_y)
        
        # Update velocity
        v = momentum * v + grad_y
        
        # Candidate point
        a_new = a - step * v
        a_new = np.maximum(a_new, 0)
        S_new = np.sum(a_new)
        if S_new < 1e-10:
            step *= 0.5
            continue
        a_new = a_new / S_new
        
        # Backtracking line search on true objective
        obj_new = compute_obj(a_new)
        if obj_new < best_obj - 1e-12:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new
            stall = 0
            # Increase step slightly
            step = min(step * 1.05, 0.5)
        else:
            # Try smaller step
            for shrink in [0.5, 0.25, 0.1]:
                a_try = a - step * shrink * v
                a_try = np.maximum(a_try, 0)
                S_try = np.sum(a_try)
                if S_try < 1e-10:
                    continue
                a_try = a_try / S_try
                obj_try = compute_obj(a_try)
                if obj_try < best_obj - 1e-12:
                    best_obj = obj_try
                    best_a = a_try.copy()
                    a = a_try
                    step *= shrink * 0.8
                    stall = 0
                    break
            else:
                stall += 1
                step *= 0.7
                if stall > 200:
                    # Perturb to escape
                    rng = np.random.default_rng(it * 31 + 7)
                    a = best_a * np.exp(rng.normal(0, 0.05, n))
                    a = np.maximum(a, 0)
                    a = a / np.sum(a)
                    v = np.zeros(n)
                    step = 0.1
                    stall = 0
                    if step < 1e-10:
                        break
    
    return best_a


def coordinate_descent(a, time_budget, max_sweeps=100):
    """Cycle through coordinates, optimizing each one at a time."""
    n = len(a)
    a = a.copy().astype(np.float64)
    S = np.sum(a)
    if S < 1e-10:
        return a
    a = a / S
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    stall = 0
    rng = np.random.default_rng(12345)
    
    for sweep in range(max_sweeps):
        if time.monotonic() - start > time_budget:
            break
        
        # Randomize order
        order = rng.permutation(n)
        improved = False
        
        for j in order:
            if time.monotonic() - start > time_budget:
                break
            
            a_old_j = a[j]
            if a_old_j < 1e-15:
                continue
            
            # Try several values for a[j]
            # Optimal: minimize max_k C[k] where C[k] = C0[k] + a[j]*(a[k-j] + a[j]*I(k-j!=j)) * I(valid)
            # We'll use a simple perturbation + local search
            candidates = [
                a_old_j * 0.5,
                a_old_j * 0.75,
                a_old_j * 0.9,
                a_old_j,
                a_old_j * 1.1,
                a_old_j * 1.25,
                a_old_j * 1.5,
                0.0,
            ]
            
            best_val = a_old_j
            best_c_obj = best_obj
            
            # Compute current conv for reference
            conv = compute_conv(a)
            M = float(np.max(conv))
            
            for candidate in candidates:
                if candidate < 0:
                    continue
                a[j] = candidate
                S_new = S - a_old_j + candidate
                if S_new < 1e-10:
                    a[j] = a_old_j
                    continue
                # Update conv efficiently
                # C[k] changes only for k where j or k-j is involved
                # C[k]_new = C[k]_old - a_old_j*(a[k-j] + a_old_j*I(2j==k)) * I(valid) 
                #                            + candidate*(a[k-j] + candidate*I(2j==k)) * I(valid)
                # Simpler: just recompute the affected entries
                for k in range(max(0, j - n + 1), min(2 * n - 1, j + n - 1) + 1):
                    if k < 0 or k >= 2 * n - 1:
                        continue
                    # Old contribution of a[j] to C[k]
                    old_contrib = 0.0
                    new_contrib = 0.0
                    kj = k - j
                    if 0 <= kj < n:
                        if kj == j:
                            old_contrib = a_old_j * a_old_j
                            new_contrib = candidate * candidate
                        else:
                            old_contrib = a_old_j * a[kj] + (a_old_j * a[kj] if k - (n - 1 - (n-1-j)) == j else 0)
                            # Actually: C[k] includes a[j]*a[k-j] and a[k-j]*a[j] (same if symmetric)
                            # C[k] = sum_i a[i]*a[k-i], so a[j]*a[k-j] appears once (when i=j)
                            # and a[k-j]*a[j] appears once (when i=k-j) if k-j != j
                            # So total contribution of a[j] to C[k] is:
                            # a[j]*a[k-j] + a[k-j]*a[j] = 2*a[j]*a[k-j] if k-j != j
                            # a[j]^2 if k-j == j
                            if kj != j:
                                old_contrib = 2.0 * a_old_j * a[kj]
                                new_contrib = 2.0 * candidate * a[kj]
                            else:
                                old_contrib = a_old_j * a_old_j
                                new_contrib = candidate * candidate
                    
                    conv[k] -= old_contrib
                    conv[k] += new_contrib
                
                M_new = float(np.max(conv))
                S2_new = S_new * S_new
                obj_candidate = 2.0 * n * M_new / S2_new
                
                if obj_candidate < best_c_obj - 1e-12:
                    best_c_obj = obj_candidate
                    best_val = candidate
                else:
                    # Undo
                    for k in range(max(0, j - n + 1), min(2 * n - 1, j + n - 1) + 1):
                        if k < 0 or k >= 2 * n - 1:
                            continue
                        kj = k - j
                        if 0 <= kj < n:
                            if kj == j:
                                conv[k] -= candidate * candidate
                                conv[k] += a_old_j * a_old_j
                            else:
                                conv[k] -= 2.0 * candidate * a[kj]
                                conv[k] += 2.0 * a_old_j * a[kj]
            
            # Apply best
            a[j] = best_val
            S = S - a_old_j + best_val
            a_norm = a / S
            
            if best_val != a_old_j:
                improved = True
                if best_c_obj < best_obj - 1e-12:
                    best_obj = best_c_obj
                    best_a = a_norm.copy()
            
            a = a_norm
        
        if not improved:
            stall += 1
            if stall > 5:
                break
        else:
            stall = 0
    
    return best_a


def build_toeplitz_sparse(a):
    """Build sparse matrix A such that A @ g = conv(a, g)."""
    n = len(a)
    rows = []
    cols = []
    vals = []
    for j in range(n):
        start = j
        end = min(j + n, 2 * n - 1)
        length = end - start
        rows.extend(range(start, end))
        cols.extend([j] * length)
        vals.extend(list(a[:length]))
    A = coo_matrix((vals, (rows, cols)), shape=(2 * n - 1, n)).tocsr()
    return A


def solve_lp(a, M, time_limit=10):
    """Solve LP: max sum(g) s.t. conv(a,g) <= M, g >= 0."""
    n = len(a)
    A = build_toeplitz_sparse(a)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': time_limit,
                               'dual_feasibility_tolerance': 1e-8})
        if res.status == 0:
            return np.maximum(res.x, 0)
        elif res.status == 3:  # timeout
            if res.x is not None:
                return np.maximum(res.x, 0)
    except Exception:
        pass
    return None


def alternating_lp(a, time_budget, max_iters=60):
    """Alternating LP with adaptive M and line search."""
    n = len(a)
    a = a.copy().astype(np.float64)
    S = np.sum(a)
    if S < 1e-10:
        return a
    a = a / S
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    stall = 0
    
    for it in range(max_iters):
        if time.monotonic() - start > time_budget:
            break
        
        conv = compute_conv(a)
        M = float(np.max(conv))
        
        # Solve LP to find optimal g given current a and M
        g = solve_lp(a, M, time_limit=min(8.0, (time_budget - (time.monotonic() - start)) * 0.3))
        if g is None:
            break
        
        S_g = np.sum(g)
        if S_g < 1e-10:
            break
        g = g / S_g
        
        # Line search
        best_t = 0.0
        best_obj_ls = best_obj
        for t in [0.3, 0.5, 0.7, 0.9, 1.0, 0.1, 0.2]:
            a_try = (1.0 - t) * a + t * g
            a_try = np.maximum(a_try, 0)
            S_try = np.sum(a_try)
            if S_try < 1e-10:
                continue
            a_try = a_try / S_try
            obj_try = compute_obj(a_try)
            if obj_try < best_obj_ls - 1e-12:
                best_obj_ls = obj_try
                best_t = t
        
        if best_t > 0.0:
            a = (1.0 - best_t) * a + best_t * g
            a = np.maximum(a, 0)
            a = a / np.sum(a)
            best_obj = best_obj_ls
            best_a = a.copy()
            stall = 0
        else:
            stall += 1
            if stall > 10:
                # Perturb
                rng = np.random.default_rng(it * 997 + 42)
                a = a * np.exp(rng.normal(0, 0.06, n))
                a = np.maximum(a, 0)
                a = a / np.sum(a)
                stall = 0
            else:
                # Small step toward g
                a = 0.95 * a + 0.05 * g
                a = np.maximum(a, 0)
                a = a / np.sum(a)
    
    return best_a


def nesterov_with_perturb(a, time_budget, seed=0):
    """Nesterov with periodic perturbations and adaptive parameters."""
    n = len(a)
    a = a.copy().astype(np.float64)
    S = np.sum(a)
    if S < 1e-10:
        return a
    a = a / S
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    rng = np.random.default_rng(seed)
    
    # Adaptive parameters
    momentum = 0.85
    step = 0.15
    T = 0.5
    pert_count = 0
    
    for it in range(10000):
        if time.monotonic() - start > time_budget:
            break
        
        # Anneal temperature
        progress = min(1.0, it / 2000.0)
        T = max(0.01, 1.0 * (0.01) ** progress)
        
        # Compute gradient
        grad, M, _ = compute_softmax_grad(a, T)
        grad = grad - np.mean(grad)
        
        # Nesterov update (simplified: just use momentum)
        a_old = a.copy()
        a_new = a - step * (momentum * grad + (1 - momentum) * grad)
        # Actually let's use proper momentum:
        # v = momentum * v + grad
        # a = a - step * v
        # We'll approximate with a simple version:
        a_new = a - step * grad
        a_new = np.maximum(a_new, 0)
        S_new = np.sum(a_new)
        if S_new < 1e-10:
            step *= 0.5
            continue
        a_new = a_new / S_new
        
        obj_new = compute_obj(a_new)
        if obj_new < best_obj - 1e-12:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new
            step = min(step * 1.02, 0.4)
        else:
            step *= 0.85
            if step < 1e-6:
                # Perturb
                pert_scale = rng.uniform(0.03, 0.15)
                a = best_a * np.exp(rng.normal(0, pert_scale, n))
                a = np.maximum(a, 0)
                a = a / np.sum(a)
                step = 0.1
                pert_count += 1
                if pert_count > 30:
                    break
    
    return best_a


def cosine_init(n, n_modes=8, seed=0):
    """Generate a smooth cosine-series initialization."""
    rng = np.random.default_rng(seed)
    k = np.arange(n)
    a = np.ones(n)
    for m in range(1, n_modes + 1):
        c = rng.uniform(-0.5, 0.5) / m
        phase = rng.uniform(0, 2 * np.pi)
        a += c * np.cos(2 * np.pi * m * k / n + phase)
    a = np.maximum(a, 0)
    S = np.sum(a)
    if S < 1e-10:
        return np.ones(n) / n
    return a / S


def ramp_init(n, seed=0):
    """Non-symmetric ramp initialization."""
    rng = np.random.default_rng(seed)
    k = np.arange(n)
    alpha = rng.uniform(0.5, 3.0)
    a = np.exp(-alpha * k / n) + 0.3 * np.exp(-alpha * (n - 1 - k) / n)
    a = np.maximum(a, 1e-10)
    S = np.sum(a)
    return a / S


def two_bump_init(n, seed=0):
    """Two bump initialization with optional gap."""
    rng = np.random.default_rng(seed)
    k = np.arange(n)
    c1 = rng.uniform(0.15, 0.35) * n
    c2 = rng.uniform(0.65, 0.85) * n
    s = rng.uniform(0.05, 0.15) * n
    a = np.exp(-0.5 * ((k - c1) / s) ** 2) + np.exp(-0.5 * ((k - c2) / s) ** 2)
    a = np.maximum(a, 1e-10)
    S = np.sum(a)
    return a / S


def interpolate_sequence(a, new_n):
    """Interpolate sequence to a new length."""
    old_n = len(a)
    x_old = np.linspace(0, 1, old_n)
    x_new = np.linspace(0, 1, new_n)
    try:
        f = interp1d(x_old, a, kind='cubic', fill_value='extrapolate', bounds_error=False)
        a_new = np.maximum(f(x_new), 0)
    except Exception:
        a_new = np.ones(new_n) / new_n
    S = np.sum(a_new)
    if S < 1e-10:
        return np.ones(new_n) / new_n
    return a_new / S


def refine_pipeline(a, time_budget, seed=0):
    """Full refinement: Nesterov -> coordinate descent -> LP -> coordinate descent."""
    n = len(a)
    a = a.copy().astype(np.float64)
    S = np.sum(a)
    if S < 1e-10:
        return a, np.inf
    a = a / S
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    
    # Phase A: Nesterov soft-max gradient (40% of budget)
    remaining = time_budget - (time.monotonic() - start)
    if remaining > 5:
        t_a = remaining * 0.4
        a_new = nesterov_softmax(a, time_budget=t_a, T_start=0.5, T_end=0.01)
        obj_new = compute_obj(a_new)
        if obj_new < best_obj:
            best_obj = obj_new
            best_a = a_new.copy()
            print(f"    Nesterov: {best_obj:.6f}")
    
    # Phase B: Nesterov with perturbation (25% of budget)
    remaining = time_budget - (time.monotonic() - start)
    if remaining > 5:
        t_b = remaining * 0.25
        a_new = nesterov_with_perturb(best_a, time_budget=t_b, seed=seed)
        obj_new = compute_obj(a_new)
        if obj_new < best_obj:
            best_obj = obj_new
            best_a = a_new.copy()
            print(f"    Nesterov+Pert: {best_obj:.6f}")
    
    # Phase C: Coordinate descent (20% of budget)
    remaining = time_budget - (time.monotonic() - start)
    if remaining > 5:
        t_c = remaining * 0.2
        a_new = coordinate_descent(best_a, time_budget=t_c)
        obj_new = compute_obj(a_new)
        if obj_new < best_obj:
            best_obj = obj_new
            best_a = a_new.copy()
            print(f"    CoordDesc: {best_obj:.6f}")
    
    # Phase D: LP (15% of budget)
    remaining = time_budget - (time.monotonic() - start)
    if remaining > 5 and n <= 600:
        t_d = remaining * 0.15
        a_new = alternating_lp(best_a, time_budget=t_d, max_iters=30)
        obj_new = compute_obj(a_new)
        if obj_new < best_obj:
            best_obj = obj_new
            best_a = a_new.copy()
            print(f"    LP: {best_obj:.6f}")
    
    return best_a, best_obj


def try_parent():
    """Try to load parent sequence from global variable."""
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64)
        parent = np.clip(parent, 0, 1000.0)
        if parent.size > 0 and np.isfinite(parent).all():
            obj = compute_obj(parent)
            print(f"Parent (n={parent.size}): obj={obj:.6f}")
            return parent.copy(), obj
    except (NameError, Exception):
        pass
    return None, np.inf


def propose_candidate(seed=42, budget_s=1500):
    """
    Main search function combining Nesterov soft-max gradient, coordinate descent,
    alternating LP, and multi-n interpolation to minimize 2*n*max(conv(a,a))/S^2.
    """
    start_time = time.monotonic()
    deadline = start_time + float(budget_s) - 30.0
    
    best_seq = None
    best_score = np.inf
    
    # Load parent
    parent, parent_obj = try_parent()
    if parent is not None:
        best_score = parent_obj
        best_seq = parent.copy()
        n_parent = len(parent)
        print(f"Starting from parent: n={n_parent}, score={best_score:.6f}")
    else:
        n_parent = 500
        best_seq = cosine_init(n_parent, seed=seed)
        best_score = compute_obj(best_seq)
        print(f"No parent found. Initialized cosine: n={n_parent}, score={best_score:.6f}")
    
    # === Phase 1: Try nearby n values by interpolation ===
    print("\n=== Phase 1: Multi-n exploration ===")
    n_base = len(best_seq)
    
    # Try reducing n slightly (exploits 2n scaling)
    n_tries = sorted(set([
        max(100, n_base - 4), max(100, n_base - 2),
        n_base,
        min(1000, n_base + 2), min(1000, n_base + 4),
    ]))
    
    for n_try in n_tries:
        if time.monotonic() > deadline - 120:
            break
        if n_try == n_base:
            continue
        
        a_interp = interpolate_sequence(best_seq, n_try)
        # Quick Nesterov refinement
        remaining = deadline - time.monotonic()
        alloc = min(remaining * 0.08, 30.0)
        a_new = nesterov_softmax(a_interp, time_budget=alloc, T_start=0.3, T_end=0.01)
        obj_new = compute_obj(a_new)
        if obj_new < best_score:
            best_score = obj_new
            best_seq = a_new.copy()
            print(f"  n={n_try}: {best_score:.6f} ** NEW BEST **")
        else:
            print(f"  n={n_try}: {obj_new:.6f}")
    
    # === Phase 2: Diverse initializations + quick optimization ===
    print("\n=== Phase 2: Diverse initializations ===")
    n_current = len(best_seq)
    
    inits = [
        ("nesterov_parent", best_seq.copy()),
        ("cosine", cosine_init(n_current, n_modes=10, seed=seed)),
        ("ramp", ramp_init(n_current, seed=seed + 1)),
        ("two_bump", two_bump_init(n_current, seed=seed + 2)),
        ("asym_ramp", ramp_init(n_current, seed=seed + 3)),
        ("two_bump2", two_bump_init(n_current, seed=seed + 4)),
    ]
    
    for name, a_init in inits:
        if time.monotonic() > deadline - 90:
            break
        obj_init = compute_obj(a_init)
        if obj_init > best_score + 0.3:
            continue  # Not competitive
        
        # Quick Nesterov
        remaining = deadline - time.monotonic()
        alloc = min(remaining * 0.06, 20.0)
        a_new = nesterov_softmax(a_init, time_budget=alloc, T_start=0.4, T_end=0.02)
        obj_new = compute_obj(a_new)
        if obj_new < best_score:
            best_score = obj_new
            best_seq = a_new.copy()
            print(f"  {name}: {best_score:.6f} ** NEW BEST **")
        else:
            print(f"  {name}: {obj_new:.6f}")
    
    # === Phase 3: Intensive refinement of best ===
    print(f"\n=== Phase 3: Intensive refinement (best={best_score:.6f}, n={len(best_seq)}) ===")
    
    n_rounds = 6
    for r in range(n_rounds):
        if time.monotonic() > deadline - 60:
            break
        remaining = deadline - time.monotonic()
        alloc = min(remaining * 0.2, 80.0)
        
        print(f"  Round {r+1}/{n_rounds}, alloc={alloc:.0f}s")
        a_new, obj_new = refine_pipeline(best_seq, time_budget=alloc, seed=seed + r * 100)
        
        if obj_new < best_score:
            best_score = obj_new
            best_seq = a_new.copy()
            print(f"    ** NEW BEST: {best_score:.6f} **")
        else:
            print(f"    No improvement ({obj_new:.6f})")
            
            # Try perturbation
            if time.monotonic() < deadline - 30:
                rng = np.random.default_rng(seed + r * 777 + 33)
                for pert_scale in [0.05, 0.1, 0.2]:
                    if time.monotonic() > deadline - 20:
                        break
                    a_pert = best_seq * np.exp(rng.normal(0, pert_scale, len(best_seq)))
                    a_pert = np.maximum(a_pert, 0)
                    a_pert = a_pert / np.sum(a_pert)
                    alloc_p = min(20.0, (deadline - time.monotonic()) * 0.15)
                    if alloc_p < 5:
                        break
                    a_p_new, obj_p = refine_pipeline(a_pert, time_budget=alloc_p, seed=seed + r * 100 + pert_scale * 1000)
                    if obj_p < best_score:
                        best_score = obj_p
                        best_seq = a_p_new.copy()
                        print(f"    Pert({pert_scale}): {best_score:.6f} ** NEW BEST **")
                        break
    
    # === Phase 4: Coordinate descent + LP polish ===
    print(f"\n=== Phase 4: Final polish (best={best_score:.6f}) ===")
    
    if time.monotonic() < deadline - 20:
        remaining = deadline - time.monotonic()
        
        # Extended coordinate descent
        cd_alloc = min(remaining * 0.3, 60.0)
        if cd_alloc > 5:

            obj_cd = compute_obj(a_cd)
            if obj_cd < best_score:
                best_score = obj_cd
                best_seq = a_cd.copy()
                print(f"  CoordDesc final: {best_score:.6f} ** NEW BEST **")
            else:
                print(f"  CoordDesc final: {obj_cd:.6f}")
        
        # LP polish
        remaining = deadline - time.monotonic()
        if remaining > 15 and len(best_seq) <= 600:
            lp_alloc = min(remaining * 0.3, 50.0)
            a_lp = alternating_lp(best_seq, time_budget=lp_alloc, max_iters=40)
            obj_lp = compute_obj(a_lp)
            if obj_lp < best_score:
                best_score = obj_lp
                best_seq = a_lp.copy()
                print(f"  LP final: {best_score:.6f} ** NEW BEST **")
            else:
                print(f"  LP final: {obj_lp:.6f}")
        
        # One more Nesterov pass
        remaining = deadline - time.monotonic()
        if remaining > 10:
            ng_alloc = min(remaining * 0.3, 40.0)
            a_ng = nesterov_softmax(best_seq, time_budget=ng_alloc, T_start=0.1, T_end=0.005)
            obj_ng = compute_obj(a_ng)
            if obj_ng < best_score:
                best_score = obj_ng
                best_seq = a_ng.copy()
                print(f"  Nesterov final: {best_score:.6f} ** NEW BEST **")
            else:
                print(f"  Nesterov final: {obj_ng:.6f}")
        
        # Final coordinate descent
        remaining = deadline - time.monotonic()
        if remaining > 8:
            a_cd2 = coordinate_descent(best_seq, time_budget=min(remaining * 0.5, 30.0), max_sweeps=30)
            obj_cd2 = compute_obj(a_cd2)
            if obj_cd2 < best_score:
                best_score = obj_cd2
                best_seq = a_cd2.copy()
                print(f"  CoordDesc2: {best_score:.6f} ** NEW BEST **")
    
    # === Phase 5: Try slightly different n ===
    print(f"\n=== Phase 5: Final n adjustment (best={best_score:.6f}, n={len(best_seq)}) ===")
    n_best = len(best_seq)
    
    if time.monotonic() < deadline - 15:
        for n_delta in [-3, -2, -1, 1, 2, 3]:
            n_try = n_best + n_delta
            if n_try < 100 or n_try > 1000:
                continue
            if time.monotonic() > deadline - 10:
                break
            
            a_interp = interpolate_sequence(best_seq, n_try)
            # Very quick refinement
            alloc = min(10.0, (deadline - time.monotonic()) * 0.3)
            if alloc < 3:
                break
            a_new = nesterov_softmax(a_interp, time_budget=alloc, T_start=0.1, T_end=0.005)
            obj_new = compute_obj(a_new)
            if obj_new < best_score:
                best_score = obj_new
                best_seq = a_new.copy()
                print(f"  n={n_try}: {best_score:.6f} ** NEW BEST **")
            else:
                print(f"  n={n_try}: {obj_new:.6f}")
    
    # Final validation
    if best_seq is None:
        best_seq = np.ones(300) / 300
        best_score = compute_obj(best_seq)
    
    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    val = evaluate_sequence(result)
    if val < np.inf and val < best_score:
        best_score = val
    elif val == np.inf:
        result = [1.0] * 300
        best_score = 2.0
    
    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_score:.6f}, time={elapsed:.1f}s ===")
    
    return result
