def evaluate_sequence(sequence: list[float]) -> float:
    """
    Evaluates a sequence of coefficients with enhanced security checks.
    Returns np.inf if the input is invalid.
    """
    # --- Security Checks ---

    # Verify that the input is a list
    if not isinstance(sequence, list):
        return np.inf

    # Reject empty lists
    if not sequence:
        return np.inf

    # Check each element in the list for validity
    for x in sequence:
        # Reject boolean types (as they are a subclass of int) and
        # any other non-integer/non-float types (like strings or complex numbers).
        if isinstance(x, bool) or not isinstance(x, (int, float)):
            return np.inf

        # Reject Not-a-Number (NaN) and infinity values.
        if np.isnan(x) or np.isinf(x):
            return np.inf

    # Convert all elements to float for consistency
    sequence = [float(x) for x in sequence]

    # Protect against negative numbers
    sequence = [max(0, x) for x in sequence]

    # Protect against numbers that are too large
    sequence = [min(1000.0, x) for x in sequence]

    n = len(sequence)
    b_sequence = np.convolve(sequence, sequence)
    max_b = max(b_sequence)
    sum_a = np.sum(sequence)

    # Protect against the case where the sum is too close to zero
    if sum_a < 0.01:
        return np.inf

    return float(2 * n * max_b / (sum_a**2))

def compute_conv(a):
    """Compute linear convolution a*a via FFT."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    conv = np.fft.irfft(fa * fa, size)
    return conv[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""

def build_toeplitz_sparse(a):
    """Build sparse Toeplitz matrix for conv(a, g)."""
    n = len(a)
    rows = []
    cols = []
    vals = []
    for j in range(n):
        start = j
        end = min(j + n, 2 * n - 1)
        length = end - start
        rows.extend(range(start, end))
        cols.extend([j] * length)
        vals.extend(list(a[:length]))
    from scipy.sparse import coo_matrix
    A = coo_matrix((vals, (rows, cols)), shape=(2 * n - 1, n)).tocsr()
    return A


def solve_lp_maxsum(a, M, time_limit=8):
    """Solve LP: max sum(g) s.t. conv(a,g) <= M, g >= 0."""

        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': time_limit})

    except Exception:
        pass
    return None


def compute_subgradient(a, topk=10):
    """Compute subgradient of max(conv(a,a)) using top-K active constraints."""
    n = len(a)
    conv = compute_conv(a)
    M = float(np.max(conv))
    
    # Find top-K active positions
    if topk >= len(conv):
        active = np.arange(len(conv))
    else:
        active = np.argsort(conv)[-topk:]
    
    # Gradient: for active position k, d(conv[k])/da[i] = 2*a[k-i] if 0<=k-i<n
    grad = np.zeros(n)
    for k in active:
        for i in range(n):
            j = k - i
            if 0 <= j < n:
                grad[i] += 2.0 * a[j]
    
    # Normalize by number of active constraints
    grad /= len(active)
    return grad, M, conv

def cosine_init(n, n_modes=8, seed=0):
    """Generate a smooth cosine-series initialization."""
    rng = np.random.default_rng(seed)
    k = np.arange(n)
    a = np.ones(n)
    for m in range(1, n_modes + 1):
        c = rng.uniform(-0.6, 0.6) / m
        phase = rng.uniform(0, 2 * np.pi)
        a += c * np.cos(2 * np.pi * m * k / n + phase)
    a = np.maximum(a, 0)
    S = np.sum(a)
    if S < 1e-10:
        return np.ones(n) / n
    return a / S


def raised_cosine_init(n, alpha=1.0):
    """Raised cosine shape (concentrated in middle)."""
    k = np.arange(n)
    x = 2.0 * k / (n - 1) - 1.0
    a = 0.5 * (1.0 + np.cos(np.pi * x)) ** alpha
    a = np.maximum(a, 1e-10)
    S = np.sum(a)
    return a / S


def gaussian_init(n, sigma_frac=0.3):
    """Gaussian bump initialization."""
    k = np.arange(n)
    center = n / 2.0
    sigma = sigma_frac * n
    a = np.exp(-0.5 * ((k - center) / sigma) ** 2)
    S = np.sum(a)
    return a / S


def parametric_optimize(n, n_modes=6, time_budget=15.0, seed=0):
    """Optimize cosine coefficients via Nelder-Mead."""
    start = time.monotonic()
    
    def objective(coeffs):
        if time.monotonic() - start > time_budget:
            return 100.0
        k = np.arange(n)
        a = np.ones(n)
        for m, c in enumerate(coeffs, 1):
            a += c * np.cos(2 * np.pi * m * k / n)
        a = np.maximum(a, 0)
        S = np.sum(a)
        if S < 1e-10:
            return 100.0
        a = a / S
        return compute_obj(a)
    
    best_obj = np.inf
    best_coeffs = None
    
    for trial in range(4):
        if time.monotonic() - start > time_budget:
            break
        rng = np.random.default_rng(seed + trial * 999)
        x0 = rng.uniform(-0.5, 0.5, n_modes)
        try:
            res = minimize(objective, x0, method='Nelder-Mead',
                          options={'maxiter': 300, 'xatol': 1e-9, 'fatol': 1e-10,
                                   'adaptive': True})
            if res.fun < best_obj:
                best_obj = res.fun
                best_coeffs = res.x.copy()
        except Exception:
            pass
    
    if best_coeffs is not None:
        k = np.arange(n)
        a = np.ones(n)
        for m, c in enumerate(best_coeffs, 1):
            a += c * np.cos(2 * np.pi * m * k / n)
        a = np.maximum(a, 0)
        S = np.sum(a)
        if S > 1e-10:
            return a / S, best_obj
    
    return np.ones(n) / n, compute_obj(np.ones(n) / n)
