Minimizes J(a) = 2*n*max(convolve(a,a))/(sum(a))² using:

import numpy as np
from scipy.optimize import linprog, minimize
from scipy.sparse import csr_matrix
import time

_A_structure_cache = {}

def fast_conv(a, g=None):
    """FFT-based convolution. If g is None, self-convolution."""
    if g is None:
        g = a
    n_a = len(a)
    n_g = len(g)
    s = 1
    while s < n_a + n_g - 1:
        s <<= 1
    fa = np.fft.rfft(a, s)
    fg = np.fft.rfft(g, s)
    return np.fft.irfft(fa * fg, s)[:n_a + n_g - 1]

def compute_J(a):
    """Compute J(a) = 2*n*max(conv(a,a))/(sum(a))^2."""
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))

def normalize(a):
    """Normalize to sum=1, clip negatives."""
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(len(a)) / len(a)
    return a / s

def get_A_structure(n):
    if n not in _A_structure_cache:
        rows = []
        cols = []
        for k in range(2 * n - 1):
            jl = max(0, k - n + 1)
            jh = min(n - 1, k)
            for j in range(jl, jh + 1):
                rows.append(k)
                cols.append(j)
        _A_structure_cache[n] = (np.array(rows, dtype=np.int32),
                                  np.array(cols, dtype=np.int32))
    return _A_structure_cache[n]

def build_A_sparse(a):
    n = len(a)
    rows_arr, cols_arr = get_A_structure(n)
    vals = a[rows_arr - cols_arr]
    return csr_matrix((vals, (rows_arr, cols_arr)), shape=(2 * n - 1, n))

def conv_row(a, k):
    """Compute row k of the convolution matrix: (A @ g)_k = sum_j A[k,j] * g_j = (a*g)_k"""
    n = len(a)
    row = np.zeros(n)
    jl = max(0, k - n + 1)
    jh = min(n - 1, k)
    for j in range(jl, jh + 1):
        row[j] = a[k - j]
    return row

def lp_best_response_cp(a, M, max_cuts=80):
    """LP best response with cutting plane method."""
    n = len(a)
    # Start with top constraints
    b = fast_conv(a)
    top_k = np.argsort(b)[-10:]
    active = set(top_k.tolist())
    
    g = None
    for _ in range(max_cuts):
        # Build constraint matrix for active set
        A_rows = []
        for k in sorted(active):
            row = conv_row(a, k)
            A_rows.append(row)
        
        A_sub = np.array(A_rows)
        b_sub = np.full(len(A_rows), M)
        c = -np.ones(n)
        
        r = linprog(c, A_ub=A_sub, b_ub=b_sub, bounds=[(0.0, None)] * n, method='highs')
        if not r.success or r.x is None:
            break
        g = np.maximum(r.x, 0)
        g_sum = g.sum()
        if g_sum < 1e-10:
            break
        g_norm = g / g_sum
        
        # Check violations
        conv_ag = fast_conv(a, g_norm)
        violated = np.where(conv_ag > M * (1 + 1e-6))[0]
        
        if len(violated) == 0:
            break
        
        # Add most violated
        worst_idx = violated[np.argmax(conv_ag[violated])]
        if worst_idx not in active:
            active.add(int(worst_idx))
    
    if g is not None and g.sum() > 1e-10:
        return g / g.sum()
    return None

def lp_best_response_full(a, M):
    """Full LP best response."""
    n = len(a)
    c = -np.ones(n)
    A = build_A_sparse(a)
    b_ub = np.full(2 * n - 1, M)
    r = linprog(c, A_ub=A, b_ub=b_ub, bounds=[(0.0, None)] * n, method='highs')
    if r.success and r.x is not None:

        if g.sum() > 1e-10:

    return None

def backtracking_line_search(a, g, J_current, max_backtracks=30):
    """Backtracking line search for J((1-t)*a + t*g)."""
    t = 1.0
    for _ in range(max_backtracks):
        a_try = (1 - t) * a + t * g
        a_try = np.maximum(a_try, 0)
        s = a_try.sum()
        if s < 1e-10:
            t *= 0.5
            continue
        a_try = a_try / s
        J_try = compute_J(a_try)
        if J_try < J_current - 1e-15:
            return a_try, J_try, t
        t *= 0.5
    return a, J_current, 0.0

def lp_mirror_descent(a, max_iters, deadline, tracker=None, use_cp=True, stall_patience=50):
    """LP mirror descent with cutting-plane LP and backtracking line search."""
    a = normalize(a.astype(np.float64))
    current_J = compute_J(a)
    if tracker is not None:
        tracker.update(a)
    
    best_J = current_J
    best_a = a.copy()
    stall = 0
    
    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        
        b = fast_conv(a)
        M = float(b.max())
        
        if use_cp:
            g = lp_best_response_cp(a, M, max_cuts=60)
        else:
            g = lp_best_response_full(a, M)
        
        if g is None:
            break
        
        # Backtracking line search
        a_new, J_new, t_used = backtracking_line_search(a, g, current_J)
        
        if J_new < current_J - 1e-14:
            a = a_new
            current_J = J_new
            stall = 0
            if J_new < best_J:
                best_J = J_new
                best_a = a.copy()
                if tracker is not None:
                    tracker.update(a)
        else:
            stall += 1
            if stall >= stall_patience:
                break
    
    if tracker is not None:
        tracker.update(best_a)
    return best_a

def lbfgs_optimize(a0, betas, deadline):
    """L-BFGS-B with beta annealing."""
    a = np.maximum(a0.astype(np.float64), 1e-8)
    a = a / a.sum()
    n = len(a)
    
    for beta, max_iters in betas:
        if time.monotonic() > deadline:
            break
        
        def func(x):
            x2 = np.maximum(x, 0.0)
            f, _ = smooth_objective_and_grad(x2, beta)
            return f if np.isfinite(f) else 1e10
        
        def jac(x):
            x2 = np.maximum(x, 0.0)
            _, g = smooth_objective_and_grad(x2, beta)
            g = np.nan_to_num(g, nan=0.0, posinf=0.0, neginf=0.0)
            g[x2 < 1e-14] = 0.0
            return g
        
        try:
            res = minimize(func, a.copy(), jac=jac, method='L-BFGS-B',
                           bounds=[(0.0, None)] * n,
                           options={'maxiter': max_iters, 'max_fun': max_iters * 15,
                                    'gtol': 1e-8})
            if res.fun < 1e10:
                new_a = np.maximum(res.x, 1e-10)
                if new_a.sum() > 1e-6:
                    a = new_a
        except Exception:
            pass
    
    return a

def multiplicative_update(a, n_iters=50, eta=0.5):
    """Fast multiplicative (mirror) update for quick local improvement."""
    a = normalize(a)
    for _ in range(n_iters):
        b = fast_conv(a)
        M = b.max()
        if M < 1e-15:
            break
        # Subgradient direction
        # d max/da_j ≈ sum_k p_k * (a_{k-j} + a_{j-k}) where p is empirical distribution on argmax
        # Use soft version: p_k ∝ exp(beta * b_k)
        beta = 5.0
        shifted = beta * (b - M)
        p = np.exp(np.clip(shifted, -500, 500))
        p = p / p.sum()
        
        # Compute gradient of max(b) using p
        n = len(a)
        S = np.convolve(a, p[::-1], mode='full')
        S = S[n - 1:2 * n - 1][::-1]
        grad = 2.0 * S  # gradient of max(b) w.r.t. a
        
        # Multiplicative update: a_j <- a_j * (1 - eta * grad_j / a_j) projected to >= 0
        # Simpler: a_j <- a_j - eta * grad_j, clip to >= 0
        a = a - eta * grad
        a = np.maximum(a, 0)
        a = normalize(a)
    return a

def resample_seq(src, dest_n):
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)

def generate_initializations(n, rng, parent=None):
    inits = []
    t = np.linspace(0, 1, n, endpoint=False)
    
    if parent is not None:
        inits.append(resample_seq(parent, n))
    
    inits.append(np.ones(n) / n)
    inits.append(rng.uniform(0.5, 1.5, n))
    
    # Various window functions
    for alpha in [2, 4, 8, 16]:
        inits.append(np.exp(-alpha * (t - 0.5) ** 2))
    
    for freq in [1, 2, 3, 4, 6, 8]:
        inits.append(1 + 0.3 * np.cos(2 * np.pi * freq * t))
        inits.append(1 + 0.5 * np.cos(2 * np.pi * freq * t + rng.uniform(0, 2*np.pi)))
    
    # Raised cosine
    inits.append(0.5 + 0.5 * np.cos(2 * np.pi * t))
    
    # Dolph-Chebyshev like
    for rho in [0.3, 0.5, 0.7]:
        inits.append(np.abs(np.cos(np.pi * (t - 0.5) / 0.5)) ** rho)
    
    # Random smooth
    for _ in range(3):
        coeffs = rng.normal(0, 1, 8)
        inits.append(np.abs(np.sum(coeffs[i] * np.cos(2*np.pi*i*t) for i in range(8))) + 0.1)
    
    # Power law
    inits.append((t + 0.01) ** rng.uniform(-1, 2))
    inits.append((1 - t + 0.01) ** rng.uniform(-1, 2))
    
    return [normalize(x) for x in inits]

def ga_evolve(pop, n, n_gens, deadline, tracker, pop_size=64):
    """Genetic algorithm with multiplicative update local search."""
    rng = np.random.default_rng(int(time.monotonic() * 1e6) % 100000)
    
    # Initialize population
    population = []
    scores = []
    for a in pop:
        a = normalize(a)
        s = compute_J(a)
        population.append(a)
        scores.append(s)
    
    # Pad population
    while len(population) < pop_size:
        a = population[rng.integers(len(population))].copy()
        a = a + rng.normal(0, 0.1, n) * a
        a = normalize(a)
        population.append(a)
        scores.append(compute_J(a))
    
    best_idx = np.argmin(scores)
    best_a = population[best_idx].copy()
    best_score = scores[best_idx]
    tracker.update(best_a)
    
    for gen in range(n_gens):
        if time.monotonic() > deadline:
            break
        
        new_pop = []
        new_scores = []
        
        # Elitism: keep top 4
        elite_idx = np.argsort(scores)[:4]
        for i in elite_idx:
            new_pop.append(population[i].copy())
            new_scores.append(scores[i])
        
        # Generate offspring
        while len(new_pop) < pop_size:
            if time.monotonic() > deadline:
                break
            
            # Tournament selection
            t_idx = rng.integers(0, len(population), size=4)
            winner = t_idx[np.argmin([scores[i] for i in t_idx])]
            parent = population[winner]
            
            # Mutation
            r = rng.random()
            if r < 0.5:
                # Gaussian mutation
                child = parent + rng.normal(0, 0.05, n) * parent
            elif r < 0.7:
                # Crossover with random
                other_idx = rng.integers(0, len(population))
                cross_point = rng.integers(n // 4, 3 * n // 4)
                child = parent.copy()
                child[cross_point:] = population[other_idx][cross_point:]
            elif r < 0.85:
                # Local perturbation
                k = rng.integers(n // 8, n // 4)
                idx = rng.choice(n, size=k, replace=False)
                child = parent.copy()
                child[idx] *= rng.uniform(0.5, 1.5, k)
            else:
                # Fresh random
                child = rng.uniform(0.5, 1.5, n)
            
            child = normalize(child)
            
            # Quick local search: a few multiplicative updates
            if rng.random() < 0.3:
                child = multiplicative_update(child, n_iters=10, eta=0.3)
            
            child_score = compute_J(child)
            new_pop.append(child)
            new_scores.append(child_score)
        
        population = new_pop
        scores = new_scores
        
        best_idx = np.argmin(scores)
        if scores[best_idx] < best_score:
            best_score = scores[best_idx]
            best_a = population[best_idx].copy()
            tracker.update(best_a)
        
        if gen % 50 == 0:
            print(f"    GA gen {gen}: best={best_score:.8f}", flush=True)
    
    return best_a

"""
Minimizes J(a) = 2*n*max(convolve(a,a))/(sum(a))² using:
1. Genetic algorithm with multiplicative update local search for global exploration
2. L-BFGS-B with log-sum-exp smoothing for fast basin finding
3. LP mirror descent with cutting-plane acceleration for precise refinement
4. Multi-length exploration and final polish
"""

import numpy as np
from scipy.optimize import linprog, minimize
from scipy.sparse import csr_matrix
import time

_A_structure_cache = {}


def fast_conv(a, g=None):
    """FFT-based convolution. If g is None, self-convolution."""
    if g is None:
        g = a
    n_a = len(a)
    n_g = len(g)
    s = 1
    while s < n_a + n_g - 1:
        s <<= 1
    fa = np.fft.rfft(a, s)
    fg = np.fft.rfft(g, s)
    return np.fft.irfft(fa * fg, s)[:n_a + n_g - 1]


def compute_J(a):
    """Compute J(a) = 2*n*max(conv(a,a))/(sum(a))^2."""
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))


def normalize(a):
    """Normalize to sum=1, clip negatives."""
    a = np.maximum(a, 0).astype(np.float64)
    s = a.sum()
    if s < 1e-15:
        return np.ones(len(a)) / len(a)
    return a / s


def get_A_structure(n):
    if n not in _A_structure_cache:
        rows = []
        cols = []
        for k in range(2 * n - 1):
            jl = max(0, k - n + 1)
            jh = min(n - 1, k)
            for j in range(jl, jh + 1):
                rows.append(k)
                cols.append(j)
        _A_structure_cache[n] = (np.array(rows, dtype=np.int32),
                                  np.array(cols, dtype=np.int32))
    return _A_structure_cache[n]


def build_A_sparse(a):
    n = len(a)
    rows_arr, cols_arr = get_A_structure(n)
    vals = a[rows_arr - cols_arr]
    return csr_matrix((vals, (rows_arr, cols_arr)), shape=(2 * n - 1, n))


def conv_row(a, k):
    """Compute row k of the convolution matrix."""
    n = len(a)
    row = np.zeros(n)
    jl = max(0, k - n + 1)
    jh = min(n - 1, k)
    for j in range(jl, jh + 1):
        row[j] = a[k - j]
    return row


def lp_best_response_cp(a, M, max_cuts=80):
    """LP best response with cutting plane method."""
    n = len(a)
    b = fast_conv(a)
    top_k = np.argsort(b)[-15:]
    active = list(top_k)

    g = None
    prev_g = None
    for it in range(max_cuts):
        A_rows = []
        for k_idx in active:
            A_rows.append(conv_row(a, int(k_idx)))

        A_sub = np.array(A_rows)
        b_sub = np.full(len(A_rows), M)
        c = -np.ones(n)

        r = linprog(c, A_ub=A_sub, b_ub=b_sub, bounds=[(0.0, None)] * n, method='highs')
        if not r.success or r.x is None:
            break
        g = np.maximum(r.x, 0)
        g_sum = g.sum()
        if g_sum < 1e-10:
            break
        g_norm = g / g_sum

        conv_ag = fast_conv(a, g_norm)
        violated = np.where(conv_ag > M * (1 + 1e-8))[0]

        if len(violated) == 0:
            break

        worst_idx = int(violated[np.argmax(conv_ag[violated])])
        if worst_idx not in active:
            active.append(worst_idx)
        elif it > max_cuts // 2:
            break

    if g is not None and g.sum() > 1e-10:
        g_final = g / g.sum()
        conv_check = fast_conv(a, g_final)
        if conv_check.max() <= M * (1 + 1e-6):
            return g_final
        # Fallback: scale down
        ratio = M / max(conv_check.max(), 1e-15)
        if ratio > 0.5:
            return g_final * ratio
    return None


def lp_best_response_full(a, M):
    """Full LP best response."""
    n = len(a)
    c = -np.ones(n)
    A = build_A_sparse(a)
    b_ub = np.full(2 * n - 1, M)
    r = linprog(c, A_ub=A, b_ub=b_ub, bounds=[(0.0, None)] * n, method='highs')
    if r.success and r.x is not None:
        g = np.maximum(r.x, 0)
        if g.sum() > 1e-10:
            return g / g.sum()
    return None


def backtracking_line_search(a, g, J_current, max_backtracks=40):
    """Backtracking line search for J((1-t)*a + t*g)."""
    best_a = a
    best_J = J_current
    t = 1.0
    for _ in range(max_backtracks):
        a_try = (1 - t) * a + t * g
        a_try = np.maximum(a_try, 0)
        s = a_try.sum()
        if s < 1e-10:
            t *= 0.5
            continue
        a_try = a_try / s
        J_try = compute_J(a_try)
        if J_try < best_J:
            best_J = J_try
            best_a = a_try.copy()
        if J_try < J_current - 1e-15:
            return best_a, best_J, t
        t *= 0.5
    return best_a, best_J, 0.0


def lp_mirror_descent(a, max_iters, deadline, tracker=None, use_cp=True, stall_patience=60):
    """LP mirror descent with cutting-plane LP and backtracking line search."""
    a = normalize(a.astype(np.float64))
    current_J = compute_J(a)
    if tracker is not None:
        tracker.update(a)

    best_J = current_J
    best_a = a.copy()
    stall = 0
    it_count = 0

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        it_count += 1

        b = fast_conv(a)
        M = float(b.max())

        if use_cp:
            g = lp_best_response_cp(a, M, max_cuts=60)
        else:
            g = lp_best_response_full(a, M)

        if g is None:
            break

        a_new, J_new, t_used = backtracking_line_search(a, g, current_J)

        if J_new < current_J - 1e-14:
            a = a_new
            current_J = J_new
            stall = 0
            if J_new < best_J:
                best_J = J_new
                best_a = a.copy()
                if tracker is not None:
                    tracker.update(a)
        else:
            # Try smaller explicit steps
            improved = False
            for t_try in [0.01, 0.005, 0.02, 0.05, 0.001]:
                a_try = (1.0 - t_try) * a + t_try * g
                a_try = np.maximum(a_try, 0)
                s = a_try.sum()
                if s < 1e-10:
                    continue
                a_try = a_try / s
                J_try = compute_J(a_try)
                if J_try < current_J - 1e-14:
                    a = a_try
                    current_J = J_try
                    stall = 0
                    improved = True
                    if J_try < best_J:
                        best_J = J_try
                        best_a = a.copy()
                        if tracker is not None:
                            tracker.update(a)
                    break
            if not improved:
                stall += 1
                if stall >= stall_patience:
                    break

        if it % 200 == 0 and it > 0:
            elapsed = time.monotonic() - (deadline - max_iters * 0.01)
            print(f"    LP iter {it}: J={current_J:.10f} stall={stall}", flush=True)

    if tracker is not None:
        tracker.update(best_a)
    return best_a


def smooth_objective_and_grad(a, beta):
    """Log-sum-exp smoothed objective and gradient."""
    n = len(a)
    s = a.sum()
    if s < 1e-10:
        return 1e10, np.zeros(n)

    b = fast_conv(a)
    b_max = float(b.max())
    shifted = beta * (b - b_max)
    shifted = np.clip(shifted, -500, 500)
    exp_s = np.exp(shifted)
    Z = exp_s.sum()
    if Z < 1e-300:
        p = np.zeros(2 * n - 1)
        p[np.argmax(b)] = 1.0
    else:
        p = exp_s / Z

    g_val = (b_max + np.log(max(Z, 1e-300))) / beta
    F = 2.0 * n * g_val / (s * s)

    S = np.convolve(a, p[::-1], mode='full')
    S = S[n - 1:2 * n - 1][::-1]
    grad_inner = 2.0 * S

    grad_F = 2.0 * n * (s * grad_inner - 2.0 * g_val) / (s ** 3)

    return F, grad_F


def lbfgs_optimize(a0, betas, deadline):
    """L-BFGS-B with beta annealing."""
    a = np.maximum(a0.astype(np.float64), 1e-8)
    a = a / a.sum()
    n = len(a)

    for beta, max_iters in betas:
        if time.monotonic() > deadline:
            break

        def func(x):
            x2 = np.maximum(x, 0.0)
            f, _ = smooth_objective_and_grad(x2, beta)
            return f if np.isfinite(f) else 1e10

        def jac(x):
            x2 = np.maximum(x, 0.0)
            _, g = smooth_objective_and_grad(x2, beta)
            g = np.nan_to_num(g, nan=0.0, posinf=0.0, neginf=0.0)
            g[x2 < 1e-14] = 0.0
            return g

        try:
            res = minimize(func, a.copy(), jac=jac, method='L-BFGS-B',
                           bounds=[(0.0, None)] * n,
                           options={'maxiter': max_iters, 'max_fun': max_iters * 15,
                                    'gtol': 1e-8})
            if res.fun < 1e10:
                new_a = np.maximum(res.x, 1e-10)
                if new_a.sum() > 1e-6:
                    a = new_a / new_a.sum()
        except Exception:
            pass

    return a


def multiplicative_update(a, n_iters=20, eta=0.3):
    """Fast gradient-based local improvement."""
    a = normalize(a)
    n = len(a)
    for _ in range(n_iters):
        b = fast_conv(a)
        M = b.max()
        if M < 1e-15:
            break
        beta = 10.0
        shifted = beta * (b - M)
        p = np.exp(np.clip(shifted, -500, 500))
        p = p / p.sum()

        S = np.convolve(a, p[::-1], mode='full')
        S = S[n - 1:2 * n - 1][::-1]
        grad = 2.0 * S

        a = a - eta * grad
        a = np.maximum(a, 0)
        a = normalize(a)
    return a


def resample_seq(src, dest_n):
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)


def generate_initializations(n, rng, parent=None):
    inits = []
    t = np.linspace(0, 1, n, endpoint=False)

    if parent is not None:
        inits.append(resample_seq(parent, n))
        p = resample_seq(parent, n)
        inits.append(np.maximum(p + rng.normal(0, 0.05, n) * p, 0.01))

    inits.append(np.ones(n))
    inits.append(rng.uniform(0.5, 1.5, n))

    for alpha in [1, 2, 4, 8, 16, 32]:
        inits.append(np.exp(-alpha * (t - 0.5) ** 2) + 0.02)

    for freq in [1, 2, 3, 4, 5, 6, 8, 10]:
        phase = rng.uniform(0, 2 * np.pi)
        inits.append(1 + 0.3 * np.cos(2 * np.pi * freq * t + phase))
        inits.append(1 + 0.5 * np.cos(2 * np.pi * freq * t + phase))

    for rho in [0.3, 0.5, 1.0, 2.0]:
        inits.append(np.abs(np.cos(np.pi * (t - 0.5))) ** rho + 0.05)

    for _ in range(4):
        coeffs = rng.normal(0, 1, 10)
        sig = np.zeros(n)
        for i in range(10):
            sig += coeffs[i] * np.cos(2 * np.pi * i * t + rng.uniform(0, 2 * np.pi))
        inits.append(np.abs(sig) + 0.1)

    inits.append((t + 0.01) ** rng.uniform(-0.5, 1.5))
    inits.append((1 - t + 0.01) ** rng.uniform(-0.5, 1.5))
    inits.append(1.0 / (t * (1 - t) + 0.1))

    return [normalize(x) for x in inits]


class ScoreTracker:
    def __init__(self):
        self.best = None
        self.best_score = float('inf')
        self.t_start = time.monotonic()

    def update(self, a):
        a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
        if a.size == 0 or a.sum() < 0.01:
            return
        score = float(evaluate_sequence(a.tolist()))
        if np.isfinite(score) and score < self.best_score:
            self.best = a.copy()
            self.best_score = score
            elapsed = time.monotonic() - self.t_start
            print(f"  * NEW BEST: J={score:.10f} n={a.size} t={elapsed:.1f}s", flush=True)


def ga_evolve(inits, n, n_gens, deadline, tracker, pop_size=48):
    """Genetic algorithm with local refinement."""
    rng = np.random.default_rng(12345)

    population = []
    scores = []
    for a in inits:
        a = normalize(a)
        population.append(a)
        scores.append(compute_J(a))

    while len(population) < pop_size:
        src = population[rng.integers(len(population))].copy()
        a = src + rng.normal(0, 0.08, n) * src
        a = normalize(a)
        population.append(a)
        scores.append(compute_J(a))

    best_score = min(scores)
    best_a = population[np.argmin(scores)].copy()
    tracker.update(best_a)

    for gen in range(n_gens):
        if time.monotonic() > deadline:
            break

        scores_arr = np.array(scores)
        elite_mask = np.zeros(pop_size, dtype=bool)
        elite_mask[np.argsort(scores_arr)[:4]] = True

        new_pop = []
        new_scores = []

        for i in range(pop_size):
            if elite_mask[i]:
                new_pop.append(population[i].copy())
                new_scores.append(scores[i])
            else:
                r = rng.random()
                if r < 0.45:
                    t_idx = rng.integers(0, pop_size, size=4)
                    winner = t_idx[np.argmin(scores_arr[t_idx])]
                    parent = population[winner]
                    mutation_scale = 0.03 + 0.04 * (1 - gen / max(n_gens, 1))
                    child = parent + rng.normal(0, mutation_scale, n) * parent
                elif r < 0.65:
                    i1 = rng.integers(0, pop_size)
                    i2 = rng.integers(0, pop_size)
                    cp = rng.integers(n // 4, 3 * n // 4)
                    child = population[i1].copy()
                    child[cp:] = population[i2][cp:]
                    child += rng.normal(0, 0.02, n) * child
                elif r < 0.80:
                    t_idx = rng.integers(0, pop_size, size=3)
                    winner = t_idx[np.argmin(scores_arr[t_idx])]
                    parent = population[winner]
                    k = rng.integers(n // 16, n // 4)
                    idx = rng.choice(n, size=k, replace=False)
                    child = parent.copy()
                    child[idx] *= rng.uniform(0.4, 1.8, k)
                else:
                    child = rng.uniform(0.3, 2.0, n)

                child = normalize(child)

                if rng.random() < 0.4:
                    child = multiplicative_update(child, n_iters=8, eta=0.25)

                new_pop.append(child)
                new_scores.append(compute_J(child))

        population = new_pop
        scores = new_scores

        gen_best = min(scores)
        if gen_best < best_score:
            best_score = gen_best
            best_a = population[scores.index(best_score)].copy()
            tracker.update(best_a)

        if gen % 100 == 0:
            print(f"    GA gen {gen}/{n_gens}: best={best_score:.8f}", flush=True)

    return best_a


def propose_candidate(seed=42, budget_s=1500, **kwargs):
    """
    Multi-phase optimization combining GA exploration, L-BFGS-B basin finding,
    and LP mirror descent with cutting-plane acceleration for final refinement.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + float(budget_s) - 15.0
    tracker = ScoreTracker()

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        if p.ndim == 1 and p.size > 0 and np.all(np.isfinite(p)):
            parent = np.maximum(p, 0.0)
            tracker.update(parent)
            print(f"Parent loaded: n={parent.size}, J={tracker.best_score:.10f}", flush=True)
    except (NameError, AttributeError):
        pass

    if parent is None:
        parent = np.ones(512)
        tracker.update(parent)
        print(f"Default parent: n=512, J={tracker.best_score:.10f}", flush=True)

    main_n = parent.size if 64 <= parent.size <= 2048 else 1024

    # === Phase 1: GA with diverse initializations ===
    p1_budget = 180.0
    deadline_p1 = time.monotonic() + p1_budget
    print(f"\n{'='*50} Phase 1: GA (budget {p1_budget:.0f}s) n={main_n} {'='*50}", flush=True)

    inits = generate_initializations(main_n, rng, parent)
    t0 = time.monotonic()
    best_ga = ga_evolve(inits, main_n, n_gens=500, deadline=deadline_p1, tracker=tracker, pop_size=48)
    n_ga = len(best_ga)
    print(f"  GA done in {time.monotonic()-t0:.1f}s, best n={n_ga}", flush=True)

    # === Phase 2: L-BFGS-B refinement ===
    if time.monotonic() < deadline:
        p2_budget = 180.0
        deadline_p2 = min(deadline, time.monotonic() + p2_budget)
        print(f"\n{'='*50} Phase 2: L-BFGS-B (budget {p2_budget:.0f}s) {'='*50}", flush=True)

        candidates = [best_ga]
        if parent is not None and parent.size == n_ga:
            candidates.append(normalize(parent))
        elif parent is not None:
            candidates.append(normalize(resample_seq(parent, n_ga)))

        t = np.linspace(0, 1, n_ga, endpoint=False)
        for freq, amp in [(3, 0.2), (5, 0.3), (2, 0.4), (7, 0.25)]:
            candidates.append(1 + amp * np.cos(2 * np.pi * freq * t))
        candidates.append(np.exp(-4 * (t - 0.5) ** 2) + 0.1)

        betas = [(3.0, 25), (8.0, 30), (15.0, 35), (30.0, 40), (60.0, 40)]

        for i, c in enumerate(candidates):
            if time.monotonic() > deadline_p2:
                break
            c = normalize(c)
            t0 = time.monotonic()
            optimized = lbfgs_optimize(c, betas, deadline_p2)
            J_val = compute_J(optimized)
            tracker.update(optimized)
            print(f"  L-BFGS-B cand {i}: J={J_val:.10f} t={time.monotonic()-t0:.1f}s", flush=True)

    # === Phase 3: LP Mirror Descent (main refinement) ===
    if tracker.best is not None and time.monotonic() < deadline:
        p3_budget = 750.0
        deadline_p3 = min(deadline, time.monotonic() + p3_budget)
        print(f"\n{'='*50} Phase 3: LP Mirror Descent (budget {p3_budget:.0f}s) {'='*50}", flush=True)

        a = tracker.best.copy()
        t0 = time.monotonic()
        a = lp_mirror_descent(a, 5000, deadline_p3, tracker=tracker, use_cp=True, stall_patience=80)
        print(f"  LP CD pass 1: {time.monotonic()-t0:.1f}s", flush=True)

        # Second pass with full LP for precision
        if tracker.best is not None and time.monotonic() < deadline_p3:
            a2 = tracker.best.copy()
            t0 = time.monotonic()
            a2 = lp_mirror_descent(a2, 3000, deadline_p3, tracker=tracker, use_cp=False, stall_patience=50)
            print(f"  LP full pass 2: {time.monotonic()-t0:.1f}s", flush=True)

    # === Phase 4: Multi-length exploration ===
    if tracker.best is not None and time.monotonic() < deadline:
        remaining = deadline - time.monotonic()
        if remaining > 50:
            print(f"\n{'='*50} Phase 4: Length Exploration ({remaining:.0f}s) {'='*50}", flush=True)

            best_n = tracker.best.size
            for n_try in sorted(set([256, 384, 512, 640, 768, 1024, 1280, 1536, 2048]) - {best_n}):
                if time.monotonic() > deadline:
                    break
                a_res = resample_seq(tracker.best, n_try)
                a_res = normalize(a_res)
                budget_here = min(35.0, deadline - time.monotonic())
                if budget_here < 10:
                    break
                a_res = lbfgs_optimize(a_res, [(15.0, 15), (40.0, 20)], time.monotonic() + budget_here * 0.4)
                tracker.update(a_res)
                rem = deadline - time.monotonic()
                if rem > 5:
                    a_res = lp_mirror_descent(a_res, int(rem * 15), deadline, tracker=tracker,
                                              use_cp=True, stall_patience=20)

    # === Phase 5: Final polish ===
    if tracker.best is not None and time.monotonic() < deadline:
        remaining = deadline - time.monotonic()
        if remaining > 15:
            print(f"\n{'='*50} Phase 5: Final Polish ({remaining:.0f}s) {'='*50}", flush=True)
            a = tracker.best.copy()
            a = lp_mirror_descent(a, int(remaining * 25), deadline, tracker=tracker,
                                  use_cp=True, stall_patience=40)

            if tracker.best is not None and time.monotonic() < deadline:
                rem2 = deadline - time.monotonic()
                if rem2 > 10:
                    a2 = tracker.best.copy()
                    a2 = lp_mirror_descent(a2, int(rem2 * 15), deadline, tracker=tracker,
                                           use_cp=False, stall_patience=30)

    # Fallback
    if tracker.best is None:
        tracker.best = np.ones(512)
        tracker.best_score = compute_J(tracker.best)

    print(f"\n{'='*50} FINAL RESULT {'='*50}", flush=True)
    print(f"J={tracker.best_score:.10f} n={tracker.best.size} "
          f"time={time.monotonic()-tracker.t_start:.1f}s", flush=True)

    result = np.clip(tracker.best, 0.0, 1000.0)
    if result.sum() < 0.01:
        result = np.ones(128)

    return [float(x) for x in result]
