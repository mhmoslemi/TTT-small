import numpy as np
from scipy.optimize import linprog, minimize
from scipy.sparse import csr_matrix
import time


def fast_conv(a):
    """Full linear self-convolution via FFT."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]


def compute_J(a):
    """Compute J(a) = 2*n*max(conv(a,a))/(sum(a))^2."""
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))


def smooth_max_and_grad(a, tau):
    """
    Compute smooth max proxy and its gradient.
    Returns (smooth_max_value, gradient, softmax_weights, conv_values).
    """
    n = len(a)
    b = fast_conv(a)
    b_max = b.max()
    exp_b = np.exp(tau * (b - b_max))
    p = exp_b / exp_b.sum()
    smooth_val = (np.log(exp_b.sum()) / tau) + b_max

    grad = np.zeros(n)
    for k in range(2 * n - 1):
        pk = p[k]
        if pk < 1e-14:
            continue
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        grad[lo:hi + 1] += 2.0 * pk * a[k - hi:k - lo + 1][::-1]

    return smooth_val, grad, p, b


def smooth_gradient_step(a, tau, lr):
    """One adaptive smooth gradient descent step."""
    _, grad, p, b = smooth_max_and_grad(a, tau)
    a_new = a - lr * grad
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s < 1e-10:
        return a
    return a_new / s

    # If they're more flat, we need higher tau
    top_k = min(20, 2 * n - 1)
    b_top = b_sorted[:top_k] / b_max  # normalized, first = 1

    # Expected decay under current tau: exp(-tau * (b_max - b_top[i]) / tau) ... 
    # Actually, let's just measure the "effective support" 
    # If effective support is small, we need higher tau
    eff_support = 1.0 / max(1e-15, (p[:top_k] ** 2).sum() / p[:top_k].sum() ** 2) \
        if p.sum() > 0 else top_k

    # Alternative: measure the ratio b[1]/b[0]
    ratio = b_sorted[1] / max(b_sorted[0], 1e-15)
    
    # If ratio is close to 1, the max is "flat" and we need higher tau
    # If ratio is well below 1, we have a sharp peak and lower tau is fine
    if ratio > 0.99:
        return min(max_tau, current_tau * 1.3)
    elif ratio > 0.95:
        return min(max_tau, current_tau * 1.1)
    elif ratio < 0.8:
        return max(min_tau, current_tau * 0.9)
    return current_tau


def build_A_rows(a, indices):
    """Build sparse A matrix rows for given convolution lag indices."""
    n = len(a)
    rows = []
    cols = []
    vals = []
    for i, k in enumerate(indices):
        j_min = max(0, k - n + 1)
        j_max = min(n - 1, k)
        for j in range(j_min, j_max + 1):
            rows.append(i)
            cols.append(j)
            vals.append(a[k - j])
    if len(rows) == 0:
        return csr_matrix((0, n))
    return csr_matrix((vals, (rows, cols)), shape=(len(indices), n))


def solve_lp_active(a, M, active_indices, time_limit=0.5):
    """Solve LP: max sum(g) s.t. (a*g)[k] <= M for k in active_indices, g >= 0."""
    n = len(a)
    if len(active_indices) == 0:
        return None

    A = build_A_rows(a, active_indices)
    c = -np.ones(n)
    b_ub = np.full(len(active_indices), M)
    bounds = [(0, None)] * n
    try:
        r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                    options={'time_limit': time_limit, 'presolve': True})
        if r.success and r.x is not None:
            return np.maximum(r.x, 0)
    except Exception:
        pass
    return None


def solve_lp_full(a, M, time_limit=5.0):
    """Solve full LP with all 2n-1 constraints."""
    n = len(a)
    all_indices = np.arange(2 * n - 1)
    return solve_lp_active(a, M, all_indices, time_limit=time_limit)


def active_set_lp_mirror_descent(a, deadline, verbose=True):
    """
    Main optimization: Active-set LP mirror descent with line search.
    Uses only near-max constraints in the LP for speed.
    """
    a = np.maximum(a.astype(np.float64), 1e-10)
    a = a / a.sum()
    n = len(a)

    best_a = a.copy()
    best_M = fast_conv(a).max()

    # Active set management
    active_tol = 0.005  # fraction of M for "active"
    active_expand_tol = 0.001  # tighter threshold for checking violations
    active_set = None  # Will be computed
    max_active = min(2 * n - 1, 500)  # Cap on active set size

    stagnant_count = 0
    stale_tol = 50  # iterations without improvement before switching strategy
    last_improve_M = best_M
    current_tau = 20.0
    use_smooth = False  # Flag to alternate to smooth gradient

    it = 0
    t0 = time.monotonic()

    while time.monotonic() < deadline:
        b = fast_conv(a)
        M = float(b.max())

        # Update active set
        active_thresh = M * (1 - active_tol)
        active_mask = b >= active_thresh
        active_indices = np.where(active_mask)[0]

        # Cap active set size (keep the largest)
        if len(active_indices) > max_active:
            b_active = b[active_indices]
            order = np.argsort(b_active)[::-1][:max_active]
            active_indices = active_indices[order]

        # Check for stagnation
        if M < last_improve_M * (1 - 1e-7):
            last_improve_M = M
            stagnant_count = 0
        else:
            stagnant_count += 1

        # Strategy switching on stagnation
        if stagnant_count > stale_tol:
            if not use_smooth:
                # Switch to smooth gradient for a few iterations
                use_smooth = True
                stagnant_count = 0
                active_tol = max(0.002, active_tol * 0.5)  # Tighten active set
                if verbose:
                    print(f"    [it {it}] Stagnant -> smooth gradient mode, "
                          f"active_tol={active_tol:.4f}", flush=True)
            else:
                # Reset: widen active set, go back to LP
                use_smooth = False
                active_tol = min(0.02, active_tol * 2.0)
                # Perturb slightly
                a = a + np.random.default_rng(it).normal(0, 0.005, n)
                a = np.maximum(a, 1e-8)
                a = a / a.sum()
                stagnant_count = 0
                if verbose:
                    print(f"    [it {it}] Reset -> LP mode, "
                          f"active_tol={active_tol:.4f}", flush=True)

        if use_smooth:
            # Smooth gradient step
            _, grad, p, b_curr = smooth_max_and_grad(a, current_tau)
            # Adaptive tau
            current_tau = adapt_tau(a, p, current_tau)
            lr = 0.15 * (1.0 / (1 + it * 0.001))
            a_new = a - lr * grad
            a_new = np.maximum(a_new, 0.0)
            s = a_new.sum()
            if s < 1e-10:
                a_new = a.copy()
            else:
                a_new = a_new / s
            a = a_new
            it += 1
            continue

        # LP Mirror Descent step
        g = solve_lp_active(a, M, active_indices, time_limit=0.3)
        if g is None:
            # LP failed, use smooth gradient as fallback
            a = smooth_gradient_step(a, current_tau, 0.1)
            it += 1
            continue

        g_sum = g.sum()
        if g_sum < 1e-12:
            break
        g = g / g_sum

        # Verify full constraint satisfaction: check (a*g)[k] <= M for all k
        # (This is expensive but ensures correctness)
        # For speed, only check a random subset or the top inactive
        ag = fast_conv(a, )  # We need a*g, not a*a
        # Actually conv(a, g) via FFT:
        size = 1
        while size < 2 * n - 1:
            size <<= 1
        fa = np.fft.rfft(a, size)
        fg = np.fft.rfft(g, size)
        ag = np.fft.irfft(fa * fg, size)[:2 * n - 1]

        violations = np.where(ag > M * (1 + active_expand_tol))[0]
        if len(violations) > 0 and len(active_indices) < max_active:
            # Add violated constraints to active set
            new_active = np.union1d(active_indices, violations[:50])
            if len(new_active) <= max_active:
                active_indices = new_active
                # Re-solve with expanded active set
                g = solve_lp_active(a, M, active_indices, time_limit=0.3)
                if g is None:
                    a = smooth_gradient_step(a, current_tau, 0.1)
                    it += 1
                    continue
                g_sum = g.sum()
                if g_sum < 1e-12:
                    break
                g = g / g_sum

        # Line search
        best_M_trial = M
        best_a_trial = None
        best_t = 0.0

        # Adaptive step sizes
        base_ts = [0.4, 0.3, 0.2, 0.15, 0.1, 0.07, 0.05, 0.03, 0.02, 0.01]
        for t in base_ts:
            a_trial = (1 - t) * a + t * g
            a_trial = np.maximum(a_trial, 0.0)
            s = a_trial.sum()
            if s < 1e-10:
                continue
            a_trial = a_trial / s
            M_trial = fast_conv(a_trial).max()
            if M_trial < best_M_trial:
                best_M_trial = M_trial
                best_a_trial = a_trial
                best_t = t
            else:
                break  # Unimodal: stop searching

        if best_a_trial is not None:
            a = best_a_trial
            if best_M_trial < best_M:
                best_M = best_M_trial
                best_a = a.copy()
                if verbose and it % 100 == 0:
                    J_val = 2 * n * best_M / best_M  # J = 2n*M/(sum^2) = 2n*M since sum=1
                    print(f"    [it {it}] M={best_M:.10f} J={2*n*best_M:.6f} "
                          f"t={best_t:.3f} active={len(active_indices)}", flush=True)
        else:
            # No improvement found; try very small step
            t = 0.005
            a_trial = (1 - t) * a + t * g
            a_trial = np.maximum(a_trial, 0.0)
            s = a_trial.sum()
            if s > 1e-10:
                a = a_trial / s

        it += 1

        if time.monotonic() - t0 > 120 and it % 500 == 0 and verbose:
            elapsed = time.monotonic() - t0
            print(f"    [it {it}] M={fast_conv(a).max():.10f} "
                  f"best_M={best_M:.10f} elapsed={elapsed:.0f}s", flush=True)

    return best_a


def smooth_gradient_descent_adaptive(a, max_iters, deadline, tau_init=10.0):
    """Adaptive smooth gradient descent with goodness-of-fit tau."""
    a = np.maximum(a.astype(np.float64), 1e-10)
    a = a / a.sum()
    n = len(a)
    best_a = a.copy()
    best_M = fast_conv(a).max()
    tau = tau_init
    v = np.zeros(n)  # momentum

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        _, grad, p, b = smooth_max_and_grad(a, tau)
        tau = adapt_tau(a, p, tau)
        M = b.max()

        if M < best_M * (1 - 1e-9):
            best_M = M
            best_a = a.copy()

        lr = 0.3 / (1 + 0.01 * it)
        beta = 0.85
        v = beta * v + lr * grad
        a_new = a - v
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s < 1e-10:
            a_new = a.copy()
        else:
            a_new = a_new / s
        a = a_new

    return best_a


def resample_seq(src, dest_n):
    """Resample 1D sequence to new length via linear interpolation."""
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)


def generate_structured_inits(n, rng, parent=None):
    """Generate diverse structured initializations."""
    inits = []
    t = np.linspace(0, 1, n, endpoint=False)

    if parent is not None:
        p = resample_seq(parent, n)
        inits.append(p)
        inits.append(np.maximum(p + rng.normal(0, 0.02, n), 0.01))
        inits.append(np.maximum(p + rng.normal(0, 0.08, n), 0.01))
        # Anti-palindromic perturbation
        pa = p.copy()
        for i in range(n):
            pa[i] = max(0.01, p[i] + 0.1 * (2 * t[i] - 1))
        inits.append(pa / pa.sum())

    # Base structures
    inits.append(np.ones(n))
    inits.append(rng.uniform(0.5, 1.0, n))

    # Anti-palindromic (key structure for this problem)
    inits.append(np.maximum(0.5 + 0.5 * (2 * t - 1), 0.01))
    inits.append(np.maximum(0.8 - 0.4 * (2 * t - 1), 0.01))
    inits.append(np.maximum(0.5 + 0.3 * (2 * t - 1) + 0.2 * np.sin(4 * np.pi * t), 0.01))

    # Oscillating (helps with middle convolution)
    for freq in [2, 3, 4, 6, 8]:
        inits.append(np.maximum(1.0 + 0.3 * np.cos(2 * np.pi * freq * t), 0.01))
    inits.append(np.maximum(1.0 + 0.2 * np.cos(2 * np.pi * 3 * t) +
                            0.2 * np.cos(2 * np.pi * 7 * t), 0.01))

    # Sigmoid-like
    inits.append(1.0 / (1.0 + np.exp(-10 * (t - 0.5))))
    inits.append(1.0 / (1.0 + np.exp(10 * (t - 0.5))))

    # Two bumps (anti-symmetric)
    inits.append(np.exp(-20 * (t - 0.3) ** 2) + 0.1 * np.exp(-20 * (t - 0.7) ** 2) + 0.05)
    inits.append(0.1 * np.exp(-20 * (t - 0.3) ** 2) + np.exp(-20 * (t - 0.7) ** 2) + 0.05)

    # Smooth random
    ksize = max(3, n // 30)
    kernel = np.ones(ksize) / ksize
    smooth = np.convolve(rng.uniform(0, 1, n), kernel, mode='same')
    inits.append(np.maximum(smooth + 0.1, 0.01))

    # Block structure
    n_blocks = rng.integers(4, max(5, n // 50))
    block_vals = rng.uniform(0.3, 1.0, n_blocks)
    block_seq = np.repeat(block_vals, n // n_blocks + 1)[:n]
    inits.append(np.maximum(block_seq, 0.01))

    # Power law
    inits.append(np.maximum(t ** 0.5 + 0.1, 0.01))
    inits.append(np.maximum((1 - t) ** 0.5 + 0.1, 0.01))

    return inits


def lp_polish(a, max_iters, deadline, verbose=True):
    """Full LP polish (all constraints) for final refinement."""
    a = np.maximum(a.astype(np.float64), 1e-10)
    a = a / a.sum()
    n = len(a)
    best_a = a.copy()
    best_M = fast_conv(a).max()
    all_indices = np.arange(2 * n - 1)

    no_improve = 0
    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        b = fast_conv(a)
        M = float(b.max())

        g = solve_lp_full(a, M, time_limit=2.0)
        if g is None:
            break

        g_sum = g.sum()
        if g_sum < 1e-12:
            break
        g = g / g_sum

        # Line search
        improved = False
        for t in [0.3, 0.2, 0.15, 0.1, 0.07, 0.05, 0.03, 0.01]:
            a_trial = (1 - t) * a + t * g
            a_trial = np.maximum(a_trial, 0.0)
            s = a_trial.sum()
            if s < 1e-10:
                continue
            a_trial = a_trial / s
            M_trial = fast_conv(a_trial).max()
            if M_trial < best_M - 1e-15:
                a = a_trial
                best_M = M_trial
                best_a = a.copy()
                improved = True
                break

        if not improved:
            no_improve += 1
            if no_improve > 30:
                break
        else:
            no_improve = 0

        if verbose and it % 20 == 0:
            print(f"      LP polish [it {it}] M={M:.10f}", flush=True)

    return best_a


def lbb_smooth_polish(a, deadline):
    """L-BFGS-B polish using smooth max as proxy."""
    n = len(a)
    bounds_list = [(0.0, 10.0)] * n

    def obj(x):
        ax = np.maximum(x, 0.0)
        s = ax.sum()
        if s < 1e-8:
            return 1e10
        ax = ax / s
        b = fast_conv(ax)
        tau = 80.0
        b_shifted = b - b.max()
        return float(np.log(np.sum(np.exp(tau * b_shifted)))) / tau

    def obj_grad(x):
        ax = np.maximum(x, 0.0)
        s = ax.sum()
        if s < 1e-8:
            return np.zeros_like(x)
        ax = ax / s
        nn = len(ax)
        b = fast_conv(ax)
        tau = 80.0
        b_max = b.max()
        exp_b = np.exp(tau * (b - b_max))
        p = exp_b / exp_b.sum()
        grad = np.zeros(nn)
        for k in range(2 * nn - 1):
            pk = p[k]
            if pk < 1e-13:
                continue
            lo = max(0, k - nn + 1)
            hi = min(nn - 1, k)
            grad[lo:hi + 1] += 2.0 * pk * ax[k - hi:k - lo + 1][::-1]
        return grad

    try:
        res = minimize(obj, a.copy(), jac=obj_grad, method='L-BFGS-B',
                       bounds=bounds_list,
                       options={'maxiter': 200, 'max_fun': 20000,
                                'ftol': 1e-20, 'gtol': 1e-12,
                                'maxls': 50})
        if res.fun < 1e8:
            return np.maximum(res.x, 0)
    except Exception:
        pass
    return a

def propose_candidate(seed=42, budget_s=1500, **kwargs):
    """
    Multi-phase hybrid search for minimizing J(a) = 2n*max(conv(a,a))/(sum a)^2.
    
    Phase 1: Quick exploration across lengths and initializations (150s)
    Phase 2: Active-set LP mirror descent on best candidates (900s)
    Phase 3: Full LP polish + L-BFGS-B + final refinement (400s)
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + float(budget_s) - 10.0
    t_start = time.monotonic()

    best_sequence = None
    best_score = float('inf')

    def update_best(a):
        nonlocal best_sequence, best_score
        a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
        if a.size == 0 or a.sum() < 0.01:
            return
        score = float(evaluate_sequence(a.tolist()))
        if np.isfinite(score) and score < best_score:
            best_sequence = a.copy()
            best_score = score
            elapsed = time.monotonic() - t_start
            print(f"  *** NEW BEST: J={score:.8f} n={a.size} t={elapsed:.0f}s",
                  flush=True)

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        if p.ndim == 1 and p.size > 0 and np.all(np.isfinite(p)):
            parent = np.maximum(p, 0.0)
            update_best(parent)
            print(f"Parent: n={parent.size}, J={best_score:.6f}", flush=True)
    except (NameError, AttributeError):
        pass

    if parent is None:
        parent = np.ones(256)
        update_best(parent)

    # === PHASE 1: Quick Exploration ===
    print(f"\n{'='*60}", flush=True)
    print(f"PHASE 1: Exploration (budget=150s)", flush=True)
    print(f"{'='*60}", flush=True)

    p1_deadline = min(time.monotonic() + 150.0, deadline - 800.0)

    # Try multiple lengths
    base_n = parent.size if parent.size >= 128 else 512
    lengths_to_try = sorted(set([
        base_n,
        max(128, base_n // 2),
        base_n * 2 if base_n * 2 <= 2048 else base_n,
        512, 768, 1024
    ]))
    lengths_to_try = [n for n in lengths_to_try if 128 <= n <= 2048]

    candidates = []  # List of (a, J, n)

    for n in lengths_to_try:
        if time.monotonic() > p1_deadline:
            break

        time_per_init = max(3.0, (p1_deadline - time.monotonic()) / (len(lengths_to_try) * 8))
        inits = generate_structured_inits(n, rng, parent)

        for a0 in inits:
            if time.monotonic() > p1_deadline:
                break
            a0 = np.maximum(a0.astype(np.float64), 1e-6)
            a0 = a0 / a0.sum()

            # Quick smooth gradient
            a = smooth_gradient_descent_adaptive(
                a0, max_iters=200,
                deadline=time.monotonic() + time_per_init,
                tau_init=10.0
            )
            J_val = compute_J(a)
            candidates.append((a, J_val, n))

    # Sort by J and keep top candidates
    candidates.sort(key=lambda x: x[1])
    top_candidates = candidates[:5]
    print(f"  Phase 1: {len(candidates)} candidates, best J={candidates[0][1]:.6f} "
          f"(n={candidates[0][2]}) [{time.monotonic()-t_start:.0f}s]", flush=True)
    for i, (a, J, n) in enumerate(top_candidates[:3]):
        print(f"    #{i+1}: J={J:.6f} n={n}", flush=True)

    update_best(top_candidates[0][0])

    # === PHASE 2: Active-Set LP Mirror Descent ===
    print(f"\n{'='*60}", flush=True)
    print(f"PHASE 2: Active-Set LP Mirror Descent (budget=900s)", flush=True)
    print(f"{'='*60}", flush=True)

    p2_deadline = min(time.monotonic() + 900.0, deadline - 200.0)

    # Process top candidates, allotting more time to better ones
    time_per_candidate = 200.0
    for idx, (a_start, J_start, n) in enumerate(top_candidates):
        if time.monotonic() > p2_deadline:
            break

        if idx == 0:
            cand_deadline = p2_deadline - 50.0
        else:
            cand_deadline = min(time.monotonic() + time_per_candidate, p2_deadline)

        print(f"  Processing candidate #{idx+1}: n={n}, J_start={J_start:.6f} "
              f"(budget={cand_deadline - time.monotonic():.0f}s)", flush=True)

        # Active-set LP mirror descent
        a_opt = active_set_lp_mirror_descent(
            a_start, deadline=cand_deadline, verbose=True
        )
        update_best(a_opt)

        # Also try a smooth gradient polish
        if time.monotonic() < cand_deadline:
            a_smooth = smooth_gradient_descent_adaptive(
                a_opt, max_iters=2000,
                deadline=cand_deadline, tau_init=30.0
            )
            update_best(a_smooth)

    # === PHASE 3: Final Polishing ===
    print(f"\n{'='*60}", flush=True)
    print(f"PHASE 3: Final Polishing (remaining time)", flush=True)
    print(f"{'='*60}", flush=True)

    if best_sequence is not None and time.monotonic() < deadline - 5:
        # Full LP polish
        p3a_deadline = min(time.monotonic() + 200.0, deadline - 50.0)
        print(f"  Full LP polish (budget={p3a_deadline - time.monotonic():.0f}s)...", flush=True)
        a_polished = lp_polish(best_sequence, max_iters=200,
                               deadline=p3a_deadline, verbose=True)
        update_best(a_polished)

        # L-BFGS-B smooth polish
        if time.monotonic() < deadline - 10:
            print(f"  L-BFGS-B smooth polish...", flush=True)
            a_lbb = lbb_smooth_polish(best_sequence, deadline - 10)
            update_best(a_lbb)

        # One more LP pass on the best
        if time.monotonic() < deadline - 5:
            print(f"  Final LP pass...", flush=True)
            a_final = lp_polish(best_sequence, max_iters=50,
                                deadline=deadline - 2, verbose=False)
            update_best(a_final)

        # Try length extension
        if time.monotonic() < deadline - 20:
            new_n = best_sequence.size * 2
            if new_n <= 3072:
                print(f"  Trying length extension to n={new_n}...", flush=True)
                a_ext = resample_seq(best_sequence, new_n)
                a_ext = np.maximum(a_ext, 1e-6)
                a_ext = a_ext / a_ext.sum()
                a_ext = smooth_gradient_descent_adaptive(
                    a_ext, max_iters=500, deadline=deadline - 5, tau_init=15.0
                )
                update_best(a_ext)

    # Fallback
    if best_sequence is None:
        best_sequence = np.ones(128)
        best_score = compute_J(best_sequence)

    print(f"\n{'='*60}", flush=True)
    print(f"FINAL: J={best_score:.8f} n={best_sequence.size} "
          f"time={time.monotonic()-t_start:.0f}s", flush=True)
    print(f"{'='*60}", flush=True)

    return [float(x) for x in best_sequence]
