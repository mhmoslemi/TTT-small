def spectral_smooth(a, n_keep=None):
    """Low-pass filter in Fourier domain."""
    n = len(a)
    size = 1
    while size < n:
        size <<= 1
    fa = np.fft.rfft(a, size)
    if n_keep is None:
        n_keep = max(4, n // 8)
    # Zero out high frequencies
    rfft_len = size // 2 + 1
    for k in range(n_keep, rfft_len):
        fa[k] = 0
    a_smooth = np.fft.irfft(fa, size)[:n]
    a_smooth = np.maximum(a_smooth, 0.0)
    s = a_smooth.sum()
    if s < 1e-10:
        return a.copy()
    return a_smooth / s


def resample_seq(src, dest_n):
    """Resample 1D sequence to new length."""
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)


def subgradient_polish(a, max_iters, deadline, step_init=0.3):
    """Subgradient descent with momentum on the max-convolution objective."""
    n = len(a)
    a = np.maximum(a.astype(np.float64), 1e-12)
    a = a / a.sum()
    best_J = compute_J(a)
    best_a = a.copy()
    momentum = np.zeros_like(a)
    beta = 0.7
    step = step_init

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        b = fast_conv(a)
        M = b.max()
        tol = 1e-10 * M + 1e-15
        active = np.where(b >= M - tol)[0]

        if len(active) == 0:
            break

        grad = np.zeros_like(a)
        for k in active:
            lo = max(0, k - n + 1)
            hi = min(n - 1, k)
            for i in range(lo, hi + 1):
                grad[i] += a[k - i]

        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-15:
            break

        # Decaying step size
        t = step / (1.0 + 0.01 * it)
        momentum = beta * momentum + (1.0 - beta) * grad / gnorm
        a_new = a - t * momentum
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s < 1e-10:
            break
        a_new = a_new / s

        J_new = compute_J(a_new)
        if J_new < best_J:
            best_J = J_new
            best_a = a_new.copy()
        a = a_new

    return best_a, best_J


def propose_candidate(seed=42, budget_s=1500, **kwargs):
    """
    Optimized LP mirror descent with restricted constraints, momentum,
    spectral smoothing, and multi-length exploration.
    """
    tracker_best = None
    tracker_best_score = float('inf')
    t_start = time.monotonic()
    deadline = t_start + float(budget_s) - 20.0
    rng = np.random.default_rng(seed)

    def track(a):
        nonlocal tracker_best, tracker_best_score
        a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
        if a.size == 0 or a.sum() < 0.01:
            return
        try:
            score = float(evaluate_sequence(a.tolist()))
        except Exception:
            return
        if np.isfinite(score) and score < tracker_best_score:
            tracker_best = a.copy()
            tracker_best_score = score
            elapsed = time.monotonic() - t_start
            print(f"  [NEW BEST] J={score:.8f} n={a.size} t={elapsed:.0f}s", flush=True)

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        if p.ndim == 1 and p.size > 0 and np.all(np.isfinite(p)):
            parent = np.maximum(p, 0.0)
            track(parent)
            print(f"Parent: n={parent.size}, J={tracker_best_score:.6f}", flush=True)
    except (NameError, AttributeError):
        pass

    if parent is None:
        parent = np.ones(512)
        track(parent)

    n0 = parent.size
    a_cur = parent.astype(np.float64).copy()
    a_cur = np.maximum(a_cur, 1e-12)
    a_cur = a_cur / a_cur.sum()

    print(f"\n{'='*60}", flush=True)
    print(f"Starting optimization: n={n0}, J={compute_J(a_cur):.8f}", flush=True)
    print(f"{'='*60}", flush=True)

    # === Phase 1: Main LP mirror descent with restricted constraints ===
    t0 = time.monotonic()
    dl1 = deadline - 250
    print(f"\nPhase 1: LP mirror descent (K=60, momentum) n={n0}", flush=True)

    a1, J1 = lp_mirror_descent_core(a_cur, 100000, dl1, K=60,
                                     use_momentum=True, momentum_init=0.4,
                                     step_init=0.5)
    track(a1)
    print(f"  Phase 1 done: J={J1:.8f} ({time.monotonic()-t0:.1f}s)", flush=True)

    if J1 < compute_J(a_cur):
        a_cur = a1.copy()

    # === Phase 2: Full LP refinement ===
    if time.monotonic() < deadline - 150:
        t0 = time.monotonic()
        dl2 = deadline - 150
        print(f"\nPhase 2: Full LP refinement n={n0}", flush=True)

        a2, J2 = lp_mirror_descent_core(a_cur, 100000, dl2, K=2 * n0 - 1,
                                         use_momentum=False,
                                         step_init=0.3)
        track(a2)
        print(f"  Phase 2 done: J={J2:.8f} ({time.monotonic()-t0:.1f}s)", flush=True)

        if J2 < compute_J(a_cur):
            a_cur = a2.copy()

    # === Phase 3: Spectral smoothing + LP ===
    if time.monotonic() < deadline - 100:
        t0 = time.monotonic()
        print(f"\nPhase 3: Spectral smoothing iterations", flush=True)

        for n_keep_frac in [0.25, 0.15, 0.35, 0.2]:
            if time.monotonic() > deadline - 80:
                break
            a_smooth = spectral_smooth(a_cur, n_keep=int(n0 * n_keep_frac))
            a_try, J_try = lp_mirror_descent_core(a_smooth, 5000, time.monotonic() + 20,
                                                   K=40, use_momentum=True,
                                                   momentum_init=0.3, step_init=0.4)
            track(a_try)
            if J_try < compute_J(a_cur):
                a_cur = a_try.copy()
                print(f"  Smoothed (frac={n_keep_frac}): J={J_try:.8f}", flush=True)

        print(f"Phase 3 done ({time.monotonic()-t0:.1f}s)", flush=True)

    # === Phase 4: Subgradient polish ===
    if time.monotonic() < deadline - 50:
        t0 = time.monotonic()
        print(f"\nPhase 4: Subgradient polish", flush=True)

        a4, J4 = subgradient_polish(a_cur, 5000, min(deadline - 30, time.monotonic() + 30))
        track(a4)
        print(f"  Phase 4 done: J={J4:.8f} ({time.monotonic()-t0:.1f}s)", flush=True)

        if J4 < compute_J(a_cur):
            a_cur = a4.copy()

    # === Phase 5: Multi-length exploration ===
    if time.monotonic() < deadline - 40:
        t0 = time.monotonic()
        print(f"\nPhase 5: Multi-length exploration (base n={n0})", flush=True)

        for n_try in [n0 - 32, n0 + 32, n0 + 64]:
            if time.monotonic() > deadline - 25:
                break
            if n_try < 64:
                continue
            a_try = resample_seq(a_cur, n_try)
            a_try = np.maximum(a_try, 1e-12)
            a_try = a_try / a_try.sum()
            remaining = max(10, deadline - 25 - time.monotonic())
            a_try, J_try = lp_mirror_descent_core(a_try, 10000, time.monotonic() + remaining,
                                                   K=50, use_momentum=True,
                                                   momentum_init=0.3, step_init=0.4)
            track(a_try)
            print(f"  n={n_try}: J={J_try:.8f}", flush=True)

        print(f"Phase 5 done ({time.monotonic()-t0:.1f}s)", flush=True)

    # === Phase 6: Final fine-tuning on best ===
    if time.monotonic() < deadline - 5 and tracker_best is not None:
        t0 = time.monotonic()
        print(f"\nPhase 6: Final fine-tuning n={tracker_best.size}", flush=True)

        a_final = tracker_best.astype(np.float64).copy()
        a_final = np.maximum(a_final, 1e-12)
        a_final = a_final / a_final.sum()
        a_final, J_final = lp_mirror_descent_core(a_final, 50000, deadline - 3,
                                                   K=2 * a_final.size - 1,
                                                   use_momentum=False, step_init=0.2)
        track(a_final)
        print(f"  Phase 6 done: J={J_final:.8f} ({time.monotonic()-t0:.1f}s)", flush=True)

    # Final
    if tracker_best is not None:
        final_a = tracker_best
    else:
        final_a = a_cur

    print(f"\n{'='*60}", flush=True)
    print(f"FINAL: J={tracker_best_score:.8f} n={final_a.size} "
          f"total_time={time.monotonic()-t_start:.0f}s", flush=True)
    print(f"{'='*60}", flush=True)

    return [float(max(0.0, min(1000.0, x))) for x in final_a]
