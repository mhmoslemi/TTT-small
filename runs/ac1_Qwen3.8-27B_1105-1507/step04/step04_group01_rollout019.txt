def xcorr(w, a):
    """Compute cross-correlation: xcorr[m] = sum_j w[m+j] * a[j] for m=0,...,len(a)-1."""
    n = len(a)
    len_w = len(w)
    # Use FFT: xcorr[m] = sum_j w[m+j] * a[j]

**Alternative: Direct QP-based approach**

At each iteration:
1. Compute current C = conv(a,a) and identify the top m lags.
2. Linearize C(k) around the current a: `C(k) ≈ C(k)_current + 2 * sum_i a[k-i]_current * da[i]`.

`sum(da) = 0`
`a + da >= 0`

                A_ub[k, i] = 2.0 * a[j]
        A_ub[k, n] = -1.0
    b_ub = -C0
    
    # A_eq: sum_i da[i] = 0
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.zeros(1)
    
    # Bounds: da[i] in [-a[i], inf], t in [0, inf]
    bounds = [(-a[i], None) for i in range(n)] + [(0.0, None)]
    
    try:
        res = linprog(c_vec, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': time_limit})
        if res.status == 0:
            da = res.x[:n]
            t_opt = res.x[n]
            
            # Line search
            best_a = a.copy()
            best_obj = compute_obj(a)
            for alpha in [1.0, 0.7, 0.5, 0.3, 0.1, 0.05]:
                a_try = a + alpha * da
                a_try = np.maximum(a_try, 0)
                s_try = np.sum(a_try)
                if s_try < 1e-10:
                    continue
                a_try = a_try / s_try
                obj_try = compute_obj(a_try)
                if obj_try < best_obj:
                    best_obj = obj_try
                    best_a = a_try.copy()
            
            return best_a, best_obj
    except Exception:
        pass
    
    return a, compute_obj(a)
```

Let me also think about whether the linearized LP approach has any theoretical guarantees. It's essentially a "linearized" version of the original non-convex problem. At each step, we solve an LP relaxation of the problem and take a step. This is similar to the "iterative reweighted" or "majorize-minimize" approach.

The key question is: does it converge to a good local minimum? I believe it should, because:
- The linearization is exact at the current point.
- The line search ensures we don't overshoot.
- The method is essentially a "first-order" method with exact line search in the linearized direction.

The main risk is that it gets stuck in a "shallow" local minimum where the linearization is a poor approximation. This is where the subgradient or L-BFGS-B steps come in to provide a different search direction.

```python
"""
Multi-strategy optimizer for minimizing 2*n*max(conv(a,a))/(sum(a))^2.
Uses linearized LP direction (cutting plane), L-BFGS-B with softmax,
LP alternating with dual weights, and subgradient methods.
Explores multiple sequence lengths around the parent.
"""

import numpy as np
from scipy.optimize import minimize, linprog
from scipy.fft import rfft, irfft
import time


def compute_conv(a):
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    conv = irfft(fa * fa, size)
    return conv[:2 * n - 1]


def compute_obj(a):
    n = len(a)
    if n == 0:
        return np.inf
    conv = compute_conv(a)
    max_b = float(np.max(conv))
    sum_a = float(np.sum(a))
    if sum_a < 0.01:
        return np.inf
    return 2.0 * n * max_b / (sum_a * sum_a)


def build_toeplitz(a):
    n = len(a)
    T = np.zeros((2 * n - 1, n), dtype=np.float64)
    for j in range(n):
        k_start = j
        k_end = min(j + n, 2 * n - 1)
        T[k_start:k_end, j] = a[:k_end - k_start]
    return T


def softmax(z):
    z = z - np.max(z)
    e = np.exp(z)
    return e / np.sum(e)


def lfbs_obj_grad(z, tau):
    """Softmax soft-max objective and gradient w.r.t. z."""
    n = len(z)
    a = softmax(z)
    conv = compute_conv(a)
    cmax = float(np.max(conv))
    shifted = np.clip((conv - cmax) / tau, -500.0, 500.0)
    exp_conv = np.exp(shifted)
    Z = float(np.sum(exp_conv))
    soft_max_val = cmax + tau * np.log(Z)
    w = exp_conv / Z
    
    w_rev = w[::-1]
    size = 1
    while size < n + len(w) - 1:
        size <<= 1
    fw = rfft(w_rev, size)
    fa = rfft(a, size)
    conv_fw = irfft(fw * fa, size)[:n + len(w) - 1]
    
    # grad_a[i] = 2 * sum_j w[i+j] * a[j] = 2 * xcorr(w, a)[i]
    # xcorr(w,a)[i] = conv(w, flip(a))[i + n - 1]
    # But I used conv(w_rev, a) which is different.
    # Let me redo: grad_a[i] = 2 * sum_{k} w[k] * a[k-i] * [0<=k-i<n]
    # = 2 * sum_j w[i+j] * a[j] (where j = k-i, k = i+j)
    # = 2 * xcorr(w, a)[i]
    
    # xcorr(w, a)[i] = conv(w, flip(a))[i + n - 1]
    flip_a = a[::-1]
    size2 = 1
    while size2 < len(w) + n - 1:
        size2 <<= 1
    fw2 = rfft(w, size2)
    fa2 = rfft(flip_a, size2)
    conv_w_fa = irfft(fw2 * fa2, size2)
    grad_a = 2.0 * conv_w_fa[n-1:n-1+n]
    
    mean_grad = float(np.dot(a, grad_a))
    grad_z = a * (grad_a - mean_grad)
    
    return soft_max_val, grad_z


def optimize_lfbs(a_init, tau_list, max_iter=200, time_budget=20.0):
    n = len(a_init)
    a = a_init.copy()
    s = np.sum(a)
    if s < 1e-10:
        a = np.ones(n) / n
    else:
        a = a / s
    a = np.maximum(a, 1e-30)
    z = np.log(a)
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    
    for tau in tau_list:
        if time.monotonic() - start > time_budget:
            break
        
        def func_and_grad(z_val):
            return lfbs_obj_grad(z_val, tau)
        
        try:
            result = minimize(func_and_grad, z, jac=True, method='L-BFGS-B',
                              options={'maxiter': max_iter, 'ftol': 1e-14, 'gtol': 1e-8})
            z_new = result.x
            a_new = softmax(z_new)
            a_new = np.maximum(a_new, 0)
            s_new = np.sum(a_new)
            if s_new < 1e-10:
                continue
            a_new = a_new / s_new
            obj = compute_obj(a_new)
            if obj < best_obj:
                best_obj = obj
                best_a = a_new.copy()
                z = z_new
        except Exception:
            continue
    
    return best_a, best_obj


def linearized_lp_step(a, time_limit=5):
    """One step: linearize conv, solve LP for direction, line search."""
    n = len(a)
    C0 = compute_conv(a)
    
    T = build_toeplitz(a) * 2.0
    A_ub = np.hstack([T, -np.ones((2*n-1, 1))])
    b_ub = -C0
    
    c_vec = np.zeros(n + 1)
    c_vec[-1] = 1.0
    
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.zeros(1)
    
    bounds = [(-a[i], None) for i in range(n)] + [(0.0, None)]
    
    try:
        res = linprog(c_vec, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': time_limit})
        if res.status == 0:
            da = res.x[:n]
            best_a = a.copy()
            best_obj = compute_obj(a)
            for alpha in [1.0, 0.8, 0.6, 0.5, 0.3, 0.1, 0.05]:
                a_try = a + alpha * da
                a_try = np.maximum(a_try, 0)
                s_try = np.sum(a_try)
                if s_try < 1e-10:
                    continue
                a_try = a_try / s_try
                obj_try = compute_obj(a_try)
                if obj_try < best_obj:
                    best_obj = obj_try
                    best_a = a_try.copy()
            return best_a, best_obj
    except Exception:
        pass
    
    return a, compute_obj(a)


def lp_alternating_step(a, time_limit=5):
    """LP alternating: fix a, find g, blend."""
    n = len(a)
    conv = compute_conv(a)
    M = float(np.max(conv))
    
    T = build_toeplitz(a)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    
    try:
        res = linprog(c, A_ub=T, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': time_limit})
        if res.status == 0:
            g = np.maximum(res.x, 0)
            s = float(np.sum(g))
            if s > 1e-10:
                g = g / s
                best_a = a.copy()
                best_obj = compute_obj(a)
                for t in [1.0, 0.7, 0.5, 0.3, 0.1]:
                    a_try = (1 - t) * a + t * g
                    a_try = np.maximum(a_try, 0)
                    s_try = np.sum(a_try)
                    if s_try < 1e-10:
                        continue
                    a_try = a_try / s_try
                    obj_try = compute_obj(a_try)
                    if obj_try < best_obj:
                        best_obj = obj_try
                        best_a = a_try.copy()
                return best_a, best_obj
    except Exception:
        pass
    
    return a, compute_obj(a)


def subgradient_step(a, tau, w_type='softmax'):
    """One subgradient step with given smoothing."""
    n = len(a)
    conv = compute_conv(a)
    cmax = float(np.max(conv))
    
    if w_type == 'softmax':
        shifted = np.clip((conv - cmax) / tau, -500.0, 500.0)
        exp_conv = np.exp(shifted)
        w = exp_conv / np.sum(exp_conv)
    else:
        w = np.zeros(len(conv))
        top_k = np.argsort(conv)[-max(1, len(conv)//10):]
        w[top_k] = 1.0 / len(top_k)
    
    # grad_a[i] = 2 * xcorr(w, a)[i] = 2 * conv(w, flip(a))[i + n - 1]
    flip_a = a[::-1]
    size = 1
    while size < len(w) + n - 1:
        size <<= 1
    fw = rfft(w, size)
    fa = rfft(flip_a, size)
    conv_w_fa = irfft(fw * fa, size)
    grad_a = 2.0 * conv_w_fa[n-1:n-1+n]
    
    return grad_a


def subgradient_optimize(a, max_iters=1000, time_budget=15.0, step=0.05, tau_start=5.0, tau_end=50.0):
    n = len(a)
    a = a.copy().astype(np.float64)
    s = np.sum(a)
    if s < 1e-10:
        a = np.ones(n) / n
    else:
        a = a / s
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    stall = 0
    current_step = step
    
    for it in range(max_iters):
        if time.monotonic() - start > time_budget:
            break
        
        frac = it / max(1, max_iters)
        tau = tau_start + (tau_end - tau_start) * frac
        
        grad_a = subgradient_step(a, tau)
        if grad_a is None:
            break
        
        # Center the gradient (since we're on the simplex)
        grad_centered = grad_a - np.mean(grad_a)
        
        a_new = a - current_step * grad_centered
        a_new = np.maximum(a_new, 0)
        s_new = np.sum(a_new)
        if s_new > 1e-10:
            a_new = a_new / s_new
        else:
            a_new = a.copy()
        
        a = a_new
        obj = compute_obj(a)
        
        if obj < best_obj - 1e-12:
            best_obj = obj
            best_a = a.copy()
            stall = 0
        else:
            stall += 1
            if stall > 50:
                current_step *= 0.8
                if current_step < 1e-8:
                    break
                stall = 0
        
        if it > 50:
            current_step *= 0.997
    
    return best_a


def resample_sequence(a, new_n):
    n = len(a)
    if n == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    return np.interp(x_new, x_old, a)


def random_init(n, seed):
    rng = np.random.default_rng(seed)
    k = np.arange(n)
    choice = rng.integers(0, 6)
    
    if choice == 0:
        f = rng.uniform(0.5, 3.0)
        phase = rng.uniform(0, 2 * np.pi)
        a = 1.0 + 0.8 * np.cos(2 * np.pi * f * k / n + phase)
    elif choice == 1:
        f1 = rng.uniform(0.3, 2.0)
        f2 = rng.uniform(2.0, 6.0)
        a = 1.0 + 0.4 * np.cos(2 * np.pi * f1 * k / n) + 0.3 * np.cos(2 * np.pi * f2 * k / n + rng.uniform(0, 2*np.pi))
    elif choice == 2:
        a = rng.random(n)
        win = max(3, n // 15)
        kernel = np.ones(win) / win
        a = np.convolve(a, kernel, mode='same')
    elif choice == 3:
        a = np.ones(n)
        for _ in range(rng.integers(1, 4)):
            f = rng.uniform(0.5, 5.0)
            amp = rng.uniform(0.1, 0.4)
            a += amp * np.cos(2 * np.pi * f * k / n + rng.uniform(0, 2 * np.pi))
    elif choice == 4:
        nblocks = rng.integers(2, 6)
        a = np.ones(n)
        for b in range(nblocks):
            start = b * n // nblocks
            end = (b + 1) * n // nblocks
            a[start:end] = rng.uniform(0.3, 1.5)
    else:
        a = 1.0 + 0.5 * np.sin(2 * np.pi * rng.uniform(0.5, 2.0) * k / n)
    
    a = np.maximum(a, 0.01)
    s = np.sum(a)
    return a / s
