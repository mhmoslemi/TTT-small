"""
Optimized LP mirror descent with adaptive constraints for minimizing
J(a) = 2*n*max(conv(a,a))/sum(a)^2.
Key improvements over previous:
1. Adaptive constraint subset (only active lags in LP)
2. Longer runs with better step size schedules
3. Dual-guided step size adaptation
4. Efficient restart strategy

import numpy as np
from scipy.optimize import linprog
from scipy.sparse import lil_matrix, csr_matrix
import time

def fast_conv(a):
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]

def compute_J(a):
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))

def build_A_sparse(n, a, lags=None):
    if lags is None:
        lags = range(2 * n - 1)
    lags = list(lags)
    rows = []
    cols = []
    vals = []
    for idx, k in enumerate(lags):
        j_min = max(0, k - n + 1)
        j_max = min(n - 1, k)
        for j in range(j_min, j_max + 1):
            rows.append(idx)
            cols.append(j)
            vals.append(a[k - j])
    return csr_matrix((vals, (rows, cols)), shape=(len(lags), n)), lags

def lp_solve(a, M, active_lags=None, n_constraints=None):
    """Solve LP: max sum(g) s.t. (a*g)_k ≤ M for active k, g ≥ 0."""
    n = len(a)
    all_lags = list(range(2 * n - 1))
    
    if active_lags is None:
        # Use all lags
        A, lags_used = build_A_sparse(n, a)
    else:
        # Use only active lags, or top n_constraints
        if n_constraints is not None and len(active_lags) > n_constraints:
            active_lags = active_lags[:n_constraints]
        A, lags_used = build_A_sparse(n, a, active_lags)
    
    c = -np.ones(n)
    b_ub = np.full(len(lags_used), M)
    bounds = [(0, None)] * n
    
    try:
        r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
        if r.success and r.x is not None:
            return np.maximum(r.x, 0)
    except Exception:
        pass
    return None

def lp_solve_topk(a, M, b, top_k=80):
    """Solve LP: max sum(g) s.t. (a*g)_k <= M for top-k lags, g >= 0."""
    n = len(a)
    # Find top-k lags by b value
    if top_k < 2 * n - 1:
        top_idx = np.argsort(b)[::-1][:top_k]
    else:
        top_idx = np.arange(2 * n - 1)

    rows = []
    cols = []
    vals = []
    for idx, k in enumerate(top_idx):
        j_min = max(0, k - n + 1)
        j_max = min(n - 1, k)
        for j in range(j_min, j_max + 1):
            rows.append(idx)
            cols.append(j)
            vals.append(a[k - j])

    if not rows:
        return None

    A = csr_matrix((vals, (rows, cols)), shape=(len(top_idx), n))
    c = -np.ones(n)
    b_ub = np.full(len(top_idx), M)
    bounds = [(0, None)] * n

    try:
        r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
        if r.success and r.x is not None:
            return np.maximum(r.x, 0)
    except Exception:
        pass
    return None


def lp_mirror_descent(a, max_iters, deadline, init_step=0.15,
                      top_k=80, use_multiplicative=False,
                      decay_every=2000, decay_factor=0.92,
                      restart_after=300):
    """LP mirror descent with adaptive top-K constraints."""
    a = np.maximum(a.astype(np.float64), 1e-10)
    a = a / a.sum()
    n = len(a)
    current_J = compute_J(a)
    best_J = current_J
    best_a = a.copy()
    no_improve = 0
    t = init_step
    it = 0

    while it < max_iters:
        if time.monotonic() > deadline:
            break

        b = fast_conv(a)
        M = float(b.max())

        g = lp_solve_topk(a, M, b, top_k=top_k)
        if g is None:
            break
        g_sum = g.sum()
        if g_sum < 1e-10:
            break
        g = g / g_sum

        # Compute trial
        if use_multiplicative:
            ratio = g / np.maximum(a, 1e-15)
            a_trial = a * np.power(ratio, t)
        else:
            a_trial = (1.0 - t) * a + t * g

        a_trial = np.maximum(a_trial, 0.0)
        s = a_trial.sum()
        if s < 1e-10:
            t *= 0.5
            it += 1
            continue
        a_trial = a_trial / s
        J_trial = compute_J(a_trial)

        if J_trial < current_J - 1e-15:
            a = a_trial
            current_J = J_trial
            if current_J < best_J:
                best_J = current_J
                best_a = a.copy()
                no_improve = 0
            else:
                no_improve += 1
        else:
            no_improve += 1
            t *= 0.75

        it += 1

        # Decay step size periodically
        if it % decay_every == 0:
            t = max(t * decay_factor, 0.005)

        # Restart if stuck
        if no_improve >= restart_after:
            t = min(t * 1.5, init_step)
            no_improve = 0

    return best_a, best_J


def subgradient_polish(a, max_iters, deadline, momentum_beta=0.7):
    """Subgradient descent with momentum on the simplex."""
    a = np.maximum(a.astype(np.float64), 1e-10)
    a = a / a.sum()
    n = len(a)
    best_M = fast_conv(a).max()
    best_a = a.copy()
    momentum = np.zeros_like(a)

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        b = fast_conv(a)
        M = float(b.max())

        # Find active set (near-max lags)
        tol = 0.02 * M
        active = np.where(b >= M - tol)[0]
        if len(active) == 0:
            active = np.array([np.argmax(b)])

        # Compute subgradient: average over active lags
        grad = np.zeros_like(a)
        for k in active:
            j_min = max(0, k - n + 1)
            j_max = min(n - 1, k)
            for j in range(j_min, j_max + 1):
                grad[j] += 2.0 * a[k - j]
        grad /= len(active)

        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-15:
            break

        step = 0.5 / (gnorm * np.sqrt(it + 1.0))
        momentum = momentum_beta * momentum + (1 - momentum_beta) * grad / gnorm
        a_new = a - step * momentum
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s < 1e-10:
            break
        a_new = a_new / s

        new_M = fast_conv(a_new).max()
        if new_M < best_M:
            best_M = new_M
            best_a = a_new.copy()
        a = a_new

    return best_a


def project_simplex(v):
    """Project v onto the simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if v.sum() < 1e-15:
        return np.ones(n) / n
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - 1))[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0)


def structured_perturb(a, rng, ptype):
    """Apply structured perturbation."""
    n = len(a)
    t_arr = np.arange(n, dtype=np.float64) / n

    if ptype == 'shift':
        s = int(rng.integers(1, max(2, n // 8)))
        a_p = np.roll(a, s)
    elif ptype == 'cosine':
        freq = rng.uniform(1, 6)
        amp = rng.uniform(0.03, 0.12)
        phase = rng.uniform(0, 2 * np.pi)
        a_p = a * (1 + amp * np.cos(2 * np.pi * freq * t_arr + phase))
    elif ptype == 'smooth':
        k = int(rng.integers(3, max(4, n // 12)))
        kernel = np.ones(k) / k
        a_p = np.convolve(a, kernel, mode='same')
    elif ptype == 'taper':
        edge = int(rng.integers(3, max(4, n // 10)))
        window = np.ones(n)
        for i in range(edge):
            frac = (i + 1) / edge
            window[i] = frac
            window[n - 1 - i] = frac
        a_p = a * window
    else:
        a_p = a + rng.normal(0, 0.02, n)

    a_p = np.maximum(a_p, 0.001)
    s = a_p.sum()
    if s > 1e-8:
        a_p = a_p / s
    return a_p


def resample_seq(src, dest_n):
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size, endpoint=False)
    t_new = np.linspace(0, 1, dest_n, endpoint=False)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)

                                   init_step=0.06, top_k=120,
                                   use_multiplicative=False,
                                   decay_every=4000, decay_factor=0.96,
                                   restart_after=500)
        tracker.update(a3)
        print(f"  J={J3:.8f} ({time.monotonic()-t0:.1f}s)", flush=True)
        if J3 < current_best_J:
            current_best = a3.copy()
            current_best_J = J3

    # === Phase 4: Subgradient polish ===
    if time.monotonic() < deadline - 350:
        t0 = time.monotonic()
        print(f"\nPhase 4: Subgradient polish (n={current_best_n})", flush=True)
        dl = min(deadline - 300, time.monotonic() + 120)
        a4 = subgradient_polish(current_best, 999999, dl, momentum_beta=0.6)
        tracker.update(a4)
        J4 = compute_J(a4)
        print(f"  J={J4:.8f} ({time.monotonic()-t0:.1f}s)", flush=True)
        if J4 < current_best_J:
            current_best = a4.copy()
            current_best_J = J4

    # === Phase 5: Long LP with very small step ===
    if time.monotonic() < deadline - 200:
        t0 = time.monotonic()
        print(f"\nPhase 5: LP long polish (n={current_best_n})", flush=True)
        dl = min(deadline - 150, time.monotonic() + 180)
        a5, J5 = lp_mirror_descent(current_best, 999999, dl,
                                   init_step=0.03, top_k=150,
                                   use_multiplicative=False,
                                   decay_every=5000, decay_factor=0.97,
                                   restart_after=600)
        tracker.update(a5)
        print(f"  J={J5:.8f} ({time.monotonic()-t0:.1f}s)", flush=True)
        if J5 < current_best_J:
            current_best = a5.copy()
            current_best_J = J5

    # === Phase 6: Perturbation + LP cycles ===
    if time.monotonic() < deadline - 120:
        t0 = time.monotonic()
        print(f"\nPhase 6: Perturb + LP cycles (n={current_best_n})", flush=True)
        ptypes = ['cosine', 'shift', 'smooth', 'cosine', 'taper']
        per_time = max(8, min(25, (deadline - 100 - time.monotonic()) / len(ptypes) / 2))

        for i, pt in enumerate(ptypes):
            if time.monotonic() > deadline - 90:
                break
            a_pert = structured_perturb(current_best, rng, pt)
            dl_p = time.monotonic() + per_time
            a_pert, J_pert = lp_mirror_descent(a_pert, 99999, dl_p,
                                               init_step=0.1, top_k=80,
                                               use_multiplicative=False,
                                               decay_every=1000, decay_factor=0.94,
                                               restart_after=200)
            tracker.update(a_pert)
            if J_pert < current_best_J:
                current_best = a_pert.copy()
                current_best_J = J_pert
                print(f"  [{pt}] J={J_pert:.8f}", flush=True)
        print(f"Phase 6 done ({time.monotonic()-t0:.1f}s)", flush=True)

    # === Phase 7: Try different lengths ===
    if time.monotonic() < deadline - 80:
        t0 = time.monotonic()
        print(f"\nPhase 7: Multi-length (base={current_best_n})", flush=True)
        base_n = current_best_n
        for n_try in [base_n // 2, base_n + 64, base_n + 128]:
            if time.monotonic() > deadline - 40:
                break
            if n_try < 64 or n_try > 2048:
                continue
            a_try = resample_seq(current_best, n_try)
            a_try = np.maximum(a_try, 1e-10)
            a_try = a_try / a_try.sum()
            dl_t = min(deadline - 30, time.monotonic() + 25)
            a_try, J_try = lp_mirror_descent(a_try, 99999, dl_t,
                                             init_step=0.1, top_k=60,
                                             use_multiplicative=False,
                                             decay_every=2000, decay_factor=0.95,
                                             restart_after=300)
            tracker.update(a_try)
            print(f"  n={n_try}: J={J_try:.8f}", flush=True)
            if J_try < current_best_J:
                current_best = a_try.copy()
                current_best_n = n_try
                current_best_J = J_try
        print(f"Phase 7 done ({time.monotonic()-t0:.1f}s)", flush=True)

    # === Phase 8: Final very long LP ===
    if time.monotonic() < deadline - 10 and tracker.best is not None:
        t0 = time.monotonic()
        print(f"\nPhase 8: Final LP (n={tracker.best.size})", flush=True)
        a_final = tracker.best.copy()
        a_final = np.maximum(a_final, 1e-10)
        a_final = a_final / a_final.sum()
        a_final, J_final = lp_mirror_descent(a_final, 999999, deadline - 5,
                                             init_step=0.02, top_k=100,
                                             use_multiplicative=False,
                                             decay_every=3000, decay_factor=0.97,
                                             restart_after=500)
        tracker.update(a_final)
        print(f"  J={J_final:.8f} ({time.monotonic()-t0:.1f}s)", flush=True)

    # Final
    if tracker.best is not None:
        final_a = tracker.best
    else:
        final_a = current_best

    print(f"\n{'='*50}", flush=True)
    print(f"FINAL: J={tracker.best_score:.8f} n={final_a.size} "
          f"time={time.monotonic()-tracker.t_start:.0f}s", flush=True)
    print(f"{'='*50}", flush=True)

    return [float(x) for x in final_a]
