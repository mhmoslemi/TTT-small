"""
Minimizes J(a) = 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Active-set LP mirror descent (much faster than full LP)

import numpy as np
from scipy.optimize import linprog, minimize
from scipy.sparse import lil_matrix, csr_matrix
import time
import sys


def fast_conv(a):
    """Full self-convolution via FFT, O(n log n)."""
    n = len(a)
    if n == 0:
        return np.array([0.0])
    s = 1
    while s < 2 * n - 1:
        s <<= 1
    fa = np.fft.rfft(a, s)
    return np.fft.irfft(fa * fa, s)[:2 * n - 1]


def compute_J(a):
    """Compute J(a) = 2*n*max(conv(a,a))/(sum(a))^2."""
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))


def conv_with_a(a, g):
    """Compute conv(a, g) efficiently."""
    return np.convolve(a, g, mode='full')

    """Solves LP: max sum(g) s.t. (a*g)_k <= M for k in active_set, g >= 0."""

        self.rows = []
        self.cols = []
        for k in range(2 * n - 1):
            j_start = max(0, k - n + 1)
            j_end = min(n - 1, k)
            for j in range(j_start, j_end + 1):
                self.rows.append(k)
                self.cols.append(j)
        self.rows = np.array(self.rows, dtype=np.int32)
        self.cols = np.array(self.cols, dtype=np.int32)
        # For each row k, the index range in the sparse arrays
        self.row_starts = np.zeros(2 * n, dtype=np.int32)
        for k in range(2 * n - 1):
            j_start = max(0, k - n + 1)
            j_end = min(n - 1, k)
            self.row_starts[k + 1] = self.row_starts[k] + (j_end - j_start + 1)
    
    def build_A_for_active(self, a, active_set):
        """Build sparse matrix for only the active constraints."""
        n = len(a)
        rows_sub = []
        cols_sub = []
        vals_sub = []
        for idx, k in enumerate(active_set):
            j_start = max(0, k - n + 1)
            j_end = min(n - 1, k)
            for j in range(j_start, j_end + 1):
                rows_sub.append(idx)
                cols_sub.append(j)
                vals_sub.append(a[k - j])
        return csr_matrix((vals_sub, (rows_sub, cols_sub)), shape=(len(active_set), n))
    
    def solve(self, a, M, active_set, max_iters=50):
        """
        Solve LP using active set method.
        Returns (g, final_active_set).
        """
        n = len(a)
        current_active = list(active_set)
        
        for iteration in range(max_iters):
            # Build reduced LP
            A = self.build_A_for_active(a, current_active)
            c = -np.ones(n)
            b_ub = np.full(len(current_active), M)
            bounds = [(0.0, None)] * n
            
            r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
            
            if not r.success or r.x is None:
                return None, current_active
            
            g = np.maximum(r.x, 0.0)
            
            # Check all constraints
            ag = conv_with_a(a, g)
            violations = np.where(ag > M * (1 + 1e-8))[0]
            
            if len(violations) == 0:
                return g, current_active
            
            # Add violating constraints (top few)
            violation_amounts = ag[violations] - M
            top_violations = violations[np.argsort(-violation_amounts)][:5]
            for k in top_violations:
                if k not in current_active:
                    current_active.append(int(k))
        
        return g, current_active


def lp_best_response_active(a, M, initial_active_set, n_lp_iters=100):
    """Solve LP best response using active set method."""
    n = len(a)
    solver = ActiveSetLP(n)
    g, final_active = solver.solve(a, M, initial_active_set, max_iters=n_lp_iters)
    if g is not None and g.sum() > 1e-10:
        return g / g.sum(), final_active
    return None, initial_active


def line_search(a, g, n_evals=15):
    """Golden section search for best t in [0,1]."""
    best_t = 0.0
    best_J = compute_J(a)
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    lo = 0.0
    hi = 1.0
    
    a_1 = g.copy()
    J_at_1 = compute_J(a_1)
    if J_at_1 < best_J:
        best_J = J_at_1
        best_t = 1.0
    
    for _ in range(n_evals):
        t1 = hi - gr * (hi - lo)
        t2 = lo + gr * (hi - lo)
        a1 = (1.0 - t1) * a + t1 * g
        a2 = (1.0 - t2) * a + t2 * g
        J1 = compute_J(a1)
        J2 = compute_J(a2)
        if J1 < J2:
            hi = t2
        else:
            lo = t1
        t_mid = (lo + hi) / 2.0
        a_mid = (1.0 - t_mid) * a + t_mid * g
        J_mid = compute_J(a_mid)
        if J_mid < best_J:
            best_J = J_mid
            best_t = t_mid
    
    return (1.0 - best_t) * a + best_t * g, best_J


def gp_mirror_descent(a, max_iters, deadline, tracker=None, n_line_evals=15,
                      stall_patience=30, lp_iters=50):
    """
    LP mirror descent with active set LP and golden section line search.
    """
    a = np.maximum(a.astype(np.float64), 1e-8)
    a = a / a.sum()
    current_J = compute_J(a)
    n = len(a)
    
    if tracker is not None:
        tracker.update(a)
    
    # Initial active set: center of convolution
    center = n - 1
    active_set = list(range(max(0, center - 3), min(2 * n - 1, center + 4)))
    
    stall_count = 0
    best_J_so_far = current_J
    best_a_so_far = a.copy()
    
    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        
        b = fast_conv(a)
        M = float(b.max())
        
        # Update active set based on current b
        # Use indices where b_k > 0.9 * M
        threshold = 0.9 * M
        near_active = np.where(b > threshold)[0]
        active_set = [int(k) for k in near_active]
        
        # Also always include the argmax
        max_idx = int(np.argmax(b))
        if max_idx not in active_set:
            active_set.append(max_idx)
        
        # Solve LP
        g, active_set = lp_best_response_active(a, M, active_set, n_lp_iters=lp_iters)
        if g is None:
            # Fallback: try with just the center
            active_set = [center]
            g, active_set = lp_best_response_active(a, M, active_set, n_lp_iters=lp_iters)
            if g is None:
                break
        
        # Line search
        a_new, J_new = line_search(a, g, n_evals=n_line_evals)
        
        if J_new < current_J - 1e-14:
            a = a_new
            current_J = J_new
            stall_count = 0
            if J_new < best_J_so_far:
                best_J_so_far = J_new
                best_a_so_far = a.copy()
                if tracker is not None:
                    tracker.update(a)
        else:
            # Try smaller steps
            found = False
            for t_try in [0.05, 0.02, 0.01, 0.1]:
                a_try = (1.0 - t_try) * a + t_try * g
                a_try = np.maximum(a_try, 0.0)
                s = a_try.sum()
                if s < 1e-10:
                    continue
                a_try = a_try / s
                J_try = compute_J(a_try)
                if J_try < current_J - 1e-14:
                    a = a_try
                    current_J = J_try
                    stall_count = 0
                    if J_try < best_J_so_far:
                        best_J_so_far = J_try
                        best_a_so_far = a.copy()
                        if tracker is not None:
                            tracker.update(a)
                    found = True
                    break
            if not found:
                stall_count += 1
                if stall_count >= stall_patience:
                    break
        
        if it % 20 == 0:
            if tracker is not None:
                tracker.update(a)
    
    if tracker is not None:
        tracker.update(best_a_so_far)
    return best_a_so_far


def smooth_conv_max_and_grad(a, beta):
    """Smooth max of conv(a,a) and its gradient."""
    n = len(a)
    b = fast_conv(a)
    b_max = float(b.max())
    shifted = beta * (b - b_max)
    shifted = np.clip(shifted, -500, 500)
    exp_shifted = np.exp(shifted)
    Z = float(exp_shifted.sum())
    if Z < 1e-300:
        p = np.zeros(2 * n - 1)
        p[np.argmax(b)] = 1.0
    else:
        p = exp_shifted / Z
    g_val = (b_max + np.log(max(Z, 1e-300))) / beta
    
    # Gradient: d/d a_i of max(b) ≈ sum_k p_k * a_{k-i}
    # conv(a, p_reversed) gives us sum_k p_k * a_{i-k} ... let me be careful
    # b_k = sum_j a_j * a_{k-j}
    # db_k / da_i = a_{k-i} (if 0 <= k-i < n)
    # So d(sum_k p_k * b_k) / da_i = sum_k p_k * a_{k-i}
    # This is the cross-correlation of a with p, evaluated at lag i
    # = (a * p_flipped)[n-1+i] ... let me think
    # sum_k p_k * a_{k-i}: let m = k-i, then k = m+i, sum_m p_{m+i} * a_m
    # This is (p_shifted * a) where p_shifted[m] = p[m+i]... 
    # Actually: sum_m a_m * p_{m+i} = (a * p)[i + 2n - 2 - ...] hmm
    # Let me just use: result[i] = sum_k p[k] * a[k-i] for valid k-i
    # This is cross-correlation: result = np.correlate(p, a, mode='full')[n-1:n-1+n]
    # Wait: np.correlate(p, a, 'full')[k] = sum_j p[j] * a[j+k-(n_a-1)]
    # Let me just compute it directly with FFT
    # sum_k p[k] * a[k-i] = sum_k' p[k'+i] * a[k'] where k' = k-i
    # = (p[i:] * a[:len(p)-i]).sum() ... this is getting complicated
    # Easiest: use the fact that it's a convolution
    # Let q = p[::-1] (reversed)
    # Then sum_k p[k] * a[k-i] = sum_k p[k] * a[k-i]
    # Let j = k-i: sum_j p[j+i] * a[j] = convolve(a, p_shifted_by_neg_i)
    # Actually the simplest: use FFT
    s = 1
    while s < 2 * n - 1:
        s <<= 1
    A_fft = np.fft.rfft(a, s)
    P_fft = np.fft.rfft(p, s)
    # sum_k p[k] * a[k-i] for each i
    # This is the cross-correlation of p and a
    # = ifft(conj(P_fft) * A_fft) ... let me verify
    # Cross-correlation: (p ⋆ a)[i] = sum_k p[k] * conj(a[k-i])
    # For real: = sum_k p[k] * a[k-i]
    # In FFT: = ifft(conj(FFT(p)) * FFT(a))
    corr = np.fft.irfft(np.conj(P_fft) * A_fft, s)
    # corr[i] = sum_k p[k] * a[k-i] (with circular convolution interpretation)
    # For our linear case, we need to extract the right part
    # Actually for linear cross-correlation:
    # (p ⋆ a)[i] = sum_{k=0}^{2n-2-i} p[k] * a[k-i] for i = 0, ..., n-1
    # Hmm, let me just use a simpler approach
    grad = 2.0 * corr[:n]  # Take the first n elements (for 0 <= i < n)
    # Actually I need to be more careful. Let me use the direct formula.
    # grad[i] = sum_k p[k] * a[k-i] where k-i is in [0, n-1]
    # = sum_{j=0}^{n-1} p[j+i] * a[j] (substituting j = k-i)
    # This is the i-th element of the cross-correlation
    # Using FFT: ifft(FFT(a) * conj(FFT(p)))[i] gives sum_j a[j] * p[j-i+shift]...
    # I'll just compute it directly for correctness
    # Actually the FFT approach should work:
    # Define extended arrays of length s (padded to power of 2, s >= 2n-1)
    # A_ext = [a, 0, 0, ..., 0] (length s)
    # P_ext = [p, 0, 0, ..., 0] (length s)
    # linear cross-correlation: ifft(conj(FFT(P_ext)) * FFT(A_ext))
    # This gives (p ⋆ a)[i] = sum_k p[k] * a[k-i] (linear, not circular)
    # for i = 0, 1, ..., s-1
    # We want i = 0, ..., n-1
    # But wait: for i > 0, a[k-i] is defined for k >= i, and p[k] for k up to 2n-2
    # So (p ⋆ a)[i] = sum_{k=i}^{2n-2} p[k] * a[k-i] = sum_{j=0}^{2n-2-i} p[j+i] * a[j]
    # For j up to n-1 (since a[j] = 0 for j >= n):
    # = sum_{j=0}^{n-1} p[j+i] * a[j] (where p[j+i] = 0 if j+i >= 2n-1)
    # This is exactly what we want!
    # So grad[i] = (p ⋆ a)[i] for i = 0, ..., n-1
    
    # The FFT computation:
    s2 = 1
    while s2 < 2 * n - 1 + n:  # need enough room
        s2 <<= 1
    A_ext = np.zeros(s2)
    A_ext[:n] = a
    P_ext = np.zeros(s2)
    P_ext[:2*n-1] = p
    corr_full = np.fft.irfft(np.conj(np.fft.rfft(P_ext, s2)) * np.fft.rfft(A_ext, s2), s2)
    grad = 2.0 * corr_full[:n]
    
    return g_val, grad


def F_and_grad(a, beta):
    """F(a) = 2n*smooth_max(conv(a,a))/(sum(a))^2 and gradient."""
    n = len(a)
    s = float(a.sum())
    if s < 1e-10:
        return 1e10, np.zeros(n)
    g_val, grad_g = smooth_conv_max_and_grad(a, beta)
    F_val = 2.0 * n * g_val / (s * s)
    grad_F = 2.0 * n * (s * grad_g - 2.0 * g_val) / (s ** 3)
    return F_val, grad_F


def gradient_polish(a, deadline, tracker=None, n_steps=200):
    """Polish using gradient descent on smoothed objective."""
    a = np.maximum(a.astype(np.float64), 1e-8)
    a = a / a.sum()
    current_J = compute_J(a)
    n = len(a)
    
    if tracker is not None:
        tracker.update(a)
    
    best_J = current_J
    best_a = a.copy()
    
    beta = 10.0
    lr = 0.01
    
    for step in range(n_steps):
        if time.monotonic() > deadline:
            break
        
        F_val, grad = F_and_grad(a, beta)
        
        # Project gradient to maintain sum(a) = 1 (approximately)
        # Subtract mean to maintain sum
        grad_centered = grad - grad.mean()
        
        # Gradient descent step
        a_new = a - lr * grad_centered
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s < 1e-10:
            break
        a_new = a_new / s
        
        J_new = compute_J(a_new)
        
        if J_new < current_J - 1e-15:
            a = a_new
            current_J = J_new
            # Decrease learning rate
            lr *= 0.99
            if J_new < best_J:
                best_J = J_new
                best_a = a.copy()
                if tracker is not None:
                    tracker.update(a)
        else:
            # Try smaller step
            lr *= 0.5
            if lr < 1e-8:
                break
        
        # Occasionally increase beta to sharpen
        if step % 50 == 0 and step > 0:
            beta *= 2.0
            beta = min(beta, 100.0)
    
    if tracker is not None:
        tracker.update(best_a)
    return best_a


def resample_seq(src, dest_n):
    """Resample 1D sequence to new length."""
    src = np.asarray(src, dtype=np.float6
4)
    if src.size == dest_n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    result = np.interp(t_new, t_old, src)
    return np.maximum(result, 0.0)


def generate_initializations(n, rng, parent=None):
    """Generate diverse initial sequences."""
    inits = []
    t_arr = np.linspace(0, 1, n, endpoint=False)
    
    if parent is not None and parent.size > 0:
        inits.append(resample_seq(parent, n))
        p = resample_seq(parent, n)
        p = p + rng.normal(0, 0.05, n) * p
        inits.append(np.maximum(p, 0.01))
    
    inits.append(np.ones(n))
    inits.append(rng.uniform(0.3, 1.0, n))
    
    # Frequency-domain inspired: try to make DFT as flat as possible
    for freq in [2, 3, 4, 5, 7, 11]:
        inits.append(1.0 + 0.3 * np.cos(2.0 * np.pi * freq * t_arr))
    
    inits.append(np.exp(-4.0 * (t_arr - 0.5) ** 2) + 0.05)
    inits.append(np.abs(np.sin(np.pi * t_arr)) + 0.15)
    inits.append(1.0 + 0.3 * np.cos(4.0 * np.pi * t_arr) + 0.1 * np.cos(8.0 * np.pi * t_arr))
    
    # Raised cosine
    inits.append(0.5 * (1.0 + np.cos(2.0 * np.pi * t_arr)))
    
    # Lee distance inspired
    inits.append(np.abs(np.sin(2.0 * np.pi * t_arr)) + np.abs(np.sin(3.0 * np.pi * t_arr)))
    
    return inits


class ScoreTracker:
    def __init__(self):
        self.best = None
        self.best_score = float('inf')
        self.t_start = time.monotonic()
    
    def update(self, a):
        a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
        if a.size == 0 or a.sum() < 0.01:
            return
        score = float(evaluate_sequence(a.tolist()))
        if np.isfinite(score) and score < self.best_score:
            self.best = a.copy()
            self.best_score = score
            elapsed = time.monotonic() - self.t_start
            print(f"  * NEW BEST: J={score:.8f} n={a.size} t={elapsed:.1f}s", flush=True)
    
    def report(self, label=""):
        if self.best is not None:
            print(f"  {label} best: J={self.best_score:.8f} n={self.best.size}", flush=True)
