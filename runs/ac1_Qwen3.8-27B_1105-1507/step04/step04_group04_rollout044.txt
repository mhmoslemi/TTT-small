def fft_conv(a):
    """Compute convolution a*a using FFT."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    conv = np.fft.irfft(fa * fa, size)
    return conv[:2 * n - 1]


def compute_obj(a):
    """Compute 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:

    conv = fft_conv(a)
    max_b = float(np.max(conv))
    sum_a = float(np.sum(a))
    if sum_a < 0.01:
        return np.inf
    return 2.0 * n * max_b / (sum_a * sum_a)

def coordinate_descent(a, time_budget=20.0):
    """
    Cyclic coordinate descent with golden section search.
    """
    n = len(a)
    a = a.copy().astype(np.float64)
    S = np.sum(a)
    if S < 1e-10:
        return a
    a = a / S
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    
    # Precompute base values
    def obj_with_change(a, i, x):
        """Compute obj if a[i] is set to x (rest unchanged)."""
        a_new = a.copy()
        a_new[i] = x
        return compute_obj(a_new)
    
    no_improve = 0
    
    while time.monotonic() - start < time_budget:
        improved = False
        for i in range(n):
            if time.monotonic() - start > time_budget:
                break
            
            current_val = a[i]
            # Search in a range around current value
            # The optimal x is probably within [0, 2*current_val + 1/n]
            lo = 0.0
            hi = 2.0 * current_val + 2.0 / n
            
            # Golden section search
            result = minimize_scalar(
                lambda x: obj_with_change(a, i, x),
                bounds=(lo, hi),
                method='bounded',
                options={'xatol': 1e-10}
            )
            
            if result.success and result.fun < best_obj - 1e-14:
                a[i] = result.x
                best_obj = result.fun
                best_a = a.copy()
                improved = True
        
        if not improved:
            no_improve += 1
            if no_improve >= 3:
                break
        else:
            no_improve = 0
    
    return best_a


def lp_pole_step(a, time_budget=30.0):
    """
    LP-based pole stepping: solve max sum(g) s.t. (a*g) <= M, g >= 0,
    then interpolate.
    """
    n = len(a)
    a = a.copy().astype(np.float64)
    S = np.sum(a)
    if S < 1e-10:
        return a
    a = a / S
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    
    stall = 0
    
    while time.monotonic() - start < time_budget - 5:
        conv = fft_conv(a)
        M = float(np.max(conv))
        
        # Build Toeplitz matrix
        A = np.zeros((2 * n - 1, n))
        for j in range(n):
            k_start = j
            k_end = min(j + n, 2 * n - 1)
            A[k_start:k_end, j] = a[:k_end - k_start]
        
        # Solve: max sum(g) s.t. A@g <= M, g >= 0
        c = -np.ones(n)
        b_ub = np.full(2 * n - 1, M)
        bounds = [(0.0, None)] * n
        
        try:
            res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                          options={'presolve': True, 'time_limit': 8})
            if res.status != 0:
                break
            g = np.maximum(res.x, 0)
            sum_g = np.sum(g)
            if sum_g < 1e-10:
                break
            g = g / sum_g
        except Exception:
            break
        
        # Line search for best interpolation
        best_t = 0.0
        best_t_obj = best_obj
        for t in [0.1, 0.2, 0.3, 0.5, 0.7, 1.0]:
            a_new = (1.0 - t) * a + t * g
            a_new = np.maximum(a_new, 0)
            s = np.sum(a_new)
            if s < 1e-10:
                continue
            a_new = a_new / s
            obj_new = compute_obj(a_new)
            if obj_new < best_t_obj:
                best_t_obj = obj_new
                best_t = t
        
        if best_t > 0 and best_t_obj < best_obj - 1e-13:
            a_new = (1.0 - best_t) * a + best_t * g
            a_new = np.maximum(a_new, 0)
            s = np.sum(a_new)
            if s > 1e-10:
                a_new = a_new / s
            a = a_new
            best_obj = best_t_obj
            best_a = a_new.copy()
            stall = 0
        else:
            stall += 1
            if stall > 10:
                break
    
    return best_a


def build_toeplitz(a):
    """Build Toeplitz matrix for convolution with a."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        k_start = j
        k_end = min(j + n, 2 * n - 1)
        A[k_start:k_end, j] = a[:k_end - k_start]
    return A


def random_init(n, seed):
    """Generate a random initialization."""
    rng = np.random.default_rng(seed)
    k = np.arange(n)
    choice = rng.integers(0, 8)
    
    if choice == 0:
        # Cosine
        f = rng.uniform(0.3, 3.0)
        phase = rng.uniform(0, 2 * np.pi)
        a = 1.0 + 0.7 * np.cos(2 * np.pi * f * k / n + phase)
    elif choice == 1:
        # Multi-freq
        a = np.ones(n)
        for _ in range(3):
            f = rng.uniform(0.3, 5.0)
            amp = rng.uniform(0.1, 0.4)
            a += amp * np.cos(2 * np.pi * f * k / n + rng.uniform(0, 2*np.pi))
    elif choice == 2:
        # Gaussian bump
        center = n * rng.uniform(0.2, 0.8)
        width = n * rng.uniform(0.1, 0.4)
        a = 1.0 + 0.5 * np.exp(-0.5 * ((k - center) / width) ** 2)
    elif choice == 3:
        # Smoothed random
        a = rng.random(max(5, n // 10))
        a = np.interp(k, np.arange(len(a)), a)
    elif choice == 4:
        # Raised cosine window
        a = 0.5 * (1 + np.cos(2 * np.pi * k / n))
    elif choice == 5:
        # Tapered
        a = np.sin(np.pi * (k + 0.5) / n)
    elif choice == 6:
        # Chirp
        f0 = rng.uniform(0.5, 1.5)
        f1 = rng.uniform(3.0, 8.0)
        t = k / n
        f = f0 + (f1 - f0) * t
        a = 1.0 + 0.4 * np.cos(2 * np.pi * np.cumsum(f) / n * n)
    else:
        # Product of windows
        a = 0.5 * (1 + np.cos(2 * np.pi * k / n))
        a *= 1.0 + 0.3 * np.cos(2 * np.pi * rng.uniform(1, 4) * k / n)
    
    a = np.maximum(a, 0)
    s = np.sum(a)
    if s < 1e-10:
        a = np.ones(n)
        s = n
    return a / s


def resample(a, new_n):
    """Resample sequence to new length."""
    n = len(a)
    if n == new_n:
        return a.copy()
    old_k = np.arange(n)
    new_k = np.linspace(0, n - 1, new_n)
    a_new = np.interp(new_k, old_k, a)
    a_new = np.maximum(a_new, 0)
    s = np.sum(a_new)
    if s > 1e-10:
        a_new = a_new / s
    return a_new
