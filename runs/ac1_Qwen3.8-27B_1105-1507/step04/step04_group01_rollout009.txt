"""
LP-alternating optimization with dual-guided subgradient for minimizing
the peak autocorrelation ratio 2*n*max(conv(a,a))/(sum(a))^2.
"""

import numpy as np
from scipy.optimize import minimize, linprog
from scipy.fft import rfft, irfft
import time


def compute_conv(a):
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    conv = irfft(fa * fa, size)
    return conv[:2 * n - 1]


def compute_obj(a):
    n = len(a)
    if n == 0:
        return np.inf
    conv = compute_conv(a)
    max_b = float(np.max(conv))
    sum_a = float(np.sum(a))
    if sum_a < 0.01:
        return np.inf
    return 2.0 * n * max_b / (sum_a * sum_a)


def fft_convolve(a, b):
    n, m = len(a), len(b)
    size = 1
    while size < n + m - 1:
        size <<= 1
    fa = rfft(a, size)
    fb = rfft(b, size)
    return irfft(fa * fb, size)[:n + m - 1]


def build_toeplitz_conv_matrix(a):
    """Build matrix T such that (T @ g) = (a * g) (full linear convolution)."""
    n = len(a)
    T = np.zeros((2 * n - 1, n))
    for j in range(n):
        lo = j
        hi = min(j + n, 2 * n - 1)
        T[lo:hi, j] = a[:hi - lo]
    return T


def solve_lp_minpeak_fixed_left(a, time_limit=30):
    """
    Solve: min M s.t. (a * g)[k] <= M for all k, sum(g) = 1, g >= 0, M >= 0.
    Returns (g, M, dual_y) or (None, None, None).
    dual_y is the dual probability distribution over lags (if available).
    """
    n = len(a)
    T = build_toeplitz_conv_matrix(a)  # (2n-1) x n
    
    # Variables: x = [g[0], ..., g[n-1], M], total n+1
    c = np.zeros(n + 1)
    c[-1] = 1.0  # minimize M
    
    # Inequality: (a*g)[k] - M <= 0 for each k
    A_ub = np.zeros((2 * n - 1, n + 1))
    A_ub[:, :n] = T
    A_ub[:, n] = -1.0
    b_ub = np.zeros(2 * n - 1)
    
    # Equality: sum(g) = 1
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])
    
    bounds = [(0.0, None)] * (n + 1)
    
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': time_limit})
        if res.status == 0:
            g = np.maximum(res.x[:n], 0)
            M = float(res.x[n])
            s = float(np.sum(g))
            if s > 1e-10:
                g = g / s
                # Re-estimate M
                conv_ag = fft_convolve(a, g)
                M = float(np.max(conv_ag))
                return g, M, None
    except Exception:
        pass
    return None, None, None


def solve_lp_minpeak(a, fixed_vec, time_limit=30):
    """
    Solve: min M s.t. (fixed_vec * x)[k] <= M, sum(x) = 1, x >= 0, M >= 0.
    Where x is the variable (either g or a).
    """
    return solve_lp_minpeak_fixed_left(fixed_vec, time_limit)


def lp_alternate(a, max_iters=50, time_budget=100.0, verbose=False):
    """
    Alternating LP: fix a → solve for g, fix g → solve for a.
    """
    n = len(a)
    a = a.copy().astype(np.float64)
    s = np.sum(a)
    if s < 1e-10:
        a = np.ones(n) / n
    else:
        a = a / s
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    
    for it in range(max_iters):
        if time.monotonic() - start > time_budget:
            break
        remaining = time_budget - (time.monotonic() - start)
        if remaining < 10:
            break
        
        # Step 1: Solve LP for g given a
        g, M_g, _ = solve_lp_minpeak_fixed_left(a, time_limit=min(30, remaining * 0.4))
        if g is None:
            break
        
        # Step 2: Solve LP for a' given g
        a_new, M_a, _ = solve_lp_minpeak_fixed_left(g, time_limit=min(30, remaining * 0.4))
        if a_new is None:
            break
        
        # Step 3: Line search - blend a with a_new
        improved = False
        for t in [1.0, 0.7, 0.5, 0.3, 0.2, 0.1]:
            candidate = (1.0 - t) * a + t * a_new
            candidate = np.maximum(candidate, 0)
            sc = np.sum(candidate)
            if sc < 1e-10:
                continue
            candidate = candidate / sc
            obj = compute_obj(candidate)
            if obj < best_obj:
                best_obj = obj
                best_a = candidate.copy()
                a = candidate
                improved = True
                break
        
        if not improved:
            # Try g directly
            candidate = g
            sc = np.sum(candidate)
            if sc > 1e-10:
                candidate = candidate / sc
                obj = compute_obj(candidate)
                if obj < best_obj:
                    best_obj = obj
                    best_a = candidate.copy()
                    a = candidate
                    improved = True
        
        if not improved:
            # Perturbation
            rng = np.random.default_rng(hash((it, n)) % 2**31)
            a = a * np.exp(rng.normal(0, 0.05, n))
            a = np.maximum(a, 0)
            s = np.sum(a)
            if s > 1e-10:
                a = a / s
    
    return best_a, best_obj


def softmax(z):
    z = z - np.max(z)
    e = np.exp(z)
    return e / np.sum(e)


def soft_max_objective_and_grad(z, tau):
    """Compute soft-max of conv(softmax(z), softmax(z)) and gradient w.r.t. z."""
    n = len(z)
    a = softmax(z)
    conv = compute_conv(a)
    cmax = float(np.max(conv))
    shifted = np.clip((conv - cmax) / tau, -500.0, 500.0)
    exp_conv = np.exp(shifted)
    Z = float(np.sum(exp_conv))
    soft_max_val = cmax + tau * np.log(Z)
    w = exp_conv / Z
    
    # grad_a[i] = 2 * sum_j w[i+j] * a[j]
    w_rev = w[::-1]
    conv_fw = fft_convolve(a, w_rev)
    grad_a = 2.0 * conv_fw[n - 1:2 * n - 1][::-1]
    
    mean_grad = float(np.dot(a, grad_a))
    grad_z = a * (grad_a - mean_grad)
    
    return soft_max_val, grad_z


def optimize_lfbs(a_init, tau_list, max_iter=200, time_budget=30.0):
    n = len(a_init)
    a = a_init.copy()
    s = np.sum(a)
    if s < 1e-10:
        a = np.ones(n) / n
    else:
        a = a / s
    a = np.maximum(a, 1e-30)
    z = np.log(a)
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    
    for tau in tau_list:
        if time.monotonic() - start > time_budget:
            break
        
        def func_and_grad(z_val):
            return soft_max_objective_and_grad(z_val, tau)
        
        try:
            result = minimize(func_and_grad, z, jac=True, method='L-BFGS-B',
                              options={'maxiter': max_iter, 'ftol': 1e-14, 'gtol': 1e-8})
            z = result.x
        except Exception:
            continue
        
        a_new = softmax(z)
        a_new = np.maximum(a_new, 0)
        s_new = np.sum(a_new)
        if s_new < 1e-10:
            continue
        a_new = a_new / s_new
        obj = compute_obj(a_new)
        if obj < best_obj:
            best_obj = obj
            best_a = a_new.copy()
    
    return best_a, best_obj


def random_init(n, seed):

    k = np.arange(n)
    choice = rng.integers(0, 12)
    
    if choice == 0:

    elif choice == 1:
        f1 = rng.uniform(0.3, 1.5)
        f2 = rng.uniform(2.0, 5.0)

    elif choice == 2:

        a = 0.5 * (1 + np.cos(2 * np.pi * k / (n - 1)))
    elif choice == 3:
        # Flat-top window
        a1, a2, a3, a4, a5 = 0.21557895, 0.41663158, 0.277263158, 0.083578947, 0.006947368
        t = 2 * np.pi * k / (n - 1) - np.pi
        a = a1 - a2 * np.cos(t) + a3 * np.cos(2*t) - a4 * np.cos(3*t) + a5 * np.cos(4*t)
    elif choice == 4:
        # Asymmetric: ramp up then flat
        mid = n // 2
        a = np.concatenate([np.linspace(0.1, 1.0, mid), np.ones(n - mid)])
    elif choice == 5:
        # Two bumps
        c1 = n * rng.uniform(0.2, 0.4)
        c2 = n * rng.uniform(0.6, 0.8)
        w = n * 0.1
        a = np.exp(-0.5 * ((k - c1) / w) ** 2) + np.exp(-0.5 * ((k - c2) / w) ** 2)
    elif choice == 6:
        # Blackman window
        a = 0.42 - 0.5 * np.cos(2 * np.pi * k / (n-1)) + 0.08 * np.cos(4 * np.pi * k / (n-1))
    elif choice == 7:
        # Random with smoothing
        a = rng.random(n)
        win = max(3, n // 30)
        kernel = np.ones(win) / win
        a = np.convolve(a, kernel, mode='same')
    elif choice == 8:
        # Anti-symmetric base + offset
        half = n // 2
        base = rng.uniform(0.5, 1.5, half)
        a = np.concatenate([base, base[::-1] * rng.uniform(0.3, 0.8)])
    elif choice == 9:
        # Chirp
        f0 = rng.uniform(0.5, 2.0)
        f1 = rng.uniform(2.0, 5.0)
        a = 1.0 + 0.4 * np.cos(2 * np.pi * (f0 * k / n + (f1-f0) * k**2 / (2*n**2)))
    elif choice == 10:
        # Geometric decay from center
        center = n / 2
        rate = rng.uniform(0.5, 2.0)
        a = np.exp(-rate * np.abs(k - center) / (n / 2))
    else:
        a = rng.uniform(0.3, 1.5, n)
    
    a = np.maximum(a, 0.01)
    s = np.sum(a)
    return a / s


def resample_sequence(a, new_n):
    n = len(a)
    if n == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0.01)

def polish_sequence(a, time_budget, verbose=False):
    """Intensive polishing combining LP and L-BFGS-B."""
    n = len(a)
    a = a.copy().astype(np.float64)
    start = time.monotonic()
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    round_num = 0
    
    while time.monotonic() - start < time_budget - 5:
        round_num += 1
        remaining = time_budget - (time.monotonic() - start)
        if remaining < 8:
            break
        
        # LP step
        if n <= 800 and remaining > 30:
            a_new, obj_new = lp_alternate(best_a, max_iters=10, 
                                           time_budget=min(remaining * 0.3, 30.0))
            if obj_new < best_obj:
                best_obj = obj_new
                best_a = a_new.copy()
                if verbose:
                    print(f"    Polish r{round_num} LP: {best_obj:.6f}")
        
        # L-BFGS-B
        remaining = time_budget - (time.monotonic() - start)
        if remaining > 15:
            tau = max(0.1 / n, 0.05 / n)
            a_new, obj_new = optimize_lfbs(best_a, [tau, tau * 0.5],
                                            max_iter=150,
                                            time_budget=min(remaining * 0.25, 20.0))
            if obj_new < best_obj:
                best_obj = obj_new
                best_a = a_new.copy()
                if verbose:
                    print(f"    Polish r{round_num} L-BFGS: {best_obj:.6f}")
        
        # Perturbation + quick LP
        remaining = time_budget - (time.monotonic() - start)
        if remaining > 20 and n <= 700:
            rng = np.random.default_rng(hash((round_num, n)) % 2**31)
            scale = rng.uniform(0.02, 0.08)
            a_pert = best_a * np.exp(rng.normal(0, scale, n))
            a_pert = np.maximum(a_pert, 0)
            s_pert = np.sum(a_pert)
            if s_pert > 1e-10:
                a_pert = a_pert / s_pert
                a_new, obj_new = lp_alternate(a_pert, max_iters=5,
                                               time_budget=min(remaining * 0.2, 15.0))
                if obj_new < best_obj:
                    best_obj = obj_new
                    best_a = a_new.copy()
    
    return best_a, best_obj
