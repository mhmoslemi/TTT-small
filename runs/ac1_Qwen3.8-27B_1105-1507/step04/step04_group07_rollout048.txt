def spectral_projection(a, alpha=0.5, n_iters=100):
    n = len(a)
    a = a.astype(np.float64).copy()
    for _ in range(n_iters):
        A = np.fft.rfft(a)
        mag = np.abs(A)
        phase = np.angle(A)
        # Compress magnitude
        mag_new = mag ** alpha  # or: mag / (1 + mag) or: 1 + log(mag + eps)
        A_new = mag_new * np.exp(1j * phase)
        a_new = np.fft.irfft(A_new, n)
        a_new = np.maximum(a_new, 0)
        s = a_new.sum()
        if s < 1e-10:
            break
        a_new /= s
        if np.allclose(a_new, a):
            break
        a = a_new
    return a
