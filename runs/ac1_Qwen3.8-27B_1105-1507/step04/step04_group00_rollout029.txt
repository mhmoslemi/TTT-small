def fast_conv(a):
    n = len(a)
    size = 1
    while size < 2*n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2*n-1]

def compute_J(a):

def fast_conv(a):
    """Full self-convolution via FFT."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]

            best_a = a.copy()
            best_conv_max = current_M
            no_improve_sweeps = 0
        else:
            no_improve_sweeps += 1

        if no_improve_sweeps > max_no_improve:
            break

    a = best_a.copy()
    a = np.maximum(a, 0)
    a_sum = a.sum()
    if a_sum > 0:
        a /= a_sum
    return a


def build_A_sparse(n, a):
    """Build sparse constraint matrix for LP: A[k,j] = a[k-j] if 0<=k-j<n."""
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        j_min = max(0, k - n + 1)
        j_max = min(n - 1, k)
        for j in range(j_min, j_max + 1):
            rows.append(k)
            cols.append(j)
            vals.append(a[k - j])
    return csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))


def lp_best_response(a, M):
    """LP: max sum(g) s.t. (a*g)[k] <= M for all k, g >= 0."""
    n = len(a)
    A = build_A_sparse(n, a)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    try:
        r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                    options={'time_limit': 30, 'presolve': True})
        if r.success and r.x is not None:
            g = np.maximum(r.x, 0)
            if g.sum() > 1e-15:
                return g
    except Exception:
        pass
    return None


def run_lp_mirror_descent(a, max_iters, deadline):
    """LP mirror descent with geometric line search."""
    a = np.maximum(a.astype(np.float64), 1e-8)
    a = a / a.sum()
    current_M = fast_conv(a).max()
    best_a = a.copy()
    best_M = current_M
    no_improve = 0
    prev_lp_time = 0.1

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        b_conv = fast_conv(a)
        M = float(b_conv.max())
        t0 = time.monotonic()
        g = lp_best_response(a, M)
        lp_time = time.monotonic() - t0
        prev_lp_time = 0.7 * prev_lp_time + 0.3 * lp_time

        if g is None:
            break
        g = np.maximum(g, 0.0)
        g_sum = g.sum()
        if g_sum < 1e-10:
            break
        g /= g_sum

        # Line search
        improved = False
        base_t = max(0.05, 0.5 * (1.0 - 0.5 * it / max(1, max_iters)))
        for t in [base_t, base_t * 0.5, base_t * 0.25, 0.02, 0.01, 0.005]:
            a_trial = np.maximum((1 - t) * a + t * g, 0)
            s = a_trial.sum()
            if s < 1e-10:
                continue
            a_trial /= s
            M_trial = fast_conv(a_trial).max()
            if M_trial < best_M - 1e-14:
                a = a_trial
                best_M = M_trial
                best_a = a.copy()
                improved = True
                break

        if not improved:
            no_improve += 1
            if no_improve > 30:
                break
        else:
            no_improve = 0

        # Check if remaining time is enough for another LP
        remaining = deadline - time.monotonic()
        if remaining < prev_lp_time * 2:
            break

    return best_a


def subgradient_step(a, tol_frac=1e-8):
    """Compute exact subgradient of max(conv(a,a))."""
    n = len(a)
    b = fast_conv(a)
    M = b.max()
    tol = tol_frac * M if M > 0 else 1e-10
    active = np.where(b >= M - tol)[0]

    grad = np.zeros(n)
    for k in active:
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        grad[lo:hi + 1] += 2.0 * a[k - hi:k - lo + 1][::-1]

    gnorm = np.linalg.norm(grad)
    if gnorm > 1e-15:
        grad /= gnorm

    return grad, b


def run_subgradient(a, max_iters, deadline, lr_init=0.15):
    """Subgradient descent with averaging."""
    a = np.maximum(a.astype(np.float64), 1e-8)
    a = a / a.sum()
    n = len(a)

    best_a = a.copy()
    best_M = fast_conv(a).max()
    running_sum = np.zeros(n)
    count = 0

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        lr = lr_init / (it + 1) ** 0.5
        grad, b = subgradient_step(a)
        M = b.max()

        if M < best_M * (1 - 1e-9):
            best_M = M
            best_a = a.copy()

        a_new = a - lr * grad
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s < 1e-10:
            a_new = a.copy()
        else:
            a_new = a_new / s

        running_sum += a_new
        count += 1
        a = a_new

    if count > 10:
        a_avg = running_sum / count
        a_avg = np.maximum(a_avg, 0.0)
        s = a_avg.sum()
        if s > 1e-10:
            a_avg /= s
            M_avg = fast_conv(a_avg).max()
            if M_avg < best_M:
                best_a = a_avg

    return best_a


def resample_seq(src, dest_n):
    """Resample 1D sequence to new length via linear interpolation."""
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)


def pairwise_coord_descent(a, max_sweeps, deadline, seed=0):
    """Coordinate descent on pairs of adjacent elements."""
    n = len(a)
    a = np.maximum(a.copy().astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-10:
        a = a / a.sum() if a.sum() > 0 else np.ones(n)
        s = a.sum()

    conv = fast_conv(a)
    best_obj = 2.0 * n * conv.max() / (s * s)
    best_a = a.copy()
    rng = np.random.default_rng(seed)

    for sweep in range(max_sweeps):
        if time.monotonic() > deadline:
            break

        for i in range(0, n - 1, 2):
            old0, old1 = a[i], a[i + 1]
            if old0 < 1e-10 and old1 < 1e-10:
                continue

            # Try a small 2D grid
            r0 = np.array([0.0, 0.5, 1.0, 1.5]) if old0 > 1e-10 else np.array([0.0, 1e-3, 1e-2])
            r1 = np.array([0.0, 0.5, 1.0, 1.5]) if old1 > 1e-10 else np.array([0.0, 1e-3, 1e-2])

            best_x0, best_x1 = old0, old1
            best_local = 2.0 * n * conv.max() / (s * s)

            # Pre-slices for incremental update
            lo0 = max(0, i)
            hi0 = min(2 * n - 2, i + n - 1)
            lo1 = max(0, i + 1)
            hi1 = min(2 * n - 2, i + 1 + n - 1)
            a_slice0 = np.array(a[lo0 - i:hi0 - i + 1], copy=True)
            a_slice1 = np.array(a[lo1 - (i + 1):hi1 - (i + 1) + 1], copy=True)

            # Fix a_slice0[i] and a_slice1[i-1] placeholders (they reference 'old' values)
            a_slice0[i - lo0 + i - lo0]  # a[i] is at position i in original, which is index (i - (lo0-i)) = 2i - lo0 in slice
            # Actually: a_slice0[j] = a[lo0 - i + j] = a[j] for j in [lo0-i, hi0-i]
            # a[i] is at index i - (lo0 - i) = 2i - lo0 in a_slice0
            idx0_in_slice0 = i - (lo0 - i)
            idx1_in_slice1 = (i + 1) - (lo1 - (i + 1))

            for x0 in r0 * old0 if old0 > 1e-10 else r0:
                for x1 in r1 * old1 if old1 > 1e-10 else r1:
                    x0 = float(x0)
                    x1 = float(x1)
                    if x0 < 0:
                        x0 = 0.0
                    if x1 < 0:
                        x1 = 0.0

                    d0 = x0 - old0
                    d1 = x1 - old1

                    if abs(d0) < 1e-12 and abs(d1) < 1e-12:
                        continue

                    # Apply updates
                    if abs(d0) > 1e-12:
                        # Update a_slice0[idx0_in_slice0] = x0 for correct computation
                        temp = a_slice0[idx0_in_slice0]
                        a_slice0[idx0_in_slice0] = x0
                        conv[lo0:hi0 + 1] += 2.0 * d0 * a_slice0
                        a_slice0[idx0_in_slice0] = temp
                        kq0 = 2 * i
                        if kq0 <= 2 * n - 2:
                            conv[kq0] += d0 * d0

                    if abs(d1) > 1e-12:
                        temp = a_slice1[idx1_in_slice1]
                        a_slice1[idx1_in_slice1] = x1
                        conv[lo1:hi1 + 1] += 2.0 * d1 * a_slice1
                        a_slice1[idx1_in_slice1] = temp
                        kq1 = 2 * (i + 1)
                        if kq1 <= 2 * n - 2:
                            conv[kq1] += d1 * d1

                    new_s = s - old0 - old1 + x0 + x1
                    if new_s < 1e-10:
                        new_obj = float('inf')
                    else:
                        new_obj = 2.0 * n * conv.max() / (new_s * new_s)

                    if new_obj < best_local:
                        best_local = new_obj
                        best_x0, best_x1 = x0, x1

                    # Rollback
                    if abs(d0) > 1e-12:
                        conv[lo0:hi0 + 1] -= 2.0 * d0 * a_slice0
                        kq0 = 2 * i
                        if kq0 <= 2 * n - 2:
                            conv[kq0] -= d0 * d0
                    if abs(d1) > 1e-12:
                        conv[lo1:hi1 + 1] -= 2.0 * d1 * a_slice1
                        kq1 = 2 * (i + 1)
                        if kq1 <= 2 * n - 2:
                            conv[kq1] -= d1 * d1

            # Apply best
            if best_x0 != old0 or best_x1 != old1:
                d0 = best_x0 - old0
                d1 = best_x1 - old1
                if abs(d0) > 1e-12:
                    conv[lo0:hi0 + 1] += 2.0 * d0 * a_slice0
                    kq0 = 2 * i
                    if kq0 <= 2 * n - 2:
                        conv[kq0] += d0 * d0
                if abs(d1) > 1e-12:
                    conv[lo1:hi1 + 1] += 2.0 * d1 * a_slice1
                    kq1 = 2 * (i + 1)
                    if kq1 <= 2 * n - 2:
                        conv[kq1] += d1 * d1
                a[i] = best_x0
                a[i + 1] = best_x1
                s = s - old0 - old1 + best_x0 + best_x1

        current_obj = 2.0 * n * conv.max() / (s * s)
        if current_obj < best_obj - 1e-12:
            best_obj = current_obj
            best_a = a.copy()

    a = best_a.copy()
    a = np.maximum(a, 0)
    if a.sum() > 0:
        a /= a.sum()
    return a


def generate_varied_initializations(n, rng, parent=None):
    """Generate diverse initial sequences."""
    inits = []
    t = np.linspace(0, 1, n, endpoint=False)

    if parent is not None:
        p = resample_seq(parent, n)
        p = p / p.sum() if p.sum() > 0 else p
        inits.append(p)
        inits.append(np.maximum(p + rng.normal(0, 0.05, n), 0.01))
        inits.append(np.maximum(p + rng.normal(0, 0.15, n), 0.01))

    inits.append(np.ones(n) / n)

    # Cosine variants
    for freq in [1, 2, 3, 4, 6]:
        for amp in [0.2, 0.4, 0.6]:
            inits.append(np.maximum(1.0 + amp * np.cos(2 * np.pi * freq * t), 0.01) / n)
            inits.append(np.maximum(1.0 + amp * np.sin(2 * np.pi * freq * t), 0.01) / n)

    # Gaussian
    for center in [0.3, 0.5, 0.7]:
        for width in [0.1, 0.2, 0.3]:
            inits.append(np.maximum(np.exp(-((t - center) / width) ** 2) + 0.1, 0.01) / n)

    # Smooth random
    ksize = max(5, n // 20)
    kernel = np.ones(ksize) / ksize
    for _ in range(3):
        smooth = np.convolve(rng.uniform(0, 1, n), kernel, mode='same')
        inits.append(np.maximum(smooth + 0.05, 0.01) / n)

    # V-shape
    inits.append(np.maximum(0.5 + 0.5 * np.abs(2 * t - 1), 0.01) / n)
    inits.append(np.maximum(1.0 - 0.5 * np.abs(2 * t - 1), 0.01) / n)

    return inits


def propose_candidate(seed=42, budget_s=1500, **kwargs):
    """
    Multi-phase search:
    Phase 1: Coordinate descent on parent + varied inits (n=512), 500s
    Phase 2: Try longer lengths (768, 1024), 300s
    Phase 3: LP mirror descent on best, 400s
    Phase 4: Subgradient + final LP polish, remaining time
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + float(budget_s) - 15.0
    tracker = ScoreTracker()

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        if p.ndim == 1 and p.size > 0 and np.all(np.isfinite(p)):
            parent = np.maximum(p, 0.0)
            tracker.update(parent)
            print(f"Parent loaded: n={parent.size}, J={tracker.best_score:.6f}", flush=True)
    except (NameError, AttributeError):
        pass

    if parent is None:
        parent = np.ones(256) / 256
        tracker.update(parent)

    current_best = parent.copy()
    current_best_n = len(current_best)

    # === PHASE 1: Coordinate descent, n = parent length ===
    p1_deadline = min(time.monotonic() + 500, deadline - 400)
    print(f"\n{'=' * 60}", flush=True)
    print(f"PHASE 1: Coordinate Descent n={current_best_n} (budget to {p1_deadline - time.monotonic():.0f}s)", flush=True)
    print(f"{'=' * 60}", flush=True)

    n_base = current_best_n
    inits = generate_varied_initializations(n_base, rng, current_best)

    for idx, a0 in enumerate(inits[:8]):  # Limit to 8 inits for speed
        if time.monotonic() > p1_deadline:
            break

        a0 = np.maximum(a0, 0)
        a0_sum = a0.sum()
        if a0_sum < 1e-10:
            continue
        a0 = a0 / a0_sum

        t0 = time.monotonic()
        a_opt = coord_descent(a0, max_sweeps=5000, deadline=p1_deadline,
                              random_order=True, seed=seed + idx)
        elapsed = time.monotonic() - t0
        J_val = compute_J(a_opt)
        print(f"  Init {idx}: J={J_val:.8f} n={n_base} ({elapsed:.1f}s)", flush=True)
        tracker.update(a_opt)
        if tracker.best_score < compute_J(current_best):
            current_best = tracker.best.copy()

    # === PHASE 1b: Pairwise coordinate descent on best ===
    if time.monotonic() < p1_deadline - 10:
        print(f"\n  Pairwise CD on best (n={len(tracker.best)})...", flush=True)
        t0 = time.monotonic()
        a_pw = pairwise_coord_descent(tracker.best, max_sweeps=2000,
                                      deadline=p1_deadline, seed=seed + 100)
        elapsed = time.monotonic() - t0
        J_val = compute_J(a_pw)
        print(f"  Pairwise: J={J_val:.8f} ({elapsed:.1f}s)", flush=True)
        tracker.update(a_pw)

    tracker.report("Phase 1")

    # === PHASE 2: Longer lengths ===
    p2_deadline = min(time.monotonic() + 300, deadline - 300)
    print(f"\n{'=' * 60}", flush=True)
    print(f"PHASE 2: Longer Lengths (budget to {p2_deadline - time.monotonic():.0f}s)", flush=True)
    print(f"{'=' * 60}", flush=True)

    base_n = tracker.best.size if tracker.best is not None else 256
    target_lengths = [l for l in [768, 1024, 1536] if l > base_n and l <= 2048]

    for new_n in target_lengths:
        if time.monotonic() > p2_deadline - 30:
            break

        t0 = time.monotonic()
        a_ext = resample_seq(tracker.best, new_n)
        a_ext = np.maximum(a_ext, 1e-6)
        a_ext = a_ext / a_ext.sum()

        a_opt = coord_descent(a_ext, max_sweeps=3000, deadline=p2_deadline - 30,
                              random_order=True, seed=seed + 200 + new_n)
        elapsed = time.monotonic() - t0
        J_val = compute_J(a_opt)
        print(f"  n={new_n}: J={J_val:.8f} ({elapsed:.1f}s)", flush=True)
        tracker.update(a_opt)

    tracker.report("Phase 2")

    # === PHASE 3: LP Mirror Descent ===
    p3_deadline = min(time.monotonic() + 400, deadline - 100)
    print(f"\n{'=' * 60}", flush=True)
    print(f"PHASE 3: LP Mirror Descent (budget to {p3_deadline - time.monotonic():.0f}s)", flush=True)
    print(f"{'=' * 60}", flush=True)

    if tracker.best is not None:
        n_best = tracker.best.size
        est_lp_time = 0.01 * (n_best / 128) ** 2
        max_iters = max(20, min(300, int(350 / max(0.01, est_lp_time))))
        print(f"  n={n_best}, est LP time={est_lp_time * 1000:.0f}ms, max_iters={max_iters}", flush=True)

        a_lp = run_lp_mirror_descent(tracker.best, max_iters, p3_deadline)
        J_val = compute_J(a_lp)
        print(f"  LP result: J={J_val:.8f}", flush=True)
        tracker.update(a_lp)

        # Second pass if time allows
        if time.monotonic() < p3_deadline - 5 * est_lp_time:
            a_lp2 = run_lp_mirror_descent(a_lp, 50, p3_deadline)
            J_val2 = compute_J(a_lp2)
            print(f"  LP pass 2: J={J_val2:.8f}", flush=True)
            tracker.update(a_lp2)

    tracker.report("Phase 3")

    # === PHASE 4: Final polishing ===
    p4_deadline = deadline
    print(f"\n{'=' * 60}", flush=True)
    print(f"PHASE 4: Final Polish (budget to {p4_deadline - time.monotonic():.0f}s)", flush=True)
    print(f"{'=' * 60}", flush=True)

    if tracker.best is not None and time.monotonic() < p4_deadline - 30:
        # Subgradient
        t0 = time.monotonic()
        a_sg = run_subgradient(tracker.best, max_iters=5000,
                               deadline=min(time.monotonic() + 60, p4_deadline - 30),
                               lr_init=0.1)
        J_val = compute_J(a_sg)
        print(f"  Subgradient: J={J_val:.8f} ({time.monotonic() - t0:.1f}s)", flush=True)
        tracker.update(a_sg)

        # Short coordinate descent
        if time.monotonic() < p4_deadline - 20:
            a_cd = coord_descent(tracker.best, max_sweeps=2000,
                                 deadline=min(time.monotonic() + 60, p4_deadline - 20),
                                 random_order=True, seed=seed + 999)
            J_val = compute_J(a_cd)
            print(f"  Coord descent: J={J_val:.8f}", flush=True)
            tracker.update(a_cd)

        # Final LP
        if time.monotonic() < p4_deadline - 5:
            n_est = tracker.best.size
            est_lp = 0.01 * (n_est / 128) ** 2
            if est_lp < (p4_deadline - time.monotonic()) / 3:
                a_final = run_lp_mirror_descent(tracker.best, 30, p4_deadline)
                J_val = compute_J(a_final)
                print(f"  Final LP: J={J_val:.8f}", flush=True)
                tracker.update(a_final)

    tracker.report("Phase 4")

    # Fallback
    if tracker.best is None:
        tracker.best = np.ones(128)
        tracker.best_score = float(evaluate_sequence(tracker.best.tolist()))

    print(f"\n{'=' * 60}", flush=True)
    print(f"FINAL: J={tracker.best_score:.8f} n={tracker.best.size} "
          f"time={time.monotonic() - tracker.t_start:.0f}s", flush=True)
    print(f"{'=' * 60}", flush=True)

    return [float(x) for x in tracker.best]
