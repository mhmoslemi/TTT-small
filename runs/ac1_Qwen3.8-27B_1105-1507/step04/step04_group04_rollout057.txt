    conv = np.fft.irfft(fa * fa, size)

def compute_obj(a):
    """Objective: 2*n*max(conv(a,a))/(sum(a))^2"""
    n = len(a)
    conv = compute_conv(a)
    max_b = np.max(conv)
    sum_a = np.sum(a)
    if sum_a < 0.01:
        return np.inf
    return 2.0 * n * max_b / (sum_a * sum_a)

def smooth_weights(conv, tau):
    """Compute softmax weights for smooth max approximation."""
    m = np.max(conv)
    shifted = tau * (conv - m)
    shifted = np.clip(shifted, -500, 50)
    w = np.exp(shifted)
    s = w.sum()
    if s < 1e-300:
        idx = np.argmax(conv)
        w = np.zeros_like(conv)
        w[idx] = 1.0
    else:
        w /= s
    return w

"""
Step function optimization using smoothed minimax with adaptive temperature,
LP-based refinement, and structured anti-palindromic exploration.
Key innovations:
- Log-sum-exp smoothed gradient with adaptive temperature scheduling
- Momentum-based multiplicative updates
- Anti-palindromic structured initialization for center-peak reduction
- Geometric (log-domain) combination with LP solutions
- Systematic n-exploration with resampling

import numpy as np
from scipy.optimize import linprog
import time


def compute_conv(a):
    """FFT-based self-convolution."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    conv = np.fft.irfft(fa * fa, size)
    return conv[:2 * n - 1]


def compute_obj(a):
    """Objective: 2*n*max(conv(a,a))/(sum(a))^2"""
    n = len(a)
    if n == 0:
        return np.inf
    conv = compute_conv(a)
    max_b = float(np.max(conv))
    sum_a = float(np.sum(a))
    if sum_a < 0.01:
        return np.inf
    return 2.0 * n * max_b / (sum_a * sum_a)


def smooth_weights(conv, tau):
    """Softmax weights for smooth max approximation."""
    m = float(np.max(conv))
    if m <= 0:
        idx = int(np.argmax(conv))
        w = np.zeros(len(conv))
        w[idx] = 1.0
        return w
    shifted = tau * (conv - m)
    shifted = np.clip(shifted, -500.0, 50.0)
    w = np.exp(shifted)
    s = float(np.sum(w))
    if s < 1e-300:
        idx = int(np.argmax(conv))
        w = np.zeros(len(conv))
        w[idx] = 1.0
    else:
        w /= s
    return w


def gradient_smoothed_max(a, tau):
    """Gradient of smooth max of conv(a,a) w.r.t. a."""
    n = len(a)
    conv = compute_conv(a)
    w = smooth_weights(conv, tau)
    corr = np.correlate(w, a, 'valid')
    return 2.0 * corr


def build_toeplitz(a):
    """Build matrix A where (A@g)[k] = full_conv(a,g)[k]."""
    n = len(a)
    size = 2 * n - 1
    A = np.zeros((size, n))
    for j in range(n):
        A[j:j + n, j] = a
    return A


def solve_lp_minpeak(a, time_limit=5):
    """LP: min M s.t. A@g <= M*1, sum(g)=1, g>=0."""
    n = len(a)
    A = build_toeplitz(a)
    c = np.zeros(n + 1)
    c[-1] = 1.0
    A_ub = np.zeros((2 * n - 1, n + 1))
    A_ub[:, :n] = A
    A_ub[:, n] = -1.0
    b_ub = np.zeros(2 * n - 1)
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])
    bounds = [(0.0, None)] * n + [(0.0, None)]
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': time_limit})
        if res.status == 0:
            g = np.maximum(res.x[:n], 0)
            s = float(np.sum(g))
            if s < 1e-10:
                return None, None
            return g / s, float(res.x[n])
    except Exception:
        pass
    return None, None


def solve_lp_maxsum(a, M, time_limit=5):
    """LP: max sum(g) s.t. A@g <= M, g>=0."""
    n = len(a)
    A = build_toeplitz(a)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': time_limit})
        if res.status == 0:
            g = np.maximum(res.x, 0)
            return g, float(np.sum(g))
    except Exception:
        pass
    return None, None


def normalize(a):
    """Normalize to sum=1, clip non-negative."""
    a = np.maximum(a, 1e-15)
    s = float(np.sum(a))
    if s < 1e-10:
        return np.ones(len(a)) / len(a)
    return a / s


def multiplicative_step_momentum(a, mom_buf, tau, step):
    """One multiplicative step with momentum."""
    n = len(a)
    grad = gradient_smoothed_max(a, tau)
    mom_buf[:] = 0.85 * mom_buf + 0.15 * grad
    log_update = -step * mom_buf
    log_update = np.clip(log_update, -50.0, 50.0)
    a_new = a * np.exp(log_update)
    return normalize(a_new)


def smoothed_gradient_optimize(a, time_budget, tau_start=5.0, tau_end=200.0,
                                step_start=0.1, step_end=0.005):
    """Smoothed gradient descent with adaptive temperature and step."""
    n = len(a)
    a = normalize(a.copy().astype(np.float64))
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    mom_buf = np.zeros(n)
    stall = 0
    patience = 150
    total_iters = 0

    while time.monotonic() - start < time_budget:
        total_iters += 1
        frac = min(1.0, total_iters / max(1, int(time_budget * 20)))
        tau = tau_start * (tau_end / tau_start) ** frac
        step = step_start * (step_end / step_start) ** frac

        a = multiplicative_step_momentum(a, mom_buf, tau, step)
        obj = compute_obj(a)

        if obj < best_obj - 1e-14:
            best_obj = obj
            best_a = a.copy()
            stall = 0
        else:

                a = a * np.exp(rng.normal(0, 0.05, n))
                a = normalize(a)
                mom_buf[:] = 0
                stall = 0

def fine_tune_active_lags(a, time_budget):
    """Fine-tune by focusing on active (near-max) convolution lags."""
    n = len(a)
    a = normalize(a.copy().astype(np.float64))
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    while time.monotonic() - start < time_budget:
        conv = compute_conv(a)
        max_conv = float(np.max(conv))

        # Find active lags (within 1% of max)
        active = np.where(conv >= 0.99 * max_conv)[0]

        # Compute gradient using only active lags (hard max approximation)
        if len(active) == 0:
            break

        # Gradient: for each active lag k, d conv[k]/d a[i] = 2*a[k-i] (if valid)
        grad = np.zeros(n)
        for k in active:
            for i in range(n):
                j = k - i
                if 0 <= j < n:
                    grad[i] += 2.0 * a[j]
        grad /= len(active)

        # Multiplicative step with very small step size
        step = 0.02
        log_update = -step * grad
        log_update = np.clip(log_update, -5.0, 5.0)
        a_new = a * np.exp(log_update)
        a_new = normalize(a_new)

        obj_new = compute_obj(a_new)
        if obj_new < best_obj - 1e-14:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new
        else:
            # Try smaller step
            step *= 0.5
            log_update = -step * grad
            log_update = np.clip(log_update, -2.5, 2.5)
            a_new = a * np.exp(log_update)
            a_new = normalize(a_new)
            obj_new = compute_obj(a_new)
            if obj_new < best_obj - 1e-14:
                best_obj = obj_new
                best_a = a_new.copy()
                a = a_new
            else:
                # Stuck, perturb
                rng = np.random.default_rng(int(time.monotonic() * 1000))
                a = a * np.exp(rng.normal(0, 0.02, n))
                a = normalize(a)

    return best_a, best_obj


def structured_init(n, seed):
    """Anti-palindromic structured initialization."""
    rng = np.random.default_rng(seed)
    k = np.arange(n)
    a = np.ones(n)

    # Add anti-palindromic sine components: sin(m*pi*(2k+1)/n)
    num_modes = int(rng.integers(2, 6))
    for idx in range(num_modes):
        m = int(rng.integers(1, max(2, n // 3)))
        amp = rng.uniform(0.05, 0.25) / (1.0 + idx * 0.3)
        a += amp * np.sin(m * np.pi * (2 * k + 1) / n)

    # Add small palindromic components (beyond constant)
    for idx in range(int(rng.integers(0, 3))):
        m = int(rng.integers(1, max(2, n // 4)))
        amp = rng.uniform(0.02, 0.1) / (1.0 + idx * 0.5)
        a += amp * np.cos(m * np.pi * (2 * k + 1) / n)

    a = np.maximum(a, 0)
    return normalize(a)


def random_smooth_init(n, seed):
    """Random smooth initialization."""
    rng = np.random.default_rng(seed)
    k = np.arange(n)
    choice = int(rng.integers(0, 5))

    if choice == 0:
        f = rng.uniform(0.5, 4.0)
        phase = rng.uniform(0, 2 * np.pi)
        a = 1.0 + 0.5 * np.cos(2 * np.pi * f * k / n + phase)
    elif choice == 1:
        f1 = rng.uniform(0.3, 2.0)
        f2 = rng.uniform(3.0, 8.0)
        a = 1.0 + 0.3 * np.cos(2 * np.pi * f1 * k / n) + 0.2 * np.cos(2 * np.pi * f2 * k / n)
    elif choice == 2:
        # Anti-palindromic heavy
        a = np.ones(n)
        for _ in range(3):
            m = int(rng.integers(1, max(2, n // 2)))
            a += rng.uniform(0.1, 0.3) * np.sin(m * np.pi * (2 * k + 1) / n)
    elif choice == 3:
        # Gaussian bump
        center = n * rng.uniform(0.3, 0.7)
        width = n * rng.uniform(0.1, 0.4)
        a = 1.0 + 0.5 * np.exp(-0.5 * ((k - center) / width) ** 2)
    else:
        # Two bumps
        c1 = n * rng.uniform(0.2, 0.4)
        c2 = n * rng.uniform(0.6, 0.8)
        w = n * rng.uniform(0.05, 0.2)
        a = 1.0 + 0.4 * np.exp(-((k - c1) / w) ** 2) + 0.4 * np.exp(-((k - c2) / w) ** 2)

    a = np.maximum(a, 0)
    return normalize(a)


def resample_sequence(a, new_n):
    """Resample sequence to new length."""
    n = len(a)
    if n == new_n:
        return a.copy()
    old_k = np.arange(n, dtype=np.float64)
    new_k = np.linspace(0, n - 1, new_n)
    a_new = np.interp(new_k, old_k, a)
    return normalize(a_new)
