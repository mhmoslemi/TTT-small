"""
Minimizes J(a) = 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Aggressive LP best-response with adaptive geometric tightening
2. Adaptive LSE (log-sum-exp) smooth gradient descent with momentum
3. Randomized coordinate descent with analytic line search
4. Multi-length search (256, 512, 768) with diverse initializations
5. Simulated annealing restarts for escaping local minima
"""

import numpy as np
from scipy.optimize import linprog
from scipy.sparse import csr_matrix
import time

_spar_cache = {}

def get_sparsity(n):
    if n not in _spar_cache:
        rows = []
        cols = []
        for k in range(2 * n - 1):
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            rows.extend([k] * (je - js + 1))
            cols.extend(range(js, je + 1))
        _spar_cache[n] = (np.array(rows, dtype=np.int32), np.array(cols, dtype=np.int32), len(rows))
    return _spar_cache[n]

def fast_conv(a):
    n = len(a)
    s = 1
    while s < 2 * n - 1:
        s <<= 1
    fa = np.fft.rfft(a, s)
    return np.fft.irfft(fa * fa, s)[:2 * n - 1]

def compute_J(a):
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))

def lp_best_response(a, M, n):
    rows, cols, nnz = get_sparsity(n)
    idx = rows.astype(np.int64) - cols.astype(np.int64)
    valid = (idx >= 0) & (idx < n)
    vals = np.zeros(nnz, dtype=np.float64)
    vals[valid] = a[idx[valid]]
    A = csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                options={'presolve': True, 'time_limit': 5, 'dual_feasibility_tolerance': 1e-9})
    if r.success and r.x is not None:
        g = np.maximum(r.x, 0.0)
        if g.sum() > 1e-10:
            return g
    return None

def run_lp_opt(a, max_iters, deadline, tracker, tighten_start=0.005, tighten_max=0.05):
    a = np.maximum(a.astype(np.float64), 1e-10)
    a /= a.sum()
    n = len(a)
    current_J = compute_J(a)
    tracker.update(a)
    last_workable_tighten = tighten_start * 0.5

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        b = fast_conv(a)
        M = float(b.max())
        
        # Adaptive tightening: try progressively tighter
        tighten = min(tighten_start + it * 0.002, tighten_max)
        M_tight = M * (1.0 - tighten)
        g = lp_best_response(a, M_tight, n)
        
        if g is None:
            # Back off
            tighten = max(0.0, last_workable_tighten - 0.005)
            M_tight = M * (1.0 - tighten)
            g = lp_best_response(a, M_tight, n)
            if g is None:
                g = lp_best_response(a, M * 1.001, n)
                if g is None:
                    break
        else:
            last_workable_tighten = tighten
        
        g = np.maximum(g, 0.0)
        gs = g.sum()
        if gs < 1e-10:
            break
        g /= gs

        best_a = None
        best_J = current_J
        progress = it / max(1, max_iters)
        base_t = max(0.1, 0.7 * (1.0 - 0.5 * progress))
        for t in [base_t, base_t * 0.5, base_t * 1.5, 0.9, 0.5, 0.3, 1.0]:
            if t > 1.0 or t <= 0:
                continue
            a_trial = (1.0 - t) * a + t * g
            a_trial = np.maximum(a_trial, 1e-10)
            ss = a_trial.sum()
            if ss < 1e-10:
                continue
            a_trial /= ss
            J_trial = compute_J(a_trial)
            if J_trial < best_J - 1e-15:
                best_J = J_trial
                best_a = a_trial
        if best_a is None:
            # Try pure g
            J_g = compute_J(g)
            if J_g < current_J - 1e-15:
                best_a = g.copy()
                best_J = J_g
            else:
                break
        a = best_a
        current_J = best_J
        if it % 5 == 0:
            tracker.update(a)
    tracker.update(a)
    return a

def lse_descent(a, max_iters, deadline, tracker, tau_start=20.0, tau_max=2000.0):
    a = np.maximum(a.astype(np.float64), 1e-8)
    a /= a.sum()
    n = len(a)
    current_J = compute_J(a)
    tracker.update(a)
    lr = 2.0 / (n * np.pi)
    momentum = 0.7
    v = np.zeros(n)
    tau = tau_start
    stall_count = 0

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        b = fast_conv(a)
        b_max = b.max()
        scaled = tau * (b - b_max)
        # Clip to avoid overflow
        scaled = np.clip(scaled, -50, 50)
        exp_b = np.exp(scaled)
        p = exp_b / exp_b.sum()

        # Gradient: 2 * correlate(p, a, 'valid')
        grad = 2.0 * np.correlate(p, a, mode='valid')
        
        v = momentum * v + (1.0 - momentum) * grad
        a_new = a - lr * v
        a_new = np.maximum(a_new, 1e-8)
        a_new /= a_new.sum()
        J_new = compute_J(a_new)
        if J_new < current_J - 1e-16:
            a = a_new
            current_J = J_new
            stall_count = 0
            # Increase tau gradually (sharper approximation to max)
            tau = min(tau_max, tau * 1.02)
            lr = 2.0 / (n * np.pi) * (tau_start / tau)**0.5
        else:
            stall_count += 1
            lr *= 0.6
            v *= 0.5
            if stall_count > 20:
                # Restart with smaller tau
                tau = max(tau_start, tau * 0.5)
                v = np.zeros(n)
                stall_count = 0
                if lr < 1e-18:
                    break
            if lr < 1e-18:
                break
        if it % 20 == 0:
            tracker.update(a)
    tracker.update(a)
    return a

def coord_descent(a, max_steps, deadline, tracker):
    a = np.maximum(a.astype(np.float64), 1e-10)
    a /= a.sum()
    n = len(a)
    current_J = compute_J(a)
    tracker.update(a)
    rng = np.random.default_rng(12345)
    stall = 0

    for step in range(max_steps):
        if time.monotonic() > deadline:
            break
        j = rng.integers(0, n)
        best_scale = 1.0
        best_J = current_J
        for scale in [0.5, 1.5, 0.25, 2.0, 0.75, 1.25, 0.1, 3.0]:
            a_trial = a.copy()
            a_trial[j] *= scale
            a_trial = np.maximum(a_trial, 1e-10)
            ss = a_trial.sum()
            if ss < 1e-10:
                continue
            a_trial /= ss
            J_trial = compute_J(a_trial)
            if J_trial < best_J - 1e-15:
                best_J = J_trial
                best_scale = scale
        if best_scale != 1.0:
            a[j] *= best_scale
            a = np.maximum(a, 1e-10)
            a /= a.sum()
            current_J = best_J
            stall = 0
        else:
            stall += 1
            if stall > 200:
                break
        if step % 50 == 0:
            tracker.update(a)
    tracker.update(a)
    return a

def flat_spectrum_init(n, rng):
    half = n // 2 + 1
    angles = rng.uniform(0, 2 * np.pi, half)
    A = np.zeros(n, dtype=complex)
    A[:half] = np.exp(1j * angles[:half])
    for k in range(1, n - n // 2):
        A[k] = np.conj(A[n - k])
    a = np.fft.ifft(A).real
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-10:
        return np.ones(n) / n
    return a / s

def resample_seq(src, dest_n):
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)

def generate_inits(n, rng, parent):
    inits = []
    if parent is not None and parent.size > 0:
        inits.append(resample_seq(parent, n))
        inits.append(resample_seq(parent[::-1], n))
        # Perturbed parent
        pert = parent + rng.normal(0, 0.1, parent.size)
        pert = np.maximum(pert, 0.0)
        inits.append(resample_seq(pert, n))

    inits.append(np.ones(n) / n)
    inits.append(rng.uniform(0.1, 1.0, n))
    inits.append(flat_spectrum_init(n, rng))
    inits.append(flat_spectrum_init(n, rng))
    inits.append(flat_spectrum_init(n, rng))

    t = np.linspace(0, 1, n, endpoint=False)
    for freq in [2, 3, 5, 7, 11, 13, 17]:
        inits.append(1.0 + 0.3 * np.cos(2 * np.pi * freq * t + rng.uniform(0, 6.28)))
    
    inits.append(1.0 + 0.25 * np.cos(2 * np.pi * 3.7 * t) + 0.15 * np.cos(2 * np.pi * 7.3 * t)
                + 0.1 * np.cos(2 * np.pi * 11.1 * t))
    inits.append(0.5 + 0.5 * np.cos(np.pi * t) ** 2)
    inits.append(0.3 + 0.7 * np.exp(-4 * (t - 0.5) ** 2))
    inits.append(0.4 * np.exp(-8 * (t - 0.3) ** 2) + 0.4 * np.exp(-8 * (t - 0.7) ** 2) + 0.1)
    
    # Multi-frequency with random phases
    inits.append(1.0 + 0.2 * np.sin(2 * np.pi * t + 0.5) + 0.15 * np.sin(4 * np.pi * t + 1.2)
                + 0.1 * np.sin(6 * np.pi * t + 2.1) + 0.08 * np.sin(10 * np.pi * t + 3.3))
    
    # Power-law
    inits.append(((t + 0.05) ** rng.uniform(-0.8, 0.8)) + 0.01)
    inits.append(((1.0 - t + 0.05) ** rng.uniform(-0.8, 0.8)) + 0.01)
    
    # Sparse smoothed
    sparse = np.zeros(n)
    n_spikes = max(10, n // 10)
    positions = rng.choice(n, size=n_spikes, replace=False)
    sparse[positions] = 1.0
    kernel_size = max(5, n // 20)
    kernel = np.exp(-0.5 * (np.arange(-kernel_size, kernel_size + 1) / (kernel_size * 0.5)) ** 2)
    kernel /= kernel.sum()
    inits.append(np.convolve(sparse, kernel, mode='same') + 0.02)
    
    # Band-limited random
    n_coeff = min(n // 2 + 1, 15)
    coeffs = rng.normal(0, 1, n_coeff)
    sig = np.zeros(n)
    for i in range(n_coeff):
        sig += coeffs[i] * np.cos(2 * np.pi * i * t + rng.uniform(0, 6.28))
    inits.append(np.abs(sig) + 0.3)

    for x in inits:
        x = np.maximum(x, 0.001).astype(np.float64)
        x /= x.sum()
    return inits

def simulated_annealing(a, deadline, tracker, n_restarts=30, t0=0.08):
    rng = np.random.default_rng(555)
    n = a.size
    a = a.copy().astype(np.float64)
    current_J = compute_J(a)
    tracker.update(a)
    temp = t0

    for r in range(n_restarts):
        if time.monotonic() > deadline:
            break
        a_pert = a + rng.normal(0, temp, n)
        a_pert = np.maximum(a_pert, 1e-8)
        a_pert /= a_pert.sum()
        
        # Short LP refinement
        a_refined = run_lp_opt(a_pert, max(3, 15 - r // 4), min(deadline, time.monotonic() + 30), tracker, tighten_start=0.01, tighten_max=0.03)
        
        J = compute_J(a_refined)
        delta = J - current_J
        if delta < 0 or rng.random() < np.exp(-delta / max(temp, 1e-10)):
            a = a_refined
            current_J = J
        temp *= 0.93
    tracker.update(a)
    return a

def multi_start_optimize(n, rng, parent, time_budget, tracker):
    """Run multiple initializations with LP + LSE + CD for a given length."""
    inits = generate_inits(n, rng, parent)
    best_a = None
    best_J = float('inf')
    
    n_inits = len(inits)
    per_init_time = time_budget / n_inits
    
    for idx, a0 in enumerate(inits):
        t_start = time.monotonic()
        t_end = t_start + per_init_time
        t_end = min(t_end, tracker.t_start + 1500 - 30)
        
        if time.monotonic() > t_end:
            break
        
        # Phase A: LP
        n_lp = max(5, int(60 / max(1, n / 300)))
        a = run_lp_opt(a0, n_lp, min(t_end, time.monotonic() + per_init_time * 0.5), tracker,
                       tighten_start=0.005, tighten_max=0.03)
        
        # Phase B: LSE
        if time.monotonic() < t_end - 2:
            a = lse_descent(a, max(100, int(50000 / n)), min(t_end - 2, time.monotonic() + per_init_time * 0.3), tracker,

            a = coord_descent(a, max(50, int(20000 / n)), min(t_end - 1, time.monotonic() + per_init_time * 0.2), tracker)
        
        J = compute_J(a)
        if J < best_J:
            best_J = J
            best_a = a.copy()
            tracker.update(a)
