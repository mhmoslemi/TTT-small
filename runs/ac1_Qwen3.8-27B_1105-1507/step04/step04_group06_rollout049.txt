import numpy as np
from scipy.optimize import linprog, minimize
from scipy.sparse import csr_matrix

# ============ Fast convolution ============

def fast_conv(a):
    """Full self-convolution via FFT."""
    n = len(a)
    s = 1
    while s < 2 * n - 1:
        s <<= 1
    fa = np.fft.rfft(a, s)
    return np.fft.irfft(fa * fa, s)[:2 * n - 1]

def fast_conv_ab(a, b):
    """Full convolution of a and b via FFT."""
    n = max(len(a), len(b))
    s = 1
    while s < len(a) + len(b) - 1:
        s <<= 1
    fa = np.fft.rfft(a, s)
    fb = np.fft.rfft(b, s)
    return np.fft.irfft(fa * fb, s)[:len(a) + len(b) - 1]

def compute_J(a):
    """Compute J(a) = 2*n*max(conv(a,a))/(sum(a))^2."""
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))

    while s < 2 * n - 1:
        s <<= 1
    fw = np.fft.rfft(w, s)
    fa = np.fft.rfft(a, s)
    pressure = np.fft.irfft(fa * fw, s)[:n]
    # Wait, this isn't quite right. Let me think...
    # pressure_j = sum_k w_k * a_{k-j} (with appropriate bounds)
    # This is the cross-correlation of w and a
    # Actually it's: pressure_j = sum_k w_k * a_{k-j}
    # = (w * a_flipped)[j] where a_flipped is a reversed
    # Or more carefully: it's the convolution w * a[::-1] evaluated at appropriate positions
    
    # Let me just do it directly for clarity (it's O(n^2) but fine for moderate n)
    # Actually for n=1024, O(n^2) = 1M ops per step, that's fine
    
    # Better: use FFT
    # pressure_j = sum_{k=0}^{2n-2} w_k * a_{k-j} (where a_out_of_range = 0)
    # = sum_{k=0}^{2n-2} w_k * a_{k-j}
    # Let's define: for fixed j, pressure_j = sum_k w_k * a_{(k-j) mod n}... no
    # a is only defined for indices 0..n-1
    # So pressure_j = sum_{k: 0<=k-j<n} w_k * a_{k-j} = sum_{i: 0<=i<n} w_{i+j} * a_i
    # = sum_i a_i * w_{i+j}
    # This is the cross-correlation!
    # pressure = correlate(a, w) where w is of length 2n-1
    
    # Using FFT: cross_correlation(a, w) = IFFT(FFT(a) * conj(FFT(w)))
    # But w is length 2n-1, a is length n... let me just use np.convolve
    
    # Actually the simplest: pressure_j = (convolve(w, a))[j + n - 1] ... no
    # Let me just think about it directly.
    # (conv(w, a))[m] = sum_i w[i] * a[m-i] for valid i
    # We want pressure_j = sum_k w[k] * a[k-j]
    # Let m = k-j, so k = m+j: pressure_j = sum_m w[m+j] * a[m]
    # = (correlate(w, a))[j] = (convolve(a, w))[j + len(w) - 1]
    
    # Hmm this is getting confusing. Let me just use a direct approach with FFT.
    
    # Alternative: pressure_j = sum_{k=0}^{2n-2} w_k * a_{k-j}
    # If we define extended_a of length 2n-1 with a_padded[k] = a[k] if 0<=k<n else 0:
    # Then pressure_j = sum_k w_k * a_padded[k-j]
    # This is (conv(w, a_padded))[2n-2-j] ... let me verify
    # (conv(w, a_padded))[m] = sum_k w[k] * a_padded[m-k]
    # Set m-k = k-j => m = 2k-j... no that's not right either.
    
    # Let me just be explicit:
    # pressure_j = sum_{k=0}^{2n-2} w_k * a_padded[k-j]
    # where a_padded[i] = a[i] if 0<=i<n else 0
    # = sum_{k: 0<=k-j<n} w_k * a[k-j]
    # Let i = k-j: = sum_{i: -j<=i<n-j} w_{i+j} * a_i
    # = sum_{i=0}^{n-1} w_{i+j} * a_i (since w is defined for 0..2n-2 and i+j is in range)
    
    # So pressure_j = dot(a, w[j:j+n]) for j = 0, ..., n-1
    # (with w having length 2n-1, so w[j:j+n] is valid for j=0..n-1 since j+n <= 2n-1)
    
    # This can be computed as a sliding dot product, or via FFT:
    # pressure = correlate(a, w) where we take the first n values
    # = FFT-based cross-correlation
    
    # Using the fact that cross_correlation(a, w)[j] = sum_i a[i] * w[i+j]
    # = IFFT(conj(FFT(a)) * FFT(w))[j] ... let me verify
    # Actually: sum_i a[i] * w[i+j] = (a * w_flipped)[j + len(w) - 1]... 
    # This is getting too complicated. Let me just use a simpler approach.
    
    # For the multiplicative weights, I'll use a simpler (but still effective) formula:
    # The "cost" of a_j is approximately a_j * (sum of nearby b values weighted by distance)
    # A simpler approximation: pressure_j = a_j * b_max (proportional to a_j at the peak)
    # But this is too crude.
    
    # Let me use the subgradient approach:
    # The subgradient of max(b) at b is e_{argmax(b)}
    # The gradient of b_k with respect to a_j is a_{k-j} (if valid)
    # So the subgradient of max(conv(a,a)) with respect to a_j is:
    #   sum_{k in argmax} a_{k-j} (symmetrized: 2 * a_{k-j} if we account for both terms)
    
    k_star = np.argmax(b)
    # Subgradient at a: grad_j = 2 * a_{k_star - j} (if 0 <= k_star - j < n)
    grad = np.zeros(n)
    for j in range(n):
        idx = k_star - j
        if 0 <= idx < n:
            grad[j] = 2.0 * a[idx]
    
    # Multiplicative update: a_j *= exp(-eta * grad_j / max(grad))
    grad_max = grad.max()
    if grad_max > 0:
        a_new = a * np.exp(-eta * grad / grad_max)
        a_new = np.maximum(a_new, 1e-10)
        a_new = a_new / a_new.sum()
    else:
        a_new = a.copy()
    
    return a_new

# ============ Smoothed optimization ============

def smooth_max_and_grad(b, beta):
    """Smooth max using log-sum-exp, return (value, weights)."""
    b_max = b.max()
    shifted = beta * (b - b_max)
    shifted = np.clip(shifted, -500, 500)
    exp_shifted = np.exp(shifted)
    Z = exp_shifted.sum()
    if Z < 1e-300:
        p = np.zeros(len(b))
        p[b_max == b] = 1.0 / (b == b_max).sum()
    else:
        p = exp_shifted / Z
    val = b_max + np.log(max(Z, 1e-300)) / beta
    return val, p

def J_smooth_and_grad(a, beta):
    """Compute smoothed J and its gradient."""
    n = len(a)
    b = fast_conv(a)
    s = a.sum()
    
    g_val, p = smooth_max_and_grad(b, beta)
    J_val = 2.0 * n * g_val / (s * s)
    
    # Gradient of g_val = sum_k p_k * b_k
    # d(b_k)/da_j = a_{k-j} + a_{j-k} (symmetric) = 2 * a_{|k-j|} (if valid)
    # Wait, b_k = sum_i a_i * a_{k-i}
    # d(b_k)/da_j = a_{k-j} (if 0 <= k-j < n) + a_{j-k} (if 0 <= j-k < n)
    # By symmetry of the convolution of a with itself, these two terms are the same
    # when both are valid, so d(b_k)/da_j = 2*a_{k-j} if 0<=k-j<n, or a_{k-j} if only one term
    # Actually: b_k = sum_i a_i * a_{k-i}
    # d(b_k)/da_j: the term a_j appears when i=j (gives a_{k-j}) and when k-i=j i.e. i=k-j (gives a_j)
    # So if j != k-j (i.e., k is even and j = k/2, or more precisely k-j != j), we get 2*a_{k-j}
    # If k-j = j (i.e., k = 2j), we get a_j (only one term)
    # Wait no: b_k = sum_i a_i * a_{k-i}
    # The term involving a_j: when i=j, we get a_j * a_{k-j}. When i=k-j, we get a_{k-j} * a_j.
    # These are the same term if k-j = j, i.e., k = 2j.
    # So d(b_k)/da_j = a_{k-j} if k-j = j (i.e., k=2j), else 2*a_{k-j} (if 0<=k-j<n)
    # But wait, we also need k-j >= 0 and k-j < n for a_{k-j} to exist.
    
    # For the gradient of the full objective:
    # J = 2n * g_val / s^2
    # dJ/da_j = 2n * (s * dg_val/da_j - 2*g_val) / s^3
    # where dg_val/da_j = sum_k p_k * d(b_k)/da_j
    
    # Compute dg_val/da_j:
    # dg_val/da_j = sum_k p_k * d(b_k)/da_j
    #             = sum_k p_k * [2*a_{k-j} (if 0<=k-j<n) or a_{k-j} (if k=2j)]
    
    # This is essentially a weighted convolution of p and a.
    # Let's compute it via FFT for efficiency.
    
    # For most k, d(b_k)/da_j = 2*a_{k-j} (when 0 <= k-j < n and k != 2j)
    # The k=2j case is a measure-zero correction that we can ignore for large n.
    
    # So approximately: dg_val/da_j ≈ 2 * sum_k p_k * a_{k-j} (with bounds)
    # This is 2 * cross_correlation(p, a)[j]
    
    # Using FFT: cross_correlation(p, a)[j] = sum_i a[i] * p[i+j] (for valid indices)
    # = IFFT(FFT(a) * conj(FFT(p))) ... let me verify
    # Actually, the cross-correlation can be computed as:
    # corr[j] = sum_i a[i] * p[i+j]
    # This equals (convolve(a, p_flipped))[j + len(p) - 1] where p_flipped = p[::-1]
    # Or: corr = np.correlate(p, a, mode='full')[offset]
    
    # Let me use FFT:
    # corr[j] = sum_i a[i] * p[i+j]
    # Let's define: state that we want sum_i a[i] * p[i+j] for j = 0, ..., n-1
    # = sum_i a[i] * p[i+j]
    # Consider the product in frequency domain:
    # FFT(a)[m] * conj(FFT(p)[m]) gives the cross-correlation
    # Specifically: IFFT(FFT(a) * conj(FFT(p)))[j] = sum_i a[i] * p[(i-j) mod N]
    # That's not quite what we want. We want sum_i a[i] * p[i+j].
    # 
    # Let me use a different approach:
    # sum_i a[i] * p[i+j] = (a * p_reversed)[j + len(p) - n + 1] ... this is getting confusing.
    
    # Let me just use the direct formula with a twist:
    # dg_val/da_j ≈ 2 * sum_{k=0}^{2n-2} p[k] * a[k-j] (where a is 0 outside [0,n-1])
    # = 2 * (convolve(p, a_padded))[2n-2-j] ... 
    
    # You know what, let me just use np.convolve for clarity:
    # pressure_j = sum_k p[k] * a[k-j]
    # = sum_k p[k] * a[k-j] where a is zero-padded
    # This is the cross-correlation of p and a:
    # cross_corr[j] = sum_k p[k] * a[k-j] = np.correlate(p, a, mode='full')[2n-2-j]
    # Wait: np.correlate(p, a)[j] = sum_k p[k+j] * a[k] (for 'full' mode, with the right offset)
    
    # Let me just compute it directly using the property:
    # pressure = np.convolve(p, a[::-1], mode='full')
    # Then pressure[n-1+j] = sum_k p[k] * a[n-1-(n-1+j-k)] = sum_k p[k] * a[k-j]
    # Hmm, let me verify with a small example.
    # a = [a0, a1, a2], p = [p0, p1, p2, p3] (lengths n=3, 2n-1=5)
    # a[::-1] = [a2, a1, a0]
    # convolve(p, a[::-1], 'full') has length 5+3-1=7
    # result[m] = sum_k p[k] * a_rev[m-k] = sum_k p[k] * a[2-(m-k)] = sum_k p[k] * a[k-m+2]
    # We want pressure_j = sum_k p[k] * a[k-j] for j=0,1,2
    # pressure_0 = sum_k p[k]*a[k] -> m such that k-m+2=k -> m=2: result[2]
    # pressure_1 = sum_k p[k]*a[k-1] -> m such that k-m+2=k-1 -> m=3: result[3]
    # pressure_2 = sum_k p[k]*a[k-2] -> m such that k-m+2=k-2 -> m=4: result[4]
    # So pressure_j = result[n-1+j] = result[2+j] for j=0,1,2
    # In general: pressure_j = convolve(p, a[::-1], 'full')[n-1+j]
    
    a_rev = a[::-1]
    pressure_full = np.convolve(p, a_rev, mode='full')
    pressure = pressure_full[n-1:2*n-1]  # n values: pressure[0] to pressure[n-1]
    
    # Now dg_val/da_j = 2 * pressure_j (approximately, ignoring the k=2j correction)
    # More precisely, for the self-convolution, the factor is 2 except at k=2j
    # But for large n, this is a negligible correction
    
    dg = 2.0 * pressure
    
    # dJ/da_j = 2n * (s * dg_j - 2*g_val) / s^3
    grad_J = 2.0 * n * (s * dg - 2.0 * g_val) / (s ** 3)
    
    return J_val, grad_J

def line_search_golden(a, g, n_evals=15):
    """Golden section search for best t in [0,1]."""
    best_t = 0.0
    best_J = compute_J(a)
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    lo = 0.0
    hi = 1.0
    
    a_1 = g.copy()
    J_at_1 = compute_J(a_1)
    if J_at_1 < best_J:
        best_J = J_at_1
        best_t = 1.0
    
    for _ in range(n_evals):
        t1 = hi - gr * (hi - lo)
        t2 = lo + gr * (hi - lo)
        a1 = (1.0 - t1) * a + t1 * g
        a2 = (1.0 - t2) * a + t2 * g
        J1 = compute_J(a1)
        J2 = compute_J(a2)
        if J1 < J2:
            hi = t2
        else:
            lo = t1
    
    # Final evaluation at midpoint
    t_mid = (lo + hi) / 2.0
    a_mid = (1.0 - t_mid) * a + t_mid * g
    J_mid = compute_J(a_mid)
    if J_mid < best_J:
        best_J = J_mid
        best_t = t_mid
    
    return (1.0 - best_t) * a + best_t * g, best_J

def lp_mirror_descent_fast(a, max_iters, deadline, tracker, 
                           n_line_evals=15, stall_patience=50,
                           n_constraints=None):
    """Fast LP mirror descent."""
    a = np.maximum(a.astype(np.float64), 1e-8)
    a = a / a.sum()
    current_J = compute_J(a)
    best_J = current_J
    best_a = a.copy()
    stall_count = 0
    
    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        
        b = fast_conv(a)
        M = float(b.max())
        
        g = lp_best_response(a, M, n_constraints=n_constraints)
        if g is None:
            break
        
        a_new, J_new = line_search_golden(a, g, n_evals=n_line_evals)
        
        if J_new < current_J - 1e-13:
            a = a_new
            current_J = J_new
            stall_count = 0
            if J_new < best_J:
                best_J = J_new
                best_a = a.copy()
                if tracker is not None:
                    tracker.update(a)
        else:
            stall_count += 1
            if stall_count >= stall_patience:
                break
    
    if tracker is not None:
        tracker.update(best_a)
    return best_a

def multiplicative_weights_loop(a, n_steps, deadline, tracker, eta=0.3):
    """Loop of multiplicative weights steps."""
    a = np.maximum(a.astype(np.float64), 1e-8)
    a = a / a.sum()
    best_J = compute_J(a)
    best_a = a.copy()
    
    for step in range(n_steps):
        if time.monotonic() > deadline:
            break
        a = multiplicative_weights_step(a, eta=eta)
        J = compute_J(a)
        if J < best_J:
            best_J = J
            best_a = a.copy()
            if tracker is not None:
                tracker.update(a)
        elif (step + 1) % 100 == 0:
            # Even if not improving, update tracker occasionally
            pass
    
    if tracker is not None:
        tracker.update(best_a)
    return best_a

def adam_smooth_optimize(a0, deadline, tracker, beta_start=1.0, beta_end=100.0, 
                          n_total_steps=2000, lr=0.001):
    """Adam optimization with annealing beta."""
    a = np.maximum(a0.astype(np.float64), 1e-6)
    a = a / a.sum()
    n = len(a)
    
    best_J = compute_J(a)
    best_a = a.copy()
    
    # Adam state
    m = np.zeros(n)
    v = np.zeros(n)
    beta1, beta2, eps = 0.9, 0.999, 1e-8
    
    for step in range(n_total_steps):
        if time.monotonic() > deadline:
            break
        
        # Anneal beta
        t_frac = step / n_total_steps
        beta = beta_start * (beta_end / beta_start) ** t_frac
        
        J_val, grad = J_smooth_and_grad(a, beta)
        grad = np.nan_to_num(grad, nan=0.0, posinf=0.0, neginf=0.0)
        
        # Adam update
        t = step + 1
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * grad ** 2
        m_hat = m / (1 - beta1 ** t)
        v_hat = v / (1 - beta2 ** t)
        a_new = a - lr * m_hat / (np.sqrt(v_hat) + eps)
        
        # Project onto feasible set (a >= 0, sum = 1)
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s < 1e-10:
            a_new = a.copy()
        else:
            a_new = a_new / s
        
        # Check if we crossed zero (sign change) - if so, use smaller step
        if np.any(a_new != a) and compute_J(a_new) > compute_J(a) * 1.01:
            # Try smaller step
            a_new = a - 0.1 * lr * m_hat / (np.sqrt(v_hat) + eps)
            a_new = np.maximum(a_new, 0.0)
            s = a_new.sum()
            if s > 1e-10:
                a_new = a_new / s
            else:
                a_new = a.copy()
        
        a = a_new
        J = compute_J(a)
        if J < best_J:
            best_J = J
            best_a = a.copy()
            if tracker is not None:
                tracker.update(a)
    
    if tracker is not None:
        tracker.update(best_a)
    return best_a

# ============ Utilities ============

class ScoreTracker:
    def __init__(self):
        self.best = None
        self.best_score = float('inf')
        self.t_start = time.monotonic()
    
    def update(self, a):
        if a is None:
            return
        a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
        if a.size == 0 or a.sum() < 0.01:
            return
        score = float(evaluate_sequence(a.tolist()))
        if np.isfinite(score) and score < self.best_score:
            self.best = a.copy()
            self.best_score = score
            elapsed = time.monotonic() - self.t_start
            print(f"  * NEW BEST: J={score:.8f} n={a.size} t={elapsed:.1f}s", flush=True)
    
    def report(self, label=""):
        if self.best is not None:
            print(f"  {label} best: J={self.best_score:.8f} n={self.best.size}", flush=True)

def resample_seq(src, dest_n):
    """Resample 1D sequence to new length."""
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    result = np.interp(t_new, t_old, src)
    return np.maximum(result, 0.0)

def generate_inits(n, rng, parent=None):
    """Generate diverse initial sequences."""
    inits = []
    t = np.linspace(0, 1, n, endpoint=False)
    
    if parent is not None:
        inits.append(resample_seq(parent, n))
    
    inits.append(np.ones(n))
    inits.append(rng.uniform(0.5, 1.5, n))
    
    # Cosine families
    for freq in [2, 3, 4, 5, 6, 8]:
        inits.append(1.0 + 0.3 * np.cos(2.0 * np.pi * freq * t))
    
    # Superposition
    inits.append(1.0 + 0.2 * np.cos(4 * np.pi * t) + 0.15 * np.cos(8 * np.pi * t))
    inits.append(1.0 + 0.25 * np.cos(6 * np.pi * t) + 0.1 * np.cos(12 * np.pi * t))
    
    # Gaussian
    inits.append(np.exp(-3.0 * (t - 0.5) ** 2) + 0.3)
    
    # Sine raised
    inits.append(np.abs(np.sin(np.pi * t)) ** 0.5 + 0.2)
    
    # Power law
    inits.append((t + 0.05) ** 0.3)
    inits.append(1.0 / (t + 0.05) ** 0.2)
    
    # Triangle
    inits.append(1.0 - 0.3 * np.abs(2.0 * (t % 0.5) - 0.5))
    
    # Riemann-like
    inits.append(1.0 + 0.15 * np.sum(np.cos(2.0 * np.pi * k * t) / k for k in [1,2,3,4,5]))
    
    return inits
