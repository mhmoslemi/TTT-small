import numpy as np
from scipy.optimize import minimize, linprog
from scipy.fft import rfft, irfft
import time


def compute_conv(a):
    """Compute linear convolution a*a using FFT. Returns array of length 2n-1."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    conv = irfft(fa * fa, size)
    return conv[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    conv = compute_conv(a)
    max_b = float(np.max(conv))
    sum_a = float(np.sum(a))
    if sum_a < 0.01:
        return np.inf
    return 2.0 * n * max_b / (sum_a * sum_a)


def fft_convolve(a, b):
    """Linear convolution of a and b using FFT."""
    n, m = len(a), len(b)
    size = 1
    while size < n + m - 1:
        size <<= 1
    fa = rfft(a, size)
    fb = rfft(b, size)
    return irfft(fa * fb, size)[:n + m - 1]


def softmax_fn(z):
    """Numerically stable softmax."""
    z = z - np.max(z)
    e = np.exp(z)
    s = np.sum(e)
    if s < 1e-300:
        return np.ones(len(z)) / len(z)
    return e / s


def soft_max_obj_and_grad_z(z, tau):
    """Compute soft-max of conv(softmax(z), softmax(z)) and gradient w.r.t. z."""
    n = len(z)
    a = softmax_fn(z)
    conv = compute_conv(a)
    cmax = float(np.max(conv))
    shifted = np.clip((conv - cmax) / tau, -500.0, 500.0)
    exp_conv = np.exp(shifted)
    Z = float(np.sum(exp_conv))
    if Z < 1e-300:
        w = np.zeros(len(conv))
        w[int(np.argmax(conv))] = 1.0
        soft_max_val = cmax
    else:
        soft_max_val = cmax + tau * np.log(Z)
        w = exp_conv / Z

    w_rev = w[::-1]
    conv_fw = fft_convolve(a, w_rev)
    f1 = conv_fw[n - 1:2 * n - 1][::-1]
    grad_a = 2.0 * f1
    mean_grad = float(np.dot(a, grad_a))
    grad_z = a * (grad_a - mean_grad)
    return soft_max_val, grad_z


def build_toeplitz(a):
    """Build Toeplitz matrix where (A@g)[k] = (conv(a,g))[k]."""
    n = len(a)
    A = np.zeros((2 * n - 1, n), dtype=np.float64)
    for j in range(n):
        A[j:j + n, j] = a[:n]
    # Actually, (conv(a,g))[k] = sum_j a[j]*g[k-j] for valid k-j
    # So A[k,j] = a[k-j] if 0 <= k-j < n, else 0
    A = np.zeros((2 * n - 1, n), dtype=np.float64)
    for j in range(n):
        for k in range(max(0, 2 * n - 1 - (n - 1)), min(2 * n - 1, j + n)):
            pass
    # More efficient:
    A = np.zeros((2 * n - 1, n), dtype=np.float64)
    for j in range(n):
        start = j
        end = min(j + n, 2 * n - 1)
        A[start:end, j] = a[0:end - start]
    return A


def solve_lp_minpeak(a, time_limit=15):
    """
    Solve LP: minimize M s.t. (a*g)[k] <= M for all k, sum(g)=1, g>=0.
    Returns (g, M_opt) or (None, None).
    """
    n = len(a)
    A = build_toeplitz(a)
    # Variables: g[0..n-1], M
    c = np.zeros(n + 1)
    c[-1] = 1.0  # minimize M
    A_ub = np.zeros((2 * n - 1, n + 1))
    A_ub[:, :n] = A
    A_ub[:, n] = -1.0  # (a*g)[k] - M <= 0
    b_ub = np.zeros(2 * n - 1)
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])
    bounds = [(0.0, None)] * n + [(0.0, None)]
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': time_limit})
        if res.status == 0:
            g = np.maximum(res.x[:n], 0)
            s = np.sum(g)
            if s > 1e-10:
                g = g / s
                M_opt = float(res.x[n])
                return g, M_opt
        elif res.status == 3:  # time limit
            # Try to extract partial solution
            if hasattr(res, 'x') and res.x is not None:
                g = np.maximum(res.x[:n], 0)
                s = np.sum(g)
                if s > 1e-10:
                    g = g / s
                    M_opt = float(res.x[n])
                    return g, M_opt
    except Exception:
        pass
    return None, None


def solve_lp_maxsum(a, M, time_limit=15):
    """Solve LP: max sum(g) s.t. (a*g)[k] <= M for all k, g >= 0."""
    n = len(a)
    A = build_toeplitz(a)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': time_limit})
        if res.status == 0:
            g = np.maximum(res.x, 0)
            s = float(np.sum(g))
            if s > 1e-10:
                return g / s, s
    except Exception:
        pass
    return None, None

                a = a * np.exp(rng.normal(0, scale, n))

                    a = a / s
                stall_count = 0

                a = np.maximum(a, 0)
                s = np.sum(a)

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    for tau in tau_list:
        if time.monotonic() - start > time_budget:
            break

        def func_and_grad(z_val):
            return soft_max_obj_and_grad_z(z_val, tau)

        try:
            result = minimize(func_and_grad, z, jac=True, method='L-BFGS-B',
                              options={'maxiter': max_iter, 'ftol': 1e-14, 'gtol': 1e-8})
            z_new = result.x
        except Exception:
            continue

        a_new = softmax_fn(z_new)
        a_new = np.maximum(a_new, 0)
        s_new = np.sum(a_new)
        if s_new < 1e-10:
            continue
        a_new = a_new / s_new
        obj = compute_obj(a_new)
        if obj < best_obj:
            best_obj = obj
            best_a = a_new.copy()
            z = z_new

    return best_a, best_obj


def subgradient_optimize(a, max_iters=2000, time_budget=20.0, tau_start=2.0, tau_end=40.0):
    """Smoothed subgradient optimization with adaptive step size."""
    n = len(a)
    a = a.copy().astype(np.float64)
    s = np.sum(a)
    if s < 1e-10:
        a = np.ones(n) / n
    else:
        a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    step = 0.1
    stall = 0

    for it in range(max_iters):
        if time.monotonic() - start > time_budget:
            break

        frac = it / max(1, max_iters)
        tau = tau_start + (tau_end - tau_start) * frac

        conv = compute_conv(a)
        max_conv = float(np.max(conv))
        shifted = np.clip(tau * (conv - max_conv), -300, 300)
        weights = np.exp(shifted)
        w_sum = np.sum(weights)
        if w_sum < 1e-300:
            idx = int(np.argmax(conv))
            weights = np.zeros(len(conv))
            weights[idx] = 1.0
        else:
            weights /= w_sum

        w_rev = weights[::-1]
        conv_fw = fft_convolve(a, w_rev)
        grad_a = 2.0 * conv_fw[n - 1:2 * n - 1][::-1]
        grad_centered = grad_a - np.mean(grad_a)

        a_new = a - step * grad_centered
        a_new = np.maximum(a_new, 0)
        s_new = np.sum(a_new)
        if s_new > 1e-10:
            a_new = a_new / s_new
        else:
            a_new = a.copy()

        a = a_new
        obj = compute_obj(a)

        if obj < best_obj - 1e-12:
            best_obj = obj
            best_a = a.copy()
            stall = 0
        else:
            stall += 1
            if stall > 80:
                step *= 0.85
                if step < 1e-12:
                    break
                rng = np.random.default_rng(it * 31 + 7)
                a = a + rng.normal(0, 0.02, n) * step * 10
                a = np.maximum(a, 0)
                s = np.sum(a)
                if s > 1e-10:
                    a = a / s
                stall = 0

        if it > 50:
            step *= 0.997

    return best_a, best_obj


def random_init(n, seed):
    """Generate diverse random initializations."""
    rng = np.random.default_rng(seed)
    k = np.arange(n)
    choice = rng.integers(0, 12)

    if choice == 0:
        f = rng.uniform(0.5, 4.0)
        phase = rng.uniform(0, 2 * np.pi)
        a = 1.0 + 0.6 * np.cos(2 * np.pi * f * k / n + phase)
    elif choice == 1:
        f1 = rng.uniform(0.3, 2.0)
        f2 = rng.uniform(2.0, 6.0)
        a = 1.0 + 0.3 * np.cos(2 * np.pi * f1 * k / n) + 0.2 * np.cos(2 * np.pi * f2 * k / n)
    elif choice == 2:
        a = rng.random(n)
    elif choice == 3:
        # Two-peak structure
        c1 = n * rng.uniform(0.2, 0.4)
        c2 = n * rng.uniform(0.6, 0.8)
        w = n * rng.uniform(0.05, 0.15)
        a = 1.0 + np.exp(-0.5 * ((k - c1) / w) ** 2) + np.exp(-0.5 * ((k - c2) / w) ** 2)
    elif choice == 4:
        # Periodic with modulation
        f = rng.uniform(1.0, 3.0)
        mod_f = rng.uniform(0.1, 0.5)
        a = 1.0 + 0.4 * np.cos(2 * np.pi * f * k / n) * np.cos(2 * np.pi * mod_f * k / n)
    elif choice == 5:
        # Smoothed random
        a = rng.random(n)
        win = max(3, n // 15)
        kernel = np.ones(win) / win
        a = np.convolve(a, kernel, mode='same')
    elif choice == 6:
        # Inverse of random (emphasize valleys)
        a = rng.uniform(0.5, 2.0, n)
        win = max(3, n // 20)
        kernel = np.ones(win) / win
        a = np.convolve(a, kernel, mode='same')
        a = 1.0 / (a + 0.1)
    elif choice == 7:
        # Chirp
        f0 = rng.uniform(0.5, 2.0)
        f1 = rng.uniform(2.0, 5.0)
        a = 1.0 + 0.4 * np.cos(2 * np.pi * (f0 * k / n + (f1 - f0) * k ** 2 / (2 * n ** 2)))
    elif choice == 8:
        # Block structure
        nblocks = rng.integers(3, 8)
        a = np.zeros(n)
        for b in range(nblocks):
            start = b * n // nblocks
            end = min((b + 1) * n // nblocks, n)
            a[start:end] = rng.uniform(0.3, 1.5)
    elif choice == 9:
        # Power-law decay from center
        center = n / 2
        alpha = rng.uniform(0.5, 2.0)
        a = 1.0 + 0.5 * np.exp(-alpha * np.abs(k - center) / (n * 0.3))
    elif choice == 10:
        # Multi-frequency
        a = np.ones(n)
        for _ in range(rng.integers(2, 5)):
            f = rng.uniform(0.5, 5.0)
            amp = rng.uniform(0.1, 0.3)
            a += amp * np.cos(2 * np.pi * f * k / n + rng.uniform(0, 2 * np.pi))
    else:
        # Gaussian mixture-like
        centers = rng.uniform(0, n, 3)
        widths = rng.uniform(n * 0.05, n * 0.2, 3)
        a = np.ones(n)
        for c, w in zip(centers, widths):
            a += 0.5 * np.exp(-0.5 * ((k - c) / w) ** 2)

    a = np.maximum(a, 0.01)
    s = np.sum(a)
    return a / s


def resample_sequence(a, new_n):
    """Resample sequence to new length via interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    # Periodic interpolation for smoother results
    x_old = np.linspace(0, 1, n, endpoint=False)
    x_new = np.linspace(0, 1, new_n, endpoint=False)
    a_ext = np.concatenate([a, a])  # periodic extension
    x_ext = np.concatenate([x_old, x_old + 1])
    a_new = np.interp(x_new, x_ext, a_ext)
    return a_new


def try_parent():
    """Try to load parent sequence."""
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64)
        parent = np.clip(parent, 0, 1000.0)
        if parent.size > 0 and np.isfinite(parent).all():
            obj = compute_obj(parent)
            print(f"Parent (n={parent.size}): obj={obj:.6f}")
            return parent.copy(), obj
    except Exception:
        pass
    return None, np.inf


def polish_sequence(a, time_budget, verbose=False):
    """Intensive polishing combining LP, L-BFGS-B, and subgradient."""
    n = len(a)
    a = a.copy().astype(np.float64)
    start = time.monotonic()

    best_a = a.copy()
    best_obj = compute_obj(a)
    round_num = 0

    while time.monotonic() - start < time_budget - 2:
        round_num += 1
        remaining = time_budget - (time.monotonic() - start)
        if remaining < 8:
            break

        # LP step (most powerful)
        if remaining > 20 and n <= 800:
            lp_budget = min(remaining * 0.4, 45.0)
            a_new, obj_new = lp_alternating(best_a, max_iters=40, time_budget=lp_budget)
            if obj_new < best_obj:
                best_obj = obj_new
                best_a = a_new.copy()
                if verbose:
                    print(f"    Polish r{round_num} LP: {best_obj:.6f}")

        # L-BFGS-B refinement
        remaining = time_budget - (time.monotonic() - start)
        if remaining > 15:
            tau_list = [1.0 / n, 0.5 / n, 0.2 / n, 0.1 / n]
            lfbs_budget = min(remaining * 0.25, 20.0)
            a_new, obj_new = lfbs_optimize(best_a, tau_list, max_iter=200, time_budget=lfbs_budget)
            if obj_new < best_obj:
                best_obj = obj_new
                best_a = a_new.copy()
                if verbose:
                    print(f"    Polish r{round_num} L-BFGS: {best_obj:.6f}")

        # Subgradient
        remaining = time_budget - (time.monotonic() - start)
        if remaining > 10:
            sg_budget = min(remaining * 0.2, 12.0)
            a_new = subgradient_optimize(best_a, max_iters=int(sg_budget * 30),
                                         time_budget=sg_budget, tau_start=5.0, tau_end=30.0)
            obj_new = compute_obj(a_new)
            if obj_new < best_obj:
                best_obj = obj_new
                best_a = a_new.copy()
                if verbose:
                    print(f"    Polish r{round_num} SG: {best_obj:.6f}")

        # Check for convergence
        remaining = time_budget - (time.monotonic() - start)
        if best_obj < 1.5030 and remaining < 30:
            break

    return best_a, best_obj


def propose_candidate(seed=42, budget_s=1500):
    """
    Main search: LP-centric optimization with L-BFGS-B refinement.
    Focuses on sequence lengths 400-900 with emphasis on 500-700.
    Uses aggressive LP alternating, diverse initializations, and perturbations.
    """
    start_time = time.monotonic()
    deadline = start_time + float(budget_s) - 30.0

    best_seq = None
    best_score = np.inf

    # Load parent
    parent, parent_obj = try_parent()
    if parent is not None and parent_obj < best_score:
        best_score = parent_obj
        best_seq = parent.copy()

    # N values to explore - focus on 400-900
    n_values = [400, 450, 500, 550, 600, 650, 700, 750, 800, 900]
    if best_seq is not None:
        np_ = len(best_seq)
        extra = [np_, max(300, np_ - 100), min(1000, np_ + 50), min(1000, np_ + 100),
                 min(1000, np_ + 150)]
        n_values = sorted(set(n_values + extra))

    print(f"Exploring n values: {n_values}")
    print(f"Initial best: {best_score:.6f}")
    print(f"Time budget: {budget_s}s, effective deadline in {deadline - start_time:.0f}s")

    candidates = []

    # Phase 1: Exploration with multiple (n, init) combinations
    for n in n_values:
        remaining = deadline - time.monotonic()
        if remaining < 30:
            break

        # Time allocation: more time for sizes near the parent
        if best_seq is not None:
            dist = abs(n - len(best_seq))
            alloc = max(20.0, 60.0 - dist * 0.05)
        else:
            alloc = 40.0
        alloc = min(alloc, remaining * 0.25)
        alloc = max(15.0, alloc)

        print(f"\nN={n}: {remaining:.0f}s remaining, alloc={alloc:.0f}s")

        n_inits = max(2, int(alloc / 18))
        n_inits = min(n_inits, 4)

        for init_idx in range(n_inits):
            if time.monotonic() > deadline - 30:
                break
            remaining_init = deadline - time.monotonic()
            if remaining_init < 15:
                break

            # Initialize
            if init_idx == 0 and best_seq is not None:
                a = resample_sequence(best_seq, n)
            else:
                a = random_init(n, seed + n * 99991 + init_idx * 79193)

            # Quick subgradient shaping
            sg_time = min(6.0, (alloc / n_inits) * 0.2)
            a = subgradient_optimize(a, max_iters=int(sg_time * 30),
                                      time_budget=sg_time, tau_start=2.0, tau_end=15.0)

            # L-BFGS-B with tau annealing
            remaining_init = deadline - time.monotonic()
            if remaining_init > 15:
                lfbs_time = min((alloc / n_inits) * 0.3, 20.0)
                tau_list = [3.0 / n, 1.5 / n, 0.7 / n, 0.3 / n, 0.1 / n]
                a, obj = lfbs_optimize(a, tau_list, max_iter=200, time_budget=lfbs_time)
            else:
                obj = compute_obj(a)

            # LP alternating (main workhorse)
            remaining_init = deadline - time.monotonic()
            if remaining_init > 20 and n <= 800:
                lp_time = min((alloc / n_inits) * 0.4, 30.0)
                a, obj = lp_alternating(a, max_iters=50, time_budget=lp_time)

            # Quick subgradient polish
            remaining_init = deadline - time.monotonic()
            if remaining_init > 10:
                a = subgradient_optimize(a, max_iters=int(remaining_init * 10),
                                          time_budget=min(remaining_init * 0.1, 8.0),
                                          tau_start=5.0, tau_end=25.0)
                obj = compute_obj(a)

            print(f"  init{init_idx}: n={n}, obj={obj:.6f}")

            if obj < best_score:
                best_score = obj
                best_seq = a.copy()
                print(f"  ** UPDATE: n={n}, obj={best_score:.6f} **")

            candidates.append((a.copy(), obj, n))

    # Phase 2: Polishing top candidates
    print(f"\n=== POLISHING PHASE ===")
    print(f"Best before polish: {best_score:.6f}")

    candidates.sort(key=lambda x: x[1])
    n_polish = min(3, len(candidates))

    for ci in range(n_polish):
        if time.monotonic() > deadline - 45:
            break
        remaining = deadline - time.monotonic()
        if remaining < 25:
            break

        a_cand = candidates[ci][0]
        n_cand = candidates[ci][2]
        print(f"  Polishing candidate {ci} (n={n_cand}, obj={candidates[ci][1]:.6f}), {remaining:.0f}s left")

        polish_budget = min(remaining * 0.35, 70.0)
        a_pol, obj_pol = polish_sequence(a_cand, time_budget=polish_budget, verbose=True)

        if obj_pol < best_score:
            best_score = obj_pol
            best_seq = a_pol.copy()
            print(f"  ** UPDATE after polish: obj={best_score:.6f} **")

        # Perturbation attempts
        remaining = deadline - time.monotonic()
        if remaining > 20 and obj_pol < best_score + 0.01:
            rng = np.random.default_rng(ci * 555 + 999 + seed)
            for ptry in range(4):
                if time.monotonic() > deadline - 20:
                    break
                scale = rng.uniform(0.02, 0.10)
                a_pert = a_cand * np.exp(rng.normal(0, scale, n_cand))
                a_pert = np.maximum(a_pert, 0)
                s = np.sum(a_pert)
                if s < 1e-10:
                    continue
                a_pert = a_pert / s
                # Quick LP polish
                a_pert, obj_pert = lp_alternating(a_pert, max_iters=30, time_budget=8.0)
                if obj_pert < best_score:
                    best_score = obj_pert
                    best_seq = a_pert.copy()
                    print(f"  ** UPDATE (perturb {ptry}): obj={best_score:.6f} **")

    # Phase 3: Final intensive polish on the global best
    if best_seq is not None and time.monotonic() < deadline - 20:
        remaining = deadline - time.monotonic()
        n_best = len(best_seq)
        print(f"\n=== FINAL POLISH: n={n_best}, {remaining:.0f}s ===")

        a_cur = best_seq.copy()
        for final_round in range(5):
            if time.monotonic() > deadline - 10:
                break
            remaining = deadline - time.monotonic()
            if remaining < 10:
                break

            # LP (most important)
            if n_best <= 800:
                lp_budget = min(remaining * 0.3, 30.0)
                a_new, obj_new = lp_alternating(a_cur, max_iters=40, time_budget=lp_budget)
                if obj_new < best_score:
                    best_score = obj_new
                    best_seq = a_new.copy()
                    a_cur = a_new.copy()
                    print(f"  Final LP {final_round}: {best_score:.6f}")

            # L-BFGS-B with very small tau
            remaining = deadline - time.monotonic()
            if remaining > 10:
                tau_vsmall = 0.05 / n_best
                a_new, obj_new = lfbs_optimize(a_cur, [0.3 / n_best, 0.15 / n_best, tau_vsmall],
                                               max_iter=300, time_budget=min(remaining * 0.2, 15.0))
                if obj_new < best_score:
                    best_score = obj_new
                    best_seq = a_new.copy()
                    a_cur = a_new.copy()
                    print(f"  Final L-BFGS {final_round}: {best_score:.6f}")

            # Subgradient
            remaining = deadline - time.monotonic()
            if remaining > 8:
                a_new = subgradient_optimize(a_cur, max_iters=int(remaining * 10),
                                              time_budget=min(remaining * 0.15, 10.0),
                                              tau_start=10.0, tau_end=50.0)
                obj_new = compute_obj(a_new)
                if obj_new < best_score:
                    best_score = obj_new
                    best_seq = a_new.copy()
                    a_cur = a_new.copy()
                    print(f"  Final SG {final_round}: {best_score:.6f}")

    # Phase 4: Try a few more lengths near the best if time permits
    if best_seq is not None and time.monotonic() < deadline - 30:
        n_best = len(best_seq)
        nearby_ns = [n for n in [n_best - 50, n_best - 25, n_best + 25, n_best + 50]
                     if 200 <= n <= 1000 and n != n_best]
        for n_try in nearby_ns[:2]:
            if time.monotonic() > deadline - 20:
                break
            remaining = deadline - time.monotonic()
            if remaining < 15:
                break
            a_try = resample_sequence(best_seq, n_try)
            a_try, obj_try = lp_alternating(a_try, max_iters=30, time_budget=min(remaining * 0.6, 20.0))
            print(f"  Nearby n={n_try}: obj={obj_try:.6f}")
            if obj_try < best_score:
                best_score = obj_try
                best_seq = a_try.copy()
                print(f"  ** UPDATE (nearby): obj={best_score:.6f} **")

    # Final validation
    if best_seq is None:
        best_seq = np.ones(600)
        best_score = compute_obj(best_seq)

    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    val = evaluate_sequence(result)
    if val < np.inf and val < best_score:
        best_score = val
    elif val == np.inf:
        result = [1.0] * 600
        best_score = 2.0

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_score:.6f}, time={elapsed:.1f}s ===")

    return result
