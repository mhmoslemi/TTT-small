def compute_conv(a):
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    conv = np.fft.irfft(fa * fa, size)
    return conv[:2 * n - 1]


def compute_obj(a):
    n = len(a)
    if n == 0:
        return np.inf
    conv = compute_conv(a)
    max_b = float(np.max(conv))
    sum_a = float(np.sum(a))
    if sum_a < 0.01:
        return np.inf
    return 2.0 * n * max_b / (sum_a * sum_a)


def build_toeplitz(a):
    n = len(a)
    A = np.zeros((2 * n - 1, n), dtype=np.float64)
    for j in range(n):
        k_start = j
        k_end = min(j + n, 2 * n - 1)
        A[k_start:k_end, j] = a[:k_end - k_start]
    return A


def solve_lp_maxsum(a, M, time_limit=5):
    """LP: max sum(g) s.t. (a*g)[k] <= M for all k, g >= 0."""
    n = len(a)
    if HAS_CVXPY:
        try:
            g_var = cp.Variable(n, nonneg=True)
            A = build_toeplitz(a)
            prob = cp.Problem(cp.Maximize(cp.sum(g_var)), [A @ g_var <= M])
            for solver in [cp.MOSEK, cp.CLARABEL, cp.SCS]:
                try:
                    prob.solve(solver=solver, time_limit=int(time_limit))
                    if g_var.value is not None:
                        return np.maximum(g_var.value, 0)
                except Exception:
                    continue
        except Exception:
            pass
    
    A = build_toeplitz(a)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': time_limit})
        if res.status == 0:
            return np.maximum(res.x, 0)
    except Exception:
        pass
    return None

def spectral_init(n, seed=42):

def compute_conv(a):
    """Compute conv(a,a) using FFT. Returns array of length 2n-1."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    conv = np.fft.irfft(fa * fa, size)
    return conv[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    conv = compute_conv(a)
    max_b = float(np.max(conv))
    sum_a = float(np.sum(a))
    if sum_a < 0.01:
        return np.inf
    return 2.0 * n * max_b / (sum_a * sum_a)


def build_toeplitz(a):
    """Build Toeplitz matrix A where (A@g)[k] = (a*g)[k] for k=0..2n-2."""
    n = len(a)
    A = np.zeros((2 * n - 1, n), dtype=np.float64)
    for j in range(n):
        k_start = j
        k_end = min(j + n, 2 * n - 1)
        A[k_start:k_end, j] = a[:k_end - k_start]
    return A


def solve_lp_maxsum(a, M, time_limit=4):
    """Solve LP: max sum(g) s.t. (a*g)[k] <= M for all k, g >= 0."""
    n = len(a)
    if HAS_CVXPY:
        try:
            g_var = cp.Variable(n, nonneg=True)
            A = build_toeplitz(a)
            prob = cp.Problem(cp.Maximize(cp.sum(g_var)), [A @ g_var <= M])
            for solver in [cp.MOSEK, cp.CLARABEL, cp.SCS]:
                try:
                    prob.solve(solver=solver, time_limit=int(time_limit))
                    if g_var.value is not None:
                        return np.maximum(g_var.value, 0)
                except Exception:
                    continue
        except Exception:
            pass

    A = build_toeplitz(a)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': time_limit})
        if res.status == 0:
            return np.maximum(res.x, 0)
    except Exception:
        pass
    return None


def solve_lp_minpeak(a, time_limit=4):

def nesterov_subgradient(a, time_budget=20.0, tau_init=15.0, lr_init=0.08):

        return a

    tau = tau_init
    lr = lr_init
    best_a = a.copy()
    best_obj = compute_obj(a)

    stall = 0

        shifted = np.clip(shifted, -500, 500)
        weights = np.exp(shifted)
        w_sum = np.sum(weights)
        if w_sum < 1e-300:
            break
        weights /= w_sum

        c = np.correlate(weights, y, mode='full')
        grad = 2.0 * c[n - 1:n - 1 + n]
        grad = grad - np.mean(grad)

        # Multiplicative update on y
        log_update = -lr * grad
        log_update = np.clip(log_update, -50, 50)
        a_new = y * np.exp(log_update)
        a_new = np.maximum(a_new, 1e-15)
        a_new = a_new / np.sum(a_new)

        # Nesterov momentum
        mu_new = (1.0 + np.sqrt(1.0 + 4.0 * mu * mu)) / 2.0
        y = a_new + ((mu_new - 1.0) / mu) * (a_new - a)
        y = np.maximum(y, 1e-15)
        y = y / np.sum(y)
        mu = mu_new
        a = a_new

                mu = 1.0
                y = a.copy()
                if lr < 0.002:
                    break
                stall = 0

        if it > 200:

def bisection_line_search(a, g, best_obj, max_iter=12):
    """Bisection line search for (1-t)*a + t*g."""
    n = len(a)
    lo = 0.0
    hi = 1.0
    best_t = 0.0
    best_obj_ls = best_obj

    for _ in range(max_iter):
        mid = (lo + hi) / 2.0
        a_try = (1.0 - mid) * a + mid * g
        a_try = np.maximum(a_try, 0)
        s_try = np.sum(a_try)
        if s_try < 1e-10:
            hi = mid
            continue
        a_try = a_try / s_try
        obj_try = compute_obj(a_try)
        if obj_try < best_obj_ls:
            best_obj_ls = obj_try
            best_t = mid
            lo = mid
        else:
            hi = mid

    return best_t, best_obj_ls

def resample_sequence(a, new_n):
    """Resample a sequence to a new length using cubic interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    old_k = np.arange(n)
    new_k = np.linspace(0, n - 1, new_n)
    # Use cubic spline-like interpolation via np.interp (linear) as fallback
    a_new = np.interp(new_k, old_k, a)
    a_new = np.maximum(a_new, 0)
    s = np.sum(a_new)
    if s > 1e-10:
        a_new = a_new / s
    return a_new

        win = max(3, n // 15)
        kernel = np.ones(win) / win
        a = np.convolve(a, kernel, mode='same')
    elif choice == 4:
        a = np.ones(n)
        for scale in range(3):
            f = rng.uniform(0.3 * (2 ** scale), 1.5 * (2 ** scale))
            amp = 0.3 / (scale + 1)
            a += amp * np.cos(2 * np.pi * f * k / n + rng.uniform(0, 2 * np.pi))
    elif choice == 5:
        # Chirp
        f0 = rng.uniform(0.3, 1.0)
        f1 = rng.uniform(3.0, 8.0)
        t = k / n
        f = f0 + (f1 - f0) * t
        a = 1.0 + 0.5 * np.cos(2 * np.pi * np.cumsum(f))
    elif choice == 6:
        # Random smooth
        a = rng.random(max(5, n // 8))
        a = np.interp(k, np.arange(len(a)), a)
    else:
        # Raised cosine with random modulation
        a = 0.5 * (1 + np.cos(2 * np.pi * k / n))
        f = rng.uniform(1, 4)
        a *= 1 + 0.2 * np.cos(2 * np.pi * f * k / n)

    a = np.maximum(a, 0)
    s = np.sum(a)
    if s < 1e-10:
        a = np.ones(n) / n
    else:
        a = a / s
    return a


def try_parent():
    """Try to load parent sequence."""
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64)
        parent = np.clip(parent, 0, 1000.0)
        if parent.size > 0 and np.isfinite(parent).all():
            obj = compute_obj(parent)
            print(f"Parent (n={parent.size}): obj={obj:.6f}")
            return parent.copy(), obj
    except Exception:
        pass
    return None, np.inf


def optimize_single(a, time_budget=60.0, verbose=False):
    """Run full optimization pipeline on a single sequence."""
    n = len(a)
    a = a.copy().astype(np.float64)
    s = np.sum(a)
    if s < 1e-10:
        a = np.ones(n) / n
        s = 1.0
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    if verbose:
        print(f"  Start: obj={best_obj:.8f}")

    # Phase 1: Quick Nesterov subgradient
    remaining = time_budget - (time.monotonic() - start)
    if remaining > 10:
        sg_time = min(10.0, remaining * 0.12)
        a_new = nesterov_subgradient(best_a, time_budget=sg_time, tau_init=15.0, lr_init=0.06)
        obj = compute_obj(a_new)
        if obj < best_obj:
            best_obj = obj
            best_a = a_new.copy()
            if verbose:
                print(f"  After Nesterov SG: {best_obj:.8f}")

    # Phase 2: Aggressive alternating LP (standard)
    remaining = time_budget - (time.monotonic() - start)
    if remaining > 15:
        lp_time = min(30.0, remaining * 0.3)
        a_new = aggressive_alternating_lp(best_a, time_budget=lp_time, use_minpeak=False)
        obj = compute_obj(a_new)
        if obj < best_obj:
            best_obj = obj
            best_a = a_new.copy()
            if verbose:
                print(f"  After LP: {best_obj:.8f}")

    # Phase 3: Peak polish
    remaining = time_budget - (time.monotonic() - start)
    if remaining > 10:
        pp_time = min(15.0, remaining * 0.2)
        a_new = peak_polish(best_a, time_budget=pp_time)
        obj = compute_obj(a_new)
        if obj < best_obj:
            best_obj = obj
            best_a = a_new.copy()
            if verbose:
                print(f"  After Peak Polish: {best_obj:.8f}")

    # Phase 4: Min-peak LP
    remaining = time_budget - (time.monotonic() - start)
    if remaining > 15:
        lp_time = min(25.0, remaining * 0.25)
        a_new = aggressive_alternating_lp(best_a, time_budget=lp_time, use_minpeak=True)
        obj = compute_obj(a_new)
        if obj < best_obj:
            best_obj = obj
            best_a = a_new.copy()
            if verbose:
                print(f"  After LP-MP: {best_obj:.8f}")

    # Phase 5: Nesterov subgradient with higher tau
    remaining = time_budget - (time.monotonic() - start)
    if remaining > 10:
        sg_time = min(12.0, remaining * 0.2)
        a_new = nesterov_subgradient(best_a, time_budget=sg_time, tau_init=40.0, lr_init=0.03)
        obj = compute_obj(a_new)
        if obj < best_obj:
            best_obj = obj
            best_a = a_new.copy()
            if verbose:
                print(f"  After SG2: {best_obj:.8f}")

    # Phase 6: Final peak polish
    remaining = time_budget - (time.monotonic() - start)
    if remaining > 8:
        pp_time = min(10.0, remaining * 0.3)
        a_new = peak_polish(best_a, time_budget=pp_time)
        obj = compute_obj(a_new)
        if obj < best_obj:
            best_obj = obj
            best_a = a_new.copy()
            if verbose:
                print(f"  After Final Polish: {best_obj:.8f}")

    return best_a, best_obj


def optimize_n(n, seed, time_budget, parent_seq=None, verbose=False):
    """Run optimization for a given n with multiple initializations."""
    start = time.monotonic()
    best_a = None
    best_obj = np.inf

    n_inits = max(2, min(4, int(time_budget / 25)))

    for init_idx in range(n_inits):
        if time.monotonic() - start > time_budget - 12:
            break
        remaining = time_budget - (time.monotonic() - start)
        if remaining < 20:
            break

        if init_idx == 0 and parent_seq is not None and len(parent_seq) != n:
            a = resample_sequence(parent_seq, n)
        elif init_idx == 0 and parent_seq is not None and len(parent_seq) == n:
            a = parent_seq.copy()
        else:
            a = random_init(n, seed + init_idx * 79193)

        sub_budget = min(remaining * 0.65, 70.0)
        a_opt, obj = optimize_single(a, time_budget=sub_budget, verbose=(verbose and init_idx == 0))

        if obj < best_obj:
            best_obj = obj
            best_a = a_opt.copy()

    if best_a is None:
        best_a = np.ones(n) / n
        best_obj = compute_obj(best_a)

    return best_a, best_obj


def propose_candidate(seed=42, budget_s=1500):
    """
    Main search combining aggressive LP, Nesterov subgradient, peak polishing,
    and multi-scale local search. Explores n values around the parent length.
    """
    start_time = time.monotonic()
    deadline = start_time + float(budget_s) - 30.0

    best_seq = None
    best_score = np.inf

    parent, parent_obj = try_parent()
    if parent is not None and parent_obj < best_score:
        best_score = parent_obj
        best_seq = parent.copy()
        n_parent = len(parent)
    else:
        n_parent = 330

    print(f"Initial best: {best_score:.6f} (n={n_parent if best_seq is not None else 'N/A'})")
    print(f"Time budget: {budget_s}s")

    # Phase 1: Intensive optimization of parent
    if best_seq is not None:
        print("\n=== Phase 1: Intensive optimization of parent ===")
        remaining = deadline - time.monotonic()
        p1_budget = min(remaining * 0.25, 300.0)

        a_opt, obj = optimize_single(best_seq, time_budget=p1_budget, verbose=True)
        if obj < best_score:
            best_score = obj
            best_seq = a_opt.copy()
            print(f"  After optimize_single: {best_score:.8f}")

        # Multi-scale local search
        remaining = deadline - time.monotonic()
        p1b_budget = min(remaining * 0.15, 150.0)
        if p1b_budget > 20:
            a_opt = multi_scale_local_search(best_seq, time_budget=p1b_budget, verbose=True)
            obj = compute_obj(a_opt)
            if obj < best_score:
                best_score = obj
                best_seq = a_opt.copy()
                print(f"  After MLS: {best_score:.8f}")

    # Phase 2: Try nearby n values
    print("\n=== Phase 2: Nearby n values ===")
    n_values = []
    for delta in range(-40, 45, 5):
        n_val = n_parent + delta
        if 150 <= n_val <= 800:
            n_values.append(n_val)
    # Deduplicate and sort by distance from parent
    n_values = sorted(set(n_values), key=lambda x: abs(x - n_parent))

    remaining = deadline - time.monotonic()
    if remaining > 30:
        for n_val in n_values:
            if time.monotonic() > deadline - 40:
                break
            remaining = deadline - time.monotonic()
            if remaining < 25:
                break
            if n_val == n_parent:
                continue  # Already done in Phase 1

            alloc = min(50.0, remaining * 0.15)
            alloc = max(alloc, 20.0)

            print(f"\n  n={n_val} ({remaining:.0f}s remaining, alloc={alloc:.0f}s)")
            a_opt, obj = optimize_n(n_val, seed=seed + n_val * 104729,
                                     time_budget=alloc, parent_seq=parent, verbose=False)
            if obj < best_score:
                best_score = obj
                best_seq = a_opt.copy()
                print(f"  ** UPDATE: n={n_val}, obj={best_score:.8f} **")

    # Phase 3: Final intensive optimization on best
    if best_seq is not None:
        remaining = deadline - time.monotonic()
        if remaining > 30:
            print(f"\n=== Phase 3: Final optimization ({remaining:.0f}s) ===")
            p3_budget = min(remaining * 0.4, remaining - 15.0)
            a_opt, obj = optimize_single(best_seq, time_budget=p3_budget, verbose=True)
            if obj < best_score:
                best_score = obj
                best_seq = a_opt.copy()
                print(f"  After Phase 3 opt: {best_score:.8f}")

            remaining = deadline - time.monotonic()
            if remaining > 20:
                p3b_budget = min(remaining * 0.35, remaining - 10.0)
                a_opt = multi_scale_local_search(best_seq, time_budget=p3b_budget, verbose=True)
                obj = compute_obj(a_opt)
                if obj < best_score:
                    best_score = obj
                    best_seq = a_opt.copy()
                    print(f"  After Phase 3 MLS: {best_score:.8f}")

    # Phase 4: Final peak polish
    if best_seq is not None:
        remaining = deadline - time.monotonic()
        if remaining > 15:
            print(f"\n=== Phase 4: Final peak polish ({remaining:.0f}s) ===")
            pp_budget = min(remaining * 0.5, remaining - 10.0)
            a_opt = peak_polish(best_seq, time_budget=pp_budget)
            obj = compute_obj(a_opt)
            if obj < best_score:
                best_score = obj
                best_seq = a_opt.copy()
                print(f"  After final polish: {best_score:.8f}")

    # Final validation
    if best_seq is None:
        best_seq = np.ones(50)
        best_score = compute_obj(best_seq)

    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    val = evaluate_sequence(result)
    if val < best_score:
        best_score = val

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(best_seq)}, score={best_score:.8f}, time={elapsed:.1f}s ===")

    return result
