Optimized step function using logsumexp-smoothed gradient descent with adaptive tau,

Key improvements over previous:

- L-BFGS-B for fast local convergence

import numpy as np
from scipy.optimize import linprog, minimize
import time

def compute_conv(a):
    """Compute conv(a,a) using FFT. Returns array of length 2n-1."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    conv = np.fft.irfft(fa * fa, size)
    return conv[:2 * n - 1]

def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    conv = compute_conv(a)
    max_b = float(np.max(conv))
    sum_a = float(np.sum(a))
    if sum_a < 0.01:
        return np.inf
    return 2.0 * n * max_b / (sum_a * sum_a)

def smoothed_obj_and_grad(a, tau):
    """Compute smoothed objective and gradient.
    s(a) = (1/tau) * log(sum_k exp(tau * conv_k(a)))
    grad[j] = 2 * sum_k w_k * a[k-j]
    where w_k = exp(tau * conv_k) / sum_k exp(tau * conv_k)
    """
    n = len(a)
    conv = compute_conv(a)
    # Numerical stability
    max_conv = np.max(conv)
    exp_conv = np.exp(tau * (conv - max_conv))
    w = exp_conv / np.sum(exp_conv)
    obj = (1.0 / tau) * (np.log(np.sum(exp_conv)) + tau * max_conv)
    # Gradient
    grad = 2.0 * np.correlate(w, a, mode='full')[n-1:2*n-1]

        if obj < best_obj - 1e-14:
            best_obj = obj
            best_a = a.copy()

"""
Step function optimization using:
1. LP-based alternating method for global optimization
2. Logsumexp-smoothed L-BFGS-B for fast local convergence  
3. Momentum-based gradient descent as backup
4. Adaptive tau annealing
5. Multi-n exploration

import numpy as np
from scipy.optimize import linprog, minimize
import time

# All helper functions at top level

def compute_conv(a):
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    conv = np.fft.irfft(fa * fa, size)
    return conv[:2 * n - 1]

def compute_obj(a):
    n = len(a)
    if n == 0:
        return np.inf
    conv = compute_conv(a)
    max_b = float(np.max(conv))
    sum_a = float(np.sum(a))
    if sum_a < 0.01:
        return np.inf
    return 2.0 * n * max_b / (sum_a * sum_a)

def build_toeplitz(a):
    n = len(a)
    A = np.zeros((2 * n - 1, n), dtype=np.float64)
    for j in range(n):
        k_start = j
        k_end = min(j + n, 2 * n - 1)
        A[k_start:k_end, j] = a[:k_end - k_start]
    return A

def solve_lp_maxsum(a, M, time_limit=5):
    n = len(a)
    A = build_toeplitz(a)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': time_limit})
        if res.status == 0:
            g = np.maximum(res.x, 0)
            return g, float(np.sum(g))
    except Exception:
        pass
    return None, None

def lp_alternating(a, max_iters=50, time_budget=100.0):
    """LP-based alternating optimization."""
    n = len(a)
    a = a.copy().astype(np.float64)
    s = np.sum(a)
    if s < 1e-10:
        return a
    a = a / s
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    stall = 0
    
    for it in range(max_iters):
        if time.monotonic() - start > time_budget - 5:
            break
        
        conv = compute_conv(a)
        M = float(np.max(conv))
        
        g, sum_g = solve_lp_maxsum(a, M, time_limit=4)
        if g is None or sum_g < 1e-10:
            break
        
        g_norm = g / sum_g
        
        best_t = 0.0
        best_obj_ls = best_obj
        for t in [0.1, 0.2, 0.3, 0.5, 0.7, 1.0]:
            a_new = (1.0 - t) * a + t * g_norm
            a_new = np.maximum(a_new, 0)
            s_new = np.sum(a_new)
            if s_new < 1e-10:
                continue
            a_new = a_new / s_new
            obj_new = compute_obj(a_new)
            if obj_new < best_obj_ls:
                best_obj_ls = obj_new
                best_t = t
        
        if best_t > 0 and best_obj_ls < best_obj - 1e-13:
            a_new = (1.0 - best_t) * a + best_t * g_norm
            a_new = np.maximum(a_new, 0)
            s_new = np.sum(a_new)
            if s_new > 1e-10:
                a_new = a_new / s_new
            best_obj = best_obj_ls
            best_a = a_new.copy()
            a = a_new
            stall = 0
        else:
            stall += 1
            if stall > 20:
                break
            # Small perturbation
            rng = np.random.default_rng(it * 12345)
            a = a * np.exp(rng.normal(0, 0.05, n))
            a = np.maximum(a, 1e-10)
            s = np.sum(a)
            a = a / s
            stall = 0
    
    return best_a

def softmax_z_to_a(z):
    z_shifted = z - np.max(z)
    ez = np.exp(z_shifted)
    return ez / np.sum(ez)

def smooth_obj_z(z, tau):
    """Smoothed objective in z-parameterization."""
    a = softmax_z_to_a(z)
    conv = compute_conv(a)
    max_c = np.max(conv)
    return float(np.log(np.sum(np.exp(tau * (conv - max_c)))) / tau + max_c)

def smooth_grad_z(z, tau):
    """Gradient of smoothed objective w.r.t. z."""
    a = softmax_z_to_a(z)
    n = len(a)
    conv = compute_conv(a)
    max_c = np.max(conv)
    exp_c = np.exp(tau * (conv - max_c))
    w = exp_c / np.sum(exp_c)
    grad_a = 2.0 * np.correlate(w, a, mode='full')[n-1:2*n-1]
    grad_z = a * (grad_a - np.dot(a, grad_a))
    return grad_z

    n = len(a)
    a = a.copy().astype(np.float64)
    s = np.sum(a)
    if s < 1e-10:
        return a
    a = a / s
    
    # Convert to z-parameterization
    z = np.log(a + 1e-15)  # initial z
    # Actually, let's use z = log(a) - mean(log(a)) for centering
    z = np.log(a + 1e-15) - np.mean(np.log(a + 1e-15))
    
    best_a = a.copy()
    best_obj = compute_obj(a)

    # Multiple rounds with increasing tau
    tau_rounds = [tau_start, tau_start*3, tau_start*9, tau_start*27, tau_start*81, tau_end]

            break
        
        remaining = time_budget - (time.monotonic() - start)
        if remaining < 15:
            break
        
        # Define objective and gradient for scipy
        def obj_func(z):
            return smooth_obj_z(z, tau)
        
        def grad_func(z):
            return smooth_grad_z(z, tau)
        
        try:
            res = minimize(obj_func, z, jac=grad_func, method='L-BFGS-B',
                          options={'maxiter': 500, 'maxfun': 10000,
                                  'ftol': 1e-14, 'gtol': 1e-10,
                                  'maxls': 50})
            if res.x is not None:
                a_new = softmax_z_to_a(res.x)
                obj_new = compute_obj(a_new)
                if obj_new < best_obj:
                    best_obj = obj_new
                    best_a = a_new.copy()
                    z = res.x.copy()
        except Exception as e:
            continue
    
    return best_a

def momentum_gradient(a, time_budget=60.0, tau=50.0):

    n = len(a)

        return a

    return best_a

def smoothed_obj_and_grad(a, tau):
    """Compute smoothed objective and gradient in a-space."""
    n = len(a)
    conv = compute_conv(a)
    max_c = np.max(conv)

    obj = float(np.log(np.sum(exp_c)) / tau + max_c)

"""
Step function optimization using logsumexp-smoothed L-BFGS-B with adaptive tau annealing,
combined with LP-based alternating optimization and intensive local search.
Key improvements over previous approach:
- Exact gradient computation for the smoothed max-convolution objective
- L-BFGS-B for superlinear convergence (vs. sublinear subgradient)
- Multi-round adaptive tau annealing for progressively sharper approximation
- Better time allocation focusing computational effort on the parent sequence
- Momentum gradient descent as complementary method
"""

import numpy as np
from scipy.optimize import linprog, minimize
import time

# Global variable for passing tau to wrapped functions (avoiding closures)
_current_tau = 1.0


def compute_conv(a):
    """Compute conv(a,a) using FFT. Returns array of length 2n-1."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    conv = np.fft.irfft(fa * fa, size)
    return conv[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    conv = compute_conv(a)
    max_b = float(np.max(conv))
    sum_a = float(np.sum(a))
    if sum_a < 0.01:
        return np.inf
    return 2.0 * n * max_b / (sum_a * sum_a)


def build_toeplitz(a):
    """Build Toeplitz matrix for convolution with a."""
    n = len(a)
    A = np.zeros((2 * n - 1, n), dtype=np.float64)
    for j in range(n):
        k_start = j
        k_end = min(j + n, 2 * n - 1)
        A[k_start:k_end, j] = a[:k_end - k_start]
    return A


def solve_lp_maxsum(a, M, time_limit=5):
    """Solve LP: max sum(g) s.t. (a*g)[k] <= M, g >= 0."""
    n = len(a)
    A = build_toeplitz(a)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': time_limit})
        if res.status == 0:
            g = np.maximum(res.x, 0)
            return g, float(np.sum(g))
    except Exception:
        pass
    return None, None


def solve_lp_minpeak(a, time_limit=5):
    """Solve LP: min M s.t. (a*g)[k] <= M, sum(g)=1, g>=0."""
    n = len(a)
    A = build_toeplitz(a)
    c = np.zeros(n + 1)
    c[-1] = 1.0
    A_ub = np.zeros((2 * n - 1, n + 1))
    A_ub[:, :n] = A
    A_ub[:, n] = -1.0
    b_ub = np.zeros(2 * n - 1)
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])
    bounds = [(0.0, None)] * n + [(0.0, None)]
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': time_limit})
        if res.status == 0:
            g = np.maximum(res.x[:n], 0)
            s = np.sum(g)
            if s < 1e-10:
                return None
            return g / s
    except Exception:
        pass
    return None


def lp_alternating(a, max_iters=60, time_budget=100.0):
    """LP-based alternating optimization with both maxsum and minpeak."""
    n = len(a)
    a = a.copy().astype(np.float64)
    s = np.sum(a)
    if s < 1e-10:
        return a
    a = a / s
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    stall = 0

    for it in range(max_iters):
        if time.monotonic() - start > time_budget - 5:
            break

        conv = compute_conv(a)
        M = float(np.max(conv))

        g, sum_g = solve_lp_maxsum(a, M, time_limit=4)
        if g is None or sum_g < 1e-10:
            # Try minpeak
            g_mp = solve_lp_minpeak(a, time_limit=4)
            if g_mp is None:
                break
            g_norm = g_mp
        else:
            g_norm = g / sum_g

        best_t = 0.0
        best_obj_ls = best_obj
        for t in [0.05, 0.1, 0.2, 0.3, 0.5, 0.7, 1.0]:
            a_new = (1.0 - t) * a + t * g_norm
            a_new = np.maximum(a_new, 0)
            s_new = np.sum(a_new)
            if s_new < 1e-10:
                continue
            a_new = a_new / s_new
            obj_new = compute_obj(a_new)
            if obj_new < best_obj_ls:
                best_obj_ls = obj_new
                best_t = t

        if best_t > 0 and best_obj_ls < best_obj - 1e-14:
            a_new = (1.0 - best_t) * a + best_t * g_norm
            a_new = np.maximum(a_new, 0)
            s_new = np.sum(a_new)
            if s_new > 1e-10:
                a_new = a_new / s_new
            best_obj = best_obj_ls
            best_a = a_new.copy()
            a = a_new
            stall = 0
        else:
            stall += 1
            if stall > 15:
                break
            rng = np.random.default_rng(it * 7919 + 31)
            a = a * np.exp(rng.normal(0, 0.04, n))
            a = np.maximum(a, 1e-10)
            a = a / np.sum(a)
            stall = 0

    return best_a


def softmax_z_to_a(z):
    """Convert z-vector to probability simplex via softmax."""
    z_shifted = z - np.max(z)
    ez = np.exp(z_shifted)
    return ez / np.sum(ez)


def smooth_obj_z(z, tau):
    """Smoothed objective in z-parameterization: (1/tau)*logsumexp(tau*conv(softmax(z)))."""
    a = softmax_z_to_a(z)
    conv = compute_conv(a)
    max_c = np.max(conv)
    return float(np.log(np.sum(np.exp(tau * (conv - max_c)))) / tau + max_c)


def smooth_grad_z(z, tau):
    """Gradient of smoothed objective w.r.t. z via chain rule through softmax."""
    a = softmax_z_to_a(z)
    n = len(a)
    conv = compute_conv(a)
    max_c = np.max(conv)
    exp_c = np.exp(tau * (conv - max_c))
    w = exp_c / np.sum(exp_c)
    # Gradient of smoothed max-conv w.r.t. a
    grad_a = 2.0 * np.correlate(w, a, mode='full')[n - 1:2 * n - 1]
    # Chain rule: dz_j = a_j * (grad_a_j - sum_i a_i * grad_a_i)
    grad_z = a * (grad_a - np.dot(a, grad_a))
    return grad_z


def _obj_wrapped(z):
    """Wrapper for scipy.minimize - uses global tau."""
    return smooth_obj_z(z, _current_tau)


def _grad_wrapped(z):
    """Wrapper for scipy.minimize - uses global tau."""
    return smooth_grad_z(z, _current_tau)


def lbfgs_optimize(a, time_budget=100.0, tau_schedule=None):
    """L-BFGS-B optimization with adaptive tau annealing via softmax parameterization."""
    global _current_tau
    n = len(a)
    a = a.copy().astype(np.float64)
    s = np.sum(a)
    if s < 1e-10:
        return a
    a = a / s

    if tau_schedule is None:
        tau_schedule = [5, 10, 20, 50, 100, 200, 400, 800]

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    no_improve_rounds = 0

    for tau in tau_schedule:
        if time.monotonic() - start > time_budget - 10:
            break
        remaining = time_budget - (time.monotonic() - start)
        if remaining < 15:
            break

        _current_tau = float(tau)

        # Initialize z from current best a
        z = np.log(best_a + 1e-15)
        z = z - np.mean(z)

        try:
            res = minimize(_obj_wrapped, z, jac=_grad_wrapped, method='L-BFGS-B',
                           options={'maxiter': 2000, 'maxfun': 50000,
                                   'ftol': 1e-16, 'gtol': 1e-12,
                                   'maxls': 50, 'maxcor': 20})
            if res.x is not None:
                a_new = softmax_z_to_a(res.x)
                obj_new = compute_obj(a_new)
                if obj_new < best_obj - 1e-15:
                    best_obj = obj_new
                    best_a = a_new.copy()
                    no_improve_rounds = 0
                else:
                    no_improve_rounds += 1
                    if no_improve_rounds >= 3:
                        break
        except Exception:
            # Fallback: try with fewer iterations
            try:
                res = minimize(_obj_wrapped, z, jac=_grad_wrapped, method='L-BFGS-B',
                               options={'maxiter': 200, 'maxfun': 5000})
                if res.x is not None:
                    a_new = softmax_z_to_a(res.x)
                    obj_new = compute_obj(a_new)
                    if obj_new < best_obj - 1e-15:
                        best_obj = obj_new
                        best_a = a_new.copy()
                        no_improve_rounds = 0
                    else:
                        no_improve_rounds += 1
            except Exception:
                no_improve_rounds += 1

    return best_a


def momentum_gradient(a, time_budget=60.0, tau=50.0):
    """Momentum-based gradient descent in a-space with normalization."""
    n = len(a)
    a = a.copy().astype(np.float64)
    s = np.sum(a)
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    v = np.zeros(n)
    momentum = 0.85
    lr = 0.05
    stall = 0

    for it in range(50000):
        if time.monotonic() - start > time_budget:
            break

        conv = compute_conv(a)
        max_c = np.max(conv)
        exp_c = np.exp(tau * (conv - max_c))
        w = exp_c / np.sum(exp_c)
        grad = 2.0 * np.correlate(w, a, mode='full')[n - 1:2 * n - 1]

        v = momentum * v - lr * grad
        a_new = a + v
        a_new = np.maximum(a_new, 1e-15)
        s_new = np.sum(a_new)
        if s_new > 1e-10:
            a_new = a_new / s_new
        a = a_new

        obj = compute_obj(a)
        if obj < best_obj - 1e-15:
            best_obj = obj
            best_a = a.copy()
            stall = 0
        else:
            stall += 1
            if stall > 500:
                lr *= 0.8
                momentum = min(momentum + 0.02, 0.95)
                stall = 0
                if lr < 0.001:
                    break

        lr *= 0.9999

    return best_a


def local_search(a, time_budget=60.0):
    """Intensive local search with diverse perturbations and re-optimization."""
    n = len(a)
    best_a = a.copy().astype(np.float64)
    best_obj = compute_obj(best_a)
    start = time.monotonic()

    attempt = 0
    while time.monotonic() - start < time_budget - 5:
        attempt += 1
        remaining = time_budget - (time.monotonic() - start)
        if remaining < 20:
            break

        rng = np.random.default_rng(attempt * 79193 + 13)

        # Try various perturbation scales
        scales = [0.01, 0.02, 0.05, 0.1]
        for scale in scales:
            if time.monotonic() - start > time_budget - 5:
                break

            a_pert = best_a * np.exp(rng.normal(0, scale, n))
            a_pert = np.maximum(a_pert, 1e-10)
            a_pert = a_pert / np.sum(a_pert)

            sub_budget = min(remaining * 0.2, 40.0)
            if sub_budget < 10:

                                       tau_schedule=[20, 50, 100, 200])

            obj = compute_obj(a_opt)
            if obj < best_obj:

                best_a = a_opt.copy()

    return best_a


def random_init(n, seed):
    """Generate a diverse random initialization."""
    rng = np.random.default_rng(seed)
    k = np.arange(n)
    f1 = rng.uniform(0.3, 2.0)
    f2 = rng.uniform(3.0, 10.0)
    a = 1.0 + 0.5 * np.cos(2 * np.pi * f1 * k / n + rng.uniform(0, 2 * np.pi))
    a += 0.3 * np.cos(2 * np.pi * f2 * k / n + rng.uniform(0, 2 * np.pi))
    a += rng.uniform(0, 0.2) * np.sin(2 * np.pi * rng.uniform(0.5, 5.0) * k / n)
    a = np.maximum(a, 0.05)
    return a / np.sum(a)


def resample(a, new_n):
    """Resample a sequence to a new length using linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    old_k = np.arange(n)
    new_k = np.linspace(0, n - 1, new_n)
    a_new = np.interp(new_k, old_k, a)
    a_new = np.maximum(a_new, 0)
    s = np.sum(a_new)
    if s > 1e-10:
        a_new = a_new / s
    return a_new

            if obj < best_score:

                best_seq = a_opt.copy()
                print(f"  After final local search: {best_score:.6f}")

        # LP polish
        remaining = deadline - time.monotonic()
        if remaining > 15:
            a_opt = lp_alternating(best_seq, max_iters=20, time_budget=remaining - 10)
            obj = compute_obj(a_opt)
            if obj < best_score:
                best_score = obj
                best_seq = a_opt.copy()
                print(f"  After final LP: {best_score:.6f}")

    # Final validation
    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    val = evaluate_sequence(result)
    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={val:.8f}, time={elapsed:.1f}s ===")

    return result
