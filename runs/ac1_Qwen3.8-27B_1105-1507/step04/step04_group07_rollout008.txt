"""
Minimizes J(a) = 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Spectral balancing (Gerchberg-Saxton) for flat autocorrelation
2. LP best-response with adaptive tightening and line search
3. Subgradient descent with active set analysis
4. LSE (log-sum-exp) smoothed gradient with momentum

7. Hierarchical time management with focus on best candidates

import numpy as np
from scipy.optimize import linprog
from scipy.sparse import csr_matrix
import time

_spar_cache = {}

def get_sparsity(n):
    if n not in _spar_cache:
        rows = []
        cols = []
        for k in range(2 * n - 1):
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            for j in range(js, je + 1):
                rows.append(k)
                cols.append(j)
        _spar_cache[n] = (np.array(rows, dtype=np.int32), np.array(cols, dtype=np.int32))
    return _spar_cache[n]

def fast_conv(a):
    n = len(a)
    s = 1
    while s < 2 * n - 1:
        s <<= 1
    fa = np.fft.rfft(a, s)
    return np.fft.irfft(fa * fa, s)[:2 * n - 1]

def compute_J(a):
    n = len(a)
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2.0 * n * b.max() / (s * s))

def spectral_balance(a, n_iters, deadline):
    """Iteratively flatten the power spectrum while maintaining non-negativity."""
    a = a.copy().astype(np.float64)
    a = np.maximum(a, 1e-10)
    a /= a.sum()
    n = len(a)
    
    for it in range(n_iters):
        if time.monotonic() > deadline:
            break
        F = np.fft.fft(a)
        mag = np.abs(F)
        # Target: geometric mean of magnitudes (preserves total power)
        target_mag = np.exp(np.mean(np.log(np.maximum(mag, 1e-15))))
        # Blend: partially flatten the spectrum
        blend = 0.5
        new_mag = (1 - blend) * mag + blend * target_mag
        new_mag = np.maximum(new_mag, 1e-15)
        # Preserve phase
        phase = F / np.maximum(mag, 1e-15)
        F_new = new_mag * phase
        a_new = np.fft.ifft(F_new).real
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s < 1e-10:
            break
        a_new /= s
        # Accept if it improves
        J_old = compute_J(a)
        J_new = compute_J(a_new)
        if J_new < J_old * (1 + 1e-10):
            a = a_new
        else:
            # Try softer blend
            for b2 in [0.3, 0.2, 0.1]:
                new_mag2 = (1 - b2) * mag + b2 * target_mag
                F_new2 = new_mag2 * phase
                a_try = np.fft.ifft(F_new2).real
                a_try = np.maximum(a_try, 0.0)
                if a_try.sum() > 1e-10:
                    a_try /= a_try.sum()
                    if compute_J(a_try) < J_old:
                        a = a_try
                        break
            else:
                break
    return a

def build_lp_matrix(n):
    rows, cols = get_sparsity(n)
    vals = np.empty(len(rows))
    for i in range(len(rows)):
        k = rows[i]
        j = cols[i]
        vals[i] = 0  # placeholder, filled later
    return rows, cols

def lp_best_response(a, M):
    """Solve LP: max sum(g) s.t. (a*g)_k <= M, g >= 0."""
    n = len(a)
    rows, cols = get_sparsity(n)
    vals = np.empty(len(rows))
    for i in range(len(rows)):
        k = rows[i]
        j = cols[i]
        vals[i] = a[k - j]
    A = csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                options={'presolve': True, 'time_limit': 15, 'mip_rel_gap': 0})
    if r.success and r.x is not None:
        g = np.maximum(r.x, 0.0)
        if g.sum() > 1e-10:
            return g
    return None

def line_search_mix(a, g, current_J):
    """Find best mixing parameter t in (0,1) for (1-t)*a + t*g."""
    n = len(a)
    best_t = None
    best_J = current_J
    # Try a range of t values
    for t in [0.1, 0.2, 0.3, 0.5, 0.7, 0.85, 0.95]:
        a_try = (1.0 - t) * a + t * g
        a_try = np.maximum(a_try, 0.0)
        s = a_try.sum()
        if s < 1e-10:
            continue
        a_try /= s
        J_try = compute_J(a_try)
        if J_try < best_J:
            best_J = J_try
            best_t = t
    if best_t is None:
        return None, current_J
    # Refine around best_t
    lo = max(0.01, best_t - 0.1)
    hi = min(0.99, best_t + 0.1)
    for _ in range(10):
        mid = (lo + hi) / 2
        a_try = (1.0 - mid) * a + mid * g
        a_try = np.maximum(a_try, 0.0)
        s = a_try.sum()
        if s < 1e-10:
            break
        a_try /= s
        J_try = compute_J(a_try)
        if J_try < best_J:
            best_J = J_try
            best_t = mid
            lo = mid
        else:
            hi = mid
    a_best = (1.0 - best_t) * a + best_t * g
    a_best = np.maximum(a_best, 0.0)
    a_best /= a_best.sum()
    return a_best, best_J

def run_lp_opt(a, max_iters, deadline, tracker, tighten=0.01):
    a = np.maximum(a.astype(np.float64), 1e-8)
    a /= a.sum()
    n = len(a)
    current_J = compute_J(a)
    if tracker:
        tracker.update(a)
    
    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        b = fast_conv(a)
        M = float(b.max())
        # Try tightening
        M_tight = M * (1.0 - tighten)
        g = lp_best_response(a, M_tight)
        if g is None:
            g = lp_best_response(a, M * 1.001)
        if g is None:
            break
        g = np.maximum(g, 0.0)
        gs = g.sum()
        if gs < 1e-10:
            break
        g /= gs
        
        a_new, new_J = line_search_mix(a, g, current_J)
        if a_new is not None and new_J < current_J - 1e-14:
            a = a_new
            current_J = new_J
        else:
            # Try without tightening
            g2 = lp_best_response(a, M)
            if g2 is not None:

                a_new, new_J = line_search_mix(a, g2, current_J)

            else:

def compute_lse_gradient(a, p):
    """Compute gradient of LSE(p · (a*a)) w.r.t. a using FFT."""
    n = len(a)
    # Need cross-correlation: (p ⊛ a)[j] = sum_m p[m+j] * a[m]
    # p has length 2n-1, a has length n
    # Use FFT with zero-padding
    N = 1
    while N < 3 * n:
        N <<= 1
    P = np.fft.rfft(p, N)
    A = np.fft.rfft(a, N)
    corr = np.fft.irfft(P * A.conj(), N)[:n]
    return 2.0 * corr

def lse_gradient_descent(a, max_iters, deadline, tracker):
    a = np.maximum(a.astype(np.float64), 1e-6)
    a /= a.sum()
    n = len(a)
    current_J = compute_J(a)
    if tracker:
        tracker.update(a)
    
    tau = 30.0
    lr = 1.0 / n
    momentum = 0.9
    v = np.zeros(n)
    no_improve = 0
    
    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        b = fast_conv(a)
        b_max = b.max()
        # Shift and scale for numerical stability
        exp_b = np.exp(tau * (b - b_max))
        p = exp_b / exp_b.sum()
        
        grad = compute_lse_gradient(a, p)
        
        v = momentum * v + (1.0 - momentum) * grad
        a_new = a - lr * v
        a_new = np.maximum(a_new, 1e-6)
        a_new /= a_new.sum()
        J_new = compute_J(a_new)
        
        if J_new < current_J - 1e-15:
            a = a_new
            current_J = J_new
            no_improve = 0
            tau = min(500.0, tau * 1.02)
            lr = min(lr * 1.01, 5.0 / n)
        else:
            no_improve += 1
            lr *= 0.7
            v *= 0.5
            if no_improve > 30 or lr < 1e-18:
                break
        if tracker and it % 50 == 0:
            tracker.update(a)
    
    if tracker:
        tracker.update(a)
    return a

def subgradient_descent(a, max_iters, deadline, tracker):
    a = np.maximum(a.astype(np.float64), 1e-8)
    a /= a.sum()
    n = len(a)
    current_J = compute_J(a)
    if tracker:
        tracker.update(a)
    lr = 1.0 / n
    rnn = 0.0
    rng = np.random.default_rng(123)
    
    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        b = fast_conv(a)
        M = float(b.max())
        # Active set: indices where b is within epsilon of max
        eps = 1e-6 * M + 1e-12
        active = np.where(b >= M - eps)[0]
        
        # Subgradient: average of gradients over active set
        # grad_k(a)_j = 2*a[k-j] for valid k-j
        # Use FFT for efficiency
        q = np.zeros(2 * n - 1)
        q[active] = 1.0 / len(active)
        # Cross-correlation of q and a
        N = 1
        while N < 3 * n:
            N <<= 1
        Q = np.fft.rfft(q, N)
        A = np.fft.rfft(a, N)
        corr = np.fft.irfft(Q * A.conj(), N)[:n]
        grad = 2.0 * corr
        
        # Polyak step size
        rnn += lr * lr
        alpha = lr / np.sqrt(rnn)
        
        a_new = a - alpha * grad
        a_new = np.maximum(a_new, 1e-8)
        a_new /= a_new.sum()
        J_new = compute_J(a_new)
        
        if J_new < current_J - 1e-15:
            a = a_new
            current_J = J_new
            lr = min(lr * 1.05, 10.0 / n)
        else:
            lr *= 0.8
        if tracker and it % 20 == 0:
            tracker.update(a)
    
    if tracker:
        tracker.update(a)
    return a

def resample_seq(src, dest_n):
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)

def generate_inits(n, rng, parent):
    inits = []
    # From parent
    if parent is not None and parent.size > 0:
        inits.append(resample_seq(parent, n))
        inits.append(resample_seq(parent[::-1], n))
        # Perturbed parent
        p = resample_seq(parent, n)
        inits.append(np.maximum(p + rng.normal(0, 0.05, n), 0.01))
    
    t = np.linspace(0, 1, n, endpoint=False)
    
    # Uniform
    inits.append(np.ones(n))
    
    # Random
    inits.append(rng.uniform(0.1, 1.0, n))
    inits.append(rng.uniform(0.0, 1.0, n))
    
    # Flat spectrum (random phases)
    for _ in range(3):
        half = n // 2 + 1
        angles = rng.uniform(0, 2 * np.pi, half)
        A = np.zeros(n, dtype=complex)
        A[:half] = np.exp(1j * angles)
        for k in range(1, n - n // 2):
            A[k] = np.conj(A[n - k])
        a = np.fft.ifft(A).real
        inits.append(np.maximum(a, 0.0))
    
    # Cosine modulated
    for freq in [1, 2, 3, 5, 7, 11]:
        amp = rng.uniform(0.1, 0.5)
        inits.append(1.0 + amp * np.cos(2 * np.pi * freq * t + rng.uniform(0, 6.28)))
    
    # Multi-frequency
    inits.append(1.0 + 0.3 * np.sin(2 * np.pi * 1.3 * t) + 0.2 * np.sin(2 * np.pi * 3.7 * t)
                + 0.1 * np.sin(2 * np.pi * 7.1 * t + 0.5))
    
    # Gaussian bumps
    for _ in range(2):
        centers = rng.uniform(0.2, 0.8, 2)
        sigmas = rng.uniform(0.05, 0.2, 2)
        sig = np.zeros(n)
        for c, s in zip(centers, sigmas):
            sig += np.exp(-0.5 * ((t - c) / s) ** 2)
        inits.append(sig + 0.1)
    
    # Power law
    inits.append((t + 0.05) ** rng.uniform(-2, 2))
    inits.append((1 - t + 0.05) ** rng.uniform(-2, 2))
    
    # Triangle
    inits.append(1.0 - np.abs(2 * t - 1.0))
    
    # Arc-sine
    inits.append(1.0 / np.sqrt(np.maximum(t * (1 - t), 0.01)))
    
    # Sparse smoothed
    sparse = np.zeros(n)
    n_spikes = max(5, n // 10)
    positions = rng.choice(n, size=n_spikes, replace=False)
    sparse[positions] = 1.0
    kernel_size = max(3, n // 20)
    kernel = np.ones(kernel_size) / kernel_size
    inits.append(np.convolve(sparse, kernel, mode='same') + 0.05)
    
    # All inits: normalize
    for i, x in enumerate(inits):
        x = np.maximum(np.asarray(x, dtype=np.float64), 0.001)
        s = x.sum()
        if s > 1e-10:
            inits[i] = x / s
        else:
            inits[i] = np.ones(n) / n
    return inits
