    n = len(a)

        k_end = min(j + n, 2 * n - 1)
        A[k_start:k_end, j] = a[:k_end - k_start]
    return A

        if max_l >= 0:
            grad_lse[j] = 2.0 * np.dot(x[:max_l + 1], w[j:j + max_l + 1])

    # Full objective: f = 2n * LSE / s^2
    obj = 2.0 * n * log_sum_exp / (s * s)

    # Gradient: df/dx[j] = 2n * (s * grad_lse[j] - 2*LSE) / s^3
    grad = 2.0 * n * (s * grad_lse - 2.0 * log_sum_exp) / (s ** 3)

    return obj, grad


def optimize_lBFGS(a, time_budget=60.0):
    """
    L-BFGS optimization with temperature annealing on the smoothed objective.
    This is the main novel contribution - uses smooth approximation of the max
    for fast, reliable convergence.
    """
    n = len(a)
    a = a.copy().astype(np.float64)
    s = np.sum(a)
    if s < 1e-10:
        a = np.ones(n) / n
        s = 1.0
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    # Temperature schedule: start smooth, end sharp
    # T * max(conv) should be O(1) to O(20) for good approximation
    # max(conv) ~ 1/n ~ 0.003 for n=330, so T ~ 300-3000 for sharp
    temperatures = [20.0, 50.0, 100.0, 200.0, 400.0, 800.0, 1500.0]

    for T in temperatures:
        if time.monotonic() - start > time_budget - 3:
            break

        elapsed_T = time.monotonic() - start
        time_per_T = time_budget / len(temperatures)

        def fun_and_jac(x):
            x = np.maximum(x, 1e-15)
            obj, grad = smoothed_obj_grad(x, T)
            if np.isinf(obj):
                return 1e10, np.zeros(len(x))
            return obj, grad

        try:
            res = minimize(fun_and_jac, a, method='L-BFGS-B',
                           jac=True, bounds=[(1e-12, None)] * n,
                           options={'maxiter': 300, 'ftol': 1e-12, 'gtol': 1e-10,
                                    'maxfun': 2000})
            if res.x is not None:
                a_new = np.maximum(res.x, 0)
                s_new = np.sum(a_new)
                if s_new > 1e-10:
                    a_new = a_new / s_new
                    obj_new = compute_obj(a_new)
                    if obj_new < best_obj:
                        best_obj = obj_new
                        best_a = a_new.copy()
                        a = a_new.copy()
        except Exception:
            pass

        if time.monotonic() - start > time_budget - 3:
            break

    return best_a, best_obj


def subgradient_step(a, tau, step):
    """One multiplicative subgradient step on the exact objective."""
    n = len(a)
    conv = compute_conv(a)
    max_conv = np.max(conv)
    shifted = tau * (conv - max_conv)
    shifted = np.clip(shifted, -500, 500)
    weights = np.exp(shifted)
    w_sum = np.sum(weights)
    if w_sum < 1e-300:
        return a
    weights /= w_sum

    # Gradient: 2 * sum_l a[l] * w[j+l]
    grad = np.zeros(n)
    for j in range(n):
        max_l = min(n - 1, 2 * n - 2 - j)
        if max_l >= 0:
            grad[j] = 2.0 * np.dot(a[:max_l + 1], weights[j:j + max_l + 1])

    a_new = a * np.exp(-step * grad)
    a_new = np.maximum(a_new, 1e-15)
    s = np.sum(a_new)
    if s > 1e-10:
        a_new = a_new / s
    return a_new


def subgradient_optimize(a, max_iters=3000, time_budget=20.0, tau=30.0, step=0.05):
    """Subgradient optimization with adaptive parameters."""
    n = len(a)
    a = a.copy().astype(np.float64)
    s = np.sum(a)
    if s < 1e-10:
        return a
    a = a / s
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    stall = 0
    patience = 100

    for it in range(max_iters):
        if time.monotonic() - start > time_budget:
            break
        a = subgradient_step(a, tau, step)
        obj = compute_obj(a)
        if obj < best_obj - 1e-13:
            best_obj = obj
            best_a = a.copy()
            stall = 0
        else:
            stall += 1
            if stall > patience:
                step = max(step * 0.7, 0.001)
                patience = max(patience // 2, 30)
                if step < 0.002:
                    break
                rng = np.random.default_rng(it * 31 + 7)
                a = a * np.exp(rng.normal(0, 0.02, n))
                a = np.maximum(a, 1e-10)
                s = np.sum(a)
                if s > 1e-10:
                    a = a / s
                stall = 0
        if it % 50 == 0:
            step *= 0.99
            tau *= 1.01

    return best_a


def alternating_lp(a, max_iters=50, time_budget=30.0):
    """Alternating LP for refinement."""
    n = len(a)
    a = a.copy().astype(np.float64)
    s = np.sum(a)
    if s < 1e-10:
        return a
    a = a / s
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    stall = 0

    for it in range(max_iters):
        if time.monotonic() - start > time_budget - 3:
            break

        conv = compute_conv(a)
        M = float(np.max(conv))

        g, sum_g = solve_lp_maxsum(a, M, time_limit=4)
        if g is None or sum_g < 1e-10:
            break

        g_norm = g / sum_g

        best_t = 0.0
        best_obj_ls = best_obj
        for t in [0.1, 0.3, 0.5, 0.7, 1.0]:
            a_new = (1.0 - t) * a + t * g_norm
            a_new = np.maximum(a_new, 0)
            s_new = np.sum(a_new)
            if s_new < 1e-10:
                continue
            a_new = a_new / s_new
            obj_new = compute_obj(a_new)
            if obj_new < best_obj_ls:
                best_obj_ls = obj_new
                best_t = t

        if best_t > 0 and best_obj_ls < best_obj - 1e-13:
            a_new = (1.0 - best_t) * a + best_t * g_norm
            a_new = np.maximum(a_new, 0)
            s_new = np.sum(a_new)
            if s_new > 1e-10:
                a_new = a_new / s_new
            best_obj = best_obj_ls
            best_a = a_new.copy()
            a = a_new
            stall = 0
        else:
            stall += 1
            if stall > 20:
                break
            a = 0.9 * a + 0.1 * g_norm
            a = np.maximum(a, 0)
            s = np.sum(a)
            if s > 1e-10:
                a = a / s

    return best_a


def fine_tune_exact(a, time_budget=20.0):
    """
    Fine-tuning using very small subgradient steps with high tau,
    combined with occasional L-BFGS micro-refinement.
    """
    n = len(a)
    a = a.copy().astype(np.float64)
    s = np.sum(a)
    if s < 1e-10:
        return a
    a = a / s
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    step = 0.005
    tau = 100.0
    stall = 0

    max_iters = int(time_budget * 80)
    for it in range(max_iters):
        if time.monotonic() - start > time_budget:
            break

        a = subgradient_step(a, tau, step)
        obj = compute_obj(a)
        if obj < best_obj - 1e-14:
            best_obj = obj
            best_a = a.copy()
            stall = 0
        else:
            stall += 1
            if stall > 80:
                step *= 0.8
                tau *= 1.3
                if step < 0.0005:
                    # Try a micro L-BFGS
                    if time.monotonic() - start < time_budget - 5:
                        a_new, obj_new = optimize_lBFGS(best_a, time_budget=5.0)
                        if obj_new < best_obj:
                            best_obj = obj_new
                            best_a = a_new.copy()
                            a = a_new.copy()
                            step = 0.005
                            stall = 0
                            continue
                    break
                stall = 0
                # Small perturbation to escape
                rng = np.random.default_rng(it)
                a = best_a * np.exp(rng.normal(0, 0.01, n))
                a = np.maximum(a, 1e-10)
                s = np.sum(a)
                a = a / s

    return best_a


def coordinate_descent(a, time_budget=15.0):
    """
    Coordinate descent: optimize one coordinate at a time.
    For fixed a with a[j] free, the objective is a quadratic in a[j] (piecewise),
    so we can find the optimal a[j] by checking critical points.
    """
    n = len(a)
    a = a.copy().astype(np.float64)
    s = np.sum(a)
    a = a / s
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    for sweep in range(200):
        if time.monotonic() - start > time_budget:
            break
        improved = False
        for j in range(n):
            if time.monotonic() - start > time_budget:
                break
            # Optimize a[j] with others fixed
            # conv[k] = sum_i a[i]*a[k-i], terms involving a[j]:
            #   appears in conv[j+l] for l=0..n-1 (as a[j]*a[l])
            #   and in conv[j+l] for l=0..n-1 (as a[l]*a[j]) -- same term if l=j
            # Actually conv[k] = a[j]*a[k-j] + (terms not involving a[j]) for k-j in [0,n-1]
            # and also a[k-j]*a[j] for k-j in [0,n-1]... 
            # Let's think: conv[k] = sum_i a[i]*a[k-i]
            # Terms with a[j]: i=j gives a[j]*a[k-j] (if 0<=k-j<n)
            #                  k-i=j gives a[k-j]*a[j] (if 0<=k-j<n, i.e., same condition)
            # If j != k-j: two terms, each a[j]*a[k-j]... no wait.
            # If i=j: term is a[j]*a[k-j]
            # If k-i=j, i.e., i=k-j: term is a[k-j]*a[j]
            # If j = k-j (k=2j): same term a[j]^2, derivative is 2*a[j]
            # Otherwise: a[j]*a[k-j] + a[k-j]*a[j] = 2*a[j]*a[k-j]
            # So conv[k] = 2*a[j]*a[k-j] + rest[k] for k != 2j
            #            = a[j]^2 + rest[k] for k = 2j

            # This makes the exact optimization of a single coordinate complex
            # because max is involved. Let's just do a 1D golden section search.
            
            old_val = a[j]
            s_others = s - old_val
            
            # Search over a[j] in [0, 3*old_val] or [0, 1] if old_val is 0
            lo = 0.0
            hi = max(3.0 * old_val, 0.5 / n) if old_val > 1e-10 else 1.0 / n
            
            # Golden section
            gr = (np.sqrt(5) - 1) / 2
            c1 = hi - gr * (hi - lo)
            c2 = lo + gr * (hi - lo)
            
            a[j] = c1
            s_curr = s_others + c1
            obj1 = compute_obj(a) if s_curr > 1e-10 else np.inf
            a[j] = c2
            s_curr = s_others + c2
            obj2 = compute_obj(a) if s_curr > 1e-10 else np.inf
            
            for _ in range(15):
                if obj1 < obj2:
                    hi = c2
                    c2 = c1
                    obj2 = obj1
                    c1 = hi - gr * (hi - lo)
                    a[j] = c1
                    obj1 = compute_obj(a) if (s_others + c1) > 1e-10 else np.inf
                else:
                    lo = c1
                    c1 = c2
                    obj1 = obj2
                    c2 = lo + gr * (hi - lo)
                    a[j] = c2
                    obj2 = compute_obj(a) if (s_others + c2) > 1e-10 else np.inf
            
            # Take the better
            if obj1 < obj2:
                a[j] = c1
            else:
                a[j] = c2
            
            s_new = np.sum(a)
            if s_new > 1e-10:
                a = a / s_new
            
            obj_new = compute_obj(a)
            if obj_new < best_obj - 1e-14:
                best_obj = obj_new
                best_a = a.copy()
                improved = True
        
        if not improved:
            break

    best_a = np.maximum(best_a, 0)
    s = np.sum(best_a)
    if s > 1e-10:
        best_a = best_a / s
    return best_a


def random_init(n, seed):
    """Generate diverse random initializations."""
    rng = np.random.default_rng(seed)
    k = np.arange(n)
    choice = rng.integers(0, 8)

    if choice == 0:
        f = rng.uniform(0.5, 4.0)
        phase = rng.uniform(0, 2 * np.pi)
        a = 1.0 + 0.8 * np.cos(2 * np.pi * f * k / n + phase)
    elif choice == 1:
        f1 = rng.uniform(0.3, 2.0)
        f2 = rng.uniform(2.0, 6.0)
        a = 1.0 + 0.4 * np.cos(2 * np.pi * f1 * k / n) + 0.3 * np.cos(2 * np.pi * f2 * k / n)
    elif choice == 2:
        center = n * rng.uniform(0.3, 0.7)
        alpha = rng.uniform(0.5, 3.0)
        a = 1.0 + 0.5 * np.exp(-alpha * np.abs(k - center) / (n * 0.3))
    elif choice == 3:
        a = rng.random(n)
        win = max(3, n // 20)
        kernel = np.ones(win) / win
        a = np.convolve(a, kernel, mode='same')
    elif choice == 4:
        a = np.ones(n)
        for _ in range(rng.integers(1, 4)):
            f = rng.uniform(0.5, 5.0)
            amp = rng.uniform(0.1, 0.4)
            a += amp * np.cos(2 * np.pi * f * k / n + rng.uniform(0, 2 * np.pi))
    elif choice == 5:
        a = np.ones(n)
        for scale in range(3):
            f = rng.uniform(0.3 * (2 ** scale), 1.5 * (2 ** scale))
            amp = 0.3 / (scale + 1)
            a += amp * np.cos(2 * np.pi * f * k / n + rng.uniform(0, 2 * np.pi))
    elif choice == 6:
        # Roughly binary with smoothing
        a = (rng.random(n) > 0.4).astype(float)
        win = max(3, n // 30)
        kernel = np.ones(win) / win
        a = np.convolve(a, kernel, mode='same')
    else:
        # Random smooth
        a = rng.random(max(5, n // 8))
        a = np.interp(k, np.arange(len(a)), a)

    a = np.maximum(a, 0)
    s = np.sum(a)
    if s < 1e-10:
        a = np.ones(n)
        s = n
    return a / s


def resample_sequence(a, new_n):
    """Resample a sequence to a new length using spline-like interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    old_k = np.arange(n, dtype=np.float64)
    new_k = np.linspace(0, n - 1, new_n)
    a_new = np.interp(new_k, old_k, a)
    a_new = np.maximum(a_new, 0)
    s = np.sum(a_new)
    if s > 1e-10:
        a_new = a_new / s
    return a_new


def try_parent():
    """Try to load parent sequence."""
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64)
        parent = np.clip(parent, 0, 1000.0)
        if parent.size > 0 and np.isfinite(parent).all():
            obj = compute_obj(parent)
            print(f"Parent (n={parent.size}): obj={obj:.8f}")
            return parent.copy(), obj
    except Exception:
        pass
    return None, np.inf


def optimize_single(a, time_budget=60.0, verbose=False):
    """Run full optimization pipeline on a single sequence."""
    n = len(a)
    a = a.copy().astype(np.float64)
    s = np.sum(a)
    if s < 1e-10:
        a = np.ones(n)
        s = n
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    if verbose:
        print(f"  Start (n={n}): obj={best_obj:.8f}")

    # Phase 1: L-BFGS with temperature annealing (fast and smooth)
    remaining = time_budget - (time.monotonic() - start)
    if remaining > 10:
        lbfgs_time = min(remaining * 0.3, 40.0)
        a_new, obj_new = optimize_lBFGS(best_a, time_budget=lbfgs_time)
        if obj_new < best_obj:
            best_obj = obj_new
            best_a = a_new.copy()
            if verbose:
                print(f"  After L-BFGS: {best_obj:.8f}")

    # Phase 2: Subgradient refinement
    remaining = time_budget - (time.monotonic() - start)
    if remaining > 8:
        sg_time = min(remaining * 0.2, 15.0)
        a_new = subgradient_optimize(best_a, time_budget=sg_time, tau=50.0, step=0.03)
        obj_new = compute_obj(a_new)
        if obj_new < best_obj:
            best_obj = obj_new
            best_a = a_new.copy()
            if verbose:
                print(f"  After SG: {best_obj:.8f}")

    # Phase 3: Alternating LP
    remaining = time_budget - (time.monotonic() - start)
    if remaining > 10:
        lp_time = min(remaining * 0.25, 20.0)
        a_new = alternating_lp(best_a, max_iters=30, time_budget=lp_time)
        obj_new = compute_obj(a_new)
        if obj_new < best_obj:
            best_obj = obj_new
            best_a = a_new.copy()
            if verbose:
                print(f"  After LP: {best_obj:.8f}")

    # Phase 4: Coordinate descent
    remaining = time_budget - (time.monotonic() - start)
    if remaining > 8:
        cd_time = min(remaining * 0.2, 12.0)
        a_new = coordinate_descent(best_a, time_budget=cd_time)
        obj_new = compute_obj(a_new)
        if obj_new < best_obj:
            best_obj = obj_new
            best_a = a_new.copy()
            if verbose:
                print(f"  After CD: {best_obj:.8f}")

    # Phase 5: Fine tune
    remaining = time_budget - (time.monotonic() - start)
    if remaining > 5:
        ft_time = min(remaining * 0.3, 10.0)
        a_new = fine_tune_exact(best_a, time_budget=ft_time)
        obj_new = compute_obj(a_new)
        if obj_new < best_obj:
            best_obj = obj_new
            best_a = a_new.copy()
            if verbose:
                print(f"  After FT: {best_obj:.8f}")

    return best_a, best_obj


def intensive_local_search(a, time_budget=100.0, verbose=False):
    """
    Intensive local search: perturb + re-optimize in many directions.
    Uses adaptive perturbation scales based on improvement history.
    """
    n = len(a)
    a = a.copy().astype(np.float64)
    s = np.sum(a)
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    if verbose:
        print(f"  Local search start: obj={best_obj:.8f}")

    attempt = 0
    base_scale = 0.03
    while time.monotonic() - start < time_budget - 5:
        attempt += 1
        remaining = time.monotonic() and (time_budget - (time.monotonic() - start))
        if remaining < 8:
            break

        rng = np.random.default_rng(attempt * 7919 + 42)

        # Adaptive scale: start large, decrease with attempts
        scale = base_scale * max(0.3, 1.0 - attempt / 50.0)
        scale *= (1.0 + 0.5 * np.sin(attempt * 0.7))  # oscillate

        # Generate perturbation
        a_pert = best_a * np.exp(rng.normal(0, scale, n))
        a_pert = np.maximum(a_pert, 1e-10)
        s_pert = np.sum(a_pert)
        if s_pert < 1e-10:
            continue
        a_pert = a_pert / s_pert

        # Quick optimization
        sub_time = min(12.0, remaining * 0.4)
        a_opt, obj = optimize_single(a_pert, time_budget=sub_time, verbose=False)

        if obj < best_obj:
            best_obj = obj
            best_a = a_opt.copy()
            if verbose and attempt % 5 == 0:
                print(f"    Attempt {attempt}: {best_obj:.8f}")

    return best_a, best_obj


def propose_candidate(seed=42, budget_s=1500):
    """
    Main search function. Uses L-BFGS with log-sum-exp smoothing as the primary
    optimizer, combined with subgradient, LP, coordinate descent, and local search.
    
    Strategy:
    1. Load and optimize parent sequence (n~330)
    2. Intensive local search on best found
    3. Try nearby n values with multiple initializations
    4. Final polish
    """
    start_time = time.monotonic()
    deadline = start_time + float(budget_s) - 20.0

    best_seq = None
    best_score = np.inf

    # Load parent
    parent, parent_obj = try_parent()
    if parent is not None and parent_obj < best_score:
        best_score = parent_obj
        best_seq = parent.copy()
        n_parent = len(parent)
    else:
        n_parent = 300
        best_seq = random_init(n_parent, seed)
        best_score = compute_obj(best_seq)

    print(f"Initial best: n={len(best_seq)}, score={best_score:.8f}")
    print(f"Time budget: {budget_s}s")

    # ============================================================
    # Phase 1: Intensive optimization of parent (L-BFGS driven)
    # ============================================================
    print("\n=== Phase 1: L-BFGS optimization of parent ===")
    remaining = deadline - time.monotonic()
    p1_budget = min(remaining * 0.25, 200.0)
    p1_budget = max(p1_budget, 30.0)

    a_opt, obj = optimize_single(best_seq, time_budget=p1_budget, verbose=True)
    if obj < best_score:
        best_score = obj
        best_seq = a_opt.copy()
        print(f"  After optimize_single: {best_score:.8f}")

    # Additional L-BFGS pass with different temperature schedule
    remaining = deadline - time.monotonic()
    if remaining > 30:
        lb_time = min(remaining * 0.1, 60.0)
        a_opt2, obj2 = optimize_lBFGS(best_seq, time_budget=lb_time)
        if obj2 < best_score:
            best_score = obj2
            best_seq = a_opt2.copy()
            print(f"  After L-BFGS2: {best_score:.8f}")

    # Subgradient polish
    remaining = deadline - time.monotonic()
    if remaining > 20:
        sg_time = min(remaining * 0.08, 30.0)
        a_opt3 = subgradient_optimize(best_seq, time_budget=sg_time, tau=80.0, step=0.02)
        obj3 = compute_obj(a_opt3)
        if obj3 < best_score:
            best_score = obj3
            best_seq = a_opt3.copy()
            print(f"  After SG polish: {best_score:.8f}")

    # Coordinate descent
    remaining = deadline - time.monotonic()
    if remaining > 15:
        cd_time = min(remaining * 0.05, 15.0)
        a_opt4 = coordinate_descent(best_seq, time_budget=cd_time)
        obj4 = compute_obj(a_opt4)
        if obj4 < best_score:
            best_score = obj4
            best_seq = a_opt4.copy()
            print(f"  After CD: {best_score:.8f}")

    # ============================================================
    # Phase 2: Intensive local search
    # ============================================================
    print(f"\n=== Phase 2: Intensive local search ===")
    remaining = deadline - time.monotonic()
    p2_budget = min(remaining * 0.3, remaining - 30)
    p2_budget = max(p2_budget, 30.0)

    a_opt, obj = intensive_local_search(best_seq, time_budget=p2_budget, verbose=True)
    if obj < best_score:
        best_score = obj
        best_seq = a_opt.copy()
        print(f"  After local search: {best_score:.8f}")

    # ============================================================
    # Phase 3: Try nearby n values
    # ============================================================
    print(f"\n=== Phase 3: Nearby n values ===")
    remaining = deadline - time.monotonic()
    if remaining > 40:
        n_values = []
        for delta in [-20, -10, 0, 10, 20, -40, 40, -60, 60]:
            n_val = n_parent + delta
            if 150 <= n_val <= 800:
                n_values.append(n_val)
        n_values = sorted(set(n_values))
        # Remove the parent n (already done)
        n_values = [x for x in n_values if x != n_parent] or n_values

        per_n = remaining / max(len(n_values), 1) * 0.8
        for n_val in n_values:
            if time.monotonic() > deadline - 30:
                break
            remaining = deadline - time.monotonic()
            if remaining < 25:
                break
            alloc = min(per_n, remaining * 0.3, 80.0)
            alloc = max(alloc, 15.0)

            # Use parent (resampled) as starting point
            a_start = resample_sequence(best_seq, n_val)
            print(f"  n={n_val} (alloc={alloc:.0f}s)")
            a_opt, obj = optimize_single(a_start, time_budget=alloc, verbose=False)
            if obj < best_score:
                best_score = obj
                best_seq = a_opt.copy()
                print(f"  ** UPDATE: n={n_val}, obj={best_score:.8f} **")

    # ============================================================
    # Phase 4: More local search on current best
    # ============================================================
    remaining = deadline - time.monotonic()
    if remaining > 30:
        print(f"\n=== Phase 4: More local search ===")
        p4_budget = min(remaining * 0.3, remaining - 20)
        p4_budget = max(p4_budget, 20.0)
        a_opt, obj = intensive_local_search(best_seq, time_budget=p4_budget, verbose=True)
        if obj < best_score:
            best_score = obj
            best_seq = a_opt.copy()
            print(f"  After LS2: {best_score:.8f}")

    # ============================================================
    # Phase 5: Final polish
    # ============================================================
    remaining = deadline - time.monotonic()
    if remaining > 15:
        print(f"\n=== Phase 5: Final polish ===")
        p5_budget = min(remaining * 0.4, remaining - 10)
        p5_budget = max(p5_budget, 10.0)

        # L-BFGS polish
        a_opt, obj = optimize_lBFGS(best_seq, time_budget=p5_budget * 0.5)
        if obj < best_score:
            best_score = obj
            best_seq = a_opt.copy()

        # Fine tune
        remaining = deadline - time.monotonic()
        if remaining > 5:
            ft_time = min(remaining * 0.5, 10.0)
            a_opt = fine_tune_exact(best_seq, time_budget=ft_time)
            obj = compute_obj(a_opt)
            if obj < best_score:
                best_score = obj
                best_seq = a_opt.copy()

        # Coordinate descent final
        remaining = deadline - time.monotonic()
        if remaining > 8:
            cd_time = min(remaining * 0.4, 10.0)
            a_opt = coordinate_descent(best_seq, time_budget=cd_time)
            obj = compute_obj(a_opt)
            if obj < best_score:
                best_score = obj
                best_seq = a_opt.copy()

        print(f"  Final: {best_score:.8f}")

    # Final validation
    if best_seq is None:
        best_seq = np.ones(50)
        best_score = compute_obj(best_seq)

    # Ensure all values are valid
    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]

    # Validate with evaluate_sequence
    val = evaluate_sequence(result)
    if np.isfinite(val) and val < best_score:
        best_score = val

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(best_seq)}, score={best_score:.8f}, time={elapsed:.1f}s ===")

    return result
