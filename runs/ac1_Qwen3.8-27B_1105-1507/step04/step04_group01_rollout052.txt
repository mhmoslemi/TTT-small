```python
import numpy as np
from scipy.optimize import minimize, linprog
from scipy.fft import rfft, irfft, fft, ifft
import time

def compute_conv(a):
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    conv = irfft(fa * fa, size)
    return conv[:2 * n - 1]

def compute_obj(a):
    n = len(a)
    conv = compute_conv(a)
    max_b = float(np.max(conv))
    sum_a = float(np.sum(a))
    if sum_a < 0.01:
        return np.inf
    return 2.0 * n * max_b / (sum_a * sum_a)

def fft_convolve(a, b):
    n, m = len(a), len(b)
    size = 1
    while size < n + m - 1:
        size <<= 1
    fa = rfft(a, size)
    fb = rfft(b, size)
    return irfft(fa * fb, size)[:n + m - 1]

def softmax(z):
    z = z - np.max(z)
    e = np.exp(z)
    return e / np.sum(e)

def soft_max_obj_and_grad_z(z, tau):
    n = len(z)
    a = softmax(z)
    conv = compute_conv(a)
    cmax = float(np.max(conv))
    shifted = np.clip((conv - cmax) / tau, -500.0, 500.0)
    exp_conv = np.exp(shifted)
    Z = float(np.sum(exp_conv))
    soft_max_val = cmax + tau * np.log(Z)
    w = exp_conv / Z
    w_rev = w[::-1]
    conv_fw = fft_convolve(a, w_rev)
    f1 = conv_fw[n - 1:2 * n - 1][::-1]
    grad_a = 2.0 * f1
    mean_grad = float(np.dot(a, grad_a))
    grad_z = a * (grad_a - mean_grad)
    return soft_max_val, grad_z
```

"""
Hybrid DFT-flattening + L-BFGS-B log-sum-exp + alternating LP optimization
for minimizing 2*n*max(conv(a,a))/(sum(a))^2 over non-negative sequences.
Key improvements: DFT-based initialization, active-set LP reduction,
adaptive time allocation favoring polishing, and multi-n exploration.

import numpy as np
from scipy.optimize import minimize, linprog
from scipy.fft import rfft, irfft, fft, ifft
import time


def compute_conv(a):
    """Compute linear convolution a*a using FFT. Returns array of length 2n-1."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    conv = irfft(fa * fa, size)
    return conv[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    conv = compute_conv(a)
    max_b = float(np.max(conv))
    sum_a = float(np.sum(a))
    if sum_a < 0.01:
        return np.inf
    return 2.0 * n * max_b / (sum_a * sum_a)


def fft_convolve(a, b):
    """Linear convolution of a and b using FFT."""
    n, m = len(a), len(b)
    size = 1
    while size < n + m - 1:
        size <<= 1
    fa = rfft(a, size)
    fb = rfft(b, size)
    return irfft(fa * fb, size)[:n + m - 1]


def softmax_fn(z):
    """Numerically stable softmax."""
    z = z - np.max(z)
    e = np.exp(z)
    return e / np.sum(e)


def soft_max_obj_and_grad_z(z, tau):
    """
    Compute soft-max of conv(softmax(z), softmax(z)) and its gradient w.r.t. z.
    """
    n = len(z)
    a = softmax_fn(z)
    conv = compute_conv(a)
    cmax = float(np.max(conv))
    shifted = np.clip((conv - cmax) / tau, -500.0, 500.0)
    exp_conv = np.exp(shifted)
    Z = float(np.sum(exp_conv))
    soft_max_val = cmax + tau * np.log(Z)
    w = exp_conv / Z
    w_rev = w[::-1]
    conv_fw = fft_convolve(a, w_rev)
    f1 = conv_fw[n - 1:2 * n - 1][::-1]
    grad_a = 2.0 * f1
    mean_grad = float(np.dot(a, grad_a))
    grad_z = a * (grad_a - mean_grad)
    return soft_max_val, grad_z

def dft_flatten_init(n, seed, n_iters=30):
    """
    Generate a non-negative sequence with approximately flat DFT magnitude
    (for non-zero frequencies) using error reduction / Gerchberg-Saxton style
    iterative projection.
    """
    rng = np.random.default_rng(seed)
    a = rng.random(n) + 0.1
    a = a / np.sum(a)

    for it in range(n_iters):
        # Compute DFT
        A = fft(a)
        # Target: flat magnitude for non-zero frequencies
        target_mag = np.ones(n)
        target_mag[0] = np.abs(A[0])  # keep DC component
        # Blend: partially flatten
        blend = 0.3 + 0.5 * (it / n_iters)  # increase flattening over iterations
        new_mag = (1 - blend) * np.abs(A) + blend * target_mag
        new_mag[0] = np.abs(A[0])
        # Preserve phase
        phase = np.angle(A)
        A_new = new_mag * np.exp(1j * phase)
        # IDFT
        a_new = np.real(ifft(A_new))
        # Project onto non-negative
        a_new = np.maximum(a_new, 0)
        s = np.sum(a_new)
        if s < 1e-10:
            a = np.ones(n) / n
            break
        a = a_new / s

    return a

def resample_sequence(a, new_n):
    """Resample sequence to new length via interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return a_new


def build_toeplitz(a):
    """Build Toeplitz matrix where (A@g)[k] = (conv(a,g))[k]."""
    n = len(a)
    A = np.zeros((2 * n - 1, n), dtype=np.float64)
    for j in range(n):
        k_start = j
        k_end = min(j + n, 2 * n - 1)
        A[k_start:k_end, j] = a[:k_end - k_start]
    return A


def solve_lp_g(a, M, time_limit=15):
    """Solve LP: max sum(g) s.t. (a*g)[k] <= M for all k, g >= 0."""
    n = len(a)
    A = build_toeplitz(a)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': time_limit})
        if res.status == 0:
            g = np.maximum(res.x, 0)
            s = float(np.sum(g))
            if s > 1e-10:
                return g / s, s
    except Exception:
        pass
    return None, None


def solve_lp_g_active(a, M, active_indices, time_limit=10):
    """Solve LP with only active constraints for speed."""
    n = len(a)
    A_full = build_toeplitz(a)
    A = A_full[active_indices, :]
    c = -np.ones(n)
    b_ub = np.full(len(active_indices), M)
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': time_limit})
        if res.status == 0:
            g = np.maximum(res.x, 0)
            s = float(np.sum(g))

    except Exception:
        pass
    return None, None


def alternating_lp(a, max_iters=80, time_budget=60.0, verbose=False):

            break

        conv = compute_conv(a)
        M = float(np.max(conv))

        # Find active constraints (those close to max)
        threshold = M * 0.95
        active = np.where(conv >= threshold)[0]
        # Always include the peak
        peak_idx = int(np.argmax(conv))
        if peak_idx not in active:
            active = np.append(active, peak_idx)

        # Try LP with active set first (faster)
        g = None
        if len(active) < len(conv):
            g, _ = solve_lp_g_active(a, M, active, time_limit=8)
            if g is not None:

                else:
                    g = None

        # Line search with more options
        best_t_obj = best_obj

        for t in t_candidates:

def optimize_lbfgs(a_init, tau_list, max_iter=300, time_budget=30.0):
    """
    Optimize using L-BFGS-B with soft-max parametrization and tau annealing.
    Returns (best_a, best_obj).
    """
    n = len(a_init)
    a = a_init.copy()
    s = np.sum(a)
    if s < 1e-10:
        a = np.ones(n) / n
    else:
        a = a / s
    a = np.maximum(a, 1e-30)
    z = np.log(a)

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    for tau in tau_list:
        if time.monotonic() - start > time_budget:
            break

        def func_and_grad(z_val):
            return soft_max_obj_and_grad_z(z_val, tau)

        try:
            result = minimize(func_and_grad, z, jac=True, method='L-BFGS-B',
                              options={'maxiter': max_iter, 'ftol': 1e-14, 'gtol': 1e-9})
            z_new = result.x
        except Exception:
            continue

        a_new = softmax_fn(z_new)
        a_new = np.maximum(a_new, 0)
        s_new = np.sum(a_new)
        if s_new < 1e-10:
            continue
        a_new = a_new / s_new
        obj = compute_obj(a_new)

        if obj < best_obj:
            best_obj = obj
            best_a = a_new.copy()
            z = z_new

    return best_a, best_obj

                                      tau_start=2.0, tau_end=25.0)

            # L-BFGS-B (medium, ~10s)
            remaining_init = deadline - time.monotonic()
            if remaining_init > 20:
                lfbs_time = min((time_alloc / n_inits) * 0.4, 18.0)
                tau_list = [3.0 / n, 1.5 / n, 0.7 / n, 0.3 / n, 0.15 / n]
                a, obj = optimize_lbfgs(a, tau_list, max_iter=200, time_budget=lfbs_time)
            else:
                obj = compute_obj(a)

            # Quick subgradient polish
            remaining_init = deadline - time.monotonic()
            if remaining_init > 15:
                a = subgradient_optimize(a, max_iters=int(remaining_init * 10),
                                          time_budget=min(remaining_init * 0.1, 8.0),
                                          tau_start=8.0, tau_end=40.0)
                obj = compute_obj(a)

            print(f"    init{init_idx}: n={n}, obj={obj:.6f}")

            if obj < best_score:
                best_score = obj
                best_seq = a.copy()
                print(f"    ** UPDATE: n={n}, obj={best_score:.6f} **")

            candidates.append((a.copy(), obj, n))

    # Phase 2: LP optimization on top candidates
    print(f"\n=== PHASE 2: LP optimization ===")
    print(f"Best before LP: {best_score:.6f}")

    candidates.sort(key=lambda x: x[1])
    n_lp = min(4, len(candidates))

    for ci in range(n_lp):
        if time.monotonic() > deadline - 120:
            break
        remaining = deadline - time.monotonic()
        if remaining < 40:
            break

        a_cand = candidates[ci][0]
        n_cand = candidates[ci][2]
        lp_time = min(remaining * 0.3, 80.0)
        print(f"  LP on candidate {ci} (n={n_cand}, obj={candidates[ci][1]:.6f}), time={lp_time:.0f}s")

        a_lp = alternating_lp(a_cand, max_iters=60, time_budget=lp_time, verbose=True)
        obj_lp = compute_obj(a_lp)
        print(f"    After LP: obj={obj_lp:.6f}")

        if obj_lp < best_score:
            best_score = obj_lp
            best_seq = a_lp.copy()
            print(f"    ** UPDATE: obj={best_score:.6f} **")

        candidates[ci] = (a_lp, obj_lp, n_cand)

    # Phase 3: Intensive polishing of the global best
    print(f"\n=== PHASE 3: Intensive polishing ===")
    print(f"Best before polish: {best_score:.6f}")

    if best_seq is not None and time.monotonic() < deadline - 30:
        remaining = deadline - time.monotonic()
        n_best = len(best_seq)
        print(f"  Polishing n={n_best}, time={remaining:.0f}s")

        a_cur = best_seq.copy()

        # Multiple rounds of polishing
        for round_idx in range(6):
            if time.monotonic() > deadline - 15:
                break
            remaining = deadline - time.monotonic()
            if remaining < 15:
                break

            # LP
            if remaining > 25:
                lp_t = min(remaining * 0.35, 50.0)
                a_new = alternating_lp(a_cur, max_iters=40, time_budget=lp_t, verbose=True)
                obj_new = compute_obj(a_new)
                if obj_new < best_score:
                    best_score = obj_new
                    best_seq = a_new.copy()
                    a_cur = a_new.copy()
                    print(f"  Polish {round_idx} LP: {best_score:.6f}")

            # L-BFGS-B
            remaining = deadline - time.monotonic()
            if remaining > 15:
                tau_vals = [0.5 / n_best, 0.2 / n_best, 0.1 / n_best, 0.05 / n_best]
                a_new, obj_new = optimize_lbfgs(a_cur, tau_vals,
                                                max_iter=300, time_budget=min(remaining * 0.2, 20.0))
                if obj_new < best_score:
                    best_score = obj_new
                    best_seq = a_new.copy()
                    a_cur = a_new.copy()
                    print(f"  Polish {round_idx} L-BFGS: {best_score:.6f}")

            # Subgradient
            remaining = deadline - time.monotonic()
            if remaining > 10:
                a_new = subgradient_optimize(a_cur, max_iters=int(remaining * 12),
                                              time_budget=min(remaining * 0.15, 15.0),
                                              tau_start=15.0, tau_end=80.0)
                obj_new = compute_obj(a_new)
                if obj_new < best_score:
                    best_score = obj_new
                    best_seq = a_new.copy()
                    a_cur = a_new.copy()
                    print(f"  Polish {round_idx} SG: {best_score:.6f}")

    # Phase 4: Perturbation-based exploration from the best
    print(f"\n=== PHASE 4: Perturbation exploration ===")
    print(f"Best before perturb: {best_score:.6f}")

    if time.monotonic() < deadline - 20:
        remaining = deadline - time.monotonic()
        n_best = len(best_seq)
        rng = np.random.default_rng(seed + 99999)

        n_perturb = min(5, int(remaining / 15))
        for ptry in range(n_perturb):
            if time.monotonic() > deadline - 15:
                break
            remaining = deadline - time.monotonic()
            if remaining < 10:
                break

            scale = rng.uniform(0.02, 0.08) * (1 + ptry * 0.3)
            a_pert = best_seq * np.exp(rng.normal(0, scale, n_best))
            a_pert = np.maximum(a_pert, 0)
            s = np.sum(a_pert)
            if s < 1e-10:
                continue
            a_pert = a_pert / s

            # Quick optimize
            a_pert = subgradient_optimize(a_pert, max_iters=400, time_budget=5.0,
                                           tau_start=5.0, tau_end=30.0)
            obj_pert = compute_obj(a_pert)

            if obj_pert < best_score:
                best_score = obj_pert
                best_seq = a_pert.copy()
                print(f"  Perturb {ptry} (scale={scale:.3f}): {best_score:.6f} ** UPDATE **")
            else:
                print(f"  Perturb {ptry} (scale={scale:.3f}): {obj_pert:.6f}")

    # Final validation
    if best_seq is None:
        best_seq = np.ones(200)
        best_score = compute_obj(best_seq)

    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    val = evaluate_sequence(result)
    if val < np.inf:
        best_score = val
    else:
        result = [1.0] * 100
        best_score = 2.0

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_score:.6f}, time={elapsed:.1f}s ===")

    return result
