"""
Hybrid optimizer for minimizing J(a) = 2*n*max(conv(a,a))/sum(a)^2.
Combines: (1) LP mirror descent with dual formulation, (2) L_p smoothed gradient descent,
(3) alternating LP/L_p phases, (4) multi-start with diverse initializations,
(5) structured perturbations for escape from local minima, (6) multi-length exploration.
"""

import numpy as np
from scipy.optimize import linprog
from scipy.sparse import csr_matrix
import time


def fast_conv(a):
    """Full self-convolution via FFT."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]


def compute_J(a):
    """Compute J(a) = 2*n*max(conv(a,a))/(sum(a))^2."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))


def project_to_simplex(x):
    """Project x onto {a >= 0, sum(a) = 1}."""
    n = len(x)
    if n == 0:
        return x
    u = np.sort(x)[::-1]
    cssv = np.cumsum(u)
    rho = np.arange(1, n + 1)
    cond = u * rho > (cssv - 1)
    if not np.any(cond):
        return np.ones(n) / n
    rho_max = np.where(cond)[0][-1] + 1
    theta = (cssv[rho_max - 1] - 1) / rho_max
    return np.maximum(x - theta, 0)


def build_dual_matrix(n, a):
    """Build the dual LP matrix. Dual: min sum(y) s.t. sum_j a_j * y_{i+j} >= 1 for all i, y >= 0."""
    # A_dual has shape (n, 2n-1), A_dual[i, k] = a_{k-i} if valid else 0
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        j_min = max(0, k - n + 1)
        j_max = min(n - 1, k)
        for j in range(j_min, j_max + 1):
            rows.append(j)
            cols.append(k)
            vals.append(a[k - j])
    return csr_matrix((vals, (rows, cols)), shape=(n, 2 * n - 1))


def solve_dual_lp(a, M):
    """Solve dual LP: min M*sum(y) s.t. A_dual @ y >= 1, y >= 0.
    Equivalent to: min sum(y) s.t. A_dual @ y >= 1/M, y >= 0 (scaled).
    Returns (y, objective_value) or (None, None).
    """
    n = len(a)
    A = build_dual_matrix(n, a)
    # Minimize 1^T y subject to -A @ y <= -1, y >= 0
    c = np.ones(2 * n - 1)
    A_ub = -A
    b_ub = -np.ones(n)
    bounds = [(0, None)] * (2 * n - 1)
    try:
        r = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs',
                    options={'time_limit': 10})
        if r.success and r.x is not None:
            y = np.maximum(r.x, 0)
            if y.sum() > 1e-15:
                return y, float(r.fun)
    except Exception:
        pass
    return None, None


def lp_mirror_descent(a, max_iters, deadline, step_decay=0.65, base_step=0.3, min_step=0.01):
    """LP mirror descent using dual LP. Returns (best_a, best_J)."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 1e-10)
    a = a / a.sum()
    n = len(a)
    current_J = compute_J(a)
    best_J = current_J
    best_a = a.copy()
    no_improve = 0

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        b = fast_conv(a)
        M = float(b.max())

        # Solve dual LP to get y, then derive best response g
        y, _ = solve_dual_lp(a, M)
        if y is None:
            # Fallback: uniform over active set
            tol = 1e-10 * M
            active = np.where(b >= M - tol)[0]
            y = np.zeros(2 * n - 1)
            y[active] = 1.0 / len(active)

        # Best response g: A @ g <= M, maximize sum(g)
        # From dual y, the best response is: g_i = M / (A^T y)_i where (A^T y)_i > 0
        # Actually, let's just solve the primal LP directly for g
        # Primal: max 1^T g s.t. A_primal @ g <= M, g >= 0
        # Where A_primal[k, i] = a[k-i] for valid k-i
        # This is the same matrix transposed: A_primal = A_dual.T
        A_primal = A_dual_matrix = build_dual_matrix(n, a).T  # shape (2n-1, n)
        c_primal = -np.ones(n)
        b_ub = np.full(2 * n - 1, M)

        try:

            else:
                g = None
        except Exception:

        if g is None or g.sum() < 1e-10:
            # Fallback: use active set gradient
            tol = 1e-10 * M
            active = np.where(b >= M - tol)[0]
            g = np.ones(n) / n
            no_improve += 1
            if no_improve > 30:
                break
            continue

        g = g / g.sum()

        # Line search with decreasing step
        t = base_step * (min_step / base_step) ** (it / max(1, max_iters / 3))

        best_trial_J = current_J
        best_trial_a = None

        for ls in range(20):

                best_J = current_J
                best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1
            if no_improve > 40:

                break
            continue

        g = g / g.sum()

        # Geometric step size
        t = max(0.02, base_step * np.exp(-0.5 * it / 50.0))
        best_trial_J = current_J
        best_trial_a = None

        for ls in range(20):
            a_trial = (1.0 - t) * a + t * g
            a_trial = np.maximum(a_trial, 0)
            s = a_trial.sum()
            if s < 1e-10:
                t *= 0.5
                continue
            a_trial = a_trial / s
            J_trial = compute_J(a_trial)
            if J_trial < best_trial_J - 1e-14:
                best_trial_J = J_trial
                best_trial_a = a_trial.copy()
                break
            t *= 0.6
            if t < 0.005:
                break

        if best_trial_a is not None:
            a = best_trial_a
            current_J = best_trial_J
            if current_J < best_J:
                best_J = current_J
                best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1
            if no_improve > 50:
                break

    return best_a, best_J


def compute_lp_gradient(a, p=10):
    """Compute gradient of sum_k (conv(a,a)_k)^p with respect to a.
    grad[i] = p * sum_j b_{i+j}^{p-1} * a[j]
    Computed via FFT.
    """
    n = len(a)
    b = fast_conv(a)  # length 2n-1
    w = b.astype(np.float64)
    w = np.maximum(w, 0)
    w = w ** (p - 1)

    # grad[i] = p * sum_{j=0}^{n-1} w[i+j] * a[j]
    # = p * convolve(a[::-1], w, mode='full')[n-1 : 2n-1]
    a_rev = a[::-1]
    # Use FFT for efficiency
    size = 1
    while size < n + (2 * n - 1) - 1:
        size <<= 1
    fa = np.fft.rfft(a_rev, size)
    fw = np.fft.rfft(w, size)
    conv = np.fft.irfft(fa * fw, size)
    grad = p * conv[n - 1: 2 * n - 1]
    return grad


def lp_gradient_descent(a, max_iters, deadline, p=10, lr0=0.1, momentum=0.9):
    """L_p smoothed gradient descent with projection onto simplex."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 1e-10)
    a = a / a.sum()
    n = len(a)
    best_J = compute_J(a)
    best_a = a.copy()
    vel = np.zeros_like(a)
    no_improve = 0

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        grad = compute_lp_gradient(a, p)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-15:
            break

        # Adaptive learning rate
        lr = lr0 / (1.0 + 0.01 * it)
        vel = momentum * vel + (1 - momentum) * grad / gnorm
        a_new = a - lr * vel
        a_new = project_to_simplex(a_new)

        J_new = compute_J(a_new)
        if J_new < best_J:
            best_J = J_new
            best_a = a_new.copy()
            no_improve = 0
        else:
            no_improve += 1
            # Reduce learning rate if no improvement
            lr *= 0.9
            if no_improve > 100:
                break

        a = a_new

    return best_a, best_J


def lp_gradient_descent_adaptive(a, max_iters, deadline, p_start=8):
    """L_p gradient descent with adaptive p (increasing over time for sharper max approximation)."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 1e-10)
    a = a / a.sum()
    n = len(a)
    best_J = compute_J(a)
    best_a = a.copy()
    vel = np.zeros_like(a)
    no_improve = 0
    momentum = 0.9

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        # Adaptive p: start lower, increase towards 20
        p = min(20.0, p_start + 2.0 * it / max(1, max_iters / 4))
        lr = 0.08 / (1.0 + 0.005 * it)

        grad = compute_lp_gradient(a, p)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-15:
            break

        vel = momentum * vel + (1 - momentum) * grad / gnorm
        a_new = a - lr * vel
        a_new = project_to_simplex(a_new)

        J_new = compute_J(a_new)
        if J_new < best_J - 1e-12:
            best_J = J_new
            best_a = a_new.copy()
            no_improve = 0
        else:
            no_improve += 1
            if no_improve > 60:
                break

        a = a_new

    return best_a, best_J


def structured_perturb(a, rng, ptype):
    """Apply a structured perturbation."""
    n = len(a)
    t = np.arange(n, dtype=np.float64) / n

    if ptype == 'shift':
        s = rng.integers(1, max(2, n // 15))
        a_p = np.roll(a, int(s))
    elif ptype == 'modulate':
        freq = rng.uniform(0.5, 6.0)
        amp = rng.uniform(0.03, 0.12)
        phase = rng.uniform(0, 2 * np.pi)
        a_p = a * (1 + amp * np.cos(2 * np.pi * freq * t + phase))
    elif ptype == 'smooth':
        k = int(rng.integers(3, max(4, n // 12)))
        kernel = np.ones(k) / k
        a_p = np.convolve(a, kernel, mode='same')
    elif ptype == 'taper':
        alpha = rng.uniform(0.6, 0.95)
        edge = max(1, int(n * (1 - alpha) / 2))
        window = np.ones(n)
        for i in range(edge):
            frac = (i + 1) / (edge + 1)
            window[i] = frac
            window[n - 1 - i] = frac
        a_p = a * window
    elif ptype == 'mirror':
        half = n // 2
        a_p = np.concatenate([a[:half], a[half:][::-1]]) if n % 2 == 0 else \
              np.concatenate([a[:half], a[half:]])
    elif ptype == 'random_smooth':
        k = max(5, n // 8)
        kernel = np.exp(-0.5 * (np.arange(-k, k + 1) / (k / 3)) ** 2)
        kernel /= kernel.sum()
        noise = rng.normal(0, 0.05, n)
        a_p = a + np.convolve(noise, kernel, mode='same')
    else:
        a_p = a + rng.normal(0, 0.03, n)

    a_p = np.maximum(a_p, 0.001)
    s = a_p.sum()
    if s > 1e-8:
        a_p = a_p / s
    return a_p


def resample_seq(src, dest_n):
    """Resample 1D sequence to new length."""
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    return np.maximum(np.interp(t_new, t_old, src), 0.001)


def generate_inits(n, rng, parent=None):
    """Generate diverse initial sequences."""
    inits = []
    t = np.arange(n, dtype=np.float64) / n

    if parent is not None:
        p = resample_seq(parent, n)
        inits.append(p)
        inits.append(np.maximum(p + rng.normal(0, 0.02, n), 0.01))
        inits.append(np.maximum(p + rng.normal(0, 0.04, n), 0.01))
        inits.append(np.maximum(p + rng.normal(0, 0.08, n), 0.01))

    # Various structured inits
    inits.append(np.ones(n) / n)
    inits.append((0.5 * (1 + np.cos(2 * np.pi * t))))
    inits.append(np.maximum(0.4 - 0.5 * np.cos(2 * np.pi * t) + 0.1 * np.cos(4 * np.pi * t), 0.01))
    inits.append(np.exp(-3 * (t - 0.5) ** 2) + 0.1)
    inits.append(np.exp(-6 * (t - 0.3) ** 2) + np.exp(-6 * (t - 0.7) ** 2) + 0.05)
    inits.append(np.maximum(1 + 0.3 * np.cos(2 * np.pi * 2 * t), 0.01))
    inits.append(np.maximum(1 + 0.2 * np.cos(2 * np.pi * 3 * t) + 0.1 * np.cos(2 * np.pi * 5 * t), 0.01))
    inits.append(np.maximum(1 - 0.3 * np.abs(2 * t - 1), 0.05))

    # Symmetric inits
    half = n // 2
    sym = np.exp(-4 * (np.arange(half) / half - 0.5) ** 2)
    inits.append(np.concatenate([sym, sym[::-1]]) if n % 2 == 0 else np.concatenate([sym, sym[::-1][1:]]))

    # Smooth random
    k = max(5, n // 10)
    kernel = np.exp(-0.5 * (np.arange(-k, k + 1) / (k / 3)) ** 2)
    kernel /= kernel.sum()
    smooth = np.convolve(rng.uniform(0, 1, n), kernel, mode='same')
    inits.append(np.maximum(smooth + 0.1, 0.01))

    # Another smooth random
    smooth2 = np.convolve(rng.uniform(0, 1, n), kernel, mode='same')
    inits.append(np.maximum(smooth2 + 0.1, 0.01))

    result = []
    for a in inits:
        a = np.asarray(a, dtype=np.float64)
        if a.size != n:
            a = resample_seq(a, n)
        a = np.maximum(a, 0.001)
        s = a.sum()
        if s > 1e-6:
            result.append(a / s)
    return result


class ScoreTracker:
    def __init__(self):
        self.best = None
        self.best_score = float('inf')

    def update(self, a):
        a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
        if a.size == 0 or a.sum() < 0.01:
            return
        score = float(evaluate_sequence(a.tolist()))
        if np.isfinite(score) and score < self.best_score:
            self.best = a.copy()
            self.best_score = score
            print(f"  [NEW BEST] J={score:.8f} n={a.size}", flush=True)

    def report(self):
        if self.best is not None:
            print(f"  Current best: J={self.best_score:.8f} n={self.best.size}", flush=True)


def alternating_optimize(a, deadline, total_budget_s, tracker):
    """Alternate between LP mirror descent and L_p gradient descent."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 1e-10)
    a = a / a.sum()
    current_J = compute_J(a)
    tracker.update(a)

    n_phases = 4
    per_phase = total_budget_s / n_phases

    for phase in range(n_phases):
        phase_deadline = min(deadline, time.monotonic() + per_phase)

        if phase % 2 == 0:
            # LP phase
            a, J = lp_primal_only(a, 500, phase_deadline, base_step=0.2)
        else:
            # L_p gradient phase
            a, J = lp_gradient_descent_adaptive(a, 3000, phase_deadline, p_start=8)

        tracker.update(a)
        if J < current_J:
            current_J = J

        if time.monotonic() > deadline:
            break

    return a, current_J


def propose_candidate(seed=42, budget_s=1500, **kwargs):
    """
    Hybrid optimizer combining LP mirror descent, L_p gradient descent,
    multi-start, and structured perturbations.
    """
    rng = np.random.default_rng(seed)
    t_start = time.monotonic()
    deadline = t_start + float(budget_s) - 30.0
    tracker = ScoreTracker()

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        if p.ndim == 1 and p.size > 0 and np.all(np.isfinite(p)):
            parent = np.maximum(p, 0.0)
            tracker.update(parent)
            print(f"Parent: n={parent.size}, J={tracker.best_score:.6f}", flush=True)
    except (NameError, AttributeError):
        pass

    if parent is None:
        parent = np.ones(640)
        tracker.update(parent)

    best_a = parent.copy()
    best_J = compute_J(best_a)
    best_n = best_a.size

    # === Phase 1: LP refinement on parent ===
    if time.monotonic() < deadline - 600:
        t0 = time.monotonic()
        print(f"\n--- Phase 1: LP on parent (n={best_n}) ---", flush=True)
        phase_dl = min(deadline - 500, t0 + 200)
        a1 = best_a.copy()
        a1, J1 = lp_primal_only(a1, 500, phase_dl, base_step=0.25)
        tracker.update(a1)
        print(f"  J={J1:.8f} ({time.monotonic()-t0:.1f}s)", flush=True)
        if J1 < best_J:
            best_a, best_J, best_n = a1.copy(), J1, a1.size

    # === Phase 2: L_p gradient on best ===
    if time.monotonic() < deadline - 500:
        t0 = time.monotonic()
        print(f"\n--- Phase 2: L_p gradient (n={best_n}) ---", flush=True)
        phase_dl = min(deadline - 400, t0 + 150)
        a2, J2 = lp_gradient_descent_adaptive(best_a, 5000, phase_dl, p_start=8)
        tracker.update(a2)
        print(f"  J={J2:.8f} ({time.monotonic()-t0:.1f}s)", flush=True)
        if J2 < best_J:
            best_a, best_J, best_n = a2.copy(), J2, a2.size

    # === Phase 3: LP on L_p result ===
    if time.monotonic() < deadline - 400:
        t0 = time.monotonic()
        print(f"\n--- Phase 3: LP after L_p (n={best_n}) ---", flush=True)
        phase_dl = min(deadline - 300, t0 + 150)
        a3 = best_a.copy()
        a3, J3 = lp_primal_only(a3, 500, phase_dl, base_step=0.2)
        tracker.update(a3)
        print(f"  J={J3:.8f} ({time.monotonic()-t0:.1f}s)", flush=True)
        if J3 < best_J:
            best_a, best_J, best_n = a3.copy(), J3, a3.size

    # === Phase 4: L_p on LP result ===
    if time.monotonic() < deadline - 300:
        t0 = time.monotonic()
        print(f"\n--- Phase 4: L_p after LP (n={best_n}) ---", flush=True)
        phase_dl = min(deadline - 200, t0 + 120)
        a4, J4 = lp_gradient_descent_adaptive(best_a, 5000, phase_dl, p_start=10)
        tracker.update(a4)
        print(f"  J={J4:.8f} ({time.monotonic()-t0:.1f}s)", flush=True)
        if J4 < best_J:
            best_a, best_J, best_n = a4.copy(), J4, a4.size

    # === Phase 5: Structured perturbations ===
    if time.monotonic() < deadline - 200:
        t0 = time.monotonic()
        print(f"\n--- Phase 5: Perturbations (n={best_n}) ---", flush=True)
        ptypes = ['shift', 'modulate', 'smooth', 'taper', 'random_smooth', 'mirror',
                  'shift', 'modulate']
        remaining = deadline - 150 - time.monotonic()
        per_pert = max(15, remaining / len(ptypes) / 2)

        for i, pt in enumerate(ptypes):
            if time.monotonic() > deadline - 130:
                break
            a_pert = structured_perturb(best_a, rng, pt)
            # Quick LP from perturbed
            dl = time.monotonic() + per_pert
            a_pert, J_pert = lp_primal_only(a_pert, 100, dl, base_step=0.3)
            tracker.update(a_pert)
            if J_pert < best_J:
                best_a, best_J, best_n = a_pert.copy(), J_pert, a_pert.size
                print(f"  [{pt}] J={J_pert:.8f} (improved!)", flush=True)
            else:
                # Try L_p from perturbed
                if time.monotonic() < deadline - 130:
                    a_pert2, J_pert2 = lp_gradient_descent_adaptive(a_pert, 2000,
                                                                     time.monotonic() + per_pert / 2,
                                                                     p_start=10)
                    tracker.update(a_pert2)
                    if J_pert2 < best_J:
                        best_a, best_J, best_n = a_pert2.copy(), J_pert2, a_pert2.size
                        print(f"  [{pt}+Lp] J={J_pert2:.8f} (improved!)", flush=True)

        print(f"Phase 5 done ({time.monotonic()-t0:.1f}s)", flush=True)

    # === Phase 6: Multi-length exploration ===
    if time.monotonic() < deadline - 150:
        t0 = time.monotonic()
        print(f"\n--- Phase 6: Multi-length (base n={best_n}) ---", flush=True)
        lengths = sorted(set([
            max(256, best_n - 128), best_n + 128, best_n + 256,
            min(2048, best_n * 2)
        ]))
        lengths = [n for n in lengths if 256 <= n <= 2048 and n != best_n]

        per_len = max(20, (deadline - 100 - time.monotonic()) / max(1, len(lengths)))

        for n_try in lengths:
            if time.monotonic() > deadline - 80:
                break
            a_try = resample_seq(best_a, n_try)
            a_try = a_try / a_try.sum()
            dl = time.monotonic() + per_len
            a_try, J_try = lp_primal_only(a_try, 300, dl, base_step=0.25)
            tracker.update(a_try)
            print(f"  n={n_try}: J={J_try:.8f}", flush=True)
            if J_try < best_J:
                best_a, best_J, best_n = a_try.copy(), J_try, n_try

        print(f"Phase 6 done ({time.monotonic()-t0:.1f}s)", flush=True)

    # === Phase 7: Multi-start with diverse inits at best length ===
    if time.monotonic() < deadline - 100:
        t0 = time.monotonic()
        print(f"\n--- Phase 7: Multi-start (n={best_n}) ---", flush=True)
        inits = generate_inits(best_n, rng, parent=best_a)
        remaining = deadline - 60 - time.monotonic()
        per_init = max(10, remaining / len(inits))

        for i, a_init in enumerate(inits[:8]):  # Limit to 8 inits
            if time.monotonic() > deadline - 50:
                break
            dl = time.monotonic() + per_init
            a_opt, J_opt = lp_primal_only(a_init, 200, dl, base_step=0.3)
            tracker.update(a_opt)
            if J_opt < best_J:
                best_a, best_J, best_n = a_opt.copy(), J_opt, a_opt.size
                print(f"  Init {i}: J={J_opt:.8f} (improved!)", flush=True)

        print(f"Phase 7 done ({time.monotonic()-t0:.1f}s)", flush=True)

    # === Phase 8: Long final optimization ===
    if time.monotonic() < deadline - 20:
        t0 = time.monotonic()
        print(f"\n--- Phase 8: Final long LP (n={best_n}) ---", flush=True)
        a_final = best_a.copy()
        a_final, J_final = lp_primal_only(a_final, 5000, deadline - 10, base_step=0.15)
        tracker.update(a_final)
        print(f"  J={J_final:.8f} ({time.monotonic()-t0:.1f}s)", flush=True)
        if J_final < best_J:
            best_a, best_J = a_final.copy(), J_final

        # One more L_p polish
        if time.monotonic() < deadline - 5:
            a_polish, J_polish = lp_gradient_descent_adaptive(best_a, 3000, deadlock := deadline - 2, p_start=12)
            tracker.update(a_polish)
            if J_polish < best_J:
                best_a, best_J = a_polish.copy(), J_polish
                print(f"  L_p polish: J={J_polish:.8f}", flush=True)

    # Final
    if tracker.best is not None:
        final_a = tracker.best
        final_score = tracker.best_score
    else:
        final_a = best_a
        final_score = compute_J(best_a)

    elapsed = time.monotonic() - t_start
    print(f"\n{'='*50}", flush=True)
    print(f"FINAL: J={final_score:.8f} n={final_a.size} time={elapsed:.0f}s", flush=True)
    print(f"{'='*50}", flush=True)

    return [float(x) for x in final_a]
