import numpy as np

def compute_conv(a):
    """Compute linear convolution a*a using FFT."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    conv = np.fft.irfft(fa * fa, size)
    return conv[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    conv = compute_conv(a)
    max_b = float(np.max(conv))
    sum_a = float(np.sum(a))
    if sum_a < 0.01:
        return np.inf
    return 2.0 * n * max_b / (sum_a * sum_a)


def build_toeplitz(a):
    """Build Toeplitz matrix for linear convolution with a."""
    n = len(a)
    A = np.zeros((2 * n - 1, n), dtype=np.float64)
    for j in range(n):
        k_end = min(j + n, 2 * n - 1)
        A[j:k_end, j] = a[:k_end - j]
    return A


def solve_lp_maxsum(a, M, time_limit=5):
    """LP: max sum(g) s.t. (a*g)[k] <= M, g >= 0."""
    n = len(a)
    A = build_toeplitz(a)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': time_limit})
        if res.status == 0:
            g = np.maximum(res.x, 0)
            return g, float(np.sum(g))
    except Exception:
        pass
    return None, None


def peak_focused_subgradient(a, tau, step, focus_top_k=5):
    """Subgradient step focused on the top peaks of the convolution."""
    n = len(a)
    conv = compute_conv(a)
    max_conv = np.max(conv)
    
    # Find the top-k peaks
    if focus_top_k and focus_top_k < len(conv):
        top_idx = np.argsort(conv)[-focus_top_k:]
        # Create weights emphasizing the peaks
        weights = np.zeros_like(conv)
        for idx in top_idx:
            weights[idx] = 1.0
    else:
        shifted = tau * (conv - max_conv)
        shifted = np.clip(shifted, -500, 500)
        weights = np.exp(shifted)
        w_sum = np.sum(weights)
        if w_sum < 1e-300:
            idx = int(np.argmax(conv))
            weights = np.zeros(len(conv))
            weights[idx] = 1.0
        else:
            weights /= w_sum
    
    # Compute gradient
    c = np.correlate(weights, a, mode='full')
    grad = 2.0 * c[n - 1:n - 1 + n]
    
    # Normalize gradient
    grad_mean = np.mean(grad)
    if abs(grad_mean) > 1e-15:
        grad = grad - grad_mean  # Center to preserve sum approximately
    
    log_update = -step * grad
    log_update = np.clip(log_update, -50, 50)
    a_new = a * np.exp(log_update)
    a_new = np.maximum(a_new, 1e-15)
    s = np.sum(a_new)
    if s > 1e-10:
        a_new = a_new / s
    return a_new


def smooth_subgradient(a, tau, step):

            stall += 1
            if stall > 50:
                step *= 0.8
                tau *= 1.3
                if step < 0.0005:
                    break

                rng = np.random.default_rng(it * 17 + 3)
                a = a * np.exp(rng.normal(0, 0.005, n))
                a = np.maximum(a, 1e-12)
                s = np.sum(a)
                if s > 1e-10:
                    a = a / s
                stall = 0
    
    return best_a


def resample_sequence(a, new_n):
    """Resample using cubic-like interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    old_k = np.arange(n, dtype=np.float64)
    new_k = np.linspace(0, n - 1, new_n)
    # Use linear interpolation (sufficient)
    a_new = np.interp(new_k, old_k, a)
    a_new = np.maximum(a_new, 0)
    s = np.sum(a_new)
    if s > 1e-10:
        a_new = a_new / s
    return a_new

        parent = np.asarray(height_sequence_1, dtype=np.float64)
        parent = np.clip(parent, 0, 1000.0)
        if parent.size > 0 and np.isfinite(parent).all():
            obj = compute_obj(parent)
            print(f"Parent (n={parent.size}): obj={obj:.6f}")

    except Exception:
        pass
    return None, np.inf
