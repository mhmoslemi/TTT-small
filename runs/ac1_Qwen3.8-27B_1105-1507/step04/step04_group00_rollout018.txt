def fast_conv(a):

    idx = np.add.outer(np.arange(n), np.arange(n))
    H = p[idx].copy()
    if reg > 0:
        H += reg * np.eye(n)
    try:
        y = np.linalg.solve(H, np.ones(n))
        if np.all(np.isfinite(y)):
            return y
    except np.linalg.LinAlgError:

def build_A_sparse(n, a):
    """Build sparse constraint matrix for LP."""
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        j_min = max(0, k - n + 1)
        j_max = min(n - 1, k)
        for j in range(j_min, j_max + 1):
            rows.append(k)
            cols.append(j)
            vals.append(a[k - j])
    return csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))


def lp_best_response(a, M):
    """LP: max sum(g) s.t. (a*g)[k] <= M for all k, g >= 0."""
    n = len(a)
    A = build_A_sparse(n, a)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    try:
        r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                    options={'time_limit': 20})
        if r.success and r.x is not None:
            g = np.maximum(r.x, 0)
            if g.sum() > 1e-15:
                return g
    except Exception:
        pass
    return None

def multiplicative_update(a, deadline, max_iters=5000, gamma=10.0, lr=0.1):

        b_shifted = b - M
        exp_b = np.exp(tau * b_shifted)

            hi = min(n - 1, k)

def resample_seq(src, dest_n):
    """Resample 1D sequence to new length via linear interpolation."""
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)


def generate_cazac_like(n, rng):
    """Generate CAZAC-inspired initialization for non-negative sequences."""
    # Zadoff-Chu: x_k = exp(-j*pi*root*k*(k+1)/N) for root odd, or exp(-j*pi*root*k^2/N)
    root = 3
    if root % 2 == 0:
        phase = -np.pi * root * np.arange(n) ** 2 / n
    else:
        phase = -np.pi * root * np.arange(n) * (np.arange(n) + 1) / n
    # Take magnitude (which is 1 for CAZAC), modulate
    # For non-negative, use 1 + cos of the phase (always >= 0)
    seq = 1.0 + 0.4 * np.cos(phase)
    # Add boundary taper
    taper = np.ones(n)
    taper[:n // 10] = 0.5
    taper[-n // 10:] = 0.5
    seq = seq * taper
    return np.maximum(seq, 0.01)


def generate_diverse_inits(n, rng, parent=None):
    """Generate diverse initial sequences."""
    inits = []
    t_arr = np.linspace(0, 1, n, endpoint=False)

    if parent is not None:
        inits.append(resample_seq(parent, n))
        p = resample_seq(parent, n)
        inits.append(np.maximum(p + rng.normal(0, 0.05, n), 0.01))
        inits.append(np.maximum(p + rng.normal(0, 0.15, n), 0.01))

    inits.append(np.ones(n))
    inits.append(generate_cazac_like(n, rng))

    # Various cosine patterns
    for freq in [1, 2, 3, 5]:
        inits.append(np.maximum(1.0 + 0.3 * np.cos(2 * np.pi * freq * t_arr), 0.01))

    # Gaussian mixtures
    inits.append(np.exp(-4.0 * (t_arr - 0.5) ** 2) + 0.1)
    inits.append(0.2 + np.exp(-8 * (t_arr - 0.3) ** 2) + np.exp(-8 * (t_arr - 0.7) ** 2))

    # Smooth random
    ksize = max(5, n // 20)
    kernel = np.ones(ksize) / ksize
    smooth = np.convolve(rng.uniform(0, 1, n), kernel, mode='same')
    inits.append(np.maximum(smooth + 0.1, 0.01))

    return inits


class ScoreTracker:
    """Tracks the best sequence found so far."""

    def __init__(self):
        self.best = None
        self.best_score = float('inf')
        self.t_start = time.monotonic()

    def update(self, a):
        a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
        if a.size == 0 or a.sum() < 0.01:
            return
        score = float(evaluate_sequence(a.tolist()))
        if np.isfinite(score) and score < self.best_score:
            self.best = a.copy()
            self.best_score = score
            elapsed = time.monotonic() - self.t_start
            print(f"  ** J={score:.8f} n={a.size} t={elapsed:.0f}s", flush=True)

    def report(self, label=""):
        if self.best is not None:
            print(f"  {label} best: J={self.best_score:.8f} n={self.best.size}", flush=True)


def propose_candidate(seed=42, budget_s=1500, **kwargs):
    """
    Multi-phase search using dual Hankel method (main innovation),
    LP mirror descent, multiplicative updates, and coordinate descent.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + float(budget_s) - 15.0
    tracker = ScoreTracker()

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        if p.ndim == 1 and p.size > 0 and np.all(np.isfinite(p)):
            parent = np.maximum(p, 0.0)
            tracker.update(parent)
            print(f"Parent loaded: n={parent.size}, J={tracker.best_score:.6f}", flush=True)
    except (NameError, AttributeError):
        pass

    if parent is None:
        parent = np.ones(256)
        tracker.update(parent)

    current_best = parent.copy()

    # === PHASE 1: Dual Hankel Method (main innovation) ===
    p1_budget = 350.0
    p1_start = time.monotonic()
    print(f"\n{'='*60}", flush=True)
    print(f"PHASE 1: Dual Hankel Descent (budget={p1_budget:.0f}s)", flush=True)
    print(f"{'='*60}", flush=True)

    # Try multiple lengths
    lengths = [128, 192, 256, 384, 512, 768]
    n_parent = parent.size
    # Prioritize lengths near the parent
    lengths.sort(key=lambda n: abs(n - n_parent))

    for n in lengths:
        if time.monotonic() > deadline - 300:
            break
        if n < 64 or n > 2048:
            continue

        # Time allocation
        time_for_n = min(120, p1_budget * (3.0 / n) / sum(3.0 / nn for nn in lengths if 64 <= nn <= 2048))
        time_for_n = max(30, time_for_n)
        phase_deadline = min(time.monotonic() + time_for_n, deadline - 250)

        if time.monotonic() >= phase_deadline:
            continue

        t0 = time.monotonic()
        inits = generate_diverse_inits(n, rng, current_best)
        best_for_n = None
        best_M_for_n = float('inf')

        for idx, a0 in enumerate(inits):
            if time.monotonic() > phase_deadline:
                break
            a0 = np.maximum(a0.astype(np.float64), 1e-8)
            a0 = a0 / a0.sum()

            a_opt, M_opt = dual_hankel_descent(
                a0, phase_deadline, max_iters=5000,
                gamma_start=1.5, gamma_end=25.0,
                blend=0.5, explore_frac=0.05
            )

            M_val = fast_conv(a_opt).max()
            if M_val < best_M_for_n:
                best_M_for_n = M_val
                best_for_n = a_opt.copy()

        if best_for_n is not None:
            J_val = compute_J(best_for_n)
            print(f"  n={n:4d}: M={best_M_for_n:.8f} J={J_val:.6f} "
                  f"({time.monotonic()-t0:.1f}s)", flush=True)
            if J_val < compute_J(current_best):
                current_best = best_for_n.copy()
            tracker.update(best_for_n)

    tracker.report("Phase 1 complete")
    print(f"Phase 1 total time: {time.monotonic()-p1_start:.1f}s", flush=True)

    # === PHASE 2: LP Mirror Descent refinement ===
    p2_start = time.monotonic()
    p2_budget = 300.0
    print(f"\n{'='*60}", flush=True)
    print(f"PHASE 2: LP Mirror Descent (budget={p2_budget:.0f}s)", flush=True)
    print(f"{'='*60}", flush=True)

    if tracker.best is not None:
        n_best = tracker.best.size
        est_lp_time = 0.01 * (n_best / 128) ** 2
        max_iters = max(50, min(1000, int(p2_budget / max(0.02, est_lp_time))))
        print(f"  n={n_best}, est LP time={est_lp_time*1000:.0f}ms, max_iters={max_iters}", flush=True)

        a = run_lp_mirror_descent(tracker.best, max_iters, min(time.monotonic() + p2_budget, deadline - 150))
        tracker.update(a)

        # Second pass
        if time.monotonic() < deadline - 100:
            a2 = run_lp_mirror_descent(a, 200, deadline - 100)
            tracker.update(a2)

    tracker.report("Phase 2 complete")
    print(f"Phase 2 total time: {time.monotonic()-p2_start:.1f}s", flush=True)

    # === PHASE 3: Multiplicative updates ===
    p3_start = time.monotonic()
    p3_budget = 200.0
    print(f"\n{'='*60}", flush=True)
    print(f"PHASE 3: Multiplicative Updates (budget={p3_budget:.0f}s)", flush=True)
    print(f"{'='*60}", flush=True)

    if tracker.best is not None:
        a = multiplicative_update(tracker.best, min(time.monotonic() + p3_budget, deadline - 100),
                                  max_iters=10000, gamma=15.0, lr=0.1)
        tracker.update(a)

        # Different hyperparameters
        if time.monotonic() < deadline - 80:
            a2 = multiplicative_update(tracker.best, deadline - 80,
                                       max_iters=5000, gamma=5.0, lr=0.2)
            tracker.update(a2)

    tracker.report("Phase 3 complete")
    print(f"Phase 3 total time: {time.monotonic()-p3_start:.1f}s", flush=True)

    # === PHASE 4: Coordinate Descent ===
    p4_start = time.monotonic()
    p4_budget = 200.0
    print(f"\n{'='*60}", flush=True)
    print(f"PHASE 4: Coordinate Descent (budget={p4_budget:.0f}s)", flush=True)
    print(f"{'='*60}", flush=True)

    if tracker.best is not None and tracker.best.size <= 600:
        a = coordinate_descent(tracker.best, min(time.monotonic() + p4_budget, deadline - 80),
                               n_outer=3, n_inner=3)
        tracker.update(a)

    tracker.report("Phase 4 complete")
    print(f"Phase 4 total time: {time.monotonic()-p4_start:.1f}s", flush=True)

    # === PHASE 5: Smooth gradient + Subgradient polish ===
    p5_start = time.monotonic()
    p5_budget = 150.0
    print(f"\n{'='*60}", flush=True)
    print(f"PHASE 5: Gradient Polish (budget={p5_budget:.0f}s)", flush=True)
    print(f"{'='*60}", flush=True)

    if tracker.best is not None:
        a = run_smooth_gradient(tracker.best, 5000, min(time.monotonic() + 80, deadline - 60),
                                initial_tau=5.0, final_tau=80.0,
                                lr_init=0.15, lr_min=0.01)
        tracker.update(a)

        # Subgradient
        n = len(tracker.best)
        best_a = tracker.best.copy()
        best_M = fast_conv(best_a).max()
        running_sum = np.zeros(n)
        p5_sub_deadline = min(time.monotonic() + 70, deadline - 50)

        for it in range(3000):
            if time.monotonic() > p5_sub_deadline:
                break
            lr = 0.1 / (it + 1) ** 0.5
            b = fast_conv(best_a)
            M = b.max()
            tol = 0.01 * M
            active = np.where(b >= M - tol)[0]

            grad = np.zeros(n)
            for k in active:
                lo = max(0, k - n + 1)
                hi = min(n - 1, k)
                grad[lo:hi + 1] += 2.0 * best_a[k - hi:k - lo + 1][::-1]
            gnorm = np.linalg.norm(grad)
            if gnorm > 1e-15:
                grad /= gnorm

            a_new = np.maximum(best_a - lr * grad, 0.0)
            s = a_new.sum()
            if s < 1e-10:
                continue
            a_new /= s
            running_sum += a_new
            best_a = a_new

            if M < best_M * (1 - 1e-9):
                best_M = M

        # Check averaged
        if it > 10:
            a_avg = running_sum / (it + 1)
            a_avg = np.maximum(a_avg, 0.0)
            s = a_avg.sum()
            if s > 1e-10:
                a_avg /= s
                tracker.update(a_avg)

    tracker.report("Phase 5 complete")
    print(f"Phase 5 total time: {time.monotonic()-p5_start:.1f}s", flush=True)

    # === PHASE 6: Final dual Hankel + LP on best ===
    p6_start = time.monotonic()
    print(f"\n{'='*60}", flush=True)
    print(f"PHASE 6: Final Refinement", flush=True)
    print(f"{'='*60}", flush=True)

    if tracker.best is not None and time.monotonic() < deadline - 20:
        # One more dual Hankel pass
        a, M = dual_hankel_descent(tracker.best, min(time.monotonic() + 60, deadline - 20),
                                   max_iters=3000, gamma_start=2.0, gamma_end=40.0,
                                   blend=0.7, explore_frac=0.02)
        tracker.update(a)

        # Final LP pass
        if time.monotonic() < deadline - 5:
            a2 = run_lp_mirror_descent(tracker.best, 50, deadline)
            tracker.update(a2)

    tracker.report("Phase 6 complete")

    # Fallback
    if tracker.best is None:
        tracker.best = np.ones(128)
        tracker.best_score = compute_J(tracker.best)

    print(f"\n{'='*60}", flush=True)
    print(f"FINAL: J={tracker.best_score:.8f} n={tracker.best.size} "
          f"time={time.monotonic()-tracker.t_start:.0f}s", flush=True)
    print(f"{'='*60}", flush=True)

    return [float(x) for x in tracker.best]
