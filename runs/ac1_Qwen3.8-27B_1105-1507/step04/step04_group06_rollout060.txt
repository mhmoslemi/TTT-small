"""
Minimizes J(a) = 2*n*max(convolve(a,a))/(sum(a))² using a hybrid approach:
1. Adam optimizer with backtracking line search and log-sum-exp smoothing
2. LP mirror descent with golden-section line search for precise refinement
3. Structured perturbations (phase shifts, frequency mods, amplitude warps)
4. Multi-length exploration (512, 1024, 2048)
5. Adaptive scheduling based on progress and stall detection

import numpy as np
from scipy.optimize import linprog
from scipy.sparse import csr_matrix
import time

_A_cache = {}


def fast_conv(a):
    n = len(a)
    s = 1
    while s < 2 * n - 1:
        s <<= 1
    fa = np.fft.rfft(a, s)
    return np.fft.irfft(fa * fa, s)[:2 * n - 1]


def compute_J(a):
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))


def smooth_max_and_grad(a, beta):
    n = len(a)
    b = fast_conv(a)
    b_max = float(b.max())
    shifted = beta * (b - b_max)
    shifted = np.clip(shifted, -500, 500)
    exp_s = np.exp(shifted)
    Z = float(exp_s.sum())
    if Z < 1e-300:
        p = np.zeros(2 * n - 1)
        p[np.argmax(b)] = 1.0
    else:
        p = exp_s / Z
    g_val = b_max + np.log(max(Z, 1e-300)) / beta
    S = np.convolve(a, p[::-1], mode='full')
    S = S[n - 1:2 * n - 1][::-1]
    grad = 2.0 * S / beta
    return g_val, grad


def J_and_grad(a, beta):
    n = len(a)
    s = float(a.sum())
    if s < 1e-10:
        return 1e10, np.zeros(n)
    g_val, grad_g = smooth_max_and_grad(a, beta)
    J_val = 2.0 * n * g_val / (s * s)
    grad_J = 2.0 * n * (s * grad_g - 2.0 * g_val) / (s ** 3)
    return J_val, grad_J


def get_lp_structure(n):
    if n not in _A_cache:
        rows = []
        cols = []
        for k in range(2 * n - 1):
            jl = max(0, k - n + 1)
            jh = min(n - 1, k)
            for j in range(jl, jh + 1):
                rows.append(k)
                cols.append(j)
        _A_cache[n] = (np.array(rows, dtype=np.int32), np.array(cols, dtype=np.int32))
    return _A_cache[n]


def lp_best_response(a, M):
    n = len(a)
    c = -np.ones(n)
    rows_arr, cols_arr = get_lp_structure(n)
    vals = a[rows_arr - cols_arr]
    A = csr_matrix((vals, (rows_arr, cols_arr)), shape=(2 * n - 1, n))
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    try:
        r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
        if r.success and r.x is not None:
            g = np.maximum(r.x, 0.0)
            g_sum = g.sum()
            if g_sum > 1e-10:
                return g / g_sum
    except Exception:
        pass
    return None

            if not np.isfinite(J_val):

            v = b2 * v + (1 - b2) * (grad ** 2)
            m_hat = m / (1 - b1 ** t)

    best_a_so_far = a.copy()

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        b = fast_conv(a)
        M = float(b.max())

        g = lp_best_response(a, M)
        if g is None:
            break

        # Golden section line search
        best_t = 0.0
        best_ls_J = current_J
        gr = (np.sqrt(5.0) - 1.0) / 2.0
        lo, hi = 0.0, 1.0
        J_at_1 = compute_J(g)
        if J_at_1 < best_ls_J:
            best_ls_J = J_at_1
            best_t = 1.0

        for _ in range(n_line_evals):
            t1 = hi - gr * (hi - lo)
            t2 = lo + gr * (hi - lo)
            a1 = (1.0 - t1) * a + t1 * g
            a2 = (1.0 - t2) * a + t2 * g
            J1 = compute_J(a1)
            J2 = compute_J(a2)
            if J1 < J2:
                hi = t2
            else:
                lo = t1

        t_mid = (lo + hi) / 2.0
        a_new = (1.0 - t_mid) * a + t_mid * g
        J_new = compute_J(a_new)

        if J_new < current_J - 1e-12:
            a = a_new
            current_J = J_new
            stall_count = 0
            if J_new < best_J_so_far:
                best_J_so_far = J_new
                best_a_so_far = a.copy()
                if tracker is not None:
                    tracker.update(a)
        else:
            improved = False
            for t_try in [0.01, 0.005, 0.02, 0.05, 0.001]:
                a_try = (1.0 - t_try) * a + t_try * g
                a_try = np.maximum(a_try, 0.0)
                s = a_try.sum()
                if s > 1e-10:
                    a_try = a_try / s
                    J_try = compute_J(a_try)
                    if J_try < current_J - 1e-14:
                        a = a_try
                        current_J = J_try
                        stall_count = 0
                        if J_try < best_J_so_far:
                            best_J_so_far = J_try
                            best_a_so_far = a.copy()
                            if tracker is not None:
                                tracker.update(a)
                        improved = True
                        break
            if not improved:
                stall_count += 1
                if stall_count >= stall_patience:
                    break

        if it % 200 == 0 and it > 0:
            elapsed_msg = f"iter {it}: J={current_J:.8f}"
            print(f"    {elapsed_msg}", flush=True)

    if tracker is not None:
        tracker.update(best_a_so_far)
    return best_a_so_far, best_J_so_far


class ScoreTracker:
    def __init__(self):
        self.best = None
        self.best_score = float('inf')
        self.t_start = time.monotonic()

    def update(self, a):
        a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
        if a.size == 0 or a.sum() < 0.01:
            return
        try:
            score = float(evaluate_sequence(a.tolist()))
        except Exception:
            return
        if np.isfinite(score) and score < self.best_score:
            self.best = a.copy()
            self.best_score = score
            elapsed = time.monotonic() - self.t_start
            print(f"  * NEW BEST: J={score:.8f} n={a.size} t={elapsed:.1f}s", flush=True)

    def report(self, label=""):
        if self.best is not None:
            print(f"  {label} best: J={self.best_score:.8f} n={self.best.size}", flush=True)


def resample_sequence(src, dest_n):
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    result = np.interp(t_new, t_old, src)
    return np.maximum(result, 0.0)


def generate_structured_perturbations(base, n, rng):
    """Generate diverse structured perturbations of base sequence."""
    t_arr = np.linspace(0, 1, n, endpoint=False)
    perturbations = []

    # Phase shifts
    for shift in [1, 2, 3, 5, 8, 13, 21]:
        perturbations.append(np.roll(base, shift))

    # Frequency modulation (multiply by cosine)
    for freq in [1.5, 2, 3, 4.5, 6]:
        mod = 1.0 + 0.15 * np.cos(2 * np.pi * freq * t_arr)
        perturbations.append(base * mod)

    # Amplitude warping
    for gamma in [0.8, 1.2, 0.7, 1.3, 1.5]:
        perturbations.append(base ** gamma * (base.sum() ** (1 - gamma)))

    # Tapered windows
    for width in [0.3, 0.5, 0.7]:
        window = np.exp(-(t_arr - 0.5) ** 2 / (2 * width ** 2))
        perturbations.append(base * (0.5 + 0.5 * window))

    # Smooth Gaussian perturbations
    for sigma in [0.03, 0.05, 0.08, 0.12]:
        noise = rng.normal(0, sigma, n)
        kernel_size = max(3, n // 64)
        kernel = np.ones(kernel_size) / kernel_size
        noise = np.convolve(noise, kernel, mode='same')
        perturbations.append(base * base + noise * base)

    # Random sparse spikes
    for n_spikes in [3, 5, 10]:
        spiked = base.copy()
        idx = rng.choice(n, size=min(n_spikes, n // 4), replace=False)
        spiked[idx] *= rng.uniform(0.3, 3.0, len(idx))
        perturbations.append(spiked)

    return perturbations


def propose_candidate(seed=42, budget_s=1500, **kwargs):
    """
    Multi-phase optimization for minimizing J(a) = 2n*max(conv(a,a))/(sum(a))^2.

    Phase 1: Adam with log-sum-exp smoothing (diverse starts, annealed beta)
    Phase 2: LP mirror descent with golden-section line search (main refinement)
    Phase 3: Structured perturbations + short LP refinement
    Phase 4: Final extended LP polish + length exploration
    """
    rng = np.random.default_rng(seed)
    total_deadline = time.monotonic() + float(budget_s) - 30.0
    tracker = ScoreTracker()

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        if p.ndim == 1 and p.size > 0 and np.all(np.isfinite(p)):
            parent = np.maximum(p, 0.0)
            tracker.update(parent)
            print(f"Parent loaded: n={parent.size}, J={tracker.best_score:.8f}", flush=True)
    except (NameError, AttributeError):
        pass

    if parent is None:
        parent = np.ones(1024)
        tracker.update(parent)
        print(f"Using default parent: n=1024, J={tracker.best_score:.8f}", flush=True)

    # === Phase 1: Adam Optimization (450s) ===
    t0 = time.monotonic()
    p1_deadline = time.monotonic() + 450.0
    print(f"\n{'='*50} Phase 1: Adam Optimization {'='*50}", flush=True)

    n_primary = parent.size
    if n_primary not in [256, 512, 1024, 2048]:
        n_primary = 1024

    # Build diverse initializations
    t_arr = np.linspace(0, 1, n_primary, endpoint=False)
    inits = []
    inits.append(('parent', parent / parent.sum()))

    # Cosine families
    for freq in [2, 3, 4, 5, 7]:
        c = 1.0 + 0.25 * np.cos(2.0 * np.pi * freq * t_arr)
        inits.append((f'cos{freq}', c / c.sum()))

    # Uniform
    inits.append(('uniform', np.ones(n_primary) / n_primary))

    # Random
    r = rng.uniform(0.5, 1.5, n_primary)
    inits.append(('random', r / r.sum()))

    # Gaussian bump
    bump = np.exp(-3.0 * (t_arr - 0.5) ** 2) + 0.1
    inits.append(('bump', bump / bump.sum()))

    # Power law
    pw = (t_arr + 0.05) ** 0.7
    inits.append(('power', pw / pw.sum()))

    # Beta schedule for Adam
    adam_beta_schedule = [
        (0.5, 60),
        (1.0, 80),
        (3.0, 100),
        (10.0, 120),
        (30.0, 150),
        (100.0, 200),
    ]

    for idx, (name, a0) in enumerate(inits):
        if time.monotonic() > p1_deadline:
            break
        if a0.sum() < 0.01:
            continue

        t_start = time.monotonic()
        # Give more time to parent-based initializations
        if idx == 0:
            schedule = adam_beta_schedule
        else:
            schedule = adam_beta_schedule[:4]  # Less time for others

        a_opt, J_opt = adam_with_linesearch(a0, p1_deadline, schedule,
                                             base_lr=0.1, tracker=tracker)
        print(f"  Adam ({name}): J={J_opt:.8f} ({time.monotonic()-t_start:.1f}s)", flush=True)

    tracker.report("Phase 1 complete")

    # === Phase 2: LP Mirror Descent (550s) ===
    if tracker.best is not None and time.monotonic() < total_deadline:
        t0 = time.monotonic()
        p2_deadline = min(total_deadline, time.monotonic() + 550.0)
        print(f"\n{'='*50} Phase 2: LP Mirror Descent n={tracker.best.size} {'='*50}", flush=True)

        # Primary run at best length
        _, _ = lp_mirror_descent(tracker.best, 8000, p2_deadline, tracker=tracker,
                                 n_line_evals=18, stall_patience=45)

        # Try other lengths
        for n_try in [512, 1024, 2048]:
            if time.monotonic() > p2_deadline:
                break
            if n_try == tracker.best.size:
                continue
            a_res = resample_sequence(tracker.best, n_try)
            a_res = a_res / max(a_res.sum(), 1e-10)
            budget_here = min(150.0, p2_deadline - time.monotonic())
            if budget_here < 20:
                continue
            max_iters = int(budget_here * 50)
            _, _ = lp_mirror_descent(a_res, max_iters, time.monotonic() + budget_here,
                                     tracker=tracker, n_line_evals=15, stall_patience=30)
            # Also run Adam on this resampled version
            if time.monotonic() < p2_deadline:
                adam_sched_short = [(1.0, 50), (5.0, 80), (20.0, 100), (100.0, 150)]
                a_adam, _ = adam_with_linesearch(a_res, min(p2_deadline, time.monotonic() + 60),
                                                  adam_sched_short, base_lr=0.08, tracker=tracker)

        tracker.report(f"Phase 2: {time.monotonic()-t0:.1f}s")

    # === Phase 3: Structured Restarts (250s) ===
    if tracker.best is not None and time.monotonic() < total_deadline:
        t0 = time.monotonic()
        p3_deadline = min(total_deadline, time.monotonic() + 250.0)
        print(f"\n{'='*50} Phase 3: Structured Restarts {'='*50}", flush=True)

        n = tracker.best.size
        perturbations = generate_structured_perturbations(tracker.best, n, rng)

        # Sort by initial J to try promising ones first
        scored_perts = []
        for pert in perturbations:
            pert = np.maximum(pert, 1e-6)
            pert = pert / pert.sum()
            J_pert = compute_J(pert)
            scored_perts.append((J_pert, pert))
        scored_perts.sort(key=lambda x: x[0])

        for J_pert, pert in scored_perts:
            if time.monotonic() > p3_deadline:
                break

            if J_pert < tracker.best_score:
                tracker.update(pert)

            # Quick LP + Adam refinement
            budget = min(25.0, p3_deadline - time.monotonic())
            if budget < 5:
                continue

            # Short LP
            max_iters = int(budget * 0.6 * 40)
            if max_iters >= 5:
                _, _ = lp_mirror_descent(pert, max_iters, time.monotonic() + budget * 0.6,
                                         tracker=tracker, n_line_evals=12, stall_patience=15)

            # Short Adam if time remains
            if time.monotonic() < p3_deadline - 5:
                adam_sched_tiny = [(1.0, 30), (10.0, 50), (50.0, 80)]
                adam_with_linesearch(pert, min(p3_deadline, time.monotonic() + 10),
                                     adam_sched_tiny, base_lr=0.05, tracker=tracker)

        tracker.report(f"Phase 3: {time.monotonic()-t0:.1f}s")

    # === Phase 4: Final Polish (120s) ===
    if tracker.best is not None and time.monotonic() < total_deadline:
        t0 = time.monotonic()
        p4_deadline = total_deadline
        remaining = p4_deadline - time.monotonic()
        print(f"\n{'='*50} Phase 4: Final Polish (budget {remaining:.0f}s) {'='*50}", flush=True)

        # Extended LP with more line search evals
        max_iters = int(remaining * 0.7 * 45)
        _, _ = lp_mirror_descent(tracker.best, max_iters,
                                  time.monotonic() + remaining * 0.7,
                                  tracker=tracker, n_line_evals=22, stall_patience=50)

        # Adam polish if time remains
        if time.monotonic() < p4_deadline - 10:
            adam_sched_final = [(10.0, 60), (50.0, 100), (200.0, 150)]
            adam_with_linesearch(tracker.best, p4_deadline,
                                  adam_sched_final, base_lr=0.03, tracker=tracker)

        # Try nearby lengths
        if time.monotonic() < p4_deadline - 20:
            n_best = tracker.best.size
            for n_try in [n_best - 32, n_best + 32, 640, 768, 1152, 1280]:
                if n_try < 128 or n_try > 4096:
                    continue
                if abs(n_try - n_best) < 16:
                    continue
                if time.monotonic() > p4_deadline - 10:
                    break
                a_res = resample_sequence(tracker.best, n_try)
                a_res = a_res / max(a_res.sum(), 1e-10)
                budget = min(15.0, p4_deadline - time.monotonic() - 5)
                if budget < 5:
                    continue
                max_iters = int(budget * 40)
                _, _ = lp_mirror_descent(a_res, max_iters, time.monotonic() + budget,
                                         tracker=tracker, n_line_evals=12, stall_patience=15)

        tracker.report(f"Phase 4: {time.monotonic()-t0:.1f}s")

    # Fallback
    if tracker.best is None:
        tracker.best = np.ones(1024)
        tracker.best_score = compute_J(tracker.best)
        print(f"  Using fallback: J={tracker.best_score:.8f}", flush=True)

    result = np.clip(tracker.best, 0.0, 1000.0)
    if result.sum() < 0.01:
        result = np.ones(128)

    final_time = time.monotonic() - tracker.t_start
    print(f"\n{'='*50} FINAL RESULT {'='*50}", flush=True)
    print(f"J={tracker.best_score:.8f} n={tracker.best.size} time={final_time:.1f}s", flush=True)

    return [float(x) for x in result]
