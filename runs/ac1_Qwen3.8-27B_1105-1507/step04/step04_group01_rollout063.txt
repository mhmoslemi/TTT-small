"""
Optimized step function sequence search using binomial coefficient initialization,
LP alternation, shape parameter sweeps, L-BFGS-B soft-max optimization, and
coordinate descent polishing. The binomial sequence C(2k,k)/4^k truncated to
length n provides a near-optimal starting point (objective ~ pi/2 for large n),
and we refine it using multiple optimization strategies.

import numpy as np
from scipy.optimize import minimize, linprog
from scipy.fft import rfft, irfft
import time
import math


def compute_conv(a):
    """Compute linear convolution a*a using FFT."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    conv = irfft(fa * fa, size)
    return conv[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    conv = compute_conv(a)
    max_b = float(np.max(conv))
    sum_a = float(np.sum(a))
    if sum_a < 0.01:
        return np.inf
    return 2.0 * n * max_b / (sum_a * sum_a)


def fft_convolve(a, b):
    """Linear convolution of a and b using FFT."""
    n, m = len(a), len(b)
    size = 1
    while size < n + m - 1:
        size <<= 1
    fa = rfft(a, size)
    fb = rfft(b, size)
    return irfft(fa * fb, size)[:n + m - 1]


def binomial_init(n):
    """Generate the binomial coefficient initialization a[k] = C(2k,k)/4^k."""
    a = np.zeros(n)
    coeff = 1.0
    for k in range(n):
        if k > 0:
            coeff = coeff * (2.0 * k) * (2.0 * k - 1.0) / (k * k)
        a[k] = coeff
    return a


def shape_modified_binomial(n, beta=0.0, mode='linear'):
    """Generate binomial sequence with shape modification."""
    a = binomial_init(n)
    k = np.arange(n, dtype=np.float64)
    if mode == 'linear':
        a = a * (1.0 + beta * k / n)
    elif mode == 'cos1':
        a = a * (1.0 + beta * np.cos(2.0 * np.pi * k / n))
    elif mode == 'cos2':
        a = a * (1.0 + beta * np.cos(4.0 * np.pi * k / n))
    elif mode == 'decay':
        a = a * (1.0 + beta * np.exp(-k / (n * 0.3)))
    elif mode == 'edge':
        x = 2.0 * k / n - 1.0
        a = a * (1.0 + beta * x * x)
    elif mode == 'sqrt':
        a = a * (1.0 + beta * np.sqrt(k / n))
    elif mode == 'quadratic':
        x = 2.0 * k / n - 1.0
        a = a * (1.0 + beta * x)
    a = np.maximum(a, 1e-10)
    return a


def softmax(z):
    """Numerically stable softmax."""
    z = z - np.max(z)
    e = np.exp(z)
    s = np.sum(e)
    return e / s


def soft_max_obj_and_grad_z(z, tau):
    """
    Compute soft-max log-sum-exp of conv(softmax(z), softmax(z)) * n / sum(softmax(z))^2
    and gradient w.r.t. z.
    """
    n = len(z)
    a = softmax(z)
    conv = compute_conv(a)
    cmax = float(np.max(conv))
    shifted = np.clip((conv - cmax) / tau, -500.0, 500.0)
    exp_conv = np.exp(shifted)
    Z = float(np.sum(exp_conv))
    soft_max_val = cmax + tau * np.log(Z)
    w = exp_conv / Z

    w_rev = w[::-1]
    conv_fw = fft_convolve(a, w_rev)
    f1 = conv_fw[n - 1:2 * n - 1][::-1]
    grad_a = 2.0 * f1

    mean_grad = float(np.dot(a, grad_a))
    grad_z = a * (grad_a - mean_grad)

    return soft_max_val, grad_z


def optimize_lbfgs_z(z_init, tau_list, max_iter=200, time_budget=20.0):

    a = np.maximum(a, 0)
    s = np.sum(a)

def build_toeplitz(a):
    """Build Toeplitz matrix where (A@g)[k] = (conv(a,g))[k]."""
    n = len(a)
    A = np.zeros((2 * n - 1, n), dtype=np.float64)
    for j in range(n):
        k_start = j
        k_end = min(j + n, 2 * n - 1)
        A[k_start:k_end, j] = a[:k_end - k_start]
    return A


def solve_lp_for_g(a, M, time_limit=30):
    """Solve LP: max sum(g) s.t. conv(a,g) <= M, g >= 0."""
    n = len(a)
    A = build_toeplitz(a)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': time_limit,
                               'dual_feasibility_tolerance': 1e-10,
                               'primal_feasibility_tolerance': 1e-10})
        if res.status == 0:
            g = np.maximum(res.x, 0)
            s = float(np.sum(g))
            if s > 1e-10:
                return g / s
    except Exception:
        pass
    return None


def lp_alternation(a, max_iters=100, time_budget=60.0, verbose=False):
    """
    LP alternation with adaptive line search.
    At each step: solve LP for g, then do line search on (1-t)*a + t*g.
    """
    n = len(a)
    a = a.copy().astype(np.float64)
    s = np.sum(a)
    if s < 1e-10:
        return a
    a = a / s

    stall = 0

        if time.monotonic() - start > time_budget:

        M = float(np.max(conv))

        g = solve_lp_for_g(a, M, time_limit=min(20, max(5, time_budget / max(1, max_iters - it))))
        if g is None:
            break

        # Adaptive line search with fine steps
        best_t_obj = best_obj
        best_a_new = best_a.copy()
        best_t = 0.0

        # Coarse search

            a_try = (1.0 - t) * a + t * g

            obj_try = compute_obj(a_try)

            best_a = best_a_new.copy()

def coordinate_descent(a, max_iters=5000, time_budget=20.0):

def resample_sequence(a, new_n):
    """Resample sequence to new length via interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 1e-10)

            if time.monotonic() > deadline - 120:

            a = shape_modified_binomial(n, beta, mode)
            s = np.sum(a)
            a = a / s
            obj = compute_obj(a)
            print(f"  Shape ({mode}, beta={beta:+.2f}): obj={obj:.8f}")

            # Quick LP polish
            remaining_init = deadline - time.monotonic()
            if remaining_init > 30:
                lp_budget = min(alloc / len(shape_configs) * 1.5, 25.0)
                a = lp_alternation(a, max_iters=int(lp_budget * 2),
                                   time_budget=lp_budget)
                obj = compute_obj(a)
                print(f"    After LP: obj={obj:.8f}")

            if obj < best_score:
                best_score = obj
                best_seq = a.copy()
                print(f"  ** UPDATE: n={n}, obj={best_score:.8f} **")

            candidates.append((a.copy(), obj, n))

    # Phase 2: Polish top candidates
    print(f"\n=== POLISHING PHASE ===")
    print(f"Best before polish: {best_score:.8f}")

    candidates.sort(key=lambda x: x[1])
    n_polish = min(5, len(candidates))

    for ci in range(n_polish):
        if time.monotonic() > deadline - 60:
            break
        remaining = deadline - time.monotonic()
        if remaining < 30:
            break

        a_cand = candidates[ci][0]
        n_cand = candidates[ci][2]
        obj_cand = candidates[ci][1]

        if obj_cand > best_score * 1.01:
            break  # Skip candidates significantly worse than best

        print(f"  Polishing candidate {ci} (n={n_cand}, obj={obj_cand:.8f}), {remaining:.0f}s left")

        polish_budget = min(remaining * 0.3, 60.0)
        a_pol, obj_pol = polish_sequence(a_cand, time_budget=polish_budget, verbose=True)

        if obj_pol < best_score:
            best_score = obj_pol
            best_seq = a_pol.copy()
            print(f"  ** UPDATE after polish: obj={best_score:.8f} **")

        # Try perturbation from this candidate
        remaining = deadline - time.monotonic()
        if remaining > 20:
            rng = np.random.default_rng(ci * 31337 + 999)
            for ptry in range(3):
                if time.monotonic() > deadline - 40:
                    break
                scale = rng.uniform(0.01, 0.05)
                a_pert = a_cand * np.exp(rng.normal(0, scale, n_cand))
                a_pert = np.maximum(a_pert, 0)
                s = np.sum(a_pert)
                if s < 1e-10:
                    continue
                a_pert = a_pert / s
                # Quick polish
                a_pert = coordinate_descent(a_pert, max_iters=2000, time_budget=8.0)
                obj_pert = compute_obj(a_pert)
                if obj_pert < best_score:
                    best_score = obj_pert
                    best_seq = a_pert.copy()
                    print(f"  ** UPDATE (perturb {ptry}): obj={best_score:.8f} **")

    # Phase 3: Final intensive polish on global best
    if best_seq is not None and time.monotonic() < deadline - 15:
        remaining = deadline - time.monotonic()
        n_best = len(best_seq)
        print(f"\n=== FINAL POLISH: n={n_best}, {remaining:.0f}s ===")

        a_cur = best_seq.copy()
        for final_round in range(6):
            if time.monotonic() > deadline - 10:
                break
            remaining = deadline - time.monotonic()
            if remaining < 10:
                break

            # LP
            if n_best <= 800:
                a_new = lp_alternation(a_cur, max_iters=30, time_budget=min(remaining * 0.25, 20.0))
                obj_new = compute_obj(a_new)
                if obj_new < best_score:
                    best_score = obj_new
                    best_seq = a_new.copy()
                    a_cur = a_new.copy()
                    print(f"  Final LP {final_round}: {best_score:.8f}")

            # L-BFGS-B
            remaining = deadline - time.monotonic()
            if remaining > 10:
                a_norm = a_cur / np.sum(a_cur)
                a_norm = np.maximum(a_norm, 1e-20)
                z = np.log(a_norm)
                tau = 0.2 / n_best
                a_new = optimize_lbfgs_z(z, [tau, tau * 0.5, tau * 0.1],
                                         max_iter=200, time_budget=min(remaining * 0.2, 12.0))
                obj_new = compute_obj(a_new)
                if obj_new < best_score:
                    best_score = obj_new
                    best_seq = a_new.copy()
                    a_cur = a_new.copy()
                    print(f"  Final L-BFGS {final_round}: {best_score:.8f}")

            # Coordinate descent (final micro-optimization)
            remaining = deadline - time.monotonic()
            if remaining > 8:
                a_new = coordinate_descent(a_cur, max_iters=int(remaining * 30),
                                           time_budget=min(remaining * 0.3, 20.0))
                obj_new = compute_obj(a_new)
                if obj_new < best_score:
                    best_score = obj_new
                    best_seq = a_new.copy()
                    a_cur = a_new.copy()
                    print(f"  Final CD {final_round}: {best_score:.8f}")

            # Subgradient with large tau (exploration)
            remaining = deadline - time.monotonic()
            if remaining > 8:
                a_new = subgradient_optimize(a_cur, max_iters=int(remaining * 5),
                                             time_budget=min(remaining * 0.15, 10.0),
                                             tau=20.0)
                obj_new = compute_obj(a_new)
                if obj_new < best_score:
                    best_score = obj_new
                    best_seq = a_new.copy()
                    a_cur = a_new.copy()
                    print(f"  Final SG {final_round}: {best_score:.8f}")

    # Final validation
    if best_seq is None:
        best_seq = binomial_init(600)
        best_score = compute_obj(best_seq)

    # Clamp values
    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    val = evaluate_sequence(result)
    if val < np.inf:
        best_score = val
    else:
        result = [1.0] * 100
        best_score = 2.0

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_score:.8f}, time={elapsed:.1f}s ===")

    return result
