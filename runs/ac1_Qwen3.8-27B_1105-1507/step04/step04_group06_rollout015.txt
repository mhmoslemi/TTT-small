def coord_descent_vectorized(a, max_sweeps, deadline, tracker=None,
                             n_deltas=30, recompute_every=10, seed=12345,
                             double_move_prob=0.05):

    Fast vectorized coordinate descent with batch line search.
    Maintains convolution incrementally, recomputes periodically.
    """
    n = len(a)
    a = a.copy().astype(np.float64)
    a = np.maximum(a, 1e-10)
    a = a / a.sum()

    rng = np.random.default_rng(seed)

        tracker.update(a)

    shift_lo = np.clip(np.arange(n), 0, 2 * n - 1)
    shift_hi = np.clip(np.arange(n) + n, 0, 2 * n - 1)

    sweep_no_improve = 0

    for sweep in range(max_sweeps):
        if time.monotonic() > deadline:
            break

        # Periodically recompute convolution from scratch (numerical stability)
        if sweep % recompute_every == 0:
            q = fast_conv(a)
        else:
            q = q  # keep incremental

        S = a.sum()
        q_max = q.max()
        current_J = 2.0 * n * q_max / (S * S)

        if current_J < best_J - 1e-14:
            best_J = current_J
            best_a = a.copy()
            if tracker:
                tracker.update(a)

        sweep_improved = False
        order = rng.permutation(n)

        for idx, i in enumerate(order):
            if time.monotonic() > deadline:
                break

            # Build direction vector d where d[k] = a[k-i] (zero-padded)
            d = np.zeros(2 * n - 1)
            lo = shift_lo[i]
            hi = shift_hi[i]
            d[lo:hi] = a[:hi - lo]

            a_i = a[i]
            S = a.sum()
            if S < 1e-10:
                break

            # Build delta candidates (vectorized)
            # Range: from -a_i (to zero) up to a reasonable positive
            scale = max(a_i, S / n * 2)
            neg_deltas = np.linspace(-a_i * 0.99, 0, n_deltas // 3 + 1)
            pos_deltas = np.linspace(0, scale, n_deltas - n_deltas // 3)
            deltas = np.concatenate([neg_deltas[1:], pos_deltas[1:]])

            # Vectorized evaluation
            # new_q[挑, k] = q[k] + 2*delta*d[k] + delta^2 * e_{2i}[k]
            deltas_col = deltas[:, np.newaxis]  # (m, 1)
            new_q_all = q[np.newaxis, :] + 2.0 * deltas_col * d[np.newaxis, :]  # (m, 2n-1)
            new_q_all[:, 2 * i] += deltas_col[:, 0] ** 2

            new_S = S + deltas
            valid = new_S > 0.01
            if not np.any(valid):
                continue

            new_J_all = np.full(len(deltas), current_J)
            new_J_all[valid] = 2.0 * n * new_q_all[valid].max(axis=1) / (new_S[valid] ** 2)

            best_idx = np.argmin(new_J_all)
            best_delta = deltas[best_idx]
            best_J_try = new_J_all[best_idx]

            if best_J_try < current_J - 1e-14:
                # Apply update
                a[i] += best_delta
                a[i] = max(a[i], 0.0)
                q += 2.0 * best_delta * d
                q[2 * i] += best_delta * best_delta
                current_J = best_J_try
                sweep_improved = True

                # Optional double move: try improving a neighboring coordinate
                if double_move_prob > 0 and rng.random() < double_move_prob:
                    j = (i + 1) % n
                    if j != i:
                        d2 = np.zeros(2 * n - 1)
                        lo2 = shift_lo[j]
                        hi2 = shift_hi[j]
                        d2[lo2:hi2] = a[:hi2 - lo2]
                        a_j = a[j]
                        S2 = a.sum()
                        if S2 > 0.01:

                                    a[j] = max(a[j], 0.0)

        jl = max(0, k - n + 1)
        jh = min(n - 1, k)
        for j in range(jl, jh + 1):
            rows.append(k)
            cols.append(j)
    return csr_matrix((np.ones(len(rows)), (rows, cols)), shape=(2 * n - 1, n))


def lp_best_response(a, M):
    """LP: max sum(g) s.t. A(a)*g <= M, g >= 0. Returns g normalized or None."""
    n = len(a)
    c = -np.ones(n)
    # A[i,j] = a[i-j] if valid, 0 otherwise (for (a*g)_i <= M)
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        jl = max(0, k - n + 1)
        jh = min(n - 1, k)
        for j in range(jl, jh + 1):
            rows.append(k)
            cols.append(j)
            vals.append(a[k - j])
    A = csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    try:
        r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                    options={'time_limit': 30})
        if r.success and r.x is not None:
            g = np.maximum(r.x, 0.0)
            gs = g.sum()
            if gs > 1e-10:
                return g / gs
    except Exception:
        pass
    return None


def lp_polish(a, deadline, tracker=None, max_iters=500):
    """LP mirror descent polish."""
    n = len(a)
    a = a.copy().astype(np.float64)
    a = np.maximum(a, 1e-10)
    a = a / a.sum()
    best_J = compute_J(a)
    best_a = a.copy()

    if tracker:
        tracker.update(a)

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        b = fast_conv(a)
        M = float(b.max())

        g = lp_best_response(a, M)
        if g is None:
            break

        # Line search
        best_t = 0.0
        best_J_ls = best_J
        for t in [0.01, 0.03, 0.05, 0.1, 0.2, 0.3, 0.5]:
            a_try = (1 - t) * a + t * g
            a_try = np.maximum(a_try, 0)
            s = a_try.sum()
            if s < 1e-10:
                continue
            a_try = a_try / s
            J_try = compute_J(a_try)
            if J_try < best_J_ls:
                best_J_ls = J_try
                best_t = t

        if best_t > 0:
            a = (1 - best_t) * a + best_t * g
            a = np.maximum(a, 0)
            a = a / a.sum()
            if best_J_ls < best_J:
                best_J = best_J_ls
                best_a = a.copy()
                if tracker:
                    tracker.update(a)

    if tracker:
        tracker.update(best_a)
    return best_a


# ===================== SMOOTH L-BFGS INIT =====================

def smooth_conv_max_and_grad(a, beta):
    """Smooth max of convolution and its gradient."""
    n = len(a)
    b = fast_conv(a)
    b_max = float(b.max())
    shifted = beta * (b - b_max)
    shifted = np.clip(shifted, -500, 500)
    exp_s = np.exp(shifted)
    Z = float(exp_s.sum())
    if Z < 1e-300:
        p = np.zeros(2 * n - 1)
        p[np.argmax(b)] = 1.0
    else:
        p = exp_s / Z
    g_val = (b_max + np.log(max(Z, 1e-300))) / beta

    # Gradient: ∂/∂a_i of b_k = a_{k-i} + a_{k-i} (symmetric)
    # grad[k] for b contributes to a via: grad_a[i] = sum_k p[k] * (a_{k-i} + a_{k-i})
    # = 2 * sum_k p[k] * a_{k-i} = 2 * (p * a)[reversed]
    S = np.convolve(p, a, mode='full')
    # S[k] = sum_j p[j] * a[k-j]
    # We want: for each i, sum_k p[k] * a[k-i]
    # This is S_full[i] where S_full = convolve(p_rev, a) ... let me think
    # sum_k p[k] * a[k-i] = (p * a)[i] in the "cross-correlation" sense
    # = sum_k p[k] * a[k-i]
    # If we define c[m] = sum_k p[k] * a[k-m], then c is the cross-correlation
    # c[m] = sum_k p[k] * a[k-m] = (p ~~ a)[m] where ~~ is cross-correlation
    # In numpy: np.correlate(a, p, mode='full') gives sum_k a[k] * p[k+m]
    # What we want is sum_k p[k] * a[k-i] for each i
    # = sum_k p[k] * a[k-i]
    # Let j = k-i: sum_j p[j+i] * a[j] = (p shifted by i) dot a
    # = correlate(p, a) at offset... 
    # Actually: if we compute cross_corr = np.correlate(a, p, 'full')
    # cross_corr[k] = sum_j a[j] * p[j+k-n+1] (for the 'full' mode)
    # This is getting confusing. Let me use FFT.

    # grad_a[i] = 2 * sum_k p[k] * a[k-i] (with a outside [0,n-1] = 0)
    # This is 2 * (p cross-correlated with a)[i]
    # Using FFT: if P = FFT(p, 2n), A = FFT(a, 2n), then
    # cross_corr = IFFT(conj(P) * A) gives sum_k p[k+j] * a[j]... 
    # Let me just do it directly for clarity (n is at most ~2048)

    grad = np.zeros(n)
    for i in range(n):
        # sum_k p[k] * a[k-i] where 0<=k<2n-1 and 0<=k-i<n
        k_lo = max(0, i)
        k_hi = min(2 * n - 1, i + n)
        if k_lo < k_hi:
            grad[i] = 2.0 * np.dot(p[k_lo:k_hi], a[k_lo - i:k_hi - i])

    return g_val, grad


def F_and_grad_smooth(a, beta):
    """Smoothed objective and gradient."""
    n = len(a)
    s = float(a.sum())
    if s < 1e-10:
        return 1e10, np.zeros(n)
    g_val, grad_g = smooth_conv_max_and_grad(a, beta)
    F_val = 2.0 * n * g_val / (s * s)
    grad_F = 2.0 * n * (s * grad_g - 2.0 * g_val) / (s ** 3)
    return F_val, grad_F


def lbfgs_init(a0, deadline, betas=(5.0, 20.0, 50.0)):
    """L-BFGS-B with smooth objective for initialization."""
    a = np.maximum(a0.astype(np.float64), 1e-8)
    n = len(a)

    for beta in betas:
        if time.monotonic() > deadline:
            break

        def func(x):
            x2 = np.maximum(x, 0.0)
            f, _ = F_and_grad_smooth(x2, beta)
            return f if np.isfinite(f) else 1e10

        def jac(x):
            x2 = np.maximum(x, 0.0)
            _, g = F_and_grad_smooth(x2, beta)
            return np.nan_to_num(g, nan=0.0, posinf=0.0, neginf=0.0)

        bounds = [(0.0, None)] * n
        try:
            res = minimize(func, a.copy(), jac=jac, method='L-BFGS-B',
                           bounds=bounds,
                           options={'maxiter': 100, 'max_fun': 2000,
                                    'gtol': 1e-6, 'ftol': 1e-10})
            if res.fun < 1e10:
                new_a = np.maximum(res.x, 1e-10)
                if new_a.sum() > 1e-6:
                    a = new_a / new_a.sum()
        except Exception:
            pass

    return a


# ===================== RESAMPLING =====================

def resample_seq(src, dest_n):
    """Resample 1D sequence to new length via linear interpolation."""
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    result = np.interp(t_new, t_old, src)
    return np.maximum(result, 0.0)

def propose_candidate(seed=42, budget_s=1500, **kwargs):
    """
    Multi-phase optimization:
    Phase 1: L-BFGS-B smooth init at multiple lengths (60s)
    Phase 2: Vectorized coordinate descent (main, ~1100s)
    Phase 3: Perturbation restarts + CD (200s)
    Phase 4: LP polish + length exploration (remaining)
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + float(budget_s) - 15.0
    tracker = ScoreTracker()

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        if p.ndim == 1 and p.size > 0 and np.all(np.isfinite(p)):
            parent = np.maximum(p, 0.0)
            tracker.update(parent)
            print(f"Parent loaded: n={parent.size}, J={tracker.best_score:.8f}", flush=True)
    except (NameError, AttributeError):
        pass

    if parent is None:
        parent = np.ones(512)
        tracker.update(parent)
        print(f"Using default parent: n=512, J={tracker.best_score:.8f}", flush=True)

    # ============ PHASE 1: L-BFGS-B initialization (60s) ============
    phase1_deadline = time.monotonic() + 60.0
    print(f"\n{'=' * 50}\nPhase 1: L-BFGS-B Init\n{'=' * 50}", flush=True)

    n_target = parent.size if parent.size <= 2048 else 1024
    a_init = resample_seq(parent, n_target)
    a_init = a_init / max(a_init.sum(), 1e-10)

    # Try a few structured initializations
    t_arr = np.linspace(0, 1, n_target, endpoint=False)
    inits = [a_init, np.ones(n_target) / n_target]
    for freq in [2, 3, 5]:
        inits.append(np.maximum(1.0 + 0.3 * np.cos(2 * np.pi * freq * t_arr), 0.01))
    inits.append(np.exp(-3.0 * (t_arr - 0.5) ** 2) + 0.05)
    inits.append(np.abs(np.sin(np.pi * t_arr)) + 0.1)

    best_init = None
    best_init_J = float('inf')
    for a0 in inits:
        if time.monotonic() > phase1_deadline:
            break
        a0 = np.maximum(a0, 1e-4)
        a0 = a0 / a0.sum()
        a_opt = lbfgs_init(a0, phase1_deadline, betas=(10.0, 30.0))
        J = compute_J(a_opt)
        if J < best_init_J:
            best_init_J = J
            best_init = a_opt.copy()
        tracker.update(a_opt)

    if best_init is None:
        best_init = a_init
    print(f"  Phase 1 done: best J={tracker.best_score:.8f}", flush=True)

    # ============ PHASE 2: Coordinate Descent (main, ~1100s) ============
    phase2_deadline = time.monotonic() + 1100.0
    n_cd = tracker.best.size
    print(f"\n{'=' * 50}\nPhase 2: Coordinate Descent n={n_cd}\n{'=' * 50}", flush=True)

    a_cd = tracker.best.copy()
    a_cd = a_cd / a_cd.sum()
    max_sweeps_p2 = 50000
    a_cd = coord_descent_vectorized(a_cd, max_sweeps_p2, phase2_deadline,
                                    tracker=tracker, n_deltas=30,
                                    recompute_every=10, seed=999,
                                    double_move_prob=0.05)
    tracker.report("Phase 2 complete")

    # ============ PHASE 3: Perturbation Restarts (200s) ============
    phase3_deadline = min(time.monotonic() + 200.0, deadline)
    if tracker.best is not None and time.monotonic() < phase3_deadline:
        print(f"\n{'=' * 50}\nPhase 3: Perturbation Restarts\n{'=' * 50}", flush=True)

        strategies = [
            (0.03, 0.0), (0.05, 0.0), (0.08, 0.0), (0.12, 0.0),
            (0.0, 5), (0.0, 10), (0.15, 0.0), (0.2, 0.0),
        ]

        attempt = 0
        for sigma, spike_k in strategies:
            for trial in range(4):
                if time.monotonic() > phase3_deadline:
                    break
                attempt += 1
                a = tracker.best.copy().astype(np.float64)
                nn = len(a)

                if sigma > 0:
                    a = a + rng.normal(0, sigma, nn) * a
                if spike_k > 0:
                    kk = min(spike_k, nn // 4)
                    idx = rng.choice(nn, size=kk, replace=False)
                    a[idx] *= rng.uniform(0.2, 4.0, kk)

                a = np.maximum(a, 1e-6)
                a = a / a.sum()

                # Quick CD refinement
                budget_here = min(30.0, phase3_deadline - time.monotonic())
                max_iters_here = int(budget_here * 500)
                a = coord_descent_vectorized(a, max_iters_here, phase3_deadline,
                                             tracker=tracker, n_deltas=25,
                                             recompute_every=10,
                                             seed=1000 + attempt,
                                             double_move_prob=0.03)

        tracker.report(f"Phase 3: {attempt} attempts")

    # ============ PHASE 4: LP Polish + Length Exploration ============
    if tracker.best is not None and time.monotonic() < deadline:
        remaining = deadline - time.monotonic()
        if remaining > 30:
            print(f"\n{'=' * 50}\nPhase 4: LP Polish + Length Exploration "
                  f"({remaining:.0f}s)\n{'=' * 50}", flush=True)

            # LP polish on best
            lp_deadline = time.monotonic() + min(120.0, remaining * 0.5)
            a_lp = lp_polish(tracker.best, lp_deadline, tracker=tracker, max_iters=300)
            tracker.report("LP polish")

            # Length exploration
            if time.monotonic() < deadline:
                n_best = tracker.best.size
                for n_try in [n_best - 64, n_best + 64, 768, 1280, 1536]:
                    if n_try < 128 or n_try > 3000 or time.monotonic() > deadline:
                        continue
                    if abs(n_try - n_best) < 32:
                        continue
                    a_res = resample_seq(tracker.best, n_try)
                    a_res = a_res / max(a_res.sum(), 1e-10)
                    budget_here = min(40.0, deadline - time.monotonic())
                    max_iters_here = int(budget_here * 400)
                    a_res = coord_descent_vectorized(a_res, max_iters_here,
                                                     deadline, tracker=tracker,
                                                     n_deltas=25, recompute_every=10,
                                                     seed=7777, double_move_prob=0.03)

                tracker.report("Length exploration")

    # Fallback
    if tracker.best is None:
        tracker.best = np.ones(512)
        tracker.best_score = compute_J(tracker.best)
        print(f"  Using fallback: J={tracker.best_score:.8f}", flush=True)

    print(f"\n{'=' * 50}\nFINAL RESULT\n{'=' * 50}", flush=True)
    print(f"J={tracker.best_score:.8f} n={tracker.best.size} "
          f"time={time.monotonic() - tracker.t_start:.1f}s", flush=True)

    result = np.clip(tracker.best, 0.0, 1000.0)
    if result.sum() < 0.01:
        result = np.ones(128)

    return [float(x) for x in result]
