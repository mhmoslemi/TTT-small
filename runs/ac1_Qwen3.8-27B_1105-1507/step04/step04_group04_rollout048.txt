We can write this as: min t subject to B[k] <= t for all k, sum(a) = 1, a >= 0.

- sum_k lambda_k * dB[k]/da[j] - mu - nu_j = 0
- For a[j] > 0: sum_k lambda_k * dB[k]/da[j] = mu

The lambda_k values form a probability distribution over the lags, and the condition says that the "expected pressure" (weighted by lambda) is the same for all active coordinates.

1. Start with some lambda distribution over lags
2. Compute the "pressure" at each coordinate: p[j] = sum_k lambda_k * dB[k]/da[j]
3. Update a to equalize the pressures (like water filling)
4. Update lambda based on which constraints are tight (B[k] close to max)

This is essentially the dual ascent method, and it's closely related to the multiplicative weights approach.

For step 3, "equalize pressures" means: if pressure at j is high, reduce a[j]; if low, increase a[j]. In the multiplicative update: a[j] *= exp(-eta * (p[j] - avg_p)).

import numpy as np
from scipy.optimize import linprog, minimize
import time


def compute_conv_fft(a):
    """Compute full convolution a*a using FFT."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    conv = np.fft.irfft(fa * fa, size)
    return conv[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = np.sum(a)
    if s < 1e-10:
        return np.inf
    B = compute_conv_fft(a)
    max_b = float(np.max(B))
    return 2.0 * n * max_b / (s * s)


def smooth_obj_and_grad(a, t):

    Compute smooth approximation to 2n*max(conv(a_norm,a_norm)) and gradient.

        smooth_max = max_B

    # g[j] = sum_k p[k] * d(B[k])/d(a_norm[j])

    # For full convolution: B[k] = sum_j a[j]*a[k-j]
    # d(B[k])/d(a[i]) = a[k-i] (if 0<=k-i<n) + a[i-k] (if 0<=i-k<n, i.e. i>k)
    # But since B is symmetric (B[k] = B[2n-2-k]), and we sum over all k with p[k],
    # the gradient simplifies to: g[j] = sum_k p[k] * a[k-j] (with a[m]=0 for m out of [0,n-1])

    A = np.fft.rfft(a_norm, size)
    # Cross-correlation via FFT: corr[k] = sum_n p[n+k]*a[n]
    # = ifft(conj(FFT(p)) * FFT(a)) ... let me verify
    # Actually: conv(p, a)[k] = sum_j p[j]*a[k-j]
    # correlate(p, a)[k] = sum_j p[j+k]*a[j] = conv(p, flip(a))[k + len(a) - 1]
    # Simpler: g[j] = sum_m p[j+m]*a[m] for m=0..n-1
    # This is the cross-correlation of p and a at lag j
    # numpy.correlate(p, a, 'full')[j] = sum_n p[n+j]*a[n] (for valid n)
    # Since len(p) = 2n-1 and len(a) = n, this gives length 3n-2
    # g[j] = correlate(p, a_norm, 'full')[j] for j=0..n-1
    
    # Use FFT for O(n log n) instead of O(n^2):
    # correlate(p, a) = ifft(FFT(p) * conj(FFT(a))) ... let me verify
    # Actually the standard relation is:

    # So correlate(p, a_norm) = ifft(conj(FFT(p)) * FFT(a_norm))

    g = np.correlate(p, a_norm, mode='full')[:n]
    
    obj = 2.0 * n * smooth_max

def multiplicative_weights_step(a, tau, step, top_k=5):
    """
    Multiplicative weights update. Focuses on the top-k lags of the convolution.
    """
    n = len(a)
    B = compute_conv_fft(a)
    max_B = float(np.max(B))
    
    # Use top-k lags for the subgradient
    if top_k >= len(B):
        indices = np.arange(len(B))
    else:
        indices = np.argsort(B)[-top_k:]
    
    # Compute gradient from selected lags: g[j] = sum_{k in indices} p[k] * a[k-j]
    # Use uniform weights over selected lags
    p_sel = np.zeros(len(B))
    p_sel[indices] = 1.0 / len(indices)
    
    g = np.correlate(p_sel, a, mode='full')[:n]
    
    # Center gradient to preserve sum
    g_centered = g - np.mean(g)
    
    # Multiplicative update
    log_update = -step * g_centered
    log_update = np.clip(log_update, -5, 5)
    a_new = a * np.exp(log_update)
    a_new = np.maximum(a_new, 1e-15)
    s_new = np.sum(a_new)
    if s_new > 1e-10:
        a_new = a_new / s_new
    return a_new


def build_toeplitz_conv(a):
    """Build matrix A where (A @ g)[k] = (a*g)[k] for k=0..2n-2."""
    n = len(a)
    A = np.zeros((2 * n - 1, n), dtype=np.float64)
    for j in range(n):
        k_start = j
        k_end = min(j + n, 2 * n - 1)
        A[k_start:k_end, j] = a[:k_end - k_start]
    return A


def solve_lp_maxsum(a, M, time_limit=5):
    """Solve LP: max sum(g) s.t. (a*g)[k] <= M for all k, g >= 0."""
    n = len(a)
    A = build_toeplitz_conv(a)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': time_limit})
        if res.status in [0, 3]:
            g = np.maximum(res.x, 0)
            return g, float(np.sum(g))
    except Exception:
        pass
    return None, None


def lp_alternation(a, max_iters=100, time_budget=60.0):

        if g is None or sum_g < 1e-10:
            break

        g_norm = g / sum_g

        # Line search for blending
        best_t = 0.0

def resample_sequence(a, new_n):
    """Resample a sequence to a new length."""
    n = len(a)
    if n == new_n:
        return a.copy()
    old_k = np.arange(n)
    new_k = np.linspace(0, n - 1, new_n)
    a_new = np.interp(new_k, old_k, a)
    a_new = np.maximum(a_new, 0)
    s = np.sum(a_new)
    if s > 1e-10:
        a_new = a_new / s
    return a_new

    n = len(a)
    a = a.copy().astype(np.float64)
    s = np.sum(a)
    if s < 1e-10:
        a = np.ones(n) / n
    else:
        a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    # Phase 1: Multiplicative weights (quick, good for rough optimization)
    remaining = time_budget - (time.monotonic() - start)
    if remaining > 10:
        t1 = min(15.0, remaining * 0.15)
        a_new = multiplicative_optimize(best_a, time_budget=t1, tau=15.0, step=0.06, top_k=5)
        obj = compute_obj(a_new)
        if obj < best_obj:
            best_obj = obj
            best_a = a_new.copy()

    # Phase 2: L-BFGS-B on smooth objective (main workhorse for polishing)
    remaining = time_budget - (time.monotonic() - start)
    if remaining > 20:
        t2 = min(60.0, remaining * 0.4)
        a_new = l_bfgsb_smooth(best_a, time_budget=t2, t_init=15.0, t_max=150.0)
        obj = compute_obj(a_new)
        if obj < best_obj:
            best_obj = obj
            best_a = a_new.copy()

    # Phase 3: LP alternation
    remaining = time_budget - (time.monotonic() - start)
    if remaining > 15:
        t3 = min(30.0, remaining * 0.25)
        a_new = lp_alternation(best_a, max_iters=60, time_budget=t3)
        obj = compute_obj(a_new)
        if obj < best_obj:
            best_obj = obj
            best_a = a_new.copy()

    # Phase 4: More L-BFGS-B with higher temperature
    remaining = time_budget - (time.monotonic() - start)
    if remaining > 10:
        t4 = min(30.0, remaining * 0.3)
        a_new = l_bfgsb_smooth(best_a, time_budget=t4, t_init=50.0, t_max=300.0)
        obj = compute_obj(a_new)
        if obj < best_obj:
            best_obj = obj
            best_a = a_new.copy()

    # Phase 5: Local search
    remaining = time_budget - (time.monotonic() - start)
    if remaining > 8:
        t5 = min(20.0, remaining * 0.4)
        a_new = local_search(best_a, time_budget=t5)
        obj = compute_obj(a_new)
        if obj < best_obj:
            best_obj = obj
            best_a = a_new.copy()

    return best_a, best_obj


def propose_candidate(seed=42, budget_s=1500):
    """
    Main search function. Uses smooth-objective L-BFGS-B, multiplicative weights,
    LP alternation, and intensive local search to minimize the step function objective.
    """
    start_time = time.monotonic()
    deadline = start_time + float(budget_s) - 20.0

    best_seq = None
    best_score = np.inf

    # Load parent
    parent, parent_obj = try_parent()
    if parent is not None and parent_obj < best_score:
        best_score = parent_obj
        best_seq = parent.copy()
        n_parent = len(parent)
    else:
        n_parent = 300
        best_seq = np.ones(n_parent) / n_parent
        best_score = compute_obj(best_seq)

    print(f"Initial best: {best_score:.8f} (n={len(best_seq)})")
    print(f"Total time budget: {budget_s}s")

    # Phase 1: Optimize the parent sequence (most promising starting point)
    print("\n=== Phase 1: Optimize parent ===")
    remaining = deadline - time.monotonic()
    p1_budget = min(remaining * 0.30, 350.0)
    a_opt, obj = optimize_sequence(best_seq, time_budget=p1_budget)
    if obj < best_score:
        best_score = obj
        best_seq = a_opt.copy()
        print(f"  After Phase 1: {best_score:.8f} (n={len(best_seq)})")

    # Phase 2: Local search on best
    print("\n=== Phase 2: Local search ===")
    remaining = deadline - time.monotonic()
    if remaining > 20:
        p2_budget = min(remaining * 0.12, 100.0)
        a_opt = local_search(best_seq, time_budget=p2_budget)
        obj = compute_obj(a_opt)
        if obj < best_score:
            best_score = obj
            best_seq = a_opt.copy()
            print(f"  After Phase 2: {best_score:.8f}")

    # Phase 3: Explore nearby n values
    print("\n=== Phase 3: Nearby n values ===")
    n_values = []
    n_best = len(best_seq)
    for delta in [-5, 5, -10, 10, -15, 15, -20, 20, -30, 30, -40, 40, -50, 50]:
        n_val = n_best + delta
        if 150 <= n_val <= 600 and n_val != n_best:
            n_values.append(n_val)
    n_values = sorted(set(n_values))

    remaining = deadline - time.monotonic()
    if remaining > 30 and n_values:
        per_n = remaining / max(len(n_values), 1) * 0.8
        for n_val in n_values:
            if time.monotonic() > deadline - 30:
                break
            remaining = deadline - time.monotonic()
            if remaining < 15:
                break
            alloc = max(10.0, min(per_n, 60.0))
            a_res = resample_sequence(best_seq, n_val)
            a_opt, obj = optimize_sequence(a_res, time_budget=alloc)
            if obj < best_score:
                best_score = obj
                best_seq = a_opt.copy()
                print(f"  ** n={n_val}: {best_score:.8f} **")

    # Phase 4: Multiple random initializations at best n
    print("\n=== Phase 4: Random inits at best n ===")
    n_best = len(best_seq)
    remaining = deadline - time.monotonic()
    if remaining > 30:
        n_inits = max(2, min(5, int(remaining / 40)))
        for idx in range(n_inits):
            if time.monotonic() > deadline - 20:
                break
            remaining = deadline - time.monotonic()
            if remaining < 20:
                break
            a_init = random_init(n_best, seed + 100000 + idx * 79193)
            alloc = min(remaining * 0.3, 60.0)
            a_opt, obj = optimize_sequence(a_init, time_budget=alloc)
            if obj < best_score:
                best_score = obj
                best_seq = a_opt.copy()
                print(f"  ** Init {idx}: {best_score:.8f} **")

    # Phase 5: Final intensive optimization
    print("\n=== Phase 5: Final optimization ===")
    remaining = deadline - time.monotonic()
    if remaining > 30:
        p5_budget = min(remaining * 0.5, remaining - 10.0)
        a_opt, obj = optimize_sequence(best_seq, time_budget=p5_budget)
        if obj < best_score:
            best_score = obj
            best_seq = a_opt.copy()
            print(f"  After Phase 5: {best_score:.8f}")

    # Phase 6: Final local search
    remaining = deadline - time.monotonic()
    if remaining > 10:
        p6_budget = min(remaining * 0.6, remaining - 5.0)
        a_opt = local_search(best_seq, time_budget=p6_budget)
        obj = compute_obj(a_opt)
        if obj < best_score:
            best_score = obj
            best_seq = a_opt.copy()
            print(f"  After Phase 6: {best_score:.8f}")

    # Final: One more L-BFGS-B polish with very high temperature
    remaining = deadline - time.monotonic()
    if remaining > 10:
        a_opt = l_bfgsb_smooth(best_seq, time_budget=min(remaining - 5, 30.0),
                               t_init=100.0, t_max=500.0)
        obj = compute_obj(a_opt)
        if obj < best_score:
            best_score = obj
            best_seq = a_opt.copy()
            print(f"  Final L-BFGS-B: {best_score:.8f}")

    # Validation
    if best_seq is None:
        best_seq = np.ones(50)
        best_score = compute_obj(best_seq)

    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    val = evaluate_sequence(result)
    if val < best_score:
        best_score = val

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(best_seq)}, score={best_score:.8f}, time={elapsed:.1f}s ===")

    return result
