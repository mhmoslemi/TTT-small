        obj = compute_obj(a_new)
        if obj < best_obj - 1e-13:
            best_obj = obj
            best_a = a_new.copy()
            a = a_new
        else:
            eta *= 0.92
            if eta < 1e-10:
                break

    return best_a


def lp_refine(a, max_iters=100, time_budget=60.0, m_factor=1.0):
    """LP-based refinement: alternate LP solves with line search."""
    n = len(a)
    a = a.copy().astype(np.float64)
    S = np.sum(a)
    if S < 1e-10:
        return a
    a = a / S

    best_a = a.copy()
    best_obj = compute_obj(a)
    conv = compute_conv(a)
    M = float(np.max(conv))
    start = time.monotonic()
    stall = 0

    for it in range(max_iters):
        if time.monotonic() - start > time_budget:
            break
        remaining = time_budget - (time.monotonic() - start)
        if remaining < 3:
            break

        # Solve LP with current M (or slightly adjusted)
        M_try = M * m_factor
        g = solve_lp_for_g(a, M_try, time_limit=min(6.0, remaining * 0.3))
        if g is None:
            break

        S_g = np.sum(g)
        if S_g < 1e-10:
            break
        g = g / S_g

        # Line search over blend parameter t
        best_t = 0.0
        best_obj_ls = best_obj
        for t in [0.1, 0.2, 0.3, 0.5, 0.7, 1.0]:
            a_try = (1.0 - t) * a + t * g
            a_try = np.maximum(a_try, 0)
            S_try = np.sum(a_try)
            if S_try < 1e-10:
                continue
            a_try = a_try / S_try
            obj_try = compute_obj(a_try)
            if obj_try < best_obj_ls:
                best_obj_ls = obj_try
                best_t = t

        if best_t > 0.0:
            a = (1.0 - best_t) * a + best_t * g
            a = np.maximum(a, 0)
            a = a / np.sum(a)
            best_obj = best_obj_ls
            best_a = a.copy()
            conv = compute_conv(a)
            M = float(np.max(conv))
            stall = 0
        else:
            stall += 1
            if stall > 10:
                # Try with relaxed M
                m_factor = 1.0 + 0.02 * (stall - 10)
            if stall > 25:
                break

    return best_a


def fourier_perturb(a, seed=0, n_modes=10, scale=0.05):
    """Structured perturbation using Fourier modes."""
    n = len(a)
    rng = np.random.default_rng(seed)
    k = np.arange(n)
    perturbation = np.zeros(n)
    for m in range(1, n_modes + 1):
        amp = rng.uniform(-scale, scale) / m
        phase = rng.uniform(0, 2 * np.pi)
        perturbation += amp * np.cos(2 * np.pi * m * k / n + phase)
    a_new = a * (1.0 + perturbation)
    a_new = np.maximum(a_new, 0)
    S = np.sum(a_new)
    if S > 1e-10:
        a_new = a_new / S
    return a_new


def gaussian_init(n, sigma_frac=0.25):
    """Gaussian bump initialization."""
    k = np.arange(n)
    center = (n - 1) / 2.0
    sigma = sigma_frac * n
    a = np.exp(-0.5 * ((k - center) / sigma) ** 2)
    a = np.maximum(a, 1e-10)
    return a / np.sum(a)


def raised_cosine_init(n, alpha=1.5):
    """Raised cosine shape."""
    k = np.arange(n)
    x = 2.0 * k / max(n - 1, 1) - 1.0
    a = 0.5 * (1.0 + np.cos(np.pi * x)) ** alpha
    a = np.maximum(a, 1e-10)
    return a / np.sum(a)


def cosine_series_init(n, n_modes=8, seed=0):
    """Cosine series initialization with random coefficients."""
    rng = np.random.default_rng(seed)
    k = np.arange(n)
    a = np.ones(n)
    for m in range(1, n_modes + 1):
        c = rng.uniform(-0.5, 0.5) / m
        phase = rng.uniform(0, 2 * np.pi)
        a += c * np.cos(2 * np.pi * m * k / n + phase)
    a = np.maximum(a, 0)
    S = np.sum(a)
    if S < 1e-10:
        return np.ones(n) / n
    return a / S


def two_hump_init(n, width=0.2, center_sep=0.5):
    """Two-hump initialization - energy at two locations."""
    k = np.arange(n)
    c1 = (1 - center_sep) / 2
    c2 = (1 + center_sep) / 2
    a = np.exp(-((k - c1 * n) / (width * n)) ** 2) + np.exp(-((k - c2 * n) / (width * n)) ** 2)
    a = np.maximum(a, 1e-10)
    return a / np.sum(a)


def refine_full(a, time_budget, do_lp=True, lp_iters=50):
    """Full refinement: subgradient + multiplicative + LP."""
    n = len(a)
    a = a.copy().astype(np.float64)
    S = np.sum(a)
    if S < 1e-10:
        return a, np.inf
    a = a / S

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    round_num = 0

    while time.monotonic() - start < time_budget - 2:
        round_num += 1
        remaining = time_budget - (time.monotonic() - start)
        if remaining < 5:
            break

        # Phase A: Subgradient
        if remaining > 8:
            alloc = min(remaining * 0.25, 20.0)
            a_sg = subgradient_descent(best_a, max_iters=int(alloc * 50), time_budget=alloc, topk=8)
            obj_sg = compute_obj(a_sg)
            if obj_sg < best_obj:
                best_obj = obj_sg
                best_a = a_sg.copy()

        # Phase B: Multiplicative update
        remaining = time_budget - (time.monotonic() - start)
        if remaining > 6:
            alloc = min(remaining * 0.15, 12.0)
            a_mu = multiplicative_update(best_a, max_iters=int(alloc * 40), time_budget=alloc, eta=0.08)
            obj_mu = compute_obj(a_mu)
            if obj_mu < best_obj:
                best_obj = obj_mu
                best_a = a_mu.copy()

        # Phase C: LP refinement (most important for fine-tuning)
        remaining = time_budget - (time.monotonic() - start)
        if do_lp and remaining > 10 and n <= 800:
            alloc = min(remaining * 0.4, 40.0)
            a_lp = lp_refine(best_a, max_iters=min(lp_iters, int(alloc * 5)), time_budget=alloc)
            obj_lp = compute_obj(a_lp)
            if obj_lp < best_obj:
                best_obj = obj_lp
                best_a = a_lp.copy()

    return best_a, best_obj


def propose_candidate(seed=42, budget_s=1500):
    """
    Main search function. Uses LP-dominant refinement with subgradient and
    multiplicative updates, structured perturbations, and multi-length exploration.
    """
    global height_sequence_1

    start_time = time.monotonic()
    deadline = start_time + float(budget_s) - 20.0

    best_seq = None
    best_score = np.inf

    # Try to load parent
    parent = None
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64)
        parent = np.clip(parent, 0, 1000.0)
        if parent.size > 0 and np.isfinite(parent).all():
            S_p = np.sum(parent)
            if S_p > 0.01:
                parent = parent / S_p
                parent_obj = compute_obj(parent)
                best_seq = parent.copy()
                best_score = parent_obj
                print(f"Parent loaded: n={parent.size}, obj={parent_obj:.6f}")
    except Exception:
        pass

    if best_seq is None:
        best_seq = gaussian_init(500, 0.25)
        best_score = compute_obj(best_seq)
        print(f"No parent. Using Gaussian init: n={len(best_seq)}, obj={best_score:.6f}")

    n_parent = len(best_seq)

    # === Phase 1: Generate diverse candidates ===
    print("\n=== Phase 1: Diverse initializations ===")
    candidate_pool = [(best_seq.copy(), best_score)]

    n_list = sorted(set([
        max(100, n_parent - 100), n_parent, min(1000, n_parent + 100),
        min(1500, n_parent + 200), max(80, n_parent - 50)
    ]))

    for n in n_list:
        if time.monotonic() > deadline - 100:
            break
        inits = [
            cosine_series_init(n, n_modes=10, seed=seed + n * 7),
            raised_cosine_init(n, alpha=1.5),
            raised_cosine_init(n, alpha=2.5),
            gaussian_init(n, 0.2),
            gaussian_init(n, 0.3),
            two_hump_init(n, width=0.15, center_sep=0.6),
        ]
        for idx, a_init in enumerate(inits):
            if time.monotonic() > deadline - 80:
                break
            obj_init = compute_obj(a_init)
            # Quick subgradient improvement
            a_quick = subgradient_descent(a_init, max_iters=300, time_budget=2.0, topk=6)
            obj_quick = compute_obj(a_quick)
            best_local = a_quick if obj_quick < obj_init else a_init
            best_local_obj = min(obj_quick, obj_init)
            candidate_pool.append((best_local.copy(), best_local_obj))
            if best_local_obj < best_score:
                best_score = best_local_obj
                best_seq = best_local.copy()
                print(f"  n={n}, init{idx}: {best_local_obj:.6f} ** NEW BEST **")

    # === Phase 2: Refine top candidates ===
    print("\n=== Phase 2: Refine top candidates ===")
    candidate_pool.sort(key=lambda x: x[1])
    n_top = min(4, len(candidate_pool))

    for idx in range(n_top):
        if time.monotonic() > deadline - 60:
            break
        remaining = deadline - time.monotonic()
        if remaining < 40:
            break
        a_cand, obj_cand = candidate_pool[idx]
        if obj_cand > best_score + 0.03 and idx > 0:
            break

        alloc = min(remaining * 0.3, 80.0)
        n_cand = len(a_cand)
        print(f"  Refining cand {idx}: n={n_cand}, obj={obj_cand:.6f}, alloc={alloc:.0f}s")

        a_ref, obj_ref = refine_full(a_cand, time_budget=alloc, do_lp=True, lp_iters=40)
        if obj_ref < best_score:
            best_score = obj_ref
            best_seq = a_ref.copy()
            print(f"    -> {obj_ref:.6f} ** NEW BEST **")

    # === Phase 3: Intensive LP refinement on best ===
    print("\n=== Phase 3: Intensive LP refinement ===")
    a_cur = best_seq.copy()
    n_best = len(a_cur)

    for round_num in range(20):
        if time.monotonic() > deadline - 60:
            break
        remaining = deadline - time.monotonic()
        if remaining < 20:
            break

        alloc = min(remaining * 0.3, 60.0)
        a_new, obj_new = refine_full(a_cur, time_budget=alloc, do_lp=True, lp_iters=60)

        if obj_new < best_score - 1e-10:
            best_score = obj_new
            best_seq = a_new.copy()
            a_cur = a_new.copy()
            print(f"  R{round_num}: {best_score:.6f} ** NEW BEST **")
        else:
            print(f"  R{round_num}: {obj_new:.6f} (best={best_score:.6f})")
            # Structured perturbation to escape
            if remaining > 30:
                for pert_seed in [round_num * 101 + 7, round_num * 203 + 13]:
                    if time.monotonic() > deadline - 25:
                        break
                    a_pert = fourier_perturb(a_cur, seed=pert_seed, n_modes=12, scale=0.04)
                    a_p, obj_p = refine_full(a_pert, time_budget=min(15.0, remaining * 0.15),
                                             do_lp=True, lp_iters=30)
                    if obj_p < best_score:
                        best_score = obj_p
                        best_seq = a_p.copy()
                        a_cur = a_p.copy()
                        print(f"  R{round_num} pertert: {best_score:.6f} ** NEW BEST **")
                        break

    # === Phase 4: Length exploration ===
    print("\n=== Phase 4: Length exploration ===")
    if time.monotonic() < deadline - 40:
        n_base = len(best_seq)
        n_tries = sorted(set([
            max(100, n_base - 100), max(100, n_base - 50),
            n_base + 50, n_base + 100, max(n_base + 150, 400),
        ]))
        n_tries = [n for n in n_tries if 80 <= n <= 1500 and n != n_base]

        for n_try in n_tries:
            if time.monotonic() > deadline - 20:
                break
            remaining = deadline - time.monotonic()
            if remaining < 20:
                break

            # Interpolate
            x_old = np.linspace(0, 1, n_base)
            x_new = np.linspace(0, 1, n_try)
            try:
                f_interp = interp1d(x_old, best_seq, kind='cubic', fill_value='extrapolate')
                a_interp = np.maximum(f_interp(x_new), 0)
            except Exception:
                a_interp = np.ones(n_try) / n_try

            S_i = np.sum(a_interp)
            if S_i > 1e-10:
                a_interp = a_interp / S_i

            alloc = min(remaining * 0.3, 35.0)
            a_new, obj_new = refine_full(a_interp, time_budget=alloc, do_lp=True, lp_iters=30)
            if obj_new < best_score:
                best_score = obj_new
                best_seq = a_new.copy()
                print(f"  n={n_try}: {obj_score:.6f} ** NEW BEST **")
            else:
                print(f"  n={n_try}: {obj_new:.6f}")

    # === Phase 5: Final polish ===
    print("\n=== Phase 5: Final polish ===")
    a_cur = best_seq.copy()
    n_final = len(a_cur)

    remaining = deadline - time.monotonic()
    if remaining > 10:
        # Extended LP
        if n_final <= 700:
            alloc = min(remaining * 0.5, 50.0)
            a_lp = lp_refine(a_cur, max_iters=80, time_budget=alloc)
            obj_lp = compute_obj(a_lp)
            if obj_lp < best_score:
                best_score = obj_lp
                best_seq = a_lp.copy()
                a_cur = a_lp.copy()
                print(f"  Final LP: {best_score:.6f} ** NEW BEST **")

        # Final subgradient
        remaining = deadline - time.monotonic()
        if remaining > 10:
            alloc = min(remaining * 0.3, 25.0)
            a_sg = subgradient_descent(a_cur, max_iters=int(alloc * 60), time_budget=alloc, topk=12)
            obj_sg = compute_obj(a_sg)
            if obj_sg < best_score:
                best_score = obj_sg
                best_seq = a_sg.copy()
                print(f"  Final SG: {best_score:.6f} ** NEW BEST **")

        # Small perturbations
        remaining = deadline - time.monotonic()
        if remaining > 8:
            rng = np.random.default_rng(12345)
            for _ in range(4):
                if time.monotonic() > deadline - 5:
                    break
                a_p = a_cur * np.exp(rng.normal(0, 0.02, n_final))
                a_p = np.maximum(a_p, 0)
                S_p = np.sum(a_p)
                if S_p > 1e-10:
                    a_p = a_p / S_p
                    a_p = subgradient_descent(a_p, max_iters=500, time_budget=min(5.0, remaining * 0.2), topk=8)
                    obj_p = compute_obj(a_p)
                    if obj_p < best_score:
                        best_score = obj_p
                        best_seq = a_p.copy()
                        print(f"  Small pert: {best_score:.6f} ** NEW BEST **")

    # Final validation
    if best_seq is None:
        best_seq = np.ones(500) / 500
        best_score = compute_obj(best_seq)

    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    val = evaluate_sequence(result)
    if val < np.inf and val < best_score:
        best_score = val
    elif val == np.inf:
        result = [1.0] * 500
        best_score = 2.0

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_score:.6f}, time={elapsed:.1f}s ===")

    return result
