def smooth_gradient_descent(a, T_init=1.0, T_min=0.01, max_iters=5000, time_budget=60.0):
    n = len(a)
    a = a.copy()
    S = np.sum(a)
    a = a / S
    v = np.zeros(n)  # momentum
    best_a = a.copy()
    best_obj = compute_obj(a)
    T = T_init
    start = time.monotonic()
    
    for it in range(max_iters):
        if time.monotonic() - start > time_budget:
            break
        
        # Compute smooth gradient
        conv = fft_conv(a, a)
        max_c = np.max(conv)
        w = np.exp((conv - max_c) / T)
        w /= np.sum(w)
        
        # Gradient via cross-correlation
        a_rev = a[::-1]
        size = 1
        while size < 3*n:
            size <<= 1
        fa = np.fft.rfft(a_rev, size)
        fw = np.fft.rfft(w, size)
        cross = np.fft.irfft(fa * fw, size)
        grad = 2.0 / T * cross[n-1 : n-1+n]
        
        # Center gradient (preserve sum constraint)
        grad = grad - np.mean(grad)
        
        # Momentum update (Nesterov)
        v = 0.9 * v + 0.1 * grad
        step = 0.1 / (1 + it * 0.001)
        a_new = a - step * v
        a_new = np.maximum(a_new, 0)
        S_new = np.sum(a_new)
        if S_new < 1e-10:
            break
        a_new = a_new / S_new
        
        obj_new = compute_obj(a_new)
        if obj_new < best_obj:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new
        else:
            # Backtrack
            step *= 0.5
        
        # Decrease temperature
        if it > 100:
            T = max(T_min, T * 0.995)
    
    return best_a, best_obj

Optimized step-function coefficient search using FFT-based smooth gradient descent
with adaptive temperature, Nesterov momentum, multi-start diverse initialization,
active-set constrained refinement, and LP polishing. Key innovations: O(n log n)
gradient via cross-correlation, hybrid smooth/exact max, and exploit symmetry.

import numpy as np
from scipy.optimize import linprog
import time


def fft_conv(a, b):
    """Linear convolution via FFT."""
    n = len(a)
    m = len(b)
    size = 1
    while size < n + m - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    return np.fft.irfft(fa * fb, size)[:n + m - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    S = float(np.sum(a))
    if S < 1e-10:
        return np.inf
    conv = fft_conv(a, a)
    M = float(np.max(conv))
    return 2.0 * n * M / (S * S)

    Returns (gradient, smooth_max_value, conv_vector).

        w[np.argmax(conv)] = 1.0
    else:
        w /= w_sum
    smooth_max_val = T * np.log(w_sum) + max_c

    # Gradient: (2/T) * (T_a^T @ w) where (T_a^T @ w)[j] = sum_m a[m] * w[j+m]
    # This is cross-correlation: convolve(a[::-1], w) at appropriate positions
    a_rev = a[::-1]
    size = 1
    while size < n + 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a_rev, size)
    fw = np.fft.rfft(w, size)
    cross = np.fft.irfft(fa * fw, size)
    grad = (2.0 / T) * cross[n - 1:n - 1 + n]
    return grad, smooth_max_val, conv


def project_simplex(a):
    """Project onto {a >= 0, sum(a) = 1}."""
    a = np.maximum(a, 0)
    S = np.sum(a)
    if S < 1e-15:
        return np.ones(len(a)) / len(a)
    return a / S


def smooth_gd(a_init, T_init=0.5, T_min=0.005, max_iters=20000, time_budget=30.0,
              momentum=0.9, verbose=False):
    """
    Smooth gradient descent with adaptive temperature, Nesterov momentum,
    and backtracking line search. Optimizes the smoothed max of convolution.
    """
    n = len(a_init)
    a = project_simplex(a_init.copy())
    v = np.zeros(n)
    best_a = a.copy()
    best_obj = compute_obj(a)
    T = T_init
    start = time.monotonic()
    stall = 0
    step = 0.5

    for it in range(max_iters):
        if time.monotonic() - start > time_budget:
            break

        # Compute smooth gradient
        grad, smooth_val, conv = smooth_gradient_fft(a, T)

        # Center gradient to preserve sum
        grad_centered = grad - np.mean(grad)

        # Nesterov momentum
        v = momentum * v + (1.0 - momentum) * grad_centered

        # Backtracking line search
        alpha = step
        improved = False
        for ls in range(10):
            a_new = a - alpha * v
            a_new = project_simplex(a_new)
            obj_new = compute_obj(a_new)
            if obj_new < best_obj - 1e-12:
                best_obj = obj_new
                best_a = a_new.copy()
                a = a_new
                improved = True
                stall = 0
                break
            alpha *= 0.5

        if not improved:
            stall += 1
            step *= 0.9
            if stall > 50:
                # Reset momentum and try smaller step
                v = np.zeros(n)
                step = max(0.01, step * 0.5)
                if step < 1e-6:
                    break

        # Adaptive temperature: decrease when making progress
        if improved:
            T = max(T_min, T * 0.997)
        elif stall > 20:
            T = min(T_init, T * 1.02)  # Slightly increase if stuck

    return best_a, best_obj


def exact_subgradient_gd(a_init, max_iters=5000, time_budget=20.0, topk=10, verbose=False):
    """
    Subgradient method using exact max (top-K active constraints).
    Used for final polishing when smooth max is no longer helpful.
    """
    n = len(a_init)
    a = project_simplex(a_init.copy())
    v = np.zeros(n)
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    step = 0.1
    stall = 0

    for it in range(max_iters):
        if time.monotonic() - start > time_budget:
            break

        conv = fft_conv(a, a)
        M = float(np.max(conv))

        # Top-K active
        active = np.argsort(conv)[-topk:]
        weights = np.zeros(len(conv))
        weights[active] = 1.0 / len(active)

        # Gradient via cross-correlation (same as smooth case but with uniform weights on active)
        a_rev = a[::-1]
        size = 1
        while size < n + 2 * n - 1:
            size <<= 1
        fa = np.fft.rfft(a_rev, size)
        fw = np.fft.rfft(weights, size)
        cross = np.fft.irfft(fa * fw, size)
        grad = 2.0 * cross[n - 1:n - 1 + n]

        grad_centered = grad - np.mean(grad)
        v = 0.85 * v + 0.15 * grad_centered

        a_new = a - step * v
        a_new = project_simplex(a_new)
        obj_new = compute_obj(a_new)

        if obj_new < best_obj - 1e-12:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new
            stall = 0
        else:
            stall += 1
            step *= 0.95
            if stall > 80:
                break

    return best_a, best_obj


def build_toeplitz_sparse(a):
    """Build sparse Toeplitz matrix T_a such that T_a @ g = conv(a, g)."""
    n = len(a)
    rows = []
    cols = []
    vals = []
    for j in range(n):
        # T_a[k,j] = a[k-j] for 0 <= k-j < n
        k_start = j
        k_end = min(j + n, 2 * n - 1)
        length = k_end - k_start
        if length > 0:
            rows.extend(range(k_start, k_end))
            cols.extend([j] * length)
            vals.extend(list(a[:length]))
    from scipy.sparse import coo_matrix
    A = coo_matrix((vals, (rows, cols)), shape=(2 * n - 1, n)).tocsr()
    return A


def solve_lp(a, M, time_limit=5):
    """Solve LP: max sum(g) s.t. conv(a,g) <= M, g >= 0."""
    n = len(a)
    A = build_toeplitz_sparse(a)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': time_limit})
        if res.status == 0:
            return np.maximum(res.x, 0)
    except Exception:
        pass
    return None


def alternating_lp(a_init, max_iters=50, time_budget=30.0, verbose=False):
    """Alternating LP method for final polishing."""
    n = len(a_init)
    a = project_simplex(a_init.copy())
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    stall = 0

    for it in range(max_iters):
        if time.monotonic() - start > time_budget:
            break

        conv = fft_conv(a, a)
        M = float(np.max(conv))

        g = solve_lp(a, M, time_limit=4)
        if g is None:
            break
        S_g = np.sum(g)
        if S_g < 1e-10:
            break
        g = g / S_g

        # Line search
        best_t = 0.0
        best_obj_ls = best_obj
        for t in [0.05, 0.1, 0.15, 0.2, 0.3, 0.5, 0.7, 1.0]:
            a_try = project_simplex((1.0 - t) * a + t * g)
            obj_try = compute_obj(a_try)
            if obj_try < best_obj_ls:
                best_obj_ls = obj_try
                best_t = t

        if best_t > 0:
            a = project_simplex((1.0 - best_t) * a + best_t * g)
            best_obj = best_obj_ls
            best_a = a.copy()
            stall = 0
            if verbose:
                print(f"    LP iter {it}: obj={best_obj:.6f}, t={best_t}")
        else:
            stall += 1
            if stall > 10:
                # Perturb
                rng = np.random.default_rng(it * 104729 + 7)
                a = project_simplex(a * np.exp(rng.normal(0, 0.05, n)))
                v = np.zeros(n)
                stall = 0
            else:
                a = project_simplex(0.95 * a + 0.05 * g)

    return best_a, best_obj


def cosine_init(n, n_modes=10, seed=0):
    """Cosine series initialization."""
    rng = np.random.default_rng(seed)
    k = np.arange(n)
    a = np.ones(n)
    for m in range(1, n_modes + 1):
        c = rng.uniform(-0.5, 0.5) / (m * 0.7)
        phase = rng.uniform(0, 2 * np.pi)
        a += c * np.cos(2 * np.pi * m * k / n + phase)
    a = np.maximum(a, 0)
    S = np.sum(a)
    if S < 1e-10:
        return np.ones(n) / n
    return a / S


def raised_cosine_init(n, alpha=1.5):
    """Raised cosine window."""
    k = np.arange(n)
    x = 2.0 * k / (n - 1) - 1.0
    a = 0.5 * (1.0 + np.cos(np.pi * x)) ** alpha
    a = np.maximum(a, 1e-10)
    S = np.sum(a)
    return a / S


def gaussian_init(n, sigma_frac=0.3):
    """Gaussian bump."""
    k = np.arange(n)
    center = n / 2.0
    sigma = sigma_frac * n
    a = np.exp(-0.5 * ((k - center) / sigma) ** 2)
    S = np.sum(a)
    return a / S


def symmetric_reflect(a):
    """Make sequence symmetric."""
    n = len(a)
    half = (n + 1) // 2
    a_sym = np.zeros(n)
    a_sym[:half] = a[:half]
    a_sym[half:] = a[:half][::-1][:n - half]
    S = np.sum(a_sym)
    if S < 1e-10:
        return np.ones(n) / n
    return a_sym / S


def propose_candidate(seed=42, budget_s=1500):
    """
    Main search: diverse multi-start smooth GD + active set + LP polish.
    """
    start_time = time.monotonic()
    deadline = start_time + float(budget_s) - 20.0

    best_seq = None
    best_score = np.inf

    # Load parent
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64)
        parent = np.clip(parent, 0, 1000.0)
        if parent.size > 0 and np.isfinite(parent).all():
            parent = project_simplex(parent)
            parent_obj = compute_obj(parent)
            best_score = parent_obj
            best_seq = parent.copy()
            print(f"Parent (n={parent.size}): obj={parent_obj:.6f}")
    except Exception:
        parent = None
        print("No parent available, starting fresh.")

    if best_seq is None:
        n0 = 500
        best_seq = cosine_init(n0, n_modes=8, seed=seed)
        best_score = compute_obj(best_seq)
        print(f"Initial cosine (n={n0}): obj={best_score:.6f}")

    n_best = len(best_seq)

    # === Phase 1: Smooth GD from diverse starts (55% of time) ===
    print("\n=== Phase 1: Smooth Gradient Descent (diverse starts) ===")
    phase1_budget = (deadline - start_time) * 0.55

    # Generate diverse initializations
    n_values = sorted(set([
        n_best,
        max(100, n_best - 100),
        min(1200, n_best + 100),
        min(1500, n_best + 200),
    ]))
    n_values = [n for n in n_values if 100 <= n <= 2000]

    candidates = []  # (a, obj) pairs

    for n in n_values:
        if time.monotonic() > deadline - 120:
            break
        remaining = deadline - time.monotonic()
        if remaining < 30:
            break

        inits = []
        if n == n_best and parent is not None:
            inits.append(("parent", parent.copy() if len(parent) == n else project_simplex(
                np.interp(np.linspace(0, 1, n), np.linspace(0, 1, len(parent)), parent))))
        inits.append((f"cos_{n}", cosine_init(n, n_modes=10, seed=seed + n)))
        inits.append((f"cos2_{n}", cosine_init(n, n_modes=15, seed=seed + n + 1000)))
        inits.append((f"rc_{n}", raised_cosine_init(n, alpha=1.5)))
        inits.append((f"rc2_{n}", raised_cosine_init(n, alpha=2.5)))
        inits.append((f"rc3_{n}", raised_cosine_init(n, alpha=3.5)))
        inits.append((f"gauss_{n}", gaussian_init(n, sigma_frac=0.25)))
        inits.append((f"gauss2_{n}", gaussian_init(n, sigma_frac=0.35)))
        # Symmetric variants
        inits.append((f"sym_rc_{n}", symmetric_reflect(raised_cosine_init(n, alpha=2.0))))
        inits.append((f"sym_g_{n}", symmetric_reflect(gaussian_init(n, sigma_frac=0.3))))

        for name, a_init in inits:
            if time.monotonic() > deadline - 100:
                break
            remaining = deadline - time.monotonic()
            if remaining < 5:
                break
            alloc = min(remaining * 0.15, 25.0)

            # Quick smooth GD
            a_result, obj_result = smooth_gd(a_init, T_init=0.8, T_min=0.01,
                                             max_iters=int(alloc * 50),
                                             time_budget=alloc, momentum=0.9)
            candidates.append((a_result, obj_result))
            if obj_result < best_score:
                best_score = obj_result
                best_seq = a_result.copy()
                print(f"  {name} (n={n}): {obj_result:.6f} ** NEW BEST **")
            else:
                print(f"  {name} (n={n}): {obj_result:.6f}")

    # Sort candidates by objective
    candidates.sort(key=lambda x: x[1])
    print(f"\n  Top 3 after Phase 1: {[f'{c[1]:.6f}' for c in candidates[:3]]}")

    # === Phase 2: Refined smooth GD + subgradient on top candidates (25% of time) ===
    print("\n=== Phase 2: Refined Optimization ===")
    phase2_budget = (deadline - start_time) * 0.25

    n_top = min(5, len(candidates))
    for idx in range(n_top):
        if time.monotonic() > deadline - 60:
            break
        remaining = deadline - time.monotonic()
        if remaining < 10:
            break
        a_cand, obj_cand = candidates[idx]
        if obj_cand > best_score + 0.02:
            break

        n_cand = len(a_cand)
        alloc = min(remaining * 0.3, 40.0)

        # Refined smooth GD with smaller initial T
        a_r, obj_r = smooth_gd(a_cand, T_init=0.3, T_min=0.003,
                               max_iters=int(alloc * 40), time_budget=alloc * 0.6,
                               momentum=0.92)
        if obj_r < best_score:
            best_score = obj_r
            best_seq = a_r.copy()
            print(f"  Refine {idx} (n={n_cand}): {obj_r:.6f} ** NEW BEST **")
            a_cand = a_r

        # Exact subgradient polish
        remaining = deadline - time.monotonic()
        if remaining > 10:
            a_sg, obj_sg = exact_subgradient_gd(a_cand, max_iters=int(remaining * 20),
                                                time_budget=min(remaining * 0.3, 15.0), topk=8)
            if obj_sg < best_score:
                best_score = obj_sg
                best_seq = a_sg.copy()
                print(f"  SG {idx} (n={n_cand}): {obj_sg:.6f} ** NEW BEST **")

        # Try with larger topk
        remaining = deadline - time.monotonic()
        if remaining > 10:
            a_sg2, obj_sg2 = exact_subgradient_gd(a_cand, max_iters=int(remaining * 15),
                                                  time_budget=min(remaining * 0.2, 10.0), topk=20)
            if obj_sg2 < best_score:
                best_score = obj_sg2
                best_seq = a_sg2.copy()
                print(f"  SG2 {idx} (n={n_cand}): {obj_sg2:.6f} ** NEW BEST **")

    # === Phase 3: LP Polish (15% of time) ===
    print("\n=== Phase 3: LP Polish ===")
    phase3_budget = (deadline - start_time) * 0.15

    a_cur = best_seq.copy()
    n_cur = len(a_cur)

    if n_cur <= 600:  # LP is feasible for moderate n
        remaining = deadline - time.monotonic()
        alloc = min(remaining * 0.5, 80.0)
        a_lp, obj_lp = alternating_lp(a_cur, max_iters=40, time_budget=alloc)
        if obj_lp < best_score:
            best_score = obj_lp
            best_seq = a_lp.copy()
            print(f"  LP: {obj_lp:.6f} ** NEW BEST **")
        else:
            print(f"  LP: {obj_lp:.6f} (best stays {best_score:.6f})")

        # Second LP round with perturbation
        remaining = deadline - time.monotonic()
        if remaining > 20:
            rng = np.random.default_rng(seed + 7777)
            a_pert = project_simplex(a_cur * np.exp(rng.normal(0, 0.04, n_cur)))
            a_lp2, obj_lp2 = alternating_lp(a_pert, max_iters=30, time_budget=min(remaining * 0.4, 40.0))
            if obj_lp2 < best_score:
                best_score = obj_lp2
                best_seq = a_lp2.copy()
                print(f"  LP2 (perturbed): {obj_lp2:.6f} ** NEW BEST **")

    # === Phase 4: Final intensive smooth GD on best (remaining time) ===
    print("\n=== Phase 4: Final Polish ===")
    a_cur = best_seq.copy()
    n_cur = len(a_cur)

    while time.monotonic() < deadline - 10:
        remaining = deadline - time.monotonic()
        if remaining < 15:
            break
        alloc = min(remaining * 0.3, 30.0)

        # Smooth GD with very small T
        a_final, obj_final = smooth_gd(a_cur, T_init=0.1, T_min=0.001,
                                       max_iters=int(alloc * 60), time_budget=alloc,
                                       momentum=0.93)
        if obj_final < best_score:
            best_score = obj_final
            best_seq = a_final.copy()
            a_cur = a_final.copy()
            print(f"  Final GD: {best_score:.6f} ** NEW BEST **")
        else:
            print(f"  Final GD: {obj_final:.6f} (no improvement)")
            # Try perturbation
            rng = np.random.default_rng(int(time.monotonic() * 1000) % 100000)
            for scale in [0.03, 0.07, 0.12]:
                if time.monotonic() > deadline - 12:
                    break
                a_p = project_simplex(a_cur * np.exp(rng.normal(0, scale, n_cur)))
                a_p, obj_p = smooth_gd(a_p, T_init=0.2, T_min=0.005,
                                       max_iters=int(alloc * 0.5), time_budget=alloc * 0.4,
                                       momentum=0.9)
                if obj_p < best_score:
                    best_score = obj_p
                    best_seq = a_p.copy()
                    a_cur = a_p.copy()
                    print(f"  Perturb({scale}): {best_score:.6f} ** NEW BEST **")

        # One more LP if time allows
        remaining = deadline - time.monotonic()
        if remaining > 30 and n_cur <= 500:
            a_lp3, obj_lp3 = alternating_lp(a_cur, max_iters=20, time_budget=min(remaining * 0.3, 30.0))
            if obj_lp3 < best_score:
                best_score = obj_lp3
                best_seq = a_lp3.copy()
                a_cur = a_lp3.copy()
                print(f"  Final LP: {best_score:.6f} ** NEW BEST **")

    # Final validation
    if best_seq is None:
        best_seq = np.ones(300) / 300
        best_score = compute_obj(best_seq)

    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    val = evaluate_sequence(result)
    if val < np.inf and val < best_score:
        best_score = val
    elif val == np.inf:
        result = [1.0] * 300
        best_score = 2.0

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_score:.6f}, time={elapsed:.1f}s ===")

    return result
