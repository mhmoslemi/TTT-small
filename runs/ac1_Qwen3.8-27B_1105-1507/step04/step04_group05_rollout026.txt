"""
Minimizes J(a) = 2*n*max(convolve(a,a))/(sum(a))^2 using:
1. LP-based mirror descent with adaptive tightening and line search
2. Simulated annealing restarts
3. Vectorized coordinate descent for final polish
4. Multi-length exploration
Key improvements over previous: better step size selection, adaptive tightening schedule,
restart mechanism to escape local optima, and more efficient coordinate descent.
"""

import numpy as np
from scipy.optimize import linprog
from scipy.sparse import lil_matrix, csr_matrix
import time

# ============ Core Functions ============

def fast_conv(a):
    """Full self-convolution via FFT."""
    n = len(a)
    if n == 0:
        return np.array([0.0])
    s = 1
    while s < 2*n - 1:
        s <<= 1
    fa = np.fft.rfft(a, s)
    return np.fft.irfft(fa*fa, s)[:2*n-1]

def compute_J(a):
    """Compute J(a) = 2*n*max(conv(a,a))/(sum(a))^2."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0)
    S = a.sum()
    if S < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2*len(a)*b.max()/(S*S))

def build_A_matrix(a):
    """Build sparse constraint matrix for LP: (A@g)_k = sum_j a_j * g_{k-j}"""
    n = len(a)
    rows, cols, vals = [], [], []
    for k in range(2*n-1):
        jl = max(0, k-n+1)
        jh = min(n-1, k)
        for j in range(jl, jh+1):
            rows.append(k)
            cols.append(j)
            vals.append(a[k-j])
    if not rows:
        return csr_matrix((2*n-1, n))
    A = lil_matrix((2*n-1, n))
    A[np.array(rows), np.array(cols)] = np.array(vals)
    return A.tocsr()

def lp_solve(a, M):
    """Solve LP: max sum(g) s.t. (a*g)_k <= M for all k, g >= 0."""
    n = len(a)
    c_obj = -np.ones(n)
    A = build_A_matrix(a)
    b_ub = np.full(2*n-1, M)
    bounds = [(0.0, None)] * n
    r = linprog(c_obj, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
    if r.success and r.x is not None:
        g = np.maximum(r.x, 0.0)
        if g.sum() > 1e-10:
            return g
    return None

# ============ LP Mirror Descent ============

def lp_mirror_descent(a, max_iters, deadline, tracker, 
                      tighten_start=0.03, tighten_end=0.002,
                      step_base=0.3):
    """LP-based mirror descent with adaptive tightening and line search."""
    a = np.maximum(a.astype(np.float64), 1e-6)
    a = a / a.sum()
    n = len(a)
    current_J = compute_J(a)
    tracker.update(a)
    
    stall_count = 0
    best_local_J = current_J
    best_local_a = a.copy()
    
    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        
        b = fast_conv(a)
        M = float(b.max())
        
        # Adaptive tightening: start large, decrease over time
        progress = min(1.0, it / max(1, max_iters))
        tighten = tighten_start * (tighten_end / tighten_start) ** (progress ** 0.7)
        
        M_target = M * (1.0 - tighten)
        
        g = lp_solve(a, M_target)
        if g is None:
            # Try less tightening
            M_target = M * (1.0 - tighten * 0.5)
            g = lp_solve(a, M_target)
        if g is None:
            M_target = M * (1.0 - tighten * 0.2)
            g = lp_solve(a, M_target)
        if g is None:
            stall_count += 1
            if stall_count > 30:
                break
            continue
        
        g = np.maximum(g, 0.0)
        g = g / g.sum()
        
        # Line search: try multiple step sizes
        t_candidates = [step_base, step_base*0.5, step_base*0.75, step_base*1.5, 
                        step_base*0.25, 0.9, 0.7, 0.5]
        t_candidates = [t for t in t_candidates if 0 < t < 1]
        
        best_a = None
        best_J = current_J
        
        for t in sorted(t_candidates, key=lambda x: abs(x - step_base)):
            a_trial = (1.0 - t) * a + t * g
            a_trial = np.maximum(a_trial, 0.0)
            s = a_trial.sum()
            if s < 1e-10:
                continue
            a_trial = a_trial / s
            J_trial = compute_J(a_trial)
            if J_trial < best_J - 1e-14:
                best_J = J_trial
                best_a = a_trial
            # Early break if we found a good improvement
            if best_J < current_J * (1 - tighten * 0.5):
                break
        
        if best_a is None:
            stall_count += 1
            if stall_count > 50:
                # Reset to best local
                a = best_local_a.copy()
                current_J = best_local_J
                stall_count = 25
            continue
        
        a = best_a
        current_J = best_J
        
        if current_J < best_local_J - 1e-10:
            best_local_J = current_J
            best_local_a = a.copy()
            stall_count = 0
        
        if it % 50 == 0:
            tracker.update(a)
            print(f"    LP iter {it}: J={current_J:.8f} tighten={tighten:.5f}", flush=True)
    
    tracker.update(a)
    return a

def coord_descent_1d_evaluate(c, a, i, x, S, old):
    """Evaluate J when a[i] = x, given current convolution c and other params."""
    n = len(a)
    new_S = S - old + x
    if new_S < 1e-10:
        return 1e10
    
    # Only need to update c in the range [i, i+n-1]
    k_lo = max(0, i)
    k_hi = min(2*n-1, i+n-1)  # inclusive
    valid_ks = np.arange(k_lo, k_hi+1)
    valid_idx = valid_ks - i
    
    # Compute delta
    delta = 2.0 * (x - old) * a[valid_idx]
    # Fix the k=2i term
    if 2*i <= k_hi and 2*i >= k_lo:
        pos = 2*i - k_lo
        # delta[pos] currently = 2*(x-old)*a[i] = 2*(x-old)*old
        # Should be (x-old)*(x+old)
        delta[pos] = (x - old) * (x + old)
    
    # Compute max of (c + delta) in the affected range, and max of c outside
    affected_max = (c[valid_ks] + delta).max()
    if k_lo > 0:
        outside_left = c[:k_lo].max()
    else:
        outside_left = -np.inf
    if k_hi < 2*n-2:
        outside_right = c[k_hi+1:].max()
    else:
        outside_right = -np.inf
    
    max_c = max(float(affected_max), float(outside_left), float(outside_right))
    return 2.0 * n * max_c / (new_S * new_S)

def coordinate_descent_pass(a):
    """One pass of vectorized coordinate descent."""
    a = np.maximum(a.astype(np.float64), 0.0)
    S = a.sum()
    if S < 1e-10:
        return a, float('inf')
    n = len(a)
    c = fast_conv(a).astype(np.float64)
    
    order = np.random.permutation(n)
    
    for i in order:
        old = a[i]
        
        # Determine search range
        if old > 1e-6:
            lo = 0.0
            hi = max(2.0 * old, S/n * 3.0)
        else:
            lo = 0.0
            hi = S/n * 3.0
        
        if hi < 1e-6:
            hi = 0.1
        
        # Golden section search
        gr = (np.sqrt(5.0) - 1.0) / 2.0
        vc = hi - gr*(hi - lo)
        vd = lo + gr*(hi - lo)
        fc = coord_descent_1d_evaluate(c, a, i, vc, S, old)
        fd = coord_descent_1d_evaluate(c, a, i, vd, S, old)
        
        for _ in range(25):
            if fc < fd:
                hi = vd
                vd = vc
                fd = fc
                vc = hi - gr*(hi - lo)
                fc = coord_descent_1d_evaluate(c, a, i, vc, S, old)
            else:
                lo = vc
                vc = vd
                fc = fd
                vd = lo + gr*(hi - lo)
                fd = coord_descent_1d_evaluate(c, a, i, vd, S, old)
        
        x_opt = max(0.0, (lo + hi) / 2.0)
        
        if abs(x_opt - old) > 1e-12:
            # Update c
            k_lo = max(0, i)
            k_hi = min(2*n-1, i+n-1)
            valid_ks = np.arange(k_lo, k_hi+1)
            valid_idx = valid_ks - i
            delta = 2.0 * (x_opt - old) * a[valid_idx]
            if 2*i <= k_hi and 2*i >= k_lo:
                pos = 2*i - k_lo
                delta[pos] = (x_opt - old) * (x_opt + old)
            c[valid_ks] += delta
            
            a[i] = x_opt
            S = S - old + x_opt
    
    new_J = 2.0*n*c.max()/(S*S)
    return a, new_J

# ============ Resampling ============

def resample_seq(src, dest_n):
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)
