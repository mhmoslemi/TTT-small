```python
import numpy as np
from scipy.optimize import linprog
import time

def fast_conv(a):
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]

def compute_J(a):
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))

def binomial_init(n):
    """Initialize with a[i] = C(2i, i) / 4^i ≈ 1/sqrt(pi*(i+1))"""
    a = np.ones(n)
    for i in range(n):
        # Use the approximation 1/sqrt(pi*(i+1)) for stability
        a[i] = 1.0 / np.sqrt(np.pi * (i + 1))
    return a

def binomial_init_reverse(n):
    """Initialize with reversed binomial coefficients"""
    return binomial_init(n)[::-1].copy()

def lp_best_response(a, M):
    """Solve LP: max sum(g) s.t. (a*g)[k] <= M for all k, g >= 0."""
    n = len(a)
    # Build constraint matrix
    # (a*g)[k] = sum_j a[k-j] * g[j] for valid j
    # A[k, j] = a[k-j] if 0 <= k-j < n, else 0
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        j_min = max(0, k - n + 1)
        j_max = min(n - 1, k)
        for j in range(j_min, j_max + 1):
            rows.append(k)
            cols.append(j)
            vals.append(a[k - j])
    
    from scipy.sparse import csr_matrix
    A = csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))
    c = -np.ones(n)  # minimize -sum(g) = maximize sum(g)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    
    r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                options={'presolve': True, 'time_limit': 60})
    if r.success and r.x is not None:
        return np.maximum(r.x, 0)
    return None
```

```python
def solve_lp(a, M):
    """Solve LP: max sum(g) s.t. (a*g)[k] <= M for all k, g >= 0."""
    n = len(a)
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        j_min = max(0, k - n + 1)
        j_max = min(n - 1, k)
        for j in range(j_min, j_max + 1):
            rows.append(k)
            cols.append(j)
            vals.append(a[k - j])
    
    A = csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    
    try:
        r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                    options={'presolve': True})
        if r.success:
            return np.maximum(r.x, 0), r.fun
    except Exception:
        pass
    return None, None
```

```python
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]


def compute_J(a):
    """Compute J(a) = 2*n*max(conv(a,a))/(sum(a))^2."""
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))


def make_initialization(n, init_type, parent=None):
    """Generate diverse initial sequences of length n."""
    i = np.arange(n, dtype=np.float64)
    if init_type == 'binomial':
        a = 1.0 / np.sqrt(np.pi * (i + 1.0))
    elif init_type == 'binomial_rev':
        a = 1.0 / np.sqrt(np.pi * (n - i))
    elif init_type == 'inv_sqrt_sym':
        a = 1.0 / np.sqrt(np.pi * (np.minimum(i + 1.0, n - i)))
    elif init_type == 'mixed_03':
        a = 1.0 / np.sqrt(np.pi * (i + 1.0)) + 0.3 / np.sqrt(np.pi * (n - i))
    elif init_type == 'mixed_10':
        a = 1.0 / np.sqrt(np.pi * (i + 1.0)) + 1.0 / np.sqrt(np.pi * (n - i))
    elif init_type == 'uniform':
        a = np.ones(n)
    elif init_type == 'linear_decay':
        a = 1.0 - 0.7 * i / max(1, n - 1)
    elif init_type == 'quadratic_decay':
        a = 1.0 / (1.0 + 0.5 * i / max(1, n))
    elif init_type == 'parent':
        if parent is not None and len(parent) > 0:
            t_old = np.linspace(0, 1, len(parent))
            t_new = np.linspace(0, 1, n)
            a = np.maximum(np.interp(t_new, t_old, parent), 0.01)
        else:
            a = np.ones(n)
    else:
        a = np.ones(n)
    a = np.maximum(a, 1e-6)
    return a / a.sum()


def build_conv_matrix(a):
    """Build sparse convolution matrix A where (A@g)[k] = (a*g)[k]."""
    n = len(a)
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        j_min = max(0, k - n + 1)
        j_max = min(n - 1, k)
        for j in range(j_min, j_max + 1):
            rows.append(k)
            cols.append(j)
            vals.append(a[k - j])
    A = csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))
    return A


def solve_lp(a, M):
    """Solve LP: max sum(g) s.t. (a*g)[k] <= M for all k, g >= 0.
    Returns (g, lp_objective_value) or (None, None)."""
    n = len(a)
    A = build_conv_matrix(a)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    try:
        r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                    options={'presolve': True})
        if r.success and r.x is not None:
            g = np.maximum(r.x, 0)
            return g, -r.fun
    except Exception:
        pass
    return None, None


def lp_mirror_descent(a, deadline, max_iters=5000, tol=1e-10):
    """LP mirror descent to minimize max(a*a) subject to sum(a)=1, a>=0."""
    a = np.maximum(a.astype(np.float64), 1e-8)
    a = a / a.sum()
    n = len(a)

    best_a = a.copy()
    best_M = fast_conv(a).max()

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        b = fast_conv(a)
        M = float(b.max())

        if M < best_M:
            best_M = M
            best_a = a.copy()

        g, lp_opt = solve_lp(a, M)
        if g is None:
            break

        g_sum = float(g.sum())
        if g_sum < 1e-10:
            break

        # Convergence check: at optimum, LP optimal = 1 (since a is feasible)
        if g_sum >= 1.0 - tol:
            break

        # Normalize g to sum to 1
        g_norm = g / g_sum

        # Adaptive step size based on gap
        gap = 1.0 - g_sum
        if gap > 0.1:
            t = 0.4
        elif gap > 0.05:
            t = 0.3
        elif gap > 0.01:
            t = 0.2
        elif gap > 0.001:
            t = 0.1
        else:
            t = 0.05

        a_new = (1.0 - t) * a + t * g_norm
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s < 1e-10:
            a_new = a.copy()
        else:
            a_new = a_new / s
        a = a_new

        # Periodic check
        if it % 100 == 0 and it > 0:
            M_check = fast_conv(a).max()
            if M_check < best_M:
                best_M = M_check
                best_a = a.copy()

    return best_a, best_M


def smooth_gradient_optimize(a, deadline, max_iters=300):
    """Smooth gradient descent for fast (less precise) optimization."""
    a = np.maximum(a.astype(np.float64), 1e-8)
    a = a / a.sum()
    n = len(a)

    best_a = a.copy()
    best_M = fast_conv(a).max()
    v = np.zeros(n)

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        frac = it / max(1, max_iters)
        tau = 10.0 + 60.0 * (frac ** 0.5)
        lr = 0.15 * (1.0 - 0.8 * frac ** 0.5)

        b = fast_conv(a)
        b_max = b.max()
        exp_b = np.exp(tau * (b - b_max))
        p = exp_b / exp_b.sum()

        grad = np.zeros(n)
        for k in range(2 * n - 1):
            pk = p[k]
            if pk < 1e-13:
                continue
            lo = max(0, k - n + 1)
            hi = min(n - 1, k)
            grad[lo:hi + 1] += 2.0 * pk * a[k - hi:k - lo + 1][::-1]

        # Momentum
        v = 0.8 * v + lr * grad
        a_new = a - v
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s < 1e-10:
            a_new = a.copy()
        else:
            a_new = a_new / s
        a = a_new

        M = fast_conv(a).max()
        if M < best_M:
            best_M = M
            best_a = a.copy()

    return best_a, best_M


def propose_candidate(seed=42, budget_s=1500, **kwargs):
    """
    Multi-phase search:
    Phase 1: Quick smooth gradient on multiple (length, initialization) combos
    Phase 2: LP mirror descent on top candidates
    Phase 3: Final polish on the overall best
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + float(budget_s) - 15.0

    best_a = None
    best_J = float('inf')

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        if p.ndim == 1 and p.size > 0 and np.all(np.isfinite(p)):
            parent = np.maximum(p, 0.0)
            J_parent = compute_J(parent)
            print(f"Parent: n={parent.size}, J={J_parent:.8f}", flush=True)
            best_a = parent.copy()
            best_J = J_parent
    except (NameError, AttributeError):
        pass

    if best_a is None:
        best_a = make_initialization(256, 'binomial')
        best_J = compute_J(best_a)

    # Phase 1: Quick smooth gradient exploration
    print(f"\n{'='*60}", flush=True)
    print(f"PHASE 1: Quick Smooth Gradient Exploration", flush=True)
    print(f"{'='*60}", flush=True)

    lengths = [256, 384, 512, 640, 768, 1024]
    init_types = ['binomial', 'binomial_rev', 'inv_sqrt_sym', 'mixed_03',
                  'mixed_10', 'linear_decay', 'quadratic_decay', 'parent']

    p1_deadline = min(time.monotonic() + 250.0, deadline - 600.0)
    candidates = []

    for n in lengths:
        if time.monotonic() > p1_deadline:
            break
        for init_type in init_types:
            if time.monotonic() > p1_deadline:
                break
            a0 = make_initialization(n, init_type, parent)
            sub_deadline = min(time.monotonic() + 15.0, p1_deadline)
            a, M = smooth_gradient_optimize(a0, sub_deadline, max_iters=150)
            J = compute_J(a)
            candidates.append((J, a.copy(), n, init_type))
            marker = " **" if J < best_J else ""
            print(f"  n={n:4d} {init_type:16s}: J={J:.8f}{marker}", flush=True)
            if J < best_J:
                best_J = J
                best_a = a.copy()

    print(f"\nPhase 1 best: J={best_J:.8f}, n={best_a.size}", flush=True)

    # Phase 2: LP mirror descent on top candidates
    print(f"\n{'='*60}", flush=True)
    print(f"PHASE 2: LP Mirror Descent (Precise Optimization)", flush=True)
    print(f"{'='*60}", flush=True)

    # Deduplicate: keep only the best per length
    best_per_length = {}
    for J, a, n, init in candidates:
        if n not in best_per_length or J < best_per_length[n][0]:
            best_per_length[n] = (J, a, init)

    # Sort by J
    sorted_candidates = sorted(best_per_length.items(), key=lambda x: x[1][0])

    # Run LP on top candidates, allocating more time to smaller n (faster LP)
    remaining = deadline - time.monotonic()
    num_to_try = min(5, len(sorted_candidates))
    time_per = max(30, remaining / (num_to_try + 1) * 0.8)

    for idx, (n, (J, a, init)) in enumerate(sorted_candidates[:num_to_try]):
        if time.monotonic() > deadline - 120:
            break

        print(f"\n  LP on n={n} (J={J:.6f}, init={init})", flush=True)
        t0 = time.monotonic()
        lp_deadline = min(time.monotonic() + time_per, deadline - 120)

        # Pre-optimize a bit more before LP
        a_pre, _ = smooth_gradient_optimize(a, min(time.monotonic() + 30, lp_deadline - 10),
                                            max_iters=200)
        a_opt, M_opt = lp_mirror_descent(a_pre, lp_deadline, max_iters=5000, tol=1e-10)
        J_opt = compute_J(a_opt)
        elapsed = time.monotonic() - t0
        marker = " ** NEW BEST!" if J_opt < best_J else ""
        print(f"    -> J={J_opt:.8f} ({elapsed:.1f}s){marker}", flush=True)

        if J_opt < best_J:
            best_J = J_opt
            best_a = a_opt.copy()

    # Phase 3: Final polish
    print(f"\n{'='*60}", flush=True)
    print(f"PHASE 3: Final Polish", flush=True)
    print(f"{'='*60}", flush=True)

    if best_a is not None and time.monotonic() < deadline - 20:
        n_best = best_a.size
        print(f"  Polishing n={n_best}, current J={best_J:.8f}", flush=True)
        a_opt, M_opt = lp_mirror_descent(best_a, deadline - 5, max_iters=10000, tol=1e-12)
        J_opt = compute_J(a_opt)
        print(f"  -> J={J_opt:.8f}", flush=True)
        if J_opt < best_J:
            best_J = J_opt
            best_a = a_opt.copy()

    # Phase 4: Try a few more lengths with the best shape
    if time.monotonic() < deadline - 30:
        print(f"\n{'='*60}", flush=True)
        print(f"PHASE 4: Length Adaptation", flush=True)
        print(f"{'='*60}", flush=True)

        n_best = best_a.size
        for new_n in [int(n_best * 0.75), int(n_best * 1.25), int(n_best * 1.5)]:
            if time.monotonic() > deadline - 20:
                break
            if new_n < 128 or new_n > 2048:
                continue
            a_new = make_initialization(new_n, 'parent', best_a)
            a_new, _ = smooth_gradient_optimize(a_new, min(time.monotonic() + 20, deadline - 10),
                                                max_iters=100)
            a_new, _ = lp_mirror_descent(a_new, min(time.monotonic() + 30, deadline - 5),
                                         max_iters=1000, tol=1e-10)
            J_new = compute_J(a_new)
            print(f"  n={new_n}: J={J_new:.8f}", flush=True)
            if J_new < best_J:
                best_J = J_new
                best_a = a_new.copy()
                print(f"    ** NEW BEST!", flush=True)

    # Final verification
    if best_a is not None:
        final_score = float(evaluate_sequence(best_a.tolist()))
        print(f"\n{'='*60}", flush=True)
        print(f"FINAL: evaluate_sequence = {final_score:.8f}, n={best_a.size}", flush=True)
        print(f"{'='*60}", flush=True)

    if best_a is None:
        best_a = make_initialization(256, 'binomial')

    return [float(x) for x in best_a]
```
