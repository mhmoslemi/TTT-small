_matrix_cache = {}

def get_matrix_structure(n):
    """Get or create the pre-allocated matrix structure for convolution."""
    if n not in _matrix_cache:
        rows = []
        cols = []
        for k in range(2*n-1):
            j_min = max(0, k-n+1)
            j_max = min(n-1, k)

def build_A(f):
    """Build sparse convolution matrix with pre-allocated structure."""
    n = len(f)

def fast_conv(a):
    """Full self-convolution via FFT."""
    n = len(a)
    size = 1
    while size < 2*n-1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa*fa, size)[:2*n-1]

def compute_J(a):

    s = a.sum()

    b = fast_conv(a)
    return float(2*len(a)*b.max()/(s*s))

def lp_best_response(f, M):

def smooth_gradient_step(f, tau):
    """One step of smooth gradient descent."""
    n = len(f)
    b = fast_conv(f)
    b_max = b.max()
    exp_b = np.exp(tau * (b - b_max))
    p = exp_b / exp_b.sum()

        lo = max(0, k-n+1)
        hi = min(n-1, k)
        grad[lo:hi+1] += 2.0 * pk * f[k-hi:k-lo+1][::-1]
    
    return grad

def fast_conv(a):
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]


def compute_J(a):
    a = np.maximum(a, 0)
    s = a.sum()

    b = fast_conv(a)

            print(f"Parent: n={parent.size}, J={best_score:.6f}", flush=True)
    except (NameError, AttributeError):
        pass

    if parent is None:
        parent = np.ones(256)
        update(parent)

    # Phase 1: Multi-res smooth gradient
    print(f"\n{'='*60}", flush=True)
    print("PHASE 1: Multi-resolution smooth gradient", flush=True)
    print(f"{'='*60}", flush=True)
    p1_dl = min(time.monotonic() + 180, deadline - 400)

    f_cur = resample_seq(parent, min(64, parent.size))
    f_cur = np.maximum(f_cur, 1e-6)
    f_cur = f_cur / f_cur.sum()
    n_target = parent.size
    res_list = []
    rn = min(64, n_target)
    while rn < n_target:
        rn = min(rn * 2, n_target)
        res_list.append(rn)
    if n_target not in res_list:
        res_list.append(n_target)
    for rn in res_list:
        if time.monotonic() > p1_dl:
            break
        f_cur = resample_seq(f_cur, rn)
        f_cur = np.maximum(f_cur, 1e-6)
        f_cur = f_cur / f_cur.sum()
        f_cur = run_smooth_descent(f_cur, 300, p1_dl)
        update(f_cur)
    print(f"Phase 1 done, J={best_score:.6f}, t={time.monotonic()-t_start:.0f}s", flush=True)

    # Phase 2: LP Mirror Descent (main)
    print(f"\n{'='*60}", flush=True)
    print("PHASE 2: LP Mirror Descent", flush=True)
    print(f"{'='*60}", flush=True)
    p2_dl = min(time.monotonic() + 750, deadline - 250)

    if best_seq is not None:
        n_main = best_seq.size
        # Time a single LP
        t_lp0 = time.monotonic()
        b_tmp = fast_conv(best_seq)
        M_tmp = float(b_tmp.max())
        g_tmp = lp_best_response(best_seq, M_tmp)
        lp_ms = (time.monotonic() - t_lp0) * 1000
        print(f"  n={n_main}, LP time={lp_ms:.0f}ms", flush=True)

        # First burst
        f_opt, J_opt = run_lp_mirror(best_seq, 80000, min(time.monotonic() + 300, p2_dl), t0_init=0.4)
        update(f_opt)
        print(f"  Burst 1: J={J_opt:.6f}, t={time.monotonic()-t_start:.0f}s", flush=True)

        # Continue with smaller steps
        if time.monotonic() < p2_dl:
            f_opt2, J_opt2 = run_lp_mirror(f_opt, 80000, min(time.monotonic() + 250, p2_dl), t0_init=0.2)
            update(f_opt2)
            print(f"  Burst 2: J={J_opt2:.6f}, t={time.monotonic()-t_start:.0f}s", flush=True)
            if time.monotonic() < p2_dl:
                f_opt3, J_opt3 = run_lp_mirror(f_opt2, 80000, p2_dl, t0_init=0.1)
                update(f_opt3)
                print(f"  Burst 3: J={J_opt3:.6f}, t={time.monotonic()-t_start:.0f}s", flush=True)
    else:
        # Fallback: run LP from scratch
        f_try = np.ones(256)
        f_opt, J_opt = run_lp_mirror(f_try, 50000, p2_dl)
        update(f_opt)

    # Phase 3: Subgradient BB polishing
    print(f"\n{'='*60}", flush=True)
    print("PHASE 3: Subgradient BB polish", flush=True)
    print(f"{'='*60}", flush=True)
    p3_dl = min(time.monotonic() + 200, deadline - 80)
    if best_seq is not None:
        f_sg, J_sg = subgrad_step_bb(best_seq, p3_dl, max_iters=50000, lr_init=0.05)
        update(f_sg)
        print(f"Subgrad: J={J_sg:.6f}, t={time.monotonic()-t_start:.0f}s", flush=True)

    # Phase 4: Coordinate descent
    print(f"\n{'='*60}", flush=True)
    print("PHASE 4: Coordinate Descent", flush=True)
    print(f"{'='*60}", flush=True)
    p4_dl = min(time.monotonic() + 80, deadline - 30)
    if best_seq is not None:
        f_cd, J_cd = coord_descent(best_seq, 200, p4_dl)
        update(f_cd)
        print(f"CoordDesc: J={J_cd:.6f}, t={time.monotonic()-t_start:.0f}s", flush=True)

    # Phase 5: Final LP burst on best
    if best_seq is not None and time.monotonic() < deadline - 5:
        print(f"\n{'='*60}", flush=True)
        print("PHASE 5: Final LP burst", flush=True)
        print(f"{'='*60}", flush=True)
        f_fin, J_fin = run_lp_mirror(best_seq, 100000, deadline, t0_init=0.05)
        update(f_fin)

    # Fallback
    if best_seq is None:
        best_seq = np.ones(128)
        best_score = compute_J(best_seq)

    print(f"\n{'='*60}", flush=True)
    print(f"FINAL: J={best_score:.8f} n={best_seq.size} time={time.monotonic()-t_start:.0f}s", flush=True)
    print(f"{'='*60}", flush=True)
    return [float(x) for x in best_seq]
