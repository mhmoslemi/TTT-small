"""
Active-set alternating LP + L-BFGS-B + multiplicative weights hybrid optimizer
for minimizing 2*n*max(conv(a,a))/(sum(a))^2 with non-negative a.

Key improvements:
1. Active-set LP: only constrain top-pct convolution entries (fewer constraints = better LP solutions)
2. Multiplicative weights (mirror descent) as an additional optimizer
3. Better structured initializations (cosine tapers, exponential, Slepian-inspired)
4. Aggressive multi-phase polishing with adaptive time allocation
5. Exploration of larger sequence lengths (up to 1000)
"""

import numpy as np
from scipy.optimize import minimize, linprog
from scipy.fft import rfft, irfft
import time


def compute_conv(a):
    """Compute linear convolution a*a using FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([0.0])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    conv = irfft(fa * fa, size)
    return conv[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    conv = compute_conv(a)
    max_b = float(np.max(conv))
    sum_a = float(np.sum(a))
    if sum_a < 0.01:
        return np.inf
    return 2.0 * n * max_b / (sum_a * sum_a)


def fft_convolve(a, b):
    """Linear convolution of a and b using FFT."""
    n, m = len(a), len(b)
    size = 1
    while size < n + m - 1:
        size <<= 1
    fa = rfft(a, size)
    fb = rfft(b, size)
    return irfft(fa * fb, size)[:n + m - 1]


def normalize_a(a):
    """Normalize a to be non-negative with sum 1."""
    a = np.maximum(a, 0.0)
    s = np.sum(a)
    if s < 1e-30:
        return np.ones(len(a)) / len(a)
    return a / s


def build_toeplitz_rows(a, row_indices):
    """Build Toeplitz matrix rows for given indices. (a*g)[k] = sum_j a[j]*g[k-j]."""
    n = len(a)
    m = len(row_indices)
    A = np.zeros((m, n), dtype=np.float64)
    for idx, k in enumerate(row_indices):
        k = int(k)
        for j in range(n):
            k_minus_j = k - j
            if 0 <= k_minus_j < n:
                A[idx, j] = a[k_minus_j]
    return A


def active_set_lp(a, M, active_pct=0.4, time_limit=10):
    """
    Solve LP: max sum(g) s.t. (a*g)[k] <= M for k in active set, g >= 0.
    Active set = top active_pct fraction of convolution entries.
    """
    n = len(a)
    conv = compute_conv(a)
    
    # Determine active set
    if active_pct >= 1.0:
        active_indices = np.arange(len(conv))
    else:
        threshold = np.percentile(conv, (1.0 - active_pct) * 100)
        active_indices = np.where(conv >= threshold)[0]
    
    if len(active_indices) == 0:
        active_indices = np.argsort(conv)[-max(1, n // 5):]
    
    A = build_toeplitz_rows(a, active_indices)
    c = -np.ones(n)
    b_ub = np.full(len(active_indices), M)
    bounds = [(0.0, None)] * n
    
    try:
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': time_limit})
        if res.status == 0:
            g = np.maximum(res.x, 0)
            s = float(np.sum(g))
            if s > 1e-10:
                return g / s
    except Exception:
        pass
    return None


def full_lp(a, M, time_limit=10):
    """Solve LP with all constraints: max sum(g) s.t. (a*g)[k] <= M for all k, g >= 0."""
    n = len(a)
    conv_len = 2 * n - 1
    A = build_toeplitz_rows(a, np.arange(conv_len))
    c = -np.ones(n)
    b_ub = np.full(conv_len, M)
    bounds = [(0.0, None)] * n
    
    try:
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': time_limit})
        if res.status == 0:
            g = np.maximum(res.x, 0)
            s = float(np.sum(g))
            if s > 1e-10:
                return g / s
    except Exception:
        pass
    return None


def subgradient_step(a, tau=1.0, step_size=0.1):
    """
    One subgradient step using soft-max weighted subgradient.
    Returns updated a (normalized, non-negative).
    """
    n = len(a)
    conv = compute_conv(a)
    max_conv = float(np.max(conv))
    
    shifted = (conv - max_conv) / tau
    shifted = np.clip(shifted, -500, 500)
    weights = np.exp(shifted)
    w_sum = np.sum(weights)
    if w_sum < 1e-300:
        idx = int(np.argmax(conv))
        weights = np.zeros(len(conv))
        weights[idx] = 1.0
    else:
        weights /= w_sum
    
    # grad_a[i] = 2 * sum_j weights[j] * a[j-i] (where valid)
    # = 2 * conv(a, weights_rev)[n-1+i] ... let me think
    # conv(a, w)[k] = sum_i a[i]*w[k-i]
    # We want grad_a[j] = 2 * sum_i w[i] * a[i-j] for valid i-j
    # = 2 * sum_i w[i] * a[i-j]
    # Let's compute conv(w, a_rev) and extract
    # Actually: grad_a[j] = 2 * sum_{i: 0<=i-j<n} w[i] * a[i-j]
    # Let m = i-j, so i = m+j: grad_a[j] = 2 * sum_m w[m+j] * a[m]
    # = 2 * (a * w_shifted)[j] where w_shifted is w reversed...
    # Simpler: use FFT
    w_rev = weights[::-1]
    conv_fw = fft_convolve(a, w_rev)
    # conv_fw[k] = sum_i a[i] * w_rev[k-i] = sum_i a[i] * w[len(w)-1-(k-i)]
    # For j in [0, n-1]: grad_a[j] = 2 * conv_fw[len(weights)-1-j]
    # len(weights) = 2n-1, so grad_a[j] = 2 * conv_fw[2n-2-j]
    grad_a = 2.0 * conv_fw[2 * n - 2: n - 1: -1]
    
    # Center gradient (for simplex projection)
    grad_centered = grad_a - np.mean(grad_a)
    
    a_new = a - step_size * grad_centered
    a_new = normalize_a(a_new)
    return a_new


def multiplicative_weights_step(a, tau=1.0, eta=0.5):
    """
    Multiplicative weights (mirror descent) step.
    a_new[i] proportional to a[i] * exp(-eta * grad_log[i])
    where grad is computed from soft-max weights.
    """
    n = len(a)
    conv = compute_conv(a)
    max_conv = float(np.max(conv))
    
    shifted = (conv - max_conv) / tau
    shifted = np.clip(shifted, -500, 500)
    weights = np.exp(shifted)
    w_sum = np.sum(weights)
    if w_sum < 1e-300:
        idx = int(np.argmax(conv))
        weights = np.zeros(len(conv))
        weights[idx] = 1.0
    else:
        weights /= w_sum
    
    # grad_a[i] = 2 * sum_j weights[j] * a[j-i]
    w_rev = weights[::-1]
    conv_fw = fft_convolve(a, w_rev)
    grad_a = 2.0 * conv_fw[2 * n - 2: n - 1: -1]
    
    # Log-space update: log(a_new) = log(a) - eta * grad_a / a = log(a) - eta * grad_a/a
    # Actually for mirror descent with entropy: a_new[i] = a[i] * exp(-eta * grad_a[i])
    # Then normalize
    log_update = -eta * grad_a
    log_update = np.clip(log_update, -50, 50)  # numerical stability
    a_new = a * np.exp(log_update)
    a_new = normalize_a(a_new)
    return a_new


def softmax(z):
    """Numerically stable softmax."""
    z = z - np.max(z)
    e = np.exp(z)
    return e / np.sum(e)


def soft_max_obj_and_grad_z(z, tau):
    """
    Compute soft-max of conv(softmax(z), softmax(z)) and its gradient w.r.t. z.
    """
    n = len(z)
    a = softmax(z)
    
    conv = compute_conv(a)
    cmax = float(np.max(conv))
    shifted = np.clip((conv - cmax) / tau, -500.0, 500.0)
    exp_conv = np.exp(shifted)
    Z = float(np.sum(exp_conv))
    soft_max_val = cmax + tau * np.log(Z)
    w = exp_conv / Z
    
    w_rev = w[::-1]
    conv_fw = fft_convolve(a, w_rev)
    grad_a = 2.0 * conv_fw[n - 1:2 * n - 1][::-1]
    
    mean_grad = float(np.dot(a, grad_a))
    grad_z = a * (grad_a - mean_grad)
    
    return soft_max_val, grad_z


def optimize_lbfgs(a_init, tau_list, max_iter=200, time_budget=30.0):
    """Optimize using L-BFGS-B with soft-max parametrization and tau annealing."""
    n = len(a_init)
    a = a_init.copy()
    a = normalize_a(a)
    a = np.maximum(a, 1e-30)
    z = np.log(a)
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    
    for tau in tau_list:
        if time.monotonic() - start > time_budget:
            break
        
        def func_and_grad(z_val):
            return soft_max_obj_and_grad_z(z_val, tau)
        
        try:
            result = minimize(func_and_grad, z, jac=True, method='L-BFGS-B',
                              options={'maxiter': max_iter, 'ftol': 1e-14, 'gtol': 1e-8})
            z_new = result.x
        except Exception:
            continue
        
        a_new = softmax(z_new)
        a_new = normalize_a(a_new)
        obj = compute_obj(a_new)
        
        if obj < best_obj:
            best_obj = obj
            best_a = a_new.copy()
            z = z_new
    
    return best_a, best_obj


def alternating_lp(a, max_iters=50, time_budget=60.0, active_pct=0.4):
    """Alternating LP with active set and line search."""
    n = len(a)
    a = normalize_a(a.copy())
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    stall_count = 0
    
    for it in range(max_iters):
        if time.monotonic() - start > time_budget - 2:
            break
        
        conv = compute_conv(a)
        M = float(np.max(conv))
        
        # Use active set LP
        if n <= 800:
            g = active_set_lp(a, M, active_pct=active_pct, time_limit=8)
        else:
            g = active_set_lp(a, M, active_pct=0.3, time_limit=12)
        
        if g is None:
            # Fall back to full LP
            if n <= 500 and it < max_iters // 2:
                g = full_lp(a, M, time_limit=10)
            if g is None:
                # Subgradient fallback
                g = subgradient_step(a, tau=5.0, step_size=0.05)
        
        # Line search
        best_t_obj = best_obj
        best_a_new = best_a.copy()
        for t in [0.3, 0.5, 0.7, 0.9, 1.0]:
            a_try = (1.0 - t) * a + t * g
            a_try = normalize_a(a_try)
            obj_try = compute_obj(a_try)
            if obj_try < best_t_obj:
                best_t_obj = obj_try
                best_a_new = a_try.copy()
        
        if best_t_obj < best_obj - 1e-12:
            best_obj = best_t_obj
            best_a = best_a_new.copy()
            a = best_a_new
            stall_count = 0
        else:
            stall_count += 1
            if stall_count > 15:
                # Perturbation restart
                rng = np.random.default_rng(it * 7919 + 12345)
                scale = min(0.1, 0.03 * stall_count)
                a = a * np.exp(rng.normal(0, scale, n))
                a = normalize_a(a)
                stall_count = 0
            else:
                a = 0.85 * a + 0.15 * g
                a = normalize_a(a)
    
    return best_a


def subgradient_optimize(a, max_iters=2000, time_budget=15.0, tau_start=3.0, tau_end=30.0):
    """Smoothed subgradient optimization with adaptive step size."""
    n = len(a)
    a = normalize_a(a.copy())
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    step = 0.1
    stall = 0
    
    for it in range(max_iters):
        if time.monotonic() - start > time_budget:
            break
        
        frac = it / max(1, max_iters)
        tau = tau_start + (tau_end - tau_start) * frac
        
        a = subgradient_step(a, tau=tau, step_size=step)
        obj = compute_obj(a)
        
        if obj < best_obj - 1e-14:
            best_obj = obj
            best_a = a.copy()
            stall = 0
        else:
            stall += 1
            if stall > 80:
                step *= 0.85
                if step < 1e-6:
                    break
                rng = np.random.default_rng(it * 31 + 7)
                a = a + rng.normal(0, 0.02, n) * step
                a = normalize_a(a)
                stall = 0
        
        if it > 100:
            step *= 0.997
    
    return best_a

def random_init(n, seed):
    """Generate diverse random initializations."""
    rng = np.random.default_rng(seed)
    k = np.arange(n, dtype=float)
    choice = rng.integers(0, 12)
    
    if choice == 0:
        f = rng.uniform(0.5, 4.0)
        phase = rng.uniform(0, 2 * np.pi)
        amp = rng.uniform(0.3, 0.9)
        a = 1.0 + amp * np.cos(2 * np.pi * f * k / n + phase)
    elif choice == 1:
        f1 = rng.uniform(0.3, 2.0)
        f2 = rng.uniform(2.0, 6.0)
        a = 1.0 + 0.4 * np.cos(2 * np.pi * f1 * k / n) + 0.3 * np.cos(2 * np.pi * f2 * k / n + rng.uniform(0, 2*np.pi))
    elif choice == 2:
        # Exponential taper
        beta = rng.uniform(0.5, 4.0)
        center = n / 2
        a = np.exp(-beta * np.abs(k - center) / (n / 2))
    elif choice == 3:
        # Two-lobe (high at ends)
        alpha = rng.uniform(0.3, 0.8)
        a = 1.0 + alpha * np.cos(4 * np.pi * k / n + rng.uniform(0, 2*np.pi))
    elif choice == 4:
        # Slepian-inspired: combination of low-frequency cosines
        n_terms = rng.integers(2, 6)
        a = np.ones(n)
        for t in range(n_terms):
            f = rng.uniform(0.3, 3 + t)
            amp = rng.uniform(0.1, 0.5) / (t + 1)
            a += amp * np.cos(2 * np.pi * f * k / n + rng.uniform(0, 2*np.pi))
    elif choice == 5:
        # Gaussian bump
        center = n * rng.uniform(0.2, 0.8)
        width = n * rng.uniform(0.1, 0.4)
        a = 1.0 + 0.6 * np.exp(-0.5 * ((k - center) / width) ** 2)
    elif choice == 6:
        # Block structure
        nblocks = rng.integers(2, 8)
        a = np.ones(n)
        for b in range(nblocks):
            start = b * n // nblocks
            end = (b + 1) * n // nblocks
            a[start:end] = rng.uniform(0.3, 1.8)
    elif choice == 7:
        # Chirp
        f0 = rng.uniform(0.5, 2.0)
        f1 = rng.uniform(2.0, 6.0)
        a = 1.0 + 0.5 * np.cos(2 * np.pi * (f0 * k / n + (f1 - f0) * k**2 / (2 * n**2)))
    elif choice == 8:
        # Smooth random (low-pass filtered)
        a = rng.random(n)
        win = max(3, n // 15)
        kernel = np.ones(win) / win
        a = np.convolve(a, kernel, mode='same')
        a = a + 0.3
    elif choice == 9:
        # Raised cosine with random phase
        f = rng.uniform(0.5, 3.0)
        phase = rng.uniform(0, 2 * np.pi)
        a = 1.0 + 0.7 * 0.5 * (1 + np.cos(2 * np.pi * f * k / n + phase))
    elif choice == 10:
        # Anti-symmetric-ish (high-low alternation with smoothing)
        f = rng.uniform(1.0, 3.0)
        a = 1.0 + rng.uniform(0.4, 0.8) * np.cos(2 * np.pi * f * k / n)
    else:
        # Pure random with floor
        a = rng.random(n) + 0.1
    
    a = np.maximum(a, 0.01)
    return a / np.sum(a)


def resample_sequence(a, new_n):
    """Resample sequence to new length via interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return a_new

                best_a = a_new.copy()
                if verbose:
                    print(f"    Polish r{round_num} LBFGS: {best_obj:.6f}")
        
        # 3. Multiplicative weights
        remaining = time_budget - (time.monotonic() - start)
        if remaining > 10:
            mw_time = min(remaining * 0.15, 12.0)
            a_new = multiplicative_weights_optimize(best_a, max_iters=int(mw_time * 30),
                                                     time_budget=mw_time, tau_start=5.0, tau_end=30.0)
            obj = compute_obj(a_new)
            if obj < best_obj:
                best_obj = obj
                best_a = a_new.copy()
                if verbose:
                    print(f"    Polish r{round_num} MW: {best_obj:.6f}")
        
        # 4. Subgradient
        remaining = time_budget - (time.monotonic() - start)
        if remaining > 8:
            sg_time = min(remaining * 0.15, 10.0)
            a_new = subgradient_optimize(best_a, max_iters=int(sg_time * 20),
                                          time_budget=sg_time, tau_start=10.0, tau_end=50.0)
            obj = compute_obj(a_new)
            if obj < best_obj:
                best_obj = obj
                best_a = a_new.copy()
                if verbose:
                    print(f"    Polish r{round_num} SG: {best_obj:.6f}")
    
    return best_a, best_obj


def propose_candidate(seed=42, budget_s=1500):
    """
    Main search: multi-phase optimization combining active-set LP, L-BFGS-B,
    multiplicative weights, and subgradient methods.
    """
    start_time = time.monotonic()
    deadline = start_time + float(budget_s) - 30.0
    
    best_seq = None
    best_score = np.inf
    
    # Load parent
    parent, parent_obj = try_parent()
    if parent is not None and parent_obj < best_score:
        best_score = parent_obj
        best_seq = parent.copy()
    
    # N values to explore
    n_values = [300, 400, 500, 600, 700, 800, 1000]
    if best_seq is not None:
        np_ = len(best_seq)
        extra = [np_, max(100, np_ - 100), min(1200, np_ + 100), min(1200, np_ + 200)]
        n_values = sorted(set(n_values + extra))
    
    print(f"Exploring n values: {n_values}")
    print(f"Initial best: {best_score:.6f}")
    print(f"Time budget: {budget_s}s, deadline in {deadline - start_time:.0f}s")
    
    # Phase 1: Exploration across n values
    candidates = []
    
    for n in n_values:
        if time.monotonic() > deadline - 300:
            break
        remaining = deadline - time.monotonic()
        if remaining < 30:
            break
        
        # Time allocation per n
        if n <= 400:
            alloc = min(40.0, remaining * 0.08)
        elif n <= 600:
            alloc = min(55.0, remaining * 0.1)
        elif n <= 800:
            alloc = min(65.0, remaining * 0.11)
        else:
            alloc = min(80.0, remaining * 0.12)
        alloc = max(15.0, alloc)
        
        print(f"\nN={n}: {remaining:.0f}s remaining, alloc={alloc:.0f}s")
        
        n_inits = max(2, int(alloc / 18))
        n_inits = min(n_inits, 4)
        
        for init_idx in range(n_inits):
            if time.monotonic() > deadline - 300:
                break
            remaining_init = deadline - time.monotonic()
            if remaining_init < 15:
                break
            
            # Initialize
            if init_idx == 0 and best_seq is not None:
                a = resample_sequence(best_seq, n)
            else:
                a = random_init(n, seed + n * 104729 + init_idx * 79193)
            
            # Quick subgradient shaping
            sg_time = min(5.0, (alloc / n_inits) * 0.15)
            a = subgradient_optimize(a, max_iters=int(sg_time * 30),
                                      time_budget=sg_time, tau_start=2.0, tau_end=15.0)
            
            # L-BFGS-B with tau annealing
            remaining_init = deadline - time.monotonic()
            if remaining_init > 15:
                lfbs_time = min((alloc / n_inits) * 0.35, 20.0)
                tau_list = [4.0 / n, 2.0 / n, 1.0 / n, 0.5 / n, 0.2 / n, 0.1 / n]
                a, obj = optimize_lbfgs(a, tau_list, max_iter=150, time_budget=lfbs_time)
            else:
                obj = compute_obj(a)
            
            # Quick active-set LP
            remaining_init = deadline - time.monotonic()
            if remaining_init > 20:
                lp_time = min((alloc / n_inits) * 0.3, 20.0)
                a = alternating_lp(a, max_iters=20, time_budget=lp_time, active_pct=0.4)
                obj = compute_obj(a)
            
            # Multiplicative weights
            remaining_init = deadline - time.monotonic()
            if remaining_init > 10:
                mw_time = min(5.0, remaining_init * 0.1)
                a = multiplicative_weights_optimize(a, max_iters=int(mw_time * 20),
                                                     time_budget=mw_time, tau_start=5.0, tau_end=25.0)
                obj = compute_obj(a)
            
            print(f"  init{init_idx}: n={n}, obj={obj:.6f}")
            
            if obj < best_score:
                best_score = obj
                best_seq = a.copy()
                print(f"  ** UPDATE: n={n}, obj={best_score:.6f} **")
            
            candidates.append((a.copy(), obj, n))
    
    # Phase 2: Polishing top candidates
    print(f"\n=== POLISHING PHASE ===")
    print(f"Best before polish: {best_score:.6f}")
    print(f"Time remaining: {deadline - time.monotonic():.0f}s")
    
    candidates.sort(key=lambda x: x[1])
    
    n_polish = min(4, len(candidates))
    for ci in range(n_polish):
        if time.monotonic() > deadline - 100:
            break
        remaining = deadline - time.monotonic()
        if remaining < 30:
            break
        
        a_cand = candidates[ci][0]
        n_cand = candidates[ci][2]
        print(f"  Polishing candidate {ci} (n={n_cand}, obj={candidates[ci][1]:.6f}), {remaining:.0f}s left")
        
        polish_budget = min(remaining * 0.35, 100.0)
        a_pol, obj_pol = polish_sequence(a_cand, time_budget=polish_budget, verbose=True)
        
        if obj_pol < best_score:
            best_score = obj_pol
            best_seq = a_pol.copy()
            print(f"  ** UPDATE after polish: obj={best_score:.6f} **")
        else:
            # Try perturbation
            remaining = deadline - time.monotonic()
            if remaining > 30:
                rng = np.random.default_rng(ci * 555 + 999)
                for ptry in range(4):
                    if time.monotonic() > deadline - 50:
                        break
                    scale = rng.uniform(0.02, 0.1)
                    a_pert = a_cand * np.exp(rng.normal(0, scale, n_cand))
                    a_pert = normalize_a(a_pert)
                    a_pert = subgradient_optimize(a_pert, max_iters=400, time_budget=4.0,
                                                   tau_start=5.0, tau_end=25.0)
                    obj_pert = compute_obj(a_pert)
                    if obj_pert < best_score:
                        best_score = obj_pert
                        best_seq = a_pert.copy()
                        print(f"  ** UPDATE (perturb {ptry}): obj={best_score:.6f} **")
    
    # Phase 3: Final intensive polish on the global best
    if best_seq is not None and time.monotonic() < deadline - 30:
        remaining = deadline - time.monotonic()
        n_best = len(best_seq)
        print(f"\n=== FINAL POLISH: n={n_best}, {remaining:.0f}s ===")
        
        a_cur = best_seq.copy()
        best_obj_cur = best_score
        
        for final_round in range(6):
            if time.monotonic() > deadline - 15:
                break
            remaining = deadline - time.monotonic()
            if remaining < 15:
                break
            
            # Active-set LP (primary)
            if remaining > 25:
                lp_time = min(remaining * 0.3, 40.0)
                a_new = alternating_lp(a_cur, max_iters=50, time_budget=lp_time, active_pct=0.35)
                obj_new = compute_obj(a_new)
                if obj_new < best_obj_cur:
                    best_obj_cur = obj_new
                    best_score = obj_new
                    best_seq = a_new.copy()
                    a_cur = a_new.copy()
                    print(f"  Final ALP r{final_round}: {best_score:.6f}")
            
            # L-BFGS-B with very small tau
            remaining = deadline - time.monotonic()
            if remaining > 20:
                tau_list = [0.5 / n_best, 0.2 / n_best, 0.08 / n_best, 0.03 / n_best]
                lfbs_time = min(remaining * 0.2, 20.0)
                a_new, obj_new = optimize_lbfgs(a_cur, tau_list, max_iter=250, time_budget=lfbs_time)
                if obj_new < best_obj_cur:
                    best_obj_cur = obj_new
                    best_score = obj_new
                    best_seq = a_new.copy()
                    a_cur = a_new.copy()
                    print(f"  Final LBFGS r{final_round}: {best_score:.6f}")
            
            # Multiplicative weights
            remaining = deadline - time.monotonic()
            if remaining > 12:
                mw_time = min(remaining * 0.15, 12.0)
                a_new = multiplicative_weights_optimize(a_cur, max_iters=int(mw_time * 25),
                                                         time_budget=mw_time, tau_start=8.0, tau_end=40.0)
                obj_new = compute_obj(a_new)
                if obj_new < best_obj_cur:
                    best_obj_cur = obj_new
                    best_score = obj_new
                    best_seq = a_new.copy()
                    a_cur = a_new.copy()
                    print(f"  Final MW r{final_round}: {best_score:.6f}")
            
            # Subgradient
            remaining = deadline - time.monotonic()
            if remaining > 10:
                sg_time = min(remaining * 0.12, 10.0)
                a_new = subgradient_optimize(a_cur, max_iters=int(sg_time * 15),
                                              time_budget=sg_time, tau_start=15.0, tau_end=60.0)
                obj_new = compute_obj(a_new)
                if obj_new < best_obj_cur:
                    best_obj_cur = obj_new
                    best_score = obj_new
                    best_seq = a_new.copy()
                    a_cur = a_new.copy()
                    print(f"  Final SG r{final_round}: {best_score:.6f}")
    
    # Final validation
    if best_seq is None:
        best_seq = np.ones(200)
        best_score = compute_obj(best_seq)
    
    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    val = evaluate_sequence(result)
    if val < np.inf:
        best_score = val
    else:
        result = [1.0] * 100
        best_score = 2.0
    
    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_score:.6f}, time={elapsed:.1f}s ===")
    
    return result
