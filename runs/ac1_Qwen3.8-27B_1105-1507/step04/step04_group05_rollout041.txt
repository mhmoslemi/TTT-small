            break
        
        t += 1
        tau = tau_schedule(step)
        p = stable_softmax(x)

import numpy as np
from scipy.optimize import linprog
from scipy.sparse import lil_matrix
import time


def fast_conv(a):
    """Full self-convolution via FFT."""
    n = len(a)
    if n == 0:
        return np.array([0.0])
    s = 1
    while s < 2 * n - 1:
        s <<= 1
    fa = np.fft.rfft(a, s)
    return np.fft.irfft(fa * fa, s)[:2 * n - 1]


def compute_J(a):
    """Compute J(a) = 2*n*max(conv(a,a))/(sum(a))^2."""
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))


def stable_softmax(x):
    """Numerically stable softmax."""
    x = x - np.max(x)
    ex = np.exp(x)
    s = ex.sum()
    if s < 1e-300:
        return np.ones(len(x)) / len(x)
    return ex / s


def compute_soft_gradient_fft(p, w):
    """Compute g_j = 2 * sum_m w[m+j] * p[m] using FFT cross-correlation."""
    n = len(p)
    L = 1
    while L < 3 * n:
        L <<= 1
    W = np.fft.fft(w, L)
    P = np.fft.fft(p, L)
    corr = np.fft.ifft(W * np.conj(P)).real
    return 2.0 * corr[:n]


def compute_soft_gradient_direct(p, w):
    """Compute g_j = 2 * sum_m w[m+j] * p[m] directly (for small n)."""
    n = len(p)
    g = np.zeros(n)
    for j in range(n):
        k_lo = j
        k_hi = min(j + n - 1, 2 * n - 2)
        g[j] = 2.0 * np.dot(w[k_lo:k_hi + 1], p[0:k_hi - k_lo + 1])
    return g


def adam_optimize(x0, n_steps, tau, lr, deadline, tracker, label=""):
    """Optimize x using Adam with log-sum-exp smooth objective in softmax parameterization."""
    n = len(x0)
    x = x0.astype(np.float64).copy()
    m = np.zeros(n)
    v = np.zeros(n)
    t = 0
    best_J = float('inf')
    
    for step in range(n_steps):
        if time.monotonic() > deadline:
            break
        
        t += 1
        p = stable_softmax(x)
        
        # Compute convolution and objective
        c = fast_conv(p)
        J = 2.0 * n * c.max()
        
        if J < best_J:
            best_J = J
            if step % 50 == 0:
                tracker.update(p)
        
        # Compute soft weights
        tau_c = tau * c
        tau_c_shifted = tau_c - tau_c.max()
        w = np.exp(tau_c_shifted)
        w_sum = w.sum()
        if w_sum < 1e-300:
            break
        w /= w_sum
        
        # Compute soft gradient
        if n <= 500:
            g = compute_soft_gradient_direct(p, w)
        else:
            g = compute_soft_gradient_fft(p, w)
        
        dot_pg = np.dot(p, g)
        grad = p * (g - dot_pg)
        
        # Adam update
        beta1, beta2, eps = 0.9, 0.999, 1e-8
        lr_t = lr * np.sqrt(1 - beta2**t) / (1 - beta1**t)
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * grad * grad
        m_hat = m / (1 - beta1**t)
        v_hat = v / (1 - beta2**t)
        x = x - lr_t * m_hat / (np.sqrt(v_hat) + eps)
        
        if step % 200 == 0:
            tracker.update(stable_softmax(x))
    
    p_final = stable_softmax(x)
    tracker.update(p_final)
    return x, best_J

    t = 0

def build_A_sparse(a):
    """Build sparse constraint matrix for LP."""
    n = len(a)
    A = lil_matrix((2 * n - 1, n))
    for k in range(2 * n - 1):
        jl = max(0, k - n + 1)
        jh = min(n - 1, k)
        for j in range(jl, jh + 1):
            A[k, j] = a[k - j]
    return A.tocsr()


def lp_best_response(a, M):
    """LP: max sum(g) s.t. (a*g)_k <= M for all k, g >= 0."""
    n = len(a)
    c = -np.ones(n)
    A = build_A_sparse(a)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    try:
        r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                    options={'presolve': True, 'time_limit': 30})
        if r.success and r.x is not None:
            g = np.maximum(r.x, 0.0)
            if g.sum() > 1e-10:
                return g
    except Exception:
        pass
    return None


def lp_mirror_descent(a, max_iters, deadline, tracker, tighten_base=0.003):
    """LP mirror descent iterations."""
    a = np.maximum(a.astype(np.float64), 0.0001)
    a = a / a.sum()
    n = len(a)
    current_J = compute_J(a)
    tracker.update(a)
    
    tighten = tighten_base
    stall_count = 0
    
    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        
        b = fast_conv(a)
        M = float(b.max())
        M_target = M * (1.0 - tighten)
        
        g = lp_best_response(a, M_target)
        if g is None:
            M_target = M * (1.0 - tighten * 0.5)
            g = lp_best_response(a, M_target)
        if g is None:
            tighten *= 0.5
            if tighten < 1e-7:
                tighten = tighten_base
            stall_count += 1
            if stall_count > 15:
                break
            continue
        
        g = np.maximum(g, 0.0)
        g_sum = g.sum()
        if g_sum < 1e-10:
            break
        g /= g_sum
        
        # Try multiple step sizes
        progress = min(1.0, it / max(1, max_iters))
        base_t = max(0.05, 0.6 * (1.0 - 0.5 * progress))
        candidates_t = [base_t, base_t * 0.5, base_t * 1.5, 0.3, 0.7]
        
        best_a = None
        best_J = current_J
        
        for t in candidates_t:
            a_trial = (1.0 - t) * a + t * g
            a_trial = np.maximum(a_trial, 0.0)
            s = a_trial.sum()
            if s < 1e-10:
                continue
            a_trial /= s
            J_trial = compute_J(a_trial)
            if J_trial < best_J - 1e-14:
                best_J = J_trial
                best_a = a_trial
        
        if best_a is None:
            stall_count += 1
            if stall_count > 20:
                break
            tighten *= 0.7
            continue
        
        a = best_a
        current_J = best_J
        stall_count = 0
        tighten = min(0.05, tighten * (1.0 - 0.97 * (1.0 - progress)))
        
        if it % 10 == 0:
            tracker.update(a)
    
    tracker.update(a)
    return a


def coordinate_descent_pass_fast(a, max_iter=3):
    """Fast coordinate descent with golden section search on each coordinate."""
    a = np.maximum(a.astype(np.float64), 0.0)
    S = a.sum()
    if S < 1e-10:
        return a, float('inf')
    n = len(a)
    b = fast_conv(a)
    
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    indices = np.arange(n)
    np.random.shuffle(indices)
    
    for _ in range(max_iter):
        for i in indices:
            ai = a[i]
            
            def f_of_x(x):
                new_S = S - ai + x
                if new_S < 1e-10:
                    return 1e10
                # Only convolution values k in [i, i+n-1] change
                max_c = 0.0
                for k in range(2 * n - 1):
                    if i <= k <= i + n - 1:
                        ki = k - i
                        if 0 <= ki < n:
                            if k == 2 * i:
                                ck = b[k] - ai * ai + x * x
                            else:
                                ck = b[k] - 2.0 * ai * a[ki] + 2.0 * x * a[ki]
                        else:
                            ck = b[k]
                    else:
                        ck = b[k]
                    if ck > max_c:
                        max_c = ck
                return 2.0 * n * max_c / (new_S * new_S)
            
            # Golden section search
            lo = 0.0
            hi = max(3.0 * ai, S / n * 2.0, 0.01)
            
            c_pt = hi - gr * (hi - lo)
            d_pt = lo + gr * (hi - lo)
            fc = f_of_x(c_pt)
            fd = f_of_x(d_pt)
            
            for _ in range(25):
                if fc < fd:
                    hi = d_pt
                    d_pt = c_pt
                    fd = fc
                    c_pt = hi - gr * (hi - lo)
                    fc = f_of_x(c_pt)
                else:
                    lo = c_pt
                    c_pt = d_pt
                    fc = fd
                    d_pt = lo + gr * (hi - lo)
                    fd = f_of_x(d_pt)
            
            x_opt = max(0.0, (lo + hi) / 2.0)
            
            if abs(x_opt - ai) > 1e-10:
                # Update convolution incrementally
                for k in range(max(0, i), min(2 * n - 1, i + n)):
                    ki = k - i
                    if 0 <= ki < n:
                        if k == 2 * i:
                            b[k] = b[k] - ai * ai + x_opt * x_opt
                        else:
                            b[k] = b[k] - 2.0 * ai * a[ki] + 2.0 * x_opt * a[ki]
                a[i] = x_opt
                S = S - ai + x_opt
                ai = x_opt
    
    # Verify
    b_check = fast_conv(a)
    new_J = 2.0 * n * b_check.max() / (S * S)
    return a, new_J


def generate_structural_inits(n, rng):
    """Generate structurally diverse initial sequences."""
    inits = []
    t = np.linspace(0, 1, n, endpoint=False)
    
    # Uniform
    inits.append(np.ones(n))
    
    # Geometric decay (left-heavy)
    for ratio in [0.97, 0.98, 0.99]:
        inits.append(np.array([ratio**i for i in range(n)]))
    
    # Geometric decay (right-heavy)
    for ratio in [0.97, 0.98, 0.99]:
        inits.append(np.array([ratio**(n-1-i) for i in range(n)]))
    
    # Parabolic
    inits.append(1.0 - t**2)
    inits.append(t**2)
    inits.append(1.0 - (2*t - 1)**2)
    
    # Sine-based
    for freq in [1, 2, 3]:
        inits.append(0.5 + 0.5 * np.sin(np.pi * t))
        inits.append(0.3 + 0.7 * np.cos(2 * np.pi * freq * t - np.pi/2)**2)
    
    # Exponential peak
    for center in [0.25, 0.5, 0.75]:
        inits.append(np.exp(-20 * (t - center)**2) + 0.05)
    
    # Asymmetric: high on one side
    inits.append(np.concatenate([np.ones(n//3), np.linspace(1.0, 0.1, 2*n//3)]))
    inits.append(np.concatenate([np.linspace(0.1, 1.0, 2*n//3), np.ones(n//3)]))
    
    # Power law
    inits.append(1.0 / (t + 0.1))
    inits.append(1.0 / (1.0 - t + 0.1))
    
    # Random with structure
    for _ in range(3):
        base = 1.0 + 0.3 * np.cos(2 * np.pi * rng.uniform(1, 5) * t + rng.uniform(0, 6.28))
        base += 0.2 * np.cos(2 * np.pi * rng.uniform(3, 10) * t)
        inits.append(np.maximum(base, 0.01))
    
    # Normalize all
    result = []
    for a in inits:
        a = np.maximum(a, 0.001)
        a = a / a.sum()
        result.append(a)
    
    return result

def resample_seq(src, dest_n):
    """Resample 1D sequence to new length via linear interpolation."""
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)

            print(f"    CD pass {cd_pass}: J={new_J:.8f}", flush=True)
    
    tracker.report(f"Phase 3 CD: {time.monotonic()-t0:.1f}s")
    
    # === Phase 4: Alternating Adam + LP + CD ===
    print(f"\n{'='*60}", flush=True)
    print(f"Phase 4: Alternating optimization", flush=True)
    print(f"{'='*60}", flush=True)
    t0 = time.monotonic()
    
    if tracker.best is not None:
        a_alt = tracker.best.copy()
    else:
        a_alt = a_cd
    
    for alt_round in range(8):
        if time.monotonic() > deadline - 80:
            break
        
        # Adam step
        x_alt = np.log(np.maximum(a_alt, 1e-6))
        sub_deadline = min(deadline - 60, time.monotonic() + 40)
        x_alt, _ = adam_optimize_adaptive(x_alt, n_steps=2000,
                                           deadline=sub_deadline,
                                           tracker=tracker, label="alt")
        a_alt = stable_softmax(x_alt)
        a_alt = np.maximum(a_alt, 0.001)
        a_alt /= a_alt.sum()
        
        # LP step
        if time.monotonic() < deadline - 40:
            a_alt = lp_mirror_descent(a_alt, max_iters=15,
                                       deadline=min(deadline - 30, time.monotonic() + 20),
                                       tracker=tracker, tighten_base=0.003)
        
        # CD step
        if time.monotonic() < deadline - 30:
            a_alt, _ = coordinate_descent_pass_fast(a_alt, max_iter=2)
            a_alt = np.maximum(a_alt, 0.001)
            a_alt /= a_alt.sum()
            tracker.update(a_alt)
        
        print(f"    Alt round {alt_round}: J={compute_J(a_alt):.8f}", flush=True)
    
    tracker.report(f"Phase 4 Alt: {time.monotonic()-t0:.1f}s")
    
    # === Phase 5: Length search ===
    print(f"\n{'='*60}", flush=True)
    print(f"Phase 5: Length Search", flush=True)
    print(f"{'='*60}", flush=True)
    t0 = time.monotonic()
    
    base = tracker.best if tracker.best is not None else a_alt
    n_base = base.size
    
    # Try a range of lengths
    length_candidates = []
    for offset in range(-20, 21, 2):
        nl = n_base + offset
        if 100 <= nl <= 800:
            length_candidates.append(nl)
    # Add some specific lengths
    for nl in [200, 256, 300, 320, 350, 400, 450, 500, 600]:
        if nl not in length_candidates:
            length_candidates.append(nl)
    
    length_candidates = sorted(set(length_candidates))
    
    for nl in length_candidates:
        if time.monotonic() > deadline - 40:
            break
        
        a_res = resample_seq(base, nl)
        a_res = np.maximum(a_res, 0.001)
        a_res /= a_res.sum()
        
        # Quick Adam
        x_res = np.log(a_res)
        x_res, _ = adam_optimize_adaptive(x_res, n_steps=1500,
                                           deadline=min(time.monotonic() + 25, deadline - 30),
                                           tracker=tracker, label=f"len{nl}")
        a_res = stable_softmax(x_res)
        
        # Quick CD
        if time.monotonic() < deadline - 30:
            a_res, _ = coordinate_descent_pass_fast(a_res, max_iter=1)
            a_res = np.maximum(a_res, 0.001)
            a_res /= a_res.sum()
            tracker.update(a_res)
        
        print(f"    n={nl}: best J={tracker.best_score:.8f}", flush=True)
    
    tracker.report(f"Phase 5 Lengths: {time.monotonic()-t0:.1f}s")
    
    # === Phase 6: Perturbation restarts ===
    print(f"\n{'='*60}", flush=True)
    print(f"Phase 6: Perturbation Restarts", flush=True)
    print(f"{'='*60}", flush=True)
    t0 = time.monotonic()
    
    if tracker.best is not None:
        a_best = tracker.best.copy()
    else:
        a_best = base
    
    n = a_best.size
    for temp in [0.05, 0.03, 0.02, 0.01, 0.005]:
        if time.monotonic() > deadline - 20:
            break
        
        a_pert = a_best.copy() + rng.normal(0, temp, n)
        a_pert = np.maximum(a_pert, 0.001)
        a_pert /= a_pert.sum()
        
        # Adam
        x_pert = np.log(a_pert)
        x_pert, _ = adam_optimize_adaptive(x_pert, n_steps=1500,
                                            deadline=min(time.monotonic() + 20, deadline - 15),
                                            tracker=tracker, label=f"pert{temp}")
        a_pert = stable_softmax(x_pert)
        
        # CD
        if time.monotonic() < deadline - 15:
            a_pert, _ = coordinate_descent_pass_fast(a_pert, max_iter=1)
            a_pert = np.maximum(a_pert, 0.001)
            a_pert /= a_pert.sum()
            tracker.update(a_pert)
    
    # Also try structured perturbations
    for _ in range(5):
        if time.monotonic() > deadline - 20:
            break
        # Random segment flip
        a_pert = a_best.copy()
        seg_start = rng.integers(0, n // 2)
        seg_end = rng.integers(seg_start + 10, n)
        a_pert[seg_start:seg_end] = a_best[seg_end-1:seg_start-1:-1] if seg_end > seg_start else a_best[seg_start:seg_end]
        a_pert = np.maximum(a_pert, 0.001)
        a_pert /= a_pert.sum()
        
        x_pert = np.log(a_pert)
        x_pert, _ = adam_optimize_adaptive(x_pert, n_steps=1000,
                                            deadline=min(time.monotonic() + 15, deadline - 15),
                                            tracker=tracker, label="flip")
        a_pert = stable_softmax(x_pert)
        tracker.update(a_pert)
    
    tracker.report(f"Phase 6 Restarts: {time.monotonic()-t0:.1f}s")
    
    # === Phase 7: Final intensive polish ===
    print(f"\n{'='*60}", flush=True)
    print(f"Phase 7: Final Polish", flush=True)
    print(f"{'='*60}", flush=True)
    t0 = time.monotonic()
    
    if tracker.best is not None:
        a_final = tracker.best.copy()
    else:
        a_final = a_best
    
    # Intensive CD
    for cd_pass in range(20):
        if time.monotonic() > deadline - 10:
            break
        a_final, new_J = coordinate_descent_pass_fast(a_final, max_iter=3)
        a_final = np.maximum(a_final, 0.0)
        a_final /= a_final.sum()
        if new_J < tracker.best_score:
            tracker.update(a_final)
    
    # Final Adam with high tau
    if time.monotonic() < deadline - 5:
        x_final = np.log(np.maximum(a_final, 1e-6))
        x_final, _ = adam_optimize_adaptive(x_final, n_steps=2000,
                                             deadline=deadline - 5,
                                             tracker=tracker, label="final")
        a_final = stable_softmax(x_final)
        a_final = np.maximum(a_final, 0.001)
        a_final /= a_final.sum()
        tracker.update(a_final)
    
    # One more CD
    if time.monotonic() < deadline - 3:
        a_final, _ = coordinate_descent_pass_fast(a_final, max_iter=2)
        a_final = np.maximum(a_final, 0.001)
        a_final /= a_final.sum()
        tracker.update(a_final)
    
    tracker.report(f"Phase 7 Final: {time.monotonic()-t0:.1f}s")
    
    # Fallback
    if tracker.best is None:
        tracker.best = np.ones(256)
        tracker.best_score = compute_J(tracker.best)
    
    print(f"\n{'='*60}", flush=True)
    print(f"FINAL RESULT", flush=True)
    print(f"{'='*60}", flush=True)
    print(f"J = {tracker.best_score:.10f}, n = {tracker.best.size}", flush=True)
    print(f"Total time: {time.monotonic()-tracker.t_start:.1f}s", flush=True)
    
    return [float(x) for x in tracker.best]
