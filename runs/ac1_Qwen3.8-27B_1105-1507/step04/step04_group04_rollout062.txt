"""
Step function optimization using:
1. L-BFGS-B with power-mean smooth approximation to the max
2. Alternating LP for direction finding
3. Multi-start local search with adaptive n
4. Nesterov-accelerated subgradient in log-domain
"""

import numpy as np
from scipy.optimize import minimize, linprog
import time

def compute_conv(a):
    """Compute conv(a,a) using FFT. Returns array of length 2n-1."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    conv = np.fft.irfft(fa * fa, size)
    return conv[:2 * n - 1]

def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    conv = compute_conv(a)
    max_b = float(np.max(conv))
    sum_a = float(np.sum(a))
    if sum_a < 0.01:
        return np.inf
    return 2.0 * n * max_b / (sum_a * sum_a)

def compute_obj_smooth(a, p=8):
    """Compute smooth objective using power mean."""
    n = len(a)
    conv = compute_conv(a)
    sum_a = float(np.sum(a))
    if sum_a < 0.01:
        return np.inf
    # Power mean of order p
    m_p = float(np.sum(conv**p) ** (1.0/p))
    return 2.0 * n * m_p / (sum_a * sum_a)

def compute_grad_smooth(a, p=8):
    """Compute gradient of smooth objective."""
    n = len(a)
    conv = compute_conv(a)
    sum_a = float(np.sum(a))
    if sum_a < 0.01:
        return np.zeros(n)
    
    # m_p = (sum(conv^p))^(1/p)
    conv_p = conv**p
    sum_conv_p = float(np.sum(conv_p))
    m_p = sum_conv_p ** (1.0/p)
    
    # d(m_p)/d(conv_k) = (1/p) * (sum(conv^p))^(1/p - 1) * p * conv_k^(p-1)
    #                    = (sum(conv^p))^((1-p)/p) * conv_k^(p-1)
    dmp_dconv = sum_conv_p ** ((1.0 - p) / p) * conv ** (p - 1)
    
    # d(conv_k)/d(a_j) = 2*a[k-j] if 0 <= k-j < n
    # So d(m_p)/d(a_j) = sum over k of dmp_dconv[k] * 2*a[k-j] (where valid)
    # This is 2 * correlate(dmp_dconv, a) [in the right range]
    
    # d(m_p)/d(a) = 2 * (dmp_dconv * a) [full correlation, take middle n]
    c = np.correlate(dmp_dconv, a, mode='full')
    grad_mp = 2.0 * c[n-1 : n-1+n]
    
    # Full objective: f = 2n * m_p / S^2
    # df/da_j = 2n/S^2 * dmp/da_j - 2 * 2n * m_p / S^3 * 1
    grad = (2.0 * n / (sum_a**2)) * grad_mp - (4.0 * n * m_p / (sum_a**3)) * np.ones(n)
    
    return grad

def build_toeplitz(a):
    """Build Toeplitz matrix A where (A@g)[k] = (a*g)[k] for k=0..2n-2."""
    n = len(a)
    A = np.zeros((2 * n - 1, n), dtype=np.float64)
    for j in range(n):
        k_start = j
        k_end = min(j + n, 2 * n - 1)
        A[k_start:k_end, j] = a[:k_end - k_start]
    return A

def solve_lp_maxsum(a, M, time_limit=5):
    """Solve LP: max sum(g) s.t. (a*g)[k] <= M for all k, g >= 0."""
    n = len(a)

def lbfgs_optimize(a, time_budget=30.0, p=8, verbose=False):
    """Optimize using L-BFGS-B with smooth power-mean objective."""
    n = len(a)
    a = a.copy().astype(np.float64)
    s = np.sum(a)
    if s < 1e-10:
        return a
    a = a / s

    start = time.monotonic()
    
    def obj_func(x):
        if time.monotonic() - start > time_budget:
            return 1e10  # Signal to stop
        x = np.maximum(x, 0)
        return compute_obj_smooth(x, p=p)
    
    def grad_func(x):
        x = np.maximum(x, 0)
        return compute_grad_smooth(x, p=p)
    
    bounds = [(0.0, 1000.0)] * n
    
    try:
        res = minimize(obj_func, a, method='L-BFGS-B', jac=grad_func,
                      bounds=bounds, options={'maxiter': 200, 'maxfun': 500,
                                             'gtol': 1e-8, 'ftol': 1e-10})
        if res.x is not None:
            a_new = np.maximum(res.x, 0)
            s_new = np.sum(a_new)

                obj_new = compute_obj(a_new)
                if obj_new < best_obj:
                    best_obj = obj_new
                    best_a = a_new.copy()
    except Exception as e:

def log_domain_subgradient(a, time_budget=20.0, tau=20.0, step_init=0.1, verbose=False):
    """Subgradient method in log-domain with Nesterov momentum."""
    n = len(a)
    a = a.copy().astype(np.float64)
    s = np.sum(a)
    if s < 1e-10:
        return a
    a = a / s
    
    # Work in log-domain
    x = np.log(a)  # x_i = log(a_i)
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    
    step = step_init
    momentum = 0.9
    y = x.copy()  # Nesterov's lookahead point
    v = np.zeros(n)  # momentum vector
    
    stall = 0
    patience = 50
    
    for it in range(10000):
        if time.monotonic() - start > time_budget:
            break
        
        # Cost update (Nesterov)
        x = y - momentum * v
        v = y - x  # (this is the standard Nesterov update)
        y = x + momentum * v  # wait, this is getting complicated
        
        # Actually let me use a simpler accelerated gradient
        # x_{k+1} = y_k - step * grad(f)(y_k)
        # y_{k+1} = x_{k+1} + ((t_k - 1)/(t_{k+1})) * (x_{k+1} - x_k)
        # where t_k = (1 + sqrt(1+4*t_{k-1}^2))/2
        
        # Let me just do plain subgradient with momentum for simplicity
        a_y = np.exp(y)
        a_y = np.maximum(a_y, 1e-15)
        s_y = np.sum(a_y)
        if s_y < 1e-10:
            break
        a_y = a_y / s_y
        
        conv = compute_conv(a_y)
        max_conv = np.max(conv)
        
        # Subgradient of max(conv) w.r.t. a (at the peak)
        peak_idx = int(np.argmax(conv))
        # grad[j] = 2 * a[peak_idx - j] if 0 <= peak_idx - j < n
        grad = np.zeros(n)
        for j in range(n):
            k = peak_idx - j
            if 0 <= k < n:
                grad[j] = 2.0 * a_y[k]
        
        # Gradient of full objective f = 2n*max/(S^2)
        # df/da_j = 2n/S^2 * grad_j - 4n*max/S^3
        obj_grad = (2.0 * n / (s_y**2)) * grad - (4.0 * n * max_conv / (s_y**3))
        
        # Transform to log-domain gradient
        # d(dx)/da * da/dx = obj_grad * a_y (chain rule: df/dx_j = df/da_j * da_j/dx_j = df/da_j * a_j)
        log_grad = obj_grad * a_y
        
        # Center the gradient (to account for scale invariance)
        log_grad -= np.mean(log_grad)
        
        # Momentum update
        v = momentum * v - step * log_grad
        y = y + v
        
        # Evaluate
        a_new = np.exp(y)
        a_new = np.maximum(a_new, 1e-15)
        s_new = np.sum(a_new)
        if s_new > 1e-10:
            a_new = a_new / s_new
        
        obj_new = compute_obj(a_new)
        if obj_new < best_obj - 1e-14:
            best_obj = obj_new
            best_a = a_new.copy()
            stall = 0
        else:
            stall += 1
            if stall > patience:
                step *= 0.7
                if step < 0.001:
                    break
                # Restart momentum
                v = np.zeros(n)
                y = np.log(np.maximum(best_a, 1e-15))
                stall = 0
                patience = max(patience // 2, 20)
    
    return best_a

def fine_local_search(a, time_budget=30.0, verbose=False):
    """Intensive fine local search with coordinate-wise and block perturbations."""
    n = len(a)
    a = a.copy().astype(np.float64)

        return a, compute_obj(a)

        attempt += 1
        remaining = time_budget - (time.monotonic() - start)
        if remaining < 5:
            break
        
        # Different perturbation strategies
        strategy = attempt % 5
        
        if strategy == 0:
            # Small global perturbation
            a_pert = best_a * np.exp(rng.normal(0, 0.01, n))

            # Block perturbation
            block_size = max(1, n // 10)
            a_pert = best_a.copy()
            start_idx = rng.integers(0, n - block_size)
            a_pert[start_idx:start_idx+block_size] *= np.exp(rng.normal(0, 0.03, block_size))
        elif strategy == 2:
            # Coordinate perturbation
            a_pert = best_a.copy()
            idx = rng.integers(0, n)
            a_pert[idx] *= np.exp(rng.normal(0, 0.05))
        elif strategy == 3:
            # Anti-symmetric perturbation (exploits reversal symmetry)
            a_pert = best_a.copy()
            delta = rng.normal(0, 0.01 * best_a)
            for i in range(n):
                a_pert[i] += delta[i]

            fa = np.fft.rfft(a_pert)

def resample_sequence(a, new_n):
    """Resample a sequence to a new length."""
    n = len(a)
    if n == new_n:
        return a.copy()
    old_k = np.arange(n)
    new_k = np.linspace(0, n - 1, new_n)
    a_new = np.interp(new_k, old_k, a)
    a_new = np.maximum(a_new, 0)
    s = np.sum(a_new)
    if s > 1e-10:
        a_new = a_new / s
    return a_new

"""
Step function optimization using:
1. L-BFGS-B with power-mean smooth approximation to the max convolution peak
2. Alternating LP for direction finding
3. Log-domain subgradient with Nesterov momentum
4. Multi-start local search with diverse perturbation strategies
5. Adaptive n exploration around the parent length

Key innovation: Using scipy L-BFGS-B on a smooth power-mean surrogate
objective provides quasi-Newton convergence that pure subgradient methods
lack, combined with LP-based block coordinate updates for global moves.

import numpy as np
from scipy.optimize import minimize, linprog
import time


def compute_conv(a):
    """Compute conv(a,a) using FFT. Returns array of length 2n-1."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    conv = np.fft.irfft(fa * fa, size)
    return conv[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    conv = compute_conv(a)
    max_b = float(np.max(np.abs(conv)))
    sum_a = float(np.sum(a))
    if sum_a < 0.01:
        return np.inf
    return 2.0 * n * max_b / (sum_a * sum_a)


def compute_obj_smooth(a, p=8.0):
    """Compute smooth objective using power mean of order p."""
    n = len(a)
    if n == 0:
        return np.inf
    conv = compute_conv(a)
    conv = np.maximum(conv, 0.0)
    sum_a = float(np.sum(a))
    if sum_a < 0.01:
        return np.inf
    sum_conv_p = float(np.sum(conv ** p))
    if sum_conv_p < 1e-300:
        return 0.0
    m_p = sum_conv_p ** (1.0 / p)
    return 2.0 * n * m_p / (sum_a * sum_a)


def compute_grad_smooth(a, p=8.0):
    """Compute gradient of smooth power-mean objective."""
    n = len(a)
    conv = compute_conv(a)
    conv = np.maximum(conv, 0.0)
    sum_a = float(np.sum(a))
    if sum_a < 0.01:
        return np.zeros(n)

    conv_p = conv ** p
    sum_conv_p = float(np.sum(conv_p))
    if sum_conv_p < 1e-300:
        return np.zeros(n)
    m_p = sum_conv_p ** (1.0 / p)

    # d(m_p)/d(conv_k) = (sum_conv_p)^((1-p)/p) * conv_k^(p-1)
    dmp_dconv = sum_conv_p ** ((1.0 - p) / p) * (conv ** (p - 1))

    # d(m_p)/d(a_j) = 2 * sum_k dmp_dconv[k] * a[k-j] (for valid k-j)
    # This is 2 * cross-correlation of dmp_dconv with a
    c = np.correlate(dmp_dconv, a, mode='full')
    grad_mp = 2.0 * c[n - 1:n - 1 + n]

    # Full gradient: f = 2n * m_p / S^2
    # df/da_j = 2n/S^2 * dmp/da_j - 4n*m_p/S^3
    grad = (2.0 * n / (sum_a ** 2)) * grad_mp - (4.0 * n * m_p / (sum_a ** 3)) * np.ones(n)

    return grad


def build_toeplitz(a):
    """Build Toeplitz matrix A where (A@g)[k] = (a*g)[k] for k=0..2n-2."""
    n = len(a)
    A = np.zeros((2 * n - 1, n), dtype=np.float64)
    for j in range(n):
        k_start = j
        k_end = min(j + n, 2 * n - 1)
        A[k_start:k_end, j] = a[:k_end - k_start]
    return A


def solve_lp_maxsum(a, M, time_limit=5):
    """Solve LP: max sum(g) s.t. (a*g)[k] <= M for all k, g >= 0."""
    n = len(a)
    A = build_toeplitz(a)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': time_limit})
        if res.status == 0:
            g = np.maximum(res.x, 0)
            return g, float(np.sum(g))
    except Exception:
        pass
    return None, None


def solve_lp_minpeak(a, time_limit=5):
    """Solve LP: min M s.t. (a*g)[k] <= M, sum(g)=1, g>=0."""
    n = len(a)
    A = build_toeplitz(a)
    c = np.zeros(n + 1)
    c[-1] = 1.0
    A_ub = np.zeros((2 * n - 1, n + 1))
    A_ub[:, :n] = A
    A_ub[:, n] = -1.0
    b_ub = np.zeros(2 * n - 1)
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])
    bounds = [(0.0, None)] * n + [(0.0, None)]
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': time_limit})
        if res.status == 0:
            g = np.maximum(res.x[:n], 0)
            s = np.sum(g)
            if s < 1e-10:
                return None, None
            return g / s, float(res.x[n])
    except Exception:
        pass
    return None, None


def lbfgs_optimize(a, time_budget=30.0, p=8.0, verbose=False):
    """Optimize using L-BFGS-B with smooth power-mean objective."""
    n = len(a)
    a = a.copy().astype(np.float64)
    s = np.sum(a)
    if s < 1e-10:
        return a.copy()
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    def obj_func(x):
        if time.monotonic() - start > time_budget:
            return 1e10
        x = np.maximum(x, 0.0)
        return compute_obj_smooth(x, p=p)

    def grad_func(x):
        x = np.maximum(x, 0.0)
        return compute_grad_smooth(x, p=p)

    # Add a tiny regularization to break scale invariance and help convergence
    def obj_func_reg(x):
        if time.monotonic() - start > time_budget:
            return 1e10
        x = np.maximum(x, 0.0)
        s = np.sum(x)
        if s < 0.01:
            return 1e10
        return compute_obj_smooth(x, p=p) + 1e-10 * (s - 1.0) ** 2

    def grad_func_reg(x):
        x = np.maximum(x, 0.0)
        g = compute_grad_smooth(x, p=p)
        s = np.sum(x)
        g = g + 2e-10 * (s - 1.0) * np.ones(n)
        return g

    bounds = [(1e-10, 1000.0)] * n

    try:
        res = minimize(obj_func_reg, a, method='L-BFGS-B', jac=grad_func_reg,
                       bounds=bounds, options={'maxiter': 300, 'maxfun': 1000,
                                               'gtol': 1e-9, 'ftol': 1e-12,
                                               'maxls': 40})
        if res.x is not None:
            a_new = np.maximum(res.x, 0.0)
            s_new = np.sum(a_new)
            if s_new > 0.01:
                a_new = a_new / s_new
                obj_new = compute_obj(a_new)
                if obj_new < best_obj:
                    best_obj = obj_new
                    best_a = a_new.copy()
    except Exception:
        pass

    return best_a


def log_domain_subgradient(a, time_budget=20.0, step_init=0.05, verbose=False):
    """Subgradient method in log-domain with momentum."""
    n = len(a)
    a = a.copy().astype(np.float64)
    s = np.sum(a)
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    step = step_init
    momentum = 0.85
    x = np.log(np.maximum(a, 1e-15))
    v = np.zeros(n)

    stall = 0
    patience = 60

    for it in range(20000):
        if time.monotonic() - start > time_budget:
            break

        a_curr = np.exp(x)
        a_curr = np.maximum(a_curr, 1e-15)
        s_curr = np.sum(a_curr)
        if s_curr < 1e-10:
            break
        a_curr = a_curr / s_curr

        conv = compute_conv(a_curr)
        max_conv = float(np.max(conv))
        peak_idx = int(np.argmax(conv))

        # Subgradient of max(conv) w.r.t. a
        grad = np.zeros(n)
        valid_start = max(0, peak_idx - (n - 1))
        valid_end = min(n - 1, peak_idx)
        for j in range(valid_start, valid_end + 1):
            grad[j] = 2.0 * a_curr[peak_idx - j]

        # Full objective gradient
        obj_grad = (2.0 * n / (s_curr ** 2)) * grad - (4.0 * n * max_conv / (s_curr ** 3))

        # Transform to log-domain: df/dx_j = df/da_j * a_j
        log_grad = obj_grad * a_curr
        log_grad -= np.mean(log_grad)

        # Momentum update
        v = momentum * v - step * log_grad
        x = x + v
        x = np.clip(x, -30, 30)

        a_new = np.exp(x)
        a_new = np.maximum(a_new, 1e-15)
        s_new = np.sum(a_new)
        if s_new > 1e-10:
            a_new = a_new / s_new

        obj_new = compute_obj(a_new)
        if obj_new < best_obj - 1e-14:
            best_obj = obj_new
            best_a = a_new.copy()
            stall = 0
        else:
            stall += 1
            if stall > patience:
                step *= 0.7
                v = np.zeros(n)
                if step < 0.001:
                    break
                x = np.log(np.maximum(best_a, 1e-15))
                stall = 0
                patience = max(patience // 2, 25)

    return best_a


def alternating_lp_refined(a, max_iters=60, time_budget=30.0, verbose=False):
    """Alternating LP with better line search and both formulations."""
    n = len(a)
    a = a.copy().astype(np.float64)
    s = np.sum(a)
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    stall = 0

    for it in range(max_iters):
        if time.monotonic() - start > time_budget - 2:
            break

        conv = compute_conv(a)
        M = float(np.max(conv))

        # Alternate between max-sum and min-peak LP
        if it % 3 == 2:
            g, M_opt = solve_lp_minpeak(a, time_limit=4)
            use_g = g
        else:
            g, sum_g = solve_lp_maxsum(a, M, time_limit=4)
            if g is not None and sum_g > 1e-10:
                use_g = g / sum_g
            else:
                use_g = None

        if use_g is None:
            break

        # Line search with more t values
        best_t = 0.0
        best_obj_ls = best_obj
        for t in [0.05, 0.1, 0.2, 0.3, 0.5, 0.7, 1.0]:
            a_new = (1.0 - t) * a + t * use_g
            a_new = np.maximum(a_new, 0)
            s_new = np.sum(a_new)
            if s_new < 1e-10:
                continue
            a_new = a_new / s_new
            obj_new = compute_obj(a_new)
            if obj_new < best_obj_ls:
                best_obj_ls = obj_new
                best_t = t

        if best_t > 0 and best_obj_ls < best_obj - 1e-15:
            a_new = (1.0 - best_t) * a + best_t * use_g
            a_new = np.maximum(a_new, 0)
            s_new = np.sum(a_new)
            if s_new > 1e-10:
                a_new = a_new / s_new
            best_obj = best_obj_ls
            best_a = a_new.copy()
            a = a_new
            stall = 0
        else:
            stall += 1
            if stall > 12:
                break
            a = 0.96 * a + 0.04 * use_g
            a = np.maximum(a, 0)
            s = np.sum(a)
            if s > 1e-10:
                a = a / s

    return best_a


def fine_local_search(a, time_budget=30.0, verbose=False):
    """Intensive fine local search with diverse perturbation strategies."""
    n = len(a)
    a = a.copy().astype(np.float64)
    s = np.sum(a)
    if s < 1e-10:
        return a, compute_obj(a)
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    rng = np.random.default_rng(42)

    attempt = 0
    while time.monotonic() - start < time_budget - 3:
        attempt += 1
        remaining = time_budget - (time.monotonic() - start)
        if remaining < 5:
            break

        strategy = attempt % 6

        if strategy == 0:
            # Small global multiplicative perturbation
            a_pert = best_a * np.exp(rng.normal(0, 0.015, n))
        elif strategy == 1:
            # Block perturbation
            block_size = max(2, n // 8)
            a_pert = best_a.copy()
            start_idx = int(rng.integers(0, n - block_size))
            a_pert[start_idx:start_idx + block_size] *= np.exp(rng.normal(0, 0.04, block_size))
        elif strategy == 2:
            # Pair perturbation (exploits reversal structure)
            a_pert = best_a.copy()
            n_pairs = max(1, n // 6)
            idx1 = rng.integers(0, n, size=n_pairs)
            idx2 = (n - 1 - idx1) % n
            delta = rng.normal(0, 0.02, size=n_pairs)
            for i in range(n_pairs):
                a_pert[idx1[i]] *= np.exp(delta[i])
                a_pert[idx2[i]] *= np.exp(-delta[i])
        elif strategy == 3:
            # Frequency-domain perturbation
            a_pert = best_a.copy()
            fa = np.fft.rfft(a_pert)
            n_pert = max(1, len(fa) // 10)
            freq_idx = rng.integers(0, len(fa), size=n_pert)
            for fi in freq_idx:
                fa[fi] *= np.exp(complex(rng.normal(0, 0.01), rng.normal(0, 0.01)))
            a_pert = np.fft.irfft(fa, n)
            a_pert = np.maximum(a_pert, 0)
        elif strategy == 4:
            # Smooth perturbation (low-pass filtered noise)
            noise = rng.normal(0, 0.01, n)
            # Simple moving average smoothing
            win = max(3, n // 30)
            kernel = np.ones(win) / win
            noise = np.convolve(noise, kernel, mode='same')
            a_pert = best_a + noise
            a_pert = np.maximum(a_pert, 0)
        else:
            # Gaussian kernel perturbation at random center
            a_pert = best_a.copy()
            center = rng.integers(0, n)
            width = max(2, n // 20)
            delta = np.exp(-0.5 * ((np.arange(n) - center) / width) ** 2)
            scale = rng.normal(0, 0.03)
            a_pert = best_a + scale * delta
            a_pert = np.maximum(a_pert, 0)

        s_pert = np.sum(a_pert)
        if s_pert < 1e-10:
            continue
        a_pert = a_pert / s_pert

        # Quick refinement
        sub_budget = min(6.0, remaining * 0.25)
        a_opt = lbfgs_optimize(a_pert, time_budget=sub_budget, p=8.0)
        obj = compute_obj(a_opt)

    return best_a, best_obj


def resample_sequence(a, new_n):
    """Resample a sequence to a new length using interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    old_k = np.arange(n)
    new_k = np.linspace(0, n - 1, new_n)
    a_new = np.interp(new_k, old_k, a)
    a_new = np.maximum(a_new, 0)
    s = np.sum(a_new)
    if s > 1e-10:
        a_new = a_new / s
    return a_new


def random_init(n, seed):
    """Generate diverse random initializations."""
    rng = np.random.default_rng(seed)
    k = np.arange(n)
    choice = int(rng.integers(0, 8))

    if choice == 0:
        f = rng.uniform(0.5, 3.0)
        phase = rng.uniform(0, 2 * np.pi)
        a = 1.0 + 0.7 * np.cos(2 * np.pi * f * k / n + phase)
    elif choice == 1:
        a = 1.0 + 0.3 * np.cos(2 * np.pi * 1.5 * k / n) + 0.2 * np.cos(2 * np.pi * 4.0 * k / n)
    elif choice == 2:
        a = rng.random(n)
        win = max(3, n // 15)
        kernel = np.ones(win) / win
        a = np.convolve(a, kernel, mode='same')
    elif choice == 3:
        a = np.ones(n)
        for _ in range(2):
            f = rng.uniform(0.5, 4.0)
            amp = rng.uniform(0.1, 0.3)
            a += amp * np.cos(2 * np.pi * f * k / n + rng.uniform(0, 2 * np.pi))
    elif choice == 4:
        # Two-bump
        c1 = n * rng.uniform(0.2, 0.4)
        c2 = n * rng.uniform(0.6, 0.8)
        w = n * 0.15
        a = 1.0 + 0.5 * np.exp(-0.5 * ((k - c1) / w) ** 2) + 0.5 * np.exp(-0.5 * ((k - c2) / w) ** 2)
    elif choice == 5:
        # Polynomial
        t = k / (n - 1)
        a = 1.0 + rng.uniform(-0.3, 0.3) * t + rng.uniform(-0.3, 0.3) * t ** 2
    elif choice == 6:
        # Chirp
        t = k / n
        freq = rng.uniform(2, 8)
        a = 1.0 + 0.4 * np.cos(2 * np.pi * freq * t ** 2)
    else:
        a = np.ones(n)
        a[rng.integers(0, n, size=n//5)] *= rng.uniform(0.5, 1.5, size=n//5)

    a = np.maximum(a, 0.0)
    s = np.sum(a)
    if s < 1e-10:
        a = np.ones(n)
        s = n
    return a / s


def optimize_single(a, time_budget=60.0, verbose=False):
    """Full optimization pipeline for a single sequence."""
    n = len(a)
    a = a.copy().astype(np.float64)
    s = np.sum(a)
    if s < 1e-10:
        a = np.ones(n)
        s = n
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    if verbose:
        print(f"    Start: obj={best_obj:.8f} (n={n})")

    # Phase 1: L-BFGS-B with p=4 (smoother, more global)
    remaining = time_budget - (time.monotonic() - start)
    if remaining > 8:
        t_budget = min(10.0, remaining * 0.2)
        a_new = lbfgs_optimize(best_a, time_budget=t_budget, p=4.0)
        obj = compute_obj(a_new)
        if obj < best_obj:
            best_obj = obj
            best_a = a_new.copy()
            if verbose:
                print(f"    L-BFGS p=4: {best_obj:.8f}")

    # Phase 2: L-BFGS-B with p=8
    remaining = time_budget - (time.monotonic() - start)
    if remaining > 8:
        t_budget = min(12.0, remaining * 0.25)
        a_new = lbfgs_optimize(best_a, time_budget=t_budget, p=8.0)
        obj = compute_obj(a_new)
        if obj < best_obj:
            best_obj = obj
            best_a = a_new.copy()
            if verbose:
                print(f"    L-BFGS p=8: {best_obj:.8f}")

    # Phase 3: L-BFGS-B with p=16 (closer to true max)
    remaining = time_budget - (time.monotonic() - start)
    if remaining > 8:
        t_budget = min(12.0, remaining * 0.25)
        a_new = lbfgs_optimize(best_a, time_budget=t_budget, p=16.0)
        obj = compute_obj(a_new)
        if obj < best_obj:
            best_obj = obj
            best_a = a_new.copy()
            if verbose:
                print(f"    L-BFGS p=16: {best_obj:.8f}")

    # Phase 4: Alternating LP
    remaining = time_budget - (time.monotonic() - start)
    if remaining > 10:
        t_budget = min(20.0, remaining * 0.3)
        a_new = alternating_lp_refined(best_a, max_iters=50, time_budget=t_budget)
        obj = compute_obj(a_new)
        if obj < best_obj:
            best_obj = obj
            best_a = a_new.copy()
            if verbose:
                print(f"    LP: {best_obj:.8f}")

    # Phase 5: Log-domain subgradient
    remaining = time_budget - (time.monotonic() - start)
    if remaining > 8:
        t_budget = min(10.0, remaining * 0.2)
        a_new = log_domain_subgradient(best_a, time_budget=t_budget, step_init=0.04)
        obj = compute_obj(a_new)
        if obj < best_obj:
            best_obj = obj
            best_a = a_new.copy()
            if verbose:
                print(f"    SG: {best_obj:.8f}")

    # Phase 6: Another LP round (polish)
    remaining = time_budget - (time.monotonic() - start)
    if remaining > 8:
        t_budget = min(10.0, remaining * 0.25)
        a_new = alternating_lp_refined(best_a, max_iters=30, time_budget=t_budget)
        obj = compute_obj(a_new)
        if obj < best_obj:
            best_obj = obj
            best_a = a_new.copy()
            if verbose:
                print(f"    LP2: {best_obj:.8f}")

    # Phase 7: Final L-BFGS with high p
    remaining = time_budget - (time.monotonic() - start)
    if remaining > 5:
        t_budget = min(8.0, remaining * 0.3)
        a_new = lbfgs_optimize(best_a, time_budget=t_budget, p=32.0)
        obj = compute_obj(a_new)
        if obj < best_obj:
            best_obj = obj
            best_a = a_new.copy()
            if verbose:
                print(f"    L-BFGS p=32: {best_obj:.8f}")

    return best_a, best_obj


def propose_candidate(seed=42, budget_s=1500):
    """
    Main optimization: combines L-BFGS-B with smooth power-mean objectives,
    alternating LP, log-domain subgradient, and multi-start local search.
    Explores n values around the parent and uses diverse perturbation strategies.
    """
    start_time = time.monotonic()
    deadline = start_time + float(budget_s) - 30.0

    best_seq = None
    best_score = np.inf

    # Load parent
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64)
        parent = np.clip(parent, 0, 1000.0)
        if parent.size > 0 and np.isfinite(parent).all():
            parent_obj = compute_obj(parent)
            if parent_obj < best_score:
                best_score = parent_obj
                best_seq = parent.copy()
                print(f"Parent: n={parent.size}, obj={parent_obj:.8f}")
    except Exception:
        pass

    if best_seq is None:
        best_seq = np.ones(330)
        best_score = compute_obj(best_seq)
        print(f"No parent, uniform n=330: obj={best_score:.8f}")

    n_parent = len(best_seq)
    print(f"Time budget: {budget_s}s, effective deadline: {deadline - start_time:.0f}s")

    # === Phase 1: Intensive optimization of parent ===
    print("\n=== Phase 1: Optimize parent ===")
    remaining = deadline - time.monotonic()
    p1_budget = min(remaining * 0.25, 250.0)
    a_opt, obj = optimize_single(best_seq, time_budget=p1_budget, verbose=True)
    if obj < best_score:
        best_score = obj
        best_seq = a_opt.copy()
        print(f"  Phase 1 result: {best_score:.8f}")

    # === Phase 2: Local search on best ===
    remaining = deadline - time.monotonic()
    if remaining > 20:
        print(f"\n=== Phase 2: Local search ({remaining:.0f}s) ===")
        p2_budget = min(remaining * 0.15, 120.0)
        a_opt, obj = fine_local_search(best_seq, time_budget=p2_budget, verbose=True)
        if obj < best_score:
            best_score = obj
            best_seq = a_opt.copy()
            print(f"  Phase 2 result: {best_score:.8f}")

    # === Phase 3: Try nearby n values ===
    print(f"\n=== Phase 3: Nearby n values ===")
    n_candidates = []
    for d in [0, -5, 5, -10, 10, -15, 15, -20, 20, -25, 25, -30, 30, 40, -40]:
        n_val = n_parent + d
        if 200 <= n_val <= 500:
            n_candidates.append(n_val)
    # Add some fixed values
    for n_val in [300, 320, 340, 350, 360, 380]:
        if n_val not in n_candidates and 200 <= n_val <= 500:
            n_candidates.append(n_val)
    n_candidates = sorted(set(n_candidates))

    remaining = deadline - time.monotonic()
    if remaining > 30:
        per_n_budget = remaining / max(len(n_candidates), 1) * 1.0
        for n_val in n_candidates:
            if time.monotonic() > deadline - 30:
                break
            remaining = deadline - time.monotonic()
            if remaining < 20:
                break
            alloc = min(per_n_budget, remaining * 0.25, 70.0)
            alloc = max(alloc, 15.0)

            if n_val == len(best_seq):
                a_start = best_seq.copy()
            else:
                a_start = resample_sequence(best_seq, n_val)

            if verbose := (n_val == n_parent):
                print(f"  n={n_val} (alloc={alloc:.0f}s)")

            a_opt, obj = optimize_single(a_start, time_budget=alloc, verbose=verbose)
            if obj < best_score:
                best_score = obj
                best_seq = a_opt.copy()
                print(f"  ** UPDATE: n={n_val}, obj={best_score:.8f} **")

    # === Phase 4: Intensive refinement of best ===
    if best_seq is not None:
        remaining = deadline - time.monotonic()
        if remaining > 30:
            print(f"\n=== Phase 4: Intensive refinement ({remaining:.0f}s) ===")
            n = len(best_seq)

            # 4a: Long L-BFGS run
            t_budget = min(remaining * 0.2, 60.0)
            a_new = lbfgs_optimize(best_seq, time_budget=t_budget, p=16.0)
            obj = compute_obj(a_new)
            if obj < best_score:
                best_score = obj
                best_seq = a_new.copy()
                print(f"  L-BFGS long p=16: {best_score:.8f}")

            # 4b: LP polish
            remaining = deadline - time.monotonic()
            if remaining > 15:
                t_budget = min(remaining * 0.15, 40.0)
                a_new = alternating_lp_refined(best_seq, max_iters=40, time_budget=t_budget)
                obj = compute_obj(a_new)
                if obj < best_score:
                    best_score = obj
                    best_seq = a_new.copy()
                    print(f"  LP polish: {best_score:.8f}")

            # 4c: Local search
            remaining = deadline - time.monotonic()
            if remaining > 15:
                t_budget = min(remaining * 0.25, 60.0)
                a_opt, obj = fine_local_search(best_seq, time_budget=t_budget, verbose=True)
                if obj < best_score:
                    best_score = obj
                    best_seq = a_opt.copy()
                    print(f"  Local search: {best_score:.8f}")

    # === Phase 5: Multi-start with diverse initializations ===
    if best_seq is not None:
        remaining = deadline - time.monotonic()
        if remaining > 40:
            print(f"\n=== Phase 5: Multi-start ({remaining:.0f}s) ===")
            n = len(best_seq)
            rng = np.random.default_rng(seed + 1000)

            n_restarts = min(6, int(remaining / 35))
            for r in range(n_restarts):
                if time.monotonic() > deadline - 15:
                    break
                remaining = deadline - time.monotonic()
                if remaining < 25:
                    break
                alloc = min(35.0, remaining * 0.35)

                # Different perturbation strategies
                if r == 0:
                    a_start = best_seq * np.exp(rng.normal(0, 0.02, n))
                elif r == 1:
                    a_start = best_seq * np.exp(rng.normal(0, 0.05, n))
                elif r == 2:
                    a_start = 0.7 * best_seq + 0.3 * random_init(n, seed + r * 1000)
                elif r == 3:
                    # Resample and back
                    a_temp = resample_sequence(best_seq, max(50, n + rng.integers(-20, 20)))
                    a_start = resample_sequence(a_temp, n)
                elif r == 4:
                    a_start = random_init(n, seed + 5000 + r)
                else:
                    # Smoothed random
                    a_r = random_init(n, seed + 9000 + r)
                    win = max(3, n // 20)
                    kernel = np.ones(win) / win
                    a_start = np.convolve(a_r, kernel, mode='same')
                    a_start = np.maximum(a_start, 0)
                    s = np.sum(a_start)
                    if s > 0.01:
                        a_start = a_start / s
                    else:
                        a_start = best_seq.copy()

                a_start = np.maximum(a_start, 0)
                s = np.sum(a_start)
                if s > 0.01:
                    a_start = a_start / s

                a_opt, obj = optimize_single(a_start, time_budget=alloc, verbose=False)
                if obj < best_score:
                    best_score = obj
                    best_seq = a_opt.copy()
                    print(f"  Restart {r}: {best_score:.8f}")

    # === Phase 6: Final polish ===
    if best_seq is not None:
        remaining = deadline - time.monotonic()
        if remaining > 10:
            print(f"\n=== Phase 6: Final polish ({remaining:.0f}s) ===")
            # Very short L-BFGS with high p
            t_budget = min(remaining * 0.4, 30.0)
            a_new = lbfgs_optimize(best_seq, time_budget=t_budget, p=32.0)
            obj = compute_obj(a_new)
            if obj < best_score:
                best_score = obj
                best_seq = a_new.copy()
                print(f"  Final L-BFGS: {best_score:.8f}")

            # Quick local search
            remaining = deadline - time.monotonic()
            if remaining > 8:
                t_budget = min(remaining * 0.5, 20.0)
                a_opt, obj = fine_local_search(best_seq, time_budget=t_budget, verbose=False)
                if obj < best_score:
                    best_score = obj
                    best_seq = a_opt.copy()
                    print(f"  Final LS: {best_score:.8f}")

    # Final validation
    if best_seq is None:
        best_seq = np.ones(50)
        best_score = compute_obj(best_seq)

    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    val = evaluate_sequence(result)
    print(f"\n=== FINAL: n={len(best_seq)}, score={val:.8f}, "
          f"internal={best_score:.8f}, time={time.monotonic()-start_time:.1f}s ===")

    return result
