def fast_conv(a):
    """Full self-convolution via FFT."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]


def compute_J(a):
    """Compute J(a) = 2*n*max(conv(a,a))/(sum(a))^2."""
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))


def build_conv_matrix(n, a):
    """Build the convolution matrix A where (A*g)[k] = (a*g)[k]."""
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        j_min = max(0, k - n + 1)
        j_max = min(n - 1, k)
        for j in range(j_min, j_max + 1):
            rows.append(k)
            cols.append(j)
            vals.append(a[k - j])
    A = csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))
    return A

def smooth_max_and_grad(a, tau):
    """Compute smooth max of conv(a,a) and its gradient."""
    n = len(a)
    b = fast_conv(a)
    b_max = b.max()
    exp_b = np.exp(tau * (b - b_max))
    p = exp_b / exp_b.sum()
    
    smooth_val = (b_max + np.log(exp_b.sum())) / tau
    
    # Gradient: d/d a_i [smooth_max] = (1/tau) * sum_k p[k] * d(b[k])/d(a_i)
    # d(b[k])/d(a_i) = a[k-i] + a[k-i] (if k-i valid) ... 
    # Actually b[k] = sum_j a[j]*a[k-j], so db[k]/da[i] = a[k-i] + a[i-(k-i)] = 2*a[k-i] (if 0<=k-i<n)
    # Wait: b[k] = sum_j a[j]*a[k-j], derivative w.r.t a[i]:
    # = a[k-i] (from j=i) + a[i] * (if k-i replaced by i, i.e., k-j=i means j=k-i) 
    # = a[k-i] + a[k-i] = 2*a[k-i] if 0 <= k-i < n
    # But wait, if i = k-i (i.e., k = 2i), then it's just a[i]^2, derivative is 2*a[i].
    # So in general, db[k]/da[i] = 2*a[k-i] if 0 <= k-i < n, else 0.
    # Hmm, that's not right either. Let me be more careful.
    # b[k] = sum_j a[j] * a[k-j]
    # db[k]/da[i] = sum_j [delta(j,i)*a[k-j] + a[j]*delta(k-j,i)]
    # = a[k-i] + a[k-i] (if 0 <= k-i < n, and k-i != i)
    # = 2*a[k-i] (if 0 <= k-i < n and k != 2i)
    # = 2*a[i] (if k = 2i, since then k-i = i, and we get a[i] + a[i] = 2a[i])
    # Actually in all cases where 0 <= k-i < n, db[k]/da[i] = 2*a[k-i].
    
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        pk = p[k]
        if pk < 1e-14:
            continue
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        # For i in [lo, hi]: contribution is 2 * pk * a[k-i]
        # a[k-i] for i=lo..hi is a[k-hi], a[k-hi+1], ..., a[k-lo]
        # which is a[k-hi : k-lo+1]
        grad[lo:hi + 1] += 2.0 * pk * a[k - hi:k - lo + 1][::-1]
    
    return smooth_val, grad, b


def run_smooth_descent(a, max_iters, deadline, tau_start=5.0, tau_end=200.0, 
                       lr_start=0.3, lr_end=0.01):
    """Smooth gradient descent with adaptive tau and learning rate + momentum."""
    a = np.maximum(a.astype(np.float64), 1e-10)
    a = a / a.sum()
    n = len(a)
    
    best_a = a.copy()
    best_M = fast_conv(a).max()
    v = np.zeros(n)
    
    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        
        frac = it / max(1, max_iters)
        tau = tau_start * (tau_end / tau_start) ** (frac ** 0.7)
        lr = lr_start * (lr_end / lr_start) ** frac
        
        _, grad, b = smooth_max_and_grad(a, tau)
        M = b.max()
        
        if M < best_M - 1e-13:
            best_M = M
            best_a = a.copy()
        
        # Heavy ball momentum
        beta = 0.9
        v = beta * v + lr * grad
        a_new = a - v
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s < 1e-10:
            a_new = a.copy()
        else:
            a_new = a_new / s
        a = a_new
    
    return best_a


def slsqp_polish(a, deadline):
    """Polish using SLSQP on the smooth approximation."""
    a = np.maximum(a.astype(np.float64), 1e-10)
    a = a / a.sum()
    n = len(a)
    tau = 500.0  # very sharp approximation
    
    def objective(x):
        ax = np.maximum(x, 0)
        s = ax.sum()
        if s < 1e-8:
            return 1e10
        ax = ax / s
        b = fast_conv(ax)
        b_shifted = b - b.max()
        return float(np.log(np.sum(np.exp(tau * b_shifted))) / tau)
    
    def gradient(x):
        ax = np.maximum(x, 0)
        s = ax.sum()
        if s < 1e-8:
            return np.zeros_like(x)
        ax = ax / s
        _, grad, _ = smooth_max_and_grad(ax, tau)
        return grad
    
    bounds = [(0.0, None)] * n
    constraints = [{'type': 'eq', 'fun': lambda x: np.sum(x) - 1.0,
                   'jac': lambda x: np.ones(n)}]
    
    try:
        res = minimize(objective, a, jac=gradient, method='SLSQP',
                       bounds=bounds, constraints=constraints,
                       options={'maxiter': 200, 'ftol': 1e-16})
        if res.success:
            return res.x
    except Exception as e:
        print(f"  SLSQP error: {e}", flush=True)
    
    return a


def resample_seq(src, dest_n):
    """Resample 1D sequence to new length."""
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)

def fast_conv(a):
    """Full self-convolution via FFT."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]


def compute_J(a):
    """Compute J(a) = 2*n*max(conv(a,a))/(sum(a))^2."""
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))


def build_conv_matrix(n, a):
    """Build sparse convolution matrix A where (A*g)[k] = sum_j a[k-j]*g[j]."""
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        j_min = max(0, k - n + 1)
        j_max = min(n - 1, k)
        for j in range(j_min, j_max + 1):
            rows.append(k)
            cols.append(j)
            vals.append(a[k - j])
    return csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))


def lp_solve_precise(a, M, time_limit=20):
    """Solve LP: max sum(g) s.t. (a*g)[k] <= M for all k, g >= 0. Returns (g, dual_vars)."""
    n = len(a)
    A = build_conv_matrix(n, a)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    try:
        r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                    options={'time_limit': time_limit,
                             'dual_feasibility_tolerance': 1e-12,
                             'primal_feasibility_tolerance': 1e-12,
                             'presolve': 'on'})
        if r.success and r.x is not None:
            g = np.maximum(r.x, 0)
            dual = r.ineqlin  # dual variables for A_ub constraints
            if g.sum() > 1e-15:
                return g, dual
    except Exception:
        pass
    return None, None


def iterative_lp_solve(a, M, max_iters=5000, tol=1e-10):
    """
    Iterative subgradient method to solve LP: max sum(g) s.t. A*g <= M, g >= 0.
    Uses the dual: min M*sum(lambda) s.t. A^T*lambda >= 1, lambda >= 0.
    Then g* is determined by complementary slackness.
    """
    n = len(a)
    # Solve the dual iteratively: min M*sum(lambda) s.t. A^T*lambda >= 1, lambda >= 0
    # Use projected subgradient on the dual
    lam = np.zeros(2 * n - 1)

    b = np.convolve(a, np.ones(n))
    active = b >= b.max() * 0.5
    lam[active] = 1.0 / np.sum(active)
    
    best_lam = lam.copy()
    best_dual_obj = M * lam.sum()
    
    # Subgradient of M*sum(lambda) is M*1
    # Constraint: A^T * lambda >= 1, violation = 1 - A^T*lambda (where positive)
    for it in range(max_iters):

        # = cross-correlation

            k_max = min(2 * n - 2, i + n - 1)
            if k_min <= k_max:

        violation = np.maximum(1.0 - At_lam, 0)
        if violation.max() < tol:

                At_v[i_min:i_max + 1] += violation[k] * a[k - i_max:k - i_min + 1][::-1]
        
        # Gradient: M*1 - mu * A * (A^T*lambda - 1)_+ ... this is getting complex
        # Let me just use a simple approach: multiplicative weights
        
        # Actually, let's just use the primal subgradient approach
        pass
    
    # Fallback: use a simple feasible solution
    g = np.ones(n)
    Ag = fast_conv(a)  # This is (a*1)[k], not (a*g)[k]... 
    # Actually (a*g)[k] with g=1 is sum_j a[k-j] = the "sliding sum"
    # Let me just return a trivial feasible solution
    g = np.ones(n) / n
    Ag = np.convolve(a, g)[:2*n-1]
    if Ag.max() > M:
        g = g * (M / Ag.max()) * 0.99

def solve_lp_direct(a, M, time_limit=15):
    """
    Try to solve the LP directly. For n <= 1024, use HiGHS.
    Returns (g, success).
    """
    n = len(a)
    g, dual = lp_solve_precise(a, M, time_limit=time_limit)
    if g is not None:
        return g, True
    return None, False


def run_lp_mirror_descent_improved(a, max_iters, deadline, aggressive=True, warm_start=None):
    """
    Improved LP mirror descent with:
    - Warm starting for LP
    - Aggressive M deflation
    - KKT verification
    - Better line search
    """
    a = np.maximum(a.astype(np.float64), 1e-12)
    a = a / a.sum()
    n = len(a)
    
    best_a = a.copy()
    best_M = fast_conv(a).max()
    best_J = 2 * n * best_M  # since sum=1
    no_improve = 0
    consecutive_fail = 0
    
    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        
        b = fast_conv(a)
        M = float(b.max())
        
        if M < best_M - 1e-14:
            best_M = M
            best_a = a.copy()
            no_improve = 0
            consecutive_fail = 0
        else:
            no_improve += 1
        
        if no_improve > 80:
            break
        
        # Try aggressive M deflation
        deflation = 1.0
        if aggressive and it > 5:
            deflation = max(0.95, 1.0 - 0.002 * (it // 10))
        M_target = M * deflation
        
        # Solve LP
        g, success = solve_lp_direct(a, M_target, time_limit=max(5, 20 - it // 50))
        
        if not success and M_target < M:
            # Retry with original M
            g, success = solve_lp_direct(a, M, time_limit=15)
        
        if not success or g is None:
            consecutive_fail += 1
            if consecutive_fail > 10:
                break
            continue
        
        g = np.maximum(g, 0.0)
        g_sum = g.sum()
        if g_sum < 1e-12:
            break
        g = g / g_sum
        
        # Determine step size via line search
        improved = False
        if aggressive:
            candidates = [0.6, 0.4, 0.3, 0.2, 0.15, 0.1, 0.05, 0.02]
        else:
            candidates = [0.4, 0.3, 0.2, 0.1, 0.05, 0.02, 0.01]
        
        for t in candidates:
            a_trial = (1 - t) * a + t * g
            a_trial = np.maximum(a_trial, 0.0)
            s = a_trial.sum()
            if s < 1e-10:
                continue
            a_trial = a_trial / s
            M_trial = fast_conv(a_trial).max()
            if M_trial < best_M * (1 - 1e-13):
                a = a_trial
                best_M = M_trial
                best_a = a.copy()
                improved = True
                break
        
        if not improved and it > 20:
            # Stagnation: try a small perturbation
            a_pert = a + 0.01 * np.random.randn(n)
            a_pert = np.maximum(a_pert, 0)
            a_pert = a_pert / a_pert.sum()
            if fast_conv(a_pert).max() < best_M * 1.01:
                a = a_pert
    
    return best_a


def smooth_max_grad(a, tau):
    """Compute (1/tau)*log(sum exp(tau*b)) and its gradient, where b = conv(a,a)."""
    n = len(a)
    b = fast_conv(a)
    b_max = b.max()
    exp_b = np.exp(tau * (b - b_max))
    p = exp_b / exp_b.sum()
    
    smooth_val = (b_max + np.log(exp_b.sum())) / tau
    
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        pk = p[k]
        if pk < 1e-14:
            continue
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        grad[lo:hi + 1] += 2.0 * pk * a[k - hi:k - lo + 1][::-1]
    
    return smooth_val, grad, b


def run_smooth_descent(a, max_iters, deadline, tau_start=5.0, tau_end=300.0,
                       lr_start=0.25, lr_end=0.005, momentum_beta=0.9):
    """Smooth gradient descent with adaptive parameters and momentum."""
    a = np.maximum(a.astype(np.float64), 1e-12)
    a = a / a.sum()
    n = len(a)
    
    best_a = a.copy()
    best_M = fast_conv(a).max()
    v = np.zeros(n)
    
    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        
        frac = it / max(1, max_iters)
        tau = tau_start * (tau_end / tau_start) ** (frac ** 0.6)
        lr = lr_start * (lr_end / lr_start) ** frac
        
        _, grad, b = smooth_max_grad(a, tau)
        M = b.max()
        
        if M < best_M - 1e-13:
            best_M = M
            best_a = a.copy()
        
        beta = momentum_beta * (1.0 - 0.3 * frac)
        v = beta * v + lr * grad
        a_new = a - v
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s < 1e-10:
            a_new = a.copy()
        else:
            a_new = a_new / s
        a = a_new
    
    return best_a


def slsqp_objective(x, tau):
    """SLSQP objective: smooth max of conv."""
    ax = np.maximum(x, 0.0)
    s = ax.sum()
    if s < 1e-8:
        return 1e10
    ax = ax / s
    b = fast_conv(ax)
    b_shifted = b - b.max()
    return float(np.log(np.sum(np.exp(tau * b_shifted))) / tau)


def slsqp_gradient(x, tau):
    """SLSQP gradient."""
    ax = np.maximum(x, 0.0)
    s = ax.sum()
    if s < 1e-8:
        return np.zeros_like(x)
    ax = ax / s
    _, grad, _ = smooth_max_grad(ax, tau)
    return grad


def slsqp_sum_constraint(x):
    return np.sum(x) - 1.0


def slsqp_sum_gradient(x):
    return np.ones_like(x)


def slsqp_polish(a, tau=500.0, maxiter=300):
    """Polish using SLSQP on smooth approximation."""
    a = np.maximum(a.astype(np.float64), 1e-10)
    a = a / a.sum()
    n = len(a)
    
    bounds = [(0.0, None)] * n
    constraints = [{'type': 'eq', 'fun': slsqp_sum_constraint, 'jac': slsqp_sum_gradient}]
    
    try:
        res = minimize(slsqp_objective, a, args=(tau,),
                       jac=lambda x: slsqp_gradient(x, tau),
                       method='SLSQP', bounds=bounds, constraints=constraints,
                       options={'maxiter': maxiter, 'ftol': 1e-18})
        if res.fun < 1e10:
            return np.maximum(res.x, 0)
    except Exception as e:
        print(f"  SLSQP error: {e}", flush=True)
    
    return a


def deflation_step(a, n_hot=50, epsilon=0.01):
    """
    Deflation-based local search: identify hot convolution coefficients
    and push them down by redistributing mass.
    """
    a = np.maximum(a.astype(np.float64), 1e-12)
    a = a / a.sum()
    n = len(a)
    b = fast_conv(a)
    M = b.max()
    
    # Find hot set
    threshold = M - epsilon * M
    hot = np.where(b >= threshold)[0]
    
    if len(hot) == 0:
        return a
    
    # Compute pressure on each coordinate from hot coefficients
    pressure = np.zeros(n)
    for k in hot:
        w = (b[k] - threshold) / max(M - threshold, 1e-10)
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        pressure[lo:hi + 1] += w * a[k - hi:k - lo + 1][::-1]
    
    # Normalize pressure
    p_max = pressure.max()
    if p_max < 1e-15:
        return a
    
    pressure /= p_max
    
    # Deflation: reduce a where pressure is high, add back where it's low
    eta = 0.05
    a_new = a * (1 - eta * pressure) + eta * a.mean() * pressure
    a_new = np.maximum(a_new, 0)
    s = a_new.sum()
    if s < 1e-10:
        return a
    a_new = a_new / s
    
    return a_new


def resample_seq(src, dest_n):
    """Resample 1D sequence to new length via linear interpolation."""
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)


class ScoreTracker:
    def __init__(self):
        self.best = None
        self.best_score = float('inf')
        self.t_start = time.monotonic()
    
    def update(self, a):
        try:
            a_clip = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
            if a_clip.size == 0 or a_clip.sum() < 0.01:
                return
            score = float(evaluate_sequence(a_clip.tolist()))
            if np.isfinite(score) and score < self.best_score:
                self.best = a_clip.copy()
                self.best_score = score
                elapsed = time.monotonic() - self.t_start
                print(f"  ** NEW BEST: J={score:.8f} n={a_clip.size} t={elapsed:.0f}s", flush=True)
                return True
        except Exception:
            pass
        return False
    
    def report(self, label=""):
        if self.best is not None:
            print(f"  {label} best: J={self.best_score:.8f} n={self.best.size}", flush=True)


def try_length(n, parent, deadline, phase_time):
    """Try to find a good sequence of length n, starting from parent."""
    a0 = resample_seq(parent, n)
    a0 = np.maximum(a0, 1e-8)
    a0 = a0 / a0.sum()
    
    # Smooth gradient descent
    iters = max(100, int(3000 / max(1, n // 64)))
    phase_dl = time.monotonic() + phase_time * 0.6
    a = run_smooth_descent(a0, iters, phase_dl,
                           tau_start=5, tau_end=200,
                           lr_start=0.2, lr_end=0.01)
    
    # LP mirror descent
    remaining = phase_time * 0.4
    if time.monotonic() < deadline - 50 and remaining > 10:
        est_lp_time = 0.004 * (n / 128) ** 2
        max_iters = max(10, int(remaining / max(0.01, est_lp_time)))
        a = run_lp_mirror_descent_improved(a, max_iters,
                                            min(time.monotonic() + remaining, deadline - 50),
                                            aggressive=True)
    
    return a

          f"time={time.monotonic()-tracker.t_start:.0f}s", flush=True)
    print(f"{'='*60}", flush=True)
    
    return [float(x) for x in tracker.best]
