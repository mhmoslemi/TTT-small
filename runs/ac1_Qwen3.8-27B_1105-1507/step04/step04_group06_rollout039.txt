import numpy as np
from scipy.optimize import linprog, minimize, minimize_scalar
from scipy.sparse import csr_matrix
import time

_A_structure_cache = {}
_N_fft_cache = {}


def _get_fft_size(n_total):
    if n_total not in _N_fft_cache:
        s = 1
        while s < n_total:
            s <<= 1
        _N_fft_cache[n_total] = s
    return _N_fft_cache[n_total]


def fast_conv(a):
    n = len(a)
    s = _get_fft_size(2 * n - 1)
    fa = np.fft.rfft(a, s)
    return np.fft.irfft(fa * fa, s)[:2 * n - 1]


def compute_J(a):
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))


def get_A_structure(n):
    if n not in _A_structure_cache:
        rows = []
        cols = []
        for k in range(2 * n - 1):
            jl = max(0, k - n + 1)
            jh = min(n - 1, k)
            for j in range(jl, jh + 1):
                rows.append(k)
                cols.append(j)
        _A_structure_cache[n] = (np.array(rows, dtype=np.int32),
                                  np.array(cols, dtype=np.int32))
    return _A_structure_cache[n]


def build_A_sparse(a):
    n = len(a)
    rows_arr, cols_arr = get_A_structure(n)
    vals = a[rows_arr - cols_arr]
    return csr_matrix((vals, (rows_arr, cols_arr)), shape=(2 * n - 1, n))


def lp_solve(a, M, x0=None):
    n = len(a)
    c = -np.ones(n)
    A = build_A_sparse(a)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    opts = {'presolve': True, 'time_limit': 5.0}
    if x0 is not None:
        opts['pivoting'] = 'devolve'
    r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                options=opts)
    if r.success and r.x is not None:
        g = np.maximum(r.x, 0.0)
        gs = g.sum()
        if gs > 1e-10:
            return g / gs
    return None


def smooth_max_and_grad(a, beta):
    n = len(a)
    b = fast_conv(a)
    b_max = float(b.max())
    shifted = beta * (b - b_max)
    shifted = np.clip(shifted, -500.0, 500.0)
    exp_s = np.exp(shifted)
    Z = float(exp_s.sum())
    if Z < 1e-300:
        p = np.zeros(2 * n - 1)
        p[np.argmax(b)] = 1.0
    else:
        p = exp_s / Z
    g_val = (b_max + np.log(max(Z, 1e-300))) / beta

    # Compute gradient: grad[m] = 2 * sum_j a[j] * p[m+j]
    N = _get_fft_size(3 * n - 2)
    A_fft = np.fft.rfft(a, N)
    P_fft = np.fft.rfft(p, N)
    R = np.fft.irfft(np.conj(A_fft) * P_fft, N)
    grad = 2.0 * R[:n]
    return g_val, grad, p, b


def F_and_grad(a, beta):
    n = len(a)
    S = float(a.sum())
    if S < 1e-10:
        return 1e10, np.zeros(n)
    g_val, grad_g, p, b = smooth_max_and_grad(a, beta)
    F_val = 2.0 * n * g_val / (S * S)
    # dF/da[j] = 2n * (S * grad_g[j] - 2 * g_val) / S^3
    grad_F = (2.0 * n / (S ** 3)) * (S * grad_g - 2.0 * g_val)
    return F_val, grad_F


def project_normalize(a):

    s = a.sum()
    if s < 1e-10:
        return np.ones_like(a) / len(a)
    return a / s


def nesterov_smooth_optimize(a0, beta_start, beta_end, deadline, max_iters=50000,
                             tracker=None):
    a = np.maximum(a0.astype(np.float64), 1e-8)
    a = a / a.sum()
    n = len(a)
    y = a.copy()
    v = np.zeros(n)
    lr = 0.01
    momentum = 0.9
    best_J = compute_J(a)
    best_a = a.copy()
    if tracker is not None:
        tracker.update(best_a)
    stall = 0
    n_iter = 0

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        # Anneal beta
        progress = min(1.0, it / max(1, max_iters * 0.8))
        beta = beta_start * (beta_end / beta_start) ** progress

        F_y, grad_y = F_and_grad(y, beta)
        if not np.isfinite(F_y):
            break

        # Backtracking line search
        accepted = False
        for _bt in range(30):
            a_new = y - lr * grad_y
            # Nesterov update
            v = momentum * v + (a_new - y)
            y_new = a_new + v
            y_new = np.maximum(y_new, 0.0)
            s_new = y_new.sum()
            if s_new < 1e-6:
                lr *= 0.5
                continue
            y_new = y_new / s_new

            F_new = compute_J(y_new)
            if F_new <= F_y + lr * 0.5 * np.dot(grad_y, grad_y) + 1e-12:
                a = y_new
                y = y_new.copy()
                accepted = True
                break
            lr *= 0.5

        if not accepted:
            # Try smaller momentum
            v = 0.5 * v
            a_new = y - lr * grad_y
            a_new = np.maximum(a_new, 0.0)
            s_new = a_new.sum()
            if s_new > 1e-6:
                a_new = a_new / s_new
                F_new = compute_J(a_new)
                if F_new < F_y:
                    y = a_new
                    a = a_new
                    accepted = True

        if not accepted:
            stall += 1
            if stall > 200:
                # Reset
                y = a.copy()
                v = np.zeros(n)
                lr = 0.005
                momentum = 0.8
                stall = 0
            continue

        a = y
        stall = 0
        n_iter += 1

        F_a = compute_J(a)
        if F_a < best_J - 1e-10:
            best_J = F_a
            best_a = a.copy()
            if tracker is not None and n_iter % 100 == 0:
                tracker.update(best_a)

        # Slowly increase momentum
        momentum = min(0.99, 0.9 + 0.09 * (it / max(1, max_iters)))

    if tracker is not None:
        tracker.update(best_a)
    return best_a


def coordinate_descent(a, deadline, tracker=None, max_passes=50):
    a = np.maximum(a.astype(np.float64), 1e-8)
    a = a / a.sum()
    n = len(a)
    best_J = compute_J(a)
    best_a = a.copy()
    if tracker is not None:
        tracker.update(best_a)

    for p in range(max_passes):
        if time.monotonic() > deadline:
            break
        improved = False
        # Randomize coordinate order
        order = np.random.permutation(n)
        for j in order:
            if time.monotonic() > deadline:
                break
            # Save a[j], try to optimize it
            a_save = a[j]
            others = a.copy()
            others[j] = 0.0

            # Compute conv for a with a[j]=0
            b_base = fast_conv(others)
            # The full conv at coordinate j:
            # b[k] = b_base[k] + a[j]*others[k-j] + a[j]*others[j-k] + a[j]^2 (if k=2j valid)
            # Actually: b[k] = sum_i a[i]*a[k-i]
            # With a[j] = x: b[k] = b_base[k] + x*(others[k-j] + others[j-k]) + x^2*I(2j==k)

            def obj_1d(x):
                x = max(0.0, x)
                # Compute max of convolution
                # b[k] = b_base[k] + x*(others[k-j] + others[j-k]) + x^2*I(k==2j)
                b_full = b_base.copy()
                # Add contributions
                for k in range(max(0, j), min(2 * n - 1, 2 * j + 1)):
                    pass  # Too slow for large n, use vectorized approach below

                # Vectorized:
                # others[k-j] for valid k-j: k in [j, j+n-1], value = others[k-j]
                # others[j-k] for valid j-k: k in [max(0,j-n+1), j], value = others[j-k]
                contrib1 = np.zeros(2 * n - 1)
                lo1 = j
                hi1 = min(j + n - 1, 2 * n - 2)
                if lo1 <= hi1:
                    contrib1[lo1:hi1 + 1] = others[0:hi1 - lo1 + 1]
                contrib2 = np.zeros(2 * n - 1)
                lo2 = max(0, j - n + 1)
                hi2 = j
                if lo2 <= hi2:
                    contrib2[lo2:hi2 + 1] = others[n - 1 - (hi2 - lo2):n - 1 + 1][::-1]
                    # others[j-k] for k from lo2 to hi2: j-k from j-lo2 to j-hi2 = 0
                    # j-lo2 = j - max(0, j-n+1), j-hi2 = j-j = 0
                    # So others[j-k] = others[0..j-lo2]

                # Simpler: just compute directly
                b_lin = x * (contrib1 + contrib2)
                b_quad = x * x * (1.0 if 2 * j < 2 * n - 1 else 0.0)
                # Actually the quadratic term: when i=j and k-i=j, so k=2j
                # b[2j] gets an extra x^2
                b_new = b_base + b_lin
                if 2 * j < 2 * n - 1:
                    b_new[2 * j] += b_quad
                return float(b_new.max())

            # Golden section search in [0, 5*a_save] or [0, 1]
            lo = 0.0
            hi = max(2.0 * a_save, 1.0 / n * 5.0)

            gr = (np.sqrt(5.0) - 1.0) / 2.0
            c = hi - gr * (hi - lo)
            d = lo + gr * (hi - lo)
            fc = obj_1d(c)
            fd = obj_1d(d)

            for _gs in range(40):
                if fc < fd:
                    hi = d
                    d = c
                    fd = fc
                    c = hi - gr * (hi - lo)
                    fc = obj_1d(c)
                else:
                    lo = c
                    c = d
                    fc = fd
                    d = lo + gr * (hi - lo)
                    fd = obj_1d(d)

            best_x = (lo + hi) / 2.0
            fx_best = obj_1d(best_x)
            f_save = obj_1d(a_save)

            if fx_best < f_save - 1e-12:
                a[j] = best_x
                improved = True
                # Renormalize
                a = a / a.sum()

        J_now = compute_J(a)
        if J_now < best_J:
            best_J = J_now
            best_a = a.copy()
            if tracker is not None:
                tracker.update(best_a)
        if not improved:
            break

    return best_a


def lp_mirror_descent(a, max_iters, deadline, tracker=None, n_line_evals=20,
                      stall_patience=30, x0_lp=None):
    a = np.maximum(a.astype(np.float64), 1e-8)
    a = a / a.sum()
    current_J = compute_J(a)
    best_J = current_J
    best_a = a.copy()
    if tracker is not None:
        tracker.update(best_a)

    stall_count = 0
    prev_g = x0_lp

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        b = fast_conv(a)
        M = float(b.max())

        g = lp_solve(a, M, x0=prev_g)
        if g is None:
            # Try with relaxed M
            g = lp_solve(a, M * 1.01)
            if g is None:
                break
        prev_g = g.copy()

        # Line search: parabolic interpolation after golden section
        a_0 = a.copy()
        a_1 = g.copy()
        f_0 = current_J
        f_1 = compute_J(a_1)

        if f_1 >= f_0:
            # g direction doesn't help, try small step
            best_t = 0.0
            best_f = f_0
            for t_try in [0.01, 0.005, 0.02, 0.05, 0.1]:
                a_try = (1.0 - t_try) * a_0 + t_try * a_1
                a_try = np.maximum(a_try, 0.0)
                s_try = a_try.sum()
                if s_try < 1e-6:
                    continue
                a_try = a_try / s_try
                f_try = compute_J(a_try)
                if f_try < best_f:
                    best_f = f_try
                    best_t = t_try
            if best_t > 0:
                a = (1.0 - best_t) * a_0 + best_t * a_1
                a = np.maximum(a, 0.0)
                a = a / a.sum()
                current_J = best_f
                if current_J < best_J:
                    best_J = current_J
                    best_a = a.copy()
                    if tracker is not None:
                        tracker.update(best_a)
                stall_count = 0
            else:
                stall_count += 1
                if stall_count >= stall_patience:
                    break
            continue

        # f_1 < f_0: do proper line search
        best_t = 1.0
        best_f = f_1

        # Golden section to bracket
        lo = 0.0
        hi = 1.0
        gr = (np.sqrt(5.0) - 1.0) / 2.0
        t1 = hi - gr * (hi - lo)
        t2 = lo + gr * (hi - lo)
        a_t1 = (1.0 - t1) * a_0 + t1 * a_1
        a_t1 = np.maximum(a_t1, 0.0)
        a_t1 = a_t1 / max(a_t1.sum(), 1e-10)
        f_t1 = compute_J(a_t1)
        a_t2 = (1.0 - t2) * a_0 + t2 * a_1
        a_t2 = np.maximum(a_t2, 0.0)
        a_t2 = a_t2 / max(a_t2.sum(), 1e-10)
        f_t2 = compute_J(a_t2)

        for _ in range(n_line_evals - 2):
            if f_t1 < f_t2:
                hi = t2
                t2 = t1
                f_t2 = f_t1
                t1 = hi - gr * (hi - lo)
            else:
                lo = t1
                t1 = t2
                f_t1 = f_t2
                t2 = lo + gr * (hi - lo)
            a_ti = (1.0 - t1) * a_0 + t1 * a_1
            a_ti = np.maximum(a_ti, 0.0)
            a_ti = a_ti / max(a_ti.sum(), 1e-10)
            f_ti = compute_J(a_ti)
            if f_ti < best_f:
                best_f = f_ti
                best_t = t1

        # Parabolic interpolation for final refinement
        if f_t1 != f_t2 and t1 != t2:
            # Use the three best points for parabolic fit
            tmid = (lo + hi) / 2.0
            a_tm = (1.0 - tmid) * a_0 + tmid * a_1
            a_tm = np.maximum(a_tm, 0.0)
            a_tm = a_tm / max(a_tm.sum(), 1e-10)
            f_tm = compute_J(a_tm)
            if f_tm < best_f:
                best_f = f_tm
                best_t = tmid

        a = (1.0 - best_t) * a_0 + best_t * a_1
        a = np.maximum(a, 1e-10)
        a = a / a.sum()
        current_J = best_f
        stall_count = 0

        if current_J < best_J:
            best_J = current_J
            best_a = a.copy()
            if tracker is not None:
                tracker.update(best_a)

    if tracker is not None:
        tracker.update(best_a)
    return best_a, prev_g


def resample_seq(src, dest_n):
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    result = np.interp(t_new, t_old, src)
    return np.maximum(result, 0.0)


def generate_fourier_init(n, rng, num_terms=8):
    t = np.arange(n) / n
    a = np.ones(n)
    for k in range(1, num_terms + 1):
        amp = rng.uniform(0.01, 0.15)
        freq = rng.integers(k, k + 3)
        phase = rng.uniform(0, 2 * np.pi)
        a += amp * np.cos(2 * np.pi * freq * t + phase)
    return np.maximum(a, 0.01)


def generate_initializations(n, rng, parent=None):
    inits = []
    t_arr = np.arange(n) / n

    if parent is not None and parent.size > 0:
        p = resample_seq(parent, n)
        inits.append(p.copy())
        p2 = p + rng.normal(0, 0.03, n) * p
        inits.append(np.maximum(p2, 0.01))
        p3 = p + rng.normal(0, 0.1, n) * p
        inits.append(np.maximum(p3, 0.01))

    inits.append(np.ones(n).copy())
    inits.append(rng.uniform(0.3, 1.0, n))

    # Fourier families
    for _ in range(4):
        inits.append(generate_fourier_init(n, rng))

    # Cosine families
    for freq in [2, 3, 4, 5, 7, 12, 16, 32, 64]:
        for amp in [0.2, 0.35, 0.5]:
            inits.append(1.0 + amp * np.cos(2.0 * np.pi * freq * t_arr))

    # Gaussian bump
    inits.append(np.exp(-3.0 * (t_arr - 0.5) ** 2) + 0.1)
    inits.append(np.exp(-8.0 * (t_arr - 0.3) ** 2) + np.exp(-8.0 * (t_arr - 0.7) ** 2) + 0.05)

    # Sine families
    inits.append(np.abs(np.sin(np.pi * t_arr)) + 0.2)
    inits.append(np.abs(np.sin(2 * np.pi * t_arr)) + 0.3)
    inits.append(np.cos(2 * np.pi * t_arr) ** 2 + 0.2)

    # Power law
    inits.append((t_arr + 0.05) ** 0.3)
    inits.append((1 - t_arr + 0.05) ** 0.3)

    # Symmetric patterns
    inits.append(1.0 + 0.3 * np.cos(4 * np.pi * t_arr) + 0.1 * np.cos(8 * np.pi * t_arr))
    inits.append(1.0 + 0.2 * np.cos(8 * np.pi * t_arr) - 0.1 * np.cos(16 * np.pi * t_arr))

    return inits


class ScoreTracker:
    def __init__(self):
        self.best = None
        self.best_score = float('inf')
        self.t_start = time.monotonic()
        self.eval_count = 0

    def update(self, a):
        a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
        if a.size == 0 or a.sum() < 0.01:
            return
        self.eval_count += 1
        score = float(evaluate_sequence(a.tolist()))
        if np.isfinite(score) and score < self.best_score:
            self.best = a.copy()
            self.best_score = score
            elapsed = time.monotonic() - self.t_start
            print(f"  * NEW BEST: J={score:.8f} n={a.size} t={elapsed:.1f}s", flush=True)

    def report(self, label=""):
        if self.best is not None:
            print(f"  {label} best: J={self.best_score:.8f} n={self.best.size} evals={self.eval_count}", flush=True)


def lbfgs_refine(a0, deadline, max_beta=50.0):
    a = np.maximum(a0.astype(np.float64), 1e-8)
    a = a / a.sum()
    n = len(a)

    betas = [(2.0, 30), (5.0, 40), (10.0, 50), (20.0, 60), (50.0, 80)]

    for beta, max_iters in betas:
        if time.monotonic() > deadline:
            break

        def func(x):
            x2 = np.maximum(x, 0.0)
            f, _ = F_and_grad(x2, beta)
            if not np.isfinite(f):
                return 1e10
            return f

        def jac(x):
            x2 = np.maximum(x, 0.0)
            _, g = F_and_grad(x2, beta)
            g = np.nan_to_num(g, nan=0.0, posinf=0.0, neginf=0.0)
            return g

        bounds = [(0.0, None)] * n
        try:
            res = minimize(func, a.copy(), jac=jac, method='L-BFGS-B',
                           bounds=bounds,
                           options={'maxiter': max_iters, 'max_fun': max_iters * 30,
                                    'gtol': 1e-8, 'ftol': 1e-14})
            if res.fun < 1e10:
                new_a = np.maximum(res.x, 1e-10)
                if new_a.sum() > 1e-6:
                    a = new_a / new_a.sum()
        except Exception:
            pass

    return a


def propose_candidate(seed=42, budget_s=1500, **kwargs):
    """
    Multi-phase optimization minimizing J(a) = 2*n*max(conv(a,a))/(sum(a))^2.
    Combines Nesterov smoothed gradient, coordinate descent, LP mirror descent,
    Fourier initialization, and multi-restart strategies.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + float(budget_s) - 30.0
    tracker = ScoreTracker()

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        if p.ndim == 1 and p.size > 0 and np.all(np.isfinite(p)):
            parent = np.maximum(p, 0.0)
            tracker.update(parent)
            print(f"Parent loaded: n={parent.size}, J={tracker.best_score:.8f}", flush=True)
    except (NameError, AttributeError):
        pass

    if parent is None:
        parent = np.ones(512)
        tracker.update(parent)
        print(f"Using default parent: n=512, J={tracker.best_score:.8f}", flush=True)

    # === Phase 1: L-BFGS-B initialization (90s) ===
    t0 = time.monotonic()
    phase1_budget = 90.0
    deadline_p1 = time.monotonic() + phase1_budget
    print(f"\n{'=' * 50} Phase 1: L-BFGS-B Init (budget {phase1_budget:.0f}s) {'=' * 50}", flush=True)

    n_main = 1024
    inits = generate_initializations(n_main, rng, tracker.best)
    n_inits = min(len(inits), 8)

    for idx in range(n_inits):
        if time.monotonic() > deadline_p1:
            break
        a0 = inits[idx].copy()
        a0 = np.maximum(a0, 0.01)
        a0 = a0 / a0.sum()
        t_start = time.monotonic()
        a_opt = lbfgs_refine(a0, deadline_p1, max_beta=30.0)
        if a_opt is not None and a_opt.sum() > 0.01:
            tracker.update(a_opt)
        print(f"  L-BFGS-B init {idx+1}/{n_inits}: J={compute_J(a_opt):.6f} "
              f"({time.monotonic()-t_start:.1f}s)", flush=True)

    tracker.report("Phase 1 complete")
    print(f"  Phase 1 took {time.monotonic()-t0:.1f}s", flush=True)

    # === Phase 2: Nesterov accelerated gradient (300s) ===
    if tracker.best is not None and time.monotonic() < deadline:
        t0 = time.monotonic()
        phase2_budget = 300.0
        deadline_p2 = min(deadline, time.monotonic() + phase2_budget)
        n_best = tracker.best.size
        print(f"\n{'=' * 50} Phase 2: Nesterov Gradient n={n_best} (budget {phase2_budget:.0f}s) {'=' * 50}", flush=True)

        # Run Nesterov from best
        a = resample_seq(tracker.best, n_best)
        a = a / a.sum()
        a = nesterov_smooth_optimize(a, beta_start=1.0, beta_end=50.0,
                                     deadline=deadline_p2, max_iters=30000,
                                     tracker=tracker)

        # Also try from a few other initializations
        for init_idx in [0, 3, 5]:
            if time.monotonic() > deadline_p2 or init_idx >= len(inits):
                break
            a0 = inits[init_idx].copy()
            a0 = resample_seq(a0, n_best)
            a0 = np.maximum(a0, 0.01)
            a0 = a0 / a0.sum()
            a = nesterov_smooth_optimize(a0, beta_start=2.0, beta_end=30.0,
                                         deadline=deadline_p2, max_iters=20000,
                                         tracker=tracker)

        tracker.report(f"Phase 2: {time.monotonic()-t0:.1f}s")

    # === Phase 3: Coordinate Descent (200s) ===
    if tracker.best is not None and time.monotonic() < deadline:
        t0 = time.monotonic()
        phase3_budget = 200.0
        deadline_p3 = min(deadline, time.monotonic() + phase3_budget)
        n_best = tracker.best.size
        print(f"\n{'=' * 50} Phase 3: Coordinate Descent n={n_best} (budget {phase3_budget:.0f}s) {'=' * 50}", flush=True)

        a = tracker.best.copy()
        # Run multiple passes with different random seeds for coordinate ordering
        for cd_pass in range(3):
            if time.monotonic() > deadline_p3:
                break
            a = coordinate_descent(a, deadline_p3, tracker=tracker, max_passes=20)

        tracker.report(f"Phase 3: {time.monotonic()-t0:.1f}s")

    # === Phase 4: LP Mirror Descent (600s) ===
    if tracker.best is not None and time.monotonic() < deadline:
        t0 = time.monotonic()
        n_best = tracker.best.size
        remaining = deadline - time.monotonic()
        phase4_budget = min(600.0, remaining)
        deadline_p4 = time.monotonic() + phase4_budget
        print(f"\n{'=' * 50} Phase 4: LP Mirror Descent n={n_best} (budget {phase4_budget:.0f}s) {'=' * 50}", flush=True)

        # Main LP run
        max_iters = int(phase4_budget * 40)
        max_iters = min(max_iters, 3000)
        a, lp_x0 = lp_mirror_descent(tracker.best, max_iters, deadline_p4,
                                     tracker=tracker, n_line_evals=22,
                                     stall_patience=40, x0_lp=None)

        # Second LP pass from result
        if time.monotonic() < deadline_p4:
            remaining2 = deadline_p4 - time.monotonic()
            max_iters2 = int(remaining2 * 40)
            max_iters2 = min(max_iters2, 2000)
            a, lp_x0 = lp_mirror_descent(tracker.best, max_iters2, deadline_p4,
                                         tracker=tracker, n_line_evals=20,
                                         stall_patience=35, x0_lp=lp_x0)

        # Try at different lengths
        for n_try in [512, 2048]:
            if time.monotonic() > deadline_p4:
                break
            if n_try == n_best:
                continue
            a_res = resample_seq(tracker.best, n_try)
            a_res = a_res / max(a_res.sum(), 1e-10)
            budget_here = min(120.0, deadline_p4 - time.monotonic())
            max_iters_here = int(budget_here * 30)
            max_iters_here = min(max_iters_here, 1500)
            a_res, _ = lp_mirror_descent(a_res, max_iters_here, deadline_p4,
                                         tracker=tracker, n_line_evals=18,
                                         stall_patience=30)

        tracker.report(f"Phase 4: {time.monotonic()-t0:.1f}s")

    # === Phase 5: Multi-restart with diverse perturbations (remaining) ===
    if tracker.best is not None and time.monotonic() < deadline:
        t0 = time.monotonic()
        n_best = tracker.best.size
        remaining = deadline - time.monotonic()
        if remaining > 60:
            print(f"\n{'=' * 50} Phase 5: Restarts (budget {remaining:.0f}s) {'=' * 50}", flush=True)

            strategies = [
                ('gaussian', 0.03),
                ('gaussian', 0.05),
                ('gaussian', 0.1),
                ('smooth', 0.05),
                ('smooth', 0.1),
                ('spike', 5),
                ('spike', 16),
                ('block', 32),
                ('fourier_perturb', 6),
            ]

            for pert_type, param in strategies:
                if time.monotonic() > deadline:
                    break
                for trial in range(3):
                    if time.monotonic() > deadline:
                        break
                    a = tracker.best.copy().astype(np.float64)
                    n = len(a)

                    if pert_type == 'gaussian':
                        a = a + rng.normal(0, param, n) * a
                    elif pert_type == 'smooth':
                        noise = rng.normal(0, param, n)
                        ksize = max(3, n // 32)
                        kernel = np.ones(ksize) / ksize
                        noise = np.convolve(noise, kernel, mode='same')
                        a = a + noise * a
                    elif pert_type == 'spike':
                        k = min(param, n // 8)
                        idx = rng.choice(n, size=k, replace=False)
                        a[idx] *= rng.uniform(0.3, 3.0, k)
                    elif pert_type == 'block':
                        block_size = param
                        n_blocks = n // block_size
                        for b in range(n_blocks):
                            if rng.random() < 0.5:
                                a[b*block_size:(b+1)*block_size] *= rng.uniform(0.5, 1.5)
                    elif pert_type == 'fourier_perturb':
                        freqs = rng.choice(np.arange(2, 32), size=6, replace=False)
                        for f in freqs:
                            amp = rng.uniform(0.01, 0.05)
                            phase = rng.uniform(0, 2*np.pi)
                            a += amp * np.cos(2*np.pi*f*np.arange(n)/n + phase)

                    a = np.maximum(a, 1e-4)
                    a = a / a.sum()

                    # Quick Nesterov polish
                    budget_here = min(40.0, deadline - time.monotonic())
                    if budget_here > 5:
                        a = nesterov_smooth_optimize(a, 5.0, 20.0,
                                                     time.monotonic() + budget_here * 0.4,
                                                     max_iters=5000, tracker=tracker)

                    # LP polish
                    budget_lp = min(30.0, deadline - time.monotonic())
                    if budget_lp > 5:
                        max_iters = int(budget_lp * 35)
                        max_iters = min(max_iters, 1000)
                        a, _ = lp_mirror_descent(a, max_iters, time.monotonic() + budget_lp,
                                                 tracker=tracker, n_line_evals=15,
                                                 stall_patience=20)

            tracker.report(f"Phase 5: {time.monotonic()-t0:.1f}s")

    # === Phase 6: Final extended LP polish ===
    if tracker.best is not None and time.monotonic() < deadline:
        remaining = deadline - time.monotonic()
        if remaining > 30:
            t0 = time.monotonic()
            print(f"\n{'=' * 50} Phase 6: Final Polish (budget {remaining:.0f}s) {'=' * 50}", flush=True)
            n_best = tracker.best.size
            max_iters = int(remaining * 40)
            max_iters = min(max_iters, 3000)
            a, _ = lp_mirror_descent(tracker.best, max_iters, deadline,
                                     tracker=tracker, n_line_evals=25,
                                     stall_patience=50)
            tracker.report(f"Phase 6: {time.monotonic()-t0:.1f}s")

    # Fallback
    if tracker.best is None:
        tracker.best = np.ones(512)
        tracker.best_score = compute_J(tracker.best)
        print(f"  Using fallback: J={tracker.best_score:.8f}", flush=True)

    print(f"\n{'=' * 50} FINAL RESULT {'=' * 50}", flush=True)
    print(f"J={tracker.best_score:.8f} n={tracker.best.size} "
          f"total_time={time.monotonic()-tracker.t_start:.1f}s "
          f"evals={tracker.eval_count}", flush=True)

    result = np.clip(tracker.best, 0.0, 1000.0)
    if result.sum() < 0.01:
        result = np.ones(128)

    return [float(x) for x in result]
