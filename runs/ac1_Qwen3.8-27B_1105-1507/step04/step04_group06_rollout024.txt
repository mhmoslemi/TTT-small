def fast_conv(a):
    """Full linear self-convolution via FFT, O(n log n)."""
    a = np.asarray(a, dtype=np.float64)
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]

    """Full linear convolution of a and b via FFT."""
    n = len(a)
    m = len(b)
    size = 1
    while size < n + m - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    return np.fft.irfft(fa * fb, size)[:n + m - 1]

    """Compute J(a) = 2*n*max(conv(a,a))/(sum(a))^2."""
    a = np.asarray(a, dtype=np.float64)

    if s < 0.01:

    return csr_matrix((vals, (rows_arr, cols_arr)), shape=(2 * n - 1, n))

def lp_best_response(a, M, time_limit=30):
    """LP: max sum(g) s.t. conv(a,g)[k] <= M, g >= 0. Returns normalized g or None."""
    n = len(a)
    c = -np.ones(n)
    A = build_A_sparse(a)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    try:
        r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                    options={'presolve': True, 'time_limit': time_limit})
        if r.success and r.x is not None:
            g = np.maximum(r.x, 0.0)
            g_sum = g.sum()
            if g_sum > 1e-10:
                return g / g_sum
    except Exception:
        pass
    return None

def smooth_obj_and_grad(a, beta):
    """
    Compute smooth approximation of J(a) = 2n*max(conv(a,a))/(sum(a))²
    using log-sum-exp, and its gradient.
    """
    n = len(a)
    b = fast_conv(a)
    b_max = float(b.max())
    shifted = beta * (b - b_max)
    shifted = np.clip(shifted, -500, 500)
    exp_vals = np.exp(shifted)
    Z = float(exp_vals.sum())
    if Z < 1e-300:
        p = np.zeros(len(b))
        p[np.argmax(b)] = 1.0
    else:
        p = exp_vals / Z
    smooth_max = b_max + np.log(max(Z, 1e-300)) / beta
    
    # c[j] = sum_{m=0}^{n-1} a[m] * p[m+j] for j=0..n-1
    # = cross-correlation (p ⋆ a)[j]
    size = 1
    while size < n + len(b) - 1:
        size <<= 1
    a_pad = np.zeros(size)
    a_pad[:n] = a
    p_pad = np.zeros(size)
    p_pad[:len(b)] = p
    
    A_fft = np.fft.rfft(a_pad)
    P_fft = np.fft.rfft(p_pad)
    
    # (p ⋆ a)[j] = ifft(FFT(p) * conj(FFT(a)))
    c = np.fft.irfft(P_fft * np.conj(A_fft), size)[:n]
    
    s = float(a.sum())
    J_val = 2.0 * n * smooth_max / (s * s)
    
    # dJ/da[j] = (4n/s²) * c[j] - (4n*smooth_max)/s³
    grad_J = (4.0 * n / (s * s)) * c - (4.0 * n * smooth_max) / (s * s * s)
    
    return J_val, grad_J

def lbfgs_smooth(a0, deadline, max_total_iters=500):
    """
    L-BFGS-B optimization with multi-stage beta annealing.
    Starts with small beta (very smooth) and increases.
    """
    a = np.maximum(np.asarray(a0, dtype=np.float64), 1e-8)
    n = len(a)
    
    # Beta schedule: start smooth, get sharper
    beta_schedule = [
        (0.5, 30),
        (2.0, 40),
        (5.0, 50),
        (10.0, 60),
        (20.0, 70),
        (50.0, 80),
        (100.0, 100),
    ]
    
    iters_used = 0
    for beta, max_iter in beta_schedule:
        if time.monotonic() > deadline or iters_used >= max_total_iters:
            break
        
        def func(x):
            x = np.maximum(x, 0.0)
            if x.sum() < 1e-10:
                return 1e10
            f, _ = smooth_obj_and_grad(x, beta)
            return f if np.isfinite(f) else 1e10
        
        def jac(x):
            x = np.maximum(x, 0.0)
            if x.sum() < 1e-10:
                return np.zeros_like(x)
            _, g = smooth_obj_and_grad(x, beta)
            g = np.nan_to_num(g, nan=0.0, posinf=0.0, neginf=0.0)
            g[x < 1e-12] = 0.0
            return g
        
        try:
            res = minimize(func, a.copy(), jac=jac, method='L-BFGS-B',
                           bounds=[(0.0, None)] * n,
                           options={'maxiter': max_iter, 'max_fun': max_iter * 30,
                                    'gtol': 1e-8, 'ftol': 1e-14,
                                    'disp': False})
            if res.fun < 1e10:
                a_new = np.maximum(res.x, 1e-10)
                if a_new.sum() > 1e-6:
                    a = a_new
            iters_used += max_iter
        except Exception:
            pass
    
    return a

def lp_mirror_descent(a, deadline, tracker, max_iters=2000, n_line_evals=14,
                      stall_patience=30):
    """
    LP-based mirror descent with line search.
    """
    a = np.maximum(a.astype(np.float64), 1e-6)
    a = a / a.sum()
    current_J = compute_J(a)
    tracker.update(a)
    
    stall_count = 0
    
    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        
        b = fast_conv(a)
        M = float(b.max())
        
        # Solve LP
        g = lp_best_response(a, M)
        if g is None:
            break
        
        # Line search: try several step sizes
        best_J = current_J
        best_a = a.copy()
        best_t = 0.0
        
        # Golden section search
        lo, hi = 0.0, 1.0
        gr = (np.sqrt(5.0) - 1.0) / 2.0
        J_lo = best_J
        a_hi = (1.0 - hi) * a + hi * g
        J_hi = compute_J(a_hi)
        if J_hi < best_J:
            best_J = J_hi
            best_a = a_hi.copy()
            best_t = 1.0
        
        for _ in range(n_line_evals):
            t1 = hi - gr * (hi - lo)
            t2 = lo + gr * (hi - lo)
            a1 = (1.0 - t1) * a + t1 * g
            a2 = (1.0 - t2) * a + t2 * g
            J1 = compute_J(a1)
            J2 = compute_J(a2)
            if J1 < J2:
                hi = t2
                if J1 < best_J:
                    best_J = J1
                    best_a = a1.copy()
                    best_t = t1
            else:
                lo = t1
                if J2 < best_J:
                    best_J = J2
                    best_a = a2.copy()
                    best_t = t2
        
        if best_t > 0 and best_J < current_J - 1e-14:
            a = best_a
            current_J = best_J
            stall_count = 0
            tracker.update(a)
        else:
            # Try smaller steps
            found = False
            for t_try in [0.01, 0.005, 0.02, 0.05, 0.1]:
                a_try = (1.0 - t_try) * a + t_try * g
                a_try = np.maximum(a_try, 0.0)
                s = a_try.sum()
                if s < 1e-10:
                    continue
                a_try = a_try / s
                J_try = compute_J(a_try)
                if J_try < current_J - 1e-14:
                    a = a_try
                    current_J = J_try
                    stall_count = 0
                    tracker.update(a)
                    found = True
                    break
            if not found:
                stall_count += 1
                if stall_count >= stall_patience:
                    break
    
    tracker.update(a)
    return a

def resample_seq(src, dest_n):
    """Resample 1D sequence to new length via linear interpolation."""
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    result = np.interp(t_new, t_old, src)
    return np.maximum(result, 0.0)

class ScoreTracker:
    def __init__(self):
        self.best = None
        self.best_score = float('inf')
        self.t_start = time.monotonic()
    
    def update(self, a):
        a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
        if a.size == 0 or a.sum() < 0.01:
            return
        score = float(evaluate_sequence(a.tolist()))
        if np.isfinite(score) and score < self.best_score:
            self.best = a.copy()
            self.best_score = score
            elapsed = time.monotonic() - self.t_start
            print(f"  * NEW BEST: J={score:.8f} n={a.size} t={elapsed:.1f}s", flush=True)
    
    def report(self, label=""):
        if self.best is not None:
            print(f"  {label} best: J={self.best_score:.8f} n={self.best.size}", flush=True)

def generate_initializations(n, rng, parent=None):
    """Generate diverse initial sequences."""
    inits = []
    t = np.linspace(0, 1, n, endpoint=False)
    
    if parent is not None and parent.size > 0:
        inits.append(resample_seq(parent, n))
        p = resample_seq(parent, n)
        p = p + rng.normal(0, 0.05, n) * p
        inits.append(np.maximum(p, 0.01))
    
    # Uniform
    inits.append(np.ones(n))
    
    # Cosine families (various frequencies)
    for freq in [1, 2, 3, 4, 5, 6, 8]:
        inits.append(np.maximum(1.0 + 0.4 * np.cos(2.0 * np.pi * freq * t), 0.01))
    
    # Gaussian bump
    inits.append(np.exp(-3.0 * (t - 0.5) ** 2) + 0.1)
    
    # Edge-emphasized (high at edges, low in middle)
    inits.append(0.5 + 0.5 * np.cos(np.pi * t))
    
    # Sine absolute
    inits.append(np.abs(np.sin(np.pi * t)) + 0.2)
    
    # Two-peak
    inits.append(1.0 + 0.3 * np.cos(4.0 * np.pi * t))
    
    # Power law
    inits.append((t + 0.1) ** 0.5)
    
    # Exponential
    inits.append(np.exp(-1.0 * t) + 0.3)
    
    # Random smooth
    noise = rng.normal(0, 1, n)
    kernel_size = max(3, n // 32)
    kernel = np.ones(kernel_size) / kernel_size
    smooth_noise = np.convolve(noise, kernel, mode='same')
    inits.append(np.maximum(1.0 + 0.3 * smooth_noise, 0.01))
    
    # Inverse cosine (high in middle)
    inits.append(0.5 + 0.5 * np.cos(np.pi * (2 * t - 1)))
    
    return inits
