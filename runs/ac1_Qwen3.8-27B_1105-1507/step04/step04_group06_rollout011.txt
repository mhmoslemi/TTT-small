"""
Minimizes J(a) = 2*n*max(convolve(a,a))/(sum(a))² using:
1. Fast L-BFGS-B with log-sum-exp smoothing and aggressive beta annealing
2. Nesterov-accelerated gradient descent with adaptive step size
3. Periodic LP best-response corrections (sparse, time-limited)
4. Multi-length exploration (512, 1024, 2048)
5. Coordinate descent polish
6. Diverse restarts with structure-aware perturbations

import numpy as np
from scipy.optimize import linprog, minimize
from scipy.sparse import csr_matrix
import time

# Cache for sparse matrix structures
_A_cache = {}
# Cache for FFT sizes
_FFT_size_cache = {}


def _get_fft_size(n):
    """Get optimal FFT size for convolution of length-n sequences."""
    if n not in _FFT_size_cache:
        s = 1
        while s < 2 * n - 1:
            s <<= 1
        _FFT_size_cache[n] = s
    return _FFT_size_cache[n]


def fast_conv(a):
    """Full self-convolution via FFT."""
    n = len(a)
    s = _get_fft_size(n)
    fa = np.fft.rfft(a, s)
    return np.fft.irfft(fa * fa, s)[:2 * n - 1]


def compute_J(a):
    """Compute J(a) = 2*n*max(conv(a,a))/(sum(a))^2."""
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))


def smooth_max_and_grad(a, beta):
    """
    Compute smooth approximation of max(conv(a,a)) and gradient w.r.t. a.
    Returns (g_val, grad_array).
    """
    n = len(a)
    b = fast_conv(a)
    b_max = float(b.max())
    shifted = beta * (b - b_max)
    shifted = np.clip(shifted, -500, 500)
    exp_s = np.exp(shifted)
    Z = float(exp_s.sum())
    if Z < 1e-300:
        p = np.zeros(2 * n - 1)
        p[np.argmax(b)] = 1.0
    else:
        p = exp_s / Z
    g_val = (b_max + np.log(max(Z, 1e-300)) / beta)

    # Gradient: d/da_i of sum_k p_k * b_k
    # b_k = sum_j a_j * a_{k-j}
    # db_k/da_i = a_{k-i} + a_{i-k} (with zero-padding)
    # grad_i = sum_k p_k * (a_{k-i} + a_{i-k})
    #   = sum_k p_k * a_{k-i} + sum_k p_k * a_{i-k}

    # Second term: S2[i] = sum_k p[k] * a[i-k] = (p * a)[i] (standard convolution)
    pa = np.convolve(p, a, mode='full')  # length 3n-2
    S2 = pa[:n]

    # First term: S1[i] = sum_k p[k] * a[k-i] = sum_j a[j] * p[j+i]
    # This is cross-correlation: np.correlate(a, p, 'full')[i] = sum_j a[j]*p[j+i]
    S1 = np.correlate(a, p, mode='full')[:n]

    grad = 2.0 * (S1 + S2)
    return g_val, grad


def F_and_grad(a, beta):
    """Compute F(a) = 2n*smooth_max/(sum(a))^2 and its gradient."""
    n = len(a)
    s = float(a.sum())
    if s < 1e-10:
        return 1e10, np.zeros(n)
    g_val, grad_g = smooth_max_and_grad(a, beta)
    F_val = 2.0 * n * g_val / (s * s)
    grad_F = 2.0 * n * (s * grad_g - 2.0 * g_val) / (s ** 3)
    return F_val, grad_F


def lbfgs_optimize(a0, beta_schedule, deadline):
    """L-BFGS-B with beta annealing schedule."""
    a = np.maximum(a0.astype(np.float64), 1e-8)
    n = len(a)

    for beta, max_iters in beta_schedule:
        if time.monotonic() > deadline:
            break

        def func(x):
            x2 = np.maximum(x, 0.0)
            f, _ = F_and_grad(x2, beta)
            if not np.isfinite(f):
                return 1e10
            return f

        def jac(x):
            x2 = np.maximum(x, 0.0)
            _, g = F_and_grad(x2, beta)
            g = np.nan_to_num(g, nan=0.0, posinf=0.0, neginf=0.0)
            g[x2 < 1e-12] = 0.0
            return g

        bounds = [(0.0, None)] * n
        try:
            res = minimize(func, a.copy(), jac=jac, method='L-BFGS-B',
                           bounds=bounds,
                           options={'maxiter': max_iters, 'max_fun': max_iters * 30,
                                    'gtol': 1e-9, 'ftol': 1e-15})
            if res.fun < 1e8:
                new_a = np.maximum(res.x, 1e-10)
                if new_a.sum() > 1e-6:
                    a = new_a
        except Exception:
            pass

    return a


def nesterov_gradient_descent(a0, deadline, beta=20.0, lr=0.01, momentum=0.9,
                               max_iters=5000, check_interval=50):
    """Nesterov-accelerated gradient descent on the smooth objective."""
    n = len(a0)
    a = np.maximum(a0.astype(np.float64), 1e-8).copy()
    v = np.zeros(n)
    best_a = a.copy()
    best_J = compute_J(a)

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        # Nesterov lookahead
        a_nest = a - momentum * v
        a_nest = np.maximum(a_nest, 0.0)

        _, grad = F_and_grad(a_nest, beta)
        grad = np.nan_to_num(grad, nan=0.0, posinf=0.0, neginf=0.0)

        # Projected gradient step
        v = momentum * v + lr * grad
        a = a - v
        a = np.maximum(a, 1e-10)

        # Periodic check
        if (it + 1) % check_interval == 0:
            J = compute_J(a)
            if J < best_J:
                best_J = J
                best_a = a.copy()

    return best_a


def build_conv_matrix_sparse(n):
    """Build the sparse convolution matrix for the LP."""
    if n in _A_cache:
        return _A_cache[n]
    rows = []
    cols = []
    for k in range(2 * n - 1):
        jl = max(0, k - n + 1)
        jh = min(n - 1, k)
        for j in range(jl, jh + 1):
            rows.append(k)
            cols.append(j)
    A = csr_matrix((np.ones(len(rows)), (np.array(rows, dtype=np.int32),
                                          np.array(cols, dtype=np.int32))),
                   shape=(2 * n - 1, n))
    _A_cache[n] = A
    return A


def lp_best_response(a, M, time_limit=5.0):
    """
    Solve LP: max sum(g) s.t. (a*g)_k <= M for all k, g >= 0.
    Returns normalized g or None.
    """
    n = len(a)
    if time.monotonic() > time_limit:
        return None

    # Build the matrix: A_ij = a_{j-i} for the constraint (a*g)_k <= M
    # (a*g)_k = sum_j a_j * g_{k-j}
    # So the constraint matrix has A[k,j] = a_j if 0 <= k-j < n, else 0
    # i.e., A[k,j] = a_j if k-n+1 <= j <= k, else 0
    # Equivalently: A[k,j] = a_{k-j} if 0 <= k-j <= n-1... 
    # Wait: (a*g)_k = sum_j a_j * g_{k-j} where g_{k-j} is 0 unless 0<=k-j<n
    # So A[k,j] = a_j * 1[0<=k-j<n] = a_j * 1[k-n+1<=j<=k]
    
    t_start = time.monotonic()
    c = -np.ones(n)
    
    # Build sparse matrix
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        jl = max(0, k - n + 1)
        jh = min(n - 1, k)
        for j in range(jl, jh + 1):
            rows.append(k)
            cols.append(j)
            vals.append(a[j])
    
    A_mat = csr_matrix((vals, (np.array(rows, dtype=np.int32),
                                np.array(cols, dtype=np.int32))),
                       shape=(2 * n - 1, n))
    
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    
    try:
        r = linprog(c, A_ub=A_mat, b_ub=b_ub, bounds=bounds,
                    method='highs', options={'time_limit': max(0.1, time_limit - (time.monotonic() - t_start))})
        if r.success and r.x is not None:
            g = np.maximum(r.x, 0.0)
            g_sum = g.sum()
            if g_sum > 1e-10:
                return g / g_sum
    except Exception:
        pass
    return None


def line_search(a, g, n_evals=12):
    """
    Efficient line search: evaluate at a few points and pick the best.
    Returns (best_a, best_J).
    """
    best_J = compute_J(a)
    best_a = a.copy()
    
    # Evaluate at several points
    ts = np.linspace(0.1, 1.0, n_evals)
    for t in ts:
        a_try = (1.0 - t) * a + t * g
        a_try = np.maximum(a_try, 0.0)
        s = a_try.sum()
        if s < 1e-10:
            continue
        a_try = a_try / s
        J = compute_J(a_try)
        if J < best_J:
            best_J = J
            best_a = a_try.copy()
    
    return best_a, best_J


def resample_seq(src, dest_n):
    """Resample 1D sequence to new length via cubic interpolation."""
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    if src.size < 2:
        return np.ones(dest_n) * src[0]
    from scipy.interpolate import CubicSpline
    t_old = np.linspace(0, 1, src.size)
    try:
        cs = CubicSpline(t_old, src, bc_type='not-a-knot')
        t_new = np.linspace(0, 1, dest_n)
        result = cs(t_new)
    except Exception:
        result = np.interp(np.linspace(0, 1, dest_n), t_old, src)
    return np.maximum(result, 0.0)


def generate_initializations(n, rng, parent=None):
    """Generate diverse initial sequences."""
    inits = []
    t = np.linspace(0, 1, n, endpoint=False)

    if parent is not None and parent.size > 1:
        inits.append(resample_seq(parent, n))
        p = resample_seq(parent, n)
        inits.append(np.maximum(p + rng.normal(0, 0.03 * p, n), 0.01))
        inits.append(np.maximum(p + rng.normal(0, 0.1 * p, n), 0.01))

    inits.append(np.ones(n))
    inits.append(rng.uniform(0.2, 1.0, n))

    # Cosine families
    for freq in [1, 2, 3, 4, 5, 6, 8]:
        inits.append(1.0 + 0.4 * np.cos(2.0 * np.pi * freq * t))

    # Power law
    inits.append((t + 0.05) ** 0.3)
    inits.append((1.0 - t + 0.05) ** 0.3)

    # Exponential
    inits.append(np.exp(-2.0 * t) + 0.15)
    inits.append(np.exp(-2.0 * (1.0 - t)) + 0.15)

    # Gaussian
    inits.append(np.exp(-6.0 * (t - 0.5) ** 2) + 0.1)

    # Two-peak
    inits.append(1.0 + 0.3 * np.cos(4.0 * np.pi * t) + 0.15 * np.cos(8.0 * np.pi * t))

    # Chirp-like
    inits.append(1.0 + 0.3 * np.sin(4.0 * np.pi * t * (0.5 + t)))

    # Sinc-like
    x = 8.0 * (t - 0.5)
    inits.append(np.abs(np.sinc(x * 0.5)) + 0.1)

    # Triangular
    inits.append(1.0 - 0.3 * np.abs(2.0 * (t % 0.5) - 0.5))

    # Bimodal
    inits.append(0.8 + 0.4 * np.exp(-8*(t-0.3)**2) + 0.4 * np.exp(-8*(t-0.7)**2))

    return inits


def coordinate_descent_polish(a, deadline, max_rounds=5):
    """Simple coordinate descent: try increasing/decreasing each coordinate."""
    a = a.astype(np.float64).copy()
    n = len(a)
    current_J = compute_J(a)
    
    for rnd in range(max_rounds):
        if time.monotonic() > deadline:
            break
        improved = False
        for i in range(n):
            if time.monotonic() > deadline:
                break
            for scale in [1.1, 0.9, 1.3, 0.7, 1.5, 0.5]:
                a_try = a.copy()
                a_try[i] *= scale
                a_try = np.maximum(a_try, 0.0)
                if a_try.sum() < 1e-6:
                    continue
                J = compute_J(a_try)
                if J < current_J - 1e-12:
                    a = a_try
                    current_J = J
                    improved = True
                    break
        if not improved:
            break
    return a


class ScoreTracker:
    """Tracks the best sequence found so far."""

    def __init__(self):
        self.best = None
        self.best_score = float('inf')
        self.t_start = time.monotonic()

    def update(self, a):
        if a is None:
            return
        a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
        if a.size == 0 or a.sum() < 0.01:
            return
        try:
            score = float(evaluate_sequence(a.tolist()))
        except Exception:
            return
        if np.isfinite(score) and score < self.best_score:
            self.best = a.copy()
            self.best_score = score
            elapsed = time.monotonic() - self.t_start
            print(f"  * NEW BEST: J={score:.8f} n={a.size} t={elapsed:.1f}s", flush=True)

    def report(self, label=""):
        if self.best is not None:
            print(f"  {label} best: J={self.best_score:.8f} n={self.best.size}", flush=True)


def lp_mirror_descent_fast(a, deadline, tracker=None, n_line_evals=12,
                           stall_patience=40, lp_time_limit=5.0,
                           lp_every=1):
    """
    LP-based mirror descent with fast line search.
    Runs LP every lp_every iterations.
    """
    a = np.maximum(a.astype(np.float64), 1e-6)
    a = a / a.sum()
    current_J = compute_J(a)
    best_J = current_J
    best_a = a.copy()
    stall_count = 0

    if tracker is not None:
        tracker.update(a)

    it = 0
    while time.monotonic() < deadline:
        it += 1
        b = fast_conv(a)
        M = float(b.max())

        # Solve LP for best response (every lp_every iterations)
        if it % lp_every == 1:
            g = lp_best_response(a, M, time_limit=lp_time_limit)
            if g is None:
                break
        else:
            # Use smooth gradient direction
            _, grad = F_and_grad(a, 50.0)
            g = np.maximum(1.0 - 0.1 * grad / (np.abs(grad).max() + 1e-10), 0.01)
            g = g / g.sum()

        # Line search
        a_new, J_new = line_search(a, g, n_evals=n_line_evals)

        if J_new < current_J - 1e-13:
            a = a_new
            current_J = J_new
            stall_count = 0
            if J_new < best_J:
                best_J = J_new
                best_a = a.copy()
                if tracker is not None:
                    tracker.update(a)
        else:
            # Try smaller steps
            found = False
            for t_try in [0.02, 0.01, 0.05, 0.005]:
                a_try = (1.0 - t_try) * a + t_try * g
                a_try = np.maximum(a_try, 0.0)
                s = a_try.sum()
                if s < 1e-10:
                    continue
                a_try = a_try / s
                J_try = compute_J(a_try)
                if J_try < current_J - 1e-14:
                    a = a_try
                    current_J = J_try
                    stall_count = 0
                    if J_try < best_J:
                        best_J = J_try
                        best_a = a.copy()
                        if tracker is not None:
                            tracker.update(a)
                    found = True
                    break
            if not found:
                stall_count += 1
                if stall_count >= stall_patience:
                    break

    if tracker is not None:
        tracker.update(best_a)
    return best_a


def accelerated_mirror_descent(a, deadline, tracker=None, beta=30.0,
                               lr_schedule=None, max_iters=10000):
    """
    Nesterov-accelerated gradient descent on smooth objective with
    periodic LP corrections.
    """
    n = len(a)
    a = np.maximum(a.astype(np.float64), 1e-8)
    a = a / a.sum()
    v = np.zeros(n)
    momentum = 0.9
    lr = 1.0 / (np.sqrt(n) * 10.0)
    current_J = compute_J(a)
    best_J = current_J
    best_a = a.copy()
    best_lr = lr

    if tracker is not None:
        tracker.update(a)

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        # Nesterov lookahead
        y = a - momentum * v
        y = np.maximum(y, 0.0)

        # Compute gradient
        _, grad = F_and_grad(y, beta)
        grad = np.nan_to_num(grad, nan=0.0, posinf=0.0, neginf=0.0)

        # Adaptive learning rate
        grad_norm = np.linalg.norm(grad)
        if grad_norm > 1e-15:
            lr = best_lr * (1.0 + 0.1 * np.cos(0.01 * it))
        else:
            lr = best_lr

        # Step
        v = momentum * v + lr * grad
        a_new = a - v
        a_new = np.maximum(a_new, 1e-10)

        # Check improvement
        J_new = compute_J(a_new)
        if J_new < current_J:
            a = a_new
            current_J = J_new
            if J_new < best_J:
                best_J = J_new
                best_a = a.copy()
                best_lr = lr * 1.05
                if tracker is not None:
                    tracker.update(a)
        else:
            # Reject step, reduce lr
            best_lr = lr * 0.7
            momentum = min(0.95, momentum + 0.01)
            a = np.maximum(a, 1e-10)

        # Periodic LP correction
        if (it + 1) % 200 == 0 and time.monotonic() < deadline - 30:
            b = fast_conv(a)
            M = float(b.max())
            g = lp_best_response(a, M, time_limit=4.0)
            if g is not None:
                a_lp, J_lp = line_search(a, g, n_evals=10)
                if J_lp < current_J:
                    a = a_lp
                    current_J = J_lp
                    v = np.zeros(n)
                    if J_lp < best_J:
                        best_J = J_lp
                        best_a = a.copy()
                        if tracker is not None:
                            tracker.update(a)

    if tracker is not None:
        tracker.update(best_a)
    return best_a

            a0 = np.maximum(inits[idx].astype(np.float64), 0.01)
            a0 = a0 / a0.sum()
            a_opt = lbfgs_optimize(a0, beta_schedule, p1_deadline)
            if a_opt is not None and a_opt.sum() > 0.01:
                tracker.update(a_opt)

        print(f"    n={n} done in {time.monotonic()-t_start:.1f}s", flush=True)

    tracker.report("Phase 1 complete")

    # === Phase 2: Nesterov accelerated gradient with LP corrections ===
    if tracker.best is not None and time.monotonic() < deadline:
        p2_budget = 400.0
        p2_deadline = min(deadline, time.monotonic() + p2_budget)
        n_best = tracker.best.size
        print(f"\n{'='*60}", flush=True)
        print(f"Phase 2: Accelerated MD + LP (n={n_best}, budget {p2_budget:.0f}s)", flush=True)
        print(f"{'='*60}", flush=True)

        t_start = time.monotonic()

        # Main run at best length
        a = accelerated_mirror_descent(tracker.best, p2_deadline, tracker=tracker,
                                       beta=30.0, max_iters=20000)

        # Try at other lengths
        for n_try in [1024, 2048]:
            if time.monotonic() > p2_deadline - 30:
                break
            if n_try == n_best:
                continue
            a_res = resample_seq(tracker.best, n_try)
            a_res = a_res / max(a_res.sum(), 1e-10)
            t_sub = time.monotonic()
            print(f"    Trying n={n_try}...", flush=True)
            a_res = accelerated_mirror_descent(a_res, p2_deadline, tracker=tracker,
                                               beta=25.0, max_iters=10000)
            print(f"    n={n_try} done in {time.monotonic()-t_sub:.1f}s", flush=True)

        tracker.report(f"Phase 2: {time.monotonic()-t_start:.1f}s")

    # === Phase 3: LP mirror descent polish ===
    if tracker.best is not None and time.monotonic() < deadline:
        p3_budget = 350.0
        p3_deadline = min(deadline, time.monotonic() + p3_budget)
        n_best = tracker.best.size
        print(f"\n{'='*60}", flush=True)
        print(f"Phase 3: LP Mirror Descent (n={n_best}, budget {p3_budget:.0f}s)", flush=True)
        print(f"{'='*60}", flush=True)

        t_start = time.monotonic()
        a = lp_mirror_descent_fast(tracker.best, p3_deadline, tracker=tracker,
                                   n_line_evals=14, stall_patience=50,
                                   lp_time_limit=6.0, lp_every=1)
        tracker.report(f"Phase 3: {time.monotonic()-t_start:.1f}s")

    # === Phase 4: L-BFGS-B final polish + coordinate descent ===
    if tracker.best is not None and time.monotonic() < deadline:
        p4_budget = 200.0
        p4_deadline = min(deadline, time.monotonic() + p4_budget)
        n_best = tracker.best.size
        print(f"\n{'='*60}", flush=True)
        print(f"Phase 4: Final Polish (n={n_best}, budget {p4_budget:.0f}s)", flush=True)
        print(f"{'='*60}", flush=True)

        t_start = time.monotonic()

        # L-BFGS-B with very high beta
        beta_final = [(50.0, 60), (100.0, 80), (200.0, 100), (500.0, 120)]
        a = lbfgs_optimize(tracker.best, beta_final, p4_deadline)
        if a is not None:
            tracker.update(a)

        # Coordinate descent polish
        if time.monotonic() < p4_deadline - 30:
            a_cd = coordinate_descent_polish(tracker.best, p4_deadline, max_rounds=3)
            tracker.update(a_cd)

        # LP correction
        if time.monotonic() < p4_deadline - 10:
            a = lp_mirror_descent_fast(tracker.best, p4_deadline, tracker=tracker,
                                       n_line_evals=10, stall_patience=30,
                                       lp_time_limit=4.0, lp_every=2)

        tracker.report(f"Phase 4: {time.monotonic()-t_start:.1f}s")

    # === Phase 5: Multi-restart with diverse perturbations ===
    if tracker.best is not None and time.monotonic() < deadline:
        p5_budget = min(150.0, deadline - time.monotonic())
        p5_deadline = time.monotonic() + p5_budget
        if p5_budget > 20:
            print(f"\n{'='*60}", flush=True)
            print(f"Phase 5: Restarts (budget {p5_budget:.0f}s)", flush=True)
            print(f"{'='*60}", flush=True)

            t_start = time.monotonic()
            n_best = tracker.best.size
            best = tracker.best.copy()

            perturbations = [
                ('gauss', 0.05), ('gauss', 0.1), ('gauss', 0.02),
                ('uniform', 0.08), ('spike', 6), ('smooth', 0.06),
                ('gauss', 0.15), ('uniform', 0.15),
            ]

            attempt = 0
            for ptype, param in perturbations:
                for trial in range(4):
                    if time.monotonic() > p5_deadline:
                        break
                    attempt += 1
                    a = best.copy().astype(np.float64)
                    n = len(a)

                    if ptype == 'gauss':
                        a = a + rng.normal(0, param, n) * a
                    elif ptype == 'uniform':
                        a = a + rng.uniform(-param, param, n) * a
                    elif ptype == 'spike':
                        k = min(param, n // 5)
                        idx = rng.choice(n, size=k, replace=False)
                        a[idx] *= rng.uniform(0.3, 3.0, k)
                    elif ptype == 'smooth':
                        noise = rng.normal(0, param, n)
                        ks = max(3, n // 32)
                        kernel = np.ones(ks) / ks
                        noise = np.convolve(noise, kernel, mode='same')
                        a = a + noise * a

                    a = np.maximum(a, 1e-4)
                    a = a / a.sum()

                    # Quick check
                    J_pert = compute_J(a)
                    if J_pert < tracker.best_score:
                        tracker.update(a)

                    # Refine with L-BFGS-B (fast)
                    if time.monotonic() < p5_deadline - 15:
                        a_opt = lbfgs_optimize(a, [(20.0, 40), (80.0, 50)],
                                                p5_deadline)
                        if a_opt is not None:
                            tracker.update(a_opt)

            tracker.report(f"Phase 5: {time.monotonic()-t_start:.1f}s, {attempt} restarts")

    # === Phase 6: Length exploration ===
    if tracker.best is not None and time.monotonic() < deadline:
        remaining = deadline - time.monotonic()
        if remaining > 30:
            print(f"\n{'='*60}", flush=True)
            print(f"Phase 6: Length Exploration (budget {remaining:.0f}s)", flush=True)
            print(f"{'='*60}", flush=True)

            n_best = tracker.best.size
            candidates = []
            for delta in [-128, -64, -32, 32, 64, 128, 256]:
                n_try = n_best + delta
                if 128 <= n_try <= 4096:
                    candidates.append(n_try)
            # Also try some fixed good lengths
            for n_fix in [768, 1280, 1536, 2048, 3072]:
                if abs(n_fix - n_best) > 64 and 128 <= n_fix <= 4096:
                    candidates.append(n_fix)

            for n_try in candidates:
                if time.monotonic() > deadline:
                    break
                a_res = resample_seq(tracker.best, n_try)
                a_res = a_res / max(a_res.sum(), 1e-10)
                budget_here = min(30.0, deadline - time.monotonic())
                if budget_here < 10:
                    break
                a_opt = lbfgs_optimize(a_res, [(30.0, 40), (100.0, 50)],
                                       time.monotonic() + budget_here)
                if a_opt is not None:
                    tracker.update(a_opt)

            tracker.report("Phase 6 complete")

    # Fallback
    if tracker.best is None:
        tracker.best = np.ones(1024)
        tracker.best_score = compute_J(tracker.best)
        print(f"  Using fallback: J={tracker.best_score:.8f}", flush=True)

    print(f"\n{'='*60}", flush=True)
    print(f"FINAL RESULT: J={tracker.best_score:.8f} n={tracker.best.size} "
          f"total_time={time.monotonic()-t_total_start:.1f}s", flush=True)
    print(f"{'='*60}", flush=True)

    # Ensure output is valid
    result = np.clip(tracker.best, 0.0, 1000.0)
    if result.sum() < 0.01:
        result = np.ones(128)

    return [float(x) for x in result]
