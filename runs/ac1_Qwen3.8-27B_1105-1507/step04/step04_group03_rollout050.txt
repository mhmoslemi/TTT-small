        for j in range(j_min, j_max + 1):
            rows.append(k)
            cols.append(j)
            vals.append(a[k - j])
    return csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))

def fast_conv(a):
    """Full self-convolution via FFT."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]


def compute_J(a):
    """Compute J(a) = 2*n*max(conv(a,a))/sum(a)^2."""
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))


def build_conv_matrix(n, a):
    """Build sparse matrix A where A[k,j] = a[k-j] for valid (k,j)."""
    rows = np.zeros(2*n-1, dtype=np.int32)  # will be overwritten

    row_list = []
    col_list = []
    val_list = []

            col_list.append(j)
            val_list.append(a[k - j])

def spectral_flatten(a, alpha=0.1):
    """Flatten the power spectrum of a."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    A = np.fft.rfft(a, size)
    P = np.abs(A)**2
    P_flat = np.full_like(P, P.mean())
    P_new = (1 - alpha) * P + alpha * P_flat
    A_new = np.sqrt(P_new) * np.exp(1j * np.angle(A))
    a_new = np.fft.irfft(A_new, size)[:n]
    a_new = np.maximum(a_new, 0)
    s = a_new.sum()
    if s > 1e-10:
        a_new = a_new / s
    return a_new

def fast_conv(a):
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]

def compute_J(a):
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))

def get_conv_structure(n):
    rows_list = []
    cols_list = []
    for k in range(2 * n - 1):
        jmin = max(0, k - n + 1)
        jmax = min(n - 1, k)
        for j in range(jmin, jmax + 1):
            rows_list.append(k)
            cols_list.append(j)
    return (np.array(rows_list, dtype=np.int32),
            np.array(cols_list, dtype=np.int32))

def solve_lp(n, a, M, structure):
    rows, cols = structure
    vals = a[rows - cols]
    A = csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    try:
        r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
        if r.success and r.x is not None:
