def fast_conv(a):
    """Full self-convolution via FFT."""
    n = len(a)
    s = 1
    while s < 2 * n - 1:
        s <<= 1
    fa = np.fft.rfft(a, s)
    return np.fft.irfft(fa * fa, s)[:2 * n - 1]


def compute_J(a):
    """Compute J(a) = 2*n*max(conv(a,a))/(sum(a))^2."""
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))

def golden_section_1d(f, lo, hi, n_iter=40):
    """Golden section minimization of 1D function."""
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    c_pt = hi - gr * (hi - lo)
    d_pt = lo + gr * (hi - lo)
    fc = f(c_pt)
    fd = f(d_pt)
    for _ in range(n_iter):
        if fc < fd:
            hi = d_pt
            d_pt = c_pt
            fd = fc
            c_pt = hi - gr * (hi - lo)
            fc = f(c_pt)
        else:
            lo = c_pt
            c_pt = d_pt
            fc = fd
            d_pt = lo + gr * (hi - lo)
            fd = f(d_pt)
    return (lo + hi) / 2.0


def cd_optimize_coord(a, i, b, S, n):
    """Find optimal value for a[i] given other coordinates fixed."""
    ai = a[i]

    def obj(x):
        new_S = S - ai + x
        if new_S < 1e-12:
            return 1e15
        # Compute new max convolution
        # b[k] changes for k in [i, i+n-1]
        max_c = 0.0
        # First, find max of unchanged b values
        # Changed range: [i, i+n-1]
        lo_k = max(0, i)
        hi_k = min(2 * n - 2, i + n - 1)
        # Max of unchanged
        if lo_k > 0:
            max_unchanged_left = b[:lo_k].max() if lo_k > 0 else 0.0
        else:
            max_unchanged_left = 0.0
        if hi_k < 2 * n - 2:
            max_unchanged_right = b[hi_k + 1:].max() if hi_k + 1 < 2 * n - 1 else 0.0
        else:
            max_unchanged_right = 0.0
        max_unchanged = max(max_unchanged_left, max_unchanged_right)

        # Compute changed values
        for k in range(lo_k, hi_k + 1):
            ki = k - i
            if 0 <= ki < n:
                if k == 2 * i:
                    ck = b[k] - ai * ai + x * x
                else:
                    ck = b[k] - 2.0 * ai * a[ki] + 2.0 * x * a[ki]
            else:
                ck = b[k]
            if ck > max_c:
                max_c = ck

        if max_unchanged > max_c:
            max_c = max_unchanged

        return 2.0 * n * max_c / (new_S * new_S)

    hi = max(4.0 * ai + 0.01, 4.0 * S / n + 0.01)
    x_opt = golden_section_1d(obj, 0.0, hi, 35)
    return max(0.0, x_opt)


def cd_pass_fast(a, max_coords=None):
    """One pass of coordinate descent. Returns (a, J)."""
    a = np.maximum(a.astype(np.float64), 0.0)
    S = a.sum()
    if S < 1e-12:
        return a, float('inf')
    a = a / S  # Normalize
    S = 1.0
    n = len(a)
    b = fast_conv(a)
    current_J = 2.0 * n * b.max() / S

    indices = list(range(n))
    np.random.shuffle(indices)
    if max_coords is not None:
        indices = indices[:min(max_coords, n)]

    for i in indices:
        new_val = cd_optimize_coord(a, i, b, S, n)
        old_val = a[i]
        if abs(new_val - old_val) > 1e-14:
            a[i] = new_val
            S = S - old_val + new_val
            # Update b incrementally
            lo_k = max(0, i)
            hi_k = min(2 * n - 2, i + n - 1)
            for k in range(lo_k, hi_k + 1):
                ki = k - i
                if 0 <= ki < n:
                    if k == 2 * i:
                        b[k] = b[k] - old_val * old_val + new_val * new_val
                    else:
                        b[k] = b[k] - 2.0 * old_val * a[ki] + 2.0 * new_val * a[ki]

    # Renormalize
    S = a.sum()
    if S < 1e-12:
        return a, float('inf')
    a = a / S
    b = fast_conv(a)
    J = 2.0 * n * b.max()

def gradient_step(a, lr, momentum=0.0):
    """One projected gradient descent step on J (with a normalized to sum=1)."""
    a = np.maximum(a.astype(np.float64), 0.0)
    S = a.sum()
    if S < 1e-12:
        return a
    a = a / S
    n = len(a)
    b = fast_conv(a)
    b_max = b.max()
    M = b_max

    # Find active set
    tol = 1e-8 * b_max + 1e-15
    active = np.where(b >= b_max - tol)[0]

    # Compute subgradient
    grad = np.zeros(n)
    for k in active:
        lo_j = max(0, k - n + 1)
        hi_j = min(n - 1, k)
        for j in range(lo_j, hi_j + 1):
            ki = k - j
            if 0 <= ki < n:
                if ki == j:
                    grad[j] += 2.0 * a[j] / len(active)
                else:
                    grad[j] += a[ki] / len(active)

    # Gradient of J = 2n*M/S^2 - 4n*M*S^{-3} (dS/da = 1)
    # With S = 1: dJ/da[i] = 2n*grad[i] - 4n*M
    grad_J = 2.0 * n * grad - 4.0 * n * M

    a_new = a - lr * grad_J
    a_new = np.maximum(a_new, 0.0)
    s_new = a_new.sum()
    if s_new < 1e-12:
        return a
    a_new = a_new / s_new
    return a_new


def resample_seq(src, dest_n):
    """Resample 1D sequence to new length."""
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)

def run_lp_optimize(a, max_iters, deadline, tracker, tighten_base=0.003):
    """LP best-response optimization with adaptive line search."""
    a = np.maximum(a.astype(np.float64), 0.001)
    a = a / a.sum()
    n = len(a)
    current_J = compute_J(a)
    tracker.update(a)

    tighten = tighten_base
    stall = 0
    prev_a = a.copy()

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        b = fast_conv(a)
        M = float(b.max())
        M_target = M * (1.0 - tighten)

        g = lp_solve(a, M_target)
        if g is None:
            M_target = M * (1.0 - tighten * 0.3)
            g = lp_solve(a, M_target)
        if g is None:
            tighten = max(0.0005, tighten * 0.5)
            stall += 1
            if stall > 30:
                break
            continue

        g = np.maximum(g, 0.0)
        g_sum = g.sum()
        if g_sum < 1e-12:
            break
        g = g / g_sum

        # Line search for best step size
        best_t = 0.0
        best_J = current_J
        best_a = a.copy()

        # Try a range of step sizes with momentum
        progress = min(1.0, it / max(1, max_iters))
        base_t = max(0.05, 0.6 * (1.0 - 0.5 * progress))

        t_candidates = [base_t, base_t * 0.5, base_t * 1.5,
                        base_t * 0.3, base_t * 2.0, 0.8, 0.3]
        # Add momentum variants
        momentum = 0.1 * (1.0 - progress)
        for t in t_candidates:
            a_trial = (1.0 - t) * a + t * g + momentum * (a - prev_a)
            a_trial = np.maximum(a_trial, 0.0)
            s = a_trial.sum()
            if s < 1e-12:
                continue
            a_trial = a_trial / s
            J_trial = compute_J(a_trial)
            if J_trial < best_J - 1e-14:
                best_J = J_trial
                best_t = t
                best_a = a_trial.copy()

        if best_a is None or best_t == 0.0:
            stall += 1
            if stall > 25:
                break
            tighten = max(0.0005, tighten * 0.7)
            continue

        prev_a = a.copy()
        a = best_a
        current_J = best_J
        stall = 0
        tracker.update(a)

        # Adapt tightening
        if it % 10 == 0:
            if current_J < compute_J(a) + 1e-10:  # progress check
                tighten = min(0.01, tighten * 0.95)
            else:
                tighten = max(0.0005, tighten * 1.1)

    return a, current_J

    tracker.update(a)

    lr = 0.01
    best_J = current_J
    best_a = a.copy()
    patience = 0

    for step in range(n_steps):
        if time.monotonic() > deadline:
            break

        a_new = gradient_step(a, lr)
        J_new = compute_J(a_new)

        if J_new < current_J - 1e-12:
            current_J = J_new
            a = a_new
            if J_new < best_J:
                best_J = J_new
                best_a = a.copy()
                tracker.update(a)
            patience = 0
        else:
            patience += 1
            lr *= 0.8
            if patience > 20:
                break

    return best_a, best_J


def run_cd_polish(a, n_passes, deadline, tracker):
    """Polish using coordinate descent."""
    a = np.maximum(a.astype(np.float64), 0.0)
    a = a / a.sum()
    current_J = compute_J(a)
    tracker.update(a)

    for p in range(n_passes):
        if time.monotonic() > deadline:
            break
        a, new_J = cd_pass_fast(a)
        if new_J < current_J - 1e-13:
            current_J = new_J
            tracker.update(a)
        else:
            if p > 2:
                break

    return a, current_J


def basin_hopping(a, deadline, tracker, n_trials=10, rng=None):
    """Basin hopping: perturb best solution, re-optimize, keep if better."""
    if rng is None:
        rng = np.random.default_rng()

    best_a = a.copy()
    best_J = compute_J(a)
    tracker.update(best_a)
    n = best_a.size

    for trial in range(n_trials):
        if time.monotonic() > deadline:
            break

        # Create perturbation
        temp = 0.02 / (1 + 0.1 * trial)
        a_pert = best_a + rng.normal(0, temp, n)
        a_pert = np.maximum(a_pert, 0.001)
        a_pert = a_pert / a_pert.sum()

        # Also try structured perturbations
        if trial % 3 == 1:
            # Swap two coordinates
            i, j = rng.choice(n, 2, replace=False)
            a_pert[j], a_pert[i] = best_a[i], best_a[j]
            a_pert = np.maximum(a_pert, 0.001)
            a_pert = a_pert / a_pert.sum()
        elif trial % 3 == 2:
            # Local smoothing
            kernel = np.ones(3) / 3.0
            a_pert = np.convolve(best_a, kernel, mode='same')
            a_pert = np.maximum(a_pert, 0.001)
            a_pert = a_pert / a_pert.sum()

        # Optimize the perturbation
        sub_deadline = min(deadline, time.monotonic() + (deadline - time.monotonic()) / max(1, n_trials - trial))
        a_opt, J_opt = run_lp_optimize(a_pert, 30, sub_deadline, tracker, tighten_base=0.002)
        a_opt, J_opt = run_cd_polish(a_opt, 3, sub_deadline, tracker)

        if J_opt < best_J - 1e-10:
            best_a = a_opt
            best_J = J_opt
            tracker.update(best_a)

    return best_a


def propose_candidate(seed=42, budget_s=1500, **kwargs):
    """
    Main search function combining LP, gradient, CD, and basin hopping.
    """
    global a_cache
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + float(budget_s) - 30.0
    tracker = ScoreTracker()

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        if p.ndim == 1 and p.size > 0 and np.all(np.isfinite(p)):
            parent = np.maximum(p, 0.0)
            tracker.update(parent)
            print(f"Parent loaded: n={parent.size}, J={tracker.best_score:.8f}", flush=True)
    except (NameError, AttributeError):
        pass

    if parent is None:
        parent = np.ones(384)
        tracker.update(parent)
        print(f"No parent found, using uniform n={parent.size}", flush=True)

    # === Phase 1: Intensive LP optimization from parent ===
    print(f"\n{'='*50} Phase 1: LP Optimization {'='*50}", flush=True)
    t0 = time.monotonic()
    phase1_deadline = deadline - (deadline - t0) * 0.45

    a = parent.copy()
    a, J = run_lp_optimize(a, 300, phase1_deadline, tracker, tighten_base=0.004)
    print(f"Phase 1 done: J={J:.8f}, t={time.monotonic()-t0:.1f}s", flush=True)

    # === Phase 2: Gradient + CD polish ===
    if time.monotonic() < deadline - (deadline - t0) * 0.3:
        print(f"\n{'='*50} Phase 2: Gradient + CD Polish {'='*50}", flush=True)
        t0 = time.monotonic()
        phase2_deadline = deadline - (deadline - tracker.t_start) * 0.25

        a, J = run_gradient_polish(tracker.best, phase2_deadline, tracker, n_steps=300)
        a, J = run_cd_polish(a, 10, phase2_deadline, tracker)
        print(f"Phase 2 done: J={J:.8f}, t={time.monotonic()-t0:.1f}s", flush=True)

    # === Phase 3: Basin Hopping ===
    if time.monotonic() < deadline - 120:
        print(f"\n{'='*50} Phase 3: Basin Hopping {'='*50}", flush=True)
        t0 = time.monotonic()
        phase3_deadline = deadline - 100

        a = basin_hopping(tracker.best, phase3_deadline, tracker, n_trials=15, rng=rng)
        print(f"Phase 3 done: J={tracker.best_score:.8f}, t={time.monotonic()-t0:.1f}s", flush=True)

    # === Phase 4: Try nearby lengths ===
    if time.monotonic() < deadline - 80:
        print(f"\n{'='*50} Phase 4: Length Exploration {'='*50}", flush=True)
        t0 = time.monotonic()
        n_parent = tracker.best.size
        phase4_deadline = deadline - 60

        nearby = set()
        for delta in [-8, -6, -4, -2, 2, 4, 6, 8, 12, -12]:
            nl = n_parent + delta
            if 200 <= nl <= 700:
                nearby.add(nl)

        for nl in sorted(nearby):
            if time.monotonic() > phase4_deadline:
                break

            a_res = resample_seq(tracker.best, nl)
            a_res = np.maximum(a_res, 0.001)
            a_res = a_res / a_res.sum()

            sub_dl = min(phase4_deadline, time.monotonic() + 20)
            a_res, _ = run_lp_optimize(a_res, 25, sub_dl, tracker, tighten_base=0.003)
            if time.monotonic() < phase4_deadline:
                a_res, _ = run_cd_polish(a_res, 3, phase4_deadline, tracker)

            print(f"  n={nl}: J={tracker.best_score:.8f}", flush=True)

        print(f"Phase 4 done: J={tracker.best_score:.8f}, t={time.monotonic()-t0:.1f}s", flush=True)

    # === Phase 5: Final intensive polish ===
    if time.monotonic() < deadline - 10:
        print(f"\n{'='*50} Phase 5: Final Polish {'='*50}", flush=True)
        t0 = time.monotonic()

        a_final = tracker.best.copy()

        # Final LP
        a_final, _ = run_lp_optimize(a_final, 30, deadline - 30, tracker, tighten_base=0.001)
        # Final gradient
        a_final, _ = run_gradient_polish(a_final, deadline - 20, tracker, n_steps=100)
        # Final CD
        a_final, _ = run_cd_polish(a_final, 5, deadline - 5, tracker)

        print(f"Phase 5 done: J={tracker.best_score:.8f}, t={time.monotonic()-t0:.1f}s", flush=True)

    # Fallback
    if tracker.best is None:
        tracker.best = np.ones(256)
        tracker.best_score = compute_J(tracker.best)

    print(f"\n{'='*50} FINAL RESULT {'='*50}", flush=True)
    print(f"J = {tracker.best_score:.10f}, n = {tracker.best.size}", flush=True)
    print(f"Total time: {time.monotonic()-tracker.t_start:.1f}s", flush=True)

    return [float(x) for x in tracker.best]
