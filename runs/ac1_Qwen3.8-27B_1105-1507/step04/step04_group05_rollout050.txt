def fast_conv(a):

    fa = np.fft.rfft(a, s)
    return np.fft.irfft(fa * fa, s)[:2 * n - 1]

    S = a.sum()
    if S < 0.01:

    b = fast_conv(a)

    A = lil_matrix((2 * n - 1, n))

        jh = min(n - 1, k)
        for j in range(jl, jh + 1):

    return A.tocsr()

def fast_conv(a):
    n = len(a)
    s = 1
    while s < 2 * n - 1:
        s <<= 1
    fa = np.fft.rfft(a, s)
    return np.fft.irfft(fa * fa, s)[:2 * n - 1]

def compute_J(a):
    a = np.maximum(a, 0)
    S = a.sum()
    if S < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (S * S))

def build_A_sparse(a):
    n = len(a)
    A = lil_matrix((2 * n - 1, n))
    for k in range(2 * n - 1):
        jl = max(0, k - n + 1)
        jh = min(n - 1, k)
        for j in range(jl, jh + 1):
            A[k, j] = a[k - j]
    return A.tocsr()

def lp_solve(a, M_target):
    """Solve LP: max sum(g) s.t. (a*g)_k <= M_target, g >= 0."""
    n = len(a)
    c_vec = -np.ones(n)
    A = build_A_sparse(a)
    b_ub = np.full(2 * n - 1, M_target)
    bounds = [(0.0, None)] * n
    r = linprog(c_vec, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                options={'presolve': True, 'dual_feasibility_tolerance': 1e-9})
    if r.success and r.x is not None:
        g = np.maximum(r.x, 0.0)
        if g.sum() > 1e-10:
            return g
    return None
