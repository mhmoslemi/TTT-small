Optimized step function search using:
1. L-BFGS-B with softmax parameterization and subgradient (primary optimizer)
2. Alternating LP with warm starts (refinement)
3. Intensive local search with simulated annealing
4. Multi-n exploration

import numpy as np
import time
from scipy.optimize import minimize, linprog
from scipy.special import softmax

def compute_conv(a):
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    conv = np.fft.irfft(fa * fa, size)
    return conv[:2 * n - 1]

def compute_obj(a):
    n = len(a)
    if n == 0:
        return np.inf
    conv = compute_conv(a)
    max_b = float(np.max(conv))
    sum_a = float(np.sum(a))
    if sum_a < 0.01:
        return np.inf
    return 2.0 * n * max_b / (sum_a * sum_a)

def grad_max_conv(a, w=None):
    """Compute subgradient of max(conv(a,a)) w.r.t. a.
    Returns grad of shape (n,)."""
    n = len(a)
    conv = compute_conv(a)
    M = np.max(conv)
    if w is None:
        # Use soft assignment
        tau = max(M * 0.01, 1e-10)
        w = np.exp(-(conv - M) / tau)
        w /= np.sum(w)
    # grad_a[m] = 2 * sum_t w[t+m] * a[t] for t=0..n-1
    # = 2 * cross_correlation(w, a)[0:n]
    N = 1
    while N < 3 * n:
        N <<= 1
    W = np.fft.rfft(w, N)
    A = np.fft.rfft(a, N)
    cross = np.fft.irfft(W * np.conj(A), N)
    grad_a = 2.0 * cross[:n]
    return grad_a, M

def softmax_obj_and_grad(x):
    """Objective and gradient in softmax parameterization."""
    x_shifted = x - np.max(x)
    a = np.exp(x_shifted)
    a /= np.sum(a)
    n = len(a)
    grad_a, M = grad_max_conv(a)
    # Chain rule: grad_x = a * (grad_a - dot(grad_a, a))
    grad_x = a * (grad_a - np.dot(grad_a, a))
    obj = 2.0 * n * M
    return obj, grad_x

def softmax_obj_only(x):
    x_shifted = x - np.max(x)
    a = np.exp(x_shifted)
    a /= np.sum(a)
    return compute_obj(a)

def lbfgs_optimize(a, time_budget=60.0, max_iter=200):
    """Optimize using L-BFGS-B in softmax space."""
    n = len(a)
    a = a.copy().astype(np.float64)
    s = np.sum(a)
    if s < 1e-10:
        a = np.ones(n) / n
    a = a / s
    
    # Initialize x from a: x_i = log(a_i)
    a_safe = np.maximum(a, 1e-15)
    x0 = np.log(a_safe)
    x0 -= np.max(x0)  # for numerical stability
    
    best_x = x0.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    
    result = minimize(softmax_obj_and_grad, x0, method='L-BFGS-B',
                      options={'maxiter': max_iter, 'maxfun': 500,
                               'fraction_lost': 1e-8})
    
    if result.fun < best_obj:
        best_obj = result.fun
        best_x = result.x.copy()
    
    # Convert back
    x_shifted = best_x - np.max(best_x)
    a_best = np.exp(x_shifted)
    a_best /= np.sum(a_best)
    
    return a_best, best_obj

def build_toeplitz_row(a, k):
    """Get the row k of the Toeplitz matrix for convolution with a."""
    n = len(a)
    row = np.zeros(n)
    # (a * g)[k] = sum_j a[j]*g[k-j]
    # For this to be valid: 0<=j<n and 0<=k-j<n => max(0, k-n+1) <= j <= min(n-1, k)
    j_start = max(0, k - n + 1)
    j_end = min(n - 1, k)
    for j in range(j_start, j_end + 1):
        row[j] = a[k - j]
    return row
