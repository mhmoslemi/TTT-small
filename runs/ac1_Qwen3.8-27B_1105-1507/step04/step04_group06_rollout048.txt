import numpy as np
from scipy.optimize import linprog, minimize
from scipy.sparse import csr_matrix
import time
import cvxpy as cp

# Global cache for LP matrix structure
_lp_structure_cache = {}
_qp_structure_cache = {}


def fast_conv(a):
    """Full self-convolution via FFT."""
    n = len(a)
    s = 1
    while s < 2 * n - 1:
        s <<= 1
    fa = np.fft.rfft(a, s)
    return np.fft.irfft(fa * fa, s)[:2 * n - 1]


def compute_J(a):
    """Compute J(a) = 2n*max(conv(a,a))/(sum(a))^2."""
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))


def get_lp_structure(n):
    """Pre-compute (row, col) arrays for LP constraint matrix."""
    if n not in _lp_structure_cache:
        rows = []
        cols = []
        for k in range(2 * n - 1):
            jl = max(0, k - n + 1)
            jh = min(n - 1, k)
            for j in range(jl, jh + 1):
                rows.append(k)
                cols.append(j)
        _lp_structure_cache[n] = (np.array(rows, dtype=np.int32),
                                  np.array(cols, dtype=np.int32))
    return _lp_structure_cache[n]


def build_lp_matrix(a):
    """Build sparse LP constraint matrix A where A[g] = conv(a, g)."""
    n = len(a)
    rows_arr, cols_arr = get_lp_structure(n)
    vals = a[rows_arr - cols_arr]
    return csr_matrix((vals, (rows_arr, cols_arr)), shape=(2 * n - 1, n))


def lp_solve(a, M, prev_g=None):
    """Solve LP: max sum(g) s.t. conv(a,g) <= M, g >= 0. Returns g or None."""
    n = len(a)
    c = -np.ones(n)
    A = build_lp_matrix(a)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    try:
        r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
        if r.success and r.x is not None:
            g = np.maximum(r.x, 0.0)
            if g.sum() > 1e-14:
                return g / g.sum()
    except Exception:
        pass
    return None


def line_search(a, g, n_evals=20):
    """Bisection + golden section hybrid line search for t in [0,1]."""
    J0 = compute_J(a)
    a1 = g.copy()
    J1 = compute_J(a1)
    if J1 < J0:
        best_t, best_J = 1.0, J1
    else:
        best_t, best_J = 0.0, J0

    gr = (np.sqrt(5.0) - 1.0) / 2.0
    lo, hi = 0.0, 1.0
    t1 = hi - gr * (hi - lo)
    t2 = lo + gr * (hi - lo)
    Jt1 = compute_J((1 - t1) * a + t1 * g)
    Jt2 = compute_J((1 - t2) * a + t2 * g)
    if Jt1 < best_J:
        best_J, best_t = Jt1, t1
    if Jt2 < best_J:
        best_J, best_t = Jt2, t2

    for _ in range(n_evals - 2):
        if Jt1 < Jt2:
            hi = t2
            t2 = t1
            Jt2 = Jt1
            t1 = hi - gr * (hi - lo)
            Jt1 = compute_J((1 - t1) * a + t1 * g)
        else:
            lo = t1
            t1 = t2
            Jt1 = Jt2
            t2 = lo + gr * (hi - lo)
            Jt2 = compute_J((1 - t2) * a + t2 * g)
        if Jt1 < best_J:
            best_J, best_t = Jt1, t1
        if Jt2 < best_J:
            best_J, best_t = Jt2, t2

    return (1 - best_t) * a + best_t * g, best_J


def lp_mirror_descent(a, max_iters, deadline, tracker=None, n_line_evals=18,
                      stall_patience=50, qp_interval=0):
    """LP mirror descent with optional periodic QP refinement."""
    a = np.maximum(a.astype(np.float64), 1e-8)
    a = a / a.sum()
    current_J = compute_J(a)
    if tracker:
        tracker.update(a)

    stall_count = 0
    best_a = a.copy()
    best_J = current_J
    prev_g = None

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        # Periodic QP refinement
        if qp_interval > 0 and it % qp_interval == 0 and it > 0:
            a_qp, t_qp = qp_solve(a, deadline, max_rounds=8)
            if a_qp is not None:
                J_qp = compute_J(a_qp)
                if J_qp < best_J:
                    best_J = J_qp
                    best_a = a_qp.copy()
                    a = a_qp.copy()
                    current_J = J_qp
                    if tracker:
                        tracker.update(a)

        b = fast_conv(a)
        M = float(b.max())

        g = lp_solve(a, M, prev_g)
        if g is None:
            break

        prev_g = g.copy()

        a_new, J_new = line_search(a, g, n_evals=n_line_evals)

        if J_new < current_J - 1e-12:
            a = a_new
            current_J = J_new
            stall_count = 0
            if J_new < best_J:
                best_J = J_new
                best_a = a.copy()
                if tracker:
                    tracker.update(a)
        else:
            # Try small explicit steps
            improved = False
            for t_try in [0.02, 0.01, 0.05, 0.005]:
                a_try = np.maximum((1 - t_try) * a + t_try * g, 0.0)
                s = a_try.sum()
                if s < 1e-10:
                    continue
                a_try = a_try / s
                J_try = compute_J(a_try)
                if J_try < current_J - 1e-14:
                    a = a_try
                    current_J = J_try
                    stall_count = 0
                    improved = True
                    if J_try < best_J:
                        best_J = J_try
                        best_a = a.copy()
                        if tracker:
                            tracker.update(a)
                    break
            if not improved:
                stall_count += 1
                if stall_count >= stall_patience:
                    break

    return best_a


def qp_solve(a_init, deadline, max_rounds=20, top_k_init=30):
    """
    Cutting-plane QP: min t s.t. a^T Q_k a <= t, sum(a)=1, a>=0.
    Uses CVXPY with CLARABEL solver.
    Returns (best_a, best_t) or (None, None).
    """
    n = len(a_init)
    a_init = np.maximum(a_init, 0)
    a_init = a_init / a_init.sum()

    conv_init = fast_conv(a_init)
    t_best = float(conv_init.max())
    a_best = a_init.copy()

    # Initial active set: top-k most active constraints
    top_idx = np.argsort(conv_init)[::-1][:top_k_init]
    active_set = set(int(k) for k in top_idx)

    a_var = cp.Variable(n)

    for rnd in range(max_rounds):
        if time.monotonic() > deadline:
            break

        t_var = cp.Variable()
        constraints = [
            cp.sum(a_var) == 1,
            a_var >= 0,
        ]

        # Add active quadratic constraints
        for k in sorted(active_set):
            jl = max(0, k - n + 1)
            jh = min(n - 1, k)
            # Build quadratic form: sum_{i=jl}^{jh} a[i]*a[k-i]
            # Express as a_var.T @ Q_k @ a_var where Q_k[i,k-i] appropriately set
            # For CVXPY efficiency, useQuadForm or manual expression
            # Manual: sum of products
            quad_expr = cp.Constant(0)
            for i in range(jl, jh + 1):
                j = k - i
                if j != i:
                    quad_expr += 0.5 * a_var[i] * a_var[j] * 2  # = a[i]*a[j]
                else:
                    quad_expr += a_var[i] * a_var[i]
            constraints.append(quad_expr <= t_var)

        objective = cp.Minimize(t_var)
        prob = cp.Problem(objective, constraints)

        try:
            prob.solve(solver=cp.CLARABEL, verbose=False)
        except Exception:
            try:
                prob.solve(solver=cp.SCS, verbose=False)
            except Exception:
                break

        if a_var.value is None:
            break

        a_new = np.maximum(np.asarray(a_var.value).flatten(), 0.0)
        s = a_new.sum()
        if s < 1e-10:
            break
        a_new = a_new / s
        conv_new = fast_conv(a_new)
        t_new = float(conv_new.max())

        if t_new < t_best - 1e-12:
            t_best = t_new
            a_best = a_new.copy()

        # Check violation
        violations = conv_new - t_best
        max_viol = float(violations.max())
        if max_viol < 1e-8:
            break

        # Add most violated constraint
        worst_k = int(np.argmax(violations))
        if worst_k not in active_set:
            active_set.add(worst_k)

        # Cap active set size
        if len(active_set) > 200:
            # Keep only top 100 most active
            sorted_k = list(active_set)
            sorted_k.sort(key=lambda k: conv_new[k], reverse=True)
            active_set = set(sorted_k[:100])

    return a_best, t_best


def smooth_obj_and_grad(a, beta):
    """Log-sum-exp smooth approximation of max(conv(a,a)) and gradient."""
    n = len(a)
    b = fast_conv(a)
    b_max = float(b.max())
    shifted = beta * (b - b_max)
    shifted = np.clip(shifted, -500, 500)
    exp_s = np.exp(shifted)
    Z = float(exp_s.sum())
    if Z < 1e-300:
        p = np.zeros(2 * n - 1)
        p[np.argmax(b)] = 1.0
    else:
        p = exp_s / Z
    g_val = b_max + np.log(max(Z, 1e-300)) / beta

    # Gradient: d/d a_j of conv(a,a) weighted by p
    # conv(a,a)[k] = sum_i a[i]*a[k-i], so d/d a_j = sum_k p[k] * (a[k-j] + a[j-k]_valid)
    # = 2 * (a * p_rev)[j] approximately (for full range)
    grad = 2.0 * np.convolve(a, p[::-1], mode='full')[n - 1:2 * n - 1][::-1]
    return g_val, grad


def F_smooth(a, beta):
    """Smoothed J and gradient."""
    n = len(a)
    s = float(a.sum())
    if s < 1e-10:
        return 1e10, np.zeros(n)
    g_val, grad_g = smooth_obj_and_grad(a, beta)
    F_val = 2.0 * n * g_val / (s * s)
    grad_F = 2.0 * n * (s * grad_g - 2.0 * g_val) / (s ** 3)
    return F_val, grad_F


def lbfgs_smooth(a0, deadline, betas=None):
    """L-BFGS-B with beta annealing on smoothed objective."""
    if betas is None:
        betas = [(2.0, 30), (10.0, 40), (50.0, 50), (200.0, 60)]
    a = np.maximum(a0.astype(np.float64), 1e-10)
    n = len(a)

    for beta, max_iters in betas:
        if time.monotonic() > deadline:
            break

        def func(x):
            x2 = np.maximum(x, 0.0)
            f, _ = F_smooth(x2, beta)
            return f if np.isfinite(f) else 1e10

        def jac(x):
            x2 = np.maximum(x, 0.0)
            _, g = F_smooth(x2, beta)
            g = np.nan_to_num(g, nan=0.0, posinf=0.0, neginf=0.0)
            g[x2 < 1e-12] = 0.0
            return g

        bounds = [(0.0, None)] * n
        try:
            res = minimize(func, a.copy(), jac=jac, method='L-BFGS-B',
                           bounds=bounds,
                           options={'maxiter': max_iters, 'max_fun': max_iters * 25,
                                    'gtol': 1e-9, 'ftol': 1e-15})
            if res.fun < 1e10:
                new_a = np.maximum(res.x, 1e-10)
                if new_a.sum() > 1e-6:
                    a = new_a
        except Exception:
            pass

    return a


def resample(src, dest_n):
    """Resample 1D sequence to new length."""
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)

def generate_inits(n, rng, parent=None):
    """Generate diverse initial sequences."""
    inits = []
    t = np.linspace(0, 1, n, endpoint=False)

    if parent is not None and parent.size > 0:
        inits.append(resample(parent, n))
        p = resample(parent, n)
        inits.append(np.maximum(p + rng.normal(0, 0.03, n) * p, 0.01))
        inits.append(np.maximum(p * (1 + 0.1 * rng.standard_normal(n)), 0.01))

    inits.append(np.ones(n))
    inits.append(rng.uniform(0.5, 1.0, n))

    # Cosine families - key for flat autocorrelation
    for freq in [3, 5, 7, 11, 13]:
        inits.append(1.0 + 0.4 * np.cos(2.0 * np.pi * freq * t))
    # Multi-frequency
    inits.append(1.0 + 0.25 * np.cos(2 * np.pi * 3 * t) + 0.15 * np.cos(2 * np.pi * 7 * t))
    inits.append(1.0 + 0.2 * np.cos(2 * np.pi * 5 * t) + 0.15 * np.cos(2 * np.pi * 11 * t))

    inits.append(np.abs(np.sin(np.pi * t)) + 0.2)
    # Gaussian
    inits.append(np.exp(-3.0 * (t - 0.5) ** 2) + 0.1)
    # Power law
    inits.append((t + 0.05) ** 0.3)
    # Alternating
    inits.append(1.0 + 0.3 * np.cos(4 * np.pi * t))

    return inits

        tracker.report("Phase 2 primary")

        # Try at n=2048 if not already
        if n_work != 2048 and time.monotonic() < deadline:
            phase2b_deadline = min(time.monotonic() + 150.0, deadline)
            print(f"  LP MD at n=2048...", flush=True)
            a_2048 = resample(tracker.best, 2048)
            a_2048 = a_2048 / a_2048.sum()
            a_2048 = lp_mirror_descent(a_2048, 3000, phase2b_deadline, tracker=tracker,
                                        n_line_evals=15, stall_patience=40)
            tracker.report("Phase 2b n=2048")

    # === Phase 3: Cutting-plane QP refinement ===
    if tracker.best is not None and time.monotonic() < deadline:
        phase3_deadline = min(time.monotonic() + 300.0, deadline)
        n_qp = tracker.best.size
        print(f"\n{'='*50} Phase 3: QP Cutting Plane n={n_qp} {'='*50}", flush=True)

        a_qp, t_qp = qp_solve(tracker.best, phase3_deadline, max_rounds=25, top_k_init=50)
        if a_qp is not None:
            tracker.update(a_qp)

        # Second QP from best
        if tracker.best is not None and time.monotonic() < phase3_deadline:
            a_qp2, t_qp2 = qp_solve(tracker.best, phase3_deadline, max_rounds=20, top_k_init=30)
            if a_qp2 is not None:
                tracker.update(a_qp2)

        tracker.report("Phase 3")

    # === Phase 4: Multi-restart with diverse perturbations ===
    if tracker.best is not None and time.monotonic() < deadline:
        phase4_deadline = min(time.monotonic() + 300.0, deadline)
        n_best = tracker.best.size
        print(f"\n{'='*50} Phase 4: Restarts n={n_best} {'='*50}", flush=True)

        strategies = [
            ('gauss', 0.05), ('gauss', 0.1), ('gauss', 0.02),
            ('gauss', 0.15), ('gauss', 0.08),
            ('smooth', 0.1), ('smooth', 0.05),
            ('spike', 10), ('spike', 5),
            ('cosine', 3.0), ('cosine', 7.0),
        ]

        for pert_type, param in strategies:
            if time.monotonic() > phase4_deadline:
                break
            a = tracker.best.copy().astype(np.float64)
            n = len(a)
            t_arr = np.linspace(0, 1, n, endpoint=False)

            if pert_type == 'gauss':
                a = a + rng.normal(0, param, n) * a
            elif pert_type == 'smooth':
                noise = rng.normal(0, param, n)
                ksize = max(3, n // 32)
                a = a + np.convolve(noise, np.ones(ksize)/ksize, 'same') * a
            elif pert_type == 'spike':
                k = min(int(param), n // 4)
                idx = rng.choice(n, size=k, replace=False)
                a[idx] *= rng.uniform(0.3, 3.0, k)
            elif pert_type == 'cosine':
                a = a * (1.0 + 0.1 * np.cos(2*np.pi*param*t_arr))

            a = np.maximum(a, 1e-6)
            a = a / a.sum()
            J_pert = compute_J(a)
            if J_pert < tracker.best_score:
                tracker.update(a)

            # Short LP refinement
            budget = min(25.0, phase4_deadline - time.monotonic())
            if budget > 5:
                a = lp_mirror_descent(a, 500, time.monotonic() + budget,
                                       tracker=tracker, n_line_evals=12, stall_patience=20)

        tracker.report("Phase 4")

    # === Phase 5: Final polish ===
    if tracker.best is not None and time.monotonic() < deadline:
        remaining = deadline - time.monotonic()
        if remaining > 30:
            n_final = tracker.best.size
            print(f"\n{'='*50} Phase 5: Final Polish n={n_final} (remaining={remaining:.0f}s) {'='*50}", flush=True)

            # Extended LP
            a = lp_mirror_descent(tracker.best, 3000, deadline, tracker=tracker,
                                   n_line_evals=22, stall_patience=80)

            # Final QP
            if time.monotonic() < deadline - 10:
                a_qp, t_qp = qp_solve(tracker.best, deadline, max_rounds=15, top_k_init=40)
                if a_qp is not None:
                    tracker.update(a_qp)

            # Try one more length
            for n_try in [n_final + 128, n_final - 128, 1536]:
                if time.monotonic() > deadline - 20:
                    break
                if n_try < 128 or n_try > 4096 or abs(n_try - n_final) < 64:
                    continue
                a_res = resample(tracker.best, n_try)
                a_res = a_res / a_res.sum()
                a_res = lp_mirror_descent(a_res, 800, deadline, tracker=tracker,
                                           n_line_evals=15, stall_patience=30)
                tracker.update(a_res)

            tracker.report("Phase 5")

    # Fallback
    if tracker.best is None:
        tracker.best = np.ones(1024)
        tracker.best_score = compute_J(tracker.best)

    total_time = time.monotonic() - total_start
    print(f"\n{'='*50} FINAL {'='*50}", flush=True)
    print(f"J={tracker.best_score:.8f} n={tracker.best.size} time={total_time:.1f}s", flush=True)

    result = np.clip(tracker.best, 0.0, 1000.0)
    if result.sum() < 0.01:
        result = np.ones(128)
    return [float(x) for x in result]
