Minimizes J(a) = 2*n*max(conv(a,a))/(sum(a))^2 using:

4. LP mirror descent for precise optimization
5. Multi-length, multi-start exploration

import numpy as np
from scipy.optimize import linprog, minimize
from scipy.sparse import csr_matrix
import time


def fast_conv(a):
    """Full self-convolution via FFT."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]


def compute_J(a):
    """Compute J(a) = 2*n*max(conv(a,a))/(sum(a))^2."""
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))


def spectral_flatten(a, num_iters, alpha=0.1, momentum=0.5, deadline=None):
    """
    Gerchberg-Saxton style spectral flattening with non-negativity projection.
    """
    n = len(a)
    a = np.maximum(a.astype(np.float64), 1e-10)
    a = a / a.sum()
    N = 1
    while N < 2 * n - 1:
        N <<= 1
    
    best_a = a.copy()
    best_M = fast_conv(a).max()
    current = a.copy()
    
    for it in range(num_iters):
        if deadline is not None and time.monotonic() > deadline:
            break
        
        # FFT
        A = np.fft.fft(current, N)
        mag = np.abs(A)
        phase = np.angle(A)
        
        # Target: flat magnitude
        target_mag = np.mean(mag)
        
        # Log-domain blending
        log_mag = np.log(mag + 1e-15)
        log_target = np.log(target_mag + 1e-15)
        log_mag_new = (1 - alpha) * log_mag + alpha * log_target
        mag_new = np.exp(log_mag_new)
        
        # Reconstruct
        A_new = mag_new * np.exp(1j * phase)
        a_new = np.real(np.fft.ifft(A_new))[:n]
        
        # Project to non-negative and normalize
        a_new = np.maximum(a_new, 0)
        s = a_new.sum()
        if s < 1e-10:

        if s > 1e-10:

            if M < best_M:

                best_a = current.copy()

def smooth_max_gradient(a, tau):
    """Compute gradient of smooth max of self-convolution."""
    n = len(a)
    b = fast_conv(a)
    b_max = b.max()
    exp_b = np.exp(tau * (b - b_max))
    p = exp_b / exp_b.sum()
    
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        pk = p[k]
        if pk < 1e-13:
            continue
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        grad[lo:hi + 1] += 2.0 * pk * a[k - hi:k - lo + 1][::-1]
    
    return grad, p, b


def run_smooth_gradient(a, max_iters, deadline, initial_tau=2.0, final_tau=80.0,
                        lr_init=0.3, lr_min=0.01):
    """Smooth gradient descent with adaptive tau."""
    a = np.maximum(a.astype(np.float64), 1e-8)
    a = a / a.sum()
    n = len(a)
    
    best_a = a.copy()
    best_M = fast_conv(a).max()
    v = np.zeros(n)
    
    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        
        frac = it / max(1, max_iters)
        tau = initial_tau + (final_tau - initial_tau) * (frac ** 0.5)
        lr = lr_init * (lr_min / lr_init) ** frac
        
        grad, p, b = smooth_max_gradient(a, tau)
        M = b.max()
        
        if M < best_M * (1 - 1e-8):
            best_M = M
            best_a = a.copy()
        
        beta = 0.9
        v = beta * v + lr * grad
        a_new = a - v
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s < 1e-10:
            a_new = a.copy()
        else:
            a_new = a_new / s
        a = a_new
    
    return best_a


def build_A_sparse(n, a):
    """Build sparse constraint matrix for LP."""
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        j_min = max(0, k - n + 1)
        j_max = min(n - 1, k)
        for j in range(j_min, j_max + 1):
            rows.append(k)
            cols.append(j)
            vals.append(a[k - j])
    A = csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))
    return A


def lp_best_response(a, M):
    """LP: max sum(g) s.t. (a*g)[k] <= M for all k, g >= 0."""
    n = len(a)
    A = build_A_sparse(n, a)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    try:
        r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                    options={'time_limit': 30})
        if r.success and r.x is not None:
            g = np.maximum(r.x, 0)
            if g.sum() > 1e-15:
                return g
    except Exception:
        pass
    return None


def run_lp_mirror_descent(a, max_iters, deadline):
    """LP mirror descent with line search."""
    a = np.maximum(a.astype(np.float64), 1e-8)
    a = a / a.sum()
    best_a = a.copy()
    best_M = fast_conv(a).max()
    no_improve = 0
    
    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        
        b = fast_conv(a)
        M = float(b.max())
        g = lp_best_response(a, M)
        if g is None:
            break
        g = np.maximum(g, 0.0)
        g_sum = g.sum()
        if g_sum < 1e-10:
            break
        g /= g_sum
        
        improved = False
        base_t = max(0.05, 0.5 * (1 - 0.3 * it / max(1, max_iters)))
        for t in [base_t, base_t * 0.5, base_t * 0.25, 0.02, 0.01]:
            a_trial = np.maximum((1 - t) * a + t * g, 0)
            s = a_trial.sum()
            if s < 1e-10:
                continue
            a_trial /= s
            M_trial = fast_conv(a_trial).max()
            if M_trial < best_M - 1e-14:
                a = a_trial
                best_M = M_trial
                best_a = a.copy()
                improved = True
                break
        
        if not improved:
            no_improve += 1
            if no_improve > 30:
                break
        else:
            no_improve = 0
    
    return best_a


def resample_seq(src, dest_n):
    """Resample 1D sequence to new length."""
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)


def generate_u_shape(n, curvature, rng):
    """Generate U-shaped (convex) sequence."""
    t = np.linspace(0, 1, n)
    # U-shape: 1 + c*(2t-1)^2
    a = 1.0 + curvature * (2 * t - 1) ** 2
    # Add some noise
    a += rng.normal(0, 0.05, n)
    a = np.maximum(a, 0.01)
    return a / a.sum()

    inits.append(np.maximum(1.0 + 1.0 * np.cos(np.pi * t), 0.01))  # Stronger U
    inits.append(np.maximum(1.0 + 2.0 * np.cos(np.pi * t), 0.01))  # Very strong U

        inits.append(np.maximum(1.0 + 0.3 * np.cos(2 * np.pi * freq * t), 0.01))
    
    # Gaussian bumps at edges
    inits.append(np.exp(-5.0 * t**2) + np.exp(-5.0 * (1-t)**2) + 0.1)
    inits.append(np.exp(-10.0 * t**2) + np.exp(-10.0 * (1-t)**2) + 0.05)

    """Mirror descent on the simplex for minimizing max(self-conv)."""
    n = len(a)
    a = np.maximum(a.astype(np.float64), 1e-10)
    a = a / a.sum()
    
    best_a = a.copy()
    best_M = fast_conv(a).max()
    
    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        
        b = fast_conv(a)
        M = b.max()
        
        if M < best_M:
            best_M = M
            best_a = a.copy()
        
        # Compute subgradient with softmax weighting
        tau = 50.0
        b_shifted = b - b.max()
        p = np.exp(tau * b_shifted)
        p /= p.sum()
        
        # Subgradient: g[i] = sum_k 2*p[k]*a[k-i]
        g = np.zeros(n)
        for k in range(2 * n - 1):
            if p[k] < 1e-12:
                continue
            lo = max(0, k - n + 1)
            hi = min(n - 1, k)
            g[lo:hi+1] += 2.0 * p[k] * a[k-hi:k-lo+1][::-1]

        a = a * np.exp(-lr * g)
        a = np.maximum(a, 1e-15)
        a = a / a.sum()
    
    return best_a
