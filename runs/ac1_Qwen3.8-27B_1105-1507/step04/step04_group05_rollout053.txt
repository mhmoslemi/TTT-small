def fast_conv(a):

    b = fast_conv(a)

    if len(b) == 0:

    bm = b.max()

    lse = bm + (1.0 / beta) * np.log(np.sum(np.exp(beta * (b - bm))))
    # Wait, logsumexp(beta * b) = beta * bm + log(sum(exp(beta*(b-bm))))
    # And (1/beta)*logsumexp(beta*b) = bm + (1/beta)*log(sum(exp(beta*(b-bm))))

def coordinate_descent_pass(a, max_evals_per_coord=100):
    """One pass of coordinate descent. For each coordinate, find optimal value."""
    a = np.maximum(a.copy(), 0)
    n = len(a)
    S = a.sum()
    if S < 1e-10:
        return a
    b = fast_conv(a)
    current_J = 2.0 * n * b.max() / (S * S)
    
    for i in range(n):
        ai = a[i]
        # Try a range of values for a[i]
        # The optimal value is likely in [0, 3*ai] or so
        # But also could be larger if the coordinate should be increased
        hi = max(3.0 * ai, S / n * 3.0, 0.01)
        
        # Evaluate J at several points
        candidates = np.concatenate([
            np.linspace(0, hi, max_evals_per_coord),
            [hi * 1.5, hi * 2.0]
        ])
        
        best_val = ai
        best_J = current_J
        
        for x in candidates:
            if x < 0:
                continue
            new_S = S - ai + x
            if new_S < 1e-10:
                continue
            # Compute new convolution max
            # Only entries in [i, i+n-1] change
            new_b = b.copy()
            for k in range(max(0, i), min(2*n-1, i+n)):
                ki = k - i
                if 0 <= ki < n:
                    if k == 2*i:
                        new_b[k] = b[k] - ai*ai + x*x
                    else:
                        new_b[k] = b[k] - 2.0*ai*a[ki] + 2.0*x*a[ki]
            max_b = new_b.max()
            J = 2.0 * n * max_b / (new_S * new_S)
            if J < best_J:
                best_J = J
                best_val = x
        
        if best_val != ai:
            # Update convolution
            old_val = a[i]
            a[i] = best_val
            S = S - old_val + best_val
            for k in range(max(0, i), min(2*n-1, i+n)):
                ki = k - i
                if 0 <= ki < n:
                    if k == 2*i:
                        b[k] = b[k] - old_val*old_val + best_val*best_val
                    else:
                        b[k] = b[k] - 2.0*old_val*a[ki] + 2.0*best_val*a[ki]
            current_J = best_J
    
    return a


def coordinate_descent_fast(a, n_passes=5, max_evals=50):
    """Multiple passes of coordinate descent."""
    a = np.maximum(a.copy(), 0)
    a /= a.sum()
    best_J = compute_J(a)
    
    for p in range(n_passes):
        a_new = coordinate_descent_pass(a, max_evals)
        new_J = compute_J(a_new)
        if new_J < best_J:
            a = a_new
            best_J = new_J
        else:
            break
    
    return a, best_J


def solve_kkt_system(n, active_set):
    """
    Solve the KKT system for a given active set.
    Active set S: set of indices k where c_k = max(c).
    
    KKT: B * a = lambda * m * 1, where B[j,l] = 1 if j+l in S
    Plus: sum(a) = 1, a >= 0
    """
    m = len(active_set)
    S_set = set(active_set)
    
    # Build B matrix: B[j,l] = 1 if j+l in active_set
    B = np.zeros((n, n))
    for j in range(n):
        for k in active_set:
            l = k - j
            if 0 <= l < n:
                B[j, l] = 1.0
    
    # System: [B  -m*1; 1^T  0] [a; lambda] = [m*lambda*1... wait
    # Actually: B*a = lambda*m*1 and 1^T*a = 1
    # Rearranging: B*a - m*lambda*1 = 0 and 1^T*a = 1
    # Matrix: [B  -m*1; 1^T  0] [a; lambda] = [0; 1]
    
    A = np.zeros((n + 1, n + 1))
    A[:n, :n] = B
    A[:n, n] = -m
    A[n, :n] = 1.0
    
    rhs = np.zeros(n + 1)
    rhs[n] = 1.0
    
    try:
        sol = np.linalg.solve(A, rhs)
    except np.linalg.LinAlgError:
        # Use least squares
        sol, _, _, _ = np.linalg.lstsq(A, rhs, rcond=None)
    
    a = sol[:n]
    
    # Check feasibility
    if np.any(a < -1e-10):
        return None, None
    
    a = np.maximum(a, 0)
    S = a.sum()
    if S < 1e-10:
        return None, None
    a /= S
    
    # Verify: compute actual convolution and check
    b = fast_conv(a)
    max_b = b.max()
    
    # Check that active set entries are at the max
    for k in active_set:
        if b[k] < max_b * (1 - 1e-6):
            return None, None
    
    # Check that non-active entries are <= max
    for k in range(2*n-1):
        if k not in S_set and b[k] > max_b * (1 + 1e-6):
            return None, None
    
    J = 2.0 * n * max_b / (a.sum()**2)
    return a, J


def generate_inits(n, rng):
    """Generate diverse initial sequences."""
    inits = []
    t = np.linspace(0, 1, n, endpoint=False)
    
    inits.append(np.ones(n))
    inits.append(rng.uniform(0.3, 1.0, n).astype(np.float64))
    
    # Tapered sequences
    inits.append(1.0 + 0.5 * np.cos(np.pi * t))
    inits.append(np.cos(np.pi * t / 2)**2 + 0.1)
    inits.append(1.0 / (1.0 + t) + 0.3)
    
    # Periodic
    for freq in [2, 3, 5]:
        inits.append(1.0 + 0.3 * np.cos(2*np.pi*freq*t))
    
    # Two-lobe
    inits.append(0.5 * np.exp(-8*(t-0.25)**2) + 0.5 * np.exp(-8*(t-0.75)**2) + 0.1)
    
    # Edge-heavy
    inits.append(1.0 + 0.5 * (2*t - 1)**2)
    inits.append(1.0 / (0.1 + np.abs(2*t - 1)))
    
    return inits


def lbfgs_smoothed(a, deadline, tracker, beta_start=1.0, beta_end=500.0, n_anneal=10):
    """
    L-BFGS-B optimization with log-sum-exp smoothing and beta annealing.
    """
    a = np.maximum(a.astype(np.float64), 0.001)
    a /= a.sum()
    n = len(a)
    current_J = compute_J(a)
    tracker.update(a)
    
    # Annealing schedule
    betas = np.geomspace(beta_start, beta_end, n_anneal)
    
    for beta in betas:
        if time.monotonic() > deadline:
            break
        if beta > 10:
            time.perf_counter()  # check deadline more frequently
        
        # L-BFGS-B optimization at this beta
        best_x = a.copy()
        best_val = compute_J(a)
        
        for attempt in range(3):
            if time.monotonic() > deadline:
                break
            
            def obj(x):
                return smoothed_objective(x, beta, n)
            
            def grad(x):
                return smoothed_gradient(x, beta)
            
            result = minimize(
                obj, best_x, method='L-BFGS-B', jac=grad,
                options={'maxiter': 100, 'ftol': 1e-12, 'gtol': 1e-8}
            )
            
            if result.x is not None:
                x_new = np.maximum(result.x, 0)
                J_new = compute_J(x_new)
                if J_new < best_val:
                    best_val = J_new
                    best_x = x_new
                # For next attempt, start from best so far
                a = best_x.copy()
        
        a = best_x
        if compute_J(a) < current_J:
            current_J = compute_J(a)
            tracker.update(a)
    
    tracker.update(a)
    return a


def resample_seq(src, dest_n):
    """Resample 1D sequence to new length via linear interpolation."""
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)


class ScoreTracker:
    def __init__(self):
        self.best = None
        self.best_score = float('inf')
        self.t_start = time.monotonic()
    
    def update(self, a):
        a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
        if a.size == 0 or a.sum() < 0.01:
            return
        score = float(evaluate_sequence(a.tolist()))
        if np.isfinite(score) and score < self.best_score:
            self.best = a.copy()
            self.best_score = score
            elapsed = time.monotonic() - self.t_start
            print(f"  * NEW BEST: J={score:.8f} n={a.size} t={elapsed:.0f}s", flush=True)
    
    def report(self, label=""):
        if self.best is not None:
            print(f"  {label} best: J={self.best_score:.8f} n={self.best.size} t={time.monotonic()-self.t_start:.0f}s", flush=True)

Wait, I have a problem. The `coordinate_descent_pass` function is O(n * max_evals * n) = O(n^2 * max_evals) because for each of the n coordinates, we evaluate J at max_evals points, and each evaluation requires updating the convolution (O(n) for the affected entries) and finding the max (O(n)). So per pass: n * max_evals * n = n^2 * max_evals.

For n=386, max_evals=100: 386^2 * 100 ≈ 15 million operations per pass. That's actually fine for a single pass, but if I do many passes, it adds up.

Actually, let me reconsider. For each coordinate i, I need to evaluate J at several x values. For each x, I need to:
1. Update the convolution entries in [i, i+n-1]: O(n)
2. Find the max of the updated convolution: O(n) (but I can track the max incrementally)

Wait, I don't need to recompute the full convolution each time. I can just track the max. When I update a_i from old to new:
- Entries in [i, i+n-1] change
- The new max is max(old_max_excluding_affected, max_of_new_affected_entries)

So I can maintain a sorted structure or just recompute the max over the affected entries and compare with the stored max of unaffected entries.

Actually, the simplest approach: for each candidate x, compute the new values of the affected entries and find their max, then compare with the stored max of the unaffected entries.

Let me optimize this:

```python
def coordinate_descent_pass_v2(a, max_evals=80):
    """Optimized coordinate descent pass."""
    a = np.maximum(a.copy(), 0)
    n = len(a)
    S = a.sum()
    if S < 1e-10:
        return a
    b = fast_conv(a)
    current_J = 2.0 * n * b.max() / (S * S)
    
    for i in range(n):
        ai = a[i]
        if ai < 1e-12:
            continue
        
        # Precompute the max of b outside [i, i+n-1]
        lo_k = max(0, i)
        hi_k = min(2*n-1, i+n-1)
        
        # Max of b outside the affected range
        if lo_k > 0 or hi_k < 2*n-2:
            outside_mask = np.ones(2*n-1, dtype=bool)
            outside_mask[lo_k:hi_k+1] = False
            max_outside = b[outside_mask].max() if outside_mask.any() else 0
        else:
            max_outside = 0.0
        
        # Range of values to try
        hi = max(3.0 * ai, S / n * 3.0, 0.01)
        candidates = np.linspace(0, hi, max_evals)
        
        best_val = ai
        best_J = current_J
        
        for x in candidates:
            new_S = S - ai + x
            if new_S < 1e-10:
                continue
            
            # Compute max of affected entries
            # For k in [lo_k, hi_k]:
            #   if k == 2*i: new_c_k = b[k] - ai^2 + x^2
            #   else: new_c_k = b[k] - 2*ai*a[k-i] + 2*x*a[k-i]
            
            max_affected = 0.0
            for k in range(lo_k, hi_k + 1):
                ki = k - i
                if 0 <= ki < n:
                    if k == 2 * i:
                        ck = b[k] - ai * ai + x * x
                    else:
                        ck = b[k] - 2.0 * ai * a[ki] + 2.0 * x * a[ki]
                else:
                    ck = b[k]
                if ck > max_affected:
                    max_affected = ck
            
            max_b = max(max_affected, max_outside)
            J = 2.0 * n * max_b / (new_S * new_S)
            if J < best_J:
                best_J = J
                best_val = x
        
        if best_val != ai:
            old_val = a[i]
            a[i] = best_val
            S = S - old_val + best_val
            for k in range(lo_k, hi_k + 1):
                ki = k - i
                if 0 <= ki < n:
                    if k == 2 * i:
                        b[k] = b[k] - old_val*old_val + best_val*best_val
                    else:
                        b[k] = b[k] - 2.0*old_val*a[ki] + 2.0*best_val*a[ki]
            current_J = best_J
    
    return a
```

This is still O(n * max_evals * n) per pass, but the constant is small. For n=386, max_evals=80: 386 * 80 * 386 ≈ 12 million inner loop iterations per pass. Each iteration is just a few floating point operations. In Python, this would be slow (~1-2 seconds per pass). With 30 passes, that's 30-60 seconds. That's acceptable.

But wait, I can vectorize the inner loop! For a given i and candidate x, the affected entries are:
- For k in [lo_k, hi_k], k ≠ 2i: new_c_k = b[k] + 2*(x-ai)*a[k-i]
- For k = 2i: new_c_k = b[k] + (x-ai)*(x+ai)

So for a vector of candidates x, I can compute all the new c_k values at once!

Let me vectorize:

```python
def coordinate_descent_pass_v3(a, max_evals=80):
    """Vectorized coordinate descent pass."""
    a = np.maximum(a.copy(), 0)
    n = len(a)
    S = a.sum()
    if S < 1e-10:
        return a
    b = fast_conv(a)
    current_J = 2.0 * n * b.max() / (S * S)
    
    for i in range(n):
        ai = a[i]
        if ai < 1e-12:
            continue
        
        lo_k = max(0, i)
        hi_k = min(2*n-1, i+n-1)
        
        # Max outside affected range
        if hi_k < 2*n-2:
            max_outside_right = b[hi_k+1:].max() if hi_k+1 < 2*n-1 else 0
        else:
            max_outside_right = 0
        if lo_k > 0:
            max_outside_left = b[:lo_k].max() if lo_k > 0 else 0
        else:
            max_outside_left = 0
        max_outside = max(max_outside_left, max_outside_right)
        
        # Get the affected b values and corresponding a indices
        ks = np.arange(lo_k, hi_k + 1)
        kis = ks - i
        valid = (kis >= 0) & (kis < n)
        
        # For vectorized computation:
        # new_c[k] = b[k] + 2*(x-ai)*a[k-i] for k != 2i
        # new_c[2i] = b[2i] + (x-ai)*(x+ai) if 2i is in range
        
        b_affected = b[lo_k:hi_k+1]
        a_affected = a[kis[valid]]
        
        # Candidates
        hi = max(3.0 * ai, S / n * 3.0, 0.01)
        candidates = np.linspace(0, hi, max_evals)
        
        # For each candidate x, compute new c_k values
        # new_c_k(x) = b_k + 2*(x - ai)*a_{k-i}  (vectorized over k)
        # delta = x - ai
        deltas = candidates[:, None] - ai  # shape: (max_evals, 1)
        # new_c = b_affected[None, :] + 2 * deltas * a_affected[None, :]
        # But for k = 2i, it's different: new_c = b_2i + (x-ai)*(x+ai) = b_2i + delta*(delta + 2*ai)
        
        new_c = b_affected[None, :] + 2.0 * deltas * a_affected[None, :]
        
        # Fix the k=2i entry if it's in range
        if lo_k <= 2*i <= hi_k:
            idx_2i = 2*i - lo_k
            new_c[:, idx_2i] = b[2*i] + deltas[:, 0] * (deltas[:, 0] + 2*ai)
        
        max_affected = new_c.max(axis=1)  # shape: (max_evals,)
        max_b = np.maximum(max_affected, max_outside)  # shape: (max_evals,)
        
        new_S = S - ai + candidates  # shape: (max_evals,)
        valid_S = new_S > 1e-10
        J_vals = np.full(max_evals, 1e10)
        J_vals[valid_S] = 2.0 * n * max_b[valid_S] / (new_S[valid_S] ** 2)
        
        best_idx = np.argmin(J_vals)
        best_val = candidates[best_idx]
        best_J = J_vals[best_idx]
        
        if best_J < current_J - 1e-12:
            old_val = a[i]
            a[i] = best_val
            S = S - old_val + best_val
            # Update b
            for k in range(lo_k, hi_k + 1):
                ki = k - i
                if 0 <= ki < n:
                    if k == 2 * i:
                        b[k] = b[k] - old_val*old_val + best_val*best_val
                    else:
                        b[k] = b[k] - 2.0*old_val*a[ki] + 2.0*best_val*a[ki]
            current_J = best_J
    
    return a
```

This is much faster! The inner loop is now vectorized. For each coordinate, we do one (max_evals × n) matrix operation, which is fast with numpy.

Let me also think about the L-BFGS-B approach. The issue with the smoothed objective is that for large β, the log-sum-exp is very peaked (essentially the max), and the gradient is concentrated on the argmax. This makes the optimization hard.

A better approach might be to use a "soft" max that's differentiable:
soft_max_β(b) = (1/β) * log(sum(exp(β*b)))

For β = 10, this is a reasonable approximation to max. For β = 100, it's very close to max.

The gradient of soft_max_β(b) w.r.t. b is the softmax: softmax(β*b) = exp(β*b) / sum(exp(β*b)).

So the gradient of the smoothed objective w.r.t. a_j is:
sum_k softmax(β*b)_k * ∂b_k/∂a_j = sum_k softmax(β*b)_k * a_{k-j}

This is a convolution! I can compute it efficiently.
