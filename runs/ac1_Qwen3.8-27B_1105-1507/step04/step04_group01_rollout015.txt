import numpy as np

def softmax(z):

def coordinate_descent_step(a, idx, conv, n):
    """
    For coordinate idx, find optimal value x >= 0 that minimizes
    2*n*max_k conv_k(x) / (S+x)^2 where conv_k(x) is the k-th element
    of the convolution when a[idx] = x.
    """
    S = float(np.sum(a)) - a[idx]  # sum of all other elements
    x0 = a[idx]
    
    if S < 1e-15:
        return 0.0  # Can't normalize, keep at 0 or handle separately
    
    # Quadratic term: (a*a)[2*idx] = x^2 + C_q
    C_q = None
    if 2 * idx < 2 * n - 1:
        C_q = conv[2 * idx] - x0 * x0
    
    # Linear terms: for k in [idx, idx+n-1], k != 2*idx
    # (a*a)[k] = 2*a[k-idx]*x + C_k
    slopes = []
    intercepts = []
    for k in range(idx, min(idx + n, 2 * n - 1)):
        if k == 2 * idx:
            continue
        ki = k - idx  # this is the index in a that multiplies x
        if 0 <= ki < n:
            slopes.append(2.0 * a[ki])
            intercepts.append(conv[k] - 2.0 * x0 * a[ki])
    
    slopes = np.array(slopes)
    intercepts = np.array(intercepts)
    
    def f_of_x(x):
        # Compute max of all conv values
        candidates = []
        if C_q is not None:
            candidates.append(x * x + C_q)
        if len(slopes) > 0:
            candidates.append(float(np.max(slopes * x + intercepts)))
        # Constant terms (k outside [idx, idx+n-1])
        # These don't change with x
        if len(candidates) == 0:
            max_val = 0.0
        else:
            max_val = max(candidates)
        
        # Also check constant terms
        # Actually all k are either in [idx, idx+n-1] or not
        # For k not in [idx, idx+n-1], conv[k] is constant
        # We need to include them in the max
        # Let's precompute the max of constant terms
        # Actually, let's just compute the full max including constants
        
        return 2.0 * n * max_val / ((S + x) * (S + x))
    
    # Better approach: precompute the max of constant terms
    # k values not in [idx, min(idx+n, 2n-1)]
    in_range = set(range(idx, min(idx + n, 2 * n - 1)))
    const_vals = []
    for k in range(2 * n - 1):
        if k not in in_range:
            const_vals.append(conv[k])
    const_max = max(const_vals) if const_vals else -np.inf
    
    def f_of_x2(x):

        if C_q is not None:
            vals.append(x * x + C_q)
        if len(slopes) > 0:

        max_val = max(vals)
        if max_val < 0:
            max_val = 0.0
        return 2.0 * n * max_val / ((S + x) * (S + x))
    
    # Find minimum using bounded scalar optimization
    x_upper = max(4.0 * S, 1.0)

                              options={'maxiter': 200, 'xatol': 1e-12})
        return max(0.0, res.x)
    except Exception:
        return x0


def coordinate_descent(a, max_passes=100, time_budget=10.0, verbose=False):
    """
    Full coordinate descent: iterate over all coordinates, optimizing each.
    """
    n = len(a)
    a = a.copy().astype(np.float64)
    s = np.sum(a)
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    
    for pass_num in range(max_passes):
        if time.monotonic() - start > time_budget:
            break
        
        conv = compute_conv(a)
        improved = False
        
        for idx in range(n):
            new_x = coordinate_descent_step(a, idx, conv, n)
            if abs(new_x - a[idx]) > 1e-14:
                a[idx] = new_x
                improved = True
                # Update conv incrementally? For small n, just recompute
                # Actually for small n (<=30), recomputing is fine
                # But for correctness, we should update conv after each change
                # For speed with small n, we recompute conv after each coordinate update
                conv = compute_conv(a)
        
        # Renormalize
        s = np.sum(a)
        if s > 1e-15:
            a = a / s
        
        obj = compute_obj(a)
        if obj < best_obj:
            best_obj = obj
            best_a = a.copy()
        
        if not improved:
            break
    
    return best_a


def solve_lp_minpeak_small(a_init, time_limit=10):
    """
    Solve LP: min M s.t. (a*g)[k] <= M, sum(g) = 1, g >= 0.
    This gives the optimal "response" g to the current a.
    For small n (<=50), this is fast.
    """
    n = len(a_init)
    a = a_init.copy()
    a = a / np.sum(a)
    
    # Build constraint matrix: A[i, j] gives coefficient of g[j] in (a*g)[i]
    # (a*g)[k] = sum_j a[k-j]*g[j] (for valid k-j)
    # Actually (a*g)[k] = sum_j a[j]*g[k-j]
    # Let's use: (a*g)[k] = sum_{j=0}^{n-1} a[j]*g[k-j] where g[m]=0 for m outside [0,n-1]
    # = sum_{j: 0<=k-j<=n-1} a[j]*g[k-j]
    # = sum_{j: max(0,k-n+1) <= j <= min(n-1,k)} a[j]*g[k-j]
    
    # Let me build it as: for each output k, the coefficient of g[j] is a[k-j] (if valid)
    A = np.zeros((2 * n - 1, n))
    for k in range(2 * n - 1):
        for j in range(n):
            ki = k - j
            if 0 <= ki < n:
                A[k, j] = a[ki]
    
    # LP: min M
    # Variables: g[0..n-1], M
    # s.t. A @ g - M <= 0 (2n-1 constraints)
    #      sum(g) = 1
    #      g >= 0, M >= 0
    
    c = np.zeros(n + 1)
    c[-1] = 1.0  # minimize M
    
    A_ub = np.zeros((2 * n - 1, n + 1))
    A_ub[:, :n] = A
    A_ub[:, n] = -1.0  # A@g - M <= 0
    b_ub = np.zeros(2 * n - 1)
    
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])
    
    bounds = [(0, None)] * n + [(0, None)]
    
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs',
                      options={'time_limit': time_limit})
        if res.status == 0:
            g = np.maximum(res.x[:n], 0)
            s = np.sum(g)
            if s > 1e-15:
                g = g / s
                M_opt = float(res.x[n])
                return g, M_opt
    except Exception:
        pass
    return None, None


def solve_lp_maxsum(a, M, time_limit=10):
    """Solve LP: max sum(g) s.t. (a*g)[k] <= M, g >= 0."""
    n = len(a)
    a = a / np.sum(a)
    
    A = np.zeros((2 * n - 1, n))
    for k in range(2 * n - 1):
        for j in range(n):
            ki = k - j
            if 0 <= ki < n:
                A[k, j] = a[ki]
    
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    
    try:
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'time_limit': time_limit})
        if res.status == 0:
            g = np.maximum(res.x, 0)
            s = np.sum(g)
            if s > 1e-15:
                return g / s, s
    except Exception:
        pass
    return None, None


def alternating_lp_small(a, max_iters=50, time_budget=15.0):
    """Alternating LP for small sequences."""
    n = len(a)
    a = a.copy().astype(np.float64)
    s = np.sum(a)
    if s < 1e-15:
        return a
    a = a / s
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    
    for it in range(max_iters):
        if time.monotonic() - start > time_budget:
            break
        
        conv = compute_conv(a)
        M = float(np.max(conv))
        
        g, _ = solve_lp_maxsum(a, M, time_limit=5)
        if g is None:
            break
        
        # Line search
        improved = False
        for t in [0.1, 0.2, 0.3, 0.5, 0.7, 1.0]:
            a_try = (1.0 - t) * a + t * g
            a_try = np.maximum(a_try, 0)
            s_try = np.sum(a_try)
            if s_try < 1e-15:
                continue
            a_try = a_try / s_try
            obj_try = compute_obj(a_try)
            if obj_try < best_obj - 1e-12:
                best_obj = obj_try
                best_a = a_try.copy()
                a = a_try.copy()
                improved = True
                break
        
        if not improved:
            a = 0.8 * a + 0.2 * g
            a = np.maximum(a, 0)
            s = np.sum(a)
            if s > 1e-15:
                a = a / s
    
    return best_a


def optimize_small(n, n_inits=5, cd_time=5.0, lp_time=5.0, seed=42):
    """
    Optimize a sequence of small length n using coordinate descent + LP.
    Returns the best sequence and its objective.
    """
    best_a = None
    best_obj = np.inf
    rng = np.random.default_rng(seed)
    
    for init_idx in range(n_inits):
        # Generate initialization
        if init_idx == 0:
            a = np.ones(n) / n
        elif init_idx == 1:
            a = rng.dirichlet(np.ones(n))
        elif init_idx == 2:
            # Cosine initialization
            k = np.arange(n)
            f = rng.uniform(0.5, 2.0)
            a = 1.0 + 0.5 * np.cos(2 * np.pi * f * k / n + rng.uniform(0, 2 * np.pi))
            a = np.maximum(a, 0.01)
            a = a / np.sum(a)
        elif init_idx == 3:
            # Two-bump
            a = np.ones(n)
            center = n * rng.uniform(0.3, 0.7)
            a += 0.5 * np.exp(-5.0 * ((np.arange(n) - center) / n) ** 2)
            a = a / np.sum(a)
        else:
            # Random with structure
            a = rng.random(n)
            # Apply low-pass
            win = max(3, n // 5)
            kernel = np.ones(win) / win
            a = np.convolve(a, kernel, mode='same')
            a = np.maximum(a, 0.01)
            a = a / np.sum(a)
        
        # Coordinate descent
        a = coordinate_descent(a, max_passes=50, time_budget=cd_time)
        obj = compute_obj(a)
        
        # LP polishing
        if obj < best_obj:
            a = alternating_lp_small(a, max_iters=30, time_budget=lp_time)
            obj = compute_obj(a)
            
            # More coordinate descent after LP
            a = coordinate_descent(a, max_passes=30, time_budget=cd_time)
            obj = compute_obj(a)
        
        if obj < best_obj:
            best_obj = obj
            best_a = a.copy()
    
    return best_a, best_obj


def kronecker_product(a, b):
    """Kronecker product of two 1D sequences."""
    return np.kron(a, b)


def try_parent():
    """Try to load parent sequence."""
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64)
        parent = np.clip(parent, 0, 1000.0)
        if parent.size > 0 and np.isfinite(parent).all():
            obj = compute_obj(parent)
            print(f"Parent (n={parent.size}): obj={obj:.6f}")
            return parent.copy(), obj
    except Exception:

    return None, np.inf


def subgradient_optimize(a, max_iters=2000, time_budget=15.0, tau_start=3.0, tau_end=50.0):
    """Smoothed subgradient optimization for larger sequences."""
    n = len(a)
    a = a.copy().astype(np.float64)
    s = np.sum(a)
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    step = 0.1
    stall = 0
    
    for it in range(max_iters):
        if time.monotonic() - start > time_budget:
            break
        
        frac = it / max(1, max_iters)
        tau = tau_start + (tau_end - tau_start) * frac
        
        conv = compute_conv(a)
        max_conv = float(np.max(conv))
        shifted = np.clip(tau * (conv - max_conv), -300, 300)
        weights = np.exp(shifted)
        w_sum = np.sum(weights)
        if w_sum < 1e-300:
            idx = int(np.argmax(conv))
            weights = np.zeros(len(conv))
            weights[idx] = 1.0
        else:
            weights /= w_sum
        
        w_rev = weights[::-1]
        conv_fw = fft_convolve(a, w_rev)
        grad_a = 2.0 * conv_fw[n - 1:2 * n - 1][::-1]
        
        grad_centered = grad_a - np.mean(grad_a)
        a_new = a - step * grad_centered
        a_new = np.maximum(a_new, 0)
        s_new = np.sum(a_new)
        if s_new > 1e-15:
            a_new = a_new / s_new
        else:
            a_new = a.copy()
        
        a = a_new
        obj = compute_obj(a)
        
        if obj < best_obj - 1e-12:
            best_obj = obj
            best_a = a.copy()
            stall = 0
        else:
            stall += 1
            if stall > 100:
                step *= 0.8
                if step < 1e-10:
                    break
                rng = np.random.default_rng(it * 31 + 7)
                a = a + rng.normal(0, 0.01, n) * step
                a = np.maximum(a, 0)
                s = np.sum(a)
                if s > 1e-15:
                    a = a / s
                stall = 0
        
        if it > 100:
            step *= 0.998
    
    return best_a


def resample_sequence(a, new_n):
    """Resample sequence to new length via interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0)
    s = np.sum(a_new)
    if s > 1e-15:
        a_new = a_new / s
    return a_new


def polish_large(a, time_budget=60.0):
    """Polish a larger sequence using subgradient + LP."""
    n = len(a)
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    
    while time.monotonic() - start < time_budget - 5:
        remaining = time_budget - (time.monotonic() - start)
        if remaining < 10:
            break
        
        # Subgradient
        if remaining > 15:
            a_new = subgradient_optimize(best_a, max_iters=int(remaining * 8),
                                         time_budget=min(remaining * 0.4, 20.0),
                                         tau_start=5.0, tau_end=40.0)
            obj = compute_obj(a_new)
            if obj < best_obj:
                best_obj = obj
                best_a = a_new.copy()
                print(f"    SG: {best_obj:.6f}")
        
        # LP if n is small enough
        remaining = time_budget - (time.monotonic() - start)
        if remaining > 20 and n <= 700:
            a_new = alternating_lp_small(best_a, max_iters=20, time_budget=min(remaining * 0.4, 30.0))
            obj = compute_obj(a_new)
            if obj < best_obj:
                best_obj = obj
                best_a = a_new.copy()
                print(f"    LP: {best_obj:.6f}")
    
    return best_a, best_obj
