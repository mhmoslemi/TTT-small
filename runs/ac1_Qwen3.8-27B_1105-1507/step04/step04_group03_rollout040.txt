def fast_conv(a):
    n = len(a)
    size = 1
    while size < 2*n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2*n-1]

def compute_J(a):

    s = a.sum()
    if s < 0.01:

import numpy as np

def fast_conv(a):
    """Full self-convolution via FFT."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]


def compute_J(a):
    """Compute J(a) = 2*n*max(conv(a,a))/(sum(a))^2."""
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))

def resample_seq(src, dest_n):
    """Resample 1D sequence to new length."""
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size, endpoint=False)
    t_new = np.linspace(0, 1, dest_n, endpoint=False)
    result = np.interp(t_new, t_old, src)
    return np.maximum(result, 0.0)

"""
Coordinate descent optimizer for minimizing J(a) = 2*n*max(conv(a,a))/sum(a)^2.
Uses Gauss-Seidel coordinate descent with 1D subproblem solving via grid+golden section.
Combined with multi-length exploration, random restarts, and diverse initializations.
Key insight: each 1D subproblem is to minimize 2n*M(x)/(S+x)^2 where M(x) is max
of affine functions and one quadratic, solvable efficiently per coordinate.

import numpy as np
import time


def fast_conv(a):
    """Full self-convolution via FFT."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]


def compute_J(a):
    """Compute J(a) = 2*n*max(conv(a,a))/(sum(a))^2."""
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))


def coord_descent_sweep(a, deadline):
    """
    One full Gauss-Seidel coordinate descent sweep.
    For each coordinate m, solve the 1D subproblem exactly.
    """
    n = len(a)
    a = a.astype(np.float64).copy()
    b = fast_conv(a)
    total_sum = float(a.sum())

    for m in range(n):
        if time.monotonic() > deadline:
            break

        old_val = a[m]
        S_other = total_sum - old_val

        # Range of k affected by a[m]
        lo_k = m
        hi_k = min(2 * n - 2, m + n - 1)
        range_len = hi_k - lo_k + 1
        k_range = np.arange(lo_k, hi_k + 1)

        # For k in range: contribution is 2*old_val*a[k-m], except k=2m where it's old_val^2
        S_k_local = b[lo_k:hi_k + 1].copy()

        # Fix k = 2m
        idx_2m = 2 * m - lo_k
        if 0 <= idx_2m < range_len:

        # Max outside affected range (unchanged by a[m])
        max_out = 0.0

            max_out = max(max_out, float(b[:lo_k].max()))
        if hi_k + 1 < 2 * n - 1:

        # Coefficients for linear part of M(x)
        coeff_arr = 2.0 * a[k_range - m].copy()
        if 0 <= idx_2m < range_len:
            coeff_arr[idx_2m] = 0.0

        # M(x) = max(S_k_local + x*coeff_arr with quadratic at idx_2m, max_out)
        S_k_2m = S_k_local[idx_2m] if 0 <= idx_2m < range_len else -np.inf

        def M_of_x(x):
            vals = S_k_local + x * coeff_arr
            if 0 <= idx_2m < range_len:
                vals[idx_2m] = S_k_2m + x * x
            mval = float(vals.max())
            return mval if mval > max_out else max_out

def coord_descent_optimize(a, deadline, max_sweeps=100000, verbose=True, label=""):
    """
    Run coordinate descent for multiple sweeps until convergence or deadline.
    Returns (best_a, best_J).
    """
    a = np.maximum(a.astype(np.float64), 0.0)
    if a.sum() < 1e-10:
        a = np.ones(len(a))
    a = a / a.sum()

    best_a = a.copy()
    best_J = compute_J(a)
    prev_J = best_J
    stagnation = 0
    sweep_count = 0

    if verbose:
        print(f"  [{label}] Start: J={best_J:.10f}, n={len(a)}", flush=True)

    for sweep in range(max_sweeps):
        if time.monotonic() > deadline:
            break

        a = coord_descent_sweep(a, deadline)
        J = compute_J(a)
        sweep_count += 1

        if J < best_J - 1e-13:
            best_J = J
            best_a = a.copy()
            stagnation = 0
            if verbose and (sweep < 10 or sweep % 100 == 0):
                print(f"    [{label}] Sweep {sweep}: J={J:.10f}", flush=True)
        else:
            J_eff = max(J, best_J)
            a = best_a  # Don't accept worsening
            stagnation += 1
            if stagnation > 50:
                if verbose:
                    print(f"    [{label}] Converged at sweep {sweep}: J={best_J:.10f}", flush=True)
                break

    if verbose:
        print(f"  [{label}] Done: J={best_J:.10f} after {sweep_count} sweeps", flush=True)

    return best_a, best_J


def resample_seq(src, dest_n):
    """Resample 1D sequence to new length using linear interpolation."""
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size, endpoint=False)
    t_new = np.linspace(0, 1, dest_n, endpoint=False)
    result = np.interp(t_new, t_old, src)
    return np.maximum(result, 0.0)


def bidirectional_resample(src, dest_n):
    """Resample with symmetric constraint."""
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    # Make symmetric
    half = (src[:len(src)//2] + src[len(src)//2:][::-1]) / 2.0
    half_n = dest_n // 2
    t_old = np.linspace(0, 1, len(half), endpoint=False)
    t_new = np.linspace(0, 1, half_n, endpoint=False)
    new_half = np.interp(t_new, t_old, half)
    if dest_n % 2 == 0:
        result = np.concatenate([new_half, new_half[::-1]])
    else:
        mid = new_half[half_n - 1] if half_n > 0 else 0.0
        result = np.concatenate([new_half, [mid], new_half[:half_n-1][::-1]]) if half_n > 0 else np.array([max(src.max(), 0.0)])
    return np.maximum(result, 0.0)


def generate_inits(n, rng, parent=None):

    inits.append(np.ones(n) / n)
    inits.append(0.5 * (1.0 + np.cos(2 * np.pi * t)))
    inits.append(np.maximum(0.5 * (1.0 - np.cos(2 * np.pi * t)) + 0.01, 0.001))
    inits.append(np.exp(-4.0 * (t - 0.5) ** 2) + 0.05)
    inits.append(np.exp(-6.0 * (t - 0.3) ** 2) + np.exp(-6.0 * (t - 0.7) ** 2) + 0.03)
    inits.append(np.maximum(1.0 + 0.25 * np.cos(2 * np.pi * t + 0.3), 0.05))
    inits.append(np.maximum(1.0 + 0.15 * np.cos(4 * np.pi * t) + 0.1 * np.cos(6 * np.pi * t), 0.05))
    inits.append(1.0 - 0.5 * np.abs(2 * t - 1.0))
    # Power law taper
    inits.append((0.5 + 0.5 * np.sin(np.pi * t)) ** 1.5)
    # Sinc-like
    x = 2 * t - 1
    inits.append(np.maximum(np.sinc(3 * x) + 0.3, 0.05))

    # Random smooth
    k = max(3, n // 8)
    kernel = np.ones(k) / k
    sm = np.convolve(rng.uniform(0.2, 1.0, n), kernel, mode='same')
    inits.append(np.maximum(sm, 0.01))

    # Normalize
    result = []
    for a in inits:
        a = np.maximum(a, 1e-8)
        s = a.sum()
        if s > 1e-6:
            result.append(a / s)
    return result


class ScoreTracker:
    def __init__(self):
        self.best = None
        self.best_score = float('inf')
        self.t_start = time.monotonic()

    def update(self, a):
        a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
        if a.size == 0 or a.sum() < 0.01:
            return
        score = float(evaluate_sequence(a.tolist()))
        if np.isfinite(score) and score < self.best_score:
            self.best = a.copy()
            self.best_score = score
            elapsed = time.monotonic() - self.t_start
            print(f"  [NEW BEST] J={score:.8f} n={a.size} t={elapsed:.0f}s", flush=True)

    def report(self):
        if self.best is not None:
            print(f"  Best: J={self.best_score:.8f} n={self.best.size}", flush=True)


def propose_candidate(seed=42, budget_s=1500, **kwargs):
    """
    Coordinate descent with multi-length search, diverse initializations,
    and random restarts. Exploits the 1D subproblem structure for exact
    per-coordinate optimization.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + float(budget_s) - 10.0
    tracker = ScoreTracker()

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        if p.ndim == 1 and p.size > 0 and np.all(np.isfinite(p)):
            parent = np.maximum(p, 0.0)
            tracker.update(parent)
            print(f"Parent: n={parent.size}, J={tracker.best_score:.6f}", flush=True)
    except (NameError, AttributeError):
        pass

    if parent is None:
        parent = np.ones(512)
        tracker.update(parent)
        print(f"No parent, using uniform n=512", flush=True)

    best_a = tracker.best.copy() if tracker.best is not None else parent.copy()
    best_n = best_a.size
    best_J = compute_J(best_a)

    # === Phase 1: Intensive coordinate descent on parent ===
    t0 = time.monotonic()
    print(f"\n{'='*55}", flush=True)
    print(f"Phase 1: Coordinate descent n={best_n}", flush=True)
    print(f"{'='*55}", flush=True)

    phase1_deadline = min(deadline - 500, time.monotonic() + 250)
    a1 = best_a.copy()
    a1, J1 = coord_descent_optimize(a1, phase1_deadline, max_sweeps=20000, label="P1")
    tracker.update(a1)

    if J1 < best_J:
        best_a = a1.copy()
        best_n = best_a.size
        best_J = J1

    print(f"  Phase 1 complete ({time.monotonic()-t0:.1f}s)", flush=True)

    # === Phase 2: Diverse initializations at best length ===
    t0 = time.monotonic()
    print(f"\n{'='*55}", flush=True)
    print(f"Phase 2: Diverse inits n={best_n}", flush=True)
    print(f"{'='*55}", flush=True)

    inits = generate_inits(best_n, rng, best_a)
    # Deduplicate (skip exact parent which is already optimized)
    inits = inits[1:]  # Skip first one (parent itself)

    remaining_p2 = max(60, deadline - 400 - time.monotonic())
    per_init_time = remaining_p2 / max(1, len(inits))

    for i, init in enumerate(inits):
        if time.monotonic() > deadline - 380:
            break
        init_dl = time.monotonic() + max(15, min(60, per_init_time))
        init = np.maximum(init, 1e-8)
        init = init / init.sum()
        a_try, J_try = coord_descent_optimize(init, init_dl, max_sweeps=3000, verbose=False)
        tracker.update(a_try)
        if J_try < best_J - 1e-10:
            best_a = a_try.copy()
            best_n = best_a.size
            best_J = J_try
            print(f"  Init {i}: J={J_try:.10f} (new best!)", flush=True)
        elif i < 5:
            print(f"  Init {i}: J={J_try:.10f}", flush=True)

    print(f"  Phase 2 complete ({time.monotonic()-t0:.1f}s)", flush=True)

    # === Phase 3: Multi-length exploration ===
    t0 = time.monotonic()
    print(f"\n{'='*55}", flush=True)
    print(f"Phase 3: Multi-length exploration (base n={best_n})", flush=True)
    print(f"{'='*55}", flush=True)

    # Generate candidate lengths
    length_candidates = set()
    for mult in [0.5, 0.75, 1.0, 1.25, 1.5, 2.0, 3.0]:
        lt = int(best_n * mult)
        if 64 <= lt <= 4096:
            length_candidates.add(lt)
    # Add some standard lengths
    for std_n in [128, 192, 256, 320, 384, 512, 640, 768, 1024, 1280, 1536, 2048, 3072]:
        if 64 <= std_n <= 4096:
            length_candidates.add(std_n)
    length_candidates = sorted(l for l in length_candidates if l != best_n)

    # Prioritize lengths near best_n
    length_candidates.sort(key=lambda x: abs(x - best_n))
    length_candidates = length_candidates[:10]

    remaining_p3 = max(100, deadline - 200 - time.monotonic())
    per_length_time = remaining_p3 / max(1, len(length_candidates))

    for n_try in length_candidates:
        if time.monotonic() > deadline - 180:
            break
        a_try = resample_seq(best_a, n_try)
        a_try = np.maximum(a_try, 1e-8)
        a_try = a_try / a_try.sum()

        length_dl = time.monotonic() + max(20, min(80, per_length_time))
        a_try, J_try = coord_descent_optimize(a_try, length_dl, max_sweeps=5000, verbose=False)
        tracker.update(a_try)
        marker = " *" if J_try < best_J else ""
        print(f"  n={n_try:5d}: J={J_try:.10f}{marker}", flush=True)

        if J_try < best_J - 1e-10:
            best_a = a_try.copy()
            best_n = n_try
            best_J = J_try

    print(f"  Phase 3 complete ({time.monotonic()-t0:.1f}s)", flush=True)

    # === Phase 4: Long coordinate descent on best ===
    t0 = time.monotonic()
    print(f"\n{'='*55}", flush=True)
    print(f"Phase 4: Long coordinate descent n={best_n}", flush=True)
    print(f"{'='*55}", flush=True)

    a_long = best_a.copy()
    a_long, J_long = coord_descent_optimize(a_long, deadline - 40, max_sweeps=200000, label="P4")
    tracker.update(a_long)

    if J_long < best_J:
        best_a = a_long.copy()
        best_n = best_a.size
        best_J = J_long

    print(f"  Phase 4 complete ({time.monotonic()-t0:.1f}s)", flush=True)

    # === Phase 5: Random restarts ===
    if time.monotonic() < deadline - 25:
        t0 = time.monotonic()
        print(f"\n{'='*55}", flush=True)
        print(f"Phase 5: Random restarts", flush=True)
        print(f"{'='*55}", flush=True)

        remaining = deadline - 30 - time.monotonic()
        n_restarts = max(1, int(remaining / 25))
        n_restarts = min(n_restarts, 8)
        per_restart = remaining / n_restarts

        for r in range(n_restarts):
            if time.monotonic() > deadline - 20:
                break
            # Create perturbed version
            noise_scale = 0.01 * (1.0 + 0.3 * r)
            a_pert = best_a + noise_scale * rng.standard_normal(best_n)
            a_pert = np.maximum(a_pert, 1e-8)
            a_pert = a_pert / a_pert.sum()

            # Also try a structural perturbation
            if r % 2 == 1:
                # Taper the edges
                edge = max(1, int(best_n * 0.1))
                for i in range(edge):
                    frac = (i + 1) / (edge + 1)
                    a_pert[i] *= frac
                    a_pert[best_n - 1 - i] *= frac
                a_pert = np.maximum(a_pert, 1e-8)
                a_pert = a_pert / a_pert.sum()

            pert_dl = time.monotonic() + max(10, per_restart)
            a_pert, J_pert = coord_descent_optimize(a_pert, pert_dl, max_sweeps=5000, verbose=False)
            tracker.update(a_pert)
            marker = " *" if J_pert < best_J else ""
            print(f"  Restart {r} (scale={noise_scale:.3f}): J={J_pert:.10f}{marker}", flush=True)

            if J_pert < best_J - 1e-10:
                best_a = a_pert.copy()
                best_n = best_a.size
                best_J = J_pert

    # === Final: One more long polish if time ===
    if time.monotonic() < deadline - 5 and tracker.best is not None:
        final_dl = deadline - 2
        a_final = tracker.best.copy()
        a_final, J_final = coord_descent_optimize(a_final, final_dl, max_sweeps=50000, verbose=False, label="Final")
        tracker.update(a_final)

    # Return best
    if tracker.best is not None:
        final_a = tracker.best
    else:
        final_a = best_a

    print(f"\n{'='*55}", flush=True)
    print(f"FINAL: J={tracker.best_score:.8f} n={final_a.size} time={time.monotonic()-tracker.t_start:.0f}s", flush=True)
    print(f"{'='*55}", flush=True)

    return [float(x) for x in final_a]
