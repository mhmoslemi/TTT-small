                a_trial = normalize((1.0 - t) * a + t * g)
                J_trial = compute_J(a_trial)
                if J_trial < best_J - 1e-15:
                    best_J = J_trial
                    best_a = a_trial

            if best_a is None:
                break
            a = best_a
            current_J = best_J
        tracker.update(a)

    tracker.update(a)
    return a

def lse_gradient_descent(a, max_iters, deadline, tracker, tau_init=20.0):
    a = normalize(a.astype(np.float64))
    n = len(a)
    current_J = compute_J(a)
    tracker.update(a)

    tau = tau_init
    lr = 0.5 / (n * tau)
    momentum = 0.8
    v = np.zeros(n)

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        b = fast_conv(a)
        b_max = b.max()
        exp_b = np.exp(tau * (b - b_max))
        p = exp_b / exp_b.sum()

        s_fft = 1
        while s_fft < 3 * n:
            s_fft <<= 1
        fp = np.fft.rfft(p, s_fft)
        fa = np.fft.rfft(a, s_fft)
        corr = np.fft.irfft(fp * np.conj(fa), s_fft)
        grad = 2.0 * corr[:n]

        v = momentum * v + (1.0 - momentum) * grad
        a_new = a - lr * v
        a_new = np.maximum(a_new, 1e-10)
        a_new = normalize(a_new)
        J_new = compute_J(a_new)
        if J_new < current_J - 1e-16:
            a = a_new
            current_J = J_new
            tau = min(500.0, tau * 1.02)
            lr = 0.5 / (n * tau)
        else:
            lr *= 0.6
            v *= 0.3
            if lr < 1e-18:
                break
        if it % 200 == 0:
            tracker.update(a)
    tracker.update(a)
    return a

def multiplicative_descent(a, max_iters, deadline, tracker):
    a = normalize(a.astype(np.float64))
    n = len(a)
    current_J = compute_J(a)
    tracker.update(a)
    lr = 2.0 / n

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        b = fast_conv(a)
        b_max = b.max()
        active = b >= b_max * 0.999
        n_active = active.sum()

        grad = np.zeros(n)
        np.add.at(grad, np.arange(2*n-1)[active], 0)
        for k in np.where(active)[0]:
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            for j in range(js, je + 1):
                grad[j] += a[k - j]
        grad /= n_active

        a_new = a * np.exp(-lr * grad)
        a_new = np.maximum(a_new, 1e-10)
        a_new = normalize(a_new)
        J_new = compute_J(a_new)
        if J_new < current_J - 1e-15:
            a = a_new
            current_J = J_new
            lr = min(lr * 1.01, 10.0 / n)
        else:
            lr *= 0.7
        if it % 100 == 0:
            tracker.update(a)
    tracker.update(a)
    return a

def subgradient_descent(a, max_iters, deadline, tracker):
    a = normalize(a.astype(np.float64))
    n = len(a)
    current_J = compute_J(a)
    tracker.update(a)
    lr = 0.3 / n
    rng = np.random.default_rng(42)
    momentum = 0.75
    v = np.zeros(n)

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        b = fast_conv(a)
        M = float(b.max())
        active = np.where(b >= M - 1e-8 * M)[0]

        grad = np.zeros(n)
        for k in active:
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            for j in range(js, je + 1):
                grad[j] += a[k - j]
        grad /= len(active)

        direction = grad - 2.0 * M * a
        v = momentum * v + (1.0 - momentum) * direction
        a_new = a - lr * v
        a_new = np.maximum(a_new, 1e-10)
        a_new = normalize(a_new)
        J_new = compute_J(a_new)
        if J_new < current_J - 1e-16:
            a = a_new
            current_J = J_new
            lr = min(lr * 1.02, 5.0 / n)
        else:
            lr *= 0.7
            v *= 0.5
        if it % 50 == 0:
            tracker.update(a)
    tracker.update(a)
    return a

def flat_spectrum_init(n, rng):
    half = n // 2 + 1
    angles = rng.uniform(0, 2 * np.pi, half)
    A = np.zeros(n, dtype=complex)
    A[:half] = np.exp(1j * angles[:half])
    for k in range(1, n - n // 2):
        A[k] = np.conj(A[n - k])
    a = np.fft.ifft(A).real
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-10:
        return np.ones(n) / n
    return a / s

def cosine_mod_init(n, rng):
    t = np.linspace(0, 1, n, endpoint=False)
    freqs = rng.uniform(0.5, 10.0, size=5)
    amps = rng.uniform(0.05, 0.4, size=5)
    phases = rng.uniform(0, 2*np.pi, size=5)
    a = np.ones(n)
    for f, amp, ph in zip(freqs, amps, phases):
        a += amp * np.cos(2 * np.pi * f * t + ph)
    a = np.maximum(a, 0.01)
    return a / a.sum()

def random_fourier_init(n, rng):
    n_coeff = min(n // 2 + 1, 20)
    t = np.arange(n) / n
    sig = np.zeros(n)
    for i in range(1, n_coeff):
        c = rng.normal(0, 1.0 / np.sqrt(i))
        sig += c * np.cos(2 * np.pi * i * t + rng.uniform(0, 2*np.pi))
    a = np.abs(sig) + 0.5
    return a / a.sum()

def resample_seq(src, dest_n):
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)

def generate_inits(n, rng, parent):
    inits = []
    if parent is not None and parent.size > 0:
        inits.append(normalize(resample_seq(parent, n)))
        inits.append(normalize(resample_seq(parent[::-1], n)))

    inits.append(np.ones(n) / n)
    inits.append(normalize(rng.uniform(0.2, 1.0, n)))
    inits.append(flat_spectrum_init(n, rng))
    inits.append(flat_spectrum_init(n, rng))
    inits.append(flat_spectrum_init(n, rng))
    inits.append(cosine_mod_init(n, rng))
    inits.append(cosine_mod_init(n, rng))
    inits.append(random_fourier_init(n, rng))
    inits.append(random_fourier_init(n, rng))

    t = np.linspace(0, 1, n, endpoint=False)
    inits.append(normalize(1.0 + 0.3 * np.cos(2 * np.pi * 3 * t)))
    inits.append(normalize(1.0 + 0.3 * np.cos(2 * np.pi * 5 * t)))
    inits.append(normalize(1.0 + 0.2 * np.cos(2 * np.pi * 7 * t) + 0.15 * np.sin(2 * np.pi * 11 * t)))
    inits.append(normalize(0.5 + 0.5 * np.cos(np.pi * t) ** 2))
    inits.append(normalize(0.3 + 0.7 * np.abs(np.sin(np.pi * t)) ** 1.5))

    # Dual-bump
    inits.append(normalize(0.3 + 0.4 * np.exp(-8 * (t - 0.25) ** 2) + 0.4 * np.exp(-8 * (t - 0.75) ** 2)))

    # Sawtooth (clipped)
    inits.append(normalize(np.abs(1.0 - 2.0 * t) + 0.3))

    return inits

class ScoreTracker:
    def __init__(self):
        self.best = None
        self.best_score = float('inf')
        self.t_start = time.monotonic()

    def update(self, a):
        a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
        if a.size == 0 or a.sum() < 0.01:
            return
        score = float(evaluate_sequence(a.tolist()))
        if np.isfinite(score) and score < self.best_score:
            self.best = a.copy()
            self.best_score = score
            elapsed = time.monotonic() - self.t_start
            print(f"  * J={score:.8f} n={a.size} t={elapsed:.1f}s", flush=True)

    def report(self, label=""):
        if self.best is not None:
            print(f"  {label} best: J={self.best_score:.8f} n={self.best.size}", flush=True)

def intensive_polish(a, deadline, tracker):
    """Multiple rounds of LP + gradient + multiplicative updates."""
    a = normalize(a.astype(np.float64))
    n = a.size

    # Round 1: LP with tight constraints
    tighten_scheds = [
        [0.005, 0.003, 0.002, 0.001],
        [0.003, 0.002, 0.001, 0.0005],
        [0.002, 0.001, 0.0005, 0.0003],
    ]
    for sched in tighten_scheds:
        if time.monotonic() > deadline - 60:
            break
        a = run_lp_opt(a, 20, deadline - 60, tracker, tighten_schedule=sched)

    # LSE polish with multiple tau values
    for tau in [10.0, 30.0, 100.0, 300.0]:
        if time.monotonic() > deadline - 40:
            break
        a = lse_gradient_descent(a, 500, deadline - 40, tracker, tau_init=tau)

    # Multiplicative descent
    if time.monotonic() < deadline - 30:
        a = multiplicative_descent(a, 500, deadline - 30, tracker)

    # Subgradient polish
    if time.monotonic() < deadline - 20:
        a = subgradient_descent(a, 300, deadline - 20, tracker)

    # Final LP
    if time.monotonic() < deadline - 5:
        a = run_lp_opt(a, 10, deadline - 3, tracker, tighten_schedule=[0.001, 0.0005])

    tracker.update(a)
    return a

def simulated_annealing(a, deadline, tracker, n_restarts=30, t0=0.08):
    rng = np.random.default_rng(12345)
    n = a.size
    a = normalize(a.copy().astype(np.float64))
    current_J = compute_J(a)
    tracker.update(a)
    temp = t0

    for r in range(n_restarts):
        if time.monotonic() > deadline:
            break
        a_pert = a + rng.normal(0, temp, n)
        a_pert = normalize(a_pert)

        a_refined = run_lp_opt(a_pert, max(3, 10 - r // 5), deadline, tracker,
                               tighten_schedule=[0.005, 0.003])

        J = compute_J(a_refined)
        delta = J - current_J
        if delta < 0 or rng.random() < np.exp(-delta / max(temp, 1e-12)):
            a = a_refined
            current_J = J
        temp *= 0.92
    tracker.update(a)
    return a

def coordinate_descent(a, max_steps, deadline, tracker):
    a = normalize(a.astype(np.float64))
    n = len(a)
    current_J = compute_J(a)
    tracker.update(a)
    rng = np.random.default_rng(77)

    for step in range(max_steps):
        if time.monotonic() > deadline:
            break
        j = rng.integers(0, n)
        best_J = current_J
        best_val = a[j]
        for scale in [0.3, 0.5, 0.7, 1.3, 1.5, 2.0, 0.1]:
            a_trial = a.copy()
            a_trial[j] *= scale
            a_trial = normalize(a_trial)
            J_trial = compute_J(a_trial)
            if J_trial < best_J - 1e-15:
                best_J = J_trial
                best_val = a_trial[j]
        if best_val != a[j]:
            a = a.copy()
            a[j] = best_val
            a = normalize(a)
            current_J = best_J
        if step % 200 == 0:
            tracker.update(a)
    tracker.update(a)
    return a

def propose_candidate(seed=42, budget_s=1500, **kwargs):
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + float(budget_s) - 30.0
    tracker = ScoreTracker()

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        if p.ndim == 1 and p.size > 0 and np.all(np.isfinite(p)):
            parent = np.maximum(p, 0.0)
            tracker.update(parent)
            print(f"Parent: n={parent.size}, J={tracker.best_score:.8f}", flush=True)
    except (NameError, AttributeError):
        pass

    # === Phase 1: Multi-length exploration ===
    # Try a range of lengths, spending limited time on each
    if parent is not None:
        lengths = [parent.size, parent.size + 64, parent.size + 128, 384, 512, 640, 768, 1024]
    else:
        lengths = [256, 320, 384, 448, 512, 640, 768, 1024]
    # Deduplicate and sort
    lengths = sorted(set(l for l in lengths if l >= 128 and l <= 1536))

    # Time allocation: 40% for phase 1, 40% for phase 2, 20% for phase 3
    phase1_end = time.monotonic() + 0.4 * (deadline - time.monotonic())
    phase2_end = time.monotonic() + 0.8 * (deadline - time.monotonic())

    print(f"\n{'='*60}", flush=True)
    print(f"Phase 1: Multi-length exploration (n in {lengths})", flush=True)
    print(f"{'='*60}", flush=True)

    best_per_length = {}
    n_lengths = len(lengths)
    time_per_length = (phase1_end - time.monotonic()) / n_lengths

    for n in lengths:
        if time.monotonic() > phase1_end - 30:
            break
        t_start_len = time.monotonic()
        t_end_len = min(phase1_end - 20, t_start_len + time_per_length)
        if time.monotonic() >= t_end_len - 20:
            continue

        print(f"\n  n={n}, budget={time_per_length:.1f}s", flush=True)
        inits = generate_inits(n, rng, parent)

        best_a = None
        best_J = float('inf')
        n_inits = len(inits)
        time_per_init = max(5.0, (t_end_len - time.monotonic()) / n_inits)

        for idx, a0 in enumerate(inits):
            if time.monotonic() > t_end_len - 15:
                break
            t_alloc = time_per_init

            # LP phase
            n_lp = max(5, int(t_alloc * 0.5 / max(0.1, n / 400)))
            a = run_lp_opt(a0, n_lp, time.monotonic() + t_alloc * 0.5, tracker,
                           tighten_schedule=[0.01, 0.008, 0.005, 0.003])

            # LSE polish
            if time.monotonic() < t_end_len - 10:
                a = lse_gradient_descent(a, max(100, int(t_alloc * 10 / max(1, n/200))),
                                         t_end_len - 10, tracker, tau_init=30.0)

            J = compute_J(a)
            if J < best_J:
                best_J = J
                best_a = a.copy()

        if best_a is not None:
            best_per_length[n] = best_J
            tracker.update(best_a)
            print(f"  n={n}: best J={best_J:.8f} ({time.monotonic()-t_start_len:.1f}s)", flush=True)

    # === Phase 2: Focused optimization on top candidates ===
    print(f"\n{'='*60}", flush=True)
    print(f"Phase 2: Focused optimization", flush=True)
    print(f"{'='*60}", flush=True)

    # Sort lengths by best J found
    if best_per_length:
        sorted_lengths = sorted(best_per_length.keys(), key=lambda x: best_per_length[x])
        top_lengths = sorted_lengths[:4]  # Focus on top 4
    else:
        top_lengths = [512]

    focus_time = (phase2_end - time.monotonic()) / max(1, len(top_lengths))

    for n in top_lengths:
        if time.monotonic() > phase2_end - 60:
            break
        print(f"\n  Focusing on n={n} (budget={focus_time:.1f}s)", flush=True)

        # Use best found for this length as starting point
        if parent is not None:
            a_start = normalize(resample_seq(parent, n))
        else:
            a_start = np.ones(n) / n

        # Try multiple starting points
        for trial in range(3):
            if time.monotonic() > phase2_end - 60:
                break
            if trial == 0:
                a0 = a_start
            elif trial == 1:
                a0 = normalize(resample_seq(parent[::-1], n)) if parent is not None else flat_spectrum_init(n, rng)
            else:
                a0 = flat_spectrum_init(n, rng)

            t_sub = min(time.monotonic() + focus_time / 3, phase2_end - 60)

            # Long LP phase
            a = run_lp_opt(a0, 40, t_sub, tracker,
                           tighten_schedule=[0.008, 0.006, 0.004, 0.003, 0.002, 0.001])

            # LSE with multiple temperatures
            if time.monotonic() < phase2_end - 50:
                a = lse_gradient_descent(a, 300, phase2_end - 50, tracker, tau_init=15.0)
            if time.monotonic() < phase2_end - 40:
                a = lse_gradient_descent(a, 200, phase2_end - 40, tracker, tau_init=200.0)

            tracker.update(a)

    # === Phase 3: Intensive polishing on global best ===
    print(f"\n{'='*60}", flush=True)
    print(f"Phase 3: Intensive polishing", flush=True)
    print(f"{'='*60}", flush=True)

    if tracker.best is not None:
        a = tracker.best.copy()
        a = intensive_polish(a, deadline - 120, tracker)

    # === Phase 4: Simulated annealing escape ===
    if tracker.best is not None and time.monotonic() < deadline - 80:
        print(f"\n  Simulated annealing...", flush=True)
        a = simulated_annealing(tracker.best, deadline - 80, tracker, n_restarts=25, t0=0.06)

        # Polish after SA
        if time.monotonic() < deadline - 30:
            a = run_lp_opt(a, 15, deadline - 30, tracker, tighten_schedule=[0.003, 0.001, 0.0005])
            if time.monotonic() < deadline - 15:
                a = lse_gradient_descent(a, 300, deadline - 15, tracker, tau_init=50.0)

    # === Phase 5: Final coordinate descent + last LP ===
    if tracker.best is not None and time.monotonic() < deadline - 15:
        print(f"\n  Final coordinate descent...", flush=True)
        a = coordinate_descent(tracker.best, 2000, deadline - 10, tracker)
        if time.monotonic() < deadline - 3:
            a = run_lp_opt(a, 5, deadline - 2, tracker, tighten_schedule=[0.001])

    # Fallback
    if tracker.best is None:
        tracker.best = np.ones(256)
        tracker.best_score = compute_J(tracker.best)
        tracker.update(tracker.best)

    print(f"\n{'='*60}", flush=True)
    print(f"FINAL: J={tracker.best_score:.8f} n={tracker.best.size} time={time.monotonic()-tracker.t_start:.1f}s", flush=True)
    print(f"{'='*60}", flush=True)

    return [float(x) for x in tracker.best]
