"""
Minimizes J(a) = 2*n*max(convolve(a,a))/(sum(a))² using:
1. Fast FFT-based Adam gradient descent with log-sum-exp β-annealing
2. LP mirror descent with golden-section line search for polish
3. Spectral perturbation multi-restart for escaping local minima
4. Multi-length exploration (512, 1024, 2048)
Key improvement over previous: Adam optimizer is O(n log n) per step vs O(n²) LP solve,

import numpy as np
from scipy.optimize import linprog
from scipy.sparse import lut_matrix, csr_matrix
import time

# Cache for constraint matrix structures
_A_structure_cache = {}


def fast_conv(a):
    """Full self-convolution via FFT."""
    n = len(a)
    s = 1
    while s < 2 * n - 1:
        s <<= 1
    fa = np.fft.rfft(a, s)
    return np.fft.irfft(fa * fa, s)[:2 * n - 1]


def compute_J(a):
    """Compute J(a) = 2*n*max(conv(a,a))/(sum(a))^2."""
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))


def smooth_max_and_weights(b, beta):
    """Compute log-sum-exp approximation of max(b) and softmax weights."""
    b_max = float(b.max())
    shifted = beta * (b - b_max)
    shifted = np.clip(shifted, -500, 500)
    exp_shifted = np.exp(shifted)
    Z = float(exp_shifted.sum())
    if Z < 1e-300:
        p = np.zeros_like(b)
        p[np.argmax(b)] = 1.0
        return b_max, p
    p = exp_shifted / Z
    val = b_max + np.log(Z) / beta
    return val, p


def J_smooth_and_grad(a, beta):
    """
    Compute smooth J(a) = 2n*logsumexp(beta*conv(a,a))/ (sum a)^2 and its gradient.
    Returns (J_value, gradient_array).
    """
    n = len(a)
    s = float(a.sum())
    if s < 1e-10:
        return 1e10, np.zeros(n)
    b = fast_conv(a)
    val, p = smooth_max_and_weights(b, beta)

    # grad_val[m] = 2 * sum_j a[j] * p[j+m]
    # This is 2 * np.correlate(p, a, mode='full')[n-1 : 3n-2]
    corr = np.correlate(p, a, mode='full')
    grad_val = 2.0 * corr[n - 1:3 * n - 2]

    J_val = 2.0 * n * val / (s * s)
    grad_J = 2.0 * n * grad_val / (s * s) - 4.0 * n * val / (s ** 3)
    grad_J = np.nan_to_num(grad_J, nan=0.0, posinf=0.0, neginf=0.0)
    return J_val, grad_J


def adam_optimize(a0, beta_schedule, deadline, tracker=None, lr=0.005,
                  beta1=0.9, beta2=0.999, eps=1e-8, stall_patience=500):
    """
    Adam gradient descent with β-annealing for the smooth J objective.
    Each β in the schedule gets an equal share of remaining time.
    """
    a = np.maximum(a0.astype(np.float64), 1e-8)
    n = len(a)
    a = a / a.sum()

    best_J = compute_J(a)
    best_a = a.copy()
    if tracker is not None:
        tracker.update(a)

    total_beta = len(beta_schedule)
    remaining = deadline - time.monotonic()
    if remaining <= 0:
        return best_a

    # Allocate time proportionally (later betas get more time since they're more important)
    weights = np.arange(1, total_beta + 1, dtype=np.float64)
    weights /= weights.sum()
    time_per_beta = remaining * weights

    for bi, beta in enumerate(beta_schedule):
        budget = time_per_beta[bi]
        deadline_beta = time.monotonic() + budget

        # Reset momentum for each new beta
        m = np.zeros(n)
        v = np.zeros(n)
        t = 0
        no_improve = 0

        while time.monotonic() < deadline_beta:
            t += 1
            J_val, grad = J_smooth_and_grad(a, beta)
            if not np.isfinite(J_val):
                break

            # Adam update
            m = beta1 * m + (1 - beta1) * grad
            v = beta2 * v + (1 - beta2) * grad * grad
            m_hat = m / (1 - beta1 ** t)
            v_hat = v / (1 - beta2 ** t)
            a_new = a - lr * m_hat / (np.sqrt(v_hat) + eps)
            a_new = np.maximum(a_new, 0.0)
            s_new = a_new.sum()
            if s_new < 1e-10:
                break
            a_new = a_new / s_new
            a = a_new

            # Check improvement periodically
            if t % 20 == 0:
                J_true = compute_J(a)
                if J_true < best_J - 1e-10:
                    best_J = J_true
                    best_a = a.copy()
                    no_improve = 0
                    if tracker is not None and t % 200 == 0:
                        tracker.update(a)
                else:
                    no_improve += 1

            # Stall detection
            if no_improve > stall_patience:
                break

        # Update tracker at end of each beta stage
        J_check = compute_J(a)
        if J_check < best_J:
            best_J = J_check
            best_a = a.copy()
        if tracker is not None:
            tracker.update(best_a)

    return best_a


def get_A_structure(n):
    """Get or pre-compute the (row, col) arrays for the constraint matrix."""
    if n not in _A_structure_cache:
        rows = []
        cols = []
        for k in range(2 * n - 1):
            jl = max(0, k - n + 1)
            jh = min(n - 1, k)
            for j in range(jl, jh + 1):
                rows.append(k)
                cols.append(j)
        _A_structure_cache[n] = (np.array(rows, dtype=np.int32),
                                 np.array(cols, dtype=np.int32))
    return _A_structure_cache[n]


def build_A_sparse(a):
    """Build sparse constraint matrix for (a*g)_k <= M constraints."""
    n = len(a)
    rows_arr, cols_arr = get_A_structure(n)
    vals = a[rows_arr - cols_arr]
    return csr_matrix((vals, (rows_arr, cols_arr)), shape=(2 * n - 1, n))


def lp_best_response(a, M):
    """LP: max sum(g) s.t. (a*g)_k <= M for all k, g >= 0. Returns normalized g or None."""
    n = len(a)
    c = -np.ones(n)
    A = build_A_sparse(a)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    try:
        r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
        if r.success and r.x is not None:
            g = np.maximum(r.x, 0.0)
            g_sum = g.sum()
            if g_sum > 1e-10:
                return g / g_sum
    except Exception:
        pass
    return None


def line_search_golden(a, g, n_evals=15):
    """Golden section search for best t in [0,1] minimizing J((1-t)*a + t*g)."""
    best_t = 0.0
    best_J = compute_J(a)
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    lo = 0.0
    hi = 1.0

    a_1 = g.copy()
    J_at_1 = compute_J(a_1)
    if J_at_1 < best_J:
        best_J = J_at_1
        best_t = 1.0

    for _ in range(n_evals):
        t1 = hi - gr * (hi - lo)
        t2 = lo + gr * (hi - lo)
        a1 = (1.0 - t1) * a + t1 * g
        a2 = (1.0 - t2) * a + t2 * g
        J1 = compute_J(a1)
        J2 = compute_J(a2)
        if J1 < J2:
            hi = t2
        else:
            lo = t1
        t_mid = (lo + hi) / 2.0
        a_mid = (1.0 - t_mid) * a + t_mid * g
        J_mid = compute_J(a_mid)
        if J_mid < best_J:
            best_J = J_mid
            best_t = t_mid

    return (1.0 - best_t) * a + best_t * g, best_J


def lp_mirror_descent(a, max_iters, deadline, tracker=None, n_line_evals=15,
                      stall_patience=30):
    """LP-based mirror descent with line search."""
    a = np.maximum(a.astype(np.float64), 1e-6)
    a = a / a.sum()
    current_J = compute_J(a)
    if tracker is not None:
        tracker.update(a)

    stall_count = 0
    best_J = current_J
    best_a = a.copy()

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        b = fast_conv(a)
        M = float(b.max())
        g = lp_best_response(a, M)
        if g is None:
            break

        a_new, J_new = line_search_golden(a, g, n_evals=n_line_evals)

        if J_new < current_J - 1e-13:
            a = a_new
            current_J = J_new
            stall_count = 0
            if J_new < best_J:
                best_J = J_new
                best_a = a.copy()
                if tracker is not None:
                    tracker.update(a)
        else:
            # Try smaller steps
            improved = False
            for t_try in [0.01, 0.005, 0.02, 0.05, 0.1]:
                a_try = (1.0 - t_try) * a + t_try * g
                a_try = np.maximum(a_try, 0.0)
                s_try = a_try.sum()
                if s_try < 1e-10:
                    continue
                a_try = a_try / s_try
                J_try = compute_J(a_try)
                if J_try < current_J - 1e-14:
                    a = a_try
                    current_J = J_try
                    stall_count = 0
                    if J_try < best_J:
                        best_J = J_try
                        best_a = a.copy()
                        if tracker is not None:
                            tracker.update(a)
                    improved = True
                    break
            if not improved:
                stall_count += 1
                if stall_count >= stall_patience:
                    break

    if tracker is not None:
        tracker.update(best_a)
    return best_a


def resample_seq(src, dest_n):
    """Resample 1D sequence to new length via linear interpolation."""
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    result = np.interp(t_new, t_old, src)
    return np.maximum(result, 0.0)


def generate_inits(n, rng, parent=None):
    """Generate diverse initial sequences of length n."""
    inits = []
    t = np.linspace(0, 1, n, endpoint=False)

    if parent is not None and parent.size > 0:
        inits.append(resample_seq(parent, n))
        p = resample_seq(parent, n)
        p = p + rng.normal(0, 0.05, n) * p
        inits.append(np.maximum(p, 0.01))

    inits.append(np.ones(n))
    inits.append(rng.uniform(0.3, 1.0, n))

    # Cosine families with various frequencies
    for freq in [1, 2, 3, 4, 5, 7, 8]:
        inits.append(1.0 + 0.4 * np.cos(2.0 * np.pi * freq * t))

    # Gaussian bump
    inits.append(np.exp(-4.0 * (t - 0.5) ** 2) + 0.1)

    # Sine absolute
    inits.append(np.abs(np.sin(np.pi * t)) + 0.2)

    # Two-peak
    inits.append(1.0 + 0.3 * np.cos(4.0 * np.pi * t))

    # Power law
    inits.append((t + 0.1) ** 0.5)

    # Exponential
    inits.append(np.exp(-1.5 * t) + 0.2)

    # Triangular
    inits.append(1.0 - 0.3 * np.abs(2.0 * (t % 0.5) - 0.5))

    # Dual cosine
    inits.append(1.0 + 0.25 * np.cos(2.0 * np.pi * 2 * t) - 0.15 * np.cos(2.0 * np.pi * 5 * t))

    return inits

def spectral_perturb(a, rng, strength=0.1):
    """Perturb a sequence by adding noise in the frequency domain."""
    n = len(a)
    s = 1
    while s < 2 * n:
        s <<= 1
    fa = np.fft.rfft(a, s)
    # Add random phase perturbation
    phase = rng.uniform(-strength, strength, fa.shape)
    magnitude = rng.normal(1.0, strength * 0.5, fa.shape)
    fa_perturbed = fa * magnitude * np.exp(1j * phase)
    a_perturbed = np.fft.irfft(fa_perturbed, s)[:n]
    return np.maximum(a_perturbed, 0.01)

        n_best = tracker.best.size
        remaining = deadline - time.monotonic()
        if remaining > 40:
            print(f"\n{'='*50} Phase 5: Length Exploration (budget {remaining:.0f}s) {'='*50}", flush=True)

            candidates = [n_best - 128, n_best - 64, n_best + 64, n_best + 128,
                          384, 640, 768, 896, 1280, 1536]
            # Filter valid and non-duplicate
            seen = set()
            valid_cands = []
            for nc in candidates:
                if 128 <= nc <= 2560 and nc != n_best and nc not in seen:
                    seen.add(nc)
                    valid_cands.append(nc)

            for n_try in valid_cands[:5]:
                if time.monotonic() > deadline:
                    break
                a_res = resample_seq(tracker.best, n_try)
                a_res = np.maximum(a_res, 0.01)
                s_res = a_res.sum()
                if s_res < 1e-10:
                    continue
                a_res = a_res / s_res
                budget_here = max(15.0, min(30.0, deadline - time.monotonic()))
                beta_sched = [50.0, 100.0, 200.0]
                a_res = adam_optimize(a_res, beta_sched, time.monotonic() + budget_here,
                                      tracker=tracker, lr=0.004, stall_patience=200)

            tracker.report(f"Phase 5: {time.monotonic()-t0:.1f}s")

    # Fallback
    if tracker.best is None:
        tracker.best = np.ones(512)
        tracker.best_score = compute_J(tracker.best)
        print(f"  Using fallback: J={tracker.best_score:.8f}", flush=True)

    print(f"\n{'='*50} FINAL RESULT {'='*50}", flush=True)
    print(f"J={tracker.best_score:.8f} n={tracker.best.size} "
          f"time={time.monotonic()-tracker.t_start:.1f}s", flush=True)

    # Ensure output is valid
    result = np.clip(tracker.best, 0.0, 1000.0)
    if result.sum() < 0.01:
        result = np.ones(128)

    return [float(x) for x in result]
