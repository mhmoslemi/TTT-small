where s = sum(a), max = max_k (a*a)_k

= 4n * [s * a_{k*-j} - max] / s³

This is a valid subgradient (one of possibly many, depending on which k achieves the max).

1. Start with a = ones(n)/n
2. At each step:

"""
Minimizes J(a) = 2*n*max(convolve(a,a))/(sum(a))^2 using:
1. LP best-response with normalized constraints (M-independent direction)
2. Efficient line search (6-point + t=1)

5. Fourier-informed diverse restarts
6. Adaptive time allocation based on progress

import numpy as np
from scipy.optimize import linprog
from scipy.sparse import csr_matrix
import time

_structure_cache = {}


def fast_conv(a):
    """Full self-convolution via FFT, O(n log n)."""
    n = len(a)
    s = 1
    while s < 2 * n - 1:
        s <<= 1
    fa = np.fft.rfft(a, s)
    return np.fft.irfft(fa * fa, s)[:2 * n - 1]


def compute_J(a):
    """Compute J(a) = 2*n*max(conv(a,a))/(sum(a))^2."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))


def get_structure(n):
    """Get or pre-compute (row, col) arrays for convolution matrix."""
    if n not in _structure_cache:
        rows = []
        cols = []
        for k in range(2 * n - 1):
            jl = max(0, k - n + 1)
            jh = min(n - 1, k)
            for j in range(jl, jh + 1):
                rows.append(k)
                cols.append(j)
        _structure_cache[n] = (np.array(rows, dtype=np.int32),
                                np.array(cols, dtype=np.int32))
    return _structure_cache[n]


def build_conv_matrix(a):
    """Build sparse convolution matrix (shape (2n-1, n))."""
    n = len(a)
    rows, cols = get_structure(n)
    vals = a[rows - cols]
    return csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))


def solve_lp_direction(a, active_rows=None):
    """
    Solve LP: max sum(g) s.t. (a*g)_k <= 1 for active rows, g >= 0.
    Returns normalized optimal g (direction) or None.
    The direction is independent of M (scale of original constraints).
    """
    n = len(a)
    A = build_conv_matrix(a)
    if active_rows is not None:
        idx = np.asarray(active_rows, dtype=np.intp)
        A = A[idx, :]
    m = A.shape[0]
    c = -np.ones(n)
    b_ub = np.ones(m)
    bounds = [(0.0, None)] * n
    try:
        r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                    options={'presolve': True})
        if r.success and r.x is not None:
            g = np.maximum(r.x, 0.0)
            g_sum = g.sum()
            if g_sum > 1e-10:
                return g / g_sum
    except Exception:
        pass
    return None


def line_search(a, g, n_pts=6):
    """
    Evaluate J at n_pts+1 points (t=0, 1/(n_pts+1), ..., 1) and return best.
    """
    best_val = compute_J(a)
    best_t = 0.0
    for i in range(1, n_pts + 1):
        t = i / (n_pts + 1)
        a_t = (1.0 - t) * a + t * g
        a_t = np.maximum(a_t, 0.0)
        s = a_t.sum()
        if s < 1e-10:
            continue
        a_t = a_t / s
        val = compute_J(a_t)
        if val < best_val:
            best_val = val
            best_t = t
    # Check t=1 (full step to LP solution)
    val_1 = compute_J(g)
    if val_1 < best_val:
        best_val = val_1
        best_t = 1.0
    result = (1.0 - best_t) * a + best_t * g
    s = result.sum()
    if s > 1e-10:
        result = result / s
    return result, best_val


def lp_mirror_loop(a, deadline, tracker, n_pts=6, min_improve=1e-12,
                   stall_limit=80, use_active=False):
    """
    Main LP mirror descent loop.
    """
    a = np.maximum(np.asarray(a, dtype=np.float64), 1e-10)
    a = a / a.sum()
    current_J = compute_J(a)
    tracker.update(a)

    stall = 0
    best_a = a.copy()
    best_J = current_J
    active_rows = None

    while time.monotonic() < deadline:
        # Optionally track active constraints
        if use_active:
            b = fast_conv(a)
            M = b.max()
            if M > 1e-15:
                # Active: those with (a*g)_k close to 1 in normalized LP
                # Use those with b_k close to max as proxy
                threshold = M * 0.95
                active_rows = np.where(b >= threshold)[0].tolist()
                if len(active_rows) < 10:
                    active_rows = None

        g = solve_lp_direction(a, active_rows=active_rows)
        if g is None:
            break

        # Check proximity to fixed point
        dist = np.linalg.norm(g - a) / max(np.linalg.norm(a), 1e-10)
        if dist < 1e-9:
            break

        a_new, J_new = line_search(a, g, n_pts=n_pts)

        if J_new < current_J - min_improve:
            a = a_new
            current_J = J_new
            stall = 0
            if J_new < best_J:
                best_J = J_new
                best_a = a.copy()
                tracker.update(a)
        else:
            stall += 1
            if stall >= stall_limit:
                break

    tracker.update(best_a)
    return best_a


def resample_seq(src, dest_n):
    """Resample 1D sequence to new length via linear interpolation."""
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    result = np.interp(t_new, t_old, src)
    return np.maximum(result, 0.0)


def generate_initializations(n, rng, parent=None):
    """Generate diverse initial sequences of length n."""
    inits = []
    t = np.linspace(0, 1, n, endpoint=False)

    if parent is not None:
        p = resample_seq(parent, n)
        p = np.maximum(p, 1e-6)
        inits.append(p / p.sum())
        # Perturbed parent
        p2 = p + rng.normal(0, 0.05, n) * p
        p2 = np.maximum(p2, 1e-6)
        inits.append(p2 / p2.sum())

    # Uniform
    inits.append(np.ones(n) / n)

    # Cosine families
    for freq in [2, 3, 4, 5, 6, 8, 10, 12, 16]:
        c = 1.0 + 0.4 * np.cos(2.0 * np.pi * freq * t)
        c = np.maximum(c, 0.01)
        inits.append(c / c.sum())

    # Two-frequency
    for f1, f2 in [(3, 7), (5, 11), (4, 9), (2, 5)]:
        c = 1.0 + 0.25 * np.cos(2 * np.pi * f1 * t) + 0.15 * np.cos(2 * np.pi * f2 * t)
        c = np.maximum(c, 0.01)
        inits.append(c / c.sum())

    # Gaussian bump
    c = np.exp(-3.0 * (t - 0.5) ** 2) + 0.1
    inits.append(c / c.sum())

    # Random smooth
    noise = rng.normal(0, 1, n)
    kernel = np.ones(max(3, n // 32)) / (n // 32)
    smooth = np.convolve(noise, kernel, mode='same')
    c = 1.0 + 0.3 * smooth
    c = np.maximum(c, 0.01)
    inits.append(c / c.sum())

    return inits

def smooth_J_and_grad(a, beta=20.0):
    """Smooth approximation of J and its gradient for L-BFGS-B."""
    n = len(a)
    s = float(a.sum())
    if s < 1e-10:
        return 1e10, np.zeros(n)
    b = fast_conv(a)
    b_max = float(b.max())
    shifted = beta * (b - b_max)
    shifted = np.clip(shifted, -400, 400)
    exp_sh = np.exp(shifted)
    Z = float(exp_sh.sum())
    if Z < 1e-300:
        p = np.zeros(2 * n - 1)
        p[int(np.argmax(b))] = 1.0
    else:
        p = exp_sh / Z
    g_val = b_max + np.log(max(Z, 1e-300)) / beta
    F_val = 2.0 * n * g_val / (s * s)
    # Gradient of max_k (a*a)_k approximation
    S = np.convolve(a, p[::-1], mode='full')
    S = S[n - 1:2 * n - 1][::-1]
    grad_g = 2.0 * S
    grad_F = 2.0 * n * (s * grad_g - 2.0 * g_val) / (s ** 3)
    return F_val, grad_F


def lbfgs_smooth(a0, deadline, max_total_iters=100):
    """L-BFGS-B with smooth objective."""
    from scipy.optimize import minimize
    a = np.maximum(np.asarray(a0, dtype=np.float64), 1e-10)
    a = a / a.sum()
    n = len(a)
    betas = [5.0, 20.0, 50.0, 100.0]
    iters_per_beta = max(10, max_total_iters // len(betas))

    for beta in betas:
        if time.monotonic() > deadline:
            break

        def func(x):
            x2 = np.maximum(x, 0.0)
            f, _ = smooth_J_and_grad(x2, beta)
            return f if np.isfinite(f) else 1e10

        def jac(x):
            x2 = np.maximum(x, 0.0)
            _, g = smooth_J_and_grad(x2, beta)
            g = np.nan_to_num(g, nan=0.0, posinf=0.0, neginf=0.0)
            g[x2 < 1e-14] = 0.0
            return g

        try:
            res = minimize(func, a.copy(), jac=jac, method='L-BFGS-B',
                           bounds=[(0.0, None)] * n,
                           options={'maxiter': iters_per_beta, 'gtol': 1e-7, 'ftol': 1e-12})
            if res.fun < 1e10:
                new_a = np.maximum(res.x, 1e-10)
                if new_a.sum() > 1e-6:
                    a = new_a
        except Exception:
            pass

    return a

                phase = rng.uniform(0, 2 * np.pi)
                a = a + amp * np.cos(2 * np.pi * k * t_arr + phase)
        else:
            # Antisymmetric perturbation (breaks symmetry)
            half = n // 2
            noise = rng.normal(0, 0.05, half)
            a[half:] += noise
            a[:half] -= noise

        a = np.maximum(a, 1e-6)
        s = a.sum()
        if s < 1e-6:
            continue
        a = a / s

        # Quick check before expensive LP
        J_pert = compute_J(a)
        if J_pert < tracker.best_score:
            tracker.update(a)

        # LP refinement (limited time per restart)
        budget_here = min(30.0, dl3 - time.monotonic())
        if budget_here > 5:
            a = lp_mirror_loop(a, time.monotonic() + budget_here, tracker,
                               n_pts=5, min_improve=1e-13, stall_limit=40)

    tracker.report(f"Phase 3 ({attempt} attempts)")

    # === Phase 4: Final polish ===
    remaining = deadline - time.monotonic()
    if remaining > 30:
        p4_budget = min(200.0, remaining)
        dl4 = time.monotonic() + p4_budget
        print(f"\n{'='*50} Phase 4: Polish (budget {p4_budget:.0f}s) {'='*50}", flush=True)

        a_polish = tracker.best.copy().astype(np.float64)
        a_polish = np.maximum(a_polish, 1e-10)
        a_polish = a_polish / a_polish.sum()

        # Long LP run with tight tolerance
        a_polish = lp_mirror_loop(a_polish, dl4, tracker, n_pts=7,
                                  min_improve=1e-14, stall_limit=150, use_active=False)

        # Verify improvement
        J_verify = compute_J(tracker.best)
        print(f"  Verify: compute_J={J_verify:.8f} vs tracker={tracker.best_score:.8f}", flush=True)

        tracker.report("Phase 4")

    # === Phase 5: Length exploration ===
    remaining = deadline - time.monotonic()
    if remaining > 20:
        print(f"\n{'='*50} Phase 5: Lengths (budget {remaining:.0f}s) {'='*50}", flush=True)
        n_best = tracker.best.size

        length_tries = []
        for delta in [-128, -64, -32, 32, 64, 128, 256]:
            lt = n_best + delta
            if 128 <= lt <= 4096 and lt != n_best:
                length_tries.append(lt)
        # Also try some common lengths
        for lt in [256, 384, 512, 768, 1024, 1536, 2048]:
            if lt not in length_tries and abs(lt - n_best) > 16:
                length_tries.append(lt)

        for lt in length_tries:
            if time.monotonic() > deadline:
                break
            a_res = resample_seq(tracker.best, lt)
            a_res = np.maximum(a_res, 1e-6)
            a_res = a_res / a_res.sum()
            J_res = compute_J(a_res)
            if J_res < tracker.best_score:
                tracker.update(a_res)

            budget_here = min(20.0, deadline - time.monotonic())
            if budget_here > 5:
                a_res = lp_mirror_loop(a_res, time.monotonic() + budget_here, tracker,
                                       n_pts=5, min_improve=1e-13, stall_limit=30)

        tracker.report("Phase 5")

    # === Final validation ===
    if tracker.best is None:
        tracker.best = np.ones(512)
        tracker.best_score = compute_J(tracker.best)

    final = np.clip(tracker.best, 0.0, 1000.0)
    if final.sum() < 0.01:
        final = np.ones(128)

    final_score = float(evaluate_sequence(final.tolist()))
    print(f"\n{'='*50} FINAL: J={final_score:.8f} n={final.size} "
          f"time={time.monotonic()-t_start:.1f}s {'='*50}", flush=True)

    return [float(x) for x in final]
