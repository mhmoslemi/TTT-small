        if R_peak <= R_mean * 1.01:
            break  # Already nearly flat
        
        # Multiplicative correction: reduce power where it contributes to peaks
        # The autocorrelation R[k] = sum_m power[m] * exp(2*pi*i*m*k/s)
        # To reduce R at its peak locations, we reduce power at all frequencies
        # weighted by their contribution to the peak
        peak_indices = np.where(R_lin >= R_peak * 0.95)[0]
        
        # Build correction: for each frequency m, compute its contribution to peaks
        correction = np.ones(s)
        for pk in peak_indices:
            # Contribution of frequency m to R[pk] is power[m] * cos(2*pi*m*pk/s)
            # We want to reduce total contribution, so reduce proportionally
            contrib = np.cos(2 * np.pi * np.arange(s) * pk / s)
            # Only reduce where contrib > 0 (positive contribution to peak)
            neg_mask = contrib < 0
            correction[neg_mask] *= np.exp(-alpha * R_peak * 0.1)
        
        # Simpler approach: just dampen high-power components
        power_max = power.max()
        if power_max > 0:
            normalized_power = power / power_max
            # Reduce components that are too concentrated
            correction = 1.0 - 0.3 * normalized_power
            correction = np.maximum(correction, 0.1)
        
        power_new = power * correction
        # Renormalize to keep total power the same
        if power_new.sum() > 1e-10:
            power_new = power_new * (power.sum() / power_new.sum())
        
        phase = np.angle(F)
        F_new = np.sqrt(power_new) * np.exp(1j * phase)
        A_new = np.fft.ifft(F_new).real
        a_new = np.maximum(A_new[:n], 0.0)
        sm = a_new.sum()
        if sm < 1e-10:
            break
        a_new = a_new / sm
        
        J_new = compute_J(a_new)
        if J_new < current_J - 1e-15:
            a = a_new
            current_J = J_new
            if current_J < best_J:
                best_J = current_J
                best_a = a.copy()
                stall = 0
            else:
                stall += 1
            alpha = min(alpha * 1.1, 5.0)
        else:
            stall += 1
            alpha *= 0.8
            if alpha < 0.05:
                alpha = 0.3
        
        if stall > 30:
            break
        if it % 20 == 0:
            tracker.update(a)
    
    # Return the best found, not the last
    if best_J < compute_J(a):
        a = best_a
    tracker.update(a)
    return a

def lse_gradient_descent(a, max_iters, deadline, tracker):
    """LSE-approximated gradient descent with momentum and backtracking."""
    a = np.maximum(a.astype(np.float64), 1e-6)
    a = a / a.sum()
    n = len(a)
    current_J = compute_J(a)
    tracker.update(a)
    
    s = 1
    while s < 3 * n:
        s <<= 1
    
    tau = 30.0
    lr = 0.5 / (n * tau)
    momentum = 0.9
    v = np.zeros(n)
    stall = 0
    
    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        
        b = fast_conv(a)
        b_max = b.max()
        # Softmax weights for the max
        exp_b = np.exp(tau * (b - b_max))
        exp_b_sum = exp_b.sum()
        if exp_b_sum < 1e-10:
            break
        p = exp_b / exp_b_sum
        
        # Gradient of LSE: sum_k p[k] * d(b[k])/d(a[j])
        # d(b[k])/d(a[j]) = a[k-j] + a[j-k] (with bounds checking)
        # Using FFT: grad = 2 * ifft(fft(p) * conj(fft(a)))[:n] (approximately)
        fp = np.fft.rfft(p, s)
        fa = np.fft.rfft(a, s)
        corr = np.fft.irfft(fp * np.conj(fa), s)
        grad = 2.0 * corr[:n]
        
        # Update with momentum
        v = momentum * v + (1.0 - momentum) * grad
        a_new = a - lr * v
        a_new = np.maximum(a_new, 1e-6)
        a_new = a_new / a_new.sum()
        
        J_new = compute_J(a_new)
        if J_new < current_J - 1e-15:
            a = a_new
            current_J = J_new
            stall = 0
            # Increase tau for sharper max approximation
            tau = min(tau * 1.02, 500.0)
            lr = 0.5 / (n * tau)
        else:
            stall += 1
            lr *= 0.5
            v *= 0.3
            tau = max(tau * 0.9, 5.0)
            if lr < 1e-18:
                break
        
        if stall > 50:
            break
        if it % 30 == 0:
            tracker.update(a)
    
    tracker.update(a)
    return a

def subgradient_descent(a, max_iters, deadline, tracker):
    """Subgradient descent with Polyak step size."""
    a = np.maximum(a.astype(np.float64), 1e-8)
    a = a / a.sum()
    n = len(a)
    current_J = compute_J(a)
    tracker.update(a)
    
    # Polyak: need lower bound L on optimal value
    L = 1.0  # theoretical lower bound
    best_J = current_J
    
    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        
        b = fast_conv(a)
        M = b.max()
        M_t = M * 0.99  # slightly below max for robustness
        
        # Find active set
        active = np.where(b >= M_t)[0]
        
        # Subgradient: average of a[k-j] over active k
        grad = np.zeros(n)
        for k in active:
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            for j in range(js, je + 1):
                grad[j] += a[k - j]
        grad /= len(active)
        
        # Polyak step size
        gap = current_J - L
        if gap < 1e-6:
            break
        lr = gap / (np.linalg.norm(grad) ** 2 + 1e-10)
        lr = min(lr, 2.0 / n)
        
        a_new = a - lr * grad
        a_new = np.maximum(a_new, 1e-8)
        sm = a_new.sum()
        if sm < 1e-10:
            break
        a_new = a_new / sm
        
        J_new = compute_J(a_new)
        if J_new < current_J:
            a = a_new
            current_J = J_new
            best_J = min(best_J, J_new)
        elif time.monotonic() > deadline - 10:
            break
    
    tracker.update(a)
    return a

def generate_inits(n, rng, parent=None):
    """Generate diverse initializations."""
    inits = []
    t = np.linspace(0, 1, n, endpoint=False)
    
    if parent is not None:
        inits.append(_resample(parent, n))
        inits.append(_resample(parent[::-1], n))
    
    inits.append(np.ones(n) / n)
    inits.append(rng.uniform(0.0, 1.0, n) + 0.1)
    
    # Flat spectrum inits
    for _ in range(3):
        inits.append(_flat_spectrum(n, rng))
    
    # Cosine-based
    for freq in [1, 2, 3, 5, 7, 11]:
        for amp in [0.2, 0.4, 0.6]:
            inits.append(1.0 + amp * np.cos(2 * np.pi * freq * t + rng.uniform(0, 6.28)))
    
    # Multi-frequency
    inits.append(1.0 + 0.3 * np.sin(2 * np.pi * t) + 0.2 * np.sin(4 * np.pi * t)
                + 0.1 * np.sin(9 * np.pi * t))
    
    # Smoothened random
    r = rng.uniform(0, 1, n)
    kernel = np.ones(max(3, n // 20)) / max(3, n // 20)
    inits.append(np.convolve(r, kernel, mode='same') + 0.1)
    
    # Triangular
    inits.append(1.0 - np.abs(2 * t - 1.0))
    
    # Gaussian bumps
    for cx in [0.3, 0.5, 0.7]:
        inits.append(np.exp(-20 * (t - cx) ** 2) + 0.1)
    
    # Power law
    inits.append((t + 0.01) ** rng.uniform(-0.5, 0.5))
    
    # Sine power
    inits.append(np.abs(np.sin(np.pi * t)) ** rng.uniform(0.5, 3.0) + 0.1)
    
    # Random Fourier
    n_c = min(10, n // 4)
    coeffs = rng.normal(0, 1, n_c)
    sig = np.zeros(n)
    for i in range(n_c):
        sig += coeffs[i] * np.cos(2 * np.pi * (i + 1) * t)
    inits.append(np.abs(sig) + 0.5)
    
    # Sparse + smooth
    n_spikes = max(5, n // 10)
    sparse = np.zeros(n)
    positions = rng.choice(n, size=n_spikes, replace=False)
    sparse[positions] = 1.0
    k = max(3, n // 25)
    kernel = np.ones(k) / k
    inits.append(np.convolve(sparse, kernel, mode='same') + 0.05)
    
    for x in inits:
        x = np.maximum(x, 0.001).astype(np.float64)
        x = x / x.sum()
    return inits

def _flat_spectrum(n, rng):
    half = n // 2 + 1
    angles = rng.uniform(0, 2 * np.pi, half)
    A = np.zeros(n, dtype=complex)
    A[:half] = np.exp(1j * angles[:half])
    for k in range(1, n - n // 2):
        A[k] = np.conj(A[n - k])
    a = np.fft.ifft(A).real
    a = np.maximum(a, 0.0)
    sm = a.sum()
    if sm < 1e-10:
        return np.ones(n) / n
    return a / sm

def _resample(src, dest_n):
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)

"""
Minimizes J(a) = 2*n*max(convolve(a,a))/(sum(a))^2 using:
1. Spectral water-filling: multiplicative power spectrum update to flatten autocorrelation (NEW)
2. Warm-started LP best-response with adaptive tightening schedule
3. LSE-smoothed gradient descent with momentum and backtracking
4. Subgradient with Polyak step size
5. Multi-length ensemble search (n=384..1024) with diverse inits
6. Targeted polishing with perturbation + re-optimization
"""

import numpy as np
from scipy.optimize import linprog
from scipy.sparse import csr_matrix
import time

_lp_build_cache = {}

def _build_lp_arrays(n):
    if n not in _lp_build_cache:
        rows = []
        cols = []
        for k in range(2 * n - 1):
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            for j in range(js, je + 1):
                rows.append(k)
                cols.append(j)
        _lp_build_cache[n] = (np.array(rows, dtype=np.int32), np.array(cols, dtype=np.int32))
    return _lp_build_cache[n]

def fast_conv(a):
    n = len(a)
    s = 1
    while s < 2 * n - 1:
        s <<= 1
    fa = np.fft.rfft(a, s)
    return np.fft.irfft(fa * fa, s)[:2 * n - 1]

def compute_J(a):
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))

def lp_solve(a, M, n, time_limit=5):
    rows, cols = _build_lp_arrays(n)
    nnz = len(rows)
    vals = np.empty(nnz)
    for i in range(nnz):
        k = rows[i]
        j = cols[i]
        vals[i] = a[k - j]
    A = csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                options={'presolve': True, 'time_limit': time_limit})
    if r.success and r.x is not None:
        return np.maximum(r.x, 0.0)
    return None

def run_lp_opt(a, max_iters, deadline, tracker, tighten_schedule=None):
    a = np.maximum(a.astype(np.float64), 1e-8)
    a = a / a.sum()
    n = len(a)
    current_J = compute_J(a)
    tracker.update(a)

    if tighten_schedule is None:
        tighten_schedule = [0.02, 0.015, 0.01, 0.008, 0.005, 0.003, 0.002]

    phase = 0
    iters_since_phase = 0
    iters_per_phase = max(3, max_iters // len(tighten_schedule))

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        if iters_since_phase >= iters_per_phase and phase < len(tighten_schedule) - 1:
            phase += 1
            iters_since_phase = 0
        iters_since_phase += 1

        tighten = tighten_schedule[phase]
        b = fast_conv(a)
        M = float(b.max())
        M_tight = M * (1.0 - tighten)

        g = lp_solve(a, M_tight, n)
        if g is None or g.sum() < 1e-14:
            g = lp_solve(a, M, n)
        if g is None or g.sum() < 1e-14:
            break
        g = g / g.sum()

        best_a = None
        best_J = current_J
        progress = it / max(1, max_iters)
        base_t = max(0.08, 0.7 * (1.0 - 0.6 * progress))
        t_candidates = sorted(set([base_t, base_t * 0.5, base_t * 1.5, 0.2, 0.5, 0.8, 0.95]))
        for t in t_candidates:
            if t <= 0 or t > 1.0:
                continue
            a_trial = np.maximum((1.0 - t) * a + t * g, 0.0)
            ss = a_trial.sum()
            if ss < 1e-10:
                continue
            a_trial = a_trial / ss
            J_trial = compute_J(a_trial)
            if J_trial < best_J - 1e-15:
                best_J = J_trial
                best_a = a_trial

        if best_a is None:
            break
        a = best_a
        current_J = best_J
        if it % 8 == 0:
            tracker.update(a)

    tracker.update(a)
    return a

def spectral_water_filling(a, max_iters, deadline, tracker, alpha_init=0.3):
    a = np.maximum(a.astype(np.float64), 1e-8)
    a = a / a.sum()
    n = len(a)
    s = 1
    while s < 2 * n - 1:
        s <<= 1

    current_J = compute_J(a)
    tracker.update(a)
    best_J = current_J
    best_a = a.copy()
    alpha = alpha_init
    stall = 0

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        A = np.zeros(s)
        A[:n] = a
        F = np.fft.fft(A)
        power = np.abs(F) ** 2
        R = np.fft.ifft(power).real[:2 * n - 1]

        R_peak = R.max()
        R_mean = R.sum() / len(R)
        if R_peak <= R_mean * 1.005:
            break

        # Multiplicative flattening: dampen frequencies contributing to peaks
        # R[k] = sum_m power[m] * exp(2*pi*i*m*k/s)
        # We want to reduce R where it's high.
        # Simple approach: reduce power where it's concentrated (high power = high R contribution)
        p_max = power.max()
        if p_max < 1e-15:
            break
        norm_p = power / p_max
        # Create correction: reduce high-power components more
        correction = 1.0 - alpha * norm_p
        correction = np.maximum(correction, 0.05)

        power_new = power * correction
        total_old = power.sum()
        total_new = power_new.sum()
        if total_new < 1e-15:
            break
        power_new = power_new * (total_old / total_new)

        phase = np.angle(F)
        F_new = np.sqrt(np.maximum(power_new, 0)) * np.exp(1j * phase)
        A_new = np.fft.ifft(F_new).real
        a_new = np.maximum(A_new[:n], 0.0)
        sm = a_new.sum()
        if sm < 1e-10:
            break
        a_new = a_new / sm

        J_new = compute_J(a_new)
        if J_new < current_J - 1e-14:
            a = a_new
            current_J = J_new
            if current_J < best_J:
                best_J = current_J
                best_a = a.copy()
                stall = 0
            else:
                stall += 1
            alpha = min(alpha * 1.08, 10.0)
        else:
            stall += 1
            alpha *= 0.85
            if alpha < 0.02:
                alpha = 0.15
            if stall > 40:
                break
        if it % 30 == 0:
            tracker.update(a)

    if best_J < compute_J(a):
        a = best_a
    tracker.update(a)
    return a

def lse_gradient_descent(a, max_iters, deadline, tracker):
    a = np.maximum(a.astype(np.float64), 1e-6)
    a = a / a.sum()
    n = len(a)
    current_J = compute_J(a)
    tracker.update(a)

    s = 1
    while s < 3 * n:
        s <<= 1

    tau = 20.0
    lr = 0.3 / (n * tau)
    momentum = 0.88
    v = np.zeros(n)
    stall = 0

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        b = fast_conv(a)
        b_max = b.max()
        exp_b = np.exp(tau * (b - b_max))
        eps_sum = exp_b.sum()
        if eps_sum < 1e-30:
            break
        p = exp_b / eps_sum

        fp = np.fft.rfft(p, s)
        fa = np.fft.rfft(a, s)
        corr = np.fft.irfft(fp * np.conj(fa), s)
        grad = 2.0 * corr[:n]

        v = momentum * v + (1.0 - momentum) * grad
        a_new = a - lr * v
        a_new = np.maximum(a_new, 1e-6)
        a_new = a_new / a_new.sum()

        J_new = compute_J(a_new)
        if J_new < current_J - 1e-14:
            a = a_new
            current_J = J_new
            stall = 0
            tau = min(tau * 1.03, 800.0)
            lr = 0.3 / (n * tau)
        else:
            stall += 1
            lr *= 0.5
            v *= 0.3
            tau = max(tau * 0.9, 5.0)
            if lr < 1e-18:
                break
            if stall > 60:
                break
        if it % 40 == 0:
            tracker.update(a)

    tracker.update(a)
    return a

def subgradient_with_polyak(a, max_iters, deadline, tracker):
    a = np.maximum(a.astype(np.float64), 1e-8)
    a = a / a.sum()
    n = len(a)
    current_J = compute_J(a)
    tracker.update(a)
    L = 1.0 + 1e-6  # lower bound

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        b = fast_conv(a)
        M = b.max()
        active = np.where(b >= M * 0.98)[0]
        if len(active) == 0:
            break

        grad = np.zeros(n)
        for k in active:
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            for j in range(js, je + 1):
                grad[j] += a[k - j]
        grad /= len(active)

        gap = current_J - L
        if gap < 1e-7:
            break
        grad_norm_sq = np.dot(grad, grad)
        lr = min(gap / (grad_norm_sq + 1e-10), 3.0 / n)

        a_new = a - lr * grad
        a_new = np.maximum(a_new, 1e-8)
        sm = a_new.sum()
        if sm < 1e-10:
            break
        a_new = a_new / sm

        J_new = compute_J(a_new)
        if J_new < current_J:
            a = a_new
            current_J = J_new
        else:
            lr *= 0.5
            a_new = a - lr * grad
            a_new = np.maximum(a_new, 1e-8)
            a_new = a_new / a_new.sum()
            J_new = compute_J(a_new)
            if J_new < current_J:
                a = a_new
                current_J = J_new

        if it % 20 == 0:
            tracker.update(a)

    tracker.update(a)
    return a

def flat_spectrum_init(n, rng):
    half = n // 2 + 1
    angles = rng.uniform(0, 2 * np.pi, half)
    A = np.zeros(n, dtype=complex)
    A[:half] = np.exp(1j * angles[:half])
    for k in range(1, n - n // 2):
        A[k] = np.conj(A[n - k])
    a = np.fft.ifft(A).real
    a = np.maximum(a, 0.0)
    sm = a.sum()
    if sm < 1e-10:
        return np.ones(n) / n
    return a / sm

def resample_seq(src, dest_n):
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)

def generate_inits(n, rng, parent=None):
    inits = []
    t = np.linspace(0, 1, n, endpoint=False)

    if parent is not None:
        inits.append(resample_seq(parent, n))
        inits.append(resample_seq(parent[::-1], n))

    inits.append(np.ones(n) / n)
    inits.append(rng.uniform(0.0, 1.0, n) + 0.05)

    for _ in range(4):
        inits.append(flat_spectrum_init(n, rng))

    for freq in [1, 2, 3, 5, 7, 11, 13]:
        for amp in [0.15, 0.3, 0.5]:
            inits.append(1.0 + amp * np.cos(2 * np.pi * freq * t + rng.uniform(0, 6.28)))

    inits.append(1.0 + 0.25 * np.sin(2 * np.pi * t) + 0.15 * np.sin(5 * np.pi * t)
                + 0.1 * np.sin(11 * np.pi * t) + 0.05 * np.sin(19 * np.pi * t))

    r = rng.uniform(0, 1, n)
    k = max(3, n // 15)
    kernel = np.ones(k) / k
    inits.append(np.convolve(r, kernel, mode='same') + 0.05)

    inits.append(1.0 - np.abs(2 * t - 1.0))
    inits.append((1.0 - np.abs(2 * t - 1.0)) ** 2)
    inits.append(np.cos(np.pi * t) ** 2 + 0.1)

    for cx in [0.25, 0.5, 0.75]:
        inits.append(np.exp(-15 * (t - cx) ** 2) + 0.1)

    inits.append((t + 0.01) ** rng.uniform(-0.5, 0.5))
    inits.append(np.abs(np.sin(np.pi * t)) ** rng.uniform(0.5, 3.0) + 0.05)

    n_c = min(12, n // 4)
    coeffs = rng.normal(0, 1, n_c)
    sig = np.zeros(n)
    for i in range(n_c):
        sig += coeffs[i] * np.cos(2 * np.pi * (i + 1) * t + rng.uniform(0, 6.28))
    inits.append(np.abs(sig) + 0.5)

    n_spikes = max(5, n // 12)
    sparse = np.zeros(n)
    positions = rng.choice(n, size=n_spikes, replace=False)
    sparse[positions] = 1.0
    k2 = max(3, n // 20)
    kernel2 = np.ones(k2) / k2
    inits.append(np.convolve(sparse, kernel2, mode='same') + 0.03)

    # Aliasing-resistant: alternating high/low
    alt = np.ones(n)
    alt[::2] = rng.uniform(0.3, 1.5)
    alt[1::2] = rng.uniform(0.3, 1.5)
    inits.append(alt)

    for x in inits:
        x = np.maximum(x, 0.001).astype(np.float64)
        x = x / x.sum()
    return inits

class ScoreTracker:
    def __init__(self):
        self.best = None
        self.best_score = float('inf')
        self.t_start = time.monotonic()

    def update(self, a):
        a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
        if a.size == 0 or a.sum() < 0.01:
            return
        score = float(evaluate_sequence(a.tolist()))
        if np.isfinite(score) and score < self.best_score:
            self.best = a.copy()
            self.best_score = score
            elapsed = time.monotonic() - self.t_start
            print(f"  *** NEW BEST: J={score:.8f} n={a.size} t={elapsed:.1f}s", flush=True)

    def report(self, label=""):
        if self.best is not None:
            print(f"  {label} best: J={self.best_score:.8f} n={self.best.size}", flush=True)

def multi_method_optimize(a, deadline, tracker, n_lp=20, n_spec=150, n_lse=500, n_sg=200):
    """Run a sequence of optimization methods on a."""
    n = len(a)

    if time.monotonic() < deadline - 30:
        a = run_lp_opt(a, n_lp, deadline - 30, tracker,
                       tighten_schedule=[0.02, 0.01, 0.005, 0.003])
    if time.monotonic() < deadline - 20:
        a = spectral_water_filling(a, n_spec, deadline - 20, tracker)
    if time.monotonic() < deadline - 10:
        a = lse_gradient_descent(a, n_lse, deadline - 10, tracker)
    if time.monotonic() < deadline - 5:
        a = run_lp_opt(a, n_lp // 2, deadline - 5, tracker,
                       tighten_schedule=[0.005, 0.003, 0.002])
    if time.monotonic() < deadline - 2:
        a = subgradient_with_polyak(a, n_sg, deadline - 2, tracker)
    if time.monotonic() < deadline - 1:
        a = lse_gradient_descent(a, 200, deadline - 1, tracker)

    return a

def propose_candidate(seed=42, budget_s=1500, **kwargs):
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + float(budget_s) - 30.0
    tracker = ScoreTracker()

    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        if p.ndim == 1 and p.size > 0 and np.all(np.isfinite(p)):
            parent = np.maximum(p, 0.0)
            tracker.update(parent)
            print(f"Parent: n={parent.size}, J={tracker.best_score:.8f}", flush=True)
    except (NameError, AttributeError):
        pass

    # Phase 1: Multi-length exploration
    if parent is not None:
        lengths = [parent.size, max(384, parent.size - 128), parent.size + 128,
                   parent.size + 256, parent.size + 512]
    else:
        lengths = [384, 512, 640, 768, 1024]

    lengths = sorted(set([n for n in lengths if 128 <= n <= 2048]))
    n_lens = len(lengths)

    total_explore_time = deadline - time.monotonic() - 400
    if total_explore_time < 50:
        total_explore_time = 50

    # Weight toward parent length
    if parent is not None:
        pn = parent.size
        weights = [1.5 if n == pn else 1.0 / (1.0 + 0.05 * abs(n - pn) / 64.0) for n in lengths]
    else:
        weights = [1.0 / (1.0 + 0.03 * abs(n - 512) / 64.0) for n in lengths]
    wsum = sum(weights)
    time_alloc = [total_explore_time * w / wsum for w in weights]

    print(f"\nPhase 1: Multi-length exploration, total budget={total_explore_time:.1f}s", flush=True)
    print(f"Lengths: {lengths}, Time alloc: {[f'{t:.0f}s' for t in time_alloc]}", flush=True)

    for i, n in enumerate(lengths):
        t_start = time.monotonic()
        t_end = t_start + time_alloc[i]
        t_end = min(t_end, deadline - 350)

        if time.monotonic() >= t_end - 15:
            continue

        print(f"\n  --- n={n}, budget={time_alloc[i]:.1f}s ---", flush=True)

        inits = generate_inits(n, rng, parent)
        n_inits = len(inits)
        per_init_time = (t_end - time.monotonic()) / n_inits

        local_best = None
        local_best_J = float('inf')

        for idx, a0 in enumerate(inits):
            t_init_end = min(time.monotonic() + per_init_time, t_end - 8)
            if time.monotonic() > t_end - 10:
                break

            # Adaptive optimization depth based on available time
            avail = t_init_end - time.monotonic()
            if avail < 5:
                # Very tight: just LP
                a = run_lp_opt(a0, max(3, int(avail / 0.5)), t_init_end, tracker,
                              tighten_schedule=[0.02, 0.01])
            elif avail < 20:
                a = multi_method_optimize(a0, t_init_end, tracker,
                                          n_lp=max(5, int(avail / 3)), n_spec=50,
                                          n_lse=max(50, int(avail * 5)), n_sg=50)
            else:
                a = multi_method_optimize(a0, t_init_end, tracker,
                                          n_lp=max(10, int(avail / 4)), n_spec=100,
                                          n_lse=max(100, int(avail * 3)), n_sg=100)

            J = compute_J(a)
            if J < local_best_J:
                local_best_J = J
                local_best = a.copy()

        if local_best is not None:
            tracker.update(local_best)
            print(f"  n={n}: local best J={local_best_J:.8f} ({time.monotonic()-t_start:.1f}s)", flush=True)

    tracker.report("After Phase 1")

    # Phase 2: Focused deep optimization
    if tracker.best is not None and time.monotonic() < deadline - 150:
        t2_start = time.monotonic()
        t2_end = deadline - 100
        n_best = tracker.best.size
        print(f"\nPhase 2: Deep focus on n={n_best}, budget={t2_end-t2_start:.1f}s", flush=True)

        a = tracker.best.copy().astype(np.float64)

        # Progressive tightening LP
        tighten_levels = [0.015, 0.01, 0.008, 0.006, 0.004, 0.003, 0.002, 0.001]
        for tl in tighten_levels:
            if time.monotonic() > t2_end - 60:
                break
            a = run_lp_opt(a, 20, t2_end - 60, tracker,
                          tighten_schedule=[tl, tl * 0.7, tl * 0.5, tl * 0.3])

        # Spectral
        if time.monotonic() < t2_end - 40:
            a = spectral_water_filling(a, 400, t2_end - 40, tracker, alpha_init=0.5)

        # LSE
        if time.monotonic() < t2_end - 30:
            a = lse_gradient_descent(a, 3000, t2_end - 30, tracker)

        # Final LP
        if time.monotonic() < t2_end - 15:
            a = run_lp_opt(a, 15, t2_end - 15, tracker,
                          tighten_schedule=[0.003, 0.002, 0.001, 0.0005])

        # Final LSE
        if time.monotonic() < t2_end - 5:
            a = lse_gradient_descent(a, 1000, t2_end - 5, tracker)

        tracker.report(f"Phase 2 ({time.monotonic()-t2_start:.1f}s)")

    # Phase 3: Perturbation and re-optimization
    if tracker.best is not None and time.monotonic() < deadline - 30:
        t3_start = time.monotonic()
        t3_end = deadline - 10
        print(f"\nPhase 3: Perturbation search, budget={t3_end-t3_start:.1f}s", flush=True)

        n_pert = tracker.best.size
        perturbations = [0.005, 0.01, 0.02, 0.05, 0.1, 0.15, 0.2, 0.3]

        for ps in perturbations:
            if time.monotonic() > t3_end - 10:
                break
            for _ in range(3):
                if time.monotonic() > t3_end - 10:
                    break
                a_p = tracker.best + rng.normal(0, ps, n_pert)
                a_p = np.maximum(a_p, 1e-6)
                a_p = a_p / a_p.sum()
                a_p = multi_method_optimize(a_p, t3_end - 5, tracker,
                                            n_lp=10, n_spec=80, n_lse=300, n_sg=50)

        # One final polish on the absolute best
        if time.monotonic() < t3_end - 5:
            a_final = tracker.best.copy()
            a_final = run_lp_opt(a_final, 10, t3_end - 3, tracker,
                                tighten_schedule=[0.002, 0.001])
            if time.monotonic() < t3_end - 1:
                a_final = lse_gradient_descent(a_final, 300, t3_end - 1, tracker)

        tracker.report(f"Phase 3 ({time.monotonic()-t3_start:.1f}s)")

    # Fallback
    if tracker.best is None:
        tracker.best = np.ones(256)
        tracker.best_score = compute_J(tracker.best)
        tracker.update(tracker.best)

    print(f"\n{'='*50} FINAL RESULT {'='*50}", flush=True)
    print(f"J = {tracker.best_score:.8f}, n = {tracker.best.size}, "
          f"time = {time.monotonic()-tracker.t_start:.1f}s", flush=True)
    return [float(x) for x in tracker.best]
