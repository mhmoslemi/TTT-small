from scipy.optimize import linprog
from scipy.sparse import csr_matrix, coo_matrix
import time

def fast_conv(a):
    n = len(a)
    s = 1
    while s < 2 * n - 1:
        s <<= 1
    fa = np.fft.rfft(a, s)
    return np.fft.irfft(fa * fa, s)[:2 * n - 1]

def compute_J(a):
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))

def build_conv_matrix(n):
    """Build the sparse matrix A where (A @ g)[k] = conv(a, g)[k] for fixed a."""
    # This depends on a, so we can't pre-build it. But the structure is fixed.
    rows = []
    cols = []
    for k in range(2 * n - 1):
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        for j in range(js, je + 1):
            rows.append(k)
            cols.append(j)
    return np.array(rows), np.array(cols)

# Pre-compute the structure
_struct_cache = {}

def get_structure(n):
    if n not in _struct_cache:
        rows = []
        cols = []
        for k in range(2 * n - 1):
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            rows.extend([k] * (je - js + 1))
            cols.extend(range(js, je + 1))
        _struct_cache[n] = (np.array(rows), np.array(cols))
    return _struct_cache[n]

def lp_best_response(a, M_budget):
    """Find g >= 0 that maximizes sum(g) subject to conv(a,g) <= M_budget."""
    n = len(a)
    rows, cols = get_structure(n)
    nnz = len(rows)
    vals = np.empty(nnz)
    for i in range(nnz):
        k = rows[i]
        j = cols[i]
        idx = k - j
        if 0 <= idx < n:
            vals[i] = a[idx]
        else:
            vals[i] = 0.0
    A = csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M_budget)
    bounds = [(0.0, None)] * n
    r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                options={'presolve': True, 'time_limit': 30})
    if r.success and r.x is not None:
        return np.maximum(r.x, 0.0)
    return None

def get_structure(n):
    if n not in _struct_cache:
        rows = []
        cols = []
        for k in range(2 * n - 1):
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            rows.extend([k] * (je - js + 1))
            cols.extend(range(js, je + 1))
        _struct_cache[n] = (np.array(rows, dtype=np.int32), np.array(cols, dtype=np.int32))
    return _struct_cache[n]

def fast_conv(a):
    n = len(a)
    s = 1
    while s < 2 * n - 1:
        s <<= 1
    fa = np.fft.rfft(a, s)
    return np.fft.irfft(fa * fa, s)[:2 * n - 1]

    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))

def lp_solve(a, M_budget, time_limit=5.0):
    """Solve LP: max sum(g) s.t. conv(a,g) <= M_budget, g >= 0."""
    n = len(a)
    rows, cols = get_structure(n)
    nnz = len(rows)
    # Build values
    vals = np.empty(nnz)
    for i in range(nnz):
        k = int(rows[i])
        j = int(cols[i])
        idx = k - j
        if 0 <= idx < n:
            vals[i] = a[idx]
        else:
            vals[i] = 0.0
    A = csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M_budget)
    bounds = [(0.0, None)] * n
    r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                options={'presolve': True, 'time_limit': time_limit})
    if r.success and r.x is not None:
        return np.maximum(r.x, 0.0)
    return None

    valid = (idx >= 0) & (idx < n)
    vals[valid] = a[idx[valid]]
    vals[~valid] = 0.0
    return vals

    b = fast_conv(a)
    M = float(b.max())
    tol = max(1e-12, 1e-6 * M)
    active = np.where(b >= M - tol)[0]
    grad = np.zeros(n)
    for k in active:
        diffs = np.abs(k - np.arange(n))
        valid = diffs < n
        contrib = np.zeros(n)
        contrib[valid] = 2.0 * a[diffs[valid]]
        grad += contrib
    grad /= len(active)
    return grad, M

def lse_gradient(a, tau):
    """Gradient of LSE approximation to max(a*a)."""
    n = len(a)
    b = fast_conv(a)
    b_max = b.max()
    exp_b = np.exp(tau * (b - b_max))
    p = exp_b / exp_b.sum()
    # grad of LSE = sum_k p[k] * d(b[k])/da
    # d(b[k])/da[m] = 2*a[|k-m|] if |k-m| < n
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        if p[k] < 1e-10:
            continue
        diffs = np.abs(k - np.arange(n))
        valid = diffs < n
        contrib = np.zeros(n)
        contrib[valid] = 2.0 * a[diffs[valid]]
        grad += p[k] * contrib
    return grad, b_max

def resample_seq(src, dest_n):
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)

def flat_spectrum_init(n, rng):
    half = n // 2 + 1
    angles = rng.uniform(0, 2 * np.pi, half)
    A = np.zeros(n, dtype=complex)
    A[:half] = np.exp(1j * angles[:half])
    for k in range(1, n - n // 2):
        A[k] = np.conj(A[n - k])
    a = np.fft.ifft(A).real
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-10:
        return np.ones(n) / n
    return a / s

def generate_inits(n, rng, parent):
    inits = []
    if parent is not None and parent.size > 0:
        inits.append(resample_seq(parent, n))
        inits.append(resample_seq(parent[::-1], n))
    
    inits.append(np.ones(n) / n)
    
    t = np.linspace(0, 1, n, endpoint=False)
    for seed in range(6):
        a = flat_spectrum_init(n, rng)
        inits.append(a)
    
    # Cosine-modulated
    for freq in [1, 2, 3, 5, 7]:
        inits.append(1.0 + 0.3 * np.cos(2 * np.pi * freq * t))
    
    # Bump functions
    for center in [0.2, 0.5, 0.8]:
        inits.append(np.exp(-50 * (t - center) ** 2) + 0.05)
    
    # Random
    inits.append(rng.uniform(0.1, 1.0, n))
    
    for x in inits:
        x = np.maximum(x, 0.001).astype(np.float64)
        x /= x.sum()
    return inits

    """Run LP best-response iterations until deadline."""

    iter_count = 0
    
    while time.monotonic() < deadline:
        iter_count += 1
        b = fast_conv(a)
        M = float(b.max())
        
        # Adaptive tightening
        M_budget = M * (1.0 - tighten)
        g = lp_solve(a, M_budget, time_limit=max_time_per_lp)
        
        if g is None or g.sum() < 1e-10:
            # Try less tightening
            tighten = max(0.001, tighten * 0.5)
            M_budget = M * (1.0 - tighten)
            g = lp_solve(a, M_budget, time_limit=max_time_per_lp)
            if g is None or g.sum() < 1e-10:
                break
        
        g = np.maximum(g, 0.0)
        g = g / g.sum()
        
        # Try different mixing ratios
        best_a = None
        best_J = current_J
        t_candidates = [0.3, 0.5, 0.7, 0.9, 1.0, 0.2, 0.4, 0.6, 0.8]
        for t in t_candidates:
            a_trial = (1.0 - t) * a + t * g
            a_trial = np.maximum(a_trial, 1e-10)
            a_trial /= a_trial.sum()
            J_trial = compute_J(a_trial)
            if J_trial < best_J:
                best_J = J_trial
                best_a = a_trial
        
        if best_a is not None:
            a = best_a
            current_J = best_J
            no_improve_count = 0
            # Gradually increase tightening when making progress
            if iter_count % 20 == 0:
                tighten = min(0.05, tighten * 1.2)
        else:
            no_improve_count += 1
            if no_improve_count > 10:
                # Reset tightening and try a different approach
                tighten = 0.001
                no_improve_count = 0
                # Perturb
                rng = np.random.default_rng(iter_count)
                a_pert = a + rng.normal(0, 0.05, n)
                a_pert = np.maximum(a_pert, 1e-10)
                a_pert /= a_pert.sum()
                if compute_J(a_pert) < current_J:
                    a = a_pert
                    current_J = compute_J(a)
                tracker.update(a)
                # Give up on LP if stuck for too long
                if tighten < 0.001:
                    break
            else:
                tighten = max(0.001, tighten * 0.7)
        
        if iter_count % 20 == 0:
            tracker.update(a)
    
    tracker.update(a)
    print(f"  LP phase: {iter_count} iters, J={compute_J(a):.8f}", flush=True)
    return a

The LP: max sum(g) s.t. conv(a,g) ≤ M_budget, g ≥ 0.

If M_budget = M (no tightening), the LP is always feasible (g=0 is feasible, and g=a is feasible since conv(a,a) ≤ M). The optimal g will have sum(g) ≥ sum(a) = 1 (after normalization).

If we tighten (M_budget < M), the LP becomes harder. The optimal g will be "more spread out" than a.

The key insight: the LP with M_budget = M gives us a g that's at least as good as a (in terms of sum). The mixing a ← (1-t)*a + t*g should improve the objective if g has a flatter convolution profile.
