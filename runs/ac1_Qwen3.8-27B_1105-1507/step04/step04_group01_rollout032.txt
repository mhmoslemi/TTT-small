```python
def solve_lp_minpeak_given_a(a, time_limit=30):
    """Solve min t s.t. (a*g)[k] <= t ∀k, sum(g)=1, g>=0."""
    n = len(a)
    # Build Toeplitz matrix T where (T[g])[k] = (a*g)[k]
    T = np.zeros((2*n-1, n))
    for j in range(n):
        # (a*g)[k] = sum_j a[j]*g[k-j], so for fixed j, g[k-j] appears in row k=j, j+1, ..., j+n-1
        # T[k, j] = a[j] if k-j is a valid index for g, i.e., 0<=k-j<n
        # So for row k, column j: a[j] if 0<=k-j<n, i.e., j<=k and k-j<n, i.e., k-n+1<=j<=k
        # For column j: rows k from j to j+n-1 (intersected with [0, 2n-2])
        k_start = j
        k_end = min(j+n, 2*n-1)
        T[k_start:k_end, j] = a[:k_end-k_start]
    
    # LP: min t
    # s.t. T@g - t <= 0 (2n-1 constraints)
    #      sum(g) = 1
    #      g >= 0, t >= 0 (t can be unrestricted actually, but for safety)
    
    c = np.zeros(n+1)
    c[n] = 1.0  # minimize t
    
    A_ub = np.zeros((2*n-1, n+1))
    A_ub[:, :n] = T
    A_ub[:, n] = -1.0
    b_ub = np.zeros(2*n-1)
    
    A_eq = np.zeros((1, n+1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])
    
    bounds = [(0, None)]*n + [(None, None)]  # t is free
    
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq, 
                  bounds=bounds, method='highs', options={'presolve': True})
    if res.status == 0:
        g = np.maximum(res.x[:n], 0)
        t = res.x[n]
        return g, t
    return None, None
```

- Direction: d = g - a
- Find t* = argmin_{t in [0,1]} max((a + t*d) * (a + t*d))
- Update: a = a + t* * d

This is a bit expensive (each evaluation requires computing the full convolution), but n is only ~600 so it's fast.

import numpy as np
from scipy.optimize import linprog
from scipy.fft import rfft, irfft
import time

def fft_convolve(a, b):
    n, m = len(a), len(b)
    size = 1
    while size < n + m - 1:
        size <<= 1
    fa = rfft(a, size)
    fb = rfft(b, size)
    return irfft(fa * fb, size)[:n + m - 1]

def compute_obj(a):
    n = len(a)
    conv = fft_convolve(a, a)
    max_b = np.max(conv)
    sum_a = np.sum(a)
    if sum_a < 0.01:
        return np.inf
    return 2.0 * n * max_b / (sum_a * sum_a)

def build_toeplitz(a):
    n = len(a)
    T = np.zeros((2*n-1, n))
    for j in range(n):
        k_start = j
        k_end = min(j+n, 2*n-1)
        T[k_start:k_end, j] = a[:k_end-k_start]
    return T

def solve_lp_given_a(a, time_limit=30):
    n = len(a)
    T = build_toeplitz(a)
    c = np.zeros(n+1)
    c[n] = 1.0
    A_ub = np.zeros((2*n-1, n+1))
    A_ub[:, :n] = T
    A_ub[:, n] = -1.0
    b_ub = np.zeros(2*n-1)
    A_eq = np.zeros((1, n+1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])
    bounds = [(0, None)]*n + [(None, None)]
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                  bounds=bounds, method='highs', options={'presolve': True, 'time_limit': time_limit})
    if res.status == 0:
        g = np.maximum(res.x[:n], 0)
        s = np.sum(g)
        if s > 1e-10:
            g = g / s
        t = res.x[n]
        return g, t
    return None, None

def alternating_lp(a, max_iters=100, time_budget=60.0):
    n = len(a)
    a = a / np.sum(a)  # normalize
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    
    for it in range(max_iters):
        if time.monotonic() - start > time_budget:
            break
        
        g, t = solve_lp_given_a(a, time_limit=15)
        if g is None:
            break
        
        # Line search
        best_t = 1.0
        best_a_ls = g.copy()
        best_obj_ls = compute_obj(g)
        for t in [0.1, 0.3, 0.5, 0.7, 0.9, 1.0]:
            a_try = (1-t)*a + t*g
            a_try = np.maximum(a_try, 0)
            s = np.sum(a_try)
            if s < 1e-10:
                continue
            a_try = a_try / s
            obj = compute_obj(a_try)
            if obj < best_obj_ls:
                best_obj_ls = obj
                best_a_ls = a_try.copy()
                best_t = t
        
        if best_obj_ls < best_obj:
            best_obj = best_obj_ls
            best_a = best_a_ls.copy()
            a = best_a_ls.copy()
        else:
            # No improvement, try smaller step or break
            a = 0.9*a + 0.1*g
            a = np.maximum(a, 0)
            a = a / np.sum(a)
    
    return best_a

def soft_max_grad(a, tau):
    """Compute soft-max of conv(a,a) and gradient w.r.t. a."""
    n = len(a)
    conv = fft_convolve(a, a)
    cmax = np.max(conv)
    shifted = np.clip((conv - cmax) / tau, -500, 500)
    exp_conv = np.exp(shifted)
    Z = np.sum(exp_conv)
    soft_max_val = cmax + tau * np.log(Z)
    w = exp_conv / Z
    
    # Correct gradient formula
    a_rev = a[::-1]
    conv1 = fft_convolve(a_rev, w)  # length n + (2n-1) - 1 = 3n-2
    conv2 = fft_convolve(a, w)      # same length
    grad = 2.0 * (conv1[n-1:2*n-1] + conv2[:n] - w[:n] * a[0])
    
    return soft_max_val, grad
