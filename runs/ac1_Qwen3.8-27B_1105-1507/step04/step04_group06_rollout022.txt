import numpy as np

# Cache for FFT sizes

def fft_conv(a):
    """Full self-convolution via FFT."""
    n = len(a)
    s = 1
    while s < 2 * n - 1:
        s <<= 1
    fa = np.fft.rfft(a, s)
    return np.fft.irfft(fa * fa, s)[:2 * n - 1]


def compute_J(a):
    """Compute J(a) = 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fft_conv(a)
    return float(2 * n * b.max() / (s * s))


def smooth_max_and_grad(b, beta):
    """
    Compute smooth max of b (log-sum-exp) and softmax weights.
    Returns (smooth_value, p) where p_k = softmax(beta*(b_k - b_max)).
    """
    b_max = float(b.max())
    shifted = beta * (b - b_max)
    shifted = np.clip(shifted, -500.0, 0.0)
    exp_shifted = np.exp(shifted)
    Z = float(exp_shifted.sum())
    if Z < 1e-300:
        p = np.zeros_like(b)
        p[np.argmax(b)] = 1.0
    else:
        p = exp_shifted / Z
    g_val = b_max + np.log(max(Z, 1e-300)) / beta
    return g_val, p


def F_and_grad(a, beta):
    """
    Compute F(a) = 2n*smooth_max(conv(a,a))/(sum(a))^2 and its gradient.
    """
    n = len(a)
    s = float(a.sum())
    if s < 1e-10:
        return 1e10, np.zeros(n)
    b = fft_conv(a)
    g_val, p = smooth_max_and_grad(b, beta)
    F_val = 2.0 * n * g_val / (s * s)
    # Gradient of smooth_max w.r.t. a
    S = np.convolve(a, p[::-1], mode='full')
    S = S[n - 1:2 * n - 1][::-1]
    grad_m = 2.0 * S
    grad_F = 2.0 * n * (s * grad_m - 2.0 * g_val) / (s ** 3)
    return F_val, grad_F


def nesterov_optimize(a0, beta_schedule, deadline, max_sweeps=2000):
    """
    Nesterov-accelerated gradient descent with backtracking line search.
    beta_schedule: list of (beta, n_iters) pairs.
    """
    a = np.maximum(a0.astype(np.float64), 1e-10)
    a = a / a.sum()
    n = len(a)
    current_F = F_and_grad(a, beta_schedule[0][0])[0]
    best_a = a.copy()
    best_F = current_F

    y = a.copy()  # Nesterov lookahead
    v = np.zeros(n)  # momentum
    gamma = 0.25  # learning rate

    for beta, n_iters in beta_schedule:
        if time.monotonic() > deadline:
            break
        gamma = 1.0 / (beta * n ** 0.5)
        t_k = 1.0  # Nesterov parameter
        y = a.copy()
        v = np.zeros(n)

        for it in range(n_iters):
            if time.monotonic() > deadline:
                break

            # Evaluate at Nesterov point
            F_y, grad_y = F_and_grad(y, beta)
            if not np.isfinite(F_y):
                gamma *= 0.5
                continue

            # Nesterov update
            t_new = (1.0 + np.sqrt(1.0 + 4.0 * t_k ** 2)) / 2.0
            theta = (t_k - 1.0) / t_new
            y_new = y + gamma * grad_y
            y_next = y_new + theta * (y_new - y)

            # Clip to nonnegative
            y_next = np.maximum(y_next, 0.0)
            s_next = y_next.sum()
            if s_next < 1e-10:
                y_next = y.copy()
                s_next = y.sum()

            # Line search (backtracking)
            alpha = gamma
            accepted = False
            for _ in range(20):
                candidate = (1.0 - alpha) * a + alpha * y_next
                candidate = np.maximum(candidate, 0.0)
                sc = candidate.sum()
                if sc < 1e-10:
                    alpha *= 0.5
                    continue
                F_cand = F_and_grad(candidate, beta)[0]
                if np.isfinite(F_cand) and F_cand <= F_y + 1e-10 * alpha * np.dot(grad_y, y_next - a):
                    accepted = True
                    break
                alpha *= 0.5

            if accepted:
                a = (1.0 - alpha) * a + alpha * y_next
                a = np.maximum(a, 1e-15)
                a = a / a.sum()
                F_a = F_and_grad(a, beta)[0]
                if F_a < best_F:
                    best_F = F_a
                    best_a = a.copy()
                y = a + theta * (a - y)
                t_k = t_new
            else:
                gamma *= 0.5
                if gamma < 1e-15:
                    break

    return best_a, best_F


def coordinate_descent(a, n_sweeps, deadline, verbose=False):
    """
    Coordinate descent: optimize one coordinate at a time.
    For coordinate i, we minimize J as a function of a_i (1D problem).
    """
    a = a.astype(np.float64)
    n = len(a)
    best_J = compute_J(a)

    for sweep in range(n_sweeps):
        if time.monotonic() > deadline:
            break
        improved = False
        for i in range(n):
            if time.monotonic() > deadline:
                break
            a_i = a[i]
            S_fix = a.sum() - a_i
            if S_fix < 1e-15:
                continue

            b = fft_conv(a)
            # Precompute slope and constant for each k
            # b_k(x) = const_k + slope_k * x for k != 2i
            # For k = 2i: b_k(x) = b_k - a_i^2 + x^2
            slope = np.zeros(2 * n - 1)
            valid = np.zeros(2 * n - 1, dtype=bool)
            for j in range(n):
                k = i + j
                if k < 2 * n - 1:
                    slope[k] = 2.0 * a[j]
                    valid[k] = True
            # For k=2i, the slope is 2*a_i but the relationship is quadratic
            # b_{2i}(x) = b_{2i} - a_i^2 + x^2
            # For k != 2i: b_k(x) = b_k - 2*a_i*a_{k-i} + 2*x*a_{k-i}
            #              = (b_k - slope_k * a_i) + slope_k * x

            # Define 1D function J(x) for x >= 0
            def j_1d(x):
                if x < 0:
                    return 1e10
                S_new = S_fix + x
                if S_new < 1e-15:
                    return 1e10
                # Compute b_k(x) for all k
                b_x = b.copy()
                # For valid k (k-i in [0,n-1]):
                # If k != 2i: b_k(x) = b_k + 2*(x-a_i)*a_{k-i}
                # If k == 2i: b_k(x) = b_k - a_i^2 + x^2
                k_2i = 2 * i
                for k in range(2 * n - 1):
                    if valid[k]:
                        if k == k_2i:
                            b_x[k] = b[k] - a_i * a_i + x * x
                        else:
                            # k-i is in [0, n-1], and a_{k-i} is the relevant element
                            j_idx = k - i
                            if 0 <= j_idx < n:
                                b_x[k] = b[k] + 2.0 * (x - a_i) * a[j_idx]
                max_b = b_x.max()
                return 2.0 * n * max_b / (S_new * S_new)

            # Optimize over x in [0, upper_bound]
            upper = max(10.0 * a_i, S_fix * 0.1) if a_i > 0 else S_fix * 0.1
            upper = max(upper, 1e-6)

            try:
                res = minimize_scalar(j_1d, bounds=(0.0, upper), method='bounded',
                                      options={'maxiter': 30})
                if res.success and res.fun < best_J - 1e-14:
                    a[i] = res.x
                    best_J = res.fun
                    improved = True
            except Exception:
                pass

        if verbose and sweep % 5 == 0:
            print(f"    CD sweep {sweep}: J={best_J:.8f}", flush=True)

    return a, best_J


def coordinate_descent_fast(a, n_sweeps, deadline):
    """
    Faster coordinate descent using vectorized 1D evaluation.
    """
    a = a.astype(np.float64)
    n = len(a)
    best_J = compute_J(a)

    for sweep in range(n_sweeps):
        if time.monotonic() > deadline:
            break
        for i in range(n):
            if time.monotonic() > deadline:
                break
            a_i = a[i]
            S_total = a.sum()
            S_fix = S_total - a_i
            if S_fix < 1e-15:
                continue

            b = fft_conv(a)
            # Precompute: for each k, the slope (2*a_{k-i}) and base
            # b_k(x) = base_k + slope_k * x  (for k != 2i)
            # b_{2i}(x) = base_2i + x^2
            slope = np.zeros(2 * n - 1)
            for j in range(n):
                k = i + j
                if k < 2 * n - 1:
                    slope[k] = 2.0 * a[j]

            base = b - slope * a_i  # b_k - slope_k * a_i = b_k when x=0 (approximately)
            # For k=2i: base should be b_{2i} - a_i^2 (since b_{2i}(x) = b_{2i} - a_i^2 + x^2)
            k_2i = 2 * i
            if k_2i < 2 * n - 1:
                base[k_2i] = b[k_2i] - a_i * a_i

            def j_fast(x):
                S_new = S_fix + x
                if S_new < 1e-15:
                    return 1e10
                b_x = base + slope * x
                b_x[k_2i] = base[k_2i] + x * x  # quadratic term
                max_b = float(b_x.max())
                return 2.0 * n * max_b / (S_new * S_new)

            upper = max(5.0 * a_i + 1e-6, S_fix * 0.05)

            try:
                res = minimize_scalar(j_fast, bounds=(0.0, upper), method='bounded',
                                      options={'maxiter': 25})
                if res.success and res.fun < best_J - 1e-13:
                    a[i] = max(res.x, 0.0)
                    best_J = res.fun
            except Exception:
                pass

    return a, best_J

            self.best = a.copy()
            self.best_score = score
            elapsed = time.monotonic() - self.t_start
            print(f"  * NEW BEST: J={score:.8f} n={a.size} t={elapsed:.1f}s", flush=True)

    def report(self, label=""):
        if self.best is not None:
            print(f"  {label} best: J={self.best_score:.8f} n={self.best.size}", flush=True)


def resample_seq(src, dest_n):
    """Resample 1D sequence to new length via linear interpolation."""
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    result = np.interp(t_new, t_old, src)
    return np.maximum(result, 0.0)


def build_A_sparse(a):
    """Build sparse constraint matrix for LP."""
    n = len(a)
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        jl = max(0, k - n + 1)
        jh = min(n - 1, k)
        for j in range(jl, jh + 1):
            rows.append(k)
            cols.append(j)
            vals.append(a[k - j])
    return csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))


def lp_direction(a, M):
    """Solve LP: max sum(g) s.t. (a*g)_k <= M, g >= 0. Return normalized g."""
    n = len(a)
    c = -np.ones(n)
    A = build_A_sparse(a)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
    if r.success and r.x is not None:
        g = np.maximum(r.x, 0.0)
        s = g.sum()
        if s > 1e-10:
            return g / s
    return None


def lp_refine(a, deadline, tracker, max_iters=200):
    """LP-based mirror descent refinement."""
    a = np.maximum(a.astype(np.float64), 1e-10)
    a = a / a.sum()
    current_J = compute_J(a)
    best_a = a.copy()
    best_J = current_J
    stall = 0

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        b = fft_conv(a)
        M = float(b.max())
        g = lp_direction(a, M)
        if g is None:
            break

        # Line search
        best_t = 0.0
        best_J_ls = current_J
        for t in [0.01, 0.02, 0.05, 0.1, 0.2, 0.5]:
            a_try = (1.0 - t) * a + t * g
            a_try = np.maximum(a_try, 0.0)
            s = a_try.sum()
            if s < 1e-10:
                continue
            a_try = a_try / s
            J_try = compute_J(a_try)
            if J_try < best_J_ls:
                best_J_ls = J_try
                best_t = t

        if best_t > 0:
            a = (1.0 - best_t) * a + best_t * g
            a = np.maximum(a, 0.0)
            a = a / a.sum()
            current_J = best_J_ls
            stall = 0
            if current_J < best_J:
                best_J = current_J
                best_a = a.copy()
                tracker.update(a)
        else:
            stall += 1
            if stall > 30:
                break

    tracker.update(best_a)
    return best_a


def generate_diverse_inits(n, rng, parent=None):
    """Generate diverse initial sequences."""
    inits = []
    t = np.linspace(0, 1, n, endpoint=False)

    if parent is not None and parent.size > 0:
        inits.append(resample_seq(parent, n))
        p = resample_seq(parent, n)
        p = p + rng.normal(0, 0.03, n) * p
        inits.append(np.maximum(p, 0.01))
        p2 = resample_seq(parent, n)
        p2 = p2 + rng.normal(0, 0.1, n) * p2
        inits.append(np.maximum(p2, 0.01))

    inits.append(np.ones(n))
    inits.append(rng.uniform(0.5, 1.0, n))

    # Various structured inits
    for freq in [1, 2, 3, 4, 5]:
        inits.append(1.0 + 0.3 * np.cos(2.0 * np.pi * freq * t))
    inits.append(1.0 + 0.2 * np.cos(2.0 * np.pi * t) + 0.1 * np.cos(4.0 * np.pi * t))
    inits.append(np.exp(-3.0 * (t - 0.5) ** 2) + 0.1)
    inits.append(np.abs(np.sin(np.pi * t)) + 0.1)
    inits.append(1.0 - 0.2 * np.cos(2.0 * np.pi * t))
    inits.append((t + 0.2) ** 0.3 + 0.3 * (1 - t) ** 0.3)
    inits.append(1.0 + 0.2 * np.sin(4.0 * np.pi * t) * np.cos(2.0 * np.pi * t))

    return inits


def propose_candidate(seed=42, budget_s=1500, **kwargs):
    """
    Multi-phase optimization:
    Phase 1: L-BFGS-B fast basin finding (60s)
    Phase 2: Nesterov accelerated smooth optimization (500s)
    Phase 3: Coordinate descent refinement (400s)
    Phase 4: LP polish (200s)
    Phase 5: Multi-length exploration (remaining)
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + float(budget_s) - 15.0
    tracker = ScoreTracker()

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        if p.ndim == 1 and p.size > 0 and np.all(np.isfinite(p)):
            parent = np.maximum(p, 0.0)
            tracker.update(parent)
            print(f"Parent loaded: n={parent.size}, J={tracker.best_score:.8f}", flush=True)
    except (NameError, AttributeError):
        pass

    if parent is None:
        parent = np.ones(1024)
        tracker.update(parent)
        print(f"Default parent: n=1024, J={tracker.best_score:.8f}", flush=True)

    n_base = parent.size if parent.size <= 2048 else 1024

    # === Phase 1: Fast L-BFGS-B ===
    print(f"\n{'='*50} Phase 1: L-BFGS-B {'='*50}", flush=True)
    p1_budget = 60.0
    deadline_p1 = time.monotonic() + p1_budget

    for n_try in [n_base]:
        if time.monotonic() > deadline_p1:
            break
        inits = generate_diverse_inits(n_try, rng, parent)
        for idx, a0 in enumerate(inits[:4]):
            if time.monotonic() > deadline_p1:
                break
            a0 = np.maximum(a0, 0.01)
            a0 = a0 / a0.sum()

            for beta, maxit in [(3.0, 30), (10.0, 30)]:
                if time.monotonic() > deadline_p1:
                    break

                def obj(x, _beta=beta):
                    x2 = np.maximum(x, 0.0)
                    f, _ = F_and_grad(x2, _beta)
                    return f if np.isfinite(f) else 1e10

                def obj_grad(x, _beta=beta):
                    x2 = np.maximum(x, 0.0)
                    _, g = F_and_grad(x2, _beta)
                    g = np.nan_to_num(g)
                    g[x2 < 1e-12] = 0.0
                    return g

                try:
                    res = minimize(obj, a0.copy(), jac=obj_grad, method='L-BFGS-B',
                                   bounds=[(0.0, None)] * n_try,
                                   options={'maxiter': maxit, 'max_fun': maxit * 15,
                                            'gtol': 1e-7, 'ftol': 1e-12})
                    if res.fun < 1e10:
                        a_opt = np.maximum(res.x, 1e-10)
                        if a_opt.sum() > 0.01:
                            tracker.update(a_opt)
                except Exception:
                    pass

    tracker.report("Phase 1 done")

    # === Phase 2: Nesterov accelerated smooth optimization ===
    print(f"\n{'='*50} Phase 2: Nesterov Optimization {'='*50}", flush=True)
    p2_budget = 500.0
    deadline_p2 = time.monotonic() + p2_budget

    a_current = tracker.best.copy()
    n_opt = len(a_current)

    # Beta schedule: start moderate, decrease for precision
    beta_schedule = [
        (3.0, 200),
        (5.0, 200),
        (8.0, 300),
        (15.0, 400),
        (25.0, 500),
        (40.0, 500),
    ]

    a_opt, F_opt = nesterov_optimize(a_current, beta_schedule, deadline_p2)
    tracker.update(a_opt)

    # Continue with higher beta for precision
    if time.monotonic() < deadline_p2:
        beta_schedule2 = [
            (50.0, 400),
            (80.0, 400),
            (150.0, 300),
        ]
        a_opt2, F_opt2 = nesterov_optimize(a_opt, beta_schedule2, deadline_p2)
        tracker.update(a_opt2)

    tracker.report(f"Phase 2 done ({time.monotonic() - (deadline_p2 - p2_budget):.0f}s)")

    # === Phase 3: Coordinate descent refinement ===
    print(f"\n{'='*50} Phase 3: Coordinate Descent {'='*50}", flush=True)
    p3_budget = 400.0
    deadline_p3 = min(deadline, time.monotonic() + p3_budget)

    a_cd = tracker.best.copy()
    n_cd = len(a_cd)

    # Estimate how many sweeps we can do
    # Each sweep is O(n^2) but vectorized; estimate ~50ms per coordinate for n=1024
    # So ~50s per sweep for n=1024, ~8 sweeps in 400s
    start_t = time.monotonic()
    n_sweeps_est = int(p3_budget / max(1.0, n_cd * 0.05))
    n_sweeps_est = max(1, min(n_sweeps_est, 20))
    print(f"  Estimated {n_sweeps_est} sweeps (n={n_cd})", flush=True)

    a_cd, J_cd = coordinate_descent_fast(a_cd, n_sweeps_est, deadline_p3)
    tracker.update(a_cd)

    # Second round of CD from best
    if time.monotonic() < deadline_p3:
        a_cd2 = tracker.best.copy()
        n_sweeps2 = max(1, int((deadline_p3 - time.monotonic()) / max(1.0, len(a_cd2) * 0.05)))
        a_cd2, J_cd2 = coordinate_descent_fast(a_cd2, n_sweeps2, deadline_p3)
        tracker.update(a_cd2)

    tracker.report(f"Phase 3 done ({time.monotonic() - start_t:.0f}s)")

    # === Phase 4: LP polish ===
    print(f"\n{'='*50} Phase 4: LP Polish {'='*50}", flush=True)
    p4_budget = 200.0
    deadline_p4 = min(deadline, time.monotonic() + p4_budget)

    if tracker.best is not None and tracker.best.size <= 1024:
        a_lp = tracker.best.copy()
        a_lp = lp_refine(a_lp, deadline_p4, tracker, max_iters=300)
    else:
        # For larger sequences, do fewer LP iters
        a_lp = tracker.best.copy()
        a_lp = lp_refine(a_lp, min(deadline_p4, time.monotonic() + 100.0), tracker, max_iters=100)

    # CD after LP
    if time.monotonic() < deadline_p4 and tracker.best is not None:
        a_after = tracker.best.copy()
        n_rem = int((deadline_p4 - time.monotonic()) / max(1.0, len(a_after) * 0.05))
        n_rem = max(1, min(n_rem, 10))
        a_after, _ = coordinate_descent_fast(a_after, n_rem, deadline_p4)
        tracker.update(a_after)

    tracker.report(f"Phase 4 done")

    # === Phase 5: Multi-length and restart ===
    print(f"\n{'='*50} Phase 5: Multi-length & Restarts {'='*50}", flush=True)
    remaining = deadline - time.monotonic()
    if remaining > 20 and tracker.best is not None:
        n_best = tracker.best.size

        # Try nearby lengths
        length_candidates = []
        for delta in [-128, -64, -32, 32, 64, 128, 256]:
            n_try = n_best + delta
            if 128 <= n_try <= 2048 and n_try != n_best:
                length_candidates.append(n_try)

        # Also try standard lengths
        for n_std in [512, 768, 1024, 1536]:
            if n_std not in length_candidates and n_std != n_best:
                length_candidates.append(n_std)

        for n_try in length_candidates[:4]:
            if time.monotonic() > deadline:
                break
            a_res = resample_seq(tracker.best, n_try)
            a_res = np.maximum(a_res, 0.01)
            a_res = a_res / a_res.sum()

            # Quick Nesterov
            beta_sched = [(10.0, 200), (30.0, 300)]
            a_opt, _ = nesterov_optimize(a_res, beta_sched, min(deadline, time.monotonic() + 60.0))
            tracker.update(a_opt)

            # CD
            if time.monotonic() < deadline:
                n_sweeps = max(1, int((deadline - time.monotonic()) / max(1.0, n_try * 0.05)))
                n_sweeps = min(n_sweeps, 5)
                a_cd, _ = coordinate_descent_fast(a_opt, n_sweeps, deadline)
                tracker.update(a_cd)

        # Perturbation restarts
        if time.monotonic() < deadline:
            for pert_std in [0.02, 0.05, 0.1]:
                if time.monotonic() > deadline:
                    break
                a_pert = tracker.best.copy() * (1.0 + rng.normal(0, pert_std, tracker.best.size))
                a_pert = np.maximum(a_pert, 1e-4)
                a_pert = a_pert / a_pert.sum()

                budget_sub = min(60.0, deadline - time.monotonic())
                if budget_sub < 10:
                    break
                beta_sched = [(20.0, 300)]
                a_opt, _ = nesterov_optimize(a_pert, beta_sched, time.monotonic() + budget_sub)
                tracker.update(a_opt)

                # Quick CD
                if time.monotonic() < deadline:
                    a_cd, _ = coordinate_descent_fast(a_opt, 3, deadline)
                    tracker.update(a_cd)

    tracker.report("Phase 5 done")

    # Final check
    if tracker.best is None:
        tracker.best = np.ones(512)
        tracker.best_score = compute_J(tracker.best)
        print(f"  Fallback: J={tracker.best_score:.8f}", flush=True)

    print(f"\n{'='*50} FINAL {'='*50}", flush=True)
    print(f"J={tracker.best_score:.8f} n={tracker.best.size} "
          f"time={time.monotonic()-tracker.t_start:.1f}s", flush=True)

    result = np.clip(tracker.best, 0.0, 1000.0)
    if result.sum() < 0.01:
        result = np.ones(128)

    return [float(x) for x in result]
