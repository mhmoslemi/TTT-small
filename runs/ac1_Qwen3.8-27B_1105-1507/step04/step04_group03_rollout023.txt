    """Gerchberg-Saxton-like step to flatten the spectrum."""
    n = len(a)

def fast_conv(a):
    """Full self-convolution via FFT."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]


def compute_J(a):
    """Compute J(a) = 2*n*max(conv(a,a))/(sum(a))^2."""
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))


def build_A_sparse(n, a):
    """Build sparse matrix A where A[k,j] = a[k-j] for valid (k,j)."""
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        j_min = max(0, k - n + 1)
        j_max = min(n - 1, k)
        for j in range(j_min, j_max + 1):
            rows.append(k)
            cols.append(j)
            vals.append(a[k - j])
    return csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))

    except Exception:

    return None, None

def fourier_to_seq_v2(params, n, n_harm):
    """Convert Fourier parameters to non-negative sequence."""

    for k in range(1, n_harm + 1):
        a += params[2*k-1] * np.cos(2.0 * np.pi * k * t)
        a += params[2*k] * np.sin(2.0 * np.pi * k * t)
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-8:
        a = np.ones(n) / n
    else:
        a = a / s
    return a

def resample_seq(src, dest_n):
    """Resample 1D sequence to new length."""
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size, endpoint=False)
    t_new = np.linspace(0, 1, dest_n, endpoint=False)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)
