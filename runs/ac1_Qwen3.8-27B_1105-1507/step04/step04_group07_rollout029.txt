Minimizes J(a) = 2*n*max(convolve(a,a))/(sum(a))^2 using:

2. LSE smoothed gradient descent with Nesterov momentum
3. Spectral initialization and multi-length search

import numpy as np
from scipy.optimize import linprog
from scipy.sparse import csr_matrix
import time

def fast_conv(a):
    n = len(a)
    s = 1
    while s < 2 * n - 1:
        s <<= 1
    fa = np.fft.rfft(a, s)
    return np.fft.irfft(fa * fa, s)[:2 * n - 1]

def compute_J(a):
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))

def solve_slp_lp(a, b):
    """Solve the SLP LP: minimize t s.t. b[k] + 2*cross(a,da)[k] <= t, sum(da)=0, da >= -a"""
    n = len(a)
    # Build constraint matrix for the peak constraints
    rows = []
    cols = []
    vals = []
    
    for k in range(2 * n - 1):
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        for j in range(js, je + 1):
            rows.append(k)
            cols.append(j)
            vals.append(2.0 * a[k - j])
        # Column for t (index n)
        rows.append(k)
        cols.append(n)
        vals.append(-1.0)
    
    A_data = np.array(vals)
    A_rows = np.array(rows)
    A_cols = np.array(cols)
    A = csr_matrix((A_data, (A_rows, A_cols)), shape=(2 * n - 1, n + 1))
    b_ub = -np.asarray(b, dtype=np.float64)
    
    # Add non-negativity constraints: -da[j] <= a[j]
    # These are: for each j, -1*da[j] <= a[j]
    # So we add n more rows
    extra_rows = list(range(2 * n - 1, 2 * n - 1 + n))
    extra_cols = list(range(n))
    extra_vals = [-1.0] * n
    
    A_full_data = np.concatenate([A_data, extra_vals])
    A_full_rows = np.concatenate([A_rows, np.array(extra_rows)])
    A_full_cols = np.concatenate([A_cols, np.array(extra_cols)])
    A_full = csr_matrix((A_full_data, (A_full_rows, A_full_cols)), shape=(3 * n - 1, n + 1))
    b_full_ub = np.concatenate([b_ub, a.astype(np.float64)])
    
    # Objective: minimize t (index n in the variable vector)
    c = np.zeros(n + 1)
    c[n] = 1.0
    
    # Equality: sum(da) = 0
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.zeros(1)
    
    # Bounds
    bounds = [(-a[i], None) for i in range(n)] + [(None, None)]
    
    result = linprog(c, A_ub=A_full, b_ub=b_full_ub, A_eq=A_eq, b_eq=b_eq,
                     bounds=bounds, method='highs',
                     options={'presolve': True, 'time_limit': 5})
    
    if result.success and result.x is not None:
        return result.x[:n], float(result.x[n])
    return None, None

            cols.append(j)

            if M_trial < best_M - 1e-13:
                best_M = M_trial
                best_alpha = alpha
        
        if best_alpha == 0.0:
            no_improve_count += 1
            if no_improve_count > 10:
                break
            # Try a smaller random perturbation
            rng_local = np.random.default_rng(it * 7919)
            a = a + rng_local.normal(0, 0.01, n)
            a = np.maximum(a, 1e-8)
            a /= a.sum()
            continue
        
        a = a + best_alpha * delta_a
        a = np.maximum(a, 0.0)
        a /= a.sum()
        current_J = compute_J(a)
        no_improve_count = 0
        
        if it % 5 == 0:
            tracker.update(a)
    
    tracker.update(a)
    return a

def lse_gradient_descent(a, max_iters, deadline, tracker, tau_init=30.0):
    """LSE smoothed gradient descent with Nesterov momentum."""
    a = np.maximum(a.astype(np.float64), 1e-8)
    a /= a.sum()
    n = len(a)
    current_J = compute_J(a)
    tracker.update(a)
    
    tau = tau_init
    lr = 1.0 / (n * tau * 0.5)
    momentum = 0.75
    v = np.zeros(n)
    
    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        
        b = fast_conv(a)
        b_max = float(b.max())
        scaled = tau * (b - b_max)
        # Numerical stability
        scaled = np.clip(scaled, -500, 500)
        exp_b = np.exp(scaled)
        p = exp_b / exp_b.sum()
        
        # Gradient of LSE: cross-correlation of p and a
        s_fft = 1
        while s_fft < 3 * n:
            s_fft <<= 1
        fp = np.fft.rfft(p, s_fft)
        fa = np.fft.rfft(a, s_fft)
        # cross(p, a)[j] = sum_k p[k]*a[k-j]
        corr = np.fft.irfft(np.conj(fp) * fa, s_fft)
        grad_lse = 2.0 * corr[:n]
        
        # Full gradient (with sum(a)=1 normalization): 2n*(grad_lse - 2*LSE*a)
        # For simplicity, just use grad_lse direction (the -2*LSE*a term is a shrinkage)
        lse_val = float(b_max + np.log(exp_b.sum()) / tau)
        grad = grad_lse - 2.0 * lse_val * a
        
        # Nesterov momentum
        v = momentum * v + (1.0 - momentum) * grad
        a_new = a - lr * v
        a_new = np.maximum(a_new, 1e-8)
        a_new /= a_new.sum()
        
        J_new = compute_J(a_new)
        if J_new < current_J - 1e-15:
            a = a_new
            current_J = J_new
            tau = min(200.0, tau * 1.005)
            lr = 1.0 / (n * tau * 0.5)
            if it % 20 == 0:
                tracker.update(a)
        else:
            lr *= 0.6
            v *= 0.3
            if lr < 1e-18:
                break
    
    tracker.update(a)
    return a

def extragradient_descent(a, max_iters, deadline, tracker):
    """Extragradient method for min max_k (a*a)[k] s.t. a in simplex."""
    a = np.maximum(a.astype(np.float64), 1e-8)
    a /= a.sum()
    n = len(a)
    current_J = compute_J(a)
    tracker.update(a)
    
    lr = 2.0 / (n + 1)
    rng = np.random.default_rng(123)
    
    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        
        b = fast_conv(a)
        M = float(b.max())
        active = np.where(b >= M - 1e-8 * M)[0]
        
        # Subgradient: average of gradients over active set
        grad = np.zeros(n)
        for k in active:
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            for j in range(js, je + 1):
                grad[j] += 2.0 * a[k - j]
        grad /= len(active)
        
        # Projected gradient step (onto simplex)
        y = a - lr * grad
        # Project onto simplex {x >= 0, sum(x) = 1}
        y = project_simplex(y)
        
        # Second subgradient at y
        b_y = fast_conv(y)
        M_y = float(b_y.max())
        active_y = np.where(b_y >= M_y - 1e-8 * M_y)[0]
        grad_y = np.zeros(n)
        for k in active_y:
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            for j in range(js, je + 1):
                grad_y[j] += 2.0 * y[k - j]
        grad_y /= len(active_y)
        
        a_new = a - lr * grad_y
        a_new = project_simplex(a_new)
        
        J_new = compute_J(a_new)
        if J_new < current_J - 1e-15:
            a = a_new
            current_J = J_new
            lr = min(lr * 1.02, 10.0 / n)
        else:
            lr *= 0.7
        if it % 30 == 0:
            tracker.update(a)
    
    tracker.update(a)
    return a

def project_simplex(v):
    """Project v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - 1))[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0)

def flat_spectrum_init(n, rng):
    """Generate sequence with approximately flat power spectrum."""
    half = n // 2 + 1
    angles = rng.uniform(0, 2 * np.pi, half)
    A = np.zeros(n, dtype=complex)
    A[:half] = np.exp(1j * angles[:half])
    for k in range(1, n - n // 2):
        A[k] = np.conj(A[n - k])
    a = np.fft.ifft(A).real
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-10:
        return np.ones(n) / n
    return a / s

def spectral_weight_init(n, rng, bandwidth=0.3):
    """Initialize with controlled spectral bandwidth."""
    half = n // 2 + 1
    freqs = np.arange(half) / n
    # Gaussian spectral weight
    weights = np.exp(-(freqs / bandwidth) ** 2)
    # Add random phase
    phases = rng.uniform(0, 2 * np.pi, half)
    A_fwd = weights * np.exp(1j * phases)
    A = np.zeros(n, dtype=complex)
    A[:half] = A_fwd
    for k in range(1, n - n // 2):
        A[k] = np.conj(A[n - k])
    a = np.fft.ifft(A).real
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-10:
        return np.ones(n) / n
    return a / s

def generate_inits(n, rng, parent):
    """Generate diverse initializations."""
    inits = []
    if parent is not None and parent.size > 0:
        # Resample parent
        if parent.size != n:
            t_old = np.linspace(0, 1, parent.size)
            t_new = np.linspace(0, 1, n)
            p_resampled = np.maximum(np.interp(t_new, t_old, parent), 0.0)
        else:
            p_resampled = parent.copy()
        p_resampled = p_resampled / p_resampled.sum()
        inits.append(p_resampled.copy())
        inits.append(p_resampled[::-1].copy())
    
    inits.append(np.ones(n) / n)
    inits.append(rng.uniform(0.1, 1.0, n) / (n * 0.55))
    inits.append(flat_spectrum_init(n, rng))
    inits.append(flat_spectrum_init(n, rng))
    inits.append(spectral_weight_init(n, rng, 0.2))
    inits.append(spectral_weight_init(n, rng, 0.4))
    inits.append(spectral_weight_init(n, rng, 0.6))
    
    t = np.linspace(0, 1, n, endpoint=False)
    # Multi-frequency cosines
    for freq_set in [[2, 5, 9], [3, 7, 13], [1, 4, 8, 12]]:
        sig = np.ones(n)
        for f in freq_set:
            sig += 0.3 * np.cos(2 * np.pi * f * t + rng.uniform(0, 6.28))
        inits.append(np.maximum(sig, 0.05))
    
    # Smooth bump functions
    inits.append(0.5 + 0.5 * np.cos(np.pi * t))
    inits.append(0.3 + 0.7 * np.exp(-4 * (t - 0.5) ** 2))
    inits.append(0.5 * np.exp(-8 * (t - 0.3) ** 2) + 0.5 * np.exp(-8 * (t - 0.7) ** 2) + 0.05)
    
    # Random Fourier with limited bandwidth
    n_coeff = min(n // 2 + 1, 15)
    coeffs = rng.normal(0, 1, n_coeff)
    sig = np.zeros(n)
    for i in range(n_coeff):
        sig += coeffs[i] * np.cos(2 * np.pi * i * t + rng.uniform(0, 6.28))
    inits.append(np.abs(sig) + 0.5)
    
    # Sparse + smooth
    sparse = np.zeros(n)
    n_spikes = max(8, n // 10)
    positions = rng.choice(n, size=n_spikes, replace=False)
    sparse[positions] = 1.0
    kernel_size = max(5, n // 20)
    kernel = np.ones(kernel_size) / kernel_size
    inits.append(np.convolve(sparse, kernel, mode='same') + 0.05)
    
    for i, x in enumerate(inits):
        x = np.array(x, dtype=np.float64)
        x = np.maximum(x, 0.001)
        x /= x.sum()
        inits[i] = x
    
    return inits

class ScoreTracker:
    def __init__(self):
        self.best = None
        self.best_score = float('inf')
        self.t_start = time.monotonic()
    
    def update(self, a):
        a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
        if a.size == 0 or a.sum() < 0.01:
            return
        score = float(evaluate_sequence(a.tolist()))
        if np.isfinite(score) and score < self.best_score:
            self.best = a.copy()
            self.best_score = score
            elapsed = time.monotonic() - self.t_start
            print(f"  * J={score:.8f} n={a.size} t={elapsed:.1f}s", flush=True)
    
    def report(self, label=""):
        if self.best is not None:
            print(f"  {label} best: J={self.best_score:.8f} n={self.best.size}", flush=True)

def coordinate_descent(a, max_steps, deadline, tracker):
    """Randomized coordinate descent."""
    a = a.copy().astype(np.float64)
    a /= a.sum()
    n = len(a)
    current_J = compute_J(a)
    tracker.update(a)
    rng = np.random.default_rng(42)
    
    for step in range(max_steps):
        if time.monotonic() > deadline:
            break
        j = rng.integers(0, n)
        b = fast_conv(a)
        M = float(b.max())
        # Try to adjust a[j] to reduce the peak
        # The peak at k is affected by a[j] through a[k-j] and a[j-k]
        best_scale = 1.0
        best_local_J = current_J
        for scale in [0.7, 1.3, 0.5, 1.5, 0.85, 1.15]:
            a_trial = a.copy()
            a_trial[j] *= scale
            a_trial = np.maximum(a_trial, 1e-8)
            ss = a_trial.sum()
            if ss < 1e-10:
                continue
            a_trial /= ss
            J_trial = compute_J(a_trial)
            if J_trial < best_local_J - 1e-13:
                best_local_J = J_trial
                best_scale = scale
        if best_scale != 1.0:
            a[j] *= best_scale
            a = np.maximum(a, 1e-8)
            a /= a.sum()
            current_J = best_local_J
        if step % 50 == 0:
            tracker.update(a)
    tracker.update(a)
    return a

def perturb_and_refine(a, deadline, tracker, n_perturbs=20):
    """Perturb the best solution and re-optimize."""
    rng = np.random.default_rng(999)
    n = a.size
    best_a = a.copy()
    best_J = compute_J(a)
    tracker.update(a)
    
    for p in range(n_perturbs):
        if time.monotonic() > deadline:
            break
        # Random perturbation
        scale = 0.05 * (1.0 + 0.5 * p / n_perturbs)
        a_pert = a + rng.normal(0, scale, n)
        a_pert = np.maximum(a_pert, 1e-8)
        a_pert /= a_pert.sum()
        
        # Quick SLP refinement
        a_ref = slp_optimize(a_pert, 30, deadline - 10, tracker)
        J_ref = compute_J(a_ref)
        if J_ref < best_J:
            best_J = J_ref
            best_a = a_ref.copy()
    
    return best_a

def propose_candidate(seed=42, budget_s=1500, **kwargs):
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + float(budget_s) - 30.0
    tracker = ScoreTracker()
    
    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        if p.ndim == 1 and p.size > 0 and np.all(np.isfinite(p)):
            parent = np.maximum(p, 0.0)
            tracker.update(parent)
            print(f"Parent loaded: n={parent.size}, J={tracker.best_score:.8f}", flush=True)
    except (NameError, AttributeError):
        pass
    
    # === Phase 1: Multi-length SLP search ===
    lengths = [512, 640, 768]
    time_per_length = (deadline - time.monotonic() - 400) / len(lengths)
    
    best_overall = None
    best_overall_J = float('inf')
    
    for n in lengths:
        t_phase_start = time.monotonic()
        t_phase_end = min(deadline - 350, t_phase_start + time_per_length)
        if time.monotonic() >= t_phase_end - 30:
            continue
        print(f"\n{'='*50} Phase1 n={n} {'='*50}", flush=True)
        
        inits = generate_inits(n, rng, parent)
        best_for_n = None
        best_J_for_n = float('inf')
        n_inits = len(inits)
        
        init_budget = max(15.0, (t_phase_end - time.monotonic() - 30) / n_inits)
        
        for idx, a0 in enumerate(inits):
            if time.monotonic() > t_phase_end - 20:
                break
            
            # SLP optimization
            n_slp = max(20, int(init_budget * 3))
            a = slp_optimize(a0, n_slp, t_phase_end - 60, tracker)
            
            # LSE polish
            if time.monotonic() < t_phase_end - 40:
                a = lse_gradient_descent(a, max(200, int(init_budget * 20)), t_phase_end - 40, tracker)
            
            J = compute_J(a)
            if J < best_J_for_n:
                best_J_for_n = J
                best_for_n = a.copy()
        
        if best_for_n is not None:
            tracker.update(best_for_n)
            if best_J_for_n < best_overall_J:
                best_overall_J = best_J_for_n
                best_overall = best_for_n.copy()
            print(f"  Phase1 n={n} done: J={best_J_for_n:.8f} t={time.monotonic()-t_phase_start:.1f}s", flush=True)
    
    # === Phase 2: Focused refinement on best solution ===
    if tracker.best is not None and time.monotonic() < deadline - 200:
        t0 = time.monotonic()
        n = tracker.best.size
        print(f"\n{'='*50} Phase2 Focus n={n} {'='*50}", flush=True)
        
        a = tracker.best.copy().astype(np.float64)
        
        # Extended SLP with many iterations
        a = slp_optimize(a, 500, deadline - 250, tracker)
        
        # LSE polish with decreasing temperature
        if time.monotonic() < deadline - 150:
            a = lse_gradient_descent(a, 3000, deadline - 150, tracker, tau_init=50.0)
        
        # Extragradient
        if time.monotonic() < deadline - 100:
            a = extragradient_descent(a, 2000, deadline - 100, tracker)
        
        # More SLP
        if time.monotonic() < deadline - 80:
            a = slp_optimize(a, 200, deadline - 80, tracker)
        
        print(f"  Phase2 done: t={time.monotonic()-t0:.1f}s", flush=True)
    
    # === Phase 3: Perturbative search ===
    if tracker.best is not None and time.monotonic() < deadline - 100:
        t0 = time.monotonic()
        print(f"\n{'='*50} Phase3 Perturb {'='*50}", flush=True)
        a = perturb_and_refine(tracker.best, deadline - 80, tracker, n_perturbs=30)
        # Final LSE
        if time.monotonic() < deadline - 50:
            a = lse_gradient_descent(a, 2000, deadline - 50, tracker, tau_init=100.0)
        print(f"  Phase3 done: t={time.monotonic()-t0:.1f}s", flush=True)
    
    # === Phase 4: Final coordinate descent polish ===
    if tracker.best is not None and time.monotonic() < deadline - 30:
        t0 = time.monotonic()
        print(f"\n{'='*50} Phase4 CoordDescent {'='*50}", flush=True)
        a = coordinate_descent(tracker.best, 5000, deadline - 20, tracker)
        # Final SLP
        if time.monotonic() < deadline - 10:
            a = slp_optimize(a, 100, deadline - 5, tracker)
        # Final LSE
        if time.monotonic() < deadline - 3:
            a = lse_gradient_descent(a, 1000, deadline - 1, tracker, tau_init=150.0)
        print(f"  Phase4 done: t={time.monotonic()-t0:.1f}s", flush=True)
    
    # Fallback
    if tracker.best is None:
        tracker.best = np.ones(256)
        tracker.best_score = compute_J(tracker.best)
        tracker.update(tracker.best)
    
    print(f"\n{'='*50} FINAL {'='*50}", flush=True)
    print(f"J={tracker.best_score:.8f} n={tracker.best.size} time={time.monotonic()-tracker.t_start:.1f}s", flush=True)
    return [float(x) for x in tracker.best]
