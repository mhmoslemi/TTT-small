    Finds g maximizing sum(g) s.t. (a*g)[:,k] <= M*weights[k], g >= 0.
    """
    n = len(a)
    rows, cols, nnz = get_sparsity(n)
    vals = np.empty(nnz)
    for i in range(nnz):
        k = rows[i]
        j = cols[i]
        vals[i] = a[k - j]
    A = csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))
    c = -np.ones(n)
    if weights is None:
        b_ub = np.full(2 * n - 1, M)
    else:
        b_ub = M * weights
    bounds = [(0.0, None)] * n
    r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                options={'presolve': True, 'time_limit': 15})
    if r.success and r.x is not None:
        g = np.maximum(r.x, 0.0)
        if g.sum() > 1e-10:
            return g
    return None

def nesterov_subgradient(a, max_iters, deadline, tracker):
    """Nesterov-accelerated subgradient descent for the exact max."""
    a = np.maximum(a.astype(np.float64), 1e-8).copy()
    a /= a.sum()
    n = len(a)
    current_J = compute_J(a)
    tracker.update(a)
    
    # Nesterov momentum
    y = a.copy()
    beta = 0.9  # momentum parameter
    lr = 2.0 / (n + 1)  # initial step size
    
    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        
        b = fast_conv(y)
        M = float(b.max())
        active = np.where(b >= M - 1e-9)[0]
        
        # Subgradient of max(b) w.r.t. y_j
        grad = np.zeros(n)
        for k in active:
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            for j in range(js, je + 1):
                grad[j] += y[k - j]
        grad /= len(active)
        
        # Full gradient of J = 2n*M/s^2 (s=1 after norm)
        # dJ/dy_j = 2n*grad_j - 4n*M*y_j (with s=1)
        direction = grad - 2.0 * M * y
        
        # Nesterov update
        z = a - lr * direction
        a_new = (1 - beta) * z + beta * y
        a_new = np.maximum(a_new, 1e-8)
        a_new /= a_new.sum()
        
        J_new = compute_J(a_new)
        if J_new < current_J - 1e-15:
            a = a_new
            y = a_new.copy()
            current_J = J_new
            lr = min(lr * 1.02, 5.0 / n)
        else:
            lr *= 0.8
            # Don't update y on rejected step
        
        if it % 100 == 0:
            tracker.update(a)
    
    tracker.update(a)
    return a

def lp_best_response(a, M, time_limit=20):
    """LP: max sum(g) s.t. (a*g)(k) <= M for all k, g >= 0."""
    n = len(a)
    rows, cols, nnz = get_sparsity(n)
    vals = np.empty(nnz)
    for i in range(nnz):
        vals[i] = a[rows[i] - cols[i]]
    A = csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                options={'presolve': True, 'time_limit': time_limit})
    if r.success and r.x is not None:
        g = np.maximum(r.x, 0.0)
        if g.sum() > 1e-10:
            return g
    return None

                J1 = compute_J(trial / ss) if ss > 1e-15 else float('inf')

def nesterov_subgrad(a, max_iters, deadline, tracker):
    """Nesterov-accelerated subgradient descent."""
    a = np.maximum(a.astype(np.float64), 1e-8).copy()
    a /= a.sum()
    n = len(a)
    current_J = compute_J(a)
    tracker.update(a)

    y = a.copy()
    beta = 0.85
    lr = 3.0 / n
    best_lr = lr

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        b = fast_conv(y)
        M = float(b.max())
        active = np.where(b >= M - 1e-9 * max(M, 1e-15))[0]

        # Gradient of J = 2n*M/s^2 (s=1): dJ/dy_j = 2n*grad_j - 4n*M*y_j
        # Simplified direction: grad_j - 2*M*y_j
        grad = np.zeros(n)
        for k in active:
            js_lo = max(0, k - n + 1)
            js_hi = min(n - 1, k + n - 1)
            for j in range(js_lo, js_hi + 1):
                grad[j] += 2.0 * y[abs(k - j)]
        grad /= len(active)

        direction = grad - 2.0 * M * y

        # Nesterov update
        z = a - lr * direction
        a_new = (1.0 - beta) * z + beta * y
        a_new = np.maximum(a_new, 1e-8)
        ss = a_new.sum()
        if ss < 1e-10:
            continue
        a_new /= ss
        J_new = compute_J(a_new)

        if J_new < current_J - 1e-15:
            a = a_new
            y = a_new.copy()
            current_J = J_new
            lr = min(lr * 1.03, 8.0 / n)
            best_lr = lr
        else:
            lr *= 0.75

        if it % 50 == 0:
            tracker.update(a)

    tracker.update(a)
    return a

def lse_momentum_descent(a, max_iters, deadline, tracker, tau_init=30.0):
    """LSE-smoothed gradient descent with momentum."""
    a = np.maximum(a.astype(np.float64), 1e-6).copy()
    a /= a.sum()
    n = len(a)
    current_J = compute_J(a)
    tracker.update(a)

    tau = tau_init
    lr = 1.0 / (n * tau)
    momentum = 0.8
    v = np.zeros(n)

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        b = fast_conv(a)
        b_max = b.max()
        shifted = tau * (b - b_max)
        exp_b = np.exp(shifted)
        p = exp_b / exp_b.sum()

        # Gradient of LSE w.r.t. a_j = sum_k p_k * 2*a[|k-j|] (using subgrad formula)
        # But for the smooth version, it's the cross-correlation:
        # d LSE / d a_j = sum_k p_k * (d b_k / d a_j) = sum_k p_k * (a[k-j] + a[j-k]) [smoothed]
        # For circular case this is 2*corr(p,a)[j]. For linear, approximate with circular.
        s_fft = 1
        while s_fft < 3 * n:
            s_fft <<= 1
        fp = np.fft.rfft(p, s_fft)
        fa = np.fft.rfft(a, s_fft)
        corr = np.fft.irfft(fp * np.conj(fa), s_fft)
        grad = 2.0 * corr[:n]

        # Full gradient of J: 2n*(LSE gradient) - 4n*LSE_val*a (approx)
        lse_val = b_max + np.log(exp_b.sum()) / tau
        direction = grad - 2.0 * lse_val * a

        v = momentum * v + (1.0 - momentum) * direction
        a_new = a - lr * v
        a_new = np.maximum(a_new, 1e-6)
        ss = a_new.sum()
        if ss < 1e-10:
            continue
        a_new /= ss
        J_new = compute_J(a_new)

        if J_new < current_J - 1e-15:
            a = a_new
            current_J = J_new
            tau = min(500.0, tau * 1.02)
            lr = 1.0 / (n * tau)
        else:
            lr *= 0.5
            v *= 0.5
            if lr < 1e-18:
                break

        if it % 100 == 0:
            tracker.update(a)

    tracker.update(a)
    return a

def coordinate_polish(a, max_steps, deadline, tracker, rng):
    """Randomized coordinate descent polishing."""
    a = a.copy().astype(np.float64)
    a /= a.sum()
    n = len(a)
    current_J = compute_J(a)
    tracker.update(a)

    for step in range(max_steps):
        if time.monotonic() > deadline:

def targeted_redistribution(a, n_target, deadline, tracker):
    """Move mass from high-contribution positions to low-contribution ones."""
    a = a.copy().astype(np.float64)
    a /= a.sum()
    n = len(a)
    current_J = compute_J(a)
    tracker.update(a)
    rng = np.random.default_rng(123)

    for it in range(n_target):
        if time.monotonic() > deadline:
            break
        b = fast_conv(a)
        k_star = int(np.argmax(b))

        # Compute contribution weights
        w = np.zeros(n)
        for j in range(n):
            d = abs(k_star - j)
            if d < n:
                w[j] = a[d]

        # Move mass from top contributors to low contributors
        n_move = max(2, n // 10)
        top_idx = np.argsort(w)[-n_move:]
        bot_idx = np.argsort(w)[:n_move]

        delta = 0.02 * a.mean()
        for src in top_idx:
            for dst in bot_idx:
                if src == dst:
                    continue
                a_trial = a.copy()
                a_trial[src] = max(a_trial[src] - delta, 0.0)
                a_trial[dst] += delta
                ss = a_trial.sum()
                if ss < 1e-10:
                    continue
                a_trial /= ss
                J_trial = compute_J(a_trial)
                if J_trial < current_J - 1e-14:
                    a = a_trial
                    current_J = J_trial
                    break

    tracker.update(a)
    return a

def flat_spectrum_init(n, rng):
    """Initialize with flat power spectrum (random phases)."""
    half = n // 2 + 1
    angles = rng.uniform(0, 2 * np.pi, half)
    A = np.zeros(n, dtype=complex)
    A[:half] = np.exp(1j * angles[:half])
    for k in range(1, n - n // 2):
        A[k] = np.conj(A[n - k])
    a = np.fft.ifft(A).real
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-10:
        return np.ones(n) / n
    return a / s

def random_fourier_init(n, rng, K=8):
    """Random Fourier series initialization."""
    t = np.arange(n) / n
    a = np.zeros(n)
    for k in range(K):
        amp = rng.uniform(0.1, 0.5)
        phase = rng.uniform(0, 2 * np.pi)
        freq = rng.choice(range(1, K + 1))
        a += amp * np.cos(2 * np.pi * freq * t + phase)
    a += 1.0  # DC offset to ensure positivity
    a = np.maximum(a, 0.01)
    a /= a.sum()
    return a

def generate_inits(n, rng, parent):
    inits = []
    if parent is not None and parent.size > 0:
        # Resample parent to this length
        t_old = np.linspace(0, 1, parent.size)
        t_new = np.linspace(0, 1, n)
        p = np.maximum(np.interp(t_new, t_old, parent), 0.0)
        p /= p.sum()
        inits.append(p)
        # Reversed
        p_rev = np.maximum(np.interp(t_new, t_old, parent[::-1]), 0.0)
        p_rev /= p_rev.sum()
        inits.append(p_rev)

    inits.append(np.ones(n) / n)
    inits.append(flat_spectrum_init(n, rng))
    inits.append(flat_spectrum_init(n, rng))
    inits.append(random_fourier_init(n, rng, K=6))
    inits.append(random_fourier_init(n, rng, K=10))
    inits.append(random_fourier_init(n, rng, K=15))

    t = np.linspace(0, 1, n, endpoint=False)
    for freq, amp in [(2, 0.4), (3, 0.3), (5, 0.5), (7, 0.35), (11, 0.25)]:
        inits.append(1.0 + amp * np.cos(2 * np.pi * freq * t + rng.uniform(0, 6.28)))
    inits.append(1.0 + 0.3 * np.cos(2 * np.pi * 3.7 * t) + 0.2 * np.cos(2 * np.pi * 7.3 * t))
    inits.append(0.5 + 0.4 * np.cos(np.pi * t) ** 2)

    # Bimodal
    inits.append(0.3 + 0.7 * np.exp(-8 * (t - 0.3) ** 2) + 0.7 * np.exp(-8 * (t - 0.7) ** 2))

    for x in inits:
        x = np.maximum(np.asarray(x, dtype=np.float64), 0.001)
        x /= x.sum()
    return inits

def resample_seq(src, dest_n):
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)

class ScoreTracker:
    def __init__(self):
        self.best = None
        self.best_score = float('inf')
        self.t_start = time.monotonic()

    def update(self, a):
        a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
        if a.size == 0 or a.sum() < 0.01:
            return
        score = float(evaluate_sequence(a.tolist()))
        if np.isfinite(score) and score < self.best_score:
            self.best = a.copy()
            self.best_score = score
            elapsed = time.monotonic() - self.t_start
            print(f"  * J={score:.8f} n={a.size} t={elapsed:.1f}s", flush=True)

    def report(self, label=""):
        if self.best is not None:
            print(f"  {label} best: J={self.best_score:.8f} n={self.best.size}", flush=True)

def propose_candidate(seed=42, budget_s=1500, **kwargs):
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + float(budget_s) - 10.0
    tracker = ScoreTracker()

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        if p.ndim == 1 and p.size > 0 and np.all(np.isfinite(p)):
            parent = np.maximum(p, 0.0)
            tracker.update(parent)
            print(f"Parent: n={parent.size}, J={tracker.best_score:.8f}", flush=True)
    except (NameError, AttributeError):
        pass

    # === Phase 1: Multi-length LP optimization ===
    lengths = [320, 384, 448, 512, 576]
    # Allocate time proportionally, giving more to mid-range lengths
    total_time = deadline - time.monotonic() - 400
    time_alloc = {320: 0.12, 384: 0.18, 448: 0.15, 512: 0.30, 576: 0.25}
    for l in lengths:
        assert l in time_alloc

    for n in lengths:
        t_phase_start = time.monotonic()
        t_phase_end = min(deadline - 300, t_phase_start + total_time * time_alloc[n])
        if time.monotonic() >= t_phase_end - 30:
            continue
        print(f"\n{'='*50} Phase1 n={n} budget={time_alloc[n]*100:.0f}% {'='*50}", flush=True)

        inits = generate_inits(n, rng, parent)
        best_for_n = None
        best_J_for_n = float('inf')
        n_inits = len(inits)
        init_budget = max(8.0, (t_phase_end - time.monotonic()) / n_inits)

        for idx, a0 in enumerate(inits):
            if time.monotonic() > t_phase_end - 20:
                break
            t0_init = time.monotonic()

            # LP phase with progressive tightening
            a = a0.copy()
            for tight in [0.03, 0.02, 0.015, 0.01, 0.008, 0.005]:
                if time.monotonic() > t_phase_end - 30:
                    break
                n_iters = max(3, int(init_budget / (6 * tight / 0.03)))
                a = run_lp_opt(a, n_iters, t_phase_end - 30, tracker,
                              tighten=tight, time_limit=15)

            # Subgradient polish
            if time.monotonic() < t_phase_end - 20:
                a = nesterov_subgrad(a, max(200, 20000 // n), t_phase_end - 20, tracker)

            # LSE polish
            if time.monotonic() < t_phase_end - 10:
                a = lse_momentum_descent(a, max(100, 20000 // n), t_phase_end - 10, tracker, tau_init=50.0)

            J = compute_J(a)
            if J < best_J_for_n:
                best_J_for_n = J
                best_for_n = a.copy()
            print(f"  init {idx}/{n_inits}: J={J:.8f} ({time.monotonic()-t0_init:.1f}s)", flush=True)

        if best_for_n is not None:
            tracker.update(best_for_n)
            if parent is None or best_J_for_n < compute_J(resample_seq(parent, best_for_n.size)):
                parent = best_for_n
            tracker.report(f"Phase1 n={n} ({time.monotonic()-t_phase_start:.1f}s): J={best_J_for_n:.8f}")

    # === Phase 2: Focused optimization on best ===
    if tracker.best is not None and time.monotonic() < deadline - 200:
        t0 = time.monotonic()
        n = tracker.best.size
        print(f"\n{'='*50} Phase2 Focus n={n} {'='*50}", flush=True)

        a = tracker.best.copy().astype(np.float64)

        # Aggressive LP with very tight constraints
        for tight in [0.004, 0.003, 0.002, 0.0015, 0.001, 0.0005]:
            if time.monotonic() > deadline - 150:
                break
            a = run_lp_opt(a, 20, deadline - 150, tracker, tighten=tight, time_limit=25)

        # Nesterov subgradient
        if time.monotonic() < deadline - 100:
            a = nesterov_subgrad(a, 5000, deadline - 100, tracker)

        # LSE polish
        if time.monotonic() < deadline - 50:
            a = lse_momentum_descent(a, 5000, deadline - 50, tracker, tau_init=100.0)

        tracker.report(f"Phase2 ({time.monotonic()-t0:.1f}s)")

    # === Phase 3: Targeted redistribution + coordinate polish ===
    if tracker.best is not None and time.monotonic() < deadline - 50:
        t0 = time.monotonic()
        n = tracker.best.size
        print(f"\n{'='*50} Phase3 Polish n={n} {'='*50}", flush=True)

        a = tracker.best.copy().astype(np.float64)
        a = targeted_redistribution(a, 200, deadline - 30, tracker)
        a = coordinate_polish(a, 3000, deadline - 20, tracker, rng)

        # Final LP pass
        if time.monotonic() < deadline - 10:
            a = run_lp_opt(a, 10, deadline - 5, tracker, tighten=0.003, time_limit=15)
            a = lse_momentum_descent(a, 2000, deadline - 2, tracker, tau_init=200.0)

        tracker.report(f"Phase3 ({time.monotonic()-t0:.1f}s)")

    # === Phase 4: Try slightly different lengths from best ===
    if tracker.best is not None and time.monotonic() < deadline - 30:
        t0 = time.monotonic()
        print(f"\n{'='*50} Phase4 Length variants {'='*50}", flush=True)
        base_n = tracker.best.size
        for delta in [-32, 32, -64, 64]:
            n_try = base_n + delta
            if n_try < 128 or n_try > 1024:
                continue
            if time.monotonic() > deadline - 20:
                break
            a_try = resample_seq(tracker.best, n_try)
            a_try /= a_try.sum()
            a_try = run_lp_opt(a_try, 8, deadline - 15, tracker, tighten=0.01, time_limit=10)
            tracker.report(f"  n={n_try}")

    # Fallback
    if tracker.best is None:
        tracker.best = np.ones(256)
        tracker.best_score = compute_J(tracker.best)
        tracker.update(tracker.best)

    print(f"\n{'='*50} FINAL {'='*50}", flush=True)
    print(f"J={tracker.best_score:.8f} n={tracker.best.size} time={time.monotonic()-tracker.t_start:.1f}s", flush=True)
    return [float(x) for x in tracker.best]
