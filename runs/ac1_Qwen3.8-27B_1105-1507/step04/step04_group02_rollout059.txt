- Cap the step size to avoid divergence

For the "flattening" idea: after subgradient steps, I can try to explicitly reduce the gap between the top convolution values by doing a small adjustment.

Let me also think about the **Nesterov accelerated subgradient** method:
- Maintain a momentum vector v
- Updated sequence: a_new = a - step * (v + grad)
- Updated momentum: v_new = (1+step/L) * v + step * grad (where L is Lipschitz constant)

Actually, for simplicity and robustness, I'll use:
- Standard subgradient with momentum (like Adam but simpler)
- Periodic "annealing" of the step size
- Perturbation escapes when stuck

Let me also consider a completely different approach: **direct optimization of the top few convolution values**.

The idea:
1. Identify the top T convolution values (say T=5)
2. Form the objective: minimize max(conv[k1], conv[k2], ..., conv[kT])
3. This is equivalent to minimizing t subject to conv[k_i] <= t for i=1..T
4. Use scipy.optimize.minimize with this objective
5. Periodically update the active set

This is a smooth (at least C^1) optimization problem when the max is achieved at a unique point, and we can use gradient methods.

Actually, I think the cleanest approach is:

**Algorithm: Smoothed max optimization**

1. Use `smooth_max(conv, t) = t * log(sum(exp(conv/k - max(conv)/k)))` as a smooth approximation of `max(conv)`
2. Optimize this smooth function using gradient descent with momentum
3. Decrease t over time for sharper approximation

The gradient of `smooth_max` is: `grad = sum_k softmax(conv/k)[k] * d(conv[k])/da`

This is exactly the weighted subgradient I was computing above, with alpha playing the role of t!

So my weighted subgradient IS the gradient of the smoothed max.

import numpy as np
from scipy.interpolate import interp1d
import time

def compute_conv(a):
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    conv = np.fft.irfft(fa * fa, size)
    return conv[:2 * n - 1]

def compute_obj(a):
    n = len(a)
    S = float(np.sum(a))
    if S < 1e-10:
        return np.inf
    conv = compute_conv(a)
    M = float(np.max(conv))
    return 2.0 * n * M / (S * S)

def smooth_max_grad(a, alpha):
    """Compute gradient of smooth_max(conv(a,a)) with parameter alpha.
    Returns (grad, M, conv) where M is the true max."""
    n = len(a)
    conv = compute_conv(a)
    M = float(np.max(conv))
    # Weights: softmax of conv/alpha, shifted for stability
    scaled = (conv - M) / alpha
    weights = np.exp(scaled)
    total_w = np.sum(weights)
    weights = weights / total_w
    
    # grad[m] = 2 * sum_j a[j] * weights[m+j] for valid m+j
    grad = np.zeros(n)
    for j in range(n):
        m_end = min(n - 1, 2 * n - 2 - j)
        if m_end >= 0:
            grad[:m_end + 1] += 2.0 * a[j] * weights[:m_end + 1]
    
    return grad, M, conv

def project_simplex(a):
    """Project onto probability simplex: a >= 0, sum(a) = 1."""
    a = np.maximum(a, 0)
    S = np.sum(a)
    if S < 1e-15:
        return np.ones(len(a)) / len(a)
    return a / S

def subgradient_descent(a, max_iters, time_budget, alpha_init=1.0, alpha_decay=0.999,
                        step_init=0.05, momentum=0.9, verbose=False):
    """Nesterov-style subgradient descent for the smooth max objective."""
    n = len(a)
    a = project_simplex(a.copy().astype(np.float64))
    best_a = a.copy()
    best_obj = compute_obj(a)
    
    v = np.zeros(n)
    alpha = alpha_init
    step = step_init
    start = time.monotonic()
    stall = 0
    
    for it in range(max_iters):
        if time.monotonic() - start > time_budget:
            break
        
        grad, M, conv = smooth_max_grad(a, alpha)
        
        # Center gradient to preserve sum (approximately)
        grad_centered = grad - np.sum(grad) / n
        
        # Nesterov momentum
        v = momentum * v + (1 - momentum) * grad_centered
        
        # Step
        a_new = a - step * v
        a_new = project_simplex(a_new)
        
        obj = compute_obj(a_new)
        
        if obj < best_obj - 1e-14:
            best_obj = obj
            best_a = a_new.copy()
            a = a_new
            stall = 0
            # Slightly decrease alpha for sharper approximation
            alpha *= alpha_decay
        else:
            stall += 1
            # Reduce step size
            step *= 0.95
            if stall > 50:
                # Reset momentum and perturb
                v = np.zeros(n)
                step = step_init * 0.5
                alpha = max(alpha, 0.1)
                rng = np.random.default_rng(it * 104729 + 12345)
                a = a * np.exp(rng.normal(0, 0.05, n))
                a = project_simplex(a)
                stall = 0
                if step < 1e-10:
                    break
    
    return best_a, best_obj

def cosine_init(n, n_modes=10, seed=0):
    rng = np.random.default_rng(seed)
    k = np.arange(n)
    a = np.ones(n)
    for m in range(1, n_modes + 1):
        c = rng.uniform(-0.5, 0.5) / (m * 0.5)
        phase = rng.uniform(0, 2 * np.pi)
        a += c * np.cos(2 * np.pi * m * k / n + phase)
    a = np.maximum(a, 1e-10)
    return a / np.sum(a)

def raised_cosine_init(n, alpha=1.0):
    k = np.arange(n)
    x = 2.0 * k / (n - 1) - 1.0
    a = 0.5 * (1.0 + np.cos(np.pi * x)) ** alpha
    a = np.maximum(a, 1e-10)
    return a / np.sum(a)

def gaussian_init(n, sigma_frac=0.3):
    k = np.arange(n)
    center = n / 2.0
    sigma = sigma_frac * n
    a = np.exp(-0.5 * ((k - center) / sigma) ** 2)
    return a / np.sum(a)

def alternating_lp(a, max_iters=20, time_budget=20.0):
    """Alternating LP method from the SOTA description."""
    n = len(a)
    a = project_simplex(a.copy().astype(np.float64))
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    
    from scipy.optimize import linprog
    from scipy.sparse import coo_matrix
    
    for it in range(max_iters):
        if time.monotonic() - start > time_budget - 2:
            break
        
        conv = compute_conv(a)
        M = float(np.max(conv))
        
        # Build Toeplitz matrix for conv(a, g) <= M
        # conv(a,g)[k] = sum_i a[i] * g[k-i]
        # For fixed a, this is linear in g
        rows = []
        cols = []
        vals = []
        for j in range(n):
            # g[j] appears in conv[k] for k = j, j+1, ..., j+n-1
            # with coefficient a[k-j] = a[0], a[1], ..., a[n-1]
            for offset in range(n):
                k = j + offset
                rows.append(k)
                cols.append(j)
                vals.append(a[offset])
        
        A = coo_matrix((vals, (rows, cols)), shape=(2*n-1, n)).tocsr()
        c = -np.ones(n)
        b_ub = np.full(2*n-1, M)
        bounds = [(0.0, None)] * n
        
        try:
            res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                         options={'presolve': True, 'time_limit': 5})
            if res.status == 0:
                g = np.maximum(res.x, 0)
                S_g = np.sum(g)
                if S_g > 1e-10:
                    g = g / S_g
                    # Line search
                    best_t = 0.0
                    best_obj_ls = best_obj
                    for t in [0.1, 0.3, 0.5, 0.7, 1.0]:
                        a_try = project_simplex((1-t)*a + t*g)
                        obj_try = compute_obj(a_try)
                        if obj_try < best_obj_ls:
                            best_obj_ls = obj_try
                            best_t = t
                    if best_t > 0:
                        a = project_simplex((1-best_t)*a + best_t*g)
                        best_obj = best_obj_ls
                        best_a = a.copy()
                        if best_obj < 1.51:
                            break
        except Exception:
            break
    
    return best_a, best_obj

"""
Optimized step-function coefficient search minimizing 2*n*max(conv(a,a))/(sum(a))^2.
Uses: smoothed-max subgradient (exponential weights on ALL peaks), adaptive alpha
scheduling, Nesterov momentum, alternating LP, asymmetric initializations,
multi-n interpolation, and iterative perturbation escapes.
Key improvement: full weighted subgradient via exponential softmax over all
convolution lags (not just top-K), combined with progressive alpha sharpening
and a richer set of asymmetric initializations to avoid local minima.
"""

import numpy as np
from scipy.optimize import linprog
from scipy.interpolate import interp1d
from scipy.sparse import coo_matrix
import time


def compute_conv(a):
    """Compute full linear convolution a*a via FFT."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    conv = np.fft.irfft(fa * fa, size)
    return conv[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    S = float(np.sum(a))
    if S < 1e-10:
        return np.inf
    conv = compute_conv(a)
    M = float(np.max(conv))
    return 2.0 * n * M / (S * S)


def project_simplex(a):
    """Project onto probability simplex: a >= 0, sum(a) = 1."""
    a = np.maximum(a, 0.0)
    S = np.sum(a)
    if S < 1e-15:
        return np.ones(len(a)) / len(a)
    return a / S


def smooth_max_grad(a, alpha):
    """
    Compute gradient of alpha*log(sum(exp(conv/alpha))) w.r.t. a.
    This is a smooth approximation of max(conv).
    Returns (grad, M, conv).
    """
    n = len(a)
    conv = compute_conv(a)
    M = float(np.max(conv))
    # Softmax weights
    scaled = (conv - M) / alpha
    weights = np.exp(scaled)
    total_w = np.sum(weights)
    weights = weights / total_w

    # grad[m] = 2 * sum_j a[j] * weights[m+j] for valid m+j in [0, 2n-2]
    grad = np.zeros(n)
    for j in range(n):
        m_end = min(n - 1, 2 * n - 2 - j)
        if m_end >= 0:
            grad[:m_end + 1] += 2.0 * a[j] * weights[:m_end + 1]

    return grad, M, conv


def subgradient_descent(a, max_iters, time_budget, alpha_init=0.5, alpha_decay=0.999,
                        step_init=0.03, momentum=0.85, verbose=False):
    """
    Nesterov-style smoothed-max subgradient descent on the simplex.
    Alpha decreases over time for progressive sharpening.
    """
    n = len(a)
    a = project_simplex(a.copy().astype(np.float64))
    best_a = a.copy()
    best_obj = compute_obj(a)

    v = np.zeros(n)
    alpha = alpha_init
    step = step_init
    start = time.monotonic()
    stall = 0
    consecutive_no_improve = 0

    for it in range(max_iters):
        if time.monotonic() - start > time_budget:
            break

        grad, M, conv = smooth_max_grad(a, alpha)

        # Center gradient to approximately preserve the sum constraint
        grad_centered = grad - np.sum(grad) / n

        # Nesterov momentum
        v = momentum * v + (1.0 - momentum) * grad_centered

        # Take step
        a_new = a - step * v
        a_new = project_simplex(a_new)

        obj = compute_obj(a_new)

        if obj < best_obj - 1e-14:
            best_obj = obj
            best_a = a_new.copy()
            a = a_new
            stall = 0
            consecutive_no_improve = 0
            alpha *= alpha_decay
            # Slowly anneal step size
            step *= (1.0 - 0.0005)
        else:
            stall += 1
            consecutive_no_improve += 1
            step *= 0.97

            if stall > 80:
                # Escape: reset momentum, increase alpha, perturb
                v = np.zeros(n)
                step = step_init * 0.6
                alpha = max(alpha * 2.0, 0.3)
                rng = np.random.default_rng(it * 104729 + 777)
                a = a * np.exp(rng.normal(0, 0.06, n))
                a = project_simplex(a)
                stall = 0
                consecutive_no_improve = 0

            if step < 1e-12:
                break

            if consecutive_no_improve > 500:
                break

    return best_a, best_obj


def cosine_init(n, n_modes=12, seed=0):
    """Smooth cosine-series initialization."""
    rng = np.random.default_rng(seed)
    k = np.arange(n)
    a = np.ones(n) * 0.5
    for m in range(1, n_modes + 1):
        c = rng.uniform(-0.4, 0.4) / (m * 0.3)
        phase = rng.uniform(0, 2 * np.pi)
        a += c * np.cos(2 * np.pi * m * k / n + phase)
    a = np.maximum(a, 1e-10)
    return a / np.sum(a)


def raised_cosine_init(n, alpha=1.0, asymmetry=0.0):
    """Raised cosine with optional asymmetry (shift peak off-center)."""
    k = np.arange(n)
    center = n / 2.0 + asymmetry * n
    x = 2.0 * (k - center) / (n - 1) - 1.0
    x = np.clip(x, -1.5, 1.5)
    a = 0.5 * (1.0 + np.cos(np.pi * x)) ** alpha
    a = np.maximum(a, 1e-10)
    return a / np.sum(a)


def gaussian_init(n, sigma_frac=0.3, asymmetry=0.0):
    """Gaussian bump with optional asymmetry."""
    k = np.arange(n)
    center = n / 2.0 + asymmetry * n
    sigma = sigma_frac * n
    a = np.exp(-0.5 * ((k - center) / sigma) ** 2)
    a = np.maximum(a, 1e-10)
    return a / np.sum(a)


def asymmetric_power_init(n, p=1.5, side=1.0):
    """Power-law asymmetric initialization.
    side > 0: more weight on the right; side < 0: more on the left."""
    k = np.arange(n)
    if side >= 0:
        a = (k + 1.0) ** p
    else:
        a = (n - k) ** p
    a = np.maximum(a, 1e-10)
    return a / np.sum(a)


def build_toeplitz(a):
    """Build sparse matrix A where A[k,j] = a[k-j] if 0<=k-j<n else 0."""
    n = len(a)
    rows = []
    cols = []
    vals = []
    for j in range(n):
        for offset in range(n):
            k = j + offset
            rows.append(k)
            cols.append(j)
            vals.append(a[offset])
    A = coo_matrix((vals, (rows, cols)), shape=(2 * n - 1, n)).tocsr()
    return A


def solve_lp_maxsum(a, M, time_limit=5):
    """Solve LP: max sum(g) s.t. conv(a,g) <= M, g >= 0."""
    n = len(a)
    A = build_toeplitz(a)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': time_limit})
        if res.status == 0:
            return np.maximum(res.x, 0)
    except Exception:
        pass
    return None


def alternating_lp(a, max_iters=25, time_budget=25.0):
    """Alternating LP: fix a, optimize g via LP, interpolate, repeat."""
    n = len(a)
    a = project_simplex(a.copy().astype(np.float64))
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    stall = 0

    for it in range(max_iters):
        if time.monotonic() - start > time_budget - 2:
            break

        conv = compute_conv(a)
        M = float(np.max(conv))

        g = solve_lp_maxsum(a, M, time_limit=4)
        if g is None:
            break

        S_g = np.sum(g)
        if S_g < 1e-10:
            break
        g = g / S_g

        # Geometric line search
        best_t = 0.0
        best_obj_ls = best_obj
        for t in [0.05, 0.1, 0.2, 0.3, 0.5, 0.7, 1.0]:
            a_try = project_simplex((1.0 - t) * a + t * g)
            obj_try = compute_obj(a_try)
            if obj_try < best_obj_ls:
                best_obj_ls = obj_try
                best_t = t

        if best_t > 0.0:
            a = project_simplex((1.0 - best_t) * a + best_t * g)
            best_obj = best_obj_ls
            best_a = a.copy()
            stall = 0
        else:
            stall += 1
            if stall > 10:
                rng = np.random.default_rng(it * 997 + 31)
                a = a * np.exp(rng.normal(0, 0.07, n))
                a = project_simplex(a)
                stall = 0
            else:
                # Small blend toward g anyway
                a = project_simplex(0.95 * a + 0.05 * g)

    return best_a, best_obj

                a_p = a_cur * np.exp(rng.normal(0, 0.02, n_final))
                a_p = project_simplex(a_p)
                a_p, obj_p = subgradient_descent(
                    a_p, max_iters=800, time_budget=min(remaining * 0.1, 8.0),
                    alpha_init=0.03, alpha_decay=0.9998, step_init=0.003)
                if obj_p < best_score:
                    best_score = obj_p
                    best_seq = a_p.copy()
                    a_cur = a_p.copy()
                    print(f"  Perturb: {best_score:.6f} ** NEW BEST **")

    # Final validation
    if best_seq is None:
        best_seq = np.ones(300) / 300
        best_score = compute_obj(best_seq)

    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    val = evaluate_sequence(result)
    if val < np.inf and val < best_score:
        best_score = val
    elif val == np.inf:
        result = [1.0] * 300
        best_score = 2.0

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_score:.6f}, time={elapsed:.1f}s ===")

    return result
