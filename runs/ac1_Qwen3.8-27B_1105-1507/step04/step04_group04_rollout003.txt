def compute_conv(a):
    """Compute linear convolution a*a using FFT."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    conv = np.fft.irfft(fa * fa, size)
    return conv[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 1e-10:
        return np.inf
    conv = compute_conv(a)
    max_b = float(np.max(conv))
    return 2.0 * n * max_b / (s * s)


def compute_grad_conv(a, w):
    """
    Compute gradient of sum_k w_k * (a*a)[k] with respect to a.
    Uses the identity: d/da_i [(a*a)[k]] = a[k-i] (if valid).
    So grad_i = sum_k w_k * a[k-i] = (w * a)[i] in circular sense, but we need linear.
    
    Actually: grad_i = sum_{k} w_k * a[k-i] where 0 <= k-i <= n-1 and 0 <= k <= 2n-2.
    Let j = k-i, so k = i+j, j in [0, n-1], k in [i, i+n-1] intersected with [0, 2n-2].
    grad_i = sum_{j=0}^{n-1} w[i+j] * a[j] (with w padded to length 2n-1, and out-of-bounds = 0)
    
    This is a cross-correlation: grad = correlate(w, a, 'full')[n-1 : n-1+n]
    """
    # w has length 2n-1, a has length n
    # grad_i = sum_j w[i+j] * a[j] for j=0..n-1 (where i+j < 2n-1)
    # This is the cross-correlation of w and a
    c = np.correlate(w, a, mode='full')
    # c[m] = sum_k w[k] * a[k - (m - (len(w)-1))] ... let me be more careful
    # np.correlate(w, a, 'full')[m] = sum_k w[k] * a[k + m - (n-1)]
    # We want: grad[i] = sum_j w[i+j] * a[j]
    # Let's use a different approach: it's the "sliding dot product"
    # grad[i] = dot(w[i:i+n], a) for i = 0, ..., n-1 (with w having length 2n-1)
    # This is a valid sliding dot product since i+n <= 2n-1 when i <= n-1.
    
    # Actually let me use FFT for this:
    # grad = ifft(fft(w, 2n) * conj(fft(a, 2n))) ... not quite.
    # 
    # Let me think again. We want grad[i] = sum_{j=0}^{n-1} w[i+j] * a[j].
    # This is the cross-correlation of w (length 2n-1) and a (length n) at specific lags.
    # 
    # Using np.correlate: np.correlate(w, a, 'full') has length (2n-1) + n - 1 = 3n-2.
    # The relevant part... let me just use the direct computation for clarity, 
    # or better, use FFT-based convolution.
    
    # Alternative: grad[i] = sum_j a[j] * w[i+j]
    # = (a reversed) convolved with w, evaluated at specific points
    # 
    # Let ar = a[::-1] (reversed)
    # conv_ar_w = convolve(ar, w) has length n + 2n-1 - 1 = 3n-2
    # conv_ar_w[k] = sum_j ar[j] * w[k-j] = sum_j a[n-1-j] * w[k-j]
    # Let i' = n-1-j, so j = n-1-i':
    # = sum_{i'} a[i'] * w[k-(n-1-i')] = sum_{i'} a[i'] * w[k-n+1+i']
    # We want: sum_j a[j] * w[i+j] = sum_j a[j] * w[(i) + j]
    # Matching: k-n+1 = i, so k = i + n - 1
    # So grad[i] = conv_ar_w[i + n - 1] for i = 0, ..., n-1
    # And conv_ar_w has length 3n-2, indices 0 to 3n-3.
    # i + n - 1 ranges from n-1 to 2n-2. This is within [0, 3n-3]. ✓
    
    ar = a[::-1]
    size = 1
    while size < len(ar) + len(w) - 1:
        size <<= 1
    fa = np.fft.rfft(ar, size)
    fw = np.fft.rfft(w, size)
    conv = np.fft.irfft(fa * fw, size)
    grad = conv[n - 1: n - 1 + n]
    return grad


def mirror_descent_step(a, tau, eta):
    """One mirror descent step on the smoothed objective."""
    n = len(a)
    conv = compute_conv(a)
    max_conv = np.max(conv)
    
    # Compute softmax weights
    shifted = tau * (conv - max_conv)
    shifted = np.clip(shifted, -500, 500)
    w = np.exp(shifted)
    w_sum = np.sum(w)
    if w_sum < 1e-300:
        idx = int(np.argmax(conv))
        w = np.zeros(len(conv))
        w[idx] = 1.0
    else:
        w /= w_sum
    
    # Compute gradient
    grad = compute_grad_conv(a, w)
    
    # Multiplicative update
    log_update = np.clip(-eta * grad, -50, 50)
    a_new = a * np.exp(log_update)
    a_new = np.maximum(a_new, 1e-15)
    s = np.sum(a_new)
    if s > 1e-10:
        a_new /= s
    return a_new, w


def build_toeplitz(a):
    """Build matrix A where (A@g)[k] = (a*g)[k] for k=0..2n-2."""
    n = len(a)
    A = np.zeros((2 * n - 1, n), dtype=np.float64)
    for j in range(n):
        k_start = j
        k_end = min(j + n, 2 * n - 1)
        A[k_start:k_end, j] = a[:k_end - k_start]
    return A


def solve_lp_minpeak(a, time_limit=5):
    """Solve LP: min t s.t. (a*g)[k] <= t, sum(g)=1, g>=0."""
    n = len(a)
    
    # Try cvxpy first
    if HAS_CVXPY:
        try:
            g_var = cp.Variable(n, nonneg=True)
            t_var = cp.Variable()
            A = build_toeplitz(a)
            constraints = [A @ g_var <= t_var, cp.sum(g_var) == 1]
            prob = cp.Problem(cp.Minimize(t_var), constraints)
            for solver in [cp.CLARABEL, cp.SCS, cp.PDLP]:
                try:
                    prob.solve(solver=solver, time_limit=int(time_limit))
                    if g_var.value is not None and t_var.value is not None:
                        g_val = np.maximum(g_var.value, 0)
                        s = np.sum(g_val)
                        if s < 1e-10:
                            continue
                        return g_val / s, float(t_var.value)
                except Exception:
                    continue
        except Exception:
            pass
    
    # Fallback to scipy linprog
    A = build_toeplitz(a)
    c = np.zeros(n + 1)
    c[-1] = 1.0
    A_ub = np.zeros((2 * n - 1, n + 1))
    A_ub[:, :n] = A
    A_ub[:, n] = -1.0
    b_ub = np.zeros(2 * n - 1)
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])
    bounds = [(0.0, None)] * n + [(None, None)]
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': time_limit})
        if res.status == 0:
            g = np.maximum(res.x[:n], 0)
            s = np.sum(g)
            if s < 1e-10:
                return None, None
            return g / s, float(res.x[n])
    except Exception:
        pass
    return None, None


def alternating_lp(a, max_iters=100, time_budget=100.0, verbose=False):
    """Alternating LP: minimize peak of a*g over g, then set a=g."""
    n = len(a)
    a = a.copy().astype(np.float64)
    s = np.sum(a)
    if s < 1e-10:
        a = np.ones(n) / n
    a = a / s
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    stall = 0
    
    for it in range(max_iters):
        if time.monotonic() - start > time_budget - 3:
            break
        
        g, M_opt = solve_lp_minpeak(a, time_limit=5)
        if g is None:
            break
        
        # Line search for mixing parameter
        best_t = 0.0
        best_obj_ls = best_obj
        for t in [1.0, 0.8, 0.6, 0.4, 0.2]:
            a_candidate = (1.0 - t) * a + t * g
            a_candidate = np.maximum(a_candidate, 0)
            s_c = np.sum(a_candidate)
            if s_c < 1e-10:
                continue
            a_candidate = a_candidate / s_c
            obj_c = compute_obj(a_candidate)
            if obj_c < best_obj_ls:
                best_obj_ls = obj_c
                best_t = t
        
        if best_t > 0 and best_obj_ls < best_obj - 1e-12:
            a_new = (1.0 - best_t) * a + best_t * g
            a_new = np.maximum(a_new, 0)
            s_new = np.sum(a_new)
            if s_new > 1e-10:
                a_new = a_new / s_new
            best_obj = best_obj_ls
            best_a = a_new.copy()
            a = a_new
            stall = 0
            if verbose:
                print(f"    LP iter {it}: obj={best_obj:.8f}, t={best_t}")
        else:
            stall += 1
            if stall > 30:
                break
            #搅动
            a = 0.95 * a + 0.05 * g
            a = np.maximum(a, 0)
            s = np.sum(a)
            if s > 1e-10:
                a = a / s
    
    return best_a, best_obj


def mirror_descent_optimize(a, time_budget=100.0, tau_start=10.0, tau_end=200.0,
                           eta_start=0.1, eta_end=0.005, verbose=False):
    """Mirror descent with adaptive temperature and step size."""
    n = len(a)
    a = a.copy().astype(np.float64)
    s = np.sum(a)
    if s < 1e-10:
        a = np.ones(n) / n
    a = a / s
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    stall = 0
    patience = 50
    
    max_iters = int(time_budget * 100)  # generous upper bound
    for it in range(max_iters):
        if time.monotonic() - start > time_budget:
            break
        
        # Adaptive parameters
        progress = min(1.0, it / max(1, max_iters // 4))
        tau = tau_start + (tau_end - tau_start) * progress
        eta = eta_start * (eta_end / eta_start) ** progress
        
        a_new, w = mirror_descent_step(a, tau, eta)
        obj = compute_obj(a_new)
        
        if obj < best_obj - 1e-13:
            best_obj = obj
            best_a = a_new.copy()
            stall = 0
            a = a_new
            if verbose and it % 100 == 0:
                print(f"    MD iter {it}: obj={best_obj:.8f}, tau={tau:.1f}, eta={eta:.4f}")
        else:
            stall += 1
            if stall > patience:
                # Perturb and reset
                rng = np.random.default_rng(it * 7919)
                a_pert = best_a * np.exp(rng.normal(0, 0.05, n))
                a_pert = np.maximum(a_pert, 1e-10)
                a_pert /= np.sum(a_pert)
                a = a_pert
                stall = 0
                patience = max(30, patience // 2)
                if patience < 20:
                    break
    
    return best_a, best_obj


def coordinate_descent(a, time_budget=50.0):
    """Coordinate descent: optimize one element at a time."""
    n = len(a)
    a = a.copy().astype(np.float64)
    s = np.sum(a)
    if s < 1e-10:
        a = np.ones(n) / n
    a = a / s
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    
    it = 0
    while time.monotonic() - start < time_budget:
        it += 1
        improved = False
        for i in range(n):
            if time.monotonic() - start > time_budget - 2:
                break
            
            # Try changing a[i] while keeping sum = 1
            # If we increase a[i] by delta, we must decrease others
            # Simple approach: try a[i] * factor for various factors
            original = a[i]
            for factor in [0.5, 0.8, 1.2, 1.5]:
                a[i] = original * factor
                s = np.sum(a)
                if s < 1e-10:
                    a[i] = original
                    continue
                a /= s
                obj = compute_obj(a)
                if obj < best_obj - 1e-13:
                    best_obj = obj
                    best_a = a.copy()
                    improved = True
                    break
            a[i] = original  # restore
            s = np.sum(a)
            a /= s
        
        if not improved and it > 20:
            break
    
    return best_a, best_obj


def diversified_init(n, seed, parent=None):
    """Generate diverse initializations."""
    rng = np.random.default_rng(seed)
    k = np.arange(n)
    choice = rng.integers(0, 8)
    
    if parent is not None:
        # Resample parent
        parent_n = len(parent)
        new_k = np.linspace(0, parent_n - 1, n)
        a = np.interp(new_k, np.arange(parent_n), parent)
        a = np.maximum(a, 0)
        s = np.sum(a)
        if s > 1e-10:
            a /= s
        return a
    
    if choice == 0:
        # Chirp
        f0, f1 = rng.uniform(1, 3), rng.uniform(5, 15)
        t = k / n
        f = f0 + (f1 - f0) * t
        a = 1.0 + 0.5 * np.cos(2 * np.pi * np.cumsum(f) - np.pi / 2)
    elif choice == 1:
        # Multi-frequency
        a = np.ones(n)
        for _ in range(rng.integers(2, 5)):
            f = rng.uniform(1, 10)
            amp = rng.uniform(0.1, 0.3)
            a += amp * np.cos(2 * np.pi * f * k / n + rng.uniform(0, 2 * np.pi))
    elif choice == 2:
        # Gaussian bump + noise
        center = n * rng.uniform(0.3, 0.7)
        width = n * rng.uniform(0.1, 0.4)
        a = 1.0 + 0.5 * np.exp(-0.5 * ((k - center) / width) ** 2)
        a += 0.1 * rng.standard_normal(n)
    elif choice == 3:
        # Sine wave
        f = rng.uniform(2, 8)
        a = 1.0 + 0.4 * np.sin(2 * np.pi * f * k / n + rng.uniform(0, 2 * np.pi))
    elif choice == 4:
        # Random smooth
        n_low = max(5, n // 20)
        a_low = rng.uniform(0.5, 2.0, n_low)
        a = np.interp(k, np.arange(n_low), a_low)
    elif choice == 5:
        # Fourier with random phases (ensuring non-negative)
        coeffs = rng.uniform(0.05, 0.2, min(n // 2, 50))
        phases = rng.uniform(0, 2 * np.pi, len(coeffs))
        a = np.ones(n)
        for m, (c, p) in enumerate(zip(coeffs, phases)):
            a += c * np.cos(2 * np.pi * m * k / n + p)
    elif choice == 6:
        # Triangle wave
        f = rng.uniform(3, 10)
        t = 2 * np.pi * f * k / n
        a = 1.0 + 0.3 * np.abs(np.sin(t))
    else:
        # Pure random
        a = rng.uniform(0.5, 2.0, n)
    
    a = np.maximum(a, 0.01)
    s = np.sum(a)
    if s < 1e-10:
        a = np.ones(n)
    return a / s

            best_a = a_new.copy()
            if verbose:
                print(f"  After MD2: {best_obj:.8f}")
    
    # Phase 4: Another LP round
    remaining = time_budget - (time.monotonic() - start)
    if remaining > 20:
        t4 = min(50.0, remaining * 0.3)
        a_new = alternating_lp(best_a, max_iters=100, time_budget=t4, verbose=verbose)
        obj = compute_obj(a_new)
        if obj < best_obj:
            best_obj = obj
            best_a = a_new.copy()
            if verbose:
                print(f"  After LP2: {best_obj:.8f}")
    
    return best_a, best_obj


def local_search_around(a, time_budget=100.0, verbose=False):
    """Intensive local search with perturbations and re-optimization."""
    n = len(a)
    a = a.copy().astype(np.float64)
    s = np.sum(a)
    if s < 1e-10:
        a = np.ones(n) / n
    a = a / s
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    
    attempt = 0
    while time.monotonic() - start < time_budget - 10:
        attempt += 1
        remaining = time_budget - (time.monotonic() - start)
        if remaining < 15:
            break
        
        rng = np.random.default_rng(attempt * 31337 + 7)
        
        # Try different perturbation types
        ptype = attempt % 4
        
        if ptype == 0:
            # Gaussian multiplicative perturbation
            scale = 0.03 + 0.04 * (attempt % 3)
            a_pert = best_a * np.exp(rng.normal(0, scale, n))
        elif ptype == 1:
            # Localized perturbation
            center = rng.integers(0, n)
            width = rng.integers(n // 10, n // 3)
            delta = rng.normal(0, 0.1)
            a_pert = best_a.copy()
            for i in range(n):
                dist = min(abs(i - center), n - abs(i - center))
                a_pert[i] *= np.exp(delta * np.exp(-0.5 * (dist / width) ** 2))
        elif ptype == 2:
            # Swap perturbation
            a_pert = best_a.copy()
            i, j = rng.integers(0, n, 2)
            tmp = a_pert[i]
            a_pert[i] = a_pert[j]
            a_pert[j] = tmp
        else:
            # Scale perturbation on a subset
            mask = rng.random(n) < 0.3
            a_pert = best_a.copy()
            a_pert[mask] *= np.exp(rng.normal(0, 0.05, mask.sum()))
        
        a_pert = np.maximum(a_pert, 1e-10)
        s_pert = np.sum(a_pert)
        if s_pert < 1e-10:
            continue
        a_pert = a_pert / s_pert
        
        # Quick optimization
        sub_time = min(20.0, remaining * 0.3)
        a_opt, obj = optimize_pipeline(a_pert, time_budget=sub_time, verbose=False)
        
        if obj < best_obj:
            best_obj = obj
            best_a = a_opt.copy()
            if verbose:
                print(f"    LS attempt {attempt}: {best_obj:.8f}")
    
    return best_a, best_obj
