import numpy as np
from scipy.optimize import linprog, minimize
from scipy.sparse import csr_matrix
import time


def fast_conv(a):
    """Full self-convolution via FFT."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]


def compute_J(a):
    """Compute J(a) = 2*n*max(conv(a,a))/(sum(a))^2."""
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))


def build_A_sparse(n, a):
    """Build sparse matrix A where A[k,j] = a[k-j] for valid j."""
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        j_min = max(0, k - n + 1)
        j_max = min(n - 1, k)
        for j in range(j_min, j_max + 1):
            rows.append(k)
            cols.append(j)
            vals.append(a[k - j])
    return csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))


def lp_best_response(a, M):
    """Solve LP: max sum(g) s.t. A@g <= M*1, g >= 0."""
    n = len(a)
    A = build_A_sparse(n, a)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    try:
        r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                    options={'presolve': True})
        if r.success and r.x is not None:
            g = np.maximum(r.x, 0)
            if g.sum() > 1e-15:
                return g
    except Exception:
        pass
    return None


def lp_mirror_descent_v2(a, max_iters, deadline, mode='linear', initial_t=0.4):
    """
    Improved LP mirror descent with adaptive step sizes and restarts.
    mode: 'linear' or 'multiplicative'
    """
    a = np.maximum(a.astype(np.float64), 1e-10)
    a = a / a.sum()
    current_J = compute_J(a)
    best_J = current_J
    best_a = a.copy()
    no_improve = 0
    t = initial_t
    consecutive_bad = 0

    for it in range(max_iters):
        if time.monotonic() > deadline:

        M = float(b.max())
        g = lp_best_response(a, M)

def gs_spectral_iterations(a, n_iters, flatten_rate, deadline):
    """
    Gerchberg-Saxton style iterations: alternate between
    time-domain (non-negativity) and frequency-domain (spectrum flattening).
    """
    a = np.maximum(a.astype(np.float64), 1e-10)
    a = a / a.sum()
    best_J = compute_J(a)
    best_a = a.copy()
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1

    for it in range(n_iters):
        if time.monotonic() > deadline:
            break

        # Frequency domain: flatten the power spectrum
        A = np.fft.rfft(a, size)
        mag = np.abs(A)
        phase = np.angle(A)

        # Adaptive flattening: compress dynamic range of magnitude
        mag_min = mag[mag > 1e-10].min() if (mag > 1e-10).any() else 1e-10
        mag_max = mag.max()
        if mag_max > 1e-10:
            # Log-compress then normalize
            log_mag = np.log(mag + 1e-15)
            log_target = np.full_like(log_mag, log_mag.mean())
            log_new = (1 - flatten_rate) * log_mag + flatten_rate * log_target
            new_mag = np.exp(log_new)
            # Normalize to preserve total energy
            new_mag = new_mag * (mag.sum() / new_mag.sum())
        else:
            new_mag = mag

        A_new = new_mag * np.exp(1j * phase)
        a_new = np.fft.irfft(A_new, size)[:n]

        # Time domain: enforce non-negativity
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s < 1e-10:
            break
        a_new = a_new / s

        J_new = compute_J(a_new)
        if J_new < best_J:
            best_J = J_new
            best_a = a_new.copy()
        a = a_new

    return best_a, best_J

def subgradient_polish(a, max_iters, deadline):
    """Subgradient descent with momentum on the max autocorrelation."""
    a = np.maximum(a.astype(np.float64), 1e-10)
    a = a / a.sum()
    best_M = fast_conv(a).max()
    best_a = a.copy()
    momentum = np.zeros_like(a)
    beta = 0.8
    n = len(a)

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        b = fast_conv(a)
        M = b.max()
        tol = 1e-10 * M
        active = np.where(b >= M - tol)[0]

        grad = np.zeros_like(a)
        for k in active:
            lo = max(0, k - n + 1)
            hi = min(n - 1, k)
            for i in range(lo, hi + 1):
                grad[i] += 2.0 * a[k - i]

        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-15:
            break

        t = 1.0 / (gnorm * (it + 1.0) ** 0.6)
        momentum = beta * momentum + (1 - beta) * grad / gnorm
        a_new = a - t * momentum
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s < 1e-10:
            break
        a_new = a_new / s

        new_M = fast_conv(a_new).max()
        if new_M < best_M:
            best_M = new_M
            best_a = a_new.copy()
        a = a_new

    return best_a


def lbb_objective(x):
    """Objective for L-BFGS-B: maximize flatness -> minimize max self-conv (scale invariant)."""
    ax = np.maximum(x, 0.0)
    s = ax.sum()
    if s < 1e-8:
        return 1e10
    ax = ax / s
    b = fast_conv(ax)
    return float(b.max())


def resample_seq(src, dest_n):
    """Resample 1D sequence to new length."""
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)

        candidates_n = []
        for dn in [-128, -64, 64, 128, 256, -256]:
            nn = base_n + dn
            if 256 <= nn <= 2048:
                candidates_n.append(nn)
        # Also try some fixed good lengths
        for nn in [384, 512, 640, 768, 896, 1024]:
            if nn != base_n and 256 <= nn <= 2048:
                candidates_n.append(nn)
        candidates_n = sorted(set(candidates_n))

        per_try_time = max(15, min(60, (deadline - 500 - time.monotonic()) / max(1, len(candidates_n))))

        for n_try in candidates_n:
            if time.monotonic() > deadline - 480:
                break
            a_try = resample_seq(current_best, n_try)
            a_try = np.maximum(a_try, 1e-10)
            a_try = a_try / a_try.sum()
            dl = time.monotonic() + per_try_time
            a_try, J_try = lp_mirror_descent_v2(a_try, 200, dl, mode='linear', initial_t=0.35)
            track(a_try)
            if J_try < compute_J(current_best):
                current_best = a_try.copy()
                print(f"  n={n_try}: J={J_try:.8f} (NEW BEST)", flush=True)
            else:
                print(f"  n={n_try}: J={J_try:.8f}", flush=True)

        print(f"Phase 5 done ({time.monotonic()-t0:.1f}s)", flush=True)

    # === Phase 6: Long LP refinement (main convergence phase) ===
    if time.monotonic() < deadline - 200:
        t0 = time.monotonic()
        print(f"\n{'='*50}", flush=True)
        print(f"Phase 6: Long LP linear (n={current_best.size})", flush=True)
        print(f"{'='*50}", flush=True)

        a6 = current_best.copy()
        a6 = np.maximum(a6, 1e-10)
        a6 = a6 / a6.sum()
        a6, J6 = lp_mirror_descent_v2(a6, 100000, deadline - 180, mode='linear', initial_t=0.45)
        track(a6)
        print(f"  J={J6:.8f} ({time.monotonic()-t0:.1f}s)", flush=True)

        if J6 < compute_J(current_best):
            current_best = a6.copy()

    # === Phase 7: LP multiplicative polish ===
    if time.monotonic() < deadline - 100:
        t0 = time.monotonic()
        print(f"\n{'='*50}", flush=True)
        print(f"Phase 7: LP multiplicative polish (n={current_best.size})", flush=True)
        print(f"{'='*50}", flush=True)

        a7 = current_best.copy()
        a7 = np.maximum(a7, 1e-10)
        a7 = a7 / a7.sum()
        a7, J7 = lp_mirror_descent_v2(a7, 50000, deadline - 80, mode='multiplicative', initial_t=0.25)
        track(a7)
        print(f"  J={J7:.8f} ({time.monotonic()-t0:.1f}s)", flush=True)

        if J7 < compute_J(current_best):
            current_best = a7.copy()

    # === Phase 8: Subgradient polish ===
    if time.monotonic() < deadline - 50:
        t0 = time.monotonic()
        print(f"\n{'='*50}", flush=True)
        print(f"Phase 8: Subgradient polish (n={current_best.size})", flush=True)
        print(f"{'='*50}", flush=True)

        a8 = subgradient_polish(current_best, 2000, min(deadline - 40, t0 + 60))
        track(a8)
        J8 = compute_J(a8)
        if J8 < compute_J(current_best):
            current_best = a8.copy()
        print(f"  J={J8:.8f} ({time.monotonic()-t0:.1f}s)", flush=True)

    # === Phase 9: L-BFGS-B final polish ===
    if time.monotonic() < deadline - 10 and best_a is not None:
        t0 = time.monotonic()
        print(f"\n{'='*50}", flush=True)
        print(f"Phase 9: L-BFGS-B final (n={best_a.size})", flush=True)
        print(f"{'='*50}", flush=True)

        # Try L-BFGS-B from the best found
        try:
            res = minimize(lbb_objective, best_a.astype(np.float64),
                           method='L-BFGS-B', bounds=[(0.0, None)] * best_a.size,
                           options={'maxiter': 200, 'max_fun': 50000, 'ftol': 1e-18, 'gtol': 1e-14})
            if res.fun < 1e10:
                track(res.x)
                print(f"  L-BFGS-B: obj={res.fun:.12f}", flush=True)
        except Exception as e:
            print(f"  L-BFGS-B error: {e}", flush=True)

        # Try from current_best too if different
        if current_best is not None and not np.allclose(current_best, best_a, atol=1e-8):
            try:
                res2 = minimize(lbb_objective, current_best.astype(np.float64),
                                method='L-BFGS-B', bounds=[(0.0, None)] * current_best.size,
                                options={'maxiter': 100, 'max_fun': 20000, 'ftol': 1e-18})
                if res2.fun < 1e10:
                    track(res2.x)
            except Exception:
                pass
        print(f"Phase 9 done ({time.monotonic()-t0:.1f}s)", flush=True)

    # === Phase 10: Final LP from L-BFGS-B result ===
    if time.monotonic() < deadline - 5 and best_a is not None:
        t0 = time.monotonic()
        print(f"\n{'='*50}", flush=True)
        print(f"Phase 10: Final LP (n={best_a.size})", flush=True)
        print(f"{'='*50}", flush=True)

        a_final = best_a.copy()
        a_final = np.maximum(a_final, 1e-10)
        a_final = a_final / a_final.sum()
        a_final, J_final = lp_mirror_descent_v2(a_final, 5000, deadline, mode='linear', initial_t=0.3)
        track(a_final)
        print(f"  J={J_final:.8f} ({time.monotonic()-t0:.1f}s)", flush=True)

    # Final decision
    if best_a is not None:
        result = best_a
    else:
        result = current_best

    print(f"\n{'='*50}", flush=True)
    print(f"FINAL: J={best_score:.8f} n={result.size} time={time.monotonic()-t_start:.0f}s", flush=True)
    print(f"{'='*50}", flush=True)

    return [float(x) for x in result]
