def fast_conv(a):
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]

    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))

    rows, cols, vals = [], [], []
    for k in range(2 * n - 1):
        j_min = max(0, k - n + 1)
        j_max = min(n - 1, k)
        for j in range(j_min, j_max + 1):
            rows.append(k)
            cols.append(j)
            vals.append(a[k - j])
    return csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))

def lp_best_response(a, M, time_limit=5.0):

                    options={'time_limit': time_limit})
        if r.success and r.x is not None:
            return np.maximum(r.x, 0)
    except:
        pass
    return None

def smooth_max_adam(a, beta_start, beta_end, max_iters, deadline, lr=0.01):
    """Adam optimizer on smooth-max objective."""
    a = np.maximum(a.astype(np.float64), 1e-10)
    a = a / a.sum()
    best_J = compute_J(a)
    best_a = a.copy()
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    t = 0
    beta1, beta2, eps = 0.8, 0.999, 1e-8

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        t += 1
        # Anneal beta
        frac = it / max(1, max_iters - 1)
        beta = beta_start * (beta_end / beta_start) ** frac

        grad = smooth_max_grad(a, beta)
        # Normalize gradient by current max C
        C_max = fast_conv(a).max()
        if C_max > 1e-15:
            grad = grad / C_max

        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * grad ** 2
        m_hat = m / (1 - beta1 ** t)
        v_hat = v / (1 - beta2 ** t)

        a_new = a - lr * m_hat / (np.sqrt(v_hat) + eps)
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s < 1e-10:
            continue
        a_new = a_new / s
        J_new = compute_J(a_new)
        if J_new < best_J:
            best_J = J_new
            best_a = a_new.copy()
        a = a_new

    return best_a, best_J
