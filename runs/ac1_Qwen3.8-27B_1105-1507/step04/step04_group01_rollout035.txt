        # cross_corr_y_w[i] = sum_d y[d] * w[d+i]
        # = (IFFT(FFT(y) * conj(FFT(w))))[i] ... for circular
        # For linear: pad both to length >= n + (2n-1) - 1 = 3n-2
        
        size = 1
        while size < 3 * n - 1:
            size <<= 1
        fy = rfft(y, size)
        fw = rfft(w, size)
        # cross_corr[i] = sum_d y[d] * w[d+i]
        # In FFT: IFFT(conj(FFT(y)) * FFT(w)) gives circular cross-correlation
        # For linear: we need to be careful with padding
        # Actually: cross_corr_linear[i] = sum_{d=0}^{n-1} y[d] * w[i+d]
        # = sum_{d=0}^{n-1} y[d] * w[i+d]
        # Let's compute (y * w_rev)[k] where w_rev = w[::-1]
        # (y * w_rev)[k] = sum_d y[d] * w_rev[k-d] = sum_d y[d] * w[(2n-2)-(k-d)]
        # = sum_d y[d] * w[2n-2-k+d]
        # We want sum_d y[d] * w[i+d] for i=0..n-1
        # So we need 2n-2-k = i, i.e., k = 2n-2-i
        # (y * w_rev)[2n-2-i] = sum_d y[d] * w[i+d] (when valid)
        
        conv_yw = fft_convolve(y, w[::-1])
        # conv_yw has length n + (2n-1) - 1 = 3n-2
        # We want indices 2n-2-i for i=0..n-1, i.e., indices n-1 to 2n-2
        # conv_yw[2n-2-i] for i=0: conv_yw[2n-2]
        # conv_yw[2n-2-i] for i=n-1: conv_yw[n-1]
        grad = 2.0 * conv_yw[n-1:2*n-1][::-1]

        # Nesterov update
        y_new = a - eta * grad
        # Project onto simplex (a >= 0, sum = 1)
        # For multiplicative update: a_new = y_new / sum(y_new) if all positive
        # Otherwise, project
        if np.min(y_new) > 0:
            a_new = y_new / np.sum(y_new)
        else:
            # Project onto simplex
            y_new = np.maximum(y_new, 0)
            s_new = np.sum(y_new)
            if s_new < 1e-15:
                a_new = a.copy()
            else:
                a_new = y_new / s_new

        # Update Nesterov momentum
        beta = (1 + np.sqrt(1 + 4 * mu * mu)) / 2  # or use standard: (1+sqrt(1+4*mu^2))/2
        # Standard Nesterov: beta_k = (1 + sqrt(1 + 4*mu^2)) / 2
        y = a_new + (beta - 1) * (a_new - a)  # This isn't quite right
        # Standard Nesterov momentum:
        # y_{k+1} = x_{k+1} + (t_{k+1}-1)/t_{k+2} * (x_{k+1} - x_k)
        # Simplified: y = a_new + momentum * (a_new - a)
        y = a_new + 0.9 * (a_new - a)
        y = np.maximum(y, 1e-15)
        y = y / np.sum(y)

        a = a_new
        conv_y = compute_conv(y)
        max_y = float(np.max(conv_y))

        # Evaluate
        obj = compute_obj(a)
        if obj < best_obj - 1e-12:
            best_obj = obj
            best_a = a.copy()

        # Adaptive step size
        if it % 100 == 99:
            # Reduce step size slowly
            eta *= 0.995

    return best_a, best_obj

def build_toeplitz_conv(a):
    """Build matrix A such that A @ g = conv(a, g)."""
    n = len(a)
    m = len(g) if 'g' in dir() else n  # We'll use n for g
    A = np.zeros((2 * n - 1, n), dtype=np.float64)
    for j in range(n):
        lo = j
        hi = min(j + n, 2 * n - 1)
        A[lo:hi, j] = a[:hi - lo]
    return A

def solve_lp_for_g(a, M, time_limit=20):
    """Solve: max sum(g) s.t. (a*g)[k] <= M, g >= 0. Returns normalized g."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        lo = j
        hi = min(j + n, 2 * n - 1)
        A[lo:hi, j] = a[:hi - lo]
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': time_limit})
        if res.status == 0:
            g = np.maximum(res.x, 0)
            s = float(np.sum(g))
            if s > 1e-10:
                return g / s
    except Exception:
        pass
    return None

def solve_lp_minpeak_direct(a, time_limit=20):
    """Solve LP: min M s.t. (a*g)[k] <= M, sum(g)=1, g>=0. Returns (g, M)."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        lo = j
        hi = min(j + n, 2 * n - 1)
        A[lo:hi, j] = a[:hi - lo]
    c = np.zeros(n + 1)
    c[-1] = 1.0
    A_ub = np.zeros((2 * n - 1, n + 1))
    A_ub[:, :n] = A
    A_ub[:, n] = -1.0
    b_ub = np.zeros(2 * n - 1)
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])
    bounds = [(0.0, None)] * n + [(0.0, None)]
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': time_limit})
        if res.status == 0:
            g = np.maximum(res.x[:n], 0)
            s = np.sum(g)
            if s > 1e-10:
                g = g / s
                M_opt = max(0.0, float(res.x[n]))
                return g, M_opt
    except Exception:
        pass
    return None, None

def alternating_lp_v2(a_init, max_iters=100, time_budget=60.0):
    """
    Improved LP alternating: alternate between solving LP for g given a,
    and LP for a given g (symmetric). Use the primal-dual gap to guide
    the blending.
    """
    n = len(a_init)
    a = a_init.copy().astype(np.float64)
    s = np.sum(a)
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    for it in range(max_iters):
        if time.monotonic() - start > time_budget:
            break

        conv_a = compute_conv(a)
        M_a = float(np.max(conv_a))

        # Solve LP for g given a
        g = solve_lp_for_g(a, M_a, time_limit=15)
        if g is None:
            break

        # Also solve LP for a' given g (symmetric role)
        a_lp = solve_lp_for_g(g, M_a, time_limit=15)

        # Try both directions and blends
        candidates = []
        for t in [0.1, 0.2, 0.3, 0.5, 0.7, 0.9, 1.0]:
            a_try = (1 - t) * a + t * g
            a_try = np.maximum(a_try, 0)
            s_try = np.sum(a_try)
            if s_try > 1e-15:
                a_try = a_try / s_try
                candidates.append(a_try)

        if a_lp is not None:
            for t in [0.3, 0.5, 0.7, 1.0]:
                a_try = (1 - t) * a + t * a_lp
                a_try = np.maximum(a_try, 0)
                s_try = np.sum(a_try)
                if s_try > 1e-15:
                    a_try = a_try / s_try
                    candidates.append(a_try)

        best_try = None
        best_try_obj = best_obj
        for a_try in candidates:
            obj_try = compute_obj(a_try)
            if obj_try < best_try_obj:
                best_try_obj = obj_try
                best_try = a_try.copy()

        if best_try is not None:
            best_obj = best_try_obj
            best_a = best_try
            a = best_try
        else:
            # Stuck: try perturbation
            if it > 10:
                rng = np.random.default_rng(it * 104729)
                a = a * np.exp(rng.normal(0, 0.05, n))
                a = np.maximum(a, 0)
                s = np.sum(a)
                if s > 1e-15:
                    a = a / s

    return best_a, best_obj

def soft_max_obj_and_grad(z, tau):
    """Compute soft-max approximation of max(conv) and gradient."""
    n = len(z)
    a = softmax(z)
    conv = compute_conv(a)
    cmax = float(np.max(conv))
    shifted = np.clip((conv - cmax) / tau, -500, 500)
    exp_c = np.exp(shifted)
    Z = float(np.sum(exp_c))
    soft_max_val = cmax + tau * np.log(Z)
    w = exp_c / Z

    # grad_a[i] = 2 * sum_j w[j] * a[j-i] (valid indices)
    conv_aw = fft_convolve(a, w[::-1])
    grad_a = 2.0 * conv_aw[n-1:2*n-1][::-1]

    # grad_z = a * (grad_a - sum(a * grad_a))
    mean_g = float(np.dot(a, grad_a))
    grad_z = a * (grad_a - mean_g)

    return soft_max_val, grad_z

def optimize_lbfgs_z(z_init, tau_list, max_iter=200, time_budget=20.0):
    """L-BFGS-B on softmax parameterization."""
    n = len(z_init)
    z = z_init.copy()
    best_a = softmax(z).copy()
    best_obj = compute_obj(best_a)
    start = time.monotonic()

    for tau in tau_list:
        if time.monotonic() - start > time_budget:
            break

        def func_and_grad(z_val):
            return soft_max_obj_and_grad(z_val, tau)

        try:
            result = minimize(func_and_grad, z, jac=True, method='L-BFGS-B',
                              options={'maxiter': max_iter, 'ftol': 1e-15, 'gtol': 1e-10})
            z = result.x
        except Exception:
            continue

        a_new = softmax(z)
        a_new = np.maximum(a_new, 1e-30)
        s = np.sum(a_new)
        if s < 1e-10:
            continue
        a_new = a_new / s
        obj = compute_obj(a_new)
        if obj < best_obj:
            best_obj = obj
            best_a = a_new.copy()

    return best_a, best_obj

def structured_init(n, seed=0):
    """Generate structured initializations."""
    rng = np.random.default_rng(seed)
    k = np.arange(n, dtype=np.float64)
    t = k / n

    choice = rng.integers(0, 12)

    if choice == 0:
        # Triangle
        a = 1.0 - 2.0 * np.abs(t - 0.5)
    elif choice == 1:
        # Sine
        f = rng.uniform(0.5, 3.0)
        a = 1.0 + 0.5 * np.cos(2 * np.pi * f * t)
    elif choice == 2:
        # Two sines
        f1 = rng.uniform(0.3, 1.5)
        f2 = rng.uniform(2.0, 5.0)
        a = 1.0 + 0.3 * np.cos(2 * np.pi * f1 * t) + 0.2 * np.cos(2 * np.pi * f2 * t)
    elif choice == 3:
        # Ramp
        a = np.linspace(0.1, 2.0, n)
    elif choice == 4:
        # Bimodal
        c1 = rng.uniform(0.2, 0.4)
        w1 = rng.uniform(0.05, 0.15)
        a = np.exp(-0.5 * ((t - c1) / w1) ** 2) + np.exp(-0.5 * ((t - (1-c1)) / w1) ** 2) + 0.1
    elif choice == 5:
        # Polynomial
        p = rng.uniform(0.5, 3.0)
        a = (1.0 - t) ** p + 0.1
    elif choice == 6:
        # Random smoothed
        a = rng.random(n)
        win = max(3, n // 30)
        kernel = np.ones(win) / win
        a = np.convolve(a, kernel, mode='same')
        a = 0.1 + a / np.max(a)
    elif choice == 7:
        # Chirp
        f0 = rng.uniform(0.5, 2.0)
        f1 = rng.uniform(2.0, 5.0)
        a = 1.0 + 0.4 * np.cos(2 * np.pi * (f0 * t + 0.5 * (f1 - f0) * t**2))
    elif choice == 8:
        # Step function
        nsteps = rng.integers(3, 8)
        a = np.repeat(rng.uniform(0.3, 2.0, nsteps), n // nsteps + 1)[:n]
    elif choice == 9:
        # Gaussian
        c = rng.uniform(0.3, 0.7)
        w = rng.uniform(0.1, 0.3)
        a = 0.2 + np.exp(-0.5 * ((t - c) / w) ** 2)
    elif choice == 10:
        # Cosine window
        a = 0.5 * (1 + np.cos(2 * np.pi * t))
    else:
        # Random
        a = rng.uniform(0.1, 2.0, n)

    a = np.maximum(a, 0.01)
    s = np.sum(a)
    return a / s

def resample(a, new_n):
    """Resample sequence to new length."""
    n = len(a)
    if n == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    return np.interp(x_new, x_old, a)

def peak_chasing(a_init, time_budget=20.0, max_iters=10000):
    """
    Direct peak-chasing: at each step, find the peak of the convolution
    and take a step to reduce it. Uses adaptive step sizes and multiple
    peak tracking.
    """
    n = len(a_init)
    a = a_init.copy().astype(np.float64)
    s = np.sum(a)
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    conv = compute_conv(a)
    max_val = float(np.max(conv))

    # Track top K peaks for smoother gradient
    K = min(10, 2 * n - 1)
    step = 0.5 / n

    for it in range(max_iters):
        if time.monotonic() - start > time_budget:
            break

        # Compute weights on top K peaks
        top_k_idx = np.argsort(conv)[-K:]
        peak_vals = conv[top_k_idx]
        # Weight by how much above the mean
        mean_val = float(np.mean(conv))
        excess = np.maximum(peak_vals - mean_val, 0)
        if np.sum(excess) < 1e-30:
            excess = peak_vals  # fallback
        w_peaks = excess / np.sum(excess)

        # Gradient: for peak at position k, grad[i] += w * a[k-i] (valid)
        # Compute using FFT
        w_full = np.zeros(2 * n - 1)
        w_full[top_k_idx] = w_peaks

        conv_aw = fft_convolve(a, w_full[::-1])
        grad = 2.0 * conv_aw[n-1:2*n-1][::-1]

        # Nesterov-style: maintain momentum
        if it == 0:
            vel = np.zeros(n)
        
        vel = 0.9 * vel - step * grad
        a_new = a + vel
        a_new = np.maximum(a_new, 0)
        s_new = np.sum(a_new)
        if s_new < 1e-15:
            a_new = a.copy()
        else:
            a_new = a_new / s_new

        a = a_new
        conv = compute_conv(a)
        max_val = float(np.max(conv))
        obj = 2.0 * n * max_val  # since sum = 1

        if obj < best_obj - 1e-12:
            best_obj = obj
            best_a = a.copy()

        # Decay step
        if it % 50 == 49:
            step *= 0.98

    return best_a, 2.0 * n * best_obj  # Convert back (we stored 2n*max with sum=1)

def peak_chasing_v2(a_init, time_budget=20.0, max_iters=10000):
    """Improved peak chasing with proper objective tracking."""
    n = len(a_init)
    a = a_init.copy().astype(np.float64)
    s = np.sum(a)
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    conv = compute_conv(a)
    K = min(20, 2 * n - 1)
    step = 2.0 / n
    vel = np.zeros(n)
    beta = 0.9

    for it in range(max_iters):
        if time.monotonic() - start > time_budget:
            break

        # Soft weights on top peaks
        sorted_conv = np.sort(conv)[::-1]
        threshold = sorted_conv[K] if K < len(conv) else sorted_conv[-1]
        excess = np.maximum(conv - threshold, 0)
        if np.sum(excess) < 1e-30:
            idx = np.argmax(conv)
            w = np.zeros(len(conv))
            w[idx] = 1.0
        else:
            w = excess / np.sum(excess)

        conv_aw = fft_convolve(a, w[::-1])
        grad = 2.0 * conv_aw[n-1:2*n-1][::-1]

        # Nesterov momentum
        vel = beta * vel - step * grad
        a_new = a + vel
        a_new = np.maximum(a_new, 0)
        s_new = np.sum(a_new)
        if s_new < 1e-15:
            a_new = a.copy()
        else:
            a_new = a_new / s_new

        a = a_new
        obj = compute_obj(a)

        if obj < best_obj - 1e-12:
            best_obj = obj
            best_a = a.copy()

        # Adaptive step
        if it % 100 == 99:
            step *= 0.95
            beta = min(0.99, beta + 0.001)

    return best_a, best_obj

def polish(a_init, time_budget=60.0):
    """Multi-method polishing."""
    n = len(a_init)
    a = a_init.copy().astype(np.float64)
    s = np.sum(a)
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    for round_num in range(20):
        if time.monotonic() - start > time_budget - 5:
            break
        remaining = time_budget - (time.monotonic() - start)
        if remaining < 5:
            break

        # 1. Peak chasing
        t1 = min(remaining * 0.25, 15.0)
        a_new, obj_new = peak_chasing_v2(best_a, time_budget=t1)
        if obj_new < best_obj:
            best_obj = obj_new
            best_a = a_new.copy()

        # 2. Mirror descent
        remaining = time_budget - (time.monotonic() - start)
        if remaining > 10:
            t2 = min(remaining * 0.2, 12.0)
            a_new, obj_new = mirror_descent_nesterov(best_a, time_budget=t2)
            if obj_new < best_obj:
                best_obj = obj_new
                best_a = a_new.copy()

        # 3. LP alternating (if n is small enough)
        remaining = time_budget - (time.monotonic() - start)
        if remaining > 15 and n <= 800:
            t3 = min(remaining * 0.25, 20.0)
            a_new, obj_new = alternating_lp_v2(best_a, max_iters=50, time_budget=t3)
            if obj_new < best_obj:
                best_obj = obj_new
                best_a = a_new.copy()

        # 4. L-BFGS-B with soft-max
        remaining = time_budget - (time.monotonic() - start)
        if remaining > 10:
            t4 = min(remaining * 0.2, 12.0)
            z = np.log(np.maximum(best_a, 1e-30))
            tau_list = [1.0/n, 0.5/n, 0.1/n, 0.05/n]
            a_new, obj_new = optimize_lbfgs_z(z, tau_list, max_iter=150, time_budget=t4)
            if obj_new < best_obj:
                best_obj = obj_new
                best_a = a_new.copy()

    return best_a, best_obj

def try_parent():
    """Try to load parent sequence."""
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64)
        parent = np.clip(parent, 0, 1000.0)
        if parent.size > 0 and np.isfinite(parent).all():
            obj = compute_obj(parent)
            print(f"Parent (n={parent.size}): obj={obj:.6f}")
            return parent.copy(), obj
    except Exception:
        pass
    return None, np.inf

def propose_candidate(seed=42, budget_s=1500):
    """
    Main search function. Combines multiple optimization strategies:
    - Peak chasing with Nesterov momentum
    - Mirror descent (multiplicative weight)
    - LP alternating optimization
    - Soft-max L-BFGS-B
    Searches over multiple sequence lengths and uses structured initializations.
    """
    start_time = time.monotonic()
    deadline = start_time + float(budget_s) - 30.0

    best_seq = None
    best_score = np.inf

    # Load parent
    parent, parent_obj = try_parent()
    if parent is not None and parent_obj < best_score:
        best_score = parent_obj
        best_seq = parent.copy()

    # N values to explore
    n_values = [200, 300, 400, 500, 600, 700, 800, 1000, 1200, 1500]
    if best_seq is not None:
        np_ = len(best_seq)
        extra = [np_, max(100, np_ - 100), min(2000, np_ + 100), min(2000, np_ + 200), min(2000, np_ + 500)]
        n_values = sorted(set(n_values + extra))

    print(f"Searching n values: {n_values}")
    print(f"Initial best: {best_score:.6f}")

    candidates = []

    for n in n_values:
        if time.monotonic() > deadline - 100:
            break
        remaining = deadline - time.monotonic()
        if remaining < 20:
            break

        # Time allocation
        if n <= 400:
            alloc = min(60.0, remaining * 0.12)
        elif n <= 800:
            alloc = min(80.0, remaining * 0.13)
        else:
            alloc = min(100.0, remaining * 0.14)
        alloc = max(20.0, alloc)

        print(f"\nN={n}: {remaining:.0f}s remaining, alloc={alloc:.0f}s")

        n_inits = max(2, int(alloc / 20))
        n_inits = min(n_inits, 4)

        for init_idx in range(n_inits):
            if time.monotonic() > deadline - 100:
                break
            remaining_init = deadline - time.monotonic()
            if remaining_init < 15:
                break

            # Generate initial sequence
            if init_idx == 0 and best_seq is not None:
                a = resample(best_seq, n)
            else:
                a = structured_init(n, seed + n * 7919 + init_idx * 31)

            # Phase 1: Peak chasing (fast, good for initial shaping)
            pc_time = min(remaining_init * 0.2, 12.0)
            a, obj = peak_chasing_v2(a, time_budget=pc_time)

            # Phase 2: Mirror descent
            remaining_init = deadline - time.monotonic()
            if remaining_init > 15:
                md_time = min(remaining_init * 0.2, 15.0)
                a, obj = mirror_descent_nesterov(a, time_budget=md_time)

            # Phase 3: LP alternating (if n allows)
            remaining_init = deadline - time.monotonic()
            if remaining_init > 20 and n <= 1000:
                lp_time = min(remaining_init * 0.3, 25.0)
                a, obj = alternating_lp_v2(a, max_iters=50, time_budget=lp_time)

            # Phase 4: L-BFGS-B polish
            remaining_init = deadline - time.monotonic()
            if remaining_init > 10:
                lbfgs_time = min(remaining_init * 0.2, 12.0)

"""
Multi-strategy optimizer for minimizing 2*n*max(conv(a,a))/(sum(a))^2.
Uses peak-chasing with Nesterov momentum, mirror descent (multiplicative weights),
LP alternating, soft-max L-BFGS-B, and active-set QP approximation.
Searches over multiple sequence lengths with structured initializations.
"""

import numpy as np
from scipy.optimize import minimize, linprog
from scipy.fft import rfft, irfft
import time


def fft_convolve(a, b):
    """Linear convolution of a and b using FFT."""
    n, m = len(a), len(b)
    size = 1
    while size < n + m - 1:
        size <<= 1
    fa = rfft(a, size)
    fb = rfft(b, size)
    return irfft(fa * fb, size)[:n + m - 1]


def compute_conv(a):
    """Compute a*a linear convolution using FFT. Returns array of length 2n-1."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    conv = irfft(fa * fa, size)
    return conv[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    conv = compute_conv(a)
    max_b = float(np.max(conv))
    sum_a = float(np.sum(a))
    if sum_a < 0.01:
        return np.inf
    return 2.0 * n * max_b / (sum_a * sum_a)


def softmax(z):
    """Numerically stable softmax."""
    z = z - np.max(z)
    e = np.exp(z)
    s = np.sum(e)
    return e / s


def normalize(a):
    """Normalize to sum=1, clip negatives."""
    a = np.maximum(a, 0.0)
    s = np.sum(a)
    if s < 1e-15:
        return np.ones(len(a)) / len(a)
    return a / s


def compute_grad_from_weights(a, w):
    """
    Compute gradient of sum_k w[k]*conv[k] w.r.t. a.
    grad[i] = 2 * sum_j w[j] * a[j-i] (valid indices only).
    Uses FFT for efficiency.
    """
    n = len(a)
    # conv[j] = sum_i a[i]*a[j-i]
    # d(sum_k w[k]*conv[k])/da[i] = 2*sum_k w[k]*a[k-i] (when 0<=k-i<n)
    # Let d = k-i: sum_{d: 0<=i+d<len(w)} w[i+d]*a[d]
    # This is a cross-correlation: (a * w_rev)[2n-2-i] type indexing
    w_rev = w[::-1]
    conv_aw = fft_convolve(a, w_rev)
    # conv_aw has length n + (2n-1) - 1 = 3n-2
    # We want indices from n-1 to 2n-2, reversed
    grad = 2.0 * conv_aw[n - 1:2 * n - 1][::-1]
    return grad


def peak_chasing_nesterov(a_init, time_budget=20.0, max_iters=50000):
    """
    Peak chasing with Nesterov momentum. Directly targets the maximum
    of the convolution using soft weights on top peaks.
    """
    n = len(a_init)
    a = normalize(a_init.copy().astype(np.float64))
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    conv = compute_conv(a)
    step = 3.0 / n
    beta = 0.85
    vel = np.zeros(n)
    K = min(30, 2 * n - 1)
    stall = 0

    for it in range(max_iters):
        if time.monotonic() - start > time_budget:
            break

        # Compute soft weights on top K peaks
        if it % 10 == 0:
            conv = compute_conv(a)
        sorted_idx = np.argsort(conv)[::-1][:K]
        peak_vals = conv[sorted_idx]
        mean_val = float(np.mean(conv))
        excess = np.maximum(peak_vals - mean_val, 1e-30)
        w_peaks = excess / np.sum(excess)

        w_full = np.zeros(2 * n - 1)
        w_full[sorted_idx] = w_peaks

        grad = compute_grad_from_weights(a, w_full)

        # Nesterov momentum update
        vel = beta * vel - step * grad
        a_new = a + vel
        a_new = np.maximum(a_new, 0)
        s_new = np.sum(a_new)
        if s_new < 1e-15:
            a_new = a.copy()
        else:
            a_new = a_new / s_new

        a = a_new
        obj = compute_obj(a)

            best_obj = obj
            best_a = a.copy()
            stall = 0
        else:
            stall += 1

            beta = min(0.99, beta + 0.002)

            a = a * np.exp(rng.normal(0, 0.03, n))
            a = normalize(a)
            vel = np.zeros(n)
            step = 3.0 / n
            stall = 0

    return best_a, best_obj


def mirror_descent(a_init, time_budget=20.0, max_iters=30000):
    """
    Multiplicative weight (mirror descent) method for the simplex-constrained
    problem. Uses exponential updates that naturally preserve non-negativity.
    """
    n = len(a_init)
    a = normalize(a_init.copy().astype(np.float64))
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    eta = 1.0 / np.sqrt(n)
    conv = compute_conv(a)
    K = min(20, 2 * n - 1)

    for it in range(max_iters):
        if time.monotonic() - start > time_budget:
            break

        if it % 5 == 0:
            conv = compute_conv(a)

        # Soft weights on peaks
        sorted_idx = np.argsort(conv)[::-1][:K]
        peak_vals = conv[sorted_idx]
        mean_val = float(np.mean(conv))
        excess = np.maximum(peak_vals - mean_val, 1e-30)
        w_peaks = excess / np.sum(excess)

        w_full = np.zeros(2 * n - 1)
        w_full[sorted_idx] = w_peaks
        grad = compute_grad_from_weights(a, w_full)

        # Multiplicative update: a_new[i] = a[i] * exp(-eta * grad[i])
        # Center grad to preserve sum
        grad_centered = grad - np.mean(grad)
        log_update = -eta * grad_centered
        log_update = np.clip(log_update, -5.0, 5.0)

        a_new = a * np.exp(log_update)
        a_new = np.maximum(a_new, 1e-30)
        a_new = a_new / np.sum(a_new)

        a = a_new
        obj = compute_obj(a)

        if obj < best_obj - 1e-12:
            best_obj = obj
            best_a = a.copy()

        # Decay
        if it % 20 == 19:
            eta *= 0.99

    return best_a, best_obj


def solve_lp_for_g(a, M, time_limit=15):
    """Solve LP: max sum(g) s.t. Toeplitz(a)@g <= M, g >= 0. Returns normalized g."""
    n = len(a)
    A = np.zeros((2 * n - 1, n), dtype=np.float64)
    for j in range(n):
        lo = j
        hi = min(j + n, 2 * n - 1)
        A[lo:hi, j] = a[:hi - lo]
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': time_limit})
        if res.status == 0:
            g = np.maximum(res.x, 0)
            s = float(np.sum(g))
            if s > 1e-10:
                return g / s
    except Exception:
        pass
    return None


def alternating_lp(a_init, max_iters=60, time_budget=40.0):
    """
    LP alternating: fix a, solve LP for optimal g; blend back.
    Also tries symmetric role (fix g, solve for a).
    """
    n = len(a_init)
    a = normalize(a_init.copy().astype(np.float64))
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    stall = 0

    for it in range(max_iters):
        if time.monotonic() - start > time_budget:
            break

        conv_a = compute_conv(a)
        M_a = float(np.max(conv_a))

        # Solve LP: best g given a
        g = solve_lp_for_g(a, M_a, time_limit=12)
        if g is None:
            break

        # Also try symmetric: best a' given g
        a_sym = solve_lp_for_g(g, M_a, time_limit=12)

        # Try blends
        best_try_obj = best_obj
        best_try_a = best_a.copy()

        for t in [0.15, 0.25, 0.4, 0.6, 0.8, 1.0]:
            a_try = (1 - t) * a + t * g
            a_try = normalize(a_try)
            obj_try = compute_obj(a_try)
            if obj_try < best_try_obj:
                best_try_obj = obj_try
                best_try_a = a_try.copy()

        if a_sym is not None:
            for t in [0.3, 0.5, 0.7, 1.0]:
                a_try = (1 - t) * a + t * a_sym
                a_try = normalize(a_try)
                obj_try = compute_obj(a_try)
                if obj_try < best_try_obj:
                    best_try_obj = obj_try
                    best_try_a = a_try.copy()

        if best_try_obj < best_obj - 1e-12:
            best_obj = best_try_obj
            best_a = best_try_a
            a = best_try_a
            stall = 0
        else:
            stall += 1
            if stall > 15:
                rng = np.random.default_rng(it * 7919 + 1)
                a = a * np.exp(rng.normal(0, 0.04, n))
                a = normalize(a)
                stall = 0

    return best_a, best_obj


def soft_max_obj_and_grad(z, tau):
    """Compute soft-max of conv(softmax(z), softmax(z)) and gradient w.r.t. z."""
    n = len(z)
    a = softmax(z)
    conv = compute_conv(a)
    cmax = float(np.max(conv))
    shifted = np.clip((conv - cmax) / tau, -500.0, 500.0)
    exp_c = np.exp(shifted)
    Z = float(np.sum(exp_c))
    soft_max_val = cmax + tau * np.log(Z)
    w = exp_c / Z

    grad_a = compute_grad_from_weights(a, w)
    mean_g = float(np.dot(a, grad_a))
    grad_z = a * (grad_a - mean_g)

    return soft_max_val, grad_z


def optimize_lbfgs(z_init, tau_list, max_iter=200, time_budget=20.0):
    """L-BFGS-B on softmax parameterization with tau annealing."""
    n = len(z_init)
    z = z_init.copy()
    best_a = softmax(z).copy()
    best_obj = compute_obj(best_a)
    start = time.monotonic()

    for tau in tau_list:
        if time.monotonic() - start > time_budget:
            break

        def func_and_grad(z_val):
            return soft_max_obj_and_grad(z_val, tau)

        try:
            result = minimize(func_and_grad, z, jac=True, method='L-BFGS-B',
                              options={'maxiter': max_iter, 'ftol': 1e-15, 'gtol': 1e-9})
            z = result.x
        except Exception:
            continue

        a_new = softmax(z)
        a_new = np.maximum(a_new, 1e-30)
        a_new = a_new / np.sum(a_new)
        obj = compute_obj(a_new)
        if obj < best_obj:
            best_obj = obj
            best_a = a_new.copy()

    return best_a, best_obj


def active_set_qp(a_init, time_budget=20.0, max_iters=2000):
    """
    Active set approach: identify the top peaks, form a quadratic
    approximation, and use L-BFGS to minimize it.
    """
    n = len(a_init)
    a = normalize(a_init.copy().astype(np.float64))
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    K = min(50, 2 * n - 1)

    for it in range(max_iters):
        if time.monotonic() - start > time_budget:
            break

        conv = compute_conv(a)
        # Identify active set: top K peaks
        sorted_idx = np.argsort(conv)[::-1][:K]
        peak_vals = conv[sorted_idx]
        mean_val = float(np.mean(conv))
        excess = np.maximum(peak_vals - mean_val, 1e-30)
        w_peaks = excess / np.sum(excess)

        w_full = np.zeros(2 * n - 1)
        w_full[sorted_idx] = w_peaks

        # The quadratic to minimize: sum_k w[k] * conv[k]
        # = sum_i a[i] * sum_k w[k] * a[k-i]
        # = a^T * (Toeplitz-like) * a
        # Gradient: 2 * sum_k w[k] * a[k-i] (valid)
        grad = compute_grad_from_weights(a, w_full)

        # Use L-BFGS on the quadratic approximation
        def quad_obj_and_grad(a_val):
            a_val = np.maximum(a_val, 0)
            s = np.sum(a_val)
            if s < 1e-15:
                return 0, np.zeros(n)
            a_val = a_val / s
            g = compute_grad_from_weights(a_val, w_full)
            # Center for simplex
            g = g - np.mean(g)
            val = float(np.dot(a_val, compute_grad_from_weights(a_val, w_full) / 2.0))
            return val, g

        # Simple gradient step with line search
        step = 1.0 / n
        for _ in range(10):
            a_new = a - step * (grad - np.mean(grad))
            a_new = np.maximum(a_new, 0)
            s_new = np.sum(a_new)
            if s_new < 1e-15:
                break
            a_new = a_new / s_new
            obj_new = compute_obj(a_new)
            if obj_new < compute_obj(a):
                a = a_new
                break
            step *= 0.5

        obj = compute_obj(a)
        if obj < best_obj - 1e-12:
            best_obj = obj
            best_a = a.copy()

    return best_a, best_obj


def structured_init(n, seed=0):
    """Generate diverse structured initializations."""
    rng = np.random.default_rng(seed)
    k = np.arange(n, dtype=np.float64)
    t = k / max(1, n - 1)
    choice = rng.integers(0, 14)

    if choice == 0:
        # Triangle
        a = 1.0 - np.abs(2.0 * t - 1.0)
    elif choice == 1:
        # Single cosine
        f = rng.uniform(0.5, 3.0)
        a = 1.0 + 0.5 * np.cos(2 * np.pi * f * t)
    elif choice == 2:
        # Two cosines
        f1 = rng.uniform(0.3, 1.5)
        f2 = rng.uniform(2.0, 5.0)
        a = 1.0 + 0.3 * np.cos(2 * np.pi * f1 * t) + 0.2 * np.cos(2 * np.pi * f2 * t)
    elif choice == 3:
        # Asymmetric ramp
        a = np.linspace(0.2, 2.0, n)
    elif choice == 4:
        # Bimodal
        c1 = rng.uniform(0.2, 0.4)
        w1 = rng.uniform(0.05, 0.15)
        a = np.exp(-0.5 * ((t - c1) / w1) ** 2) + np.exp(-0.5 * ((t - (1 - c1)) / w1) ** 2) + 0.1
    elif choice == 5:
        # Power law
        p = rng.uniform(0.3, 3.0)
        a = (1.0 - t) ** p + 0.05
    elif choice == 6:
        # Smoothed random
        a = rng.random(n)
        win = max(5, n // 20)
        kernel = np.ones(win) / win
        a = np.convolve(a, kernel, mode='same')
        a = 0.1 + a / (np.max(a) + 1e-10)
    elif choice == 7:
        # Chirp
        f0 = rng.uniform(0.5, 2.0)
        f1 = rng.uniform(2.0, 5.0)
        a = 1.0 + 0.4 * np.cos(2 * np.pi * (f0 * t + 0.5 * (f1 - f0) * t ** 2))
    elif choice == 8:
        # Step blocks
        nsteps = rng.integers(3, 8)
        vals = rng.uniform(0.3, 2.0, nsteps)
        a = np.repeat(vals, n // nsteps + 1)[:n]
    elif choice == 9:
        # Gaussian bump
        c = rng.uniform(0.3, 0.7)
        w = rng.uniform(0.1, 0.3)
        a = 0.2 + np.exp(-0.5 * ((t - c) / w) ** 2)
    elif choice == 10:
        # Cosine window
        a = 0.5 * (1.0 + np.cos(2 * np.pi * t))
    elif choice == 11:
        # Three-term Fourier
        f1, f2, f3 = rng.uniform(0.5, 8, 3)
        a1, a2, a3 = rng.uniform(0.1, 0.4, 3)
        p1, p2, p3 = rng.uniform(0, 2 * np.pi, 3)
        a = 1.0 + a1 * np.cos(2 * np.pi * f1 * t + p1) + a2 * np.cos(2 * np.pi * f2 * t + p2) + a3 * np.cos(2 * np.pi * f3 * t + p3)
    elif choice == 12:
        # Wedge (V-shape)
        c = rng.uniform(0.3, 0.7)
        a = 0.2 + 1.5 * np.abs(t - c)
    else:
        # Random positive
        a = rng.uniform(0.1, 2.0, n)

    a = np.maximum(a, 0.01)
    return a / np.sum(a)


def resample(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    return np.interp(x_new, x_old, a)


def polish_sequence(a_init, time_budget=60.0):
    """Multi-method polishing cycle."""
    n = len(a_init)
    a = normalize(a_init.copy().astype(np.float64))
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    for rnd in range(30):
        if time.monotonic() - start > time_budget - 5:
            break
        remaining = time_budget - (time.monotonic() - start)
        if remaining < 5:
            break

        # 1. Peak chasing
        t1 = min(remaining * 0.2, 10.0)
        if t1 > 2:
            a_new, obj_new = peak_chasing_nesterov(best_a, time_budget=t1)
            if obj_new < best_obj:
                best_obj = obj_new
                best_a = a_new.copy()

        # 2. Mirror descent
        remaining = time_budget - (time.monotonic() - start)
        if remaining > 8:
            t2 = min(remaining * 0.15, 8.0)
            if t2 > 2:
                a_new, obj_new = mirror_descent(best_a, time_budget=t2)
                if obj_new < best_obj:
                    best_obj = obj_new
                    best_a = a_new.copy()

        # 3. LP alternating (size-limited)
        remaining = time_budget - (time.monotonic() - start)
        if remaining > 12 and n <= 900:
            t3 = min(remaining * 0.25, 18.0)
            a_new, obj_new = alternating_lp(best_a, max_iters=40, time_budget=t3)
            if obj_new < best_obj:
                best_obj = obj_new
                best_a = a_new.copy()

        # 4. L-BFGS-B soft-max
        remaining = time_budget - (time.monotonic() - start)
        if remaining > 8:
            t4 = min(remaining * 0.2, 10.0)
            z = np.log(np.maximum(best_a, 1e-30))
            tau_list = [1.0 / n, 0.5 / n, 0.2 / n, 0.1 / n, 0.05 / n]
            a_new, obj_new = optimize_lbfgs(z, tau_list, max_iter=150, time_budget=t4)
            if obj_new < best_obj:
                best_obj = obj_new
                best_a = a_new.copy()

        # 5. Active set
        remaining = time_budget - (time.monotonic() - start)
        if remaining > 5:
            t5 = min(remaining * 0.1, 5.0)
            if t5 > 1:
                a_new, obj_new = active_set_qp(best_a, time_budget=t5)
                if obj_new < best_obj:
                    best_obj = obj_new
                    best_a = a_new.copy()

    return best_a, best_obj


def try_parent():
    """Try to load parent sequence."""
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64)
        parent = np.clip(parent, 0, 1000.0)
        if parent.size > 0 and np.isfinite(parent).all():
            obj = compute_obj(parent)
            print(f"Parent (n={parent.size}): obj={obj:.6f}")
            return parent.copy(), obj
    except Exception:
        pass
    return None, np.inf


def propose_candidate(seed=42, budget_s=1500):
    """
    Main search: multi-strategy optimization combining peak chasing,
    mirror descent, LP alternating, soft-max L-BFGS-B, and active-set QP.
    Searches over multiple sequence lengths with diverse initializations.
    """
    start_time = time.monotonic()
    deadline = start_time + float(budget_s) - 30.0

    best_seq = None
    best_score = np.inf

    # Load parent
    parent, parent_obj = try_parent()
    if parent is not None and parent_obj < best_score:
        best_score = parent_obj
        best_seq = parent.copy()

    # N values to explore
    n_values = [200, 300, 400, 500, 600, 700, 800, 1000, 1200, 1500]
    if best_seq is not None:
        np_ = len(best_seq)
        extra = [np_, max(100, np_ - 100), min(2000, np_ + 100),
                 min(2000, np_ + 300), min(2000, np_ + 500)]
        n_values = sorted(set(n_values + extra))

    print(f"Searching n values: {n_values}")
    print(f"Initial best: {best_score:.6f}")

    candidates = []

    for n in n_values:
        if time.monotonic() > deadline - 80:
            break
        remaining = deadline - time.monotonic()
        if remaining < 20:
            break

        # Time allocation per n
        if n <= 400:
            alloc = min(50.0, remaining * 0.12)
        elif n <= 800:
            alloc = min(70.0, remaining * 0.13)
        elif n <= 1200:
            alloc = min(90.0, remaining * 0.13)
        else:
            alloc = min(110.0, remaining * 0.14)
        alloc = max(20.0, alloc)

        print(f"\nN={n}: {remaining:.0f}s remaining, alloc={alloc:.0f}s")

        n_inits = max(2, int(alloc / 25))
        n_inits = min(n_inits, 4)

        for init_idx in range(n_inits):
            if time.monotonic() > deadline - 80:
                break
            remaining_init = deadline - time.monotonic()
            if remaining_init < 15:
                break

            # Generate initial sequence
            if init_idx == 0 and best_seq is not None and abs(len(best_seq) - n) <= 200:
                a = resample(best_seq, n)
            else:
                a = structured_init(n, seed + n * 7919 + init_idx * 31627)

            # Phase 1: Peak chasing (fast shaping)
            pc_time = min(remaining_init * 0.15, 10.0)
            a, obj = peak_chasing_nesterov(a, time_budget=pc_time)

            # Phase 2: Mirror descent
            remaining_init = deadline - time.monotonic()
            if remaining_init > 12:
                md_time = min(remaining_init * 0.15, 10.0)
                a, obj = mirror_descent(a, time_budget=md_time)

            # Phase 3: LP alternating (size-limited)
            remaining_init = deadline - time.monotonic()
            if remaining_init > 20 and n <= 1000:
                lp_time = min(remaining_init * 0.25, 20.0)
                a, obj = alternating_lp(a, max_iters=50, time_budget=lp_time)

            # Phase 4: L-BFGS-B soft-max polish
            remaining_init = deadline - time.monotonic()
            if remaining_init > 8:
                lbfgs_time = min(remaining_init * 0.2, 12.0)
                z = np.log(np.maximum(a, 1e-30))
                tau_list = [2.0 / n, 1.0 / n, 0.5 / n, 0.2 / n, 0.1 / n]
                a, obj = optimize_lbfgs(z, tau_list, max_iter=200, time_budget=lbfgs_time)

            # Phase 5: Active set
            remaining_init = deadline - time.monotonic()
            if remaining_init > 5:
                aq_time = min(remaining_init * 0.1, 5.0)
                a, obj = active_set_qp(a, time_budget=aq_time)

            print(f"  init{init_idx}: n={n}, obj={obj:.6f}")

            if obj < best_score:
                best_score = obj
                best_seq = a.copy()
                print(f"  ** UPDATE: n={n}, obj={best_score:.6f} **")

            candidates.append((a.copy(), obj, n))

    # Polish phase
    print(f"\n=== POLISHING ===")
    print(f"Best before polish: {best_score:.6f}")

    candidates.sort(key=lambda x: x[1])
    n_polish = min(4, len(candidates))

    for ci in range(n_polish):
        if time.monotonic() > deadline - 50:
            break
        remaining = deadline - time.monotonic()
        if remaining < 20:
            break

        a_cand = candidates[ci][0]
        n_cand = candidates[ci][2]
        print(f"  Polishing cand {ci} (n={n_cand}, obj={candidates[ci][1]:.6f}), {remaining:.0f}s left")

        polish_budget = min(remaining * 0.35, 70.0)
        a_pol, obj_pol = polish_sequence(a_cand, time_budget=polish_budget)

        if obj_pol < best_score:
            best_score = obj_pol
            best_seq = a_pol.copy()
            print(f"  ** UPDATE after polish: obj={best_score:.6f} **")

        # Perturbation + re-optimize
        remaining = deadline - time.monotonic()
        if remaining > 25 and n_cand <= 1000:
            rng = np.random.default_rng(ci * 7919 + 42)
            for ptry in range(3):
                if time.monotonic() > deadline - 25:
                    break
                scale = rng.uniform(0.02, 0.08)
                a_pert = a_pol * np.exp(rng.normal(0, scale, n_cand))
                a_pert = normalize(a_pert)
                # Quick optimization
                a_pert, obj_pert = peak_chasing_nesterov(a_pert, time_budget=5.0)
                remaining = deadline - time.monotonic()
                if remaining > 10 and n_cand <= 800:
                    a_pert, obj_pert = alternating_lp(a_pert, max_iters=20, time_budget=min(remaining * 0.3, 10.0))
                if obj_pert < best_score:
                    best_score = obj_pert
                    best_seq = a_pert.copy()
                    print(f"  ** UPDATE (perturb {ptry}): obj={best_score:.6f} **")

    # Final intensive polish on global best
    if best_seq is not None and time.monotonic() < deadline - 15:
        remaining = deadline - time.monotonic()
        n_best = len(best_seq)
        print(f"\n=== FINAL POLISH: n={n_best}, {remaining:.0f}s ===")

        a_cur = best_seq.copy()
        for fr in range(6):
            if time.monotonic() > deadline - 10:
                break
            remaining = deadline - time.monotonic()
            if remaining < 12:
                break

            # Peak chasing
            t = min(remaining * 0.15, 12.0)
            a_new, obj_new = peak_chasing_nesterov(a_cur, time_budget=t)
            if obj_new < best_score:
                best_score = obj_new
                best_seq = a_new.copy()
                a_cur = a_new.copy()
                print(f"  Final PC {fr}: {best_score:.6f}")

            # Mirror descent
            remaining = deadline - time.monotonic()
            if remaining > 10:
                t = min(remaining * 0.12, 8.0)
                a_new, obj_new = mirror_descent(a_cur, time_budget=t)
                if obj_new < best_score:
                    best_score = obj_new
                    best_seq = a_new.copy()
                    a_cur = a_new.copy()
                    print(f"  Final MD {fr}: {best_score:.6f}")

            # LP
            remaining = deadline - time.monotonic()
            if remaining > 15 and n_best <= 800:
                t = min(remaining * 0.2, 15.0)
                a_new, obj_new = alternating_lp(a_cur, max_iters=30, time_budget=t)
                if obj_new < best_score:
                    best_score = obj_new
                    best_seq = a_new.copy()
                    a_cur = a_new.copy()
                    print(f"  Final LP {fr}: {best_score:.6f}")

            # L-BFGS
            remaining = deadline - time.monotonic()
            if remaining > 8:
                t = min(remaining * 0.15, 10.0)
                z = np.log(np.maximum(a_cur, 1e-30))
                tau_list = [0.3 / n_best, 0.1 / n_best, 0.05 / n_best, 0.02 / n_best]
                a_new, obj_new = optimize_lbfgs(z, tau_list, max_iter=200, time_budget=t)
                if obj_new < best_score:
                    best_score = obj_new
                    best_seq = a_new.copy()
                    a_cur = a_new.copy()
                    print(f"  Final LBFGS {fr}: {best_score:.6f}")

    # Validation
    if best_seq is None:
        best_seq = np.ones(200)
        best_score = compute_obj(best_seq)

    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    val = evaluate_sequence(result)
    if val < np.inf:
        best_score = val
    else:
        result = [1.0] * 100
        best_score = 2.0

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_score:.6f}, time={elapsed:.1f}s ===")
    return result
