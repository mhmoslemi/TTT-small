Hybrid optimizer combining smooth-max gradient descent (Adam), LP mirror descent,
and multi-start exploration for minimizing J(a) = 2*n*max(conv(a,a))/sum(a)^2.
"""

import numpy as np
from scipy.optimize import linprog
from scipy.sparse import csr_matrix
import time


def fast_conv(a):
    """Full self-convolution via FFT."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]


def compute_J(a):
    """Compute J(a) = 2*n*max(conv(a,a))/(sum(a))^2."""
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))


def smooth_max_grad(a, tau):
    """
    Compute gradient and value of (1/tau)*log(sum_k exp(tau*c[k]))
    where c = conv(a,a). Returns (grad_a, smooth_max_value, p).
    """
    n = len(a)
    c = fast_conv(a)
    c_max_val = float(c.max())
    exp_c = np.exp(tau * (c - c_max_val))
    p = exp_c / exp_c.sum()
    smooth_val = (c_max_val + np.log(exp_c.sum())) / tau
    
    # grad_a[i] = 2 * sum_j a[j] * p[i+j] for valid j (0 <= j < n, i+j < 2n-1)
    grad = np.zeros(n)
    # Use matrix-free approach: for each i, grad[i] = 2 * dot(a[:n], p[i:i+n]) (with bounds)
    # Actually p has length 2n-1, so p[i:i+n] might go out of bounds
    # Let's use a convolution-like approach
    # grad[i] = 2 * sum_{j=0}^{n-1} a[j] * p[i+j] where i+j < 2n-1
    # This is 2 * (cross-correlation of a and p at lag i)
    
    # Efficient computation using FFT:
    # Define q as p reversed and zero-padded to length 2n-1
    # Then cross-correlation can be done via FFT
    # Actually, let's just do it directly for n <= 2048
    for i in range(n):
        j_end = min(n, 2 * n - 1 - i)
        if j_end > 0:
            grad[i] = 2.0 * np.dot(a[:j_end], p[i:i + j_end])
    
    return grad, p, c_max_val


def build_A_sparse(n, a):
    """Build sparse matrix A where A[k,j] = a[k-j] for valid j."""
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        j_min = max(0, k - n + 1)
        j_max = min(n - 1, k)
        for j in range(j_min, j_max + 1):
            rows.append(k)
            cols.append(j)
            vals.append(a[k - j])
    return csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))


def lp_best_response(a, M):
    """Solve LP: max sum(g) s.t. A@g <= M*1, g >= 0."""
    n = len(a)
    A = build_A_sparse(n, a)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    try:
        r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
        if r.success and r.x is not None:
            g = np.maximum(r.x, 0)
            if g.sum() > 1e-15:
                return g
    except Exception:
        pass
    return None


def adam_smooth_max(a, max_iters, deadline, tau_start=5.0, tau_end=50.0, lr=0.02):
    """Adam optimization on smooth max of self-convolution with log parameterization."""
    n = len(a)
    a = np.maximum(a, 1e-10)
    a = a / a.sum()
    
    x = np.log(a)  # Log parameterization
    m = np.zeros(n)
    v = np.zeros(n)
    beta1, beta2 = 0.85, 0.999
    eps = 1e-8
    
    best_J = compute_J(a)
    best_a = a.copy()
    no_improve = 0
    
    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        
        # Current a
        x_shift = x - x.max()
        a = np.exp(x_shift)
        a = a / a.sum()
        
        # Adaptive tau
        tau = tau_start + (tau_end - tau_start) * min(1.0, it / (max_iters * 0.7))
        
        # Gradient
        grad_a, p, c_max = smooth_max_grad(a, tau)
        
        # Convert to x-space gradient
        grad_x = a * (grad_a - np.dot(grad_a, a))
        
        # Adam update
        m = beta1 * m + (1 - beta1) * grad_x
        v = beta2 * v + (1 - beta2) * grad_x ** 2
        m_hat = m / (1 - beta1 ** (it + 1))
        v_hat = v / (1 - beta2 ** (it + 1))
        
        # Adaptive learning rate
        lr_eff = lr / (1 + it * 0.01)
        x = x - lr_eff * m_hat / (np.sqrt(v_hat) + eps)
        
        # Compute J
        a_new = np.exp(x - x.max())
        a_new = a_new / a_new.sum()
        J_new = compute_J(a_new)
        
        if J_new < best_J - 1e-12:
            best_J = J_new
            best_a = a_new.copy()
            no_improve = 0
        else:
            no_improve += 1
            # If stuck, increase tau or reduce learning rate
            if no_improve > 50:
                lr *= 0.7
                if lr < 1e-6:
                    break
        
    return best_a, best_J


def lp_mirror_descent(a, max_iters, deadline):
    """LP mirror descent with adaptive line search."""
    a = np.maximum(a.astype(np.float64), 1e-10)
    a = a / a.sum()
    current_J = compute_J(a)
    best_J = current_J
    best_a = a.copy()
    no_improve = 0
    
    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        
        b = fast_conv(a)
        M = float(b.max())
        g = lp_best_response(a, M)
        if g is None:
            break
        g = np.maximum(g, 0.0)
        g_sum = g.sum()
        if g_sum < 1e-10:
            break
        g = g / g_sum
        
        base_t = 0.3 * np.exp(-0.2 * it / 100.0) + 0.05
        t = min(1.0, base_t)
        
        best_trial_J = current_J
        best_trial_a = None
        
        for ls in range(20):
            a_trial = (1.0 - t) * a + t * g
            a_trial = np.maximum(a_trial, 0.0)
            s = a_trial.sum()
            if s < 1e-10:
                t *= 0.5
                continue
            a_trial = a_trial / s
            J_trial = compute_J(a_trial)
            if J_trial < best_trial_J - 1e-14:
                best_trial_J = J_trial
                best_trial_a = a_trial.copy()
                break
            t *= 0.65
            if t < 0.002:
                break
        
        if best_trial_a is not None:
            a = best_trial_a
            current_J = best_trial_J
            if current_J < best_J:
                best_J = current_J
                best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1
            if no_improve > 30:
                break
    
    return best_a, best_J


def generate_diverse_inits(n, rng, parent=None):
    """Generate diverse initial sequences."""
    inits = []
    t = np.arange(n, dtype=np.float64) / n
    
    if parent is not None:

    inits.append(0.5 * (1 - np.cos(2 * np.pi * t)))

    inits.append(np.exp(-4 * (t - 0.5) ** 2) + 0.05)

    # Various frequencies
    for f in [1, 2, 3, 4, 5, 7, 11, 13]:
        for amp in [0.2, 0.4, 0.6]:

    for _ in range(5):

    # Smooth random
    k = max(3, n // 12)

    for _ in range(3):
        smooth = np.convolve(rng.uniform(0, 1, n), kernel, mode='same')
        inits.append(np.maximum(smooth + 0.1, 0.01))
    
    # Random with structure

    result = []
    for a in inits:
        a = np.maximum(a, 0.001).astype(np.float64)
        s = a.sum()

def resample_seq(src, dest_n):
    """Resample 1D sequence to new length."""
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(1, dest_n, dest_n)  # Avoid endpoints
    return np.maximum(np.interp(t_new, t_old, src), 0.0)

def subgradient_polish(a, max_iters, deadline):
    """Subgradient descent with momentum on the max convolution."""
    a = np.maximum(a.astype(np.float64), 1e-10)
    a = a / a.sum()
    best_M = float(fast_conv(a).max())
    best_a = a.copy()
    momentum = np.zeros_like(a)
    beta = 0.8
    n = len(a)
    
    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        
        b = fast_conv(a)
        M = float(b.max())
        tol = 1e-10 * M + 1e-15
        active = np.where(b >= M - tol)[0]
        
        grad = np.zeros(n)
        for k in active:
            lo = max(0, k - n + 1)
            hi = min(n - 1, k)
            for i in range(lo, hi + 1):
                grad[i] += 2.0 * a[k - i]
        
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-15:
            break
        
        t = 1.0 / (gnorm * np.sqrt(it + 1.0))
        momentum = beta * momentum + (1 - beta) * grad / gnorm
        a_new = a - t * momentum
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s < 1e-10:
            break
        a_new = a_new / s
        
        new_M = float(fast_conv(a_new).max())
        if new_M < best_M:
            best_M = new_M
            best_a = a_new.copy()
        a = a_new
    
    return best_a

def fast_conv(a):
    """Full self-convolution via FFT."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]


def compute_J(a):
    """Compute J(a) = 2*n*max(conv(a,a))/(sum(a))^2."""
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))


def smooth_max_grad(a, tau):
    """
    Compute gradient of (1/tau)*log(sum_k exp(tau*c[k])) where c=conv(a,a).
    Returns (grad, p, c_max).
    """
    n = len(a)
    c = fast_conv(a)
    c_max_val = float(c.max())
    exp_c = np.exp(tau * (c - c_max_val))
    p = exp_c / exp_c.sum()

    grad = np.zeros(n)
    for i in range(n):
        j_end = min(n, 2 * n - 1 - i)
        if j_end > 0:
            grad[i] = 2.0 * np.dot(a[:j_end], p[i:i + j_end])

    return grad, p, c_max_val


def build_A_sparse(n, a):
    """Build sparse matrix A where A[k,j] = a[k-j] for valid j."""
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        j_min = max(0, k - n + 1)
        j_max = min(n - 1, k)
        for j in range(j_min, j_max + 1):
            rows.append(k)
            cols.append(j)
            vals.append(a[k - j])
    return csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))


def lp_best_response(a, M):
    """Solve LP: max sum(g) s.t. A@g <= M*1, g >= 0."""
    n = len(a)
    A = build_A_sparse(n, a)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    try:
        r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
        if r.success and r.x is not None:
            g = np.maximum(r.x, 0)
            if g.sum() > 1e-15:
                return g
    except Exception:
        pass
    return None


def lp_min_peak(a):
    """Solve LP: min t s.t. A@g <= t, sum(g)=1, g>=0."""
    n = len(a)
    A = build_A_sparse(n, a)
    # Variables: g[0..n-1], t. Minimize t.
    # Constraints: A@g - t <= 0 (2n-1 constraints)
    # sum(g) = 1
    # g >= 0, t free
    c_obj = np.zeros(n + 1)
    c_obj[n] = 1.0  # minimize t

    # A_ub: [A, -I_{2n-1}] @ [g; t] <= 0
    eye_neg = -np.eye(2 * n - 1)
    A_full = np.hstack([np.asarray(A.todense()), eye_neg])

    # Equality: sum(g) = 1
    from scipy.sparse import spmatrix
    Ae = np.zeros((1, n + 1))
    Ae[0, :n] = 1.0
    be = [1.0]

    bounds = [(0, None)] * n + [(None, None)]
    try:

def adam_smooth_max(a, max_iters, deadline, tau_start=5.0, tau_end=80.0, lr=0.015):

    m = np.zeros(n)
    v = np.zeros(n)

        v_hat = v / (1 - beta2 ** (it + 1))

def topk_sum_grad(a, K):

def generate_diverse_inits(n, rng, parent=None):

    # Multi-harmonic
    for _ in range(6):
        n_harm = min(6, n // 4)

def resample_seq(src, dest_n):
    """Resample 1D sequence to new length."""
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0.5 / dest_n, 1 - 0.5 / dest_n, dest_n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)

def subgradient_polish(a, max_iters, deadline):
    """Subgradient descent with momentum."""
    a = np.maximum(a.astype(np.float64), 1e-10)
    a = a / a.sum()
    best_M = float(fast_conv(a).max())
    best_a = a.copy()
    momentum = np.zeros_like(a)
    beta = 0.8
    n = len(a)

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        b = fast_conv(a)
        M = float(b.max())
        tol = 1e-10 * M + 1e-15
        active = np.where(b >= M - tol)[0]

        grad = np.zeros(n)
        for k in active:
            lo = max(0, k - n + 1)
            hi = min(n - 1, k)
            for i in range(lo, hi + 1):
                grad[i] += 2.0 * a[k - i]

        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-15:
            break

        t = 1.0 / (gnorm * np.sqrt(it + 1.0))
        momentum = beta * momentum + (1 - beta) * grad / gnorm
        a_new = a - t * momentum
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s < 1e-10:
            break
        a_new = a_new / s

        new_M = float(fast_conv(a_new).max())
        if new_M < best_M:
            best_M = new_M
            best_a = a_new.copy()
        a = a_new

    return best_a


def coordinate_descent_polish(a, max_iters, deadline):
    """Coordinate descent on individual elements with random ordering."""
    n = len(a)
    a = np.maximum(a.astype(np.float64), 1e-10)
    a = a / a.sum()
    best_J = compute_J(a)
    best_a = a.copy()
    rng = np.random.default_rng(123)

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        # Randomly permute coordinates
        idx = rng.permutation(n)

        # Process in chunks to time-check
        chunk_size = 50
        for start in range(0, n, chunk_size):
            if time.monotonic() > deadline:
                break
            for i in idx[start:start + chunk_size]:
                a_orig = a[i]
                if a_orig < 1e-8:
                    # Try setting to small positive
                    for scale in [0.001, 0.01]:
                        a_trial = a.copy()
                        a_trial[i] = scale * a.mean()
                        s = a_trial.sum()
                        if s > 0.01:
                            a_trial = a_trial / s
                            J_trial = compute_J(a_trial)
                            if J_trial < best_J - 1e-12:
                                a = a_trial
                                best_J = J_trial
                                best_a = a.copy()
                                break
                else:
                    # Try scaling
                    for scale in [0.5, 0.75, 1.25, 1.5]:
                        a_trial = a.copy()
                        a_trial[i] = a_orig * scale
                        s = a_trial.sum()
                        if s < 0.01:
                            continue
                        a_trial = a_trial / s
                        J_trial = compute_J(a_trial)
                        if J_trial < best_J - 1e-12:
                            a = a_trial
                            best_J = J_trial
                            best_a = a.copy()
                            break

    return best_a, best_J


def structured_perturb(a, rng, ptype):
    """Apply structured perturbation."""
    n = len(a)
    t = np.arange(n, dtype=np.float64) / n

    if ptype == 'shift':
        s = rng.integers(1, max(2, n // 8))
        a_p = np.roll(a, s)
    elif ptype == 'modulate':
        freq = rng.uniform(0.5, 6.0)
        amp = rng.uniform(0.03, 0.12)
        phase = rng.uniform(0, 2 * np.pi)
        a_p = a * (1 + amp * np.cos(2 * np.pi * freq * t + phase))
    elif ptype == 'smooth':
        k = rng.integers(3, max(4, n // 8))
        kernel = np.ones(k) / k
        a_p = np.convolve(a, kernel, mode='same')
    elif ptype == 'taper':
        alpha = rng.uniform(0.6, 0.95)
        window = np.ones(n)
        edge = max(1, int(n * (1 - alpha) / 2))
        for i in range(edge):
            frac = (i + 1) / (edge + 1)
            window[i] = frac
            window[n - 1 - i] = frac
        a_p = a * window
    elif ptype == 'bump':
        center = rng.uniform(0.2, 0.8)
        width = rng.uniform(0.05, 0.3)
        a_p = a * (1 + 0.1 * np.exp(-((t - center) ** 2) / (2 * width ** 2)))
    else:
        a_p = a + rng.normal(0, 0.02, n)

    a_p = np.maximum(a_p, 0.001)
    s = a_p.sum()
    if s > 1e-8:
        a_p = a_p / s
    return a_p


def propose_candidate(seed=42, budget_s=1500, **kwargs):
    """
    Hybrid optimizer: smooth-max Adam, top-K smooth optimization, LP mirror descent,
    coordinate descent, multi-start exploration with diverse initializations,
    structured perturbations, and subgradient polishing.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + float(budget_s) - 30.0
    tracker = ScoreTracker()

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        if p.ndim == 1 and p.size > 0 and np.all(np.isfinite(p)):
            parent = np.maximum(p, 0.0)
            tracker.update(parent)
            print(f"Parent: n={parent.size}, J={tracker.best_score:.6f}", flush=True)
    except (NameError, AttributeError):
        pass

    if parent is None:
        parent = np.ones(512) / 512
        tracker.update(parent)

    current_best = parent.copy()
    current_best_n = current_best.size
    current_best_J = compute_J(current_best)

    # === Phase 1: Multi-start exploration with Adam ===
    t0 = time.monotonic()
    print(f"\n{'='*50}", flush=True)
    print(f"Phase 1: Multi-start exploration", flush=True)
    print(f"{'='*50}", flush=True)

    lengths = sorted(set([
        current_best_n,
        current_best_n + 64,
        current_best_n + 128,
        max(256, current_best_n - 64),
    ]))
    lengths = [l for l in lengths if 256 <= l <= 2048]

    all_candidates = []
    time_budget_phase1 = min(500, (deadline - time.monotonic()) * 0.45)
    time_per_length = max(30, time_budget_phase1 / len(lengths))

    for n_try in lengths:
        if time.monotonic() > deadline - 600:
            break

        t_len = time.monotonic()
        inits = generate_diverse_inits(n_try, rng, parent)

        # Quick score all inits
        scored = []
        for a in inits:
            J = compute_J(a)
            scored.append((J, a))
        scored.sort(key=lambda x: x[0])

        print(f"  n={n_try}: {len(inits)} inits, best J={scored[0][0]:.6f}", flush=True)

        # Run Adam on top 4
        top_k = min(4, len(scored))
        time_per_candidate = time_per_length / (top_k + 1)

        for rank in range(top_k):
            if time.monotonic() > deadline - 500:
                break
            a_init = scored[rank][1]
            dl = time.monotonic() + time_per_candidate
            a_opt, J_opt = adam_smooth_max(a_init, 300, dl,
                                           tau_start=5.0, tau_end=60.0, lr=0.015)
            all_candidates.append((J_opt, a_opt))
            tracker.update(a_opt)
            if J_opt < current_best_J:
                current_best = a_opt.copy()
                current_best_n = a_opt.size
                current_best_J = J_opt

        # Also try topk smooth objective on best init
        if time.monotonic() < deadline - 500:
            a_init2 = scored[0][1]
            dl2 = time.monotonic() + time_per_candidate
            a_opt2, J_opt2 = topk_adam(a_init2, 200, dl2, K=3,
                                       tau_start=3.0, tau_end=20.0, lr=0.02)
            all_candidates.append((J_opt2, a_opt2))
            tracker.update(a_opt2)
            if J_opt2 < current_best_J:
                current_best = a_opt2.copy()
                current_best_n = a_opt2.size
                current_best_J = J_opt2

        print(f"  n={n_try} done ({time.monotonic()-t_len:.1f}s), "
              f"current best J={current_best_J:.8f}", flush=True)

    all_candidates.sort(key=lambda x: x[0])
    print(f"Phase 1 complete. Best: J={all_candidates[0][0]:.8f} "
          f"({time.monotonic()-t0:.0f}s)", flush=True)

    # === Phase 2: LP mirror descent on top candidates ===
    if time.monotonic() < deadline - 300:
        t0 = time.monotonic()
        print(f"\n{'='*50}", flush=True)
        print(f"Phase 2: LP mirror descent on top candidates", flush=True)
        print(f"{'='*50}", flush=True)

        n_lp = min(5, len(all_candidates))
        if n_lp == 0:
            n_lp = 1
            all_candidates = [(current_best_J, current_best.copy())]

        # Deduplicate by length
        seen_lengths = set()
        lp_candidates = []
        for J_c, a_c in all_candidates:
            if a_c.size not in seen_lengths or len(lp_candidates) < 2:
                seen_lengths.add(a_c.size)
                lp_candidates.append((J_c, a_c))
            if len(lp_candidates) >= n_lp:
                break

        time_per_lp = max(20, min(100, (deadline - 300 - time.monotonic()) / len(lp_candidates)))

        for i, (J_c, a_c) in enumerate(lp_candidates):
            if time.monotonic() > deadline - 250:
                break
            a_c = np.maximum(a_c, 1e-10)
            a_c = a_c / a_c.sum()
            a_lp, J_lp = lp_mirror_descent(a_c, 800, time.monotonic() + time_per_lp)
            tracker.update(a_lp)
            if J_lp < current_best_J:
                current_best = a_lp.copy()
                current_best_n = a_lp.size
                current_best_J = J_lp
                print(f"  LP cand {i} (n={a_lp.size}): J={J_lp:.8f}", flush=True)

        print(f"Phase 2 done ({time.monotonic()-t0:.1f}s), "
              f"current best J={current_best_J:.8f}", flush=True)

    # === Phase 3: Extended Adam on best ===
    if time.monotonic() < deadline - 200:
        t0 = time.monotonic()
        print(f"\n{'='*50}", flush=True)
        print(f"Phase 3: Extended Adam (n={current_best_n})", flush=True)
        print(f"{'='*50}", flush=True)

        a_ext = np.maximum(current_best, 1e-10)
        a_ext = a_ext / a_ext.sum()
        a_ext, J_ext = adam_smooth_max(a_ext, 3000, min(deadline - 150, time.monotonic() + 250),
                                       tau_start=10.0, tau_end=150.0, lr=0.008)
        tracker.update(a_ext)
        if J_ext < current_best_J:
            current_best = a_ext.copy()
            current_best_n = a_ext.size
            current_best_J = J_ext
            print(f"  Adam: J={J_ext:.8f}", flush=True)
        print(f"Phase 3 done ({time.monotonic()-t0:.1f}s)", flush=True)

    # === Phase 4: LP refinement (linear) ===
    if time.monotonic() < deadline - 120:
        t0 = time.monotonic()
        print(f"\n{'='*50}", flush=True)
        print(f"Phase 4: Long LP linear (n={current_best_n})", flush=True)
        print(f"{'='*50}", flush=True)

        a_lp2 = np.maximum(current_best, 1e-10)
        a_lp2 = a_lp2 / a_lp2.sum()
        a_lp2, J_lp2 = lp_mirror_descent(a_lp2, 10000, deadline - 80, use_multiplicative=False)
        tracker.update(a_lp2)
        if J_lp2 < current_best_J:
            current_best = a_lp2.copy()
            current_best_n = a_lp2.size
            current_best_J = J_lp2
            print(f"  LP: J={J_lp2:.8f}", flush=True)
        print(f"Phase 4 done ({time.monotonic()-t0:.1f}s)", flush=True)

    # === Phase 5: LP multiplicative ===
    if time.monotonic() < deadline - 40:
        t0 = time.monotonic()
        print(f"\n{'='*50}", flush=True)
        print(f"Phase 5: LP multiplicative (n={current_best_n})", flush=True)
        print(f"{'='*50}", flush=True)

        a_lp3 = np.maximum(current_best, 1e-10)
        a_lp3 = a_lp3 / a_lp3.sum()
        a_lp3, J_lp3 = lp_mirror_descent(a_lp3, 2000, deadline - 20, use_multiplicative=True)
        tracker.update(a_lp3)
        if J_lp3 < current_best_J:
            current_best = a_lp3.copy()
            current_best_n = a_lp3.size
            current_best_J = J_lp3
            print(f"  LP mult: J={J_lp3:.8f}", flush=True)
        print(f"Phase 5 done ({time.monotonic()-t0:.1f}s)", flush=True)

    # === Phase 6: Structured perturbations + LP ===
    if time.monotonic() < deadline - 80:
        t0 = time.monotonic()
        print(f"\n{'='*50}", flush=True)
        print(f"Phase 6: Perturbations + LP (n={current_best_n})", flush=True)
        print(f"{'='*50}", flush=True)

        ptypes = ['shift', 'modulate', 'smooth', 'taper', 'bump', 'modulate', 'shift', 'smooth']
        remaining = deadline - 80 - time.monotonic()
        per_pert = max(10, min(40, remaining / len(ptypes)))

        for ptype in ptypes:
            if time.monotonic() > deadline - 60:
                break
            a_pert = structured_perturb(current_best, rng, ptype)
            a_pert, J_pert = lp_mirror_descent(a_pert, 200, time.monotonic() + per_pert)
            tracker.update(a_pert)
            if J_pert < current_best_J:
                current_best = a_pert.copy()
                current_best_n = a_pert.size
                current_best_J = J_pert
                print(f"  [{ptype}]: J={J_pert:.8f}", flush=True)

        print(f"Phase 6 done ({time.monotonic()-t0:.1f}s)", flush=True)

    # === Phase 7: Coordinate descent ===
    if time.monotonic() < deadline - 30:
        t0 = time.monotonic()
        print(f"\n{'='*50}", flush=True)
        print(f"Phase 7: Coordinate descent (n={current_best_n})", flush=True)
        print(f"{'='*50}", flush=True)

        a_cd, J_cd = coordinate_descent_polish(current_best, 50000, deadline - 15)
        tracker.update(a_cd)
        if J_cd < current_best_J:
            current_best = a_cd.copy()
            current_best_J = J_cd
            print(f"  CD: J={J_cd:.8f}", flush=True)
        print(f"Phase 7 done ({time.monotonic()-t0:.1f}s)", flush=True)

    # === Phase 8: Subgradient ===
    if time.monotonic() < deadline - 5:
        t0 = time.monotonic()
        print(f"\n{'='*50}", flush=True)
        print(f"Phase 8: Subgradient (n={current_best_n})", flush=True)
        print(f"{'='*50}", flush=True)

        a_sub = subgradient_polish(current_best, 5000, deadline - 3)
        tracker.update(a_sub)
        J_sub = compute_J(a_sub)
        if J_sub < current_best_J:
            current_best = a_sub.copy()
            current_best_J = J_sub
            print(f"  SG: J={J_sub:.8f}", flush=True)
        print(f"Phase 8 done ({time.monotonic()-t0:.1f}s)", flush=True)

    # Final
    final_a = tracker.best if tracker.best is not None else current_best
    print(f"\n{'='*50}", flush=True)
    print(f"FINAL: J={tracker.best_score:.8f} n={final_a.size} "
          f"time={time.monotonic()-tracker.t_start:.0f}s", flush=True)
    print(f"{'='*50}", flush=True)

    return [float(x) for x in final_a]
