                best_score = config['evaluator'](best_sequence.tolist())
                print(f"Gradient descent: J={best_score:.6f}")
                
        # Fallback if the best_sequence was not updated significantly in the time window.
        if best_sequence is None:
            best_sequence = np.ones(320) / 320.0
            best_score = config['evaluator'](best_sequence.tolist())

    # Fallback: If no candidate was found, return benchmark.
    if best_sequence is None:
        best_sequence = benchmark_sample
    else:
        best_sequence = best_sequence / best_sequence.sum()

    return best_sequence.tolist()
