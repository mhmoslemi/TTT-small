def fast_conv(a):

    n = len(a)
    s = 1
    while s < 2 * n - 1:
        s <<= 1
    fa = np.fft.rfft(a, s)
    return np.fft.irfft(fa * fa, s)[:2 * n - 1]

    """Compute J(a) = 2*n*max(conv(a,a))/(sum(a))^2."""
    a = np.maximum(a, 0)
    s = float(a.sum())
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))


def get_A_structure(n):
    """Get or pre-compute the (row, col) arrays for the constraint matrix."""
    if n not in _A_structure_cache:
        rows = []
        cols = []
        for k in range(2 * n - 1):
            jl = max(0, k - n + 1)
            jh = min(n - 1, k)
            for j in range(jl, jh + 1):
                rows.append(k)
                cols.append(j)
        _A_structure_cache[n] = (np.array(rows, dtype=np.int32),
                                  np.array(cols, dtype=np.int32))
    return _A_structure_cache[n]

    """Build sparse constraint matrix."""
    n = len(a)
    rows_arr, cols_arr = get_A_structure(n)
    vals = a[rows_arr - cols_arr]
    return csr_matrix((vals, (rows_arr, cols_arr)), shape=(2 * n - 1, n))


def lp_best_response(a, M):
    """LP: max sum(g) s.t. (a*g)_k <= M for all k, g >= 0. Returns normalized g or None."""
    n = len(a)
    c = -np.ones(n)
    A = build_A_sparse(a)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    try:
        r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
        if r.success and r.x is not None:
            g = np.maximum(r.x, 0.0)
            g_sum = g.sum()
            if g_sum > 1e-10:
                return g / g_sum
    except Exception:
        pass
    return None

def smooth_J_and_grad(a, beta):
    """Compute smoothed J and its gradient."""

    s = float(a.sum())

    grad_J = (2.0 * n * grad_sm / (s * s)) - (4.0 * n * sm_val / (s ** 3))
    return J_val, grad_J


def fista_optimize(a0, betas, deadline, max_iters_per_beta=100):
    """FISTA (accelerated proximal gradient) on the smoothed objective."""
    a = np.maximum(a0.astype(np.float64), 1e-10)
    n = len(a)
    S = a.sum()
    a = a / S  # normalize to sum=1 for optimization (scale invariant)

    best_a = a.copy()
    best_J = compute_J(a)

    for beta, max_it in betas:
        if time.monotonic() > deadline:
            break

        x = a.copy()
        y = a.copy()
        tk = 1.0

        # Estimate Lipschitz constant
        # For the smoothed function, L is approximately 4n/beta * max(a)
        # Use adaptive Lipschitz estimation
        L = 4.0 * n / beta * np.max(a) + 1.0
        L = max(L, 1.0)

        for it in range(max_it):
            if time.monotonic() > deadline:
                break

            # Compute gradient at y
            _, grad_y = smooth_J_and_grad(y, beta)
            grad_y = np.nan_to_num(grad_y, nan=0.0, posinf=0.0, neginf=0.0)

            # Gradient step
            step = 1.0 / L
            z = y - step * grad_y

            # Project onto simplex (sum=1, a>=0)
            # Use bisection for the threshold
            z = np.maximum(z, 0.0)
            S_z = z.sum()
            if S_z < 1e-15:
                x_new = np.ones(n) / n
            else:
                # We need to project z onto {a >= 0, sum(a) = 1}
                # This is: a = max(z - lambda, 0) where lambda is chosen so sum = 1
                z_sorted = np.sort(z)[::-1]
                cz = np.cumsum(z_sorted) - 1.0
                rho = np.max(np.where(z_sorted - cz / np.arange(1, n+1) > 0)[0])
                theta = (z_sorted[rho] - 1.0 / (rho + 1))
                x_new = np.maximum(z - theta, 0.0)
                S_xn = x_new.sum()
                if S_xn < 1e-10:
                    x_new = np.ones(n) / n
                else:
                    x_new = x_new / S_xn

            # FISTA update
            tk_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * tk * tk))
            y = x_new + ((tk - 1.0) / tk_new) * (x_new - x)
            y = np.maximum(y, 0.0)
            S_y = y.sum()
            if S_y > 1e-10:
                y = y / S_y
            else:
                y = x_new.copy()

            x = x_new
            tk = tk_new

            # Check improvement
            J_x = compute_J(x)
            if J_x < best_J:
                best_J = J_x
                best_a = x.copy()

            # Adaptive L: if next step would be larger, increase L
            if it % 10 == 9:
                _, grad_check = smooth_J_and_grad(y, beta)
                diff = np.linalg.norm(y - (y - (1.0/L) * grad_check))
                if diff > 1.0 / L:
                    L *= 1.5

        a = x.copy()

    return best_a, best_J


def parabolic_line_search(a, g, J0, n_evals=12):
    """
    Parabolic interpolation line search minimizing J((1-t)*a + t*g) for t in [0,1].
    Returns (best_t, best_a, best_J).
    """
    best_t = 0.0
    best_J = J0
    best_a = a.copy()

    # Initial evaluations
    t1 = 0.25
    t2 = 0.5
    a1 = (1.0 - t1) * a + t1 * g
    a2 = (1.0 - t2) * a + t2 * g
    J1 = compute_J(a1)
    J2 = compute_J(a2)

    if J1 < best_J:
        best_J = J1
        best_t = t1
        best_a = a1.copy()
    if J2 < best_J:
        best_J = J2
        best_t = t2
        best_a = a2.copy()

    # Also check t=1
    J_full = compute_J(g)
    if J_full < best_J:
        best_J = J_full
        best_t = 1.0
        best_a = g.copy()

    # Parabolic interpolation: use 3 points to find minimum
    # Points: (0, J0), and the two best among t1, t2, 1
    for _ in range(n_evals):
        # Use the 3 most informative points
        points = [(0.0, J0), (t1, J1), (t2, J2)]
        if J_full < best_J or abs(1.0 - best_t) < 0.1:
            points.append((1.0, J_full))

        # Find the minimum of any quadratic through 3 of these points
        # Use the 3 points with the best J values
        points.sort(key=lambda x: x[1])
        if len(points) >= 3:
            (x0, y0), (x1, y1), (x2, y2) = points[0], points[1], points[2]

            # Fit parabola through these 3 points
            denom = (x0 - x1) * (x0 - x2) * (x1 - x2)
            if abs(denom) > 1e-15:
                a_par = (x2*(y1-y0) + x1*(y0-y2) + x0*(y2-y1)) / denom
                b_par = (x2*x2*(y0-y1) + x1*x1*(y2-y0) + x0*x0*(y1-y2)) / denom
                if abs(a_par) > 1e-15:
                    t_new = -b_par / (2.0 * a_par)
                    t_new = np.clip(t_new, 0.01, 0.99)
                else:
                    t_new = 0.5
            else:
                t_new = 0.5
        else:
            t_new = 0.5

        a_new = (1.0 - t_new) * a + t_new * g
        J_new = compute_J(a_new)

        # Update the points
        if J_new < J1:
            t1, J1 = t_new, J_new
        elif J_new < J2:
            t2, J2 = t_new, J_new

        if J_new < best_J:
            best_J = J_new
            best_t = t_new
            best_a = a_new.copy()

    return best_t, best_a, best_J


def golden_line_search(a, g, J0, n_evals=15):
    """Golden section search for t in [0,1] minimizing J((1-t)*a + t*g)."""
    best_t = 0.0
    best_J = J0
    best_a = a.copy()

    gr = (np.sqrt(5.0) - 1.0) / 2.0
    lo = 0.0
    hi = 1.0

    J_at_1 = compute_J(g)
    if J_at_1 < best_J:
        best_J = J_at_1
        best_t = 1.0
        best_a = g.copy()

    t1 = hi - gr * (hi - lo)
    t2 = lo + gr * (hi - lo)
    J1 = compute_J((1.0 - t1) * a + t1 * g)
    J2 = compute_J((1.0 - t2) * a + t2 * g)

    if J1 < best_J:
        best_J = J1
        best_t = t1
        best_a = ((1.0 - t1) * a + t1 * g).copy()
    if J2 < best_J:
        best_J = J2
        best_t = t2
        best_a = ((1.0 - t2) * a + t2 * g).copy()

    for _ in range(n_evals - 2):
        if J1 < J2:
            hi = t2
            t2 = t1
            J2 = J1
            t1 = hi - gr * (hi - lo)
            J1 = compute_J((1.0 - t1) * a + t1 * g)
        else:
            lo = t1
            t1 = t2
            J1 = J2
            t2 = lo + gr * (hi - lo)
            J2 = compute_J((1.0 - t2) * a + t2 * g)

        if J1 < best_J:
            best_J = J1
            best_t = t1
            best_a = ((1.0 - t1) * a + t1 * g).copy()
        if J2 < best_J:
            best_J = J2
            best_t = t2
            best_a = ((1.0 - t2) * a + t2 * g).copy()

    return best_t, best_a, best_J


def lp_mirror_descent(a, max_iters, deadline, tracker=None, n_line_evals=15,
                      stall_patience=30, use_parabolic=True):
    """
    LP-based mirror descent with line search.
    Includes stall detection, momentum, and adaptive step.
    """
    a = np.maximum(a.astype(np.float64), 1e-8)
    a = a / a.sum()
    current_J = compute_J(a)
    if tracker is not None:
        tracker.update(a)

    stall_count = 0
    best_J_so_far = current_J
    best_a_so_far = a.copy()

    # Momentum buffer
    momentum = np.zeros_like(a)
    beta_mom = 0.3

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        b = fast_conv(a)
        M = float(b.max())

        # Solve LP for best response
        g = lp_best_response(a, M)
        if g is None:
            break

        # Apply momentum
        momentum = beta_mom * momentum + (1.0 - beta_mom) * g
        g_mom = momentum / momentum.sum() if momentum.sum() > 1e-10 else g

        # Line search: try both g and g_mom
        if use_parabolic:
            t1, a1, J1 = parabolic_line_search(a, g, current_J, n_evals=n_line_evals)
        else:
            t1, a1, J1 = golden_line_search(a, g, current_J, n_evals=n_line_evals)

        # Also try momentum version
        if use_parabolic:
            t2, a2, J2 = parabolic_line_search(a, g_mom, current_J, n_evals=max(5, n_line_evals//2))
        else:
            t2, a2, J2 = golden_line_search(a, g_mom, current_J, n_evals=max(5, n_line_evals//2))

        # Choose the better
        if J2 < J1:
            a_new, J_new, t_used = a2, J2, t2
            momentum = momentum * 0.5  # reset momentum partially
        else:
            a_new, J_new, t_used = a1, J1, t1

        if J_new < current_J - 1e-13:
            a = a_new
            current_J = J_new
            stall_count = 0
            if J_new < best_J_so_far:
                best_J_so_far = J_new
                best_a_so_far = a.copy()
                if tracker is not None:
                    tracker.update(a)
        else:
            # Try smaller steps
            improved = False
            for t_try in [0.01, 0.005, 0.02, 0.05]:
                a_try = (1.0 - t_try) * a + t_try * g
                a_try = np.maximum(a_try, 0.0)
                s = a_try.sum()
                if s < 1e-10:
                    continue
                a_try = a_try / s
                J_try = compute_J(a_try)
                if J_try < current_J - 1e-14:
                    a = a_try
                    current_J = J_try
                    stall_count = 0
                    improved = True
                    if J_try < best_J_so_far:
                        best_J_so_far = J_try
                        best_a_so_far = a.copy()
                        if tracker is not None:
                            tracker.update(a)
                    break
            if not improved:
                stall_count += 1
                if stall_count >= stall_patience:
                    break

    if tracker is not None:
        tracker.update(best_a_so_far)
    return best_a_so_far


def resample_seq(src, dest_n):
    """Resample 1D sequence to new length via linear interpolation."""
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    result = np.interp(t_new, t_old, src)
    return np.maximum(result, 0.0)


def generate_u_shaped(n, alpha, rng):
    """Generate U-shaped sequence: high at edges, low in middle."""
    t = np.linspace(0, 1, n)
    # U-shape: 1 + alpha * (2*t - 1)^2
    seq = 1.0 + alpha * (2.0 * t - 1.0) ** 2
    # Add some noise
    seq += rng.normal(0, 0.02, n)
    return np.maximum(seq, 0.01)


def generate_initializations(n, rng, parent=None):
    """Generate diverse initial sequences of length n."""
    inits = []
    t_arr = np.linspace(0, 1, n, endpoint=False)

    if parent is not None and parent.size > 0:
        inits.append(resample_seq(parent, n))
        p = resample_seq(parent, n)
        p = p + rng.normal(0, 0.03, n) * p
        inits.append(np.maximum(p, 0.01))

    # U-shaped families (key insight: mass at edges flattens convolution)
    for alpha in [0.5, 1.0, 2.0, 3.0, 5.0]:
        inits.append(generate_u_shaped(n, alpha, rng))

    # Uniform
    inits.append(np.ones(n))

    # Random
    inits.append(rng.uniform(0.3, 1.0, n))

    # Cosine families
    for freq in [2, 3, 4]:
        inits.append(1.0 + 0.3 * np.cos(2.0 * np.pi * freq * t_arr))

    # Bathtub (inverse of Gaussian)
    inits.append(1.0 + 2.0 * np.exp(-8.0 * (t_arr - 0.5) ** 2) * (-1) + 1.5)
    inits.append(np.maximum(2.0 - np.exp(-10.0 * (t_arr - 0.5) ** 2), 0.1))

    # Power law edges
    inits.append(1.0 + 1.5 * (np.minimum(t_arr, 1 - t_arr)) ** (-0.3))
    inits.append(1.0 + 2.0 * (np.minimum(t_arr, 1 - t_arr)) ** (-0.5))

    # Two-peak
    inits.append(1.0 + 0.5 * (np.exp(-20*(t_arr-0.1)**2) + np.exp(-20*(t_arr-0.9)**2)))

    # Triangular U
    inits.append(1.0 + 1.0 * np.abs(2.0 * t_arr - 1.0))

    return inits


class ScoreTracker:
    """Tracks the best sequence found so far."""

    def __init__(self):
        self.best = None
        self.best_score = float('inf')
        self.t_start = time.monotonic()

    def update(self, a):
        a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
        if a.size == 0 or a.sum() < 0.01:
            return
        score = float(evaluate_sequence(a.tolist()))
        if np.isfinite(score) and score < self.best_score:
            self.best = a.copy()
            self.best_score = score
            elapsed = time.monotonic() - self.t_start
            print(f"  * NEW BEST: J={score:.8f} n={a.size} t={elapsed:.1f}s", flush=True)

    def report(self, label=""):
        if self.best is not None:
            print(f"  {label} best: J={self.best_score:.8f} n={self.best.size}", flush=True)


def subgradient_step(a, top_k=5, step_size=None):
    """
    One subgradient step: push down the top-k lags of the convolution.
    Returns updated a and new J.
    """
    n = len(a)
    b = fast_conv(a)
    S = a.sum()
    if S < 1e-10:
        return a, compute_J(a)

    # Get top-k lags
    top_indices = np.argsort(b)[::-1][:top_k]
    # Create concentrated distribution on top lags
    p = np.zeros(2 * n - 1)
    p[top_indices] = 1.0 / top_k

    # Compute gradient: grad[i] = sum_k p[k]*(a[k-i] + a[i-k]) - 2*M/S
    # where M = max(b) (or weighted average)
    M_weighted = float(np.dot(p, b))
    grad = np.zeros(n)

    # Using FFT for efficiency
    s = 1
    while s < 4 * n:
        s <<= 1
    Fa = np.fft.rfft(a, s)
    Fp = np.fft.rfft(p, s)

    # T1[i] = sum_k p[k]*a[k-i] = cross_corr(a, p)[i]
    corr = np.fft.irfft(np.conj(Fa) * Fp, s)[:2*n-1]
    # T2[i] = sum_k p[k]*a[i-k] = conv(p, a)[i]
    conv = np.fft.irfft(Fp * Fa, s)[:2*n-1]

    grad = corr[:n] + conv[:n] - 2.0 * M_weighted / S

    if step_size is None:
        # Adaptive step size
        grad_norm = np.linalg.norm(grad)
        if grad_norm < 1e-15:
            step_size = 0.0
        else:
            step_size = 0.01 * S / grad_norm

    a_new = a - step_size * grad
    a_new = np.maximum(a_new, 0.0)
    a_new = a_new / a_new.sum()

    return a_new, compute_J(a_new)


def subgradient_method(a, deadline, tracker=None, n_steps=5000,
                       top_k_schedule=None, cooling=0.999):
    """
    Subgradient method with adaptive step size and top-k scheduling.
    """
    a = np.maximum(a.astype(np.float64), 1e-8)
    a = a / a.sum()
    current_J = compute_J(a)
    best_a = a.copy()
    best_J = current_J

    if tracker is not None:
        tracker.update(a)

    step_size = 0.01
    top_k = 5

    for step in range(n_steps):
        if time.monotonic() > deadline:
            break

        # Schedule top_k
        if top_k_schedule:
            frac = step / max(n_steps - 1, 1)
            # Start with large top_k (smoother gradient), end with small (sharper)
            top_k = max(1, int(np.round(len(top_k_schedule[0]) +
                                         frac * (len(top_k_schedule[1]) - len(top_k_schedule[0])))))
            top_k = max(1, min(top_k, 2 * len(a)))

        a_new, J_new = subgradient_step(a, top_k=top_k, step_size=step_size)

        if J_new < current_J:
            a = a_new
            current_J = J_new
            if J_new < best_J:
                best_J = J_new
                best_a = a.copy()
                if tracker is not None:
                    tracker.update(a)
        else:
            # Reduce step size
            step_size *= 0.9

        # Cool down
        step_size *= cooling

        if step % 200 == 0 and tracker is not None:
            tracker.update(a)

    if tracker is not None:
        tracker.update(best_a)
    return best_a


def propose_candidate(seed=42, budget_s=1500, **kwargs):
    """
    Multi-phase optimization for minimizing J(a) = 2n*max(conv(a,a))/(sum(a))^2.

    Phase 1: Diverse initialization with U-shaped profiles + FISTA basin finding
    Phase 2: Subgradient method with top-k scheduling
    Phase 3: LP mirror descent with parabolic line search (main refinement)
    Phase 4: Multi-restart with structured perturbations
    Phase 5: Final LP polish + length exploration
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + float(budget_s) - 15.0
    tracker = ScoreTracker()

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        if p.ndim == 1 and p.size > 0 and np.all(np.isfinite(p)):
            parent = np.maximum(p, 0.0)
            tracker.update(parent)
            print(f"Parent loaded: n={parent.size}, J={tracker.best_score:.8f}", flush=True)
    except (NameError, AttributeError):
        pass

    if parent is None:
        parent = np.ones(512)
        tracker.update(parent)
        print(f"Using default parent: n={parent.size}", flush=True)

    n_base = parent.size

    # === Phase 1: FISTA with diverse initializations ===
    phase1_budget = 200.0
    deadline_p1 = time.monotonic() + phase1_budget
    print(f"\n{'='*50} Phase 1: FISTA (budget {phase1_budget:.0f}s) {'='*50}", flush=True)
    t0 = time.monotonic()

    # Use the best length from parent
    for n_try in [n_base]:
        if time.monotonic() > deadline_p1:
            break
        inits = generate_initializations(n_try, rng, parent)
        # Select top initializations by score
        init_scores = []
        for a0 in inits:
            a0 = np.maximum(a0, 0.01)
            a0 = a0 / a0.sum()
            j0 = compute_J(a0)
            init_scores.append((j0, a0))
        init_scores.sort(key=lambda x: x[0])

        # Run FISTA on top 5 initializations
        for idx, (j0, a0) in enumerate(init_scores[:5]):
            if time.monotonic() > deadline_p1:
                break
            beta_schedule = [(2.0, 50), (5.0, 50), (10.0, 60), (20.0, 80), (40.0, 100)]
            remaining = deadline_p1 - time.monotonic()
            if remaining < 5:
                break
            a_opt, J_opt = fista_optimize(a0, beta_schedule, time.monotonic() + remaining,
                                          max_iters_per_beta=80)
            if a_opt is not None:
                tracker.update(a_opt)
            print(f"    init {idx}: J0={j0:.6f} -> J={J_opt:.6f}", flush=True)

    tracker.report(f"Phase 1 complete ({time.monotonic()-t0:.1f}s)")

    # === Phase 2: Subgradient method ===
    if tracker.best is not None and time.monotonic() < deadline:
        phase2_budget = 200.0
        deadline_p2 = time.monotonic() + phase2_budget
        print(f"\n{'='*50} Phase 2: Subgradient (budget {phase2_budget:.0f}s) {'='*50}", flush=True)
        t0 = time.monotonic()

        n_best = tracker.best.size
        # Run subgradient from best
        remaining = deadline_p2 - time.monotonic()
        n_steps = int(remaining * 200)  # ~200 steps per second estimate
        n_steps = min(n_steps, 10000)
        a_sub = subgradient_method(tracker.best, time.monotonic() + remaining,
                                   tracker=tracker, n_steps=n_steps,
                                   top_k_schedule=[10, 3], cooling=0.9995)
        tracker.update(a_sub)

        # Try from a few other initializations
        for a0 in init_scores[1:3]:
            if time.monotonic() > deadline_p2:
                break
            a_sub2 = subgradient_method(a0[1], deadline_p2, tracker=tracker,
                                        n_steps=3000, top_k_schedule=[5, 2], cooling=0.999)
            tracker.update(a_sub2)

        tracker.report(f"Phase 2 complete ({time.monotonic()-t0:.1f}s)")

    # === Phase 3: LP mirror descent (MAIN PHASE) ===
    if tracker.best is not None and time.monotonic() < deadline:
        n_best = tracker.best.size
        phase3_budget = 500.0
        deadline_p3 = min(deadline, time.monotonic() + phase3_budget)
        print(f"\n{'='*50} Phase 3: LP Mirror Descent n={n_best} (budget {phase3_budget:.0f}s) {'='*50}", flush=True)
        t0 = time.monotonic()

        remaining = deadline_p3 - time.monotonic()
        max_iters = int(remaining * 50)
        max_iters = min(max_iters, 4000)

        # Primary LP run
        a_lp = lp_mirror_descent(tracker.best, max_iters, deadline_p3, tracker=tracker,
                                  n_line_evals=16, stall_patience=40, use_parabolic=True)
        tracker.update(a_lp)

        # Second run from best with different parameters
        if tracker.best is not None and time.monotonic() < deadline_p3:
            remaining2 = deadline_p3 - time.monotonic()
            max_iters2 = int(remaining2 * 50)
            max_iters2 = min(max_iters2, 3000)
            a_lp2 = lp_mirror_descent(tracker.best, max_iters2, deadline_p3, tracker=tracker,
                                       n_line_evals=14, stall_patience=30, use_parabolic=False)
            tracker.update(a_lp2)

        tracker.report(f"Phase 3 complete ({time.monotonic()-t0:.1f}s)")

    # === Phase 4: Multi-restart with structured perturbations ===
    if tracker.best is not None and time.monotonic() < deadline:
        n_best = tracker.best.size
        phase4_budget = 350.0
        deadline_p4 = min(deadline, time.monotonic() + phase4_budget)
        print(f"\n{'='*50} Phase 4: Multi-Restart (budget {phase4_budget:.0f}s) {'='*50}", flush=True)
        t0 = time.monotonic()

        # Define perturbation strategies
        strategies = []
        # Gaussian perturbations with various scales
        for scale in [0.03, 0.05, 0.08, 0.12, 0.15]:
            strategies.append(('gaussian', scale))
        # U-shaped perturbations (push mass to edges or center)
        strategies.append(('push_edges', 0.2))
        strategies.append(('push_center', 0.2))
        # Spike perturbations
        for n_spikes in [4, 8, 16]:
            strategies.append(('spike', n_spikes))
        # Smooth noise
        strategies.append(('smooth', 0.05))
        strategies.append(('smooth', 0.1))

        attempt = 0
        for pert_type, param in strategies:
            if time.monotonic() > deadline_p4:
                break
            for trial in range(4):
                if time.monotonic() > deadline_p4:
                    break
                attempt += 1
                a = tracker.best.copy().astype(np.float64)
                n = len(a)
                t_arr = np.linspace(0, 1, n)

                if pert_type == 'gaussian':
                    a = a + rng.normal(0, param, n) * (a / a.max())
                elif pert_type == 'push_edges':
                    # Push mass to edges
                    edge_weight = param * np.minimum(t_arr, 1 - t_arr) ** (-0.5)
                    a = a + edge_weight * 0.1
                elif pert_type == 'push_center':
                    # Push mass to center
                    center_weight = param * np.exp(-5 * (t_arr - 0.5) ** 2)
                    a = a + center_weight * 0.1
                elif pert_type == 'spike':
                    k = min(param, n // 4)
                    idx = rng.choice(n, size=k, replace=False)
                    a[idx] *= rng.uniform(0.3, 3.0, k)
                elif pert_type == 'smooth':
                    noise = rng.normal(0, param, n)
                    kernel_size = max(3, n // 32)
                    kernel = np.ones(kernel_size) / kernel_size
                    noise = np.convolve(noise, kernel, mode='same')
                    a = a + noise * (a / a.max())

                a = np.maximum(a, 1e-4)
                a = a / a.sum()

                # Quick FISTA polish
                beta_sched = [(5.0, 30), (15.0, 40), (30.0, 50)]
                a_opt, _ = fista_optimize(a, beta_sched, time.monotonic() + 30)
                tracker.update(a_opt)

                # LP polish
                if time.monotonic() < deadline_p4:
                    budget_here = min(40.0, deadline_p4 - time.monotonic())
                    max_iters = int(budget_here * 50)
                    max_iters = min(max_iters, 1000)
                    a_lp = lp_mirror_descent(a_opt, max_iters, deadline_p4, tracker=tracker,
                                              n_line_evals=12, stall_patience=25,
                                              use_parabolic=True)
                    tracker.update(a_lp)

        tracker.report(f"Phase 4 complete ({time.monotonic()-t0:.1f}s), {attempt} attempts")

    # === Phase 5: Final polish + length exploration ===
    if tracker.best is not None and time.monotonic() < deadline:
        t0 = time.monotonic()
        n_best = tracker.best.size
        remaining = deadline - time.monotonic()
        print(f"\n{'='*50} Phase 5: Final Polish (budget {remaining:.0f}s) {'='*50}", flush=True)

        # Extended LP on best
        phase5a_budget = min(200.0, remaining * 0.6)
        deadline_p5a = time.monotonic() + phase5a_budget
        max_iters = int(phase5a_budget * 50)
        max_iters = min(max_iters, 5000)
        a_final = lp_mirror_descent(tracker.best, max_iters, deadline_p5a, tracker=tracker,
                                     n_line_evals=20, stall_patience=60, use_parabolic=True)
        tracker.update(a_final)

        # Try nearby lengths
        if time.monotonic() < deadline:
            for n_try in [n_best - 32, n_best + 32, n_best - 64, n_best + 64,
                          768, 896, 1152, 1280]:
                if n_try < 128 or n_try > 2048 or time.monotonic() > deadline:
                    continue
                if abs(n_try - n_best) < 16:
                    continue
                a_res = resample_seq(tracker.best, n_try)
                a_res = np.maximum(a_res, 0.01)
                a_res = a_res / a_res.sum()
                budget_here = min(30.0, deadline - time.monotonic())
                if budget_here < 5:
                    break
                max_iters = int(budget_here * 40)
                max_iters = min(max_iters, 800)
                a_res = lp_mirror_descent(a_res, max_iters, deadline, tracker=tracker,
                                           n_line_evals=12, stall_patience=20,
                                           use_parabolic=True)
                tracker.update(a_res)

        tracker.report(f"Phase 5 complete ({time.monotonic()-t0:.1f}s)")

    # Fallback
    if tracker.best is None:
        tracker.best = np.ones(256)
        tracker.best_score = compute_J(tracker.best)
        print(f"  Using fallback: J={tracker.best_score:.8f}", flush=True)

    print(f"\n{'='*50} FINAL RESULT {'='*50}", flush=True)
    print(f"J={tracker.best_score:.8f} n={tracker.best.size} "
          f"time={time.monotonic()-tracker.t_start:.1f}s", flush=True)

    # Ensure output is valid
    result = np.clip(tracker.best, 0.0, 1000.0)
    if result.sum() < 0.01:
        result = np.ones(128)

    return [float(x) for x in result]
