def fast_conv(a):
    """Full self-convolution via FFT."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]


def compute_J(a):
    """Compute J(a) = 2*n*max(conv(a,a))/(sum(a))^2."""
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))


def project_simplex(v):
    """Euclidean projection onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - 1))[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0)


def subgradient_exact(a, top_k=None):
    """
    Compute subgradient of max(conv(a,a)) using top_k peaks.
    Returns (gradient_direction, active_set, b, M).
    """
    n = len(a)
    b = fast_conv(a)
    M = b.max()

    if top_k is not None and top_k < len(b):
        # Select top K peaks
        tol = M * 0.001
        # Get indices sorted by value
        idx = np.argsort(b)[::-1][:top_k]
        active = idx
        # Re-normalize weights to sum to 1
        weights = b[active] / b[active].sum()
    else:
        tol = 1e-6 * M if M > 0 else 1e-10
        active = np.where(b >= M - tol)[0]
        weights = np.ones(len(active)) / len(active)

    grad = np.zeros(n)
    for k, w in zip(active, weights):
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        grad[lo:hi + 1] += 2.0 * w * a[k - hi:k - lo + 1][::-1]

    return grad, active, b, M


def run_nesterov_subgradient(a, max_iters, deadline, lr_init=0.3, top_k=None,
                              momentum_init=0.5):
    """
    Nesterov-accelerated projected subgradient descent.
    Uses Euclidean projection onto simplex.
    """
    a = np.maximum(a.astype(np.float64), 1e-10)
    a = a / a.sum()
    n = len(a)

    best_a = a.copy()
    best_M = fast_conv(a).max()

    # Nesterov: maintain y (lookahead) and v (velocity)
    y = a.copy()
    v = np.zeros(n)
    t = 1.0

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        t = it + 1
        gamma = (1 + np.sqrt(1 + 4 * t * t)) / 2  # Nesterov momentum
        gamma_prev = (1 + np.sqrt(1 + 4 * (t - 1) * (t - 1))) / 2 if t > 1 else 1.0

        lr = lr_init / (t ** 0.6)

        # Compute subgradient at y
        grad, active, b, M = subgradient_exact(y, top_k=top_k)

        # Nesterov update
        y_new = y - lr * grad
        y_new = project_simplex(y_new)

        # Update x (the actual iterate)
        a_new = y_new + ((t - 1) / gamma - 1) * (y_new - y)
        a_new = project_simplex(a_new)

        M_val = b.max()
        if M_val < best_M * (1 - 1e-10):
            best_M = M_val
            best_a = a_new.copy()

        a = a_new
        y = y_new

    M_final = fast_conv(a).max()
    if M_final < best_M:
        best_a = a.copy()

    return best_a


def build_A_matrix(n, a, active_set=None):
    """Build constraint matrix for LP. A[k,j] = a[k-j] if valid."""
    if active_set is not None:
        rows_all = active_set
    else:
        rows_all = range(2 * n - 1)

    rows = []
    cols = []
    vals = []
    for k in rows_all:
        j_min = max(0, k - n + 1)
        j_max = min(n - 1, k)
        for j in range(j_min, j_max + 1):
            rows.append(k)
            cols.append(j)
            vals.append(a[k - j])

    if active_set is not None:
        # Remap row indices to 0..len(active_set)-1
        row_map = {k: i for i, k in enumerate(active_set)}
        rows = [row_map[r] for r in rows]
        shape = (len(active_set), n)
    else:
        shape = (2 * n - 1, n)

    A = csr_matrix((vals, (rows, cols)), shape=shape)
    return A

def run_peak_shaving(a, max_iters, deadline, top_k=5, lr_init=0.1):
    """
    Focused peak shaving: use only top-K peaks for subgradient,
    with backtracking line search.
    """
    a = np.maximum(a.astype(np.float64), 1e-10)
    a = a / a.sum()
    n = len(a)

    best_a = a.copy()
    best_M = fast_conv(a).max()

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        b = fast_conv(a)
        M = b.max()

        # Get gradient from top K peaks only
        K = min(top_k, len(b))
        top_idx = np.argsort(b)[::-1][:K]
        weights = b[top_idx]
        weights = weights / weights.sum()

        grad = np.zeros(n)
        for k, w in zip(top_idx, weights):
            lo = max(0, k - n + 1)
            hi = min(n - 1, k)
            grad[lo:hi + 1] += 2.0 * w * a[k - hi:k - lo + 1][::-1]

        # Backtracking line search
        lr = lr_init / (1 + 0.1 * it)
        for factor in [1.0, 0.5, 0.25, 0.1, 0.05, 0.01]:
            a_trial = a - lr * factor * grad
            a_trial = project_simplex(a_trial)
            b_trial = fast_conv(a_trial)
            M_trial = b_trial.max()

            if M_trial < best_M - 1e-14:
                a = a_trial
                best_M = M_trial
                best_a = a.copy()
                break
        else:
            # No improvement even with smallest step
            if lr * 0.01 < 1e-8:
                break

    return best_a


def run_log_domain_descent(a, max_iters, deadline, lr_init=0.5):

        if time.monotonic() > deadline:
            break

        # da/dx = a * (I - a*1^T) for softmax
        # grad_x = (I - a*1^T)^T @ (grad_a / a) = grad_a/a - a * sum(grad_a)
        # Simplified: grad_x[i] = grad_a[i]/a[i] - sum(grad_a[j])
        # But this can be unstable. Let's use: grad_x = grad_a - a * sum(grad_a)
        # (first-order approximation of the softmax Jacobian)
        grad_x = grad_a - a_cur * grad_a.sum()

        lr = lr_init / (1 + 0.3 * it ** 0.5)
        v = 0.9 * v + lr * grad_x
        x = x - v

        # Periodic renormalization to prevent overflow
        x = x - x.mean()

    # Final conversion
    x_shifted = x - x.max()
    a_final = np.exp(x_shifted)
    a_final = a_final / a_final.sum()

    if fast_conv(a_final).max() < best_M:
        best_a = a_final

    return best_a


def resample_seq(src, dest_n):
    """Resample 1D sequence to new length via cubic interpolation."""
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    if src.size < 2:
        return np.full(dest_n, src[0])
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    # Use cubic interpolation for smoother results
    from scipy.interpolate import CubicSpline
    try:
        cs = CubicSpline(t_old, src, bc_type='natural')
        result = cs(t_new)
    except Exception:
        result = np.interp(t_new, t_old, src)
    return np.maximum(result, 0.0)


def generate_diverse_inits(n, rng, parent=None):

    inits.append(np.exp(-8.0 * (t_arr - 0.3) ** 2) + np.exp(-8.0 * (t_arr - 0.7) ** 2) + 0.05)

"""
Minimizes J(a) = 2*n*max(conv(a,a))/(sum(a))^2 using a novel hybrid approach:
1. Nesterov-accelerated projected subgradient with exact max (fast exploration)

4. Log-domain gradient descent for better conditioning near the boundary

Key innovations: warm-started LP dual, geometric mean updates, focused peak

import numpy as np

def fast_conv(a):
    """Full self-convolution via FFT."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]


def compute_J(a):
    """Compute J(a) = 2*n*max(conv(a,a))/(sum(a))^2."""
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))


def project_simplex(v):
    """Euclidean projection onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho_candidates = np.nonzero(u * np.arange(1, n + 1) > (cssv - 1))[0]
    if len(rho_candidates) == 0:
        return np.ones(n) / n
    rho = rho_candidates[-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0)


def subgradient_exact(a, top_k=None):
    """
    Compute subgradient of max(conv(a,a)) using top_k peaks.
    Returns (gradient, active_set, b, M).
    """
    n = len(a)
    b = fast_conv(a)
    M = b.max()

    if top_k is not None and top_k < len(b):
        idx = np.argsort(b)[::-1][:top_k]
        active = idx
        weights = b[active]
        w_sum = weights.sum()
        if w_sum > 0:
            weights = weights / w_sum
        else:
            weights = np.ones(len(active)) / len(active)
    else:
        tol = 1e-6 * M if M > 0 else 1e-10
        active = np.where(b >= M - tol)[0]
        weights = np.ones(len(active)) / max(1, len(active))

    grad = np.zeros(n)
    for k, w in zip(active, weights):
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        grad[lo:hi + 1] += 2.0 * w * a[k - hi:k - lo + 1][::-1]

    return grad, active, b, M


def run_nesterov_subgradient(a, max_iters, deadline, lr_init=0.3, top_k=None):
    """
    Nesterov-accelerated projected subgradient descent.
    Uses Euclidean projection onto simplex.
    """
    a = np.maximum(a.astype(np.float64), 1e-10)
    a = a / a.sum()
    n = len(a)

    best_a = a.copy()
    best_M = fast_conv(a).max()

    y = a.copy()

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        t_val = it + 1
        t_prev = it

        lr = lr_init / (t_val ** 0.6)

        grad, active, b, M = subgradient_exact(y, top_k=top_k)

        # Standard Nesterov update
        y_new = y - lr * grad
        y_new = project_simplex(y_new)

        # Compute momentum coefficient
        if t_prev > 0:
            gamma = (1.0 + np.sqrt(1.0 + 4.0 * t_val * t_val)) / 2.0
            gamma_prev = (1.0 + np.sqrt(1.0 + 4.0 * t_prev * t_prev)) / 2.0
            a_new = y_new + ((t_prev / gamma) - 1.0) * (y_new - y)
        else:
            a_new = y_new

        a_new = project_simplex(a_new)

        # Check improvement
        if M < best_M * (1 - 1e-10):
            best_M = M
            best_a = a_new.copy()

        a = a_new
        y = y_new

    # Final check
    M_final = fast_conv(a).max()
    if M_final < best_M:
        best_a = a.copy()

    return best_a


def build_A_matrix(n, a, active_set=None):
    """Build constraint matrix for LP. A[k,j] = a[k-j] if valid."""
    if active_set is not None:
        rows_all = list(active_set)
    else:
        rows_all = list(range(2 * n - 1))

    rows = []
    cols = []
    vals = []
    for k in rows_all:
        j_min = max(0, k - n + 1)
        j_max = min(n - 1, k)
        for j in range(j_min, j_max + 1):
            rows.append(k)
            cols.append(j)
            vals.append(a[k - j])

    if active_set is not None:
        row_map = {k: i for i, k in enumerate(rows_all)}
        rows = [row_map[r] for r in rows]
        shape = (len(rows_all), n)
    else:
        shape = (2 * n - 1, n)

    if len(rows) == 0:
        return csr_matrix((shape[0], n))
    return csr_matrix((vals, (rows, cols)), shape=shape)


def lp_best_response(a, M, active_set=None, time_limit=15):
    """
    LP: max sum(g) s.t. (a*g)[k] <= M for k in active_set (or all k), g >= 0.
    Returns g or None.
    """
    n = len(a)
    if active_set is not None:
        A = build_A_matrix(n, a, active_set)
        m = len(active_set)
    else:
        A = build_A_matrix(n, a)
        m = 2 * n - 1

    c = -np.ones(n)
    b_ub = np.full(m, M)
    bounds = [(0, None)] * n

    try:
        r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                    options={'time_limit': time_limit, 'presolve': True})
        if r.success and r.x is not None:
            g = np.maximum(r.x, 0)
            if g.sum() > 1e-15:
                return g
    except Exception:
        pass
    return None

def run_peak_shaving(a, max_iters, deadline, top_k=5, lr_init=0.1):
    """
    Focused peak shaving: use only top-K peaks for subgradient,
    with backtracking line search.
    """
    a = np.maximum(a.astype(np.float64), 1e-10)
    a = a / a.sum()
    n = len(a)

    best_a = a.copy()
    best_M = fast_conv(a).max()
    stuck_count = 0

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        b = fast_conv(a)
        M = b.max()

        K = min(top_k, len(b))
        top_idx = np.argsort(b)[::-1][:K]
        weights = b[top_idx]
        w_sum = weights.sum()
        if w_sum > 0:
            weights = weights / w_sum
        else:
            weights = np.ones(K) / K

        grad = np.zeros(n)
        for k, w in zip(top_idx, weights):
            lo = max(0, k - n + 1)
            hi = min(n - 1, k)
            grad[lo:hi + 1] += 2.0 * w * a[k - hi:k - lo + 1][::-1]

        # Backtracking line search
        lr_base = lr_init / (1 + 0.05 * it)
        found_improvement = False
        for factor in [1.0, 0.5, 0.25, 0.1, 0.05, 0.01]:
            a_trial = a - lr_base * factor * grad
            a_trial = project_simplex(a_trial)
            b_trial = fast_conv(a_trial)
            M_trial = b_trial.max()

            if M_trial < best_M - 1e-14:
                a = a_trial
                best_M = M_trial
                best_a = a.copy()
                found_improvement = True
                stuck_count = 0
                break

        if not found_improvement:
            stuck_count += 1
            if stuck_count > 50:
                break

    return best_a


def run_log_domain_descent(a, max_iters, deadline, lr_init=0.5):
    """
    Gradient descent in the log domain: a[i] = softmax(x[i]).
    """
    a_init = np.maximum(a.astype(np.float64), 1e-10)
    a_init = a_init / a_init.sum()
    n = len(a_init)

    x = np.log(a_init)
    best_a = a_init.copy()
    best_M = fast_conv(a_init).max()

    v = np.zeros(n)

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        x_shifted = x - x.max()
        a_cur = np.exp(x_shifted)
        a_cur = a_cur / a_cur.sum()

        grad_a, _, b, M = subgradient_exact(a_cur, top_k=10)

        if M < best_M * (1 - 1e-10):
            best_M = M
            best_a = a_cur.copy()

        # Gradient in log domain (approximate)
        grad_x = grad_a - a_cur * grad_a.sum()

        lr = lr_init / (1 + 0.5 * it ** 0.5)
        v = 0.85 * v + lr * grad_x
        x = x - v

        # Renormalize to prevent overflow
        x = x - x.mean()

    # Final
    x_shifted = x - x.max()
    a_final = np.exp(x_shifted)
    a_final = a_final / a_final.sum()
    if fast_conv(a_final).max() < best_M:
        best_a = a_final

    return best_a


def resample_seq(src, dest_n):
    """Resample 1D sequence to new length via cubic interpolation."""
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    if src.size < 2:
        return np.full(dest_n, src[0])
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    try:
        cs = CubicSpline(t_old, src, bc_type='natural')
        result = cs(t_new)
    except Exception:
        result = np.interp(t_new, t_old, src)
    return np.maximum(result, 0.0)
