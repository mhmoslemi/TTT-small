            S = np.sum(g)
            if S > 1e-10:
                g = g / S
            return g
    except Exception:
        pass
    return None

def lp_max_sum(a, M, time_limit=10):
    """
    LP1: max sum(g) s.t. conv(a,g)[k] <= M for all k, g >= 0.
    """
    n = len(a)
    A = build_toeplitz_sparse(a)
    c = -np.ones(n)
    b_ub = np.full(2*n-1, M)
    bounds = [(0, None)] * n
    try:
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': time_limit})
        if res.status == 0:
            g = np.maximum(res.x, 0)
            S = np.sum(g)
            if S > 1e-10:
                g = g / S
            return g
    except Exception:
        pass
    return None

def subgradient_step(a, topk=20):
    """Compute subgradient of max(conv(a,a)) w.r.t. a."""
    n = len(a)
    conv = compute_conv(a)
    M = float(np.max(conv))
    
    # Threshold: use all k where conv[k] >= M * (1 - 0.05)
    threshold = M * 0.95
    active = np.where(conv >= threshold)[0]
    if len(active) == 0:
        active = np.argsort(conv)[-topk:]
    
    # Gradient of (aa)[k] w.r.t. a[i] = a[k-i] + a[i-k] (if both in range)
    # More precisely: d/d a[i] sum_j a[j] a[k-j] = a[k-i] (if 0<=k-i<n) + a[i-k] (if 0<=i-k<n)
    # But since we're differentiating a^T T_k a, it's 2 * (T_k a)[i] (for symmetric T_k)
    # Actually for linear conv: (aa)[k] = sum_j a[j] a[k-j]
    # d/d a[i] = a[k-i] (if 0<=k-i<n) + a[i-k] (if 0<=i-k<n, i.e., k<=i<n+k, but since k can be up to 2n-2...)
    # Wait, let me think more carefully.
    # (aa)[k] = sum_{j: 0<=j<n, 0<=k-j<n} a[j] * a[k-j]
    # d/d a[i]: 
    #   - a[i] appears as a[j] when j=i: contributes a[k-i] if 0<=k-i<n
    #   - a[i] appears as a[k-j] when k-j=i, i.e., j=k-i: contributes a[k-i] if 0<=k-i<n
    #   Wait, that's the same term counted twice? No:
    #   (aa)[k] = sum_j a[j] * a[k-j]
    #   If we differentiate w.r.t. a[i]:
    #     Term where j=i: a[k-i] (valid if 0 <= k-i < n)
    #     Term where k-j=i, i.e., j=k-i: a[k-i] (valid if 0 <= k-i < n)
    #   But these are the same when j=i and k-j=i, which means k=2i.
    #   Actually no, they're different terms in the sum (unless k=2i).
    #   So: d/d a[i] (aa)[k] = (a[k-i] if 0<=k-i<n else 0) + (a[k-i] if 0<=k-i<n else 0)
    #   Wait, that's 2*a[k-i] when 0<=k-i<n.
    #   
    #   Hmm, let me verify with a small example.
    #   a = [a0, a1], k=1: (aa)[1] = a[0]*a[1] + a[1]*a[0] = 2*a0*a1
    #   d/d a0 = 2*a1. Using formula: 2*a[1-0] = 2*a1. ✓
    #   d/d a1 = 2*a0. Using formula: 2*a[1-1] = 2*a0. ✓
    #   
    #   a = [a0, a1, a2], k=1: (aa)[1] = a[0]*a[1] + a[1]*a[0] = 2*a0*a1
    #   d/d a0 = 2*a1. Formula: 2*a[1] = 2*a1. ✓
    #   d/d a1 = 2*a0. Formula: 2*a[0] = 2*a0. ✓
    #   d/d a2 = 0. Formula: k-i = 1-2 = -1, not in [0,3), so 0. ✓
    #
    #   a = [a0, a1, a2], k=2: (aa)[2] = a[0]*a[2] + a[1]*a[1] + a[2]*a[0] = 2*a0*a2 + a1^2
    #   d/d a0 = 2*a2. Formula: 2*a[2] = 2*a2. ✓
    #   d/d a1 = 2*a1. Formula: 2*a[1] = 2*a1. ✓
    #   d/d a2 = 2*a0. Formula: 2*a[0] = 2*a0. ✓
    #
    # Great, so the gradient is: for each active k, grad[i] += 2*a[k-i] if 0 <= k-i < n

    grad = np.zeros(n)
    for k in active:
        for i in range(n):
            j = k - i
            if 0 <= j < n:
                grad[i] += 2.0 * a[j]
    
    # Average over active set
    grad /= len(active)
    return grad, M, conv

def momentum_subgradient(a, max_iters, time_budget, topk=15, lr0=0.1):
    """Momentum subgradient descent."""
    n = len(a)
    a = a.copy().astype(np.float64)
    S = np.sum(a)
    if S < 1e-10:
        return a
    a = a / S
    
    v = np.zeros(n)
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    momentum = 0.8
    stall = 0
    
    for it in range(max_iters):
        if time.monotonic() - start > time_budget:
            break
        
        grad, M, conv = subgradient_step(a, topk)
        grad_centered = grad - np.mean(grad)
        
        v = momentum * v + (1.0 - momentum) * grad_centered
        lr = lr0 / (1.0 + 0.005 * it)
        
        a_new = a - lr * v
        a_new = np.maximum(a_new, 0)
        S_new = np.sum(a_new)
        if S_new < 1e-10:
            break
        a_new = a_new / S_new
        
        obj = compute_obj(a_new)
        if obj < best_obj - 1e-12:
            best_obj = obj
            best_a = a_new.copy()
            a = a_new
            stall = 0
        else:
            stall += 1
            if stall > 50:
                rng = np.random.default_rng(it * 31 + 7)
                a = a * np.exp(rng.normal(0, 0.05, n))
                a = np.maximum(a, 0)
                a = a / np.sum(a)
                v = np.zeros(n)
                stall = 0
                lr0 *= 0.8
                if lr0 < 1e-8:
                    break
    
    return best_a

def spectral_init(n, n_modes=12, seed=0):
    """Spectral initialization: design a sequence with approximately flat power spectrum."""
    rng = np.random.default_rng(seed)
    k = np.arange(n)
    
    # Start with a raised cosine (smooth, concentrated)
    x = 2.0 * k / (n - 1) - 1.0
    a = 0.5 * (1.0 + np.cos(np.pi * x)) ** 1.2
    
    # Add random cosine modes
    for m in range(1, n_modes + 1):
        c = rng.uniform(-0.4, 0.4) / m
        phase = rng.uniform(0, 2 * np.pi)
        a += c * np.cos(2 * np.pi * m * k / n + phase)
    
    a = np.maximum(a, 1e-10)
    S = np.sum(a)
    return a / S

def differentiators_init(n, seed=0):
    """Initialize using a differentiator-like pattern (alternating peaks)."""
    rng = np.random.default_rng(seed)
    a = np.ones(n)
    # Create a pattern with alternating high and low values
    for i in range(n):
        a[i] *= (1.0 + 0.3 * np.sin(2 * np.pi * rng.uniform(2, 5) * i / n))
    a = np.maximum(a, 1e-10)
    S = np.sum(a)
    return a / S

def refine_with_lp(a, max_iters=30, time_budget=40.0, use_lp2=True):
    """Refine using alternating LP."""
    n = len(a)
    a = a.copy().astype(np.float64)
    S = np.sum(a)
    if S < 1e-10:
        return a, np.inf
    a = a / S
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    stall = 0
    
    for it in range(max_iters):
        if time.monotonic() - start > time_budget:
            break
        
        conv = compute_conv(a)
        M = float(np.max(conv))
        
        if use_lp2:
            g = lp_min_peak(a, time_limit=min(5.0, max(2.0, time_budget - (time.monotonic() - start))))
        else:
            g = lp_max_sum(a, M, time_limit=min(5.0, max(2.0, time_budget - (time.monotonic() - start))))
        
        if g is None:
            break
        
        S_g = np.sum(g)
        if S_g < 1e-10:
            break
        g = g / S_g
        
        # Line search
        best_t = 0.0
        best_obj_ls = best_obj
        for t in np.concatenate([np.linspace(0.05, 0.5, 5), np.linspace(0.5, 1.0, 6)]):
            a_try = (1.0 - t) * a + t * g
            a_try = np.maximum(a_try, 0)
            S_try = np.sum(a_try)
            if S_try < 1e-10:
                continue
            a_try = a_try / S_try
            obj_try = compute_obj(a_try)
            if obj_try < best_obj_ls:
                best_obj_ls = obj_try
                best_t = t
        
        if best_t > 0.0:
            a = (1.0 - best_t) * a + best_t * g
            a = np.maximum(a, 0)
            a = a / np.sum(a)
            best_obj = best_obj_ls
            best_a = a.copy()
            stall = 0
        else:
            stall += 1
            if stall > 8:
                rng = np.random.default_rng(it * 7919 + 123)
                a = a * np.exp(rng.normal(0, 0.1, n))
                a = np.maximum(a, 0)
                a = a / np.sum(a)
                stall = 0
            else:
                # Small blend to keep moving
                a = 0.95 * a + 0.05 * g
                a = np.maximum(a, 0)
                a = a / np.sum(a)
    
    return best_a, best_obj

def full_refine(a, time_budget):
    """Full refinement pipeline combining LP and subgradient."""
    n = len(a)
    a = a.copy().astype(np.float64)
    S = np.sum(a)
    if S < 1e-10:
        return a, np.inf
    a = a / S
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    rnd = 0
    
    while time.monotonic() - start < time_budget - 5:
        rnd += 1
        remaining = time_budget - (time.monotonic() - start)
        if remaining < 10:
            break
        
        # LP phase
        lp_budget = min(remaining * 0.4, 30.0)
        if n <= 800:
            a_lp, obj_lp = refine_with_lp(best_a, max_iters=20, time_budget=lp_budget, use_lp2=True)
            if obj_lp < best_obj:
                best_obj = obj_lp
                best_a = a_lp.copy()
        
        # Subgradient phase
        remaining = time_budget - (time.monotonic() - start)
        if remaining > 10:
            sg_budget = min(remaining * 0.25, 15.0)
            a_sg = momentum_subgradient(best_a, max_iters=int(sg_budget * 30), time_budget=sg_budget, topk=15)
            obj_sg = compute_obj(a_sg)
            if obj_sg < best_obj:
                best_obj = obj_sg
                best_a = a_sg.copy()
        
        # Perturbation
        remaining = time_budget - (time.monotonic() - start)
        if remaining > 15:
            rng = np.random.default_rng(rnd * 42 + 17)
            a_pert = best_a * np.exp(rng.normal(0, 0.08, n))
            a_pert = np.maximum(a_pert, 0)
            S_p = np.sum(a_pert)
            if S_p > 1e-10:
                a_pert = a_pert / S_p
                # Quick refinement
                if n <= 600:
                    a_p, obj_p = refine_with_lp(a_pert, max_iters=10, time_budget=min(10.0, remaining * 0.2))
                    if obj_p < best_obj:
                        best_obj = obj_p
                        best_a = a_p.copy()
    
    return best_a, best_obj

"""
Step-function coefficient optimization using LP2 (min-peak) formulation,
FFT-based subgradient with threshold active sets, spectral initialization,
multi-scale exploration, and adaptive perturbation escapes.
Key innovations over previous: LP2 (min t s.t. conv(a,g)<=t, sum(g)=1)
gives more targeted descent directions; threshold-based active sets capture
the true non-smoothness; multi-n interpolation with cubic splines explores
the length dimension effectively.

import numpy as np
from scipy.optimize import linprog
from scipy.sparse import coo_matrix, hstack, eye as speye
from scipy.interpolate import interp1d
import time


def compute_conv(a):
    """Compute linear convolution a*a via FFT."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    conv = np.fft.irfft(fa * fa, size)
    return conv[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    S = float(np.sum(a))
    if S < 1e-10:
        return np.inf
    conv = compute_conv(a)
    M = float(np.max(conv))
    return 2.0 * n * M / (S * S)


def build_toeplitz_sparse(a):
    """Build sparse Toeplitz matrix T where (T@g)[k] = conv(a,g)[k]."""
    n = len(a)
    rows = []
    cols = []
    vals = []
    for j in range(n):
        start = j
        end = min(j + n, 2 * n - 1)
        length = end - start
        rows.extend(range(start, end))
        cols.extend([j] * length)
        vals.extend(list(a[:length]))
    A = coo_matrix((vals, (rows, cols)), shape=(2 * n - 1, n)).tocsr()
    return A


def lp_min_peak(a, time_limit=8):
    """
    LP2: min t s.t. conv(a,g)[k] <= t for all k, sum(g)=1, g>=0.
    Returns optimal g normalized to sum=1.
    """
    n = len(a)
    A = build_toeplitz_sparse(a)
    c = np.zeros(n + 1)
    c[-1] = 1.0

    # A_aug: (2n-1) x (n+1), last column is -1
    rows_t = list(range(2 * n - 1))
    cols_t = [n] * (2 * n - 1)
    vals_t = [-1.0] * (2 * n - 1)
    A_t = coo_matrix((vals_t, (rows_t, cols_t)), shape=(2 * n - 1, n + 1)).tocsr()
    A_aug = hstack([A, A_t]).tocsr()

    b_ub = np.zeros(2 * n - 1)
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])
    bounds = [(0, None)] * (n + 1)

    try:
        res = linprog(c, A_ub=A_aug, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': time_limit})
        if res.status == 0:
            g = np.maximum(res.x[:n], 0)
            S = np.sum(g)
            if S > 1e-10:
                return g / S
    except Exception:
        pass
    return None


def lp_max_sum(a, M, time_limit=8):
    """
    LP1: max sum(g) s.t. conv(a,g)[k] <= M for all k, g>=0.
    Returns optimal g normalized to sum=1.
    """
    n = len(a)
    A = build_toeplitz_sparse(a)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    try:
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': time_limit})
        if res.status == 0:
            g = np.maximum(res.x, 0)
            S = np.sum(g)
            if S > 1e-10:
                return g / S
    except Exception:
        pass
    return None


def compute_subgrad(a, threshold_frac=0.93):
    """Compute subgradient of max(conv(a,a)) using threshold-based active set."""
    n = len(a)
    conv = compute_conv(a)
    M = float(np.max(conv))
    threshold = M * threshold_frac
    active_idx = np.where(conv >= threshold)[0]

    if len(active_idx) == 0:
        active_idx = np.argsort(conv)[-10:]

    grad = np.zeros(n)
    for k in active_idx:
        i_start = max(0, k - n + 1)
        i_end = min(k, n - 1)
        if i_start <= i_end:
            grad[i_start:i_end + 1] += 2.0 * a[k - i_end:k - i_start - 1:-1]

    grad /= len(active_idx)
    return grad, M, conv


def momentum_subgradient(a, max_iters=5000, time_budget=10.0, lr0=0.08, momentum=0.82):
    """Momentum subgradient descent with adaptive step and perturbation escapes."""
    n = len(a)
    a = a.copy().astype(np.float64)
    S = np.sum(a)
    if S < 1e-10:
        return a
    a = a / S

    v = np.zeros(n)
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    stall = 0

    for it in range(max_iters):
        if time.monotonic() - start > time_budget:
            break

        grad, M, conv = compute_subgrad(a)
        grad_centered = grad - np.mean(grad)

        v = momentum * v + (1.0 - momentum) * grad_centered
        lr = lr0 / (1.0 + 0.003 * it)

        a_new = a - lr * v
        a_new = np.maximum(a_new, 0)
        S_new = np.sum(a_new)
        if S_new < 1e-10:
            break
        a_new = a_new / S_new

        obj = compute_obj(a_new)
        if obj < best_obj - 1e-13:
            best_obj = obj
            best_a = a_new.copy()
            a = a_new
            stall = 0
        else:
            stall += 1
            if stall > 40:
                rng = np.random.default_rng(it * 31 + 7)
                scale = min(0.08, 0.02 + 0.01 * (stall // 40))

                a = np.maximum(a, 0)
                a = a / np.sum(a)
                v = np.zeros(n)
                stall = 0
                lr0 *= 0.85
                if lr0 < 1e-9:
                    break

def spectral_init(n, n_modes=12, seed=0):
    """Spectral initialization with raised cosine envelope + random harmonics."""
    rng = np.random.default_rng(seed)
    k = np.arange(n)
    x = 2.0 * k / (n - 1) - 1.0
    alpha = rng.uniform(0.8, 2.0)
    a = 0.5 * (1.0 + np.cos(np.pi * x)) ** alpha
    for m in range(1, n_modes + 1):
        c = rng.uniform(-0.35, 0.35) / m
        phase = rng.uniform(0, 2 * np.pi)
        a += c * np.cos(2 * np.pi * m * k / n + phase)
    a = np.maximum(a, 1e-10)
    S = np.sum(a)
    return a / S


def random_smooth_init(n, seed=0):
    """Random smooth initialization via spline interpolation of random points."""
    rng = np.random.default_rng(seed)
    n_pts = rng.integers(5, max(6, n // 10))
    pts_x = np.sort(rng.uniform(0, 1, n_pts))
    pts_y = np.abs(rng.normal(1.0, 0.4, n_pts))
    pts_y = np.maximum(pts_y, 0.1)
    try:
        f = interp1d(pts_x, pts_y, kind='cubic', fill_value='extrapolate')
        a = np.maximum(f(np.linspace(0, 1, n)), 1e-10)
    except Exception:
        a = np.ones(n)
    S = np.sum(a)
    return a / S

                a = np.maximum(a, 0)
                a = a / np.sum(a)

    return best_a, best_obj


def full_refine(a, time_budget, verbose=False):
    """Combined LP + subgradient refinement pipeline."""
    n = len(a)
    a = a.copy().astype(np.float64)
    S = np.sum(a)
    if S < 1e-10:
        return a, np.inf
    a = a / S

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    rnd = 0

    while time.monotonic() - start < time_budget - 5:
        rnd += 1
        remaining = time_budget - (time.monotonic() - start)
        if remaining < 10:
            break

        # LP phase
        if n <= 900:
            lp_t = min(remaining * 0.4, 25.0)
            a_lp, obj_lp = refine_with_lp(best_a, max_iters=20, time_budget=lp_t, use_lp2=True)
            if obj_lp < best_obj:
                best_obj = obj_lp
                best_a = a_lp.copy()
                if verbose:
                    print(f"    R{rnd} LP2: {best_obj:.6f}")

        # Subgradient phase
        remaining = time_budget - (time.monotonic() - start)
        if remaining > 8:
            sg_t = min(remaining * 0.2, 12.0)
            a_sg = momentum_subgradient(best_a, max_iters=int(sg_t * 40),
                                         time_budget=sg_t, lr0=0.06)
            obj_sg = compute_obj(a_sg)
            if obj_sg < best_obj:
                best_obj = obj_sg
                best_a = a_sg.copy()
                if verbose:
                    print(f"    R{rnd} SG: {best_obj:.6f}")

        # Perturbation escape
        remaining = time_budget - (time.monotonic() - start)
        if remaining > 15:
            rng = np.random.default_rng(rnd * 42 + 17)
            a_pert = best_a * np.exp(rng.normal(0, 0.07, n))
            a_pert = np.maximum(a_pert, 0)
            S_p = np.sum(a_pert)
            if S_p > 1e-10:
                a_pert = a_pert / S_p
                if n <= 700:
                    a_p, obj_p = refine_with_lp(a_pert, max_iters=12,
                                                 time_budget=min(10.0, remaining * 0.2),
                                                 use_lp2=True)
                    if obj_p < best_obj:
                        best_obj = obj_p
                        best_a = a_p.copy()
                        if verbose:
                            print(f"    R{rnd} Pert: {best_obj:.6f}")

    return best_a, best_obj

    """
    Main search: LP2-based refinement with spectral init, multi-n exploration,

    """

    deadline = start_time + float(budget_s) - 20.0

    best_seq = None
    best_score = np.inf

    # Load parent
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64)
        parent = np.clip(parent, 0, 1000.0)
        if parent.size > 0 and np.isfinite(parent).all():
            S = np.sum(parent)
            if S > 1e-10:
                best_seq = parent / S
                best_score = compute_obj(best_seq)
                print(f"Parent (n={parent.size}): obj={best_score:.6f}")
    except Exception:
        pass

    if best_seq is None:
        best_seq = spectral_init(500, n_modes=10, seed=seed)
        best_score = compute_obj(best_seq)
        print(f"Initial spectral (n=500): obj={best_score:.6f}")

    n_best = len(best_seq)

    print("\n=== Phase 1: Intensive LP2 refinement of parent ===")
    remaining = deadline - time.monotonic()
    if remaining > 30:
        alloc = min(remaining * 0.3, 120.0)
        a_ref, obj_ref = full_refine(best_seq, time_budget=alloc, verbose=True)
        if obj_ref < best_score:
            best_score = obj_ref
            best_seq = a_ref.copy()
            print(f"  After Phase 1: {best_score:.6f} ** NEW BEST **")

    # === Phase 2: Multi-n exploration ===
    print("\n=== Phase 2: Multi-n exploration ===")

        300, 400, 500, 600, 700, 800, 900,

            obj_quick = compute_obj(a_quick)

            best_local = a_quick if obj_quick < obj_init else a_init
            obj_local = min(obj_quick, obj_init)
            candidates.append((best_local, obj_local, n))

            if obj_local < best_score:
                best_score = obj_local
