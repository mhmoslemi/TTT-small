"""
LP-based alternating optimization for minimizing 2*n*max(conv(a,a))/(sum(a))^2.
Uses the best-response LP to iteratively improve the sequence, with adaptive
perturbation and multiple initialization strategies.
"""

import numpy as np
from scipy.optimize import linprog
from scipy.fft import rfft, irfft
import time


def compute_conv(a):
    """Compute linear convolution a*a using FFT."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    conv = irfft(fa * fa, size)
    return conv[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    conv = compute_conv(a)
    max_b = float(np.max(conv))
    sum_a = float(np.sum(a))
    if sum_a < 0.01:
        return np.inf
    return 2.0 * n * max_b / (sum_a * sum_a)


def build_toeplitz(a):
    """Build matrix A where (A@g)[k] = (conv(a,g))[k]."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        A[j:j + n, j] = a
    return A


def solve_lp(a, M, time_limit=15):
    """LP: max sum(g) s.t. (a*g)[k] <= M for all k, g >= 0. Returns normalized g or None."""
    n = len(a)
    A = build_toeplitz(a)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': time_limit})
        if res.status == 0:
            g = np.maximum(res.x, 0)
            s = float(np.sum(g))
            if s > 1e-10:
                return g / s
    except Exception:
        pass
    return None

    stall = 0

        if time.monotonic() - start > time_budget:
            break
        
        conv = compute_conv(a)
        M = float(np.max(conv))
        
        remaining_lp = time_budget - (time.monotonic() - start)
        if remaining_lp < 2:
            break
        lp_time = min(20.0, max(3.0, remaining_lp * 0.1))
        
        g = solve_lp(a, M, time_limit=lp_time)
        if g is None:
            stall += 1
            if stall > 10:
                break
            continue
        
        # Line search
        improved = False
        # Try full step first, then back off

            a_try = (1.0 - t) * a + t * g
