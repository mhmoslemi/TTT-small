def generate_asymmetric_inits(n, rng, parent=None):
    """Generate diverse asymmetric initializations."""
    inits = []
    t = np.linspace(0, 1, n, endpoint=False)

    if parent is not None:
        p = resample_seq(parent, n)
        inits.append(p)
        inits.append(np.maximum(p + rng.normal(0, 0.05, n), 0.01))
        inits.append(np.maximum(p + rng.normal(0, 0.15, n), 0.01))

    # Exponential decays
    for rate in [1.0, 2.0, 3.0, 5.0]:
        inits.append(np.exp(-rate * t) + 0.01)
        inits.append(np.exp(-rate * (1 - t)) + 0.01)

    # Power ramps
    for pwr in [0.5, 1.0, 2.0, 3.0]:
        inits.append(t ** pwr + 0.01)
        inits.append((1 - t) ** pwr + 0.01)

    # Sine sweeps
    for freq in [1, 2, 3]:
        inits.append(np.maximum(1.0 + 0.5 * np.sin(np.pi * t * freq), 0.01))
        inits.append(np.maximum(0.5 + 0.5 * np.cos(np.pi * t * freq), 0.01))

    # Gaussian bumps at various positions
    for center in [0.2, 0.3, 0.5, 0.7, 0.8]:
        for width in [0.1, 0.2, 0.3]:
            inits.append(np.exp(-((t - center) / width) ** 2) + 0.05)

    # Double peaks (asymmetric)
    inits.append(np.exp(-((t - 0.2) / 0.15) ** 2) + 0.3 * np.exp(-((t - 0.8) / 0.1) ** 2) + 0.01)
    inits.append(0.3 * np.exp(-((t - 0.2) / 0.1) ** 2) + np.exp(-((t - 0.8) / 0.15) ** 2) + 0.01)

    # Step functions
    for split in [0.3, 0.4, 0.5, 0.6, 0.7]:
        step = np.where(t < split, 1.0, 0.3)
        inits.append(step)

    # Smooth random
    ksize = max(3, n // 20)
    kernel = np.ones(ksize) / ksize
    smooth = np.convolve(rng.uniform(0.01, 1.0, n), kernel, mode='same')
    inits.append(np.maximum(smooth, 0.01))

    # Chirp
    inits.append(np.maximum(1.0 + 0.4 * np.cos(2 * np.pi * 5 * t ** 2), 0.01))

    return inits


class ScoreTracker:
    """Tracks the best sequence found so far."""

    def __init__(self):
        self.best = None
        self.best_score = float('inf')
        self.t_start = time.monotonic()

    def update(self, a, label=""):
        a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
        if a.size == 0 or a.sum() < 0.01:
            return
        score = float(evaluate_sequence(a.tolist()))
        if np.isfinite(score) and score < self.best_score:
            self.best = a.copy()
            self.best_score = score
            elapsed = time.monotonic() - self.t_start
            print(f"  ** J={score:.10f} n={a.size} t={elapsed:.0f}s {label}", flush=True)

    def report(self, label=""):
        if self.best is not None:
            print(f"  {label} best: J={self.best_score:.10f} n={self.best.size}", flush=True)


def polish_with_cd(a, max_sweeps=200, deadline=None):
    """Polish a sequence with coordinate descent, returning best found."""
    a = np.maximum(a.astype(np.float64), 0.0)
    a = a / a.sum()
    result, J = coord_descent(a, max_sweeps=max_sweeps, deadline=deadline)
    return result, J


def propose_candidate(seed=42, budget_s=1500, **kwargs):
    """
    Multi-phase search using exact coordinate descent:
    Phase 1: CD on parent sequence
    Phase 2: Multi-length exploration with diverse initializations
    Phase 3: Refined CD on best candidate with "shakes"
    Phase 4: Targeted asymmetric optimization
    Phase 5: Final ultra-polish
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + float(budget_s) - 10.0
    tracker = ScoreTracker()

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        if p.ndim == 1 and p.size > 0 and np.all(np.isfinite(p)):
            parent = np.maximum(p, 0.0)
            tracker.update(parent, "(parent)")
            print(f"Parent loaded: n={parent.size}, J={tracker.best_score:.8f}", flush=True)
    except (NameError, AttributeError):
        pass

    if parent is None:
        parent = np.ones(256)
        tracker.update(parent, "(default)")

    best = tracker.best.copy()
    best_n = best.size

    # === PHASE 1: CD on parent ===
    p1_deadline = min(time.monotonic() + 180, deadline - 300)
    print(f"\n{'='*60}", flush=True)
    print(f"PHASE 1: CD on parent (n={best_n}, budget={180}s)", flush=True)
    print(f"{'='*60}", flush=True)
    t0 = time.monotonic()

    a, J = polish_with_cd(best, max_sweeps=500, deadline=p1_deadline)
    tracker.update(a, "(CD parent)")
    best = a.copy()

    print(f"Phase 1 done: J={tracker.best_score:.10f} ({time.monotonic()-t0:.1f}s)", flush=True)

    # === PHASE 2: Multi-length exploration ===
    p2_deadline = min(time.monotonic() + 400, deadline - 200)
    print(f"\n{'='*60}", flush=True)
    print(f"PHASE 2: Multi-length exploration (budget=400s)", flush=True)
    print(f"{'='*60}", flush=True)
    t0 = time.monotonic()

    lengths = [256, 384, 448, 512, 576, 640, 768, 896]
    # Allocate time per length
    time_per = 350.0 / len(lengths)

    for n in lengths:
        if time.monotonic() > p2_deadline:
            break
        n_deadline = min(time.monotonic() + time_per, p2_deadline)

        inits = generate_asymmetric_inits(n, rng, best)
        best_for_n = None
        best_J_for_n = float('inf')

        for idx, a0 in enumerate(inits):
            if time.monotonic() > n_deadline:
                break
            a0 = np.maximum(a0.astype(np.float64), 0.01)
            a0 = a0 / a0.sum()

            # Quick CD (fewer sweeps for exploration)
            a, J = coord_descent(a0, max_sweeps=80, deadline=n_deadline)
            if J < best_J_for_n:
                best_J_for_n = J
                best_for_n = a.copy()

        if best_for_n is not None:
            tracker.update(best_for_n, f"(n={n})")
            if tracker.best_score < compute_J(best):
                best = tracker.best.copy()

    print(f"Phase 2 done: best J={tracker.best_score:.10f} ({time.monotonic()-t0:.1f}s)", flush=True)

    # === PHASE 3: Refined CD with shakes on best ===
    p3_deadline = min(time.monotonic() + 350, deadline - 100)
    print(f"\n{'='*60}", flush=True)
    print(f"PHASE 3: Refined CD + shakes (n={best.size}, budget=350s)", flush=True)
    print(f"{'='*60}", flush=True)
    t0 = time.monotonic()

    n_best = best.size
    shake_temp = 0.1

    for shake_round in range(15):
        if time.monotonic() > p3_deadline:
            break

        # CD polish
        a, J = coord_descent(best, max_sweeps=200, deadline=p3_deadline)
        tracker.update(a, f"(shake {shake_round})")

        # Shake
        a_shake = best.copy()
        noise = rng.normal(0, shake_temp * 0.02, n_best)
        a_shake = np.maximum(a_shake + noise, 0.01)
        a_shake = a_shake / a_shake.sum()

        a, J = coord_descent(a_shake, max_sweeps=100, deadline=p3_deadline)
        tracker.update(a, f"(shake {shake_round} recover)")

        if tracker.best is not None:
            best = tracker.best.copy()

        shake_temp *= 0.85

    print(f"Phase 3 done: best J={tracker.best_score:.10f} ({time.monotonic()-t0:.1f}s)", flush=True)

    # === PHASE 4: Targeted asymmetric + zeroing experiments ===
    p4_deadline = min(time.monotonic() + 200, deadline - 50)
    print(f"\n{'='*60}", flush=True)
    print(f"PHASE 4: Asymmetric + zeroing (budget=200s)", flush=True)
    print(f"{'='*60}", flush=True)
    t0 = time.monotonic()

    n_best = best.size

    # Try zeroing certain positions and re-optimizing
    for zero_strategy in range(5):
        if time.monotonic() > p4_deadline:
            break

        a = best.copy()
        # Zero out a random subset
        n_zero = n_best // (4 + zero_strategy)
        zero_mask = np.zeros(n_best, dtype=bool)
        zero_idx = rng.choice(n_best, size=n_zero, replace=False)
        zero_mask[zero_idx] = True
        a[zero_mask] = 0.0
        a = a / max(a.sum(), 1e-10)

        a, J = coord_descent(a, max_sweeps=150, deadline=p4_deadline)
        tracker.update(a, f"(zero {zero_strategy})")

    # Try strongly asymmetric starts
    for asym in range(4):
        if time.monotonic() > p4_deadline:
            break
        t = np.linspace(0, 1, n_best, endpoint=False)
        if asym == 0:
            a0 = np.exp(-4 * t) + 0.01
        elif asym == 1:
            a0 = np.exp(-4 * (1 - t)) + 0.01
        elif asym == 2:
            a0 = (1 - t) ** 2 + 0.01
        else:
            a0 = t ** 2 + 0.01
        a0 = a0 / a0.sum()
        a, J = coord_descent(a0, max_sweeps=150, deadline=p4_deadline)
        tracker.update(a, f"(asym {asym})")

    if tracker.best is not None:
        best = tracker.best.copy()

    print(f"Phase 4 done: best J={tracker.best_score:.10f} ({time.monotonic()-t0:.1f}s)", flush=True)

    # === PHASE 5: Final ultra-polish ===
    remaining = max(0, deadline - time.monotonic())
    print(f"\n{'='*60}", flush=True)
    print(f"PHASE 5: Final ultra-polish (remaining={remaining:.0f}s)", flush=True)
    print(f"{'='*60}", flush=True)
    t0 = time.monotonic()

    if tracker.best is not None:
        # Long CD run
        a, J = coord_descent(tracker.best, max_sweeps=2000, deadline=deadline)
        tracker.update(a, "(final CD)")

        # One more shake + CD
        if time.monotonic() < deadline - 30:
            n_b = tracker.best.size
            a_shake = tracker.best + rng.normal(0, 0.005, n_b)
            a_shake = np.maximum(a_shake, 0.001)
            a_shake = a_shake / a_shake.sum()
            a, J = coord_descent(a_shake, max_sweeps=300, deadline=deadline)
            tracker.update(a, "(final shake)")

    print(f"Phase 5 done: ({time.monotonic()-t0:.1f}s)", flush=True)

    # Fallback
    if tracker.best is None:
        tracker.best = parent
        tracker.best_score = compute_J(parent)

    print(f"\n{'='*60}", flush=True)
    print(f"FINAL: J={tracker.best_score:.10f} n={tracker.best.size} "
          f"time={time.monotonic()-tracker.t_start:.0f}s", flush=True)
    print(f"{'='*60}", flush=True)

    return [float(x) for x in tracker.best]
