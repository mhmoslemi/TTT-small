**Spectral approach:**

For a sequence a of length n, the linear convolution a*a has length 2n-1. If we compute the DFT of size N ≥ 2n-1:
FFT(a*a) = FFT(a) * FFT(a) (element-wise, since the FFT of a convolution is the product of FFTs)

So |FFT(a)|^2 = FFT(a*a) (for the linear convolution, with zero-padding).

This means: the peak of the linear convolution is the peak of |FFT(a)|^2 (in the frequency domain).

So we want to minimize max_ω |A(ω)|^2 where A(ω) = Σ a_k e^{-iωk/A_N} for N-point DFT.

This is equivalent to minimizing max_ω |A(ω)|, which is a "peak amplitude" problem in the frequency domain.

By Parseval's theorem: Σ_k c_k = Σ_ω |A(ω)|^2 = N * Σ_k a_k^2 (for N-point DFT with appropriate normalization).

So we want to make |A(ω)| as flat as possible while keeping Σ a_k^2 fixed (or minimizing the peak while keeping the integral fixed).

This is related to "flat-top" filter design! The optimal solution would be a sequence whose DFT has a flat magnitude response.

For the DFT to be perfectly flat, we'd need a "Dirichlet kernel" inverse, which gives a rectangular window in frequency = a sinc-like function in time. But we need a ≥ 0, which constrains the solution.

**Practical spectral approach:**
1. Compute A(ω) = DFT(a)
2. The peak of |A(ω)|^2 corresponds to the peak of the convolution
3. To reduce the peak, we can:
   a. Apply a frequency-domain "notch" at the peak frequency
   b. Then transform back and enforce non-negativity

This is similar to the "iterative reweighted" methods used in compressed sensing.

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using LP-alternating with dual-guided
subgradient, Barzilai-Borwein adaptive steps, multi-start with diverse

import numpy as np

def compute_conv(a):
    """Compute full linear convolution a*a via FFT."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    conv = np.fft.irfft(fa * fa, size)
    return conv[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    S = float(np.sum(a))
    if S < 1e-10:
        return np.inf
    conv = compute_conv(a)
    M = float(np.max(conv))
    return 2.0 * n * M / (S * S)


def build_toeplitz(a):
    """Build sparse matrix T where T[k,j] = a[k-j] if 0<=k-j<n else 0."""
    n = len(a)
    rows = np.concatenate([np.arange(lo, hi + 1) for k in range(2 * n - 1)
                           for lo, hi in [(max(0, k - (n - 1)), min(n - 1, k))]]
                          if False) or None
    # More efficient construction
    all_rows = []
    all_cols = []
    all_vals = []
    for k in range(2 * n - 1):
        lo = max(0, k - (n - 1))
        hi = min(n - 1, k)
        js = np.arange(lo, hi + 1)
        all_rows.extend(np.full(len(js), k).tolist())
        all_cols.extend(js.tolist())
        all_vals.extend(a[ks].tolist() for ks in (k - js))
    return coo_matrix((all_vals, (all_rows, all_cols)), shape=(2 * n - 1, n)).tocsr()


def solve_lp(a, M, time_limit=8):
    """Solve LP: max sum(g) s.t. T(a)*g <= M, g >= 0. Returns (g, dual)."""
    n = len(a)
    T = build_toeplitz(a)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c, A_ub=T, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': time_limit})
        if res.status == 0:
            g = np.maximum(res.x, 0)
            dual = getattr(res, 'ineqlin', None)
            return g, dual
    except Exception:
        pass
    return None, None


def subgradient_at(a, conv, topk=15):
    """Compute average subgradient from top-K active lags."""
    n = len(a)
    active = np.argsort(conv)[-topk:]
    grad = np.zeros(n)
    for k in active:
        lo = max(0, k - (n - 1))
        hi = min(n - 1, k)
        for j in range(lo, hi + 1):
            grad[j] += 2.0 * a[k - j]
    grad /= len(active)
    return grad


def dual_weighted_gradient(a, conv, dual, topk=20):
    """Compute subgradient weighted by LP dual variables."""
    n = len(a)
    if dual is None or len(dual) < 2 * n - 1:
        return subgradient_at(a, conv, topk)
    weights = np.maximum(dual, 0)
    total = np.sum(weights)
    if total < 1e-10:
        return subgradient_at(a, conv, topk)
    weights = weights / total
    # Only use top contributions
    top_idx = np.argsort(weights)[-topk:]
    grad = np.zeros(n)
    for k in top_idx:
        w = weights[k]
        if w < 1e-12:
            continue
        lo = max(0, k - (n - 1))
        hi = min(n - 1, k)
        for j in range(lo, hi + 1):
            grad[j] += w * 2.0 * a[k - j]
    return grad

def ramp_init(n):
    """Linear ramp initialization."""
    a = np.arange(1, n + 1, dtype=np.float64)
    return a / np.sum(a)


def exp_decay_init(n, alpha=0.5):
    """Exponential decay initialization."""
    a = np.exp(-alpha * np.arange(n, dtype=np.float64))
    return a / np.sum(a)


def cosine_init(n, seed=0):
    """Smooth cosine-series initialization."""
    rng = np.random.default_rng(seed)
    k = np.arange(n, dtype=np.float64)
    a = np.ones(n)
    for m in range(1, 7):
        c = rng.uniform(-0.4, 0.4) / m
        phase = rng.uniform(0, 2 * np.pi)
        a += c * np.cos(2 * np.pi * m * k / n + phase)
    a = np.maximum(a, 1e-10)
    return a / np.sum(a)


def random_init(n, seed=0):
    """Random smooth initialization."""
    rng = np.random.default_rng(seed)
    a = rng.random(n)
    # Smooth it
    kernel = np.ones(5) / 5
    a = np.convolve(a, kernel, mode='same')
    a = np.maximum(a, 1e-10)
    return a / np.sum(a)


def quadratic_init(n):
    """Quadratic (parabola) initialization."""
    k = np.arange(n, dtype=np.float64)
    center = (n - 1) / 2.0
    a = 1.0 + (k - center) / center
    a = np.maximum(a, 0.01)
    return a / np.sum(a)
