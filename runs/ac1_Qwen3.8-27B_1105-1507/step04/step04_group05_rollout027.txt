```python
        if np.isfinite(score) and score < self.best_score:

def cd_pass_vectorized(a, rng, n_golden=25):
    """One pass of coordinate descent with vectorized 1D search."""
    a = np.maximum(a.astype(np.float64), 0.0)
    S = a.sum()
    if S < 1e-10:
        return a, compute_J(a)
    n = len(a)
    b = fast_conv(a).copy()
    current_max = b.max()
    current_J = 2.0 * n * current_max / (S * S)
    
    indices = rng.permutation(n)
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    
    for i in indices:
        ai = a[i]
        
        # Precompute unchanged max
        # Unchanged: k < i or k >= i+n (capped at 2n-2)
        k_end = min(2*n-1, i+n)
        if i > 0:
            left_max = b[:i].max()
        else:
            left_max = 0.0
        if k_end < 2*n-1:
            right_max = b[k_end:].max()
        else:
            right_max = 0.0
        unchanged_max = max(left_max, right_max)
        
        # Search range for x (new value of a[i])
        if ai > 1e-10:
            lo = max(0.0, ai * 0.1)
            hi = max(ai * 3.0, S / n * 4.0)
        else:
            lo = 0.0
            hi = max(S / n * 4.0, 0.01)
        
        # Precompute the coefficient vectors for the affected range
        ks = np.arange(i, k_end)
        kis = np.arange(0, k_end - i)
        a_kis = a[kis]
        b_ks = b[ks].copy()
        
        # Determine which position is 2i in the array
        pos_2i = None
        if 2*i >= i and 2*i < k_end:
            pos_2i = 2*i - i  # = i
        
        c1 = 2.0 * a_kis  # coefficient of x
        c0 = b_ks - 2.0 * ai * a_kis  # constant term (for non-2i positions)
        # For 2i position, the formula is different: b[2i] - ai^2 + x^2
        if pos_2i is not None:
            c0[pos_2i] = b_ks[pos_2i] - ai * ai
            # For this position, the term is x^2 instead of 2*x*a[ki]
            # So we need to handle it separately
        
        def obj(x):
            new_S = S - ai + x
            if new_S < 1e-12:
                return 1e15
            vals = c0 + c1 * x
            if pos_2i is not None:
                vals[pos_2i] = c0[pos_2i] + x * x
            max_c = max(unchanged_max, vals.max())
            return 2.0 * n * max_c / (new_S * new_S)
        
        # Golden section search
        c_pt = hi - gr * (hi - lo)
        d_pt = lo + gr * (hi - lo)
        fc = obj(c_pt)
        fd = obj(d_pt)
        
        for _ in range(n_golden):
            if fc < fd:
                hi = d_pt
                d_pt = c_pt
                fd = fc
                c_pt = hi - gr * (hi - lo)
                fc = obj(c_pt)
            else:
                lo = c_pt
                c_pt = d_pt
                fc = fd
                d_pt = lo + gr * (hi - lo)
                fd = obj(d_pt)
        
        x_opt = max(0.0, (lo + hi) / 2.0)
        
        if abs(x_opt - ai) > 1e-12:
            # Update a[i]
            a[i] = x_opt
            # Update S
            S = S + (x_opt - ai)
            # Update b incrementally
            for k_idx in range(len(ks)):
                k = ks[k_idx]
                ki = kis[k_idx]
                if k == 2*i:
                    b[k] = b[k] - ai*ai + x_opt*x_opt
                else:
                    b[k] = b[k] - 2.0*ai*a[ki] + 2.0*x_opt*a[ki]
    
    # Final verification
    b_check = fast_conv(a)
    new_J = 2.0 * n * b_check.max() / (S * S)
    return a, new_J

def run_cd(a, n_passes, deadline, tracker, rng):
    """Run multiple passes of coordinate descent."""
    a = np.maximum(a.astype(np.float64), 0.0)
    a = a / a.sum()
    current_J = compute_J(a)
    tracker.update(a)
    
    no_improve = 0
    for p in range(n_passes):
        if time.monotonic() > deadline:
            break
        t0 = time.monotonic()
        a, new_J = cd_pass_vectorized(a, rng)
        dt = time.monotonic() - t0
        
        if new_J < current_J - 1e-14:
            current_J = new_J
            tracker.update(a)
            no_improve = 0
        else:
            no_improve += 1
        
        if p % 10 == 0 or p == n_passes - 1:
            print(f"    CD pass {p}: J={current_J:.8f} ({dt:.3f}s/pass)", flush=True)
        
        if no_improve >= 15:
            break
    
    tracker.update(a)
    return a, current_J

def build_A_sparse(a):
    n = len(a)
    A = lil_matrix((2*n-1, n))
    for k in range(2*n-1):
        jl = max(0, k - n + 1)
        jh = min(n - 1, k)
        for j in range(jl, jh + 1):
            A[k, j] = a[k - j]
    return A.tocsr()

def lp_best_response(a, M):
    n = len(a)
    c = -np.ones(n)
    A = build_A_sparse(a)
    b_ub = np.full(2*n-1, M)
    bounds = [(0.0, None)] * n
    r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
    if r.success and r.x is not None:
        g = np.maximum(r.x, 0.0)
        if g.sum() > 1e-10:
            return g
    return None

def resample_seq(src, dest_n):
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)

def generate_inits(n, rng):
    """Generate diverse initial sequences."""
    inits = []
    t = np.linspace(0, 1, n, endpoint=False)
    
    inits.append(np.ones(n))
    inits.append(rng.uniform(0.5, 1.0, n))
    
    # Cosine variations
    for freq in [2, 3, 5]:
        for amp in [0.2, 0.4]:
            inits.append(1.0 + amp * np.cos(2*np.pi*freq*t + rng.uniform(0, 2*np.pi)))
    
    # Two peaks
    inits.append(np.exp(-20*(t-0.3)**2) + np.exp(-20*(t-0.7)**2) + 0.2)
    
    # Edge-enhanced
    inits.append(0.5 + 0.5*np.cos(2*np.pi*t)**2)
    
    # Random smooth
    x = rng.normal(0, 1, n)
    kernel = np.exp(-np.arange(-10, 11)**2 / 20.0)
    kernel /= kernel.sum()
    inits.append(np.convolve(x, kernel, mode='same'))
    
    return [np.maximum(x, 0.01) for x in inits]
```

```python
def subgradient_step(a, lr, rng):
    a = np.maximum(a, 0.0)
    S = a.sum()
    n = len(a)
    b = fast_conv(a)
    max_c = b.max()
    
    # Active set
    tol = 1e-8 * max_c
    active = np.where(b >= max_c - tol)[0]
    
    # Subgradient
    s = np.zeros(n)
    for k in active:
        # s[i] += 2*a[k-i] for valid i
        i_min = max(0, k - n + 1)
        i_max = min(n-1, k)
        for i in range(i_min, i_max + 1):
            s[i] += 2.0 * a[k-i]
    
    # Gradient of J
    grad = 2.0 * n * (s * S - 2.0 * max_c) / (S**3)
    
    # Projected gradient step on simplex
    a_new = a - lr * grad
    # Project onto simplex (a >= 0, sum = 1)
    a_new = np.maximum(a_new, 0)
    if a_new.sum() > 0:
        a_new = a_new / a_new.sum()
    else:
        a_new = a.copy()
    
    return a_new
```

Actually, for the simplex projection, the correct projection of y onto {x >= 0, sum(x) = 1} is not just normalization after clipping. The correct projection solves:
min ||x - y||^2 s.t. x >= 0, sum(x) = 1

This can be solved by: x[i] = max(y[i] - tau, 0) where tau is chosen so that sum(x) = 1. We can find tau by bisection.

Let me implement this properly:

```python
def project_simplex(y):
    """Project y onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(y)
    if y.sum() <= 0:
        return np.ones(n) / n
    # Bisection for tau
    lo = y.min() - 1
    hi = y.max()
    for _ in range(60):
        tau = (lo + hi) / 2
        x = np.maximum(y - tau, 0)
        s = x.sum()
        if s > 1:
            lo = tau
        else:
            hi = tau
    x = np.maximum(y - tau, 0)
    return x
```

This is O(n) per iteration of bisection, 60 iterations, so O(60n) total. Fine.

```python
"""
Minimizes J(a) = 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Fast vectorized coordinate descent with incremental convolution (primary)
2. LP mirror descent for global escape from local minima
3. Subgradient exploration with adaptive step sizes
4. Multi-length and multi-start diversification
Key innovation: O(n) vectorized 1D coordinate search enabling thousands of CD passes.

import numpy as np
from scipy.optimize import linprog
from scipy.sparse import lil_matrix
import time


def fast_conv(a):
    n = len(a)
    s = 1
    while s < 2 * n - 1:
        s <<= 1
    fa = np.fft.rfft(a, s)
    return np.fft.irfft(fa * fa, s)[:2 * n - 1]


def compute_J(a):
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))


def project_simplex(y):
    n = len(y)
    if y.sum() <= 0:
        return np.ones(n) / n
    lo = y.min() - 1.0
    hi = y.max() + 1.0
    for _ in range(80):
        tau = (lo + hi) / 2.0
        x = np.maximum(y - tau, 0.0)
        s = x.sum()
        if s > 1.0:
            lo = tau
        else:
            hi = tau
    x = np.maximum(y - tau, 0.0)
    return x

def cd_pass_vectorized(a, rng, n_golden=28):
    a = np.maximum(a.astype(np.float64), 0.0)
    S = a.sum()
    if S < 1e-10:
        return a, compute_J(a)
    n = len(a)
    b = fast_conv(a).copy()
    current_max = b.max()
    current_J = 2.0 * n * current_max / (S * S)

    indices = rng.permutation(n)
    gr = (np.sqrt(5.0) - 1.0) / 2.0

    for i in indices:
        ai = a[i]
        k_end = min(2 * n - 1, i + n)
        k_start = i

        if k_start > 0:
            left_max = b[:k_start].max()
        else:
            left_max = 0.0
        if k_end < 2 * n - 1:
            right_max = b[k_end:].max()
        else:
            right_max = 0.0
        unchanged_max = max(left_max, right_max)

        if ai > 1e-10:
            lo = max(0.0, ai * 0.05)
            hi = max(ai * 4.0, S / n * 5.0)
        else:
            lo = 0.0
            hi = max(S / n * 5.0, 0.01)

        kis = np.arange(0, k_end - k_start)
        a_kis = a[kis]
        b_ks = b[k_start:k_end]

        pos_2i = None
        if 2 * i >= k_start and 2 * i < k_end:
            pos_2i = 2 * i - k_start

        c1 = 2.0 * a_kis
        c0 = b_ks.copy() - 2.0 * ai * a_kis
        if pos_2i is not None:
            c0[pos_2i] = b_ks[pos_2i] - ai * ai

        def obj(x):
            new_S = S - ai + x
            if new_S < 1e-12:
                return 1e15
            vals = c0 + c1 * x
            if pos_2i is not None:
                vals[pos_2i] = c0[pos_2i] + x * x
            max_c = unchanged_max
            vm = vals.max()
            if vm > max_c:
                max_c = vm
            return 2.0 * n * max_c / (new_S * new_S)

        c_pt = hi - gr * (hi - lo)
        d_pt = lo + gr * (hi - lo)
        fc = obj(c_pt)
        fd = obj(d_pt)

        for _ in range(n_golden):
            if fc < fd:
                hi = d_pt
                d_pt = c_pt
                fd = fc
                c_pt = hi - gr * (hi - lo)
                fc = obj(c_pt)
            else:
                lo = c_pt
                c_pt = d_pt
                fc = fd
                d_pt = lo + gr * (hi - lo)
                fd = obj(d_pt)

        x_opt = max(0.0, (lo + hi) / 2.0)
        f_opt = obj(x_opt)
        f_cur = obj(ai) if ai >= lo and ai <= hi else f_opt

        if f_opt < f_cur - 1e-15:
            a[i] = x_opt
            S = S + (x_opt - ai)
            for idx in range(len(kis)):
                k = k_start + idx
                ki = kis[idx]
                if k == 2 * i:
                    b[k] = b[k] - ai * ai + x_opt * x_opt
                else:
                    b[k] = b[k] - 2.0 * ai * a[ki] + 2.0 * x_opt * a[ki]

    b_check = fast_conv(a)
    new_J = 2.0 * n * b_check.max() / (S * S)
    return a, new_J


def run_cd(a, n_passes, deadline, tracker, rng, verbose_every=20):
    a = np.maximum(a.astype(np.float64), 0.0)
    if a.sum() < 1e-10:
        return a, compute_J(a)
    a = a / a.sum()
    current_J = compute_J(a)
    tracker.update(a)

    no_improve = 0
    for p in range(n_passes):
        if time.monotonic() > deadline:
            break
        t0 = time.monotonic()
        a, new_J = cd_pass_vectorized(a, rng)
        dt = time.monotonic() - t0

        if new_J < current_J - 1e-14:
            current_J = new_J
            tracker.update(a)
            no_improve = 0
        else:
            no_improve += 1

        if p % verbose_every == 0 or p == n_passes - 1:
            print(f"    CD pass {p}: J={current_J:.8f} ({dt:.3f}s/pass)", flush=True)

        if no_improve >= 20:
            break

    tracker.update(a)
    return a, current_J


def build_A_sparse(a):
    n = len(a)
    A = lil_matrix((2 * n - 1, n))
    for k in range(2 * n - 1):
        jl = max(0, k - n + 1)
        jh = min(n - 1, k)
        for j in range(jl, jh + 1):
            A[k, j] = a[k - j]
    return A.tocsr()


def lp_best_response(a, M):
    n = len(a)
    c_vec = -np.ones(n)
    A = build_A_sparse(a)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    r = linprog(c_vec, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
    if r.success and r.x is not None:
        g = np.maximum(r.x, 0.0)
        if g.sum() > 1e-10:
            return g
    return None

def subgradient_explore(a, deadline, tracker, rng):
    a = np.maximum(a.astype(np.float64), 0.0)
    a = a / a.sum()
    n = len(a)
    current_J = compute_J(a)
    tracker.update(a)

    lr = 0.1 / np.sqrt(n)
    patience = 0

    while time.monotonic() < deadline:
        b = fast_conv(a)
        max_c = b.max()
        tol = 1e-6 * max_c
        active = np.where(b >= max_c - tol)[0]

        s = np.zeros(n)
        for k in active:
            i_min = max(0, k - n + 1)
            i_max = min(n - 1, k)
            if i_min <= i_max:
                s[i_min:i_max + 1] += 2.0 * a[k - i_max:i_min - 1 - 1][::-1] if i_max >= i_min else 0

        # Simpler:
        s = np.zeros(n)
        for k in active:
            for i in range(max(0, k - n + 1), min(n, k + 1)):
                ki = k - i
                if 0 <= ki < n:
                    s[i] += 2.0 * a[ki]

        S = a.sum()
        grad = 2.0 * n * (s * S - 2.0 * max_c) / (S ** 3)

        a_new = a - lr * grad
        a_new = project_simplex(a_new)

        J_new = compute_J(a_new)
        if J_new < current_J - 1e-14:
            a = a_new
            current_J = J_new
            tracker.update(a)
            patience = 0
            lr = min(lr * 1.1, 0.5)
        else:
            patience += 1
            lr *= 0.8

        if patience > 50:
            break

    tracker.update(a)
    return a, current_J


def resample_seq(src, dest_n):
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)


def generate_inits(n, rng):
    inits = []
    t = np.linspace(0, 1, n, endpoint=False)
    inits.append(np.ones(n))
    inits.append(rng.uniform(0.4, 1.0, n))
    for freq in [2, 3, 4, 5, 7]:
        inits.append(1.0 + 0.3 * np.cos(2 * np.pi * freq * t + rng.uniform(0, 6.28)))
    inits.append(np.exp(-15 * (t - 0.3) ** 2) + np.exp(-15 * (t - 0.7) ** 2) + 0.15)
    inits.append(0.5 + 0.5 * np.cos(2 * np.pi * t) ** 2)
    x = rng.normal(0, 1, n)
    kernel = np.exp(-np.arange(-15, 16) ** 2 / 30.0)
    kernel /= kernel.sum()
    inits.append(np.convolve(x, kernel, mode='same'))
    inits.append(1.0 - 0.4 * np.abs(2 * t - 1.0))
    return [np.maximum(x, 0.01) for x in inits]
```
