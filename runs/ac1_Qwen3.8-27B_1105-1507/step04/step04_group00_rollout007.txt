import numpy as np
from scipy.optimize import linprog
from scipy.sparse import csr_matrix
import time


def fast_conv(a):
    """Full self-convolution via FFT."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]


def compute_J(a):
    """Compute J(a) = 2*n*max(conv(a,a))/(sum(a))^2."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))


def build_convolution_matrix(n, a):
    """Build sparse matrix A where A[k,j] = a[k-j] if 0<=k-j<n, else 0."""
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        j_min = max(0, k - n + 1)
        j_max = min(n - 1, k)
        for j in range(j_min, j_max + 1):
            rows.append(k)
            cols.append(j)
            vals.append(a[k - j])
    A = csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))
    return A


def solve_lp(a, M):
    """
    Solve LP: max sum(g) s.t. (a*g)[k] <= M for all k, g >= 0.
    Returns (g, dual_y) or (None, None) on failure.
    """
    n = len(a)
    A = build_convolution_matrix(n, a)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    try:
        r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                    options={'time_limit': 30, 'dual_feasibility_tolerance': 1e-10,
                             'primal_feasibility_tolerance': 1e-10})
        if r.success and r.x is not None:
            g = np.maximum(r.x, 0)
            dual_y = r.ineqlin.marginals if r.ineqlin.marginals is not None else None
            if g.sum() > 1e-15:
                return g, dual_y
    except Exception:
        pass
    return None, None


def smooth_max_grad(a, tau):
    """Compute gradient of smooth max (log-sum-exp / tau) of conv(a,a)."""
    n = len(a)
    b = fast_conv(a)
    b_max = b.max()
    exp_b = np.exp(tau * (b - b_max))
    p = exp_b / exp_b.sum()

    grad = np.zeros(n)
    for k in range(2 * n - 1):
        pk = p[k]
        if pk < 1e-14:
            continue
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        grad[lo:hi + 1] += 2.0 * pk * a[k - hi:k - lo + 1][::-1]

    return grad, b


def run_smooth_gd(a, max_iters, deadline, tau_start=3.0, tau_end=60.0,
                  lr_start=0.25, lr_end=0.02, momentum_beta=0.85):
    """Smooth gradient descent with momentum and annealed tau/lr."""
    a = np.maximum(a.astype(np.float64).copy(), 1e-9)
    a = a / a.sum()
    n = len(a)
    best_a = a.copy()
    best_M = fast_conv(a).max()
    v = np.zeros(n)

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        frac = (it + 1) / max(1, max_iters)
        tau = tau_start + (tau_end - tau_start) * frac
        lr = lr_start * (lr_end / lr_start) ** frac

        grad, b = smooth_max_grad(a, tau)
        M = b.max()
        if M < best_M - 1e-12:
            best_M = M
            best_a = a.copy()

        v = momentum_beta * v + lr * grad
        a_new = a - v
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s < 1e-10:
            a_new = a.copy()
        else:
            a_new = a_new / s
        a = a_new

    return best_a


def run_lp_mirror_descent(a, max_iters, deadline, aggressive=False):
    """
    LP mirror descent with backtracking line search.
    If aggressive=True, uses tighter M targets to force faster convergence.
    """
    a = np.maximum(a.astype(np.float64).copy(), 1e-9)
    a = a / a.sum()
    n = len(a)
    best_a = a.copy()
    best_M = fast_conv(a).max()
    no_improve = 0
    stall_threshold = 30 if not aggressive else 15

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        b = fast_conv(a)
        M = float(b.max())

        # For aggressive mode, target slightly below current max
        M_target = M * (0.999 if aggressive else 1.0)

        g, dual_y = solve_lp(a, M_target)
        if g is None:
            # Try with slightly looser constraint
            g, dual_y = solve_lp(a, M * 1.001)
            if g is None:
                no_improve += 1
                if no_improve > stall_threshold:
                    break
                continue
            normal = it < max_iters // 2

        g = np.maximum(g, 0.0)
        g_sum = g.sum()
        if g_sum < 1e-12:
            no_improve += 1
            continue
        g = g / g_sum

        # Backtracking line search
        improved = False
        # Try larger steps first for faster convergence
        base_t = 0.4 if not aggressive else 0.3
        t_candidates = [base_t, base_t * 0.7, base_t * 0.5, base_t * 0.3,
                        base_t * 0.2, 0.1, 0.05, 0.02, 0.01, 0.005]

        for t in t_candidates:
            a_trial = np.maximum((1.0 - t) * a + t * g, 0.0)
            s = a_trial.sum()
            if s < 1e-10:
                continue
            a_trial = a_trial / s
            M_trial = fast_conv(a_trial).max()
            if M_trial < best_M - 1e-14:
                a = a_trial
                best_M = M_trial
                best_a = a.copy()
                improved = True
                no_improve = 0
                break

        if not improved:
            no_improve += 1
            # Perturb to escape local minimum
            if no_improve == stall_threshold:
                a = a + 0.02 * np.random.default_rng(it).normal(0, 1, n)
                a = np.maximum(a, 1e-6)
                a = a / a.sum()
                no_improve = 0
            elif no_improve > stall_threshold * 2:
                break

    return best_a


def coordinate_descent(a, deadline, max_sweeps=50):
    """
    Coordinate descent: for each index i, try shifting mass from i to j
    to reduce the max convolution value.
    """
    a = np.maximum(a.astype(np.float64).copy(), 0.0)
    a = a / a.sum()
    n = len(a)
    best_a = a.copy()
    best_M = fast_conv(a).max()
    rng = np.random.default_rng(12345)

    for sweep in range(max_sweeps):
        if time.monotonic() > deadline:
            break
        improved = False
        # Random subset of coordinates to try (for speed)
        if n <= 600:
            indices = list(range(n))
            rng.shuffle(indices)
        else:
            indices = rng.choice(n, size=min(n, 400), replace=False)

        for i in indices:
            if time.monotonic() > deadline:
                break
            if a[i] < 1e-8:
                continue

            # Try moving small amount from i to a few other indices
            # Choose targets: indices where a is small (to spread mass)
            # Or indices that would reduce the peak
            peak_idx = np.argmax(fast_conv(a))
            # Pick 3 random targets
            targets = rng.choice([j for j in range(n) if j != i and a[j] < a[i]],
                                  size=min(3, n - 1), replace=False)
            for j in targets:
                for epsilon in [0.01, 0.05, 0.1, 0.2]:
                    delta = epsilon * a[i]
                    if delta > a[i] * 0.9:
                        continue
                    a_trial = a.copy()
                    a_trial[i] -= delta
                    a_trial[j] += delta
                    a_trial = np.maximum(a_trial, 0.0)
                    s = a_trial.sum()
                    if s < 1e-10:
                        continue
                    a_trial = a_trial / s
                    M_trial = fast_conv(a_trial).max()
                    if M_trial < best_M - 1e-14:
                        a = a_trial
                        best_M = M_trial
                        best_a = a.copy()
                        improved = True
                        break
                if improved:
                    break

        if not improved and sweep > 5:
            break

    return best_a


def resample_seq(src, dest_n):
    """Resample 1D sequence to new length via linear interpolation."""
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)

def generate_parametric_inits(n, rng, parent=None):
    """Generate diverse parametric initializations."""
    inits = []
    t = np.linspace(0, 1, n, endpoint=False)

    if parent is not None:
        p = resample_seq(parent, n)
        inits.append(p.copy())
        for noise in [0.01, 0.05, 0.1, 0.2]:
            inits.append(np.maximum(p + rng.normal(0, noise, n), 0.01))

    # Power ramps with various exponents
    for p_exp in [0.5, 1.0, 1.5, 2.0, 3.0]:
        ramp = (1 - t) ** p_exp
        inits.append(ramp / ramp.sum())
        ramp2 = t ** p_exp
        inits.append(ramp2 / ramp2.sum())
        # Double ramp
        dr = (1 - t) ** p_exp + t ** p_exp
        inits.append(dr / dr.sum())

    # Sinusoidal
    for freq in [1, 2, 3, 4]:
        for amp in [0.3, 0.5, 0.7]:
            s = 1.0 + amp * np.cos(2 * np.pi * freq * t)
            inits.append(np.maximum(s, 0.01) / np.maximum(s, 0.01).sum())
            s2 = 1.0 + amp * np.sin(2 * np.pi * freq * t + 0.5)
            inits.append(np.maximum(s2, 0.01) / np.maximum(s2, 0.01).sum())

    # Gaussian bumps
    for center in [0.3, 0.5, 0.7]:
        for width in [0.1, 0.2, 0.3]:
            g = np.exp(-((t - center) / width) ** 2)
            inits.append(g / g.sum())

    # Sum of two Gaussians (asymmetric)
    g1 = np.exp(-((t - 0.25) / 0.15) ** 2)
    g2 = np.exp(-((t - 0.75) / 0.1) ** 2)
    inits.append((g1 + g2) / (g1 + g2).sum())

    # Triangle
    tri = 1.0 - 2 * np.abs(t - 0.5)
    inits.append(np.maximum(tri, 0.01) / np.maximum(tri, 0.01).sum())

    # Random smooth
    ksize = max(5, n // 30)
    kernel = np.ones(ksize) / ksize
    smooth = np.convolve(rng.uniform(0, 1, n), kernel, mode='same')
    inits.append(np.maximum(smooth, 0.01) / np.maximum(smooth, 0.01).sum())

    # Block structure
    n_blocks = rng.integers(3, max(4, n // 50))
    block_vals = rng.uniform(0.2, 1.0, n_blocks)
    block_seq = np.repeat(block_vals, n // n_blocks + 1)[:n]
    inits.append(block_seq / block_seq.sum())

    return inits

            a_res = resample_seq(tracker.best, cn)
            a_res = np.maximum(a_res, 1e-7)
            a_res = a_res / a_res.sum()

            # Quick smooth GD
            a_res = run_smooth_gd(a_res, 50, deadline - 20,
                                   tau_start=3, tau_end=40,
                                   lr_start=0.15, lr_end=0.03)
            tracker.update(a_res)

            # Short LP
            if time.monotonic() < deadline - 15:
                lp_time_est = 0.01 * (cn / 512) ** 2
                iters = max(10, min(100, int(20 / max(0.005, lp_time_est))))
                a_res = run_lp_mirror_descent(a_res, iters, deadline - 15, aggressive=False)
                tracker.update(a_res)

    # === FINAL: One more LP pass on the overall best ===
    if tracker.best is not None and time.monotonic() < deadline - 2:
        n_f = tracker.best.size
        lp_time_f = 0.01 * (n_f / 512) ** 2
        iters_f = max(10, min(200, int((deadline - 2 - time.monotonic()) / max(0.005, lp_time_f))))
        a_f = run_lp_mirror_descent(tracker.best, iters_f, deadline, aggressive=True)
        tracker.update(a_f)

    # Fallback
    if tracker.best is None:
        tracker.best = np.ones(256)
        tracker.best_score = compute_J(tracker.best)

    print(f"\n{'='*60}", flush=True)
    print(f"FINAL: J={tracker.best_score:.8f} n={tracker.best.size} "
          f"time={time.monotonic()-tracker.t_start:.1f}s", flush=True)
    print(f"{'='*60}", flush=True)

    return [float(x) for x in tracker.best]
