And by the rearrangement inequality for non-negative sequences, if a is sorted in decreasing order, the sum a[j]*a[n-1-j] is minimized when one is increasing and the other decreasing. But since both are the same sequence (just reversed), the sum is at its minimum when the sequence is "anti-sorted" in some sense.

Actually, for ANY non-negative sequence:
conv[n-1] = sum_{j=0}^{n-1} a[j]*a[n-1-j]

If we let b[j] = a[n-1-j], then conv[n-1] = sum a[j]*b[j]. By C-S: (sum a[j]*b[j])^2 <= (sum a[j]^2)(sum b[j]^2) = (sum a[j]^2)^2.

So conv[n-1] <= sum a[j]^2.

But we also have: conv[n-1] >= 0 (since all terms are non-negative).

The minimum of conv[n-1] over all non-negative a with sum = S is... well, we can make it arbitrarily close to 0 by putting all mass at one end (e.g., a[0] = S, rest = 0). In that case, conv[n-1] = 0 but conv[0] = S^2.

So the constraint is: max(conv) >= max(conv[0], conv[n-1], ..., conv[2n-2]) >= max(a[0]^2, a[n-1]^2, sum a[j]*a[n-1-j], ...)

To minimize the max, we want all convolution values to be as equal as possible. The ideal is conv[k] = S^2/(2n-1) for all k. But this is impossible for non-negative sequences because:
- conv[0] = a[0]^2 must equal S^2/(2n-1), so a[0] = S/sqrt(2n-1)
- Similarly a[n-1] = S/sqrt(2n-1)
- conv[1] = a[0]*a[1] = S^2/(2n-1), so a[1] = S/sqrt(2n-1)
- By induction, all a[j] = S/sqrt(2n-1)
- Then S = n*S/sqrt(2n-1), so sqrt(2n-1) = n, i.e., 2n-1 = n^2, i.e., n^2 - 2n + 1 = 0, i.e., (n-1)^2 = 0, i.e., n=1.

So for n > 1, we can't have a perfectly flat convolution with non-negative entries. The best we can do is make the convolution as flat as possible while maintaining non-negativity.

This is a well-studied problem in the context of "low autocorrelation sequences" and "Golay sequences" (though those typically allow negative values).

For non-negative sequences, the problem is related to "minimum peak-to-mean envelope power ratio" (PMEPR) for real signals, or "apodization" in optics.

"""

1. Softmax L-BFGS-B with aggressive adaptive tau annealing
2. Reduced LP (top-K tightest constraints) for fast polishing
3. Novel peak-reduction local search (mass transfer heuristic)
4. Multiplicative weight updates with momentum

6. Adaptive strategy switching when methods stall

import numpy as np
from scipy.optimize import minimize, linprog
from scipy.fft import rfft, irfft
import time


def compute_conv(a):
    """Compute linear convolution a*a using FFT. Returns array of length 2n-1."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    conv = irfft(fa * fa, size)
    return conv[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    conv = compute_conv(a)
    max_b = float(np.max(conv))
    sum_a = float(np.sum(a))
    if sum_a < 0.01:
        return np.inf
    return 2.0 * n * max_b / (sum_a * sum_a)


def fft_convolve(a, b):
    """Linear convolution of a and b using FFT."""
    n, m = len(a), len(b)
    size = 1
    while size < n + m - 1:
        size <<= 1
    fa = rfft(a, size)
    fb = rfft(b, size)
    return irfft(fa * fb, size)[:n + m - 1]


def softmax(z):
    """Numerically stable softmax."""
    z = z - np.max(z)
    e = np.exp(z)
    return e / np.sum(e)


def build_toeplitz_reduced(a, k_indices):
    """Build reduced Toeplitz matrix using only specified row indices."""
    n = len(a)
    K = len(k_indices)
    A = np.zeros((K, n), dtype=np.float64)
    for row_idx, k in enumerate(k_indices):
        j_start = max(0, k - n + 1)
        j_end = min(n, k + 1)
        for j in range(j_start, j_end):
            A[row_idx, j] = a[k - j]
    return A


def solve_lp_reduced(a, M, top_k=200, time_limit=10):
    """Solve reduced LP: max sum(g) s.t. (a*g)[k] <= M for top-k tightest k, g >= 0."""
    n = len(a)
    conv = compute_conv(a)
    # Find the tightest constraints (largest conv values)
    k_indices = np.argsort(conv)[::-1][:top_k]
    A = build_toeplitz_reduced(a, k_indices)
    c = -np.ones(n)
    b_ub = np.full(len(k_indices), M)
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': time_limit})
        if res.status == 0:
            g = np.maximum(res.x, 0)
            s = float(np.sum(g))
            if s > 1e-10:
                return g / s, s
    except Exception:
        pass
    return None, None


def solve_lp_full(a, M, time_limit=15):
    """Solve full LP: max sum(g) s.t. (a*g)[k] <= M for all k, g >= 0."""
    n = len(a)
    A = np.zeros((2 * n - 1, n), dtype=np.float64)
    for j in range(n):
        k_start = j
        k_end = min(j + n, 2 * n - 1)
        A[k_start:k_end, j] = a[:k_end - k_start]
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': time_limit})
        if res.status == 0:
            g = np.maximum(res.x, 0)
            s = float(np.sum(g))
            if s > 1e-10:
                return g / s, s
    except Exception:
        pass
    return None, None


def soft_max_obj_and_grad_z(z, tau):
    """Compute soft-max of conv(softmax(z), softmax(z)) and gradient w.r.t. z."""
    n = len(z)
    a = softmax(z)
    conv = compute_conv(a)
    cmax = float(np.max(conv))
    shifted = np.clip((conv - cmax) / tau, -500.0, 500.0)
    exp_conv = np.exp(shifted)
    Z = float(np.sum(exp_conv))
    soft_max_val = cmax + tau * np.log(Z)
    w = exp_conv / Z

    w_rev = w[::-1]
    conv_fw = fft_convolve(a, w_rev)
    f1 = conv_fw[n - 1:2 * n - 1][::-1]
    grad_a = 2.0 * f1

    mean_grad = float(np.dot(a, grad_a))
    grad_z = a * (grad_a - mean_grad)

    return soft_max_val, grad_z


def optimize_lbfgs(a_init, tau_list, max_iter=300, time_budget=30.0):
    """Optimize using L-BFGS-B with soft-max parametrization and tau annealing."""
    n = len(a_init)
    a = a_init.copy()
    s = np.sum(a)
    if s < 1e-10:
        a = np.ones(n) / n
    else:
        a = a / s
    a = np.maximum(a, 1e-30)
    z = np.log(a)

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    for tau in tau_list:
        if time.monotonic() - start > time_budget:
            break

        def func_and_grad(z_val):
            return soft_max_obj_and_grad_z(z_val, tau)

        try:
            result = minimize(func_and_grad, z, jac=True, method='L-BFGS-B',
                              options={'maxiter': max_iter, 'ftol': 1e-14, 'gtol': 1e-8})
            z_new = result.x
        except Exception:
            continue

        a_new = softmax(z_new)
        a_new = np.maximum(a_new, 0)
        s_new = np.sum(a_new)
        if s_new < 1e-10:
            continue
        a_new = a_new / s_new
        obj = compute_obj(a_new)

        if obj < best_obj:
            best_obj = obj
            best_a = a_new.copy()
            z = z_new

    return best_a, best_obj


def peak_reduction_search(a, max_iters=500, time_budget=20.0, n_trials_per_iter=3):
    """
    Novel peak-reduction local search: identify the convolution peak,
    then transfer mass from positions that contribute most to the peak
    to positions that reduce it.
    """
    n = len(a)
    a = a.copy().astype(np.float64)
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    for it in range(max_iters):
        if time.monotonic() - start > time_budget:
            break

        conv = compute_conv(a)
        peak_k = int(np.argmax(conv))
        peak_val = conv[peak_k]

        # Identify contributors to the peak: a[j] * a[peak_k - j]
        j_start = max(0, peak_k - n + 1)
        j_end = min(n, peak_k + 1)
        contributors = np.arange(j_start, j_end)
        contrib_vals = a[contributors] * a[peak_k - contributors]
        total_contrib = np.sum(contrib_vals)

        if total_contrib < 1e-15:
            break

        # Find the largest contributor pair
        max_idx = np.argmax(contrib_vals)
        j_star = contributors[max_idx]
        partner = peak_k - j_star

        # Transfer mass: reduce the larger of a[j_star], a[partner]
        # and add to a position that doesn't contribute to the peak
        if a[j_star] >= a[partner]:
            source = j_star
        else:
            source = partner

        # Find target: a position far from the peak contribution
        # Positions that DON'T contribute to conv[peak_k] are those
        # where neither j nor peak_k-j equals source
        # Actually ALL positions contribute to SOME conv value.
        # We want to add mass where it causes the least increase in max conv.
        # Approximation: add to a position with low "influence"
        # = position i where sum over affected conv values is lowest

        # Simple heuristic: add to a random position or the position
        # with minimum current value (spreading mass)
        # Better: add to the position that least increases the new max
        epsilon = 0.01 * a[source]
        if epsilon < 1e-10:
            break

        # Try multiple target positions
        best_target = -1
        best_new_obj = best_obj

        # Candidate targets: positions with low values
        low_mask = a < np.mean(a) * 0.8
        if np.sum(low_mask) > 0:
            targets = np.where(low_mask)[0]
            # Sample a few
            if len(targets) > n_trials_per_iter * 10:
                rng = np.random.default_rng(it * 31 + 7)
                targets = rng.choice(targets, n_trials_per_iter * 10, replace=False)
        else:
            targets = np.where(a == np.min(a))[0]

        for target_idx in targets[:n_trials_per_iter * 2]:
            if target_idx == source:
                continue
            a_try = a.copy()
            a_try[source] -= epsilon
            a_try[target_idx] += epsilon
            if np.min(a_try) < 0:
                continue
            s_try = np.sum(a_try)
            if s_try < 1e-10:
                continue
            a_try = a_try / s_try
            obj_try = compute_obj(a_try)
            if obj_try < best_new_obj:
                best_new_obj = obj_try
                best_target = target_idx
                best_a_try = a_try.copy()

        if best_target >= 0:
            a = best_a_try
            if best_new_obj < best_obj:
                best_obj = best_new_obj
                best_a = a.copy()

    return best_a, best_obj


def multiplicative_weight_update(a, max_iters=2000, time_budget=15.0, tau_start=5.0, tau_end=50.0):
    """
    Multiplicative weight update (Sinkhorn-like) for the positive orthant.
    a[i] <- a[i] * exp(-eta * subgrad[i]) / normalization
    """
    n = len(a)
    a = a.copy().astype(np.float64)
    s = np.sum(a)
    if s < 1e-10:
        a = np.ones(n) / n
    else:
        a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    for it in range(max_iters):
        if time.monotonic() - start > time_budget:
            break

        frac = it / max(1, max_iters)
        tau = tau_start + (tau_end - tau_start) * frac
        eta = 0.5 / (1.0 + 0.1 * it)

        conv = compute_conv(a)
        max_conv = float(np.max(conv))
        shifted = np.clip(tau * (conv - max_conv), -300, 300)
        weights = np.exp(shifted)
        w_sum = np.sum(weights)
        if w_sum < 1e-300:
            idx = int(np.argmax(conv))
            weights = np.zeros(len(conv))
            weights[idx] = 1.0
        else:
            weights /= w_sum

        # Compute subgradient
        w_rev = weights[::-1]
        conv_fw = fft_convolve(a, w_rev)
        grad_a = 2.0 * conv_fw[n - 1:2 * n - 1][::-1]

        # Multiplicative update: a[i] *= exp(-eta * grad_a[i])
        # This is equivalent to exponentiated gradient
        log_scale = eta * grad_a
        log_scale = np.clip(log_scale, -50, 50)
        a = a * np.exp(-log_scale)
        a = np.maximum(a, 1e-30)
        s = np.sum(a)
        if s > 1e-10:
            a = a / s

        obj = compute_obj(a)
        if obj < best_obj - 1e-12:
            best_obj = obj
            best_a = a.copy()

    return best_a, best_obj


def alternating_lp(a, max_iters=80, time_budget=50.0, use_reduced=True, top_k=300):
    """Alternating LP with line search using reduced LP for speed."""
    n = len(a)
    a = a.copy().astype(np.float64)
    s = np.sum(a)
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    stall_count = 0
    patience = 30

    for it in range(max_iters):
        if time.monotonic() - start > time_budget:
            break

        conv = compute_conv(a)
        M = float(np.max(conv))

        if use_reduced and n > 400:
            g, _ = solve_lp_reduced(a, M, top_k=top_k, time_limit=8)
        else:
            g, _ = solve_lp_full(a, M, time_limit=12)

        if g is None:
            break

        # Line search
        found_improvement = False
        for t in [0.1, 0.2, 0.3, 0.5, 0.7, 1.0]:
            a_try = (1.0 - t) * a + t * g
            a_try = np.maximum(a_try, 0)
            s_try = np.sum(a_try)
            if s_try < 1e-10:
                continue
            a_try = a_try / s_try
            obj_try = compute_obj(a_try)
            if obj_try < best_obj - 1e-10:
                best_obj = obj_try
                best_a = a_try.copy()
                a = a_try.copy()
                found_improvement = True
                break

        if found_improvement:
            stall_count = 0
        else:
            stall_count += 1
            if stall_count > patience:
                rng = np.random.default_rng(it * 7919 + 12345)
                a = a * np.exp(rng.normal(0, 0.05, n))
                a = np.maximum(a, 0)
                s = np.sum(a)
                if s > 1e-10:
                    a = a / s
                stall_count = 0
                patience = 15
            else:
                a = 0.95 * a + 0.05 * g
                a = np.maximum(a, 0)
                s = np.sum(a)
                if s > 1e-10:
                    a = a / s

    return best_a


def subgradient_optimize(a, max_iters=3000, time_budget=20.0, tau_start=3.0, tau_end=50.0):
    """Smoothed subgradient optimization with adaptive step size."""
    n = len(a)
    a = a.copy().astype(np.float64)
    s = np.sum(a)
    if s < 1e-10:
        a = np.ones(n) / n
    else:
        a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    step = 0.1
    stall = 0
    momentum = np.zeros(n)
    beta = 0.9

    for it in range(max_iters):
        if time.monotonic() - start > time_budget:
            break

        frac = it / max(1, max_iters)
        tau = tau_start + (tau_end - tau_start) * frac

        conv = compute_conv(a)
        max_conv = float(np.max(conv))
        shifted = np.clip(tau * (conv - max_conv), -300, 300)
        weights = np.exp(shifted)
        w_sum = np.sum(weights)
        if w_sum < 1e-300:
            idx = int(np.argmax(conv))
            weights = np.zeros(len(conv))
            weights[idx] = 1.0
        else:
            weights /= w_sum

        w_rev = weights[::-1]
        conv_fw = fft_convolve(a, w_rev)
        grad_a = 2.0 * conv_fw[n - 1:2 * n - 1][::-1]

        # Center the gradient (for simplex constraint)
        grad_centered = grad_a - np.dot(a, grad_a)

        # Nesterov momentum
        momentum = beta * momentum - step * grad_centered
        a_new = a + momentum
        a_new = np.maximum(a_new, 0)
        s_new = np.sum(a_new)
        if s_new > 1e-10:
            a_new = a_new / s_new
        else:
            a_new = a.copy()

        a = a_new
        obj = compute_obj(a)

        if obj < best_obj - 1e-12:
            best_obj = obj
            best_a = a.copy()
            stall = 0
        else:
            stall += 1
            if stall > 50:
                step *= 0.85
                if step < 1e-10:
                    break
                rng = np.random.default_rng(it * 31 + 7)
                a = a + rng.normal(0, 0.005, n)
                a = np.maximum(a, 0)
                s = np.sum(a)
                if s > 1e-10:
                    a = a / s
                momentum = np.zeros(n)
                stall = 0

        if it > 50:
            step *= 0.997

    return best_a


def random_init(n, seed):
    """Generate diverse random initializations."""
    rng = np.random.default_rng(seed)
    k = np.arange(n)
    choice = rng.integers(0, 12)

    if choice == 0:
        f = rng.uniform(0.5, 4.0)
        phase = rng.uniform(0, 2 * np.pi)
        a = 1.0 + 0.8 * np.cos(2 * np.pi * f * k / n + phase)
    elif choice == 1:
        f1 = rng.uniform(0.3, 2.0)
        f2 = rng.uniform(2.0, 6.0)
        a = 1.0 + 0.4 * np.cos(2 * np.pi * f1 * k / n) + 0.3 * np.cos(2 * np.pi * f2 * k / n)
    elif choice == 2:
        # Dip in the middle
        center = n * 0.5
        alpha = rng.uniform(1.0, 5.0)
        depth = rng.uniform(0.2, 0.7)
        a = 1.0 - depth * np.exp(-alpha * ((k - center) / (n * 0.3)) ** 2)
    elif choice == 3:
        a = rng.random(n)
        win = max(3, n // 20)
        kernel = np.ones(win) / win
        a = np.convolve(a, kernel, mode='same')
    elif choice == 4:
        a = np.ones(n)
        for _ in range(rng.integers(1, 5)):
            f = rng.uniform(0.5, 5.0)
            amp = rng.uniform(0.1, 0.4)
            a += amp * np.cos(2 * np.pi * f * k / n + rng.uniform(0, 2 * np.pi))
    elif choice == 5:
        # Two-peak structure
        f = rng.uniform(0.3, 2.0)
        a = 1.0 + 0.5 * np.cos(2 * np.pi * f * k / n)
    elif choice == 6:
        nblocks = rng.integers(2, 7)
        a = np.ones(n)
        for b in range(nblocks):
            start = b * n // nblocks
            end = (b + 1) * n // nblocks
            a[start:end] = rng.uniform(0.3, 1.5)
    elif choice == 7:
        # Chirp-like
        f0 = rng.uniform(0.5, 2.0)
        f1 = rng.uniform(2.0, 6.0)
        a = 1.0 + 0.5 * np.cos(2 * np.pi * (f0 * k / n + (f1 - f0) * k**2 / (2 * n**2)))
    elif choice == 8:
        # Edge peaks (valley in middle)
        center = n * 0.5
        width = n * rng.uniform(0.2, 0.5)
        a = 1.0 - 0.5 * np.exp(-0.5 * ((k - center) / width) ** 2)
    elif choice == 9:
        # Sine with harmonic
        f = rng.uniform(0.5, 3.0)
        a = 1.0 + 0.3 * np.sin(2 * np.pi * f * k / n) + 0.2 * np.sin(4 * np.pi * f * k / n)
    elif choice == 10:
        # Legendre-like (approximate)
        x = 2.0 * k / (n - 1) - 1.0
        a = 1.0 + 0.4 * (1.0 - x**2)  # Second Legendre polynomial shape
    else:
        a = rng.random(n) + 0.1

    a = np.maximum(a, 0.01)
    s = np.sum(a)
    return a / s


def resample_sequence(a, new_n):
    """Resample sequence to new length via interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return a_new


def try_parent():
    """Try to load parent sequence."""
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64)
        parent = np.clip(parent, 0, 1000.0)
        if parent.size > 0 and np.isfinite(parent).all():
            obj = compute_obj(parent)
            print(f"Parent (n={parent.size}): obj={obj:.6f}")
            return parent.copy(), obj
    except Exception:
        pass
    return None, np.inf


def coordinate_descent_polish(a, max_iters=5000, time_budget=30.0):
    """Coordinate descent: optimize one coordinate at a time on the simplex."""
    n = len(a)
    a = a.copy().astype(np.float64)
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    conv = compute_conv(a)
    max_conv = float(np.max(conv))

    for it in range(max_iters):
        if time.monotonic() - start > time_budget:
            break

        # Pick coordinate to update (cyclic or random)
        if it % 3 == 0:
            i = it % n
        else:
            rng = np.random.default_rng(it)
            i = rng.integers(0, n)

        # Find optimal a[i] by 1D search
        # We need to adjust a[i] while keeping sum constant
        # So we need to reduce some other coordinate or all others proportionally
        # Simpler: just try a few values of a[i] and renormalize
        a[i] = max(a[i] * 0.5, 1e-10)
        s = np.sum(a)
        if s > 1e-10:
            a = a / s
            obj = compute_obj(a)
            if obj < best_obj:
                best_obj = obj
                best_a = a.copy()

        a[i] = a[i] * 1.5  # partially restore
        s = np.sum(a)
        if s > 1e-10:
            a = a / s
            obj = compute_obj(a)
            if obj < best_obj:
                best_obj = obj
                best_a = a.copy()

    return best_a


def polish_sequence(a, time_budget, verbose=False):
    """Intensive polishing combining all methods."""
    n = len(a)
    a = a.copy().astype(np.float64)
    s = np.sum(a)
    if s < 1e-10:
        return a, compute_obj(a)
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    round_num = 0

    while time.monotonic() - start < time_budget - 3:
        round_num += 1
        remaining = time_budget - (time.monotonic() - start)
        if remaining < 5:
            break

        # LP step
        if remaining > 10:
            lp_budget = min(remaining * 0.3, 30.0)
            top_k = 200 if n > 400 else 2 * n - 1
            a_new = alternating_lp(best_a, max_iters=40, time_budget=lp_budget,
                                   use_reduced=(n > 300), top_k=top_k)
            obj = compute_obj(a_new)
            if obj < best_obj:
                best_obj = obj
                best_a = a_new.copy()
                if verbose:
                    print(f"    r{round_num} LP: {best_obj:.8f}")

        # L-BFGS-B
        remaining = time_budget - (time.monotonic() - start)
        if remaining > 8:
            tau_final = 0.2 / n
            tau_list = [0.5 / n, 0.3 / n, 0.1 / n]
            a_new, obj_new = optimize_lbfgs(best_a, tau_list, max_iter=200,
                                            time_budget=min(remaining * 0.2, 15.0))
            if obj_new < best_obj:
                best_obj = obj_new
                best_a = a_new.copy()
                if verbose:
                    print(f"    r{round_num} L-BFGS: {best_obj:.8f}")

        # Peak reduction
        remaining = time_budget - (time.monotonic() - start)
        if remaining > 8:
            a_new, obj_new = peak_reduction_search(best_a, max_iters=200,
                                                    time_budget=min(remaining * 0.15, 10.0))
            if obj_new < best_obj:
                best_obj = obj_new
                best_a = a_new.copy()
                if verbose:
                    print(f"    r{round_num} PeakRed: {best_obj:.8f}")

        # Multiplicative weight
        remaining = time_budget - (time.monotonic() - start)
        if remaining > 8:
            a_new, obj_new = multiplicative_weight_update(best_a, max_iters=int(remaining * 3),
                                                           time_budget=min(remaining * 0.15, 10.0),
                                                           tau_start=10.0, tau_end=40.0)
            if obj_new < best_obj:
                best_obj = obj_new
                best_a = a_new.copy()
                if verbose:
                    print(f"    r{round_num} MWU: {best_obj:.8f}")

        # Subgradient
        remaining = time_budget - (time.monotonic() - start)
        if remaining > 6:
            a_new = subgradient_optimize(best_a, max_iters=int(remaining * 5),
                                          time_budget=min(remaining * 0.15, 10.0),
                                          tau_start=8.0, tau_end=30.0)
            obj = compute_obj(a_new)
            if obj < best_obj:
                best_obj = obj
                best_a = a_new.copy()
                if verbose:
                    print(f"    r{round_num} SG: {best_obj:.8f}")

    return best_a, best_obj


def propose_candidate(seed=42, budget_s=1500):
    """
    Main search: explores multiple sequence lengths using diverse optimization methods.
    Key improvements over previous:
    - Peak reduction local search (novel)
    - Multiplicative weight updates (novel)
    - Reduced LP for faster iterations
    - Nesterov momentum in subgradient
    - More diverse initializations with "dip" patterns
    - Better time allocation and adaptive strategy switching
    """
    start_time = time.monotonic()
    deadline = start_time + float(budget_s) - 30.0

    best_seq = None
    best_score = np.inf

    # Load parent
    parent, parent_obj = try_parent()
    if parent is not None and parent_obj < best_score:
        best_score = parent_obj
        best_seq = parent.copy()

    # N values to explore - focus around parent length
    n_base = len(parent) if parent is not None else 600
    n_values = sorted(set([
        max(100, n_base - 200),
        max(100, n_base - 100),
        n_base,
        n_base + 100,
        n_base + 200,
        min(1200, n_base + 400),
        400, 500, 700, 800, 1000
    ]))

    print(f"Exploring n values: {n_values}")
    print(f"Initial best: {best_score:.8f}")

    # Phase 1: Exploration with diverse methods
    candidates = []

    for n in n_values:
        if time.monotonic() > deadline - 200:
            break
        remaining = deadline - time.monotonic()
        if remaining < 15:
            break

        # Time allocation
        if n <= 300:
            alloc = min(25.0, remaining * 0.08)
        elif n <= 600:
            alloc = min(45.0, remaining * 0.1)
        elif n <= 900:
            alloc = min(55.0, remaining * 0.12)
        else:
            alloc = min(60.0, remaining * 0.12)
        alloc = max(15.0, alloc)

        print(f"\nN={n}: {remaining:.0f}s remaining, alloc={alloc:.0f}s")

        n_inits = max(2, min(4, int(alloc / 12)))

        for init_idx in range(n_inits):
            if time.monotonic() > deadline - 200:
                break
            remaining_init = deadline - time.monotonic()
            if remaining_init < 15:
                break

            # Different initialization strategies
            if init_idx == 0 and best_seq is not None and abs(n - len(best_seq)) < 200:
                a = resample_sequence(best_seq, n)
            elif init_idx == 1:
                a = random_init(n, seed + n * 104729 + 79193)
            elif init_idx == 2:
                a = random_init(n, seed + n * 209150 + 12345)
            else:
                # Structured initialization: dip pattern optimized
                k = np.arange(n)
                depth = 0.3 + 0.3 * (init_idx % 3) * 0.5
                width = 0.2 + 0.1 * (init_idx % 3)
                a = 1.0 - depth * np.exp(-0.5 * ((k - n * 0.5) / (n * width)) ** 2)
                a = np.maximum(a, 0.05)
                a = a / np.sum(a)

            # Optimization pipeline for this init
            # Step 1: Quick subgradient shaping
            sg_time = min(6.0, (alloc / n_inits) * 0.15)
            a = subgradient_optimize(a, max_iters=int(sg_time * 30),
                                      time_budget=sg_time, tau_start=3.0, tau_end=15.0)

            # Step 2: L-BFGS-B with tau annealing
            remaining_init = deadline - time.monotonic()
            if remaining_init > 10:
                lfbs_time = min((alloc / n_inits) * 0.35, 20.0)
                tau_list = [5.0 / n, 2.0 / n, 1.0 / n, 0.5 / n, 0.2 / n]
                a, obj = optimize_lbfgs(a, tau_list, max_iter=200, time_budget=lfbs_time)
            else:
                obj = compute_obj(a)

            # Step 3: LP polishing
            remaining_init = deadline - time.monotonic()
            if remaining_init > 15 and n <= 800:
                lp_time = min((alloc / n_inits) * 0.3, 20.0)
                a = alternating_lp(a, max_iters=50, time_budget=lp_time,
                                   use_reduced=(n > 300), top_k=250)
                obj = compute_obj(a)

            # Step 4: Peak reduction
            remaining_init = deadline - time.monotonic()
            if remaining_init > 10:
                a, obj = peak_reduction_search(a, max_iters=300,
                                                time_budget=min(remaining_init * 0.15, 8.0))

            # Step 5: MWU
            remaining_init = deadline - time.monotonic()
            if remaining_init > 8:
                a, obj = multiplicative_weight_update(a, max_iters=int(remaining_init * 3),
                                                       time_budget=min(remaining_init * 0.1, 6.0),
                                                       tau_start=10.0, tau_end=40.0)

            print(f"  init{init_idx}: n={n}, obj={obj:.8f}")

            if obj < best_score:
                best_score = obj
                best_seq = a.copy()
                print(f"  ** UPDATE: n={n}, obj={best_score:.8f} **")

            candidates.append((a.copy(), obj, n))

    # Phase 2: Polishing the best candidates
    print(f"\n=== POLISHING PHASE ===")
    print(f"Best before polish: {best_score:.8f}")

    candidates.sort(key=lambda x: x[1])
    n_polish = min(4, len(candidates))

    for ci in range(n_polish):
        if time.monotonic() > deadline - 100:
            break
        remaining = deadline - time.monotonic()
        if remaining < 20:
            break

        a_cand = candidates[ci][0]
        n_cand = candidates[ci][2]
        print(f"  Polishing candidate {ci} (n={n_cand}, obj={candidates[ci][1]:.8f}), {remaining:.0f}s left")

        polish_budget = min(remaining * 0.35, 80.0)
        a_pol, obj_pol = polish_sequence(a_cand, time_budget=polish_budget, verbose=True)

        if obj_pol < best_score:
            best_score = obj_pol
            best_seq = a_pol.copy()
            print(f"  ** UPDATE after polish: obj={best_score:.8f} **")
        else:
            # Try perturbation from the polished candidate
            remaining = deadline - time.monotonic()
            if remaining > 20:
                rng = np.random.default_rng(ci * 555 + 999)
                for ptry in range(4):
                    if time.monotonic() > deadline - 50:
                        break
                    scale = rng.uniform(0.02, 0.10)
                    a_pert = a_pol * np.exp(rng.normal(0, scale, n_cand))
                    a_pert = np.maximum(a_pert, 0.01)
                    a_pert = a_pert / np.sum(a_pert)

                    # Quick optimization of perturbation
                    a_pert, obj_pert = optimize_lbfgs(a_pert, [2.0 / n_cand, 0.5 / n_cand],
                                                      max_iter=100, time_budget=5.0)
                    if obj_pert < best_score:
                        best_score = obj_pert
                        best_seq = a_pert.copy()
                        print(f"  ** UPDATE (perturb {ptry}): obj={best_score:.8f} **")

    # Phase 3: Final intensive polish on the global best
    if best_seq is not None and time.monotonic() < deadline - 20:
        remaining = deadline - time.monotonic()
        n_best = len(best_seq)
        print(f"\n=== FINAL POLISH: n={n_best}, {remaining:.0f}s ===")

        a_cur = best_seq.copy()
        for final_round in range(6):
            if time.monotonic() > deadline - 10:
                break
            remaining = deadline - time.monotonic()
            if remaining < 10:
                break

            # L-BFGS-B with very small tau
            if remaining > 8:
                tau_vsmall = 0.05 / n_best
                a_new, obj_new = optimize_lbfgs(a_cur, [0.3 / n_best, tau_vsmall],
                                                max_iter=300, time_budget=min(remaining * 0.2, 15.0))
                if obj_new < best_score:
                    best_score = obj_new
                    best_seq = a_new.copy()
                    a_cur = a_new.copy()
                    print(f"  Final L-BFGS r{final_round}: {best_score:.8f}")

            # LP
            remaining = deadline - time.monotonic()
            if remaining > 15 and n_best <= 800:
                a_new = alternating_lp(a_cur, max_iters=50, time_budget=min(remaining * 0.25, 25.0),
                                       use_reduced=(n_best > 300), top_k=300)
                obj_new = compute_obj(a_new)
                if obj_new < best_score:
                    best_score = obj_new
                    best_seq = a_new.copy()
                    a_cur = a_new.copy()
                    print(f"  Final LP r{final_round}: {best_score:.8f}")

            # Peak reduction
            remaining = deadline - time.monotonic()
            if remaining > 8:
                a_new, obj_new = peak_reduction_search(a_cur, max_iters=500,
                                                        time_budget=min(remaining * 0.15, 10.0))
                if obj_new < best_score:
                    best_score = obj_new
                    best_seq = a_new.copy()
                    a_cur = a_new.copy()
                    print(f"  Final PeakRed r{final_round}: {best_score:.8f}")

            # MWU
            remaining = deadline - time.monotonic()
            if remaining > 6:
                a_new, obj_new = multiplicative_weight_update(a_cur, max_iters=int(remaining * 4),
                                                               time_budget=min(remaining * 0.12, 8.0),
                                                               tau_start=5.0, tau_end=30.0)
                if obj_new < best_score:
                    best_score = obj_new
                    best_seq = a_new.copy()
                    a_cur = a_new.copy()
                    print(f"  Final MWU r{final_round}: {best_score:.8f}")

            # Subgradient
            remaining = deadline - time.monotonic()
            if remaining > 6:
                a_new = subgradient_optimize(a_cur, max_iters=int(remaining * 8),
                                              time_budget=min(remaining * 0.12, 8.0),
                                              tau_start=15.0, tau_end=60.0)
                obj_new = compute_obj(a_new)
                if obj_new < best_score:
                    best_score = obj_new
                    best_seq = a_new.copy()
                    a_cur = a_new.copy()
                    print(f"  Final SG r{final_round}: {best_score:.8f}")

    # Final validation
    if best_seq is None:
        best_seq = np.ones(200)
        best_score = compute_obj(best_seq)

    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    val = evaluate_sequence(result)
    if val < np.inf:
        best_score = val
    else:
        result = [1.0] * 100
        best_score = 2.0

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_score:.8f}, time={elapsed:.1f}s ===")

    return result
