import numpy as np
from scipy.optimize import linprog

import time

def fast_conv(a):
    """Full self-convolution via FFT."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]

def conv_with(a, b):
    """Full convolution of a and b via FFT."""
    total = len(a) + len(b) - 1
    size = 1
    while size < total:
        size <<= 1
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    return np.fft.irfft(fa * fb, size)[:total]

def compute_J(a):
    """Compute J(a) = 2*n*max(conv(a,a))/(sum(a))^2."""
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))

def lse_smooth_max(a, tau):
    """Compute smooth max of conv(a,a) and its gradient using log-sum-exp."""
    n = len(a)
    b = fast_conv(a)
    b_max = b.max()
    # Numerically stable
    exp_b = np.exp(tau * (b - b_max))
    exp_sum = exp_b.sum()
    p = exp_b / exp_sum  # softmax weights
    
    # Gradient: d/da[i] of smooth_max = sum_k p[k] * d(a*a)[k]/da[i]
    # d(a*a)[k]/da[i] = 2*a[k-i] (if 0 <= k-i < n)
    # So grad[i] = 2 * sum_k p[k] * a[k-i]
    # This is 2 * (p_star * a) where * is cross-correlation
    # grad[i] = 2 * sum_k p[k] * a[k-i]
    # = 2 * conv(p, a_flipped)[offset]
    
    a_flip = a[::-1]
    c = conv_with(p, a_flip)
    # c[m] = sum_k p[k] * a_flip[m-k] = sum_k p[k] * a[n-1-m+k]
    # We want sum_k p[k] * a[k-i], which is c[n-1+i]
    grad = 2.0 * c[n-1:n-1+n]
    
    smooth_val = b_max + np.log(exp_sum) / tau
    return smooth_val, grad, b, p

def subgradient_max(a):
    """Compute exact subgradient of max(conv(a,a))."""
    n = len(a)
    b = fast_conv(a)
    M = b.max()
    tol = 1e-10 * M + 1e-15
    active = np.where(b >= M - tol)[0]
    
    # grad[i] = sum_{k in active} 2*a[k-i] (normalized by |active|)
    grad = np.zeros(n)
    a_flip = a[::-1]
    indicator = np.zeros(2*n-1)
    indicator[active] = 1.0 / len(active)
    c = conv_with(indicator, a_flip)
    grad = 2.0 * c[n-1:n-1+n]
    
    return grad, M, active, b

def frequency_domain_gs(n, rng, n_iters=50):
    """Gerchberg-Saxton style frequency domain initialization."""
    L = 1
    while L < 2 * n - 1:
        L <<= 1
    
    # Start with a smooth window
    t = np.linspace(0, 1, n, endpoint=False)
    a = np.maximum(0.5 + 0.5 * np.cos(2 * np.pi * t), 0.01)
    a = a / a.sum()
    
    for it in range(n_iters):
        # Zero-pad and FFT
        a_pad = np.zeros(L)
        a_pad[:n] = a
        A = np.fft.rfft(a_pad)
        
        # Flatten magnitude
        mag = np.abs(A)
        mag = np.where(mag > 1e-15, 1.0, mag)  # set to 1 where non-zero
        phase = np.angle(A)
        A_new = mag * np.exp(1j * phase)
        
        # IFFT and extract
        a_new = np.fft.irfft(A_new, L)[:n]
        a = np.maximum(a_new, 0.0)
        s = a.sum()
        if s > 1e-10:
            a = a / s
        else:
            break
    
    return a

def resample_seq(src, dest_n):
    """Resample 1D sequence to new length via linear interpolation."""
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)

def build_A_sparse(n, a):
    """Build sparse convolution matrix A where (A*g)[k] = (a*g)[k]."""
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        j_min = max(0, k - n + 1)
        j_max = min(n - 1, k)
        for j in range(j_min, j_max + 1):
            rows.append(k)
            cols.append(j)
            vals.append(a[k - j])
    return csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))

def lp_best_response(a, M):

    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    try:

                    options={'time_limit': 60})
        if r.success and r.x is not None:
            g = np.maximum(r.x, 0)
            if g.sum() > 1e-15:
                return g
    except Exception:
        pass
    return None

"""
Minimizes J(a) = 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Diverse initializations (frequency-domain GS, structured, random, parent)
2. Nesterov-accelerated log-sum-exp gradient descent with adaptive temperature
3. LP mirror descent with Armijo line search and restarts
4. Coordinate descent for final local polishing
5. Multi-length exploration (256-1536)
Key differences from previous: Nesterov acceleration, Armijo line search in LP,
coordinate descent phase, frequency-domain initialization, LP restarts.

import numpy as np
from scipy.optimize import linprog
from scipy.sparse import csr_matrix
import time


def fast_conv(a):
    """Full self-convolution via FFT."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]


def conv_with(a, b):
    """Full linear convolution of a and b via FFT."""
    total = len(a) + len(b) - 1
    size = 1
    while size < total:
        size <<= 1
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    return np.fft.irfft(fa * fb, size)[:total]


def compute_J(a):
    """Compute J(a) = 2*n*max(conv(a,a))/(sum(a))^2."""
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))


def lse_smooth_max_grad(a, tau):
    """
    Compute log-sum-exp smooth max of conv(a,a) and its gradient.
    Returns (smooth_value, gradient, b, p).
    """
    n = len(a)
    b = fast_conv(a)
    b_max = b.max()
    exp_b = np.exp(tau * (b - b_max))
    exp_sum = exp_b.sum()
    p = exp_b / exp_sum

    # grad[i] = 2 * sum_k p[k] * a[k-i] (cross-correlation)
    a_flip = a[::-1]
    c = conv_with(p, a_flip)
    grad = 2.0 * c[n - 1:n - 1 + n]

    smooth_val = b_max + np.log(exp_sum) / tau
    return smooth_val, grad, b, p


def subgradient_of_max(a, tol_frac=1e-7):
    """Compute a subgradient of max(conv(a,a)). Returns (grad, M, active_set)."""
    n = len(a)
    b = fast_conv(a)
    M = b.max()
    tol = tol_frac * M + 1e-15
    active = np.where(b >= M - tol)[0]

    # Unnormalized subgradient: sum over active of d(a*a)[k]/da
    # grad[i] = sum_{k in active} 2*a[k-i]
    indicator = np.zeros(2 * n - 1)
    indicator[active] = 1.0
    a_flip = a[::-1]
    c = conv_with(indicator, a_flip)
    grad = 2.0 * c[n - 1:n - 1 + n]

    return grad, M, active


def frequency_domain_gs(n, rng, n_iters=60):
    """Gerchberg-Saxton style frequency domain initialization for flat spectrum."""
    L = 1
    while L < 2 * n - 1:
        L <<= 1

    t = np.linspace(0, 1, n, endpoint=False)
    # Start with a raised cosine window
    a = np.maximum(0.5 + 0.5 * np.cos(2 * np.pi * t), 0.01)
    a = a / a.sum()

    for it in range(n_iters):
        a_pad = np.zeros(L)
        a_pad[:n] = a
        A = np.fft.rfft(a_pad)

        # Flatten magnitude (keep phase)
        mag = np.abs(A)
        phase = np.angle(A)
        target_mag = np.ones_like(mag)
        # Blend: partially flatten
        blend = 0.3 + 0.5 * (it / max(1, n_iters))
        mag_new = (1 - blend) * mag + blend * target_mag
        # Avoid zeroing out DC
        if mag_new[0] < 1e-10:
            mag_new[0] = 1e-10

        A_new = mag_new * np.exp(1j * phase)
        a_new = np.fft.irfft(A_new, L)[:n]
        a = np.maximum(a_new, 0.0)
        s = a.sum()
        if s > 1e-10:
            a = a / s
        else:
            break

    return a


def resample_seq(src, dest_n):
    """Resample 1D sequence to new length via linear interpolation."""
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)


def build_A_sparse(n, a):
    """Build sparse matrix A where (A@g)[k] = (a*g)[k] for all k."""
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        j_min = max(0, k - n + 1)
        j_max = min(n - 1, k)
        if j_min <= j_max:
            js = np.arange(j_min, j_max + 1)
            rows.extend([k] * len(js))
            cols.extend(js.tolist())
            vals.extend(a[k - js].tolist())
    return csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))


def lp_best_response(a, M, time_limit=45):
    """LP: max sum(g) s.t. (a*g)[k] <= M for all k, g >= 0. Returns g or None."""
    n = len(a)
    A = build_A_sparse(n, a)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    try:
        r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                    options={'time_limit': time_limit})
        if r.success and r.x is not None:
            g = np.maximum(r.x, 0)
            if g.sum() > 1e-15:
                return g
    except Exception:
        pass
    return None


def run_nesterov_lse(a, max_iters, deadline, tau_start=2.0, tau_end=200.0):
    """
    Nesterov-accelerated gradient descent on LSE smooth max.
    Uses adaptive tau (annealing) and backtracking line search.
    """
    a = np.maximum(a.astype(np.float64), 1e-8)
    a = a / a.sum()
    n = len(a)

    best_a = a.copy()
    best_M = fast_conv(a).max()

    # Nesterov state
    x = a.copy()  # current point
    y = a.copy()  # Nesterov lookahead point
    x_prev = a.copy()
    mu = 1.0  # momentum parameter (2/(k+2))

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        frac = it / max(1, max_iters)
        tau = tau_start * (tau_end / tau_start) ** (frac ** 0.6)

        # Compute gradient at lookahead point y
        smooth_val, grad, b, p = lse_smooth_max_grad(y, tau)
        M_current = b.max()

        if M_current < best_M * (1 - 1e-10):
            best_M = M_current
            best_a = y.copy()

        # Backtracking line search
        lr = 0.3
        f_y = smooth_val
        accepted = False
        for _ls in range(20):
            x_trial = y - lr * grad
            x_trial = np.maximum(x_trial, 0.0)
            s = x_trial.sum()
            if s < 1e-10:
                lr *= 0.5
                continue
            x_trial = x_trial / s

            # Armijo condition (approximate)
            f_trial = fast_conv(x_trial).max()
            if f_trial < M_current + 1e-12 * M_current:
                accepted = True
                break
            lr *= 0.5

        if not accepted:
            # Take a tiny step anyway to avoid stagnation
            x_trial = y - 0.001 * grad
            x_trial = np.maximum(x_trial, 0.0)
            s = x_trial.sum()
            if s > 1e-10:
                x_trial = x_trial / s
            else:
                x_trial = x.copy()

        # Nesterov update
        mu_new = 2.0 / (it + 2)
        y = x_trial + (mu - mu_new) / mu_new * (x_trial - x_prev)
        y = np.maximum(y, 0.0)
        s = y.sum()
        if s > 1e-10:
            y = y / s
        else:
            y = x_trial.copy()

        x_prev = x_trial.copy()
        x = x_trial
        mu = mu_new

    # Final check
    M_final = fast_conv(x).max()
    if M_final < best_M:
        best_a = x.copy()
        best_M = M_final

    return best_a


def run_subgradient_avg(a, max_iters, deadline, lr_init=0.3):
    """Subgradient descent with Polyak-like step and running average."""
    a = np.maximum(a.astype(np.float64), 1e-8)
    a = a / a.sum()
    n = len(a)

    best_a = a.copy()
    best_M = fast_conv(a).max()
    running_sum = np.zeros(n)

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        lr = lr_init / (1 + 0.05 * it) ** 0.7

        grad, M, active = subgradient_of_max(a)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-15:
            break
        grad = grad / gnorm

        if M < best_M * (1 - 1e-10):
            best_M = M
            best_a = a.copy()

        a_new = a - lr * grad
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s < 1e-10:
            a_new = a.copy()
        else:
            a_new = a_new / s

        running_sum += a_new
        a = a_new

    # Check averaged solution

        s = a_avg.sum()
        if s > 1e-10:
            a_avg = a_avg / s
            M_avg = fast_conv(a_avg).max()

def run_lp_mirror_descent(a, max_iters, deadline, t_init=0.4):
    """
    LP mirror descent with Armijo-style line search and adaptive step.
    More robust than geometric decay: increases t on success, decreases on failure.
    """
    a = np.maximum(a.astype(np.float64), 1e-8)
    a = a / a.sum()
    n = len(a)

    b = fast_conv(a)
    current_M = float(b.max())
    best_a = a.copy()
    best_M = current_M
    no_improve = 0
    t = t_init

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        g = lp_best_response(a, current_M, time_limit=40)
        if g is None:
            break
        g = np.maximum(g, 0.0)
        g_sum = g.sum()
        if g_sum < 1e-10:
            break
        g = g / g_sum

        # Check if a is already close to g (convergence)
        if np.linalg.norm(a - g) < 1e-6:
            break

        # Armijo line search: try decreasing t
        improved = False
        t_candidates = [t, t * 0.75, t * 0.5, t * 0.35, t * 0.2, 0.08, 0.04, 0.02, 0.01]
        for tt in t_candidates:
            if tt <= 0:
                continue
            a_trial = (1 - tt) * a + tt * g
            a_trial = np.maximum(a_trial, 0.0)
            s = a_trial.sum()
            if s < 1e-10:
                continue
            a_trial = a_trial / s
            M_trial = float(fast_conv(a_trial).max())

            if M_trial < best_M - 1e-14:
                a = a_trial
                best_M = M_trial
                best_a = a.copy()
                current_M = M_trial
                improved = True
                # Increase step size slightly on success
                t = min(0.8, tt * 1.05)
                break

        if not improved:
            no_improve += 1
            t = max(0.005, t * 0.85)
            if no_improve > 80:
                break
        else:
            no_improve = 0

    return best_a


def coordinate_descent(a, deadline, n_sweeps=5, step_multipliers=None):
    """
    Coordinate descent: cyclically try scaling each element.
    Accept change if it reduces max convolution value.
    """
    if step_multipliers is None:
        step_multipliers = [0.5, 0.7, 0.85, 1.15, 1.3, 1.5, 0.3, 1.7, 2.0, 0.15]

    a = np.maximum(a.astype(np.float64), 0.0)
    a = a / a.sum()
    n = len(a)
    best_M = float(fast_conv(a).max())

    for sweep in range(n_sweeps):
        if time.monotonic() > deadline:
            break
        sweep_improved = False
        rng_indices = np.random.permutation(n)

        for i in rng_indices:
            if time.monotonic() > deadline:
                break
            original = a[i]
            if original < 1e-8:
                # Try setting to small positive values
                for val in [0.001, 0.005, 0.01, 0.02]:
                    a[i] = val
                    s = a.sum()
                    if s < 0.01:
                        a[i] = 0
                        continue
                    a_n = a / s
                    M_t = float(fast_conv(a_n).max())
                    if M_t < best_M - 1e-12:
                        a = a_n.copy()
                        best_M = M_t
                        sweep_improved = True
                        break
                a[i] = 0 if original < 1e-8 else original
                continue

            for mult in step_multipliers:
                a[i] = original * mult
                s = a.sum()
                if s < 0.01:
                    a[i] = original
                    continue
                a_n = a / s
                M_t = float(fast_conv(a_n).max())
                if M_t < best_M - 1e-12:
                    a = a_n.copy()
                    best_M = M_t
                    sweep_improved = True
                    break
            a[i] = a[i]  # already updated if improved

        if not sweep_improved:
            break

    return a


class ScoreTracker:
    """Tracks the best sequence found so far."""

    def __init__(self):
        self.best = None
        self.best_score = float('inf')
        self.t_start = time.monotonic()

    def update(self, a):
        a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
        if a.size == 0 or a.sum() < 0.01:
            return
        score = float(evaluate_sequence(a.tolist()))
        if np.isfinite(score) and score < self.best_score:
            self.best_score = score
            self.best = a.copy()
            elapsed = time.monotonic() - self.t_start
            print(f"  ** NEW BEST: J={score:.8f} n={a.size} t={elapsed:.0f}s", flush=True)

    def report(self, label=""):
        if self.best is not None:
            print(f"  {label} best: J={self.best_score:.8f} n={self.best.size}", flush=True)


def generate_initializations(n, rng, parent=None):
    """Generate diverse initial sequences of length n."""
    inits = []
    t_arr = np.linspace(0, 1, n, endpoint=False)

    if parent is not None:
        p = resample_seq(parent, n)
        inits.append(p)
        inits.append(np.maximum(p + rng.normal(0, 0.03, n), 0.01))
        inits.append(np.maximum(p + rng.normal(0, 0.08, n), 0.01))
        inits.append(np.maximum(p * rng.uniform(0.7, 1.3, n), 0.01))

    # Frequency domain
    inits.append(frequency_domain_gs(n, rng, n_iters=40))

    # Basic structures
    inits.append(np.ones(n))
    inits.append(np.maximum(1.0 + 0.3 * np.cos(2 * np.pi * 2 * t_arr), 0.01))
    inits.append(np.maximum(1.0 + 0.4 * np.cos(2 * np.pi * 3 * t_arr + 0.5), 0.01))
    inits.append(np.maximum(1.0 + 0.3 * np.sin(2 * np.pi * 5 * t_arr), 0.01))
    inits.append(np.maximum(1.0 + 0.2 * np.cos(2 * np.pi * 1 * t_arr) + 0.2 * np.cos(2 * np.pi * 4 * t_arr), 0.01))

    # Windows
    inits.append(np.exp(-3.0 * (t_arr - 0.5) ** 2) + 0.05)
    inits.append(np.maximum(1.0 - 0.5 * np.abs(2 * t_arr - 1), 0.05))
    inits.append(np.maximum(1.0 - 0.3 * np.abs(2 * t_arr - 1) ** 2, 0.05))

    # Smooth random
    ksize = max(3, n // 30)
    kernel = np.ones(ksize) / ksize
    smooth = np.convolve(rng.uniform(0, 1, n), kernel, mode='same')
    inits.append(np.maximum(smooth + 0.1, 0.01))

    # Block with transition
    n_blocks = max(4, n // 64)
    block_vals = rng.uniform(0.3, 1.0, n_blocks)
    block_seq = np.repeat(block_vals, n // n_blocks + 1)[:n]
    # Smooth the transitions
    ksize2 = max(3, n // (n_blocks * 4))
    kernel2 = np.ones(ksize2) / ksize2
    inits.append(np.maximum(np.convolve(block_seq, kernel2, mode='same') + 0.05, 0.01))

    return inits
