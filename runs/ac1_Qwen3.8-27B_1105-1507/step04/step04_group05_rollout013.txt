import numpy as np
from scipy.optimize import linprog
from scipy.sparse import lil_matrix, csr_matrix
import time

def fast_conv(a):
    """Full self-convolution via FFT."""
    n = len(a)
    s = 1
    while s < 2 * n - 1:
        s <<= 1
    fa = np.fft.rfft(a, s)
    return np.fft.irfft(fa * fa, s)[:2 * n - 1]

def compute_J(a):
    """Compute J(a) = 2*n*max(conv(a,a))/(sum(a))^2."""
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))

def build_conv_matrix(a):
    """Build sparse matrix A where (A @ g)[k] = (a*g)[k]."""
    n = len(a)
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        jl = max(0, k - n + 1)
        jh = min(n - 1, k)
        if jl <= jh:
            cols_seg = np.arange(jl, jh + 1)
            rows_seg = np.full(len(cols_seg), k)
            vals_seg = a[k - cols_seg]
            rows.extend(rows_seg.tolist())
            cols.extend(cols_seg.tolist())
            vals.extend(vals_seg.tolist())
    if len(rows) == 0:
        return csr_matrix((2*n-1, n))
    A = lil_matrix((2 * n - 1, n))
    A[rows, cols] = vals
    return A.tocsr()

def lp_best_response(a, M):
    """LP: max sum(g) s.t. (a*g)[k] <= M for all k, g >= 0."""
    n = len(a)
    c = -np.ones(n)
    A = build_conv_matrix(a)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
    if r.success and r.x is not None:
        g = np.maximum(r.x, 0.0)
        if g.sum() > 1e-10:
            return g
    return None

def subgradient_of_max_conv(a):
    """Compute subgradient of max_k (a*a)[k] w.r.t. a."""
    n = len(a)
    b = fast_conv(a)
    b_max = b.max()
    tol = 1e-8 * b_max + 1e-12
    active = np.where(b >= b_max - tol)[0]
    grad = np.zeros(n)
    for k in active:
        for j in range(max(0, k - n + 1), min(n, k + 1)):
            ki = k - j
            if 0 <= ki < n:
                grad[j] += a[ki]
    return grad, b_max

def lp_mirror_descent(a, max_iters, deadline, tracker, tighten_base=0.003):
    """LP mirror descent with line search and adaptive tightening."""
    a = np.maximum(a.astype(np.float64), 1e-6)
    a /= a.sum()
    n = len(a)
    current_J = compute_J(a)
    tracker.update(a)
    
    tighten = tighten_base
    stall = 0
    
    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        
        b = fast_conv(a)
        M = float(b.max())
        M_target = M * (1.0 - tighten)
        
        g = lp_best_response(a, M_target)
        if g is None:
            M_target = M * (1.0 - tighten * 0.3)
            g = lp_best_response(a, M_target)
        if g is None:
            tighten *= 0.5
            if tighten < 1e-7:
                tighten = tighten_base
            stall += 1
            if stall > 30:
                break
            continue
        
        g = np.maximum(g, 0.0)
        g_sum = g.sum()
        if g_sum < 1e-10:
            break
        g /= g_sum
        
        # Line search: try multiple step sizes
        best_a = None
        best_J = current_J
        
        progress = it / max(1, max_iters)
        base_t = max(0.02, 0.4 * (1.0 - 0.5 * progress))
        candidates = [base_t, base_t*0.5, base_t*0.7, base_t*1.3, base_t*2.0, 0.1, 0.3]
        
        for t in candidates:
            a_trial = (1.0 - t) * a + t * g
            a_trial = np.maximum(a_trial, 1e-10)
            s = a_trial.sum()
            if s < 1e-10:
                continue
            a_trial /= s
            J_trial = compute_J(a_trial)
            if J_trial < best_J - 1e-13:
                best_J = J_trial
                best_a = a_trial.copy()
        
        if best_a is not None:
            a = best_a
            current_J = best_J
            stall = 0
        else:
            stall += 1
            if stall > 20:
                break
            tighten *= 0.8
        
        if it % 50 == 0:
            tracker.update(a)
    
    tracker.update(a)
    return a, current_J

def multiplicative_weights(a, max_iters, deadline, tracker, eta_init=0.1):
    """Multiplicative weights update for the minimax problem."""
    a = np.maximum(a.astype(np.float64), 1e-6)
    a /= a.sum()
    n = len(a)
    current_J = compute_J(a)
    tracker.update(a)
    
    eta = eta_init
    patience = 0
    
    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        
        grad, b_max = subgradient_of_max_conv(a)
        
        # Normalize gradient
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-15:
            break
        grad = grad / gnorm
        
        # MW update
        a_new = a * np.exp(-eta * grad)
        a_new = np.maximum(a_new, 1e-10)
        s = a_new.sum()
        if s < 1e-10:
            break
        a_new /= s
        
        J_new = compute_J(a_new)

            current_J = J_new

            eta = max(0.001, eta * 0.999)
        else:

            if patience > 100:

        if it % 50 == 0:
            tracker.update(a)
    
    tracker.update(a)
    return a, current_J

def coord_descent_pass(a, n_iter=30):
    """One pass of coordinate descent with golden section search."""
    a = np.maximum(a.astype(np.float64), 0.0)
    S = a.sum()
    if S < 1e-10:
        return a
    n = len(a)
    b = fast_conv(a).copy()
    
    indices = list(range(n))
    np.random.shuffle(indices)
    
    for i in indices:
        ai_old = a[i]
        
        def obj(x):
            new_S = S - ai_old + x
            if new_S < 1e-10:
                return 1e10
            # Compute max conv with a[i] = x
            max_c = 0.0
            # Only k in [i, i+n-1] change
            for k in range(max(0, i), min(2*n-1, i+n)):
                ki = k - i
                if 0 <= ki < n:
                    if k == 2 * i:
                        ck = b[k] - ai_old * ai_old + x * x
                    else:
                        ck = b[k] - 2.0 * ai_old * a[ki] + 2.0 * x * a[ki]
                else:
                    ck = b[k]
                if ck > max_c:
                    max_c = ck
            # Also check unchanged k's
            for k in range(2*n-1):
                if k < i or k >= i + n:
                    if b[k] > max_c:
                        max_c = b[k]
            return 2.0 * n * max_c / (new_S * new_S)
        
        # Golden section search
        gr = (np.sqrt(5.0) - 1.0) / 2.0
        lo = 0.0
        hi = max(3.0 * ai_old, S / n * 5.0, 0.01)
        
        c_pt = hi - gr * (hi - lo)
        d_pt = lo + gr * (hi - lo)
        fc = obj(c_pt)
        fd = obj(d_pt)
        
        for _ in range(25):
            if fc < fd:
                hi = d_pt
                d_pt = c_pt
                fd = fc
                c_pt = hi - gr * (hi - lo)
                fc = obj(c_pt)
            else:
                lo = c_pt
                c_pt = d_pt
                fc = fd
                d_pt = lo + gr * (hi - lo)
                fd = obj(d_pt)
        
        x_opt = max(0.0, (lo + hi) / 2.0)
        
        if abs(x_opt - ai_old) > 1e-12:
            # Update b incrementally
            for k in range(max(0, i), min(2*n - 1, i + n)):
                ki = k - i
                if 0 <= ki < n:
                    if k == 2 * i:
                        b[k] = b[k] - ai_old * ai_old + x_opt * x_opt
                    else:
                        b[k] = b[k] - 2.0 * ai_old * a[ki] + 2.0 * x_opt * a[ki]
            a[i] = x_opt
            S = S - ai_old + x_opt
    
    return a

def coord_descent(a, n_passes, deadline, tracker):

    a = np.maximum(a.astype(np.float64), 0.0)
    a /= a.sum()

        if p % 3 == 0:
            print(f"  CD pass {p}: J={current_J:.8f} ({time.monotonic()-t0:.1f}s)", flush=True)
    
    return a

def smooth_potential_optimize(a, max_iters, deadline, tracker, lr=0.01):
    """Optimize using smooth potential: sum of (conv - t)^2 for conv > t."""
    a = np.maximum(a.astype(np.float64), 1e-6)
    a /= a.sum()
    n = len(a)
    current_J = compute_J(a)
    tracker.update(a)
    
    # Use Adam optimizer
    m = np.zeros(n)
    v = np.zeros(n)
    beta1, beta2, eps = 0.8, 0.999, 1e-8
    t_adam = 0
    lr_t = lr
    
    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        
        b = fast_conv(a)
        b_max = b.max()
        
        # Smooth potential: sum_k (b_k - t)^2 where t = b_max
        # But this is non-differentiable. Use soft version:
        # P = sum_k max(0, b_k - 0.9*b_max)^2
        threshold = 0.85 * b_max
        active = b > threshold
        if not np.any(active):
            break
        
        # Gradient of sum_k max(0, b_k - threshold)^2 w.r.t. a
        grad = np.zeros(n)
        for k in np.where(active)[0]:
            diff = b[k] - threshold
            for j in range(max(0, k - n + 1), min(n, k + 1)):
                ki = k - j
                if 0 <= ki < n:
                    grad[j] += 2.0 * diff * a[ki]
        
        # Adam update
        t_adam += 1
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * grad**2
        m_hat = m / (1 - beta1**t_adam)
        v_hat = v / (1 - beta2**t_adam)
        
        a_new = a - lr_t * m_hat / (np.sqrt(v_hat) + eps)
        a_new = np.maximum(a_new, 1e-8)
        s = a_new.sum()
        if s < 1e-10:
            break
        a_new /= s
        
        J_new = compute_J(a_new)
        
        if J_new < current_J - 1e-12:
            a = a_new
            current_J = J_new
            lr_t = lr_t * 1.01  # increase slightly
        else:
            lr_t = max(1e-6, lr_t * 0.8)
        
        if it % 100 == 0:
            tracker.update(a)
            # Decay learning rate
            lr_t = lr * (0.95 ** (it // 100))
    
    tracker.update(a)
    return a

def resample_seq(src, dest_n):
    """Resample 1D sequence to new length via linear interpolation."""
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    return np.maximum(np.interp(t_new, t_old, src), 1e-6)

class ScoreTracker:
    def __init__(self):
        self.best = None
        self.best_score = float('inf')
        self.t_start = time.monotonic()
    
    def update(self, a):
        a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
        if a.size == 0 or a.sum() < 0.01:
            return
        score = float(evaluate_sequence(a.tolist()))
        if np.isfinite(score) and score < self.best_score:
            self.best = a.copy()
            self.best_score = score
            elapsed = time.monotonic() - self.t_start
            print(f"  * NEW BEST: J={score:.8f} n={a.size} t={elapsed:.0f}s", flush=True)
    
    def report(self, label=""):
        if self.best is not None:
            print(f"  {label} best: J={self.best_score:.8f} n={self.best.size} t={time.monotonic()-self.t_start:.0f}s", flush=True)
