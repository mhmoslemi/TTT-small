    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))


def build_A_coo(n, a):
    """Build COO-sparse matrix A where A[k,j] = a[k-j] for valid j."""
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        j_min = max(0, k - n + 1)
        j_max = min(n - 1, k)
        for j in range(j_min, j_max + 1):
            rows.append(k)
            cols.append(j)
            vals.append(a[k - j])
    return coo_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))


def lp_best_response(a, M):
    """Solve LP: max sum(g) s.t. A@g <= M, g >= 0. A[k,j] = a[|k-j|] structure."""
    n = len(a)
    A = build_A_coo(n, a)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    try:
        r = linprog(c, A_ub=A.tocsr(), b_ub=b_ub, bounds=bounds,
                    method='highs', options={'presolve': True})
        if r.success and r.x is not None:
            g = np.maximum(r.x, 0)
            if g.sum() > 1e-15:
                return g
    except Exception:
        pass
    return None


def lp_mirror_descent(a, deadline, max_iters=5000):
    """LP mirror descent with adaptive step size. Returns (best_a, best_J)."""
    a = np.maximum(a.astype(np.float64), 1e-10)
    s = a.sum()
    a = a / s
    current_J = compute_J(a)
    best_J = current_J
    best_a = a.copy()
    no_improve = 0
    n = len(a)

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        b = fast_conv(a)
        M = float(b.max())
        g = lp_best_response(a, M)
        if g is None:
            break
        g = np.maximum(g, 0.0)
        g_sum = g.sum()
        if g_sum < 1e-10:
            break
        g = g / g_sum

        # Adaptive step size: start larger, decay over time
        base_t = 0.3 * np.exp(-0.2 * it / 200.0) + 0.08
        t = min(1.0, base_t)

        best_trial_J = current_J
        best_trial_a = None

        for _ in range(20):
            a_trial = (1.0 - t) * a + t * g
            a_trial = np.maximum(a_trial, 0.0)
            st = a_trial.sum()
            if st < 1e-10:
                t *= 0.5
                continue
            a_trial = a_trial / st
            J_trial = compute_J(a_trial)
            if J_trial < best_trial_J - 1e-14:
                best_trial_J = J_trial
                best_trial_a = a_trial.copy()
                break
            t *= 0.6
            if t < 0.001:
                break

        if best_trial_a is not None:
            a = best_trial_a
            current_J = best_trial_J
            if current_J < best_J:
                best_J = current_J
                best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1
            if no_improve > 50:
                break

    return best_a, best_J


def nesterov_subgradient(a, deadline, max_iters=10000):
    """Nesterov-accelerated subgradient descent on the simplex."""
    a = np.maximum(a.astype(np.float64), 1e-10)
    a = a / a.sum()
    n = len(a)
    best_M = fast_conv(a).max()
    best_a = a.copy()
    v = np.zeros_like(a)  # velocity for Nesterov
    eta0 = 1.0
    gamma = 0.9

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        # Nesterov lookahead
        a_look = a - gamma * v
        a_look = np.maximum(a_look, 0)
        sl = a_look.sum()
        if sl > 1e-10:
            a_look = a_look / sl
        else:
            a_look = a

        b = fast_conv(a_look)
        M = b.max()
        tol = 1e-10 * max(M, 1e-15)
        active = np.where(b >= M - tol)[0]

        # Subgradient of max_k conv(a,a)_k
        grad = np.zeros_like(a)
        for k in active:
            # d/d(a_j) conv(k) = a[j-k] + a[j+k] (clipped to valid range)
            for sign in [1, -1]:
                shift = sign * k
                if shift >= 0:
                    grad[:n - shift] += a[shift:]
                else:
                    grad[-shift:] += a[:n + shift]

        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-15:
            break

        # Adaptive step size
        eta = eta0 / np.sqrt(it + 1) * 0.5
        v = gamma * v + (1 - gamma) * grad / gnorm * eta
        a_new = a - v
        a_new = np.maximum(a_new, 0)
        s = a_new.sum()
        if s < 1e-10:
            break
        a_new = a_new / s

        new_M = fast_conv(a_new).max()
        if new_M < best_M:
            best_M = new_M
            best_a = a_new.copy()
        a = a_new

    return best_a


def multiplicative_weights(a, deadline, max_iters=5000):
    """Multiplicative weights update for the LP mirror ascent."""
    a = np.maximum(a.astype(np.float64), 1e-10)
    a = a / a.sum()
    best_J = compute_J(a)
    best_a = a.copy()
    no_improve = 0
    n = len(a)

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        b = fast_conv(a)
        M = float(b.max())
        g = lp_best_response(a, M)
        if g is None:
            break
        g = np.maximum(g, 0.0)
        gs = g.sum()
        if gs < 1e-10:
            break
        g = g / gs

        eta = 0.15 * np.exp(-0.15 * it / 200.0) + 0.03
        best_trial_J = compute_J(a)
        best_trial_a = None

        for _ in range(15):
            ratio = np.where(a > 1e-12, g / np.maximum(a, 1e-12), 1.0)
            a_trial = a * np.power(ratio, eta)
            a_trial = np.maximum(a_trial, 0)
            st = a_trial.sum()
            if st < 1e-10:
                eta *= 0.5
                continue
            a_trial = a_trial / st
            J_trial = compute_J(a_trial)
            if J_trial < best_trial_J - 1e-14:
                best_trial_J = J_trial
                best_trial_a = a_trial.copy()
                break
            eta *= 0.6
            if eta < 0.001:
                break

        if best_trial_a is not None:
            a = best_trial_a
            if best_trial_J < best_J:
                best_J = best_trial_J
                best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1
            if no_improve > 40:
                break

    return best_a, best_J


def resample_seq(src, dest_n):
    """Resample 1D sequence to new length."""
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)


def generate_initializations(n, parent=None):
    """Generate diverse initial sequences of length n."""
    inits = []
    t = np.arange(n, dtype=np.float64) / n

    # Uniform (optimal for this problem)
    inits.append(np.ones(n) / n)

    # Slight perturbations of uniform
    rng = np.random.default_rng(12345 + n)
    inits.append(np.maximum(1 + 0.1 * rng.standard_normal(n), 0.01))
    inits.append(np.maximum(1 + 0.05 * rng.standard_normal(n), 0.01))

    # Smooth waveforms
    inits.append(np.maximum(1 + 0.3 * np.cos(2 * np.pi * t), 0.01))
    inits.append(np.maximum(1 + 0.3 * np.cos(4 * np.pi * t), 0.01))
    inits.append(np.maximum(1 + 0.2 * np.cos(2 * np.pi * t) + 0.1 * np.cos(4 * np.pi * t), 0.01))
    inits.append(np.exp(-8 * (t - 0.5) ** 2) + 0.2)
    inits.append(np.maximum(1 - 0.5 * np.abs(2 * t - 1), 0.05))

    # Parent (resampled)
    if parent is not None:
        p = resample_seq(parent, n)
        inits.append(p)
        inits.append(np.maximum(p + 0.03 * rng.standard_normal(n), 0.01))

    # Normalize all
    result = []
    for a in inits:
        a = np.maximum(a, 0.001)
        s = a.sum()
        if s > 1e-6:
            result.append(a / s)
    return result


class ScoreTracker:
    def __init__(self):
        self.best = None
        self.best_score = float('inf')
        self.t_start = time.monotonic()

    def update(self, a):
        a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
        if a.size == 0 or a.sum() < 0.01:
            return
        score = float(evaluate_sequence(a.tolist()))
        if np.isfinite(score) and score < self.best_score:
            self.best = a.copy()
            self.best_score = score
            elapsed = time.monotonic() - self.t_start
            print(f"  [NEW BEST] J={score:.8f} n={a.size} t={elapsed:.0f}s", flush=True)

    def report(self, label=""):
        if self.best is not None:
            print(f"  {label} best: J={self.best_score:.8f} n={self.best.size}", flush=True)


def lbb_objective(x):
    """Smooth approximation of max(conv(x,x)) for L-BFGS-B."""
    ax = np.maximum(x, 0.0)
    s = ax.sum()
    if s < 1e-8:
        return 1e10
    ax = ax / s
    b = fast_conv(ax)
    # Smooth max via softmax
    scale = 100.0
    shifted = b / scale
    max_val = shifted.max()
    log_sum = max_val + np.log(np.sum(np.exp(shifted - max_val)))
    return float(log_sum * scale)


def propose_candidate(seed=42, budget_s=1500, **kwargs):
    """
    Optimizer: LP mirror descent + Nesterov subgradient + multiplicative weights.
    Multi-length search with aggressive time allocation to LP.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + float(budget_s) - 10.0
    tracker = ScoreTracker()

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        if p.ndim == 1 and p.size > 0 and np.all(np.isfinite(p)):
            parent = np.maximum(p, 0.0)
            tracker.update(parent)
            print(f"Parent: n={parent.size}, J={tracker.best_score:.6f}", flush=True)
    except (NameError, AttributeError):
        pass

    # Determine which lengths to try
    if parent is not None:
        base_n = parent.size
        lengths = sorted(set([
            256, 384, 512, max(128, base_n - 64), base_n,
            base_n + 64, base_n + 128,
            min(2048, base_n * 2)
        ]))
    else:
        lengths = [256, 512, 640, 1024]

    # Filter reasonable lengths
    lengths = [n for n in lengths if 64 <= n <= 4096]

    # Allocate time: give most time to the best-performing length
    # Phase 1: Quick scan of all lengths with subgradient (fast)
    print(f"\n{'='*50}", flush=True)
    print(f"Phase 1: Quick subgradient scan ({len(lengths)} lengths)", flush=True)
    print(f"{'='*50}", flush=True)

    length_results = {}
    per_length_time = max(5, min(30, (budget_s * 0.15) / len(lengths)))

    for n in lengths:
        if time.monotonic() > deadline - 100:
            break
        inits = generate_initializations(n, parent)
        best_a = None
        best_J = float('inf')
        t0 = time.monotonic()

        for init in inits[:4]:  # Try at most 4 inits per length
            if time.monotonic() > t0 + per_length_time:
                break
            a_sub = nesterov_subgradient(init, min(time.monotonic() + per_length_time / 2, deadline - 50),
                                         max_iters=3000)
            J_sub = compute_J(a_sub)
            if J_sub < best_J:
                best_J = J_sub
                best_a = a_sub.copy()

        if best_a is not None:
            length_results[n] = (best_a, best_J)
            tracker.update(best_a)
            print(f"  n={n}: J={best_J:.8f} ({time.monotonic()-t0:.1f}s)", flush=True)

    if not length_results:
        # Fallback: use parent or uniform
        if parent is not None:
            final = parent / parent.sum()
        else:
            final = np.ones(512) / 512
        tracker.update(final)
        return [float(x) for x in tracker.best]

    # Sort lengths by performance
    sorted_lengths = sorted(length_results.keys(), key=lambda n: length_results[n][1])
    best_n = sorted_lengths[0]
    best_a = length_results[best_n][0]
    best_J = length_results[best_n][1]
    print(f"\nBest so far: n={best_n}, J={best_J:.8f}", flush=True)

    # Phase 2: LP mirror descent on top candidates (most time here)
    print(f"\n{'='*50}", flush=True)
    print(f"Phase 2: LP mirror descent (top lengths)", flush=True)
    print(f"{'='*50}", flush=True)

    # Give 60% of remaining time to LP
    remaining = deadline - time.monotonic()
    lp_time = remaining * 0.55
    top_lengths = sorted_lengths[:min(4, len(sorted_lengths))]
    per_lp_time = lp_time / len(top_lengths)

    for n in top_lengths:
        if time.monotonic() > deadline - 80:
            break
        a_start = length_results[n][0].copy()
        a_start = np.maximum(a_start, 1e-10)
        a_start = a_start / a_start.sum()

        lp_deadline = min(time.monotonic() + per_lp_time, deadline - 70)
        print(f"  LP on n={n} (budget {per_lp_time:.0f}s)...", flush=True)
        t0 = time.monotonic()
        a_lp, J_lp = lp_mirror_descent(a_start, lp_deadline, max_iters=10000)
        tracker.update(a_lp)
        print(f"    n={n}: J={J_lp:.8f} ({time.monotonic()-t0:.1f}s)", flush=True)

        if J_lp < best_J:
            best_J = J_lp
            best_a = a_lp.copy()
            best_n = n
            length_results[n] = (a_lp, J_lp)

    # Phase 3: Multiplicative weights on best
    if time.monotonic() < deadline - 50:
        print(f"\n{'='*50}", flush=True)
        print(f"Phase 3: Multiplicative weights (n={best_n})", flush=True)
        print(f"{'='*50}", flush=True)

        a_mw = best_a.copy()
        a_mw = np.maximum(a_mw, 1e-10)
        a_mw = a_mw / a_mw.sum()
        mw_deadline = min(time.monotonic() + (deadline - time.monotonic()) * 0.3, deadline - 40)
        t0 = time.monotonic()
        a_mw, J_mw = multiplicative_weights(a_mw, mw_deadline, max_iters=10000)
        tracker.update(a_mw)
        print(f"  J={J_mw:.8f} ({time.monotonic()-t0:.1f}s)", flush=True)

        if J_mw < best_J:
            best_J = J_mw
            best_a = a_mw.copy()

    # Phase 4: Another LP round on the best
    if time.monotonic() < deadline - 30:
        print(f"\n{'='*50}", flush=True)
        print(f"Phase 4: Final LP refinement (n={best_n})", flush=True)
        print(f"{'='*50}", flush=True)

        a_final = best_a.copy()
        a_final = np.maximum(a_final, 1e-10)
        a_final = a_final / a_final.sum()
        t0 = time.monotonic()
        a_final, J_final = lp_mirror_descent(a_final, deadline - 25, max_iters=10000)
        tracker.update(a_final)
        print(f"  J={J_final:.8f} ({time.monotonic()-t0:.1f}s)", flush=True)

        if J_final < best_J:
            best_J = J_final
            best_a = a_final.copy()

    # Phase 5: L-BFGS-B polishing (if time allows)
    if time.monotonic() < deadline - 5 and tracker.best is not None:
        t0 = time.monotonic()
        n_lb = tracker.best.size
        try:
            x0 = tracker.best.astype(np.float64)
            res = minimize(lbb_objective, x0, method='L-BFGS-B',
                           bounds=[(0.0, None)] * n_lb,
                           options={'maxiter': 200, 'max_fun': 50000,
                                    'ftol': 1e-16, 'gtol': 1e-14})
            if res.fun < 1e10:
                tracker.update(res.x)
        except Exception as e:
            print(f"  L-BFGS-B error: {e}", flush=True)
        print(f"  L-BFGS-B done ({time.monotonic()-t0:.1f}s)", flush=True)

    # Final
    if tracker.best is not None:
        final_a = tracker.best
    else:
        final_a = best_a

    print(f"\n{'='*50}", flush=True)
    print(f"FINAL: J={tracker.best_score:.8f} n={final_a.size} "
          f"time={time.monotonic()-tracker.t_start:.0f}s", flush=True)
    print(f"{'='*50}", flush=True)

    return [float(x) for x in final_a]
