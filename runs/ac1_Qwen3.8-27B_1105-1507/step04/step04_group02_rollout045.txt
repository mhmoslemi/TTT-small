from scipy.optimize import linprog
from scipy.sparse import coo_matrix
import time

def compute_conv(a):
    n = len(a)
    size = 1
    while size < 2*n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    conv = np.fft.irfft(fa * fa, size)
    return conv[:2*n-1]

def compute_obj(a):
    n = len(a)
    S = np.sum(a)
    if S < 1e-12:
        return np.inf
    conv = compute_conv(a)
    M = np.max(conv)
    return 2.0 * n * M / (S * S)

def build_toeplitz(a):
    """Build sparse matrix T such that (T @ g)[k] = (a*g)[k]."""
    n = len(a)
    rows, cols, vals = [], [], []
    for k in range(2*n - 1):
        i_min = max(0, k - (n-1))
        i_max = min(n-1, k)
        for i in range(i_min, i_max + 1):
            rows.append(k)
            cols.append(i)
            vals.append(a[k - i])
    return coo_matrix((vals, (rows, cols)), shape=(2*n-1, n)).tocsr()

def solve_lp(a, M, time_limit=5):
    n = len(a)
    T = build_toeplitz(a)
    c = -np.ones(n)
    b_ub = np.full(2*n-1, M)
    bounds = [(0, None)] * n
    try:
        res = linprog(c, A_ub=T, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': time_limit})
        if res.status == 0:
            return np.maximum(res.x, 0)
    except:
        pass
    return None

sum_{k: 0<=k-i<n} lambda_k * a[k-i] >= 1

Let j = k-i, this becomes:
sum_{j: 0<=j<n, 0<=j+i<2n-1} lambda_{j+i} * a[j] >= 1

This is a "weighted autocorrelation" of lambda with a, evaluated at shift i.

By strong duality, at the optimum:
- Primal: sum(g) = M * sum(lambda)
- Complementary slackness: g_i * (dual_slack_i) = 0, lambda_k * (M - (a*g)[k]) = 0

So the optimal g has support only where the dual constraints are tight, and the optimal lambda has support only where (a*g)[k] = M.

This gives us a way to check convergence: if the KKT conditions are approximately satisfied, we're near the optimum.

Uses: efficient LP alternation with adaptive M-scheduling, FFT-based convolution,

import numpy as np
from scipy.optimize import linprog
from scipy.sparse import coo_matrix
import time


def compute_conv(a):
    """Compute linear convolution a*a via FFT."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    conv = np.fft.irfft(fa * fa, size)
    return conv[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    S = float(np.sum(a))
    if S < 1e-10:
        return np.inf
    conv = compute_conv(a)
    M = float(np.max(conv))
    return 2.0 * n * M / (S * S)


def build_toeplitz_sparse(a):
    """Build sparse Toeplitz matrix T such that (T @ g)[k] = (a*g)[k]."""
    n = len(a)
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        i_min = max(0, k - (n - 1))
        i_max = min(n - 1, k)
        for i in range(i_min, i_max + 1):
            rows.append(k)
            cols.append(i)
            vals.append(a[k - i])
    return coo_matrix((vals, (rows, cols)), shape=(2 * n - 1, n)).tocsr()


def solve_lp_maxsum(a, M, time_limit=5.0):
    """Solve LP: max sum(g) s.t. conv(a,g) <= M, g >= 0."""
    n = len(a)
    T = build_toeplitz_sparse(a)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c, A_ub=T, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': time_limit})
        if res.status == 0:
            return np.maximum(res.x, 0.0)
    except Exception:
        pass
    return None


def solve_lp_reduced(a, M, active_indices, time_limit=4.0):
    """Solve LP restricted to active_indices (variables outside are fixed to 0)."""
    n = len(a)
    n_active = len(active_indices)
    if n_active == 0:
        return None
    a_active = a[active_indices]
    # Build reduced Toeplitz
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        for idx_local, i in enumerate(active_indices):
            j = k - i
            if 0 <= j < n:
                rows.append(k)
                cols.append(idx_local)
                vals.append(a[j])
    T = coo_matrix((vals, (rows, cols)), shape=(2 * n - 1, n_active)).tocsr()
    c = -np.ones(n_active)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n_active
    try:
        res = linprog(c, A_ub=T, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': time_limit})
        if res.status == 0:
            g_full = np.zeros(n)
            g_full[active_indices] = np.maximum(res.x, 0.0)
            return g_full
    except Exception:
        pass
    return None


def lp_alternation(a, max_iters=200, time_budget=60.0, M_reduction=0.999, topk_lp=None):
    """
    LP alternation with adaptive M scheduling.
    At each step: solve LP for g given current a and target M, then update a.
    """
    n = len(a)
    a = a.copy().astype(np.float64)
    S = np.sum(a)
    if S < 1e-10:
        return a, compute_obj(a)
    a = a / S

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    stall = 0
    current_M_target = None

    # Track active set (indices where a[i] > threshold)
    active_mask = a > 1e-8 * a.max()
    active_indices = np.where(active_mask)[0]

    for it in range(max_iters):
        if time.monotonic() - start > time_budget:
            break

        conv = compute_conv(a)
        M_cur = float(np.max(conv))

        # Schedule M target: initially use current M, then try to reduce
        if current_M_target is None:
            M_target = M_cur
        else:
            M_target = max(M_cur * M_reduction, current_M_target * 0.995)

        # Solve LP (use reduced LP if active set is much smaller)
        if topk_lp and len(active_indices) > n // 2:
            g = None  # Use full LP
        elif len(active_indices) < n * 0.7 and len(active_indices) > 0:
            g = solve_lp_reduced(a, M_target, active_indices, time_limit=3.0)
            if g is None:
                g = solve_lp_maxsum(a, M_target, time_limit=3.0)
        else:
            g = solve_lp_maxsum(a, M_target, time_limit=4.0)

        if g is None:
            M_target = M_cur  # Fall back to current M
            g = solve_lp_maxsum(a, M_target, time_limit=4.0)
            if g is None:
                break

        S_g = np.sum(g)
        if S_g < 1e-10:
            break
        g = g / S_g

        # Line search
        best_t = 0.0
        best_obj_ls = best_obj
        for t in [0.1, 0.2, 0.3, 0.5, 0.7, 0.9, 1.0]:
            a_try = (1.0 - t) * a + t * g
            a_try = np.maximum(a_try, 0.0)
            S_try = np.sum(a_try)
            if S_try < 1e-10:
                continue
            a_try = a_try / S_try
            obj_try = compute_obj(a_try)
            if obj_try < best_obj_ls:
                best_obj_ls = obj_try
                best_t = t

        if best_t > 0.0:
            a = (1.0 - best_t) * a + best_t * g
            a = np.maximum(a, 0.0)
            a = a / np.sum(a)
            best_obj = best_obj_ls
            best_a = a.copy()
            stall = 0
            # Update M target
            conv_new = compute_conv(a)
            current_M_target = float(np.max(conv_new))
        else:
            stall += 1
            if stall > 10:
                # Try with higher M target (optimistic)

                g2 = solve_lp_maxsum(a, M_target, time_limit=4.0)
                if g2 is not None:
                    S_g2 = np.sum(g2)
                    if S_g2 > 1e-10:
                        g2 = g2 / S_g2
                        a = 0.5 * a + 0.5 * g2
                        a = np.maximum(a, 0.0)
                        a = a / np.sum(a)
                    stall = 0
                else:
                    # Perturb
                    rng = np.random.default_rng(it * 104729 + 7)
                    a = a * np.exp(rng.normal(0, 0.05, n))
                    a = np.maximum(a, 0.0)
                    a = a / np.sum(a)
                    stall = 0
                    current_M_target = None

        # Update active set periodically
        if it % 20 == 0:
            active_mask = a > 1e-10 * a.max()
            active_indices = np.where(active_mask)[0]

    return best_a, best_obj


def subgradient_polish(a, max_iters=500, time_budget=10.0, topk=10):
    """Subgradient descent with momentum for fine polishing."""
    n = len(a)
    a = a.copy().astype(np.float64)
    S = np.sum(a)
    if S < 1e-10:
        return a, compute_obj(a)
    a = a / S

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    v = np.zeros(n)
    momentum = 0.8
    step = 0.08
    stall = 0

    for it in range(max_iters):
        if time.monotonic() - start > time_budget:
            break

        conv = compute_conv(a)
        M = float(np.max(conv))

        # Find top-k active positions
        k_active = np.argsort(conv)[-topk:]

        # Compute subgradient
        grad = np.zeros(n)
        for k in k_active:
            i_min = max(0, k - (n - 1))
            i_max = min(n - 1, k)
            for i in range(i_min, i_max + 1):
                grad[i] += 2.0 * a[k - i]
        grad /= len(k_active)

        # Center gradient (preserve sum)
        grad_centered = grad - np.mean(grad)

        v = momentum * v + (1.0 - momentum) * grad_centered
        a_new = a - step * v
        a_new = np.maximum(a_new, 0.0)
        S_new = np.sum(a_new)
        if S_new < 1e-10:
            break
        a_new = a_new / S_new

        obj = compute_obj(a_new)
        if obj < best_obj - 1e-12:
            best_obj = obj
            best_a = a_new.copy()
            a = a_new
            stall = 0
        else:
            stall += 1
            if stall > 50:
                step *= 0.7
                v = 0.5 * v
                if step < 1e-10:
                    break
                stall = 0

    return best_a, best_obj


def multiplicative_polish(a, max_iters=300, time_budget=8.0, topk=8):
    """Multiplicative (exponential) weight update."""
    n = len(a)
    a = a.copy().astype(np.float64)
    S = np.sum(a)
    if S < 1e-10:
        return a, compute_obj(a)
    a = a / S

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    eta = 0.3

    for it in range(max_iters):
        if time.monotonic() - start > time_budget:
            break

        conv = compute_conv(a)
        k_active = np.argsort(conv)[-topk:]

        grad = np.zeros(n)
        for k in k_active:
            i_min = max(0, k - (n - 1))
            i_max = min(n - 1, k)
            for i in range(i_min, i_max + 1):
                grad[i] += 2.0 * a[k - i]
        grad /= len(k_active)
        grad_centered = grad - np.mean(grad)

        a_new = a * np.exp(-eta * grad_centered)
        a_new = np.maximum(a_new, 1e-15)
        a_new = a_new / np.sum(a_new)

        obj = compute_obj(a_new)
        if obj < best_obj - 1e-12:
            best_obj = obj
            best_a = a_new.copy()
            a = a_new
        else:
            eta *= 0.85
            if eta < 1e-10:
                break

    return best_a, best_obj


def cosine_init(n, n_modes=8, seed=0):
    """Generate a smooth cosine-series initialization."""
    rng = np.random.default_rng(seed)
    k = np.arange(n)
    a = np.ones(n)
    for m in range(1, n_modes + 1):
        c = rng.uniform(-0.5, 0.5) / m
        phase = rng.uniform(0, 2 * np.pi)
        a += c * np.cos(2 * np.pi * m * k / n + phase)
    a = np.maximum(a, 0)
    S = np.sum(a)
    if S < 1e-10:
        return np.ones(n) / n
    return a / S


def asymmetric_init(n, seed=0):

    t = k / (n - 1)

        best_seq = np.ones(300) / 300
        best_score = compute_obj(best_seq)

    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    val = evaluate_sequence(result)
    if val < np.inf:
        if val < best_score:
            best_score = val
    else:
        result = [1.0] * 300
        best_score = 2.0

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_score:.8f}, time={elapsed:.1f}s ===")

    return result
