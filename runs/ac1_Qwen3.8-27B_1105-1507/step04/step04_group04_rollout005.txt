        # Let me just do it directly for small n, or use a smarter FFT trick.
        # For n=300, direct O(n^2) is 90000 ops, which is fine.
        
        # Actually, let's use the FFT trick:
        # sum_j u[i+j] * v[j] = sum_j u[i+j] * v[j]
        # If we define w[i+j] = u[i+j] * v[j], this is like a "sliding dot product".
        # This is the same as the (i)-th element of the correlation of u and v.
        
        # np.correlate(u, v, 'full')[k] = sum_j u[j+k] * v[j] (for valid j)
        # Wait, numpy's correlate: np.correlate(a, v, mode) computes
        # c[k] = sum_n a[n+k] * conj(v[n])
        # So np.correlate(u, v, 'full')[i] = sum_j u[j+i] * v[j]
        # That's exactly what we want!
        
        corr = np.correlate(weights, v, mode='full')
        # corr[i] = sum_j weights[j+i] * v[j] for i = -(n-1) to (2n-2)
        # We want the part where i = 0 to n-1 (for the i-th row of H)
        # corr has length (2n-1) + n - 1 = 3n-2
        # The offset is (n-1), so corr[n-1 + i] = sum_j weights[j+i] * v[j]
        # Wait, for full mode: length = len(u) + len(v) - 1 = (2n-1) + n - 1 = 3n-2
        # corr[k] for k = 0 to 3n-3
        # The "zero lag" is at k = len(v) - 1 = n-1
        # corr[n-1 + i] = sum_j weights[j+i] * v[j] (when j+i < 2n-1 and j < n)
        
        result = corr[n-1:n-1+n]
        return 2.0 * result
    
    # FISTA (fast iterative shrinkage-thresholding) for the QP
    # minimize f(a) = 0.5 * a^T H a subject to a >= 0, sum(a) = 1
    # (the 0.5 factor is just for convenience)
    
    # Lipschitz constant estimate: ||H||_2 <= sum of absolute row sums
    # For now, use a rough estimate
    L = 2.0 * np.sum(weights) * 2  # rough upper bound on ||H||
    
    # Actually, let's compute H @ ones to get a better estimate
    # ||H|| <= max row sum of |H|
    # H[i,j] = 2*weights[i+j], so row i sum = 2*sum_{j: i+j<2n-1} weights[i+j]
    # Max over i of this.
    row_sums = np.array([2.0 * np.sum(weights[i:i+n]) for i in range(n)])
    L = np.max(row_sums)
    
    step = 1.0 / L
    momentum = 0.0
    k_momentum = 1.0
    
    a_prev = a.copy()
    
    for it in range(max_iters):
        # Gradient step
        grad = Hv(a)  # This is H @ a = 2 * sum_k weights[k] * c[k-direction]
        # Actually, grad[i] = 2 * sum_j H[i,j]/2 * a[j] ... 
        # Wait, I defined H[i,j] = 2*weights[i+j], so grad = H @ a
        # But the actual gradient of 0.5*a^T H a is H a (if H is symmetric)
        # Our H is symmetric (H[i,j] = weights[i+j] = H[j,i]), so grad = H@a
        
        # Wait, I had H[i,j] = 2*weights[i+j]. Let me redo.
        # f(a) = sum_k weights[k] * c[k] = sum_k weights[k] * sum_i a[i]*a[k-i]
        # = sum_{i,j} a[i]*a[j]*weights[i+j]
        # = a^T W a where W[i,j] = weights[i+j]
        # gradient = (W + W^T) a = 2W a (since W is symmetric)
        # So grad[i] = 2 * sum_j weights[i+j] * a[j]
        
        # That's what Hv computes. Good.
        
        a_new = a - step * grad
        a_new = np.maximum(a_new, 0)
        
        # Project onto sum = 1
        s = np.sum(a_new)
        if s < 1e-10:
            a_new = np.ones(n) / n
        else:
            # Need to project onto {a >= 0, sum(a) = 1}
            # If s != 1, we need to find tau such that sum(max(a_new - tau, 0)) = 1
            # Use bisection
            if abs(s - 1.0) > 1e-8:
                lo, hi = -1.0, max(a_new) + 1.0
                for _ in range(50):
                    mid = (lo + hi) / 2
                    s_mid = np.sum(np.maximum(a_new - mid, 0))
                    if s_mid > 1.0:
                        lo = mid
                    else:
                        hi = mid
                a_new = np.maximum(a_new - (lo + hi) / 2, 0)
                s = np.sum(a_new)
                if s > 0:
                    a_new = a_new / s
        
        # FISTA momentum
        k_new = (1.0 + np.sqrt(1.0 + 4.0 * k_momentum**2)) / 2.0
        a = a_new + ((k_momentum - 1.0) / k_new) * (a_new - a_prev)
        a = np.maximum(a, 0)
        s = np.sum(a)
        if s > 0:
            a = a / s
        else:
            a = np.ones(n) / n
        
        a_prev = a_new
        k_momentum = k_new
    
    return a
