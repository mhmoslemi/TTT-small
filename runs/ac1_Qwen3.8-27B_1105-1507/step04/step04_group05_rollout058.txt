"""
Minimizes J(a) = 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Fast vectorized coordinate descent (O(n) per 1D eval via numpy)
2. LP-based best response with adaptive tightening and line search
3. Subgradient descent with active-set gradient
4. Multi-length exploration via resampling
5. Perturbation restarts with re-optimization
6. Pair coordinate descent exploiting symmetry

import numpy as np
from scipy.optimize import linprog
from scipy.sparse import lil_matrix
import time


def fast_conv(a):
    """Full self-convolution via FFT."""
    n = len(a)
    s = 1
    while s < 2 * n - 1:
        s <<= 1
    fa = np.fft.rfft(a, s)
    return np.fft.irfft(fa * fa, s)[:2 * n - 1]


def compute_J(a):
    """Compute J(a) = 2*n*max(conv(a,a))/(sum(a))^2."""
    a = np.maximum(a, 0)
    S = a.sum()
    if S < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (S * S))


def build_conv_matrix(n, a):
    """Build sparse matrix A such that A@g = convolve(a, g)."""
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        jl = max(0, k - n + 1)
        jh = min(n - 1, k)
        if jl <= jh:
            cnt = jh - jl + 1
            rows.extend([k] * cnt)
            cols.extend(range(jl, jh + 1))
            seg = a[k - jh:k - jl + 1][::-1]
            vals.extend(seg.tolist())
    A = lil_matrix((2 * n - 1, n))
    for idx in range(len(rows)):
        A[rows[idx], cols[idx]] = vals[idx]
    return A.tocsr()


def lp_best_response(a, M):
    """LP: max sum(g) s.t. (a*g)_k <= M for all k, g >= 0."""
    n = len(a)
    c = -np.ones(n)
    A = build_conv_matrix(n, a)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
    if r.success and r.x is not None:
        g = np.maximum(r.x, 0.0)
        if g.sum() > 1e-10:
            return g
    return None

            ki_vals = k - np.arange(jl, jh + 1)
            grad[jl:jh + 1] += a[ki_vals]

        S = a.sum()
        grad -= 2.0 * b_max / S

        a_new = a - step_size * grad
        a_new = np.maximum(a_new, 0.0)
        if a_new.sum() < 1e-10:
            break
        a_new = a_new / a_new.sum()

        new_J = compute_J(a_new)
        if new_J < current_J:
            a = a_new
            current_J = new_J
            tracker_update(a, tracker)
        else:
            step_size *= 0.5

    tracker_update(a, tracker)
    return a


def lp_iterations(a, max_iters, tracker, deadline, tighten_base=0.005):
    """LP mirror descent iterations."""
    a = np.maximum(a.astype(np.float64), 0.001)
    a = a / a.sum()
    n = len(a)
    current_J = compute_J(a)
    tracker_update(a, tracker)
    tighten = tighten_base
    stall = 0

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        b = fast_conv(a)
        M = float(b.max())
        M_target = M * (1.0 - tighten)

        g = lp_best_response(a, M_target)
        if g is None:
            M_target = M * (1.0 - tighten * 0.5)
            g = lp_best_response(a, M_target)
        if g is None:
            tighten *= 0.5
            if tighten < 1e-7:
                tighten = tighten_base
            stall += 1
            if stall > 30:
                break
            continue

        g = np.maximum(g, 0.0)
        g = g / g.sum()

        best_a = None
        best_J = current_J
        progress = min(1.0, it / max(1, max_iters))
        base_t = max(0.01, 0.3 * (1.0 - 0.5 * progress))
        for t in [base_t, base_t * 0.5, base_t * 1.5, base_t * 2.0, 0.8 * base_t]:
            a_trial = (1.0 - t) * a + t * g
            a_trial = np.maximum(a_trial, 0.0)
            if a_trial.sum() < 1e-10:
                continue
            a_trial = a_trial / a_trial.sum()
            J_trial = compute_J(a_trial)
            if J_trial < best_J - 1e-14:
                best_J = J_trial
                best_a = a_trial

        if best_a is None:
            stall += 1
            if stall > 30:
                break
            tighten *= 0.7
            continue

        a = best_a
        current_J = best_J
        stall = 0
        tighten = min(0.08, tighten * 1.02)

        if it % 15 == 0:
            tracker_update(a, tracker)

    tracker_update(a, tracker)
    return a


def resample_seq(src, dest_n):
    """Resample 1D sequence to new length."""
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)


def tracker_update(a, tracker):
    """Update tracker with current best."""
    a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
    if a.size == 0 or a.sum() < 0.01:
        return
    score = evaluate_sequence(a.tolist())
    if np.isfinite(score) and score < tracker['best_score']:
        tracker['best'] = a.copy()
        tracker['best_score'] = score
        tracker['best_n'] = a.size
        elapsed = time.monotonic() - tracker['t0']
        print(f"  * NEW BEST: J={score:.8f} n={a.size} t={elapsed:.0f}s", flush=True)


def propose_candidate(seed=42, budget_s=1500):
    """
    Main search function: intensive coordinate descent + LP + multi-length exploration.
    """
    rng = np.random.default_rng(seed)
    t0 = time.monotonic()
    deadline = t0 + float(budget_s) - 20.0
    tracker = {'best': None, 'best_score': float('inf'), 'best_n': 0, 't0': t0}

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        if p.ndim == 1 and p.size > 0 and np.all(np.isfinite(p)):
            parent = np.maximum(p, 0.0)
            tracker_update(parent, tracker)
            print(f"Parent loaded: n={parent.size}, J={tracker['best_score']:.8f}", flush=True)
    except (NameError, AttributeError):
        pass

    if parent is None:
        parent = np.ones(386)
        tracker_update(parent, tracker)
        print(f"No parent, using uniform n=386, J={tracker['best_score']:.8f}", flush=True)

    n_parent = parent.size

    # Phase 1: Intensive coordinate descent on parent
    print(f"\n{'='*50} Phase 1: CD on parent (n={n_parent}) {'='*50}", flush=True)
    t_ph = time.monotonic()
    a = parent.copy()
    a = a / a.sum()
    cd_deadline = min(deadline - 200, t_ph + 300)
    a = cd_multi_pass(a, n_passes=30, tracker=tracker, deadline=cd_deadline)
    print(f"Phase 1 done: J={tracker['best_score']:.8f}, t={time.monotonic()-t_ph:.1f}s", flush=True)

    # Phase 2: LP iterations
    if time.monotonic() < deadline - 150:
        print(f"\n{'='*50} Phase 2: LP iterations {'='*50}", flush=True)
        t_ph = time.monotonic()
        a_lp = tracker['best'].copy()
        a_lp = np.maximum(a_lp, 0.001)
        a_lp = a_lp / a_lp.sum()
        lp_deadline = min(deadline - 100, t_ph + 180)
        a = lp_iterations(a_lp, max_iters=80, tracker=tracker, deadline=lp_deadline, tighten_base=0.004)
        print(f"Phase 2 done: J={tracker['best_score']:.8f}, t={time.monotonic()-t_ph:.1f}s", flush=True)

    # Phase 3: More CD after LP
    if time.monotonic() < deadline - 100:
        print(f"\n{'='*50} Phase 3: CD after LP {'='*50}", flush=True)
        t_ph = time.monotonic()
        a = tracker['best'].copy()
        a = a / a.sum()
        cd_deadline = min(deadline - 80, t_ph + 200)
        a = cd_multi_pass(a, n_passes=40, tracker=tracker, deadline=cd_deadline)
        print(f"Phase 3 done: J={tracker['best_score']:.8f}, t={time.monotonic()-t_ph:.1f}s", flush=True)

    # Phase 4: Alternate LP and CD
    if time.monotonic() < deadline - 80:
        print(f"\n{'='*50} Phase 4: Alternate LP+CD {'='*50}", flush=True)
        t_ph = time.monotonic()
        for round_i in range(5):
            if time.monotonic() > deadline - 80:
                break
            # LP step
            a = tracker['best'].copy()
            a = np.maximum(a, 0.001)
            a = a / a.sum()
            a = lp_iterations(a, max_iters=20, tracker=tracker,
                             deadline=min(time.monotonic() + 40, deadline - 60),
                             tighten_base=0.003 / (1 + 0.3 * round_i))
            if time.monotonic() > deadline - 60:
                break
            # CD step
            a = tracker['best'].copy()
            a = a / a.sum()
            a = cd_multi_pass(a, n_passes=15, tracker=tracker,
                             deadline=min(time.monotonic() + 60, deadline - 50))
        print(f"Phase 4 done: J={tracker['best_score']:.8f}, t={time.monotonic()-t_ph:.1f}s", flush=True)

    # Phase 5: Multi-length exploration
    if time.monotonic() < deadline - 60:
        print(f"\n{'='*50} Phase 5: Multi-length exploration {'='*50}", flush=True)
        t_ph = time.monotonic()
        test_lengths = [n_parent - 4, n_parent - 2, n_parent + 2, n_parent + 4,
                        n_parent + 6, n_parent + 8, n_parent - 6, n_parent + 10]
        test_lengths = [l for l in test_lengths if 100 <= l <= 700]
        seen = set()
        unique_lengths = []
        for l in test_lengths:
            if l not in seen:
                seen.add(l)
                unique_lengths.append(l)

        for nl in unique_lengths:
            if time.monotonic() > deadline - 40:
                break
            a_res = resample_seq(tracker['best'], nl)
            a_res = np.maximum(a_res, 0.001)
            a_res = a_res / a_res.sum()
            a_res = cd_multi_pass(a_res, n_passes=10, tracker=tracker,
                                  deadline=min(time.monotonic() + 40, deadline - 30))
            if time.monotonic() > deadline - 40:
                break
            a_res = lp_iterations(a_res, max_iters=10, tracker=tracker,
                                  deadline=min(time.monotonic() + 30, deadline - 25),
                                  tighten_base=0.003)
            print(f"  Length {nl}: J={tracker['best_score']:.8f}", flush=True)
        print(f"Phase 5 done: J={tracker['best_score']:.8f}, t={time.monotonic()-t_ph:.1f}s", flush=True)

    # Phase 6: Perturbation restarts
    if time.monotonic() < deadline - 30:
        print(f"\n{'='*50} Phase 6: Perturbation restarts {'='*50}", flush=True)
        t_ph = time.monotonic()
        n_best = tracker['best'].size
        for temp in [0.08, 0.05, 0.03, 0.015, 0.01]:
            if time.monotonic() > deadline - 20:
                break
            a_pert = tracker['best'].copy() + rng.normal(0, temp, n_best)
            a_pert = np.maximum(a_pert, 0.001)
            a_pert = a_pert / a_pert.sum()
            a_pert = cd_multi_pass(a_pert, n_passes=15, tracker=tracker,
                                   deadline=min(time.monotonic() + 50, deadline - 15))
            if time.monotonic() > deadline - 20:
                break
            a_pert = lp_iterations(a_pert, max_iters=10, tracker=tracker,
                                   deadline=min(time.monotonic() + 30, deadline - 10),
                                   tighten_base=0.002)
            print(f"  Pert temp={temp}: J={tracker['best_score']:.8f}", flush=True)
        print(f"Phase 6 done: J={tracker['best_score']:.8f}, t={time.monotonic()-t_ph:.1f}s", flush=True)

    # Phase 7: Subgradient + CD polish
    if time.monotonic() < deadline - 15:
        print(f"\n{'='*50} Phase 7: Subgradient + final CD {'='*50}", flush=True)
        t_ph = time.monotonic()
        a = tracker['best'].copy()
        a = a / a.sum()
        a = subgradient_descent(a, n_steps=50, step_size=0.05, tracker=tracker,
                                deadline=min(time.monotonic() + 40, deadline - 10))
        if time.monotonic() < deadline - 5:
            a = cd_multi_pass(a, n_passes=20, tracker=tracker, deadline=deadline - 3)
        print(f"Phase 7 done: J={tracker['best_score']:.8f}, t={time.monotonic()-t_ph:.1f}s", flush=True)

    # Phase 8: Final intensive CD
    if tracker['best'] is not None and time.monotonic() < deadline - 2:
        a_final = tracker['best'].copy()
        a_final = a_final / a_final.sum()
        a_final = cd_multi_pass(a_final, n_passes=10, tracker=tracker, deadline=deadline - 1)

    # Fallback
    if tracker['best'] is None:
        tracker['best'] = np.ones(256)
        tracker['best'] = tracker['best'] / tracker['best'].sum()
        tracker['best_score'] = compute_J(tracker['best'])

    print(f"\n{'='*50} FINAL RESULT {'='*50}", flush=True)
    print(f"J = {tracker['best_score']:.10f}, n = {tracker['best'].size}", flush=True)
    print(f"Total time: {time.monotonic() - t0:.1f}s", flush=True)

    return [float(x) for x in tracker['best']]
