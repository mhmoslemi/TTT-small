import numpy as np

def compute_conv(a):
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    conv = irfft(fa * fa, size)
    return conv[:2 * n - 1]

def compute_obj(a):
    n = len(a)
    if n == 0:
        return np.inf
    conv = compute_conv(a)
    max_b = float(np.max(conv))
    sum_a = float(np.sum(a))
    if sum_a < 0.01:
        return np.inf
    return 2.0 * n * max_b / (sum_a * sum_a)

def fft_convolve(a, b):
    n, m = len(a), len(b)
    size = 1
    while size < n + m - 1:
        size <<= 1
    fa = rfft(a, size)
    fb = rfft(b, size)
    return irfft(fa * fb, size)[:n + m - 1]

def softmax(z):
    z = z - np.max(z)
    e = np.exp(z)
    s = np.sum(e)
    if s < 1e-300:
        return np.ones(len(z)) / len(z)
    return e / s

def adaptive_tau(a, k_top=5):
    """Compute adaptive tau based on the spread of top autocorrelation values."""
    conv = compute_conv(a)
    sorted_vals = np.sort(conv)[::-1]
    top_k = sorted_vals[:min(k_top, len(sorted_vals))]
    # tau proportional to the gap
    gap = top_k[0] - top_k[-1] if len(top_k) > 1 else 0
    n = len(a)
    base_tau = 2.0 / n
    # Scale by the gap relative to the max
    max_val = top_k[0]
    if max_val < 1e-10:
        return base_tau
    ratio = gap / max_val
    tau = base_tau * (1.0 + 5.0 * ratio)
    return tau

    conv = compute_conv(a)

def build_toeplitz(a):
    n = len(a)
    A = np.zeros((2 * n - 1, n), dtype=np.float64)
    for j in range(n):
        k_start = j
        k_end = min(j + n, 2 * n - 1)
        A[k_start:k_end, j] = a[:k_end - k_start]
    return A

def solve_lp_g(a, M, time_limit=15):
    n = len(a)
    A = build_toeplitz(a)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': time_limit})
        if res.status == 0:
            g = np.maximum(res.x, 0)
            s = float(np.sum(g))
            if s > 1e-10:
                return g / s, s
    except Exception:
        pass
    return None, None

def peak_reduction(a, max_iters=200, time_budget=15.0):
    """Targeted peak reduction: identify peak lag and reduce contributing products."""
    n = len(a)
    a = a.copy().astype(np.float64)
    s = np.sum(a)
    if s < 1e-10:
        return a
    a = a / s
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    
    for it in range(max_iters):
        if time.monotonic() - start > time_budget:
            break
        
        conv = compute_conv(a)
        peak_lag = int(np.argmax(conv))
        peak_val = conv[peak_lag]
        
        if peak_lag == 0:
            # Peak at lag 0: try to make the sequence more "flat"
            # This means reducing variance while keeping sum constant
            mean_a = np.sum(a) / n
            a = a * 0.95 + 0.05 * mean_a
            a = np.maximum(a, 0)
            s = np.sum(a)
            if s > 1e-10:
                a = a / s
            continue
        
        # For peak at lag k > 0:
        # conv[k] = sum_i a[i] * a[i+k]
        # Find the pairs with largest contribution
        contributions = np.zeros(n)
        for i in range(n - peak_lag):
            contributions[i] = a[i] * a[i + peak_lag]
            contributions[i + peak_lag] += a[i] * a[i + peak_lag]
        
        # Reduce the largest contributors, increase the smallest
        n_reduce = max(1, n // 20)
        n_increase = max(1, n // 20)
        
        top_indices = np.argsort(contributions)[::-1][:n_reduce]
        bottom_indices = np.argsort(contributions)[:n_increase]
        
        total_reduce = np.sum(a[top_indices]) * 0.02
        a[top_indices] -= total_reduce / n_reduce
        a[bottom_indices] += total_reduce / n_increase
        
        a = np.maximum(a, 0)
        s = np.sum(a)
        if s > 1e-10:
            a = a / s
        
        obj = compute_obj(a)
        if obj < best_obj - 1e-12:
            best_obj = obj
            best_a = a.copy()
    
    return best_a

def subgradient_optimize(a, max_iters=2000, time_budget=20.0):
    """Smoothed subgradient optimization with adaptive parameters."""
    n = len(a)
    a = a.copy().astype(np.float64)
    s = np.sum(a)
    if s < 1e-10:
        a = np.ones(n) / n
        s = 1.0
    else:
        a = a / s
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    step = min(0.5 / np.sqrt(n), 0.05)
    
    for it in range(max_iters):
        if time.monotonic() - start > time_budget:
            break
        
        frac = it / max(1, max_iters)
        tau = (3.0 + 40.0 * frac) / max(1, n // 100)
        tau = max(tau, 0.5 / n)
        
        conv = compute_conv(a)
        max_conv = float(np.max(conv))
        shifted = np.clip(tau * (conv - max_conv) / max_conv, -300, 300) if max_conv > 0 else np.zeros_like(conv)
        weights = np.exp(shifted)
        w_sum = np.sum(weights)
        if w_sum < 1e-300:
            idx = int(np.argmax(conv))
            weights = np.zeros(len(conv))
            weights[idx] = 1.0
        else:
            weights /= w_sum
        
        w_rev = weights[::-1]
        conv_fw = fft_convolve(a, w_rev)
        grad_a = 2.0 * conv_fw[n - 1:2 * n - 1][::-1]
        
        # Normalize gradient
        grad_norm = np.linalg.norm(grad_a)
        if grad_norm > 1e-15:
            grad_a = grad_a / grad_norm
        
        grad_centered = grad_a - np.mean(grad_a)
        a_new = a - step * grad_centered
        a_new = np.maximum(a_new, 0)
        s_new = np.sum(a_new)
        if s_new > 1e-10:
            a_new = a_new / s_new
        else:
            a_new = a.copy()
        
        a = a_new
        obj = compute_obj(a)
        
        if obj < best_obj - 1e-12:
            best_obj = obj
            best_a = a.copy()
        
        # Decay step size
        step *= 0.999
        if step < 1e-8:
            break
    
    return best_a

def spectral_perturbation(a, rng):
    """Perturb in frequency domain to escape local minima."""
    n = len(a)
    a = a.copy()
    
    # Get FFT
    size = 1
    while size < 2 * n:
        size <<= 1
    A = rfft(a, size)
    
    # Perturb the magnitude
    n_pert = rng.integers(3, max(4, n // 10))
    indices = rng.choice(size // 2, n_pert, replace=False)
    indices = np.delete(indices, 0)  # Keep DC
    gains = 1.0 + rng.normal(0, 0.15, n_pert)
    gains = np.maximum(gains, 0.3)
    
    for i, idx in enumerate(indices):
        A[idx] *= gains[i]
    
    a_new = irfft(A, size)[:n]
    a_new = np.maximum(a_new, 0)
    s = np.sum(a_new)
    if s > 1e-10:
        a_new = a_new / s
    else:
        a_new = a.copy()
    
    return a_new

import numpy as np
from scipy.optimize import minimize, linprog
from scipy.fft import rfft, irfft

def compute_conv(a):
    """Compute linear convolution a*a using FFT. Returns array of length 2n-1."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    conv = irfft(fa * fa, size)
    return conv[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    conv = compute_conv(a)
    max_b = float(np.max(conv))
    sum_a = float(np.sum(a))
    if sum_a < 0.01:
        return np.inf
    return 2.0 * n * max_b / (sum_a * sum_a)


def fft_convolve(a, b):
    """Linear convolution of a and b using FFT."""
    n, m = len(a), len(b)
    size = 1
    while size < n + m - 1:
        size <<= 1
    fa = rfft(a, size)
    fb = rfft(b, size)
    return irfft(fa * fb, size)[:n + m - 1]


def softmax_fn(z):
    """Numerically stable softmax."""
    z = z - np.max(z)
    e = np.exp(z)
    s = np.sum(e)
    if s < 1e-300:
        return np.ones(len(z)) / len(z)
    return e / s


def adaptive_tau(a, k_top=5):
    """Compute adaptive tau based on the spread of top autocorrelation values."""
    conv = compute_conv(a)
    sorted_vals = np.sort(conv)[::-1]
    top_k = sorted_vals[:min(k_top, len(sorted_vals))]
    gap = top_k[0] - top_k[-1] if len(top_k) > 1 else 0.0
    n = len(a)
    base_tau = 2.0 / n
    max_val = top_k[0]
    if max_val < 1e-10:
        return base_tau
    ratio = gap / max_val
    tau = base_tau * (1.0 + 5.0 * ratio)
    return max(tau, 0.1 / n)


def soft_max_obj_and_grad_z(z, tau):
    """Compute soft-max of conv(softmax(z), softmax(z)) and gradient w.r.t. z."""
    n = len(z)
    a = softmax_fn(z)
    conv = compute_conv(a)
    cmax = float(np.max(conv))
    shifted = np.clip((conv - cmax) / tau, -500.0, 500.0)
    exp_conv = np.exp(shifted)
    Z = float(np.sum(exp_conv))
    soft_max_val = cmax + tau * np.log(Z)
    w = exp_conv / Z

    w_rev = w[::-1]
    conv_fw = fft_convolve(a, w_rev)
    f1 = conv_fw[n - 1:2 * n - 1][::-1]
    grad_a = 2.0 * f1

    mean_grad = float(np.dot(a, grad_a))
    grad_z = a * (grad_a - mean_grad)
    return soft_max_val, grad_z


def optimize_lbfgs(a_init, tau_list, max_iter=200, time_budget=30.0):
    """Optimize using L-BFGS-B with soft-max parametrization and tau annealing."""
    n = len(a_init)
    a = a_init.copy()
    s = np.sum(a)
    if s < 1e-10:
        a = np.ones(n) / n
    else:
        a = a / s
    a = np.maximum(a, 1e-30)
    z = np.log(a)

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    for tau in tau_list:
        if time.monotonic() - start > time_budget:
            break

        def func_and_grad(z_val):
            return soft_max_obj_and_grad_z(z_val, tau)

        try:
            result = minimize(func_and_grad, z, jac=True, method='L-BFGS-B',
                              options={'maxiter': max_iter, 'ftol': 1e-15, 'gtol': 1e-10})
            z_new = result.x
        except Exception:
            continue

        a_new = softmax_fn(z_new)
        a_new = np.maximum(a_new, 0)
        s_new = np.sum(a_new)
        if s_new < 1e-10:
            continue
        a_new = a_new / s_new
        obj = compute_obj(a_new)

        if obj < best_obj:
            best_obj = obj
            best_a = a_new.copy()
            z = z_new.copy()

    return best_a, best_obj


def build_toeplitz(a):
    """Build Toeplitz matrix A where (A@g)[k] = (conv(a,g))[k]."""
    n = len(a)
    A = np.zeros((2 * n - 1, n), dtype=np.float64)
    for j in range(n):
        k_start = j
        k_end = min(j + n, 2 * n - 1)
        A[k_start:k_end, j] = a[:k_end - k_start]
    return A


def solve_lp_g(a, M, time_limit=15):
    """Solve LP: max sum(g) s.t. (a*g)[k] <= M for all k, g >= 0."""
    n = len(a)
    A = build_toeplitz(a)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': time_limit})
        if res.status == 0:
            g = np.maximum(res.x, 0)
            s = float(np.sum(g))
            if s > 1e-10:
                return g / s, s
    except Exception:
        pass
    return None, None


def solve_lp_g_extended(a, M_base, n_m_values=3, time_limit=10):
    """Try multiple M values around M_base for more diverse LP solutions."""
    n = len(a)
    best_g = None
    best_obj = np.inf

    for mult in [0.95, 1.0, 1.1]:
        M = M_base * mult
        g, _ = solve_lp_g(a, M, time_limit=time_limit)
        if g is not None:
            # Check quality: max cross-correlation with a
            cross = fft_convolve(a, g)
            # cross has length 2n-1, but we want the "autocorrelation-like" part
            # Actually for the LP, the constraint is already satisfied
            if g is not None and best_g is None:
                best_g = g
            elif g is not None:
                # Prefer the one with lower max cross-correlation
                max_cross_new = float(np.max(np.abs(cross[:2*n-1])))
                if best_g is not None:
                    cross_best = fft_convolve(a, best_g)
                    max_cross_best = float(np.max(np.abs(cross_best[:2*n-1])))
                    if max_cross_new < max_cross_best:
                        best_g = g

    return best_g


def alternating_lp_v2(a, max_iters=80, time_budget=100.0):
    """Enhanced alternating LP with adaptive blending, M-range, and spectral perturbation."""
    n = len(a)
    a = a.copy().astype(np.float64)
    s = np.sum(a)
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    stall_count = 0
    patience = 12
    rng = np.random.default_rng(12345)

    for it in range(max_iters):
        if time.monotonic() - start > time_budget:
            break

        conv = compute_conv(a)
        M = float(np.max(conv))

        # Try standard LP
        g, _ = solve_lp_g(a, M, time_limit=8)
        if g is None:
            # Try with relaxed M
            g, _ = solve_lp_g(a, M * 1.05, time_limit=8)
            if g is None:
                break

        # Adaptive blending: try multiple t values
        best_t_obj = float('inf')
        best_a_new = None
        for t in [0.05, 0.1, 0.15, 0.2, 0.3, 0.4, 0.5, 0.7, 1.0]:
            a_try = (1.0 - t) * a + t * g
            a_try = np.maximum(a_try, 0)
            s_try = np.sum(a_try)
            if s_try < 1e-10:
                continue
            a_try = a_try / s_try
            obj_try = compute_obj(a_try)
            if obj_try < best_t_obj:
                best_t_obj = obj_try
                best_a_new = a_try.copy()

        if best_a_new is None:
            break

        if best_t_obj < best_obj - 1e-12:
            best_obj = best_t_obj
            best_a = best_a_new.copy()
            a = best_a_new
            stall_count = 0
        else:
            stall_count += 1
            if stall_count > patience:
                # Spectral perturbation
                a_pert = spectral_perturbation(a, rng)
                obj_pert = compute_obj(a_pert)
                if obj_pert < best_obj:
                    a = a_pert
                else:
                    # Cosine perturbation
                    k = np.arange(n)
                    freq = rng.uniform(0.3, 4.0)
                    phase = rng.uniform(0, 2 * np.pi)
                    pert = 1.0 + 0.2 * np.cos(2 * np.pi * freq * k / n + phase)
                    a_new = a * pert
                    a_new = np.maximum(a_new, 0)
                    s_new = np.sum(a_new)
                    if s_new > 1e-10:
                        a = a_new / s_new
                    # Also try random noise
                    a_noise = a * np.exp(rng.normal(0, 0.1, n))
                    a_noise = np.maximum(a_noise, 0)
                    s_n = np.sum(a_noise)
                    if s_n > 1e-10:
                        a_noise = a_noise / s_n
                        if compute_obj(a_noise) < compute_obj(a):
                            a = a_noise
                stall_count = 0
                patience = max(8, patience - 2)
            else:
                # Small step toward g
                a = 0.9 * a + 0.1 * g
                a = np.maximum(a, 0)
                s = np.sum(a)
                if s > 1e-10:
                    a = a / s

    return best_a


def spectral_perturbation(a, rng):
    """Perturb in frequency domain to escape local minima."""
    n = len(a)
    size = 1
    while size < 2 * n:
        size <<= 1
    A = rfft(a, size)

    n_pert = rng.integers(5, max(6, n // 8))
    indices = rng.choice(size // 2 - 1, n_pert, replace=False) + 1
    gains = 1.0 + rng.normal(0, 0.2, n_pert)
    gains = np.maximum(gains, 0.3)

    for i, idx in enumerate(indices):
        A[idx] *= gains[i]

    a_new = irfft(A, size)[:n]
    a_new = np.maximum(a_new, 0)
    s = np.sum(a_new)
    if s > 1e-10:
        a_new = a_new / s
    else:
        a_new = a.copy()
    return a_new


def peak_reduction(a, max_iters=300, time_budget=15.0):
    """Targeted peak reduction: identify peak lag and reduce contributing products."""
    n = len(a)
    a = a.copy().astype(np.float64)
    s = np.sum(a)
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    for it in range(max_iters):
        if time.monotonic() - start > time_budget:
            break

        conv = compute_conv(a)
        peak_lag = int(np.argmax(conv))
        peak_val = conv[peak_lag]

        if peak_lag == 0:
            # Peak at lag 0: reduce variance (make more flat)
            mean_a = np.sum(a) / n
            a = a * 0.97 + 0.03 * mean_a
            a = np.maximum(a, 0)
            s = np.sum(a)
            if s > 1e-10:
                a = a / s
            obj = compute_obj(a)
            if obj < best_obj - 1e-12:
                best_obj = obj
                best_a = a.copy()
            continue

        # Compute contributions to the peak
        contributions = np.zeros(n)
        for i in range(n - peak_lag):
            prod = a[i] * a[i + peak_lag]
            contributions[i] += prod
            contributions[i + peak_lag] += prod

        n_reduce = max(1, n // 15)
        n_increase = max(1, n // 15)

        top_indices = np.argsort(contributions)[::-1][:n_reduce]
        bottom_indices = np.argsort(contributions)[:n_increase]

        # Reduce high contributors, increase low ones
        reduce_amount = 0.015
        for idx in top_indices:
            a[idx] -= reduce_amount * a[idx]
        total_added = 0.0
        for idx in bottom_indices:
            add = reduce_amount * a[top_indices[0]]
            a[idx] += add
            total_added += add

        a = np.maximum(a, 0)
        s = np.sum(a)
        if s > 1e-10:
            a = a / s

        obj = compute_obj(a)
        if obj < best_obj - 1e-12:
            best_obj = obj
            best_a = a.copy()

    return best_a


def mirror_descent(a, max_iters=5000, time_budget=20.0):
    """Mirror descent on the probability simplex with adaptive step size."""
    n = len(a)
    a = a.copy().astype(np.float64)
    s = np.sum(a)
    if s < 1e-10:
        a = np.ones(n) / n
    else:
        a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    eta = 1.0 / np.sqrt(n)

    for it in range(max_iters):
        if time.monotonic() - start > time_budget:
            break

        # Compute smoothed max gradient
        tau = max(1.0, 20.0 / max(1, int((it + 1) / 100))) / n
        conv = compute_conv(a)
        cmax = float(np.max(conv))
        shifted = np.clip((conv - cmax) / tau, -300.0, 300.0)
        exp_conv = np.exp(shifted)
        w = exp_conv / np.sum(exp_conv)

        # Gradient of the smoothed max at a
        w_rev = w[::-1]
        conv_fw = fft_convolve(a, w_rev)
        grad = 2.0 * conv_fw[n - 1:2 * n - 1][::-1]

        # Mirror descent update: a <- exp(log(a) - eta * grad) / Z
        # This is equivalent to: a <- a * exp(-eta * grad) / sum(a * exp(-eta * grad))
        log_a = np.log(np.maximum(a, 1e-30))
        new_log_a = log_a - eta * grad
        new_log_a = new_log_a - np.max(new_log_a)
        a_new = np.exp(new_log_a)
        a_new = a_new / np.sum(a_new)

        a = a_new
        obj = compute_obj(a)
        if obj < best_obj - 1e-12:
            best_obj = obj
            best_a = a.copy()

        # Decay eta
        eta *= 0.9995
        if eta < 1e-10:
            break

    return best_a


def subgradient_optimize(a, max_iters=3000, time_budget=20.0):
    """Smoothed subgradient optimization with adaptive parameters."""
    n = len(a)
    a = a.copy().astype(np.float64)
    s = np.sum(a)
    if s < 1e-10:
        a = np.ones(n) / n
    else:
        a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    step = min(0.3 / np.sqrt(n), 0.03)

    for it in range(max_iters):
        if time.monotonic() - start > time_budget:
            break

        frac = it / max(1, max_iters)
        tau = max(0.5 / n, (2.0 + 50.0 * frac) / max(1, n // 50))

        conv = compute_conv(a)
        max_conv = float(np.max(conv))
        if max_conv > 0:
            shifted = np.clip(tau * (conv - max_conv) / max_conv, -300, 300)
        else:
            shifted = np.zeros_like(conv)
        weights = np.exp(shifted)
        w_sum = np.sum(weights)
        if w_sum < 1e-300:
            idx = int(np.argmax(conv))
            weights = np.zeros(len(conv))
            weights[idx] = 1.0
        else:
            weights /= w_sum

        w_rev = weights[::-1]
        conv_fw = fft_convolve(a, w_rev)
        grad_a = 2.0 * conv_fw[n - 1:2 * n - 1][::-1]

        grad_norm = np.linalg.norm(grad_a)
        if grad_norm > 1e-15:
            grad_a = grad_a / grad_norm

        grad_centered = grad_a - np.mean(grad_a)
        a_new = a - step * grad_centered
        a_new = np.maximum(a_new, 0)
        s_new = np.sum(a_new)
        if s_new > 1e-10:
            a_new = a_new / s_new
        else:
            a_new = a.copy()

        a = a_new
        obj = compute_obj(a)
        if obj < best_obj - 1e-12:
            best_obj = obj
            best_a = a.copy()

        step *= 0.999
        if step < 1e-9:
            break

    return best_a


def resample_sequence(a, new_n):
    """Resample sequence to new length via interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return a_new

        remaining = deadline - time.monotonic()
        if remaining > 20:
            a6 = alternating_lp_v2(best_seq, max_iters=30, time_budget=min(remaining * 0.15, 20.0))
            obj6 = compute_obj(a6)
            if obj6 < best_score:
                best_score = obj6
                best_seq = a6.copy()
                print(f"Phase 6 LP r{round_idx}: {best_score:.8f}")

        # Mirror descent (very small step)
        remaining = deadline - time.monotonic()
        if remaining > 15:
            a6 = mirror_descent(best_seq, max_iters=int(remaining * 8),
                                 time_budget=min(remaining * 0.1, 10.0))
            obj6 = compute_obj(a6)
            if obj6 < best_score:
                best_score = obj6
                best_seq = a6.copy()
                print(f"Phase 6 mirror r{round_idx}: {best_score:.8f}")

    # === Phase 7: Nearby Lengths (if time permits) ===
    remaining = deadline - time.monotonic()
    if remaining > 150:
        print(f"\n{'='*50}")
        print(f"Phase 7: Nearby Lengths ({remaining:.0f}s available)")
        print(f"{'='*50}")

        for n_try in [n_best - 50, n_best + 50, n_best + 100, n_best - 100]:
            if time.monotonic() > deadline - 30:
                break
            if n_try < 100:
                continue
            remaining = deadline - time.monotonic()
            if remaining < 50:
                break

            a_resample = resample_sequence(best_seq, n_try)

            # Quick L-BFGS-B
            tau_t = [3.0 / n_try, 1.0 / n_try, 0.3 / n_try]
            a_try, obj_try = optimize_lbfgs(a_resample, tau_t, max_iter=200,
                                            time_budget=min(remaining * 0.2, 25.0))

            # LP
            remaining = deadline - time.monotonic()
            if remaining > 30:
                a_try = alternating_lp_v2(a_try, max_iters=30, time_budget=min(remaining * 0.2, 25.0))
                obj_try = compute_obj(a_try)

            if obj_try < best_score:
                best_score = obj_try
                best_seq = a_try.copy()
                n_best = n_try
                print(f"  n={n_try}: {best_score:.8f} ** IMPROVED **")
            else:
                print(f"  n={n_try}: {obj_try:.8f} (no improvement)")

    # === Final Validation ===
    if best_seq is None:
        best_seq = np.ones(200)

    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    val = evaluate_sequence(result)
    if val < np.inf:
        final_score = val
    else:
        result = [1.0] * 100
        final_score = 2.0

    elapsed = time.monotonic() - start_time
    print(f"\n{'='*50}")
    print(f"FINAL: n={len(result)}, score={final_score:.8f}, time={elapsed:.1f}s")
    print(f"{'='*50}")

    return result
