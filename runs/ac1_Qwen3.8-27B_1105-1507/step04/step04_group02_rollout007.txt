import numpy as np

def compute_conv(a):
    """Compute linear convolution a*a via FFT."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    conv = np.fft.irfft(fa * fa, size)
    return conv[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    S = float(np.sum(a))
    if S < 1e-10:
        return np.inf
    conv = compute_conv(a)
    M = float(np.max(conv))
    return 2.0 * n * M / (S * S)


def softmax_smooth_obj(a, tau):
    """Compute smooth approximation of the objective using soft-max."""
    n = len(a)
    S = float(np.sum(a))
    if S < 1e-10:
        return 1e10, np.zeros(n)
    conv = compute_conv(a)
    conv_max = np.max(conv)
    shifted = (conv - conv_max) / tau
    # Numerically stable log-sum-exp
    max_shifted = np.max(shifted)
    log_sum_exp = max_shifted + np.log(np.sum(np.exp(shifted - max_shifted)))
    soft_max = conv_max + tau * log_sum_exp
    obj = 2.0 * n * soft_max / (S * S)
    return obj, conv


def softmax_smooth_grad(a, tau):
    """Compute gradient of smooth soft-max objective using FFT cross-correlation."""
    n = len(a)
    S = float(np.sum(a))
    if S < 1e-10:
        return 1e10, np.zeros(n)
    
    conv = compute_conv(a)
    conv_max = np.max(conv)
    
    # Stable softmax weights
    shifted = (conv - conv_max) / tau
    max_shifted = np.max(shifted)
    exp_shifted = np.exp(shifted - max_shifted)
    Z = np.sum(exp_shifted)
    w = exp_shifted / Z  # softmax weights, sum to 1
    
    soft_max = conv_max + tau * (max_shifted + np.log(Z))
    obj = 2.0 * n * soft_max / (S * S)
    
    # Cross-correlation: c[i] = sum_k w[k] * a[k-i] (valid indices only)
    # Using FFT: ifft(FFT(w, L) * conj(FFT(a, L)))[:n]
    L = 1
    while L < (2 * n - 1) + n - 1:
        L <<= 1
    W = np.fft.fft(w, L)
    A = np.fft.fft(a, L)
    cross_full = np.fft.ifft(W * np.conj(A))
    cross = cross_full[:n].real
    
    # Gradient: 4n/S^2 * cross[i] - 4n*soft_max/S^3
    grad = (4.0 * n / (S * S)) * cross - (4.0 * n * soft_max) / (S ** 3)
    
    return obj, grad


def l_bfgs_b_optimize(a, tau_list, max_time=60.0):
    """Optimize using L-BFGS-B with multiple temperature levels."""
    n = len(a)
    a = a.copy().astype(np.float64)
    S = np.sum(a)

            # Renormalize to prevent scale drift

def build_toeplitz_sparse(a):
    """Build sparse Toeplitz matrix for conv(a, g)."""
    n = len(a)
    rows = []
    cols = []
    vals = []
    for j in range(n):
        start = j
        end = min(j + n, 2 * n - 1)
        length = end - start
        rows.extend(range(start, end))
        cols.extend([j] * length)
        vals.extend(list(a[:length]))
    A = coo_matrix((vals, (rows, cols)), shape=(2 * n - 1, n)).tocsr()
    return A


def solve_lp_maxsum(a, M, time_limit=5):
    """Solve LP: max sum(g) s.t. conv(a,g) <= M, g >= 0."""
    n = len(a)
    A = build_toeplitz_sparse(a)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': time_limit})
        if res.status == 0:
            return np.maximum(res.x, 0)
    except Exception:
        pass
    return None


def lp_polish(a, max_iters=20, time_budget=30.0):

    n = len(a)
    a = a.copy().astype(np.float64)

    if S < 1e-10:
        return a
    a = a / S
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    stall = 0
    
    for it in range(max_iters):
        if time.monotonic() - start > time_budget:

        conv = compute_conv(a)
        M = float(np.max(conv))
        
        g = solve_lp_maxsum(a, M, time_limit=4)
        if g is None:
            break
        
        S_g = np.sum(g)
        if S_g < 1e-10:
            break
        g = g / S_g
        
        # Line search
        best_t = 0.0
        best_obj_ls = best_obj
        for t in [0.1, 0.2, 0.3, 0.5, 0.7, 0.9, 1.0]:
            a_try = (1.0 - t) * a + t * g
            a_try = np.maximum(a_try, 0)
            S_try = np.sum(a_try)
            if S_try < 1e-10:
                continue
            a_try = a_try / S_try
            obj_try = compute_obj(a_try)
            if obj_try < best_obj_ls:
                best_obj_ls = obj_try
                best_t = t

            a = (1.0 - best_t) * a + best_t * g
            a = np.maximum(a, 0)
            a = a / np.sum(a)
            best_obj = best_obj_ls
            best_a = a.copy()
            stall = 0
        else:
            stall += 1
            if stall > 8:
                break
    
    return best_a


def cosine_init(n, n_modes=6, seed=0):
    """Cosine series initialization."""
    rng = np.random.default_rng(seed)
    k = np.arange(n)
    a = np.ones(n)
    for m in range(1, n_modes + 1):
        c = rng.uniform(-0.5, 0.5) / m
        phase = rng.uniform(0, 2 * np.pi)
        a += c * np.cos(2 * np.pi * m * k / n + phase)
    a = np.maximum(a, 0)
    S = np.sum(a)
    if S < 1e-10:
        return np.ones(n) / n
    return a / S


def raised_cosine_init(n, alpha=1.5):
    """Raised cosine initialization."""
    k = np.arange(n)
    x = 2.0 * k / (n - 1) - 1.0
    a = 0.5 * (1.0 + np.cos(np.pi * x)) ** alpha
    a = np.maximum(a, 1e-10)
    S = np.sum(a)
    return a / S


def gaussian_init(n, sigma_frac=0.3, center_frac=0.5):
    """Gaussian bump initialization."""
    k = np.arange(n)
    center = center_frac * n
    sigma = sigma_frac * n
    a = np.exp(-0.5 * ((k - center) / sigma) ** 2)
    S = np.sum(a)
    return a / S


def tent_init(n):
    """Triangular (tent) initialization."""
    k = np.arange(n)
    mid = (n - 1) / 2.0
    a = 1.0 - np.abs(k - mid) / (mid + 1)
    a = np.maximum(a, 0)
    S = np.sum(a)
    return a / S


def random_smooth_init(n, seed=0, smoothness=3):
    """Random smooth initialization via low-frequency noise."""
    rng = np.random.default_rng(seed)
    freqs = np.fft.rfftfreq(n)
    mask = freqs < 1.0 / smoothness
    spectrum = rng.normal(0, 1, n // 2 + 1)
    spectrum *= mask
    a = np.fft.irfft(spectrum * np.sqrt(n), n)
    a = np.abs(a) + 0.1
    S = np.sum(a)
    return a / S

            a_quick = l_bfgs_b_optimize(a_init, [tau, tau * 0.5], max_time=3.0)
            obj_quick = compute_obj(a_quick)
            best_local = a_quick if obj_quick < obj_init else a_init
            best_local_obj = min(obj_quick, obj_init)
            candidate_pool.append((best_local, best_local_obj, name, n))
            
            if best_local_obj < best_score:
                best_score = best_local_obj
                best_seq = best_local.copy()
                print(f"  {name} (n={n}): {best_local_obj:.6f} ** NEW BEST **")
    
    # Sort by objective
    candidate_pool.sort(key=lambda x: x[1])
    print(f"\n  Top 5 candidates after screening:")
    for i, (a_c, o_c, nm, nc) in enumerate(candidate_pool[:5]):
        print(f"    {i}: {nm} obj={o_c:.6f}")
    
    # === Phase 2: Refine top candidates ===
    print("\n=== Phase 2: Refine top candidates ===")
    
    n_refine = min(4, len(candidate_pool))
    for idx in range(n_refine):
        if time.monotonic() > deadline - 60:
            break
        remaining = deadline - time.monotonic()
        if remaining < 40:
            break
        
        a_cand, obj_cand, name, n_cand = candidate_pool[idx]
        if obj_cand > best_score + 0.02:
            break
        
        alloc = min(remaining * 0.3, 80.0)
        print(f"  Refining {name} (n={n_cand}, obj={obj_cand:.6f}), alloc={alloc:.0f}s")
        
        a_ref, obj_ref = full_refine(a_cand, total_time=alloc)
        if obj_ref < best_score:
            best_score = obj_ref
            best_seq = a_ref.copy()
            print(f"    ** NEW BEST: n={n_cand}, obj={best_score:.6f} **")
        else:
            print(f"    After refine: {obj_ref:.6f} (best stays {best_score:.6f})")
    
    # === Phase 3: Intensive refinement with perturbation ===
    print("\n=== Phase 3: Intensive refinement ===")
    
    a_cur = best_seq.copy()
    n_best = len(a_cur)
    
    for round_num in range(20):
        if time.monotonic() > deadline - 40:
            break
        remaining = deadline - time.monotonic()
        if remaining < 25:
            break
        
        alloc = min(remaining * 0.25, 50.0)
        a_new, obj_new = full_refine(a_cur, total_time=alloc)
        
        if obj_new < best_score:
            best_score = obj_new
            best_seq = a_new.copy()
            a_cur = a_new.copy()
            print(f"  R{round_num}: {best_score:.6f} ** NEW BEST **")
        else:
            print(f"  R{round_num}: {obj_new:.6f} (stays {best_score:.6f})")
            
            # Perturbation escape
            if remaining > 30:
                a_pert, obj_pert = perturb_and_refine(
                    a_cur, n_perturbs=3, time_budget=min(remaining * 0.3, 30.0),
                    seed=round_num * 1000 + seed)
                if obj_pert < best_score:
                    best_score = obj_pert
                    best_seq = a_pert.copy()
                    a_cur = a_pert.copy()
                    print(f"  R{round_num} perturb: {best_score:.6f} ** NEW BEST **")
    
    # === Phase 4: Multi-n interpolation ===
    print("\n=== Phase 4: Multi-n interpolation ===")
    
    if time.monotonic() < deadline - 30:
        n_base = len(best_seq)
        n_tries = sorted(set([
            max(100, n_base - 100),
            max(100, n_base - 50),
            n_base + 50,
            n_base + 100,
            n_base + 200,
        ]))
        n_tries = [n for n in n_tries if 100 <= n <= 2000 and n != n_base]
        
        for n_try in n_tries:
            if time.monotonic() > deadline - 20:
                break
            remaining = deadline - time.monotonic()
            if remaining < 20:
                break
            
            # Interpolate best sequence to new length
            x_old = np.linspace(0, 1, n_base)
            x_new = np.linspace(0, 1, n_try)
            try:
                f_interp = interp1d(x_old, best_seq, kind='cubic', fill_value='extrapolate')
                a_interp = np.maximum(f_interp(x_new), 0)
            except Exception:
                a_interp = raised_cosine_init(n_try, alpha=2.0)
            
            S_i = np.sum(a_interp)
            if S_i > 1e-10:
                a_interp = a_interp / S_i
            
            alloc = min(remaining * 0.4, 40.0)
            a_new, obj_new = full_refine(a_interp, total_time=alloc)
            if obj_new < best_score:
                best_score = obj_new
                best_seq = a_new.copy()
                print(f"  n={n_try}: {best_score:.6f} ** NEW BEST **")
            else:
                print(f"  n={n_try}: {obj_new:.6f}")
    
    # === Phase 5: Stochastic hill-climbing polish ===
    print("\n=== Phase 5: Stochastic polish ===")
    
    a_cur = best_seq.copy()
    n_final = len(a_cur)
    
    if time.monotonic() < deadline - 15:
        remaining = deadline - time.monotonic()
        rng = np.random.default_rng(seed + 77777)
        T = 0.01  # Temperature for simulated annealing
        current_obj = compute_obj(a_cur)
        
        max_iters = int(remaining * 8)
        for it in range(max_iters):
            if time.monotonic() > deadline - 10:
                break
            
            # Small random perturbation
            scale = T * 2
            a_try = a_cur * np.exp(rng.normal(0, scale, n_final))
            a_try = np.maximum(a_try, 0)
            S_try = np.sum(a_try)
            if S_try < 1e-10:
                continue
            a_try = a_try / S_try
            
            obj_try = compute_obj(a_try)
            delta = obj_try - current_obj
            
            if delta < 0 or rng.random() < np.exp(-delta / T):
                a_cur = a_try
                current_obj = obj_try
                if current_obj < best_score:
                    best_score = current_obj
                    best_seq = a_cur.copy()
                    print(f"  SA iter {it}: {best_score:.6f} ** NEW BEST **")
            
            # Cool down
            T *= 0.999
            if T < 0.001:
                break
    
    # === Phase 6: Final gradient polish ===
    print("\n=== Phase 6: Final polish ===")
    
    if time.monotonic() < deadline - 8:
        remaining = deadline - time.monotonic()
        a_cur = best_seq.copy()
        n_final = len(a_cur)
        
        conv = compute_conv(a_cur)
        M = np.max(conv)
        tau_list = [M * 0.02, M * 0.01, M * 0.005, M * 0.002]
        a_new = l_bfgs_b_optimize(a_cur, tau_list, max_time=min(remaining * 0.5, 20.0))
        obj_new = compute_obj(a_new)
        if obj_new < best_score:
            best_score = obj_new
            best_seq = a_new.copy()
            print(f"  Final grad: {best_score:.6f} ** NEW BEST **")
        
        # LP if small enough
        remaining = deadline - time.monotonic()
        if remaining > 10 and len(best_seq) <= 600:
            a_lp = lp_polish(best_seq, max_iters=10, time_budget=min(remaining * 0.5, 20.0))
            obj_lp = compute_obj(a_lp)
            if obj_lp < best_score:
                best_score = obj_lp
                best_seq = a_lp.copy()
                print(f"  Final LP: {best_score:.6f} ** NEW BEST **")
    
    # Final validation
    if best_seq is None:
        best_seq = np.ones(300) / 300
        best_score = compute_obj(best_seq)
    
    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    val = evaluate_sequence(result)
    if val < np.inf and val < best_score:
        best_score = val
    elif val == np.inf:
        result = [1.0] * 300
        best_score = 2.0
    
    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_score:.6f}, time={elapsed:.1f}s ===")
    
    return result
