R[m] = sum_k w[k] * a[k-m]  (for real)

In terms of convolution: R = ifft(fft(w, N) * conj(fft(a, N))) where N >= L + n - 1.

But the indexing is: R[m] = sum_k w[k] * a[k-m].

If we compute C = ifft(fft(w, N) * conj(fft(a, N))), then:
C[k] = sum_j w[j] * a[(k-j) mod N]

For k in [0, N-1], this gives:
C[k] = sum_j w[j] * a[k-j] (when 0 <= k-j < n)
     + sum_j w[j] * a[k-j+N] (when k-j < 0, wrapping around)

To avoid wrap-around, we need N >= L + n - 1. Then for k in [0, L+n-2]:
C[k] = sum_j w[j] * a[k-j] (with a being zero-padded)

So R[m] = C[m] for m in [0, L-1] (approximately, with some edge effects).

Actually, R[m] = sum_k w[k] * a[k-m]. Let's substitute j = k - m:
R[m] = sum_j w[j+m] * a[j]

This is the cross-correlation. In convolution terms:
R[m] = (w_rev * a)[m + L - 1] where w_rev is the reversed w.

Hmm, this is getting confusing. Let me just use the direct O(n^2) computation since it's fast enough.

from scipy.optimize import linprog
import time

def fast_conv(a):
    n = len(a)
    size = 1
    while size < 2*n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2*n-1]

def compute_J(a):
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))

def cross_corr_1d(w, a):
    """Compute (w cross_corr a)[m] = sum_k w[k] * a[k-m] for m=0..n-1."""
    n = len(a)
    L = len(w)
    result = np.zeros(n)
    for m in range(n):
        end = min(L, m + n)
        if end > m:
            result[m] = np.dot(w[m:end], a[:end-m])
    return result

def mw_step(a, eta, beta):
    """One multiplicative weight / exponential gradient step."""
    n = len(a)
    c = fast_conv(a)
    M = c.max()
    
    # Soft weights
    w = np.exp(beta * (c - M))
    w_sum = w.sum()
    if w_sum < 1e-30:
        w = np.ones_like(c) / len(c)
    else:
        w /= w_sum
    
    # Subgradient: 2 * cross_corr(w, a)
    grad = 2.0 * cross_corr_1d(w, a)
    grad = np.maximum(grad, 0.0)
    
    # Multiplicative update
    a_new = a * np.exp(-eta * grad)
    a_new = np.maximum(a_new, 1e-10)
    a_new /= a_new.sum()
    
    return a_new

def lp_solve(a, M):
    """Solve LP: max sum(g) s.t. (a*g)[k] <= M for all k, g >= 0."""
    n = len(a)
    # Build dense constraint matrix
    A = np.zeros((2*n-1, n))
    for k in range(2*n-1):
        js = max(0, k - n + 1)
        je = min(n-1, k)
        if js <= je:
            A[k, js:je+1] = a[k-je:k-js+1][::-1]
    
    c_obj = -np.ones(n)
    b_ub = np.full(2*n-1, M)
    bounds = [(0.0, None)] * n
    
    r = linprog(c_obj, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
    if r.success and r.x is not None:
        return np.maximum(r.x, 0.0)
    return None

def coordinate_descent_pass(a, max_iters=200):
    """One pass of coordinate descent with golden section search."""
    a = np.maximum(a.astype(np.float64), 0.0)
    S = a.sum()
    if S < 1e-10:
        return a
    n = len(a)
    b = fast_conv(a)
    
    indices = list(range(n))
    np.random.shuffle(indices)
    
    for i in indices[:max_iters]:
        v = a[i]
        if v < 1e-10:
            # Try setting to a small value
            candidates = [0.0, 0.01 * S / n, 0.1 * S / n]
        else:
            candidates = [0.0, v * 0.5, v, v * 1.5, v * 2.0, S / n]
        
        best_val = v
        best_J = 2.0 * n * b.max() / (S * S)
        
        for x in candidates:
            if x == v:
                continue
            # Compute new convolution
            b_new = b.copy()
            for k in range(max(0, i), min(2*n-1, i+n)):
                ki = k - i
                if 0 <= ki < n:
                    b_new[k] = b[k] - 2.0 * v * a[ki] + 2.0 * x * a[ki]
            # Diagonal correction
            if 2*i < 2*n-1:
                b_new[2*i] = b[2*i] - v*v + x*x
            
            S_new = S - v + x
            if S_new < 1e-10:
                continue
            J_new = 2.0 * n * b_new.max() / (S_new * S_new)
            if J_new < best_J:
                best_J = J_new
                best_val = x
        
        if abs(best_val - v) > 1e-10:
            # Apply update
            for k in range(max(0, i), min(2*n-1, i+n)):
                ki = k - i
                if 0 <= ki < n:
                    b[k] = b[k] - 2.0 * v * a[ki] + 2.0 * best_val * a[ki]

                b[2*i] = b[2*i] - v*v + best_val*best_val
            S = S - v + best_val
            a[i] = best_val
    
    return a
