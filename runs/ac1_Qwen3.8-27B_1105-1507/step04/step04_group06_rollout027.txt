```python
"""
Minimizes J(a) = 2*n*max(convolve(a,a))/(sum(a))² using:
1. Mirror descent with negative entropy (fast, O(n log n) per step)
2. Beta annealing for log-sum-exp smoothing of the max
3. Multi-length exploration (1024, 1536, 2048)
4. Multi-restart with diverse perturbations
5. Final L-BFGS polish at high beta
"""

import numpy as np
from scipy.optimize import minimize
import time

def fast_conv(a):
    n = len(a)
    s = 1
    while s < 2 * n - 1:
        s <<= 1
    fa = np.fft.rfft(a, s)
    return np.fft.irfft(fa * fa, s)[:2 * n - 1]

def compute_J(a):
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))

def smooth_obj_and_grad(a, beta):
    """Compute smooth approximation of max(conv(a,a)) and gradient w.r.t. a."""
    n = len(a)
    b = fast_conv(a)
    M = b.max()
    shifted = beta * (b - M)
    shifted = np.clip(shifted, -700, 700)
    exp_shifted = np.exp(shifted)
    Z = exp_shifted.sum()
    p = exp_shifted / Z
    f_val = M + np.log(Z) / beta
    # grad_j = 2 * sum_i a_i * p_{i+j}
    # = 2 * convolve(a[::-1], p, mode='full')[n-1 : 2n-1]
    conv = np.convolve(a[::-1], p, mode='full')
    grad = 2.0 * conv[n-1:2*n-1]
    return f_val, grad, p

def mirror_descent(a, beta, n_iters, lr_init, deadline, tracker=None):
    """Run mirror descent for n_iters steps at given beta."""
    a = np.maximum(a.astype(np.float64), 1e-10)
    s = a.sum()
    a = a / s * n  # normalize sum to n
    n = len(a)
    
    f_old, grad_old, _ = smooth_obj_and_grad(a, beta)
    lr = lr_init
    
    for it in range(n_iters):
        if time.monotonic() > deadline:
            break
        
        # Mirror descent update
        # a_new_j ∝ a_j * exp(-lr * grad_j)
        log_ratio = -lr * grad_old
        # Subtract max for numerical stability
        log_ratio -= log_ratio.max()
        weights = np.exp(log_ratio)
        a_new = a * weights
        a_new_sum = a_new.sum()
        if a_new_sum < 1e-10:
            break
        a_new = a_new / a_new_sum * n  # renormalize
        
        # Evaluate
        f_new, grad_new, _ = smooth_obj_and_grad(a_new, beta)
        
        if f_new < f_old:
            a = a_new
            grad_old = grad_new
            f_old = f_new
        else:
            # Reduce learning rate
            lr *= 0.5
    
    return a, f_old

def F_and_grad_full(a, beta):
    """Full objective F(a) = 2n*smooth_max/(sum)^2 and gradient."""
    n = len(a)
    s = a.sum()
    if s < 1e-10:
        return 1e10, np.zeros(n)
    f_val, grad_g, _ = smooth_obj_and_grad(a, beta)
    F = 2.0 * n * f_val / (s * s)
    grad_F = (2.0 * n * (s * grad_g - 2.0 * f_val * np.ones(n))) / (s ** 3)
    return F, grad_F

def lbfgs_smooth(a0, beta, max_iters, deadline):
    """L-BFGS-B on the smoothed full objective."""
    a = np.maximum(a0.astype(np.float64), 1e-10)
    n = len(a)
    
    def func(x):
        x2 = np.maximum(x, 1e-10)
        f, _ = F_and_grad_full(x2, beta)
        return f if np.isfinite(f) else 1e10
    
    def jac(x):
        x2 = np.maximum(x, 1e-10)
        _, g = F_and_grad_full(x2, beta)
        return np.nan_to_num(g, nan=0.0, posinf=0.0, neginf=0.0)
    
    bounds = [(0.0, None)] * n
    try:
        res = minimize(func, a.copy(), jac=jac, method='L-BFGS-B', bounds=bounds,

def resample_seq(src, dest_n):
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    result = np.interp(t_new, t_old, src)
    return np.maximum(result, 0.0)
```

```python
def _next_pow2(m):
    s = 1
    while s < m:
        s <<= 1
    return s


def fast_conv(a):
    """Full self-convolution via FFT, O(n log n)."""
    n = len(a)
    s = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, s)
    return np.fft.irfft(fa * fa, s)[:2 * n - 1]

    """Compute J(a) = 2*n*max(conv(a,a))/(sum(a))^2."""
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))

        return 1e10, np.zeros(n)
    f_val, grad_f, _ = smooth_obj_grad(a, beta)
    F = 2.0 * n * f_val / (s * s)
    grad_F = (2.0 * n / (s ** 3)) * (s * grad_f - 2.0 * f_val)
    return F, grad_F


def mirror_descent(a, beta, max_iters, lr_init, deadline):
    """
    Entropy mirror descent for minimizing smooth objective f(a)
    under constraints a >= 0, sum(a) = n.
    Update: a_j_new ∝ a_j * exp(-lr * grad_j), then renormalize.
    Returns (a, f_value).
    """
    n = len(a)
    a = np.maximum(a.astype(np.float64), 1e-10)
    a = a / a.sum() * n

    f_old, grad_old, _ = smooth_obj_grad(a, beta)
    lr = lr_init
    best_a = a.copy()
    best_f = f_old
    no_improve = 0

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        # Mirror descent update
        log_ratio = -lr * grad_old
        log_ratio -= log_ratio.max()
        weights = np.exp(log_ratio)
        a_new = a * weights
        a_new_sum = a_new.sum()
        if a_new_sum < 1e-10:
            break
        a_new = a_new / a_new_sum * n

        f_new, grad_new, _ = smooth_obj_grad(a_new, beta)

        if f_new < f_old - 1e-15:
            a = a_new
            grad_old = grad_new
            f_old = f_new
            no_improve = 0
            if f_old < best_f:
                best_f = f_old
                best_a = a.copy()
        else:
            lr *= 0.5
            no_improve += 1
            if no_improve > 50:
                break
            if lr < 1e-10:
                break

    return best_a, best_f


def lbfgs_smooth(a0, beta, max_iters, deadline):
    """L-BFGS-B on the smoothed full objective F(a)."""
    a = np.maximum(a0.astype(np.float64), 1e-10)
    n = len(a)

    def func_wrapper(x):
        x2 = np.maximum(x, 1e-10)
        f, _ = full_obj_grad(x2, beta)
        return f if np.isfinite(f) else 1e10

    def jac_wrapper(x):
        x2 = np.maximum(x, 1e-10)
        _, g = full_obj_grad(x2, beta)
        return np.nan_to_num(g, nan=0.0, posinf=0.0, neginf=0.0)

    bounds = [(0.0, None)] * n
    try:
        res = minimize(func_wrapper, a.copy(), jac=jac_wrapper, method='L-BFGS-B',
                       bounds=bounds,
                       options={'maxiter': max_iters, 'max_fun': max_iters * 15,
                                'gtol': 1e-8, 'ftol': 1e-14})
        if res.fun < 1e10 and np.isfinite(res.fun):
            return np.maximum(res.x, 1e-10)
    except Exception:
        pass
    return a


def resample_seq(src, dest_n):
    """Resample 1D sequence to new length via linear interpolation."""
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    result = np.interp(t_new, t_old, src)
    return np.maximum(result, 0.0)


def get_initializations(n, rng, parent=None):
    """Generate diverse initial sequences of length n."""
    inits = []
    t = np.linspace(0, 1, n, endpoint=False)

    if parent is not None:
        p = resample_seq(parent, n)
        inits.append(p)
        inits.append(np.maximum(p + rng.normal(0, 0.05, n) * p, 0.01))
        inits.append(np.maximum(p + rng.normal(0, 0.1, n) * p, 0.01))

    inits.append(np.ones(n))
    inits.append(rng.uniform(0.3, 1.0, n))

    # Cosine families
    for freq in [1, 2, 3, 4, 5, 7, 11]:
        inits.append(1.0 + 0.25 * np.cos(2.0 * np.pi * freq * t))

    # Multi-frequency
    inits.append(1.0 + 0.15 * np.cos(2 * np.pi * 3 * t) +
                 0.12 * np.cos(2 * np.pi * 7 * t + 0.5))
    inits.append(1.0 + 0.1 * np.cos(2 * np.pi * 5 * t) +
                 0.08 * np.cos(2 * np.pi * 11 * t + 1.0) +
                 0.08 * np.cos(2 * np.pi * 13 * t + 2.0))

    # Gaussian bump
    inits.append(np.exp(-3.0 * (t - 0.5) ** 2) + 0.1)

    # Random smooth (sum of random cosines)
    n_freqs = 6
    freqs = rng.choice(range(1, n // 4 + 1), size=n_freqs, replace=False)
    amps = rng.uniform(0.03, 0.15, n_freqs)
    phases = rng.uniform(0, 2 * np.pi, n_freqs)
    sig = np.ones(n)
    for f_, a_, p_ in zip(freqs, amps, phases):
        sig += a_ * np.cos(2 * np.pi * f_ * t + p_)
    inits.append(np.maximum(sig, 0.05))

    # Block structure (on-off blocks)
    block_len = max(2, n // 32)
    blocks = rng.integers(1, 3, size=block_len * 2)
    pat = np.repeat(blocks, block_len)[:n]
    inits.append(np.maximum(pat.astype(float) * 0.5 + 0.5, 0.1))

    return inits


def run_mirror_annealing(a, deadline, tracker, betas=None):
    """Run mirror descent with beta annealing on sequence a."""
    n = len(a)
    a = np.maximum(a, 1e-10).astype(np.float64)
    a = a / a.sum() * n

    if betas is None:
        betas = [1, 2, 4, 8, 16, 32, 64, 128, 256, 512, 1024, 2048, 4096]

    for beta in betas:
        if time.monotonic() > deadline:
            break

        # Adaptive learning rate
        max_a = float(np.max(a))
        lr = 0.8 / (max_a * max_a + 1e-8)
        lr = min(lr, 2.0)

        # Number of iterations based on remaining time and beta
        remaining = deadline - time.monotonic()
        if beta <= 8:
            n_iters = 400
        elif beta <= 64:
            n_iters = 300
        elif beta <= 512:
            n_iters = 200
        else:
            n_iters = 150
        n_iters = max(20, int(n_iters * min(1.0, remaining / 60.0)))

        a, f_val = mirror_descent(a, beta, n_iters, lr, deadline)
        tracker.update(a)

    return a


class ScoreTracker:
    def __init__(self):
        self.best = None
        self.best_score = float('inf')
        self.t_start = time.monotonic()

    def update(self, a):
        a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
        if a.size == 0 or a.sum() < 0.01:
            return
        score = float(evaluate_sequence(a.tolist()))
        if np.isfinite(score) and score < self.best_score:
            self.best = a.copy()
            self.best_score = score
            elapsed = time.monotonic() - self.t_start
            print(f"  * NEW BEST: J={score:.8f} n={a.size} t={elapsed:.1f}s", flush=True)


def propose_candidate(seed=42, budget_s=1500, **kwargs):
    """
    Multi-phase optimization minimizing J(a) = 2n*max(conv(a,a))/(sum(a))^2.
    Uses entropy mirror descent with beta annealing as the core engine,
    supplemented by L-BFGS polish and multi-restart exploration.
    """
    global _best_global
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + float(budget_s) - 15.0
    tracker = ScoreTracker()

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        if p.ndim == 1 and p.size > 0 and np.all(np.isfinite(p)):
            parent = np.maximum(p, 0.0)
            tracker.update(parent)
            print(f"Parent loaded: n={parent.size}, J={tracker.best_score:.8f}", flush=True)
    except (NameError, AttributeError):
        pass

    if parent is None:
        parent = np.ones(1024)
        tracker.update(parent)
        print(f"Using default parent: n=1024, J={tracker.best_score:.8f}", flush=True)

    # === Phase 1: Quick L-BFGS at a few lengths ===
    print(f"\n{'=' * 50}", flush=True)
    print(f"Phase 1: L-BFGS Quick Warm Start", flush=True)
    print(f"{'=' * 50}", flush=True)
    p1_deadline = time.monotonic() + 80.0

    for n in [1024, 1536]:
        if time.monotonic() > p1_deadline:
            break
        print(f"  L-BFGS n={n}...", flush=True)
        inits = get_initializations(n, rng, parent)
        for a0 in inits[:6]:
            if time.monotonic() > p1_deadline:
                break
            a0 = np.maximum(a0, 0.01)
            for beta in [5.0, 20.0]:
                a_opt = lbfgs_smooth(a0, beta, max_iters=80, deadline=p1_deadline)
                tracker.update(a_opt)
    print(f"Phase 1 done: best={tracker.best_score:.8f}", flush=True)

    # === Phase 2: Mirror descent with beta annealing (MAIN) ===
    if tracker.best is not None:
        print(f"\n{'=' * 50}", flush=True)
        print(f"Phase 2: Mirror Descent Annealing (Main)", flush=True)
        print(f"{'=' * 50}", flush=True)
        p2_deadline = time.monotonic() + 850.0

        # Try the best length and a couple of alternatives
        n_best = tracker.best.size
        lengths = sorted(set([n_best, 1024, 1536, 2048]))
        lengths = [l for l in lengths if 128 <= l <= 3072][:4]

        for n_target in lengths:
            if time.monotonic() > p2_deadline:
                break

            a = resample_seq(tracker.best, n_target)
            a = np.maximum(a, 1e-10)
            a = a / a.sum() * n_target

            print(f"  Mirror annealing n={n_target}...", flush=True)
            t_start = time.monotonic()

            run_mirror_annealing(a, p2_deadline, tracker)

            # L-BFGS polish after mirror descent
            if time.monotonic() < p2_deadline:
                for beta_polish in [50.0, 200.0]:
                    if time.monotonic() > p2_deadline:
                        break
                    a = lbfgs_smooth(a, beta_polish, max_iters=150, deadline=p2_deadline)
                    tracker.update(a)

            print(f"    n={n_target} done in {time.monotonic()-t_start:.1f}s, "
                  f"best={tracker.best_score:.8f}", flush=True)

    # === Phase 3: Multi-restart with diverse perturbations ===
    if tracker.best is not None and time.monotonic() < deadline:
        p3_deadline = min(time.monotonic() + 250.0, deadline)
        print(f"\n{'=' * 50}", flush=True)
        print(f"Phase 3: Multi-Restart", flush=True)
        print(f"{'=' * 50}", flush=True)

        n_best = tracker.best.size
        attempt = 0

        for trial in range(30):
            if time.monotonic() > p3_deadline:
                break
            attempt += 1

            a = tracker.best.copy().astype(np.float64)
            n = len(a)

            # Diverse perturbation strategies
            strategy = trial % 6
            if strategy == 0:
                a = a + rng.normal(0, 0.05, n) * a
            elif strategy == 1:
                a = a + rng.normal(0, 0.12, n) * a
            elif strategy == 2:
                # Smooth noise
                noise = rng.normal(0, 0.08, n)
                ksz = max(3, n // 32)
                kernel = np.ones(ksz) / ksz
                noise = np.convolve(noise, kernel, mode='same')
                a = a + noise * a
            elif strategy == 3:
                # Spike perturbation
                k = min(n // 8, 64)
                idx = rng.choice(n, size=k, replace=False)
                a[idx] *= rng.uniform(0.3, 2.5, k)
            elif strategy == 4:
                # Frequency-based perturbation
                t = np.linspace(0, 1, n, endpoint=False)
                freq = rng.integers(2, n // 4)
                phase = rng.uniform(0, 2 * np.pi)
                a = a * (1.0 + 0.1 * np.cos(2 * np.pi * freq * t + phase))
            else:
                # Block perturbation
                block = max(4, n // 64)
                for _ in range(3):
                    start = rng.integers(0, n - block)
                    a[start:start + block] *= rng.uniform(0.5, 1.8)

            a = np.maximum(a, 1e-6)
            a = a / a.sum() * n

            # Quick mirror descent (shorter schedule)
            remaining = p3_deadline - time.monotonic()
            if remaining > 5:
                sub_deadline = min(p3_deadline, time.monotonic() + min(remaining, 30.0))
                run_mirror_annealing(a, sub_deadline, tracker,
                                     betas=[16, 64, 256, 1024])

            if attempt % 5 == 0:
                print(f"  Restart {attempt}: best={tracker.best_score:.8f}", flush=True)

        print(f"Phase 3 done: {attempt} attempts, best={tracker.best_score:.8f}", flush=True)

    # === Phase 4: Final high-precision polish ===
    if tracker.best is not None and time.monotonic() < deadline:
        p4_deadline = deadline
        print(f"\n{'=' * 50}", flush=True)
        print(f"Phase 4: Final High-Beta Polish", flush=True)
        print(f"{'=' * 50}", flush=True)

        a = tracker.best.copy().astype(np.float64)
        n = len(a)
        a = a / a.sum() * n

        # Very high beta mirror descent for precision
        for beta in [512, 2048, 8192, 16384]:
            if time.monotonic() > p4_deadline:
                break
            max_a = float(np.max(a))
            lr = 0.5 / (max_a * max_a + 1e-8)
            lr = min(lr, 1.0)
            remaining = p4_deadline - time.monotonic()
            n_iters = max(50, int(remaining * 50))
            a, _ = mirror_descent(a, beta, n_iters, lr, p4_deadline)
            tracker.update(a)

        # Final L-BFGS
        if time.monotonic() < p4_deadline:
            for beta_f in [100.0, 500.0]:
                if time.monotonic() > p4_deadline:
                    break
                a = lbfgs_smooth(a, beta_f, max_iters=300, deadline=p4_deadline)
                tracker.update(a)

        print(f"Phase 4 done: best={tracker.best_score:.8f}", flush=True)

    # === Phase 5: Try nearby lengths ===
    if tracker.best is not None and time.monotonic() < deadline:
        p5_deadline = deadline
        n_best = tracker.best.size
        remaining = p5_deadline - time.monotonic()
        if remaining > 20:
            print(f"\n{'=' * 50}", flush=True)
            print(f"Phase 5: Length Exploration ({remaining:.0f}s left)", flush=True)
            print(f"{'=' * 50}", flush=True)

            for n_try in [n_best - 64, n_best + 64, n_best + 128,
                          768, 896, 1152, 1280, 1408, 1664]:
                if n_try < 128 or n_try > 3072:
                    continue
                if abs(n_try - n_best) < 32:
                    continue
                if time.monotonic() > p5_deadline:
                    break

                a_res = resample_seq(tracker.best, n_try)
                a_res = np.maximum(a_res, 1e-10)
                a_res = a_res / a_res.sum() * n_try

                sub_dl = min(p5_deadline, time.monotonic() + 30.0)
                run_mirror_annealing(a_res, sub_dl, tracker,
                                     betas=[32, 128, 512, 2048])
                print(f"  n={n_try}: best={tracker.best_score:.8f}", flush=True)

    # Fallback
    if tracker.best is None:
        tracker.best = np.ones(1024)
        tracker.best_score = compute_J(tracker.best)
        print(f"  Using fallback: J={tracker.best_score:.8f}", flush=True)

    print(f"\n{'=' * 50}", flush=True)
    print(f"FINAL RESULT", flush=True)
    print(f"{'=' * 50}", flush=True)
    print(f"J = {tracker.best_score:.8f}", flush=True)
    print(f"n = {tracker.best.size}", flush=True)
    print(f"Total time = {time.monotonic() - tracker.t_start:.1f}s", flush=True)

    result = np.clip(tracker.best, 0.0, 1000.0)
    if result.sum() < 0.01:
        result = np.ones(128)

    # Final validation
    final_score = float(evaluate_sequence(result.tolist()))
    if not np.isfinite(final_score):
        result = np.ones(1024)
        final_score = float(evaluate_sequence(result.tolist()))

    print(f"Validated final score: {final_score:.8f}", flush=True)
    return [float(x) for x in result]
```
