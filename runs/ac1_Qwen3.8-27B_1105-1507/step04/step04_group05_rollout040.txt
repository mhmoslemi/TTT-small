Minimizes J(a) = 2*n*max(conv(a,a))/(sum(a))^2 using:

import numpy as np
from scipy.optimize import linprog
from scipy.sparse import lil_matrix
import time

def fast_conv(a):
    n = len(a)
    s = 1
    while s < 2*n - 1:
        s <<= 1
    fa = np.fft.rfft(a, s)
    return np.fft.irfft(fa * fa, s)[:2*n-1]

def compute_J(a):
    a = np.maximum(a, 0)
    S = a.sum()
    if S < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (S * S))

def build_A_sparse(a):
    n = len(a)
    rows = []
    cols = []
    vals = []
    for k in range(2*n - 1):
        jl = max(0, k - n + 1)
        jh = min(n-1, k)
        if jl <= jh:
            cnt = jh - jl + 1
            rows.extend([k]*cnt)
            cols.extend(range(jl, jh+1))
            seg = a[k-jh:k-jl+1][::-1]
            vals.extend(seg.tolist())
    A = lil_matrix((2*n-1, n))
    for idx in range(len(rows)):
        A[rows[idx], cols[idx]] = vals[idx]
    return A.tocsr()

def lp_best_response(a, M):
    n = len(a)
    c = -np.ones(n)
    A = build_A_sparse(a)
    b_ub = np.full(2*n-1, M)
    bounds = [(0.0, None)] * n
    r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
    if r.success and r.x is not None:
        g = np.maximum(r.x, 0.0)
        if g.sum() > 1e-10:
            return g
    return None

"""
Minimizes J(a) = 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Exact 1D coordinate descent via active constraint analysis (key innovation)
2. LP-based alternating minimization with adaptive tightening
3. Multi-restart with diverse structures and length exploration

import numpy as np
from scipy.optimize import linprog
from scipy.sparse import lil_matrix, csr_matrix
import time


def fast_conv(a):
    """Full self-convolution via FFT."""
    n = len(a)
    s = 1
    while s < 2 * n - 1:
        s <<= 1
    fa = np.fft.rfft(a, s)
    return np.fft.irfft(fa * fa, s)[:2 * n - 1]


def compute_J(a):
    """Compute J(a) = 2*n*max(conv(a,a))/(sum(a))^2."""
    a = np.maximum(a, 0)
    S = a.sum()
    if S < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (S * S))


def build_A_sparse(a):
    """Build sparse constraint matrix for LP: (a * g)[k] <= M for all k."""
    n = len(a)
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        jl = max(0, k - n + 1)
        jh = min(n - 1, k)
        if jl <= jh:
            cnt = jh - jl + 1
            rows.extend([k] * cnt)
            cols.extend(range(jl, jh + 1))
            seg = a[k - jh:k - jl + 1][::-1]
            vals.extend(seg.tolist())
    A = lil_matrix((2 * n - 1, n))
    for idx in range(len(rows)):
        A[rows[idx], cols[idx]] = vals[idx]
    return A.tocsr()


def lp_best_response(a, M):
    """LP: max sum(g) s.t. (a*g)[k] <= M for all k, g >= 0."""
    n = len(a)
    c = -np.ones(n)
    A = build_A_sparse(a)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
    if r.success and r.x is not None:
        g = np.maximum(r.x, 0.0)
        if g.sum() > 1e-10:
            return g
    return None


def optimize_coordinate(a, i):
    """Optimize coordinate i using active constraint fixed-point iteration.
    Returns (updated_a, improved_flag)."""
    n = len(a)
    S_rest = a.sum() - a[i]
    ai_orig = a[i]

    best_J = float('inf')
    best_val = ai_orig

    for it in range(40):
        S_total = S_rest + a[i]
        if S_total < 1e-10:
            break
        b = fast_conv(a)
        max_val = b.max()
        max_h = int(np.argmax(b))

        current_J = 2.0 * n * max_val / (S_total * S_total)
        if current_J < best_J - 1e-15:
            best_J = current_J
            best_val = a[i]

        hi = max_h - i

        if i <= max_h <= i + n - 1:
            if max_h == 2 * i:
                # Quadratic: result = C + x^2
                C = b[max_h] - a[i] * a[i]
                if S_rest > 1e-10:
                    x_new = C / S_rest
                else:
                    x_new = a[i]
            else:
                # Linear: result = D + 2*a[hi]*x
                coeff = 2.0 * a[hi]
                D = b[max_h] - 2.0 * a[i] * a[hi]
                if coeff > 1e-15:
                    x_new = S_rest - D / coeff
                else:
                    x_new = a[i]
        else:
            # Active constraint not affected by x: increase x to grow denominator
            x_new = a[i] + S_rest / n

        x_new = max(0.0, min(x_new, 20.0 * S_rest / n + 1.0))

        if abs(x_new - a[i]) < 1e-12 * max(1.0, a[i]):
            break

        a[i] = x_new

    a[i] = best_val
    return a, (best_val != ai_orig)


def coordinate_descent_pass(a, order=None):
    """One full pass of coordinate descent with exact 1D subproblem solving."""
    a = a.copy()
    a = np.maximum(a, 0.0)
    n = len(a)
    if a.sum() < 1e-10:
        return a

    if order is None:
        order = list(range(n))

    any_improved = False
    for i in order:
        a, improved = optimize_coordinate(a, i)
        if improved:
            any_improved = True

    return a, any_improved


def coordinate_descent(a, n_passes, deadline, tracker):
    """Multiple passes of exact coordinate descent."""
    a = np.maximum(a.astype(np.float64), 0.0)
    S = a.sum()
    if S < 1e-10:
        return a
    a = a / S
    current_J = compute_J(a)
    tracker.update(a)

    for p in range(n_passes):
        if time.monotonic() > deadline:
            break
        t0 = time.monotonic()
        order = list(range(len(a)))
        np.random.shuffle(order)
        a, improved = coordinate_descent_pass(a, order)
        if a.sum() > 1e-10:
            a = a / a.sum()
        new_J = compute_J(a)
        if new_J < current_J - 1e-13:
            current_J = new_J
            tracker.update(a)
            if p % 3 == 0:
                print(f"    CD pass {p}: J={current_J:.8f} ({time.monotonic()-t0:.1f}s)", flush=True)
        elif not improved and p > 2:
            break

    tracker.update(a)
    return a


def run_lp_iterations(a, max_iters, deadline, tracker, tighten_base=0.003):
    """LP mirror descent with adaptive tightening."""
    a = np.maximum(a.astype(np.float64), 1e-6)
    a = a / a.sum()
    n = len(a)
    current_J = compute_J(a)
    tracker.update(a)

    tighten = tighten_base
    stall_count = 0

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        b = fast_conv(a)
        M = float(b.max())
        M_target = M * (1.0 - tighten)

        g = lp_best_response(a, M_target)
        if g is None:
            M_target = M * (1.0 - tighten * 0.3)
            g = lp_best_response(a, M_target)
        if g is None:
            tighten = max(1e-5, tighten * 0.5)
            stall_count += 1
            if stall_count > 30:
                break
            continue

        g = np.maximum(g, 0.0)
        g_sum = g.sum()
        if g_sum < 1e-10:
            break
        g = g / g_sum

        progress = min(1.0, it / max(1, max_iters))
        base_t = max(0.02, 0.6 * (1.0 - progress * 0.8))
        candidates_t = [base_t, base_t * 0.5, base_t * 1.5, 0.8, 0.4]

        best_a = None
        best_J = current_J

        for t in candidates_t:
            a_trial = (1.0 - t) * a + t * g
            a_trial = np.maximum(a_trial, 0.0)
            s = a_trial.sum()
            if s < 1e-10:
                continue
            a_trial = a_trial / s
            J_trial = compute_J(a_trial)
            if J_trial < best_J - 1e-14:
                best_J = J_trial
                best_a = a_trial

        if best_a is None:
            stall_count += 1
            tighten = max(1e-5, tighten * 0.6)
            if stall_count > 40:
                break
            continue

        a = best_a
        current_J = best_J
        stall_count = 0
        tighten = min(0.05, tighten * 1.02)

        if it % 15 == 0:
            tracker.update(a)
            print(f"    LP iter {it}: J={current_J:.8f}", flush=True)

    tracker.update(a)
    return a


def resample_seq(src, dest_n):
    """Resample 1D sequence to new length via linear interpolation."""
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)

def generate_structured_inits(n, rng):
    """Generate diverse structured initial sequences."""
    inits = []
    t = np.linspace(0, 1, n, endpoint=False)

    inits.append(np.ones(n))

    for freq in [1, 2, 3, 5, 7, 11]:
        amp = 0.3 / freq
        inits.append(np.maximum(1.0 + amp * np.cos(2.0 * np.pi * freq * t + rng.uniform(0, 6.28)), 0.1))

    inits.append(np.maximum(1.0 + 0.2 * np.cos(2*np.pi*2*t) + 0.15 * np.cos(2*np.pi*5*t), 0.1))
    inits.append(np.abs(np.sin(np.pi * t)) + 0.2)
    inits.append(1.0 - 0.3 * np.abs(2*t - 1.0))
    inits.append(0.5 * np.exp(-8*(t-0.3)**2) + 0.5 * np.exp(-8*(t-0.7)**2) + 0.15)
    inits.append(rng.uniform(0.3, 1.0, n))

    # Thinned uniform (random zeros)
    a = np.ones(n)
    mask = rng.random(n) < 0.3
    a[mask] = 0
    if a.sum() > 0:
        inits.append(a)

    # Ramp-like
    inits.append(np.maximum(1.0 + 0.5*(t - 0.5), 0.1))

    return inits

    for p in range(n_passes):
        if time.monotonic() > deadline:
            break
        improved = False
        for i in range(half):
            j = n - 1 - i
            # Optimize a[i] and a[j] together (they must be equal)
            # Try a few values
            S_rest = a.sum() - a[i] - (a[j] if i != j else 0)
            ai = a[i]
            best_x = ai
            best_J_val = float('inf')

            for scale in [0.5, 0.7, 0.85, 1.0, 1.15, 1.3, 1.5, 2.0]:
                x = ai * scale
                a_test = a.copy()
                a_test[i] = x
                if i != j:
                    a_test[j] = x
                S_t = a_test.sum()
                if S_t < 1e-10:
                    continue
                b = fast_conv(a_test)
                J_val = 2.0 * n * b.max() / (S_t * S_t)
                if J_val < best_J_val:
                    best_J_val = J_val
                    best_x = x

            if best_x != ai:
                a[i] = best_x
                if i != j:
                    a[j] = best_x
                improved = True

        a = a / a.sum()
        new_J = compute_J(a)
        if new_J < current_J - 1e-13:
            current_J = new_J
            tracker.update(a)
        elif not improved:
            break

    tracker.update(a)
    return a


def propose_candidate(seed=42, budget_s=1500, **kwargs):
    """
    Main search function combining exact coordinate descent, LP optimization,
    and multi-restart exploration.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + float(budget_s) - 30.0
    tracker = ScoreTracker()

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        if p.ndim == 1 and p.size > 0 and np.all(np.isfinite(p)):
            parent = np.maximum(p, 0.0)
            tracker.update(parent)
            print(f"Parent loaded: n={parent.size}, J={tracker.best_score:.8f}", flush=True)
    except (NameError, AttributeError):
        pass

    if parent is None:
        parent = np.ones(384)
        tracker.update(parent)
        print(f"No parent, using uniform n=384", flush=True)

    n_parent = parent.size
    print(f"\n{'='*60}", flush=True)
    print(f"Starting optimization with n={n_parent}", flush=True)
    print(f"{'='*60}", flush=True)

    # === Phase 1: Quick LP to get a good starting point ===
    print(f"\n--- Phase 1: LP warmup ---", flush=True)
    t0 = time.monotonic()
    budget1 = min(60, (deadline - time.monotonic()) * 0.1)
    a = run_lp_iterations(parent.copy(), 30, time.monotonic() + budget1, tracker,
                          tighten_base=0.004)
    print(f"  Phase 1 done: {time.monotonic()-t0:.1f}s, J={tracker.best_score:.8f}", flush=True)

    # === Phase 2: Exact coordinate descent ===
    print(f"\n--- Phase 2: Exact Coordinate Descent ---", flush=True)
    t0 = time.monotonic()
    budget2 = min(120, (deadline - time.monotonic()) * 0.2)
    a = coordinate_descent(tracker.best.copy(), n_passes=30,
                           deadline=time.monotonic() + budget2, tracker=tracker)
    print(f"  Phase 2 done: {time.monotonic()-t0:.1f}s, J={tracker.best_score:.8f}", flush=True)

    # === Phase 3: Alternate LP + CD ===
    print(f"\n--- Phase 3: Alternating LP + CD ---", flush=True)
    t0 = time.monotonic()
    phase3_deadline = time.monotonic() + min(300, (deadline - time.monotonic()) * 0.35)
    iteration = 0
    while time.monotonic() < phase3_deadline:
        # LP step
        remaining = phase3_deadline - time.monotonic()
        lp_time = min(remaining * 0.5, 20)
        a = run_lp_iterations(tracker.best.copy(), 15, time.monotonic() + lp_time, tracker,
                              tighten_base=0.002 / (1 + 0.1 * iteration))

        if time.monotonic() > phase3_deadline:
            break

        # CD step
        cd_time = min(phase3_deadline - time.monotonic(), 30)
        a = coordinate_descent(tracker.best.copy(), n_passes=5,
                               deadline=time.monotonic() + cd_time, tracker=tracker)
        iteration += 1
        if iteration % 5 == 0:
            print(f"    Alt iter {iteration}: J={tracker.best_score:.8f}", flush=True)

    print(f"  Phase 3 done: {time.monotonic()-t0:.1f}s, J={tracker.best_score:.8f}", flush=True)

    # === Phase 4: Try different lengths ===
    print(f"\n--- Phase 4: Length Exploration ---", flush=True)
    t0 = time.monotonic()
    phase4_deadline = time.monotonic() + min(200, (deadline - time.monotonic()) * 0.25)

    lengths_to_try = []
    base = n_parent
    for delta in [-6, -4, -2, 0, 2, 4, 6, 8, 10, 12]:
        L = base + delta
        if 200 <= L <= 600 and L not in lengths_to_try:
            lengths_to_try.append(L)

    for nl in lengths_to_try:
        if time.monotonic() > phase4_deadline - 30:
            break
        if nl == n_parent:
            a_trial = tracker.best.copy()
        else:
            a_trial = resample_seq(tracker.best, nl)
        a_trial = np.maximum(a_trial, 0.001)
        a_trial = a_trial / a_trial.sum()

        # Quick LP + CD
        lp_dead = time.monotonic() + min(40, (phase4_deadline - time.monotonic()) * 0.4)
        a_trial = run_lp_iterations(a_trial, 15, lp_dead, tracker, tighten_base=0.003)

        if time.monotonic() < phase4_deadline - 20:
            cd_dead = time.monotonic() + min(30, (phase4_deadline - time.monotonic()) * 0.5)
            a_trial = coordinate_descent(a_trial, n_passes=5, deadline=cd_dead, tracker=tracker)

        print(f"    Length {nl}: J={tracker.best_score:.8f}", flush=True)

    print(f"  Phase 4 done: {time.monotonic()-t0:.1f}s, J={tracker.best_score:.8f}", flush=True)

    # === Phase 5: Symmetric optimization ===
    print(f"\n--- Phase 5: Symmetric Optimization ---", flush=True)
    t0 = time.monotonic()
    phase5_deadline = time.monotonic() + min(120, (deadline - time.monotonic()) * 0.15)

    a_sym = tracker.best.copy()
    a_sym = optimize_symmetric(a_sym, n_passes=20, deadline=phase5_deadline, tracker=tracker)

    # Also try CD on the symmetric result without symmetry constraint
    if time.monotonic() < phase5_deadline:
        a_sym_asym = coordinate_descent(a_sym, n_passes=10,
                                        deadline=phase5_deadline, tracker=tracker)

    print(f"  Phase 5 done: {time.monotonic()-t0:.1f}s, J={tracker.best_score:.8f}", flush=True)

    # === Phase 6: Multi-restart with perturbations ===
    print(f"\n--- Phase 6: Perturbation Restarts ---", flush=True)
    t0 = time.monotonic()
    phase6_deadline = time.monotonic() + min(150, (deadline - time.monotonic()) * 0.2)

    n = tracker.best.size
    perturbations = [
        (0.05, "Gauss 0.05"),
        (0.03, "Gauss 0.03"),
        (0.02, "Gauss 0.02"),
        (0.01, "Gauss 0.01"),
    ]

    for temp, label in perturbations:
        if time.monotonic() > phase6_deadline - 30:
            break

        a_pert = tracker.best.copy() + rng.normal(0, temp, n)
        a_pert = np.maximum(a_pert, 0.001)
        a_pert = a_pert / a_pert.sum()

        # CD polish
        cd_dead = time.monotonic() + min(40, (phase6_deadline - time.monotonic()) * 0.5)
        a_pert = coordinate_descent(a_pert, n_passes=10, deadline=cd_dead, tracker=tracker)

        # Quick LP
        if time.monotonic() < phase6_deadline - 10:
            lp_dead = time.monotonic() + min(20, (phase6_deadline - time.monotonic()) * 0.4)
            a_pert = run_lp_iterations(a_pert, 10, lp_dead, tracker, tighten_base=0.002)
            a_pert = coordinate_descent(a_pert, n_passes=5,
                                        deadline=min(lp_dead + 10, phase6_deadline), tracker=tracker)

        print(f"    {label}: J={tracker.best_score:.8f}", flush=True)

    print(f"  Phase 6 done: {time.monotonic()-t0:.1f}s, J={tracker.best_score:.8f}", flush=True)

    # === Phase 7: Final intensive coordinate descent ===
    print(f"\n--- Phase 7: Final Polish ---", flush=True)
    t0 = time.monotonic()
    final_deadline = deadline

    # Long CD on best
    a_final = coordinate_descent(tracker.best.copy(), n_passes=50,
                                 deadline=time.monotonic() + (final_deadline - time.monotonic()) * 0.5,
                                 tracker=tracker)

    if time.monotonic() < final_deadline - 10:
        # One more LP + CD
        a_final = run_lp_iterations(tracker.best, 20,
                                    time.monotonic() + (final_deadline - time.monotonic()) * 0.3,
                                    tracker, tighten_base=0.001)
        a_final = coordinate_descent(tracker.best, n_passes=20,
                                     deadline=final_deadline - 5, tracker=tracker)

    print(f"  Phase 7 done: {time.monotonic()-t0:.1f}s, J={tracker.best_score:.8f}", flush=True)

    # Fallback
    if tracker.best is None:
        tracker.best = np.ones(256)
        tracker.best_score = compute_J(tracker.best)

    print(f"\n{'='*60}", flush=True)
    print(f"FINAL RESULT: J = {tracker.best_score:.10f}, n = {tracker.best.size}", flush=True)
    print(f"Total time: {time.monotonic()-tracker.t_start:.1f}s", flush=True)
    print(f"{'='*60}", flush=True)

    return [float(x) for x in tracker.best]
