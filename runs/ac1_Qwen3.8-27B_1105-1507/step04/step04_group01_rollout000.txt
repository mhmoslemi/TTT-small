    # Or more simply: grad_a[i] = 2 * (cross_correlation of a and v)[i]
    # where cross_correlation[a,v][i] = sum_j a[j]*v[i+j]
    
    # Using FFT: cross_correlation can be computed as:
    # corr[i] = sum_j a[j] * v[i+j] = (a_rev * v)[len(a)-1-i] ... hmm
    # Let me just use the direct formula with FFT:
    # Define a_padded of length >= 3n-2
    # FFT approach: corr = ifft(fft(a, N) * fft(v_flipped, N)) where v_flipped appropriately
    
    # Actually, the simplest way: 
    # grad_a[i] = 2 * sum_{j=0}^{n-1} a[j] * w[i+j]
    # This is the i-th element of the "running sum" or "sliding dot product"
    # We can compute it as: for each i, dot(a, w[i:i+n])
    # Using convolution: if we define w_ext = w (length 2n-1), then
    # grad_a[i] = dot(a, w_ext[i:i+n]) = (reverse_convolve)(a, w_ext)[i]
    # = convolve(a, w_ext_flipped)[(2n-2)-i + n - 1 - ... ]
    
    # Let me just use the FFT approach directly:
    # Define u of length 2n-1 where u[k] = w[k]
    # grad_a[i] = sum_{j=0}^{n-1} a[j] * u[i+j]
    # Let's flip a: a_flip[j] = a[n-1-j]
    # Then grad_a[i] = sum_{j=0}^{n-1} a_flip[n-1-j] * u[i+j]
    # Hmm, this is getting convoluted. Let me just use the direct O(n^2) approach
    # for now since n=600 gives 360K operations which is fast.
    
    # Actually, let me use the correlation via FFT properly:
    # grad_a[i] = sum_{j=0}^{n-1} a[j] * w[i+j]
    # Let's define the cross-correlation of w and a:
    # (w ⊛ a)[i] = sum_j w[j] * a[j-i] (standard cross-correlation)
    # That's not quite what we want.
    # We want: sum_j a[j] * w[i+j] = sum_j a[j] * w[i+j]
    # Let k = i+j, so j = k-i: sum_{k=i}^{i+n-1} a[k-i] * w[k]
    # = (a * w)[i] where * denotes cross-correlation in a specific sense
    # Actually this IS the i-th element of the linear cross-correlation of a and w:
    # (a ⊛ w)[i] = sum_j a[j] * conj(w[j-i]) for complex, or sum_j a[j]*w[j-i] for real
    # Hmm, that's sum_j a[j]*w[j-i], not sum_j a[j]*w[i+j].
    
    # Let me just compute it directly using a different approach:
    # grad_a[i] = sum_{j=0}^{n-1} a[j]*w[i+j]
    # Define b[j] = a[n-1-j] (reversed a)
    # Then grad_a[i] = sum_{j=0}^{n-1} b[n-1-j]*w[i+j]
    # Let m = n-1-j, so j = n-1-m:
    # = sum_{m=0}^{n-1} b[m]*w[i+n-1-m]
    # = (b * w)[i+n-1] where * is standard convolution
    # So grad_a = 2 * convolve(b, w)[n-1 : 2n-1]
    # where convolve(b,w) has length (n-1)+(2n-2)+1 = 3n-2
    
    # Let's verify: convolve(b, w)[k] = sum_m b[m]*w[k-m]
    # convolve(b, w)[i+n-1] = sum_m b[m]*w[i+n-1-m]
    # = sum_{m=0}^{n-1} a[n-1-m]*w[i+n-1-m]
    # Let j = n-1-m: = sum_{j=0}^{n-1} a[j]*w[i+j]  ✓
    
    b = a[::-1].copy()
    conv_bw = fft_convolve(b, w)  # length n + (2n-1) - 1 = 3n-2
    grad_a = 2.0 * conv_bw[n-1:2*n-1]
    
    # Project gradient onto the simplex {a>=0, sum(a)=1}
    # Simple approach: subtract mean, clip
    grad_centered = grad_a - np.mean(grad_a)
    a_new = a - step_size * grad_centered
    a_new = np.maximum(a_new, 0)
    s = np.sum(a_new)
    if s > 1e-10:
        a_new = a_new / s
    else:
        a_new = a.copy()
    
    return a_new


def polyak_subgradient(a, f_best, max_iters=500, time_budget=10.0, tau=5.0):
    """Subgradient with Polyak step size."""
    n = len(a)
    a = a.copy().astype(np.float64)
    s = np.sum(a)
    if s < 1e-10:
        return a
    a = a / s
    
    f_current = compute_obj(a)
    f_best_local = min(f_best, f_current)
    start = time.monotonic()
    stall = 0
    
    for it in range(max_iters):
        if time.monotonic() - start > time_budget:
            break
        
        a_new = subgradient_step(a, tau, 1.0)  # unit step, we'll scale
        # Actually let's just do a proper step
        # Compute gradient properly
        conv = compute_conv(a)
        max_conv = float(np.max(conv))
        shifted = np.clip((conv - max_conv) / tau, -500.0, 500.0)
        weights = np.exp(shifted)
        w_sum = np.sum(weights)
        if w_sum < 1e-300:
            idx = int(np.argmax(conv))
            weights = np.zeros(len(conv))
            weights[idx] = 1.0
        else:
            weights /= w_sum
        
        b = a[::-1].copy()
        conv_bw = fft_convolve(b, weights)
        grad_a = 2.0 * conv_bw[n-1:2*n-1]
        grad_centered = grad_a - np.mean(grad_a)
        
        # Polyak step size
        f_now = 2.0 * n * max_conv / (s * s)  # current objective (using normalized a)
        # For normalized a (sum=1): obj = 2n * max_conv
        f_now_norm = 2.0 * n * max_conv
        gap = f_now_norm - 2.0 * n * (f_best_local / (2.0 * n))  # = f_now_norm - f_best_normalized
        # Actually f_best is the full objective = 2n*max_r/(sum_a)^2
        # For sum_a = 1: f = 2n*max_r
        # So the "target" max_r = f_best / (2n)
        # gap in terms of max_r: max_conv - f_best/(2n)
        
        target_max = f_best_local / (2.0 * n)
        gap = max_conv - target_max
        
        if gap < 1e-10:
            break
        
        grad_norm_sq = float(np.dot(grad_centered, grad_centered))
        if grad_norm_sq < 1e-30:
            break
        
        # Polyak step: step = gap / ||grad||^2 (approximately)
        step = gap / (grad_norm_sq + 1e-20)
        step = min(step, 0.5)  # cap the step
        
        a_try = a - step * grad_centered
        a_try = np.maximum(a_try, 0)
        s_try = np.sum(a_try)
        if s_try > 1e-10:
            a_try = a_try / s_try
            obj_try = compute_obj(a_try)
            if obj_try < f_best_local:
                f_best_local = obj_try
                a = a_try
                stall = 0
            else:
                # Try smaller step
                for scale in [0.5, 0.25, 0.1]:
                    a_try2 = a - step * scale * grad_centered
                    a_try2 = np.maximum(a_try2, 0)
                    s_try2 = np.sum(a_try2)
                    if s_try2 > 1e-10:
                        a_try2 = a_try2 / s_try2
                        obj_try2 = compute_obj(a_try2)
                        if obj_try2 < f_best_local:
                            f_best_local = obj_try2
                            a = a_try2
                            stall = 0
                            break
                else:
                    stall += 1
                    if stall > 50:
                        break
        else:
            stall += 1
            if stall > 50:
                break
    
    return a, f_best_local


def lp_polish(a, max_iters=200, time_budget=60.0, verbose=False):
    """Intensive LP polishing with geometric M reduction."""
    n = len(a)
    a = a.copy().astype(np.float64)
    s = np.sum(a)
    if s < 1e-10:
        return a
    a = a / s
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    stall = 0
    
    for it in range(max_iters):
        if time.monotonic() - start > time_budget:
            break
        
        conv = compute_conv(a)
        M = float(np.max(conv))
        
        # Solve LP: max sum(g) s.t. (a*g)[k] <= M, g >= 0
        g, sum_g = solve_lp_for_g(a, M)
        if g is None:
            break
        
        # Normalize g to have same sum as a
        g_norm = g / sum_g * s
        
        # Line search for best blend
        best_t_obj = best_obj
        best_a_new = best_a.copy()
        for t in [0.05, 0.1, 0.2, 0.3, 0.5, 0.7, 1.0]:
            a_try = (1.0 - t) * a + t * g_norm
            a_try = np.maximum(a_try, 0)
            s_try = np.sum(a_try)
            if s_try < 1e-10:
                continue
            a_try = a_try / s_try
            obj_try = compute_obj(a_try)
            if obj_try < best_t_obj:
                best_t_obj = obj_try
                best_a_new = a_try.copy()
        
        if best_t_obj < best_obj - 1e-12:
            best_obj = best_t_obj
            best_a = best_a_new.copy()
            a = best_a
            stall = 0
            if verbose and it % 20 == 0:
                print(f"    LP iter {it}: obj={best_obj:.6f}")
        else:
            # Try with reduced M
            stall += 1
            if stall > 10:
                # Perturb and restart
                rng = np.random.default_rng(it * 7919 + 42)
                a = a * np.exp(rng.normal(0, 0.05, n))
                a = np.maximum(a, 0)
                s = np.sum(a)
                if s > 1e-10:
                    a = a / s
                stall = 0
            elif stall > 5:
                # Try smaller M
                M_reduced = M * 0.99
                g2, sum_g2 = solve_lp_for_g(a, M_reduced)
                if g2 is not None and sum_g2 > 1e-10:
                    g2_norm = g2 / sum_g2 * s
                    for t in [0.1, 0.3, 0.5]:
                        a_try = (1.0 - t) * a + t * g2_norm
                        a_try = np.maximum(a_try, 0)
                        s_try = np.sum(a_try)
                        if s_try > 1e-10:
                            a_try = a_try / s_try
                            obj_try = compute_obj(a_try)
                            if obj_try < best_obj:
                                best_obj = obj_try
                                best_a = a_try.copy()
                                a = best_a
                                stall = 0
                                break
    
    return best_a, best_obj


def spectral_perturb(a, seed):
    """Perturb using spectral information to flatten the power spectrum."""
    n = len(a)
    rng = np.random.default_rng(seed)
    
    # Compute power spectrum
    fa = rfft(a, n)
    power = np.abs(fa)**2
    
    # Target: flatten the power spectrum (make it more uniform)
    # Current spectrum has peaks; we want to reduce them
    # Gradient of sum(power^2) w.r.t. a gives direction to flatten
    # d(sum_k |F[k]|^4)/da = 4 * ifft(F * conj(F^3)) ... complex
    # Simpler: just add a small random spectral perturbation
    
    # Generate random phase perturbation in frequency domain
    fa_pert = fa.copy()
    for k in range(1, len(fa)):
        fa_pert[k] *= np.exp(1j * rng.normal(0, 0.3))
    
    a_pert = irfft(fa_pert, n)[:n]
    a_pert = np.maximum(a_pert, 0)
    s = np.sum(a_pert)
    if s > 1e-10:
        a_pert = a_pert / s
        return a_pert
    
    return a.copy()


def resample_sequence(a, new_n):
    n = len(a)
    if n == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0.001)
