```python
"""
Minimizes J(a) = 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Fast coordinate descent (Gauss-Seidel with golden section search per coordinate)
2. Smooth gradient descent with adaptive high-tau log-sum-exp
3. LP mirror descent for precise local refinement
4. Multi-length exploration with informed initialization (Gaussian bumps, resampling)
5. L-BFGS-B on smoothed objective for additional polishing
Key innovation: O(n^2) coordinate descent sweeps are ~1000x cheaper per iteration
than LP-based methods, enabling thousands of refinement passes.
"""

import numpy as np
from scipy.optimize import linprog, minimize
from scipy.sparse import csr_matrix
import time


def fast_conv(a):
    """Full self-convolution via FFT."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]


def compute_J(a):
    """Compute J(a) = 2*n*max(conv(a,a))/(sum(a))^2."""
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))


def resample_seq(src, dest_n):
    """Resample 1D sequence to new length via linear interpolation."""
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)


def coordinate_descent(a, max_sweeps, deadline, rng, verbose=False):
    """
    Gauss-Seidel coordinate descent with golden section search per coordinate.
    Maintains b = conv(a,a) incrementally for O(n) per coordinate evaluation.
    """
    a = np.maximum(a.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-10:
        return a
    a = a / s
    n = len(a)
    b = fast_conv(a)
    S = 1.0  # normalized sum

    best_a = a.copy()
    best_J = compute_J(a)
    no_improve_sweeps = 0

    for sweep in range(max_sweeps):
        if time.monotonic() > deadline:
            break

        order = rng.permutation(n)
        sweep_changed = False

        for i in order:
            ai = a[i]
            # Precompute a_shifted: a[k-i] for valid k, zero elsewhere
            a_shifted = np.zeros(2 * n - 1)
            hi = min(2 * n - 1, i + n)
            a_shifted[i:hi] = a[0:hi - i]

            delta_sq_pos = 2 * i if 2 * i < 2 * n - 1 else -1

            # Golden section search for optimal t >= 0
            # J(t) = 2n * max(b + 2*(t-ai)*a_shifted + (t-ai)^2 * e_{2i}) / (S + t - ai)^2
            t_lo = 0.0
            t_hi = max(4.0 * ai + 5.0 / n, 15.0 / n)

            # Quick check: is t=0 better?
            def J_at(t):
                d = t - ai
                mb = (b + 2.0 * d * a_shifted).max()
                if delta_sq_pos >= 0:
                    val = b[delta_sq_pos] + 2.0 * d * a_shifted[delta_sq_pos] + d * d
                    if val > mb:
                        mb = val
                ns = S + d
                if ns < 1e-10:
                    return 1e10
                return 2.0 * n * mb / (ns * ns)

            gr = (np.sqrt(5.0) - 1.0) / 2.0
            t1 = t_hi - gr * (t_hi - t_lo)
            t2 = t_hi - (1.0 - gr) * (t_hi - t_lo)
            f1 = J_at(t1)
            f2 = J_at(t2)

            for _ in range(45):
                if f1 < f2:
                    t_hi = t2
                    t2 = t1
                    f2 = f1
                    t1 = t_hi - gr * (t_hi - t_lo)
                    f1 = J_at(t1)
                else:
                    t_lo = t1
                    t1 = t2
                    f1 = f2
                    t2 = t_hi - (1.0 - gr) * (t_hi - t_lo)
                    f2 = J_at(t2)

            t_opt = (t_lo + t_hi) / 2.0

            # Also check endpoints
            f_lo = J_at(0.0)
            f_hi = J_at(t_hi)
            if f_lo < min(f1, f2):
                t_opt = 0.0
            elif f_hi < min(f1, f2):
                t_opt = t_hi

            delta = t_opt - ai
            if abs(delta) > 1e-10:
                # Update b incrementally
                hi = min(2 * n - 1, i + n)
                b[i:hi] += 2.0 * delta * a[0:hi - i]
                if delta_sq_pos >= 0:
                    b[delta_sq_pos] += delta * delta
                a[i] = t_opt
                S += delta
                sweep_changed = True

            # Track best
            curr_J = 2.0 * n * b.max() / (S * S)
            if curr_J < best_J * (1.0 - 1e-9):
                best_J = curr_J
                best_a = a.copy()
                no_improve_sweeps = 0

        if not sweep_changed:
            no_improve_sweeps += 1
            if no_improve_sweeps >= 5:
                break
        else:
            no_improve_sweeps = 0

            print(f"    CD sweep {sweep}: J={2.0*n*b.max()/(S*S):.8f} best={best_J:.8f}", flush=True)

    best_a = np.maximum(best_a, 0.0)
    s = best_a.sum()
    if s > 1e-10:
        best_a /= s
    return best_a


def smooth_max_grad(a, tau):
    """Compute gradient of (1/tau)*log(sum exp(tau*(a*a))) w.r.t. a."""
    n = len(a)
    b = fast_conv(a)
    b_max = b.max()
    exp_b = np.exp(tau * (b - b_max))
    p = exp_b / exp_b.sum()

    grad = np.zeros(n)
    for k in range(2 * n - 1):
        pk = p[k]
        if pk < 1e-14:
            continue
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        grad[lo:hi + 1] += 2.0 * pk * a[k - hi:k - lo + 1][::-1]
    return grad, p, b


def run_smooth_gradient(a, max_iters, deadline, initial_tau=5.0, final_tau=300.0):
    """Smooth gradient descent with adaptive tau and learning rate."""
    a = np.maximum(a.astype(np.float64), 1e-8)
    a = a / a.sum()
    n = len(a)

    best_a = a.copy()
    best_J = compute_J(a)
    v = np.zeros(n)

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        frac = it / max(1, max_iters)
        tau = initial_tau + (final_tau - initial_tau) * (frac ** 0.6)
        lr = 0.15 * (0.02 / 0.15) ** frac

        grad, p, b = smooth_max_grad(a, tau)
        curr_J = compute_J(a)
        if curr_J < best_J * (1 - 1e-9):
            best_J = curr_J
            best_a = a.copy()

        beta = 0.9
        v = beta * v + lr * grad
        a_new = a - v
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s < 1e-10:
            a_new = a.copy()
        else:
            a_new = a_new / s
        a = a_new

    return best_a


def build_A_sparse(n, a):
    """Build sparse matrix A where A[k,j] = a[k-j] if 0<=k-j<n."""
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        j_min = max(0, k - n + 1)
        j_max = min(n - 1, k)
        for j in range(j_min, j_max + 1):
            rows.append(k)
            cols.append(j)
            vals.append(a[k - j])
    return csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))


def lp_best_response(a, M, timeout=30):
    """LP: max sum(g) s.t. (a*g)[k] <= M for all k, g >= 0."""
    n = len(a)
    A = build_A_sparse(n, a)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    try:
        r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                    options={'time_limit': timeout})
        if r.success and r.x is not None:
            g = np.maximum(r.x, 0)
            if g.sum() > 1e-15:
                return g
    except Exception:
        pass
    return None


def run_lp_mirror(a, max_iters, deadline, timeout=20):
    """LP mirror descent with geometric line search."""
    a = np.maximum(a.astype(np.float64), 1e-8)
    a = a / a.sum()
    best_a = a.copy()
    best_M = fast_conv(a).max()
    no_improve = 0

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        b = fast_conv(a)
        M = float(b.max())
        g = lp_best_response(a, M, timeout)
        if g is None:
            break
        g = np.maximum(g, 0.0)
        g_sum = g.sum()
        if g_sum < 1e-10:
            break
        g /= g_sum

        improved = False
        base_t = max(0.03, 0.5 * (1 - 0.5 * it / max(1, max_iters)))
        for t in [base_t, base_t * 0.5, base_t * 0.25, 0.01, 0.005]:
            a_trial = np.maximum((1 - t) * a + t * g, 0)
            s = a_trial.sum()
            if s < 1e-10:
                continue
            a_trial /= s
            M_trial = fast_conv(a_trial).max()
            if M_trial < best_M - 1e-14:
                a = a_trial
                best_M = M_trial
                best_a = a.copy()
                improved = True
                break

        if not improved:
            no_improve += 1
            if no_improve > 15:
                break
        else:
            no_improve = 0

    return best_a


def _lbb_obj(x, tau=200.0):
    """Smooth max objective for L-BFGS-B."""
    ax = np.maximum(x, 0.0)
    s = ax.sum()
    if s < 1e-8:
        return 1e10
    ax = ax / s
    b = fast_conv(ax)
    b_shifted = b - b.max()
    return float(np.log(np.sum(np.exp(tau * b_shifted))) / tau)


def _lbb_grad(x, tau=200.0):
    """Gradient for L-BFGS-B smooth objective."""
    ax = np.maximum(x, 0.0)
    s = ax.sum()
    if s < 1e-8:
        return np.zeros_like(x)
    ax = ax / s
    n = len(ax)
    b = fast_conv(ax)
    b_max = b.max()
    exp_b = np.exp(tau * (b - b_max))
    p = exp_b / exp_b.sum()

    grad = np.zeros(n)
    for k in range(2 * n - 1):
        pk = p[k]
        if pk < 1e-12:
            continue
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        grad[lo:hi + 1] += 2.0 * pk * ax[k - hi:k - lo + 1][::-1]
    return grad


def generate_gaussian_init(n, sigma_frac, rng):
    """Generate a Gaussian bump centered in the array."""
    t = np.arange(n)
    center = (n - 1) / 2.0
    sigma = sigma_frac * n
    a = np.exp(-(t - center) ** 2 / (2.0 * sigma ** 2))
    # Add some randomness
    if rng is not None:
        a += rng.normal(0, 0.02 * a.max(), n)
    a = np.maximum(a, 1e-6)
    a /= a.sum()
    return a


def generate_raised_cosine_init(n, width_frac, rng):
    """Generate a raised cosine window."""
    t = np.arange(n) / (n - 1)
    center = 0.5
    half_width = width_frac / 2.0
    x = np.abs(t - center) / half_width
    a = np.where(x <= 1, 0.5 * (1 + np.cos(np.pi * x)), 0.0)
    a += 1e-6
    a /= a.sum()
    return a


def propose_candidate(seed=42, budget_s=1500, **kwargs):
    """
    Multi-phase hybrid search with emphasis on fast coordinate descent:
    Phase 1: Coordinate descent on parent (fast convergence)
    Phase 2: Smooth gradient + more coordinate descent
    Phase 3: LP mirror descent for precise refinement
    Phase 4: Multi-length exploration (n=1024, n=768)
    Phase 5: L-BFGS-B + final polishing
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + float(budget_s) - 10.0
    t_start = time.monotonic()

    best_sequence = None
    best_score = float('inf')

    def update_tracker(a, label=""):
        nonlocal best_sequence, best_score
        a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
        if a.size == 0 or a.sum() < 0.01:
            return
        score = float(evaluate_sequence(a.tolist()))
        if np.isfinite(score) and score < best_score:
            best_sequence = a.copy()
            best_score = score
            elapsed = time.monotonic() - t_start
            print(f"  ** NEW BEST: J={score:.8f} n={a.size} t={elapsed:.0f}s {label}", flush=True)

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        if p.ndim == 1 and p.size > 0 and np.all(np.isfinite(p)):
            parent = np.maximum(p, 0.0)
            update_tracker(parent, "(parent)")
            print(f"Parent: n={parent.size}, J={best_score:.6f}", flush=True)
    except (NameError, AttributeError):
        pass

    if parent is None:
        parent = generate_gaussian_init(512, 0.15, rng)
        update_tracker(parent, "(default)")

    # === PHASE 1: Coordinate descent on parent ===
    p1_deadline = t_start + 200.0
    print(f"\n{'='*60}", flush=True)
    print(f"PHASE 1: Coordinate Descent n={parent.size} (budget=200s)", flush=True)
    print(f"{'='*60}", flush=True)

    a = parent.copy()
    a = a / a.sum()
    a = coordinate_descent(a, 100000, min(p1_deadline, deadline - 300), rng, verbose=True)
    update_tracker(a, "(CD phase1)")

    # === PHASE 2: Smooth gradient + CD ===
    p2_deadline = t_start + 350.0
    print(f"\n{'='*60}", flush=True)
    print(f"PHASE 2: Smooth Gradient + CD (budget=150s)", flush=True)
    print(f"{'='*60}", flush=True)

    a = run_smooth_gradient(a, 2000, min(p2_deadline, deadline - 250),
                            initial_tau=5.0, final_tau=200.0)
    update_tracker(a, "(smooth grad)")

    a = coordinate_descent(a, 50000, min(p2_deadline + 100, deadline - 200), rng)
    update_tracker(a, "(CD phase2)")

    # === PHASE 3: LP Mirror Descent ===
    p3_deadline = t_start + 600.0
    print(f"\n{'='*60}", flush=True)
    print(f"PHASE 3: LP Mirror Descent n={a.size} (budget=250s)", flush=True)
    print(f"{'='*60}", flush=True)

    n_cur = a.size
    est_lp_time = 0.005 * (n_cur / 128) ** 2
    max_lp_iters = max(20, min(300, int(250 / max(0.01, est_lp_time))))
    print(f"  n={n_cur}, est LP time={est_lp_time*1000:.0f}ms, max_iters={max_lp_iters}", flush=True)

    a = run_lp_mirror(a, max_lp_iters, min(p3_deadline, deadline - 150), timeout=15)
    update_tracker(a, "(LP)")

    # Quick coordinate descent after LP
    a = coordinate_descent(a, 20000, min(p3_deadline + 50, deadline - 100), rng)
    update_tracker(a, "(CD after LP)")

    # === PHASE 4: Multi-length exploration ===
    print(f"\n{'='*60}", flush=True)
    print(f"PHASE 4: Multi-Length Exploration", flush=True)
    print(f"{'='*60}", flush=True)

    base_n = a.size
    target_lengths = []
    if base_n <= 600:
        target_lengths = [int(base_n * 2), int(base_n * 1.5), int(base_n * 3)]
    else:
        target_lengths = [int(base_n * 2), int(base_n // 2)]
    # Filter to reasonable range
    target_lengths = [n for n in target_lengths if 128 <= n <= 2048]
    # Remove duplicates
    seen = set()
    target_lengths = [n for n in target_lengths if n not in seen and not seen.add(n)]

    remaining = deadline - time.monotonic()
    time_per_len = max(60.0, remaining / max(1, len(target_lengths)) - 100)

    for target_n in target_lengths:
        if time.monotonic() > deadline - 80:
            break
        print(f"\n  Trying n={target_n} (budget={time_per_len:.0f}s)", flush=True)
        len_deadline = min(time.monotonic() + time_per_len, deadline - 80)

        # Initialize from current best (resampled) + Gaussian perturbation
        a_ext = resample_seq(a, target_n)
        a_ext = np.maximum(a_ext, 1e-6)
        a_ext /= a_ext.sum()

        # Also try Gaussian initialization
        a_gauss = generate_gaussian_init(target_n, 0.12, rng)
        a_gauss2 = generate_gaussian_init(target_n, 0.08, rng)
        a_cos = generate_raised_cosine_init(target_n, 0.7, rng)

        candidates = [a_ext, a_gauss, a_gauss2, a_cos]
        best_for_len = None
        best_J_for_len = float('inf')

        for idx, a0 in enumerate(candidates):
            if time.monotonic() > len_deadline - 30:
                break
            a0 = np.maximum(a0, 1e-6)
            a0 = a0 / a0.sum()

            # Coordinate descent (fast)
            cd_time = max(20.0, (len_deadline - time.monotonic()) * 0.5)
            a_try = coordinate_descent(a0, 50000, time.monotonic() + cd_time, rng)
            J_try = compute_J(a_try)

            if J_try < best_J_for_len:
                best_J_for_len = J_try
                best_for_len = a_try.copy()
                print(f"    candidate {idx}: J={J_try:.8f}", flush=True)

        if best_for_len is not None:
            # Polish with smooth gradient
            if time.monotonic() < len_deadline - 20:
                a_try = run_smooth_gradient(best_for_len, 1000, len_deadline - 20,
                                            initial_tau=10, final_tau=150)
                if compute_J(a_try) < best_J_for_len:
                    best_for_len = a_try
                    best_J_for_len = compute_J(a_try)

            # Final CD polish
            a_try = coordinate_descent(best_for_len, 20000, len_deadline, rng)
            if compute_J(a_try) < best_J_for_len:
                best_for_len = a_try
                best_J_for_len = compute_J(a_try)

            update_tracker(best_for_len, f"(n={target_n})")

    # === PHASE 5: L-BFGS-B + Final polish ===
    print(f"\n{'='*60}", flush=True)
    print(f"PHASE 5: L-BFGS-B + Final Polish (remaining time)", flush=True)
    print(f"{'='*60}", flush=True)

    if best_sequence is not None and time.monotonic() < deadline - 30:
        n_best = best_sequence.size
        # L-BFGS-B
        if time.monotonic() < deadline - 60:
            bounds_list = [(0.0, None)] * n_best
            try:
                res = minimize(_lbb_obj, best_sequence.astype(np.float64),
                               jac=_lbb_grad, method='L-BFGS-B',
                               bounds=bounds_list,
                               options={'maxiter': 200, 'max_fun': 20000,
                                        'ftol': 1e-16, 'gtol': 1e-10})
                if res.fun < 1e10:
                    update_tracker(res.x, "(L-BFGS-B)")
            except Exception as e:
                print(f"  L-BFGS-B error: {e}", flush=True)

        # Final coordinate descent
        if time.monotonic() < deadline - 10:
            a = coordinate_descent(best_sequence, 30000, deadline - 5, rng)
            update_tracker(a, "(final CD)")

        # One more LP pass if time allows
        if time.monotonic() < deadline - 5 and best_sequence is not None:
            n_b = best_sequence.size
            est_t = 0.005 * (n_b / 128) ** 2
            if est_t < (deadline - time.monotonic() - 5) * 0.5:
                a = run_lp_mirror(best_sequence, 20, deadline - 2, timeout=5)
                update_tracker(a, "(final LP)")

    # Fallback
    if best_sequence is None:
        best_sequence = parent
        best_score = compute_J(best_sequence)

    print(f"\n{'='*60}", flush=True)
    print(f"FINAL: J={best_score:.8f} n={best_sequence.size} "
          f"time={time.monotonic()-t_start:.0f}s", flush=True)
    print(f"{'='*60}", flush=True)

    return [float(x) for x in best_sequence]
```
