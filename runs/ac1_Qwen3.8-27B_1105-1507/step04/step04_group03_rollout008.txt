def lp_best_response(a, M):
    """Solve LP: max sum(g) s.t. A@g <= M*1, g >= 0."""
    n = len(a)
    A = build_A_sparse(n, a)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    try:
        r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
        if r.success and r.x is not None:
            g = np.maximum(r.x, 0)
            if g.sum() > 1e-15:
                return g
    except Exception:
        pass
    return None


def smooth_max_obj(x, beta):
    """Log-sum-exp approximation of max(conv(x,x)). Returns (value, grad)."""
    a = np.maximum(x, 0)
    s = a.sum()
    if s < 1e-8:
        return 1e10, np.zeros_like(x)
    a = a / s
    b = fast_conv(a)
    # Numerically stable log-sum-exp
    b_max = b.max()
    if b_max < 0:
        b_max = 0
    scaled = beta * (b - b_max)
    # Clip for numerical stability
    scaled = np.clip(scaled, -500, 500)
    exp_s = np.exp(scaled)
    log_sum = b_max + np.log(exp_s.sum()) / beta
    # Gradient
    p = exp_s / exp_s.sum()  # softmax weights
    n = len(a)
    grad = np.zeros(n)
    # grad_m = 2 * sum_j a_j * p_{m+j}
    for j in range(n):
        if a[j] > 1e-15:
            grad += 2.0 * a[j] * p[j:j + n]
    # Scale by 1/beta and normalize (since we normalized a)
    grad /= beta
    return float(log_sum), grad

def lp_mirror_descent_v2(a, max_iters, deadline, mode='auto', m_shrink=0.0):
    """
    Enhanced LP mirror descent.
    mode: 'linear', 'multiplicative', or 'auto' (alternates)
    m_shrink: fraction to shrink M (0 = no shrink, 0.05 = use 95% of M)
    """
    a = np.maximum(a.astype(np.float64), 1e-12)
    a = a / a.sum()
    current_J = compute_J(a)
    best_J = current_J
    best_a = a.copy()
    no_improve = 0
    use_mult = (mode == 'multiplicative')
    if mode == 'auto':
        use_mult = False

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        b = fast_conv(a)
        M = float(b.max())

        # M-shrinking: occasionally use smaller M for aggressive updates
        M_eff = M * (1.0 - m_shrink) if m_shrink > 0 else M
        if M_eff <= 0:
            M_eff = M

        g = lp_best_response(a, M_eff)
        if g is None:
            if m_shrink > 0:
                g = lp_best_response(a, M)
                if g is None:
                    break
            else:
                break
        g = np.maximum(g, 0.0)
        g_sum = g.sum()
        if g_sum < 1e-10:
            no_improve += 1
            if no_improve > 30:
                break
            continue
        g = g / g_sum

        # Adaptive step size
        if it < 50:
            base_t = 0.35
        elif it < 150:
            base_t = 0.25
        elif it < 300:
            base_t = 0.15
        else:
            base_t = 0.08
        t = base_t

        # In auto mode, switch between linear and multiplicative every 50 iters
        if mode == 'auto' and it % 50 == 0:
            use_mult = not use_mult

        best_trial_J = current_J
        best_trial_a = None

        for ls in range(30):
            if use_mult:
                ratio = g / np.maximum(a, 1e-15)
                a_trial = a * np.power(ratio, t)
            else:
                a_trial = (1.0 - t) * a + t * g
            a_trial = np.maximum(a_trial, 0.0)
            s = a_trial.sum()
            if s < 1e-10:
                t *= 0.5
                continue
            a_trial = a_trial / s
            J_trial = compute_J(a_trial)
            if J_trial < best_trial_J - 1e-13:
                best_trial_J = J_trial
                best_trial_a = a_trial.copy()
                break
            t *= 0.6
            if t < 0.005:
                break

        if best_trial_a is not None:
            a = best_trial_a
            current_J = best_trial_J
            if current_J < best_J - 1e-12:
                best_J = current_J
                best_a = a.copy()
                no_improve = 0
            else:
                no_improve += 1
        else:
            no_improve += 1
            if no_improve > 50:
                break

    return best_a, best_J


def smooth_optimize(a, deadline, max_betas=5, initial_beta=5.0):
    """
    Smooth optimization using log-sum-exp with increasing beta.
    Returns best (a, J) found.
    """
    a = np.maximum(a.astype(np.float64), 1e-10)
    a = a / a.sum()
    best_a = a.copy()
    best_J = compute_J(a)

    betas = [initial_beta * (2.0 ** i) for i in range(max_betas)]
    betas = [b for b in betas if b <= 200.0]

    for beta in betas:
        if time.monotonic() > deadline:
            break

        def obj_and_grad(x):
            a_x = np.maximum(x, 0)
            s = a_x.sum()
            if s < 1e-8:
                return 1e10, np.zeros_like(x)
            a_x = a_x / s
            b = fast_conv(a_x)
            b_max = b.max()
            if b_max < 0:
                b_max = 0
            scaled = beta * (b - b_max)
            scaled = np.clip(scaled, -500, 500)
            exp_s = np.exp(scaled)
            val = b_max + np.log(exp_s.sum()) / beta
            p = exp_s / exp_s.sum()
            n = len(a_x)
            grad = np.zeros(n)
            for j in range(n):
                if a_x[j] > 1e-15:
                    grad += 2.0 * a_x[j] * p[j:j + n]
            grad /= beta
            return float(val), grad

        n = len(a)
        try:
            res = minimize(
                lambda x: obj_and_grad(x)[0],
                a.copy(),
                jac=lambda x: obj_and_grad(x)[1],
                method='L-BFGS-B',
                bounds=[(0.0, None)] * n,
                options={'maxiter': 200, 'max_fun': 10000, 'ftol': 1e-16, 'gtol': 1e-14}
            )
            if res.fun < 1e9:
                a_opt = np.maximum(res.x, 0)
                s = a_opt.sum()
                if s > 1e-8:
                    a_opt = a_opt / s
                    J_opt = compute_J(a_opt)
                    if J_opt < best_J:
                        best_J = J_opt
                        best_a = a_opt.copy()
        except Exception:
            pass

    return best_a, best_J


def projected_subgradient_nesterov(a, max_iters, deadline, lr_init=0.01):
    """
    Projected subgradient with Nesterov acceleration on smooth max approx.
    """
    a = np.maximum(a.astype(np.float64), 1e-10)
    a = a / a.sum()
    best_a = a.copy()
    best_J = compute_J(a)
    v = np.zeros_like(a)  # momentum
    a_prev = a.copy()
    beta = 20.0  # fixed beta for subgradient

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        # Nesterov look-ahead
        gamma = (1.0 + np.sqrt(1.0 + 4.0 * it * it)) / 2.0
        a_look = np.maximum((1 + 1.0 / gamma) * a - (1.0 / gamma) * a_prev, 0)
        s = a_look.sum()
        if s < 1e-8:
            break
        a_look = a_look / s

        # Compute gradient of smooth max at a_look
        b = fast_conv(a_look)
        b_max = b.max()
        if b_max < 0:
            b_max = 0
        scaled = beta * (b - b_max)
        scaled = np.clip(scaled, -500, 500)
        exp_s = np.exp(scaled)
        p = exp_s / exp_s.sum()
        n = len(a_look)
        grad = np.zeros(n)
        for j in range(n):
            if a_look[j] > 1e-15:
                grad += 2.0 * a_look[j] * p[j:j + n]
        grad /= beta

        # Learning rate
        lr = lr_init / (1.0 + 0.05 * it)

        # Update
        v = 0.9 * v + (1.0 - 0.9) * grad
        a_new = a - lr * v
        a_new = np.maximum(a_new, 0)
        s = a_new.sum()
        if s < 1e-8:
            break
        a_new = a_new / s

        J_new = compute_J(a_new)
        if J_new < best_J:
            best_J = J_new
            best_a = a_new.copy()

        a_prev = a
        a = a_new

    return best_a, best_J


def resample_seq(src, dest_n):
    """Resample 1D sequence to new length."""
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)


def perturb_seq(a, rng, ptype):
    """Apply perturbation to sequence."""
    n = len(a)
    t = np.arange(n, dtype=np.float64) / n

    if ptype == 0:  # circulant shift
        s = int(rng.integers(1, max(2, n // 8)))
        a_p = np.roll(a, s)
    elif ptype == 1:  # cosine modulation
        freq = rng.uniform(0.5, 6.0)
        amp = rng.uniform(0.03, 0.15)
        a_p = a * (1 + amp * np.cos(2 * np.pi * freq * t + rng.uniform(0, 2 * np.pi)))
    elif ptype == 2:  # smoothing
        k = int(rng.integers(3, max(4, n // 8)))
        kernel = np.ones(k) / k
        a_p = np.convolve(a, kernel, mode='same')
    elif ptype == 3:  # Gaussian perturbation
        sigma = rng.uniform(0.01, 0.08)
        a_p = a + rng.normal(0, sigma, n)
    elif ptype == 4:  # taper edges
        alpha = rng.uniform(0.3, 0.8)
        edge = max(1, int(n * (1 - alpha) / 2))
        window = np.ones(n)
        for i in range(edge):
            frac = (i + 1) / (edge + 1)
            window[i] = frac
            window[n - 1 - i] = frac
        a_p = a * window
    elif ptype == 5:  # frequency domain perturbation
        fa = np.fft.rfft(a)
        mask = np.ones_like(fa)
        n_mask = int(rng.integers(1, max(2, len(fa) // 4)))
        idx = rng.choice(len(fa), n_mask, replace=False)
        mask[idx] = rng.uniform(0.5, 1.5, n_mask)
        a_p = np.fft.irfft(fa * mask, n)
    else:
        a_p = a.copy()

    a_p = np.maximum(a_p, 0.001)
    s = a_p.sum()
    if s > 1e-8:
        a_p = a_p / s
    return a_p


class Tracker:
    def __init__(self):
        self.best = None
        self.best_score = float('inf')
        self.t_start = time.monotonic()

    def update(self, a):
        a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
        if a.size == 0 or a.sum() < 0.01:
            return
        score = float(evaluate_sequence(a.tolist()))
        if np.isfinite(score) and score < self.best_score:
            self.best = a.copy()
            self.best_score = score
            elapsed = time.monotonic() - self.t_start
            print(f"  [NEW BEST] J={score:.8f} n={a.size} t={elapsed:.0f}s", flush=True)

    def report(self):
        if self.best is not None:
            elapsed = time.monotonic() - self.t_start
            print(f"  Current best: J={self.best_score:.8f} n={self.best.size} t={elapsed:.0f}s", flush=True)
