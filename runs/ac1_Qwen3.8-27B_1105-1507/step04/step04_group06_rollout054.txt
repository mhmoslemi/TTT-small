def get_A_structure(n):
    """Get or pre-compute the (row, col) arrays for the constraint matrix."""
    if n not in _A_structure_cache:
        rows = []
        cols = []
        for k in range(2 * n - 1):

            jh = min(n - 1, k)
            for j in range(jl, jh + 1):
                rows.append(k)
                cols.append(j)
        _A_structure_cache[n] = (np.array(rows, dtype=np.int32),

def build_A_sparse(a):
    """Build sparse constraint matrix for LP."""
    n = len(a)
    rows_arr, cols_arr = get_A_structure(n)
    vals = a[rows_arr - cols_arr]
    return csr_matrix((vals, (rows_arr, cols_arr)), shape=(2 * n - 1, n))


def lp_best_response(a, M):
    """LP: max sum(g) s.t. (a*g)_k <= M for all k, g >= 0. Returns normalized g or None."""
    n = len(a)
    c = -np.ones(n)
    A = build_A_sparse(a)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    try:
        r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
        if r.success and r.x is not None:
            g = np.maximum(r.x, 0.0)
            g_sum = g.sum()
            if g_sum > 1e-10:
                return g / g_sum
    except Exception:
        pass
    return None

    hi = 1.0
    for _ in range(n_evals):
        t1 = hi - gr * (hi - lo)
        t2 = lo + gr * (hi - lo)
        a1 = (1.0 - t1) * a + t1 * g
        a2 = (1.0 - t2) * a + t2 * g
        J1 = compute_J(a1)
        J2 = compute_J(a2)
        if J1 < J2:
            hi = t2
        else:
            lo = t1
        t_mid = (lo + hi) / 2.0
        a_mid = (1.0 - t_mid) * a + t_mid * g
        J_mid = compute_J(a_mid)
        if J_mid < best_J:
            best_J = J_mid
            best_a = a_mid
            best_t = t_mid

    return best_a, best_J


def lp_mirror_descent(a, max_iters, deadline, tracker=None, n_line_evals=18,
                      stall_patience=30):
    """LP-based mirror descent with line search."""
    a = np.maximum(a.astype(np.float64), 1e-8)
    a = a / a.sum()
    current_J = compute_J(a)
    if tracker is not None:
        tracker.update(a)

    stall_count = 0
    best_J_so_far = current_J
    best_a_so_far = a.copy()

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        b = fast_conv(a)
        M = float(b.max())

        g = lp_best_response(a, M)
        if g is None:
            break

        a_new, J_new = line_search_convex(a, g, n_evals=n_line_evals)

        if J_new < current_J - 1e-13:
            a = a_new
            current_J = J_new
            stall_count = 0
            if J_new < best_J_so_far:
                best_J_so_far = J_new
                best_a_so_far = a.copy()
                if tracker is not None:
                    tracker.update(a)
        else:
            # Try smaller steps
            improved = False
            for t_try in [0.01, 0.005, 0.02, 0.05, 0.1]:
                a_try = (1.0 - t_try) * a + t_try * g
                a_try = np.maximum(a_try, 0.0)
                s = a_try.sum()
                if s < 1e-10:
                    continue
                a_try = a_try / s
                J_try = compute_J(a_try)
                if J_try < current_J - 1e-14:
                    a = a_try
                    current_J = J_try
                    stall_count = 0
                    improved = True
                    if J_try < best_J_so_far:
                        best_J_so_far = J_try
                        best_a_so_far = a.copy()
                        if tracker is not None:
                            tracker.update(a)
                    break
            if not improved:
                stall_count += 1
                if stall_count >= stall_patience:
                    break

        if tracker is not None and it % 100 == 0:
            tracker.update(a)

    if tracker is not None:
        tracker.update(best_a_so_far)
    return best_a_so_far


def generate_initializations(n, rng, parent=None):
    """Generate diverse initial sequences of length n."""
    inits = []
    t_arr = np.linspace(0, 1, n, endpoint=False)

    if parent is not None and parent.size > 0:
        p = resample_seq(parent, n)
        p = np.maximum(p, 1e-6)
        inits.append(p / p.sum())
        # Perturbed parent
        p2 = p + rng.normal(0, 0.05, n) * p
        p2 = np.maximum(p2, 0.01)
        inits.append(p2 / p2.sum())

    # Uniform
    inits.append(np.ones(n) / n)

    # Random
    r = rng.uniform(0.3, 1.0, n)
    inits.append(r / r.sum())

    # Cosine families
    for freq in [2, 3, 4, 5, 7, 11]:
        c = 1.0 + 0.4 * np.cos(2.0 * np.pi * freq * t_arr)
        c = np.maximum(c, 0.01)
        inits.append(c / c.sum())

    # Gaussian bump
    g = np.exp(-3.0 * (t_arr - 0.5) ** 2) + 0.1
    inits.append(g / g.sum())

    # Sine absolute
    s = np.abs(np.sin(np.pi * t_arr)) + 0.2
    inits.append(s / s.sum())

    # Two-peak
    tp = 1.0 + 0.4 * np.cos(4.0 * np.pi * t_arr) + 0.15 * np.cos(8.0 * np.pi * t_arr)
    tp = np.maximum(tp, 0.01)
    inits.append(tp / tp.sum())

    # Power law
    pl = (t_arr + 0.1) ** 0.5
    inits.append(pl / pl.sum())

    # Exponential decay
    ed = np.exp(-2.0 * t_arr) + 0.1
    inits.append(ed / ed.sum())

    # Block structure (alternating high/low)
    blk = np.ones(n)
    blk[::2] = 0.3
    inits.append(blk / blk.sum())

    # Sparse peaks
    sp = np.ones(n) * 0.1
    n_peaks = max(4, n // 32)
    peak_idx = rng.choice(n, size=n_peaks, replace=False)
    sp[peak_idx] = 1.0
    inits.append(sp / sp.sum())

    return inits


def resample_seq(src, dest_n):
    """Resample 1D sequence to new length via linear interpolation."""
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    if src.size == 0:
        return np.ones(dest_n) / dest_n
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    result = np.interp(t_new, t_old, src)
    return np.maximum(result, 0.0)


class ScoreTracker:
    """Tracks the best sequence found so far."""

    def __init__(self):
        self.best = None
        self.best_score = float('inf')
        self.t_start = time.monotonic()

    def update(self, a):
        a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
        if a.size == 0 or a.sum() < 0.01:
            return
        score = float(evaluate_sequence(a.tolist()))
        if np.isfinite(score) and score < self.best_score:
            self.best = a.copy()
            self.best_score = score
            elapsed = time.monotonic() - self.t_start
            print(f"  * NEW BEST: J={score:.8f} n={a.size} t={elapsed:.1f}s", flush=True)

    def report(self, label=""):
        if self.best is not None:
            print(f"  {label} best: J={self.best_score:.8f} n={self.best.size}", flush=True)


def propose_candidate(seed=42, budget_s=1500, **kwargs):
    """
    Multi-phase optimization for minimizing J(a) = 2n*max(conv(a,a))/(sum(a))^2.

    Phase 1: Softmax+Adam with log-sum-exp smoothing (fast basin finding, ~500s)
    Phase 2: LP mirror descent polish (main refinement, ~500s)
    Phase 3: Multi-restart with diverse perturbations (~300s)
    Phase 4: Final LP polish + length exploration (~150s)
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + float(budget_s) - 15.0
    tracker = ScoreTracker()

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        if p.ndim == 1 and p.size > 0 and np.all(np.isfinite(p)):
            parent = np.maximum(p, 0.0)
            if parent.sum() > 0.01:
                tracker.update(parent)
            print(f"Parent loaded: n={parent.size}, J={tracker.best_score:.8f}", flush=True)
    except (NameError, AttributeError):
        pass

    if parent is None or parent.size == 0:
        parent = np.ones(1024)
        tracker.update(parent)
        print(f"Using default parent: n=1024, J={tracker.best_score:.8f}", flush=True)

    n_parent = parent.size

    # === Phase 1: Softmax + Adam with beta annealing ===
    t0 = time.monotonic()
    phase1_budget = 450.0
    deadline_p1 = time.monotonic() + phase1_budget
    print(f"\n{'=' * 60}", flush=True)
    print(f"Phase 1: Softmax+Adam (budget {phase1_budget:.0f}s)", flush=True)
    print(f"{'=' * 60}", flush=True)

    # Try multiple lengths
    lengths_p1 = sorted(set([n_parent, 512, 1024, 2048]))
    lengths_p1 = [l for l in lengths_p1 if 64 <= l <= 2048]

    # Beta schedule: progressive sharpening
    beta_schedule_full = [
        (1.0, 300),
        (2.0, 300),
        (5.0, 300),
        (10.0, 300),
        (25.0, 300),
        (50.0, 300),
        (100.0, 400),
    ]

    # Shorter schedule for non-primary lengths
    beta_schedule_short = [
        (1.0, 200),
        (5.0, 200),
        (15.0, 200),
        (50.0, 300),
    ]

    for n_len in lengths_p1:
        if time.monotonic() > deadline_p1:
            break

        t_start = time.monotonic()
        is_primary = (n_len == n_parent)
        schedule = beta_schedule_full if is_primary else beta_schedule_short

        print(f"\n  Optimizing n={n_len} ({'primary' if is_primary else 'secondary'})...", flush=True)

        inits = generate_initializations(n_len, rng, parent)
        # Limit number of initializations based on time
        max_inits = 5 if is_primary else 3
        # Always include the (resampled) parent as first init
        inits_sorted = sorted(inits, key=lambda x: compute_J(x))
        inits_use = inits_sorted[:max_inits]

        best_a_for_len = None
        best_J_for_len = float('inf')

        for idx, a0 in enumerate(inits_use):
            if time.monotonic() > deadline_p1:
                break

            a0 = np.maximum(a0, 1e-8)
            a0 = a0 / a0.sum()

            # Determine how many steps to give this init
            remaining_frac = max(0.1, 1.0 - (time.monotonic() - t_start) / phase1_budget * len(inits_use))
            adj_schedule = [(b, int(s * remaining_frac)) for b, s in schedule]

            a_opt = adam_softmax_optimize(a0, adj_schedule, deadline_p1,
                                          lr_init=0.3, beta1=0.9, beta2=0.999)
            if a_opt is not None and a_opt.sum() > 0.01:
                tracker.update(a_opt)
                J_opt = compute_J(a_opt)
                if J_opt < best_J_for_len:
                    best_J_for_len = J_opt
                    best_a_for_len = a_opt.copy()

        print(f"    n={n_len} done in {time.monotonic()-t_start:.1f}s, best_J={best_J_for_len:.8f}", flush=True)

    tracker.report("Phase 1 complete")

    # === Phase 2: LP mirror descent polish ===
    if tracker.best is not None and time.monotonic() < deadline:
        t0 = time.monotonic()
        n_best = tracker.best.size
        phase2_budget = 500.0
        deadline_p2 = min(deadline, time.monotonic() + phase2_budget)
        print(f"\n{'=' * 60}", flush=True)
        print(f"Phase 2: LP Mirror Descent n={n_best} (budget {phase2_budget:.0f}s)", flush=True)
        print(f"{'=' * 60}", flush=True)

        # Primary run at best length
        remaining = deadline_p2 - time.monotonic()
        max_iters = int(remaining * 40)  # ~40 LP iters/sec estimated
        max_iters = min(max_iters, 8000)
        a = lp_mirror_descent(tracker.best, max_iters, deadline_p2, tracker=tracker,
                              n_line_evals=20, stall_patience=40)

        # Try at a few other lengths
        for n_try in [512, 1024, 2048]:
            if time.monotonic() > deadline_p2:
                break
            if n_try == n_best:
                continue
            a_res = resample_seq(tracker.best, n_try)
            a_res = np.maximum(a_res, 1e-8)
            a_res = a_res / a_res.sum()
            budget_here = min(150.0, deadline_p2 - time.monotonic())
            if budget_here < 10:
                continue
            max_iters_here = int(budget_here * 30)
            max_iters_here = min(max_iters_here, 4000)
            a_res = lp_mirror_descent(a_res, max_iters_here, deadline_p2, tracker=tracker,
                                      n_line_evals=15, stall_patience=30)

        tracker.report(f"Phase 2: {time.monotonic()-t0:.1f}s")

    # === Phase 3: Multi-restart with diverse perturbations ===
    if tracker.best is not None and time.monotonic() < deadline:
        t0 = time.monotonic()
        n_best = tracker.best.size
        phase3_budget = 350.0
        deadline_p3 = min(deadline, time.monotonic() + phase3_budget)
        print(f"\n{'=' * 60}", flush=True)
        print(f"Phase 3: Multi-Restart (budget {phase3_budget:.0f}s)", flush=True)
        print(f"{'=' * 60}", flush=True)

        # Define perturbation strategies
        strategies = []
        # Gaussian perturbations of various scales
        for scale in [0.03, 0.05, 0.08, 0.12, 0.2]:
            strategies.append(('gaussian', scale))
        # Smooth noise
        for scale in [0.05, 0.1, 0.15]:
            strategies.append(('smooth', scale))
        # Spike perturbations
        for k in [4, 8, 16]:
            strategies.append(('spike', k))
        # Block perturbations
        strategies.append(('block', 0.15))
        # Phase shift in frequency domain
        strategies.append(('phase', 0.3))

        attempt = 0
        for pert_type, param in strategies:
            for trial in range(4):
                if time.monotonic() > deadline_p3:
                    break
                attempt += 1
                a = tracker.best.copy().astype(np.float64)
                n = len(a)

                if pert_type == 'gaussian':
                    a = a + rng.normal(0, param, n) * a
                elif pert_type == 'smooth':
                    noise = rng.normal(0, param, n)
                    kernel_size = max(3, n // 64)
                    kernel = np.ones(kernel_size) / kernel_size
                    noise = np.convolve(noise, kernel, mode='same')
                    a = a + noise * a
                elif pert_type == 'spike':
                    k = min(param, n // 4)
                    idx = rng.choice(n, size=k, replace=False)
                    a[idx] *= rng.uniform(0.3, 3.0, k)
                elif pert_type == 'block':
                    # Perturb a random block
                    block_size = max(16, n // 8)
                    start = rng.integers(0, n - block_size)
                    a[start:start + block_size] *= rng.uniform(0.5, 1.5, block_size)
                elif pert_type == 'phase':
                    # Circular shift in frequency domain
                    fa = np.fft.rfft(a)
                    angles = np.linspace(0, 2 * np.pi, len(fa), endpoint=False)
                    shift = rng.uniform(-param, param)
                    fa_phase = fa * np.exp(1j * shift * angles)
                    a = np.fft.irfft(fa_phase, n)

                a = np.maximum(a, 1e-6)
                a = a / a.sum()

                # Quick check
                J_pert = compute_J(a)
                if J_pert < tracker.best_score - 1e-6:
                    tracker.update(a)
                    print(f"    Perturbed directly improved! J={J_pert:.8f}", flush=True)

                # Run Adam refinement (fast)
                beta_sched_quick = [(5.0, 100), (20.0, 100), (50.0, 100)]
                budget_adam = min(60.0, deadline_p3 - time.monotonic())
                if budget_adam > 5:
                    a_adam = adam_softmax_optimize(a, beta_sched_quick,
                                                   time.monotonic() + budget_adam,
                                                   lr_init=0.3)
                    if a_adam is not None:
                        tracker.update(a_adam)

                # Run LP polish (slower but more precise)
                budget_lp = min(40.0, deadline_p3 - time.monotonic())
                if budget_lp > 5:
                    start_a = tracker.best if tracker.best.size == a.size else a
                    start_a = np.maximum(start_a, 1e-8)
                    start_a = start_a / start_a.sum()
                    max_iters = int(budget_lp * 30)
                    max_iters = min(max_iters, 1000)
                    a_lp = lp_mirror_descent(start_a, max_iters, time.monotonic() + budget_lp,
                                             tracker=tracker, n_line_evals=12,
                                             stall_patience=20)

        tracker.report(f"Phase 3: {time.monotonic()-t0:.1f}s, {attempt} attempts")

    # === Phase 4: Final polish + length exploration ===
    if tracker.best is not None and time.monotonic() < deadline:
        t0 = time.monotonic()
        n_best = tracker.best.size
        remaining = deadline - time.monotonic()
        if remaining > 20:
            print(f"\n{'=' * 60}", flush=True)
            print(f"Phase 4: Final Polish (budget {remaining:.0f}s)", flush=True)
            print(f"{'=' * 60}", flush=True)

            # Extended LP on best
            max_iters = int(remaining * 0.5 * 35)
            max_iters = min(max_iters, 5000)
            deadline_p4a = time.monotonic() + remaining * 0.5
            a = lp_mirror_descent(tracker.best, max_iters, deadline_p4a, tracker=tracker,
                                  n_line_evals=22, stall_patience=60)

            # Try adjacent lengths
            remaining2 = deadline - time.monotonic()
            if remaining2 > 15:
                candidate_lengths = [n_best - 64, n_best + 64, n_best - 128,
                                     n_best + 128, 768, 1152, 1536]
                candidate_lengths = [l for l in candidate_lengths
                                     if 64 <= l <= 3000 and abs(l - n_best) >= 32]
                for n_try in candidate_lengths[:4]:
                    if time.monotonic() > deadline:
                        break
                    budget_here = min(20.0, deadline - time.monotonic())
                    if budget_here < 5:
                        break
                    a_res = resample_seq(tracker.best, n_try)
                    a_res = np.maximum(a_res, 1e-8)
                    a_res = a_res / a_res.sum()
                    max_iters = int(budget_here * 25)
                    max_iters = min(max_iters, 800)
                    a_res = lp_mirror_descent(a_res, max_iters, deadline, tracker=tracker,
                                              n_line_evals=12, stall_patience=20)

            tracker.report(f"Phase 4: {time.monotonic()-t0:.1f}s")

    # Fallback
    if tracker.best is None:
        tracker.best = np.ones(512)
        tracker.best_score = compute_J(tracker.best)
        print(f"  Using fallback: J={tracker.best_score:.8f}", flush=True)

    print(f"\n{'=' * 60}", flush=True)
    print(f"FINAL RESULT", flush=True)
    print(f"J={tracker.best_score:.8f} n={tracker.best.size} "
          f"time={time.monotonic()-tracker.t_start:.1f}s", flush=True)
    print(f"{'=' * 60}", flush=True)

    # Ensure valid output
    result = np.clip(tracker.best, 0.0, 1000.0)
    if result.sum() < 0.01:
        result = np.ones(128)

    return [float(x) for x in result]
