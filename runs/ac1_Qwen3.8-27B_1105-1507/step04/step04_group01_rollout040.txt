Coordinate descent + LP alternating + L-BFGS-B hybrid optimizer
for minimizing 2*n*max(conv(a,a))/(sum(a))^2.
Uses the structure of the 1-D subproblem in coordinate descent
for fast, precise local optimization.

import numpy as np
from scipy.optimize import minimize, minimize_scalar, linprog
from scipy.fft import rfft, irfft
import time


def compute_conv(a):
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    conv = irfft(fa * fa, size)
    return conv[:2 * n - 1]


def compute_obj(a):
    n = len(a)
    if n == 0:
        return np.inf
    conv = compute_conv(a)
    max_b = float(np.max(conv))
    sum_a = float(np.sum(a))
    if sum_a < 0.01:
        return np.inf
    return 2.0 * n * max_b / (sum_a * sum_a)


def fft_convolve(a, b):
    n, m = len(a), len(b)
    size = 1
    while size < n + m - 1:
        size <<= 1
    fa = rfft(a, size)
    fb = rfft(b, size)
    return irfft(fa * fb, size)[:n + m - 1]

def coord_optim(a, b, i):
    """Optimize coordinate i of a given current convolution b. Returns new value of a[i]."""
    n = len(a)
    old_val = a[i]
    S = float(np.sum(a))
    S_prime = S - old_val
    
    m_low = i
    m_high = min(i + n - 1, 2 * n - 1 - 1)  # 2n-2
    
    if m_low > m_high:
        return old_val
    
    affected = np.arange(m_low, m_high + 1)
    num_affected = len(affected)
    
    # M_unchanged: max of b outside affected range
    if m_low > 0 or m_high < 2 * n - 2:
        mask = np.ones(2 * n - 1, dtype=bool)
        mask[m_low:m_high + 1] = False
        M_unch = float(np.max(b[mask])) if np.any(mask) else 0.0
    else:
        M_unch = 0.0
    
    # C and coeff for affected positions
    C = np.zeros(num_affected)
    coeff = np.zeros(num_affected)  # linear coefficient of t
    quad_mask = np.zeros(num_affected, dtype=bool)  # True where it's t^2
    
    for idx, m in enumerate(affected):
        if m == 2 * i:
            C[idx] = b[m] - old_val * old_val
            coeff[idx] = 0.0
            quad_mask[idx] = True
        else:
            C[idx] = b[m] - 2.0 * old_val * a[m - i]
            coeff[idx] = 2.0 * a[m - i]
    
    def h_func(t):
        vals = C + t * coeff
        vals[quad_mask] = C[quad_mask] + t * t
        new_max = max(M_unch, float(np.max(vals)))
        denom = (S_prime + t) ** 2
        if denom < 1e-30:
            return np.inf
        return 2.0 * n * new_max / denom
    
    # Evaluate at strategic points
    candidates = [0.0, old_val]
    # Add some points around old_val
    candidates.extend([old_val * 0.5, old_val * 1.5, old_val * 0.1, old_val * 2.0])
    candidates = [max(0.0, c) for c in candidates]
    
    best_val = np.inf
    best_t = old_val
    for t in candidates:

    return best_a


def softmax(z):
    z = z - np.max(z)
    e = np.exp(z)
    return e / np.sum(e)


def soft_max_obj_and_grad_z(z, tau):
    n = len(z)
    a = softmax(z)
    conv = compute_conv(a)
    cmax = float(np.max(conv))
    shifted = np.clip((conv - cmax) / tau, -500.0, 500.0)
    exp_conv = np.exp(shifted)
    Z = float(np.sum(exp_conv))
    soft_max_val = cmax + tau * np.log(Z)
    w = exp_conv / Z

    w_rev = w[::-1]
    conv_fw = fft_convolve(a, w_rev)
    f1 = conv_fw[n - 1:2 * n - 1][::-1]
    grad_a = 2.0 * f1

    mean_grad = float(np.dot(a, grad_a))
    grad_z = a * (grad_a - mean_grad)
    return soft_max_val, grad_z


def optimize_lbfgs(a_init, tau_list, max_iter=200, time_budget=20.0):
    n = len(a_init)
    a = a_init.copy()
    s = np.sum(a)
    if s < 1e-10:
        a = np.ones(n) / n
    else:
        a = a / s
    a = np.maximum(a, 1e-30)
    z = np.log(a)

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    for tau in tau_list:
        if time.monotonic() - start > time_budget:
            break

        def func_and_grad(z_val):
            return soft_max_obj_and_grad_z(z_val, tau)

        try:
            result = minimize(func_and_grad, z, jac=True, method='L-BFGS-B',
                              options={'maxiter': max_iter, 'ftol': 1e-15, 'gtol': 1e-9})
            z_new = result.x
        except Exception:
            continue

        a_new = softmax(z_new)
        a_new = np.maximum(a_new, 0)
        s_new = np.sum(a_new)
        if s_new < 1e-10:
            continue
        a_new = a_new / s_new
        obj = compute_obj(a_new)

        if obj < best_obj:
            best_obj = obj
            best_a = a_new.copy()
            z = z_new

    return best_a, best_obj


def alternating_lp(a, max_iters=40, time_budget=40.0):
    n = len(a)
    a = a.copy().astype(np.float64)
    S = np.sum(a)
    if S < 1e-10:
        return a
    a = a / S

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    stall_count = 0
    patience = 25

    for it in range(max_iters):
        if time.monotonic() - start > time_budget:
            break

        conv = compute_conv(a)
        M = float(np.max(conv))

        g, _ = solve_lp_g(a, M, time_limit=8)
        if g is None:
            break

        best_t_obj = best_obj
        best_a_new = best_a.copy()
        for t in [0.05, 0.1, 0.2, 0.3, 0.5, 0.7, 1.0]:
            a_try = (1.0 - t) * a + t * g
            a_try = np.maximum(a_try, 0)
            s_try = np.sum(a_try)
            if s_try < 1e-10:
                continue
            a_try = a_try / s_try
            obj_try = compute_obj(a_try)
            if obj_try < best_t_obj:
                best_t_obj = obj_try
                best_a_new = a_try.copy()

        if best_t_obj < best_obj - 1e-13:
            best_obj = best_t_obj
            best_a = best_a_new.copy()
            a = best_a_new
            stall_count = 0
        else:
            stall_count += 1
            if stall_count > patience:
                rng = np.random.default_rng(it * 7919 + 12345)
                a = a * np.exp(rng.normal(0, 0.04, n))
                a = np.maximum(a, 0)
                s = np.sum(a)
                if s > 1e-10:
                    a = a / s
                stall_count = 0
                patience = 40
            else:
                a = 0.9 * a + 0.1 * g
                a = np.maximum(a, 0)
                s = np.sum(a)
                if s > 1e-10:
                    a = a / s

    return best_a


def resample_sequence(a, new_n):
    n = len(a)
    if n == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return a_new


def try_parent():
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64)
        parent = np.clip(parent, 0, 1000.0)
        if parent.size > 0 and np.isfinite(parent).all():
            obj = compute_obj(parent)
            print(f"Parent (n={parent.size}): obj={obj:.6f}")
            return parent.copy(), obj
    except Exception:
        pass
    return None, np.inf


def random_init(n, seed):
    rng = np.random.default_rng(seed)
    k = np.arange(n)
    choice = rng.integers(0, 8)

    if choice == 0:
        f = rng.uniform(0.5, 4.0)
        phase = rng.uniform(0, 2 * np.pi)
        a = 1.0 + 0.7 * np.cos(2 * np.pi * f * k / n + phase)
    elif choice == 1:
        f1 = rng.uniform(0.3, 2.0)
        f2 = rng.uniform(2.0, 6.0)
        a = 1.0 + 0.4 * np.cos(2 * np.pi * f1 * k / n) + 0.3 * np.cos(2 * np.pi * f2 * k / n)
    elif choice == 2:
        a = rng.random(n)
        win = max(3, n // 15)
        kernel = np.ones(win) / win
        a = np.convolve(a, kernel, mode='same')
    elif choice == 3:
        a = np.ones(n)
        for _ in range(rng.integers(1, 4)):
            f = rng.uniform(0.5, 5.0)
            amp = rng.uniform(0.1, 0.4)
            a += amp * np.cos(2 * np.pi * f * k / n + rng.uniform(0, 2 * np.pi))
    elif choice == 4:
        f = rng.uniform(0.3, 2.0)
        a = 1.0 + 0.5 * np.sin(2 * np.pi * f * k / n)
    elif choice == 5:
        nblocks = rng.integers(2, 6)
        a = np.ones(n)
        for b in range(nblocks):
            start = b * n // nblocks
            end = (b + 1) * n // nblocks
            a[start:end] = rng.uniform(0.3, 1.5)
    elif choice == 6:
        center = n * rng.uniform(0.3, 0.7)
        width = n * rng.uniform(0.1, 0.3)
        a = 1.0 + 0.5 * np.exp(-0.5 * ((k - center) / width) ** 2)
    else:
        a = rng.random(n) + 0.1

    a = np.maximum(a, 0.01)
    s = np.sum(a)
    return a / s


def propose_candidate(seed=42, budget_s=1500):
    """
    Main search: Stochastic coordinate descent (exploiting 1-D convex substructure)
    + LP alternating + L-BFGS-B soft-max. Multi-n exploration with intensive polishing.
    """
    start_time = time.monotonic()
    deadline = start_time + float(budget_s) - 30.0

    best_seq = None
    best_score = np.inf

    # Load parent
    parent, parent_obj = try_parent()
    if parent is not None and parent_obj < best_score:
        best_score = parent_obj
        best_seq = parent.copy()
        print(f"Starting from parent: n={len(best_seq)}, obj={best_score:.6f}")

    # === PHASE 1: Intensive coordinate descent on parent ===
    if best_seq is not None and time.monotonic() < deadline - 200:
        n_best = len(best_seq)
        alloc = min(80.0, (deadline - time.monotonic()) * 0.2)
        print(f"\nPhase 1: Stochastic CD on n={n_best}, budget={alloc:.0f}s")
        a_cd = stochastic_coord_descent(best_seq, max_iters=50000,
                                         time_budget=alloc, seed=seed)
        obj_cd = compute_obj(a_cd)
        print(f"  After CD: obj={obj_cd:.6f}")
        if obj_cd < best_score:
            best_score = obj_cd
            best_seq = a_cd.copy()

    # === PHASE 2: LP alternating ===
    if best_seq is not None and time.monotonic() < deadline - 150:
        n_best = len(best_seq)
        alloc = min(60.0, (deadline - time.monotonic()) * 0.15)
        if n_best <= 800:
            print(f"\nPhase 2: LP alternating on n={n_best}, budget={alloc:.0f}s")
            a_lp = alternating_lp(best_seq, max_iters=40, time_budget=alloc)
            obj_lp = compute_obj(a_lp)
            print(f"  After LP: obj={obj_lp:.6f}")
            if obj_lp < best_score:
                best_score = obj_lp
                best_seq = a_lp.copy()

    # === PHASE 3: Coordinate descent after LP ===
    if best_seq is not None and time.monotonic() < deadline - 120:
        n_best = len(best_seq)
        alloc = min(60.0, (deadline - time.monotonic()) * 0.15)
        print(f"\nPhase 3: CD after LP on n={n_best}, budget={alloc:.0f}s")
        a_cd2 = stochastic_coord_descent(best_seq, max_iters=50000,
                                          time_budget=alloc, seed=seed + 1000)
        obj_cd2 = compute_obj(a_cd2)
        print(f"  After CD2: obj={obj_cd2:.6f}")
        if obj_cd2 < best_score:
            best_score = obj_cd2
            best_seq = a_cd2.copy()

    # === PHASE 4: L-BFGS-B polish ===
    if best_seq is not None and time.monotonic() < deadline - 100:
        n_best = len(best_seq)
        alloc = min(40.0, (deadline - time.monotonic()) * 0.1)
        print(f"\nPhase 4: L-BFGS-B on n={n_best}, budget={alloc:.0f}s")
        tau_list = [1.0 / n_best, 0.5 / n_best, 0.2 / n_best, 0.1 / n_best]
        a_lb, obj_lb = optimize_lbfgs(best_seq, tau_list, max_iter=200, time_budget=alloc)
        print(f"  After L-BFGS: obj={obj_lb:.6f}")
        if obj_lb < best_score:
            best_score = obj_lb
            best_seq = a_lb.copy()

    # === PHASE 5: More CD + LP cycling ===
    if best_seq is not None and time.monotonic() < deadline - 80:
        n_best = len(best_seq)
        remaining = deadline - time.monotonic()
        print(f"\nPhase 5: CD+LP cycling, {remaining:.0f}s remaining")
        for cycle in range(4):
            if time.monotonic() > deadline - 60:
                break
            rem = deadline - time.monotonic()
            if rem < 30:
                break

            # CD
            cd_budget = min(rem * 0.35, 40.0)
            a_new = stochastic_coord_descent(best_seq, max_iters=50000,
                                              time_budget=cd_budget, seed=seed + 2000 + cycle * 100)
            obj_new = compute_obj(a_new)
            if obj_new < best_score:
                best_score = obj_new
                best_seq = a_new.copy()
                print(f"  Cycle {cycle} CD: {best_score:.6f}")

            # LP
            rem = deadline - time.monotonic()
            if rem > 30 and len(best_seq) <= 800:
                lp_budget = min(rem * 0.3, 30.0)
                a_new = alternating_lp(best_seq, max_iters=30, time_budget=lp_budget)
                obj_new = compute_obj(a_new)
                if obj_new < best_score:
                    best_score = obj_new
                    best_seq = a_new.copy()
                    print(f"  Cycle {cycle} LP: {best_score:.6f}")

    # === PHASE 6: Try different n values ===
    n_candidates = [400, 500, 700, 800]
    if best_seq is not None:
        n_best = len(best_seq)
        n_candidates = [x for x in n_candidates if abs(x - n_best) > 30]

    print(f"\nPhase 6: Multi-n exploration, {deadline - time.monotonic():.0f}s remaining")
    for n_try in n_candidates[:3]:
        if time.monotonic() > deadline - 50:
            break
        rem = deadline - time.monotonic()
        if rem < 40:
            break

        alloc = min(rem * 0.25, 50.0)
        print(f"  Trying n={n_try}, budget={alloc:.0f}s")

        # Initialize from best_seq if possible
        if best_seq is not None:
            a_try = resample_sequence(best_seq, n_try)
        else:
            a_try = random_init(n_try, seed + n_try)

        # Quick CD
        cd_b = min(alloc * 0.5, 20.0)
        a_try = stochastic_coord_descent(a_try, max_iters=30000,
                                          time_budget=cd_b, seed=seed + n_try * 7)
        obj_try = compute_obj(a_try)

        # Quick LP if n allows
        rem = deadline - time.monotonic()
        if rem > 30 and n_try <= 700:
            lp_b = min(rem * 0.3, 20.0)
            a_try = alternating_lp(a_try, max_iters=25, time_budget=lp_b)
            obj_try = compute_obj(a_try)

        # More CD
        rem = deadline - time.monotonic()
        if rem > 20:
            cd_b2 = min(rem * 0.3, 15.0)
            a_try = stochastic_coord_descent(a_try, max_iters=30000,
                                              time_budget=cd_b2, seed=seed + n_try * 13)
            obj_try = compute_obj(a_try)

        print(f"    n={n_try}: obj={obj_try:.6f}")
        if obj_try < best_score:
            best_score = obj_try
            best_seq = a_try.copy()
            print(f"    ** UPDATE: n={n_try}, obj={best_score:.6f} **")

    # === PHASE 7: Final intensive polish on best ===
    if best_seq is not None and time.monotonic() < deadline - 20:
        n_final = len(best_seq)
        remaining = deadline - time.monotonic()
        print(f"\nPhase 7: Final polish n={n_final}, {remaining:.0f}s")

        # Multiple CD passes with different seeds
        for pass_idx in range(5):
            if time.monotonic() > deadline - 10:
                break
            rem = deadline - time.monotonic()
            if rem < 15:
                break
            cd_b = min(rem * 0.3, 30.0)
            a_new = stochastic_coord_descent(best_seq, max_iters=100000,
                                              time_budget=cd_b, seed=seed + 5000 + pass_idx * 31)
            obj_new = compute_obj(a_new)
            if obj_new < best_score:
                best_score = obj_new
                best_seq = a_new.copy()
                print(f"  Final CD pass {pass_idx}: {best_score:.6f}")

            # Small perturbation + CD
            rem = deadline - time.monotonic()
            if rem > 15:
                rng = np.random.default_rng(seed + 9000 + pass_idx)
                perturb_scale = rng.uniform(0.02, 0.06)
                a_pert = best_seq * np.exp(rng.normal(0, perturb_scale, n_final))
                a_pert = np.maximum(a_pert, 0)
                s_pert = np.sum(a_pert)
                if s_pert > 1e-10:
                    a_pert = a_pert / s_pert
                    cd_b2 = min(rem * 0.25, 20.0)
                    a_pert = stochastic_coord_descent(a_pert, max_iters=50000,
                                                       time_budget=cd_b2,
                                                       seed=seed + 9000 + pass_idx * 7)
                    obj_pert = compute_obj(a_pert)
                    if obj_pert < best_score:
                        best_score = obj_pert
                        best_seq = a_pert.copy()
                        print(f"  Final perturb {pass_idx}: {best_score:.6f}")

        # One final LP if time permits
        rem = deadline - time.monotonic()
        if rem > 20 and n_final <= 700:
            a_new = alternating_lp(best_seq, max_iters=20, time_budget=min(rem * 0.4, 30.0))
            obj_new = compute_obj(a_new)
            if obj_new < best_score:
                best_score = obj_new
                best_seq = a_new.copy()
                print(f"  Final LP: {best_score:.6f}")

            # CD after final LP
            rem = deadline - time.monotonic()
            if rem > 10:
                a_new = stochastic_coord_descent(best_seq, max_iters=50000,
                                                  time_budget=min(rem * 0.5, 20.0),
                                                  seed=seed + 99999)
                obj_new = compute_obj(a_new)
                if obj_new < best_score:
                    best_score = obj_new
                    best_seq = a_new.copy()
                    print(f"  Final CD: {best_score:.6f}")

    # Final validation
    if best_seq is None:
        best_seq = np.ones(200)
        best_score = compute_obj(best_seq)

    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    val = evaluate_sequence(result)
    if val < np.inf:
        best_score = val
    else:
        result = [1.0] * 100
        best_score = 2.0

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_score:.6f}, time={elapsed:.1f}s ===")

    return result
