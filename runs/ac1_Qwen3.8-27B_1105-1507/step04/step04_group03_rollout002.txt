import numpy as np
from scipy.optimize import linprog, minimize
from scipy.sparse import csr_matrix

def fast_conv(a):
    """Full self-convolution via FFT."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]


def compute_J(a):
    """Compute J(a) = 2*n*max(conv(a,a))/(sum(a))^2."""
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-10:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))


def build_lp_data(n):
    """Pre-compute LP sparse structure for convolution matrix A."""
    rows = []
    cols = []
    for k in range(2 * n - 1):
        j_min = max(0, k - n + 1)
        j_max = min(n - 1, k)
        for j in range(j_min, j_max + 1):
            rows.append(k)
            cols.append(j)
    return np.array(rows, dtype=np.int32), np.array(cols, dtype=np.int32), 2 * n - 1


def solve_lp(a, M, lp_data):
    """Solve LP: max sum(g) s.t. A@g <= M, g >= 0."""
    rows, cols, n_rows = lp_data
    n = len(a)
    vals = np.empty(len(rows), dtype=np.float64)
    for idx in range(len(rows)):
        k = rows[idx]
        j = cols[idx]
        vals[idx] = a[k - j]
    A = csr_matrix((vals, (rows, cols)), shape=(n_rows, n))
    c = -np.ones(n)
    b_ub = np.full(n_rows, M)
    bounds = [(0, None)] * n
    try:
        r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                    options={'presolve': True})
        if r.success and r.x is not None:
            return np.maximum(r.x, 0)
    except Exception:
        pass
    return None

                grad[i] += 2.0 * a[k - i]

        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-15:
            break

        # Adaptive step size
        step = step_scale / (gnorm * np.sqrt(it + 1.0))
        step = min(step, 0.1)

        a_new = a - step * grad
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s < 1e-10:
            step_scale *= 0.5
            continue
        a_new = a_new / s

        new_M = fast_conv(a_new).max()
        new_J = 2 * n * new_M / s / s if s > 0 else float('inf')

        if new_M < M - 1e-12:
            a = a_new
            if new_J < best_J:
                best_J = new_J
                best_a = a_new.copy()
        else:
            step_scale *= 0.7

    return best_a


def resample_seq(src, dest_n):
    """Resample 1D sequence to new length using linear interpolation."""
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)


def structured_perturb(a, rng, ptype):
    """Apply a structured perturbation."""
    n = len(a)
    t = np.arange(n, dtype=np.float64) / n

    if ptype == 'shift':
        s = int(rng.integers(1, max(2, n // 8)))
        a_p = np.roll(a, s)
    elif ptype == 'cosine':
        freq = rng.uniform(1, 6)
        amp = rng.uniform(0.02, 0.08)
        a_p = a * (1 + amp * np.cos(2 * np.pi * freq * t + rng.uniform(0, 2 * np.pi)))
    elif ptype == 'smooth':
        k = int(rng.integers(5, max(6, n // 8)))
        kernel = np.ones(k) / k
        a_p = np.convolve(a, kernel, mode='same')
    elif ptype == 'taper':
        alpha = rng.uniform(0.6, 0.95)
        edge = max(1, int(n * (1 - alpha) / 2))
        window = np.ones(n)
        for i in range(edge):
            frac = (i + 1) / edge
            window[i] = frac
            window[n - 1 - i] = frac
        a_p = a * window
    elif ptype == 'anti_peak':
        # Reduce at high-autocorrelation positions
        b = fast_conv(a)
        M = b.max()
        active = np.where(b >= 0.98 * M)[0]
        a_p = a.copy()
        for k in active:
            lo = max(0, k - n + 1)
            hi = min(n - 1, k)
            for i in range(lo, hi + 1):
                a_p[i] *= 0.95
    else:
        a_p = a + rng.normal(0, 0.02, n)

    a_p = np.maximum(a_p, 1e-6)
    s = a_p.sum()
    if s > 1e-8:
        a_p = a_p / s
    return a_p


def lbb_objective(x):
    """Objective for L-BFGS-B: max of self-convolution (scale invariant)."""
    ax = np.maximum(x, 0.0)
    s = ax.sum()
    if s < 1e-8:
        return 1e10
    ax = ax / s
    b = fast_conv(ax)
    return float(b.max())


def lbb_noise_landscape(x, a_current, M_current, n_active):
    """Noise landscape for L-BFGS-B: penalize deviation from flat convolution."""
    ax = np.maximum(x, 0.0)
    s = ax.sum()
    if s < 1e-8:
        return 1e10
    ax = ax / s
    b = fast_conv(ax)
    M = b.max()
    # Penalize peaks above current M
    excess = np.maximum(b - M_current, 0)
    return M + 0.5 * np.sum(excess ** 2)


def propose_candidate(seed=42, budget_s=1500, **kwargs):
    """
    Hybrid optimizer: LP mirror descent (linear + multiplicative), momentum LP,
    water-filling equalization, multi-start perturbations, multi-length search,
    and L-BFGS-B final refinement.
    """
    rng = np.random.default_rng(seed)
    t_start = time.monotonic()
    deadline = t_start + float(budget_s) - 30.0
    best_a = None
    best_score = float('inf')
    best_n = 0

    def update_best(a):
        nonlocal best_a, best_score, best_n
        a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
        if a.size == 0 or a.sum() < 0.01:
            return
        score = float(evaluate_sequence(a.tolist()))
        if np.isfinite(score) and score < best_score:
            best_a = a.copy()
            best_score = score
            best_n = a.size
            elapsed = time.monotonic() - t_start
            print(f"  [NEW BEST] J={score:.8f} n={a.size} t={elapsed:.0f}s", flush=True)

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        if p.ndim == 1 and p.size > 0 and np.all(np.isfinite(p)):
            parent = np.maximum(p, 0.0)
            update_best(parent)
            print(f"Parent loaded: n={parent.size}, J={best_score:.6f}", flush=True)
    except (NameError, AttributeError):
        pass

    if parent is None:
        parent = np.ones(512)
        update_best(parent)

    parent_n = parent.size
    print(f"Starting optimization, parent n={parent_n}, target J<1.5027", flush=True)

    # ===================== Phase 1: LP Mirror Descent (Linear) =====================
    if time.monotonic() < deadline - 1100:
        t0 = time.monotonic()
        print(f"\n{'='*55}", flush=True)
        print(f"Phase 1: LP Linear (n={parent_n})", flush=True)
        print(f"{'='*55}", flush=True)

        phase_dl = min(deadline - 1100, t0 + 150)
        a1 = np.maximum(parent, 1e-12)
        a1 = a1 / a1.sum()
        a1, J1 = lp_mirror_descent(a1, 500, phase_dl, mode='linear')
        update_best(a1)
        print(f"  J={J1:.8f} ({time.monotonic()-t0:.1f}s)", flush=True)

    # ===================== Phase 2: LP Mirror Descent (Multiplicative) =====================
    if time.monotonic() < deadline - 950:
        t0 = time.monotonic()
        print(f"\n{'='*55}", flush=True)
        print(f"Phase 2: LP Multiplicative (n={parent_n})", flush=True)
        print(f"{'='*55}", flush=True)

        phase_dl = min(deadline - 950, t0 + 120)
        a2 = np.maximum(best_a, 1e-12) if best_a is not None else np.maximum(parent, 1e-12)
        a2 = a2 / a2.sum()
        a2, J2 = lp_mirror_descent(a2, 500, phase_dl, mode='multiplicative')
        update_best(a2)
        print(f"  J={J2:.8f} ({time.monotonic()-t0:.1f}s)", flush=True)

    # ===================== Phase 3: LP with Momentum =====================
    if time.monotonic() < deadline - 750:
        t0 = time.monotonic()
        print(f"\n{'='*55}", flush=True)
        print(f"Phase 3: LP + Momentum (n={best_a.size if best_a is not None else parent_n})", flush=True)
        print(f"{'='*55}", flush=True)

        phase_dl = min(deadline - 750, t0 + 200)
        a3 = np.maximum(best_a, 1e-12) if best_a is not None else np.maximum(parent, 1e-12)
        a3 = a3 / a3.sum()
        a3, J3 = lp_mirror_descent(a3, 2000, phase_dl, mode='linear', momentum_beta=0.3)
        update_best(a3)
        print(f"  J={J3:.8f} ({time.monotonic()-t0:.1f}s)", flush=True)

    # ===================== Phase 4: Water Filling =====================
    if time.monotonic() < deadline - 500:
        t0 = time.monotonic()
        print(f"\n{'='*55}", flush=True)
        print(f"Phase 4: Water Filling (n={best_a.size if best_a is not None else parent_n})", flush=True)
        print(f"{'='*55}", flush=True)

        phase_dl = min(deadline - 500, t0 + 150)
        a4 = water_filling(best_a, phase_dl, max_iters=3000)
        update_best(a4)
        J4 = compute_J(a4)
        print(f"  J={J4:.8f} ({time.monotonic()-t0:.1f}s)", flush=True)

    # ===================== Phase 5: Multi-Start Perturbations =====================
    if time.monotonic() < deadline - 300:
        t0 = time.monotonic()
        print(f"\n{'='*55}", flush=True)
        print(f"Phase 5: Multi-Start Perturbations", flush=True)
        print(f"{'='*55}", flush=True)

        base = best_a if best_a is not None else parent
        perturbs = ['shift', 'cosine', 'anti_peak', 'smooth', 'taper', 'shift', 'anti_peak', 'cosine']
        per_pert_time = max(15, min(40, (deadline - 300 - time.monotonic()) / len(perturbs)))

        for i, ptype in enumerate(perturbs):
            if time.monotonic() > deadline - 280:
                break
            a_pert = structured_perturb(base, rng, ptype)
            dl = time.monotonic() + per_pert_time
            a_pert, J_pert = lp_mirror_descent(a_pert, 200, dl, mode='linear')
            update_best(a_pert)
            # Quick water filling
            if time.monotonic() < dl:
                a_wf = water_filling(a_pert, min(time.monotonic() + 10, deadline - 280), 500)
                update_best(a_wf)
            print(f"  [{ptype}] J={compute_J(a_pert):.8f} ({time.monotonic()-t0:.1f}s)", flush=True)

    # ===================== Phase 6: Long LP on Best =====================
    if time.monotonic() < deadline - 100:
        t0 = time.monotonic()
        n_long = best_a.size if best_a is not None else parent_n
        print(f"\n{'='*55}", flush=True)
        print(f"Phase 6: Long LP (n={n_long})", flush=True)
        print(f"{'='*55}", flush=True)

        a_long = np.maximum(best_a, 1e-12)
        a_long = a_long / a_long.sum()
        a_long, J_long = lp_mirror_descent(a_long, 50000, deadline - 80, mode='linear')
        update_best(a_long)
        print(f"  J={J_long:.8f} ({time.monotonic()-t0:.1f}s)", flush=True)

        # Try multiplicative on the result
        if time.monotonic() < deadline - 30:
            a_long2 = np.maximum(best_a, 1e-12)
            a_long2 = a_long2 / a_long2.sum()
            a_long2, J_long2 = lp_mirror_descent(a_long2, 10000, deadline - 20, mode='multiplicative')
            update_best(a_long2)
            if J_long2 < J_long:
                print(f"  Mult improved: J={J_long2:.8f}", flush=True)

    # ===================== Phase 7: Water Filling on Best =====================
    if time.monotonic() < deadline - 30:
        t0 = time.monotonic()
        print(f"\n{'='*55}", flush=True)
        print(f"Phase 7: Final Water Filling", flush=True)
        print(f"{'='*55}", flush=True)

        a_wf = water_filling(best_a, min(deadline - 10, t0 + 60), 5000)
        update_best(a_wf)
        J_wf = compute_J(a_wf)
        print(f"  J={J_wf:.8f} ({time.monotonic()-t0:.1f}s)", flush=True)

    # ===================== Phase 8: L-BFGS-B Final Polish =====================
    if time.monotonic() < deadline - 5 and best_a is not None:
        t0 = time.monotonic()
        print(f"\n{'='*55}", flush=True)
        print(f"Phase 8: L-BFGS-B Polish (n={best_a.size})", flush=True)
        print(f"{'='*55}", flush=True)

        try:
            n = best_a.size
            res = minimize(lbb_objective, best_a.astype(np.float64),
                           method='L-BFGS-B', bounds=[(0.0, None)] * n,
                           options={'maxiter': 200, 'max_fun': 50000, 'ftol': 1e-18,
                                    'gtol': 1e-15})
            if res.fun < 1e10:
                update_best(res.x)
                print(f"  L-BFGS-B: max_conv={res.fun:.10f} J={compute_J(res.x):.8f}", flush=True)
        except Exception as e:
            print(f"  L-BFGS-B error: {e}", flush=True)
        print(f"  ({time.monotonic()-t0:.1f}s)", flush=True)

    # ===================== Phase 9: Multi-Length Check =====================
    if time.monotonic() < deadline and best_a is not None:
        t0 = time.monotonic()
        print(f"\n{'='*55}", flush=True)
        print(f"Phase 9: Multi-Length Check", flush=True)
        print(f"{'='*55}", flush=True)

        base_n = best_a.size
        for n_try in [base_n - 32, base_n + 32, base_n - 64, base_n + 64]:
            if n_try < 64 or n_try > 2048 or time.monotonic() > deadline - 5:
                continue
            a_try = resample_seq(best_a, n_try)
            a_try = np.maximum(a_try, 1e-12)
            a_try = a_try / a_try.sum()
            # Quick LP
            dl = min(deadline - 3, t0 + 20)
            a_try, J_try = lp_mirror_descent(a_try, 200, dl, mode='linear')
            update_best(a_try)
            J_try = compute_J(a_try)
            print(f"  n={n_try}: J={J_try:.8f}", flush=True)

    # ===================== Final =====================
    if best_a is not None:
        final_score = float(evaluate_sequence(best_a.tolist()))
        print(f"\n{'='*55}", flush=True)
        print(f"FINAL: J={final_score:.8f} n={best_a.size} time={time.monotonic()-t_start:.0f}s", flush=True)
        print(f"{'='*55}", flush=True)
        return [float(x) for x in best_a]
    else:
        return [1.0] * 64
