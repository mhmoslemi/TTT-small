import time


def compute_conv(a):
    """Compute linear convolution a*a using FFT. Returns array of length 2n-1."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    conv = irfft(fa * fa, size)
    return conv[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    conv = compute_conv(a)
    max_b = float(np.max(conv))
    sum_a = float(np.sum(a))
    if sum_a < 0.01:
        return np.inf
    return 2.0 * n * max_b / (sum_a * sum_a)


def fft_convolve(a, b):
    """Linear convolution of a and b using FFT."""
    n, m = len(a), len(b)
    size = 1
    while size < n + m - 1:
        size <<= 1
    fa = rfft(a, size)
    fb = rfft(b, size)
    return irfft(fa * fb, size)[:n + m - 1]


def softmax(z):
    """Numerically stable softmax."""
    z = z - np.max(z)
    e = np.exp(z)
    return e / np.sum(e)


def smooth_obj_and_grad_z(z, tau):
    """
    Compute smoothed max of conv(softmax(z), softmax(z)) and gradient w.r.t. z.
    Uses log-sum-exp smoothing.
    """
    n = len(z)
    a = softmax(z)
    conv = compute_conv(a)
    cmax = float(np.max(conv))
    shifted = np.clip((conv - cmax) / tau, -500.0, 500.0)
    exp_conv = np.exp(shifted)
    Z = float(np.sum(exp_conv))
    soft_max_val = cmax + tau * np.log(Z)
    w = exp_conv / Z

    # grad_a[i] = 2 * sum_j w[i+j] * a[j]
    w_rev = w[::-1]
    conv_fw = fft_convolve(a, w_rev)
    grad_a = 2.0 * conv_fw[n - 1:2 * n - 1][::-1]

    mean_grad = float(np.dot(a, grad_a))
    grad_z = a * (grad_a - mean_grad)
    return soft_max_val, grad_z


def optimize_smooth(a_init, tau_list, max_iter_per_tau=200, time_budget=30.0):
    """L-BFGS-B with softmax parametrization and tau annealing."""
    n = len(a_init)
    a = a_init.copy()
    s = np.sum(a)
    if s < 1e-10:
        a = np.ones(n) / n
    else:
        a = a / s
    a = np.maximum(a, 1e-30)
    z = np.log(a)

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    for tau in tau_list:
        if time.monotonic() - start > time_budget:
            break

        try:
            result = minimize(smooth_obj_and_grad_z, z, args=(tau,), jac=True,
                              method='L-BFGS-B',
                              options={'maxiter': max_iter_per_tau, 'ftol': 1e-15, 'gtol': 1e-9})
            z = result.x
        except Exception:
            continue

        a_new = softmax(z)
        a_new = np.maximum(a_new, 1e-30)
        s_new = np.sum(a_new)
        if s_new < 1e-10:
            continue
        a_new = a_new / s_new
        obj = compute_obj(a_new)
        if obj < best_obj:
            best_obj = obj
            best_a = a_new.copy()

    return best_a, best_obj


def lp_direct_peak_shave(a, time_limit=30):
    """
    LP-based direct peak shaving. Linearizes the quadratic constraints at current a
    and solves for a direction that minimizes the peak.
    Returns (delta, M_new) or (None, None).
    """
    n = len(a)
    a = a.copy()
    s = np.sum(a)
    if s < 1e-10:
        return None, None
    a = a / s

    conv = compute_conv(a)
    M_current = float(np.max(conv))

    # Build LP: min M_new
    # s.t. for each k: sum_i 2*a[k-i]*delta[i] - M_new <= -conv[k]
    #      sum_i delta[i] = 0
    #      delta[i] >= -a[i]  =>  -delta[i] <= a[i]
    #
    # Variables: [delta_0, ..., delta_{n-1}, M_new]  (n+1 total)

    num_vars = n + 1
    c = np.zeros(num_vars)
    c[-1] = 1.0  # minimize M_new

    # Inequality constraints: A_ub @ x <= b_ub
    # (2n-1) constraints from linearized conv + n constraints from non-negativity
    num_ineq = (2 * n - 1) + n
    A_ub = np.zeros((num_ineq, num_vars))
    b_ub = np.zeros(num_ineq)

    for k in range(2 * n - 1):
        # row k: sum_i 2*a[k-i]*delta[i] - M_new <= -conv[k]
        for i in range(n):
            ji = k - i
            if 0 <= ji < n:
                A_ub[k, i] = 2.0 * a[ji]
        A_ub[k, n] = -1.0
        b_ub[k] = -conv[k]

    # Non-negativity: -delta[i] <= a[i]
    for i in range(n):
        row = 2 * n - 1 + i
        A_ub[row, i] = -1.0
        b_ub[row] = a[i]

    # Equality: sum delta = 0
    A_eq = np.zeros((1, num_vars))
    A_eq[0, :n] = 1.0
    b_eq = np.array([0.0])

    bounds = [(None, None)] * n + [(0.0, None)]  # M_new >= 0

    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': time_limit})
        if res.status == 0:
            delta = res.x[:n]
            M_new = float(res.x[n])
            # Ensure delta doesn't violate non-negativity
            delta = np.maximum(delta, -a + 1e-15)
            # Renormalize delta to have sum 0
            dsum = np.sum(delta)
            delta = delta - dsum / n
            return delta, M_new
    except Exception:
        pass
    return None, None


def lp_shave_and_update(a, max_iters=50, time_budget=50.0, verbose=False):
    """Iteratively apply LP peak shaving with line search."""
    n = len(a)
    a = a.copy()
    s = np.sum(a)
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    stall = 0

    for it in range(max_iters):
        if time.monotonic() - start > time_budget:
            break

        delta, M_new = lp_direct_peak_shave(a, time_limit=20)
        if delta is None:
            break

        # Line search
        improved = False
        for t in [1.0, 0.7, 0.5, 0.3, 0.2, 0.1]:
            a_try = a + t * delta
            a_try = np.maximum(a_try, 0)
            s_try = np.sum(a_try)
            if s_try < 1e-10:
                continue
            a_try = a_try / s_try
            obj_try = compute_obj(a_try)
            if obj_try < best_obj - 1e-12:
                best_obj = obj_try
                best_a = a_try.copy()
                a = a_try.copy()
                improved = True
                break

        if improved:
            stall = 0
            if verbose:
                print(f"    LP-shave iter {it}: obj={best_obj:.6f}")
        else:
            stall += 1
            if stall > 15:
                # Perturb and restart
                rng = np.random.default_rng(it * 7919 + 42)
                a = a * np.exp(rng.normal(0, 0.05, n))
                a = np.maximum(a, 1e-15)
                a = a / np.sum(a)
                stall = 0
                if stall > 30:
                    break

    return best_a


def subgradient_optimize(a, max_iters=5000, time_budget=20.0, tau_start=2.0, tau_end=50.0,
                         momentum=0.9):
    """Smoothed subgradient with momentum."""
    n = len(a)
    a = a.copy()
    s = np.sum(a)
    if s < 1e-10:
        a = np.ones(n) / n
    else:
        a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    step = 0.15
    vel = np.zeros(n)
    stall = 0

    for it in range(max_iters):
        if time.monotonic() - start > time_budget:
            break

        frac = it / max(1, max_iters)
        tau = tau_start + (tau_end - tau_start) * frac * frac  # quadratic schedule

        conv = compute_conv(a)
        max_conv = float(np.max(conv))
        shifted = np.clip(tau * (conv - max_conv), -300, 300)
        weights = np.exp(shifted)
        w_sum = np.sum(weights)
        if w_sum < 1e-300:
            idx = int(np.argmax(conv))
            weights = np.zeros(len(conv))
            weights[idx] = 1.0
        else:
            weights /= w_sum

        w_rev = weights[::-1]
        conv_fw = fft_convolve(a, w_rev)
        grad_a = 2.0 * conv_fw[n - 1:2 * n - 1][::-1]

        grad_centered = grad_a - np.mean(grad_a)
        vel = momentum * vel - step * grad_centered
        a_new = a + vel
        a_new = np.maximum(a_new, 1e-15)
        s_new = np.sum(a_new)
        if s_new > 1e-10:
            a_new = a_new / s_new
        else:
            a_new = a.copy()
            vel = np.zeros(n)

        a = a_new
        obj = compute_obj(a)

        if obj < best_obj - 1e-12:
            best_obj = obj
            best_a = a.copy()
            stall = 0
        else:
            stall += 1
            if stall > 200:
                step *= 0.7
                if step < 1e-12:
                    break
                rng = np.random.default_rng(it * 31 + 7)
                a = a + rng.normal(0, 0.02, n) * step
                a = np.maximum(a, 1e-15)
                s = np.sum(a)
                if s > 1e-10:
                    a = a / s
                vel = np.zeros(n)
                stall = 0

        if it > 100:
            step *= 0.997

    return best_a


def alternating_lp(a, max_iters=40, time_budget=40.0):
    """Alternating LP: solve max sum(g) s.t. (a*g)[k] <= M, g >= 0, then blend."""
    n = len(a)
    a = a.copy()
    s = np.sum(a)
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    stall = 0

    for it in range(max_iters):
        if time.monotonic() - start > time_budget:
            break

        conv = compute_conv(a)
        M = float(np.max(conv))

        # Build Toeplitz for a*g
        A = np.zeros((2 * n - 1, n))
        for j in range(n):
            k_start = j
            k_end = min(j + n, 2 * n - 1)
            A[k_start:k_end, j] = a[:k_end - k_start]

        c = -np.ones(n)
        b_ub = np.full(2 * n - 1, M)
        bounds = [(0.0, None)] * n

        try:
            res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                          options={'presolve': True, 'time_limit': 15})
            if res.status != 0:
                break
            g = np.maximum(res.x, 0)
            s_g = float(np.sum(g))
            if s_g < 1e-10:
                break
            g = g / s_g
        except Exception:
            break

        # Line search for blend
        improved = False
        for t in [0.5, 0.3, 0.7, 1.0, 0.2]:
            a_try = (1.0 - t) * a + t * g
            a_try = np.maximum(a_try, 1e-15)
            s_try = np.sum(a_try)
            if s_try < 1e-10:
                continue
            a_try = a_try / s_try
            obj_try = compute_obj(a_try)
            if obj_try < best_obj - 1e-12:
                best_obj = obj_try
                best_a = a_try.copy()
                a = a_try.copy()
                improved = True
                break

        if improved:
            stall = 0
        else:
            stall += 1
            if stall > 20:
                rng = np.random.default_rng(it * 104729 + 1)
                a = a * np.exp(rng.normal(0, 0.08, n))
                a = np.maximum(a, 1e-15)
                a = a / np.sum(a)
                stall = 0
                if stall > 40:
                    break

    return best_a


def random_init(n, seed):
    """Generate diverse random initializations."""
    rng = np.random.default_rng(seed)
    k = np.arange(n)
    choice = rng.integers(0, 12)

    if choice == 0:
        f = rng.uniform(0.5, 4.0)
        phase = rng.uniform(0, 2 * np.pi)
        a = 1.0 + 0.8 * np.cos(2 * np.pi * f * k / n + phase)
    elif choice == 1:
        f1 = rng.uniform(0.3, 2.0)
        f2 = rng.uniform(2.0, 6.0)
        a = 1.0 + 0.4 * np.cos(2 * np.pi * f1 * k / n) + 0.3 * np.cos(2 * np.pi * f2 * k / n)
    elif choice == 2:
        center = n * rng.uniform(0.3, 0.7)
        alpha = rng.uniform(0.5, 3.0)
        a = 1.0 + 0.5 * np.exp(-alpha * np.abs(k - center) / (n * 0.3))
    elif choice == 3:
        a = rng.random(n)
        win = max(3, n // 20)
        kernel = np.ones(win) / win
        a = np.convolve(a, kernel, mode='same')
    elif choice == 4:
        a = np.ones(n)
        for _ in range(rng.integers(1, 5)):
            f = rng.uniform(0.5, 5.0)
            amp = rng.uniform(0.1, 0.4)
            a += amp * np.cos(2 * np.pi * f * k / n + rng.uniform(0, 2 * np.pi))
    elif choice == 5:
        f = rng.uniform(0.3, 2.0)
        a = 1.0 + 0.5 * np.sin(2 * np.pi * f * k / n)
    elif choice == 6:
        nblocks = rng.integers(2, 7)
        a = np.ones(n)
        for b in range(nblocks):
            start = b * n // nblocks
            end = (b + 1) * n // nblocks
            a[start:end] = rng.uniform(0.3, 1.5)
    elif choice == 7:
        f0 = rng.uniform(0.5, 2.0)
        f1 = rng.uniform(2.0, 6.0)
        a = 1.0 + 0.5 * np.cos(2 * np.pi * (f0 * k / n + (f1 - f0) * k**2 / (2 * n**2)))
    elif choice == 8:
        center = n * rng.uniform(0.2, 0.8)
        width = n * rng.uniform(0.1, 0.4)
        a = 1.0 + 0.6 * np.exp(-0.5 * ((k - center) / width) ** 2)
    elif choice == 9:
        # Fourier-based: random flat-ish spectrum
        L = 2 * n
        freqs = np.arange(L)
        phases = rng.uniform(0, 2 * np.pi, L)
        mags = np.ones(L)
        for _ in range(rng.integers(1, 4)):
            mags += rng.uniform(0.1, 0.5) * np.cos(2 * np.pi * rng.uniform(1, 10) * freqs / L)
        mags = np.maximum(mags, 0.1)
        a = irfft(mags * np.exp(1j * phases), L)[:n]
        a = np.maximum(a, 0.01)
    elif choice == 10:
        # Two-scale: base pattern repeated with variation
        block_len = max(2, n // 10)
        base = np.maximum(rng.uniform(0.5, 2.0, block_len), 0.1)
        a = np.tile(base, n // block_len + 1)[:n]
        a += rng.normal(0, 0.1, n)
        a = np.maximum(a, 0.01)
    else:
        a = rng.random(n) + 0.1

    a = np.maximum(a, 0.01)
    s = np.sum(a)
    return a / s


def resample_sequence(a, new_n):
    """Resample sequence to new length via interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    return np.interp(x_new, x_old, a)


def try_parent():
    """Try to load parent sequence."""
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64)
        parent = np.clip(parent, 0, 1000.0)
        if parent.size > 0 and np.isfinite(parent).all():
            obj = compute_obj(parent)
            print(f"Parent (n={parent.size}): obj={obj:.6f}")
            return parent.copy(), obj
    except Exception:
        pass
    return None, np.inf


def polish_sequence(a, time_budget, verbose=False):
    """Intensive polishing combining all methods."""
    n = len(a)
    a = a.copy()
    start = time.monotonic()

    best_a = a.copy()
    best_obj = compute_obj(a)
    round_num = 0

    while time.monotonic() - start < time_budget - 3:
        round_num += 1
        remaining = time_budget - (time.monotonic() - start)
        if remaining < 5:
            break

        # LP direct peak shaving (new, primary method)
        if n <= 1000 and remaining > 20:
            a_new = lp_shave_and_update(best_a, max_iters=min(40, int(remaining)),
                                        time_budget=min(remaining * 0.4, 40.0),
                                        verbose=verbose)
            obj = compute_obj(a_new)
            if obj < best_obj:
                best_obj = obj
                best_a = a_new.copy()
                if verbose:
                    print(f"    Polish r{round_num} LP-shave: {best_obj:.6f}")

        # Smooth L-BFGS-B
        remaining = time_budget - (time.monotonic() - start)
        if remaining > 15:
            tau_list = [1.0 / n, 0.5 / n, 0.2 / n, 0.1 / n, 0.05 / n]
            a_new, obj_new = optimize_smooth(best_a, tau_list, max_iter_per_tau=150,
                                             time_budget=min(remaining * 0.25, 20.0))
            if obj_new < best_obj:
                best_obj = obj_new
                best_a = a_new.copy()
                if verbose:
                    print(f"    Polish r{round_num} L-BFGS: {best_obj:.6f}")

        # Subgradient
        remaining = time_budget - (time.monotonic() - start)
        if remaining > 10:
            a_new = subgradient_optimize(best_a, max_iters=int(remaining * 8),
                                         time_budget=min(remaining * 0.2, 12.0),
                                         tau_start=5.0, tau_end=40.0)
            obj = compute_obj(a_new)
            if obj < best_obj:
                best_obj = obj
                best_a = a_new.copy()
                if verbose:
                    print(f"    Polish r{round_num} SG: {best_obj:.6f}")

        # Alternating LP
        remaining = time_budget - (time.monotonic() - start)
        if n <= 800 and remaining > 20:
            a_new = alternating_lp(best_a, max_iters=30,
                                   time_budget=min(remaining * 0.25, 25.0))
            obj = compute_obj(a_new)
            if obj < best_obj:
                best_obj = obj
                best_a = a_new.copy()
                if verbose:
                    print(f"    Polish r{round_num} AltLP: {best_obj:.6f}")

    return best_a, best_obj


def propose_candidate():
    """
    Main search function. Explores multiple sequence lengths using:
    - Smoothed L-BFGS-B with geometric tau annealing
    - LP-based direct peak shaving (linearized QCQP)
    - Momentum subgradient method
    - Alternating LP polishing
    Multi-start with diverse initializations.
    """
    start_time = time.monotonic()
    deadline = start_time + 1475.0  # 25s buffer

    best_seq = None
    best_score = np.inf

    # Load parent
    parent, parent_obj = try_parent()
    if parent is not None and parent_obj < best_score:
        best_score = parent_obj
        best_seq = parent.copy()

    # Sequence lengths to explore
    n_values = [500, 550, 600, 650, 700, 800]
    if best_seq is not None:
        np_ = len(best_seq)
        extra = [np_, max(100, np_ - 100), min(1200, np_ + 100)]
        n_values = sorted(set(n_values + extra))

    print(f"Exploring n values: {n_values}")
    print(f"Initial best: {best_score:.6f}")
    print(f"Deadline in {deadline - start_time:.0f}s")

    candidates = []  # (a, obj, n)

    # Phase 1: Exploration
    for n in n_values:
        if time.monotonic() > deadline - 200:
            break
        remaining = deadline - time.monotonic()
        if remaining < 15:
            break

        # Time allocation
        if n <= 550:
            alloc = min(45.0, remaining * 0.12)
        elif n <= 700:
            alloc = min(60.0, remaining * 0.14)
        else:
            alloc = min(70.0, remaining * 0.16)
        alloc = max(20.0, alloc)

        print(f"\nN={n}: {remaining:.0f}s remaining, alloc={alloc:.0f}s")

        n_inits = max(2, int(alloc / 20))
        n_inits = min(n_inits, 4)

        for init_idx in range(n_inits):
            if time.monotonic() > deadline - 200:
                break
            remaining_init = deadline - time.monotonic()
            if remaining_init < 15:
                break

            # Generate initialization
            if init_idx == 0 and best_seq is not None:
                a = resample_sequence(best_seq, n)
            else:
                a = random_init(n, 42 + n * 104729 + init_idx * 79193)

            # Step 1: Quick subgradient shaping
            sg_time = min(6.0, (alloc / n_inits) * 0.15)
            a = subgradient_optimize(a, max_iters=int(sg_time * 30),
                                      time_budget=sg_time, tau_start=3.0, tau_end=25.0)

            # Step 2: Smooth L-BFGS-B with tau annealing
            remaining_init = deadline - time.monotonic()
            if remaining_init > 20:
                lfbs_time = min((alloc / n_inits) * 0.35, 30.0)
                tau_list = [5.0/n, 3.0/n, 1.5/n, 0.75/n, 0.4/n, 0.2/n, 0.1/n, 0.05/n]
                a, obj = optimize_smooth(a, tau_list, max_iter_per_tau=150,
                                         time_budget=lfbs_time)
            else:
                obj = compute_obj(a)

            # Step 3: LP direct peak shaving
            remaining_init = deadline - time.monotonic()
            if remaining_init > 20 and n <= 1000:
                a = lp_shave_and_update(a, max_iters=min(30, int(remaining_init // 2)),
                                        time_budget=min(remaining_init * 0.3, 30.0))
                obj = compute_obj(a)

            # Step 4: Subgradient polish
            remaining_init = deadline - time.monotonic()
            if remaining_init > 10:
                a = subgradient_optimize(a, max_iters=int(remaining_init * 10),
                                          time_budget=min(remaining_init * 0.15, 10.0),
                                          tau_start=8.0, tau_end=50.0)
                obj = compute_obj(a)

            # Step 5: Alternating LP
            remaining_init = deadline - time.monotonic()
            if remaining_init > 15 and n <= 800:
                a = alternating_lp(a, max_iters=25, time_budget=min(remaining_init * 0.3, 20.0))
                obj = compute_obj(a)

            print(f"  init{init_idx}: n={n}, obj={obj:.6f}")

            if obj < best_score:
                best_score = obj
                best_seq = a.copy()
                print(f"  ** UPDATE: n={n}, obj={best_score:.6f} **")

            candidates.append((a.copy(), obj, n))

    # Phase 2: Intensive polishing of top candidates
    print(f"\n=== POLISHING PHASE ===")
    print(f"Best before polish: {best_score:.6f}")
    print(f"Time remaining: {deadline - time.monotonic():.0f}s")

    candidates.sort(key=lambda x: x[1])
    n_polish = min(4, len(candidates))

    for ci in range(n_polish):
        if time.monotonic() > deadline - 80:
            break
        remaining = deadline - time.monotonic()
        if remaining < 25:
            break

        a_cand = candidates[ci][0]
        n_cand = candidates[ci][2]
        print(f"  Polishing candidate {ci} (n={n_cand}, obj={candidates[ci][1]:.6f}), "
              f"{remaining:.0f}s left")

        polish_budget = min(remaining * 0.35, 100.0)
        a_pol, obj_pol = polish_sequence(a_cand, time_budget=polish_budget, verbose=True)

        if obj_pol < best_score:
            best_score = obj_pol
            best_seq = a_pol.copy()
            print(f"  ** UPDATE after polish: obj={best_score:.6f} **")

        # Perturbation-based restart
        remaining = deadline - time.monotonic()
        if remaining > 25:
            rng = np.random.default_rng(ci * 555 + 999)
            for ptry in range(4):
                if time.monotonic() > deadline - 40:
                    break
                scale = rng.uniform(0.02, 0.10)
                a_pert = a_cand * np.exp(rng.normal(0, scale, n_cand))
                a_pert = np.maximum(a_pert, 1e-15)
                a_pert = a_pert / np.sum(a_pert)

                # Quick optimization
                a_pert = subgradient_optimize(a_pert, max_iters=300, time_budget=4.0,
                                              tau_start=5.0, tau_end=30.0)
                a_pert, obj_pert = optimize_smooth(a_pert,
                                                   [1.0/n_cand, 0.3/n_cand, 0.1/n_cand],
                                                   max_iter_per_tau=100, time_budget=8.0)
                if obj_pert < best_score:
                    best_score = obj_pert
                    best_seq = a_pert.copy()
                    print(f"  ** UPDATE (perturb {ptry}): obj={best_score:.6f} **")

    # Phase 3: Final intensive polish
    if best_seq is not None and time.monotonic() < deadline - 30:
        remaining = deadline - time.monotonic()
        n_best = len(best_seq)
        print(f"\n=== FINAL POLISH: n={n_best}, {remaining:.0f}s ===")

        a_cur = best_seq.copy()
        best_final_obj = best_score

        for final_round in range(6):
            if time.monotonic() > deadline - 10:
                break
            remaining = deadline - time.monotonic()
            if remaining < 15:
                break

            # LP peak shaving
            if n_best <= 1000 and remaining > 20:
                a_new = lp_shave_and_update(a_cur, max_iters=30,
                                            time_budget=min(remaining * 0.25, 25.0),
                                            verbose=True)
                obj_new = compute_obj(a_new)
                if obj_new < best_final_obj:
                    best_final_obj = obj_new
                    best_seq = a_new.copy()
                    a_cur = a_new.copy()
                    print(f"  Final LP-shave {final_round}: {best_final_obj:.6f}")

            # L-BFGS-B
            remaining = deadline - time.monotonic()
            if remaining > 15:
                tau_list = [0.5/n_best, 0.2/n_best, 0.1/n_best, 0.05/n_best, 0.02/n_best]
                a_new, obj_new = optimize_smooth(a_cur, tau_list, max_iter_per_tau=200,
                                                 time_budget=min(remaining * 0.2, 15.0))
                if obj_new < best_final_obj:
                    best_final_obj = obj_new
                    best_seq = a_new.copy()
                    a_cur = a_new.copy()
                    print(f"  Final L-BFGS {final_round}: {best_final_obj:.6f}")

            # Subgradient
            remaining = deadline - time.monotonic()
            if remaining > 10:
                a_new = subgradient_optimize(a_cur, max_iters=int(remaining * 8),
                                             time_budget=min(remaining * 0.15, 12.0),
                                             tau_start=10.0, tau_end=60.0)
                obj_new = compute_obj(a_new)
                if obj_new < best_final_obj:
                    best_final_obj = obj_new
                    best_seq = a_new.copy()
                    a_cur = a_new.copy()
                    print(f"  Final SG {final_round}: {best_final_obj:.6f}")

            # Alternating LP
            remaining = deadline - time.monotonic()
            if n_best <= 800 and remaining > 15:
                a_new = alternating_lp(a_cur, max_iters=25,
                                       time_budget=min(remaining * 0.2, 15.0))
                obj_new = compute_obj(a_new)
                if obj_new < best_final_obj:
                    best_final_obj = obj_new
                    best_seq = a_new.copy()
                    a_cur = a_new.copy()
                    print(f"  Final AltLP {final_round}: {best_final_obj:.6f}")

        best_score = best_final_obj

    # Final validation
    if best_seq is None:
        best_seq = np.ones(600) / 600
        best_score = compute_obj(best_seq)

    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    val = evaluate_sequence(result)
    print(f"\nFinal evaluate_sequence: {val:.6f}")
    if np.isfinite(val) and val < best_score:
        best_score = val
    elif not np.isfinite(val):
        result = [1.0] * 100
        best_score = 2.0

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_score:.6f}, time={elapsed:.1f}s ===")

    return result
