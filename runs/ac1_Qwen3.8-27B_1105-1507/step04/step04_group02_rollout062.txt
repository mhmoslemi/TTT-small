"""
Optimized step-function coefficient search using soft-max smoothed objective
with gradient-based optimization, adaptive temperature annealing, LP polishing,
and multi-start exploration to minimize 2*n*max(conv(a,a))/(sum(a))^2.
"""

import numpy as np
from scipy.optimize import linprog
from scipy.interpolate import interp1d
from scipy.sparse import coo_matrix
import time

def compute_conv(a):
    """Compute linear convolution a*a via FFT."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    conv = np.fft.irfft(fa * fa, size)
    return conv[:2 * n - 1]

def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    S = float(np.sum(a))
    if S < 1e-10:
        return np.inf
    conv = compute_conv(a)
    M = float(np.max(conv))
    return 2.0 * n * M / (S * S)

def softmax_obj(a, tau):

def build_toeplitz_sparse(a):
    """Build sparse Toeplitz matrix for conv(a, g)."""
    n = len(a)
    rows = []
    cols = []
    vals = []
    for j in range(n):
        start = j
        end = min(j + n, 2 * n - 1)
        length = end - start
        rows.extend(range(start, end))
        cols.extend([j] * length)
        vals.extend(list(a[:length]))
    A = coo_matrix((vals, (rows, cols)), shape=(2 * n - 1, n)).tocsr()
    return A

def solve_lp_maxsum(a, M, time_limit=8):
    """Solve LP: max sum(g) s.t. conv(a,g) <= M, g >= 0."""
    n = len(a)
    A = build_toeplitz_sparse(a)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': time_limit})
        if res.status == 0:
            return np.maximum(res.x, 0)
    except Exception:
        pass
    return None

def softmax_gradient_descent(a, time_budget, tau_start=5.0, tau_end=200.0,
                              max_iters=5000, lr_start=0.1, momentum=0.9):
    """Gradient descent on soft-max smoothed objective with annealed temperature."""
    n = len(a)
    a = a.copy().astype(np.float64)
    S = np.sum(a)
    if S < 1e-10:
        return a, np.inf
    a = a / S
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    v = np.zeros(n)
    lr = lr_start
    stall = 0
    
    for it in range(max_iters):
        if time.monotonic() - start > time_budget:
            break
        
        # Anneal temperature
        frac = min(1.0, it / max_iters)
        tau = tau_start * (tau_end / tau_start) ** frac
        
        obj, grad = softmax_obj(a, tau)
        
        # Center gradient (preserve sum)
        grad_centered = grad - np.mean(grad)
        
        # Momentum update
        v = momentum * v + (1.0 - momentum) * grad_centered
        a_new = a - lr * v
        a_new = np.maximum(a_new, 0)
        S_new = np.sum(a_new)
        if S_new < 1e-10:
            break
        a_new = a_new / S_new
        
        # Check improvement
        obj_new = compute_obj(a_new)
        if obj_new < best_obj - 1e-10:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new
            stall = 0
        else:
            a = a_new  # Keep moving but track best
            stall += 1
            if stall > 50:
                lr *= 0.8
                if lr < 1e-8:
                    break
    
    return best_a, best_obj

        if time.monotonic() - start > time_budget:
            break
        
        conv = compute_conv(a)
        M = float(np.max(conv))

        best_t = 0.0
        best_obj_ls = best_obj
        for t in [0.05, 0.1, 0.2, 0.3, 0.5, 0.7, 0.9, 1.0]:
            a_try = (1.0 - t) * a + t * g
            a_try = np.maximum(a_try, 0)
            S_try = np.sum(a_try)
            if S_try < 1e-10:
                continue
            a_try = a_try / S_try
            obj_try = compute_obj(a_try)
            if obj_try < best_obj_ls:
                best_obj_ls = obj_try
                best_t = t
        
        if best_t > 0.0:
            a = (1.0 - best_t) * a + best_t * g
            a = np.maximum(a, 0)
            a = a / np.sum(a)
            best_obj = best_obj_ls
            best_a = a.copy()
            stall = 0
        else:

            if stall > 10:
                # Try with slightly relaxed M
                M_relaxed = M * 1.005
                g = solve_lp_maxsum(a, M_relaxed, time_limit=5)
                if g is not None and np.sum(g) > 1e-10:

                    a = 0.95 * a + 0.05 * g

                    a = a / np.sum(a)
                    obj = compute_obj(a)
                    if obj < best_obj:
                        best_obj = obj
                        best_a = a.copy()
                stall = 0

def cosine_init(n, n_modes=8, seed=0):
    """Generate a smooth cosine-series initialization."""
    rng = np.random.default_rng(seed)
    k = np.arange(n)
    a = np.ones(n)
    for m in range(1, n_modes + 1):
        c = rng.uniform(-0.6, 0.6) / m
        phase = rng.uniform(0, 2 * np.pi)
        a += c * np.cos(2 * np.pi * m * k / n + phase)
    a = np.maximum(a, 0)
    S = np.sum(a)
    if S < 1e-10:
        return np.ones(n) / n
    return a / S

def raised_cosine_init(n, alpha=1.0):
    """Raised cosine shape."""
    k = np.arange(n)
    x = 2.0 * k / (n - 1) - 1.0
    a = 0.5 * (1.0 + np.cos(np.pi * x)) ** alpha
    a = np.maximum(a, 1e-10)
    S = np.sum(a)
    return a / S

def gaussian_init(n, sigma_frac=0.3):
    """Gaussian bump initialization."""
    k = np.arange(n)
    center = n / 2.0
    sigma = sigma_frac * n
    a = np.exp(-0.5 * ((k - center) / sigma) ** 2)
    S = np.sum(a)
    return a / S

def random_smooth_init(n, seed=0, bandwidth=3):
    """Random initialization smoothed with a Gaussian kernel."""
    rng = np.random.default_rng(seed)
    a = rng.random(n)
    # Smooth with moving average
    kernel = np.ones(bandwidth) / bandwidth
    a = np.convolve(a, kernel, mode='same')
    a = np.maximum(a, 0)
    S = np.sum(a)
    if S < 1e-10:
        return np.ones(n) / n
    return a / S

def refine_sequence(a, time_budget, verbose=False):
    """Full refinement pipeline for a single sequence."""
    n = len(a)

    S = np.sum(a)

    a = a / S
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    round_num = 0
    
    while time.monotonic() - start < time_budget - 2:
        round_num += 1
        remaining = time_budget - (time.monotonic() - start)
        if remaining < 4:
            break

        if remaining > 5:

                                                     tau_start=5.0, tau_end=150.0,

                best_obj = obj_sg
                best_a = a_sg.copy()
                if verbose:

"""
Optimized step-function coefficient search using soft-max smoothed objective
with gradient-based optimization, adaptive temperature annealing, LP polishing,
coordinate descent fine-tuning, and multi-start exploration to minimize
2*n*max(conv(a,a))/(sum(a))^2.
"""

import numpy as np
from scipy.optimize import linprog
from scipy.interpolate import interp1d
from scipy.sparse import coo_matrix
import time


def compute_conv(a):
    """Compute linear convolution a*a via FFT."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    conv = np.fft.irfft(fa * fa, size)
    return conv[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    S = float(np.sum(a))
    if S < 1e-10:
        return np.inf
    conv = compute_conv(a)
    M = float(np.max(conv))
    return 2.0 * n * M / (S * S)


def softmax_grad(a, tau):
    """
    Compute soft-max smoothed max(conv(a,a)) and its gradient w.r.t. a.
    Uses the convention sum(a)=1. Returns (soft_max_value, gradient).
    """
    n = len(a)
    conv = compute_conv(a)
    conv_max = np.max(conv)
    conv_shifted = conv - conv_max
    exp_conv = np.exp(tau * conv_shifted)
    Z = np.sum(exp_conv)
    p = exp_conv / Z
    soft_max = conv_max + np.log(Z) / tau

    # Gradient: d(soft_max)/da[i] = 2 * sum_{j=0}^{n-1} p[i+j] * a[j]
    # Computed via FFT: reverse p, convolve with a, extract appropriate part
    p_reversed = p[::-1]
    size = 1
    while size < (2 * n - 1) + n - 1:
        size <<= 1
    fp = np.fft.rfft(p_reversed, size)
    fa = np.fft.rfft(a, size)
    cross = np.fft.irfft(fp * fa, size)
    # grad[i] = 2 * cross[2n-2-i] for i=0..n-1
    grad_sm = 2.0 * cross[n - 1:2 * n - 1][::-1]

    return soft_max, grad_sm, conv


def full_objective_and_grad(a, tau):
    """
    Full objective 2n*SM/S^2 and its gradient w.r.t. a.
    Returns (objective, gradient).
    """
    n = len(a)
    S = float(np.sum(a))
    if S < 1e-10:
        return np.inf, np.zeros(n)
    soft_max, grad_sm, conv = softmax_grad(a, tau)
    obj = 2.0 * n * soft_max / (S * S)
    # d(obj)/da[i] = 2n/S^2 * grad_sm[i] - 4n*soft_max/S^3
    total_grad = (2.0 * n / (S * S)) * (grad_sm - 2.0 * soft_max / S)
    return obj, total_grad


def build_toeplitz_sparse(a):
    """Build sparse Toeplitz matrix for conv(a, g)."""
    n = len(a)
    rows = []
    cols = []
    vals = []
    for j in range(n):
        start = j
        end = min(j + n, 2 * n - 1)
        length = end - start
        rows.extend(range(start, end))
        cols.extend([j] * length)
        vals.extend(list(a[:length]))
    A = coo_matrix((vals, (rows, cols)), shape=(2 * n - 1, n)).tocsr()
    return A


def solve_lp_maxsum(a, M, time_limit=6):
    """Solve LP: max sum(g) s.t. conv(a,g) <= M, g >= 0."""
    n = len(a)
    A = build_toeplitz_sparse(a)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': time_limit})
        if res.status == 0:
            g = np.maximum(res.x, 0)
            if np.sum(g) > 1e-12:
                return g
    except Exception:
        pass
    return None


def softmax_gradient_descent(a, time_budget, tau_start=2000.0, tau_end=60000.0,
                             max_iters=5000, lr_start=0.08, momentum=0.85):
    """Gradient descent on soft-max smoothed objective with annealed temperature."""
    n = len(a)
    a = a.copy().astype(np.float64)
    S = np.sum(a)
    if S < 1e-10:
        return a, np.inf
    a = a / S

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    v = np.zeros(n)
    lr = lr_start
    stall = 0
    stall_threshold = 80

    for it in range(max_iters):
        if time.monotonic() - start > time_budget:
            break

        frac = min(1.0, (it + 1) / max_iters)
        tau = tau_start * (tau_end / tau_start) ** frac

        obj, grad = full_objective_and_grad(a, tau)
        if not np.isfinite(obj):
            break

        # Center gradient to preserve sum
        grad_centered = grad - np.mean(grad)

        # Momentum
        v = momentum * v + (1.0 - momentum) * grad_centered
        a_new = a - lr * v
        a_new = np.maximum(a_new, 0)
        S_new = np.sum(a_new)
        if S_new < 1e-10:
            break
        a_new = a_new / S_new

        obj_new = compute_obj(a_new)
        if obj_new < best_obj - 1e-12:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new
            stall = 0
        else:
            a = a_new
            stall += 1
            if stall > stall_threshold:
                lr *= 0.75
                v = 0.5 * v
                stall = 0
                if lr < 1e-10:
                    break

    return best_a, best_obj


def multiplicative_update_sm(a, time_budget, tau_start=2000.0, tau_end=60000.0,

        else:

def alternating_lp(a, max_iters=30, time_budget=30.0):
    """Alternating LP with multi-M line search."""
    n = len(a)
    a = a.copy().astype(np.float64)
    S = np.sum(a)
    if S < 1e-10:
        return a, np.inf
    a = a / S

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    stall = 0

    for it in range(max_iters):
        if time.monotonic() - start > time_budget:
            break

        conv = compute_conv(a)
        M = float(np.max(conv))

        # Try multiple M values for diversity
        M_values = [M, M * 1.002, M * 0.998]
        best_g = None
        for M_try in M_values:
            g = solve_lp_maxsum(a, M_try, time_limit=4)
            if g is not None:
                S_g = np.sum(g)
                if S_g > best_a.sum() * 1e-6:
                    g = g / S_g
                    if best_g is None or np.sum(g * a) > np.sum(best_g * a):
                        best_g = g

        if best_g is None:
            break

        g = best_g

        # Line search with geometric progression
        best_t = 0.0
        best_obj_ls = best_obj
        for t in [0.03, 0.07, 0.1, 0.15, 0.2, 0.3, 0.5, 0.7, 1.0]:
            a_try = (1.0 - t) * a + t * g
            a_try = np.maximum(a_try, 0)
            S_try = np.sum(a_try)
            if S_try < 1e-10:
                continue
            a_try = a_try / S_try
            obj_try = compute_obj(a_try)
            if obj_try < best_obj_ls:
                best_obj_ls = obj_try
                best_t = t

        if best_t > 0.0:
            a = (1.0 - best_t) * a + best_t * g
            a = np.maximum(a, 0)
            a = a / np.sum(a)
            best_obj = best_obj_ls
            best_a = a.copy()
            stall = 0
        else:
            stall += 1
            if stall > 8:
                # Perturb and try again
                rng = np.random.default_rng(it * 104729 + 7)
                a = a * np.exp(rng.normal(0, 0.05, n))
                a = np.maximum(a, 0)
                a = a / np.sum(a)
                stall = 0
            elif stall > 15:
                break

def cosine_init(n, n_modes=8, seed=0):
    """Generate a smooth cosine-series initialization."""
    rng = np.random.default_rng(seed)
    k = np.arange(n)
    a = np.ones(n)
    for m in range(1, n_modes + 1):
        c = rng.uniform(-0.6, 0.6) / m
        phase = rng.uniform(0, 2 * np.pi)
        a += c * np.cos(2 * np.pi * m * k / n + phase)
    a = np.maximum(a, 0)
    S = np.sum(a)
    if S < 1e-10:
        return np.ones(n) / n
    return a / S


def raised_cosine_init(n, alpha=1.0):
    """Raised cosine shape."""
    k = np.arange(n)
    x = 2.0 * k / (n - 1) - 1.0
    a = 0.5 * (1.0 + np.cos(np.pi * x)) ** alpha
    a = np.maximum(a, 1e-10)
    S = np.sum(a)
    return a / S


def gaussian_init(n, sigma_frac=0.3):
    """Gaussian bump initialization."""
    k = np.arange(n)
    center = n / 2.0
    sigma = sigma_frac * n
    a = np.exp(-0.5 * ((k - center) / sigma) ** 2)
    S = np.sum(a)
    return a / S


def two_peak_init(n, gap_frac=0.3):
    """Two Gaussian peaks."""
    k = np.arange(n)
    c1 = n * (0.5 - gap_frac / 2)
    c2 = n * (0.5 + gap_frac / 2)
    sigma = 0.15 * n
    a = np.exp(-0.5 * ((k - c1) / sigma) ** 2) + np.exp(-0.5 * ((k - c2) / sigma) ** 2)
    S = np.sum(a)
    return a / S


def random_smooth_init(n, seed=0, bandwidth=5):
    """Random initialization smoothed."""
    rng = np.random.default_rng(seed)
    a = rng.random(n)
    kernel = np.ones(bandwidth) / bandwidth

    a = np.maximum(a, 0)
    S = np.sum(a)
    if S < 1e-10:

    return a / S

                if verbose:
                    print(f"    R{round_num} LP: {best_obj:.6f}")

        # Coordinate descent polish
        remaining = time_budget - (time.monotonic() - start)
        if remaining > 6:
            cd_t = min(remaining * 0.1, 5.0)
            a_cd, obj_cd = coordinate_descent_polish(best_a, time_budget=cd_t)
            if obj_cd < best_obj:
                best_obj = obj_cd
                best_a = a_cd.copy()
                if verbose:
                    print(f"    R{round_num} CD: {best_obj:.6f}")

        # Perturbation escape
        remaining = time_budget - (time.monotonic() - start)
        if remaining > 8:
            rng = np.random.default_rng(round_num * 42 + 17)
            for pert_idx in range(2):
                if time.monotonic() - start > time_budget - 3:
                    break
                scale = rng.uniform(0.03, 0.12)
                a_pert = best_a * np.exp(rng.normal(0, scale, n))
                a_pert = np.maximum(a_pert, 0)
                S_p = np.sum(a_pert)
                if S_p < 1e-10:
                    continue
                a_pert = a_pert / S_p
                a_pert, obj_pert = softmax_gradient_descent(a_pert, time_budget=2.5,
                                                             tau_start=2000.0, tau_end=40000.0,
                                                             max_iters=250)
                if obj_pert < best_obj:
                    best_obj = obj_pert
                    best_a = a_pert.copy()
                    if verbose:
                        print(f"    R{round_num} Pert{pert_idx}: {best_obj:.6f}")

    return best_a, best_obj


def try_parent():
    """Try to load parent sequence from global variable."""
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64)
        parent = np.clip(parent, 0, 1000.0)
        if parent.size > 0 and np.isfinite(parent).all():
            obj = compute_obj(parent)
            print(f"Parent (n={parent.size}): obj={obj:.6f}")
            return parent.copy(), obj
    except Exception:
        pass
    return None, np.inf


def propose_candidate(seed=42, budget_s=1500):
    """
    Main search function. Combines soft-max smoothed gradient descent,
    multiplicative updates, LP-based polishing, coordinate descent,
    and multi-start exploration to minimize 2*n*max(conv(a,a))/(sum(a))^2.
    """
    start_time = time.monotonic()
    deadline = start_time + float(budget_s) - 20.0

    best_seq = None
    best_score = np.inf

    # Load parent
    parent, parent_obj = try_parent()
    if parent is not None:
        best_score = parent_obj
        best_seq = parent.copy()
        n_parent = len(parent)
    else:
        n_parent = 400

    print(f"Initial best: {best_score:.6f} (n={n_parent if best_seq is not None else 'N/A'})")

    # === Phase 1: Diverse initialization search ===
    print("\n=== Phase 1: Diverse initializations ===")
    n_candidates = sorted(set([
        max(100, n_parent - 150), max(100, n_parent - 50),
        n_parent,
        min(1000, n_parent + 100), min(1200, n_parent + 200)
    ]))

    candidate_pool = []

    for n in n_candidates:
        if time.monotonic() > deadline:
            break
        remaining = deadline - time.monotonic()
        if remaining < 15:
            break

        inits = [
            (f"cosine_{n}", cosine_init(n, n_modes=8, seed=seed + n)),
            (f"rcos_{n}", raised_cosine_init(n, alpha=1.5)),
            (f"rcos2_{n}", raised_cosine_init(n, alpha=2.5)),
            (f"rcos3_{n}", raised_cosine_init(n, alpha=4.0)),
            (f"gauss_{n}", gaussian_init(n, sigma_frac=0.25)),
            (f"gauss2_{n}", gaussian_init(n, sigma_frac=0.4)),
            (f"tpk_{n}", two_peak_init(n, gap_frac=0.3)),
            (f"rand_{n}", random_smooth_init(n, seed=seed + n, bandwidth=5)),
        ]

        for name, a_init in inits:
            if time.monotonic() > deadline:
                break
            obj_init = compute_obj(a_init)
            a_quick, obj_quick = softmax_gradient_descent(a_init, time_budget=2.0,
                                                          tau_start=2000.0, tau_end=30000.0,
                                                          max_iters=200)
            best_local = a_quick if obj_quick < obj_init else a_init
            best_local_obj = min(obj_quick, obj_init)
            candidate_pool.append((best_local, best_local_obj))
            if best_local_obj < best_score:
                best_score = best_local_obj
                best_seq = best_local.copy()
                print(f"  {name} (n={n}): {best_local_obj:.6f} ** NEW BEST **")

    # === Phase 2: Refine top candidates ===
    print("\n=== Phase 2: Refine top candidates ===")
    candidate_pool.sort(key=lambda x: x[1])

    n_refine = min(5, len(candidate_pool))
    for idx in range(n_refine):
        if time.monotonic() > deadline:
            break
        remaining = deadline - time.monotonic()
        if remaining < 25:
            break
        a_cand, obj_cand = candidate_pool[idx]
        if obj_cand > best_score + 0.04:
            break

        n_cand = len(a_cand)
        alloc = min(remaining * 0.2, 50.0)
        print(f"  Refining candidate {idx} (n={n_cand}, obj={obj_cand:.6f}), alloc={alloc:.0f}s")

        a_ref, obj_ref = refine_sequence(a_cand, time_budget=alloc, verbose=True)
        if obj_ref < best_score:
            best_score = obj_ref
            best_seq = a_ref.copy()
            print(f"    ** NEW BEST: n={n_cand}, obj={best_score:.6f} **")

    # === Phase 3: Intensive refinement loop ===
    print("\n=== Phase 3: Intensive refinement ===")
    a_cur = best_seq.copy()
    n_best = len(a_cur)

    for round_num in range(25):
        if time.monotonic() > deadline - 25:
            break
        remaining = deadline - time.monotonic()
        if remaining < 15:
            break

        alloc = min(remaining * 0.17, 45.0)
        a_new, obj_new = refine_sequence(a_cur, time_budget=alloc, verbose=False)

        if obj_new < best_score:
            best_score = obj_new
            best_seq = a_new.copy()
            a_cur = a_new.copy()
            print(f"  R{round_num}: {best_score:.6f} ** NEW BEST **")
        else:
            if round_num % 3 == 0:
                print(f"  R{round_num}: {obj_new:.6f} (best={best_score:.6f})")
            # Perturbation escape
            rng = np.random.default_rng(round_num * 777 + 333)
            improved = False
            for pert_scale in [0.05, 0.1, 0.15, 0.25]:
                if time.monotonic() > deadline - 15:
                    break
                a_pert = a_cur * np.exp(rng.normal(0, pert_scale, n_best))
                a_pert = np.maximum(a_pert, 0)
                S_p = np.sum(a_pert)
                if S_p < 1e-10:
                    continue
                a_pert = a_pert / S_p
                a_pert, obj_pert = refine_sequence(a_pert, time_budget=min(15.0, remaining * 0.1),
                                                    verbose=False)
                if obj_pert < best_score:
                    best_score = obj_pert
                    best_seq = a_pert.copy()
                    a_cur = a_pert.copy()
                    print(f"  R{round_num} pert({pert_scale}): {best_score:.6f} ** NEW BEST **")
                    improved = True
                    break

    # === Phase 4: Multi-n interpolation ===
    print("\n=== Phase 4: Multi-n exploration ===")
    if time.monotonic() < deadline - 25:
        n_base = len(best_seq)
        n_tries = sorted(set([
            max(80, n_base - 150), max(80, n_base - 75),
            n_base + 50, n_base + 100, n_base + 200,
            max(100, n_base + 400)
        ]))
        n_tries = [n for n in n_tries if 80 <= n <= 2000]

        for n_try in n_tries:
            if time.monotonic() > deadline - 12:
                break
            remaining = deadline - time.monotonic()
            if remaining < 12:
                break

            x_old = np.linspace(0, 1, n_base)
            x_new = np.linspace(0, 1, n_try)
            try:
                f_interp = interp1d(x_old, best_seq, kind='cubic', fill_value='extrapolate')
                a_interp = np.maximum(f_interp(x_new), 0)
            except Exception:
                a_interp = np.ones(n_try) / n_try

            S_i = np.sum(a_interp)
            if S_i > 1e-10:
                a_interp = a_interp / S_i

            alloc = min(remaining * 0.2, 25.0)
            a_new, obj_new = refine_sequence(a_interp, time_budget=alloc, verbose=False)
            if obj_new < best_score:
                best_score = obj_new
                best_seq = a_new.copy()
                print(f"  n={n_try}: {best_score:.6f} ** NEW BEST **")
            else:
                print(f"  n={n_try}: {obj_new:.6f}")

    # === Phase 5: Final intensive polish ===
    print("\n=== Phase 5: Final polish ===")
    a_cur = best_seq.copy()
    n_final = len(a_cur)

    if time.monotonic() < deadline - 10:
        remaining = deadline - time.monotonic()

        # High-temperature soft-max gradient (nearly exact subgradient)
        a_sg, obj_sg = softmax_gradient_descent(a_cur, time_budget=remaining * 0.2,
                                                 tau_start=5000.0, tau_end=200000.0,
                                                 max_iters=int(remaining * 0.2 * 30))
        if obj_sg < best_score:
            best_score = obj_sg
            best_seq = a_sg.copy()
            a_cur = a_sg.copy()
            print(f"  SG high-tau: {best_score:.6f} ** NEW BEST **")

        # LP polish
        remaining = deadline - time.monotonic()
        if remaining > 12 and n_final <= 600:
            a_lp, obj_lp = alternating_lp(a_cur, max_iters=35, time_budget=min(remaining * 0.35, 35.0))
            if obj_lp < best_score:
                best_score = obj_lp
                best_seq = a_lp.copy()
                a_cur = a_lp.copy()
                print(f"  Final LP: {best_score:.6f} ** NEW BEST **")

        # Coordinate descent
        remaining = deadline - time.monotonic()
        if remaining > 8:
            a_cd, obj_cd = coordinate_descent_polish(a_cur, time_budget=min(remaining * 0.2, 8.0))
            if obj_cd < best_score:
                best_score = obj_cd
                best_seq = a_cd.copy()
                print(f"  Final CD: {best_score:.6f} ** NEW BEST **")

        # Final perturbation rounds
        remaining = deadline - time.monotonic()
        if remaining > 12:
            rng = np.random.default_rng(99999)
            for _ in range(4):
                if time.monotonic() > deadline - 8:
                    break
                a_p = a_cur * np.exp(rng.normal(0, 0.04, n_final))
                a_p = np.maximum(a_p, 0)
                S_p = np.sum(a_p)
                if S_p > 1e-10:
                    a_p = a_p / S_p
                    a_p, obj_p = softmax_gradient_descent(a_p, time_budget=min(remaining * 0.15, 10.0),
                                                          tau_start=5000.0, tau_end=150000.0,
                                                          max_iters=800)
                    if obj_p < best_score:
                        best_score = obj_p
                        best_seq = a_p.copy()
                        a_cur = a_p.copy()
                        print(f"  Perturb: {best_score:.6f} ** NEW BEST **")

    # Final validation
    if best_seq is None:
        best_seq = np.ones(300) / 300
        best_score = compute_obj(best_seq)

    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    val = evaluate_sequence(result)
    if val < np.inf and val < best_score:
        best_score = val
    elif val == np.inf:
        result = [1.0] * 300
        best_score = 2.0

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_score:.6f}, time={elapsed:.1f}s ===")

    return result
