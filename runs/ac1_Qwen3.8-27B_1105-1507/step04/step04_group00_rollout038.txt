def fast_conv(a):
    """Full self-convolution via FFT."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]


def compute_J(a):
    """Compute J(a) = 2*n*max(conv(a,a))/(sum(a))^2."""
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))


def smooth_max_gradient(a, tau):
    """Compute gradient of smooth max of autocorrelation."""
    n = len(a)
    b = fast_conv(a)
    b_max = b.max()
    exp_b = np.exp(tau * (b - b_max))
    p = exp_b / exp_b.sum()
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        pk = p[k]
        if pk < 1e-13:
            continue
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        grad[lo:hi + 1] += 2.0 * pk * a[k - hi:k - lo + 1][::-1]
    return grad, p, b

        tau = initial_tau + (final_tau - initial_tau) * (frac ** 0.5)

        a = a_new

            # For the mask part, we need max(b_without[mask] + 2*xp*a[a_idx])
            # This is max of linear functions - can be computed as a dot product with max

            def eval_obj(xp):
                S_new = S - x + xp
                if S_new < 1e-14:
                    return 1e30
                # Compute max of b_trial
                # Part 1: k not in mask -> b_without[k] (max is b_without_max, but might include mask elements)
                # Simpler: just compute the full max
                # b_trial = b_without + 2*xp*a[a_idx] on mask, + xp^2 - 2*xp*x at 2*j0
                # Max over all k:
                # For k in mask, k != 2*j0: b_without[k] + 2*xp*a[k-j0]
                # For k = 2*j0 (if in mask): b_without[2*j0] + xp^2
                # For k not in mask: b_without[k]

                # Vectorized computation:
                # max_mask = max(b_without[mask] + 2*xp*a[a_idx])
                # Fix for k=2*j0
                # max_out = max(b_without[k] for k not in mask) -- approx by b_without.max()
                #   (this is an upper bound; the true max_out <= b_without.max())
                # Actually, for correctness, let's just compute:
                # candidate1 = max(b_without[mask] + 2*xp*a[a_idx])
                # candidate2 = b_without[2*j0] + xp^2 (if 2*j0 in mask)
                # candidate3 = max of b_without outside mask
                # result = max(candidate1_with_fix, candidate2, candidate3)

                # Efficient approach:
                # vals_mask = b_without[mask] + 2*xp*a[a_idx]
                # If 2*j0 in mask: vals_mask[2*j0-k_lo] = b_without[2*j0] + xp^2 (recompute)
                # max_val = max(vals_mask.max(), b_without_max)
                #   But b_without_max might be from a mask element, in which case vals_mask already covers it
                #   Actually b_without_max = max of ALL elements including mask
                #   The non-mask elements have unchanged values, so max_outside = max(b_without outside mask)
                #   This is <= b_without.max()

                # For a tight bound (which we need for correctness):
                # max_trial = max(max_inside, max_outside)
                # max_inside = max(b_without[mask] + 2*xp*a[a_idx], with fix at 2*j0)
                # max_outside = max(b_without[k] for k not in mask) <= b_without.max()

                # Since b_without.max() is an upper bound for max_outside, and max_inside >= b_without[mask].max()
                # (because 2*xp*a[a_idx] >= 0), we have:
                # max_trial = max(max_inside, b_without.max())
                # Wait, that's not right either. b_without.max() could be from a mask element.
                # Let's just say: max_trial >= max_inside (always true)
                # And max_trial = max(max_inside, max_outside) where max_outside <= b_without.max()
                # So max_trial = max(max_inside, min(b_without.max(), max_outside))
                # This is getting circular. Let me just compute it directly.

                # Direct computation (O(n) but only for the max):
                # Build b_trial efficiently
                # Actually, let's precompute b_without_outside_max once outside the loop
                pass

            # Let me restructure: precompute what we can
            # b_without is fixed. The only thing that changes with xp is:
            #   b_trial[k] = b_without[k] + 2*xp*a[a_idx] for k in mask (k!=2j0)
            #   b_trial[2*j0] = b_without[2*j0] + xp^2
            #   b_trial[k] = b_without[k] for k not in mask

            # So: max(b_trial) = max(
            #     max(b_without[k] + 2*xp*a[a_idx] for k in mask, k != 2*j0),
            #     b_without[2*j0] + xp^2  (if 2*j0 in mask),
            #     max(b_without[k] for k not in mask)
            #   )

            # Precompute: max_out = max(b_without[k] for k not in mask)
            # For this, create a mask array:
            in_mask = np.zeros(2 * n - 1, dtype=bool)
            in_mask[mask] = True
            max_out = np.max(b_without[~in_mask]) if np.any(~in_mask) else 0.0

            # For the mask part (excluding 2*j0):
            if k_lo <= 2 * j0 <= k_hi:
                # Exclude 2*j0 from mask for the linear part
                mask_idx = mask - k_lo  # indices into mask array
                two_j0_pos = 2 * j0 - k_lo
                # Linear part: all mask positions except two_j0_pos
                linear_mask_a = np.delete(a[a_idx], two_j0_pos)
                linear_mask_b = np.delete(b_without[mask], two_j0_pos)
            else:
                linear_mask_a = a[a_idx].copy()
                linear_mask_b = b_without[mask].copy()

            def eval_obj2(xp):
                S_new = S - x + xp
                if S_new < 1e-14:
                    return 1e30
                # Max of linear part on mask
                if len(linear_mask_a) > 0:
                    max_linear = np.max(linear_mask_b + 2.0 * xp * linear_mask_a)
                else:
                    max_linear = 0.0
                # Quadratic part at 2*j0
                if k_lo <= 2 * j0 <= k_hi:
                    max_quad = b_without[2 * j0] + xp * xp
                else:
                    max_quad = 0.0
                # Outside mask
                max_total = max(max_linear, max_quad, max_out)
                return max_total / (S_new * S_new)

            # Golden section search on [0, x_upper]
            lo = 0.0
            hi = x_upper
            c = hi - gr * (hi - lo)
            d = lo + gr * (hi - lo)
            fc = eval_obj2(c)
            fd = eval_obj2(d)

            for _ in range(50):
                if fc < fd:
                    hi = d
                    d = c
                    fd = fc
                    c = hi - gr * (hi - lo)
                    fc = eval_obj2(c)
                else:
                    lo = c
                    c = d
                    fc = fd
                    d = lo + gr * (hi - lo)
                    fd = eval_obj2(d)

            x_opt = (lo + hi) / 2.0
            f_opt = eval_obj2(x_opt)

            # Also check boundaries
            f_lo = eval_obj2(0.0)
            f_hi = eval_obj2(x_upper)
            if f_lo < f_opt:
                x_opt = 0.0
                f_opt = f_lo
            if f_hi < f_opt:
                x_opt = x_upper
                f_opt = f_hi

            if f_opt < sweep_best_obj * (1.0 - 1e-13):
                # Apply update
                old_x = x
                a[j0] = x_opt
                S = S - old_x + x_opt
                b = fast_conv(a)
                current_obj = b.max() / (S * S)
                if current_obj < best_obj:
                    best_obj = current_obj
                    best_a = a.copy()
                sweep_best_obj = min(sweep_best_obj, current_obj)

        # Track convergence
        if sweep_best_obj >= best_obj * (1.0 - 1e-12):
            no_improve_sweeps += 1
        else:
            no_improve_sweeps = 0

        if no_improve_sweeps > max_no_improve:
            break

        if verbose and sweep % 200 == 0:
            print(f"      CD sweep {sweep}: obj={best_obj:.10f} (J={2*n*best_obj:.8f})", flush=True)

    # Return normalized
    s_final = best_a.sum()
    if s_final > 1e-10:
        best_a = best_a / s_final
    return best_a


def build_A_sparse(n, a):
    """Build sparse constraint matrix for LP: A[k,j] = a[k-j] if 0<=k-j<n."""
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        j_min = max(0, k - n + 1)
        j_max = min(n - 1, k)
        for j in range(j_min, j_max + 1):
            rows.append(k)
            cols.append(j)
            vals.append(a[k - j])
    A = csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))
    return A


def lp_best_response(a, M):
    """LP: max sum(g) s.t. (a*g)[k] <= M for all k, g >= 0."""
    n = len(a)
    A = build_A_sparse(n, a)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    try:
        r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                    options={'time_limit': 20, 'feasibility_tolerance': 1e-9})
        if r.success and r.x is not None:
            g = np.maximum(r.x, 0)
            if g.sum() > 1e-15:
                return g
    except Exception:
        pass
    return None


def run_lp_mirror_descent(a, max_iters, deadline):
    """LP mirror descent with geometric line search."""
    a = np.maximum(a.astype(np.float64), 1e-8)
    a = a / a.sum()
    best_a = a.copy()
    best_M = fast_conv(a).max()
    no_improve = 0

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        b = fast_conv(a)
        M = float(b.max())
        g = lp_best_response(a, M)
        if g is None:
            break
        g = np.maximum(g, 0.0)
        g_sum = g.sum()
        if g_sum < 1e-10:
            break
        g /= g_sum

        improved = False
        base_t = max(0.05, 0.5 * (1 - 0.5 * it / max(1, max_iters)))
        for t in [base_t, base_t * 0.5, base_t * 0.25, 0.02, 0.01]:
            a_trial = np.maximum((1 - t) * a + t * g, 0)
            s = a_trial.sum()
            if s < 1e-10:
                continue
            a_trial /= s
            M_trial = fast_conv(a_trial).max()
            if M_trial < best_M - 1e-14:
                a = a_trial
                best_M = M_trial
                best_a = a.copy()
                improved = True
                break

        if not improved:
            no_improve += 1
            if no_improve > 15:
                break
        else:
            no_improve = 0

    return best_a


def resample_seq(src, dest_n):
    """Resample 1D sequence to new length via linear interpolation."""
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)


def generate_initializations(n, rng, parent=None):
    """Generate diverse initial sequences of length n."""
    inits = []
    t_arr = np.linspace(0, 1, n, endpoint=False)

    if parent is not None:
        p = resample_seq(parent, n)
        inits.append(p)
        inits.append(np.maximum(p + rng.normal(0, 0.02, n), 0.01))
        inits.append(np.maximum(p + rng.normal(0, 0.08, n), 0.01))

    inits.append(np.ones(n))
    inits.append(rng.uniform(0.5, 1.0, n))

    for freq in [1, 2, 3, 5, 8]:
        inits.append(np.maximum(1.0 + 0.3 * np.cos(2 * np.pi * freq * t_arr), 0.01))
        inits.append(np.maximum(1.0 + 0.4 * np.sin(2 * np.pi * freq * t_arr), 0.01))

    inits.append(np.maximum(1.0 + 0.3 * np.cos(2 * np.pi * 3 * t_arr ** 2), 0.01))
    inits.append(np.exp(-3.0 * (t_arr - 0.5) ** 2) + 0.1)
    inits.append(np.exp(-6.0 * (t_arr - 0.3) ** 2) + np.exp(-6.0 * (t_arr - 0.7) ** 2) + 0.05)
    inits.append(0.2 + np.exp(-10 * (t_arr - 0.2) ** 2) + np.exp(-10 * (t_arr - 0.8) ** 2))
    inits.append(np.maximum(1.0 - 0.5 * np.abs(2 * t_arr - 1), 0.05))
    inits.append(0.3 + 0.7 * np.abs(2 * t_arr - 1))

    ksize = max(3, n // 25)
    kernel = np.ones(ksize) / ksize
    smooth = np.convolve(rng.uniform(0, 1, n), kernel, mode='same')
    inits.append(np.maximum(smooth + 0.1, 0.01))

    # Alternating pattern
    alt = np.ones(n)
    alt[::2] = 0.3
    inits.append(alt)

    # Three-level pattern
    three = np.tile([1.0, 0.5, 0.8], n // 3 + 1)[:n]
    inits.append(three)

    return inits


class ScoreTracker:
    """Tracks the best sequence found so far."""

    def __init__(self):
        self.best = None
        self.best_score = float('inf')
        self.t_start = time.monotonic()

    def update(self, a):
        a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
        if a.size == 0 or a.sum() < 0.01:
            return
        score = float(evaluate_sequence(a.tolist()))
        if np.isfinite(score) and score < self.best_score:
            self.best = a.copy()
            self.best_score = score
            elapsed = time.monotonic() - self.t_start
            print(f"  ** J={score:.8f} n={a.size} t={elapsed:.0f}s", flush=True)

    def report(self, label=""):
        if self.best is not None:
            print(f"  {label} best: J={self.best_score:.8f} n={self.best.size}", flush=True)


def propose_candidate(seed=42, budget_s=1500, **kwargs):
    """
    Multi-phase search:
    Phase 1: Coordinate descent on multiple lengths (primary, fast)
    Phase 2: Smooth gradient for exploration
    Phase 3: LP mirror descent polishing
    Phase 4: Final coordinate descent refinement
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + float(budget_s) - 20.0
    tracker = ScoreTracker()

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        if p.ndim == 1 and p.size > 0 and np.all(np.isfinite(p)):
            parent = np.maximum(p, 0.0)
            tracker.update(parent)
            print(f"Parent loaded: n={parent.size}, J={tracker.best_score:.6f}", flush=True)
    except (NameError, AttributeError):
        pass

    if parent is None:
        parent = np.ones(256)
        tracker.update(parent)

    current_best = parent.copy()

    # === PHASE 1: Coordinate descent across multiple lengths ===
    lengths = [128, 192, 256, 384, 512, 640, 768, 1024]
    lengths = [n for n in lengths if 64 <= n <= 2048]

    p1_budget = 800.0
    p1_start = time.monotonic()
    print(f"\n{'='*60}", flush=True)
    print(f"PHASE 1: Coordinate Descent (budget={p1_budget:.0f}s)", flush=True)
    print(f"{'='*60}", flush=True)

    for n in lengths:
        if time.monotonic() > deadline - 200:
            break

        # Time allocation: more time for lengths near the best
        if tracker.best is not None and abs(n - tracker.best.size) < 200:
            time_for_n = min(120, p1_budget * (6.0 / n) / sum(6.0 / nn for nn in lengths) * 2)
        else:
            time_for_n = min(80, p1_budget * (4.0 / n) / sum(4.0 / nn for nn in lengths))
        time_for_n = max(15, time_for_n)
        phase_deadline = min(time.monotonic() + time_for_n, deadline - 150)

        if time.monotonic() >= phase_deadline:
            continue

        t0 = time.monotonic()
        inits = generate_initializations(n, rng, current_best)
        best_for_n = None
        best_M_for_n = float('inf')

        for idx, a0 in enumerate(inits):
            if time.monotonic() > phase_deadline:
                break
            a0 = np.maximum(a0.astype(np.float64), 1e-6)
            s0 = a0.sum()
            if s0 < 1e-10:
                continue
            a0 = a0 / s0

            # First do a quick smooth gradient to get into the basin
            iters_sm = max(30, int(1000 / max(1, n // 128)))
            a_sm = run_smooth_gradient(a0, iters_sm, min(time.monotonic() + 5, phase_deadline),
                                       initial_tau=3.0, final_tau=30.0,
                                       lr_init=0.15, lr_min=0.03)

            # Then coordinate descent (the main workhorse)
            max_cd_sweeps = int(5000 / max(1, n // 256))
            a_cd = coordinate_descent(a_sm, max_cd_sweeps, phase_deadline, seed=seed + idx)

            M_val = fast_conv(a_cd / a_cd.sum()).max()
            if M_val < best_M_for_n:
                best_M_for_n = M_val
                best_for_n = a_cd.copy()

        if best_for_n is not None:
            J_val = compute_J(best_for_n)
            print(f"  n={n:4d}: M={best_M_for_n:.8f} J={J_val:.6f} "
                  f"({time.monotonic()-t0:.1f}s)", flush=True)
            if J_val < compute_J(current_best):
                current_best = best_for_n.copy()
            tracker.update(best_for_n)

    tracker.report("Phase 1 complete")
    print(f"Phase 1 total time: {time.monotonic()-p1_start:.1f}s", flush=True)

    # === PHASE 2: More coordinate descent on the best length ===
    p2_start = time.monotonic()
    print(f"\n{'='*60}", flush=True)
    print(f"PHASE 2: Extended CD on Best (budget=~200s)", flush=True)
    print(f"{'='*60}", flush=True)

    if tracker.best is not None:
        n_best = tracker.best.size
        a = coordinate_descent(tracker.best, 20000, min(time.monotonic() + 200, deadline - 100),
                               verbose=True, seed=seed + 100)
        tracker.update(a)

        # Try with small perturbation
        a_pert = tracker.best.copy() + rng.normal(0, 0.01, tracker.best.size)
        a_pert = np.maximum(a_pert, 0.001)
        a_pert = a_pert / a_pert.sum()
        a = coordinate_descent(a_pert, 10000, min(time.monotonic() + 100, deadline - 80),
                               seed=seed + 200)
        tracker.update(a)

    tracker.report("Phase 2 complete")
    print(f"Phase 2 total time: {time.monotonic()-p2_start:.1f}s", flush=True)

    # === PHASE 3: LP Mirror Descent polishing ===
    p3_start = time.monotonic()
    print(f"\n{'='*60}", flush=True)
    print(f"PHASE 3: LP Mirror Descent (budget=~200s)", flush=True)
    print(f"{'='*60}", flush=True)

    if tracker.best is not None:
        n_best = tracker.best.size
        est_lp_time = 0.01 * (n_best / 128) ** 2
        max_iters = max(20, min(300, int(180 / max(0.01, est_lp_time))))
        print(f"  n={n_best}, est LP iter time={est_lp_time*1000:.0f}ms, max_iters={max_iters}", flush=True)

        a = run_lp_mirror_descent(tracker.best, max_iters, min(time.monotonic() + 180, deadline - 60))
        tracker.update(a)

        # CD after LP
        if time.monotonic() < deadline - 40:
            a2 = coordinate_descent(a, 10000, min(time.monotonic() + 80, deadline - 40),
                                    seed=seed + 300)
            tracker.update(a2)

    tracker.report("Phase 3 complete")
    print(f"Phase 3 total time: {time.monotonic()-p3_start:.1f}s", flush=True)

    # === PHASE 4: Final refinement ===
    p4_start = time.monotonic()
    print(f"\n{'='*60}", flush=True)
    print(f"PHASE 4: Final Refinement (remaining time)", flush=True)
    print(f"{'='*60}", flush=True)

    if tracker.best is not None and time.monotonic() < deadline - 10:
        # Long CD run
        a = coordinate_descent(tracker.best, 50000, min(time.monotonic() + 150, deadline - 30),
                               verbose=True, seed=seed + 400)
        tracker.update(a)

        # One more LP pass if time allows
        if time.monotonic() < deadline - 15:
            n_best = tracker.best.size
            a2 = run_lp_mirror_descent(tracker.best, 50, deadline - 10)
            tracker.update(a2)
            # CD after
            if time.monotonic() < deadline - 5:
                a3 = coordinate_descent(a2, 5000, deadline, seed=seed + 500)
                tracker.update(a3)

    tracker.report("Phase 4 complete")
    print(f"Phase 4 total time: {time.monotonic()-p4_start:.1f}s", flush=True)

    # Fallback
    if tracker.best is None:
        tracker.best = np.ones(128)
        tracker.best_score = compute_J(tracker.best)

    print(f"\n{'='*60}", flush=True)
    print(f"FINAL: J={tracker.best_score:.8f} n={tracker.best.size} "
          f"time={time.monotonic()-tracker.t_start:.0f}s", flush=True)
    print(f"{'='*60}", flush=True)

    return [float(x) for x in tracker.best]
