"""

2. Minimax primal-dual method with QP inner loop (theoretically grounded)

Key innovation: the gradient of the log-sum-exp smoothed objective is computed
via a single FFT convolution, enabling 100x more iterations in the same time.

import numpy as np
from scipy.optimize import linprog
from scipy.sparse import csr_matrix
import time
import cvxpy as cp


def fast_conv(a):
    """Full self-convolution via FFT."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]


def compute_J(a):
    """Compute J(a) = 2*n*max(conv(a,a))/(sum(a))^2."""
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))


def fft_smooth_gradient(a, p):
    """
    Compute grad[i] = 2 * sum_{k=0}^{2n-2} p[k] * a[k-i] for valid indices.
    Equivalent to: grad[i] = 2 * sum_{j=0}^{n-1} a[j] * p[i+j]
    Computed via FFT in O(n log n).
    """
    n = len(a)
    size = 1
    while size < 3 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a[::-1], size)
    fp = np.fft.rfft(p, size)
    conv_result = np.fft.irfft(fa * fp, size)
    grad = 2.0 * conv_result[n - 1:2 * n - 1]
    return grad


def compute_smooth_objective_and_grad(a, tau):
    """Compute smoothed objective and its gradient via FFT."""
    n = len(a)
    b = fast_conv(a)
    b_max = b.max()
    exp_b = np.exp(tau * (b - b_max))
    p = exp_b / exp_b.sum()
    obj = (np.log(exp_b.sum()) / tau) + b_max
    grad = fft_smooth_gradient(a, p)
    return obj, grad, p, b


def run_fft_smooth_gradient(a, max_iters, deadline, tau_init=5.0, tau_final=200.0,
                            lr_init=0.5, lr_min=0.05, momentum_beta=0.9):
    """
    Fast smooth gradient descent with FFT-accelerated gradients,
    Nesterov momentum, and adaptive tau/lr schedules.
    """
    a = np.maximum(a.astype(np.float64), 1e-10)
    a = a / a.sum()
    n = len(a)

    best_a = a.copy()
    best_M = fast_conv(a).max()
    v = np.zeros(n)
    y = a.copy()  # Nesterov lookahead

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        frac = (it + 1) / max(1, max_iters)
        # Anneal tau: smooth → sharp
        tau = tau_init * (tau_final / tau_init) ** (frac ** 0.7)
        # Learning rate: decay
        lr = lr_init * (lr_min / lr_init) ** frac

        # Nesterov: evaluate gradient at lookahead point
        y = a + momentum_beta * v
        y = np.maximum(y, 1e-10)
        y = y / y.sum()

        obj, grad, p, b = compute_smooth_objective_and_grad(y, tau)
        M = b.max()

        if M < best_M * (1 - 1e-9):
            best_M = M
            best_a = a.copy()

        # Nesterov update
        v = momentum_beta * v - lr * grad
        a_new = a + v
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s < 1e-10:
            a_new = a.copy()
        else:
            a_new = a_new / s
        a = a_new
        v = v * 0.995  # slight damping for stability

    # Final check
    M_final = fast_conv(a).max()
    if M_final < best_M:
        best_a = a.copy()
    return best_a


def build_hankel_matrix(p):
    """Build the n x n matrix M where M[i,j] = p[i+j]."""
    n = p.size // 2 + (p.size % 2)  # p has length 2n-1, so n = (len(p)+1)//2
    n = (p.size + 1) // 2
    M = np.empty((n, n), dtype=np.float64)
    for i in range(n):
        for j in range(n):
            M[i, j] = p[i + j]
    return M

def minimax_primal_dual(a_init, max_iters, deadline, gamma_init=2.0, gamma_decay=0.995):
    """
    Minimax primal-dual algorithm:
    - Dual variable p (distribution over convolution positions)
    - Primal: solve QP min a^T M(p) a s.t. a>=0, sum(a)=1
    - Dual update: p[k] ∝ p[k] * exp(gamma * (a*a)[k])
    """
    n = len(a_init)
    a = np.maximum(a_init.astype(np.float64), 1e-8)
    a = a / a.sum()

    p = np.ones(2 * n - 1) / (2 * n - 1)
    gamma = gamma_init

    best_a = a.copy()
    best_M = fast_conv(a).max()

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        # Primal step: solve inner QP
        M = build_hankel_matrix(p)
        n_q = len(a)
        a_var = cp.Variable(n_q)
        objective = cp.Minimize(cp.quad_form(a_var, cp.psd_wrap(M)))
        constraints = [a_var >= 0, cp.sum(a_var) == 1]
        prob = cp.Problem(objective, constraints)
        try:
            prob.solve(solver=cp.CLARARABEL, verbose=False)
        except Exception:
            try:
                prob.solve(solver=cp.SCS, verbose=False)
            except Exception:
                break

        if a_var.value is None:
            break
        a_new = np.maximum(a_var.value, 0)
        s = a_new.sum()
        if s < 1e-10:
            break
        a_new = a_new / s

        M_val = fast_conv(a_new).max()
        if M_val < best_M:
            best_M = M_val
            best_a = a_new.copy()

        # Dual update: exponentiated gradient
        b = fast_conv(a_new)
        b_scaled = gamma * (b - b.min())  # shift for stability
        p_new = p * np.exp(b_scaled)
        p_sum = p_new.sum()
        if p_sum < 1e-10 or not np.isfinite(p_sum):
            break
        p = p_new / p_sum

        a = a_new
        gamma *= gamma_decay

    return best_a


def build_A_sparse(n, a):
    """Build sparse constraint matrix for LP: A[k,j] = a[k-j] if 0<=k-j<n."""
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        j_min = max(0, k - n + 1)
        j_max = min(n - 1, k)
        for j in range(j_min, j_max + 1):
            rows.append(k)
            cols.append(j)
            vals.append(a[k - j])
    A = csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))
    return A


def lp_best_response(a, M):
    """LP: max sum(g) s.t. (a*g)[k] <= M for all k, g >= 0."""
    n = len(a)
    A = build_A_sparse(n, a)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    try:
        r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                    options={'time_limit': 30})
        if r.success and r.x is not None:
            g = np.maximum(r.x, 0)
            if g.sum() > 1e-15:
                return g
    except Exception:
        pass
    return None


def run_lp_mirror_descent(a, max_iters, deadline):
    """LP mirror descent with geometric line search."""
    a = np.maximum(a.astype(np.float64), 1e-8)
    a = a / a.sum()
    best_a = a.copy()
    best_M = fast_conv(a).max()
    no_improve = 0

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        b = fast_conv(a)
        M = float(b.max())
        g = lp_best_response(a, M)
        if g is None:
            break
        g = np.maximum(g, 0.0)
        g_sum = g.sum()
        if g_sum < 1e-10:
            break
        g /= g_sum

        improved = False
        base_t = max(0.05, 0.5 * (1 - 0.3 * it / max(1, max_iters)))
        for t in [base_t, base_t * 0.5, base_t * 0.25, 0.02, 0.01]:
            a_trial = np.maximum((1 - t) * a + t * g, 0)
            s = a_trial.sum()
            if s < 1e-10:
                continue
            a_trial /= s
            M_trial = fast_conv(a_trial).max()
            if M_trial < best_M - 1e-14:
                a = a_trial
                best_M = M_trial
                best_a = a.copy()
                improved = True
                break

        if not improved:
            no_improve += 1
            if no_improve > 30:
                break
        else:
            no_improve = 0

    return best_a


def resample_seq(src, dest_n):
    """Resample 1D sequence to new length via cubic interpolation."""
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    # Use scipy for cubic
    from scipy.interpolate import interp1d
    f = interp1d(t_old, src, kind='cubic', fill_value="extrapolate")
    result = f(t_new)
    return np.maximum(result, 0.0)

    smooth = np.convolve(rng.uniform(0, 1, n), kernel, mode='same')

    # Block structure
    for n_blocks in [4, 8, 16, 32]:

    inits.append(np.maximum(0.3 + 0.7 * t_arr ** 2, 0.01))
    inits.append(np.maximum(0.3 + 0.7 * (1 - t_arr) ** 2, 0.01))

    inits.append(np.hamming(n).astype(np.float64) + 0.05)
    inits.append(np.blackman(n).astype(np.float64) + 0.05)

    # Random with structure
    inits.append(0.5 + 0.3 * np.sin(np.pi * t_arr) + 0.2 * rng.uniform(0.8, 1.0, n))

class ScoreTracker:
    """Tracks the best sequence found so far."""

    def __init__(self):
        self.best = None
        self.best_score = float('inf')
        self.t_start = time.monotonic()

    def update(self, a):
        a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
        if a.size == 0 or a.sum() < 0.01:
            return
        score = float(evaluate_sequence(a.tolist()))
        if np.isfinite(score) and score < self.best_score:
            self.best = a.copy()
            self.best_score = score
            elapsed = time.monotonic() - self.t_start
            print(f"  ** J={score:.8f} n={a.size} t={elapsed:.0f}s", flush=True)

    def report(self, label=""):
        if self.best is not None:
            print(f"  {label} best: J={self.best_score:.8f} n={self.best.size}", flush=True)

    print(f"{'='*60}", flush=True)

    lengths = [128, 192, 256, 384, 512, 640, 768, 1024, 1280, 1536, 2048]
    lengths = [n for n in lengths if 64 <= n <= 2048]

    candidates = []  # list of (J, n, a)

    for n in lengths:
        if time.monotonic() > deadline - 400:
            break

        time_for_n = min(60.0, max(15.0, total_time_p1 * 256.0 / n))
        phase_deadline = min(time.monotonic() + time_for_n, deadline - 350)

        if time.monotonic() >= phase_deadline:
            continue

        t0 = time.monotonic()
        inits = generate_initializations(n, rng, parent)

        # Shuffle and take a subset for speed
        rng_init = np.random.default_rng(seed + n)
        shuffle_idx = rng_init.permutation(len(inits))
        max_inits = max(4, min(len(inits), int(30 * 256 / n)))
        selected_inits = [inits[i] for i in shuffle_idx[:max_inits]]

        best_for_n = None
        best_M_for_n = float('inf')

        for a0 in selected_inits:
            if time.monotonic() > phase_deadline:
                break
            a0 = np.maximum(a0.astype(np.float64), 1e-6)
            a0 = a0 / a0.sum()

            # Quick FFT smooth gradient
            iters = max(100, min(2000, int(150000 / n)))
            a = run_fft_smooth_gradient(a0, iters, phase_deadline,
                                        tau_init=3.0, tau_final=80.0,
                                        lr_init=0.3, lr_min=0.03,
                                        momentum_beta=0.85)

            M_val = fast_conv(a).max()
            if M_val < best_M_for_n:
                best_M_for_n = M_val
                best_for_n = a.copy()

        if best_for_n is not None:
            J_val = compute_J(best_for_n)
            elapsed = time.monotonic() - t0
            print(f"  n={n:4d}: M={best_M_for_n:.8f} J={J_val:.6f} "
                  f"({elapsed:.1f}s)", flush=True)
            candidates.append((J_val, n, best_for_n.copy()))
            tracker.update(best_for_n)

    # Sort candidates by J
    candidates.sort(key=lambda x: x[0])
    tracker.report("Phase 1 complete")
    print(f"Phase 1 total time: {time.monotonic()-p1_start:.1f}s", flush=True)

    # === PHASE 2: Refined smooth gradient on top candidates ===
    p2_start = time.monotonic()
    total_time_p2 = 350.0
    print(f"\n{'='*60}", flush=True)
    print(f"PHASE 2: Refined FFT Smooth Gradient (budget={total_time_p2:.0f}s)", flush=True)
    print(f"{'='*60}", flush=True)

    top_k = min(5, len(candidates))
    for rank in range(top_k):
        if time.monotonic() > deadline - 350:
            break
        J_val, n, a = candidates[rank]
        time_remaining = min(total_time_p2 / (top_k - rank), deadline - 300 - time.monotonic())
        if time_remaining < 10:
            break

        phase_deadline = time.monotonic() + time_remaining
        print(f"  Refining candidate {rank}: n={n}, J={J_val:.6f}", flush=True)

        # Longer, more careful smooth gradient
        a_refined = run_fft_smooth_gradient(a, 5000, phase_deadline,
                                            tau_init=5.0, tau_final=300.0,
                                            lr_init=0.2, lr_min=0.01,
                                            momentum_beta=0.9)
        J_ref = compute_J(a_refined)
        print(f"    -> J={J_ref:.6f}", flush=True)
        tracker.update(a_refined)
        candidates[rank] = (J_ref, n, a_refined)

    # Re-sort
    candidates.sort(key=lambda x: x[0])
    tracker.report("Phase 2 complete")
    print(f"Phase 2 total time: {time.monotonic()-p2_start:.1f}s", flush=True)

    # === PHASE 3: Minimax primal-dual ===
    p3_start = time.monotonic()
    total_time_p3 = 300.0
    print(f"\n{'='*60}", flush=True)
    print(f"PHASE 3: Minimax Primal-Dual (budget={total_time_p3:.0f}s)", flush=True)
    print(f"{'='*60}", flush=True)

    if tracker.best is not None and time.monotonic() < deadline - 250:
        a_pd = tracker.best.copy()
        n_pd = a_pd.size
        print(f"  Running primal-dual on n={n_pd}, J={tracker.best_score:.6f}", flush=True)

        pd_deadline = min(time.monotonic() + total_time_p3, deadline - 200)
        # Estimate QP time
        t_qp_start = time.monotonic()
        M_test = build_hankel_matrix(np.ones(2 * n_pd - 1) / (2 * n_pd - 1))
        a_var = cp.Variable(n_pd)
        objective = cp.Minimize(cp.quad_form(a_var, cp.psd_wrap(M_test)))
        constraints = [a_var >= 0, cp.sum(a_var) == 1]
        prob = cp.Problem(objective, constraints)
        try:
            prob.solve(solver=cp.CLARARABEL, verbose=False)
        except Exception:
            try:
                prob.solve(solver=cp.SCS, verbose=False)
            except Exception:
                pass
        qp_time = time.monotonic() - t_qp_start
        print(f"  QP solve time: {qp_time*1000:.1f}ms", flush=True)

        max_iters_pd = max(5, min(500, int((pd_deadline - time.monotonic()) / max(0.01, qp_time * 2))))
        print(f"  Max primal-dual iterations: {max_iters_pd}", flush=True)

        a_pd = minimax_primal_dual(a_pd, max_iters_pd, pd_deadline,
                                   gamma_init=5.0, gamma_decay=0.99)
        tracker.update(a_pd)

    tracker.report("Phase 3 complete")
    print(f"Phase 3 total time: {time.monotonic()-p3_start:.1f}s", flush=True)

    # === PHASE 4: LP Mirror Descent ===
    p4_start = time.monotonic()
    total_time_p4 = 250.0
    print(f"\n{'='*60}", flush=True)
    print(f"PHASE 4: LP Mirror Descent (budget={total_time_p4:.0f}s)", flush=True)
    print(f"{'='*60}", flush=True)

    if tracker.best is not None and time.monotonic() < deadline - 100:
        n_best = tracker.best.size
        lp_deadline = min(time.monotonic() + total_time_p4, deadline - 80)
        est_lp_time = 0.01 * (n_best / 128) ** 2
        max_iters_lp = max(20, min(300, int(200 / max(0.005, est_lp_time))))
        print(f"  n={n_best}, est LP time={est_lp_time*1000:.0f}ms, max_iters={max_iters_lp}", flush=True)

        a_lp = run_lp_mirror_descent(tracker.best, max_iters_lp, lp_deadline)
        tracker.update(a_lp)

        # Second LP pass
        if time.monotonic() < deadline - 60:
            a_lp2 = run_lp_mirror_descent(a_lp, 50, deadline - 60)
            tracker.update(a_lp2)

    tracker.report("Phase 4 complete")
    print(f"Phase 4 total time: {time.monotonic()-p4_start:.1f}s", flush=True)

    # === PHASE 5: Final perturbations and extended smooth gradient ===
    p5_start = time.monotonic()
    print(f"\n{'='*60}", flush=True)
    print(f"PHASE 5: Final Perturbations (remaining time)", flush=True)
    print(f"{'='*60}", flush=True)

    if tracker.best is not None and time.monotonic() < deadline - 30:
        n_best = tracker.best.size
        temperature = 0.03
        max_restarts = 15

        for r in range(max_restarts):
            if time.monotonic() > deadline - 20:
                break

            a = tracker.best.copy().astype(np.float64)
            noise = rng.normal(0, temperature * 0.05, n_best)
            a = a + noise
            a = np.maximum(a, 0.001)
            a = a / a.sum()

            # Quick smooth gradient
            a = run_fft_smooth_gradient(a, 200, deadline - 20,
                                        tau_init=10, tau_final=100,
                                        lr_init=0.15, lr_min=0.03,
                                        momentum_beta=0.9)
            tracker.update(a)

            temperature *= 0.85

    # Try length extension
    if tracker.best is not None and time.monotonic() < deadline - 15:
        base_n = tracker.best.size
        for mult in [2, 3]:
            if time.monotonic() > deadline - 10:
                break
            new_n = base_n * mult
            if new_n > 4096:
                continue
            a_ext = resample_seq(tracker.best, new_n)
            a_ext = np.maximum(a_ext, 1e-6)
            a_ext = a_ext / a_ext.sum()

            a_ext = run_fft_smooth_gradient(a_ext, 300, deadline - 10,
                                            tau_init=5, tau_final=100,
                                            lr_init=0.2, lr_min=0.03,
                                            momentum_beta=0.9)
            tracker.update(a_ext)

    # === Final: one more LP pass if time remains ===
    if tracker.best is not None and time.monotonic() < deadline - 5:
        a = run_lp_mirror_descent(tracker.best, 30, deadline)
        tracker.update(a)

    # Fallback
    if tracker.best is None:
        tracker.best = np.ones(128)
        tracker.best_score = compute_J(tracker.best)

    print(f"\n{'='*60}", flush=True)
    print(f"FINAL: J={tracker.best_score:.8f} n={tracker.best.size} "
          f"time={time.monotonic()-tracker.t_start:.0f}s", flush=True)
    print(f"{'='*60}", flush=True)

    return [float(x) for x in tracker.best]
