def resample_seq(src, dest_n):
    """Resample 1D sequence to new length via linear interpolation."""
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)

            dist = np.abs(np.arange(n) - i)
            candidate_js = np.concatenate([candidate_js, np.argsort(dist)[-n_trials_per_coord:]])
            candidate_js = np.unique(candidate_js[:2 * n_trials_per_coord])

            for j in candidate_js:
                if j == i or a[j] >= a[i]:
                    continue
                # Try transferring a fraction of a[i] to j
                for frac in [0.1, 0.2, 0.3, 0.5]:
                    delta = frac * a[i]
                    a_trial = a.copy()
                    a_trial[i] -= delta
                    a_trial[j] += delta
                    if a_trial[i] < 0:
                        continue
                    M_trial = fast_conv(a_trial).max()
                    if M_trial < best_M - 1e-14:
                        a = a_trial
                        best_M = M_trial
                        best_a = a.copy()
                        any_improve = True
                        break

        if not any_improve:
            # Try the reverse: add mass to low-mass positions
            low_order = np.argsort(a)
            for j in low_order[:n // 4]:
                if time.monotonic() > deadline:
                    break
                for i in order[:n // 4]:
                    if i == j:
                        continue
                    for frac in [0.05, 0.1]:
                        delta = frac * a[i]
                        a_trial = a.copy()
                        a_trial[i] -= delta
                        a_trial[j] += delta
                        if a_trial[i] < 0:
                            continue
                        M_trial = fast_conv(a_trial).max()
                        if M_trial < best_M - 1e-14:
                            a = a_trial
                            best_M = M_trial
                            best_a = a.copy()
                            any_improve = True
                            break
            if not any_improve:
                break

    return best_a


def subgradient_polish(a, max_iters, deadline):
    """Subgradient descent with Polyak step."""
    a = np.maximum(a.astype(np.float64), 1e-10)
    a = a / a.sum()
    n = len(a)
    best_a = a.copy()
    best_M = fast_conv(a).max()
    running_sum = np.zeros(n)
    count = 0

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        b = fast_conv(a)
        M = b.max()
        tol = 1e-7 * M
        active = np.where(b >= M - tol)[0]

        grad = np.zeros(n)
        for k in active:
            lo = max(0, k - n + 1)
            hi = min(n - 1, k)
            seg = a[k - hi:k - lo + 1][::-1].copy()
            # Handle k=2i case: the middle element should be counted once
            # Actually for subgradient of max, we sum over active k
            # d(conv_k)/d(a_i) = 2*a[k-i] for k!=2i, a[i] for k=2i
            # But for simplicity, use 2*a[k-i] (slightly overcounts the diagonal)
            grad[lo:hi + 1] += 2.0 * seg

        gnorm = np.linalg.norm(grad)
        if gnorm > 1e-15:
            grad /= gnorm
        else:
            break

        # Polyak step: lr = (M - M_lower) / ||grad||^2
        # M_lower = average of conv = 1/(2n-1) (when sum(a)=1)
        M_lower = 1.0 / (2 * n - 1)
        lr = max(0.01, (M - M_lower) / max(gnorm * gnorm, 1e-10))
        lr = min(lr, 0.5)

        a_new = a - lr * grad
        a_new = np.maximum(a_new, 0)
        s = a_new.sum()
        if s < 1e-10:
            a_new = a.copy()
        else:
            a_new /= s
        a = a_new
        running_sum += a
        count += 1

        M_now = fast_conv(a).max()
        if M_now < best_M - 1e-12:
            best_M = M_now
            best_a = a.copy()

    # Check averaged solution
    if count > 10:
        a_avg = running_sum / count
        a_avg = np.maximum(a_avg, 0)
        s = a_avg.sum()
        if s > 1e-10:
            a_avg /= s
            M_avg = fast_conv(a_avg).max()
            if M_avg < best_M:
                best_a = a_avg

    return best_a


def generate_diverse_inits(n, rng, parent=None):
    """Generate diverse initial sequences."""
    inits = []
    t = np.linspace(0, 1, n, endpoint=False)

    if parent is not None:
        p = resample_seq(parent, n)
        inits.append(p)
        inits.append(np.maximum(p + rng.normal(0, 0.05, n), 0.01))
        inits.append(np.maximum(p + rng.normal(0, 0.15, n), 0.01))
        # Symmetrize parent
        sym = (p + p[::-1]) / 2
        inits.append(np.maximum(sym, 0.01))

    inits.append(np.ones(n))
    inits.append(rng.uniform(0.3, 1.0, n))

    # Bimodal at ends
    inits.append(np.maximum(1.0 - 2.0 * t + 0.5 * np.exp(-20 * t**2) + 0.5 * np.exp(-20 * (1-t)**2), 0.01))
    # Cosine variants
    for freq in [0.5, 1, 1.5, 2, 3]:
        inits.append(np.maximum(1.0 + 0.4 * np.cos(2 * np.pi * freq * t), 0.01))
    # Linear ramp
    inits.append(0.2 + 0.8 * t)
    inits.append(0.2 + 0.8 * (1 - t))
    # V-shape
    inits.append(0.3 + 0.7 * np.abs(2 * t - 1))
    # Inverted V
    inits.append(0.3 + 0.7 * (1 - np.abs(2 * t - 1)))
    # Smooth random
    ksize = max(3, n // 20)
    kernel = np.ones(ksize) / ksize
    smooth = np.convolve(rng.uniform(0, 1, n), kernel, mode='same')
    inits.append(np.maximum(smooth + 0.1, 0.01))
    # Block structure
    for nb in [4, 8, 16]:
        bv = rng.uniform(0.2, 1.0, nb)
        bs = np.repeat(bv, n // nb + 1)[:n]
        inits.append(bs)

    return [np.maximum(x, 1e-6) / np.maximum(x, 1e-6).sum() for x in inits]


def propose_candidate(seed=42, budget_s=1500, **kwargs):
    """
    Multi-phase search for optimal sequence:
    Phase 1: Smooth gradient across lengths (fast exploration)
    Phase 2: LP mirror descent with kicks (main optimization)
    Phase 3: Subgradient polish
    Phase 4: Pair transfer refinement
    Phase 5: Final LP polish
    """
    t0 = time.monotonic()
    deadline = t0 + float(budget_s) - 25.0
    rng = np.random.default_rng(seed)

    best_score = float('inf')
    best_seq = None

    def track(a, label=""):
        nonlocal best_score, best_seq
        a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
        if a.size == 0 or a.sum() < 0.01:
            return
        score = float(evaluate_sequence(a.tolist()))
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_seq = a.copy()
            elapsed = time.monotonic() - t0
            print(f"  ** {label} J={score:.8f} n={a.size} t={elapsed:.0f}s", flush=True)
        return score

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        if p.ndim == 1 and p.size > 0 and np.all(np.isfinite(p)):
            parent = np.maximum(p, 0.0)
            s_parent = parent.sum()
            if s_parent > 0.01:
                track(parent, "parent")
                print(f"Parent: n={parent.size}, J={best_score:.6f}", flush=True)
    except (NameError, AttributeError):
        pass

    if parent is None:
        parent = np.ones(256)
        track(parent, "default")

    # === PHASE 1: Smooth gradient exploration across lengths ===
    print(f"\n{'='*60}", flush=True)
    print(f"PHASE 1: Smooth Gradient Exploration", flush=True)
    print(f"{'='*60}", flush=True)

    p1_deadline = min(t0 + 200, deadline - 600)
    lengths = [parent.size]  # Start with parent length
    # Add some other lengths
    for mult in [0.5, 0.75, 1.25, 1.5, 2.0]:
        ln = int(parent.size * mult)
        ln = max(128, min(1024, ln))
        if ln not in lengths:
            lengths.append(ln)
    lengths = sorted(set(lengths))

    for n in lengths:
        if time.monotonic() > p1_deadline:
            break
        time_per_n = max(15, min(50, (p1_deadline - time.monotonic()) / max(1, len(lengths))))
        phase_dl = min(time.monotonic() + time_per_n, p1_deadline)

        inits = generate_diverse_inits(n, rng, parent)
        best_n = None
        best_M_n = float('inf')

        for a0 in inits:
            if time.monotonic() > phase_dl:
                break
            a = run_smooth_grad_descent(a0, 200, phase_dl,
                                        tau_start=2.0, tau_end=40.0,
                                        lr_start=0.2, lr_end=0.03)
            M_val = fast_conv(a).max()
            if M_val < best_M_n:
                best_M_n = M_val
                best_n = a.copy()

        if best_n is not None:
            J = compute_J(best_n)
            print(f"  n={n:4d}: J={J:.6f} ({time.monotonic()-t0:.0f}s)", flush=True)
            track(best_n, f"P1 n={n}")

    tracker_best = best_seq.copy() if best_seq is not None else parent.copy()
    print(f"Phase 1 done. Best J={best_score:.6f}", flush=True)

    # === PHASE 2: LP Mirror Descent (main optimization) ===
    print(f"\n{'='*60}", flush=True)
    print(f"PHASE 2: LP Mirror Descent", flush=True)
    print(f"{'='*60}", flush=True)

    p2_deadline = min(t0 + 800, deadline - 200)
    n_best = tracker_best.size
    est_lp_time = 0.005 * (n_best / 128) ** 2
    max_lp_iters = max(50, int((p2_deadline - time.monotonic()) / max(0.05, est_lp_time)))
    print(f"  n={n_best}, est LP iter={est_lp_time*1000:.0f}ms, max_iters={max_lp_iters}", flush=True)

    a_opt = run_lp_mirror(tracker_best, max_lp_iters, p2_deadline,
                          kick_interval=20, kick_noise=0.02)
    track(a_opt, "P2 LP")

    # === PHASE 3: Subgradient polish ===
    print(f"\n{'='*60}", flush=True)
    print(f"PHASE 3: Subgradient Polish", flush=True)
    print(f"{'='*60}", flush=True)

    if time.monotonic() < deadline - 100:
        p3_dl = min(time.monotonic() + 80, deadline - 150)
        a_sg = subgradient_polish(tracker_best, 3000, p3_dl)
        track(a_sg, "P3 SG")

    # === PHASE 4: Pair transfer refinement ===
    print(f"\n{'='*60}", flush=True)
    print(f"PHASE 4: Pair Transfer Refinement", flush=True)
    print(f"{'='*60}", flush=True)

    if time.monotonic() < deadline - 80:
        p4_dl = min(time.monotonic() + 120, deadline - 80)
        a_pt = pair_transfer_optimize(tracker_best, 50, p4_dl, n_trials_per_coord=6)
        track(a_pt, "P4 PT")

    # === PHASE 5: LP on best found + different starting points ===
    print(f"\n{'='*60}", flush=True)
    print(f"PHASE 5: Final LP Polish + Restarts", flush=True)
    print(f"{'='*60}", flush=True)

    if time.monotonic() < deadline - 30:
        # LP on current best
        remaining_iters = max(10, int((deadline - 30 - time.monotonic()) / max(0.05, est_lp_time)))
        a_final = run_lp_mirror(tracker_best, remaining_iters, deadline - 30,
                                kick_interval=15, kick_noise=0.01)
        track(a_final, "P5 LP best")

        # Try LP from a few perturbed starts
        if time.monotonic() < deadline - 60:
            for trial in range(3):
                if time.monotonic() > deadline - 50:
                    break
                a_pert = tracker_best + rng.normal(0, 0.03, tracker_best.size)
                a_pert = np.maximum(a_pert, 0)
                a_pert /= a_pert.sum()
                # Quick smooth gradient
                a_pert = run_smooth_grad_descent(a_pert, 50, deadline - 50,
                                                 tau_start=10, tau_end=40,
                                                 lr_start=0.1, lr_end=0.02)
                # Short LP
                a_pert = run_lp_mirror(a_pert, 15, deadline - 50,
                                       kick_interval=10, kick_noise=0.01)
                track(a_pert, f"P5 restart {trial}")

    # === Phase 6: Try different lengths for final attempt ===
    if time.monotonic() < deadline - 15 and best_seq is not None:
        print(f"\n{'='*60}", flush=True)
        print(f"PHASE 6: Length Exploration", flush=True)
        print(f"{'='*60}", flush=True)

        base_n = best_seq.size
        for mult in [1.5, 2]:
            if time.monotonic() > deadline - 10:
                break
            new_n = int(base_n * mult)
            if new_n > 1536 or new_n < 128:
                continue
            a_ext = resample_seq(best_seq, new_n)
            a_ext = np.maximum(a_ext, 1e-8)
            a_ext /= a_ext.sum()
            a_ext = run_smooth_grad_descent(a_ext, 80, deadline - 10,
                                            tau_start=5, tau_end=40,
                                            lr_start=0.15, lr_end=0.03)
            track(a_ext, f"P6 n={new_n}")
            if time.monotonic() < deadline - 15:
                a_ext = run_lp_mirror(a_ext, 10, deadline - 10)
                track(a_ext, f"P6 LP n={new_n}")

    # Final report
    if best_seq is None:
        best_seq = parent
        best_score = compute_J(best_seq)

    elapsed = time.monotonic() - t0
    print(f"\n{'='*60}", flush=True)
    print(f"FINAL: J={best_score:.8f} n={best_seq.size} time={elapsed:.0f}s", flush=True)
    print(f"{'='*60}", flush=True)

    return [float(x) for x in best_seq]
