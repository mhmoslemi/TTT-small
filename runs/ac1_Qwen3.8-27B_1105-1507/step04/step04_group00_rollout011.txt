By the uncertainty principle, a sequence can't be both concentrated in time and concentrated in frequency. But here we're looking at a different trade-off.

Actually, (a*a)[n-1] = sum a[i]*a[n-1-i] is the inner product of a and its reversal. If a is very different from its reversal, this inner product can be small.

For example, if a[i] = 1 for i < n/2 and 0 for i >= n/2:
(a*a)[n-1] = sum_{i=0}^{n/2-1} a[i]*a[n-1-i] = sum_{i=0}^{n/2-1} 1 * a[n-1-i]
For i < n/2, n-1-i > n/2-1 >= n/2 (for n even), so a[n-1-i] = 0.
Therefore (a*a)[n-1] = 0!

I think the key insight is: to minimize J = 2*n*max_b/S^2, we want:
1. Large S (so we want most entries to be large)
2. Small max_b (so we want the autocorrelation to be flat)
3. The ratio max_b * n / S^2 should be minimized

These are in tension: making entries large increases both S and max_b.

The optimal balance is to have entries that are "as equal as possible" (to maximize S for a given max entry) while having a "flat" autocorrelation (to minimize max_b).

For the all-ones sequence, the autocorrelation is triangular (peaked at center), giving J=2.
To get J<2, we need to reduce the center of the autocorrelation below the "average" level.

Since (a*a)[n-1] = <a, a_reversed>, and for symmetric a this equals ||a||^2 >= S^2/n, the sequence must be asymmetric.

The question is: what's the most effective asymmetry?

= n + eps*S_f + eps^2*<f, f_rev>

where S_f = sum(f). If we choose f with sum(f)=0 (zero mean), then:
(a*a)[n-1] = n + eps^2*<f, f_rev>

S = n + eps*S_f = n (if sum(f)=0)

So max_b ≈ n + eps^2*<f,f_rev> (assuming n-1 is the peak lag)

J ≈ 2*n*(n + eps^2*<f,f_rev>)/n^2 = 2 + 2*eps^2*<f,f_rev>/n

To minimize J, we want <f, f_rev> < 0, i.e., f should be "anti-correlated" with its reversal.

For example, f[i] = i - (n-1)/2 (linear):

"""
Minimizes J(a) = 2*n*max(conv(a,a))/(sum(a))^2 using an improved hybrid approach:
1. Asymmetric initializations exploiting the insight that J<2 requires asymmetry
2. Nesterov-accelerated smooth gradient descent with adaptive tau
3. Variance-flattening pre-optimization
4. Active-set LP refinement for final polishing
5. Multi-length exploration

import numpy as np
from scipy.optimize import linprog
from scipy.sparse import csr_matrix
import time


def fast_conv(a):
    """Full self-convolution via FFT."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]


def compute_J(a):
    """Compute J(a) = 2*n*max(conv(a,a))/(sum(a))^2."""
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))


def smooth_max_and_grad(a, tau):
    """Compute smooth max (log-sum-exp) and its gradient."""
    n = len(a)
    b = fast_conv(a)
    b_max = b.max()
    exp_b = np.exp(tau * (b - b_max))
    p = exp_b / exp_b.sum()
    
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        pk = p[k]
        if pk < 1e-14:
            continue
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        grad[lo:hi + 1] += 2.0 * pk * a[k - hi:k - lo + 1][::-1]
    
    smooth_val = b_max + np.log(exp_b.sum()) / tau
    return smooth_val, grad, b, p


def run_nesterov_smooth(a0, max_iters, deadline, tau_start=5.0, tau_end=100.0,
                         lr_init=0.5, momentum_init=0.9):
    """Nesterov accelerated smooth gradient descent with adaptive tau."""
    a = np.maximum(a0.astype(np.float64), 1e-10)
    a = a / a.sum()
    n = len(a)
    
    best_a = a.copy()
    best_M = fast_conv(a).max()
    
    y = a.copy()  # Nesterov lookahead
    v = np.zeros(n)
    beta = momentum_init
    
    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        
        frac = it / max(1, max_iters)
        # Adaptive tau: start smooth, end sharp
        tau = tau_start * (tau_end / tau_start) ** (frac ** 0.7)
        # Learning rate: 1/sqrt(t) schedule
        lr = lr_init / (it + 1) ** 0.4
        # Momentum: slowly increase
        beta = momentum_init + 0.05 * frac
        
        # Nesterov step
        y = a + beta * v
        y = np.maximum(y, 1e-10)
        y = y / y.sum()
        
        val, grad, b, p = smooth_max_and_grad(y, tau)
        M = b.max()
        
        if M < best_M * (1 - 1e-10):
            best_M = M
            best_a = y.copy()
        
        # Update
        v = beta * v - lr * grad
        a_new = a + v
        a_new = np.maximum(a_new, 1e-10)
        a_new = a_new / a_new.sum()
        a = a_new
    
    # Final evaluation
    M_final = fast_conv(a).max()
    if M_final < best_M:
        best_a = a.copy()
        best_M = M_final
    
    return best_a


def run_variance_flatten(a0, max_iters, deadline):
    """Minimize variance of autocorrelation (encourages flatness)."""
    a = np.maximum(a0.astype(np.float64), 1e-10)
    a = a / a.sum()
    n = len(a)
    
    best_a = a.copy()
    best_var = np.var(fast_conv(a))
    
    lr = 0.1
    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        
        b = fast_conv(a)
        b_mean = b.mean()
        # Gradient of Var(b) w.r.t. a
        # Var(b) = mean(b^2) - mean(b)^2
        # d(Var)/da = 2*mean(db/da * (b - mean(b)))
        # db[k]/da[i] = a[k-i] + a[i] (if valid)... actually db[k]/da[i] = a[k-i] (for the k-i index)
        # More precisely: b[k] = sum_j a[j]*a[k-j], so db[k]/da[i] = a[k-i] + a[i-k] (if both valid)
        # But since b[k] = sum_j a[j]*a[k-j], db[k]/da[i] = a[k-i] (when 0<=k-i<n) + a[i] (when 0<=i<n, which is always)

        # = a[k-i] (if 0<=k-i<n) + a[k-i] (if 0<=k-i<n) = 2*a[k-i] (if 0<=k-i<n)
        # Wait, that's not right either. Let me be more careful.
        # b[k] = sum_{j: 0<=j<n, 0<=k-j<n} a[j] * a[k-j]
        # d b[k] / d a[i] = sum_j [delta_{j,i} * a[k-j] + a[j] * delta_{k-j,i}]
        # = a[k-i] (if 0<=k-i<n) + a[k-i] (if 0<=k-i<n)
        # = 2*a[k-i] (if 0<=k-i<n)
        # Otherwise 0.
        
        # So d(Var(b))/da[i] = sum_k 2*(b[k]-b_mean)/(2n-1) * 2*a[k-i] * (1/(2n-1) ... )
        # Actually, let's just use the chain rule numerically or with the formula above.
        
        # For efficiency, let's compute the gradient:
        # g[i] = sum_k c[k] * 2*a[k-i] where c[k] = 2*(b[k]-b_mean)/(2*(2n-1)) ... 
        # Let me just do it the straightforward way:
        # g[i] = sum_{k: |k-i| < n} 2*(b[k]-b_mean)/(2*(2n-1)) * 2*a[k-i]
        # Hmm, this is getting complicated. Let me use a simpler approach:
        # Just minimize sum((b[k]-b_mean)^2) which has a clean gradient.
        
        # Actually, let's minimize the sum of squares of b (which is related to variance since mean(b) is fixed for fixed sum(a))
        # sum(b[k]^2) = sum_k (sum_j a[j]*a[k-j])^2
        # By Parseval: = (1/N) * sum_k |A(k)|^4 where N is FFT size
        # Gradient: d/da[i] sum_k b[k]^2 = 2 * sum_k b[k] * db[k]/da[i] = 4 * sum_k b[k]*a[k-i] (valid k)
        # = 4 * (b * a_reversed)[i] ... this is a convolution!
        
        a_rev = a[::-1]
        grad_bsq = 4.0 * np.convolve(b, a_rev, mode='same')[:n]
        
        # Actually let me just use the direct formula:
        # g[i] = 4 * sum_{k: 0<=k<2n-1, 0<=k-i<n} b[k] * a[k-i]
        # This is 4 * (b * a)[i + n - 1] ... no
        # sum_k b[k]*a[k-i] is the cross-correlation of b and a at lag i
        # = (b * a_rev)[i + len(b) - len(a)] ... this is getting complicated
        
        # Let me just do it directly:
        g = np.zeros(n)
        for i in range(n):
            # g[i] = 4 * sum_{k=i}^{i+n-1} b[k] * a[k-i] (where a[k-i] is valid)
            # k-i ranges from 0 to n-1, so k ranges from i to i+n-1
            # But b has length 2n-1, so k goes from i to min(i+n-1, 2n-2)
            ks = np.arange(i, min(i + n, 2 * n - 1))
            g[i] = 4.0 * np.dot(b[ks], a[ks - i])

        # g[i] = 4 * sum_{j=0}^{n-1} b[i+j] * a[j] (for valid i+j < 2n-1)
        # This is the cross-correlation of b and a, which can be computed via FFT.
        
        # Actually, let me just use the convolution approach:
        # b_full has length 2n-1, a has length n
        # cross_corr[i] = sum_j b[i+j] * a[j] for j where i+j < 2n-1
        # This is a "sliding dot product" which is a form of cross-correlation
        # cross_corr = np.correlate(b, a, mode='valid') ... no, that's the other direction
        # Let me think: cross_corr[i] = sum_j b[i+j]*a[j] = (b * a_flipped)[i + n - 1]
        # where a_flipped = a[::-1] and * denotes full convolution
        
        a_flip = a[::-1]
        bc = np.convolve(b, a_flip, mode='full')
        # cross_corr[i] = bc[i + n - 1] for i = 0..n-1
        g = 4.0 * bc[n-1:n-1+n]
        
        # Gradient step
        a_new = a - lr * g / (n * np.linalg.norm(g))  # normalize
        a_new = np.maximum(a_new, 1e-10)
        a_new = a_new / a_new.sum()
        
        var_new = np.var(fast_conv(a_new))
        if var_new < best_var * (1 - 1e-8):
            best_var = var_new
            best_a = a_new.copy()
        
        a = a_new
        lr *= 0.999
    
    return best_a


def active_set_refine(a0, max_iters, deadline, tol=1e-6):
    """
    Active set method: identify peak lags, solve LP for best response,
    update a via mirror descent.
    """
    a = np.maximum(a0.astype(np.float64), 1e-10)
    a = a / a.sum()
    n = len(a)
    
    best_a = a.copy()
    best_M = fast_conv(a).max()
    
    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        
        b = fast_conv(a)
        M = b.max()
        
        if M < best_M - 1e-12:
            best_M = M
            best_a = a.copy()
        
        # Identify active set (lags within tol of max)
        threshold = M * (1 - tol)
        active = np.where(b >= threshold)[0]
        
        # Solve LP: max sum(g) s.t. sum_j a[k-j]*g[j] <= M for k in active, g >= 0
        # Build constraint matrix (only for active rows)
        rows = []
        cols = []
        vals = []
        for k in active:
            j_min = max(0, k - n + 1)
            j_max = min(n - 1, k)
            for j in range(j_min, j_max + 1):
                rows.append(k - active[0] if False else len(rows) // max(1, (j_max - j_min + 1)))  # this is wrong
                # Let me just build it properly
                pass
        
        # Proper sparse matrix construction
        row_idx = []
        col_idx = []
        val_idx = []
        row_counter = 0
        for k in active:
            j_min = max(0, k - n + 1)
            j_max = min(n - 1, k)
            for j in range(j_min, j_max + 1):
                row_idx.append(row_counter)
                col_idx.append(j)
                val_idx.append(a[k - j])
            row_counter += 1
        
        if len(val_idx) == 0:
            break
            
        A = csr_matrix((val_idx, (row_idx, col_idx)), shape=(len(active), n))
        c_vec = -np.ones(n)
        b_ub = np.full(len(active), M)
        bounds = [(0, None)] * n
        
        try:
            r = linprog(c_vec, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                        options={'time_limit': 10, 'presolve': True})
            if r.success and r.x is not None:
                g = np.maximum(r.x, 0)
                g_sum = g.sum()
                if g_sum < 1e-10:
                    break
                g = g / g_sum
            else:
                break
        except Exception:
            break
        
        # Mirror descent update with line search
        improved = False
        for t in [0.3, 0.2, 0.1, 0.05, 0.02]:
            a_trial = (1 - t) * a + t * g
            a_trial = np.maximum(a_trial, 0)
            s = a_trial.sum()
            if s < 1e-10:
                continue
            a_trial = a_trial / s
            M_trial = fast_conv(a_trial).max()
            if M_trial < best_M - 1e-12:
                a = a_trial
                best_M = M_trial
                best_a = a.copy()
                improved = True
                break
        
        if not improved:
            # Try smaller step
            a_trial = (1 - 0.01) * a + 0.01 * g
            a_trial = np.maximum(a_trial, 0)
            s = a_trial.sum()
            if s > 1e-10:
                a_trial /= s
                M_trial = fast_conv(a_trial).max()
                if M_trial < best_M:
                    a = a_trial
                    best_M = M_trial
                    best_a = a.copy()
                else:
                    break
    
    return best_a


def generate_asymmetric_inits(n, rng, parent=None):
    """Generate initializations that exploit the asymmetry requirement."""
    inits = []
    t = np.linspace(0, 1, n, endpoint=False)
    
    if parent is not None:
        from scipy.interpolate import interp1d
        p = np.asarray(parent, dtype=np.float64)
        if p.size != n:
            t_old = np.linspace(0, 1, p.size, endpoint=False)
            f = interp1d(t_old, p, kind='linear', fill_value='extrapolate')
            p_resampled = np.maximum(f(t), 0.01)
        else:
            p_resampled = np.maximum(p, 0.01)
        p_resampled = p_resampled / p_resampled.sum()
        inits.append(p_resampled.copy())
        inits.append(np.maximum(p_resampled + rng.normal(0, 0.02, n), 0.01))
        inits.append(np.maximum(p_resampled + rng.normal(0, 0.05, n), 0.01))
    
    # Linear ramp (asymmetric)
    for slope in [0.3, 0.5, 0.7, 1.0]:
        inits.append(np.maximum(1.0 + slope * (2*t - 1), 0.01))
        inits.append(np.maximum(1.0 - slope * (2*t - 1), 0.01))
    
    # Exponential decay
    for rate in [2, 5, 10]:
        inits.append(np.maximum(np.exp(-rate * t) + 0.1, 0.01))
        inits.append(np.maximum(np.exp(-rate * (1-t)) + 0.1, 0.01))
    
    # Sigmoid transition
    for mid in [0.3, 0.5, 0.7]:
        sigmoid = 1.0 / (1.0 + np.exp(-10*(t - mid)))
        inits.append(0.3 + 0.7 * sigmoid)
        inits.append(0.3 + 0.7 * (1 - sigmoid))
    
    # Quadratic (parabolic, asymmetric)
    inits.append(np.maximum(1.0 + 0.3*(t - 0.3)**2 - 0.1, 0.01))
    inits.append(np.maximum(1.0 + 0.3*(t - 0.7)**2 - 0.1, 0.01))
    
    # Random smooth (asymmetric by construction)
    ksize = max(3, n // 30)
    kernel = np.ones(ksize) / ksize
    smooth = np.convolve(rng.uniform(0, 1, n), kernel, mode='same')
    inits.append(np.maximum(smooth + 0.2, 0.01))
    
    #Freq-structured asymmetric
    for freq in [1, 2, 3]:
        inits.append(np.maximum(1.0 + 0.3*np.cos(2*np.pi*freq*t + np.pi/4), 0.01))
        inits.append(np.maximum(1.0 + 0.3*np.sin(2*np.pi*freq*t), 0.01))
    
    # "Barker-like" patterns scaled up
    # For the merit factor problem, good sequences have specific structures
    # Try sequences based on quadratic residues
    if n <= 128:
        # Quadratic residue sequence (converted to [0,1])
        qr = np.zeros(n)
        for i in range(n):
            if i == 0:
                qr[i] = 0.5
            else:
                qr[i] = 0.5 + 0.5 * (1 if pow(i, 2, n) in set(pow(j, 2, n) for j in range(1, n)) else -1) * 0.5
        inits.append(np.maximum(qr, 0.01))
    
    # Step function (piecewise constant, asymmetric)
    for n_steps in [3, 5, 8, 13]:
        step_sizes = rng.dirichlet(np.ones(n_steps) * 2) * n
        step_vals = rng.uniform(0.3, 1.5, n_steps)
        step_seq = np.zeros(n)
        pos = 0
        for i, (ss, sv) in enumerate(zip(step_sizes, step_vals)):
            length = max(1, int(ss))
            step_seq[pos:pos+length] = sv
            pos += length
        if pos < n:
            step_seq[pos:] = step_vals[-1]
        inits.append(np.maximum(step_seq, 0.01))
    
    # Triangular (asymmetric peak)
    for peak in [0.3, 0.5, 0.7]:
        tri = 1.0 - np.abs(t - peak) / max(peak, 1-peak)
        inits.append(np.maximum(0.3 + 0.7 * tri, 0.01))
    
    return inits

    def __init__(self):
        self.best = None
        self.best_score = float('inf')
        self.t_start = time.monotonic()
    
    def update(self, a):
        a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
        if a.size == 0 or a.sum() < 0.01:
            return
        score = float(evaluate_sequence(a.tolist()))
        if np.isfinite(score) and score < self.best_score:
            self.best = a.copy()
            self.best_score = score
            elapsed = time.monotonic() - self.t_start
            print(f"  ** J={score:.8f} n={a.size} t={elapsed:.0f}s", flush=True)
    
    def report(self, label=""):
        if self.best is not None:
            print(f"  {label} best: J={self.best_score:.8f} n={self.best.size}", flush=True)
