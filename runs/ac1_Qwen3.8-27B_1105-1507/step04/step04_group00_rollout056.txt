import numpy as np
from scipy.optimize import minimize
from scipy.sparse import csr_matrix, lil_matrix
import time


def fast_conv(a):
    """Full self-convolution via FFT."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]


def compute_J(a):
    """Compute J(a) = 2*n*max(conv(a,a))/(sum(a))^2."""
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))


def smooth_max_obj_and_grad(a, tau):
    """
    Objective and gradient of (1/tau)*log(sum exp(tau*(a*a)_k)).
    Returns (value, gradient).
    """
    b = fast_conv(a)
    n = len(a)
    b_max = b.max()
    exp_b = np.exp(tau * (b - b_max))
    p = exp_b / exp_b.sum()
    val = b_max + np.log(exp_b.sum()) / tau

    grad = np.zeros(n)
    for k in range(2 * n - 1):
        pk = p[k]
        if pk < 1e-14:
            continue
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        grad[lo:hi + 1] += 2.0 * pk * a[k - hi:k - lo + 1][::-1]

    return val, grad


def smooth_max_obj(x, tau):
    """Wrapper for scipy minimize: objective only."""
    ax = np.maximum(x, 0.0)
    s = ax.sum()
    if s < 1e-15:
        return 1e10
    ax = ax / s
    val, _ = smooth_max_obj_and_grad(ax, tau)
    return val


def smooth_max_grad(x, tau):
    """Wrapper for scipy minimize: gradient only."""
    ax = np.maximum(x, 0.0)
    s = ax.sum()
    if s < 1e-15:
        return np.zeros_like(x)
    ax = ax / s
    _, grad = smooth_max_obj_and_grad(ax, tau)
    return grad

                method='L-BFGS-B', bounds=bounds,
                options={'maxiter': max_iter, 'max_fun': max_iter * 10,
                         'ftol': 1e-18, 'gtol': 1e-12, 'gtol_rel': 1e-12}
            )
            if res.fun < best_val and res.x is not None:
                ax = np.maximum(res.x, 0.0)
                s = ax.sum()
                if s > 1e-10:
                    ax /= s
                    best_val = res.fun
                    best_a = ax.copy()
            break
        except Exception:
            if deadline is not None and time.monotonic() > deadline:
                break
            a = a + 0.01 * np.random.default_rng().uniform(-1, 1, len(a))
            a = np.maximum(a, 1e-8)
            a /= a.sum()

    return best_a


def identify_active_set(a, rel_tol=1e-6):
    """Identify the active set where (a*a)_k is close to the max."""
    b = fast_conv(a)
    M = b.max()
    tol = rel_tol * M
    active = np.where(b >= M - tol)[0]
    return active, M, b


def active_set_newton(a, max_iters=50, deadline=None):
    """
    Newton method on the KKT system for a given active set.
    Variables: a (length n), dual weights mu (length |A|), multiplier lam, and M.
    Equations:
      F_k = (a*a)_k - M = 0 for k in A
      G_i = 2*sum_k mu_k * a_{k-i} - lam = 0 for all i
      H1 = sum(a) - 1 = 0
      H2 = sum(mu) - 1 = 0
    """
    a = np.maximum(a.astype(np.float64), 1e-10)
    a = a / a.sum()
    n = len(a)

    best_a = a.copy()
    best_M = fast_conv(a).max()

    for it in range(max_iters):
        if deadline is not None and time.monotonic() > deadline:
            break

        active, M, b = identify_active_set(a)
        A = len(active)
        if A == 0 or A > n:
            break

        # Build the KKT system
        # Unknowns: da (n), dmu (A), dlam (1), dM (1) -> total n+A+2
        # Equations: F (A), G (n), H1 (1), H2 (1) -> total A+n+2

        # Current mu: uniform over active set
        mu = np.ones(A) / A
        # Current lam: from G equation (should be approximately constant)
        G_vals = np.zeros(n)
        for idx, k in enumerate(active):
            lo = max(0, k - n + 1)
            hi = min(n - 1, k)
            G_vals[lo:hi + 1] += 2.0 * mu[idx] * a[k - hi:k - lo + 1][::-1]
        lam = G_vals.mean()

        # Build Jacobian J (sparse) and residual
        # Residual:
        F_res = b[active] - M  # should be ~0
        G_res = G_vals - lam   # should be ~0
        H1_res = a.sum() - 1.0
        H2_res = mu.sum() - 1.0

        res = np.concatenate([F_res, G_res, [H1_res, H2_res]])
        res_norm = np.linalg.norm(res)

        if res_norm < 1e-12:
            break

        # Build Jacobian (dense for simplicity, n+A+2 is at most ~1500)
        # J structure:
        # Row F_k: dF/d(da_i) = 2*a_{k-i}, dF/dM = -1
        # Row G_i: dG/d(da_j) = 2*sum_k mu_k * delta_{j,k-i}, dG/d(dmu_k) = 2*a_{k-i}, dG/dlam = -1
        # Row H1: d/d(da_i) = 1
        # Row H2: d/d(dmu_k) = 1

        J_size = n + A + 2
        # Use dense for small problems, structured for large
        if J_size <= 2000:
            J = np.zeros((J_size, J_size))

            # F rows (first A rows)
            row_offset = 0
            for idx, k in enumerate(active):
                lo = max(0, k - n + 1)
                hi = min(n - 1, k)
                J[row_offset, lo:hi + 1] = 2.0 * a[k - hi:k - lo + 1][::-1]
                J[row_offset, n + A] = -1.0  # dF/dM
                row_offset += 1

            # G rows (next n rows)
            row_offset = A
            for i in range(n):
                # dG_i/da_j = 2 * sum_k mu_k * delta_{j, k-i}
                for idx, k in enumerate(active):
                    j = k - i
                    if 0 <= j < n:
                        J[row_offset, j] += 2.0 * mu[idx]
                # dG_i/dmu_k = 2 * a_{k-i}
                for idx, k in enumerate(active):
                    j = k - i
                    if 0 <= j < n:
                        J[row_offset, n + idx] = 2.0 * a[j]
                # dG_i/dlam = -1
                J[row_offset, n + A] = -1.0
                if i == 0:  # H1 in same row? No, separate
                    pass
                row_offset += 1

            # H1 row
            J[n + A, :n] = 1.0
            # H2 row
            J[n + A + 1, n:n + A] = 1.0

            # Solve J * dx = -res
            try:
                dx = np.linalg.solve(J, -res)
            except np.linalg.LinAlgError:
                break

            da = dx[:n]
            dmu = dx[n:n + A]
            dlam = dx[n + A]
            dM = dx[n + A + 1]

            # Damped update
            for step_frac in [1.0, 0.5, 0.25, 0.1, 0.05]:
                a_trial = a + step_frac * da
                a_trial = np.maximum(a_trial, 1e-10)
                a_trial /= a_trial.sum()
                mu_trial = mu + step_frac * dmu
                mu_trial = np.maximum(mu_trial, 0.0)
                if mu_trial.sum() > 1e-10:
                    mu_trial /= mu_trial.sum()

                M_trial = fast_conv(a_trial).max()
                if M_trial < best_M:
                    best_M = M_trial
                    best_a = a_trial.copy()
                    a = a_trial
                    mu = mu_trial
                    break
            else:
                break
        else:
            # Too large for dense Newton, use simpler update
            break

    return best_a


def multiplicative_perturbation(a, rng, scale=0.05):
    """Apply a structured multiplicative perturbation."""
    n = len(a)
    t = np.linspace(0, 1, n, endpoint=False)
    mode = rng.integers(0, 6)

    if mode == 0:
        freq = rng.uniform(0.5, 5.0)
        pert = 1.0 + scale * np.cos(2 * np.pi * freq * t + rng.uniform(0, 2 * np.pi))
    elif mode == 1:
        freq = rng.uniform(0.5, 3.0)
        pert = 1.0 + scale * np.sin(2 * np.pi * freq * t) ** 2
    elif mode == 2:
        center = rng.uniform(0.2, 0.8)
        width = rng.uniform(0.05, 0.3)
        pert = 1.0 + scale * np.exp(-((t - center) / width) ** 2)
    elif mode == 3:
        # Edge taper adjustment
        edge = np.minimum(t, 1 - t)
        pert = 1.0 + scale * (1.0 - np.exp(-10 * edge))
    elif mode == 4:
        # Polynomial perturbation
        deg = rng.integers(2, 6)
        coeffs = rng.uniform(-1, 1, deg + 1) * scale
        pert = 1.0 + np.polyval(coeffs, t)
    else:
        # Random smooth perturbation
        ksize = max(3, n // 20)
        kernel = np.ones(ksize) / ksize
        noise = np.convolve(rng.normal(0, 1, n), kernel, mode='same')
        noise = noise - noise.mean()
        pert = 1.0 + scale * noise / (np.abs(noise).max() + 1e-10)

    return np.maximum(a * pert, 1e-10)


def generate_analytic_inits(n, rng):
    """Generate diverse analytic initializations."""
    inits = []
    t = np.linspace(0, 1, n, endpoint=False)

    # Raised cosine (Hann window)
    inits.append(0.5 * (1 + np.cos(2 * np.pi * t)))

    # Sin^2
    inits.append(np.sin(np.pi * (t + 1.0 / (2 * n))) ** 2)

    # Flattop approximation
    inits.append(np.maximum(np.cos(np.pi * (t - 0.5) * 1.2), 0.01))

    # Power tapers
    for p in [0.5, 1.0, 2.0, 4.0]:
        x = np.minimum(t, 1 - t)
        inits.append(x ** p + 0.05)

    # Gaussian centers
    for center in [0.3, 0.5, 0.7]:
        for width in [0.1, 0.2, 0.3]:
            inits.append(np.exp(-((t - center) / width) ** 2) + 0.02)

    # Double bump
    inits.append(np.exp(-8 * (t - 0.3) ** 2) + np.exp(-8 * (t - 0.7) ** 2) + 0.02)
    inits.append(np.exp(-5 * (t - 0.25) ** 2) + np.exp(-5 * (t - 0.75) ** 2) + 0.05)

    # Cosine power (interpolates between flat and peaked)
    for p in [0.3, 0.5, 0.7, 1.0, 1.5, 2.0, 3.0]:
        inits.append((0.5 * (1 + np.cos(2 * np.pi * t))) ** p + 0.01)

    # Polynomial with positive coeff
    for deg in [2, 3, 4]:
        x = rng.uniform(0.5, 2.0, deg + 1)
        x = x / x.sum()
        poly = np.zeros(deg + 1)
        poly[0] = x[0]
        for i in range(1, deg + 1):
            poly[i] = x[i] * (poly[i - 1] if i == 1 else 1.0)
        # Just use simple polynomials
        if deg == 2:
            inits.append(1.0 + 0.3 * (2 * t - 1) ** 2)
        elif deg == 3:
            inits.append(1.0 + 0.2 * (2 * t - 1) * (1 - (2 * t - 1) ** 2))

    # Smooth random
    ksize = max(5, n // 15)
    kernel = np.ones(ksize) / ksize
    for _ in range(3):
        noise = np.convolve(rng.uniform(0, 1, n), kernel, mode='same')
        inits.append(np.maximum(noise + 0.1, 0.01))

    # Flat with small perturbation
    inits.append(1.0 + 0.05 * rng.normal(0, 1, n))
    inits[-1] = np.maximum(inits[-1], 0.01)

    # Sawtooth smooth
    inits.append(np.maximum(0.5 + 0.3 * np.sin(2 * np.pi * 2 * t), 0.01))
    inits.append(np.maximum(0.5 + 0.3 * np.sin(2 * np.pi * 3 * t) + 0.1 * np.sin(2 * np.pi * 7 * t), 0.01))

    return inits


def run_smooth_grad_descent(a, max_iters, deadline, tau_init=5.0, tau_final=100.0,
                            lr_init=0.3):
    """Smooth gradient descent with increasing tau."""
    a = np.maximum(a.astype(np.float64), 1e-8)
    a = a / a.sum()
    n = len(a)
    best_a = a.copy()
    best_M = fast_conv(a).max()
    v = np.zeros(n)

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        frac = it / max(1, max_iters)
        tau = tau_init + (tau_final - tau_init) * (frac ** 0.7)
        lr = lr_init * (0.02 / lr_init) ** frac

        b = fast_conv(a)
        b_max = b.max()
        exp_b = np.exp(tau * (b - b_max))
        p = exp_b / exp_b.sum()

        grad = np.zeros(n)
        for k in range(2 * n - 1):
            pk = p[k]
            if pk < 1e-13:
                continue
            lo = max(0, k - n + 1)
            hi = min(n - 1, k)
            grad[lo:hi + 1] += 2.0 * pk * a[k - hi:k - lo + 1][::-1]

        v = 0.9 * v + lr * grad
        a = np.maximum(a - v, 0.0)
        s = a.sum()
        if s < 1e-10:
            a = best_a.copy()
        else:
            a /= s

        M = fast_conv(a).max()
        if M < best_M:
            best_M = M
            best_a = a.copy()

    return best_a


def resample_seq(src, dest_n):
    """Resample 1D sequence to new length via cubic interpolation."""
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    from scipy.interpolate import CubicSpline
    try:
        cs = CubicSpline(t_old, src, bc_type='not-a-knot')
        result = cs(t_new)
    except Exception:
        result = np.interp(t_new, t_old, src)
    return np.maximum(result, 0.0)


def propose_candidate(seed=42, budget_s=1500, **kwargs):
    """
    Multi-phase optimization:
    Phase 1: Multi-length exploration with analytic inits + smooth gradient (300s)
    Phase 2: L-BFGS-B refinement with high tau on top candidates (400s)
    Phase 3: Active set Newton refinement (250s)
    Phase 4: Structured perturbation + re-optimization (350s)
    Phase 5: Final polish and length adaptation (150s)
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + float(budget_s) - 20.0
    t_start = time.monotonic()

    best_score = float('inf')
    best_a = None

    def track_candidate(a):
        nonlocal best_score, best_a
        a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
        if a.size == 0 or a.sum() < 0.01:
            return
        score = float(evaluate_sequence(a.tolist()))
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_a = a.copy()
            elapsed = time.monotonic() - t_start
            print(f"  ** NEW BEST: J={score:.8f} n={a.size} t={elapsed:.0f}s", flush=True)

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        if p.ndim == 1 and p.size > 0 and np.all(np.isfinite(p)):
            parent = np.maximum(p, 0.0)
            track_candidate(parent)
            print(f"Parent: n={parent.size}, J={best_score:.6f}", flush=True)
    except (NameError, AttributeError):
        pass

    if parent is None:
        parent = np.ones(256)
        track_candidate(parent)

    # === PHASE 1: Exploration ===
    p1_deadline = t_start + 280
    print(f"\n{'='*60}", flush=True)
    print(f"PHASE 1: Multi-length Exploration", flush=True)
    print(f"{'='*60}", flush=True)

    lengths = [256, 320, 384, 448, 512, 576, 640, 768, 896, 1024]
    # If parent exists, include its length
    if parent is not None and parent.size not in lengths:
        lengths.append(parent.size)
    lengths = sorted(set(l for l in lengths if 128 <= l <= 1200))

    candidates = []  # (score, a) pairs

    for n in lengths:
        if time.monotonic() > p1_deadline - 30:
            break

        # Time per length: less for longer
        time_budget = min(40.0, 200.0 * (300.0 / n))
        phase_dl = min(time.monotonic() + time_budget, p1_deadline - 20)

        inits = generate_analytic_inits(n, rng)
        # Add parent-based init
        if parent is not None:
            inits.append(resample_seq(parent, n))
            p_resampled = resample_seq(parent, n)
            inits.append(np.maximum(p_resampled + rng.normal(0, 0.05, n), 0.01))

        best_M_for_n = float('inf')
        best_a_for_n = None

        for a0 in inits:
            if time.monotonic() > phase_dl:
                break
            a0 = np.maximum(a0, 1e-8)
            a0 /= a0.sum()

            # Quick smooth gradient
            max_it = max(80, min(300, int(15000 / n)))
            a = run_smooth_grad_descent(a0, max_it, phase_dl,
                                        tau_init=3.0, tau_final=60.0, lr_init=0.25)
            M = fast_conv(a).max()
            if M < best_M_for_n:
                best_M_for_n = M
                best_a_for_n = a.copy()

        if best_a_for_n is not None:
            J_val = compute_J(best_a_for_n)
            print(f"  n={n:4d}: J={J_val:.6f} ({time.monotonic()-t_start:.0f}s)", flush=True)
            candidates.append((J_val, best_a_for_n.copy()))
            track_candidate(best_a_for_n)

    # Sort candidates by score
    candidates.sort(key=lambda x: x[0])
    print(f"  Top candidates: {min(len(candidates),5)} found", flush=True)
    for i, (s, a) in enumerate(candidates[:5]):
        print(f"    #{i+1}: J={s:.6f} n={a.size}", flush=True)

    # === PHASE 2: L-BFGS-B Refinement ===
    p2_deadline = t_start + 650
    print(f"\n{'='*60}", flush=True)
    print(f"PHASE 2: L-BFGS-B Refinement (tau=100-300)", flush=True)
    print(f"{'='*60}", flush=True)

    top_k = min(6, len(candidates))
    if best_a is not None and best_a not in [c[1] for c in candidates[:top_k]]:
        candidates.insert(0, (best_score, best_a.copy()))
        top_k = min(top_k + 1, len(candidates))

    for idx in range(top_k):
        if time.monotonic() > p2_deadline - 30:
            break
        J_val, a = candidates[idx]
        n = a.size
        print(f"  Refining candidate {idx+1} (n={n}, J={J_val:.6f})...", flush=True)

        # Progressive tau increase
        for tau in [50, 150, 300]:
            remaining = p2_deadline - time.monotonic()
            if remaining < 30:
                break
            max_it = max(100, int(remaining * 2))
            a = run_lbfgs(a, tau, max_iter=max_it, deadline=p2_deadline - 10)
            J_new = compute_J(a)
            if J_new < J_val:
                J_val = J_new
            candidates[idx] = (J_val, a.copy())
            track_candidate(a)

        print(f"    -> J={J_val:.8f}", flush=True)

    # Update best
    if best_a is not None:
        candidates = [(0, best_a.copy())] + [c for c in candidates if c[1] is not best_a]

    # === PHASE 3: Active Set Newton ===
    p3_deadline = t_start + 900
    print(f"\n{'='*60}", flush=True)
    print(f"PHASE 3: Active Set Newton Refinement", flush=True)
    print(f"{'='*60}", flush=True)

    if best_a is not None:
        n_best = best_a.size
        print(f"  n={n_best}", flush=True)

        # Multiple active set Newton runs with different tolerances
        for rel_tol in [1e-4, 1e-5, 1e-6]:
            if time.monotonic() > p3_deadline - 30:
                break
            a_ns = active_set_newton(best_a, max_iters=30, deadline=p3_deadline - 10)
            J_ns = compute_J(a_ns)
            if J_ns < best_score:
                print(f"    Newton (tol={rel_tol}): J={J_ns:.8f}", flush=True)
            track_candidate(a_ns)

            # Follow with L-BFGS-B
            if time.monotonic() < p3_deadline - 60:
                a = run_lbfgs(a_ns, tau=200, max_iter=200, deadline=p3_deadline - 20)
                track_candidate(a)

        # Also try on second best candidate
        if len(candidates) > 1:
            J2, a2 = candidates[1]
            if time.monotonic() < p3_deadline - 60:
                a_ns2 = active_set_newton(a2, max_iters=20, deadline=p3_deadline - 30)
                track_candidate(a_ns2)
                a = run_lbfgs(a_ns2, tau=200, max_iter=150, deadline=p3_deadline - 10)
                track_candidate(a)

    # === PHASE 4: Structured Perturbation ===
    p4_deadline = t_start + 1200
    print(f"\n{'='*60}", flush=True)
    print(f"PHASE 4: Structured Perturbation + Re-optimization", flush=True)
    print(f"{'='*60}", flush=True)

    if best_a is not None:
        n_best = best_a.size
        n_perturb = min(30, max(10, int((p4_deadline - time.monotonic()) / 10)))

        for p in range(n_perturb):
            if time.monotonic() > p4_deadline - 30:
                break

            a = multiplicative_perturbation(best_a, rng, scale=0.03 * (1 + 0.5 * p / n_perturb))
            a /= a.sum()

            # Quick L-BFGS-B
            a = run_lbfgs(a, tau=100, max_iter=100, deadline=p4_deadline - 20)
            track_candidate(a)

            # If improved, do active set Newton
            if best_score < float('inf') and time.monotonic() < p4_deadline - 10:
                a_ns = active_set_newton(best_a, max_iters=10, deadline=p4_deadline - 5)
                track_candidate(a_ns)

        # Larger perturbations
        for scale in [0.1, 0.2, 0.3]:
            if time.monotonic() > p4_deadline - 30:
                break
            a = multiplicative_perturbation(best_a, rng, scale=scale)
            a /= a.sum()
            a = run_smooth_grad_descent(a, 100, p4_deadline - 30, tau_init=5, tau_final=50)
            a = run_lbfgs(a, tau=100, max_iter=100, deadline=p4_deadline - 10)
            track_candidate(a)

    # === PHASE 5: Final Polish ===
    print(f"\n{'='*60}", flush=True)
    print(f"PHASE 5: Final Polish", flush=True)
    print(f"{'='*60}", flush=True)

    if best_a is not None:
        # Final L-BFGS-B with very high tau
        if time.monotonic() < deadline - 30:
            a = run_lbfgs(best_a, tau=500, max_iter=300, deadline=deadline - 20)
            track_candidate(a)

        # Active set Newton one more time
        if time.monotonic() < deadline - 20:
            a_ns = active_set_newton(best_a, max_iters=30, deadline=deadline - 10)
            track_candidate(a_ns)
            if time.monotonic() < deadline - 5:
                a = run_lbfgs(a_ns, tau=300, max_iter=100, deadline=deadline - 2)
                track_candidate(a)

        # Try different n (scale up/down)
        n_best = best_a.size
        for new_n in [n_best + 64, n_best - 64, n_best * 2]:
            if time.monotonic() > deadline - 20:
                break
            if new_n < 128 or new_n > 2000:
                continue
            a_ext = resample_seq(best_a, new_n)
            a_ext = np.maximum(a_ext, 1e-8)
            a_ext /= a_ext.sum()
            a_ext = run_lbfgs(a_ext, tau=80, max_iter=100, deadline=deadline - 10)
            track_candidate(a_ext)

    # Final report
    if best_a is not None:
        final_score = float(evaluate_sequence(best_a.tolist()))
        print(f"\n{'='*60}", flush=True)
        print(f"FINAL: J={final_score:.8f} n={best_a.size}", flush=True)
        print(f"{'='*60}", flush=True)
    else:
        best_a = np.ones(256)
        final_score = compute_J(best_a)
        print(f"FALLBACK: J={final_score:.6f}", flush=True)

    return [float(x) for x in best_a]
