from scipy.optimize import linprog
from scipy.sparse import csr_matrix
import time

def fast_conv(a):
    """Full self-convolution via FFT."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]

def compute_M(a):
    """Compute max of self-convolution."""
    return float(fast_conv(a).max())

def compute_J(a):
    """Compute J(a) = 2*n*max(conv(a,a))/(sum(a))^2."""
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))

def line_search_analytic(a, g):
    """
    Find best t in [0,1] minimizing max_k ((1-t)*a + t*g *sum(a)/sum(g))^2.
    Uses the fact that each (a_t * a_t)[k] is quadratic in t.
    """
    # Normalize both to sum 1
    sa = a.sum()
    sg = g.sum()
    if sa < 1e-15 or sg < 1e-15:
        return 0.0, compute_M(a)
    a = a / sa
    g = g / sg

    A = fast_conv(a)  # (a*a)[k]
    B = np.convolve(a, g)  # (a*g)[k]
    C = fast_conv(g)  # (g*g)[k]

    #        = A_k + 2(B_k - A_k)*t + (A_k - 2*B_k + C_k)*t^2
    # = A_k + 2(B_k - A_k)*t + (C_k - 2*B_k + A_k)*t^2

    b_coef = 2 * (B - A)  # linear term
    c_coef = (C - 2 * B + A)  # quadratic term (should be >= 0 since it's (g-k*a-g_k)^2... not quite)

    # The quadratic coefficient c_coef[k] = C_k - 2*B_k + A_k = (a_k - g_k)^2 essentially... 
    # No, it's not exactly that. But f_k(t) >= 0 for all t, so the quadratic is bounded below.
    
    # Ternary search on [0, 1]
    lo, hi = 0.0, 1.0
    for _ in range(80):

        m2 = hi - (hi - lo) / 3
        f1 = np.max(a_coef + b_coef * m1 + c_coef * m1**2)
        f2 = np.max(a_coef + b_coef * m2 + c_coef * m2**2)
        if f1 < f2:

        else:

    # Clamp to valid range
    t = max(0.0, min(1.0, t))
    best_val = float(np.max(a_coef + b_coef * t + c_coef * t**2))

def fast_conv(a):
    """Full self-convolution via FFT."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]


def cross_conv(a, b):

    n = len(a)

    fb = np.fft.rfft(b, size)
    return np.fft.irfft(fa * fb, size)[:2 * n - 1]

    t_start = time.monotonic()
    deadline = t_start + float(budget_s) - 15.0
    tracker = Tracker(t_start)

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        if p.ndim == 1 and p.size > 0 and np.all(np.isfinite(p)):
            parent = np.maximum(p, 0.0)
            tracker.update(parent, "(parent)")
            print(f"Parent loaded: n={parent.size}, J={tracker.best_score:.8f}", flush=True)
    except (NameError, AttributeError):
        pass

    if parent is None:
        parent = np.ones(256)
        tracker.update(parent, "(default)")

    # Phase 1: LP mirror descent on parent
    print(f"\n{'='*60}", flush=True)
    print(f"PHASE 1: LP Mirror Descent n={parent.size}", flush=True)
    print(f"{'='*60}", flush=True)

    n0 = parent.size
    row_idx, col_idx = build_lp_matrix(n0)
    a = parent.copy()
    a = a / a.sum()

    phase1_deadline = min(t_start + budget_s * 0.35, deadline)
    a_opt = run_lp_mirror_descent(a, 50000, phase1_deadline, row_idx, col_idx, verbose=True)
    tracker.update(a_opt, "(P1 LP-MD)")

    # Phase 2: Smooth gradient from best
    print(f"\n{'='*60}", flush=True)
    print(f"PHASE 2: Smooth Gradient Polish", flush=True)
    print(f"{'='*60}", flush=True)

    if tracker.best is not None:
        n_cur = tracker.best.size
        if n_cur != n0:
            row_idx2, col_idx2 = build_lp_matrix(n_cur)
        else:
            row_idx2, col_idx2 = row_idx, col_idx

        a_sg = tracker.best.copy()
        a_sg = a_sg / a_sg.sum()
        a_sg = run_smooth_gradient(a_sg, 2000, min(t_start + budget_s * 0.45, deadline),
                                   initial_tau=5.0, final_tau=150.0)
        tracker.update(a_sg, "(P2 smooth)")

        # LP polish after smooth
        a_lp = a_sg.copy()
        a_lp = a_lp / a_lp.sum()
        a_lp = run_lp_mirror_descent(a_lp, 10000, min(t_start + budget_s * 0.55, deadline),
                                     row_idx2, col_idx2, verbose=False)
        tracker.update(a_lp, "(P2 LP polish)")

    # Phase 3: Multi-length exploration
    print(f"\n{'='*60}", flush=True)
    print(f"PHASE 3: Multi-length Exploration", flush=True)
    print(f"{'='*60}", flush=True)

    if tracker.best is not None:
        base_n = tracker.best.size
        trial_lengths = []
        for mult in [0.5, 0.75, 1.25, 1.5, 2.0]:
            n_trial = int(base_n * mult)
            if 64 <= n_trial <= 4096 and n_trial != base_n:
                trial_lengths.append(n_trial)
        trial_lengths = sorted(set(trial_lengths))

        phase3_deadline = min(t_start + budget_s * 0.75, deadline)
        for n_trial in trial_lengths:
            if time.monotonic() > phase3_deadline:
                break

            a_init = resample_seq(tracker.best, n_trial)
            a_init = np.maximum(a_init, 1e-6)
            a_init = a_init / a_init.sum()

            r_idx, c_idx = build_lp_matrix(n_trial)
            sub_deadline = min(time.monotonic() + 80, phase3_deadline)
            a_opt = run_lp_mirror_descent(a_init, 5000, sub_deadline, r_idx, c_idx, verbose=False)
            tracker.update(a_opt, f"(P3 n={n_trial})")

    # Phase 4: Perturbation restarts
    print(f"\n{'='*60}", flush=True)
    print(f"PHASE 4: Perturbation Restarts", flush=True)
    print(f"{'='*60}", flush=True)

    if tracker.best is not None:
        n_best = tracker.best.size
        r_idx_b, c_idx_b = build_lp_matrix(n_best)
        temperature = 0.08
        phase4_deadline = min(t_start + budget_s * 0.90, deadline)

        for r in range(40):
            if time.monotonic() > phase4_deadline:
                break

            a = tracker.best.copy().astype(np.float64)
            noise_scale = temperature * 0.015
            a = a + rng.normal(0, noise_scale, n_best)
            a = np.maximum(a, 1e-6)
            a = a / a.sum()

            # Quick smooth gradient first
            sub_dl = min(time.monotonic() + 15, phase4_deadline)
            a = run_smooth_gradient(a, 200, sub_dl, initial_tau=10, final_tau=80)

            # Then LP
            sub_dl2 = min(time.monotonic() + 30, phase4_deadline)
            a_opt = run_lp_mirror_descent(a, 3000, sub_dl2, r_idx_b, c_idx_b, verbose=False)
            tracker.update(a_opt, f"(P4 r={r} T={temperature:.4f})")

            temperature *= 0.88

    # Phase 5: Final extended LP polish
    print(f"\n{'='*60}", flush=True)
    print(f"PHASE 5: Final LP Polish", flush=True)
    print(f"{'='*60}", flush=True)

    if tracker.best is not None:
        n_final = tracker.best.size
        r_f, c_f = build_lp_matrix(n_final)
        a = tracker.best.copy()
        a = a / a.sum()
        a_opt = run_lp_mirror_descent(a, 50000, deadline, r_f, c_f, verbose=True)
        tracker.update(a_opt, "(P5 final)")

    # Fallback
    if tracker.best is None:
        tracker.best = parent.copy()
        tracker.best_score = compute_J(tracker.best)

    print(f"\n{'='*60}", flush=True)
    print(f"FINAL: J={tracker.best_score:.8f} n={tracker.best.size} "
          f"time={time.monotonic()-t_start:.0f}s", flush=True)
    print(f"{'='*60}", flush=True)

    return [float(x) for x in tracker.best]
