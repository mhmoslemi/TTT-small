    """Find optimal value of a[i] given all other coordinates fixed."""
    n = len(a)
    t = a[i]

    # Terms involving a[i]:

    #   k-j=i: a[k-i]*a[i] (valid if 0<=k-i<n, i.e., i<=k<i+n)
    
    # Full convolution
    conv = np.convolve(a, a)
    
    # Remove a[i]'s contribution
    # a[i] contributes to conv[k] for k in [i, i+n-1] (from j=i: a[i]*a[k-i])

    #   Otherwise: total = 2*a[i]*a[k-i]

        elif order_mode == 'sorted':
            order = np.argsort(a)[::-1]  # largest first
        elif order_mode == 'sorted_asc':
            order = np.argsort(a)  # smallest first
        else:
            order = np.arange(n)

        a = coord_descent_pass(a, order, damping=damping)
        a = np.maximum(a, 0.0)
        s = np.sum(a)
        if s > 1e-10:
            a = a / s

        obj = compute_obj(a)
        if obj < best_obj - 1e-12:
            best_obj = obj
            best_a = a.copy()
            stall = 0
        else:
            stall += 1
            if stall > patience:
                break

    return best_a


def build_toeplitz(a):
    """Build Toeplitz matrix for convolution with a."""
    n = len(a)
    A = np.zeros((2 * n - 1, n), dtype=np.float64)
    for j in range(n):
        k_start = j
        k_end = min(j + n, 2 * n - 1)
        A[k_start:k_end, j] = a[:k_end - k_start]
    return A


def solve_lp_maxsum(a, M, time_limit=5):
    """Solve LP: max sum(g) s.t. (a*g)[k] <= M, g >= 0."""
    n = len(a)
    A = build_toeplitz(a)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': time_limit})
        if res.status == 0:
            g = np.maximum(res.x, 0)
            return g, float(np.sum(g))
    except Exception:
        pass
    return None, None


def lp_refine(a, time_budget=20.0, max_iters=20):
    """Alternating LP refinement."""
    n = len(a)
    a = a.copy().astype(np.float64)
    s = np.sum(a)
    if s < 1e-10:
        return a
    a = a / s
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    stall = 0

    for it in range(max_iters):
        if time.monotonic() - start > time_budget - 2:
            break

        conv = compute_conv(a)
        M = float(np.max(conv))

        g, sum_g = solve_lp_maxsum(a, M, time_limit=min(4.0, (time.monotonic() - start + time_budget) * 0.1))
        if g is None or sum_g < 1e-10:
            break

        g_norm = g / sum_g

        best_t = 0.0
        best_obj_ls = best_obj
        for t in [0.1, 0.3, 0.5, 0.7, 1.0]:
            a_new = (1.0 - t) * a + t * g_norm
            a_new = np.maximum(a_new, 0)
            s_new = np.sum(a_new)
            if s_new < 1e-10:
                continue
            a_new = a_new / s_new
            obj_new = compute_obj(a_new)
            if obj_new < best_obj_ls:
                best_obj_ls = obj_new
                best_t = t

        if best_t > 0 and best_obj_ls < best_obj - 1e-12:
            a_new = (1.0 - best_t) * a + best_t * g_norm
            a_new = np.maximum(a_new, 0)
            s_new = np.sum(a_new)
            if s_new > 1e-10:
                a_new = a_new / s_new
            best_obj = best_obj_ls
            best_a = a_new.copy()
            a = a_new
            stall = 0
        else:
            stall += 1
            if stall > 10:
                break
            a = 0.9 * a + 0.1 * g_norm
            a = np.maximum(a, 0)
            s = np.sum(a)
            if s > 1e-10:
                a = a / s

    return best_a


def multiplicative_subgradient(a, time_budget=15.0, tau=20.0, step=0.05, max_iters=2000):
    """Multiplicative subgradient descent."""
    n = len(a)
    a = a.copy().astype(np.float64)
    s = np.sum(a)
    if s < 1e-10:
        return a
    a = a / s
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    stall = 0
    patience = 50

    for it in range(max_iters):
        if time.monotonic() - start > time_budget:
            break

        conv = compute_conv(a)
        max_conv = np.max(conv)
        shifted = tau * (conv - max_conv)
        shifted = np.clip(shifted, -500, 500)
        weights = np.exp(shifted)
        w_sum = np.sum(weights)
        if w_sum < 1e-300:
            weights = np.ones(len(conv)) / len(conv)
        else:
            weights /= w_sum

        c = np.correlate(weights, a, mode='full')
        grad = 2.0 * c[n - 1:n - 1 + n]

        log_update = -step * grad
        log_update = np.clip(log_update, -30, 30)
        a_new = a * np.exp(log_update)
        a_new = np.maximum(a_new, 1e-15)
        s_new = np.sum(a_new)
        if s_new > 1e-10:
            a_new = a_new / s_new
        a = a_new

        obj = compute_obj(a)
        if obj < best_obj - 1e-13:
            best_obj = obj
            best_a = a.copy()
            stall = 0
        else:
            stall += 1
            if stall > patience:
                step *= 0.7
                if step < 0.001:
                    break
                rng = np.random.default_rng(it * 7 + 3)
                a = a * np.exp(rng.normal(0, 0.05, n))
                a = np.maximum(a, 1e-10)
                s = np.sum(a)
                if s > 1e-10:
                    a = a / s
                stall = 0
                patience = max(patience // 2, 20)
        if it > 500:
            step *= 0.998
            tau *= 1.003

    return best_a


def random_init(n, seed):
    """Generate diverse random initializations."""
    rng = np.random.default_rng(seed)
    k = np.arange(n)
    choice = rng.integers(0, 8)

    if choice == 0:
        f = rng.uniform(0.5, 4.0)
        phase = rng.uniform(0, 2 * np.pi)
        a = 1.0 + 0.8 * np.cos(2 * np.pi * f * k / n + phase)
    elif choice == 1:
        f1 = rng.uniform(0.3, 2.0)
        f2 = rng.uniform(2.0, 6.0)
        a = 1.0 + 0.4 * np.cos(2 * np.pi * f1 * k / n) + 0.3 * np.cos(2 * np.pi * f2 * k / n)
    elif choice == 2:
        a = rng.random(n)
        win = max(3, n // 20)
        kernel = np.ones(win) / win
        a = np.convolve(a, kernel, mode='same')
    elif choice == 3:
        a = np.ones(n)
        for _ in range(rng.integers(1, 4)):
            f = rng.uniform(0.5, 5.0)
            amp = rng.uniform(0.1, 0.4)
            a += amp * np.cos(2 * np.pi * f * k / n + rng.uniform(0, 2 * np.pi))
    elif choice == 4:
        center = n * rng.uniform(0.3, 0.7)
        width = n * rng.uniform(0.1, 0.5)
        a = 1.0 + 0.7 * np.exp(-0.5 * ((k - center) / width) ** 2)
    elif choice == 5:
        a = np.ones(n)
        for scale in range(3):
            f = rng.uniform(0.3 * (2 ** scale), 1.5 * (2 ** scale))
            amp = 0.3 / (scale + 1)
            a += amp * np.cos(2 * np.pi * f * k / n + rng.uniform(0, 2 * np.pi))
    elif choice == 6:
        a = rng.random(max(5, n // 10))
        a = np.interp(k, np.arange(len(a)), a)
    else:
        a = np.ones(n)
        a += 0.3 * np.sin(2 * np.pi * 1.3 * k / n) + 0.2 * np.sin(2 * np.pi * 3.7 * k / n + 0.5)

    a = np.maximum(a, 0)
    s = np.sum(a)
    if s < 1e-10:
        a = np.ones(n)
        s = n
    return a / s


def resample_sequence(a, new_n):
    """Resample a sequence to a new length."""
    n = len(a)
    if n == new_n:
        return a.copy()
    old_k = np.arange(n, dtype=np.float64)
    new_k = np.linspace(0, n - 1, new_n)
    a_new = np.interp(new_k, old_k, a)
    a_new = np.maximum(a_new, 0)
    s = np.sum(a_new)
    if s > 1e-10:
        a_new = a_new / s
    return a_new


def optimize_sequence(a, time_budget=60.0):
    """Run full optimization pipeline on a single sequence."""
    n = len(a)
    a = a.copy().astype(np.float64)
    s = np.sum(a)
    if s < 1e-10:
        a = np.ones(n)
        s = n
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    # Phase 1: Coordinate descent (fast, many passes)
    remaining = time_budget - (time.monotonic() - start)
    if remaining > 5:
        cd_time = min(remaining * 0.3, 30.0)
        a_new = coord_descent_optimize(best_a, max_passes=2000,
                                       time_budget=cd_time, damping=1.0,
                                       order_mode='random', seed=123)
        obj = compute_obj(a_new)
        if obj < best_obj:
            best_obj = obj
            best_a = a_new.copy()

    # Phase 2: More coordinate descent with different ordering
    remaining = time_budget - (time.monotonic() - start)
    if remaining > 5:
        cd_time = min(remaining * 0.25, 25.0)
        a_new = coord_descent_optimize(best_a, max_passes=2000,
                                       time_budget=cd_time, damping=0.9,
                                       order_mode='sorted', seed=456)
        obj = compute_obj(a_new)
        if obj < best_obj:
            best_obj = obj
            best_a = a_new.copy()

    # Phase 3: LP refinement
    remaining = time_budget - (time.monotonic() - start)
    if remaining > 10:
        lp_time = min(remaining * 0.2, 20.0)
        a_new = lp_refine(best_a, time_budget=lp_time, max_iters=15)
        obj = compute_obj(a_new)
        if obj < best_obj:
            best_obj = obj
            best_a = a_new.copy()

    # Phase 4: Subgradient polish
    remaining = time_budget - (time.monotonic() - start)
    if remaining > 5:
        sg_time = min(remaining * 0.2, 15.0)
        a_new = multiplicative_subgradient(best_a, time_budget=sg_time,
                                           tau=30.0, step=0.04)
        obj = compute_obj(a_new)
        if obj < best_obj:
            best_obj = obj
            best_a = a_new.copy()

    # Phase 5: Final coordinate descent polish
    remaining = time_budget - (time.monotonic() - start)
    if remaining > 3:
        cd_time = min(remaining * 0.3, 20.0)
        a_new = coord_descent_optimize(best_a, max_passes=5000,
                                       time_budget=cd_time, damping=1.0,
                                       order_mode='random', seed=789)
        obj = compute_obj(a_new)
        if obj < best_obj:
            best_obj = obj
            best_a = a_new.copy()

    return best_a, best_obj


def try_parent():
    """Try to load parent sequence."""
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64)
        parent = np.clip(parent, 0, 1000.0)
        if parent.size > 0 and np.isfinite(parent).all():
            obj = compute_obj(parent)
            return parent.copy(), obj
    except Exception:
        pass
    return None, np.inf


def propose_candidate(seed=42, budget_s=1500):
    """
    Main search function. Uses coordinate descent as the primary optimizer,
    combined with LP refinement, subgradient methods, and n-exploration.
    """
    start_time = time.monotonic()
    deadline = start_time + float(budget_s) - 30.0

    best_seq = None
    best_score = np.inf

    # Load parent
    parent, parent_obj = try_parent()
    if parent is not None and parent_obj < best_score:
        best_score = parent_obj
        best_seq = parent.copy()
        n_parent = len(parent)
    else:
        n_parent = 330

    print(f"Initial best: {best_score:.6f} (n={n_parent if best_seq is not None else 'N/A'})")
    print(f"Time budget: {budget_s}s")

    # Phase 1: Intensive coordinate descent on parent
    if best_seq is not None:
        remaining = deadline - time.monotonic()
        p1_budget = min(remaining * 0.3, 350.0)
        print(f"\n=== Phase 1: Coordinate descent on parent (budget={p1_budget:.0f}s) ===")

        # Multiple rounds of coordinate descent with different settings
        a_current = best_seq.copy()
        s = np.sum(a_current)
        if s > 1e-10:
            a_current = a_current / s

        for round_idx in range(5):
            remaining = deadline - time.monotonic()
            if remaining < 20:
                break
            sub_budget = min(remaining * 0.2, 60.0)

            modes = ['random', 'sorted', 'sorted_asc', 'random', 'sorted']
            order_mode = modes[round_idx % len(modes)]
            dampings = [1.0, 0.95, 1.0, 0.9, 1.0]
            damping = dampings[round_idx % len(dampings)]

            a_new = coord_descent_optimize(a_current, max_passes=5000,
                                           time_budget=sub_budget, damping=damping,
                                           order_mode=order_mode,
                                           seed=seed + round_idx * 137)
            obj = compute_obj(a_new)
            if obj < best_score:
                best_score = obj
                best_seq = a_new.copy()
                a_current = a_new.copy()
                print(f"  Round {round_idx} ({order_mode}, d={damping}): {best_score:.8f}")

    # Phase 2: LP refinement (less frequent, more expensive)
    if best_seq is not None:
        remaining = deadline - time.monotonic()
        if remaining > 30:
            print(f"\n=== Phase 2: LP refinement (remaining={remaining:.0f}s) ===")
            lp_budget = min(remaining * 0.15, 120.0)
            a_new = lp_refine(best_seq, time_budget=lp_budget, max_iters=30)
            obj = compute_obj(a_new)
            if obj < best_score:
                best_score = obj
                best_seq = a_new.copy()
                print(f"  After LP: {best_score:.8f}")

            # Coordinate descent after LP
            remaining = deadline - time.monotonic()
            if remaining > 20:
                cd_budget = min(remaining * 0.1, 60.0)
                a_new = coord_descent_optimize(best_seq, max_passes=5000,
                                               time_budget=cd_budget, damping=1.0,
                                               order_mode='random', seed=seed + 999)
                obj = compute_obj(a_new)
                if obj < best_score:
                    best_score = obj
                    best_seq = a_new.copy()
                    print(f"  After CD post-LP: {best_score:.8f}")

    # Phase 3: Explore nearby n values
    if best_seq is not None:
        remaining = deadline - time.monotonic()
        if remaining > 40:
            print(f"\n=== Phase 3: Nearby n values (remaining={remaining:.0f}s) ===")
            n_values = []
            for delta in [-10, -5, 5, 10, -20, 20, -30, 30, -50, 50]:
                n_val = n_parent + delta
                if 150 <= n_val <= 800:
                    n_values.append(n_val)
            n_values = sorted(set(n_values))

            per_n_budget = min(remaining / max(len(n_values), 1) * 0.8, 80.0)
            for n_val in n_values:
                if time.monotonic() > deadline - 30:
                    break
                remaining = deadline - time.monotonic()
                if remaining < 15:
                    break
                alloc = max(min(per_n_budget, remaining * 0.25), 10.0)

                a_resampled = resample_sequence(best_seq, n_val)
                a_opt, obj = optimize_sequence(a_resampled, time_budget=alloc)
                if obj < best_score:
                    best_score = obj
                    best_seq = a_opt.copy()
                    n_parent = n_val
                    print(f"  ** n={n_val}: {best_score:.8f} **")
                else:
                    print(f"  n={n_val}: {obj:.8f} (no improvement)")

    # Phase 4: Intensive perturbation search on best
    if best_seq is not None:
        remaining = deadline - time.monotonic()
        if remaining > 30:
            print(f"\n=== Phase 4: Perturbation search (remaining={remaining:.0f}s) ===")
            n = len(best_seq)
            rng = np.random.default_rng(seed + 55555)
            attempt = 0
            while time.monotonic() < deadline - 20:
                attempt += 1
                remaining = deadline - time.monotonic()
                if remaining < 15:
                    break

                scale = 0.01 + 0.03 * (attempt % 5) / 5.0
                a_pert = best_seq * np.exp(rng.normal(0, scale, n))
                a_pert = np.maximum(a_pert, 1e-10)
                s_pert = np.sum(a_pert)
                if s_pert < 1e-10:
                    continue
                a_pert = a_pert / s_pert

                sub_time = min(15.0, remaining * 0.3)
                a_opt, obj = optimize_sequence(a_pert, time_budget=sub_time)
                if obj < best_score:
                    best_score = obj
                    best_seq = a_opt.copy()
                    print(f"  Perturb {attempt} (scale={scale:.3f}): {best_score:.8f}")
                if attempt >= 20:
                    break

    # Phase 5: Try different n values from scratch
    if best_seq is not None:
        remaining = deadline - time.monotonic()
        if remaining > 40:
            print(f"\n=== Phase 5: Different n values (remaining={remaining:.0f}s) ===")
            for n_val in [250, 350, 400, 500]:
                if time.monotonic() > deadline - 20:
                    break
                if len(best_seq) == n_val:
                    continue
                remaining = deadline - time.monotonic()
                if remaining < 25:
                    break
                alloc = min(30.0, remaining * 0.25)
                a_init = random_init(n_val, seed + n_val * 7919)
                a_opt, obj = optimize_sequence(a_init, time_budget=alloc)
                if obj < best_score:
                    best_score = obj
                    best_seq = a_opt.copy()
                    print(f"  ** n={n_val}: {best_score:.8f} **")

    # Phase 6: Final polish
    if best_seq is not None:
        remaining = deadline - time.monotonic()
        if remaining > 10:
            print(f"\n=== Phase 6: Final polish (remaining={remaining:.0f}s) ===")
            # Multiple short coordinate descent rounds
            for r in range(3):
                remaining = deadline - time.monotonic()
                if remaining < 5:
                    break
                fb = min(remaining * 0.3, 20.0)
                a_new = coord_descent_optimize(best_seq, max_passes=10000,
                                               time_budget=fb, damping=1.0,
                                               order_mode='random',
                                               seed=seed + 1000 + r * 31)
                obj = compute_obj(a_new)
                if obj < best_score:
                    best_score = obj
                    best_seq = a_new.copy()
                    print(f"  Polish {r}: {best_score:.8f}")

    # Final validation
    if best_seq is None:
        best_seq = np.ones(100) / 100
        best_score = compute_obj(best_seq)

    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    val = evaluate_sequence(result)
    if val < best_score:
        best_score = val

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(best_seq)}, score={best_score:.8f}, time={elapsed:.1f}s ===")

    return result
