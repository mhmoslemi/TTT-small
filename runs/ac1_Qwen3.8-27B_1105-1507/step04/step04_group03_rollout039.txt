J(a) = 2*n*max(conv(a,a))/(sum(a))^2.

from scipy.optimize import linprog, minimize

def fast_conv(a):
    """Full self-convolution via FFT."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]

        #               = 0 otherwise

        # i.e., it's the "cross-correlation" of delta_k with a, which is just a shifted/reversed a
        # grad_bk[j] = 2*a[k-j] (with bounds handling)
        # This is equivalent to: (reversed a) shifted by k
        # Let's just do it directly with numpy
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)

        # Handle the diagonal case where j = k-j (i.e., k = 2j)
        if k % 2 == 0:
            mid = k // 2
            if 0 <= mid < n and lo <= mid <= hi:
                vals[mid - lo] = a[mid]  # only count once
        grad[indices] += vals

Wait, I'm overcomplicating the gradient. Let me think again.

b[k] = sum_{i} a[i] * a[k-i] where the sum is over i such that 0 <= i < n and 0 <= k-i < n.

So the valid range of i is: max(0, k-(n-1)) <= i <= min(n-1, k).

db[k]/da[j] for a specific j:
- j appears in the sum as a[j] when i = j (if j is in valid range)
- j appears in the sum as a[k-i] when k-i = j, i.e., i = k-j (if k-j is in valid range)

So:
- If max(0, k-n+1) <= j <= min(n-1, k) AND 0 <= k-j < n:
  - Contribution from first role: a[k-j]
  - If k-j != j AND max(0, k-n+1) <= k-j <= min(n-1, k):
    - Contribution from second role: a[j]
  - If k-j == j (i.e., k = 2j):
    - Only one contribution: a[j] (since it's the same term a[j]*a[j])

Wait, I need to be more careful. In the sum sum_i a[i]*a[k-i], when i=j, the term is a[j]*a[k-j]. When i=k-j (and k-j != j), the term is a[k-j]*a[j]. These are two separate terms in the sum (they come from different values of i). So:

db[k]/da[j] = a[k-j] (from i=j term) + a[k-j] (from i=k-j term, if k-j != j)
= 2*a[k-j] if j != k-j
= a[j] if j == k-j (i.e., k = 2j)

And this is valid when both j and k-j are in [0, n-1].

So the gradient vector (for a fixed k) has entries:
- grad[j] = 2*a[k-j] if 0 <= j < n, 0 <= k-j < n, and j != k-j
- grad[j] = a[j] if k = 2j (and 0 <= j < n)  
- grad[j] = 0 otherwise

For the averaged subgradient over active lags L:
grad_total[j] = (1/|L|) * sum_{k in L} grad_k[j]

This can be computed efficiently:
For each active k, we add 2*a[k-j] to grad[j] for all valid j.
This is like a "smeared" version of a.

Actually, I realize that for each k, the gradient is essentially a "reflected and shifted" copy of a. Specifically:
- Define a_rev[j] = a[k-j] for valid j (this is a reflected copy of a, shifted by k)
- Then grad_k = 2 * a_rev (with the diagonal correction)

So computing the gradient for all active lags is: for each active k, create a reflected/shifted copy of a and add it to the accumulator. This is O(n * |L|) which is fine.

    Uses active set = lags where b[k] >= max(b) - tol_frac * max(b)."""
    n = len(a)

    a = np.maximum(a.astype(np.float64), 1e-10)

        if s < 1e-10:

def lp_refine(a, M, max_iters=50, deadline=None):
    """LP-based best response refinement.
    Solve: max sum(g) s.t. conv(a,g) <= M, g >= 0.
    Then blend a towards g."""
    n = len(a)
    a = np.maximum(a, 1e-10)
    a = a / a.sum()
    
    best_a = a.copy()
    best_M = M
    
    for it in range(max_iters):
        if deadline is not None and time.monotonic() > deadline:
            break
        
        b = fast_conv(a)
        M = b.max()
        
        # Build the LP: max sum(g) s.t. A@g <= M, g >= 0
        # A is (2n-1) x n, A[k,j] = a[k-j] if valid else 0
        # For efficiency, use the structure
        
        # Actually, let's use a different formulation that's easier to solve:
        # The constraint is: for all k, sum_j a[|k-j|] * g[j] <= M
        # (approximately, using the symmetric structure)
        
        # Let me just build the sparse matrix directly
        from scipy.sparse import csr_matrix
        rows = []
        cols = []
        vals = []
        for k in range(2 * n - 1):
            for j in range(n):
                kj = k - j
                if 0 <= kj < n:
                    rows.append(k)
                    cols.append(j)
                    vals.append(a[kj])
        
        A = csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))
        
        try:
            c = -np.ones(n)
            res = linprog(c, A_ub=A, b_ub=np.full(2*n-1, M), 
                         bounds=[(0, None)]*n, method='highs',
                         options={'presolve': True})
            if res.success:
                g = np.maximum(res.x, 0)
                gs = g.sum()
                if gs > 1e-10:
                    g = g / gs

                        a_trial = (1-t)*a + t*g
                        a_trial = np.maximum(a_trial, 0)

                            best_a = a_trial.copy()
                            a = a_trial
                            break
        except Exception:
            pass
        
        if fast_conv(a).max() >= best_M * (1 - 1e-8):
            # Converged
            break
    
    return best_a, best_M

"""
Hybrid optimizer for minimizing J(a) = 2*n*max(conv(a,a))/(sum(a))^2.
Uses: aggressive LP mirror descent with adaptive step sizes, subgradient
polishing with momentum, perturbation-based restarts to escape local minima,

import numpy as np
from scipy.optimize import linprog
from scipy.sparse import csr_matrix
import time


def fast_conv(a):
    """Full self-convolution via FFT."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]


def compute_J(a):
    """Compute J(a) = 2*n*max(conv(a,a))/(sum(a))^2."""
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-10:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))


def build_conv_matrix(n, a):
    """Build sparse matrix A where A[k,j] = a[k-j] for valid j."""
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        j_min = max(0, k - n + 1)
        j_max = min(n - 1, k)
        if j_min <= j_max:
            js = np.arange(j_min, j_max + 1)
            rows.extend([k] * len(js))
            cols.extend(js.tolist())
            vals.extend(a[k - js].tolist())
    return csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))


def lp_best_response(a, M, relaxation=1.0):
    """Solve LP: max sum(g) s.t. A@g <= M*relaxation, g >= 0."""
    n = len(a)
    A = build_conv_matrix(n, a)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M * relaxation)
    bounds = [(0, None)] * n
    try:
        r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                    options={'presolve': True, 'time_limit': 30})
        if r.success and r.x is not None:
            g = np.maximum(r.x, 0)
            if g.sum() > 1e-15:
                return g
    except Exception:
        pass
    return None


def lp_mirror_descent(a, max_iters, deadline, use_multiplicative=False,
                      t_init=0.5, t_decay=0.97, t_min=0.02,
                      relaxation=1.0, subgrad_every=50):
    """Enhanced LP mirror descent with adaptive step sizes and periodic subgradient."""
    a = np.maximum(a.astype(np.float64), 1e-10)
    a = a / a.sum()
    current_M = fast_conv(a).max()
    best_M = current_M
    best_a = a.copy()
    no_improve = 0
    n = len(a)

    t = t_init
    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        # Periodic subgradient step to escape stagnation
        if it > 0 and it % subgrad_every == 0:
            b = fast_conv(a)
            M = b.max()
            tol = 1e-6 * M
            active = np.where(b >= M - tol)[0]
            grad = np.zeros(n)
            for k in active:
                j_min = max(0, k - n + 1)
                j_max = min(n - 1, k)
                js = np.arange(j_min, j_max + 1)
                contrib = 2.0 * a[k - js]
                if k % 2 == 0:
                    mid = k // 2
                    if j_min <= mid <= j_max:
                        contrib[mid - j_min] = a[mid]
                grad[js] += contrib
            grad /= max(1, len(active))
            gnorm = np.linalg.norm(grad)
            if gnorm > 1e-12:
                step = 0.05 * M / gnorm
                a_sub = a - step * grad / gnorm
                a_sub = np.maximum(a_sub, 0)
                s = a_sub.sum()
                if s > 1e-10:
                    a_sub = a_sub / s
                    sub_M = fast_conv(a_sub).max()
                    if sub_M < current_M * 1.001:
                        a = a_sub
                        current_M = sub_M

        b = fast_conv(a)
        M = float(b.max())
        g = lp_best_response(a, M, relaxation=relaxation)
        if g is None:
            break
        g = np.maximum(g, 0.0)
        g_sum = g.sum()
        if g_sum < 1e-10:
            break
        g = g / g_sum

        # Try a range of step sizes (aggressive to conservative)
        best_trial_M = current_M
        best_trial_a = None
        t_try = t
        for ls in range(30):
            if use_multiplicative:
                ratio = g / np.maximum(a, 1e-15)
                a_trial = a * np.power(ratio, t_try)
            else:
                a_trial = (1.0 - t_try) * a + t_try * g
            a_trial = np.maximum(a_trial, 0.0)
            s = a_trial.sum()
            if s < 1e-10:
                t_try *= 0.5
                continue
            a_trial = a_trial / s
            trial_M = fast_conv(a_trial).max()
            if trial_M < best_trial_M - 1e-14:
                best_trial_M = trial_M
                best_trial_a = a_trial.copy()
                break
            t_try *= 0.6
            if t_try < 0.001:
                break

        if best_trial_a is not None:
            a = best_trial_a
            current_M = best_trial_M
            if current_M < best_M:
                best_M = current_M
                best_a = a.copy()
                no_improve = 0
            else:
                no_improve += 1
            # Decay step size slowly
            t = max(t_min, t * t_decay)
        else:
            no_improve += 1
            # Try larger step (might be oscillating)
            t = min(1.0, t * 1.3) if no_improve % 10 == 0 else t * 0.8
            t = max(t_min, min(1.0, t))
            if no_improve > 80:
                break

    return best_a, best_M


def subgradient_polish(a, max_iters, deadline, momentum_beta=0.85):
    """Subgradient descent with momentum for polishing."""
    a = np.maximum(a.astype(np.float64), 1e-10)
    a = a / a.sum()
    n = len(a)
    b = fast_conv(a)
    best_M = b.max()
    best_a = a.copy()
    momentum = np.zeros(n)
    beta = momentum_beta
    it_count = 0

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        it_count += 1

        b = fast_conv(a)
        M = b.max()
        if M < best_M:
            best_M = M
            best_a = a.copy()

        tol = 1e-8 * M
        active = np.where(b >= M - tol)[0]
        if len(active) == 0:
            active = np.array([np.argmax(b)])

        grad = np.zeros(n)
        for k in active:
            j_min = max(0, k - n + 1)
            j_max = min(n - 1, k)
            js = np.arange(j_min, j_max + 1)
            contrib = 2.0 * a[k - js]
            if k % 2 == 0:
                mid = k // 2
                if j_min <= mid <= j_max:
                    contrib[mid - j_min] = a[mid]
            grad[js] += contrib
        grad /= len(active)

        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-14:
            break

        step = 0.3 / (gnorm * np.sqrt(it + 1.0) + 1e-10)
        momentum = beta * momentum + (1 - beta) * grad / gnorm
        a_new = a - step * momentum
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s < 1e-10:
            break
        a_new = a_new / s
        a = a_new
        beta = max(0.5, beta * 0.9995)

    return best_a, best_M


def resample_seq(src, dest_n):
    """Resample 1D sequence to new length."""
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)


def structured_perturb(a, rng, ptype):
    """Apply structured perturbation to escape local minima."""
    n = len(a)
    t = np.arange(n, dtype=np.float64) / n

    if ptype == 'shift':
        s = rng.integers(1, max(2, n // 8))
        a_p = np.roll(a, s)
    elif ptype == 'cosine':
        freq = rng.uniform(1, 10)
        amp = rng.uniform(0.03, 0.15)
        a_p = a * (1 + amp * np.cos(2 * np.pi * freq * t + rng.uniform(0, 2 * np.pi)))
    elif ptype == 'smooth':
        k = rng.integers(3, max(4, n // 8))
        kernel = np.ones(k) / k
        a_p = np.convolve(a, kernel, mode='same')
    elif ptype == 'taper':
        alpha = rng.uniform(0.4, 0.9)
        window = np.ones(n)
        edge = max(1, int(n * (1 - alpha) / 2))
        for i in range(edge):
            frac = (i + 1) / (edge + 1)
            window[i] = frac
            window[n - 1 - i] = frac
        a_p = a * window
    elif ptype == 'noise':
        sigma = rng.uniform(0.02, 0.1)
        a_p = a + rng.normal(0, sigma, n)
    elif ptype == 'mirror':
        half = n // 2
        a_p = np.concatenate([a[:half], a[half::-1]])
        if len(a_p) < n:
            a_p = np.concatenate([a_p, a[:n - len(a_p)]])
        a_p = a_p[:n]
    else:
        a_p = a + rng.normal(0, 0.05, n)

    a_p = np.maximum(a_p, 0.001)
    s = a_p.sum()
    if s > 1e-8:
        a_p = a_p / s
    return a_p

                dv2 = min(deadline - 280, time.monotonic() + per_pert * 0.4)
                a_p2, _ = lp_mirror_descent(a_pert2, 150, dv2, use_multiplicative=False,
                                             t_init=0.4, t_decay=0.97, t_min=0.02)
                J_p2 = compute_J(a_p2)
                tracker.update(a_p2)

                cand_J = min(J_p1, J_p2)
                cand_a = a_p1 if J_p1 <= J_p2 else a_p2
            else:
                cand_J = J_p1
                cand_a = a_p1

            if cand_J < best_J:
                best_a = cand_a
                best_J = cand_J
                print(f"  [{ptype}] J={cand_J:.8f}", flush=True)

        print(f"Phase 4 done ({time.monotonic()-t0:.1f}s)", flush=True)

    # === Phase 5: Multi-length exploration ===
    if time.monotonic() < deadline - 200:
        t0 = time.monotonic()
        print(f"\n{'='*50}", flush=True)
        print(f"Phase 5: Multi-length (base n={best_n})", flush=True)
        print(f"{'='*50}", flush=True)

        base_n = best_n
        lengths = sorted(set([
            max(128, base_n - 64),
            max(128, base_n // 2),
            base_n + 32, base_n + 64, base_n + 128,
            min(2048, base_n * 2)
        ]))
        lengths = [x for x in lengths if 128 <= x <= 2048 and x != base_n]

        for n_try in lengths:
            if time.monotonic() > deadline - 150:
                break

            a_try = resample_seq(best_a, n_try)
            a_try = np.maximum(a_try, 1e-10)
            a_try = a_try / a_try.sum()

            perf = max(20, min(60, (deadline - 150 - time.monotonic()) / max(1, len(lengths) - lengths.index(n_try))))
            a_try, _ = lp_mirror_descent(a_try, 300, time.monotonic() + perf,
                                          use_multiplicative=False,
                                          t_init=0.4, t_decay=0.97, t_min=0.02)
            J_try = compute_J(a_try)
            tracker.update(a_try)
            print(f"  n={n_try}: J={J_try:.8f}", flush=True)

            if J_try < best_J:
                best_a = a_try
                best_n = n_try
                best_J = J_try

        print(f"Phase 5 done ({time.monotonic()-t0:.1f}s)", flush=True)

    # === Phase 6: Diverse starts + subgradient ===
    if time.monotonic() < deadline - 150:
        t0 = time.monotonic()
        print(f"\n{'='*50}", flush=True)
        print(f"Phase 6: Diverse starts (n={best_n})", flush=True)
        print(f"{'='*50}", flush=True)

        inits = generate_inits(best_n, rng, parent=best_a)
        # Remove duplicates (first one is parent)
        inits = inits[2:]  # skip exact parent and small noise
        n_inits = len(inits)
        remaining = deadline - 120 - time.monotonic()
        per_init = max(10, remaining / max(1, n_inits))

        for i, a_init in enumerate(inits):
            if time.monotonic() > deadline - 100:
                break

            # Subgradient first (faster)
            dl_sg = time.monotonic() + per_init * 0.5
            a_sg, _ = subgradient_polish(a_init, 5000, dl_sg)
            J_sg = compute_J(a_sg)
            tracker.update(a_sg)

            # Then LP
            if time.monotonic() < deadline - 100:
                dl_lp = min(deadline - 100, time.monotonic() + per_init * 0.5)
                a_lp, _ = lp_mirror_descent(a_sg, 200, dl_lp, use_multiplicative=False,
                                             t_init=0.4, t_decay=0.97, t_min=0.02)
                J_lp = compute_J(a_lp)
                tracker.update(a_lp)
                J_cand = min(J_sg, J_lp)
            else:
                J_cand = J_sg

            if J_cand < best_J:
                best_a = a_sg if J_sg <= J_lp else a_lp
                best_J = J_cand
                print(f"  init {i}: J={J_cand:.8f}", flush=True)

        print(f"Phase 6 done ({time.monotonic()-t0:.1f}s)", flush=True)

    # === Phase 7: Final long LP refinement ===
    if time.monotonic() < deadline - 40:
        t0 = time.monotonic()
        print(f"\n{'='*50}", flush=True)
        print(f"Phase 7: Final long LP (n={best_n})", flush=True)
        print(f"{'='*50}", flush=True)

        a_final = best_a.copy()
        a_final, M_final = lp_mirror_descent(a_final, 10000, deadline - 20,
                                              use_multiplicative=False,
                                              t_init=0.3, t_decay=0.995,
                                              t_min=0.01, relaxation=1.0)
        J_final = compute_J(a_final)
        tracker.update(a_final)

        # Subgradient polish on the result
        if time.monotonic() < deadline - 10:
            a_polish, _ = subgradient_polish(a_final, 5000, deadline - 5)
            J_polish = compute_J(a_polish)
            tracker.update(a_polish)
            if J_polish < J_final:
                a_final = a_polish
                J_final = J_polish

        print(f"  J={J_final:.8f} ({time.monotonic()-t0:.1f}s)", flush=True)

        if J_final < best_J:
            best_a = a_final
            best_J = J_final

    # Final answer
    if tracker.best is not None:
        final_a = tracker.best
    else:
        final_a = best_a

    print(f"\n{'='*50}", flush=True)
    print(f"FINAL: J={tracker.best_score:.8f} n={final_a.size} time={time.monotonic()-tracker.t_start:.0f}s", flush=True)
    print(f"{'='*50}", flush=True)

    return [float(x) for x in final_a]
