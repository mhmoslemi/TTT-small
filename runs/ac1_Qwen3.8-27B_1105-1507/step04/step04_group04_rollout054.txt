def compute_conv(a):
    """Compute conv(a,a) using FFT. Returns array of length 2n-1."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    conv = np.fft.irfft(fa * fa, size)
    return conv[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    conv = compute_conv(a)
    max_b = float(np.max(conv))
    sum_a = float(np.sum(a))
    if sum_a < 0.01:
        return np.inf
    return 2.0 * n * max_b / (sum_a * sum_a)


def build_toeplitz(a):
    """Build Toeplitz matrix A where (A@g)[k] = (a*g)[k] for k=0..2n-2."""
    n = len(a)
    A = np.zeros((2 * n - 1, n), dtype=np.float64)
    for j in range(n):
        k_start = j
        k_end = min(j + n, 2 * n - 1)
        A[k_start:k_end, j] = a[:k_end - k_start]
    return A


def solve_lp_direction(a, M, time_limit=5):
    """Solve LP: max sum(g) s.t. (a*g)[k] <= M for all k, g >= 0."""
    n = len(a)
    A = build_toeplitz(a)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    try:

                      options={'presolve': True, 'time_limit': time_limit})

            return g, float(np.sum(g))

        pass
    return None, None

    s = np.sum(a_new)
    if s > 1e-10:

    return a_new

"""
Step function optimization using coordinate descent, adaptive subgradient,
and alternating LP. Key innovation: efficient 1D coordinate-wise optimization
of the non-convex objective, combined with subgradient and LP polishing.
"""

import numpy as np
from scipy.optimize import linprog, minimize_scalar
import time

try:
    import cvxpy as cp
    HAS_CVXPY = True
except ImportError:
    HAS_CVXPY = False


def compute_conv(a):
    """Compute conv(a,a) using FFT. Returns array of length 2n-1."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    conv = np.fft.irfft(fa * fa, size)
    return conv[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    conv = compute_conv(a)
    max_b = float(np.max(conv))
    sum_a = float(np.sum(a))
    if sum_a < 0.01:
        return np.inf
    return 2.0 * n * max_b / (sum_a * sum_a)


def build_toeplitz(a):
    """Build Toeplitz matrix A where (A@g)[k] = (a*g)[k] for k=0..2n-2."""
    n = len(a)
    A = np.zeros((2 * n - 1, n), dtype=np.float64)
    for j in range(n):
        k_start = j
        k_end = min(j + n, 2 * n - 1)
        A[k_start:k_end, j] = a[:k_end - k_start]
    return A


def solve_lp_direction(a, M, time_limit=3):
    """Solve LP: max sum(g) s.t. (a*g)[k] <= M for all k, g >= 0."""
    n = len(a)
    A = build_toeplitz(a)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': time_limit})
        if res.status == 0:
            g = np.maximum(res.x, 0)
            return g, float(np.sum(g))
    except Exception:
        pass
    return None, None


def subgradient_step(a, tau, step):
    """One multiplicative subgradient step."""
    n = len(a)
    conv = compute_conv(a)
    max_conv = np.max(conv)
    shifted = tau * (conv - max_conv)
    shifted = np.clip(shifted, -500, 500)
    weights = np.exp(shifted)
    w_sum = np.sum(weights)
    if w_sum < 1e-300:
        return a
    weights /= w_sum
    c = np.correlate(weights, a, mode='full')
    grad = 2.0 * c[n - 1:n - 1 + n]
    log_update = -step * grad
    log_update = np.clip(log_update, -50, 50)
    a_new = a * np.exp(log_update)
    a_new = np.maximum(a_new, 1e-15)
    s = np.sum(a_new)
    if s > 1e-10:
        a_new = a_new / s
    return a_new


def adaptive_subgradient(a, time_budget=30.0, tau_start=10.0, tau_end=300.0,
                         step_start=0.12):

    if s < 1e-10:
        return a
    a = a / s
    best_a = a.copy()

    tau = tau_start

    """

        return a

            break

            d_start = j
            d_end = min(j + n, 2 * n - 1)
            d_len = d_end - d_start
            d = a[:d_len].copy()

            if d_start > 0:
                max_inactive = max(max_inactive, float(np.max(conv[:d_start])))
            if d_end < 2 * n - 1:
                max_inactive = max(max_inactive, float(np.max(conv[d_end:])))

            conv_active = conv[d_start:d_end].copy()
            s = 1.0  # we maintain unit sum

                max_all = max(max_active, max_inactive)

                return 2.0 * n * max_all / denom

                                      method='bounded', options={'xatol': 1e-9})

            except Exception:
                x_opt = a_j

            best_obj = obj
            best_a = a.copy()
        elif not any_change:

def resample_sequence(a, new_n):
    """Resample a sequence to a new length using interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    old_k = np.arange(n, dtype=np.float64)
    new_k = np.linspace(0, n - 1, new_n)
    a_new = np.interp(new_k, old_k, a)
    a_new = np.maximum(a_new, 0)
    s = np.sum(a_new)
    if s > 1e-10:
        a_new = a_new / s
    return a_new

        a_new = adaptive_subgradient(best_a, time_budget=sg_time)
        obj = compute_obj(a_new)
        if obj < best_obj:
            best_obj = obj
            best_a = a_new.copy()
            if verbose:
                print(f"  After SG: {best_obj:.8f}")

    # Phase C: LP polish
    remaining = time_budget - (time.monotonic() - start)
    if remaining > 5:
        lp_time = min(remaining * 0.25, 20.0)
        a_new = lp_polish(best_a, time_budget=lp_time)
        obj = compute_obj(a_new)
        if obj < best_obj:
            best_obj = obj
            best_a = a_new.copy()
            if verbose:
                print(f"  After LP: {best_obj:.8f}")

    # Phase D: Final coordinate descent polish
    remaining = time_budget - (time.monotonic() - start)
    if remaining > 5:
        cd_time = min(remaining * 0.3, 30.0)
        a_new = coordinate_descent(best_a, time_budget=cd_time, seed=99999)
        obj = compute_obj(a_new)
        if obj < best_obj:
            best_obj = obj
            best_a = a_new.copy()
            if verbose:
                print(f"  After CD2: {best_obj:.8f}")

    return best_a, best_obj


def local_search(a, time_budget=30.0, seed=0):
    """Perturbation-based local search."""
    n = len(a)
    a = a.copy().astype(np.float64)
    s = np.sum(a)
    if s < 1e-10:
        return a
    a = a / s
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    rng = np.random.default_rng(seed)

    attempt = 0
    while time.monotonic() - start < time_budget - 3:
        attempt += 1
        remaining = time_budget - (time.monotonic() - start)
        if remaining < 5:
            break

        scale = 0.01 + 0.04 * (attempt % 5) / 5.0
        a_pert = best_a * np.exp(rng.normal(0, scale, n))
        a_pert = np.maximum(a_pert, 1e-10)
        s_pert = np.sum(a_pert)
        if s_pert < 1e-10:
            continue
        a_pert = a_pert / s_pert

        sub_time = min(8.0, remaining * 0.3)
        a_opt, obj = full_optimize(a_pert, time_budget=sub_time)

        if obj < best_obj:
            best_obj = obj
            best_a = a_opt.copy()

    return best_a, best_obj


def propose_candidate(seed=42, budget_s=1500):
    """
    Main search function. Uses coordinate descent (primary), adaptive subgradient,
    and LP polishing. Explores multiple n values around the parent.
    """
    start_time = time.monotonic()
    deadline = start_time + float(budget_s) - 20.0

    best_seq = None
    best_score = np.inf
    n_parent = 330

    # Load parent
    parent, parent_obj = try_load_parent()
    if parent is not None and parent_obj < best_score:
        best_score = parent_obj
        best_seq = parent.copy()
        n_parent = len(parent)

    print(f"Initial best: {best_score:.6f} (n={n_parent})")
    print(f"Time budget: {budget_s}s")

    # === Phase 1: Intensive coordinate descent on parent ===
    if best_seq is not None:
        remaining = deadline - time.monotonic()
        p1_budget = min(remaining * 0.3, 400.0)
        print(f"\n=== Phase 1: Full optimize on parent (budget={p1_budget:.0f}s) ===")
        a_opt, obj = full_optimize(best_seq, time_budget=p1_budget, verbose=True)
        if obj < best_score:
            best_score = obj
            best_seq = a_opt.copy()
            print(f"  After Phase 1: {best_score:.8f}")

    # === Phase 2: Local search around best ===
    if best_seq is not None:
        remaining = deadline - time.monotonic()
        p2_budget = min(remaining * 0.2, 200.0)
        print(f"\n=== Phase 2: Local search (budget={p2_budget:.0f}s) ===")
        a_opt, obj = local_search(best_seq, time_budget=p2_budget, seed=seed)
        if obj < best_score:
            best_score = obj
            best_seq = a_opt.copy()
            print(f"  After Phase 2: {best_score:.8f}")
        else:
            print(f"  No improvement (still {best_score:.8f})")

    # === Phase 3: Try nearby n values ===
    print(f"\n=== Phase 3: Nearby n values ===")
    n_candidates = []
    for delta in [-10, -5, 0, 5, 10, 20, 30, 40, -20, -30, 50, -50]:
        n_val = n_parent + delta
        if 150 <= n_val <= 800:
            n_candidates.append(n_val)
    n_candidates = sorted(set(n_candidates))

    remaining = deadline - time.monotonic()
    if remaining > 30:
        per_n = remaining / max(len(n_candidates), 1) * 0.8
        for n_val in n_candidates:
            if time.monotonic() > deadline - 30:
                break
            remaining = deadline - time.monotonic()
            if remaining < 15:
                break
            alloc = min(max(per_n, 15.0), remaining * 0.25, 80.0)

            if best_seq is not None:
                a_init = resample_sequence(best_seq, n_val)
            else:
                a_init = np.ones(n_val) / n_val

            print(f"  n={n_val} (alloc={alloc:.0f}s)")
            a_opt, obj = full_optimize(a_init, time_budget=alloc)
            if obj < best_score:
                best_score = obj
                best_seq = a_opt.copy()
                print(f"  ** NEW BEST: n={n_val}, obj={obj:.8f} **")

    # === Phase 4: Second round of local search on best ===
    if best_seq is not None:
        remaining = deadline - time.monotonic()
        if remaining > 20:
            p4_budget = min(remaining * 0.35, remaining - 10.0)
            print(f"\n=== Phase 4: Second local search (budget={p4_budget:.0f}s) ===")
            a_opt, obj = local_search(best_seq, time_budget=p4_budget, seed=seed + 777)
            if obj < best_score:
                best_score = obj
                best_seq = a_opt.copy()
                print(f"  After Phase 4: {best_score:.8f}")

    # === Phase 5: Final coordinate descent polish ===
    if best_seq is not None:
        remaining = deadline - time.monotonic()
        if remaining > 15:
            p5_budget = min(remaining * 0.4, remaining - 5.0)
            print(f"\n=== Phase 5: Final CD polish (budget={p5_budget:.0f}s) ===")
            a_opt = coordinate_descent(best_seq, time_budget=p5_budget, seed=424242)
            obj = compute_obj(a_opt)
            if obj < best_score:
                best_score = obj
                best_seq = a_opt.copy()
                print(f"  After Phase 5: {best_score:.8f}")

    # === Final: one more full optimize if time allows ===
    if best_seq is not None:
        remaining = deadline - time.monotonic()
        if remaining > 15:
            final_budget = min(remaining * 0.5, remaining - 5.0)
            print(f"\n=== Final optimize (budget={final_budget:.0f}s) ===")
            a_opt, obj = full_optimize(best_seq, time_budget=final_budget, verbose=True)
            if obj < best_score:
                best_score = obj
                best_seq = a_opt.copy()
                print(f"  Final: {best_score:.8f}")

    # Fallback
    if best_seq is None:
        best_seq = np.ones(300)
        best_score = compute_obj(best_seq)

    # Validate with evaluate_sequence
    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    val = evaluate_sequence(result)
    if val < best_score:
        best_score = val
    print(f"\n=== FINAL: n={len(best_seq)}, score={best_score:.8f} ===")
    print(f"    (evaluate_sequence confirms: {val:.8f})")

    return result
