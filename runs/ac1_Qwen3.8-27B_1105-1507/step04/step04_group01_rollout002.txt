    bounds = [(0.0, None)] * n + [(0.0, None)]
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': time_limit})
        if res.status == 0:
            g = np.maximum(res.x[:n], 0)
            s = np.sum(g)
            if s > 1e-10:
                g = g / s
                M_opt = float(res.x[n])
                return g, M_opt
    except Exception:
        pass
    return None, None

def golden_section_search(f, a, b, tol=1e-6, max_iter=50):
    """Golden section search for minimum of f on [a, b]."""
    gr = (np.sqrt(5) - 1) / 2
    c = b - gr * (b - a)
    d = a + gr * (b - a)
    fc = f(c)
    fd = f(d)
    for _ in range(max_iter):
        if b - a < tol:
            break
        if fc < fd:
            b = d
            d = c
            fd = fc
            c = b - gr * (b - a)
            fc = f(c)
        else:
            a = c
            c = d
            fc = fd
            d = a + gr * (b - a)
            fd = f(d)
    return (a + b) / 2

def alternating_lp_refined(a, max_iters=100, time_budget=100.0):
    """Alternating LP with golden section line search."""
    n = len(a)
    a = a.copy().astype(np.float64)
    s = np.sum(a)
    if s < 1e-10:
        return a
    a = a / s
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    stall_count = 0
    
    for it in range(max_iters):
        if time.monotonic() - start > time_budget:
            break
        
        conv = compute_conv(a)
        M = float(np.max(conv))
        
        # Solve LP
        g, M_lp = solve_lp_given_a(a, time_limit=max(5.0, time_budget * 0.1))
        if g is None:
            break
        
        # Golden section search for step size
        def path_obj(t):
            a_try = (1.0 - t) * best_a + t * g
            a_try = np.maximum(a_try, 0)
            s_try = np.sum(a_try)
            if s_try < 1e-10:
                return np.inf
            a_try = a_try / s_try
            return compute_obj(a_try)
        
        best_t = 0.0
        best_t_obj = best_obj
        # Try a few points first
        for t in [0.1, 0.2, 0.3, 0.5, 0.7, 1.0]:
            obj_t = path_obj(t)
            if obj_t < best_t_obj:
                best_t_obj = obj_t
                best_t = t
        
        # If improvement found, refine with golden section
        if best_t > 0 and best_t < 1:
            t_opt = golden_section_search(path_obj, max(0, best_t - 0.2), min(1, best_t + 0.2), max_iter=20)
            obj_opt = path_obj(t_opt)
            if obj_opt < best_t_obj:
                best_t_obj = obj_opt
                best_t = t_opt
        
        if best_t_obj < best_obj - 1e-12:
            best_obj = best_t_obj
            a_try = (1.0 - best_t) * best_a + best_t * g
            a_try = np.maximum(a_try, 0)
            s_try = np.sum(a_try)
            best_a = a_try / s_try
            a = best_a.copy()
            stall_count = 0
        else:
            stall_count += 1
            if stall_count > 15:
                # Perturb
                rng = np.random.default_rng(hash((it, n)) % 2**31)
                scale = 0.05 + 0.05 * stall_count / 20
                a = best_a * np.exp(rng.normal(0, scale, n))
                a = np.maximum(a, 0)
                s = np.sum(a)
                if s > 1e-10:
                    a = a / s
                stall_count = 0
        
        # Also try the LP solution directly
        obj_g = compute_obj(g)
        if obj_g < best_obj - 1e-12:
            best_obj = obj_g
            best_a = g.copy()
            a = g.copy()
            stall_count = 0
    
    return best_a

def softmax(z):
    z = z - np.max(z)
    e = np.exp(z)
    return e / np.sum(e)

def soft_max_obj_and_grad(z, tau):
    """Compute soft-max of conv(softmax(z), softmax(z)) and gradient w.r.t. z."""
    n = len(z)
    a = softmax(z)
    conv = compute_conv(a)
    cmax = float(np.max(conv))
    shifted = np.clip((conv - cmax) / tau, -500.0, 500.0)
    exp_conv = np.exp(shifted)
    Z = float(np.sum(exp_conv))
    soft_max_val = cmax + tau * np.log(Z)
    w = exp_conv / Z
    
    w_rev = w[::-1]
    conv_fw = fft_convolve(a, w_rev)
    f1 = conv_fw[n - 1:2 * n - 1][::-1]
    grad_a = 2.0 * f1
    
    mean_grad = float(np.dot(a, grad_a))
    grad_z = a * (grad_a - mean_grad)
    
    return soft_max_val, grad_z

def lbfgs_soft_max(a_init, tau_list, time_budget=30.0, max_iter=200):
    """L-BFGS-B optimization with softmax parametrization."""
    n = len(a_init)
    a = a_init.copy()
    s = np.sum(a)
    if s < 1e-10:
        a = np.ones(n) / n
    else:
        a = a / s
    a = np.maximum(a, 1e-30)
    z = np.log(a)
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    
    for tau in tau_list:
        if time.monotonic() - start > time_budget:
            break
        
        def func_and_grad(z_val):
            return soft_max_obj_and_grad(z_val, tau)
        
        try:
            result = minimize(func_and_grad, z, jac=True, method='L-BFGS-B',
                              options={'maxiter': max_iter, 'ftol': 1e-14, 'gtol': 1e-8})
            z_new = result.x
        except Exception:
            continue
        
        a_new = softmax(z_new)
        a_new = np.maximum(a_new, 0)
        s_new = np.sum(a_new)
        if s_new < 1e-10:
            continue
        a_new = a_new / s_new
        obj = compute_obj(a_new)
        
        if obj < best_obj:
            best_obj = obj
            best_a = a_new.copy()
            z = z_new
    
    return best_a, best_obj

def subgradient_optimize(a, max_iters=2000, time_budget=20.0, tau=10.0):
    """Subgradient optimization with log-sum-exp smoothing."""
    n = len(a)
    a = a.copy().astype(np.float64)
    s = np.sum(a)
    if s < 1e-10:
        a = np.ones(n) / n
    else:
        a = a / s
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    step = 0.05
    stall = 0
    
    for it in range(max_iters):
        if time.monotonic() - start > time_budget:
            break
        
        conv = compute_conv(a)
        max_conv = float(np.max(conv))
        shifted = np.clip(tau * (conv - max_conv), -300, 300)
        weights = np.exp(shifted)
        w_sum = np.sum(weights)
        if w_sum < 1e-300:
            idx = int(np.argmax(np.log(weights + 1e-300)))
            weights = np.zeros(len(conv))
            weights[idx] = 1.0
        else:
            weights /= w_sum
        
        w_rev = weights[::-1]
        conv_fw = fft_convolve(a, w_rev)
        grad_a = 2.0 * conv_fw[n - 1:2 * n - 1][::-1]
        
        # Project gradient onto sum=0 hyperplane
        grad_centered = grad_a - np.mean(grad_a)
        
        a_new = a - step * grad_centered
        a_new = np.maximum(a_new, 0)
        s_new = np.sum(a_new)
        if s_new > 1e-10:
            a_new = a_new / s_new
        else:
            a_new = a.copy()
        
        a = a_new
        obj = compute_obj(a)
        
        if obj < best_obj - 1e-12:
            best_obj = obj
            best_a = a.copy()
            stall = 0
        else:
            stall += 1
            if stall > 200:
                step *= 0.9
                if step < 1e-12:
                    break
                rng = np.random.default_rng(it * 31 + 7)
                a = a + rng.normal(0, step, n)
                a = np.maximum(a, 0)
                s = np.sum(a)
                if s > 1e-10:
                    a = a / s
                stall = 0
        
        # Decrease step slowly
        step *= 0.9995
    
    return best_a

def random_init(n, seed):
    """Generate diverse random initializations."""
    rng = np.random.default_rng(seed)
    k = np.arange(n)
    choice = rng.integers(0, 12)
    
    if choice == 0:
        # Pure chirp
        f = rng.uniform(0.5, 5.0)
        a = 1.0 + 0.5 * np.cos(2 * np.pi * f * k**2 / (2*n))
    elif choice == 1:
        # Multiple chirps
        a = np.ones(n)
        for _ in range(rng.integers(2, 4)):
            f = rng.uniform(0.3, 8.0)
            a += 0.3 * np.cos(2 * np.pi * f * k**2 / (2*n) + rng.uniform(0, 2*np.pi))
    elif choice == 2:
        # Frank-Zadoff-Chu inspired
        root = rng.integers(1, n, dtype=np.float64)
        a = 1.0 + 0.3 * np.cos(2 * np.pi * root * k * (k+1) / (2*n))
    elif choice == 3:
        # Gaussian mixture
        a = np.ones(n) * 0.3
        for _ in range(rng.integers(2, 6)):
            center = rng.uniform(0, n)
            width = rng.uniform(n*0.05, n*0.3)
            a += np.exp(-0.5 * ((k - center) / width) ** 2)
    elif choice == 4:
        # Piecewise sinusoidal
        a = np.ones(n)
        nseg = rng.integers(2, 5)
        for s in range(nseg):
            start = s * n // nseg
            end = (s+1) * n // nseg
            f = rng.uniform(0.5, 5.0)
            a[start:end] = 1.0 + rng.uniform(0.2, 0.6) * np.sin(2*np.pi*f*(k[start:end]-start)/(end-start))
    elif choice == 5:
        # Edge-emphasized
        center = n/2
        a = 1.0 + 0.5 * np.cos(np.pi * (k - center) / center)
    elif choice == 6:
        # Random with smoothing
        a = rng.random(n)
        win = max(3, n // 15)
        kernel = np.ones(win) / win
        a = np.convolve(a, kernel, mode='same')
    elif choice == 7:
        # Anti-symmetric around center
        half = n // 2
        left = rng.uniform(0.5, 2.0, half)
        right = rng.uniform(0.5, 2.0, n - half)
        a = np.concatenate([left, right])
        # Make it "anti-symmetric": large at edges, small at center
        a = 1.0 + 0.5 * (1 - np.abs(k - center) / center)
    elif choice == 8:
        # Modulated
        f_c = rng.uniform(1.0, 4.0)
        f_m = rng.uniform(0.5, 3.0)
        a = 1.0 + 0.4 * np.cos(2*np.pi*f_c*k/n) * np.cos(2*np.pi*f_m*k/n)
    elif choice == 9:
        # Spline-like
        n_knots = rng.integers(3, 8)
        knots = np.sort(rng.uniform(0, 1, n_knots))
        values = rng.uniform(0.3, 1.5, n_knots)
        a = np.interp(k/n, knots, values)
    elif choice == 10:
        # Exponential decay from edges
        center = n/2
        a = np.exp(-0.5 * ((k - center) / (n*0.4))**2) + 0.3
    else:
        # Random Fourier
        a = np.ones(n)
        for _ in range(rng.integers(2, 5)):
            f = rng.uniform(0.5, 10.0)
            amp = rng.uniform(0.1, 0.4)
            phase = rng.uniform(0, 2*np.pi)
            a += amp * np.cos(2*np.pi*f*k/n + phase)
    
    a = np.maximum(a, 0.01)
    s = np.sum(a)
    return a / s

def resample_sequence(a, new_n):
    n = len(a)
    if n == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    return np.interp(x_new, x_old, a)

def coordinate_descent(a, max_iters=5000, time_budget=30.0):
    """Coordinate descent: try moving mass from peak-contributing positions."""
    n = len(a)
    a = a.copy().astype(np.float64)
    s = np.sum(a)
    a = a / s
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    
    conv = compute_conv(a)
    peak_idx = int(np.argmax(conv))
    
    for _ in range(max_iters):
        if time.monotonic() - start > time_budget:
            break
        
        conv = compute_conv(a)
        peak_idx = int(np.argmax(conv))
        peak_val = conv[peak_idx]
        
        # Find which coordinate contributes most to the peak
        # conv[peak_idx] = sum_i a[i] * a[peak_idx - i] (with bounds)
        i_low = max(0, peak_idx - n + 1)
        i_high = min(peak_idx, n - 1)
        
        # Contribution of a[i] to conv[peak_idx] is a[i] * a[peak_idx - i]
        contributions = np.zeros(n)
        for i in range(i_low, i_high + 1):
            j = peak_idx - i
            if 0 <= j < n:
                contributions[i] = a[i] * a[j]
        
        # Move a small amount from the highest contributor to a random other position
        src = int(np.argmax(contributions))
        if a[src] < 1e-8:
            continue
        
        # Move delta from src to a random position
        rng = np.random.default_rng(int(time.monotonic() * 1e6) % 2**31)
        delta = a[src] * rng.uniform(0.01, 0.05)
        dst = int(rng.integers(0, n))
        if dst == src:
            continue
        
        a_new = a.copy()
        a_new[src] -= delta
        a_new[dst] += delta
        a_new = np.maximum(a_new, 0)
        s_new = np.sum(a_new)
        a_new = a_new / s_new
        
        obj_new = compute_obj(a_new)
        if obj_new < best_obj - 1e-14:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new
    
    return best_a

def try_parent():
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64)
        parent = np.clip(parent, 0, 1000.0)

    return None, np.inf
