def fast_conv(a):
    """Full self-convolution via FFT."""
    n = len(a)
    s = 1
    while s < 2 * n - 1:
        s <<= 1
    fa = np.fft.rfft(a, s)
    return np.fft.irfft(fa * fa, s)[:2 * n - 1]


def compute_J(a):
    """Compute J(a) = 2*n*max(conv(a,a))/(sum(a))^2."""
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))


def smooth_obj_and_grad(a, beta):
    """
    Compute F(a) = 2n * softmax_beta(conv(a,a)) / (sum a)^2 and its gradient.
    """
    n = len(a)
    b = fast_conv(a)
    b_max = float(b.max())
    shifted = beta * (b - b_max)
    shifted = np.clip(shifted, -500, 500)
    exp_shifted = np.exp(shifted)
    Z = float(exp_shifted.sum())
    if Z < 1e-300:
        p = np.zeros(2 * n - 1)
        p[np.argmax(b)] = 1.0
    else:
        p = exp_shifted / Z
    
    smooth_max = b_max + np.log(max(Z, 1e-300)) / beta
    s = float(a.sum())
    F_val = 2.0 * n * smooth_max / (s * s)
    
    # Gradient: dF/da_i = 2n/(s^2) * 2*sum_k p_k * a_{k-i} - 2*F*a_i/s
    # conv(p, a[::-1])[n-1:2n-1] gives sum_k p_k * a_{k-i} for i=0..n-1
    S = np.convolve(p, a[::-1], mode='full')[n - 1:2 * n - 1]
    grad_smooth = 2.0 * S
    grad = 2.0 * n * grad_smooth / (s * s) - 2.0 * F_val * a / s
    
    return F_val, grad

def get_partial_A(a, K):

            cols.append(int(j))
            vals.append(float(a[k - j]))

def line_search_partial(a, g, n_evals=12):
    """Faster line search with fewer evaluations."""
    best_t = 0.0
    best_J = compute_J(a)
    a_1 = g.copy()
    J_at_1 = compute_J(a_1)
    if J_at_1 < best_J:
        best_J = J_at_1
        best_t = 1.0
    
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    lo = 0.0
    hi = 1.0
    
    for _ in range(n_evals):
        t1 = hi - gr * (hi - lo)
        t2 = lo + gr * (hi - lo)
        a1 = (1.0 - t1) * a + t1 * g
        a2 = (1.0 - t2) * a + t2 * g
        J1 = compute_J(a1)
        J2 = compute_J(a2)
        if J1 < J2:
            hi = t2
        else:
            lo = t1
        t_mid = (lo + hi) / 2.0
        a_mid = (1.0 - t_mid) * a + t_mid * g
        J_mid = compute_J(a_mid)
        if J_mid < best_J:
            best_J = J_mid
            best_t = t_mid
    
    return (1.0 - best_t) * a + best_t * g, best_J, best_t

def resample_seq(src, dest_n):
    """Resample 1D sequence to new length via linear interpolation."""
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    result = np.interp(t_new, t_old, src)
    return np.maximum(result, 0.0)
