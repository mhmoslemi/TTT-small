"""
Hybrid optimization for minimizing J(a) = 2*n*max(conv(a,a))/(sum(a))^2:
1. Multi-length smooth gradient descent with FFT-based convolution
2. Pairwise coordinate descent for precise local optimization
3. LP mirror descent for final refinement
4. Multi-start with diverse initializations and length exploration
"""

import numpy as np
from scipy.optimize import linprog, minimize
from scipy.sparse import csr_matrix
import time

def fast_conv(a):
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]

def compute_J(a):
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))

def smooth_max_and_grad(a, tau):
    n = len(a)
    b = fast_conv(a)
    b_max = b.max()
    exp_b = np.exp(tau * (b - b_max))
    p = exp_b / exp_b.sum()
    
    grad = np.zeros(n)
    active = np.where(p > 1e-12)[0]
    for k in active:
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        grad[lo:hi + 1] += 2.0 * p[k] * a[k - hi:k - lo + 1][::-1]
    
    return grad, p, b, b_max

def run_smooth_descent(a, deadline, max_iters=5000, tau_start=1.0, tau_end=200.0):
    a = np.maximum(a.astype(np.float64), 1e-8)

def pairwise_cd(a, max_iters=2000, deadline=None):
    """Pairwise coordinate descent: swap mass between pairs to reduce peak."""
    a = np.maximum(a.astype(np.float64), 0.0)
    a = a / a.sum()
    n = len(a)
    best_a = a.copy()
    best_M = fast_conv(a).max()
    
    for it in range(max_iters):
        if deadline is not None and time.monotonic() > deadline:
            break
        
        b = fast_conv(a)
        M = b.max()
        if M < best_M * (1 - 1e-10):
            best_M = M
            best_a = a.copy()
        
        # Find the peak location
        peak_k = int(np.argmax(b))
        
        # Find the pair (i, j) that best reduces the peak
        # Moving mass from peak-contributing positions to others
        # The peak is at conv[peak_k] = sum_j a[j]*a[peak_k-j]
        # To reduce it, move mass from positions that contribute to the peak
        # to positions that don't.
        
        # Simple heuristic: find the position i that most contributes to the peak
        # and move mass to a position j that least contributes.
        
        lo = max(0, peak_k - n + 1)
        hi = min(n - 1, peak_k)
        
        # Contribution of a[i] to conv[peak_k] is roughly 2*a[i]*a[peak_k-i]
        # (or a[i]^2 if 2i = peak_k)
        
        # Find i in [lo, hi] with largest contribution
        contrib = np.zeros(hi - lo + 1)
        for idx, i in enumerate(range(lo, hi + 1)):
            j = peak_k - i
            if 0 <= j < n and j != i:
                contrib[idx] = 2 * a[i] * a[j]
            elif j == i:
                contrib[idx] = a[i] ** 2
        
        # Find position to move mass to (outside [lo, hi] or with low contribution)
        # Try moving from max contrib to min contrib
        if hi - lo + 1 >= 2:
            max_idx = lo + int(np.argmax(contrib))
            min_idx = lo + int(np.argmin(contrib))
            
            if max_idx != min_idx and a[max_idx] > 1e-6:
                # Try a small move
                delta = 0.01 * a[max_idx]
                a_trial = a.copy()
                a_trial[max_idx] -= delta
                a_trial[min_idx] += delta
                if a_trial[max_idx] < 0:
                    a_trial[max_idx] = 0
                    a_trial[min_idx] += a[max_idx]
                
                M_trial = fast_conv(a_trial).max()
                if M_trial < M:
                    a = a_trial
                else:
                    # Try smaller delta
                    delta = 0.001 * a[max_idx]
                    a_trial = a.copy()
                    a_trial[max_idx] -= delta
                    a_trial[min_idx] += delta
                    if a_trial[max_idx] < 0:
                        a_trial[max_idx] = 0
                    M_trial = fast_conv(a_trial).max()
                    if M_trial < M:
                        a = a_trial
    
    return best_a

def build_A_sparse(n, a):
    rows, cols, vals = [], [], []
    for k in range(2 * n - 1):
        for j in range(max(0, k - n + 1), min(n - 1, k) + 1):
            rows.append(k)
            cols.append(j)
            vals.append(a[k - j])
    return csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))

def lp_best_response(a, M):
    n = len(a)
    A = build_A_sparse(n, a)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    try:
        r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                    options={'time_limit': 20})
        if r.success and r.x is not None:
            g = np.maximum(r.x, 0)
            if g.sum() > 1e-15:
                return g
    except:
        pass
    return None

def run_lp_mirror(a, max_iters=200, deadline=None):

def resample(src, n):
    src = np.asarray(src, dtype=np.float64)
    if src.size == n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)

        frac = it / max_iters
        tau = tau_start * (tau_end / tau_start) ** (frac ** 0.6)
        lr = lr_init * (lr_end / lr_init) ** frac

        grad, p, b, b_max = smooth_max_grad(y, tau)
        if b_max < best_M * (1 - 1e-10):
            best_M = b_max
            best_a = a.copy()

        t_k = (1.0 + np.sqrt(1.0 + 4.0 * beta * beta)) / 2.0
        a_new = np.maximum(y - lr * (grad + beta * (y - a)), 0.0)
        s = a_new.sum()
        if s < 1e-10:
            a_new = a.copy()
        else:
            a_new /= s
        y = a_new + ((t_k - 1.0) / (t_k + 1.0)) * (a_new - a)
        y = np.maximum(y, 0.0)
        sy = y.sum()
        if sy < 1e-10:
            y = a_new
        else:
            y /= sy
        a = a_new

    return best_a


def run_subgradient_avg(a, deadline, max_iters=3000, lr_const=0.3):
    a = np.maximum(a.astype(np.float64), 1e-8)
    a = a / a.sum()
    n = len(a)
    best_a = a.copy()
    best_M = fast_conv(a).max()
    running = np.zeros(n)
    count = 0

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        b = fast_conv(a)
        M = b.max()
        if M < best_M * (1 - 1e-10):
            best_M = M
            best_a = a.copy()

        tol = 1e-6 * M
        active = np.where(b >= M - tol)[0]
        grad = np.zeros(n)
        for k in active:
            lo = max(0, k - n + 1)
            hi = min(n - 1, k)
            grad[lo:hi + 1] += 2.0 * a[k - hi:k - lo + 1][::-1]
        gn = np.linalg.norm(grad)
        if gn > 1e-15:
            grad /= gn

        lr = lr_const / np.sqrt(it + 1)
        a_new = np.maximum(a - lr * grad, 0.0)
        s = a_new.sum()
        if s < 1e-10:
            a_new = a.copy()
        else:
            a_new /= s
        running += a_new
        count += 1
        a = a_new

    if count > 50:
        avg = running / count
        avg = np.maximum(avg, 0.0)
        s = avg.sum()
        if s > 1e-10:
            avg /= s
            M_avg = fast_conv(avg).max()
            if M_avg < best_M:
                best_a = avg
    return best_a


def build_A_sparse(n, a):
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        j_min = max(0, k - n + 1)
        j_max = min(n - 1, k)
        for j in range(j_min, j_max + 1):
            rows.append(k)
            cols.append(j)
            vals.append(a[k - j])
    return csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))


def lp_best_response(a, M, time_limit=15):
    n = len(a)
    A = build_A_sparse(n, a)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    try:
        r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                    options={'time_limit': time_limit, 'presolve': True})
        if r.success and r.x is not None and r.x.sum() > 1e-15:
            return np.maximum(r.x, 0)
    except Exception:
        pass
    return None


def run_lp_mirror(a, deadline, max_iters=300):
    a = np.maximum(a.astype(np.float64), 1e-8)
    a = a / a.sum()
    n = len(a)
    best_a = a.copy()
    best_M = fast_conv(a).max()
    no_imp = 0

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        b = fast_conv(a)
        M = float(b.max())
        g = lp_best_response(a, M)
        if g is None:
            break
        gs = g.sum()
        if gs < 1e-10:
            break
        g /= gs

        improved = False
        for t in [0.5, 0.35, 0.2, 0.1, 0.05, 0.02, 0.01]:
            a_t = np.maximum((1.0 - t) * a + t * g, 0.0)
            s = a_t.sum()
            if s < 1e-10:
                continue
            a_t /= s
            M_t = fast_conv(a_t).max()
            if M_t < best_M - 1e-14:
                a = a_t
                best_M = M_t
                best_a = a.copy()
                improved = True
                break
        if improved:
            no_imp = 0
        else:
            no_imp += 1
            if no_imp > 20:
                break
    return best_a


def resample_seq(src, n):
    src = np.asarray(src, dtype=np.float64)
    if src.size == n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)


def make_inits(n, rng, parent):
    inits = []
    t = np.linspace(0, 1, n, endpoint=False)
    if parent is not None:
        p = resample_seq(parent, n)
        inits.append(p.copy())
        inits.append(np.maximum(p + rng.normal(0, 0.015, n), 0.005))
        inits.append(np.maximum(p + rng.normal(0, 0.04, n), 0.005))
        inits.append(np.maximum(p * (1 + 0.08 * rng.normal(0, 1, n)), 0.005))
    inits.append(np.ones(n))
    for alpha in [0.7, 1.0, 1.4, 2.0]:
        inits.append(np.maximum(0.3 + 0.7 * (t**alpha + (1 - t)**alpha) / 2, 0.01))
    for f in [1, 2, 3]:
        inits.append(np.maximum(0.6 + 0.25 * np.cos(2 * np.pi * f * t), 0.01))
    inits.append(np.exp(-3 * (t - 0.5)**2) + 0.1)
    inits.append(np.exp(-6*(t-0.25)**2) + np.exp(-6*(t-0.75)**2) + 0.05)
    k = max(3, n // 40)
    kern = np.ones(k) / k
    inits.append(np.maximum(np.convolve(rng.uniform(0, 1, n), kern, 'same') + 0.1, 0.01))
    half_n = n // 2 + 1
    half = np.maximum(rng.uniform(0.3, 1.0, half_n), 0.01)
    full = np.concatenate([half, half[-2::-1]])[:n]
    inits.append(full)
    return inits


class BestTracker:
    def __init__(self):
        self.seq = None
        self.score = float('inf')
        self.t0 = time.monotonic()

    def update(self, a):
        if a is None:
            return
        a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
        if a.size == 0 or a.sum() < 0.01:
            return
        s = float(evaluate_sequence(a.tolist()))
        if np.isfinite(s) and s < self.score:
            self.score = s
            self.seq = a.copy()
            el = time.monotonic() - self.t0
            print(f"  ** J={s:.8f} n={a.size} t={el:.0f}s", flush=True)

    def report(self, label):
        if self.seq is not None:
            print(f"  {label}: J={self.score:.8f} n={self.seq.size}", flush=True)


def propose_candidate(seed=42, budget_s=1500, **kw):
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + budget_s - 25.0
    best = BestTracker()

    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        if p.ndim == 1 and p.size > 0 and np.all(np.isfinite(p)):
            parent = np.maximum(p, 0.0)
            best.update(parent)
            print(f"Parent: n={parent.size}, J={best.score:.6f}", flush=True)
    except (NameError, AttributeError, Exception):
        pass

    if parent is None:
        parent = np.ones(256)
        best.update(parent)

    # Determine lengths to explore
    base_n = parent.size
    candidate_lengths = set()
    for mult in [0.75, 0.875, 1.0, 1.125, 1.25, 1.5, 1.75, 2.0, 2.5, 3.0]:
        n_try = int(base_n * mult)
        n_try = max(128, min(2048, (n_try // 16) * 16))
        candidate_lengths.add(n_try)
    # Add some specific values
    for n_try in [448, 512, 576, 640, 768, 896, 1024, 1152, 1280]:
        if 128 <= n_try <= 2048:
            candidate_lengths.add(n_try)
    lengths = sorted(candidate_lengths)
    print(f"Lengths: {lengths}", flush=True)

    # === PHASE 1: Smooth gradient descent across lengths ===
    p1_budget = 380.0
    p1_start = time.monotonic()
    print(f"\n{'='*50}", flush=True)
    print(f"PHASE 1: Smooth Gradient Descent ({p1_budget:.0f}s)", flush=True)
    print(f"{'='*50}", flush=True)

    # Weight shorter lengths more heavily (they're faster per iteration)
    weights = np.array([1.0 / np.sqrt(n) for n in lengths])
    weights /= weights.sum()

    best_per_length = {}
    for i, n in enumerate(lengths):
        if time.monotonic() > deadline - 250:
            break
        allot = p1_budget * weights[i]
        allot = max(10, min(80, allot))
        dl = min(time.monotonic() + allot, deadline - 200)
        if time.monotonic() >= dl:
            continue

        t0 = time.monotonic()
        inits = make_inits(n, rng, best.seq if best.seq is not None else parent)
        best_n = None
        best_M = float('inf')

        n_iters = max(80, int(4000 * (256.0 / n)))
        for a0 in inits:
            if time.monotonic() > dl:
                break
            a0 = np.maximum(a0.astype(np.float64), 1e-6)
            a0 /= a0.sum()
            a = run_smooth_nesterov(a0, dl, max_iters=n_iters,
                                    tau_start=1.0, tau_end=150.0,
                                    lr_init=0.35, lr_end=0.003)
            M = fast_conv(a).max()
            if M < best_M:
                best_M = M
                best_n = a.copy()

        if best_n is not None:
            J = compute_J(best_n)
            print(f"  n={n:4d}: M={best_M:.7f} J={J:.6f} ({time.monotonic()-t0:.1f}s)", flush=True)
            best.update(best_n)
            best_per_length[n] = best_n.copy()

    best.report("Phase 1")
    print(f"Phase 1 time: {time.monotonic()-p1_start:.1f}s", flush=True)

    # === PHASE 2: Extended optimization on top candidates ===
    p2_start = time.monotonic()
    print(f"\n{'='*50}", flush=True)
    print(f"PHASE 2: Extended Polish", flush=True)
    print(f"{'='*50}", flush=True)

    if best.seq is not None:
        # Subgradient with averaging
        a = run_subgradient_avg(best.seq, min(time.monotonic() + 60, deadline - 150),
                                max_iters=2000, lr_const=0.25)
        best.update(a)

        # More smooth gradient from best
        a = run_smooth_nesterov(best.seq, min(time.monotonic() + 60, deadline - 150),
                                max_iters=3000, tau_start=3.0, tau_end=200.0,
                                lr_init=0.2, lr_end=0.002)
        best.update(a)

    best.report("Phase 2")
    print(f"Phase 2 time: {time.monotonic()-p2_start:.1f}s", flush=True)

    # === PHASE 3: LP Mirror Descent ===
    p3_start = time.monotonic()
    print(f"\n{'='*50}", flush=True)
    print(f"PHASE 3: LP Mirror Descent", flush=True)
    print(f"{'='*50}", flush=True)

    if best.seq is not None:
        n = best.seq.size
        # Estimate per-LP time based on n
        est_lp_ms = 4.0 * (n / 128.0)**2
        print(f"  n={n}, est LP time={est_lp_ms:.0f}ms", flush=True)

        # Do as many LP iterations as possible within time budget
        lp_budget = min(250.0, max(30.0, (deadline - 80) - time.monotonic()))
        max_iters = max(20, int(lp_budget / max(0.01, est_lp_ms / 1000.0)))
        print(f"  LP budget={lp_budget:.0f}s, max_iters={max_iters}", flush=True)

        a = run_lp_mirror(best.seq, min(time.monotonic() + lp_budget, deadline - 60),
                          max_iters=max_iters)
        best.update(a)

        # Second LP pass if time remains
        if time.monotonic() < deadline - 40:
            a2 = run_lp_mirror(a, deadline - 40, max_iters=100)
            best.update(a2)

    best.report("Phase 3")
    print(f"Phase 3 time: {time.monotonic()-p3_start:.1f}s", flush=True)

    # === PHASE 4: Multi-start with structured perturbations ===
    p4_start = time.monotonic()
    print(f"\n{'='*50}", flush=True)
    print(f"PHASE 4: Multi-start Perturbations", flush=True)
    print(f"{'='*50}", flush=True)

    if best.seq is not None and time.monotonic() < deadline - 40:
        n = best.seq.size
        perturb_schemes = [
            (0.02, 'gauss'),
            (0.01, 'gauss'),
            (0.005, 'gauss'),
            (0.03, 'lowfreq'),
            (0.02, 'lowfreq'),
        ]
        for scale, ptype in perturb_schemes:
            if time.monotonic() > deadline - 30:
                break
            for _ in range(2):
                if time.monotonic() > deadline - 25:
                    break
                if ptype == 'gauss':
                    noise = rng.normal(0, scale * 0.01, n)
                else:
                    # Low-frequency perturbation
                    freqs = rng.integers(1, 5, 3)
                    phases = rng.uniform(0, 2 * np.pi, 3)
                    t_arr = np.linspace(0, 1, n)
                    noise = sum(scale * 0.01 * np.cos(2 * np.pi * f * t_arr + ph)
                                for f, ph in zip(freqs, phases))
                a = best.seq + noise
                a = np.maximum(a, 1e-4)
                a /= a.sum()
                a = run_smooth_nesterov(a, min(time.monotonic() + 25, deadline - 25),
                                        max_iters=400, tau_start=5.0, tau_end=100.0,
                                        lr_init=0.15, lr_end=0.01)
                best.update(a)
                # Quick LP if time
                if time.monotonic() < deadline - 15:
                    a2 = run_lp_mirror(a, deadline - 15, max_iters=20)
                    best.update(a2)

    best.report("Phase 4")
    print(f"Phase 4 time: {time.monotonic()-p4_start:.1f}s", flush=True)

    # === PHASE 5: Try slightly different lengths around best ===
    if best.seq is not None and time.monotonic() < deadline - 20:
        p5_start = time.monotonic()
        print(f"\n{'='*50}", flush=True)
        print(f"PHASE 5: Length Refinement", flush=True)
        print(f"{'='*50}", flush=True)
        n_best = best.seq.size
        for delta in [-32, -16, 16, 32, 64]:
            if time.monotonic() > deadline - 15:
                break
            n_try = n_best + delta
            if n_try < 128 or n_try > 2048:
                continue
            a_ext = resample_seq(best.seq, n_try)
            a_ext = np.maximum(a_ext, 1e-6)
            a_ext /= a_ext.sum()
            a_ext = run_smooth_nesterov(a_ext, min(time.monotonic() + 20, deadline - 10),
                                        max_iters=300, tau_start=3, tau_end=80,
                                        lr_init=0.2, lr_end=0.02)
            best.update(a_ext)
        best.report("Phase 5")
        print(f"Phase 5 time: {time.monotonic()-p5_start:.1f}s", flush=True)

    # === Final LP polish ===
    if best.seq is not None and time.monotonic() < deadline - 3:
        a = run_lp_mirror(best.seq, deadline, max_iters=50)
        best.update(a)

    # Fallback
    if best.seq is None:
        best.seq = np.ones(128)
        best.score = compute_J(best.seq)

    print(f"\n{'='*50}", flush=True)
    print(f"FINAL: J={best.score:.8f} n={best.seq.size} time={time.monotonic()-best.t0:.0f}s", flush=True)
    print(f"{'='*50}", flush=True)
    return [float(x) for x in best.seq]
