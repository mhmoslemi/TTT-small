to LP iterations.

import numpy as np
from scipy.optimize import linprog, minimize
from scipy.sparse import csr_matrix
import time


def fast_conv(a):
    """Full self-convolution via FFT."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]


def compute_J(a):
    """Compute J(a) = 2*n*max(conv(a,a))/(sum(a))^2."""
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))


def compute_subgradient(a, active_indices):
    """Compute subgradient of max_k c[k] at point a for active set."""
    n = len(a)
    grad = np.zeros(n)
    for k in active_indices:
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        for j in range(lo, hi + 1):
            grad[j] += 2.0 * a[k - j]
    return grad


def lp_best_response(a, M, active_indices=None):
    """Solve restricted LP: max sum(g) s.t. A_g@g <= M*1, g >= 0.
    Only use constraints from active_indices if provided."""
    n = len(a)
    if active_indices is None:
        active_indices = list(range(2 * n - 1))
    
    rows = []
    cols = []
    vals = []
    row_index = []
    for idx, k in enumerate(active_indices):
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        for j in range(lo, hi + 1):
            rows.append(idx)
            cols.append(j)
            vals.append(a[k - j])
        row_index.append(idx)
    
    n_cons = len(active_indices)
    A = csr_matrix((vals, (rows, cols)), shape=(n_cons, n))
    c = -np.ones(n)
    b_ub = np.full(n_cons, M)
    bounds = [(0, None)] * n
    try:
        r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                    options={'presolve': True, 'time_limit': 30})
        if r.success and r.x is not None:
            g = np.maximum(r.x, 0)
            if g.sum() > 1e-15:
                return g
    except Exception:
        pass
    return None

        if gnorm < 1e-15:
            break
        
        # Step size with decay
        step = t / (it + 1)
        
        # Gradient step
        y_new = y - step * grad / gnorm
        y_new = np.maximum(y_new, 0)
        s = y_new.sum()
        if s < 1e-10:
            break
        y_new = y_new / s
        
        # Nesterov momentum
        v = momentum * v + (1 - momentum) * (y_new - y)
        y = y_new + v
        y = np.maximum(y, 0)
        s = y.sum()
        if s < 1e-10:
            y = y_new
        else:
            y = y / s
        
        # Track best
        M_y = fast_conv(y).max()
        if M_y < best_M:
            best_M = M_y
            best_a = y.copy()
        
        # Decay parameters
        if it > 100:
            momentum = min(0.999, momentum + 0.001)
        if it % 200 == 0 and it > 0:
            t *= 0.7
    
    return best_a, best_M


def lp_mirror_descent_fast(a, max_iters, deadline, use_active_set=True):
    """LP mirror descent with active-set reduction."""
    a = np.maximum(a.astype(np.float64), 1e-15)
    a = a / a.sum()
    current_J = compute_J(a)
    best_J = current_J
    best_a = a.copy()
    no_improve = 0
    
    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        
        b = fast_conv(a)
        M = float(b.max())
        
        # Active set: indices within 5% of max
        if use_active_set:
            tol = 0.05 * M
            active = list(np.where(b >= M - tol)[0])
            # Always include argmax
            argmax_k = int(np.argmax(b))
            if argmax_k not in active:
                active.append(argmax_k)
        else:
            active = None
        
        g = lp_best_response(a, M, active)
        if g is None:
            # Try full LP
            g = lp_best_response(a, M, None)
            if g is None:
                break
        
        g = np.maximum(g, 0)
        g_sum = g.sum()
        if g_sum < 1e-10:
            break
        g = g / g_sum
        
        # Adaptive step size with backtracking
        base_t = 0.2 * np.exp(-0.2 * it / 50.0) + 0.03
        t = min(1.0, base_t)
        
        best_trial_J = current_J
        best_trial_a = None
        
        for ls in range(20):
            a_trial = (1.0 - t) * a + t * g
            a_trial = np.maximum(a_trial, 0.0)
            s = a_trial.sum()
            if s < 1e-10:
                t *= 0.5
                continue
            a_trial = a_trial / s
            J_trial = compute_J(a_trial)
            if J_trial < best_trial_J - 1e-14:
                best_trial_J = J_trial
                best_trial_a = a_trial.copy()
                break
            t *= 0.6
            if t < 0.001:
                break
        
        if best_trial_a is not None:
            a = best_trial_a
            current_J = best_trial_J
            if current_J < best_J:
                best_J = current_J
                best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1
            if no_improve > 30:
                break
    
    return best_a, best_J


def lp_multiplicative(a, max_iters, deadline):
    """LP with multiplicative (geometric) update."""
    a = np.maximum(a.astype(np.float64), 1e-15)
    a = a / a.sum()
    current_J = compute_J(a)
    best_J = current_J
    best_a = a.copy()
    no_improve = 0
    
    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        
        b = fast_conv(a)
        M = float(b.max())
        tol = 0.05 * M
        active = list(np.where(b >= M - tol)[0])
        argmax_k = int(np.argmax(b))
        if argmax_k not in active:
            active.append(argmax_k)
        
        g = lp_best_response(a, M, active)
        if g is None:
            break
        
        g = np.maximum(g, 1e-15)
        g_sum = g.sum()
        if g_sum < 1e-10:
            break
        g = g / g_sum
        
        base_t = 0.15 * np.exp(-0.2 * it / 50.0) + 0.02
        t = min(1.0, base_t)
        
        best_trial_J = current_J
        best_trial_a = None
        
        for ls in range(20):
            ratio = g / np.maximum(a, 1e-15)
            a_trial = a * np.power(ratio, t)
            a_trial = np.maximum(a_trial, 0.0)
            s = a_trial.sum()
            if s < 1e-10:
                t *= 0.5
                continue
            a_trial = a_trial / s
            J_trial = compute_J(a_trial)
            if J_trial < best_trial_J - 1e-14:
                best_trial_J = J_trial
                best_trial_a = a_trial.copy()
                break
            t *= 0.6
            if t < 0.001:
                break
        
        if best_trial_a is not None:
            a = best_trial_a
            current_J = best_trial_J
            if current_J < best_J:
                best_J = current_J
                best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1
            if no_improve > 30:
                break
    
    return best_a, best_J


def lbb_objective(x):
    """Objective for L-BFGS-B: max of self-convolution (scale invariant)."""
    ax = np.maximum(x, 0.0)
    s = ax.sum()
    if s < 1e-8:
        return 1e10
    ax = ax / s
    b = fast_conv(ax)
    return float(b.max())


def resample_seq(src, dest_n):
    """Resample 1D sequence to new length."""
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)


def generate_structured_inits(n, rng):
    """Generate diverse structured initial sequences."""
    inits = []
    t = np.arange(n, dtype=np.float64) / n
    
    inits.append(np.ones(n))
    inits.append(0.5 * (1 + np.cos(2 * np.pi * t)))
    inits.append(np.maximum(0.42 - 0.5 * np.cos(2 * np.pi * t) + 0.08 * np.cos(4 * np.pi * t), 0.01))
    inits.append(np.exp(-4 * (t - 0.5) ** 2) + 0.1)
    inits.append(np.exp(-8 * (t - 0.3) ** 2) + np.exp(-8 * (t - 0.7) ** 2) + 0.05)
    
    for f in [1, 2, 3, 5, 7, 11, 13]:
        inits.append(np.maximum(1.0 + 0.3 * np.cos(2 * np.pi * f * t + rng.uniform(0, 2 * np.pi)), 0.01))
    
    # Sum of two sinusoids
    for f1 in [2, 3, 5]:
        for f2 in [7, 11, 13]:
            phase = rng.uniform(0, 2 * np.pi)
            inits.append(np.maximum(1.0 + 0.25 * np.cos(2 * np.pi * f1 * t) + 0.15 * np.cos(2 * np.pi * f2 * t + phase), 0.01))
    
    # Triangle wave
    inits.append(1.0 - 0.5 * np.abs(2 * t - 1))
    
    result = []
    for a in inits:
        a = np.maximum(a, 0.001)
        s = a.sum()
        if s > 1e-6:
            result.append(a / s)
    return result


class ScoreTracker:
    def __init__(self):
        self.best = None
        self.best_score = float('inf')
        self.t_start = time.monotonic()

    def update(self, a):
        a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
        if a.size == 0 or a.sum() < 0.01:
            return
        score = float(evaluate_sequence(a.tolist()))
        if np.isfinite(score) and score < self.best_score:
            self.best = a.copy()
            self.best_score = score
            elapsed = time.monotonic() - self.t_start
            print(f"  [NEW BEST] J={score:.8f} n={a.size} t={elapsed:.0f}s", flush=True)

    def report(self, label=""):
        if self.best is not None:
            print(f"  {label} best: J={self.best_score:.8f} n={self.best.size}", flush=True)


def propose_candidate(seed=42, budget_s=1500):
    """
    Hybrid optimizer: Nesterov subgradient + LP mirror descent (active set) +
    multiplicative LP + structured perturbations + L-BFGS-B final polish.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + float(budget_s) - 20.0
    tracker = ScoreTracker()

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        if p.ndim == 1 and p.size > 0 and np.all(np.isfinite(p)):
            parent = np.maximum(p, 0.0)
            tracker.update(parent)
            print(f"Parent: n={parent.size}, J={tracker.best_score:.6f}", flush=True)
    except (NameError, AttributeError):
        pass

    if parent is None:
        parent = np.ones(640)
        tracker.update(parent)

    current_best = parent.copy()
    current_best_n = current_best.size
    current_best_J = compute_J(current_best)
    print(f"Initial J (compute): {current_best_J:.8f}", flush=True)

    # === Phase 1: Nesterov accelerated subgradient (fast, many iterations) ===
    t0 = time.monotonic()
    print(f"\n{'='*50}", flush=True)
    print(f"Phase 1: Nesterov subgradient (n={current_best_n})", flush=True)
    print(f"{'='*50}", flush=True)

    phase1_dl = min(deadline - 900, time.monotonic() + 180)
    a1, M1 = nesterov_subgradient(current_best, 50000, phase1_dl)
    J1 = compute_J(a1)
    tracker.update(a1)
    print(f"  J={J1:.8f} ({time.monotonic()-t0:.1f}s)", flush=True)

    if J1 < current_best_J:
        current_best = a1.copy()
        current_best_J = J1

    # === Phase 2: LP mirror descent with active set (main workhorse) ===
    if time.monotonic() < deadline - 700:
        t0 = time.monotonic()
        print(f"\n{'='*50}", flush=True)
        print(f"Phase 2: LP mirror descent active-set (n={current_best_n})", flush=True)
        print(f"{'='*50}", flush=True)

        phase2_dl = min(deadline - 600, time.monotonic() + 250)
        a2 = current_best.copy()
        a2 = np.maximum(a2, 1e-15)
        a2 = a2 / a2.sum()
        a2, J2 = lp_mirror_descent_fast(a2, 5000, phase2_dl, use_active_set=True)
        tracker.update(a2)
        print(f"  J={J2:.8f} ({time.monotonic()-t0:.1f}s)", flush=True)

        if J2 < current_best_J:
            current_best = a2.copy()
            current_best_J = J2

    # === Phase 3: LP multiplicative mode ===
    if time.monotonic() < deadline - 500:
        t0 = time.monotonic()
        print(f"\n{'='*50}", flush=True)
        print(f"Phase 3: LP multiplicative (n={current_best_n})", flush=True)
        print(f"{'='*50}", flush=True)

        phase3_dl = min(deadline - 400, time.monotonic() + 200)
        a3 = current_best.copy()
        a3 = np.maximum(a3, 1e-15)
        a3 = a3 / a3.sum()
        a3, J3 = lp_multiplicative(a3, 3000, phase3_dl)
        tracker.update(a3)
        print(f"  J={J3:.8f} ({time.monotonic()-t0:.1f}s)", flush=True)

        if J3 < current_best_J:
            current_best = a3.copy()
            current_best_J = J3

    # === Phase 4: Alternating Nesterov + LP with perturbations ===
    if time.monotonic() < deadline - 250:
        t0 = time.monotonic()
        print(f"\n{'='*50}", flush=True)
        print(f"Phase 4: Alternating + perturb (n={current_best_n})", flush=True)
        print(f"{'='*50}", flush=True)

        n = current_best_n
        t_arr = np.arange(n, dtype=np.float64) / n
        
        perturb_schedule = [
            ('nesterov', 0),
            ('lp', 0),
            ('perturb_cosine', 3.0),
            ('lp', 0),
            ('nesterov', 0),
            ('lp', 0),
            ('perturb_taper', 0.7),
            ('lp', 0),
            ('nesterov', 0),
            ('perturb_shift', 5),
            ('lp', 0),
        ]
        
        per_step_time = max(15, (deadline - 230 - time.monotonic()) / len(perturb_schedule))
        
        for step_type, param in perturb_schedule:
            if time.monotonic() > deadline - 230:
                break
            dl = time.monotonic() + min(per_step_time, 60)
            
            if step_type == 'nesterov':
                a_try = current_best.copy()
                a_try, M_try = nesterov_subgradient(a_try, 20000, dl)
                J_try = compute_J(a_try)
                tracker.update(a_try)
                if J_try < current_best_J:
                    current_best = a_try.copy()
                    current_best_J = J_try
                print(f"  [nesterov] J={J_try:.8f}", flush=True)
                
            elif step_type == 'lp':
                a_try = current_best.copy()
                a_try = np.maximum(a_try, 1e-15)
                a_try = a_try / a_try.sum()
                a_try, J_try = lp_mirror_descent_fast(a_try, 2000, dl, use_active_set=True)
                tracker.update(a_try)
                if J_try < current_best_J:
                    current_best = a_try.copy()
                    current_best_J = J_try
                print(f"  [lp] J={J_try:.8f}", flush=True)
                
            elif step_type == 'perturb_cosine':
                freq = param
                amp = rng.uniform(0.03, 0.08)
                phase = rng.uniform(0, 2 * np.pi)
                a_try = current_best * (1 + amp * np.cos(2 * np.pi * freq * t_arr + phase))
                a_try = np.maximum(a_try, 1e-10)
                a_try = a_try / a_try.sum()
                # Quick LP from perturbed
                a_try, J_try = lp_mirror_descent_fast(a_try, 500, dl, use_active_set=True)
                tracker.update(a_try)
                if J_try < current_best_J:
                    current_best = a_try.copy()
                    current_best_J = J_try
                print(f"  [perturb_cos f={freq}] J={J_try:.8f}", flush=True)
                
            elif step_type == 'perturb_taper':
                alpha = param
                window = np.ones(n)
                edge = max(1, int(n * (1 - alpha) / 2))
                for i in range(edge):
                    frac = i / edge
                    window[i] = frac
                    window[n - 1 - i] = frac
                a_try = current_best * window
                a_try = np.maximum(a_try, 1e-10)
                a_try = a_try / a_try.sum()
                a_try, J_try = lp_mirror_descent_fast(a_try, 500, dl, use_active_set=True)
                tracker.update(a_try)
                if J_try < current_best_J:
                    current_best = a_try.copy()
                    current_best_J = J_try
                print(f"  [perturb_taper a={alpha}] J={J_try:.8f}", flush=True)
                
            elif step_type == 'perturb_shift':
                s = param
                a_try = np.roll(current_best, s)
                a_try = np.maximum(a_try, 1e-10)
                a_try = a_try / a_try.sum()
                a_try, J_try = lp_mirror_descent_fast(a_try, 500, dl, use_active_set=True)
                tracker.update(a_try)
                if J_try < current_best_J:
                    current_best = a_try.copy()
                    current_best_J = J_try
                print(f"  [perturb_shift s={s}] J={J_try:.8f}", flush=True)
        
        print(f"Phase 4 done ({time.monotonic()-t0:.1f}s)", flush=True)

    # === Phase 5: Multi-length exploration ===
    if time.monotonic() < deadline - 150:
        t0 = time.monotonic()
        print(f"\n{'='*50}", flush=True)
        print(f"Phase 5: Multi-length (base n={current_best_n})", flush=True)
        print(f"{'='*50}", flush=True)

        base_n = current_best_n
        lengths_to_try = sorted(set([
            base_n - 64, base_n + 32, base_n + 64, base_n + 128,
        ]))
        lengths_to_try = [n for n in lengths_to_try if 128 <= n <= 2048 and n != base_n and n % 2 == 0]
        
        for n_try in lengths_to_try:
            if time.monotonic() > deadline - 130:
                break
            a_try = resample_seq(current_best, n_try)
            a_try = np.maximum(a_try, 1e-10)
            a_try = a_try / a_try.sum()
            perf = max(15, min(50, (deadline - 120 - time.monotonic()) / max(1, len(lengths_to_try))))
            a_try, J_try = lp_mirror_descent_fast(a_try, 2000, time.monotonic() + perf, use_active_set=True)
            tracker.update(a_try)
            print(f"  n={n_try}: J={J_try:.8f}", flush=True)

            if J_try < current_best_J:
                current_best = a_try.copy()
                current_best_n = n_try
                current_best_J = J_try

        print(f"Phase 5 done ({time.monotonic()-t0:.1f}s)", flush=True)

    # === Phase 6: Long LP refinement (linear mode) ===
    if time.monotonic() < deadline - 80:
        t0 = time.monotonic()
        print(f"\n{'='*50}", flush=True)
        print(f"Phase 6: Long LP linear (n={current_best_n})", flush=True)
        print(f"{'='*50}", flush=True)

        a_long = current_best.copy()
        a_long = np.maximum(a_long, 1e-15)
        a_long = a_long / a_long.sum()
        a_long, J_long = lp_mirror_descent_fast(a_long, 10000, deadline - 60, use_active_set=True)
        tracker.update(a_long)
        print(f"  J={J_long:.8f} ({time.monotonic()-t0:.1f}s)", flush=True)

        if J_long < current_best_J:
            current_best = a_long.copy()
            current_best_J = J_long

    # === Phase 7: Final multiplicative LP ===
    if time.monotonic() < deadline - 40:
        t0 = time.monotonic()
        print(f"\n{'='*50}", flush=True)
        print(f"Phase 7: Final LP multiplicative (n={current_best_n})", flush=True)
        print(f"{'='*50}", flush=True)

        a_fin = current_best.copy()
        a_fin = np.maximum(a_fin, 1e-15)
        a_fin = a_fin / a_fin.sum()
        a_fin, J_fin = lp_multiplicative(a_fin, 5000, deadline - 30)
        tracker.update(a_fin)
        print(f"  J={J_fin:.8f} ({time.monotonic()-t0:.1f}s)", flush=True)

        if J_fin < current_best_J:
            current_best = a_fin.copy()
            current_best_J = J_fin

    # === Phase 8: L-BFGS-B final polish ===
    if time.monotonic() < deadline - 5 and tracker.best is not None:
        t0 = time.monotonic()
        print(f"\n{'='*50}", flush=True)
        print(f"Phase 8: L-BFGS-B polish (n={tracker.best.size})", flush=True)
        print(f"{'='*50}", flush=True)
        n = tracker.best.size
        try:
            res = minimize(lbb_objective, tracker.best.astype(np.float64),
                           method='L-BFGS-B', bounds=[(0.0, None)] * n,
                           options={'maxiter': 200, 'max_fun': 50000, 'ftol': 1e-16, 'gtol': 1e-14})
            if res.fun < 1e10:
                tracker.update(res.x)
            print(f"  L-BFGS-B J={2*n*res.fun:.8f}", flush=True)
        except Exception as e:
            print(f"  L-BFGS-B error: {e}", flush=True)
        print(f"  ({time.monotonic()-t0:.1f}s)", flush=True)

    # Final
    if tracker.best is not None:
        final_a = tracker.best
    else:
        final_a = current_best

    print(f"\n{'='*50}", flush=True)
    print(f"FINAL: J={tracker.best_score:.8f} n={final_a.size} time={time.monotonic()-tracker.t_start:.0f}s", flush=True)
    print(f"{'='*50}", flush=True)

    return [float(x) for x in final_a]
