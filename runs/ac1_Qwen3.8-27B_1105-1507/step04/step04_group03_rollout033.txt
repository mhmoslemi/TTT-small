```python
import numpy as np
from scipy.optimize import linprog
from scipy.sparse import csr_matrix
import time

def fast_conv(a):
    """Full self-convolution via FFT."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]

def compute_J(a):
    """Compute J(a) = 2*n*max(conv(a,a))/(sum(a))^2."""
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))

def build_A_sparse(n, a):
    """Build sparse matrix A where A[k,j] = a[k-j] for valid j."""
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        j_min = max(0, k - n + 1)
        j_max = min(n - 1, k)
        for j in range(j_min, j_max + 1):
            rows.append(k)
            cols.append(j)
            vals.append(a[k - j])
    return csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))

def lp_best_response(a, M):
    """Solve LP: max sum(g) s.t. A@g <= M, g >= 0. Returns (g, sum_g)."""
    n = len(a)
    A = build_A_sparse(n, a)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    try:
        r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
        if r.success and r.x is not None:
            g = np.maximum(r.x, 0)
            return g, float(g.sum())
    except Exception:
        pass
    return None, 0.0

def active_set_lp(a, M, tol=1e-3):
    """Solve LP using only active constraints."""
    n = len(a)
    b = fast_conv(a)
    # Only include constraints where b[k] >= M - tol*M
    active = np.where(b >= M - tol * M)[0]
    
    if len(active) == 0:
        active = np.array([np.argmax(b)])
    
    # Build sparse matrix with only active constraints
    rows = []
    cols = []
    vals = []
    for idx, k in enumerate(active):
        j_min = max(0, k - n + 1)
        j_max = min(n - 1, k)
        for j in range(j_min, j_max + 1):
            rows.append(idx)
            cols.append(j)
            vals.append(a[k - j])
    
    A = csr_matrix((vals, (rows, cols)), shape=(len(active), n))
    c = -np.ones(n)
    b_ub = np.full(len(active), M)
    bounds = [(0, None)] * n
    
    try:
        r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
        if r.success and r.x is not None:
            g = np.maximum(r.x, 0)
            # Verify feasibility against ALL constraints
            test_conv = np.convolve(a, g, 'full')
            if test_conv.max() <= M * (1 + 1e-6):
                return g, float(g.sum())
            # If infeasible, fall back to full LP
        else:
            # Fall back to full LP
            return lp_best_response(a, M)
    except Exception:
        return lp_best_response(a, M)
    
    return lp_best_response(a, M)

def resample_seq(src, dest_n):
    """Resample 1D sequence to new length."""
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)
```

```python
def prox_simplex(v):
    """Project v onto the probability simplex {a: a >= 0, sum(a) = 1}."""
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n+1) > (cssv - 1))[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0)
```

```python
                # valid i: 0 <= k-i < n => k-n+1 <= i <= k
                i_start = max(0, k - n + 1)
                i_end = min(n - 1, k)
                for i in range(i_start, i_end + 1):
                    grad[i] += 2.0 * w * a[k - i]
            
            # Update with proximal step
            a_new = a - eta * grad
            a_new = prox_simplex(a_new)
            a_new = np.maximum(a_new, 1e-15)
            s = a_new.sum()
            if s < 1e-10:
                break
            a_new = a_new / s
            
            total_iters += 1
            J_new = compute_J(a_new)
            if J_new < best_J:
                best_J = J_new
                best_a = a_new.copy()
            
            a = a_new
        
        # Decay step size
        eta *= 0.8
        if eta < 1e-6:
            break
    
    return best_a, best_J
```

```python
"""
Improved optimizer for minimizing J(a) = 2*n*max(conv(a,a))/sum(a)^2.
Uses aggressive LP mirror descent with active-set acceleration, smoothed proximal gradient,
and extended final polishing. Key improvements over previous version:
- Active-set LP for faster solves
- Smoothed proximal gradient (FISTA) as independent method
- Better step size schedules with backtracking
- Extended final LP polishing phase
- Dual gap tracking for convergence detection

import numpy as np
from scipy.optimize import linprog
from scipy.sparse import csr_matrix
import time


def fast_conv(a):
    """Full self-convolution via FFT."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]


def compute_J(a):
    """Compute J(a) = 2*n*max(conv(a,a))/(sum(a))^2."""
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))


def build_A_sparse(n, a):
    """Build sparse matrix A where A[k,j] = a[k-j] for valid j."""
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        j_min = max(0, k - n + 1)
        j_max = min(n - 1, k)
        for j in range(j_min, j_max + 1):
            rows.append(k)
            cols.append(j)
            vals.append(a[k - j])
    return csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))


def lp_best_response_full(a, M):
    """Solve full LP: max sum(g) s.t. conv(a,g) <= M, g >= 0."""
    n = len(a)
    A = build_A_sparse(n, a)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    try:
        r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
        if r.success and r.x is not None:
            g = np.maximum(r.x, 0)
            dual = r.ineqbnd if hasattr(r, 'ineqbnd') else None
            return g, float(g.sum())
    except Exception:
        pass
    return None, 0.0


def lp_best_response_active(a, M, tol_frac=0.05):
    """Solve LP using only near-active constraints for speed."""
    n = len(a)
    b = fast_conv(a)
    threshold = M - tol_frac * M
    active = np.where(b >= threshold)[0]
    
    if len(active) < 3:
        active = np.where(b >= b.max() - 0.2 * b.max())[0]
    
    if len(active) == 0:
        active = np.array([int(np.argmax(b))])
    
    # Build sparse matrix with only active constraints
    rows = []
    cols = []
    vals = []
    for idx, k in enumerate(active):
        j_min = max(0, k - n + 1)
        j_max = min(n - 1, k)
        for j in range(j_min, j_max + 1):
            rows.append(idx)
            cols.append(j)
            vals.append(a[k - j])
    
    A = csr_matrix((vals, (rows, cols)), shape=(len(active), n))
    c = -np.ones(n)
    b_ub = np.full(len(active), M)
    bounds = [(0, None)] * n
    
    try:
        r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
        if r.success and r.x is not None:
            g = np.maximum(r.x, 0)
            # Verify against ALL constraints
            if g.sum() > 1e-15:
                test_conv = np.convolve(a, g, 'full')
                if test_conv.max() <= M * (1 + 1e-5):
                    return g, float(g.sum())
    except Exception:
        pass
    
    # Fall back to full LP
    return lp_best_response_full(a, M)


def prox_simplex(v):
    """Project v onto the probability simplex {a: a >= 0, sum(a) = 1}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - 1))[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho = rho[-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0)


def smoothed_prox_grad(a, deadline, tau_start=10.0, tau_end=100.0):
    """
    Smoothed proximal gradient (FISTA-style) for minimizing max_k (a*a)_k
    subject to a >= 0, sum(a) = 1.
    Uses log-sum-exp smoothing with increasing temperature.
    """
    a = np.maximum(a.astype(np.float64), 1e-15)
    a = a / a.sum()
    n = len(a)
    best_J = compute_J(a)
    best_a = a.copy()
    
    # FISTA momentum
    y = a.copy()
    t_mom = 1.0
    eta = 0.5 / np.sqrt(n)
    total_iters = 0
    max_total = 200000
    
    while time.monotonic() < deadline and total_iters < max_total:
        # Adaptive temperature
        frac = min(1.0, total_iters / 50000.0)
        tau = tau_start + (tau_end - tau_start) * frac
        
        b = fast_conv(y)
        b_max = b.max()
        
        # Softmax weights
        scaled = tau * (b - b_max)
        weights = np.exp(scaled)
        w_sum = weights.sum()
        if w_sum < 1e-30:
            break
        weights = weights / w_sum
        
        # Compute gradient: grad_i = 2 * sum_k weights[k] * a[k-i] for valid indices
        grad = np.zeros(n)
        # Use threshold to only process significant lags
        active_mask = weights > 1e-8
        active_lags = np.where(active_mask)[0]
        
        for k in active_lags:
            w = weights[k]
            i_start = max(0, k - n + 1)
            i_end = min(n - 1, k)
            # Vectorized: grad[i_start:i_end+1] += 2*w*a[k - i for i in range(i_start, i_end+1)]
            indices = np.arange(i_start, i_end + 1)
            grad[indices] += 2.0 * w * a[k - indices]
        
        # FISTA update
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t_mom * t_mom)) / 2.0
        y_new = a + ((t_mom - 1.0) / t_new) * (a - y)
        y_new = np.maximum(y_new, 0.0)
        s_y = y_new.sum()
        if s_y > 1e-15:
            y_new = y_new / s_y
        
        # Gradient step with backtracking
        step = eta
        for bt in range(20):
            candidate = y_new - step * grad
            candidate = prox_simplex(candidate)
            candidate = np.maximum(candidate, 1e-15)
            s_c = candidate.sum()
            if s_c < 1e-10:
                step *= 0.5
                continue
            candidate = candidate / s_c
            J_cand = compute_J(candidate)
            if J_cand <= best_J + 0.01 * step * np.linalg.norm(grad):
                break
            step *= 0.5
        
        a = candidate
        t_mom = t_new
        total_iters += 1
        
        J_a = compute_J(a)
        if J_a < best_J:
            best_J = J_a
            best_a = a.copy()
        
        # Decay step size slowly
        if total_iters % 1000 == 0:
            eta *= 0.95
    
    return best_a, best_J


def lp_mirror_descent(a, max_iters, deadline, mode='linear', full_step_period=50, tol=1e-6):
    """
    LP mirror descent for minimizing max convolution.
    mode='linear': convex combination update
    mode='multiplicative': exponentiated gradient update
    """
    a = np.maximum(a.astype(np.float64), 1e-12)
    a = a / a.sum()
    n = len(a)
    current_J = compute_J(a)
    best_J = current_J
    best_a = a.copy()
    no_improve = 0
    iteration = 0
    use_active = True
    
    while time.monotonic() < deadline:
        if iteration >= max_iters:
            break
        iteration += 1
        
        b = fast_conv(a)
        M = float(b.max())
        
        # Solve LP (active set for speed, fallback to full)
        if use_active:
            g, sum_g = lp_best_response_active(a, M, tol_frac=0.05)
        else:
            g, sum_g = lp_best_response_full(a, M)
        
        if g is None:
            break
        g = np.maximum(g, 0.0)
        sum_g = float(g.sum())
        if sum_g < 1e-10:
            break
        
        # Gap: 1 - sum_g (since sum(a) = 1 and a is feasible)
        gap = 1.0 - sum_g
        if gap < tol:
            break
        
        g = g / sum_g
        
        # Step size
        if iteration % full_step_period == 0:
            t = 1.0
        else:
            t = min(0.8, 0.15 + 0.7 * min(1.0, gap * 10))
        
        # Line search
        best_trial_J = current_J
        best_trial_a = None
        t_try = t
        
        for ls in range(20):
            if mode == 'linear':
                a_trial = (1.0 - t_try) * a + t_try * g
            else:
                ratio = g / np.maximum(a, 1e-15)
                a_trial = a * np.power(ratio, t_try)
            
            a_trial = np.maximum(a_trial, 0.0)
            s = a_trial.sum()
            if s < 1e-10:
                t_try *= 0.5
                continue
            a_trial = a_trial / s
            J_trial = compute_J(a_trial)
            
            if J_trial < best_trial_J - 1e-13:
                best_trial_J = J_trial
                best_trial_a = a_trial.copy()
                break
            t_try *= 0.6
            if t_try < 0.003:
                break
        
        if best_trial_a is not None:
            a = best_trial_a
            current_J = best_trial_J
            if current_J < best_J - 1e-10:
                best_J = current_J
                best_a = a.copy()
                no_improve = 0
            else:
                no_improve += 1
        else:
            no_improve += 1
        
        # Escape mechanism
        if no_improve > 25:
            if mode == 'linear':
                a = g.copy()
            else:
                a = np.sqrt(a * g)
            a = np.maximum(a, 1e-12)
            s = a.sum()
            if s > 1e-10:
                a = a / s
                current_J = compute_J(a)
                if current_J < best_J:
                    best_J = current_J
                    best_a = a.copy()
            no_improve = 0
        elif no_improve > 80:
            break
    
    return best_a, best_J


def resample_seq(src, dest_n):
    """Resample 1D sequence to new length."""
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)

        a_final = np.maximum(a_final, 1e-12)
        a_final = a_final / a_final.sum()
        
        # Long linear LP
        a_final, J_final = lp_mirror_descent(a_final, 200000, deadline - 30.0,
                                             mode='linear', full_step_period=80, tol=1e-8)
        do_update(a_final)
        print(f"  Linear: J={J_final:.8f} ({time.monotonic()-t0:.1f}s)", flush=True)
        
        # Follow with multiplicative from the best
        if time.monotonic() < deadline - 10.0:
            a_mult = a_final.copy() if J_final < best_J else (best_a.copy() if best_a is not None else a_final)
            a_mult = np.maximum(a_mult, 1e-12)
            a_mult = a_mult / a_mult.sum()
            a_mult, J_mult = lp_mirror_descent(a_mult, 50000, deadline - 5.0,
                                               mode='multiplicative', full_step_period=100, tol=1e-8)
            do_update(a_mult)
            print(f"  Mult: J={J_mult:.8f} ({time.monotonic()-t0:.1f}s)", flush=True)
    
    # === Final output ===
    if best_a is not None:
        print(f"\n{'='*50}", flush=True)
        print(f"FINAL: J={best_J:.8f} n={best_a.size} time={time.monotonic()-t_start:.0f}s", flush=True)
        print(f"{'='*50}", flush=True)
        return [float(x) for x in best_a]
    else:
        print(f"\nFALLBACK: returning parent", flush=True)
        return [float(x) for x in current]
```
