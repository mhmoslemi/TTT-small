"""

import numpy as np
from scipy.optimize import linprog
import time


def compute_conv(a):
    """Compute conv(a,a) using FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([0.0])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    conv = np.fft.irfft(fa * fa, size)
    return conv[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    conv = compute_conv(a)
    max_b = float(np.max(conv))
    sum_a = float(np.sum(a))
    if sum_a < 0.01:
        return np.inf
    return 2.0 * n * max_b / (sum_a * sum_a)


def build_toeplitz(a):
    """Build Toeplitz matrix A where (A@g)[k] = (a*g)[k] for k=0..2n-2."""
    n = len(a)
    A = np.zeros((2 * n - 1, n), dtype=np.float64)
    for j in range(n):
        k_start = j
        k_end = min(j + n, 2 * n - 1)
        A[k_start:k_end, j] = a[:k_end - k_start]
    return A


def solve_lp_maxsum(a, M, time_limit=8):
    """Solve LP: max sum(g) s.t. (a*g)[k] <= M for all k, g >= 0."""
    n = len(a)
    A = build_toeplitz(a)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': time_limit})
        if res.status == 0:
            g = np.maximum(res.x, 0)
            return g, float(np.sum(g))
        elif res.x is not None:
            g = np.maximum(res.x, 0)
            s = float(np.sum(g))
            if s > 1e-10:
                return g, s
    except Exception:
        pass
    return None, None


def solve_lp_minpeak(a, time_limit=8):
    """Solve LP: min M s.t. (a*g)[k] <= M, sum(g)=1, g>=0. Returns (g, M)."""
    n = len(a)
    A = build_toeplitz(a)
    c = np.zeros(n + 1)
    c[-1] = 1.0
    A_ub = np.zeros((2 * n - 1, n + 1))
    A_ub[:, :n] = A
    A_ub[:, n] = -1.0
    b_ub = np.zeros(2 * n - 1)
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])
    bounds = [(0.0, None)] * n + [(0.0, None)]
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': time_limit})
        if res.status == 0:
            g = np.maximum(res.x[:n], 0)
            s = np.sum(g)
            if s < 1e-10:
                return None, None
            g = g / s
            return g, float(res.x[n])
        elif res.x is not None:
            g = np.maximum(res.x[:n], 0)
            s = np.sum(g)
            if s > 1e-10:
                g = g / s
                return g, float(res.x[n])
    except Exception:
        pass
    return None, None


def analytical_subgradient(a):
    """
    Compute subgradient of f(a) = 2*n*max_k conv_k / (sum a)^2.
    At the current maximizer k*, the subgradient is:
    grad[i] = 4*n*(a[k*-i]*S - C) / S^3  if 0 <= k*-i < n
             = -4*n*C / S^3                otherwise
    """
    n = len(a)
    conv = compute_conv(a)
    k_star = int(np.argmax(conv))
    C = conv[k_star]
    S = float(np.sum(a))
    if S < 1e-10:
        return np.zeros(n)
    grad = np.full(n, -4.0 * n * C / (S * S * S))
    # For indices i where 0 <= k_star - i < n
    i_min = max(0, k_star - (n - 1))
    i_max = min(n - 1, k_star)
    for i in range(i_min, i_max + 1):
        idx = k_star - i
        grad[i] = 4.0 * n * (a[idx] * S - C) / (S * S * S)
    return grad


def subgradient_step(a, step, max_iters=200, time_budget=10.0):
    """Subgradient optimization with adaptive step size."""
    n = len(a)
    a = a.copy().astype(np.float64)
    S = np.sum(a)
    if S < 1e-10:
        return a
    a = a / S
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    stall = 0
    patience = 50

    for it in range(max_iters):
        if time.monotonic() - start > time_budget:
            break

        grad = analytical_subgradient(a)
        # Normalize gradient
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-15:
            break
        grad = grad / gnorm

        # Line search
        best_t = step
        best_f = best_obj
        for t in [step, step * 0.5, step * 0.25, step * 2.0]:
            a_new = a - t * grad
            a_new = np.maximum(a_new, 0)
            s_new = np.sum(a_new)
            if s_new < 1e-10:
                continue
            a_new = a_new / s_new
            obj_new = compute_obj(a_new)
            if obj_new < best_f:
                best_f = obj_new
                best_t = t

        if best_f < best_obj - 1e-14:
            a_new = a - best_t * grad
            a_new = np.maximum(a_new, 0)
            s_new = np.sum(a_new)
            if s_new > 1e-10:
                a_new = a_new / s_new
            a = a_new
            best_obj = best_f
            best_a = a.copy()
            stall = 0
            step = min(step * 1.05, 0.5)
        else:
            stall += 1
            step *= 0.8
            if step < 1e-5:
                break
            if stall > patience:
                # Add noise for escape
                rng = np.random.default_rng(it * 7 + 3)
                a = a * np.exp(rng.normal(0, 0.05, n))
                a = np.maximum(a, 1e-10)
                a = a / np.sum(a)
                stall = 0
                patience = max(patience // 2, 20)
                step = 0.1

    return best_a

def multiplicative_subgradient(a, tau=30.0, step=0.05, max_iters=500, time_budget=15.0):
    """Multiplicative subgradient with the exponential reweighting trick."""
    n = len(a)
    a = a.copy().astype(np.float64)
    S = np.sum(a)
    if S < 1e-10:
        return a
    a = a / S
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    stall = 0

    for it in range(max_iters):
        if time.monotonic() - start > time_budget:
            break

        conv = compute_conv(a)
        max_conv = np.max(conv)
        shifted = tau * (conv - max_conv)
        shifted = np.clip(shifted, -500, 500)
        weights = np.exp(shifted)
        w_sum = np.sum(weights)
        if w_sum < 1e-300:
            break
        weights /= w_sum

        # Gradient via correlation
        c = np.correlate(weights, a, mode='full')
        grad = 2.0 * c[n - 1:n - 1 + n]

        a_new = a * np.exp(-step * grad)
        a_new = np.maximum(a_new, 1e-15)
        s_new = np.sum(a_new)
        if s_new > 1e-10:
            a_new = a_new / s_new
        obj = compute_obj(a_new)

        if obj < best_obj - 1e-14:
            best_obj = obj
            best_a = a_new.copy()
            a = a_new
            stall = 0
        else:
            a = a_new  # still update a for momentum
            stall += 1
            if stall > 40:
                step *= 0.8
                tau *= 1.1
                if step < 0.001:
                    break
                stall = 0

    return best_a


def resample_sequence(a, new_n):
    """Resample a sequence to a new length using interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    old_k = np.arange(n, dtype=np.float64)
    new_k = np.linspace(0, n - 1, new_n)
    a_new = np.interp(new_k, old_k, a)
    a_new = np.maximum(a_new, 0)
    s = np.sum(a_new)
    if s > 1e-10:
        a_new = a_new / s
    return a_new


def random_init(n, seed):
    """Generate diverse random initializations optimized for this problem."""
    rng = np.random.default_rng(seed)
    k = np.arange(n, dtype=np.float64)
    t = k / n
    choice = rng.integers(0, 8)

    if choice == 0:

    elif choice == 1:
        # Multi-frequency
        f1 = rng.uniform(0.5, 2.0)
        f2 = rng.uniform(2.0, 5.0)
        a = 1.0 + 0.5 * np.cos(2 * np.pi * f1 * t) + 0.3 * np.cos(2 * np.pi * f2 * t)
    elif choice == 2:
        # Smooth random
        m = max(5, n // 15)
        vals = rng.uniform(0.3, 2.0, m)
        a = np.interp(t, np.linspace(0, 1, m), vals)
    elif choice == 3:
        # W-shape
        f = rng.uniform(2, 5)
        a = 1.0 + 0.6 * np.cos(2 * np.pi * f * t) ** 2
    elif choice == 4:
        # Exponential decay from center
        center = n * 0.5
        alpha = rng.uniform(1.0, 4.0)
        a = 1.0 + 0.8 * np.exp(-alpha * (t - 0.5) ** 2)
    elif choice == 5:
        # Random smooth with enforcement
        m = max(3, n // 20)
        vals = rng.uniform(0.1, 3.0, m)
        a = np.interp(t, np.linspace(0, 1, m), vals)
    elif choice == 6:
        # Alternating with decay
        freq = rng.uniform(3, 8)
        a = 1.0 + 0.4 * np.exp(-2.0 * t * (1 - t)) * np.cos(2 * np.pi * freq * t)
    else:
        # Pure U-shape variant
        power = rng.uniform(1.5, 4.0)
        a = 1.0 + 0.8 * np.abs(2.0 * t - 1.0) ** power

    a = np.maximum(a, 0)
    s = np.sum(a)
    if s < 1e-10:
        a = np.ones(n) / n
    else:

    if remaining > 5:
        t3 = min(8.0, remaining * 0.15)
        a_new = multiplicative_subgradient(best_a, tau=40.0, step=0.03,
                                           max_iters=300, time_budget=t3)
        obj = compute_obj(a_new)
        if obj < best_obj:
            best_obj = obj
            best_a = a_new.copy()

    # Phase 4: Another LP round
    remaining = time_budget - (time.monotonic() - start)
    if remaining > 10:
        t4 = min(20.0, remaining * 0.3)
        a_new = alternating_lp_refined(best_a, max_iters=50, time_budget=t4)
        obj = compute_obj(a_new)
        if obj < best_obj:
            best_obj = obj
            best_a = a_new.copy()

    # Phase 5: Subgradient polish
    remaining = time_budget - (time.monotonic() - start)
    if remaining > 5:
        t5 = min(10.0, remaining * 0.2)
        a_new = subgradient_step(best_a, step=0.05, max_iters=300, time_budget=t5)
        obj = compute_obj(a_new)
        if obj < best_obj:
            best_obj = obj
            best_a = a_new.copy()

    return best_a, best_obj


def intensive_local_search(a, time_budget=120.0):
    """Intensive local search combining coordinate, block, and LP-based methods."""
    n = len(a)
    a = a.copy().astype(np.float64)
    S = np.sum(a)
    if S < 1e-10:
        return a, np.inf
    a = a / S

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    # Coordinate search
    remaining = time_budget - (time.monotonic() - start)
    if remaining > 10:
        t = min(20.0, remaining * 0.2)
        a_new = coordinate_local_search(best_a, time_budget=t)
        obj = compute_obj(a_new)
        if obj < best_obj:
            best_obj = obj
            best_a = a_new.copy()

    # Block search
    remaining = time_budget - (time.monotonic() - start)
    if remaining > 10:
        t = min(20.0, remaining * 0.2)
        a_new = block_local_search(best_a, time_budget=t, block_size=3)
        obj = compute_obj(a_new)
        if obj < best_obj:
            best_obj = obj
            best_a = a_new.copy()

    # Iterative: small perturbations + quick LP
    attempt = 0
    while time.monotonic() - start < time_budget - 10:
        attempt += 1
        rng = np.random.default_rng(attempt * 99983 + 7)
        remaining = time_budget - (time.monotonic() - start)
        if remaining < 8:
            break

        # Perturbation
        scale = 0.01 + 0.04 * (attempt % 5) / 5.0
        a_pert = best_a * np.exp(rng.normal(0, scale, n))
        a_pert = np.maximum(a_pert, 1e-10)
        a_pert = a_pert / np.sum(a_pert)

        # Quick optimization
        sub_time = min(10.0, remaining * 0.3)
        a_opt, obj = optimize_pipeline(a_pert, time_budget=sub_time)

        if obj < best_obj:
            best_obj = obj
            best_a = a_opt.copy()
            # Reset perturbation scale when we improve
            scale = 0.01

    # Final coordinate polish
    remaining = time_budget - (time.monotonic() - start)
    if remaining > 5:
        t = min(remaining - 2, 15.0)
        a_new = coordinate_local_search(best_a, time_budget=t)
        obj = compute_obj(a_new)
        if obj < best_obj:
            best_obj = obj
            best_a = a_new.copy()

    return best_a, best_obj


def try_parent():
    """Try to load parent sequence."""
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64)
        parent = np.clip(parent, 0, 1000.0)
        if parent.size > 0 and np.isfinite(parent).all():
            obj = compute_obj(parent)
            print(f"Parent (n={parent.size}): obj={obj:.6f}")
            return parent.copy(), obj
    except Exception:
        pass
    return None, np.inf


def propose_candidate(seed=42, budget_s=1500):
    """
    Main search function. Uses multi-phase optimization:
    1. Load parent and try nearby n values
    2. Intensive optimization pipeline
    3. Fine local search
    Returns the best sequence found within the time budget.
    """
    start_time = time.monotonic()
    deadline = start_time + float(budget_s) - 30.0

    best_seq = None
    best_score = np.inf

    # Load parent
    parent, parent_obj = try_parent()
    if parent is not None and parent_obj < best_score:
        best_score = parent_obj
        best_seq = parent.copy()
        n_parent = len(parent)
    else:
        n_parent = 330

    print(f"Initial best: {best_score:.6f} (n={len(best_seq) if best_seq is not None else 'N/A'})")
    print(f"Time budget: {budget_s}s, deadline in {deadline - start_time:.0f}s")

    # === Phase 1: n-sweep with quick optimization (200s) ===
    print("\n=== Phase 1: n-sweep ===")
    phase1_end = start_time + 200.0
    n_values = sorted(set([
        n_parent, n_parent - 5, n_parent + 5,
        n_parent - 10, n_parent + 10,
        n_parent - 20, n_parent + 20,
        n_parent - 30, n_parent + 30,
        300, 350, 400, 250
    ]))
    n_values = [n for n in n_values if 150 <= n <= 800]
    n_values = list(dict.fromkeys(n_values))  # deduplicate preserving order

    for n_val in n_values:
        if time.monotonic() > phase1_end - 5 or time.monotonic() > deadline - 600:
            break
        remaining = phase1_end - time.monotonic()
        if remaining < 15:
            break
        alloc = min(20.0, remaining * 0.5)
        alloc = max(alloc, 10.0)

        # Try up to 2 initializations
        for init_i in range(2):
            if time.monotonic() > phase1_end - 3:
                break
            sub_time = min(alloc, phase1_end - time.monotonic() - 2)
            if sub_time < 5:
                break
            if init_i == 0 and n_val == n_parent and parent is not None:
                a0 = resample_sequence(parent, n_val)
            else:
                a0 = random_init(n_val, seed + n_val * 104729 + init_i * 31)
            a_opt, obj = optimize_pipeline(a0, time_budget=sub_time)
            if obj < best_score:
                best_score = obj
                best_seq = a_opt.copy()
                print(f"  n={n_val}, init={init_i}: {obj:.6f} (UPDATE)")

    print(f"After Phase 1: best={best_score:.6f}, n={len(best_seq)}")

    # === Phase 2: Intensive optimization (500s) ===
    print("\n=== Phase 2: Intensive optimization ===")
    phase2_end = min(start_time + 700.0, deadline - 300.0)
    if best_seq is not None:
        sub_budget = min(phase2_end - time.monotonic() - 5, 400.0)
        if sub_budget > 20:
            a_opt, obj = optimize_pipeline(best_seq, time_budget=sub_budget)
            if obj < best_score:
                best_score = obj
                best_seq = a_opt.copy()
                print(f"  After pipeline: {best_score:.6f}")

    # Second round with different parameters
    if best_seq is not None:
        remaining = phase2_end - time.monotonic()
        if remaining > 30:
            sub_budget = min(remaining - 5, 200.0)
            a_opt, obj = optimize_pipeline(best_seq, time_budget=sub_budget)
            if obj < best_score:
                best_score = obj
                best_seq = a_opt.copy()
                print(f"  After pipeline 2: {best_score:.6f}")

    # Try a few more n values around the best
    if best_seq is not None:
        n_best = len(best_seq)
        remaining = phase2_end - time.monotonic()
        if remaining > 30:
            for delta in [-3, -2, -1, 1, 2, 3]:
                n_try = n_best + delta
                if 200 <= n_try <= 600:
                    if time.monotonic() > phase2_end - 10 or time.monotonic() > deadline - 300:
                        break
                    sub_time = min(25.0, phase2_end - time.monotonic() - 5)
                    if sub_time < 10:
                        break
                    a0 = resample_sequence(best_seq, n_try)
                    a_opt, obj = optimize_pipeline(a0, time_budget=sub_time)
                    if obj < best_score:
                        best_score = obj
                        best_seq = a_opt.copy()
                        print(f"  n={n_try}: {obj:.6f} (UPDATE)")

    print(f"After Phase 2: best={best_score:.6f}, n={len(best_seq)}")

    # === Phase 3: Fine local search (remaining time) ===
    print("\n=== Phase 3: Fine local search ===")
    remaining = deadline - time.monotonic()
    if best_seq is not None and remaining > 30:
        sub_budget = min(remaining - 10, 400.0)
        a_opt, obj = intensive_local_search(best_seq, time_budget=sub_budget)
        if obj < best_score:
            best_score = obj
            best_seq = a_opt.copy()
            print(f"  After local search: {best_score:.6f}")

        # One more pipeline round
        remaining = deadline - time.monotonic()
        if remaining > 30:
            sub_budget = min(remaining - 10, 200.0)
            a_opt, obj = optimize_pipeline(best_seq, time_budget=sub_budget)
            if obj < best_score:
                best_score = obj
                best_seq = a_opt.copy()
                print(f"  After final pipeline: {best_score:.6f}")

        # Final coordinate polish
        remaining = deadline - time.monotonic()
        if remaining > 10:
            sub_budget = min(remaining - 5, 50.0)
            a_new = coordinate_local_search(best_seq, time_budget=sub_budget)
            obj = compute_obj(a_new)
            if obj < best_score:
                best_score = obj
                best_seq = a_new.copy()
                print(f"  After coordinate polish: {best_score:.6f}")

    # Final validation
    if best_seq is None:
        best_seq = np.ones(50) / 50
        best_score = compute_obj(best_seq)

    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    val = evaluate_sequence(result)
    if val < best_score:
        best_score = val

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(best_seq)}, score={best_score:.6f}, time={elapsed:.1f}s ===")

    return result
