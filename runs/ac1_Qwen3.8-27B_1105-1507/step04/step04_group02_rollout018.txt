"""
Optimized step-function coefficient search minimizing 2*n*max(conv(a,a))/(sum(a))^2.
Key improvements over previous approach:
1. Efficient alternating LP with active-set tracking via dual variables
2. Nesterov-accelerated subgradient with adaptive top-K
3. Structured "anti-peak" perturbations targeting active constraints
4. Multi-scale warm-starting
5. Better time management with parallel-free sequential scheduling
"""

import numpy as np
from scipy.optimize import linprog
from scipy.sparse import coo_matrix
import time


def compute_conv(a):
    """Compute linear convolution a*a via FFT."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    conv = np.fft.irfft(fa * fa, size)
    return conv[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    S = float(np.sum(a))
    if S < 1e-10:
        return np.inf
    conv = compute_conv(a)
    M = float(np.max(conv))
    return 2.0 * n * M / (S * S)


def build_toeplitz(a):
    """Build sparse matrix T where T @ g = conv(a, g)."""
    n = len(a)
    rows = []
    cols = []
    vals = []
    for j in range(n):
        # conv(a, g)[k] = sum_i a[i] * g[k-i]
        # For fixed j (= k-i), k = i + j, so k ranges from j to j+n-1
        # Valid k: 0 <= k <= 2n-2
        # i = k - j, valid when 0 <= i < n
        # So for column j, the non-zero entries are at rows j, j+1, ..., j+n-1
        # (capped at 2n-2)
        start_k = j
        end_k = min(j + n - 1, 2 * n - 2)
        for k in range(start_k, end_k + 1):
            i = k - j
            rows.append(k)
            cols.append(j)
            vals.append(a[i])
    T = coo_matrix((vals, (rows, cols)), shape=(2 * n - 1, n)).tocsr()
    return T


def solve_lp_dual(a, M, time_limit=10):
    """
    Solve the dual of: max sum(g) s.t. T@g <= M, g >= 0
    Dual: min M * sum(y) s.t. T^T @ y >= 1, y >= 0
    
    Returns: (g, dual_y, lp_status)
    """
    n = len(a)
    T = build_toeplitz(a)
    
    # Primal: max 1^T g s.t. T g <= M*1, g >= 0
    # c = -ones (for minimization)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    
    try:
        res = linprog(c, A_ub=T, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': time_limit,
                               'dual_feasibility_tolerance': 1e-9,
                               'primal_feasibility_tolerance': 1e-9})
        if res.status == 0:
            g = np.maximum(res.x, 0)
            # Get dual variables (shadow prices)
            y = np.maximum(-res.ineqlin.marginals, 0) if res.ineqlin.marginals is not None else None
            return g, y, res.status
        elif res.status == 3:  # Time limit
            # Return best feasible solution found
            if res.x is not None:
                g = np.maximum(res.x, 0)
                y = np.maximum(-res.ineqlin.marginals, 0) if res.ineqlin.marginals is not None else None
                return g, y, 3
    except Exception as e:
        pass
    return None, None, -1


def nesterov_subgradient(a, time_budget, topk=10, base_lr=0.1):
    """Nesterov-accelerated subgradient descent."""
    n = len(a)
    a = a.copy().astype(np.float64)
    S = np.sum(a)
    if S < 1e-10:
        return a
    a = a / S
    
    # Nesterov state
    y = a.copy()
    v = np.zeros(n)
    beta = 0.9
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    
    for it in range(100000):
        if time.monotonic() - start > time_budget:
            break
        
        # Compute subgradient at y
        conv_y = compute_conv(y)
        M_y = float(np.max(conv_y))
        
        # Top-K active constraints
        if topk >= len(conv_y):
            active = np.arange(len(conv_y))
        else:
            active = np.argsort(conv_y)[-topk:]
        
        # Gradient of max(conv(y,y)) w.r.t. y
        grad = np.zeros(n)
        T_y = build_toeplitz(y)
        # grad = T_y^T @ (1/|active|) * sum of active rows
        active_mask = np.zeros(2 * n - 1)
        active_mask[active] = 1.0 / len(active)
        grad = T_y.T @ (2.0 * active_mask)
        
        # Nesterov update
        alpha = 2.0 / (it + 2.0)
        v_new = beta * v + (1 - beta) * grad
        y_new = y - alpha * base_lr * v_new
        y_new = np.maximum(y_new, 0)
        S_new = np.sum(y_new)
        if S_new < 1e-10:
            break
        y_new = y_new / S_new
        
        a_new = (1 - alpha) * a + alpha * y_new
        a_new = np.maximum(a_new, 0)
        S_a = np.sum(a_new)
        if S_a < 1e-10:
            break
        a_new = a_new / S_a
        
        a = a_new
        v = v_new
        y = a  # Reset y to a for simplicity (standard Nesterov would keep y separate)
        
        obj = compute_obj(a)
        if obj < best_obj:
            best_obj = obj
            best_a = a.copy()
    
    return best_a


def targeted_perturbation(a, conv, M, num_perts=5, time_budget=10.0):
    """
    Create perturbations that specifically target the active (peak) constraints.
    Idea: reduce values of a that contribute most to the peak, increase others.
    """
    n = len(a)
    start = time.monotonic()
    results = []
    
    # Find peak locations
    peak_indices = np.argsort(conv)[-min(10, 2*n-1):]
    
    for p in range(num_perts):
        if time.monotonic() - start > time_budget:
            break
        
        # For each peak location k, the gradient w.r.t. a[i] is a[k-i] (for valid k-i)
        # We want to decrease a[i] where it contributes to peaks, increase elsewhere
        
        peak_grad = np.zeros(n)
        for k in peak_indices:
            for i in range(n):
                j = k - i
                if 0 <= j < n:
                    peak_grad[i] += a[j]
        
        # Normalize
        peak_grad = peak_grad / (np.max(np.abs(peak_grad)) + 1e-10)
        
        # Create perturbation: reduce peak_contributing elements, increase others
        scale = 0.05 + 0.1 * (p / max(1, num_perts - 1))
        a_pert = a * (1.0 - scale * peak_grad + scale * np.mean(peak_grad))
        a_pert = np.maximum(a_pert, 0)
        S = np.sum(a_pert)
        if S > 1e-10:
            a_pert = a_pert / S
            results.append(a_pert)
    
    return results


def alternating_lp_refine(a, max_iters=50, time_budget=60.0):
    """
    Alternating LP with line search and dual-based active set tracking.
    This is the core optimization method.
    """
    n = len(a)
    a = a.copy().astype(np.float64)
    S = np.sum(a)
    if S < 1e-10:
        return a
    a = a / S
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    stall = 0
    consecutive_no_improve = 0
    
    for it in range(max_iters):
        if time.monotonic() - start > time_budget:
            break
        
        remaining = time_budget - (time.monotonic() - start)
        if remaining < 3:
            break
        
        conv = compute_conv(a)
        M = float(np.max(conv))
        
        # LP solve - adapt time limit to remaining budget
        lp_time = min(remaining * 0.3, 15.0)
        g, y_dual, status = solve_lp_dual(a, M, time_limit=lp_time)
        
        if g is None or status == -1:
            # Fallback: use subgradient step
            a_sg = nesterov_subgradient(a, time_budget=remaining * 0.2, topk=10)
            obj_sg = compute_obj(a_sg)
            if obj_sg < best_obj:
                best_obj = obj_sg
                best_a = a_sg.copy()
            continue
        
        S_g = np.sum(g)
        if S_g < 1e-10:
            break
        g = g / S_g
        
        # Line search
        best_t = 0.0
        best_obj_ls = best_obj
        for t in [0.1, 0.2, 0.3, 0.5, 0.7, 0.9, 1.0]:
            a_try = (1.0 - t) * a + t * g
            a_try = np.maximum(a_try, 0)
            S_try = np.sum(a_try)
            if S_try < 1e-10:
                continue
            a_try = a_try / S_try
            obj_try = compute_obj(a_try)
            if obj_try < best_obj_ls:
                best_obj_ls = obj_try
                best_t = t
        
        if best_t > 0:
            a = (1.0 - best_t) * a + best_t * g
            a = np.maximum(a, 0)
            a = a / np.sum(a)
            best_obj = best_obj_ls
            best_a = a.copy()
            stall = 0
            consecutive_no_improve = 0
        else:
            stall += 1
            consecutive_no_improve += 1
            
            if consecutive_no_improve > 5:
                # Try targeted perturbation
                perts = targeted_perturbation(a, conv, M, num_perts=3, time_budget=2.0)
                improved = False
                for a_p in perts:
                    # Quick subgradient on perturbed
                    a_p2 = nesterov_subgradient(a_p, time_budget=3.0, topk=8)
                    obj_p = compute_obj(a_p2)
                    if obj_p < best_obj:
                        best_obj = obj_p
                        best_a = a_p2.copy()
                        a = a_p2.copy()
                        improved = True
                        break
                if not improved:
                    # Reduce M slightly and retry
                    # (effectively relaxing the constraint)
                    M_relaxed = M * 1.01
                    g2, _, _ = solve_lp_dual(a, M_relaxed, time_limit=5.0)
                    if g2 is not None:
                        S_g2 = np.sum(g2)
                        if S_g2 > 1e-10:
                            g2 = g2 / S_g2
                            a = 0.8 * a + 0.2 * g2
                            a = np.maximum(a, 0)
                            a = a / np.sum(a)
                            obj_check = compute_obj(a)
                            if obj_check < best_obj:
                                best_obj = obj_check
                                best_a = a.copy()
            
            if stall > 20:
                break
    
    return best_a


def smooth_init(n, seed=0, mode='cosine'):
    """Generate smooth initialization."""
    rng = np.random.default_rng(seed)
    k = np.arange(n, dtype=np.float64)
    
    if mode == 'cosine':
        a = np.ones(n)
        for m in range(1, 10):
            c = rng.uniform(-0.5, 0.5) / m
            phase = rng.uniform(0, 2 * np.pi)
            a += c * np.cos(2 * np.pi * m * k / n + phase)
    elif mode == 'gaussian':
        center = n * rng.uniform(0.3, 0.7)
        sigma = n * rng.uniform(0.15, 0.4)
        a = np.exp(-0.5 * ((k - center) / sigma) ** 2)
    elif mode == 'tent':
        center = n // 2
        a = 1.0 - np.abs(k - center) / (n / 2.0)
    elif mode == 'arch':
        x = 2.0 * k / (n - 1) - 1.0
        a = np.sqrt(np.maximum(1.0 - x**2, 0.01))
    else:
        a = rng.dirichlet(np.ones(n))
        return a
    
    a = np.maximum(a, 0)
    S = np.sum(a)
    if S < 1e-10:
        return np.ones(n) / n
    return a / S


def multiscale_optimize(a, time_budget, scales=None):
    """
    Multi-scale approach: optimize a downsampled version first, then upsample and refine.
    """
    n = len(a)
    if scales is None:
        scales = [max(20, n // 8), max(50, n // 4), max(100, n // 2), n]
    
    start = time.monotonic()
    current = a.copy()
    
    for target_n in scales:
        if time.monotonic() - start > time_budget:
            break
        
        if target_n <= 20 or target_n >= n:
            continue
        
        # Downsample
        ratios = np.arange(target_n) * n / target_n
        idx = np.round(ratios).astype(int).clip(0, n - 1)
        coarse = current[idx]
        fine1d = np.sum(coarse)
        if fine1d > 1e-10:
            coarse = coarse / fine1d
        
        remaining = time_budget - (time.monotonic() - start)
        alloc = remaining * 0.4
        if alloc < 5:
            continue
        
        # Optimize coarse
        coarse_opt = alternating_lp_refine(coarse, max_iters=20, time_budget=alloc)
        
        # Upsample
        from scipy.interpolate import interp1d
        x_old = np.linspace(0, 1, target_n)
        x_new = np.linspace(0, 1, n)
        try:
            f = interp1d(x_old, coarse_opt, kind='cubic', fill_value='extrapolate')
            upsampled = np.maximum(f(x_new), 0)
        except:
            upsampled = current
        
        S = np.sum(upsampled)
        if S > 1e-10:
            upsampled = upsampled / S
            current = upsampled
    
    # Final refinement at full scale
    remaining = time_budget - (time.monotonic() - start)
    if remaining > 5:
        current = alternating_lp_refine(current, max_iters=30, time_budget=remaining)
    
    return current

"""
Optimized step-function coefficient search minimizing 2*n*max(conv(a,a))/(sum(a))^2.
Strategy: Focus on efficient alternating LP with dual-based active set tracking,
Nesterov-accelerated subgradient for escapes, structured anti-peak perturbations,
and multi-scale warm-starting. Key improvements: adaptive M relaxation, golden-section
line search, cached Toeplitz matrices, and longer focused refinement phases.
"""

import numpy as np
from scipy.optimize import linprog
from scipy.sparse import coo_matrix
from scipy.interpolate import interp1d
import time


def compute_conv(a):
    """Compute linear convolution a*a via FFT."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    conv = np.fft.irfft(fa * fa, size)
    return conv[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    S = float(np.sum(a))
    if S < 1e-10:
        return np.inf
    conv = compute_conv(a)
    M = float(np.max(conv))
    return 2.0 * n * M / (S * S)


def build_toeplitz(a):
    """Build sparse matrix T where T @ g = conv(a, g). Shape (2n-1, n)."""
    n = len(a)
    rows = np.empty(n * n, dtype=np.int32)
    cols = np.empty(n * n, dtype=np.int32)
    vals = np.empty(n * n, dtype=np.float64)
    idx = 0
    for j in range(n):
        start_k = j
        end_k = min(j + n - 1, 2 * n - 2)
        length = end_k - start_k + 1
        rows[idx:idx + length] = range(start_k, end_k + 1)
        cols[idx:idx + length] = j
        vals[idx:idx + length] = a[:length]
        idx += length
    T = coo_matrix((vals[:idx], (rows[:idx], cols[:idx])), shape=(2 * n - 1, n)).tocsr()
    return T


def solve_lp(a, M, time_limit=10.0):
    """
    Solve LP: max sum(g) s.t. T(a)@g <= M, g >= 0.
    Returns (g, active_constraint_indices, status).
    """
    n = len(a)
    T = build_toeplitz(a)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n

    try:
        res = linprog(c, A_ub=T, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': time_limit,
                               'dual_feasibility_tolerance': 1e-10,
                               'primal_feasibility_tolerance': 1e-10})
        if res.status == 0:
            g = np.maximum(res.x, 0)
            # Get marginal values to identify active constraints
            marginals = res.ineqlin.marginals if res.ineqlin.marginals is not None else np.zeros(2 * n - 1)
            active = np.where(np.abs(marginals) > 1e-8)[0]
            return g, active, 0
        elif res.status == 3 and res.x is not None:
            g = np.maximum(res.x, 0)
            return g, np.array([], dtype=int), 3
    except Exception:
        pass
    return None, np.array([], dtype=int), -1


def subgradient_descent(a, time_budget, topk=10, base_lr=0.15, momentum=0.85):
    """Momentum subgradient descent with adaptive step size."""
    n = len(a)
    a = a.copy().astype(np.float64)
    S = np.sum(a)
    if S < 1e-10:
        return a
    a = a / S

    v = np.zeros(n)
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    stall = 0

    for it in range(1000000):
        if time.monotonic() - start > time_budget:
            break

        conv = compute_conv(a)
        M = float(np.max(conv))

        # Top-K active
        if topk >= len(conv):
            active = np.arange(len(conv))
        else:
            active = np.argsort(conv)[-topk:]

        # Gradient computation using Toeplitz transpose
        T = build_toeplitz(a)
        active_vec = np.zeros(2 * n - 1)
        active_vec[active] = 1.0 / len(active)
        grad = T.T @ (2.0 * active_vec)

        # Center gradient to preserve sum
        grad_centered = grad - np.mean(grad)

        v = momentum * v + (1.0 - momentum) * grad_centered
        step = base_lr / (1.0 + 0.005 * it)

        a_new = a - step * v
        a_new = np.maximum(a_new, 0)
        S_new = np.sum(a_new)
        if S_new < 1e-10:
            break
        a_new = a_new / S_new

        obj = compute_obj(a_new)
        if obj < best_obj - 1e-14:
            best_obj = obj
            best_a = a_new.copy()
            a = a_new
            stall = 0
        else:
            stall += 1
            if stall > 80:
                rng = np.random.default_rng(it * 31 + 7)
                a = a * np.exp(rng.normal(0, 0.06, n))
                a = np.maximum(a, 0)
                S = np.sum(a)
                if S > 1e-10:
                    a = a / S
                v = np.zeros(n)
                stall = 0
                base_lr *= 0.8
                if base_lr < 1e-9:
                    break

    return best_a


def golden_section_line_search(a, g, best_obj, max_evals=8):
    """Golden section search for optimal t in (1-t)*a + t*g."""
    n = len(a)
    gr = (np.sqrt(5.0) - 1.0) / 2.0  # golden ratio ~0.618

    lo = 0.0
    hi = 1.0
    c = hi - gr * (hi - lo)
    d = lo + gr * (hi - lo)

    def obj_at(t):
        if t <= 0:
            return best_obj
        a_t = (1.0 - t) * a + t * g
        a_t = np.maximum(a_t, 0)
        S = np.sum(a_t)
        if S < 1e-10:
            return best_obj
        a_t = a_t / S
        return compute_obj(a_t)

    fc = obj_at(c)
    fd = obj_at(d)
    evals = 2

    for _ in range(max_evals - 2):
        if fc < fd:
            hi = d
            d = c
            fd = fc
            c = hi - gr * (hi - lo)
            fc = obj_at(c)
        else:
            lo = c
            c = d
            fc = fd
            d = lo + gr * (hi - lo)
            fd = obj_at(d)
        evals += 1

    best_t = (c + d) / 2.0
    best_obj_ls = min(fc, fd)
    # Also check endpoints
    for t_check in [0.1, 0.5, 1.0]:
        oc = obj_at(t_check)
        if oc < best_obj_ls:
            best_obj_ls = oc
            best_t = t_check

    return best_t, best_obj_ls


def anti_peak_perturbation(a, conv, num_perts=4, base_scale=0.08):
    """Create perturbations targeting peak reduction."""
    n = len(a)
    results = []
    peak_indices = np.argsort(conv)[-min(15, 2 * n - 1):]

    for p in range(num_perts):
        scale = base_scale * (1.0 + 0.5 * p)
        # Compute which a[i] contribute most to peaks
        T = build_toeplitz(a)
        peak_vec = np.zeros(2 * n - 1)
        peak_vec[peak_indices] = 1.0 / len(peak_indices)
        contrib = T.T @ peak_vec  # contribution of each a[i] to peaks

        # Normalize contribution
        max_c = np.max(contrib)
        if max_c > 1e-10:
            contrib = contrib / max_c

        # Reduce high-contribution elements, boost low-contribution
        a_pert = a * (1.0 - scale * contrib + scale * np.mean(contrib))
        a_pert = np.maximum(a_pert, 1e-15)
        S = np.sum(a_pert)
        if S > 1e-10:
            results.append(a_pert / S)

    return results


def alternating_lp_refine(a, time_budget, max_iters=60, verbose=False):
    """
    Core optimization: alternating LP with adaptive M, line search, and perturbation escapes.
    """
    n = len(a)
    a = a.copy().astype(np.float64)
    S = np.sum(a)
    if S < 1e-10:
        return a
    a = a / S

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    stall = 0

    for it in range(max_iters):
        if time.monotonic() - start > time_budget:
            break

        remaining = time_budget - (time.monotonic() - start)
        if remaining < 2.0:
            break

        conv = compute_conv(a)
        M = float(np.max(conv))

        # Try LP with current M and slightly relaxed M
        lp_time = min(remaining * 0.25, 12.0)
        g, active, status = solve_lp(a, M, time_limit=lp_time)

        if g is None:
            # Fallback to subgradient
            rem2 = time_budget - (time.monotonic() - start)
            if rem2 > 3:
                a_sg = subgradient_descent(a, time_budget=rem2 * 0.3, topk=10)
                obj_sg = compute_obj(a_sg)
                if obj_sg < best_obj:
                    best_obj = obj_sg
                    best_a = a_sg.copy()
                    a = a_sg.copy()
            continue

        S_g = np.sum(g)
        if S_g < 1e-10:
            break
        g = g / S_g

        # Golden section line search
        best_t, best_obj_ls = golden_section_line_search(a, g, best_obj, max_evals=6)

        if best_t > 0.01 and best_obj_ls < best_obj:
            a = (1.0 - best_t) * a + best_t * g
            a = np.maximum(a, 0)
            a = a / np.sum(a)
            best_obj = best_obj_ls
            best_a = a.copy()
            stall = 0
            if verbose:
                print(f"    LP it={it}: obj={best_obj:.8f}, t={best_t:.3f}, |active|={len(active)}")
        else:
            stall += 1
            if stall > 8:
                # Try relaxed M
                M_relax = M * 1.005
                g2, _, status2 = solve_lp(a, M_relax, time_limit=5.0)
                if g2 is not None and np.sum(g2) > 1e-10:
                    g2 = g2 / np.sum(g2)
                    a_try = 0.7 * a + 0.3 * g2
                    a_try = np.maximum(a_try, 0)
                    a_try = a_try / np.sum(a_try)
                    obj_try = compute_obj(a_try)
                    if obj_try < best_obj:
                        best_obj = obj_try
                        best_a = a_try.copy()
                        a = a_try.copy()
                        stall = 0
                elif stall > 15:
                    # Anti-peak perturbation
                    perts = anti_peak_perturbation(a, conv, num_perts=3, base_scale=0.05)
                    improved = False
                    for a_p in perts:
                        rem3 = time_budget - (time.monotonic() - start)
                        if rem3 < 3:
                            break
                        a_p2 = subgradient_descent(a_p, time_budget=min(4.0, rem3 * 0.3), topk=8)
                        obj_p = compute_obj(a_p2)
                        if obj_p < best_obj:
                            best_obj = obj_p
                            best_a = a_p2.copy()
                            a = a_p2.copy()
                            improved = True
                            stall = 0
                            break
                    if not improved and stall > 25:
                        break

    return best_a


def smooth_init(n, seed=0, mode='cosine'):
    """Generate diverse smooth initializations."""
    rng = np.random.default_rng(seed)
    k = np.arange(n, dtype=np.float64)

    if mode == 'cosine':
        a = np.ones(n) * 0.8
        for m in range(1, 12):
            c = rng.uniform(-0.4, 0.4) / m
            phase = rng.uniform(0, 2 * np.pi)
            a += c * np.cos(2 * np.pi * m * k / n + phase)
    elif mode == 'gaussian':
        center = n * rng.uniform(0.25, 0.75)
        sigma = n * rng.uniform(0.15, 0.45)
        a = np.exp(-0.5 * ((k - center) / sigma) ** 2)
        a += 0.05 * np.ones(n)
    elif mode == 'arch':
        x = 2.0 * k / (n - 1) - 1.0
        a = np.sqrt(np.maximum(1.0 - x ** 2, 0.01))
    elif mode == 'bump':
        # Two bumps
        c1 = n * rng.uniform(0.2, 0.4)
        c2 = n * rng.uniform(0.6, 0.8)
        s = n * 0.1
        a = np.exp(-0.5 * ((k - c1) / s) ** 2) + np.exp(-0.5 * ((k - c2) / s) ** 2)
        a += 0.03 * np.ones(n)
    elif mode == 'sawtooth':
        a = np.abs(2.0 * (k / n - 0.5))
        a = a ** rng.uniform(0.5, 3.0)
        a += 0.05 * np.ones(n)
    else:
        a = rng.dirichlet(np.ones(n)) * n
        return a

    a = np.maximum(a, 0)
    S = np.sum(a)
    if S < 1e-10:
        return np.ones(n) / n
    return a / S


def multiscale_optimize(a, time_budget):
    """Multi-scale: optimize coarser versions first, then refine."""
    n = len(a)
    if n < 50:
        return alternating_lp_refine(a, time_budget)

    start = time.monotonic()
    current = a.copy()

    scales = [max(20, n // 10), max(40, n // 5), max(80, n // 2)]
    scales = [s for s in scales if s < n]

    for target_n in scales:
        if time.monotonic() - start > time_budget:
            break

        remaining = time_budget - (time.monotonic() - start)
        if remaining < 5:
            break

        # Downsample
        idx = np.round(np.arange(target_n) * n / target_n).astype(int).clip(0, n - 1)
        coarse = current[idx]
        S_c = np.sum(coarse)
        if S_c > 1e-10:
            coarse = coarse / S_c

        alloc = remaining * 0.35
        coarse_opt = alternating_lp_refine(coarse, time_budget=alloc)
        obj_coarse = compute_obj(coarse_opt)

        # Upsample
        x_old = np.linspace(0, 1, target_n)
        x_new = np.linspace(0, 1, n)
        try:
            f = interp1d(x_old, coarse_opt, kind='cubic', fill_value='extrapolate')
            upsampled = np.maximum(f(x_new), 0)
        except:
            upsampled = current

        S = np.sum(upsampled)
        if S > 1e-10:
            upsampled = upsampled / S
            current = upsampled

    # Final full-scale refinement
    remaining = time_budget - (time.monotonic() - start)
    if remaining > 3:
        current = alternating_lp_refine(current, time_budget=remaining * 0.8)

    return current


def propose_candidate(seed=42, budget_s=1500):
    """
    Main search function combining:
    1. Diverse initialization with quick subgradient warm-up
    2. Multi-scale optimization for promising candidates
    3. Intensive alternating LP refinement
    4. Anti-peak targeted perturbations for escape
    5. Final high-precision polish
    """
    start_time = time.monotonic()
    deadline = start_time + float(budget_s) - 30.0

    best_seq = None
    best_score = np.inf

    # Load parent
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64)
        parent = np.clip(parent, 0, 1000.0)
        if parent.size > 0 and np.isfinite(parent).all() and np.sum(parent) > 0.01:
            parent = parent / np.sum(parent)
            obj = compute_obj(parent)
            print(f"Parent (n={parent.size}): obj={obj:.8f}")
            best_score = obj
            best_seq = parent.copy()
    except:
        pass

    if best_seq is None:
        best_seq = np.ones(400) / 400
        best_score = compute_obj(best_seq)
        print(f"Default init: obj={best_score:.8f}")

    n_parent = len(best_seq)
    print(f"Starting best: {best_score:.8f} (n={n_parent})")

    # === Phase 1: Diverse exploration ===
    print("\n=== Phase 1: Diverse exploration ===")

    n_list = sorted(set([
        max(100, n_parent - 150),
        max(150, n_parent - 50),
        n_parent,
        min(800, n_parent + 100),
        min(1000, n_parent + 250)
    ]))

    candidates = []

    for n in n_list:
        if time.monotonic() > deadline - 120:
            break

        modes = ['cosine', 'gaussian', 'arch', 'bump', 'sawtooth']
        for mode in modes:
            if time.monotonic() > deadline - 60:
                break

            s = seed + n * 7 + (hash(mode) % 997)
            a_init = smooth_init(n, seed=s, mode=mode)
            obj_init = compute_obj(a_init)

            # Quick subgradient warm-up
            a_quick = subgradient_descent(a_init, time_budget=4.0, topk=8, base_lr=0.12)
            obj_quick = compute_obj(a_quick)

            if obj_quick < obj_init:
                a_best_local = a_quick
                obj_best_local = obj_quick
            else:
                a_best_local = a_init
                obj_best_local = obj_init

            candidates.append((a_best_local, obj_best_local, f"n{n}_{mode}"))

            if obj_best_local < best_score:
                best_score = obj_best_local
                best_seq = a_best_local.copy()
                print(f"  {f'n{n}_{mode}'}: {obj_best_local:.8f} ** NEW BEST **")

    # Parent perturbations
    if best_seq is not None and time.monotonic() < deadline - 60:
        n_p = len(best_seq)
        for scale in [0.03, 0.06, 0.12, 0.2]:
            if time.monotonic() > deadline - 30:
                break
            rng = np.random.default_rng(seed + int(scale * 10000))
            a_pert = best_seq * np.exp(rng.normal(0, scale, n_p))
            a_pert = np.maximum(a_pert, 0)
            S = np.sum(a_pert)
            if S < 1e-10:
                continue
            a_pert = a_pert / S
            a_quick = subgradient_descent(a_pert, time_budget=4.0, topk=10)
            obj_quick = compute_obj(a_quick)
            if obj_quick < best_score:
                best_score = obj_quick
                best_seq = a_quick.copy()
                print(f"  Parent pert({scale}): {obj_quick:.8f} ** NEW BEST **")

    # === Phase 2: Multi-scale on top candidates ===
    print(f"\n=== Phase 2: Multi-scale (best={best_score:.8f}) ===")

    candidates.sort(key=lambda x: x[1])

    for idx in range(min(4, len(candidates))):
        if time.monotonic() > deadline - 90:
            break

        a_cand, obj_cand, name = candidates[idx]
        if obj_cand > best_score + 0.02:
            break

        remaining = deadline - time.monotonic()
        alloc = min(remaining * 0.2, 50.0)
        n_cand = len(a_cand)
        print(f"  Multi-scale {name} (n={n_cand}, obj={obj_cand:.8f}), alloc={alloc:.0f}s")

        a_ms = multiscale_optimize(a_cand, time_budget=alloc)
        obj_ms = compute_obj(a_ms)

        if obj_ms < best_score:
            best_score = obj_ms
            best_seq = a_ms.copy()
            print(f"    ** NEW BEST: n={n_cand}, obj={obj_ms:.8f} **")

    # === Phase 3: Intensive alternating LP (main workhorse) ===
    print(f"\n=== Phase 3: Intensive LP (best={best_score:.8f}) ===")

    a_cur = best_seq.copy()

    for round_num in range(25):
        if time.monotonic() > deadline - 40:
            break

        remaining = deadline - time.monotonic()
        if remaining < 15:
            break

        alloc = min(remaining * 0.18, 40.0)
        a_new = alternating_lp_refine(a_cur, time_budget=alloc, max_iters=50)
        obj_new = compute_obj(a_new)

        if obj_new < best_score - 1e-12:
            best_score = obj_new
            best_seq = a_new.copy()
            a_cur = a_new.copy()
            print(f"  LP R{round_num}: {best_score:.8f} ** NEW BEST **")
        else:
            print(f"  LP R{round_num}: {obj_new:.8f} (best={best_score:.8f})")

            # Try subgradient escape
            if remaining > 20:
                a_sg = subgradient_descent(a_cur, time_budget=min(8.0, remaining * 0.12), topk=12)
                obj_sg = compute_obj(a_sg)
                if obj_sg < best_score:
                    best_score = obj_sg
                    best_seq = a_sg.copy()
                    a_cur = a_sg.copy()
                    print(f"    SG escape: {obj_sg:.8f} ** NEW BEST **")

    # === Phase 4: Anti-peak targeted escape ===
    print(f"\n=== Phase 4: Anti-peak escape (best={best_score:.8f}) ===")

    a_cur = best_seq.copy()

    for esc in range(15):
        if time.monotonic() > deadline - 20:
            break

        remaining = deadline - time.monotonic()
        if remaining < 8:
            break

        conv = compute_conv(a_cur)
        perts = anti_peak_perturbation(a_cur, conv, num_perts=4, base_scale=0.04)

        for a_p in perts:
            if time.monotonic() > deadline - 10:
                break
            alloc = min(remaining * 0.2, 18.0)
            a_ref = alternating_lp_refine(a_p, time_budget=alloc, max_iters=30)
            obj_ref = compute_obj(a_ref)

            if obj_ref < best_score:
                best_score = obj_ref
                best_seq = a_ref.copy()
                a_cur = a_ref.copy()
                print(f"  Escape {esc}: {obj_ref:.8f} ** NEW BEST **")
                break

        # Also try random perturbation + refine
        if time.monotonic() < deadline - 10:
            rng = np.random.default_rng(seed + esc * 7919)
            a_rand = a_cur * np.exp(rng.normal(0, 0.05, len(a_cur)))
            a_rand = np.maximum(a_rand, 0)
            a_rand = a_rand / np.sum(a_rand)
            alloc = min(remaining * 0.15, 15.0)
            a_r = alternating_lp_refine(a_rand, time_budget=alloc, max_iters=25)
            obj_r = compute_obj(a_r)
            if obj_r < best_score:
                best_score = obj_r
                best_seq = a_r.copy()
                a_cur = a_r.copy()
                print(f"  Rand {esc}: {obj_r:.8f} ** NEW BEST **")

    # === Phase 5: Final polish ===
    print(f"\n=== Phase 5: Final polish (best={best_score:.8f}) ===")

    remaining = deadline - time.monotonic()
    if remaining > 10:
        # Long subgradient with small steps
        a_sg = subgradient_descent(best_seq, time_budget=remaining * 0.4, topk=15, base_lr=0.05, momentum=0.9)
        obj_sg = compute_obj(a_sg)
        if obj_sg < best_score:
            best_score = obj_sg
            best_seq = a_sg.copy()
            print(f"  Final SG: {obj_sg:.8f} ** NEW BEST **")

    remaining = deadline - time.monotonic()
    if remaining > 8:
        # Final LP with more iterations
        a_lp = alternating_lp_refine(best_seq, time_budget=remaining * 0.6, max_iters=40)
        obj_lp = compute_obj(a_lp)
        if obj_lp < best_score:
            best_score = obj_lp
            best_seq = a_lp.copy()
            print(f"  Final LP: {obj_lp:.8f} ** NEW BEST **")

    # Validate and return
    if best_seq is None:
        best_seq = np.ones(300) / 300
        best_score = compute_obj(best_seq)

    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    val = evaluate_sequence(result)
    if val < np.inf and val < best_score:
        best_score = val
    elif val == np.inf:
        result = [1.0] * 300
        best_score = evaluate_sequence(result)

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_score:.8f}, time={elapsed:.1f}s ===")

    return result
