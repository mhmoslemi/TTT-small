import numpy as np
from scipy.optimize import linprog
from scipy.sparse import csr_matrix, lil_matrix

def fast_conv(a):
    """Full self-convolution via FFT."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]


def cross_conv(a, g):
    """Cross-convolution a*g via FFT."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    fg = np.fft.rfft(g, size)
    return np.fft.irfft(fa * fg, size)[:2 * n - 1]


def compute_J(a):
    """Compute J(a) = 2*n*max(conv(a,a))/(sum(a))^2."""
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))


def build_convolution_matrix(n, a):
    """Build sparse matrix A where (A@g)[k] = (a*g)[k] for k=0..2n-2."""
    # For each output position k, (a*g)[k] = sum_j a[k-j]*g[j]
    # where j ranges over max(0, k-n+1) to min(n-1, k)
    rows = []
    cols = []
    vals = []
    
    # More efficient: for each j (column), a[j] contributes to positions k=j, j+1, ..., j+n-1
    # with values a[j], a[j-1], ..., a[0] (for k >= j)
    # and a[0], a[1], ..., a[k-j] (for k < j) -- wait, let me think again.
    
    # (a*g)[k] = sum_{j} a[k-j] * g[j] where 0 <= k-j < n and 0 <= j < n
    # So for fixed j: a[k-j] * g[j] appears in row k for k = j, j+1, ..., j+n-1
    # i.e., for k-j = 0, 1, ..., n-1, so a[k-j] = a[0], a[1], ..., a[n-1]
    
    # Wait: k-j ranges from 0 to n-1, so k = j+0, j+1, ..., j+n-1
    # And a[k-j] = a[0], a[1], ..., a[n-1]

            rows.append(k)
            cols.append(j)
            vals.append(a[m])
    
    return csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))


def lp_best_response_fast(a, M):
    """Solve LP: max sum(g) s.t. A@g <= M*1, g >= 0. Returns g or None."""
    n = len(a)
    A = build_convolution_matrix(n, a)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    try:
        r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
        if r.success and r.x is not None:
            g = np.maximum(r.x, 0)
            if g.sum() > 1e-15:
                return g
    except Exception:
        pass
    return None

def directed_peak_reduction(a, deadline, max_iters=500):

            continue

def resample_seq(src, dest_n):

"""
Optimized hybrid optimizer for minimizing J(a) = 2*n*max(conv(a,a))/sum(a)^2.
Uses: (1) LP mirror descent with cached matrix structure and adaptive step sizes,
(2) wide length sweep, (3) peak-targeted directed perturbations,
(4) subgradient refinement, (5) multi-restart with diverse perturbations.

import numpy as np
from scipy.optimize import linprog
from scipy.sparse import csr_matrix
import time


def fast_conv(a):
    """Full self-convolution via FFT."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]


def compute_J(a):
    """Compute J(a) = 2*n*max(conv(a,a))/(sum(a))^2."""
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))


def build_conv_structure(n):
    """Pre-compute row/col indices for convolution matrix."""
    rows = []
    cols = []
    # Column j has entries a[0..n-1] at rows j, j+1, ..., j+n-1
    for j in range(n):
        rows.extend(range(j, j + n))
        cols.extend([j] * n)
    return np.array(rows, dtype=np.int32), np.array(cols, dtype=np.int32)


def make_conv_matrix(a, rows, cols):
    """Build sparse convolution matrix with pre-computed structure."""
    n = len(a)
    vals = np.tile(a, n)
    return csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))


def lp_best_response(a, M, rows, cols):
    """Solve LP: max sum(g) s.t. A@g <= M, g >= 0."""
    n = len(a)
    A = make_conv_matrix(a, rows, cols)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    try:
        r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
        if r.success and r.x is not None:
            g = np.maximum(r.x, 0)
            if g.sum() > 1e-15:
                return g
    except Exception:
        pass
    return None

def subgradient_polish(a, deadline, max_iters=2000):
    """Subgradient descent with momentum for final polish."""
    a = np.maximum(a.astype(np.float64), 1e-12)
    a = a / a.sum()
    n = len(a)
    best_J = compute_J(a)
    best_a = a.copy()
    momentum = np.zeros(n)
    beta = 0.9
    
    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        
        b = fast_conv(a)
        M = b.max()
        tol = 1e-10 * M
        active = np.where(b >= M - tol)[0]
        
        grad = np.zeros(n)
        for k in active:
            lo = max(0, k - n + 1)
            hi = min(n - 1, k)
            for i in range(lo, hi + 1):
                grad[i] += 2.0 * a[k - i]
        
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-15:
            break
        
        t_step = 2.0 / (gnorm * np.sqrt(it + 1.0))
        momentum = beta * momentum + (1 - beta) * grad / gnorm
        a_new = a - t_step * momentum
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s < 1e-10:
            break
        a_new = a_new / s
        
        J_new = compute_J(a_new)
        if J_new < best_J:
            best_J = J_new
            best_a = a_new.copy()
        a = a_new
    
    return best_a, best_J


def peak_reduction(a, deadline, max_iters=3000, rng=None):
    """Targeted peak reduction by moving mass from peak contributors."""
    if rng is None:
        rng = np.random.default_rng(123)
    a = np.maximum(a.astype(np.float64), 1e-12)
    a = a / a.sum()
    n = len(a)
    best_J = compute_J(a)
    best_a = a.copy()
    
    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        
        b = fast_conv(a)
        M = b.max()
        peak_idx = int(np.argmax(b))
        
        j_min = max(0, peak_idx - n + 1)
        j_max = min(n - 1, peak_idx)
        
        # Find top contributor
        max_prod = 0
        max_j1 = -1
        max_j2 = -1
        for j in range(j_min, j_max + 1):
            prod = a[j] * a[peak_idx - j]
            if prod > max_prod:
                max_prod = prod
                max_j1 = j
                max_j2 = peak_idx - j
        
        if max_j1 < 0:
            break
        
        src = max_j1 if a[max_j1] >= a[max_j2] else max_j2
        dst = int(rng.integers(0, n))
        
        if src == dst or a[src] < 1e-8:
            continue
        
        best_delta = 0
        best_trial_J = best_J
        best_trial_a = None
        
        for frac in [0.005, 0.01, 0.02, 0.05, 0.1, 0.2]:
            delta = frac * a[src]
            a_trial = a.copy()
            a_trial[src] -= delta
            a_trial[dst] += delta
            if a_trial[src] < -1e-10 or a_trial[dst] < -1e-10:
                continue
            a_trial = np.maximum(a_trial, 0)
            s = a_trial.sum()
            if s < 1e-10:
                continue
            a_trial = a_trial / s
            J_trial = compute_J(a_trial)
            if J_trial < best_trial_J:
                best_trial_J = J_trial
                best_delta = delta
                best_trial_a = a_trial.copy()
        
        if best_trial_a is not None:
            a = best_trial_a
            best_J = best_trial_J
            best_a = a.copy()
    
    return best_a, best_J

    """Resample 1D sequence to new length."""
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)


def update_tracker(tracker, a):
    """Update the score tracker."""
    a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
    if a.size == 0 or a.sum() < 0.01:
        return
    score = float(evaluate_sequence(a.tolist()))
    if np.isfinite(score) and score < tracker['best_score']:
        tracker['best'] = a.copy()
        tracker['best_score'] = score
        elapsed = time.monotonic() - tracker['t_start']
        print(f"  [NEW BEST] J={score:.8f} n={a.size} t={elapsed:.0f}s", flush=True)
