"""
Minimizes J(a) = 2*n*max(convolve(a,a))/(sum(a))^2 using:
1. Fast LP mirror descent with precomputed constraint structure
2. Efficient coordinate descent with incremental convolution updates
3. Adaptive step size and tightening parameters

import numpy as np
from scipy.optimize import linprog
import time


def fast_conv(a):
    """Full self-convolution via FFT."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]


def compute_J(a):
    """Compute J(a) = 2*n*max(conv(a,a))/(sum(a))^2."""
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))


class LP_Solver:
    """Precomputed LP solver for the best-response problem."""
    
    def __init__(self, n):
        self.n = n
        # Precompute the structure of the constraint matrix
        # A[k, j] = a[k-j] if 0 <= k-j < n
        # For each row k, the non-zero columns are j in [max(0,k-n+1), min(n-1,k)]
        # and A[k,j] = a[k-j]
        
        # Store as lists of (col, a_index) for each row
        self.row_data = []
        for k in range(2 * n - 1):
            j_start = max(0, k - n + 1)
            j_end = min(n - 1, k)
            # A[k,j] = a[k-j]
            data = []
            for j in range(j_start, j_end + 1):
                data.append((j, k - j))  # (column_index, index_into_a)
            self.row_data.append(data)
        
        # Precompute the dense matrix structure (which columns are non-zero per row)
        self.row_cols = []
        self.row_a_idx = []
        for k in range(2 * n - 1):
            j_start = max(0, k - n + 1)
            j_end = min(n - 1, k)
            cols = list(range(j_start, j_end + 1))
            a_idx = [k - j for j in cols]
            self.row_cols.append(cols)
            self.row_a_idx.append(a_idx)
    
    def build_matrix(self, a):
        """Build the dense constraint matrix."""
        n = self.n
        A = np.zeros((2 * n - 1, n))
        for k in range(2 * n - 1):
            for idx, j in enumerate(self.row_cols[k]):
                A[k, j] = a[self.row_a_idx[k][idx]]
        return A
    
    def solve(self, a, M):
        """Solve: max sum(g) s.t. A(a) @ g <= M, g >= 0."""
        A = self.build_matrix(a)
        c = -np.ones(self.n)
        b_ub = np.full(2 * self.n - 1, M)
        bounds = [(0.0, None)] * self.n
        result = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
        if result.success and result.x is not None:
            g = np.maximum(result.x, 0.0)
            if g.sum() > 1e-10:
                return g
        return None


def lp_mirror_descent(a, deadline, tracker, max_iters=500, eps0=0.005):
    """LP-based mirror descent optimization."""
    a = np.maximum(a.astype(np.float64), 1e-6)
    a = a / a.sum()
    n = len(a)
    
    solver = LP_Solver(n)
    current_J = compute_J(a)
    tracker.update(a)
    
    eps = eps0
    stall_count = 0
    prev_M = None
    
    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        
        b = fast_conv(a)
        M = float(b.max())
        
        if prev_M is not None and abs(M - prev_M) < 1e-10:
            stall_count += 1
        else:
            stall_count = max(0, stall_count - 1)
        
        prev_M = M
        
        # Try to tighten
        M_target = M * (1.0 - eps)
        g = solver.solve(a, M_target)
        
        if g is None:
            # Back off
            eps *= 0.5
            if eps < 0.0005:
                break
            continue
        
        g = np.maximum(g, 0.0)
        g_sum = g.sum()
        if g_sum < 1e-10:
            break
        g = g / g_sum
        
        # Line search
        best_t = 0.0
        best_J = current_J
        for t in [1.0, 0.7, 0.5, 0.3, 0.2, 0.1, 0.05]:
            a_trial = (1.0 - t) * a + t * g
            a_trial = np.maximum(a_trial, 0.0)
            s = a_trial.sum()
            if s < 1e-10:
                continue
            a_trial = a_trial / s
            J_trial = compute_J(a_trial)
            if J_trial < best_J - 1e-14:
                best_J = J_trial
                best_t = t
        
        if best_t == 0.0:
            # No improvement, try even smaller steps
            for t in [0.03, 0.02, 0.01]:
                a_trial = (1.0 - t) * a + t * g
                a_trial = np.maximum(a_trial, 0.0)
                s = a_trial.sum()
                if s < 1e-10:
                    continue
                a_trial = a_trial / s
                J_trial = compute_J(a_trial)
                if J_trial < best_J - 1e-14:
                    best_J = J_trial
                    best_t = t
        
        if best_t == 0.0:
            stall_count += 1
            if stall_count > 30:
                break
            eps *= 0.7
            continue
        
        a = (1.0 - best_t) * a + best_t * g
        a = np.maximum(a, 0.0)
        a = a / a.sum()
        current_J = best_J
        stall_count = 0
        
        # Adapt eps
        if best_t > 0.3:
            eps = min(0.02, eps * 1.1)  # Can tighten more
        elif best_t < 0.1:
            eps = max(0.001, eps * 0.8)  # Need to loosen
        
        if it % 50 == 0:
            tracker.update(a)
    
    tracker.update(a)
    return a


def coordinate_descent_pass(a, deadline=None):
    """One pass of coordinate descent with efficient updates."""
    a = np.maximum(a.astype(np.float64), 0.0)
    S = a.sum()
    if S < 1e-10:
        return a
    n = len(a)
    b = fast_conv(a)
    current_J = 2.0 * n * b.max() / (S * S)
    
    indices = np.random.permutation(n)
    
    for i in indices:
        if deadline is not None and time.monotonic() > deadline:
            break
        
        old_val = a[i]
        
        # Find optimal value for a[i] using golden section search
        # The affected convolution values are for k in [i, i+n-1] (clamped to [0, 2n-2])
        
        def obj_at_x(x):
            new_S = S - old_val + x
            if new_S < 1e-10:
                return 1e10
            max_c = 0.0
            # Only need to check affected lags
            k_start = max(0, i)
            k_end = min(2 * n - 2, i + n - 1)
            for k in range(k_start, k_end + 1):
                ki = k - i  # offset into a
                if 0 <= ki < n:
                    if k == 2 * i:
                        ck = b[k] - old_val * old_val + x * x
                    else:
                        ck = b[k] + 2.0 * (x - old_val) * a[ki]
                else:
                    ck = b[k]
                if ck > max_c:
                    max_c = ck
            # Also need to check unaffected lags (take the max of those)
            # Pre-compute max of unaffected lags
            # Actually, for efficiency, let's just check all lags
            # The unaffected lags have the same b[k] value
            # So max_c should also consider the global max of unaffected b[k]
            # We can precompute this, but for now let's just do it inline
            
            return 2.0 * n * max_c / (new_S * new_S)
        
        # Golden section search
        gr = (np.sqrt(5.0) - 1.0) / 2.0
        lo = 0.0
        hi = max(3.0 * old_val if old_val > 0 else 0.1, S / n * 2.0)
        
        c_pt = hi - gr * (hi - lo)
        d_pt = lo + gr * (hi - lo)
        fc = obj_at_x(c_pt)
        fd = obj_at_x(d_pt)
        
        for _ in range(30):
            if fc < fd:
                hi = d_pt
                d_pt = c_pt
                fd = fc
                c_pt = hi - gr * (hi - lo)
                fc = obj_at_x(c_pt)
            else:
                lo = c_pt
                c_pt = d_pt
                fc = fd
                d_pt = lo + gr * (hi - lo)
                fd = obj_at_x(d_pt)
        
        x_opt = max(0.0, (lo + hi) / 2.0)
        
        if abs(x_opt - old_val) > 1e-12:
            # Update a and b
            a[i] = x_opt
            S = S - old_val + x_opt
            for k in range(max(0, i), min(2 * n - 1, i + n)):
                ki = k - i
                if 0 <= ki < n:
                    if k == 2 * i:
                        b[k] = b[k] - old_val * old_val + x_opt * x_opt
                    else:
                        b[k] = b[k] + 2.0 * (x_opt - old_val) * a[ki]
    
    # Verify
    b_check = fast_conv(a)
    new_J = 2.0 * n * b_check.max() / (S * S)
    return a, new_J
