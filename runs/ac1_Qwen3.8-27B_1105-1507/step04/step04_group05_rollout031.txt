import numpy as np
from scipy.optimize import linprog
from scipy.sparse import lil_matrix, csr_matrix
import time


def fast_conv(a):
    """Full self-convolution via FFT."""
    n = len(a)
    s = 1
    while s < 2 * n - 1:
        s <<= 1
    fa = np.fft.rfft(a, s)
    return np.fft.irfft(fa * fa, s)[:2 * n - 1]


def compute_J(a):
    """Compute J(a) = 2*n*max(conv(a,a))/(sum(a))^2."""
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))


def build_A_sparse(a):
    """Build sparse constraint matrix for LP: (a*g)[k] <= M for all k."""
    n = len(a)
    m = 2 * n - 1
    A = lil_matrix((m, n))
    for k in range(m):
        jl = max(0, k - n + 1)
        jh = min(n - 1, k)
        if jl <= jh:
            seg = a[k - jh:k - jl + 1][::-1]
            for jj in range(jl, jh + 1):
                A[k, jj] = seg[jj - jl]
    return A.tocsr()


def lp_best_response(a, M):
    """LP: max sum(g) s.t. (a*g)_k <= M for all k, g >= 0."""
    n = len(a)
    c_obj = -np.ones(n)
    A = build_A_sparse(a)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    r = linprog(c_obj, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
    if r.success and r.x is not None:
        g = np.maximum(r.x, 0.0)
        if g.sum() > 1e-10:
            return g
    return None

    """LP best-response iterations."""
    a = a.astype(np.float64).copy()
    a = np.maximum(a, 1e-6)
    a /= a.sum()
    n = len(a)
    current_J = compute_J(a)
    tracker.update(a)
    
    tighten = tighten_base
    stall = 0
    last_print = 0
    
    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        
        c = fast_conv(a)
        M = float(c.max())
        
        for attempt in range(3):
            M_target = M * (1.0 - tighten * (1.0 - attempt * 0.3))
            g = lp_best_response(a, M_target)
            if g is not None:
                break
        
        if g is None:
            tighten *= 0.5
            stall += 1
            if stall > 30:
                break
            continue
        
        g = np.maximum(g, 0.0)
        g /= g.sum()
        
        # Try multiple step sizes
        progress = min(1.0, it / max(1, max_iters))
        base_t = max(0.02, 0.4 * (1.0 - 0.5 * progress))
        candidates = [base_t, base_t * 0.5, base_t * 0.75, base_t * 1.5, 0.1]
        
        best_a = None
        best_J = current_J
        
        for t in candidates:
            a_trial = (1.0 - t) * a + t * g
            a_trial = np.maximum(a_trial, 0.0)
            s = a_trial.sum()
            if s < 1e-10:
                continue
            a_trial /= s
            J_trial = compute_J(a_trial)
            if J_trial < best_J - 1e-14:
                best_J = J_trial
                best_a = a_trial
        
        if best_a is not None:
            a = best_a
            current_J = best_J
            stall = 0
            tighten = min(0.03, tighten * 1.02)
            if (time.monotonic() - last_print) > 5.0:
                print(f"    LP iter {it}: J={current_J:.8f}, t={tighten:.5f}", flush=True)
                last_print = time.monotonic()
                tracker.update(a)
        else:
            stall += 1
            tighten *= 0.6
            if stall > 40:
                break
    
    tracker.update(a)
    return a


def coordinate_update(a, i, b, S):
    """Golden section search for optimal value of a[i]."""
    n = len(a)
    ai = a[i]
    
    def obj(x):
        new_S = S - ai + x
        if new_S < 1e-10:
            return 1e10
        max_c = 0.0
        for k in range(max(0, i), min(2 * n - 1, i + n)):
            ki = k - i
            if 0 <= ki < n:
                if k == 2 * i:
                    ck = b[k] - ai * ai + x * x
                else:
                    ck = b[k] - 2.0 * ai * a[ki] + 2.0 * x * a[ki]
            else:
                ck = b[k]
            if ck > max_c:
                max_c = ck
        # Need max over all k, not just affected ones
        unaffected_max = 0.0
        for k in range(2 * n - 1):
            if k < i or k >= i + n:
                if b[k] > unaffected_max:
                    unaffected_max = b[k]
        return 2.0 * n * max(max_c, unaffected_max) / (new_S * new_S)
    
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    lo = 0.0
    hi = max(3.0 * ai, 3.0 * S / n, 0.01)
    
    c_pt = hi - gr * (hi - lo)
    d_pt = lo + gr * (hi - lo)
    fc = obj(c_pt)
    fd = obj(d_pt)
    
    for _ in range(30):
        if fc < fd:
            hi = d_pt
            d_pt = c_pt
            fd = fc
            c_pt = hi - gr * (hi - lo)
            fc = obj(c_pt)
        else:
            lo = c_pt
            c_pt = d_pt
            fc = fd
            d_pt = lo + gr * (hi - lo)
            fd = obj(d_pt)
    
    return max(0.0, (lo + hi) / 2.0)


def coordinate_descent_pass(a):
    """One full pass of coordinate descent."""
    a = np.maximum(a.astype(np.float64), 0.0)
    S = a.sum()
    if S < 1e-10:
        return a, compute_J(a)
    n = len(a)
    b = fast_conv(a)
    current_J = 2.0 * n * b.max() / (S * S)
    
    indices = list(range(n))
    np.random.shuffle(indices)
    
    for i in indices:
        new_val = coordinate_update(a, i, b, S)
        if abs(new_val - a[i]) > 1e-12:
            old_val = a[i]
            a[i] = new_val
            S = S - old_val + new_val
            for k in range(max(0, i), min(2 * n - 1, i + n)):
                ki = k - i
                if 0 <= ki < n:
                    if k == 2 * i:
                        b[k] = b[k] - old_val * old_val + new_val * new_val
                    else:
                        b[k] = b[k] - 2.0 * old_val * a[ki] + 2.0 * new_val * a[ki]
    
    b_check = fast_conv(a)
    new_J = 2.0 * n * b_check.max() / (S * S)
    return a, new_J


def coordinate_descent_phase(a, deadline, tracker, max_passes=20):
    """Multiple passes of coordinate descent."""
    a = np.maximum(a.astype(np.float64), 0.0)
    a /= a.sum()
    current_J = compute_J(a)
    tracker.update(a)
    
    for p in range(max_passes):
        if time.monotonic() > deadline:
            break
        t0 = time.monotonic()
        a, new_J = coordinate_descent_pass(a)
        elapsed = time.monotonic() - t0
        if new_J < current_J - 1e-12:
            current_J = new_J
            tracker.update(a)
            print(f"    CD pass {p}: J={current_J:.8f} ({elapsed:.1f}s)", flush=True)
        else:
            if p > 2:
                break
    
    tracker.update(a)
    return a


def resample_seq(src, dest_n):
    """Resample 1D sequence to new length via linear interpolation."""
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)


class ScoreTracker:
    def __init__(self):
        self.best = None
        self.best_score = float('inf')
        self.t_start = time.monotonic()
    
    def update(self, a):
        if a is None:
            return
        a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
        if a.size == 0 or a.sum() < 0.01:
            return
        score = float(evaluate_sequence(a.tolist()))
        if np.isfinite(score) and score < self.best_score:
            self.best = a.copy()
            self.best_score = score
            elapsed = time.monotonic() - self.t_start
            print(f"  *** NEW BEST: J={score:.8f} n={a.size} t={elapsed:.0f}s ***", flush=True)
    
    def report(self, label=""):
        if self.best is not None:
            print(f"  {label} best: J={self.best_score:.8f} n={self.best.size} t={time.monotonic()-self.t_start:.0f}s", flush=True)


def combined_gradient_lp_phase(a, deadline, tracker, grad_steps_per_lp=5):
    """Alternate between gradient flow steps and LP best-response."""
    a = a.astype(np.float64).copy()
    a = np.maximum(a, 1e-6)
    a /= a.sum()
    n = len(a)
    current_J = compute_J(a)
    tracker.update(a)
    
    c_range = fast_conv(a).max()
    beta = max(c_range * 0.05, 1e-4)
    eta = 1.0 / (n * 0.2)
    tighten = 0.004
    last_print = 0
    cycle = 0
    
    while time.monotonic() < deadline:
        # Gradient flow steps
        for _ in range(grad_steps_per_lp):
            if time.monotonic() > deadline:
                break
            a_new, smoothmax = gradient_flow_step(a, beta, eta)
            J_new = compute_J(a_new)
            if J_new < current_J - 1e-14:
                a = a_new
                current_J = J_new
                eta = min(eta * 1.03, 3.0 / n)
            else:
                eta *= 0.7
        
        # LP step
        if time.monotonic() > deadline:
            break
        c = fast_conv(a)
        M = float(c.max())
        g = lp_best_response(a, M * (1.0 - tighten))
        if g is None:
            g = lp_best_response(a, M * (1.0 - tighten * 0.5))
        if g is not None:
            g /= g.sum()
            t = 0.15
            a_trial = (1.0 - t) * a + t * g
            a_trial = np.maximum(a_trial, 1e-8)
            a_trial /= a_trial.sum()
            J_trial = compute_J(a_trial)
            if J_trial < current_J - 1e-14:
                a = a_trial
                current_J = J_trial
                tracker.update(a)
        
        cycle += 1
        if cycle % 10 == 0:
            if (time.monotonic() - last_print) > 5.0:
                print(f"    Combined cycle {cycle}: J={current_J:.8f}", flush=True)
                last_print = time.monotonic()
    
    tracker.update(a)
    return a
