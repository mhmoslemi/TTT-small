    n = len(a)
    conv = compute_conv(a)
    M = np.max(conv)

        idx = np.argsort(conv)[-topk:]

        conv_top = conv

def cosine_init(n, n_modes, seed):
    rng = np.random.default_rng(seed)

        phase = rng.uniform(0, 2 * np.pi)
        a += c * np.cos(2 * np.pi * m * k / n + phase)
    a = np.maximum(a, 0)
    S = np.sum(a)

def gaussian_init(n, sigma_frac=0.3):

                print(f"  n={n_try}: {obj_new:.6f}")
    # Phase 5: final polish
    print(f"\n=== Phase 5: Final polish (best={best_score:.6f}) ===")
    a_cur = best_seq.copy()
    n_final = len(a_cur)
    if time.monotonic() < deadline - 8:
        remaining = deadline - time.monotonic()
        a_sg = momentum_subgrad(a_cur, time_budget=remaining * 0.35,
                                alpha0=20.0, alpha1=100.0, topk=30)
        obj_sg = compute_obj(a_sg)
        if obj_sg < best_score:
            best_score = obj_sg
            best_seq = a_sg.copy()
            print(f"  SG: {best_score:.6f} ** NEW BEST **")
        remaining = deadline - time.monotonic()
        if remaining > 10 and n_final <= 500:
            a_lp = alternating_lp(a_cur, time_budget=min(remaining * 0.5, 30.0))
            obj_lp = compute_obj(a_lp)
            if obj_lp < best_score:
                best_score = obj_lp
                best_seq = a_lp.copy()
                print(f"  Final LP: {best_score:.6f} ** NEW BEST **")
        remaining = deadline - time.monotonic()
        if remaining > 10:
            rng = np.random.default_rng(77777)
            for _ in range(5):
                if time.monotonic() > deadline - 5:
                    break
                a_p = a_cur * np.exp(rng.normal(0, 0.04, n_final))
                a_p = np.maximum(a_p, 0)
                Sp = np.sum(a_p)
                if Sp > 1e-10:
                    a_p = a_p / Sp
                    a_p = momentum_subgrad(a_p, time_budget=min(remaining * 0.15, 10.0),
                                           alpha0=30.0, alpha1=120.0, topk=40)
                    obj_p = compute_obj(a_p)
                    if obj_p < best_score:
                        best_score = obj_p
                        best_seq = a_p.copy()
                        print(f"  Perturb: {best_score:.6f} ** NEW BEST **")
    # Final validation
    if best_seq is None:
        best_seq = np.ones(300) / 300
        best_score = compute_obj(best_seq)
    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    val = evaluate_sequence(result)
    if val < np.inf and val < best_score:
        best_score = val
    elif val == np.inf:
        result = [1.0] * 300
        best_score = 2.0
    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_score:.6f}, time={elapsed:.1f}s ===")
    return result
