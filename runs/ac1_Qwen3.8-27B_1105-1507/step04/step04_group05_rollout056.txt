import numpy as np

def fast_conv(a):
    """Full self-convolution via FFT."""
    n = len(a)
    s = 1
    while s < 2 * n - 1:
        s <<= 1
    fa = np.fft.rfft(a, s)
    return np.fft.irfft(fa * fa, s)[:2 * n - 1]


def compute_J(a):
    """Compute J(a) = 2*n*max(conv(a,a))/(sum(a))^2."""
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))


def build_A_sparse(a):
    """Build sparse constraint matrix for LP."""
    n = len(a)
    from scipy.sparse import lil_matrix
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        jl = max(0, k - n + 1)
        jh = min(n - 1, k)
        for j in range(jl, jh + 1):
            rows.append(k)
            cols.append(j)
            vals.append(a[k - j])
    A = lil_matrix((2 * n - 1, n))
    A[rows, cols] = vals
    return A.tocsr()


def lp_best_response(a, M):
    """LP: max sum(g) s.t. (a*g)_k <= M for all k, g >= 0."""
    n = len(a)
    c = -np.ones(n)
    A = build_A_sparse(a)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
    if r.success and r.x is not None:
        g = np.maximum(r.x, 0.0)
        if g.sum() > 1e-10:
            return g
    return None


def softmax_conv_gradient(a, beta=100.0):
    """Compute gradient of soft-max smoothed J with respect to a.
    J_smooth = 2n * M_smooth / S^2
    M_smooth = (1/beta) * log(sum_k exp(beta * conv_k))
    """
    n = len(a)
    b = fast_conv(a)
    S = a.sum()
    
    # Softmax weights
    b_shifted = b - b.max()  # for numerical stability
    p = np.exp(beta * b_shifted)
    p_sum = p.sum()
    p = p / p_sum  # softmax weights over convolution entries
    
    # M_smooth approximation
    M_smooth = b.max() + np.log(p_sum) / beta
    
    # grad_M[i] = sum_k p[k] * 2 * a[k-i] (if 0 <= k-i < n)
    grad_M = np.zeros(n)
    for k in range(2 * n - 1):
        if p[k] < 1e-15:
            continue
        # a[k-i] for i in [max(0, k-n+1), min(k, n-1)]
        il = max(0, k - n + 1)
        ih = min(n - 1, k)
        # a[k-i] for i in [il, ih] is a[k-ih : k-il+1] reversed
        grad_M[il:ih+1] += 2.0 * p[k] * a[k-ih:k-il+1][::-1]
    
    # grad_J[i] = 2n/S * (grad_M[i] - 2*M_smooth/S)
    grad_J = (2.0 * n / S) * (grad_M - 2.0 * M_smooth / S)
    
    J_smooth = 2.0 * n * M_smooth / (S * S)
    
    return J_smooth, grad_J

def resample_seq(src, dest_n):

        a_momentum = a + v
        a_momentum = np.maximum(a_momentum, 0.0001)
        a_momentum = a_momentum / a_momentum.sum()

        # Line search
        best_J = current_J
        best_a = a
        for alpha in [1.0, 0.7, 0.5, 0.3, 0.15, 0.05]:
            a_ls = (1.0 - alpha) * a + alpha * a_momentum
            a_ls = np.maximum(a_ls, 0.0001)
            a_ls = a_ls / a_ls.sum()
            J_ls = compute_J(a_ls)
            if J_ls < best_J:
                best_J = J_ls
                best_a = a_ls

        if best_J < current_J - 1e-13:
            a = best_a
            current_J = best_J
            stall_count = 0
            # Slightly increase tightening when making progress
            tighten = min(0.015, tighten * 1.02)
        else:
            v *= 0.3
            stall_count += 1
            tighten = max(0.001, tighten * 0.9)
            if stall_count > 25:
                break

        if it % 30 == 0:
            tracker.update(a)

    tracker.update(a)
    return a


def lbfgsb_refine(a, deadline, tracker, beta=100.0, max_iter=300):
    """Refine using L-BFGS-B with soft-max smoothed objective."""
    a = np.maximum(a.astype(np.float64), 0.0001)
    a = a / a.sum()
    n = len(a)
    current_J = compute_J(a)
    tracker.update(a)

    if time.monotonic() > deadline:
        return a

    def obj_and_grad(x):
        x = np.maximum(x, 0.0)
        s = x.sum()
        if s < 1e-10:
            return 1e10, np.zeros(len(x))
        J_val, grad = softmax_objective_and_grad(x, beta=beta)
        if not np.isfinite(J_val):
            return 1e10, np.zeros(len(x))
        return J_val, grad

    bounds = [(0.0, None)] * n

    try:
        result = minimize(obj_and_grad, a, method='L-BFGS-B',
                          bounds=bounds,
                          options={'maxiter': max_iter, 'ftol': 1e-16,
                                   'gtol': 1e-14, 'maxls': 50})
        a_new = np.maximum(result.x, 0.0001)
        a_new = a_new / a_new.sum()
        J_new = compute_J(a_new)
        if J_new < current_J:
            a = a_new
            current_J = J_new
    except Exception as e:
        print(f"    L-BFGS-B error: {e}", flush=True)

    tracker.update(a)
    return a


def coordinate_descent_pass(a):
    """One pass of 1D coordinate descent with golden section search."""
    a = np.maximum(a.astype(np.float64), 0.0)
    S = a.sum()
    if S < 1e-10:
        return a, compute_J(a)
    n = len(a)
    b = fast_conv(a)
    gr = (np.sqrt(5.0) - 1.0) / 2.0

    indices = list(range(n))
    np.random.shuffle(indices)

    for i in indices:
        ai = a[i]
        max_c_static = 0.0
        for k in range(2 * n - 1):
            ki = k - i
            if not (0 <= ki < n):
                if b[k] > max_c_static:
                    max_c_static = b[k]

        def f_1d(x):
            new_S = S - ai + x
            if new_S < 1e-10:
                return 1e10
            max_c_local = max_c_static
            for k in range(max(0, i), min(2 * n - 1, i + n)):
                ki = k - i
                if 0 <= ki < n:
                    if k == 2 * i:
                        ck = b[k] - ai * ai + x * x
                    else:
                        ck = b[k] - 2.0 * ai * a[ki] + 2.0 * x * a[ki]
                else:
                    ck = b[k]
                if ck > max_c_local:
                    max_c_local = ck
            return 2.0 * n * max_c_local / (new_S * new_S)

        lo = 0.0
        hi = max(3.0 * ai, 3.0 * S / n, 0.01)
        c_point = hi - gr * (hi - lo)
        d_point = lo + gr * (hi - lo)
        fc = f_1d(c_point)
        fd = f_1d(d_point)

        for _ in range(35):
            if fc < fd:
                hi = d_point
                d_point = c_point
                fd = fc
                c_point = hi - gr * (hi - lo)
                fc = f_1d(c_point)
            else:
                lo = c_point
                c_point = d_point
                fc = fd
                d_point = lo + gr * (hi - lo)
                fd = f_1d(d_point)

        x_opt = max(0.0, (lo + hi) / 2.0)

        if abs(x_opt - ai) > 1e-12:
            old_val = ai
            a[i] = x_opt
            S = S - old_val + x_opt
            for k in range(max(0, i), min(2 * n, i + n)):
                ki = k - i
                if 0 <= ki < n:
                    if k == 2 * i:
                        b[k] = b[k] - old_val * old_val + x_opt * x_opt
                    else:
                        b[k] = b[k] - 2.0 * old_val * a[ki] + 2.0 * x_opt * a[ki]

    b_check = fast_conv(a)
    new_J = 2.0 * n * b_check.max() / (a.sum() ** 2)
    return a, new_J


def block_coordinate_descent(a, block_size=4, n_passes=3, deadline=None, tracker=None):
    """Block coordinate descent using Nelder-Mead for each block."""
    a = np.maximum(a.astype(np.float64), 0.0001)
    a = a / a.sum()
    n = len(a)

    if tracker:
        tracker.update(a)

    for pass_idx in range(n_passes):
        if deadline and time.monotonic() > deadline:
            break

        blocks = []
        start = 0
        while start < n:
            end = min(start + block_size, n)
            blocks.append(list(range(start, end)))
            start = end
        np.random.shuffle(blocks)

        for block in blocks:
            if deadline and time.monotonic() > deadline:
                break

            x0 = a[block].copy()

            def block_obj(x, blk=block, base=a):
                a_trial = base.copy()
                a_trial[blk] = np.maximum(x, 0.0)
                s = a_trial.sum()
                if s < 1e-10:
                    return 1e10
                return compute_J(a_trial)

            try:
                result = minimize_nelder_mead(block_obj, x0, maxiter=100,
                                              xatol=1e-12, fatol=1e-14)
                x_new = np.maximum(result.x, 0.0)
                a[block] = x_new
                a = a / a.sum()
            except Exception:
                pass

    if tracker:
        tracker.update(a)
    return a


def resample_seq(src, dest_n):
    """Resample 1D sequence to new length via linear interpolation."""
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0001)


def generate_initial(n, rng, style=0):
    """Generate diverse initial sequences."""
    t = np.linspace(0, 1, n, endpoint=False)
    if style == 0:
        return np.ones(n)
    elif style == 1:
        return np.maximum(1.0 + 0.3 * np.cos(2 * np.pi * 3 * t), 0.1)
    elif style == 2:
        return np.maximum(1.0 + 0.3 * np.cos(2 * np.pi * 5 * t + 1.0), 0.1)
    elif style == 3:
        return np.maximum(0.5 + 0.5 * np.cos(2 * np.pi * t) ** 2, 0.01)
    elif style == 4:
        return np.maximum(rng.uniform(0.5, 1.0, n), 0.01)
    elif style == 5:
        return np.maximum(1.0 + 0.2 * np.cos(2 * np.pi * 2 * t) + 0.2 * np.cos(2 * np.pi * 7 * t), 0.01)
    else:
        return np.ones(n)


class ScoreTracker:
    def __init__(self):
        self.best = None
        self.best_score = float('inf')
        self.t_start = time.monotonic()

    def update(self, a):
        a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
        if a.size == 0 or a.sum() < 0.01:
            return
        score = float(evaluate_sequence(a.tolist()))
        if np.isfinite(score) and score < self.best_score:
            self.best = a.copy()
            self.best_score = score
            elapsed = time.monotonic() - self.t_start
            print(f"  * NEW BEST: J={score:.8f} n={a.size} t={elapsed:.0f}s", flush=True)

    def report(self, label=""):
        if self.best is not None:
            print(f"  {label} best: J={self.best_score:.8f} n={self.best.size} t={time.monotonic()-self.t_start:.0f}s", flush=True)


def optimize_at_length(a, deadline, tracker, n_lp=80, n_cd_passes=3, beta_lbfgs=150.0):
    """Full optimization pipeline at a given length."""
    a = np.maximum(a.astype(np.float64), 0.0001)
    a = a / a.sum()
    n = len(a)

    # Phase A: LP mirror descent
    if time.monotonic() < deadline - 60:
        a = accelerated_lp_mirror_descent(a, n_lp, deadline - 60, tracker,
                                          tighten_base=0.003)

    # Phase B: L-BFGS-B refinement
    if time.monotonic() < deadline - 30:
        a = lbfgsb_refine(a, deadline - 30, tracker, beta=beta_lbfgs, max_iter=200)

    # Phase C: Coordinate descent
    for _ in range(n_cd_passes):
        if time.monotonic() > deadline - 10:
            break
        a, new_J = coordinate_descent_pass(a)
        a = a / a.sum()
        tracker.update(a)

    # Phase D: Block CD
    if time.monotonic() < deadline - 10:
        a = block_coordinate_descent(a, block_size=3, n_passes=2,
                                     deadline=deadline - 10, tracker=tracker)

    # Phase E: Final L-BFGS-B with higher beta
    if time.monotonic() < deadline - 5:
        a = lbfgsb_refine(a, deadline - 5, tracker, beta=300.0, max_iter=100)

    # Phase F: Final CD
    if time.monotonic() < deadline - 2:
        a, _ = coordinate_descent_pass(a)
        a = a / a.sum()
        tracker.update(a)

    return a


def propose_candidate(seed=42, budget_s=1500, **kwargs):
    """
    Main search function combining accelerated LP mirror descent,
    L-BFGS-B with analytic gradient, coordinate descent, and block CD.
    """
    rng = np.random.default_rng(seed)
    total_deadline = time.monotonic() + float(budget_s) - 30.0
    tracker = ScoreTracker()

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        if p.ndim == 1 and p.size > 0 and np.all(np.isfinite(p)):
            parent = np.maximum(p, 0.0)
            tracker.update(parent)
            print(f"Parent loaded: n={parent.size}, J={tracker.best_score:.8f}", flush=True)
    except (NameError, AttributeError):
        pass

    if parent is None:
        parent = np.ones(400)
        tracker.update(parent)
        print(f"No parent found, using uniform n=400", flush=True)

    n_parent = parent.size

    # === Phase 1: Optimize at parent length ===
    print(f"\n{'='*50} Phase 1: Optimize at n={n_parent} {'='*50}", flush=True)
    t0 = time.monotonic()
    deadline_p1 = total_deadline - 600

    a = optimize_at_length(parent.copy(), deadline_p1, tracker,
                           n_lp=100, n_cd_passes=4, beta_lbfgs=150.0)
    tracker.report(f"Phase 1 (t={time.monotonic()-t0:.1f}s)")

    # === Phase 2: Try nearby lengths ===
    print(f"\n{'='*50} Phase 2: Nearby Lengths {'='*50}", flush=True)
    t0 = time.monotonic()

    nearby_lengths = [n_parent - 4, n_parent - 2, n_parent + 2, n_parent + 4,
                      n_parent + 8, n_parent - 8, n_parent + 12]
    nearby_lengths = [l for l in nearby_lengths if 200 <= l <= 800]
    seen = set()
    unique_lengths = []
    for l in nearby_lengths:
        if l not in seen:
            seen.add(l)
            unique_lengths.append(l)

    per_length_budget = min(60, (total_deadline - 300 - time.monotonic()) / max(1, len(unique_lengths)))

    for nl in unique_lengths:
        if time.monotonic() > total_deadline - 200:
            break
        a_res = resample_seq(tracker.best, nl)
        a_res = np.maximum(a_res, 0.0001)
        a_res = a_res / a_res.sum()
        dl = min(time.monotonic() + per_length_budget, total_deadline - 100)
        a_res = optimize_at_length(a_res, dl, tracker, n_lp=30, n_cd_passes=2, beta_lbfgs=100.0)
        print(f"    Length {nl}: J={tracker.best_score:.8f}", flush=True)

    tracker.report(f"Phase 2 (t={time.monotonic()-t0:.1f}s)")

    # === Phase 3: Try very different lengths ===
    print(f"\n{'='*50} Phase 3: Diverse Lengths {'='*50}", flush=True)
    t0 = time.monotonic()

    diverse_lengths = [256, 320, 448, 512, 640]
    diverse_lengths = [l for l in diverse_lengths if l != n_parent and 150 <= l <= 1000]

    for nl in diverse_lengths:
        if time.monotonic() > total_deadline - 150:
            break
        # Resample from best
        a_div = resample_seq(tracker.best, nl)
        a_div = np.maximum(a_div, 0.0001)
        a_div = a_div / a_div.sum()
        dl = min(time.monotonic() + 40, total_deadline - 100)
        a_div = optimize_at_length(a_div, dl, tracker, n_lp=20, n_cd_passes=2, beta_lbfgs=80.0)
        print(f"    Length {nl}: J={tracker.best_score:.8f}", flush=True)

    tracker.report(f"Phase 3 (t={time.monotonic()-t0:.1f}s)")

    # === Phase 4: Intensive optimization on best ===
    print(f"\n{'='*50} Phase 4: Intensive Polish {'='*50}", flush=True)
    t0 = time.monotonic()
    best_n = tracker.best.size

    # More LP iterations
    a_best = tracker.best.copy()
    a_best = accelerated_lp_mirror_descent(a_best, 80, total_deadline - 80, tracker,
                                           tighten_base=0.002)

    # L-BFGS-B with multiple beta values
    for beta in [50.0, 100.0, 200.0, 400.0]:
        if time.monotonic() > total_deadline - 30:
            break
        a_best = lbfgsb_refine(a_best, total_deadline - 30, tracker, beta=beta, max_iter=150)

    # Multiple CD passes
    for _ in range(5):
        if time.monotonic() > total_deadline - 15:
            break
        a_best, _ = coordinate_descent_pass(a_best)
        a_best = a_best / a_best.sum()
        tracker.update(a_best)

    # Block CD final
    if time.monotonic() < total_deadline - 10:
        a_best = block_coordinate_descent(a_best, block_size=3, n_passes=2,
                                          deadline=total_deadline - 10, tracker=tracker)

    # Final L-BFGS-B
    if time.monotonic() < total_deadline - 5:
        a_best = lbfgsb_refine(a_best, total_deadline - 5, tracker, beta=500.0, max_iter=100)

    # Very final CD
    if time.monotonic() < total_deadline - 2:
        a_best, _ = coordinate_descent_pass(a_best)
        a_best = a_best / a_best.sum()
        tracker.update(a_best)

    tracker.report(f"Phase 4 (t={time.monotonic()-t0:.1f}s)")

    # === Phase 5: Perturbation restarts ===
    if time.monotonic() < total_deadline - 20:
        print(f"\n{'='*50} Phase 5: Perturbation Restarts {'='*50}", flush=True)
        t0 = time.monotonic()
        n = tracker.best.size
        for temp in [0.03, 0.02, 0.01]:
            if time.monotonic() > total_deadline - 15:
                break
            a_pert = tracker.best.copy() + rng.normal(0, temp, n)
            a_pert = np.maximum(a_pert, 0.0001)
            a_pert = a_pert / a_pert.sum()
            dl = min(time.monotonic() + 25, total_deadline - 10)
            a_pert = optimize_at_length(a_pert, dl, tracker, n_lp=20, n_cd_passes=2,
                                        beta_lbfgs=150.0)
        tracker.report(f"Phase 5 (t={time.monotonic()-t0:.1f}s)")

    # Fallback
    if tracker.best is None:
        tracker.best = np.ones(256)
        tracker.best_score = compute_J(tracker.best)

    print(f"\n{'='*50} FINAL RESULT {'='*50}", flush=True)
    print(f"J = {tracker.best_score:.10f}, n = {tracker.best.size}", flush=True)
    print(f"Total time: {time.monotonic()-tracker.t_start:.1f}s", flush=True)

    return [float(x) for x in tracker.best]
