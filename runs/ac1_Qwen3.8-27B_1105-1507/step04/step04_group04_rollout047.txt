def compute_conv(a):
    """Compute full convolution a*a using FFT. Returns array of length 2n-1."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    conv = np.fft.irfft(fa * fa, size)
    return conv[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    conv = compute_conv(a)
    max_b = float(np.max(conv))
    sum_a = float(np.sum(a))
    if sum_a < 0.01:
        return np.inf
    return 2.0 * n * max_b / (sum_a * sum_a)


def compute_subgradient(a, tau=100.0):
    """
    Compute subgradient of the objective.
    Uses soft-max approximation with parameter tau.
    Returns (objective, gradient) where gradient has same length as a.
    """
    n = len(a)
    conv = compute_conv(a)
    max_conv = float(np.max(conv))
    S = float(np.sum(a))
    f = 2.0 * n * max_conv / (S * S)

    # Soft weights on the convolution peaks
    shifted = tau * (conv - max_conv)
    shifted = np.clip(shifted, -500, 0)
    weights = np.exp(shifted)
    w_sum = np.sum(weights)
    if w_sum < 1e-300:
        idx = int(np.argmax(conv))
        weights = np.zeros(len(conv))
        weights[idx] = 1.0
    else:
        weights /= w_sum

    # Compute h[i] = sum_j a[j] * weights[i+j] for i = 0,...,n-1
    # This is the cross-correlation of a and weights
    N = 1
    while N < n + len(weights):
        N <<= 1
    A = np.fft.rfft(a, N)
    W = np.fft.rfft(weights, N)
    h_full = np.fft.irfft(np.conj(A) * W, N)
    h = h_full[:n]

    # Gradient: df/da_i = (4n/S^3) * (S * h[i] - M)
    grad = (4.0 * n / (S ** 3)) * (S * h - max_conv)
    return f, grad


def build_toeplitz(a):
    """Build Toeplitz matrix where (A@g)[k] = (a*g)[k] for k=0..2n-2."""
    n = len(a)
    A = np.zeros((2 * n - 1, n), dtype=np.float64)
    for j in range(n):
        k_end = min(j + n, 2 * n - 1)
        A[j:k_end, j] = a[:k_end - j]
    return A


def solve_lp_maxsum(a, M, time_limit=10):
    """Solve LP: max sum(g) s.t. (a*g)[k] <= M for all k, g >= 0."""
    n = len(a)
    A = build_toeplitz(a)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': time_limit})
        if res.status == 0:
            g = np.maximum(res.x, 0)
            return g, float(np.sum(g))
    except Exception:
        pass
    return None, None


def backtracking_line_search(a, grad, f_current, tau, max_iter=30):
    """
    Find step size for multiplicative update a_i <- a_i * exp(-step * grad_i).
    Uses backtracking. Returns (new_a, new_f, step_used).
    """
    n = len(a)
    step = 0.1
    best_a = a.copy()
    best_f = f_current

    for _ in range(max_iter):
        log_update = -step * grad
        log_update = np.clip(log_update, -100, 100)
        a_try = a * np.exp(log_update)
        a_try = np.maximum(a_try, 1e-15)
        S = np.sum(a_try)
        if S < 1e-10:
            step *= 0.5
            continue
        a_try = a_try / S
        f_try = compute_obj(a_try)

        if f_try < f_current - 1e-15:
            return a_try, f_try, step
        step *= 0.5
        if step < 1e-8:
            break

    return best_a, best_f, 0.0

    a = a / S

    best_f = compute_obj(a)

    step_boost = 1.0

        a_new, f_new, step_used = backtracking_line_search(

        if f_new < best_f - 1e-14:
            best_f = f_new
            best_a = a_new.copy()
            a = a_new
            stall = 0
            if verbose and it % 100 == 0:
                elapsed = time.monotonic() - start

            stall += 1
            if stall > patience:

                    a = a / np.sum(a)

                step_boost *= 0.7

def resample_sequence(a, new_n):
    """Resample a sequence to a new length using interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    old_k = np.arange(n)
    new_k = np.linspace(0, n - 1, new_n)
    a_new = np.interp(new_k, old_k, a)
    a_new = np.maximum(a_new, 0)
    S = np.sum(a_new)
    if S > 1e-10:
        a_new = a_new / S
    return a_new

def propc2_optimize(a, time_budget, verbose=False):
    """
    PROXC2-inspired optimization: iterative projection-like updates.
    Alternates between:
    1. Computing the "dual" weights from the current convolution
    2. Updating a using the dual weights
    This is similar to the multiplicative water-filling algorithm.
    """
    n = len(a)
    a = a.copy().astype(np.float64)
    S = np.sum(a)
    if S < 1e-10:
        return a
    a = a / S

    best_a = a.copy()
    best_f = compute_obj(a)
    start = time.monotonic()

    # PROXC2-like iteration
    for it in range(1000):
        if time.monotonic() - start > time_budget:
            break

        conv = compute_conv(a)
        max_conv = float(np.max(conv))

        # Dual variables: weights proportional to how much each lag exceeds threshold
        # Use a "soft threshold" approach
        tau = 20.0 + it * 0.5  # increasing sharpness
        excess = np.maximum(conv - max_conv * 0.5, 0)
        if np.max(excess) < 1e-15:
            break
        dual = excess / np.max(excess)

        # Update: a_i proportional to 1 / (sum_k dual[k] * a[k-i])^alpha
        # This is a "repulsion" update that reduces peaks
        h = np.array([np.dot(a, dual[i:i+n]) for i in range(n)])
        h = np.maximum(h, 1e-15)

        alpha = 0.5 + 0.3 * np.sin(it * 0.1)  # oscillating exponent
        a_new = 1.0 / (h ** alpha)
        a_new = np.maximum(a_new, 1e-10)
        a_new = a_new / np.sum(a_new)

        # Blend with current to avoid oscillation
        t = 0.3 + 0.2 * np.exp(-it * 0.01)
        a_try = (1 - t) * a + t * a_new
        a_try = np.maximum(a_try, 0)
        a_try = a_try / np.sum(a_try)

        f_try = compute_obj(a_try)
        if f_try < best_f:
            best_f = f_try
            best_a = a_try.copy()
            a = a_try
            if verbose and it % 50 == 0:
                elapsed = time.monotonic() - start
                print(f"    PROXC2 it={it}, obj={best_f:.8f}, elapsed={elapsed:.1f}s")

    return best_a

        parent = np.clip(parent, 0, 1000.0)
        if parent.size > 0 and np.isfinite(parent).all():
            p_obj = compute_obj(parent)
            if p_obj < best_score:
                best_score = p_obj
                best_seq = parent.copy()
            print(f"Parent (n={parent.size}): obj={p_obj:.8f}")
    except Exception:
        pass

    if best_seq is None:
        # Fallback: create a reasonable initial sequence
        n = 330
        rng = np.random.default_rng(seed)
        k = np.arange(n)
        best_seq = 1.0 + 0.3 * np.cos(2 * np.pi * 1.5 * k / n) + 0.2 * np.cos(2 * np.pi * 4.0 * k / n)
        best_seq = np.maximum(best_seq, 0)
        best_seq = best_seq / np.sum(best_seq)
        best_score = compute_obj(best_seq)
        print(f"Generated initial (n={n}): obj={best_score:.8f}")

    n_parent = len(best_seq)
    print(f"Starting best: {best_score:.8f} (n={n_parent})")
    print(f"Total time budget: {budget_s}s")

    # Phase 1: Intensive optimization of parent at its native length
    print("\n=== Phase 1: Optimize parent (n={}) ===".format(n_parent))
    remaining = deadline - time.monotonic()
    p1_budget = min(remaining * 0.25, 300.0)
    a_opt = alternating_optimize(best_seq, p1_budget, verbose=True)
    f_opt = compute_obj(a_opt)
    if f_opt < best_score:
        best_score = f_opt
        best_seq = a_opt.copy()
        print(f"  Phase 1 result: {best_score:.8f}")

    # Phase 2: PROXC2 on the best so far
    remaining = deadline - time.monotonic()
    if remaining > 30:
        print(f"\n=== Phase 2: PROXC2 ({remaining:.0f}s remaining) ===")
        p2_budget = min(remaining * 0.15, 120.0)
        a_opt = propc2_optimize(best_seq, p2_budget, verbose=True)
        f_opt = compute_obj(a_opt)
        if f_opt < best_score:
            best_score = f_opt
            best_seq = a_opt.copy()
            print(f"  Phase 2 result: {best_score:.8f}")

        # Follow up with gradient
        remaining = deadline - time.monotonic()
        if remaining > 20:
            p2b_budget = min(remaining * 0.1, 60.0)
            a_opt = gradient_optimize(best_seq, p2b_budget, tau_init=100.0, verbose=True)
            f_opt = compute_obj(a_opt)
            if f_opt < best_score:
                best_score = f_opt
                best_seq = a_opt.copy()
                print(f"  Phase 2b result: {best_score:.8f}")

    # Phase 3: Try nearby n values
    print(f"\n=== Phase 3: Nearby n values ({deadline - time.monotonic():.0f}s remaining) ===")
    n_values = sorted(set([
        n_parent - 10, n_parent - 5, n_parent, n_parent + 5, n_parent + 10,
        n_parent + 20, n_parent - 20, n_parent + 30, n_parent - 30
    ]))
    n_values = [n for n in n_values if 150 <= n <= 800]

    remaining = deadline - time.monotonic()
    if remaining > 40:
        # Allocate time: more time to n closest to parent
        n_values.sort(key=lambda x: abs(x - n_parent))
        total_alloc = remaining * 0.35
        # Give more time to first few
        weights = [1.0 / (i + 1) for i in range(len(n_values))]
        total_w = sum(weights)
        allocs = [total_alloc * w / total_w for w in weights]

        for i, n_val in enumerate(n_values):
            if time.monotonic() > deadline - 30:
                break
            remaining = deadline - time.monotonic()
            if remaining < 20:
                break
            alloc = min(allocs[i], remaining * 0.3, 80.0)
            alloc = max(alloc, 15.0)

            # Resample parent to this n
            a_init = resample_sequence(parent if parent is not None else best_seq, n_val)

            print(f"  n={n_val} (alloc={alloc:.0f}s, {remaining:.0f}s left)")
            a_opt = alternating_optimize(a_init, alloc, verbose=(n_val == n_parent))
            f_opt = compute_obj(a_opt)
            if f_opt < best_score:
                best_score = f_opt
                best_seq = a_opt.copy()
                print(f"  ** UPDATE: n={n_val}, obj={best_score:.8f} **")

    # Phase 4: Local perturbation search on best
    if best_seq is not None:
        remaining = deadline - time.monotonic()
        if remaining > 30:
            print(f"\n=== Phase 4: Local perturbation search ({remaining:.0f}s) ===")
            p4_budget = min(remaining * 0.3, 200.0)
            a_opt = local_perturbation_search(best_seq, p4_budget, verbose=True)
            f_opt = compute_obj(a_opt)
            if f_opt < best_score:
                best_score = f_opt
                best_seq = a_opt.copy()
                print(f"  Phase 4 result: {best_score:.8f}")

    # Phase 5: Final intensive optimization on best
    if best_seq is not None:
        remaining = deadline - time.monotonic()
        if remaining > 30:
            print(f"\n=== Phase 5: Final optimization ({remaining:.0f}s) ===")
            p5_budget = min(remaining * 0.5, remaining - 15.0)
            a_opt = alternating_optimize(best_seq, p5_budget, verbose=True)
            f_opt = compute_obj(a_opt)
            if f_opt < best_score:
                best_score = f_opt
                best_seq = a_opt.copy()
                print(f"  Phase 5 result: {best_score:.8f}")

            # One more perturbation round
            remaining = deadline - time.monotonic()
            if remaining > 20:
                p5b_budget = min(remaining * 0.4, remaining - 10.0)
                a_opt = local_perturbation_search(best_seq, p5b_budget, verbose=True)
                f_opt = compute_obj(a_opt)
                if f_opt < best_score:
                    best_score = f_opt
                    best_seq = a_opt.copy()
                    print(f"  Phase 5b result: {best_score:.8f}")

    # Phase 6: Final gradient polish
    if best_seq is not None:
        remaining = deadline - time.monotonic()
        if remaining > 10:
            p6_budget = min(remaining * 0.6, remaining - 5.0)
            a_opt = gradient_optimize(best_seq, p6_budget, tau_init=200.0, verbose=True)
            f_opt = compute_obj(a_opt)
            if f_opt < best_score:
                best_score = f_opt
                best_seq = a_opt.copy()
                print(f"  Final polish: {best_score:.8f}")

    # Final validation
    if best_seq is None:
        best_seq = np.ones(100)
        best_score = compute_obj(best_seq)

    # Ensure valid output
    best_seq = np.clip(best_seq, 0.0, 1000.0)
    result = [float(x) for x in best_seq]

    # Verify with evaluate_sequence
    val = evaluate_sequence(result)
    if val < best_score:
        best_score = val

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(best_seq)}, score={best_score:.8f}, time={elapsed:.1f}s ===")

    return result
