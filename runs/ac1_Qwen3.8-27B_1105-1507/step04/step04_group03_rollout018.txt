def fourier_to_seq(params, n, n_harm):
    t = np.arange(n, dtype=np.float64) / n
    a = np.full(n, params[0], dtype=np.float64)
    for k in range(1, n_harm + 1):
        a += params[2 * k - 1] * np.cos(2.0 * np.pi * k * t)
        a += params[2 * k] * np.sin(2.0 * np.pi * k * t)
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-8:
        a = np.ones(n) / n
    else:
        a = a / s
    return a

import numpy as np

def fast_conv(a):
    """Full self-convolution via FFT."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]


def compute_J(a):
    """Compute J(a) = 2*n*max(conv(a,a))/(sum(a))^2."""
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))

def build_active_lp(a, M, n_active):
    """Build sparse matrix for LP using only the most binding constraints."""
    n = len(a)
    b = fast_conv(a)
    # Select most binding constraints (top n_active by b value)
    sorted_idx = np.argsort(b)[::-1]
    active_idx = sorted_idx[:min(n_active, len(b))]
    active_idx = np.sort(active_idx)

    rows = []
    cols = []
    vals = []
    for i, k in enumerate(active_idx):
        j_min = max(0, k - n + 1)
        j_max = min(n - 1, k)
        for j in range(j_min, j_max + 1):
            rows.append(i)
            cols.append(j)
            vals.append(a[k - j])

    if not rows:
        return None, None
    A = csr_matrix((vals, (rows, cols)), shape=(len(active_idx), n))
    b_ub = np.full(len(active_idx), M)
    return A, b_ub


def active_lp_solve(a, M, n_active):
    """Solve LP: max sum(g) s.t. A_active@g <= M, g >= 0."""
    n = len(a)

    if A is None:
        return None
    c = -np.ones(n)
    bounds = [(0, None)] * n

                    options={'presolve': True, 'dual_feasibility_tolerance': 1e-8})
        if r.success and r.x is not None:
            g = np.maximum(r.x, 0)

def subgradient_polish(a, max_iters, deadline):
    """Subgradient descent with momentum for final polishing."""
    a = np.maximum(a.astype(np.float64), 1e-12)
    a = a / a.sum()
    best_M = fast_conv(a).max()
    best_a = a.copy()
    momentum = np.zeros_like(a)
    beta = 0.8
    n = len(a)

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        b = fast_conv(a)
        M = b.max()
        tol = 1e-10 * M
        active = np.where(b >= M - tol)[0]

        grad = np.zeros_like(a)
        for k in active:
            for j in range(n):
                diff = k - j
                if 0 <= diff < n:
                    grad[j] += 2.0 * a[diff]

        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-15:
            break

        t = 1.0 / (gnorm * np.sqrt(it + 1.0))
        momentum = beta * momentum + (1 - beta) * grad / gnorm
        a_new = a - t * momentum
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s < 1e-10:
            break
        a_new = a_new / s

        new_M = fast_conv(a_new).max()
        if new_M < best_M:
            best_M = new_M
            best_a = a_new.copy()
        a = a_new

    return best_a


def resample_seq(src, dest_n):
    """Resample 1D sequence to new length."""
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)


def fourier_to_seq(params, n, n_harm):
    """Convert Fourier parameters to non-negative sequence."""
    t = np.arange(n, dtype=np.float64) / n
    a = np.full(n, params[0], dtype=np.float64)
    for k in range(1, n_harm + 1):
        a += params[2 * k - 1] * np.cos(2.0 * np.pi * k * t)
        a += params[2 * k] * np.sin(2.0 * np.pi * k * t)
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-8:
        a = np.ones(n) / n
    else:
        a = a / s
    return a


def seq_to_fourier(a, n_harm):
    """Extract Fourier parameters from a sequence."""
    n = len(a)
    a_norm = a / a.sum()
    t = np.arange(n, dtype=np.float64) / n
    params = np.zeros(2 * n_harm + 1)
    params[0] = np.mean(a_norm)
    for k in range(1, n_harm + 1):
        params[2 * k - 1] = 2.0 * np.mean(a_norm * np.cos(2.0 * np.pi * k * t))
        params[2 * k] = 2.0 * np.mean(a_norm * np.sin(2.0 * np.pi * k * t))
    return params


def cmaes_eig_half(C, z, dim):
    """Apply C^(1/2) @ z using eigendecomposition."""
    eigvals, eigvecs = np.linalg.eigh(C)
    eigvals = np.maximum(eigvals, 1e-12)
    root = np.sqrt(eigvals)
    return (eigvecs * root) @ (eigvecs.T @ z)


def run_cmaes_fourier(n, n_harm, params0, sigma0, deadline, tracker, rng):
    """CMA-ES on Fourier parameters."""
    dim = 2 * n_harm + 1
    lam = 4 + int(3 * np.log(dim))
    mu = lam // 2

    weights = np.log(mu + 0.5) - np.log(np.arange(1, mu + 1))
    weights = weights / weights.sum()
    mu_eff = 1.0 / np.sum(weights ** 2)
    c_sigma = (mu_eff + 2) / (dim + mu_eff + 5)
    c_c = (4 + mu_eff / dim) / (dim + 4 + 2 * mu_eff / dim)
    d_sigma = 1 + 2 * max(0, np.sqrt((mu_eff - 1) / (dim + 1) - 1) - 1) + c_sigma
    c_1 = 2 / ((dim + 1.3) ** 2 + mu_eff)
    c_mu = min(1 - c_1, 2 * (mu_eff - 2 + 1.0 / mu_eff) / ((dim + 2) ** 2 + mu_eff))
    chi_N = np.sqrt(dim) * (1 - 1 / (4 * dim) + 1 / (21 * dim ** 2))

    C = np.eye(dim)
    Cs = np.zeros((dim, dim))
    p_sigma = np.zeros(dim)
    p_c = np.zeros(dim)

    x = params0.copy()
    sigma = sigma0
    best_local_J = compute_J(fourier_to_seq(params0, n, n_harm))
    best_local_x = params0.copy()
    tracker.update(fourier_to_seq(params0, n, n_harm))

    for gen in range(2000):
        if time.monotonic() > deadline:
            break

        offspring = []
        offspring_J = []
        for i in range(lam):
            z = rng.standard_normal(dim)
            y = cmaes_eig_half(C, z, dim)
            x_trial = x + sigma * y
            a_trial = fourier_to_seq(x_trial, n, n_harm)
            J_trial = compute_J(a_trial)
            offspring.append(x_trial)
            offspring_J.append(J_trial)

        idx = np.argsort(offspring_J)

        if offspring_J[idx[0]] < best_local_J:
            best_local_J = offspring_J[idx[0]]
            best_local_x = offspring[idx[0]].copy()
            a_best = fourier_to_seq(best_local_x, n, n_harm)
            tracker.update(a_best)

        # Update mean
        x_new = np.zeros(dim)
        for i in range(mu):
            x_new += weights[i] * offspring[idx[i]]

        # Update sigma
        if sigma > 1e-30:
            diff = (x_new - x) / (sigma * chi_N)
        else:
            diff = np.zeros(dim)
        p_sigma = (1 - c_sigma) * p_sigma + np.sqrt(c_sigma * (2 - c_sigma) * mu_eff) * diff
        sigma *= np.exp(c_sigma / d_sigma * (np.linalg.norm(p_sigma) / chi_N - 1))
        sigma = max(sigma, 1e-12)
        sigma = min(sigma, 0.5)

        # Update covariance
        h_sigma = 1.0
        if np.linalg.norm(p_sigma) / chi_N > (1.4 + c_c) * np.sqrt((1 + c_sigma) / (1 - c_sigma)):
            h_sigma = 0.0
        p_c = (1 - c_c) * p_c + h_sigma * np.sqrt(c_c * (2 - c_c) * mu_eff) * (x_new - x) / max(sigma, 1e-30)

        rank_1 = np.outer(p_c, p_c)
        rank_mu = np.zeros((dim, dim))
        for i in range(mu):
            d = offspring[idx[i]] - x
            rank_mu += weights[i] * np.outer(d, d)

        Cs = (1 - c_1 - c_mu) * Cs + c_1 * (rank_1 + h_sigma * (1 - h_sigma) * np.outer(p_c, p_c) * c_c) + c_mu * rank_mu
        C = (1 - c_1 - c_mu) * C + c_1 * (rank_1 + h_sigma * (1 - h_sigma) * np.outer(p_c, p_c) * c_c) + c_mu * rank_mu
        C = (C + C.T) / 2
        x = x_new

    return best_local_x, best_local_J


def lbb_objective(x):
    """Objective for L-BFGS-B: max of self-convolution (scale invariant)."""
    ax = np.maximum(x, 0.0)
    s = ax.sum()
    if s < 1e-8:
        return 1e10
    ax = ax / s
    b = fast_conv(ax)
    return float(b.max())
