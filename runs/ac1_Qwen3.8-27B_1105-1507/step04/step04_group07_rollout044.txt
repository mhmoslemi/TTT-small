"""
Minimizes J(a) = 2*n*max(convolve(a,a))/(sum(a))^2 using:
1. LP best-response with adaptive tightening
2. LSE (Log-Sum-Exp) gradient descent with adaptive temperature
3. Softmax reparameterization for unconstrained gradient steps
4. Multi-length search (512-1536) with diverse initializations
5. Spectral shaping for flat-autocorrelation initializations
6. Micro-perturbation local search for final polishing

import numpy as np
from scipy.optimize import linprog
from scipy.sparse import csr_matrix
import time

_spar_cache = {}

def get_sparsity(n):
    if n not in _spar_cache:
        rows = []
        cols = []
        for k in range(2 * n - 1):
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            rows.extend([k] * (je - js + 1))
            cols.extend(range(js, je + 1))
        _spar_cache[n] = (np.array(rows), np.array(cols), len(rows))
    return _spar_cache[n]

def fast_conv(a):
    n = len(a)
    s = 1
    while s < 2 * n - 1:
        s <<= 1
    fa = np.fft.rfft(a, s)
    return np.fft.irfft(fa * fa, s)[:2 * n - 1]

def compute_J(a):
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))

def lp_best_response(a, M, time_limit=15):
    n = len(a)
    rows, cols, nnz = get_sparsity(n)
    A = csr_matrix((a[rows - cols + (2*n-2 - (2*n-2))], (rows, cols)), shape=(2 * n - 1, n))
    # Build A properly: A[k, j] = a[k-j] if 0 <= k-j < n
    vals = np.empty(nnz)
    for i in range(nnz):
        k = rows[i]
        j = cols[i]
        idx = k - j
        vals[i] = a[idx] if 0 <= idx < n else 0.0
    A = csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                options={'presolve': True, 'time_limit': time_limit})
    if r.success and r.x is not None:
        g = np.maximum(r.x, 0.0)
        if g.sum() > 1e-10:
            return g
    return None

    a /= a.sum()
    n = len(a)

        gs = g.sum()
        if gs < 1e-10:

        g /= gs

        best_a = None

            for j in range(js, je + 1):
                grad_a[j] += a[k - j]
        grad_a /= len(active)

        # Chain rule for softmax: dJ/dx[j] = a[j]*(grad_a[j] - sum_i a[i]*grad_a[i])
        mean_grad = np.dot(a, grad_a)
        grad_x = a * (grad_a - mean_grad)

        v = momentum * v + (1.0 - momentum) * grad_x
        x_new = x - lr * v
        # Keep x in reasonable range
        x_new = np.clip(x_new, -30, 10)

        # Check improvement
        x_shifted_new = x_new - x_new.max()
        exp_new = np.exp(x_shifted_new)
        a_new = exp_new / exp_new.sum()
        J_new = compute_J(a_new)
        if J_new < current_J - 1e-15:
            x = x_new
            a = a_new
            current_J = J_new
            lr = min(lr * 1.05, 50.0 / n)
        else:
            lr *= 0.7
            v *= 0.5

        if it % 20 == 0:
            tracker.update(a)
    tracker.update(a)
    return a

def spectral_shaping_init(n, rng, dc_weight=1.0, sidelobe_level=0.1):
    """Generate sequence with shaped spectrum: strong DC, suppressed sidelobes."""
    half = n // 2 + 1
    magnitudes = np.ones(half) * sidelobe_level
    magnitudes[0] = dc_weight
    phases = rng.uniform(0, 2 * np.pi, half)
    A = magnitudes * np.exp(1j * phases)
    # Conjugate symmetry
    A_full = np.zeros(n, dtype=complex)
    A_full[:half] = A
    for k in range(1, n - n // 2):
        A_full[k] = np.conj(A_full[n - k])
    a = np.fft.ifft(A_full).real
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-10:
        return np.ones(n) / n
    return a / s

def flat_spectrum_init(n, rng):
    half = n // 2 + 1
    angles = rng.uniform(0, 2 * np.pi, half)
    A = np.zeros(n, dtype=complex)
    A[:half] = np.exp(1j * angles[:half])
    for k in range(1, n - n // 2):
        A[k] = np.conj(A[n - k])
    a = np.fft.ifft(A).real
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-10:
        return np.ones(n) / n
    return a / s

def resample_seq(src, dest_n):
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)

def generate_inits(n, rng, parent):
    inits = []
    if parent is not None and parent.size > 0:
        inits.append(resample_seq(parent, n))
        inits.append(resample_seq(parent[::-1], n))

    inits.append(np.ones(n) / n)
    inits.append(flat_spectrum_init(n, rng))
    inits.append(flat_spectrum_init(n, rng))
    inits.append(spectral_shaping_init(n, rng, dc_weight=1.0, sidelobe_level=0.05))
    inits.append(spectral_shaping_init(n, rng, dc_weight=1.0, sidelobe_level=0.1))
    inits.append(spectral_shaping_init(n, rng, dc_weight=1.0, sidelobe_level=0.2))

    t = np.linspace(0, 1, n, endpoint=False)
    for freq in [2, 3, 5, 7]:
        inits.append(1.0 + 0.3 * np.cos(2 * np.pi * freq * t + rng.uniform(0, 6.28)))
    inits.append(0.5 + 0.5 * np.cos(np.pi * t) ** 2)
    inits.append(1.0 + 0.2 * np.sin(2 * np.pi * t) + 0.15 * np.sin(4 * np.pi * t)
                + 0.1 * np.sin(7 * np.pi * t))

    for x in inits:
        x = np.maximum(x, 0.001).astype(np.float64)
        x /= x.sum()
    return inits

class ScoreTracker:
    def __init__(self):
        self.best = None
        self.best_score = float('inf')
        self.t_start = time.monotonic()

    def update(self, a):
        a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
        if a.size == 0 or a.sum() < 0.01:
            return
        score = float(evaluate_sequence(a.tolist()))
        if np.isfinite(score) and score < self.best_score:
            self.best = a.copy()
            self.best_score = score
            elapsed = time.monotonic() - self.t_start
            print(f"  * J={score:.8f} n={a.size} t={elapsed:.1f}s", flush=True)

    def report(self, label=""):
        if self.best is not None:
            print(f"  {label} best: J={self.best_score:.8f} n={self.best.size}", flush=True)

def micro_perturbation_search(a, max_steps, deadline, tracker):
    """Try small random perturbations to find local improvements."""
    a = a.copy().astype(np.float64)
    a /= a.sum()
    n = len(a)
    current_J = compute_J(a)
    tracker.update(a)
    rng = np.random.default_rng(12345)

    for step in range(max_steps):
        if time.monotonic() > deadline:
            break
        # Random small modification
        n_modify = rng.integers(1, max(2, n // 20) + 1)
        indices = rng.choice(n, size=n_modify, replace=False)
        a_trial = a.copy()
        for idx in indices:
            factor = rng.uniform(0.7, 1.3)
            a_trial[idx] *= factor
        a_trial = np.maximum(a_trial, 1e-10)
        a_trial /= a_trial.sum()
        J_trial = compute_J(a_trial)
        if J_trial < current_J - 1e-14:
            a = a_trial
            current_J = J_trial
            tracker.update(a)
    return a

def subgradient_polish(a, max_iters, deadline, tracker):
    """Polished subgradient descent for the exact max objective."""
    a = np.maximum(a.astype(np.float64), 1e-10)
    a /= a.sum()
    n = len(a)
    current_J = compute_J(a)
    tracker.update(a)
    lr = 2.0 / n
    rng = np.random.default_rng(777)

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        b = fast_conv(a)
        M = float(b.max())
        active = np.where(b >= M - 1e-12 * M)[0]

        grad = np.zeros(n)
        for k in active:
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            for j in range(js, je + 1):
                grad[j] += a[k - j]
        grad /= len(active)

        direction = grad - 2.0 * M * a
        a_new = a - lr * direction
        a_new = np.maximum(a_new, 1e-10)
        a_new /= a_new.sum()
        J_new = compute_J(a_new)
        if J_new < current_J - 1e-15:
            a = a_new
            current_J = J_new
            lr = min(lr * 1.02, 10.0 / n)
        else:
            lr *= 0.6
        if it % 30 == 0:
            tracker.update(a)
    tracker.update(a)
    return a

def optimize_sequence(a0, deadline, tracker, phase_budget):
    """Full optimization pipeline for a single sequence."""
    a = a0.copy().astype(np.float64)
    a = np.maximum(a, 1e-10)
    a /= a.sum()
    n = len(a)

    # LP phase
    n_lp = max(5, min(40, int(phase_budget * 0.35 / max(0.5, n / 256))))
    t_end_lp = min(deadline, time.monotonic() + phase_budget * 0.35)
    a = run_lp_opt(a, n_lp, t_end_lp, tracker, tighten=0.01)

    # LSE phase
    n_lse = max(100, int(phase_budget * 0.35 * 30000 / max(1, n)))
    t_end_lse = min(deadline, time.monotonic() + phase_budget * 0.35)
    a = lse_gradient_descent(a, n_lse, t_end_lse, tracker)

    # Softmax GD
    n_sm = max(50, int(phase_budget * 0.2 * 5000 / max(1, n)))
    t_end_sm = min(deadline, time.monotonic() + phase_budget * 0.2)
    a = softmax_gradient_descent(a, n_sm, t_end_sm, tracker)

    # Final LSE
    if time.monotonic() < deadline - 5:
        a = lse_gradient_descent(a, max(200, int(20000 / n)), deadline, tracker)

    return a

def propose_candidate(seed=42, budget_s=1500, **kwargs):
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + float(budget_s) - 15.0
    tracker = ScoreTracker()

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        if p.ndim == 1 and p.size > 0 and np.all(np.isfinite(p)):
            parent = np.maximum(p, 0.0)
            tracker.update(parent)
            print(f"Parent loaded: n={parent.size}, J={tracker.best_score:.8f}", flush=True)
    except (NameError, AttributeError):
        pass

    # === Phase 1: Improve parent (n=512) ===
    if parent is not None and time.monotonic() < deadline - 600:
        t0 = time.monotonic()
        n = parent.size
        print(f"\n{'='*50} Phase1: Improve parent n={n} {'='*50}", flush=True)

        a = parent.copy().astype(np.float64)
        a = np.maximum(a, 1e-10)
        a /= a.sum()

        # Aggressive LP
        t_end = min(deadline - 500, t0 + 180)
        a = run_lp_opt(a, 50, t_end, tracker, tighten=0.008)

        # LSE
        if time.monotonic() < t_end + 120:
            a = lse_gradient_descent(a, 5000, min(deadline - 400, t0 + 300), tracker)

        # Softmax
        if time.monotonic() < deadline - 300:
            a = softmax_gradient_descent(a, 2000, min(deadline - 200, t0 + 420), tracker)

        # Subgradient polish
        if time.monotonic() < deadline - 200:
            a = subgradient_polish(a, 3000, min(deadline - 100, t0 + 450), tracker)

        tracker.report(f"Phase1 ({time.monotonic()-t0:.1f}s): J={tracker.best_score:.8f}")

    # === Phase 2: Multi-length exploration ===
    # Try different lengths, allocating time based on what's available
    lengths_to_try = [640, 768, 1024]
    if parent is not None and parent.size != 512:
        lengths_to_try = [parent.size, parent.size + 128, parent.size + 256]

    remaining = deadline - time.monotonic() - 250  # reserve 250s for final phases
    if remaining > 100:
        time_per_length = remaining / len(lengths_to_try)

        for n in lengths_to_try:
            if n < 64 or time.monotonic() > deadline - 200:
                continue
            t_start = time.monotonic()
            t_end = min(deadline - 200, t_start + time_per_length)
            print(f"\n{'='*50} Phase2: n={n} budget={time_per_length:.0f}s {'='*50}", flush=True)

            inits = generate_inits(n, rng, parent)
            n_inits = min(len(inits), max(2, int(time_per_length / 30)))

            best_for_n = None
            best_J_n = float('inf')

            for idx in range(n_inits):
                if time.monotonic() > t_end - 10:
                    break
                a0 = inits[idx % len(inits)].copy()
                budget_init = max(15.0, (t_end - time.monotonic()))
                a = optimize_sequence(a0, t_end, tracker, budget_init)
                J = compute_J(a)
                if J < best_J_n:
                    best_J_n = J
                    best_for_n = a.copy()

            if best_for_n is not None:
                tracker.update(best_for_n)
                if parent is None or best_J_n < tracker.best_score:
                    parent = best_for_n
                print(f"  Phase2 n={n}: best J={best_J_n:.8f} ({time.monotonic()-t_start:.1f}s)", flush=True)

    # === Phase 3: Focused optimization on global best ===
    if tracker.best is not None and time.monotonic() < deadline - 150:
        t0 = time.monotonic()
        n = tracker.best.size
        print(f"\n{'='*50} Phase3: Focus n={n} {'='*50}", flush=True)

        a = tracker.best.copy().astype(np.float64)
        a = np.maximum(a, 1e-10)
        a /= a.sum()

        # Multiple LP passes with decreasing tighten
        tighten_sched = [0.005, 0.008, 0.012, 0.015, 0.02]
        for tight in tighten_sched:
            if time.monotonic() > deadline - 100:
                break
            a = run_lp_opt(a, 20, deadline - 80, tracker, tighten=tight)

        # LSE
        if time.monotonic() < deadline - 60:
            a = lse_gradient_descent(a, 5000, deadline - 40, tracker)

        # Softmax
        if time.monotonic() < deadline - 30:
            a = softmax_gradient_descent(a, 2000, deadline - 15, tracker)

        tracker.report(f"Phase3 ({time.monotonic()-t0:.1f}s)")

    # === Phase 4: Micro-perturbation + final LSE ===
    if tracker.best is not None and time.monotonic() < deadline - 10:
        t0 = time.monotonic()
        print(f"\n{'='*50} Phase4: Micro search {'='*50}", flush=True)

        a = tracker.best.copy().astype(np.float64)
        a = micro_perturbation_search(a, 5000, deadline - 20, tracker)

        # Final LSE polish
        if time.monotonic() < deadline - 5:
            a = lse_gradient_descent(a, 1000, deadline - 2, tracker)

        # Final micro
        if time.monotonic() < deadline - 1:
            a = micro_perturbation_search(a, 2000, deadline - 0.5, tracker)

        tracker.report(f"Phase4 ({time.monotonic()-t0:.1f}s)")

    # Fallback
    if tracker.best is None:
        fallback = np.ones(512) / 512
        tracker.best = fallback
        tracker.best_score = compute_J(fallback)
        tracker.update(tracker.best)

    print(f"\n{'='*50} FINAL {'='*50}", flush=True)
    print(f"J={tracker.best_score:.8f} n={tracker.best.size} time={time.monotonic()-tracker.t_start:.1f}s", flush=True)
    return [float(x) for x in tracker.best]
