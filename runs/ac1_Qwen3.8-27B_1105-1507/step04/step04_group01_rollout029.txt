"""
Alternating LP optimization for minimizing 2*n*max(conv(a,a))/(sum(a))^2.
Key insight: conditional on one sequence being fixed, the subproblem is a
linear program. We alternate between solving LPs for each sequence, which
is exact (not a gradient approximation). Combined with multiple restarts
and symmetry exploitation, this should outperform gradient-based methods.
"""

import numpy as np
from scipy.optimize import linprog
from scipy.fft import rfft, irfft
import time


def compute_conv(a):
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    conv = irfft(fa * fa, size)
    return conv[:2 * n - 1]


def compute_obj(a):
    n = len(a)
    if n == 0:
        return np.inf
    conv = compute_conv(a)
    max_b = float(np.max(conv))
    sum_a = float(np.sum(a))
    if sum_a < 0.01:
        return np.inf
    return 2.0 * n * max_b / (sum_a * sum_a)


def fft_convolve(a, b):
    n, m = len(a), len(b)
    size = 1
    while size < n + m - 1:
        size <<= 1
    fa = rfft(a, size)
    fb = rfft(b, size)
    return irfft(fa * fb, size)[:n + m - 1]


def build_conv_matrix(a):
    """Build matrix T (2n-1 x n) such that T @ x = a * x (full linear convolution)."""
    n = len(a)
    T = np.zeros((2 * n - 1, n))
    for i in range(n):
        start = i
        end = min(i + n, 2 * n - 1)
        T[start:end, i] = a[:end - start]
    return T


def solve_lp_given_fixed(fixed):
    """
    Given fixed sequence f (length n), solve:
    min M
    s.t. (f * x)[k] <= M for k = 0, ..., 2n-2
         sum(x) = 1
         x >= 0
    Returns (x_opt, M_opt) or (None, None).
    """
    n = len(fixed)
    # Build constraint matrix: (f * x)[k] = sum_i f[k-i] * x[i]
    # T[k, i] = f[k-i] if 0 <= k-i < n
    T = np.zeros((2 * n - 1, n))
    for i in range(n):
        start = i
        end = min(i + n, 2 * n - 1)
        T[start:end, i] = fixed[:end - start]
    
    # Variables: x[0], ..., x[n-1], M
    # Objective: min M
    c = np.zeros(n + 1)
    c[n] = 1.0
    
    # Inequality: T @ x - M <= 0
    A_ub = np.zeros((2 * n - 1, n + 1))
    A_ub[:, :n] = T
    A_ub[:, n] = -1.0
    b_ub = np.zeros(2 * n - 1)
    
    # Equality: sum(x) = 1
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])
    
    bounds = [(0.0, None)] * n + [(0.0, None)]
    
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': 60})
        if res.status == 0:
            x = np.maximum(res.x[:n], 0)
            M = float(max(res.x[n], 0))
            s = np.sum(x)
            if s < 1e-10:
                return None, None
            x = x / s  # ensure sum = 1 (numerical)
            return x, M
    except Exception:
        pass
    return None, None


def alternating_lp(a_init, max_iters=50, time_budget=60.0, verbose=False):
    """
    Alternating LP: alternate between solving LPs for a and g.
    At each step, fix one sequence and optimally solve for the other.
    """
    n = len(a_init)
    a = a_init.copy().astype(np.float64)
    s = np.sum(a)
    if s < 1e-10:
        a = np.ones(n) / n
    else:
        a = a / s
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    stall = 0
    
    for it in range(max_iters):
        if time.monotonic() - start > time_budget:
            break
        
        elapsed = time.monotonic() - start
        remaining = time_budget - elapsed
        if remaining < 2:
            break
        
        # Step 1: Given a, find best g
        g, M1 = solve_lp_given_fixed(a)
        if g is None:
            break
        
        # Step 2: Given g, find best a_new
        a_new, M2 = solve_lp_given_fixed(g)
        if a_new is None:
            break
        
        # Evaluate
        obj_new = compute_obj(a_new)
        if verbose:
            print(f"    ALP iter {it}: M1={M1:.8f}, M2={M2:.8f}, obj={obj_new:.8f}, best={best_obj:.8f}")
        
        if obj_new < best_obj - 1e-14:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new
            stall = 0
        else:
            # Try blending
            improved = False
            for t in [0.3, 0.5, 0.7]:
                a_blend = (1 - t) * best_a + t * a_new
                a_blend = a_blend / max(np.sum(a_blend), 1e-10)
                obj_blend = compute_obj(a_blend)
                if obj_blend < best_obj - 1e-14:
                    best_obj = obj_blend
                    best_a = a_blend.copy()
                    a = a_blend
                    improved = True
                    break
            if not improved:
                stall += 1
                if stall >= 3:
                    break
    
    return best_a, best_obj

        # sum(x) = 1 means: 2*sum(x[0:half-1]) + (x[half-1] if n is even else 0) = 1
        # Actually for symmetric x of length n:
        # If n is odd: x = [x0, x1, ..., x_{(n-1)/2}, ..., x1, x0]
        #   sum = 2*(x0+...+x_{(n-3)/2}) + x_{(n-1)/2}
        # If n is even: x = [x0, x1, ..., x_{n/2-1}, x_{n/2-1}, ..., x1, x0]
        #   sum = 2*(x0+...+x_{n/2-1})
        
        # For the LP, I'll just use the standard formulation but project back to symmetric
        # after each LP solve. This is simpler and still effective.
        
        g, M1 = solve_lp_given_fixed(a)
        if g is None:
            break
        # Project to symmetric
        g = 0.5 * (g + g[::-1])
        g = np.maximum(g, 0)
        gs = np.sum(g)
        if gs < 1e-10:
            break
        g = g / gs
        
        a_new, M2 = solve_lp_given_fixed(g)
        if a_new is None:
            break
        a_new = 0.5 * (a_new + a_new[::-1])
        a_new = np.maximum(a_new, 0)
        a_ns = np.sum(a_new)
        if a_ns < 1e-10:
            break
        a_new = a_new / a_ns
        
        obj_new = compute_obj(a_new)
        
        if obj_new < best_obj - 1e-14:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new
            stall = 0
        else:
            improved = False
            for t in [0.3, 0.5, 0.7]:
                a_blend = (1 - t) * best_a + t * a_new
                a_blend = 0.5 * (a_blend + a_blend[::-1])
                a_blend = np.maximum(a_blend, 0)
                s_b = np.sum(a_blend)
                if s_b < 1e-10:

                if obj_blend < best_obj - 1e-14:
                    best_obj = obj_blend
                    best_a = a_blend.copy()

                    break

    return best_a, best_obj


def random_init_sym(n, seed):
    """Generate symmetric random initialization."""
    rng = np.random.default_rng(seed)
    k = np.arange(n)
    choice = rng.integers(0, 6)
    
    if choice == 0:
        f = rng.uniform(0.5, 3.0)
        phase = rng.uniform(0, np.pi)
        a = 1.0 + 0.7 * np.cos(2 * np.pi * f * k / n + phase)
    elif choice == 1:
        f1 = rng.uniform(0.3, 1.5)
        f2 = rng.uniform(2.0, 5.0)
        a = 1.0 + 0.3 * np.cos(2 * np.pi * f1 * k / n) + 0.3 * np.cos(2 * np.pi * f2 * k / n)
    elif choice == 2:
        # Uniform with noise
        a = 1.0 + 0.5 * rng.standard_normal(n)
    elif choice == 3:
        # Bump
        center = n * rng.uniform(0.3, 0.7)
        width = n * rng.uniform(0.1, 0.3)
        a = 1.0 + 0.8 * np.exp(-0.5 * ((k - center) / width) ** 2)
    elif choice == 4:
        # Multi-frequency
        a = np.ones(n)
        for _ in range(rng.integers(2, 5)):
            f = rng.uniform(0.5, 6.0)
            amp = rng.uniform(0.1, 0.4)
            a += amp * np.cos(2 * np.pi * f * k / n + rng.uniform(0, 2 * np.pi))
    else:
        a = rng.random(n) + 0.2
    
    # Enforce symmetry
    a = 0.5 * (a + a[::-1])
    a = np.maximum(a, 0.01)
    a = a / np.sum(a)
    return a


def resample_sequence(a, new_n):
    n = len(a)
    if n == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return a_new

def subgradient_step(a, tau, step_size):
    """One subgradient step with smoothing."""
    n = len(a)
    conv = compute_conv(a)
    max_conv = float(np.max(conv))
    shifted = np.clip(tau * (conv - max_conv), -300, 300)
    weights = np.exp(shifted)
    w_sum = np.sum(weights)
    if w_sum < 1e-300:
        idx = int(np.argmax(conv))
        weights = np.zeros(len(conv))
        weights[idx] = 1.0
    else:
        weights /= w_sum
    
    # grad_a[i] = 2 * sum_j weights[i+j] * a[j] (for the smoothed max)
    # More precisely: grad of soft_max(conv) w.r.t. a[i]
    # d conv[k] / d a[i] = a[k-i] + a[i-(k-n+1)] ... actually it's 2*sum_j w[k]*a[k-i]
    # Let me use: grad_a[i] = 2 * (w_convolve_a)[n-1-i] where w_convolve_a = convolve(w, a)
    # Actually: d/d a[i] sum_k w[k] * conv[k] = 2 * sum_k w[k] * a[k-i] (for valid k-i)
    # = 2 * (w * a)[n-1-i] ... let me just compute it directly
    
    # d conv[k] / d a[i] = a[k-i] (if 0<=k-i<n) + a[i+k-n+1] (if 0<=i+k-n+1<n)
    # For the first part: sum_k w[k] * a[k-i] for valid k-i = (w * a_rev)[i] ... 
    
    # Simpler: use the fact that d/d a[i] of conv[k] = a[k-i] (when k-i in [0,n-1])
    # So grad_a[i] = 2 * sum_k w[k] * a[k-i] (for k-i in [0,n-1])
    #              = 2 * sum_j w[i+j] * a[j] (for j in [0,n-1], i+j in [0,2n-2])
    
    w_rev = weights[::-1]
    conv_wa = fft_convolve(a, w_rev)
    # conv_wa[m] = sum_j a[j] * w_rev[m-j] = sum_j a[j] * weights[2n-2-m+j]
    # We want: sum_j w[i+j] * a[j] for j=0..n-1
    # = sum_j a[j] * w[i+j]
    # Let i+j = p, so j = p-i: sum_p a[p-i] * w[p] for p = i..i+n-1 (capped)
    # This is (w * a)[i+n-1] ... hmm let me just do it the standard way
    
    # Actually the gradient of soft_max(conv(a,a)) w.r.t. a[i]:
    # = 2 * sum_k w[k] * d conv[k] / d a[i]
    # d conv[k] / d a[i] = a[k-i] if 0 <= k-i < n, else 0
    # So grad_a[i] = 2 * sum_{k: 0<=k-i<n} w[k] * a[k-i]
    #              = 2 * sum_{j=0}^{n-1} w[i+j] * a[j]  (where w[i+j] = 0 if i+j >= 2n-1)
    
    # This is the cross-correlation of w and a, evaluated at lag i
    # = 2 * (w cross-correl a)[i]
    # Using FFT: cross-correlation can be computed as ifft(conj(fft(w)) * fft(a))
    # But w has length 2n-1 and a has length n...
    
    # Let me just use the direct formula via convolution:
    # sum_j w[i+j] * a[j] = (w * a_rev)[i + n - 1] where a_rev = a[::-1]
    # Wait: (w * a_rev)[m] = sum_j w[j] * a_rev[m-j] = sum_j w[j] * a[n-1-(m-j)]
    # = sum_j w[j] * a[n-1-m+j]
    # Set n-1-m+j = j' → j = j' + m - n + 1
    # = sum_{j'} w[j'+m-n+1] * a[j']
    
    # Hmm, let me just use the approach from the previous code which worked:
    w_rev = weights[::-1]
    conv_fw = fft_convolve(a, w_rev)
    # From previous code: grad_a = 2.0 * conv_fw[n-1:2*n-1][::-1]
    grad_a = 2.0 * conv_fw[n - 1:2 * n - 1][::-1]
    
    # Center the gradient (since we optimize on the simplex)
    grad_centered = grad_a - np.mean(grad_a)
    a_new = a - step_size * grad_centered
    a_new = np.maximum(a_new, 0)
    s_new = np.sum(a_new)
    if s_new > 1e-10:
        a_new = a_new / s_new
    return a_new


def subgradient_optimize(a, max_iters=1000, time_budget=20.0, tau_start=3.0, tau_end=50.0):
    """Smoothed subgradient optimization."""
    n = len(a)
    a = a.copy().astype(np.float64)
    s = np.sum(a)
    if s < 1e-10:
        a = np.ones(n) / n
    else:
        a = a / s
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    step = 0.1
    stall = 0
    
    for it in range(max_iters):
        if time.monotonic() - start > time_budget:
            break
        
        frac = it / max(1, max_iters)
        tau = tau_start + (tau_end - tau_start) * frac
        
        a = subgradient_step(a, tau, step)
        obj = compute_obj(a)
        
        if obj < best_obj - 1e-12:
            best_obj = obj
            best_a = a.copy()
            stall = 0
        else:
            stall += 1
            if stall > 50:
                step *= 0.9
                if step < 1e-8:
                    break
        
        if it > 50:
            step *= 0.999
    
    return best_a

"""
Alternating LP optimization for minimizing 2*n*max(conv(a,a))/(sum(a))^2.
Key insight: conditional on one sequence being fixed, the subproblem is a
linear program solved exactly. We alternate between solving LPs, combined with
multiple restarts, symmetry exploitation, and subgradient polishing.

import numpy as np
from scipy.optimize import linprog
from scipy.fft import rfft, irfft
import time


def compute_conv(a):
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    conv = irfft(fa * fa, size)
    return conv[:2 * n - 1]


def compute_obj(a):
    n = len(a)
    if n == 0:
        return np.inf
    conv = compute_conv(a)
    max_b = float(np.max(conv))
    sum_a = float(np.sum(a))
    if sum_a < 0.01:
        return np.inf
    return 2.0 * n * max_b / (sum_a * sum_a)


def fft_convolve(a, b):
    n, m = len(a), len(b)
    size = 1
    while size < n + m - 1:
        size <<= 1
    fa = rfft(a, size)
    fb = rfft(b, size)
    return irfft(fa * fb, size)[:n + m - 1]


def solve_lp_given_fixed(fixed, time_limit=30):
    """
    Given fixed sequence f (length n), solve:
    min M
    s.t. (f * x)[k] <= M for k = 0, ..., 2n-2
         sum(x) = 1
         x >= 0, M >= 0
    Returns (x_opt, M_opt) or (None, None).
    """
    n = len(fixed)
    T = np.zeros((2 * n - 1, n))
    for i in range(n):
        start = i
        end = min(i + n, 2 * n - 1)
        T[start:end, i] = fixed[:end - start]

    c = np.zeros(n + 1)
    c[n] = 1.0

    A_ub = np.zeros((2 * n - 1, n + 1))
    A_ub[:, :n] = T
    A_ub[:, n] = -1.0
    b_ub = np.zeros(2 * n - 1)

    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])

    bounds = [(0.0, None)] * n + [(0.0, None)]

    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': time_limit})
        if res.status == 0:
            x = np.maximum(res.x[:n], 0)
            M = float(max(res.x[n], 0))
            s = np.sum(x)
            if s < 1e-10:
                return None, None
            x = x / s
            return x, M
    except Exception:
        pass
    return None, None


def alternating_lp(a_init, max_iters=50, time_budget=60.0, verbose=False):
    """Alternating LP: fix one sequence, solve LP for the other, iterate."""
    n = len(a_init)
    a = a_init.copy().astype(np.float64)
    s = np.sum(a)
    if s < 1e-10:
        a = np.ones(n) / n
    else:
        a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    stall = 0

    for it in range(max_iters):
        if time.monotonic() - start > time_budget:
            break
        remaining = time_budget - (time.monotonic() - start)
        if remaining < 3:
            break

        g, M1 = solve_lp_given_fixed(a, time_limit=max(5, int(remaining * 0.4)))
        if g is None:
            break

        a_new, M2 = solve_lp_given_fixed(g, time_limit=max(5, int(remaining * 0.4)))
        if a_new is None:
            break

        obj_new = compute_obj(a_new)
        if verbose:
            print(f"    ALP iter {it}: M1={M1:.8f}, M2={M2:.8f}, obj={obj_new:.8f}, best={best_obj:.8f}")

        if obj_new < best_obj - 1e-14:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new
            stall = 0
        else:
            # Try blending
            improved = False
            for t in [0.3, 0.5, 0.7, 1.0]:
                a_blend = (1 - t) * best_a + t * a_new
                s_b = np.sum(a_blend)
                if s_b < 1e-10:
                    continue
                a_blend = a_blend / s_b
                obj_blend = compute_obj(a_blend)
                if obj_blend < best_obj - 1e-14:
                    best_obj = obj_blend
                    best_a = a_blend.copy()
                    a = a_blend
                    improved = True
                    break
            if not improved:
                stall += 1
                if stall >= 4:
                    break

    return best_a, best_obj

def subgradient_step(a, tau, step_size):
    """One smoothed subgradient step."""
    n = len(a)
    conv = compute_conv(a)
    max_conv = float(np.max(conv))
    shifted = np.clip(tau * (conv - max_conv), -300, 300)
    weights = np.exp(shifted)
    w_sum = np.sum(weights)
    if w_sum < 1e-300:
        idx = int(np.argmax(conv))
        weights = np.zeros(len(conv))
        weights[idx] = 1.0
    else:
        weights /= w_sum

    w_rev = weights[::-1]
    conv_fw = fft_convolve(a, w_rev)
    grad_a = 2.0 * conv_fw[n - 1:2 * n - 1][::-1]

    grad_centered = grad_a - np.mean(grad_a)
    a_new = a - step_size * grad_centered
    a_new = np.maximum(a_new, 0)
    s_new = np.sum(a_new)
    if s_new > 1e-10:
        a_new = a_new / s_new
    return a_new


def subgradient_optimize(a, max_iters=1000, time_budget=20.0, tau_start=3.0, tau_end=50.0):
    """Smoothed subgradient optimization."""
    n = len(a)
    a = a.copy().astype(np.float64)
    s = np.sum(a)
    if s < 1e-10:
        a = np.ones(n) / n
    else:
        a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    step = 0.1
    stall = 0

    for it in range(max_iters):
        if time.monotonic() - start > time_budget:
            break

        frac = it / max(1, max_iters)
        tau = tau_start + (tau_end - tau_start) * frac

        a = subgradient_step(a, tau, step)
        obj = compute_obj(a)

        if obj < best_obj - 1e-12:
            best_obj = obj
            best_a = a.copy()
            stall = 0
        else:
            stall += 1
            if stall > 50:
                step *= 0.9
                if step < 1e-8:
                    break

        if it > 50:
            step *= 0.999

    return best_a

def resample_sequence(a, new_n):
    n = len(a)
    if n == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return a_new

            remaining_init = deadline - time.monotonic()
            if remaining_init > 5:
                a = subgradient_optimize(a_alp, max_iters=int(remaining_init * 8),
                                          time_budget=min(4.0, remaining_init * 0.15),
                                          tau_start=8.0, tau_end=35.0)
                obj_final = compute_obj(a)
                if obj_final < obj_alp:
                    a_alp = a
                    obj_alp = obj_final

            print(f"  init{init_idx}: n={n}, obj={obj_alp:.6f}")

            if obj_alp < best_score:
                best_score = obj_alp
                best_seq = a_alp.copy()
                print(f"  ** UPDATE: n={n}, obj={best_score:.6f} **")

            candidates.append((a_alp.copy(), obj_alp))

    # === PHASE 2: Polishing top candidates ===
    print(f"\n=== POLISHING PHASE ===")
    print(f"Best before polish: {best_score:.6f}")

    candidates.sort(key=lambda x: x[1])

    n_polish = min(4, len(candidates))
    for ci in range(n_polish):
        if time.monotonic() > deadline - 50:
            break
        remaining = deadline - time.monotonic()
        if remaining < 20:
            break

        a_cand = candidates[ci][0]
        n_cand = len(a_cand)
        print(f"  Polishing cand {ci} (n={n_cand}, obj={candidates[ci][1]:.6f}), {remaining:.0f}s left")

        polish_budget = min(remaining * 0.35, 70.0)
        a_pol, obj_pol = polish_candidate(a_cand, polish_budget, verbose=True)

        if obj_pol < best_score:
            best_score = obj_pol
            best_seq = a_pol.copy()
            print(f"  ** UPDATE: obj={best_score:.6f} **")

    # === PHASE 3: Final intensive polish ===
    if best_seq is not None and time.monotonic() < deadline - 20:
        remaining = deadline - time.monotonic()
        n_best = len(best_seq)
        print(f"\n=== FINAL POLISH: n={n_best}, {remaining:.0f}s ===")

        a_cur = best_seq.copy()
        for rnd in range(6):
            if time.monotonic() > deadline - 10:
                break
            remaining = deadline - time.monotonic()
            if remaining < 12:
                break

            a_new, obj_new = alternating_lp(a_cur, max_iters=30,
                                            time_budget=min(remaining * 0.4, 35.0))
            if obj_new < best_score:
                best_score = obj_new
                best_seq = a_new.copy()
                a_cur = a_new
                print(f"  Final ALP r{rnd}: {best_score:.6f}")

            remaining = deadline - time.monotonic()
            if remaining > 15:
                a_sym, obj_sym = alternating_lp_symmetric(a_cur, max_iters=25,
                                                          time_budget=min(remaining * 0.3, 25.0))
                if obj_sym < best_score:
                    best_score = obj_sym
                    best_seq = a_sym.copy()
                    a_cur = a_sym
                    print(f"  Final SymALP r{rnd}: {best_score:.6f}")

            remaining = deadline - time.monotonic()
            if remaining > 10:
                a_sg = subgradient_optimize(a_cur, max_iters=int(remaining * 6),
                                             time_budget=min(remaining * 0.2, 12.0),
                                             tau_start=12.0, tau_end=50.0)
                obj_sg = compute_obj(a_sg)
                if obj_sg < best_score:
                    best_score = obj_sg
                    best_seq = a_sg.copy()
                    a_cur = a_sg
                    print(f"  Final SG r{rnd}: {best_score:.6f}")

    # Final validation
    if best_seq is None:
        best_seq = np.ones(200)
        best_score = compute_obj(best_seq)

    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    val = evaluate_sequence(result)
    if val < np.inf:
        best_score = val
    else:
        result = [1.0] * 100
        best_score = 2.0

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_score:.6f}, time={elapsed:.1f}s ===")

    return result
