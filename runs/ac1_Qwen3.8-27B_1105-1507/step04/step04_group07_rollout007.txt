import numpy as np
from scipy.optimize import linprog
from scipy.sparse import csr_matrix

# Helper functions (top-level, no closures)

_spar_cache = {}

def get_sparsity(n):
    if n not in _spar_cache:
        rows = []
        cols = []
        for k in range(2 * n - 1):
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            rows.extend([k] * (je - js + 1))
            cols.extend(range(js, je + 1))
        _spar_cache[n] = (np.array(rows), np.array(cols), len(rows))
    return _spar_cache[n]

def fast_conv(a):
    n = len(a)
    s = 1
    while s < 2 * n - 1:
        s <<= 1
    fa = np.fft.rfft(a, s)
    return np.fft.irfft(fa * fa, s)[:2 * n - 1]

def compute_J(a):
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))

def fft_cross_correlation(c, a):
    """Compute cross-correlation sum_m c[j+m]*a[m] for j=0,...,len(a)-1"""
    n = len(a)
    nc = len(c)
    size = 1
    while size < nc + n - 1:
        size <<= 1
    C = np.fft.rfft(c, size)
    A = np.fft.rfft(a, size)
    cross = np.fft.irfft(np.conj(C) * A, size)
    return cross[:n]

def lse_gradient(b, tau, a):
    """Compute LSE gradient w.r.t. a using FFT cross-correlation."""
    b_shifted = b - b.max()
    exp_b = np.exp(tau * b_shifted)
    p = exp_b / exp_b.sum()
    # LSE gradient w.r.t. a[j] = sum_k p[k] * db[k]/da[j]
    # = sum_k p[k] * a[k-j] (for valid k-j)
    # This is cross-correlation of p and a
    return fft_cross_correlation(p, a)

def lp_best_response(a, M):
    n = len(a)
    rows, cols, nnz = get_sparsity(n)

def flat_spectrum_init(n, rng):
    half = n // 2 + 1
    angles = rng.uniform(0, 2 * np.pi, half)
    A = np.zeros(n, dtype=complex)
    A[:half] = np.exp(1j * angles[:half])
    for k in range(1, n - n // 2):
        A[k] = np.conj(A[n - k])
    a = np.fft.ifft(A).real
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-10:
        return np.ones(n) / n
    return a / s

def resample_seq(src, dest_n):
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)

    return a

"""
Minimizes J(a) = 2*n*max(convolve(a,a))/(sum(a))^2 using:
1. FFT-based vectorized LSE gradient descent (major speedup over explicit loops)
2. LP best-response with adaptive tightening schedules
3. Multi-length exploration (256, 512, 768) with diverse inits
4. Simulated annealing with LSE polishing
5. Coordinate descent final polish
Key improvement: O(n log n) gradient via FFT cross-correlation instead of O(n*|active|)
"""

import numpy as np
from scipy.optimize import linprog
from scipy.sparse import csr_matrix
import time

_spar_cache = {}

def get_sparsity(n):
    if n not in _spar_cache:
        rows = []
        cols = []
        for k in range(2 * n - 1):
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            rows.extend([k] * (je - js + 1))
            cols.extend(range(js, je + 1))
        _spar_cache[n] = (np.array(rows), np.array(cols), len(rows))
    return _spar_cache[n]

def fast_conv(a):
    n = len(a)
    s = 1
    while s < 2 * n - 1:
        s <<= 1
    fa = np.fft.rfft(a, s)
    return np.fft.irfft(fa * fa, s)[:2 * n - 1]

def compute_J(a):
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))

def fft_cross_correlation(c, a):
    """Compute sum_m c[j+m]*a[m] for j=0,...,n-1 using FFT."""
    n = len(a)
    nc = len(c)
    size = 1
    while size < nc + n - 1:
        size <<= 1
    C = np.fft.rfft(c, size)
    A = np.fft.rfft(a, size)
    cross = np.fft.irfft(np.conj(C) * A, size)
    return cross[:n]

def lse_optimize(a, max_iters, deadline, tracker, tau_start=30.0, tau_end=1000.0):
    """LSE (Log-Sum-Exp) smoothed optimization with FFT-based gradients."""
    a = np.maximum(a.astype(np.float64), 1e-8)
    a /= a.sum()
    n = len(a)
    current_J = compute_J(a)
    tracker.update(a)
    tau = tau_start
    lr = 1.0 / (n * tau)
    momentum = 0.8
    v = np.zeros(n)
    stall_count = 0

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        b = fast_conv(a)
        b_max = b.max()
        exp_b = np.exp(tau * (b - b_max))
        p = exp_b / exp_b.sum()
        grad = fft_cross_correlation(p, a)

        v = momentum * v + (1.0 - momentum) * grad
        a_new = a - lr * v
        a_new = np.maximum(a_new, 1e-8)
        a_new /= a_new.sum()
        J_new = compute_J(a_new)

        if J_new < current_J - 1e-16:
            a = a_new
            current_J = J_new
            stall_count = 0
            tau = min(tau_end, tau * 1.015)
            lr = 1.0 / (n * tau)
        else:
            stall_count += 1
            lr = max(lr * 0.6, 1e-10 / n)
            v *= 0.3
            if stall_count > 50 or lr < 1e-12 / n:
                break
        if it % 100 == 0:
            tracker.update(a)
    tracker.update(a)
    return a

def subgradient_optimize(a, max_iters, deadline, tracker):
    """Subgradient descent on the exact max using FFT cross-correlation."""
    a = np.maximum(a.astype(np.float64), 1e-8)
    a /= a.sum()
    n = len(a)
    current_J = compute_J(a)
    tracker.update(a)
    lr = 2.0 / n

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        b = fast_conv(a)
        M = float(b.max())
        # Use high-tau LSE as approximate subgradient
        p = np.exp(200.0 * (b - M))
        p /= p.sum()
        grad = fft_cross_correlation(p, a)
        # Direction: minimize J = 2n*M (with s=1), grad_J = 2n*grad
        # But we also want to maintain normalization, so use: grad - 2*M*a (projected)
        direction = grad - 2.0 * M * a
        a_new = a - lr * direction
        a_new = np.maximum(a_new, 1e-8)
        a_new /= a_new.sum()
        J_new = compute_J(a_new)
        if J_new < current_J - 1e-15:
            a = a_new
            current_J = J_new
            lr = min(lr * 1.01, 10.0 / n)
        else:
            lr *= 0.7
        if it % 50 == 0:
            tracker.update(a)
    tracker.update(a)
    return a

def lp_best_response(a, M, time_limit=6):
    n = len(a)
    rows, cols, nnz = get_sparsity(n)
    vals = np.empty(nnz)
    for i in range(nnz):
        k = rows[i]
        j = cols[i]
        vals[i] = a[k - j]
    A = csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                options={'presolve': True, 'time_limit': time_limit})
    if r.success and r.x is not None:
        g = np.maximum(r.x, 0.0)
        if g.sum() > 1e-10:
            return g / g.sum()
    return None

def run_lp_opt(a, max_iters, deadline, tracker, tighten=0.01):
    """LP best-response iteration with adaptive step size."""
    a = np.maximum(a.astype(np.float64), 1e-8)
    a /= a.sum()
    n = len(a)
    current_J = compute_J(a)
    tracker.update(a)
    fail_count = 0

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        b = fast_conv(a)
        M = float(b.max())
        M_tight = M * (1.0 - tighten)

        g = lp_best_response(a, M_tight, time_limit=5)
        if g is None:
            g = lp_best_response(a, M, time_limit=5)
        if g is None:
            fail_count += 1
            if fail_count > 3:
                break
            tighten = max(0.001, tighten * 0.5)
            continue
        fail_count = 0

        # Line search over step sizes
        best_a = None
        best_J = current_J
        progress = it / max(1, max_iters)
        base_t = max(0.05, 0.5 * (1.0 - progress))
        candidates = [base_t, base_t * 1.5, base_t * 0.5, 0.8, 0.3, 0.1, 0.95]
        for t in candidates:
            if t <= 0 or t > 1.0:
                continue
            a_trial = (1.0 - t) * a + t * g
            a_trial = np.maximum(a_trial, 0.0)
            ss = a_trial.sum()
            if ss < 1e-10:
                continue
            a_trial /= ss
            J_trial = compute_J(a_trial)
            if J_trial < best_J - 1e-15:
                best_J = J_trial
                best_a = a_trial

        if best_a is None:
            tighten = max(0.001, tighten * 0.7)
            if tighten < 0.002:
                fail_count += 1
                if fail_count > 5:
                    break
            continue

        a = best_a
        current_J = best_J
        if it % 5 == 0:
            tracker.update(a)
    tracker.update(a)
    return a

def flat_spectrum_init(n, rng):
    half = n // 2 + 1
    angles = rng.uniform(0, 2 * np.pi, half)
    A = np.zeros(n, dtype=complex)
    A[:half] = np.exp(1j * angles[:half])
    for k in range(1, n - n // 2):
        A[k] = np.conj(A[n - k])
    a = np.fft.ifft(A).real
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-10:
        return np.ones(n) / n
    return a / s

def resample_seq(src, dest_n):
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)

def generate_inits(n, rng, parent):
    inits = []
    if parent is not None and parent.size > 0:
        inits.append(resample_seq(parent, n))
        inits.append(resample_seq(parent[::-1], n))

    inits.append(np.ones(n) / n)
    inits.append(rng.uniform(0.1, 1.0, n))
    for _ in range(4):
        inits.append(flat_spectrum_init(n, rng))

    t = np.linspace(0, 1, n, endpoint=False)
    for freq in [2, 3, 5, 7, 11, 13, 17]:
        inits.append(1.0 + 0.35 * np.cos(2 * np.pi * freq * t + rng.uniform(0, 6.28)))
    inits.append(1.0 + 0.2 * np.cos(2 * np.pi * 3.7 * t) + 0.15 * np.cos(2 * np.pi * 7.3 * t))
    inits.append(0.5 + 0.5 * np.cos(np.pi * t) ** 2)

    # Multi-sine
    inits.append(1.0 + 0.2 * np.sin(2 * np.pi * t) + 0.15 * np.sin(4 * np.pi * t)
                + 0.1 * np.sin(6 * np.pi * t) + 0.05 * np.sin(10 * np.pi * t))

    # Gaussian bumps
    inits.append(0.3 + 0.7 * np.exp(-8 * (t - 0.25) ** 2) + 0.3 * np.exp(-8 * (t - 0.75) ** 2))

    for x in inits:
        x = np.maximum(x, 0.001).astype(np.float64)
        x /= x.sum()
    return inits

def simulated_annealing(a, deadline, tracker, n_restarts=20, t0=0.08):
    rng = np.random.default_rng(777)
    n = a.size
    a = a.copy().astype(np.float64)
    current_J = compute_J(a)
    tracker.update(a)
    temp = t0

    for r in range(n_restarts):
        if time.monotonic() > deadline:
            break
        a_pert = a + rng.normal(0, temp, n)
        a_pert = np.maximum(a_pert, 1e-6)
        a_pert /= a_pert.sum()

        # LSE refinement from perturbed state
        a_refined = lse_optimize(a_pert, max(30, 80 - r * 2), deadline, tracker,
                                 tau_start=50, tau_end=300)
        J = compute_J(a_refined)
        delta = J - current_J
        if delta < 0 or rng.random() < np.exp(-delta / max(temp, 1e-10)):
            a = a_refined
            current_J = J
        temp *= 0.92
    tracker.update(a)
    return a

def coordinate_descent(a, max_steps, deadline, tracker):
    a = a.copy().astype(np.float64)
    a /= a.sum()
    n = len(a)
    current_J = compute_J(a)
    tracker.update(a)
    rng = np.random.default_rng(42)
    stall = 0

    for step in range(max_steps):
        if time.monotonic() > deadline:
            break
        if stall > 200:
            break
        j = rng.integers(0, n)
        improved = False
        for scale in [0.8, 1.2, 0.6, 1.4, 0.9, 1.1, 0.5, 1.5]:
            a_trial = a.copy()
            a_trial[j] *= scale
            a_trial = np.maximum(a_trial, 1e-8)
            ss = a_trial.sum()
            if ss < 1e-10:
                continue
            a_trial /= ss
            J_trial = compute_J(a_trial)
            if J_trial < current_J - 1e-14:
                a = a_trial
                current_J = J_trial
                improved = True
                break
        if not improved:
            stall += 1
        else:
            stall = 0
        if step % 300 == 0:
            tracker.update(a)
    tracker.update(a)
    return a

                break
            # LP phase
            n_lp = max(8, int(40.0 / max(1, n / 256)))
            a = run_lp_opt(a0, n_lp, t_end - 30, tracker, tighten=0.01)
            # LSE phase
            if time.monotonic() < t_end - 15:
                a = lse_optimize(a, max(200, int(8000 / n)), t_end - 10, tracker,
                                 tau_start=30, tau_end=400)
            J = compute_J(a)
            if J < best_J_for_n:
                best_J_for_n = J
                best_for_n = a.copy()
            if (idx + 1) % 3 == 0:
                print(f"    init {idx+1}/{n_inits}: best_J={best_J_for_n:.8f}", flush=True)

        if best_for_n is not None:
            tracker.update(best_for_n)
            if parent is None or best_J_for_n < compute_J(resample_seq(parent, best_for_n.size)):
                parent = best_for_n
            tracker.report(f"Phase1 n={n}: J={best_J_for_n:.8f}")

    # === Phase 2: Deep optimization on best found ===
    if tracker.best is not None and time.monotonic() < deadline - 80:
        t0 = time.monotonic()
        n = tracker.best.size
        print(f"\n{'='*50} Phase2 Deep n={n} {'='*50}", flush=True)

        a = tracker.best.copy().astype(np.float64)

        # LP with progressive tightening
        for tight in [0.02, 0.015, 0.01, 0.008, 0.005, 0.004, 0.003]:
            if time.monotonic() > deadline - 50:
                break
            a = run_lp_opt(a, 15, deadline - 50, tracker, tighten=tight)

        # LSE with increasing tau
        if time.monotonic() < deadline - 40:
            a = lse_optimize(a, 3000, deadline - 35, tracker, tau_start=50, tau_end=2000)

        # Subgradient
        if time.monotonic() < deadline - 20:
            a = subgradient_optimize(a, 5000, deadline - 15, tracker)

        tracker.report(f"Phase2 ({time.monotonic()-t0:.1f}s)")

    # === Phase 3: Simulated annealing for escape ===
    if tracker.best is not None and time.monotonic() < deadline - 50:
        t0 = time.monotonic()
        print(f"\n{'='*50} Phase3 SA {'='*50}", flush=True)
        a = simulated_annealing(tracker.best, deadline - 35, tracker, n_restarts=15, t0=0.06)
        if time.monotonic() < deadline - 15:
            a = run_lp_opt(a, 10, deadline - 12, tracker, tighten=0.005)
            if time.monotonic() < deadline - 8:
                a = lse_optimize(a, 500, deadline - 5, tracker, tau_start=100, tau_end=800)
        tracker.report(f"Phase3 ({time.monotonic()-t0:.1f}s)")

    # === Phase 4: Final polish ===
    if tracker.best is not None and time.monotonic() < deadline - 8:
        t0 = time.monotonic()
        print(f"\n{'='*50} Phase4 Polish {'='*50}", flush=True)
        a = coordinate_descent(tracker.best, 3000, deadline - 6, tracker)
        if time.monotonic() < deadline - 2:
            a = lse_optimize(a, 500, deadline - 1, tracker, tau_start=200, tau_end=2000)
        tracker.report(f"Phase4 ({time.monotonic()-t0:.1f}s)")

    # Fallback
    if tracker.best is None:
        tracker.best = np.ones(256) / 256
        tracker.update(tracker.best)

    print(f"\n{'='*50} FINAL {'='*50}", flush=True)
    print(f"J={tracker.best_score:.8f} n={tracker.best.size} time={time.monotonic()-tracker.t_start:.1f}s", flush=True)
    return [float(x) for x in tracker.best]
