    n = len(a)

    for it in range(max_iters):

            je = min(n - 1, k)
            for j in range(js, je + 1):

import numpy as np
from scipy.optimize import linprog
from scipy.sparse import csr_matrix

_spar_cache = {}

def get_sparsity(n):
    if n not in _spar_cache:
        rows = []
        cols = []
        for k in range(2 * n - 1):
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            rows.extend([k] * (je - js + 1))
            cols.extend(range(js, je + 1))
        _spar_cache[n] = (np.array(rows), np.array(cols), len(rows))
    return _spar_cache[n]

def fast_conv(a):
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]

def compute_J(a):
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))

    F = np.fft.rfft(f, size)
    A = np.fft.rfft(a, size)
    return np.fft.irfft(F * np.conj(A), size)[:n]

        j = cols[i]
        vals[i] = a[k - j]
    A = csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                options={'presolve': True, 'time_limit': 15})
    if r.success and r.x is not None:
        g = np.maximum(r.x, 0.0)
        if g.sum() > 1e-10:
            return g
    return None

def active_set_update(a, max_iters, deadline, tracker, alpha=0.5, eps_frac=1e-4):
    """Multiplicative update based on active set structure."""
    a = a.copy().astype(np.float64)
    a = np.maximum(a, 1e-10)
    a /= a.sum()
    n = len(a)
    current_J = compute_J(a)
    tracker.update(a)
    
    support_threshold = 1e-6
    
    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        
        b = fast_conv(a)
        M = float(b.max())
        
        # Active set: positions where b is close to max
        active_threshold = M * (1.0 - eps_frac)
        f = (b >= active_threshold).astype(np.float64)
        
        # Cross-correlation: h[j] = sum_k f[k] * a[k-j]
        h = cross_corr_fft(f, a, n)
        h = np.maximum(h, 1e-10)
        
        # Support of a
        support = a > support_threshold * a.max()
        if not np.any(support):
            break
        
        # Target: make h constant on support
        target = h[support].mean()
        
        # Multiplicative update
        a_new = a * np.power(target / h, alpha)
        a_new = np.maximum(a_new, 1e-10)
        
        # Also try a simpler version: push down where h is large
        a_new2 = a * np.exp(-0.1 * (h - target) / max(target, 1e-10))
        a_new2 = np.maximum(a_new2, 1e-10)
        
        best = a
        best_J = current_J
        for trial in [a_new, a_new2]:
            trial = trial / trial.sum()
            J_trial = compute_J(trial)
            if J_trial < best_J:
                best_J = J_trial
                best = trial
        
        if np.all(best == a):
            # Try different alpha
            for alt_alpha in [0.1, 0.3, 0.8, 1.0]:
                a_try = a * np.power(target / h, alt_alpha)
                a_try = np.maximum(a_try, 1e-10)
                a_try /= a_try.sum()
                J_try = compute_J(a_try)
                if J_try < best_J:
                    best_J = J_try
                    best = a_try
            if np.all(best == a):
                break
        
        a = best
        current_J = best_J
        
        if it % 20 == 0:
            tracker.update(a)
    
    tracker.update(a)
    return a, current_J

def lp_iteration(a, M_tight, deadline, tracker, n):
    """Single LP best-response step."""
    g = lp_best_response(a, M_tight, n)
    if g is None:
        return a
    g = np.maximum(g, 0.0)
    gs = g.sum()
    if gs < 1e-10:
        return a
    g /= gs
    
    best_a = a
    best_J = compute_J(a)
    for t in [0.3, 0.5, 0.7, 1.0]:
        a_trial = (1.0 - t) * a + t * g

        if ss < 1e-10:

        a_trial /= ss

def lse_refine(a, deadline, tracker, max_iters=500):
    """LSE-based gradient descent to smooth out the peaks."""
    a = a.copy().astype(np.float64)
    a = np.maximum(a, 1e-8)
    a /= a.sum()
    n = len(a)
    current_J = compute_J(a)
    tracker.update(a)
    
    tau = 100.0
    lr = 1.0 / (n * tau)
    momentum = 0.9
    v = np.zeros(n)
    
    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        b = fast_conv(a)
        b_max = b.max()
        # Soft-max weights
        exp_b = np.exp(-tau * (b_max - b) / (2*n-1))  # weight more on peaks
        p = exp_b / exp_b.sum()
        
        # Gradient: d/d a_j [sum_k p_k (a*a)_k] = 2 * (p cross_corr a)_j
        grad = 2.0 * cross_corr_fft(p, a, n)
        
        v = momentum * v + (1.0 - momentum) * grad
        a_new = a - lr * v
        a_new = np.maximum(a_new, 1e-8)
        a_new /= a_new.sum()
        
        J_new = compute_J(a_new)
        if J_new < current_J - 1e-15:
            a = a_new
            current_J = J_new
            tau = min(500.0, tau * 1.02)
            lr = 1.0 / (n * tau)
        else:
            lr *= 0.5
            v *= 0.5
            if lr < 1e-18:
                break
        
        if it % 50 == 0:
            tracker.update(a)
    
    tracker.update(a)
    return a

def flat_spectrum_init(n, rng, amplitude_var=1.0):
    """Generate sequence with approximately flat power spectrum."""
    half = n // 2 + 1
    angles = rng.uniform(0, 2 * np.pi, half)
    amps = 1.0 + amplitude_var * rng.standard_normal(half)
    amps = np.abs(amps)
    A = np.zeros(n, dtype=complex)
    A[0] = amps[0].real
    A[1:half] = amps[1:half] * np.exp(1j * angles[1:half])
    for k in range(1, n - n // 2):
        A[k] = np.conj(A[n - k])
    a = np.fft.ifft(A).real
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-10:
        return np.ones(n) / n
    return a / s

def resample_seq(src, dest_n):
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)

def wrap_init(x, n):
    """Wrap an array to length n, ensuring non-negative and normalized."""

    if x.size != n:
        x = resample_seq(x, n)
    x = np.maximum(x, 0.001)
    x /= x.sum()
    return x

_spar_cache = {}


def get_sparsity(n):
    if n not in _spar_cache:
        rows = []
        cols = []
        for k in range(2 * n - 1):
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            rows.extend([k] * (je - js + 1))
            cols.extend(range(js, je + 1))
        _spar_cache[n] = (np.array(rows), np.array(cols), len(rows))
    return _spar_cache[n]


def fast_conv(a):
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]


def compute_J(a):
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))


def cross_corr_fft(f, a, n):
    """Compute g[j] = sum_k f[k] * a[k-j] for j in [0, n-1]."""
    size = 1
    while size < 2 * n:
        size <<= 1
    F = np.fft.rfft(f, size)
    A = np.fft.rfft(a, size)
    return np.fft.irfft(F * np.conj(A), size)[:n]


def lp_best_response(a, M, n):

        # Update dual: emphasize high-convolution positions
        b_norm = b / b.sum()
        dual = beta * dual + (1.0 - beta) * b_norm
        dual = np.maximum(dual, 1e-15)

        # h[j] = sum_k dual[k] * a[k-j]
        h = cross_corr_fft(dual, a, n)
        h = np.maximum(h, 1e-10)

        # Target: make h uniform
        target = h.mean()

        # Multiplicative update
        a_new = a * np.power(target / h, alpha)
        a_new = np.maximum(a_new, 1e-10)
        a_new /= a_new.sum()

        J_new = compute_J(a_new)
        if J_new < best_J:
            best_J = J_new
            best_a = a_new.copy()
            a = a_new
            current_J = J_new
            # Increase aggressiveness slightly
            alpha = min(0.8, alpha * 1.01)
            if it % 30 == 0:
                tracker.update(a)
        else:
            # Try smaller step
            improved = False
            for alt_a in [0.2, 0.1, 0.05]:
                a_try = a * np.power(target / h, alt_a)
                a_try = np.maximum(a_try, 1e-10)
                a_try /= a_try.sum()
                J_try = compute_J(a_try)
                if J_try < best_J:
                    a = a_try
                    best_J = J_try
                    best_a = a.copy()
                    current_J = J_try
                    improved = True
                    break
            if not improved:
                alpha = max(0.05, alpha * 0.8)
                beta = min(0.85, beta * 1.03)
                if alpha < 0.06 and beta > 0.8:
                    break

    tracker.update(best_a)
    return best_a, best_J


def lse_gradient_descent(a, deadline, tracker, max_iters=1000):
    """Smooth the convolution using LSE approximation and gradient descent."""
    a = a.copy().astype(np.float64)
    a = np.maximum(a, 1e-8)
    a /= a.sum()
    n = len(a)
    current_J = compute_J(a)
    tracker.update(a)

    tau = 50.0
    lr = 2.0 / (n * tau)
    momentum = 0.85
    v = np.zeros(n)
    best_J = current_J
    best_a = a.copy()
    stall = 0

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        b = fast_conv(a)
        b_max = b.max()
        # Weights emphasizing peaks
        exp_b = np.exp(tau * (b - b_max) / (2 * n - 1))
        p = exp_b / exp_b.sum()

        # Gradient of sum_k p_k * b_k w.r.t. a
        grad = 2.0 * cross_corr_fft(p, a, n)

        v = momentum * v + (1.0 - momentum) * grad
        a_new = a - lr * v
        a_new = np.maximum(a_new, 1e-8)
        a_new /= a_new.sum()

        J_new = compute_J(a_new)
        if J_new < best_J - 1e-15:
            a = a_new
            best_J = J_new
            best_a = a.copy()
            current_J = J_new
            tau = min(400.0, tau * 1.02)
            lr = 2.0 / (n * tau)
            stall = 0
            if it % 50 == 0:
                tracker.update(a)
        else:
            lr *= 0.6
            v *= 0.3
            stall += 1
            if stall > 30 or lr < 1e-18:
                break

    tracker.update(best_a)
    return best_a, best_J


def peak_shaving(a, deadline, tracker, max_iters=200):
    """Targeted peak reduction: identify the max position and locally adjust."""
    a = a.copy().astype(np.float64)
    a = np.maximum(a, 1e-10)
    a /= a.sum()
    n = len(a)
    current_J = compute_J(a)
    tracker.update(a)
    rng = np.random.default_rng(123)

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        b = fast_conv(a)
        M = float(b.max())
        peak_idx = int(np.argmax(b))

        # Find all j that contribute to b[peak_idx]
        js_start = max(0, peak_idx - n + 1)
        js_end = min(n - 1, peak_idx)

        # For each contributing j, reducing a[j] reduces b[peak_idx]
        # But also affects other b[k]. Do a local search.
        best_improvement = None
        best_a = a
        best_J = current_J

        # Try reducing random subsets of contributing indices
        n_try = min(20, js_end - js_start + 1)
        candidates = rng.choice(range(js_start, js_end + 1), size=n_try, replace=False)

        for j in candidates:
            for factor in [0.5, 0.7, 0.3, 0.8]:
                a_trial = a.copy()
                a_trial[j] *= factor
                a_trial = np.maximum(a_trial, 1e-10)
                ss = a_trial.sum()
                if ss < 1e-10:
                    continue
                a_trial /= ss
                J_trial = compute_J(a_trial)
                if J_trial < best_J:
                    best_J = J_trial
                    best_a = a_trial.copy()

        if best_J < current_J - 1e-14:
            a = best_a
            current_J = best_J
            tracker.update(a)
        else:
            if it > 50:
                break

    tracker.update(a)
    return a, current_J


def flat_spectrum_init(n, rng, var=1.0):
    """Generate sequence with approximately flat power spectrum (non-negative)."""
    half = n // 2 + 1
    angles = rng.uniform(0, 2 * np.pi, half)
    amps = np.abs(1.0 + var * rng.standard_normal(half))
    A = np.zeros(n, dtype=complex)
    A[0] = amps[0]
    for k in range(1, half):
        A[k] = amps[k] * np.exp(1j * angles[k])
    for k in range(1, n - n // 2):
        A[k] = np.conj(A[n - k])
    a = np.fft.ifft(A).real
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-10:
        return np.ones(n) / n
    return a / s


def resample_seq(src, dest_n):
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)


def wrap_init(x, n):
    x = np.asarray(x, dtype=np.float64)
    if x.size != n:
        x = resample_seq(x, n)
    x = np.maximum(x, 0.001)
    x /= x.sum()
    return x


def full_optimize(a, deadline, tracker):
    """Run all optimization methods in sequence with time budgeting."""
    a = np.maximum(a.astype(np.float64), 1e-10)
    a /= a.sum()
    n = len(a)
    remaining = deadline - time.monotonic()
    if remaining < 10:
        return a

    # Allocate time: LP 30%, dual-avg 30%, LSE 20%, peak-shave 20%
    t_lp = min(remaining * 0.3, 150.0)
    t_dual = min(remaining * 0.3, 150.0)
    t_lse = min(remaining * 0.2, 100.0)
    t_peak = max(remaining - t_lp - t_dual - t_lse - 5, 10.0)

    # LP phase
    dl1 = time.monotonic() + t_lp
    a, J = run_lp_phase(a, dl1, tracker, max_time=t_lp)

    # Dual averaging phase
    dl2 = time.monotonic() + t_dual
    a, J = dual_averaging_update(a, dl2, tracker)

    # LSE phase
    dl3 = time.monotonic() + t_lse
    a, J = lse_gradient_descent(a, dl3, tracker)

    # Peak shaving
    dl4 = time.monotonic() + t_peak
    a, J = peak_shaving(a, dl4, tracker)

    # Quick LP polish
    if time.monotonic() < deadline - 5:
        b = fast_conv(a)
        M = float(b.max())
        g = lp_best_response(a, M * 0.99, n)
        if g is not None:
            g = np.maximum(g, 0.0)
            gs = g.sum()
            if gs > 1e-10:
                g /= gs
                for t in [0.3, 0.5, 1.0]:
                    a_try = (1 - t) * a + t * g
                    a_try = np.maximum(a_try, 0.0)
                    ss = a_try.sum()
                    if ss > 1e-10:
                        a_try /= ss
                        if compute_J(a_try) < J:
                            a = a_try
                            J = compute_J(a)

    tracker.update(a)
    return a


class ScoreTracker:
    def __init__(self):
        self.best = None
        self.best_score = float('inf')
        self.t_start = time.monotonic()

    def update(self, a):
        a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
        if a.size == 0 or a.sum() < 0.01:
            return
        score = float(evaluate_sequence(a.tolist()))
        if np.isfinite(score) and score < self.best_score:
            self.best = a.copy()
            self.best_score = score
            elapsed = time.monotonic() - self.t_start
            print(f"  * NEW BEST: J={score:.8f} n={a.size} t={elapsed:.1f}s", flush=True)

    def report(self, label=""):
        if self.best is not None:
            print(f"  {label} best: J={self.best_score:.8f} n={self.best.size} t={time.monotonic()-self.t_start:.1f}s", flush=True)


def propose_candidate(seed=42, budget_s=1500, **kwargs):
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + float(budget_s) - 30.0
    tracker = ScoreTracker()

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        if p.ndim == 1 and p.size > 0 and np.all(np.isfinite(p)):
            parent = np.maximum(p, 0.0)
            if parent.sum() > 0.01:
                parent /= parent.sum()
                tracker.update(parent)
                print(f"Parent: n={parent.size}, J={tracker.best_score:.8f}", flush=True)
    except (NameError, AttributeError):
        pass

    if parent is None:
        parent = np.ones(256) / 256.0
        tracker.update(parent)

    # === Phase 1: Improve parent aggressively ===
    print(f"\n{'='*50} Phase 1: Improve parent (n={parent.size}) {'='*50}", flush=True)
    t1_end = min(deadline, time.monotonic() + 400.0)
    a_parent = full_optimize(parent, t1_end, tracker)
    print(f"  Phase 1 done: J={compute_J(a_parent):.8f}", flush=True)

    # === Phase 2: Try larger lengths ===
    larger_lengths = []
    pn = parent.size
    for n_cand in [int(pn * 1.5), int(pn * 2), int(pn * 3), 2048]:
        n_cand = max(n_cand, 64)
        n_cand = min(n_cand, 2048)
        if n_cand > pn and n_cand not in larger_lengths:
            larger_lengths.append(n_cand)

    # Also try some standard sizes
    for std in [384, 512, 768, 1024, 1536, 2048]:
        if std > pn and std not in larger_lengths:
            larger_lengths.append(std)

    larger_lengths = sorted(set(larger_lengths))

    # Filter by time
    time_remaining = deadline - time.monotonic()
    time_per_len = max(50.0, (time_remaining - 100) / max(1, len(larger_lengths)))
    # Only try lengths that fit in budget
    n_lens_to_try = min(len(larger_lengths), int((time_remaining - 100) / time_per_len))
    larger_lengths = larger_lengths[:max(1, n_lens_to_try)]

    print(f"\n{'='*50} Phase 2: Larger lengths {larger_lengths} {'='*50}", flush=True)

    for n in larger_lengths:
        if time.monotonic() > deadline - 60:
            break
        t0 = time.monotonic()
        t_phase_end = min(deadline - 60, t0 + time_per_len)
        print(f"\n  --- n={n} (budget {time_per_len:.0f}s) ---", flush=True)

        # Generate inits
        inits = []
        inits.append(wrap_init(parent, n))
        inits.append(wrap_init(parent[::-1], n))
        inits.append(flat_spectrum_init(n, rng, var=0.8))
        inits.append(flat_spectrum_init(n, rng, var=1.5))

        t_arr = np.linspace(0, 1, n, endpoint=False)
        inits.append(wrap_init(1.0 + 0.3 * np.cos(2 * np.pi * 3 * t_arr) + 0.2 * np.cos(2 * np.pi * 7 * t_arr + 1.0), n))
        inits.append(wrap_init(0.5 + 0.5 * np.cos(2 * np.pi * t_arr) ** 2, n))
        inits.append(wrap_init(rng.uniform(0.5, 1.5, n), n))

        # Deduplicate
        unique_inits = [inits[0]]
        for x in inits[1:]:
            Jx = compute_J(x)
            if all(abs(Jx - compute_J(u)) > 0.005 for u in unique_inits):
                unique_inits.append(x)

        n_inits = min(len(unique_inits), max(2, int(time_per_len / 60)))
        if n_inits == 2048:  # safety
            n_inits = 3

        init_budget = time_per_len / n_inits
        best_for_n = None
        best_J_for_n = float('inf')

        for idx in range(n_inits):
            t_init_end = min(t_phase_end - 10, time.monotonic() + init_budget)
            if time.monotonic() > t_phase_end - 20:
                break
            a0 = unique_inits[idx % len(unique_inits)]
            a_opt = full_optimize(a0, t_init_end, tracker)
            J = compute_J(a_opt)
            if J < best_J_for_n:
                best_J_for_n = J
                best_for_n = a_opt.copy()

        if best_for_n is not None:
            tracker.update(best_for_n)
            print(f"  n={n} best: J={best_J_for_n:.8f} (took {time.monotonic()-t0:.1f}s)", flush=True)

    # === Phase 3: Final polish on overall best ===
    if tracker.best is not None and time.monotonic() < deadline - 30:
        print(f"\n{'='*50} Phase 3: Final Polish {'='*50}", flush=True)
        a_final = full_optimize(tracker.best, deadline - 5, tracker)
        tracker.update(a_final)

    tracker.report("FINAL")

    # Fallback
    if tracker.best is None:
        fallback = np.ones(256)
        tracker.update(fallback)

    print(f"\n{'='*50} RESULT {'='*50}", flush=True)
    print(f"J={tracker.best_score:.8f} n={tracker.best.size} time={time.monotonic()-tracker.t_start:.1f}s", flush=True)
    return [float(x) for x in tracker.best]
