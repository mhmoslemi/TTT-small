"""
Aggressive LP-alternating + subgradient + coordinate descent optimizer
for minimizing 2*n*max(conv(a,a))/(sum(a))^2.
Key improvements: iterated min-peak LP, coordinate descent polish,
better time allocation, and diverse initializations including Thue-Morse.

import numpy as np
from scipy.optimize import minimize, linprog
from scipy.fft import rfft, irfft
import time

def fft_convolve(a, b):
    n, m = len(a), len(b)
    size = 1
    while size < n + m - 1:
        size <<= 1
    fa = rfft(a, size)
    fb = rfft(b, size)
    return irfft(fa * fb, size)[:n + m - 1]

def compute_conv(a):
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]

def compute_obj(a):
    n = len(a)
    if n == 0:
        return np.inf
    conv = compute_conv(a)
    max_b = float(np.max(conv))
    sum_a = float(np.sum(a))
    if sum_a < 0.01:
        return np.inf
    return 2.0 * n * max_b / (sum_a * sum_a)

def build_toeplitz(a):
    n = len(a)
    # A has shape (2n-1, n)
    # A[k, j] = a[k-j] if 0 <= k-j < n, else 0
    # (a*g)[k] = sum_j A[k,j] * g[j]
    A = np.zeros((2 * n - 1, n), dtype=np.float64)
    for j in range(n):
        start = j
        end = min(j + n, 2 * n - 1)
        A[start:end, j] = a[:end - start]
    return A

def solve_lp_minpeak(a, time_limit=30):
    """
    Solve: min M s.t. (a*g)[k] <= M for all k, sum(g)=1, g >= 0
    Variables: g[0..n-1], M
    """
    n = len(a)
    A = build_toeplitz(a)
    # Variables: [g_0, ..., g_{n-1}, M]
    c = np.zeros(n + 1)
    c[-1] = 1.0  # minimize M
    A_ub = np.zeros((2 * n - 1, n + 1))
    A_ub[:, :n] = A
    A_ub[:, n] = -1.0
    b_ub = np.zeros(2 * n - 1)
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])
    bounds = [(0.0, None)] * n + [(0.0, None)]
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                  bounds=bounds, method='highs',
                  options={'presolve': True, 'time_limit': time_limit})
    if res.status == 0:
        g = np.maximum(res.x[:n], 0)
        M_opt = float(res.x[n])
        return g, M_opt
    return None, None

def solve_lp_maxsum(a, M, time_limit=30):
    """
    Solve: max sum(g) s.t. (a*g)[k] <= M for all k, g >= 0
    """
    n = len(a)
    A = build_toeplitz(a)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                  options={'presolve': True, 'time_limit': time_limit})
    if res.status == 0:
        g = np.maximum(res.x, 0)
        s = float(np.sum(g))
        if s > 1e-10:
            return g / s, s
    return None, None

def softmax(z):
    z = z - np.max(z)
    e = np.exp(z)
    return e / np.sum(e)

def soft_max_obj_and_grad(z, tau):
    n = len(z)
    a = softmax(z)
    conv = compute_conv(a)
    cmax = float(np.max(conv))
    shifted = np.clip((conv - cmax) / tau, -500.0, 500.0)
    exp_conv = np.exp(shifted)
    Z = float(np.sum(exp_conv))
    soft_max_val = cmax + tau * np.log(Z)
    w = exp_conv / Z
    
    w_rev = w[::-1]
    conv_fw = fft_convolve(a, w_rev)
    f1 = conv_fw[n - 1:2 * n - 1][::-1]
    grad_a = 2.0 * f1
    
    mean_grad = float(np.dot(a, grad_a))
    grad_z = a * (grad_a - mean_grad)
    
    return soft_max_val, grad_z

def optimize_lbfgs(a_init, tau_list, max_iter=200, time_budget=20.0):
    n = len(a_init)
    a = a_init.copy()
    s = np.sum(a)
    if s < 1e-10:
        a = np.ones(n) / n
    a = a / s
    a = np.maximum(a, 1e-30)
    z = np.log(a)
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    
    for tau in tau_list:
        if time.monotonic() - start > time_budget:
            break
        
        def func_and_grad(z_val):
            return soft_max_obj_and_grad(z_val, tau)
        
        try:
            result = minimize(func_and_grad, z, jac=True, method='L-BFGS-B',
                              options={'maxiter': max_iter, 'ftol': 1e-14, 'gtol': 1e-8})
            z_new = result.x
        except Exception:
            continue
        
        a_new = softmax(z_new)
        a_new = np.maximum(a_new, 0)
        s_new = np.sum(a_new)
        if s_new < 1e-10:
            continue
        a_new = a_new / s_new
        obj = compute_obj(a_new)
        
        if obj < best_obj:
            best_obj = obj
            best_a = a_new.copy()
            z = z_new
    
    return best_a, best_obj

def subgradient_optimize(a, max_iters=2000, time_budget=15.0, tau_start=5.0, tau_end=50.0):
    n = len(a)
    a = a.copy().astype(np.float64)
    s = np.sum(a)
    if s < 1e-10:
        a = np.ones(n) / n
    a = a / s
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    step = 0.05
    
    for it in range(max_iters):
        if time.monotonic() - start > time_budget:
            break
        
        frac = it / max(1, max_iters)
        tau = tau_start + (tau_end - tau_start) * frac
        
        conv = compute_conv(a)
        max_conv = float(np.max(conv))
        shifted = np.clip(tau * (conv - max_conv), -300, 300)
        weights = np.exp(shifted)
        w_sum = np.sum(weights)
        if w_sum < 1e-300:
            idx = int(np.argmax(conv))
            weights = np.zeros(len(conv))
            weights[idx] = 1.0
        else:
            weights /= w_sum
        
        w_rev = weights[::-1]
        conv_fw = fft_convolve(a, w_rev)
        grad_a = 2.0 * conv_fw[n - 1:2 * n - 1][::-1]
        
        grad_centered = grad_a - np.mean(grad_a)
        a_new = a - step * grad_centered
        a_new = np.maximum(a_new, 0)
        s_new = np.sum(a_new)
        if s_new > 1e-10:
            a_new = a_new / s_new
        else:
            a_new = a.copy()
        
        a = a_new
        obj = compute_obj(a)
        
        if obj < best_obj - 1e-12:
            best_obj = obj
            best_a = a.copy()
        
        if it > 50:
            step *= 0.997
    
    return best_a

def coordinate_descent(a, time_budget=10.0, n_sweeps=50):
    """
    Coordinate descent: for each i, try to optimize a[i] while keeping others fixed.
    The objective as a function of a[i] (with sum(a)=1) is:
    f(a[i]) = 2n * max_k (a*a)[k] / (1-a[i]+a[i])^2 = 2n * max_k (a*a)[k] (since we normalize)
    
    Actually, let's not normalize during the sweep. Fix sum of other elements = S_other.
    Then a[i] = a_i, sum_total = S_other + a_i.
    (a*a)[k] is linear in a_i for each k (since it's a sum of products, and a_i appears
    in at most 2 terms: a[i]*a[k-i] and a[k-i]*a[i]).
    
    So (a*a)[k](a_i) = c_k + d_k * a_i where:
    c_k = (a*a)[k] - contribution from a_i
    d_k = coefficient of a_i
    
    The max of (a*a)[k] as a function of a_i is the max of a set of linear functions,
    so it's convex in a_i. We want to minimize max_k (c_k + d_k * a_i) / (S_other + a_i)^2.
    
    This is a 1D convex-ish problem we can solve with ternary search or golden section.
    """
    n = len(a)
    a = a.copy().astype(np.float64)
    s = np.sum(a)
    a = a / s
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    
    for sweep in range(n_sweeps):
        if time.monotonic() - start > time_budget - 2:
            break
        
        for i in range(n):
            if time.monotonic() - start > time_budget - 1:
                break
            
            a_save = a[i]
            S_other = s - a_save  # sum of all other elements (should be ~1 since we normalize)
            
            # Try a few values
            # The optimal a[i] is somewhere in [0, ~2*a_save] perhaps
            candidates = [0.0, a_save * 0.5, a_save * 0.8, a_save, a_save * 1.1, 
                         a_save * 1.25, a_save * 1.5, a_save * 2.0,
                         S_other / n]  # average value
            
            best_local = a_save
            best_local_obj = compute_obj(a)
            
            for val in candidates:
                if val < 0 or val > 100:
                    continue
                a[i] = val
                total = np.sum(a)
                if total < 1e-10:
                    a[i] = a_save
                    continue
                # Don't normalize - the objective handles it
                obj = compute_obj(a)
                if obj < best_local_obj:
                    best_local_obj = obj
                    best_local = val
            
            a[i] = best_local
            s = np.sum(a)
            if s > 1e-10:
                a = a / s
            
            if compute_obj(a) > best_obj:
                a[i] = a_save
                s = np.sum(a)
                a = a / s
    
    return a

def thue_morse(n):
    """Generate Thue-Morse sequence of length n (as 0/1 values)."""
    t = np.zeros(n)
    for i in range(n):
        t[i] = bin(i).count('1') % 2
    return t

def diverse_init(n, seed, parent=None):
    """Generate diverse initializations."""
    rng = np.random.default_rng(seed)
    k = np.arange(n)
    choice = rng.integers(0, 12)
    
    if parent is not None and choice < 3:
        # Resample parent
        x_old = np.linspace(0, 1, len(parent))
        x_new = np.linspace(0, 1, n)
        a = np.interp(x_new, x_old, parent)
        # Add some noise
        a = a * np.exp(rng.normal(0, 0.1, n))
    elif choice == 3:
        # Thue-Morse smoothed
        t = thue_morse(n)
        a = 0.3 + t.astype(float)
        # Smooth
        win = max(3, n // 30)
        kernel = np.ones(win) / win
        a = np.convolve(a, kernel, mode='same')
    elif choice == 4:
        # All ones
        a = np.ones(n)
    elif choice == 5:
        # Cosine
        f = rng.uniform(0.5, 3.0)
        phase = rng.uniform(0, 2 * np.pi)
        a = 1.0 + 0.5 * np.cos(2 * np.pi * f * k / n + phase)
    elif choice == 6:
        # Two cosines
        f1 = rng.uniform(0.5, 2.0)
        f2 = rng.uniform(2.0, 5.0)
        a = 1.0 + 0.3 * np.cos(2 * np.pi * f1 * k / n) + 0.3 * np.cos(2 * np.pi * f2 * k / n)
    elif choice == 7:
        # Random smoothed
        a = rng.random(n)
        win = max(3, n // 20)
        kernel = np.ones(win) / win
        a = np.convolve(a, kernel, mode='same')
    elif choice == 8:
        # Power law
        alpha = rng.uniform(0.5, 2.0)
        a = (k + 1) ** (-alpha)
    elif choice == 9:
        # Bimodal
        c1 = n * rng.uniform(0.2, 0.4)
        c2 = n * rng.uniform(0.6, 0.8)
        w1 = n * rng.uniform(0.05, 0.15)
        w2 = n * rng.uniform(0.05, 0.15)
        a = 1.0 + np.exp(-0.5 * ((k - c1) / w1) ** 2) + np.exp(-0.5 * ((k - c2) / w2) ** 2)
    elif choice == 10:
        # Sawtooth
        f = rng.uniform(1.0, 4.0)
        a = 1.0 + 0.3 * (2 * np.abs((k * f / n) % 1 - 0.5) - 0.5)
    else:
        # Random binary smoothed
        a = (rng.random(n) > 0.5).astype(float)
        win = max(3, n // 25)
        kernel = np.ones(win) / win
        a = np.convolve(a, kernel, mode='same')
    
    a = np.maximum(a, 0.01)
    s = np.sum(a)
    return a / s

"""
Aggressive LP-alternating + L-BFGS-B + coordinate descent + subgradient optimizer
for minimizing 2*n*max(conv(a,a))/(sum(a))^2.
Key improvements over parent: iterated min-peak LP as primary optimizer,
coordinate descent polish, better time allocation, diverse initializations
including Thue-Morse and frequency-domain flat sequences.

import numpy as np
from scipy.optimize import minimize, linprog
from scipy.fft import rfft, irfft
import time


def fft_convolve(a, b):
    """Linear convolution of a and b using FFT."""
    n, m = len(a), len(b)
    size = 1
    while size < n + m - 1:
        size <<= 1
    fa = rfft(a, size)
    fb = rfft(b, size)
    return irfft(fa * fb, size)[:n + m - 1]


def compute_conv(a):
    """Compute linear convolution a*a using FFT. Returns array of length 2n-1."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    conv = compute_conv(a)
    max_b = float(np.max(conv))
    sum_a = float(np.sum(a))
    if sum_a < 0.01:
        return np.inf
    return 2.0 * n * max_b / (sum_a * sum_a)


def build_toeplitz(a):
    """Build Toeplitz matrix A where (A@g)[k] = (conv(a,g))[k]. Shape (2n-1, n)."""
    n = len(a)
    A = np.zeros((2 * n - 1, n), dtype=np.float64)
    idx = np.arange(n)
    for j in range(n):
        start = j
        end = min(j + n, 2 * n - 1)
        A[start:end, j] = a[:end - start]
    return A


def solve_lp_minpeak(a, time_limit=25):
    """
    Solve: min M s.t. (a*g)[k] <= M for all k, sum(g)=1, g >= 0.
    Variables: [g_0,...,g_{n-1}, M]
    Returns (g, M_opt) or (None, None).
    """
    n = len(a)
    A = build_toeplitz(a)
    c = np.zeros(n + 1)
    c[-1] = 1.0
    A_ub = np.zeros((2 * n - 1, n + 1))
    A_ub[:, :n] = A
    A_ub[:, n] = -1.0
    b_ub = np.zeros(2 * n - 1)
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])
    bounds = [(0.0, None)] * n + [(0.0, None)]
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': time_limit})
        if res.status == 0:
            g = np.maximum(res.x[:n], 0)
            M_opt = float(res.x[n])
            return g, M_opt
    except Exception:
        pass
    return None, None


def solve_lp_maxsum(a, M, time_limit=25):
    """
    Solve: max sum(g) s.t. (a*g)[k] <= M for all k, g >= 0.
    Returns (g_normalized, sum_g) or (None, None).
    """
    n = len(a)
    A = build_toeplitz(a)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': time_limit})
        if res.status == 0:
            g = np.maximum(res.x, 0)
            s = float(np.sum(g))
            if s > 1e-10:
                return g / s, s
    except Exception:
        pass
    return None, None


def alternating_lp_hybrid(a, max_iters=80, time_budget=60.0):
    """
    Alternating LP: alternates between min-peak LP and max-sum LP.
    More aggressive than previous version with better line search.
    """
    n = len(a)
    a = a.copy().astype(np.float64)
    s = np.sum(a)
    if s < 1e-10:
        return np.ones(n) / n
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    stall = 0

    for it in range(max_iters):
        if time.monotonic() - start > time_budget - 5:
            break

        use_minpeak = (it % 2 == 0)
        if use_minpeak:
            g, M_opt = solve_lp_minpeak(a, time_limit=min(20, max(5, time_budget - (time.monotonic() - start))))
        else:
            conv = compute_conv(a)
            M = float(np.max(conv)) * 1.001
            g, _ = solve_lp_maxsum(a, M, time_limit=min(20, max(5, time_budget - (time.monotonic() - start))))

        if g is None:
            break

        g = np.maximum(g, 0)
        sg = np.sum(g)
        if sg < 1e-10:
            break
        g = g / sg

        # Line search with geometric spacing
        improved = False
        for t in [0.4, 0.6, 0.8, 1.0, 0.3, 0.5, 0.7, 0.9, 0.2, 0.1]:
            a_try = (1.0 - t) * a + t * g
            a_try = np.maximum(a_try, 1e-12)
            st = np.sum(a_try)
            if st < 1e-10:
                continue
            a_try = a_try / st
            obj_try = compute_obj(a_try)
            if obj_try < best_obj - 1e-9:
                best_obj = obj_try
                best_a = a_try.copy()
                a = a_try.copy()
                improved = True
                break

        if improved:
            stall = 0
        else:
            stall += 1
            if stall > 15:
                # Perturb to escape
                rng = np.random.default_rng(it * 6151 + 42)
                a = a * np.exp(rng.normal(0, 0.05, n))
                a = np.maximum(a, 1e-10)
                a = a / np.sum(a)
                stall = 0
            else:
                break

    return best_a


def softmax_fn(z):
    """Numerically stable softmax."""
    z = z - np.max(z)
    e = np.exp(z)
    return e / np.sum(e)


def soft_max_obj_grad(z, tau):
    """
    Compute soft-max of conv(softmax(z), softmax(z)) and gradient w.r.t. z.
    """
    n = len(z)
    a = softmax_fn(z)
    conv = compute_conv(a)
    cmax = float(np.max(conv))
    shifted = np.clip((conv - cmax) / tau, -500.0, 500.0)
    exp_conv = np.exp(shifted)
    Z = float(np.sum(exp_conv))
    soft_max_val = cmax + tau * np.log(Z)
    w = exp_conv / Z

    w_rev = w[::-1]
    conv_fw = fft_convolve(a, w_rev)
    f1 = conv_fw[n - 1:2 * n - 1][::-1]
    grad_a = 2.0 * f1

    mean_grad = float(np.dot(a, grad_a))
    grad_z = a * (grad_a - mean_grad)

    return soft_max_val, grad_z


def optimize_lbfgs(a_init, tau_list, max_iter=200, time_budget=20.0):
    """Optimize using L-BFGS-B with soft-max parametrization and tau annealing."""
    n = len(a_init)
    a = a_init.copy()
    s = np.sum(a)
    if s < 1e-10:
        a = np.ones(n) / n
    a = a / s
    a = np.maximum(a, 1e-30)
    z = np.log(a)

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    for tau in tau_list:
        if time.monotonic() - start > time_budget:
            break

        def func_and_grad(z_val):
            return soft_max_obj_grad(z_val, tau)

        try:
            result = minimize(func_and_grad, z, jac=True, method='L-BFGS-B',
                              options={'maxiter': max_iter, 'ftol': 1e-14, 'gtol': 1e-8})
            z_new = result.x
        except Exception:
            continue

        a_new = softmax_fn(z_new)
        a_new = np.maximum(a_new, 0)
        s_new = np.sum(a_new)
        if s_new < 1e-10:
            continue
        a_new = a_new / s_new
        obj = compute_obj(a_new)

        if obj < best_obj:
            best_obj = obj
            best_a = a_new.copy()
            z = z_new

    return best_a, best_obj


def subgradient_optimize(a, max_iters=2000, time_budget=15.0, tau_start=5.0, tau_end=50.0):
    """Smoothed subgradient optimization with adaptive step size."""
    n = len(a)
    a = a.copy().astype(np.float64)
    s = np.sum(a)
    if s < 1e-10:
        a = np.ones(n) / n
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    step = 0.05
    stall = 0

    for it in range(max_iters):
        if time.monotonic() - start > time_budget:
            break

        frac = it / max(1, max_iters)
        tau = tau_start + (tau_end - tau_start) * frac

        conv = compute_conv(a)
        max_conv = float(np.max(conv))
        shifted = np.clip(tau * (conv - max_conv), -300, 300)
        weights = np.exp(shifted)
        w_sum = np.sum(weights)
        if w_sum < 1e-300:
            idx = int(np.argmax(conv))
            weights = np.zeros(len(conv))
            weights[idx] = 1.0
        else:
            weights /= w_sum

        w_rev = weights[::-1]
        conv_fw = fft_convolve(a, w_rev)
        grad_a = 2.0 * conv_fw[n - 1:2 * n - 1][::-1]

        grad_centered = grad_a - np.mean(grad_a)
        a_new = a - step * grad_centered
        a_new = np.maximum(a_new, 0)
        s_new = np.sum(a_new)
        if s_new > 1e-10:
            a_new = a_new / s_new
        else:
            a_new = a.copy()

        a = a_new
        obj = compute_obj(a)

        if obj < best_obj - 1e-12:
            best_obj = obj
            best_a = a.copy()
            stall = 0
        else:
            stall += 1
            if stall > 100:
                step *= 0.7
                if step < 1e-10:
                    break

        if it > 50:
            step *= 0.998

    return best_a


def coordinate_descent(a, time_budget=10.0, n_sweeps=30):
    """
    Coordinate descent: for each coordinate i, try to find the best value
    by evaluating a small set of candidates.
    """
    n = len(a)
    a = a.copy().astype(np.float64)
    s = np.sum(a)
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    for sweep in range(n_sweeps):
        if time.monotonic() - start > time_budget - 2:
            break

        improved = False
        for i in range(n):
            if time.monotonic() - start > time_budget - 1:
                break
            if n > 500 and (i % 100 == 99) and (time.monotonic() - start > time_budget * 0.8):
                break

            a_save = a[i]
            S_other = 1.0 - a_save

            # Candidate values to try
            base = a_save
            candidates = [0.0, base * 0.3, base * 0.6, base * 0.8, base,
                         base * 1.1, base * 1.3, base * 1.6, base * 2.0,
                         S_other / n]

            best_local = base
            best_local_obj = best_obj

            for val in candidates:
                if val < -0.01 or val > 1.0:
                    continue
                val = max(0.0, val)
                a[i] = val
                total = np.sum(a)
                if total < 1e-10:
                    a[i] = a_save
                    continue
                obj = compute_obj(a)
                if obj < best_local_obj:
                    best_local_obj = obj
                    best_local = val

            if best_local != base:
                a[i] = best_local
                a = a / np.sum(a)
                improved = True

            elif a[i] != a_save:
                a[i] = a_save
                a = a / np.sum(a)

        if not improved:
            break

    final_obj = compute_obj(a)
    if final_obj < best_obj:
        best_obj = final_obj
        best_a = a.copy()

    return best_a


def thue_morse_smooth(n):
    """Generate a smoothed Thue-Morse inspired sequence."""
    t = np.zeros(n)
    for i in range(n):
        t[i] = bin(i).count('1') % 2
    a = 0.2 + t.astype(float)
    # Gaussian smoothing
    win = max(3, n // 25)
    x = np.arange(win) - win // 2
    sigma = win / 6.0
    kernel = np.exp(-0.5 * (x / sigma) ** 2)
    kernel /= np.sum(kernel)
    a = np.convolve(a, kernel, mode='same')
    return a


def diverse_init(n, seed, parent=None):
    """Generate diverse initializations."""
    rng = np.random.default_rng(seed)
    k = np.arange(n)
    choice = rng.integers(0, 12)

    if parent is not None and choice < 2:
        # Resample parent with perturbation
        x_old = np.linspace(0, 1, len(parent))
        x_new = np.linspace(0, 1, n)
        a = np.interp(x_new, x_old, parent)
        a = a * np.exp(rng.normal(0, 0.08, n))
    elif choice == 2:
        # Thue-Morse smoothed
        a = thue_morse_smooth(n)
    elif choice == 3:
        # All ones (uniform)
        a = np.ones(n)
    elif choice == 4:
        # Single cosine
        f = rng.uniform(0.5, 3.0)
        phase = rng.uniform(0, 2 * np.pi)
        a = 1.0 + 0.5 * np.cos(2 * np.pi * f * k / n + phase)
    elif choice == 5:
        # Two cosines
        f1 = rng.uniform(0.3, 2.0)
        f2 = rng.uniform(2.0, 6.0)
        a = 1.0 + 0.3 * np.cos(2 * np.pi * f1 * k / n) + 0.2 * np.cos(2 * np.pi * f2 * k / n + rng.uniform(0, 2*np.pi))
    elif choice == 6:
        # Random smoothed
        a = rng.random(n) + 0.1
        win = max(3, n // 20)
        kernel = np.ones(win) / win
        a = np.convolve(a, kernel, mode='same')
    elif choice == 7:
        # Power law decay
        alpha = rng.uniform(0.3, 1.5)
        a = (k + 1) ** (-alpha) + 0.1
    elif choice == 8:
        # Bimodal
        c1 = n * rng.uniform(0.2, 0.4)
        c2 = n * rng.uniform(0.6, 0.8)
        w1 = n * rng.uniform(0.05, 0.15)
        w2 = n * rng.uniform(0.05, 0.15)
        a = 1.0 + np.exp(-0.5 * ((k - c1) / w1) ** 2) + np.exp(-0.5 * ((k - c2) / w2) ** 2)
    elif choice == 9:
        # Chirp
        f0 = rng.uniform(0.5, 2.0)
        f1 = rng.uniform(2.0, 5.0)
        a = 1.0 + 0.4 * np.cos(2 * np.pi * (f0 * k / n + (f1 - f0) * k**2 / (2 * n**2)))
    elif choice == 10:
        # Binary smoothed
        a = (rng.random(n) > 0.5).astype(float)
        win = max(3, n // 25)
        kernel = np.ones(win) / win
        a = np.convolve(a, kernel, mode='same')
    else:
        # Frequency-domain flat: inverse FFT of random phases on flat magnitude
        # This gives a sequence with (approximately) flat power spectrum
        spec = np.ones(n) * (1.0 + 0.3 * rng.random(n))
        phases = rng.uniform(0, 2 * np.pi, n)
        complex_spec = spec * np.exp(1j * phases)
        # Make it Hermitian for real output
        complex_spec[0] = spec[0]
        if n % 2 == 0:
            complex_spec[n // 2] = spec[n // 2]
        complex_spec[n // 2 + 1:] = np.conj(complex_spec[1:n // 2][::-1])
        a = np.fft.ifft(complex_spec).real
        a = np.abs(a) + 0.1

    a = np.maximum(a, 0.01)
    s = np.sum(a)
    return a / s

                a = diverse_init(n, seed + n * 100003 + init_idx * 7919, parent=None)

            # Step 1: Quick subgradient shaping
            sg_time = min(5.0, (alloc / n_inits) * 0.12)
            a = subgradient_optimize(a, max_iters=int(sg_time * 50),
                                      time_budget=sg_time, tau_start=3.0, tau_end=20.0)

            # Step 2: L-BFGS-B
            remaining_init = deadline - time.monotonic()
            if remaining_init > 20:
                tau_list = [4.0 / n, 2.0 / n, 1.0 / n, 0.5 / n, 0.25 / n]
                a, obj = optimize_lbfgs(a, tau_list, max_iter=200,
                                         time_budget=min((alloc / n_inits) * 0.25, 15.0))
            else:
                obj = compute_obj(a)

            # Step 3: Hybrid LP
            remaining_init = deadline - time.monotonic()
            if remaining_init > 25 and n <= 900:
                a = alternating_lp_hybrid(a, max_iters=40,
                                           time_budget=min((alloc / n_inits) * 0.4, 30.0))
                obj = compute_obj(a)

            # Step 4: Coordinate descent
            remaining_init = deadline - time.monotonic()
            if remaining_init > 12:
                a = coordinate_descent(a, time_budget=min(8.0, remaining_init * 0.15),
                                       n_sweeps=15)
                obj = compute_obj(a)

            # Step 5: Final subgradient
            remaining_init = deadline - time.monotonic()
            if remaining_init > 10:
                a = subgradient_optimize(a, max_iters=int(remaining_init * 25),
                                          time_budget=min(6.0, remaining_init * 0.12),
                                          tau_start=10.0, tau_end=35.0)
                obj = compute_obj(a)

            print(f"  init{init_idx}: n={n}, obj={obj:.7f}")

            if obj < best_score:
                best_score = obj
                best_seq = a.copy()
                print(f"  ** NEW BEST: n={n}, obj={best_score:.7f} **")

            candidates.append((a.copy(), obj, n))

    # Phase 2: Polish top candidates
    print(f"\n{'='*50}")
    print(f"=== POLISHING PHASE ===")
    print(f"Best before polish: {best_score:.7f}")

    candidates.sort(key=lambda x: x[1])
    n_polish = min(3, len(candidates))

    for ci in range(n_polish):
        if time.monotonic() > deadline - 100:
            break
        remaining = deadline - time.monotonic()
        if remaining < 40:
            break

        a_cand = candidates[ci][0]
        n_cand = candidates[ci][2]
        print(f"\n  Polishing candidate {ci} (n={n_cand}, obj={candidates[ci][1]:.7f}), {remaining:.0f}s left")

        polish_budget = min(remaining * 0.4, 120.0)
        a_pol, obj_pol = polish_sequence(a_cand, time_budget=polish_budget, verbose=True)

        if obj_pol < best_score:
            best_score = obj_pol
            best_seq = a_pol.copy()
            print(f"  ** UPDATE after polish: obj={best_score:.7f} **")

        # Perturbation restarts
        remaining = deadline - time.monotonic()
        if remaining > 35:
            rng = np.random.default_rng(ci * 555 + 999)
            for ptry in range(4):
                if time.monotonic() > deadline - 25:
                    break
                scale = rng.uniform(0.02, 0.10)
                a_pert = best_seq * np.exp(rng.normal(0, scale, len(best_seq)))
                a_pert = np.maximum(a_pert, 1e-10)
                s = np.sum(a_pert)
                if s < 1e-10:
                    continue
                a_pert = a_pert / s

                # Quick optimize pipeline
                a_pert = subgradient_optimize(a_pert, max_iters=200, time_budget=3.0,
                                               tau_start=5.0, tau_end=25.0)
                if len(a_pert) <= 900:
                    a_pert = alternating_lp_hybrid(a_pert, max_iters=15, time_budget=8.0)
                a_pert = coordinate_descent(a_pert, time_budget=3.0, n_sweeps=10)

                obj_pert = compute_obj(a_pert)
                if obj_pert < best_score:
                    best_score = obj_pert
                    best_seq = a_pert.copy()
                    print(f"  ** UPDATE (perturb {ptry}, scale={scale:.3f}): obj={best_score:.7f} **")

    # Phase 3: Final intensive polish on the global best
    if best_seq is not None and time.monotonic() < deadline - 25:
        remaining = deadline - time.monotonic()
        n_best = len(best_seq)
        print(f"\n{'='*50}")
        print(f"=== FINAL POLISH: n={n_best}, {remaining:.0f}s ===")

        a_cur = best_seq.copy()
        for rnd in range(6):
            if time.monotonic() > deadline - 12:
                break
            remaining = deadline - time.monotonic()
            if remaining < 18:
                break

            # LP
            if n_best <= 900 and remaining > 25:
                a_new = alternating_lp_hybrid(a_cur, max_iters=50,
                                               time_budget=min(remaining * 0.22, 28.0))
                obj_new = compute_obj(a_new)
                if obj_new < best_score:
                    best_score = obj_new
                    best_seq = a_new.copy()
                    a_cur = a_new.copy()
                    print(f"  Final LP {rnd}: {best_score:.7f}")

            # L-BFGS-B with very small tau
            remaining = deadline - time.monotonic()
            if remaining > 18:
                tau_list = [1.0 / n_best, 0.5 / n_best, 0.2 / n_best, 0.1 / n_best]
                a_new, obj_new = optimize_lbfgs(a_cur, tau_list, max_iter=250,
                                                 time_budget=min(remaining * 0.18, 15.0))
                if obj_new < best_score:
                    best_score = obj_new
                    best_seq = a_new.copy()
                    a_cur = a_new.copy()
                    print(f"  Final L-BFGS {rnd}: {best_score:.7f}")

            # Coordinate descent (more sweeps)
            remaining = deadline - time.monotonic()
            if remaining > 15:
                a_new = coordinate_descent(a_cur, time_budget=min(remaining * 0.15, 12.0),
                                           n_sweeps=40)
                obj_new = compute_obj(a_new)
                if obj_new < best_score:
                    best_score = obj_new
                    best_seq = a_new.copy()
                    a_cur = a_new.copy()
                    print(f"  Final CoordDesc {rnd}: {best_score:.7f}")

            # Subgradient
            remaining = deadline - time.monotonic()
            if remaining > 12:
                a_new = subgradient_optimize(a_cur, max_iters=int(remaining * 20),
                                              time_budget=min(remaining * 0.12, 12.0),
                                              tau_start=20.0, tau_end=70.0)
                obj_new = compute_obj(a_new)
                if obj_new < best_score:
                    best_score = obj_new
                    best_seq = a_new.copy()
                    a_cur = a_new.copy()
                    print(f"  Final SG {rnd}: {best_score:.7f}")

    # Final validation
    if best_seq is None:
        best_seq = np.ones(200)
        best_score = compute_obj(best_seq)

    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    val = evaluate_sequence(result)
    if val < np.inf:
        best_score = val
    else:
        result = [1.0] * 100
        best_score = 2.0

    elapsed = time.monotonic() - start_time
    print(f"\n{'='*50}")
    print(f"=== FINAL RESULT: n={len(result)}, score={best_score:.7f}, time={elapsed:.1f}s ===")

    return result
