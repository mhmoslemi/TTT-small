**Alternative: Direct "peak shaving" in time domain**

The idea: for each peak in the autocorrelation, identify which a_i values are most responsible and reduce them.

For r_k = sum_i a_i * a_{k-i}, the "responsibility" of a_i is proportional to a_{k-i} (or a_i * a_{k-i} for the pair).

So: for the peak at lag k*, reduce a_i most for those i where a_{k*-i} is large.

This is: a_i <- a_i - eta * a_{k*-i} (for i where 0 <= k*-i < n)

Minimizes J(a) = 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Exponential-family (mirror) peak shaving with adaptive step
2. Spectral-domain Gerchberg-Saxton peak reduction

4. Multi-scale progression (solve small n, upsample to large n)
5. Population-based diversification with parametric inits
6. Final coordinate-descent polish

import numpy as np
from scipy.optimize import linprog
from scipy.sparse import csr_matrix
import time

_spar_cache = {}

def get_sparsity(n):
    if n not in _spar_cache:
        rows = []
        cols = []
        for k in range(2 * n - 1):
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            rows.extend([k] * (je - js + 1))
            cols.extend(range(js, je + 1))
        _spar_cache[n] = (np.array(rows), np.array(cols), len(rows))
    return _spar_cache[n]

def fast_conv(a):
    n = len(a)
    s = 1
    while s < 2 * n - 1:
        s <<= 1
    fa = np.fft.rfft(a, s)
    return np.fft.irfft(fa * fa, s)[:2 * n - 1]

def compute_J(a):
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))

def mirror_peak_shave(a, max_iters, deadline, tracker=None):
    """Exponential-family updates targeting top autocorrelation peaks."""
    n = len(a)
    a = np.maximum(a.astype(np.float64), 1e-10)
    a = a / a.sum()
    current_J = compute_J(a)
    if tracker:
        tracker.update(a)
    eta = 0.3
    best_eta = 0.3
    stuck = 0

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        r = fast_conv(a)
        r_max = r.max()

        # Find top-k peaks (adaptive k)
        n_peaks = min(8, len(r))
        sorted_idx = np.argsort(r)[::-1]
        peak_indices = sorted_idx[:n_peaks]

        # Compute weighted gradient
        grad = np.zeros(n)
        peak_weights = r[peak_indices] / max(r_max, 1e-15)
        for idx, ki in enumerate(peak_indices):
            w = peak_weights[idx]
            js = max(0, ki - n + 1)
            je = min(n - 1, ki)
            for j in range(js, je + 1):
                grad[j] += w * a[ki - j]

        grad_max = grad.max()
        if grad_max < 1e-15:
            break
        grad /= grad_max

        # Exponential (mirror) update
        a_new = a * np.exp(-eta * grad)
        a_new = np.maximum(a_new, 1e-12)
        ss = a_new.sum()
        if ss < 1e-10:
            break
        a_new /= ss

        J_new = compute_J(a_new)
        if J_new < current_J - 1e-14:
            a = a_new
            current_J = J_new
            stuck = 0
            best_eta = eta
            eta = min(eta * 1.03, 2.0)
        else:
            stuck += 1
            eta = max(eta * 0.75, 0.005)
            if stuck > 50:
                break

        if tracker and it % 20 == 0:
            tracker.update(a)

    if tracker:
        tracker.update(a)
    return a

def gs_spectral_shaping(a, max_iters, deadline, tracker=None):
    """Gerchberg-Saxton style spectral shaping to flatten autocorrelation."""
    n = len(a)
    N = 1
    while N < 2 * n - 1:
        N <<= 1

    a = np.maximum(a.astype(np.float64), 1e-10)
    a = a / a.sum()
    current_J = compute_J(a)
    if tracker:
        tracker.update(a)

    alpha = 0.5  # shaping strength
    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        A = np.fft.rfft(a, N)
        power = np.abs(A) ** 2
        r = np.fft.irfft(power, N)[:2 * n - 1]

        r_max = r.max()
        r_min = r.min()
        r_range = r_max - r_min
        if r_range < 1e-12:
            break

        # Shape power spectrum: reduce power where autocorrelation is peaked
        # r = IFFT(power), so reducing power at freqs contributing to peaks
        # Approximate: scale power by exp(-alpha * normalized_r)
        r_norm = (r - r_min) / r_range
        # Extend r_norm to full power length
        power_shape = np.exp(-alpha * r_norm)
        # Pad power_shape to match power length
        if len(power_shape) < len(power):
            power_shape = np.concatenate([power_shape, np.ones(len(power) - len(power_shape))])
        else:
            power_shape = power_shape[:len(power)]

        power_new = power * power_shape
        # Ensure non-negative and not all zero
        if power_new.max() < 1e-15:
            break

        # Reconstruct with original phase
        phase = np.angle(A)
        A_new = np.sqrt(power_new) * np.exp(1j * phase)

        # Build full spectrum
        A_full = np.zeros(N, dtype=complex)
        A_full[:len(A)] = A_new
        for k in range(1, min(len(A) - 1, N // 2)):
            A_full[N - k] = np.conj(A_new[k])
        if N % 2 == 0 and len(A) > N // 2:
            A_full[N // 2] = A_new[N // 2]

        a_new = np.fft.irfft(A_full, N)[:n]
        a_new = np.maximum(a_new, 1e-10)
        ss = a_new.sum()
        if ss < 1e-10:
            break
        a_new /= ss

        J_new = compute_J(a_new)
        if J_new < current_J - 1e-14:
            a = a_new
            current_J = J_new
            alpha = min(alpha * 1.02, 3.0)
        else:
            alpha = max(alpha * 0.85, 0.01)
            if alpha < 0.01:
                break

        if tracker and it % 30 == 0:
            tracker.update(a)

    if tracker:
        tracker.update(a)
    return a

def lp_best_response(a, M):
    n = len(a)
    rows, cols, nnz = get_sparsity(n)
    vals = np.empty(nnz)
    for i in range(nnz):
        k = rows[i]
        j = cols[i]
        vals[i] = a[k - j]
    A = csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                options={'presolve': True, 'time_limit': 8})
    if r.success and r.x is not None:
        g = np.maximum(r.x, 0.0)
        if g.sum() > 1e-10:
            return g
    return None

def run_lp_opt(a, max_iters, deadline, tracker, tighten=0.01):
    a = np.maximum(a.astype(np.float64), 1e-10)
    a = a / a.sum()
    n = len(a)
    current_J = compute_J(a)
    if tracker:
        tracker.update(a)

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        b = fast_conv(a)
        M = float(b.max())
        M_tight = M * (1.0 - tighten)
        g = lp_best_response(a, M_tight)
        if g is None:
            g = lp_best_response(a, M)
        if g is None:
            break
        g = np.maximum(g, 0.0)
        gs = g.sum()
        if gs < 1e-10:
            break
        g = g / gs

        best_a = None
        best_J = current_J
        progress = it / max(1, max_iters)
        base_t = max(0.03, 0.4 * (1.0 - progress))
        for t in [base_t, base_t * 1.5, base_t * 0.5, 0.7, 0.2]:
            if t > 1.0 or t < 0.01:
                continue
            a_trial = (1.0 - t) * a + t * g
            a_trial = np.maximum(a_trial, 1e-10)
            ss = a_trial.sum()
            if ss < 1e-10:
                continue
            a_trial = a_trial / ss
            J_trial = compute_J(a_trial)
            if J_trial < best_J - 1e-14:
                best_J = J_trial
                best_a = a_trial

        if best_a is None:
            break
        a = best_a
        current_J = best_J
        if tracker and it % 10 == 0:
            tracker.update(a)
    if tracker:
        tracker.update(a)
    return a

def multi_peak_subgradient(a, max_iters, deadline, tracker=None):
    """Subgradient method targeting multiple peaks simultaneously."""
    n = len(a)
    a = np.maximum(a.astype(np.float64), 1e-10)
    a = a / a.sum()
    current_J = compute_J(a)
    if tracker:
        tracker.update(a)
    lr = 1.0 / n
    rng = np.random.default_rng(123)
    stuck = 0

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        b = fast_conv(a)
        b_max = float(b.max())

        # Find active set (peaks within epsilon of max)
        eps = 0.01 * b_max
        active = np.where(b >= b_max - eps)[0]

        # Combined subgradient
        grad = np.zeros(n)
        for k in active:
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            for j in range(js, je + 1):
                grad[j] += a[k - j]
        grad /= max(len(active), 1)

        # Add regularization toward uniform (prevents collapse)
        reg = 0.01 * (a - 1.0 / n)

        direction = grad + reg
        a_new = a - lr * direction
        a_new = np.maximum(a_new, 1e-10)
        a_new = a_new / a_new.sum()

        J_new = compute_J(a_new)
        if J_new < current_J - 1e-14:
            a = a_new
            current_J = J_new
            stuck = 0
            lr = min(lr * 1.02, 10.0 / n)
        else:
            stuck += 1
            lr *= 0.7
            if stuck > 30:
                break

        if tracker and it % 50 == 0:
            tracker.update(a)

    if tracker:
        tracker.update(a)
    return a

def resample_seq(src, dest_n):
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size, endpoint=False)
    t_new = np.linspace(0, 1, dest_n, endpoint=False)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)

def generate_diverse_inits(n, rng, parent=None):
    """Generate diverse initial sequences."""
    inits = []
    t = np.linspace(0, 1, n, endpoint=False)

    if parent is not None and parent.size > 0:
        inits.append(resample_seq(parent, n))
        inits.append(resample_seq(parent[::-1], n))

    # Uniform
    inits.append(np.ones(n) / n)

    # Raised cosine (Hann)
    inits.append(0.5 + 0.5 * np.cos(2 * np.pi * t))

    # Blackman-like
    inits.append(0.42 + 0.5 * np.cos(2 * np.pi * t) + 0.08 * np.cos(4 * np.pi * t))

    # Gaussian centered
    for sigma in [0.2, 0.3, 0.4, 0.5]:
        inits.append(np.exp(-((t - 0.5) ** 2) / (2 * sigma ** 2)))

    # Power-law edges
    inits.append(np.sin(np.pi * t) ** 2)
    inits.append(np.sin(np.pi * t) ** 4)
    inits.append(np.sqrt(np.sin(np.pi * t)))

    # Multi-frequency modulations
    for f in [1, 2, 3, 5]:
        for amp in [0.2, 0.4, 0.6]:
            inits.append(1.0 + amp * np.cos(2 * np.pi * f * t))

    # Tailored: high in middle, low at edges, with ripple
    base = np.sin(np.pi * t)
    for freq in [4, 8, 12, 16]:
        inits.append(base * (1.0 + 0.3 * np.cos(2 * np.pi * freq * t)))

    # Flat spectrum (random phases)
    for _ in range(3):
        half = n // 2 + 1
        angles = rng.uniform(0, 2 * np.pi, half)
        A = np.zeros(n, dtype=complex)
        A[:half] = np.exp(1j * angles)
        for k in range(1, min(n // 2, n - 1)):
            A[n - k] = np.conj(A[k])
        a = np.fft.ifft(A).real
        inits.append(np.maximum(a, 0.0) + 0.01)

    # Random smooth
    for _ in range(2):
        n_coeff = min(n // 4 + 1, 10)
        coeffs = rng.normal(0, 1, n_coeff)
        sig = np.zeros(n)
        for i in range(n_coeff):
            sig += coeffs[i] * np.cos(2 * np.pi * i * t + rng.uniform(0, 6.28))
        inits.append(np.abs(sig) + 0.3)

    # Two-peak structure
    inits.append(np.exp(-8 * (t - 0.25) ** 2) + np.exp(-8 * (t - 0.75) ** 2) + 0.05)

    # Asymmetric
    inits.append(np.exp(-3 * (t - 0.4) ** 2) + 0.05)
    inits.append(np.exp(-3 * (t - 0.6) ** 2) + 0.05)

    # Boxcar + taper
    for taper_w in [0.1, 0.2, 0.3]:
        box = np.ones(n)
        n_t = int(taper_w * n)
        for i in range(n_t):
            box[i] = i / n_t
            box[n - 1 - i] = i / n_t
        inits.append(box)

    # Normalize all
    valid_inits = []
    for x in inits:
        x = np.maximum(np.asarray(x, dtype=np.float64), 0.001)
        s = x.sum()
        if s > 1e-10:
            valid_inits.append(x / s)
    return valid_inits

def coordinate_descent(a, max_steps, deadline, tracker):
    """Randomized coordinate descent with multiplicative updates."""
    a = a.copy().astype(np.float64)
    a = a / a.sum()
    n = len(a)
    current_J = compute_J(a)
    tracker.update(a)
    rng = np.random.default_rng(42)

    for step in range(max_steps):
        if time.monotonic() > deadline:
            break
        j = rng.integers(0, n)
        improved = False
        for scale in [0.7, 1.3, 0.5, 1.5, 0.85, 1.15]:
            a_trial = a.copy()
            a_trial[j] *= scale
            if a_trial[j] < 0:
                continue
            a_trial = np.maximum(a_trial, 1e-10)
            ss = a_trial.sum()
            if ss < 1e-10:
                continue
            a_trial /= ss
            J_trial = compute_J(a_trial)
            if J_trial < current_J - 1e-14:
                a = a_trial
                current_J = J_trial
                improved = True
                break
        if not improved and step % 500 == 0 and step > 0:
            # Try swapping two coordinates
            j1, j2 = rng.integers(0, n, 2)
            for mix in [0.3, 0.7]:
                a_trial = a.copy()
                a_trial[j1], a_trial[j2] = a[j2], a[j1]
                a_trial = np.maximum(a_trial, 1e-10)
                a_trial /= a_trial.sum()
                J_trial = compute_J(a_trial)
                if J_trial < current_J - 1e-14:
                    a = a_trial
                    current_J = J_trial
                    break
        if step % 200 == 0:
            tracker.update(a)
    tracker.update(a)
    return a

def lse_gradient_descent(a, max_iters, deadline, tracker=None):
    """Log-sum-exp smoothed gradient descent (differentiable surrogate for max)."""
    n = len(a)
    a = np.maximum(a.astype(np.float64), 1e-6)
    a = a / a.sum()
    current_J = compute_J(a)
    if tracker:
        tracker.update(a)

    tau = 30.0
    lr = 1.0 / (n * tau)
    momentum = 0.8
    v = np.zeros(n)
    stuck = 0

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        b = fast_conv(a)
        b_max = b.max()
        exp_b = np.exp(tau * (b - b_max))
        p = exp_b / exp_b.sum()

        # Gradient of LSE proxy: cross-correlation of p with a
        N = 1
        while N < 3 * n:
            N <<= 1
        fp = np.fft.rfft(p, N)
        fa = np.fft.rfft(a, N)
        corr = np.fft.irfft(fp * np.conj(fa), N)
        grad = 2.0 * corr[:n]

        v = momentum * v + (1.0 - momentum) * grad
        a_new = a - lr * v
        a_new = np.maximum(a_new, 1e-6)
        a_new = a_new / a_new.sum()

        J_new = compute_J(a_new)
        if J_new < current_J - 1e-14:
            a = a_new
            current_J = J_new
            stuck = 0
            tau = min(tau * 1.01, 200.0)
            lr = 1.0 / (n * tau)
        else:
            stuck += 1
            lr *= 0.5
            v *= 0.5
            if stuck > 40 or lr < 1e-18:
                break

        if tracker and it % 100 == 0:
            tracker.update(a)

    if tracker:
        tracker.update(a)
    return a

            parent = np.maximum(p, 0.0)
            tracker.update(parent)
            print(f"Parent: n={parent.size}, J={tracker.best_score:.8f}", flush=True)
    except (NameError, AttributeError):
        pass

    # === Phase 0: Multi-scale warm start ===
    # Solve at small n first to find good structure, then upsample
    if parent is None:
        small_n = 64
        inits_small = generate_diverse_inits(small_n, rng, None)
        best_small = None
        best_small_J = float('inf')
        for a0 in inits_small[:8]:
            a_t = mirror_peak_shave(a0, 200, time.monotonic() + 5, tracker)
            J_t = compute_J(a_t)
            if J_t < best_small_J:
                best_small_J = J_t
                best_small = a_t.copy()
        if best_small is not None:
            tracker.update(best_small)
            print(f"Phase 0 small-n best: J={best_small_J:.6f}", flush=True)

    # === Phase 1: Multi-length optimization ===
    lengths = [256, 320, 384, 448, 512]
    remaining = deadline - time.monotonic() - 200
    time_per_length = remaining / len(lengths)

    for n in lengths:
        t_phase_start = time.monotonic()
        t_phase_end = min(deadline - 180, t_phase_start + time_per_length)
        if time.monotonic() >= t_phase_end - 20:
            continue
        print(f"\n{'='*50} Phase1 n={n} budget={time_per_length:.1f}s {'='*50}", flush=True)

        inits = generate_diverse_inits(n, rng, parent)
        best_for_n = None
        best_J_for_n = float('inf')
        n_inits = len(inits)
        init_budget = max(8.0, (t_phase_end - time.monotonic()) / n_inits)

        for idx, a0 in enumerate(inits):
            if time.monotonic() > t_phase_end - 15:
                break
            t0_init = time.monotonic()
            budget_left = t_phase_end - 15
            third = budget_left / 3.0

            # Step 1: Mirror peak shave (primary method - different from previous)
            a = mirror_peak_shave(a0, int(200 * third / max(1, (t_phase_end - t0_init) / n_inits * 3)),
                                  t0_init + third, tracker)

            # Step 2: GS spectral shaping
            if time.monotonic() < t0_init + 2 * third - 5:
                a = gs_spectral_shaping(a, int(100 * third / max(1, (t_phase_end - t0_init) / n_inits * 3)),
                                        t0_init + 2 * third - 5, tracker)

            # Step 3: LP best-response
            if time.monotonic() < t_phase_end - 15:
                n_lp = max(5, int(30 * third / max(1, (t_phase_end - t0_init) / n_inits * 3)))
                a = run_lp_opt(a, n_lp, t_phase_end - 15, tracker, tighten=0.01)

            J = compute_J(a)
            if J < best_J_for_n:
                best_J_for_n = J
                best_for_n = a.copy()

        if best_for_n is not None:
            tracker.update(best_for_n)
            if parent is None or best_J_for_n < compute_J(resample_seq(parent, best_for_n.size)):
                parent = best_for_n.copy()
            tracker.report(f"Phase1 n={n} ({time.monotonic()-t_phase_start:.1f}s)")

    # === Phase 2: Focused deep optimization on best ===
    if tracker.best is not None and time.monotonic() < deadline - 150:
        t0 = time.monotonic()
        n = tracker.best.size
        print(f"\n{'='*50} Phase2 Focus n={n} {'='*50}", flush=True)

        a = tracker.best.copy().astype(np.float64)

        # Aggressive mirror shaving
        a = mirror_peak_shave(a, 500, min(deadline - 120, t0 + 60), tracker)

        # GS spectral
        if time.monotonic() < deadline - 100:
            a = gs_spectral_shaping(a, 300, min(deadline - 100, t0 + 100), tracker)

        # LP with progressive tightening
        tighten_sched = [0.005, 0.01, 0.015, 0.02, 0.025, 0.03, 0.04]
        for tight in tighten_sched:
            if time.monotonic() > deadline - 80:
                break
            a = run_lp_opt(a, 20, deadline - 80, tracker, tighten=tight)

        # Multi-peak subgradient
        if time.monotonic() < deadline - 60:
            a = multi_peak_subgradient(a, 3000, deadline - 60, tracker)

        # LSE polish
        if time.monotonic() < deadline - 40:
            a = lse_gradient_descent(a, 2000, deadline - 40, tracker)

        # Final mirror shave
        if time.monotonic() < deadline - 20:
            a = mirror_peak_shave(a, 300, deadline - 20, tracker)

        tracker.report(f"Phase2 ({time.monotonic()-t0:.1f}s)")
        parent = tracker.best.copy()

    # === Phase 3: Diversification + re-optimization ===
    if tracker.best is not None and time.monotonic() < deadline - 80:
        t0 = time.monotonic()
        n = tracker.best.size
        print(f"\n{'='*50} Phase3 Diversify n={n} {'='*50}", flush=True)

        # Create perturbed versions and re-optimize
        for pert in [0.05, 0.1, 0.15]:
            if time.monotonic() > deadline - 50:
                break
            a_pert = tracker.best + rng.normal(0, pert, n)
            a_pert = np.maximum(a_pert, 1e-8)
            a_pert = a_pert / a_pert.sum()

            a = mirror_peak_shave(a_pert, 200, deadline - 40, tracker)
            if time.monotonic() < deadline - 30:
                a = run_lp_opt(a, 15, deadline - 30, tracker, tighten=0.01)
            if time.monotonic() < deadline - 20:
                a = lse_gradient_descent(a, 500, deadline - 20, tracker)

        # Try different lengths near the best
        for n_try in [n - 32, n + 32, n + 64]:
            if n_try < 32 or time.monotonic() > deadline - 40:
                continue
            a_resamp = resample_seq(tracker.best, n_try)
            a = mirror_peak_shave(a_resamp, 150, deadline - 30, tracker)
            if time.monotonic() < deadline - 20:
                a = run_lp_opt(a, 10, deadline - 20, tracker, tighten=0.01)

        tracker.report(f"Phase3 ({time.monotonic()-t0:.1f}s)")

    # === Phase 4: Final coordinate descent polish ===
    if tracker.best is not None and time.monotonic() < deadline - 10:
        t0 = time.monotonic()
        print(f"\n{'='*50} Phase4 CoordDescent {'='*50}", flush=True)
        a = coordinate_descent(tracker.best, 5000, deadline - 5, tracker)

        # Final LP
        if time.monotonic() < deadline - 3:
            a = run_lp_opt(a, 10, deadline - 2, tracker, tighten=0.005)

        # Final LSE
        if time.monotonic() < deadline - 1:
            a = lse_gradient_descent(a, 300, deadline - 0.5, tracker)

        tracker.report(f"Phase4 ({time.monotonic()-t0:.1f}s)")

    # Fallback
    if tracker.best is None:
        a = np.ones(256)
        a = a / a.sum()
        tracker.best = a
        tracker.best_score = compute_J(a)
        tracker.update(tracker.best)

    print(f"\n{'='*50} FINAL {'='*50}", flush=True)
    print(f"J={tracker.best_score:.8f} n={tracker.best.size} time={time.monotonic()-tracker.t_start:.1f}s", flush=True)
    return [float(x) for x in tracker.best]
