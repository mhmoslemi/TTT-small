            diag[j] = p[k] * a[j]
    
    grad = 2.0 * corr - diag
    return smooth_val, grad

Let me also think about the **coordinate descent** improvement. The key insight is:
- When we change a_i, only O(n) convolution values change
- We can compute the new max in O(n) time
- We can do a 1D optimization (golden section or even cubic interpolation)

For the 1D problem: minimize J(x) = 2*n*max(max_unaff, max_aff(x)) / (S - a_i + x)^2

where max_aff(x) is the max of the affected convolution values as a function of x.

Each affected value is of the form:
- b[i+off] + 2*(x-ai)*a[off] (linear in x) for off != i
- b[2i] + x^2 - ai^2 (quadratic in x) for off = i (i.e., at position i in the affected array)

So max_aff(x) is the max of O(n) linear functions and one quadratic function. This is a piecewise linear/quadratic function. The minimum of J(x) occurs where:
1. The derivative of the active constraint equals the derivative of the denominator, OR
2. Two constraints intersect (kink point)

This is actually a well-structured 1D problem! We could solve it exactly by:
1. Finding all the "breakpoints" where the argmax changes
2. Checking the minimum in each interval

But finding all breakpoints is O(n^2) in the worst case. For practical purposes, golden section search with 30-40 iterations should be sufficient.

"""

import numpy as np
from scipy.optimize import linprog
from scipy.sparse import lil_matrix
import time


def fast_conv(a):
    """Full self-convolution via FFT."""
    n = len(a)
    if n == 0:
        return np.array([0.0])
    s = 1
    while s < 2 * n - 1:
        s <<= 1
    fa = np.fft.rfft(a, s)
    return np.fft.irfft(fa * fa, s)[:2 * n - 1]


def compute_J(a):
    """Compute J(a) = 2*n*max(conv(a,a))/(sum(a))^2."""
    a = np.maximum(a, 0)
    S = a.sum()
    if S < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (S * S))


def smooth_max_grad_fast(a, tau):
    """Compute smooth max of self-convolution and gradient using FFT-based correlation."""
    n = len(a)
    b = fast_conv(a)
    max_b = b.max()
    exp_b = np.exp(tau * (b - max_b))
    p = exp_b / exp_b.sum()
    smooth_val = max_b + np.log(exp_b.sum()) / tau

    # Compute (p correlation a)_j = sum_{k=j}^{min(j+n,2n-1)-1} p[k]*a[k-j]
    # via FFT: convolve(p, a_flipped)[n-1+j]
    size = 1
    while size < (2 * n - 1) + n:
        size <<= 1
    P = np.fft.rfft(p, size)
    A_flip = np.fft.rfft(a[::-1].astype(np.float64), size)
    conv = np.fft.irfft(P * A_flip, size)
    corr = conv[n - 1:2 * n - 1]

    # Diagonal correction: where k=2j, we used 2*a[k-j] but should use a[k-j]
    diag = np.zeros(n)
    for j in range(n):
        k = 2 * j
        if 0 <= k < 2 * n - 1:
            diag[j] = p[k] * a[j]

    grad = 2.0 * corr - diag
    return smooth_val, grad


def adam_smooth_max(a, n_steps, deadline, tracker, tau_start=5.0, tau_end=80.0, lr=0.04):
    """Adam optimizer on smooth max approximation with tau annealing."""
    n = len(a)
    a = a.copy().astype(np.float64)
    a = np.maximum(a, 1e-8)
    a /= a.sum()

    m = np.zeros(n)
    v = np.zeros(n)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8

    best_J = compute_J(a)
    tracker.update(a)
    stall = 0

    for step in range(n_steps):
        if time.monotonic() > deadline:
            break

        progress = step / max(1, n_steps)
        tau = tau_start + (tau_end - tau_start) * (progress ** 2)

        _, grad = smooth_max_grad_fast(a, tau)

        # Adam update
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad * grad)
        m_hat = m / (1 - beta1 ** (step + 1))
        v_hat = v / (1 - beta2 ** (step + 1))

        # Adaptive learning rate
        effective_lr = lr * (1.0 - 0.6 * progress)
        a_new = a - effective_lr * m_hat / (np.sqrt(v_hat) + eps)

        # Project to simplex
        a_new = np.maximum(a_new, 0.0)
        if a_new.sum() < 1e-10:
            a_new = a.copy()
        else:
            a_new /= a_new.sum()

        a = a_new
        J_new = compute_J(a)
        if J_new < best_J - 1e-12:
            best_J = J_new
            stall = 0
            tracker.update(a)
        else:
            stall += 1
            if stall > 5000:
                break

        if step % 2000 == 0:
            print(f"    Adam step {step}: J={best_J:.8f}, tau={tau:.1f}", flush=True)

    return a


def fista_smooth_max(a, n_steps, deadline, tracker, tau_start=10.0, tau_end=60.0):
    """FISTA (accelerated proximal gradient) on smooth max."""
    n = len(a)
    a_curr = a.copy().astype(np.float64)
    a_curr = np.maximum(a_curr, 1e-8)
    a_curr /= a_curr.sum()
    a_prev = a_curr.copy()

    best_J = compute_J(a_curr)
    tracker.update(a_curr)

    t_k = 1.0
    base_eta = 0.02

    for step in range(n_steps):
        if time.monotonic() > deadline:
            break

        progress = step / max(1, n_steps)
        tau = tau_start + (tau_end - tau_start) * (progress ** 2)
        eta = base_eta * (1.0 - 0.4 * progress)

        _, grad = smooth_max_grad_fast(a_curr, tau)

        # FISTA momentum
        a_y = a_prev + ((t_k - 1.0) / (t_k + 2.0)) * (a_curr - a_prev)
        a_new = a_y - eta * grad

        # Project to simplex
        a_new = np.maximum(a_new, 0.0)
        if a_new.sum() < 1e-10:
            a_new = a_curr.copy()
        else:
            a_new /= a_new.sum()

        J_new = compute_J(a_new)
        if J_new < best_J:
            best_J = J_new
            tracker.update(a_new)

        a_prev = a_curr
        a_curr = a_new
        t_k = (1.0 + np.sqrt(1.0 + 4.0 * t_k * t_k)) / 2.0

        if step % 2000 == 0:
            print(f"    FISTA step {step}: J={best_J:.8f}, tau={tau:.1f}, t={t_k:.1f}", flush=True)

    return a_curr


def build_A_sparse(a):
    """Build sparse constraint matrix for LP: A[g] = (a*g)_k for all k."""
    n = len(a)
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        jl = max(0, k - n + 1)
        jh = min(n - 1, k)
        for j in range(jl, jh + 1):
            rows.append(k)
            cols.append(j)
            vals.append(a[k - j])
    A = lil_matrix((2 * n - 1, n))
    for idx in range(len(rows)):
        A[rows[idx], cols[idx]] = vals[idx]
    return A.tocsr()


def lp_best_response(a, M):
    """LP: max sum(g) s.t. (a*g)_k <= M for all k, g >= 0."""
    n = len(a)
    c = -np.ones(n)
    A = build_A_sparse(a)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
    if r.success and r.x is not None:
        g = np.maximum(r.x, 0.0)
        if g.sum() > 1e-10:
            return g
    return None


def lp_mirror_descent(a, max_iters, deadline, tracker, tighten_base=0.003):
    """LP-based mirror descent with adaptive tightening."""
    a = np.maximum(a.astype(np.float64), 0.001)
    a /= a.sum()
    n = len(a)
    current_J = compute_J(a)
    tracker.update(a)

    tighten = tighten_base
    stall_count = 0

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        b = fast_conv(a)
        M = float(b.max())
        M_target = M * (1.0 - tighten)

        g = lp_best_response(a, M_target)
        if g is None:
            M_target = M * (1.0 - tighten * 0.4)
            g = lp_best_response(a, M_target)
        if g is None:
            tighten *= 0.5
            if tighten < 1e-6:
                break
            continue

        g = np.maximum(g, 0.0)
        g_sum = g.sum()
        if g_sum < 1e-10:
            break
        g /= g_sum

        # Try multiple step sizes
        progress = min(1.0, it / max(1, max_iters))
        base_t = max(0.02, 0.6 * (1.0 - 0.7 * progress))
        candidates_t = [base_t, base_t * 0.5, base_t * 1.5, base_t * 0.3]

        best_a = None
        best_J = current_J

        for t in candidates_t:
            a_trial = (1.0 - t) * a + t * g
            a_trial = np.maximum(a_trial, 0.0)
            s = a_trial.sum()
            if s < 1e-10:
                continue
            a_trial /= s
            J_trial = compute_J(a_trial)
            if J_trial < best_J - 1e-14:
                best_J = J_trial
                best_a = a_trial

        if best_a is None:
            stall_count += 1
            if stall_count > 30:
                break
            tighten *= 0.7
            continue

        a = best_a
        current_J = best_J
        stall_count = 0

        if it % 3 == 0:
            tighten = min(0.02, tighten * 1.02)
        else:
            tighten = max(0.001, tighten * 0.97)

        if it % 10 == 0:
            tracker.update(a)

    tracker.update(a)
    return a


def coord_opt_fast(a, i, b, S):
    """Faster coordinate optimization for index i using vectorized updates."""
    n = len(a)
    ai = a[i]

    # Precompute max of unaffected b values
    if n == 1:
        max_unaffected = 0.0
    else:
        unaffected = np.ones(2 * n - 1, dtype=bool)
        unaffected[i:i + n] = False
        if unaffected.any():
            max_unaffected = b[unaffected].max()
        else:
            max_unaffected = 0.0

    # Affected b values: b[i], b[i+1], ..., b[i+n-1]
    b_aff = b[i:i + n].copy()

    def eval_obj(x):
        new_S = S - ai + x
        if new_S < 1e-12:
            return 1e10
        if x == ai:
            max_c = max_unaffected
            if b_aff.max() > max_c:
                max_c = b_aff.max()
        else:
            # Vectorized update
            new_b_aff = b_aff + 2.0 * (x - ai) * a
            # Fix diagonal: at position i in b_aff (which is b[2i])
            if i < n:
                new_b_aff[i] = b_aff[i] + x * x - ai * ai
            max_c = max_unaffected
            if new_b_aff.max() > max_c:
                max_c = new_b_aff.max()
        return 2.0 * n * max_c / (new_S * new_S)

    # Golden section search
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    lo = 0.0
    hi = max(3.0 * ai, (S / n) * 3.0, 0.01)

    c_pt = hi - gr * (hi - lo)
    d_pt = lo + gr * (hi - lo)
    fc = eval_obj(c_pt)
    fd = eval_obj(d_pt)

    for _ in range(35):
        if fc < fd:
            hi = d_pt
            d_pt = c_pt
            fd = fc
            c_pt = hi - gr * (hi - lo)
            fc = eval_obj(c_pt)
        else:
            lo = c_pt
            c_pt = d_pt
            fc = fd
            d_pt = lo + gr * (hi - lo)
            fd = eval_obj(d_pt)

    x_opt = (lo + hi) / 2.0
    return max(0.0, x_opt)


def coordinate_descent_pass(a, rng, max_coords=None):
    """One pass of coordinate descent with vectorized updates."""
    a = np.maximum(a.astype(np.float64), 0.0)
    S = a.sum()
    if S < 1e-10:
        return a, compute_J(a)
    n = len(a)
    b = fast_conv(a)

    indices = list(range(n))
    rng.shuffle(indices)
    if max_coords is not None:
        indices = indices[:min(max_coords, n)]

    for i in indices:
        old_val = a[i]
        new_val = coord_opt_fast(a, i, b, S)
        if abs(new_val - old_val) > 1e-12:
            S = S - old_val + new_val
            a[i] = new_val
            # Update convolution incrementally
            for k in range(max(0, i), min(2 * n - 1, i + n)):
                ki = k - i
                if 0 <= ki < n:
                    if k == 2 * i:
                        b[k] = b[k] - old_val * old_val + new_val * new_val
                    else:
                        b[k] = b[k] - 2.0 * old_val * a[ki] + 2.0 * new_val * a[ki]

    b_check = fast_conv(a)
    new_J = 2.0 * n * b_check.max() / (S * S)
    return a, new_J


def resample_seq(src, dest_n):
    """Resample 1D sequence to new length via linear interpolation."""
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)


def make_palindromic(a, n_target):
    """Create a palindromic sequence of length n_target from a."""
    half = n_target // 2
    # Resample to get half the values
    src = np.asarray(a, dtype=np.float64)
    if src.size > half:
        t_old = np.linspace(0, 1, src.size)
        t_new = np.linspace(0, 1, half)
        half_vals = np.maximum(np.interp(t_new, t_old, src), 0.0)
    else:
        t_old = np.linspace(0, 1, src.size)
        t_new = np.linspace(0, 1, half)
        half_vals = np.maximum(np.interp(t_new, t_old, src), 0.0)

    if n_target % 2 == 0:
        pal = np.concatenate([half_vals, half_vals[::-1]])
    else:
        pal = np.concatenate([half_vals, half_vals[-1], half_vals[-1:0:-1]])
    return pal


def generate_structured_inits(n, rng):
    """Generate diverse structured initial sequences."""
    inits = []
    t = np.linspace(0, 1, n, endpoint=False)

    inits.append(np.ones(n))
    inits.append(rng.uniform(0.3, 1.0, n))

    for freq in [2, 3, 5, 7, 11]:
        inits.append(1.0 + 0.4 * np.cos(2.0 * np.pi * freq * t + rng.uniform(0, 6.28)))

    inits.append(1.0 + 0.3 * np.cos(2 * np.pi * 3 * t) + 0.2 * np.cos(2 * np.pi * 7 * t))
    inits.append(np.abs(np.sin(np.pi * t)) + 0.3)
    inits.append(0.5 + 0.5 * np.cos(2 * np.pi * t) ** 2)
    inits.append(1.0 - 0.5 * np.abs(2 * t - 1.0))
    inits.append(0.5 * np.exp(-10 * (t - 0.3) ** 2) + 0.5 * np.exp(-10 * (t - 0.7) ** 2) + 0.1)

    # Power law
    inits.append(1.0 / (0.5 * n * t + 0.5) ** 0.5)
    # Gaussian
    inits.append(np.exp(-4.0 * (t - 0.5) ** 2) + 0.1)
    # Ramp
    inits.append(np.linspace(0.5, 1.0, n))
    # V-shaped
    inits.append(np.abs(2 * t - 1.0) + 0.5)

    return inits

                if time.monotonic() > deadline - 280:
                    break
                a_pal, Jp = coordinate_descent_pass(a_pal, rng)
                a_pal /= a_pal.sum()
                tracker.update(a_pal)

        tracker.report(f"Phase 5 Palindromic: {time.monotonic() - t0:.1f}s")

    # === Phase 6: Multi-length exploration ===
    if time.monotonic() < deadline - 250:
        print(f"\n{'=' * 50} Phase 6: Multi-Length Exploration {'=' * 50}", flush=True)
        t0 = time.monotonic()

        # Try nearby lengths
        nearby_lengths = []
        for delta in [-6, -4, -2, 2, 4, 6, 8, 10, -8, 12]:
            nl = n_parent + delta
            if 150 <= nl <= 700 and nl not in nearby_lengths:
                nearby_lengths.append(nl)

        # Also try some "round" numbers
        for nl in [256, 300, 320, 384, 400, 448, 512, 512]:
            if nl not in nearby_lengths and 150 <= nl <= 700:
                nearby_lengths.append(nl)

        budget_per_length = max(20, (deadline - time.monotonic() - 200) / max(1, len(nearby_lengths)))

        for nl in nearby_lengths:
            if time.monotonic() > deadline - 180:
                break
            if nl == n_parent:
                continue

            a_resampled = resample_seq(tracker.best, nl)
            a_resampled = np.maximum(a_resampled, 1e-8)
            a_resampled /= a_resampled.sum()
            tracker.update(a_resampled)

            # Quick Adam
            a_resampled = adam_smooth_max(a_resampled, 5000,
                                          time.monotonic() + min(budget_per_length * 0.6, 60),
                                          tracker, tau_start=20.0, tau_end=60.0, lr=0.04)

            # Quick CD
            if time.monotonic() < deadline - 180:
                a_resampled, Jr = coordinate_descent_pass(a_resampled, rng)
                a_resampled /= a_resampled.sum()
                tracker.update(a_resampled)

            print(f"    Length {nl}: J={tracker.best_score:.8f}", flush=True)

        tracker.report(f"Phase 6 Multi-Length: {time.monotonic() - t0:.1f}s")

    # === Phase 7: Diverse restarts ===
    if time.monotonic() < deadline - 150:
        print(f"\n{'=' * 50} Phase 7: Diverse Restarts {'=' * 50}", flush=True)
        t0 = time.monotonic()

        n_best = tracker.best.size
        inits = generate_structured_inits(n_best, rng)

        # Also add perturbed versions of the best
        for temp in [0.02, 0.05, 0.1]:
            perturb = tracker.best.copy() + rng.normal(0, temp, n_best)
            perturb = np.maximum(perturb, 1e-8)
            perturb /= perturb.sum()
            inits.append(perturb)

        budget_per_init = max(15, (deadline - time.monotonic() - 120) / max(1, len(inits)))

        for init_a in inits:
            if time.monotonic() > deadline - 120:
                break

            init_a = np.maximum(init_a.astype(np.float64), 1e-8)
            init_a /= init_a.sum()
            tracker.update(init_a)

            # Adam
            init_a = adam_smooth_max(init_a, 5000,
                                     min(time.monotonic() + budget_per_init * 0.7, deadline - 110),
                                     tracker, tau_start=10.0, tau_end=50.0, lr=0.04)
            # CD
            if time.monotonic() < deadline - 110:
                init_a, Jd = coordinate_descent_pass(init_a, rng)
                init_a /= init_a.sum()
                tracker.update(init_a)

        tracker.report(f"Phase 7 Restarts: {time.monotonic() - t0:.1f}s")

    # === Phase 8: Final intensive polish ===
    if time.monotonic() < deadline - 30:
        print(f"\n{'=' * 50} Phase 8: Final Polish {'=' * 50}", flush=True)
        t0 = time.monotonic()

        a_final = tracker.best.copy().astype(np.float64)
        a_final = np.maximum(a_final, 1e-8)
        a_final /= a_final.sum()

        # LP polish
        if time.monotonic() < deadline - 50:
            a_final = lp_mirror_descent(a_final, 50, deadline - 50, tracker, tighten_base=0.002)

        # Adam polish with high tau
        if time.monotonic() < deadline - 30:
            a_final = adam_smooth_max(a_final, 10000, deadline - 25, tracker,
                                      tau_start=50.0, tau_end=120.0, lr=0.02)

        # Final CD
        if time.monotonic() < deadline - 10:
            a_final = np.maximum(a_final.astype(np.float64), 1e-8)
            a_final /= a_final.sum()
            for p in range(5):
                if time.monotonic() > deadline - 8:
                    break
                a_final, Jf = coordinate_descent_pass(a_final, rng)
                a_final /= a_final.sum()
                tracker.update(a_final)

        tracker.report(f"Phase 8 Final: {time.monotonic() - t0:.1f}s")

    # Fallback
    if tracker.best is None:
        tracker.best = np.ones(256)
        tracker.best_score = compute_J(tracker.best)

    print(f"\n{'=' * 50} FINAL RESULT {'=' * 50}", flush=True)
    print(f"J = {tracker.best_score:.10f}, n = {tracker.best.size}", flush=True)
    print(f"Total time: {time.monotonic() - tracker.t_start:.1f}s", flush=True)

    return [float(x) for x in tracker.best]
