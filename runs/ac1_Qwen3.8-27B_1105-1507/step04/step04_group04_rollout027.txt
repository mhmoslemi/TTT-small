def compute_conv(a):

def build_toeplitz(a):
    """Build Toeplitz matrix A where (A@g)[k] = (a*g)[k] for k=0..2n-2."""
    n = len(a)

    for j in range(n):
        k_start = j
        k_end = min(j + n, 2 * n - 1)
        A[k_start:k_end, j] = a[:k_end - k_start]
    return A

    # cross_corr = ifft(fft(w, size) * conj(fft(a, size))) ... hmm

    if n <= 2000:

def alternating_lp_v2(a, max_iters=100, time_budget=60.0):
    """Improved alternating LP with adaptive step and faster convergence."""

        M = float(np.max(conv))

def subgradient_optimize_v2(a, max_iters=10000, time_budget=30.0, tau=10.0):

            grad[i] = 2.0 * a[k - i]
        
        ada_grad = beta * ada_grad + (1 - beta) * grad**2
        adaptive_step = ada_lr / (np.sqrt(ada_grad) + 1e-8)
        adaptive_step = np.minimum(adaptive_step, 1.0)
        
        # Use the adaptive step instead
        # Actually, let's stick with the fixed step for now and just track stall
        obj = compute_obj(a_new)
        if obj < best_obj - 1e-14:

            best_a = a_new.copy()

            stall = 0
        else:

                step *= 0.8

                rng = np.random.default_rng(it * 31 + 7)

def coordinate_descent(a, max_iters=5000, time_budget=30.0):
    """Coordinate descent: optimize one variable at a time."""
    n = len(a)
    a = a.copy().astype(np.float64)
    s = np.sum(a)
    if s < 1e-10:
        return a
    a = a / s
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    rng = np.random.default_rng(42)
    
    # Precompute the Toeplitz structure for fast evaluation
    # When we change a[i] by delta, the change in (a*a)[k] is:
    # delta * (a[k-i] + a[i+k] - a[i] if k-i==i or i+k==i)
    # More precisely: (a*a)[k] = sum_j a[j]*a[k-j]
    # d/d(a[i]) (a*a)[k] = a[k-i] + a[i-... 
    # Actually d/d(a[i]) [sum_j a[j]*a[k-j]] = a[k-i] + (if i != k-i) a[i] * ... 
    # No wait: sum_j a[j]*a[k-j]. The terms involving a[i] are:
    # j=i: a[i]*a[k-i]
    # k-j=i => j=k-i: a[k-i]*a[i]
    # So derivative = a[k-i] + a[k-i] = 2*a[k-i] if i != k-i, or a[i] if i=k-i
    
    for it in range(max_iters):
        if time.monotonic() - start > time_budget:
            break
        if it % 1000 == 0 and it > 0:
            print(f"    CD iter {it}: obj={best_obj:.6f}")
        
        i = it % n  # or random
        
        # Try a few values for a[i]
        current_val = a[i]
        best_new_val = current_val
        best_new_obj = best_obj
        
        # Try scaling up and down
        for factor in [0.5, 0.7, 0.9, 1.1, 1.3, 1.5]:
            new_val = current_val * factor
            a[i] = new_val
            s_new = np.sum(a)
            if s_new < 1e-10:
                a[i] = current_val
                continue
            a_norm = a / s_new
            obj_new = compute_obj(a_norm)
            if obj_new < best_new_obj:
                best_new_obj = obj_new
                best_new_val = new_val
        a[i] = current_val  # restore
        
        if best_new_val != current_val:
            a[i] = best_new_val
            s = np.sum(a)
            if s > 1e-10:
                a = a / s
            if best_new_obj < best_obj - 1e-14:
                best_obj = best_new_obj
                best_a = a.copy()
    
    return best_a

def random_init(n, seed):
    """Generate diverse random initializations."""
    rng = np.random.default_rng(seed)
    k = np.arange(n)
    choice = rng.integers(0, 8)

    if choice == 0:
        f = rng.uniform(0.5, 4.0)
        phase = rng.uniform(0, 2 * np.pi)
        a = 1.0 + 0.8 * np.cos(2 * np.pi * f * k / n + phase)
    elif choice == 1:
        # U-shape
        center = n / 2
        a = 1.0 + 0.8 * ((k - center) / (n / 2)) ** 2
    elif choice == 2:
        # Two bumps
        a = (np.exp(-((k - n*0.25)/(n*0.1))**2) + np.exp(-((k - n*0.75)/(n*0.1))**2) + 0.3)
    elif choice == 3:
        f1 = rng.uniform(0.3, 2.0)
        f2 = rng.uniform(2.0, 6.0)
        a = 1.0 + 0.4 * np.cos(2 * np.pi * f1 * k / n) + 0.3 * np.cos(2 * np.pi * f2 * k / n)
    elif choice == 4:
        a = rng.random(n)
        win = max(3, n // 20)
        kernel = np.ones(win) / win
        a = np.convolve(a, kernel, mode='same')
    elif choice == 5:
        # Ramp up then down
        center = n * rng.uniform(0.3, 0.7)
        width = n * rng.uniform(0.2, 0.5)
        a = np.exp(-0.5 * ((k - center) / width) ** 2) + 0.2
    elif choice == 6:
        # Sine modulated
        f = rng.uniform(1, 5)
        a = 1.0 + 0.5 * np.sin(2 * np.pi * f * k / n) ** 2
    else:
        a = 1.0 + 0.3 * rng.random(n)

    a = np.maximum(a, 0)
    s = np.sum(a)
    if s < 1e-10:
        a = np.ones(n)
        s = n
    return a / s


def resample_sequence(a, new_n):
    """Resample a sequence to a new length using interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    old_k = np.arange(n)
    new_k = np.linspace(0, n - 1, new_n)
    a_new = np.interp(new_k, old_k, a)
    a_new = np.maximum(a_new, 0)
    s = np.sum(a_new)
    if s > 1e-10:
        a_new = a_new / s
    return a_new

import numpy as np
from scipy.optimize import linprog

def compute_conv(a):
    """Compute conv(a,a) using FFT. Returns array of length 2n-1."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    conv = np.fft.irfft(fa * fa, size)
    return conv[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    conv = compute_conv(a)
    max_b = float(np.max(conv))
    sum_a = float(np.sum(a))
    if sum_a < 0.01:
        return np.inf
    return 2.0 * n * max_b / (sum_a * sum_a)


def build_toeplitz(a):
    """Build Toeplitz matrix A where (A@g)[k] = (a*g)[k] for k=0..2n-2."""
    n = len(a)
    A = np.zeros((2 * n - 1, n), dtype=np.float64)
    for j in range(n):
        k_start = j
        k_end = min(j + n, 2 * n - 1)
        A[k_start:k_end, j] = a[:k_end - k_start]
    return A


def solve_lp_maxsum(a, M, time_limit=5):
    """Solve LP: max sum(g) s.t. (a*g)[k] <= M for all k, g >= 0."""
    n = len(a)
    A = build_toeplitz(a)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': time_limit})
        if res.status == 0:
            g = np.maximum(res.x, 0)
            return g, float(np.sum(g))
    except Exception:
        pass
    return None, None


def compute_smoothed_grad(a, weights):

        M[valid] = weights[k[valid]]

    else:

def smoothed_subgradient_step(a, step, tau=10.0):
    """Subgradient step using softmax-weighted lags of the convolution."""
    n = len(a)
    conv = compute_conv(a)
    max_conv = np.max(conv)
    shifted = tau * (conv - max_conv)
    shifted = np.clip(shifted, -100, 100)
    weights = np.exp(shifted)
    w_sum = np.sum(weights)
    if w_sum < 1e-300:
        weights = np.zeros(len(conv))
        weights[int(np.argmax(conv))] = 1.0
    else:
        weights /= w_sum

    grad = compute_smoothed_grad(a, weights)
    grad *= 2.0

    log_update = -step * grad
    log_update = np.clip(log_update, -50, 50)
    a_new = a * np.exp(log_update)
    a_new = np.maximum(a_new, 1e-15)
    s = np.sum(a_new)
    if s > 1e-10:
        a_new = a_new / s
    return a_new

def alternating_lp(a, max_iters=100, time_budget=60.0):
    """Alternating LP with adaptive step and convergence detection."""
    n = len(a)
    a = a.copy().astype(np.float64)
    s = np.sum(a)

        return a
    a = a / s

    stall = 0

        conv = compute_conv(a)
        M = float(np.max(conv))

        g, sum_g = solve_lp_maxsum(a, M, time_limit=4)
        if g is None or sum_g < 1e-10:

        g_norm = g / sum_g

        best_t = 0.0
        best_obj_ls = best_obj
        for t in [0.5, 0.7, 0.85, 1.0]:
            a_new = (1.0 - t) * a + t * g_norm
            a_new = np.maximum(a_new, 0)
            s_new = np.sum(a_new)
            if s_new < 1e-10:
                continue
            a_new = a_new / s_new
            obj_new = compute_obj(a_new)
            if obj_new < best_obj_ls:
                best_obj_ls = obj_new
                best_t = t

        if best_t > 0 and best_obj_ls < best_obj - 1e-14:
            a_new = (1.0 - best_t) * a + best_t * g_norm
            a_new = np.maximum(a_new, 0)
            s_new = np.sum(a_new)
            if s_new > 1e-10:
                a_new = a_new / s_new
            best_obj = best_obj_ls
            best_a = a_new.copy()
            a = a_new
            stall = 0
        else:
            stall += 1
            if stall > 40:
                break
            a = 0.95 * a + 0.05 * g_norm
            a = np.maximum(a, 0)
            s = np.sum(a)
            if s > 1e-10:
                a = a / s

    return best_a

                    continue
                a_test = a / s_new
                obj_test = compute_obj(a_test)
                if obj_test < best_local_obj:
                    best_local_obj = obj_test
                    best_val = current_val * factor
            a[i] = current_val
            if best_val != current_val:
                a[i] = best_val
                s = np.sum(a)
                if s > 1e-10:
                    a = a / s
                if best_local_obj < best_obj:
                    best_obj = best_local_obj
                    best_a = a.copy()
                    improved = True
        if not improved:
            break
    return best_a


def try_parent():
    """Try to load parent sequence."""
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64)
        parent = np.clip(parent, 0, 1000.0)
        if parent.size > 0 and np.isfinite(parent).all():
            obj = compute_obj(parent)
            print(f"Parent (n={parent.size}): obj={obj:.6f}")
            return parent.copy(), obj
    except Exception:
        pass
    return None, np.inf


def random_init(n, seed):
    """Generate diverse random initializations."""
    rng = np.random.default_rng(seed)
    k = np.arange(n)
    choice = rng.integers(0, 10)

    if choice == 0:
        f = rng.uniform(0.5, 4.0)
        phase = rng.uniform(0, 2 * np.pi)
        a = 1.0 + 0.8 * np.cos(2 * np.pi * f * k / n + phase)
    elif choice == 1:
        center = n / 2
        a = 1.0 + 0.8 * ((k - center) / (n / 2)) ** 2
    elif choice == 2:
        a = (np.exp(-((k - n * 0.25) / (n * 0.1)) ** 2) +
             np.exp(-((k - n * 0.75) / (n * 0.1)) ** 2) + 0.3)
    elif choice == 3:
        f1 = rng.uniform(0.3, 2.0)
        f2 = rng.uniform(2.0, 6.0)
        a = 1.0 + 0.4 * np.cos(2 * np.pi * f1 * k / n) + 0.3 * np.cos(2 * np.pi * f2 * k / n)
    elif choice == 4:
        a = rng.random(n)
        win = max(3, n // 20)
        kernel = np.ones(win) / win
        a = np.convolve(a, kernel, mode='same')
    elif choice == 5:
        center = n * rng.uniform(0.3, 0.7)
        width = n * rng.uniform(0.2, 0.5)
        a = np.exp(-0.5 * ((k - center) / width) ** 2) + 0.2
    elif choice == 6:
        f = rng.uniform(1, 5)
        a = 1.0 + 0.5 * np.sin(2 * np.pi * f * k / n) ** 2
    elif choice == 7:
        f0 = rng.uniform(0.3, 1.0)
        f1 = rng.uniform(3.0, 8.0)
        t = k / n
        f = f0 + (f1 - f0) * t
        a = 1.0 + 0.5 * np.cos(2 * np.pi * np.cumsum(f))
    elif choice == 8:
        a = np.ones(n)
        for _ in range(2):
            f = rng.uniform(0.5, 5.0)
            a += rng.uniform(0.1, 0.4) * np.cos(2 * np.pi * f * k / n + rng.uniform(0, 2 * np.pi))
    else:
        a = 1.0 + 0.3 * rng.random(n)

    a = np.maximum(a, 0)
    s = np.sum(a)
    if s < 1e-10:
        a = np.ones(n)
        s = n
    return a / s


def resample_sequence(a, new_n):
    """Resample a sequence to a new length using interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    old_k = np.arange(n)
    new_k = np.linspace(0, n - 1, new_n)
    a_new = np.interp(new_k, old_k, a)
    a_new = np.maximum(a_new, 0)
    s = np.sum(a_new)
    if s > 1e-10:
        a_new = a_new / s
    return a_new


def optimize_pipeline(a, time_budget=60.0, verbose=False):
    """Run full optimization pipeline on a single sequence."""
    n = len(a)
    a = a.copy().astype(np.float64)
    s = np.sum(a)
    if s < 1e-10:
        a = np.ones(n)
        s = n
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    if verbose:
        print(f"  Pipeline start: obj={best_obj:.8f}")

    # Phase 1: Subgradient (quick initial progress)
    remaining = time_budget - (time.monotonic() - start)
    if remaining > 8:
        sg_time = min(8.0, remaining * 0.12)
        a_new = subgradient_optimize(best_a, max_iters=int(sg_time * 50),
                                      time_budget=sg_time, tau=10.0, step=0.08)
        obj = compute_obj(a_new)
        if obj < best_obj:
            best_obj = obj
            best_a = a_new.copy()
            if verbose:
                print(f"  After SG1: {best_obj:.8f}")

    # Phase 2: Alternating LP (main workhorse)
    remaining = time_budget - (time.monotonic() - start)
    if remaining > 15:
        lp_time = min(25.0, remaining * 0.35)
        a_new = alternating_lp(best_a, max_iters=80, time_budget=lp_time)
        obj = compute_obj(a_new)
        if obj < best_obj:
            best_obj = obj
            best_a = a_new.copy()
            if verbose:
                print(f"  After LP1: {best_obj:.8f}")

    # Phase 3: Subgradient with higher tau (fine-tuning)
    remaining = time_budget - (time.monotonic() - start)
    if remaining > 8:
        sg_time = min(8.0, remaining * 0.15)
        a_new = subgradient_optimize(best_a, max_iters=int(sg_time * 40),
                                      time_budget=sg_time, tau=40.0, step=0.04)
        obj = compute_obj(a_new)
        if obj < best_obj:
            best_obj = obj
            best_a = a_new.copy()
            if verbose:
                print(f"  After SG2: {best_obj:.8f}")

    # Phase 4: Another LP round
    remaining = time_budget - (time.monotonic() - start)
    if remaining > 12:
        lp_time = min(18.0, remaining * 0.3)
        a_new = alternating_lp(best_a, max_iters=50, time_budget=lp_time)
        obj = compute_obj(a_new)
        if obj < best_obj:
            best_obj = obj
            best_a = a_new.copy()
            if verbose:
                print(f"  After LP2: {best_obj:.8f}")

    # Phase 5: Coordinate search
    remaining = time_budget - (time.monotonic() - start)
    if remaining > 10:
        cd_time = min(10.0, remaining * 0.2)
        a_new = coordinate_search(best_a, time_budget=cd_time)
        obj = compute_obj(a_new)
        if obj < best_obj:
            best_obj = obj
            best_a = a_new.copy()
            if verbose:
                print(f"  After CD: {best_obj:.8f}")

    # Phase 6: Final LP polish
    remaining = time_budget - (time.monotonic() - start)
    if remaining > 12:
        lp_time = min(15.0, remaining * 0.3)
        a_new = alternating_lp(best_a, max_iters=40, time_budget=lp_time)
        obj = compute_obj(a_new)
        if obj < best_obj:
            best_obj = obj
            best_a = a_new.copy()
            if verbose:
                print(f"  After LP3: {best_obj:.8f}")

    return best_a, best_obj


def local_search(a, time_budget=60.0, verbose=False):
    """Local search with diverse perturbations and re-optimization."""
    n = len(a)
    a = a.copy().astype(np.float64)
    s = np.sum(a)
    if s < 1e-10:
        return a, compute_obj(a)
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    attempt = 0
    while time.monotonic() - start < time_budget - 5:
        attempt += 1
        remaining = time_budget - (time.monotonic() - start)
        if remaining < 10:
            break

        rng = np.random.default_rng(attempt * 7919 + 1)

        # Vary the perturbation scale
        scale = 0.01 + 0.06 * (attempt % 5) / 5.0
        # Sometimes use a different type of perturbation
        if attempt % 3 == 0:
            # Sparse perturbation
            n_pert = max(1, n // 10)
            idx = rng.choice(n, size=n_pert, replace=False)
            a_pert = best_a.copy()
            a_pert[idx] *= np.exp(rng.normal(0, 0.2, n_pert))
        elif attempt % 3 == 1:
            # Dense small perturbation
            a_pert = best_a * np.exp(rng.normal(0, scale, n))
        else:
            # Medium perturbation
            a_pert = best_a * np.exp(rng.normal(0, scale * 1.5, n))

        a_pert = np.maximum(a_pert, 1e-10)
        s_pert = np.sum(a_pert)
        if s_pert < 1e-10:
            continue
        a_pert = a_pert / s_pert

        sub_time = min(10.0, remaining * 0.3)
        a_opt, obj = optimize_pipeline(a_pert, time_budget=sub_time, verbose=False)

        if obj < best_obj:
            best_obj = obj
            best_a = a_opt.copy()
            if verbose:
                print(f"    LS attempt {attempt}: {best_obj:.8f}")

    return best_a, best_obj
