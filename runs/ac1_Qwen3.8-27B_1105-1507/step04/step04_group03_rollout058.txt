def fast_conv(a):
    """Full self-convolution via FFT."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]


def compute_J(a):
    """Compute J(a) = 2*n*max(conv(a,a))/(sum(a))^2."""
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))

def fast_conv(a):

    size = 1
    while size < 2 * n - 1:

    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]

    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))

def build_A_sparse(n, a):
    """Build sparse matrix A where A[k,j] = a[k-j] for valid j (a used as kernel)."""
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        j_min = max(0, k - n + 1)
        j_max = min(n - 1, k)
        for j in range(j_min, j_max + 1):
            rows.append(k)
            cols.append(j)
            vals.append(a[k - j])
    return csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))


def lp_best_response(a, M):
    """Solve LP: max sum(g) s.t. A(a) @ g <= M, g >= 0.
    A(a) is the convolution matrix: (A(a) @ g)[k] = sum_j a[k-j]*g[j] = (a*g)[k]
    """
    n = len(a)
    A = build_A_sparse(n, a)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    try:
        r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                    options={'presolve': True})
        if r.success and r.x is not None:
            g = np.maximum(r.x, 0)
            if g.sum() > 1e-15:
                return g
    except Exception:
        pass
    return None

            a_trial = np.maximum(a_trial, 0.0)

            if s < 1e-10:

                continue
            a_trial = a_trial / s

                best_a = a.copy()

def fast_conv(a):
    """Full self-convolution via FFT."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]


def compute_J(a):
    """Compute J(a) = 2*n*max(conv(a,a))/(sum(a))^2."""
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))


def cd_pass(a, conv, order):
    """One pass of coordinate descent. Updates a and conv in place partially.
    Returns (a_new, conv_new)."""
    n = len(a)
    a = a.copy()
    conv = conv.copy()

    for i in order:
        old_val = a[i]

        # Affected k range: [i, i+n-1] clipped to [0, 2n-2]
        k_min = i
        k_max = min(2 * n - 2, i + n - 1)

        valid_k = np.arange(k_min, k_max + 1)
        valid_j = valid_k - i  # these are in [0, n-1] by construction

        # Contribution of a_i to conv
        is_diag = (valid_k == 2 * i)
        contrib = np.empty(len(valid_k))
        nd_idx = ~is_diag
        contrib[is_diag] = old_val * old_val
        contrib[nd_idx] = 2.0 * old_val * a[valid_j[nd_idx]]

        # Reduced convolution (a_i set to 0)
        conv_red = conv - np.zeros(2 * n - 1)
        conv_red[valid_k] -= contrib

        # Bisection for optimal t = min max_k (conv_red[k] + contrib_k(x))
        t0_base = np.max(conv_red)
        has_quad = (2 * i <= 2 * n - 2)
        quad_offset = conv_red[2 * i] if has_quad else 0.0

        # Linear terms for non-diagonal affected k
        nd_k = valid_k[nd_idx]
        nd_j = valid_j[nd_idx]
        ld_coeff = 2.0 * a[nd_j]
        ld_const = conv_red[nd_k]

        nz = ld_coeff > 1e-18
        ld_coeff = ld_coeff[nz]
        ld_const = ld_const[nz]

        t_lo = t0_base
        t_hi = np.max(conv)

        if t_hi <= t_lo + 1e-18:
            x_opt = old_val
        else:
            # Bisection: ~45 iterations gives ~1e-13 relative precision
            for _ in range(45):
                t_mid = 0.5 * (t_lo + t_hi)

                if has_quad:
                    x_bq = np.sqrt(max(0.0, t_mid - quad_offset))
                else:
                    x_bq = np.inf

                if len(ld_coeff) > 0:
                    x_bl = np.min((t_mid - ld_const) / ld_coeff)
                else:
                    x_bl = np.inf

                if min(x_bq, x_bl) >= -1e-12:
                    t_hi = t_mid
                else:
                    t_lo = t_mid

            t_opt = t_hi
            if has_quad:
                x_bq = np.sqrt(max(0.0, t_opt - quad_offset))
            else:
                x_bq = np.inf
            if len(ld_coeff) > 0:
                x_bl = np.min((t_opt - ld_const) / ld_coeff)
            else:
                x_bl = np.inf
            x_opt = min(x_bq, x_bl)
            x_opt = max(0.0, x_opt)

        # Update a and conv
        a[i] = x_opt
        conv = conv_red
        contrib_new = np.empty(len(valid_k))
        contrib_new[is_diag] = x_opt * x_opt
        contrib_new[nd_idx] = 2.0 * x_opt * a[valid_j[nd_idx]]
        conv[valid_k] += contrib_new

    return a, conv


def build_A_sparse(n, a):
    """Build sparse matrix A where (A@g)[k] = (a*g)[k] for self-conv structure."""
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        j_min = max(0, k - n + 1)
        j_max = min(n - 1, k)
        for j in range(j_min, j_max + 1):
            rows.append(k)
            cols.append(j)
            vals.append(a[k - j])
    return csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))


def lp_best_response(a, M):
    """Solve LP: max sum(g) s.t. (a*g)[k] <= M for all k, g >= 0."""
    n = len(a)
    A = build_A_sparse(n, a)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    try:
        r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                    options={'presolve': True})
        if r.success and r.x is not None:
            g = np.maximum(r.x, 0)
            if g.sum() > 1e-15:
                return g
    except Exception:
        pass
    return None


def lp_mirror_descent(a, max_iters, deadline):
    """LP mirror descent with adaptive line search."""
    a = np.maximum(a.astype(np.float64), 1e-10)
    a = a / a.sum()
    current_J = compute_J(a)
    best_J = current_J
    best_a = a.copy()
    no_improve = 0

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        b = fast_conv(a)
        M = float(b.max())
        g = lp_best_response(a, M)
        if g is None:
            break
        g = np.maximum(g, 0.0)
        g_sum = g.sum()
        if g_sum < 1e-10:
            break
        g = g / g_sum

        base_t = 0.35 * np.exp(-0.15 * it / 50.0) + 0.05
        t = min(1.0, base_t)

        best_trial_J = current_J
        best_trial_a = None

        for ls in range(20):
            a_trial = (1.0 - t) * a + t * g
            a_trial = np.maximum(a_trial, 0.0)
            s = a_trial.sum()
            if s < 1e-10:
                t *= 0.5
                continue
            a_trial = a_trial / s
            J_trial = compute_J(a_trial)
            if J_trial < best_trial_J - 1e-14:
                best_trial_J = J_trial
                best_trial_a = a_trial.copy()
                break
            t *= 0.6
            if t < 0.001:
                break

        if best_trial_a is not None:
            a = best_trial_a
            current_J = best_trial_J
            if current_J < best_J:
                best_J = current_J
                best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1
            if no_improve > 35:
                break

    return best_a, best_J


def resample_seq(src, dest_n):
    """Resample 1D sequence to new length."""
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)


class ScoreTracker:
    def __init__(self):
        self.best = None
        self.best_score = float('inf')
        self.t_start = time.monotonic()

    def update(self, a):
        a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
        if a.size == 0 or a.sum() < 0.01:
            return
        score = float(evaluate_sequence(a.tolist()))
        if np.isfinite(score) and score < self.best_score:
            self.best = a.copy()
            self.best_score = score
            elapsed = time.monotonic() - self.t_start
            print(f"  [NEW BEST] J={score:.8f} n={a.size} t={elapsed:.1f}s", flush=True)

    def report(self):
        if self.best is not None:
            print(f"  best: J={self.best_score:.8f} n={self.best.size}", flush=True)


def propose_candidate(seed=42, budget_s=1500, **kwargs):
    """
    Hybrid optimizer: coordinate descent (primary) + LP mirror descent (secondary).
    CD solves 1D convex subproblems exactly via bisection.
    Multi-length exploration and restarts for global improvement.
    """
    rng = np.random.default_rng(seed)
    start_time = time.monotonic()
    deadline = start_time + float(budget_s) - 20.0
    tracker = ScoreTracker()

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        if p.ndim == 1 and p.size > 0 and np.all(np.isfinite(p)):
            parent = np.maximum(p, 0.0)
            tracker.update(parent)
            print(f"Parent: n={parent.size}, J={tracker.best_score:.8f}", flush=True)
    except (NameError, AttributeError):
        pass

    if parent is None:
        parent = np.ones(512) / 512
        tracker.update(parent)

    # === Phase 1: Coordinate descent on parent ===
    print(f"\n{'='*60}", flush=True)
    print(f"Phase 1: Coordinate descent (n={parent.size})", flush=True)
    print(f"{'='*60}", flush=True)

    a = parent.copy()
    a = np.maximum(a, 1e-10)
    conv = fast_conv(a)
    current_max = conv.max()
    print(f"  Initial: max_conv={current_max:.10f}, sum={a.sum():.6f}", flush=True)

    cd_deadline = min(deadline - 300, start_time + 700)
    pass_count = 0
    best_cd_max = current_max
    best_cd_a = a.copy()
    stagnation_count = 0

    while time.monotonic() < cd_deadline:
        order = rng.permutation(len(a))
        a, conv = cd_pass(a, conv, order)
        pass_count += 1
        current_max = conv.max()

        if current_max < best_cd_max - 1e-12 * best_cd_max:
            improvement = (best_cd_max - current_max) / best_cd_max
            best_cd_max = current_max
            best_cd_a = a.copy()
            stagnation_count = 0
            if pass_count <= 10 or pass_count % 100 == 1 or improvement > 0.002:
                J_val = compute_J(a)
                print(f"  Pass {pass_count}: max_conv={current_max:.10f}, J={J_val:.8f}, "
                      f"improvement={improvement*100:.4f}%", flush=True)
                tracker.update(a)
        else:
            stagnation_count += 1

        # Periodic reporting
        if pass_count % 200 == 0:
            elapsed = time.monotonic() - start_time
            J_now = compute_J(a)
            print(f"  Pass {pass_count}: max_conv={current_max:.10f}, J={J_now:.8f}, "
                  f"elapsed={elapsed:.0f}s", flush=True)

        # If stagnating, apply a small perturbation
        if stagnation_count > 300 and pass_count % 500 == 0:
            noise = rng.normal(0, 0.01 * a.mean(), len(a))
            a = np.maximum(a + noise, 1e-10)
            conv = fast_conv(a)
            current_max = conv.max()
            if current_max < best_cd_max:
                best_cd_max = current_max
                best_cd_a = a.copy()
            print(f"  [RESTART] Pass {pass_count}: perturbed, max_conv={current_max:.10f}",
                  flush=True)

    tracker.update(best_cd_a)
    J_cd = compute_J(best_cd_a)
    print(f"  CD done: {pass_count} passes, best J={J_cd:.8f}", flush=True)

    # === Phase 2: LP mirror descent from CD result ===
    if time.monotonic() < deadline - 80:
        print(f"\n{'='*60}", flush=True)
        print(f"Phase 2: LP mirror descent (n={best_cd_a.size})", flush=True)
        print(f"{'='*60}", flush=True)

        lp_deadline = min(deadline - 40, time.monotonic() + 250)
        a_lp, J_lp = lp_mirror_descent(best_cd_a, 500, lp_deadline)
        tracker.update(a_lp)
        print(f"  LP result: J={J_lp:.8f}", flush=True)

    # === Phase 3: CD from LP result (or best so far) ===
    if time.monotonic() < deadline - 40:
        print(f"\n{'='*60}", flush=True)
        print(f"Phase 3: CD refinement from best", flush=True)
        print(f"{'='*60}", flush=True)

        a3 = np.maximum(tracker.best, 1e-10).copy()
        conv3 = fast_conv(a3)
        cd3_deadline = min(deadline - 20, time.monotonic() + 200)
        passes3 = 0

        while time.monotonic() < cd3_deadline:
            order = rng.permutation(len(a3))
            a3, conv3 = cd_pass(a3, conv3, order)
            passes3 += 1
            if passes3 % 50 == 0:
                J3 = compute_J(a3)
                print(f"  CD3 pass {passes3}: J={J3:.8f}", flush=True)
                tracker.update(a3)

        tracker.update(a3)
        print(f"  CD3 done: {passes3} passes, best J={tracker.best_score:.8f}", flush=True)

    # === Phase 4: Multi-length exploration ===
    if time.monotonic() < deadline - 30:
        print(f"\n{'='*60}", flush=True)
        print(f"Phase 4: Multi-length exploration", flush=True)
        print(f"{'='*60}", flush=True)

        base = tracker.best
        base_n = base.size
        # Try nearby and somewhat different lengths
        candidate_lengths = sorted(set([
            max(128, base_n - 256),
            max(128, base_n - 128),
            base_n + 128,
            base_n + 256,
            base_n + 512,
        ]))
        candidate_lengths = [n for n in candidate_lengths if 128 <= n <= 4096 and n != base_n]

        per_len_time = max(20, (deadline - 25 - time.monotonic()) / max(1, len(candidate_lengths)))

        for n_try in candidate_lengths:
            if time.monotonic() > deadline - 25:
                break
            a_try = resample_seq(base, n_try)
            a_try = np.maximum(a_try, 1e-10)
            conv_try = fast_conv(a_try)

            t0 = time.monotonic()
            passes = 0
            while time.monotonic() < t0 + per_len_time * 0.8:
                order = rng.permutation(n_try)
                a_try, conv_try = cd_pass(a_try, conv_try, order)
                passes += 1

            J_try = compute_J(a_try)
            tracker.update(a_try)
            print(f"  n={n_try}: J={J_try:.8f} ({passes} passes, "
                  f"{time.monotonic()-t0:.0f}s)", flush=True)

    # === Phase 5: Final CD polish on the best ===
    if time.monotonic() < deadline - 5 and tracker.best is not None:
        print(f"\n{'='*60}", flush=True)
        print(f"Phase 5: Final CD polish", flush=True)
        print(f"{'='*60}", flush=True)

        a5 = np.maximum(tracker.best, 1e-10).copy()
        conv5 = fast_conv(a5)
        dl5 = deadline - 3
        passes5 = 0
        while time.monotonic() < dl5:
            order = rng.permutation(len(a5))
            a5, conv5 = cd_pass(a5, conv5, order)
            passes5 += 1
            if passes5 % 100 == 0:
                J5 = compute_J(a5)
                print(f"  Final pass {passes5}: J={J5:.8f}", flush=True)
                tracker.update(a5)

        tracker.update(a5)
        print(f"  Final: {passes5} passes, J={tracker.best_score:.8f}", flush=True)

    # Final output
    if tracker.best is not None:
        final_a = tracker.best
    else:
        final_a = parent

    print(f"\n{'='*60}", flush=True)
    print(f"FINAL: J={tracker.best_score:.8f} n={final_a.size} "
          f"time={time.monotonic()-start_time:.0f}s", flush=True)
    print(f"{'='*60}", flush=True)

    return [float(x) for x in final_a]
