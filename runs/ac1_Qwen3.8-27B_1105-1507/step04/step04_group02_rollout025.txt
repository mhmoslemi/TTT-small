with L-BFGS optimization in softmax parameterization, multi-scale t-annealing,

import numpy as np
from scipy.optimize import minimize, linprog
from scipy.fft import rfft, irfft
import time

def compute_conv(a):
    """Compute linear convolution a*a via FFT."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    conv = irfft(fa * fa, size)
    return conv[:2 * n - 1]

def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    S = float(np.sum(a))
    if S < 1e-10:
        return np.inf
    conv = compute_conv(a)
    M = float(np.max(conv))
    return 2.0 * n * M / (S * S)

def softmax_obj_and_grad(x, t):
    """
    Compute the log-sum-exp approximated objective and its gradient.
    x: unconstrained parameter vector (length n)
    a = softmax(x) (so sum(a) = 1, a > 0)
    Objective: (2n/t) * log(sum_k exp(t * (a*a)[k]))
    """
    n = len(x)
    # Compute a = softmax(x)
    x_shift = x - np.max(x)
    a = np.exp(x_shift)
    S = a.sum()
    a = a / S
    
    # Compute convolution
    conv = compute_conv(a)
    
    # Log-sum-exp of t * conv
    tconv = t * conv
    tconv_max = tconv.max()
    log_sum_exp = tconv_max + np.log(np.sum(np.exp(tconv - tconv_max)))
    obj = (2.0 * n / t) * log_sum_exp
    
    # Softmax weights over convolution positions
    exp_tconv = np.exp(tconv - tconv_max)
    w = exp_tconv / exp_tconv.sum()
    
    # Gradient of obj w.r.t. a[i]:
    # d(obj)/da[i] = (2n) * sum_k w[k] * d(conv[k])/da[i]
    # d(conv[k])/da[i] = 2*a[k-i] if 0 <= k-i < n
    # So d(obj)/da[i] = 4n * sum_k w[k] * a[k-i] * I(0<=k-i<n)
    # = 4n * sum_j w[i+j] * a[j] (where j = k-i, so k = i+j)
    
    # Compute grad_a[i] = sum_j w[i+j] * a[j] for valid j
    grad_a = np.zeros(n)
    for i in range(n):
        # j ranges from 0 to n-1, and i+j must be < 2n-1 (always true since i<n, j<n, so i+j < 2n-1)
        # w[i+j] for j=0,...,n-1 gives w[i], w[i+1], ..., w[i+n-1]
        grad_a[i] = np.dot(w[i:i+n], a)
    
    d_obj_da = 4.0 * n * grad_a
    
    # Chain rule through softmax: a = exp(x)/sum(exp(x))
    # da[i]/dx[j] = a[i]*(delta[i,j] - a[j])
    # d(obj)/dx[j] = sum_i d(obj)/da[i] * da[i]/dx[j]
    # = sum_i d_obj_da[i] * a[i] * (delta[i,j] - a[j])
    # = d_obj_da[j]*a[j] - a[j]*sum_i d_obj_da[i]*a[i]
    
    dot = np.dot(d_obj_da, a)
    grad_x = d_obj_da * a - a * dot
    
    return obj, grad_x

    """Optimize using L-BFGS with log-sum-exp objective."""
    a = a_init.copy()
    S = a.sum()
    if S < 1e-10:
        return a_init
    a = a / S
    # Convert to log-space (with small offset for numerical stability)
    a_safe = np.maximum(a, 1e-15)
    x = np.log(a_safe)
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    
    for t in t_schedule:
        if time.monotonic() - start > time_budget:
            break
        
        def func_and_grad(x_curr):
            return softmax_obj_and_grad(x_curr, t)
        
        try:
            res = minimize(func_and_grad, x, method='L-BFGS-B', 
                         options={'maxiter': 500, 'gtol': 1e-8, 'maxls': 50})
            # Convert back
            x_new = res.x
            x_shift = x_new - np.max(x_new)
            a_new = np.exp(x_shift)
            a_new = a_new / a_new.sum()
            obj_new = compute_obj(a_new)
            if obj_new < best_obj:
                best_obj = obj_new
                best_a = a_new.copy()
                x = x_new.copy()
        except Exception as e:
            print(f"  L-BFGS error at t={t}: {e}")
    
    return best_a

with L-BFGS in softmax parameterization, t-annealing, FFT-based gradients,
clipping-and-filtering (Fienup) refinement, and simplex-projected subgradient.
Multi-n exploration with parent-sequence initialization and random restarts.
"""

import numpy as np
from scipy.optimize import minimize
from scipy.fft import rfft, irfft, fft, ifft
import time


def compute_conv(a):
    """Compute linear convolution a*a via FFT."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    conv = irfft(fa * fa, size)
    return conv[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    S = float(np.sum(a))
    if S < 1e-10:
        return np.inf
    conv = compute_conv(a)
    M = float(np.max(conv))
    return 2.0 * n * M / (S * S)


def compute_grad_a_fft(w, a):
    """Compute grad_a[i] = sum_j w[i+j] * a[j] using FFT cross-correlation."""
    n = len(a)
    a_rev = a[::-1]
    size = 1
    while size < len(w) + n - 1:
        size <<= 1
    conv = irfft(rfft(w, size) * rfft(a_rev, size), size)
    return conv[n - 1:2 * n - 1]


def softmax_obj_and_grad(x, t):
    """
    Log-sum-exp approximated objective and gradient.
    x: unconstrained parameter (length n), a = softmax(x).
    Returns (obj, grad_x).
    """
    n = len(x)
    x_shift = x - np.max(x)
    a = np.exp(x_shift)
    S = a.sum()
    a = a / S

    conv = compute_conv(a)

    tconv = t * conv
    tconv_max = tconv.max()
    log_sum_exp = tconv_max + np.log(np.sum(np.exp(tconv - tconv_max)))
    obj = (2.0 * n / t) * log_sum_exp

    exp_tconv = np.exp(tconv - tconv_max)
    w = exp_tconv / exp_tconv.sum()

    grad_a = compute_grad_a_fft(w, a)
    d_obj_da = 4.0 * n * grad_a

    dot = np.dot(d_obj_da, a)
    grad_x = d_obj_da * a - a * dot

    return obj, grad_x


def project_simplex(v):
    """Project v onto {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.where(u * np.arange(1, n + 1) > (cssv - 1))[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho = rho[-1]
    theta = (cssv[rho] - 1) / (rho + 1)
    return np.maximum(v - theta, 0)


def lbfgs_optimize(a_init, t_schedule, maxiter_per_t=300, time_budget=60.0, seed=0):
    """L-BFGS optimization with log-sum-exp objective and t-annealing."""
    n = len(a_init)
    a = a_init.copy()
    S = a.sum()
    if S < 1e-10:
        return a_init
    a = a / S
    a_safe = np.maximum(a, 1e-15)
    x = np.log(a_safe)

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    for t in t_schedule:
        if time.monotonic() - start > time_budget:
            break
        remaining = time_budget - (time.monotonic() - start)
        if remaining < 2:
            break

        maxit = int(maxiter_per_t * min(1.0, remaining / 30.0))
        if maxit < 10:
            maxit = 10

        def func_wrapper(x_curr):
            return softmax_obj_and_grad(x_curr, t)

        try:
            res = minimize(func_wrapper, x, method='L-BFGS-B',
                          jac=True,
                          options={'maxiter': maxit, 'gtol': 1e-7, 'maxls': 30,
                                   'maxfun': maxit * 5})
            x = res.x
            x_shift = x - np.max(x)
            a_new = np.exp(x_shift)
            a_new = a_new / a_new.sum()
            obj_new = compute_obj(a_new)
            if obj_new < best_obj:
                best_obj = obj_new
                best_a = a_new.copy()
            else:
                # Reheat: add noise
                rng = np.random.default_rng(seed + int(t))
                x = x + rng.normal(0, 0.05, n)
        except Exception:
            pass

    return best_a


def fienup_refine(a_init, n_iters=200, alpha=0.5, time_budget=30.0, seed=0):
    """Clipping-and-filtering (Fienup/Gerchberg-Saxton) PAPR reduction."""
    n = len(a_init)
    a = a_init.copy()
    S = a.sum()
    if S < 1e-10:
        return a_init
    a = a / S

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    size = 1
    while size < 2 * n - 1:
        size <<= 1

    for it in range(n_iters):
        if time.monotonic() - start > time_budget:
            break

        A = fft(a, size)
        power = np.abs(A) ** 2

        conv = compute_conv(a)
        target = np.median(conv)

        # Clip peaks in power spectrum
        power_clipped = np.minimum(power, target * alpha + np.max(power) * (1 - alpha))
        # Ensure DC is preserved
        power_clipped[0] = power[0]

        # Recover A with modified magnitude, same phase
        phase = np.angle(A)
        A_new = np.sqrt(power_clipped) * np.exp(1j * phase)

        a_new = np.real(ifft(A_new, size))[:n]
        a_new = project_simplex(a_new)

        obj_new = compute_obj(a_new)
        if obj_new < best_obj:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new

        # Adaptive alpha
        if it % 20 == 19:
            alpha = max(0.1, alpha * 0.85)

    return best_a


def subgradient_simplex(a_init, max_iters=5000, time_budget=30.0, seed=0):
    """Subgradient method with simplex projection and adaptive step size."""
    n = len(a_init)
    a = a_init.copy()
    S = a.sum()
    if S < 1e-10:
        return a_init
    a = a / S

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    step = 0.1 / np.sqrt(n)
    stall = 0

    for it in range(max_iters):
        if time.monotonic() - start > time_budget:
            break

        conv = compute_conv(a)
        M = np.max(conv)
        # Find active set (positions within 5% of max)
        threshold = M * 0.95
        active = np.where(conv >= threshold)[0]

        # Average subgradient over active set
        grad_a = np.zeros(n)
        for k in active:
            lo = max(0, k - n + 1)
            hi = min(k, n - 1)
            grad_a[lo:hi + 1] += 2.0 * a[k - hi:k - lo + 1][::-1]
        grad_a /= len(active)

        # Center gradient (preserve sum)
        grad_a -= np.mean(grad_a)

        a_new = a - step * grad_a
        a_new = project_simplex(a_new)

        obj_new = compute_obj(a_new)
        if obj_new < best_obj - 1e-10:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new
            step = min(step * 1.02, 5.0 / np.sqrt(n))
            stall = 0
        else:
            stall += 1
            step *= 0.97
            if stall > 500:
                rng = np.random.default_rng(seed + it)
                a = project_simplex(a + rng.normal(0, 0.05, n))
                stall = 0

    return best_a


def interpolate_sequence(a, new_n):
    """Interpolate a sequence to a new length."""
    n = len(a)
    if n == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_interp = np.interp(x_new, x_old, a)
    a_interp = np.maximum(a_interp, 0)
    S = a_interp.sum()
    if S < 1e-10:
        return np.ones(new_n) / new_n
    return a_interp / S

def cosine_init(n, n_modes=10, seed=0):
    """Generate a cosine-series initialization."""
    rng = np.random.default_rng(seed)
    k = np.arange(n)
    a = np.ones(n)
    for m in range(1, n_modes + 1):
        c = rng.uniform(-0.5, 0.5) / (m * 0.5)
        phase = rng.uniform(0, 2 * np.pi)
        a += c * np.cos(2 * np.pi * m * k / n + phase)
    a = np.maximum(a, 1e-10)
    S = a.sum()
    return a / S


def random_init(n, seed=0):
    """Generate a random non-negative initialization."""
    rng = np.random.default_rng(seed)
    a = rng.uniform(0.1, 1.0, n)
    # Add some structure
    k = np.arange(n)
    a *= 1.0 + 0.3 * np.cos(2 * np.pi * k / n * rng.integers(1, 5))
    a = np.maximum(a, 1e-10)
    return a / a.sum()


def full_refine(a, time_budget, seed=0):
    """Full refinement pipeline: L-BFGS + Fienup + subgradient."""
    n = len(a)
    a = a.copy()
    S = a.sum()
    if S < 1e-10:
        return a, np.inf
    a = a / S

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    # Phase A: L-BFGS with t-annealing
    remaining = time_budget - (time.monotonic() - start)
    if remaining > 10:
        t_schedule = [20, 50, 100, 200, 400]
        alloc = min(remaining * 0.5, 80.0)
        a_lbfgs = lbfgs_optimize(best_a, t_schedule, maxiter_per_t=200,
                                  time_budget=alloc, seed=seed)
        obj_lbfgs = compute_obj(a_lbfgs)
        if obj_lbfgs < best_obj:
            best_obj = obj_lbfgs
            best_a = a_lbfgs.copy()
            print(f"    L-BFGS: {best_obj:.6f}")

    # Phase B: Fienup clipping
    remaining = time_budget - (time.monotonic() - start)
    if remaining > 5:
        alloc = min(remaining * 0.2, 20.0)
        a_fien = fienup_refine(best_a, n_iters=300, alpha=0.6,
                               time_budget=alloc, seed=seed)
        obj_fien = compute_obj(a_fien)
        if obj_fien < best_obj:
            best_obj = obj_fien
            best_a = a_fien.copy()
            print(f"    Fienup: {best_obj:.6f}")

    # Phase C: Subgradient with simplex projection
    remaining = time_budget - (time.monotonic() - start)
    if remaining > 5:
        alloc = min(remaining * 0.2, 20.0)
        a_sg = subgradient_simplex(best_a, max_iters=3000,
                                    time_budget=alloc, seed=seed)
        obj_sg = compute_obj(a_sg)
        if obj_sg < best_obj:
            best_obj = obj_sg
            best_a = a_sg.copy()
            print(f"    Subgrad: {best_obj:.6f}")

    # Phase D: Second L-BFGS pass (finer t)
    remaining = time_budget - (time.monotonic() - start)
    if remaining > 5:
        t_schedule2 = [100, 200, 500, 800]
        alloc = min(remaining * 0.3, 40.0)
        a_lbfgs2 = lbfgs_optimize(best_a, t_schedule2, maxiter_per_t=200,
                                   time_budget=alloc, seed=seed + 1000)
        obj_lbfgs2 = compute_obj(a_lbfgs2)
        if obj_lbfgs2 < best_obj:
            best_obj = obj_lbfgs2
            best_a = a_lbfgs2.copy()
            print(f"    L-BFGS2: {best_obj:.6f}")

    return best_a, best_obj


def propose_candidate(seed=42, budget_s=1500):
    """
    Main search function. Uses L-BFGS with log-sum-exp smoothed objective,
    Fienup clipping-and-filtering, subgradient with simplex projection,
    multi-n exploration, and random restarts to minimize the target objective.
    """
    start_time = time.monotonic()
    deadline = start_time + float(budget_s) - 30.0

    best_seq = None
    best_score = np.inf

    # Load parent
    parent, parent_obj = get_parent_sequence()
    if parent is not None:
        best_score = parent_obj
        best_seq = parent.copy()
        n_parent = len(parent)
    else:
        n_parent = 500
        best_seq = np.ones(n_parent) / n_parent
        best_score = compute_obj(best_seq)

    print(f"Starting best: n={len(best_seq)}, score={best_score:.6f}")

    # === Phase 1: Main optimization on parent ===
    print("\n=== Phase 1: Main optimization on parent ===")
    remaining = deadline - time.monotonic()
    if remaining > 30:
        alloc = min(remaining * 0.35, 300.0)
        print(f"  Refining parent (n={n_parent}), alloc={alloc:.0f}s")
        a_ref, obj_ref = full_refine(best_seq, time_budget=alloc, seed=seed)
        if obj_ref < best_score:
            best_score = obj_ref
            best_seq = a_ref.copy()
            print(f"  ** NEW BEST: n={n_parent}, score={best_score:.6f} **")

    # === Phase 2: Try nearby n values ===
    print("\n=== Phase 2: Nearby n exploration ===")
    n_candidates = sorted(set([
        max(100, n_parent - 100),
        max(100, n_parent - 50),
        n_parent + 50,
        n_parent + 100,
        n_parent + 200,
        min(1200, n_parent + 300),
    ]))
    n_candidates = [n for n in n_candidates if 100 <= n <= 1500]

    for n_try in n_candidates:
        if time.monotonic() > deadline - 60:
            break
        remaining = deadline - time.monotonic()
        if remaining < 30:
            break

        a_interp = interpolate_sequence(best_seq, n_try)
        alloc = min(remaining * 0.2, 80.0)
        print(f"  n={n_try}: refining...")
        a_new, obj_new = full_refine(a_interp, time_budget=alloc, seed=seed + n_try)
        if obj_new < best_score:
            best_score = obj_new
            best_seq = a_new.copy()
            print(f"  ** NEW BEST: n={n_try}, score={best_score:.6f} **")
        else:
            print(f"  n={n_try}: {obj_new:.6f} (best={best_score:.6f})")

    # === Phase 3: Different initializations ===
    print("\n=== Phase 3: Diverse initializations ===")
    n_best = len(best_seq)

    inits = []
    for n_try in [n_best, max(200, n_best - 50), min(1000, n_best + 100)]:
        inits.append((n_try, f"cosine", cosine_init(n_try, n_modes=12, seed=seed + n_try)))
        inits.append((n_try, f"random", random_init(n_try, seed=seed + n_try + 1000)))

    for n_try, name, a_init in inits:
        if time.monotonic() > deadline - 40:
            break
        remaining = deadline - time.monotonic()
        if remaining < 20:
            break

        alloc = min(remaining * 0.15, 50.0)
        a_new, obj_new = full_refine(a_init, time_budget=alloc, seed=seed + hash(name) % 10000)
        if obj_new < best_score:
            best_score = obj_new
            best_seq = a_new.copy()
            print(f"  {name} n={n_try}: ** NEW BEST: {best_score:.6f} **")
        else:
            print(f"  {name} n={n_try}: {obj_new:.6f}")

    # === Phase 4: Intensive refinement with perturbations ===
    print("\n=== Phase 4: Intensive refinement ===")
    n_final = len(best_seq)

    for round_num in range(10):
        if time.monotonic() > deadline - 30:
            break
        remaining = deadline - time.monotonic()
        if remaining < 20:
            break

        # Try perturbations of different magnitudes
        rng = np.random.default_rng(seed + round_num * 7717)
        best_this_round = best_score

        for pert_scale in [0.03, 0.08, 0.15]:
            if time.monotonic() > deadline - 20:
                break
            remaining = deadline - time.monotonic()
            if remaining < 15:
                break

            a_pert = best_seq * np.exp(rng.normal(0, pert_scale, n_final))
            a_pert = project_simplex(a_pert)

            alloc = min(remaining * 0.25, 40.0)
            a_new, obj_new = full_refine(a_pert, time_budget=alloc,
                                          seed=seed + round_num * 100 + int(pert_scale * 100))
            if obj_new < best_this_round:
                best_this_round = obj_new
            if obj_new < best_score:
                best_score = obj_new
                best_seq = a_new.copy()
                print(f"  R{round_num} pert({pert_scale}): ** NEW BEST: {best_score:.6f} **")

        if best_this_round >= best_score:
            # No improvement, try a more aggressive restart
            remaining = deadline - time.monotonic()
            if remaining > 30 and round_num % 3 == 2:
                a_restart = project_simplex(best_seq + rng.normal(0, 0.3, n_final))
                alloc = min(remaining * 0.2, 50.0)
                a_new, obj_new = full_refine(a_restart, time_budget=alloc,
                                              seed=seed + round_num * 2000)
                if obj_new < best_score:
                    best_score = obj_new
                    best_seq = a_new.copy()
                    print(f"  R{round_num} restart: ** NEW BEST: {best_score:.6f} **")

    # === Phase 5: Final polish ===
    print("\n=== Phase 5: Final polish ===")
    remaining = deadline - time.monotonic()
    if remaining > 10:
        alloc = min(remaining * 0.6, 60.0)
        a_pol, obj_pol = full_refine(best_seq, time_budget=alloc, seed=seed + 99999)
        if obj_pol < best_score:
            best_score = obj_pol
            best_seq = a_pol.copy()
            print(f"  Polish: ** NEW BEST: {best_score:.6f} **")

    # === Final validation ===
    if best_seq is None:
        best_seq = np.ones(300) / 300
        best_score = compute_obj(best_seq)

    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    val = evaluate_sequence(result)
    if np.isfinite(val):
        if val < best_score:
            best_score = val
    else:
        result = [1.0] * 300
        best_score = 2.0

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_score:.6f}, time={elapsed:.1f}s ===")

    return result
