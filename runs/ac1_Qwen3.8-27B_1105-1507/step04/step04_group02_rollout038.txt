```python
from scipy.optimize import linprog
from scipy.interpolate import interp1d
import time

def compute_conv(a):
    n = len(a)
    size = 1
    while size < 2*n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    conv = np.fft.irfft(fa * fa, size)
    return conv[:2*n-1]

def obj_value(a):
    n = len(a)
    S = np.sum(a)
    if S < 1e-10:
        return np.inf
    conv = compute_conv(a)
    return 2.0 * n * np.max(conv) / (S * S)

def solve_dual_lp(a, M):
    """
    Solve the dual LP:
    min M * sum(lambda)
    s.t. sum_k lambda[k] * a[k-i] >= 1 for i = 0, ..., n-1
         lambda[k] >= 0
    
    The constraint matrix (A^T) has (A^T)[i, k] = a[k-i] if 0 <= k-i < n.
    """
    n = len(a)
    # Dual: min c^T * lambda s.t. B^T * lambda >= 1, lambda >= 0
    # where B^T has shape (n, 2n-1) and B^T[i,k] = a[k-i] if valid
    
    # For linprog: minimize c^T x s.t. A_ub @ x <= b_ub, bounds
    # We need: B^T @ lambda >= 1  =>  -B^T @ lambda <= -1
    # c = M * ones(2n-1)
    
    # Build B^T as sparse matrix
    from scipy.sparse import coo_matrix
    rows = []
    cols = []
    vals = []
    for k in range(2*n - 1):
        # For each k, find i such that 0 <= k-i < n, i.e., k-n+1 <= i <= k
        i_start = max(0, k - n + 1)
        i_end = min(n - 1, k)
        for i in range(i_start, i_end + 1):
            rows.append(i)
            cols.append(k)
            vals.append(a[k - i])
    
    Bt = coo_matrix((vals, (rows, cols)), shape=(n, 2*n-1)).tocsr()
    
    c = M * np.ones(2*n - 1)
    A_ub = -Bt  # -B^T @ lambda <= -1
    b_ub = -np.ones(n)
    bounds = [(0, None)] * (2*n - 1)
    
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs',
                  options={'presolve': True, 'time_limit': 5})
    
    if res.status == 0:
        return res.x
    return None

def extract_primal_from_dual(a, M, lambdas):
    """
    Extract the primal solution g from the dual solution using complementary slackness.
    g[i] > 0 only if the i-th dual constraint is tight.
    lambda[k] > 0 only if the k-th primal constraint is tight.
    
    The active set of dual constraints (tight ones) gives us the support of g.
    Then we solve a small LP on the active set.
    """
    n = len(a)
    # Check which dual constraints are tight
    Bt_vals = np.zeros(n)
    for i in range(n):
        s = 0.0
        for k in range(max(0, i), min(2*n-1, i+n)):
            s += lambdas[k] * a[k-i]
        Bt_vals[i] = s
    
    tol = 1e-6
    active = np.where(Bt_vals >= 1.0 - tol)[0]
    
    if len(active) == 0:
        return np.ones(n) / n
    
    # Solve reduced LP: max sum(g[i]) for i in active
    # s.t. conv(a,g)[k] <= M for all k, g[i] = 0 for i not in active
    # This is a smaller LP with only len(active) variables
    
    # If active is small, solve directly
    m = len(active)
    if m == n:
        # All active, solve the full primal
        return solve_primal_lp(a, M)
    
    # Build the reduced constraint matrix
    # conv(a,g)[k] = sum_{i in active} a[k-i] * g[i] <= M
    from scipy.sparse import coo_matrix
    rows_r = []
    cols_r = []
    vals_r = []
    for i_idx, i in enumerate(active):
        for k in range(max(0, i), min(2*n-1, i+n)):
            rows_r.append(k)
            cols_r.append(i_idx)
            vals_r.append(a[k-i])
    
    Ar = coo_matrix((vals_r, (rows_r, cols_r)), shape=(2*n-1, m)).tocsr()
    c_r = -np.ones(m)
    b_ub_r = np.full(2*n-1, M)
    bounds_r = [(0, None)] * m
    
    res = linprog(c_r, A_ub=Ar, b_ub=b_ub_r, bounds=bounds_r, method='highs',
                  options={'preserve': True, 'time_limit': 5})
    
    if res.status == 0:
        g = np.zeros(n)
        g[active] = np.maximum(res.x, 0)
        return g
    return np.ones(n) / n

def solve_primal_lp(a, M):
    """Solve the primal LP directly: max sum(g) s.t. conv(a,g) <= M, g >= 0."""

    from scipy.sparse import coo_matrix
```

```python
"""
Best-response dynamics with dual LP and projected subgradient for minimizing
2*n*max(conv(a,a))/(sum(a))^2.
"""

import numpy as np
from scipy.optimize import linprog
from scipy.sparse import diags
import time

def compute_conv(a):
    n = len(a)
    size = 2*n - 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)

def obj_value(a):
    n = len(a)
    S = a.sum()
    if S < 1e-10: return np.inf
    return 2.0 * n * compute_conv(a).max() / (S * S)

def build_conv_matrix(a):
    """Build sparse matrix A (2n-1 x n) where A[k,i] = a[k-i] if 0<=k-i<n."""
    n = len(a)
    diags_list = [np.full(n, a[d]) for d in range(n)]
    offsets = list(range(n))
    return diags(diags_list, offsets, shape=(2*n-1, n), format='csr')

def solve_dual(a, M):
    """Solve dual LP: min M*sum(lam) s.t. A^T lam >= 1, lam >= 0."""
    n = len(a)
    A = build_conv_matrix(a)  # (2n-1, n)
    At = A.T  # (n, 2n-1)
    c = M * np.ones(2*n-1)
    res = linprog(c, A_ub=-At, b_ub=-np.ones(n), bounds=[(0,None)]*(2*n-1),
                  method='highs', options={'time_limit': 5})
    return res.x if res.status == 0 else None

def best_response(a):
    """Compute best response g to a."""
    n = len(a)
    conv = compute_conv(a)
    M = conv.max()
    
    # Try dual LP
    lam = solve_dual(a, M)
    if lam is not None:
        # Extract primal from complementary slackness
        dual_vals = (build_conv_matrix(a).T).dot(lam)  # n-vector
        tol = 1e-6
        active = np.where(dual_vals >= 1.0 - tol)[0]
        
        if len(active) > 0 and len(active) < n:
            # Solve reduced primal
            A = build_conv_matrix(a)[:, active]  # (2n-1, len(active))
            res = linprog(-np.ones(len(active)), A_ub=A, b_ub=np.full(2*n-1, M),
                         bounds=[(0,None)]*len(active), method='highs',
                         options={'time_limit': 5})
            if res.status == 0:
                g = np.zeros(n)
                g[active] = np.maximum(res.x, 0)
                S = g.sum()
                if S > 1e-10:
                    return g / S
        elif len(active) == n:
            # Full primal
            A = build_conv_matrix(a)
            res = linprog(-np.ones(n), A_ub=A, b_ub=np.full(2*n-1, M),
                         bounds=[(0,None)]*n, method='highs',
                         options={'time_limit': 5})
            if res.status == 0:
                g = np.maximum(res.x, 0)
                S = g.sum()
                if S > 1e-10:
                    return g / S
    
    # Fallback: uniform
    return np.ones(n) / n

def project_simplex(v):
    """Project v onto the probability simplex."""
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n+1) > (cssv - 1))[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0)

def subgradient_step(a, step_size, topk=20):
    """One step of projected subgradient descent."""
    n = len(a)
    conv = compute_conv(a)
    M = conv.max()
    
    # Active set: top-k conv values
    active = np.argsort(conv)[-topk:]
    
    # Subgradient
    g = np.zeros(n)
    for k in active:
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        g[j_start:j_end+1] += 2.0 * a[k - np.arange(j_start, j_end+1)]
    g /= len(active)
    
    # Step
    a_new = a - step_size * g
    a_new = project_simplex(a_new)
    return a_new

    rng = np.random.default_rng(seed)
```

```python
"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using best-response dynamics with dual LP
acceleration, projected subgradient polishing, multi-n exploration, and
perturbation-based escape from local minima. Key innovation: efficient dual LP
with pre-built sparse structure and adaptive line search for fast per-iteration
throughput enabling thousands of best-response steps.

import numpy as np
from scipy.optimize import linprog
from scipy.sparse import diags, csr_matrix
from scipy.interpolate import interp1d
import time


def compute_conv(a):
    """Compute full linear convolution a*a via FFT."""
    n = len(a)
    size = 2 * n - 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)


def obj_value(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    S = float(np.sum(a))
    if S < 1e-10:
        return np.inf
    conv = compute_conv(a)
    M = float(np.max(conv))
    return 2.0 * n * M / (S * S)


def build_conv_matrix(a):
    """Build sparse Toeplitz matrix A (2n-1 x n) where A[k,i] = a[k-i] if 0<=k-i<n."""
    n = len(a)
    diags_list = []
    offsets = []
    for d in range(n):
        length = min(n, 2 * n - 1 - d)
        diags_list.append(np.full(length, a[d]))
        offsets.append(d)
    return diags(diags_list, offsets, shape=(2 * n - 1, n), format='csr')


def solve_best_response(a):
    """
    Compute best response g to a via dual LP.
    Returns (g_normalized, M, sum_g_before_norm).
    """
    n = len(a)
    conv = compute_conv(a)
    M = float(np.max(conv))

    A = build_conv_matrix(a)
    At = A.T.tocsr()  # (n, 2n-1)

    # Dual: min M * sum(lambda) s.t. At @ lambda >= 1, lambda >= 0
    c = M * np.ones(2 * n - 1)
    b_ub = -np.ones(n)

    try:
        res = linprog(c, A_ub=-At, b_ub=b_ub,
                      bounds=[(0, None)] * (2 * n - 1),
                      method='highs', options={'time_limit': 8, 'presolve': True})
        if res.status != 0:
            return np.ones(n) / n, M, 1.0
        lam = res.x
    except Exception:
        return np.ones(n) / n, M, 1.0

    # Complementary slackness to find active set of primal
    dual_vals = At.dot(lam)  # n-vector
    tol = 1e-6 * max(1.0, M)
    active = np.where(dual_vals >= 1.0 - tol)[0]

    if len(active) == 0:
        active = np.where(dual_vals == np.max(dual_vals))[0]

    if len(active) == n:
        # All active: solve full primal
        try:
            res2 = linprog(-np.ones(n), A_ub=A, b_ub=np.full(2 * n - 1, M),
                          bounds=[(0, None)] * n, method='highs',
                          options={'time_limit': 8, 'presolve': True})
            if res2.status == 0:
                g = np.maximum(res2.x, 0)
                Sg = float(np.sum(g))
                if Sg > 1e-10:
                    return g / Sg, M, Sg
        except Exception:
            pass
        return np.ones(n) / n, M, 1.0
    else:
        # Solve reduced primal on active set
        A_red = A[:, active].tocsr()
        m = len(active)
        try:
            res2 = linprog(-np.ones(m), A_ub=A_red, b_ub=np.full(2 * n - 1, M),
                          bounds=[(0, None)] * m, method='highs',
                          options={'time_limit': 8, 'presolve': True})
            if res2.status == 0:
                g = np.zeros(n)
                g[active] = np.maximum(res2.x, 0)
                Sg = float(np.sum(g))
                if Sg > 1e-10:
                    return g / Sg, M, Sg
        except Exception:
            pass
        return np.ones(n) / n, M, 1.0


def project_simplex(v):
    """Project v onto probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    rho = np.nonzero(cond)[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho = rho[-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0)


def subgradient_direction(a, topk=20):
    """Compute subgradient direction of max_k conv(a,a)[k]."""
    n = len(a)
    conv = compute_conv(a)
    M = float(np.max(conv))

    # Get active constraints (top-k)
    if topk >= 2 * n - 1:
        active = np.arange(2 * n - 1)
    else:
        # Use values close to max
        threshold = M - 1e-8
        active = np.where(conv >= threshold)[0]
        if len(active) == 0:
            active = np.argsort(conv)[-topk:]
        elif len(active) > topk:
            active = active[np.argsort(conv[active])[-topk:]]

    # Subgradient: for each active k, d(conv[k])/da[i] = 2*a[k-i] if 0<=k-i<n
    g = np.zeros(n)
    for k in active:
        for i in range(max(0, k - n + 1), min(n, k + 1)):
            j = k - i
            if 0 <= j < n:
                g[i] += 2.0 * a[j]
    g /= max(1, len(active))
    return g, M, conv

def sg_refine(a, time_budget, topk=10):

    n = len(a)

    momentum = 0.7
    stall = 0

    while time.monotonic() - start < time_budget:
        g, M, conv = subgradient_direction(a, topk=topk)
        # Center to preserve sum
        g_centered = g - np.mean(g)
        momentum_v = momentum * momentum_v + (1 - momentum) * g_centered
        a_new = a - step * momentum_v
        a_new = project_simplex(a_new)

        if obj_new < best_obj - 1e-13:

            best_a = a_new.copy()
            a = a_new
            stall = 0
        else:
            stall += 1
            step *= 0.97
            if stall > 200:
                break

def cosine_init(n, seed=0, n_modes=10):
    """Cosine series initialization."""
    rng = np.random.default_rng(seed)
    k = np.arange(float(n))
    a = np.ones(n)
    for m in range(1, n_modes + 1):
        c = rng.uniform(-0.5, 0.5) / m
        phase = rng.uniform(0, 2 * np.pi)
        a += c * np.cos(2 * np.pi * m * k / n + phase)
    a = np.maximum(a, 1e-6)
    return a / a.sum()


def raised_cosine_init(n, alpha=2.0):
    k = np.arange(float(n))
    x = 2.0 * k / (n - 1) - 1.0
    a = (0.5 * (1.0 + np.cos(np.pi * x))) ** alpha
    a = np.maximum(a, 1e-8)
    return a / a.sum()


def interpolate_sequence(a, new_n):
    """Interpolate sequence to new length."""
    n = len(a)
    if n == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    try:
        f = interp1d(x_old, a, kind='cubic', fill_value='extrapolate')
        a_new = np.maximum(f(x_new), 0)
    except Exception:
        a_new = np.ones(new_n) / new_n
    S = a_new.sum()
    if S < 1e-10:
        return np.ones(new_n) / new_n
    return a_new / S


def perturb_sequence(a, scale, rng):
    """Multiplicative perturbation."""
    n = len(a)
    a_p = a * np.exp(rng.normal(0, scale, n))
    a_p = np.maximum(a_p, 0)
    S = a_p.sum()
    if S < 1e-10:
        return a.copy()
    return a_p / S

            print(f"  Phase 1: n={n_parent}, obj={best_score:.10f} ** IMPROVED **")
        else:
            print(f"  Phase 1: n={n_parent}, obj={obj_br:.10f} (best={best_score:.10f})")

    # === Phase 2: Subgradient polish on current best ===
    print("\n=== Phase 2: SG polish ===")
    remaining = deadline - time.monotonic()
    t2 = min(remaining * 0.1, 80.0)
    if t2 > 10:
        a_sg, obj_sg = sg_refine(best_seq, time_budget=t2)
        if obj_sg < best_score:
            best_score = obj_sg
            best_seq = a_sg.copy()
            print(f"  SG polish: obj={best_score:.10f} ** IMPROVED **")
        else:
            print(f"  SG polish: obj={obj_sg:.10f} (best={best_score:.10f})")

    # === Phase 3: Multi-n exploration ===
    print("\n=== Phase 3: Multi-n exploration ===")
    n_base = len(best_seq)
    n_options = sorted(set([
        200, 250, 300, 350, 400, 450, n_base, 550, 600, 700, 800, 1000, 1200
    ]))
    n_options = [n for n in n_options if 100 <= n <= 1500 and n != n_base]
    # Add interpolated lengths around base
    for delta in [-100, -50, 50, 100, 150]:
        n_try = n_base + delta
        if 100 <= n_try <= 1500:
            n_options.append(n_try)
    n_options = sorted(set(n_options))

    candidates = []
    for n in n_options:
        if time.monotonic() > deadline - 60:
            break
        remaining = deadline - time.monotonic()
        if remaining < 30:
            break

        # Initialize from interpolation of current best
        a_interp = interpolate_sequence(best_seq, n)
        obj_interp = obj_value(a_interp)

        # Also try fresh initializations
        init_options = [
            a_interp,
            cosine_init(n, seed=seed + n * 7),
            raised_cosine_init(n, alpha=2.0),
        ]

        best_local = None
        best_local_obj = np.inf
        for a_init in init_options:
            obj_init = obj_value(a_init)
            if obj_init < best_local_obj:
                best_local_obj = obj_init
                best_local = a_init.copy()

        # Quick best-response (limited time)
        t_quick = min(20.0, remaining * 0.08)
        if t_quick > 5:
            a_br, obj_br = best_response_iteration(best_local, time_budget=t_quick)
            if obj_br < best_local_obj:
                best_local_obj = obj_br
                best_local = a_br.copy()

        candidates.append((best_local, best_local_obj, n))
        if best_local_obj < best_score:
            best_score = best_local_obj
            best_seq = best_local.copy()
            print(f"  n={n}: obj={best_score:.10f} ** NEW BEST **")
        else:
            print(f"  n={n}: obj={best_local_obj:.10f}")

    # Sort and refine top candidates
    candidates.sort(key=lambda x: x[1])
    print("\n=== Phase 4: Refine top candidates ===")
    for idx in range(min(4, len(candidates))):
        if time.monotonic() > deadline - 40:
            break
        remaining = deadline - time.monotonic()
        if remaining < 30:
            break
        a_cand, obj_cand, n_cand = candidates[idx]
        if obj_cand > best_score + 0.01:
            break
        t_ref = min(remaining * 0.15, 100.0)
        a_ref, obj_ref = best_response_iteration(a_cand, time_budget=t_ref, verbose=True)
        if obj_ref < best_score:
            best_score = obj_ref
            best_seq = a_ref.copy()
            print(f"  Refine n={n_cand}: obj={best_score:.10f} ** NEW BEST **")
        else:
            print(f"  Refine n={n_cand}: obj={obj_ref:.10f} (best={best_score:.10f})")

    # === Phase 5: Perturbation + re-optimize ===
    print("\n=== Phase 5: Perturbation escape ===")
    rng = np.random.default_rng(seed + 99999)
    n_best = len(best_seq)
    for pert_scale in [0.15, 0.25, 0.4, 0.6]:
        if time.monotonic() > deadline - 25:
            break
        remaining = deadline - time.monotonic()
        if remaining < 20:
            break
        a_pert = perturb_sequence(best_seq, pert_scale, rng)
        t_pert = min(remaining * 0.12, 60.0)
        a_new, obj_new = best_response_iteration(a_pert, time_budget=t_pert)
        if obj_new < best_score:
            best_score = obj_new
            best_seq = a_new.copy()
            print(f"  Perturb({pert_scale}): obj={best_score:.10f} ** NEW BEST **")
        else:
            print(f"  Perturb({pert_scale}): obj={obj_new:.10f} (best={best_score:.10f})")

    # === Phase 6: Final intensive best-response on best ===
    print("\n=== Phase 6: Final intensive BR ===")
    remaining = deadline - time.monotonic()
    if remaining > 30:
        t_final = min(remaining * 0.5, 200.0)
        a_fin, obj_fin = best_response_iteration(best_seq, time_budget=t_final, verbose=True)
        if obj_fin < best_score:
            best_score = obj_fin
            best_seq = a_fin.copy()
            print(f"  Final BR: obj={best_score:.10f} ** NEW BEST **")

    # === Phase 7: Final SG polish ===
    remaining = deadline - time.monotonic()
    if remaining > 10:
        a_pol, obj_pol = sg_refine(best_seq, time_budget=min(remaining * 0.6, 50.0), topk=8)
        if obj_pol < best_score:
            best_score = obj_pol
            best_seq = a_pol.copy()
            print(f"  Final SG: obj={best_score:.10f} ** NEW BEST **")

    # Final validation
    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    val = evaluate_sequence(result)
    if val < np.inf:
        best_score = min(best_score, val)
        print(f"\n=== FINAL: n={len(result)}, eval_score={val:.10f} ===")
    else:
        print("WARNING: evaluate_sequence returned inf, using fallback")
        result = [1.0] * 300
        best_score = 2.0

    elapsed = time.monotonic() - start_time
    print(f"Total time: {elapsed:.1f}s")
    return result
```
