        if residual < 1e-10:
            return f
        
        # Anderson step
        if len(X) >= 2:
            # Build the Anderson system
            k = min(m, len(X) - 1)
            Xk = np.array(X[-k:])  # shape (k, n)
            Fk = np.array(Fx[-k:])  # shape (k, n)
            # d_k = F(x_k) - x_k
            D = Fk - Xk  # shape (k, n)
            # Solve min ||beta|| s.t. D @ beta = D[-1]
            # Actually: find beta such that sum beta_i * d_i = d_k
            # This is: D @ beta = d_k where d_k = D[-1]
            d_k = D[-1]
            G = D @ D.T  # k x k
            rhs = D @ d_k  # k vector
            try:
                beta = np.linalg.solve(G + 1e-12 * np.eye(k), rhs)
            except np.linalg.LinAlgError:
                beta = np.zeros(k)
            
            x_new = f - D @ beta  # Anderson extrapolation
            x_new = np.maximum(x_new, 0)
            s_new = x_new.sum()
            if s_new < 1e-10:
                x_new = f.copy()
            else:
                x_new = x_new / s_new
        else:
            x_new = f.copy()
        
        x = x_new
        X.append(x.copy())
        Fx.append(F(x))
        
        # Keep only last m+1 in history

            X = X[-(m+1):]
            Fx = Fx[-(m+1):]
    
    return np.maximum(x, 0) / max(x.sum(), 1e-10)

def build_conv_matrix(n, a):
    """Build the (2n-1) x n convolution matrix T(a) as sparse."""
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        j_min = max(0, k - n + 1)
        j_max = min(n - 1, k)
        for j in range(j_min, j_max + 1):
            rows.append(k)
            cols.append(j)
            vals.append(a[k - j])
    return csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))

def solve_lp_best_response(a, M):
    """Solve: max sum(g) s.t. T(a) @ g <= M, g >= 0. Returns g."""
    n = len(a)
    A = build_conv_matrix(n, a)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    try:
        r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                    options={'time_limit': 60, 'presolve': True})
        if r.success and r.x is not None:
            g = np.maximum(r.x, 0)
            if g.sum() > 1e-15:
                return g, r
    except Exception:
        pass
    return None, None

def subgradient_step(a):
    """Compute exact subgradient of max(conv(a,a))."""
    n = len(a)
    b = fast_conv(a)
    M = b.max()
    tol = 1e-8 * M if M > 0 else 1e-10
    active = np.where(b >= M - tol)[0]
    
    grad = np.zeros(n)
    for k in active:
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        grad[lo:hi + 1] += 2.0 * a[k - hi:k - lo + 1][::-1]
    
    gnorm = np.linalg.norm(grad)
    if gnorm > 1e-15:
        grad /= gnorm
    
    return grad, active, b

def run_subgradient_polish(a, max_iters, deadline, lr_init=0.1):

    a = np.maximum(a.astype(np.float64), 1e-8)
    a = a / a.sum()
    n = len(a)
    
    best_a = a.copy()
    best_M = fast_conv(a).max()
    running_sum = np.zeros(n)
    count = 0
    
    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        grad, active, b = subgradient_step(a)
        M = b.max()

            best_M = M
            best_a = a.copy()

        s = a_new.sum()

        else:
            a_new = a_new / s

            a_avg /= s

def resample_seq(src, dest_n):
    """Resample 1D sequence to new length."""
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)

At the fixed point, we have:
- (a*a)_k <= M for all k
- There exist y_k >= 0 with y_k > 0 only where (a*a)_k = M

"""
Minimizes J(a) = 2*n*max(conv(a,a))/(sum(a))^2 using:
1. LP mirror descent (best response dynamics) as the primary optimizer
2. Anderson acceleration to speed up fixed-point convergence
3. Symmetry exploitation (palindromic sequences)

import numpy as np
from scipy.optimize import linprog
from scipy.sparse import csr_matrix
import time


def fast_conv(a):
    """Full self-convolution via FFT."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]


def compute_J(a):
    """Compute J(a) = 2*n*max(conv(a,a))/(sum(a))^2."""
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))


def enforce_symmetry(a):
    """Make sequence palindromic (a[i] = a[n-1-i])."""
    return 0.5 * (a + a[::-1])


def build_conv_matrix(n, a):
    """Build the (2n-1) x n convolution matrix T(a) as sparse."""
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        j_min = max(0, k - n + 1)
        j_max = min(n - 1, k)
        for j in range(j_min, j_max + 1):
            rows.append(k)
            cols.append(j)
            vals.append(a[k - j])
    return csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))


def solve_lp_best_response(a, M):
    """Solve: max sum(g) s.t. T(a) @ g <= M, g >= 0. Returns g or None."""
    n = len(a)
    A = build_conv_matrix(n, a)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    try:
        r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                    options={'time_limit': 120, 'presolve': True})
        if r.success and r.x is not None:
            g = np.maximum(r.x, 0)
            if g.sum() > 1e-15:
                return g
    except Exception:
        pass
    return None


def lp_best_response_symmetric(a, M):
    """
    LP best response exploiting palindromic symmetry of a.
    If a is palindromic, T(a) has symmetry: T[k,j] = T[2n-2-k, n-1-j]
    so constraints come in pairs. We can halve the constraint count.
    """
    n = len(a)
    # Check if a is approximately symmetric
    if np.max(np.abs(a - a[::-1])) > 1e-8:
        return solve_lp_best_response(a, M)
    
    # Only need constraints for k = 0, ..., n-1 (the rest are symmetric)
    # T(a)[k,j] for k=0..n-1, j=0..n-1
    rows = []
    cols = []
    vals = []
    half = n  # k = 0 to n-1 (n constraints instead of 2n-1)
    for k in range(half):
        j_min = max(0, k - n + 1)
        j_max = min(n - 1, k)
        for j in range(j_min, j_max + 1):
            rows.append(k)
            cols.append(j)
            vals.append(a[k - j])
    A = csr_matrix((vals, (rows, cols)), shape=(half, n))
    c = -np.ones(n)
    b_ub = np.full(half, M)
    bounds = [(0, None)] * n
    try:
        r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                    options={'time_limit': 120, 'presolve': True})
        if r.success and r.x is not None:
            g = np.maximum(r.x, 0)
            if g.sum() > 1e-15:
                return g
    except Exception:
        pass
    return None


def run_lp_mirror_descent(a, max_iters, deadline, use_symmetry=False,
                          anderson_m=0, hard_update=True):
    """
    LP mirror descent with optional Anderson acceleration.
    
    Parameters:
    - a: initial sequence (will be normalized)
    - max_iters: maximum number of iterations
    - deadline: time deadline (monotonic clock)
    - use_symmetry: enforce palindromic symmetry
    - anderson_m: Anderson acceleration memory (0 = no acceleration)
    - hard_update: if True, use t=1 (a_new = g); if False, use adaptive t
    """
    a = np.maximum(a.astype(np.float64), 1e-8)
    a = a / a.sum()
    n = len(a)
    
    best_a = a.copy()
    best_M = fast_conv(a).max()
    
    # Anderson history
    X_hist = []
    F_hist = []
    
    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        
        b = fast_conv(a)
        M = float(b.max())
        
        if M < best_M * (1 - 1e-10):
            best_M = M
            best_a = a.copy()
        
        # Solve LP
        if use_symmetry:
            g = lp_best_response_symmetric(a, M)
        else:
            g = solve_lp_best_response(a, M)
        
        if g is None:
            break
        
        g = np.maximum(g, 0)
        s = g.sum()
        if s < 1e-10:
            break
        g = g / s
        
        if use_symmetry:
            g = enforce_symmetry(g)
            g = np.maximum(g, 0)
            s = g.sum()
            if s > 1e-10:
                g = g / s
        
        # Anderson acceleration
        if anderson_m > 0 and len(X_hist) >= 1:
            X_hist.append(a.copy())
            F_hist.append(g.copy())
            
            if len(X_hist) > anderson_m + 1:
                X_hist = X_hist[-(anderson_m + 1):]
                F_hist = F_hist[-(anderson_m + 1):]
            
            if len(X_hist) >= 2:
                k = min(anderson_m, len(X_hist) - 1)
                Xk = np.array(X_hist[-k:])
                Fk = np.array(F_hist[-k:])
                D = Fk - Xk
                d_k = D[-1]
                G = D @ D.T
                rhs = D @ d_k
                try:
                    beta = np.linalg.solve(G + 1e-12 * np.eye(k), rhs)
                    # Clip beta to prevent divergence
                    beta = np.clip(beta, -2.0, 2.0)
                except np.linalg.LinAlgError:
                    beta = np.zeros(k)
                
                a_new = g - D @ beta
                a_new = np.maximum(a_new, 0)
                s_new = a_new.sum()
                if s_new < 1e-10:
                    a_new = g.copy()
                else:
                    a_new = a_new / s_new
            else:
                a_new = g.copy()
        elif hard_update:
            a_new = g.copy()
        else:
            # Adaptive soft update
            frac = it / max(1, max_iters)
            t = 0.5 + 0.5 * frac
            a_new = (1 - t) * a + t * g
            a_new = np.maximum(a_new, 0)
            s_new = a_new.sum()
            if s_new < 1e-10:
                a_new = a.copy()
            else:
                a_new = a_new / s_new
        
        if use_symmetry:
            a_new = enforce_symmetry(a_new)
            a_new = np.maximum(a_new, 0)
            s_new = a_new.sum()
            if s_new > 1e-10:
                a_new = a_new / s_new
        
        # Convergence check
        if np.linalg.norm(a_new - a) / max(a.sum(), 1e-10) < 1e-12:
            if fast_conv(a_new).max() < best_M:
                best_a = a_new.copy()
                best_M = fast_conv(a_new).max()
            break
        
        a = a_new
    
    M_final = fast_conv(a).max()
    if M_final < best_M:
        best_a = a.copy()
    
    return best_a


def smooth_gradient_step(a, tau):
    """Compute gradient of smooth max of conv(a,a)."""
    n = len(a)
    b = fast_conv(a)
    b_max = b.max()
    exp_b = np.exp(tau * (b - b_max))
    p = exp_b / exp_b.sum()
    
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        pk = p[k]
        if pk < 1e-13:
            continue
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        grad[lo:hi + 1] += 2.0 * pk * a[k - hi:k - lo + 1][::-1]
    
    return grad, p, b


def run_fista(a, deadline, initial_tau=2.0, final_tau=60.0, max_iters=3000):
    """Nesterov-accelerated (FISTA) smooth gradient descent."""
    a = np.maximum(a.astype(np.float64), 1e-8)
    a = a / a.sum()
    n = len(a)
    
    best_a = a.copy()
    best_M = fast_conv(a).max()
    
    y = a.copy()
    t_mom = 1.0
    lr = 0.3
    
    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        
        frac = it / max(1, max_iters)
        tau = initial_tau + (final_tau - initial_tau) * (frac ** 0.6)
        
        grad, p, b = smooth_gradient_step(y, tau)
        M = b.max()
        
        if M < best_M * (1 - 1e-9):
            best_M = M
            best_a = y.copy()
        
        # Adaptive learning rate
        lr = 0.3 * (0.05 / 0.3) ** frac
        
        x = y - lr * grad / tau
        x = np.maximum(x, 0.0)
        s = x.sum()
        if s < 1e-10:
            x = y.copy()
        else:
            x = x / s
        
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t_mom * t_mom)) / 2.0
        y = x + ((t_mom - 1.0) / t_new) * (x - y)
        y = np.maximum(y, 0.0)
        sy = y.sum()
        if sy < 1e-10:
            y = x.copy()
        else:
            y = y / sy
        t_mom = t_new
    
    return best_a


def run_subgradient(a, max_iters, deadline, lr_init=0.1):
    """Subgradient descent with averaging."""
    a = np.maximum(a.astype(np.float64), 1e-8)
    a = a / a.sum()
    n = len(a)
    
    best_a = a.copy()
    best_M = fast_conv(a).max()
    running_sum = np.zeros(n)
    count = 0
    
    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        
        b = fast_conv(a)
        M = b.max()
        tol = 1e-8 * M if M > 0 else 1e-10
        active = np.where(b >= M - tol)[0]
        
        grad = np.zeros(n)
        for k in active:
            lo = max(0, k - n + 1)
            hi = min(n - 1, k)
            grad[lo:hi + 1] += 2.0 * a[k - hi:k - lo + 1][::-1]
        
        gnorm = np.linalg.norm(grad)
        if gnorm > 1e-15:
            grad /= gnorm
        
        if M < best_M * (1 - 1e-9):
            best_M = M
            best_a = a.copy()
        
        lr = lr_init / (it + 1) ** 0.5
        a_new = a - lr * grad
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s < 1e-10:
            a_new = a.copy()
        else:
            a_new = a_new / s
        
        running_sum += a_new
        count += 1
        a = a_new
    
    if count > 10:
        a_avg = running_sum / count
        a_avg = np.maximum(a_avg, 0.0)
        s = a_avg.sum()
        if s > 1e-10:
            a_avg /= s
            M_avg = fast_conv(a_avg).max()
            if M_avg < best_M:
                best_a = a_avg
    
    return best_a


def resample_seq(src, dest_n):
    """Resample 1D sequence to new length via linear interpolation."""
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)


def estimate_lp_time(a):
    """Estimate time for one LP solve."""

    t0 = time.monotonic()
    M = fast_conv(a).max()
    g = solve_lp_best_response(a, M)
    elapsed = time.monotonic() - t0
    return elapsed
