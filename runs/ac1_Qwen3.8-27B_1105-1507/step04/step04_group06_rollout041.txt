"""
Minimizes J(a) = 2*n*max(convolve(a,a))/(sum(a))² using enhanced LP mirror descent with:
- Backtracking line search (faster than golden section)
- Momentum on LP directions for smoother convergence

import numpy as np
from scipy.optimize import linprog, minimize
from scipy.sparse import csr_matrix
import time

# Global caches
_A_cache = {}
_start_time = None
_best_sequence = None
_best_score = float('inf')


def fast_conv(a):
    """Full self-convolution via FFT."""
    n = len(a)
    s = 1
    while s < 2 * n - 1:
        s <<= 1
    fa = np.fft.rfft(a, s)
    return np.fft.irfft(fa * fa, s)[:2 * n - 1]


def compute_J(a):
    """Compute J(a) = 2*n*max(conv(a,a))/(sum(a))^2."""
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))


def get_A_structure(n):
    """Get or pre-compute row/col arrays for the constraint matrix."""
    if n not in _A_cache:
        rows = []
        cols = []
        for k in range(2 * n - 1):
            jl = max(0, k - n + 1)
            jh = min(n - 1, k)
            for j in range(jl, jh + 1):
                rows.append(k)
                cols.append(j)
        _A_cache[n] = (np.array(rows, dtype=np.int32),
                       np.array(cols, dtype=np.int32))
    return _A_cache[n]


def build_A_sparse(a):
    """Build sparse constraint matrix A where (A*g)_k = (a*g)_k."""
    n = len(a)
    rows_arr, cols_arr = get_A_structure(n)
    vals = a[rows_arr - cols_arr]
    return csr_matrix((vals, (rows_arr, cols_arr)), shape=(2 * n - 1, n))


def lp_solve(a, M):
    """
    Solve LP: max sum(g) s.t. A(a)*g <= M, g >= 0.
    Returns (g_normalized, dual_y) or (None, None).
    """
    n = len(a)
    c = -np.ones(n)
    A = build_A_sparse(a)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
    if r.success and r.x is not None:
        g = np.maximum(r.x, 0.0)
        gs = g.sum()
        if gs > 1e-10:
            g = g / gs
            dual_y = None
            if hasattr(r, 'ineqlin') and r.ineqlin is not None:
                dual_y = np.maximum(r.ineqlin, 0.0)
            return g, dual_y
    return None, None


def smooth_max_and_subgrad(a, beta):
    """
    Compute smooth approximation of max(conv(a,a)) and its subgradient.
    Returns (smooth_value, subgradient_array).
    """
    n = len(a)
    b = fast_conv(a)
    b_max = float(b.max())
    shifted = beta * (b - b_max)
    shifted = np.clip(shifted, -500, 500)
    exp_s = np.exp(shifted)
    Z = float(exp_s.sum())
    if Z < 1e-300:
        p = np.zeros(2 * n - 1)
        p[int(np.argmax(b))] = 1.0
    else:
        p = exp_s / Z
    g_val = (b_max + np.log(max(Z, 1e-300))) / beta

    # Subgradient: subgrad[i] = 2 * sum_l a[l] * p[l+i] for valid indices
    subgrad = np.zeros(n)
    # Pad p for easy slicing
    pp = np.zeros(3 * n)
    pp[:2 * n - 1] = p
    for i in range(n):
        end = min(n, 2 * n - 1 - i)
        if end > 0:
            subgrad[i] = 2.0 * np.dot(a[:end], pp[i:i + end])

    return g_val, subgrad


def F_and_grad(a, beta):
    """Compute F(a) = 2n*smooth_max(conv(a,a))/(sum(a))^2 and gradient."""
    n = len(a)
    s = float(a.sum())
    if s < 1e-10:
        return 1e10, np.zeros(n)
    g_val, subgrad_g = smooth_max_and_subgrad(a, beta)
    F_val = 2.0 * n * g_val / (s * s)
    grad_F = 2.0 * n * (s * subgrad_g - 2.0 * g_val) / (s ** 3)
    return F_val, grad_F


def lbfgs_smooth(a0, beta_schedule, deadline):
    """L-BFGS-B with beta annealing on the smoothed objective."""
    a = np.maximum(a0.astype(np.float64), 1e-8)
    n = len(a)

    for beta, max_iters in beta_schedule:
        if time.monotonic() > deadline:
            break

        def func(x):
            x2 = np.maximum(x, 0.0)
            f, _ = F_and_grad(x2, beta)
            return f if np.isfinite(f) else 1e10

        def jac(x):
            x2 = np.maximum(x, 0.0)
            _, g = F_and_grad(x2, beta)
            g = np.nan_to_num(g, nan=0.0, posinf=0.0, neginf=0.0)
            g[x2 < 1e-12] = 0.0
            return g

        bounds = [(0.0, None)] * n
        try:
            res = minimize(func, a.copy(), jac=jac, method='L-BFGS-B',
                           bounds=bounds,
                           options={'maxiter': max_iters,
                                    'max_fun': max_iters * 25,
                                    'gtol': 1e-8, 'ftol': 1e-14})
            if res.fun < 1e10:
                new_a = np.maximum(res.x, 1e-10)
                if new_a.sum() > 1e-6:
                    a = new_a
        except Exception:
            pass
    return a


def backtracking_line_search(a, direction, current_J, max_alpha=2.0,
                             min_alpha=1e-4, c1=1e-4, rho=0.5):
    """
    Backtracking line search along direction = g - a.
    Tries to find alpha that gives sufficient decrease.
    Returns (best_a, best_J, best_alpha).
    """
    best_a = a.copy()
    best_J = current_J
    best_alpha = 0.0

    # Try alpha = 1 first
    a_try = a + 1.0 * direction
    a_try = np.maximum(a_try, 0)
    s = a_try.sum()
    if s < 1e-10:
        return best_a, best_J, 0.0
    a_try = a_try / s
    J_try = compute_J(a_try)

    if J_try < best_J:
        best_J = J_try
        best_a = a_try.copy()
        best_alpha = 1.0
        # Try extrapolating
        for alpha_try in [1.5, 2.0]:
            a_ext = a + alpha_try * direction
            a_ext = np.maximum(a_ext, 0)
            s_ext = a_ext.sum()
            if s_ext < 1e-10:
                break
            a_ext = a_ext / s_ext
            J_ext = compute_J(a_ext)
            if J_ext < best_J:
                best_J = J_ext
                best_a = a_ext.copy()
                best_alpha = alpha_try
            else:
                break
        return best_a, best_J, best_alpha

    # Backtrack
    alpha = 0.5
    while alpha > min_alpha:
        a_try = a + alpha * direction
        a_try = np.maximum(a_try, 0)
        s = a_try.sum()
        if s < 1e-10:
            alpha *= rho
            continue
        a_try = a_try / s
        J_try = compute_J(a_try)
        if J_try < best_J:
            best_J = J_try
            best_a = a_try.copy()
            best_alpha = alpha
            # Try slightly larger
            alpha_new = min(alpha * 1.3, max_alpha)
            a_try2 = a + alpha_new * direction
            a_try2 = np.maximum(a_try2, 0)
            s2 = a_try2.sum()
            if s2 > 1e-10:
                a_try2 = a_try2 / s2
                J_try2 = compute_J(a_try2)
                if J_try2 < best_J:
                    best_J = J_try2
                    best_a = a_try2.copy()
                    best_alpha = alpha_new
            return best_a, best_J, best_alpha
        alpha *= rho

    return best_a, best_J, best_alpha

def resample_seq(src, dest_n):
    """Resample 1D sequence to new length via linear interpolation."""
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    result = np.interp(t_new, t_old, src)
    return np.maximum(result, 0.0)


def generate_diverse_inits(n, rng, parent=None):
    """Generate diverse initial sequences."""
    inits = []
    t = np.linspace(0, 1, n, endpoint=False)

    if parent is not None:
        p = resample_seq(parent, n)
        inits.append(p)
        inits.append(np.maximum(p + rng.normal(0, 0.05, n) * p, 0.01))
        inits.append(np.maximum(p + rng.normal(0, 0.1, n) * p, 0.01))

    inits.append(np.ones(n))
    inits.append(rng.uniform(0.3, 1.0, n))

    for freq in [2, 3, 4, 5, 8]:
        inits.append(1.0 + 0.3 * np.cos(2.0 * np.pi * freq * t))

    inits.append(np.exp(-3.0 * (t - 0.5) ** 2) + 0.05)
    inits.append(np.abs(np.sin(np.pi * t)) + 0.1)
    inits.append(1.0 + 0.25 * np.cos(4 * np.pi * t) + 0.1 * np.cos(9 * np.pi * t))
    inits.append((t + 0.1) ** 0.3)
    inits.append(np.exp(-2.0 * t) + 0.15)
    inits.append(1.0 - 0.25 * np.abs(2.0 * (t % 0.5) - 0.5))
    # Anti-symmetric perturbation (helps break symmetry)
    asym = np.ones(n)
    asym[:n//2] += 0.1
    asym[n//2:] -= 0.1
    inits.append(asym)

    return inits

def targeted_perturbation(a, rng, strategy):
    """Generate targeted perturbations to reduce convolution peaks."""
    n = len(a)
    b = fast_conv(a)
    b_max = b.max()
    M = b_max

    new_a = a.copy()

    if strategy == 'reduce_peak_contributors':
        # Find top 3 peak positions
        peak_positions = np.argsort(b)[-3:]
        for k in peak_positions:
            if b[k] < M * 0.9:
                continue
            jl = max(0, k - n + 1)
            jh = min(n - 1, k)
            # Find the largest contributing pair
            contribs = []
            for j in range(jl, jh + 1):
                other = k - j
                if 0 <= other < n:
                    contribs.append((a[j] * a[other], j, other))
            if contribs:
                contribs.sort(reverse=True)
                # Reduce the largest contributor
                _, j_big, o_big = contribs[0]
                new_a[j_big] *= rng.uniform(0.7, 0.9)
                if o_big != j_big:
                    new_a[o_big] *= rng.uniform(0.85, 0.95)

    elif strategy == 'anti_peak':
        # Reduce all values that contribute to peaks proportionally
        peak_positions = np.where(b > M * 0.85)[0]
        for k in peak_positions[:10]:
            jl = max(0, k - n + 1)
            jh = min(n - 1, k)
            for j in range(jl, jh + 1):
                other = k - j
                if 0 <= other < n:
                    weight = a[j] * a[other] / (M + 1e-10)
                    new_a[j] -= 0.02 * weight * a[j]

    elif strategy == 'spread':
        # Spread mass from high to low values
        mean_a = a.mean()
        high_mask = a > 1.2 * mean_a
        low_mask = a < 0.8 * mean_a
        n_high = high_mask.sum()
        n_low = low_mask.sum()
        if n_high > 0 and n_low > 0:
            transfer = 0.05 * a[high_mask].sum() / n_low
            new_a[high_mask] -= 0.05 * a[high_mask]
            new_a[low_mask] += transfer

    elif strategy == 'random_spike':
        k = min(5, n // 4)
        idx = rng.choice(n, size=k, replace=False)
        new_a[idx] *= rng.uniform(0.5, 2.0, k)

    elif strategy == 'smooth_noise':
        noise = rng.normal(0, 0.06, n)
        kernel_size = max(3, n // 64)
        kernel = np.ones(kernel_size) / kernel_size
        noise = np.convolve(noise, kernel, mode='same')
        new_a = new_a + noise * new_a

    new_a = np.maximum(new_a, 1e-6)
    new_a = new_a / new_a.sum()
    return new_a

            budget_here = min(30.0, p3_deadline - time.monotonic())
            if budget_here > 5:
                a_ref = lp_mirror_descent_enhanced(
                    a_pert, time.monotonic() + budget_here, tracker,
                    max_iters=5000, momentum_beta=0.3,
                    aggressive_every=30, grad_hybrid_every=20, stall_patience=30
                )

        print(f"  {attempt} targeted attempts done", flush=True)
        tracker.report(f"Phase 3 done ({time.monotonic()-t0:.1f}s)")

    # === Phase 4: Multi-restart with diverse perturbations ===
    if tracker.best is not None and time.monotonic() < deadline:
        t0 = time.monotonic()
        p4_deadline = min(time.monotonic() + 250.0, deadline)
        print(f"\n{'='*50}\nPhase 4: Multi-Restart\n{'='*50}", flush=True)

        restart_configs = [
            ('gaussian', 0.03), ('gaussian', 0.06), ('gaussian', 0.12),
            ('uniform', 0.08), ('uniform', 0.15),
            ('gaussian', 0.02), ('smooth', 0.1), ('spike_rand', 10),
            ('gaussian', 0.05), ('uniform', 0.12),
        ]

        attempt = 0
        for pert_type, param in restart_configs:
            if time.monotonic() > p4_deadline:
                break
            attempt += 1
            a = tracker.best.copy().astype(np.float64)
            n = len(a)

            if pert_type == 'gaussian':
                a = a + rng.normal(0, param, n) * a
            elif pert_type == 'uniform':
                a = a + rng.uniform(-param, param, n) * a
            elif pert_type == 'smooth':
                noise = rng.normal(0, param, n)
                ks = max(3, n // 32)
                kern = np.ones(ks) / ks
                noise = np.convolve(noise, kern, mode='same')
                a = a + noise * a
            elif pert_type == 'spike_rand':
                k = min(param, n // 3)
                idx = rng.choice(n, size=k, replace=False)
                a[idx] *= rng.uniform(0.4, 2.5, k)

            a = np.maximum(a, 1e-5)
            a = a / a.sum()
            J_pert = compute_J(a)
            if J_pert < tracker.best_score:
                tracker.update(a)

            budget_here = min(25.0, p4_deadline - time.monotonic())
            if budget_here > 3:
                lp_mirror_descent_enhanced(
                    a, time.monotonic() + budget_here, tracker,
                    max_iters=3000, momentum_beta=0.3,
                    aggressive_every=40, grad_hybrid_every=25, stall_patience=40
                )

        print(f"  {attempt} restarts done", flush=True)
        tracker.report(f"Phase 4 done ({time.monotonic()-t0:.1f}s)")

    # === Phase 5: Final polish ===
    if tracker.best is not None and time.monotonic() < deadline:
        t0 = time.monotonic()
        remaining = deadline - time.monotonic()
        print(f"\n{'='*50}\nPhase 5: Final Polish ({remaining:.0f}s remaining)\n{'='*50}", flush=True)

        # Extended LP run on best
        a = lp_mirror_descent_enhanced(
            tracker.best, deadline, tracker,
            max_iters=100000, momentum_beta=0.2,
            aggressive_every=30, grad_hybrid_every=20, stall_patience=80
        )

        # Try a few different lengths near the best
        n_best = tracker.best.size
        for n_try in [n_best - 32, n_best + 32, n_best + 64]:
            if time.monotonic() > deadline or n_try < 64:
                continue
            a_res = resample_seq(tracker.best, n_try)
            a_res = a_res / max(a_res.sum(), 1e-10)
            budget_here = min(30.0, deadline - time.monotonic())
            if budget_here > 5:
                lp_mirror_descent_enhanced(
                    a_res, time.monotonic() + budget_here, tracker,
                    max_iters=5000, momentum_beta=0.3,
                    aggressive_every=40, grad_hybrid_every=30, stall_patience=40
                )

        tracker.report(f"Phase 5 done ({time.monotonic()-t0:.1f}s)")

    # Final validation
    if tracker.best is None:
        tracker.best = np.ones(512)
        tracker.best_score = compute_J(tracker.best)
        print(f"  Fallback: J={tracker.best_score:.8f}", flush=True)

    final_score = float(evaluate_sequence(tracker.best.tolist()))
    print(f"\n{'='*50}\nFINAL: J={final_score:.8f} n={tracker.best.size} "
          f"time={time.monotonic()-_start_time:.1f}s\n{'='*50}", flush=True)

    result = np.clip(tracker.best, 0.0, 1000.0)
    if result.sum() < 0.01:
        result = np.ones(128)
    return [float(x) for x in result]
