def coordinate_descent(a, time_budget=30.0, max_sweeps=50, verbose=False):
    n = len(a)
    a = a.copy().astype(np.float64)
    s = np.sum(a)
    if s < 1e-10:
        return a
    a = a / s
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    
    sweep = 0
    while sweep < max_sweeps:
        if time.monotonic() - start > time_budget - 2:
            break
        
        improved = False
        for i in range(n):
            if time.monotonic() - start > time_budget - 2:
                break
            
            # Optimize coordinate i
            old_val = a[i]
            if old_val < 1e-15:
                # Try setting to small positive
                a[i] = 1e-10
                obj = compute_obj(a)
                if obj < best_obj:
                    best_obj = obj
                    best_a = a.copy()
                    improved = True
                else:
                    a[i] = 0
                continue
            
            # Golden section search for optimal a[i]
            # The function is f(x) = 2n * max_k (a*a)[k] / (sum a)^2
            # as a function of x = a[i]
            
            def func(x):
                if x < 0:
                    return np.inf
                a[i] = x
                obj = compute_obj(a)
                return obj
            
            # Search range: [0, 2*old_val]
            res = minimize_scalar(func, bounds=(0, 3 * old_val), method='bounded',
                                  options={'xatol': 1e-10, 'maxiter': 50})
            if res.success and res.fun < a[i] * 0 + best_obj:
                a[i] = res.x
                s_cur = np.sum(a)
                if s_cur > 1e-10:
                    a = a / s_cur
                obj = compute_obj(a)
                if obj < best_obj:
                    best_obj = obj
                    best_a = a.copy()
                    improved = True
        
        sweep += 1
        if not improved:
            break
    
    return best_a

def multiplicative_subgradient_step(a, tau, step):
    n = len(a)
    conv = compute_conv(a)
    max_conv = np.max(conv)
    # Find positions where conv is close to max
    threshold = max_conv * 0.95
    active = conv[conv >= threshold]
    shifted = tau * (conv - max_conv)
    shifted = np.clip(shifted, -500, 500)
    weights = np.exp(shifted)
    w_sum = np.sum(weights)
    if w_sum < 1e-300:
        idx = int(np.argmax(conv))
        weights = np.zeros(len(conv))
        weights[idx] = 1.0
    else:
        weights /= w_sum
    c = np.correlate(weights, a, mode='full')
    grad = 2.0 * c[n - 1:n - 1 + n]
    log_update = -step * grad
    log_update = np.clip(log_update, -50, 50)
    a_new = a * np.exp(log_update)
    a_new = np.maximum(a_new, 1e-15)
    s = np.sum(a_new)
    if s > 1e-10:
        a_new = a_new / s
    return a_new

def resample_sequence(a, new_n):
    n = len(a)
    if n == new_n:
        return a.copy()
    old_k = np.arange(n)
    new_k = np.linspace(0, n - 1, new_n)
    a_new = np.interp(new_k, old_k, a)
    a_new = np.maximum(a_new, 0)
    s = np.sum(a_new)
    if s > 1e-10:
        a_new = a_new / s
    return a_new

"""
Step function optimization using alternating LP, fast coordinate descent,
and multiplicative subgradient methods. Key improvements:
- Fast coordinate descent using cross-term decomposition (no FFT needed per evaluation)
- More aggressive alternating LP with adaptive step sizes
- Systematic exploration of n values around parent
- Peak-aware coordinate ordering for faster convergence
- Better time management with adaptive budget allocation

import numpy as np
from scipy.optimize import linprog
import time


def compute_conv(a):
    """Compute conv(a,a) using FFT. Returns array of length 2n-1."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    conv = np.fft.irfft(fa * fa, size)
    return conv[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    conv = compute_conv(a)
    max_b = float(np.max(conv))
    sum_a = float(np.sum(a))
    if sum_a < 0.01:
        return np.inf
    return 2.0 * n * max_b / (sum_a * sum_a)


def build_toeplitz(a):
    """Build Toeplitz matrix A where (A@g)[k] = (a*g)[k] for k=0..2n-2."""
    n = len(a)
    A = np.zeros((2 * n - 1, n), dtype=np.float64)
    for j in range(n):
        k_start = j
        k_end = min(j + n, 2 * n - 1)
        A[k_start:k_end, j] = a[:k_end - k_start]
    return A


def solve_lp_maxsum(a, M, time_limit=8):
    """Solve LP: max sum(g) s.t. (a*g)[k] <= M for all k, g >= 0."""
    n = len(a)
    A = build_toeplitz(a)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': time_limit})
        if res.status == 0:
            g = np.maximum(res.x, 0)
            return g, float(np.sum(g))
    except Exception:
        pass
    return None, None


def alternating_lp(a, max_iters=100, time_budget=60.0, verbose=False):
    """Alternating LP with line search."""
    n = len(a)
    a = a.copy().astype(np.float64)
    s = np.sum(a)
    if s < 1e-10:
        return a
    a = a / s
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    stall = 0

    for it in range(max_iters):
        if time.monotonic() - start > time_budget - 3:
            break

        conv = compute_conv(a)
        M = float(np.max(conv))

        g, sum_g = solve_lp_maxsum(a, M, time_limit=6)
        if g is None or sum_g < 1e-10:
            break

        g_norm = g / sum_g

        # Line search with more options
        best_t = 0.0
        best_obj_ls = best_obj
        for t in [0.2, 0.4, 0.6, 0.8, 0.9, 1.0]:
            a_new = (1.0 - t) * a + t * g_norm
            a_new = np.maximum(a_new, 0)
            s_new = np.sum(a_new)
            if s_new < 1e-10:
                continue
            a_new = a_new / s_new
            obj_new = compute_obj(a_new)
            if obj_new < best_obj_ls:
                best_obj_ls = obj_new
                best_t = t

        if best_t > 0 and best_obj_ls < best_obj - 1e-14:
            a_new = (1.0 - best_t) * a + best_t * g_norm
            a_new = np.maximum(a_new, 0)
            s_new = np.sum(a_new)
            if s_new > 1e-10:
                a_new = a_new / s_new
            best_obj = best_obj_ls
            best_a = a_new.copy()
            a = a_new
            stall = 0
            if verbose and it % 15 == 0:
                print(f"    LP iter {it}: obj={best_obj:.8f}")
        else:
            stall += 1
            if stall > 35:
                break
            a = 0.9 * a + 0.1 * g_norm
            a = np.maximum(a, 0)
            s = np.sum(a)
            if s > 1e-10:
                a = a / s

    return best_a


def fast_coordinate_descent(a, time_budget=30.0, max_sweeps=100, verbose=False):
    """
    Fast coordinate descent using cross-term decomposition.
    For each coordinate i, the change in conv[k] when a[i] changes by delta is:
    - 2*delta*a[k-i] for k where 0<=k-i<n and k!=2i
    - 2*a[i]*delta + delta^2 for k=2i
    This avoids FFT calls during the 1D search.
    """
    n = len(a)
    a = a.copy().astype(np.float64)
    s = np.sum(a)
    if s < 1e-10:
        return a
    a = a / s
    conv = compute_conv(a)
    best_a = a.copy()
    best_obj = 2.0 * n * np.max(conv) / (s * s)
    start = time.monotonic()

    total_len = 2 * n - 1

    for sweep in range(max_sweeps):
        if time.monotonic() - start > time_budget - 2:
            break

        # Order coordinates by their contribution to peak positions
        # (do peak-contributing coordinates first for faster convergence)
        max_pos = np.argmax(conv)
        # Coordinates that contribute to the peak
        peak_contrib = []
        for i in range(n):
            if 0 <= max_pos - i < n:
                peak_contrib.append(a[i])
        peak_contrib.sort(reverse=True)
        # Sort indices: high contributors to peak first, then rest
        order = list(range(n))
        order.sort(key=lambda i: (-a[i] if (0 <= max_pos - i < n) else 0, i))

        improved = False
        for i in order:
            if time.monotonic() - start > time_budget - 2:
                break

            old_val = a[i]
            if old_val < 1e-15:
                continue

            # Precompute cross terms for coordinate i
            # cross[k] = a[k-i] if 0 <= k-i < n else 0
            k_min = max(0, i)  # min k such that k-i >= 0
            k_max = min(total_len - 1, i + n - 1)  # max k such that k-i < n
            cross = np.zeros(total_len)
            cross[k_min:k_max + 1] = a[:k_max + 1 - k_min]

            # For k = 2i, the term is special: 2*old_val*delta + delta^2
            k_special = 2 * i
            has_special = (0 <= k_special < total_len)

            # Search for optimal delta in range [-0.9*old_val, 2*old_val]
            # Use golden section search
            gr = (np.sqrt(5) - 1) / 2  # golden ratio ~0.618
            lo = -0.9 * old_val
            hi = 2.0 * old_val
            x1 = hi - gr * (hi - lo)
            x2 = lo + gr * (hi - lo)

            def eval_at(delta):
                """Evaluate objective at a[i] + delta."""
                new_val = old_val + delta
                if new_val < 0:
                    return np.inf
                # Compute new max of conv
                # conv_new[k] = conv[k] + 2*delta*cross[k] for most k
                # conv_new[2i] = conv[2i] + 2*old_val*delta + delta^2
                new_conv_max = np.max(conv + 2 * delta * cross)
                if has_special:
                    special_val = conv[k_special] - 2 * delta * cross[k_special] + 2 * old_val * delta + delta * delta
                    # Wait, this double-counts. Let me redo.
                    # conv[k_special] already includes old_val * a[k_special - i] for the l=i term
                    # and a[k_special - i] * old_val for the k-l=i term (if k_special-i != i)
                    # Actually conv[2i] includes a[i]*a[i] = old_val^2 from the l=i, k-l=i term
                    # When we add 2*delta*cross[2i] = 2*delta*a[i] = 2*delta*old_val
                    # That accounts for the cross terms: (old_val+delta)*a[0... no
                    #
                    # Let me be more careful.
                    # conv[k] = sum_l a[l]*a[k-l]
                    # The terms involving a[i] are:
                    #   l=i: a[i]*a[k-i] = old_val * a[k-i]
                    #   k-l=i: a[k-i]*a[i] = a[k-i]*old_val (same as above if k-i != i)
                    #   l=i AND k-l=i (k=2i): a[i]*a[i] = old_val^2 (counted once)
                    #
                    # So conv[k] = rest[k] + old_val * a[k-i] + a[k-i] * old_val (if k-i valid, k!=2i)
                    #            = rest[k] + 2*old_val*a[k-i] (if k!=2i, k-i valid)
                    #            = rest[k] + old_val^2 (if k=2i)
                    # where rest[k] = conv[k] - contribution from a[i]
                    #
                    # new_conv[k] = rest[k] + new_val * a[k-i] + a[k-i] * new_val (if k!=2i)
                    #            = rest[k] + 2*new_val*a[k-i]
                    #            = conv[k] + 2*delta*a[k-i]
                    # new_conv[2i] = rest[2i] + new_val^2
                    #            = conv[2i] - old_val^2 + (old_val+delta)^2
                    #            = conv[2i] + 2*old_val*delta + delta^2
                    #
                    # BUT: cross[2i] = a[2i-i] = a[i] = old_val
                    # So conv[2i] + 2*delta*cross[2i] = conv[2i] + 2*delta*old_val
                    # And the correct value is conv[2i] + 2*old_val*delta + delta^2
                    # So we need to add delta^2 extra at position 2i.
                    #
                    # For the max computation:
                    # new_conv[k] for k != 2i: conv[k] + 2*delta*cross[k]
                    # new_conv[2i]: conv[2i] + 2*old_val*delta + delta^2

                    # The np.max(conv + 2*delta*cross) already computed the wrong value at 2i
                    # It computed conv[2i] + 2*delta*old_val instead of conv[2i] + 2*old_val*delta + delta^2
                    # The difference is delta^2
                    # So the true max is max(np.max(conv + 2*delta*cross) adjusted, ...)
                    # At position 2i, the true value is one delta^2 more than what we computed
                    # But other positions might be the max, so we need:
                    max_without_special = np.max(np.where(np.arange(total_len) == k_special,
                                                         -np.inf, conv + 2 * delta * cross))
                    true_special = conv[k_special] + 2 * old_val * delta + delta * delta
                    new_conv_max = max(max_without_special, true_special)

                new_sum = s + delta
                if new_sum <= 0:
                    return np.inf
                return 2.0 * n * new_conv_max / (new_sum * new_sum)

            f1 = eval_at(x1)
            f2 = eval_at(x2)

            for _ in range(30):
                if f1 < f2:
                    hi = x2
                    x2 = x1
                    f2 = f1
                    x1 = hi - gr * (hi - lo)
                    f1 = eval_at(x1)
                else:
                    lo = x1
                    x1 = x2
                    f1 = f2
                    x2 = lo + gr * (hi - lo)
                    f2 = eval_at(x2)

            best_delta = (lo + hi) / 2.0
            new_val = old_val + best_delta
            if new_val < 0:
                new_val = 1e-15

            if abs(best_delta) > 1e-12:
                a[i] = new_val
                s += best_delta
                if s > 1e-10:
                    a = a / s
                    s = 1.0
                    # Update conv
                    conv = compute_conv(a)
                    obj = 2.0 * n * np.max(conv)
                    if obj < best_obj - 1e-15:
                        best_obj = obj
                        best_a = a.copy()
                        improved = True
                else:
                    a[i] = 0
            else:
                a[i] = old_val  # no change
                if s > 1e-10:
                    pass  # s unchanged since delta ~ 0

        if verbose and sweep % 5 == 0:
            print(f"    CD sweep {sweep}: obj={best_obj:.8f}")
        if not improved and sweep > 3:
            # Try a broader perturbation
            if time.monotonic() - start > time_budget - 5:
                break
            rng = np.random.default_rng(sweep * 7919)
            a = a * np.exp(rng.normal(0, 0.01, n))
            a = np.maximum(a, 1e-10)
            s = np.sum(a)
            if s > 1e-10:
                a = a / s
                s = 1.0
            conv = compute_conv(a)

    return best_a


def multiplicative_subgradient_step(a, tau, step):
    """One multiplicative subgradient step."""
    n = len(a)
    conv = compute_conv(a)
    max_conv = np.max(conv)
    shifted = tau * (conv - max_conv)
    shifted = np.clip(shifted, -500, 500)
    weights = np.exp(shifted)
    w_sum = np.sum(weights)
    if w_sum < 1e-300:
        idx = int(np.argmax(conv))
        weights = np.zeros(len(conv))
        weights[idx] = 1.0
    else:
        weights /= w_sum
    c = np.correlate(weights, a, mode='full')
    grad = 2.0 * c[n - 1:n - 1 + n]
    log_update = -step * grad
    log_update = np.clip(log_update, -50, 50)
    a_new = a * np.exp(log_update)
    a_new = np.maximum(a_new, 1e-15)
    s = np.sum(a_new)
    if s > 1e-10:
        a_new = a_new / s
    return a_new


def subgradient_optimize(a, max_iters=5000, time_budget=15.0, tau=20.0, step=0.05):
    """Subgradient optimization with adaptive parameters."""
    n = len(a)
    a = a.copy().astype(np.float64)
    s = np.sum(a)
    if s < 1e-10:
        return a
    a = a / s
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    stall = 0

    for it in range(max_iters):
        if time.monotonic() - start > time_budget:
            break
        a = multiplicative_subgradient_step(a, tau, step)
        obj = compute_obj(a)
        if obj < best_obj - 1e-13:
            best_obj = obj
            best_a = a.copy()
            stall = 0
        else:
            stall += 1
            if stall > 50:
                step = max(step * 0.7, 0.002)
                if step < 0.003:
                    break
                rng = np.random.default_rng(it * 31 + 7)
                a = a * np.exp(rng.normal(0, 0.02, n))
                a = np.maximum(a, 1e-10)
                s = np.sum(a)
                if s > 1e-10:
                    a = a / s
                stall = 0
        if it > 300:
            step *= 0.998
            tau *= 1.003
    return best_a


def resample_sequence(a, new_n):
    """Resample a sequence to a new length using interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    old_k = np.arange(n)
    new_k = np.linspace(0, n - 1, new_n)
    a_new = np.interp(new_k, old_k, a)
    a_new = np.maximum(a_new, 0)
    s = np.sum(a_new)
    if s > 1e-10:
        a_new = a_new / s
    return a_new


def optimize_single(a, time_budget=60.0, verbose=False):
    """Run full optimization pipeline on a single sequence."""
    n = len(a)
    a = a.copy().astype(np.float64)
    s = np.sum(a)
    if s < 1e-10:
        a = np.ones(n)
        s = n
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    if verbose:
        print(f"  Start: obj={best_obj:.8f}")

    # Phase 1: Quick subgradient
    remaining = time_budget - (time.monotonic() - start)
    if remaining > 5:
        sg_time = min(5.0, remaining * 0.1)
        a_new = subgradient_optimize(best_a, max_iters=int(sg_time * 40),
                                      time_budget=sg_time, tau=15.0, step=0.06)
        obj = compute_obj(a_new)
        if obj < best_obj:
            best_obj = obj
            best_a = a_new.copy()
            if verbose:
                print(f"  After SG: {best_obj:.8f}")

    # Phase 2: Alternating LP (main workhorse)
    remaining = time_budget - (time.monotonic() - start)
    if remaining > 10:
        lp_time = min(remaining * 0.35, 35.0)
        a_new = alternating_lp(best_a, max_iters=80, time_budget=lp_time, verbose=verbose)
        obj = compute_obj(a_new)
        if obj < best_obj:
            best_obj = obj
            best_a = a_new.copy()
            if verbose:
                print(f"  After LP: {best_obj:.8f}")

    # Phase 3: Coordinate descent (fast fine-tuning)
    remaining = time_budget - (time.monotonic() - start)
    if remaining > 8:
        cd_time = min(remaining * 0.2, 20.0)
        a_new = fast_coordinate_descent(best_a, time_budget=cd_time, max_sweeps=80)
        obj = compute_obj(a_new)
        if obj < best_obj:
            best_obj = obj
            best_a = a_new.copy()
            if verbose:
                print(f"  After CD: {best_obj:.8f}")

    # Phase 4: Another LP round
    remaining = time_budget - (time.monotonic() - start)
    if remaining > 10:
        lp_time = min(remaining * 0.3, 25.0)
        a_new = alternating_lp(best_a, max_iters=60, time_budget=lp_time)
        obj = compute_obj(a_new)
        if obj < best_obj:
            best_obj = obj
            best_a = a_new.copy()
            if verbose:
                print(f"  After LP2: {best_obj:.8f}")

    # Phase 5: Final coordinate descent
    remaining = time_budget - (time.monotonic() - start)
    if remaining > 5:
        cd_time = min(remaining * 0.4, 15.0)
        a_new = fast_coordinate_descent(best_a, time_budget=cd_time, max_sweeps=60)
        obj = compute_obj(a_new)
        if obj < best_obj:
            best_obj = obj
            best_a = a_new.copy()
            if verbose:
                print(f"  After CD2: {best_obj:.8f}")

    return best_a, best_obj


def intensive_local_search(a, time_budget=60.0, verbose=False):
    """Intensive local search with perturbations and re-optimizations."""
    n = len(a)
    a = a.copy().astype(np.float64)
    s = np.sum(a)
    if s < 1e-10:
        return a, compute_obj(a)
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    if verbose:
        print(f"  Local search start: obj={best_obj:.8f}")

    attempt = 0
    while time.monotonic() - start < time_budget - 15:
        attempt += 1
        remaining = time_budget - (time.monotonic() - start)
        if remaining < 20:
            break

        rng = np.random.default_rng(attempt * 12345 + 6789)
        scale = 0.02 + 0.04 * (attempt % 4) / 4.0

        a_pert = best_a * np.exp(rng.normal(0, scale, n))
        a_pert = np.maximum(a_pert, 1e-10)
        s_pert = np.sum(a_pert)
        if s_pert < 1e-10:
            continue
        a_pert = a_pert / s_pert

        sub_time = min(20.0, remaining * 0.4)
        a_opt, obj = optimize_single(a_pert, time_budget=sub_time, verbose=False)

        if obj < best_obj - 1e-12:
            best_obj = obj
            best_a = a_opt.copy()
            if verbose:
                print(f"    Attempt {attempt}: {best_obj:.8f}")

    return best_a, best_obj


def propose_candidate(seed=42, budget_s=1500):
    """
    Main search function. Combines alternating LP, fast coordinate descent,
    and multiplicative subgradient to minimize the step function objective.
    """
    start_time = time.monotonic()
    deadline = start_time + float(budget_s) - 30.0

    best_seq = None
    best_score = np.inf

    # Load parent
    parent = None
    n_parent = 330
    try:
        parent_arr = np.asarray(height_sequence_1, dtype=np.float64)
        parent_arr = np.clip(parent_arr, 0, 1000.0)
        if parent_arr.size > 0 and np.isfinite(parent_arr).all():
            obj = compute_obj(parent_arr)
            print(f"Parent (n={parent_arr.size}): obj={obj:.6f}")
            if obj < best_score:
                best_score = obj
                best_seq = parent_arr.copy()
                n_parent = len(parent_arr)
            parent = parent_arr.copy()
    except NameError:
        print("No parent available, starting fresh.")
    except Exception as e:
        print(f"Could not load parent: {e}")

    print(f"Initial best: {best_score:.6f} (n={n_parent})")
    print(f"Time budget: {budget_s}s")

    # Phase 1: Intensive optimization of parent
    if best_seq is not None:
        print("\n=== Phase 1: Intensive optimization of parent ===")
        remaining = deadline - time.monotonic()
        p1_budget = min(remaining * 0.3, 350.0)
        if p1_budget > 30:
            a_opt, obj = optimize_single(best_seq, time_budget=p1_budget, verbose=True)
            if obj < best_score:
                best_score = obj
                best_seq = a_opt.copy()
                print(f"  After optimize: {best_score:.6f}")

        # Local search
        remaining = deadline - time.monotonic()
        if remaining > 30:
            ls_budget = min(remaining * 0.2, 180.0)
            a_opt, obj = intensive_local_search(best_seq, time_budget=ls_budget, verbose=True)
            if obj < best_score:
                best_score = obj
                best_seq = a_opt.copy()
                print(f"  After local search: {best_score:.6f}")

    # Phase 2: Try nearby n values
    print("\n=== Phase 2: Nearby n values ===")
    n_values = []
    for delta in range(-50, 51, 5):
        n_val = n_parent + delta
        if 200 <= n_val <= 600:
            n_values.append(n_val)
    n_values = sorted(set(n_values))

    remaining = deadline - time.monotonic()
    if remaining > 30 and parent is not None:
        per_n_budget = remaining / max(len(n_values), 1) * 1.1
        for n_val in n_values:
            if time.monotonic() > deadline - 30:
                break
            remaining = deadline - time.monotonic()
            if remaining < 20:
                break
            # Skip the parent's n (already optimized)
            if n_val == n_parent:
                continue
            alloc = min(per_n_budget, remaining * 0.2, 60.0)
            alloc = max(alloc, 15.0)

            a_init = resample_sequence(parent, n_val)
            a_opt, obj = optimize_single(a_init, time_budget=alloc, verbose=False)
            if obj < best_score:
                best_score = obj
                best_seq = a_opt.copy()
                print(f"  ** UPDATE: n={n_val}, obj={best_score:.6f} **")

    # Phase 3: Final intensive optimization on best
    if best_seq is not None:
        remaining = deadline - time.monotonic()
        if remaining > 40:
            print(f"\n=== Phase 3: Final optimization ({remaining:.0f}s remaining) ===")
            p3_budget = min(remaining * 0.4, remaining - 20.0)
            a_opt, obj = optimize_single(best_seq, time_budget=p3_budget, verbose=True)
            if obj < best_score:
                best_score = obj
                best_seq = a_opt.copy()
                print(f"  Final opt: {best_score:.6f}")

            remaining = deadline - time.monotonic()
            if remaining > 20:
                ls_budget = min(remaining * 0.5, remaining - 10.0)
                a_opt, obj = intensive_local_search(best_seq, time_budget=ls_budget, verbose=True)
                if obj < best_score:
                    best_score = obj
                    best_seq = a_opt.copy()
                    print(f"  Final LS: {best_score:.6f}")

            # One more round of CD
            remaining = deadline - time.monotonic()
            if remaining > 10:
                cd_time = min(remaining * 0.5, remaining - 5.0)
                a_new = fast_coordinate_descent(best_seq, time_budget=cd_time, max_sweeps=50)
                obj = compute_obj(a_new)
                if obj < best_score:
                    best_score = obj
                    best_seq = a_new.copy()
                    print(f"  Final CD: {best_score:.6f}")

    # Final validation
    if best_seq is None:
        best_seq = np.ones(330)
        best_score = compute_obj(best_seq)

    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    val = evaluate_sequence(result)
    if val < best_score:
        best_score = val

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(best_seq)}, score={best_score:.6f}, time={elapsed:.1f}s ===")

    return result
