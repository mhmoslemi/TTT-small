    # Objective: minimize M -> c = [0, ..., 0, 1]

def fast_conv(a):
    """Full self-convolution via FFT."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]


def compute_J(a):
    """Compute J(a) = 2*n*max(conv(a,a))/(sum(a))^2."""
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))


def smooth_max_gradient(a, tau):
    """Compute gradient of soft-max of conv(a,a) w.r.t. a."""
    n = len(a)
    b = fast_conv(a)
    b_max = b.max()
    exp_b = np.exp(tau * (b - b_max))
    p = exp_b / exp_b.sum()
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        pk = p[k]
        if pk < 1e-13:
            continue
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        grad[lo:hi + 1] += 2.0 * pk * a[k - hi:k - lo + 1][::-1]
    return grad, p, b


def multiplicative_descent(a, max_iters, deadline, tau_init=5.0, tau_final=120.0,
                           lr_init=0.08, lr_min=0.003):
    """Multiplicative gradient descent for smooth max of conv."""
    a = np.maximum(a, 1e-8).astype(np.float64)
    a = a / a.sum()
    n = len(a)
    best_a = a.copy()
    best_M = fast_conv(a).max()

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        frac = it / max(1, max_iters)
        tau = tau_init + (tau_final - tau_init) * (frac ** 0.7)
        lr = lr_init * (lr_min / lr_init) ** frac

        grad, p, b = smooth_max_gradient(a, tau)
        M = b.max()
        if M < best_M * (1 - 1e-10):
            best_M = M
            best_a = a.copy()

        log_update = -lr * grad
        log_update = np.clip(log_update, -3.0, 3.0)
        a = a * np.exp(log_update)
        a = np.maximum(a, 1e-10)
        a = a / a.sum()

    return best_a


def lp_best_response_g(a, M):
    """LP: max sum(g) s.t. (a*g)[k] <= M for all k, g >= 0. Returns g or None."""
    n = len(a)
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        j_min = max(0, k - n + 1)
        j_max = min(n - 1, k)
        for j in range(j_min, j_max + 1):
            rows.append(k)
            cols.append(j)
            vals.append(a[k - j])
    A = csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    try:
        r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                    options={'time_limit': 20})
        if r.success and r.x is not None:
            g = np.maximum(r.x, 0)
            if g.sum() > 1e-15:
                return g / g.sum()
    except Exception:
        pass
    return None

def alternating_lp(a, max_iters, deadline, m_offsets=(0.0, -0.005, -0.01, -0.02, -0.03)):
    """
    Alternating LP best-response:
    - Fix a, solve LP for g (max sum(g) s.t. (a*g)_k <= M)
    - Fix g, solve LP for a' (min M' s.t. (a'*g)_k <= M', sum(a')=1)
    - Update a via convex combination with line search
    """
    a = np.maximum(a.astype(np.float64), 1e-8)
    a = a / a.sum()
    n = len(a)
    best_a = a.copy()
    best_M = fast_conv(a).max()
    no_improve = 0

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        b = fast_conv(a)
        M = float(b.max())

        improved = False
        for offset in m_offsets:
            M_try = M * (1.0 + offset)
            if M_try <= 0:
                continue

            g = lp_best_response_g(a, M_try)
            if g is None:
                continue

            a_new, M_lp = lp_best_response_a(g)
            if a_new is None:
                continue

            # Line search: try different step sizes
            for t in [1.0, 0.7, 0.5, 0.3, 0.2, 0.1]:
                a_trial = np.maximum((1 - t) * a + t * a_new, 0)
                s = a_trial.sum()
                if s < 1e-10:
                    continue
                a_trial = a_trial / s
                M_trial = fast_conv(a_trial).max()
                if M_trial < best_M - 1e-12:
                    a = a_trial
                    best_M = M_trial
                    best_a = a.copy()
                    improved = True
                    break

            if improved:
                break

            no_improve += 1
            if no_improve > 25:

        else:
            no_improve = 0

def resample_seq(src, dest_n):
    """Resample 1D sequence to new length via linear interpolation."""
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    return np.maximum(np.interp(t_new, t_old, src), 1e-8)

    print(f"Phase 1 time: {time.monotonic()-t0:.1f}s", flush=True)

    t0 = time.monotonic()
    print(f"\n{'='*60}", flush=True)
    print(f"PHASE 2: Alternating LP Best-Response", flush=True)
    print(f"{'='*60}", flush=True)

    if tracker.best is not None:
        n_best = tracker.best.size
        # Estimate LP time per iteration
        est_lp_time = 0.005 * (n_best / 128) ** 2
        print(f"  n={n_best}, est LP iter time={est_lp_time*1000:.0f}ms", flush=True)

        # Run alternating LP with generous budget
        p2_deadline = min(time.monotonic() + 800, deadline - 200)
        a = alternating_lp(tracker.best, 5000, p2_deadline,
                           m_offsets=(0.0, -0.005, -0.01, -0.02, -0.03, -0.05))
        tracker.update(a)

        # Continue with one-sided LP if time remains
        if time.monotonic() < deadline - 100:
            a2 = one_sided_lp(a, 2000, min(time.monotonic() + 200, deadline - 100),
                              m_offsets=(0.0, -0.01, -0.02, -0.03))
            tracker.update(a2)

    tracker.report("Phase 2")
    print(f"Phase 2 time: {time.monotonic()-t0:.1f}s", flush=True)

    # === PHASE 3: Random restarts + LP polish ===
    t0 = time.monotonic()
    print(f"\n{'='*60}", flush=True)
    print(f"PHASE 3: Random Restarts + Polish", flush=True)
    print(f"{'='*60}", flush=True)

    if tracker.best is not None and time.monotonic() < deadline - 50:
        n_best = tracker.best.size
        base = tracker.best.copy()
        temps = [0.03, 0.02, 0.015, 0.01, 0.005, 0.003, 0.002, 0.001]

        for temp in temps:
            if time.monotonic() > deadline - 30:
                break

            # Generate perturbed version
            a_pert = base + rng.normal(0, temp * 0.1, n_best)
            a_pert = np.maximum(a_pert, 1e-4)
            a_pert = a_pert / a_pert.sum()

            # Quick gradient warm-up
            a_pert = multiplicative_descent(a_pert, 30, deadline - 30,
                                            tau_init=10, tau_final=60,
                                            lr_init=0.05, lr_min=0.01)
            tracker.update(a_pert)

            # Alternating LP polish
            a_pol = alternating_lp(a_pert, 200, min(time.monotonic() + 60, deadline - 30),
                                   m_offsets=(0.0, -0.01, -0.02))
            tracker.update(a_pol)

            # One-sided LP
            a_pol2 = one_sided_lp(a_pol, 100, min(time.monotonic() + 30, deadline - 30),
                                  m_offsets=(0.0, -0.01))
            tracker.update(a_pol2)

    tracker.report("Phase 3")
    print(f"Phase 3 time: {time.monotonic()-t0:.1f}s", flush=True)

    # === PHASE 4: Length extension ===
    if tracker.best is not None and time.monotonic() < deadline - 30:
        t0 = time.monotonic()
        print(f"\n{'='*60}", flush=True)
        print(f"PHASE 4: Length Extension", flush=True)
        print(f"{'='*60}", flush=True)

        base_n = tracker.best.size
        for mult in [2, 3]:
            if time.monotonic() > deadline - 15:
                break
            new_n = base_n * mult
            if new_n > 3072:
                continue
            a_ext = resample_seq(tracker.best, new_n)
            a_ext = a_ext / a_ext.sum()
            a_ext = multiplicative_descent(a_ext, 50, deadline - 15,
                                           tau_init=5, tau_final=50,
                                           lr_init=0.05, lr_min=0.01)
            tracker.update(a_ext)
            # Short LP
            a_ext2 = alternating_lp(a_ext, 50, deadline - 15,
                                    m_offsets=(0.0, -0.01))
            tracker.update(a_ext2)

        # Also try slightly different lengths
        for delta in [-64, -32, 32, 64]:
            if time.monotonic() > deadline - 15:
                break
            new_n = base_n + delta
            if new_n < 128 or new_n > 3072:
                continue
            a_ext = resample_seq(tracker.best, new_n)
            a_ext = a_ext / a_ext.sum()
            a_ext = multiplicative_descent(a_ext, 40, deadline - 15,
                                           tau_init=5, tau_final=50,
                                           lr_init=0.04, lr_min=0.01)
            tracker.update(a_ext)

        tracker.report("Phase 4")
        print(f"Phase 4 time: {time.monotonic()-t0:.1f}s", flush=True)

    # === FINAL: Extended LP on best ===
    if tracker.best is not None and time.monotonic() < deadline - 5:
        print(f"\nFinal LP polish...", flush=True)
        a_final = alternating_lp(tracker.best, 200, deadline,
                                 m_offsets=(0.0, -0.005, -0.01, -0.02))
        tracker.update(a_final)

    # Fallback
    if tracker.best is None:
        tracker.best = np.ones(128)
        tracker.best_score = compute_J(tracker.best)

    print(f"\n{'='*60}", flush=True)
    print(f"FINAL: J={tracker.best_score:.8f} n={tracker.best.size} "
          f"time={time.monotonic()-tracker.t_start:.0f}s", flush=True)
    print(f"{'='*60}", flush=True)

    return [float(x) for x in tracker.best]
