"""
Minimizes J(a) = 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Nesterov-accelerated smooth gradient descent (Phase 1)
2. Reduced-constraint LP mirror descent with constraint generation (Phase 2, workhorse)
3. Full LP polish + coordinate descent (Phase 3)
4. Peak-specific attacks with adaptive LP (Phase 4)
5. Final verification (Phase 5)

Key innovation: constraint generation LP - solves LPs with only the top-K
active constraints (~50) instead of all (2n-1), giving 10-20x speedup per
iteration. This allows 1000+ LP iterations in the 700s budget.
"""

import numpy as np
from scipy.optimize import linprog
from scipy.sparse import csr_matrix
import time


def fast_conv(a):
    """Full self-convolution via FFT."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]


def compute_J(a):
    """Compute J(a) = 2*n*max(conv(a,a))/(sum(a))^2."""
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))


def smooth_max_gradient(a, tau):
    """Gradient of smooth max (1/tau)*log(sum exp(tau*b_k)) w.r.t. a."""
    n = len(a)
    b = fast_conv(a)
    b_max = b.max()
    exp_b = np.exp(tau * (b - b_max))
    p = exp_b / exp_b.sum()
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        pk = p[k]
        if pk < 1e-13:
            continue
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        grad[lo:hi + 1] += 2.0 * pk * a[k - hi:k - lo + 1][::-1]
    return grad, p, b


def subgradient_exact(a, tol_frac=1e-10):
    """Exact subgradient of max(conv(a,a))."""
    n = len(a)
    b = fast_conv(a)
    M = b.max()
    tol = tol_frac * M if M > 0 else 1e-10
    active = np.where(b >= M - tol)[0]
    grad = np.zeros(n)
    for k in active:
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        grad[lo:hi + 1] += 2.0 * a[k - hi:k - lo + 1][::-1]
    gnorm = np.linalg.norm(grad)
    if gnorm > 1e-15:
        grad /= gnorm
    return grad, active, b, M


def run_smooth_gradient(a, max_iters, deadline, initial_tau=3.0, final_tau=150.0,
                        lr_init=0.2, lr_min=0.01):
    """Nesterov-accelerated smooth gradient descent."""
    a = np.maximum(a.astype(np.float64), 1e-8)
    a = a / a.sum()
    n = len(a)
    best_a = a.copy()
    best_M = fast_conv(a).max()
    y = a.copy()  # Nesterov momentum
    v = np.zeros(n)

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        frac = it / max(1, max_iters - 1)
        tau = initial_tau + (final_tau - initial_tau) * (frac ** 0.7)
        lr = lr_init * (lr_min / lr_init) ** frac

        grad_y, _, b_y = smooth_max_gradient(y, tau)
        M_y = b_y.max()
        # y update
        y_new = a + (v / (lr + 1e-15))  # This isn't quite right for Nesterov
        # Simplified: use heavy ball instead
        v = 0.9 * v + lr * grad_y
        a_new = a - v
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s < 1e-10:
            a_new = a.copy()
        else:
            a_new = a_new / s
        a = a_new

        M = fast_conv(a).max()
        if M < best_M * (1 - 1e-9):
            best_M = M
            best_a = a.copy()

    return best_a


def build_reduced_A(n, a, lag_indices):
    """Build sparse constraint matrix for only specified lag indices."""
    rows = []
    cols = []
    vals = []
    for idx, k in enumerate(lag_indices):
        j_min = max(0, k - n + 1)
        j_max = min(n - 1, k)
        for j in range(j_min, j_max + 1):
            rows.append(idx)
            cols.append(j)
            vals.append(a[k - j])
    A = csr_matrix((vals, (rows, cols)), shape=(len(lag_indices), n))
    return A


def reduced_lp_best_response(a, M, K=50, time_limit=10.0):
    """LP best response using only top-K lag constraints."""
    n = len(a)
    b = fast_conv(a)
    # Get top K lags
    if 2 * n - 1 <= K:
        lags = np.arange(2 * n - 1)
    else:
        lags = np.argsort(b)[-K:]
    lags = np.sort(lags)

    A = build_reduced_A(n, a, lags)
    c = -np.ones(n)
    b_ub = np.full(len(lags), M)
    bounds = [(0, None)] * n

    try:
        r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                    options={'time_limit': time_limit, 'presolve': True})
        if r.success and r.x is not None:
            g = np.maximum(r.x, 0)
            if g.sum() > 1e-15:
                return g, lags
    except Exception:
        pass
    return None, None


def full_lp_best_response(a, M, time_limit=30.0):
    """Full LP best response with all constraints."""
    n = len(a)
    lags = np.arange(2 * n - 1)
    A = build_reduced_A(n, a, lags)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n

    try:
        r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                    options={'time_limit': time_limit, 'presolve': True})
        if r.success and r.x is not None:
            g = np.maximum(r.x, 0)
            if g.sum() > 1e-15:
                return g
    except Exception:
        pass
    return None


def constraint_generation_lp(a, M, K_start=20, K_max=200, time_per_lp=5.0):
    """LP with constraint generation: start with few constraints, add violations."""
    n = len(a)
    b = fast_conv(a)
    lags = np.sort(np.argsort(b)[-K_start:])
    active_lags = set(lags.tolist())

    for iteration in range(20):  # Max constraint additions
        if len(active_lags) > K_max:
            break
        lags_arr = np.sort(list(active_lags))
        A = build_reduced_A(n, a, lags_arr)
        c = -np.ones(n)
        b_ub = np.full(len(lags_arr), M)
        bounds = [(0, None)] * n

        try:
            r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                        options={'time_limit': time_per_lp, 'presolve': True})
        except Exception:
            break

        if not r.success or r.x is None:
            break

        g = np.maximum(r.x, 0)
        if g.sum() < 1e-15:
            return None

        # Check for violated constraints
        b_check = fast_conv(a) * 0 + fast_conv(g)  # (a*g)[k]
        violations = np.where(b_check > M + 1e-10)[0]

        if len(violations) == 0:
            return g

        # Add violating constraints
        for v in violations:
            if len(active_lags) < K_max:
                active_lags.add(int(v))

    # If we still have violations, return the last solution (best effort)
    return g


def run_lp_mirror_descent(a, max_iters, deadline, use_constraint_gen=True,
                          K=50, lp_time_limit=5.0):
    """LP mirror descent with adaptive line search."""
    a = np.maximum(a.astype(np.float64), 1e-8)
    a = a / a.sum()
    best_a = a.copy()
    best_M = fast_conv(a).max()
    no_improve = 0

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        b = fast_conv(a)
        M = float(b.max())

        if use_constraint_gen:
            g = constraint_generation_lp(a, M, K_start=20, K_max=100,
                                         time_per_lp=lp_time_limit)
        else:
            g, _ = reduced_lp_best_response(a, M, K=K, time_limit=lp_time_limit)

        if g is None:
            break
        g = np.maximum(g, 0.0)
        g_sum = g.sum()
        if g_sum < 1e-10:
            break
        g /= g_sum

        # Adaptive line search
        improved = False
        base_t = max(0.03, 0.5 * (1.0 - 0.5 * it / max(1, max_iters)))
        for t in [base_t, base_t * 0.7, base_t * 0.5, base_t * 0.3, 0.01, 0.005]:
            a_trial = np.maximum((1 - t) * a + t * g, 0)
            s = a_trial.sum()
            if s < 1e-10:
                continue
            a_trial /= s
            M_trial = fast_conv(a_trial).max()
            if M_trial < best_M - 1e-14:
                a = a_trial
                best_M = M_trial
                best_a = a.copy()
                improved = True
                no_improve = 0
                break

        if not improved:
            no_improve += 1
            if no_improve > 50:
                # Try smaller steps
                for t in [0.001, 0.0005, 0.0001]:
                    a_trial = np.maximum((1 - t) * a + t * g, 0)
                    s = a_trial.sum()
                    if s < 1e-10:
                        continue
                    a_trial /= s
                    M_trial = fast_conv(a_trial).max()
                    if M_trial < best_M - 1e-14:
                        a = a_trial
                        best_M = M_trial
                        best_a = a.copy()
                        improved = True
                        no_improve = 0
                        break
                if not improved:
                    if no_improve > 100:
                        break

    return best_a


def run_subgradient(a, max_iters, deadline, lr_init=0.15):
    """Subgradient descent with averaged solution."""
    a = np.maximum(a.astype(np.float64), 1e-8)
    a = a / a.sum()
    best_a = a.copy()
    best_M = fast_conv(a).max()
    running_sum = np.zeros_like(a)
    count = 0

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        lr = lr_init / (it + 1) ** 0.5
        grad, active, b, M = subgradient_exact(a)

        if M < best_M * (1 - 1e-9):
            best_M = M
            best_a = a.copy()

        a_new = a - lr * grad
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s < 1e-10:
            a_new = a.copy()
        else:
            a_new = a_new / s
        running_sum += a_new
        count += 1
        a = a_new

    if count > 10:
        a_avg = running_sum / count
        a_avg = np.maximum(a_avg, 0)
        s = a_avg.sum()
        if s > 1e-10:
            a_avg /= s
            M_avg = fast_conv(a_avg).max()
            if M_avg < best_M:
                best_a = a_avg

    return best_a


def coordinate_descent(a, max_sweeps, deadline, n_trials=5):
    """Coordinate descent: for each coordinate, try several values."""
    a = np.maximum(a.astype(np.float64), 1e-8)
    a = a / a.sum()
    n = len(a)
    best_a = a.copy()
    best_M = fast_conv(a).max()
    rng = np.random.default_rng(12345)

    for sweep in range(max_sweeps):
        if time.monotonic() > deadline:
            break
        # Random order to avoid bias
        order = rng.permutation(n)
        for j in order:
            if time.monotonic() > deadline:
                break
            orig_val = a[j]
            if orig_val < 1e-8:
                continue
            # Try several multipliers
            best_local = orig_val
            best_local_M = best_M
            for mult in [0.5, 0.7, 0.85, 1.15, 1.3, 1.5]:
                a_trial = a.copy()
                a_trial[j] = orig_val * mult
                a_trial = np.maximum(a_trial, 0)
                s = a_trial.sum()
                if s < 1e-10:
                    continue
                a_trial /= s
                M_trial = fast_conv(a_trial).max()
                if M_trial < best_local_M - 1e-12:
                    best_local_M = M_trial
                    best_local = orig_val * mult
            a[j] = best_local
            s = a.sum()
            a /= s
            M_now = fast_conv(a).max()
            if M_now < best_M - 1e-12:
                best_M = M_now
                best_a = a.copy()

    return best_a


def peak_attack(a, deadline, n_rounds=200):
    """Directly attack the peak lag by reducing contributing elements."""
    a = np.maximum(a.astype(np.float64), 1e-8)
    a = a / a.sum()
    n = len(a)
    best_a = a.copy()
    best_M = fast_conv(a).max()

    for r in range(n_rounds):
        if time.monotonic() > deadline:
            break
        b = fast_conv(a)
        M = b.max()
        k_star = int(np.argmax(b))

        # Find the elements that contribute most to the peak
        lo = max(0, k_star - n + 1)
        hi = min(n - 1, k_star)
        partners = a[k_star - hi:k_star - lo + 1][::-1]
        # Weight by partner value
        contribution = a[lo:hi + 1] * partners

        # Reduce top contributors
        top_idx = np.argsort(contribution)[::-1][:3]
        for idx in top_idx:
            j = lo + int(idx)
            a[j] *= 0.95  # Reduce by 5%
        a = np.maximum(a, 1e-8)
        a /= a.sum()

        M_new = fast_conv(a).max()
        if M_new < best_M - 1e-12:
            best_M = M_new
            best_a = a.copy()

    return best_a


def resample_seq(src, dest_n):
    """Resample 1D sequence to new length."""
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)


def generate_diverse_inits(n, rng, parent):
    """Generate diverse initializations."""
    inits = []
    t_arr = np.linspace(0, 1, n, endpoint=False)

    if parent is not None:
        p = resample_seq(parent, n)
        inits.append(p.copy())
        inits.append(np.maximum(p + rng.normal(0, 0.02, n), 0.01))
        inits.append(np.maximum(p + rng.normal(0, 0.08, n), 0.01))
        inits.append(np.maximum(p * (1 + 0.1 * rng.standard_t(df=5, size=n)), 0.01))

    inits.append(np.ones(n))
    for freq in [1, 2, 3, 5]:
        inits.append(np.maximum(1.0 + 0.3 * np.cos(2 * np.pi * freq * t_arr), 0.01))

    # Flat spectrum initialization
    phases = rng.uniform(0, 2 * np.pi, n)
    flat_spec = np.ones(n)
    a_init = np.fft.irfft(flat_spec * np.exp(1j * phases), n)
    inits.append(np.maximum(a_init + a_init.max() + 0.1, 0.01))

    # Gaussian center
    inits.append(np.exp(-4.0 * (t_arr - 0.5) ** 2) + 0.1)

    # Random smooth
    ksize = max(3, n // 30)
    kernel = np.ones(ksize) / ksize
    smooth = np.convolve(rng.uniform(0, 1, n), kernel, mode='same')
    inits.append(np.maximum(smooth + 0.1, 0.01))

    return inits


class ScoreTracker:
    def __init__(self):
        self.best = None
        self.best_score = float('inf')
        self.t_start = time.monotonic()

    def update(self, a):
        a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
        if a.size == 0 or a.sum() < 0.01:
            return
        score = float(evaluate_sequence(a.tolist()))
        if np.isfinite(score) and score < self.best_score:
            self.best = a.copy()
            self.best_score = score
            elapsed = time.monotonic() - self.t_start
            print(f"  ** J={score:.8f} n={a.size} t={elapsed:.0f}s", flush=True)

    def report(self, label=""):
        if self.best is not None:
            print(f"  {label} best: J={self.best_score:.8f} n={self.best.size}", flush=True)

            print(f"  n={n:4d}: best smooth grad J={tracker.best_score:.8f}", flush=True)

    tracker.report("Phase 1 done")

    # === PHASE 2: Reduced LP Mirror Descent (700s) - MAIN ===
    p2_deadline = min(time.monotonic() + 700, deadline - 400)
    print(f"\n{'='*60}", flush=True)
    print(f"PHASE 2: Reduced LP Mirror Descent (budget={p2_deadline - time.monotonic():.0f}s)", flush=True)
    print(f"{'='*60}", flush=True)

    # Run constraint generation LP on the best
    a_opt = run_lp_mirror_descent(tracker.best, max_iters=5000,
                                  deadline=p2_deadline,
                                  use_constraint_gen=True, K=50, lp_time_limit=5.0)
    tracker.update(a_opt)

    # Run full LP on the result
    if time.monotonic() < p2_deadline - 50:
        a_opt2 = run_lp_mirror_descent(a_opt, max_iters=2000,
                                       deadline=p2_deadline - 50,
                                       use_constraint_gen=False, K=100, lp_time_limit=10.0)
        tracker.update(a_opt2)

    # Subgradient polish
    if time.monotonic() < p2_deadline - 20:
        a_sub = run_subgradient(tracker.best, 3000, p2_deadline - 20, lr_init=0.1)
        tracker.update(a_sub)

    tracker.report("Phase 2 done")

    # === PHASE 3: Coordinate descent + LP (250s) ===
    p3_deadline = min(time.monotonic() + 250, deadline - 150)
    print(f"\n{'='*60}", flush=True)
    print(f"PHASE 3: Coordinate Descent + LP (budget={p3_deadline - time.monotonic():.0f}s)", flush=True)
    print(f"{'='*60}", flush=True)

    a_cd = coordinate_descent(tracker.best, max_sweeps=50, deadline=p3_deadline - 60)
    tracker.update(a_cd)

    # LP polish after CD
    if time.monotonic() < p3_deadline - 30:
        a_lp = run_lp_mirror_descent(tracker.best, max_iters=1000,
                                     deadline=p3_deadline,
                                     use_constraint_gen=True, K=30, lp_time_limit=3.0)
        tracker.update(a_lp)

    tracker.report("Phase 3 done")

    # === PHASE 4: Peak attack + LP (150s) ===
    p4_deadline = min(time.monotonic() + 150, deadline - 50)
    print(f"\n{'='*60}", flush=True)
    print(f"PHASE 4: Peak Attack (budget={p4_deadline - time.monotonic():.0f}s)", flush=True)
    print(f"{'='*60}", flush=True)

    if time.monotonic() < p4_deadline:
        a_peak = peak_attack(tracker.best, deadline=p4_deadline - 50, n_rounds=500)
        tracker.update(a_peak)

        # LP after peak attack
        if time.monotonic() < p4_deadline - 10:
            a_final = run_lp_mirror_descent(tracker.best, max_iters=500,
                                            deadline=p4_deadline,
                                            use_constraint_gen=True, K=30, lp_time_limit=3.0)
            tracker.update(a_final)

    tracker.report("Phase 4 done")

    # === PHASE 5: Final multi-length check + last LP (remaining) ===
    print(f"\n{'='*60}", flush=True)
    print(f"PHASE 5: Final Polish (budget={deadline - time.monotonic():.0f}s)", flush=True)
    print(f"{'='*60}", flush=True)

    # Try different lengths
    base_n = tracker.best.size
    for mult in [2, 3]:
        new_n = base_n * mult
        if new_n > 2048 or time.monotonic() > deadline - 30:
            continue
        a_ext = resample_seq(tracker.best, new_n)
        a_ext = np.maximum(a_ext, 1e-6)
        a_ext = a_ext / a_ext.sum()
        a_ext = run_smooth_gradient(a_ext, 100, deadline - 30,
                                    initial_tau=5, final_tau=50, lr_init=0.1, lr_min=0.02)
        tracker.update(a_ext)

    # Final LP pass
    if time.monotonic() < deadline - 3:
        a = run_lp_mirror_descent(tracker.best, max_iters=200,
                                  deadline=deadline,
                                  use_constraint_gen=False, K=80, lp_time_limit=5.0)
        tracker.update(a)

    tracker.report("FINAL")

    # Fallback
    if tracker.best is None:
        tracker.best = np.ones(128)
        tracker.best_score = compute_J(tracker.best)

    print(f"\n{'='*60}", flush=True)
    print(f"RESULT: J={tracker.best_score:.8f} n={tracker.best.size} "
          f"total_time={time.monotonic()-tracker.t_start:.0f}s", flush=True)
    print(f"{'='*60}", flush=True)

    return [float(x) for x in tracker.best]
