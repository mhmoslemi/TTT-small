"""
Step function optimization using L-BFGS with log-sum-exp smoothing,
softmax parameterization, and coordinate descent polishing.
"""

import numpy as np
from scipy.optimize import minimize
import time

def compute_conv_fft(a):
    """Compute full self-convolution using FFT."""
    n = len(a)
    size = 1
    while size < 2*n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    conv = np.fft.irfft(fa * fa, size)
    return conv[:2*n-1]

def obj_val(a):
    """Compute the objective: 2n * max(conv(a,a)) / (sum(a))^2."""
    n = len(a)
    s = np.sum(a)
    if s < 1e-10:
        return np.inf
    c = compute_conv_fft(a)
    return 2.0 * n * np.max(c) / (s * s)

def softmax(x):
    x_shifted = x - np.max(x)
    exp_x = np.exp(x_shifted)
    return exp_x / np.sum(exp_x)

def smooth_obj_grad(x, t):
    """Smoothed objective and gradient for L-BFGS."""
    n = len(x)
    a = softmax(x)
    c = compute_conv_fft(a)
    
    # Log-sum-exp
    tc = t * c
    tc_max = np.max(tc)
    exp_tc = np.exp(tc - tc_max)
    log_Z = tc_max + np.log(np.sum(exp_tc))
    obj = 2.0 * n * log_Z / t
    
    # Weights
    w = exp_tc / np.sum(exp_tc)
    
    # Compute S[i] = sum_m a[m] * w[m+i] (cross-correlation)
    S = np.correlate(w, a, mode='full')
    
    # h[i] = 2*S[i] - w[2i]*a[i]
    h = 2.0 * S.copy()
    diag_idx = 2 * np.arange(n)
    valid = diag_idx < len(w)
    h[valid] -= w[diag_idx[valid]] * a[valid]
    
    # Gradient: grad_x[j] = 2n * a[j] * (h[j] - dot(h, a))
    mean_h = np.dot(h, a)
    grad = 2.0 * n * a * (h - mean_h)
    
    return obj, grad
