"""
Hybrid LP + Spectral optimization for step-function coefficient search.
Combines alternating LP with spectral initialization, adaptive step sizes,
and multi-scale exploration to minimize 2*n*max(conv(a,a))/(sum(a))^2.
"""

import numpy as np
from scipy.optimize import linprog
from scipy.interpolate import interp1d
from scipy.sparse import coo_matrix
import time

def compute_conv_fftp(a):
    """Compute full linear convolution a*a via FFT."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    conv = np.fft.irfft(fa * fa, size)
    return conv[:2 * n - 1]

def compute_obj(a):
    """Compute objective value."""
    n = len(a)
    S = float(np.sum(a))
    if S < 1e-10:
        return np.inf
    conv = compute_conv_fftp(a)
    M = float(np.max(conv))
    return 2.0 * n * M / (S * S)

def build_conv_matrix(a):
    """Build sparse matrix A such that A @ g = conv(a, g)."""
    n = len(a)
    n_conv = 2 * n - 1
    rows = []
    cols = []
    vals = []
    for j in range(n):
        # g[j] contributes to conv positions j, j+1, ..., j+n-1
        # conv[k] += a[k-j] * g[j] for k = j, ..., j+n-1
        for k in range(j, min(j + n, n_conv)):
            rows.append(k)
            cols.append(j)
            vals.append(a[k - j])
    A = coo_matrix((vals, (rows, cols)), shape=(n_conv, n)).tocsr()
    return A

def solve_lp_response(a, t, time_limit=10):
    """
    Solve: min t (already fixed) -> actually we want to maximize sum(g)
    s.t. (a*g)_k <= t for all k, g >= 0.
    This gives the best response direction.
    """
    n = len(a)
    A = build_conv_matrix(a)
    c = -np.ones(n)  # minimize -sum(g) = maximize sum(g)
    b_ub = np.full(2 * n - 1, t)
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': time_limit, 'dual_feasibility_tolerance': 1e-8})
        if res.status == 0:
            return np.maximum(res.x, 0)
    except Exception:
        pass
    return None

def solve_lp_min_peak(a, time_limit=10):
    """
    Solve: min t s.t. (a*g)_k <= t for all k, sum(g) = 1, g >= 0.
    This directly minimizes the peak of a*g.
    """
    n = len(a)
    A_conv = build_conv_matrix(a)
    # Variables: [g_0, ..., g_{n-1}, t]
    n_vars = n + 1
    # Objective: minimize t -> c = [0,...,0,1]
    c = np.zeros(n_vars)
    c[-1] = 1.0
    
    # Constraints: (a*g)_k - t <= 0 for all k
    # A_ub: [A_conv, -ones(2n-1, 1)]
    n_conv = 2 * n - 1
    A_ub_rows = list(range(n_conv))
    A_ub_cols = []
    A_ub_vals = []
    # For each row k: sum_i A_conv[k,i] * g_i - t <= 0
    # We need to add -1 in the t column for each row
    # Build augmented matrix
    from scipy.sparse import hstack, csc_matrix
    A_aug = hstack([A_conv, -np.ones((n_conv, 1))], format='csr')
    
    b_ub = np.zeros(n_conv)
    
    # Equality: sum(g) = 1
    A_eq = np.zeros((1, n_vars))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])
    
    bounds = [(0.0, None)] * n + [(None, None)]
    
    try:
        res = linprog(c, A_ub=A_aug, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': time_limit})
        if res.status == 0:
            g = np.maximum(res.x[:n], 0)
            return g
    except Exception:
        pass
    return None

```python
"""
Hybrid L-BFGS + LP optimization for step-function coefficient search.
Uses log-sum-exp smoothing of the max-convolution objective, softmax 
reparameterization for the simplex constraint, adaptive temperature 
annealing, and LP-based finishing to minimize 2*n*max(conv(a,a))/(sum(a))^2.
"""

import numpy as np
from scipy.optimize import linprog, minimize
from scipy.interpolate import interp1d
from scipy.sparse import coo_matrix, hstack
import time


def compute_conv_fftp(a):
    """Compute full linear convolution a*a via FFT."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    conv = np.fft.irfft(fa * fa, size)
    return conv[:2 * n - 1]


def compute_obj(a):
    """Compute exact objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    S = float(np.sum(a))
    if S < 1e-15:
        return np.inf
    conv = compute_conv_fftp(a)
    M = float(np.max(conv))
    return 2.0 * n * M / (S * S)


def lse_max_conv(a, L):
    """
    Compute log-sum-exp smoothed max of conv(a,a) and its gradient.
    Returns (smoothed_max, grad, actual_max)
    """
    n = len(a)
    conv = compute_conv_fftp(a)
    
    max_c = float(np.max(conv))
    # Numerically stable log-sum-exp
    shifted = L * (conv - max_c)
    # Clip to avoid overflow
    shifted = np.clip(shifted, -700, 700)
    exp_shifted = np.exp(shifted)
    sum_exp = np.sum(exp_shifted)
    
    lse = max_c + np.log(sum_exp) / L
    weights = exp_shifted / sum_exp
    
    # Gradient: d(lse)/da_i = 2 * sum_{j=0}^{n-1} a[j] * weights[i+j]
    # Compute via FFT: r = conv(a_reversed, weights)[n-1 : 2n-1]
    ar = a[::-1]
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa_r = np.fft.rfft(ar, size)
    fw = np.fft.rfft(weights, size)
    conv_ar_w = np.fft.irfft(fa_r * fw, size)[:2*n-1]
    r = conv_ar_w[n-1:]  # length n
    grad = 2.0 * r
    
    return lse, grad, max_c


def softmax_of_b(b):
    """Convert b to a = softmax(b) (sums to 1, all positive)."""
    b_shifted = b - np.max(b)
    exp_b = np.exp(b_shifted)
    return exp_b / np.sum(exp_b)


def objective_b(b, L, n_fixed=None):
    """
    Objective in b-space: J(softmax(b)) with log-sum-exp smoothing.
    Returns (value, gradient)
    """
    a = softmax_of_b(b)
    n = len(a)
    lse, grad_a, _ = lse_max_conv(a, L)
    # J = 2n * lse / (sum(a))^2 = 2n * lse (since sum(a) = 1)
    J = 2.0 * n * lse
    
    # Gradient: dJ/db_i = a_i * (dJa_i - sum_j a_j * dJa_j)
    # where dJa_j = 2n * d(lse)/da_j = 2n * grad_a[j]
    grad_J_a = 2.0 * n * grad_a
    # Project to tangent space of simplex
    grad_b = a * (grad_J_a - np.dot(a, grad_J_a))
    
    return J, grad_b


def lse_lbgfs(a0, L_schedule=None, maxiter=2000, time_budget=30.0, seed=0):
    """
    Optimize using L-BFGS-B on the log-sum-exp smoothed objective.
    a0: initial sequence (will be converted to b-space via log)
    L_schedule: list of L values to use (temperature annealing)
    """
    start = time.monotonic()
    n = len(a0)
    a0 = np.maximum(a0, 1e-15)
    S = np.sum(a0)
    a0 = a0 / S
    
    b = np.log(a0)  # b-space parametrization
    
    if L_schedule is None:
        L_schedule = [10, 30, 50, 100, 200, 400]
    
    best_a = a0.copy()
    best_obj = compute_obj(a0)
    
    for L in L_schedule:
        if time.monotonic() - start > time_budget:
            break
        
        def fun_and_grad(b_curr):
            val, grad = objective_b(b_curr, L)
            return val, grad
        
        try:
            res = minimize(fun_and_grad, b, method='L-BFGS-B',
                          jac=True, options={'maxiter': int(maxiter / len(L_schedule)),
                                             'gtol': 1e-10, 'maxls': 50,
                                             'iprint': -1})
            b = res.x
            a = softmax_of_b(b)
            obj = compute_obj(a)
            if obj < best_obj:
                best_obj = obj
                best_a = a.copy()
        except Exception:
            pass
    
    return best_a, best_obj


def build_conv_matrix(a):
    """Build sparse matrix A such that A @ g = conv(a, g)."""
    n = len(a)
    n_conv = 2 * n - 1
    rows = np.zeros(n * n, dtype=int)
    cols = np.zeros(n * n, dtype=int)
    vals = np.zeros(n * n)
    
    idx = 0
    for j in range(n):
        for k in range(j, min(j + n, n_conv)):
            rows[idx] = k
            cols[idx] = j
            vals[idx] = a[k - j]
            idx += 1
    
    A = coo_matrix((vals[:idx], (rows[:idx], cols[:idx])), shape=(n_conv, n)).tocsr()
    return A


def solve_lp_min_peak(a, time_limit=15):
    """
    Solve: min t s.t. (a*g)_k <= t for all k, sum(g) = 1, g >= 0.
    Returns optimal g.
    """
    n = len(a)
    n_conv = 2 * n - 1
    A_conv = build_conv_matrix(a)
    
    # Variables: g (length n), t (scalar)
    n_vars = n + 1
    c = np.zeros(n_vars)
    c[-1] = 1.0  # minimize t
    
    # A_ub: [A_conv | -ones(n_conv, 1)], b_ub = 0
    A_ub = hstack([A_conv, -np.ones((n_conv, 1))], format='csr')
    b_ub = np.zeros(n_conv)
    
    # A_eq: sum(g) = 1
    A_eq = np.zeros((1, n_vars))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])
    
    bounds = [(0.0, None)] * n + [(None, None)]
    
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': time_limit})
        if res.status == 0:
            return np.maximum(res.x[:n], 0)
    except Exception:
        pass
    return None


def alternating_lp(a, n_iters=50, time_per_lp=10.0, time_budget=100.0, blend=0.5):
    """
    Alternating LP: iteratively solve min-peak LP and blend.
    """
    start = time.monotonic()
    n = len(a)
    a = a.copy()
    S = np.sum(a)
    if S < 1e-15:
        return a, np.inf
    a = a / S
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    stall = 0
    
    for it in range(n_iters):
        if time.monotonic() - start > time_budget:
            break
        
        g = solve_lp_min_peak(a, time_limit=time_per_lp)
        if g is None:
            break
        
        S_g = np.sum(g)
        if S_g < 1e-15:
            break
        g = g / S_g
        
        # Blend
        a_new = (1.0 - blend) * a + blend * g
        a_new = np.maximum(a_new, 0)
        S_new = np.sum(a_new)
        if S_new < 1e-15:
            break
        a_new = a_new / S_new
        
        obj = compute_obj(a_new)
        if obj < best_obj - 1e-12:
            best_obj = obj
            best_a = a_new.copy()
            a = a_new
            stall = 0
        else:
            stall += 1
            if stall > 20:
                # Try stronger blend
                blend = min(1.0, blend * 1.5)
                if blend >= 1.0:
                    a = g.copy()
                    S_a = np.sum(a)
                    a = a / S_a
                    stall = 0
                    blend = 0.5
            elif stall > 40:
                break
    
    return best_a, best_obj


def try_parent():
    """Try to load parent sequence."""
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64)
        parent = np.clip(parent, 0, 1000.0)
        if parent.size > 0 and np.isfinite(parent).all():
            obj = compute_obj(parent)
            print(f"Parent (n={parent.size}): obj={obj:.6f}")
            return parent.copy(), obj
    except Exception:
        pass
    return None, np.inf


def cosine_init(n, seed=0, n_modes=10):
    """Generate cosine-series initialization."""
    rng = np.random.default_rng(seed)
    k = np.arange(n)
    a = np.ones(n)
    for m in range(1, n_modes + 1):
        c = rng.uniform(-0.5, 0.5) / np.sqrt(m)
        phase = rng.uniform(0, 2 * np.pi)
        a += c * np.cos(2 * np.pi * m * k / n + phase)
    a = np.maximum(a, 1e-10)
    S = np.sum(a)
    return a / S


def interpolate_sequence(a, n_new):
    """Interpolate sequence to new length."""
    n_old = len(a)
    if n_old == n_new:
        return a.copy()
    x_old = np.linspace(0, 1, n_old)
    x_new = np.linspace(0, 1, n_new)
    try:
        f = interp1d(x_old, a, kind='cubic', fill_value='extrapolate')
        a_new = np.maximum(f(x_new), 0)
    except Exception:
        a_new = np.ones(n_new) / n_new
    S = np.sum(a_new)
    if S < 1e-15:
        return np.ones(n_new) / n_new
    return a_new / S


def propose_candidate(seed=42, budget_s=1500):
    """
    Main search function combining L-BFGS with log-sum-exp smoothing,
    LP-based alternating refinement, and multi-scale exploration.
    """
    start_time = time.monotonic()
    deadline = start_time + budget_s - 30.0
    
    best_score = np.inf
    best_seq = None
    
    # Load parent
    parent, parent_obj = try_parent()
    if parent is not None:
        best_score = parent_obj
        best_seq = parent.copy()
        n_base = len(parent)
    else:
        n_base = 500
        best_seq = np.ones(n_base) / n_base
        best_score = compute_obj(best_seq)
        print(f"No parent. Starting uniform (n={n_base}): obj={best_score:.6f}")
    
    print(f"Initial best: n={n_base}, score={best_score:.6f}")
    
    # === Phase 1: L-BFGS with log-sum-exp smoothing ===
    print("\n=== Phase 1: L-BFGS with LSE smoothing ===")
    
    if time.monotonic() < deadline - 60:
        remaining = deadline - time.monotonic()
        alloc = min(remaining * 0.25, 200.0)
        
        # Try different L schedules
        L_schedules = [
            [5, 15, 30, 60, 100, 150, 200],
            [10, 20, 40, 80, 160, 300],
            [20, 50, 100, 200, 400],
            [8, 16, 32, 64, 128, 256, 512],
        ]
        
        for ls_idx, L_sched in enumerate(L_schedules):
            if time.monotonic() > deadline - 60:
                break
            remaining = deadline - time.monotonic()
            alloc_ls = min(remaining * 0.2, 80.0)
            
            # Start from current best
            a_init = best_seq.copy()
            a_new, obj_new = lse_lbgfs(a_init, L_schedule=L_sched, 
                                        maxiter=3000, time_budget=alloc_ls, seed=seed+ls_idx)
            if obj_new < best_score:
                best_score = obj_new
                best_seq = a_new.copy()
                print(f"  L-BFGS schedule {ls_idx}: {best_score:.6f} ** NEW BEST **")
            else:
                print(f"  L-BFGS schedule {ls_idx}: {obj_new:.6f}")
        
        # Also try from cosine initializations
        for n_try in [n_base, n_base - 50, n_base + 50]:
            if time.monotonic() > deadline - 60:
                break
            if n_try < 50:
                continue
            remaining = deadline - time.monotonic()
            alloc_try = min(remaining * 0.15, 40.0)
            
            for cos_seed in [seed + 100 + n_try, seed + 200 + n_try]:
                if time.monotonic() > deadline - 40:
                    break
                a_cos = cosine_init(n_try, seed=cos_seed, n_modes=12)
                a_new, obj_new = lse_lbgfs(a_cos, L_schedule=[10, 30, 80, 200, 400],
                                            maxiter=2000, time_budget=alloc_try, seed=cos_seed)
                if obj_new < best_score:
                    best_score = obj_new
                    best_seq = a_new.copy()
                    print(f"  Cosine init (n={n_try}, seed={cos_seed}): {best_score:.6f} ** NEW BEST **")
    
    # === Phase 2: LP-based alternating refinement ===
    print("\n=== Phase 2: LP-based alternating ===")
    
    if time.monotonic() < deadline - 60:
        remaining = deadline - time.monotonic()
        alloc = min(remaining * 0.3, 300.0)
        
        # Try different blend factors
        for blend in [0.3, 0.5, 0.7, 1.0]:
            if time.monotonic() > deadline - 60:
                break
            remaining = deadline - time.monotonic()
            alloc_blend = min(remaining * 0.25, 100.0)
            
            a_lp, obj_lp = alternating_lp(best_seq, n_iters=100, time_per_lp=8.0,
                                          time_budget=alloc_blend, blend=blend)
            if obj_lp < best_score:
                best_score = obj_lp
                best_seq = a_lp.copy()
                print(f"  LP blend={blend}: {best_score:.6f} ** NEW BEST **")
            else:
                print(f"  LP blend={blend}: {obj_lp:.6f}")
    
    # === Phase 3: Multi-scale exploration ===
    print("\n=== Phase 3: Multi-scale exploration ===")
    
    if time.monotonic() < deadline - 30:
        n_best = len(best_seq)
        n_tries = sorted(set([
            max(100, n_best - 100), max(100, n_best - 50),
            n_best, n_best + 25, n_best + 50, n_best + 100,
            max(100, n_best + 200)
        ]))
        n_tries = [n for n in n_tries if 100 <= n <= 3000]
        
        for n_try in n_tries:
            if time.monotonic() > deadline - 20:
                break
            remaining = deadline - time.monotonic()
            if remaining < 20:
                break
            
            a_interp = interpolate_sequence(best_seq, n_try)
            
            # Quick L-BFGS refinement
            alloc = min(remaining * 0.2, 40.0)
            a_new, obj_new = lse_lbgfs(a_interp, L_schedule=[20, 50, 100, 200],
                                        maxiter=1000, time_budget=alloc, seed=seed + n_try)
            
            # Quick LP refinement

                                        maxiter=2000, time_budget=alloc_sg,

            # LP pass (if n is small enough)
            if time.monotonic() < deadline - 15 and n_best <= 600:

                if obj_lp < obj_new:
