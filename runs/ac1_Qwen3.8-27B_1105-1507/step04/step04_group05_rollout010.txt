"""

1. Soft-subgradient descent with backtracking line search
2. Fast coordinate descent with incremental convolution updates
3. LP-based mirror descent for larger moves

import numpy as np
import time
from scipy.optimize import linprog


def fast_conv(a):
    """Full self-convolution via FFT."""
    n = len(a)
    s = 1
    while s < 2 * n - 1:
        s <<= 1
    fa = np.fft.rfft(a, s)
    return np.fft.irfft(fa * fa, s)[:2 * n - 1]


def compute_J(a):
    """Compute J(a) = 2*n*max(conv(a,a))/(sum(a))^2."""
    a = np.maximum(a, 0.0)
    S = a.sum()
    if S < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (S * S))


def soft_subgradient(a, tau):
    """Compute soft subgradient of max(conv(a,a))."""
    n = len(a)
    b = fast_conv(a)
    b_max = b.max()
    # Soft weights (Gumbel-softmax style)
    if tau < 1e-10:
        # Hard max
        w = np.zeros(2 * n - 1)
        w[np.argmax(b)] = 1.0
    else:
        w = np.exp((b - b_max) / tau)
        w /= w.sum()
    # grad_j = sum_k w_k * a[k-j] (if 0 <= k-j < n)
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        if w[k] < 1e-12:
            continue
        jl = max(0, k - n + 1)
        jh = min(n - 1, k)
        if jl <= jh:
            # grad[jl:jh+1] += w[k] * a[k-jh : k-jl+1][::-1]
            seg = a[k - jh:k - jl + 1][::-1]
            grad[jl:jh + 1] += w[k] * seg
    return grad


def subgradient_step_backtrack(a, tau, max_iter=50):
    """One subgradient step with backtracking line search."""
    n = len(a)
    S = a.sum()
    if S < 1e-10:
        return a
    a_norm = a / S
    b = fast_conv(a_norm)
    b_max = b.max()
    J_cur = 2.0 * n * b_max  # since sum=1, J = 2n*max_b
    
    grad = soft_subgradient(a_norm, tau)
    # Remove mean component (to preserve normalization approximately)
    grad -= grad.mean()
    gnorm = np.linalg.norm(grad)
    if gnorm < 1e-15:
        return a
    
    # Backtracking line search
    eta = 0.1
    for _ in range(max_iter):
        a_trial = a_norm - eta * grad
        a_trial = np.maximum(a_trial, 0.0)
        S_trial = a_trial.sum()
        if S_trial < 1e-10:
            eta *= 0.5
            continue
        a_trial = a_trial / S_trial
        b_trial = fast_conv(a_trial)
        J_trial = 2.0 * n * b_trial.max()
        if J_trial < J_cur - 1e-12:
            return a_trial * S  # rescale back
        eta *= 0.5
        if eta < 1e-8:
            break
    return a


def cd_update_coord(a, b, i, S):
    """Find optimal value for coordinate i. Returns (new_val, new_b, new_S, new_J)."""
    n = len(a)
    old = a[i]
    
    # Precompute unaffected max
    k_start = i
    k_end = min(2 * n - 2, i + n - 1)
    
    # Track the max outside affected range
    if k_start > 0:
        M_unaff_left = b[:k_start].max() if k_start > 0 else 0.0
    else:
        M_unaff_left = 0.0
    if k_end < 2 * n - 2:
        M_unaff_right = b[k_end + 1:].max() if k_end + 1 < 2 * n - 1 else 0.0
    else:
        M_unaff_right = 0.0
    M_unaff = max(M_unaff_left, M_unaff_right)
    
    # Coefficients for affected b_k as function of x = a_i
    # b_k(x) = b_k + 2*(x-old)*a[k-i] for k != 2i
    # b_{2i}(x) = b_{2i} + x^2 - old^2 for k = 2i
    # So b_k(x) = base_k + slope_k * x for k != 2i (linear)
    #            b_{2i}(x) = quad_base + x^2 (quadratic)
    
    bases = np.zeros(n)  # for k in [k_start, k_end], index by local = k - k_start
    slopes = np.zeros(n)
    quad_active = False
    quad_base = 0.0
    quad_local = -1
    
    for local, k in enumerate(range(k_start, k_end + 1)):
        ki = k - i
        if 0 <= ki < n:
            if k == 2 * i:
                quad_active = True
                quad_base = b[k] - old * old
                quad_local = local
                bases[local] = 0  # handled separately
                slopes[local] = 0
            else:
                bases[local] = b[k] - 2.0 * old * a[ki]
                slopes[local] = 2.0 * a[ki]
        else:
            bases[local] = b[k]
            slopes[local] = 0.0
    
    n_aff = k_end - k_start + 1
    
    def eval_J(x):
        """Evaluate J when a[i] = x."""
        new_S = S - old + x
        if new_S < 1e-10:
            return 1e10
        # Max of affected
        vals_rec = bases[:n_aff] + slopes[:n_aff] * x
        if quad_active:
            vals_rec[quad_local] = quad_base + x * x
        max_aff = vals_rec.max()
        max_total = max(M_unaff, max_aff)
        return 2.0 * n * max_total / (new_S * new_S)
    
    # Golden section search
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    lo = 0.0
    hi = max(3.0 * old, S / n * 5.0, 0.5)
    
    c_pt = hi - gr * (hi - lo)
    d_pt = lo + gr * (hi - lo)
    fc = eval_J(c_pt)
    fd = eval_J(d_pt)
    
    for _ in range(30):
        if fc < fd:
            hi = d_pt
            d_pt = c_pt
            fd = fc
            c_pt = hi - gr * (hi - lo)
            fc = eval_J(c_pt)
        else:
            lo = c_pt
            c_pt = d_pt
            fc = fd
            d_pt = lo + gr * (hi - lo)
            fd = eval_J(d_pt)
    
    x_opt = max(0.0, (lo + hi) / 2.0)
    
    # Update b incrementally
    delta = x_opt - old
    for k in range(k_start, k_end + 1):
        ki = k - i
        if 0 <= ki < n:
            if k == 2 * i:
                b[k] = b[k] - old * old + x_opt * x_opt
            else:
                b[k] = b[k] + 2.0 * delta * a[ki]
    
    new_S = S - old + x_opt
    J_new = 2.0 * n * b.max() / (new_S * new_S)
    return x_opt, b, new_S, J_new


def cd_pass_fast(a, max_coords=None):
    """One full pass of fast coordinate descent. Returns (a, b, S, J)."""
    a = np.maximum(a.astype(np.float64), 0.0)
    S = a.sum()
    if S < 1e-10:
        return a
    n = len(a)
    b = fast_conv(a)
    J_cur = 2.0 * n * b.max() / (S * S)
    
    indices = list(range(n))
    np.random.shuffle(indices)
    if max_coords is not None:
        indices = indices[:min(max_coords, n)]
    
    for i in indices:
        new_val, b, S, J_new = cd_update_coord(a, b, i, S)
        if abs(new_val - a[i]) > 1e-12:
            a[i] = new_val
    
    # Final verification
    b_check = fast_conv(a)
    J_final = 2.0 * n * b_check.max() / (S * S)
    return a, b_check, S, J_final

def lp_best_response(a, M):
    """LP: max sum(g) s.t. (a*g)_k <= M for all k, g >= 0."""
    n = len(a)
    # Build constraint matrix: A[k, j] = a[k-j] if 0 <= k-j < n else 0
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        jl = max(0, k - n + 1)
        jh = min(n - 1, k)
        for j in range(jl, jh + 1):
            ki = k - j
            rows.append(k)
            cols.append(j)
            vals.append(a[ki])
    
    if not rows:
        return None
    
    # Use dense for small problems
    A = np.zeros((2 * n - 1, n))
    A[np.array(rows), np.array(cols)] = np.array(vals)
    
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    
    try:
        r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                    options={'maxiter': 5000})
        if r.success and r.x is not None:
            g = np.maximum(r.x, 0.0)
            if g.sum() > 1e-10:
                return g
    except Exception:
        pass
    return None

def resample_seq(src, dest_n):
    """Resample 1D sequence to new length via linear interpolation."""
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)


def generate_inits(n, rng):
    """Generate diverse initial sequences."""
    inits = []
    t = np.linspace(0, 1, n, endpoint=False)
    
    inits.append(np.ones(n))
    inits.append(rng.uniform(0.3, 1.0, n))
    
    # U-shaped (larger at edges)
    inits.append(1.0 + 0.5 * (2 * t - 1.0) ** 2)
    inits.append(1.0 / np.sqrt(t + 0.1) + 1.0 / np.sqrt(1 - t + 0.1))
    inits.append(0.5 + 0.5 * np.cos(np.pi * t) ** 2)
    
    # Cosine windows of various widths
    for alpha in [0.5, 1.0, 2.0]:
        inits.append(np.cos(np.pi * alpha * (t - 0.5)) ** 2 + 0.1)
    
    # Sinusoidal
    for freq in [2, 3, 5]:
        inits.append(1.0 + 0.3 * np.cos(2 * np.pi * freq * t + rng.uniform(0, 6.28)))
    
    # Random smooth
    for _ in range(3):
        raw = rng.normal(0, 1, n)
        # Smooth with a small kernel
        kernel = np.ones(5) / 5
        smooth = np.convolve(raw, kernel, mode='same')
        inits.append(np.abs(smooth) + 0.3)
    
    for init in inits:
        init = np.maximum(init, 0.01)
        init = init / init.sum()
        inits[-1] = init
    return inits

                J_cur = J_pert
                a = a_pert
                tracker.update(a)
    
    tracker.update(a)
    return a


def propose_candidate(seed=42, budget_s=1500, **kwargs):
    """
    Main search function: combines subgradient descent, coordinate descent,
    LP mirror descent, and multi-length exploration.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + float(budget_s) - 30.0
    tracker = ScoreTracker()
    
    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        if p.ndim == 1 and p.size > 0 and np.all(np.isfinite(p)):
            parent = np.maximum(p, 0.0)
            tracker.update(parent)
            print(f"Parent loaded: n={parent.size}, J={tracker.best_score:.8f}", flush=True)
    except (NameError, AttributeError):
        pass
    
    if parent is None:
        parent = np.ones(384)
        tracker.update(parent)
        print(f"No parent, using uniform n=384, J={tracker.best_score:.8f}", flush=True)
    
    n_parent = parent.size
    
    # === Phase 1: Subgradient descent on parent (300s) ===
    print(f"\n{'='*50} Phase 1: Subgradient (n={n_parent}) {'='*50}", flush=True)
    t0 = time.monotonic()
    a = parent.copy()
    a = subgradient_optimize(a, time.monotonic() + 200, tracker, tau_init=0.03)
    tracker.report(f"Phase 1 ({time.monotonic()-t0:.0f}s):")
    
    # === Phase 2: Intensive coordinate descent (400s) ===
    if time.monotonic() < deadline - 300:
        print(f"\n{'='*50} Phase 2: Coordinate Descent {'='*50}", flush=True)
        t0 = time.monotonic()
        a = cd_optimize(tracker.best, time.monotonic() + 350, tracker, max_passes=500)
        tracker.report(f"Phase 2 ({time.monotonic()-t0:.0f}s):")
    
    # === Phase 3: LP mirror descent (300s) ===
    if time.monotonic() < deadline - 200:
        print(f"\n{'='*50} Phase 3: LP Mirror Descent {'='*50}", flush=True)
        t0 = time.monotonic()
        a = lp_mirror_descent(tracker.best, time.monotonic() + 250, tracker, max_iters=80)
        tracker.report(f"Phase 3 ({time.monotonic()-t0:.0f}s):")
    
    # === Phase 4: CD after LP (150s) ===
    if time.monotonic() < deadline - 100:
        print(f"\n{'='*50} Phase 4: CD Polish after LP {'='*50}", flush=True)
        t0 = time.monotonic()
        a = cd_optimize(tracker.best, time.monotonic() + 120, tracker, max_passes=300)
        tracker.report(f"Phase 4 ({time.monotonic()-t0:.0f}s):")
    
    # === Phase 5: Length exploration (200s) ===
    if time.monotonic() < deadline - 80:
        print(f"\n{'='*50} Phase 5: Length Exploration {'='*50}", flush=True)
        t0 = time.monotonic()
        
        lengths_to_try = sorted(set([
            n_parent - 4, n_parent - 2, n_parent + 2, n_parent + 4,
            n_parent + 6, n_parent + 8, n_parent + 10, n_parent - 6,
            n_parent + 12, n_parent + 20, n_parent - 10
        ]), key=lambda x: abs(x - n_parent))
        
        for nl in lengths_to_try:
            if time.monotonic() > deadline - 60:
                break
            if nl < 50 or nl > 800:
                continue
            
            a_res = resample_seq(tracker.best, nl)
            a_res = np.maximum(a_res, 0.01)
            a_res = a_res / a_res.sum()
            
            # Quick CD
            a_res, _, _, _ = cd_pass_fast(a_res)
            a_res = a_res / a_res.sum()
            tracker.update(a_res)
            
            # One more CD pass if time
            if time.monotonic() < deadline - 60:
                a_res, _, _, _ = cd_pass_fast(a_res)
                a_res = a_res / a_res.sum()
                tracker.update(a_res)
            
            print(f"    n={nl}: J={tracker.best_score:.8f}", flush=True)
        
        tracker.report(f"Phase 5 ({time.monotonic()-t0:.0f}s):")
    
    # === Phase 6: Perturbation restarts (150s) ===
    if time.monotonic() < deadline - 50:
        print(f"\n{'='*50} Phase 6: Perturbation Restarts {'='*50}", flush=True)
        t0 = time.monotonic()
        n_best = tracker.best.size
        
        for temp in [0.03, 0.02, 0.01, 0.005, 0.05]:
            if time.monotonic() > deadline - 40:
                break
            
            a_pert = tracker.best + rng.normal(0, temp, n_best)
            a_pert = np.maximum(a_pert, 0.01)
            a_pert = a_pert / a_pert.sum()
            
            # Short CD
            a_pert, _, _, _ = cd_pass_fast(a_pert)
            a_pert = a_pert / a_pert.sum()
            tracker.update(a_pert)
            
            # Another CD pass
            if time.monotonic() < deadline - 40:
                a_pert, _, _, _ = cd_pass_fast(a_pert)
                a_pert = a_pert / a_pert.sum()
                tracker.update(a_pert)
            
            print(f"    temp={temp}: J={tracker.best_score:.8f}", flush=True)
        
        tracker.report(f"Phase 6 ({time.monotonic()-t0:.0f}s):")
    
    # === Phase 7: Final intensive CD (remaining time) ===
    if time.monotonic() < deadline - 5:
        print(f"\n{'='*50} Phase 7: Final Polish {'='*50}", flush=True)
        t0 = time.monotonic()
        a = cd_optimize(tracker.best, deadline - 5, tracker, max_passes=1000)
        tracker.report(f"Phase 7 ({time.monotonic()-t0:.0f}s):")
    
    # Fallback
    if tracker.best is None:
        tracker.best = np.ones(256)
        tracker.best_score = compute_J(tracker.best)
    
    print(f"\n{'='*50} FINAL RESULT {'='*50}", flush=True)
    print(f"J = {tracker.best_score:.10f}, n = {tracker.best.size}", flush=True)
    print(f"Total time: {time.monotonic()-tracker.t_start:.1f}s", flush=True)
    
    return [float(x) for x in tracker.best]
