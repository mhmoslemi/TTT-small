"""
Optimized step-function coefficient search using:
1. FFT-accelerated log-sum-exp smoothed gradient
2. Mirror descent (exponential weights) on the simplex
3. Adaptive temperature scheduling
4. Periodic LP refinement for provable improvements
5. Multi-start with diverse initializations and lengths
6. Perturbation-based escapes from local minima

import numpy as np
from scipy.optimize import linprog
from scipy.sparse import coo_matrix
import time

def compute_conv(a):
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    conv = np.fft.irfft(fa * fa, size)
    return conv[:2 * n - 1]

def compute_obj(a):
    n = len(a)
    S = float(np.sum(a))
    if S < 1e-10:
        return np.inf
    conv = compute_conv(a)
    M = float(np.max(conv))
    return 2.0 * n * M / (S * S)

def smooth_gradient_fft(a, tau=0.01):
    """Compute gradient of log-sum-exp smoothed max of conv(a,a) using FFT."""
    n = len(a)
    conv = compute_conv(a)
    max_c = np.max(conv)
    exp_c = np.exp((conv - max_c) / tau)
    w_sum = np.sum(exp_c)
    if w_sum < 1e-30:
        w = np.zeros_like(conv)
        w[np.argmax(conv)] = 1.0
    else:
        w = exp_c / w_sum
    
    # Cross-correlation of w (len 2n-1) and a (len n)
    size = 1
    while size < 3 * n - 2:
        size <<= 1
    fw = np.fft.rfft(w, size)
    fa = np.fft.rfft(a, size)
    cross = np.fft.irfft(fw * np.conj(fa), size)
    
    grad = 2.0 * cross[:n]
    return grad, max_c, conv

def topk_gradient(a, topk=20):
    """Compute subgradient using top-K active constraints."""
    n = len(a)
    conv = compute_conv(a)
    M = np.max(conv)
    
    # Get top-K indices
    if topk >= len(conv):
        idx = np.arange(len(conv))
    else:
        idx = np.argpartition(conv, -topk)[-topk:]
    
    # Compute gradient: for active k, grad[i] += 2*a[k-i]
    # Using FFT: grad = 2 * cross_corr(weights, a) where weights is one-hot on active set
    w = np.zeros(len(conv))
    w[idx] = 1.0 / len(idx)
    
    size = 1
    while size < 3 * n - 2:
        size <<= 1
    fw = np.fft.rfft(w, size)
    fa = np.fft.rfft(a, size)
    cross = np.fft.irfft(fw * np.conj(fa), size)
    
    grad = 2.0 * cross[:n]
    return grad, M, conv

def mirror_descent(a, max_iters=2000, time_budget=30.0, tau_init=0.003, tau_range=0.08, topk=30):
    """Mirror descent with adaptive temperature, using log-sum-exp or top-K gradient."""
    n = len(a)
    a = a.copy().astype(np.float64)
    S = np.sum(a)
    if S < 1e-10:
        return a
    a = a / S
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    
    eta = 0.25
    stall = 0
    use_smooth = n <= 2000  # Use smooth gradient for reasonable n
    
    for it in range(max_iters):
        if time.monotonic() - start > time_budget:
            break
        
        # Adaptive tau
        progress = min(1.0, it / max(1, max_iters * 0.8))
        tau = tau_init + tau_range * progress
        
        if use_smooth and tau > 0.001:
            grad, max_c, conv = smooth_gradient_fft(a, tau)
        else:
            grad, max_c, conv = topk_gradient(a, topk=topk)
        
        # Center gradient
        grad_c = grad - np.mean(grad)
        
        # Adaptive eta based on gradient magnitude
        grad_norm = np.sqrt(np.mean(grad_c**2))
        if grad_norm < 1e-10:
            break
        
        # Mirror descent update
        a_new = a * np.exp(-eta * grad_c / grad_norm)
        a_new = np.maximum(a_new, 1e-15)
        S_new = np.sum(a_new)
        if S_new < 1e-10:
            break
        a_new = a_new / S_new
        
        obj = compute_obj(a_new)
        
        if obj < best_obj - 1e-11:
            best_obj = obj
            best_a = a_new.copy()
            a = a_new
            stall = 0
        else:
            stall += 1
            eta *= 0.95
            if eta < 1e-6:
                eta = 0.25
                # Perturb to escape
                rng = np.random.default_rng(it * 31 + 7)
                a = best_a * np.exp(rng.normal(0, 0.08, n))
                a = np.maximum(a, 0)
                a = a / np.sum(a)
                stall = 0
    
    return best_a

def build_toeplitz_sparse(a):
    """Build sparse matrix for conv(a, g) as linear in g."""
    n = len(a)
    rows = []
    cols = []
    vals = []
    for j in range(n):
        # conv(a,g)[k] = sum_i a[i]*g[k-i]
        # For fixed j (which is g[j]), it contributes to conv[k] where k = i + j, i.e., a[i]*g[j]
        # So for column j, the non-zero entries are at rows k = j, j+1, ..., j+n-1
        # with values a[0], a[1], ..., a[n-1]
        start = j
        end = min(j + n, 2 * n - 1)
        length = end - start
        rows.extend(range(start, end))
        cols.extend([j] * length)
        vals.extend(a[:length].tolist())
    
    A = coo_matrix((vals, (rows, cols)), shape=(2 * n - 1, n)).tocsr()
    return A

def solve_lp(a, M, time_limit=10):
    """Solve LP: max sum(g) s.t. conv(a,g) <= M, g >= 0."""
    n = len(a)
    A = build_toeplitz_sparse(a)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': time_limit, 'mip_gap': 0})
        if res.status == 0:
            return np.maximum(res.x, 0)
    except Exception:
        pass
    return None

def lp_refine(a, max_iters=20, time_budget=30.0):
    """Alternating LP refinement."""
    n = len(a)
    a = a.copy().astype(np.float64)
    S = np.sum(a)
    if S < 1e-10:
        return a
    a = a / S
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    stall = 0
    
    for it in range(max_iters):
        if time.monotonic() - start > time_budget:
            break
        
        conv = compute_conv(a)
        M = float(np.max(conv))
        
        # Slightly relaxed M for better LP solution
        M_lp = M * 1.001
        
        g = solve_lp(a, M_lp, time_limit=5)
        if g is None:
            break
        
        S_g = np.sum(g)
        if S_g < 1e-10:
            break
        g = g / S_g
        
        # Line search
        best_t = 0.0
        best_obj_ls = best_obj
        for t in [0.1, 0.2, 0.3, 0.5, 0.7, 0.9, 1.0]:
            a_try = (1.0 - t) * a + t * g
            a_try = np.maximum(a_try, 0)
            S_try = np.sum(a_try)
            if S_try < 1e-10:
                continue
            a_try = a_try / S_try
            obj_try = compute_obj(a_try)
            if obj_try < best_obj_ls - 1e-12:
                best_obj_ls = obj_try
                best_t = t
        
        if best_t > 0.0:
            a = (1.0 - best_t) * a + best_t * g
            a = np.maximum(a, 0)
            a = a / np.sum(a)
            best_obj = best_obj_ls
            best_a = a.copy()
            stall = 0
        else:
            stall += 1
            if stall > 10:
                break
    
    return best_a

def cosine_init(n, n_modes=8, seed=0):
    rng = np.random.default_rng(seed)
    k = np.arange(n)
    a = np.ones(n)
    for m in range(1, n_modes + 1):
        c = rng.uniform(-0.5, 0.5) / m
        phase = rng.uniform(0, 2 * np.pi)
        a += c * np.cos(2 * np.pi * m * k / n + phase)
    a = np.maximum(a, 0)
    S = np.sum(a)
    if S < 1e-10:
        return np.ones(n) / n
    return a / S

def smoothed_bump_init(n, sigma=0.25, center=0.5):
    k = np.arange(n)
    x = k / (n - 1) if n > 1 else np.array([0.5])
    a = np.exp(-0.5 * ((x - center) / sigma) ** 2)
    a = a + 0.1  # Small floor to avoid zeros
    S = np.sum(a)
    return a / S

def raised_cosine_init(n, alpha=2.0):
    k = np.arange(n)
    x = 2.0 * k / (n - 1) - 1.0 if n > 1 else np.array([0.0])
    a = (0.5 * (1.0 + np.cos(np.pi * x))) ** alpha
    a = np.maximum(a, 1e-6)
    S = np.sum(a)
    return a / S

"""
Optimized step-function coefficient search using:
1. FFT-accelerated log-sum-exp smoothed gradient (O(n log n))
2. Mirror descent (exponential weights) on the simplex
3. Adaptive temperature based on convolution spread
4. Parametric shape optimization for diverse initializations
5. Periodic LP refinement for provable improvements
6. Multi-start with diverse initializations and length exploration
7. Perturbation-based escapes from local minima
"""

import numpy as np
from scipy.optimize import linprog, minimize
from scipy.sparse import coo_matrix
import time


def compute_conv(a):
    """Compute full convolution a*a via FFT."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    conv = np.fft.irfft(fa * fa, size)
    return conv[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    S = float(np.sum(a))
    if S < 1e-10:
        return np.inf
    conv = compute_conv(a)
    M = float(np.max(conv))
    return 2.0 * n * M / (S * S)


def adaptive_tau(conv, n_effective=30):
    """Compute adaptive temperature for log-sum-exp based on convolution spread."""
    c_max = float(np.max(conv))
    c_min = float(np.min(conv))
    spread = c_max - c_min
    if spread < 1e-15:
        return 0.001
    return spread / np.log(max(2.0, float(n_effective)))


def smooth_gradient_fft(a, tau=0.01):
    """Compute gradient of log-sum-exp smoothed max of conv(a,a) using FFT."""
    n = len(a)
    conv = compute_conv(a)
    max_c = float(np.max(conv))
    # Clip exponent for numerical stability
    exp_arg = (conv - max_c) / tau
    exp_arg = np.maximum(exp_arg, -50.0)
    exp_c = np.exp(exp_arg)
    w_sum = float(np.sum(exp_c))
    if w_sum < 1e-30:
        w = np.zeros_like(conv)
        w[int(np.argmax(conv))] = 1.0
    else:
        w = exp_c / w_sum

    # Cross-correlation of w (len 2n-1) and a (len n) via FFT
    size = 1
    while size < 3 * n - 2:
        size <<= 1
    fw = np.fft.rfft(w, size)
    fa = np.fft.rfft(a, size)
    cross = np.fft.irfft(fw * np.conj(fa), size)

    grad = 2.0 * cross[:n]
    return grad, max_c, conv


def topk_gradient(a, topk=30):
    """Compute subgradient using top-K active constraints (FFT-accelerated)."""
    n = len(a)
    conv = compute_conv(a)
    M = float(np.max(conv))

    if topk >= len(conv):
        idx = np.arange(len(conv))
    else:
        idx = np.argpartition(conv, -topk)[-topk:]

    w = np.zeros(len(conv))
    w[idx] = 1.0 / len(idx)

    size = 1
    while size < 3 * n - 2:
        size <<= 1
    fw = np.fft.rfft(w, size)
    fa = np.fft.rfft(a, size)
    cross = np.fft.irfft(fw * np.conj(fa), size)

    grad = 2.0 * cross[:n]
    return grad, M, conv


def mirror_descent(a, max_iters=3000, time_budget=30.0):
    """Mirror descent with adaptive log-sum-exp temperature."""
    n = len(a)
    a = a.copy().astype(np.float64)
    S = np.sum(a)
    if S < 1e-10:
        return a
    a = a / S

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    stall = 0

        conv = compute_conv(a)
        tau = adaptive_tau(conv, n_eff)
        tau = max(tau, 1e-6)

        grad, max_c, _ = smooth_gradient_fft(a, tau)
        grad_c = grad - np.mean(grad)

        grad_norm = float(np.sqrt(np.mean(grad_c ** 2)))
        if grad_norm < 1e-12:
            break

        # Mirror descent (exponential weights) update
        a_new = a * np.exp(-eta * grad_c / grad_norm)
        a_new = np.maximum(a_new, 1e-15)
        S_new = float(np.sum(a_new))
        if S_new < 1e-10:

        a_new = a_new / S_new

            a = a_new

        else:

            eta *= 0.95
            if eta < 1e-6:
                eta = 0.3

def build_toeplitz_sparse(a):
    """Build sparse matrix representing conv(a, g) as linear in g."""
    n = len(a)
    rows = []
    cols = []
    vals = []
    for j in range(n):
        start = j
        end = min(j + n, 2 * n - 1)
        length = end - start
        rows.extend(range(start, end))
        cols.extend([j] * length)
        vals.extend(a[:length].tolist())
    A = coo_matrix((vals, (rows, cols)), shape=(2 * n - 1, n)).tocsr()
    return A


def solve_lp(a, M, time_limit=8):
    """Solve LP: max sum(g) s.t. conv(a,g) <= M, g >= 0."""
    n = len(a)
    A = build_toeplitz_sparse(a)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': time_limit})
        if res.status == 0:
            return np.maximum(res.x, 0)
    except Exception:
        pass
    return None


def lp_refine(a, max_iters=25, time_budget=30.0):
    """Alternating LP refinement with line search."""
    n = len(a)
    a = a.copy().astype(np.float64)
    S = np.sum(a)
    if S < 1e-10:
        return a
    a = a / S

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    stall = 0

    for it in range(max_iters):
        if time.monotonic() - start > time_budget:
            break

        conv = compute_conv(a)
        M = float(np.max(conv))
        M_lp = M * 1.002

        g = solve_lp(a, M_lp, time_limit=5)
        if g is None:
            break

        S_g = float(np.sum(g))
        if S_g < 1e-10:
            break
        g = g / S_g

        best_t = 0.0
        best_obj_ls = best_obj
        for t in [0.1, 0.2, 0.3, 0.5, 0.7, 0.9, 1.0]:
            a_try = (1.0 - t) * a + t * g
            a_try = np.maximum(a_try, 0)
            S_try = float(np.sum(a_try))
            if S_try < 1e-10:
                continue
            a_try = a_try / S_try
            obj_try = compute_obj(a_try)
            if obj_try < best_obj_ls - 1e-12:
                best_obj_ls = obj_try
                best_t = t

        if best_t > 0.0:
            a = (1.0 - best_t) * a + best_t * g
            a = np.maximum(a, 0)
            a = a / np.sum(a)
            best_obj = best_obj_ls
            best_a = a.copy()
            stall = 0
        else:
            stall += 1
            if stall > 12:
                break

    return best_a

        c = rng.uniform(-0.5, 0.5) / m
        phase = rng.uniform(0, 2 * np.pi)
        a += c * np.cos(2 * np.pi * m * k / n + phase)
    a = np.maximum(a, 0)
    S = np.sum(a)
    if S < 1e-10:
        return np.ones(n) / n
    return a / S


def gaussian_init(n, sigma=0.3, center=0.5):
    """Gaussian bump initialization."""
    k = np.arange(n)
    x = k / max(1, n - 1)
    a = np.exp(-0.5 * ((x - center) / sigma) ** 2) + 0.05
    S = np.sum(a)
    return a / S


def raised_cosine_init(n, alpha=2.0):
    """Raised cosine shape."""
    k = np.arange(n)
    x = 2.0 * k / max(1, n - 1) - 1.0
    a = (0.5 * (1.0 + np.cos(np.pi * x))) ** alpha
    a = np.maximum(a, 1e-6)
    S = np.sum(a)
    return a / S


def parametric_search(n, time_budget=15.0, seed=0):
    """Optimize cosine shape parameters via Nelder-Mead."""
    start = time.monotonic()
    best_obj = np.inf
    best_params = None
    n_modes = 5
    n_params = 1 + 2 * n_modes

    for trial in range(6):
        if time.monotonic() - start > time_budget:
            break
        rng = np.random.default_rng(seed + trial * 999)
        x0 = np.concatenate([
            [1.0],
            rng.uniform(-0.4, 0.4, n_modes),
            rng.uniform(0, 2 * np.pi, n_modes)
        ])

        def obj_func(params):
            if time.monotonic() - start > time_budget:
                return 100.0
            base = params[0]
            k_arr = np.arange(n)
            aa = np.full(n, base)
            for m in range(1, n_modes + 1):
                c = params[m]
                phi = params[n_modes + m]
                aa += c * np.cos(2 * np.pi * m * k_arr / n + phi)
            aa = np.maximum(aa, 0)
            S_val = np.sum(aa)
            if S_val < 1e-10:
                return 100.0
            aa = aa / S_val
            return compute_obj(aa)

        try:
            res = minimize(obj_func, x0, method='Nelder-Mead',
                          options={'maxiter': 200, 'xatol': 1e-8, 'fatol': 1e-10,
                                   'adaptive': True})
            if res.fun < best_obj:
                best_obj = res.fun
                best_params = res.x.copy()
        except Exception:
            pass

    if best_params is not None:
        k = np.arange(n)
        a = np.full(n, best_params[0])
        for m in range(1, n_modes + 1):
            c = best_params[m]
            phi = best_params[n_modes + m]
            a += c * np.cos(2 * np.pi * m * k / n + phi)
        a = np.maximum(a, 0)
        S = np.sum(a)
        if S > 1e-10:
            return a / S, best_obj

    return None, np.inf


def refine_full(a, time_budget, use_lp=True, verbose=False, method='auto'):
    """Full refinement pipeline combining multiple methods."""
    n = len(a)
    a = a.copy().astype(np.float64)
    S = np.sum(a)
    if S < 1e-10:
        return a, np.inf
    a = a / S

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    round_num = 0

    while time.monotonic() - start < time_budget - 2:
        round_num += 1
        remaining = time_budget - (time.monotonic() - start)
        if remaining < 3:
            break

        # Phase A: Mirror descent (log-sum-exp)
        md_t = min(remaining * 0.4, 25.0)
        a_md = mirror_descent(best_a, max_iters=int(md_t * 40), time_budget=md_t)
        obj_md = compute_obj(a_md)
        if obj_md < best_obj:
            best_obj = obj_md
            best_a = a_md.copy()
            if verbose:
                print(f"    R{round_num} MD-smooth: {best_obj:.6f}")

        # Phase B: Top-K mirror descent (different gradient)
        remaining = time_budget - (time.monotonic() - start)
        if remaining > 8:
            tk_t = min(remaining * 0.25, 15.0)
            a_tk = mirror_descent_topk(best_a, max_iters=int(tk_t * 40),
                                       time_budget=tk_t, topk=50)
            obj_tk = compute_obj(a_tk)
            if obj_tk < best_obj:
                best_obj = obj_tk
                best_a = a_tk.copy()
                if verbose:
                    print(f"    R{round_num} MD-topk: {best_obj:.6f}")

        # Phase C: LP refinement
        remaining = time_budget - (time.monotonic() - start)
        if remaining > 10 and use_lp and n <= 600:
            lp_t = min(remaining * 0.3, 20.0)
            a_lp = lp_refine(best_a, max_iters=15, time_budget=lp_t)
            obj_lp = compute_obj(a_lp)
            if obj_lp < best_obj:
                best_obj = obj_lp
                best_a = a_lp.copy()
                if verbose:
                    print(f"    R{round_num} LP: {best_obj:.6f}")

        # Phase D: Perturbation escape
        remaining = time_budget - (time.monotonic() - start)
        if remaining > 6:
            rng = np.random.default_rng(round_num * 42 + 17)
            a_pert = best_a * np.exp(rng.normal(0, 0.07, n))
            a_pert = np.maximum(a_pert, 0)
            S_p = np.sum(a_pert)
            if S_p > 1e-10:
                a_pert = a_pert / S_p
                a_pert = mirror_descent(a_pert, max_iters=400,
                                        time_budget=min(5.0, remaining * 0.3))
                obj_pert = compute_obj(a_pert)
                if obj_pert < best_obj:
                    best_obj = obj_pert
                    best_a = a_pert.copy()
                    if verbose:
                        print(f"    R{round_num} Pert: {best_obj:.6f}")

    return best_a, best_obj


def propose_candidate(seed=42, budget_s=1500):
    """
    Main search function. Combines FFT-accelerated smooth gradient mirror descent,
    parametric shape optimization, LP refinement, and multi-start exploration
    to minimize 2*n*max(conv(a,a))/(sum(a))^2.
    """
    start_time = time.monotonic()
    deadline = start_time + float(budget_s) - 20.0

    best_seq = None
    best_score = np.inf

    # Load parent sequence
    n_parent = 500
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64)
        parent = np.clip(parent, 0, 1000.0)
        if parent.size > 0 and np.isfinite(parent).all():
            parent = parent / np.sum(parent)
            parent_obj = compute_obj(parent)
            best_score = parent_obj
            best_seq = parent.copy()
            n_parent = len(parent)
            print(f"Parent (n={n_parent}): obj={parent_obj:.6f}")
    except Exception:
        pass

    if best_seq is None:
        best_seq = np.ones(500) / 500
        best_score = compute_obj(best_seq)
        print(f"Default (n=500): obj={best_score:.6f}")

    # === Phase 1: Diverse initialization and quick optimization ===
    print("\n=== Phase 1: Diverse initializations ===")
    candidate_pool = []

    # Include parent
    candidate_pool.append((best_seq.copy(), best_score))

    lengths = sorted(set([
        max(100, n_parent - 100), max(100, n_parent - 50),
        n_parent,
        min(1200, n_parent + 50), min(1200, n_parent + 100), min(1200, n_parent + 150)
    ]))

    for n in lengths:
        if time.monotonic() > deadline:
            break
        remaining = deadline - time.monotonic()
        if remaining < 15:
            break

        inits = [
            cosine_init(n, n_modes=8, seed=seed + n),
            gaussian_init(n, sigma=0.25, center=0.5),
            gaussian_init(n, sigma=0.35, center=0.5),
            raised_cosine_init(n, alpha=2.0),
            raised_cosine_init(n, alpha=3.0),
        ]

        for i, a_init in enumerate(inits):
            if time.monotonic() > deadline:
                break
            obj_init = compute_obj(a_init)

            # Quick mirror descent
            a_quick = mirror_descent(a_init, max_iters=800, time_budget=min(4.0, remaining * 0.04))
            obj_quick = compute_obj(a_quick)

            best_local = a_quick if obj_quick < obj_init else a_init
            best_local_obj = min(obj_quick, obj_init)
            candidate_pool.append((best_local, best_local_obj))

            if best_local_obj < best_score:
                best_score = best_local_obj
                best_seq = best_local.copy()
                print(f"  n={n}, init{i}: {best_local_obj:.6f} ** NEW BEST **")

    # Parametric search
    for n in [n_parent, max(100, n_parent - 50), min(1000, n_parent + 100)]:
        if time.monotonic() > deadline:
            break
        remaining = deadline - time.monotonic()
        if remaining < 10:
            break
        a_param, obj_param = parametric_search(n, time_budget=min(8.0, remaining * 0.08),
                                               seed=seed + n)
        if a_param is not None:
            candidate_pool.append((a_param, obj_param))
            if obj_param < best_score:
                best_score = obj_param
                best_seq = a_param.copy()
                print(f"  Parametric n={n}: {best_score:.6f} ** NEW BEST **")

    # Sort candidates
    candidate_pool.sort(key=lambda x: x[1])

    # === Phase 2: Refine top candidates ===
    print("\n=== Phase 2: Refine top candidates ===")
    n_refine = min(4, len(candidate_pool))
    for idx in range(n_refine):
        if time.monotonic() > deadline:
            break
        remaining = deadline - time.monotonic()
        if remaining < 30:
            break

        a_cand, obj_cand = candidate_pool[idx]
        if obj_cand > best_score + 0.03:
            break

        n_cand = len(a_cand)
        alloc = min(remaining * 0.3, 60.0)
        print(f"  Refining cand {idx} (n={n_cand}, obj={obj_cand:.6f}), alloc={alloc:.0f}s")

        a_ref, obj_ref = refine_full(a_cand, time_budget=alloc,
                                     use_lp=(n_cand <= 600), verbose=True)
        if obj_ref < best_score:
            best_score = obj_ref
            best_seq = a_ref.copy()
            print(f"    ** NEW BEST: n={n_cand}, obj={best_score:.6f} **")

    # === Phase 3: Intensive refinement loop ===
    print("\n=== Phase 3: Intensive refinement ===")
    a_cur = best_seq.copy()
    n_best = len(a_cur)

    for round_num in range(20):
        if time.monotonic() > deadline - 20:
            break
        remaining = deadline - time.monotonic()
        if remaining < 15:
            break

        alloc = min(remaining * 0.2, 40.0)
        a_new, obj_new = refine_full(a_cur, time_budget=alloc,
                                     use_lp=(n_best <= 600), verbose=False)

        if obj_new < best_score:
            best_score = obj_new
            best_seq = a_new.copy()
            a_cur = a_new.copy()
            print(f"  R{round_num}: {best_score:.6f} ** NEW BEST **")
        else:
            print(f"  R{round_num}: {obj_new:.6f} (best {best_score:.6f})")
            # Perturbation escapes
            rng = np.random.default_rng(round_num * 777 + 333)
            for pert_scale in [0.05, 0.1, 0.2]:
                if time.monotonic() > deadline - 10:
                    break
                a_pert = a_cur * np.exp(rng.normal(0, pert_scale, n_best))
                a_pert = np.maximum(a_pert, 0)
                S_p = np.sum(a_pert)
                if S_p < 1e-10:
                    continue
                a_pert = a_pert / S_p
                a_pert, obj_pert = refine_full(a_pert, time_budget=min(12.0, remaining * 0.1),
                                               use_lp=False, verbose=False)
                if obj_pert < best_score:
                    best_score = obj_pert
                    best_seq = a_pert.copy()
                    a_cur = a_pert.copy()
                    print(f"  R{round_num} pert({pert_scale}): {best_score:.6f} ** NEW BEST **")

    # === Phase 4: Length exploration via interpolation ===
    print("\n=== Phase 4: Length exploration ===")
    if time.monotonic() < deadline - 30:
        n_base = len(best_seq)
        n_tries = sorted(set([
            max(100, n_base - 100), max(100, n_base - 50),
            n_base + 50, n_base + 100, n_base + 150
        ]))
        n_tries = [n for n in n_tries if 100 <= n <= 1500]

        for n_try in n_tries:
            if time.monotonic() > deadline - 10:
                break
            remaining = deadline - time.monotonic()
            if remaining < 15:
                break

            x_old = np.linspace(0, 1, n_base)
            x_new = np.linspace(0, 1, n_try)
            a_interp = np.interp(x_new, x_old, best_seq)
            a_interp = np.maximum(a_interp, 0)
            S_i = np.sum(a_interp)
            if S_i > 1e-10:
                a_interp = a_interp / S_i

            alloc = min(remaining * 0.3, 25.0)
            a_new, obj_new = refine_full(a_interp, time_budget=alloc,
                                         use_lp=(n_try <= 600), verbose=False)
            if obj_new < best_score:
                best_score = obj_new
                best_seq = a_new.copy()
                print(f"  n={n_try}: {best_score:.6f} ** NEW BEST **")
            else:
                print(f"  n={n_try}: {obj_new:.6f}")

    # === Phase 5: Final polish ===
    print("\n=== Phase 5: Final polish ===")
    a_cur = best_seq.copy()
    n_final = len(a_cur)

    if time.monotonic() < deadline - 5:
        remaining = deadline - time.monotonic()

        # Extended smooth mirror descent
        a_md = mirror_descent(a_cur, max_iters=int(remaining * 0.4 * 30),
                              time_budget=remaining * 0.4)
        obj_md = compute_obj(a_md)
        if obj_md < best_score:
            best_score = obj_md
            best_seq = a_md.copy()
            a_cur = a_md.copy()
            print(f"  Final MD: {best_score:.6f} ** NEW BEST **")

        # Top-K variant
        remaining = deadline - time.monotonic()
        if remaining > 10:
            a_tk = mirror_descent_topk(a_cur, max_iters=int(remaining * 0.3 * 30),
                                       time_budget=remaining * 0.3, topk=60)
            obj_tk = compute_obj(a_tk)
            if obj_tk < best_score:
                best_score = obj_tk
                best_seq = a_tk.copy()
                a_cur = a_tk.copy()
                print(f"  Final TopK: {best_score:.6f} ** NEW BEST **")

        # Final LP
        remaining = deadline - time.monotonic()
        if remaining > 10 and n_final <= 600:
            a_lp = lp_refine(a_cur, max_iters=30, time_budget=min(remaining * 0.5, 40.0))
            obj_lp = compute_obj(a_lp)
            if obj_lp < best_score:
                best_score = obj_lp
                best_seq = a_lp.copy()
                print(f"  Final LP: {best_score:.6f} ** NEW BEST **")

    # Final validation with evaluate_sequence
    if best_seq is None:
        best_seq = np.ones(300) / 300
        best_score = compute_obj(best_seq)

    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    val = evaluate_sequence(result)
    if val < np.inf and val < best_score:
        best_score = val
    elif val == np.inf:
        result = [1.0] * 300
        best_score = 2.0

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_score:.6f}, time={elapsed:.1f}s ===")

    return result
