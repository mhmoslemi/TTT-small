def compute_conv(a):
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    conv = np.fft.irfft(fa * fa, size)
    return conv[:2 * n - 1]

def compute_obj(a):

def get_gradient_and_weights(a, T, var_weight=0.0):
    """Compute gradient of smoothed objective.
    
    Objective (for sum(a)=1): 2n * LSE(conv/T) + var_weight * 2n * Var(conv)
    Returns (gradient, weights, conv, lse_value).

def resample(a, new_n):
    """Resample sequence to new length."""
    n = len(a)

    old_k = np.linspace(0, 1, n)

    a_new = np.interp(new_k, old_k, a)
    a_new = np.maximum(a_new, 0.001)
    a_new = a_new / np.sum(a_new)
    return a_new
