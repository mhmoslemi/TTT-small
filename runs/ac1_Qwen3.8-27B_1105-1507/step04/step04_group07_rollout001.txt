import numpy as np
from scipy.optimize import linprog
from scipy.sparse import csr_matrix
import time

_spar_cache = {}

def get_sparsity(n):
    if n not in _spar_cache:
        rows = []
        cols = []
        for k in range(2 * n - 1):
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            rows.extend([k] * (je - js + 1))
            cols.extend(range(js, je + 1))
        _spar_cache[n] = (np.array(rows), np.array(cols), len(rows))
    return _spar_cache[n]

def get_A_matrix(n):
    rows, cols, nnz = get_sparsity(n)
    # Build the full matrix values lazily
    return rows, cols, nnz

def fast_conv(a):
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]

def compute_J(a):
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))

def lp_best_response(a, M, time_limit=5.0):
    n = len(a)
    rows, cols, nnz = get_sparsity(n)
    vals = np.empty(nnz)
    for i in range(nnz):
        k = rows[i]
        j = cols[i]
        idx = k - j
        if 0 <= idx < n:
            vals[i] = a[idx]
        else:
            vals[i] = 0.0
    A = csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                options={'presolve': True, 'time_limit': time_limit})
    if r.success and r.x is not None:
        g = np.maximum(r.x, 0.0)
        if g.sum() > 1e-10:
            return g, r
    return None, None

def vectorized_subgradient_step(a, lr, num_active_tol=1e-8):

def vect_subgrad(a, lr, momentum=0.0):
    """Vectorized subgradient using FFT-based cross-correlation."""
    b = fast_conv(a)
    M = b.max()
    active = np.where(b >= M - 1e-10 * max(M, 1e-15))[0]
    n = len(a)
    # For self-convolution: d(b_k)/d(a_j) = a[k-j] (if 0<=k-j<n)
    # Sum over active k: grad[j] = sum_{k in active} a[k-j] (if 0<=k-j<n)
    # This is a cross-correlation of 'active indicator' with 'a'
    active_vec = np.zeros(len(b))
    active_vec[active] = 1.0 / len(active)
    # grad[j] = sum_k active_vec[k] * a[k-j] for 0<=k-j<n
    # = cross_correlation(active_vec, a)[j] where cross_corr[x][j] = sum_k active_vec[k]*a[k-j]
    # This is the convolution of active_vec with reversed a, shifted appropriately
    # Actually: sum_k active_vec[k] * a[k-j] = (active_vec * reverse(a))[n-1-j] ... let me think
    # Let's just use the direct formula via FFT
    # grad[j] = sum_{k: 0<=k-j<n} active_vec[k] * a[k-j]
    # Let m = k-j, then k = j+m, so: grad[j] = sum_m active_vec[j+m] * a[m] for m in range
    # This is a correlation! grad = correlate(active_vec, a)
    s = 1
    while s < 3 * n:
        s <<= 1
    fa = np.fft.rfft(a, s)
    fm = np.fft.rfft(active_vec, s)
    # Correlation: sum_k fm[k] * fa[k-j] -> IFFT of FM * conj(FA)
    corr = np.fft.irfft(fm * np.conj(fa), s)
    # corr[j] = sum_k active_vec[k] * a[k-j] (circular), but we want linear
    # For our purpose, since active_vec is zero-padded to length 2n-1 and a is length n,
    # the linear result should be in corr[0:n] or a shifted portion
    grad = corr[:n].copy()
    # Normalize
    grad /= max(len(active), 1)
    return grad, M, active

def run_lp_opt(a, max_iters, deadline, tracker, tighten=0.01, lp_time=5.0):
    """Core LP optimization loop with progressive tightening."""
    a = np.maximum(a.astype(np.float64), 1e-8)
    a /= a.sum()
    n = len(a)
    current_J = compute_J(a)
    tracker.update(a)
    no_improve_count = 0
    current_tighten = tighten * 0.5

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        b = fast_conv(a)
        M = float(b.max())
        # Progressive tightening: start loose, get tighter over iterations
        progress = it / max(1, max_iters)
        effective_tighten = current_tighten + progress * tighten * 1.5
        M_tight = M * (1.0 - effective_tighten)
        
        g, lp_result = lp_best_response(a, M_tight, time_limit=lp_time)
        if g is None:
            M_tight = M * (1.0 - effective_tighten * 0.5)
            g, lp_result = lp_best_response(a, M_tight, time_limit=lp_time)
        if g is None:
            g, lp_result = lp_best_response(a, M, time_limit=lp_time)
        if g is None:
            break
        g = np.maximum(g, 0.0)
        gs = g.sum()
        if gs < 1e-10:
            break
        g /= gs

        # Adaptive line search
        best_a = None
        best_J = current_J
        # Try multiple step sizes with golden-section-like spacing
        candidate_ts = []
        base_t = max(0.1, 0.6 * (1.0 - 0.5 * progress))
        candidate_ts = [base_t, base_t * 0.5, base_t * 1.5, 0.3, 0.7, 0.9, 0.1]
        candidate_ts = sorted(set([t for t in candidate_ts if 0 < t <= 1.0]), reverse=True)
        
        for t in candidate_ts:
            a_trial = (1.0 - t) * a + t * g
            a_trial = np.maximum(a_trial, 1e-10)
            ss = a_trial.sum()
            if ss < 1e-10:
                continue
            a_trial /= ss
            J_trial = compute_J(a_trial)
            if J_trial < best_J - 1e-14:
                best_J = J_trial
                best_a = a_trial.copy()
            if J_trial > best_J:
                break  # Past minimum in this direction

        if best_a is None:
            no_improve_count += 1
            if no_improve_count > 5:
                break
            # Try smaller steps
            for t in [0.05, 0.02, 0.01]:
                a_trial = (1.0 - t) * a + t * g
                a_trial = np.maximum(a_trial, 1e-10)
                a_trial /= a_trial.sum()
                J_trial = compute_J(a_trial)
                if J_trial < best_J - 1e-14:
                    best_J = J_trial
                    best_a = a_trial.copy()
                    break
            if best_a is None:
                continue
        else:
            no_improve_count = 0

        a = best_a
        current_J = best_J
        if it % 5 == 0:
            tracker.update(a)
    tracker.update(a)
    return a

def subgradient_descent(a, max_iters, deadline, tracker):
    """Vectorized subgradient descent on the max convolution value."""
    a = np.maximum(a.astype(np.float64), 1e-8)
    a /= a.sum()
    n = len(a)
    current_J = compute_J(a)
    tracker.update(a)
    lr = 2.0 / n
    v = np.zeros(n)
    momentum = 0.7
    stall = 0

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        b = fast_conv(a)
        M = b.max()
        active = np.where(b >= M - 1e-10 * max(M, 1e-15))[0]
        
        # Compute subgradient via FFT correlation
        n_b = len(b)
        active_vec = np.zeros(n_b)
        active_vec[active] = 1.0 / len(active)
        
        s = 1
        while s < n_b + n:
            s <<= 1
        fa = np.fft.rfft(a, s)
        fm = np.fft.rfft(active_vec, s)
        corr = np.fft.irfft(fm * np.conj(fa), s)
        grad = corr[:n].copy()
        
        # Direction: decrease max -> move against subgradient, but also penalize large a
        # Full gradient of J (normalized s=1): dJ/da_j proportional to grad_j - 2*M*a_j
        direction = grad - 2.0 * M * a
        
        v = momentum * v + (1.0 - momentum) * direction
        a_new = a - lr * v
        a_new = np.maximum(a_new, 1e-8)
        a_new /= a_new.sum()
        J_new = compute_J(a_new)
        
        if J_new < current_J - 1e-15:
            a = a_new
            current_J = J_new
            lr = min(lr * 1.02, 10.0 / n)
            stall = 0
        else:
            lr *= 0.7
            v *= 0.5
            stall += 1
            if stall > 30:
                break
        if it % 20 == 0:
            tracker.update(a)
    tracker.update(a)
    return a

def lse_gradient_descent(a, max_iters, deadline, tracker):
    """Log-sum-exp smoothed gradient descent for differentiability."""
    a = np.maximum(a.astype(np.float64), 1e-6)
    a /= a.sum()
    n = len(a)
    current_J = compute_J(a)
    tracker.update(a)
    tau = 30.0
    lr = 1.0 / (n * tau)
    v = np.zeros(n)
    momentum = 0.8
    stall = 0

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        b = fast_conv(a)
        b_max = b.max()
        exp_b = np.exp(tau * (b - b_max))
        p = exp_b / exp_b.sum()
        
        # Gradient of LSE(a*a) w.r.t. a: 2 * cross_correlation(p, a)[:n]
        n_b = len(b)
        s = 1
        while s < n_b + n:
            s <<= 1
        fp = np.fft.rfft(p, s)
        fa = np.fft.rfft(a, s)
        corr = np.fft.irfft(fp * np.conj(fa), s)
        grad = 2.0 * corr[:n]
        
        v = momentum * v + (1.0 - momentum) * grad
        a_new = a - lr * v
        a_new = np.maximum(a_new, 1e-6)
        a_new /= a_new.sum()
        J_new = compute_J(a_new)
        
        if J_new < current_J - 1e-15:
            a = a_new
            current_J = J_new
            tau = min(200.0, tau * 1.01)
            lr = 1.0 / (n * tau)
            stall = 0
        else:
            lr *= 0.5
            v *= 0.3
            stall += 1
            if stall > 40:
                break
        if it % 50 == 0:
            tracker.update(a)
    tracker.update(a)
    return a

def flat_spectrum_init(n, rng):
    """Generate sequence with approximately flat power spectrum."""
    half = n // 2 + 1
    angles = rng.uniform(0, 2 * np.pi, half)
    A = np.zeros(n, dtype=complex)
    A[:half] = np.exp(1j * angles[:half])
    for k in range(1, n - n // 2):
        A[k] = np.conj(A[n - k])
    a = np.fft.ifft(A).real
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-10:
        return np.ones(n) / n
    return a / s

def resample_seq(src, dest_n):
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)

    inits.append(np.exp(-8 * (t - 0.5) ** 2) + 0.1)
    # Double bell
    inits.append(np.exp(-15 * (t - 0.25) ** 2) + np.exp(-15 * (t - 0.75) ** 2) + 0.05)
    # S-curve
    inits.append(1.0 / (1.0 + np.exp(-10 * (t - 0.5))) + 0.1)
    
    for x in inits:
        x = np.maximum(x, 0.001).astype(np.float64)
        x /= x.sum()
    return inits

class ScoreTracker:
    def __init__(self):
        self.best = None
        self.best_score = float('inf')
        self.t_start = time.monotonic()

    def update(self, a):
        a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
        if a.size == 0 or a.sum() < 0.01:
            return
        score = float(evaluate_sequence(a.tolist()))
        if np.isfinite(score) and score < self.best_score:
            self.best = a.copy()
            self.best_score = score
            elapsed = time.monotonic() - self.t_start
            print(f"  * J={score:.8f} n={a.size} t={elapsed:.1f}s", flush=True)

    def report(self, label=""):
        if self.best is not None:
            print(f"  {label} best: J={self.best_score:.8f} n={self.best.size}", flush=True)

def simulated_annealing(a, deadline, tracker, n_restarts=30, t0=0.12):
    rng = np.random.default_rng(777)
    n = a.size
    a = a.copy().astype(np.float64)
    current_J = compute_J(a)
    tracker.update(a)
    temp = t0

    for r in range(n_restarts):
        if time.monotonic() > deadline:
            break
        a_pert = a + rng.normal(0, temp, n)
        a_pert = np.maximum(a_pert, 1e-6)
        a_pert /= a_pert.sum()

        a_refined = run_lp_opt(a_pert, max(5, 15 - r // 8), deadline, tracker, tighten=0.02, lp_time=3.0)

        J = compute_J(a_refined)
        delta = J - current_J
        if delta < 0 or rng.random() < np.exp(-delta / max(temp, 1e-10)):
            a = a_refined
            current_J = J
        temp *= 0.93
    tracker.update(a)
    return a

def coordinate_descent(a, max_steps, deadline, tracker):
    a = a.copy().astype(np.float64)
    a /= a.sum()
    n = len(a)
    current_J = compute_J(a)
    tracker.update(a)
    rng = np.random.default_rng(42)
    stall = 0

    for step in range(max_steps):
        if time.monotonic() > deadline:
            break
        j = rng.integers(0, n)
        improved = False
        for scale in [0.5, 1.5, 0.1, 2.0, 0.8, 1.2, 0.7]:
            a_trial = a.copy()
            a_trial[j] *= scale
            a_trial = np.maximum(a_trial, 1e-8)
            ss = a_trial.sum()
            if ss < 1e-10:
                continue
            a_trial /= ss
            J_trial = compute_J(a_trial)
            if J_trial < current_J - 1e-14:
                a = a_trial
                current_J = J_trial
                improved = True
                break
        if not improved:
            stall += 1
            if stall > 100:
                break
        else:
            stall = 0
        if step % 200 == 0:
            tracker.update(a)
    tracker.update(a)
    return a

def multi_start_search(n, rng, parent, deadline, tracker, time_budget):
    """Search from multiple initializations for a given length."""
    inits = generate_inits(n, rng, parent)
    best_for_n = None
    best_J_for_n = float('inf')
    
    # Time per init
    n_inits = len(inits)
    per_init_time = time_budget / n_inits
    
    for idx, a0 in enumerate(inits):
        if time.monotonic() > deadline - 10:
            break
        t_start = time.monotonic()
        elapsed_in_init = time.monotonic() - t_start
        
        # LP phase - main optimization
        n_lp_iters = max(8, int(40 * per_init_time / 15.0))
        lp_deadline = min(deadline - 30, t_start + elapsed_in_init + per_init_time * 0.5)
        a = run_lp_opt(a0, n_lp_iters, lp_deadline, tracker, tighten=0.012, lp_time=4.0)
        
        # Subgradient polish
        if time.monotonic() < deadline - 20:
            subgrad_deadline = min(deadline - 20, t_start + per_init_time * 0.8)
            a = subgradient_descent(a, max(20, int(5000 / n)), subgrad_deadline, tracker)
        
        # LSE polish
        if time.monotonic() < deadline - 10:
            lse_deadline = min(deadline - 10, t_start + per_init_time)
            a = lse_gradient_descent(a, max(50, int(10000 / n)), lse_deadline, tracker)
        
        J = compute_J(a)
        if J < best_J_for_n:
            best_J_for_n = J
            best_for_n = a.copy()
    
    if best_for_n is not None:
        tracker.update(best_for_n)
    return best_for_n, best_J_for_n

def propose_candidate(seed=42, budget_s=1500, **kwargs):
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + float(budget_s) - 10.0
    tracker = ScoreTracker()

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        if p.ndim == 1 and p.size > 0 and np.all(np.isfinite(p)):
            parent = np.maximum(p, 0.0)
            tracker.update(parent)
            print(f"Parent: n={parent.size}, J={tracker.best_score:.8f}", flush=True)
    except (NameError, AttributeError):
        pass

    # === Phase 1: Multi-length LP optimization ===
    lengths = [256, 512]
    # Allocate time: give more to 512 since previous best was there
    remaining = deadline - time.monotonic() - 200
    time_256 = remaining * 0.3
    time_512 = remaining * 0.7

    all_best = {}
    
    for n in lengths:
        t_budget = time_256 if n == 256 else time_512
        t_end = time.monotonic() + t_budget
        print(f"\n{'='*50} Phase1 n={n} budget={t_budget:.0f}s {'='*50}", flush=True)
        
        best_n, best_J_n = multi_start_search(n, rng, parent, t_end, tracker, t_budget)
        all_best[n] = (best_n, best_J_n)
        
        if best_n is not None:
            tracker.report(f"Phase1 n={n}: J={best_J_n:.8f}")
    
    # Determine which length to focus on
    if 512 in all_best and all_best[512][1] < float('inf'):
        focus_n = 512
    else:
        focus_n = 256
    
    # === Phase 2: Focused deep optimization ===
    if tracker.best is not None and time.monotonic() < deadline - 150:
        t0 = time.monotonic()
        n = tracker.best.size
        print(f"\n{'='*50} Phase2 Deep n={n} {'='*50}", flush=True)
        
        a = tracker.best.copy().astype(np.float64)
        
        # Multiple LP passes with varying tightening
        tighten_sched = [0.003, 0.006, 0.01, 0.015, 0.02, 0.025, 0.03, 0.035]
        for tight in tighten_sched:
            if time.monotonic() > deadline - 120:
                break
            a = run_lp_opt(a, 25, deadline - 120, tracker, tighten=tight, lp_time=4.0)
        
        # Subgradient
        if time.monotonic() < deadline - 80:
            a = subgradient_descent(a, max(100, int(20000 / n)), deadline - 80, tracker)
        
        # LSE
        if time.monotonic() < deadline - 50:
            a = lse_gradient_descent(a, max(200, int(50000 / n)), deadline - 50, tracker)
        
        # Another LP round
        if time.monotonic() < deadline - 30:
            a = run_lp_opt(a, 20, deadline - 30, tracker, tighten=0.01, lp_time=3.0)
        
        tracker.report(f"Phase2 ({time.monotonic()-t0:.1f}s)")

    # === Phase 3: Simulated annealing ===
    if tracker.best is not None and time.monotonic() < deadline - 80:
        t0 = time.monotonic()
        n = tracker.best.size
        print(f"\n{'='*50} Phase3 SA n={n} {'='*50}", flush=True)
        a = simulated_annealing(tracker.best, deadline - 60, tracker, n_restarts=30, t0=0.08)
        # Polish after SA
        if time.monotonic() < deadline - 30:
            a = run_lp_opt(a, 15, deadline - 30, tracker, tighten=0.01, lp_time=3.0)
            if time.monotonic() < deadline - 15:
                a = lse_gradient_descent(a, max(100, int(20000 / n)), deadline - 15, tracker)
        tracker.report(f"Phase3 ({time.monotonic()-t0:.1f}s)")

    # === Phase 4: Coordinate descent + final polish ===
    if tracker.best is not None and time.monotonic() < deadline - 15:
        t0 = time.monotonic()
        print(f"\n{'='*50} Phase4 Final Polish {'='*50}", flush=True)
        a = coordinate_descent(tracker.best, 5000, deadline - 10, tracker)
        if time.monotonic() < deadline - 5:
            a = run_lp_opt(a, 10, deadline - 3, tracker, tighten=0.005, lp_time=2.0)
            if time.monotonic() < deadline - 1:
                a = lse_gradient_descent(a, 300, deadline - 0.5, tracker)
        tracker.report(f"Phase4 ({time.monotonic()-t0:.1f}s)")

    # Fallback
    if tracker.best is None:
        tracker.best = np.ones(256)
        tracker.best_score = compute_J(tracker.best)
        tracker.update(tracker.best)

    print(f"\n{'='*50} FINAL {'='*50}", flush=True)
    print(f"J={tracker.best_score:.8f} n={tracker.best.size} time={time.monotonic()-tracker.t_start:.1f}s", flush=True)
    return [float(x) for x in tracker.best]
