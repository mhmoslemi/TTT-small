def lp_mirror_descent(a, max_iters, deadline, tracker, tighten=0.003):
    """LP mirror descent with adaptive tightening."""
    a = np.maximum(a.astype(np.float64), 0.001)
    a /= a.sum()
    n = len(a)
    current_J = compute_J(a)
    tracker.update(a)
    stall = 0
    
    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        
        b = fast_conv(a)
        M = float(b.max())
        M_target = M * (1.0 - tighten)
        
        # Build LP: max sum(g) s.t. (a*g)[k] <= M_target, g >= 0
        # A * g <= M_target * ones
        # a*g has (a*g)[k] = sum_j a[j]*g[k-j] for valid j, k-j
        # This is a Toeplitz system: A[k,j] = a[k-j] if 0<=k-j<n else 0
        
        rows = []
        cols = []
        vals = []
        for k in range(2 * n - 1):
            jl = max(0, k - n + 1)
            jh = min(n - 1, k)
            for j in range(jl, jh + 1):
                rows.append(k)
                cols.append(j)
                vals.append(a[k - j])
        
        A = csr_matrix((vals, (rows, cols)), shape=(2*n-1, n))
        c = -np.ones(n)
        b_ub = np.full(2*n-1, M_target)
        bounds = [(0.0, None)] * n
        
        r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
        
        if r.success and r.x is not None:
            g = np.maximum(r.x, 0.0)
            g_sum = g.sum()
            if g_sum > 1e-10:
                g /= g_sum
                # Try step sizes
                best_a = None
                best_J = current_J
                for t in [0.5, 0.3, 0.2, 0.1, 0.7, 0.8]:
                    a_trial = (1-t)*a + t*g
                    a_trial = np.maximum(a_trial, 0.0)
                    if a_trial.sum() < 1e-10:
                        continue
                    a_trial /= a_trial.sum()
                    J_trial = compute_J(a_trial)
                    if J_trial < best_J:
                        best_J = J_trial
                        best_a = a_trial
                
                if best_a is not None:
                    a = best_a
                    current_J = best_J
                    stall = 0
                    if it % 10 == 0:
                        tracker.update(a)
                        print(f"    LP iter {it}: J={current_J:.8f}", flush=True)
                else:
                    stall += 1
                    if stall > 10:
                        break
            else:
                break
        else:
            tighten *= 0.5
            if tighten < 1e-6:
                break
    
    tracker.update(a)
    return a


def resample_seq(src, dest_n):
    """Resample 1D sequence to new length via linear interpolation."""
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)

"""

import numpy as np
from scipy.optimize import linprog
from scipy.sparse import csr_matrix
import time


def fast_conv(a):
    """Full self-convolution via FFT."""
    n = len(a)
    s = 1
    while s < 2 * n - 1:
        s <<= 1
    fa = np.fft.rfft(a, s)
    return np.fft.irfft(fa * fa, s)[:2 * n - 1]


def compute_J(a):
    """Compute J(a) = 2*n*max(conv(a,a))/(sum(a))^2."""
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))


def cd_optimize_1d(a, i, b, S, v):
    """
    Find optimal value for coordinate i using vectorized evaluation.
    a: current sequence, i: coordinate index, b: current conv, S: current sum, v: current a[i].
    Returns optimal x >= 0.
    """
    n = len(a)
    k_lo = max(0, i)
    k_hi = min(i + n - 1, 2 * n - 2)

    # C = max of b for unaffected k
    C = 0.0
    if k_lo > 0:
        C = max(C, float(b[:k_lo].max()))
    if k_hi < 2 * n - 2:
        C = max(C, float(b[k_hi + 1:].max()))

    # Affected positions (k in [k_lo, k_hi], k != 2i)
    two_i = 2 * i
    ks_list = []
    for k in range(k_lo, k_hi + 1):
        if k == two_i:
            continue
        ki = k - i
        if 0 <= ki < n:
            ks_list.append(k)

    if len(ks_list) > 0:
        ks_arr = np.array(ks_list)
        kis = ks_arr - i
        alphas = 2.0 * a[kis]
        bas = b[ks_arr] - 2.0 * v * a[kis]
    else:
        alphas = np.array([0.0])
        bas = np.array([0.0])

    has_quad = (0 <= two_i < 2 * n - 1)
    quad_const = float(b[two_i] - v * v) if has_quad else -1e30

    S_prime = S - v
    if S_prime + v < 1e-20:
        return v

    # Generate candidates
    candidates = [v, 0.0, max(0.0, v * 0.5), max(0.0, v * 1.5), S_prime / n * 1.5]

    if len(ks_list) > 0:
        for j in range(len(alphas)):
            if alphas[j] > 1e-10:
                # Optimal of (alpha*x+beta)/(S'+x)^2
                x_star = (alphas[j] * S_prime - 2.0 * bas[j]) / alphas[j]
                if x_star >= 0 and x_star < 1e6:
                    candidates.append(x_star)
                # Where linear = C
                if C > bas[j]:
                    x_c = (C - bas[j]) / alphas[j]
                    if x_c >= 0 and x_c < 1e6:
                        candidates.append(x_c)
            if has_quad and alphas[j] > 1e-10:
                # Where linear = quad: x^2 - alpha*x + (quad_const - beta) = 0
                disc = alphas[j] ** 2 - 4.0 * (quad_const - bas[j])
                if disc >= 0:
                    sq = np.sqrt(disc)
                    x1 = (alphas[j] + sq) / 2.0
                    x2 = (alphas[j] - sq) / 2.0
                    if 0 <= x1 < 1e6:
                        candidates.append(x1)
                    if 0 <= x2 < 1e6:
                        candidates.append(x2)

    if has_quad:
        if S_prime > 1e-10:
            x_star = quad_const / S_prime
            if 0 <= x_star < 1e6:
                candidates.append(x_star)
        if C > quad_const:
            x_c = np.sqrt(C - quad_const)
            if x_c < 1e6:
                candidates.append(x_c)

    candidates = np.array([max(0.0, c) for c in candidates], dtype=np.float64)
    candidates = candidates[candidates < 1e4]

    if len(candidates) == 0:
        return v

    # Vectorized evaluation
    if len(ks_list) > 0:
        linear_vals = np.outer(candidates, alphas) + bas[np.newaxis, :]
        max_linear = linear_vals.max(axis=1)
    else:
        max_linear = np.zeros(len(candidates))

    if has_quad:
        max_quad = candidates ** 2 + quad_const
    else:
        max_quad = np.full(len(candidates), -1e30)

    max_num = np.maximum.reduce([np.full(len(candidates), C), max_linear, max_quad])
    denom = (S_prime + candidates) ** 2
    denom = np.maximum(denom, 1e-30)
    J_vals = 2.0 * n * max_num / denom

    best_idx = int(np.argmin(J_vals))
    best_x = float(candidates[best_idx])
    best_J = float(J_vals[best_idx])

    # Golden section refinement
    if best_x > 1e-8:
        lo = max(0.0, best_x * 0.5)
        hi = best_x * 2.0 + 0.01

        def _J_scalar(x):
            if len(ks_list) > 0:
                ml = float((alphas * x + bas).max())
            else:
                ml = -1e30
            if has_quad:
                mq = x * x + quad_const
            else:
                mq = -1e30
            mn = max(C, ml, mq)
            d = (S_prime + x) ** 2
            if d < 1e-30:
                return 1e20
            return 2.0 * n * mn / d

        gr = (np.sqrt(5.0) - 1.0) / 2.0
        c_pt = hi - gr * (hi - lo)
        d_pt = lo + gr * (hi - lo)
        fc = _J_scalar(c_pt)
        fd = _J_scalar(d_pt)
        for _ in range(15):
            if fc < fd:
                hi = d_pt
                d_pt = c_pt
                fd = fc
                c_pt = hi - gr * (hi - lo)
                fc = _J_scalar(c_pt)
            else:
                lo = c_pt
                c_pt = d_pt
                fc = fd
                d_pt = lo + gr * (hi - lo)
                fd = _J_scalar(d_pt)

        x_ref = (lo + hi) / 2.0
        J_ref = _J_scalar(x_ref)
        if J_ref < best_J:
            best_x = x_ref
            best_J = J_ref

    return max(0.0, best_x)


def cd_pass(a):
    """One full pass of coordinate descent. Returns (a, J)."""
    a = a.astype(np.float64).copy()
    S = a.sum()
    if S < 1e-10:
        return a, float('inf')
    n = len(a)
    b = fast_conv(a)

    indices = list(range(n))
    np.random.shuffle(indices)

    for i in indices:
        v = a[i]
        x_opt = cd_optimize_1d(a, i, b, S, v)
        if abs(x_opt - v) > 1e-12:
            a[i] = x_opt
            S = S - v + x_opt
            # Update b incrementally
            two_i = 2 * i
            k_start = max(0, i)
            k_end = min(2 * n - 1, i + n)
            for k in range(k_start, k_end):
                ki = k - i
                if 0 <= ki < n:
                    if k == two_i:
                        b[k] += x_opt * x_opt - v * v
                    else:
                        b[k] += 2.0 * (x_opt - v) * a[ki]

    b_check = fast_conv(a)
    new_J = 2.0 * n * b_check.max() / (S * S)
    return a, new_J


def cd_until_converge(a, max_passes, deadline, tracker, track_every=10):
    """Run CD passes until convergence or deadline."""
    a = np.maximum(a.astype(np.float64), 1e-8)
    a /= a.sum()
    current_J = compute_J(a)
    tracker.update(a)
    stall = 0

    for p in range(max_passes):
        if time.monotonic() > deadline:
            break
        a, new_J = cd_pass(a)
        if new_J < current_J - 1e-12:
            improvement = current_J - new_J
            current_J = new_J
            stall = 0
            if p % track_every == 0:
                tracker.update(a)
                print(f"    CD pass {p}: J={current_J:.8f}", flush=True)
        else:
            stall += 1
            if stall > 5:
                break

    tracker.update(a)
    return a


def lp_mirror_step(a, tighten=0.003):
    """One LP mirror descent step. Returns (a_new, J_new) or None."""
    n = len(a)
    a = np.maximum(a, 1e-8)
    b = fast_conv(a)
    M = float(b.max())
    M_target = M * (1.0 - tighten)

    # Build LP: max sum(g) s.t. (a*g)[k] <= M_target, g >= 0
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        jl = max(0, k - n + 1)
        jh = min(n - 1, k)
        for j in range(jl, jh + 1):
            rows.append(k)
            cols.append(j)
            vals.append(a[k - j])

    A = csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M_target)
    bounds = [(0.0, None)] * n

    r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')

    if not r.success or r.x is None:
        return None

    g = np.maximum(r.x, 0.0)
    g_sum = g.sum()
    if g_sum < 1e-10:
        return None
    g /= g_sum

    return g


def lp_run(a, max_iters, deadline, tracker, tighten=0.003):
    """Run LP mirror descent for multiple iterations."""
    a = np.maximum(a.astype(np.float64), 1e-8)
    a /= a.sum()
    current_J = compute_J(a)
    tracker.update(a)
    stall = 0
    current_tighten = tighten

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        g = lp_mirror_step(a, current_tighten)
        if g is None:
            current_tighten *= 0.5
            if current_tighten < 1e-6:
                break
            continue

        # Try step sizes
        best_a = None
        best_J = current_J
        for t in [0.5, 0.3, 0.2, 0.7, 0.1, 0.8]:
            a_trial = (1.0 - t) * a + t * g
            a_trial = np.maximum(a_trial, 0.0)
            s = a_trial.sum()
            if s < 1e-10:
                continue
            a_trial /= s
            J_trial = compute_J(a_trial)
            if J_trial < best_J:
                best_J = J_trial
                best_a = a_trial

        if best_a is not None:
            a = best_a
            current_J = best_J
            stall = 0
            if it % 5 == 0:
                tracker.update(a)
                print(f"    LP iter {it}: J={current_J:.8f}", flush=True)
        else:
            stall += 1
            current_tighten *= 0.7
            if stall > 15:
                break

    tracker.update(a)
    return a


def resample_seq(src, dest_n):
    """Resample 1D sequence to new length via linear interpolation."""
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)


def anneal_and_cd(a, n_perturbs, deadline, tracker, base_temp=0.05):
    """Simulated annealing with CD polish at each step."""
    a = np.maximum(a.astype(np.float64), 1e-8)
    a /= a.sum()
    n = len(a)
    current_J = compute_J(a)
    tracker.update(a)

    for step in range(n_perturbs):
        if time.monotonic() > deadline:
            break

        T = base_temp * (0.9 ** step)

        # Perturb: randomly change a block of coordinates
        block_size = max(3, n // 20)
        start = np.random.randint(0, n - block_size)
        a_trial = a.copy()
        for idx in range(start, start + block_size):
            a_trial[idx] = max(0.0, a_trial[idx] * np.exp(np.random.normal(0, T)))

        s = a_trial.sum()
        if s < 1e-10:
            continue
        a_trial /= s

        J_trial = compute_J(a_trial)

        # Accept with probability
        if J_trial < current_J:
            a = a_trial
            current_J = J_trial
            tracker.update(a)
        else:
            delta = J_trial - current_J
            if delta < 10 * T and np.random.random() < np.exp(-delta / max(T, 1e-10)):
                a = a_trial
                current_J = J_trial

        # Quick CD polish every few steps
        if step % 3 == 0 and time.monotonic() < deadline:
            for _ in range(3):
                if time.monotonic() > deadline:
                    break
                a, new_J = cd_pass(a)
                if new_J < current_J:
                    current_J = new_J
                    tracker.update(a)

    tracker.update(a)
    return a


class ScoreTracker:
    def __init__(self):
        self.best = None
        self.best_score = float('inf')
        self.t_start = time.monotonic()

    def update(self, a):
        a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
        if a.size == 0 or a.sum() < 0.01:
            return
        score = float(evaluate_sequence(a.tolist()))
        if np.isfinite(score) and score < self.best_score:
            self.best = a.copy()
            self.best_score = score
            elapsed = time.monotonic() - self.t_start
            print(f"  * NEW BEST: J={score:.8f} n={a.size} t={elapsed:.0f}s", flush=True)

    def report(self, label=""):
        if self.best is not None:
            print(f"  {label} best: J={self.best_score:.8f} n={self.best.size} t={time.monotonic()-self.t_start:.0f}s", flush=True)

        tracker.report(f"Phase 7 Restarts: {time.monotonic()-t0:.1f}s")

    # === Phase 8: Final intensive CD ===
    print(f"\n{'='*50} Phase 8: Final CD {'='*50}", flush=True)
    t0 = time.monotonic()
    a_final = cd_until_converge(tracker.best, max_passes=100, deadline=deadline - 5, tracker=tracker, track_every=20)
    tracker.report(f"Phase 8 Final: {time.monotonic()-t0:.1f}s")

    # Fallback
    if tracker.best is None:
        tracker.best = np.ones(256)
        tracker.best_score = compute_J(tracker.best)

    # Final verification
    final_score = float(evaluate_sequence([float(x) for x in tracker.best]))
    print(f"\n{'='*50} FINAL RESULT {'='*50}", flush=True)
    print(f"J = {final_score:.10f}, n = {tracker.best.size}", flush=True)
    print(f"Total time: {time.monotonic()-tracker.t_start:.1f}s", flush=True)

    return [float(x) for x in tracker.best]
