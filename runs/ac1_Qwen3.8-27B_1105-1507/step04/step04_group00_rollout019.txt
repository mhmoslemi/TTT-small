    # p has length 2n-1, a has length n
    # correlate(p, a, 'valid')[j] = sum_i p[j+i] * a[i]
    grad = 2.0 * np.correlate(p, a, mode='valid')
    
    return grad, b, p


def nesterov_smooth_descent(a, max_iters, deadline, initial_tau=5.0, final_tau=200.0,
                            lr_init=0.5, lr_min=0.05, momentum_init=0.9, momentum_final=0.999):
    """
    Nesterov accelerated smooth gradient descent.
    """
    a = np.maximum(a.astype(np.float64), 1e-10)
    a = a / a.sum()
    n = len(a)
    
    best_a = a.copy()
    best_M = fast_conv(a).max()
    
    # Nesterov: y = a + v; grad at y; a_new = a - lr*grad; v_new = m*(a_new - a)
    v = np.zeros(n)
    
    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        
        frac = it / max(1, max_iters)
        tau = initial_tau + (final_tau - initial_tau) * (frac ** 0.7)
        lr = lr_init * (lr_min / lr_init) ** frac
        m = momentum_init + (momentum_final - momentum_init) * frac
        
        # Nesterov lookahead
        y = a + m * v
        y = np.maximum(y, 0.0)
        sy = y.sum()
        if sy < 1e-10:
            y = a
        else:
            y = y / sy
        
        grad, b, p = smooth_grad(y, tau)
        M = b.max()
        
        if M < best_M * (1 - 1e-10):
            best_M = M
            best_a = a.copy()
        
        # Update
        a_new = a - lr * grad
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s < 1e-10:
            a_new = a.copy()
        else:
            a_new = a_new / s
        
        v = a_new - a
        a = a_new
    
    if fast_conv(best_a).max() <= fast_conv(a).max():
        return best_a
    return a


def subgradient_descent(a, max_iters, deadline, lr_init=0.3):
    """Subgradient descent with exact max and Polyak step."""
    a = np.maximum(a.astype(np.float64), 1e-10)
    a = a / a.sum()
    n = len(a)
    
    best_a = a.copy()
    best_M = fast_conv(a).max()
    running_sum = np.zeros(n)
    count = 0
    
    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        
        b = fast_conv(a)
        M = b.max()
        tol = 1e-8 * M
        
        active = np.where(b >= M - tol)[0]
        
        # Subgradient: sum of 2*a[k-j] over active k
        # Use correlate trick: for each active k, add 2*a shifted
        grad = np.zeros(n)
        for k in active:
            lo = max(0, k - n + 1)
            hi = min(n - 1, k)
            grad[lo:hi+1] += 2.0 * a[k-hi:k-lo+1][::-1]
        
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-15:
            break
        grad /= gnorm
        
        lr = lr_init / np.sqrt(it + 1)
        a_new = a - lr * grad
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s < 1e-10:
            a_new = a.copy()
        else:
            a_new = a_new / s
        
        running_sum += a_new
        count += 1
        a = a_new
        
        if M < best_M * (1 - 1e-10):
            best_M = M
            best_a = a.copy()
    
    if count > 10:
        a_avg = running_sum / count
        a_avg = np.maximum(a_avg, 0.0)
        s = a_avg.sum()
        if s > 1e-10:
            a_avg /= s
            M_avg = fast_conv(a_avg).max()
            if M_avg < best_M:
                best_a = a_avg
    
    return best_a


def build_conv_matrix(n, a):
    """Build sparse matrix where row k represents (a*x)[k] = sum_j A[k,j]*x[j]."""
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        j_min = max(0, k - n + 1)
        j_max = min(n - 1, k)
        for j in range(j_min, j_max + 1):
            rows.append(k)
            cols.append(j)
            vals.append(a[k - j])
    A = csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))
    return A


def lp_best_response(a, M):
    """Solve LP: max sum(g) s.t. A@ g <= M, g >= 0."""
    n = len(a)
    A = build_conv_matrix(n, a)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    try:
        r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                    options={'time_limit': 20, 'presolve': True})
        if r.success and r.x is not None:
            g = np.maximum(r.x, 0)
            if g.sum() > 1e-15:
                return g
    except Exception:
        pass
    return None


def lp_mirror_descent(a, max_iters, deadline):
    """LP mirror descent with adaptive step."""
    a = np.maximum(a.astype(np.float64), 1e-10)
    a = a / a.sum()
    
    b = fast_conv(a)
    current_M = float(b.max())
    best_a = a.copy()
    best_M = current_M
    no_improve = 0
    
    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        
        b = fast_conv(a)
        M = float(b.max())
        g = lp_best_response(a, M)
        if g is None:
            break
        g = np.maximum(g, 0.0)
        g_sum = g.sum()
        if g_sum < 1e-10:
            break
        g /= g_sum
        
        # Line search with multiple step sizes
        improved = False
        base_t = max(0.1, 0.8 * (1.0 - 0.5 * it / max(1, max_iters)))
        for t in [base_t, base_t*0.5, base_t*0.25, 0.05, 0.02, 0.01]:
            a_trial = np.maximum((1 - t) * a + t * g, 0)
            s = a_trial.sum()
            if s < 1e-10:
                continue
            a_trial /= s
            M_trial = fast_conv(a_trial).max()
            if M_trial < best_M - 1e-14:
                a = a_trial
                best_M = M_trial
                best_a = a.copy()
                improved = True
                break
        
        if not improved:
            no_improve += 1
            if no_improve > 15:
                break
        else:
            no_improve = 0
    
    return best_a

"""
Minimizes J(a) = 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Vectorized smooth gradient descent (Nesterov momentum, adaptive tau)
2. Exact subgradient descent with Polyak averaging
3. LP mirror descent for precise local refinement
4. High-tau smooth gradient for final convergence to exact max
Key improvement over previous: O(n^2) gradient via numpy correlate (C-optimized),
enabling 5-10x more gradient iterations; better momentum schedule; adaptive step sizes.
"""

import numpy as np
from scipy.optimize import linprog
from scipy.sparse import csr_matrix
import time


def fast_conv(a):
    """Full self-convolution via FFT."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]


def compute_J(a):
    """Compute J(a) = 2*n*max(conv(a,a))/(sum(a))^2."""
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))


def smooth_gradient(a, tau):
    """
    Compute gradient of (1/tau)*log(sum_k exp(tau*b_k)) w.r.t. a.
    Uses vectorized correlate for O(n^2) in C.
    Returns (grad, b, p).
    """
    n = len(a)
    b = fast_conv(a)
    b_max = b.max()
    exp_b = np.exp(tau * (b - b_max))
    p = exp_b / exp_b.sum()
    # grad_j = 2 * sum_k p[k] * a[k-j] = 2 * correlate(p, a, 'valid')
    # p has length 2n-1, a has length n; 'valid' gives length n
    grad = 2.0 * np.correlate(p, a, mode='valid')
    return grad, b, p


def nesterov_smooth_descent(a, max_iters, deadline, initial_tau=5.0, final_tau=150.0,
                            lr_init=0.5, lr_min=0.02, mom_init=0.85, mom_final=0.999):
    """Nesterov accelerated smooth gradient descent with adaptive tau and learning rate."""
    a = np.maximum(a.astype(np.float64), 1e-10)
    a = a / a.sum()
    n = len(a)

    best_a = a.copy()
    best_M = fast_conv(a).max()
    v = np.zeros(n)

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        frac = min(1.0, it / max(1, max_iters))
        tau = initial_tau + (final_tau - initial_tau) * (frac ** 0.6)
        lr = lr_init * (lr_min / lr_init) ** frac
        m = mom_init + (mom_final - mom_init) * (frac ** 0.8)

        # Nesterov lookahead
        y = a + m * v
        y = np.maximum(y, 0.0)
        sy = y.sum()
        if sy < 1e-10:
            y = a.copy()
        else:
            y = y / sy

        grad, b, p = smooth_gradient(y, tau)
        M = b.max()

        if M < best_M * (1 - 1e-10):
            best_M = M
            best_a = a.copy()

        a_new = a - lr * grad
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s < 1e-10:
            a_new = a.copy()
        else:
            a_new = a_new / s

        v = a_new - a
        a = a_new

    M_final = fast_conv(a).max()
    if fast_conv(best_a).max() <= M_final:
        return best_a
    return a


def subgradient_descent(a, max_iters, deadline, lr_init=0.4):

def resample_seq(src, dest_n):
    """Resample 1D sequence to new length via linear interpolation."""
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)

    inits.append(np.ones(n))
    inits.append(rng.uniform(0.5, 1.0, n))

    for freq in [1, 2, 3, 5]:
        inits.append(np.maximum(1.0 + 0.3 * np.cos(2 * np.pi * freq * t_arr), 0.01))
        inits.append(np.maximum(1.0 + 0.4 * np.sin(2 * np.pi * freq * t_arr), 0.01))

    inits.append(np.maximum(1.0 + 0.3 * np.cos(2 * np.pi * 2.5 * t_arr ** 2), 0.01))
    inits.append(np.exp(-4.0 * (t_arr - 0.5) ** 2) + 0.1)
    inits.append(np.exp(-8.0 * (t_arr - 0.3) ** 2) + np.exp(-8.0 * (t_arr - 0.7) ** 2) + 0.05)
    inits.append(np.maximum(1.0 - 0.4 * np.abs(2 * t_arr - 1), 0.05))
    inits.append(0.3 + 0.7 * np.abs(2 * t_arr - 1))

    # Smooth random
    ksize = max(3, n // 20)
    kernel = np.ones(ksize) / ksize
    smooth = np.convolve(rng.uniform(0, 1, n), kernel, mode='same')
    inits.append(np.maximum(smooth + 0.1, 0.01))

    # Block structure
    n_blocks = rng.integers(3, max(4, n // 50))
    block_vals = rng.uniform(0.3, 1.0, n_blocks)
    block_seq = np.repeat(block_vals, n // n_blocks + 1)[:n]
    inits.append(block_seq)

    # Power law
    inits.append(np.maximum(0.2 + 0.8 * t_arr ** 0.5, 0.01))
    inits.append(np.maximum(0.2 + 0.8 * (1 - t_arr) ** 0.5, 0.01))

    return inits


class ScoreTracker:
    def __init__(self):
        self.best = None
        self.best_score = float('inf')
        self.t_start = time.monotonic()

    def update(self, a):
        a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
        if a.size == 0 or a.sum() < 0.01:
            return
        try:
            score = float(evaluate_sequence(a.tolist()))
        except Exception:
            return
        if np.isfinite(score) and score < self.best_score:
            self.best = a.copy()
            self.best_score = score
            elapsed = time.monotonic() - self.t_start
            print(f"  ** J={score:.8f} n={a.size} t={elapsed:.0f}s", flush=True)

    def report(self, label=""):
        if self.best is not None:
            print(f"  {label} best: J={self.best_score:.8f} n={self.best.size}", flush=True)


def propose_candidate(seed=42, budget_s=1500, **kwargs):
    """
    Multi-phase hybrid search for minimizing J(a) = 2n*max(a*a)/(sum a)^2.
    Uses vectorized gradient, Nesterov momentum, subgradient, LP mirror descent.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + float(budget_s) - 20.0
    tracker = ScoreTracker()

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        if p.ndim == 1 and p.size > 0 and np.all(np.isfinite(p)):
            parent = np.maximum(p, 0.0)
            tracker.update(parent)
            print(f"Parent loaded: n={parent.size}, J={tracker.best_score:.6f}", flush=True)
    except (NameError, AttributeError):
        pass

    if parent is None:
        parent = np.ones(512) * 1.0
        tracker.update(parent)

    current_best = parent.copy()

    # === PHASE 1: Smooth gradient descent across multiple lengths ===
    p1_budget = 400.0
    p1_start = time.monotonic()
    print(f"\n{'='*60}", flush=True)
    print(f"PHASE 1: Nesterov Smooth Gradient Descent (budget={p1_budget:.0f}s)", flush=True)
    print(f"{'='*60}", flush=True)

    lengths = [256, 384, 512, 640, 768, 1024]
    # Allocate time proportionally to 1/n (shorter = more time per init)
    weights = [1.0 / n for n in lengths]
    total_w = sum(weights)

    for n in lengths:
        if time.monotonic() > deadline - 180:
            break

        time_for_n = p1_budget * (weights[lengths.index(n)] / total_w)
        time_for_n = max(30, min(120, time_for_n))
        phase_deadline = min(time.monotonic() + time_for_n, deadline - 150)

        if time.monotonic() >= phase_deadline:
            continue

        t0 = time.monotonic()
        inits = generate_initializations(n, rng, current_best)
        best_for_n = None
        best_M_for_n = float('inf')

        for idx, a0 in enumerate(inits):
            if time.monotonic() > phase_deadline:
                break
            a0 = np.maximum(a0.astype(np.float64), 1e-8)
            a0 = a0 / a0.sum()

            # Nesterov smooth descent
            a = nesterov_smooth_descent(a0, 100000, phase_deadline,
                                        initial_tau=3.0, final_tau=100.0,
                                        lr_init=0.4, lr_min=0.02,
                                        mom_init=0.85, mom_final=0.999)

            M_val = fast_conv(a).max()
            if M_val < best_M_for_n:
                best_M_for_n = M_val
                best_for_n = a.copy()

        if best_for_n is not None:
            J_val = compute_J(best_for_n)
            parent_J = compute_J(resample_seq(current_best, n)) if current_best.size != n else compute_J(current_best)
            print(f"  n={n:4d}: J={J_val:.6f} (parent_resampled={parent_J:.6f}) "
                  f"({time.monotonic()-t0:.1f}s)", flush=True)
            if J_val < compute_J(current_best):
                current_best = best_for_n.copy()
            tracker.update(best_for_n)

    tracker.report("Phase 1 complete")
    print(f"Phase 1 total: {time.monotonic()-p1_start:.1f}s", flush=True)

    # === PHASE 2: Subgradient refinement on best ===
    p2_start = time.monotonic()
    p2_budget = min(150.0, (deadline - time.monotonic()) * 0.2)
    print(f"\n{'='*60}", flush=True)
    print(f"PHASE 2: Subgradient Refinement (budget={p2_budget:.0f}s)", flush=True)
    print(f"{'='*60}", flush=True)

    if tracker.best is not None:
        p2_deadline = time.monotonic() + p2_budget
        a = subgradient_descent(tracker.best, 50000, p2_deadline, lr_init=0.3)
        tracker.update(a)

        # Also try on current_best if different
        if not np.allclose(tracker.best, current_best, atol=1e-6):
            a2 = subgradient_descent(current_best, 30000, min(p2_deadline, time.monotonic() + p2_budget * 0.4),
                                     lr_init=0.25)
            tracker.update(a2)

    tracker.report("Phase 2 complete")
    print(f"Phase 2 total: {time.monotonic()-p2_start:.1f}s", flush=True)

    # === PHASE 3: LP Mirror Descent ===
    p3_start = time.monotonic()
    p3_budget = min(400.0, (deadline - time.monotonic()) * 0.35)
    print(f"\n{'='*60}", flush=True)
    print(f"PHASE 3: LP Mirror Descent (budget={p3_budget:.0f}s)", flush=True)
    print(f"{'='*60}", flush=True)

    if tracker.best is not None:
        n_best = tracker.best.size
        p3_deadline = time.monotonic() + p3_budget

        # Estimate LP time
        t0_lp = time.monotonic()
        b_test = fast_conv(tracker.best)
        M_test = float(b_test.max())
        g_test = lp_best_response(tracker.best, M_test)
        lp_time = time.monotonic() - t0_lp
        print(f"  n={n_best}, LP time={lp_time*1000:.0f}ms", flush=True)

        max_iters = max(20, int(p3_budget / max(0.01, lp_time * 1.5)))
        print(f"  Estimated max LP iterations: {max_iters}", flush=True)

        a = lp_mirror_descent(tracker.best, max_iters, p3_deadline)
        tracker.update(a)

        # Second pass if time allows
        if time.monotonic() < p3_deadline - 5:
            a2 = lp_mirror_descent(a, max(10, int((p3_deadline - time.monotonic()) / max(0.01, lp_time * 1.5))),
                                   p3_deadline)
            tracker.update(a2)

    tracker.report("Phase 3 complete")
    print(f"Phase 3 total: {time.monotonic()-p3_start:.1f}s", flush=True)

    # === PHASE 4: High-tau smooth gradient (converges to exact max) ===
    p4_start = time.monotonic()
    p4_budget = min(200.0, (deadline - time.monotonic()) * 0.3)
    print(f"\n{'='*60}", flush=True)
    print(f"PHASE 4: High-tau Smooth Gradient (budget={p4_budget:.0f}s)", flush=True)
    print(f"{'='*60}", flush=True)

    if tracker.best is not None:
        p4_deadline = time.monotonic() + p4_budget
        # Very high tau to approximate exact max
        a = nesterov_smooth_descent(tracker.best, 200000, p4_deadline,
                                    initial_tau=50.0, final_tau=500.0,
                                    lr_init=0.15, lr_min=0.005,
                                    mom_init=0.9, mom_final=0.999)
        tracker.update(a)

        # Subgradient follow-up
        if time.monotonic() < p4_deadline - 10:
            a2 = subgradient_descent(a, 20000, p4_deadline, lr_init=0.2)
            tracker.update(a2)

    tracker.report("Phase 4 complete")
    print(f"Phase 4 total: {time.monotonic()-p4_start:.1f}s", flush=True)

    # === PHASE 5: Perturbation + re-optimize (SA-like) ===
    p5_start = time.monotonic()
    p5_budget = max(30.0, deadline - time.monotonic())
    print(f"\n{'='*60}", flush=True)
    print(f"PHASE 5: Perturbation Restarts (budget={p5_budget:.0f}s)", flush=True)
    print(f"{'='*60}", flush=True)

    if tracker.best is not None and p5_budget > 30:
        n_best = tracker.best.size
        temperature = 0.03
        max_restarts = 30

        for r in range(max_restarts):
            if time.monotonic() > deadline - 15:
                break

            a = tracker.best.copy().astype(np.float64)
            noise_scale = temperature * (1.0 + 0.5 * np.sin(r * 0.7))
            a = a + rng.normal(0, noise_scale * 0.02, n_best)
            a = np.maximum(a, 0.001)
            a = a / a.sum()

            # Quick smooth gradient
            a = nesterov_smooth_descent(a, 5000, min(time.monotonic() + 10, deadline - 15),
                                        initial_tau=10, final_tau=100,
                                        lr_init=0.1, lr_min=0.02,
                                        mom_init=0.9, mom_final=0.999)
            tracker.update(a)

            # If improved, do LP polish
            if time.monotonic() < deadline - 30 and r % 3 == 0:
                est_lp = 0.1 * (n_best / 512) ** 2
                if est_lp < 5:
                    a = lp_mirror_descent(a, 20, min(time.monotonic() + 30, deadline - 15))
                    tracker.update(a)

            temperature *= 0.93

    tracker.report("Phase 5 complete")
    print(f"Phase 5 total: {time.monotonic()-p5_start:.1f}s", flush=True)

    # === PHASE 6: Length extension ===
    if tracker.best is not None and time.monotonic() < deadline - 10:
        p6_start = time.monotonic()
        print(f"\n{'='*60}", flush=True)
        print(f"PHASE 6: Length Extension", flush=True)
        print(f"{'='*60}", flush=True)

        base_n = tracker.best.size
        for mult in [2, 3]:
            if time.monotonic() > deadline - 10:
                break
            new_n = base_n * mult
            if new_n > 3072 or new_n < 128:
                continue
            a_ext = resample_seq(tracker.best, new_n)
            a_ext = np.maximum(a_ext, 1e-6)
            a_ext = a_ext / a_ext.sum()
            a_ext = nesterov_smooth_descent(a_ext, 20000, deadline - 10,
                                            initial_tau=5, final_tau=80,
                                            lr_init=0.2, lr_min=0.03,
                                            mom_init=0.9, mom_final=0.999)
            tracker.update(a_ext)

        tracker.report("Phase 6 complete")

    # === Final LP pass ===
    if tracker.best is not None and time.monotonic() < deadline - 3:
        a = lp_mirror_descent(tracker.best, 50, deadline)
        tracker.update(a)

    # Fallback
    if tracker.best is None:
        tracker.best = np.ones(512)
        tracker.best_score = compute_J(tracker.best)

    print(f"\n{'='*60}", flush=True)
    print(f"FINAL: J={tracker.best_score:.8f} n={tracker.best.size} "
          f"time={time.monotonic()-tracker.t_start:.0f}s", flush=True)
    print(f"{'='*60}", flush=True)

    return [float(x) for x in tracker.best]
