"""
Minimizes J(a) = 2*n*max(convolve(a,a))/(sum(a))^2 using:
1. Adam optimizer with adaptive L_p-norm smoothing (p annealed from 10 to 10000)
2. Multiplicative (mirror) gradient descent for positivity-preserving updates
3. Nesterov-accelerated gradient descent
4. LP-based final polish for precise max reduction
5. Multi-restart with diverse initializations and adaptive scheduling
"""

import numpy as np
from scipy.optimize import linprog, minimize
from scipy.sparse import csr_matrix
import time

_A_structure_cache = {}


def get_A_structure(n):
    if n not in _A_structure_cache:
        rows = []
        cols = []
        for k in range(2 * n - 1):
            jl = max(0, k - n + 1)
            jh = min(n - 1, k)
            for j in range(jl, jh + 1):
                rows.append(k)
                cols.append(j)
        _A_structure_cache[n] = (np.array(rows, dtype=np.int32),
                                  np.array(cols, dtype=np.int32))
    return _A_structure_cache[n]


def build_A_sparse_fast(a):
    n = len(a)
    rows_arr, cols_arr = get_A_structure(n)
    vals = a[rows_arr - cols_arr]
    return csr_matrix((vals, (rows_arr, cols_arr)), shape=(2 * n - 1, n))


def fast_autocorr(a):
    """Compute full autocorrelation via FFT."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    A = np.fft.rfft(a, size)
    corr = np.fft.irfft(A * A, size)
    return corr[:2 * n - 1]


def compute_J_exact(a):
    """Compute exact J(a) = 2*n*max(conv(a,a))/(sum(a))^2"""
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_autocorr(a)
    return float(2 * len(a) * b.max() / (s * s))


def pnorm_obj_and_grad(a, p):
    """
    Compute J_p(a) = 2*n * ||c||_p / (sum(a))^2 and its gradient.
    Uses numerically stable log-space computations.
    """
    n = len(a)
    s = float(a.sum())
    if s < 1e-10:
        return 1e10, np.zeros(n)

    c = fast_autocorr(a)
    c = np.maximum(c, 1e-300)

    # ||c||_p in log space
    log_c = np.log(c)
    max_log_c = log_c.max()
    p_shifted = np.clip(p * (log_c - max_log_c), -700, 700)
    log_sum_exp_p = p * max_log_c + np.log(np.exp(p_shifted).sum())
    log_norm_p = log_sum_exp_p / p
    norm_p = np.exp(log_norm_p)

    # w_l = c_l^(p-1) / ||c||_p^(p-1)
    log_w = (p - 1) * (log_c - log_norm_p)
    log_w = np.clip(log_w, -700, 0)
    w = np.exp(log_w)

    obj = 2.0 * n * norm_p / (s * s)

    # Gradient via FFT convolution
    w_flip = w[::-1]
    size_fft = 1
    while size_fft < len(w_flip) + n:
        size_fft <<= 1
    Wf = np.fft.rfft(w_flip, size_fft)
    Aft = np.fft.rfft(a, size_fft)
    conv = np.fft.irfft(Wf * Aft, size_fft)
    grad_unnorm = conv[n - 1:2 * n - 1]

    # dJ/da_i = 4*n * (grad_unnorm[i] - norm_p/s) / s^2
    grad = 4.0 * n * (grad_unnorm - norm_p / s) / (s * s)

    return obj, grad


def adam_step(state, grad, lr, beta1, beta2, eps, t):
    """Single Adam update step."""
    m = state[0] * beta1 + grad * (1 - beta1)
    v = state[1] * beta2 + grad * grad * (1 - beta2)
    m_hat = m / (1 - beta1 ** t)
    v_hat = v / (1 - beta2 ** t)
    step = lr * m_hat / (np.sqrt(v_hat) + eps)
    return step, (m, v)


def adam_optimize(a0, p_schedule, total_iters, deadline, tracker=None,
                  lr=0.01, beta1=0.9, beta2=0.999, eps=1e-8):
    """
    Adam optimizer with p-norm smoothing and p annealing.
    p_schedule: list of (p_value, fraction_of_total) tuples.
    """
    n = len(a0)
    a = np.maximum(a0.astype(np.float64), 1e-6)
    a = a / a.sum()

    m = np.zeros(n)
    v = np.zeros(n)
    t = 0
    iters_done = 0

    for p_val, frac in p_schedule:
        n_iters_p = int(total_iters * frac)
        for i in range(n_iters_p):
            if time.monotonic() > deadline:
                break
            if iters_done >= total_iters:
                break

            obj, grad = pnorm_obj_and_grad(a, p_val)
            if not np.isfinite(obj):
                break

            # Project gradient: zero out where a is at boundary
            grad[a < 1e-8] = max(grad[a < 1e-8], 0)

            step, (m, v) = adam_step((m, v), grad, lr, beta1, beta2, eps, t + 1)
            t += 1

            a_new = a - step
            a_new = np.maximum(a_new, 0.0)
            s_new = a_new.sum()
            if s_new < 1e-10:
                a_new = a.copy()
            else:
                a_new = a_new / s_new
            a = a_new
            iters_done += 1

            if tracker is not None and iters_done % 500 == 0:
                tracker.update(a)

    if tracker is not None:
        tracker.update(a)
    return a


def nesterov_optimize(a0, p_schedule, total_iters, deadline, tracker=None,
                       lr=0.005, momentum=0.9):
    """Nesterov-accelerated gradient descent with p-norm smoothing."""
    n = len(a0)
    a = np.maximum(a0.astype(np.float64), 1e-6)
    a = a / a.sum()
    v = np.zeros(n)
    t = 0

    for p_val, frac in p_schedule:
        n_iters_p = int(total_iters * frac)
        for i in range(n_iters_p):
            if time.monotonic() > deadline:
                break
            if t >= total_iters:
                break

            # Nesterov lookahead
            a_look = a - momentum * v
            a_look = np.maximum(a_look, 0)
            s_look = a_look.sum()
            if s_look > 1e-10:
                a_look = a_look / s_look

            obj, grad = pnorm_obj_and_grad(a_look, p_val)
            if not np.isfinite(obj):
                break

            # Adaptive learning rate based on gradient norm
            gnorm = np.linalg.norm(grad)
            if gnorm < 1e-15:
                step_lr = lr
            else:
                step_lr = lr / (1 + gnorm * 0.1)

            v = momentum * v + step_lr * grad
            a_new = a - v
            a_new = np.maximum(a_new, 0)
            s_new = a_new.sum()
            if s_new < 1e-10:
                a_new = a.copy()
            else:
                a_new = a_new / s_new
            a = a_new
            t += 1

            if tracker is not None and t % 500 == 0:
                tracker.update(a)

    if tracker is not None:
        tracker.update(a)
    return a


def multiplicative_descent(a0, p_schedule, total_iters, deadline, tracker=None,
                           base_lr=0.01):
    """
    Multiplicative (mirror) gradient descent.
    Update: a_i *= exp(-lr * grad_i), then normalize.
    Naturally maintains positivity.
    """
    n = len(a0)
    a = np.maximum(a0.astype(np.float64), 1e-6)
    a = a / a.sum()
    t = 0

    for p_val, frac in p_schedule:
        n_iters_p = int(total_iters * frac)
        for i in range(n_iters_p):
            if time.monotonic() > deadline:
                break
            if t >= total_iters:
                break

            obj, grad = pnorm_obj_and_grad(a, p_val)
            if not np.isfinite(obj):
                break

            # Scale gradient for stability
            gnorm = np.linalg.norm(grad)
            if gnorm < 1e-15:
                break
            effective_lr = base_lr / (gnorm * n ** 0.5 + 1)

            log_update = -effective_lr * grad
            log_update = np.clip(log_update, -5, 5)
            a_new = a * np.exp(log_update)
            a_new = np.maximum(a_new, 1e-10)
            a_new = a_new / a_new.sum()
            a = a_new
            t += 1

            if tracker is not None and t % 500 == 0:
                tracker.update(a)

    if tracker is not None:
        tracker.update(a)
    return a


def lp_best_response(a, M):
    """Solve LP: max sum(g) s.t. conv(a,g)_k <= M, g >= 0."""
    n = len(a)
    c = -np.ones(n)
    A = build_A_sparse_fast(a)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    try:
        r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                    options={'time_limit': 30})
        if r.success and r.x is not None:
            g = np.maximum(r.x, 0.0)
            g_sum = g.sum()
            if g_sum > 1e-10:
                return g / g_sum
    except Exception:
        pass
    return None


def lp_refine(a, max_iters, deadline, tracker=None):
    """LP-based mirror descent refinement."""
    a = np.maximum(a.astype(np.float64), 1e-8)
    a = a / a.sum()
    current_J = compute_J_exact(a)
    best_J = current_J
    best_a = a.copy()
    stall = 0

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        b = fast_autocorr(a)
        M = float(b.max())

        g = lp_best_response(a, M)
        if g is None:
            break

        # Try several step sizes
        improved = False
        for t_try in [1.0, 0.5, 0.3, 0.1, 0.05, 0.01]:
            a_try = (1.0 - t_try) * a + t_try * g
            a_try = np.maximum(a_try, 0)
            s = a_try.sum()
            if s < 1e-10:
                continue
            a_try = a_try / s
            J_try = compute_J_exact(a_try)
            if J_try < current_J - 1e-14:
                a = a_try
                current_J = J_try
                improved = True
                if J_try < best_J:
                    best_J = J_try
                    best_a = a.copy()
                    if tracker is not None:
                        tracker.update(a)
                break

        if not improved:
            stall += 1
            if stall > 20:
                break
        else:
            stall = 0

    if tracker is not None:
        tracker.update(best_a)
    return best_a


def generate_initializations(n, rng, parent=None):
    """Generate diverse initial sequences."""
    inits = []
    t_arr = np.linspace(0, 1, n, endpoint=False)

    if parent is not None and parent.size > 0:

        inits.append(np.maximum(p, 0.01))

        inits.append(np.maximum(p2, 0.01))

    inits.append(np.ones(n))
    inits.append(rng.uniform(0.3, 1.0, n))

    for freq in [2, 3, 4, 5, 7, 11]:
        inits.append(1.0 + 0.3 * np.cos(2.0 * np.pi * freq * t_arr))

    inits.append(np.exp(-3.0 * (t_arr - 0.5) ** 2) + 0.1)
    inits.append(np.abs(np.sin(np.pi * t_arr)) + 0.2)
    inits.append(1.0 + 0.25 * np.cos(4 * np.pi * t_arr) + 0.15 * np.cos(7 * np.pi * t_arr))
    inits.append((t_arr + 0.05) ** 0.7)
    inits.append(np.exp(-2.0 * t_arr) + 0.3)

    # Random smooth
    coeffs = rng.standard_normal(5)
    smooth = np.zeros(n)
    for k, c in enumerate(coeffs):
        smooth += c * np.cos(2 * np.pi * (k + 1) * t_arr)
    inits.append(1.0 + 0.2 * smooth)

    return inits


def resample_seq(src, dest_n):
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)


class ScoreTracker:
    def __init__(self):
        self.best = None
        self.best_score = float('inf')
        self.t_start = time.monotonic()

    def update(self, a):
        a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
        if a.size == 0 or a.sum() < 0.01:
            return
        score = compute_J_exact(a)
        if np.isfinite(score) and score < self.best_score:
            self.best = a.copy()
            self.best_score = score
            elapsed = time.monotonic() - self.t_start
            print(f"  * NEW BEST: J={score:.8f} n={a.size} t={elapsed:.1f}s", flush=True)

    def report(self, label=""):
        if self.best is not None:
            print(f"  {label}: J={self.best_score:.8f} n={self.best.size}", flush=True)

    n_inits = min(len(inits), 8)
    print(f"  Trying {n_inits} initializations at n={n_base}...", flush=True)

    for idx, a0 in enumerate(inits[:n_inits]):
        if time.monotonic() > deadline_p1:
            break
        a0 = np.maximum(a0, 0.01)
        a0 = a0 / a0.sum()
        t_start = time.monotonic()
        a_opt = adam_optimize(a0, p_schedule_main, total_iters_per_run,
                              deadline_p1, tracker, lr=0.005)
        if a_opt is not None:
            tracker.update(a_opt)
        print(f"  Init {idx+1}/{n_inits}: J={tracker.best_score:.8f} "
              f"({time.monotonic()-t_start:.1f}s)", flush=True)

    # Also try at different lengths
    for n_try in [640, 1024, 1280]:
        if time.monotonic() > deadline_p1:
            break
        if n_try == n_base:
            continue
        a0 = resample_seq(tracker.best if tracker.best is not None else parent, n_try)
        a0 = np.maximum(a0, 0.01)
        a0 = a0 / a0.sum()
        t_start = time.monotonic()
        a_opt = adam_optimize(a0, p_schedule_main, total_iters_per_run,
                              deadline_p1, tracker, lr=0.005)
        if a_opt is not None:
            tracker.update(a_opt)
        print(f"  n={n_try}: J={tracker.best_score:.8f} "
              f"({time.monotonic()-t_start:.1f}s)", flush=True)

    tracker.report("Phase 1")

    # === Phase 2: Nesterov + Multiplicative descent ===
    if tracker.best is not None and time.monotonic() < deadline:
        t0 = time.monotonic()
        p2_budget = 300.0
        deadline_p2 = min(deadline, time.monotonic() + p2_budget)
        print(f"\n{'='*50} Phase 2: Nesterov+Multiplicative (budget {p2_budget:.0f}s) {'='*50}", flush=True)

        n_best = tracker.best.size
        p_schedule_p2 = [
            (50, 0.2),
            (100, 0.25),
            (300, 0.25),
            (1000, 0.2),
            (5000, 0.1),
        ]
        total_iters_p2 = 6000

        # Nesterov on best
        a_nes = nesterov_optimize(tracker.best, p_schedule_p2, total_iters_p2,
                                  deadline_p2, tracker, lr=0.003, momentum=0.95)
        if a_nes is not None:
            tracker.update(a_nes)
        print(f"  Nesterov done: J={tracker.best_score:.8f}", flush=True)

        # Multiplicative descent
        if time.monotonic() < deadline_p2:
            a_mult = multiplicative_descent(tracker.best, p_schedule_p2, total_iters_p2,
                                            deadline_p2, tracker, base_lr=0.005)
            if a_mult is not None:
                tracker.update(a_mult)
            print(f"  Multiplicative done: J={tracker.best_score:.8f}", flush=True)

        # Restarts from best with perturbations
        if time.monotonic() < deadline_p2:
            for pert_scale in [0.03, 0.05, 0.1]:
                if time.monotonic() > deadline_p2:
                    break
                a_pert = tracker.best * (1 + pert_scale * rng.standard_normal(n_best))
                a_pert = np.maximum(a_pert, 0.01)
                a_pert = a_pert / a_pert.sum()
                a_opt = adam_optimize(a_pert, p_schedule_p2, total_iters_p2,
                                      deadline_p2, tracker, lr=0.005)
                if a_opt is not None:
                    tracker.update(a_opt)
            print(f"  Restarts done: J={tracker.best_score:.8f}", flush=True)

        tracker.report("Phase 2")

    # === Phase 3: LP-based refinement ===
    if tracker.best is not None and time.monotonic() < deadline:
        t0 = time.monotonic()
        p3_budget = 400.0
        deadline_p3 = min(deadline, time.monotonic() + p3_budget)
        print(f"\n{'='*50} Phase 3: LP Refinement (budget {p3_budget:.0f}s) {'='*50}", flush=True)

        n_best = tracker.best.size
        # Run LP refine
        a_lp = lp_refine(tracker.best, max_iters=200, deadline=deadline_p3, tracker=tracker)
        if a_lp is not None:
            tracker.update(a_lp)
        print(f"  LP refine 1: J={tracker.best_score:.8f} ({time.monotonic()-t0:.1f}s)", flush=True)

        # Continue LP from updated best
        if time.monotonic() < deadline_p3:
            a_lp2 = lp_refine(tracker.best, max_iters=150, deadline=deadline_p3, tracker=tracker)
            if a_lp2 is not None:
                tracker.update(a_lp2)
            print(f"  LP refine 2: J={tracker.best_score:.8f} ({time.monotonic()-t0:.1f}s)", flush=True)

        tracker.report("Phase 3")

    # === Phase 4: Final polish - combine best of all methods ===
    if tracker.best is not None and time.monotonic() < deadline:
        t0 = time.monotonic()
        remaining = deadline - time.monotonic()
        p4_budget = min(200.0, remaining)
        deadline_p4 = time.monotonic() + p4_budget
        print(f"\n{'='*50} Phase 4: Final Polish (budget {p4_budget:.0f}s) {'='*50}", flush=True)

        n_best = tracker.best.size

        # Fine-tune with very high p Adam
        p_schedule_final = [
            (1000, 0.3),
            (5000, 0.3),
            (20000, 0.4),
        ]
        a_fin = adam_optimize(tracker.best, p_schedule_final, 4000,
                              deadline_p4, tracker, lr=0.001)
        if a_fin is not None:
            tracker.update(a_fin)

        # One more LP pass
        if time.monotonic() < deadline_p4:
            a_lp3 = lp_refine(tracker.best, max_iters=50, deadline=deadline_p4, tracker=tracker)
            if a_lp3 is not None:
                tracker.update(a_lp3)

        tracker.report("Phase 4")

    # === Phase 5: Try length variations ===
    if tracker.best is not None and time.monotonic() < deadline:
        t0 = time.monotonic()
        remaining = deadline - time.monotonic()
        if remaining > 30:
            print(f"\n{'='*50} Phase 5: Length Variations ({remaining:.0f}s) {'='*50}", flush=True)
            n_best = tracker.best.size
            for n_try in [n_best - 64, n_best + 64, n_best - 128, n_best + 128]:
                if n_try < 128 or n_try > 4096 or time.monotonic() > deadline:
                    continue
                if n_try == n_best:
                    continue
                a_res = resample_seq(tracker.best, n_try)
                a_res = np.maximum(a_res, 0.01)
                a_res = a_res / a_res.sum()
                p_sched = [(50, 0.3), (200, 0.4), (2000, 0.3)]
                a_opt = adam_optimize(a_res, p_sched, 3000, deadline, tracker, lr=0.003)
                if a_opt is not None:
                    tracker.update(a_opt)
                # Quick LP polish
                if time.monotonic() < deadline:
                    a_lp = lp_refine(tracker.best, max_iters=30, deadline=deadline, tracker=tracker)
                    if a_lp is not None:
                        tracker.update(a_lp)
            tracker.report("Phase 5")

    # Fallback
    if tracker.best is None:
        tracker.best = np.ones(512)
        tracker.best_score = compute_J_exact(tracker.best)

    print(f"\n{'='*50} FINAL RESULT {'='*50}", flush=True)
    print(f"J={tracker.best_score:.8f} n={tracker.best.size} "
          f"time={time.monotonic()-tracker.t_start:.1f}s", flush=True)

    result = np.clip(tracker.best, 0.0, 1000.0)
    if result.sum() < 0.01:
        result = np.ones(128)

    return [float(x) for x in result]
