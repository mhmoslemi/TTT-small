Step function optimization using smooth (log-sum-exp) objective with L-BFGS-B,

def compute_conv(a):

    n = len(a)

    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    conv = np.fft.irfft(fa * fa, size)
    return conv[:2 * n - 1]

    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""

import numpy as np
from scipy.optimize import minimize, minimize_scalar
import time


def compute_conv(a):
    """Compute conv(a,a) using FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    conv = np.fft.irfft(fa * fa, size)
    return conv[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    conv = compute_conv(a)
    max_b = float(np.max(conv))
    sum_a = float(np.sum(a))
    if sum_a < 0.01:
        return np.inf
    return 2.0 * n * max_b / (sum_a * sum_a)


def smooth_obj_and_grad(a, T):
    """Compute smoothed objective (log-sum-exp) and its gradient."""
    n = len(a)
    conv = compute_conv(a)
    sum_a = float(np.sum(a))

    if sum_a < 1e-10:
        return 1e10, np.zeros(n)

    # Compute smooth max and softmax weights
    x_max = float(np.max(conv))
    if T < 1e-8:
        S = x_max
        w = np.zeros(len(conv))
        idx_max = np.where(conv == x_max)
        w[idx_max] = 1.0 / len(idx_max[0])
    else:
        exp_shifted = np.exp((conv - x_max) / T)
        sum_exp = float(np.sum(exp_shifted))
        S = x_max + T * np.log(sum_exp)
        w = exp_shifted / sum_exp

    # Gradient of S w.r.t. a[j] = 2 * correlate(w, a, 'valid')[j]
    # w has length 2n-1, a has length n
    # np.correlate(w, a, 'valid')[j] = sum_i w[i+j] * a[i] for j=0..n-1
    grad_S = 2.0 * np.correlate(w, a, mode='valid')

    # Full objective: f = 2n*S / sum_a^2
    obj = 2.0 * n * S / (sum_a * sum_a)
    grad = (2.0 * n * grad_S) / (sum_a * sum_a) - \
           (4.0 * n * S * a) / (sum_a * sum_a * sum_a)

    return obj, grad


def lbfgs_anneal(a, T_start=5.0, T_end=0.005, n_anneal_steps=25,
                 time_budget=60.0, verbose=False):
    """L-BFGS-B optimization with geometric temperature annealing."""
    n = len(a)
    a = a.copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        a = np.ones(n) / n
        s = 1.0
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    if n_anneal_steps <= 1:
        ratio = 1.0
    else:
        ratio = (T_end / T_start) ** (1.0 / n_anneal_steps)

    for step in range(n_anneal_steps + 1):
        if time.monotonic() - start > time_budget - 3:
            break

        T_cur = T_start * (ratio ** step)
        if verbose:
            print(f"    Anneal step {step}/{n_anneal_steps}, T={T_cur:.6f}, obj={best_obj:.8f}")

        def fun_grad(x):
            obj, grad = smooth_obj_and_grad(x, T_cur)
            return obj, grad

        try:
            res = minimize(fun_grad, a, method='L-BFGS-B',
                           jac=True, bounds=[(0.0, None)] * n,
                           options={'maxiter': 500, 'maxfun': 1000,
                                    'ftol': 1e-14, 'gtol': 1e-10,
                                    'iprint': -1})
            if res.x is not None:
                a_new = np.maximum(res.x, 0.0)
                s_new = float(np.sum(a_new))
                if s_new > 1e-10:
                    a_new = a_new / s_new
                    obj = compute_obj(a_new)
                    if obj < best_obj:
                        best_obj = obj
                        best_a = a_new.copy()
                        a = a_new
        except Exception:
            pass

    return best_a, best_obj


def coordinate_descent(a, time_budget=30.0, max_sweeps=100, verbose=False):
    """Exact 1D coordinate descent with bounded scalar minimization."""
    n = len(a)
    a = a.copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a, np.inf
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    sweep = 0
    for sweep in range(max_sweeps):
        if time.monotonic() - start > time_budget - 3:
            break

        conv = compute_conv(a)
        sum_a = float(np.sum(a))
        max_conv = float(np.max(conv))
        obj_before = compute_obj(a)

        improved = False
        for j in range(n):
            if time.monotonic() - start > time_budget - 2:
                break

            old_val = a[j]
            S = sum_a - old_val  # sum of all other coordinates

            # Compute base, d, e vectors for the 1D problem
            # conv_t[k] = base[k] + d[k]*t + e[k]*t^2
            base = conv.copy()
            d = np.zeros(2 * n - 1)
            e = np.zeros(2 * n - 1)

            # Remove a[j] contributions from conv
            # For k where 0 <= k-j < n:
            km_range = range(max(0, j), min(2 * n - 1, j + n))
            for k in km_range:
                km = k - j  # 0 <= km < n
                if k == 2 * j:
                    base[k] -= old_val * old_val
                    e[k] = 1.0
                    d[k] = 0.0
                else:
                    base[k] -= 2.0 * old_val * a[km]
                    d[k] = 2.0 * a[km]

            # 1D objective: f(t) = 2n * max_k(base[k] + d[k]*t + e[k]*t^2) / (S+t)^2
            # Use bounded minimization
            t_upper = max(5.0 * (old_val + 1.0 / n), 2.0 / n)

            def f_1d(t):
                val = base + d * t + e * t * t
                max_val = float(np.max(val))
                S_t = S + t
                if S_t < 1e-10:
                    return 1e10
                return 2.0 * n * max_val / (S_t * S_t)

            try:
                res = minimize_scalar(f_1d, bounds=(0.0, t_upper),
                                      method='bounded',
                                      options={'xatol': 1e-12, 'maxiter': 150})
                if res.success and res.fun < f_1d(old_val) - 1e-12:
                    a[j] = res.x
                    improved = True
            except Exception:
                pass

        # Update best after full sweep
        obj_after = compute_obj(a)
        if obj_after < best_obj:
            best_obj = obj_after
            best_a = a.copy()
            if verbose and sweep % 10 == 0:
                print(f"    CD sweep {sweep}: obj={best_obj:.8f}")
        elif not improved:
            break

    return best_a, best_obj


def multiplicative_subgradient(a, tau, step, n_iters=200, time_budget=10.0):
    """Multiplicative subgradient descent (fast heuristic)."""
    n = len(a)
    a = a.copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    for it in range(n_iters):
        if time.monotonic() - start > time_budget:
            break

        conv = compute_conv(a)
        max_conv = float(np.max(conv))
        shifted = tau * (conv - max_conv)
        shifted = np.clip(shifted, -500, 500)
        weights = np.exp(shifted)
        w_sum = float(np.sum(weights))
        if w_sum < 1e-300:
            break
        weights /= w_sum

        grad = 2.0 * np.correlate(weights, a, mode='valid')
        a = a * np.exp(-step * grad)
        a = np.maximum(a, 1e-15)
        s = float(np.sum(a))
        if s > 1e-10:
            a = a / s

        obj = compute_obj(a)
        if obj < best_obj:
            best_obj = obj
            best_a = a.copy()

    return best_a


def random_init(n, seed):
    """Generate diverse random initializations with frequency structure."""
    rng = np.random.default_rng(seed)
    k = np.arange(n)
    choice = rng.integers(0, 8)

    if choice == 0:
        # Single dominant frequency
        f = rng.uniform(1.0, 3.0)
        phase = rng.uniform(0, 2 * np.pi)
        a = 1.0 + 0.7 * np.cos(2 * np.pi * f * k / n + phase)
    elif choice == 1:
        # Two frequencies
        f1 = rng.uniform(0.5, 2.0)
        f2 = rng.uniform(2.0, 5.0)
        a = 1.0 + 0.4 * np.cos(2 * np.pi * f1 * k / n) + \
            0.3 * np.cos(2 * np.pi * f2 * k / n + rng.uniform(0, 2 * np.pi))
    elif choice == 2:
        # Three frequencies (harmonic-like)
        f0 = rng.uniform(0.5, 1.5)
        a = 1.0
        for m in range(3):
            f = f0 * (m + 1)
            amp = 0.5 / (m + 1)
            a += amp * np.cos(2 * np.pi * f * k / n + rng.uniform(0, 2 * np.pi))
    elif choice == 3:
        # Gaussian bump
        center = n * rng.uniform(0.3, 0.7)
        width = n * rng.uniform(0.1, 0.4)
        a = 1.0 + 0.6 * np.exp(-0.5 * ((k - center) / width) ** 2)
    elif choice == 4:
        # Smooth random (low-pass filtered noise)
        a_raw = rng.random(max(5, n // 15))
        a = np.interp(k, np.arange(len(a_raw)), a_raw)
        a = 0.5 + 0.5 * a
    elif choice == 5:
        # Chirp
        f0 = rng.uniform(0.5, 1.0)
        f1 = rng.uniform(3.0, 7.0)
        t = k / n
        f = f0 + (f1 - f0) * t
        a = 1.0 + 0.5 * np.cos(2 * np.pi * np.cumsum(f))
    elif choice == 6:
        # Random smooth with multiple scales
        a = np.ones(n)
        for scale in range(4):
            f = rng.uniform(0.3 * (2 ** scale), 1.0 * (2 ** scale))
            amp = 0.3 / (scale + 1)
            a += amp * np.cos(2 * np.pi * f * k / n + rng.uniform(0, 2 * np.pi))
    else:
        # Windowed random
        a = rng.uniform(0.3, 1.0, n)
        win = 1.0 - 0.5 * np.cos(2 * np.pi * k / (n - 1))
        a = a * win
        a = 0.5 + 0.5 * a

    a = np.maximum(a, 0.0)
    s = float(np.sum(a))
    if s < 1e-10:
        a = np.ones(n)
        s = n
    return a / s


def resample_sequence(a, new_n):
    """Resample a sequence to a new length."""
    n = len(a)
    if n == new_n:
        return a.copy()
    old_k = np.arange(n, dtype=np.float64)
    new_k = np.linspace(0, n - 1, new_n)
    a_new = np.interp(new_k, old_k, a)
    a_new = np.maximum(a_new, 0)
    s = float(np.sum(a_new))
    if s > 1e-10:
        a_new = a_new / s
    return a_new


def optimize_sequence(a, time_budget=60.0, verbose=False):
    """Full optimization pipeline on a single sequence."""
    n = len(a)
    a = a.copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        a = np.ones(n) / n
        s = 1.0
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    if verbose:
        print(f"  Initial: obj={best_obj:.8f}")

    # Phase A: Quick subgradient (warm start)
    remaining = time_budget - (time.monotonic() - start)
    if remaining > 8:
        sg_time = min(5.0, remaining * 0.1)
        a_sg = multiplicative_subgradient(a, tau=20.0, step=0.05,
                                          n_iters=int(sg_time * 50),
                                          time_budget=sg_time)
        obj_sg = compute_obj(a_sg)
        if obj_sg < best_obj:
            best_obj = obj_sg
            best_a = a_sg.copy()
            a = a_sg.copy()

    # Phase B: L-BFGS with annealing (main workhorse)
    remaining = time_budget - (time.monotonic() - start)
    if remaining > 15:
        lbfgs_time = min(remaining * 0.4, 40.0)
        a_lbfgs, obj_lbfgs = lbfgs_anneal(a, T_start=3.0, T_end=0.01,
                                           n_anneal_steps=15,
                                           time_budget=lbfgs_time,
                                           verbose=verbose)
        if obj_lbfgs < best_obj:
            best_obj = obj_lbfgs
            best_a = a_lbfgs.copy()
            a = a_lbfgs.copy()
            if verbose:
                print(f"  After L-BFGS: obj={best_obj:.8f}")

    # Phase C: Coordinate descent (precise refinement)
    remaining = time_budget - (time.monotonic() - start)
    if remaining > 10:
        cd_time = min(remaining * 0.3, 25.0)
        a_cd, obj_cd = coordinate_descent(a, time_budget=cd_time,
                                          max_sweeps=int(cd_time * 3),
                                          verbose=verbose)
        if obj_cd < best_obj:
            best_obj = obj_cd
            best_a = a_cd.copy()
            a = a_cd.copy()
            if verbose:
                print(f"  After CD: obj={best_obj:.8f}")

    # Phase D: Second L-BFGS round with lower temperature
    remaining = time_budget - (time.monotonic() - start)
    if remaining > 10:
        lbfgs_time = min(remaining * 0.3, 20.0)
        a_lbfgs2, obj_lbfgs2 = lbfgs_anneal(a, T_start=0.5, T_end=0.001,
                                             n_anneal_steps=10,
                                             time_budget=lbfgs_time,
                                             verbose=False)
        if obj_lbfgs2 < best_obj:
            best_obj = obj_lbfgs2
            best_a = a_lbfgs2.copy()
            a = a_lbfgs2.copy()
            if verbose:
                print(f"  After L-BFGS2: obj={best_obj:.8f}")

    # Phase E: Final coordinate descent
    remaining = time_budget - (time.monotonic() - start)
    if remaining > 5:
        cd_time = min(remaining * 0.5, remaining - 2)
        a_cd2, obj_cd2 = coordinate_descent(a, time_budget=cd_time,
                                            max_sweeps=int(cd_time * 2))
        if obj_cd2 < best_obj:
            best_obj = obj_cd2
            best_a = a_cd2.copy()
            if verbose:
                print(f"  After CD2: obj={best_obj:.8f}")

    return best_a, best_obj


def try_parent():
    """Try to load parent sequence."""
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64)
        parent = np.clip(parent, 0, 1000.0)
        if parent.size > 0 and np.isfinite(parent).all():
            obj = compute_obj(parent)
            print(f"Parent (n={parent.size}): obj={obj:.8f}")
            return parent.copy(), obj
    except Exception:
        pass
    return None, np.inf


def propose_candidate(seed=42, budget_s=1500):
    """
    Main search function. Uses L-BFGS with log-sum-exp smoothing,
    coordinate descent, temperature annealing, and systematic n exploration.
    """
    start_time = time.monotonic()
    deadline = start_time + float(budget_s) - 20.0

    best_seq = None
    best_score = np.inf

    # Load parent
    parent, parent_obj = try_parent()
    n_parent = 330
    if parent is not None:
        n_parent = len(parent)
        if parent_obj < best_score:
            best_score = parent_obj
            best_seq = parent.copy()

    print(f"Initial best: {best_score:.8f} (n={n_parent})")
    print(f"Time budget: {budget_s}s, deadline in {deadline - start_time:.0f}s")

    # Phase 1: Heavy optimization on parent (most important)
    print("\n=== Phase 1: Optimize parent ===")
    if best_seq is not None:
        remaining = deadline - time.monotonic()
        p1_budget = min(remaining * 0.3, 350.0)
        print(f"  Budget: {p1_budget:.0f}s")
        a_opt, obj = optimize_sequence(best_seq, time_budget=p1_budget, verbose=True)
        if obj < best_score:
            best_score = obj
            best_seq = a_opt.copy()
            print(f"  ** Update: {best_score:.8f} **")

    # Phase 2: Try nearby n values
    print("\n=== Phase 2: Nearby n values ===")
    n_candidates = []
    for delta in [-20, -15, -10, -5, 0, 5, 10, 15, 20, 25, 30, -30, 35, 40]:
        nv = n_parent + delta
        if 200 <= nv <= 500:
            n_candidates.append(nv)
    # Also try some fixed values
    for nv in [280, 300, 320, 340, 360, 380]:
        if nv not in n_candidates:
            n_candidates.append(nv)
    n_candidates = sorted(set(n_candidates))

    remaining = deadline - time.monotonic()
    if remaining > 60:
        # Budget allocation: give more time to n values near the parent
        weights = []
        for nv in n_candidates:
            dist = abs(nv - n_parent)
            w = 1.0 / (1.0 + dist * 0.1)
            weights.append(w)
        total_w = sum(weights)
        weights = [w / total_w for w in weights]

        for i, nv in enumerate(n_candidates):
            if time.monotonic() > deadline - 30:
                break
            remaining = deadline - time.monotonic()
            if remaining < 20:
                break
            alloc = remaining * weights[i] * 1.5  # 1.5x to prioritize
            alloc = min(alloc, 80.0)
            alloc = max(alloc, 10.0)

            if nv == n_parent and best_seq is not None:
                a_init = best_seq.copy()
            elif parent is not None:
                a_init = resample_sequence(parent, nv)
            else:
                a_init = random_init(nv, seed + nv * 104729)

            print(f"\n  n={nv} (budget={alloc:.0f}s, {remaining:.0f}s left)")
            a_opt, obj = optimize_sequence(a_init, time_budget=alloc,
                                           verbose=(nv == n_parent))
            if obj < best_score:
                best_score = obj
                best_seq = a_opt.copy()
                print(f"  ** UPDATE: n={nv}, obj={best_score:.8f} **")

    # Phase 3: Intensive refinement on best
    print(f"\n=== Phase 3: Intensive refinement (best={best_score:.8f}) ===")
    if best_seq is not None:
        remaining = deadline - time.monotonic()
        if remaining > 30:
            p3_budget = min(remaining * 0.4, 200.0)
            print(f"  Budget: {p3_budget:.0f}s")

            # Multiple rounds of L-BFGS + CD with different T schedules
            a = best_seq.copy()
            s = float(np.sum(a))
            a = a / s

            for round_idx in range(4):
                if time.monotonic() > deadline - 20:
                    break
                rem = deadline - time.monotonic()
                if rem < 15:
                    break

                # L-BFGS with progressively lower T
                T_s = [1.0, 0.3, 0.1, 0.05][round_idx % 4]
                T_e = [0.01, 0.003, 0.001, 0.0005][round_idx % 4]
                t_budget = min(rem * 0.4, 40.0)
                a_new, obj_new = lbfgs_anneal(a, T_start=T_s, T_end=T_e,
                                               n_anneal_steps=12,
                                               time_budget=t_budget)
                if obj_new < best_score:
                    best_score = obj_new
                    best_seq = a_new.copy()
                    a = a_new.copy()
                    print(f"  Round {round_idx} L-BFGS: {best_score:.8f}")

                # Coordinate descent
                rem = deadline - time.monotonic()
                if rem > 10:
                    cd_budget = min(rem * 0.4, 30.0)
                    a_new, obj_new = coordinate_descent(a, time_budget=cd_budget,
                                                       max_sweeps=int(cd_budget * 3))
                    if obj_new < best_score:
                        best_score = obj_new
                        best_seq = a_new.copy()
                        a = a_new.copy()
                        print(f"  Round {round_idx} CD: {best_score:.8f}")

    # Phase 4: Random restarts exploring different structures
    print(f"\n=== Phase 4: Random restarts (best={best_score:.8f}) ===")
    if best_seq is not None:
        remaining = deadline - time.monotonic()
        n_best = len(best_seq)

        if remaining > 40:
            restart_budget = remaining * 0.5
            n_restarts = max(3, int(restart_budget / 30))
            per_restart = restart_budget / n_restarts

            for r in range(n_restarts):
                if time.monotonic() > deadline - 25:
                    break
                rem = deadline - time.monotonic()
                if rem < 20:
                    break

                alloc = min(per_restart, rem * 0.4)
                alloc = max(alloc, 10.0)

                # Different strategies for each restart
                if r % 3 == 0:
                    # Perturb best
                    rng = np.random.default_rng(seed + r * 999983)
                    a_init = best_seq * np.exp(rng.normal(0, 0.05, n_best))
                    a_init = np.maximum(a_init, 0)
                    s = float(np.sum(a_init))
                    a_init = a_init / s
                elif r % 3 == 1:
                    # New random init at same n
                    a_init = random_init(n_best, seed + r * 4711)
                else:
                    # Slightly different n
                    n_try = n_best + rng.integers(-10, 11) if r % 2 == 0 else n_best
                    n_try = max(200, min(500, n_try))
                    if parent is not None:
                        a_init = resample_sequence(parent, n_try)
                    else:
                        a_init = random_init(n_try, seed + r * 7919)

                a_opt, obj = optimize_sequence(a_init, time_budget=alloc)
                if obj < best_score:
                    best_score = obj
                    best_seq = a_opt.copy()
                    print(f"  Restart {r}: {best_score:.8f}")

    # Phase 5: Final polish
    print(f"\n=== Phase 5: Final polish ===")
    if best_seq is not None:
        remaining = deadline - time.monotonic()
        if remaining > 10:
            # Very fine coordinate descent
            cd_budget = min(remaining * 0.5, 60.0)
            a_final, obj_final = coordinate_descent(best_seq, time_budget=cd_budget,
                                                   max_sweeps=int(cd_budget * 5))
            if obj_final < best_score:
                best_score = obj_final
                best_seq = a_final.copy()
                print(f"  Final CD: {best_score:.8f}")

            # One more L-BFGS with very low T
            remaining = deadline - time.monotonic()
            if remaining > 8:
                a_final, obj_final = lbfgs_anneal(best_seq, T_start=0.1, T_end=0.0001,
                                                  n_anneal_steps=8,
                                                  time_budget=min(remaining * 0.4, 30.0))
                if obj_final < best_score:
                    best_score = obj_final
                    best_seq = a_final.copy()
                    print(f"  Final L-BFGS: {best_score:.8f}")

    # Final validation
    if best_seq is None:
        best_seq = np.ones(50)
        best_score = compute_obj(best_seq)

    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    val = evaluate_sequence(result)
    if val < best_score:
        best_score = val

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(best_seq)}, score={best_score:.8f}, time={elapsed:.1f}s ===")

    return result
