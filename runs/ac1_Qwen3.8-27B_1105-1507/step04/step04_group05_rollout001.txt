"""
Minimizes J(a) = 2*n*max(convolve(a,a))/(sum(a))^2 using:
1. Fast coordinate descent with O(n) per-coordinate updates and golden section
2. LP mirror descent with adaptive tightening and line search
3. Gradient descent with smooth max approximation
4. Multi-length exploration and restarts
5. Pairwise coordinate descent
"""

import numpy as np
from scipy.optimize import linprog
from scipy.sparse import lil_matrix, csr_matrix
import time


def fast_conv(a):
    """Full self-convolution via FFT."""
    n = len(a)
    s = 1
    while s < 2 * n - 1:
        s <<= 1
    fa = np.fft.rfft(a, s)
    return np.fft.irfft(fa * fa, s)[:2 * n - 1]


def compute_J(a):
    """Compute J(a) = 2*n*max(conv(a,a))/(sum(a))^2."""
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))


def build_conv_matrix(a):
    """Build the convolution constraint matrix A where (A@g)[k] = (a*g)[k]."""
    n = len(a)
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        jl = max(0, k - n + 1)
        jh = min(n - 1, k)
        for j in range(jl, jh + 1):
            rows.append(k)
            cols.append(j)
            vals.append(a[k - j])
    A = lil_matrix((2 * n - 1, n))
    for r, c, v in zip(rows, cols, vals):
        A[r, c] = v
    return A.tocsr()


def lp_solve(a, M):
    """Solve LP: max sum(g) s.t. (a*g)[k] <= M for all k, g >= 0."""
    n = len(a)
    c = -np.ones(n)
    A = build_conv_matrix(a)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    result = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
    if result.success and result.x is not None:
        g = np.maximum(result.x, 0.0)
        if g.sum() > 1e-10:
            return g
    return None

    """
    n = len(a)
    old = a[i]
    new_S = S + x - old
    if new_S < 1e-15:
        return 1e20
    
    # Find max of affected convolution entries
    max_affected = 0.0
    delta = x - old
    for k in range(i, i + n):
        ki = k - i
        if 0 <= ki < n:
            if k == 2 * i:
                bk = b[k] + x * x - old * old
            else:
                bk = b[k] + 2.0 * delta * a[ki]
            if bk > max_affected:
                max_affected = bk
    
    max_b = max(max_affected, b_unaffected_max)
    if max_b < 0:
        max_b = 0.0
    
    return 2.0 * n * max_b / (new_S * new_S)


def golden_section(func, lo, hi, n_iter=50):
    """Golden section minimization."""
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    c = hi - gr * (hi - lo)
    d = lo + gr * (hi - lo)
    fc = func(c)
    fd = func(d)
    for _ in range(n_iter):
        if fc < fd:
            hi = d
            d = c
            fd = fc
            c = hi - gr * (hi - lo)
            fc = func(c)
        else:
            lo = c
            c = d
            fc = fd
            d = lo + gr * (hi - lo)
            fd = func(d)
    return (lo + hi) / 2.0


def cd_pass(a, b, S, max_iters_per_coord=50):
    """One pass of coordinate descent. Returns updated (a, b, S, J)."""
    n = len(a)
    a = a.copy()
    b = b.copy()
    
    # Precompute unaffected max for each coordinate (approximate: use global max initially)
    # Actually, the "unaffected" set depends on which coordinate we're updating
    # For coordinate i, unaffected are k < i or k >= i+n
    # Prefix max and suffix max:
    # prefix_max[i] = max(b[0], ..., b[i-1])
    # suffix_max[i] = max(b[i], ..., b[2n-2])
    prefix_max = np.zeros(2 * n - 1 + 1)
    prefix_max[0] = -np.inf
    for k in range(2 * n - 1):
        prefix_max[k + 1] = max(prefix_max[k], b[k])
    
    suffix_max = np.zeros(2 * n - 1 + 1)
    suffix_max[2 * n - 1] = -np.inf
    for k in range(2 * n - 2, -1, -1):
        suffix_max[k] = max(suffix_max[k + 1], b[k])
    
    # For coordinate i, unaffected max = max(prefix_max[i], suffix_max[i+n])
    # (entries k < i and k >= i+n are unaffected)
    
    ordering = np.arange(n)
    np.random.shuffle(ordering)
    
    for i in ordering:
        old = a[i]
        
        # Compute unaffected max
        unaff_lo = prefix_max[max(0, i)]  # max of b[0..i-1]
        unaff_hi = suffix_max[min(2 * n - 1, i + n)]  # max of b[i+n..2n-2]
        b_unaff = max(unaff_lo if np.isfinite(unaff_lo) else 0.0,
                      unaff_hi if np.isfinite(unaff_hi) else 0.0)
        
        # Define objective for golden section
        def obj(x):
            return coord_obj(x, a, i, b, S, b_unaff)
        
        # Determine search range
        hi = max(3.0 * old, S / n * 3.0, 1e-6)
        lo = 0.0
        
        x_opt = golden_section(obj, lo, hi, n_iter=max_iters_per_coord)
        x_opt = max(0.0, x_opt)
        
        if abs(x_opt - old) > 1e-12:
            # Update convolution
            delta = x_opt - old
            for k in range(i, min(i + n, 2 * n - 1)):
                ki = k - i
                if 0 <= ki < n:
                    if k == 2 * i:
                        b[k] += x_opt * x_opt - old * old
                    else:
                        b[k] += 2.0 * delta * a[ki]
            
            S += delta
            a[i] = x_opt
    
    # Recompute prefix/suffix and get final J
    b_max = b.max()
    J = 2.0 * n * b_max / (S * S)
    return a, b, S, J


def cd_optimize(a, n_passes, deadline, tracker):
    """Run multiple passes of coordinate descent."""
    a = np.maximum(a.astype(np.float64), 0.0)
    S = a.sum()
    n = len(a)
    b = fast_conv(a)
    current_J = 2.0 * n * b.max() / (S * S)
    tracker.update(a)
    
    for p in range(n_passes):
        if time.monotonic() > deadline:
            break
        t0 = time.monotonic()
        a, b, S, new_J = cd_pass(a, b, S)
        if new_J < current_J - 1e-14:
            current_J = new_J
            tracker.update(a)
            if p % 10 == 0:
                print(f"  CD pass {p}: J={current_J:.8f} ({time.monotonic()-t0:.1f}s)", flush=True)
        else:
            if p > 10:
                break
    
    tracker.update(a)
    return a


def lp_mirror_descent(a, n_iters, deadline, tracker, tighten_base=0.003):
    """LP mirror descent with adaptive tightening."""
    a = np.maximum(a.astype(np.float64), 1e-6)
    a /= a.sum()
    n = len(a)
    current_J = compute_J(a)
    tracker.update(a)
    
    tighten = tighten_base
    stall = 0
    
    for it in range(n_iters):
        if time.monotonic() > deadline:
            break
        
        b = fast_conv(a)
        M = float(b.max())
        M_target = M * (1.0 - tighten)
        
        g = lp_solve(a, M_target)
        if g is None:
            tighten *= 0.5
            if tighten < 1e-7:
                break
            continue
        
        g /= g.sum()
        
        # Line search over step sizes
        best_t = 0.0
        best_J = current_J
        for t in [0.3, 0.5, 0.7, 0.1, 0.2, 0.4, 0.6, 0.8]:
            a_try = (1.0 - t) * a + t * g
            a_try = np.maximum(a_try, 0.0)
            s_try = a_try.sum()
            if s_try < 1e-10:
                continue
            a_try /= s_try
            J_try = compute_J(a_try)
            if J_try < best_J:
                best_J = J_try
                best_t = t
        
        if best_t > 0:
            a = (1.0 - best_t) * a + best_t * g
            a = np.maximum(a, 0.0)
            a /= a.sum()
            current_J = best_J
            stall = 0
        else:
            stall += 1
            tighten *= 0.8
            if stall > 30:
                break
        
        if it % 20 == 0:
            tracker.update(a)
            print(f"  LP iter {it}: J={current_J:.8f} t={best_t:.2f} tighten={tighten:.5f}", flush=True)
    
    tracker.update(a)
    return a


def gradient_descent_smooth(a, n_iters, deadline, tracker, alpha_init=0.05):
    """Gradient descent with smooth max approximation."""
    a = np.maximum(a.astype(np.float64), 1e-6)
    a /= a.sum()
    n = len(a)
    current_J = compute_J(a)
    tracker.update(a)
    alpha = alpha_init
    
    for it in range(n_iters):
        if time.monotonic() > deadline:
            break
        
        b = fast_conv(a)
        S = a.sum()
        b_max = b.max()
        
        # Smooth max: use top-k entries
        k_val = min(5, 2 * n - 1)
        top_k_idx = np.argsort(b)[-k_val:][::-1]
        top_k_vals = b[top_k_idx]
        
        # Soft weights (exponential)
        t_smooth = 10.0 / b_max
        weights = np.exp(t_smooth * (top_k_vals - top_k_vals[0]))
        weights /= weights.sum()
        
        # Gradient of soft-max w.r.t. a
        grad = np.zeros(n)
        for w, k in zip(weights, top_k_idx):
            for j in range(max(0, k - n + 1), min(n, k + 1)):
                ki = k - j
                if 0 <= ki < n:
                    if ki == j:
                        grad[j] += w * 2.0 * a[j]
                    else:
                        grad[j] += w * 2.0 * a[ki]
        
        # Subtract Lagrange multiplier for sum constraint
        # J ~ max_b / S^2, dJ/da[i] ~ (grad_max[i] * S^2 - max_b * 2 * S) / S^4
        # Simplified: grad_J[i] = grad_max[i] / S^2 - 2 * b_max / S^3
        grad -= 2.0 * b_max / (S * S)
        
        a_new = a + alpha * (-grad)  # Wait, we want to minimize, so subtract gradient
        # Actually J = 2n * b_max / S^2
        # dJ/da[i] = 2n * (db_max/da[i] * S - b_max * 2) / S^2
        # = 2n/S^2 * (S * grad_max[i] - 2*b_max)
        # Let me redo:
        # J = 2n * b_max / S^2
        # dJ/da[i] = 2n * (grad_bmax[i] * S^2 - b_max * 2*S) / S^4
        # = 2n/S^2 * (grad_bmax[i] - 2*b_max/S)
        
        grad_J = 2.0 * n / (S * S) * (grad - 2.0 * b_max / S)
        
        a_new = a - alpha * grad_J
        a_new = np.maximum(a_new, 0.0)
        s_new = a_new.sum()
        if s_new < 1e-10:
            alpha *= 0.5
            continue
        a_new /= s_new
        
        J_new = compute_J(a_new)
        if J_new < current_J:
            a = a_new
            current_J = J_new
            alpha = min(alpha * 1.05, 0.5)
            tracker.update(a)
        else:
            alpha *= 0.7
        
        if it % 50 == 0:
            print(f"  GD iter {it}: J={current_J:.8f} alpha={alpha:.4f}", flush=True)
    
    tracker.update(a)
    return a


def resample_seq(src, dest_n):
    """Resample 1D sequence to new length via linear interpolation."""
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)


class ScoreTracker:
    def __init__(self):
        self.best = None
        self.best_score = float('inf')
        self.t_start = time.monotonic()
    
    def update(self, a):
        a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
        if a.size == 0 or a.sum() < 0.01:
            return
        score = float(evaluate_sequence(a.tolist()))
        if np.isfinite(score) and score < self.best_score:
            self.best = a.copy()
            self.best_score = score
            elapsed = time.monotonic() - self.t_start
            print(f"  *** NEW BEST: J={score:.8f} n={a.size} t={elapsed:.0f}s ***", flush=True)
    
    def report(self, label=""):
        if self.best is not None:
            print(f"  {label} best: J={self.best_score:.8f} n={self.best.size} t={time.monotonic()-self.t_start:.0f}s", flush=True)


def pairwise_cd_pass(a, b, S, n_pairs=100):
    """Pairwise coordinate descent: update pairs of coordinates simultaneously."""
    n = len(a)
    a = a.copy()
    b = b.copy()
    
    for _ in range(n_pairs):
        i, j = np.random.choice(n, 2, replace=False)
        if i == j:
            continue
        
        old_i, old_j = a[i], a[j]
        S_old = S
        
        def obj_params(xi, xj):
            new_S = S_old - old_i - old_j + xi + xj
            if new_S < 1e-15:
                return 1e20
            max_b = b.max()  # crude: use global max as upper bound for unaffected
            # Actually need to be more careful, but for simplicity use full recompute
            # This is expensive, so limit usage
            return 2.0 * n * b.max() / (new_S * new_S)
        
        # Simpler: just try a few random perturbations
        best_j_val = 2.0 * n * b.max() / (S * S)
        best_xi, best_xj = old_i, old_j
        
        for trial in range(10):
            scale = 1.0 + np.random.uniform(-0.3, 0.3)
            xi = max(0.0, old_i * scale)
            xj = max(0.0, old_j * (2.0 - scale))  # try to preserve sum
            a[i], a[j] = xi, xj
            S_trial = S_old - old_i - old_j + xi + xj
            if S_trial < 1e-15:
                a[i], a[j] = old_i, old_j
                continue
            b_trial = fast_conv(a)
            J_trial = 2.0 * n * b_trial.max() / (S_trial * S_trial)
            if J_trial < best_j_val:
                best_j_val = J_trial
                best_xi, best_xj = xi, xj
                b = b_trial.copy()
                S = S_trial
            else:
                a[i], a[j] = old_i, old_j
        
        a[i], a[j] = best_xi, best_xj
        S = S_old - old_i - old_j + best_xi + best_xj
        # Need to update b
        b = fast_conv(a)
    
    J_final = 2.0 * n * b.max() / (S * S)
    return a, b, S, J_final


def generate_structural_inits(n, rng):
    """Generate diverse structural initializations."""
    inits = []
    t = np.linspace(0, 1, n, endpoint=False)
    
    # Uniform
    inits.append(np.ones(n))
    
    # Chirp-like
    inits.append(0.5 + 0.5 * np.cos(2 * np.pi * n * t * t / 2))
    inits.append(0.5 + 0.5 * np.cos(2 * np.pi * (n * t - n * t * t / 2)))
    
    # Frequency modulated
    for f in [2, 3, 5]:
        inits.append(np.maximum(0.5 + 0.5 * np.cos(2 * np.pi * f * t), 0.01))
    
    # Ramp with oscillation
    inits.append(np.maximum(1.0 + 0.5 * np.sin(2 * np.pi * 3 * t) * t, 0.01))
    
    # Two-scale
    inits.append(0.5 + 0.3 * np.cos(2 * np.pi * 2 * t) + 0.2 * np.cos(2 * np.pi * 7 * t))
    inits = [np.maximum(x, 0.01) for x in inits]
    
    return inits
