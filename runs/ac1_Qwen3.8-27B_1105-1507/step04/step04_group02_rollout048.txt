"""
Optimized step-function coefficient search minimizing 2*n*max(conv(a,a))/(sum(a))^2.
Uses: soft-max smoothed gradient descent with adaptive beta, spectral peak reduction,
LP-based refinement (HiGHS), frequency-domain shaping, and multi-start exploration.
Key innovation: smooth soft-max objective enables faster convergence than hard-max
subgradient, combined with spectral domain operations for global peak reduction.
"""

import numpy as np
from scipy.optimize import linprog
from scipy.sparse import coo_matrix
import time


def compute_conv(a):
    """Compute linear convolution a*a via FFT."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    conv = np.fft.irfft(fa * fa, size)
    return conv[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    S = float(np.sum(a))
    if S < 1e-10:
        return np.inf
    conv = compute_conv(a)
    M = float(np.max(conv))
    return 2.0 * n * M / (S * S)


def compute_softmax_gradient(a, beta):
    """Compute gradient of soft-max approximation to max(conv(a,a)).
    
    Returns (grad, conv, weights) where grad is df/da for f = (1/beta)*log(sum exp(beta*conv_k)).
    """
    n = len(a)
    conv = compute_conv(a)
    conv_max = conv.max()
    # For numerical stability, shift
    scaled = beta * (conv - conv_max)
    # Clip for overflow protection
    scaled = np.clip(scaled, -500, 500)
    weights = np.exp(scaled)
    weights_sum = weights.sum()
    if weights_sum < 1e-300:
        weights = np.zeros_like(conv)
        weights[np.argmax(conv)] = 1.0
        weights_sum = 1.0
    weights /= weights_sum

    # grad[m] = 2 * sum_j a[j] * weights[m+j] for j in [0, n-1], m+j in [0, 2n-2]
    # This is cross-correlation, computed via FFT
    N = 1
    while N < 2 * n:
        N <<= 1
    fa = np.fft.rfft(a, N)
    fw = np.fft.rfft(weights, N)
    # circular cross-correlation: ifft(conj(fa) * fw)[m] = sum_n a[n] * weights[(n+m) mod N]
    cross = np.fft.irfft(np.conj(fa) * fw, N)[:n]
    grad = 2.0 * cross
    return grad, conv, weights


def project_simplex(a, target_sum=1.0):
    """Project onto {x >= 0, sum(x) = target_sum}."""
    a = np.maximum(a, 0.0)
    S = a.sum()
    if S < 1e-15:
        return np.full_like(a, target_sum / len(a))
    return a * (target_sum / S)


def softmax_gradient_descent(a, max_iters, time_budget, beta_start=50.0, beta_end=500.0,
                              lr=0.02, momentum=0.9, topk_smooth=20):
    """Soft-max gradient descent with adaptive beta and momentum."""
    n = len(a)
    a = project_simplex(a.astype(np.float64).copy())
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    v = np.zeros(n)
    stall = 0
    max_stall = 50

    for it in range(max_iters):
        if time.monotonic() - start > time_budget:
            break

        # Adaptive beta: increase over iterations
        frac = min(1.0, it / max(1, max_iters * 0.8))
        beta = beta_start + (beta_end - beta_start) * frac

        grad, conv, weights = compute_softmax_gradient(a, beta)

        # Center gradient to preserve sum
        grad_centered = grad - grad.mean()

        # Momentum update
        v = momentum * v + (1.0 - momentum) * grad_centered

        # Adaptive learning rate
        lr_adapt = lr / (1.0 + 0.001 * it)

        a_new = a - lr_adapt * v
        a_new = project_simplex(a_new)

        obj = compute_obj(a_new)
        if obj < best_obj - 1e-12:
            best_obj = obj
            best_a = a_new.copy()
            a = a_new
            stall = 0
        else:
            stall += 1
            if stall >= max_stall:
                # Escape: perturb and reduce step
                rng = np.random.default_rng(hash((n, it)) % (2**31))
                a = a * np.exp(rng.normal(0, 0.1, n))
                a = project_simplex(a)
                v = np.zeros(n)
                stall = 0
                lr *= 0.5
                if lr < 1e-6:
                    break

    return best_a, best_obj


def build_toeplitz_sparse(a):
    """Build sparse Toeplitz matrix for conv(a, g)."""
    n = len(a)
    rows = []
    cols = []
    vals = []
    for j in range(n):
        start = j
        end = min(j + n, 2 * n - 1)
        length = end - start
        rows.extend(range(start, end))
        cols.extend([j] * length)
        vals.extend(list(a[:length]))
    A = coo_matrix((vals, (rows, cols)), shape=(2 * n - 1, n)).tocsr()
    return A


def solve_lp_maxsum(a, M, time_limit=5):
    """Solve LP: max sum(g) s.t. conv(a,g) <= M, g >= 0."""
    n = len(a)
    A = build_toeplitz_sparse(a)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': time_limit,
                               'dual_feasibility_tolerance': 1e-9})
        if res.status == 0:
            return np.maximum(res.x, 0)
    except Exception:
        pass
    return None


def lp_refine(a, max_iters=30, time_budget=30.0):
    """Alternating LP refinement with line search."""
    n = len(a)
    a = project_simplex(a.astype(np.float64).copy())
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    stall = 0

    for it in range(max_iters):
        if time.monotonic() - start > time_budget:
            break

        conv = compute_conv(a)
        M = float(np.max(conv))

        g = solve_lp_maxsum(a, M, time_limit=4)
        if g is None:
            stall += 1
            if stall > 5:
                break
            # Try slightly relaxed M
            g = solve_lp_maxsum(a, M * 1.05, time_limit=4)
            if g is None:
                break

        S_g = np.sum(g)
        if S_g < 1e-10:
            break
        g = g / S_g

        # Line search
        best_t = 0.0
        best_obj_ls = best_obj
        for t in [0.05, 0.1, 0.2, 0.3, 0.5, 0.7, 1.0]:
            a_try = project_simplex((1.0 - t) * a + t * g)
            obj_try = compute_obj(a_try)
            if obj_try < best_obj_ls:
                best_obj_ls = obj_try
                best_t = t

        if best_t > 0.0:
            a = project_simplex((1.0 - best_t) * a + best_t * g)
            best_obj = best_obj_ls
            best_a = a.copy()
            stall = 0
        else:
            stall += 1
            if stall > 10:
                # Perturb
                rng = np.random.default_rng(it * 6151 + 7)
                a = a * np.exp(rng.normal(0, 0.08, n))
                a = project_simplex(a)
                stall = 0

    return best_a, best_obj


def spectral_peak_reduction(a, n_iters=50, time_budget=10.0):
    """Reduce peaks in autocorrelation via frequency-domain phase manipulation.
    
    Key idea: modify the spectrum to flatten the time-domain autocorrelation.
    We work with the magnitude of the DFT and try to make it more uniform.
    """
    n = len(a)
    a = project_simplex(a.astype(np.float64).copy())
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    # Get initial spectrum
    fa = np.fft.rfft(a, 2 * n)
    mag = np.abs(fa)
    phase = np.angle(fa)

    for it in range(n_iters):
        if time.monotonic() - start > time_budget:
            break

        # Compute current autocorrelation
        conv = compute_conv(a)
        M = conv.max()

        # Identify peak lags
        active = np.where(conv > 0.9 * M)[0]

        # In frequency domain: |F|^2 has IFFT = autocorrelation
        # We want to reduce peaks -> make |F|^2 more uniform
        # Strategy: apply water-filling to |F|^2 subject to sum constraint
        power = mag ** 2
        total_power = power.sum()

        # Soft cap: reduce large frequency components
        cap = total_power / len(power) * 3.0  # Allow up to 3x average
        power_reduced = np.minimum(power, cap)
        # Renormalize to preserve total power (approximately)
        if power_reduced.sum() > 0:
            power_reduced = power_reduced * (total_power / power_reduced.sum())

        # Update magnitude
        mag_new = np.sqrt(power_reduced)

        # Reconstruct
        fa_new = mag_new * np.exp(1j * phase)
        a_new = np.fft.irfft(fa_new, 2 * n)[:n]
        a_new = project_simplex(a_new)

        obj_new = compute_obj(a_new)
        if obj_new < best_obj:
            best_obj = obj_new
            best_a = a_new.copy()
            mag = np.abs(np.fft.rfft(a_new, 2 * n))
            phase = np.angle(np.fft.rfft(a_new, 2 * n))

    return best_a, best_obj


def cosine_init(n, n_modes=12, seed=0):
    """Generate a smooth cosine-series initialization."""
    rng = np.random.default_rng(seed)
    k = np.arange(n)
    a = np.ones(n) * 0.8
    for m in range(1, n_modes + 1):
        c = rng.uniform(-0.3, 0.3) / m
        phase = rng.uniform(0, 2 * np.pi)
        a += c * np.cos(2 * np.pi * m * k / n + phase)
    a = np.maximum(a, 0)
    return project_simplex(a)


def gaussian_init(n, sigma_frac=0.3, seed=0):
    """Gaussian bump with noise."""
    rng = np.random.default_rng(seed)
    k = np.arange(n)
    center = n / 2.0
    sigma = sigma_frac * n
    a = np.exp(-0.5 * ((k - center) / sigma) ** 2)
    a += 0.05 * rng.random(n)
    return project_simplex(a)


def periodic_init(n, period=7, seed=0):
    """Periodic envelope - may help with flat autocorrelation."""
    rng = np.random.default_rng(seed)
    k = np.arange(n)
    # Create a periodic pattern modulated by envelope
    base = 1.0 + 0.3 * np.cos(2 * np.pi * k / period)
    envelope = 0.5 + 0.5 * np.cos(np.pi * (k - n/2) / (n/2))
    a = base * envelope + 0.05 * rng.random(n)
    return project_simplex(a)


def multi_envelope_init(n, n_humps=3, seed=0):
    """Multiple Gaussian humps."""
    rng = np.random.default_rng(seed)
    k = np.arange(n)
    a = np.zeros(n)

        sigma = rng.uniform(0.1 * n, 0.3 * n)

            break
        remaining = phase1_deadline - time.monotonic()
        if remaining < 10:
            break

        # Multiple initializations for this n
        inits = [
            cosine_init(n, n_modes=10, seed=seed + n),
            gaussian_init(n, sigma_frac=0.3, seed=seed + n + 1),
            gaussian_init(n, sigma_frac=0.5, seed=seed + n + 2),
            periodic_init(n, period=7, seed=seed + n + 3),
            multi_envelope_init(n, n_humps=4, seed=seed + n + 4),
        ]

        # If we have a parent and n is close, try interpolation
        if parent is not None and abs(n - n_parent) <= 100:
            if n != n_parent:
                x_old = np.linspace(0, 1, n_parent)
                x_new = np.linspace(0, 1, n)
                from scipy.interpolate import interp1d
                f_interp = interp1d(x_old, parent, kind='cubic', fill_value='extrapolate')
                a_interp = np.maximum(f_interp(x_new), 0)
                a_interp = project_simplex(a_interp)
                inits.append(a_interp)
            else:
                inits.append(parent.copy())

        for idx_name, a_init in enumerate(inits):
            if time.monotonic() > phase1_deadline:
                break
            obj_init = compute_obj(a_init)

            # Quick gradient descent
            alloc = min(remaining * 0.3, 8.0)
            a_quick, obj_quick = softmax_gradient_descent(
                a_init, max_iters=1000, time_budget=alloc,
                beta_start=30.0, beta_end=150.0, lr=0.02)

            best_local = a_quick if obj_quick < obj_init else a_init
            best_local_obj = min(obj_quick, obj_init)
            candidates.append((best_local, best_local_obj, n))

            if best_local_obj < best_score:
                best_score = best_local_obj
                best_seq = best_local.copy()
                print(f"  n={n}, init{idx_name}: {best_local_obj:.6f} ** NEW BEST **")

    print(f"Phase 1 done. Best: {best_score:.6f}")

    # === Phase 2: Deep refinement of top candidates (55% time) ===
    print("\n=== Phase 2: Deep refinement ===")
    phase2_time = min(825.0, (deadline - start_time) * 0.55)
    phase2_deadline = start_time + phase2_time + phase1_time  # Adjust
    phase2_deadline = min(deadline - 300, start_time + phase1_time + phase2_time)

    candidates.sort(key=lambda x: x[1])
    n_refine = min(4, len(candidates))

    for idx in range(n_refine):
        if time.monotonic() > phase2_deadline:
            break
        remaining = phase2_deadline - time.monotonic()
        if remaining < 30:
            break

        a_cand, obj_cand, n_cand = candidates[idx]
        if obj_cand > best_score + 0.02:
            break

        alloc = min(remaining * 0.4, 120.0)
        print(f"  Refining cand {idx} (n={n_cand}, obj={obj_cand:.6f}), alloc={alloc:.0f}s")

        a_ref, obj_ref = refine_full(a_cand, time_budget=alloc)
        if obj_ref < best_score:
            best_score = obj_ref
            best_seq = a_ref.copy()
            print(f"    ** NEW BEST: n={n_cand}, obj={best_score:.6f} **")

    # === Phase 3: Intensive refinement of global best (20% time) ===
    print("\n=== Phase 3: Intensive refinement ===")
    phase3_deadline = min(deadline - 100, start_time + (budget_s - 20) * 0.8)
    a_cur = best_seq.copy()
    n_best = len(a_cur)

    for round_num in range(20):
        if time.monotonic() > phase3_deadline:
            break
        remaining = phase3_deadline - time.monotonic()
        if remaining < 20:
            break

        alloc = min(remaining * 0.2, 40.0)
        a_new, obj_new = refine_full(a_cur, time_budget=alloc)

        if obj_new < best_score:
            best_score = obj_new
            best_seq = a_new.copy()
            a_cur = a_new.copy()
            print(f"  R{round_num}: {best_score:.6f} ** NEW BEST **")
        else:
            print(f"  R{round_num}: best stays {best_score:.6f}")
            # Try perturbation
            remaining = phase3_deadline - time.monotonic()
            if remaining > 10:
                rng = np.random.default_rng(round_num * 4441 + 77)
                for scale in [0.08, 0.15, 0.25]:
                    if time.monotonic() > phase3_deadline - 5:
                        break
                    a_pert = a_cur * np.exp(rng.normal(0, scale, n_best))
                    a_pert = project_simplex(a_pert)
                    a_try, obj_try = softmax_gradient_descent(
                        a_pert, max_iters=2000, time_budget=min(10.0, remaining * 0.2),
                        beta_start=50.0, beta_end=400.0, lr=0.01)
                    if obj_try < best_score:
                        best_score = obj_try
                        best_seq = a_try.copy()
                        a_cur = a_try.copy()
                        print(f"  R{round_num} pert({scale}): {best_score:.6f} ** NEW BEST **")
                        break

    # === Phase 4: Try different n values (5% time) ===
    print("\n=== Phase 4: n-value exploration ===")
    phase4_deadline = deadline - 50
    if time.monotonic() < phase4_deadline:
        n_base = len(best_seq)
        n_tries = [n_base - 50, n_base - 25, n_base + 25, n_base + 50, n_base + 100]
        n_tries = [n for n in n_tries if 100 <= n <= 2000 and n != n_base]

        from scipy.interpolate import interp1d
        for n_try in n_tries:
            if time.monotonic() > phase4_deadline:
                break
            remaining = phase4_deadline - time.monotonic()
            if remaining < 10:
                break

            x_old = np.linspace(0, 1, n_base)
            x_new = np.linspace(0, 1, n_try)
            f_interp = interp1d(x_old, best_seq, kind='cubic', fill_value='extrapolate')
            a_interp = np.maximum(f_interp(x_new), 0)
            a_interp = project_simplex(a_interp)

            alloc = min(remaining * 0.4, 20.0)
            a_new, obj_new = refine_full(a_interp, time_budget=alloc)
            if obj_new < best_score:
                best_score = obj_new
                best_seq = a_new.copy()
                print(f"  n={n_try}: {best_score:.6f} ** NEW BEST **")
            else:
                print(f"  n={n_try}: {obj_new:.6f}")

    # === Phase 5: Final ultra-fine polish (remaining time) ===
    print("\n=== Phase 5: Final polish ===")
    remaining = deadline - time.monotonic()
    if remaining > 10:
        a_cur = best_seq.copy()
        n_final = len(a_cur)

        # High-beta gradient descent
        alloc = min(remaining * 0.4, 60.0)
        a_sg, obj_sg = softmax_gradient_descent(
            a_cur, max_iters=5000, time_budget=alloc,
            beta_start=200.0, beta_end=1000.0, lr=0.005, momentum=0.9)
        if obj_sg < best_score:
            best_score = obj_sg
            best_seq = a_sg.copy()
            a_cur = a_sg.copy()
            print(f"  High-beta SG: {best_score:.6f} ** NEW BEST **")

        # Final LP
        remaining = deadline - time.monotonic()
        if remaining > 10 and n_final <= 600:
            a_lp, obj_lp = lp_refine(a_cur, max_iters=30, time_budget=min(remaining * 0.4, 40.0))
            if obj_lp < best_score:
                best_score = obj_lp
                best_seq = a_lp.copy()
                print(f"  Final LP: {best_score:.6f} ** NEW BEST **")

        # Tiny perturbations
        remaining = deadline - time.monotonic()
        if remaining > 5:
            rng = np.random.default_rng(123456789)
            for _ in range(5):
                if time.monotonic() > deadline - 3:
                    break
                a_p = a_cur * np.exp(rng.normal(0, 0.02, n_final))
                a_p = project_simplex(a_p)
                a_p, obj_p = softmax_gradient_descent(
                    a_p, max_iters=3000, time_budget=min(remaining * 0.15, 10.0),
                    beta_start=300.0, beta_end=800.0, lr=0.003, momentum=0.92)
                if obj_p < best_score:
                    best_score = obj_p
                    best_seq = a_p.copy()
                    print(f"  Micro-perturb: {best_score:.6f} ** NEW BEST **")

    # Final validation
    if best_seq is None:
        best_seq = np.ones(500) / 500

    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    val = evaluate_sequence(result)
    if val < np.inf:
        if val < best_score:
            best_score = val
    else:
        # Fallback
        result = [1.0] * 300
        best_score = evaluate_sequence(result)

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_score:.6f}, time={elapsed:.1f}s ===")

    return result
