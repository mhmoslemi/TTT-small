import numpy as np
from scipy.optimize import linprog
import time

def compute_conv(a):
    n = len(a)
    size = 2*n - 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)

def compute_obj(a):
    n = len(a)
    S = np.sum(a)
    if S < 1e-10:
        return np.inf
    conv = compute_conv(a)
    return 2.0 * n * np.max(conv) / (S * S)

def linearized_lp(a, M, eps_frac=0.05, time_limit=5):
    """
    Solve the linearized LP:
    min t
    s.t. c_k + 2*(a*delta)[k] <= t, k in active set
         sum(delta) = 0
         delta >= -a
    """
    n = len(a)
    conv = compute_conv(a)
    M = np.max(conv)
    
    # Active set: positions where conv is within eps_frac*M of max
    threshold = M * (1 - eps_frac)
    active = np.where(conv >= threshold)[0]
    
    if len(active) == 0:
        active = np.argmax(conv)
        active = np.array([active])
    
    # Build constraint matrix: A_act[i, j] = 2 * a[active[i] - j] if 0 <= active[i]-j < n else 0
    m = len(active)
    A = np.zeros((m, n))
    for idx, k in enumerate(active):
        for j in range(n):
            r = k - j
            if 0 <= r < n:
                A[idx, j] = 2.0 * a[r]
    
    # LP: min [t, 0, ..., 0] (t is the first variable, delta is the rest)
    # Variables: [t, delta_0, ..., delta_{n-1}]
    c_obj = np.zeros(n + 1)
    c_obj[0] = 1.0  # minimize t
    
    # Constraints: A @ delta - t <= -c_k  =>  -t + A @ delta <= -c_k
    # Or: t - A @ delta >= c_k  =>  -t + A @ delta <= -c_k
    # In standard form for linprog (A_ub @ x <= b_ub):
    # A_ub rows: [-1, A[0,:]], [-1, A[1,:]], ..., [-1, A[m-1,:]]
    # b_ub: [-c[active[0]], -c[active[1]], ...]
    
    A_ub = np.zeros((m, n + 1))
    A_ub[:, 0] = -1.0
    A_ub[:, 1:] = A
    b_ub = -conv[active]
    
    # Equality constraint: sum(delta) = 0
    A_eq = np.zeros((1, n + 1))
    A_eq[0, 1:] = 1.0
    b_eq = np.zeros(1)
    
    # Bounds: t is free, delta[i] >= -a[i], delta[i] <= some upper bound
    bounds = [(None, None)] + [(-a[i], 1.0) for i in range(n)]
    
    res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                  bounds=bounds, method='highs',
                  options={'time_limit': time_limit, 'presolve': True})
    
    if res.status == 0:
        return res.x[1:].copy(), active, res.x[0]  # delta, active set, t value
    return None, active, None

def lp_direction(a, M, time_limit=5):
    """
    Original LP: max sum(g) s.t. (a*g)[k] <= M, g >= 0
    """
    n = len(a)
    conv_len = 2 * n - 1
    
    # A[i, j] = a[i-j] if 0 <= i-j < n, else 0 (for the convolution a*g)
    # (a*g)[k] = sum_j a[k-j] * g[j]
    A = np.zeros((conv_len, n))
    for k in range(conv_len):
        for j in range(n):
            r = k - j
            if 0 <= r < n:
                A[k, j] = a[r]
    
    c = -np.ones(n)
    b_ub = np.full(conv_len, M)
    bounds = [(0, None)] * n
    
    res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                  options={'time_limit': time_limit, 'presolve': True})
    
    if res.status == 0:
        return np.maximum(res.x, 0)
    return None

def refine(a, time_budget, verbose=False):
    """Main refinement loop combining linearized LP and original LP."""
    n = len(a)
    a = a.astype(np.float64).copy()
    S = np.sum(a)
    a = a / S
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    
    iteration = 0
    while time.monotonic() - start < time_budget - 2:
        iteration += 1
        remaining = time_budget - (time.monotonic() - start)
        if remaining < 5:
            break
        
        a_cur = a.copy()
        conv = compute_conv(a_cur)
        M = np.max(conv)
        
        improved = False
        
        # Method 1: Linearized LP
        if remaining > 10:
            delta, active, t_new = linearized_lp(a_cur, M, eps_frac=0.05, time_limit=min(remaining*0.3, 10))
            if delta is not None:
                # Line search
                for step in [1.0, 0.5, 0.25, 0.1, 0.05]:
                    a_try = a_cur + step * delta
                    a_try = np.maximum(a_try, 0)
                    S_try = np.sum(a_try)
                    if S_try < 1e-10:
                        continue
                    a_try = a_try / S_try
                    obj_try = compute_obj(a_try)
                    if obj_try < best_obj * (1 - 1e-9):
                        best_obj = obj_try
                        best_a = a_try.copy()
                        a = a_try.copy()
                        improved = True
                        break
        
        # Method 2: Original LP (g direction)
        if remaining > 10 and not improved:
            g = lp_direction(a_cur, M, time_limit=min(remaining*0.3, 10))
            if g is not None:
                S_g = np.sum(g)
                if S_g > 1e-10:
                    g = g / S_g
                    for step in [0.3, 0.2, 0.1, 0.05]:
                        a_try = (1-step)*a_cur + step*g
                        a_try = np.maximum(a_try, 0)
                        S_try = np.sum(a_try)
                        if S_try < 1e-10:
                            continue
                        a_try = a_try / S_try
                        obj_try = compute_obj(a_try)
                        if obj_try < best_obj * (1 - 1e-9):
                            best_obj = obj_try
                            best_a = a_try.copy()
                            a = a_try.copy()
                            improved = True
                            break
        
        # Method 3: Subgradient fallback
        if remaining > 5 and not improved:
            # Find active peaks
            topk = min(10, len(conv))
            active_k = np.argsort(conv)[-topk:]
            grad = np.zeros(n)
            for k in active_k:
                for i in range(n):
                    j = k - i
                    if 0 <= j < n:
                        grad[i] += 2.0 * a_cur[j]
            grad /= len(active_k)
            grad_centered = grad - np.mean(grad)
            
            for step in [0.1, 0.05, 0.02, 0.01]:
                a_try = a_cur - step * grad_centered
                a_try = np.maximum(a_try, 0)
                S_try = np.sum(a_try)
                if S_try < 1e-10:
                    continue
                a_try = a_try / S_try
                obj_try = compute_obj(a_try)
                if obj_try < best_obj * (1 - 1e-9):
                    best_obj = obj_try
                    best_a = a_try.copy()
                    a = a_try.copy()
                    improved = True
                    break
        
        if not improved and verbose:
            if iteration % 10 == 0:
                print(f"    Iter {iteration}: obj={best_obj:.6f}, no improvement")
    
    return best_a, best_obj

def compute_conv(a):
    n = len(a)
    size = 2 * n - 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)

def compute_obj(a):
    n = len(a)
    S = float(np.sum(a))
    if S < 1e-10:
        return np.inf
    conv = compute_conv(a)
    M = float(np.max(conv))
    return 2.0 * n * M / (S * S)

def solve_linearized_lp(a, active_k, time_limit=5.0):
    """
    Solve: min t
           s.t. conv_k + 2*(a*delta)_k <= t, k in active_k
                sum(delta) = 0
                delta >= -a
    Returns delta (or None if infeasible).
    """
    n = len(a)
    conv = compute_conv(a)
    
    m = len(active_k)
    # Build constraint matrix
    # Row for k: [-1, 2*a[k-0], 2*a[k-1], ..., 2*a[k-(n-1)]]
    # where a[j] = 0 if j < 0 or j >= n
    A_ub = np.zeros((m, n + 1))
    A_ub[:, 0] = -1.0
    for idx, k in enumerate(active_k):
        for j in range(min(k+1, n)):
            r = k - j
            if 0 <= r < n:
                A_ub[idx, j+1] = 2.0 * a[r]
    
    b_ub = -conv[active_k]
    
    # Equality: sum(delta) = 0
    A_eq = np.zeros((1, n+1))
    A_eq[0, 1:] = 1.0
    b_eq = np.zeros(1)
    
    bounds = [(None, None)] + [(-a[i], a[i]*2) for i in range(n)]
    
    try:
        res = linprog(np.zeros(n+1) + np.concatenate([[1]] + [0]*n),
                     A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                     bounds=bounds, method='highs',
                     options={'time_limit': time_limit, 'presolve': True})
        if res.status == 0:
            return res.x[1:]
    except:
        pass
    return None

def solve_g_lp(a, M, time_limit=5.0):
    """
    Solve: max sum(g)
           s.t. (a*g)[k] <= M for all k
                g >= 0
    """
    n = len(a)
    conv_len = 2*n - 1
    A = np.zeros((conv_len, n))
    for k in range(conv_len):
        lo = max(0, k - (n-1))
        hi = min(k, n-1)
        for j in range(lo, hi+1):
            A[k, j] = a[k-j]
    
    c = -np.ones(n)
    b_ub = np.full(conv_len, M)
    bounds = [(0, None)] * n
    
    try:
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds,
                     method='highs', options={'time_limit': time_limit, 'presolve': True})
        if res.status == 0:
            return np.maximum(res.x, 0)
    except:
        pass
    return None

def subgradient_step(a, topk=10):
    """One subgradient descent step (centered)."""
    n = len(a)
    conv = compute_conv(a)
    active = np.argsort(conv)[-topk:]
    grad = np.zeros(n)
    for k in active:
        for i in range(n):
            j = k - i
            if 0 <= j < n:
                grad[i] += 2.0 * a[j]
    grad /= len(active)
    grad -= np.mean(grad)  # center
    return grad
