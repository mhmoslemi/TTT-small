Now let me put together the full algorithm.

One more consideration: the normalize step. Since J is scale-invariant, I'll always normalize a to have sum=1 during optimization, and the final answer can be returned as-is (any positive scale works).

Let me also think about the LP step more carefully. The LP is:
maximize sum(g)
subject to: (a*g)[k] <= M for all k = 0, ..., 2n-2
            g[j] >= 0 for all j

where M is the current max of (a*a).

The key insight from the "state of the art" paragraph is to use a tightening factor: instead of M, use M*(1-tighten). This forces the LP to find a solution that not only satisfies the current constraint but also has some margin, which leads to better progress.

After finding g, we rescale it to sum=1 and do a line search:
a_new = (1-t)*a + t*g, find best t.

Let me now write the complete code:

```python
"""
Minimizes J(a) = 2*n*max(convolve(a,a))/(sum(a))^2 using:
1. FISTA on smooth LSE approximation with adaptive temperature
2. LP best-response with adaptive tightening and line search
3. Multi-length exploration with diverse spectral initializations
4. Progressive refinement: coarse-to-fine tau schedule
5. Symmetry-aware optimization (optional)
"""

import numpy as np
from scipy.optimize import linprog
from scipy.sparse import csr_matrix
import time

_spar_cache = {}

def get_sparsity(n):
    if n not in _spar_cache:
        rows = []
        cols = []
        for k in range(2 * n - 1):
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            rows.extend([k] * (je - js + 1))
            cols.extend(range(js, je + 1))
        _spar_cache[n] = (np.array(rows), np.array(cols), len(rows))
    return _spar_cache[n]

def fast_conv(a):
    n = len(a)
    s = 1
    while s < 2 * n - 1:
        s <<= 1
    fa = np.fft.rfft(a, s)
    return np.fft.irfft(fa * fa, s)[:2 * n - 1]

def compute_J(a):
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))

def lse_grad(a, p):
    """Gradient of LSE: grad[j] = sum_m a[m]*p[j+m]"""
    n = len(a)
    a_rev = a[::-1]
    s = 1
    while s < 3 * n:
        s <<= 1
    result = np.fft.irfft(np.fft.rfft(a_rev, s) * np.fft.rfft(p, s), s)
    return result[np.arange(n) + n - 1]

def fista_lse(a, max_iters, deadline, tracker, tau_start, tau_end):
    """FISTA on LSE approximation of the max-convolution objective."""
    a = a.copy().astype(np.float64)
    a = np.maximum(a, 1e-10)
    a /= a.sum()
    n = len(a)
    current_J = compute_J(a)
    tracker.update(a)
    
    # FISTA state
    x = a.copy()
    y = a.copy()
    t = 1.0
    lr = 1.0 / np.sqrt(n)  # initial step size
    best_x = a.copy()
    best_J = current_J
    
    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        
        # Adaptive tau
        progress = min(1.0, it / max(1, max_iters))
        tau = tau_start + (tau_end - tau_start) * progress
        
        # Compute LSE gradient at y
        b = fast_conv(y)
        b_max = b.max()
        # Numerical stability: shift b so max is 0
        shifted = tau * (b - b_max)
        # Clip to avoid overflow
        shifted = np.clip(shifted, -500, 500)
        exp_b = np.exp(shifted)
        p = exp_b / exp_b.sum()
        
        grad = lse_grad(y, p)
        
        # FISTA update
        t_old = t
        x_new = y - lr * grad
        x_new = np.maximum(x_new, 1e-10)
        x_new /= x_new.sum()
        
        t = (1.0 + np.sqrt(1.0 + 4.0 * t_old * t_old)) / 2.0
        y = x_new + ((t_old - 1.0) / t) * (x_new - x)
        y = np.maximum(y, 1e-10)
        y /= y.sum()
        x = x_new
        
        # Evaluate and update best
        if it % 10 == 0 or it == max_iters - 1:
            J_trial = compute_J(x)
            if J_trial < best_J:
                best_J = J_trial
                best_x = x.copy()
                tracker.update(x)
        
        # Adaptive step size: if progress is stalled, reduce lr
        if it % 50 == 0 and it > 0:
            J_check = compute_J(x)
            if J_check > best_J * 1.001:
                lr *= 0.5
                y = x.copy()
                t = 1.0
    
    tracker.update(best_x)
    return best_x

def lp_best_response(a, M, time_limit=5):
    """Solve LP: max sum(g) s.t. (a*g)[k] <= M, g >= 0"""
    n = len(a)
    rows, cols, nnz = get_sparsity(n)
    
    # Build the matrix values
    vals = np.empty(nnz)
    for i in range(nnz):
        k = rows[i]
        j = cols[i]
        vals[i] = a[k - j]
    
    A = csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    
    r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                options={'presolve': True, 'time_limit': time_limit})
    if r.success and r.x is not None:
        g = np.maximum(r.x, 0.0)
        if g.sum() > 1e-10:
            return g
    return None

def lp_refine(a, max_iters, deadline, tracker, tighten_schedule=None):
    """LP-based refinement with adaptive tightening."""
    a = a.copy().astype(np.float64)
    a = np.maximum(a, 1e-10)
    a /= a.sum()
    n = len(a)
    current_J = compute_J(a)
    tracker.update(a)
    
    if tighten_schedule is None:
        tighten_schedule = [0.005, 0.01, 0.015, 0.02, 0.025, 0.03, 0.04, 0.05]
    
    it = 0
    for tighten in tighten_schedule:
        while it < max_iters:
            if time.monotonic() > deadline:
                break
            if it >= max_iters:
                break
            
            b = fast_conv(a)
            M = float(b.max())
            M_tight = M * (1.0 - tighten)
            
            g = lp_best_response(a, M_tight, time_limit=3)
            if g is None:
                g = lp_best_response(a, M, time_limit=3)
            if g is None:
                break
            g /= g.sum()
            
            # Line search
            best_t = 0
            best_J = current_J
            base_t = max(0.1, 0.5 * (1.0 - it / max(1, max_iters)))
            for t in [base_t, base_t * 0.5, base_t * 1.5, 0.2, 0.4, 0.6, 0.8]:
                if t > 0.95:
                    continue
                a_trial = (1.0 - t) * a + t * g
                a_trial = np.maximum(a_trial, 1e-10)
                a_trial /= a_trial.sum()
                J_trial = compute_J(a_trial)
                if J_trial < best_J - 1e-15:
                    best_J = J_trial
                    best_t = t
            
            if best_t == 0:
                break
            
            a = (1.0 - best_t) * a + best_t * g
            a = np.maximum(a, 1e-10)
            a /= a.sum()
            current_J = best_J
            it += 1
        
        if time.monotonic() > deadline:
            break
    
    tracker.update(a)
    return a

def flat_spectrum_init(n, rng, n_freqs=20):
    """Generate sequence with approximately flat power spectrum."""
    half = n // 2 + 1
    # Random phases, flat amplitude
    angles = rng.uniform(0, 2 * np.pi, half)
    A = np.zeros(n, dtype=complex)
    A[:half] = np.exp(1j * angles[:half])
    for k in range(1, n - n // 2):
        A[k] = np.conj(A[n - k])
    a = np.fft.ifft(A).real
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-10:
        return np.ones(n) / n
    return a / s

def windowed_cosine_init(n, rng):
    """Initialize with windowed cosine pattern."""
    t = np.arange(n) / n
    freq = rng.uniform(1, 20)
    phase = rng.uniform(0, 2 * np.pi)
    a = 1.0 + 0.3 * np.cos(2 * np.pi * freq * t + phase)
    a *= 0.5 * (1 + np.cos(2 * np.pi * t))  # taper
    a = np.maximum(a, 0.01)
    return a / a.sum()

def multi_freq_init(n, rng):
    """Initialize with multiple cosine frequencies."""
    t = np.arange(n) / n
    a = np.ones(n)
    n_freqs = rng.integers(3, 8)
    for _ in range(n_freqs):
        f = rng.uniform(0.5, n / 4)
        amp = rng.uniform(0.05, 0.3)
        phase = rng.uniform(0, 2 * np.pi)
        a += amp * np.cos(2 * np.pi * f * t + phase)
    a = np.maximum(a, 0.01)
    return a / a.sum()

def generate_inits(n, rng, parent):
    """Generate diverse initializations."""
    inits = []
    t = np.arange(n) / n
    
    if parent is not None:
        # Resample parent
        src = parent.astype(np.float64)
        if src.size != n:
            old_t = np.linspace(0, 1, src.size, endpoint=False)
            new_t = np.linspace(0, 1, n, endpoint=False)
            src = np.interp(new_t, old_t, src)
        inits.append(np.maximum(src, 0.001))
        inits.append(np.maximum(src[::-1], 0.001))  # reversed
    
    inits.append(np.ones(n))
    inits.append(rng.uniform(0.1, 1.0, n))
    inits.append(flat_spectrum_init(n, rng))
    inits.append(flat_spectrum_init(n, rng))
    inits.append(windowed_cosine_init(n, rng))
    inits.append(windowed_cosine_init(n, rng))
    inits.append(multi_freq_init(n, rng))
    inits.append(multi_freq_init(n, rng))
    
    # Parametric forms
    inits.append(1.0 + 0.2 * np.cos(2 * np.pi * t))
    inits.append(1.0 + 0.15 * np.cos(4 * np.pi * t) + 0.1 * np.cos(6 * np.pi * t))
    inits.append(0.5 + 0.5 * np.cos(np.pi * t)**2)
    inits.append(np.exp(-4*(t-0.5)**2) + 0.1)
    
    for x in inits:
        x = np.maximum(x, 0.001).astype(np.float64)
        x /= x.sum()
    return inits

class ScoreTracker:
    def __init__(self):
        self.best = None
        self.best_score = float('inf')
        self.t_start = time.monotonic()
    
    def update(self, a):
        a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
        if a.size == 0 or a.sum() < 0.01:
            return
        score = float(evaluate_sequence(a.tolist()))
        if np.isfinite(score) and score < self.best_score:

def resample_seq(src, dest_n):
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size, endpoint=False)
    t_new = np.linspace(0, 1, dest_n, endpoint=False)
    return np.maximum(np.interp(t_new, t_old, src), 0.001)

def optimize_single(a0, n, deadline, tracker, budget_s):
    """Run full optimization pipeline on a single initialization."""
    a = a0.copy().astype(np.float64)
    a = np.maximum(a, 1e-8)
    a /= a.sum()
    
    remaining = deadline - time.monotonic()
    if remaining < 10:
        return a
    
    # Stage 1: LP refinement (40% of budget)
    lp_deadline = time.monotonic() + 0.4 * budget_s
    a = lp_refine(a, max(10, int(30 * budget_s / 30)), min(lp_deadline, deadline - 5), tracker)
    
    # Stage 2: FISTA on LSE (40% of budget)
    fista_deadline = min(time.monotonic() + 0.4 * budget_s, deadline - 5)
    n_iters = max(100, int(5000 * budget_s / 30))
    a = fista_lse(a, n_iters, fista_deadline, tracker,
                  tau_start=n/50, tau_end=n/5)
    
    # Stage 3: LP again with tighter constraints (15% of budget)
    lp2_deadline = min(time.monotonic() + 0.15 * budget_s, deadline - 3)
    a = lp_refine(a, max(5, int(20 * budget_s / 30)), lp2_deadline, tracker,
                  tighten_schedule=[0.005, 0.01, 0.015, 0.02])
    
    # Stage 4: Final FISTA (5% of budget)
    fista2_deadline = min(time.monotonic() + 0.05 * budget_s, deadline - 1)
    a = fista_lse(a, max(50, int(2000 * budget_s / 30)), fista2_deadline, tracker,
                  tau_start=n/5, tau_end=n)
    
    return a
```

```python
"""

import numpy as np
from scipy.optimize import linprog
from scipy.sparse import csr_matrix
import time

_spar_cache = {}

def get_sparsity(n):
    if n not in _spar_cache:
        rows = []
        cols = []
        for k in range(2 * n - 1):
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            rows.extend([k] * (je - js + 1))
            cols.extend(range(js, je + 1))
        _spar_cache[n] = (np.array(rows), np.array(cols), len(rows))
    return _spar_cache[n]

def fast_conv(a):
    n = len(a)
    s = 1
    while s < 2 * n - 1:
        s <<= 1
    fa = np.fft.rfft(a, s)
    return np.fft.irfft(fa * fa, s)[:2 * n - 1]

def compute_J(a):
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))

def lse_grad(a, p):
    """Compute gradient of LSE objective w.r.t. a.
    grad[j] = sum_{m=0}^{n-1} a[m] * p[j+m]
    using FFT-based cross-correlation.
    """
    n = len(a)
    a_rev = a[::-1]
    s = 1
    while s < 3 * n:
        s <<= 1
    result = np.fft.irfft(np.fft.rfft(a_rev, s) * np.fft.rfft(p, s), s)
    return result[np.arange(n) + n - 1]

def fista_lse(a_init, max_iters, deadline, tracker, tau_start, tau_end):
    """FISTA on LSE approximation with adaptive tau and backtracking step size."""
    a = a_init.copy().astype(np.float64)
    a = np.maximum(a, 1e-10)
    a /= a.sum()
    n = len(a)
    current_J = compute_J(a)
    tracker.update(a)

    x = a.copy()
    y = a.copy()
    x_old = a.copy()
    t = 1.0
    lr = 2.0 / np.sqrt(n)
    best_x = a.copy()
    best_J = current_J
    stagnant = 0

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        progress = min(1.0, it / max(1, max_iters))
        tau = tau_start + (tau_end - tau_start) * progress

        b = fast_conv(y)
        b_max = b.max()
        shifted = np.clip(tau * (b - b_max), -500, 500)
        exp_b = np.exp(shifted)
        p = exp_b / exp_b.sum()

        grad = lse_grad(y, p)

        t_old = t
        x_new = y - lr * grad
        x_new = np.maximum(x_new, 1e-10)
        ss = x_new.sum()
        if ss < 1e-10:
            x_new = a.copy()
        else:
            x_new /= ss

        t = (1.0 + np.sqrt(1.0 + 4.0 * t_old * t_old)) / 2.0
        y = x_new + ((t_old - 1.0) / t) * (x_new - x_old)
        y = np.maximum(y, 1e-10)
        y /= y.sum()
        x_old = x
        x = x_new

        if it % 15 == 0 or it == max_iters - 1:
            J_trial = compute_J(x)
            if J_trial < best_J - 1e-15:
                best_J = J_trial
                best_x = x.copy()
                stagnant = 0
                tracker.update(x)
            else:
                stagnant += 1

        if stagnant >= 4 and it > 30:
            lr *= 0.6
            y = x.copy()
            x_old = x.copy()
            t = 1.0
            stagnant = 0

        if lr < 1e-12:
            break

    tracker.update(best_x)
    return best_x

def solve_lp(a, M, time_limit=2.0):
    """Solve LP: max sum(g) s.t. (a*g)[k] <= M for all k, g >= 0."""
    n = len(a)
    rows, cols, nnz = get_sparsity(n)

    vals = np.empty(nnz)
    for i in range(nnz):
        k = rows[i]
        j = cols[i]
        vals[i] = a[k - j]

    A = csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n

    r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                options={'presolve': True, 'time_limit': time_limit})
    if r.success and r.x is not None:
        g = np.maximum(r.x, 0.0)
        gs = g.sum()
        if gs > 1e-10:
            return g / gs
    return None

def lp_refine(a_init, max_iters, deadline, tracker, tighten_schedule=None):
    """LP-based refinement with progressive tightening."""
    a = a_init.copy().astype(np.float64)
    a = np.maximum(a, 1e-10)
    a /= a.sum()
    n = len(a)
    current_J = compute_J(a)
    tracker.update(a)

    if tighten_schedule is None:
        tighten_schedule = [0.005, 0.01, 0.015, 0.02, 0.025, 0.03]

    total_it = 0
    for tighten in tighten_schedule:
        if time.monotonic() > deadline:
            break
        for _ in range(max(1, max_iters // len(tighten_schedule))):
            if time.monotonic() > deadline:
                break
            if total_it >= max_iters:
                break

            b = fast_conv(a)
            M = float(b.max())
            M_tight = M * (1.0 - tighten)

            g = solve_lp(a, M_tight, time_limit=2.0)
            if g is None:
                g = solve_lp(a, M, time_limit=2.0)
            if g is None:
                break

            # Line search over multiple step sizes
            best_t = 0.0
            best_J_local = current_J
            progress = total_it / max(1, max_iters)
            base_t = max(0.1, 0.6 * (1.0 - progress))
            candidates = [base_t, base_t * 0.5, base_t * 1.5, 0.15, 0.3, 0.5, 0.7, 0.85]
            for t_val in candidates:
                if t_val > 0.95 or t_val < 0.01:
                    continue
                a_trial = (1.0 - t_val) * a + t_val * g
                a_trial = np.maximum(a_trial, 1e-10)
                a_trial /= a_trial.sum()
                J_trial = compute_J(a_trial)
                if J_trial < best_J_local - 1e-15:
                    best_J_local = J_trial
                    best_t = t_val

            if best_t == 0.0:
                break

            a = (1.0 - best_t) * a + best_t * g
            a = np.maximum(a, 1e-10)
            a /= a.sum()
            current_J = best_J_local
            total_it += 1
            tracker.update(a)

    tracker.update(a)
    return a

def flat_spectrum_init(n, rng):
    """Generate sequence with approximately flat magnitude spectrum."""
    half = n // 2 + 1
    angles = rng.uniform(0, 2 * np.pi, half)
    A = np.zeros(n, dtype=complex)
    A[:half] = np.exp(1j * angles[:half])
    for k in range(1, n - n // 2):
        A[k] = np.conj(A[n - k])
    a = np.fft.ifft(A).real
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-10:
        return np.ones(n) / n
    return a / s

def multi_freq_init(n, rng):
    """Initialize with sum of cosines at random frequencies."""
    t = np.arange(n) / n
    a = np.ones(n)
    n_freqs = int(rng.integers(3, 9))
    for _ in range(n_freqs):
        f = rng.uniform(0.5, n / 3)
        amp = rng.uniform(0.05, 0.35)
        phase = rng.uniform(0, 2 * np.pi)
        a += amp * np.cos(2 * np.pi * f * t + phase)
    a = np.maximum(a, 0.01)
    return a / a.sum()

def weighted_random_init(n, rng):

    t = np.arange(n) / n
    a = rng.uniform(0.3, 1.0)
    smoothing = int(max(3, n // 20))
    if smoothing > 1:
        kernel = np.ones(smoothing) / smoothing
        a = np.convolve(a, kernel, mode='same')
    return np.maximum(a, 0.01)

def resample_seq(src, dest_n):
    """Resample a sequence to a different length using linear interpolation."""
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size, endpoint=False)
    t_new = np.linspace(0, 1, dest_n, endpoint=False)
    return np.maximum(np.interp(t_new, t_old, src), 0.001)

    def report(self, label=""):
        if self.best is not None:
            print(f"  {label} best: J={self.best_score:.8f} n={self.best.size}", flush=True)

def propose_candidate(seed=42, budget_s=1500, **kwargs):
    """Main search function. Explores multiple lengths and optimization strategies
    to minimize J(a) = 2n*max(convolve(a,a))/(sum(a))^2.
    Uses FISTA on LSE approximation combined with LP best-response refinement.
    """
    rng = np.random.default_rng(seed)
    total_deadline = time.monotonic() + float(budget_s) - 25.0
    tracker = ScoreTracker()

    # Load parent if available
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        if p.ndim == 1 and p.size > 0 and np.all(np.isfinite(p)):
            parent = np.maximum(p, 0.0)
            tracker.update(parent)
            print(f"Parent loaded: n={parent.size}, J={tracker.best_score:.8f}", flush=True)
    except (NameError, AttributeError, TypeError):
        pass

    # === Phase 1: Multi-length exploration (35% of time) ===
    phase1_frac = 0.35
    phase1_end = time.monotonic() + phase1_frac * (float(budget_s) - 30)

    print(f"\n{'='*60}", flush=True)
    print(f"Phase 1: Multi-length exploration (deadline={phase1_end - time.monotonic():.0f}s)", flush=True)
    print(f"{'='*60}", flush=True)

    lengths = [256, 384, 512, 640, 768]

    # Adjust lengths if parent is available
    if parent is not None:
        pn = parent.size
        if pn not in lengths:
            lengths = [l for l in lengths if abs(l - pn) < 200]
            if not lengths:
                lengths = [max(128, pn // 2), pn, min(1024, pn * 2)]
        # Ensure parent's length is included
        lengths = list(dict.fromkeys(lengths))  # dedupe preserving order

    best_overall = None
    best_overall_J = float('inf')

    for n in lengths:
        if time.monotonic() > phase1_end - 20:
            break

        t_start = time.monotonic()
        n_time_available = max(15, phase1_end - time.monotonic() - 5)

        inits = generate_inits(n, rng, parent)
        n_inits = len(inits)
        budget_per_init = max(8.0, n_time_available / max(1, n_inits) - 3)

        best_for_n = None
        best_J_for_n = float('inf')

        for idx, a0 in enumerate(inits):
            if time.monotonic() > phase1_end - 5:
                break
            remaining_for_init = min(budget_per_init, phase1_end - 5 - time.monotonic())
            if remaining_for_init < 5:
                break

            a = optimize_pipeline(a0, time.monotonic() + remaining_for_init, tracker, remaining_for_init)
            J = compute_J(a)

            if J < best_J_for_n:
                best_J_for_n = J
                best_for_n = a.copy()

        if best_for_n is not None:
            tracker.update(best_for_n)
            if best_J_for_n < best_overall_J:
                best_overall_J = best_J_for_n
                best_overall = best_for_n.copy()
            print(f"  n={n:4d}: best J={best_J_for_n:.8f} ({time.monotonic()-t_start:.1f}s)", flush=True)

    if tracker.best is not None:
        parent = tracker.best.copy()
    elif best_overall is not None:
        parent = best_overall

    # === Phase 2: Deep optimization on best (50% of time) ===
    phase2_deadline = time.monotonic() + 0.50 * (float(budget_s) - 30)
    phase2_deadline = min(phase2_deadline, total_deadline - 100)

    if parent is not None and time.monotonic() < phase2_deadline - 30:
        n = parent.size
        t_start = time.monotonic()

        print(f"\n{'='*60}", flush=True)
        print(f"Phase 2: Deep optimization n={n} (time={phase2_deadline - time.monotonic():.0f}s)", flush=True)
        print(f"{'='*60}", flush=True)

        a = parent.copy().astype(np.float64)
        a = np.maximum(a, 1e-10)
        a /= a.sum()

        # Progressive tightening LP passes
        tighten_phases = [
            [0.005, 0.01, 0.015],
            [0.01, 0.02, 0.03],
            [0.015, 0.025, 0.035, 0.045],
            [0.02, 0.035, 0.05, 0.065],
            [0.025, 0.04, 0.055, 0.07, 0.08],
        ]

        n_passes = len(tighten_phases) + 3  # +3 for FISTA phases
        budget_per_pass = (phase2_deadline - time.monotonic()) / n_passes

        for pi, sched in enumerate(tighten_phases):
            if time.monotonic() > phase2_deadline - 10:
                break
            pass_deadline = min(time.monotonic() + budget_per_pass * 1.2, phase2_deadline - 10)
            n_lp = max(10, int(40 * budget_per_pass / 30.0))
            a = lp_refine(a, n_lp, pass_deadline, tracker, tighten_schedule=sched)
            print(f"    LP pass {pi+1}/{len(tighten_phases)}: J={compute_J(a):.8f}", flush=True)

            # Interleaved FISTA
            if time.monotonic() < phase2_deadline - 20:
                fista_deadline = min(time.monotonic() + budget_per_pass * 0.8, phase2_deadline - 15)
                n_fista = max(200, int(5000 * budget_per_pass / 30.0))
                tau_s = n / (50.0 - 5.0 * pi)
                tau_e = n / (5.0 - 0.5 * pi)
                a = fista_lse(a, n_fista, fista_deadline, tracker, tau_start=tau_s, tau_end=tau_e)
                print(f"    FISTA after LP {pi+1}: J={compute_J(a):.8f}", flush=True)

        tracker.update(a)
        print(f"  Phase 2 done: J={compute_J(a):.8f} ({time.monotonic()-t_start:.1f}s)", flush=True)

        if compute_J(a) < tracker.best_score - 1e-15:
            parent = a.copy()

    # === Phase 3: Final polish (15% of time) ===
    phase3_deadline = total_deadline - 5

    if tracker.best is not None and time.monotonic() < phase3_deadline - 10:
        n = tracker.best.size
        t_start = time.monotonic()

        print(f"\n{'='*60}", flush=True)
        print(f"Phase 3: Final polish n={n} (time={phase3_deadline - time.monotonic():.0f}s)", flush=True)
        print(f"{'='*60}", flush=True)

        a = tracker.best.copy().astype(np.float64)
        a = np.maximum(a, 1e-10)
        a /= a.sum()

        # Very tight LP
        a = lp_refine(a, 15, min(time.monotonic() + 0.3 * (phase3_deadline - time.monotonic()), phase3_deadline - 20), tracker,
                      tighten_schedule=[0.003, 0.007, 0.012, 0.018, 0.025])

        # High-tau FISTA
        if time.monotonic() < phase3_deadline - 10:
            fista_dl = min(time.monotonic() + 0.35 * (phase3_deadline - time.monotonic()), phase3_deadline - 8)
            a = fista_lse(a, 5000, fista_dl, tracker, tau_start=n * 0.5, tau_end=n * 3.0)

        # Final tight LP
        if time.monotonic() < phase3_deadline - 5:
            a = lp_refine(a, 10, min(time.monotonic() + 0.25 * (phase3_deadline - time.monotonic()), phase3_deadline - 3), tracker,
                          tighten_schedule=[0.003, 0.008, 0.015, 0.022])

        # Final FISTA with very high tau
        if time.monotonic() < phase3_deadline - 2:
            a = fista_lse(a, 3000, phase3_deadline - 1, tracker, tau_start=n * 2.0, tau_end=n * 8.0)

        tracker.update(a)
        print(f"  Phase 3 done: J={compute_J(a):.8f} ({time.monotonic()-t_start:.1f}s)", flush=True)

    # Fallback
    if tracker.best is None:
        print("WARNING: No valid solution found, using fallback", flush=True)
        fallback = np.ones(256)
        tracker.update(fallback)

    print(f"\n{'='*60}", flush=True)
    print(f"FINAL: J={tracker.best_score:.8f} n={tracker.best.size} time={time.monotonic()-tracker.t_start:.1f}s", flush=True)
    print(f"{'='*60}", flush=True)

    return [float(x) for x in tracker.best]
```
