"""
Hybrid mirror descent + sparsity promotion + LP polishing for minimizing
2*n*max(conv(a,a))/(sum(a))^2. Uses multiplicative updates for natural
non-negativity handling, explicit zero-introduction for structural sparsity,
frequency-domain shaping, and LP polishing for final refinement.
"""

import numpy as np
from scipy.optimize import minimize, linprog
from scipy.fft import rfft, irfft
import time

def compute_conv(a):
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    conv = irfft(fa * fa, size)
    return conv[:2 * n - 1]

def compute_obj(a):
    n = len(a)
    if n == 0:
        return np.inf
    conv = compute_conv(a)
    max_b = float(np.max(conv))
    sum_a = float(np.sum(a))
    if sum_a < 0.01:
        return np.inf
    return 2.0 * n * max_b / (sum_a * sum_a)

def fft_cross_correlate(a, b):
    """Compute correlation: result[i] = sum_j a[j] * b[i+j] for valid indices."""
    na, nb = len(a), len(b)
    size = 1
    while size < na + nb - 1:
        size <<= 1
    fa = rfft(a, size)
    fb = rfft(b, size)
    return irfft(fa * fb, size)[:na + nb - 1]

def mirror_descent_step(a, conv, tau, eta):
    """One step of mirror descent on the soft-max of conv values."""
    n = len(a)
    cmax = float(np.max(conv))
    shifted = np.clip((conv - cmax) / tau, -500.0, 500.0)
    exp_conv = np.exp(shifted)
    Z = float(np.sum(exp_conv))
    w = exp_conv / Z
    
    # Gradient: grad_a[i] = 2 * sum_{m=0}^{n-1} w[i+m] * a[m]
    w_rev = w[::-1]
    conv_fw = fft_cross_correlate(a, w_rev)
    # grad_a[i] = 2 * conv_fw[2n-2-i]
    grad_a = 2.0 * conv_fw[n-1:2*n-1][::-1]
    
    # Center gradient (scale invariance)
    grad_centered = grad_a - np.sum(a * grad_a) / np.sum(a)
    
    # Multiplicative update
    a_new = a * np.exp(-eta * grad_centered)
    a_new = np.maximum(a_new, 0)
    s = np.sum(a_new)
    if s > 1e-10:
        a_new = a_new / s
    return a_new

def mirror_descent_optimize(a_init, time_budget=30.0, tau_schedule=None, eta_schedule=None):
    """Mirror descent optimization with adaptive parameters."""
    n = len(a_init)
    a = a_init.copy().astype(np.float64)
    s = np.sum(a)
    if s < 1e-10:
        a = np.ones(n) / n
    else:
        a = a / s
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    
    if tau_schedule is None:
        tau_schedule = np.linspace(10.0/n, 1.0/n, 5)
    
    total_iters = max(100, int(time_budget * 50))
    
    for it in range(total_iters):
        if time.monotonic() - start > time_budget:
            break
        
        frac = it / total_iters
        # Interpolate tau
        tau_idx = min(int(frac * len(tau_schedule)), len(tau_schedule) - 1)
        tau = tau_schedule[tau_idx]
        
        # Adaptive eta: start large, decay
        eta = 0.5 * (1.0 + 0.5 * np.cos(np.pi * frac)) / np.sqrt(n)
        
        conv = compute_conv(a)
        a_new = mirror_descent_step(a, conv, tau, eta)
        a = a_new
        
        if it % 50 == 0:
            obj = compute_obj(a)
            if obj < best_obj:
                best_obj = obj
                best_a = a.copy()
    
    obj = compute_obj(a)
    if obj < best_obj:
        best_obj = obj
        best_a = a.copy()
    
    return best_a, best_obj

def sparsify(a, sparsity_level, threshold_factor=2.0):
    """Set small elements to zero to promote sparsity."""
    n = len(a)
    a = a.copy()
    s = np.sum(a)
    if s < 1e-10:
        return a
    a_norm = a / s
    
    # Set elements below threshold to zero
    threshold = sparsity_level * np.median(a_norm[a_norm > 0]) if np.any(a_norm > 0) else 0
    a[a_norm < threshold] = 0
    
    s = np.sum(a)
    if s > 1e-10:
        a = a / s
    
    return a

def sparsity_promotion(a, time_budget=20.0):
    """Try different sparsity levels and keep the best."""
    n = len(a)
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    
    sparsity_levels = [0.05, 0.1, 0.15, 0.2, 0.25, 0.3]
    
    for sp in sparsity_levels:
        if time.monotonic() - start > time_budget - 5:
            break
        
        a_sp = sparsify(a, sp)
        
        # Quick mirror descent to recover from sparsification damage
        a_opt, obj = mirror_descent_optimize(a_sp, time_budget=min(5.0, (time_budget - (time.monotonic() - start)) * 0.5))
        
        if obj < best_obj:
            best_obj = obj
            best_a = a_opt.copy()
    
    return best_a, best_obj

def peak_reduction(a, time_budget=20.0):
    """Focus optimization on reducing the peak convolution value."""
    n = len(a)
    a = a.copy().astype(np.float64)
    s = np.sum(a)
    if s < 1e-10:
        return a
    a = a / s
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    
    for it in range(max(50, int(time_budget * 20))):
        if time.monotonic() - start > time_budget:
            break
        
        conv = compute_conv(a)
        peak_lag = int(np.argmax(conv))
        peak_val = conv[peak_lag]
        
        # Find elements that contribute most to the peak
        # C[peak] = sum_j a[j]*a[peak-j]
        lo = max(0, peak_lag - n + 1)
        hi = min(peak_lag, n - 1)
        contributions = a[lo:hi+1] * a[peak_lag - lo:peak_lag - hi]
        
        # Reduce the largest contributors (multiplicatively)
        order = np.argsort(contributions)[::-1]
        n_reduce = max(1, n // 10)
        for idx in order[:n_reduce]:
