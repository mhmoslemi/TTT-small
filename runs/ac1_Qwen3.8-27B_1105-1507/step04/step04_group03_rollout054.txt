Hybrid optimizer combining Gerchberg-Saxton flat-spectrum method,
Nesterov-accelerated subgradient descent, and LP mirror descent with
bisection-based M reduction. Multi-length exploration with adaptive
time allocation.
"""

import numpy as np
from scipy.optimize import linprog
from scipy.sparse import csr_matrix
import time


def fast_conv(a):
    """Full self-convolution via FFT."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]


def compute_J(a):
    """Compute J(a) = 2*n*max(conv(a,a))/(sum(a))^2."""
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))


def project_simplex(v):
    """Project v onto {x >= 0, sum(x) = 1} using the standard algorithm."""
    n = len(v)
    if n == 1:
        return np.array([1.0])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    rho = np.max(np.where(cond)[0]) if np.any(cond) else 0
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0)


def gerchberg_saxton(a, n_outer, n_inner, alpha, deadline):
    """Gerchberg-Saxton flat spectrum method with non-negativity projection."""
    a = np.maximum(a.copy(), 0.0)
    s = a.sum()
    if s < 1e-10:
        return a
    a = a / s

    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1

    best_a = a.copy()
    best_M = fast_conv(a).max()
    no_improve = 0

    for outer in range(n_outer):
        if time.monotonic() > deadline:
            break

        for inner in range(n_inner):
            A = np.fft.rfft(a, size)
            A_mag = np.abs(A)
            A_phase = A / np.maximum(A_mag, 1e-15)

            avg_mag = np.mean(A_mag[1:]) if len(A_mag) > 1 else A_mag[0]
            target_mag = np.full(len(A_mag), avg_mag)
            target_mag[0] = A_mag[0]

            # Blend

def subgradient_nesterov(a, max_iters, deadline, lr0=0.05):
    """Nesterov-accelerated subgradient descent on the simplex."""
    a = np.maximum(a.astype(np.float64), 1e-12)
    a = a / a.sum()
    n = len(a)

    best_M = fast_conv(a).max()
    best_a = a.copy()

    x = a.copy()
    y = x.copy()
    t_k = 1.0

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        b = fast_conv(y)
        k_star = int(b.argmax())

        # Subgradient: grad[i] = 2*y[k_star - i] for valid i
        grad = np.zeros(n)
        i_min = max(0, k_star - (n - 1))
        i_max = min(n - 1, k_star)
        if i_min <= i_max:

        x = x_new
        t_k = t_new

        M = fast_conv(x).max()

def build_lp_matrix(n, a, M, b_rhs):
    """Build sparse constraint matrix for LP: (a*g)[k] <= b_rhs[k]."""
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        j_min = max(0, k - n + 1)
        j_max = min(n - 1, k)
        for j in range(j_min, j_max + 1):
            rows.append(k)
            cols.append(j)
            vals.append(a[k - j])
    return csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))


def lp_best_response(a, M):
    """Solve LP: max sum(g) s.t. (a*g)[k] <= M for all k, g >= 0."""
    n = len(a)
    A = build_lp_matrix(n, a, M, None)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    try:

        if r.success and r.x is not None:

    except Exception:
        pass
    return None

def resample_seq(src, dest_n):
    """Resample 1D sequence to new length."""
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)

    print(f"Phase 4: LP multiplicative (n={len(current_best)})", flush=True)
    print(f"{'='*50}", flush=True)
    t0 = time.monotonic()
    lp2_deadline = min(deadline - 250, time.monotonic() + 100)
    a_lp2, J_lp2 = lp_mirror_descent_v2(current_best, 400, lp2_deadline,
                                         use_multiplicative=True,
                                         step_decay=0.2, step_base=0.4, step_min=0.05)
    tracker.update(a_lp2)
    print(f"  LP mult: J={J_lp2:.8f} ({time.monotonic()-t0:.1f}s)", flush=True)
    if J_lp2 < current_best_J:
        current_best = a_lp2.copy()
        current_best_J = J_lp2

    # === Phase 5: Bisection LP ===
    print(f"\n{'='*50}", flush=True)
    print(f"Phase 5: Bisection LP (n={len(current_best)})", flush=True)
    print(f"{'='*50}", flush=True)
    t0 = time.monotonic()
    bis_deadline = min(deadline - 200, time.monotonic() + 80)
    a_bis, M_bis = bisection_lp(current_best, 30, bis_deadline)
    J_bis = compute_J(a_bis)
    tracker.update(a_bis)
    print(f"  Bisection LP: J={J_bis:.8f} ({time.monotonic()-t0:.1f}s)", flush=True)
    if J_bis < current_best_J:
        current_best = a_bis.copy()
        current_best_J = J_bis

    # Polish bisection result with LP
    if time.monotonic() < deadline - 150:
        t0 = time.monotonic()
        a_bis_lp, J_bis_lp = lp_mirror_descent_v2(a_bis, 200, min(deadline - 150, time.monotonic() + 60),
                                                   use_multiplicative=False)
        tracker.update(a_bis_lp)
        print(f"  Bis+LP: J={J_bis_lp:.8f} ({time.monotonic()-t0:.1f}s)", flush=True)
        if J_bis_lp < current_best_J:
            current_best = a_bis_lp.copy()
            current_best_J = J_bis_lp

    # === Phase 6: Multi-length exploration ===
    print(f"\n{'='*50}", flush=True)
    print(f"Phase 6: Multi-length (base n={len(current_best)})", flush=True)
    print(f"{'='*50}", flush=True)
    t0 = time.monotonic()

    base_n = len(current_best)
    candidate_lengths = []
    for delta in [-64, -32, 32, 64, 96, 128]:
        n_try = base_n + delta
        if 192 <= n_try <= 2048 and n_try != base_n:
            candidate_lengths.append(n_try)
    candidate_lengths = sorted(set(candidate_lengths))

    remaining = deadline - 120 - time.monotonic()
    per_len_time = max(10, remaining / max(1, len(candidate_lengths)))

    for n_try in candidate_lengths:
        if time.monotonic() > deadline - 120:
            break
        a_try = resample_seq(current_best, n_try)
        a_try = np.maximum(a_try, 1e-10)
        a_try = a_try / a_try.sum()

        # Quick LP
        a_try, J_try = lp_mirror_descent_v2(a_try, 200, time.monotonic() + per_len_time * 0.7,
                                             use_multiplicative=False)
        tracker.update(a_try)
        J_try = compute_J(a_try)
        print(f"  n={n_try}: J={J_try:.8f}", flush=True)

        if J_try < current_best_J:
            current_best = a_try.copy()
            current_best_J = J_try

    # === Phase 7: Long LP refinement on absolute best ===
    print(f"\n{'='*50}", flush=True)
    print(f"Phase 7: Long LP polish (n={len(current_best)})", flush=True)
    print(f"{'='*50}", flush=True)
    t0 = time.monotonic()
    a_final, J_final = lp_mirror_descent_v2(current_best, 5000, deadline - 20,
                                             use_multiplicative=False,
                                             step_decay=0.15, step_base=0.2, step_min=0.02)
    tracker.update(a_final)
    print(f"  Final LP linear: J={J_final:.8f} ({time.monotonic()-t0:.1f}s)", flush=True)

    if J_final < current_best_J:
        current_best = a_final.copy()
        current_best_J = J_final

    # Try multiplicative one more time
    if time.monotonic() < deadline - 10:
        a_final2, J_final2 = lp_mirror_descent_v2(current_best, 2000, deadline - 10,
                                                   use_multiplicative=True,
                                                   step_decay=0.1, step_base=0.3, step_min=0.02)
        tracker.update(a_final2)
        print(f"  Final LP mult: J={J_final2:.8f}", flush=True)
        if J_final2 < current_best_J:
            current_best = a_final2.copy()
            current_best_J = J_final2

    # === Phase 8: Final subgradient polish ===
    if time.monotonic() < deadline - 5:
        a_sp = subgradient_nesterov(current_best, 10000, deadline - 5, lr0=0.02)
        tracker.update(a_sp)
        J_sp = compute_J(a_sp)
        if J_sp < current_best_J:
            current_best = a_sp.copy()
            current_best_J = J_sp
        print(f"  Final subgradient: J={J_sp:.8f}", flush=True)

    # Final decision
    if tracker.best is not None:
        final_a = tracker.best
    else:
        final_a = current_best

    print(f"\n{'='*50}", flush=True)
    print(f"FINAL: J={tracker.best_score:.8f} n={final_a.size} time={time.monotonic()-tracker.t_start:.0f}s", flush=True)
    print(f"{'='*50}", flush=True)

    return [float(x) for x in final_a]
