def project_simplex(v):
    """Project v onto {x >= 0, sum(x) = 1}."""

    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0)

        active = np.array([np.argmax(b)])
    
    # Build reduced A matrix
    rows = []
    cols = []
    vals = []
    for idx, k in enumerate(active):
        j_min = max(0, k - n + 1)
        j_max = min(n - 1, k)
        for j in range(j_min, j_max + 1):
            rows.append(idx)
            cols.append(j)
            vals.append(a[k - j])
    
    A_red = csr_matrix((vals, (rows, cols)), shape=(len(active), n))
    c = -np.ones(n)
    b_ub = np.full(len(active), M)
    bounds = [(0, None)] * n
    
    try:
        r = linprog(c, A_ub=A_red, b_ub=b_ub, bounds=bounds, method='highs')
        if r.success and r.x is not None:
            g = np.maximum(r.x, 0)
            if g.sum() > 1e-15:
                return g
    except Exception:
        pass
    return None
