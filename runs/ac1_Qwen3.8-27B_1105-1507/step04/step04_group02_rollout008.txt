"""
Optimized step-function coefficient search minimizing 2*n*max(conv(a,a))/(sum(a))^2.
Uses soft-max smoothed subgradient with Nesterov momentum, adaptive step size,
multi-scale exploration, and diverse initializations including chirp-like sequences.
"""

import numpy as np
import time
from scipy.interpolate import interp1d


def compute_conv(a):
    """FFT-based linear convolution of a with itself."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    S = float(np.sum(a))
    if S < 1e-10:
        return np.inf
    conv = compute_conv(a)
    M = float(np.max(conv))
    return 2.0 * n * M / (S * S)


def normalize(a):
    """Clip to non-negative and normalize to sum 1."""
    a = np.maximum(a, 0.0)
    S = np.sum(a)
    if S < 1e-15:
        return np.ones(len(a) + 1) / (len(a) + 1)  # shouldn't happen
    return a / S


def compute_soft_gradient(a, temperature=0.02, topk=50):
    """
    Compute gradient of the soft-max approximation of max(conv(a,a)).
    Uses importance sampling: only compute contributions from top-k lags.
    """
    n = len(a)
    conv = compute_conv(a)
    
    # Soft weights
    conv_max = conv.max()
    # Use top-k lags for efficiency
    if topk < len(conv):
        topk_idx = np.argsort(conv)[-topk:]
    else:
        topk_idx = np.arange(len(conv))
    
    # Softmax weights over top-k
    w = np.exp((conv[topk_idx] - conv_max) / temperature)
    w_sum = w.sum()
    if w_sum < 1e-30:
        w = np.ones(len(topk_idx)) / len(topk_idx)
    else:
        w = w / w_sum
    
    # Gradient: grad[i] = sum_k w_k * 2 * a[k-i] (for valid k-i)
    grad = np.zeros(n)
    for idx, k in enumerate(topk_idx):
        wi = 2.0 * w[idx]
        i_lo = max(0, k - (n - 1))
        i_hi = min(n - 1, k)
        i_vals = np.arange(i_lo, i_hi + 1)
        j_vals = k - i_vals
        grad[i_vals] += wi * a[j_vals]
    
    return grad, conv_max, conv


def nesterov_optimize(a, time_budget, topk=30, beta_init=0.05, 
                       momentum_init=0.9, step_init=0.1, 
                       perturbation_interval=200, perturbation_scale=0.1):
    """
    Nesterov accelerated subgradient descent with adaptive parameters.
    """
    n = len(a)
    a = normalize(a.copy())
    v = np.zeros(n)
    y = a.copy()
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    
    step = step_init
    momentum = momentum_init
    beta = beta_init  # temperature for soft-max
    
    start = time.monotonic()
    stall = 0
    it = 0
    
    while time.monotonic() - start < time_budget:
        it += 1
        # Nesterov lookahead
        y = a + momentum * v
        
        # Compute gradient at y
        grad, M, conv = compute_soft_gradient(y, temperature=beta, topk=topk)
        
        # Center gradient (to preserve sum)
        grad = grad - grad.mean()
        
        # Update velocity
        v = momentum * v - step * grad
        
        # Update position
        a_new = y + v
        a_new = normalize(a_new)
        
        # Evaluate
        obj = compute_obj(a_new)
        
        if obj < best_obj:
            best_a = a_new.copy()
            best_obj = obj
            a = a_new
            stall = 0
        else:
            stall += 1
            
            if stall % 10 == 0:
                # Adaptive step size
                step *= 0.95
                momentum = min(0.999, momentum * 1.001)
                # Reduce temperature (sharpen the max approximation)
                beta = max(0.001, beta * 0.98)
            
            if stall > perturbation_interval:
                # Perturbation
                rng = np.random.default_rng(it * 7919)
                scale = perturbation_scale * (1.0 + 0.5 * np.sin(it * 0.1))
                a = a * np.exp(rng.normal(0, scale, n))
                a = normalize(a)
                v = np.zeros(n)
                step = step_init * 0.5
                momentum = momentum_init
                beta = beta_init
                stall = 0
    
    return best_a, best_obj


def cosine_init(n, n_modes=10, seed=0, alpha_max=0.8):
    """Cosine series initialization with random phases."""
    rng = np.random.default_rng(seed)
    k = np.arange(n)
    a = np.ones(n)
    for m in range(1, n_modes + 1):
        c = rng.uniform(-alpha_max, alpha_max) / (0.5 + 0.5 * m)
        phase = rng.uniform(0, 2 * np.pi)
        a += c * np.cos(2.0 * np.pi * m * k / n + phase)
    return normalize(a)


def chirp_init(n, alpha=0.5, seed=0):
    """Chirp (quadratic phase) initialization."""
    rng = np.random.default_rng(seed)
    k = np.arange(n)
    phi0 = rng.uniform(0, 2 * np.pi)
    a = 1.0 + alpha * np.cos(np.pi * k * k / (n * 0.5) + phi0)
    return normalize(a)


def linear_phase_init(n, freq, alpha=0.5, seed=0):
    """Linear phase (complex exponential real part) initialization."""
    rng = np.random.default_rng(seed)
    k = np.arange(n)
    phase = rng.uniform(0, 2 * np.pi)
    a = 1.0 + alpha * np.cos(2.0 * np.pi * freq * k / n + phase)
    return normalize(a)


def interpolate_sequence(a, new_n):
    """Interpolate a sequence to a new length."""
    n = len(a)
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    try:
        f = interp1d(x_old, a, kind='cubic', fill_value='extrapolate')
        a_new = np.maximum(f(x_new), 0)
    except Exception:
        a_new = np.ones(new_n) / new_n
    return normalize(a_new)


def try_load_parent():
    """Try to load parent sequence."""
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64)
        parent = np.clip(parent, 0, 1000.0)
        if parent.size > 0 and np.isfinite(parent).all():
            return parent.copy()
    except (NameError, Exception):
        pass
    return None
