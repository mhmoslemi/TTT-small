```
def project_simplex(v, s=1):
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.findlast((u - (cssv - s) / (np.arange(1, n+1))) > 0)
    theta = (cssv[rho] - s) / (rho + 1)
    return np.maximum(v - theta, 0)
```

```python
import numpy as np
from scipy.optimize import linprog
from scipy.sparse import csr_matrix
import time


def fast_conv(a):
    n = len(a)
    size = 1
    while size < 2*n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2*n-1]


def smooth_max_grad(a, tau):
    """Compute gradient of (1/tau)*log(sum exp(tau*conv(a,a))) w.r.t. a."""
    n = len(a)
    c = fast_conv(a)
    c_max = c.max()
    exp_c = np.exp(tau * (c - c_max))
    p = exp_c / exp_c.sum()
    
    # grad[i] = 2 * sum_j p[i+j] * a[j] = 2 * (p cross-corr a)[i]
    N = 1
    while N < (2*n-1) + n - 1:
        N <<= 1
    fp = np.fft.fft(p, N)
    fa = np.fft.fft(a, N)
    cross = np.fft.ifft(fp * np.conj(fa)).real[:n]
    
    return 2.0 * cross, p, c


def project_simplex(v, s=1.0):
    """Project v onto {x: x>=0, sum(x)=s}."""
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.findlast((u - (cssv - s) / np.arange(1, n+1)) > 0)
    if rho is None:
        rho = 0
    theta = (cssv[rho] - s) / (rho + 1)
    return np.maximum(v - theta, 0.0)


def compute_J(a):
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))


def nesterov_smooth_max(a0, max_iters, deadline, tau_start=50.0, tau_end=5000.0):
    """Nesterov accelerated projected gradient on smooth max."""
    a = np.maximum(a0.astype(np.float64), 1e-15)
    a = a / a.sum()
    n = len(a)
    
    best_a = a.copy()
    best_val = fast_conv(a).max()
    
    y = a.copy()  # Nesterov lookahead
    a_prev = a.copy()
    
    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        
        # Adaptively increase tau
        frac = min(1.0, it / max(1, max_iters))
        tau = tau_start * (tau_end / tau_start) ** frac
        
        # Compute gradient at y
        grad, p, c = smooth_max_grad(y, tau)
        c_max = c.max()
        
        if c_max < best_val:
            best_val = c_max
            best_a = a.copy()
        
        # Nesterov step
        momentum = (it - 1) / (it + 2) if it > 0 else 0.0
        y = a + momentum * (a - a_prev)
        y = project_simplex(y)
        
        # Backtracking line search
        # f(y) = (1/tau) * log(sum exp(tau * c(y)))  (shifted by c_max for stability)
        def f_at(x):
            cx = fast_conv(x)
            cx_max = cx.max()
            return float(np.log(np.sum(np.exp(tau * (cx - cx_max)))) / tau + cx_max)
        
        f_y = f_at(y)
        step = 1.0 / np.linalg.norm(grad)
        if step > 1.0:
            step = 1.0
        
        improved = False
        for _ in range(30):
            x_new = project_simplex(y - step * grad)
            f_new = f_at(x_new)
            if f_new < f_y - 1e-12:
                improved = True
                break
            step *= 0.5
        
        if not improved:
            # Try a smaller fixed step
            x_new = project_simplex(y - 0.01 * grad / np.linalg.norm(grad))
        
        a_prev = a
        a = x_new
    
    # Compare best with final
    if fast_conv(best_a).max() < fast_conv(a).max():
        return best_a
    return a
```

Final note on the simplex projection: The standard algorithm (Duchi et al.) is:
1. Sort v in descending order: v_(1) >= v_(2) >= ... >= v_(n)

4. theta = (S_k - s) / k
5. Projected point: max(v - theta, 0)

"""
Minimizes J(a) = 2*n*max(conv(a,a))/(sum(a))^2 using:
1. FFT-accelerated Nesterov projected gradient on smooth max (O(n log n) per step)
2. LP best-response dynamics for precise refinement
3. Multi-length exploration (n=512, 1024)
4. SLSQP final polish
Key: very high tau (up to 50000) for near-exact max, Nesterov O(1/k^2) rate,
   FFT cross-correlation for gradient, simplex projection via sorting.

import numpy as np
from scipy.optimize import linprog, minimize
from scipy.sparse import csr_matrix
import time


def fast_conv(a):
    """Full self-convolution via FFT. O(n log n)."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]


def smooth_max_gradient_fft(a, tau):
    """
    Compute gradient of (1/tau)*log(sum_k exp(tau*conv(a,a)[k])) w.r.t. a.
    Uses FFT cross-correlation: grad[i] = 2 * sum_j p[i+j]*a[j].
    Returns (gradient, softmax_weights, convolution).
    """
    n = len(a)
    c = fast_conv(a)
    c_max = c.max()
    exp_c = np.exp(tau * (c - c_max))
    p = exp_c / exp_c.sum()

    # grad[i] = 2 * (p cross-corr a)[i] = 2 * ifft(FFT(p)*conj(FFT(a))).real[i]
    N = 1
    while N < (2 * n - 1) + n - 1:
        N <<= 1
    fp = np.fft.fft(p, N)
    fa = np.fft.fft(a, N)
    cross = np.fft.ifft(fp * np.conj(fa)).real[:n]

    return 2.0 * cross, p, c


def project_simplex(v, s=1.0):
    """Project v onto {x >= 0, sum(x) = s} in O(n log n)."""
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = (u - (cssv - s) / np.arange(1, n + 1)) > 0
    idx = np.where(cond)[0]
    if len(idx) == 0:
        return np.full(n, s / n)
    rho = idx[-1]
    theta = (cssv[rho] - s) / (rho + 1)
    return np.maximum(v - theta, 0.0)


def compute_J(a):
    """Compute J(a) = 2*n*max(conv(a,a))/(sum(a))^2."""
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))


def nesterov_smooth(a0, max_iters, deadline, tau_start=30.0, tau_end=20000.0,
                    lr=None):
    """
    Nesterov accelerated projected gradient on smooth max over the simplex.
    Uses FFT-based gradient computation.
    """
    a = np.maximum(a0.astype(np.float64), 1e-15)
    a = a / a.sum()
    n = len(a)

    best_a = a.copy()
    best_val = fast_conv(a).max()
    y = a.copy()
    a_prev = a.copy()
    it_count = 0

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        # Adaptive tau: exponential ramp
        frac = min(1.0, (it + 1) / max(1, max_iters))
        tau = tau_start * (tau_end / tau_start) ** frac

        # Compute gradient at y (Nesterov lookahead point)
        grad, p, c = smooth_max_gradient_fft(y, tau)
        c_max = c.max()

        if c_max < best_val:
            best_val = c_max
            best_a = a.copy()

        # Nesterov momentum
        if it == 0:
            y = a.copy()
        else:
            momentum = (it - 1 + 1.0) / (it + 1 + 2.0)
            y = a + momentum * (a - a_prev)
            y = project_simplex(y)
            # Recompute gradient at updated y
            grad, p, c = smooth_max_gradient_fft(y, tau)
            c_max = c.max()
            if c_max < best_val:
                best_val = c_max
                best_a = a.copy()

        # Step size: based on Lipschitz constant estimate
        if lr is None:
            # L ~ 2*tau*max(c) + 2*max(a) roughly
            L_est = 2.0 * tau * c.max() + 2.0 * a.max() + 1e-8
            step = 1.0 / L_est
        else:
            step = lr

        a_prev = a.copy()
        a_new = project_simplex(y - step * grad)
        a = a_new
        it_count += 1

    # Final comparison
    val_final = fast_conv(a).max()
    val_best = fast_conv(best_a).max()
    if val_best < val_final:
        return best_a
    return a


def subgradient_nesterov(a0, max_iters, deadline, lr_base=0.5):
    """
    Nesterov accelerated subgradient method on the exact max of conv(a,a).
    Uses random active subgradients for variance reduction.
    """
    a = np.maximum(a0.astype(np.float64), 1e-15)
    a = a / a.sum()
    n = len(a)
    rng = np.random.default_rng(12345)

    best_a = a.copy()
    best_val = fast_conv(a).max()
    y = a.copy()
    a_prev = a.copy()

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        # Nesterov lookahead
        if it == 0:
            y = a.copy()
        else:
            momentum = it / (it + 2.0)
            y = a + momentum * (a - a_prev)
            y = project_simplex(y)

        # Compute max and active set
        c = fast_conv(y)
        M = c.max()

        if M < best_val:
            best_val = M
            best_a = a.copy()

        # Get active set (within tolerance of max)
        tol = 1e-6 * M + 1e-15
        active = np.where(c >= M - tol)[0]

        # Random subgradient: pick a random active lag
        k = active[rng.integers(len(active))]

        # Gradient of c[k] w.r.t. a: grad[i] = 2*a[k-i] for i in valid range
        grad = np.zeros(n)
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        grad[lo:hi + 1] = 2.0 * a[k - hi:k - lo + 1][::-1]

        # Also use the lookahead y for the gradient direction
        grad_y = np.zeros(n)
        grad_y[lo:hi + 1] = 2.0 * y[k - hi:k - lo + 1][::-1]

        # Step size (diminishing)
        step = lr_base / (it + 1) ** 0.5

        a_prev = a.copy()
        a = project_simplex(y - step * grad_y)

    val_final = fast_conv(a).max()
    val_best = fast_conv(best_a).max()
    if val_best < val_final:
        return best_a
    return a


def lp_best_response(a, M):
    """
    LP: max sum(g) s.t. (a*g)[k] <= M for all k, g >= 0.
    Returns optimal g or None.
    """
    n = len(a)
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        i_lo = max(0, k - n + 1)
        i_hi = min(n - 1, k)
        for i in range(i_lo, i_hi + 1):
            rows.append(k)
            cols.append(i)
            vals.append(a[k - i])

    if len(rows) == 0:
        return None

    B = csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))
    c_obj = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n

    try:
        r = linprog(c_obj, A_ub=B, b_ub=b_ub, bounds=bounds, method='highs',
                    options={'time_limit': 60, 'presolve': True})
        if r.success and r.x is not None:
            g = np.maximum(r.x, 0.0)
            if g.sum() > 1e-15:
                return g
    except Exception:
        pass
    return None


def lp_mirror_descent(a0, max_iters, deadline):
    """
    LP mirror descent (best-response dynamics).
    At each step: solve LP for best response g given current a and M,
    then mix a <- (1-alpha)*a + alpha*g/sum(g).
    """
    a = np.maximum(a0.astype(np.float64), 1e-15)
    a = a / a.sum()
    n = len(a)

    best_a = a.copy()
    best_val = fast_conv(a).max()
    no_improve = 0

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        c = fast_conv(a)
        M = float(c.max())

        g = lp_best_response(a, M)
        if g is None:
            break

        g = np.maximum(g, 0.0)
        g_sum = g.sum()
        if g_sum < 1e-12:
            break
        g /= g_sum

        # Diminishing step size
        alpha = max(0.05, 0.8 / (1.0 + 0.1 * it))

        improved = False
        for t in [alpha, alpha * 0.7, alpha * 0.5, alpha * 0.3, 0.05, 0.02]:
            a_trial = (1.0 - t) * a + t * g
            a_trial = np.maximum(a_trial, 0.0)
            s = a_trial.sum()
            if s < 1e-10:
                continue
            a_trial /= s
            M_trial = fast_conv(a_trial).max()
            if M_trial < best_val - 1e-15:
                a = a_trial
                best_val = M_trial
                best_a = a.copy()
                improved = True
                break

        if improved:
            no_improve = 0
        else:
            no_improve += 1
            if no_improve > 30:
                break

    return best_a


def resample_seq(src, dest_n):
    """Resample 1D sequence to new length via linear interpolation."""
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)


def slsqp_polish(a0, deadline, tau=200.0):
    """SLSQP refinement on smooth max objective."""
    a = np.maximum(a0.astype(np.float64), 1e-10)
    a = a / a.sum()
    n = len(a)

    def obj(x):
        ax = np.maximum(x, 0.0)
        s = ax.sum()
        if s < 1e-8:
            return 1e10
        ax = ax / s
        c = fast_conv(ax)
        c_max = c.max()
        exp_c = np.exp(tau * (c - c_max))
        return float(np.log(exp_c.sum()) / tau + c_max)

    def obj_grad(x):
        ax = np.maximum(x, 0.0)
        s = ax.sum()
        if s < 1e-8:
            return np.zeros_like(x)
        ax = ax / s
        c = fast_conv(ax)
        c_max = c.max()
        exp_c = np.exp(tau * (c - c_max))
        p = exp_c / exp_c.sum()
        N = 1
        while N < (2 * len(ax) - 1) + len(ax) - 1:
            N <<= 1
        fp = np.fft.fft(p, N)
        fa = np.fft.fft(ax, N)
        cross = np.fft.ifft(fp * np.conj(fa)).real[:len(ax)]
        return 2.0 * cross

    bounds = [(0.0, None)] * n
    cons = [{'type': 'eq', 'fun': lambda x: x.sum() - 1.0,
             'jac': lambda x: np.ones(n)}]

    try:
        res = minimize(obj, a, jac=obj_grad, method='SLSQP', bounds=bounds,
                       constraints=cons,
                       options={'maxiter': 200, 'ftol': 1e-16, 'disp': False})
        if res.success or res.fun < 1e10:
            return np.maximum(res.x, 0.0)
    except Exception:
        pass
    return a
