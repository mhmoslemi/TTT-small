Minimizes J(a) = 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Gerchberg-Saxton spectral equalization with non-negativity
2. Frequency-domain water-filling
3. LSE-smoothed gradient descent (different parameters)

5. Phase randomization for escape from local minima

import numpy as np
import time

def fast_conv(a):
    n = len(a)
    s = 1
    while s < 2 * n - 1:
        s <<= 1
    fa = np.fft.rfft(a, s)
    return np.fft.irfft(fa * fa, s)[:2 * n - 1]

def compute_J(a):
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))

def gs_spectral_equalize(a, max_iters, deadline, tracker, alpha_schedule=None):
    """Gerchberg-Saxton: alternate between flat spectrum in freq domain and non-negativity in time domain."""
    a = np.maximum(a.astype(np.float64), 1e-10)
    n = len(a)
    a = a / a.sum()
    current_J = compute_J(a)
    tracker.update(a)
    
    alpha = 0.3  # blending factor
    
    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        
        A = np.fft.fft(a)
        mag = np.abs(A)
        phase = np.angle(A)
        
        # Target: geometric mean or mean of magnitudes
        # Use log-mean to avoid zero issues
        log_mag = np.log(mag + 1e-15)
        target_log = np.mean(log_mag)
        target_mag = np.exp(target_log)
        
        # Blend
        new_log_mag = (1 - alpha) * log_mag + alpha * target_log
        new_mag = np.exp(new_log_mag)
        
        # Reconstruct with original phases
        new_A = new_mag * np.exp(1j * phase)
        
        # Enforce conjugate symmetry
        a_new = np.fft.ifft(new_A).real
        a_new = np.maximum(a_new, 0.0)
        
        if a_new.sum() < 1e-10:
            alpha *= 0.5
            continue
        
        a_new = a_new / a_new.sum()
        J_new = compute_J(a_new)
        
        if J_new < current_J - 1e-15:
            a = a_new
            current_J = J_new
            alpha = min(0.5, alpha * 1.05)
        else:
            alpha *= 0.7
            if alpha < 0.01:
                break
        
        if it % 20 == 0:
            tracker.update(a)
    
    tracker.update(a)
    return a

def lse_gradient(a, max_iters, deadline, tracker, tau0=20.0):
    """LSE-smoothed gradient descent with momentum."""
    a = np.maximum(a.astype(np.float64), 1e-8)
    n = len(a)
    a = a / a.sum()
    current_J = compute_J(a)
    tracker.update(a)
    
    tau = tau0
    lr = 2.0 / (n * tau)
    momentum = 0.9
    v = np.zeros(n)
    
    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        
        b = fast_conv(a)
        b_max = b.max()
        # Softmax weights for the "max"
        exp_b = np.exp(tau * (b - b_max))
        p = exp_b / exp_b.sum()
        
        # Gradient of LSE(tau * max(b)) w.r.t. a
        # d LSE / d a_j = 2 * sum_k p_k * a_{k-j} (where valid)
        # This is a cross-correlation of p with a
        s_fft = 1
        while s_fft < 3 * n:
            s_fft <<= 1
        fp = np.fft.rfft(p, s_fft)
        fa = np.fft.rfft(a, s_fft)
        corr = np.fft.irfft(fp * np.conj(fa), s_fft)
        grad = 2.0 * corr[:n]
        
        # Full gradient of J (with sum normalization s=1):
        # J = 2n * M / s^2, dJ/da_j = 2n*(grad_j * s^2 - M * 2 * s * a_j) / s^4
        # With s=1: dJ/da_j = 2n*(grad_j - 2*M*a_j)
        # Direction: grad_j - 2*b_max*a_j
        direction = grad - 2.0 * b_max * a
        
        v = momentum * v + (1 - momentum) * direction
        a_new = a - lr * v
        a_new = np.maximum(a_new, 1e-8)
        a_new = a_new / a_new.sum()
        
        J_new = compute_J(a_new)
        if J_new < current_J - 1e-15:
            a = a_new
            current_J = J_new
            tau = min(500.0, tau * 1.02)
            lr = 2.0 / (n * tau)
        else:
            lr *= 0.5
            v *= 0.3
            if lr < 1e-18:
                break
        
        if it % 100 == 0:
            tracker.update(a)
    
    tracker.update(a)
    return a

    inits.append(t**2 + 0.1)  # quadratic
    inits.append((1-t)**2 + 0.1)

    inits.append(1.0 - np.abs(2*t - 1.0) + 0.1)
    
    # Normalize all
    result = []
    for x in inits:
        x = np.maximum(np.asarray(x, dtype=np.float64), 0.001)
        x = x / x.sum()
        result.append(x)

            break

        b = fast_conv(a)
        b_max = b.max()
        exp_b = np.exp(tau * (b - b_max))
        p = exp_b / exp_b.sum()

        fp = np.fft.rfft(p, s_fft)
        fa = np.fft.rfft(a, s_fft)
        corr = np.fft.irfft(fp * np.conj(fa), s_fft)
        grad = 2.0 * corr[:n]

        direction = grad - 2.0 * b_max * a
        v = momentum * v + (1.0 - momentum) * direction
        a_new = a - lr * v
        a_new = np.maximum(a_new, 1e-8)
        a_new = a_new / a_new.sum()

        J_new = compute_J(a_new)
        if J_new < current_J - 1e-15:
            a = a_new
            current_J = J_new
            tau = min(800.0, tau * 1.015)
            lr = 2.0 / (n * tau)
        else:
            lr *= 0.5
            v *= 0.3
            if lr < 1e-18:
                break

        if it % 100 == 0:
            tracker.update(a)

    tracker.update(a)
    return a

def peak_directed_descent(a, max_iters, deadline, tracker):
    """Directly attack the peak lag of the autocorrelation."""
    a = np.maximum(a.astype(np.float64), 1e-10)
    n = len(a)
    a = a / a.sum()
    current_J = compute_J(a)
    tracker.update(a)
    eta = 0.5 / n

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        b = fast_conv(a)
        k_peak = int(np.argmax(b))
        b_max = b[k_peak]

        # Gradient of b_{k_peak} w.r.t. a_j
        grad = np.zeros(n)
        js = max(0, k_peak - n + 1)
        je = min(n - 1, k_peak)
        for j in range(js, je + 1):
            grad[j] = a[k_peak - j]

        # Move to reduce peak: decrease a_j proportional to a_{k_peak - j}
        # But also maintain sum: add uniform compensation
        direction = grad - b_max * 2.0 * a
        a_new = a - eta * direction
        a_new = np.maximum(a_new, 1e-10)
        a_new = a_new / a_new.sum()

        J_new = compute_J(a_new)
        if J_new < current_J - 1e-15:
            a = a_new
            current_J = J_new
            eta = min(5.0 / n, eta * 1.01)
        else:
            eta *= 0.7

        if it % 100 == 0:
            tracker.update(a)

    tracker.update(a)
    return a

def phase_escape(a, n_restarts, deadline, tracker):
    """Random phase perturbations in frequency domain to escape local minima."""
    rng = np.random.default_rng(54321)
    n = a.size
    a = a.copy().astype(np.float64)
    a = a / a.sum()
    current_J = compute_J(a)
    tracker.update(a)

    for r in range(n_restarts):
        if time.monotonic() > deadline:
            break

        A = np.fft.fft(a)
        mag = np.abs(A)
        phase = np.angle(A)

        perturbation = rng.normal(0, 0.8, n)
        perturbation[0] = 0.0
        if n % 2 == 0:
            perturbation[n // 2] = 0.0
        for k in range(1, (n + 1) // 2):
            perturbation[n - k] = -perturbation[k]

        new_phase = phase + perturbation
        new_A = mag * np.exp(1j * new_phase)
        a_new = np.fft.ifft(new_A).real
        a_new = np.maximum(a_new, 0.0)

        if a_new.sum() < 1e-10:
            continue
        a_new = a_new / a_new.sum()

        a_new = gs_spectral_equalize(a_new, 50, deadline - 20, tracker, alpha0=0.4)
        J_new = compute_J(a_new)
        if J_new < current_J:
            a = a_new
            current_J = J_new
            tracker.update(a)

    return a

def generate_inits(n, rng, parent=None):
    """Generate diverse initial candidates."""
    inits = []
    t = np.arange(n, dtype=np.float64) / n

    if parent is not None and parent.size > 0:
        t_old = np.arange(len(parent), dtype=np.float64) / len(parent)
        pr = np.interp(t, t_old, parent)
        inits.append(np.maximum(pr, 0.001))
        inits.append(np.maximum(pr[::-1], 0.001))

    # Uniform
    inits.append(np.ones(n))

    # Chirps (always non-negative)
    inits.append(1.0 + np.cos(np.pi * t * t))
    inits.append(1.0 + 0.5 * np.cos(np.pi * 3.0 * t * t))
    inits.append(1.0 + 0.7 * np.cos(np.pi * 1.5 * t * t))
    inits.append(1.0 + 0.4 * np.sin(np.pi * 2.0 * t * t))

    # Random flat spectrum
    for _ in range(4):
        half = n // 2 + 1
        angles = rng.uniform(0, 2 * np.pi, half)
        A = np.zeros(n, dtype=complex)
        A[:half] = np.exp(1j * angles[:half])
        for k in range(1, n - n // 2):
            A[k] = np.conj(A[n - k])
        a = np.fft.ifft(A).real
        inits.append(np.maximum(a, 0.0) + 0.05)

    # Asymmetric (key for breaking below J=2)
    inits.append(np.exp(-2.0 * t) + 0.1)
    inits.append(np.exp(-2.0 * (1.0 - t)) + 0.1)
    inits.append(t * t + 0.1)
    inits.append((1.0 - t) * (1.0 - t) + 0.1)
    inits.append(1.0 / (t + 0.3))
    inits.append(1.0 / (1.0 - t + 0.3))

    # Bumps
    inits.append(np.exp(-((t - 0.3) ** 2) / 0.04) + 0.05)
    inits.append(np.exp(-((t - 0.7) ** 2) / 0.04) + 0.05)
    inits.append(np.exp(-((t - 0.2) ** 2) / 0.02) + 0.3 * np.exp(-((t - 0.8) ** 2) / 0.05) + 0.05)

    # Multi-frequency
    inits.append(1.0 + 0.3 * np.cos(2 * np.pi * 2 * t) + 0.2 * np.cos(2 * np.pi * 5 * t)
                + 0.1 * np.cos(2 * np.pi * 11 * t) + 0.05 * np.cos(2 * np.pi * 17 * t))

    # Power laws
    inits.append((t + 0.05) ** 0.5)
    inits.append((1.0 - t + 0.05) ** 0.5)
    inits.append((t + 0.05) ** 2.0)

    # Random smooth
    for _ in range(3):
        n_coeff = min(8, n // 4 + 1)
        coeffs = rng.normal(0, 1, n_coeff)
        sig = np.zeros(n)
        for i in range(n_coeff):
            sig += coeffs[i] * np.cos(2 * np.pi * i * t + rng.uniform(0, 6.28))
        inits.append(np.abs(sig) + 0.5)

    # Binary-like
    for _ in range(2):
        bits = rng.choice([0.0, 1.0], size=n)
        inits.append(bits + 0.2)

    # Normalize
    result = []
    for x in inits:
        x = np.maximum(np.asarray(x, dtype=np.float64), 0.001)
        x = x / x.sum()
        result.append(x)
    return result

def resample_seq(src, dest_n):
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    return np.maximum(np.interp(t_new, t_old, src), 0.001)

def propose_candidate(seed=42, budget_s=1500, **kwargs):
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + float(budget_s) - 30.0
    tracker = ScoreTracker()

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        if p.ndim == 1 and p.size > 0 and np.all(np.isfinite(p)):
            parent = np.maximum(p, 0.0)
            tracker.update(parent)
            print(f"Parent loaded: n={parent.size}, J={tracker.best_score:.8f}", flush=True)
    except (NameError, AttributeError):
        pass

    # === Phase 1: Multi-length, multi-init optimization ===
    lengths = [256, 384, 512, 640]
    time_per_length = (deadline - time.monotonic() - 400) / len(lengths)

    for n in lengths:
        t_start = time.monotonic()
        t_end = min(deadline - 400, t_start + time_per_length)
        if time.monotonic() >= t_end - 40:
            continue

        print(f"\n{'='*50} Phase1 n={n} {'='*50}", flush=True)

        inits = generate_inits(n, rng, parent)
        best_for_n = None
        best_J_for_n = float('inf')
        n_inits = len(inits)

        for idx, a0 in enumerate(inits):
            if time.monotonic() > t_end - 30:
                break

            # GS spectral equalization
            a = gs_spectral_equalize(a0, 150, t_end - 80, tracker, alpha0=0.3)

            # LSE gradient
            if time.monotonic() < t_end - 50:
                a = lse_gradient(a, 800, t_end - 50, tracker, tau0=40.0)

            # Peak-directed
            if time.monotonic() < t_end - 30:
                a = peak_directed_descent(a, 300, t_end - 30, tracker)

            J = compute_J(a)
            if J < best_J_for_n:
                best_J_for_n = J
                best_for_n = a.copy()

            if (idx + 1) % 5 == 0:
                print(f"    init {idx+1}/{n_inits}: best={best_J_for_n:.8f} t={time.monotonic()-t_start:.1f}s", flush=True)

        if best_for_n is not None:
            tracker.update(best_for_n)
            print(f"  Phase1 n={n}: J={best_J_for_n:.8f} time={time.monotonic()-t_start:.1f}s", flush=True)

    # === Phase 2: Focused deep optimization ===
    if tracker.best is not None and time.monotonic() < deadline - 250:
        n = tracker.best.size
        print(f"\n{'='*50} Phase2 Deep n={n} {'='*50}", flush=True)

        a = tracker.best.copy().astype(np.float64)

        # Multiple GS passes with different alphas
        for alpha_try in [0.15, 0.25, 0.4, 0.6]:
            if time.monotonic() > deadline - 250:
                break
            a_temp = gs_spectral_equalize(a.copy(), 300, deadline - 250, tracker, alpha0=alpha_try)
            if compute_J(a_temp) < compute_J(a):
                a = a_temp

        # LSE with multiple tau
        for tau in [15, 50, 150, 400]:
            if time.monotonic() > deadline - 200:
                break
            a_temp = lse_gradient(a.copy(), 1500, deadline - 200, tracker, tau0=tau)
            if compute_J(a_temp) < compute_J(a):
                a = a_temp

        # Peak-directed
        if time.monotonic() < deadline - 150:
            a_temp = peak_directed_descent(a.copy(), 1000, deadline - 150, tracker)
            if compute_J(a_temp) < compute_J(a):
                a = a_temp

        # Phase escape
        if time.monotonic() < deadline - 100:
            a = phase_escape(a, 40, deadline - 100, tracker)

        # Final LSE polish
        if time.monotonic() < deadline - 50:
            a = lse_gradient(a, 3000, deadline - 50, tracker, tau0=200.0)

        # GS + peak-directed alternation
        if time.monotonic() < deadline - 20:
            for _ in range(3):
                if time.monotonic() > deadline - 20:
                    break
                a = gs_spectral_equalize(a, 100, deadline - 20, tracker, alpha0=0.2)
                a = peak_directed_descent(a, 200, deadline - 20, tracker)

        tracker.report("Phase2")

    # === Phase 3: Multi-start perturbation ===
    if tracker.best is not None and time.monotonic() < deadline - 60:
        n = tracker.best.size
        print(f"\n{'='*50} Phase3 MultiStart {'='*50}", flush=True)
        rng3 = np.random.default_rng(4242)
        a = tracker.best.copy()

        for restart in range(30):
            if time.monotonic() > deadline - 30:
                break

            scale = 0.03 * (1.0 + restart * 0.1)
            a_pert = a + rng3.normal(0, scale, n)
            a_pert = np.maximum(a_pert, 0.001)
            a_pert = a_pert / a_pert.sum()

            a_pert = gs_spectral_equalize(a_pert, 60, deadline - 30, tracker, alpha0=0.3)
            a_pert = lse_gradient(a_pert, 500, deadline - 30, tracker, tau0=60.0)

            if compute_J(a_pert) < compute_J(a):
                a = a_pert.copy()
            tracker.update(a)

        tracker.report("Phase3")

    # === Phase 4: Coordinate descent polish ===
    if tracker.best is not None and time.monotonic() < deadline - 15:
        print(f"\n{'='*50} Phase4 CoordDesc {'='*50}", flush=True)
        a = tracker.best.copy().astype(np.float64)
        n = a.size
        rng4 = np.random.default_rng(888)
        current_J = compute_J(a)

        for step in range(5000):
            if time.monotonic() > deadline - 5:
                break
            j = int(rng4.integers(0, n))
            best_trial = None
            best_trial_J = current_J
            for scale in [0.4, 0.6, 0.8, 1.25, 1.5, 2.0]:
                a_trial = a.copy()
                a_trial[j] *= scale
                a_trial = np.maximum(a_trial, 1e-8)
                ss = a_trial.sum()
                if ss < 1e-10:
                    continue
                a_trial /= ss
                J_trial = compute_J(a_trial)
                if J_trial < best_trial_J:
                    best_trial_J = J_trial
                    best_trial = a_trial.copy()
            if best_trial is not None:
                a = best_trial
                current_J = best_trial_J
            if step % 500 == 0:
                tracker.update(a)

        # Final micro-polish with LSE
        if time.monotonic() < deadline - 3:
            a = lse_gradient(a, 500, deadline - 2, tracker, tau0=500.0)
        tracker.update(a)
        tracker.report("Phase4")

    # Fallback
    if tracker.best is None:
        tracker.best = np.ones(256)
        tracker.update(tracker.best)

    print(f"\n{'='*50} FINAL {'='*50}", flush=True)
    print(f"J={tracker.best_score:.8f} n={tracker.best.size} time={time.monotonic()-tracker.t_start:.1f}s", flush=True)
    return [float(x) for x in tracker.best]
