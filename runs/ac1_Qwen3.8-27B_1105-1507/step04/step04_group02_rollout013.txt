def solve_lp(a, M, time_limit=5.0):
    """Solve LP: max sum(g) s.t. conv(a,g) <= M, g >= 0."""
    n = len(a)
    # Build constraint matrix: (2n-1) x n, where row k, col i = a[k-i] if valid
    rows = []
    cols = []
    vals = []
    for i in range(n):
        for k in range(max(0, i - n + 1), min(i + n, 2*n - 1)):
            j = k - i
            if 0 <= j < n:
                rows.append(k)
                cols.append(i)
                vals.append(a[j])
    
    A = coo_matrix((vals, (rows, cols)), shape=(2*n-1, n)).tocsr()

    try:

                     options={'presolve': True, 'time_limit': time_limit})
        if res.status == 0:
            return np.maximum(res.x, 0)
    except Exception:
        pass
    return None

def multiplicative_wu(a, max_iters=5000, time_budget=20.0, eta_init=0.5):

        for k in active:

            hi = min(k, n - 1)

                j = k - i
                if j == i:
                    grad[i] += a[j]
                else:
                    grad[i] += 2.0 * a[j]

        a_new = a * np.exp(-eta * grad_cent)
        a_new = np.maximum(a_new, 1e-15)

"""
Optimized step-function coefficient search minimizing 2*n*max(conv(a,a))/(sum(a))^2.
Uses: exact full-active-set subgradient with backtracking line search, aggressive
alternating LP (HiGHS), multiplicative Wu updates, Fejér-kernel and multi-profile
initializations, adaptive multi-n exploration, and iterative perturbation-escape
refinement. Key improvements over prior: full active set (not top-K), backtracking
line search, more aggressive LP usage with multiple M values, Fejér kernel init,
and better time allocation between phases.
"""

import numpy as np
from scipy.optimize import linprog
from scipy.sparse import coo_matrix
from scipy.interpolate import interp1d
import time


def compute_conv(a):
    """Compute linear convolution a*a via FFT."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    conv = np.fft.irfft(fa * fa, size)
    return conv[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    S = float(np.sum(a))
    if S < 1e-10:
        return np.inf
    conv = compute_conv(a)
    M = float(np.max(conv))
    return 2.0 * n * M / (S * S)


def normalize(a):
    """Normalize to sum=1, ensure non-negative."""
    a = np.maximum(a, 0.0).astype(np.float64)
    S = float(np.sum(a))
    if S < 1e-10:
        return np.ones(len(a)) / len(a)
    return a / S


def subgradient_full_active(a, eps_frac=1e-3):
    """Compute exact subgradient using full active set (positions near max)."""
    n = len(a)
    conv = compute_conv(a)
    M = float(np.max(conv))
    if M <= 0:
        return np.zeros(n), 0.0, conv, np.array([0])

    threshold = M * (1.0 - eps_frac)
    active = np.where(conv >= threshold)[0]

    if len(active) == 0:
        active = np.array([int(np.argmax(conv))])

    grad = np.zeros(n)
    for k in active:
        lo = max(0, k - n + 1)
        hi = min(k, n - 1)
        for i in range(lo, hi + 1):
            j = k - i
            if j == i:
                grad[i] += a[j]
            else:
                grad[i] += 2.0 * a[j]

    grad /= float(len(active))
    return grad, M, conv, active


def projected_subgradient(a, max_iters=10000, time_budget=30.0, eps_frac=1e-3):
    """Projected subgradient descent with momentum and backtracking."""
    n = len(a)
    a = normalize(a.copy())
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    v = np.zeros(n)
    momentum = 0.75
    base_step = 0.08
    stall = 0

    for it in range(max_iters):
        if time.monotonic() - start > time_budget:
            break

        grad, M, conv, active = subgradient_full_active(a, eps_frac)
        grad_cent = grad - np.mean(grad)

        v = momentum * v + (1.0 - momentum) * grad_cent
        v_norm = np.linalg.norm(v)
        if v_norm < 1e-15:
            v_dir = -grad_cent
            v_norm = np.linalg.norm(v_dir)
        else:
            v_dir = -v

        if v_norm < 1e-15:
            break

        # Backtracking line search
        step = base_step
        improved = False
        for _ in range(40):
            if step < 1e-12:
                break
            a_try = a + step * v_dir / v_norm
            a_try = np.maximum(a_try, 0.0)
            S_try = float(np.sum(a_try))
            if S_try < 1e-10:
                step *= 0.5
                continue
            a_try = a_try / S_try
            obj_try = compute_obj(a_try)

                best_obj = obj_try

def build_toeplitz(a):
    """Build sparse Toeplitz matrix A where (A*g)[k] = conv(a,g)[k]."""
    n = len(a)
    rows = []
    cols = []
    vals = []
    for i in range(n):
        lo = i
        hi = min(i + n, 2 * n - 1)
        for k in range(lo, hi):
            j = k - i
            if 0 <= j < n:
                rows.append(k)
                cols.append(i)
                vals.append(a[j])
    A = coo_matrix((vals, (rows, cols)), shape=(2 * n - 1, n)).tocsr()
    return A


def solve_lp(a, M, time_limit=5.0):
    """Solve LP: max sum(g) s.t. conv(a,g) <= M, g >= 0."""
    n = len(a)
    if n > 800:
        return None
    A = build_toeplitz(a)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': time_limit})
        if res.status == 0:
            return np.maximum(res.x, 0.0)
    except Exception:
        pass
    return None

    while time.monotonic() - start < time_budget - 5:
        rnd += 1
        remaining = time_budget - (time.monotonic() - start)
        if remaining < 12:
            break

        # Subgradient
        t_sg = min(remaining * 0.35, 25.0)
        a2, obj2 = projected_subgradient(best_a, max_iters=int(t_sg * 50), time_budget=t_sg)
        if obj2 < best_obj:
            best_obj = obj2
            best_a = a2
            if verbose:
                print(f"    R{rnd} SG: {best_obj:.6f}")

        # Multiplicative
        remaining = time_budget - (time.monotonic() - start)
        if remaining > 15:
            t_mu = min(remaining * 0.2, 12.0)
            a3, obj3 = multiplicative_wu(best_a, max_iters=int(t_mu * 40), time_budget=t_mu)
            if obj3 < best_obj:
                best_obj = obj3
                best_a = a3
                if verbose:
                    print(f"    R{rnd} MU: {best_obj:.6f}")

        # LP
        remaining = time_budget - (time.monotonic() - start)
        if remaining > 20 and n <= 600:
            t_lp = min(remaining * 0.3, 35.0)
            a4, obj4 = alternating_lp(best_a, max_iters=25, time_budget=t_lp)
            if obj4 < best_obj:
                best_obj = obj4
                best_a = a4
                if verbose:
                    print(f"    R{rnd} LP: {best_obj:.6f}")

    return best_a, best_obj


def fejér_kernel(n):
    """Discrete Fejér kernel (triangular window in freq domain)."""
    k = np.arange(n)
    # Fejér kernel: sum_{j=0}^{n-1} (1 - j/n) * cos(2*pi*j*k/n)
    j = np.arange(n)
    F = np.zeros(n)
    for jj in range(n):
        F += (1.0 - jj / n) * np.cos(2.0 * np.pi * jj * k / n)
    F = np.maximum(F, 1e-10)
    return F / np.sum(F)


def raised_cosine_init(n, alpha=1.0):
    """Raised cosine shape."""
    k = np.arange(n)
    x = 2.0 * k / (n - 1) - 1.0
    a = 0.5 * (1.0 + np.cos(np.pi * x)) ** alpha
    a = np.maximum(a, 1e-10)
    return a / np.sum(a)


def gaussian_init(n, sigma_frac=0.35):
    """Gaussian bump initialization."""
    k = np.arange(n)
    center = n / 2.0
    sigma = sigma_frac * n / 2.0
    a = np.exp(-0.5 * ((k - center) / sigma) ** 2)
    return a / np.sum(a)


def cosine_mix_init(n, seed=0, n_modes=10):
    """Random cosine series initialization."""
    rng = np.random.default_rng(seed)
    k = np.arange(n)
    a = np.ones(n)
    for m in range(1, n_modes + 1):
        c = rng.uniform(-0.5, 0.5) / (m ** 0.7)
        phase = rng.uniform(0, 2 * np.pi)
        a += c * np.cos(2 * np.pi * m * k / n + phase)
    a = np.maximum(a, 1e-10)
    return a / np.sum(a)


def triangle_init(n):
    """Triangular (Bartlett window) initialization."""
    k = np.arange(n)
    a = np.maximum(1.0 - np.abs(2.0 * k / (n - 1) - 1.0), 0.0)
    a = np.maximum(a, 1e-10)
    return a / np.sum(a)


def all_initializations(n, seed=0):
    """Generate all initialization candidates for a given n."""
    inits = {
        'uniform': np.ones(n) / n,
        'fejér': fejér_kernel(n),
        'rcos1': raised_cosine_init(n, 1.0),
        'rcos2': raised_cosine_init(n, 2.0),
        'rcos3': raised_cosine_init(n, 3.0),
        'gauss03': gaussian_init(n, 0.3),
        'gauss05': gaussian_init(n, 0.5),
        'triangle': triangle_init(n),
        'cosmix1': cosine_mix_init(n, seed=seed + 1, n_modes=8),
        'cosmix2': cosine_mix_init(n, seed=seed + 2, n_modes=12),
    }
    return inits


def propose_candidate(seed=42):
    """
    Main search function. Multi-phase optimization:
    Phase 1: Broad multi-n, multi-profile exploration with quick refinement.
    Phase 2: Focused refinement of top candidates.
    Phase 3: Intensive iterative refinement with perturbation escapes.
    Phase 4: Final polish with extended runs.
    """
    start_time = time.monotonic()
    total_budget = 1500.0
    deadline = start_time + total_budget - 35.0

    best_seq = None
    best_score = np.inf

    # Load parent sequence
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64)
        parent = np.clip(parent, 0.0, 1000.0)
        if parent.size > 0 and np.isfinite(parent).all():
            best_seq = parent.copy()
            best_score = compute_obj(best_seq)
            print(f"Parent loaded (n={parent.size}): obj={best_score:.6f}")
    except (NameError, Exception) as e:
        print(f"Could not load parent: {e}")

    if best_seq is None:
        best_seq = np.ones(400) / 400.0
        best_score = compute_obj(best_seq)
        print(f"Default init (n=400): obj={best_score:.6f}")

    n_parent = len(best_seq)

    # === Phase 1: Multi-n, multi-profile exploration ===
    print("\n=== Phase 1: Multi-n exploration ===")
    n_values = sorted(set([
        max(100, n_parent - 200),
        max(100, n_parent - 100),
        n_parent,
        n_parent + 50,
        n_parent + 100,
        n_parent + 200,
        min(1200, n_parent + 300),
    ]))
    n_values = [n for n in n_values if 100 <= n <= 1500]

    candidates = []

    for n in n_values:
        if time.monotonic() > deadline:
            break
        remaining = deadline - time.monotonic()
        if remaining < 40:
            break

        # Add interpolated parent as an init
        inits = all_initializations(n, seed=seed + n)
        n_base = len(best_seq)
        if n_base != n and n_base > 10:
            x_old = np.linspace(0, 1, n_base)
            x_new = np.linspace(0, 1, n)
            try:
                f_interp = interp1d(x_old, best_seq, kind='cubic', fill_value='extrapolate')
                a_interp = np.maximum(f_interp(x_new), 1e-10)
                inits['interp_parent'] = a_interp / np.sum(a_interp)
            except Exception:
                pass

        for name, a_init in inits.items():
            if time.monotonic() > deadline:
                break
            a_init = normalize(a_init)
            obj_init = compute_obj(a_init)

            # Quick subgradient refinement
            a_quick, obj_quick = projected_subgradient(
                a_init, max_iters=3000, time_budget=4.0)
            best_local = a_quick if obj_quick < obj_init else a_init
            best_local_obj = min(obj_quick, obj_init)

            candidates.append((best_local, best_local_obj, n, name))

            if best_local_obj < best_score:
                best_score = best_local_obj
                best_seq = best_local.copy()
                print(f"  n={n} {name}: {best_local_obj:.6f} ** NEW BEST **")

    # === Phase 2: Focused refinement of top candidates ===
    print("\n=== Phase 2: Focused refinement ===")
    candidates.sort(key=lambda x: x[1])

    n_refine = min(8, len(candidates))
    for idx in range(n_refine):
        if time.monotonic() > deadline:
            break
        remaining = deadline - time.monotonic()
        if remaining < 50:
            break

        a_cand, obj_cand, n_cand, name = candidates[idx]
        if obj_cand > best_score + 0.03:
            break

        alloc = min(remaining * 0.18, 50.0)
        print(f"  Refine {idx} (n={n_cand}, {name}, obj={obj_cand:.6f}), alloc={alloc:.0f}s")

        a_ref, obj_ref = refine_sequence(a_cand, time_budget=alloc, verbose=True)
        if obj_ref < best_score:
            best_score = obj_ref
            best_seq = a_ref.copy()
            print(f"    ** NEW BEST: n={n_cand}, obj={best_score:.6f} **")

    # === Phase 3: Intensive iterative refinement ===
    print("\n=== Phase 3: Intensive refinement ===")
    a_cur = best_seq.copy()
    n_cur = len(a_cur)

    for rnd in range(25):
        if time.monotonic() > deadline - 40:
            break
        remaining = deadline - time.monotonic()
        if remaining < 30:
            break

        alloc = min(remaining * 0.12, 40.0)
        a_new, obj_new = refine_sequence(a_cur, time_budget=alloc, verbose=False)

        if obj_new < best_score:
            best_score = obj_new
            best_seq = a_new.copy()
            a_cur = a_new.copy()
            print(f"  R{rnd}: {best_score:.6f} ** NEW BEST **")
        else:
            # Perturbation escape
            rng = np.random.default_rng(rnd * 48271 + 999)
            for scale in [0.05, 0.1, 0.15, 0.25]:
                if time.monotonic() > deadline - 25:
                    break
                a_p = a_cur * np.exp(rng.normal(0, scale, n_cur))
                a_p = normalize(a_p)
                a_p, obj_p = projected_subgradient(
                    a_p, max_iters=4000, time_budget=min(12.0, remaining * 0.08))
                if obj_p < best_score:
                    best_score = obj_p
                    best_seq = a_p.copy()
                    a_cur = a_p.copy()
                    print(f"  R{rnd} pert({scale}): {best_score:.6f} ** NEW BEST **")
                    break

    # === Phase 4: Final polish ===
    print("\n=== Phase 4: Final polish ===")
    remaining = deadline - time.monotonic()
    if remaining > 25:
        n_fin = len(best_seq)

        # Extended subgradient
        t_sg = min(remaining * 0.35, 60.0)
        a_sg, obj_sg = projected_subgradient(best_seq, max_iters=int(t_sg * 50), time_budget=t_sg)
        if obj_sg < best_score:
            best_score = obj_sg
            best_seq = a_sg
            print(f"  Final SG: {best_score:.6f} ** NEW BEST **")

        # Extended LP
        remaining = deadline - time.monotonic()
        if remaining > 25 and n_fin <= 550:
            t_lp = min(remaining * 0.5, 50.0)
            a_lp, obj_lp = alternating_lp(best_seq, max_iters=40, time_budget=t_lp)
            if obj_lp < best_score:
                best_score = obj_lp
                best_seq = a_lp
                print(f"  Final LP: {best_score:.6f} ** NEW BEST **")

        # Final multiplicative
        remaining = deadline - time.monotonic()
        if remaining > 15:
            t_mu = min(remaining * 0.4, 25.0)
            a_mu, obj_mu = multiplicative_wu(best_seq, max_iters=int(t_mu * 50), time_budget=t_mu)
            if obj_mu < best_score:
                best_score = obj_mu
                best_seq = a_mu
                print(f"  Final MU: {best_score:.6f} ** NEW BEST **")

    # Validate and return
    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    val = evaluate_sequence(result)
    if val < np.inf and val < best_score:
        best_score = val
    elif val == np.inf:
        print("WARNING: final evaluation returned inf, using fallback")
        result = [1.0] * 300
        best_score = 2.0

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_score:.6f}, time={elapsed:.1f}s ===")

    return result
