def build_toeplitz(a):
    """Build Toeplitz matrix A where (A@g)[k] = sum_j a[j]*g[k-j] for valid j."""
    n = len(a)
    A = np.zeros((2 * n - 1, n), dtype=np.float64)
    for j in range(n):
        k_start = j
        k_end = min(j + n, 2 * n - 1)
        A[k_start:k_end, j] = a[:k_end - k_start]
    return A

    A_ub[:, :n] = A
    A_ub[:, n] = -1.0
    b_ub = np.zeros(2 * n - 1)
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])
    bounds = [(0.0, None)] * n + [(0.0, None)]
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': time_limit})
        if res.status == 0:
            g = np.maximum(res.x[:n], 0)
            s = np.sum(g)
            if s > 1e-10:
                g = g / s
                return g, float(res.x[n])
    except Exception:
        pass
    return None, None


def solve_lp_maxsum(a, M, time_limit=5):
    """Solve LP: max sum(g) s.t. (a*g)[k] <= M, g >= 0."""
    n = len(a)
    A = build_toeplitz(a)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': time_limit})
        if res.status == 0:
            g = np.maximum(res.x, 0)
            return g, float(np.sum(g))
    except Exception:
        pass
    return None, None


def alternating_lp(a, time_budget=30.0, verbose=False):
    """Alternating LP optimization with line search."""
    n = len(a)
    a = a.copy().astype(np.float64)
    S = float(np.sum(a))
    if S < 1e-10:
        return a, np.inf
    a = a * (n / S)
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    stall = 0

    for it in range(200):
        if time.monotonic() - start > time_budget - 3:
            break

        g, M_opt = solve_lp_minpeak(a, time_limit=4)
        if g is None:
            break

        best_t = 0.0
        best_obj_ls = best_obj
        for t in [0.1, 0.2, 0.3, 0.5, 0.7, 1.0]:
            a_new = (1.0 - t) * a + t * g
            a_new = np.maximum(a_new, 0)
            s_new = np.sum(a_new)
            if s_new < 1e-10:
                continue
            a_new = a_new * (n / s_new)
            obj_new = compute_obj(a_new)
            if obj_new < best_obj_ls:
                best_obj_ls = obj_new
                best_t = t

        if best_t > 0 and best_obj_ls < best_obj - 1e-12:
            a_new = (1.0 - best_t) * a + best_t * g
            a_new = np.maximum(a_new, 0)
            s_new = np.sum(a_new)
            if s_new > 1e-10:
                a_new = a_new * (n / s_new)
            best_obj = best_obj_ls
            best_a = a_new.copy()
            a = a_new
            stall = 0
        else:
            stall += 1
            if stall > 15:
                break
            a = 0.9 * a + 0.1 * g
            a = np.maximum(a, 0)
            s = np.sum(a)
            if s > 1e-10:
                a = a * (n / s)

    return best_a, best_obj


def smooth_gradient_optimize(a, time_budget=30.0, verbose=False):
    """Smooth gradient descent using soft-max approximation of the peak."""
    n = len(a)
    a = a.copy().astype(np.float64)
    S = float(np.sum(a))
    if S < 1e-10:
        return a, np.inf
    a = a * (n / S)
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    beta = 5.0  # soft-max parameter
    step = 0.05
    stall = 0

    max_iters = max(100, int(time_budget * 30))
    for it in range(max_iters):
        if time.monotonic() - start > time_budget:
            break

        conv = compute_conv(a)
        max_conv = np.max(conv)
        # Soft-max weights
        shifted = beta * (conv - max_conv)
        shifted = np.clip(shifted, -200, 200)
        weights = np.exp(shifted)
        w_sum = np.sum(weights)
        if w_sum < 1e-300:
            break
        weights /= w_sum

        # Gradient of max(conv) w.r.t. a[i]:
        # d(conv[k])/d(a[i]) = 2*a[k-i] for k in [i, i+n-1], k != 2i
        # d(conv[2i])/d(a[i]) = 2*a[i]
        # grad[i] = sum_k weights[k] * d(conv[k])/d(a[i])
        grad = np.zeros(n)
        for k in range(2 * n - 1):
            if weights[k] < 1e-10:
                continue
            k_min_contrib = max(0, k - n + 1)
            k_max_contrib = min(k, n - 1)
            for i in range(k_min_contrib, k_max_contrib + 1):
                j = k - i
                if i == j:
                    grad[i] += weights[k] * 2.0 * a[i]
                else:
                    grad[i] += weights[k] * 2.0 * a[j]

        # Normalize gradient
        grad_mean = np.mean(grad)
        grad_centered = grad - grad_mean

        # Multiplicative update
        a_new = a * np.exp(-step * grad_centered)
        a_new = np.maximum(a_new, 1e-15)
        s_new = np.sum(a_new)
        if s_new > 1e-10:
            a_new = a_new * (n / s_new)
        a = a_new

        obj = compute_obj(a)
        if obj < best_obj - 1e-12:
            best_obj = obj
            best_a = a.copy()
            stall = 0
        else:
            stall += 1
            if stall > 50:
                step *= 0.7
                beta *= 1.3
                if step < 0.005:
                    break
                stall = 0
                # Add small noise
                rng = np.random.default_rng(it)
                a = a * np.exp(rng.normal(0, 0.02, n))
                a = np.maximum(a, 1e-10)
                a = a * (n / np.sum(a))

        if it > 100:
            step *= 0.998
            beta *= 1.005

    return best_a, best_obj


def random_init(n, seed):
    """Generate diverse random initializations."""
    rng = np.random.default_rng(seed)
    k = np.arange(n)
    choice = rng.integers(0, 8)

    if choice == 0:
        f = rng.uniform(0.5, 4.0)
        phase = rng.uniform(0, 2 * np.pi)
        a = 1.0 + 0.5 * np.cos(2 * np.pi * f * k / n + phase)
    elif choice == 1:
        f1 = rng.uniform(0.3, 2.0)
        f2 = rng.uniform(3.0, 8.0)
        a = 1.0 + 0.3 * np.cos(2 * np.pi * f1 * k / n) + 0.2 * np.cos(2 * np.pi * f2 * k / n)
    elif choice == 2:
        a = 1.0 + 0.5 * np.sin(np.pi * k / n) ** 2
    elif choice == 3:
        a = 1.0 + 0.4 * np.cos(2 * np.pi * rng.uniform(1, 5) * k / n)
        a += 0.3 * np.cos(2 * np.pi * rng.uniform(5, 15) * k / n + rng.uniform(0, 6.28))
    elif choice == 4:
        # Smooth random
        a = rng.random(max(5, n // 15))
        a = np.interp(k, np.arange(len(a)), a)
        a = 1.0 + 0.5 * (a - 0.5)
    elif choice == 5:
        # Two bumps
        c1 = n * rng.uniform(0.2, 0.4)
        c2 = n * rng.uniform(0.6, 0.8)
        a = 1.0 + 0.4 * np.exp(-0.5 * ((k - c1) / (n * 0.1)) ** 2) + 0.4 * np.exp(-0.5 * ((k - c2) / (n * 0.1)) ** 2)
    elif choice == 6:
        # sawtooth-like
        a = 1.0 + 0.3 * np.abs(np.sin(2 * np.pi * rng.uniform(1, 3) * k / n))
    else:
        a = 1.0 + 0.4 * np.sin(2 * np.pi * rng.uniform(0.5, 2) * k / n + rng.uniform(0, 6.28))
        a += 0.2 * np.cos(2 * np.pi * rng.uniform(2, 6) * k / n)

    a = np.maximum(a, 0.01)
    a = a * (n / np.sum(a))
    return a


def resample_sequence(a, new_n):
    """Resample a sequence to a new length using interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    old_k = np.arange(n, dtype=float)
    new_k = np.linspace(0, n - 1, new_n)
    a_new = np.interp(new_k, old_k, a)
    a_new = np.maximum(a_new, 0.01)
    s = np.sum(a_new)
    if s > 1e-10:
        a_new = a_new * (new_n / s)
    return a_new
