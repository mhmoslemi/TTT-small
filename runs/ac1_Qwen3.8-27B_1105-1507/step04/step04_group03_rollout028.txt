import numpy as np
from scipy.optimize import linprog, minimize
from scipy.sparse import csr_matrix
import time

def fast_conv(a, size=None):
    """Full self-convolution via FFT."""
    n = len(a)
    if size is None:
        size = 1
        while size < 2 * n - 1:
            size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]

def compute_J(a):
    """Compute J(a) = 2*n*max(conv(a,a))/(sum(a))^2."""
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))

def mwu_update(a, eta, tol=0.05):
    """Single MWU iteration."""
    n = len(a)
    r = fast_conv(a)
    M = r.max()
    if M < 1e-15:
        return a
    active = r >= M * (1 - tol)
    # Compute pressure
    P = np.zeros(n)
    active_idx = np.where(active)[0]
    for k in active_idx:
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        P[lo:hi+1] += a[k-hi:k-lo+1][::-1]
    P_max = P.max()
    if P_max < 1e-15:
        P[:] = 1.0
        P_max = 1.0
    a_new = a * np.exp(-eta * P / P_max)
    a_new = np.maximum(a_new, 1e-15)
    a_new = a_new / a_new.sum()
    return a_new

def mwu_descent(a, max_iters, eta0, deadline):
    """Multiplicative weights descent."""
    a = a.copy()
    best_J = compute_J(a)
    best_a = a.copy()
    no_improve = 0
    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        eta = eta0 * (0.99 ** (it // 50))
        a_new = mwu_update(a, eta)
        J_new = compute_J(a_new)
        if J_new < best_J:
            best_J = J_new
            best_a = a_new.copy()
            no_improve = 0
        else:
            no_improve += 1
        a = a_new
        if no_improve > 100:
            # Restart with smaller eta
            eta0 *= 0.5
            no_improve = 0
    return best_a, best_J
