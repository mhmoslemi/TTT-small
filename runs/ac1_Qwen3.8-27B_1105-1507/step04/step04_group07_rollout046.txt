    try:
        if hasattr(r, 'ub') and r.ub is not None:
            dual = np.asarray(r.ub, dtype=np.float64)
        elif hasattr(r, 'ineqlin') and r.ineqlin is not None:
            dual = np.asarray(r.ineqlin, dtype=np.float64)
    except Exception:
        pass
    return g, dual

def spectral_init(n, rng, n_iter=300):
    """Iterative projection: flat spectrum + non-negativity."""
    half = n // 2 + 1
    phases = rng.uniform(0, 2 * np.pi, half)
    a = np.ones(n) / n
    for it in range(n_iter):
        # Compute current spectrum
        F = np.fft.rfft(a)
        mags = np.abs(F[:half])
        # Target: flat magnitude
        target_mag = np.sum(mags) / half
        # Blended magnitude (partial step toward flat)
        alpha = 0.3 + 0.7 * (it / n_iter)
        new_mag = (1 - alpha) * mags + alpha * target_mag
        # Reconstruct with current phases
        A = new_mag * np.exp(1j * np.angle(F[:half]))
        A_full = np.zeros(n, dtype=complex)
        A_full[:half] = A
        for k in range(1, half):
            if n - k >= 0 and n - k < n:
                A_full[n - k] = np.conj(A[k])
        a = np.fft.ifft(A_full).real
        a = np.maximum(a, 1e-10)
        a /= a.sum()
    return a

def cosine_init(n, rng):
    """Initialize with multi-frequency cosine structure."""
    t = np.linspace(0, 1, n, endpoint=False)
    n_freq = rng.integers(3, 9)
    freqs = rng.uniform(0.5, 15.0, n_freq)
    amplitudes = rng.uniform(0.1, 0.5, n_freq)
    phases = rng.uniform(0, 2 * np.pi, n_freq)
    sig = np.ones(n) * 0.5
    for f, a_, p in zip(freqs, amplitudes, phases):
        sig += a_ * np.cos(2 * np.pi * f * t + p)
    sig = np.maximum(sig, 0.01)
    sig /= sig.sum()
    return sig

def random_init(n, rng):
    """Simple random non-negative initialization."""
    a = rng.uniform(0.05, 1.0, n)
    a /= a.sum()
    return a

def bimodal_init(n, rng):
    """Two-Gaussian mixture initialization."""
    t = np.linspace(0, 1, n)
    c1 = rng.uniform(0.2, 0.4)
    c2 = 1.0 - c1
    w1 = rng.uniform(0.05, 0.15)
    w2 = rng.uniform(0.05, 0.15)
    sig = np.exp(-0.5 * ((t - c1) / w1) ** 2) + np.exp(-0.5 * ((t - c2) / w2) ** 2) + 0.05
    sig /= sig.sum()
    return sig

def resample_seq(src, dest_n):
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)

def generate_inits(n, rng, parent):
    """Generate diverse initializations."""
    inits = []
    if parent is not None and parent.size > 0:
        inits.append(resample_seq(parent, n))
        inits.append(resample_seq(parent[::-1], n))
    inits.append(np.ones(n) / n)
    inits.append(random_init(n, rng))
    inits.append(random_init(n, rng))
    inits.append(spectral_init(n, rng, n_iter=100))
    inits.append(spectral_init(n, rng, n_iter=100))
    inits.append(cosine_init(n, rng))
    inits.append(cosine_init(n, rng))
    inits.append(bimodal_init(n, rng))
    inits.append(bimodal_init(n, rng))
    inits.append(bimodal_init(n, rng))
    # U-shaped
    t = np.linspace(0, 1, n)
    inits.append((1.0 + 0.5 * (2 * t - 1) ** 2) / (1.0 + 0.5 * (2 * np.linspace(0, 1, n) - 1) ** 2).sum())
    # W-shaped
    inits.append((1.0 + 0.4 * np.cos(4 * np.pi * t)) / (n * (1.0 + 0.4 * np.cos(4 * np.pi * np.linspace(0,1,n)))) )
    inits[-1] = np.maximum(inits[-1], 0.01)
    inits[-1] /= inits[-1].sum()
    for x in inits:
        x = np.maximum(x, 1e-6).astype(np.float64)
        x /= x.sum()
    return inits

class Tracker:
    def __init__(self):
        self.best = None
        self.best_score = float('inf')
        self.t0 = time.monotonic()

    def update(self, a):
        a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
        if a.size == 0 or a.sum() < 0.01:
            return
        score = float(evaluate_sequence(a.tolist()))
        if np.isfinite(score) and score < self.best_score:
            self.best = a.copy()
            self.best_score = score
            el = time.monotonic() - self.t0
            print(f"  * NEW BEST: J={score:.8f} n={a.size} t={el:.1f}s", flush=True)

    def report(self, label=""):
        if self.best is not None:
            print(f"  {label}: J={self.best_score:.8f} n={self.best.size} t={time.monotonic()-self.t0:.1f}s", flush=True)

def lp_optimize(a, max_iters, deadline, tracker, eps_start=0.05, eps_end=0.001, eps_decay=0.97):
    """Core LP alternating optimization with adaptive tightening."""
    a = np.maximum(a.astype(np.float64), 1e-8)
    a /= a.sum()
    n = len(a)
    current_J = compute_J(a)
    tracker.update(a)
    eps = eps_start

    for it in range(max_iters):
        if time.monotonic() > deadline - 5:
            break
        b = fast_conv(a)
        M = float(b.max())
        if M < 1e-14:
            break

        # Try tightened LP first
        M_tight = M * (1.0 - eps)
        g, dual = solve_lp(a, M_tight, n, tlimit=6)
        if g is None or g.sum() < 1e-10:
            g, dual = solve_lp(a, M, n, tlimit=6)
        if g is None or g.sum() < 1e-10:
            break
        g = np.maximum(g, 0.0)
        g /= g.sum()

        # Line search for mixing parameter
        best_a = None
        best_J = current_J
        progress = it / max(1, max_iters)
        base_t = max(0.1, 0.8 * (1.0 - progress) ** 0.5)
        t_candidates = sorted(set([base_t, base_t * 1.5, base_t * 0.5, 0.6, 0.3, 0.9, 0.15]))
        for t in t_candidates:
            if t > 1.0 or t < 0:
                continue
            a_trial = (1.0 - t) * a + t * g
            a_trial = np.maximum(a_trial, 1e-8)
            a_trial /= a_trial.sum()
            J_trial = compute_J(a_trial)
            if J_trial < best_J - 1e-15:
                best_J = J_trial
                best_a = a_trial

        if best_a is None:
            # Try without tightening
            eps = max(eps * 0.8, eps_end * 0.5)
            if eps < eps_end * 0.3:
                break
            continue

        a = best_a
        current_J = best_J
        eps = max(eps * eps_decay, eps_end)

        if it % 5 == 0:
            tracker.update(a)

    tracker.update(a)
    return a

def dual_weighted_subgradient(a, max_iters, deadline, tracker):
    """Subgradient descent using LP dual for weighted gradient."""
    a = np.maximum(a.astype(np.float64), 1e-8)
    a /= a.sum()
    n = len(a)
    current_J = compute_J(a)
    tracker.update(a)
    lr = 2.0 / n
    momentum = 0.7
    v = np.zeros(n)
    rng = np.random.default_rng(123)

    for it in range(max_iters):
        if time.monotonic() > deadline - 5:
            break
        b = fast_conv(a)
        M = float(b.max())
        if M < 1e-14:
            break

        # Find active set (within tolerance of max)
        tol = max(1e-8 * M, 1e-12)
        active = np.where(b >= M - tol)[0]
        n_active = len(active)

        # Compute subgradient: average of a[k-j] over active k
        grad = np.zeros(n)
        for k in active:
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = k - np.arange(js, je + 1)
            valid = (idx >= 0) & (idx < n)
            grad[js:je+1][valid] += a[idx[valid]]
        grad /= n_active

        # Full gradient of J (with sum_a=1 normalization):
        # dJ/da_j proportional to 2*n*grad_j - 4*n*M*a_j (since d(sum_a^2)/da_j = 2*sum_a*... but sum_a=1)
        # Actually J = 2n*M where M = max(conv(a,a)), and dM/da_j = average subgradient
        # With normalization sum(a)=1: use Lagrangian, gradient = 2n*grad - 2n*M*2*a (from d(M/s^2))
        # Simplified: direction to decrease = grad - 2*M*a (normalized)
        direction = grad - 2.0 * M * a
        # Center to maintain sum=1
        direction -= direction.mean()

        v = momentum * v + (1.0 - momentum) * direction
        a_new = a - lr * v
        a_new = np.maximum(a_new, 1e-8)
        a_new /= a_new.sum()
        J_new = compute_J(a_new)
        if J_new < current_J - 1e-15:
            a = a_new
            current_J = J_new
            lr = min(lr * 1.02, 10.0 / n)
        else:
            lr *= 0.6
            v *= 0.5
            if lr < 1e-14:
                break
        if it % 20 == 0:
            tracker.update(a)
    tracker.update(a)
    return a

def lse_smooth(a, max_iters, deadline, tracker):
    """Log-sum-exp smoothed gradient descent."""
    a = np.maximum(a.astype(np.float64), 1e-6)
    a /= a.sum()
    n = len(a)
    current_J = compute_J(a)
    tracker.update(a)
    tau = 100.0
    lr = 1.0 / (n * tau)
    momentum = 0.8
    v = np.zeros(n)

    for it in range(max_iters):
        if time.monotonic() > deadline - 5:
            break
        b = fast_conv(a)
        b_max = b.max()
        # Clip for numerical stability
        b_clipped = np.clip(b - b_max, -50.0, 0.0)
        exp_b = np.exp(tau * b_clipped)
        p = exp_b / exp_b.sum()

        # Gradient of LSE: cross-correlation of p and a
        s = 1
        while s < 3 * n:
            s <<= 1
        fp = np.fft.rfft(p, s)
        fa = np.fft.rfft(a, s)
        # cross-correlation: ifft(conj(fp) * fa) gives correlation
        corr = np.fft.irfft(np.conj(fp) * fa, s)
        grad = 2.0 * corr[:n]

        v = momentum * v + (1.0 - momentum) * grad
        a_new = a - lr * v
        a_new = np.maximum(a_new, 1e-6)
        a_new /= a_new.sum()
        J_new = compute_J(a_new)
        if J_new < current_J - 1e-15:
            a = a_new
            current_J = J_new
            tau = min(500.0, tau * 1.01)
            lr = 1.0 / (n * tau)
        else:
            lr *= 0.5
            v *= 0.5
            if lr < 1e-16:
                break
        if it % 50 == 0:
            tracker.update(a)
    tracker.update(a)
    return a

def simulated_annealing_lp(a, deadline, tracker, n_restarts=30):
    """SA with LP refinement at each step."""
    rng = np.random.default_rng(42)
    n = a.size
    a = a.copy().astype(np.float64)
    current_J = compute_J(a)
    tracker.update(a)
    temp = 0.1

    for r in range(n_restarts):
        if time.monotonic() > deadline - 10:
            break
        # Perturb
        sigma = temp * max(1.0, n ** (-0.5))
        a_pert = a + rng.normal(0, sigma, n)
        a_pert = np.maximum(a_pert, 1e-6)
        a_pert /= a_pert.sum()

        # Short LP refinement
        a_ref = lp_optimize(a_pert, max(5, 15 - r // 4), deadline - 10, tracker,
                           eps_start=0.03, eps_end=0.005, eps_decay=0.9)
        J_ref = compute_J(a_ref)
        delta = J_ref - current_J
        if delta < 0 or rng.random() < np.exp(-delta / max(temp, 1e-10)):
            a = a_ref
            current_J = J_ref
        temp *= 0.93
    tracker.update(a)
    return a

def propose_candidate(seed=42, budget_s=1500, **kwargs):
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + float(budget_s) - 25.0
    tracker = Tracker()

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        if p.ndim == 1 and p.size > 0 and np.all(np.isfinite(p)):
            parent = np.maximum(p, 0.0)
            parent /= parent.sum()
            tracker.update(parent)
            print(f"Parent loaded: n={parent.size}, J={tracker.best_score:.8f}", flush=True)
    except (NameError, AttributeError):
        pass

    # === Phase 1: Tournament - many short optimizations ===
    print(f"\n{'='*60} Phase 1: Tournament {'='*60}", flush=True)
    lengths = [256, 320, 384, 448, 512, 576, 640]
    # Filter: if parent exists, include its length
    if parent is not None and parent.size not in lengths:
        lengths.append(parent.size)
    lengths = sorted(set(lengths))

    # Allocate time per length proportional to 1/n (smaller n = cheaper)
    inv_n = np.array([1.0 / l for l in lengths])
    inv_n /= inv_n.sum()
    phase1_time = (deadline - time.monotonic()) * 0.30
    phase1_end = time.monotonic() + phase1_time

    candidates = []  # list of (J, n, a)

    for n in lengths:
        t_alloc = max(15.0, phase1_time * inv_n[n // lengths[0] if n in lengths else 0])
        t_end = min(phase1_end, time.monotonic() + t_alloc)
        if time.monotonic() > phase1_end - 10:
            continue
        if t_end - time.monotonic() < 15:
            continue

        print(f"  n={n}, budget={t_end - time.monotonic():.0f}s", flush=True)
        inits = generate_inits(n, rng, parent)
        n_inits = len(inits)
        t_per_init = max(8.0, (t_end - time.monotonic() - 5) / n_inits)

        best_n_J = float('inf')
        best_n_a = None

        for idx, a0 in enumerate(inits):
            if time.monotonic() > t_end - 5:
                break
            t_init_end = time.monotonic() + t_per_init
            # Short LP
            a = lp_optimize(a0, max(5, int(t_per_init / 3)), t_init_end, tracker,
                           eps_start=0.05, eps_end=0.01, eps_decay=0.9)
            J = compute_J(a)
            if J < best_n_J:
                best_n_J = J
                best_n_a = a.copy()
            tracker.update(a)

        if best_n_a is not None:
            candidates.append((best_n_J, n, best_n_a))
            print(f"  n={n} best: J={best_n_J:.8f}", flush=True)

    # Rank candidates
    candidates.sort(key=lambda x: x[0])
    print(f"  Candidates: {[(c[0]:.6f, c[1]) for c in candidates[:5]]}", flush=True)

    # === Phase 2: Long optimization on top candidates ===
    print(f"\n{'='*60} Phase 2: Long Optimization {'='*60}", flush=True)
    phase2_time = (deadline - time.monotonic()) * 0.35
    phase2_end = time.monotonic() + phase2_time

    # Pick top 3 candidates (or fewer if time is tight)
    n_top = min(3, len(candidates))
    if time.monotonic() > phase2_end - 60:
        n_top = 1

    for ci in range(n_top):
        if time.monotonic() > phase2_end - 30:
            break
        J_c, n_c, a_c = candidates[ci]
        t_alloc_c = max(30.0, (phase2_end - time.monotonic()) / (n_top - ci))
        t_end_c = time.monotonic() + t_alloc_c
        print(f"  Candidate {ci}: n={n_c}, J_start={J_c:.8f}, budget={t_alloc_c:.0f}s", flush=True)

        # Long LP optimization
        a = lp_optimize(a_c, 200, t_end_c - 60, tracker,
                       eps_start=0.04, eps_end=0.001, eps_decay=0.96)

        # Subgradient polishing
        if time.monotonic() < t_end_c - 30:
            a = dual_weighted_subgradient(a, 2000, t_end_c - 30, tracker)

        # LSE smoothing
        if time.monotonic() < t_end_c - 10:
            a = lse_smooth(a, 1500, t_end_c - 5, tracker)

        J_final = compute_J(a)
        candidates[ci] = (J_final, n_c, a.copy())
        print(f"  Candidate {ci} done: J={J_final:.8f}", flush=True)
        tracker.update(a)

    # === Phase 3: Simulated annealing escape ===
    print(f"\n{'='*60} Phase 3: Simulated Annealing {'='*60}", flush=True)
    phase3_end = time.monotonic() + (deadline - time.monotonic()) * 0.20

    if tracker.best is not None and time.monotonic() < phase3_end - 20:
        a_sa = simulated_annealing_lp(tracker.best, phase3_end - 10, tracker, n_restarts=25)
        tracker.update(a_sa)

    # === Phase 4: Final polish ===
    print(f"\n{'='*60} Phase 4: Final Polish {'='*60}", flush=True)
    if tracker.best is not None:
        a = tracker.best.copy()
        # Final LP with very tight epsilon
        a = lp_optimize(a, 50, deadline - 30, tracker,
                       eps_start=0.02, eps_end=0.0005, eps_decay=0.95)
        # Final subgradient
        if time.monotonic() < deadline - 20:
            a = dual_weighted_subgradient(a, 3000, deadline - 15, tracker)
        # Final LSE
        if time.monotonic() < deadline - 5:
            a = lse_smooth(a, 2000, deadline - 2, tracker)
        tracker.update(a)

    # Fallback
    if tracker.best is None:
        tracker.best = np.ones(256) / 256
        tracker.best_score = compute_J(tracker.best)
        tracker.update(tracker.best)

    tracker.report("FINAL")
    print(f"Took {time.monotonic() - tracker.t0:.1f}s total", flush=True)
    return [float(x) for x in tracker.best]
