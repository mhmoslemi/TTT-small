"""
Improved LP mirror descent for minimizing J(a) = 2*n*max(conv(a,a))/sum(a)^2.
Key improvements over previous: active-set LP (reduced constraints), adaptive
step size with aggressive backtracking, multi-length search, diverse restarts,
and tuned hyperparameters for faster convergence.
"""

import numpy as np
from scipy.optimize import linprog
from scipy.sparse import csr_matrix
import time


def fast_conv(a):
    """Full self-convolution via FFT."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]


def compute_J(a):
    """Compute J(a) = 2*n*max(conv(a,a))/(sum(a))^2."""
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))


def solve_best_response(a, M, gap_frac=0.05):
    """Solve LP: max sum(g) s.t. A(g)_k <= M for k where conv_a_k >= (1-gap_frac)*M, g >= 0."""
    n = len(a)
    b = fast_conv(a)
    threshold = M * (1.0 - gap_frac)
    active = np.where(b >= threshold)[0]

    if len(active) == 0:
        return np.ones(n) / n
    if len(active) == 2 * n - 1:
        active = np.where(b >= M * 0.5)[0]
        if len(active) == 0:
            active = np.where(b >= 0)[0]

    rows = []
    cols = []
    vals = []
    for idx, k in enumerate(active):
        j_min = max(0, k - n + 1)
        j_max = min(n - 1, k)
        for j in range(j_min, j_max + 1):
            rows.append(idx)
            cols.append(j)
            vals.append(a[k - j])

    A = csr_matrix((vals, (rows, cols)), shape=(len(active), n))
    c = -np.ones(n)
    b_ub = np.full(len(active), M)
    bounds = [(0, None)] * n

    try:
        r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                    options={'presolve': True})
        if r.success and r.x is not None:
            g = np.maximum(r.x, 0)
            gs = g.sum()
            if gs > 1e-15:
                return g / gs
    except Exception:
        pass
    return a.copy()


def mirror_descent(a, max_iters, deadline, multiplicative=True, init_t=1.0, gap_frac=0.05):
    """LP mirror descent with adaptive backtracking step size."""
    n = len(a)
    a = np.maximum(a.astype(np.float64), 1e-12)
    a = a / a.sum()
    current_J = compute_J(a)
    best_J = current_J
    best_a = a.copy()
    no_improve = 0
    t = init_t
    it = 0

    while it < max_iters and time.monotonic() < deadline:
        b = fast_conv(a)
        M = float(b.max())
        g = solve_best_response(a, M, gap_frac)

        if g is None or not np.all(np.isfinite(g)):
            break
        g = np.maximum(g, 0)
        gs = g.sum()
        if gs < 1e-12:
            break
        g = g / gs

        found = False
        t_local = t
        for _ in range(25):
            if multiplicative:
                a_trial = a * np.power(g / np.maximum(a, 1e-15), t_local)
            else:
                a_trial = (1.0 - t_local) * a + t_local * g

            a_trial = np.maximum(a_trial, 0)
            s = a_trial.sum()
            if s < 1e-12:
                t_local *= 0.5
                continue
            a_trial = a_trial / s
            J_trial = compute_J(a_trial)

            if J_trial < current_J - 1e-15:
                a = a_trial
                current_J = J_trial
                found = True
                t = t_local
                break
            t_local *= 0.5
            if t_local < 0.001:
                break

        it += 1
        if found:
            if current_J < best_J:
                best_J = current_J
                best_a = a.copy()
            no_improve = 0
            # Occasionally try to increase step size
            if it % 200 == 0 and t < 0.5:
                t = min(1.0, t * 1.3)
        else:
            no_improve += 1
            if no_improve > 60:
                break

    return best_a, best_J


def resample(src, dest_n):
    """Resample 1D sequence to new length."""
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)


def make_inits(n, rng, parent=None):

    inits = []
    t = np.arange(n, dtype=np.float64) / n

    if parent is not None:

        p = np.maximum(p, 0.01)

        inits.append(p)

        inits.append(np.maximum(p + rng.normal(0, 0.05, n), 0.01))

    kernel = np.ones(k) / k
    r = np.convolve(rng.uniform(0, 1, n), kernel, mode='same')
    inits.append(np.maximum(r + 0.2, 0.01))

    result = []
    for a in inits:
        a = np.maximum(np.asarray(a, dtype=np.float64), 0.001)
        s = a.sum()
        if s > 1e-6:
            result.append(a / s)
