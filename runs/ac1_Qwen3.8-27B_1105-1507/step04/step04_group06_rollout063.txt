Search for non-negative coefficients minimizing
    J(a) = 2*n*max(convolve(a,a)) / (sum(a))**2.

Main ideas:
  * Use the invariant q_k = binom(2k,k)/4^k as a strong family of initial
   isations.  These are the coefficients of (1-x)**(-1/2), and their finite
    truncations make the first half of the polynomial-square coefficients
    unusually flat.
  * Run LP-based best-response mirror descent.  For fixed a, the linear program
    maximizes sum(g) subject to convolve(a,g) <= M, g >= 0.  This gives the
    steepest feasible direction for reducing max convolve after normalisation.
  * Add smooth log-sum-exp / softmax L-BFGS runs to escape LP basins.
  * Explore several lengths, including 1024 and 2048, with adaptive budgeting.
"""

import time
import numpy as np
from scipy.fft import rfft, irfft, next_fast_len
from scipy.optimize import linprog, minimize
from scipy.sparse import csr_matrix

_A_cache = {}
MAX_LP_N = 2048


def fft_conv(a):
    """Fast full linear convolution a * a via FFT."""
    a = np.asarray(a, dtype=np.float64)
    n = a.size
    if n == 0:
        return np.empty(0, dtype=np.float64)
    m = 2 * n - 1
    nfft = next_fast_len(m)
    fa = rfft(a, nfft)
    c = irfft(fa * fa, nfft)[:m]
    return c


def compute_J(a):
    """Fast internal objective value."""
    a = np.asarray(a, dtype=np.float64)
    if a.size == 0:
        return float("inf")
    a = np.clip(a, 0.0, 1000.0)
    s = float(a.sum())
    if s < 0.01:
        return float("inf")
    c = fft_conv(a)
    M = float(np.max(c))
    if M < 0.0:
        M = 0.0
    J = 2.0 * a.size * M / (s * s)
    if not np.isfinite(J):
        return float("inf")
    return float(J)


def q_sequence(n):
    """Normalised sequence with coefficients of (1-x)^(-1/2)."""
    if n <= 0:
        return np.empty(0, dtype=np.float64)
    q = np.empty(n, dtype=np.float64)
    q[0] = 1.0
    for k in range(1, n):
        q[k] = q[k - 1] * (2.0 * k - 1.0) / (2.0 * k)
    s = q.sum()
    if s <= 0.0:
        q = np.ones(n, dtype=np.float64)
    else:
        q /= s
    return q


def resample_seq(src, dest_n):
    """Linearly resample a 1-D sequence to dest_n."""
    src = np.asarray(src, dtype=np.float64)
    if src.size == 0:
        return np.ones(dest_n, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    t_old = np.linspace(0.0, 1.0, src.size)
    t_new = np.linspace(0.0, 1.0, dest_n)
    out = np.interp(t_new, t_old, src)
    return np.maximum(out, 0.0)


def perturb_sequence(a, rng, ptype, param):
    """Return a perturbed normalised copy of a."""
    a = np.asarray(a, dtype=np.float64)
    n = a.size
    out = a.copy()

    if ptype == "gauss":
        out = out + rng.normal(0.0, float(param), n) * out
    elif ptype == "uniform":
        out = out + rng.uniform(-float(param), float(param), n) * out
    elif ptype == "spike":
        k = int(param)
        if k <= 0:
            k = 1
        k = min(k, max(1, n // 4))
        idx = rng.choice(n, size=k, replace=False)
        out[idx] *= rng.uniform(0.25, 3.5, k)
    elif ptype == "smooth":
        noise = rng.normal(0.0, float(param), n)
        kk = max(3, n // 64)
        kernel = np.ones(kk, dtype=np.float64) / float(kk)
        noise = np.convolve(noise, kernel, mode="same")
        out = out + noise * out
    elif ptype == "fourier":
        t = np.linspace(0.0, 1.0, n, endpoint=False)
        f = rng.uniform(0.75, 8.0)
        ph = rng.uniform(0.0, 2.0 * np.pi)
        amp = float(param)
        mod = 1.0 + amp * np.cos(2.0 * np.pi * f * t + ph)
        out = out * mod

    out = np.maximum(out, 1e-9)
    s = out.sum()
    if s <= 0.0:
        return np.ones(n, dtype=np.float64) / n
    return out / s


class ScoreTracker:
    def __init__(self):
        self.best = None
        self.best_score = float("inf")
        self.t0 = time.monotonic()

    def update(self, a):
        a = np.asarray(a, dtype=np.float64)
        if a.size == 0:
            return
        a = np.clip(a, 0.0, 1000.0)
        s = a.sum()
        if s < 0.01:
            return
        score = compute_J(a)
        if np.isfinite(score) and score < self.best_score:
            self.best = a.copy()
            self.best_score = float(score)
            print(
                "  * NEW BEST: J=%.8f n=%d t=%.1fs"
                % (self.best_score, self.best.size, time.monotonic() - self.t0),
                flush=True,
            )


def ta_product(p, a):
    """
    Compute h = T(p) a where T(p) has anti-diagonal weights p:
        h_i = sum_j p_{i+j} a_j.
    This is the gradient (up to factor 2) of sum_k p_k conv(a,a)_k.
    """
    n = a.size
    L = 2 * n - 1
    q = p[::-1]
    length = L + n - 1
    nfft = next_fast_len(length)
    fq = rfft(q, nfft)
    fa = rfft(a, nfft)
    conv = np.real(irfft(fq * fa, nfft))
    idx = np.arange(L - 1, L - 1 - n, -1, dtype=np.intp)
    return conv[idx]


def smooth_function_and_grad(z, eta):
    """
    Smooth objective for softmax parameterisation a = softmax(z).
    eta is dimensionless sharpness of log-sum-exp approximation to max.
    Returns (value, gradient w.r.t. z).
    """
    z = np.asarray(z, dtype=np.float64)
    if z.size == 0:
        return 1.0e100, z
    if not np.all(np.isfinite(z)):
        return 1.0e100, np.zeros_like(z)

    zm = z - np.max(z)
    ez = np.exp(np.clip(zm, -745.0, 0.0))
    ez_sum = ez.sum()
    if ez_sum <= 1e-300:
        return 1.0e100, np.zeros_like(z)

    a = ez / ez_sum
    n = a.size

    c = fft_conv(a)
    cmax = float(np.max(c))
    if cmax <= 1e-300:
        return 0.0, np.zeros_like(z)

    scale = max(cmax, 1e-300)
    shifted = (float(eta) * (c - cmax)) / scale
    shifted = np.clip(shifted, -700.0, 700.0)
    ep = np.exp(shifted)
    Z = ep.sum()
    if Z <= 1e-300:
        p = np.zeros_like(c)
        p[int(np.argmax(c))] = 1.0
        smooth = cmax
    else:
        p = ep / Z
        smooth = cmax + scale * np.log(Z) / float(eta)

    h = ta_product(p, a)
    grad_smooth = 2.0 * h
    grad_a = 2.0 * n * grad_smooth
    val = 2.0 * n * smooth

    grad_z = a * (grad_a - np.dot(a, grad_a))
    grad_z = np.nan_to_num(grad_z, nan=0.0, posinf=0.0, neginf=0.0)
    return float(val), grad_z


def smooth_lbfgs_fun(z, eta):
    val, _ = smooth_function_and_grad(z, eta)
    return val


def smooth_lbfgs_jac(z, eta):
    _, g = smooth_function_and_grad(z, eta)
    return g


def lbfgs_smooth_optimize(a0, eta_schedule, deadline, tracker=None):
    """Optimise smooth approximation using softmax + L-BFGS-B."""
    a = np.maximum(np.asarray(a0, dtype=np.float64), 1e-12)
    s = a.sum()
    if s <= 1e-12:
        return None
    a = a / s
    if a.size == 0:
        return None

    z = np.log(np.clip(a, 1e-300, 1.0))
    n = a.size

    # Limit maxiter for very large n to keep L-BFGS cheap.
    for eta, maxiter in eta_schedule:
        if time.monotonic() > deadline:
            break
        mi = int(maxiter)
        if n > 1536:
            mi = min(mi, 35)
        else:
            mi = min(mi, 70)
        if mi <= 0:
            continue
        try:
            res = minimize(
                smooth_lbfgs_fun,
                z,
                args=(float(eta),),
                jac=smooth_lbfgs_jac,
                method="L-BFGS-B",
                options={
                    "maxiter": mi,
                    "ftol": 1e-14,
                    "gtol": 1e-9,
                    "maxcor": 12,
                    "maxls": 20,
                },
            )
            z = np.asarray(res.x, dtype=np.float64)
            if np.all(np.isfinite(z)):
                zm = z - np.max(z)
                ez = np.exp(np.clip(zm, -745.0, 0.0))
                ez_sum = ez.sum()
                if ez_sum > 1e-12:
                    a = ez / ez_sum
                    a = np.maximum(a, 0.0)
                    s = a.sum()
                    if s > 1e-12:
                        a /= s
                    if tracker is not None:
                        tracker.update(a)
        except Exception:
            pass

    return a


def get_A_structure(n):
    """Triangle Toeplitz row/column indices and row offsets."""
    if n in _A_cache:
        return _A_cache[n]

    rows = []
    cols = []
    offsets = [0]
    total = 0
    for k in range(2 * n - 1):
        jl = max(0, k - n + 1)
        jh = min(n - 1, k)
        L = jh - jl + 1
        total += L
        rows.extend([k] * L)
        cols.extend(range(jl, jh + 1))
        offsets.append(total)

    rows = np.asarray(rows, dtype=np.int32)
    cols = np.asarray(cols, dtype=np.int32)
    offsets = np.asarray(offsets, dtype=np.int64)
    _A_cache[n] = (rows, cols, offsets)
    return _A_cache[n]


def select_lp_rows(c, n, K):
    """Select rows for the LP.  K=None means all rows."""
    L = 2 * n - 1
    if K is None or K >= L:
        return None
    K = max(1, int(K))
    if K >= L:
        return None

    kth = L - K
    idx = np.argpartition(c, kth)[kth:].astype(np.int64)
    crit = np.array([0, n - 1, 2 * n - 2], dtype=np.int64)
    comb = np.unique(np.concatenate((idx, crit)))
    return np.sort(comb).astype(np.int32)


def build_A_selected(a, n, selected):
    """Build sparse B for selected convolution rows."""
    if n > MAX_LP_N:
        return None
    rows_all, cols_all, offsets = get_A_structure(n)
    total_rows = 2 * n - 1

    if selected is None:
        vals = a[rows_all - cols_all]
        return csr_matrix((vals, (rows_all, cols_all)), shape=(total_rows, n))

    # If almost all rows are selected, it is simpler to use all rows.
    if selected.size > 0.80 * total_rows:
        vals = a[rows_all - cols_all]
        return csr_matrix((vals, (rows_all, cols_all)), shape=(total_rows, n))

    K = selected.size
    if K == 0:
        return None

    # Fast path: selected rows are exactly a prefix.
    if selected.size > 0 and selected[0] == 0 and selected[-1] < selected.size:
        end = offsets[selected.size]
        rows_sel = rows_all[:end]
        cols_sel = cols_all[:end]
    else:
        row_parts = []
        col_parts = []
        for k in selected:
            lo = offsets[k]
            hi = offsets[k + 1]
            L = hi - lo
            if L > 0:
                row_parts.append(np.full(L, int(k), dtype=np.int32))
                col_parts.append(cols_all[lo:hi])
        if not row_parts:
            return None
        rows_sel = np.concatenate(row_parts)
        cols_sel = np.concatenate(col_parts)

    vals = a[rows_sel - cols_sel]
    local_rows = np.arange(K, dtype=np.int32)
    return csr_matrix((vals, (local_rows, cols_sel)), shape=(K, n))


def lp_best_response(a, c, M, K):
    """
    LP: max sum(g) s.t. convolve(a, g) <= M on selected rows, g >= 0.
    Returns (g_normalized, raw_sum).
    """
    n = len(a)
    if n > MAX_LP_N or M <= 0.0:
        return None, 0.0

    selected = select_lp_rows(c, n, K)
    A = build_A_selected(a, n, selected)
    if A is None or A.nnz == 0:
        return None, 0.0

    cvec = -np.ones(n, dtype=np.float64)
    bvec = np.full(A.shape[0], M * (1.0 + 1e-12), dtype=np.float64)
    bounds = [(0.0, None)] * n

    try:
        res = linprog(
            cvec,
            A_ub=A,
            b_ub=bvec,
            bounds=bounds,
            method="highs",
            options={
                "presolve": True,
                "primal_feasibility_tolerance": 1e-8,
                "dual_feasibility_tolerance": 1e-8,
            },
        )
    except Exception:
        return None, 0.0

    if res.success and res.x is not None:
        g = np.maximum(res.x, 0.0)
        raw = float(g.sum())
        if raw > 1e-12:
            return g / raw, raw
    return None, 0.0


def line_search(a, g, n_evals=12):
    """Golden-section line search along (1-t)*a + t*g."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    g = np.maximum(np.asarray(g, dtype=np.float64), 0.0)
    sa = a.sum()
    sg = g.sum()
    if sa <= 1e-12 or sg <= 1e-12:
        return a.copy(), compute_J(a)
    a = a / sa
    g = g / sg

    best_a = a.copy()
    best_J = compute_J(a)
    lo = 0.0
    hi = 1.0
    gr = (np.sqrt(5.0) - 1.0) / 2.0

    g1 = compute_J(g)
    if g1 < best_J:
        best_J = g1
        best_a = g.copy()

    for _ in range(max(1, int(n_evals))):
        if hi - lo < 1e-8:
            break
        t1 = hi - gr * (hi - lo)
        t2 = lo + gr * (hi - lo)

        a1 = (1.0 - t1) * a + t1 * g
        a1 = np.maximum(a1, 0.0)
        s1 = a1.sum()
        if s1 > 1e-12:
            a1 /= s1
            J1 = compute_J(a1)
        else:
            J1 = float("inf")

        a2 = (1.0 - t2) * a + t2 * g
        a2 = np.maximum(a2, 0.0)
        s2 = a2.sum()
        if s2 > 1e-12:
            a2 /= s2
            J2 = compute_J(a2)
        else:
            J2 = float("inf")

        if J1 < best_J:
            best_J = J1
            best_a = a1.copy()
        if J2 < best_J:
            best_J = J2
            best_a = a2.copy()

        if J1 < J2:
            hi = t2
        else:
            lo = t1

    return best_a, float(best_J)


def lp_mirror_descent(a, deadline, tracker, K=None, n_line_evals=12,
                      stall_patience=25, max_iters=100000):
    """LP best-response mirror descent with line search."""
    a = np.asarray(a, dtype=np.float64)
    n = a.size
    if n == 0 or n > MAX_LP_N:
        return a

    a = np.maximum(a, 0.0)
    s = a.sum()
    if s <= 1e-12:
        return a
    a = a / s
    tracker.update(a)

    current_J = compute_J(a)
    stall = 0

    for _ in range(int(max_iters)):
        if time.monotonic() > deadline:
            break

        c = fft_conv(a)
        M = float(np.max(c))
        if M <= 0.0:
            break

        g, raw = lp_best_response(a, c, M, K)
        if g is None:
            break

        # If the LP barely improves over g=a, this direction is stagnant.
        if raw <= 1.0 + 1e-8:
            stall += 1
            if stall >= stall_patience:
                break
            continue

        est_J = current_J / max(raw, 1.0)
        if est_J >= current_J * (1.0 - 1e-7):
            stall += 1
            if stall >= stall_patience:
                break
            continue

        new_a, new_J = line_search(a, g, n_evals=int(n_line_evals))
        if new_J < current_J - 1e-12:
            a = new_a
            current_J = new_J
            stall = 0
            tracker.update(a)
            continue

        # Try small fixed convex steps.
        improved = False
        for t in (0.02, 0.05, 0.01, 0.1, 0.2):
            bt = (1.0 - t) * a + t * g
            bt = np.maximum(bt, 0.0)
            sb = bt.sum()
            if sb <= 1e-12:
                continue
            bt /= sb
            Jb = compute_J(bt)
            if Jb < current_J - 1e-12:
                a = bt
                current_J = Jb
                improved = True
                stall = 0
                tracker.update(a)
                break

        if not improved:
            stall += 1
            if stall >= stall_patience:
                break

    tracker.update(a)
    return a


def make_initial_candidates(n, rng, best=None, parent=None, include_cos=True):
    """Create diverse normalised initialisations of length n."""
    inits = []

    def maybe_add(a):
        a = np.asarray(a, dtype=np.float64)
        if a.size != n:
            return
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s > 1e-12:
            inits.append(a / s)

    if best is not None:
        maybe_add(resample_seq(best, n))

    if parent is not None:
        maybe_add(resample_seq(parent, n))

    q = q_sequence(n)
    maybe_add(q)

    if best is not None:
        rb = resample_seq(best, n)
        maybe_add(0.5 * (q + rb))

    if parent is not None:
        rp = resample_seq(parent, n)
        maybe_add(0.5 * (q + rp))

    maybe_add(np.ones(n, dtype=np.float64))

    t = np.linspace(0.0, 1.0, n, endpoint=False)
    idx = np.arange(n, dtype=np.float64)

    for beta in (0.25, 0.5, 0.75):
        maybe_add((idx + 1.0) ** (-beta))

    for alpha in (0.02, 0.08, 0.25):
        maybe_add(np.exp(-alpha * t))

    for gamma in (0.35, 0.8, 1.6):
        maybe_add((idx + 1.0) ** (-0.5) * (1.0 - 0.75 * t) ** gamma)

    if include_cos:
        for freq in (1, 2, 3, 5):
            for amp in (0.08, 0.18):
                maybe_add(q * (1.0 + amp * np.cos(2.0 * np.pi * freq * t)))

    # A few random positive low-pass sequences.
    for _ in range(2):
        r = rng.random(n)
        r = np.convolve(r, np.ones(max(3, n // 128), dtype=np.float64), mode="same")
        maybe_add(r + 1e-3)

    # Unique-ish by tuple hash only for short n; otherwise simply return list.
    if n <= 2048 and len(inits) > 8:
        seen = set()
        uniq = []
        for a in inits:
            key = tuple(np.round(a, 8).astype(np.float64))
            if key not in seen:
                seen.add(key)
                uniq.append(a)
        inits = uniq[:12]
    else:
        inits = inits[:12]

    return inits


def propose_candidate(seed=42, budget_s=1500.0):
    """
    Entry point.  Searches for a low-scoring non-negative sequence.
    Uses about budget_s seconds and returns the best sequence found.
    """
    budget_s = float(max(60.0, budget_s))
    rng = np.random.default_rng(seed)
    start_time = time.monotonic()
    deadline = start_time + budget_s - 20.0

    tracker = ScoreTracker()

    # Load parent if available.
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        if p.ndim == 1 and p.size > 0 and np.all(np.isfinite(p)):
            parent = np.clip(p, 0.0, 1000.0)
            tracker.update(parent)
            print("Parent loaded: n=%d J=%.8f" % (parent.size, tracker.best_score), flush=True)
    except Exception:
        pass

    if parent is None:
        parent = q_sequence(1024)
        tracker.update(parent)
        print("Using q(1024) as default parent: J=%.8f" % tracker.best_score, flush=True)

    # --- Initial q-family candidates ---
    for n in (256, 512, 1024, 2048):
        if time.monotonic() > deadline - 250.0:
            break
        tracker.update(q_sequence(n))
        if parent is not None and parent.size > 0:
            rp = resample_seq(parent, n)
            tracker.update(0.5 * (q_sequence(n) + rp))
            tracker.update(rp)

    # --- Phase 1: smooth L-BFGS basin finding ---
    if tracker.best is not None and time.monotonic() < deadline:
        now = time.monotonic()
        phase1_budget = min(140.0, max(30.0, deadline - now - 850.0))
        phase1_deadline = time.monotonic() + phase1_budget
        print("\n=== Phase 1: smooth L-BFGS, budget %.0fs ===" % phase1_budget, flush=True)

        lengths = set()
        if tracker.best.size <= 2048 and tracker.best.size >= 128:
            lengths.add(tracker.best.size)
        for n in (512, 1024, 2048):
            lengths.add(n)
        lengths = sorted([x for x in lengths if 128 <= x <= 2048])

        eta_schedule = [
            (10.0, 30),
            (50.0, 40),
            (200.0, 45),
            (700.0, 35),
        ]

        for n in lengths:
            if time.monotonic() > phase1_deadline:
                break
            t0 = time.monotonic()
            inits = make_initial_candidates(n, rng, best=tracker.best, parent=parent, include_cos=True)
            # Limit initialisations to the best available time.
            per_init_budget = max(8.0, min(35.0, (phase1_deadline - time.monotonic()) / max(1, len(inits))))
            for a0 in inits[:6]:
                if time.monotonic() > phase1_deadline:
                    break
                init_deadline = min(phase1_deadline, time.monotonic() + per_init_budget)
                aopt = lbfgs_smooth_optimize(a0, eta_schedule, init_deadline, tracker=tracker)
                if aopt is not None:
                    tracker.update(aopt)
            print("  Phase1 n=%d done in %.1fs" % (n, time.monotonic() - t0), flush=True)

    # --- Phase 2: main LP mirror descent ---
    if tracker.best is not None and time.monotonic() < deadline:
        now = time.monotonic()
        remaining = deadline - now
        main_budget = min(900.0, max(60.0, remaining - 350.0))
        main_deadline = time.monotonic() + main_budget
        print("\n=== Phase 2: LP mirror descent, budget %.0fs ===" % main_budget, flush=True)

        if tracker.best.size <= MAX_LP_N:
            K = None if tracker.best.size <= 1024 else min(tracker.best.size, 2 * tracker.best.size - 1)
            a = lp_mirror_descent(
                tracker.best,
                main_deadline,
                tracker,
                K=K,
                n_line_evals=12,
                stall_patience=25,
                max_iters=100000,
            )
            tracker.update(a)

        # Try / refine nearby feasible lengths if time remains.
        n_best = tracker.best.size if tracker.best is not None else 1024
        for n_try in (1024, 1536, 2048, 768):
            if time.monotonic() > main_deadline:
                break
            if n_try == n_best:
                continue
            if n_try > MAX_LP_N:
                continue
            a_try = resample_seq(tracker.best, n_try)
            tracker.update(a_try)
            K2 = None if n_try <= 1024 else min(n_try, 2 * n_try - 1)
            a_try = lp_mirror_descent(
                a_try,
                main_deadline,
                tracker,
                K=K2,
                n_line_evals=10,
                stall_patience=20,
                max_iters=50000,
            )
            tracker.update(a_try)

    # --- Phase 3: diverse restarts and short polishing ---
    if tracker.best is not None and time.monotonic() < deadline:
        now = time.monotonic()
        restart_budget = min(180.0, max(40.0, deadline - now - 120.0))
        restart_deadline = time.monotonic() + restart_budget
        print("\n=== Phase 3: restarts, budget %.0fs ===" % restart_budget, flush=True)

        # Symmetries / reversible combinations.
        rev = tracker.best[::-1].copy()
        tracker.update(rev)
        mix_rev = 0.5 * (tracker.best + rev)
        tracker.update(mix_rev)

        strategies = [
            ("gauss", 0.03),
            ("gauss", 0.07),
            ("gauss", 0.14),
            ("uniform", 0.08),
            ("spike", 6),
            ("smooth", 0.07),
            ("fourier", 0.18),
            ("gauss", 0.015),
        ]

        for ptype, param in strategies:
            for _ in range(6):
                if time.monotonic() > restart_deadline:
                    break
                a0 = tracker.best.copy()
                a0 = perturb_sequence(a0, rng, ptype, param)
                if a0.size <= MAX_LP_N:
                    K = None if a0.size <= 1024 else min(a0.size, 2 * a0.size - 1)
                    sub_dead = min(restart_deadline, time.monotonic() + 18.0)
                    aopt = lp_mirror_descent(
                        a0,
                        sub_dead,
                        tracker,
                        K=K,
                        n_line_evals=8,
                        stall_patience=10,
                        max_iters=30000,
                    )
                    tracker.update(aopt)
                else:
                    sub_dead = min(restart_deadline, time.monotonic() + 15.0)
                    aopt = lbfgs_smooth_optimize(a0, [(20.0, 25), (120.0, 30)], sub_dead, tracker=tracker)
                    if aopt is not None:
                        tracker.update(aopt)

    # --- Phase 4: final long polish ---
    if tracker.best is not None and time.monotonic() < deadline:
        now = time.monotonic()
        final_deadline = deadline
        print("\n=== Phase 4: final polish until %.1fs ===" % (final_deadline - start_time), flush=True)

        if tracker.best.size <= MAX_LP_N:
            K = None if tracker.best.size <= 1024 else min(tracker.best.size, 2 * tracker.best.size - 1)
            a = lp_mirror_descent(
                tracker.best,
                final_deadline,
                tracker,
                K=K,
                n_line_evals=14,
                stall_patience=40,
                max_iters=200000,
            )
            tracker.update(a)

            if tracker.best is not None and time.monotonic() < final_deadline:
                # A second shorter polish on best found.
                a = lp_mirror_descent(
                    tracker.best,
                    final_deadline,
                    tracker,
                    K=K,
                    n_line_evals=12,
                    stall_patience=30,
                    max_iters=100000,
                )
                tracker.update(a)
        else:
            aopt = lbfgs_smooth_optimize(
                tracker.best,
                [(20.0, 50), (120.0, 60), (500.0, 60)],
                final_deadline,
                tracker=tracker,
            )
            if aopt is not None:
                tracker.update(aopt)

    # Fallback if something failed.
    if tracker.best is None:
        tracker.best = q_sequence(1024)
        tracker.best_score = compute_J(tracker.best)

    result = np.clip(np.asarray(tracker.best, dtype=np.float64), 0.0, 1000.0)
    if result.size == 0 or result.sum() < 0.01:
        result = q_sequence(1024)

    # Optional final verification using the evaluator if available.
    try:
        final_eval = evaluate_sequence([float(x) for x in result])
        print("\nFINAL: internal J=%.8f, evaluator=%.8f, n=%d, time=%.1fs"
              % (tracker.best_score, final_eval, result.size, time.monotonic() - start_time), flush=True)
    except Exception:
        print("\nFINAL: internal J=%.8f, n=%d, time=%.1fs"
              % (tracker.best_score, result.size, time.monotonic() - start_time), flush=True)

    return [float(x) for x in result]
