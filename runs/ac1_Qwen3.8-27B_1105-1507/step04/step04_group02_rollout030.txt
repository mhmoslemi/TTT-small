Optimized step-function search using coordinate descent, softmax subgradient,

def build_toeplitz_sparse(a):
    n = len(a)
    rows = []
    cols = []
    vals = []
    for j in range(n):
        start = j
        end = min(j + n, 2 * n - 1)
        length = end - start
        rows.extend(range(start, end))
        cols.extend([j] * length)
        vals.extend(a[:length].tolist())
    A = coo_matrix((vals, (rows, cols)), shape=(2 * n - 1, n)).tocsr()
    return A


def solve_lp(a, M, time_limit=5):
    """Solve LP: max sum(g) s.t. T_a * g <= M, g >= 0 where T_a is the Toeplitz of a."""
    n = len(a)
    A = build_toeplitz_sparse(a)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': time_limit})
        if res.status == 0:
            return np.maximum(res.x, 0)
    except Exception:
        pass
    return None

        conv = compute_conv(a)

        g = solve_lp(a, M * 1.001, time_limit=4)
        if g is None:

        S_g = np.sum(g)
        if S_g < 1e-10:

        g = g / S_g
        
        best_t = 0.0
        best_obj_ls = best_obj
        for t in [0.1, 0.3, 0.5, 0.7, 1.0]:

            a_try = np.maximum(a_try, 0)
            S_try = np.sum(a_try)

"""
Optimized step-function search using coordinate descent, softmax subgradient,
LP alternating, and adaptive multi-start. Key innovation: efficient coordinate descent
exploiting the 1D structure of the convolution objective for fast sub-optimization.
"""

import numpy as np
from scipy.optimize import linprog, minimize_scalar
from scipy.sparse import coo_matrix
import time


def compute_conv(a):
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    conv = np.fft.irfft(fa * fa, size)
    return conv[:2 * n - 1]


def compute_obj(a):
    n = len(a)
    S = float(np.sum(a))
    if S < 1e-10:
        return np.inf
    conv = compute_conv(a)
    M = float(np.max(conv))
    return 2.0 * n * M / (S * S)


def softmax_subgradient(a, max_iters=20000, time_budget=60.0, init_temp=0.05):
    """Softmax-weighted subgradient descent with momentum and adaptive temperature."""
    n = len(a)
    a = a.copy().astype(np.float64)
    S = np.sum(a)
    if S < 1e-10:
        return a, np.inf
    a = a / S
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    v = np.zeros(n)
    momentum = 0.82
    base_lr = 0.15
    temp = init_temp
    stall = 0
    no_improve = 0

    for it in range(max_iters):
        if time.monotonic() - start > time_budget:
            break

        conv = compute_conv(a)
        M = np.max(conv)

        # Softmax weights
        scaled = (conv - M) / (temp * max(M, 1e-10))
        weights = np.exp(scaled)
        wsum = np.sum(weights)
        if wsum < 1e-100:
            weights = np.zeros(len(conv))
            weights[np.argmax(conv)] = 1.0
        else:
            weights /= wsum

        # Gradient via cross-correlation
        grad = 2.0 * np.correlate(weights, a, mode='valid')
        grad_centered = grad - np.mean(grad)

        v = momentum * v + (1.0 - momentum) * grad_centered
        lr = base_lr / (1.0 + 0.003 * it)

        a_new = a - lr * v
        a_new = np.maximum(a_new, 0)
        S_new = np.sum(a_new)
        if S_new < 1e-10:
            break
        a_new = a_new / S_new

        obj = compute_obj(a_new)
        if obj < best_obj - 1e-11:
            best_obj = obj
            best_a = a_new.copy()
            a = a_new
            stall = 0
            no_improve = 0
        else:
            stall += 1
            no_improve += 1
            if no_improve % 100 == 0:
                temp *= 0.92
            if stall > 80:
                rng = np.random.default_rng(it * 31 + 7)
                a = a * np.exp(rng.normal(0, 0.04, n))
                a = np.maximum(a, 0)
                a = a / np.sum(a)
                v = np.zeros(n)
                stall = 0
                base_lr *= 0.85
                if base_lr < 1e-9:
                    break

    return best_a, best_obj


def coordinate_descent(a, n_sweeps=10, time_budget=120.0, max_inner=40):
    """Coordinate descent exploiting 1D structure: for each coordinate, minimize
    the objective treating it as a 1D problem (max of lines + one quadratic over quadratic)."""
    n = len(a)
    a = a.copy().astype(np.float64)
    S = np.sum(a)
    if S < 1e-10:
        return a, np.inf
    a = a / S
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    d_vals = np.arange(n)

    for sweep in range(n_sweeps):
        if time.monotonic() - start > time_budget:
            break
        obj_before = best_obj
        improved_count = 0

        for i in range(n):
            if time.monotonic() - start > time_budget:
                break

            ai = a[i]
            S_base = S - ai
            if S_base < 1e-10:
                continue

            # Build base and linear coefficients
            conv_full = compute_conv(a)
            base = conv_full.copy()
            lin_coeff = np.zeros(2 * n - 1)

            k_vals = i + d_vals  # positions [i, i+1, ..., i+n-1]
            base[k_vals] -= 2.0 * ai * a
            lin_coeff[k_vals] = 2.0 * a

            quad_k = 2 * i
            has_quad = quad_k < 2 * n - 1
            if has_quad:
                base[quad_k] += ai * ai
                lin_coeff[quad_k] = 0.0

            # Define 1D objective
            x_hi = max(3.0 * ai + 0.05 * S, 1e-8)

            def f_1d(x):
                Sx = S_base + x
                if Sx < 1e-10:
                    return 1e10
                vals = base + lin_coeff * x
                if has_quad:
                    vals[quad_k] = base[quad_k] + x * x
                Mv = np.max(vals)
                return 2.0 * n * Mv / (Sx * Sx)

            # Check current value
            f_current = f_1d(ai)
            if f_current > best_obj * 1.001:
                # Something inconsistent, skip
                continue

            # Bounded minimization
            try:
                res = minimize_scalar(f_1d, bounds=(0.0, x_hi), method='bounded',
                                      options={'xatol': 1e-12, 'maxiter': max_inner})
                if res.fun < best_obj - 1e-12:
                    new_ai = res.x
                    a[i] = new_ai
                    S = S_base + new_ai
                    # Verify
                    actual_obj = compute_obj(a)
                    if actual_obj < best_obj:
                        best_obj = actual_obj
                        best_a = a.copy()
                        improved_count += 1
            except Exception:
                pass

        if improved_count == 0:
            # Try a few more coordinates with wider bounds
            if sweep < n_sweeps - 2:
                rng = np.random.default_rng(sweep * 127 + 3)
                for _ in range(min(n, 50)):
                    i = rng.integers(0, n)
                    ai = a[i]
                    S_base = S - ai
                    if S_base < 1e-10:
                        continue
                    conv_full = compute_conv(a)
                    base = conv_full.copy()
                    lin_coeff = np.zeros(2 * n - 1)
                    k_vals = i + d_vals
                    base[k_vals] -= 2.0 * ai * a
                    lin_coeff[k_vals] = 2.0 * a
                    quad_k = 2 * i
                    has_quad = quad_k < 2 * n - 1
                    if has_quad:
                        base[quad_k] += ai * ai
                        lin_coeff[quad_k] = 0.0
                    x_hi = max(5.0 * ai + 0.1 * S, 1e-6)

                    def f_1d_w(x):
                        Sx = S_base + x
                        if Sx < 1e-10:
                            return 1e10
                        vals = base + lin_coeff * x
                        if has_quad:
                            vals[quad_k] = base[quad_k] + x * x
                        return 2.0 * n * np.max(vals) / (Sx * Sx)

                    try:
                        res = minimize_scalar(f_1d_w, bounds=(0.0, x_hi), method='bounded',
                                              options={'xatol': 1e-12, 'maxiter': max_inner})
                        if res.fun < best_obj - 1e-12:
                            a[i] = res.x
                            S = S_base + res.x
                            actual_obj = compute_obj(a)
                            if actual_obj < best_obj:
                                best_obj = actual_obj
                                best_a = a.copy()
                    except Exception:
                        pass
        if improved_count < 3 and sweep > 2:
            # Try small perturbations to escape
            rng = np.random.default_rng(sweep * 7919 + 11)
            a = best_a * np.exp(rng.normal(0, 0.03, n))
            a = np.maximum(a, 0)
            S = np.sum(a)
            a = a / S
            best_obj = compute_obj(a)
            best_a = a.copy()

    return best_a, best_obj


def build_toeplitz_sparse(a):
    n = len(a)
    rows = []
    cols = []
    vals = []
    for j in range(n):
        start = j
        end = min(j + n, 2 * n - 1)
        length = end - start
        rows.extend(range(start, end))
        cols.extend([j] * length)
        vals.extend(a[:length].tolist())
    A = coo_matrix((vals, (rows, cols)), shape=(2 * n - 1, n)).tocsr()
    return A


def solve_lp(a, M, time_limit=5):
    n = len(a)
    A = build_toeplitz_sparse(a)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': time_limit})
        if res.status == 0:
            return np.maximum(res.x, 0)
    except Exception:
        pass
    return None


def alternating_lp(a, max_iters=30, time_budget=40.0):
    n = len(a)
    a = a.copy().astype(np.float64)
    S = np.sum(a)
    if S < 1e-10:
        return a, np.inf
    a = a / S
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    stall = 0

    for it in range(max_iters):
        if time.monotonic() - start > time_budget:
            break
        conv = compute_conv(a)
        M = float(np.max(conv))
        g = solve_lp(a, M * 1.0005, time_limit=4)
        if g is None:
            break
        S_g = np.sum(g)
        if S_g < 1e-10:
            break
        g = g / S_g
        best_t = 0.0
        best_obj_ls = best_obj
        for t in [0.05, 0.15, 0.3, 0.5, 0.7, 0.9, 1.0]:
            a_try = (1.0 - t) * a + t * g
            a_try = np.maximum(a_try, 0)
            S_try = np.sum(a_try)
            if S_try < 1e-10:
                continue
            a_try = a_try / S_try
            obj_try = compute_obj(a_try)
            if obj_try < best_obj_ls:
                best_obj_ls = obj_try
                best_t = t
        if best_t > 0.0:
            a = (1.0 - best_t) * a + best_t * g
            a = np.maximum(a, 0)
            a = a / np.sum(a)
            best_obj = best_obj_ls
            best_a = a.copy()
            stall = 0
        else:
            stall += 1
            if stall > 12:
                break

    return best_a, best_obj


def cosine_init(n, n_modes=10, seed=0):
    rng = np.random.default_rng(seed)
    k = np.arange(n)
    a = np.ones(n)
    for m in range(1, n_modes + 1):
        c = rng.uniform(-0.7, 0.7) / np.sqrt(m)
        phase = rng.uniform(0, 2 * np.pi)
        a += c * np.cos(2 * np.pi * m * k / n + phase)
    a = np.maximum(a, 0)
    S = np.sum(a)
    if S < 1e-10:
        return np.ones(n) / n
    return a / S


def tapered_init(n, alpha=1.5, seed=0):
    """Raised-cosine taper with random modulation."""
    rng = np.random.default_rng(seed)
    k = np.arange(n)
    x = 2.0 * k / (n - 1) - 1.0
    base = 0.5 * (1.0 + np.cos(np.pi * x)) ** alpha
    freq = rng.uniform(1, 4)
    a = base * (1.0 + 0.3 * np.cos(2 * np.pi * freq * k / n + rng.uniform(0, 2 * np.pi)))
    a = np.maximum(a, 0)
    S = np.sum(a)
    if S < 1e-10:
        return np.ones(n) / n
    return a / S


def random_smooth_init(n, seed=0):
    """Random smooth initialization via low-pass filtered noise."""
    rng = np.random.default_rng(seed)
    a = rng.uniform(0.5, 1.5, n)
    # Smooth with a window
    w = np.hanning(min(20, n))
    a = np.convolve(a, w, mode='same')
    a = np.maximum(a, 0.01)
    S = np.sum(a)
    return a / S

            break
        n_cand = len(a_cand)
        alloc = min(remaining * 0.2, 80.0)
        print(f"  Refining cand {idx} (n={n_cand}, obj={obj_cand:.8f}, alloc={alloc:.0f}s)")
        a_ref, obj_ref = refine_sequence(a_cand, time_budget=alloc)
        if obj_ref < best_score:
            best_score = obj_ref
            best_seq = a_ref.copy()
            print(f"    ** NEW BEST: n={n_cand}, obj={best_score:.8f} **")

    # Phase 3: Intensive refinement loop
    print("\n=== Phase 3: Intensive refinement ===")
    a_cur = best_seq.copy()
    for round_num in range(20):
        if time.monotonic() > deadline - 60:
            break
        remaining = deadline - time.monotonic()
        if remaining < 30:
            break
        alloc = min(remaining * 0.25, 60.0)
        a_new, obj_new = refine_sequence(a_cur, time_budget=alloc)
        if obj_new < best_score:
            best_score = obj_new
            best_seq = a_new.copy()
            a_cur = a_new.copy()
            print(f"  R{round_num}: {best_score:.8f} ** NEW BEST **")
        else:
            print(f"  R{round_num}: {obj_new:.8f} (best={best_score:.8f})")
            # Stronger perturbation
            rng = np.random.default_rng(round_num * 777 + 333)
            n_cur = len(a_cur)
            for ps in [0.1, 0.15, 0.25]:
                if time.monotonic() > deadline - 30:
                    break
                a_p = a_cur * np.exp(rng.normal(0, ps, n_cur))
                a_p = np.maximum(a_p, 0)
                S_p = np.sum(a_p)
                if S_p < 1e-10:
                    continue
                a_p = a_p / S_p
                a_p, obj_p = coordinate_descent(a_p, n_sweeps=5, time_budget=min(15.0, remaining * 0.1), max_inner=30)
                if obj_p < best_score:
                    best_score = obj_p
                    best_seq = a_p.copy()
                    a_cur = a_p.copy()
                    print(f"  R{round_num} pert({ps}): {best_score:.8f} ** NEW BEST **")

    # Phase 4: Try different lengths
    print("\n=== Phase 4: Multi-length exploration ===")
    if time.monotonic() < deadline - 60:
        n_base = len(best_seq)
        n_tries = sorted(set([
            max(100, n_base - 100), max(100, n_base - 50),
            n_base + 50, n_base + 100, n_base + 150,
            max(100, n_base + 300)
        ]))
        n_tries = [n for n in n_tries if 100 <= n <= 1500 and n != n_base]

        for n_try in n_tries:
            if time.monotonic() > deadline - 30:
                break
            remaining = deadline - time.monotonic()
            if remaining < 20:
                break
            # Interpolate
            x_old = np.linspace(0, 1, n_base)
            x_new = np.linspace(0, 1, n_try)
            from scipy.interpolate import interp1d
            try:
                f_interp = interp1d(x_old, best_seq, kind='cubic', fill_value='extrapolate')
                a_interp = np.maximum(f_interp(x_new), 0)
            except Exception:
                a_interp = cosine_init(n_try, n_modes=8, seed=seed + n_try)
            S_i = np.sum(a_interp)
            if S_i > 1e-10:
                a_interp = a_interp / S_i
            alloc = min(remaining * 0.3, 40.0)
            a_new, obj_new = refine_sequence(a_interp, time_budget=alloc)
            if obj_new < best_score:
                best_score = obj_new
                best_seq = a_new.copy()
                print(f"  n={n_try}: {best_score:.8f} ** NEW BEST **")
            else:
                print(f"  n={n_try}: {obj_new:.8f}")

    # Phase 5: Final coordinate descent polish
    print("\n=== Phase 5: Final polish ===")
    a_cur = best_seq.copy()
    n_final = len(a_cur)
    remaining = deadline - time.monotonic()
    if remaining > 10:
        print(f"  Final CD with {remaining:.0f}s budget")
        a_final, obj_final = coordinate_descent(a_cur, n_sweeps=20, time_budget=remaining * 0.6, max_inner=50)
        if obj_final < best_score:
            best_score = obj_final
            best_seq = a_final.copy()
            a_cur = a_final.copy()
            print(f"  CD: {best_score:.8f} ** NEW BEST **")

        remaining = deadline - time.monotonic()
        if remaining > 10:
            a_sg, obj_sg = softmax_subgradient(a_cur, time_budget=remaining * 0.3, init_temp=0.02)
            if obj_sg < best_score:
                best_score = obj_sg
                best_seq = a_sg.copy()
                a_cur = a_sg.copy()
                print(f"  SG: {best_score:.8f} ** NEW BEST **")

        remaining = deadline - time.monotonic()
        if remaining > 10 and n_final <= 600:
            a_lp, obj_lp = alternating_lp(a_cur, max_iters=30, time_budget=remaining * 0.4)
            if obj_lp < best_score:
                best_score = obj_lp
                best_seq = a_lp.copy()
                print(f"  LP: {best_score:.8f} ** NEW BEST **")

        # One final CD
        remaining = deadline - time.monotonic()
        if remaining > 10:
            a_final2, obj_final2 = coordinate_descent(best_seq, n_sweeps=10, time_budget=remaining * 0.8, max_inner=50)
            if obj_final2 < best_score:
                best_score = obj_final2
                best_seq = a_final2.copy()
                print(f"  Final CD2: {best_score:.8f} ** NEW BEST **")

    # Validate
    if best_seq is None:
        best_seq = np.ones(300) / 300
        best_score = compute_obj(best_seq)

    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    val = evaluate_sequence(result)
    if val < np.inf and val < best_score:
        best_score = val
    elif val == np.inf:
        result = [1.0] * 300

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_score:.8f}, time={elapsed:.1f}s ===")
    return result
