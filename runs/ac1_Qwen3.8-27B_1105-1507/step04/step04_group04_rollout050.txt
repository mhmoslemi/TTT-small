Optimization of step function sequences using smooth-max L-BFGS in log-space.
Key innovation: parameterize a = softmax(x) and use smooth approximation of max
with adaptive temperature, solved via L-BFGS-B. This provides a smooth landscape
that handles the non-differentiability of the max operator gracefully.

import numpy as np
import time
from scipy.optimize import minimize

def fft_convolve(a, b):
    """FFT-based convolution."""
    n = len(a)
    m = len(b)
    size = 1
    while size < n + m - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    conv = np.fft.irfft(fa * fb, size)
    return conv[:n + m - 1]

def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = np.sum(a)
    if s < 0.01:
        return np.inf
    conv = fft_convolve(a, a)
    return 2.0 * n * np.max(conv) / (s * s)

def smooth_max_and_weights(conv, tau):

        weights[idx] = 1.0
        return c_max, weights

    smooth_val = c_max + np.log(exp_sum) / tau

    """Optimize in log-space with adaptive tau."""
    start = time.monotonic()

        elapsed = time.monotonic() - start
        # Adaptive tau: interpolate from start to end over time_budget

        return obj, grad

def alternating_lp_step(a, time_limit=5.0):

    a = a / np.sum(a)

    A = np.zeros((2 * n - 1, n))

        for k in range(max(0, j), min(2*n-1, j+n)):
            A[k, j] = a[k-j]

            s = np.sum(g)
            if s > 1e-10:
                return g / s
    except Exception:
        pass
    return None

        if remaining < 8:
            break

        stage_time = min(remaining * 0.5, 50.0)
        max_iter = max(50, int(stage_time * 30))

        def closure(xx):
            return obj_and_grad_logspace(xx, tau, n)

        try:
            result = minimize(closure, x, jac=True, method='L-BFGS-B',
                              options={'maxiter': max_iter, 'maxfun': max_iter * 2,
                                       'ftol': 1e-14, 'gtol': 1e-11,
                                       'maxls': 50})
            x_new = result.x
            # Verify improvement
            a_new = np.exp(x_new - np.max(x_new))
            a_new = a_new / np.sum(a_new)
            obj_new = compute_obj(a_new)
            a_old_obj = compute_obj(a)
            if obj_new <= a_old_obj * 1.001:
                x = x_new
                a = a_new
        except Exception:
            pass

    a_final = np.exp(x - np.max(x))
    a_final = np.maximum(a_final, 1e-15)
    a_final = a_final / np.sum(a_final)
    return a_final, compute_obj(a_final)


def multiplicative_subgradient(a, tau=50.0, step=0.05, max_iters=3000, time_budget=15.0):
    """Multiplicative subgradient descent with adaptive parameters."""
    n = len(a)
    a = a.copy().astype(np.float64)
    a = np.maximum(a, 1e-15)
    a = a / np.sum(a)
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    stall = 0
    patience = 60

    for it in range(max_iters):
        if time.monotonic() - start > time_budget:
            break

        conv = fft_convolve(a, a)
        c_max = np.max(conv)
        shifted = np.clip(tau * (conv - c_max), -500, 500)
        weights = np.exp(shifted)
        w_sum = np.sum(weights)
        if w_sum < 1e-300:
            idx = int(np.argmax(conv))
            weights = np.zeros(len(conv))
            weights[idx] = 1.0
        else:
            weights /= w_sum

        grad_a = np.correlate(weights, a, mode='full')[:n]
        grad_centered = grad_a - np.mean(grad_a)

        a_new = a * np.exp(-step * grad_centered)
        a_new = np.maximum(a_new, 1e-15)
        a_new = a_new / np.sum(a_new)
        a = a_new

        obj = compute_obj(a)
        if obj < best_obj - 1e-13:
            best_obj = obj
            best_a = a.copy()
            stall = 0
        else:
            stall += 1
            if stall > patience:
                step *= 0.7
                tau *= 1.3
                if step < 0.001:
                    break
                rng = np.random.default_rng(it * 31 + 7)
                a = a * np.exp(rng.normal(0, 0.04, n))
                a = np.maximum(a, 1e-10)
                a = a / np.sum(a)
                stall = 0
                patience = max(patience // 2, 25)

        if it > 100:
            step *= 0.998
            tau *= 1.003

    return best_a


def alternating_lp(a, time_limit=4.0):
    """One alternating LP step: find g maximizing sum(g) s.t. A@g <= M."""
    n = len(a)
    a = a / np.sum(a)
    conv = fft_convolve(a, a)
    M = float(np.max(conv))

    # Build constraint matrix
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        k_end = min(j + n, 2 * n - 1)
        A[j:k_end, j] = a[:k_end - j]

    c = -np.ones(n)
    try:
        res = linprog(c, A_ub=A, b_ub=np.full(2 * n - 1, M),
                      bounds=[(0.0, None)] * n, method='highs',
                      options={'presolve': True, 'time_limit': time_limit})
        if res.status == 0:
            g = np.maximum(res.x, 0)
            s = np.sum(g)
            if s > 1e-10:
                return g / s
    except Exception:
        pass
    return None


def alternating_lp_optimize(a, max_iters=30, time_budget=30.0):
    """Alternating LP with line search."""
    n = len(a)
    a = a.copy().astype(np.float64)
    a = np.maximum(a, 1e-15)
    a = a / np.sum(a)
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    stall = 0

    for it in range(max_iters):
        if time.monotonic() - start > time_budget - 5:
            break

        g = alternating_lp(a, time_limit=3.0)
        if g is None:
            break

        best_t = 0.0
        best_obj_ls = best_obj
        for t in [0.3, 0.5, 0.7, 1.0]:
            a_new = (1.0 - t) * a + t * g
            a_new = np.maximum(a_new, 0)
            s_new = np.sum(a_new)
            if s_new < 1e-10:
                continue
            a_new = a_new / s_new
            obj_new = compute_obj(a_new)
            if obj_new < best_obj_ls:
                best_obj_ls = obj_new
                best_t = t

        if best_t > 0 and best_obj_ls < best_obj - 1e-13:
            a_new = (1.0 - best_t) * a + best_t * g
            a_new = np.maximum(a_new, 0)
            a_new = a_new / np.sum(a_new)
            best_obj = best_obj_ls
            best_a = a_new.copy()
            a = a_new
            stall = 0
        else:
            stall += 1
            if stall > 15:
                break
            a = 0.9 * a + 0.1 * g
            a = np.maximum(a, 0)
            a = a / np.sum(a)

    return best_a, best_obj


def local_search(a, time_budget=40.0, seed_offset=0):
    """Local search: perturb + re-optimize with continuation method."""
    n = len(a)
    a = a.copy().astype(np.float64)
    a = np.maximum(a, 1e-15)
    a = a / np.sum(a)
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    attempt = 0

    while time.monotonic() - start < time_budget - 10:
        attempt += 1
        remaining = time_budget - (time.monotonic() - start)
        if remaining < 15:
            break

        rng = np.random.default_rng(attempt * 7919 + seed_offset + 42)

        # Vary perturbation scale
        scales = [0.02, 0.04, 0.06, 0.03, 0.05]
        scale = scales[attempt % len(scales)]

        for sub in range(2):
            if time.monotonic() - start > time_budget - 10:
                break

            a_pert = best_a * np.exp(rng.normal(0, scale, n))
            a_pert = np.maximum(a_pert, 1e-15)
            a_pert = a_pert / np.sum(a_pert)

            sub_budget = min(remaining * 0.35, 25.0)
            a_opt, obj = optimize_continuation(a_pert,
                                               tau_values=[50, 200, 500],
                                               time_budget=sub_budget)
            if obj < best_obj:
                best_obj = obj
                best_a = a_opt.copy()

    return best_a, best_obj


def random_init(n, seed):
    """Generate diverse random initializations."""
    rng = np.random.default_rng(seed)
    k = np.arange(n)
    choice = rng.integers(0, 8)

    if choice == 0:
        f = rng.uniform(0.5, 4.0)
        phase = rng.uniform(0, 2 * np.pi)
        a = 1.0 + 0.8 * np.cos(2 * np.pi * f * k / n + phase)
    elif choice == 1:
        f1 = rng.uniform(0.3, 2.0)
        f2 = rng.uniform(2.0, 6.0)
        a = 1.0 + 0.4 * np.cos(2 * np.pi * f1 * k / n) + 0.3 * np.cos(2 * np.pi * f2 * k / n)
    elif choice == 2:
        center = n * rng.uniform(0.3, 0.7)
        alpha = rng.uniform(0.5, 3.0)
        a = 1.0 + 0.5 * np.exp(-alpha * np.abs(k - center) / (n * 0.3))
    elif choice == 3:
        a = rng.random(n)
        win = max(3, n // 20)
        kernel = np.ones(win) / win
        a = np.convolve(a, kernel, mode='same')
    elif choice == 4:
        a = np.ones(n)
        for _ in range(rng.integers(1, 4)):
            f = rng.uniform(0.5, 5.0)
            amp = rng.uniform(0.1, 0.4)
            a += amp * np.cos(2 * np.pi * f * k / n + rng.uniform(0, 2 * np.pi))
    elif choice == 5:
        a = np.ones(n)
        for scale in range(3):
            f = rng.uniform(0.3 * (2 ** scale), 1.5 * (2 ** scale))
            amp = 0.3 / (scale + 1)
            a += amp * np.cos(2 * np.pi * f * k / n + rng.uniform(0, 2 * np.pi))
    elif choice == 6:
        # Tapered window (Hann-like)
        a = 0.5 + 0.5 * np.cos(2 * np.pi * k / (n - 1))
        a = np.maximum(a, 0.01)
    else:
        a = rng.random(max(5, n // 10))
        a = np.interp(k, np.arange(len(a)), a)

    a = np.maximum(a, 1e-6)
    a = a / np.sum(a)
    return a


def resample_sequence(a, new_n):
    """Resample sequence to new length via interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    old_k = np.arange(n)
    new_k = np.linspace(0, n - 1, new_n)
    a_new = np.interp(new_k, old_k, a)
    a_new = np.maximum(a_new, 0)
    s = np.sum(a_new)
    if s > 1e-10:
        a_new = a_new / s
    return a_new


def propose_candidate(seed=42, budget_s=1500):
    """
    Main search function. Uses smooth-max L-BFGS with continuation as the
    primary optimizer, complemented by subgradient, alternating LP, and
    local search. Explores multiple sequence lengths around the parent.
    """
    start_time = time.monotonic()
    deadline = start_time + float(budget_s) - 30.0

    best_seq = None
    best_score = np.inf

    # Load parent
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64)
        parent = np.clip(parent, 0, 1000.0)
        if parent.size > 0 and np.isfinite(parent).all():
            parent_obj = compute_obj(parent)
            best_score = parent_obj
            best_seq = parent.copy()
            print(f"Parent (n={parent.size}): obj={parent_obj:.8f}")
    except Exception:
        pass

    if best_seq is None:
        n = 300
        best_seq = random_init(n, seed)
        best_score = compute_obj(best_seq)
        print(f"Random init (n=300): obj={best_score:.8f}")

    n_parent = len(best_seq)
    print(f"Starting optimization. Budget: {budget_s}s")

    # === Phase 1: Continuation optimization of parent ===
    print(f"\n=== Phase 1: Continuation L-BFGS on parent (n={n_parent}) ===")
    remaining = deadline - time.monotonic()
    if remaining > 60:
        p1_budget = min(remaining * 0.3, 350.0)
        print(f"  Budget: {p1_budget:.0f}s")
        a_opt, obj = optimize_continuation(best_seq,
                                           tau_values=[10, 30, 100, 300, 1000, 3000],
                                           time_budget=p1_budget)
        if obj < best_score:
            best_score = obj
            best_seq = a_opt.copy()
            print(f"  After continuation: {best_score:.8f}")

    # === Phase 2: Subgradient polish ===
    remaining = deadline - time.monotonic()
    if remaining > 30:
        print(f"\n=== Phase 2: Subgradient polish ===")
        p2_budget = min(remaining * 0.12, 80.0)
        a_opt = multiplicative_subgradient(best_seq, tau=80, step=0.02,
                                           max_iters=5000, time_budget=p2_budget)
        obj = compute_obj(a_opt)
        if obj < best_score:
            best_score = obj
            best_seq = a_opt.copy()
            print(f"  After subgradient: {best_score:.8f}")

    # === Phase 3: Alternating LP ===
    remaining = deadline - time.monotonic()
    if remaining > 40:
        print(f"\n=== Phase 3: Alternating LP ===")
        p3_budget = min(remaining * 0.15, 100.0)
        a_opt, obj = alternating_lp_optimize(best_seq, max_iters=25,
                                             time_budget=p3_budget)
        if obj < best_score:
            best_score = obj
            best_seq = a_opt.copy()
            print(f"  After LP: {best_score:.8f}")

    # === Phase 4: Local search on current best ===
    remaining = deadline - time.monotonic()
    if remaining > 40:
        print(f"\n=== Phase 4: Local search ===")
        p4_budget = min(remaining * 0.2, 200.0)
        a_opt, obj = local_search(best_seq, time_budget=p4_budget,
                                  seed_offset=seed)
        if obj < best_score:
            best_score = obj
            best_seq = a_opt.copy()
            print(f"  After local search: {best_score:.8f}")

    # === Phase 5: Try different n values ===
    print(f"\n=== Phase 5: Explore different n values ===")
    remaining = deadline - time.monotonic()
    if remaining > 50:
        n_values = []
        for delta in [-15, -10, -5, 5, 10, 15, 20, 25, 30, 40, 50, 60, 80, 100, -30, -20]:
            nv = n_parent + delta
            if 150 <= nv <= 800 and nv != n_parent:
                n_values.append(nv)
        for nv in [250, 350, 400, 500]:
            if nv not in n_values and nv != n_parent:
                n_values.append(nv)
        n_values = sorted(set(n_values))

        # Prioritize n values close to parent
        n_values.sort(key=lambda nv: abs(nv - n_parent))

        per_n = max(15.0, remaining * 0.6 / max(len(n_values), 1))

        for nv in n_values:
            if time.monotonic() > deadline - 30:
                break
            remaining = deadline - time.monotonic()
            if remaining < 25:
                break
            alloc = min(per_n, remaining * 0.25, 60.0)
            alloc = max(alloc, 12.0)

            # Resample parent to new n
            a_res = resample_sequence(best_seq, nv)
            a_opt, obj = optimize_continuation(a_res,
                                               tau_values=[30, 100, 500],
                                               time_budget=alloc)
            if obj < best_score:
                best_score = obj
                best_seq = a_opt.copy()
                print(f"  ** UPDATE: n={nv}, obj={best_score:.8f} (remaining={remaining:.0f}s)")
            else:
                print(f"  n={nv}: {obj:.8f} (no improvement)")

    # === Phase 6: Final intensive optimization ===
    remaining = deadline - time.monotonic()
    if remaining > 30 and best_seq is not None:
        print(f"\n=== Phase 6: Final optimization (n={len(best_seq)}) ===")

        # Continuation with very fine tau schedule
        p6_budget = min(remaining * 0.35, 120.0)
        a_opt, obj = optimize_continuation(best_seq,
                                           tau_values=[50, 200, 800, 3000, 10000],
                                           time_budget=p6_budget)
        if obj < best_score:
            best_score = obj
            best_seq = a_opt.copy()
            print(f"  After final continuation: {best_score:.8f}")

        # Subgradient with small steps
        remaining = deadline - time.monotonic()
        if remaining > 20:
            p6b_budget = min(remaining * 0.25, remaining - 15.0)
            a_opt = multiplicative_subgradient(best_seq, tau=200, step=0.01,
                                               max_iters=5000, time_budget=p6b_budget)
            obj = compute_obj(a_opt)
            if obj < best_score:
                best_score = obj
                best_seq = a_opt.copy()
                print(f"  After final subgradient: {best_score:.8f}")

        # Final local search
        remaining = deadline - time.monotonic()
        if remaining > 20:
            p6c_budget = min(remaining * 0.3, remaining - 10.0)
            a_opt, obj = local_search(best_seq, time_budget=p6c_budget,
                                      seed_offset=seed + 9999)
            if obj < best_score:
                best_score = obj
                best_seq = a_opt.copy()
                print(f"  After final local search: {best_score:.8f}")

    # === Final validation ===
    if best_seq is None:
        best_seq = np.ones(50)
        best_score = compute_obj(best_seq)

    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]

    # Validate with the actual evaluator
    val = evaluate_sequence(result)
    if val < best_score:
        best_score = val

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(best_seq)}, score={best_score:.8f}, time={elapsed:.1f}s ===")

    return result
