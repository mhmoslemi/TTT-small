```python
"""
Minimizes J(a) = 2*n*max(convolve(a,a))/(sum(a))² using:
1. Fast LP mirror descent with adaptive step size
2. L-BFGS-B with log-sum-exp smoothing for polishing
3. Multi-length exploration (256, 512, 768, 1024, 1536)
4. Diverse restarts with perturbations
5. Dual-LP analysis for convergence detection
"""

import numpy as np
from scipy.optimize import linprog, minimize
from scipy.sparse import csr_matrix
import time

# Cache for pre-computed structures
_structure_cache = {}

def fast_conv(a):
    """Full self-convolution via FFT."""
    n = len(a)
    s = 1
    while s < 2 * n - 1:
        s <<= 1
    fa = np.fft.rfft(a, s)
    return np.fft.irfft(fa * fa, s)[:2 * n - 1]

def compute_J(a):
    """Compute J(a) = 2*n*max(conv(a,a))/(sum(a))^2."""
    a = np.maximum(a, 0)
    s = float(a.sum())
    if s < 1e-10:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))

def get_structure(n):
    """Get (row, col) arrays for the convolution constraint matrix."""
    if n not in _structure_cache:
        rows = []
        cols = []
        for k in range(2 * n - 1):
            jl = max(0, k - n + 1)
            jh = min(n - 1, k)
            for j in range(jl, jh + 1):
                rows.append(k)
                cols.append(j)
        _structure_cache[n] = (np.array(rows, dtype=np.int32),
                                np.array(cols, dtype=np.int32))
    return _structure_cache[n]

def build_A(a):
    """Build sparse constraint matrix for LP."""
    n = len(a)
    rows_arr, cols_arr = get_structure(n)
    # A[k,j] = a[k-j] for valid (k,j)
    vals = a[rows_arr - cols_arr]
    return csr_matrix((vals, (rows_arr, cols_arr)), shape=(2 * n - 1, n))

def lp_best_response(a, M, n):
    """Solve LP: max sum(g) s.t. (a*g)_k <= M, g >= 0. Returns normalized g."""
    A = build_A(a)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    try:
        r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                    options={'presolve': True, 'dual_feasibility_tolerance': 1e-9})
        if r.success and r.x is not None:
            g = np.maximum(r.x, 0.0)
            g_sum = float(g.sum())
            if g_sum > 1e-10:
                return g / g_sum
    except Exception:
        pass
    return None

def fast_line_search(a, g, current_J):
    """
    Fast line search: try a few t values and pick the best.
    Uses the structure that J is a ratio of quadratic forms.
    """
    candidates_t = [0.1, 0.2, 0.3, 0.5, 0.7, 1.0]
    best_J = current_J
    best_t = 0.0
    best_a = a.copy()
    
    for t in candidates_t:
        a_try = (1.0 - t) * a + t * g
        a_try = np.maximum(a_try, 0.0)
        s = float(a_try.sum())
        if s < 1e-10:
            continue
        a_try = a_try / s
        J_try = compute_J(a_try)
        if J_try < best_J:
            best_J = J_try
            best_t = t
            best_a = a_try.copy()
    
    # If t=1.0 was best, try even further (though we can't go beyond g)
    # If no improvement, return original
    return best_a, best_J, best_t

def lp_mirror_descent_fast(a, deadline, tracker, max_iters=10000, stall_patience=50):
    """
    Fast LP mirror descent with adaptive step size.
    """
    a = np.maximum(a.astype(np.float64), 1e-8)
    a = a / a.sum()
    n = len(a)
    current_J = compute_J(a)
    best_a = a.copy()
    best_J = current_J
    stall_count = 0
    
    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        
        # Compute current convolution and max
        b = fast_conv(a)
        M = float(b.max())
        
        # Solve LP
        g = lp_best_response(a, M, n)
        if g is None:
            break
        
        # Check if g is similar to a (convergence)
        sim = float(np.dot(a, g))  # Cosine-like similarity (both normalized)
        if sim > 0.9999:
            stall_count += 1
            if stall_count >= stall_patience:
                break
            continue
        
        # Line search
        a_new, J_new, t_used = fast_line_search(a, g, current_J)
        
        if J_new < current_J - 1e-14:
            a = a_new
            current_J = J_new
            stall_count = 0
            if J_new < best_J:
                best_J = J_new
                best_a = a.copy()
                tracker.update(a)
        else:
            # Try smaller steps
            improved = False
            for t_try in [0.05, 0.02, 0.01]:
                a_try = (1.0 - t_try) * a + t_try * g
                a_try = np.maximum(a_try, 0.0)
                s = float(a_try.sum())
                if s < 1e-10:
                    continue
                a_try = a_try / s
                J_try = compute_J(a_try)
                if J_try < current_J - 1e-15:
                    a = a_try
                    current_J = J_try
                    stall_count = 0
                    if J_try < best_J:
                        best_J = J_try
                        best_a = a.copy()
                        tracker.update(a)
                    improved = True
                    break
            if not improved:
                stall_count += 1
                if stall_count >= stall_patience:
                    break
    
    tracker.update(best_a)
    return best_a
```

"""
Minimizes J(a) = 2*n*max(convolve(a,a))/(sum(a))^2 using:
1. Fast LP mirror descent with adaptive step-size (core refinement)
2. L-BFGS-B with log-sum-exp smoothed objective for polishing
3. Multi-length exploration (256, 512, 768, 1024, 1536)
4. Bisection-on-M strategy to escape LP MD plateaus
5. Diverse restarts with structured perturbations
6. Global best tracking with validity checks via evaluate_sequence

import numpy as np
from scipy.optimize import linprog, minimize
from scipy.sparse import csr_matrix
import time

# Global cache for convolution matrix structures
_structure_cache = {}


def _fast_conv(a):
    """Full self-convolution via FFT, O(n log n)."""
    n = len(a)
    if n == 1:
        return np.array([a[0] * a[0]])
    s = 1
    while s < 2 * n - 1:
        s <<= 1
    fa = np.fft.rfft(a, s)
    return np.fft.irfft(fa * fa, s)[:2 * n - 1]


def _compute_J(a):
    """Compute J(a) = 2*n*max(conv(a,a))/(sum(a))^2. Assumes a >= 0."""
    a = np.maximum(a, 0.0)
    s = float(a.sum())
    if s < 1e-10:
        return float('inf')
    b = _fast_conv(a)
    return float(2.0 * len(a) * b.max() / (s * s))


def _get_structure(n):
    """Get or precompute (row, col) index arrays for convolution matrix."""
    if n not in _structure_cache:
        rows = []
        cols = []
        for k in range(2 * n - 1):
            jl = max(0, k - n + 1)
            jh = min(n - 1, k)
            for j in range(jl, jh + 1):
                rows.append(k)
                cols.append(j)
        _structure_cache[n] = (np.array(rows, dtype=np.int32),
                               np.array(cols, dtype=np.int32))
    return _structure_cache[n]


def _build_A(a):
    """Build sparse A matrix where (A*g)_k = (a*g)_k (convolution)."""
    n = len(a)
    rows_arr, cols_arr = _get_structure(n)
    vals = a[rows_arr - cols_arr]
    return csr_matrix((vals, (rows_arr, cols_arr)), shape=(2 * n - 1, n))


def _lp_best_response(a, M, n):
    """
    Solve LP: max sum(g) s.t. (a*g)_k <= M for all k, g >= 0.
    Returns g normalized to sum=1, or None if infeasible/degenerate.
    """
    A = _build_A(a)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    try:
        r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                    options={'presolve': True, 'time_limit': 10.0})
        if r.success and r.x is not None:
            g = np.maximum(r.x, 0.0)
            g_sum = float(g.sum())
            if g_sum > 1e-10:
                return g / g_sum
    except Exception:
        pass
    return None


def _fast_line_search(a, g, current_J):
    """
    Fast line search: evaluate J at a set of candidate t values.
    Returns (best_a, best_J, best_t).
    """
    t_candidates = [0.1, 0.2, 0.3, 0.5, 0.7, 1.0]
    best_J = current_J
    best_t = 0.0
    best_a = None

    for t in t_candidates:
        a_try = (1.0 - t) * a + t * g
        a_try = np.maximum(a_try, 0.0)
        s = float(a_try.sum())
        if s < 1e-10:
            continue
        a_try = a_try / s
        J_try = _compute_J(a_try)
        if J_try < best_J - 1e-15:
            best_J = J_try
            best_t = t
            best_a = a_try.copy()

    if best_a is None:
        return a, current_J, 0.0
    return best_a, best_J, best_t


def _lp_mirror_descent(a, deadline, tracker, max_iters=20000, stall_patience=60):
    """
    LP-based mirror descent with adaptive step and bisection-on-M escape.
    Normalizes a to sum=1 throughout.
    """
    a = np.maximum(a.astype(np.float64), 1e-8)
    a = a / a.sum()
    n = len(a)
    current_J = _compute_J(a)
    best_a = a.copy()
    best_J = current_J
    stall_count = 0
    m_reduction = 1.0  # Factor applied to M for escape

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        b = _fast_conv(a)
        M = float(b.max()) * m_reduction

        # Solve LP for best response
        g = _lp_best_response(a, M, n)
        if g is None:
            break

        # Check similarity (convergence detection)
        sim = float(np.dot(a, g))
        if sim > 0.99995:
            # Near fixed point: try reducing M to escape
            m_reduction = max(0.5, m_reduction * 0.9)
            stall_count += 1
            if stall_count >= stall_patience:
                break
            continue

        # Line search
        a_new, J_new, t_used = _fast_line_search(a, g, current_J)

        if J_new < current_J - 1e-14:
            a = a_new
            current_J = J_new
            stall_count = 0
            m_reduction = min(1.0, m_reduction + 0.02)  # Gradually restore M
            if J_new < best_J:
                best_J = J_new
                best_a = a.copy()
                tracker.update(a)
        else:
            # Try smaller steps
            improved = False
            for t_try in [0.05, 0.02, 0.01, 0.005]:
                a_try = (1.0 - t_try) * a + t_try * g
                a_try = np.maximum(a_try, 0.0)
                s = float(a_try.sum())
                if s < 1e-10:
                    continue
                a_try = a_try / s
                J_try = _compute_J(a_try)
                if J_try < current_J - 1e-15:
                    a = a_try
                    current_J = J_try
                    stall_count = 0
                    m_reduction = min(1.0, m_reduction + 0.01)
                    if J_try < best_J:
                        best_J = J_try
                        best_a = a.copy()
                        tracker.update(a)
                    improved = True
                    break
            if not improved:
                stall_count += 1
                m_reduction = max(0.5, m_reduction * 0.95)
                if stall_count >= stall_patience:
                    break

    tracker.update(best_a)
    return best_a


def _lse_and_grad(a, beta):
    """
    Compute log-sum-exp of conv(a,a) and its gradient w.r.t. a.
    Returns (lse_value, gradient).
    """
    n = len(a)
    b = _fast_conv(a)
    b_max = float(b.max())
    shifted = beta * (b - b_max)
    shifted = np.clip(shifted, -500.0, 500.0)
    exp_shifted = np.exp(shifted)
    Z = float(exp_shifted.sum())
    if Z < 1e-300:
        p = np.zeros(2 * n - 1)
        p[np.argmax(b)] = 1.0
    else:
        p = exp_shifted / Z
    lse_val = (b_max + np.log(max(Z, 1e-300))) / beta

    # Gradient: grad[i] = 2 * sum_k p_k * a[k-i] (valid range)
    # Computed via: 2 * (convolve(a, p[::-1], 'full')[n-1:2n-1])[::-1]
    S = np.convolve(a, p[::-1], mode='full')
    S = S[n - 1:2 * n - 1][::-1]
    grad = 2.0 * S
    return lse_val, grad


def _F_and_grad(a, beta):
    """Compute smoothed objective F and its gradient."""
    n = len(a)
    s = float(a.sum())
    if s < 1e-10:
        return 1e10, np.zeros(n)
    lse_val, grad_lse = _lse_and_grad(a, beta)
    F_val = 2.0 * n * lse_val / (s * s)
    grad_F = 2.0 * n * (s * grad_lse - 2.0 * lse_val) / (s ** 3)
    return F_val, grad_F


def _lbfgs_polish(a0, deadline, beta_list=None):
    """L-BFGS-B optimization with beta annealing."""
    if beta_list is None:
        beta_list = [(2.0, 30), (5.0, 40), (10.0, 50), (20.0, 60), (50.0, 80)]

    a = np.maximum(a0.astype(np.float64), 1e-8)
    n = len(a)
    a = a / a.sum()

    for beta, max_iters in beta_list:
        if time.monotonic() > deadline:
            break

        def func(x):
            x2 = np.maximum(x, 0.0)
            f, _ = _F_and_grad(x2, beta)
            if not np.isfinite(f):
                return 1e10
            return f

        def jac(x):
            x2 = np.maximum(x, 0.0)
            if x2.sum() < 1e-10:
                return np.zeros_like(x)
            _, g = _F_and_grad(x2, beta)
            g = np.nan_to_num(g, nan=0.0, posinf=0.0, neginf=0.0)
            g[x2 < 1e-12] = 0.0
            return g

        bounds = [(0.0, None)] * n
        try:
            res = minimize(func, a.copy(), jac=jac, method='L-BFGS-B',
                           bounds=bounds,
                           options={'maxiter': max_iters,
                                    'max_fun': max_iters * 30,
                                    'gtol': 1e-9, 'ftol': 1e-15})
            if res.fun < 1e10:
                new_a = np.maximum(res.x, 1e-10)
                if new_a.sum() > 1e-6:
                    a = new_a
        except Exception:
            pass

    return a


def _resample(src, dst_n):
    """Resample 1D sequence to new length via linear interpolation."""
    src = np.asarray(src, dtype=np.float64)
    if src.size == dst_n:
        return src.copy()
    if src.size == 0:
        return np.ones(dst_n)
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dst_n)
    result = np.interp(t_new, t_old, src)
    return np.maximum(result, 0.0)


def _generate_inits(n, rng, parent):
    """Generate diverse initial sequences of length n."""
    inits = []
    t_arr = np.linspace(0, 1, n, endpoint=False)

    if parent is not None and parent.size > 0:
        inits.append(_resample(parent, n))
        p = _resample(parent, n)
        p = p + rng.normal(0, 0.05, n) * p
        inits.append(np.maximum(p, 0.01))
        p2 = _resample(parent, n)
        p2 = p2 + rng.normal(0, 0.12, n) * p2
        inits.append(np.maximum(p2, 0.01))

    # Uniform
    inits.append(np.ones(n))

    # Geometric (one-sided decay)
    r = 1.0 - 1.0 / (n + 1)
    inits.append(r ** np.arange(n, dtype=np.float64))

    # Reverse geometric
    inits.append((r ** np.arange(n, dtype=np.float64))[::-1])

    # Cosine families
    for freq in [2, 3, 5]:
        inits.append(1.0 + 0.4 * np.cos(2.0 * np.pi * freq * t_arr))

    # Asymmetric ramp with oscillation
    inits.append((1.0 - t_arr) + 0.2 * np.sin(4.0 * np.pi * t_arr))

    # Two-peak asymmetric
    inits.append(np.exp(-8.0 * (t_arr - 0.3) ** 2) + 0.5 * np.exp(-8.0 * (t_arr - 0.7) ** 2) + 0.05)

    # Exponential decay
    inits.append(np.exp(-2.0 * t_arr) + 0.1)

    return inits

        a = _lp_mirror_descent(tracker.best, deadline_p2, tracker,
                               max_iters=50000, stall_patience=80)
        tracker.report(f"LP MD primary: {time.monotonic()-t0:.1f}s")

        # Try at other lengths
        for n_try in [1024, 512, 768, 1536]:
            if time.monotonic() > deadline_p2:
                break
            if n_try == n_best:
                continue
            a_res = _resample(tracker.best, n_try)
            a_res = a_res / max(a_res.sum(), 1e-10)
            budget_here = min(120.0, deadline_p2 - time.monotonic())
            if budget_here < 10:
                break
            _lp_mirror_descent(a_res, deadline_p2, tracker,
                               max_iters=20000, stall_patience=50)

        tracker.report(f"Phase 2 total: {time.monotonic()-t0:.1f}s")

    # === Phase 3: Diverse restarts ===
    if tracker.best is not None and time.monotonic() < deadline:
        t0 = time.monotonic()
        phase3_budget = 350.0
        deadline_p3 = min(deadline, time.monotonic() + phase3_budget)
        print(f"\n{'='*50} Phase 3: Restarts (budget {phase3_budget:.0f}s) {'='*50}", flush=True)

        strategies = [
            ('gaussian', 0.06),
            ('gaussian', 0.12),
            ('gaussian', 0.20),
            ('uniform', 0.1),
            ('center_boost', 0.15),
            ('edge_trim', 0.1),
            ('spike', 5),
            ('smooth_noise', 0.1),
            ('reverse', 0.0),
            ('gaussian', 0.03),
            ('multiplicative', 0.15),
        ]

        attempt = 0
        for pert_type, param in strategies:
            for trial in range(4):
                if time.monotonic() > deadline_p3:
                    break
                attempt += 1
                a = tracker.best.copy().astype(np.float64)
                n = len(a)

                if pert_type == 'gaussian':
                    a = a + rng.normal(0, param, n) * a
                elif pert_type == 'uniform':
                    a = a + rng.uniform(-param, param, n) * a
                elif pert_type == 'center_boost':
                    center = n // 2
                    width = max(1, n // 8)
                    for i in range(max(0, center - width), min(n, center + width)):
                        a[i] *= (1.0 + param * rng.normal(0, 1))
                elif pert_type == 'edge_trim':
                    k = max(1, int(n * param))
                    a[:k] *= rng.uniform(0.5, 1.5, k)
                    a[-k:] *= rng.uniform(0.5, 1.5, k)
                elif pert_type == 'spike':
                    k = min(param, n // 8)
                    idx = rng.choice(n, size=k, replace=False)
                    a[idx] *= rng.uniform(0.3, 3.0, k)
                elif pert_type == 'smooth_noise':
                    noise = rng.normal(0, param, n)
                    kernel_size = max(3, n // 32)
                    kernel = np.ones(kernel_size) / kernel_size
                    noise = np.convolve(noise, kernel, mode='same')
                    a = a + noise * a
                elif pert_type == 'reverse':
                    # Blend with reversed sequence
                    t_mix = rng.uniform(0.3, 0.7)
                    a = t_mix * a + (1.0 - t_mix) * a[::-1]
                elif pert_type == 'multiplicative':
                    a = a * np.exp(rng.normal(0, param, n))

                a = np.maximum(a, 1e-4)
                a = a / a.sum()

                # Quick check
                J_pert = _compute_J(a)
                if J_pert < tracker.best_score:
                    tracker.update(a)

                # LP refinement (short)
                budget_here = min(30.0, deadline_p3 - time.monotonic())
                if budget_here < 5:
                    continue
                _lp_mirror_descent(a, deadline_p3, tracker,
                                   max_iters=8000, stall_patience=30)

        tracker.report(f"Phase 3: {time.monotonic()-t0:.1f}s, {attempt} attempts")

    # === Phase 4: Final polish ===
    if tracker.best is not None and time.monotonic() < deadline:
        t0 = time.monotonic()
        remaining = deadline - time.monotonic()
        if remaining > 20:
            print(f"\n{'='*50} Phase 4: Final Polish (budget {remaining:.0f}s) {'='*50}", flush=True)

            # L-BFGS polish on the smoothed objective
            n_best = tracker.best.size
            deadline_p4a = time.monotonic() + min(60.0, remaining * 0.3)
            a_polished = _lbfgs_polish(tracker.best, deadline_p4a,
                                        [(10.0, 40), (30.0, 50), (80.0, 60)])
            if a_polished is not None and a_polished.sum() > 0.01:
                tracker.update(a_polished)

            # Extended LP MD
            deadline_p4b = min(deadline, time.monotonic() + min(150.0, remaining * 0.5))
            _lp_mirror_descent(tracker.best, deadline_p4b, tracker,
                               max_iters=30000, stall_patience=100)

            # Very small perturbations + LP
            for _ in range(3):
                if time.monotonic() >= deadline:
                    break
                a = tracker.best.copy().astype(np.float64)
                n = len(a)
                a = a + rng.normal(0, 0.01, n) * a
                a = np.maximum(a, 1e-4)
                a = a / a.sum()
                budget_here = min(40.0, deadline - time.monotonic())
                if budget_here < 5:
                    break
                _lp_mirror_descent(a, time.monotonic() + budget_here, tracker,
                                   max_iters=10000, stall_patience=40)

            # Try adjacent lengths
            for n_try in [n_best - 128, n_best + 128, 768, 1280]:
                if n_try < 128 or n_try > 2048 or time.monotonic() > deadline:
                    continue
                if abs(n_try - n_best) < 64:
                    continue
                a_res = _resample(tracker.best, n_try)
                a_res = a_res / max(a_res.sum(), 1e-10)
                budget_here = min(30.0, deadline - time.monotonic())
                if budget_here < 5:
                    continue
                _lp_mirror_descent(a_res, time.monotonic() + budget_here, tracker,
                                   max_iters=5000, stall_patience=20)

            tracker.report(f"Phase 4: {time.monotonic()-t0:.1f}s")

    # Fallback
    if tracker.best is None:
        tracker.best = np.ones(512)
        tracker.best_score = _compute_J(tracker.best)
        print(f"  Using fallback: J={tracker.best_score:.8f}", flush=True)

    print(f"\n{'='*50} FINAL RESULT {'='*50}", flush=True)
    print(f"J={tracker.best_score:.8f} n={tracker.best.size} "
          f"time={time.monotonic()-tracker.t_start:.1f}s", flush=True)

    # Ensure valid output
    result = np.clip(tracker.best, 0.0, 1000.0)
    if result.size == 0 or result.sum() < 0.01:
        result = np.ones(128)

    return [float(x) for x in result]
