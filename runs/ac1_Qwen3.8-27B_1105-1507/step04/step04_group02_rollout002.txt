"""
Step-function coefficient search using bundle subgradient method, 
CVXPY-based LP alternation, and multi-scale spectral initialization.
Key improvements over previous: bundle method for better convergence,
proper line search, active set management, and systematic n exploration.
"""

import numpy as np
from scipy.optimize import linprog, minimize
from scipy.interpolate import interp1d
from scipy.sparse import coo_matrix, csr_matrix
import time

try:
    import cvxpy as cp
    HAS_CVXPY = True
except ImportError:
    HAS_CVXPY = False


def compute_conv(a):
    """Compute linear convolution a*a via FFT."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    conv = np.fft.irfft(fa * fa, size)
    return conv[:2 * n - 1]


def compute_conv_with_b(a, b):
    """Compute linear convolution a*b via FFT."""
    n = max(len(a), len(b))
    size = 1
    while size < len(a) + len(b) - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    conv = np.fft.irfft(fa * fb, size)
    return conv[:len(a) + len(b) - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    S = float(np.sum(a))
    if S < 1e-10:
        return np.inf
    conv = compute_conv(a)
    M = float(np.max(conv))
    return 2.0 * n * M / (S * S)


def build_toeplitz_sparse(a):
    """Build sparse matrix T such that T @ g = conv(a, g)."""
    n_a = len(a)
    n_g = n_a  # assuming same length
    rows = []
    cols = []
    vals = []
    for j in range(n_g):
        start = j
        end = min(j + n_a, n_a + n_g - 1)
        length = end - start
        rows.extend(range(start, end))
        cols.extend([j] * length)
        vals.extend(list(a[:length]))
    A = coo_matrix((vals, (rows, cols)), shape=(n_a + n_g - 1, n_g)).tocsr()
    return A


def solve_lp_cvxpy(a, M, time_limit=10):
    """Solve LP: max sum(g) s.t. conv(a,g) <= M, g >= 0 using CVXPY."""
    n = len(a)
    g = cp.Variable(n)
    A = build_toeplitz_sparse(a)
    
    constraints = [A @ g <= M, g >= 0]
    objective = cp.Maximize(cp.sum(g))
    prob = cp.Problem(objective, constraints)
    
    solvers_to_try = ['GLOP', 'GUROBI', 'MOSEK', 'SCIP', 'ECOS', 'CLARABEL']
    for solver in solvers_to_try:
        try:
            prob.solve(solver=solver, time_limit=time_limit)
            if g.value is not None:
                return np.maximum(g.value, 0)
        except Exception:
            continue
    return None


def solve_lp_scipy(a, M, time_limit=8):
    """Solve LP using scipy linprog with HiGHS."""
    n = len(a)
    A = build_toeplitz_sparse(a)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': time_limit})
        if res.status == 0:
            return np.maximum(res.x, 0)
    except Exception:
        pass
    return None

    """Solve LP with best available solver."""
    if HAS_CVXPY:
        result = solve_lp_cvxpy(a, M, time_limit)
        if result is not None:
            return result
    return solve_lp_scipy(a, M, time_limit)


def get_active_set(conv, threshold_frac=0.999):
    """Get indices where conv is close to its maximum."""
    M = np.max(conv)
    threshold = M * threshold_frac
    active = np.where(conv >= threshold)[0]
    return active, M


def bundle_subgradient(a, max_iters=200, time_budget=30.0, verbose=False):
    """
    Bundle method for minimizing max(conv(a,a))[k] subject to sum(a)=1, a>=0.
    Uses a collection of subgradients to build a cutting-plane model.
    """
    n = len(a)
    a = a.copy().astype(np.float64)
    S = np.sum(a)
    if S < 1e-10:
        return a
    a = a / S
    
    # Normalize: we work with sum(a) = 1, objective is max(conv(a,a))
    # Then final obj = 2n * max_conv
    
    best_a = a.copy()
    best_M = float(np.max(compute_conv(a)))
    start = time.monotonic()
    
    # Bundle: list of (a_k, conv_k, f_k)
    bundle = []
    
    #锹 Initialize with current point
    conv0 = compute_conv(a)
    M0 = float(np.max(conv0))
    bundle.append((a.copy(), conv0.copy(), M0))
    
    # We'll use a simpler approach: focus on the active set
    # and do projected subgradient with smart step size
    
    v = np.zeros(n)  # momentum
    momentum = 0.9
    base_step = 0.1
    stall = 0
    
    for it in range(max_iters):
        if time.monotonic() - start > time_budget:
            break
        
        # Compute subgradient from active set
        conv = compute_conv(a)
        M = float(np.max(conv))
        
        # Active set: top positions
        n_active = min(2 * n, max(5, int(0.1 * (2 * n - 1))))
        top_idx = np.argsort(conv)[-n_active:]
        
        # Subgradient of max_k conv[k] w.r.t. a
        # d conv[k] / da[i] = a[k-i] (if 0 <= k-i < n)
        # For max, we use the weighted average over active set
        weights = np.exp(conv[top_idx] / (0.01 + M * 0.01))  # softmax weights
        weights /= np.sum(weights)
        
        grad = np.zeros(n)
        for w, k in zip(weights, top_idx):
            for i in range(n):
                j = k - i
                if 0 <= j < n:
                    grad[i] += w * a[j]
        
        # Project gradient onto sum=0 hyperplane (to preserve sum)
        grad = grad - np.mean(grad)
        
        # Polyak step size: use gap between current M and known lower bound
        # Lower bound on M: by average, mean(conv) = sum(a)^2 / (2n-1) ≈ 1/(2n-1)
        # Actually mean(conv) = (1/(2n-1)) * sum_k conv[k] = (1/(2n-1)) * (sum a)^2 = 1/(2n-1)
        # So M >= 1/(2n-1)
        lower_bound = 1.0 / (2 * n - 1)
        gap = M - lower_bound
        if gap < 1e-10:
            break
        polyak_step = gap / (np.dot(grad, grad) + 1e-10)
        step = min(base_step, 0.5 * polyak_step)
        
        # Update with momentum
        v = momentum * v + (1 - momentum) * grad
        a_new = a - step * v
        a_new = np.maximum(a_new, 0)
        S_new = np.sum(a_new)
        if S_new < 1e-10:
            break
        a_new = a_new / S_new
        
        # Evaluate
        M_new = float(np.max(compute_conv(a_new)))
        
        if M_new < best_M - 1e-10:
            best_M = M_new
            best_a = a_new.copy()
            a = a_new
            stall = 0
        else:
            stall += 1
            if stall % 50 == 0:
                # Perturb
                rng = np.random.default_rng(it * 104729)
                a = a * np.exp(rng.normal(0, 0.05, n))
                a = np.maximum(a, 0)
                a = a / np.sum(a)
                v = np.zeros(n)
                base_step *= 0.8
                stall = 0
                if base_step < 1e-10:
                    break
    
    return best_a


def alternating_lp_refine(a, max_iters=50, time_budget=60.0, verbose=False):
    """
    Alternating LP with proper line search and adaptive M.
    """
    n = len(a)
    a = a.copy().astype(np.float64)
    S = np.sum(a)
    if S < 1e-10:
        return a
    a = a / S
    
    best_a = a.copy()
    best_M = float(np.max(compute_conv(a)))
    start = time.monotonic()
    stall = 0
    
    for it in range(max_iters):
        if time.monotonic() - start > time_budget:
            break
        
        conv = compute_conv(a)
        M = float(np.max(conv))
        
        # Solve LP: max sum(g) s.t. conv(a, g) <= M, g >= 0
        remaining = time_budget - (time.monotonic() - start)
        if remaining < 5:
            break
        lp_time = min(remaining * 0.3, 15.0)
        g = solve_lp(a, M, time_limit=lp_time)
        if g is None:
            break
        
        S_g = np.sum(g)
        if S_g < 1e-10:
            break
        g = g / S_g
        
        # Line search: find best t in [0,1]
        best_t = 0.0
        best_M_ls = best_M
        prev_Ms = []
        prev_ts = []
        
        for t in [0.1, 0.2, 0.3, 0.5, 0.7, 0.85, 1.0]:
            a_try = (1.0 - t) * a + t * g
            a_try = np.maximum(a_try, 0)
            S_try = np.sum(a_try)
            if S_try < 1e-10:
                continue
            a_try = a_try / S_try
            M_try = float(np.max(compute_conv(a_try)))
            prev_Ms.append(M_try)
            prev_ts.append(t)
            if M_try < best_M_ls:
                best_M_ls = M_try
                best_t = t
        
        # If we found improvement, do a finer search
        if best_t > 0 and len(prev_Ms) >= 3:
            # Parabolic interpolation
            ts = np.array(prev_ts)
            Ms = np.array(prev_Ms)
            # Find the three best points around the minimum
            idx = np.argmin(Ms)
            if 0 < idx < len(ts) - 1:
                t1, t2, t3 = ts[idx-1], ts[idx], ts[idx+1]
                f1, f2, f3 = Ms[idx-1], Ms[idx], Ms[idx+1]
                denom = (f1*(t2-t3) + f2*(t3-t1) + f3*(t1-t2))
                if abs(denom) > 1e-15:
                    t_par = 0.5 * (f1*(t2*t2-t3*t3) + f2*(t3*t3-t1*t1) + f3*(t1*t1-t2*t2)) / denom
                    if 0 < t_par < 1:
                        a_par = (1.0 - t_par) * a + t_par * g
                        a_par = np.maximum(a_par, 0)
                        S_par = np.sum(a_par)
                        if S_par > 1e-10:
                            a_par = a_par / S_par
                            M_par = float(np.max(compute_conv(a_par)))
                            if M_par < best_M_ls:
                                best_M_ls = M_par
                                best_t = t_par
        
        if best_t > 0:
            a = (1.0 - best_t) * a + best_t * g
            a = np.maximum(a, 0)
            a = a / np.sum(a)
            best_M = best_M_ls
            best_a = a.copy()
            stall = 0
            if verbose and it % 10 == 0:
                print(f"    LP iter {it}: M={best_M:.8f}, t={best_t:.3f}")
        else:
            stall += 1
            if stall > 10:
                # Try perturbations
                rng = np.random.default_rng(it * 7919 + 42)
                improved = False
                for scale in [0.05, 0.1, 0.2]:
                    a_pert = a * np.exp(rng.normal(0, scale, n))
                    a_pert = np.maximum(a_pert, 0)
                    S_p = np.sum(a_pert)
                    if S_p < 1e-10:
                        continue
                    a_pert = a_pert / S_p
                    M_pert = float(np.max(compute_conv(a_pert)))
                    if M_pert < best_M:
                        a = a_pert
                        best_M = M_pert
                        best_a = a_pert.copy()
                        improved = True
                        break
                if not improved:
                    break
                stall = 0
    
    return best_a


def cosine_init(n, n_modes=10, seed=0):
    """Generate a smooth cosine-series initialization."""
    rng = np.random.default_rng(seed)
    k = np.arange(n)
    a = np.ones(n)
    for m in range(1, n_modes + 1):
        c = rng.uniform(-0.5, 0.5) / (m * 0.5 + 1)
        phase = rng.uniform(0, 2 * np.pi)
        a += c * np.cos(2 * np.pi * m * k / n + phase)
    a = np.maximum(a, 1e-10)
    S = np.sum(a)
    return a / S


def gaussian_init(n, sigma_frac=0.3):
    """Gaussian bump initialization."""
    k = np.arange(n)
    center = (n - 1) / 2.0
    sigma = sigma_frac * n
    a = np.exp(-0.5 * ((k - center) / sigma) ** 2)
    S = np.sum(a)
    return a / S


def raised_cosine_init(n, alpha=2.0):
    """Raised cosine shape."""
    k = np.arange(n)
    x = 2.0 * k / (n - 1) - 1.0
    a = (0.5 * (1.0 + np.cos(np.pi * x))) ** alpha
    a = np.maximum(a, 1e-10)
    S = np.sum(a)
    return a / S


def bell_init(n, alpha=3.0):
    """Generalized bell curve."""
    k = np.arange(n)
    x = 2.0 * k / (n - 1) - 1.0
    a = (1.0 + x**2) ** (-alpha)
    S = np.sum(a)
    return a / S


def optimize_n_with_methods(a_init, n_methods_time, time_budget):
    """Apply multiple optimization methods to a sequence and return the best."""
    n = len(a_init)
    a = a_init.copy()
    S = np.sum(a)
    a = a / S
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    
    # Method 1: Subgradient
    remaining = time_budget - (time.monotonic() - start)
    if remaining > 5:
        t1 = min(remaining * 0.3, 20.0)
        a1 = bundle_subgradient(best_a, max_iters=int(t1 * 50), time_budget=t1)
        obj1 = compute_obj(a1)
        if obj1 < best_obj:
            best_obj = obj1
            best_a = a1.copy()
    
    # Method 2: LP alternation
    remaining = time_budget - (time.monotonic() - start)
    if remaining > 10 and n <= 600:
        t2 = min(remaining * 0.4, 40.0)
        a2 = alternating_lp_refine(best_a, max_iters=int(t2 * 2), time_budget=t2)
        obj2 = compute_obj(a2)
        if obj2 < best_obj:
            best_obj = obj2
            best_a = a2.copy()
    
    # Method 3: Subgradient again (refinement)
    remaining = time_budget - (time.monotonic() - start)
    if remaining > 5:
        t3 = min(remaining * 0.3, 15.0)
        a3 = bundle_subgradient(best_a, max_iters=int(t3 * 50), time_budget=t3)
        obj3 = compute_obj(a3)
        if obj3 < best_obj:
            best_obj = obj3
            best_a = a3.copy()
    
    return best_a, best_obj

def multi_start_search(n_list, seed, time_budget):
    """Try multiple initializations for each n and return best candidates."""
    start = time.monotonic()
    candidates = []
    
    for n in n_list:
        if time.monotonic() - start > time_budget:
            break
        remaining = time_budget - (time.monotonic() - start)
        if remaining < 10:
            break
        
        inits = [
            (f"cos_{n}", cosine_init(n, n_modes=8, seed=seed + n)),
            (f"cos_{n}_2", cosine_init(n, n_modes=12, seed=seed + n + 1000)),
            (f"gauss_{n}", gaussian_init(n, sigma_frac=0.25)),
            (f"gauss_{n}_2", gaussian_init(n, sigma_frac=0.35)),
            (f"gauss_{n}_3", gaussian_init(n, sigma_frac=0.45)),
            (f"rcos_{n}", raised_cosine_init(n, alpha=2.0)),
            (f"rcos_{n}_2", raised_cosine_init(n, alpha=3.0)),
            (f"bell_{n}", bell_init(n, alpha=3.0)),
            (f"bell_{n}_2", bell_init(n, alpha=4.0)),
        ]
        
        for name, a_init in inits:
            if time.monotonic() - start > time_budget:
                break
            remaining = time_budget - (time.monotonic() - start)
            if remaining < 5:
                break
            obj_init = compute_obj(a_init)
            # Quick subgradient
            a_quick = bundle_subgradient(a_init, max_iters=200, time_budget=min(3.0, remaining * 0.2))
            obj_quick = compute_obj(a_quick)
            best_local = a_quick if obj_quick < obj_init else a_init
            best_local_obj = min(obj_quick, obj_init)
            candidates.append((best_local, best_local_obj, name))
    
    return candidates


def interpolate_to_n(a, n_target):
    """Interpolate sequence to target length."""
    n_src = len(a)
    if n_src == n_target:
        return a.copy()
    x_old = np.linspace(0, 1, n_src)
    x_new = np.linspace(0, 1, n_target)
    try:
        from scipy.interpolate import interp1d
        f = interp1d(x_old, a, kind='cubic', fill_value='extrapolate')
        a_new = np.maximum(f(x_new), 0)
    except Exception:
        # Fallback: simple linear interpolation
        from scipy.interpolate import interp1d
        f = interp1d(x_old, a, kind='linear', fill_value=0)
        a_new = np.maximum(f(x_new), 0)
    S = np.sum(a_new)
    if S > 1e-10:
        a_new = a_new / S
    return a_new


def active_set_optimization(a, time_budget, verbose=False):
    """
    Optimization that focuses on the active set.
    Identifies which convolution positions are active (near max)
    and uses this structure to guide optimization.
    """
    n = len(a)
    a = a.copy().astype(np.float64)
    S = np.sum(a)
    a = a / S
    
    best_a = a.copy()
    best_M = float(np.max(compute_conv(a)))
    start = time.monotonic()
    stall = 0
    
    while time.monotonic() - start < time_budget:
        remaining = time_budget - (time.monotonic() - start)
        if remaining < 3:
            break
        
        conv = compute_conv(a)
        M = float(np.max(conv))
        
        # Identify active set with tight threshold
        threshold = M * (1.0 - 0.001)
        active = np.where(conv >= threshold)[0]
        
        # Compute subgradient from active set only
        grad = np.zeros(n)
        for k in active:
            for i in range(n):
                j = k - i
                if 0 <= j < n:
                    grad[i] += a[j]
        grad /= len(active)
        
        # Project to maintain sum
        grad = grad - np.mean(grad)
        
        # Adaptive step size
        step = 0.05 / (1 + 0.001 * (time.monotonic() - start))
        
        a_new = a - step * grad
        a_new = np.maximum(a_new, 0)
        S_new = np.sum(a_new)
        if S_new < 1e-10:
            break
        a_new = a_new / S_new
        
        M_new = float(np.max(compute_conv(a_new)))
        if M_new < best_M - 1e-12:
            best_M = M_new
            best_a = a_new.copy()
            a = a_new
            stall = 0
        else:
            stall += 1
            if stall > 30:
                # Try LP escape
                if remaining > 10:
                    g = solve_lp(a_new, M, time_limit=min(5, remaining * 0.2))
                    if g is not None:
                        S_g = np.sum(g)
                        if S_g > 1e-10:
                            g = g / S_g
                            for t in [0.3, 0.5, 0.7]:
                                a_try = (1-t)*a_new + t*g
                                a_try = np.maximum(a_try, 0)
                                S_try
