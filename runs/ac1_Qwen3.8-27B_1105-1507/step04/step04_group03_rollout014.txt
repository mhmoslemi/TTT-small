def fast_conv(a):

def compute_J(a):

def simplex_project(v):
    """Project v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not np.any(cond):

    rho = np.nonzero(cond)[0][-1]

    return np.maximum(v - theta, 0)


def build_conv_matrix(n, a):

    cols = []

def resample_seq(src, dest_n):
    """Resample 1D sequence to new length."""
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)

        if J2 < compute_J(a):
            a = a2
        print(f"  J={compute_J(a):.8f} ({time.monotonic()-t0:.1f}s)", flush=True)

    # === Phase 3: LP mirror descent - multiplicative mode (250s) ===
    if time.monotonic() < deadline - 250:
        t0 = time.monotonic()
        print(f"\nPhase 3: LP mirror descent multiplicative (n={n_parent})", flush=True)
        remaining = min(250, deadline - 250 - time.monotonic())
        a3, J3 = lp_mirror_descent(a, t0 + remaining, max_iters=5000, mode='multiplicative')
        update_tracker(a3)
        if J3 < compute_J(a):
            a = a3
        print(f"  J={compute_J(a):.8f} ({time.monotonic()-t0:.1f}s)", flush=True)

    # === Phase 4: Subgradient polish (60s) ===
    if time.monotonic() < deadline - 60:
        t0 = time.monotonic()
        print(f"\nPhase 4: Subgradient polish (n={n_parent})", flush=True)
        a4 = subgradient_polish(a, min(deadline - 55, t0 + 55))
        update_tracker(a4)
        if compute_J(a4) < compute_J(a):
            a = a4
        print(f"  J={compute_J(a):.8f} ({time.monotonic()-t0:.1f}s)", flush=True)

    # === Phase 5: LP mirror descent - auto mode (200s) ===
    if time.monotonic() < deadline - 100:
        t0 = time.monotonic()
        print(f"\nPhase 5: LP mirror descent auto (n={n_parent})", flush=True)
        remaining = min(200, deadline - 100 - time.monotonic())
        a5, J5 = lp_mirror_descent(a, t0 + remaining, max_iters=5000, mode='auto')
        update_tracker(a5)
        if J5 < compute_J(a):
            a = a5
        print(f"  J={compute_J(a):.8f} ({time.monotonic()-t0:.1f}s)", flush=True)

    # === Phase 6: L-BFGS-B with decreasing beta (120s) ===
    if time.monotonic() < deadline - 30:
        t0 = time.monotonic()
        print(f"\nPhase 6: L-BFGS-B smoothed (n={n_parent})", flush=True)
        for beta in [5, 20, 50, 100]:
            if time.monotonic() > deadline - 25:
                break
            a6 = lbb_smoothed(a, beta, max_iter=100)
            update_tracker(a6)
            if compute_J(a6) < compute_J(a):
                a = a6
        print(f"  J={compute_J(a):.8f} ({time.monotonic()-t0:.1f}s)", flush=True)

    # === Phase 7: Multi-length exploration (100s) ===
    if time.monotonic() < deadline - 60:
        t0 = time.monotonic()
        print(f"\nPhase 7: Multi-length (base n={n_parent})", flush=True)
        lengths = sorted(set([n_parent - 64, n_parent + 32, n_parent + 64, n_parent + 128]))
        lengths = [n for n in lengths if 128 <= n <= 2048 and n != n_parent]

        for n_try in lengths[:3]:
            if time.monotonic() > deadline - 50:
                break
            a_try = resample_seq(a, n_try)
            a_try = np.maximum(a_try, 1e-10)
            a_try = a_try / a_try.sum()
            update_tracker(a_try)

            # Quick LP refinement
            dl = min(deadline - 45, time.monotonic() + 25)
            a_try, J_try = lp_mirror_descent(a_try, dl, max_iters=300, mode='linear')
            update_tracker(a_try)
            print(f"  n={n_try}: J={compute_J(a_try):.8f}", flush=True)

        print(f"Phase 7 done ({time.monotonic()-t0:.1f}s)", flush=True)

    # === Phase 8: Final long LP on best (remaining time) ===
    if time.monotonic() < deadline - 15 and best_a is not None:
        t0 = time.monotonic()
        print(f"\nPhase 8: Final LP refinement (n={best_a.size})", flush=True)
        a_final = best_a.copy()
        a_final = np.maximum(a_final, 1e-10)
        a_final = a_final / a_final.sum()
        a_final, J_final = lp_mirror_descent(a_final, deadline - 10, max_iters=20000, mode='auto')
        update_tracker(a_final)
        print(f"  J={compute_J(a_final):.8f} ({time.monotonic()-t0:.1f}s)", flush=True)

    # Final output
    if best_a is not None:
        print(f"\n{'='*50}", flush=True)
        print(f"FINAL: J={best_score:.8f} n={best_a.size} time={time.monotonic()-t_start:.0f}s", flush=True)
        print(f"{'='*50}", flush=True)
        return [float(x) for x in best_a]
    else:
        print(f"WARNING: No valid candidate found, returning parent", flush=True)
        return [float(max(x, 0)) for x in parent]
