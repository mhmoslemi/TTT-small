```python
"""
Hybrid optimizer using coordinate descent, smoothed-max gradient descent,
and LP mirror descent with adaptive M to minimize J(a) = 2*n*max(conv(a,a))/sum(a)^2.
"""

import numpy as np
from scipy.optimize import linprog, minimize
from scipy.sparse import csr_matrix
import time


def fast_conv(a):
    """Full self-convolution via FFT."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]


def compute_J(a):
    """Compute J(a) = 2*n*max(conv(a,a))/(sum(a))^2."""
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))


def smooth_max(x, tau=1.0):
    """Smooth approximation of max using log-sum-exp."""
    x_max = x.max()
    return x_max + np.log(np.sum(np.exp(x - x_max))) / tau * tau  # wait this is wrong
    # Correct: (1/tau) * log(sum(exp(tau * x)))
    # = (1/tau) * log(sum(exp(tau*(x - x_max)))) + x_max
    return x_max + np.log(np.sum(np.exp(tau * (x - x_max)))) / tau


def smooth_max_grad(x, tau=1.0):
    """Gradient of smooth max."""
    x_max = x.max()
    weights = np.exp(tau * (x - x_max))
    weights = weights / weights.sum()
    return weights


def build_A_sparse(n, a):
    """Build sparse matrix A where A[k,j] = a[k-j] for valid j."""
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        j_min = max(0, k - n + 1)
        j_max = min(n - 1, k)
        for j in range(j_min, j_max + 1):
            rows.append(k)
            cols.append(j)
            vals.append(a[k - j])
    return csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))


def lp_best_response(a, M):
    """Solve LP: max sum(g) s.t. A@g <= M*1, g >= 0."""
    n = len(a)
    A = build_A_sparse(n, a)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    try:
        r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
        if r.success and r.x is not None:
            g = np.maximum(r.x, 0)
            if g.sum() > 1e-15:
                return g
    except Exception:
        pass
    return None

                        lo, hi = best_val, mid  # hmm this is getting complicated
                    # Just use the best of the three
                    break  # Skip fine search for now, keep it simple
                
                a[j] = best_val
                improved = True
            
            a[j] = best_val  # Make sure we set it
            
            if best_trial_M < best_M:
                best_M = best_trial_M
                improved = True

        if not improved:
            no_improve += 1
            if no_improve >= 3:
                break
        else:
            no_improve = 0
            # Normalize
            s = a.sum()
            if s > 1e-10:
                a = a / s
                best_M = fast_conv(a).max()
    
    return a


def smoothed_max_objective(x, tau=50.0):
    """Objective: smooth max of self-convolution (scale invariant)."""
    ax = np.maximum(x, 0.0)
    s = ax.sum()
    if s < 1e-8:
        return 1e10
    ax = ax / s
    b = fast_conv(ax)
    return smooth_max(b, tau)


def smoothed_max_gradient(x, tau=50.0):
    """Gradient of smooth max of self-convolution."""
    ax = np.maximum(x, 1e-15)
    s = ax.sum()
    ax = ax / s
    b = fast_conv(ax)
    weights = smooth_max_grad(b, tau)
    
    # Gradient of b[k] w.r.t. ax[j]
    # b[k] = sum_i ax[i] * ax[k-i]
    # db[k]/dax[j] = ax[k-j] + ax[k-j] = 2*ax[k-j] (if valid)
    # But we need to account for the normalization ax = x/s
    # For now, just compute the gradient of max(conv) w.r.t. unnormalized a
    grad = np.zeros_like(ax)
    n = len(ax)
    for k in range(2 * n - 1):
        w = weights[k]
        if w < 1e-10:
            continue
        j_min = max(0, k - n + 1)
        j_max = min(n - 1, k)
        for j in range(j_min, j_max + 1):
            grad[j] += 2.0 * w * ax[k - j]
    
    return grad


def gradient_descent_smoothed(a, max_iters, deadline, tau=50.0):
    """Gradient descent with momentum on smoothed max."""
    a = np.maximum(a.astype(np.float64), 1e-10)
    a = a / a.sum()
    best_a = a.copy()
    best_J = compute_J(a)
    momentum = np.zeros_like(a)
    beta = 0.9
    lr = 0.01

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        grad = smoothed_max_gradient(a, tau)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-15:
            break

        momentum = beta * momentum + (1 - beta) * grad / gnorm
        a_new = a - lr * momentum
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s < 1e-10:
            break
        a_new = a_new / s

        J_new = compute_J(a_new)
        if J_new < best_J:
            best_J = J_new
            best_a = a_new.copy()
            lr = min(lr * 1.05, 0.1)
        else:
            lr *= 0.9

        a = a_new

    return best_a


def nelder_mead_smoothed(a0, max_iters, deadline, tau=50.0):
    """Nelder-Mead on smoothed max objective."""
    a = np.maximum(a0.astype(np.float64), 1e-10)
    a = a / a.sum()
    n = len(a)
    
    # Create initial simplex
    simplex = [a.copy()]
    for i in range(n):
        v = a.copy()
        v[i] *= 1.01
        if v[i] < 0:
            v[i] = 0
        v = np.maximum(v, 0)
        s = v.sum()
        if s > 1e-10:
            v = v / s
        simplex.append(v)

    # Evaluate objective
    obj = lambda x: smoothed_max_objective(np.maximum(x, 0), tau)
    
    best_a = a.copy()
    best_val = compute_J(a)

    # Use scipy's Nelder-Mead
    try:
        res = minimize(lambda x: smoothed_max_objective(np.maximum(x, 0), tau),
                      a, method='Nelder-Mead',
                      options={'maxiter': max_iters, 'xatol': 1e-8, 'fatol': 1e-10})
        if res.fun < 1e10:
            a_nm = np.maximum(res.x, 0)
            s = a_nm.sum()

    except Exception:

    return best_a


def fourier_optimize(a0, n_harm, deadline, method='L-BFGS-B'):
    """Optimize in Fourier parameter space."""
    n = len(a0)
    a_norm = a0 / a0.sum()
    
    t = np.arange(n, dtype=np.float64) / n
    params = np.zeros(2 * n_harm + 1)
    params[0] = np.mean(a_norm)
    for k in range(1, n_harm + 1):
        params[2 * k - 1] = 2.0 * np.mean(a_norm * np.cos(2.0 * np.pi * k * t))
        params[2 * k] = 2.0 * np.mean(a_norm * np.sin(2.0 * np.pi * k * t))

    def fourier_to_seq(p):
        a = np.full(n, p[0], dtype=np.float64)
        for k in range(1, n_harm + 1):
            a += p[2 * k - 1] * np.cos(2.0 * np.pi * k * t)
            a += p[2 * k] * np.sin(2.0 * np.pi * k * t)
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s < 1e-8:
            return np.ones(n) / n
        return a / s

    def objective(p):
        a = fourier_to_seq(p)
        return compute_J(a)

    x0 = params.copy()
    f0 = objective(x0)
    best_a = fourier_to_seq(x0)
    best_f = f0

    try:
        bounds = [(0, None)] + [(-1, 1)] * (2 * n_harm)
        res = minimize(objective, x0, method=method, bounds=bounds,
                      options={'maxiter': 200, 'max_fun': 50000, 'ftol': 1e-12})
        if res.fun < best_f - 1e-12:
            best_a = fourier_to_seq(res.x)
            best_f = res.fun
            best_a_norm = best_a
    except Exception:
        pass

    return best_a, best_f


def resample_seq(src, dest_n):
    """Resample 1D sequence to new length."""
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)

def coordinate_descent_v2(a, max_passes, deadline):
    """
    Improved coordinate descent with binary search for optimal value.
    """
    a = np.maximum(a.astype(np.float64), 1e-10)
    a = a / a.sum()
    n = len(a)
    best_a = a.copy()
    best_M = fast_conv(a).max()
    no_improve_passes = 0

    for p in range(max_passes):
        if time.monotonic() > deadline:
            break
        pass_improved = False
        
        for j in range(n):
            if time.monotonic() > deadline:
                break
            
            orig = a[j]
            
            # Compute current conv
            b = fast_conv(a)
            M_cur = b.max()
            
            # Try to find better value for a[j]
            # The effect of changing a[j] to x on conv[k]:
            # conv[k] new = conv[k] - 2*orig*ax[k-j] + 2*x*ax[k-j] (approximately, ignoring the x^2 term at k=2j)
            # More precisely: conv[k] = sum_i a[i]*a[k-i]
            # Terms with a[j]: a[j]*a[k-j] + a[k-j]*a[j] = 2*a[j]*a[k-j] (if j != k-j)
            # or a[j]^2 (if j == k-j, i.e., k = 2j)
            
            # For each k, the new conv[k] = old_conv[k] - 2*orig*a[k-j] + 2*x*a[k-j] 
            #   (for k where j is valid and k-j is valid and j != k-j)
            # = old_conv[k] + 2*(x - orig)*a[k-j]
            
            # For k = 2j: conv[2j] new = old_conv[2j] - orig^2 + x^2
            
            # We want to minimize max_k new_conv[k]
            # This is: min_x max_k (C_k + 2*(x-orig)*a[k-j]) s.t. x >= 0
            # where C_k = old_conv[k] - 2*orig*a[k-j] (for k != 2j)
            # and C_{2j} = old_conv[2j] - orig^2
            
            # This is a 1D convex problem. We can solve it with binary search.
            
            # Valid range for x: let's try [0, 3*orig]
            lo, hi = 0.0, max(3.0 * orig, 1.0)
            
            def objective_1d(x):
                # Compute max conv when a[j] = x
                # conv[k] = b[k] - 2*orig*a_valid[k-j] + 2*x*a_valid[k-j] for k != 2j
                # conv[2j] = b[2j] - orig^2 + x^2 (if 2j < 2n-1)
                
                # Use FFT: just directly compute
                a_trial = a.copy()
                a_trial[j] = x
                b_trial = fast_conv(a_trial)
                return b_trial.max()
            
            # Grid search first
            grid = np.linspace(0, hi, 20)
            best_x = orig
            best_obj = objective_1d(orig)
            for x in grid:
                obj = objective_1d(x)
                if obj < best_obj:
                    best_obj = obj
                    best_x = x
            
            # Refine around best_x with finer grid
            if best_x != orig:
                span = hi / 20
                fine_grid = np.linspace(max(0, best_x - span), min(hi, best_x + span), 20)
                for x in fine_grid:
                    obj = objective_1d(x)
                    if obj < best_obj:
                        best_obj = obj
                        best_x = x
            
            if best_x != orig:
                a[j] = best_x
                pass_improved = True
                if best_obj < best_M:
                    best_M = best_obj
        
        if pass_improved:
            s = a.sum()
            if s > 1e-10:
                a = a / s
            no_improve_passes = 0
        else:
            no_improve_passes += 1
            if no_improve_passes >= 2:
                break
    
    return a
```

Wait, I realize the coordinate descent with FFT at each step is going to be very slow. For n=640, each FFT-based conv takes O(n log n) ≈ O(640*10) = 6400 operations. But doing it for 20 grid points × 640 coordinates × 50 passes = 640,000 FFT calls. That's way too slow.

I need to optimize the coordinate descent. Instead of recomputing the full convolution each time, I can use the formula:
conv_new[k] = conv_old[k] + 2*(x - orig) * a[k-j] (for valid k, k-j)

This is O(n) per candidate value, and we have O(n) k values, so O(n) per candidate. With 20 candidates per coordinate, that's O(20*n) per coordinate, O(20*n^2) per pass. For n=640, that's about 8M operations per pass. With 50 passes, that's 400M operations. Still might be slow in Python.

Let me vectorize this:

For coordinate j, the change in conv when a[j] changes from orig to x:
- For each k, conv[k] changes by 2*(x-orig)*a[k-j] where both j and k-j are in [0, n-1]
- The valid k range is: j <= k <= j + n - 1 (i.e., k-j ranges from 0 to n-1, and j is fixed)
  Wait: k-j must be in [0, n-1] AND j must be in [0, n-1] (which it is).
  So k-j in [0, n-1] means k in [j, j+n-1]. But k must also be in [0, 2n-2].
  So k in [max(0, j), min(2n-2, j+n-1)] = [j, j+n-1] (since j >= 0 and j+n-1 <= 2n-2 for j <= n-1).

Actually wait, I need to be more careful. The convolution is:
conv[k] = sum_{i=0}^{n-1} a[i] * a[k-i] where a[m] = 0 for m not in [0, n-1]

The terms where a[j] appears:
- i = j: a[j] * a[k-j] (valid when 0 <= k-j <= n-1, i.e., j <= k <= j+n-1)
- k-i = j: a[k-j] * a[j] (valid when 0 <= k-j <= n-1, i.e., j <= k <= j+n-1)

These are the same term when j = k-j, i.e., k = 2j.

So for k != 2j: conv[k] gets contribution 2*a[j]*a[k-j]
For k = 2j: conv[k] gets contribution a[j]^2

When we change a[j] from orig to x:
- For k != 2j (and j <= k <= j+n-1): conv[k] changes by 2*(x-orig)*a[k-j]
- For k = 2j: conv[k] changes by x^2 - orig^2

So the new conv[k] for k in [j, j+n-1]:
- If k != 2j: conv_new[k] = conv_old[k] + 2*(x-orig)*a[k-j]
- If k = 2j: conv_new[k] = conv_old[k] + x^2 - orig^2

And for k outside [j, j+n-1], conv doesn't change.

This is great! We can compute the new max in O(n) time for each candidate x.

Let me rewrite the coordinate descent with this optimization:

```python
def coordinate_descent_v2(a, max_passes, deadline):
    a = np.maximum(a.astype(np.float64), 1e-10)
    a = a / a.sum()
    n = len(a)
    best_a = a.copy()
    conv = fast_conv(a).copy()
    best_M = conv.max()
    no_improve_passes = 0

    for p in range(max_passes):
        if time.monotonic() > deadline:
            break
        pass_improved = False
        
        for j in range(n):
            if time.monotonic() > deadline:
                break
            
            orig = a[j]
            # Valid k range: [j, j+n-1]
            k_lo = j
            k_hi = j + n - 1
            k_hi = min(k_hi, 2*n-2)
            
            # The "affected" convolution values
            # For k in [k_lo, k_hi], k != 2j: new_conv[k] = conv[k] + 2*(x-orig)*a[k-j]
```

```python
"""
Hybrid optimizer minimizing J(a) = 2*n*max(conv(a,a))/sum(a)^2.
Combines: (1) LP mirror descent with adaptive M and cached sparse structure,
(2) vectorized coordinate descent with O(n) per-candidate evaluation,
(3) Fourier-parameterized L-BFGS-B for global escape,
(4) multi-length exploration.
Key innovation: precomputed LP matrix structure + adaptive M target for stronger
LP descent steps, plus exact O(n) coordinate descent without repeated FFTs.

import numpy as np
from scipy.optimize import linprog, minimize
from scipy.sparse import csr_matrix
import time


def fast_conv(a):
    """Full self-convolution via FFT."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]


def compute_J(a):
    """Compute J(a) = 2*n*max(conv(a,a))/(sum(a))^2."""
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))


def build_A_structure(n):
    """Precompute row/col indices for the convolution matrix A[k,j] = a[k-j]."""
    rows = []
    cols = []
    for k in range(2 * n - 1):
        j_min = max(0, k - n + 1)
        j_max = min(n - 1, k)
        for j in range(j_min, j_max + 1):
            rows.append(k)
            cols.append(j)
    return np.array(rows, dtype=np.int32), np.array(cols, dtype=np.int32)


def make_A(rows, cols, a):
    """Build sparse convolution matrix using precomputed structure."""
    n = len(a)
    vals = a[rows - cols]
    return csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))


def lp_best_response_fast(rows, cols, a, M):
    """Solve LP: max sum(g) s.t. A@g <= M, g >= 0 using precomputed structure."""
    n = len(a)
    A = make_A(rows, cols, a)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    try:
        r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
        if r.success and r.x is not None:
            g = np.maximum(r.x, 0)
            if g.sum() > 1e-15:
                return g
    except Exception:
        pass
    return None


def lp_mirror_descent(a, rows, cols, max_iters, deadline, multiplicative=False, adaptive_M=False):
    """LP mirror descent with precomputed structure and adaptive M."""
    a = np.maximum(a.astype(np.float64), 1e-10)
    a = a / a.sum()
    n = len(a)
    current_J = compute_J(a)
    best_J = current_J
    best_a = a.copy()
    no_improve = 0

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        b = fast_conv(a)
        M = float(b.max())

        # Adaptive M: try slightly below current max for stronger step
        if adaptive_M:
            # Oscillate between slightly below and at current M
            frac = 1.0 - 0.005 * (1.0 + np.sin(0.05 * it))
            M_target = M * frac
            # Ensure feasibility: M must be >= sum(a)^2/(2n-1) approximately
            M_floor = a.sum() ** 2 / (2.0 * n - 1) * 1.005
            M_target = max(M_target, M_floor)
        else:
            M_target = M

        g = lp_best_response_fast(rows, cols, a, M_target)
        if g is None:
            g = lp_best_response_fast(rows, cols, a, M)
        if g is None:
            break
        g = np.maximum(g, 0.0)
        g_sum = g.sum()
        if g_sum < 1e-10:
            break
        g = g / g_sum

        # Adaptive step size: start large, decay slowly
        base_t = 0.4 * np.exp(-0.1 * it / 500.0) + 0.08
        t = min(1.0, base_t)

        best_trial_J = current_J
        best_trial_a = None

        for ls in range(30):
            if multiplicative:
                ratio = g / np.maximum(a, 1e-15)
                a_trial = a * np.power(ratio, t)
            else:
                a_trial = (1.0 - t) * a + t * g
            a_trial = np.maximum(a_trial, 0.0)
            s = a_trial.sum()
            if s < 1e-10:
                t *= 0.5
                continue
            a_trial = a_trial / s
            J_trial = compute_J(a_trial)
            if J_trial < best_trial_J - 1e-15:
                best_trial_J = J_trial
                best_trial_a = a_trial.copy()
                break
            t *= 0.55
            if t < 0.001:
                break

        if best_trial_a is not None:
            a = best_trial_a
            current_J = best_trial_J
            if current_J < best_J:
                best_J = current_J
                best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1
            if no_improve > 60:
                break

    return best_a, best_J


def coordinate_descent(a, max_passes, deadline):
    """
    Fast coordinate descent using O(n) per-candidate evaluation.
    For coordinate j changing from orig to x:
      conv[k] changes by 2*(x-orig)*a[k-j] for k in [j, j+n-1], k != 2j
      conv[2j] changes by x^2 - orig^2
    """
    a = np.maximum(a.astype(np.float64), 1e-10)
    a = a / a.sum()
    n = len(a)
    conv = fast_conv(a).copy()
    best_M = conv.max()
    no_improve = 0

    for p in range(max_passes):
        if time.monotonic() > deadline:
            break
        pass_improved = False

        for j in range(n):
            if time.monotonic() > deadline:
                break

            orig = a[j]
            # Affected k range: [j, j+n-1]
            k_lo = j
            k_hi = min(j + n - 1, 2 * n - 2)
            if k_lo > k_hi:
                continue

            # Coefficients: for k in [k_lo, k_hi], linear coeff is 2*a[k-j]
            affected_k = np.arange(k_lo, k_hi + 1)
            affected_a = a[affected_k - j]  # = a[0], a[1], ..., a[n-1] (or subset)
            coeffs = 2.0 * affected_a

            # Unaffected max (precompute once per j)
            if k_lo == 0 and k_hi == 2 * n - 2:
                unaffected_max = 0.0
            else:
                # Build mask
                mask = np.ones(2 * n - 1, dtype=bool)
                mask[k_lo:k_hi + 1] = False
                unaffected_max = conv[mask].max() if mask.any() else 0.0

            # Current affected values
            aff_vals = conv[k_lo:k_hi + 1].copy()

            # Try candidate values
            candidates = np.array([0.0, orig * 0.2, orig * 0.5, orig * 0.8,
                                   orig * 1.0, orig * 1.2, orig * 1.5, orig * 2.0, orig * 3.0])

            best_x = orig
            best_trial_max = best_M

            for x in candidates:
                delta = x - orig
                new_aff = aff_vals + delta * coeffs
                # Quadratic correction at k = 2j
                if k_lo <= 2 * j <= k_hi:
                    pos = 2 * j - k_lo
                    new_aff[pos] = aff_vals[pos] + x * x - orig * orig
                trial_max = max(new_aff.max(), unaffected_max)
                if trial_max < best_trial_max:
                    best_trial_max = trial_max
                    best_x = x

            if best_x != orig:
                a[j] = best_x
                # Update conv in place
                delta = best_x - orig
                conv[k_lo:k_hi + 1] += delta * coeffs
                if k_lo <= 2 * j <= k_hi:
                    pos = 2 * j - k_lo
                    # Correct: the linear update already added delta*coeffs[pos] = delta*2*a[0]
                    # But we need x^2 - orig^2 instead. So subtract the linear part and add quadratic.
                    conv[2 * j] = conv[2 * j] - delta * 2.0 * a[0] + (best_x * best_x - orig * orig)
                pass_improved = True
                if best_trial_max < best_M:
                    best_M = best_trial_max

        if pass_improved:
            s = a.sum()
            if s > 1e-10:
                a = a / s
            # Recompute conv to avoid drift
            conv = fast_conv(a)
            best_M = conv.max()
            no_improve = 0
        else:
            no_improve += 1
            if no_improve >= 3:
                break

    return a


def fourier_optimize(a0, n_harm, deadline):
    """Optimize in Fourier parameter space using L-BFGS-B."""
    n = len(a0)
    a_norm = a0 / a0.sum()
    t = np.arange(n, dtype=np.float64) / n

    params = np.zeros(2 * n_harm + 1)
    params[0] = np.mean(a_norm)
    for k in range(1, n_harm + 1):
        params[2 * k - 1] = 2.0 * np.mean(a_norm * np.cos(2.0 * np.pi * k * t))
        params[2 * k] = 2.0 * np.mean(a_norm * np.sin(2.0 * np.pi * k * t))

    def fourier_to_seq(p):
        a = np.full(n, p[0], dtype=np.float64)
        for k in range(1, n_harm + 1):
            a += p[2 * k - 1] * np.cos(2.0 * np.pi * k * t)
            a += p[2 * k] * np.sin(2.0 * np.pi * k * t)
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s < 1e-8:
            return np.ones(n) / n
        return a / s

    def objective(p):
        a = fourier_to_seq(p)
        b = fast_conv(a)
        return float(b.max())

    x0 = params.copy()
    f0 = objective(x0)
    best_a = fourier_to_seq(x0)
    best_f = f0

    try:
        bounds = [(0, None)] + [(-0.5, 0.5)] * (2 * n_harm)
        res = minimize(objective, x0, method='L-BFGS-B', bounds=bounds,
                       options={'maxiter': 300, 'max_fun': 80000, 'ftol': 1e-14})
        if res.fun < best_f - 1e-14:
            best_a = fourier_to_seq(res.x)
            best_f = res.fun
    except Exception:
        pass

    return best_a, compute_J(best_a)


def resample_seq(src, dest_n):
    """Resample 1D sequence to new length."""
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)

            a_four_lp = np.maximum(a_four_lp, 1e-10)
            a_four_lp = a_four_lp / a_four_lp.sum()
            a_four_lp, J_four_lp = lp_mirror_descent(a_four_lp, rows, cols, 2000,
                                                      time.monotonic() + min(60, deadline - time.monotonic() - 30),
                                                      multiplicative=False, adaptive_M=True)
            tracker.update(a_four_lp)
            print(f"  LP from Fourier: J={J_four_lp:.8f} ({time.monotonic()-t0b:.1f}s)", flush=True)
            if J_four_lp < J_best:
                current_best = a_four_lp.copy()
                J_best = J_four_lp

    # === Phase 6: Multi-length exploration ===
    remaining = deadline - time.monotonic()
    if remaining > 60:
        print(f"\n{'='*60}", flush=True)
        print(f"Phase 6: Multi-length (base n={n_best})", flush=True)
        print(f"{'='*60}", flush=True)

        n_base = n_best
        lengths_to_try = []
        for delta in [-32, 32, -64, 64, 128]:
            n_try = n_base + delta
            if 128 <= n_try <= 2048 and n_try != n_base:
                lengths_to_try.append(n_try)
        if n_base * 2 <= 2048:
            lengths_to_try.append(n_base * 2)

        lengths_to_try = sorted(set(lengths_to_try))

        for n_try in lengths_to_try:
            if time.monotonic() > deadline - 30:
                break
            a_try = resample_seq(current_best, n_try)
            a_try = np.maximum(a_try, 1e-10)
            a_try = a_try / a_try.sum()

            # Quick CD
            per_budget = max(10, min(30, (deadline - 30 - time.monotonic()) / max(1, len(lengths_to_try))))
            a_try = coordinate_descent(a_try, 20, time.monotonic() + per_budget * 0.4)
            a_try = np.maximum(a_try, 1e-10)
            a_try = a_try / a_try.sum()

            # Quick LP
            rows_t, cols_t = build_A_structure(n_try)
            a_try, J_try = lp_mirror_descent(a_try, rows_t, cols_t, 500,
                                             time.monotonic() + per_budget * 0.6,
                                             multiplicative=False, adaptive_M=True)
            tracker.update(a_try)
            print(f"  n={n_try}: J={J_try:.8f}", flush=True)

            if J_try < J_best:
                current_best = a_try.copy()
                n_best = n_try
                J_best = J_try

    # === Phase 7: Final long LP + CD ===
    remaining = deadline - time.monotonic()
    if remaining > 20:
        print(f"\n{'='*60}", flush=True)
        print(f"Phase 7: Final LP+CD (n={n_best}, budget={remaining:.0f}s)", flush=True)
        print(f"{'='*60}", flush=True)

        # Rebuild structure if length changed
        if len(current_best) != n_best:
            n_best = len(current_best)
        rows_f, cols_f = build_A_structure(n_best)

        a_fin = current_best.copy()
        a_fin = np.maximum(a_fin, 1e-10)
        a_fin = a_fin / a_fin.sum()
        t0 = time.monotonic()
        a_fin, J_fin = lp_mirror_descent(a_fin, rows_f, cols_f, 10000, deadline - 10,
                                         multiplicative=False, adaptive_M=True)
        tracker.update(a_fin)
        print(f"  Final LP: J={J_fin:.8f} ({time.monotonic()-t0:.1f}s)", flush=True)
        if J_fin < J_best:
            current_best = a_fin.copy()
            J_best = J_fin

        # Final CD
        if time.monotonic() < deadline - 5:
            a_fin_cd = coordinate_descent(current_best, 50, deadline - 2)
            J_fin_cd = compute_J(a_fin_cd)
            tracker.update(a_fin_cd)
            print(f"  Final CD: J={J_fin_cd:.8f}", flush=True)
            if J_fin_cd < J_best:
                current_best = a_fin_cd.copy()
                J_best = J_fin_cd

    # Final decision
    if tracker.best is not None:
        final_a = tracker.best
    else:
        final_a = current_best

    print(f"\n{'='*60}", flush=True)
    print(f"FINAL: J={tracker.best_score:.8f} n={final_a.size} time={time.monotonic()-tracker.t_start:.0f}s", flush=True)
    print(f"{'='*60}", flush=True)

    return [float(x) for x in final_a]
```
