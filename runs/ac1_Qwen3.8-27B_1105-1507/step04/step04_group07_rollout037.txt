def fast_conv(a):
    n = len(a)
    s = 1
    while s < 2 * n - 1:
        s <<= 1
    fa = np.fft.rfft(a, s)
    return np.fft.irfft(fa * fa, s)[:2 * n - 1]

def compute_J(a):

    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)

def coord_descent(a, max_passes, deadline, tracker, tol=1e-10, verbose=False):
    """
    Fast coordinate descent with golden-section 1D minimization.
    Each pass goes through all coordinates in random order.
    """
    a = a.copy().astype(np.float64)
    n = len(a)
    s = a.sum()
    if s < 1e-10:
        return a
    a /= s
    
    current_J = compute_J(a)
    tracker.update(a)
    
    rng = np.random.default_rng(42)
    gr = (1.0 - 5.0 ** 0.5) / 2.0  # ~-0.618, will use abs
    gr_abs = -gr  # ~0.618
    
    for p in range(max_passes):
        if time.monotonic() > deadline:
            break
        
        b = fast_conv(a)
        # Prefix and suffix max for quick "max outside" queries
        prefix_max = np.maximum.accumulate(b)
        suffix_max = np.maximum.accumulate(b[::-1])[::-1]
        
        order = rng.permutation(n)
        pass_improvement = 0.0
        
        for i in order:
            if time.monotonic() > deadline:
                break
            
            # k range where a[i] affects conv
            k_lo = i
            k_hi = min(2 * n - 2, i + n - 1)
            
            # Max of b outside [k_lo, k_hi]
            max_out = 0.0
            if k_lo > 0:
                max_out = max(max_out, prefix_max[k_lo - 1])
            if k_hi < 2 * n - 2:
                max_out = max(max_out, suffix_max[k_hi + 1])

            sum_base = 1.0 - ai  # since sum(a) = 1
            
            # Prepare arrays for vectorized evaluation
            ks = np.arange(k_lo, k_hi + 1)
            ds = ks - i  # indices into a for the "other" factor
            
            # base[k] = b[k] - current contribution of a[i]
            base = b[ks].copy()
            mask_2i = (ks == 2 * i)
            base[mask_2i] -= ai * ai
            base[~mask_2i] -= 2.0 * ai * a[ds[~mask_2i]]
            
            # Linear coefficient: c[k] = 2*a[k-i] for k != 2i, 0 for k = 2i
            c_coef = np.zeros(len(ks))
            c_coef[~mask_2i] = 2.0 * a[ds[~mask_2i]]
            
            # Now new_b[k] = base[k] + c_coef[k]*t + (t^2 if k==2i else 0)
            
            # Golden section search for t in [0, t_hi]
            t_hi = max(2.5 * ai, 0.02) if ai > 1e-8 else 0.05
            
            def J_t(t):
                new_vals = base + c_coef * t
                new_vals[mask_2i] += t * t
                m = max(max_out, new_vals.max())
                s_new = sum_base + t
                return 2.0 * n * m / (s_new * s_new)
            
            # Evaluate at current
            J_cur = J_t(ai)
            
            # Golden section
            lo = 0.0
            hi = t_hi
            t1 = hi - gr_abs * (hi - lo)
            t2 = lo + gr_abs * (hi - lo)
            f1 = J_t(t1)
            f2 = J_t(t2)
            
            for _ in range(40):
                if f1 < f2:
                    hi = t2
                    t2 = t1
                    f2 = f1
                    t1 = hi - gr_abs * (hi - lo)
                    f1 = J_t(t1)
                else:
                    lo = t1
                    t1 = t2
                    f1 = f2
                    t2 = lo + gr_abs * (hi - lo)
                    f2 = J_t(t2)
            
            t_opt = (lo + hi) / 2.0
            J_opt = J_t(t_opt)
            
            if J_opt < J_cur - 1e-12:
                a[i] = t_opt
                pass_improvement += (J_cur - J_opt)
            
            # Update b and auxiliary arrays (incremental update)
            # b changes at positions where a[i] contributed
            # Instead of updating b in place (error-prone), just mark that
            # we'll recompute b at the start of the next pass
            # For now, update b locally for subsequent coordinates in this pass
            delta = t_opt - ai
            if abs(delta) > 1e-12:
                for k in range(k_lo, k_hi + 1):
                    d = k - i
                    if k == 2 * i:

                    else:
                        b[k] += 2 * a[d] * delta

                # This is getting complicated. Let's just recompute at the start of each sub-block.

        current_J = 2.0 * n * b.max() / 1.0  # sum = 1

        tracker.update(a)

            print(f"    Pass {p}: J={current_J:.8f}, improvement={pass_improvement:.2e}", flush=True)

            break
    
    return a

Minimizes J(a) = 2n*max(convolve(a,a))/(sum(a))^2 using:
1. Fast vectorized coordinate descent with golden-section 1D minimization
2. Diverse multi-start from structured initializations (chirp, cosine, gaussian, etc.)

4. LSE smooth gradient polish
5. Multi-length exploration with parent resampling
6. Sparse LP best-response for final escape attempts

import numpy as np
from scipy.optimize import linprog
from scipy.sparse import csr_matrix
import time

def fast_conv(a):
    """Compute linear convolution a*a using FFT."""
    n = len(a)
    s = 1
    while s < 2 * n - 1:
        s <<= 1
    fa = np.fft.rfft(a, s)
    return np.fft.irfft(fa * fa, s)[:2 * n - 1]

def compute_J(a):
    """Compute the objective value J(a)."""
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))

class ScoreTracker:
    def __init__(self):

        self.best_score = float('inf')

            print(f"  {label} best: J={self.best_score:.8f} n={self.best.size}", flush=True)

def coordinate_descent(a, max_passes, deadline, tracker):
    """
    Fast coordinate descent with golden-section 1D minimization.
    For each coordinate i, find optimal t = a[i] minimizing J, holding others fixed.
    """
    a = a.copy().astype(np.float64)
    n = len(a)
    s = a.sum()
    if s < 1e-10:
        return a
    a /= s
    
    current_J = compute_J(a)
    tracker.update(a)
    
    rng = np.random.default_rng(42)
    gr_abs = (5.0 ** 0.5 - 1.0) / 2.0  # golden ratio conjugate ~0.618
    
    for p in range(max_passes):
        if time.monotonic() > deadline:
            break
        
        b = fast_conv(a)
        prefix_max = np.maximum.accumulate(b)
        suffix_max = np.maximum.accumulate(b[::-1])[::-1]
        
        order = rng.permutation(n)
        pass_improvement = 0.0
        
        for i in order:
            if time.monotonic() > deadline:
                break
            
            k_lo = i
            k_hi = min(2 * n - 2, i + n - 1)
            length = k_hi - k_lo + 1
            if length <= 1:
                continue
            
            max_out = 0.0
            if k_lo > 0:
                max_out = float(prefix_max[k_lo - 1])
            if k_hi < 2 * n - 2:
                m2 = float(suffix_max[k_hi + 1])
                if m2 > max_out:
                    max_out = m2
            
            ai = a[i]
            sum_base = 1.0 - ai
            
            ks = np.arange(k_lo, k_hi + 1)
            ds = ks - i  # = k - i, valid since k in [i, i+n-1] => k-i in [0, n-1]
            
            # base[k] = b[k] minus a[i]'s current contribution
            base = b[ks].copy()
            mask_2i = (ks == 2 * i)
            base[~mask_2i] -= 2.0 * ai * a[ds[~mask_2i]]
            if mask_2i.any():
                base[mask_2i] -= ai * ai
            
            # Linear coefficient for t
            c_coef = 2.0 * a[ds].copy()
            c_coef[mask_2i] = 0.0
            
            # Golden section search range
            t_hi = max(4.0 * ai, 8.0 / n) + 0.01
            lo = 0.0
            hi = t_hi
            
            # Precompute for fast evaluation
            has_2i = mask_2i.any()
            
            def eval_J(t):
                new_vals = base + c_coef * t
                if has_2i:
                    new_vals = new_vals.copy()
                    new_vals[mask_2i] += t * t
                m = max(max_out, float(new_vals.max()))
                s_new = sum_base + t
                if s_new < 1e-10:
                    return float('inf')
                return 2.0 * n * m / (s_new * s_new)
            
            f_cur = eval_J(ai)
            
            t1 = hi - gr_abs * (hi - lo)
            t2 = lo + gr_abs * (hi - lo)
            f1 = eval_J(t1)
            f2 = eval_J(t2)
            
            for _ in range(28):
                if f1 < f2:
                    hi = t2
                    t2 = t1
                    f2 = f1
                    t1 = hi - gr_abs * (hi - lo)
                    f1 = eval_J(t1)
                else:
                    lo = t1
                    t1 = t2
                    f1 = f2
                    t2 = lo + gr_abs * (hi - lo)
                    f2 = eval_J(t2)
            
            t_opt = (lo + hi) * 0.5
            f_opt = eval_J(t_opt)
            
            if f_opt < f_cur - 1e-13:
                a[i] = t_opt
                pass_improvement += (f_cur - f_opt)
                # Update b locally for subsequent coordinates
                delta = t_opt - ai
                if abs(delta) > 1e-14:
                    b[~mask_2i & (np.arange(2*n-1) >= k_lo) & (np.arange(2*n-1) <= k_hi)] += 0  # placeholder
                    for k in range(k_lo, k_hi + 1):
                        d = k - i
                        if k == 2 * i:
                            b[k] += 2.0 * ai * delta + delta * delta
                        else:
                            b[k] += 2.0 * a[d] * delta
        
        current_J = compute_J(a)
        tracker.update(a)
        
        if pass_improvement < 1e-14:
            break
    
    return a

def subgradient_descent(a, max_iters, deadline, tracker):
    """Subgradient descent on the exact max of convolution."""
    a = a.copy().astype(np.float64)
    n = len(a)
    a /= a.sum()
    current_J = compute_J(a)
    tracker.update(a)
    
    lr = 2.0 / (n ** 0.5)
    best_lr = lr
    rng = np.random.default_rng(999)
    
    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        
        b = fast_conv(a)
        M = float(b.max())
        active = np.where(b >= M - 1e-12 * max(M, 1e-10))[0]
        
        # Compute subgradient
        grad = np.zeros(n)
        for k in active:
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            for j in range(js, je + 1):
                d = k - j
                if k == 2 * j:
                    grad[j] += 2.0 * a[j]
                else:
                    grad[j] += 2.0 * a[d]
        grad /= len(active)
        
        # Full gradient of J (with sum normalization)
        # J = 2n * M / s^2, direction = grad_M - 2*M*a/s (after normalizing s=1)
        direction = grad - 2.0 * M * a
        
        a_new = a - lr * direction
        a_new = np.maximum(a_new, 1e-10)
        a_new /= a_new.sum()
        
        J_new = compute_J(a_new)
        if J_new < current_J - 1e-15:
            a = a_new
            current_J = J_new
            best_lr = lr
            lr = min(lr * 1.005, 10.0 / n)
        else:
            lr *= 0.8
            if lr < best_lr * 0.01:
                lr = best_lr * 0.5
        
        if it % 100 == 0:
            tracker.update(a)
    
    tracker.update(a)
    return a

def lse_gradient_descent(a, max_iters, deadline, tracker):
    """LSE (Log-Sum-Exp) smooth approximation gradient descent."""
    a = a.copy().astype(np.float64)
    n = len(a)
    a /= a.sum()
    current_J = compute_J(a)
    tracker.update(a)
    
    tau = 80.0
    lr = 1.0 / (n * tau)
    momentum = 0.8
    v = np.zeros(n)
    
    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        
        b = fast_conv(a)
        b_max = b.max()
        exp_b = np.exp(tau * (b - b_max))
        p = exp_b / exp_b.sum()
        
        # Gradient of LSE w.r.t. a: 2 * cross_correlation(p, a)
        s = 1
        while s < 3 * n:
            s <<= 1
        fp = np.fft.rfft(p, s)
        fa = np.fft.rfft(a, s)
        corr = np.fft.irfft(fp * np.conj(fa), s)
        grad = 2.0 * corr[:n]
        
        v = momentum * v + (1.0 - momentum) * grad
        a_new = a - lr * v
        a_new = np.maximum(a_new, 1e-8)
        a_new /= a_new.sum()
        
        J_new = compute_J(a_new)
        if J_new < current_J - 1e-15:
            a = a_new
            current_J = J_new
            tau = min(500.0, tau * 1.005)
            lr = 1.5 / (n * tau)
        else:
            lr *= 0.6
            v *= 0.3
            if lr < 1e-14:
                break
        
        if it % 200 == 0:
            tracker.update(a)
    
    tracker.update(a)
    return a

def resample_seq(src, dest_n):
    """Resample a sequence to a different length."""
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)

def generate_inits(n, rng, parent):
    """Generate diverse initializations."""
    inits = []
    t = np.linspace(0, 1, n, endpoint=False)
    
    # Cosine window
    inits.append(0.5 + 0.5 * np.cos(np.pi * t))
    
    # Cosine squared
    inits.append(np.cos(np.pi * (t - 0.5)) ** 2)
    
    # Gaussian
    for sigma in [0.15, 0.25, 0.35]:
        inits.append(np.exp(-((t - 0.5) ** 2) / (2 * sigma ** 2)))
    
    # Triangular
    inits.append(1.0 - np.abs(2.0 * t - 1.0))
    
    # Chirp-like (quadratic phase)
    for freq in [1.0, 2.0, 3.0]:
        inits.append(1.0 + 0.4 * np.cos(2.0 * np.pi * freq * t ** 2))
    
    # Multi-frequency
    inits.append(1.0 + 0.3 * np.cos(2.0 * np.pi * 3.0 * t) + 0.2 * np.cos(2.0 * np.pi * 7.0 * t))
    inits.append(1.0 + 0.25 * np.sin(2.0 * np.pi * 2.0 * t) + 0.2 * np.sin(2.0 * np.pi * 5.0 * t)
                + 0.15 * np.sin(2.0 * np.pi * 9.0 * t))
    
    # Random smooth
    for _ in range(4):
        n_coeff = min(n // 2 + 1, 8 + rng.integers(0, 8))
        coeffs = rng.normal(0, 1, n_coeff)
        phases = rng.uniform(0, 2 * np.pi, n_coeff)
        sig = np.zeros(n)
        for i in range(n_coeff):
            sig += coeffs[i] * np.cos(2.0 * np.pi * i * t + phases[i])
        inits.append(np.abs(sig) + 0.5)
    
    # Random with smoothing
    for _ in range(3):
        raw = rng.uniform(0.2, 1.0, n)
        kernel_size = max(3, n // 20)
        kernel = np.ones(kernel_size) / kernel_size
        inits.append(np.convolve(raw, kernel, mode='same'))
    
    # Power law
    inits.append((t + 0.05) ** rng.uniform(-1.5, 1.5))
    inits.append((1.0 - t + 0.05) ** rng.uniform(-1.5, 1.5))
    inits.append(1.0 + 0.3 * ((t - 0.3) ** 2))
    inits.append(1.0 + 0.3 * ((t - 0.6) ** 2))
    
    # Bimodal
    inits.append(np.exp(-8 * (t - 0.3) ** 2) + np.exp(-8 * (t - 0.7) ** 2) + 0.1)
    inits.append(np.exp(-10 * (t - 0.2) ** 2) + np.exp(-10 * (t - 0.8) ** 2) + 0.1)
    
    # Flat spectrum (random phases)
    for _ in range(3):
        half = n // 2 + 1
        angles = rng.uniform(0, 2 * np.pi, half)
        A = np.zeros(n, dtype=complex)
        A[:half] = np.exp(1j * angles)
        for k in range(1, n - n // 2):
            A[k] = np.conj(A[n - k])
        a = np.fft.ifft(A).real
        inits.append(np.maximum(a, 0.0) + 0.1)
    
    # Parent resampled
    if parent is not None and parent.size > 0:
        inits.append(resample_seq(parent, n))
        inits.append(resample_seq(parent[::-1], n))
        # Perturbed parent
        p_res = resample_seq(parent, n)
        p_res = p_res + rng.normal(0, 0.05, n)
        inits.append(np.maximum(p_res, 0.01))
    
    # Ensure all valid
    final_inits = []
    for x in inits:
        x = np.maximum(x, 0.001).astype(np.float64)
        s = x.sum()
        if s > 1e-10:
            x /= s
            final_inits.append(x)
    
    return final_inits

def lp_best_response(a, M, time_limit=5):
    """LP to find best response: maximize sum(b) s.t. a*b <= M, b >= 0."""
    n = len(a)
    # Build sparse matrix: for each lag k, the constraint is sum_j a[j]*b[k-j] <= M
    # But this couples a and b. The standard trick: fix a, optimize b.
    # (a*b)[k] = sum_j a[j] * b[k-j]
    # For fixed a, this is linear in b.
    rows = []
    cols = []
    for k in range(2 * n - 1):
        for j in range(max(0, k - n + 1), min(n, k + 1)):
            b_idx = k - j
            if 0 <= b_idx < n:
                rows.append(k)
                cols.append(b_idx)
    
    rows = np.array(rows)
    cols = np.array(cols)
    vals = np.array([a[j] for k in range(2 * n - 1) for j in range(max(0, k - n + 1), min(n, k + 1)) if 0 <= k - j < n])
    
    A = csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    
    result = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                     options={'presolve': True, 'time_limit': time_limit})
    if result.success and result.x is not None:
        g = np.maximum(result.x, 0.0)
        if g.sum() > 1e-10:
            return g / g.sum()
    return None

def lp_escape(a, deadline, tracker, n_iters=20):
    """Use LP best-response to escape local minima."""
    a = a.copy().astype(np.float64)
    a /= a.sum()
    n = len(a)
    current_J = compute_J(a)
    tracker.update(a)
    
    for it in range(n_iters):
        if time.monotonic() > deadline:
            break
        b = fast_conv(a)
        M = float(b.max())
        # Try tightening
        M_tight = M * 0.99
        g = lp_best_response(a, M_tight, time_limit=3)
        if g is None:
            g = lp_best_response(a, M, time_limit=3)
        if g is None:
            break
        
        # Try several mixing ratios
        best_J = current_J
        best_a = a
        for t in [0.3, 0.5, 0.7, 0.9]:
            a_trial = (1 - t) * a + t * g
            a_trial = np.maximum(a_trial, 0)
            s = a_trial.sum()
            if s < 1e-10:
                continue
            a_trial /= s
            J = compute_J(a_trial)
            if J < best_J:
                best_J = J
                best_a = a_trial
        
        if best_a is a and best_J >= current_J:
            break
        a = best_a
        current_J = best_J
        tracker.update(a)
    
    return a

    if best_phase1 is not None:
        tracker.update(best_phase1)
    
    # ============ Phase 2: Deep polish on best ============
    if tracker.best is not None and time.monotonic() < deadline - 300:
        print(f"\n{'='*60}", flush=True)
        print(f"Phase 2: Deep polish, n={tracker.best.size}", flush=True)
        print(f"{'='*60}", flush=True)
        
        a = tracker.best.copy()
        t_end2 = deadline - 300
        
        # Extended coordinate descent
        a = coordinate_descent(a, max_passes=500, deadline=min(t_end2 - 60, time.monotonic() + 120), tracker=tracker)
        
        # Subgradient
        if time.monotonic() < t_end2 - 40:
            a = subgradient_descent(a, max_iters=3000, deadline=min(t_end2 - 40, time.monotonic() + 80), tracker=tracker)
        
        # LSE polish
        if time.monotonic() < t_end2 - 10:
            a = lse_gradient_descent(a, max_iters=2000, deadline=t_end2 - 5, tracker=tracker)
        
        # Final coordinate descent
        if time.monotonic() < t_end2:
            a = coordinate_descent(a, max_passes=100, deadline=t_end2, tracker=tracker)
        
        tracker.report("Phase 2")
    
    # ============ Phase 3: Try larger lengths ============
    if time.monotonic() < deadline - 200:
        print(f"\n{'='*60}", flush=True)
        print(f"Phase 3: Larger lengths", flush=True)
        print(f"{'='*60}", flush=True)
        
        base = tracker.best if tracker.best is not None else (parent if parent is not None else np.ones(512) / 512)
        
        larger_lengths = [640, 768, 1024]
        t_avail = deadline - time.monotonic() - 100
        time_per_len = t_avail / len(larger_lengths)
        
        for n_large in larger_lengths:
            if time.monotonic() > deadline - 100:
                break
            t_end3 = time.monotonic() + time_per_len
            t_end3 = min(t_end3, deadline - 100)
            
            print(f"  Trying n={n_large}...", flush=True)
            
            # Generate inits from resampled base
            inits_large = []
            inits_large.append(resample_seq(base, n_large))
            inits_large.append(resample_seq(base[::-1], n_large))
            
            # Also add some native inits
            t = np.linspace(0, 1, n_large, endpoint=False)
            inits_large.append(0.5 + 0.5 * np.cos(np.pi * t))
            inits_large.append(np.exp(-((t - 0.5) ** 2) / (2 * 0.3 ** 2)))
            inits_large.append(1.0 + 0.3 * np.cos(2 * np.pi * 3.0 * t ** 2))
            
            for x in inits_large:
                x = np.maximum(x, 0.001)
                x /= x.sum()
            
            best_large = None
            best_J_large = float('inf')
            
            time_per_init_l = (t_end3 - time.monotonic()) / len(inits_large)
            time_per_init_l = max(5.0, time_per_init_l)
            
            for idx, a0 in enumerate(inits_large):
                if time.monotonic() > t_end3 - 5:
                    break
                t_end_init = min(time.monotonic() + time_per_init_l, t_end3 - 3)
                a = coordinate_descent(a0, max_passes=100, deadline=t_end_init, tracker=tracker)
                J = compute_J(a)
                if J < best_J_large:
                    best_J_large = J
                    best_large = a.copy()
            
            if best_large is not None:
                print(f"  n={n_large}: best J={best_J_large:.8f}", flush=True)
                tracker.update(best_large)
    
    # ============ Phase 4: LP escape attempts ============
    if tracker.best is not None and time.monotonic() < deadline - 60:
        print(f"\n{'='*60}", flush=True)
        print(f"Phase 4: LP escape", flush=True)
        print(f"{'='*60}", flush=True)
        
        t_end4 = deadline - 30
        a = tracker.best.copy()
        a = lp_escape(a, deadline=t_end4, tracker=tracker, n_iters=10)
        
        # Polish after LP
        if time.monotonic() < t_end4 - 5:
            a = coordinate_descent(a, max_passes=50, deadline=t_end4 - 2, tracker=tracker)
        
        tracker.report("Phase 4")
    
    # ============ Phase 5: Final perturb-and-reoptimize ============
    if tracker.best is not None and time.monotonic() < deadline - 30:
        print(f"\n{'='*60}", flush=True)
        print(f"Phase 5: Final perturb and reoptimize", flush=True)
        print(f"{'='*60}", flush=True)
        
        t_end5 = deadline - 5
        rng2 = np.random.default_rng(777)
        a = tracker.best.copy()
        n = a.size
        t_avail = t_end5 - time.monotonic()
        n_attempts = max(3, int(t_avail / 20))
        
        for attempt in range(n_attempts):
            if time.monotonic() > t_end5 - 10:
                break
            # Perturb
            noise_scale = 0.02 + 0.03 * (attempt / max(1, n_attempts))
            a_pert = a + rng2.normal(0, noise_scale, n)
            a_pert = np.maximum(a_pert, 1e-6)
            a_pert /= a_pert.sum()
            
            # Re-optimize
            t_end_att = min(time.monotonic() + t_avail / n_attempts - 2, t_end5 - 5)
            a_pert = coordinate_descent(a_pert, max_passes=50, deadline=t_end_att, tracker=tracker)
        
        tracker.report("Phase 5")
    
    # Final
    tracker.report("FINAL")
    print(f"\nFinal: J={tracker.best_score:.8f}, n={tracker.best.size if tracker.best is not None else 0}", flush=True)
    
    if tracker.best is None:
        # Fallback
        fallback = np.ones(256) / 256
        tracker.best = fallback
        tracker.best_score = compute_J(fallback)
    
    return [float(x) for x in tracker.best]
