def compute_conv(a):
    """Compute full linear convolution a*a using FFT."""
    n = len(a)
    if n == 1:
        return np.array([a[0]**2])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    conv = irfft(fa * fa, size)
    return conv[:2 * n - 1]

def compute_obj(a):

    sum_a = float(np.sum(a))

def softmax(z, tau=1.0):

def project_simplex(v):
    """Project vector onto probability simplex {x: sum x = 1, x >= 0}."""

    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0)

def softmax_grad(a, tau):
    """
    Compute soft-max of conv(a,a) and its gradient w.r.t. a.
    Returns (soft_max_value, gradient).

    conv = compute_conv(a)
    cmax = float(np.max(conv))
    shifted = np.clip((conv - cmax) / tau, -500.0, 500.0)
    exp_conv = np.exp(shifted)
    Z = float(np.sum(exp_conv))
    soft_max_val = cmax + tau * np.log(Z)
    w = exp_conv / Z
    grad = conv_gradient(a, w)
    return soft_max_val, grad

            y = a + theta * (a - v)
            y = project_simplex(y)

            # = a_j * (grad_a_j - sum_i a_i * grad_a_i)

            return val, grad_z

def build_toeplitz_conv(a):
    """
    Build matrix A (2n-1 x n) such that A @ g = conv(a, g).
    """
    n = len(a)
    m = 2 * n - 1
    A = np.zeros((m, n), dtype=np.float64)
    for j in range(n):
        for k in range(n):
            A[j + k, j] = a[k]
    return A

def solve_lp_max_sum(a, M, time_limit=20):
    """
    Solve LP: max sum(g) s.t. conv(a,g) <= M, g >= 0.
    Returns normalized g and the optimal sum.
    """
    n = len(a)
    m = 2 * n - 1
    # Build A such that (A @ g)[k] = conv(a,g)[k]
    A = np.zeros((m, n), dtype=np.float64)
    for j in range(n):
        A[j:j+n, j] = a[:n]
    
    c = -np.ones(n)
    b_ub = np.full(m, M)
    bounds = [(0.0, None)] * n
    
    try:
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                     options={'presolve': True, 'time_limit': time_limit})

            g = np.maximum(res.x, 0)
            s = float(np.sum(g))
            if s > 1e-10:
                return g / s, s
    except Exception:
        pass
    return None, None

def subgradient_optimize(a_init, max_iters=2000, time_budget=20.0, tau_start=2.0, tau_end=30.0, step_init=0.3):

    best_a = a.copy()

    step = step_init

    for it in range(max_iters):

            break
        
        frac = it / max(1, max_iters)

def random_init(n, seed):

def resample_sequence(a, new_n):
    """Resample sequence to new length via interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, n, endpoint=False)
    x_new = np.linspace(0, 1, new_n, endpoint=False)
    a_new = np.interp(x_new, x_old, a)
    return normalize(a_new)

        a_new, obj_new = lbfgs_softmax(z, tau_list, time_budget=lfbs_time, max_iter=200)
        if obj_new < best_obj:
            best_obj = obj_new
            best_a = a_new.copy()
        a = best_a.copy()
    
    # Step 3: Alternating LP
    remaining = time_budget - (time.monotonic() - start)
    if remaining > 15 and n <= 800:
        lp_time = min(remaining * 0.3, 30.0)
        a_new = alternating_lp(a, max_iters=40, time_budget=lp_time)
        obj = compute_obj(a_new)
        if obj < best_obj:
            best_obj = obj
            best_a = a_new.copy()
        a = best_a.copy()
    
    # Step 4: Subgradient polish
    remaining = time_budget - (time.monotonic() - start)
    if remaining > 5:
        a_new = subgradient_optimize(a, max_iters=int(remaining * 10),
                                      time_budget=min(remaining * 0.3, 10.0),
                                      tau_start=8.0, tau_end=30.0, step_init=0.1)
        obj = compute_obj(a_new)
        if obj < best_obj:
            best_obj = obj
            best_a = a_new.copy()
    
    return best_a, best_obj

def try_parent():
    """Try to load parent sequence."""
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64)
        parent = np.clip(parent, 0, 1000.0)
        if parent.size > 0 and np.isfinite(parent).all():
            obj = compute_obj(parent)
            print(f"Parent (n={parent.size}): obj={obj:.6f}")
            return parent.copy(), obj
    except Exception:
        pass
    return None, np.inf

def propose_candidate(seed=42, budget_s=1500):
    """
    Main search function. Uses multi-phase approach:
    1. Load parent and evaluate
    2. Explore multiple n values with diverse initializations
    3. Polish top candidates
    4. Final intensive optimization on the global best
    """
    start_time = time.monotonic()
    deadline = start_time + float(budget_s) - 30.0
    
    best_seq = None
    best_score = np.inf
    
    # Load parent
    parent, parent_obj = try_parent()
    if parent is not None and parent_obj < best_score:
        best_score = parent_obj
        best_seq = parent.copy()
    
    # N values to explore
    base_n_values = [400, 500, 600, 700, 800, 1000]
    if best_seq is not None:
        np_ = len(best_seq)
        extra = [np_, max(50, np_ - 100), min(1500, np_ + 100), min(1500, np_ + 200), min(1500, np_ + 500)]
        n_values = sorted(set(base_n_values + extra))
    else:
        n_values = base_n_values
    
    print(f"Exploring n values: {n_values}")
    print(f"Initial best: {best_score:.6f}")
    
    candidates = []  # (a, obj, n)
    
    # Phase 1: Exploration
    for n in n_values:
        if time.monotonic() > deadline - 200:
            break
        remaining = deadline - time.monotonic()
        if remaining < 30:
            break
        
        # Time allocation
        if n <= 500:
            alloc = min(40.0, remaining * 0.12)
        elif n <= 800:
            alloc = min(55.0, remaining * 0.14)
        else:
            alloc = min(65.0, remaining * 0.15)
        alloc = max(15.0, alloc)
        
        print(f"\nN={n}: {remaining:.0f}s remaining, alloc={alloc:.0f}s")
        
        n_inits = max(2, int(alloc / 18))
        n_inits = min(n_inits, 5)
        
        for init_idx in range(n_inits):
            if time.monotonic() > deadline - 200:
                break
            remaining_init = deadline - time.monotonic()
            if remaining_init < 15:
                break
            
            # Generate initial point
            if init_idx == 0 and best_seq is not None:
                a = resample_sequence(best_seq, n)

    # Phase 2: Polish top candidates
    print(f"\n=== POLISHING PHASE ===")
    print(f"Best before polish: {best_score:.6f}")
    
    candidates.sort(key=lambda x: x[1])
    
    n_polish = min(4, len(candidates))
    for ci in range(n_polish):

        n_cand = candidates[ci][2]
