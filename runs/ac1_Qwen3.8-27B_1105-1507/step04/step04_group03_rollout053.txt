The dual-guided perturbation idea: after solving the LP, the dual variables y_k tell us which convolution positions are "expensive." We can perturb a in directions that specifically reduce the convolution at those positions.

Specifically, if y_k > 0, it means constraint k is tight. To reduce (a*a)_k, we should decrease a at positions where a[k-i] is large (for i near the peak of g). This gives us a targeted perturbation direction.

def dual_guided_perturb(a, y, g, rng, scale=0.1):
    """Perturb a in a direction guided by LP dual variables."""
    n = len(a)
    b = fast_conv(a)
    M = b.max()
    
    # Weight the perturbation by dual importance
    # Positions where dual is high are the most constrained
    w = np.maximum(y, 0) if y is not None else np.ones(2*n-1)
    w = w / (w.sum() + 1e-10)
    
    # For each active constraint k, the gradient of (a*a)_k w.r.t. a is:
    # grad_k[i] = 2*a[k-i] (for valid i)
    # We want to decrease a where this gradient is large AND the constraint is tight
    perturb = np.zeros(n)
    for k in range(2*n-1):

            j_min = max(0, k - n + 1)
            j_max = min(n - 1, k)
            for j in range(j_min, j_max + 1):

    a_new = np.maximum(a_new, 0.01)
    a_new = a_new / a_new.sum()

import numpy as np
from scipy.optimize import linprog, minimize
from scipy.sparse import csr_matrix
import time


def fast_conv(a):
    """Full self-convolution via FFT."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]


def compute_J(a):
    """Compute J(a) = 2*n*max(conv(a,a))/(sum(a))^2."""
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))


def precompute_A_structure(n):
    """Precompute (row, col) indices for the convolution-mixing matrix."""
    rows = []
    cols = []
    for k in range(2 * n - 1):
        j_min = max(0, k - n + 1)
        j_max = min(n - 1, k)
        for j in range(j_min, j_max + 1):
            rows.append(k)
            cols.append(j)
    return np.array(rows, dtype=np.int32), np.array(cols, dtype=np.int32)


def make_A_sparse(a, rows, cols):
    """Build sparse matrix A where A[k,j] = a[k-j]."""
    values = a[rows - cols]
    n = len(a)
    return csr_matrix((values, (rows, cols)), shape=(2 * n - 1, n))


def lp_best_response(a, M, rows, cols):
    """Solve LP: max sum(g) s.t. A@g <= M, g >= 0. Returns (g, dual_y)."""
    n = len(a)
    A = make_A_sparse(a, rows, cols)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    try:
        r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
        if r.success and r.x is not None:
            g = np.maximum(r.x, 0)
            # Try to extract dual variables
            y = None
            try:
                y = r.ineqlin.marginals
            except (AttributeError, TypeError):
                pass
            if y is None:
                # Fallback: estimate from slacks
                try:
                    slacks = b_ub - A @ r.x
                    y = np.zeros(2 * n - 1)
                    y[slacks < 1e-6 * M] = 1.0
                except Exception:
                    y = None
            return g, y
    except Exception:
        pass
    return None, None


def lp_mirror_descent(a, rows, cols, max_time_s, mode='linear', init_step=0.3, verbose=True):
    """LP mirror descent with adaptive step size. Returns (best_a, best_J)."""
    a = np.maximum(a.astype(np.float64), 1e-12)
    a = a / a.sum()
    n = len(a)
    current_J = compute_J(a)
    best_J = current_J
    best_a = a.copy()
    no_improve = 0
    t_start = time.monotonic()

    it = 0
    while time.monotonic() - t_start < max_time_s:
        b = fast_conv(a)
        M = float(b.max())
        
        g, y = lp_best_response(a, M, rows, cols)
        if g is None:
            break
        g_sum = g.sum()
        if g_sum < 1e-12:
            break
        g = g / g_sum

        # Adaptive step size
        base_t = init_step * np.exp(-0.5 * it / 500.0) + 0.02
        t = min(1.0, base_t)

        improved = False
        for ls in range(30):

                ratio = g / np.maximum(a, 1e-15)
                a_trial = a * np.power(ratio, t)

            a_trial = np.maximum(a_trial, 0.0)
            s = a_trial.sum()
            if s < 1e-10:

                continue
            a_trial = a_trial / s

        else:

def block_perturb(a, rng, n_blocks=3):
    """Apply block-wise perturbations to escape local minima."""
    n = len(a)
    a_new = a.copy()
    for _ in range(n_blocks):
        start = rng.integers(0, max(1, n - 10))
        length = rng.integers(5, max(10, n // 10))
        length = min(length, n - start)
        factor = rng.uniform(0.5, 1.5)
        a_new[start:start+length] *= factor
    a_new = np.maximum(a_new, 0.01)
    a_new = a_new / a_new.sum()
    return a_new


def resample_seq(src, dest_n):
    """Resample 1D sequence to new length."""
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)

"""
LP-centric optimizer for minimizing J(a) = 2*n*max(conv(a,a))/sum(a)^2.
Key innovations: precomputed sparse structure for fast LP, adaptive step scheduling,

import numpy as np
from scipy.optimize import linprog
from scipy.sparse import csr_matrix
import time


def fast_conv(a):
    """Full self-convolution via FFT."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]


def compute_J(a):
    """Compute J(a) = 2*n*max(conv(a,a))/(sum(a))^2."""
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))


def precompute_A_structure(n):
    """Precompute (row, col) indices for the mixing matrix A where A[k,j]=a[k-j]."""
    rows = []
    cols = []
    for k in range(2 * n - 1):
        j_min = max(0, k - n + 1)
        j_max = min(n - 1, k)
        for j in range(j_min, j_max + 1):
            rows.append(k)
            cols.append(j)
    return np.array(rows, dtype=np.int32), np.array(cols, dtype=np.int32)


def make_A_sparse(a, rows, cols):
    """Build sparse matrix with values a[k-j] at precomputed positions."""
    values = a[rows - cols]
    n = len(a)
    return csr_matrix((values, (rows, cols)), shape=(2 * n - 1, n))


def lp_best_response(a, M, rows, cols):
    """Solve LP: max sum(g) s.t. A@g <= M*1, g >= 0. Returns (g, tight_mask)."""
    n = len(a)
    A = make_A_sparse(a, rows, cols)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    try:
        r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
        if r.success and r.x is not None:
            g = np.maximum(r.x, 0)
            # Estimate tight constraints from residuals
            residual = b_ub - A @ r.x
            tight = residual < 1e-4 * max(M, 1e-10)
            return g, tight
    except Exception:
        pass
    return None, None


def lp_mirror_descent(a, rows, cols, max_time_s, mode='linear', init_step=0.35, verbose=False):

    a = np.maximum(a.astype(np.float64), 1e-12)

        M = float(b.max())

        g, tight = lp_best_response(a, M, rows, cols)

def cross_corr_gradient(a):
    """Compute subgradient of max_k (a*a)_k at a using the argmax constraint."""
    n = len(a)
    b = fast_conv(a)
    k_star = int(np.argmax(b))
    # Gradient of (a*a)_{k*} w.r.t. a[i] = 2*a[k*-i] if valid
    grad = np.zeros(n)
    for i in range(n):
        j = k_star - i
        if 0 <= j < n:
            grad[i] = 2.0 * a[j]
    return grad, k_star


def smoothed_gradient_step(a, T, step_size):
    """One step of smoothed gradient descent on max(conv(a,a)) with scale invariance."""
    n = len(a)
    b = fast_conv(a)
    # Softmax weights
    b_shifted = b - b.max()
    p = np.exp(T * b_shifted)
    p = p / p.sum()
    # Gradient of smoothed objective
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        if p[k] > 1e-10:
            j_min = max(0, k - n + 1)
            j_max = min(n - 1, k)
            for j in range(j_min, j_max + 1):
                grad[j] += 2.0 * p[k] * a[k - j]
    # Projected gradient step on simplex
    a_new = a - step_size * grad
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s < 1e-10:
        return a
    a_new = a_new / s
    return a_new


def generate_diverse_inits(n, parent, rng, count=15):
    """Generate diverse initial sequences."""
    inits = []
    t = np.arange(n, dtype=np.float64) / n

    if parent is not None:
        if len(parent) == n:
            p = parent.copy()
        else:
            t_old = np.linspace(0, 1, len(parent))
            p = np.interp(t, t_old, parent)
        inits.append(np.maximum(p, 0.01) / max(np.maximum(p, 0.01).sum(), 1e-10))
        # Perturbed parent
        for scale in [0.05, 0.1, 0.2]:
            a_p = p + rng.normal(0, scale, n)
            a_p = np.maximum(a_p, 0.01)
            inits.append(a_p / a_p.sum())

    # Fourier modes
    for freq in [1, 2, 3, 4, 5, 7, 8, 12]:
        amp = rng.uniform(0.15, 0.45)
        phase = rng.uniform(0, 2 * np.pi)
        a = 1.0 + amp * np.cos(2 * np.pi * freq * t + phase)
        a = np.maximum(a, 0.05)
        inits.append(a / a.sum())

    # Two-tone
    for f1, f2 in [(1, 3), (2, 5), (1, 7), (3, 8)]:
        a = 1.0 + 0.2 * np.cos(2 * np.pi * f1 * t) + 0.15 * np.cos(2 * np.pi * f2 * t + rng.uniform(0, 3))
        a = np.maximum(a, 0.05)
        inits.append(a / a.sum())

    # Gaussian mixtures
    for _ in range(4):
        centers = rng.uniform(0.1, 0.9, 2)
        widths = rng.uniform(0.08, 0.25, 2)
        a = np.zeros(n)
        for c, w in zip(centers, widths):
            a += np.exp(-((t - c) ** 2) / (2 * w ** 2))
        a += 0.1
        inits.append(a / a.sum())

    # Random smooth
    k = max(3, n // 25)
    kernel = np.ones(k) / k
    a = np.convolve(rng.uniform(0.3, 1.5, n), kernel, mode='same')
    inits.append(a / a.sum())

    # W-shaped
    a = 0.5 + 0.4 * np.cos(4 * np.pi * t)
    a = np.maximum(a, 0.05)
    inits.append(a / a.sum())

    # Trim to count
    result = []
    for a in inits:
        a = np.asarray(a, dtype=np.float64)
        if a.size != n:
            continue
        a = np.maximum(a, 0.01)
        s = a.sum()
        if s > 1e-6:
            result.append(a / s)
        if len(result) >= count:
            break
    return result


def block_perturb(a, rng, n_blocks=3):
    """Apply block-wise multiplicative perturbations."""
    n = len(a)
    a_new = a.copy()
    for _ in range(n_blocks):
        start = int(rng.integers(0, max(1, n - 5)))
        length = int(rng.integers(5, max(8, n // 12)))
        length = min(length, n - start)
        factor = rng.uniform(0.4, 1.8)
        a_new[start:start + length] *= factor
    a_new = np.maximum(a_new, 0.01)
    a_new = a_new / a_new.sum()
    return a_new


def smooth_perturb(a, rng):
    """Apply smooth low-frequency perturbation."""
    n = len(a)
    t = np.arange(n, dtype=np.float64) / n
    freq = rng.uniform(1, 6)
    amp = rng.uniform(0.05, 0.2)
    phase = rng.uniform(0, 2 * np.pi)
    a_new = a * (1 + amp * np.cos(2 * np.pi * freq * t + phase))
    a_new = np.maximum(a_new, 0.01)
    a_new = a_new / a_new.sum()
    return a_new


def resample_seq(src, dest_n):
    """Resample 1D sequence to new length."""
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)

            a_res, J_res = lp_mirror_descent(a_try, rows_t, cols_t, per_len * 0.85, 'linear', init_step=0.3)
            tracker.update(a_res)
            tag = " <== NEW BEST" if J_res < current_best_J else ""
            print(f"  n={n_try}: J={J_res:.8f}{tag}", flush=True)
            if J_res < current_best_J:
                current_best = a_res.copy()
                current_best_n = n_try
                current_best_J = J_res

        print(f"Phase 6 done ({time.monotonic()-t0:.1f}s)", flush=True)

    # ================================================================
    # Phase 7: Final extended LP (remaining time)
    # ================================================================
    if time.monotonic() < deadline - 5:
        print(f"\nPhase 7: Final LP (n={current_best_n})", flush=True)
        t0 = time.monotonic()
        n7 = current_best_n
        rows7, cols7 = get_structure(n7)
        remaining = deadline - time.monotonic() - 5
        a7, J7 = lp_mirror_descent(current_best, rows7, cols7, remaining * 0.75, 'linear', init_step=0.15)
        tracker.update(a7)
        print(f"  linear: J={J7:.8f} ({time.monotonic()-t0:.1f}s)", flush=True)
        if J7 < current_best_J:
            current_best = a7.copy()
            current_best_J = J7

        if time.monotonic() < deadline - 3:
            a7b, J7b = lp_mirror_descent(current_best, rows7, cols7, deadline - time.monotonic() - 2, 'multiplicative', init_step=0.1)
            tracker.update(a7b)
            print(f"  mult: J={J7b:.8f}", flush=True)
            if J7b < current_best_J:
                current_best = a7b.copy()
                current_best_J = J7b

    # Final output
    final_a = tracker.best if tracker.best is not None else current_best
    print(f"\n{'='*55}", flush=True)
    print(f"FINAL: J={tracker.best_score:.8f} n={final_a.size} total_time={time.monotonic()-tracker.t_start:.0f}s", flush=True)
    print(f"{'='*55}", flush=True)

    return [float(x) for x in final_a]
