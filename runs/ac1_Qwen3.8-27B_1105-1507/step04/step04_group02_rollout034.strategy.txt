def compute_convolve(a):
    """Compute full linear convolution using FFT with 0-padding."""
    n = len(a)
    size = 2 * n - 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)  # First 2*n-1 values

def compute_objective(a):
    """Compute the evaluation objective for sequence a."""
    n = len(a)
    conv = compute_convolve(a)
    M = float(np.max(conv))
    S = np.sum(a)
    return float(2 * n * M / (S**2)) if abs(S) > 1e-15 else np.inf

def cosine_init(n, n_modes=8, seed=0.0):
    """Generate smooth normalized cosine initialization."""
    seq = np.ones(n)
    rng = np.random.RandomState(int(seed))  # Hash seed to deterministic seed
    for m in range(1, n_modes+1):
        c = rng.uniform(-0.6, 0.6) / m
        phase = rng.uniform(0, 2*np.pi)
        for k in range(n):
            seq[k] += c * np.cos(2*np.pi*m*k/n + phase)
    seq = np.clip(seq, 0, None) / np.sum(seq)  # Normalize to sum 1
    return seq

def raised_cosine_init(n, exponent=1.2, seed=0.0):
    """Generate normalized raised cosine initialization."""

    seed_rng = np.random.RandomState(int(seed))

    seq = seq**(1/spread)
    seq = seq / np.sum(seq)
    return seq
