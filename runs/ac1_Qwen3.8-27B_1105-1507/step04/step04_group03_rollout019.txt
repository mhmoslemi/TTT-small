Key improvements over previous:

import numpy as np

def fast_conv(a):
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]


def compute_J(a):
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))


def build_constraint_matrix(n, a, k_indices=None):
    """Build sparse constraint matrix for LP. A[k,j] = a[k-j]."""
    if k_indices is None:
        k_indices = range(2 * n - 1)
    else:
        k_indices = list(k_indices)
    n_k = len(k_indices)
    rows = np.zeros(n_k * n, dtype=np.int32)
    cols = np.zeros(n_k * n, dtype=np.int32)
    vals = np.zeros(n_k * n)
    count = 0
    for idx, k in enumerate(k_indices):
        j_min = max(0, k - n + 1)
        j_max = min(n - 1, k)
        for j in range(j_min, j_max + 1):
            rows[count] = idx
            cols[count] = j
            vals[count] = a[k - j]
            count += 1
    return csr_matrix((vals[:count], (rows[:count], cols[:count])), shape=(n_k, n))


def lp_best_response_fast(a, M, active_indices, use_all=False):
    """Fast LP: max sum(g) s.t. A@g <= M*1, g >= 0. Uses active set or all lags."""
    n = len(a)
    if use_all or len(active_indices) > n:
        A = build_constraint_matrix(n, a)
        b_ub = np.full(2 * n - 1, M)
    else:
        A = build_constraint_matrix(n, a, active_indices)
        b_ub = np.full(len(active_indices), M)
    c = -np.ones(n)
    bounds = [(0, None)] * n
    try:
        r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                    options={'presolve': True, 'dual_feasibility_tolerance': 1e-9})
        if r.success and r.x is not None:
            return np.maximum(r.x, 0)
    except Exception:
        pass
    return None


def lp_mirror_descent_v2(a, max_iters, deadline, use_multiplicative=False,
                         active_tol=1e-4, verbose=False):
    """Improved LP mirror descent with adaptive step size and active set."""
    a = np.maximum(a.astype(np.float64), 1e-12)
    a = a / a.sum()
    current_J = compute_J(a)
    best_J = current_J
    best_a = a.copy()
    no_improve = 0
    step_scale = 1.0

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        b = fast_conv(a)
        M = float(b.max())

        # Active set: lags within tolerance of max
        threshold = M - active_tol * M
        active = np.where(b >= threshold)[0].tolist()
        # Always include the global max
        global_max_idx = int(np.argmax(b))
        if global_max_idx not in active:
            active.append(global_max_idx)

        # Use all constraints if active set is large
        use_all = len(active) > n * 0.8
        g = lp_best_response_fast(a, M, active, use_all=use_all)
        if g is None:
            g = lp_best_response_fast(a, M, active, use_all=True)
            if g is None:
                break

        g = np.maximum(g, 0.0)
        g_sum = g.sum()
        if g_sum < 1e-10:
            break

        # Dual gap: how much improvement is possible
        dual_gap = g_sum / a.sum() - 1.0
        if dual_gap < 1e-10:
            no_improve += 1
            if no_improve > 20:
                break
            # Try with all constraints for confirmation
            g_all = lp_best_response_fast(a, M, active, use_all=True)
            if g_all is not None and g_all.sum() > g_sum + 1e-8:
                g = g_all
                g_sum = g.sum()
                dual_gap = g_sum / a.sum() - 1.0
                if dual_gap < 1e-10:
                    continue
            else:
                if no_improve > 50:
                    break
                continue

        # Adaptive step size based on dual gap
        if dual_gap < 0.01:
            t = min(1.0, 0.1 + 0.9 * dual_gap / 0.01)
        elif dual_gap < 0.1:
            t = 0.4
        else:
            t = 0.7
        t = t * step_scale

        g = g / g_sum

        # Line search
        best_trial_J = current_J
        best_trial_a = None
        for ls in range(15):
            if use_multiplicative:
                ratio = g / np.maximum(a, 1e-15)
                a_trial = a * np.power(ratio, t)
            else:
                a_trial = (1.0 - t) * a + t * g
            a_trial = np.maximum(a_trial, 0.0)
            s = a_trial.sum()
            if s < 1e-10:
                t *= 0.5
                continue
            a_trial = a_trial / s
            J_trial = compute_J(a_trial)
            if J_trial < best_trial_J - 1e-14:
                best_trial_J = J_trial
                best_trial_a = a_trial.copy()
                break
            t *= 0.6
            if t < 0.001:
                break

        if best_trial_a is not None:
            a = best_trial_a
            current_J = best_trial_J
            if current_J < best_J - 1e-12:
                best_J = current_J
                best_a = a.copy()
                no_improve = 0
            else:
                no_improve += 1
        else:
            no_improve += 1
            step_scale *= 0.9
            if no_improve > 100:
                break

    return best_a, best_J


def spectral_reshape(a, rng, strength=0.3):
    """Reshape sequence in frequency domain to reduce peak."""
    n = len(a)
    A = np.fft.rfft(a)
    # Compute power spectrum
    P = np.abs(A) ** 2
    # Subtract a smoothed version to create a "flattening" correction
    P_smooth = np.convolve(P, np.ones(5) / 5, mode='same')
    # Blend: reduce peaks in spectrum
    P_new = P * (1 - strength) + P_smooth * strength
    # Also add a small random perturbation in phase
    phase = np.angle(A) + rng.normal(0, 0.01, len(A))
    A_new = np.sqrt(np.maximum(P_new, 0)) * np.exp(1j * phase)
    a_new = np.fft.irfft(A_new, 2 * n)
    a_new = a_new[:n]
    a_new = np.maximum(a_new, 0)
    s = a_new.sum()
    if s < 1e-8:
        return a.copy()
    return a_new / s


def kkt_direct_solve(a, deadline, max_iters=50):
    """Try to solve KKT conditions directly for final polish."""
    n = len(a)
    a = np.maximum(a, 1e-12)
    a = a / a.sum()
    best_a = a.copy()
    best_J = compute_J(a)

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        b = fast_conv(a)
        M = b.max()
        # Active set with tight tolerance
        tol = 1e-8 * M
        active = np.where(b >= M - tol)[0]

        if len(active) == 0:
            active = np.array([int(np.argmax(b))])

        # Compute "pressure" vector: p_i = sum_{k in active} a[k-i]
        p = np.zeros(n)
        for k in active:
            j_min = max(0, k - n + 1)
            j_max = min(n - 1, k)
            for j in range(j_min, j_max + 1):
                p[j] += a[k - j]

        # Target: p should be constant (= lambda) for support of a
        # Use multiplicative update: a_i *= (lambda / p_i)^t
        # Estimate lambda as the mean of p over support
        support = a > 1e-10
        if support.sum() < 2:
            break
        lam = p[support].mean()
        if lam < 1e-15:
            break

        t = 0.1 / (1 + 0.1 * it)
        ratio = np.ones(n)
        ratio[support] = np.power(lam / np.maximum(p[support], 1e-15), t)
        a_new = a * ratio
        a_new = np.maximum(a_new, 0)
        s = a_new.sum()
        if s < 1e-10:
            break
        a_new = a_new / s
        J_new = compute_J(a_new)
        if J_new < best_J:
            best_J = J_new
            best_a = a_new.copy()
        a = a_new

    return best_a, best_J


def resample_seq(src, dest_n):
    """Resample 1D sequence to new length using cubic interpolation."""
    src = np.asarray(src, dtype=np.from_pyfloat if False else np.float64)
    if src.size == dest_n:
        return src.copy()
    # Use linear interpolation for simplicity and speed
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    result = np.interp(t_new, t_old, src)
    return np.maximum(result, 0.0)


def generate_structural_inits(n, rng, parent=None):
    """Generate diverse structural initializations."""
    inits = []
    t = np.arange(n, dtype=np.float64) / n

    if parent is not None:
        p = resample_seq(parent, n)
        s = p.sum()
        if s > 1e-8:
            p = p / s
            inits.append(p.copy())
            inits.append(np.maximum(p + rng.normal(0, 0.03, n), 0.001))
            inits.append(np.maximum(p + rng.normal(0, 0.08, n), 0.001))

    # Uniform
    inits.append(np.ones(n) / n)

    # Raised cosine
    inits.append(np.maximum(0.5 * (1 + np.cos(2 * np.pi * t)), 0.001))

    # Two bumps
    inits.append(np.maximum(np.exp(-8 * (t - 0.3) ** 2) + np.exp(-8 * (t - 0.7) ** 2), 0.001))

    # Three bumps
    inits.append(np.maximum(np.exp(-6 * (t - 0.2) ** 2) + np.exp(-6 * (t - 0.5) ** 2) + np.exp(-6 * (t - 0.8) ** 2), 0.001))

    # Smoothed random
    k = max(3, n // 20)
    kernel = np.ones(k) / k
    smooth = np.convolve(rng.uniform(0, 1, n), kernel, mode='same')
    inits.append(np.maximum(smooth, 0.001))

    # Tapered (higher in middle)
    inits.append(np.maximum(1.0 - 0.3 * np.abs(2 * t - 1) ** 2, 0.001))

    # Modulated
    inits.append(np.maximum(1.0 + 0.2 * np.cos(4 * np.pi * t), 0.001))
    inits.append(np.maximum(1.0 + 0.15 * np.cos(6 * np.pi * t + 0.5), 0.001))

    # Shifted concentration
    inits.append(np.maximum(np.exp(-5 * (t - 0.6) ** 2), 0.001))
    inits.append(np.maximum(np.exp(-5 * (t - 0.4) ** 2), 0.001))

    # Normalize all
    result = []
    for a in inits:
        a = np.maximum(a, 0.001).astype(np.float64)
        s = a.sum()
        if s > 1e-6:
            result.append(a / s)
    return result

def subgradient_polish_v2(a, max_iters, deadline):
    """Subgradient descent with momentum and adaptive step."""
    a = np.maximum(a.astype(np.float64), 1e-12)
    a = a / a.sum()
    best_a = a.copy()
    best_J = compute_J(a)
    momentum = np.zeros_like(a)
    beta = 0.8
    step_mult = 1.0

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        b = fast_conv(a)
        M = b.max()
        tol = 1e-9 * M
        active = np.where(b >= M - tol)[0]

        # Subgradient of max at a
        grad = np.zeros_like(a)
        for k in active:
            j_min = max(0, k - len(a) + 1)
            j_max = min(len(a) - 1, k)
            for i in range(j_min, j_max + 1):
                grad[i] += 2.0 * a[k - i]

        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-15:
            break

        t = step_mult / (gnorm * np.sqrt(it + 1.0))
        momentum = beta * momentum + (1 - beta) * grad / gnorm
        a_new = a - t * momentum
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s < 1e-10:
            step_mult *= 0.5
            continue
        a_new = a_new / s
        J_new = compute_J(a_new)
        if J_new < best_J:
            best_J = J_new
            best_a = a_new.copy()
        a = a_new

    return best_a, best_J

    if time.monotonic() < deadline - 500:

        print(f"\n{'='*50}", flush=True)
        print(f"Phase 4: Spectral reshape + LP (n={n_base})", flush=True)
        print(f"{'='*50}", flush=True)

        n_spectral = 4
        per_time = max(20, min(60, (deadline - 500 - time.monotonic()) / n_spectral / 2))

        for i in range(n_spectral):
            if time.monotonic() > deadline - 480:
                break
            strength = 0.1 + 0.15 * i
            a_spec = spectral_reshape(current_best, rng, strength=strength)
            a_spec, J_spec = lp_mirror_descent_v2(a_spec, 1000, time.monotonic() + per_time,
                                                  use_multiplicative=False, active_tol=1e-4)
            tracker.update(a_spec)
            print(f"  [spec {i}, s={strength:.2f}] J={J_spec:.8f}", flush=True)

            if J_spec < current_best_J:
                current_best = a_spec.copy()
                current_best_J = J_spec

        print(f"Phase 4 done ({time.monotonic()-t0:.1f}s)", flush=True)

    # === Phase 5: Multi-length search ===
    if time.monotonic() < deadline - 300:
        t0 = time.monotonic()
        print(f"\n{'='*50}", flush=True)
        print(f"Phase 5: Multi-length (base n={n_base})", flush=True)
        print(f"{'='*50}", flush=True)

        base_n = current_best.size
        # Try a range of lengths around the base
        length_candidates = [base_n + d for d in [-16, -8, 8, 16, 32, -32, 64, -64, 128]
                             if base_n + d >= 64 and base_n + d <= 2048]
        length_candidates = sorted(set(length_candidates))

        remaining = deadline - 300 - time.monotonic()
        per_len_time = max(15, min(60, remaining / max(1, len(length_candidates)) / 2.5))

        for n_try in length_candidates:
            if time.monotonic() > deadline - 280:
                break
            a_try = resample_seq(current_best, n_try)
            a_try = np.maximum(a_try, 1e-10)
            s = a_try.sum()
            if s > 1e-8:
                a_try = a_try / s
            a_try, J_try = lp_mirror_descent_v2(a_try, 2000, time.monotonic() + per_len_time,
                                                use_multiplicative=False, active_tol=1e-4)
            tracker.update(a_try)
            print(f"  n={n_try}: J={J_try:.8f}", flush=True)

            if J_try < current_best_J:
                current_best = a_try.copy()
                current_best_J = J_try

        # Also try some significantly different lengths
        for n_try in [512, 768, 1024]:
            if time.monotonic() > deadline - 260:
                break
            if n_try == base_n:
                continue
            a_try = resample_seq(current_best, n_try)
            a_try = np.maximum(a_try, 1e-10)
            s = a_try.sum()
            if s > 1e-8:
                a_try = a_try / s
            a_try, J_try = lp_mirror_descent_v2(a_try, 1000, time.monotonic() + 30,
                                                use_multiplicative=False, active_tol=1e-4)
            tracker.update(a_try)
            print(f"  n={n_try}: J={J_try:.8f}", flush=True)

            if J_try < current_best_J:
                current_best = a_try.copy()
                current_best_J = J_try

        print(f"Phase 5 done ({time.monotonic()-t0:.1f}s)", flush=True)

    # === Phase 6: Diversified restarts ===
    if time.monotonic() < deadline - 200:
        t0 = time.monotonic()
        print(f"\n{'='*50}", flush=True)
        print(f"Phase 6: Diversified restarts (n={current_best.size})", flush=True)
        print(f"{'='*50}", flush=True)

        n_current = current_best.size
        inits = generate_structural_inits(n_current, rng, parent=current_best)

        remaining = deadline - 200 - time.monotonic()
        per_init_time = max(10, min(40, remaining / max(1, len(inits)) / 2))

        for init_i, a_init in enumerate(inits):
            if time.monotonic() > deadline - 180:
                break
            a_r, J_r = lp_mirror_descent_v2(a_init, 500, time.monotonic() + per_init_time,
                                            use_multiplicative=(init_i % 2 == 0), active_tol=1e-4)
            tracker.update(a_r)
            if J_r < current_best_J:
                current_best = a_r.copy()
                current_best_J = J_r
                print(f"  [init {init_i}] J={J_r:.8f} IMPROVED!", flush=True)

        print(f"Phase 6 done ({time.monotonic()-t0:.1f}s)", flush=True)

    # === Phase 7: Long LP refinement on best found ===
    if time.monotonic() < deadline - 100:
        t0 = time.monotonic()
        print(f"\n{'='*50}", flush=True)
        print(f"Phase 7: Long LP refinement (n={current_best.size})", flush=True)
        print(f"{'='*50}", flush=True)

        a_long = current_best.copy()
        phase7_dl = min(deadline - 60, time.monotonic() + 200)
        a_long, J_long = lp_mirror_descent_v2(a_long, 50000, phase7_dl,
                                              use_multiplicative=False, active_tol=1e-5)
        tracker.update(a_long)
        print(f"  LP linear: J={J_long:.8f} ({time.monotonic()-t0:.1f}s)", flush=True)

        if J_long < current_best_J:
            current_best = a_long.copy()
            current_best_J = J_long

        # Multiplicative on the result
        if time.monotonic() < deadline - 30:
            t1 = time.monotonic()
            a_long2 = current_best.copy()
            a_long2, J_long2 = lp_mirror_descent_v2(a_long2, 20000, deadline - 20,
                                                    use_multiplicative=True, active_tol=1e-5)
            tracker.update(a_long2)
            print(f"  LP multi: J={J_long2:.8f} ({time.monotonic()-t1:.1f}s)", flush=True)
            if J_long2 < current_best_J:
                current_best = a_long2.copy()
                current_best_J = J_long2

    # === Phase 8: KKT + subgradient final polish ===
    if time.monotonic() < deadline - 15:
        t0 = time.monotonic()
        print(f"\n{'='*50}", flush=True)
        print(f"Phase 8: Final polish (n={current_best.size})", flush=True)
        print(f"{'='*50}", flush=True)

        a_final = current_best.copy()
        # KKT solve
        a_kkt2, J_kkt2 = kkt_direct_solve(a_final, min(deadline - 10, time.monotonic() + 30), max_iters=100)
        tracker.update(a_kkt2)
        if J_kkt2 < current_best_J:
            current_best = a_kkt2.copy()
            current_best_J = J_kkt2

        # Subgradient
        a_sub, J_sub = subgradient_polish_v2(current_best, 500, min(deadline - 5, time.monotonic() + 20))
        tracker.update(a_sub)
        if J_sub < current_best_J:
            current_best = a_sub.copy()
            current_best_J = J_sub
        print(f"  Final: J={compute_J(current_best):.8f} ({time.monotonic()-t0:.1f}s)", flush=True)

    # === Phase 9: L-BFGS-B on best ===
    if time.monotonic() < deadline - 3 and tracker.best is not None:
        t0 = time.monotonic()
        n = tracker.best.size
        try:
            res = minimize(lbb_objective, tracker.best.astype(np.float64),
                           method='L-BFGS-B', bounds=[(0.0, None)] * n,
                           options={'maxiter': 200, 'max_fun': 50000, 'ftol': 1e-16,
                                    'gtol': 1e-10, 'maxls': 50})
            if res.fun < 1e10:
                tracker.update(res.x)
        except Exception as e:
            print(f"  L-BFGS-B error: {e}", flush=True)
        print(f"  L-BFGS-B done ({time.monotonic()-t0:.1f}s)", flush=True)

    # Final decision: use tracker best (which uses evaluate_sequence)
    if tracker.best is not None:
        final_a = tracker.best
    else:
        final_a = current_best

    print(f"\n{'='*50}", flush=True)
    print(f"FINAL: J={tracker.best_score:.8f} n={final_a.size} time={time.monotonic()-tracker.t_start:.0f}s", flush=True)
    print(f"{'='*50}", flush=True)

    return [float(x) for x in final_a]
