LP-alternating optimization with KKT-guided updates for minimizing

import numpy as np
from scipy.optimize import linprog
from scipy.fft import rfft, irfft
import time


def compute_conv(a):
    """Compute linear convolution a*a using FFT."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    conv = irfft(fa * fa, size)
    return conv[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    conv = compute_conv(a)
    max_b = float(np.max(conv))
    sum_a = float(np.sum(a))
    if sum_a < 0.01:
        return np.inf
    return 2.0 * n * max_b / (sum_a * sum_a)


def build_toeplitz(a):
    """Build matrix A where (A@g)[k] = (conv(a,g))[k] for k=0..2n-2."""
    n = len(a)
    A = np.zeros((2 * n - 1, n), dtype=np.float64)
    for j in range(n):
        k_start = j
        k_end = min(j + n, 2 * n - 1)
        A[k_start:k_end, j] = a[:k_end - k_start]
    return A


def solve_lp_max_sum(a, M, time_limit=30):
    """Solve LP: max sum(g) s.t. conv(a,g) <= M, g >= 0."""
    n = len(a)
    A = build_toeplitz(a)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': time_limit})
        if res.status == 0:
            g = np.maximum(res.x, 0)
            s = float(np.sum(g))
            if s > 1e-10:
                return g, s, res
    except Exception:
        pass
    return None, None, None


def solve_lp_min_M(a, time_limit=30):
    """Solve LP: min M s.t. conv(a,g) <= M, sum(g)=1, g >= 0."""
    n = len(a)
    A = build_toeplitz(a)
    c = np.zeros(n + 1)
    c[-1] = 1.0
    A_ub = np.zeros((2 * n - 1, n + 1))
    A_ub[:, :n] = A
    A_ub[:, n] = -1.0
    b_ub = np.zeros(2 * n - 1)
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])
    bounds = [(0.0, None)] * n + [(0.0, None)]
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': time_limit})
        if res.status == 0:
            g = np.maximum(res.x[:n], 0)
            s = np.sum(g)
            if s > 1e-10:
                M_opt = float(res.x[n])
                return g / s, M_opt, res
    except Exception:
        pass
    return None, None, None

    conv = compute_conv(a)

    # where y are dual variables. This is hard to check directly, so we just
    # return the spread of the "effective gradient"

def lp_alternate(a, max_iters=100, time_budget=100.0, verbose=False):
    """
    Alternating LP optimization with backtracking line search.
    At each step: fix a, solve LP for optimal g, then blend.
    """
    n = len(a)
    a = a.copy().astype(np.float64)
    s = np.sum(a)
    if s < 1e-10:
        a = np.ones(n) / n
    else:
        a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    stall = 0
    best_stall = 0
    step_scale = 1.0

    for it in range(max_iters):
        if time.monotonic() - start > time_budget:
            break

        conv = compute_conv(a)
        M = float(np.max(conv))

        # Solve LP
        g, s_g, _ = solve_lp_max_sum(a, M, time_limit=10)
        if g is None:
            break
        g = g / s_g  # normalize

        # Backtracking line search
        best_t = 0.0
        best_t_obj = best_obj
        t = step_scale
        while t > 1e-4:
            a_try = (1.0 - t) * a + t * g
            a_try = np.maximum(a_try, 0)
            s_try = np.sum(a_try)
            if s_try < 1e-10:
                t *= 0.5
                continue
            a_try = a_try / s_try
            obj_try = compute_obj(a_try)
            if obj_try < best_t_obj:
                best_t_obj = obj_try
                best_t = t
                best_a_new = a_try.copy()
            t *= 0.7

        if best_t > 0 and best_t_obj < best_obj - 1e-12:
            best_obj = best_t_obj
            best_a = best_a_new.copy()
            a = best_a_new
            stall = 0
            step_scale = min(1.0, best_t * 1.5)
            if verbose:
                print(f"    LP iter {it}: obj={best_obj:.8f}, t={best_t:.3f}")
        else:
            # Try smaller step
            a = 0.95 * a + 0.05 * g
            a = np.maximum(a, 0)
            s = np.sum(a)
            if s > 1e-10:
                a = a / s
            stall += 1
            step_scale = max(0.1, step_scale * 0.8)

            if stall > 30:
                # Perturb and restart
                rng = np.random.default_rng(it * 7919 + 42)
                a = a * np.exp(rng.normal(0, 0.05, n))
                a = np.maximum(a, 0)
                s = np.sum(a)
                if s > 1e-10:
                    a = a / s
                stall = 0
                best_stall += 1
                if best_stall > 2:
                    break

    return best_a, best_obj


def lp_alternate_symmetric(a, max_iters=100, time_budget=100.0, verbose=False):
    """
    Symmetric alternating LP: alternate between fixing a and fixing g.
    This can escape some local minima that one-directional alternation gets stuck in.
    """
    n = len(a)
    a = a.copy().astype(np.float64)
    s = np.sum(a)
    if s < 1e-10:
        a = np.ones(n) / n
    else:
        a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    for it in range(max_iters):
        if time.monotonic() - start > time_budget:
            break

        if it % 2 == 0:
            # Fix a, optimize g
            conv = compute_conv(a)
            M = float(np.max(conv))
            g, s_g, _ = solve_lp_max_sum(a, M, time_limit=8)
            if g is None:
                break
            g = g / s_g
            a = 0.7 * a + 0.3 * g
        else:
            # Fix g (use current a as g), optimize new a
            g = a.copy()
            conv_g = compute_conv(g)
            M = float(np.max(conv_g))
            a_new, s_a, _ = solve_lp_max_sum(g, M, time_limit=8)
            if a_new is None:
                break
            a_new = a_new / s_a
            a = 0.7 * a + 0.3 * a_new

        a = np.maximum(a, 0)
        s = np.sum(a)
        if s < 1e-10:
            break
        a = a / s
        obj = compute_obj(a)
        if obj < best_obj:
            best_obj = obj
            best_a = a.copy()
            if verbose:
                print(f"    SymLP iter {it}: obj={best_obj:.8f}")

    return best_a, best_obj

def generate_structural_init(n, seed, n_parent=None, a_parent=None):
    """Generate diverse initializations, including resampling from parent."""
    rng = np.random.default_rng(seed)
    k = np.arange(n)

    choice = rng.integers(0, 8)

    if choice == 0 and a_parent is not None:
        # Resample from parent
        x_old = np.linspace(0, 1, len(a_parent))
        x_new = np.linspace(0, 1, n)
        a = np.interp(x_new, x_old, a_parent)
    elif choice == 1:
        # Smooth random
        a = rng.random(n)
        win = max(3, n // 15)
        kernel = np.ones(win) / win
        a = np.convolve(a, kernel, mode='same')
    elif choice == 2:
        # Modulated
        f = rng.uniform(0.5, 3.0)
        phase = rng.uniform(0, 2 * np.pi)
        amp = rng.uniform(0.3, 0.7)
        a = 1.0 + amp * np.cos(2 * np.pi * f * k / n + phase)
    elif choice == 3:
        # Multi-frequency
        a = np.ones(n)
        for _ in range(rng.integers(2, 5)):
            f = rng.uniform(0.3, 5.0)
            amp = rng.uniform(0.1, 0.4)
            a += amp * np.cos(2 * np.pi * f * k / n + rng.uniform(0, 2 * np.pi))
    elif choice == 4:
        # Bump
        center = n * rng.uniform(0.2, 0.8)
        width = n * rng.uniform(0.1, 0.3)
        a = 1.0 + 0.5 * np.exp(-0.5 * ((k - center) / width) ** 2)
    elif choice == 5:
        # Block structure
        nblocks = rng.integers(3, 8)
        a = np.ones(n)
        for b in range(nblocks):
            start = b * n // nblocks
            end = (b + 1) * n // nblocks
            a[start:end] = rng.uniform(0.3, 1.8)
    elif choice == 6:

        f0 = rng.uniform(0.5, 2.0)

        a = 1.0 + 0.4 * np.cos(2 * np.pi * (f0 + 0.5 * (f1-f0) * k / n) * k / n)

            a += amp * np.exp(-0.5 * ((k - c) / w) ** 2)
