"""
Minimizes J(a) = 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Newton method with log-sum-exp continuation (tau continuation)
2. Exploits Hankel structure of the Hessian
3. LP alternating direction for polishing
4. Multi-length exploration with diverse initializations
5. Symmetric sequence exploitation

import numpy as np
from scipy.optimize import linprog
from scipy.sparse import csr_matrix
import time

def fast_conv(a):
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]

def compute_J(a):
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))

def softmax_stable(x):
    x_max = x.max()
    exp_x = np.exp(x - x_max)
    return exp_x / exp_x.sum()

def smooth_max_and_grad(a, tau):
    """Compute smooth max value, softmax weights p, and gradient."""
    n = len(a)
    b = fast_conv(a)
    b_max = b.max()
    exp_b = np.exp(tau * (b - b_max))
    p = exp_b / exp_b.sum()
    val = b_max + np.log(exp_b.sum()) / tau
    
    # Gradient: grad[m] = sum_j a[j] * p[m+j] + a[m]
    # (since sum p = 1)
    grad = np.zeros(n)
    for m in range(n):
        grad[m] = np.dot(a, p[m:m+n]) + a[m]
    
    return val, p, grad, b

def smooth_max_hessian(p):
    """Build the Hessian matrix (Hankel structure)."""
    n = len(p) // 2 + (len(p) - 1) % 2  # Actually n = len(p[0..n-1]) gives first col
    # p has length 2n-1
    n = (len(p) + 1) // 2
    return np.hankel(p[:n], p[n-1:])

def newton_step(a, tau, max_iters=100, verbose=False):
    """
    Newton method for smooth max minimization.
    a is normalized to sum=1, a >= 0.
    """
    n = len(a)
    a = np.maximum(a, 1e-10).astype(np.float64)
    a = a / a.sum()
    
    best_a = a.copy()
    best_val = smooth_max_and_grad(a, tau)[0]
    no_improve = 0
    
    for it in range(max_iters):
        val, p, grad, b = smooth_max_and_grad(a, tau)
        
        if val < best_val:
            best_val = val
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1
            if no_improve > 20:
                break
        
        # Build Hessian
        H = np.hankel(p[:n], p[n-1:])
        # Regularize
        H += 1e-6 * np.eye(n)
        
        # Solve for Newton direction
        try:
            d = np.linalg.solve(H, -grad)
        except np.linalg.LinAlgError:
            d = -grad
            d /= (np.linalg.norm(d) + 1e-15)
        
        # Limit step size
        d_norm = np.linalg.norm(d)
        if d_norm > 1.0:
            d /= d_norm
        
        # Backtracking line search
        alpha = 1.0
        f_current = val
        success = False
        for _ in range(30):
            a_trial = a - alpha * d
            a_trial = np.maximum(a_trial, 0)
            s = a_trial.sum()
            if s < 1e-10:
                alpha *= 0.5
                continue
            a_trial /= s
            f_trial = smooth_max_and_grad(a_trial, tau)[0]
            if f_trial < f_current - 1e-14:
                success = True
                break
            alpha *= 0.5
        
        if not success:
            break
        
        a = a_trial
    
    return best_a

def continuation_newton(a, tau_schedule, iters_per_tau=30, deadline=None):
    """
    Run Newton with continuation on tau.
    tau_schedule: list of tau values, increasing.
    """
    a = np.maximum(a.astype(np.float64), 1e-10)
    a = a / a.sum()
    
    for tau in tau_schedule:
        if deadline is not None and time.monotonic() > deadline:
            break
        a = newton_step(a, tau, max_iters=iters_per_tau)
    
    return a

def lp_best_response(a, M):
    """LP: max sum(g) s.t. (a*g)[k] <= M for all k, g >= 0."""
    n = len(a)
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        j_min = max(0, k - n + 1)
        j_max = min(n - 1, k)
        for j in range(j_min, j_max + 1):
            rows.append(k)
            cols.append(j)
            vals.append(a[k - j])
    A = csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    try:
        r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                    options={'time_limit': 20})
        if r.success and r.x is not None:
            g = np.maximum(r.x, 0)
            if g.sum() > 1e-15:
                return g / g.sum()
    except Exception:
        pass
    return None

def lp_alternating(a, max_iters=100, deadline=None):
    """LP-based alternating direction method."""
    a = np.maximum(a.astype(np.float64), 1e-10)
    a = a / a.sum()
    best_a = a.copy()
    best_M = fast_conv(a).max()
    
    for it in range(max_iters):
        if deadline is not None and time.monotonic() > deadline:
            break
        
        b = fast_conv(a)
        M = float(b.max())
        
        g = lp_best_response(a, M)
        if g is None:
            break
        
        # Blend
        improved = False
        for t in [0.3, 0.2, 0.1, 0.05, 0.02]:
            a_trial = (1 - t) * a + t * g
            a_trial = np.maximum(a_trial, 0)
            s = a_trial.sum()
            if s < 1e-10:

            a_trial /= s
            M_trial = fast_conv(a_trial).max()
            if M_trial < best_M - 1e-14:
                a = a_trial
                best_M = M_trial
                best_a = a.copy()
                improved = True
                break
        
        if not improved:
            break
    
    return best_a

def resample_seq(src, dest_n):
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)

import numpy as np
from scipy.optimize import linprog
from scipy.sparse import csr_matrix
import time


def fast_conv(a):
    """Full self-convolution via FFT."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]


def compute_J(a):
    """Compute J(a) = 2*n*max(conv(a,a))/(sum(a))^2."""
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))


def smooth_max_val_grad(a, tau):
    """
    Compute smooth max value, softmax weights p, and gradient.
    Returns (value, p, grad).
    """
    n = len(a)
    b = fast_conv(a)
    b_max = b.max()

    p = exp_b / exp_b.sum()
    val = b_max + np.log(exp_b.sum()) / tau

    grad = np.zeros(n)
    for m in range(n):
        grad[m] = a[m] + np.dot(a, p[m:m + n])

    return val, p, grad, b

        for _ in range(40):
            a_trial = a - alpha * d
            a_trial = np.maximum(a_trial, 0)
            s = a_trial.sum()
            if s < 1e-10:
                alpha *= 0.5
                continue
            a_trial /= s
            f_trial = smooth_max_val_grad(a_trial, tau)[0]
            if f_trial <= f_current - c1 * alpha * grad_norm * np.linalg.norm(d):
                success = True
                break
            alpha *= 0.5

        if not success or alpha < 1e-10:
            break

        a = a_trial

    return best_a


def continuation_newton(a, tau_list, iters_per_tau=25, deadline=None):
    """Run Newton with continuation over a schedule of tau values."""
    a = np.maximum(a.astype(np.float64), 1e-10)
    a = a / a.sum()
    for tau in tau_list:
        if deadline is not None and time.monotonic() > deadline:
            break
        a = newton_optimize(a, tau, max_iters=iters_per_tau)
    return a


def lp_best_response(a, M):
    """LP: max sum(g) s.t. (a*g)[k] <= M for all k, g >= 0. Returns normalized g."""
    n = len(a)
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        j_min = max(0, k - n + 1)
        j_max = min(n - 1, k)
        for j in range(j_min, j_max + 1):
            rows.append(k)
            cols.append(j)
            vals.append(a[k - j])
    A = csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    try:
        r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                    options={'time_limit': 15})
        if r.success and r.x is not None:
            g = np.maximum(r.x, 0)
            if g.sum() > 1e-15:
                return g / g.sum()
    except Exception:
        pass
    return None


def lp_alternating(a, max_iters=80, deadline=None):
    """LP-based alternating direction method."""
    a = np.maximum(a.astype(np.float64), 1e-10)
    a = a / a.sum()
    best_a = a.copy()
    best_M = fast_conv(a).max()

    for it in range(max_iters):
        if deadline is not None and time.monotonic() > deadline:
            break

        b = fast_conv(a)
        M = float(b.max())
        g = lp_best_response(a, M)
        if g is None:
            break

        improved = False
        for t in [0.4, 0.25, 0.15, 0.08, 0.04, 0.02]:
            a_trial = (1 - t) * a + t * g
            a_trial = np.maximum(a_trial, 0)
            s = a_trial.sum()
            if s < 1e-10:
                continue
            a_trial /= s
            M_trial = fast_conv(a_trial).max()
            if M_trial < best_M - 1e-14:
                a = a_trial
                best_M = M_trial
                best_a = a.copy()
                improved = True
                break

        if not improved:
            break

    return best_a


def resample_seq(src, dest_n):
    """Resample 1D sequence to new length via linear interpolation."""
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)


def generate_inits(n, rng, parent=None):
    """Generate diverse initializations including asymmetric ones."""
    inits = []
    t = np.linspace(0, 1, n, endpoint=False)

    if parent is not None:
        p = resample_seq(parent, n)
        inits.append(p.copy())
        inits.append(np.maximum(p + rng.normal(0, 0.03, n), 0.01))
        inits.append(np.maximum(p + rng.normal(0, 0.1, n), 0.01))
        inits.append(np.maximum(p * (1 + 0.2 * rng.normal(0, 1, n)), 0.01))

    # Uniform
    inits.append(np.ones(n) * (1.0 / n))
    # Random uniform
    r = rng.uniform(0.5, 1.5, n)
    inits.append(r / r.sum())

    # Linearly increasing (tilted) - key asymmetric candidate
    lin = np.arange(1, n + 1, dtype=np.float64)
    inits.append(lin / lin.sum())
    # Linearly decreasing
    lin_dec = np.arange(n, 0, -1, dtype=np.float64)
    inits.append(lin_dec / lin_dec.sum())
    # Quadratically increasing
    quad = np.arange(1, n + 1, dtype=np.float64) ** 2
    inits.append(quad / quad.sum())

    # U-shaped (high at edges)
    u_shape = 0.3 + np.cos(np.pi * t)
    inits.append(np.maximum(u_shape, 0.05) / np.maximum(u_shape, 0.05).sum())
    # V-shaped (low at edges)
    v_shape = 0.3 + np.sin(np.pi * t)
    inits.append(np.maximum(v_shape, 0.05) / np.maximum(v_shape, 0.05).sum())

    # S-curve (sigmoid-like tilt)
    for steep in [5, 10, 20]:
        s = 1.0 / (1.0 + np.exp(-steep * (t - 0.5)))
        inits.append(np.maximum(s + 0.1, 0.01))

    # Double bump
    for pos in [0.3, 0.25]:
        db = 0.2 + np.exp(-40 * (t - pos) ** 2) + np.exp(-40 * (t - (1 - pos)) ** 2)
        inits.append(np.maximum(db, 0.01) / np.maximum(db, 0.01).sum())

    # Triple bump (asymmetric)
    tb = 0.15 + np.exp(-50 * (t - 0.15) ** 2) + 0.5 * np.exp(-50 * (t - 0.5) ** 2) + np.exp(-50 * (t - 0.85) ** 2)
    inits.append(np.maximum(tb, 0.01) / np.maximum(tb, 0.01).sum())

    # Frequency structures
    for freq in [2, 3, 5, 7]:
        c = 1.0 + 0.4 * np.cos(2 * np.pi * freq * t)
        inits.append(np.maximum(c, 0.05) / np.maximum(c, 0.05).sum())

    # Smooth random
    k = max(3, n // 15)
    kern = np.ones(k) / k
    smooth = np.convolve(rng.uniform(0, 1, n), kern, mode='same')
    inits.append(np.maximum(smooth + 0.1, 0.01))

    # Power law
    inits.append(np.maximum(t ** 2 + 0.1, 0.01))
    inits.append(np.maximum((1 - t) ** 2 + 0.1, 0.01))
    inits.append(np.maximum(t ** 0.5 + 0.1, 0.01))

    return inits


class Tracker:
    """Tracks the best sequence found so far."""

    def __init__(self):
        self.best = None
        self.best_score = float('inf')
        self.t0 = time.monotonic()

    def update(self, a):
        a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
        if a.size == 0 or a.sum() < 0.01:
            return
        score = evaluate_sequence(a.tolist())
        if np.isfinite(score) and score < self.best_score:
            self.best = a.copy()
            self.best_score = score
            print(f"  ** J={score:.8f} n={a.size} t={time.monotonic() - self.t0:.1f}s", flush=True)

    def report(self, label=""):
        if self.best is not None:
            print(f"  {label} best: J={self.best_score:.8f} n={self.best.size}", flush=True)


def propose_candidate(seed=42, budget_s=1500, **kwargs):
    """
    Multi-phase search:
    Phase 1: Newton continuation across multiple n and initializations
    Phase 2: LP alternating direction polishing
    Phase 3: Final high-tau Newton polish
    Phase 4: Length neighborhood exploration
    Phase 5: Final LP pass
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + budget_s - 20
    tracker = Tracker()

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        if p.ndim == 1 and p.size > 0 and np.all(np.isfinite(p)):
            parent = np.maximum(p, 0)
            tracker.update(parent)
            print(f"Parent: n={parent.size}, J={tracker.best_score:.6f}", flush=True)
    except (NameError, AttributeError):
        pass

    if parent is None:
        parent = np.ones(256)
        tracker.update(parent)

    # === PHASE 1: Newton Continuation ===
    print(f"\n{'=' * 60}", flush=True)
    print(f"PHASE 1: Newton Continuation", flush=True)
    print(f"{'=' * 60}", flush=True)

    lengths = [256, 384, 512, 640, 768, 896, 1024, 1280, 1536, 2048]
    tau_schedule = [3, 5, 8, 12, 20, 35, 60, 100]

    for n in lengths:
        if time.monotonic() > deadline - 180:
            break

        # Limit n based on remaining time (Newton solve is O(n^3))
        est_solve_time = (n / 512) ** 3 * 0.05  # rough estimate in seconds
        max_iters_est = int(60 / max(0.01, est_solve_time))
        if max_iters_est < 10:
            continue

        time_budget = min(120.0, (deadline - 180 - time.monotonic()) /
                          max(1, len(lengths) - lengths.index(n)))
        if time_budget < 15:
            continue

        t0 = time.monotonic()
        inits = generate_inits(n, rng, parent)
        best_for_n = None
        best_M = float('inf')

        iters_per_tau = max(10, min(30, int(60 / max(0.05, est_solve_time))))

        for idx, a0 in enumerate(inits):
            if time.monotonic() > deadline - 180:
                break
            if time.monotonic() - t0 > time_budget:
                break

            a0 = np.maximum(a0, 1e-8)
            a0 = a0 / a0.sum()

            # Adaptive tau schedule: fewer high-tau steps for larger n
            if n > 1024:
                tau_sched = tau_schedule[:5]
            elif n > 512:
                tau_sched = tau_schedule[:7]
            else:
                tau_sched = tau_schedule

            a = continuation_newton(a0, tau_sched, iters_per_tau=iters_per_tau,
                                    deadline=deadline - 180)

            M = fast_conv(a).max()
            if M < best_M:
                best_M = M
                best_for_n = a.copy()

        if best_for_n is not None:
            J = compute_J(best_for_n)
            print(f"  n={n:4d}: M={best_M:.8f} J={J:.6f} "
                  f"({time.monotonic() - t0:.1f}s, {len(inits)} inits)", flush=True)
            tracker.update(best_for_n)

    tracker.report("Phase 1")

    # === PHASE 2: LP Polishing ===
    print(f"\n{'=' * 60}", flush=True)
    print(f"PHASE 2: LP Polishing", flush=True)
    print(f"{'=' * 60}", flush=True)

    if tracker.best is not None:
        n_best = tracker.best.size
        # Estimate LP time and set iteration limit
        est_lp_time = 0.005 * (n_best / 128) ** 2
        max_lp_iters = max(20, min(200, int(200 / max(0.01, est_lp_time))))
        print(f"  n={n_best}, est LP time={est_lp_time * 1000:.0f}ms, max_iters={max_lp_iters}",
              flush=True)

        a = lp_alternating(tracker.best, max_iters=max_lp_iters, deadline=deadline - 100)
        tracker.update(a)

        if time.monotonic() < deadline - 80:
            a = lp_alternating(a, max_iters=max(lp_iters // 3, 10), deadline=deadline - 60)
            tracker.update(a)

    tracker.report("Phase 2")

    # === PHASE 3: Final Newton Polish (high tau) ===
    print(f"\n{'=' * 60}", flush=True)
    print(f"PHASE 3: High-Tau Newton Polish", flush=True)
    print(f"{'=' * 60}", flush=True)

    if tracker.best is not None:
        n_best = tracker.best.size
        est_solve_time = (n_best / 512) ** 3 * 0.05
        iters = max(10, min(50, int(80 / max(0.05, est_solve_time))))
        a = continuation_newton(tracker.best, [80, 120, 180, 250],
                                iters_per_tau=iters, deadline=deadline - 40)
        tracker.update(a)

    tracker.report("Phase 3")

    # === PHASE 4: Length Neighborhood ===
    print(f"\n{'=' * 60}", flush=True)
    print(f"PHASE 4: Length Exploration", flush=True)
    print(f"{'=' * 60}", flush=True)

    if tracker.best is not None and time.monotonic() < deadline - 30:
        base_n = tracker.best.size
        for delta in [0, -128, 128, -256, 256, -64, 64]:
            new_n = base_n + delta
            if new_n < 128 or new_n > 2560:
                continue
            if new_n == base_n:
                continue
            if time.monotonic() > deadline - 20:
                break

            a = resample_seq(tracker.best, new_n)
            a = np.maximum(a, 1e-8)
            a = a / a.sum()
            est_solve_time = (new_n / 512) ** 3 * 0.05
            iters = max(8, min(30, int(20 / max(0.05, est_solve_time))))
            a = continuation_newton(a, [20, 40, 80], iters_per_tau=iters,
                                    deadline=deadline - 15)
            tracker.update(a)

    tracker.report("Phase 4")

    # === PHASE 5: Final LP ===
    if tracker.best is not None and time.monotonic() < deadline - 5:
        a = lp_alternating(tracker.best, max_iters=20, deadline=deadline)
        tracker.update(a)

    # Fallback
    if tracker.best is None:
        tracker.best = np.ones(128)
        tracker.best_score = compute_J(tracker.best)

    print(f"\n{'=' * 60}", flush=True)
    print(f"FINAL: J={tracker.best_score:.8f} n={tracker.best.size} "
          f"time={time.monotonic() - tracker.t0:.0f}s", flush=True)
    print(f"{'=' * 60}", flush=True)

    return [float(x) for x in tracker.best]
