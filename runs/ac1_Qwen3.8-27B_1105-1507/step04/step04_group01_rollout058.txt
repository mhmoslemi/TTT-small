Dual: min M * sum(y) s.t. sum_k A[k,i]*y[k] >= 1 for all i, y >= 0.

At the optimum, by strong duality: sum(g) = M * sum(y).

The complementary slackness: g[i] > 0 implies sum_k A[k,i]*y[k] = 1.

Now, the "ideal" a would satisfy: for the optimal g from the LP, a = g (fixed point). At a fixed point, a = g means:
- a[i] > 0 implies sum_k A[k,i]*y[k] = 1 (from complementary slackness)
- conv(a, a)[k] <= M for all k (primal feasibility)
- conv(a, a)[k] < M implies y[k] = 0 (complementary slackness)

This is similar to the KKT conditions of the original non-convex problem!

def compute_conv(a):

    size = 1
    while size < 2*n-1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa*fa, size)[:2*n-1]

    conv = compute_conv(a)
    max_b = np.max(conv)
    sum_a = np.sum(a)
    if sum_a < 0.01:
        return np.inf
    return 2.0*n*max_b/(sum_a*sum_a)

def build_toeplitz(a):
    n = len(a)
    # A[k, j] = a[k-j] if 0 <= k-j < n, else 0
    # k ranges 0..2n-2, j ranges 0..n-1
    # A[k,j] = a[k-j] if max(0, k-n+1) <= j <= min(n-1, k)
    idx = np.arange(n)
    # Use broadcasting
    k_idx = np.arange(2*n-1)[:, None]
    j_idx = idx[None, :]
    diff = k_idx - j_idx
    A = np.zeros((2*n-1, n))
    mask = (diff >= 0) & (diff < n)
    A[mask] = a[diff[mask]]
    return A

def solve_lp_for_g(a, M, time_limit=10):
    """LP: max sum(g) s.t. (a*g)[k] <= M for all k, g >= 0."""
    n = len(a)
    A = build_toeplitz(a)
    c = -np.ones(n)
    b_ub = np.full(2*n-1, M)
    bounds = [(0, None)]*n
    try:
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': time_limit})
        if res.status == 0:
            g = np.maximum(res.x, 0)
            s = np.sum(g)
            if s > 1e-10:
                return g/s, g  # normalized g and raw g
    except:
        pass
    return None, None

def lp_alternate(a, n_iters=200, t=0.1, time_budget=60.0):

    n = len(a)

        return a
    a = a / s
    best_a = a.copy()

    start = time.monotonic()

        obj = compute_obj(a_new)
        if obj < best_obj:
            best_obj = obj
            best_a = a_new.copy()
        a = a_new
    
    return best_a

        a = a / s
        obj = compute_obj(a)
        if obj < best_obj - 1e-12:
            best_obj = obj
            best_a = a.copy()
    
    return best_a

"""
LP-dominant alternating optimization with adaptive blending and peak shaving.
Uses the alternating LP (fix one variable, optimize the other) as the primary
optimization tool, with adaptive blend parameters, peak-shaving coordinate
descent, and spectral initialization. Designed to beat 1.5027 on the
minimization of 2*n*max(conv(a,a))/(sum(a))^2.
"""

import numpy as np
from scipy.optimize import linprog
from scipy.fft import rfft, irfft
import time

def compute_conv(a):
    """FFT-based linear convolution of a with itself."""
    n = len(a)
    size = 1
    while size < 2*n-1:
        size <<= 1
    fa = rfft(a, size)
    conv = irfft(fa*fa, size)
    return conv[:2*n-1]

def compute_obj(a):
    """Compute 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    conv = compute_conv(a)
    max_b = float(np.max(conv))
    sum_a = float(np.sum(a))
    if sum_a < 0.01:
        return np.inf
    return 2.0*n*max_b/(sum_a*sum_a)

def build_toeplitz(a):
    """Build matrix A where (A@g)[k] = (conv(a,g))[k]."""
    n = len(a)
    k_idx = np.arange(2*n-1)[:, None]
    j_idx = np.arange(n)[None, :]
    diff = k_idx - j_idx
    mask = (diff >= 0) & (diff < n)
    A = np.zeros((2*n-1, n), dtype=np.float64)
    A[mask] = a[diff[mask]]
    return A

def solve_lp_maxsum(a, M, time_limit=5):
    """LP: max sum(g) s.t. conv(a,g)[k] <= M, g >= 0. Returns normalized g."""
    n = len(a)
    A = build_toeplitz(a)
    c = -np.ones(n)
    b_ub = np.full(2*n-1, M)
    bounds = [(0.0, None)]*n
    try:
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds,
                      method='highs', options={'presolve': True, 'time_limit': time_limit})
        if res.status == 0:
            g = np.maximum(res.x, 0.0)
            s = float(np.sum(g))
            if s > 1e-10:
                return g / s
    except Exception:
        pass
    return None

def solve_lp_minpeak(a, time_limit=10):
    """LP: min M s.t. conv(a,g)[k] <= M, sum(g)=1, g >= 0. Returns (g, M)."""
    n = len(a)
    A = build_toeplitz(a)
    # Variables: g[0..n-1], M
    c = np.zeros(n+1)
    c[n] = 1.0
    A_ub = np.zeros((2*n-1, n+1))
    A_ub[:, :n] = A
    A_ub[:, n] = -1.0
    b_ub = np.zeros(2*n-1)
    A_eq = np.zeros((1, n+1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])
    bounds = [(0.0, None)]*n + [(None, None)]
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': time_limit})
        if res.status == 0:
            g = np.maximum(res.x[:n], 0.0)
            M_opt = float(res.x[n])
            return g, M_opt
    except Exception:
        pass
    return None, None

def peak_shave(a, max_iters=300, time_budget=15.0):

    a = a.copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    step = 0.05
    
    for it in range(max_iters):
        if time.monotonic() - start > time_budget:

        # Get top 3 peaks

        grad = np.zeros(n)
        for k in top_idx:

            i_lo = max(0, k - n + 1)
            i_hi = min(n - 1, k)
            for i in range(i_lo, i_hi + 1):
                grad[i] += a[k - i]
        gn = np.linalg.norm(grad)
        if gn < 1e-10:

        grad /= gn

        a_new = np.maximum(a_new, 0.0)

        if sn < 1e-10:

        obj = compute_obj(a_new)
        if obj < best_obj - 1e-12:
            best_obj = obj
            best_a = a_new.copy()
            a = a_new
        else:

def resample(a, new_n):
    """Resample sequence to new length."""
    old_n = len(a)

    x_old = np.linspace(0, 1, old_n)
    x_new = np.linspace(0, 1, new_n)
    return np.interp(x_new, x_old, a)
