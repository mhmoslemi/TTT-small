"""
Minimizes J(a) = 2*n*max(convolve(a,a))/(sum(a))² using:
1. FISTA with simplex projection on smooth log-sum-exp objective (fast basin finding)
2. LP mirror descent with FISTA-style extrapolation and adaptive line search
3. Active-set identification + L-BFGS-B with very high beta for precise refinement
4. Multi-restart with diverse perturbations and coordinate ascent polish

import numpy as np
from scipy.optimize import linprog, minimize
from scipy.sparse import csr_matrix
import time

_A_structure_cache = {}


def fast_conv(a):
    """Full self-convolution via FFT, O(n log n)."""
    n = len(a)
    s = 1
    while s < 2 * n - 1:
        s <<= 1
    fa = np.fft.rfft(a, s)
    return np.fft.irfft(fa * fa, s)[:2 * n - 1]


def compute_J(a):
    """Compute J(a) = 2*n*max(conv(a,a))/(sum(a))^2."""
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))


def get_A_structure(n):
    """Get or pre-compute the (row, col) arrays for the constraint matrix."""
    if n not in _A_structure_cache:
        rows = []
        cols = []
        for k in range(2 * n - 1):
            jl = max(0, k - n + 1)
            jh = min(n - 1, k)
            for j in range(jl, jh + 1):
                rows.append(k)
                cols.append(j)
        _A_structure_cache[n] = (np.array(rows, dtype=np.int32),
                                  np.array(cols, dtype=np.int32))
    return _A_structure_cache[n]


def build_A_sparse_fast(a):
    """Build sparse constraint matrix using pre-computed structure."""
    n = len(a)
    rows_arr, cols_arr = get_A_structure(n)
    vals = a[rows_arr - cols_arr]
    return csr_matrix((vals, (rows_arr, cols_arr)), shape=(2 * n - 1, n))


def lp_best_response(a, M):
    """LP: max sum(g) s.t. (a*g)_k <= M for all k, g >= 0. Returns normalized g or None."""
    n = len(a)
    c = -np.ones(n)
    A = build_A_sparse_fast(a)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    try:
        r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
        if r.success and r.x is not None:
            g = np.maximum(r.x, 0.0)
            g_sum = g.sum()
            if g_sum > 1e-10:
                return g / g_sum
    except Exception:
        pass
    return None


def smooth_max_and_grad(a, beta):
    """
    Compute smooth approximation of max(conv(a,a)) and its gradient.
    Returns (smooth_max_value, gradient_array).
    """
    n = len(a)
    b = fast_conv(a)
    b_max = float(b.max())
    shifted = beta * (b - b_max)
    shifted = np.clip(shifted, -500, 500)
    exp_shifted = np.exp(shifted)
    Z = float(exp_shifted.sum())
    if Z < 1e-300:
        p = np.zeros(2 * n - 1)
        p[np.argmax(b)] = 1.0
    else:
        p = exp_shifted / Z
    g_val = (b_max + np.log(max(Z, 1e-300))) / beta

    # Gradient via FFT cross-correlation
    s = 1
    while s < 2 * n - 1:
        s <<= 1
    fa = np.fft.rfft(a, s)
    fp = np.fft.rfft(p, s)
    corr = np.fft.irfft(fp * np.conj(fa), s)[:2 * n - 1]
    grad = 2.0 * corr
    return g_val, grad


def F_and_grad(a, beta):
    """Compute F(a) = 2n*smooth_max(conv(a,a))/(sum(a))^2 and gradient."""
    n = len(a)
    s = float(a.sum())
    if s < 1e-10:
        return 1e10, np.zeros(n)
    g_val, grad_g = smooth_max_and_grad(a, beta)
    F_val = 2.0 * n * g_val / (s * s)
    grad_F = 2.0 * n * (s * grad_g - 2.0 * g_val * np.ones(n)) / (s ** 3)
    return F_val, grad_F


def project_simplex(v):
    """Project v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.argmax(1 - u / np.arange(1, n + 1) > 0)
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0)


def fista_optimize(a0, beta, max_iters, deadline, alpha=None):
    """
    FISTA on the smooth objective F(a) subject to a in simplex.
    a0 should already be normalized to sum to 1.
    """
    n = len(a0)
    a = np.maximum(a0, 0).copy()
    s = a.sum()
    if s < 1e-10:
        return a
    a = a / s

    y = a.copy()
    t_k = 1.0
    a_prev = a.copy()

    if alpha is None:
        # Estimate L from the function
        # For our problem on the simplex, a reasonable initial step
        alpha = 0.01

    best_a = a.copy()
    best_F = F_and_grad(a, beta)[0]

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        _, grad_y = F_and_grad(y, beta)

        # Gradient step + projection
        x_new = project_simplex(y - alpha * grad_y)

        # FISTA update
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t_k * t_k)) / 2.0
        momentum = (t_k - 1.0) / t_new
        y_new = x_new + momentum * (x_new - a_prev)
        y_new = project_simplex(y_new)

        a_prev = a
        a = x_new
        y = y_new
        t_k = t_new

        # Backtracking: check if we made progress
        F_new = F_and_grad(a, beta)[0]
        if not np.isfinite(F_new):
            alpha *= 0.5
            continue
        if F_new < best_F:
            best_F = F_new
            best_a = a.copy()

    return best_a


def fista_anneal(a0, betas, deadline):
    """FISTA with beta annealing."""
    a = a0.copy()
    for beta, max_iters in betas:
        if time.monotonic() > deadline:
            break
        a = fista_optimize(a, beta, max_iters, deadline)
    return a


def line_search_golden(a, g, n_evals=25):
    """
    Golden section search for best t in [0,1] minimizing J((1-t)*a + t*g).
    Returns (best_a, best_J, best_t).
    """
    best_t = 0.0
    best_J = compute_J(a)
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    lo = 0.0
    hi = 1.0
    J_at_0 = best_J
    a_1 = g.copy()
    J_at_1 = compute_J(a_1)
    if J_at_1 < J_at_0:
        best_J = J_at_1
        best_t = 1.0

    for _ in range(n_evals):
        t1 = hi - gr * (hi - lo)
        t2 = lo + gr * (hi - lo)
        a1 = (1.0 - t1) * a + t1 * g
        a2 = (1.0 - t2) * a + t2 * g
        J1 = compute_J(a1)
        J2 = compute_J(a2)
        if J1 < J2:
            hi = t2
        else:
            lo = t1
        t_mid = (lo + hi) / 2.0
        a_mid = (1.0 - t_mid) * a + t_mid * g
        J_mid = compute_J(a_mid)
        if J_mid < best_J:
            best_J = J_mid
            best_t = t_mid

    return (1.0 - best_t) * a + best_t * g, best_J, best_t


def lp_mirror_descent_extrap(a, max_iters, deadline, tracker=None,
                              n_line_evals=22, stall_patience=50,
                              use_extrap=True):
    """
    LP mirror descent with FISTA-style extrapolation.
    Keeps a momentum field to accelerate convergence.
    """
    a = np.maximum(a.astype(np.float64), 1e-8)
    a = a / a.sum()
    a_prev = a.copy()
    y = a.copy()  # extrapolated point
    t_k = 1.0

    current_J = compute_J(a)
    if tracker is not None:
        tracker.update(a)

    stall_count = 0
    best_J_so_far = current_J
    best_a_so_far = a.copy()

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        # Use extrapolated point for LP if enabled
        a_lp = y if use_extrap else a
        a_lp = np.maximum(a_lp, 1e-8)
        a_lp = a_lp / a_lp.sum()

        b = fast_conv(a_lp)
        M = float(b.max())

        g = lp_best_response(a_lp, M)
        if g is None:
            break

        # Line search from current a towards g
        a_new, J_new, t_used = line_search_golden(a, g, n_evals=n_line_evals)

        if J_new < current_J - 1e-13:
            a_old = a
            a = a_new
            current_J = J_new
            stall_count = 0

            # FISTA momentum update
            t_new = (1.0 + np.sqrt(1.0 + 4.0 * t_k * t_k)) / 2.0
            momentum = (t_k - 1.0) / t_new
            y = a + momentum * (a - a_old)
            y = np.maximum(y, 1e-8)
            y = y / y.sum()
            t_k = t_new

            if J_new < best_J_so_far:
                best_J_so_far = J_new
                best_a_so_far = a.copy()
                if tracker is not None:
                    tracker.update(a)
        else:
            # Try smaller steps
            improved = False
            for t_try in [0.01, 0.005, 0.02, 0.003, 0.05]:
                a_try = (1.0 - t_try) * a + t_try * g
                a_try = np.maximum(a_try, 0.0)
                s = a_try.sum()
                if s < 1e-10:
                    continue
                a_try = a_try / s
                J_try = compute_J(a_try)
                if J_try < current_J - 1e-14:
                    a_old = a
                    a = a_try
                    current_J = J_try
                    stall_count = 0
                    improved = True

                    t_new = (1.0 + np.sqrt(1.0 + 4.0 * t_k * t_k)) / 2.0
                    momentum = (t_k - 1.0) / t_new
                    y = a + momentum * (a - a_old)
                    y = np.maximum(y, 1e-8)
                    y = y / y.sum()
                    t_k = t_new

                    if J_try < best_J_so_far:
                        best_J_so_far = J_try
                        best_a_so_far = a.copy()
                        if tracker is not None:
                            tracker.update(a)
                    break

            if not improved:
                stall_count += 1
                # Reset momentum on stall
                y = a.copy()
                t_k = 1.0
                if stall_count >= stall_patience:
                    break

        if tracker is not None and it % 100 == 0:
            tracker.update(a)

    if tracker is not None:
        tracker.update(best_a_so_far)
    return best_a_so_far


def lbfgs_polish(a0, beta, max_iters, deadline):
    """L-BFGS-B optimization on the smooth objective (sum-normalized)."""
    a = np.maximum(a0.astype(np.float64), 1e-10)
    a = a / a.sum()
    n = len(a)

    def func(x):
        x2 = np.maximum(x, 0.0)
        s = x2.sum()
        if s < 1e-10:
            return 1e10
        x2 = x2 / s
        f, _ = F_and_grad(x2, beta)
        if not np.isfinite(f):
            return 1e10
        return f

    def jac(x):
        x2 = np.maximum(x, 0.0)
        s = x2.sum()
        if s < 1e-10:
            return np.zeros(n)
        x2 = x2 / s
        _, g = F_and_grad(x2, beta)
        g = np.nan_to_num(g, nan=0.0, posinf=0.0, neginf=0.0)
        g[x2 < 1e-12] = 0.0
        return g

    bounds = [(0.0, None)] * n
    try:
        res = minimize(func, a.copy(), jac=jac, method='L-BFGS-B',
                       bounds=bounds,
                       options={'maxiter': max_iters, 'max_fun': max_iters * 30,
                                'gtol': 1e-10, 'ftol': 1e-16})
        if res.fun < 1e10:
            new_a = np.maximum(res.x, 0.0)
            if new_a.sum() > 1e-6:
                return new_a / new_a.sum()
    except Exception:
        pass
    return a


def resample_seq(src, dest_n):
    """Resample 1D sequence to new length via cubic interpolation."""
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    if src.size < 2:
        return np.full(dest_n, src[0] if src.size == 1 else 0.0)
    from scipy.interpolate import PchipInterpolator
    try:
        t_old = np.linspace(0, 1, src.size)
        interp = PchipInterpolator(t_old, src, extrapolate=False)
        t_new = np.linspace(0, 1, dest_n)
        result = interp(t_new)
    except Exception:
        t_old = np.linspace(0, 1, src.size)
        t_new = np.linspace(0, 1, dest_n)
        result = np.interp(t_new, t_old, src)
    return np.maximum(result, 0.0)


def generate_initializations(n, rng, parent=None):
    """Generate diverse initial sequences of length n."""
    inits = []
    t_arr = np.linspace(0, 1, n, endpoint=False)

    if parent is not None and parent.size > 0:
        inits.append(resample_seq(parent, n))
        p = resample_seq(parent, n)
        p = p + rng.normal(0, 0.05, n) * p
        inits.append(np.maximum(p, 0.01))

    inits.append(np.ones(n))
    inits.append(rng.uniform(0.3, 1.0, n))

    # Cosine families with various frequencies
    for freq in [2, 3, 4, 5, 6, 7, 8]:
        inits.append(1.0 + 0.3 * np.cos(2.0 * np.pi * freq * t_arr))

    # Combined cosines
    inits.append(1.0 + 0.25 * np.cos(2.0 * np.pi * 3 * t_arr) + 0.15 * np.cos(2.0 * np.pi * 7 * t_arr))
    inits.append(1.0 + 0.2 * np.cos(2.0 * np.pi * 2 * t_arr) + 0.2 * np.cos(2.0 * np.pi * 5 * t_arr))

    # Gaussian bump
    inits.append(np.exp(-3.0 * (t_arr - 0.5) ** 2) + 0.1)

    # Sine absolute
    inits.append(np.abs(np.sin(np.pi * t_arr)) + 0.2)

    # Wavelength-varying
    freq_var = 3 + 5 * t_arr
    inits.append(np.cos(2.0 * np.pi * np.cumsum(freq_var) / n) + 1.3)

    # Symmetric pattern
    inits.append(1.0 + 0.3 * np.cos(2.0 * np.pi * 4 * t_arr) * np.cos(np.pi * t_arr))

    # Sawtooth-like (smoothed)
    inits.append(1.0 + 0.2 * np.sin(4.0 * np.pi * t_arr) * np.sin(2.0 * np.pi * t_arr))

    return inits

def coordinate_polish(a, n_sweeps=5, deadline=None):
    """
    Coordinate descent polish: for each coordinate, optimize its value
    while keeping others fixed. Uses the exact (non-smooth) objective.
    """
    a = np.maximum(a.copy(), 0.0)
    s = a.sum()
    if s < 1e-10:
        return a
    a = a / s
    current_J = compute_J(a)

    for sweep in range(n_sweeps):
        if deadline is not None and time.monotonic() > deadline:
            break
        improved = False
        for i in range(len(a)):
            if deadline is not None and time.monotonic() > deadline:
                break
            a_orig = a[i]
            # Try several values for coordinate i
            # The optimal a_i is somewhere around the current value
            # Try: current * factor for several factors
            for factor in [0.5, 0.7, 0.85, 1.15, 1.3, 1.5, 0.3, 2.0, 0.9, 1.1]:
                a[i] = a_orig * factor
                s_new = a.sum()
                if s_new < 1e-10:
                    a[i] = a_orig
                    continue
                J_try = compute_J(a)
                if J_try < current_J - 1e-12:
                    current_J = J_try
                    improved = True
                    break
            a[i] = a_orig  # Reset (we already tried modifications)
        if not improved:
            break

    return a

    if tracker.best is not None and time.monotonic() < deadline:
        phase4_budget = 350.0
        deadline_p4 = min(deadline, time.monotonic() + phase4_budget)
        n_best = tracker.best.size
        print(f"\n{'='*50} Phase 4: Restarts n={n_best} (budget {phase4_budget:.0f}s) {'='*50}", flush=True)

        strategies = [
            ('gaussian', 0.03),
            ('gaussian', 0.05),
            ('gaussian', 0.08),
            ('gaussian', 0.01),
            ('uniform', 0.05),
            ('uniform', 0.1),
            ('spike', 6),
            ('spike', 12),
            ('smooth_noise', 0.05),
            ('smooth_noise', 0.1),
            ('gaussian', 0.02),
            ('localized', 0.15),
        ]

        attempt = 0
        for pert_type, param in strategies:
            for trial in range(4):
                if time.monotonic() > deadline_p4:
                    break
                attempt += 1
                a = tracker.best.copy().astype(np.float64)
                n = len(a)

                if pert_type == 'gaussian':
                    a = a + rng.normal(0, param, n) * a
                elif pert_type == 'uniform':
                    a = a + rng.uniform(-param, param, n) * a
                elif pert_type == 'spike':
                    k = min(param, n // 4)
                    idx = rng.choice(n, size=k, replace=False)
                    a[idx] *= rng.uniform(0.3, 3.0, k)
                elif pert_type == 'smooth_noise':
                    noise = rng.normal(0, param, n)
                    kernel_size = max(3, n // 32)
                    kernel = np.exp(-0.5 * (np.arange(kernel_size) - kernel_size // 2) ** 2 / (kernel_size // 4 + 1))
                    kernel /= kernel.sum()
                    noise = np.convolve(noise, kernel, mode='same')
                    a = a + noise * a
                elif pert_type == 'localized':
                    # Modify a random contiguous block
                    block_size = rng.integers(n // 20, n // 5)
                    start = rng.integers(0, n - block_size)
                    a[start:start + block_size] *= rng.uniform(0.5, 2.0)

                a = np.maximum(a, 1e-4)
                s = a.sum()
                if s < 1e-10:
                    continue
                a = a / s

                J_pert = compute_J(a)
                if J_pert < tracker.best_score:
                    tracker.update(a)

                # Short LP refinement
                budget_here = min(45.0, deadline_p4 - time.monotonic())
                max_iters = int(budget_here * 50)
                max_iters = min(max_iters, 1000)
                a = lp_mirror_descent_extrap(a, max_iters, deadline_p4, tracker=tracker,
                                              n_line_evals=18, stall_patience=20,
                                              use_extrap=True)

        print(f"  Phase 4 done: {attempt} attempts, best: J={tracker.best_score:.8f}", flush=True)

    # === Phase 5: Final high-precision polish ===
    if tracker.best is not None and time.monotonic() < deadline:
        remaining = deadline - time.monotonic()
        phase5_budget = min(150.0, remaining)
        deadline_p5 = time.monotonic() + phase5_budget
        n_p5 = tracker.best.size
        print(f"\n{'='*50} Phase 5: Final Polish n={n_p5} (budget {phase5_budget:.0f}s) {'='*50}", flush=True)

        # Very high precision LP
        max_iters = int(phase5_budget * 60)
        max_iters = min(max_iters, 5000)
        a = lp_mirror_descent_extrap(tracker.best, max_iters, deadline_p5, tracker=tracker,
                                      n_line_evals=30, stall_patience=60, use_extrap=True)

        # L-BFGS-B with extremely high beta
        if time.monotonic() < deadline_p5:
            for beta in [500.0, 1000.0]:
                if time.monotonic() > deadline_p5:
                    break
                a_res = lbfgs_polish(tracker.best, beta, max_iters=300, deadline=deadline_p5)
                tracker.update(a_res)

        # Coordinate descent polish
        if time.monotonic() < deadline_p5:
            a_coord = coordinate_polish(tracker.best, n_sweeps=3, deadline=deadline_p5)
            tracker.update(a_coord)

        print(f"  Phase 5 done, best: J={tracker.best_score:.8f}", flush=True)

    # === Phase 6: Length exploration ===
    if tracker.best is not None and time.monotonic() < deadline:
        remaining = deadline - time.monotonic()
        if remaining > 30:
            n_best = tracker.best.size
            print(f"\n{'='*50} Phase 6: Length Exploration (budget {remaining:.0f}s) {'='*50}", flush=True)

            candidate_lengths = []
            for delta in [-32, -16, 16, 32, 64, -64]:
                nl = n_best + delta
                if 128 <= nl <= 2048:
                    candidate_lengths.append(nl)
            # Also try some "natural" lengths
            for nl in [512, 768, 1024, 1280, 1536]:
                if nl not in candidate_lengths and abs(nl - n_best) > 8:
                    candidate_lengths.append(nl)

            for n_try in candidate_lengths[:8]:
                if time.monotonic() > deadline:
                    break
                if n_try == n_best:
                    continue
                a_res = resample_seq(tracker.best, n_try)
                a_res = np.maximum(a_res, 0.01)
                a_res = a_res / a_res.sum()
                J_res = compute_J(a_res)
                if J_res < tracker.best_score:
                    tracker.update(a_res)

                # Quick LP refinement
                budget_here = min(30.0, deadline - time.monotonic())
                max_iters = int(budget_here * 50)
                max_iters = min(max_iters, 800)
                a_res = lp_mirror_descent_extrap(a_res, max_iters, deadline, tracker=tracker,
                                                  n_line_evals=15, stall_patience=20,
                                                  use_extrap=True)

            print(f"  Phase 6 done, best: J={tracker.best_score:.8f}", flush=True)

    # Fallback
    if tracker.best is None:
        tracker.best = np.ones(512)
        tracker.best_score = compute_J(tracker.best)
        print(f"  Using fallback: J={tracker.best_score:.8f}", flush=True)

    print(f"\n{'='*50} FINAL RESULT {'='*50}", flush=True)
    print(f"J={tracker.best_score:.8f} n={tracker.best.size} "
          f"time={time.monotonic()-tracker.t_start:.1f}s", flush=True)

    # Ensure output is valid
    result = np.clip(tracker.best, 0.0, 1000.0)
    if result.sum() < 0.01:
        result = np.ones(128)

    return [float(x) for x in result]
