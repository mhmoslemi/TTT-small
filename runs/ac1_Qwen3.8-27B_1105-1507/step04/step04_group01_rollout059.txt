Hybrid optimization for minimizing 2*n*max(conv(a,a))/(sum(a))^2.
Uses spectral initialization, soft-max L-BFGS-B, LP alternating,
subgradient with backtracking, and coordinate descent polishing.
Improves over parent by: better spectral init, adaptive tau,
coordinate descent for small n, dual-weighted subgradient, and

import numpy as np
from scipy.optimize import minimize, linprog, minimize_scalar
from scipy.fft import rfft, irfft
import time


def compute_conv(a):
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    conv = irfft(fa * fa, size)
    return conv[:2 * n - 1]


def compute_obj(a):
    n = len(a)
    if n == 0:
        return np.inf
    conv = compute_conv(a)
    max_b = float(np.max(conv))
    sum_a = float(np.sum(a))
    if sum_a < 0.01:
        return np.inf
    return 2.0 * n * max_b / (sum_a * sum_a)


def fft_convolve(a, b):
    n, m = len(a), len(b)
    size = 1
    while size < n + m - 1:
        size <<= 1
    fa = rfft(a, size)
    fb = rfft(b, size)
    return irfft(fa * fb, size)[:n + m - 1]

def softmax(z):
    z = z - np.max(z)
    e = np.exp(z)
    return e / np.sum(e)

    cmax = float(np.max(conv))
    shifted = np.clip((conv - cmax) / tau, -500.0, 500.0)
    exp_conv = np.exp(shifted)
    Z = float(np.sum(exp_conv))
    soft_max_val = cmax + tau * np.log(Z)
    w = exp_conv / Z
    w_rev = w[::-1]
    conv_fw = fft_convolve(a, w_rev)
    f1 = conv_fw[n - 1:2 * n - 1][::-1]
    grad_a = 2.0 * f1
    mean_grad = float(np.dot(a, grad_a))
    grad_z = a * (grad_a - mean_grad)
    return soft_max_val, grad_z


def optimize_lfbs(a_init, tau_list, max_iter=300, time_budget=30.0):
    """L-BFGS-B with soft-max parametrization and tau annealing."""
    n = len(a_init)
    a = a_init.copy()
    s = np.sum(a)
    if s < 1e-10:
        a = np.ones(n) / n
    else:
        a = a / s
    a = np.maximum(a, 1e-30)
    z = np.log(a)
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    for tau in tau_list:
        if time.monotonic() - start > time_budget:
            break

        def func_and_grad(z_val):
            return soft_max_obj_and_grad_z(z_val, tau)

        try:
            result = minimize(func_and_grad, z, jac=True, method='L-BFGS-B',
                              options={'maxiter': max_iter, 'ftol': 1e-15, 'gtol': 1e-10})
            z_new = result.x
        except Exception:
            continue

        a_new = softmax(z_new)
        a_new = np.maximum(a_new, 0)
        s_new = np.sum(a_new)
        if s_new < 1e-10:
            continue
        a_new = a_new / s_new
        obj = compute_obj(a_new)
        if obj < best_obj:
            best_obj = obj
            best_a = a_new.copy()
            z = z_new

    return best_a, best_obj


def build_toeplitz(a):
    n = len(a)
    A = np.zeros((2 * n - 1, n), dtype=np.float64)
    for j in range(n):
        k_start = j
        k_end = min(j + n, 2 * n - 1)
        A[k_start:k_end, j] = a[:k_end - k_start]
    return A


def solve_lp_g(a, M, time_limit=20):
    """Solve LP: max sum(g) s.t. (a*g)[k] <= M for all k, g >= 0."""
    n = len(a)
    A = build_toeplitz(a)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': time_limit})
        if res.status == 0:
            g = np.maximum(res.x, 0)
            s = float(np.sum(g))
            if s > 1e-10:
                return g / s, s
    except Exception:
        pass
    return None, None


def alternating_lp(a, max_iters=50, time_budget=40.0):
    """Alternating LP with line search."""
    n = len(a)
    a = a.copy().astype(np.float64)
    s = np.sum(a)
    if s < 1e-10:
        return a
    a = a / s
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    stall_count = 0
    patience = 15

    for it in range(max_iters):
        if time.monotonic() - start > time_budget:
            break
        conv = compute_conv(a)
        M = float(np.max(conv))
        g, _ = solve_lp_g(a, M, time_limit=12)
        if g is None:
            break
        best_t_obj = best_obj
        best_a_new = best_a.copy()
        for t in [0.05, 0.1, 0.2, 0.3, 0.5, 0.7, 1.0]:
            a_try = (1.0 - t) * a + t * g
            a_try = np.maximum(a_try, 0)
            s_try = np.sum(a_try)
            if s_try < 1e-10:
                continue
            a_try = a_try / s_try
            obj_try = compute_obj(a_try)
            if obj_try < best_t_obj:
                best_t_obj = obj_try
                best_a_new = a_try.copy()
        if best_t_obj < best_obj - 1e-12:
            best_obj = best_t_obj
            best_a = best_a_new.copy()
            a = best_a_new
            stall_count = 0
        else:
            stall_count += 1
            if stall_count > patience:
                rng = np.random.default_rng(it * 7919 + 12345)
                a = a * np.exp(rng.normal(0, 0.05, n))
                a = np.maximum(a, 0)
                s = np.sum(a)
                if s > 1e-10:
                    a = a / s
                stall_count = 0
                patience = 8
            else:
                a = 0.85 * a + 0.15 * g
                a = np.maximum(a, 0)
                s = np.sum(a)
                if s > 1e-10:
                    a = a / s
    return best_a


def subgradient_with_backtracking(a, max_iters=2000, time_budget=20.0,
                                   tau_start=2.0, tau_end=50.0):
    """Subgradient with adaptive tau and backtracking line search."""
    n = len(a)
    a = a.copy().astype(np.float64)
    s = np.sum(a)
    if s < 1e-10:
        a = np.ones(n) / n
    else:
        a = a / s
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    step = 0.5
    stall = 0

    for it in range(max_iters):
        if time.monotonic() - start > time_budget:
            break
        frac = min(1.0, it / max(1, max_iters))
        tau = tau_start + (tau_end - tau_start) * frac

        conv = compute_conv(a)
        max_conv = float(np.max(conv))
        shifted = np.clip(tau * (conv - max_conv), -300, 300)
        weights = np.exp(shifted)
        w_sum = np.sum(weights)
        if w_sum < 1e-300:
            idx = int(np.argmax(conv))
            weights = np.zeros(len(conv))
            weights[idx] = 1.0
        else:
            weights /= w_sum

        w_rev = weights[::-1]
        conv_fw = fft_convolve(a, w_rev)
        grad_a = 2.0 * conv_fw[n - 1:2 * n - 1][::-1]
        grad_centered = grad_a - np.mean(grad_a)

        # Backtracking line search
        best_step_obj = best_obj
        best_step_a = a.copy()
        for trial_step in [step, step * 0.5, step * 0.25, step * 0.1]:
            a_new = a - trial_step * grad_centered
            a_new = np.maximum(a_new, 0)
            s_new = np.sum(a_new)
            if s_new < 1e-10:
                continue
            a_new = a_new / s_new
            obj_new = compute_obj(a_new)
            if obj_new < current_obj if (current_obj := compute_obj(a)) else best_obj:
                pass
            if obj_new < best_obj:
                best_step_obj = obj_new
                best_step_a = a_new.copy()

        if best_step_obj < best_obj - 1e-14:
            best_obj = best_step_obj
            best_a = best_step_a.copy()
            a = best_step_a
            stall = 0
            step = min(step * 1.05, 2.0)
        else:
            stall += 1
            cur_obj = compute_obj(a)
            a = a - step * 0.1 * grad_centered
            a = np.maximum(a, 0)
            s = np.sum(a)
            if s > 1e-10:
                a = a / s
            if stall > 50:
                step *= 0.7
                if step < 1e-8:
                    break
                rng = np.random.default_rng(it * 31 + 7)
                a = a + rng.normal(0, 0.02, n) * step
                a = np.maximum(a, 0)
                s = np.sum(a)
                if s > 1e-10:
                    a = a / s
                stall = 0
        if it > 200:
            step *= 0.997
    return best_a


def coordinate_descent(a, time_budget=20.0, max_sweeps=50, verbose=False):
    """Coordinate descent with 1D minimization per coordinate."""
    n = len(a)
    a = a.copy().astype(np.float64)
    s = np.sum(a)
    if s < 1e-10:
        a = np.ones(n) / n
    else:
        a = a / s
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    sweep = 0

    while time.monotonic() - start < time_budget and sweep < max_sweeps:
        sweep += 1
        improved = 0
        cur_obj = compute_obj(a)
        S = float(np.sum(a))
        order = np.random.permutation(n) if sweep % 3 == 0 else np.arange(n)

        for j in order:
            if time.monotonic() - start > time_budget:
                return best_a, best_obj
            old_val = a[j]

            def obj_1d(x):
                a[j] = x
                c = compute_conv(a)
                max_c = float(np.max(c))
                s_new = S - old_val + x
                if s_new < 1e-10:
                    return 1e10
                return 2.0 * n * max_c / (s_new * s_new)

            x_hi = max(3.0 * old_val + 1e-6, S / n * 2.0)
            res = minimize_scalar(obj_1d, bounds=(0.0, x_hi), method='bounded',
                                  options={'xatol': 1e-10})
            if res.success:
                a[j] = float(np.clip(res.x, 0, x_hi))
                if abs(a[j] - old_val) > 1e-10:
                    improved += 1
            else:
                a[j] = old_val

        s = np.sum(a)
        if s < 1e-10:
            break
        a = a / s
        obj = compute_obj(a)
        if obj < best_obj - 1e-13:
            best_obj = obj
            best_a = a.copy()
            if verbose and sweep % 5 == 0:
                print(f"    CD sweep {sweep}: {best_obj:.6f}")
        elif improved == 0:
            break
    return best_a, best_obj


def resample_sequence(a, new_n):
    n = len(a)
    if n == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return a_new


def try_parent():
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64)
        parent = np.clip(parent, 0, 1000.0)
        if parent.size > 0 and np.isfinite(parent).all():
            obj = compute_obj(parent)
            print(f"Parent (n={parent.size}): obj={obj:.6f}")
            return parent.copy(), obj
    except Exception:
        pass
    return None, np.inf


def polish_sequence(a, time_budget, verbose=False):
    """Intensive polishing of a single sequence."""
    n = len(a)
    a = a.copy().astype(np.float64)
    start = time.monotonic()
    best_a = a.copy()
    best_obj = compute_obj(a)
    round_num = 0

    while time.monotonic() - start < time_budget - 3:
        round_num += 1
        remaining = time_budget - (time.monotonic() - start)
        if remaining < 5:
            break

        # LP step
        if n <= 800 and remaining > 20:
            a_new = alternating_lp(best_a, max_iters=25, time_budget=min(remaining * 0.3, 30.0))
            obj = compute_obj(a_new)
            if obj < best_obj:
                best_obj = obj
                best_a = a_new.copy()
                if verbose:
                    print(f"    Polish r{round_num} LP: {best_obj:.6f}")

        # L-BFGS-B
        remaining = time_budget - (time.monotonic() - start)
        if remaining > 12:
            tau_small = max(0.05 / n, 0.001)
            a_new, obj_new = optimize_lfbs(best_a, [tau_small, tau_small * 0.3],
                                           max_iter=200, time_budget=min(remaining * 0.2, 12.0))
            if obj_new < best_obj:
                best_obj = obj_new
                best_a = a_new.copy()
                if verbose:
                    print(f"    Polish r{round_num} L-BFGS: {best_obj:.6f}")

        # Subgradient
        remaining = time_budget - (time.monotonic() - start)
        if remaining > 10:
            a_new = subgradient_with_backtracking(best_a, max_iters=int(remaining * 8),
                                                  time_budget=min(remaining * 0.15, 12.0),
                                                  tau_start=10.0, tau_end=50.0)
            obj = compute_obj(a_new)
            if obj < best_obj:
                best_obj = obj
                best_a = a_new.copy()
                if verbose:
                    print(f"    Polish r{round_num} SG: {best_obj:.6f}")

        # Coordinate descent for small n
        remaining = time_budget - (time.monotonic() - start)
        if n <= 400 and remaining > 15:
            a_new, obj_new = coordinate_descent(best_a, time_budget=min(remaining * 0.25, 15.0))
            if obj_new < best_obj:
                best_obj = obj_new
                best_a = a_new.copy()
                if verbose:
                    print(f"    Polish r{round_num} CD: {best_obj:.6f}")

    return best_a, best_obj


def propose_candidate(seed=42, budget_s=1500):
    """
    Main search: explores multiple sequence lengths using L-BFGS-B, LP alternating,
    subgradient, and coordinate descent. Uses spectral initialization, adaptive tau
    annealing, and multi-start with diverse initializations.
    """
    start_time = time.monotonic()
    deadline = start_time + float(budget_s) - 30.0

    best_seq = None
    best_score = np.inf
    rng = np.random.default_rng(seed)

    # Load parent
    parent, parent_obj = try_parent()
    if parent is not None and parent_obj < best_score:
        best_score = parent_obj
        best_seq = parent.copy()

    # N values to explore
    n_values = [300, 400, 500, 600, 700, 800, 1000, 1200]
    if best_seq is not None:
        np_ = len(best_seq)
        extra = [np_, max(200, np_ - 100), min(1500, np_ + 100),
                 min(1500, np_ + 200), min(1500, np_ + 400)]
        n_values = sorted(set(n_values + extra))

    print(f"Exploring n values: {n_values}")
    print(f"Initial best: {best_score:.6f}")

    candidates = []

    for n in n_values:
        if time.monotonic() > deadline - 120:
            break
        remaining = deadline - time.monotonic()
        if remaining < 15:
            break

        # Time allocation
        if n <= 500:
            alloc = min(50.0, remaining * 0.12)
        elif n <= 800:
            alloc = min(65.0, remaining * 0.13)
        else:
            alloc = min(75.0, remaining * 0.14)
        alloc = max(15.0, alloc)

        print(f"\nN={n}: {remaining:.0f}s remaining, alloc={alloc:.0f}s")

        n_inits = max(2, int(alloc / 18))
        n_inits = min(n_inits, 5)

        for init_idx in range(n_inits):
            if time.monotonic() > deadline - 120:
                break
            remaining_init = deadline - time.monotonic()
            if remaining_init < 12:
                break

            if init_idx == 0 and best_seq is not None:
                a = resample_sequence(best_seq, n)
            else:
                a = random_init(n, np.random.default_rng(seed + n * 104729 + init_idx * 79193))

            # Phase A: Subgradient shaping
            sg_time = min(6.0, (alloc / n_inits) * 0.15)
            if remaining_init > sg_time + 10:
                a = subgradient_with_backtracking(a, max_iters=int(sg_time * 30),
                                                  time_budget=sg_time,
                                                  tau_start=2.0, tau_end=25.0)

            # Phase B: L-BFGS-B with tau annealing
            remaining_init = deadline - time.monotonic()
            if remaining_init > 15:
                lfbs_time = min((alloc / n_inits) * 0.35, 25.0)
                tau_list = [8.0 / n, 4.0 / n, 2.0 / n, 1.0 / n,
                            0.5 / n, 0.25 / n, 0.1 / n, 0.05 / n]
                a, obj = optimize_lfbs(a, tau_list, max_iter=250, time_budget=lfbs_time)
            else:
                obj = compute_obj(a)

            # Phase C: LP alternating
            remaining_init = deadline - time.monotonic()
            if remaining_init > 15 and n <= 900:
                lp_time = min((alloc / n_inits) * 0.3, 25.0)
                a = alternating_lp(a, max_iters=35, time_budget=lp_time)
                obj = compute_obj(a)

            # Phase D: Subgradient polish
            remaining_init = deadline - time.monotonic()
            if remaining_init > 8:
                a = subgradient_with_backtracking(a, max_iters=int(remaining_init * 10),
                                                  time_budget=min(remaining_init * 0.1, 10.0),
                                                  tau_start=8.0, tau_end=35.0)
                obj = compute_obj(a)

            # Phase E: Coordinate descent for small n
            remaining_init = deadline - time.monotonic()
            if remaining_init > 10 and n <= 400:
                a, obj = coordinate_descent(a, time_budget=min(remaining_init * 0.15, 10.0))

            print(f"  init{init_idx}: n={n}, obj={obj:.6f}")

            if obj < best_score:
                best_score = obj
                best_seq = a.copy()
                print(f"  ** UPDATE: n={n}, obj={best_score:.6f} **")

            candidates.append((a.copy(), obj, n))

    # Phase 2: Polishing
    print(f"\n=== POLISHING PHASE ===")
    print(f"Best before polish: {best_score:.6f}")

    candidates.sort(key=lambda x: x[1])
    n_polish = min(4, len(candidates))

    for ci in range(n_polish):
        if time.monotonic() > deadline - 80:
            break
        remaining = deadline - time.monotonic()
        if remaining < 25:
            break

        a_cand = candidates[ci][0]
        n_cand = candidates[ci][2]
        print(f"  Polishing candidate {ci} (n={n_cand}, obj={candidates[ci][1]:.6f}), {remaining:.0f}s left")

        polish_budget = min(remaining * 0.35, 90.0)
        a_pol, obj_pol = polish_sequence(a_cand, time_budget=polish_budget, verbose=True)

        if obj_pol < best_score:
            best_score = obj_pol
            best_seq = a_pol.copy()
            print(f"  ** UPDATE after polish: obj={best_score:.6f} **")
        else:
            # Perturbation restart
            remaining = deadline - time.monotonic()
            if remaining > 20:
                pert_rng = np.random.default_rng(ci * 555 + 999 + seed)
                for ptry in range(4):
                    if time.monotonic() > deadline - 40:
                        break
                    scale = pert_rng.uniform(0.02, 0.15)
                    a_pert = a_cand * np.exp(pert_rng.normal(0, scale, n_cand))
                    a_pert = np.maximum(a_pert, 0)
                    s = np.sum(a_pert)
                    if s < 1e-10:
                        continue
                    a_pert = a_pert / s
                    a_pert = subgradient_with_backtracking(a_pert, max_iters=400,
                                                           time_budget=5.0,
                                                           tau_start=5.0, tau_end=30.0)
                    obj_pert = compute_obj(a_pert)
                    if obj_pert < best_score:
                        best_score = obj_pert
                        best_seq = a_pert.copy()
                        print(f"  ** UPDATE (perturb {ptry}): obj={best_score:.6f} **")
                    # Quick LP
                    remaining = deadline - time.monotonic()
                    if remaining > 15 and n_cand <= 700 and obj_pert < best_score:
                        a_pert = alternating_lp(a_pert, max_iters=15, time_budget=min(10.0, remaining * 0.3))
                        obj_pert2 = compute_obj(a_pert)
                        if obj_pert2 < best_score:
                            best_score = obj_pert2
                            best_seq = a_pert.copy()
                            print(f"  ** UPDATE (perturb+LP {ptry}): obj={best_score:.6f} **")

    # Phase 3: Final intensive polish
    if best_seq is not None and time.monotonic() < deadline - 25:
        remaining = deadline - time.monotonic()
        n_best = len(best_seq)
        print(f"\n=== FINAL POLISH: n={n_best}, {remaining:.0f}s ===")

        a_cur = best_seq.copy()
        for final_round in range(5):
            if time.monotonic() > deadline - 15:
                break
            remaining = deadline - time.monotonic()
            if remaining < 18:
                break

            # L-BFGS-B with very small tau
            tau_vsmall = 0.03 / n_best
            a_new, obj_new = optimize_lfbs(a_cur, [0.2 / n_best, tau_vsmall, tau_vsmall * 0.3],
                                           max_iter=400, time_budget=min(remaining * 0.2, 18.0))
            if obj_new < best_score:
                best_score = obj_new
                best_seq = a_new.copy()
                a_cur = a_new.copy()
                print(f"  Final L-BFGS {final_round}: {best_score:.6f}")

            # LP
            remaining = deadline - time.monotonic()
            if remaining > 20 and n_best <= 900:
                a_new = alternating_lp(a_cur, max_iters=30, time_budget=min(remaining * 0.25, 25.0))
                obj_new = compute_obj(a_new)
                if obj_new < best_score:
                    best_score = obj_new
                    best_seq = a_new.copy()
                    a_cur = a_new.copy()
                    print(f"  Final LP {final_round}: {best_score:.6f}")

            # Subgradient
            remaining = deadline - time.monotonic()
            if remaining > 15:
                a_new = subgradient_with_backtracking(a_cur, max_iters=int(remaining * 8),
                                                      time_budget=min(remaining * 0.15, 15.0),
                                                      tau_start=15.0, tau_end=60.0)
                obj_new = compute_obj(a_new)
                if obj_new < best_score:
                    best_score = obj_new
                    best_seq = a_new.copy()
                    a_cur = a_new.copy()
                    print(f"  Final SG {final_round}: {best_score:.6f}")

            # Coordinate descent
            remaining = deadline - time.monotonic()
            if remaining > 12 and n_best <= 500:
                a_new, obj_new = coordinate_descent(a_cur, time_budget=min(remaining * 0.2, 12.0))
                if obj_new < best_score:
                    best_score = obj_new
                    best_seq = a_new.copy()
                    a_cur = a_new.copy()
                    print(f"  Final CD {final_round}: {best_score:.6f}")

    # Final validation
    if best_seq is None:
        best_seq = np.ones(200)
        best_score = compute_obj(best_seq)

    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    val = evaluate_sequence(result)
    if val < np.inf:
        best_score = val
    else:
        result = [1.0] * 100
        best_score = 2.0

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_score:.6f}, time={elapsed:.1f}s ===")

    return result
