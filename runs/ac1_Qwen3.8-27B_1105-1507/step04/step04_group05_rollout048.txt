import numpy as np
from scipy.optimize import linprog
from scipy.sparse import lil_matrix, csr_matrix
import time

def fast_conv(a):
    """Full self-convolution via FFT."""
    n = len(a)
    s = 1
    while s < 2 * n - 1:
        s <<= 1
    fa = np.fft.rfft(a, s)
    return np.fft.irfft(fa * fa, s)[:2 * n - 1]

def compute_J(a):
    """Compute J(a) = 2*n*max(conv(a,a))/(sum(a))^2."""
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))

def build_convolution_matrix(a):
    """Build the convolution matrix A where (A @ g)_k = (a*g)_k."""
    n = len(a)
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        jl = max(0, k - n + 1)
        jh = min(n - 1, k)
        for j in range(jl, jh + 1):
            rows.append(k)
            cols.append(j)
            vals.append(a[k - j])
    A = lil_matrix((2 * n - 1, n))
    for idx in range(len(rows)):
        A[rows[idx], cols[idx]] = vals[idx]
    return A.tocsr()

def lp_best_response_maxsum(a, M):
    """LP: max sum(g) s.t. (a*g)_k <= M for all k, g >= 0."""
    n = len(a)
    c = -np.ones(n)
    A = build_convolution_matrix(a)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                options={'presolve': True, 'time_limit': 10})
    if r.success and r.x is not None:
        g = np.maximum(r.x, 0.0)
        if g.sum() > 1e-10:
            return g
    return None

def lp_best_response_minpeak(a):
    """LP: min t s.t. (a*g)_k <= t for all k, sum(g) = 1, g >= 0."""
    n = len(a)
    # Variables: g_0, ..., g_{n-1}, t (total n+1 variables)
    # Minimize t -> c = [0, ..., 0, 1]
    c = np.zeros(n + 1)
    c[-1] = 1.0
    
    # Constraints:
    # (a*g)_k - t <= 0 for all k (2n-1 constraints)
    # sum(g) = 1 (1 equality constraint)
    # g_i >= 0, t free (but t >= 0 implicitly)
    
    A_ub_rows = []
    b_ub_vals = []
    for k in range(2 * n - 1):
        row = np.zeros(n + 1)
        jl = max(0, k - n + 1)
        jh = min(n - 1, k)
        for j in range(jl, jh + 1):
            row[j] = a[k - j]
        row[-1] = -1.0  # -t
        A_ub_rows.append(row)
        b_ub_vals.append(0.0)
    
    A_ub = np.array(A_ub_rows)
    b_ub = np.array(b_ub_vals)
    
    # Equality: sum(g) = 1
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])
    
    bounds = [(0.0, None)] * n + [(0.0, None)]  # g >= 0, t >= 0
    
    r = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                bounds=bounds, method='highs',
                options={'presolve': True, 'time_limit': 10})
    if r.success and r.x is not None:
        g = np.maximum(r.x[:n], 0.0)
        if g.sum() > 1e-10:
            return g / g.sum()
    return None

def coordinate_descent_pass(a, b=None, S=None):
    """One full pass of coordinate descent with incremental updates."""
    a = np.maximum(a.astype(np.float64), 0.0)
    if S is None:
        S = a.sum()

        return a, float('inf')
    if b is None:
        b = fast_conv(a).copy()
    else:
        b = b.copy()
    n = len(a)
    
    for i in range(n):
        new_val = coordinate_descent_1d(a, i, b, S)
        if abs(new_val - a[i]) > 1e-12:
            old_val = a[i]
            delta = new_val - old_val
            a[i] = new_val
            S = S - old_val + new_val
            # Update convolution
            for k in range(i, min(2*n-1, i+n)):
                ki = k - i
                if 0 <= ki < n:
                    if k == 2 * i:
                        b[k] += delta * (new_val + old_val)  # new^2 - old^2 = (new-old)(new+old)
                    else:
                        b[k] += 2.0 * delta * a[ki]
    
    # Verify
    b_check = fast_conv(a)
    new_J = 2.0 * n * b_check.max() / (S * S)
    return a, new_J

import numpy as np

def fast_conv(a):
    """Full self-convolution via FFT."""
    n = len(a)
    s = 1
    while s < 2 * n - 1:
        s <<= 1
    fa = np.fft.rfft(a, s)
    return np.fft.irfft(fa * fa, s)[:2 * n - 1]


def compute_J(a):
    """Compute J(a) = 2*n*max(conv(a,a))/(sum(a))^2."""
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))


def build_conv_matrix(a):
    """Build sparse convolution constraint matrix."""
    n = len(a)
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        jl = max(0, k - n + 1)
        jh = min(n - 1, k)
        for j in range(jl, jh + 1):
            rows.append(k)
            cols.append(j)
            vals.append(a[k - j])
    A = lil_matrix((2 * n - 1, n))
    for idx in range(len(rows)):
        A[rows[idx], cols[idx]] = vals[idx]
    return A.tocsr()


def lp_maxsum(a, M, time_limit=5):
