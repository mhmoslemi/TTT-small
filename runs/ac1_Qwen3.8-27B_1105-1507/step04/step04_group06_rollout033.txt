                # Evaluate at a few more points near the optimum
                for t_try in [t_star, max(0, t_star - 0.01 * t_ub), t_star + 0.01 * t_ub, 0.0]:
                    t_try = max(0.0, min(t_try, t_ub))
                    J_try = eval_1d(t_try)
                    if t_try == t_star or J_try < eval_1d(t_star):
                        if t_try == t_star:
                            continue
                # Just use the golden section result
                new_val = t_star

            else:
                new_val = old_val

            # Update
            delta = new_val - old_val
            if abs(delta) < 1e-12:
                continue

            a[i] = new_val
            s_total += delta

            # Update B
            if k_max >= k_min:
                B[k_range] += delta * D[k_range]
                if 2 * i < 2 * n - 1:
                    B[2 * i] += new_val * new_val - old_val * old_val

            # Check improvement
            J_now = float(2 * n * B.max() / (s_total * s_total))
            if J_now < best_J:
                best_J = J_now
                best_a = a.copy()
                if tracker is not None:
                    tracker.update(a)

    if tracker is not None:
        tracker.update(best_a)
    return best_a


def smooth_obj_and_grad(a, beta):
    """Smooth approximation of max(conv(a,a)) using log-sum-exp, plus gradient."""
    n = len(a)
    b = fast_conv(a)
    b_max = float(b.max())
    shifted = beta * (b - b_max)
    shifted = np.clip(shifted, -500, 500)
    exp_s = np.exp(shifted)
    Z = float(exp_s.sum())
    if Z < 1e-300:
        p = np.zeros(2 * n - 1)
        p[int(np.argmax(b))] = 1.0
    else:
        p = exp_s / Z
    g_val = b_max + np.log(max(Z, 1e-300)) / beta

    # Gradient of max(conv(a,a)) w.r.t. a: grad_i = 2 * sum_k p_k * a[k-i] (with range checks)
    # (conv(a,a))[k] = sum_j a[j]*a[k-j]
    # d/dc a_i = sum_k p_k * 2*a[k-i] (with valid range)
    # = 2 * (p_reversed * a) at position i... let me think
    # d/dc a_i = sum_k p_k * [a[k-i] (if valid) + a[i-k] (if valid, different term)]
    # Actually for self-convolution: d/dc a_i = 2 * sum_k p_k * a[k-i] (counting both j=i and k-j=i)
    # More precisely: d/dc a_i = sum_k p_k * (a[k-i] * 1(0<=k-i<n) + a[i-k] * 1(0<=i-k<n))
    # But a[k-i] + a[i-k] counts the i term twice when k-i = i-k, i.e., k=2i.
    # For self-conv: d/dc a_i = 2 * sum_k p_k * a[k-i] * 1(0<=k-i<n) - p_{2i} * a[i]
    # Actually let's just use: d/dc a_i = (p_flipped * a)[i] + (p * a)[i] ... no.
    # 
    # Simplest: d/dc a_i = 2 * (np.convolve(p, a, 'full'))[i + (n-1)] ... no that's cross-corr.
    # 
    # Let me just do it directly:
    # d/d a_i [sum_j a_j * a_{k-j}] = a_{k-i} (if 0<=k-i<n) + a_{i-k} (if 0<=i-k<n and i-k != k-i)
    # For self-conv, a_{k-i} and a_{i-k} are the same when we sum over all k with p_k weights:
    # d/d a_i = sum_k p_k * a[k-i] * 1(0<=k-i<n) + sum_k p_k * a[i-k] * 1(0<=i-k<n, i-k!=k-i)
    # The second sum: let m = i-k, so k = i-m. Sum over m: p_{i-m} * a[m] * 1(0<=m<n, m!=i)
    # = sum_{m != i} p[i-m] * a[m] (with range checks)
    # The first sum: sum_{j=0}^{n-1} p[k] * a[k-i] where k-i=j, so k=j+i: sum_j p[j+i]*a[j]
    # 
    # Combined: d/d a_i = sum_j p[j+i]*a[j] + sum_{m!=i} p[i-m]*a[m]
    # = sum_j p[j+i]*a[j] + sum_m p[i-m]*a[m] - p[0]*a[i] (removing m=i term which has p[i-i]=p[0])
    
    # Use cross-correlation:
    # First term: sum_j p[j+i]*a[j] for j in [0,n-1] and j+i in [0,2n-2]
    # This is (a * p_rev)[i] where p_rev is p reversed... let me just use FFT.
    
    # Actually the simplest correct formula:
    # d/d a_i of conv(a,a) [dot] p = 2 * (cross_corr(p, a))[i]
    # where cross_corr(p, a)[i] = sum_k p[k] * a[k-i] (valid range)
    # But we need to be careful about the factor of 2 and the self-term.
    
    # Let me verify: conv(a,a)[k] = sum_j a[j]*a[k-j]
    # d/dc a_i = sum_j [delta(j,i)*a[k-j] + a[j]*delta(k-j,i)]
    # = a[k-i] (if 0<=k-i<n) + sum_j a[j]*delta(k-j,i)
    # = a[k-i] (if 0<=k-i<n) + a[k-i] (if 0<=k-i<n)
    # Wait no: sum_j a[j]*delta(k-j,i) = a[k-i] (if 0<=k-i<n)
    # So d/dc a_i = a[k-i]*1(0<=k-i<n) + a[k-i]*1(0<=k-i<n) = 2*a[k-i]*1(0<=k-i<n)
    # But when k-i=i (k=2i), we get 2*a[i], not a[i]^2. That's because we're differentiating
    # a[j]*a[k-j] w.r.t. a_i, and when j=i AND k-j=i, it's the same variable, so:
    # d/dc a_i [a[i]*a[i]] = 2*a[i]
    # So the formula d/dc a_i = 2*a[k-i]*1(0<=k-i<n) is correct for all k.
    
    # Therefore: grad_i = sum_k p_k * 2 * a[k-i] * 1(0<=k-i<n)
    # = 2 * sum_{j=0}^{n-1} p[j+i] * a[j]  (where j = k-i, k=j+i)
    # This is a cross-correlation of p and a.
    
    # Using FFT: cross_corr(p, a)[i] = sum_j p[j+i]*a[j] = (IFFT(FFT(p) * conj(FFT(a))))[i + (2n-2)]
    # Actually, let's use np.correlate or direct computation.
    
    # p has length 2n-1, a has length n.
    # cross_corr[i] = sum_{j: 0<=j<n, 0<=j+i<2n-1} p[j+i] * a[j]
    # For i in [0, n-1]: j+i ranges from i to i+n-1, all in [0, 2n-2] since i+n-1 <= 2n-2.
    
    if n <= 1:
        grad = np.array([2.0 * a[0] * p.sum()]) if n == 1 else np.zeros(n)
    else:
        # Use FFT for cross-correlation
        size = 1
        while size < 3 * n:
            size <<= 1
        F_p = np.fft.rfft(p, size)
        F_a = np.fft.rfft(a, size)
        # We want sum_j p[j+i]*a[j] for each i
        # This is (p * a_flipped)[i + (n-1)] in full convolution... 
        # Actually: sum_j p[j+i]*a[j] = (a * p_rev_full)[i + length_a - 1] ... 
        # Let me just use: result[i] = sum_j a[j] * p[j+i]
        # = convolve(a, p)[i + (n-1)] ... no.
        # convolve(a, p, 'full')[m] = sum_j a[j] * p[m-j]
        # We want sum_j a[j] * p[j+i] = sum_j a[j] * p[(j+i)]
        # Let m-j = j+i => m = 2j+i. That doesn't work.
        # 
        # Alternative: sum_j a[j]*p[j+i] = <a, shift_{-i}(p)> = cross-correlation
        # = IFFT(conj(FFT(a)) * FFT(p)) ... no, that's circular.
        # 
        # Let me just do it directly with the FFT trick:
        # sum_j a[j]*p[j+i] for j=0..n-1
        # = sum_j a[j]*p[j+i]
        # Pad p to length size, take the segment p[i:i+n], multiply element-wise with a, sum.
        # For all i simultaneously: this is a Toeplitz matrix-vector product.
        # 
        # FFT approach: embed in circular convolution.
        # If we pad a to length size (zeros after n), then:
        # circular_conv(a, p)[k] = sum_j a[j] * p[(k-j) mod size]
        # We want sum_j a[j] * p[j+i] = sum_j a[j] * p[(j+i) mod size] (if size is large enough)
        # = circular_conv(a, p_shifted)[?]
        # Hmm, let me just use:
        # result = np.correlate(p, a, mode='valid') ... no, that gives sum_j p[j+i]*a[j] for i=0..(len(p)-len(a))
        # p has length 2n-1, a has length n. valid gives 2n-1-n+1 = n values.
        # np.correlate(p, a, 'valid')[i] = sum_j p[j+i]*a[j] for j=0..n-1. Yes!
        
        grad = 2.0 * np.correlate(p, a, mode='valid')
    
    return g_val, grad


def F_and_grad(a, beta):
    """Compute smooth F(a) = 2n*logsumexp(beta*conv(a,a))/(sum a)^2 and gradient."""
    n = len(a)
    s = float(a.sum())
    if s < 1e-10:
        return 1e10, np.zeros(n)
    g_val, grad_g = smooth_obj_and_grad(a, beta)
    F_val = 2.0 * n * g_val / (s * s)
    grad_F = 2.0 * n * (s * grad_g - 2.0 * g_val * np.ones(n)) / (s ** 3)
    return F_val, grad_F


def lbfgs_optimize(a0, deadline):
    """L-BFGS-B with beta annealing."""
    a = np.maximum(a0.astype(np.float64), 1e-8).copy()
    n = len(a)
    betas = [(0.5, 30), (2.0, 30), (10.0, 40), (30.0, 50), (100.0, 60)]

    for beta, max_iters in betas:
        if time.monotonic() > deadline:
            break

        def func(x):
            x2 = np.maximum(x, 0.0)
            f, _ = F_and_grad(x2, beta)
            return f if np.isfinite(f) else 1e10

        def jac(x):
            x2 = np.maximum(x, 0.0)
            _, g = F_and_grad(x2, beta)
            g = np.nan_to_num(g, nan=0.0, posinf=0.0, neginf=0.0)
            g[x2 < 1e-14] = 0.0
            return g

        bounds = [(0.0, None)] * n
        try:
            res = minimize(func, a.copy(), jac=jac, method='L-BFGS-B',
                           bounds=bounds,
                           options={'maxiter': max_iters, 'max_fun': max_iters * 30,
                                    'gtol': 1e-9, 'ftol': 1e-15})
            if res.fun < 1e10:
                new_a = np.maximum(res.x, 1e-10)
                if new_a.sum() > 1e-6:
                    a = new_a
        except Exception:
            pass

    return a


def resample_seq(src, dest_n):
    """Resample 1D sequence to new length via linear interpolation."""
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)


def generate_inits(n, rng, parent=None):
    """Generate diverse initial sequences."""
    inits = []
    t = np.linspace(0, 1, n, endpoint=False)

    if parent is not None and parent.size > 0:
        inits.append(resample_seq(parent, n))
        p = resample_seq(parent, n)
        inits.append(np.maximum(p + rng.normal(0, 0.05, n) * p, 0.01))

    inits.append(np.ones(n))
    inits.append(rng.uniform(0.3, 1.0, n))
    for freq in [2, 3, 4, 5, 7, 11]:
        inits.append(1.0 + 0.35 * np.cos(2.0 * np.pi * freq * t))
    inits.append(np.exp(-4.0 * (t - 0.5) ** 2) + 0.05)
    inits.append(np.abs(np.sin(np.pi * t)) + 0.15)
    inits.append(1.0 + 0.3 * np.cos(4.0 * np.pi * t) + 0.1 * np.cos(8.0 * np.pi * t))
    # Hamming window
    inits.append(0.54 - 0.46 * np.cos(2.0 * np.pi * t))
    # Blackman
    inits.append(0.42 - 0.5 * np.cos(2.0 * np.pi * t) + 0.08 * np.cos(4.0 * np.pi * t))
    # Sawtooth
    inits.append(1.0 + 0.3 * np.sin(4.0 * np.pi * t))
    return inits


class ScoreTracker:
    def __init__(self):
        self.best = None
        self.best_score = float('inf')
        self.t_start = time.monotonic()

    def update(self, a):
        a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
        if a.size == 0 or a.sum() < 0.01:
            return
        score = float(evaluate_sequence(a.tolist()))
        if np.isfinite(score) and score < self.best_score:
            self.best = a.copy()
            self.best_score = score
            elapsed = time.monotonic() - self.t_start
            print(f"  * NEW BEST: J={score:.8f} n={a.size} t={elapsed:.1f}s", flush=True)

    def report(self, label=""):
        if self.best is not None:
            print(f"  {label} best: J={self.best_score:.8f} n={self.best.size}", flush=True)


def propose_candidate(seed=42, budget_s=1500, **kwargs):
    """
    Hybrid optimization: L-BFGS-B basin finding -> LP mirror descent ->
    Coordinate descent -> Interleaved refinement -> Final polish.
    """
    rng = np.random.default_rng(seed)
    total_deadline = time.monotonic() + float(budget_s) - 15.0
    tracker = ScoreTracker()

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        if p.ndim == 1 and p.size > 0 and np.all(np.isfinite(p)):
            parent = np.maximum(p, 0.0)
            tracker.update(parent)
            print(f"Parent loaded: n={parent.size}, J={tracker.best_score:.8f}", flush=True)
    except (NameError, AttributeError):
        pass

    if parent is None:
        parent = np.ones(512)
        tracker.update(parent)
        print(f"Default parent: n=512, J={tracker.best_score:.8f}", flush=True)

    # === Phase 1: L-BFGS-B basin finding (100s) ===
    p1_budget = 100.0
    deadline_p1 = time.monotonic() + p1_budget
    print(f"\n{'='*50} Phase 1: L-BFGS-B ({p1_budget:.0f}s) {'='*50}", flush=True)

    n_main = tracker.best.size
    lengths_p1 = sorted(set([n_main, 512, 1024]))
    lengths_p1 = [n for n in lengths_p1 if 128 <= n <= 2048]

    for n in lengths_p1:
        if time.monotonic() > deadline_p1:
            break
        t_start = time.monotonic()
        print(f"  L-BFGS-B n={n}...", flush=True)
        inits = generate_inits(n, rng, parent)
        n_inits = min(len(inits), max(4, len(inits) // 2))
        for a0 in inits[:n_inits]:
            if time.monotonic() > deadline_p1:
                break
            a0 = np.maximum(a0, 0.01)
            a0 = a0 / a0.sum()
            a_opt = lbfgs_optimize(a0, deadline_p1)
            if a_opt is not None and a_opt.sum() > 0.01:
                tracker.update(a_opt)
        print(f"    n={n} done in {time.monotonic()-t_start:.1f}s", flush=True)

    tracker.report("Phase 1")

    # === Phase 2: LP Mirror Descent (550s) ===
    if tracker.best is not None and time.monotonic() < total_deadline:
        n_main = tracker.best.size
        p2_budget = 550.0
        deadline_p2 = min(total_deadline, time.monotonic() + p2_budget)
        print(f"\n{'='*50} Phase 2: LP-MD n={n_main} ({p2_budget:.0f}s) {'='*50}", flush=True)

        # Primary at best length
        max_iters = 5000
        a = lp_mirror_descent(tracker.best, max_iters, deadline_p2, tracker=tracker)

        # Try other lengths
        for n_try in [512, 768, 1024, 1536, 2048]:
            if time.monotonic() > deadline_p2:
                break
            if n_try == n_main:
                continue
            a_res = resample_seq(tracker.best, n_try)
            a_res = a_res / max(a_res.sum(), 1e-10)
            budget_here = min(120.0, deadline_p2 - time.monotonic())
            if budget_here < 20:
                break
            max_iters_here = int(budget_here * 3)
            a_res = lp_mirror_descent(a_res, max_iters_here, deadline_p2, tracker=tracker)

        tracker.report("Phase 2")

    # === Phase 3: Coordinate Descent (400s) ===
    if tracker.best is not None and time.monotonic() < total_deadline:
        n_main = tracker.best.size
        p3_budget = 400.0
        deadline_p3 = min(total_deadline, time.monotonic() + p3_budget)
        print(f"\n{'='*50} Phase 3: Coord Descent n={n_main} ({p3_budget:.0f}s) {'='*50}", flush=True)

        # Estimate sweeps: each sweep ~ 0.3-0.5s for n=1024
        est_sweep_time = 0.4
        n_sweeps = int(p3_budget / est_sweep_time)
        n_sweeps = max(5, min(n_sweeps, 2000))
        print(f"  Planning {n_sweeps} sweeps", flush=True)

        t_start = time.monotonic()
        a_cd = coordinate_descent(tracker.best, n_sweeps, deadline_p3,
                                  tracker=tracker, rng=rng)
        elapsed = time.monotonic() - t_start
        print(f"  CD done in {elapsed:.1f}s", flush=True)

        # Second pass at different length
        if time.monotonic() < total_deadline:
            for n_try in [n_main - 128, n_main + 128, 1024]:
                if n_try < 128 or n_try > 2048 or time.monotonic() > deadline_p3:
                    continue
                if n_try == n_main:
                    continue
                a_res = resample_seq(tracker.best, n_try)
                a_res = a_res / max(a_res.sum(), 1e-10)
                n_sweeps2 = max(5, int(min(60.0, deadline_p3 - time.monotonic()) / est_sweep_time))
                a_res = coordinate_descent(a_res, n_sweeps2, deadline_p3,
                                           tracker=tracker, rng=rng)

        tracker.report("Phase 3")

    # === Phase 4: Interleaved LP-MD + CD (300s) ===
    if tracker.best is not None and time.monotonic() < total_deadline:
        n_main = tracker.best.size
        p4_budget = 300.0
        deadline_p4 = min(total_deadline, time.monotonic() + p4_budget)
        print(f"\n{'='*50} Phase 4: Interleaved ({p4_budget:.0f}s) {'='*50}", flush=True)

        a = tracker.best.copy().astype(np.float64)
        a = a / a.sum()

        # Alternate between LP-MD (short) and CD (short)
        while time.monotonic() < deadline_p4:
            # LP-MD chunk (30s)
            dl = min(time.monotonic() + 30.0, deadline_p4)
            a = lp_mirror_descent(a, 500, dl, tracker=tracker, n_stall=20)

            # CD chunk (20s)
            dl = min(time.monotonic() + 20.0, deadline_p4)
            n_sw = max(2, int(20.0 / 0.4))
            a = coordinate_descent(a, n_sw, dl, tracker=tracker, rng=rng)

            # Perturb and restart if stuck
            if time.monotonic() < deadline_p4:
                a_pert = a + rng.normal(0, 0.03, len(a)) * a
                a_pert = np.maximum(a_pert, 1e-6)
                a_pert = a_pert / a_pert.sum()
                J_pert = compute_J(a_pert)
                if J_pert < tracker.best_score:
                    tracker.update(a_pert)
                a = a_pert if J_pert < compute_J(a) else a

        tracker.report("Phase 4")

    # === Phase 5: Final polish (remaining) ===
    if tracker.best is not None and time.monotonic() < total_deadline:
        n_main = tracker.best.size
        remaining = total_deadline - time.monotonic()
        print(f"\n{'='*50} Phase 5: Final Polish (remaining {remaining:.0f}s) {'='*50}", flush=True)

        if remaining > 30:
            # Final LP-MD with high precision line search
            max_iters = int(min(200.0, remaining * 0.5) * 3)
            a = lp_mirror_descent(tracker.best, max_iters, total_deadline, tracker=tracker, n_stall=50)

            # Final CD
            if time.monotonic() < total_deadline:
                n_sw = max(3, int((total_deadline - time.monotonic()) * 0.3 / 0.4))
                a = coordinate_descent(tracker.best, n_sw, total_deadline, tracker=tracker, rng=rng)

        tracker.report("Phase 5")

    # Fallback
    if tracker.best is None:
        tracker.best = np.ones(512)
        tracker.best_score = float(evaluate_sequence(tracker.best.tolist()))
        print(f"  Fallback: J={tracker.best_score:.8f}", flush=True)

    print(f"\n{'='*50} FINAL {'='*50}", flush=True)
    print(f"J={tracker.best_score:.8f} n={tracker.best.size} "
          f"time={time.monotonic()-tracker.t_start:.1f}s", flush=True)

    result = np.clip(tracker.best, 0.0, 1000.0)
    if result.sum() < 0.01:
        result = np.ones(128)
    return [float(x) for x in result]
