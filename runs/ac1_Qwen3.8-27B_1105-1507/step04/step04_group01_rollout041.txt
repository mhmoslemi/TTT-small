"""
Optimized step-function search combining soft-max L-BFGS-B, LP alternating,
adaptive coordinate descent, and perturbation-based restarts to minimize
2*n*max(conv(a,a))/(sum(a))^2 for non-negative sequences.
"""

import numpy as np
from scipy.optimize import minimize, linprog
from scipy.fft import rfft, irfft
import time


def compute_conv(a):
    """Compute linear convolution a*a using FFT. Returns array of length 2n-1."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    conv = irfft(fa * fa, size)
    return conv[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    conv = compute_conv(a)
    max_b = float(np.max(conv))
    sum_a = float(np.sum(a))
    if sum_a < 0.01:
        return np.inf
    return 2.0 * n * max_b / (sum_a * sum_a)


def fft_convolve(a, b):
    """Linear convolution of a and b using FFT."""
    n, m = len(a), len(b)
    size = 1
    while size < n + m - 1:
        size <<= 1
    fa = rfft(a, size)
    fb = rfft(b, size)
    return irfft(fa * fb, size)[:n + m - 1]


def softmax(z):
    """Numerically stable softmax."""
    z = z - np.max(z)
    e = np.exp(z)
    s = np.sum(e)
    return e / s


def soft_max_obj_and_grad_z(z, tau):
    """
    Compute soft-max of conv(softmax(z), softmax(z)) and its gradient w.r.t. z.
    """
    n = len(z)
    a = softmax(z)
    conv = compute_conv(a)
    cmax = float(np.max(conv))
    shifted = np.clip((conv - cmax) / tau, -500.0, 500.0)
    exp_conv = np.exp(shifted)
    Z = float(np.sum(exp_conv))
    soft_max_val = cmax + tau * np.log(Z)
    w = exp_conv / Z

    # grad_a[i] = 2 * sum_j w[i+j] * a[j]
    w_rev = w[::-1]
    conv_fw = fft_convolve(a, w_rev)
    f1 = conv_fw[n - 1:2 * n - 1][::-1]
    grad_a = 2.0 * f1

    mean_grad = float(np.dot(a, grad_a))
    grad_z = a * (grad_a - mean_grad)
    return soft_max_val, grad_z


def optimize_lbfgs(a_init, tau_list, max_iter=200, time_budget=30.0):
    """Optimize using L-BFGS-B with soft-max parametrization and tau annealing."""
    n = len(a_init)
    a = a_init.copy()
    s = np.sum(a)
    if s < 1e-10:
        a = np.ones(n) / n
    else:
        a = a / s
    a = np.maximum(a, 1e-30)
    z = np.log(a)

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    for tau in tau_list:
        if time.monotonic() - start > time_budget:
            break

        def func_and_grad(z_val):
            return soft_max_obj_and_grad_z(z_val, tau)

        try:
            result = minimize(func_and_grad, z, jac=True, method='L-BFGS-B',
                              options={'maxiter': max_iter, 'ftol': 1e-15, 'gtol': 1e-9})
            z_new = result.x
        except Exception:
            continue

        a_new = softmax(z_new)
        a_new = np.maximum(a_new, 0)
        s_new = np.sum(a_new)
        if s_new < 1e-10:
            continue
        a_new = a_new / s_new
        obj = compute_obj(a_new)

        if obj < best_obj:
            best_obj = obj
            best_a = a_new.copy()
            z = z_new
    return best_a, best_obj


def build_toeplitz(a):
    """Build Toeplitz matrix A where (A@g)[k] = (conv(a,g))[k]."""
    n = len(a)
    A = np.zeros((2 * n - 1, n), dtype=np.float64)
    for j in range(n):
        k_start = j
        k_end = min(j + n, 2 * n - 1)
        A[k_start:k_end, j] = a[:k_end - k_start]
    return A


def solve_lp_g(a, M, time_limit=15):
    """Solve LP: max sum(g) s.t. (a*g)[k] <= M for all k, g >= 0."""
    n = len(a)
    A = build_toeplitz(a)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': time_limit})
        if res.status == 0:
            g = np.maximum(res.x, 0)
            s = float(np.sum(g))
            if s > 1e-10:
                return g / s, s
    except Exception:
        pass
    return None, None


def solve_lp_minpeak(a, time_limit=15):
    """Solve LP: min M s.t. (a*g)[k] <= M for all k, sum(g)=1, g>=0."""
    n = len(a)
    A = build_toeplitz(a)
    c = np.zeros(n + 1)
    c[-1] = 1.0
    A_ub = np.zeros((2 * n - 1, n + 1))
    A_ub[:, :n] = A
    A_ub[:, n] = -1.0
    b_ub = np.zeros(2 * n - 1)
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])
    bounds = [(0.0, None)] * n + [(0.0, None)]
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': time_limit})
        if res.status == 0:
            g = np.maximum(res.x[:n], 0)
            s = np.sum(g)
            if s > 1e-10:
                g = g / s
                M_opt = float(res.x[n])
                return g, M_opt
    except Exception:
        pass
    return None, None

def subgradient_optimize(a, max_iters=2000, time_budget=15.0, tau_start=5.0, tau_end=50.0):

    best_obj = compute_obj(a)

    step = 0.05
    stall = 0

    for it in range(max_iters):
        if time.monotonic() - start > time_budget:
            break

        frac = it / max(1, max_iters)
        tau = tau_start + (tau_end - tau_start) * frac

        conv = compute_conv(a)
        max_conv = float(np.max(conv))
        shifted = np.clip(tau * (conv - max_conv), -300, 300)
        weights = np.exp(shifted)
        w_sum = np.sum(weights)
        if w_sum < 1e-300:
            idx = int(np.argmax(conv))

        grad_centered = grad_a - np.mean(grad_a)

        a_new = np.maximum(a_new, 0)
        s_new = np.sum(a_new)
        if s_new > 1e-10:

            stall = 0
        else:
            stall += 1
            if stall > 80:

                    break
                rng = np.random.default_rng(it * 31 + 7)
                a = a + rng.normal(0, 0.005 * step, n)
                a = np.maximum(a, 0)
                s = np.sum(a)
                if s > 1e-10:
                    a = a / s
                stall = 0

def coordinate_transfer_search(a, max_iters=500, time_budget=20.0):
    """
    Adaptive coordinate transfer local search.
    Tries small transfers of mass between coordinates to reduce max(conv).
    """
    n = len(a)
    a = a.copy().astype(np.float64)
    s = np.sum(a)
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    best_conv = compute_conv(a)
    best_max = float(np.max(best_conv))
    start = time.monotonic()

    eps_base = 1.0 / n
    accepted = 0
    rejected = 0

    for it in range(max_iters):
        if time.monotonic() - start > time_budget:
            break

        # Find the bottleneck index
        k_star = int(np.argmax(best_conv))
        current_max = best_conv[k_star]

        # Find indices i that contribute most to conv[k_star]
        # conv[k_star] = sum_j a[j] * a[k_star - j]
        contributions = np.zeros(n)
        for j in range(n):
            k_j = k_star - j
            if 0 <= k_j < n:
                contributions[j] = a[j] * a[k_j]

        # Sort by contribution (descending)
        top_idx = np.argsort(contributions)[::-1][:20]

        found_improvement = False
        for i in top_idx:
            if time.monotonic() - start > time_budget:
                break
            if contributions[i] < 1e-15:
                continue

            # Try transferring mass FROM i TO some j
            # We want to reduce conv[k_star], so decrease a[i] (if i contributes)
            # and increase a[j] where j does NOT contribute to conv[k_star]

            # Find good j candidates: those that don't contribute to conv[k_star]
            # a[j] contributes to conv[k_star] if 0 <= k_star - j < n, i.e., j in [k_star-n+1, k_star]
            # We want j NOT in that range
            j_candidates = []
            for j in range(n):
                if j == i:
                    continue
                k_j = k_star - j
                if not (0 <= k_j < n):
                    j_candidates.append(j)

            if not j_candidates:
                continue

            # Try a few j candidates
            rng = np.random.default_rng(it * 1000 + i)
            j_trials = rng.choice(j_candidates, size=min(5, len(j_candidates)), replace=False)

            for j in j_trials:
                # Transfer: a[i] -= eps, a[j] += eps (keep sum constant)
                for scale in [0.5, 1.0, 2.0]:
                    eps = eps_base * scale
                    if a[i] < eps:
                        continue
                    a_try = a.copy()
                    a_try[i] -= eps
                    a_try[j] += eps
                    obj_try = compute_obj(a_try)
                    if obj_try < best_obj - 1e-14:
                        a = a_try
                        best_obj = obj_try
                        best_a = a.copy()
                        best_conv = compute_conv(a)
                        best_max = float(np.max(best_conv))
                        found_improvement = True
                        accepted += 1
                        break
                if found_improvement:
                    break
            if found_improvement:
                break

        if not found_improvement:
            rejected += 1
            # Occasionally perturb to escape local minimum
            if rejected > 100:
                rng = np.random.default_rng(it * 137 + 7)
                n_perturb = max(3, n // 50)
                idx = rng.choice(n, size=n_perturb, replace=False)
                a = a.copy()
                for idx_j in idx:
                    if a[idx_j] > 1e-10:
                        a[idx_j] *= np.exp(rng.normal(0, 0.03))
                a = np.maximum(a, 0)
                s = np.sum(a)
                if s > 1e-10:
                    a = a / s
                rejected = 0

        # Adapt eps
        if accepted > 0 and it % 50 == 0:
            eps_base *= 0.9

    return best_a


def pairwise_transfer(a, max_iters=300, time_budget=15.0):
    """
    Pairwise mass transfer: try moving mass from one coordinate to another
    to reduce the peak of the convolution.
    """
    n = len(a)
    a = a.copy().astype(np.float64)
    s = np.sum(a)
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    eps = 0.5 / n

    for it in range(max_iters):
        if time.monotonic() - start > time_budget:
            break

        conv = compute_conv(a)
        k_star = int(np.argmax(conv))
        max_val = conv[k_star]

        # Find top contributors to conv[k_star]
        contrib = np.zeros(n)
        for j in range(n):
            k_j = k_star - j
            if 0 <= k_j < n:
                contrib[j] = a[j] * a[k_j]

        # Top contributor to decrease
        i = int(np.argmax(contrib))
        if contrib[i] < 1e-15:
            break

        # Find best j to receive mass
        # j should not contribute to conv[k_star]
        best_j = -1
        best_obj_j = best_obj
        for j in range(n):
            if j == i:
                continue
            k_j = k_star - j
            if 0 <= k_j < n:
                continue  # j also contributes to conv[k_star], skip
            # Try transfer
            if a[i] < eps:
                break
            a_try = a.copy()
            a_try[i] -= eps
            a_try[j] += eps
            obj_try = compute_obj(a_try)
            if obj_try < best_obj_j:
                best_obj_j = obj_try
                best_j = j

        if best_j >= 0:
            a[i] -= eps
            a[best_j] += eps
            a = np.maximum(a, 0)
            s = np.sum(a)
            if s > 1e-10:
                a = a / s
            obj = compute_obj(a)
            if obj < best_obj:
                best_obj = obj
                best_a = a.copy()
        else:
            # Try smaller eps
            eps *= 0.8
            if eps < 1e-6 / n:
                # Perturb
                rng = np.random.default_rng(it * 7 + 3)
                n_p = max(2, n // 100)
                idx = rng.choice(n, size=n_p, replace=False)
                a = a.copy()
                for idx_j in idx:
                    a[idx_j] = max(0, a[idx_j] * np.exp(rng.normal(0, 0.02)))
                s = np.sum(a)
                if s > 1e-10:
                    a = a / s
                eps = 0.5 / n

    return best_a


def random_init(n, seed):
    """Generate diverse random initializations."""
    rng = np.random.default_rng(seed)

    if choice == 0:
        f = rng.uniform(0.5, 4.0)
        phase = rng.uniform(0, 2 * np.pi)
        a = 1.0 + 0.7 * np.cos(2 * np.pi * f * k / n + phase)
    elif choice == 1:
        f1 = rng.uniform(0.3, 2.0)
        f2 = rng.uniform(2.0, 6.0)
        a = 1.0 + 0.3 * np.cos(2 * np.pi * f1 * k / n) + 0.3 * np.cos(2 * np.pi * f2 * k / n)
    elif choice == 2:
        # Parabolic (edge-weighted)
        t = 2.0 * k / (n - 1) - 1.0
        alpha = rng.uniform(0.3, 1.5)
        a = 1.0 + alpha * t * t
    elif choice == 3:
        # Inverted parabolic (center-weighted)
        t = 2.0 * k / (n - 1) - 1.0
        alpha = rng.uniform(0.3, 1.5)
        a = 1.0 + alpha * (1.0 - t * t)
    elif choice == 4:
        # Two bumps
        c1 = n * rng.uniform(0.1, 0.4)
        c2 = n * rng.uniform(0.6, 0.9)
        w = n * rng.uniform(0.05, 0.2)
        a = np.exp(-0.5 * ((k - c1) / w) ** 2) + np.exp(-0.5 * ((k - c2) / w) ** 2) + 0.1
    elif choice == 5:
        # Cosine window
        a = 0.5 + 0.5 * np.cos(np.pi * k / (n - 1))
        a += 0.1 * rng.random(n)
    elif choice == 6:
        # Random smoothed
        a = rng.random(n)
        win = max(3, n // 30)
        kernel = np.ones(win) / win
        a = np.convolve(a, kernel, mode='same')
    elif choice == 7:
        # Geometric decay from edges
        alpha = rng.uniform(0.01, 0.05)
        a = np.exp(-alpha * np.minimum(k, n - 1 - k))
    elif choice == 8:
        # Sine wave
        f = rng.uniform(1.0, 5.0)
        a = 1.0 + 0.4 * np.sin(2 * np.pi * f * k / n)
    elif choice == 9:
        # Block structure
        nblocks = rng.integers(3, 10)
        a = np.ones(n)
        for b in range(nblocks):
            start = b * n // nblocks
            end = (b + 1) * n // nblocks
            a[start:end] = rng.uniform(0.3, 2.0)
    elif choice == 10:
        # Difference-set inspired: use modular arithmetic
        v = n  # approximate
        a = np.zeros(n)
        # Simple difference set heuristic
        step = max(1, int(np.sqrt(n)))
        for m in range(0, n, step):
            a[m % n] += 1.0
            if m + step < n:
                a[m + step] += 0.5
        a += 0.1
    else:
        a = 1.0 + 0.3 * rng.random(n)

    a = np.maximum(a, 0.05)
    s = np.sum(a)
    return a / s


def resample_sequence(a, new_n):
    """Resample sequence to new length via interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    s = np.sum(a_new)
    if s > 1e-10:
        a_new = a_new / s
    return a_new


def try_parent():
    """Try to load parent sequence."""
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64)
        parent = np.clip(parent, 0, 1000.0)
        if parent.size > 0 and np.isfinite(parent).all():
            obj = compute_obj(parent)
            print(f"Parent (n={parent.size}): obj={obj:.8f}")
            return parent.copy(), obj
    except Exception:
        pass
    return None, np.inf


def polish_sequence(a, time_budget, verbose=False):
    """Intensive multi-method polishing of a single sequence."""
    n = len(a)
    a = a.copy().astype(np.float64)
    start = time.monotonic()

    best_a = a.copy()
    best_obj = compute_obj(a)
    round_num = 0

    while time.monotonic() - start < time_budget - 2:
        round_num += 1
        remaining = time_budget - (time.monotonic() - start)
        if remaining < 4:
            break

        # Coordinate transfer search (new!)
        if remaining > 8:
            a_new = coordinate_transfer_search(best_a, max_iters=int(remaining * 15),
                                               time_budget=min(remaining * 0.25, 15.0))
            obj = compute_obj(a_new)
            if obj < best_obj:
                best_obj = obj
                best_a = a_new.copy()
                if verbose:
                    print(f"    Polish r{round_num} CoordTransfer: {best_obj:.8f}")

        # L-BFGS-B with small tau
        remaining = time_budget - (time.monotonic() - start)
        if remaining > 10:
            tau_small = max(0.05 / n, 0.01)
            a_new, obj_new = optimize_lbfgs(best_a, [0.3 / n, tau_small, tau_small * 0.5],
                                            max_iter=150, time_budget=min(remaining * 0.2, 12.0))
            if obj_new < best_obj:
                best_obj = obj_new
                best_a = a_new.copy()
                if verbose:
                    print(f"    Polish r{round_num} L-BFGS: {best_obj:.8f}")

        # LP step
        remaining = time_budget - (time.monotonic() - start)
        if n <= 800 and remaining > 15:
            a_new = alternating_lp(best_a, max_iters=25, time_budget=min(remaining * 0.25, 20.0))
            obj = compute_obj(a_new)
            if obj < best_obj:
                best_obj = obj
                best_a = a_new.copy()
                if verbose:
                    print(f"    Polish r{round_num} LP: {best_obj:.8f}")

        # Subgradient
        remaining = time_budget - (time.monotonic() - start)
        if remaining > 8:
            a_new = subgradient_optimize(best_a, max_iters=int(remaining * 8),
                                         time_budget=min(remaining * 0.15, 12.0),
                                         tau_start=10.0, tau_end=40.0)
            obj = compute_obj(a_new)
            if obj < best_obj:
                best_obj = obj
                best_a = a_new.copy()
                if verbose:
                    print(f"    Polish r{round_num} SG: {best_obj:.8f}")

        # Pairwise transfer
        remaining = time_budget - (time.monotonic() - start)
        if remaining > 6:
            a_new = pairwise_transfer(best_a, max_iters=200, time_budget=min(remaining * 0.15, 10.0))
            obj = compute_obj(a_new)
            if obj < best_obj:
                best_obj = obj
                best_a = a_new.copy()
                if verbose:
                    print(f"    Polish r{round_num} Pairwise: {best_obj:.8f}")

    return best_a, best_obj


def propose_candidate(seed=42, budget_s=1500):
    """
    Main search function. Explores multiple sequence lengths and initialization
    strategies, using soft-max L-BFGS-B, LP alternating, coordinate transfer
    search, and subgradient methods.
    """
    start_time = time.monotonic()
    deadline = start_time + float(budget_s) - 30.0

    best_seq = None
    best_score = np.inf

    # Load parent
    parent, parent_obj = try_parent()
    if parent is not None and parent_obj < best_score:
        best_score = parent_obj
        best_seq = parent.copy()

    # N values to explore
    n_values = [500, 550, 600, 650, 700, 800]
    if best_seq is not None:
        np_ = len(best_seq)
        extra = [np_, max(400, np_ - 50), min(1000, np_ + 50), min(1000, np_ + 100)]
        n_values = sorted(set(n_values + extra))

    print(f"Exploring n values: {n_values}")
    print(f"Initial best: {best_score:.8f}")

    candidates = []

    # Phase 1: Exploration
    for n in n_values:
        if time.monotonic() > deadline - 120:
            break
        remaining = deadline - time.monotonic()
        if remaining < 15:
            break

        alloc = min(60.0, remaining * 0.15)
        alloc = max(15.0, alloc)

        print(f"\nN={n}: {remaining:.0f}s remaining, alloc={alloc:.0f}s")

        n_inits = max(2, int(alloc / 20))
        n_inits = min(n_inits, 5)

        for init_idx in range(n_inits):
            if time.monotonic() > deadline - 120:
                break
            remaining_init = deadline - time.monotonic()
            if remaining_init < 12:
                break

            # Choose initialization
            if init_idx == 0 and best_seq is not None:
                a = resample_sequence(best_seq, n)
            else:
                a = random_init(n, seed + n * 10000 + init_idx * 79193)

            # Quick subgradient shaping
            sg_time = min(5.0, (alloc / n_inits) * 0.15)
            a = subgradient_optimize(a, max_iters=int(sg_time * 30),
                                      time_budget=sg_time,
                                      tau_start=5.0, tau_end=25.0)

            # L-BFGS-B with tau annealing
            remaining_init = deadline - time.monotonic()
            if remaining_init > 15:
                lfbs_time = min((alloc / n_inits) * 0.4, 25.0)
                tau_list = [8.0/n, 4.0/n, 2.0/n, 1.0/n, 0.5/n, 0.2/n, 0.1/n, 0.05/n]
                a, obj = optimize_lbfgs(a, tau_list, max_iter=200, time_budget=lfbs_time)
            else:
                obj = compute_obj(a)

            # Coordinate transfer (new component)
            remaining_init = deadline - time.monotonic()
            if remaining_init > 10:
                a = coordinate_transfer_search(a, max_iters=200,
                                               time_budget=min(remaining_init * 0.15, 10.0))
                obj = compute_obj(a)

            # LP
            remaining_init = deadline - time.monotonic()
            if remaining_init > 15 and n <= 800:
                a = alternating_lp(a, max_iters=30, time_budget=min(remaining_init * 0.3, 25.0))
                obj = compute_obj(a)

            # Subgradient polish
            remaining_init = deadline - time.monotonic()
            if remaining_init > 8:
                a = subgradient_optimize(a, max_iters=int(remaining_init * 10),
                                          time_budget=min(remaining_init * 0.12, 8.0),
                                          tau_start=15.0, tau_end=50.0)
                obj = compute_obj(a)

            print(f"  init{init_idx}: n={n}, obj={obj:.8f}")

            if obj < best_score:
                best_score = obj
                best_seq = a.copy()
                print(f"  ** UPDATE: n={n}, obj={best_score:.8f} **")

            candidates.append((a.copy(), obj, n))

    # Phase 2: Polishing top candidates
    print(f"\n=== POLISHING PHASE ===")
    print(f"Best before polish: {best_score:.8f}")

    candidates.sort(key=lambda x: x[1])
    n_polish = min(4, len(candidates))

    for ci in range(n_polish):
        if time.monotonic() > deadline - 50:
            break
        remaining = deadline - time.monotonic()
        if remaining < 20:
            break

        a_cand = candidates[ci][0]
        n_cand = candidates[ci][2]
        print(f"  Polishing candidate {ci} (n={n_cand}, obj={candidates[ci][1]:.8f}), {remaining:.0f}s left")

        polish_budget = min(remaining * 0.35, 60.0)
        a_pol, obj_pol = polish_sequence(a_cand, time_budget=polish_budget, verbose=True)

        if obj_pol < best_score:
            best_score = obj_pol
            best_seq = a_pol.copy()
            print(f"  ** UPDATE after polish: obj={best_score:.8f} **")

        # Perturbation + re-optimize
        remaining = deadline - time.monotonic()
        if remaining > 25:
            rng = np.random.default_rng(ci * 555 + 999)
            for ptry in range(3):
                if time.monotonic() > deadline - 30:
                    break
                scale = rng.uniform(0.02, 0.08)
                a_pert = a_cand * np.exp(rng.normal(0, scale, n_cand))
                a_pert = np.maximum(a_pert, 0)
                s = np.sum(a_pert)
                if s < 1e-10:
                    continue
                a_pert = a_pert / s

                # Quick optimization
                a_pert, obj_pert = optimize_lbfgs(a_pert, [2.0/n_cand, 0.5/n_cand, 0.1/n_cand],
                                                   max_iter=100, time_budget=min(remaining * 0.1, 10.0))
                a_pert = coordinate_transfer_search(a_pert, max_iters=100, time_budget=5.0)
                obj_pert = compute_obj(a_pert)

                if obj_pert < best_score:
                    best_score = obj_pert
                    best_seq = a_pert.copy()
                    print(f"  ** UPDATE (perturb {ptry}): obj={best_score:.8f} **")

    # Phase 3: Final intensive polish
    if best_seq is not None and time.monotonic() < deadline - 15:
        remaining = deadline - time.monotonic()
        n_best = len(best_seq)
        print(f"\n=== FINAL POLISH: n={n_best}, {remaining:.0f}s ===")

        a_cur = best_seq.copy()
        for final_round in range(5):
            if time.monotonic() > deadline - 10:
                break
            remaining = deadline - time.monotonic()
            if remaining < 10:
                break

            # L-BFGS-B with tiny tau
            tau_tiny = max(0.02 / n_best, 0.005)
            a_new, obj_new = optimize_lbfgs(a_cur, [0.2/n_best, tau_tiny, tau_tiny*0.5],
                                             max_iter=300, time_budget=min(remaining * 0.2, 15.0))
            if obj_new < best_score:
                best_score = obj_new
                best_seq = a_new.copy()
                a_cur = a_new.copy()
                print(f"  Final L-BFGS {final_round}: {best_score:.8f}")

            # Coordinate transfer
            remaining = deadline - time.monotonic()
            if remaining > 8:
                a_new = coordinate_transfer_search(a_cur, max_iters=300,
                                                    time_budget=min(remaining * 0.2, 12.0))
                obj_new = compute_obj(a_new)
                if obj_new < best_score:
                    best_score = obj_new
                    best_seq = a_new.copy()
                    a_cur = a_new.copy()
                    print(f"  Final CoordTransfer {final_round}: {best_score:.8f}")

            # LP
            remaining = deadline - time.monotonic()
            if remaining > 15 and n_best <= 800:
                a_new = alternating_lp(a_cur, max_iters=25, time_budget=min(remaining * 0.25, 20.0))
                obj_new = compute_obj(a_new)
                if obj_new < best_score:
                    best_score = obj_new
                    best_seq = a_new.copy()
                    a_cur = a_new.copy()
                    print(f"  Final LP {final_round}: {best_score:.8f}")

            # Subgradient
            remaining = deadline - time.monotonic()
            if remaining > 8:
                a_new = subgradient_optimize(a_cur, max_iters=int(remaining * 8),
                                              time_budget=min(remaining * 0.15, 10.0),
                                              tau_start=20.0, tau_end=60.0)
                obj_new = compute_obj(a_new)
                if obj_new < best_score:
                    best_score = obj_new
                    best_seq = a_new.copy()
                    a_cur = a_new.copy()
                    print(f"  Final SG {final_round}: {best_score:.8f}")

            # Pairwise transfer
            remaining = deadline - time.monotonic()
            if remaining > 5:
                a_new = pairwise_transfer(a_cur, max_iters=200, time_budget=min(remaining * 0.15, 8.0))
                obj_new = compute_obj(a_new)
                if obj_new < best_score:
                    best_score = obj_new
                    best_seq = a_new.copy()
                    a_cur = a_new.copy()
                    print(f"  Final Pairwise {final_round}: {best_score:.8f}")

    # Final validation
    if best_seq is None:
        best_seq = np.ones(600)
        best_score = compute_obj(best_seq)

    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    val = evaluate_sequence(result)
    if val < np.inf:
        best_score = val
    else:
        result = [1.0] * 100
        best_score = 2.0

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_score:.8f}, time={elapsed:.1f}s ===")

    return result
