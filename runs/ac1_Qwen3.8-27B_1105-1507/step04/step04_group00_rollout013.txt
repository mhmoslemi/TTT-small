    for it in range(200000):
        if time.monotonic() > deadline:
            break

        b = fast_conv(a)
        M = float(b.max())
        tol = max(1e-12 * M, 1e-15)
        active = np.where(b >= M - tol)[0]

        grad = np.zeros(n)
        for k in active:
            lo = max(0, k - n + 1)
            hi = min(n - 1, k)
            seg = a[k - hi:k - lo + 1][::-1]
            grad[lo:hi + 1] += 2.0 * seg

        lr = 0.05 / (1 + it * 0.01) ** 0.6
        a_new = a - lr * grad
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s < 1e-10:
            continue
        a_new /= s
        a = a_new
        running_sum += a
        count += 1

        if M < best_M - 1e-15:
            best_M = M
            best_a = a.copy()

    if count > 20:
        a_avg = running_sum / count
        a_avg = np.maximum(a_avg, 0.0)
        s = a_avg.sum()
        if s > 1e-10:
            a_avg /= s
            M_avg = fast_conv(a_avg).max()
            if M_avg < best_M:
                best_a = a_avg

    return best_a


def smooth_gradient_step(a, tau, deadline_check):
    """One smooth gradient step with log-sum-exp approximation."""
    n = len(a)
    b = fast_conv(a)
    b_max = b.max()
    exp_b = np.exp(tau * (b - b_max))
    p = exp_b / exp_b.sum()

    grad = np.zeros(n)
    for k in range(2 * n - 1):
        pk = p[k]
        if pk < 1e-14:
            continue
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        grad[lo:hi + 1] += 2.0 * pk * a[k - hi:k - lo + 1][::-1]

    return grad, b


def quick_smooth_optimize(a, max_iters, deadline, tau_init=3.0, tau_final=60.0):
    """Quick smooth gradient descent for exploration."""
    a = np.maximum(a.astype(np.float64), 1e-8)
    a = a / a.sum()
    n = len(a)
    best_a = a.copy()
    best_M = fast_conv(a).max()

    v = np.zeros(n)
    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        frac = it / max(1, max_iters)
        tau = tau_init + (tau_final - tau_init) * frac
        lr = 0.2 * (0.1 ** frac)

        grad, b = smooth_gradient_step(a, tau, deadline)
        M = b.max()
        if M < best_M - 1e-14:
            best_M = M
            best_a = a.copy()

        v = 0.9 * v + lr * grad
        a_new = np.maximum(a - v, 0.0)
        s = a_new.sum()
        if s > 1e-10:
            a = a_new / s

    return best_a


def resample_seq(src, dest_n):
    """Resample 1D sequence to new length via linear interpolation."""
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)


def propose_candidate(budget_s=1500, **kwargs):
    """
    Multi-phase LP-based optimization:
    Phase 1: Quick smooth gradient exploration across lengths (~4 min)
    Phase 2: Aggressive LP alternating minimization (~70% of time)
    Phase 3: Subgradient polishing + final LP passes (~remaining time)
    """
    t_start = time.monotonic()
    deadline = t_start + budget_s - 15.0
    rng = np.random.default_rng(42)

    best = None
    best_score = float('inf')

    def try_update(a):
        nonlocal best, best_score
        a = np.clip(np.asarray(a, dtype=np.float64), 0, 1000)
        if a.size == 0 or a.sum() < 0.01:
            return
        score = float(evaluate_sequence(a.tolist()))
        if np.isfinite(score) and score < best_score:
            best = a.copy()
            best_score = score
            elapsed = time.monotonic() - t_start
            print(f"  ** NEW BEST: J={score:.8f} n={a.size} t={elapsed:.0f}s", flush=True)

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        if p.ndim == 1 and p.size > 0 and np.all(np.isfinite(p)):
            parent = np.maximum(p, 0)
            try_update(parent)
            print(f"Parent loaded: n={parent.size}, J={best_score:.8f}", flush=True)
    except (NameError, AttributeError):
        pass

    if parent is None:
        parent = np.ones(512)
        try_update(parent)

    # === PHASE 1: Quick Exploration ===
    p1_deadline = t_start + min(240, budget_s * 0.15)
    print(f"\n{'='*60}", flush=True)
    print(f"PHASE 1: Quick Exploration (budget={p1_deadline - t_start:.0f}s)", flush=True)
    print(f"{'='*60}", flush=True)

    lengths = [384, 512, 640]
    for n in lengths:
        if time.monotonic() > p1_deadline - 20:
            break
        # Initialize from parent if possible
        if parent is not None:
            a0 = resample_seq(parent, n)
        else:
            a0 = np.ones(n)
        a0 = np.maximum(a0, 0.01)
        a0 /= a0.sum()

        # Quick smooth gradient
        a1 = quick_smooth_optimize(a0, 300, p1_deadline - 20, tau_init=3, tau_final=50)
        try_update(a1)

        # Also try with perturbation
        a2 = np.maximum(a0 + rng.normal(0, 0.05, n), 0.01)
        a2 /= a2.sum()
        a2 = quick_smooth_optimize(a2, 200, p1_deadline - 20, tau_init=5, tau_final=80)
        try_update(a2)

    print(f"Phase 1 done: best J={best_score:.8f}, n={best.size if best is not None else 0}", flush=True)

    # === PHASE 2: LP Optimization (main workhorse) ===
    p2_deadline = t_start + budget_s * 0.78
    print(f"\n{'='*60}", flush=True)
    print(f"PHASE 2: LP Optimization (budget={p2_deadline - t_start:.0f}s)", flush=True)
    print(f"{'='*60}", flush=True)

    if best is not None:
        n_best = best.size
        print(f"  Starting LP on n={n_best}, J={best_score:.8f}", flush=True)

        # Main LP run
        a = lp_optimize(best, p2_deadline, verbose=True)
        try_update(a)

        # Second LP pass if time remains and we improved
        if time.monotonic() < deadline - 90:
            print(f"  Second LP pass... J={best_score:.8f}", flush=True)
            a = lp_optimize(a, deadline - 90, verbose=True)
            try_update(a)

    # === PHASE 3: Polishing ===
    p3_deadline = deadline
    print(f"\n{'='*60}", flush=True)
    print(f"PHASE 3: Polishing (deadline in {p3_deadline - time.monotonic():.0f}s)", flush=True)
    print(f"{'='*60}", flush=True)

    if best is not None and time.monotonic() < p3_deadline - 5:
        n_best = best.size

        # Subgradient polish
        polish_deadline = min(time.monotonic() + 60, p3_deadline - 30)
        if time.monotonic() < polish_deadline:
            a_sub = subgradient_polish(best, polish_deadline)
            try_update(a_sub)

        # Final LP pass
        if time.monotonic() < p3_deadline - 5:
            n_cur = best.size
            a_final = lp_optimize(best, p3_deadline - 2, verbose=False)
            try_update(a_final)

    # === Final: try a few different lengths ===
    if best is not None and time.monotonic() < deadline - 5:
        for mult in [0.75, 1.25]:
            if time.monotonic() > deadline - 5:
                break
            new_n = int(best.size * mult)
            if new_n < 64 or new_n > 2048:
                continue
            a_ext = resample_seq(best, new_n)
            a_ext = np.maximum(a_ext, 0.01)
            a_ext /= a_ext.sum()
            a_ext = lp_optimize(a_ext, min(time.monotonic() + 30, deadline - 3), verbose=False)
            try_update(a_ext)

    # Fallback
    if best is None:
        best = np.ones(512)
        best_score = compute_J(best)

    # Final report
    final_score = float(evaluate_sequence([float(x) for x in best]))
    elapsed = time.monotonic() - t_start
    print(f"\n{'='*60}", flush=True)
    print(f"FINAL: J={final_score:.8f} n={best.size} time={elapsed:.0f}s", flush=True)
    print(f"{'='*60}", flush=True)

    return [float(x) for x in best]
