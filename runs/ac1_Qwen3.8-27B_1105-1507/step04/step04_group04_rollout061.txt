```python
"""
Step function optimization using alternating LP, coordinate descent,
and smoothed gradient descent with adaptive parameters.
"""

import numpy as np
from scipy.optimize import linprog
import time

# Global state
global_best = None
global_best_obj = np.inf

def compute_conv(a):
    n = len(a)
    size = 1
    while size < 2*n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    conv = np.fft.irfft(fa * fa, size)
    return conv[:2*n-1]

def compute_obj(a):
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    conv = compute_conv(a)
    return 2.0 * n * float(np.max(conv)) / (s * s)

def build_toeplitz(a):
    n = len(a)
    A = np.zeros((2*n-1, n))
    for j in range(n):
        start = j
        end = min(j+n, 2*n-1)
        A[start:end, j] = a[:end-start]
    return A

def solve_lp_maxsum(a, M, time_limit=5):
    n = len(a)
    A = build_toeplitz(a)
    c = -np.ones(n)
    b_ub = np.full(2*n-1, M)
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': time_limit})
        if res.status == 0:
            g = np.maximum(res.x, 0)
            return g, float(np.sum(g))
    except Exception:
        pass
    return None, None

def solve_lp_minpeak(a, time_limit=5):
    n = len(a)
    A = build_toeplitz(a)
    # min M s.t. A@g <= M, sum(g)=1, g>=0
    # Variables: g[0..n-1], M
    c = np.zeros(n+1)
    c[-1] = 1.0
    A_ub = np.zeros((2*n-1, n+1))
    A_ub[:, :n] = A
    A_ub[:, n] = -1.0
    b_ub = np.zeros(2*n-1)
    A_eq = np.zeros((1, n+1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])
    bounds = [(0.0, None)] * n + [(0.0, None)]
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': time_limit})
        if res.status == 0:
            g = np.maximum(res.x[:n], 0)
            s = np.sum(g)
            if s < 1e-10:
                return None, None
            g = g / s
            return g, float(res.x[n])
    except Exception:
        pass
    return None, None

def alternating_lp(a, time_budget, verbose=False):
    n = len(a)
    a = a.copy().astype(np.float64)
    s = np.sum(a)
    if s < 1e-10:
        return a
    a = a / s
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    stall = 0
    
    while time.monotonic() - start < time_budget - 2:
        conv = compute_conv(a)
        M = float(np.max(conv))
        
        g, sum_g = solve_lp_maxsum(a, M, time_limit=3)
        if g is None or sum_g < 1e-10:
            break
        g = g / sum_g
        
        # Line search
        best_t = 0
        best_obj_ls = best_obj
        for t in [0.1, 0.2, 0.3, 0.5, 0.7, 1.0]:
            a_new = (1-t)*a + t*g
            a_new = np.maximum(a_new, 0)
            s_new = np.sum(a_new)
            if s_new < 1e-10:
                continue
            a_new = a_new / s_new
            obj_new = compute_obj(a_new)
            if obj_new < best_obj_ls:
                best_obj_ls = obj_new
                best_t = t
        
        if best_t > 0 and best_obj_ls < best_obj - 1e-14:
            a_new = (1-best_t)*a + best_t*g
            a_new = np.maximum(a_new, 0)
            a_new = a_new / np.sum(a_new)
            a = a_new
            best_obj = best_obj_ls
            best_a = a.copy()
            stall = 0
        else:
            stall += 1
            if stall > 30:
                # Try noise
                rng = np.random.default_rng(int(time.monotonic()*1000))
                a = a * np.exp(rng.normal(0, 0.05, n))
                a = np.maximum(a, 1e-10)
                a = a / np.sum(a)
                stall = 0
                if time.monotonic() - start > time_budget * 0.8:
                    break
    
    return best_a

def coordinate_descent(a, time_budget, verbose=False):
    n = len(a)
    a = a.copy().astype(np.float64)
    s = np.sum(a)
    if s < 1e-10:
        return a
    a = a / s
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    sweep = 0
    no_improve = 0
    
    while time.monotonic() - start < time_budget - 2:
        conv_full = compute_conv(a)
        
        for j in range(n):
            aj = a[j]
            if aj < 1e-15:
                continue
            
            # Compute conv0 (conv with a[j]=0)
            conv0 = conv_full.copy()
            start_k = max(0, j)
            end_k = min(2*n-1, j+n)
            if end_k > start_k:
                conv0[start_k:end_k] -= 2*aj*a[:end_k-start_k]
            if 2*j < 2*n-1:
                conv0[2*j] += aj*aj  # fix: we subtracted 2*aj^2, should subtract aj^2
        
        # Compute slopes
        slopes = np.zeros(2*n-1)
        if end_k > start_k:
            slopes[start_k:end_k] = 2*a[:end_k-start_k]
        if 2*j < 2*n-1:
            slopes[2*j] = 0  # quadratic handled separately
        
        # Ternary search
        lo = 0.0
        hi = max(2*aj + 0.01, 0.01)
        for _ in range(40):
            m1 = lo + (hi-lo)/3
            m2 = hi - (hi-lo)/3
            f1 = np.max(conv0 + slopes*m1)
            q1 = conv0[2*j] + m1*m1 if 2*j < 2*n-1 else -np.inf
            f1 = max(f1, q1)
            f2 = np.max(conv0 + slopes*m2)
            q2 = conv0[2*j] + m2*m2 if 2*j < 2*n-1 else -np.inf
            f2 = max(f2, q2)
            if f1 < f2:
                hi = m2
            else:
                lo = m1
        
        a[j] = max(0.0, (lo+hi)/2)
        
        # Update conv_full incrementally? Or just recompute at end of sweep.
        
        # Renormalize periodically
        if (j+1) % 50 == 0:
            s = np.sum(a)
            if s > 1e-10:
                a = a / s
        
        s = np.sum(a)
        if s > 1e-10:
            a = a / s
        conv_full = compute_conv(a)  # update for next j
        
        sweep += 1
        obj = compute_obj(a)
        if obj < best_obj:
            best_obj = obj
            best_a = a.copy()
            no_improve = 0
        else:

def smoothed_grad_descent(a, time_budget, tau_start=20, tau_end=300):
    n = len(a)
    a = a.copy().astype(np.float64)
    s = np.sum(a)
    if s < 1e-10:
        return a
    a = a / s
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    v = np.zeros(n)
    
    while time.monotonic() - start < time_budget - 2:
        frac = (time.monotonic() - start) / time_budget
        tau = tau_start * (tau_end/tau_start)**min(frac, 1.0)
        
        conv = compute_conv(a)
        shifted = tau * (conv - np.max(conv))
        p = np.exp(shifted)
        p_sum = np.sum(p)
        if p_sum < 1e-300:
            break
        p = p / p_sum
        
        # Gradient: 2 * correlate(p, a, 'full')[n-1:n-1+n]
        corr = np.correlate(p, a, mode='full')
        grad = 2.0 * corr[n-1:n-1+n]
        
        # Momentum
        v = 0.9 * v + 0.1 * grad
        v_norm = np.linalg.norm(v)
        if v_norm < 1e-15:
            break
        v = v / v_norm
        
        # Step with backtracking
        step = 0.05
        obj_curr = compute_obj(a)
        for _ in range(15):
            a_new = a - step * v
            a_new = np.maximum(a_new, 0)
            s_new = np.sum(a_new)
            if s_new < 1e-10:
                step *= 0.5
                continue
            a_new = a_new / s_new
            obj_new = compute_obj(a_new)
            if obj_new < obj_curr:
                a = a_new
                break
            step *= 0.5
        
        obj = compute_obj(a)
        if obj < best_obj:
            best_obj = obj
            best_a = a.copy()
    
    return best_a

def random_init(n, seed):

def resample(a, new_n):
    n = len(a)
    if n == new_n:
        return a.copy()
    old_k = np.arange(n)
    new_k = np.linspace(0, n-1, new_n)
    a_new = np.interp(new_k, old_k, a)
    a_new = np.maximum(a_new, 0)
    s = np.sum(a_new)
