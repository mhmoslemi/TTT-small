def build_conv_matrix(a):
    """Build the constraint matrix for the LP: A * g <= b where (a*g)_k <= b_k."""
    n = len(a)
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        jl = max(0, k - n + 1)
        jh = min(n - 1, k)
        if jl <= jh:
            cnt = jh - jl + 1
            rows.extend([k] * cnt)
            cols.extend(range(jl, jh + 1))
            vals.extend(a[jl:jh+1].tolist())
    from scipy.sparse import csr_matrix
    A = csr_matrix((vals, (rows, cols)), shape=(2*n-1, n))
    return A

def lp_best_response(a, M):

def resample_seq(src, dest_n):
    """Resample 1D sequence to new length."""
    src = np.asarray(src, dtype=np.float64)

    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)

"""
Minimizes J(a) = 2*n*max(convolve(a,a))/(sum(a))^2 using:
1. LP mirror descent with adaptive tightening for initial optimization
2. Smoothed log-sum-exp gradient with FFT-accelerated cross-correlation
3. Progressive refinement with increasing smoothing parameter (L-BFGS-B)
4. Multi-length exploration and perturbation restarts
Key innovation: FFT-based gradient computation for the smoothed max, enabling
efficient quasi-Newton optimization in the critical final phase.
"""

import numpy as np
from scipy.optimize import linprog, minimize
from scipy.sparse import csr_matrix
import time


def fast_conv(a):
    """Full self-convolution via FFT."""
    n = len(a)
    s = 1
    while s < 2 * n - 1:
        s <<= 1
    fa = np.fft.rfft(a, s)
    return np.fft.irfft(fa * fa, s)[:2 * n - 1]


def compute_J(a):
    """Compute J(a) = 2*n*max(conv(a,a))/(sum(a))^2."""
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))


def smoothed_obj_grad(a, alpha):
    """Compute smoothed J and gradient using log-sum-exp.
    
    Uses FFT for the cross-correlation in the gradient computation.
    Returns (J_smoothed, gradient).
    """
    n = len(a)
    b = fast_conv(a)
    S = a.sum()
    if S < 0.01:
        return 1e10, np.zeros(n)
    
    b_max = b.max()
    alpha_b = alpha * (b - b_max)
    alpha_b = np.clip(alpha_b, -500, 0)
    exp_vals = np.exp(alpha_b)
    sum_exp = np.sum(exp_vals)
    if sum_exp < 1e-300:
        return 1e10, np.zeros(n)
    log_sum = np.log(sum_exp)
    LSE = b_max + log_sum / alpha
    w = exp_vals / sum_exp
    
    # Cross-correlation: corr[i] = sum_j w[i+j] * a[j]
    # This equals (w * a) cross-correlation, computable via FFT
    size = 1
    while size < 3 * n:
        size <<= 1
    W = np.fft.rfft(w, size)
    A = np.fft.rfft(a, size)
    corr = np.fft.irfft(W * np.conj(A), size)[:n]
    
    grad_LSE = 2.0 * corr
    J = 2.0 * n * LSE / (S * S)
    grad_J = 2.0 * n * (grad_LSE * S - 2.0 * LSE * a) / (S ** 3)
    
    return J, grad_J


def build_conv_matrix(a):
    """Build sparse constraint matrix for LP: (a*g)_k for all k."""
    n = len(a)
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        jl = max(0, k - n + 1)
        jh = min(n - 1, k)
        if jl <= jh:
            cnt = jh - jl + 1
            rows.extend([k] * cnt)
            cols.extend(range(jl, jh + 1))
            vals.extend(a[jl:jh + 1].tolist())
    A = csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))
    return A


def lp_best_response(a, M):
    """LP: max sum(g) s.t. (a*g)_k <= M for all k, g >= 0."""
    n = len(a)
    c = -np.ones(n)
    A = build_conv_matrix(a)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
    if r.success and r.x is not None:
        g = np.maximum(r.x, 0.0)
        if g.sum() > 1e-10:
            return g
    return None


def run_lp_phase(a, time_budget, tracker):
    """Run LP mirror descent for the given time budget."""
    a = np.maximum(a.astype(np.float64), 0.001)
    a /= a.sum()
    n = len(a)
    current_J = compute_J(a)
    tracker.update(a)
    
    deadline = time.monotonic() + time_budget
    stall = 0
    last_improvement = time.monotonic()
    
    while time.monotonic() < deadline:
        b = fast_conv(a)
        M = float(b.max())
        
        g = None
        for frac in [0.004, 0.002, 0.001, 0.0005, 0.0002]:
            g = lp_best_response(a, M * (1.0 - frac))
            if g is not None:
                break
        
        if g is None:
            stall += 1
            if stall > 40 or (time.monotonic() - last_improvement) > 60:
                break
            continue
        
        g = np.maximum(g, 0.0)
        g_sum = g.sum()
        if g_sum < 1e-10:
            break
        g /= g_sum
        
        # Adaptive line search
        best_t = 0.0
        best_J = current_J
        for t in [0.15, 0.3, 0.5, 0.7, 0.85, 1.0]:
            a_trial = (1.0 - t) * a + t * g
            a_trial = np.maximum(a_trial, 0.0)
            s = a_trial.sum()
            if s < 0.01:
                continue
            a_trial /= s
            J_trial = compute_J(a_trial)
            if J_trial < best_J - 1e-14:
                best_J = J_trial
                best_t = t
        
        if best_t > 0:
            a = (1.0 - best_t) * a + best_t * g
            a = np.maximum(a, 0.0)
            a /= a.sum()
            current_J = best_J
            stall = 0
            last_improvement = time.monotonic()
            tracker.update(a)
        else:
            stall += 1
            if stall > 50 or (time.monotonic() - last_improvement) > 90:
                break
    
    tracker.update(a)
    return a


def obj_for_lbfgsb(x, alpha):
    """Objective wrapper for L-BFGS-B."""
    x = np.maximum(x, 0.0)
    if x.sum() < 0.01:
        return 1e10
    val, _ = smoothed_obj_grad(x, alpha)
    return val


def grad_for_lbfgsb(x, alpha):
    """Gradient wrapper for L-BFGS-B."""
    x = np.maximum(x, 0.0)
    if x.sum() < 0.01:
        return np.zeros(len(x))
    _, g = smoothed_obj_grad(x, alpha)
    return g


def run_smoothed_phase(a, time_budget, tracker):
    """Run smoothed gradient descent with progressive alpha."""
    a = np.maximum(a.astype(np.float64), 0.001)
    a /= a.sum()
    n = len(a)
    current_J = compute_J(a)
    tracker.update(a)
    
    deadline = time.monotonic() + time_budget
    
    alphas = [50, 100, 200, 400, 800, 1600, 3200, 6400]
    
    for alpha in alphas:
        if time.monotonic() > deadline - 5:
            break
        
        t0 = time.monotonic()
        
        # Determine max iterations based on remaining time
        max_iter = 300
        max_fun = 800
        
        result = minimize(
            obj_for_lbfgsb, a, method='L-BFGS-B',
            jac=lambda x: grad_for_lbfgsb(x, alpha),
            args=(alpha,),
            bounds=[(0.0, None)] * n,
            options={'maxiter': max_iter, 'maxfun': max_fun,
                     'ftol': 1e-15, 'gtol': 1e-10, 'gtol_rel': 1e-10}
        )
        
        a_new = np.maximum(result.x, 0.0)
        if a_new.sum() > 0.01:
            a_new /= a_new.sum()
            J_new = compute_J(a_new)
            if J_new < current_J - 1e-14:
                current_J = J_new
                a = a_new
                tracker.update(a)
            print(f"    alpha={alpha}: true_J={compute_J(a_new):.8f} smoothed={result.fun:.8f} "
                  f"iters={result.nit} time={time.monotonic()-t0:.1f}s", flush=True)
        else:
            print(f"    alpha={alpha}: invalid, skipping", flush=True)
        
        # Check time
        elapsed = time.monotonic() - t0
        if elapsed > time_budget * 0.8 / len(alphas):
            break
    
    tracker.update(a)
    return a


def resample_seq(src, dest_n):
    """Resample 1D sequence to new length via linear interpolation."""
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)
