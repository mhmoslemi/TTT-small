        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]


def compute_J(a):
    """Compute J(a) = 2*n*max(conv(a,a))/(sum(a))^2."""
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))


def eval_delta_fast(a, b, i, delta, n, S, outside_max):
    """Evaluate J when a[i] += delta, using precomputed outside_max. O(n)."""
    new_S = S + delta
    if new_S < 0.01:
        return float('inf')
    k_lo = max(0, i)
    k_hi = min(2 * n - 2, i + n - 1)
    a_seg = a[k_lo - i:k_hi + 1 - i]
    b_seg = b[k_lo:k_hi + 1] + 2.0 * delta * a_seg
    if 2 * i >= k_lo and 2 * i <= k_hi:
        b_seg = b_seg.copy()
        b_seg[2 * i - k_lo] += delta * delta
    inside_max = float(b_seg.max())
    total_max = max(inside_max, outside_max)
    return 2.0 * n * total_max / (new_S * new_S)


def coordinate_descent(a, deadline, max_sweeps=5000, seed=42, verbose=True):
    """Fast coordinate descent with incremental b updates. O(n) per coordinate."""
    a = np.maximum(a.copy(), 1e-10).astype(np.float64)
    a /= a.sum()
    n = len(a)
    S = float(a.sum())
    b = fast_conv(a)
    best_a = a.copy()
    best_J = 2.0 * n * float(b.max()) / (S * S)
    rng = np.random.default_rng(seed)
    no_improve = 0

    for sweep in range(max_sweeps):
        if time.monotonic() > deadline:
            break
        sweep_best_before = 2.0 * n * float(b.max()) / (S * S)
        order = rng.permutation(n)

        for i in order:
            v = a[i]
            S = float(a.sum())

            # Precompute outside max (independent of delta)
            k_lo = max(0, i)
            k_hi = min(2 * n - 2, i + n - 1)
            outside_max = 0.0
            if k_lo > 0:
                outside_max = max(outside_max, float(b[:k_lo].max()))
            if k_hi < 2 * n - 2:
                outside_max = max(outside_max, float(b[k_hi + 1:].max()))

            # Ternary search for optimal delta
            lo = -v * 0.9999
            hi = max(v * 3.0, 0.5)
            for _ in range(35):
                m1 = lo + (hi - lo) / 3.0
                m2 = hi - (hi - lo) / 3.0
                f1 = eval_delta_fast(a, b, i, m1, n, S, outside_max)
                f2 = eval_delta_fast(a, b, i, m2, n, S, outside_max)
                if f1 < f2:
                    hi = m2
                else:
                    lo = m1
                if hi - lo < 1e-14 * max(1.0, v):
                    break

            delta = (lo + hi) / 2.0
            new_v = v + delta
            if new_v < 0:
                new_v = 0.0
                delta = new_v - v

            if abs(delta) < 1e-14:
                continue

            # Incremental b update (vectorized)
            old_a_i = a[i]
            a_seg = a[k_lo - i:k_hi + 1 - i]
            b[k_lo:k_hi + 1] += 2.0 * delta * a_seg
            if 2 * i >= k_lo and 2 * i <= k_hi:
                b[2 * i] += delta * delta
            a[i] = new_v
            S += delta

        # End of sweep
        new_J = 2.0 * n * float(b.max()) / (S * S)
        if new_J < best_J:
            best_J = new_J
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        if no_improve >= 5:
            break
        if verbose and sweep % 50 == 0:
            print(f"    CD sweep {sweep}: J={best_J:.8f}", flush=True)

    # Final verification
    final_J = compute_J(best_a)
    if final_J < best_J:
        best_J = final_J
    return best_a


def smooth_gradient_step(a, tau, deadline_budget=0.1):
    """One smooth gradient descent step. Returns updated a."""
    n = len(a)
    b = fast_conv(a)
    b_max = b.max()
    exp_b = np.exp(tau * (b - b_max))
    p = exp_b / exp_b.sum()

    grad = np.zeros(n)
    for k in range(2 * n - 1):
        pk = p[k]
        if pk < 1e-12:
            continue
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        grad[lo:hi + 1] += 2.0 * pk * a[k - hi:k - lo + 1][::-1]

    return grad, b


def run_smooth_gradient(a, max_iters, deadline, initial_tau=3.0, final_tau=60.0,
                        lr_init=0.3, lr_min=0.02):
    """Smooth gradient descent with adaptive parameters."""
    a = np.maximum(a.astype(np.float64), 1e-8)
    a = a / a.sum()
    n = len(a)
    best_a = a.copy()
    best_M = fast_conv(a).max()
    v = np.zeros(n)

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        frac = it / max(1, max_iters)
        tau = initial_tau + (final_tau - initial_tau) * (frac ** 0.5)
        lr = lr_init * (lr_min / lr_init) ** frac

        grad, b = smooth_gradient_step(a, tau)
        M = b.max()
        if M < best_M * (1 - 1e-8):
            best_M = M
            best_a = a.copy()

        beta = 0.9
        v = beta * v + lr * grad
        a_new = np.maximum(a - v, 0.0)
        s = a_new.sum()
        if s < 1e-10:
            a_new = a.copy()
        else:
            a_new = a_new / s
        a = a_new

    if compute_J(best_a) < compute_J(a):
        return best_a
    return a


def build_A_sparse(n, a):
    """Build sparse constraint matrix for LP."""
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        j_min = max(0, k - n + 1)
        j_max = min(n - 1, k)
        for j in range(j_min, j_max + 1):
            rows.append(k)
            cols.append(j)
            vals.append(a[k - j])
    A = csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))
    return A


def lp_best_response(a, M, time_limit=30):
    """LP: max sum(g) s.t. (a*g)[k] <= M for all k, g >= 0."""
    n = len(a)
    A = build_A_sparse(n, a)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    try:
        r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                    options={'time_limit': time_limit})
        if r.success and r.x is not None:
            g = np.maximum(r.x, 0)
            if g.sum() > 1e-15:
                return g
    except Exception:
        pass
    return None


def run_lp_mirror_descent(a, max_iters, deadline, verbose=True):
    """LP mirror descent with geometric line search."""
    a = np.maximum(a.astype(np.float64), 1e-8)
    a = a / a.sum()
    best_a = a.copy()
    best_M = fast_conv(a).max()
    best_J = compute_J(a)
    no_improve = 0

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        b = fast_conv(a)
        M = float(b.max())
        g = lp_best_response(a, M, time_limit=20)
        if g is None:
            break
        g = np.maximum(g, 0.0)
        g_sum = g.sum()
        if g_sum < 1e-10:
            break
        g /= g_sum

        improved = False
        base_t = max(0.05, 0.5 * (1 - 0.3 * it / max(1, max_iters)))
        for t in [base_t, base_t * 0.5, base_t * 0.25, 0.02, 0.01]:
            a_trial = np.maximum((1 - t) * a + t * g, 0)
            s = a_trial.sum()
            if s < 1e-10:
                continue
            a_trial /= s
            J_trial = compute_J(a_trial)
            if J_trial < best_J - 1e-14:
                a = a_trial
                best_J = J_trial
                best_a = a.copy()
                improved = True
                if verbose:
                    print(f"    LP iter {it}: J={best_J:.8f} (t={t:.3f})", flush=True)
                break

        if not improved:
            no_improve += 1
            if no_improve > 15:
                break
        else:
            no_improve = 0

    return best_a


def resample_seq(src, dest_n):
    """Resample 1D sequence to new length via linear interpolation."""
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)


def generate_diverse_inits(n, rng, parent=None):
    """Generate diverse initial sequences of length n."""
    inits = []
    t = np.linspace(0, 1, n, endpoint=False)

    if parent is not None:
        p = resample_seq(parent, n)
        inits.append(p)
        inits.append(np.maximum(p + rng.normal(0, 0.05, n), 0.01))
        inits.append(np.maximum(p + rng.normal(0, 0.15, n), 0.01))

    inits.append(np.ones(n))
    inits.append(rng.uniform(0.5, 1.0, n))

    # Cosine/sine structures
    for freq in [1, 2, 3, 5]:
        inits.append(np.maximum(1.0 + 0.3 * np.cos(2 * np.pi * freq * t), 0.01))
        inits.append(np.maximum(1.0 + 0.4 * np.sin(2 * np.pi * freq * t), 0.01))

    # Gaussian bumps
    inits.append(np.exp(-4.0 * (t - 0.5) ** 2) + 0.1)
    inits.append(np.exp(-8.0 * (t - 0.3) ** 2) + np.exp(-8.0 * (t - 0.7) ** 2) + 0.05)

    # Triangle
    inits.append(np.maximum(1.0 - 0.4 * np.abs(2 * t - 1), 0.05))
    inits.append(0.3 + 0.7 * np.abs(2 * t - 1))

    # Chirp
    inits.append(np.maximum(1.0 + 0.3 * np.cos(2 * np.pi * 4 * t ** 2), 0.01))

    # Smooth random
    ksize = max(3, n // 20)
    kernel = np.ones(ksize) / ksize
    smooth = np.convolve(rng.uniform(0, 1, n), kernel, mode='same')
    inits.append(np.maximum(smooth + 0.1, 0.01))

    # Asymmetric
    inits.append(np.maximum(0.3 + 0.7 * t ** 1.5, 0.01))
    inits.append(np.maximum(0.3 + 0.7 * (1 - t) ** 1.5, 0.01))

    return inits


class ScoreTracker:
    def __init__(self):
        self.best = None
        self.best_score = float('inf')
        self.t_start = time.monotonic()

    def update(self, a):
        a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
        if a.size == 0 or a.sum() < 0.01:
            return
        score = float(evaluate_sequence(a.tolist()))
        if np.isfinite(score) and score < self.best_score:
            self.best = a.copy()
            self.best_score = score
            elapsed = time.monotonic() - self.t_start
            print(f"  ** J={score:.8f} n={a.size} t={elapsed:.0f}s", flush=True)

    def report(self, label=""):
        if self.best is not None:
            print(f"  {label} best: J={self.best_score:.8f} n={self.best.size}", flush=True)


def propose_candidate(seed=42, budget_s=1500, **kwargs):
    """
    Multi-phase hybrid search combining coordinate descent, LP mirror descent,
    smooth gradient, and multi-length exploration.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + float(budget_s) - 10.0
    tracker = ScoreTracker()

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        if p.ndim == 1 and p.size > 0 and np.all(np.isfinite(p)):
            parent = np.maximum(p, 0.0)
            tracker.update(parent)
            print(f"Parent loaded: n={parent.size}, J={tracker.best_score:.6f}", flush=True)
    except (NameError, AttributeError):
        pass

    if parent is None:
        parent = np.ones(256)
        tracker.update(parent)

    current_best = parent.copy()

    # === PHASE 1: Coordinate descent on parent (fast, precise) ===
    p1_deadline = min(time.monotonic() + 200, deadline - 300)
    print(f"\n{'='*60}", flush=True)
    print(f"PHASE 1: Coordinate Descent on parent (n={parent.size})", flush=True)
    print(f"{'='*60}", flush=True)

    n_parent = parent.size
    a_cd = coordinate_descent(parent, p1_deadline, max_sweeps=10000,
                              seed=seed, verbose=True)
    tracker.update(a_cd)
    if tracker.best is not None:
        current_best = tracker.best.copy()

    # === PHASE 2: LP mirror descent for structural refinement ===
    p2_deadline = min(time.monotonic() + 250, deadline - 200)
    print(f"\n{'='*60}", flush=True)
    print(f"PHASE 2: LP Mirror Descent (n={current_best.size})", flush=True)
    print(f"{'='*60}", flush=True)

    n_cur = current_best.size
    est_lp_time = 0.005 * (n_cur / 128) ** 2
    max_lp_iters = max(20, min(300, int(250 / max(0.01, est_lp_time))))
    print(f"  n={n_cur}, est_lp_time={est_lp_time*1000:.0f}ms, max_iters={max_lp_iters}", flush=True)

    a_lp = run_lp_mirror_descent(current_best, max_lp_iters, p2_deadline, verbose=True)
    tracker.update(a_lp)
    if tracker.best is not None:
        current_best = tracker.best.copy()

    # === PHASE 3: CD + LP alternation (refinement loop) ===
    p3_deadline = min(time.monotonic() + 200, deadline - 100)
    print(f"\n{'='*60}", flush=True)
    print(f"PHASE 3: CD + LP Alternation", flush=True)
    print(f"{'='*60}", flush=True)

    for round_i in range(5):
        if time.monotonic() > p3_deadline:
            break
        a_cd = coordinate_descent(current_best, min(time.monotonic() + 40, p3_deadline - 50),
                                  max_sweeps=5000, seed=seed + round_i, verbose=False)
        tracker.update(a_cd)
        if tracker.best is not None:
            current_best = tracker.best.copy()

        if time.monotonic() > p3_deadline - 30:
            break
        a_lp = run_lp_mirror_descent(current_best, 30, min(time.monotonic() + 30, p3_deadline - 10),
                                     verbose=False)
        tracker.update(a_lp)
        if tracker.best is not None:
            current_best = tracker.best.copy()

    tracker.report("After Phase 3")

    # === PHASE 4: Multi-length exploration ===
    p4_deadline = min(time.monotonic() + 300, deadline - 50)
    print(f"\n{'='*60}", flush=True)
    print(f"PHASE 4: Multi-length exploration", flush=True)
    print(f"{'='*60}", flush=True)

    base_n = current_best.size
    candidate_lengths = []
    for mult in [0.5, 1.0, 1.5, 2.0, 3.0]:
        ln = int(base_n * mult)
        ln = max(64, min(ln, 2048))
        if ln not in candidate_lengths:
            candidate_lengths.append(ln)
    # Also try some standard lengths
    for ln in [128, 192, 256, 384, 512, 768, 1024, 1536]:
        if ln not in candidate_lengths and 64 <= ln <= 2048:
            candidate_lengths.append(ln)
    candidate_lengths = sorted(candidate_lengths)

    for n_try in candidate_lengths:
        if time.monotonic() > p4_deadline:
            break

        # Resample current best to new length
        a_res = resample_seq(current_best, n_try)
        a_res = np.maximum(a_res, 1e-6)
        a_res = a_res / a_res.sum()

        # Quick smooth gradient to adapt to new length
        a_res = run_smooth_gradient(a_res, 100, min(time.monotonic() + 20, p4_deadline - 10),
                                    initial_tau=3, final_tau=40, lr_init=0.2, lr_min=0.03)

        # Coordinate descent refinement
        cd_deadline = min(time.monotonic() + 60, p4_deadline - 10)
        a_cd = coordinate_descent(a_res, cd_deadline, max_sweeps=5000,
                                  seed=seed + n_try, verbose=False)
        tracker.update(a_cd)

    tracker.report("After Phase 4")
    if tracker.best is not None:
        current_best = tracker.best.copy()

    # === PHASE 5: Random restarts with perturbation ===
    p5_deadline = min(time.monotonic() + 200, deadline - 20)
    print(f"\n{'='*60}", flush=True)
    print(f"PHASE 5: Random restarts + polished CD", flush=True)
    print(f"{'='*60}", flush=True)

    n_best = current_best.size
    temperature = 0.1
    max_restarts = 30

    for r in range(max_restarts):
        if time.monotonic() > p5_deadline:
            break

        # Perturb best
        noise = rng.normal(0, temperature * 0.05, n_best)
        a_pert = current_best + noise
        a_pert = np.maximum(a_pert, 0.001)
        a_pert = a_pert / a_pert.sum()

        # Quick smooth gradient
        a_pert = run_smooth_gradient(a_pert, 30, min(time.monotonic() + 10, p5_deadline - 5),
                                     initial_tau=5, final_tau=40, lr_init=0.1, lr_min=0.02)

        # CD refinement
        cd_dl = min(time.monotonic() + 30, p5_deadline - 5)
        a_cd = coordinate_descent(a_pert, cd_dl, max_sweeps=3000,
                                  seed=seed + 1000 + r, verbose=False)
        tracker.update(a_cd)

        temperature *= 0.93

    tracker.report("After Phase 5")
    if tracker.best is not None:
        current_best = tracker.best.copy()

    # === PHASE 6: Final intensive CD + LP on the best ===
    if time.monotonic() < deadline - 5:
        print(f"\n{'='*60}", flush=True)
        print(f"PHASE 6: Final polish", flush=True)
        print(f"{'='*60}", flush=True)

        a_cd = coordinate_descent(current_best, min(time.monotonic() + 60, deadline - 15),
                                  max_sweeps=10000, seed=seed + 9999, verbose=True)
        tracker.update(a_cd)

        if tracker.best is not None and time.monotonic() < deadline - 5:
            current_best = tracker.best.copy()
            a_lp = run_lp_mirror_descent(current_best, 20, deadline, verbose=True)
            tracker.update(a_lp)

    # === Final: one more CD pass if time remains ===
    if tracker.best is not None and time.monotonic() < deadline - 2:
        a_final = coordinate_descent(tracker.best, deadline, max_sweeps=5000,
                                     seed=seed + 7777, verbose=False)
        tracker.update(a_final)

    # Fallback
    if tracker.best is None:
        tracker.best = np.ones(128)
        tracker.best_score = compute_J(tracker.best)

    print(f"\n{'='*60}", flush=True)
    print(f"FINAL: J={tracker.best_score:.8f} n={tracker.best.size} "
          f"time={time.monotonic()-tracker.t_start:.0f}s", flush=True)
    print(f"{'='*60}", flush=True)

    return [float(x) for x in tracker.best]
