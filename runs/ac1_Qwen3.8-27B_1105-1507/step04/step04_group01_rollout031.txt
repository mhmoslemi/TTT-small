```python
"""
Spectral-Aware Hybrid Optimizer: Combines L-BFGS-B with soft-max parametrization,
LP alternating, and frequency-domain spectral equalization to minimize
2*n*max(conv(a,a))/(sum(a))^2 for non-negative sequences.

Key improvements:
1. Spectral equalization (Griffin-Lim style) to flatten power spectrum
2. Aggressive tau annealing in L-BFGS-B
3. Min-peak LP and max-sum LP alternating
4. Larger sequence lengths explored
5. Peak-shaving direct projection
6. Better time management with diverse restarts
"""

import numpy as np
from scipy.optimize import minimize, linprog
from scipy.fft import rfft, irfft
import time


def compute_conv(a):
    """Compute linear convolution a*a using FFT. Returns array of length 2n-1."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    conv = irfft(fa * fa, size)
    return conv[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    conv = compute_conv(a)
    max_b = float(np.max(conv))
    sum_a = float(np.sum(a))
    if sum_a < 0.01:
        return np.inf
    return 2.0 * n * max_b / (sum_a * sum_a)


def fft_convolve(a, b):
    """Linear convolution of a and b using FFT."""
    n, m = len(a), len(b)
    size = 1
    while size < n + m - 1:
        size <<= 1
    fa = rfft(a, size)
    fb = rfft(b, size)
    return irfft(fa * fb, size)[:n + m - 1]


def softmax(z):
    """Numerically stable softmax."""
    z = z - np.max(z)
    e = np.exp(z)
    return e / np.sum(e)


def soft_max_obj_and_grad_z(z, tau):
    """Compute soft-max of conv(softmax(z), softmax(z)) and gradient w.r.t. z."""
    n = len(z)
    a = softmax(z)

    conv = compute_conv(a)
    cmax = float(np.max(conv))
    shifted = np.clip((conv - cmax) / tau, -500.0, 500.0)
    exp_conv = np.exp(shifted)
    Z = float(np.sum(exp_conv))
    soft_max_val = cmax + tau * np.log(Z)
    w = exp_conv / Z

    # grad_a[i] = 2 * sum_j w[i+j] * a[j] (for valid indices)
    w_rev = w[::-1]
    conv_fw = fft_convolve(a, w_rev)
    f1 = conv_fw[n - 1:2 * n - 1][::-1]
    grad_a = 2.0 * f1

    mean_grad = float(np.dot(a, grad_a))
    grad_z = a * (grad_a - mean_grad)

    return soft_max_val, grad_z


def optimize_lfbs(a_init, tau_list, max_iter=300, time_budget=30.0):
    """Optimize using L-BFGS-B with soft-max parametrization and tau annealing."""
    n = len(a_init)
    a = a_init.copy()
    s = np.sum(a)
    if s < 1e-10:
        a = np.ones(n) / n
    else:
        a = a / s
    a = np.maximum(a, 1e-30)
    z = np.log(a)

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    for tau in tau_list:
        if time.monotonic() - start > time_budget:
            break

        def func_and_grad(z_val):
            return soft_max_obj_and_grad_z(z_val, tau)

        try:
            result = minimize(func_and_grad, z, jac=True, method='L-BFGS-B',
                              options={'maxiter': max_iter, 'ftol': 1e-14, 'gtol': 1e-8})
            z_new = result.x
        except Exception:
            continue

        a_new = softmax(z_new)
        a_new = np.maximum(a_new, 0)
        s_new = np.sum(a_new)
        if s_new < 1e-10:
            continue
        a_new = a_new / s_new
        obj = compute_obj(a_new)

        if obj < best_obj:
            best_obj = obj
            best_a = a_new.copy()
            z = z_new

    return best_a, best_obj


def build_toeplitz(a):
    """Build matrix A where (A@g)[k] = (conv(a,g))[k]."""
    n = len(a)
    A = np.zeros((2 * n - 1, n), dtype=np.float64)
    for j in range(n):
        k_start = j
        k_end = min(j + n, 2 * n - 1)
        A[k_start:k_end, j] = a[:k_end - k_start]
    return A


def solve_lp_maxsum(a, M, time_limit=15):
    """Solve LP: max sum(g) s.t. (a*g)[k] <= M for all k, g >= 0."""
    n = len(a)
    A = build_toeplitz(a)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': time_limit})
        if res.status == 0:
            g = np.maximum(res.x, 0)
            s = float(np.sum(g))
            if s > 1e-10:
                return g / s, s
    except Exception:
        pass
    return None, None


def solve_lp_minpeak(a, time_limit=15):
    """Solve LP: min M s.t. (a*g)[k] <= M for all k, sum(g)=1, g>=0."""
    n = len(a)
    A = build_toeplitz(a)
    c = np.zeros(n + 1)
    c[-1] = 1.0
    A_ub = np.zeros((2 * n - 1, n + 1))
    A_ub[:, :n] = A
    A_ub[:, n] = -1.0
    b_ub = np.zeros(2 * n - 1)
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])
    bounds = [(0.0, None)] * n + [(0.0, None)]
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': time_limit})
        if res.status == 0:
            g = np.maximum(res.x[:n], 0)
            s = np.sum(g)
            if s > 1e-10:
                g = g / s
                M_opt = float(res.x[n])
                return g, M_opt
    except Exception:
        pass
    return None, None


def alternating_lp(a, max_iters=40, time_budget=40.0):
    """Alternating LP with line search."""
    n = len(a)
    a = a.copy().astype(np.float64)
    s = np.sum(a)
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    stall_count = 0
    patience = 15

    for it in range(max_iters):
        if time.monotonic() - start > time_budget:
            break

        conv = compute_conv(a)
        M = float(np.max(conv))

        g, _ = solve_lp_maxsum(a, M, time_limit=12)
        if g is None:
            g2, _ = solve_lp_minpeak(a, time_limit=12)
            if g2 is None:
                break
            g = g2

        best_t_obj = best_obj
        best_a_new = best_a.copy()
        for t in [0.05, 0.1, 0.2, 0.3, 0.5, 0.7, 1.0]:
            a_try = (1.0 - t) * a + t * g
            a_try = np.maximum(a_try, 0)
            s_try = np.sum(a_try)
            if s_try < 1e-10:
                continue
            a_try = a_try / s_try
            obj_try = compute_obj(a_try)
            if obj_try < best_t_obj:
                best_t_obj = obj_try
                best_a_new = a_try.copy()

        if best_t_obj < best_obj - 1e-12:
            best_obj = best_t_obj
            best_a = best_a_new.copy()
            a = best_a_new
            stall_count = 0
        else:
            stall_count += 1
            if stall_count > patience:
                rng = np.random.default_rng(it * 7919 + 12345)
                a = a * np.exp(rng.normal(0, 0.05, n))
                a = np.maximum(a, 0)
                s = np.sum(a)
                if s > 1e-10:
                    a = a / s
                stall_count = 0
                patience = 8
            else:
                a = 0.9 * a + 0.1 * g
                a = np.maximum(a, 0)
                s = np.sum(a)
                if s > 1e-10:
                    a = a / s

    return best_a


def spectral_equalize(a, num_iters=50, gamma=1.5, time_budget=5.0):
    """
    Griffin-Lim style spectral equalization.
    Flattens the power spectrum of the zero-padded sequence.
    """
    n = len(a)
    a = a.copy().astype(np.float64)
    s = np.sum(a)
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    N = 1
    while N < 2 * n - 1:
        N <<= 1

    for _ in range(num_iters):
        if time.monotonic() - start > time_budget:
            break

        A = rfft(a, N)
        P = np.abs(A) ** 2
        total_power = float(np.sum(P))
        target = total_power / N

        # Reduce peaks
        mask = P > target * gamma
        if np.any(mask):
            ratio = np.sqrt(target * gamma / np.maximum(P[mask], 1e-30))
            A[mask] *= ratio[:, np.newaxis] if ratio.ndim > 0 else ratio

        a_new = irfft(A, N)[:n]
        a_new = np.maximum(a_new, 0)
        s_new = np.sum(a_new)
        if s_new < 1e-10:
            break
        a_new = a_new / s_new

        obj = compute_obj(a_new)
        if obj < best_obj:
            best_obj = obj
            best_a = a_new.copy()
            a = a_new

    return best_a


def peak_shave(a, num_iters=30, time_budget=5.0):
    """Direct peak shaving: iteratively reduce the maximum convolution value."""
    n = len(a)
    a = a.copy().astype(np.float64)
    s = np.sum(a)
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    for _ in range(num_iters):
        if time.monotonic() - start > time_budget:
            break

        conv = compute_conv(a)
        k_max = int(np.argmax(conv))
        max_val = float(conv[k_max])

        # Gradient of conv[k_max] w.r.t. a[i] is a[k_max - i] (for valid i)
        # To reduce conv[k_max], decrease a[i] proportional to a[k_max-i]
        grad = np.zeros(n)
        for i in range(n):
            j = k_max - i
            if 0 <= j < n:
                grad[i] = a[j]
            else:
                j2 = k_max + i
                if 0 <= j2 < n:
                    pass  # already handled by symmetry in convolution

        # Actually, d(conv[k])/da[i] = a[k-i] if 0<=k-i<n, else a[k+n-1-i]... 
        # Let me use the correct formula:
        # conv[k] = sum_{i} a[i]*a[k-i] (valid indices)
        # d(conv[k])/da[i] = a[k-i] if 0<=k-i<n, plus a[i-k+n-1]... 
        # Simpler: use FFT-based approach
        # conv[k] = sum_j a[j] * a[k-j]
        # d/d a[i] = a[k-i] (if 0<=k-i<n) + a[k-i+n] ... no
        # For linear conv: conv[k] = sum_{j=max(0,k-n+1)}^{min(k,n-1)} a[j]*a[k-j]
        # d(conv[k])/da[i] = a[k-i] (if 0<=k-i<n and k-i is in valid range for second factor)
        # Actually: conv[k] = sum_j a[j]*a[k-j], so d(conv[k])/da[i] = a[k-i] (if valid)

        # Let me just compute it directly
        grad = np.zeros(n)
        for i in range(n):
            j = k_max - i
            if 0 <= j < n:
                grad[i] += a[j]

        # Step: reduce a in direction of grad
        step = 0.02 * max_val / (np.max(grad) + 1e-10)
        a_new = a - step * grad
        a_new = np.maximum(a_new, 0)
        s_new = np.sum(a_new)
        if s_new < 1e-10:
            break
        a_new = a_new / s_new

        obj = compute_obj(a_new)
        if obj < best_obj:
            best_obj = obj
            best_a = a_new.copy()
            a = a_new
        else:
            # Try smaller step
            a_new2 = a - 0.5 * step * grad
            a_new2 = np.maximum(a_new2, 0)
            s2 = np.sum(a_new2)
            if s2 > 1e-10:
                a_new2 = a_new2 / s2
                obj2 = compute_obj(a_new2)
                if obj2 < best_obj:
                    best_obj = obj2
                    best_a = a_new2.copy()
                    a = a_new2
                else:
                    break
            else:
                break

    return best_a


def subgradient_optimize(a, max_iters=3000, time_budget=20.0, tau_start=3.0, tau_end=50.0):
    """Smoothed subgradient optimization."""
    n = len(a)
    a = a.copy().astype(np.float64)
    s = np.sum(a)
    if s < 1e-10:
        a = np.ones(n) / n
    else:
        a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    step = 0.1
    stall = 0

    for it in range(max_iters):
        if time.monotonic() - start > time_budget:
            break

        frac = it / max(1, max_iters)
        tau = tau_start + (tau_end - tau_start) * frac

        conv = compute_conv(a)
        max_conv = float(np.max(conv))
        shifted = np.clip(tau * (conv - max_conv), -300, 300)
        weights = np.exp(shifted)
        w_sum = np.sum(weights)
        if w_sum < 1e-300:
            idx = int(np.argmax(conv))
            weights = np.zeros(len(conv))
            weights[idx] = 1.0
        else:
            weights /= w_sum

        w_rev = weights[::-1]
        conv_fw = fft_convolve(a, w_rev)
        grad_a = 2.0 * conv_fw[n - 1:2 * n - 1][::-1]

        grad_centered = grad_a - np.mean(grad_a)
        a_new = a - step * grad_centered
        a_new = np.maximum(a_new, 0)
        s_new = np.sum(a_new)
        if s_new > 1e-10:
            a_new = a_new / s_new
        else:
            a_new = a.copy()

        a = a_new
        obj = compute_obj(a)

        if obj < best_obj - 1e-12:
            best_obj = obj
            best_a = a.copy()
            stall = 0
        else:
            stall += 1
            if stall > 80:
                step *= 0.85
                if step < 1e-10:
                    break
                rng = np.random.default_rng(it * 31 + 7)
                a = a + rng.normal(0, 0.01, n) * step
                a = np.maximum(a, 0)
                s = np.sum(a)
                if s > 1e-10:
                    a = a / s
                stall = 0

        if it > 100:
            step *= 0.997

    return best_a


def random_init(n, seed):
    """Generate diverse random initializations."""
    rng = np.random.default_rng(seed)
    k = np.arange(n)
    choice = rng.integers(0, 12)

    if choice == 0:
        f = rng.uniform(0.5, 4.0)
        phase = rng.uniform(0, 2 * np.pi)
        a = 1.0 + 0.8 * np.cos(2 * np.pi * f * k / n + phase)
    elif choice == 1:
        f1 = rng.uniform(0.3, 2.0)
        f2 = rng.uniform(2.0, 6.0)
        a = 1.0 + 0.4 * np.cos(2 * np.pi * f1 * k / n) + 0.3 * np.cos(2 * np.pi * f2 * k / n)
    elif choice == 2:
        center = n * rng.uniform(0.3, 0.7)
        alpha = rng.uniform(0.5, 3.0)
        a = 1.0 + 0.5 * np.exp(-alpha * np.abs(k - center) / (n * 0.3))
    elif choice == 3:
        a = rng.random(n)
        win = max(3, n // 20)
        kernel = np.ones(win) / win
        a = np.convolve(a, kernel, mode='same')
    elif choice == 4:
        a = np.ones(n)
        for _ in range(rng.integers(1, 5)):
            f = rng.uniform(0.5, 5.0)
            amp = rng.uniform(0.1, 0.4)
            a += amp * np.cos(2 * np.pi * f * k / n + rng.uniform(0, 2 * np.pi))
    elif choice == 5:
        f = rng.uniform(0.3, 2.0)
        a = 1.0 + 0.5 * np.sin(2 * np.pi * f * k / n)
    elif choice == 6:
        nblocks = rng.integers(2, 7)
        a = np.ones(n)
        for b in range(nblocks):
            start = b * n // nblocks
            end = (b + 1) * n // nblocks
            a[start:end] = rng.uniform(0.3, 1.5)
    elif choice == 7:
        f0 = rng.uniform(0.5, 2.0)
        f1 = rng.uniform(2.0, 6.0)
        a = 1.0 + 0.5 * np.cos(2 * np.pi * (f0 * k / n + (f1 - f0) * k**2 / (2 * n**2)))
    elif choice == 8:
        center = n * rng.uniform(0.2, 0.8)
        width = n * rng.uniform(0.1, 0.4)
        a = 1.0 + 0.6 * np.exp(-0.5 * ((k - center) / width) ** 2)
    elif choice == 9:
        # Multi-frequency with random phases
        a = np.ones(n)
        for _ in range(rng.integers(2, 5)):
            f = rng.uniform(0.5, 8.0)
            amp = rng.uniform(0.05, 0.3)
            ph = rng.uniform(0, 2 * np.pi)
            a += amp * np.cos(2 * np.pi * f * k / n + ph)
    elif choice == 10:
        # Smooth random via RFF
        a = np.ones(n)
        num_f = rng.integers(3, 8)
        for _ in range(num_f):
            f = rng.uniform(1.0, 10.0)
            amp = rng.uniform(0.05, 0.2)
            ph = rng.uniform(0, 2 * np.pi)
            a += amp * np.cos(2 * np.pi * f * k / n + ph)
    else:
        a = rng.random(n) + 0.1

    a = np.maximum(a, 0.01)
    s = np.sum(a)
    return a / s


def resample_sequence(a, new_n):
    """Resample sequence to new length via interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return a_new


def try_parent():
    """Try to load parent sequence."""
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64)
        parent = np.clip(parent, 0, 1000.0)
        if parent.size > 0 and np.isfinite(parent).all():
            obj = compute_obj(parent)
            print(f"Parent (n={parent.size}): obj={obj:.6f}")
            return parent.copy(), obj
    except Exception:
        pass
    return None, np.inf


def polish_sequence(a, time_budget, verbose=False):
    """Intensive polishing combining all methods."""
    n = len(a)
    a = a.copy().astype(np.float64)
    start = time.monotonic()

    best_a = a.copy()
    best_obj = compute_obj(a)
    round_num = 0

    while time.monotonic() - start < time_budget - 3:
        round_num += 1
        remaining = time_budget - (time.monotonic() - start)
        if remaining < 5:
            break

        # LP step
        if n <= 900 and remaining > 20:
            a_new = alternating_lp(best_a, max_iters=25, time_budget=min(remaining * 0.3, 35.0))
            obj = compute_obj(a_new)
            if obj < best_obj:
                best_obj = obj
                best_a = a_new.copy()
                if verbose:
                    print(f"    Polish r{round_num} LP: {best_obj:.6f}")

        # L-BFGS-B with small tau
        remaining = time_budget - (time.monotonic() - start)
        if remaining > 12:
            tau_final = 0.2 / n
            tau_small = max(0.05 / n, tau_final * 0.5)
            a_new, obj_new = optimize_lfbs(best_a, [tau_small, tau_final * 0.3],
                                           max_iter=150, time_budget=min(remaining * 0.2, 12.0))
            if obj_new < best_obj:
                best_obj = obj_new
                best_a = a_new.copy()
                if verbose:
                    print(f"    Polish r{round_num} L-BFGS: {best_obj:.6f}")

        # Spectral equalization
        remaining = time_budget - (time.monotonic() - start)
        if remaining > 8:
            a_new = spectral_equalize(best_a, num_iters=30, gamma=1.3,
                                      time_budget=min(remaining * 0.15, 6.0))
            obj = compute_obj(a_new)
            if obj < best_obj:
                best_obj = obj
                best_a = a_new.copy()
                if verbose:
                    print(f"    Polish r{round_num} Spectral: {best_obj:.6f}")

        # Peak shaving
        remaining = time_budget - (time.monotonic() - start)
        if remaining > 6:
            a_new = peak_shave(best_a, num_iters=20,
                               time_budget=min(remaining * 0.1, 4.0))
            obj = compute_obj(a_new)
            if obj < best_obj:
                best_obj = obj
                best_a = a_new.copy()
                if verbose:
                    print(f"    Polish r{round_num} PeakShave: {best_obj:.6f}")

        # Subgradient polish
        remaining = time_budget - (time.monotonic() - start)
        if remaining > 8:
            a_new = subgradient_optimize(best_a, max_iters=int(remaining * 4),
                                         time_budget=min(remaining * 0.15, 10.0),
                                         tau_start=10.0, tau_end=40.0)
            obj = compute_obj(a_new)
            if obj < best_obj:
                best_obj = obj
                best_a = a_new.copy()
                if verbose:
                    print(f"    Polish r{round_num} SG: {best_obj:.6f}")

    return best_a, best_obj


def propose_candidate(seed=42, budget_s=1500):
    """
    Main search function. Explores multiple sequence lengths using L-BFGS-B,
    LP alternating, spectral equalization, and subgradient methods.
    """
    start_time = time.monotonic()
    deadline = start_time + float(budget_s) - 20.0

    best_seq = None
    best_score = np.inf

    # Load parent
    parent, parent_obj = try_parent()
    if parent is not None and parent_obj < best_score:
        best_score = parent_obj
        best_seq = parent.copy()

    # N values to explore - focus on larger sizes for better bounds
    n_values = [400, 500, 600, 700, 800, 1000, 1200]
    if best_seq is not None:
        np_ = len(best_seq)
        extra = [np_, max(200, np_ - 100), min(1400, np_ + 100), min(1400, np_ + 200)]
        n_values = sorted(set(n_values + extra))

    print(f"Exploring n values: {n_values}")
    print(f"Initial best: {best_score:.6f}")

    candidates = []

    # Phase 1: Exploration
    for n in n_values:
        if time.monotonic() > deadline - 120:
            break
        remaining = deadline - time.monotonic()
        if remaining < 15:
            break

        # Time allocation: more time for larger n (better potential)
        if n <= 500:
            alloc = min(25.0, remaining * 0.08)
        elif n <= 800:
            alloc = min(40.0, remaining * 0.10)
        else:
            alloc = min(55.0, remaining * 0.12)
        alloc = max(15.0, alloc)

        print(f"\nN={n}: {remaining:.0f}s remaining, alloc={alloc:.0f}s")

        n_inits = max(2, int(alloc / 12))
        n_inits = min(n_inits, 4)

        for init_idx in range(n_inits):
            if time.monotonic() > deadline - 120:
                break
            remaining_init = deadline - time.monotonic()
            if remaining_init < 15:
                break

            a = random_init(n, seed + n * 104729 + init_idx * 79193)

            # Use parent if similar length
            if init_idx == 0 and best_seq is not None and abs(len(best_seq) - n) < 200:
                a = resample_sequence(best_seq, n)

            # Step 1: Quick subgradient shaping
            sg_time = min(5.0, (alloc / n_inits) * 0.15)
            a = subgradient_optimize(a, max_iters=int(sg_time * 30),
                                      time_budget=sg_time,
                                      tau_start=3.0, tau_end=15.0)

            # Step 2: L-BFGS-B with tau annealing
            remaining_init = deadline - time.monotonic()
            if remaining_init > 12:
                lfbs_time = min((alloc / n_inits) * 0.35, 20.0)
                tau_list = [8.0 / n, 4.0 / n, 2.0 / n, 1.0 / n, 0.5 / n, 0.2 / n, 0.1 / n]
                a, obj = optimize_lfbs(a, tau_list, max_iter=250, time_budget=lfbs_time)
            else:
                obj = compute_obj(a)

            # Step 3: Spectral equalization
            remaining_init = deadline - time.monotonic()
            if remaining_init > 8:
                a = spectral_equalize(a, num_iters=20, gamma=1.4,
                                      time_budget=min(remaining_init * 0.1, 5.0))
                obj = compute_obj(a)

            # Step 4: LP if n allows
            remaining_init = deadline - time.monotonic()
            if remaining_init > 12 and n <= 900:
                a = alternating_lp(a, max_iters=30, time_budget=min(remaining_init * 0.3, 25.0))
                obj = compute_obj(a)

            # Step 5: Peak shaving
            remaining_init = deadline - time.monotonic()
            if remaining_init > 5:
                a = peak_shave(a, num_iters=15, time_budget=min(remaining_init * 0.08, 3.0))
                obj = compute_obj(a)

            print(f"  init{init_idx}: n={n}, obj={obj:.6f}")

            if obj < best_score:
                best_score = obj
                best_seq = a.copy()
                print(f"  ** UPDATE: n={n}, obj={best_score:.6f} **")

            candidates.append((a.copy(), obj, n))

    # Phase 2: Polishing top candidates
    print(f"\n=== POLISHING PHASE ===")
    print(f"Best before polish: {best_score:.6f}")

    candidates.sort(key=lambda x: x[1])

    n_polish = min(4, len(candidates))
    for ci in range(n_polish):
        if time.monotonic() > deadline - 50:
            break
        remaining = deadline - time.monotonic()
        if remaining < 20:
            break

        a_cand = candidates[ci][0]
        n_cand = candidates[ci][2]
        print(f"  Polishing candidate {ci} (n={n_cand}, obj={candidates[ci][1]:.6f}), {remaining:.0f}s left")

        polish_budget = min(remaining * 0.35, 70.0)
        a_pol, obj_pol = polish_sequence(a_cand, time_budget=polish_budget, verbose=True)

        if obj_pol < best_score:
            best_score = obj_pol
            best_seq = a_pol.copy()
            print(f"  ** UPDATE after polish: obj={best_score:.6f} **")
        else:
            # Perturbation restarts
            remaining = deadline - time.monotonic()
            if remaining > 15:
                rng = np.random.default_rng(ci * 555 + 999)
                for ptry in range(4):
                    if time.monotonic() > deadline - 30:
                        break
                    scale = rng.uniform(0.02, 0.10)
                    a_pert = a_cand * np.exp(rng.normal(0, scale, n_cand))
                    a_pert = np.maximum(a_pert, 0)
                    s = np.sum(a_pert)
                    if s < 1e-10:
                        continue
                    a_pert = a_pert / s
                    a_pert = subgradient_optimize(a_pert, max_iters=400, time_budget=3.0,
                                                   tau_start=5.0, tau_end=25.0)
                    a_pert = spectral_equalize(a_pert, num_iters=15, gamma=1.3, time_budget=3.0)
                    obj_pert = compute_obj(a_pert)
                    if obj_pert < best_score:
                        best_score = obj_pert
                        best_seq = a_pert.copy()
                        print(f"  ** UPDATE (perturb {ptry}): obj={best_score:.6f} **")
                        break

    # Phase 3: Final intensive polish
    if best_seq is not None and time.monotonic() < deadline - 15:
        remaining = deadline - time.monotonic()
        n_best = len(best_seq)
        print(f"\n=== FINAL POLISH: n={n_best}, {remaining:.0f}s ===")

        a_cur = best_seq.copy()
        for final_round in range(5):
            if time.monotonic() > deadline - 8:
                break
            remaining = deadline - time.monotonic()
            if remaining < 10:
                break

            # L-BFGS-B with very small tau
            remaining = deadline - time.monotonic()
            if remaining > 10:
                tau_vsmall = 0.03 / n_best
                a_new, obj_new = optimize_lfbs(a_cur, [0.2 / n_best, tau_vsmall],
                                               max_iter=300, time_budget=min(remaining * 0.2, 15.0))
                if obj_new < best_score:
                    best_score = obj_new
                    best_seq = a_new.copy()
                    a_cur = a_new.copy()
                    print(f"  Final L-BFGS {final_round}: {best_score:.6f}")

            # LP
            remaining = deadline - time.monotonic()
            if remaining > 15 and n_best <= 900:
                a_new = alternating_lp(a_cur, max_iters=30, time_budget=min(remaining * 0.25, 25.0))
                obj_new = compute_obj(a_new)
                if obj_new < best_score:
                    best_score = obj_new
                    best_seq = a_new.copy()
                    a_cur = a_new.copy()
                    print(f"  Final LP {final_round}: {best_score:.6f}")

            # Spectral
            remaining = deadline - time.monotonic()
            if remaining > 8:
                a_new = spectral_equalize(a_cur, num_iters=40, gamma=1.2,
                                          time_budget=min(remaining * 0.15, 8.0))
                obj_new = compute_obj(a_new)
                if obj_new < best_score:
                    best_score = obj_new
                    best_seq = a_new.copy()
                    a_cur = a_new.copy()
                    print(f"  Final Spectral {final_round}: {best_score:.6f}")

            # Peak shave
            remaining = deadline - time.monotonic()
            if remaining > 5:
                a_new = peak_shave(a_cur, num_iters=30, time_budget=min(remaining * 0.1, 4.0))
                obj_new = compute_obj(a_new)
                if obj_new < best_score:
                    best_score = obj_new
                    best_seq = a_new.copy()
                    a_cur = a_new.copy()
                    print(f"  Final PeakShave {final_round}: {best_score:.6f}")

            # Subgradient
            remaining = deadline - time.monotonic()
            if remaining > 8:
                a_new = subgradient_optimize(a_cur, max_iters=int(remaining * 5),
                                              time_budget=min(remaining * 0.15, 12.0),
                                              tau_start=15.0, tau_end=60.0)
                obj_new = compute_obj(a_new)
                if obj_new < best_score:
                    best_score = obj_new
                    best_seq = a_new.copy()
                    a_cur = a_new.copy()
                    print(f"  Final SG {final_round}: {best_score:.6f}")

    # Final validation
    if best_seq is None:
        best_seq = np.ones(200)
        best_score = compute_obj(best_seq)

    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    val = evaluate_sequence(result)
    if val < np.inf:
        best_score = val
    else:
        result = [1.0] * 100
        best_score = 2.0

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_score:.6f}, time={elapsed:.1f}s ===")

    return result
```
