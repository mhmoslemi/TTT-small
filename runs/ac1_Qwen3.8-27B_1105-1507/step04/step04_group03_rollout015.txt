def fast_conv(a):
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]

def compute_J(a):
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))

def build_A_sparse(n, a):
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        j_min = max(0, k - n + 1)
        j_max = min(n - 1, k)
        for j in range(j_min, j_max + 1):
            rows.append(k)
            cols.append(j)
            vals.append(a[k - j])
    return csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))

def lp_solve(a, M):
    n = len(a)
    A = build_A_sparse(n, a)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    try:
        r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
        if r.success and r.x is not None:
            return np.maximum(r.x, 0)
    except Exception:
        pass
    return None

def lp_mirror_descent(a, max_iters, deadline, use_mult=False):

def lse_gradient(a, tau=50.0):
    """Compute gradient of log-sum-exp approximation to max(conv(a,a))."""
    n = len(a)
    b = fast_conv(a)
    # Shift b to be positive for numerical stability
    b_shift = b - b.max()
    weights = np.exp(tau * b_shift)
    weights = weights / weights.sum()
    
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        w = weights[k]
        if w > 1e-15:
            lo = max(0, k - n + 1)
            hi = min(n - 1, k)
            for i in range(lo, hi + 1):
                grad[i] += 2.0 * w * a[k - i]
    
    return grad

def adam_refine(a, max_iters, deadline, tau=50.0, lr=0.001):
    """Adam optimization on log-sum-exp smoothed objective."""
    a = np.maximum(a.astype(np.float64), 1e-10).copy()
    a = a / a.sum()
    best_J = compute_J(a)
    best_a = a.copy()
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1, beta2, eps = 0.8, 0.999, 1e-8
    it = 0
    
    while it < max_iters and time.monotonic() < deadline:
        it += 1
        # Adaptive tau
        current_tau = min(tau * (1 + it / 200.0), 500.0)
        grad = lse_gradient(a, current_tau)
        
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * grad ** 2
        m_hat = m / (1 - beta1 ** it)
        v_hat = v / (1 - beta2 ** it)
        
        a_new = a - lr * m_hat / (np.sqrt(v_hat) + eps)
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s < 1e-8:
            break
        a_new = a_new / s
        
        J_new = compute_J(a_new)
        if J_new < best_J:
            best_J = J_new
            best_a = a_new.copy()
        a = a_new
    
    return best_a, best_J

def fourier_parametrize(params, n):
    """Convert Fourier parameters to non-negative sequence."""
    t = np.arange(n, dtype=np.float64) / n
    a = np.full(n, params[0], dtype=np.float64)
    for k in range(1, len(params) // 2 + 1):
        if 2*k-1 < len(params):
            a += params[2*k-1] * np.cos(2.0 * np.pi * k * t)
        if 2*k < len(params):
            a += params[2*k] * np.sin(2.0 * np.pi * k * t)
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-8:
        return np.ones(n) / n
    return a / s

def seq_to_fourier(a, n_harm):
    n = len(a)
    a_norm = a / a.sum()
    t = np.arange(n, dtype=np.float64) / n
    params = np.zeros(2 * n_harm + 1)
    params[0] = np.mean(a_norm)
    for k in range(1, n_harm + 1):
        params[2*k-1] = 2.0 * np.mean(a_norm * np.cos(2.0 * np.pi * k * t))
        params[2*k] = 2.0 * np.mean(a_norm * np.sin(2.0 * np.pi * k * t))
    return params

def cmaes_optimize(n, n_harm, params0, sigma0, max_gens, deadline):
    """Simplified CMA-ES."""
    dim = len(params0)
    lam = max(4, 4 + int(3 * np.log(dim)))
    mu = lam // 2
    weights = np.log(mu + 0.5) - np.log(np.arange(1, mu + 1))
    weights /= weights.sum()
    mu_eff = 1.0 / np.sum(weights ** 2)
    c_sigma = (mu_eff + 2) / (dim + mu_eff + 5)
    c_c = (4 + mu_eff / dim) / (dim + 4 + 2 * mu_eff / dim)
    d_sigma = 1 + 2 * max(0, np.sqrt((mu_eff - 1) / (dim + 1) - 1) - 1) + c_sigma
    c_1 = 2 / ((dim + 1.3) ** 2 + mu_eff)
    c_mu = min(1 - c_1, 2 * (mu_eff - 2 + 1.0 / mu_eff) / ((dim + 2) ** 2 + mu_eff))
    chi_N = np.sqrt(dim) * (1 - 1/(4*dim) + 1/(21*dim**2))
    
    C = np.eye(dim)
    Cs = np.zeros((dim, dim))
    p_sigma = np.zeros(dim)
    p_c = np.zeros(dim)
    x = params0.copy()
    sigma = sigma0
    
    best_J = compute_J(fourier_parametrize(params0, n))
    best_x = params0.copy()
    
    for gen in range(max_gens):
        if time.monotonic() > deadline:
            break
        
        offspring = []
        offspring_J = []
        for i in range(lam):
            z = np.random.standard_normal(dim)
            # Cheap covariance application: eigendecomposition
            eigvals, eigvecs = np.linalg.eigh(C)
            eigvals = np.maximum(eigvals, 1e-14)
            y = (eigvecs * np.sqrt(eigvals)) @ (eigvecs.T @ z)
            x_trial = x + sigma * y
            a_trial = fourier_parametrize(x_trial, n)
            J_trial = compute_J(a_trial)
            offspring.append(x_trial)
            offspring_J.append(J_trial)
        
        idx = np.argsort(offspring_J)
        sorted_x = [offspring[i] for i in idx[:mu]]
        
        x_new = sum(weights[i] * sorted_x[i] for i in range(mu))
        
        if sigma > 1e-30:
            diff = (x_new - x) / (sigma * chi_N)
        else:
            diff = np.zeros(dim)
        p_sigma = (1 - c_sigma) * p_sigma + np.sqrt(c_sigma * (2-c_sigma) * mu_eff) * diff
        sigma *= np.exp(c_sigma / d_sigma * (np.linalg.norm(p_sigma)/chi_N - 1))
        sigma = max(sigma, 1e-12)
        
        h_sigma = 1.0
        if np.linalg.norm(p_sigma)/chi_N > (1.4+c_c)*np.sqrt((1+c_sigma)/(1-c_sigma)):

        p_c = (1-c_c)*p_c + h_sigma*np.sqrt(c_c*(2-c_c)*mu_eff)*(x_new-x)/max(sigma,1e-30)
        
        rank_1 = np.outer(p_c, p_c)
        rank_mu = sum(weights[i]*np.outer(sorted_x[i]-x, sorted_x[i]-x) for i in range(mu))
        
        Cs = (1-c_1-c_mu)*Cs + c_1*(rank_1 + h_sigma*(1-h_sigma)*np.outer(p_c,p_c)*c_c) + c_mu*rank_mu
        C = (1-c_1-c_mu)*C + c_1*(rank_1 + h_sigma*(1-h_sigma)*np.outer(p_c,p_c)*c_c) + c_mu*rank_mu
        C = (C + C.T) / 2
        x = x_new
        
        if offspring_J[idx[0]] < best_J:
            best_J = offspring_J[idx[0]]
            best_x = offspring[idx[0]].copy()
    
    return best_x, best_J

def resample(src, dest_n):
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)

        rank_mu = np.zeros((dim, dim))
        for i in range(mu):
            d = sorted_x[i] - x
            rank_mu += weights[i] * np.outer(d, d)
        
        Cs = (1-c_1-c_mu)*Cs + c_1*(rank_1 + h_sigma*(1-h_sigma)*np.outer(p_c,p_c)*c_c) + c_mu*rank_mu
        C = (1-c_1-c_mu)*C + c_1*(rank_1 + h_sigma*(1-h_sigma)*np.outer(p_c,p_c)*c_c) + c_mu*rank_mu
        C = (C + C.T) / 2
        x = x_new
        
        if offspring_M[idx[0]] < best_M:
            best_M = offspring_M[idx[0]]
            best_x = offspring[idx[0]].copy()
    
    return best_x, best_M

def resample(src, dest_n):
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)

class Tracker:
    def __init__(self):
        self.best = None
        self.best_score = float('inf')
        self.t_start = time.monotonic()
    
    def update(self, a):
        a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
        if a.size == 0 or a.sum() < 0.01:
            return
        score = float(evaluate_sequence(a.tolist()))
        if np.isfinite(score) and score < self.best_score:
            self.best_score = score
            self.best = a.copy()
            print(f"  [NEW BEST] J={score:.8f} n={a.size} t={time.monotonic()-self.t_start:.0f}s", flush=True)
