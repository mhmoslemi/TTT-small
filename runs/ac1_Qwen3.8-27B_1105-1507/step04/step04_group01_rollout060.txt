"""
Optimized subgradient + L-BFGS-B + LP hybrid for minimizing
2*n*max(conv(a,a))/(sum(a))^2 over non-negative sequences.
Uses the key insight that the gradient of max(conv) at the peak lag
is simply 2*a reversed and shifted, enabling very fast iterations.

import numpy as np
from scipy.optimize import minimize, linprog
from scipy.fft import rfft, irfft
import time


def compute_conv(a):
    """Compute full linear convolution a*a using FFT."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    conv = irfft(fa * fa, size)
    return conv[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 1e-10:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def fft_convolve(a, b):
    """Linear convolution of a and b using FFT."""
    n, m = len(a), len(b)
    size = 1
    while size < n + m - 1:
        size <<= 1
    fa = rfft(a, size)
    fb = rfft(b, size)
    return irfft(fa * fb, size)[:n + m - 1]


def project_simplex(v):
    """Project v onto {x: sum(x)=1, x>=0}."""
    v = np.asarray(v, dtype=np.float64)
    n = len(v)
    # Sort in decreasing order
    u = np.sort(v)[::-1]
    # Find rho
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - 1))[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0)

def exact_subgradient(a):
    """Compute the exact subgradient at the maximizing lag."""
    n = len(a)
    conv = compute_conv(a)
    k_star = int(np.argmax(conv))
    # grad[j] = 2 * a[k_star - j] for j in [max(0, k_star-(n-1)), min(k_star, n-1)]
    grad = np.zeros(n)
    j_min = max(0, k_star - (n - 1))
    j_max = min(k_star, n - 1)
    for j in range(j_min, j_max + 1):
        grad[j] = 2.0 * a[k_star - j]
    return grad, float(conv[k_star])


def subgradient_descent(a, time_budget=10.0, tau_init=5.0, tau_growth=1.02,
                        max_iters=10000, use_smoothing=True):
    """
    Subgradient descent with adaptive step size and momentum.
    a is normalized to sum=1.
    """
    n = len(a)
    a = a.copy().astype(np.float64)
    s = np.sum(a)
    if s < 1e-10:
        return np.ones(n) / n
    a = a / s
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    
    # Momentum
    v = np.zeros(n)
    mu = 0.9
    
    # Adaptive step size
    alpha = 1.0 / np.sqrt(n)
    
    # Smoothing parameter
    tau = tau_init / n
    
    stall = 0
    best_stall_obj = best_obj
    
    for it in range(max_iters):
        if time.monotonic() - start > time_budget:
            break
        
        # Compute gradient
        if use_smoothing:
            grad, _ = smooth_gradient(a, tau)
        else:
            grad, _ = exact_subgradient(a)
        
        # Center the gradient (since we're on the simplex, only the component
        # orthogonal to the all-ones vector matters)
        grad_centered = grad - np.mean(grad)
        
        # Momentum update
        v = mu * v + (1 - mu) * grad_centered
        a_new = a - alpha * v
        a_new = project_simplex(a_new)
        
        obj = compute_obj(a_new)
        
        if obj < best_obj - 1e-12:
            best_obj = obj
            best_a = a_new.copy()
            stall = 0
            a = a_new
        else:
            a = a_new  # still move (for momentum)
            stall += 1
        
        # Adapt step size
        if stall > 50:
            alpha *= 0.8
            v *= 0.5
            stall = 0
            if alpha < 1e-10:
                break
        elif stall == 0:
            alpha = min(alpha * 1.05, 1.0)
        
        # Adapt tau (increase smoothing over time to get a more global gradient)
        if use_smoothing:
            tau *= tau_growth
        
        # Periodic momentum decay
        if it % 500 == 0 and it > 0:
            mu = max(0.5, mu - 0.05)
    
    return best_a


def lbfgs_softmax(a_init, tau_list, time_budget=20.0, max_iter=200):
    """L-BFGS-B with softmax parametrization."""
    n = len(a_init)
    a = a_init.copy().astype(np.float64)
    s = np.sum(a)
    if s < 1e-10:
        a = np.ones(n) / n
    else:
        a = a / s
    a = np.maximum(a, 1e-30)
    z = np.log(a)
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    
    for tau in tau_list:
        if time.monotonic() - start > time_budget:
            break
        
        # Define objective and gradient in z-space
        # a = softmax(z), obj = 2*n*max(conv(a,a))
        # grad_z = a * (grad_a - sum(a*grad_a))
        
        def func_grad(z_val):
            a_val = _softmax(z_val)
            conv = compute_conv(a_val)
            cmax = float(np.max(conv))
            shifted = np.clip((conv - cmax) / tau, -500, 500)
            exp_c = np.exp(shifted)
            Z = float(np.sum(exp_c))
            soft_val = cmax + tau * np.log(Z)
            w = exp_c / Z
            
            w_rev = w[::-1]
            conv_fw = fft_convolve(a_val, w_rev)
            grad_a = 2.0 * conv_fw[n-1:2*n-1][::-1]
            
            mean_g = float(np.dot(a_val, grad_a))
            grad_z = a_val * (grad_a - mean_g)
            
            return 2.0 * n * soft_val, grad_z
        
        try:
            result = minimize(func_grad, z, jac=True, method='L-BFGS-B',
                              options={'maxiter': max_iter, 'ftol': 1e-15, 'gtol': 1e-8})
            z_new = result.x
            a_new = _softmax(z_new)
            a_new = np.maximum(a_new, 0)
            s_new = np.sum(a_new)
            if s_new < 1e-10:
                continue
            a_new = a_new / s_new
            obj = compute_obj(a_new)
            if obj < best_obj:
                best_obj = obj
                best_a = a_new.copy()
                z = z_new
        except Exception:
            continue
    
    return best_a, best_obj


def _softmax(z):
    z = z - np.max(z)
    e = np.exp(z)
    return e / np.sum(e)


def build_toeplitz(a):
    """Build matrix A where (A@g)[k] = (conv(a,g))[k]."""
    n = len(a)
    A = np.zeros((2 * n - 1, n), dtype=np.float64)
    for j in range(n):
        k_start = j
        k_end = min(j + n, 2 * n - 1)
        A[k_start:k_end, j] = a[:k_end - k_start]
    return A


def lp_alternate(a, time_budget=20.0, max_iters=30):

    n = len(a)
    a = a.copy().astype(np.float64)
    s = np.sum(a)
    if s < 1e-10:
        return a
    a = a / s
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    stall = 0
    
    for it in range(max_iters):
        if time.monotonic() - start > time_budget:
            break
        
        conv = compute_conv(a)
        M = float(np.max(conv))
        
        # Solve LP: max sum(g) s.t. conv(a,g)[k] <= M, g >= 0
        A = build_toeplitz(a)
        c = -np.ones(n)
        b_ub = np.full(2 * n - 1, M)
        bounds = [(0.0, None)] * n
        try:
            res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                          options={'presolve': True, 'time_limit': 10})
            if res.status == 0:
                g = np.maximum(res.x, 0)
                sg = float(np.sum(g))
                if sg > 1e-10:
                    g = g / sg
                    # Line search
                    improved = False
                    for t in [1.0, 0.7, 0.5, 0.3, 0.1]:

                        a_try = np.maximum(a_try, 0)
                        st = np.sum(a_try)
                        if st < 1e-10:
                            continue

                            break
                    if improved:

                        stall = 0
                    else:
                        stall += 1
                        if stall > 10:
                            break
        except Exception:
            break
    
    return best_a


def resample(a, new_n):
    """Resample sequence to new length."""
    n = len(a)
    if n == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    return np.interp(x_new, x_old, a)


def make_init(n, seed):
    """Generate diverse initializations."""
    rng = np.random.default_rng(seed)
    k = np.arange(n)
    choice = rng.integers(0, 12)
    
    if choice == 0:
        # Linear increasing
        a = np.arange(1, n + 1, dtype=np.float64)
    elif choice == 1:
        # Geometric
        r = rng.uniform(1.01, 2.0)
        a = r ** k
    elif choice == 2:
        # Quadratic
        a = (k + 1) ** 2
    elif choice == 3:
        # Exponential
        rate = rng.uniform(0.5, 3.0)
        a = np.exp(rate * k / n)
    elif choice == 4:
        # Cosine window (raised cosine)
        a = 0.5 * (1 + np.cos(2 * np.pi * k / (n - 1))) + 0.1
    elif choice == 5:
        # Inverse cosine (peaked in middle)
        a = 1.0 + 0.8 * np.cos(np.pi * k / (n - 1))
    elif choice == 6:
        # Random smooth
        a = rng.random(n)
        win = max(3, n // 30)
        kernel = np.ones(win) / win
        a = np.convolve(a, kernel, mode='same')
    elif choice == 7:
        # Two-sided geometric (peak in middle)
        center = n / 2
        rate = rng.uniform(0.5, 2.0)
        a = np.exp(-rate * np.abs(k - center) / n)
    elif choice == 8:
        # Power law
        p = rng.uniform(0.5, 3.0)
        a = (k + 1) ** p
    elif choice == 9:
        # Sinusoidal modulation
        f = rng.uniform(0.5, 3.0)
        a = 1.0 + 0.5 * np.cos(2 * np.pi * f * k / n)
    elif choice == 10:
        # Step function
        nsteps = rng.integers(2, 8)
        a = np.ones(n)

            end = (b + 1) * n // nsteps

    else:
        # Random
        a = rng.random(n) + 0.01
    
    a = np.maximum(a, 1e-6)
    s = np.sum(a)
    return a / s

"""
Hybrid subgradient + L-BFGS-B + LP optimization for minimizing
2*n*max(conv(a,a))/(sum(a))^2 over non-negative sequences.
Uses the key structural insight that the subgradient of max(conv) at the
peak lag is simply 2*a reversed and shifted, enabling very fast iterations.
Exploits scale invariance (fixes sum=1) and uses diverse initializations
including monotone, geometric, and power-law profiles.

import numpy as np
from scipy.optimize import minimize, linprog
from scipy.fft import rfft, irfft
import time


def compute_conv(a):
    """Compute full linear convolution a*a using FFT. Returns array of length 2n-1."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    conv = irfft(fa * fa, size)
    return conv[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 1e-10:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def fft_convolve(a, b):
    """Linear convolution of a and b using FFT."""
    n, m = len(a), len(b)
    size = 1
    while size < n + m - 1:
        size <<= 1
    fa = rfft(a, size)
    fb = rfft(b, size)
    return irfft(fa * fb, size)[:n + m - 1]


def project_simplex(v):
    """Project v onto {x: sum(x)=1, x>=0} using Duchi et al. algorithm."""
    v = np.asarray(v, dtype=np.float64)
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    idx = np.arange(1, n + 1)
    cond = u * idx > (cssv - 1)
    if not np.any(cond):
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0)


def _softmax(z):
    """Numerically stable softmax."""
    z = z - np.max(z)
    e = np.exp(z)
    return e / np.sum(e)


def smooth_gradient(a, tau):
    """
    Compute smoothed gradient of max(conv(a,a)) w.r.t. a using softmax weights.
    Returns (gradient array of length n, max_conv value).
    """
    n = len(a)
    conv = compute_conv(a)
    cmax = float(np.max(conv))
    shifted = np.clip((conv - cmax) / tau, -500, 500)
    exp_c = np.exp(shifted)
    total = float(np.sum(exp_c))
    if total < 1e-300:
        w = np.zeros(len(conv))
        w[int(np.argmax(conv))] = 1.0
    else:
        w = exp_c / total

    # grad[j] = 2 * sum_{m=0}^{n-1} a[m] * w[j+m]
    # = 2 * conv(a, w_rev)[j + 2n - 2]
    w_rev = w[::-1]
    conv_aw = fft_convolve(a, w_rev)
    grad = 2.0 * conv_aw[2 * n - 2: 3 * n - 2]
    return grad, cmax


def exact_subgradient(a):
    """Compute exact subgradient at the maximizing lag. Very fast."""
    n = len(a)
    conv = compute_conv(a)
    k_star = int(np.argmax(conv))
    j_min = max(0, k_star - (n - 1))
    j_max = min(k_star, n - 1)
    grad = np.zeros(n)
    # grad[j] = 2 * a[k_star - j] for j in [j_min, j_max]
    # k_star - j ranges from k_star - j_max to k_star - j_min (decreasing)

    grad[j_min:j_max + 1] = 2.0 * a[j_min:j_max + 1][::-1]
    return grad, float(conv[k_star])

    v = np.zeros(n)
    mu = 0.85

    while time.monotonic() - start < time_budget:
        if use_smoothing:
            grad, _ = smooth_gradient(a, tau)
            tau *= 1.02
        else:
            grad, _ = exact_subgradient(a)

        grad_c = grad - np.mean(grad)
        v = mu * v + (1.0 - mu) * grad_c
        a_new = project_simplex(a - alpha * v)

        obj = compute_obj(a_new)

        if obj < best_obj - 1e-14:
            best_obj = obj
            best_a = a_new.copy()
            a = a_new
            stall = 0
            alpha = min(alpha * 1.03, 0.5)
        else:
            a = 0.9 * a + 0.1 * a_new
            stall += 1
            if stall > 30:
                alpha *= 0.7
                v *= 0.3
                stall = 0
                if alpha < 1e-10:
                    break

        if int((time.monotonic() - start) * 10) % 50 == 0:
            mu = max(0.5, mu * 0.99)

    return best_a


def lbfgs_softmax(a_init, tau_list, time_budget=20.0, max_iter=200):
    """L-BFGS-B with softmax parametrization and tau annealing."""
    n = len(a_init)
    a = a_init.copy().astype(np.float64)
    s = np.sum(a)
    if s < 1e-10:
        a = np.ones(n) / n
    else:
        a = a / s
    a = np.maximum(a, 1e-30)
    z = np.log(a)

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    for tau in tau_list:
        if time.monotonic() - start > time_budget:
            break

        def func_grad(z_val):
            a_val = _softmax(z_val)
            conv = compute_conv(a_val)
            cmax = float(np.max(conv))
            shifted = np.clip((conv - cmax) / tau, -500, 500)
            exp_c = np.exp(shifted)
            Z = float(np.sum(exp_c))
            soft_val = cmax + tau * np.log(Z)
            w = exp_c / Z
            w_rev = w[::-1]
            conv_fw = fft_convolve(a_val, w_rev)
            grad_a = 2.0 * conv_fw[n - 1:2 * n - 1][::-1]
            mean_g = float(np.dot(a_val, grad_a))
            grad_z = a_val * (grad_a - mean_g)
            return 2.0 * n * soft_val, grad_z

        try:
            result = minimize(func_grad, z, jac=True, method='L-BFGS-B',
                              options={'maxiter': max_iter, 'ftol': 1e-15, 'gtol': 1e-8})
            z_new = result.x
            a_new = _softmax(z_new)
            a_new = np.maximum(a_new, 0)
            s_new = np.sum(a_new)
            if s_new < 1e-10:
                continue
            a_new = a_new / s_new
            obj = compute_obj(a_new)
            if obj < best_obj:
                best_obj = obj
                best_a = a_new.copy()
                z = z_new
        except Exception:
            continue

    return best_a, best_obj


def build_toeplitz(a):
    """Build matrix A where (A@g)[k] = (conv(a,g))[k]."""
    n = len(a)
    A = np.zeros((2 * n - 1, n), dtype=np.float64)
    for j in range(n):
        k_start = j
        k_end = min(j + n, 2 * n - 1)
        A[k_start:k_end, j] = a[:k_end - k_start]
    return A


def lp_alternate(a, time_budget=20.0, max_iters=30):
    """LP alternating optimization."""
    n = len(a)
    a = a.copy().astype(np.float64)
    s = np.sum(a)
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    stall = 0

    for it in range(max_iters):
        if time.monotonic() - start > time_budget:
            break

        conv = compute_conv(a)
        M = float(np.max(conv))

        A = build_toeplitz(a)
        c = -np.ones(n)
        b_ub = np.full(2 * n - 1, M)
        bounds = [(0.0, None)] * n
        try:
            res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                          options={'presolve': True, 'time_limit': 10})
            if res.status == 0:
                g = np.maximum(res.x, 0)
                sg = float(np.sum(g))
                if sg > 1e-10:
                    g = g / sg
                    improved = False
                    for t in [1.0, 0.7, 0.5, 0.3, 0.1]:
                        a_try = (1 - t) * a + t * g
                        a_try = np.maximum(a_try, 0)
                        st = np.sum(a_try)
                        if st < 1e-10:
                            continue
                        a_try = a_try / st
                        obj_try = compute_obj(a_try)
                        if obj_try < best_obj - 1e-12:
                            best_obj = obj_try
                            best_a = a_try.copy()
                            improved = True
                            break
                    if improved:
                        a = best_a
                        stall = 0
                    else:
                        stall += 1
                        if stall > 8:
                            break
        except Exception:
            break

    return best_a


def resample_seq(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    return np.interp(x_new, x_old, a)


def make_init(n, seed):
    """Generate diverse structured initializations."""
    rng = np.random.default_rng(seed)
    k = np.arange(n, dtype=np.float64)
    choice = rng.integers(0, 14)

    if choice == 0:
        # Linear increasing
        a = k + 1
    elif choice == 1:
        # Geometric
        r = rng.uniform(1.005, 1.5)
        a = r ** (k / n * n)
    elif choice == 2:
        # Quadratic increasing
        a = (k + 1) ** 2
    elif choice == 3:
        # Cubic
        p = rng.uniform(1.5, 3.0)
        a = (k + 1) ** p
    elif choice == 4:
        # Exponential increasing
        rate = rng.uniform(1.0, 5.0)
        a = np.exp(rate * k / n)
    elif choice == 5:
        # Inverse (decreasing)
        a = (n - k) ** rng.uniform(1.0, 2.5)
    elif choice == 6:
        # Raised cosine
        a = 0.3 + 0.7 * (1 + np.cos(np.pi * k / (n - 1))) / 2
    elif choice == 7:
        # Gaussian bump off-center
        center = n * rng.uniform(0.2, 0.8)
        width = n * rng.uniform(0.1, 0.4)
        a = 0.1 + np.exp(-0.5 * ((k - center) / width) ** 2)
    elif choice == 8:
        # Two-sided geometric (peak at random position)
        center = n * rng.uniform(0.2, 0.8)
        rate = rng.uniform(1.0, 4.0)
        a = np.exp(-rate * np.abs(k - center) / n)
    elif choice == 9:
        # Sinusoidal
        f = rng.uniform(0.5, 2.0)
        phase = rng.uniform(0, 2 * np.pi)
        a = 1.0 + 0.6 * np.cos(2 * np.pi * f * k / n + phase)
    elif choice == 10:
        # Step function with random levels
        nsteps = rng.integers(2, 6)
        a = np.ones(n)
        for b in range(nsteps):
            start = b * n // nsteps
            end = (b + 1) * n // nsteps
            a[start:end] = rng.uniform(0.2, 2.0)
    elif choice == 11:
        # Smooth random
        a = rng.random(n)
        win = max(5, n // 20)
        kernel = np.hanning(win)
        kernel /= kernel.sum()
        a = np.convolve(a, kernel, mode='same')
    elif choice == 12:
        # Power law with offset
        p = rng.uniform(0.3, 2.0)
        a = 0.1 + (k / n) ** p
    else:
        # Pure random
        a = rng.random(n) + 0.01

    a = np.maximum(a, 1e-6)
    s = np.sum(a)
    return a / s


def polish_sequence(a, time_budget=40.0):
    """Multi-stage polishing of a sequence."""
    n = len(a)
    a = a.copy().astype(np.float64)
    start = time.monotonic()
    best_a = a.copy()
    best_obj = compute_obj(a)

    for round_num in range(8):
        remaining = time_budget - (time.monotonic() - start)
        if remaining < 5:
            break

        # Fast subgradient
        t1 = min(remaining * 0.2, 8.0)
        if t1 > 2:
            a1 = fast_subgradient(best_a, time_budget=t1, use_smoothing=False)
            obj1 = compute_obj(a1)
            if obj1 < best_obj:
                best_obj = obj1
                best_a = a1.copy()

        # Smoothed subgradient
        remaining = time_budget - (time.monotonic() - start)
        if remaining > 5:
            t2 = min(remaining * 0.25, 12.0)
            a2 = fast_subgradient(best_a, time_budget=t2, use_smoothing=True, tau_init=2.0)
            obj2 = compute_obj(a2)
            if obj2 < best_obj:
                best_obj = obj2
                best_a = a2.copy()

        # L-BFGS-B with small tau
        remaining = time_budget - (time.monotonic() - start)
        if remaining > 10 and n <= 900:
            t3 = min(remaining * 0.3, 15.0)
            tau_list = [3.0 / n, 1.0 / n, 0.5 / n]
            a3, obj3 = lbfgs_softmax(best_a, tau_list, time_budget=t3, max_iter=150)
            if obj3 < best_obj:
                best_obj = obj3
                best_a = a3.copy()

        # LP
        remaining = time_budget - (time.monotonic() - start)
        if remaining > 15 and n <= 450:
            t4 = min(remaining * 0.3, 20.0)
            a4 = lp_alternate(best_a, time_budget=t4, max_iters=20)
            obj4 = compute_obj(a4)
            if obj4 < best_obj:
                best_obj = obj4
                best_a = a4.copy()

    return best_a, best_obj


def propose_candidate(seed=42, budget_s=1500):
    """
    Main search: explores multiple sequence lengths using fast subgradient
    (exploiting the simple gradient structure), L-BFGS-B with softmax,
    LP alternating, and diverse structural initializations.
    """
    start_time = time.monotonic()
    deadline = start_time + float(budget_s) - 30.0

    best_seq = None
    best_score = np.inf

    # Load parent
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64)
        parent = np.clip(parent, 0, 1000.0)
        if parent.size > 0 and np.isfinite(parent).all():
            s = np.sum(parent)
            if s > 0.01:
                parent_norm = parent / s
                obj = compute_obj(parent_norm)
                print(f"Parent (n={parent.size}): obj={obj:.6f}")
                best_score = obj
                best_seq = parent_norm.copy()
    except Exception:
        pass

    # N values to explore
    n_values = [100, 150, 200, 250, 300, 400, 500, 600, 700, 800]
    if best_seq is not None:
        np_ = len(best_seq)
        extra = [np_, max(80, np_ - 50), min(1000, np_ + 50), min(1000, np_ + 100),
                 min(1000, np_ + 200)]
        n_values = sorted(set(n_values + extra))

    print(f"Exploring n values: {n_values}")
    print(f"Initial best: {best_score:.6f}")

    candidates = []

    # Phase 1: Exploration with diverse initializations
    for n in n_values:
        if time.monotonic() > deadline - 180:
            break
        remaining = deadline - time.monotonic()
        if remaining < 25:
            break

        alloc = min(70.0, remaining * 0.14)
        alloc = max(12.0, alloc)
        print(f"\nN={n}: {remaining:.0f}s remaining, alloc={alloc:.0f}s")

        n_inits = max(2, min(4, int(alloc / 14)))

        for idx in range(n_inits):
            if time.monotonic() > deadline - 180:
                break
            remain = deadline - time.monotonic()
            if remain < 10:
                break

            if idx == 0 and best_seq is not None and abs(len(best_seq) - n) < 200:
                a = resample_seq(best_seq, n)
            else:
                a = make_init(n, seed + n * 104729 + idx * 79193)

            # Fast exact subgradient (very cheap, many iterations)
            t_sg = min(4.0, (alloc / n_inits) * 0.2)
            a = fast_subgradient(a, time_budget=t_sg, use_smoothing=False)

            # Smoothed subgradient
            remain = deadline - time.monotonic()
            if remain > 10:
                t_sg2 = min((alloc / n_inits) * 0.3, 10.0)
                a = fast_subgradient(a, time_budget=t_sg2, use_smoothing=True, tau_init=3.0)

            # L-BFGS-B refinement
            remain = deadline - time.monotonic()
            if remain > 10:
                tau_list = [5.0 / n, 2.0 / n, 1.0 / n, 0.5 / n]
                t_lb = min((alloc / n_inits) * 0.35, 18.0)
                a, obj = lbfgs_softmax(a, tau_list, time_budget=t_lb, max_iter=200)
            else:
                obj = compute_obj(a)

            # LP if feasible
            remain = deadline - time.monotonic()
            if remain > 12 and n <= 400:
                t_lp = min(remain * 0.25, 15.0)
                a = lp_alternate(a, time_budget=t_lp, max_iters=25)
                obj = compute_obj(a)

            print(f"  init{idx}: n={n}, obj={obj:.6f}")

            if obj < best_score:
                best_score = obj
                best_seq = a.copy()
                print(f"  ** NEW BEST: n={n}, obj={best_score:.6f} **")

            candidates.append((a.copy(), obj, n))

    # Phase 2: Polish top candidates
    print(f"\n=== POLISHING ===")
    print(f"Best before polish: {best_score:.6f}")

    candidates.sort(key=lambda x: x[1])
    n_polish = min(4, len(candidates))

    for ci in range(n_polish):
        if time.monotonic() > deadline - 80:
            break
        remaining = deadline - time.monotonic()
        if remaining < 20:
            break

        a_c = candidates[ci][0]
        n_c = candidates[ci][2]
        print(f"  Polishing cand {ci} (n={n_c}, obj={candidates[ci][1]:.6f}), {remaining:.0f}s left")

        polish_budget = min(remaining * 0.3, 50.0)
        a_p, obj_p = polish_sequence(a_c, time_budget=polish_budget)

        if obj_p < best_score:
            best_score = obj_p
            best_seq = a_p.copy()
            print(f"  ** NEW BEST after polish: obj={best_score:.6f} **")
        else:
            # Perturbation restarts
            remain = deadline - time.monotonic()
            if remain > 25:
                rng = np.random.default_rng(ci * 12345 + 67890)
                for ptry in range(5):
                    if time.monotonic() > deadline - 30:
                        break
                    scale = rng.uniform(0.02, 0.1)
                    a_pert = a_c * np.exp(rng.normal(0, scale, n_c))
                    a_pert = np.maximum(a_pert, 0)
                    s_p = np.sum(a_pert)
                    if s_p < 1e-10:
                        continue
                    a_pert = a_pert / s_p
                    a_pert = fast_subgradient(a_pert, time_budget=3.0, use_smoothing=False)
                    a_pert = fast_subgradient(a_pert, time_budget=4.0, use_smoothing=True, tau_init=2.0)
                    obj_pert = compute_obj(a_pert)
                    if obj_pert < best_score:
                        best_score = obj_pert
                        best_seq = a_pert.copy()
                        print(f"  ** NEW BEST (perturb {ptry}): obj={best_score:.6f} **")

    # Phase 3: Final intensive polish on global best
    if best_seq is not None and time.monotonic() < deadline - 15:
        remaining = deadline - time.monotonic()
        n_best = len(best_seq)
        print(f"\n=== FINAL POLISH: n={n_best}, {remaining:.0f}s ===")

        a_cur = best_seq.copy()
        for rd in range(8):
            if time.monotonic() > deadline - 5:
                break
            remaining = deadline - time.monotonic()
            if remaining < 8:
                break

            # Fast exact SG
            a_new = fast_subgradient(a_cur, time_budget=min(remaining * 0.12, 8.0), use_smoothing=False)
            obj_new = compute_obj(a_new)
            if obj_new < best_score:
                best_score = obj_new
                best_seq = a_new.copy()
                a_cur = a_new.copy()
                print(f"  Final FSG {rd}: {best_score:.6f}")

            # Smoothed SG
            remaining = deadline - time.monotonic()
            if remaining > 10:
                a_new = fast_subgradient(a_cur, time_budget=min(remaining * 0.15, 10.0),
                                         use_smoothing=True, tau_init=1.5)
                obj_new = compute_obj(a_new)
                if obj_new < best_score:
                    best_score = obj_new
                    best_seq = a_new.copy()
                    a_cur = a_new.copy()
                    print(f"  Final SSG {rd}: {best_score:.6f}")

            # L-BFGS-B with very small tau
            remaining = deadline - time.monotonic()
            if remaining > 12 and n_best <= 900:
                tau_list = [2.0 / n_best, 1.0 / n_best, 0.5 / n_best, 0.2 / n_best]
                a_new, obj_new = lbfgs_softmax(a_cur, tau_list,
                                               time_budget=min(remaining * 0.25, 20.0),
                                               max_iter=300)
                if obj_new < best_score:
                    best_score = obj_new
                    best_seq = a_new.copy()
                    a_cur = a_new.copy()
                    print(f"  Final LB {rd}: {best_score:.6f}")

            # LP
            remaining = deadline - time.monotonic()
            if remaining > 18 and n_best <= 500:
                a_new = lp_alternate(a_cur, time_budget=min(remaining * 0.25, 20.0), max_iters=25)
                obj_new = compute_obj(a_new)
                if obj_new < best_score:
                    best_score = obj_new
                    best_seq = a_new.copy()
                    a_cur = a_new.copy()
                    print(f"  Final LP {rd}: {best_score:.6f}")

    # Final validation
    if best_seq is None:
        best_seq = np.ones(200) / 200
        best_score = compute_obj(best_seq)

    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    val = evaluate_sequence(result)
    if np.isfinite(val):
        best_score = val
    else:
        result = [1.0] * 100
        best_score = 2.0

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_score:.6f}, time={elapsed:.1f}s ===")
    return result
