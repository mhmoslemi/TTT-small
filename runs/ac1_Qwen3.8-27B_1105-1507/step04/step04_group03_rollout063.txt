"""
Hybrid optimizer combining LP mirror descent, novel coordinate descent with
incremental convolution updates, and smoothed L-BFGS-B. The key innovation is
a fast coordinate descent that exploits the 1D convex structure of each
coordinate subproblem, enabling thousands of steps per second for fast local
exploration and escape from local minima found by LP methods alone.

import numpy as np
from scipy.optimize import linprog, minimize
from scipy.sparse import csr_matrix
import time


def fast_conv(a):
    """Full self-convolution via FFT."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]


def compute_J(a):
    """Compute J(a) = 2*n*max(conv(a,a))/(sum(a))^2."""
    a = np.maximum(a.astype(np.float64), 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))


def build_A_sparse(n, a):
    """Build sparse matrix A where A[k,j] = a[k-j] for valid j."""
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        j_min = max(0, k - n + 1)
        j_max = min(n - 1, k)
        for j in range(j_min, j_max + 1):
            rows.append(k)
            cols.append(j)
            vals.append(a[k - j])
    return csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))


def lp_best_response(a, M):
    """Solve LP: max sum(g) s.t. A@g <= M*1, g >= 0."""
    n = len(a)
    A = build_A_sparse(n, a)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    try:
        r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                    options={'presolve': True, 'time_limit': 5.0})
        if r.success and r.x is not None:
            g = np.maximum(r.x, 0)
            if g.sum() > 1e-15:
                return g
    except Exception:
        pass
    return None


def lp_mirror_descent(a, max_iters, deadline, use_multiplicative=False, verbose=False):
    """LP mirror descent with adaptive line search."""
    a = np.maximum(a.astype(np.float64), 1e-10)
    s = a.sum()
    a = a / s
    current_J = compute_J(a)
    best_J = current_J
    best_a = a.copy()
    no_improve = 0
    it_count = 0

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        it_count += 1

        b = fast_conv(a)
        M = float(b.max())
        g = lp_best_response(a, M)
        if g is None:
            break
        g = np.maximum(g, 0.0)
        g_sum = g.sum()
        if g_sum < 1e-10:
            break
        g = g / g_sum

        # Adaptive step size
        base_t = 0.3 * np.exp(-0.2 * it / 150.0) + 0.05
        t = min(1.0, base_t)

        best_trial_J = current_J
        best_trial_a = None

        for ls in range(20):
            if use_multiplicative:
                ratio = g / np.maximum(a, 1e-15)
                a_trial = a * np.power(ratio, t)
            else:
                a_trial = (1.0 - t) * a + t * g
            a_trial = np.maximum(a_trial, 0.0)
            st = a_trial.sum()
            if st < 1e-10:
                t *= 0.5
                continue
            a_trial = a_trial / st
            J_trial = compute_J(a_trial)
            if J_trial < best_trial_J - 1e-14:
                best_trial_J = J_trial
                best_trial_a = a_trial.copy()
                break
            t *= 0.6
            if t < 0.001:
                break

        if best_trial_a is not None:
            a = best_trial_a
            current_J = best_trial_J
            if current_J < best_J - 1e-15:
                best_J = current_J
                best_a = a.copy()
                if verbose and it % 100 == 0:
                    print(f"    LP it={it}, J={best_J:.8f}", flush=True)
            no_improve = 0
        else:
            no_improve += 1
            if no_improve > 50:
                break

    return best_a, best_J


def coordinate_descent(a, max_sweeps, deadline, verbose=False):
    """
    Coordinate descent minimizing max(conv(a,a)) with incremental updates.
    For each coordinate j, solve the 1D convex subproblem exactly via binary search.
    """
    a = np.maximum(a.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-10:
        a = np.ones(len(a)) / len(a)
        s = 1.0
    a = a / s
    n = len(a)
    c = fast_conv(a)
    current_max = c.max()
    best_max = current_max
    best_a = a.copy()

    # Precompute valid index ranges for each j
    # For coordinate j, the affected k values are: k = j + idx for idx in [0, n-1]
    # i.e., k in [j, j+n-1], clipped to [0, 2n-2]

    for sweep in range(max_sweeps):
        if time.monotonic() > deadline:
            break

        # Random shuffle order for diversity
        order = np.random.permutation(n)

        for j in order:
            aj = a[j]
            if aj < 1e-15:
                continue

            # Binary search on target M
            # Current max is a valid upper bound
            M_hi = current_max
            M_lo = 0.0

            # Theoretical lower bound: we can't go below the "forced" values
            # For k where a[k-j] = 0 (and k != 2j), c[k] is unchanged
            # So M_lo = max of those c[k] values
            # But computing this each time is O(n), so let's just use 0

            best_delta = 0.0
            best_M = M_hi

            for _ in range(40):
                M_mid = (M_lo + M_hi) / 2.0

                # Check quadratic feasibility (k = 2j)
                if 2 * j >= 2 * n - 1:
                    # 2j is out of range, no quadratic constraint
                    quad_lo = float('-inf')
                    quad_hi = float('inf')
                else:
                    val = a[j] * a[j] + M_mid - c[2 * j]
                    if val < 0:
                        M_lo = M_mid
                        continue
                    sqrt_val = np.sqrt(val)
                    quad_lo = -a[j] - sqrt_val
                    quad_hi = -a[j] + sqrt_val

                # Linear constraints: for k != 2j where 0 <= k-j < n and a[k-j] > 0
                # delta <= (M_mid - c[k]) / (2 * a[k-j])
                upper = float('inf')
                idx = k - j  # not right, need to loop
                # For k in [j, j+n-1] intersect [0, 2n-2], k != 2j:
                k_lo = max(0, j)
                k_hi = min(2 * n - 2, j + n - 1)

                # Vectorized computation
                if k_lo <= k_hi:
                    k_vals = np.arange(k_lo, k_hi + 1)
                    # Exclude k = 2j
                    mask = k_vals != 2 * j
                    if np.any(mask):
                        k_masked = k_vals[mask]
                        idx_masked = k_masked - j  # in [0, n-1]
                        a_vals = a[idx_masked]
                        c_vals = c[k_masked]
                        # For a_vals > 0: upper bound on delta
                        pos_mask = a_vals > 1e-15
                        if np.any(pos_mask):
                            bounds = (M_mid - c_vals[pos_mask]) / (2.0 * a_vals[pos_mask])
                            upper = min(upper, float(np.min(bounds)))
                        # For a_vals = 0: need c[k] <= M_mid
                        if np.any(~pos_mask):
                            if np.max(c_vals[~pos_mask]) > M_mid + 1e-12:
                                # Infeasible
                                M_lo = M_mid
                                continue

                # Final feasibility check
                lo_bound = max(-a[j], quad_lo)
                hi_bound = min(upper, quad_hi)

                if lo_bound <= hi_bound + 1e-12:
                    # Feasible
                    best_M = M_mid
                    best_delta = (lo_bound + hi_bound) / 2.0
                    M_hi = M_mid
                else:
                    M_lo = M_mid

            # Apply the delta
            if abs(best_delta) > 1e-14:
                new_aj = a[j] + best_delta
                if new_aj < 0:
                    new_aj = 0.0
                    best_delta = -a[j]

                # Update convolution incrementally
                k_lo = max(0, j)
                k_hi = min(2 * n - 2, j + n - 1)
                if k_lo <= k_hi:
                    k_vals = np.arange(k_lo, k_hi + 1)
                    idx_vals = k_vals - j
                    a_idx = a[idx_vals]
                    # For k = 2j: delta is 2*delta*old_aj + delta^2
                    # For others: delta is 2*delta*a[k-j]
                    updates = 2.0 * best_delta * a_idx
                    if 2 * j >= k_lo and 2 * j <= k_hi:
                        pos_2j = np.where(k_vals == 2 * j)[0]
                        if len(pos_2j) > 0:
                            updates[pos_2j[0]] = 2.0 * best_delta * a[j] + best_delta * best_delta
                    c[k_lo:k_hi + 1] += updates

                a[j] = new_aj

                # Renormalize
                s = a.sum()
                if s > 1e-10:
                    a = a / s
                    c = c / (s * s)

                current_max = c.max()
                if current_max < best_max - 1e-15:
                    best_max = current_max
                    best_a = a.copy()

            # Periodic full recompute to avoid drift
            if (j + 1) % 64 == 0:
                c = fast_conv(a)
                s = a.sum()
                if s > 1e-10:
                    a = a / s
                    c = c / s  # wait, c = conv(a/s, a/s) = conv(a,a)/s^2
                    # Actually we already normalized a, so recompute
                c = fast_conv(a)
                current_max = c.max()
                if current_max < best_max:
                    best_max = current_max
                    best_a = a.copy()

        if verbose and sweep % 50 == 0:
            print(f"    CD sweep={sweep}, max_conv={c.max():.10f}", flush=True)

    # Final normalization
    s = a.sum()
    if s > 1e-10:
        a = a / s
    return best_a

def smoothed_max_objective(x, tau):
    """Smoothed max of self-convolution (scale-invariant)."""
    ax = np.maximum(x, 0.0)
    s = ax.sum()
    if s < 1e-8:
        return 1e10
    ax = ax / s
    b = fast_conv(ax)
    # Soft-max approximation
    b_max = b.max()
    shifted = tau * (b - b_max)
    log_sum_exp = b_max + np.log(np.sum(np.exp(shifted))) / tau
    return float(log_sum_exp)


def smoothed_max_grad(x, tau):
    """Gradient of smoothed max of self-convolution."""
    ax = np.maximum(x, 0.0)
    s = ax.sum()
    if s < 1e-8:
        return 1e10, np.zeros_like(x)
    ax = ax / s
    b = fast_conv(ax)
    n = len(ax)
    b_max = b.max()
    shifted = tau * (b - b_max)
    weights = np.exp(shifted)
    weights = weights / weights.sum()  # softmax weights

    # Gradient of sum_k w_k * b_k w.r.t. ax
    # b_k = sum_i ax_i * ax_{k-i}
    # db_k/dax_j = 2 * ax_{k-j} (with boundary handling)
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        if weights[k] < 1e-10:
            continue
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        for i in range(lo, hi + 1):
            grad[i] += 2.0 * weights[k] * ax[k - i]

    return smoothed_max_objective(x, tau), grad


def lbb_smoothed(a, deadline):
    """L-BFGS-B with adaptive smoothing parameter."""
    n = len(a)
    best_a = a.copy()

    for tau in [50.0, 20.0, 10.0, 5.0, 2.0, 1.0]:
        if time.monotonic() > deadline:
            break

        def obj(x):
            return smoothed_max_objective(x, tau)

        # Use L-BFGS-B
        try:
            res = minimize(obj, best_a.astype(np.float64), method='L-BFGS-B',
                           bounds=[(0.0, None)] * n,

            if res.fun < 1e9:

        except Exception:
            pass

def resample_seq(src, dest_n):
    """Resample 1D sequence to new length."""
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)


def structured_perturb(a, rng, perturb_type):
    """Apply a structured perturbation."""
    n = len(a)
    t = np.arange(n, dtype=np.float64) / n

    if perturb_type == 'shift':
        s = int(rng.integers(1, max(2, n // 10)))
        a_p = np.roll(a, s)
    elif perturb_type == 'cosine':
        freq = rng.uniform(1, 6)
        amp = rng.uniform(0.02, 0.08)
        phase = rng.uniform(0, 2 * np.pi)
        a_p = a * (1 + amp * np.cos(2 * np.pi * freq * t + phase))
    elif perturb_type == 'smooth':
        k = int(rng.integers(3, max(4, n // 12)))
        kernel = np.ones(k) / k
        a_p = np.convolve(a, kernel, mode='same')
    elif perturb_type == 'taper':
        alpha = rng.uniform(0.6, 0.9)
        edge = max(1, int(n * (1 - alpha) / 2))
        window = np.ones(n)
        for i in range(edge):
            frac = (i + 1) / (edge + 1)
            window[i] = frac
            window[n - 1 - i] = frac
        a_p = a * window
    elif perturb_type == 'antiphase':
        # Add a component that's antiphase at the peak lags
        freq = int(rng.integers(1, 4))
        amp = rng.uniform(0.01, 0.05)
        a_p = a * (1 - amp * np.cos(2 * np.pi * freq * t))
    else:
        a_p = a + rng.normal(0, 0.02, n)

    a_p = np.maximum(a_p, 0.001)
    s = a_p.sum()
    if s > 1e-8:
        a_p = a_p / s
    return a_p


class ScoreTracker:
    def __init__(self):
        self.best = None
        self.best_score = float('inf')
        self.t_start = time.monotonic()

    def update(self, a):
        a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
        if a.size == 0 or a.sum() < 0.01:
            return
        score = float(evaluate_sequence(a.tolist()))
        if np.isfinite(score) and score < self.best_score:
            self.best = a.copy()
            self.best_score = score
            elapsed = time.monotonic() - self.t_start
            print(f"  [NEW BEST] J={score:.8f} n={a.size} t={elapsed:.0f}s", flush=True)


def propose_candidate(seed=42, budget_s=1500, **kwargs):
    """
    Hybrid optimizer: LP mirror descent + coordinate descent + smoothed L-BFGS-B.
    Key innovation: fast coordinate descent exploiting 1D convex subproblems.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + float(budget_s) - 20.0
    tracker = ScoreTracker()

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        if p.ndim == 1 and p.size > 0 and np.all(np.isfinite(p)):
            parent = np.maximum(p, 0.0)
            tracker.update(parent)
            print(f"Parent: n={parent.size}, J={tracker.best_score:.6f}", flush=True)
    except (NameError, AttributeError):
        pass

    if parent is None:
        parent = np.ones(640)
        tracker.update(parent)

    current_best = parent.copy()
    current_best_n = current_best.size
    current_best_J = compute_J(current_best)

    # === Phase 1: LP mirror descent from parent (linear) ===
    t0 = time.monotonic()
    print(f"\n{'='*50}", flush=True)
    print(f"Phase 1: LP linear (n={current_best_n})", flush=True)
    print(f"{'='*50}", flush=True)

    phase1_dl = min(deadline - 900, time.monotonic() + 150)
    a1 = current_best.copy()
    a1 = np.maximum(a1, 1e-10)
    a1 = a1 / a1.sum()
    a1, J1 = lp_mirror_descent(a1, 500, phase1_dl, use_multiplicative=False, verbose=True)
    tracker.update(a1)
    print(f"  J={J1:.8f} ({time.monotonic()-t0:.1f}s)", flush=True)

    if J1 < current_best_J:
        current_best = a1.copy()
        current_best_J = J1

    # === Phase 2: LP multiplicative ===
    if time.monotonic() < deadline - 800:
        t0 = time.monotonic()
        print(f"\n{'='*50}", flush=True)
        print(f"Phase 2: LP multiplicative (n={current_best_n})", flush=True)
        print(f"{'='*50}", flush=True)

        phase2_dl = min(deadline - 700, time.monotonic() + 80)
        a2 = current_best.copy()
        a2 = np.maximum(a2, 1e-10)
        a2 = a2 / a2.sum()
        a2, J2 = lp_mirror_descent(a2, 300, phase2_dl, use_multiplicative=True)
        tracker.update(a2)
        print(f"  J={J2:.8f} ({time.monotonic()-t0:.1f}s)", flush=True)

        if J2 < current_best_J:
            current_best = a2.copy()
            current_best_J = J2

    # === Phase 3: Coordinate descent (NEW) ===
    if time.monotonic() < deadline - 500:
        t0 = time.monotonic()
        print(f"\n{'='*50}", flush=True)
        print(f"Phase 3: Coordinate descent (n={current_best_n})", flush=True)
        print(f"{'='*50}", flush=True)

        phase3_dl = min(deadline - 400, time.monotonic() + 350)
        a3 = coordinate_descent_fast(current_best, 10000, phase3_dl)
        tracker.update(a3)
        J3 = compute_J(a3)
        print(f"  CD J={J3:.8f} ({time.monotonic()-t0:.1f}s)", flush=True)

        if J3 < current_best_J:
            current_best = a3.copy()
            current_best_J = J3

        # Follow up with LP from CD result
        if time.monotonic() < deadline - 350 and J3 < current_best_J + 0.01:
            t0b = time.monotonic()
            a3_lp = current_best.copy()
            a3_lp = np.maximum(a3_lp, 1e-10)
            a3_lp = a3_lp / a3_lp.sum()
            lp_dl = min(deadline - 300, time.monotonic() + 120)
            a3_lp, J3_lp = lp_mirror_descent(a3_lp, 500, lp_dl, use_multiplicative=False)
            tracker.update(a3_lp)
            print(f"  LP from CD: J={J3_lp:.8f} ({time.monotonic()-t0b:.1f}s)", flush=True)
            if J3_lp < current_best_J:
                current_best = a3_lp.copy()
                current_best_J = J3_lp

    # === Phase 4: Structured perturbations + LP + CD ===
    if time.monotonic() < deadline - 250:
        t0 = time.monotonic()
        print(f"\n{'='*50}", flush=True)
        print(f"Phase 4: Perturb + LP + CD (n={current_best_n})", flush=True)
        print(f"{'='*50}", flush=True)

        perturb_types = ['shift', 'cosine', 'smooth', 'taper', 'antiphase',
                         'shift', 'cosine', 'shift', 'antiphase']
        n_pert = len(perturb_types)
        remaining = deadline - 250 - time.monotonic()
        per_pert_time = max(10, min(50, remaining / n_pert / 2))

        for i, ptype in enumerate(perturb_types):
            if time.monotonic() > deadline - 230:
                break
            a_pert = structured_perturb(current_best, rng, ptype)

            # LP step
            lp_dl = time.monotonic() + per_pert_time / 2
            a_pert, J_pert = lp_mirror_descent(a_pert, 200, lp_dl, use_multiplicative=False)

            # CD step
            cd_dl = time.monotonic() + per_pert_time / 2
            a_pert = coordinate_descent_fast(a_pert, 500, cd_dl)
            J_pert = compute_J(a_pert)
            tracker.update(a_pert)

            if J_pert < current_best_J:
                current_best = a_pert.copy()
                current_best_J = J_pert
                print(f"  [{ptype}] J={J_pert:.8f}", flush=True)

        print(f"Phase 4 done ({time.monotonic()-t0:.1f}s)", flush=True)

    # === Phase 5: Smoothed L-BFGS-B ===
    if time.monotonic() < deadline - 150:
        t0 = time.monotonic()
        print(f"\n{'='*50}", flush=True)
        print(f"Phase 5: Smoothed L-BFGS-B (n={current_best_n})", flush=True)
        print(f"{'='*50}", flush=True)

        smb_dl = min(deadline - 100, time.monotonic() + 120)
        a5, J5 = lbb_smoothed(current_best, smb_dl)
        tracker.update(a5)
        print(f"  S-L-BFGS J={J5:.8f} ({time.monotonic()-t0:.1f}s)", flush=True)

        if J5 < current_best_J:
            current_best = a5.copy()
            current_best_J = J5

            # Polish with LP
            if time.monotonic() < deadline - 50:
                a5_lp = np.maximum(current_best, 1e-10)
                a5_lp = a5_lp / a5_lp.sum()
                a5_lp, J5_lp = lp_mirror_descent(a5_lp, 300, time.monotonic() + 40, use_multiplicative=False)
                tracker.update(a5_lp)
                if J5_lp < current_best_J:
                    current_best = a5_lp.copy()
                    current_best_J = J5_lp

    # === Phase 6: Multi-length exploration ===
    if time.monotonic() < deadline - 100:
        t0 = time.monotonic()
        print(f"\n{'='*50}", flush=True)
        print(f"Phase 6: Multi-length (base n={current_best_n})", flush=True)
        print(f"{'='*50}", flush=True)

        base_n = current_best_n
        lengths_to_try = sorted(set([
            max(256, base_n - 64), min(1280, base_n + 64),
            min(1280, base_n + 128), min(1024, base_n * 2)
        ]))
        lengths_to_try = [n for n in lengths_to_try if 256 <= n <= 1280 and abs(n - base_n) >= 32]

        for n_try in lengths_to_try:
            if time.monotonic() > deadline - 80:
                break
            a_try = resample_seq(current_best, n_try)
            a_try = np.maximum(a_try, 1e-10)
            a_try = a_try / a_try.sum()

            # Quick LP
            perf = max(10, min(40, (deadline - 80 - time.monotonic()) / max(1, len(lengths_to_try))))
            a_try, J_try = lp_mirror_descent(a_try, 200, time.monotonic() + perf, use_multiplicative=False)

            # Quick CD
            a_try = coordinate_descent_fast(a_try, 200, time.monotonic() + perf)
            J_try = compute_J(a_try)
            tracker.update(a_try)
            print(f"  n={n_try}: J={J_try:.8f}", flush=True)

            if J_try < current_best_J:
                current_best = a_try.copy()
                current_best_n = n_try
                current_best_J = J_try

        print(f"Phase 6 done ({time.monotonic()-t0:.1f}s)", flush=True)

    # === Phase 7: Long LP + CD refinement on best ===
    if time.monotonic() < deadline - 30:
        t0 = time.monotonic()
        print(f"\n{'='*50}", flush=True)
        print(f"Phase 7: Final LP + CD (n={current_best_n})", flush=True)
        print(f"{'='*50}", flush=True)

        a7 = current_best.copy()
        a7 = np.maximum(a7, 1e-10)
        a7 = a7 / a7.sum()

        # Long LP
        lp_dl = time.monotonic() + min(120, (deadline - 30 - time.monotonic()) * 0.5)
        a7, J7 = lp_mirror_descent(a7, 5000, lp_dl, use_multiplicative=False, verbose=True)
        tracker.update(a7)
        J7 = compute_J(a7)
        print(f"  LP J={J7:.8f}", flush=True)

        if J7 < current_best_J:
            current_best = a7.copy()
            current_best_J = J7

        # CD polish
        if time.monotonic() < deadline - 15:
            cd_dl = min(deadline - 5, time.monotonic() + 60)
            a7_cd = coordinate_descent_fast(current_best, 2000, cd_dl)
            tracker.update(a7_cd)
            J7_cd = compute_J(a7_cd)
            print(f"  CD J={J7_cd:.8f}", flush=True)
            if J7_cd < current_best_J:
                current_best = a7_cd.copy()
                current_best_J = J7_cd

        # Very short multiplicative LP
        if time.monotonic() < deadline - 5:
            a7m = current_best.copy()
            a7m = np.maximum(a7m, 1e-10)
            a7m = a7m / a7m.sum()
            a7m, J7m = lp_mirror_descent(a7m, 500, deadline - 3, use_multiplicative=True)
            tracker.update(a7m)
            if J7m < current_best_J:
                current_best = a7m.copy()
                current_best_J = J7m
                print(f"  Mult LP J={J7m:.8f}", flush=True)

        print(f"Phase 7 done ({time.monotonic()-t0:.1f}s)", flush=True)

    # Final decision
    if tracker.best is not None:
        final_a = tracker.best
    else:
        final_a = current_best

    # Ensure valid output
    final_a = np.maximum(final_a, 0.0)
    s = final_a.sum()
    if s < 0.01:
        final_a = np.ones(640)
    final_a = np.minimum(final_a, 1000.0)

    print(f"\n{'='*50}", flush=True)
    print(f"FINAL: J={tracker.best_score:.8f} n={final_a.size} time={time.monotonic()-tracker.t_start:.0f}s", flush=True)
    print(f"{'='*50}", flush=True)

    return [float(x) for x in final_a]
