"""
Hybrid optimizer: LP mirror descent + smoothed L-BFGS + multi-start.
Focuses on local refinement with second-order methods after coarse LP optimization.
"""

import numpy as np
from scipy.optimize import linprog, minimize
from scipy.sparse import csr_matrix
import time

def fast_conv(a):
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]

def compute_J(a):
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))

def build_A_sparse(n, a):
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        j_min = max(0, k - n + 1)
        j_max = min(n - 1, k)
        for j in range(j_min, j_max + 1):
            rows.append(k)
            cols.append(j)
            vals.append(a[k - j])
    return csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))

def lp_best_response(a, M):
    n = len(a)
    A = build_A_sparse(n, a)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    try:
        r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
        if r.success and r.x is not None:
            g = np.maximum(r.x, 0)
            if g.sum() > 1e-15:
                return g
    except Exception:
        pass
    return None

def lp_mirror_descent(a, max_iters, deadline, use_multiplicative=False):
    a = np.maximum(a.astype(np.float64), 1e-10)
    a = a / a.sum()
    current_J = compute_J(a)
    best_J = current_J
    best_a = a.copy()
    no_improve = 0

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        b = fast_conv(a)
        M = float(b.max())
        g = lp_best_response(a, M)
        if g is None:
            break
        g = np.maximum(g, 0.0)
        g_sum = g.sum()
        if g_sum < 1e-10:
            break
        g = g / g_sum

        base_t = 0.3 * np.exp(-0.2 * it / 50.0) + 0.03
        t = min(1.0, base_t)

        best_trial_J = current_J
        best_trial_a = None

        for ls in range(20):
            if use_multiplicative:
                ratio = g / np.maximum(a, 1e-15)
                a_trial = a * np.power(ratio, t)
            else:
                a_trial = (1.0 - t) * a + t * g
            a_trial = np.maximum(a_trial, 0.0)
            s = a_trial.sum()
            if s < 1e-10:
                t *= 0.5
                continue
            a_trial = a_trial / s
            J_trial = compute_J(a_trial)
            if J_trial < best_trial_J - 1e-14:
                best_trial_J = J_trial
                best_trial_a = a_trial.copy()
                break
            t *= 0.6
            if t < 0.001:
                break

            a = best_trial_a
            current_J = best_trial_J
            if current_J < best_J:
                best_J = current_J
                best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1
            if no_improve > 30:
                break

    return best_a, best_J


def smooth_max_conv_obj(x, n, tau):
    """Objective for L-BFGS: smooth max of conv(softmax(x), softmax(x))."""
    dx = x - x.max()
    a = np.exp(dx)
    a = a / a.sum()
    b = fast_conv(a)
    # Log-sum-exp for numerical stability
    bm = b.max()
    sm = np.sum(np.exp(tau * (b - bm)))
    return bm + np.log(sm) / tau

def smooth_max_conv_grad(x, n, tau):
    """Gradient of smooth max conv objective w.r.t. x (log-parameters)."""
    dx = x - x.max()
    a = np.exp(dx)
    s = a.sum()
    a = a / s
    b = fast_conv(a)
    bm = b.max()
    w = np.exp(tau * (b - bm))
    w = w / w.sum()
    
    # Gradient of smooth_max w.r.t. a
    g_a = np.zeros(n)
    for k in range(2 * n - 1):
        if w[k] > 1e-15:
            wk = w[k]
            for i in range(n):
                if 0 <= k - i < n:
                    if 2 * i == k:
                        g_a[i] += wk * a[i]
                    else:
                        g_a[i] += wk * 2 * a[k - i]
    
    # Chain rule: da/dx = a * (I - a/s) = a * (I - a) (since s=1 after normalization)
    # Actually with softmax: a_i = exp(x_i) / sum(exp(x_j))
    # da_i/dx_j = a_i * (delta_ij - a_j)
    # So grad_x = a * (g_a - a * sum(g_a * a))
    g_x = a * (g_a - a * np.dot(g_a, a))
    return g_x


def lbb_smooth(x0, n, tau, deadline, maxiter=500):
    """L-BFGS on smoothed objective in log-space."""
    def obj(x):
        return smooth_max_conv_obj(x, n, tau)
    
    def grad(x):
        return smooth_max_conv_grad(x, n, tau)
    
    try:
        res = minimize(obj, x0, jac=grad, method='L-BFGS-B',
                       options={'maxiter': maxiter, 'max_fun': 50000,
                                'ftol': 1e-18, 'gtol': 1e-15})
        dx = res.x - res.x.max()
        a = np.exp(dx)
        a = a / a.sum()
        return a

"""
Optimizer combining LP mirror descent, coordinate descent, and smoothed L-BFGS.
Exploits 1D convexity structure of coordinate updates and uses second-order
methods for final refinement.

import numpy as np
from scipy.optimize import linprog, minimize, minimize_scalar
from scipy.sparse import csr_matrix
import time


def fast_conv(a):
    """Full self-convolution via FFT."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]


def compute_J(a):
    """Compute J(a) = 2*n*max(conv(a,a))/(sum(a))^2."""
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))


def build_A_sparse(n, a):
    """Build sparse matrix A where A[k,j] = a[k-j] for valid j."""
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        j_min = max(0, k - n + 1)
        j_max = min(n - 1, k)
        for j in range(j_min, j_max + 1):
            rows.append(k)
            cols.append(j)
            vals.append(a[k - j])
    return csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))


def lp_best_response(a, M):
    """Solve LP: max sum(g) s.t. A@g <= M*1, g >= 0."""
    n = len(a)
    A = build_A_sparse(n, a)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    try:
        r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
        if r.success and r.x is not None:
            g = np.maximum(r.x, 0)
            if g.sum() > 1e-15:
                return g
    except Exception:
        pass
    return None


def lp_mirror_descent(a, max_iters, deadline, use_multiplicative=False):
    """LP mirror descent with adaptive line search."""
    a = np.maximum(a.astype(np.float64), 1e-10)
    a = a / a.sum()

    best_J = current_J

    no_improve = 0

    for it in range(max_iters):
        if time.monotonic() > deadline:

        b = fast_conv(a)
        M = float(b.max())
        g = lp_best_response(a, M)
        if g is None:
            break
        g = np.maximum(g, 0.0)
        g_sum = g.sum()
        if g_sum < 1e-10:
            break
        g = g / g_sum

        base_t = 0.3 * np.exp(-0.25 * it / 80.0) + 0.02
        t = min(1.0, base_t)

        best_trial_J = current_J
        best_trial_a = None

        for ls in range(20):
            if use_multiplicative:
                ratio = g / np.maximum(a, 1e-15)
                a_trial = a * np.power(ratio, t)
            else:

            a_trial = np.maximum(a_trial, 0.0)
            s = a_trial.sum()
            if s < 1e-10:

                continue
            a_trial = a_trial / s

import numpy as np
from scipy.optimize import linprog, minimize, minimize_scalar
from scipy.sparse import csr_matrix
import time


def fast_conv(a):
    """Full self-convolution via FFT."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]


def compute_J(a):
    """Compute J(a) = 2*n*max(conv(a,a))/(sum(a))^2."""
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))


def build_A_sparse(n, a):
    """Build sparse matrix A where A[k,j] = a[k-j] for valid j."""
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        j_min = max(0, k - n + 1)
        j_max = min(n - 1, k)
        for j in range(j_min, j_max + 1):
            rows.append(k)
            cols.append(j)
            vals.append(a[k - j])
    return csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))


def lp_best_response(a, M):
    """Solve LP: max sum(g) s.t. A@g <= M*1, g >= 0."""
    n = len(a)
    A = build_A_sparse(n, a)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    try:
        r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
        if r.success and r.x is not None:
            g = np.maximum(r.x, 0)
            if g.sum() > 1e-15:
                return g
    except Exception:
        pass
    return None


def lp_mirror_descent(a, max_iters, deadline, use_multiplicative=False):
    """LP mirror descent with adaptive line search."""
    a = np.maximum(a.astype(np.float64), 1e-10)
    a = a / a.sum()
    current_J = compute_J(a)
    best_J = current_J
    best_a = a.copy()
    no_improve = 0

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        b = fast_conv(a)
        M = float(b.max())
        g = lp_best_response(a, M)
        if g is None:
            break
        g = np.maximum(g, 0.0)
        g_sum = g.sum()
        if g_sum < 1e-10:
            break
        g = g / g_sum

        base_t = 0.25 * np.exp(-0.2 * it / 60.0) + 0.03
        t = min(1.0, base_t)
        best_trial_J = current_J
        best_trial_a = None

        for ls in range(20):
            if use_multiplicative:
                ratio = g / np.maximum(a, 1e-15)
                a_trial = a * np.power(ratio, t)
            else:
                a_trial = (1.0 - t) * a + t * g
            a_trial = np.maximum(a_trial, 0.0)
            s = a_trial.sum()
            if s < 1e-10:
                t *= 0.5
                continue
            a_trial = a_trial / s
            J_trial = compute_J(a_trial)
            if J_trial < best_trial_J - 1e-14:
                best_trial_J = J_trial
                best_trial_a = a_trial.copy()
                break
            t *= 0.6
            if t < 0.001:
                break

        if best_trial_a is not None:
            a = best_trial_a
            current_J = best_trial_J
            if current_J < best_J:
                best_J = current_J
                best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1
            if no_improve > 25:
                break

    return best_a, best_J


def coordinate_descent(a, max_sweeps, deadline, verbose=True):
    """Coordinate descent: optimize one element at a time via 1D bounded search."""
    a = a.copy()
    n = len(a)
    best_J = compute_J(a)
    best_a = a.copy()

    for sweep in range(max_sweeps):
        if time.monotonic() > deadline:
            break

        for i in range(n):
            if time.monotonic() > deadline:
                break

            a_i = a[i]
            if a_i < 1e-12:
                # Try setting to small positive value
                a[i] = 1e-6
                s = a.sum()
                if s > 1e-10:
                    J_test = compute_J(a / s)
                    if J_test < best_J:
                        a = a / s
                        best_J = J_test
                        best_a = a.copy()
                    else:
                        a[i] = 0.0
                continue

            t_hi = max(3.0 * a_i, 0.05)

            def obj_1d(t_val):
                a[i] = t_val
                s = a.sum()
                if s < 1e-10:
                    return 1e10
                b = fast_conv(a / s)
                return b.max()

            try:
                res = minimize_scalar(obj_1d, bounds=(0.0, t_hi), method='bounded',
                                      options={'xatol': 1e-8, 'maxiter': 40})
                a[i] = max(0.0, res.x)
            except Exception:
                a[i] = a_i

        s = a.sum()
        if s > 1e-10:
            a = a / s
        J_now = compute_J(a)
        if J_now < best_J - 1e-14:
            best_J = J_now
            best_a = a.copy()
            if verbose and sweep % 10 == 0:
                print(f"      CD sweep {sweep}: J={best_J:.8f}", flush=True)
        elif J_now >= best_J - 1e-14 and sweep > 20:
            # Check if we're stuck
            if sweep % 10 == 0:
                a = best_a.copy()
                a = a / a.sum()

    return best_a, best_J


def smooth_max_obj(x, n, tau):
    """Smooth max of conv(softmax(x), softmax(x)) using log-sum-exp."""
    dx = x - x.max()
    a = np.exp(dx)
    s = a.sum()
    if s < 1e-30:
        return 1e10
    a = a / s
    b = fast_conv(a)
    bm = b.max()
    # Log-sum-exp for stability
    shifted = tau * (b - bm)
    max_shifted = shifted.max()
    return float(bm + (max_shifted + np.log(np.sum(np.exp(shifted - max_shifted)))) / tau)


def smooth_max_grad(x, n, tau):
    """Gradient of smooth max objective w.r.t. x (log-parameters)."""
    dx = x - x.max()
    a = np.exp(dx)
    s = a.sum()
    if s < 1e-30:
        return np.zeros(n)
    a = a / s
    b = fast_conv(a)
    bm = b.max()
    shifted = tau * (b - bm)
    max_shifted = shifted.max()
    w = np.exp(shifted - max_shifted)
    w = w / w.sum()

    # Gradient of smooth_max w.r.t. a
    g_a = np.zeros(n)
    n_b = len(b)
    for k in range(n_b):
        wk = w[k]
        if wk < 1e-14:
            continue
        for i in range(n):
            j = k - i
            if 0 <= j < n:
                if 2 * i == k:
                    g_a[i] += wk * a[i]
                else:
                    g_a[i] += 2.0 * wk * a[j]

    # Chain rule for softmax: d(softmax(x)_i)/dx_j = a_i*(delta_ij - a_j)
    # grad_x_j = sum_i g_a_i * a_i * (delta_ij - a_i) = a_j * (g_a_j - sum_i g_a_i * a_i)
    dot = np.dot(g_a, a)
    g_x = a * (g_a - dot * a)
    return g_x


def lbb_smoothed(x0, n, deadline, tau_start=50, tau_end=200):
    """L-BFGS on smoothed max-conv objective in log-space with increasing tau."""
    a_current = None
    for tau in [tau_start, int(tau_start * 2), int(tau_end * 0.7), tau_end]:
        if time.monotonic() > deadline:
            break

        def obj(x):
            return smooth_max_obj(x, n, tau)

        def grad(x):
            return smooth_max_grad(x, n, tau)

        try:
            res = minimize(obj, x0, jac=grad, method='L-BFGS-B',
                           options={'maxiter': 200, 'max_fun': 30000,
                                    'ftol': 1e-16, 'gtol': 1e-14})
            dx = res.x - res.x.max()
            a_candidate = np.exp(dx)
            a_candidate = a_candidate / a_candidate.sum()
            J_candidate = compute_J(a_candidate)

            if a_current is None or J_candidate < compute_J(a_current):
                a_current = a_candidate
            x0 = res.x
        except Exception:
            break

    return a_current


def resample_seq(src, dest_n):
    """Resample 1D sequence to new length."""
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)


def generate_inits(n, parent=None, rng=None):

    inits.append(np.maximum(1.0 - 0.5 * np.abs(2 * t - 1), 0.1))
    inits[-1] = inits[-1] / inits[-1].sum()

    # Edge-concentrated
    edge = np.zeros(n)
    decay = int(n * 0.15)
    for i in range(decay):
        edge[i] = np.exp(-i / decay * 3)
        edge[n - 1 - i] = np.exp(-i / decay * 3)
    edge[n // 3:2 * n // 3] += 0.3
    inits.append(edge / edge.sum())

    # Random smooth
    k = max(5, n // 10)

    smooth = np.convolve(rng.uniform(0, 1, n), kernel, mode='same')
    inits.append(np.maximum(smooth + 0.1, 0.01))
    inits[-1] = inits[-1] / inits[-1].sum()

    result = []
    for a in inits:
        a = np.maximum(np.asarray(a, dtype=np.float64), 0.001)
        s = a.sum()
        if s > 1e-6:
            result.append(a / s)
    return result

            x0 = np.log(np.maximum(best_a, 1e-10))
            x0 = x0 - x0.mean()
            a_final = lbb_smoothed(x0, best_n, deadline - 2, tau_start=100, tau_end=500)
            if a_final is not None:
                J_final3 = compute_J(a_final)
                tracker.update(a_final)
                print(f"  Final L-BFGS: J={J_final3:.8f}", flush=True)
                if J_final3 < compute_J(best_a):
                    best_a = a_final.copy()

    print(f"  Phase 7 done ({time.monotonic()-t0:.1f}s)", flush=True)

    # Final output
    final_a = tracker.best if tracker.best is not None else best_a
    print(f"\n{'='*50}", flush=True)
    print(f"FINAL: J={tracker.best_score:.8f} n={final_a.size}", flush=True)
    print(f"Total time: {time.monotonic()-tracker.t_start:.0f}s", flush=True)
    print(f"{'='*50}", flush=True)

    return [float(x) for x in final_a]
