import numpy as np
from scipy.optimize import minimize, linprog
from scipy.fft import rfft, irfft

def compute_conv(a):
    """Compute linear convolution a*a using FFT."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    conv = irfft(fa * fa, size)
    return conv[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    conv = compute_conv(a)
    max_b = float(np.max(conv))
    sum_a = float(np.sum(a))
    if sum_a < 0.01:
        return np.inf
    return 2.0 * n * max_b / (sum_a * sum_a)


def fft_convolve(a, b):
    """Linear convolution of a and b using FFT."""
    n, m = len(a), len(b)
    size = 1
    while size < n + m - 1:
        size <<= 1
    fa = rfft(a, size)
    fb = rfft(b, size)
    return irfft(fa * fb, size)[:n + m - 1]


def softmax(z):
    """Numerically stable softmax."""
    z = z - np.max(z)
    e = np.exp(z)
    s = np.sum(e)
    if s < 1e-300:
        idx = int(np.argmax(z))
        out = np.zeros_like(z)
        out[idx] = 1.0
        return out
    return e / s


def soft_max_obj_and_grad_z(z, tau):
    """Compute soft-max of conv(softmax(z), softmax(z)) and grad w.r.t. z."""
    n = len(z)
    a = softmax(z)
    conv = compute_conv(a)
    cmax = float(np.max(conv))
    shifted = np.clip((conv - cmax) / tau, -500.0, 500.0)
    exp_conv = np.exp(shifted)
    Z = float(np.sum(exp_conv))
    soft_max_val = cmax + tau * np.log(Z)
    w = exp_conv / Z

    w_rev = w[::-1]
    conv_fw = fft_convolve(a, w_rev)
    f1 = conv_fw[n - 1:2 * n - 1][::-1]
    grad_a = 2.0 * f1

    mean_grad = float(np.dot(a, grad_a))
    grad_z = a * (grad_a - mean_grad)

    return soft_max_val, grad_z


def gradient_from_weights(a, w):
    """Compute grad of sum_k w_k*(a*a)[k] w.r.t. a. Returns grad_a of length n."""
    n = len(a)
    w_rev = w[::-1]
    conv_fw = fft_convolve(a, w_rev)
    return 2.0 * conv_fw[n - 1:2 * n - 1][::-1]


def build_toeplitz(a):
    """Build Toeplitz matrix A where (A@g)[k] = (conv(a,g))[k]."""
    n = len(a)
    A = np.zeros((2 * n - 1, n), dtype=np.float64)
    for j in range(n):
        k_start = j
        k_end = min(j + n, 2 * n - 1)
        A[k_start:k_end, j] = a[:k_end - k_start]
    return A


def solve_lp_dual(a, time_limit=20):
    """Solve dual LP: max nu s.t. sum(y)=1, y>=0, sum_k y_k*A[k,j]>=nu for all j.
    Returns (y_dual, nu_opt) or (None, None)."""
    n = len(a)
    A = build_toeplitz(a)
    # Dual: max nu s.t. A^T @ y >= nu * 1, sum(y) = 1, y >= 0
    # Variables: y (2n-1 vars) + nu (1 var)
    # min -nu (since linprog minimizes)
    # s.t. -A^T @ y + nu * 1 <= 0  (i.e., A^T @ y >= nu)
    #      sum(y) = 1
    #      y >= 0, nu free (but will be >= 0 at optimum)
    num_y = 2 * n - 1
    c = np.zeros(num_y + 1)
    c[-1] = -1.0  # minimize -nu = maximize nu
    # Inequality: -A^T @ y + nu <= 0
    A_ub = np.zeros((n, num_y + 1))
    A_ub[:, :num_y] = -A.T
    A_ub[:, num_y] = 1.0
    b_ub = np.zeros(n)
    # Equality: sum(y) = 1
    A_eq = np.zeros((1, num_y + 1))
    A_eq[0, :num_y] = 1.0
    b_eq = np.array([1.0])
    bounds = [(0.0, None)] * num_y + [(None, None)]
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': time_limit})
        if res.status == 0:
            y = np.maximum(res.x[:num_y], 0)
            s = np.sum(y)
            if s > 1e-10:
                y = y / s
                nu = float(res.x[num_y])
                return y, nu
    except Exception:
        pass
    return None, None


def solve_lp_primal(a, M, time_limit=20):
    """Solve LP: max sum(g) s.t. (a*g)[k] <= M for all k, g >= 0.
    Returns (g_normalized, sum_g) or (None, None)."""
    n = len(a)
    A = build_toeplitz(a)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': time_limit})
        if res.status == 0:
            g = np.maximum(res.x, 0)
            s = float(np.sum(g))
            if s > 1e-10:
                return g / s, s
    except Exception:
        pass
    return None, None

def optimize_lfbs(a_init, tau_list, max_iter=400, time_budget=30.0):
    """L-BFGS-B with soft-max parametrization and tau annealing."""

            break

            return soft_max_obj_and_grad_z(z_val, tau)

        try:
            result = minimize(func_and_grad, z, jac=True, method='L-BFGS-B',
                              options={'maxiter': max_iter, 'ftol': 1e-15,
                                       'gtol': 1e-10, 'maxfun': 15000})

def subgradient_optimize(a, max_iters=5000, time_budget=25.0, tau_start=5.0, tau_end=50.0):
    """Smoothed subgradient with adaptive step size."""
    n = len(a)
    a = a.copy().astype(np.float64)
    s = np.sum(a)
    if s < 1e-10:
        a = np.ones(n) / n
    else:
        a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    step = 0.15
    stall = 0

    for it in range(max_iters):
        if time.monotonic() - start > time_budget:
            break

        frac = it / max(1, max_iters)
        tau = tau_start + (tau_end - tau_start) * frac

        conv = compute_conv(a)
        max_conv = float(np.max(conv))
        shifted = np.clip(tau * (conv - max_conv) / max_conv, -300, 300)
        weights = np.exp(shifted)
        w_sum = np.sum(weights)
        if w_sum < 1e-300:
            idx = int(np.argmax(conv))
            weights = np.zeros(len(conv))
            weights[idx] = 1.0
        else:
            weights = weights / w_sum

        grad_a = gradient_from_weights(a, weights)
        grad_centered = grad_a - np.mean(grad_a)
        grad_norm = np.linalg.norm(grad_centered)
        if grad_norm < 1e-15:
            break
        grad_centered = grad_centered / grad_norm

        a_new = a - step * grad_centered
        a_new = np.maximum(a_new, 0)
        s_new = np.sum(a_new)
        if s_new > 1e-10:
            a_new = a_new / s_new
        else:
            a_new = a.copy()

        a = a_new
        obj = compute_obj(a)

        if obj < best_obj - 1e-12:
            best_obj = obj
            best_a = a.copy()
            stall = 0
        else:
            stall += 1
            if stall > 150:
                step *= 0.7
                if step < 1e-12:
                    break
                rng = np.random.default_rng(it * 31 + 7)
                a = a + rng.normal(0, 0.02, n) * step
                a = np.maximum(a, 0)
                s = np.sum(a)
                if s > 1e-10:
                    a = a / s
                stall = 0

    return best_a

def resample_sequence(a, new_n):
    """Resample sequence to new length via interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0.01)

def combined_optimize(a, time_budget, n):
    """Run combined optimization: subgradient -> L-BFGS-B -> LP -> L-BFGS-B -> LP."""
    start = time.monotonic()
    best_a = a.copy().astype(np.float64)
    best_obj = compute_obj(best_a)

    # Phase 1: Quick subgradient shaping
    sg_time = min(8.0, time_budget * 0.1)
    if time_budget - (time.monotonic() - start) > sg_time:
        a_new = subgradient_optimize(best_a, max_iters=int(sg_time * 50),
                                     time_budget=sg_time, tau_start=3.0, tau_end=20.0)
        obj_new = compute_obj(a_new)
        if obj_new < best_obj:
            best_obj = obj_new
            best_a = a_new

    # Phase 2: L-BFGS-B with tau annealing
    remaining = time_budget - (time.monotonic() - start)
    if remaining > 15:
        tau_list = [10.0 / n, 5.0 / n, 2.0 / n, 1.0 / n, 0.5 / n, 0.2 / n, 0.1 / n, 0.05 / n]
        lfbs_time = min(remaining * 0.4, 30.0)
        a_new, obj_new = optimize_lfbs(best_a, tau_list, max_iter=400, time_budget=lfbs_time)
        if obj_new < best_obj:
            best_obj = obj_new
            best_a = a_new

    # Phase 3: LP polishing
    remaining = time_budget - (time.monotonic() - start)
    if remaining > 15 and n <= 800:
        lp_time = min(remaining * 0.35, 45.0)
        a_new = lp_alt_reads(best_a, max_iters=100, time_budget=lp_time)
        obj_new = compute_obj(a_new)
        if obj_new < best_obj:
            best_obj = obj_new
            best_a = a_new

    # Phase 4: Final L-BFGS-B with tiny tau
    remaining = time_budget - (time.monotonic() - start)
    if remaining > 10:
        tau_list = [0.3 / n, 0.1 / n, 0.03 / n, 0.01 / n]
        lfbs_time = min(remaining * 0.3, 20.0)
        a_new, obj_new = optimize_lfbs(best_a, tau_list, max_iter=500, time_budget=lfbs_time)
        if obj_new < best_obj:
            best_obj = obj_new
            best_a = a_new

    # Phase 5: Final LP
    remaining = time_budget - (time.monotonic() - start)
    if remaining > 10 and n <= 800:
        lp_time = min(remaining * 0.5, 30.0)
        a_new = lp_alt_reads(best_a, max_iters=50, time_budget=lp_time)
        obj_new = compute_obj(a_new)
        if obj_new < best_obj:
            best_obj = obj_new
            best_a = a_new

    return best_a, best_obj


def propose_candidate(seed=42, budget_s=1500):
    """
    Main search function. Explores multiple sequence lengths using combined
    LP + L-BFGS-B + subgradient optimization with aggressive tau annealing
    and dual LP exploitation.
    """
    start_time = time.monotonic()
    deadline = start_time + float(budget_s) - 30.0

    best_seq = None
    best_score = np.inf

    # Load parent
    parent, parent_obj = try_parent()
    if parent is not None and parent_obj < best_score:
        best_score = parent_obj
        best_seq = parent.copy()

    # Sequence lengths to explore
    n_values = [100, 200, 300, 400, 500, 600, 700, 800]
    if best_seq is not None:
        np_ = len(best_seq)
        extra = [np_, max(50, np_ - 100), min(1000, np_ + 100), min(1000, np_ + 200)]
        n_values = sorted(set(n_values + extra))

    print(f"Exploring n values: {n_values}")
    print(f"Initial best: {best_score:.7f}")

    # Phase 1: Exploration
    candidates = []

    for n in n_values:
        if time.monotonic() > deadline - 120:
            break
        remaining = deadline - time.monotonic()
        if remaining < 15:
            break

        # Time allocation
        if n <= 300:
            alloc = min(40.0, remaining * 0.12)
        elif n <= 600:
            alloc = min(55.0, remaining * 0.14)
        else:
            alloc = min(65.0, remaining * 0.15)
        alloc = max(15.0, alloc)

        print(f"\nN={n}: {remaining:.0f}s remaining, alloc={alloc:.1f}s")

        n_inits = max(2, int(alloc / 18))
        n_inits = min(n_inits, 5)

        for init_idx in range(n_inits):
            if time.monotonic() > deadline - 120:
                break
            remaining_init = deadline - time.monotonic()
            if remaining_init < 12:
                break

            # Generate initial point
            if init_idx == 0 and best_seq is not None:
                a = resample_sequence(best_seq, n)
            else:
                a = random_init(n, seed + n * 104729 + init_idx * 79193)

            # Run combined optimization
            init_time = min(alloc / n_inits - 2, remaining_init - 10)
            if init_time < 8:
                init_time = 8
            a_opt, obj_opt = combined_optimize(a, init_time, n)

            print(f"  init{init_idx}: n={n}, obj={obj_opt:.7f}")

            if obj_opt < best_score:
                best_score = obj_opt
                best_seq = a_opt.copy()
                print(f"  ** UPDATE: n={n}, obj={best_score:.7f} **")

            candidates.append((a_opt.copy(), obj_opt, n))

    # Phase 2: Polish top candidates
    print(f"\n=== POLISHING PHASE ===")
    print(f"Best before polish: {best_score:.7f}")

    candidates.sort(key=lambda x: x[1])

    n_polish = min(3, len(candidates))
    for ci in range(n_polish):
        if time.monotonic() > deadline - 50:
            break
        remaining = deadline - time.monotonic()
        if remaining < 20:
            break

        a_cand = candidates[ci][0]
        n_cand = candidates[ci][2]
        print(f"  Polishing candidate {ci} (n={n_cand}, obj={candidates[ci][1]:.7f}), {remaining:.0f}s left")

        polish_budget = min(remaining * 0.4, 90.0)
        a_pol, obj_pol = polish_sequence(a_cand, time_budget=polish_budget, verbose=True)

        if obj_pol < best_score:
            best_score = obj_pol
            best_seq = a_pol.copy()
            print(f"  ** UPDATE after polish: obj={best_score:.7f} **")
        else:
            # Try perturbations
            remaining = deadline - time.monotonic()
            if remaining > 20:
                rng = np.random.default_rng(ci * 555 + 999)
                for ptry in range(4):
                    if time.monotonic() > deadline - 30:
                        break
                    scale = rng.uniform(0.03, 0.15)
                    a_pert = a_cand * np.exp(rng.normal(0, scale, n_cand))
                    a_pert = np.maximum(a_pert, 0)
                    s = np.sum(a_pert)
                    if s < 1e-10:
                        continue
                    a_pert = a_pert / s
                    # Quick optimization from perturbed point
                    pert_time = min(15.0, remaining - 20)
                    if pert_time > 5:
                        a_pert, obj_pert = combined_optimize(a_pert, pert_time, n_cand)
                    else:
                        obj_pert = compute_obj(a_pert)
                    if obj_pert < best_score:
                        best_score = obj_pert
                        best_seq = a_pert.copy()
                        print(f"  ** UPDATE (perturb {ptry}): obj={best_score:.7f} **")

    # Phase 3: Final intensive polish
    if best_seq is not None and time.monotonic() < deadline - 15:
        remaining = deadline - time.monotonic()
        n_best = len(best_seq)
        print(f"\n=== FINAL POLISH: n={n_best}, {remaining:.0f}s ===")

        a_cur = best_seq.copy()
        for final_round in range(5):
            if time.monotonic() > deadline - 10:
                break
            remaining = deadline - time.monotonic()
            if remaining < 12:
                break

            # LP
            if n_best <= 800:
                lp_time = min(remaining * 0.25, 25.0)
                a_new = lp_alt_reads(a_cur, max_iters=60, time_budget=lp_time)
                obj_new = compute_obj(a_new)
                if obj_new < best_score:
                    best_score = obj_new
                    best_seq = a_new.copy()
                    a_cur = a_new.copy()
                    print(f"  Final LP {final_round}: {best_score:.7f}")

            # L-BFGS-B tiny tau
            remaining = deadline - time.monotonic()
            if remaining > 10:
                tau_list = [0.2 / n_best, 0.05 / n_best, 0.02 / n_best, 0.01 / n_best]
                lfbs_time = min(remaining * 0.2, 18.0)
                a_new, obj_new = optimize_lfbs(a_cur, tau_list, max_iter=500, time_budget=lfbs_time)
                if obj_new < best_score:
                    best_score = obj_new
                    best_seq = a_new.copy()
                    a_cur = a_new.copy()
                    print(f"  Final L-BFGS {final_round}: {best_score:.7f}")

            # Subgradient
            remaining = deadline - time.monotonic()
            if remaining > 8:
                sg_time = min(remaining * 0.15, 12.0)
                a_new = subgradient_optimize(a_cur, max_iters=int(sg_time * 50),
                                             time_budget=sg_time,
                                             tau_start=10.0, tau_end=40.0)
                obj_new = compute_obj(a_new)
                if obj_new < best_score:
                    best_score = obj_new
                    best_seq = a_new.copy()
                    a_cur = a_new.copy()
                    print(f"  Final SG {final_round}: {best_score:.7f}")

    # Final validation
    if best_seq is None:
        best_seq = np.ones(200)
        best_score = compute_obj(best_seq)

    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    val = evaluate_sequence(result)
    if val < np.inf:
        best_score = val
    else:
        result = [1.0] * 100
        best_score = 2.0

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_score:.7f}, time={elapsed:.1f}s ===")

    return result
