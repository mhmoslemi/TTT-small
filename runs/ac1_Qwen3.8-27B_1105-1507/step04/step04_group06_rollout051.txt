_A_cache = {}

def fast_conv(a):
    """Full self-convolution via FFT."""
    n = len(a)
    s = 1
    while s < 2 * n - 1:
        s <<= 1
    fa = np.fft.rfft(a, s)
    return np.fft.irfft(fa * fa, s)[:2 * n - 1]

def compute_J(a):
    """Compute J(a) = 2*n*max(conv(a,a))/(sum(a))^2."""
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))

def get_conv_matrix_structure(n):
    """Pre-compute (row, col) for the convolution constraint matrix."""
    if n not in _A_cache:
        rows = []
        cols = []
        for k in range(2 * n - 1):
            jl = max(0, k - n + 1)
            jh = min(n - 1, k)
            for j in range(jl, jh + 1):
                rows.append(k)
                cols.append(j)
        _A_cache[n] = (np.array(rows, dtype=np.int32),
                        np.array(cols, dtype=np.int32))
    return _A_cache[n]

def build_conv_matrix(a, active_lags=None):
    """Build sparse matrix (or just the active subset) for LP."""
    n = len(a)
    if active_lags is not None:
        rows_arr = np.array(active_lags, dtype=np.int32)
        all_rows, all_cols = get_conv_matrix_structure(n)
        # Find indices for active lags
        mask = np.zeros(2 * n - 1, dtype=bool)
        mask[active_lags] = True
        idx = np.where(mask)[0]
        rows_arr_subset = all_rows[idx]
        cols_arr_subset = all_cols[idx]
        vals = a[rows_arr_subset - cols_arr_subset]
        return csr_matrix((vals, (rows_arr_subset - min(active_lags), cols_arr_subset)),
                          shape=(len(active_lags), n))
    else:
        rows_arr, cols_arr = get_conv_matrix_structure(n)
        vals = a[rows_arr - cols_arr]
        return csr_matrix((vals, (rows_arr, cols_arr)), shape=(2 * n - 1, n))

    return None

def F_and_grad(a, beta):
    """Compute smooth objective and gradient."""
    n = len(a)
    s = float(a.sum())
    if s < 1e-10:
        return 1e10, np.zeros(n)
    g_val, grad_g = smooth_max_conv_and_grad(a, beta)
    F_val = 2.0 * n * g_val / (s * s)
    grad_F = 2.0 * n * (s * grad_g - 2.0 * g_val) / (s ** 3)
    return F_val, grad_F

def lbfgs_with_annealing(a0, betas, deadline):
    """L-BFGS-B with beta annealing."""
    a = np.maximum(a0.astype(np.float64), 1e-8)
    n = len(a)
    
    for beta, max_iters in betas:
        if time.monotonic() > deadline:
            break
        
        def func(x):
            x2 = np.maximum(x, 0.0)
            f, _ = F_and_grad(x2, beta)
            return f if np.isfinite(f) else 1e10
        
        def jac(x):
            x2 = np.maximum(x, 0.0)
            _, g = F_and_grad(x2, beta)
            g = np.nan_to_num(g, nan=0.0, posinf=0.0, neginf=0.0)
            g[x2 < 1e-12] = 0.0
            return g
        
        bounds = [(0.0, None)] * n
        try:
            res = minimize(func, a.copy(), jac=jac, method='L-BFGS-B',
                           bounds=bounds,
                           options={'maxiter': max_iters, 'max_fun': max_iters * 30,
                                    'gtol': 1e-9, 'ftol': 1e-15})
            if res.fun < 1e10:
                new_a = np.maximum(res.x, 1e-10)
                if new_a.sum() > 1e-6:
                    a = new_a
        except Exception:
            pass
    
    return a

def targeted_peak_shifting(a, rng, n_shifts=5):
    """Target perturbation to reduce specific convolution peaks."""
    n = len(a)
    a = a.copy()
    b = fast_conv(a)
    b_max = b.max()
    
    # Find peaks
    threshold = b_max * 0.95
    peak_lags = np.where(b >= threshold)[0]
    
    for lag in peak_lags[:n_shifts]:
        # The contribution to (a*a)_lag comes from pairs (j, lag-j)
        # Reduce the larger of each pair
        jl = max(0, lag - n + 1)
        jh = min(n - 1, lag)
        for j in range(jl, jh + 1):
            other = lag - j
            if 0 <= other < n:
                # Reduce the larger element slightly
                if a[j] > a[other]:
                    a[j] *= (1 - 0.02 * rng.random())
                else:
                    a[other] *= (1 - 0.02 * rng.random())
    
    a = np.maximum(a, 1e-6)
    return a / a.sum()

def fourier_subspace_optimize(a0, n_modes, deadline, max_iters=200):
    """
    Project a onto first n_modes Fourier modes + DC, then optimize in that subspace.
    """
    n = len(a0)
    a = a0.copy()
    
    # Get frequency components
    A = np.fft.rfft(a)
    # Keep DC + first n_modes
    mask = np.zeros(len(A), dtype=bool)
    mask[:min(n_modes+1, len(A))] = True
    A_sub = A.copy()
    A_sub[~mask] = 0
    a_sub = np.fft.irfft(A_sub, n)
    a_sub = np.maximum(a_sub, 0)
    if a_sub.sum() < 1e-6:
        return a0
    
    # Optimize in the subspace by varying the amplitudes/phases of kept modes
    # Simpler: just do L-BFGS on the full problem but initialized from a_sub
    betas = [(5.0, 30), (20.0, 40), (50.0, 50)]
    a_opt = lbfgs_with_annealing(a_sub, betas, deadline)
    return a_opt

def resample_seq(src, dest_n):
    """Resample 1D sequence to new length."""
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    result = np.interp(t_new, t_old, src)
    return np.maximum(result, 0.0)

def generate_initializations(n, rng, parent=None):
    """Generate diverse initial sequences."""
    inits = []
    t = np.linspace(0, 1, n, endpoint=False)
    
    if parent is not None and parent.size > 0:
        p = resample_seq(parent, n)
        inits.append(p / max(p.sum(), 1e-10))
        p2 = p * (1 + 0.05 * rng.standard_normal(n))
        inits.append(np.maximum(p2, 0.01) / max(np.maximum(p2, 0.01).sum(), 1e-10))
    
    inits.append(np.ones(n) / n)
    inits.append(rng.uniform(0.3, 1.0, n) / n)
    
    # Cosine families
    for f in [2, 3, 4, 5, 6, 8]:
        seq = 1.0 + 0.4 * np.cos(2 * np.pi * f * t)
        inits.append(np.maximum(seq, 0.1) / max(np.maximum(seq, 0.1).sum(), 1e-10))
    
    # Multi-frequency
    seq = 1.0 + 0.25*np.cos(2*np.pi*3*t) + 0.15*np.cos(2*np.pi*7*t)
    inits.append(np.maximum(seq, 0.1) / max(np.maximum(seq, 0.1).sum(), 1e-10))
    
    # Gaussian
    seq = np.exp(-3*(t-0.5)**2) + 0.1
    inits.append(seq / seq.sum())
    
    # Sine
    seq = np.abs(np.sin(np.pi * t)) + 0.2
    inits.append(seq / seq.sum())
    
    # Power law
    seq = (t + 0.05) ** 0.7
    inits.append(seq / seq.sum())
    
    # Exponential
    seq = np.exp(-1.0 * t) + 0.15
    inits.append(seq / seq.sum())
    
    # Symmetric bump
    seq = 1.0 + 0.3 * np.cos(np.pi * t)
    inits.append(np.maximum(seq, 0.1) / max(np.maximum(seq, 0.1).sum(), 1e-10))
    
    return inits

class ScoreTracker:
