        j_max = min(n - 1, k)

            rows.append(k)
            cols.append(j)
            vals.append(a[k-j])
    
    A = csr_matrix((vals, (rows, cols)), shape=(2*n-1, n))
    
    constraints = [A @ g <= M]
    objective = cp.Maximize(cp.sum(g))
    prob = cp.Problem(objective, constraints)
    
    try:

import numpy as np
import time
import cvxpy as cp
from scipy.sparse import csr_matrix
from scipy.fft import next_fast_len

def fast_conv(a):
    n = len(a)
    size = next_fast_len(2*n - 1)
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2*n-1]

def compute_J(a):
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))

def build_A(n, a):
    """Build sparse matrix A where A[k,j] = a[k-j] if 0<=k-j<n."""
    rows, cols, vals = [], [], []
    for k in range(2*n - 1):
        j_min = max(0, k - n + 1)
        j_max = min(n - 1, k)
        for j in range(j_min, j_max + 1):
            rows.append(k)
            cols.append(j)
            vals.append(a[k-j])
    return csr_matrix((vals, (rows, cols)), shape=(2*n-1, n))

def solve_lp(a, M):
    """Solve LP: max sum(g) s.t. A@g <= M, g >= 0. Returns g or None."""
    n = len(a)
    g = cp.Variable(n, nonneg=True)
    A = build_A(n, a)
    constraints = [A @ g <= M]
    prob = cp.Problem(cp.Maximize(cp.sum(g)), constraints)
    try:
        prob.solve(solver=cp.GLOP)
        if g.value is not None:
            return np.maximum(g.value, 0)
    except:
        pass
    try:
        prob.solve(solver=cp.ECOS)
        if g.value is not None:
            return np.maximum(g.value, 0)
    except:
        pass
    try:
        prob.solve(solver=cp.CBC)
        if g.value is not None:
            return np.maximum(g.value, 0)
    except:
        pass
    return None

def mirror_descent(a, max_iters, deadline, use_anderson=True, anderson_m=4):
    """LP mirror descent with optional Anderson acceleration."""
    n = len(a)
    a = np.maximum(a.astype(np.float64), 1e-10)
    a = a / a.sum()
    
    best_a = a.copy()
    best_M = fast_conv(a).max()
    
    # Anderson acceleration
    history_x = [a.copy()]
    history_F = [None]
    
    step_size = 0.5
    
    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        
        g = solve_lp(a, best_M)
        if g is None or g.sum() < 1e-10:
            break
        g = g / g.sum()
        
        if use_anderson and it >= anderson_m:
            # Anderson acceleration
            m = min(anderson_m, it)
            # F(x) = T(x) - x where T is the map
            # We store F values
            F_current = g - a
            valid = all(history_F[i] is not None for i in range(-m, 0))
            if valid:
                F_mat = np.array([history_F[it - m + i] for i in range(m)])
                G = F_mat @ F_mat.T + 1e-10 * np.eye(m)
                kkt = np.zeros((m+1, m+1))
                kkt[:m, :m] = 2 * G
                kkt[:m, m] = 1
                kkt[m, :m] = 1
                rhs = np.zeros(m+1)
                rhs[m] = 1
                try:
                    sol = np.linalg.solve(kkt, rhs)
                    alpha = np.maximum(sol[:m], 0)
                    alpha_sum = alpha.sum()
                    if alpha_sum > 1e-10:
                        alpha = alpha / alpha_sum
                        a = sum(alpha[i] * history_x[it - m + i] for i in range(m))
                        a = np.maximum(a, 1e-10)
                        a = a / a.sum()
                except:
                    pass
        
        # Update: a -> (1-t)*a + t*g
        t = step_size * (0.5 + 0.5 * (1 - it / max(1, max_iters)))
        a_new = (1 - t) * a + t * g
        a_new = np.maximum(a_new, 1e-10)
        a_new = a_new / a_new.sum()
        
        M_new = fast_conv(a_new).max()
        if M_new < best_M - 1e-13:
            best_M = M_new
            best_a = a_new.copy()
        
        a = a_new
        history_x.append(a.copy())
        history_F.append(g - a)
        if len(history_x) > 30:
            history_x = history_x[-30:]
            history_F = history_F[-30:]
        
        # Decay step size
        step_size = max(0.1, 0.5 * (1 - it / max(1, max_iters)))

                    a_new = np.maximum(a_new, 1e-10)
                    a_new = a_new / a_new.sum()
                    a = a_new
            except:
                a = T_x  # fallback to plain Picard
        else:
            a = T_x
        
        X_hist.append(a.copy())
        
        M_new = fast_conv(a).max()
        if M_new < best_M - 1e-13:
            best_M = M_new
            best_a = a.copy()
        
        # Keep history bounded
        if len(X_hist) > 20:
            X_hist = X_hist[-20:]
            F_hist = F_hist[-20:]
    
    return best_a

Now let me also think about the coordinate descent approach more carefully.

For coordinate descent, the key operation is: given all coordinates except a[i], find the optimal a[i].

The objective is max_k (a*a)[k]. For fixed other coordinates, this is:
max_k {C[k] + a[i]*a[k-i]*[0≤k-i<n and k≠2i] * 2 + a[i]^2*[k=2i]}

This is a convex function of a[i] (it's the max of linear/quadratic functions, all convex in a[i] ≥ 0).

To minimize it, we can use the fact that the optimal a[i] is where the "active" constraints are balanced. Specifically, if we sort the constraints by their slope (coefficient of a[i]), the optimum is where the highest constraint crosses the second-highest, etc.

But a simpler approach is to just do a 1D convex optimization:
- The function is convex in a[i] ≥ 0
- Use ternary search or bisection on the derivative

The derivative of the objective w.r.t. a[i] is:
If the max is achieved at k*:
  d/da[i] (a*a)[k*] = 2*a[k*-i] (if k* ≠ 2i) or 2*a[i] (if k* = 2i)

But the max might change as a[i] changes. So we need to track which constraint is active.

A practical approach:
1. Start with a[i] = 0
2. Evaluate the objective
3. Determine the active constraint (which k gives the max)
4. Since the objective is convex and piecewise linear (or has at most one quadratic piece), the optimum is either:
   - At a[i] = 0
   - At a point where two constraints are equal (a "kink")
   - At the point where the derivative of the active constraint is 0 (only possible for the quadratic case, k=2i)

For the piecewise linear case (no k=2i active), the optimum is where two linear constraints cross:
C[k1] + 2*a[i]*a[k1-i] = C[k2] + 2*a[i]*a[k2-i]
a[i] = (C[k2] - C[k1]) / (2*(a[k1-i] - a[k2-i]))

For the quadratic case (k=2i active), the optimum might be where:
C[2i] + a[i]^2 = C[k*] + 2*a[i]*a[k*-i]
This is a quadratic equation in a[i].

This is getting complicated. Let me just use a simple approach: evaluate the objective at a few candidate points and pick the best.

Candidate points for a[i]:
1. a[i] = 0
2. a[i] = current value
3. For each k, the "break-even" point where constraint k becomes active:
   a[i] = (M - C[k]) / (2*a[k-i]) for linear constraints
   a[i] = sqrt(M - C[k]) for quadratic constraint
4. The point that equalizes the top 2-3 constraints

Actually, the simplest approach that works well in practice is:
1. Compute the current max M
2. For each k, compute the maximum a[i] allowed: a[i] ≤ (M - C[k]) / coeff[k]
3. The optimal a[i] is the minimum of these upper bounds (clipped at 0)
