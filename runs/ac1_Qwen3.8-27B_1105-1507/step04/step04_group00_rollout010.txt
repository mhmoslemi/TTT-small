def resample_seq(src, dest_n):
    """Resample 1D sequence to new length via linear interpolation."""
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)

    """Make a sequence symmetric: a[i] = a[n-1-i]."""
    n = len(a)

    return result

            M_val = fast_conv(a).max()
            if M_val < best_M_for_n:
                best_M_for_n = M_val
                best_for_n = a.copy()

        if best_for_n is not None:
            J_val = compute_J(best_for_n)
            print(f"  Phase1 n={n:4d}: J={J_val:.6f} ({time.monotonic()-t_global_start:.0f}s)", flush=True)
            best_per_length[n] = best_for_n.copy()
            update_best(best_for_n)

    print(f"Phase 1 complete at t={time.monotonic()-t_global_start:.0f}s", flush=True)

    # === PHASE 2: LP Mirror Descent (70% of budget) ===
    p2_deadline = t_global_start + 0.90 * float(budget_s)
    print(f"\n{'='*60}", flush=True)
    print(f"PHASE 2: LP Mirror Descent (until t={p2_deadline - t_global_start:.0f}s)", flush=True)
    print(f"{'='*60}", flush=True)

    # Sort candidates by score, try best few
    sorted_lengths = sorted(best_per_length.keys(),
                            key=lambda n: compute_J(best_per_length[n]))

    # Also include parent's length
    if parent is not None:
        pn = parent.size
        if pn not in best_per_length:
            best_per_length[pn] = parent.copy()
            if pn not in sorted_lengths:
                sorted_lengths.append(pn)
                sorted_lengths.sort(key=lambda n: compute_J(best_per_length[n]))

    # Allocate time to top lengths, weighted by inverse score
    top_lengths = sorted_lengths[:min(5, len(sorted_lengths))]
    # Give more time to shorter sequences (faster LP)
    weights = [1.0 / max(0.1, compute_J(best_per_length[n])) * (512.0 / n) for n in top_lengths]
    total_w = sum(weights)
    available_time = max(10.0, p2_deadline - time.monotonic() - 10.0)

    for idx, n in enumerate(top_lengths):
        if time.monotonic() > p2_deadline:
            break

        time_budget = available_time * weights[idx] / total_w
        time_budget = max(15.0, min(600.0, time_budget))
        phase_dl = min(time.monotonic() + time_budget, p2_deadline - 5.0)

        if time.monotonic() >= phase_dl:
            continue

        print(f"\n  Running LP mirror descent on n={n} (budget={time_budget:.0f}s)", flush=True)
        a_start = best_per_length[n].copy()
        a_start = np.maximum(a_start, 1e-10)
        a_start = a_start / a_start.sum()

        a_result, M_result = lp_mirror_descent(a_start, phase_dl, max_iters=500000,
                                               initial_t=0.3, verbose=True)
        J_result = compute_J(a_result)
        print(f"  LP done n={n}: J={J_result:.8f} (M={M_result:.10f})", flush=True)

        best_per_length[n] = a_result.copy()
        update_best(a_result)

        # Also try symmetric version
        a_sym = make_symmetric(a_result)
        a_sym = np.maximum(a_sym, 1e-10)
        a_sym = a_sym / a_sym.sum()
        J_sym = compute_J(a_sym)
        if J_sym < J_result:
            print(f"  Symmetric improved! J_sym={J_sym:.8f}", flush=True)
            best_per_length[n] = a_sym.copy()
            update_best(a_sym)
            a_result = a_sym

    print(f"Phase 2 complete at t={time.monotonic()-t_global_start:.0f}s", flush=True)

    # === PHASE 3: Final Polish (10% of budget) ===
    p3_deadline = deadline
    print(f"\n{'='*60}", flush=True)
    print(f"PHASE 3: Final Polish (until t={p3_deadline - t_global_start:.0f}s)", flush=True)
    print(f"{'='*60}", flush=True)

    if best_overall is not None and time.monotonic() < p3_deadline - 10:
        # Run one more round of LP on the absolute best
        n_best = best_overall.size
        remaining = p3_deadline - time.monotonic() - 5.0
        if remaining > 30:
            print(f"  Final LP polish on n={n_best} (budget={remaining:.0f}s)", flush=True)
            a_pol = lp_mirror_descent(best_overall, time.monotonic() + remaining,
                                      max_iters=200000, initial_t=0.15, verbose=True)
            update_best(a_pol)

        # Try a few random perturbations
        if time.monotonic() < p3_deadline - 10:
            for attempt in range(5):
                if time.monotonic() > p3_deadline - 5:
                    break
                a_pert = best_overall + rng.normal(0, 0.01 * best_overall.max(), best_overall.size)
                a_pert = np.maximum(a_pert, 0.0)
                a_pert = a_pert / a_pert.sum()
                # Quick smooth gradient
                a_pert = smooth_gradient_descent(a_pert, min(time.monotonic() + 20, p3_deadline - 5),
                                                 max_iters=200, tau_init=10, tau_end=80,
                                                 lr_init=0.1, lr_end=0.02)
                update_best(a_pert)

    # Fallback
    if best_overall is None:
        best_overall = np.ones(128)
        best_overall_score = compute_J(best_overall)

    print(f"\n{'='*60}", flush=True)
    print(f"FINAL: J={best_overall_score:.8f}, n={best_overall.size}, "
          f"time={time.monotonic()-t_global_start:.0f}s", flush=True)
    print(f"{'='*60}", flush=True)

    return [float(x) for x in best_overall]
