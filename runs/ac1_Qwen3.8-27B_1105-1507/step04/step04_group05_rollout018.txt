Actually, I realize the multiplicative update with the full gradient computation is O(n^2) per iteration, which is too slow for n=386. Let me use FFT to compute the gradient.

The gradient of b[k] = (a*a)[k] w.r.t. a[j] is a[k-j] (for valid k-j). So the gradient of sum_k w[k]*b[k] w.r.t. a[j] is sum_k w[k]*a[k-j] = (w*a)[j] (correlation/convolution).

So grad[j] = (w * a)[j] where * is the correlation (not convolution). This can be computed with FFT!

```python
def multiplicative_update(a, eta, n_iter=50):
    a = normalize(a.copy())
    n = len(a)
    
    for it in range(n_iter):
        b = fast_conv(a)
        b_max = b.max()
        
        # Soft max weights
        w = np.exp(eta * (b - b_max))
        w /= w.sum()
        
        # Gradient: grad[j] = sum_k w[k] * a[k-j] = (w correlated with a)[j]
        # This is the cross-correlation of w and a
        # (w*a)[j] = sum_k w[k] * conj(a[k-j]) for complex
        # For real: sum_k w[k] * a[k-j]
        # This is the convolution of w_reversed and a, evaluated at position j
        # Or: it's the correlation
        
        # Using FFT:
        s = 1
        while s < 2*n-1:
            s <<= 1
        fw = np.fft.rfft(w, s)
        fa = np.fft.rfft(a, s)
        # Correlation of w and a: ifft(conj(fw) * fa)
        grad = np.fft.irfft(np.conj(fw) * fa, s)[:n]
        
        # Multiplicative update
        a = a * np.exp(-eta * grad)
        a = normalize(a)
    
    return a
```

```python
def spectral_update(a, eta, n_iter):
    """Multiplicative weight update using FFT-based gradient computation.
    Minimizes max(conv(a,a)) by iteratively reducing mass at positions
    that contribute to convolution peaks."""
    a = normalize(a.copy())
    n = len(a)
    s = 1
    while s < 2 * n - 1:
        s <<= 1

    for it in range(n_iter):
        b = fast_conv(a)
        b_max = b.max()
        if b_max < 1e-15:
            break

        # Soft-max weights: emphasize peak positions
        w = np.exp(eta * (b - b_max))
        w_sum = w.sum()
        if w_sum < 1e-15:
            break
        w /= w_sum

        # Gradient: grad[j] = sum_k w[k]*a[k-j] = correlation(w, a)[j]
        fw = np.fft.rfft(w, s)
        fa = np.fft.rfft(a, s)
        grad = np.fft.irfft(np.conj(fw) * fa, s)[:n]

        # Multiplicative update with damping
        a = a * np.exp(-eta * grad)
        a = normalize(a)

    return a

def coordinate_descent_pass(a):
    """One pass of coordinate descent with O(n) update per coordinate."""
    a = normalize(a.copy())
    n = len(a)
    b = fast_conv(a)
    S = a.sum()  # = 1 after normalization
    current_J = 2.0 * n * b.max()

    # Random order
    indices = np.arange(n)
    np.random.shuffle(indices)

    for i in indices:
        old_val = a[i]
        if old_val < 1e-12:
            # Try setting to a small value
            candidates = [S / n * 0.5, S / n, S / (2 * n)]
        else:
            candidates = [0.0, old_val * 0.5, old_val * 1.5, S / n,
                          old_val * 0.7, old_val * 1.3, S / (n * 0.5)]

        best_val = old_val
        best_J = current_J

        for new_val in candidates:
            new_val = max(0.0, new_val)
            delta = new_val - old_val
            if abs(delta) < 1e-14:
                continue

            # Compute new max convolution efficiently
            # Only positions k in [i, i+n-1] change
            unchanged_max = 0.0
            for k in range(2 * n - 1):
                if k < i or k > i + n - 1:
                    if b[k] > unchanged_max:
                        unchanged_max = b[k]

            changed_max = 0.0
            for k in range(max(0, i), min(2 * n - 1, i + n)):
                ki = k - i
                if 0 <= ki < n:
                    if k == 2 * i:
                        new_ck = b[k] - old_val * old_val + new_val * new_val
                    else:
                        new_ck = b[k] - 2.0 * old_val * a[ki] + 2.0 * new_val * a[ki]
                    if new_ck > changed_max:
                        changed_max = new_ck

            new_max = max(unchanged_max, changed_max)
            new_S = S  # sum unchanged (we're not renormalizing per-step)
            new_J = 2.0 * n * new_max / (new_S * new_S)

            if new_J < best_J:
                best_J = new_J
                best_val = new_val

        if best_val != old_val:
            delta = best_val - old_val
            a[i] = best_val
            # Update convolution
            for k in range(max(0, i), min(2 * n - 1, i + n)):
                ki = k - i
                if 0 <= ki < n:
                    if k == 2 * i:
                        b[k] += best_val * best_val - old_val * old_val
                    else:
                        b[k] += 2.0 * delta * a[ki]
            current_J = best_J

    return a, current_J

    a = normalize(a.copy())
    current_J = compute_J(a)
    tracker.update(a)

    for p in range(n_passes):
        if time.monotonic() > deadline:

        a, new_J = coordinate_descent_pass(a)
        if new_J < current_J - 1e-14:
            current_J = new_J
            a = normalize(a)
            tracker.update(a)
            if p % 5 == 0:
                print(f"    CD pass {p}: J={current_J:.8f}", flush=True)
        else:
            if p > 3:
                break

    return a


def resample_seq(src, dest_n):
    """Resample 1D sequence to new length via linear interpolation."""
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)


def optimize_at_length(a, deadline, tracker, n_spectral=100, n_lp=100, n_cd=10):
    """Full optimization pipeline at a given length."""
    a = normalize(a.copy())

    # Phase A: Spectral update (fast)
    t_deadline_a = min(deadline, time.monotonic() + 60)
    if time.monotonic() < t_deadline_a:
        a, _ = spectral_update_adaptive(a, 2.0, t_deadline_a, tracker)

    # Phase B: LP mirror descent
    t_deadline_b = min(deadline, time.monotonic() + 120)
    if time.monotonic() < t_deadline_b:
        a = lp_mirror_descent(a, t_deadline_b, tracker, max_iters=n_lp)

    # Phase C: Spectral refinement
    t_deadline_c = min(deadline, time.monotonic() + 30)
    if time.monotonic() < t_deadline_c:
        a, _ = spectral_update_adaptive(a, 1.0, t_deadline_c, tracker)

    # Phase D: Coordinate descent
    t_deadline_d = min(deadline, time.monotonic() + 60)
    if time.monotonic() < t_deadline_d:
        a = coordinate_descent(a, t_deadline_d, tracker, n_passes=n_cd)

    # Phase E: Final LP
    if time.monotonic() < deadline - 5:
        a = lp_mirror_descent(a, deadline - 2, tracker, max_iters=30)

    return a


class ScoreTracker:
    def __init__(self):
        self.best = None
        self.best_score = float('inf')
        self.t_start = time.monotonic()

    def update(self, a):
        if a is None:
            return
        a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
        if a.size == 0 or a.sum() < 0.01:
            return
        score = float(evaluate_sequence(a.tolist()))
        if np.isfinite(score) and score < self.best_score:
            self.best = a.copy()
            self.best_score = score
            elapsed = time.monotonic() - self.t_start
            print(f"  * NEW BEST: J={score:.8f} n={a.size} t={elapsed:.1f}s", flush=True)

    def report(self, label=""):
        if self.best is not None:
            print(f"  {label} best: J={self.best_score:.8f} n={self.best.size}", flush=True)


def propose_candidate(seed=42, budget_s=1500, **kwargs):
    """
    Main search function combining spectral updates, LP mirror descent,
    coordinate descent, and multi-length exploration.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + float(budget_s) - 15.0
    tracker = ScoreTracker()

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        if p.ndim == 1 and p.size > 0 and np.all(np.isfinite(p)):
            parent = np.maximum(p, 0.0)
            tracker.update(parent)
            print(f"Parent loaded: n={parent.size}, J={tracker.best_score:.8f}", flush=True)
    except (NameError, AttributeError):
        pass

    if parent is None:
        parent = np.ones(384)
        tracker.update(parent)
        print(f"No parent, using uniform n=384, J={tracker.best_score:.8f}", flush=True)

    n_parent = parent.size

    # === Phase 1: Intensive optimization on parent length ===
    print(f"\n{'='*50} Phase 1: Optimize parent (n={n_parent}) {'='*50}", flush=True)
    t0 = time.monotonic()
    a_best = optimize_at_length(parent, deadline - 400, tracker,
                                 n_spectral=200, n_lp=200, n_cd=15)
    tracker.report(f"Phase 1: {time.monotonic()-t0:.1f}s")

    # === Phase 2: Explore nearby lengths ===
    print(f"\n{'='*50} Phase 2: Length Exploration {'='*50}", flush=True)
    t0 = time.monotonic()

    # Try lengths around the parent
    candidate_lengths = []
    for delta in [-8, -6, -4, -2, 2, 4, 6, 8, 10, -10]:
        nl = n_parent + delta
        if 150 <= nl <= 800 and nl not in candidate_lengths:
            candidate_lengths.append(nl)

    # Also try some strategic lengths
    for nl in [256, 300, 350, 400, 450, 500, 512, 600]:
        if 150 <= nl <= 800 and nl not in candidate_lengths:
            candidate_lengths.append(nl)

    # Sort by proximity to parent
    candidate_lengths.sort(key=lambda x: abs(x - n_parent))

    for nl in candidate_lengths[:10]:
        if time.monotonic() > deadline - 200:
            break

        a_resampled = resample_seq(tracker.best, nl)
        a_resampled = np.maximum(a_resampled, 0.001)
        a_resampled = normalize(a_resampled)

        # Quick optimization
        t_sub = min(deadline - 200, time.monotonic() + 60)
        a_trial = optimize_at_length(a_resampled, t_sub, tracker,
                                      n_spectral=50, n_lp=50, n_cd=5)
        print(f"    Length {nl}: J={tracker.best_score:.8f}", flush=True)

    tracker.report(f"Phase 2: {time.monotonic()-t0:.1f}s")

    # === Phase 3: Perturbation restarts on best ===
    print(f"\n{'='*50} Phase 3: Perturbation Restarts {'='*50}", flush=True)
    t0 = time.monotonic()
    n = tracker.best.size

    for temp in [0.1, 0.05, 0.03, 0.02, 0.01]:
        if time.monotonic() > deadline - 80:
            break

        a_pert = tracker.best.copy() + rng.normal(0, temp, n)
        a_pert = np.maximum(a_pert, 0.001)
        a_pert = normalize(a_pert)

        t_sub = min(deadline - 80, time.monotonic() + 40)
        a_trial = optimize_at_length(a_pert, t_sub, tracker,
                                      n_spectral=50, n_lp=50, n_cd=10)

    tracker.report(f"Phase 3: {time.monotonic()-t0:.1f}s")

    # === Phase 4: Intensive final optimization ===
    print(f"\n{'='*50} Phase 4: Final Intensive Optimization {'='*50}", flush=True)
    t0 = time.monotonic()

    # Long LP run
    a_final = lp_mirror_descent(tracker.best, min(deadline - 30, time.monotonic() + 120),
                                 tracker, max_iters=300)

    # Spectral polish
    if time.monotonic() < deadline - 20:
        a_final, _ = spectral_update_adaptive(a_final, 0.5, min(deadline - 20, time.monotonic() + 30), tracker)

    # CD polish
    if time.monotonic() < deadline - 10:
        a_final = coordinate_descent(a_final, min(deadline - 10, time.monotonic() + 20), tracker, n_passes=5)

    # Final LP
    if time.monotonic() < deadline - 2:
        a_final = lp_mirror_descent(a_final, deadline - 1, tracker, max_iters=50)

    tracker.report(f"Phase 4: {time.monotonic()-t0:.1f}s")

    # === Phase 5: Try one more length at the very end ===
    if time.monotonic() < deadline - 10 and tracker.best is not None:
        n_best = tracker.best.size
        for nl in [n_best + 2, n_best - 2, n_best + 4]:
            if time.monotonic() > deadline - 5:
                break
            if 100 <= nl <= 1000:
                a_try = resample_seq(tracker.best, nl)
                a_try = normalize(np.maximum(a_try, 0.001))
                J_try = compute_J(a_try)
                tracker.update(a_try)

    # Fallback
    if tracker.best is None:
        tracker.best = np.ones(256)
        tracker.best_score = compute_J(tracker.best)

    print(f"\n{'='*50} FINAL RESULT {'='*50}", flush=True)
    print(f"J = {tracker.best_score:.10f}, n = {tracker.best.size}", flush=True)
    print(f"Total time: {time.monotonic()-tracker.t_start:.1f}s", flush=True)

    return [float(x) for x in tracker.best]
```
