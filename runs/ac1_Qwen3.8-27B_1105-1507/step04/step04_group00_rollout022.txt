    """Compute gradient of smooth max of conv(a,a) using softmax."""

                    options={'time_limit': time_limit, 'presolve': True})

            g = np.maximum(r.x, 0)
            if g.sum() > 1e-15:

    return None

    a = a.copy().astype(np.float64)
    b = fast_conv(a)
    best_M = b.max()
    best_a = a.copy()
    eps = 0.001

    # Precompute index ranges for all pairs would be too much memory.
    # Instead, compute on the fly but efficiently.

    for sweep in range(n_sweeps):
        if deadline is not None and time.monotonic() > deadline:
            break

        # Random permutation for diversity
        perm = rng.permutation(2 * n)  # 2n random draws for pairs
        num_pairs = n

        for p in range(num_pairs):
            i = perm[2 * p] % n
            j = perm[2 * p + 1] % n
            if i == j:
                j = (i + 1) % n

            # Affected range
            k_lo = min(i, j)
            k_hi = min(2 * n - 2, max(i + n - 1, j + n - 1))
            k_len = k_hi - k_lo + 1

            # External max
            ext_max = 0.0
            if k_lo > 0:
                ext_max = float(np.max(b[:k_lo]))
            if k_hi < 2 * n - 2:
                ext_max = max(ext_max, float(np.max(b[k_hi + 1:])))

            b_aff = b[k_lo:k_hi + 1]

            # t bounds
            t_min = -a[i] * (1 - eps)
            t_max = a[j] * (1 - eps)
            if t_max <= t_min:
                continue

            # Precompute the linear coefficient array: L[k] = a[k-i] - a[k-j] (0 if out of range)
            L = np.zeros(k_len)
            ki_lo = max(k_lo, i) - k_lo
            ki_hi = min(k_hi, i + n - 1) - k_lo + 1
            if ki_hi > ki_lo:
                L[ki_lo:ki_hi] += a[max(k_lo, i) - i:min(k_hi, i + n - 1) - i + 1]
            kj_lo = max(k_lo, j) - k_lo
            kj_hi = min(k_hi, j + n - 1) - k_lo + 1
            if kj_hi > kj_lo:
                L[kj_lo:kj_hi] -= a[max(k_lo, j) - j:min(k_hi, j + n - 1) - j + 1]

            # Quadratic position indicators
            quad_pos = {}
            if 2 * i <= k_hi and 2 * i >= k_lo:
                quad_pos[2 * i - k_lo] = quad_pos.get(2 * i - k_lo, 0) + 1.0
            if 2 * j <= k_hi and 2 * j >= k_lo:
                quad_pos[2 * j - k_lo] = quad_pos.get(2 * j - k_lo, 0) - 1.0
            if i != j and i + j <= k_hi and i + j >= k_lo:
                quad_pos[i + j - k_lo] = quad_pos.get(i + j - k_lo, 0) - 2.0

            # Grid search
            best_t = 0.0
            current_M = max(float(b_aff.max()), ext_max)
            best_M_trial = current_M

            ts = np.linspace(t_min, t_max, n_grid)
            for t in ts:
                new_vals = b_aff + 2.0 * t * L
                # Add quadratic terms
                for pos, coeff in quad_pos.items():
                    new_vals[pos] += coeff * t * t
                M_trial = max(float(new_vals.max()), ext_max)
                if M_trial < best_M_trial - 1e-15:
                    best_M_trial = M_trial
                    best_t = t

            if best_t != 0.0:
                a[i] = max(0.0, a[i] + best_t)
                a[j] = max(0.0, a[j] - best_t)
                # Incrementally update b
                for k_idx in range(k_lo, k_hi + 1):
                    new_val = b_aff[k_idx - k_lo] + 2.0 * best_t * L[k_idx - k_lo]
                    for pos, coeff in quad_pos.items():
                        if k_idx - k_lo == pos:
                            new_val += coeff * best_t * best_t
                    b[k_idx] = new_val

        # Periodic full recompute to prevent drift
        if (sweep + 1) % 10 == 0:
            b = fast_conv(a)

        # Track best
        cur_M = b.max()
        if cur_M < best_M - 1e-14:
            best_M = cur_M
            best_a = a.copy()

    return best_a


def resample_seq(src, dest_n):
    """Resample 1D sequence to new length via linear interpolation."""
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)


def generate_initializations(n, rng, parent=None):
    """Generate diverse initial sequences of length n."""
    inits = []
    t_arr = np.linspace(0, 1, n, endpoint=False)

    if parent is not None:
        p = resample_seq(parent, n)
        inits.append(p)
        inits.append(np.maximum(p + rng.normal(0, 0.05, n), 0.01))
        inits.append(np.maximum(p + rng.normal(0, 0.15, n), 0.01))

    inits.append(np.ones(n))
    inits.append(rng.uniform(0.5, 1.0, n))

    # Raised cosine
    inits.append(0.5 + 0.5 * np.cos(2 * np.pi * t_arr))
    # Various harmonics
    for freq in [1, 2, 3, 4, 5]:
        inits.append(np.maximum(1.0 + 0.4 * np.cos(2 * np.pi * freq * t_arr), 0.01))
    # Chirp
    inits.append(np.maximum(1.0 + 0.3 * np.cos(2 * np.pi * 4 * t_arr ** 2), 0.01))
    # Gaussian bumps
    inits.append(np.exp(-4.0 * (t_arr - 0.5) ** 2) + 0.1)
    inits.append(np.exp(-8.0 * (t_arr - 0.3) ** 2) + np.exp(-8.0 * (t_arr - 0.7) ** 2) + 0.05)
    # U-shape
    inits.append(0.3 + 0.7 * (2 * t_arr - 1) ** 2)
    # Inverted U
    inits.append(0.7 - 0.4 * (2 * t_arr - 1) ** 2)
    inits[-1] = np.maximum(inits[-1], 0.01)
    # Smooth random
    ksize = max(3, n // 20)
    kernel = np.ones(ksize) / ksize
    smooth = np.convolve(rng.uniform(0, 1, n), kernel, mode='same')
    inits.append(np.maximum(smooth + 0.1, 0.01))
    # Block
    n_blocks = max(4, n // 64)
    block_vals = rng.uniform(0.3, 1.0, n_blocks)
    inits.append(np.repeat(block_vals, n // n_blocks + 1)[:n])
    # Power law
    inits.append(np.maximum(0.2 + 0.8 * t_arr ** 0.5, 0.01))
    inits.append(np.maximum(0.2 + 0.8 * (1 - t_arr) ** 0.5, 0.01))

    return inits


class ScoreTracker:
    def __init__(self):
        self.best = None
        self.best_score = float('inf')

    def update(self, a):
        a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
        if a.size == 0 or a.sum() < 0.01:
            return
        score = float(evaluate_sequence(a.tolist()))
        if np.isfinite(score) and score < self.best_score:
            self.best = a.copy()
            self.best_score = score
            print(f"  ** J={score:.8f} n={a.size}", flush=True)


def run_coord_as(a, deadline, rng, n_grid=25):
    """Run coordinate descent with simulated annealing acceptance."""
    n = len(a)
    a = a.copy().astype(np.float64)
    b = fast_conv(a)
    current_M = b.max()
    best_M = current_M
    best_a = a.copy()
    eps = 0.001
    temperature = 0.01 * current_M

    it = 0
    while time.monotonic() < deadline:
        it += 1
        # Pick random pair
        i = rng.integers(n)
        j = rng.integers(n)
        if i == j:
            j = (i + 1) % n

        k_lo = min(i, j)
        k_hi = min(2 * n - 2, max(i + n - 1, j + n - 1))
        k_len = k_hi - k_lo + 1

        ext_max = 0.0
        if k_lo > 0:
            ext_max = float(np.max(b[:k_lo]))
        if k_hi < 2 * n - 2:
            ext_max = max(ext_max, float(np.max(b[k_hi + 1:])))

        b_aff = b[k_lo:k_hi + 1]
        t_min = -a[i] * (1 - eps)
        t_max = a[j] * (1 - eps)
        if t_max <= t_min:
            continue

        # Build linear coefficient
        L = np.zeros(k_len)
        ki_lo = max(k_lo, i) - k_lo
        ki_hi = min(k_hi, i + n - 1) - k_lo + 1
        if ki_hi > ki_lo:
            L[ki_lo:ki_hi] += a[max(k_lo, i) - i:min(k_hi, i + n - 1) - i + 1]
        kj_lo = max(k_lo, j) - k_lo
        kj_hi = min(k_hi, j + n - 1) - k_lo + 1
        if kj_hi > kj_lo:
            L[kj_lo:kj_hi] -= a[max(k_lo, j) - j:min(k_hi, j + n - 1) - j + 1]

        quad_pos = {}
        if 2 * i <= k_hi and 2 * i >= k_lo:
            quad_pos[2 * i - k_lo] = quad_pos.get(2 * i - k_lo, 0) + 1.0
        if 2 * j <= k_hi and 2 * j >= k_lo:
            quad_pos[2 * j - k_lo] = quad_pos.get(2 * j - k_lo, 0) - 1.0
        if i != j and i + j <= k_hi and i + j >= k_lo:
            quad_pos[i + j - k_lo] = quad_pos.get(i + j - k_lo, 0) - 2.0

        # Try a few random t values and the grid optimum
        candidates = [0.0]
        candidates.extend(rng.uniform(t_min, t_max, 5).tolist())
        # Also add the grid optimum
        best_t = 0.0
        cur_M = max(float(b_aff.max()), ext_max)
        for t in np.linspace(t_min, t_max, 15):
            new_vals = b_aff + 2.0 * t * L
            for pos, coeff in quad_pos.items():
                new_vals[pos] += coeff * t * t
            M_t = max(float(new_vals.max()), ext_max)
            if M_t < cur_M:
                cur_M = M_t
                best_t = t
        candidates.append(best_t)

        # Evaluate best candidate
        best_new_M = current_M
        best_apply_t = 0.0
        for t in candidates:
            new_vals = b_aff + 2.0 * t * L
            for pos, coeff in quad_pos.items():
                new_vals[pos] += coeff * t * t
            M_t = max(float(new_vals.max()), ext_max)
            if M_t < best_new_M:
                best_new_M = M_t
                best_apply_t = t

        # Accept if improved (or with SA probability if worse)
        delta = best_new_M - current_M
        if delta < 0 or rng.random() < np.exp(-delta / max(temperature, 1e-10)):
            # Apply
            a[i] = max(0.0, a[i] + best_apply_t)
            a[j] = max(0.0, a[j] - best_apply_t)
            for k_idx in range(k_lo, k_hi + 1):
                new_val = b_aff[k_idx - k_lo] + 2.0 * best_apply_t * L[k_idx - k_lo]
                for pos, coeff in quad_pos.items():
                    if k_idx - k_lo == pos:
                        new_val += coeff * best_apply_t * best_apply_t
                b[k_idx] = new_val
            current_M = best_new_M

        # Cool down
        temperature *= 0.9995
        if temperature < 1e-6 * current_M:
            temperature = 1e-6 * current_M

        # Track best
        if current_M < best_M - 1e-14:
            best_M = current_M
            best_a = a.copy()

        # Periodic recompute
        if it % 500 == 0:
            b = fast_conv(a)
            current_M = b.max()

    return best_a


def propose_candidate(budget_s=1500):
    """
    Multi-phase search combining coordinate descent (fast, many iterations),
    LP mirror descent (few large moves), and smooth gradient (initialization).
    """
    rng = np.random.default_rng(42)
    deadline = time.monotonic() + budget_s - 20.0
    tracker = ScoreTracker()

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        if p.ndim == 1 and p.size > 0 and np.all(np.isfinite(p)):
            parent = np.maximum(p, 0.0)
            tracker.update(parent)
            print(f"Parent loaded: n={parent.size}, J={tracker.best_score:.6f}", flush=True)
    except (NameError, AttributeError):
        pass

    if parent is None:
        parent = np.ones(256)
        tracker.update(parent)

    best_seq = parent.copy()

    # === PHASE 1: Smooth gradient initialization across lengths ===
    print(f"\n{'='*60}", flush=True)
    print(f"PHASE 1: Smooth Gradient Initialization", flush=True)
    print(f"{'='*60}", flush=True)
    p1_deadline = time.monotonic() + 120.0

    lengths = [128, 192, 256, 384, 512, 640, 768, 1024]
    for n in lengths:
        if time.monotonic() > p1_deadline:
            break
        inits = generate_initializations(n, rng, best_seq)
        best_M_n = float('inf')
        best_a_n = None
        for a0 in inits:
            if time.monotonic() > p1_deadline:
                break
            a0 = np.maximum(a0, 1e-6)
            a0 = a0 / a0.sum()
            a = run_smooth_gradient(a0, 200, p1_deadline,
                                    initial_tau=3.0, final_tau=50.0,
                                    lr_init=0.2, lr_min=0.02)
            M_val = fast_conv(a).max()
            if M_val < best_M_n:
                best_M_n = M_val
                best_a_n = a.copy()
        if best_a_n is not None:
            J_val = compute_J(best_a_n)
            print(f"  n={n:4d}: J={J_val:.6f}", flush=True)
            tracker.update(best_a_n)
            if tracker.best_score < compute_J(best_seq):
                best_seq = best_a_n.copy()

    print(f"Phase 1 done. Best: J={tracker.best_score:.6f}, n={tracker.best.size}", flush=True)

    # === PHASE 2: Coordinate descent (main workhorse) ===
    print(f"\n{'='*60}", flush=True)
    print(f"PHASE 2: Coordinate Descent (fast iterative refinement)", flush=True)
    print(f"{'='*60}", flush=True)

    n_main = tracker.best.size
    a = tracker.best.copy().astype(np.float64)
    p2_deadline = time.monotonic() + 500.0

    # Multiple coordinate descent passes with different strategies
    for round_num in range(6):
        if time.monotonic() > p2_deadline:
            break
        rd = p2_deadline - 60 * (6 - round_num)  # leave time for later rounds
        if round_num == 0:
            a = coordinate_descent_fast(a, n_sweeps=1000, n_grid=25,
                                        rng=rng, deadline=rd)
        elif round_num == 1:
            a = run_coord_as(a, deadline=rd, rng=rng)
        elif round_num == 2:
            a = coordinate_descent_fast(a, n_sweeps=1000, n_grid=40,
                                        rng=rng, deadline=rd)
        elif round_num == 3:
            a = run_coord_as(a, deadline=rd, rng=rng)
        elif round_num == 4:
            a = coordinate_descent_fast(a, n_sweeps=2000, n_grid=30,
                                        rng=rng, deadline=rd)
        else:
            a = run_coord_as(a, deadline=rd, rng=rng)

        a = np.maximum(a, 1e-10)
        a = a / a.sum()
        tracker.update(a)
        if tracker.best is not None:
            a = tracker.best.copy()
        print(f"  Round {round_num+1}: J={tracker.best_score:.8f}, n={n_main}", flush=True)

    # === PHASE 3: LP mirror descent for large moves ===
    print(f"\n{'='*60}", flush=True)
    print(f"PHASE 3: LP Mirror Descent", flush=True)
    print(f"{'='*60}", flush=True)

    if tracker.best is not None:
        n_lp = tracker.best.size
        est_time = 0.005 * (n_lp / 128) ** 2
        max_iters = max(10, min(100, int(200 / max(0.05, est_time))))
        print(f"  n={n_lp}, est LP time={est_time*1000:.0f}ms, max_iters={max_iters}", flush=True)
        p3_deadline = min(time.monotonic() + 200, deadline - 100)
        a = run_lp_mirror_descent(tracker.best, max_iters, p3_deadline)
        tracker.update(a)
        if tracker.best is not None:
            a = tracker.best.copy()
            print(f"  After LP: J={tracker.best_score:.8f}", flush=True)

    # === PHASE 4: More coordinate descent on LP result ===
    print(f"\n{'='*60}", flush=True)
    print(f"PHASE 4: Coordinate Descent Polish", flush=True)
    print(f"{'='*60}", flush=True)

    if tracker.best is not None:
        p4_deadline = min(time.monotonic() + 200, deadline - 80)
        a = coordinate_descent_fast(tracker.best, n_sweeps=2000, n_grid=30,
                                    rng=rng, deadline=p4_deadline)
        tracker.update(a)
        a = run_coord_as(tracker.best, deadline=deadline - 50, rng=rng)
        tracker.update(a)
        print(f"  After CD polish: J={tracker.best_score:.8f}", flush=True)

    # === PHASE 5: Try different lengths ===
    print(f"\n{'='*60}", flush=True)
    print(f"PHASE 5: Length Exploration", flush=True)
    print(f"{'='*60}", flush=True)

    if tracker.best is not None:
        base_n = tracker.best.size
        candidate_lengths = set()
        for mult in [0.5, 0.75, 1.5, 2.0]:
            new_n = int(base_n * mult)
            if 64 <= new_n <= 2048:
                candidate_lengths.add(new_n)

        for new_n in sorted(candidate_lengths):
            if time.monotonic() > deadline - 30:
                break
            a_new = resample_seq(tracker.best, new_n)
            a_new = np.maximum(a_new, 1e-6)
            a_new = a_new / a_new.sum()
            # Quick smooth gradient
            a_new = run_smooth_gradient(a_new, 100, deadline - 30,
                                        initial_tau=5, final_tau=40,
                                        lr_init=0.15, lr_min=0.03)
            # Short coordinate descent
            a_new = coordinate_descent_fast(a_new, n_sweeps=200, n_grid=20,
                                            rng=rng, deadline=deadline - 30)
            tracker.update(a_new)
            print(f"  n={new_n}: J={tracker.best_score:.8f}", flush=True)

    # === PHASE 6: Final LP + CD alternating ===
    print(f"\n{'='*60}", flush=True)
    print(f"PHASE 6: Final Alternating Polish", flush=True)
    print(f"{'='*60}", flush=True)

    if tracker.best is not None and time.monotonic() < deadline - 5:
        remaining = deadline - time.monotonic()
        # Split time between LP and CD
        n_final = tracker.best.size
        est_lp = 0.005 * (n_final / 128) ** 2

        for alt in range(10):
            if time.monotonic() > deadline - 3:
                break
            # LP step
            lp_deadline = min(time.monotonic() + est_lp * 3, deadline - 10)
            a = run_lp_mirror_descent(tracker.best, 10, lp_deadline)
            tracker.update(a)
            # CD step
            cd_deadline = min(time.monotonic() + 30, deadline - 3)
            a = coordinate_descent_fast(a, n_sweeps=500, n_grid=30,
                                        rng=rng, deadline=cd_deadline)
            tracker.update(a)
            print(f"  Alt {alt+1}: J={tracker.best_score:.8f}", flush=True)

    # Final
    if tracker.best is None:
        tracker.best = np.ones(128)
        tracker.best_score = compute_J(tracker.best)

    print(f"\n{'='*60}", flush=True)
    print(f"FINAL: J={tracker.best_score:.8f} n={tracker.best.size}", flush=True)
    print(f"{'='*60}", flush=True)

    return [float(x) for x in tracker.best]
