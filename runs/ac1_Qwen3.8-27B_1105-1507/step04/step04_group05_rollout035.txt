def solve_kkt_system(n, active_set):
    """Solve M @ a = C * 1, sum(a) = 1, where M[i,j] = 1_{i+j in active_set}"""
    # Build M
    M = np.zeros((n, n))
    for i in range(n):
        for j in range(n):
            if (i + j) in active_set:
                M[i, j] = 1.0
    
    # Augmented system: [M  -1; 1^T 0] [a; C] = [0; 1]
    A_aug = np.zeros((n+1, n+1))
    A_aug[:n, :n] = M
    A_aug[:n, n] = -1.0
    A_aug[n, :n] = 1.0
    
    b_aug = np.zeros(n+1)
    b_aug[n] = 1.0
    
    try:
        sol = np.linalg.solve(A_aug, b_aug)
        a_sol = sol[:n]
        C = sol[n]
    except np.linalg.LinAlgError:
        sol, _, _, _ = np.linalg.lstsq(A_aug, b_aug, rcond=None)
        a_sol = sol[:n]
        C = sol[n]
    
    return a_sol, C

import numpy as np

def fast_conv(a):
    """Full self-convolution via FFT."""
    n = len(a)
    s = 1
    while s < 2 * n - 1:
        s <<= 1
    fa = np.fft.rfft(a, s)
    return np.fft.irfft(fa * fa, s)[:2 * n - 1]


def compute_J(a):
    """Compute J(a) = 2*n*max(conv(a,a))/(sum(a))^2."""
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))


def compute_kkt_gradient(a, active_mask):
    """Compute g[i] = sum_j a[j] * s[i+j] where s is active_mask (length 2n-1)."""
    n = len(a)
    # g[i] = sum_{j=0}^{n-1} active_mask[i+j] * a[j]
    # = convolve(active_mask, reverse(a))[n-1 : 2n-1]
    B = a[::-1]

    fB = np.fft.rfft(B, N)
    conv = np.fft.irfft(fS * fB, N)

def kkt_multiplicative_update(a, eta, active_eps):
    """One multiplicative update step guided by KKT gradient.
    Returns (new_a, c, max_c, J)."""
    n = len(a)
    c = fast_conv(a)
    max_c = c.max()
    if max_c < 1e-15:
        return a, c, max_c, compute_J(a)
    
    threshold = max_c * (1.0 - active_eps)
    active_mask = (c >= threshold).astype(np.float64)
    
    g = compute_kkt_gradient(a, active_mask)

    g_mean = g.mean()
    if g_mean > 1e-15:

    s = a_new.sum()
    if s < 1e-10:
        return a, c, max_c, compute_J(a)
    a_new /= s
    J = 2.0 * n * max_c / (a.sum() ** 2)  # approximate with old max_c
    return a_new, c, max_c, J


def kkt_linear_solve(n, active_set):
    """Solve the KKT linear system M@a = C*1, sum(a)=1.
    M[i,j] = 1 if (i+j) in active_set, else 0.
    Returns solution a (may have negative entries)."""
    M = np.zeros((n, n))
    for i in range(n):
        for j in range(n):
            if (i + j) in active_set:
                M[i, j] = 1.0
    
    # Augmented: [M  -1; 1^T  0] [a; C] = [0; 1]
    A_aug = np.zeros((n + 1, n + 1))
    A_aug[:n, :n] = M
    A_aug[:n, n] = -1.0
    A_aug[n, :n] = 1.0
    b_aug = np.zeros(n + 1)
    b_aug[n] = 1.0
    
    try:
        sol = np.linalg.solve(A_aug, b_aug)
        a_sol = sol[:n]
    except np.linalg.LinAlgError:
        sol, _, _, _ = np.linalg.lstsq(A_aug, b_aug, rcond=None)
        a_sol = sol[:n]
    
    return a_sol

def kkt_linear_iteration(a, max_iters, deadline, tracker):
    """Use KKT linear system solving with active set refinement."""
    a = np.maximum(a.copy(), 0.0)
    a /= a.sum()
    n = len(a)
    current_J = compute_J(a)
    tracker.update(a)

        if time.monotonic() > deadline:
            break
        
        c = fast_conv(a)
        max_c = c.max()
        
        # Active set: lags within 5% of max
        tol = 0.05 * max_c
        active_idx = np.where(c >= max_c - tol)[0]
        active_set = set(active_idx.tolist())
        
        # Solve KKT system
        a_sol = kkt_linear_solve(n, active_set)
        
        # Check if solution is valid
        if np.any(a_sol < -0.1):
            # Too many negatives - try wider active set
            tol = 0.1 * max_c
            active_idx = np.where(c >= max_c - tol)[0]
            active_set = set(active_idx.tolist())
            a_sol = kkt_linear_solve(n, active_set)
        
        if np.any(a_sol < -0.5):
            # Still bad - skip this iteration
            continue
        
        # Project and normalize
        a_new = np.maximum(a_sol, 0.0)
        s = a_new.sum()
        if s < 1e-10:
            continue
        a_new /= s
        
        # Blend with current (damping)
        alpha = 0.7 if it < 20 else 0.9

                current_J = J_proj
                a = a_new
                tracker.update(a)
    
    tracker.update(a)
    return a

def resample_seq(src, dest_n):
    """Resample 1D sequence to new length via linear interpolation."""
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)

                                                 eps_start=eps_val, eps_end=eps_val * 0.5)
            print(f"    eps={eps_val:.2f}: J={compute_J(a_trial):.8f}", flush=True)
        
        tracker.report(f"Phase 5 Sweep: {time.monotonic()-t0:.1f}s")
    
    # === Phase 6: Multi-length exploration ===
    if time.monotonic() < deadline - 150:
        print(f"\n{'='*50} Phase 6: Multi-Length Exploration {'='*50}", flush=True)
        t0 = time.monotonic()
        
        base = tracker.best.copy() if tracker.best is not None else a
        n_base = base.size
        
        # Try nearby lengths
        lengths_to_try = []
        for delta in [-8, -4, -2, 2, 4, 8, 16, 32]:
            nl = n_base + delta
            if 100 <= nl <= 1000 and nl not in lengths_to_try:
                lengths_to_try.append(nl)
        
        # Also try some "natural" lengths
        for nl in [256, 300, 384, 400, 512, 500]:
            if 100 <= nl <= 1000 and nl not in lengths_to_try:
                lengths_to_try.append(nl)
        
        for nl in sorted(lengths_to_try):
            if time.monotonic() > deadline - 100:
                break
            
            a_resampled = resample_seq(base, nl)
            a_resampled = np.maximum(a_resampled, 0.001)
            a_resampled /= a_resampled.sum()
            
            # Quick KKT-MU
            a_resampled = kkt_fixed_point_iteration(a_resampled, max_iters=5000,
                                                     deadline=time.monotonic() + 50, tracker=tracker,
                                                     eps_start=0.05, eps_end=0.01)
            # Quick CD
            if time.monotonic() < deadline - 100:
                a_resampled = coordinate_descent_fast(a_resampled, n_passes=1,
                                                       deadline=deadline - 100)
                tracker.update(a_resampled)
            
            print(f"    n={nl}: J_best={tracker.best_score:.8f}", flush=True)
        
        tracker.report(f"Phase 6 Lengths: {time.monotonic()-t0:.1f}s")
    
    # === Phase 7: Perturbation restarts ===
    if time.monotonic() < deadline - 50:
        print(f"\n{'='*50} Phase 7: Perturbation Restarts {'='*50}", flush=True)
        t0 = time.monotonic()
        
        best = tracker.best.copy() if tracker.best is not None else a
        n = best.size
        
        for temp in [0.03, 0.02, 0.01, 0.005]:
            if time.monotonic() > deadline - 30:
                break
            
            a_pert = best + rng.normal(0, temp, n)
            a_pert = np.maximum(a_pert, 0.001)
            a_pert /= a_pert.sum()
            
            a_pert = kkt_fixed_point_iteration(a_pert, max_iters=5000,
                                                deadline=time.monotonic() + 40, tracker=tracker,
                                                eps_start=0.05, eps_end=0.01)
            if time.monotonic() < deadline - 30:
                a_pert = coordinate_descent_fast(a_pert, n_passes=1, deadline=deadline - 30)
                tracker.update(a_pert)
        
        tracker.report(f"Phase 7 Restarts: {time.monotonic()-t0:.1f}s")
    
    # === Phase 8: Final intensive polish ===
    if time.monotonic() < deadline - 5:
        print(f"\n{'='*50} Phase 8: Final Polish {'='*50}", flush=True)
        t0 = time.monotonic()
        
        a_final = tracker.best.copy() if tracker.best is not None else a
        a_final = kkt_fixed_point_iteration(a_final, max_iters=10000, deadline=deadline - 20,
                                             tracker=tracker, eps_start=0.03, eps_end=0.005)
        
        if time.monotonic() < deadline - 10:
            a_final = coordinate_descent_fast(a_final, n_passes=2, deadline=deadline - 5)
            tracker.update(a_final)
        
        tracker.report(f"Phase 8 Final: {time.monotonic()-t0:.1f}s")
    
    # Fallback
    if tracker.best is None:
        tracker.best = np.ones(256)
        tracker.best_score = compute_J(tracker.best)
    
    print(f"\n{'='*50} FINAL RESULT {'='*50}", flush=True)
    print(f"J = {tracker.best_score:.10f}, n = {tracker.best.size}", flush=True)
    print(f"Total time: {time.monotonic()-tracker.t_start:.1f}s", flush=True)
    
    return [float(x) for x in tracker.best]
