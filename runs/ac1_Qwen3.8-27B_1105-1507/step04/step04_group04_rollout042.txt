```
def softmax_vals(z):
    z = np.asarray(z, dtype=np.float64)
    z = np.nan_to_num(z, nan=0.0, posinf=0.0, neginf=0.0)
    if z.size == 0:
        return z.copy()
    m = float(np.max(z))
    ez = np.exp(z - m)
    s = float(np.sum(ez))
    if s <= 0.0 or not np.isfinite(s):
        return np.full(z.shape, 1.0 / z.size, dtype=np.float64)
    return ez / s
```

```
def lbfgs_continuation(a, time_budget, taus=None, verbose=False):
    start = time.monotonic()
    n = len(a)
    a = normalize_to_S(a, n)
    best_a = a.copy()
    best_score = score_sequence(a)
    if time_budget <= 2.0 or n <= 0:
        return best_a, best_score
    if taus is None:
        if time_budget < 12.0:
            taus = [20.0, 60.0]
        elif time_budget < 35.0:
            taus = [10.0, 40.0, 100.0]
        elif time_budget < 90.0:
            taus = [5.0, 20.0, 60.0, 160.0]
        else:
            taus = [5.0, 15.0, 40.0, 100.0, 200.0, 500.0]
    lw = len(taus)
    if lw == 2: ws = [1.0, 2.0]
    elif lw == 3: ws = [1.0, 1.0, 2.0]
    elif lw == 4: ws = [0.8, 1.0, 1.2, 2.0]
    else: ws = [0.7, 0.8, 1.0, 1.2, 1.6, 2.8]
    wsum = sum(ws)
    S = np.sqrt(2.0*n)
    u = np.maximum(a/S, 0.0)
    us = float(np.sum(u))
    if us <= 1e-12:
        u = np.ones(n); us = n
    else:
        u = u / us
    z = np.log(u + 1e-12)
    z = z - float(np.mean(z))
    for idx, tau_val in enumerate(taus):
        tau_budget = float(time_budget) * ws[idx] / wsum
        tau_end = time.monotonic() + tau_budget
        stall = 0
        while time.monotonic() < min(tau_end, start + time_budget - 1.0):
            rem = tau_end - time.monotonic()
            if rem < 0.15: break
            _CUR_N = n
            _CUR_TAU = float(tau_val)
            if n <= 250: maxiter = 300
            elif n <= 600: maxiter = 180
            else: maxiter = 90
            if rem < 2.0: maxiter = min(maxiter, 40)
            elif rem < 8.0: maxiter = min(maxiter, 120)
            options = {'maxiter': int(maxiter), 'maxfun': int(maxiter*3), 'ftol': 1e-12, 'gtol': 1e-7, 'maxls': 50, 'disp': False}
            try:
                res = minimize(lbfgs_objective, z, method='L-BFGS-B', jac=True, options=options)
            except Exception:
                break
            if getattr(res, 'x', None) is None:
                break
            z2 = np.asarray(res.x, dtype=np.float64)
            if not np.all(np.isfinite(z2)):
                break
            z2 = z2 - float(np.mean(z2))
            z2 = np.clip(z2, -100.0, 100.0)
            u2 = softmax_vals(z2)
            a2 = S * u2
            sc = score_sequence(a2)
            if sc < best_score - 1e-12:
                best_score = sc
                best_a = a2.copy()
                stall = 0
            else:
                stall += 1
            z = z2
            if stall >= 3:
                break
            status = getattr(res, 'status', -1)
            if status == 0 and stall > 0:
                break
        u_best = np.maximum(best_a / S, 0.0)
        ub = float(np.sum(u_best))
        if ub <= 1e-12:
            u_best = np.ones(n) / n
        else:
            u_best = u_best / ub
        z = np.log(u_best + 1e-12)
        z = z - float(np.mean(z))
        if verbose:
            print(f"      LBFGS tau={tau_val:.0f} best={best_score:.6f} elapsed={time.monotonic()-start:.1f}s")
    return best_a, float(best_score)
```

```python
"""
Optimize nonnegative step-function heights by minimizing
    2*n*max(np.convolve(a,a)) / (sum(a))^2.

Strategy:
- Work internally with sum(a) = sqrt(2n), so the score is the peak
  convolution coefficient.
- Main local optimizer: L-BFGS on the log-simplex minimizing a
  log-sum-exp surrogate of the peak convolution.  Tau is annealed.
- Secondary global/local tools: alternating LP directions using
  max sum(g) s.t. a*g <= M, random perturbations, re-optimization, and
  length exploration around the parent construction.
"""

import numpy as np
import time
from scipy.optimize import minimize, linprog

try:
    import cvxpy as cp
    HAS_CVXPY = True
except Exception:
    HAS_CVXPY = False

np.seterr(over="ignore", invalid="ignore", divide="ignore", under="ignore")

# Globals used by the L-BFGS objective function (kept top-level to avoid closures).
_CUR_N = 0
_CUR_TAU = 10.0


def softmax_vals(z):
    z = np.asarray(z, dtype=np.float64)
    z = np.nan_to_num(z, nan=0.0, posinf=0.0, neginf=0.0)
    if z.size == 0:
        return z.copy()
    m = float(np.max(z))
    ez = np.exp(z - m)
    s = float(np.sum(ez))
    if s <= 0.0 or not np.isfinite(s):
        return np.full(z.shape, 1.0 / z.size, dtype=np.float64)
    return ez / s


def compute_conv(a):
    a = np.asarray(a, dtype=np.float64)
    n = len(a)
    if n == 0:
        return np.array([], dtype=np.float64)

    if n <= 128:
        conv = np.convolve(a, a, mode="full")
    else:
        need = 2 * n - 1
        size = 1
        while size < need:
            size <<= 1
        fa = np.fft.rfft(a, size)
        conv = np.fft.irfft(fa * fa, size)[:need]

    conv = np.asarray(conv, dtype=np.float64)
    if conv.size and conv.min() < 0.0:
        conv = np.maximum(conv, 0.0)
    return conv


def score_sequence(a):
    a = np.asarray(a, dtype=np.float64)
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s <= 1e-12:
        return np.inf
    c = compute_conv(a)
    if c.size == 0:
        return np.inf
    return float(2.0 * n * max(0.0, np.max(c)) / (s * s))


def normalize_to_S(a, n):
    a = np.asarray(a, dtype=np.float64).copy()
    if a.size != n:
        if a.size == 0:
            a = np.ones(n, dtype=np.float64)
        else:
            a = a[:n] if a.size >= n else np.resize(a, n)
    a = np.maximum(a, 0.0)
    s = float(np.sum(a))
    target = np.sqrt(2.0 * max(1, n))
    if s <= 1e-12:
        a = np.ones(n, dtype=np.float64)
        s = float(n)
    a *= target / s
    return a


def _lbfgs_core(z):
    n = _CUR_N
    tau = float(_CUR_TAU)
    if n <= 0:
        return np.inf, np.zeros(0, dtype=np.float64)

    z_arr = np.asarray(z, dtype=np.float64)
    if z_arr.size != n:
        z_arr = np.resize(z_arr, n)

    u = softmax_vals(z_arr)
    S = np.sqrt(2.0 * n)
    a = S * u
    c = compute_conv(a)

    if c.size == 0 or not np.all(np.isfinite(c)):
        return np.inf, np.zeros(n, dtype=np.float64)

    maxc = float(np.max(c))
    if not np.isfinite(maxc):
        return np.inf, np.zeros(n, dtype=np.float64)

    if tau <= 0.0:
        tau = 1e-6

    x = tau * (c - maxc)
    m = float(np.max(x))
    ex = np.exp(np.clip(x - m, -745.0, 0.0))
    wsum = float(np.sum(ex))

    if wsum <= 1e-300 or not np.isfinite(wsum):
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
        obj = maxc + m / tau
    else:
        w = ex / wsum
        obj = maxc + (m + np.log(wsum)) / tau

    # Gradient of log-sum-exp(c) wrt a:
    # gcoker = sum_m w_m a_{m-i}, then g_a_i = 2 * gcoker_i + symmetric term,
    # which is exactly 2 * correlate(w, a) at the middle n coordinates.
    corr = np.correlate(w, a, mode="full")
    ga = 2.0 * corr[n - 1:2 * n - 1]
    if ga.size != n:
        ga = np.resize(ga, n)

    s = float(np.sum(a))
    if s <= 1e-12:
        return float(obj), np.zeros(n, dtype=np.float64)

    # Gradient wrt z for a = S * softmax(z):
    # g_z_i = S u_i (g_a_i - (sum_j a_j g_a_j) / S)
    t = float(np.dot(a, ga)) / s
    gz = a * (ga - t)
    gz = np.nan_to_num(gz, nan=0.0, posinf=0.0, neginf=0.0)

    if not np.isfinite(obj):
        return np.inf, np.zeros(n, dtype=np.float64)

    return float(obj), gz


def lbfgs_objective(z):
    return _lbfgs_core(z)


def lbfgs_continuation(a, time_budget, taus=None, verbose=False):
    start = time.monotonic()
    n = len(a)
    a = normalize_to_S(a, n)
    best_a = a.copy()
    best_score = score_sequence(a)

    if time_budget <= 2.0 or n <= 0:
        return best_a, float(best_score)

    if taus is None:
        if time_budget < 12.0:
            taus = [20.0, 60.0]
        elif time_budget < 35.0:
            taus = [10.0, 40.0, 100.0]
        elif time_budget < 90.0:
            taus = [5.0, 20.0, 60.0, 160.0]
        else:
            taus = [5.0, 15.0, 40.0, 100.0, 200.0, 500.0]

    lw = len(taus)
    if lw == 2:
        ws = [1.0, 2.0]
    elif lw == 3:
        ws = [1.0, 1.0, 2.0]
    elif lw == 4:
        ws = [0.8, 1.0, 1.2, 2.0]
    else:
        ws = [0.7, 0.8, 1.0, 1.2, 1.6, 2.8]
    wsum = float(sum(ws))

    S = float(np.sqrt(2.0 * n))
    u = np.maximum(a / S, 0.0)
    us = float(np.sum(u))
    if us <= 1e-12:
        u = np.ones(n, dtype=np.float64)
        us = float(n)
    else:
        u = u / us
    z = np.log(u + 1e-12)
    z = z - float(np.mean(z))

    if n <= 160:
        base_iter = 320
    elif n <= 350:
        base_iter = 180
    elif n <= 700:
        base_iter = 100
    else:
        base_iter = 60

    for idx, tau_val in enumerate(taus):
        tau_budget = float(time_budget) * ws[idx] / wsum
        tau_end = time.monotonic() + tau_budget
        stall = 0

        while time.monotonic() < min(tau_end, start + time_budget - 1.0):
            rem = tau_end - time.monotonic()
            if rem < 0.15:
                break

            _CUR_N = n
            _CUR_TAU = float(tau_val)

            maxiter = base_iter
            if rem < 2.0:
                maxiter = min(maxiter, 35)
            elif rem < 8.0:
                maxiter = min(maxiter, 90)

            options = {
                "maxiter": int(maxiter),
                "maxfun": int(max(1, maxiter * 3)),
                "ftol": 1e-10,
                "gtol": 1e-6,
                "maxls": 50,
                "disp": False,
            }

            try:
                res = minimize(lbfgs_objective, z, method="L-BFGS-B", jac=True, options=options)
            except Exception:
                break

            if getattr(res, "x", None) is None:
                break

            z2 = np.asarray(res.x, dtype=np.float64)
            if not np.all(np.isfinite(z2)):
                break

            z2 = z2 - float(np.mean(z2))
            z2 = np.clip(z2, -100.0, 100.0)

            u2 = softmax_vals(z2)
            a2 = S * u2
            sc = score_sequence(a2)

            if sc < best_score - 1e-12:
                best_score = sc
                best_a = a2.copy()
                stall = 0
            else:
                stall += 1

            z = z2

            if stall >= 3:
                break
            if getattr(res, "status", -1) == 0 and stall > 0:
                break

        # For the next tau, restart from the best exact point found so far.
        u_best = np.maximum(best_a / S, 0.0)
        ub = float(np.sum(u_best))
        if ub <= 1e-12:
            u_best = np.ones(n, dtype=np.float64) / n
        else:
            u_best = u_best / ub
        z = np.log(u_best + 1e-12)
        z = z - float(np.mean(z))

        if verbose:
            print(f"      LBFGS tau={float(tau_val):.0f} best={best_score:.6f} elapsed={time.monotonic()-start:.1f}s")

    return best_a, float(best_score)


def build_toeplitz(a):
    n = len(a)
    A = np.zeros((2 * n - 1, n), dtype=np.float64)
    for j in range(n):
        end = min(j + n, 2 * n - 1)
        A[j:end, j] = a[:end - j]
    return A


def solve_lp_maxsum(a, M, time_limit=3.0):
    n = len(a)
    if M <= 0.0 or not np.isfinite(M):
        return None, None

    A = build_toeplitz(a)

    if HAS_CVXPY and n <= 500:
        try:
            g_var = cp.Variable(n, nonneg=True)
            prob = cp.Problem(cp.Maximize(cp.sum(g_var)), [A @ g_var <= M])
            solved = False
            for solver in ("MOSEK", "CLARABEL", "SCS"):
                try:
                    prob.solve(solver=solver, time_limit=float(time_limit))
                    if g_var.value is not None:
                        solved = True
                        break
                except Exception:
                    continue

            if solved and g_var.value is not None:
                gv = np.maximum(np.asarray(g_var.value, dtype=np.float64), 0.0)
                s = float(np.sum(gv))
                if s > 1e-12 and np.isfinite(s):
                    return gv, s
        except Exception:
            pass

    c = -np.ones(n, dtype=np.float64)
    b_ub = np.full(2 * n - 1, M, dtype=np.float64)
    bounds = [(0.0, None)] * n
    try:
        res = linprog(
            c,
            A_ub=A,
            b_ub=b_ub,
            bounds=bounds,
            method="highs",
            options={"presolve": True, "time_limit": float(time_limit)},
        )
        if res.status == 0 and res.x is not None:
            gv = np.maximum(np.asarray(res.x, dtype=np.float64), 0.0)
            s = float(np.sum(gv))
            if s > 1e-12 and np.isfinite(s):
                return gv, s
    except Exception:
        pass

    return None, None


def alternating_lp(a, time_budget, max_iters=8):
    n = len(a)
    start = time.monotonic()
    a = normalize_to_S(a, n)
    S = float(np.sqrt(2.0 * n))
    best_a = a.copy()
    best_score = score_sequence(a)

    for _ in range(max_iters):
        if time.monotonic() > start + time_budget - 4.0:
            break

        a_safe = np.maximum(a, 1e-8)
        ss = float(np.sum(a_safe))
        if ss <= 1e-12:
            break
        a_safe = a_safe / ss * S

        c = compute_conv(a_safe)
        M = float(np.max(c))
        if M <= 1e-12 or not np.isfinite(M):
            break

        tl = min(3.0, max(0.5, (start + time_budget - 5.0 - time.monotonic()) / 5.0))
        if tl < 0.3:
            break

        g, sumg = solve_lp_maxsum(a_safe, M, time_limit=tl)
        if g is None or not np.isfinite(sumg):
            break
        if sumg <= S * 1.0001:
            break

        g = normalize_to_S(g, n)
        best_t = 0.0
        best_sc_ls = best_score

        for t in (0.05, 0.10, 0.15, 0.25, 0.40, 0.60, 0.80, 1.00):
            a_new = (1.0 - t) * a + t * g
            a_new = normalize_to_S(a_new, n)
            sc = score_sequence(a_new)
            if sc < best_sc_ls:
                best_sc_ls = sc
                best_t = t

        if best_t > 0.0 and best_sc_ls < best_score - 1e-12:
            a_new = (1.0 - best_t) * a + best_t * g
            a = normalize_to_S(a_new, n)
            best_score = score_sequence(a)
            best_a = a.copy()
        else:
            break

    return best_a, float(best_score)


def random_init(n, seed):
    rng = np.random.default_rng(seed)
    k = np.arange(n, dtype=np.float64)
    choice = int(rng.integers(0, 7))

    if choice == 0:
        a = np.ones(n, dtype=np.float64)
    elif choice == 1:
        f = float(rng.uniform(0.5, 3.0))
        ph = float(rng.uniform(0.0, 2.0 * np.pi))
        amp = float(rng.uniform(0.2, 0.7))
        a = 1.0 + amp * np.cos(2.0 * np.pi * f * k / n + ph)
    elif choice == 2:
        f1 = float(rng.uniform(0.3, 2.0))
        f2 = float(rng.uniform(2.0, 6.0))
        a = (
            1.0
            + 0.3 * np.cos(2.0 * np.pi * f1 * k / n)
            + 0.25 * np.sin(2.0 * np.pi * f2 * k / n + float(rng.uniform(0.0, 2.0 * np.pi)))
        )
    elif choice == 3:
        a = 0.2 * np.ones(n, dtype=np.float64)
        sig1 = max(1.0, n * float(rng.uniform(0.02, 0.08)))
        amp1 = float(rng.uniform(3.0, 10.0))
        a += amp1 * np.exp(-0.5 * (k / sig1) ** 2)
        sig2 = max(1.0, n * float(rng.uniform(0.02, 0.08)))
        amp2 = float(rng.uniform(3.0, 10.0))
        a += amp2 * np.exp(-0.5 * ((n - 1.0 - k) / sig2) ** 2)
    elif choice == 4:
        a = rng.random(n, dtype=np.float64)
        win = max(3, n // 20)
        kern = np.ones(win, dtype=np.float64) / win
        a = np.convolve(a, kern, mode="same")
    elif choice == 5:
        a = 1.0 + float(rng.uniform(0.2, 0.8)) * (k / max(1.0, n - 1.0))
    else:
        f = float(rng.uniform(0.8, 4.0))
        ph = float(rng.uniform(0.0, 2.0 * np.pi))
        a = 1.0 + 0.6 * np.sin(2.0 * np.pi * f * k / n + ph)

    a = np.maximum(a, 0.01)
    return normalize_to_S(a, n)


def resample_sequence(a, new_n):
    a = np.asarray(a, dtype=np.float64)
    if a.size == 0:
        return np.ones(new_n, dtype=np.float64)
    if a.size == 1:
        return np.full(new_n, float(a[0]), dtype=np.float64)
    old_k = np.arange(a.size, dtype=np.float64)
    new_k = np.linspace(0.0, float(a.size - 1), new_n, dtype=np.float64)
    b = np.interp(new_k, old_k, a)
    b = np.maximum(b, 0.0)
    return b


def optimize_sequence(a, time_budget, verbose=False):
    start = time.monotonic()
    n = len(a)
    a = normalize_to_S(a, n)
    best_a = a.copy()
    best_score = score_sequence(a)

    if verbose:
        print(f"    optimize_sequence n={n} start={best_score:.6f} budget={float(time_budget):.0f}")

    # Phase 1: main L-BFGS continuation.
    p1 = min(float(time_budget) * 0.55, float(time_budget) - 5.0)
    if p1 > 5.0:
        a1, sc1 = lbfgs_continuation(best_a, p1, verbose=verbose)
        if sc1 < best_score:
            best_score = sc1
            best_a = a1.copy()

    # Phase 2: LP-based direction.
    rem = (start + time_budget - 10.0) - time.monotonic()
    if rem > 18.0 and n <= 700:
        p2 = min(rem * 0.25, 50.0)
        if p2 > 6.0:
            a2, sc2 = alternating_lp(best_a, p2, max_iters=6)
            if sc2 < best_score:
                best_score = sc2
                best_a = a2.copy()

    # Phase 3: randomized local re-optimization.
    rem = (start + time_budget - 10.0) - time.monotonic()
    if rem > 18.0:
        local_budget = min(rem * 0.35, 80.0)
        local_deadline = time.monotonic() + local_budget
        max_attempts = max(1, min(6, int(float(time_budget) / 25.0) + 1))
        attempt = 0

        S_local = float(np.sqrt(2.0 * n))
        u_base = np.maximum(best_a / S_local, 0.0)
        ub = float(np.sum(u_base))
        if ub <= 1e-12:
            u_base = np.ones(n, dtype=np.float64) / n
        else:
            u_base = u_base / ub

        while attempt < max_attempts and time.monotonic() < local_deadline - 4.0:
            attempt += 1
            sub = min(20.0, local_deadline - time.monotonic() - 2.0)
            if sub < 5.0:
                break

            seed_local = 12345 + n * 7919 + attempt * 104729
            rng = np.random.default_rng(seed_local)
            zz = np.log(u_base + 1e-9)
            zz = zz - float(np.mean(zz))
            scale = 0.03 + 0.03 * (attempt % 4)
            zz = zz + rng.normal(0.0, scale, n)
            zz = zz - float(np.mean(zz))
            u_try = softmax_vals(zz)
            a_try = normalize_to_S(u_try, n)

            a3, sc3 = lbfgs_continuation(a_try, sub, taus=[40.0, 100.0, 250.0])
            if sc3 < best_score:
                best_score = sc3
                best_a = a3.copy()

    # Phase 4: final higher-tau L-BFGS polish.
    rem = (start + time_budget - 8.0) - time.monotonic()
    if rem > 12.0:
        p4 = min(rem * 0.6, 70.0)
        a4, sc4 = lbfgs_continuation(best_a, p4, taus=[80.0, 200.0, 500.0])
        if sc4 < best_score:
            best_score = sc4
            best_a = a4.copy()

    # Phase 5: one more short LP polish if it is still likely cheap.
    rem = (start + time_budget - 6.0) - time.monotonic()
    if rem > 12.0 and n <= 500:
        p5 = min(rem * 0.5, 25.0)
        a5, sc5 = alternating_lp(best_a, p5, max_iters=3)
        if sc5 < best_score:
            best_score = sc5
            best_a = a5.copy()

    if verbose:
        print(f"    optimize_sequence final n={n} best={best_score:.6f}")

    return best_a, float(best_score)


def propose_candidate(seed=42, budget_s=1500):
    start = time.monotonic()
    deadline = start + float(budget_s) - 30.0

    best_seq = None
    best_score = np.inf
    base_n = 330

    # Try to load the supplied parent construction.
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        p = np.clip(p, 0.0, None)
        if p.size > 0 and np.all(np.isfinite(p)):
            base_n = int(p.size)
            best_seq = normalize_to_S(p, base_n)
            best_score = score_sequence(best_seq)
            print(f"Parent loaded: n={base_n}, score={best_score:.6f}")
    except Exception:
        best_seq = normalize_to_S(np.ones(base_n, dtype=np.float64), base_n)
        best_score = score_sequence(best_seq)
        print(f"No valid parent; using uniform n={base_n}, score={best_score:.6f}")

    # Phase 1: intensive optimization of the current best / parent.
    p1 = min(600.0, max(30.0, (deadline - time.monotonic()) * 0.40))
    if p1 > 10.0:
        a1, sc1 = optimize_sequence(best_seq, p1, verbose=True)
        if sc1 < best_score:
            best_score = sc1
            best_seq = a1.copy()
            print(f"  Phase1 update: n={len(best_seq)}, score={best_score:.6f}")

    # Phase 2: explore nearby lengths by resampling the current best,
    # plus a small number of independent random smooth starts.
    n_candidates = []
    for delta in (0, -5, 5, -10, 10, -20, 20, -35, 35, -50, 50):
        nv = base_n + delta
        if 100 <= nv <= 1000:
            n_candidates.append(nv)
    for nv in (300, 320, 340, 360, 380, 400):
        if 100 <= nv <= 1000:
            n_candidates.append(nv)

    n_candidates = sorted(set(n_candidates))

    explore_cap = min(450.0, max(0.0, deadline - start - 80.0))
    explored = 0.0

    for nv in n_candidates:
        if time.monotonic() > deadline - 70.0:
            break
        if explored > explore_cap:
            break
        if len(best_seq) == nv:
            continue

        remaining = deadline - time.monotonic() - 70.0
        if remaining < 20.0:
            break

        alloc = min(55.0, max(18.0, remaining / 5.0))
        explored += alloc

        init1 = resample_sequence(best_seq, nv)
        init1 = normalize_to_S(init1, nv)

        t1 = min(alloc * 0.65, remaining - 20.0)
        if t1 > 10.0:
            a_opt, sc = optimize_sequence(init1, t1, verbose=False)
            if sc < best_score:
                best_score = sc
                best_seq = a_opt.copy()
                print(f"  Phase2 resample update: n={nv}, score={best_score:.6f}")

        remaining2 = deadline - time.monotonic() - 70.0
        if remaining2 > 25.0:
            init2 = random_init(nv, int(seed) + nv * 104729)
            t2 = min(alloc * 0.35, remaining2 - 15.0)
            if t2 > 8.0:
                a_opt, sc = lbfgs_continuation(init2, t2, verbose=False)
                if sc < best_score:
                    best_score = sc
                    best_seq = a_opt.copy()
                    print(f"  Phase2 random update: n={nv}, score={best_score:.6f}")

    # Phase 3: final intensive polish on the global best.
    remaining = deadline - time.monotonic()
    if remaining > 40.0 and best_seq is not None:
        final_budget = min(remaining - 15.0, 400.0)
        a_final, sc_final = optimize_sequence(best_seq, final_budget, verbose=True)
        if sc_final < best_score:
            best_score = sc_final
            best_seq = a_final.copy()

    # Safety fallback.
    if best_seq is None:
        best_seq = normalize_to_S(np.ones(200, dtype=np.float64), 200)

    # Convert to the exact output type required by evaluate_sequence.
    result = []
    for x in best_seq:
        vx = float(x)
        if not np.isfinite(vx):
            vx = 0.0
        if vx < 0.0:
            vx = 0.0
        if vx > 1000.0:
            vx = 1000.0
        result.append(vx)

    # Final validation with the official evaluator.
    try:
        val = float(evaluate_sequence(result))
        if np.isfinite(val) and val < best_score:
            best_score = val
    except Exception:
        pass

    print(f"FINAL: n={len(result)}, score={best_score:.6f}, elapsed={time.monotonic()-start:.1f}s")
    return result
```
