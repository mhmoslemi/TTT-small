3. θ = (1/ρ) * Σ_{i=1}^ρ v_(i)
4. Projected point: x_i = max(v_i - θ, 0)

"""
Minimizes J(a) = 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Smooth gradient descent with log-sum-exp approximation and adaptive temperature annealing
2. Projected gradient onto the probability simplex
3. Momentum for faster convergence
4. Multiple diverse restarts with different initializations and lengths
5. LP best-response polish for final improvement
6. Coordinate descent for final polish
Key innovation: Smooth differentiable approximation of the max via log-sum-exp,
enabling efficient gradient-based optimization with temperature annealing.
"""

import numpy as np
from scipy.optimize import linprog
from scipy.sparse import lil_matrix
import time

def fast_conv(a):
    """Full self-convolution via FFT."""
    n = len(a)
    if n == 0:
        return np.array([])
    s = 1
    while s < 2 * n - 1:
        s <<= 1
    fa = np.fft.rfft(a, s)
    return np.fft.irfft(fa * fa, s)[:2 * n - 1]

def compute_J_normalized(a):
    """Compute J = 2*n*max(conv(a,a))/(sum(a))^2. Assumes a >= 0."""
    s = a.sum()
    if s < 1e-10:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))

def logsumexp(x, tau=1.0):
    """Compute log(sum(exp(tau*x))) / tau with numerical stability."""
    x_max = x.max()
    if x_max < -1e300:
        return 0.0
    shifted = tau * (x - x_max)
    log_sum = x_max + np.log(np.sum(np.exp(shifted))) / tau
    return log_sum

def smooth_grad(a, tau):
    """Compute gradient of logsumexp(tau * conv(a)) w.r.t. a.
    Returns (value, gradient)."""
    n = len(a)
    b = fast_conv(a)
    b_max = b.max()
    shifted = tau * (b - b_max)
    exp_vals = np.exp(shifted)
    w = exp_vals / exp_vals.sum()
    value = b_max + np.log(exp_vals.sum()) / tau
    # Gradient: 2 * tau * correlate(w, a, 'valid')
    grad = 2.0 * tau * np.correlate(w, a, mode='valid')
    return value, grad

def project_simplex(v):
    """Project v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - 1))[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    return np.maximum(v - theta, 0)

def smooth_gradient_descent(a, tau_schedule, max_iters_per_tau, momentum=0.9, deadline=None):
    """Run smooth gradient descent with given tau schedule.
    a: initial point (will be normalized to sum=1)
    tau_schedule: list of tau values to use (annealing)
    Returns optimized a and final J."""
    a = np.maximum(a.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-10:
        return a, float('inf')
    a = a / s
    v = np.zeros_like(a)  # velocity for momentum
    
    best_a = a.copy()
    best_J = compute_J_normalized(a)
    
    for tau in tau_schedule:
        for it in range(max_iters_per_tau):
            if deadline is not None and time.monotonic() > deadline:
                return best_a, best_J
            
            value, grad = smooth_grad(a, tau)
            v = momentum * v - grad  # heavy ball
            a_new = a + v
            a_new = project_simplex(a_new)
            
            J = compute_J_normalized(a_new)
            if J < best_J:
                best_J = J
                best_a = a_new.copy()
            
            a = a_new
    
    return best_a, best_J

def build_A_sparse(a):
    """Build sparse constraint matrix for LP."""
    n = len(a)
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        jl = max(0, k - n + 1)
        jh = min(n - 1, k)
        if jl <= jh:
            cnt = jh - jl + 1
            rows.extend([k] * cnt)
            cols.extend(range(jl, jh + 1))
            seg = a[k - jh:k - jl + 1][::-1]
            vals.extend(seg.tolist())
    A = lil_matrix((2 * n - 1, n))
    for idx in range(len(rows)):
        A[rows[idx], cols[idx]] = vals[idx]
    return A.tocsr()

def lp_best_response(a, M):
    """LP: max sum(g) s.t. (a*g)_k <= M for all k, g >= 0."""
    n = len(a)
    c = -np.ones(n)
    A = build_A_sparse(a)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
    if r.success and r.x is not None:
        g = np.maximum(r.x, 0.0)
        if g.sum() > 1e-10:
            return g
    return None

def lp_polish(a, max_iters=50, deadline=None):
    """LP-based polish. Returns improved a."""
    a = np.maximum(a.astype(np.float64), 0.0001)
    a = a / a.sum()
    best_a = a.copy()
    best_J = compute_J_normalized(a)
    
    for it in range(max_iters):
        if deadline is not None and time.monotonic() > deadline:
            break
        b = fast_conv(a)
        M = b.max()
        for tighten in [0.01, 0.005, 0.002, 0.001]:
            g = lp_best_response(a, M * (1 - tighten))
            if g is not None:
                break
        if g is None:
            break
        g = g / g.sum()
        for t in [0.3, 0.5, 0.7, 0.8]:
            a_try = (1 - t) * a + t * g
            a_try = np.maximum(a_try, 0)
            s = a_try.sum()
            if s < 1e-10:
                continue
            a_try = a_try / s
            J_try = compute_J_normalized(a_try)
            if J_try < best_J:
                best_J = J_try
                best_a = a_try.copy()
                a = a_try
                break
    
    return best_a

def resample_seq(src, dest_n):
    """Resample 1D sequence to new length."""
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)

def generate_inits(n, rng):
    """Generate diverse initial sequences of length n."""
    inits = []
    t = np.linspace(0, 1, n, endpoint=False)
    
    # Uniform
    inits.append(np.ones(n) / n)
    
    # Linear ramp
    ramp = np.arange(1, n + 1, dtype=np.float64)
    inits.append(ramp / ramp.sum())
    
    # Quadratic
    quad = np.arange(1, n + 1, dtype=np.float64) ** 2
    inits.append(quad / quad.sum())
    
    # Sine
    s = 1.0 + 0.5 * np.sin(2 * np.pi * t)
    inits.append(s / s.sum())
    
    # Cosine
    c = 1.0 + 0.5 * np.cos(2 * np.pi * t)
    inits.append(c / c.sum())
    
    # Two-frequency
    two = 1.0 + 0.3 * np.cos(2*np.pi*3*t) + 0.2 * np.cos(2*np.pi*7*t)
    inits.append(np.maximum(two, 0.01) / np.maximum(two, 0.01).sum())
    
    # Random smooth
    for _ in range(3):
        r = rng.random(n)
        # Smooth it
        kernel = np.ones(5) / 5
        r_smooth = np.convolve(r, kernel, mode='same')
        r_smooth = np.maximum(r_smooth, 0.01)
        inits.append(r_smooth / r_smooth.sum())
    
    # Exponential decay
    exp_dec = np.exp(-3 * t)
    inits.append(exp_dec / exp_dec.sum())
    
    # Triangle
    tri = 1.0 - 2.0 * np.abs(t - 0.5)
    inits.append(np.maximum(tri, 0.01) / np.maximum(tri, 0.01).sum())
    
    return inits
