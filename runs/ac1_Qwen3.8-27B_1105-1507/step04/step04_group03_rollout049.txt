            break
        
        # Anneal temperature
        frac = step / max(1, n_steps - 1)
        T = T_start * (T_end / T_start) ** frac
        
        # Anneal learning rate
        lr = lr_start * (lr_end / lr_start) ** frac
        
        # Compute a from x
        x_max = x.max()
        a = np.exp(x - x_max)
        a = a / a.sum()
        
        # Compute gradient of S_T with respect to a
        S_T, grad_a = softmax_objective_and_grad(a, T)
        
        # Convert gradient to x-space
        # dS/dx_j = a_j * grad_a_j - a_j * sum_i(grad_a_i * a_i)
        grad_x = a * (grad_a - np.dot(grad_a, a))
        
        # Adam update
        m = beta1 * m + (1 - beta1) * grad_x
        v = beta2 * v + (1 - beta2) * grad_x ** 2
        m_hat = m / (1 - beta1 ** (step + 1))
        v_hat = v / (1 - beta2 ** (step + 1))
        x = x - lr * m_hat / (np.sqrt(v_hat) + eps_adam)
        
        # Track best
        if step % 100 == 0:
            J = compute_J(a)
            if J < best_J:
                best_J = J
                best_a = a.copy()
                tracker.update(a)
    
    # Final computation
    x_max = x.max()
    a = np.exp(x - x_max)
    a = a / a.sum()
    J = compute_J(a)
    if J < best_J:
        best_J = J
        best_a = a.copy()
    tracker.update(best_a)
    
    return best_a, best_J
