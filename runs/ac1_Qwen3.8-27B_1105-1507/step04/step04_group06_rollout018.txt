        s <<= 1
    fa = np.fft.rfft(a, s)
    return np.fft.irfft(fa * fa, s)[:2 * n - 1]


def compute_J(a):
    """Compute J(a) = 2*n*max(conv(a,a))/(sum(a))^2."""
    a = np.maximum(a, 0)
    s = float(a.sum())
    if s < 1e-10:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))


def coordinate_descent_1d(a, i, s):
    """
    Solve the 1D subproblem for coordinate i.
    Minimize J(x) = 2n * max_conv(x) / (x + s)^2 over x >= 0.
    
    Parameters:
    - a: current sequence (length n)
    - i: coordinate index to optimize
    - s: sum of all other coordinates (sum(a) - a[i])
    
    Returns optimal x.
    """
    n = len(a)
    
    # Compute base convolution (a with a[i] = 0)
    a_base = a.copy()
    a_base[i] = 0.0
    base_conv = fast_conv(a_base)
    
    # Precompute max of unaffected convolution values
    # Unaffected: k < i or k >= i + n
    if i > 0:
        max_left = base_conv[:i].max()
    else:
        max_left = 0.0
    if i + n < 2 * n - 1:
        max_right = base_conv[i + n:].max()
    else:
        max_right = 0.0
    max_unaffected = max(max_left, max_right)
    
    # Precompute affected base values and coefficients
    # For k in [i, i+n-1]: base_conv[k] + 2*x*a[k-i]
    # where k-i ranges from 0 to n-1, i.e., a[0] to a[n-1] = a
    affected_base = base_conv[i:i + n].copy()
    coeffs = a.copy()  # a[k-i] for k = i..i+n-1
    quad_index = i  # index in affected_base where k = 2i, i.e., 2i - i = i
    quad_base = base_conv[2 * i]  # base_conv at k = 2i
    
    def obj(x):
        """Compute J with a[i] = x."""
        # Compute max of affected convolution values
        if x == 0:
            max_affected = max(affected_base.max(), quad_base)
        else:
            vals = affected_base + 2.0 * x * coeffs
            vals[quad_index] = quad_base + x * x
            max_affected = max(vals.max(), quad_base)  # quad_base already included in vals[quad_index]
            # Actually vals[quad_index] = quad_base + x^2, so we just need vals.max()
            max_affected = vals.max()
        total_max = max(max_unaffected, max_affected)
        return 2.0 * n * total_max / ((x + s) * (x + s))
    
    # Golden section search on [0, x_upper]
    # Choose x_upper: large enough that J(x_upper) > J(0)
    x_upper = max(10.0 * a[i] if a[i] > 0 else 10.0 * s / n, 1.0)
    # Expand if needed
    while obj(x_upper) < obj(x_upper * 0.5) and x_upper < 1e6:
        x_upper *= 2
    
    # Use golden section
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    lo = 0.0
    hi = x_upper
    best_x = 0.0
    best_val = obj(0.0)
    
    for _ in range(60):
        t1 = hi - gr * (hi - lo)
        t2 = lo + gr * (hi - lo)
        f1 = obj(t1)
        f2 = obj(t2)
        if f1 < best_val:
            best_val = f1
            best_x = t1
        if f2 < best_val:
            best_val = f2
            best_x = t2
        if f1 < f2:
            hi = t2
        else:
            lo = t1
    
    return best_x


def cd_pass(a, n_max_evals_per_coord=60, verbose=False):
    """One pass of coordinate descent over all coordinates."""
    n = len(a)
    a = a.copy()
    order = np.random.permutation(n)
    improvement = 0.0
    current_J = compute_J(a)
    
    for i in order:
        s = float(a.sum() - a[i])
        if s < 1e-10:
            continue
        x_opt = coordinate_descent_1d(a, i, s)
        if x_opt > a[i]:
            improvement += x_opt - a[i]
        a[i] = max(0.0, x_opt)
    
    # Normalize (doesn't change J but keeps values in reasonable range)
    total = a.sum()
    if total > 1e-10:
        a = a / total
    
    new_J = compute_J(a)
    return a, new_J


def coordinate_descent_full(a0, deadline, tracker, n_passes_max=2000, random_restart_every=50):
    """Run coordinate descent until deadline or convergence."""
    a = np.maximum(a0.astype(np.float64), 0.0)
    a = a / a.sum()
    current_J = compute_J(a)
    best_J = current_J
    best_a = a.copy()
    tracker.update(a)
    
    stall_count = 0
    consecutive_worse = 0
    
    for pass_num in range(n_passes_max):
        if time.monotonic() > deadline:
            break
        
        a, new_J = cd_pass(a)
        
        if new_J < best_J - 1e-12:
            best_J = new_J
            best_a = a.copy()
            tracker.update(a)
            stall_count = 0
            consecutive_worse = 0
        else:
            consecutive_worse += 1
            if new_J > current_J + 1e-10:
                # Revert
                a = best_a.copy()
                new_J = best_J
            stall_count += 1
        
        current_J = new_J
        
        if stall_count % 10 == 0 and verbose:
            elapsed = time.monotonic() - tracker.t_start
            print(f"    CD pass {pass_num}: J={best_J:.8f} stall={stall_count} t={elapsed:.1f}s", flush=True)
        
        # Random perturbation to escape local minima
        if stall_count >= random_restart_every:
            a = best_a.copy()
            n = len(a)
            # Add smooth perturbation
            noise = np.random.normal(0, 0.05, n)
            kernel_size = max(3, n // 32)
            kernel = np.ones(kernel_size) / kernel_size
            noise = np.convolve(noise, kernel, mode='same')
            a = a + noise * a
            a = np.maximum(a, 0.01)
            a = a / a.sum()
            stall_count = 0
        
        if stall_count > 200:
            break
    
    return best_a, best_J


def smooth_obj_and_grad(x, beta):
    """Compute smooth objective and gradient for L-BFGS-B."""
    n = len(x)
    x = np.maximum(x, 0)
    s = float(x.sum())
    if s < 1e-10:
        return 1e10, np.zeros(n)
    
    b = fast_conv(x)
    b_max = float(b.max())
    shifted = beta * (b - b_max)
    shifted = np.clip(shifted, -500, 0)
    exp_shifted = np.exp(shifted)
    Z = float(exp_shifted.sum())
    if Z < 1e-300:
        p = np.zeros(2 * n - 1)
        p[np.argmax(b)] = 1.0
        g_val = b_max
    else:
        p = exp_shifted / Z
        g_val = b_max + np.log(Z) / beta
    
    F = 2.0 * n * g_val / (s * s)
    
    # Gradient: grad_i = 2*summation of p[i+j]*a[j] - 2*g_val/s * (from quotient rule)
    # dF/dx_i = 2n * (s * grad_g_i - 2*g_val) / s^3
    # where grad_g_i = sum_j p_{i+j} * x[j] * 2 (from the convolution gradient)
    
    # Compute sum_j p[i+j] * x[j] for each i
    # This is a cross-correlation-like operation
    # p has length 2n-1, x has length n
    # For each i, sum_j p[i+j] * x[j] where 0 <= j < n and 0 <= i+j < 2n-1
    
    # Use the same trick as before:
    # S[i] = sum_j x[j] * p[i+j]
    # This is the (n-1+i)-th element of convolve(x, p[::-1])
    S = np.convolve(x, p[::-1], mode='full')
    S = S[n-1:2*n-1]  # length n, S[i] = sum_j x[j]*p[n-1-i+j]... 
    
    # Wait, let me re-derive. np.convolve(x, p_rev, 'full')[m] = sum_j x[j] * p_rev[m-j]
    # p_rev[k] = p[2n-2-k]
    # So conv[m] = sum_j x[j] * p[2n-2-m+j]
    # We want S[i] = sum_j x[j] * p[i+j]
    # Set 2n-2-m+j = i+j => 2n-2-m = i => m = 2n-2-i
    # So S[i] = conv[2n-2-i]
    # conv has length n + (2n-1) - 1 = 3n-2
    # m = 2n-2-i, for i in [0, n-1], m in [n-1, 2n-2]
    
    S = np.convolve(x, p[::-1], mode='full')
    grad_g = 2.0 * S[2*n-2-n+1:2*n-2+1][::-1]
    # S[2n-2-i] for i=0..n-1: indices 2n-2, 2n-3, ..., n-1
    # That's S[n-1:2n-1][::-1]
    
    grad_g = 2.0 * S[n-1:2*n-1][::-1]
    
    grad_F = 2.0 * n * (s * grad_g - 2.0 * g_val) / (s ** 3)
    grad_F = np.where(x > 1e-12, grad_F, 0.0)
    
    return F, grad_F


def lbfgs_smooth(a0, deadline, max_time=120):
    """L-BFGS-B optimization with beta annealing."""
    a = np.maximum(a0.astype(np.float64), 1e-8)
    n = len(a)
    
    beta_schedule = [
        (2.0, 30),
        (5.0, 40),
        (10.0, 50),
        (20.0, 60),
        (50.0, 80),
        (100.0, 100),
    ]
    
    for beta, max_iters in beta_schedule:
        if time.monotonic() > deadline:
            break
        
        def func(x):
            f, _ = smooth_obj_and_grad(x, beta)
            return f if np.isfinite(f) else 1e10
        
        def jac(x):
            _, g = smooth_obj_and_grad(x, beta)
            g = np.nan_to_num(g, nan=0.0, posinf=0.0, neginf=0.0)
            return g
        
        bounds = [(0.0, None)] * n
        try:
            res = minimize(lambda x: func(x), a.copy(), jac=lambda x: jac(x),
                          method='L-BFGS-B', bounds=bounds,
                          options={'maxiter': max_iters, 'max_fun': max_iters * 30,
                                   'gtol': 1e-8, 'ftol': 1e-14})
            if res.fun < 1e10:
                new_a = np.maximum(res.x, 1e-10)
                if new_a.sum() > 1e-6:
                    a = new_a
        except Exception:
            pass
    
    return a
