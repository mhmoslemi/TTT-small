"""
Hybrid optimizer combining FFT-based subgradient descent, spectral nudging,
LP mirror descent, and multi-length search to minimize J(a) = 2n*max(conv(a,a))/sum(a)^2.
Key innovations over previous: (1) FFT-accelerated subgradient with momentum,
(2) spectral equalization in Fourier domain, (3) adaptive multi-phase strategy,
(4) diverse structured initializations.

import numpy as np
from scipy.optimize import linprog
from scipy.sparse import csr_matrix
import time


def fast_conv(a):
    """Full self-convolution via FFT."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]


def compute_J(a):
    """Compute J(a) = 2*n*max(conv(a,a))/(sum(a))^2."""
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))


def build_A_sparse(n, a):
    """Build sparse matrix A where A[k,j] = a[k-j] for valid j."""
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        j_min = max(0, k - n + 1)
        j_max = min(n - 1, k)
        for j in range(j_min, j_max + 1):
            rows.append(k)
            cols.append(j)
            vals.append(a[k - j])
    return csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))


def lp_best_response(a, M):
    """Solve LP: max sum(g) s.t. A@g <= M*1, g >= 0."""
    n = len(a)
    A = build_A_sparse(n, a)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    try:
        r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                    options={"presolve": True, "time_limit": 10})
        if r.success and r.x is not None:
            g = np.maximum(r.x, 0)
            if g.sum() > 1e-15:
                return g
    except Exception:
        pass
    return None


def lp_mirror_descent(a, max_iters, deadline, use_multiplicative=False):

    a = np.maximum(a.astype(np.float64), 1e-10)
    a = a / a.sum()
    current_J = compute_J(a)
    best_J = current_J
    best_a = a.copy()
    no_improve = 0

    for it in range(max_iters):

        M = float(b.max())
        g = lp_best_response(a, M)
        if g is None:

        g = np.maximum(g, 0.0)
        g_sum = g.sum()
        if g_sum < 1e-10:
            break
        g = g / g_sum

        base_t = 0.3 * np.exp(-0.4 * it / 150.0) + 0.03
        t = min(1.0, base_t)

        best_trial_J = current_J
        best_trial_a = None

                a_trial = a * np.power(ratio, t)

            a_trial = np.maximum(a_trial, 0.0)

            if s < 1e-10:

            a_trial = a_trial / s

def fft_subgradient_descent(a, max_iters, deadline, momentum_beta=0.7):
    """FFT-accelerated subgradient descent with momentum for min max(conv(a,a))."""
    a = np.maximum(a.astype(np.float64), 1e-10)
    a = a / a.sum()
    n = len(a)
    best_J = compute_J(a)
    best_a = a.copy()
    momentum = np.zeros_like(a)

    size = 1
    while size < 2 * n - 1:
        size <<= 1

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        b = fast_conv(a)
        M = b.max()
        # Active set: constraints within 5% of max
        tol = 0.05 * M
        active = b >= (M - tol)

        # Compute subgradient via FFT
        # g[i] = (2/|K|) * sum_{k in active} a[k-i] (circular)
        indicator = np.zeros(size)
        indicator[:2 * n - 1] = active.astype(np.float64)
        active_count = max(1, int(active.sum()))

        fa = np.fft.rfft(a, size)
        fi = np.fft.rfft(indicator, size)
        g = np.fft.irfft(fa * fi, size)[:n] * (2.0 / active_count)

        # Project gradient onto tangent space (sum = 0)
        g = g - g.mean()

        # Momentum update
        gnorm = np.linalg.norm(g)
        if gnorm < 1e-15:
            break
        g = g / gnorm
        momentum = momentum_beta * momentum + (1 - momentum_beta) * g

        # Adaptive step size
        eta = 2.0 / (np.sqrt(it + 1.0) * (1 + 0.1 * it / max_iters))

        a_new = a - eta * momentum
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s < 1e-10:
            break
        a_new = a_new / s

        J_new = compute_J(a_new)
        if J_new < best_J:
            best_J = J_new
            best_a = a_new.copy()
        a = a_new

    return best_a, best_J

            flat_mag = np.zeros_like(mag)
            flat_mag[0] = np.sqrt(np.sum(mag**2))
            for alpha in [0.1, 0.2, 0.3]:
                blend_mag = (1 - alpha) * mag + alpha * flat_mag
                new_A = blend_mag * phase
                a_trial = np.fft.irfft(new_A, size)[:n]
                a_trial = np.maximum(a_trial, 0.0)
                s_trial = a_trial.sum()
                if s_trial < 1e-10:
                    continue
                a_trial = a_trial / s_trial
                J_trial = compute_J(a_trial)
                if J_trial < best_J - 1e-10:
                    best_J = J_trial
                    best_a = a_trial.copy()
                    a = a_trial.copy()
                    improved = True
                    break

        if not improved and it > 20:
            break

    return best_a, best_J


def cmaes_eigen(C, z, dim):
    """Apply C^(1/2) @ z using eigendecomposition."""
    eigvals, eigvecs = np.linalg.eigh(C)
    eigvals = np.maximum(eigvals, 1e-12)
    root = np.sqrt(eigvals)
    return (eigvecs * root) @ (eigvecs.T @ z)


def fourier_to_seq(params, n, n_harm):
    """Convert Fourier parameters to a non-negative sequence (normalized)."""
    t = np.arange(n, dtype=np.float64) / n
    a = np.full(n, params[0], dtype=np.float64)
    for k in range(1, n_harm + 1):
        a += params[2 * k - 1] * np.cos(2.0 * np.pi * k * t)
        a += params[2 * k] * np.sin(2.0 * np.pi * k * t)
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-8:
        a = np.ones(n) / n
    else:
        a = a / s
    return a


def seq_to_fourier(a, n_harm):
    """Extract Fourier parameters from a sequence."""
    n = len(a)
    a_norm = a / a.sum()
    t = np.arange(n, dtype=np.float64) / n
    params = np.zeros(2 * n_harm + 1)
    params[0] = np.mean(a_norm)
    for k in range(1, n_harm + 1):
        params[2 * k - 1] = 2.0 * np.mean(a_norm * np.cos(2.0 * np.pi * k * t))
        params[2 * k] = 2.0 * np.mean(a_norm * np.sin(2.0 * np.pi * k * t))
    return params


def run_cmaes_fourier(n, n_harm, params0, sigma0, max_gens, deadline):
    """CMA-ES on Fourier parameters."""
    dim = 2 * n_harm + 1
    lam = 4 + int(3 * np.log(dim))
    mu = lam // 2

    weights = np.log(mu + 0.5) - np.log(np.arange(1, mu + 1))
    weights = weights / weights.sum()
    mu_eff = 1.0 / np.sum(weights ** 2)
    c_sigma = (mu_eff + 2) / (dim + mu_eff + 5)
    c_c = (4 + mu_eff / dim) / (dim + 4 + 2 * mu_eff / dim)
    d_sigma = 1 + 2 * max(0, np.sqrt((mu_eff - 1) / (dim + 1) - 1) - 1) + c_sigma
    c_1 = 2 / ((dim + 1.3) ** 2 + mu_eff)
    c_mu = min(1 - c_1, 2 * (mu_eff - 2 + 1.0 / mu_eff) / ((dim + 2) ** 2 + mu_eff))
    chi_N = np.sqrt(dim) * (1 - 1 / (4 * dim) + 1 / (21 * dim ** 2))

    C = np.eye(dim)
    Cs = np.zeros((dim, dim))
    p_sigma = np.zeros(dim)
    p_c = np.zeros(dim)

    x = params0.copy()
    sigma = sigma0

    best_x = params0.copy()
    best_J = compute_J(fourier_to_seq(params0, n, n_harm))

    for gen in range(max_gens):
        if time.monotonic() > deadline:
            break

        offspring = []
        offspring_J = []
        for i in range(lam):
            z = np.random.standard_normal(dim)
            y = cmaes_eigen(C, z, dim)
            x_trial = x + sigma * y
            a_trial = fourier_to_seq(x_trial, n, n_harm)
            J_trial = compute_J(a_trial)
            offspring.append(x_trial)
            offspring_J.append(J_trial)

        idx = np.argsort(offspring_J)
        sorted_x = [offspring[i] for i in idx[:mu]]

        x_new = np.zeros(dim)
        for i in range(mu):
            x_new += weights[i] * sorted_x[i]

        if sigma > 1e-30:
            diff = (x_new - x) / (sigma * chi_N)
        else:
            diff = np.zeros(dim)
        p_sigma = (1 - c_sigma) * p_sigma + np.sqrt(c_sigma * (2 - c_sigma) * mu_eff) * diff
        sigma *= np.exp(c_sigma / d_sigma * (np.linalg.norm(p_sigma) / chi_N - 1))
        sigma = max(sigma, 1e-12)
        sigma = min(sigma, 1.0)

        h_sigma = 1.0
        if np.linalg.norm(p_sigma) / chi_N > (1.4 + c_c) * np.sqrt((1 + c_sigma) / (1 - c_sigma)):
            h_sigma = 0.0
        p_c = (1 - c_c) * p_c + h_sigma * np.sqrt(c_c * (2 - c_c) * mu_eff) * (x_new - x) / max(sigma, 1e-30)

        rank_1 = np.outer(p_c, p_c)
        rank_mu = np.zeros((dim, dim))
        for i in range(mu):
            d = sorted_x[i] - x
            rank_mu += weights[i] * np.outer(d, d)

        Cs = (1 - c_1 - c_mu) * Cs + c_1 * (rank_1 + h_sigma * (1 - h_sigma) * np.outer(p_c, p_c) * c_c) + c_mu * rank_mu
        C = (1 - c_1 - c_mu) * C + c_1 * (rank_1 + h_sigma * (1 - h_sigma) * np.outer(p_c, p_c) * c_c) + c_mu * rank_mu
        C = (C + C.T) / 2

        x = x_new

        if offspring_J[idx[0]] < best_J:
            best_J = offspring_J[idx[0]]
            best_x = offspring[idx[0]].copy()

    return best_x, best_J


def resample_seq(src, dest_n):
    """Resample 1D sequence to new length."""
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)


def structured_perturb(a, rng, perturb_type):
    """Apply a structured perturbation to the sequence."""
    n = len(a)
    t = np.arange(n, dtype=np.float64) / n

    if perturb_type == 'shift':
        s = rng.integers(1, max(2, n // 8))
        a_p = np.roll(a, s)
    elif perturb_type == 'cosine':
        freq = rng.uniform(1, 10)
        amp = rng.uniform(0.03, 0.15)
        a_p = a * (1 + amp * np.cos(2 * np.pi * freq * t + rng.uniform(0, 2 * np.pi)))
    elif perturb_type == 'smooth':
        k = rng.integers(3, max(4, n // 8))
        kernel = np.ones(k) / k
        a_p = np.convolve(a, kernel, mode='same')
    elif perturb_type == 'taper':
        alpha = rng.uniform(0.4, 0.9)
        window = np.ones(n)
        edge = max(1, int(n * (1 - alpha) / 2))
        for i in range(edge):
            frac = i / edge
            window[i] = frac
            window[n - 1 - i] = frac
        a_p = a * window
    elif perturb_type == 'bump':
        center = rng.uniform(0.2, 0.8)
        width = rng.uniform(0.05, 0.3)
        amp = rng.uniform(0.1, 0.5)
        a_p = a + amp * np.exp(-((t - center) ** 2) / (2 * width ** 2))
    else:
        a_p = a + rng.normal(0, 0.05, n)

    a_p = np.maximum(a_p, 0.001)
    s = a_p.sum()
    if s > 1e-8:
        a_p = a_p / s
    return a_p


def generate_inits(n, rng, parent=None):
    """Generate diverse initial sequences."""
    inits = []
    t = np.arange(n, dtype=np.float64) / n

    if parent is not None:
        p = resample_seq(parent, n)
        inits.append(p.copy())
        inits.append(np.maximum(p + rng.normal(0, 0.02, n), 0.01))
        inits.append(np.maximum(p + rng.normal(0, 0.05, n), 0.01))
        inits.append(np.maximum(p + rng.normal(0, 0.1, n), 0.01))

    # Constant
    inits.append(np.ones(n))
    # Cosine bump
    inits.append(0.5 * (1 + np.cos(2 * np.pi * t)))
    # Multi-frequency
    inits.append(np.maximum(0.5 - 0.4 * np.cos(2 * np.pi * t) + 0.1 * np.cos(4 * np.pi * t), 0.01))
    # Gaussian
    inits.append(np.exp(-4 * (t - 0.5) ** 2) + 0.1)
    # Double Gaussian
    inits.append(np.exp(-8 * (t - 0.3) ** 2) + np.exp(-8 * (t - 0.7) ** 2) + 0.05)
    # Triangular
    inits.append(np.maximum(1.0 - 2 * np.abs(t - 0.5), 0.05))
    # Sine wave
    inits.append(np.maximum(1.0 + 0.3 * np.sin(2 * np.pi * 3 * t), 0.01))
    # Barkar-like (5-term made non-negative and extended)
    barkar = np.array([1, 1, 1, -1, 1, -1, 1])
    barkar_pos = np.abs(barker)
    barkar_ext = np.tile(barker_pos, n // len(barker_pos) + 1)[:n]
    inits.append(barker_ext.astype(np.float64))
    # Smooth random
    k = max(5, n // 10)
    kernel = np.ones(k) / k
    smooth = np.convolve(rng.uniform(0, 1, n), kernel, mode='same')
    inits.append(np.maximum(smooth + 0.3, 0.01))
    # Decaying
    inits.append(np.exp(-3 * t) + 0.1)
    # Periodic with random phase
    inits.append(np.maximum(1.0 + 0.2 * np.cos(2 * np.pi * 5 * t + rng.uniform(0, 2 * np.pi)), 0.01))

    result = []
    for a in inits:
        a = np.maximum(a, 0.001)
        s = a.sum()
        if s > 1e-6:
            result.append(a / s)
    return result


class ScoreTracker:
    def __init__(self):
        self.best = None
        self.best_score = float('inf')
        self.t_start = time.monotonic()

    def update(self, a):
        a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
        if a.size == 0 or a.sum() < 0.01:
            return
        score = float(evaluate_sequence(a.tolist()))
        if np.isfinite(score) and score < self.best_score:
            self.best = a.copy()
            self.best_score = score
            elapsed = time.monotonic() - self.t_start
            print(f"  [NEW BEST] J={score:.8f} n={a.size} t={elapsed:.0f}s", flush=True)


def propose_candidate(seed=42, budget_s=1500, **kwargs):
    """

    Multi-phase, multi-length search with diverse initializations.

    deadline = time.monotonic() + float(budget_s) - 30.0
    tracker = ScoreTracker()

    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        if p.ndim == 1 and p.size > 0 and np.all(np.isfinite(p)):
            parent = np.maximum(p, 0.0)
            tracker.update(parent)
            print(f"Parent: n={parent.size}, J={tracker.best_score:.6f}", flush=True)
    except (NameError, AttributeError):
        pass

    if parent is None:
        parent = np.ones(512)
        tracker.update(parent)

    # === Phase 1: Multi-initialization + spectral nudging + FFT subgradient ===
    print(f"\n{'='*50}", flush=True)
    print(f"Phase 1: Multi-init + spectral + FFT-subgrad", flush=True)
    print(f"{'='*50}", flush=True)

    base_n = parent.size
    n_options = sorted(set([
        max(200, base_n - 100),
        base_n,
        min(2000, base_n + 100),
        min(2000, base_n * 2)
    ]))
    n_options = [n for n in n_options if 200 <= n <= 2000]

    best_a = parent.copy()
    best_J = compute_J(parent)

    for n_try in n_options[:3]:
        if time.monotonic() > deadline - 600:

            # FFT subgradient
            a_sub, J_sub = fft_subgradient_descent(a_spec, 300, time.monotonic() + time_per_init)

            # Track
            tracker.update(a_sub)
            if J_sub < best_J:
                best_J = J_sub
                best_a = a_sub.copy()
                print(f"    n={n_try}: J={J_sub:.8f}", flush=True)

    print(f"Phase 1 done, best J={best_J:.8f} (t={time.monotonic()-tracker.t_start:.0f}s)", flush=True)

    # === Phase 2: LP mirror descent on best ===
    if time.monotonic() < deadline - 400:
        print(f"\n{'='*50}", flush=True)
        print(f"Phase 2: LP mirror descent (n={best_a.size})", flush=True)
        print(f"{'='*50}", flush=True)

        a_lp = np.maximum(best_a, 1e-10)
        a_lp = a_lp / a_lp.sum()
        lp_dl = min(deadline - 350, time.monotonic() + 250)
        a_lp, J_lp = lp_mirror_descent(a_lp, 500, lp_dl, use_multiplicative=False)
        tracker.update(a_lp)
        print(f"  LP linear: J={J_lp:.8f}", flush=True)

        if J_lp < best_J:
            best_J = J_lp
            best_a = a_lp.copy()

        # Try multiplicative
        if time.monotonic() < deadline - 320:
            a_lp2 = np.maximum(best_a, 1e-10)
            a_lp2 = a_lp2 / a_lp2.sum()
            a_lp2, J_lp2 = lp_mirror_descent(a_lp2, 200, min(deadline - 300, time.monotonic() + 100), use_multiplicative=True)
            tracker.update(a_lp2)
            print(f"  LP mult: J={J_lp2:.8f}", flush=True)
            if J_lp2 < best_J:
                best_J = J_lp2
                best_a = a_lp2.copy()

    # === Phase 3: Spectral nudging again on LP result ===
    if time.monotonic() < deadline - 250:
        print(f"\n{'='*50}", flush=True)
        print(f"Phase 3: Spectral + FFT-subgrad refinement", flush=True)
        print(f"{'='*50}", flush=True)

        a_spec2, J_spec2 = spectral_nudging(best_a, n_iters=100)
        tracker.update(a_spec2)
        a_sub2, J_sub2 = fft_subgradient_descent(a_spec2, 500, min(deadline - 200, time.monotonic() + 150))
        tracker.update(a_sub2)
        print(f"  Spectral+Subgrad: J={J_sub2:.8f}", flush=True)

        if J_sub2 < best_J:
            best_J = J_sub2
            best_a = a_sub2.copy()

    # === Phase 4: CMA-ES on Fourier parameters ===
    if time.monotonic() < deadline - 200:
        print(f"\n{'='*50}", flush=True)
        print(f"Phase 4: CMA-ES Fourier (n={best_a.size})", flush=True)
        print(f"{'='*50}", flush=True)

        n_cma = best_a.size
        n_harm = min(20, n_cma // 5)
        params0 = seq_to_fourier(best_a, n_harm)

        best_params, J_cma = run_cmaes_fourier(n_cma, n_harm, params0, 0.03, 200, cma_dl)
        a_cma = fourier_to_seq(best_params, n_cma, n_harm)
        tracker.update(a_cma)
        print(f"  CMA-ES: J={J_cma:.8f}", flush=True)

        if J_cma < best_J:
            best_J = J_cma
            best_a = a_cma.copy()

        # LP from CMA
        if time.monotonic() < deadline - 120:
            a_cma_lp = np.maximum(best_a, 1e-10)
            a_cma_lp = a_cma_lp / a_cma_lp.sum()
            a_cma_lp, J_cma_lp = lp_mirror_descent(a_cma_lp, 200, min(deadline - 100, time.monotonic() + 80))
            tracker.update(a_cma_lp)
            if J_cma_lp < best_J:
                best_J = J_cma_lp
                best_a = a_cma_lp.copy()

    # === Phase 5: Structured perturbations + local optimization ===
    if time.monotonic() < deadline - 100:
        print(f"\n{'='*50}", flush=True)
        print(f"Phase 5: Perturbations (n={best_a.size})", flush=True)
        print(f"{'='*50}", flush=True)

        perturb_types = ['shift', 'cosine', 'smooth', 'taper', 'bump', 'cosine', 'shift', 'bump']
        remaining = deadline - 100 - time.monotonic()
        per_pert = max(10, min(40, remaining / len(perturb_types)))

        for ptype in perturb_types:
            if time.monotonic() > deadline - 90:
                break
            a_pert = structured_perturb(best_a, rng, ptype)
            # Quick FFT subgradient
            a_pert, J_pert = fft_subgradient_descent(a_pert, 100, time.monotonic() + per_pert)
            # Quick LP
            a_pert = np.maximum(a_pert, 1e-10)
            a_pert = a_pert / a_pert.sum()
            a_pert, J_pert2 = lp_mirror_descent(a_pert, 50, time.monotonic() + per_pert * 0.5)
            tracker.update(a_pert)
            J_min = min(J_pert, J_pert2)
            if J_min < best_J:
                best_J = J_min
                best_a = a_pert.copy()
                print(f"  [{ptype}] J={J_min:.8f}", flush=True)

    # === Phase 6: Multi-length resampling ===
    if time.monotonic() < deadline - 80:
        print(f"\n{'='*50}", flush=True)
        print(f"Phase 6: Multi-length (base n={best_a.size})", flush=True)
        print(f"{'='*50}", flush=True)

        base_n = best_a.size
        lengths = sorted(set([
            max(256, base_n - 128),
            base_n + 64,
            base_n + 128,
            min(2500, base_n + 256),
            min(2500, base_n * 2)
        ]))
        lengths = [n for n in lengths if 256 <= n <= 2500 and n != base_n]

        for n_try in lengths:
            if time.monotonic() > deadline - 60:
                break
            a_try = resample_seq(best_a, n_try)
            a_try = np.maximum(a_try, 1e-10)
            a_try = a_try / a_try.sum()
            perf = max(10, min(50, (deadline - 60 - time.monotonic()) / max(1, len(lengths) - lengths.index(n_try))))
            a_try, J_try = lp_mirror_descent(a_try, 200, time.monotonic() + perf)
            tracker.update(a_try)
            print(f"  n={n_try}: J={J_try:.8f}", flush=True)
            if J_try < best_J:
                best_J = J_try
                best_a = a_try.copy()

    # === Phase 7: Final long LP + FFT subgradient ===
    if time.monotonic() < deadline - 20:
        print(f"\n{'='*50}", flush=True)
        print(f"Phase 7: Final polish (n={best_a.size})", flush=True)
        print(f"{'='*50}", flush=True)

        a_final = np.maximum(best_a, 1e-10)
        a_final = a_final / a_final.sum()

        # Long LP
        a_final, J_final = lp_mirror_descent(a_final, 5000, deadline - 10, use_multiplicative=False)
        tracker.update(a_final)
        print(f"  Final LP: J={J_final:.8f}", flush=True)

        if J_final < best_J:
            best_J = J_final
            best_a = a_final.copy()

        # FFT subgradient polish
        if time.monotonic() < deadline - 5:
            a_sub_final, J_sub_final = fft_subgradient_descent(best_a, 500, deadline - 2)
            tracker.update(a_sub_final)
            if J_sub_final < best_J:
                best_J = J_sub_final
                best_a = a_sub_final.copy()
                print(f"  Final Subgrad: J={J_sub_final:.8f}", flush=True)

    # Final output
    final_a = tracker.best if tracker.best is not None else best_a
    print(f"\n{'='*50}", flush=True)
    print(f"FINAL: J={tracker.best_score:.8f} n={final_a.size} time={time.monotonic()-tracker.t_start:.0f}s", flush=True)
    print(f"{'='*50}", flush=True)

    return [float(x) for x in final_a]
