def fast_conv(a):
    """Full self-convolution via FFT."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]


def compute_J(a):
    """Compute J(a) = 2*n*max(conv(a,a))/(sum(a))^2."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))


def project_simplex(v):
    """Project v onto probability simplex {x>=0, sum(x)=1}."""
    n = len(v)
    if v.sum() <= 1e-15:
        return np.ones(n) / n
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    rho_idx = np.nonzero(cond)[0]
    if len(rho_idx) == 0:
        return np.ones(n) / n
    rho = rho_idx[-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0)


def smooth_max_grad(a, tau):
    """
    Compute (1/tau)*log(sum exp(tau*c[k])) and its gradient.
    Returns (smooth_value, gradient, raw_conv).
    """
    n = len(a)
    b = fast_conv(a)
    b_max = b.max()
    shifted = tau * (b - b_max)
    # Clamp for numerical safety
    shifted = np.clip(shifted, -500, 500)
    exp_b = np.exp(shifted)
    p = exp_b / exp_b.sum()
    smooth_val = (b_max + np.log(exp_b.sum())) / tau

    grad = np.zeros(n)
    for k in range(2 * n - 1):
        pk = p[k]
        if pk < 1e-14:
            continue
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        # dc[k]/da[j] = 2*a[k-j] for j in [lo, hi]
        grad[lo:hi + 1] += 2.0 * pk * a[k - hi:k - lo + 1][::-1]
    return smooth_val, grad, b


def run_adam_smooth(a, max_iters, deadline, tau_start=30.0, tau_end=400.0,
                    lr=0.05, beta1=0.9, beta2=0.999):
    """Adam optimizer on smooth log-sum-exp objective with projected steps."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 1e-10)
    a = a / a.sum()
    n = len(a)
    m = np.zeros(n)
    v = np.zeros(n)
    best_a = a.copy()
    best_M = fast_conv(a).max()
    t = 0

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        t += 1
        frac = it / max(1, max_iters)
        tau = tau_start + (tau_end - tau_start) * (frac ** 0.7)
        # Adaptive LR with warmup
        lr_eff = lr * min(1.0, t / 10.0) * (1.0 - 0.3 * frac)

        _, grad, b = smooth_max_grad(a, tau)
        M = b.max()
        if M < best_M * (1 - 1e-9):
            best_M = M
            best_a = a.copy()

        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * grad ** 2
        m_hat = m / (1 - beta1 ** t)
        v_hat = v / (1 - beta2 ** t)
        step = lr_eff * m_hat / (np.sqrt(v_hat) + 1e-8)

        a_new = a - step
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s < 1e-10:
            a_new = a.copy()
        else:
            a_new = a_new / s
        a = a_new

    # Check which is better
    if fast_conv(best_a).max() < fast_conv(a).max():
        return best_a
    return a


def build_lp_matrix(n, a):
    """Build sparse matrix A[k,j] = a[k-j] for valid (k,j)."""
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        for j in range(lo, hi + 1):
            rows.append(k)
            cols.append(j)
            vals.append(a[k - j])
    return csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))


def solve_lp(a, M, time_limit=20):
    """Solve LP: max sum(g) s.t. A@g <= M*1, g >= 0. Returns g or None."""
    n = len(a)
    A = build_lp_matrix(n, a)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    try:
        r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                    options={'time_limit': time_limit, 'presolve': True})
        if r.success and r.x is not None:
            g = np.maximum(r.x, 0)
            gs = g.sum()
            if gs > 1e-12:
                return g / gs
    except Exception:
        pass
    return None


def run_lp_bisection(a, max_iters, deadline, M_scale=1.0):
    """LP Mirror Descent with bisection step size."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 1e-10)
    a = a / a.sum()
    n = len(a)
    best_a = a.copy()
    best_M = fast_conv(a).max()
    no_improve = 0

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        b = fast_conv(a)
        M = float(b.max()) * M_scale

        g = solve_lp(a, M, time_limit=15)
        if g is None:
            break

        # Bisection for optimal step size
        lo, hi = 0.0, 1.0
        best_t = 0.0
        for _ in range(18):
            mid = (lo + hi) / 2.0
            a_trial = (1 - mid) * a + mid * g
            a_trial = np.maximum(a_trial, 0.0)
            s = a_trial.sum()
            if s < 1e-10:
                hi = mid
                continue
            a_trial /= s
            M_trial = fast_conv(a_trial).max()
            if M_trial < best_M - 1e-14:
                lo = mid
                best_t = mid
            else:
                hi = mid

        if best_t < 1e-6:
            no_improve += 1
            if no_improve > 15:
                break
            # Try a smaller step
            best_t = 0.01

        a_new = (1 - best_t) * a + best_t * g
        a_new = np.maximum(a_new, 0.0)

        M_new = fast_conv(a_new).max()

            best_M = M_new
            best_a = a_new.copy()
            no_improve = 0
        a = a_new

    return best_a

    # Add delta * a[k-j] for valid k (k-j in [0, n-1])
    k_lo = j
    k_hi = min(2 * n - 2, j + n - 1)
    if k_lo <= k_hi:
        conv_new[k_lo:k_hi + 1] += delta * a[j:j + (k_hi - k_lo + 1)]
    # Correct k = 2j: the change should be x^2 - old^2, not 2*(x-old)*a[j]
    if 0 <= 2 * j <= 2 * n - 2:
        conv_new[2 * j] = conv_old[2 * j] + x * x - old_val * old_val
    s = S_other + x
    if s < 0.01:
        return float('inf')
    return 2.0 * n * conv_new.max() / (s * s)


def coord_descent(a, max_sweeps, deadline, search_range_mult=3.0):
    """Coordinate descent with golden section search per coordinate."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 1e-10)
    n = len(a)
    S = a.sum()
    best_a = a.copy()
    best_M = fast_conv(a).max()
    gr = (np.sqrt(5.0) - 1.0) / 2.0  # ~0.618

    for sweep in range(max_sweeps):
        if time.monotonic() > deadline:
            break
        conv_cur = fast_conv(a)
        improved = False

        for j in range(n):
            if time.monotonic() > deadline:
                break
            old_val = a[j]
            S_other = S - old_val
            # Search range
            lo = 0.0
            hi = max(search_range_mult * old_val, 10.0 / n, 1e-6)
            best_x = old_val
            best_obj = coord_objective(a, j, old_val, conv_cur, S_other)

            # Golden section search
            for _ in range(30):
                x1 = lo + (1.0 - gr) * (hi - lo)
                x2 = lo + gr * (hi - lo)
                obj1 = coord_objective(a, j, x1, conv_cur, S_other)
                obj2 = coord_objective(a, j, x2, conv_cur, S_other)
                if obj1 < best_obj:
                    best_obj = obj1
                    best_x = x1
                if obj2 < best_obj:
                    best_obj = obj2
                    best_x = x2
                if obj1 < obj2:
                    hi = x2
                else:
                    lo = x1
                if hi - lo < 1e-10:
                    break

            if abs(best_x - old_val) > 1e-8:
                a[j] = best_x
                improved = True
                S = S_other + best_x
                conv_cur = fast_conv(a)

        M = conv_cur.max()
        if M < best_M * (1 - 1e-9):
            best_M = M
            best_a = a.copy()
        elif not improved:
            break

    return best_a


def resample_seq(src, dest_n):
    """Resample 1D sequence to new length via linear interpolation."""
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)


class ScoreTracker:
    def __init__(self):
        self.best = None
        self.best_score = float('inf')
        self.t_start = time.monotonic()

    def update(self, a):
        a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
        if a.size == 0 or a.sum() < 0.01:
            return
        score = float(evaluate_sequence(a.tolist()))
        if np.isfinite(score) and score < self.best_score:
            self.best = a.copy()
            self.best_score = score
            elapsed = time.monotonic() - self.t_start
            print(f"  ** J={score:.8f} n={a.size} t={elapsed:.0f}s", flush=True)

    def report(self, label=""):
        if self.best is not None:
            print(f"  {label} best: J={self.best_score:.8f} n={self.best.size}", flush=True)


def propose_candidate(seed=42, budget_s=1500, **kwargs):
    """
    Multi-phase search:
    Phase 1: Adam smooth optimization across multiple lengths
    Phase 2: LP Mirror Descent with bisection step sizes
    Phase 3: Coordinate Descent
    Phase 4: Combined LP + Coord + perturbation
    Phase 5: Final polish and length extension
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + float(budget_s) - 10.0
    tracker = ScoreTracker()

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        if p.ndim == 1 and p.size > 0 and np.all(np.isfinite(p)):
            parent = np.maximum(p, 0.0)
            tracker.update(parent)
            print(f"Parent: n={parent.size}, J={tracker.best_score:.6f}", flush=True)
    except (NameError, AttributeError):
        pass

    if parent is None:
        parent = np.ones(256)
        tracker.update(parent)

    # ============================================================
    # PHASE 1: Adam Smooth Optimization (400s)
    # ============================================================
    p1_deadline = time.monotonic() + 400.0
    p1_deadline = min(p1_deadline, deadline - 600.0)
    print(f"\n{'='*60}", flush=True)
    print(f"PHASE 1: Adam Smooth (deadline at t={p1_deadline - time.monotonic():.0f}s)", flush=True)
    print(f"{'='*60}", flush=True)

    lengths = [320, 384, 448, 512, 576, 640, 768]
    for n in lengths:
        if time.monotonic() > p1_deadline - 30:
            break
        n_deadline = min(time.monotonic() + 50.0, p1_deadline - 20)
        t0 = time.monotonic()

        # Initialize from parent (resampled) and random
        inits = []
        inits.append(resample_seq(parent, n))
        inits.append(np.maximum(rng.uniform(0.3, 1.0, n), 0.01))
        # Asymmetric init (important since symmetric can't beat J=2)
        t_arr = np.linspace(0, 1, n)
        inits.append(np.maximum(0.5 + 0.4 * t_arr + 0.1 * np.sin(4 * np.pi * t_arr), 0.01))
        inits.append(np.maximum(0.5 + 0.3 * (1 - t_arr) ** 2 + 0.1 * np.cos(6 * np.pi * t_arr), 0.01))
        # Two-bump asymmetric
        inits.append(np.maximum(0.2 + np.exp(-8 * (t_arr - 0.3) ** 2) + 0.5 * np.exp(-6 * (t_arr - 0.7) ** 2), 0.01))

        best_n_M = float('inf')
        best_n_a = None

        for a0 in inits:
            if time.monotonic() > n_deadline:
                break
            a0 = np.maximum(a0, 1e-8)
            a0 = a0 / a0.sum()
            iters = max(100, min(800, int(30000 / n)))
            a = run_adam_smooth(a0, iters, n_deadline, tau_start=30, tau_end=350, lr=0.04)
            M_val = fast_conv(a).max()
            if M_val < best_n_M:
                best_n_M = M_val
                best_n_a = a.copy()

        if best_n_a is not None:
            J_val = compute_J(best_n_a)
            print(f"  n={n:4d}: M={best_n_M:.8f} J={J_val:.6f} ({time.monotonic()-t0:.1f}s)", flush=True)
            tracker.update(best_n_a)

    tracker.report("Phase 1 done")

    # ============================================================
    # PHASE 2: LP Mirror Descent with Bisection (450s)
    # ============================================================
    p2_deadline = time.monotonic() + 450.0
    p2_deadline = min(p2_deadline, deadline - 350.0)
    print(f"\n{'='*60}", flush=True)
    print(f"PHASE 2: LP Bisection (deadline at t={p2_deadline - time.monotonic():.0f}s)", flush=True)
    print(f"{'='*60}", flush=True)

    if tracker.best is not None:
        best_a = tracker.best.copy()
        n_best = best_a.size
        # If n is too large for LP, resample to 512
        if n_best > 768:
            best_a = resample_seq(best_a, 640)
            best_a = best_a / best_a.sum()
            tracker.update(best_a)
            n_best = best_a.size

        # First LP pass with scale = 1.0
        a_lp = run_lp_bisection(best_a, 200, min(time.monotonic() + 200, p2_deadline - 100),
                                M_scale=1.0)
        tracker.update(a_lp)

        # Second LP pass with slightly relaxed M
        if time.monotonic() < p2_deadline - 50:
            a_lp2 = run_lp_bisection(tracker.best, 100, min(time.monotonic() + 150, p2_deadline - 30),
                                     M_scale=1.02)
            tracker.update(a_lp2)

    tracker.report("Phase 2 done")

    # ============================================================
    # PHASE 3: Coordinate Descent (300s)
    # ============================================================
    p3_deadline = time.monotonic() + 300.0
    p3_deadline = min(p3_deadline, deadline - 200.0)
    print(f"\n{'='*60}", flush=True)
    print(f"PHASE 3: Coordinate Descent (deadline at t={p3_deadline - time.monotonic():.0f}s)", flush=True)
    print(f"{'='*60}", flush=True)

    if tracker.best is not None:
        n_best = tracker.best.size
        # Coordinate descent is O(n^2) per sweep, so limit n
        if n_best > 640:
            a_cd = resample_seq(tracker.best, 512)
            a_cd = a_cd / a_cd.sum()
        else:
            a_cd = tracker.best.copy()
            a_cd = a_cd / a_cd.sum()

        max_sweeps = max(10, int(200 / max(1, n_best // 64)))
        a_cd = coord_descent(a_cd, max_sweeps, p3_deadline, search_range_mult=2.0)
        tracker.update(a_cd)

        # Second pass with different search range
        if time.monotonic() < p3_deadline - 30:
            a_cd2 = coord_descent(tracker.best, max_sweeps // 2, p3_deadline,
                                  search_range_mult=4.0)
            tracker.update(a_cd2)

    tracker.report("Phase 3 done")

    # ============================================================
    # PHASE 4: Combined LP + Coord + Perturbation (300s)
    # ============================================================
    p4_deadline = time.monotonic() + 300.0
    p4_deadline = min(p4_deadline, deadline - 80.0)
    print(f"\n{'='*60}", flush=True)
    print(f"PHASE 4: Combined (deadline at t={p4_deadline - time.monotonic():.0f}s)", flush=True)
    print(f"{'='*60}", flush=True)

    if tracker.best is not None:
        n_best = tracker.best.size
        if n_best > 768:
            n_best = 640

        for cycle in range(10):
            if time.monotonic() > p4_deadline - 20:
                break

            a = tracker.best.copy().astype(np.float64)
            n = a.size

            # Small perturbation
            noise_scale = 0.02 * (1.0 - 0.08 * cycle)
            a = a + rng.normal(0, noise_scale * 0.01, n)
            a = np.maximum(a, 0.0)
            a = a / a.sum()

            # Short LP
            sub_dl = min(time.monotonic() + 40, p4_deadline - 40)
            a = run_lp_bisection(a, 30, sub_dl, M_scale=1.0)
            tracker.update(a)

            # Short coordinate descent
            if time.monotonic() < p4_deadline - 30 and a.size <= 640:
                a2 = coord_descent(a, 5, min(time.monotonic() + 30, p4_deadline - 30),
                                   search_range_mult=2.0)
                tracker.update(a2)

            # Check for improvement
            elapsed = time.monotonic() - tracker.t_start
            print(f"  cycle {cycle}: J={tracker.best_score:.8f} t={elapsed:.0f}s", flush=True)

    tracker.report("Phase 4 done")

    # ============================================================
    # PHASE 5: Final Polish + Length Extension (remaining)
    # ============================================================
    print(f"\n{'='*60}", flush=True)
    print(f"PHASE 5: Final Polish", flush=True)
    print(f"{'='*60}", flush=True)

    if tracker.best is not None and time.monotonic() < deadline - 5:
        # Final LP pass
        a_final = run_lp_bisection(tracker.best, 50, deadline - 3, M_scale=1.0)
        tracker.update(a_final)

        # Try length extension (2x)
        if time.monotonic() < deadline - 10:
            n_cur = tracker.best.size
            n_ext = min(n_cur * 2, 1024)
            if n_ext > n_cur:
                a_ext = resample_seq(tracker.best, n_ext)
                a_ext = a_ext / a_ext.sum()
                a_ext = run_adam_smooth(a_ext, 100, deadline - 5, tau_start=50,
                                        tau_end=200, lr=0.03)
                tracker.update(a_ext)

        # Try slightly shorter length
        if time.monotonic() < deadline - 5 and tracker.best.size > 256:
            n_short = max(256, tracker.best.size // 2 + 64)
            a_short = resample_seq(tracker.best, n_short)
            a_short = a_short / a_short.sum()
            a_short = run_adam_smooth(a_short, 100, deadline - 3, tau_start=50,
                                      tau_end=200, lr=0.03)
            tracker.update(a_short)

    # Fallback
    if tracker.best is None:
        tracker.best = np.ones(128)
        tracker.best_score = compute_J(tracker.best)

    print(f"\n{'='*60}", flush=True)
    print(f"FINAL: J={tracker.best_score:.8f} n={tracker.best.size} "
          f"time={time.monotonic()-tracker.t_start:.0f}s", flush=True)
    print(f"{'='*60}", flush=True)

    return [float(x) for x in tracker.best]
