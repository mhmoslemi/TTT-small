def frank_wolfe(a, max_iters=200, time_budget=600.0, verbose=False):

        if g is None:

                a = a * np.exp(rng.normal(0, 0.08, n))

            a = np.maximum(a, 0)

            stall = 0
            stagnation = 0
            if verbose and it % 20 == 0:

def subgradient_polish(a, max_iters=5000, time_budget=60.0):
    """Subgradient descent with full active set and adaptive step."""
    n = len(a)
    a = a.copy().astype(np.float64)
    a = a / np.sum(a)
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    
    v = np.zeros(n)
    momentum = 0.9
    step = 0.05
    stall = 0
    
    for it in range(max_iters):
        if time.monotonic() - start > time_budget:
            break
        
        conv = compute_conv(a)
        M = float(np.max(conv))
        
        # Active set: lags within 5% of max
        threshold = M * 0.95
        active = np.where(conv >= threshold)[0]
        
        # Subgradient
        grad = np.zeros(n)
        for k in active:
            for i in range(n):
                j = k - i
                if 0 <= j < n:
                    grad[i] += 2.0 * a[j]
        grad /= len(active)
        
        # Center (preserve sum)
        grad_c = grad - np.mean(grad)
        
        v = momentum * v + (1 - momentum) * grad_c
        a_new = a - step * v
        a_new = np.maximum(a_new, 0)
        S = np.sum(a_new)
        if S < 1e-10:
            break
        a_new = a_new / S
        
        obj = compute_obj(a_new)
        if obj < best_obj - 1e-14:
            best_obj = obj
            best_a = a_new.copy()
            a = a_new
            stall = 0
        else:
            stall += 1
            if stall % 50 == 0:
                step *= 0.95
            if stall > 500:
                # Escape
                rng = np.random.default_rng(it * 31)
                a = a * np.exp(rng.normal(0, 0.03, n))
                a = np.maximum(a, 0)
                a = a / np.sum(a)
                v = np.zeros(n)
                step = 0.05
                stall = 0
    
    return best_a, best_obj

def compute_conv(a):
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    conv = np.fft.irfft(fa * fa, size)
    return conv[:2 * n - 1]


def compute_obj(a):
    n = len(a)
    S = float(np.sum(a))
    if S < 1e-10:
        return np.inf
    conv = compute_conv(a)
    M = float(np.max(conv))
    return 2.0 * n * M / (S * S)


def build_toeplitz_sparse(a):
    n = len(a)
    m = 2 * n - 1
    nnnz = n * n
    rows = np.empty(nnnz, dtype=np.int32)
    cols = np.empty(nnnz, dtype=np.int32)
    vals = np.empty(nnnz, dtype=np.float64)
    idx = 0
    for j in range(n):
        for i in range(n):
            rows[idx] = j + i
            cols[idx] = j
            vals[idx] = a[i]
            idx += 1
    return coo_matrix((vals, (rows, cols)), shape=(m, n)).tocsr()


_toeplitz_cache = {}


def get_toeplitz(a):

        a_c += coeff * np.cos(2 * np.pi * m * k / n)
    a_c = np.maximum(a_c, 0)
    S = np.sum(a_c)
    if S < 1e-10:
        return a
    return a_c / S


def propose_candidate():
    start_time = time.monotonic()
    deadline = start_time + 1470.0

    # Get parent
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64)
        parent = np.clip(parent, 0, 1000.0)
    except Exception:
        parent = np.ones(500) / 500

    n_parent = len(parent)
    a = parent / np.sum(parent)
    best_a = a.copy()
    best_obj = compute_obj(a)
    print(f"Parent (n={n_parent}): obj={best_obj:.8f}")

    # Generate initial candidates from various transformations
    candidates = []
    transforms = [
        ("identity", a.copy()),
        ("symmetric", symmetrize(a)),
        ("cosine_smooth", cosine_smooth(a)),
        ("windowed_1", windowed(a, 1.0)),
        ("windowed_2", windowed(a, 2.0)),
    ]
    for name, a_c in transforms:
        S = np.sum(a_c)
        if S > 1e-10:
            a_c = a_c / S
            obj_c = compute_obj(a_c)
            candidates.append((name, a_c, obj_c))
            if obj_c < best_obj:
                best_obj = obj_c
                best_a = a_c.copy()
                print(f"  {name}: {obj_c:.8f} ** NEW BEST **")
            else:
                print(f"  {name}: {obj_c:.8f}")

    # === Phase 1: Frank-Wolfe main optimization ===
    print("\n=== Phase 1: Frank-Wolfe (multi-target LP) ===")
    remaining = deadline - time.monotonic()
    fw_budget = min(600.0, remaining * 0.45)
    a_fw, obj_fw = frank_wolfe_multi_target(best_a, max_iters=250, time_budget=fw_budget, verbose=True)
    if obj_fw < best_obj:
        best_obj = obj_fw
        best_a = a_fw.copy()
        print(f"  After Phase 1 FW: {best_obj:.8f}")
    else:
        print(f"  Phase 1 FW: {obj_fw:.8f} (no improvement)")

    # === Phase 2: Subgradient polish ===
    print("\n=== Phase 2: Full active-set subgradient ===")
    remaining = deadline - time.monotonic()
    sg_budget = min(150.0, remaining * 0.15)
    a_sg, obj_sg = full_active_subgradient(best_a, max_iters=30000, time_budget=sg_budget)
    if obj_sg < best_obj:
        best_obj = obj_sg
        best_a = a_sg.copy()
        print(f"  After Phase 2 SG: {best_obj:.8f}")
    else:
        print(f"  Phase 2 SG: {obj_sg:.8f} (best stays {best_obj:.8f})")

    # === Phase 3: Second FW pass ===
    print("\n=== Phase 3: Second FW pass ===")
    remaining = deadline - time.monotonic()
    if remaining > 120:
        fw2_budget = min(300.0, remaining * 0.35)
        a_fw2, obj_fw2 = frank_wolfe_multi_target(best_a, max_iters=150, time_budget=fw2_budget, verbose=True)
        if obj_fw2 < best_obj:
            best_obj = obj_fw2
            best_a = a_fw2.copy()
            print(f"  After Phase 3 FW: {best_obj:.8f}")
    else:
        print("  Skipping (insufficient time)")

    # === Phase 4: Try alternative starting points ===
    print("\n=== Phase 4: Alternative starts ===")
    remaining = deadline - time.monotonic()
    if remaining > 80:
        # Try the second-best initial candidate
        candidates.sort(key=lambda x: x[2])
        tried = set()
        for name, a_c, obj_c in candidates:
            if name in tried:
                continue
            if time.monotonic() > deadline - 50:
                break
            if obj_c > best_obj + 0.02:
                break
            tried.add(name)
            n_c = len(a_c)
            alloc = min(remaining * 0.2, 60.0)
            a_opt, obj_opt = frank_wolfe_multi_target(a_c, max_iters=50, time_budget=alloc)
            if obj_opt < best_obj:
                best_obj = obj_opt
                best_a = a_opt.copy()
                print(f"  {name} (n={n_c}): {best_obj:.8f} ** NEW BEST **")
            else:
                print(f"  {name} (n={n_c}): {obj_opt:.8f}")

    # === Phase 5: Multi-n exploration ===
    print("\n=== Phase 5: Multi-n exploration ===")
    remaining = deadline - time.monotonic()
    if remaining > 100:
        n_base = len(best_a)
        n_tries = sorted(set([
            max(100, n_base - 50), max(100, n_base - 100),
            n_base + 50, n_base + 100, n_base + 150
        ]))
        n_tries = [n for n in n_tries if 100 <= n <= 2000]

        for n_try in n_tries:
            if time.monotonic() > deadline - 40:
                break
            x_old = np.linspace(0, 1, n_base)
            x_new = np.linspace(0, 1, n_try)
            try:
                f_interp = interp1d(x_old, best_a, kind='cubic', fill_value='extrapolate')
                a_new = np.maximum(f_interp(x_new), 0)
            except Exception:
                a_new = np.ones(n_try) / n_try

            S = np.sum(a_new)
            if S < 1e-10:
                continue
            a_new = a_new / S

            # Symmetrize after interpolation
            a_new = symmetrize(a_new)
            S = np.sum(a_new)
            if S > 1e-10:
                a_new = a_new / S

            alloc = min((deadline - time.monotonic() - 30) * 0.3, 50.0)
            if alloc < 10:
                break
            a_opt, obj_opt = frank_wolfe_multi_target(a_new, max_iters=40, time_budget=alloc)
            if obj_opt < best_obj:
                best_obj = obj_opt
                best_a = a_opt.copy()
                print(f"  n={n_try}: {best_obj:.8f} ** NEW BEST **")
            else:
                print(f"  n={n_try}: {obj_opt:.8f}")

    # === Phase 6: Final intensive polish ===
    print("\n=== Phase 6: Final polish ===")
    remaining = deadline - time.monotonic()
    if remaining > 30:
        # Short FW
        alloc = min(remaining * 0.5, 60.0)
        a_fw3, obj_fw3 = frank_wolfe_multi_target(best_a, max_iters=80, time_budget=alloc)
        if obj_fw3 < best_obj:
            best_obj = obj_fw3
            best_a = a_fw3.copy()
            print(f"  Final FW: {best_obj:.8f}")

        # Short SG
        remaining = deadline - time.monotonic()
        if remaining > 20:
            alloc = min(remaining * 0.7, 40.0)
            a_sg2, obj_sg2 = full_active_subgradient(best_a, max_iters=20000, time_budget=alloc)
            if obj_sg2 < best_obj:
                best_obj = obj_sg2
                best_a = a_sg2.copy()
                print(f"  Final SG: {best_obj:.8f}")

    # Convert to output list
    result = [float(max(0.0, min(1000.0, x))) for x in best_a]
    val = evaluate_sequence(result)
    if val < np.inf and val < best_obj:
        best_obj = val

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_obj:.8f}, time={elapsed:.1f}s ===")

    return result
