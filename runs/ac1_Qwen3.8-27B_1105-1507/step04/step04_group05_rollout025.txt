def coordinate_descent_pass_fast(a):
    """Fast coordinate descent with exact 1D optimization per coordinate."""
    a = np.maximum(a.astype(np.float64), 0.0)
    S = a.sum()
    if S < 1e-10:
        return a
    n = len(a)
    b = fast_conv(a)
    
    # Process coordinates in random order
    indices = np.random.permutation(n)
    
    for i in indices:
        ai = a[i]
        
        # Precompute the parts that don't depend on x
        # For k in [i, i+n-1]:
        #   if k == 2i: b_new[k] = b[k] - ai^2 + x^2
        #   else: b_new[k] = b[k] - 2*ai*a[k-i] + 2*x*a[k-i]
        # For k outside: b_new[k] = b[k]
        
        # C = max of b[k] for k outside [i, i+n-1]
        k_start = max(0, i)
        k_end = min(2*n-1, i+n)  # exclusive
        # Actually k goes from i to i+n-1 (inclusive) for the changed ones
        # But k must be in [0, 2n-2]
        k_lo = i
        k_hi = min(2*n-2, i+n-1)
        
        # Max of unchanged values
        if k_lo > 0:
            c_left = b[:k_lo].max()
        else:
            c_left = -np.inf
        if k_hi < 2*n-2:
            c_right = b[k_hi+1:].max()
        else:
            c_right = -np.inf
        C = max(c_left, c_right)
        
        # For the changed values, precompute:
        # b_new[k] = base[k] + slope[k]*x for k != 2i
        # b_new[2i] = base_2i + x^2
        bases = []
        slopes = []
        for k in range(k_lo, k_hi+1):
            ki = k - i
            if 0 <= ki < n:
                if k == 2*i:
                    bases.append(b[k] - ai*ai)  # + x^2 (handled separately)
                    slopes.append(0.0)
                else:
                    bases.append(b[k] - 2.0*ai*a[ki])
                    slopes.append(2.0*a[ki])
        
        bases = np.array(bases)
        slopes = np.array(slopes)
        D = S - ai  # denominator base: (D + x)^2
        
        # Golden section search for optimal x
        gr = (np.sqrt(5.0) - 1.0) / 2.0
        lo = 0.0
        hi = max(5.0 * ai, S / n * 3.0, 0.01)
        
        def J_of_x(x):
            # Compute max convolution
            max_c = C
            if len(bases) > 0:
                # Linear parts
                lin_vals = bases + slopes * x
                max_c = max(max_c, lin_vals.max())
                # Quadratic part (k = 2i)
                quad_idx = np.where(slopes == 0.0)[0]
                for qi in quad_idx:
                    max_c = max(max_c, bases[qi] + x*x)
            new_S = D + x
            if new_S < 1e-10:
                return 1e10
            return 2.0 * n * max_c / (new_S * new_S)
        
        c_point = hi - gr * (hi - lo)
        d_point = lo + gr * (hi - lo)
        fc = J_of_x(c_point)
        fd = J_of_x(d_point)
        
        for _ in range(50):
            if fc < fd:
                hi = d_point
                d_point = c_point
                fd = fc
                c_point = hi - gr * (hi - lo)
                fc = J_of_x(c_point)
            else:
                lo = c_point
                c_point = d_point
                fc = fd
                d_point = lo + gr * (hi - lo)
                fd = J_of_x(d_point)
        
        x_opt = (lo + hi) / 2.0
        x_opt = max(0.0, x_opt)
        
        if abs(x_opt - ai) > 1e-10:

            a[i] = x_opt

                ki = k - i
                if 0 <= ki < n:
                    if k == 2*i:
                        b[k] = b[k] - ai*ai + x_opt*x_opt
                    else:

    b_check = fast_conv(a)

def build_A_sparse(a):
    n = len(a)
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        jl = max(0, k - n + 1)
        jh = min(n - 1, k)
        if jl <= jh:
            for j in range(jl, jh + 1):
                rows.append(k)
                cols.append(j)
                vals.append(a[k - j])
    A = lil_matrix((2 * n - 1, n))
    for idx in range(len(rows)):
        A[rows[idx], cols[idx]] = vals[idx]
    return A.tocsr()

def lp_best_response(a, M):
    n = len(a)
    c = -np.ones(n)
    A = build_A_sparse(a)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
    if r.success and r.x is not None:
        g = np.maximum(r.x, 0.0)
        if g.sum() > 1e-10:
            return g
    return None

    stall = 0

        if time.monotonic() > deadline:
            break
        
        b = fast_conv(a)
        M = float(b.max())
        
        # Try to tighten

            tighten *= 0.7
            if tighten < 1e-5:

        g = g / g.sum()
        
        # Line search on step size

            if s < 1e-10:

        current_J = best_J

        # Decrease tightening as we progress
        tighten = max(0.001, tighten * 0.995)

    return a

Minimizes J(a) = 2*n*max(convolve(a,a))/(sum(a))^2 using:

import numpy as np
from scipy.optimize import minimize, linprog
from scipy.sparse import lil_matrix

def fast_conv(a):
    """Full self-convolution via FFT."""
    n = len(a)
    s = 1
    while s < 2 * n - 1:
        s <<= 1
    fa = np.fft.rfft(a, s)
    return np.fft.irfft(fa * fa, s)[:2 * n - 1]


def compute_J(a):
    """Compute J(a) = 2*n*max(conv(a,a))/(sum(a))^2."""
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))


def smooth_obj_and_grad(z, tau):

    a = softmax(z), so a > 0 and sum(a) = 1."""

    n = len(a)

    b = fast_conv(a)

    # Log-sum-exp of tau * b
    tau_b = tau * b
    max_tb = tau_b.max()
    exp_tb = np.exp(tau_b - max_tb)
    sum_exp = exp_tb.sum()
    p = exp_tb / sum_exp

    f_val = 2.0 * n * (max_tb + np.log(sum_exp)) / tau

    # Gradient of smooth max w.r.t. a[i] = 2 * sum_j p[i+j] * a[j]
    grad_f = np.zeros(n)
    for i in range(n):
        max_j = min(n, 2 * n - 1 - i)
        if max_j > 0:
            grad_f[i] = 2.0 * np.dot(p[i:i + max_j], a[:max_j])

    # Chain rule through softmax: grad_z[j] = grad_f[j]*a[j] - a[j]*sum_i(grad_f[i]*a[i])
    dot_prod = np.dot(grad_f, a)
    grad_z = grad_f * a - a * dot_prod

    return f_val, grad_z


def smooth_obj_wrapper(z, tau):
    val, _ = smooth_obj_and_grad(z, tau)
    return val


def smooth_grad_wrapper(z, tau):
    _, grad = smooth_obj_and_grad(z, tau)
    return grad


def run_smooth_optimization(a_init, deadline, tracker):
    """Run L-BFGS smooth optimization with continuation on tau."""
    a = np.maximum(a_init.astype(np.float64), 1e-10)
    a = a / a.sum()
    n = len(a)
    z = np.log(a)

    current_J = compute_J(a)
    tracker.update(a)

    tau_schedule = [1.0, 3.0, 6.0, 12.0, 25.0, 50.0, 100.0, 200.0, 400.0, 800.0]

    for tau in tau_schedule:
        if time.monotonic() > deadline:
            break

        t0 = time.monotonic()
        try:
            result = minimize(
                smooth_obj_wrapper,
                z,
                args=(tau,),
                method='L-BFGS-B',
                jac=smooth_grad_wrapper,
                options={'maxiter': 800, 'maxfun': 20000, 'ftol': 1e-16, 'gtol': 1e-14}
            )
            z = result.x
        except Exception:
            break

        z_shifted = z - z.max()
        exp_z = np.exp(z_shifted)
        a = exp_z / exp_z.sum()
        a = np.maximum(a, 1e-10)
        a = a / a.sum()

        J_new = compute_J(a)
        if J_new < current_J:
            current_J = J_new
            tracker.update(a)

        elapsed = time.monotonic() - t0
        print(f"    tau={tau}: J={compute_J(a):.8f}, time={elapsed:.1f}s, iters={result.nit}", flush=True)

    return a


def J_of_x(x, bases, slopes, C, D, n):
    """Compute J when a[i]=x given precomputed arrays."""
    max_c = C
    if len(bases) > 0:
        lin_vals = bases + slopes * x
        max_c = max(max_c, np.max(lin_vals))
        quad_mask = (slopes == 0.0)
        if np.any(quad_mask):
            quad_vals = bases[quad_mask] + x * x
            max_c = max(max_c, np.max(quad_vals))
    new_S = D + x
    if new_S < 1e-10:
        return 1e10
    return 2.0 * n * max_c / (new_S * new_S)


def coordinate_descent_pass(a, temperature=0.0):
    """One full pass of exact 1D coordinate descent with optional annealing."""
    a = np.maximum(a.astype(np.float64), 0.0)
    S = a.sum()
    if S < 1e-10:
        return a, float('inf')
    n = len(a)
    b = fast_conv(a)

    indices = np.random.permutation(n)
    gr = (np.sqrt(5.0) - 1.0) / 2.0

    for i in indices:
        ai = a[i]
        k_lo = i
        k_hi = min(2 * n - 2, i + n - 1)

        # Max of unchanged convolution values
        if k_lo > 0:
            c_left = b[:k_lo].max()
        else:
            c_left = -np.inf
        if k_hi < 2 * n - 2:
            c_right = b[k_hi + 1:].max()
        else:
            c_right = -np.inf
        C = max(c_left, c_right)

        bases = np.zeros(k_hi - k_lo + 1)
        slopes = np.zeros(k_hi - k_lo + 1)
        for idx, k in enumerate(range(k_lo, k_hi + 1)):
            ki = k - i
            if 0 <= ki < n:
                if k == 2 * i:
                    bases[idx] = b[k] - ai * ai
                    slopes[idx] = 0.0
                else:
                    bases[idx] = b[k] - 2.0 * ai * a[ki]
                    slopes[idx] = 2.0 * a[ki]

        D = S - ai

        lo = 0.0
        hi = max(5.0 * ai, S / n * 4.0, 0.01)

        c_pt = hi - gr * (hi - lo)
        d_pt = lo + gr * (hi - lo)
        fc = J_of_x(c_pt, bases, slopes, C, D, n)
        fd = J_of_x(d_pt, bases, slopes, C, D, n)

        for _ in range(50):
            if fc < fd:
                hi = d_pt
                d_pt = c_pt
                fd = fc
                c_pt = hi - gr * (hi - lo)
                fc = J_of_x(c_pt, bases, slopes, C, D, n)
            else:
                lo = c_pt
                c_pt = d_pt
                fc = fd
                d_pt = lo + gr * (hi - lo)
                fd = J_of_x(d_pt, bases, slopes, C, D, n)

        x_opt = (lo + hi) / 2.0
        x_opt = max(0.0, x_opt)

        # Simulated annealing: occasionally perturb
        if temperature > 0.0:
            if np.random.random() < 0.1:
                x_opt = max(0.0, x_opt * (1.0 + np.random.normal(0, temperature)))

        if abs(x_opt - ai) > 1e-12:
            a[i] = x_opt
            for k in range(k_lo, k_hi + 1):
                ki = k - i
                if 0 <= ki < n:
                    if k == 2 * i:
                        b[k] = b[k] - ai * ai + x_opt * x_opt
                    else:
                        b[k] = b[k] - 2.0 * ai * a[ki] + 2.0 * x_opt * a[ki]
            S = S - ai + x_opt

    b_check = fast_conv(a)
    new_J = 2.0 * n * b_check.max() / (S * S)
    return a, new_J


def run_coordinate_descent(a, deadline, tracker, max_passes=100, use_annealing=False):
    """Run multiple passes of coordinate descent."""
    a = np.maximum(a.astype(np.float64), 0.0)
    a = a / a.sum()
    current_J = compute_J(a)
    tracker.update(a)

    no_improve = 0
    for p in range(max_passes):
        if time.monotonic() > deadline:
            break

        temp = 0.0
        if use_annealing:
            progress = p / max_passes
            temp = 0.05 * (1.0 - progress) ** 2

        t0 = time.monotonic()
        a, new_J = coordinate_descent_pass(a, temperature=temp)
        elapsed = time.monotonic() - t0

        if new_J < current_J - 1e-11:
            current_J = new_J
            tracker.update(a)
            no_improve = 0
        else:
            no_improve += 1

        if p % 10 == 0:
            print(f"    CD pass {p}: J={current_J:.8f} ({elapsed:.2f}s)", flush=True)

        if no_improve > 8 and not use_annealing:
            break

    tracker.update(a)
    return a


def build_A_sparse(a):
    """Build sparse constraint matrix for LP."""
    n = len(a)
    A = lil_matrix((2 * n - 1, n), dtype=np.float64)
    for k in range(2 * n - 1):
        jl = max(0, k - n + 1)
        jh = min(n - 1, k)
        for j in range(jl, jh + 1):
            A[k, j] = a[k - j]
    return A.tocsr()


def lp_best_response(a, M):
    """LP: max sum(g) s.t. (a*g)[k] <= M for all k, g >= 0."""
    n = len(a)
    c = -np.ones(n)
    A = build_A_sparse(a)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
    if r.success and r.x is not None:
        g = np.maximum(r.x, 0.0)
        if g.sum() > 1e-10:
            return g
    return None


def run_lp_refinement(a, deadline, tracker, max_iters=300):
    """LP-based best response refinement with adaptive tightening."""
    a = np.maximum(a.astype(np.float64), 0.001)
    a = a / a.sum()
    current_J = compute_J(a)
    tracker.update(a)

    tighten = 0.004
    stall = 0

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        b = fast_conv(a)
        M = float(b.max())

        M_target = M * (1.0 - tighten)
        g = lp_best_response(a, M_target)
        if g is None:
            M_target = M * (1.0 - tighten * 0.5)
            g = lp_best_response(a, M_target)
        if g is None:
            tighten *= 0.7
            if tighten < 1e-6:
                break
            continue

        g = np.maximum(g, 0.0)
        gs = g.sum()
        if gs < 1e-10:
            break
        g = g / gs

        # Line search
        best_t = 0.0
        best_J = current_J
        for t in [0.05, 0.1, 0.2, 0.3, 0.5, 0.7, 0.85]:
            a_trial = (1.0 - t) * a + t * g
            a_trial = np.maximum(a_trial, 0.0)
            s = a_trial.sum()
            if s < 1e-10:
                continue
            a_trial = a_trial / s
            J_trial = compute_J(a_trial)
            if J_trial < best_J - 1e-14:
                best_J = J_trial
                best_t = t

        if best_t == 0.0:
            stall += 1
            if stall > 40:
                break
            tighten *= 0.8
            continue

        a = (1.0 - best_t) * a + best_t * g
        a = np.maximum(a, 0.0)
        a = a / a.sum()
        current_J = best_J
        stall = 0
        tighten = max(0.001, tighten * 0.997)

        if it % 25 == 0:
            tracker.update(a)

    tracker.update(a)
    return a


def resample_seq(src, dest_n):
    """Resample 1D sequence to new length via linear interpolation."""
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)


class ScoreTracker:
    def __init__(self):
        self.best = None
        self.best_score = float('inf')
        self.t_start = time.monotonic()

    def update(self, a):
        a_arr = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
        if a_arr.size == 0 or a_arr.sum() < 0.01:
            return
        score = float(evaluate_sequence(a_arr.tolist()))
        if np.isfinite(score) and score < self.best_score:
            self.best = a_arr.copy()
            self.best_score = score
            elapsed = time.monotonic() - self.t_start
            print(f"  * NEW BEST: J={score:.8f} n={a_arr.size} t={elapsed:.0f}s", flush=True)

    def report(self, label=""):
        if self.best is not None:
            print(f"  {label} best: J={self.best_score:.8f} n={self.best.size} t={time.monotonic()-self.t_start:.0f}s", flush=True)


def propose_candidate(seed=42, budget_s=1500, **kwargs):
    """
    Main search function combining smooth L-BFGS, coordinate descent, LP refinement,
    multi-length exploration, and perturbation restarts.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + float(budget_s) - 30.0
    tracker = ScoreTracker()

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        if p.ndim == 1 and p.size > 0 and np.all(np.isfinite(p)):
            parent = np.maximum(p, 0.0)
            tracker.update(parent)
            print(f"Parent loaded: n={parent.size}, J={tracker.best_score:.8f}", flush=True)
    except (NameError, AttributeError):
        pass

    if parent is None:
        parent = np.ones(384)
        tracker.update(parent)
        print(f"No parent, using uniform n=384", flush=True)

    a = parent.copy()
    a = a / a.sum()

    # Phase 1: Smooth L-BFGS with continuation
    if time.monotonic() < deadline - 600:
        print(f"\n{'='*50} Phase 1: Smooth L-BFGS (n={a.size}) {'='*50}", flush=True)
        t0 = time.monotonic()
        a = run_smooth_optimization(a, deadline - 600, tracker)
        tracker.report(f"Phase 1 L-BFGS: {time.monotonic()-t0:.1f}s")

    # Phase 2: Coordinate descent (exact)
    if time.monotonic() < deadline - 500:
        print(f"\n{'='*50} Phase 2: Coordinate Descent {'='*50}", flush=True)
        t0 = time.monotonic()
        a = run_coordinate_descent(tracker.best, deadline - 500, tracker, max_passes=80)
        tracker.report(f"Phase 2 CD: {time.monotonic()-t0:.1f}s")

    # Phase 3: LP refinement
    if time.monotonic() < deadline - 400:
        print(f"\n{'='*50} Phase 3: LP Refinement {'='*50}", flush=True)
        t0 = time.monotonic()
        a = run_lp_refinement(tracker.best, deadline - 400, tracker, max_iters=250)
        tracker.report(f"Phase 3 LP: {time.monotonic()-t0:.1f}s")

    # Phase 4: More CD after LP
    if time.monotonic() < deadline - 300:
        print(f"\n{'='*50} Phase 4: CD after LP {'='*50}", flush=True)
        t0 = time.monotonic()
        a = run_coordinate_descent(tracker.best, deadline - 300, tracker, max_passes=60)
        tracker.report(f"Phase 4 CD: {time.monotonic()-t0:.1f}s")

    # Phase 5: LP again on CD result
    if time.monotonic() < deadline - 250:
        print(f"\n{'='*50} Phase 5: LP on CD result {'='*50}", flush=True)
        t0 = time.monotonic()
        a = run_lp_refinement(tracker.best, deadline - 250, tracker, max_iters=150)
        # Follow with CD
        if time.monotonic() < deadline - 200:
            a = run_coordinate_descent(tracker.best, deadline - 200, tracker, max_passes=40)
        tracker.report(f"Phase 5 LP+CD: {time.monotonic()-t0:.1f}s")

    # Phase 6: Nearby lengths
    if time.monotonic() < deadline - 100:
        print(f"\n{'='*50} Phase 6: Nearby Lengths {'='*50}", flush=True)
        t0 = time.monotonic()

        n_best = tracker.best.size
        nearby_lengths = sorted(set([
            n_best - 2, n_best - 4, n_best + 2, n_best + 4,
            n_best - 6, n_best + 6, n_best + 8, n_best + 12
        ]))
        nearby_lengths = [l for l in nearby_lengths if 250 <= l <= 550]

        for nl in nearby_lengths:
            if time.monotonic() > deadline - 80:
                break

            a_res = resample_seq(tracker.best, nl)
            a_res = np.maximum(a_res, 0.001)
            a_res = a_res / a_res.sum()

            # Quick smooth + CD
            a_trial = run_smooth_optimization(a_res, time.monotonic() + 20, tracker)
            if time.monotonic() < deadline - 80:
                a_trial, _ = coordinate_descent_pass(a_trial)
                a_trial = a_trial / a_trial.sum()
                tracker.update(a_trial)

            print(f"    Length {nl}: best J={tracker.best_score:.8f}", flush=True)

        tracker.report(f"Phase 6 Lengths: {time.monotonic()-t0:.1f}s")

    # Phase 7: Perturbation restarts with annealing
    if time.monotonic() < deadline - 60:
        print(f"\n{'='*50} Phase 7: Perturbation Restarts {'='*50}", flush=True)
        t0 = time.monotonic()
        n = tracker.best.size

        for temp in [0.03, 0.02, 0.01, 0.005, 0.002]:
            if time.monotonic() > deadline - 50:
                break

            a_pert = tracker.best.copy() * (1.0 + rng.normal(0, temp, n))
            a_pert = np.maximum(a_pert, 1e-10)
            a_pert = a_pert / a_pert.sum()

            # CD with annealing
            a_pert, new_J = coordinate_descent_pass(a_pert, temperature=temp)
            a_pert = a_pert / a_pert.sum()
            tracker.update(a_pert)

            if time.monotonic() < deadline - 50:
                a_pert, new_J = coordinate_descent_pass(a_pert)
                a_pert = a_pert / a_pert.sum()
                tracker.update(a_pert)

        tracker.report(f"Phase 7 Restarts: {time.monotonic()-t0:.1f}s")

    # Phase 8: Final intensive CD
    if time.monotonic() < deadline - 5:
        print(f"\n{'='*50} Phase 8: Final CD Polish {'='*50}", flush=True)
        t0 = time.monotonic()
        a = run_coordinate_descent(tracker.best, deadline - 3, tracker, max_passes=100)
        tracker.report(f"Phase 8 Final: {time.monotonic()-t0:.1f}s")

    # Fallback
    if tracker.best is None:
        tracker.best = np.ones(256)
        tracker.best_score = compute_J(tracker.best)

    print(f"\n{'='*50} FINAL RESULT {'='*50}")
    print(f"J = {tracker.best_score:.10f}, n = {tracker.best.size}", flush=True)
    print(f"Total time: {time.monotonic()-tracker.t_start:.1f}s", flush=True)

    return [float(x) for x in tracker.best]
