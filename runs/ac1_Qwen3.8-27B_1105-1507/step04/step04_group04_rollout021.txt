"""
Step function optimization using SDP relaxation for initialization,
combined with alternating LP polishing and adaptive local search.
Key innovation: SDP provides a globally-informed starting point,
avoiding the local minima that plague purely local methods.
"""

import numpy as np
from scipy.optimize import linprog
import time
import cvxpy as cp

def compute_conv(a):
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    conv = np.fft.irfft(fa * fa, size)
    return conv[:2 * n - 1]

def compute_obj(a):
    n = len(a)
    if n == 0:
        return np.inf
    conv = compute_conv(a)
    max_b = float(np.max(conv))
    sum_a = float(np.sum(a))
    if sum_a < 0.01:
        return np.inf
    return 2.0 * n * max_b / (sum_a * sum_a)

def build_toeplitz(a):
    n = len(a)
    A = np.zeros((2 * n - 1, n), dtype=np.float64)
    for j in range(n):
        k_start = j
        k_end = min(j + n, 2 * n - 1)
        A[k_start:k_end, j] = a[:k_end - k_start]
    return A

def solve_sdp(n, time_limit=120):
    """Solve SDP relaxation: min M s.t. max(conv(a,a)) <= M, sum(a)=1, a>=0."""
    X = cp.Variable((n, n), symmetric=True)
    M = cp.Variable(nonneg=True)
    
    constraints = [X >> 0]
    
    # Convolution constraints: for each lag k, sum of X[i,k-i] <= M
    for k in range(2 * n - 1):
        i_start = max(0, k - n + 1)
        i_end = min(n - 1, k)
        total = 0
        for i in range(i_start, i_end + 1):
            j = k - i
            total += X[i, j]
        constraints.append(total <= M)
    
    # Normalization: sum of all entries of X = 1 (equals (sum a)^2 = 1)
    constraints.append(cp.sum(X) == 1)
    
    prob = cp.Problem(cp.Minimize(M), constraints)
    
    for solver in [cp.MOSEK, cp.CLARABEL, cp.SCS]:
        try:
            prob.solve(solver=solver, time_limit=time_limit)
            if prob.status in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
                if X.value is not None:
                    X_val = X.value.copy()
                    X_val = (X_val + X_val.T) / 2.0
                    # Ensure PSD (clip negative eigenvalues)
                    eigvals, eigvecs = np.linalg.eigh(X_val)
                    eigvals = np.maximum(eigvals, 0)
                    X_psd = eigvecs @ np.diag(eigvals) @ eigvecs.T
                    X_psd = (X_psd + X_psd.T) / 2.0
                    # Extract a from top eigenvector
                    idx = np.argmax(eigvals)
                    a = eigvecs[:, idx] * np.sqrt(max(eigvals[idx], 0))
                    if np.sum(a) < 0:
                        a = -a
                    a = np.maximum(a, 0)
                    s = np.sum(a)
                    if s > 1e-10:
                        a = a / s
                    obj = compute_obj(a)
                    return a, obj, float(M.value)
        except Exception:
            continue
    
    return None, np.inf, None

def solve_lp_maxsum(a, M, time_limit=5):
    n = len(a)
    A = build_toeplitz(a)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': time_limit})
        if res.status == 0:
            g = np.maximum(res.x, 0)
            return g, float(np.sum(g))
    except Exception:
        pass
    return None, None

    s = np.sum(a)

        g_norm = g / sum_g

        best_t = 0.0

def resample_sequence(a, new_n):
    n = len(a)
    if n == new_n:
        return a.copy()

    a_new = np.maximum(a_new, 0)
    s = np.sum(a_new)
    if s > 1e-10:

    return a_new

def compute_conv(a):
    """Compute conv(a,a) using FFT. Returns array of length 2n-1."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    conv = np.fft.irfft(fa * fa, size)
    return conv[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    conv = compute_conv(a)
    max_b = float(np.max(conv))
    sum_a = float(np.sum(a))
    if sum_a < 0.01:
        return np.inf
    return 2.0 * n * max_b / (sum_a * sum_a)


def build_toeplitz(a):
    """Build Toeplitz matrix A where (A@g)[k] = (a*g)[k] for k=0..2n-2."""
    n = len(a)
    A = np.zeros((2 * n - 1, n), dtype=np.float64)
    for j in range(n):
        k_start = j
        k_end = min(j + n, 2 * n - 1)
        A[k_start:k_end, j] = a[:k_end - k_start]
    return A


def solve_sdp(n, time_limit=90):
    """Solve SDP relaxation: min M s.t. (a*a)[k]<=M, sum(a)=1, a>=0."""
    if not HAS_CVXPY:
        return None, np.inf, None

    X = cp.Variable((n, n), symmetric=True)
    M = cp.Variable(nonneg=True)

    constraints = [X >> 0]

    for k in range(2 * n - 1):
        i_start = max(0, k - n + 1)
        i_end = min(n - 1, k)
        total = None
        for i in range(i_start, i_end + 1):
            j = k - i
            if total is None:
                total = X[i, j]
            else:
                total = total + X[i, j]
        constraints.append(total <= M)

    constraints.append(cp.sum(X) == 1)

    prob = cp.Problem(cp.Minimize(M), constraints)

    for solver in [cp.MOSEK, cp.CLARABEL, cp.SCS, cp.PDLP]:
        try:
            prob.solve(solver=solver, time_limit=time_limit)
            if prob.status in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
                if X.value is not None:
                    X_val = X.value.copy()
                    X_val = (X_val + X_val.T) / 2.0
                    eigvals, eigvecs = np.linalg.eigh(X_val)
                    eigvals_clipped = np.maximum(eigvals, 0)
                    idx = np.argmax(eigvals_clipped)
                    a = eigvecs[:, idx] * np.sqrt(max(eigvals_clipped[idx], 0))
                    if np.sum(a) < 0:
                        a = -a
                    a = np.maximum(a, 0)
                    s = np.sum(a)
                    if s > 1e-10:
                        a = a / s
                    obj = compute_obj(a)
                    return a, obj, float(M.value)
        except Exception:
            continue
    return None, np.inf, None


def solve_lp_maxsum(a, M, time_limit=5):
    """Solve LP: max sum(g) s.t. (a*g)[k] <= M for all k, g >= 0."""
    n = len(a)
    A = build_toeplitz(a)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': time_limit})
        if res.status == 0:
            g = np.maximum(res.x, 0)
            return g, float(np.sum(g))
    except Exception:
        pass
    return None, None


def solve_lp_minpeak(a, time_limit=5):
    """Solve LP: min M s.t. (a*g)[k] <= M, sum(g)=1, g>=0."""
    n = len(a)
    A = build_toeplitz(a)
    c = np.zeros(n + 1)
    c[-1] = 1.0
    A_ub = np.zeros((2 * n - 1, n + 1))
    A_ub[:, :n] = A
    A_ub[:, n] = -1.0
    b_ub = np.zeros(2 * n - 1)
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])
    bounds = [(0.0, None)] * n + [(0.0, None)]
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': time_limit})
        if res.status == 0:
            g = np.maximum(res.x[:n], 0)
            s = np.sum(g)
            if s < 1e-10:
                return None, None
            g = g / s
            return g, float(res.x[n])
    except Exception:
        pass
    return None, None


def alternating_lp_advanced(a, max_iters=120, time_budget=40.0):
    """Enhanced alternating LP with adaptive steps and restart."""
    n = len(a)
    a = a.copy().astype(np.float64)
    s = np.sum(a)
    if s < 1e-10:
        return a
    a = a / s
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    stall = 0
    consecutive_fail = 0

    for it in range(max_iters):
        if time.monotonic() - start > time_budget - 3:
            break

        conv = compute_conv(a)
        M = float(np.max(conv))

        # Alternate between max-sum and min-peak LP
        if it % 3 == 2:
            g, _ = solve_lp_minpeak(a, time_limit=4)
        else:
            g, sum_g = solve_lp_maxsum(a, M, time_limit=4)
            if g is not None and sum_g > 1e-10:
                g = g / sum_g

        if g is None:
            consecutive_fail += 1
            if consecutive_fail > 5:
                break
            continue
        consecutive_fail = 0

        # Adaptive line search
        best_t = 0.0
        best_obj_ls = best_obj
        for t in [0.05, 0.1, 0.2, 0.3, 0.5, 0.7, 1.0]:
            a_new = (1.0 - t) * a + t * g
            a_new = np.maximum(a_new, 0)
            s_new = np.sum(a_new)
            if s_new < 1e-10:
                continue
            a_new = a_new / s_new
            obj_new = compute_obj(a_new)
            if obj_new < best_obj_ls:
                best_obj_ls = obj_new
                best_t = t

        if best_t > 0 and best_obj_ls < best_obj - 1e-13:
            a_new = (1.0 - best_t) * a + best_t * g
            a_new = np.maximum(a_new, 0)
            s_new = np.sum(a_new)
            if s_new > 1e-10:
                a_new = a_new / s_new
            best_obj = best_obj_ls
            best_a = a_new.copy()
            a = a_new
            stall = 0
        else:
            stall += 1
            if stall > 20:
                # Perturb and restart
                rng = np.random.default_rng(it * 7919 + 42)
                a = a * np.exp(rng.normal(0, 0.03, n))
                a = np.maximum(a, 1e-10)
                s = np.sum(a)
                a = a / s
                stall = 0
                if stall > 60:
                    break
    return best_a


def subgradient_optimize(a, max_iters=5000, time_budget=20.0, tau=20.0, step=0.05):
    """Multiplicative subgradient with adaptive parameters."""
    n = len(a)
    a = a.copy().astype(np.float64)
    s = np.sum(a)
    if s < 1e-10:
        return a
    a = a / s
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    stall = 0
    patience = 100

    for it in range(max_iters):
        if time.monotonic() - start > time_budget:
            break

        conv = compute_conv(a)
        max_conv = np.max(conv)
        shifted = tau * (conv - max_conv)
        shifted = np.clip(shifted, -500, 500)
        weights = np.exp(shifted)
        w_sum = np.sum(weights)
        if w_sum < 1e-300:
            idx = int(np.argmax(conv))
            weights = np.zeros(len(conv))
            weights[idx] = 1.0
        else:
            weights /= w_sum

        c = np.correlate(weights, a, mode='full')
        grad = 2.0 * c[n - 1:n - 1 + n]
        grad_centered = grad - np.mean(grad)

        a_new = a * np.exp(-step * grad_centered)
        a_new = np.maximum(a_new, 1e-15)
        s_new = np.sum(a_new)
        if s_new > 1e-10:
            a_new = a_new / s_new
        a = a_new

        obj = compute_obj(a)
        if obj < best_obj - 1e-14:
            best_obj = obj
            best_a = a.copy()
            stall = 0
        else:
            stall += 1
            if stall > patience:
                step = max(step * 0.72, 0.001)
                tau = min(tau * 1.25, 300.0)
                if step < 0.002:
                    break
                rng = np.random.default_rng(it * 31 + 7)
                a = a * np.exp(rng.normal(0, 0.04, n))
                a = np.maximum(a, 1e-10)
                s = np.sum(a)
                a = a / s
                stall = 0
                patience = max(patience // 2, 40)

        if it > 500 and it % 200 == 0:
            step *= 0.94
            tau *= 1.06

    return best_a


def fine_tune(a, time_budget=15.0):
    """High-tau fine tuning with very small steps."""
    n = len(a)
    a = a.copy().astype(np.float64)
    s = np.sum(a)
    if s < 1e-10:
        return a
    a = a / s
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    step = 0.004
    tau = 100.0
    stall = 0

    max_iters = int(time_budget * 80)
    for it in range(max_iters):
        if time.monotonic() - start > time_budget:
            break
        conv = compute_conv(a)
        max_conv = np.max(conv)
        shifted = tau * (conv - max_conv)
        shifted = np.clip(shifted, -500, 500)
        weights = np.exp(shifted)
        w_sum = np.sum(weights)
        if w_sum < 1e-300:
            break
        weights /= w_sum
        c = np.correlate(weights, a, mode='full')
        grad = 2.0 * c[n - 1:n - 1 + n]
        grad_centered = grad - np.mean(grad)
        a_new = a * np.exp(-step * grad_centered)
        a_new = np.maximum(a_new, 1e-15)
        s_new = np.sum(a_new)
        if s_new > 1e-10:
            a_new = a_new / s_new
        a = a_new
        obj = compute_obj(a)
        if obj < best_obj - 1e-15:
            best_obj = obj
            best_a = a.copy()
            stall = 0
        else:
            stall += 1
            if stall > 300:
                step *= 0.7
                tau *= 1.15
                if step < 0.0005:
                    break
                stall = 0
    return best_a


def resample_sequence(a, new_n):
    """Resample a sequence to a new length using interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    old_k = np.arange(n, dtype=np.float64)
    new_k = np.linspace(0, n - 1, new_n)
    a_new = np.interp(new_k, old_k, a)
    a_new = np.maximum(a_new, 0)
    s = np.sum(a_new)
    if s > 1e-10:
        a_new = a_new / s
    return a_new


def random_init(n, seed):
    """Generate diverse random initializations."""
    rng = np.random.default_rng(seed)
    k = np.arange(n, dtype=np.float64)
    choice = rng.integers(0, 8)

    if choice == 0:
        f = rng.uniform(0.5, 4.0)
        phase = rng.uniform(0, 2 * np.pi)
        a = 1.0 + 0.7 * np.cos(2 * np.pi * f * k / n + phase)
    elif choice == 1:
        f1 = rng.uniform(0.3, 2.0)
        f2 = rng.uniform(2.0, 6.0)
        a = 1.0 + 0.3 * np.cos(2 * np.pi * f1 * k / n) + 0.3 * np.cos(2 * np.pi * f2 * k / n)
    elif choice == 2:
        a = rng.random(n)
        win = max(3, n // 15)
        kernel = np.ones(win) / win
        a = np.convolve(a, kernel, mode='same')
    elif choice == 3:
        a = np.ones(n)
        for _ in range(rng.integers(1, 4)):
            f = rng.uniform(0.5, 5.0)
            amp = rng.uniform(0.1, 0.4)
            a += amp * np.cos(2 * np.pi * f * k / n + rng.uniform(0, 2 * np.pi))
    elif choice == 4:
        center = n * rng.uniform(0.3, 0.7)
        width = n * rng.uniform(0.15, 0.4)
        a = 1.0 + 0.5 * np.exp(-0.5 * ((k - center) / width) ** 2)
    elif choice == 5:
        a = rng.choice([0.3, 1.2, 2.0], size=n)
    elif choice == 6:
        a = rng.random(max(5, n // 8))
        a = np.interp(k, np.arange(len(a), dtype=np.float64), a)
    else:
        a = 1.0 + 0.4 * np.sin(2 * np.pi * rng.uniform(1, 4) * k / n + rng.uniform(0, 2 * np.pi))

    a = np.maximum(a, 0)
    s = np.sum(a)
    if s < 1e-10:
        a = np.ones(n)
        s = n
    return a / s


def optimize_single(a, time_budget=60.0):
    """Run full optimization pipeline on a single sequence."""
    n = len(a)
    a = a.copy().astype(np.float64)
    s = np.sum(a)
    if s < 1e-10:
        a = np.ones(n)
        s = n
    a = a / s
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    # Phase 1: Quick subgradient
    remaining = time_budget - (time.monotonic() - start)
    if remaining > 5:
        sg_time = min(10.0, remaining * 0.15)
        a_new = subgradient_optimize(best_a, time_budget=sg_time, tau=15.0, step=0.06)
        obj = compute_obj(a_new)
        if obj < best_obj:
            best_obj = obj
            best_a = a_new.copy()

    # Phase 2: Alternating LP
    remaining = time_budget - (time.monotonic() - start)
    if remaining > 10:
        lp_time = min(30.0, remaining * 0.4)
        a_new = alternating_lp_advanced(best_a, max_iters=100, time_budget=lp_time)
        obj = compute_obj(a_new)
        if obj < best_obj:
            best_obj = obj
            best_a = a_new.copy()

    # Phase 3: Subgradient polish
    remaining = time_budget - (time.monotonic() - start)
    if remaining > 8:
        sg_time = min(12.0, remaining * 0.25)
        a_new = subgradient_optimize(best_a, time_budget=sg_time, tau=40.0, step=0.025)
        obj = compute_obj(a_new)
        if obj < best_obj:
            best_obj = obj
            best_a = a_new.copy()

    # Phase 4: Fine tuning
    remaining = time_budget - (time.monotonic() - start)
    if remaining > 5:
        ft_time = min(10.0, remaining * 0.3)
        a_new = fine_tune(best_a, time_budget=ft_time)
        obj = compute_obj(a_new)
        if obj < best_obj:
            best_obj = obj
            best_a = a_new.copy()

    return best_a, best_obj


def local_search(a, time_budget=50.0):
    """Intensive local search with perturbations and re-optimizations."""
    n = len(a)
    a = a.copy().astype(np.float64)
    s = np.sum(a)
    if s < 1e-10:
        return a, compute_obj(a)
    a = a / s
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    attempt = 0

    while time.monotonic() - start < time_budget - 5:
        attempt += 1
        remaining = time_budget - (time.monotonic() - start)
        if remaining < 8:
            break
        rng = np.random.default_rng(attempt * 12345 + 6789)

        # Vary perturbation strategy
        if attempt % 3 == 1:
            scale = 0.02
            a_pert = best_a * np.exp(rng.normal(0, scale, n))
        elif attempt % 3 == 2:
            # Sparse perturbation
            a_pert = best_a.copy()
            n_sparse = max(3, n // 20)
            idx = rng.choice(n, size=n_sparse, replace=False)
            a_pert[idx] *= rng.uniform(0.5, 1.8, n_sparse)
        else:
            # Blend with a random smooth function
            a_rand = random_init(n, attempt * 31)
            t_blend = rng.uniform(0.05, 0.15)
            a_pert = (1 - t_blend) * best_a + t_blend * a_rand

        a_pert = np.maximum(a_pert, 1e-10)
        s_pert = np.sum(a_pert)
        if s_pert < 1e-10:
            continue
        a_pert = a_pert / s_pert

        sub_time = min(15.0, remaining * 0.3)
        a_opt, obj = optimize_single(a_pert, time_budget=sub_time)
        if obj < best_obj:
            best_obj = obj
            best_a = a_opt.copy()

    return best_a, best_obj


def try_parent():
    """Try to load parent sequence."""

    return None, np.inf

    enhanced alternating LP, adaptive subgradient, and multi-length exploration.
    """
    start_time = time.monotonic()
    deadline = start_time + float(budget_s) - 30.0

    best_seq = None
    best_score = np.inf

    parent, parent_obj = try_parent()

        best_score = parent_obj

    else:

    sdp_results = {}
    print("\n=== Phase 1: SDP Relaxation ===")
    if HAS_CVXPY:
        sdp_n_values = [100, 150, 200]
        for n_sdp in sdp_n_values:
            if time.monotonic() > deadline - 120:
                break
            remaining_sdp = deadline - time.monotonic()
            if remaining_sdp < 30:
                break
            alloc = min(80.0, remaining_sdp * 0.3)

            t0 = time.monotonic()
            a_sdp, obj_sdp, m_val = solve_sdp(n_sdp, time_limit=int(alloc))

            if a_sdp is not None:
                sdp_results[n_sdp] = (a_sdp, obj_sdp)
                print(f"    obj={obj_sdp:.6f}, M={m_val:.6f}, time={elapsed_sdp:.1f}s")
            else:
                print(f"    Failed, time={elapsed_sdp:.1f}s")

    print(f"\n=== Phase 2: Optimize parent (n={n_parent}) ===")
    if best_seq is not None:

        p2_budget = min(remaining * 0.25, 250.0)

    print(f"\n=== Phase 3: Multi-n exploration ===")
    target_ns = sorted(set([
        n_parent,
        n_parent - 10, n_parent + 10,
        n_parent - 20, n_parent + 20,
        n_parent - 30, n_parent + 30,
        n_parent - 50, n_parent + 50,
    ]))
    target_ns = [nv for nv in target_ns if 150 <= nv <= 600]

    remaining = deadline - time.monotonic()
    if remaining > 40:
        per_n_budget = remaining / max(len(target_ns), 1) * 1.2

                break
            remaining = deadline - time.monotonic()
            if remaining < 25:
                break
            alloc = min(per_n_budget, remaining * 0.3, 130.0)

            best_n_seq = None

                a_opt, obj = optimize_single(a_init, time_budget=t_sub)
                if obj < best_n_obj:

                if time.monotonic() > deadline - 20:
                    break
                a_init = resample_sequence(a_sdp, n_val)

                a_opt = fine_tune(best_seq, time_budget=ft_budget)

                    print(f"  After final fine-tune: {best_score:.6f}")

    # Final validation
    if best_seq is None:
        best_seq = np.ones(50)
        best_score = compute_obj(best_seq)

    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    val = evaluate_sequence(result)
    if val < best_score:
        best_score = val

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(best_seq)}, score={best_score:.6f}, time={elapsed:.1f}s ===")

    return result
