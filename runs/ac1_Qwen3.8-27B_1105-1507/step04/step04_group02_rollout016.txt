from scipy.optimize import linprog
from scipy.sparse import csr_matrix
import time

def compute_conv(a):
    """FFT-based convolution."""
    n = len(a)
    size = 1
    while size < 2*n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2*n-1]

def compute_obj(a):
    n = len(a)
    S = np.sum(a)
    if S < 1e-10:
        return np.inf
    conv = compute_conv(a)
    return 2.0 * n * max(conv) / (S * S)

def build_toeplitz(a):
    """Build the Toeplitz matrix T such that T @ g = conv(a, g)."""
    n = len(a)
    # conv(a, g)[k] = sum_j a[k-j] * g[j] for valid j
    # = sum_j T[k,j] * g[j]
    # T[k,j] = a[k-j] if 0 <= k-j < n, else 0
    rows = []
    cols = []
    vals = []
    for j in range(n):
        # For this j, T[k,j] = a[k-j] for k = j, j+1, ..., j+n-1
        k_start = j
        k_end = j + n  # exclusive
        for k in range(k_start, min(k_end, 2*n-1)):
            rows.append(k)
            cols.append(j)
            vals.append(a[k-j])
    return csr_matrix((vals, (rows, cols)), shape=(2*n-1, n))

def solve_lp(a, M, time_limit=10):
    """Solve: max sum(g) s.t. T(a)g <= M, g >= 0."""
    n = len(a)
    T = build_toeplitz(a)
    c = -np.ones(n)  # minimize -sum(g) = maximize sum(g)
    res = linprog(c, A_ub=T, b_ub=np.full(2*n-1, M), 
                  bounds=[(0, None)]*n, method='highs',
                  options={'time_limit': time_limit, 'presolve': True})
    if res.status == 0:
        return np.maximum(res.x, 0)
    return None

def lp_optimize(a, time_budget, eps_list=None):
    """LP-based alternating optimization."""
    if eps_list is None:
        eps_list = [0.0, 0.005, 0.01, 0.02, 0.05]

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    
    it = 0
    stall = 0
    
    while time.monotonic() - start < time_budget - 2:
        remaining = time_budget - (time.monotonic() - start)
        if remaining < 5:
            break
        
        conv = compute_conv(a)
        M = max(conv)
        
        # Try different epsilon values
        eps = eps_list[it % len(eps_list)]
        M_relaxed = M * (1.0 + eps)
        
        g = solve_lp(a, M_relaxed, time_limit=min(10, remaining * 0.3))
        if g is None or np.sum(g) < 1e-10:
            it += 1
            continue
        
        g = g / np.sum(g)
        
        # Line search
        best_t = 0
        best_obj_ls = best_obj
        for t in [0.1, 0.2, 0.3, 0.5, 0.7, 0.9, 1.0]:
            a_try = (1-t)*a + t*g
            a_try = np.maximum(a_try, 0)
            S = np.sum(a_try)
            if S < 1e-10:

            a_try = a_try / S
            obj_try = compute_obj(a_try)
            if obj_try < best_obj_ls:

                a = a / np.sum(a)
                stall = 0
        
        it += 1
    
    return best_a, best_obj

"""
LP-driven step-function optimization minimizing 2*n*max(conv(a,a))/(sum(a))^2.
Core method: alternating LP (maximize sum(g) s.t. conv(a,g) <= M, g>=0) with adaptive
relaxation, cubic line search, dual-weighted subgradient polish, multi-n exploration,
and perturbation-based escapes. Focused on making the LP the primary engine with
subgradient as a secondary polishing tool.
"""

import numpy as np
from scipy.optimize import linprog
from scipy.sparse import coo_matrix, csr_matrix
from scipy.interpolate import interp1d
import time


def compute_conv(a):
    """Compute linear convolution a*a via FFT."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    conv = np.fft.irfft(fa * fa, size)
    return conv[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    S = float(np.sum(a))
    if S < 1e-10:
        return np.inf
    conv = compute_conv(a)
    M = float(np.max(conv))
    return 2.0 * n * M / (S * S)


def build_toeplitz_sparse(a):
    """Build sparse matrix T such that T @ g = conv(a, g)."""
    n = len(a)
    rows = np.empty(n * n, dtype=np.int32)
    cols = np.empty(n * n, dtype=np.int32)
    vals = np.empty(n * n, dtype=np.float64)
    idx = 0
    for j in range(n):
        k_max = min(j + n, 2 * n - 1)
        for k in range(j, k_max):
            rows[idx] = k
            cols[idx] = j
            vals[idx] = a[k - j]
            idx += 1
    rows = rows[:idx]
    cols = cols[:idx]
    vals = vals[:idx]
    return csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))


def solve_lp(a, M, time_limit=10.0):
    """Solve LP: max sum(g) s.t. T(a)g <= M*1, g >= 0. Returns g or None."""
    n = len(a)
    T = build_toeplitz_sparse(a)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c, A_ub=T, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': time_limit})
        if res.status == 0:
            g = np.maximum(res.x, 0.0)
            if np.sum(g) > 1e-12:
                return g
    except Exception:
        pass
    return None


def cubic_line_search(a, g, best_obj, n_points=12):
    """Cubic line search along direction (1-t)*a + t*g. Returns best t and obj."""
    n = len(a)
    a_norm = a / np.sum(a)
    g_norm = g / np.sum(g)
    
    best_t = 0.0
    best_val = best_obj
    ts = np.linspace(0.0, 1.0, n_points)
    objs = np.full(n_points, np.inf)
    
    for i, t in enumerate(ts):
        a_try = (1.0 - t) * a_norm + t * g_norm
        a_try = np.maximum(a_try, 0.0)
        S = np.sum(a_try)
        if S < 1e-10:
            continue
        a_try = a_try / S
        objs[i] = compute_obj(a_try)
        if objs[i] < best_val:
            best_val = objs[i]
            best_t = t
    
    # Refine around best
    if 0 < best_t < 1:
        lo = max(0.0, best_t - 0.5 / n_points)
        hi = min(1.0, best_t + 0.5 / n_points)
        ts2 = np.linspace(lo, hi, 8)
        for t in ts2:
            a_try = (1.0 - t) * a_norm + t * g_norm
            a_try = np.maximum(a_try, 0.0)
            S = np.sum(a_try)
            if S < 1e-10:
                continue
            a_try = a_try / S
            obj_try = compute_obj(a_try)
            if obj_try < best_val:
                best_val = obj_try
                best_t = t
    
    return best_t, best_val


def subgradient_step(a, alpha, topk=20):
    """One subgradient step with centering. Returns updated a."""
    n = len(a)
    conv = compute_conv(a)
    M = float(np.max(conv))
    
    # Find top-k active positions
    if topk >= len(conv):
        active = np.arange(len(conv))
    else:
        threshold = np.sort(conv)[-topk]
        active = np.where(conv >= threshold * 0.999)[0]
    
    # Compute gradient
    grad = np.zeros(n)
    for k in active:
        lo = max(0, k - n + 1)
        hi = min(n, k + 1)
        for i in range(lo, hi):
            j = k - i
            if 0 <= j < n:
                grad[i] += 2.0 * a[j]
    
    grad /= max(len(active), 1)
    
    # Center to preserve sum
    grad_centered = grad - np.mean(grad)
    
    a_new = a - alpha * grad_centered
    a_new = np.maximum(a_new, 0.0)
    S = np.sum(a_new)
    if S < 1e-10:
        return a
    return a_new / S


def dual_weighted_subgradient(a, y_weights, topk=20):
    """Subgradient weighted by LP dual variables."""
    n = len(a)
    conv = compute_conv(a)
    M = float(np.max(conv))
    
    if topk >= len(conv):
        active = np.arange(len(conv))
    else:
        threshold = np.sort(conv)[-topk]
        active = np.where(conv >= threshold * 0.999)[0]
    
    grad = np.zeros(n)
    total_w = 0.0
    for k in active:
        w = y_weights[k] if k < len(y_weights) else 1.0
        if w < 0:
            w = 1.0
        lo = max(0, k - n + 1)
        hi = min(n, k + 1)
        for i in range(lo, hi):
            j = k - i
            if 0 <= j < n:
                grad[i] += 2.0 * a[j] * w
        total_w += w
    
    if total_w > 0:
        grad /= total_w
    else:
        grad /= max(len(active), 1)
    
    grad_centered = grad - np.mean(grad)
    return grad_centered

    return best_a, best_obj


def subgradient_polish(a, time_budget, verbose=False):
    """Polish with momentum subgradient descent."""
    n = len(a)
    a = a.copy().astype(np.float64)
    S = np.sum(a)
    if S < 1e-10:
        return a, compute_obj(a)
    a = a / S
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    
    v = np.zeros(n)
    momentum = 0.8
    base_alpha = 0.15
    stall = 0
    
    it = 0
    while time.monotonic() - start < time_budget - 2.0:
        alpha = base_alpha / (1.0 + 0.005 * it)
        a_new = subgradient_step(a, alpha, topk=min(30, n // 3))
        
        # Momentum
        v = momentum * v + (1.0 - momentum) * (a_new - a)
        a_mom = a - 0.5 * v
        a_mom = np.maximum(a_mom, 0.0)
        S_mom = np.sum(a_mom)
        if S_mom > 1e-10:
            a_mom = a_mom / S_mom
            obj_mom = compute_obj(a_mom)
        else:
            obj_mom = np.inf
        
        obj_new = compute_obj(a_new)
        
        if obj_mom < obj_new and obj_mom < best_obj:
            a = a_mom
            best_obj = obj_mom
            best_a = a.copy()
        elif obj_new < best_obj:
            a = a_new
            best_obj = obj_new
            best_a = a.copy()
        else:
            stall += 1
            if stall > 50:
                rng = np.random.default_rng(it * 31 + 7)
                a = a * np.exp(rng.normal(0, 0.04, n))
                a = np.maximum(a, 0.0)
                a = a / np.sum(a)
                v = np.zeros(n)
                stall = 0
                base_alpha *= 0.8
        
        it += 1
    
    return best_a, best_obj


def cosine_init(n, n_modes=10, seed=0):
    """Cosine series initialization."""
    rng = np.random.default_rng(seed)
    k = np.arange(n)
    a = np.ones(n)
    for m in range(1, n_modes + 1):
        c = rng.uniform(-0.5, 0.5) / m
        phase = rng.uniform(0, 2 * np.pi)
        a += c * np.cos(2 * np.pi * m * k / n + phase)
    a = np.maximum(a, 0.0)
    S = np.sum(a)
    if S < 1e-10:
        return np.ones(n) / n
    return a / S


def raised_cosine_init(n, alpha=1.5, seed=0):
    """Raised cosine with random perturbation."""
    k = np.arange(n)
    x = 2.0 * k / (n - 1) - 1.0
    a = 0.5 * (1.0 + np.cos(np.pi * x)) ** alpha
    rng = np.random.default_rng(seed)
    a += rng.uniform(-0.1, 0.1, n)
    a = np.maximum(a, 0.0)
    S = np.sum(a)
    if S < 1e-10:
        return np.ones(n) / n
    return a / S


def gaussian_init(n, sigma_frac=0.3, seed=0):
    """Gaussian bump with random perturbation."""
    k = np.arange(n)
    center = n / 2.0
    sigma = sigma_frac * n
    a = np.exp(-0.5 * ((k - center) / sigma) ** 2)
    rng = np.random.default_rng(seed)
    a += 0.05 * rng.uniform(0, 1, n)
    S = np.sum(a)
    if S < 1e-10:
        return np.ones(n) / n
    return a / S


def periodic_init(n, freq=3, seed=0):
    """Periodic non-negative pattern."""
    k = np.arange(n)
    rng = np.random.default_rng(seed)
    a = 1.0 + 0.4 * np.cos(2 * np.pi * freq * k / n + rng.uniform(0, 2 * np.pi))
    a += 0.2 * np.cos(2 * np.pi * (freq + 1) * k / n + rng.uniform(0, 2 * np.pi))
    a = np.maximum(a, 0.0)
    S = np.sum(a)
    return a / S


def try_load_parent():
    """Try to load parent sequence."""
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64)
        parent = np.clip(parent, 0, 1000.0)
        if parent.size > 0 and np.isfinite(parent).all():
            obj = compute_obj(parent)
            print(f"Loaded parent: n={parent.size}, obj={obj:.6f}")
            return parent.copy(), obj
    except Exception:
        pass
    return None, np.inf


def interpolate_sequence(a, n_new):
    """Interpolate sequence to new length."""
    n_old = len(a)
    if n_old == n_new:
        return a.copy()
    x_old = np.linspace(0, 1, n_old)
    x_new = np.linspace(0, 1, n_new)
    try:
        f = interp1d(x_old, a, kind='cubic', fill_value='extrapolate')
        a_new = np.maximum(f(x_new), 0.0)
    except Exception:
        try:
            f = interp1d(x_old, a, kind='linear', fill_value='extrapolate')
            a_new = np.maximum(f(x_new), 0.0)
        except Exception:
            a_new = np.ones(n_new) / n_new
    S = np.sum(a_new)
    if S < 1e-10:
        return np.ones(n_new) / n_new
    return a_new / S


def propose_candidate(seed=42, budget_s=1500):
    """
    Main search function. Uses LP-driven optimization as the primary engine,
    with subgradient polishing, multi-n exploration, and perturbation escapes.
    """
    start_time = time.monotonic()
    deadline = start_time + float(budget_s) - 20.0
    
    best_seq = None
    best_score = np.inf
    
    # Load parent
    parent, parent_obj = try_load_parent()
    if parent is not None:
        best_score = parent_obj
        best_seq = parent.copy()
        n_parent = len(parent)
    else:
        n_parent = 500
    
    print(f"Initial best: {best_score:.6f} (n={n_parent if best_seq is not None else 'N/A'})")
    
    # === Phase 1: Quick exploration (~250s) ===
    print("\n=== Phase 1: Quick exploration ===")
    t_phase1_end = start_time + 250.0
    
    n_candidates = sorted(set([
        max(200, n_parent - 100), n_parent, min(1200, n_parent + 100),
        300, 400, 500, 600, 700, 800, 1000
    ]))
    n_candidates = [n for n in n_candidates if 200 <= n <= 1200]
    
    candidate_pool = []
    
    for n in n_candidates:
        if time.monotonic() > t_phase1_end:
            break
        remaining = t_phase1_end - time.monotonic()
        if remaining < 15:
            break
        
        # Multiple inits per n
        inits = [
            cosine_init(n, n_modes=8, seed=seed + n),
            raised_cosine_init(n, alpha=1.5, seed=seed + n + 1),
            gaussian_init(n, sigma_frac=0.3, seed=seed + n + 2),
            periodic_init(n, freq=3, seed=seed + n + 3),
        ]
        
        for idx, a_init in enumerate(inits):
            if time.monotonic() > t_phase1_end:
                break
            remaining = t_phase1_end - time.monotonic()
            if remaining < 10:
                break
            
            obj_init = compute_obj(a_init)
            alloc = min(remaining * 0.2, 25.0)
            
            # Quick LP optimization
            a_quick, obj_quick = lp_optimize(a_init, time_budget=alloc, verbose=False)
            
            candidate_pool.append((a_quick, obj_quick))
            if obj_quick < best_score:
                best_score = obj_quick
                best_seq = a_quick.copy()
                print(f"  n={n}, init{idx}: {best_score:.6f} ** NEW BEST **")
    
    # Sort candidates
    candidate_pool.sort(key=lambda x: x[1])
    print(f"  Phase 1 done. Best: {best_score:.6f}, pool size: {len(candidate_pool)}")
    
    # === Phase 2: Focused LP refinement (~600s) ===
    print(f"\n=== Phase 2: Focused LP refinement ===")
    t_phase2_end = min(start_time + 850.0, deadline)
    
    # Refine top candidates
    n_refine = min(4, len(candidate_pool))
    for idx in range(n_refine):
        if time.monotonic() > t_phase2_end - 60:
            break
        remaining = t_phase2_end - time.monotonic()
        if remaining < 30:
            break
        
        a_cand, obj_cand = candidate_pool[idx]
        if obj_cand > best_score + 0.02:
            break
        
        n_cand = len(a_cand)
        alloc = min(remaining * 0.3, 120.0)
        print(f"  Refining candidate {idx} (n={n_cand}, obj={obj_cand:.6f}), alloc={alloc:.0f}s")
        
        a_ref, obj_ref = lp_optimize(a_cand, time_budget=alloc, verbose=True)
        
        if obj_ref < best_score:
            best_score = obj_ref
            best_seq = a_ref.copy()
            print(f"    ** NEW BEST: n={n_cand}, obj={best_score:.6f} **")
    
    # === Phase 3: Intensive iteration on best (~300s) ===
    print(f"\n=== Phase 3: Intensive iteration ===")
    t_phase3_end = min(start_time + 1150.0, deadline)
    
    a_cur = best_seq.copy()
    n_best = len(a_cur)
    round_num = 0
    
    while time.monotonic() < t_phase3_end:
        remaining = t_phase3_end - time.monotonic()
        if remaining < 30:
            break
        
        round_num += 1
        alloc = min(remaining * 0.2, 60.0)
        
        # LP optimization
        a_new, obj_new = lp_optimize(a_cur, time_budget=alloc, verbose=False)
        
        if obj_new < best_score:
            best_score = obj_new
            best_seq = a_new.copy()
            a_cur = a_new.copy()
            print(f"  R{round_num} LP: {best_score:.6f} ** NEW BEST **")
        else:
            # Subgradient polish
            sg_alloc = min(remaining * 0.1, 20.0)
            a_sg, obj_sg = subgradient_polish(a_cur, time_budget=sg_alloc)
            if obj_sg < best_score:
                best_score = obj_sg
                best_seq = a_sg.copy()
                a_cur = a_sg.copy()
                print(f"  R{round_num} SG: {best_score:.6f} ** NEW BEST **")
            else:
                print(f"  R{round_num}: LP={obj_new:.6f}, SG={obj_sg:.6f} (best={best_score:.6f})")
                
                # Perturbation
                rng = np.random.default_rng(round_num * 104729 + 999)
                for scale in [0.05, 0.1, 0.15]:
                    if time.monotonic() > t_phase3_end - 15:
                        break
                    a_pert = a_cur * np.exp(rng.normal(0, scale, n_best))
                    a_pert = np.maximum(a_pert, 0.0)
                    a_pert = a_pert / np.sum(a_pert)
                    
                    # Quick refine
                    pert_alloc = min(remaining * 0.08, 15.0)
                    a_pr, obj_pr = lp_optimize(a_pert, time_budget=pert_alloc, verbose=False)
                    if obj_pr < best_score:
                        best_score = obj_pr
                        best_seq = a_pr.copy()
                        a_cur = a_pr.copy()
                        print(f"  R{round_num} pert({scale}): {best_score:.6f} ** NEW BEST **")
                        break
    
    # === Phase 4: Multi-n exploration (~150s) ===
    print(f"\n=== Phase 4: Multi-n exploration ===")
    t_phase4_end = min(start_time + 1300.0, deadline)
    
    n_base = len(best_seq)
    n_tries = sorted(set([
        max(200, n_base - 100), max(200, n_base - 50),
        n_base + 50, n_base + 100, n_base + 150, n_base + 200,
        max(200, n_base + 300)
    ]))
    n_tries = [n for n in n_tries if 200 <= n <= 1500]
    # Remove duplicates and the base n
    n_tries = [n for n in n_tries if n != n_base]
    
    for n_try in n_tries:
        if time.monotonic() > t_phase4_end:
            break
        remaining = t_phase4_end - time.monotonic()
        if remaining < 20:
            break
        
        a_interp = interpolate_sequence(best_seq, n_try)
        alloc = min(remaining * 0.25, 40.0)
        
        a_new, obj_new = lp_optimize(a_interp, time_budget=alloc, verbose=False)
        if obj_new < best_score:
            best_score = obj_new
            best_seq = a_new.copy()
            print(f"  n={n_try}: {best_score:.6f} ** NEW BEST **")
        else:
            print(f"  n={n_try}: {obj_new:.6f}")
    
    # === Phase 5: Final polish (~120s) ===
    print(f"\n=== Phase 5: Final polish ===")
    
    a_cur = best_seq.copy()
    n_final = len(a_cur)
    
    while time.monotonic() < deadline - 10:
        remaining = deadline - time.monotonic()
        if remaining < 20:
            break
        
        alloc = min(remaining * 0.4, 50.0)
        a_new, obj_new = lp_optimize(a_cur, time_budget=alloc, verbose=True)
        
        if obj_new < best_score:
            best_score = obj_new
            best_seq = a_new.copy()
            a_cur = a_new.copy()
        else:
            # Try subgradient
            sg_alloc = min(remaining * 0.2, 20.0)
            a_sg, obj_sg = subgradient_polish(a_cur, time_budget=sg_alloc)
            if obj_sg < best_score:
                best_score = obj_sg
                best_seq = a_sg.copy()
                a_cur = a_sg.copy()
            else:
                # Small perturbation and retry
                rng = np.random.default_rng(int(time.monotonic() * 1000))
                a_pert = a_cur * np.exp(rng.normal(0, 0.03, n_final))
                a_pert = np.maximum(a_pert, 0.0)
                a_pert = a_pert / np.sum(a_pert)
                a_cur = a_pert
    
    # Final validation
    if best_seq is None:
        best_seq = np.ones(300)
        best_score = compute_obj(best_seq)
    
    # Normalize to sum=1 for safety (objective is scale-invariant)
    S = np.sum(best_seq)
    if S > 1e-10:
        best_seq = best_seq / S
        # Scale back to reasonable range
        best_seq = best_seq * 100.0  # sum = 100, values in [0, 100]
    
    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    val = evaluate_sequence(result)
    if val < np.inf:
        if val < best_score:
            best_score = val
        print(f"\n=== FINAL: n={len(result)}, score={val:.6f} ===")
    else:
        print("WARNING: evaluate_sequence returned inf, using fallback")
        result = [1.0] * 300
        best_score = 2.0
    
    elapsed = time.monotonic() - start_time
    print(f"Total time: {elapsed:.1f}s")
    
    return result
