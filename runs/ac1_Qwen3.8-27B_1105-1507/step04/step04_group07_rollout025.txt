            je = min(n - 1, k)
            rows.extend([k] * (je - js + 1))
            cols.extend(range(js, je + 1))
        _spar_cache[n] = (np.array(rows), np.array(cols), len(rows))
    return _spar_cache[n]

def fast_conv(a):
    n = len(a)
    s = 1
    while s < 2 * n - 1:
        s <<= 1
    fa = np.fft.rfft(a, s)
    return np.fft.irfft(fa * fa, s)[:2 * n - 1]

def fast_cross_conv(a, g):
    """Compute cross-convolution a * g (not self-conv)."""
    n = len(a)
    ng = len(g)
    s = 1
    while s < n + ng - 1:
        s <<= 1
    fa = np.fft.rfft(a, s)
    fg = np.fft.rfft(g, s)
    return np.fft.irfft(fa * fg, s)[:n + ng - 1]

def compute_J(a):
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))

def mw_best_response(a, M, n_iters=200, eta=0.05):
    """
    Multiplicative Weights best response.
    Finds g >= 0 maximizing sum(g) subject to (a*g)_k <= M for all k.
    Returns approximate solution normalized to sum=1.
    """
    n = len(a)
    g = np.ones(n)
    
    for t in range(n_iters):
        b = fast_cross_conv(a, g)
        # Only consider the first 2n-1 elements (valid cross-conv range)
        if len(b) > 2 * n - 1:
            b = b[:2 * n - 1]
        k_star = int(np.argmax(b))
        v = b[k_star]
        if v <= M * (1 + 1e-6):
            break
        
        # Update: reduce g[j] for j contributing to constraint k_star
        js = max(0, k_star - n + 1)
        je = min(n - 1, k_star)
        if je < js:
            break
        
        # g[j] *= (1 - eta * a[k_star - j] / M)
        # Vectorized:
        indices = np.arange(js, je + 1)
        factors = 1.0 - eta * a[k_star - indices] / v  # use v (current violation) for normalization
        g[indices] *= factors
        g[indices] = np.maximum(g[indices], 0.0)
    
    total = g.sum()
    if total < 1e-10:
        return None
    g = g / total
    return g

def lp_best_response(a, M, time_limit=5):
    """LP best response using scipy linprog."""
    n = len(a)
    rows, cols, nnz = get_sparsity(n)
    
    # Build the values for the sparse matrix
    vals = np.empty(nnz)
    for i in range(nnz):
        k = rows[i]
        j = cols[i]
        idx = k - j
        if 0 <= idx < n:
            vals[i] = a[idx]
        else:
            vals[i] = 0.0
    
    A = csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                options={'presolve': True, 'time_limit': time_limit})
    if r.success and r.x is not None:
        g = np.maximum(r.x, 0.0)
        if g.sum() > 1e-10:
            return g / g.sum()
    return None

def lse_gradient_descent(a, max_iters, deadline, tracker):
    """LSE gradient descent with adaptive tau and learning rate."""
    a = np.maximum(a.astype(np.float64), 1e-6)
    a = a / a.sum()
    n = len(a)
    current_J = compute_J(a)
    tracker.update(a)
    
    tau = 30.0
    lr = 0.1 / n
    momentum = 0.9
    v = np.zeros(n)
    
    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        
        b = fast_conv(a)
        b_max = b.max()
        
        # Softmax weights
        exp_b = np.exp(tau * (b - b_max))
        p = exp_b / exp_b.sum()
        
        # Gradient: cross-correlation of p and a
        s = 1
        while s < 3 * n:
            s <<= 1
        fp = np.fft.rfft(p, s)
        fa = np.fft.rfft(a, s)
        corr = np.fft.irfft(fp * np.conj(fa), s)
        grad = 2.0 * corr[:n]
        
        # Nesterov momentum
        v = momentum * v + (1.0 - momentum) * grad
        a_new = a - lr * v
        a_new = np.maximum(a_new, 1e-6)
        a_new = a_new / a_new.sum()
        
        J_new = compute_J(a_new)
        if J_new < current_J - 1e-14:
            a = a_new
            current_J = J_new
            # Increase tau slowly
            tau = min(200.0, tau * 1.02)
            lr = 0.1 / (n * max(1.0, tau / 30.0))
        else:
            lr *= 0.5
            v *= 0.3
            if lr < 1e-18:
                break
        
        if it % 20 == 0:
            tracker.update(a)
    
    tracker.update(a)
    return a

def subgradient_descent(a, max_iters, deadline, tracker):

    a = np.maximum(a.astype(np.float64), 1e-8)
    a = a / a.sum()
    n = len(a)
    current_J = compute_J(a)

        if time.monotonic() > deadline:
            break
        
        b = fast_conv(a)
        M = float(b.max())
        
        # Find active set
        eps = 1e-6 * M
        active = np.where(b >= M - eps)[0]
        
        # Subgradient of max(b_k) w.r.t. a[j]
        # d(b_k)/da[j] = a[k-j] (for self-conv, symmetric)
        grad = np.zeros(n)
        for k in active:
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            if je >= js:
                grad[js:je+1] += a[k - np.arange(js, je + 1)]
        grad /= len(active)
        
        # Direction for minimizing 2n*M (with sum(a)=1 constraint)
        # We want to decrease a where it contributes to peak, increase elsewhere
        # Effective gradient of J = 2n*max_b (with sum=1): 
        # dJ/da[j] = 2n * grad[j] - 2n*M * 2*a[j] (from denominator, but sum=1 so denom is fixed)
        # Actually with sum=1, J = 2n*max_b, so dJ/da[j] = 2n * grad[j]
        # But we need to maintain sum=1, so the projected gradient is:
        # grad_proj = grad - mean(grad) (to maintain sum)
        # But also need to account for the fact that changing a[j] affects all b_k
        
        # Simpler: just use grad as the direction to subtract (to reduce peak)
        # and add a uniform component to maintain sum
        grad_centered = grad - grad.mean()
        a_new = a - lr * grad_centered
        a_new = np.maximum(a_new, 1e-8)
        a_new = a_new / a_new.sum()
        
        J_new = compute_J(a_new)
        if J_new < current_J - 1e-15:
            a = a_new
            current_J = J_new
            lr = min(lr * 1.05, 5.0 / n)
        else:
            lr *= 0.7
        
        if it % 50 == 0:
            tracker.update(a)
    
    tracker.update(a)
    return a

    n = len(a)
    s = 1
    while s < 2 * n - 1:
        s <<= 1
    fa = np.fft.rfft(a, s)
    return np.fft.irfft(fa * fa, s)[:2 * n - 1]


def fast_cross_conv(a, g):
    n = len(a)
    s = 1
    while s < 2 * n - 1:
        s <<= 1
    fa = np.fft.rfft(a, s)
    fg = np.fft.rfft(g, s)
    return np.fft.irfft(fa * fg, s)[:2 * n - 1]


def compute_J(a):
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))


def mw_best_response_global(a, M, n_iters=300, eta=0.08):
    """
    Global Multiplicative Weights best response.
    Finds g >= 0 maximizing sum(g) subject to (a*g)_k <= M for all k.
    Uses a global update: all coordinates are updated based on weighted sum of violations.
    """
    n = len(a)
    g = np.ones(n)
    
    for t in range(n_iters):
        b = fast_cross_conv(a, g)
        if len(b) > 2 * n - 1:
            b = b[:2 * n - 1]
        
        excess = np.maximum(b - M, 0.0)
        total_excess = excess.sum()
        if total_excess < 1e-12 * M:
            break
        
        # Compute "badness" of each coordinate: cross-correlation of excess and a
        s = 1
        while s < 3 * n:
            s <<= 1
        fe = np.fft.rfft(excess, s)
        fa = np.fft.rfft(a, s)
        badness = np.fft.irfft(fe * fa, s)[:n]
        
        max_bad = badness.max()
        if max_bad < 1e-15:
            break
        
        factors = 1.0 - eta * badness / max_bad
        factors = np.maximum(factors, 0.0)
        g *= factors
        g = np.maximum(g, 1e-12)
    
    total = g.sum()
    if total < 1e-10:
        return None
    return g / total


def mw_best_response_single(a, M, n_iters=200, eta=0.06):
    """
    Single-constraint MW best response (alternates to the most violated constraint).
    """
    n = len(a)
    g = np.ones(n)
    
    for t in range(n_iters):
        b = fast_cross_conv(a, g)
        if len(b) > 2 * n - 1:
            b = b[:2 * n - 1]
        k_star = int(np.argmax(b))
        v = b[k_star]
        if v <= M * (1.0 + 1e-6):
            break
        
        js = max(0, k_star - n + 1)
        je = min(n - 1, k_star)
        if je < js:
            break
        
        indices = np.arange(js, je + 1)
        a_vals = a[k_star - indices]
        factors = 1.0 - eta * a_vals / v
        factors = np.maximum(factors, 0.0)
        g[indices] *= factors
        g[indices] = np.maximum(g[indices], 1e-12)
    
    total = g.sum()
    if total < 1e-10:
        return None
    return g / total


def lp_best_response(a, M, time_limit=8):
    """LP best response using scipy linprog with HiGHS."""
    n = len(a)
    rows, cols, nnz = get_sparsity(n)
    
    vals = np.empty(nnz)
    for i in range(nnz):
        k = rows[i]
        j = cols[i]
        idx = k - j
        if 0 <= idx < n:
            vals[i] = a[idx]
        else:
            vals[i] = 0.0
    
    A = csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                options={'presolve': True, 'time_limit': time_limit})
    if r.success and r.x is not None:
        g = np.maximum(r.x, 0.0)
        if g.sum() > 1e-10:
            return g / g.sum()
    return None


def flat_spectrum_project(a, n_iters=50, alpha=0.5):
    """Iteratively project toward flat spectrum while maintaining non-negativity."""
    a = np.maximum(a, 0.0).copy()
    if a.sum() < 1e-10:
        return a
    a = a / a.sum()
    n = len(a)
    
    for _ in range(n_iters):
        A = np.fft.rfft(a)
        target = np.mean(np.abs(A))
        phases = np.angle(A)
        A_flat = target * np.exp(1j * phases)
        a_raw = np.fft.irfft(A_flat, n)
        a_proj = np.maximum(a_raw, 0.0)
        if a_proj.sum() < 1e-10:
            break
        a_proj = a_proj / a_proj.sum()
        a = (1.0 - alpha) * a + alpha * a_proj
    
    return a


def run_mw_opt(a, max_iters, deadline, tracker, eta=0.08, tighten=0.03, use_global=True):
    """Optimize using MW best-response (global or single-constraint)."""
    a = np.maximum(a.astype(np.float64), 1e-8)
    a = a / a.sum()
    n = len(a)
    current_J = compute_J(a)
    tracker.update(a)
    
    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        
        b = fast_conv(a)
        M = float(b.max())
        M_tight = M * (1.0 - tighten)
        
        if use_global:
            g = mw_best_response_global(a, M_tight, n_iters=200, eta=eta)
        else:
            g = mw_best_response_single(a, M_tight, n_iters=150, eta=eta)
        
        if g is None:
            if use_global:
                g = mw_best_response_global(a, M, n_iters=100, eta=eta)
            else:
                g = mw_best_response_single(a, M, n_iters=100, eta=eta)
        if g is None:
            break
        
        best_a = None
        best_J = current_J
        progress = it / max(1, max_iters)
        base_t = max(0.1, 0.6 * (1.0 - progress * 0.5))
        for t in [base_t, base_t * 0.5, base_t * 1.5, 0.9, 0.2]:
            if t > 1.0 or t <= 0:
                continue
            a_trial = (1.0 - t) * a + t * g
            a_trial = np.maximum(a_trial, 0.0)
            ss = a_trial.sum()
            if ss < 1e-10:
                continue
            a_trial = a_trial / ss
            J_trial = compute_J(a_trial)
            if J_trial < best_J - 1e-12:
                best_J = J_trial
                best_a = a_trial
        
        if best_a is None:
            break
        a = best_a
        current_J = best_J
        if it % 3 == 0:
            tracker.update(a)
    
    tracker.update(a)
    return a


def run_lp_opt(a, max_iters, deadline, tracker, tighten=0.01):
    """Optimize using LP best-response."""
    a = np.maximum(a.astype(np.float64), 1e-8)
    a = a / a.sum()
    n = len(a)
    current_J = compute_J(a)
    tracker.update(a)
    
    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        
        b = fast_conv(a)
        M = float(b.max())
        M_tight = M * (1.0 - tighten)
        
        g = lp_best_response(a, M_tight, time_limit=8)
        if g is None:
            g = lp_best_response(a, M, time_limit=5)
        if g is None:
            break
        
        best_a = None
        best_J = current_J
        for t in [0.3, 0.5, 0.7, 0.9, 0.15]:
            a_trial = (1.0 - t) * a + t * g
            a_trial = np.maximum(a_trial, 0.0)
            ss = a_trial.sum()
            if ss < 1e-10:
                continue
            a_trial = a_trial / ss
            J_trial = compute_J(a_trial)
            if J_trial < best_J - 1e-12:
                best_J = J_trial
                best_a = a_trial
        
        if best_a is None:
            break
        a = best_a
        current_J = best_J
        tracker.update(a)
    
    tracker.update(a)
    return a


def lse_gradient_descent(a, max_iters, deadline, tracker, tau_init=30.0):
    """LSE gradient descent with Nesterov momentum and adaptive tau."""
    a = np.maximum(a.astype(np.float64), 1e-6)
    a = a / a.sum()
    n = len(a)
    current_J = compute_J(a)
    tracker.update(a)
    
    tau = tau_init
    lr = 0.5 / n
    momentum = 0.85
    v = np.zeros(n)
    
    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        
        b = fast_conv(a)
        b_max = b.max()
        exp_b = np.exp(tau * (b - b_max))
        p = exp_b / (exp_b.sum() + 1e-30)
        
        s = 1
        while s < 3 * n:
            s <<= 1
        fp = np.fft.rfft(p, s)
        fa = np.fft.rfft(a, s)
        corr = np.fft.irfft(fp * np.conj(fa), s)
        grad = 2.0 * corr[:n]
        
        v = momentum * v + (1.0 - momentum) * grad
        a_new = a - lr * v
        a_new = np.maximum(a_new, 1e-6)
        a_new = a_new / a_new.sum()
        
        J_new = compute_J(a_new)
        if J_new < current_J - 1e-14:
            a = a_new
            current_J = J_new
            tau = min(300.0, tau * 1.015)
            lr = min(lr * 1.01, 2.0 / n)
        else:
            lr *= 0.6
            v *= 0.3
            if lr < 1e-18:
                break
        
        if it % 10 == 0:
            tracker.update(a)
    
    tracker.update(a)
    return a


def subgradient_descent(a, max_iters, deadline, tracker):
    """Subgradient descent on the exact max function."""
    a = np.maximum(a.astype(np.float64), 1e-8)
    a = a / a.sum()
    n = len(a)
    current_J = compute_J(a)
    tracker.update(a)
    lr = 2.0 / n
    
    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        
        b = fast_conv(a)
        M = float(b.max())
        eps = max(1e-8, 1e-6 * M)
        active = np.where(b >= M - eps)[0]
        
        grad = np.zeros(n)
        for k in active:
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            if je >= js:
                grad[js:je + 1] += a[k - np.arange(js, je + 1)]
        grad /= len(active)
        
        grad_centered = grad - grad.mean()
        a_new = a - lr * grad_centered
        a_new = np.maximum(a_new, 1e-8)
        a_new = a_new / a_new.sum()
        
        J_new = compute_J(a_new)
        if J_new < current_J - 1e-15:
            a = a_new
            current_J = J_new
            lr = min(lr * 1.03, 8.0 / n)
        else:
            lr *= 0.7
        
        if it % 30 == 0:
            tracker.update(a)
    
    tracker.update(a)
    return a


def generate_inits(n, rng, parent=None):
    """Generate diverse initializations."""
    inits = []
    
    if parent is not None and parent.size > 0:
        t_old = np.linspace(0, 1, parent.size)
        t_new = np.linspace(0, 1, n)
        p = np.maximum(np.interp(t_new, t_old, parent), 0.0)
        if p.sum() > 1e-10:
            inits.append(p / p.sum())
        pr = p[::-1]
        if pr.sum() > 1e-10:
            inits.append(pr / pr.sum())
    
    inits.append(np.ones(n) / n)
    inits.append(rng.uniform(0.1, 1.0, n))
    
    # Flat spectrum (multiple random phase realizations)
    for _ in range(3):
        half = n // 2 + 1
        angles = rng.uniform(0, 2 * np.pi, half)
        A = np.zeros(n, dtype=complex)
        A[:half] = np.exp(1j * angles[:half])
        for k in range(1, n - n // 2):
            A[k] = np.conj(A[n - k])
        a = np.fft.ifft(A).real
        a = np.maximum(a, 0.0)
        if a.sum() > 1e-10:
            inits.append(a / a.sum())
    
    t = np.linspace(0, 1, n, endpoint=False)
    
    # Single frequency modulations
    for freq in [2, 3, 5, 7, 11]:
        inits.append(1.0 + 0.4 * np.cos(2 * np.pi * freq * t + rng.uniform(0, 6.28)))
    
    # Multi-frequency
    inits.append(1.0 + 0.25 * np.sin(2 * np.pi * t) + 0.2 * np.sin(4 * np.pi * t)
                + 0.15 * np.sin(7 * np.pi * t) + 0.1 * np.sin(11 * np.pi * t))
    inits.append(1.0 + 0.3 * np.cos(2 * np.pi * 3.7 * t) + 0.2 * np.cos(2 * np.pi * 7.3 * t)
                + 0.15 * np.cos(2 * np.pi * 11.1 * t))
    
    # Bell shapes
    inits.append(0.5 + 0.5 * np.cos(np.pi * t) ** 2)
    inits.append(0.3 + 0.7 * (1.0 - np.abs(2 * t - 1.0)) ** 0.5)
    
    # Two-lobe Gaussian
    inits.append(0.5 * np.exp(-8 * (t - 0.3) ** 2) + 0.5 * np.exp(-8 * (t - 0.7) ** 2) + 0.05)
    
    # Sparse + smooth
    sparse = np.zeros(n)
    n_spikes = max(5, n // 12)
    positions = rng.choice(n, size=n_spikes, replace=False)
    sparse[positions] = 1.0
    kernel_size = max(3, n // 25)
    kernel = np.ones(kernel_size) / kernel_size
    inits.append(np.convolve(sparse, kernel, mode='same') + 0.05)
    
    # Power law
    inits.append((t + 0.05) ** rng.uniform(-0.5, 1.5))
    inits.append((1.0 - t + 0.05) ** rng.uniform(-0.5, 1.5))
    
    # Random Fourier (limited bandwidth)
    n_coeff = min(n // 2 + 1, 10)
    coeffs = rng.normal(0, 1, n_coeff)
    sig = np.zeros(n)
    for i in range(n_coeff):
        sig += coeffs[i] * np.cos(2 * np.pi * i * t / n + rng.uniform(0, 6.28))
    inits.append(np.abs(sig) + 0.5)
    
    # Triangular wave
    inits.append(1.0 - 2.0 * np.abs(np.sin(np.pi * t)))
    
    # Cosine with zero crossing
    inits.append(0.5 + 0.5 * np.cos(2 * np.pi * t * rng.uniform(1, 3)))
    
    cleaned = []
    for x in inits:
        x = np.maximum(np.asarray(x, dtype=np.float64), 0.001)
        s = x.sum()
        if s > 1e-10:
            cleaned.append(x / s)
    return cleaned


class ScoreTracker:
    def __init__(self):
        self.best = None
        self.best_score = float('inf')
        self.t_start = time.monotonic()
    
    def update(self, a):
        global _global_best, _global_best_score
        a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
        if a.size == 0 or a.sum() < 0.01:
            return
        try:
            score = float(evaluate_sequence(a.tolist()))
        except Exception:
            return
        if np.isfinite(score) and score < self.best_score:
            self.best = a.copy()
            self.best_score = score
            _global_best = a.copy()
            _global_best_score = score
            elapsed = time.monotonic() - self.t_start
            print(f"  * J={score:.8f} n={a.size} t={elapsed:.1f}s", flush=True)
    
    def report(self, label=""):
        if self.best is not None:
            print(f"  {label} best: J={self.best_score:.8f} n={self.best.size}", flush=True)


def coordinate_descent(a, max_steps, deadline, tracker):
    """Randomized coordinate descent with multiple scales."""
    a = a.copy().astype(np.float64)
    a = a / a.sum()
    n = len(a)
    current_J = compute_J(a)
    tracker.update(a)
    rng = np.random.default_rng(42)
    
    for step in range(max_steps):
        if time.monotonic() > deadline:
            break
        j = int(rng.integers(0, n))
        improved = False
        for scale in [0.5, 1.5, 0.2, 2.0, 0.7, 1.3]:
            a_trial = a.copy()
            a_trial[j] *= scale
            a_trial = np.maximum(a_trial, 1e-8)
            ss = a_trial.sum()
            if ss < 1e-10:
                continue
            a_trial = a_trial / ss
            J_trial = compute_J(a_trial)
            if J_trial < current_J - 1e-13:
                a = a_trial
                current_J = J_trial
                improved = True
                break
        if step % 200 == 0:
            tracker.update(a)
    tracker.update(a)
    return a


def propose_candidate(seed=42, budget_s=1500, **kwargs):
    """Main search function. Returns the best sequence found."""
    global _global_best, _global_best_score
    
    rng = np.random.default_rng(seed)
    total_time = float(budget_s)
    deadline = time.monotonic() + total_time - 25.0
    tracker = ScoreTracker()
    
    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        if p.ndim == 1 and p.size > 0 and np.all(np.isfinite(p)):
            parent = np.maximum(p, 0.0)
            tracker.update(parent)
            print(f"Parent: n={parent.size}, J={tracker.best_score:.8f}", flush=True)
    except (NameError, AttributeError):
        pass
    
    # === Phase 1: Fast MW exploration (55% of time) ===
    t_phase1_end = time.monotonic() + 0.55 * (total_time - 60)
    
    lengths = [384, 512, 640]
    time_per_length = max(30.0, (t_phase1_end - time.monotonic() - 60) / len(lengths))
    
    for n in lengths:
        if time.monotonic() > t_phase1_end - 60:
            break
        t_phase_end = min(t_phase1_end - 30, time.monotonic() + time_per_length)
        print(f"\n{'='*50} Phase1 n={n} {'='*50}", flush=True)
        
        inits = generate_inits(n, rng, parent)
        best_for_n = None
        best_J_for_n = float('inf')
        n_inits = len(inits)
        
        init_budget = max(15.0, (t_phase_end - time.monotonic()) / n_inits)
        
        for idx, a0 in enumerate(inits):
            if time.monotonic() > t_phase_end - 30:
                break
            t0_init = time.monotonic()
            print(f"  Init {idx}/{n_inits} (budget {init_budget:.0f}s)", flush=True)
            
            # MW optimization (fast)
            a = run_mw_opt(a0, 30, t0_init + init_budget * 0.5, tracker, eta=0.08, tighten=0.03)
            
            # Flat spectrum projection (quick fix)
            if time.monotonic() < t_phase_end - 20:
                a = flat_spectrum_project(a, n_iters=30, alpha=0.3)
                tracker.update(a)
            
            # More MW with tighter constraint
            if time.monotonic() < t_phase_end - 20:
                a = run_mw_opt(a, 20, t_phase_end - 20, tracker, eta=0.06, tighten=0.06)
            
            # LSE polish
            if time.monotonic() < t_phase_end - 10:
                a = lse_gradient_descent(a, 500, t_phase_end - 10, tracker, tau_init=20.0)
            
            J = compute_J(a)
            if J < best_J_for_n:
                best_J_for_n = J
                best_for_n = a.copy()
        
        if best_for_n is not None:
            tracker.update(best_for_n)
            print(f"  Phase1 n={n} best local: J={best_J_for_n:.8f}", flush=True)
    
    # === Phase 2: LP refinement (30% of time) ===
    t_phase2_start = time.monotonic()
    t_phase2_end = t_phase2_start + 0.30 * (total_time - 60)
    
    if tracker.best is not None and time.monotonic() < t_phase2_end - 30:
        print(f"\n{'='*50} Phase2 LP Refinement n={tracker.best.size} {'='*50}", flush=True)
        
        a = tracker.best.copy().astype(np.float64)
        a = a / a.sum()
        
        # Multiple LP passes with increasing tightening
        tighten_sched = [0.005, 0.01, 0.015, 0.02, 0.025, 0.03, 0.04, 0.05]
        for tight in tighten_sched:
            if time.monotonic() > t_phase2_end - 30:
                break
            a = run_lp_opt(a, 15, t_phase2_end - 30, tracker, tighten=tight)
        
        # Interleave with MW for speed
        if time.monotonic() < t_phase2_end - 20:
            a = run_mw_opt(a, 20, t_phase2_end - 20, tracker, eta=0.05, tighten=0.02, use_global=False)
        
        # Subgradient polish
        if time.monotonic() < t_phase2_end - 10:
            a = subgradient_descent(a, 2000, t_phase2_end - 10, tracker)
        
        tracker.report("Phase2")
    
    # === Phase 3: Fine polishing (15% of time) ===
    t_phase3_end = deadline
    
    if tracker.best is not None and time.monotonic() < t_phase3_end - 10:
        print(f"\n{'='*50} Phase3 Polishing n={tracker.best.size} {'='*50}", flush=True)
        
        a = tracker.best.copy().astype(np.float64)
        a = a / a.sum()
        
        # LSE with small tau (smooth)
        if time.monotonic() < t_phase3_end - 40:
            a = lse_gradient_descent(a, 3000, min(t_phase3_end - 40, time.monotonic() + 120), tracker, tau_init=15.0)
        
        # Subgradient
        if time.monotonic() < t_phase3_end - 30:
            a = subgradient_descent(a, 3000, min(t_phase3_end - 30, time.monotonic() + 100), tracker)
        
        # Flat spectrum projection (gentle)
        if time.monotonic() < t_phase3_end - 25:
            a = flat_spectrum_project(a, n_iters=20, alpha=0.2)
            tracker.update(a)
        
        # Final LP
        if time.monotonic() < t_phase3_end - 20:
            a = run_lp_opt(a, 10, min(t_phase3_end - 20, time.monotonic() + 60), tracker, tighten=0.005)
        
        # Coordinate descent
        if time.monotonic() < t_phase3_end - 10:
            a = coordinate_descent(a, 2000, min(t_phase3_end - 10, time.monotonic() + 60), tracker)
        
        # Final LSE
        if time.monotonic() < t_phase3_end - 5:
            a = lse_gradient_descent(a, 1000, t_phase3_end - 5, tracker, tau_init=10.0)
        
        tracker.report("Phase3")
    
    # Fallback
    if tracker.best is None:
        tracker.best = np.ones(256)
        tracker.best_score = compute_J(tracker.best)
        tracker.update(tracker.best)
    
    print(f"\n{'='*50} FINAL {'='*50}", flush=True)
    print(f"J={tracker.best_score:.8f} n={tracker.best.size} time={time.monotonic()-tracker.t_start:.1f}s", flush=True)
    
    result = [float(max(0.0, min(1000.0, x))) for x in tracker.best]
    return result
