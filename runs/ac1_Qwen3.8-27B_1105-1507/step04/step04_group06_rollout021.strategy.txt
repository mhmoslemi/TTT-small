def create_initial_sequence(length, rng, family_index=None):
    """Generate an initial sequence of specified length and family type."""
    t_arr = np.linspace(0, 1, length, endpoint=False)

def tournament_selection(population, fitnesses, tournament_size=4, rng=None):
    """Select `tournament_size` candidates and return the best one determined by their fitnesses."""
    if rng is None:
        rng = np.random.default_rng()
    indices = []
    for _ in range(tournament_size):
        sample_indices = rng.choice(population.size, size=tournament_size)
        best_idx = sample_indices[np.argmin([fitnesses[i] for i in sample_indices])]
        indices.append(best_idx)
    winner_idx = np.mean(indices)  # average of indices for deterministic child
