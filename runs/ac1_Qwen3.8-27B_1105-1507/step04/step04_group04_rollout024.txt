"""

import numpy as np
from scipy.optimize import minimize, linprog
import time


def compute_conv(a):
    """Compute linear convolution a*a using FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    conv = np.fft.irfft(fa * fa, size)
    return conv[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    conv = compute_conv(a)
    max_b = float(np.max(conv))
    sum_a = float(np.sum(a))
    if sum_a < 0.01:
        return np.inf
    return 2.0 * n * max_b / (sum_a * sum_a)


def normalize(a):
    """Normalize a to sum 1, ensuring non-negativity."""
    a = np.maximum(a, 0.0)
    s = np.sum(a)
    if s < 1e-15:
        return np.ones(len(a)) / len(a)
    return a / s


def lse_smooth_obj_grad(a, tau):
    """
    Compute smooth objective and its gradient.
    Uses log-sum-exp approximation of max over all 2n-1 convolution terms.
    Assumes a is already normalized (sum=1).
    Returns (obj, grad) where grad is d(obj)/d(a).
    """
    n = len(a)
    conv = compute_conv(a)
    max_c = float(np.max(conv))
    shifted = tau * (conv - max_c)
    # Numerically stable log-sum-exp
    max_shifted = np.max(shifted)
    lse = np.log(np.sum(np.exp(shifted - max_shifted))) + max_shifted + tau * max_c
    obj = 2.0 * n * lse / tau

    # Weights for the gradient
    weights = np.exp(shifted - max_shifted)
    weights /= np.sum(weights)

    # Gradient computation:
    # grad[j] = tau * sum_k w[k] * (a[k-j] + a[j-k]) with boundary conditions
    # term1[j] = sum_k w[k]*a[k-j] (valid when 0 <= k-j < n)
    # term2[j] = sum_k w[k]*a[j-k] (valid when 0 <= j-k < n)

    # term2 = convolve(weights, a)[:n]
    term2 = np.convolve(weights, a, mode='full')[:n]

    # term1[j] = sum_m a[m]*w[j+m]
    # = (a * w_flipped)[2n-2-j]
    w_flipped = weights[::-1]
    conv_awf = np.convolve(a, w_flipped, mode='full')
    term1 = conv_awf[n - 1:2 * n - 1][::-1]

    grad = tau * (term1 + term2)
    return obj, grad


def softmax_logspace_obj(x, tau):
    """
    Objective in log-space using softmax parameterization.
    x -> a = softmax(x), then compute smooth objective.
    """
    x_shifted = x - np.max(x)
    a = np.exp(x_shifted)
    a /= np.sum(a)
    obj, _ = lse_smooth_obj_grad(a, tau)
    return obj


def softmax_logspace_grad(x, tau):
    """
    Gradient of the smooth objective with respect to x (log-space).
    Uses chain rule: d_obj/dx_k = a_k * (grad_a_k - dot(a, grad_a))
    """
    x_shifted = x - np.max(x)
    a = np.exp(x_shifted)
    a /= np.sum(a)
    _, grad_a = lse_smooth_obj_grad(a, tau)
    dot_val = np.dot(a, grad_a)
    grad_x = a * (grad_a - dot_val)
    return grad_x


def lbfgs_optimize_in_logspace(a_init, tau, maxiter=500, time_budget=60.0):
    """
    Optimize using L-BFGS-B in log-space (softmax parameterization).
    a_init should be positive (will be log-transformed).
    Returns optimized a (normalized).
    """
    n = len(a_init)
    a_init = np.maximum(a_init, 1e-15)
    x0 = np.log(a_init)
    x0 -= np.max(x0)  # for numerical stability

    start = time.monotonic()

    def fun(x):
        if time.monotonic() - start > time_budget:
            return np.inf
        return softmax_logspace_obj(x, tau)

    def fprime(x):
        if time.monotonic() - start > time_budget:
            return np.zeros(n)
        return softmax_logspace_grad(x, tau)

    res = minimize(fun, x0, jac=fprime, method='L-BFGS-B',
                   options={'maxiter': maxiter, 'ftol': 1e-15, 'gtol': 1e-10,
                            'maxfun': int(time_budget * 20), 'maxls': 50})

    x_opt = res.x
    a_opt = np.exp(x_opt - np.max(x_opt))
    a_opt /= np.sum(a_opt)
    return a_opt


def multiplicative_subgradient_step(a, tau, step):
    """One multiplicative subgradient step (operates on normalized a)."""
    n = len(a)
    conv = compute_conv(a)
    max_conv = np.max(conv)
    shifted = tau * (conv - max_conv)
    shifted = np.clip(shifted, -100, 100)
    weights = np.exp(shifted)
    w_sum = np.sum(weights)
    if w_sum < 1e-300:
        return a
    weights /= w_sum

    # Compute gradient
    w_flipped = weights[::-1]
    conv_awf = np.convolve(a, w_flipped, mode='full')
    term1 = conv_awf[n - 1:2 * n - 1][::-1]
    term2 = np.convolve(weights, a, mode='full')[:n]
    grad = 2.0 * (term1 + term2)

    # Center gradient to preserve sum approximately
    grad_centered = grad - np.mean(grad)

    log_update = -step * grad_centered
    log_update = np.clip(log_update, -50, 50)
    a_new = a * np.exp(log_update)
    a_new = np.maximum(a_new, 1e-15)
    a_new = a_new / np.sum(a_new)
    return a_new


def subgradient_optimize(a, max_iters=3000, time_budget=20.0, tau=20.0, step=0.05):
    """Subgradient optimization with adaptive parameters."""
    n = len(a)

            best_obj = obj
            best_a = a.copy()
            stall = 0
        else:
            stall += 1
            if stall > patience:

                tau *= 1.1
                if step < 0.003:

                # Perturb to escape

                stall = 0

def build_toeplitz(a):
    """Build Toeplitz matrix A where (A@g)[k] = (a*g)[k] for k=0..2n-2."""
    n = len(a)
    A = np.zeros((2 * n - 1, n), dtype=np.float64)
    for j in range(n):
        k_start = j
        k_end = min(j + n, 2 * n - 1)
        A[k_start:k_end, j] = a[:k_end - k_start]
    return A


def solve_lp_maxsum(a, M, time_limit=5):
    """Solve LP: max sum(g) s.t. (a*g)[k] <= M for all k, g >= 0."""
    n = len(a)
    A = build_toeplitz(a)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': time_limit})
        if res.status == 0:
            g = np.maximum(res.x, 0)
            return g, float(np.sum(g))
    except Exception:
        pass
    return None, None


def solve_lp_minpeak(a, time_limit=5):
    """Solve LP: min M s.t. (a*g)[k] <= M for all k, sum(g)=1, g>=0."""
    n = len(a)
    A = build_toeplitz(a)
    c = np.zeros(n + 1)
    c[-1] = 1.0
    A_ub = np.zeros((2 * n - 1, n + 1))
    A_ub[:, :n] = A
    A_ub[:, n] = -1.0
    b_ub = np.zeros(2 * n - 1)
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])
    bounds = [(0.0, None)] * n + [(0.0, None)]
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': time_limit})
        if res.status == 0:
            g = np.maximum(res.x[:n], 0)
            s = np.sum(g)
            if s < 1e-10:
                return None, None
            g = g / s
            return g, float(res.x[n])
    except Exception:
        pass
    return None, None


def alternating_lp(a, max_iters=60, time_budget=30.0):
    """Alternating LP with adaptive line search."""
    n = len(a)
    a = normalize(a.copy().astype(np.float64))
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    stall = 0

    for it in range(max_iters):
        if time.monotonic() - start > time_budget - 2:
            break

        conv = compute_conv(a)
        M = float(np.max(conv))

        g, sum_g = solve_lp_maxsum(a, M, time_limit=4)
        if g is None or sum_g < 1e-10:
            break
        g_norm = g / sum_g

        # Line search
        best_t = 0.0
        best_obj_ls = best_obj
        for t in np.linspace(0.1, 1.0, 10):
            a_new = (1.0 - t) * a + t * g_norm
            a_new = np.maximum(a_new, 0)
            s_new = np.sum(a_new)
            if s_new < 1e-10:
                continue
            a_new = a_new / s_new
            obj_new = compute_obj(a_new)
            if obj_new < best_obj_ls:
                best_obj_ls = obj_new
                best_t = t

        if best_t > 0 and best_obj_ls < best_obj - 1e-14:
            a_new = (1.0 - best_t) * a + best_t * g_norm
            a_new = np.maximum(a_new, 0)
            a_new = a_new / np.sum(a_new)
            best_obj = best_obj_ls
            best_a = a_new.copy()
            a = a_new
            stall = 0
        else:
            stall += 1
            if stall > 30:
                break
            a = 0.9 * a + 0.1 * g_norm
            a = normalize(a)
    return best_a

def pair_swap_search(a, time_budget=30.0, rng_seed=42):
    """
    Local search: randomly swap small amounts between pairs of coordinates.
    Accept if the smooth objective decreases.
    """
    n = len(a)
    a = normalize(a.copy().astype(np.float64))
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    rng = np.random.default_rng(rng_seed)

    tau = 30.0
    eps_start = 0.02
    it = 0

    while time.monotonic() - start < time_budget - 1:
        it += 1
        # Decrease epsilon over time
        frac = (time.monotonic() - start) / time_budget
        eps = eps_start * (1.0 - 0.8 * frac) + 0.001

        i = rng.integers(0, n)
        j = rng.integers(0, n)
        if i == j:
            continue

        # Try moving eps from a[i] to a[j] (and vice versa)
        a_trial = a.copy()
        a_trial[i] -= eps
        a_trial[j] += eps
        if a_trial[i] < 0:
            a_trial[i] = 0
            a_trial[j] += a_trial[j]  # just skip
            continue
        a_trial = normalize(a_trial)

        obj_trial = compute_obj(a_trial)
        if obj_trial < best_obj - 1e-15:
            best_obj = obj_trial
            best_a = a_trial.copy()
            a = a_trial

        # Also try the reverse
        a_trial2 = a.copy()
        a_trial2[j] -= eps
        a_trial2[i] += eps
        if a_trial2[j] < 0:
            continue
        a_trial2 = normalize(a_trial2)
        obj_trial2 = compute_obj(a_trial2)
        if obj_trial2 < best_obj - 1e-15:
            best_obj = obj_trial2
            best_a = a_trial2.copy()
            a = a_trial2

    return best_a


def random_init(n, seed):
    """Generate diverse random initializations."""
    rng = np.random.default_rng(seed)
    k = np.arange(n)
    choice = rng.integers(0, 8)

    if choice == 0:
        f = rng.uniform(0.5, 4.0)
        phase = rng.uniform(0, 2 * np.pi)
        a = 1.0 + 0.7 * np.cos(2 * np.pi * f * k / n + phase)
    elif choice == 1:
        f1 = rng.uniform(0.3, 2.0)
        f2 = rng.uniform(2.0, 6.0)
        a = 1.0 + 0.3 * np.cos(2 * np.pi * f1 * k / n) + 0.3 * np.cos(2 * np.pi * f2 * k / n)
    elif choice == 2:
        center = n * rng.uniform(0.3, 0.7)
        alpha = rng.uniform(0.5, 3.0)
        a = 1.0 + 0.5 * np.exp(-alpha * np.abs(k - center) / (n * 0.3))
    elif choice == 3:
        a = rng.random(n)
        win = max(3, n // 15)
        kernel = np.ones(win) / win
        a = np.convolve(a, kernel, mode='same')
    elif choice == 4:
        a = np.ones(n)
        for _ in range(rng.integers(1, 4)):
            f = rng.uniform(0.5, 5.0)
            amp = rng.uniform(0.1, 0.4)
            a += amp * np.cos(2 * np.pi * f * k / n + rng.uniform(0, 2 * np.pi))
    elif choice == 5:
        a = np.ones(n)
        for scale in range(3):
            f = rng.uniform(0.3 * (2 ** scale), 1.5 * (2 ** scale))
            amp = 0.25 / (scale + 1)
            a += amp * np.cos(2 * np.pi * f * k / n + rng.uniform(0, 2 * np.pi))
    elif choice == 6:
        # Multi-bump
        n_bumps = rng.integers(2, 5)
        a = np.ones(n)
        for _ in range(n_bumps):
            center = rng.uniform(0, n)
            width = n * rng.uniform(0.05, 0.3)
            amp = rng.uniform(0.2, 0.8) * (rng.random() > 0.5 ? 1 : -1)
            a += amp * np.exp(-0.5 * ((k - center) / width) ** 2)
    else:
        # Sine with random harmonics
        a = np.ones(n)
        n_harm = rng.integers(2, 6)
        for h in range(n_harm):
            f = rng.uniform(0.5, 3.0) * (h + 1)
            amp = rng.uniform(0.05, 0.3) / (h + 1)

    return normalize(a)


def resample_sequence(a, new_n):
    """Resample a sequence to a new length using interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    old_k = np.linspace(0, 1, n)
    new_k = np.linspace(0, 1, new_n)
    a_new = np.interp(new_k, old_k, a)
    a_new = np.maximum(a_new, 0)
    return normalize(a_new)


def try_parent():
    """Try to load parent sequence."""
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64)
        parent = np.clip(parent, 0, 1000.0)
        if parent.size > 0 and np.isfinite(parent).all():
            obj = compute_obj(parent)
            print(f"Parent (n={parent.size}): obj={obj:.6f}")
            return parent.copy(), obj
    except Exception:
        pass
    return None, np.inf


def full_pipeline_optimize(a, time_budget=90.0, verbose=False):
    """Run full optimization pipeline: L-BFGS -> subgradient -> LP -> pair swap."""
    n = len(a)
    a = normalize(a.copy().astype(np.float64))
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    if verbose:
        print(f"    Start: obj={best_obj:.8f}")

    # Phase 1: L-BFGS in log-space (broad convergence)
    remaining = time_budget - (time.monotonic() - start)
    if remaining > 10:
        for tau in [15.0, 30.0, 60.0]:
            remaining = time_budget - (time.monotonic() - start)
            if remaining < 8:
                break
            t_budget = min(remaining * 0.25, 20.0)
            a_new = lbfgs_optimize_in_logspace(best_a, tau, maxiter=200, time_budget=t_budget)
            obj = compute_obj(a_new)
            if obj < best_obj:
                best_obj = obj
                best_a = a_new.copy()
                if verbose:
                    print(f"    After L-BFGS (tau={tau}): {best_obj:.8f}")
            else:
                break

    # Phase 2: Subgradient refinement
    remaining = time_budget - (time.monotonic() - start)
    if remaining > 8:
        sg_time = min(remaining * 0.2, 15.0)
        a_new = subgradient_optimize(best_a, max_iters=int(sg_time * 30),
                                     time_budget=sg_time, tau=25.0, step=0.04)
        obj = compute_obj(a_new)
        if obj < best_obj:
            best_obj = obj
            best_a = a_new.copy()
            if verbose:
                print(f"    After SG: {best_obj:.8f}")

    # Phase 3: Alternating LP
    remaining = time_budget - (time.monotonic() - start)
    if remaining > 15:
        lp_time = min(remaining * 0.3, 25.0)
        a_new = alternating_lp(best_a, max_iters=40, time_budget=lp_time)
        obj = compute_obj(a_new)
        if obj < best_obj:
            best_obj = obj
            best_a = a_new.copy()
            if verbose:
                print(f"    After LP: {best_obj:.8f}")

    # Phase 4: Min-peak LP
    remaining = time_budget - (time.monotonic() - start)
    if remaining > 15:
        lp_time = min(remaining * 0.2, 20.0)
        a_new = alternating_lp_minpeak(best_a, max_iters=30, time_budget=lp_time)
        obj = compute_obj(a_new)
        if obj < best_obj:
            best_obj = obj
            best_a = a_new.copy()
            if verbose:
                print(f"    After LP-MP: {best_obj:.8f}")

    # Phase 5: Pair swap local search
    remaining = time_budget - (time.monotonic() - start)
    if remaining > 10:
        ps_time = min(remaining * 0.3, 20.0)
        a_new = pair_swap_search(best_a, time_budget=ps_time, rng_seed=12345)
        obj = compute_obj(a_new)
        if obj < best_obj:
            best_obj = obj
            best_a = a_new.copy()
            if verbose:
                print(f"    After PS: {best_obj:.8f}")

    # Phase 6: Final L-BFGS polish with high tau
    remaining = time_budget - (time.monotonic() - start)
    if remaining > 5:
        t_budget = min(remaining * 0.4, 15.0)
        a_new = lbfgs_optimize_in_logspace(best_a, tau=100.0, maxiter=300, time_budget=t_budget)
        obj = compute_obj(a_new)
        if obj < best_obj:
            best_obj = obj
            best_a = a_new.copy()
            if verbose:
                print(f"    Final L-BFGS: {best_obj:.8f}")

    return best_a, best_obj


def intensive_refine(a, time_budget=60.0, verbose=False):
    """
    Intensive refinement: multiple rounds of perturb + optimize.
    Tries different perturbation scales and types.
    """
    n = len(a)
    a = normalize(a.copy().astype(np.float64))
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    if verbose:
        print(f"  Intensive refine start: {best_obj:.8f}")

    attempt = 0
    while time.monotonic() - start < time_budget - 10:
        attempt += 1
        remaining = time_budget - (time.monotonic() - start)
        if remaining < 15:
            break

        rng = np.random.default_rng(attempt * 7919 + 31)

        # Different perturbation strategies
        strategy = attempt % 4

        if strategy == 0:
            # Multiplicative noise
            scale = 0.03 + 0.04 * (attempt % 3) / 3.0
            a_pert = best_a * np.exp(rng.normal(0, scale, n))
        elif strategy == 1:
            # Additive noise
            a_pert = best_a + rng.normal(0, 0.01 + 0.02 * (attempt % 3) / 3.0, n)
        elif strategy == 2:
            # Smooth perturbation (low-frequency)
            noise = rng.normal(0, 1, max(3, n // 10))
            noise_full = np.interp(np.arange(n), np.arange(len(noise)), noise)
            a_pert = best_a + 0.05 * noise_full
        else:
            # Targeted: increase values where autocorrelation is low
            conv = compute_conv(best_a)
            target = 1.0 / (n + 1)
            deficit = np.maximum(0, target - conv[:n])
            a_pert = best_a + 0.02 * deficit[:n] if len(deficit) >= n else best_a

        a_pert = np.maximum(a_pert, 1e-10)
        a_pert = normalize(a_pert)

        sub_budget = min(15.0, remaining * 0.35)
        a_opt, obj = full_pipeline_optimize(a_pert, time_budget=sub_budget, verbose=False)

        if obj < best_obj:
            best_obj = obj
            best_a = a_opt.copy()
            if verbose:
                print(f"    Attempt {attempt} (strat={strategy}): {best_obj:.8f}")

    return best_a, best_obj


def propose_candidate(seed=42, budget_s=1500):
    """
    Main search function. Combines L-BFGS, subgradient, alternating LP,
    and pair-swap search across multiple n values.
    Returns the best sequence found within the time budget.
    """
    start_time = time.monotonic()
    deadline = start_time + float(budget_s) - 30.0

    best_seq = None
    best_score = np.inf

    # Load parent
    parent, parent_obj = try_parent()
    if parent is not None and parent_obj < best_score:
        best_score = parent_obj
        best_seq = parent.copy()
        n_parent = len(parent)
    else:
        n_parent = 330
        print("No parent available, starting fresh.")

    print(f"Initial: n={n_parent}, score={best_score:.6f}, budget={budget_s}s")

    # === Phase 1: Intensive optimization of parent ===
    if best_seq is not None:
        remaining = deadline - time.monotonic()
        p1_budget = min(remaining * 0.3, 350.0)
        print(f"\n=== Phase 1: Optimize parent (budget={p1_budget:.0f}s) ===")

        a_opt, obj = full_pipeline_optimize(best_seq, time_budget=p1_budget * 0.5, verbose=True)
        if obj < best_score:
            best_score = obj
            best_seq = a_opt.copy()
            print(f"  After full pipeline: {best_score:.8f}")

        remaining = deadline - time.monotonic()
        p1b_budget = min(remaining * 0.2, 150.0)
        if p1b_budget > 20:
            a_opt, obj = intensive_refine(best_seq, time_budget=p1b_budget, verbose=True)
            if obj < best_score:
                best_score = obj
                best_seq = a_opt.copy()
                print(f"  After intensive refine: {best_score:.8f}")

    # === Phase 2: Try nearby n values ===
    print(f"\n=== Phase 2: Nearby n values ===")
    n_values = []
    for delta in range(-30, 31, 10):
        n_val = n_parent + delta
        if 150 <= n_val <= 800:
            n_values.append(n_val)
    # Add some strategic values
    for n_val in [250, 400, 450, 500]:
        if n_val not in n_values and 150 <= n_val <= 800:
            n_values.append(n_val)
    n_values = sorted(set(n_values))

    remaining = deadline - time.monotonic()
    if remaining > 60:
        per_n = remaining / max(len(n_values), 1) * 1.5
        for n_val in n_values:
            if time.monotonic() > deadline - 60:
                break
            remaining = deadline - time.monotonic()
            if remaining < 25:
                break
            alloc = min(per_n, remaining * 0.25, 80.0)
            alloc = max(alloc, 20.0)

            print(f"\n  n={n_val} ({remaining:.0f}s left, alloc={alloc:.0f}s)")
            # Use parent as warm start
            a_init = resample_sequence(best_seq, n_val) if best_seq is not None else random_init(n_val, seed + n_val)
            a_opt, obj = full_pipeline_optimize(a_init, time_budget=alloc, verbose=(n_val == n_parent))
            if obj < best_score:
                best_score = obj
                best_seq = a_opt.copy()
                print(f"  ** UPDATE: n={n_val}, score={best_score:.8f} **")

    # === Phase 3: Final intensive refinement ===
    if best_seq is not None:
        remaining = deadline - time.monotonic()
        if remaining > 40:
            print(f"\n=== Phase 3: Final refinement ({remaining:.0f}s) ===")
            p3_budget = min(remaining * 0.5, remaining - 20.0)
            a_opt, obj = intensive_refine(best_seq, time_budget=p3_budget, verbose=True)
            if obj < best_score:
                best_score = obj
                best_seq = a_opt.copy()
                print(f"  After final refine: {best_score:.8f}")

            # One more full pipeline
            remaining = deadline - time.monotonic()
            if remaining > 20:
                p3b = min(remaining * 0.4, remaining - 10.0)
                a_opt, obj = full_pipeline_optimize(best_seq, time_budget=p3b, verbose=True)
                if obj < best_score:
                    best_score = obj
                    best_seq = a_opt.copy()
                    print(f"  After final pipeline: {best_score:.8f}")

            # Pair swap polish
            remaining = deadline - time.monotonic()
            if remaining > 15:
                ps_budget = min(remaining * 0.3, remaining - 10.0)
                a_opt = pair_swap_search(best_seq, time_budget=ps_budget, rng_seed=99999)
                obj = compute_obj(a_opt)
                if obj < best_score:
                    best_score = obj
                    best_seq = a_opt.copy()
                    print(f"  After pair swap: {best_score:.8f}")

    # === Phase 4: Try different n if time allows ===
    if best_seq is not None:
        remaining = deadline - time.monotonic()
        if remaining > 50:
            print(f"\n=== Phase 4: Different n ({remaining:.0f}s) ===")
            other_ns = [200, 280, 350, 500, 600]
            for n_val in other_ns:
                if time.monotonic() > deadline - 30:
                    break
                if len(best_seq) == n_val:
                    continue
                remaining = deadline - time.monotonic()
                if remaining < 30:
                    break
                alloc = min(35.0, remaining * 0.25)
                a_init = resample_sequence(best_seq, n_val)
                a_opt, obj = full_pipeline_optimize(a_init, time_budget=alloc, verbose=False)
                if obj < best_score:
                    best_score = obj
                    best_seq = a_opt.copy()
                    print(f"  ** UPDATE: n={n_val}, score={best_score:.8f} **")

    # Final validation
    if best_seq is None:
        best_seq = np.ones(50)
        best_score = compute_obj(best_seq)

    # Ensure valid output
    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    val = evaluate_sequence(result)
    if val < best_score:
        best_score = val

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(best_seq)}, score={best_score:.8f}, time={elapsed:.1f}s ===")

    return result
