import numpy as np
import time
from scipy.optimize import linprog

def compute_conv(a):
    n = len(a)
    size = max(2*n-1, 1)
    while size < 2*n-1:
        size *= 2
    fa = np.fft.rfft(a, size)
    conv = np.fft.irfft(fa * fa, size)
    return conv[:2*n-1]

def compute_obj(a):
    n = len(a)
    conv = compute_conv(a)
    max_b = np.max(conv)
    sum_a = np.sum(a)
    if sum_a < 0.01:

    n = len(a)

    S = np.sum(a)
    if S < 1e-10:
        return a
    a = a / S  # normalize to sum = 1
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    
    step = 0.1
    stall = 0
    patience = 50
    
    while time.monotonic() - start < time_budget:
        # Anneal temperature
        elapsed = time.monotonic() - start
        progress = min(elapsed / time_budget, 1.0)
        T = T_start * (T_end / T_start) ** progress
        
        grad, L_T = softmax_grad(a, T)
        
        # Gradient step with normalization
        grad_norm = np.linalg.norm(grad)
        if grad_norm < 1e-15:
            break
        
        # Step size: we want the relative change to be small
        a_new = a - step * grad / grad_norm * np.linalg.norm(a)
        a_new = np.maximum(a_new, 1e-15)
        S_new = np.sum(a_new)
        if S_new > 1e-10:
            a_new = a_new / S_new
        
        obj_new = compute_obj(a_new)
        if obj_new < best_obj:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new
            stall = 0
            step = min(step * 1.05, 1.0)
        else:
            stall += 1
            if stall > patience:
                step *= 0.5
                if step < 1e-4:
                    break

                a = a * np.exp(rng.normal(0, 0.05, n))
                a = np.maximum(a, 1e-10)
                a = a / np.sum(a)
                stall = 0
                patience = max(patience // 2, 20)
    
    return best_a

import numpy as np
import time
from scipy.optimize import linprog

def compute_conv(a):
    """Compute full linear convolution of a with itself using FFT."""
    n = len(a)
    size = 1
    while size < 2*n-1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    conv = np.fft.irfft(fa * fa, size)
    return conv[:2*n-1]

def compute_obj(a):
    """Compute the objective value."""
    n = len(a)
    conv = compute_conv(a)
    max_b = float(np.max(conv))
    sum_a = float(np.sum(a))
    if sum_a < 0.01:
        return np.inf
    return 2.0 * n * max_b / (sum_a * sum_a)

def softmax_gradient(a, T):
    """Compute gradient of the soft-max smoothed objective."""
    n = len(a)
    conv = compute_conv(a)
    S = float(np.sum(a))
    
    # Soft-max weights
    max_conv = float(np.max(conv))
    p = np.exp((conv - max_conv) / T)
    p_sum = float(np.sum(p))
    L_T = max_conv + T * np.log(p_sum)
    p /= p_sum
    
    # Gradient computation using correlate and convolve
    # G[i] = sum_k p[k] * (a[k-i] + a[i-k])
    G1 = np.correlate(p, a, 'full')[n-1:n-1+n]
    G2 = np.convolve(p, a, 'full')[:n]
    G = G1 + G2
    
    # Full gradient of 2n * L_T / S^2
    grad = 2.0 * n * (G - 2.0 * L_T / S) / (S * S)
    return grad, L_T
