```python
def project_simplex(v, s=1.0):
    """Project v onto {x : x >= 0, sum(x) = s}."""
    n = len(v)
    # Find the threshold
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.where(u * np.arange(1, n+1) > (cssv - s))[0][-1]
    theta = (cssv[rho] - s) / (rho + 1)
    return np.maximum(v - theta, 0)
```

Uses combination of: FFT-based soft-gradient descent with geometric tau annealing,

import numpy as np
from scipy.fft import rfft, irfft
from scipy.optimize import linprog
import time


def compute_conv(a):
    """Compute linear convolution a*a using FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([0.0])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    conv = irfft(fa * fa, size)
    return conv[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    conv = compute_conv(a)
    max_b = float(np.max(conv))
    sum_a = float(np.sum(a))
    if sum_a < 0.01:
        return np.inf
    return 2.0 * n * max_b / (sum_a * sum_a)


def fft_cross_convolve(a, b):
    """Linear convolution of a and b using FFT."""
    n, m = len(a), len(b)
    size = 1
    while size < n + m - 1:
        size <<= 1
    fa = rfft(a, size)
    fb = rfft(b, size)
    return irfft(fa * fb, size)[:n + m - 1]


def softmax(z):
    """Numerically stable softmax."""
    z = z - np.max(z)
    e = np.exp(z)
    s = np.sum(e)
    if s < 1e-300:
        return np.ones(len(z)) / len(z)
    return e / s


def soft_max_obj_and_grad_z(z, tau):
    """
    Compute soft-max of conv(softmax(z), softmax(z)) and gradient w.r.t. z.
    """
    n = len(z)
    a = softmax(z)
    conv = compute_conv(a)
    cmax = float(np.max(conv))
    shifted = np.clip((conv - cmax) / tau, -500.0, 500.0)
    exp_conv = np.exp(shifted)
    Z = float(np.sum(exp_conv))
    soft_max_val = cmax + tau * np.log(Z)
    w = exp_conv / Z

    # grad_a[i] = 2 * sum_k w[k] * a[k-i] (with bounds)
    # = 2 * cross_corr(a, w)[i] where cross_corr[i] = sum_j a[j]*w[j+i]
    w_rev = w[::-1]
    conv_fw = fft_cross_convolve(a, w_rev)
    # cross_corr[i] = conv_fw[2n-2-i]
    grad_a = 2.0 * conv_fw[n - 1:2 * n - 1][::-1]

    # grad_z for softmax parametrization
    mean_grad = float(np.dot(a, grad_a))
    grad_z = a * (grad_a - mean_grad)

    return soft_max_val, grad_z


def project_simplex(v, s=1.0):
    """Project v onto {x : x >= 0, sum(x) = s}."""
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho_candidates = np.where(u * np.arange(1, n + 1) > (cssv - s))[0]
    if len(rho_candidates) == 0:
        return np.full(n, s / n)
    rho = rho_candidates[-1]
    theta = (cssv[rho] - s) / (rho + 1)
    return np.maximum(v - theta, 0)


def soft_gradient_step(a, tau, step_size):
    """One soft-gradient step with simplex projection."""
    n = len(a)
    conv = compute_conv(a)
    cmax = float(np.max(conv))
    shifted = np.clip(tau * (conv - cmax), -300.0, 300.0)
    exp_vals = np.exp(shifted)
    w = exp_vals / np.sum(exp_vals)

    # Gradient of soft-max w.r.t. a
    w_rev = w[::-1]
    conv_fw = fft_cross_convolve(a, w_rev)
    grad_a = 2.0 * conv_fw[n - 1:2 * n - 1][::-1]

    # Projected gradient step
    a_new = a - step_size * grad_a
    a_new = project_simplex(a_new, 1.0)
    return a_new


def build_toeplitz(a):
    """Build Toeplitz matrix where (A@g)[k] = (conv(a,g))[k]."""
    n = len(a)
    A = np.zeros((2 * n - 1, n), dtype=np.float64)
    for j in range(n):
        k_start = j
        k_end = min(j + n, 2 * n - 1)
        A[k_start:k_end, j] = a[:k_end - k_start]
    return A


def solve_lp_g(a, M, time_limit=20):
    """Solve LP: max sum(g) s.t. (a*g)[k] <= M for all k, g >= 0."""
    n = len(a)
    A = build_toeplitz(a)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': time_limit})
        if res.status == 0:
            g = np.maximum(res.x, 0)
            s = float(np.sum(g))
            if s > 1e-10:
                return g / s, s
    except Exception:
        pass
    return None, None


def solve_lp_feasible_M(a, time_limit=20):
    """Binary search for minimum M such that LP max sum(g) s.t. a*g<=M, g>=0 is feasible with sum(g)>0."""
    n = len(a)
    conv = compute_conv(a)
    M_max = float(np.max(conv))
    M_min = M_max * 0.3

    # First check M_max is feasible
    g, s = solve_lp_g(a, M_max, time_limit=time_limit)
    if g is None:
        return M_max, None, None

    # Binary search
    lo, hi = M_min, M_max
    best_g, best_M = g, M_max
    for _ in range(8):
        if time.monotonic() > time_limit:
            break
        mid = (lo + hi) / 2
        g_try, s_try = solve_lp_g(a, mid, time_limit=5)
        if g_try is not None and s_try > 1e-10:
            best_g = g_try
            best_M = mid
            hi = mid
        else:
            lo = mid
    return best_M, best_g, None

def coordinate_descent_polish(a, max_passes=200, time_budget=30.0, verbose=False):
    """Exact coordinate descent for minimizing 2n*max(conv)/(sum)^2."""
    n = len(a)
    a = a.copy().astype(np.float64)
    s = np.sum(a)
    if s < 1e-10:
        a = np.ones(n) / n
    else:
        a = a / s

    start = time.monotonic()
    best_obj = compute_obj(a)
    best_a = a.copy()
    no_improve = 0

    for p in range(max_passes):
        if time.monotonic() - start > time_budget:
            break

        conv = compute_conv(a)
        k_star = int(np.argmax(conv))
        C_k = float(conv[k_star])
        S = float(np.sum(a))

        improved_this_pass = False

        for i in range(n):
            if time.monotonic() - start > time_budget:
                break

            ai = a[i]
            j = k_star - i

            if k_star == 2 * i:
                # Diagonal case
                D = S - ai
                if D < 1e-15:
                    continue
                E = C_k - ai * ai
                x = E / D
                if x < 0:
                    x = 0.0
            elif 0 <= j < n and a[j] > 1e-15:
                # Off-diagonal
                aj = a[j]
                D = S - ai
                B = C_k - 2.0 * aj * ai
                x = D - B / aj
                if x < 0:
                    x = 0.0
            else:
                # a[i] doesn't affect conv[k*], skip
                continue

            if abs(x - ai) < 1e-14:
                continue

            a[i] = x
            improved_this_pass = True

        # Renormalize
        s_new = np.sum(a)
        if s_new < 1e-10:
            a = best_a.copy()
            break
        a = a / s_new

        obj = compute_obj(a)
        if obj < best_obj - 1e-12:
            best_obj = obj
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1
            if no_improve > 30:
                break

        if verbose and p % 50 == 0:
            print(f"  CD pass {p}: obj={obj:.8f}")

    return best_a

    start = time.monotonic()

    for tau in tau_list:
        if time.monotonic() - start > time_budget:
            break

        def func_and_grad(z_val):
            return soft_max_obj_and_grad_z(z_val, tau)

        try:
            result = minimize(func_and_grad, z, jac=True, method='L-BFGS-B',
                              options={'maxiter': max_iter, 'ftol': 1e-14, 'gtol': 1e-8})
            z_new = result.x
        except Exception:
            continue

        a_new = softmax(z_new)
        a_new = np.maximum(a_new, 0)
        s_new = np.sum(a_new)
        if s_new < 1e-10:
            continue
        a_new = a_new / s_new
        obj = compute_obj(a_new)

        if obj < best_obj:
            best_obj = obj
            best_a = a_new.copy()
            z = z_new

    return best_a, best_obj


def random_init(n, seed):
    """Generate diverse random initializations."""
    rng = np.random.default_rng(seed)
    k = np.arange(n)
    choice = rng.integers(0, 10)

    if choice == 0:
        f = rng.uniform(0.5, 4.0)
        phase = rng.uniform(0, 2 * np.pi)
        a = 1.0 + 0.8 * np.cos(2 * np.pi * f * k / n + phase)
    elif choice == 1:
        f1 = rng.uniform(0.3, 2.0)
        f2 = rng.uniform(2.0, 6.0)
        a = 1.0 + 0.4 * np.cos(2 * np.pi * f1 * k / n) + 0.3 * np.cos(2 * np.pi * f2 * k / n)
    elif choice == 2:
        center = n * rng.uniform(0.3, 0.7)
        alpha = rng.uniform(0.5, 3.0)
        a = 1.0 + 0.5 * np.exp(-alpha * np.abs(k - center) / (n * 0.3))
    elif choice == 3:
        a = rng.random(n)
        win = max(3, n // 20)
        kernel = np.ones(win) / win
        a = np.convolve(a, kernel, mode='same')
    elif choice == 4:
        a = np.ones(n)
        for _ in range(rng.integers(1, 5)):
            f = rng.uniform(0.5, 5.0)
            amp = rng.uniform(0.1, 0.4)
            a += amp * np.cos(2 * np.pi * f * k / n + rng.uniform(0, 2 * np.pi))
    elif choice == 5:
        f = rng.uniform(0.3, 2.0)
        a = 1.0 + 0.5 * np.sin(2 * np.pi * f * k / n)
    elif choice == 6:
        nblocks = rng.integers(2, 7)
        a = np.ones(n)
        for b in range(nblocks):
            start = b * n // nblocks
            end = (b + 1) * n // nblocks
            a[start:end] = rng.uniform(0.3, 1.5)
    elif choice == 7:
        f0 = rng.uniform(0.5, 2.0)
        f1 = rng.uniform(2.0, 6.0)
        a = 1.0 + 0.5 * np.cos(2 * np.pi * (f0 * k / n + (f1 - f0) * k**2 / (2 * n**2)))
    elif choice == 8:
        center = n * rng.uniform(0.2, 0.8)
        width = n * rng.uniform(0.1, 0.4)
        a = 1.0 + 0.6 * np.exp(-0.5 * ((k - center) / width) ** 2)
    else:
        a = rng.random(n) + 0.1

    a = np.maximum(a, 0.01)
    s = np.sum(a)
    return a / s


def resample_sequence(a, new_n):
    """Resample sequence to new length via interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return a_new


def try_parent():
    """Try to load parent sequence."""
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64)
        parent = np.clip(parent, 0, 1000.0)
        if parent.size > 0 and np.isfinite(parent).all():
            obj = compute_obj(parent)
            print(f"Parent (n={parent.size}): obj={obj:.6f}")
            return parent.copy(), obj
    except Exception:
        pass
    return None, np.inf


def polish_sequence(a, time_budget, verbose=False):
    """Intensive polishing: alternating LP, soft-gradient, coordinate descent."""
    n = len(a)
    a = a.copy().astype(np.float64)
    start = time.monotonic()

    best_a = a.copy()
    best_obj = compute_obj(a)
    round_num = 0

    while time.monotonic() - start < time_budget - 3:
        round_num += 1
        remaining = time_budget - (time.monotonic() - start)
        if remaining < 5:
            break

        # LP step
        if n <= 700 and remaining > 25:
            a_new = alternating_lp_adaptive(best_a, max_iters=30, time_budget=min(remaining * 0.3, 40.0))
            obj = compute_obj(a_new)
            if obj < best_obj:
                best_obj = obj
                best_a = a_new.copy()
                if verbose:
                    print(f"    Polish r{round_num} LP: {best_obj:.8f}")

        # L-BFGS-B
        remaining = time_budget - (time.monotonic() - start)
        if remaining > 15:
            tau_final = 0.2 / n
            tau_small = max(0.05 / n, tau_final * 0.5)
            a_new, obj_new = optimize_lfbs(best_a, [tau_small, tau_final * 0.3],
                                           max_iter=150, time_budget=min(remaining * 0.2, 15.0))
            if obj_new < best_obj:
                best_obj = obj_new
                best_a = a_new.copy()
                if verbose:
                    print(f"    Polish r{round_num} L-BFGS: {best_obj:.8f}")

        # Soft-gradient
        remaining = time_budget - (time.monotonic() - start)
        if remaining > 10:
            tau_list = [0.1 / n, 0.05 / n, 0.02 / n]
            a_new = optimize_soft_gradient(best_a, tau_list, time_budget=min(remaining * 0.2, 12.0))
            obj = compute_obj(a_new)
            if obj < best_obj:
                best_obj = obj
                best_a = a_new.copy()
                if verbose:
                    print(f"    Polish r{round_num} SG: {best_obj:.8f}")

        # Coordinate descent
        remaining = time_budget - (time.monotonic() - start)
        if remaining > 8:
            a_new = coordinate_descent_polish(best_a, max_passes=100,
                                              time_budget=min(remaining * 0.2, 10.0))
            obj = compute_obj(a_new)
            if obj < best_obj:
                best_obj = obj
                best_a = a_new.copy()
                if verbose:
                    print(f"    Polish r{round_num} CD: {best_obj:.8f}")

    return best_a, best_obj


def propose_candidate(seed=42, budget_s=1500):
    """
    Main search function. Uses multi-resolution initialization, LP alternating
    with adaptive M, soft-gradient descent, L-BFGS-B, and coordinate descent
    polishing to minimize 2*n*max(conv(a,a))/(sum(a))^2.
    """
    start_time = time.monotonic()
    deadline = start_time + float(budget_s) - 30.0

    best_seq = None
    best_score = np.inf

    # Load parent
    parent, parent_obj = try_parent()
    if parent is not None and parent_obj < best_score:
        best_score = parent_obj
        best_seq = parent.copy()
        n_target = len(parent)
    else:
        n_target = 600

    print(f"Initial best: {best_score:.8f}, target n={n_target}")

    # Phase 1: Multi-resolution initialization and optimization
    # Start small, grow to target size
    n_stages = []
    if n_target <= 200:
        n_stages = [n_target]
    elif n_target <= 400:
        n_stages = [100, 200, n_target]
    else:
        n_stages = [100, 200, 400, n_target]

    # Add nearby sizes for exploration
    extra_ns = [n_target, max(50, n_target - 50), min(800, n_target + 50), min(800, n_target + 100)]
    all_ns = sorted(set(n_stages + extra_ns))

    print(f"N values to explore: {all_ns}")

    candidates = []  # list of (a, obj, n)

    # Optimize at each stage
    for n in all_ns:
        if time.monotonic() > deadline - 200:
            break
        remaining = deadline - time.monotonic()
        if remaining < 15:
            break

        # Time allocation
        if n <= 200:
            alloc = min(30.0, remaining * 0.08)
        elif n <= 400:
            alloc = min(45.0, remaining * 0.1)
        else:
            alloc = min(60.0, remaining * 0.12)
        alloc = max(15.0, alloc)

        print(f"\nN={n}: {remaining:.0f}s remaining, alloc={alloc:.0f}s")

        # Initialize: from parent if available, or random
        if best_seq is not None and abs(len(best_seq) - n) // n < 0.5:
            a = resample_sequence(best_seq, n)
        else:
            a = random_init(n, seed + n * 104729)

        # Quick optimization
        # Step 1: Soft-gradient
        sg_time = min(alloc * 0.25, 15.0)
        tau_list = [2.0/n, 1.0/n, 0.5/n, 0.2/n]
        a = optimize_soft_gradient(a, tau_list, time_budget=sg_time)

        # Step 2: L-BFGS-B
        remaining_init = deadline - time.monotonic()
        if remaining_init > 20:
            lfbs_time = min(alloc * 0.35, 25.0)
            tau_list2 = [1.0/n, 0.5/n, 0.2/n, 0.1/n, 0.05/n]
            a, obj = optimize_lfbs(a, tau_list2, max_iter=200, time_budget=lfbs_time)
        else:
            obj = compute_obj(a)

        # Step 3: LP
        remaining_init = deadline - time.monotonic()
        if remaining_init > 20 and n <= 700:
            lp_time = min(alloc * 0.3, 30.0)
            a = alternating_lp_adaptive(a, max_iters=30, time_budget=lp_time)
            obj = compute_obj(a)

        # Step 4: Coordinate descent
        remaining_init = deadline - time.monotonic()
        if remaining_init > 10:
            a = coordinate_descent_polish(a, max_passes=200, time_budget=min(remaining_init * 0.2, 10.0))
            obj = compute_obj(a)

        print(f"  N={n}: obj={obj:.8f}")

        if obj < best_score:
            best_score = obj
            best_seq = a.copy()
            print(f"  ** UPDATE: n={n}, obj={best_score:.8f} **")

        candidates.append((a.copy(), obj, n))

        # Try a second initialization for larger n
        if n == n_target and n <= 800:
            remaining_init = deadline - time.monotonic()
            if remaining_init > 40:
                a2 = random_init(n, seed + n * 209459 + 7)
                tau_list3 = [1.0/n, 0.5/n, 0.2/n]
                a2 = optimize_soft_gradient(a2, tau_list3, time_budget=10.0)
                a2, obj2 = optimize_lfbs(a2, [0.2/n, 0.1/n, 0.05/n], max_iter=150, time_budget=15.0)
                if n <= 700:
                    a2 = alternating_lp_adaptive(a2, max_iters=20, time_budget=15.0)
                obj2 = compute_obj(a2)
                a2 = coordinate_descent_polish(a2, max_passes=100, time_budget=8.0)
                obj2 = compute_obj(a2)
                print(f"  N={n} init2: obj={obj2:.8f}")
                if obj2 < best_score:
                    best_score = obj2
                    best_seq = a2.copy()
                    print(f"  ** UPDATE (init2): obj={best_score:.8f} **")
                candidates.append((a2.copy(), obj2, n))

    # Phase 2: Intensive polishing of top candidates
    print(f"\n=== POLISHING PHASE ===")
    print(f"Best before polish: {best_score:.8f}")

    candidates.sort(key=lambda x: x[1])
    n_polish = min(4, len(candidates))

    for ci in range(n_polish):
        if time.monotonic() > deadline - 80:
            break
        remaining = deadline - time.monotonic()
        if remaining < 20:
            break

        a_cand = candidates[ci][0]
        n_cand = candidates[ci][2]
        print(f"\n  Polishing candidate {ci} (n={n_cand}, obj={candidates[ci][1]:.8f}), {remaining:.0f}s left")

        polish_budget = min(remaining * 0.35, 120.0)
        a_pol, obj_pol = polish_sequence(a_cand, time_budget=polish_budget, verbose=True)

        if obj_pol < best_score:
            best_score = obj_pol
            best_seq = a_pol.copy()
            print(f"  ** UPDATE after polish: obj={best_score:.8f} **")

        # Try perturbation
        remaining = deadline - time.monotonic()
        if remaining > 30:
            rng = np.random.default_rng(ci * 555 + 999)
            for ptry in range(4):
                if time.monotonic() > deadline - 40:
                    break
                scale = rng.uniform(0.02, 0.08)
                a_pert = a_pol * np.exp(rng.normal(0, scale, n_cand))
                a_pert = np.maximum(a_pert, 0)
                s = np.sum(a_pert)
                if s < 1e-10:
                    continue
                a_pert = a_pert / s

                # Quick optimization of perturbed
                tau_pert = [0.5/n_cand, 0.2/n_cand, 0.1/n_cand]
                a_pert = optimize_soft_gradient(a_pert, tau_pert, time_budget=8.0)
                if n_cand <= 700:
                    a_pert = alternating_lp_adaptive(a_pert, max_iters=20, time_budget=15.0)
                a_pert = coordinate_descent_polish(a_pert, max_passes=100, time_budget=8.0)
                obj_pert = compute_obj(a_pert)

                if obj_pert < best_score:
                    best_score = obj_pert
                    best_seq = a_pert.copy()
                    print(f"  ** UPDATE (perturb {ptry}): obj={best_score:.8f} **")

    # Phase 3: Final intensive polish on global best
    if best_seq is not None and time.monotonic() < deadline - 20:
        remaining = deadline - time.monotonic()
        n_best = len(best_seq)
        print(f"\n=== FINAL POLISH: n={n_best}, {remaining:.0f}s remaining ===")

        a_cur = best_seq.copy()

        for final_round in range(6):
            if time.monotonic() > deadline - 10:
                break
            remaining = deadline - time.monotonic()
            if remaining < 15:
                break

            # LP
            if n_best <= 700 and remaining > 25:
                a_new = alternating_lp_adaptive(a_cur, max_iters=30,
                                                time_budget=min(remaining * 0.25, 30.0))
                obj_new = compute_obj(a_new)
                if obj_new < best_score:
                    best_score = obj_new
                    best_seq = a_new.copy()
                    a_cur = a_new.copy()
                    print(f"  Final LP {final_round}: {best_score:.8f}")

            # L-BFGS-B with very small tau
            remaining = deadline - time.monotonic()
            if remaining > 15:
                tau_vsmall = [0.1/n_best, 0.05/n_best, 0.02/n_best]
                a_new, obj_new = optimize_lfbs(a_cur, tau_vsmall,
                                               max_iter=300, time_budget=min(remaining * 0.2, 15.0))
                if obj_new < best_score:
                    best_score = obj_new
                    best_seq = a_new.copy()
                    a_cur = a_new.copy()
                    print(f"  Final L-BFGS {final_round}: {best_score:.8f}")

            # Soft-gradient with small tau
            remaining = deadline - time.monotonic()
            if remaining > 12:
                tau_sg = [0.05/n_best, 0.02/n_best, 0.01/n_best]
                a_new = optimize_soft_gradient(a_cur, tau_sg, time_budget=min(remaining * 0.2, 12.0))
                obj_new = compute_obj(a_new)
                if obj_new < best_score:
                    best_score = obj_new
                    best_seq = a_new.copy()
                    a_cur = a_new.copy()
                    print(f"  Final SG {final_round}: {best_score:.8f}")

            # Coordinate descent (most precise for final polish)
            remaining = deadline - time.monotonic()
            if remaining > 10:
                a_new = coordinate_descent_polish(a_cur, max_passes=500,
                                                  time_budget=min(remaining * 0.25, 20.0))
                obj_new = compute_obj(a_new)
                if obj_new < best_score:
                    best_score = obj_new
                    best_seq = a_new.copy()
                    a_cur = a_new.copy()
                    print(f"  Final CD {final_round}: {best_score:.8f}")

    # Final validation
    if best_seq is None:
        best_seq = np.ones(200)
        best_score = compute_obj(best_seq)

    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    val = evaluate_sequence(result)
    if val < np.inf:
        best_score = val
    else:
        result = [1.0] * 100
        best_score = 2.0

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_score:.8f}, time={elapsed:.1f}s ===")

    return result
