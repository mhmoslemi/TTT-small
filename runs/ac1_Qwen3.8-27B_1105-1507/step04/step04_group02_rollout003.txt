    conv = compute_conv(a)
    M = float(np.max(conv))
    return 2.0 * n * M / (S * S)


def build_toeplitz_sparse(a):
    """Build sparse Toeplitz matrix T where T @ g = conv(a, g)."""
    n = len(a)
    rows = []
    cols = []
    vals = []
    for j in range(n):
        start = j
        end = min(j + n, 2 * n - 1)
        length = end - start
        rows.extend(range(start, end))
        cols.extend([j] * length)
        vals.extend(list(a[:length]))
    A = coo_matrix((vals, (rows, cols)), shape=(2 * n - 1, n)).tocsr()
    return A


def soft_subgradient(a, T=0.02):
    """Compute softmax-weighted subgradient of max(conv(a,a)).
    
    Returns: (gradient_vector, max_conv, conv_values)
    The gradient uses soft attention over all convolution peaks weighted by exp(conv/T).
    """
    n = len(a)
    conv = compute_conv(a)
    M = float(np.max(conv))
    
    # Compute softmax weights with numerical stability
    shifted = conv - M
    w = np.exp(shifted / T)
    w_sum = np.sum(w)
    if w_sum < 1e-30:
        w = np.zeros(len(conv))
        w[np.argmax(conv)] = 1.0
    else:
        w /= w_sum
    
    # Only use significant weights for efficiency
    threshold = np.max(w) * 1e-4
    active = np.where(w > threshold)[0]
    
    if len(active) == 0:
        active = np.array([np.argmax(conv)])
        w_active = np.array([1.0])
    else:
        w_active = w[active]
    
    # Compute weighted gradient
    grad = np.zeros(n)
    for idx, k in enumerate(active):
        wk = w_active[idx]
        # ∂conv[k]/∂a[i] = 2*a[k-i] for 0 <= k-i < n
        i_min = max(0, k - n + 1)
        i_max = min(n, k + 1)
        j_min = k - i_max + 1
        j_max = k - i_min
        grad[i_min:i_max] += 2.0 * wk * a[j_min:j_max]
    
    return grad, M, conv


def mirror_descent_step(a, grad, eta):
    """Exponential weights (mirror descent) step with projection to simplex-like constraint."""
    n = len(a)
    grad_centered = grad - np.mean(grad)
    a_new = a * np.exp(-eta * grad_centered)
    a_new = np.maximum(a_new, 0.0)
    S = np.sum(a_new)
    if S < 1e-15:
        return a.copy()
    return a_new / S


def linear_step(a, grad, eta):
    """Linear gradient step with projection."""
    n = len(a)
    grad_centered = grad - np.mean(grad)
    a_new = a - eta * grad_centered
    a_new = np.maximum(a_new, 0.0)
    S = np.sum(a_new)
    if S < 1e-15:
        return a.copy()
    return a_new / S


def solve_lp_maxsum(a, M, time_limit=5):
    """Solve LP: max sum(g) s.t. conv(a,g) <= M, g >= 0."""
    n = len(a)
    A = build_toeplitz_sparse(a)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': time_limit})
        if res.status == 0:
            return np.maximum(res.x, 0)
        elif res.status == 3:  # Time limit
            # Return best found so far if available
            if res.x is not None:
                return np.maximum(res.x, 0)
    except Exception:
        pass
    return None


def lp_alternate(a, max_iters=30, time_budget=30.0, m_spreads=None):
    """LP alternating method with multiple M values and line search.
    
    m_spreads: list of multipliers to try for M (e.g., [0.98, 1.0, 1.02])
    """
    n = len(a)
    a = a.copy().astype(np.float64)
    S = np.sum(a)
    if S < 1e-10:
        return a
    a = a / S
    
    if m_spreads is None:
        m_spreads = [0.97, 0.985, 1.0, 1.015, 1.03]
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    stall = 0
    
    for it in range(max_iters):
        if time.monotonic() - start > time_budget:
            break
        
        conv = compute_conv(a)
        M = float(np.max(conv))
        
        # Try different M values
        g_best = None
        for ms in m_spreads:
            M_try = M * ms
            g = solve_lp_maxsum(a, M_try, time_limit=4)
            if g is not None and np.sum(g) > 0:
                if g_best is None or np.sum(g) > np.sum(g_best):
                    g_best = g
        
        if g_best is None or np.sum(g_best) < 1e-10:
            break
        
        g_best = g_best / np.sum(g_best)
        
        # Line search: try multiple step sizes
        best_t = 0.0
        best_obj_ls = best_obj
        for t in [0.03, 0.08, 0.15, 0.3, 0.5, 0.7, 0.9]:
            a_try = (1.0 - t) * a + t * g_best
            a_try = np.maximum(a_try, 0)
            S_try = np.sum(a_try)
            if S_try < 1e-10:
                continue
            a_try = a_try / S_try
            obj_try = compute_obj(a_try)
            if obj_try < best_obj_ls:
                best_obj_ls = obj_try
                best_t = t
        
        if best_t > 0.0:
            a = (1.0 - best_t) * a + best_t * g_best
            a = np.maximum(a, 0)
            a = a / np.sum(a)
            best_obj = best_obj_ls
            best_a = a.copy()
            stall = 0
        else:
            stall += 1
            if stall > 10:
                break
    
    return best_a


def soft_sg_phase(a, time_budget, T_start=0.05, T_end=0.005, topk_fallback=15):
    """Soft subgradient descent with adaptive temperature annealing.
    
    Uses mirror descent (exponential weights) as the primary update,
    with periodic linear steps for faster local convergence.
    """
    n = len(a)
    a = a.copy().astype(np.float64)
    S = np.sum(a)
    if S < 1e-10:
        return a
    a = a / S
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    eta = 0.1
    stall = 0
    it = 0
    
    while time.monotonic() - start < time_budget:
        it += 1
        remaining_frac = max(0, 1.0 - (time.monotonic() - start) / time_budget)
        T = T_start * (T_end / T_start) ** (1.0 - remaining_frac)
        T = max(T, T_end)
        
        grad, M, conv = soft_subgradient(a, T)
        
        # Use mirror descent for odd iterations, linear for even
        if it % 3 == 0:
            a_new = linear_step(a, grad, eta)
        else:
            a_new = mirror_descent_step(a, grad, eta)
        
        obj_new = compute_obj(a_new)
        if obj_new < best_obj - 1e-14:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new
            stall = 0
            eta = min(eta * 1.02, 0.3)
        else:
            stall += 1
            eta *= 0.95
            if stall > 50:
                # Adaptive temperature boost
                T = T * 0.5
                eta = max(eta * 0.5, 0.01)
                if eta < 0.005:
                    break
                stall = 0
    
    return best_a


def fourier_perturb(a, rng, scale=0.1):
    """Add structured noise in the frequency domain."""
    n = len(a)
    size = 1
    while size < n:
        size <<= 1
    fa = np.fft.rfft(a, size)
    
    # Add complex Gaussian noise in frequency domain
    noise = rng.normal(0, scale, fa.shape).astype(np.float64)
    # Preserve the DC component (sum)
    noise[0] = 0.0
    fa_noisy = fa + noise
    
    a_new = np.fft.irfft(fa_noisy, size)[:n]
    a_new = np.maximum(a_new, 0)
    S = np.sum(a_new)
    if S < 1e-10:
        return a.copy()
    return a_new / S


def time_domain_perturb(a, rng, scale=0.05):
    """Multiplicative random perturbation in time domain."""
    n = len(a)
    a_new = a * np.exp(rng.normal(0, scale, n))
    a_new = np.maximum(a_new, 0)
    S = np.sum(a_new)
    if S < 1e-10:
        return a.copy()
    return a_new / S


def cosine_init(n, n_modes=10, seed=0):
    """Generate a smooth cosine-series initialization."""
    rng = np.random.default_rng(seed)
    k = np.arange(n)
    a = np.ones(n)
    for m in range(1, n_modes + 1):
        c = rng.uniform(-0.5, 0.5) / m
        phase = rng.uniform(0, 2 * np.pi)
        a += c * np.cos(2 * np.pi * m * k / n + phase)
    a = np.maximum(a, 1e-10)
    S = np.sum(a)
    return a / S


def raised_cosine_init(n, alpha=1.5):
    """Raised cosine shape."""
    k = np.arange(n)
    x = 2.0 * k / (n - 1) - 1.0
    a = 0.5 * (1.0 + np.cos(np.pi * x)) ** alpha
    a = np.maximum(a, 1e-10)
    S = np.sum(a)
    return a / S


def spline_init(n, seed=0):
    """Smooth random spline-like initialization."""
    rng = np.random.default_rng(seed)
    # Generate few random control points and interpolate
    n_ctrl = rng.integers(5, 15)
    ctrl_x = np.linspace(0, n - 1, n_ctrl)
    ctrl_y = rng.uniform(0.5, 2.0, n_ctrl)
    # Ensure smoothness
    from scipy.interpolate import CubicSpline
    x = np.arange(n)
    cs = CubicSpline(ctrl_x, ctrl_y)
    a = np.maximum(cs(x), 1e-10)
    S = np.sum(a)
    return a / S


def resize_sequence(a, new_n):
    """Resize a sequence using cubic interpolation."""
    from scipy.interpolate import CubicSpline
    old_n = len(a)
    x_old = np.linspace(0, old_n - 1, old_n)
    cs = CubicSpline(x_old, a)
    x_new = np.linspace(0, new_n - 1, new_n)
    a_new = np.maximum(cs(x_new), 0.0)
    S = np.sum(a_new)
    if S < 1e-10:
        return np.ones(new_n) / new_n
    return a_new / S


def combined_refine(a, time_budget, use_lp=True, lp_n_max=600):
    """Combined refinement: soft SG + LP alternating + perturbations."""
    n = len(a)
    a = a.copy().astype(np.float64)
    S = np.sum(a)
    a = a / S
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    round_num = 0
    
    while time.monotonic() - start < time_budget - 5:
        round_num += 1
        remaining = time_budget - (time.monotonic() - start)
        if remaining < 5:
            break
        
        # Soft subgradient phase
        sg_time = min(remaining * 0.35, 20.0)
        if sg_time > 2:
            a_sg = soft_sg_phase(best_a, time_budget=sg_time,
                                 T_start=0.05, T_end=0.005)
            obj_sg = compute_obj(a_sg)
            if obj_sg < best_obj:
                best_obj = obj_sg
                best_a = a_sg.copy()
                if round_num <= 5:
                    print(f"      SG r{round_num}: {best_obj:.6f}")
        
        # LP phase
        remaining = time_budget - (time.monotonic() - start)
        if remaining > 10 and use_lp and n <= lp_n_max:
            lp_time = min(remaining * 0.3, 25.0)
            if lp_time > 5:
                a_lp = lp_alternate(best_a, max_iters=20, time_budget=lp_time)
                obj_lp = compute_obj(a_lp)
                if obj_lp < best_obj:
                    best_obj = obj_lp
                    best_a = a_lp.copy()
                    if round_num <= 5:
                        print(f"      LP r{round_num}: {best_obj:.6f}")
        
        # Perturbation escape
        remaining = time_budget - (time.monotonic() - start)
        if remaining > 8:
            rng = np.random.default_rng(round_num * 104729 + 7)
            # Try Fourier perturbation
            a_f = fourier_perturb(best_a, rng, scale=0.08)
            a_f = soft_sg_phase(a_f, time_budget=min(4.0, remaining * 0.2),
                                T_start=0.03, T_end=0.01)
            obj_f = compute_obj(a_f)
            if obj_f < best_obj:
                best_obj = obj_f
                best_a = a_f.copy()
                if round_num <= 5:
                    print(f"      F4 r{round_num}: {best_obj:.6f}")
            
            # Try time-domain perturbation
            remaining = time_budget - (time.monotonic() - start)
            if remaining > 5:
                a_t = time_domain_perturb(best_a, rng, scale=0.06)
                a_t = soft_sg_phase(a_t, time_budget=min(3.0, remaining * 0.15),
                                    T_start=0.03, T_end=0.01)
                obj_t = compute_obj(a_t)
                if obj_t < best_obj:
                    best_obj = obj_t
                    best_a = a_t.copy()
    
    return best_a, best_obj
