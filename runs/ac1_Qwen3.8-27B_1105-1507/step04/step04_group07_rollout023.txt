"""
Minimizes J(a) = 2*n*max(convolve(a,a))/(sum(a))^2 using:
1. LP best-response with adaptive tightening (main workhorse)
2. LSE (Log-Sum-Exp) gradient smoothing for polishing
3. Spectral domain refinement
4. Multi-length search (512 to 2048)
5. Multiplicative weight updates as alternative to additive mixing
"""

import numpy as np
from scipy.optimize import linprog
from scipy.sparse import csr_matrix
import time

# Cache for sparsity patterns
_spar_cache = {}

def get_sparsity(n):
    """Get the sparsity pattern for the convolution constraint matrix."""
    if n not in _spar_cache:
        rows = []
        cols = []
        for k in range(2 * n - 1):
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            rows.extend([k] * (je - js + 1))
            cols.extend(range(js, je + 1))
        _spar_cache[n] = (np.array(rows), np.array(cols), len(rows))
    return _spar_cache[n]

def fast_conv(a):
    """Fast linear convolution via FFT."""
    n = len(a)
    s = 1
    while s < 2 * n - 1:
        s <<= 1
    fa = np.fft.rfft(a, s)
    return np.fft.irfft(fa * fa, s)[:2 * n - 1]

def compute_J(a):
    """Compute the objective J(a) = 2*n*max(a*a)/(sum(a))^2."""
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))

def lp_solve(a, M, n, timeout=5):
    """Solve the LP: max sum(g) s.t. (a*g)[k] <= M, g >= 0."""
    rows, cols, nnz = get_sparsity(n)
    vals = np.empty(nnz)
    for i in range(nnz):
        k = rows[i]
        j = cols[i]
        vals[i] = a[k - j]
    A = csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                options={'presolve': True, 'time_limit': timeout})
    if r.success and r.x is not None:
        return np.maximum(r.x, 0.0)
    return None

def lp_optimize(a, deadline, tracker, max_iters=200, base_tighten=0.02):
    """Iterative LP best-response optimization."""
    a = np.maximum(a.astype(np.float64), 1e-10)
    a = a / a.sum()
    n = len(a)
    current_J = compute_J(a)
    tracker.update(a)
    
    no_improve = 0
    tighten = base_tighten
    
    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        
        b = fast_conv(a)
        M = float(b.max())
        
        # Try with tightening
        M_tight = M * (1.0 - tighten)
        g = lp_solve(a, M_tight, n, timeout=3)
        
        if g is None or g.sum() < 1e-10:
            # Try without tightening
            g = lp_solve(a, M, n, timeout=3)
            if g is None or g.sum() < 1e-10:
                # Try looser
                M_loose = M * 1.01
                g = lp_solve(a, M_loose, n, timeout=3)
                if g is None or g.sum() < 1e-10:
                    break
        
        g_sum = g.sum()
        if g_sum < 1e-10:
            break
        
        g_norm = g / g_sum
        
        # Try multiple mixing factors
        best_a = None
        best_J = current_J
        progress = it / max(1, max_iters)
        base_t = max(0.1, 0.6 * (1.0 - progress))
        
        candidates = [base_t, base_t * 2, base_t * 0.5, 0.3, 0.5, 0.7, 0.9]
        for t in candidates:
            if t > 1.0:
                continue
            a_trial = (1.0 - t) * a + t * g_norm
            a_trial = np.maximum(a_trial, 0.0)
            ss = a_trial.sum()
            if ss < 1e-10:
                continue
            a_trial = a_trial / ss
            J_trial = compute_J(a_trial)
            if J_trial < best_J - 1e-15:
                best_J = J_trial
                best_a = a_trial
        
        if best_a is None:
            # Try multiplicative update
            mix = 0.5
            a_mult = a * (g_norm + 1e-10) ** mix
            a_mult = np.maximum(a_mult, 1e-10)
            a_mult = a_mult / a_mult.sum()
            J_mult = compute_J(a_mult)
            if J_mult < best_J - 1e-15:
                best_J = J_mult
                best_a = a_mult
            else:
                no_improve += 1
                # Adjust tightening

            a = best_a

def lse_polish(a, deadline, tracker, max_iters=2000, tau0=20.0):
    """Polish using Log-Sum-Exp smoothed gradient descent."""
    a = np.maximum(a.astype(np.float64), 1e-8)
    a = a / a.sum()
    n = len(a)
    current_J = compute_J(a)
    tracker.update(a)
    
    tau = tau0
    lr = 2.0 / (n * tau)
    momentum = 0.9
    v = np.zeros(n)
    no_improve = 0
    
    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        
        b = fast_conv(a)
        b_max = b.max()
        exp_b = np.exp(tau * (b - b_max))
        p = exp_b / exp_b.sum()
        
        # Gradient of LSE: d/d a[j] sum_k p_k * (a*a)[k] = 2 * (p*a)[cross]
        # (p * a)[j] = sum_k p[k] * a[k-j] (cross-correlation)
        s = 1
        while s < 3 * n:
            s <<= 1
        fp = np.fft.rfft(p, s)
        fa = np.fft.rfft(a, s)
        # Cross-correlation: ifft( conj(F(p)) * F(a) )
        corr = np.fft.irfft(np.conj(fp) * fa, s)
        grad = 2.0 * corr[:n]
        
        v = momentum * v + (1.0 - momentum) * grad
        a_new = a - lr * v
        a_new = np.maximum(a_new, 1e-8)
        a_new = a_new / a_new.sum()
        
        J_new = compute_J(a_new)
        if J_new < current_J - 1e-16:
            a = a_new
            current_J = J_new
            no_improve = 0
            tau = min(500.0, tau * 1.02)
            lr = 2.0 / (n * tau)
        else:
            no_improve += 1
            lr *= 0.7
            v *= 0.5
            if no_improve > 50:
                break
        
        if it % 100 == 0:
            tracker.update(a)
    
    tracker.update(a)
    return a

def spectral_refine(a, deadline, tracker, max_iters=500):
    """Try to flatten the spectrum to reduce correlation peak."""
    a = np.maximum(a.astype(np.float64), 1e-8)
    a = a / a.sum()
    n = len(a)
    current_J = compute_J(a)
    tracker.update(a)
    
    lr = 0.1
    no_improve = 0
    
    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        
        # Compute DFT
        A = np.fft.rfft(a, n)
        # Power spectrum
        P = np.abs(A) ** 2
        P_max = P.max()
        
        # We want to reduce peaks in P. Gradient of max(P) w.r.t. a:
        # P[k] = |sum_j a[j] exp(-2pi i j k/n)|^2
        # dP[k]/da[j] = 2 * Re(exp(-2pi i j k/n) * conj(A[k]))
        # For the max frequency:
        k_max = np.argmax(P)
        phase = np.exp(-2j * np.pi * k_max * np.arange(n) / n)
        grad = 2.0 * np.real(phase * np.conj(A[k_max]))
        
        a_new = a - lr * grad
        a_new = np.maximum(a_new, 1e-8)
        a_new = a_new / a_new.sum()
        
        J_new = compute_J(a_new)
        if J_new < current_J - 1e-15:
            a = a_new
            current_J = J_new
            no_improve = 0
            lr = min(0.5, lr * 1.05)
        else:
            no_improve += 1
            lr *= 0.8
            if no_improve > 30:
                break
    
    tracker.update(a)
    return a

def generate_inits(n, rng, parent):
    """Generate diverse initializations."""
    inits = []
    t = np.linspace(0, 1, n, endpoint=False)
    
    if parent is not None and parent.size > 0:
        # Resample parent
        t_old = np.linspace(0, 1, parent.size)
        t_new = np.linspace(0, 1, n)
        p_resampled = np.interp(t_new, t_old, parent)
        inits.append(np.maximum(p_resampled, 0.001))
        # Reversed
        inits.append(np.maximum(p_resampled[::-1], 0.001))
    
    # Uniform
    inits.append(np.ones(n))
    
    # Raised cosine
    inits.append(0.5 * (1 + np.cos(2 * np.pi * t)))
    
    # Triangular
    inits.append(1.0 - np.abs(2 * t - 1.0))
    
    # Gaussian center
    inits.append(np.exp(-4 * (t - 0.5) ** 2))
    
    # Two bumps
    inits.append(np.exp(-10 * (t - 0.3) ** 2) + np.exp(-10 * (t - 0.7) ** 2) + 0.01)
    
    # Cosine window
    inits.append(np.cos(np.pi * (t - 0.5)) ** 2)
    
    # Sine raised
    inits.append(np.sin(np.pi * t) ** 0.5)
    
    # Power law
    inits.append((t + 0.05) ** 0.5)
    inits.append((1 - t + 0.05) ** 0.5)
    
    # Multi-freq
    inits.append(1.0 + 0.3 * np.cos(2 * np.pi * 3 * t) + 0.2 * np.cos(2 * np.pi * 7 * t))
    inits.append(1.0 + 0.2 * np.sin(2 * np.pi * 5 * t + 0.5) + 0.15 * np.cos(2 * np.pi * 11 * t))
    
    # Flat spectrum init
    half = n // 2 + 1
    angles = rng.uniform(0, 2 * np.pi, half)
    A = np.zeros(n, dtype=complex)
    A[:half] = np.exp(1j * angles)
    for k in range(1, n - n // 2):
        A[k] = np.conj(A[n - k])
    a_flat = np.fft.ifft(A).real
    inits.append(np.maximum(a_flat, 0.0))
    
    # Random smooth
    n_coeff = min(n // 4, 20)
    coeffs = rng.normal(0, 1, n_coeff)
    sig = np.zeros(n)
    for i in range(n_coeff):
        sig += coeffs[i] * np.cos(2 * np.pi * i * t + rng.uniform(0, 6.28))
    inits.append(np.maximum(1.0 + 0.5 * sig, 0.0))
    
    # Sparse smoothed
    n_spikes = max(10, n // 10)
    positions = rng.choice(n, size=n_spikes, replace=False)
    sparse = np.zeros(n)
    sparse[positions] = 1.0
    kernel_size = max(5, n // 20)
    kernel = np.exp(-0.5 * (np.arange(kernel_size) - kernel_size // 2) ** 2 / (kernel_size // 4) ** 2)
    kernel /= kernel.sum()
    inits.append(np.convolve(sparse, kernel, mode='same') + 0.01)
    
    # Normalize all
    for i, x in enumerate(inits):
        x = np.maximum(x, 1e-4).astype(np.float64)
        x = x / x.sum()
        inits[i] = x
    
    return inits

"""
Minimizes J(a) = 2*n*max(convolve(a,a))/(sum(a))^2 using:
1. LP best-response with adaptive tightening (primary method)
2. LSE (Log-Sum-Exp) gradient smoothing for polishing
3. Peak-shaving heuristic to identify and reduce contributing elements
4. Multi-length search (384 to 2048) with diverse initializations
5. Simulated restarts for escaping local optima
"""

import numpy as np
from scipy.optimize import linprog
from scipy.sparse import csr_matrix
import time

# ============ Caches and utilities ============

_spar_cache = {}

def get_sparsity(n):
    """Get the sparsity pattern for the convolution constraint matrix."""
    if n not in _spar_cache:
        rows = []
        cols = []
        for k in range(2 * n - 1):
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            rows.extend([k] * (je - js + 1))
            cols.extend(range(js, je + 1))
        _spar_cache[n] = (np.array(rows, dtype=np.int32),
                          np.array(cols, dtype=np.int32),
                          len(rows))
    return _spar_cache[n]

def fast_conv(a):
    """Fast linear convolution via FFT."""
    n = len(a)
    s = 1
    while s < 2 * n - 1:
        s <<= 1
    fa = np.fft.rfft(a, s)
    return np.fft.irfft(fa * fa, s)[:2 * n - 1]

def compute_J(a):
    """Compute J(a) = 2*n*max(a*a)/(sum(a))^2."""
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))

def lp_solve_conv(a, M, n, timeout=5.0):
    """Solve LP: max sum(g) s.t. (a*g)[k] <= M for all k, g >= 0."""
    rows, cols, nnz = get_sparsity(n)
    vals = np.empty(nnz)
    for i in range(nnz):
        k = rows[i]
        j = cols[i]
        vals[i] = a[k - j]
    A = csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                options={'presolve': True, 'time_limit': timeout})
    if r.success and r.x is not None:
        return np.maximum(r.x, 0.0)
    return None

# ============ Score tracking ============

class ScoreTracker:
    """Tracks the best solution found."""
    def __init__(self):
        self.best = None
        self.best_score = float('inf')
        self.t_start = time.monotonic()

    def update(self, a):
        a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
        if a.size == 0 or a.sum() < 0.01:
            return
        a_list = a.tolist()
        score = evaluate_sequence(a_list)
        if np.isfinite(score) and score < self.best_score:
            self.best = a.copy()
            self.best_score = score
            elapsed = time.monotonic() - self.t_start
            print(f"  ** NEW BEST: J={score:.8f} n={a.size} t={elapsed:.1f}s", flush=True)

    def report(self, label=""):
        if self.best is not None:
            print(f"  {label}: J={self.best_score:.8f} n={self.best.size}", flush=True)

# ============ Core optimization methods ============

def lp_optimize(a, deadline, tracker, max_iters=100, base_tighten=0.02, verbose=False):
    """Iterative LP best-response optimization with adaptive tightening."""
    a = np.maximum(a.astype(np.float64), 1e-10)
    a = a / a.sum()
    n = len(a)
    current_J = compute_J(a)
    tracker.update(a)

    no_improve = 0
    tighten = base_tighten

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        b = fast_conv(a)
        M = float(b.max())

        # Try with tightening
        M_tight = M * (1.0 - tighten)
        g = lp_solve_conv(a, M_tight, n, timeout=3.0)

        if g is None or g.sum() < 1e-10:
            g = lp_solve_conv(a, M, n, timeout=3.0)
            if g is None or g.sum() < 1e-10:
                M_loose = M * 1.005
                g = lp_solve_conv(a, M_loose, n, timeout=3.0)
                if g is None or g.sum() < 1e-10:
                    break

        g_sum = g.sum()
        if g_sum < 1e-10:
            break

        g_norm = g / g_sum

        # Try multiple mixing factors
        best_a = None
        best_J = current_J
        progress = it / max(1, max_iters)
        base_t = max(0.1, 0.6 * (1.0 - progress))

        candidates = sorted(set([base_t, base_t * 1.5, base_t * 0.5, 0.2, 0.4, 0.6, 0.8]))
        for t in candidates:
            if t <= 0.0 or t > 1.0:
                continue
            a_trial = (1.0 - t) * a + t * g_norm
            a_trial = np.maximum(a_trial, 1e-10)
            ss = a_trial.sum()
            if ss < 1e-10:
                continue
            a_trial = a_trial / ss
            J_trial = compute_J(a_trial)
            if J_trial < best_J - 1e-15:
                best_J = J_trial
                best_a = a_trial

        if best_a is not None:
            a = best_a
            current_J = best_J
            no_improve = 0
            # Gradually increase tightening
            tighten = min(0.12, base_tighten * (1.0 + 0.03 * it))
        else:
            no_improve += 1
            # Try multiplicative update
            a_mult = a * (g_norm + 1e-10) ** 0.5
            a_mult = np.maximum(a_mult, 1e-10)
            a_mult = a_mult / a_mult.sum()
            J_mult = compute_J(a_mult)
            if J_mult < current_J - 1e-15:
                a = a_mult
                current_J = J_mult
                no_improve = 0
            else:
                tighten = min(0.15, tighten * 1.2)
                if no_improve > 25:
                    break

        if verbose and it % 10 == 0:
            tracker.update(a)

    tracker.update(a)
    return a

def lse_polish(a, deadline, tracker, max_iters=3000, tau0=30.0, tau_max=800.0):
    """Polish using Log-Sum-Exp smoothed gradient descent with momentum."""
    a = np.maximum(a.astype(np.float64), 1e-8)
    a = a / a.sum()
    n = len(a)
    current_J = compute_J(a)
    tracker.update(a)

    tau = tau0
    lr = 2.0 / (n * tau)
    momentum = 0.88
    v = np.zeros(n)
    no_improve = 0

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        b = fast_conv(a)
        b_max = b.max()
        # Clip to avoid overflow
        exp_b = np.exp(np.minimum(tau * (b - b_max), 50.0))
        p = exp_b / exp_b.sum()

        # Gradient: d/d a[j] of sum_k p_k * (a*a)[k]
        # = 2 * sum_k p_k * a[k-j] = 2 * (p reversed cross a)[j]
        s = 1
        while s < 3 * n:
            s <<= 1
        fp = np.fft.rfft(p, s)
        fa = np.fft.rfft(a, s)
        corr = np.fft.irfft(np.conj(fp) * fa, s)
        grad = 2.0 * corr[:n]

        v = momentum * v + (1.0 - momentum) * grad
        # Adaptive step size
        grad_norm = np.linalg.norm(grad)
        if grad_norm > 1e-15:
            step = lr * np.linalg.norm(v) / grad_norm
            step = min(step, 0.5)
            v_scaled = v * (step / (np.linalg.norm(v) + 1e-30))
        else:
            v_scaled = v

        a_new = a - v_scaled
        a_new = np.maximum(a_new, 1e-8)
        a_new = a_new / a_new.sum()

        J_new = compute_J(a_new)
        if J_new < current_J - 1e-16:
            a = a_new
            current_J = J_new
            no_improve = 0
            tau = min(tau_max, tau * 1.03)
            lr = 2.0 / (n * tau)
        else:
            no_improve += 1
            lr *= 0.7
            v *= 0.5
            if no_improve > 80:
                break

        if it % 200 == 0:
            tracker.update(a)

    tracker.update(a)
    return a

def peak_shave(a, deadline, tracker, strength=0.3):
    """Reduce values at positions contributing most to peak correlation."""
    a = np.maximum(a.astype(np.float64), 1e-10)
    a = a / a.sum()
    n = len(a)
    current_J = compute_J(a)
    tracker.update(a)

    for _ in range(20):
        if time.monotonic() > deadline:
            break

        b = fast_conv(a)
        M = b.max()
        active = np.where(b >= M * 0.9)[0]

        contributions = np.zeros(n)
        for k in active:
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            for j in range(js, je + 1):
                contributions[j] += a[k - j]

        if contributions.max() < 1e-15:
            break

        # Reduce high-contribution elements
        a_new = a * (1.0 - strength * contributions / contributions.max())
        a_new = np.maximum(a_new, 1e-10)
        a_new = a_new / a_new.sum()

        J_new = compute_J(a_new)
        if J_new < current_J - 1e-14:
            a = a_new
            current_J = J_new
            strength = min(0.5, strength * 1.05)
        else:
            strength *= 0.8
            if strength < 0.01:
                break

    tracker.update(a)
    return a

def subgradient_step(a, deadline, tracker, max_iters=2000):
    """Subgradient descent on the exact max function."""
    a = np.maximum(a.astype(np.float64), 1e-10)
    a = a / a.sum()
    n = len(a)
    current_J = compute_J(a)
    tracker.update(a)
    lr = 0.3 / n
    rng = np.random.default_rng(123)

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        b = fast_conv(a)
        M = float(b.max())
        active = np.where(b >= M - 1e-8)[0]

        grad = np.zeros(n)
        for k in active:
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            for j in range(js, je + 1):
                grad[j] += a[k - j]
        grad /= len(active)

        # Direction: reduce the gradient (which points toward increasing peak)
        direction = grad - M * 2.0 * a
        a_new = a - lr * direction
        a_new = np.maximum(a_new, 1e-10)
        a_new = a_new / a_new.sum()

        J_new = compute_J(a_new)
        if J_new < current_J - 1e-15:
            a = a_new
            current_J = J_new
            lr = min(lr * 1.02, 1.0 / n)
        else:
            lr *= 0.75
            if lr < 1e-10 / n:
                break

        if it % 100 == 0:
            tracker.update(a)

    tracker.update(a)
    return a

# ============ Initialization ============

def generate_inits(n, rng, parent):
    """Generate diverse initializations for length n."""
    inits = []
    t = np.linspace(0, 1, n, endpoint=False)

    if parent is not None and parent.size > 0:
        t_old = np.linspace(0, 1, parent.size)
        t_new = np.linspace(0, 1, n)
        p_res = np.interp(t_new, t_old, parent)
        inits.append(np.maximum(p_res, 0.001))
        inits.append(np.maximum(p_res[::-1], 0.001))
        # Slightly perturbed parent
        inits.append(np.maximum(p_res + rng.normal(0, 0.02, n), 0.001))

    # Uniform
    inits.append(np.ones(n))

    # Raised cosine
    inits.append(0.5 * (1.0 + np.cos(2.0 * np.pi * t)))

    # Triangular
    inits.append(1.0 - np.abs(2.0 * t - 1.0))

    # Gaussian center
    inits.append(np.exp(-3.0 * (t - 0.5) ** 2 / (0.5 ** 2)))

    # Two bumps
    inits.append(np.exp(-15.0 * (t - 0.25) ** 2) + np.exp(-15.0 * (t - 0.75) ** 2) + 0.01)

    # Cosine window
    inits.append(np.cos(np.pi * (t - 0.5)) ** 2)

    # Sine power
    inits.append(np.sin(np.pi * t) ** 0.7)

    # Power laws
    inits.append((t + 0.05) ** 0.4)
    inits.append((1.0 - t + 0.05) ** 0.4)

    # Multi-frequency
    inits.append(1.0 + 0.25 * np.cos(2.0 * np.pi * 3.0 * t) + 0.15 * np.cos(2.0 * np.pi * 7.0 * t))
    inits.append(1.0 + 0.2 * np.sin(2.0 * np.pi * 5.0 * t + 0.5) + 0.1 * np.cos(2.0 * np.pi * 11.0 * t + 1.0))
    inits.append(1.0 + 0.15 * np.cos(4.0 * np.pi * t) - 0.1 * np.cos(8.0 * np.pi * t))

    # Flat spectrum (random phases)
    half = n // 2 + 1
    angles = rng.uniform(0, 2.0 * np.pi, half)
    A = np.zeros(n, dtype=complex)
    A[:half] = np.exp(1j * angles)
    for k in range(1, n - n // 2):
        A[k] = np.conj(A[n - k])
    a_flat = np.fft.ifft(A).real
    inits.append(np.maximum(a_flat, 0.0))

    # Another flat spectrum with amplitude variation
    amps = rng.uniform(0.5, 1.5, half)
    angles2 = rng.uniform(0, 2.0 * np.pi, half)
    A2 = np.zeros(n, dtype=complex)
    A2[:half] = amps * np.exp(1j * angles2)
    for k in range(1, n - n // 2):
        A2[k] = np.conj(A2[n - k])
    a_flat2 = np.fft.ifft(A2).real
    inits.append(np.maximum(a_flat2, 0.0))

    # Low-pass random
    n_coeff = min(n // 4, 15)
    coeffs = rng.normal(0, 1, n_coeff)
    sig = np.zeros(n)
    for i in range(n_coeff):
        coeff_decay = 1.0 / (1.0 + 0.3 * i)
        sig += coeff_decay * coeffs[i] * np.cos(2.0 * np.pi * i * t + rng.uniform(0, 6.28))
    inits.append(np.maximum(1.0 + 0.4 * sig, 0.0))

    # Sparse + smooth
    n_spikes = max(10, n // 8)
    positions = rng.choice(n, size=n_spikes, replace=False)
    sparse = np.zeros(n)
    sparse[positions] = 1.0
    kernel_size = max(5, n // 15)
    k_range = np.arange(kernel_size) - kernel_size // 2
    kernel = np.exp(-0.5 * (k_range / (kernel_size * 0.25)) ** 2)
    kernel /= kernel.sum()
    inits.append(np.convolve(sparse, kernel, mode='same') + 0.01)

    #normalize all
    final_inits = []
    for x in inits:
        x = np.maximum(x, 1e-4).astype(np.float64)
        x = x / x.sum()
        final_inits.append(x)

    return final_inits

# ============ Main search function ============

def propose_candidate(seed=42, budget_s=1500, **kwargs):
    """
    Search for the best non-negative sequence minimizing
    J(a) = 2*n*max(convolve(a,a))/(sum(a))^2.

    Strategy:
    - Phase 1 (55% time): Explore multiple lengths with LP + LSE
    - Phase 2 (35% time): Exploit best found with aggressive LP + polishing
    - Phase 3 (10% time): Final polish and fallback
    """
    rng = np.random.default_rng(seed)
    t0 = time.monotonic()
    total_budget = float(budget_s)
    deadline = t0 + total_budget - 25.0
    tracker = ScoreTracker()

    # Load parent sequence
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        if p.ndim == 1 and p.size > 0 and np.all(np.isfinite(p)):
            parent = np.maximum(p, 0.0)
            tracker.update(parent)
            print(f"Parent loaded: n={parent.size}, J={tracker.best_score:.8f}", flush=True)
    except (NameError, AttributeError):
        pass

    # ============ Phase 1: Multi-length exploration ============
    explore_deadline = t0 + 0.55 * (total_budget - 25.0)
    print(f"\n{'='*55}", flush=True)
    print(f"Phase 1: Multi-length exploration (deadline at t={explore_deadline-t0:.0f}s)", flush=True)
    print(f"{'='*55}", flush=True)

    lengths_to_try = [384, 512, 640, 768, 1024, 1280, 1536, 2048]

    # Prioritize lengths near parent size and larger
    if parent is not None:
        pn = parent.size
        lengths_to_try.sort(key=lambda x: abs(x - pn) + 0.1 * x)

    for n in lengths_to_try:
        if time.monotonic() > explore_deadline - 45:
            break

        phase_start = time.monotonic()
        print(f"\n  --- Exploring n={n} ---", flush=True)

        # Estimate LP time
        est_lp_time = 0.2 * (n / 512.0) ** 1.6
        remaining = explore_deadline - time.monotonic() - 30
        max_total_lps = max(5, int(remaining / max(est_lp_time, 0.05)))

        # Number of initializations to try
        n_inits = min(8, max(3, max_total_lps // 15))
        inits = generate_inits(n, rng, parent)
        n_inits = min(n_inits, len(inits))

        best_J_n = float('inf')
        best_a_n = None

        for idx in range(n_inits):
            if time.monotonic() > explore_deadline - 30:
                break

            a0 = inits[idx]
            iters_per_init = max(8, min(60, max_total_lps // n_inits))

            # LP optimization
            a = lp_optimize(a0, explore_deadline - 30, tracker,
                           max_iters=iters_per_init, base_tighten=0.02)

            # LSE polish
            if time.monotonic() < explore_deadline - 15:
                a = lse_polish(a, explore_deadline - 15, tracker,
                             max_iters=max(200, 3000 // (n // 128)), tau0=25.0)

            J = compute_J(a)
            if J < best_J_n:
                best_J_n = J
                best_a_n = a.copy()

        if best_a_n is not None:
            tracker.update(best_a_n)
        print(f"  n={n}: best J={best_J_n:.8f} ({time.monotonic()-phase_start:.1f}s)", flush=True)

    # ============ Phase 2: Exploit best ============
    if tracker.best is not None:
        exploit_deadline = t0 + 0.90 * (total_budget - 25.0)
        print(f"\n{'='*55}", flush=True)
        print(f"Phase 2: Exploit best (n={tracker.best.size})", flush=True)
        print(f"{'='*55}", flush=True)

        a = tracker.best.copy().astype(np.float64)
        n = a.size

        # Multiple LP passes with increasing tightening
        tighten_schedule = [0.01, 0.015, 0.02, 0.025, 0.03, 0.04, 0.05, 0.06]
        lps_per_tight = max(5, int(40 * (512.0 / n) ** 0.5))

        for tight in tighten_schedule:
            if time.monotonic() > exploit_deadline - 40:
                break
            a = lp_optimize(a, exploit_deadline - 40, tracker,
                          max_iters=lps_per_tight, base_tighten=tight)

        # Subgradient refinement
        if time.monotonic() < exploit_deadline - 30:
            a = subgradient_step(a, exploit_deadline - 30, tracker,
                               max_iters=max(200, int(2000 * 512.0 / n)))

        # Peak shaving
        if time.monotonic() < exploit_deadline - 20:
            a = peak_shave(a, exploit_deadline - 20, tracker, strength=0.25)
            # LP after peak shaving
            if time.monotonic() < exploit_deadline - 15:
                a = lp_optimize(a, exploit_deadline - 15, tracker,
                              max_iters=20, base_tighten=0.02)

        # LSE polish
        if time.monotonic() < exploit_deadline - 10:
            a = lse_polish(a, exploit_deadline - 10, tracker,
                         max_iters=max(500, int(5000 * 512.0 / n)), tau0=40.0, tau_max=600.0)

        tracker.report("Phase 2 complete")

    # ============ Phase 3: Final polish and restarts ============
    if tracker.best is not None and time.monotonic() < deadline - 15:
        print(f"\n{'='*55}", flush=True)
        print(f"Phase 3: Final polish", flush=True)
        print(f"{'='*55}", flush=True)

        a = tracker.best.copy().astype(np.float64)
        n = a.size

        # Try a few random restarts from current best
        n_restarts = 3
        for r in range(n_restarts):
            if time.monotonic() > deadline - 15:
                break
            a_pert = a + rng.normal(0, 0.03 / np.sqrt(n), n)
            a_pert = np.maximum(a_pert, 1e-8)
            a_pert = a_pert / a_pert.sum()

            a_trial = lp_optimize(a_pert, deadline - 15, tracker,
                                 max_iters=15, base_tighten=0.02)
            if time.monotonic() < deadline - 10:
                a_trial = lse_polish(a_trial, deadline - 10, tracker,
                                    max_iters=500, tau0=30.0)
            a = a_trial

        # Final LSE with high tau
        if time.monotonic() < deadline - 8:
            a = lse_polish(tracker.best.copy(), deadline - 8, tracker,
                         max_iters=2000, tau0=100.0, tau_max=1000.0)

        # Very final LP
        if time.monotonic() < deadline - 4:
            a = lp_optimize(tracker.best.copy(), deadline - 4, tracker,
                          max_iters=10, base_tighten=0.01)

        tracker.report("Phase 3 complete")

    # ============ Fallback ============
    if tracker.best is None:
        print("FALLBACK: No valid solution found, using uniform", flush=True)
        n_fb = 512
        tracker.best = np.ones(n_fb) / n_fb
        tracker.best_score = compute_J(tracker.best)
        tracker.update(tracker.best)

    # Final output
    final = np.clip(tracker.best, 0.0, 1000.0)
    # Ensure at least one positive element
    if final.max() < 1e-6:
        final = np.ones(final.size) / final.size

    final_score = evaluate_sequence(final.tolist())
    print(f"\n{'='*55}", flush=True)
    print(f"FINAL: J={final_score:.8f} n={final.size} time={time.monotonic()-t0:.1f}s", flush=True)
    print(f"{'='*55}", flush=True)

    return [float(x) for x in final]
