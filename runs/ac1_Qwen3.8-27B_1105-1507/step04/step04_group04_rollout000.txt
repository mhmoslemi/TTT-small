import time

def compute_conv(a):
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    conv = np.fft.irfft(fa * fa, size)
    return conv[:2 * n - 1]

def compute_obj(a):
    n = len(a)
    if n == 0:
        return np.inf
    conv = compute_conv(a)
    max_b = float(np.max(conv))
    sum_a = float(np.sum(a))
    if sum_a < 0.01:
        return np.inf
    return 2.0 * n * max_b / (sum_a * sum_a)

def build_toeplitz(a):
    n = len(a)
    A = np.zeros((2 * n - 1, n), dtype=np.float64)
    for j in range(n):
        k_start = j
        k_end = min(j + n, 2 * n - 1)
        A[k_start:k_end, j] = a[:k_end - k_start]
    return A

def solve_lp_maxsum(a, M, time_limit=10):
    n = len(a)
    A = build_toeplitz(a)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)

        if res.status == 0:

- Subgradient polishing with dual guidance
- Multi-scale local search
- Systematic n-value exploration around parent
"""

import numpy as np
from scipy.optimize import linprog
import time

try:
    import cvxpy as cp
    HAS_CVXPY = True
except ImportError:
    HAS_CVXPY = False


def compute_conv(a):
    """Compute full self-convolution using FFT."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    conv = np.fft.irfft(fa * fa, size)
    return conv[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    conv = compute_conv(a)
    max_b = float(np.max(conv))
    sum_a = float(np.sum(a))
    if sum_a < 0.01:
        return np.inf
    return 2.0 * n * max_b / (sum_a * sum_a)


def build_toeplitz(a):
    """Build matrix A where (A@g)[k] = (a*g)[k] for k=0..2n-2."""
    n = len(a)
    A = np.zeros((2 * n - 1, n), dtype=np.float64)
    for j in range(n):
        k_start = j
        k_end = min(j + n, 2 * n - 1)
        A[k_start:k_end, j] = a[:k_end - k_start]
    return A


def solve_lp_maxsum(a, M, time_limit=10):
    """Maximize sum(g) s.t. (a*g)[k] <= M for all k, g >= 0."""
    n = len(a)
    A = build_toeplitz(a)
    
    if HAS_CVXPY:
        try:
            g_var = cp.Variable(n, nonneg=True)
            constraints = [A @ g_var <= M]
            prob = cp.Problem(cp.Maximize(cp.sum(g_var)), constraints)
            for solver in [cp.MOSEK, cp.CLARABEL, cp.SCS, cp.PDLP]:
                try:
                    prob.solve(solver=solver, time_limit=int(time_limit))
                    break
                except Exception:
                    continue
            if g_var.value is not None:
                g_val = np.maximum(g_var.value, 0)
                return g_val, float(np.sum(g_val))
        except Exception:
            pass
    
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': time_limit})
        if res.status == 0:
            g = np.maximum(res.x, 0)
            return g, float(np.sum(g))
    except Exception:
        pass
    return None, None


def solve_lp_minpeak(a, time_limit=10):
    """Minimize M s.t. (a*g)[k] <= M, sum(g)=1, g>=0. Returns (g, M_opt)."""
    n = len(a)
    A = build_toeplitz(a)
    
    if HAS_CVXPY:
        try:
            g_var = cp.Variable(n, nonneg=True)
            M_var = cp.Variable()
            constraints = [A @ g_var <= M_var, cp.sum(g_var) == 1]
            prob = cp.Problem(cp.Minimize(M_var), constraints)
            for solver in [cp.MOSEK, cp.CLARABEL, cp.SCS, cp.PDLP]:
                try:
                    prob.solve(solver=solver, time_limit=int(time_limit))
                    break
                except Exception:
                    continue
            if g_var.value is not None and M_var.value is not None:
                g_val = np.maximum(g_var.value, 0)
                s = np.sum(g_val)
                if s > 1e-10:
                    g_val = g_val / s
                    return g_val, float(M_var.value)
        except Exception:
            pass
    
    c = np.zeros(n + 1)
    c[-1] = 1.0
    A_ub = np.zeros((2 * n - 1, n + 1))
    A_ub[:, :n] = A
    A_ub[:, n] = -1.0
    b_ub = np.zeros(2 * n - 1)
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])
    bounds = [(0.0, None)] * n + [(0.0, None)]
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': time_limit})
        if res.status == 0:
            g = np.maximum(res.x[:n], 0)
            s = np.sum(g)
            if s > 1e-10:
                g = g / s
                return g, float(res.x[n])
    except Exception:
        pass
    return None, None


def subgradient_step(a, tau=30.0, step=0.05):
    """Multiplicative subgradient step using convolution gradient."""
    n = len(a)
    conv = compute_conv(a)
    max_conv = np.max(conv)
    
    # Softmax weights over convolution lags (focus on peaks)
    shifted = tau * (conv - max_conv)
    shifted = np.clip(shifted, -400, 400)
    weights = np.exp(shifted)
    w_sum = np.sum(weights)
    if w_sum < 1e-300:
        return a
    weights /= w_sum
    
    # Gradient: d(conv[aargmax])/da[i] = weighted sum of a[k-i] over active lags
    # Use correlate: (weights * a)[i] = sum_k weights[k] * a[k] shifted
    c = np.correlate(weights, a, mode='full')
    grad = 2.0 * c[n - 1:n - 1 + n]
    
    # Center to preserve sum approximately
    grad_centered = grad - np.mean(grad)
    
    log_update = -step * grad_centered
    log_update = np.clip(log_update, -30, 30)
    a_new = a * np.exp(log_update)
    a_new = np.maximum(a_new, 1e-15)
    s = np.sum(a_new)
    if s > 1e-10:
        a_new = a_new / s
    return a_new


def alternating_lp_optimize(a, max_iters=100, time_budget=60.0, use_minpeak=False):
    """Alternating LP with adaptive line search."""
    n = len(a)
    a = a.copy().astype(np.float64)
    s = np.sum(a)
    if s < 1e-10:
        return a
    a = a / s
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    stall = 0
    stall_limit = 30

    for it in range(max_iters):
        if time.monotonic() - start > time_budget - 5:
            break

        conv = compute_conv(a)
        M = float(np.max(conv))

        if use_minpeak:
            g, M_opt = solve_lp_minpeak(a, time_limit=8)
            if g is None:
                break
        else:
            g, sum_g = solve_lp_maxsum(a, M, time_limit=8)
            if g is None or sum_g < 1e-10:
                break
            g = g / sum_g

        # Adaptive line search
        best_t = 0.0
        best_obj_ls = best_obj
        # Try a range of step sizes
        for t in [1.0, 0.7, 0.5, 0.3, 0.2, 0.1, 0.05]:
            a_new = (1.0 - t) * a + t * g
            a_new = np.maximum(a_new, 0)
            s_new = np.sum(a_new)
            if s_new < 1e-10:
                continue
            a_new = a_new / s_new
            obj_new = compute_obj(a_new)
            if obj_new < best_obj_ls - 1e-15:
                best_obj_ls = obj_new
                best_t = t

        if best_t > 0 and best_obj_ls < best_obj - 1e-14:
            a_new = (1.0 - best_t) * a + best_t * g
            a_new = np.maximum(a_new, 0)
            s_new = np.sum(a_new)
            if s_new > 1e-10:
                a_new = a_new / s_new
            best_obj = best_obj_ls
            best_a = a_new.copy()
            a = a_new
            stall = 0
            stall_limit = max(15, stall_limit - 1)
        else:
            stall += 1
            if stall > stall_limit:
                # Perturb to escape
                rng = np.random.default_rng(it * 1000 + 7)
                a = a * np.exp(rng.normal(0, 0.02, n))
                a = np.maximum(a, 1e-10)
                s = np.sum(a)
                a = a / s
                stall = 0
                stall_limit = max(10, stall_limit - 5)
            else:
                # Small nudge
                a = 0.95 * a + 0.05 * g
                a = np.maximum(a, 0)
                s = np.sum(a)
                if s > 1e-10:
                    a = a / s

    return best_a


def subgradient_polish(a, time_budget=20.0, tau=40.0, step=0.04):
    """Polish with subgradient method, adaptive parameters."""
    n = len(a)
    a = a.copy().astype(np.float64)
    s = np.sum(a)
    if s < 1e-10:
        return a
    a = a / s
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    stall = 0

    while time.monotonic() - start < time_budget - 2:
        a = subgradient_step(a, tau=tau, step=step)
        obj = compute_obj(a)
        if obj < best_obj - 1e-14:
            best_obj = obj
            best_a = a.copy()
            stall = 0
        else:
            stall += 1
            if stall > 50:
                step *= 0.8
                tau *= 1.15
                if step < 0.002:
                    break
                # Perturb
                rng = np.random.default_rng(int(time.monotonic() * 1000))
                a = a * np.exp(rng.normal(0, 0.03, n))
                a = np.maximum(a, 1e-10)
                a = a / np.sum(a)
                stall = 0
    return best_a


def fine_polish(a, time_budget=15.0):
    """Very fine polishing with tiny steps and high tau."""
    n = len(a)
    a = a.copy().astype(np.float64)
    s = np.sum(a)
    if s < 1e-10:
        return a
    a = a / s
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    step = 0.01
    tau = 80.0
    stall = 0

    while time.monotonic() - start < time_budget - 1:
        a = subgradient_step(a, tau=tau, step=step)
        obj = compute_obj(a)
        if obj < best_obj - 1e-15:
            best_obj = obj
            best_a = a.copy()
            stall = 0
        else:
            stall += 1
            if stall > 80:
                step *= 0.7
                if step < 0.001:
                    break
                stall = 0
    return best_a


def local_search(a, time_budget=60.0):
    """Multi-scale local search with re-optimization."""
    n = len(a)
    a = a.copy().astype(np.float64)
    s = np.sum(a)
    if s < 1e-10:
        return a
    a = a / s
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    attempt = 0

    while time.monotonic() - start < time_budget - 10:
        attempt += 1
        remaining = time_budget - (time.monotonic() - start)
        if remaining < 15:
            break

        rng = np.random.default_rng(attempt * 7919 + 31)
        
        # Multi-scale perturbation
        scale = 0.01 + 0.04 * (attempt % 5) / 5.0
        
        a_pert = best_a * np.exp(rng.normal(0, scale, n))
        a_pert = np.maximum(a_pert, 1e-10)
        a_pert = a_pert / np.sum(a_pert)

        # Quick optimization
        sub_time = min(12.0, remaining * 0.35)
        a_opt = alternating_lp_optimize(a_pert, max_iters=20, time_budget=sub_time)
        obj = compute_obj(a_opt)
        
        if obj < best_obj:
            best_obj = obj
            best_a = a_opt.copy()
            print(f"    LS attempt {attempt}: {best_obj:.8f}")

    return best_a


def full_optimize(a, time_budget=80.0):
    """Full optimization pipeline for a single sequence."""
    n = len(a)
    a = a.copy().astype(np.float64)
    s = np.sum(a)
    if s < 1e-10:
        a = np.ones(n)
        s = n
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    # Phase 1: MaxSum LP
    remaining = time_budget - (time.monotonic() - start)
    if remaining > 15:
        t1 = min(remaining * 0.4, 35.0)
        a_new = alternating_lp_optimize(best_a, max_iters=60, time_budget=t1, use_minpeak=False)
        obj = compute_obj(a_new)
        if obj < best_obj:
            best_obj = obj
            best_a = a_new.copy()
            print(f"      MaxSum LP: {best_obj:.8f}")

    # Phase 2: Subgradient polish
    remaining = time_budget - (time.monotonic() - start)
    if remaining > 8:
        t2 = min(remaining * 0.2, 15.0)
        a_new = subgradient_polish(best_a, time_budget=t2, tau=30.0, step=0.05)
        obj = compute_obj(a_new)
        if obj < best_obj:
            best_obj = obj
            best_a = a_new.copy()
            print(f"      Subgrad: {best_obj:.8f}")

    # Phase 3: MinPeak LP
    remaining = time_budget - (time.monotonic() - start)
    if remaining > 15:
        t3 = min(remaining * 0.35, 30.0)
        a_new = alternating_lp_optimize(best_a, max_iters=40, time_budget=t3, use_minpeak=True)
        obj = compute_obj(a_new)
        if obj < best_obj:
            best_obj = obj
            best_a = a_new.copy()
            print(f"      MinPeak LP: {best_obj:.8f}")

    # Phase 4: Subgradient again
    remaining = time_budget - (time.monotonic() - start)
    if remaining > 8:
        t4 = min(remaining * 0.2, 12.0)
        a_new = subgradient_polish(best_a, time_budget=t4, tau=50.0, step=0.03)
        obj = compute_obj(a_new)
        if obj < best_obj:
            best_obj = obj
            best_a = a_new.copy()
            print(f"      Subgrad2: {best_obj:.8f}")

    # Phase 5: Fine polish
    remaining = time_budget - (time.monotonic() - start)
    if remaining > 8:
        t5 = min(remaining * 0.4, 20.0)
        a_new = fine_polish(best_a, time_budget=t5)
        obj = compute_obj(a_new)
        if obj < best_obj:
            best_obj = obj
            best_a = a_new.copy()
            print(f"      Fine: {best_obj:.8f}")

    return best_a, best_obj


def random_init(n, seed):
    """Generate diverse initializations."""
    rng = np.random.default_rng(seed)
    k = np.arange(n)
    choice = rng.integers(0, 8)

    if choice == 0:
        f = rng.uniform(0.5, 3.0)
        phase = rng.uniform(0, 2 * np.pi)
        a = 1.0 + 0.7 * np.cos(2 * np.pi * f * k / n + phase)
    elif choice == 1:
        f1 = rng.uniform(0.3, 2.0)
        f2 = rng.uniform(2.0, 5.0)
        a = 1.0 + 0.4 * np.cos(2 * np.pi * f1 * k / n) + 0.3 * np.cos(2 * np.pi * f2 * k / n)
    elif choice == 2:
        center = n * rng.uniform(0.3, 0.7)
        width = n * rng.uniform(0.1, 0.4)
        a = 1.0 + 0.6 * np.exp(-0.5 * ((k - center) / width) ** 2)
    elif choice == 3:
        a = np.ones(n)
        for _ in range(rng.integers(2, 5)):
            f = rng.uniform(0.5, 6.0)
            amp = rng.uniform(0.1, 0.4)
            a += amp * np.cos(2 * np.pi * f * k / n + rng.uniform(0, 2 * np.pi))
    elif choice == 4:
        # Chirp
        f0 = rng.uniform(0.3, 1.5)
        f1 = rng.uniform(3.0, 8.0)
        t = k / n
        phase = 2 * np.pi * (f0 * t + (f1 - f0) * t**2 / 2)
        a = 1.0 + 0.5 * np.cos(phase)
    elif choice == 5:
        # Smooth random
        m = max(5, n // 8)
        a_coarse = rng.uniform(0.5, 2.0, m)
        a = np.interp(k, np.arange(m), a_coarse)
    elif choice == 6:
        # Two-tone
        f1 = rng.uniform(1.0, 3.0)
        f2 = rng.uniform(3.0, 7.0)
        a = 1.0 + 0.3 * np.sin(2 * np.pi * f1 * k / n) + 0.3 * np.sin(2 * np.pi * f2 * k / n + np.pi / 3)
    else:
        a = np.ones(n) * (1.0 + 0.5 * rng.uniform(0, 1) * np.sin(np.pi * k / (n - 1)))

    a = np.maximum(a, 0.01)
    s = np.sum(a)
    if s > 1e-10:
        a = a / s
    return a


def resample_sequence(a, new_n):
    """Resample sequence to new length."""
    n = len(a)
    if n == new_n:
        return a.copy()
    old_k = np.arange(n)
    new_k = np.linspace(0, n - 1, new_n)
    a_new = np.interp(new_k, old_k, a)
    a_new = np.maximum(a_new, 0.01)
    s = np.sum(a_new)
    if s > 1e-10:
        a_new = a_new / s
    return a_new


def try_parent():
    """Try to load parent sequence."""
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64)
        parent = np.clip(parent, 0, 1000.0)
        if parent.size > 0 and np.isfinite(parent).all():
            s = np.sum(parent)
            if s > 0.01:
                obj = compute_obj(parent)
                print(f"Parent (n={parent.size}): obj={obj:.8f}")
                return parent.copy(), obj
    except Exception:
        pass
    return None, np.inf


def propose_candidate(seed=42, budget_s=1500):
    """
    Main search function. Optimizes step function to minimize 2*n*max(conv)/(sum)^2.
    Uses alternating LP (maxsum & minpeak), subgradient polishing, and local search.
    Focuses on n values around the parent (330) with systematic exploration.
    """
    start_time = time.monotonic()
    deadline = start_time + float(budget_s) - 30.0

    best_seq = None
    best_score = np.inf

    # Load parent
    parent, parent_obj = try_parent()
    if parent is not None:
        best_score = parent_obj
        best_seq = parent.copy()
        n_parent = len(parent)
    else:
        n_parent = 300
        best_seq = random_init(n_parent, seed)
        best_score = compute_obj(best_seq)

    print(f"Best: {best_score:.8f} (n={len(best_seq)})")
    print(f"Budget: {budget_s}s, deadline in {deadline - start_time:.0f}s")

    # Phase 1: Intensive optimization of parent
    print("\n=== Phase 1: Optimize parent ===")
    remaining = deadline - time.monotonic()
    if remaining > 30:
        p1_budget = min(remaining * 0.35, 350.0)
        a_opt, obj = full_optimize(best_seq, time_budget=p1_budget)
        if obj < best_score:
            best_score = obj
            best_seq = a_opt.copy()
            print(f"  After full_optimize: {best_score:.8f}")

    # Phase 2: Local search on best
    remaining = deadline - time.monotonic()
    if remaining > 30:
        print(f"\n=== Phase 2: Local search ({remaining:.0f}s) ===")
        p2_budget = min(remaining * 0.25, 200.0)
        a_opt = local_search(best_seq, time_budget=p2_budget)
        obj = compute_obj(a_opt)
        if obj < best_score:
            best_score = obj
            best_seq = a_opt.copy()
            print(f"  After local search: {best_score:.8f}")

    # Phase 3: Try nearby n values
    print(f"\n=== Phase 3: Nearby n values ===")
    n_candidates = []
    for delta in [0, -10, 10, -20, 20, -30, 30, -50, 50, -70, 70, 90, -90]:
        nv = n_parent + delta
        if 150 <= nv <= 800:
            n_candidates.append(nv)
    # Add some fixed values
    for nv in [300, 350, 400, 250, 450]:
        if nv not in n_candidates and 150 <= nv <= 800:
            n_candidates.append(nv)
    n_candidates = sorted(set(n_candidates))
    # Remove if too many
    if len(n_candidates) > 10:
        # Keep closest to n_parent
        n_candidates = sorted(n_candidates, key=lambda x: abs(x - n_parent))[:10]
        n_candidates = sorted(n_candidates)

    remaining = deadline - time.monotonic()
    if remaining > 40:
        for nv in n_candidates:
            if time.monotonic() > deadline - 40:
                break
            remaining = deadline - time.monotonic()
            if remaining < 30:
                break
            if nv == len(best_seq):
                continue
            
            alloc = min(remaining / max(len(n_candidates), 1) * 1.5, 100.0, remaining * 0.25)
            alloc = max(alloc, 20.0)
            
            print(f"\n  n={nv} (remaining={remaining:.0f}s, alloc={alloc:.0f}s)")
            
            # Resample parent to this n
            a_init = resample_sequence(parent, nv) if parent is not None else random_init(nv, seed + nv)
            
            a_opt, obj = full_optimize(a_init, time_budget=alloc)
            
            if obj < best_score:
                best_score = obj
                best_seq = a_opt.copy()
                print(f"  ** NEW BEST: n={nv}, obj={best_score:.8f} **")

    # Phase 4: Final intensive optimization
    remaining = deadline - time.monotonic()
    if remaining > 30:
        print(f"\n=== Phase 4: Final optimization ({remaining:.0f}s) ===")
        p4_budget = min(remaining * 0.45, remaining - 15.0)
        a_opt, obj = full_optimize(best_seq, time_budget=p4_budget)
        if obj < best_score:
            best_score = obj
            best_seq = a_opt.copy()
            print(f"  After final full_opt: {best_score:.8f}")

    # Phase 5: Final local search
    remaining = deadline - time.monotonic()
    if remaining > 20:
        print(f"\n=== Phase 5: Final local search ({remaining:.0f}s) ===")
        p5_budget = min(remaining * 0.5, remaining - 10.0)
        a_opt = local_search(best_seq, time_budget=p5_budget)
        obj = compute_obj(a_opt)
        if obj < best_score:
            best_score = obj
            best_seq = a_opt.copy()
            print(f"  After final LS: {best_score:.8f}")

    # Phase 6: One more round of LP if time allows
    remaining = deadline - time.monotonic()
    if remaining > 15:
        print(f"\n=== Phase 6: Final LP polish ({remaining:.0f}s) ===")
        p6_budget = min(remaining * 0.6, remaining - 5.0)
        a_opt = alternating_lp_optimize(best_seq, max_iters=30, time_budget=p6_budget, use_minpeak=False)
        obj = compute_obj(a_opt)
        if obj < best_score:
            best_score = obj
            best_seq = a_opt.copy()
            print(f"  After final LP: {best_score:.8f}")
        
        remaining = deadline - time.monotonic()
        if remaining > 10:
            a_opt = fine_polish(best_seq, time_budget=min(remaining * 0.5, 15.0))
            obj = compute_obj(a_opt)
            if obj < best_score:
                best_score = obj
                best_seq = a_opt.copy()
                print(f"  After final fine: {best_score:.8f}")

    # Final validation
    if best_seq is None:
        best_seq = np.ones(50)
        best_score = compute_obj(best_seq)

    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    
    # Validate with actual evaluation
    try:
        val = evaluate_sequence(result)
        if val < best_score:
            best_score = val
    except Exception:
        pass

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(best_seq)}, score={best_score:.8f}, time={elapsed:.1f}s ===")

    return result
