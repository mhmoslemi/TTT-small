    """

        if np.isnan(x) or np.isinf(x):

    sequence = [float(x) for x in sequence]

def fast_conv(a):
    """Full self-convolution via FFT."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]


def compute_J(a):
    """Compute J(a) = 2*n*max(conv(a,a))/(sum(a))^2."""
    a = np.maximum(a, 0)
    s = a.sum()

    return float(2 * len(a) * b.max() / (s * s))


def build_A_sparse(n, a):
    """Build sparse matrix A where A[k,j] = a[k-j] for valid j."""
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        j_min = max(0, k - n + 1)
        j_max = min(n - 1, k)
        for j in range(j_min, j_max + 1):
            rows.append(k)
            cols.append(j)
            vals.append(a[k - j])
    return csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))


def lp_best_response(a, M):

def cmaes_eigen(C, z, dim):
    """Apply C^(1/2) @ z using eigendecomposition."""
    eigvals, eigvecs = np.linalg.eigh(C)
    eigvals = np.maximum(eigvals, 1e-12)
    root = np.sqrt(eigvals)
    return (eigvecs * root) @ (eigvecs.T @ z)


def fourier_to_seq(params, n, n_harm):
    """Convert Fourier parameters to a non-negative sequence (normalized)."""
    t = np.arange(n, dtype=np.float64) / n
    a = np.full(n, params[0], dtype=np.float64)
    for k in range(1, n_harm + 1):
        a += params[2 * k - 1] * np.cos(2.0 * np.pi * k * t)
        a += params[2 * k] * np.sin(2.0 * np.pi * k * t)
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-8:
        a = np.ones(n) / n
    else:
        a = a / s
    return a


def seq_to_fourier(a, n_harm):

        params[2 * k] = 2.0 * np.mean(a_norm * np.sin(2.0 * np.pi * k * t))
    return params

def subgradient_polish(a, max_iters, deadline):
    """Subgradient descent with momentum."""
    a = np.maximum(a.astype(np.float64), 1e-10)
    a = a / a.sum()
    best_M = fast_conv(a).max()
    best_a = a.copy()
    momentum = np.zeros_like(a)
    beta = 0.85

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        b = fast_conv(a)
        M = b.max()
        tol = 1e-11 * M
        active = np.where(b >= M - tol)[0]

        grad = np.zeros_like(a)
        n = len(a)
        for k in active:
            lo = max(0, k - n + 1)
            hi = min(n - 1, k)
            for i in range(lo, hi + 1):
                grad[i] += 2.0 * a[k - i]

        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-15:
            break

        t = 1.5 / (gnorm * np.sqrt(it + 1.0))
        momentum = beta * momentum + (1 - beta) * grad / gnorm
        a_new = a - t * momentum
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s < 1e-10:
            break
        a_new = a_new / s

        new_M = fast_conv(a_new).max()
        if new_M < best_M:
            best_M = new_M
            best_a = a_new.copy()
        a = a_new

    return best_a


def resample_seq(src, dest_n):
    """Resample 1D sequence to new length."""
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)


def structured_perturb(a, rng, perturb_type):
    """Apply a structured perturbation to the sequence."""
    n = len(a)
    t = np.arange(n, dtype=np.float64) / n

    if perturb_type == 'shift':
        s = rng.integers(1, max(2, n // 10))
        a_p = np.roll(a, s)
    elif perturb_type == 'cosine':
        freq = rng.uniform(1, 8)
        amp = rng.uniform(0.02, 0.1)
        a_p = a * (1 + amp * np.cos(2 * np.pi * freq * t + rng.uniform(0, 2 * np.pi)))
    elif perturb_type == 'smooth':
        k = rng.integers(3, max(4, n // 10))
        kernel = np.ones(k) / k
        a_p = np.convolve(a, kernel, mode='same')
    elif perturb_type == 'taper':
        alpha = rng.uniform(0.5, 0.9)
        window = np.ones(n)
        edge = max(1, int(n * (1 - alpha) / 2))
        for i in range(edge):
            frac = i / edge
            window[i] = frac
            window[n - 1 - i] = frac
        a_p = a * window
    else:
        a_p = a + rng.normal(0, 0.03, n)

    a_p = np.maximum(a_p, 0.001)
    s = a_p.sum()
    if s > 1e-8:
        a_p = a_p / s
    return a_p


def generate_inits(n, rng, parent=None):
    """Generate diverse initial sequences."""
    inits = []
    t = np.arange(n, dtype=np.float64) / n

    if parent is not None:
        p = resample_seq(parent, n)
        inits.append(p)
        inits.append(np.maximum(p + rng.normal(0, 0.02, n), 0.01))
        inits.append(np.maximum(p + rng.normal(0, 0.05, n), 0.01))

    inits.append(np.ones(n))
    inits.append(0.5 * (1 + np.cos(2 * np.pi * t)))
    inits.append(np.maximum(0.42 - 0.5 * np.cos(2 * np.pi * t) + 0.08 * np.cos(4 * np.pi * t), 0.01))
    inits.append(np.exp(-4 * (t - 0.5) ** 2) + 0.1)
    inits.append(np.exp(-8 * (t - 0.3) ** 2) + np.exp(-8 * (t - 0.7) ** 2) + 0.05)
    for f in [1, 2, 3, 5, 7, 11]:
        inits.append(np.maximum(1.0 + 0.3 * np.cos(2 * np.pi * f * t), 0.01))
    inits.append(np.maximum(1.0 - 0.4 * np.abs(2 * t - 1), 0.05))

    # Smooth random
    k = max(3, n // 16)
    kernel = np.ones(k) / k
    smooth = np.convolve(rng.uniform(0, 1, n), kernel, mode='same')
    inits.append(np.maximum(smooth + 0.2, 0.01))

    result = []
    for a in inits:
        a = np.maximum(a, 0.001)
        s = a.sum()
        if s > 1e-6:
            result.append(a / s)
    return result


class ScoreTracker:
    def __init__(self):
        self.best = None
        self.best_score = float('inf')
        self.t_start = time.monotonic()

    def update(self, a):
        a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
        if a.size == 0 or a.sum() < 0.01:
            return
        score = float(evaluate_sequence(a.tolist()))
        if np.isfinite(score) and score < self.best_score:
            self.best = a.copy()
            self.best_score = score
            elapsed = time.monotonic() - self.t_start
            print(f"  [NEW BEST] J={score:.8f} n={a.size} t={elapsed:.0f}s", flush=True)

    def report(self, label=""):
        if self.best is not None:
            print(f"  {label} best: J={self.best_score:.8f} n={self.best.size}", flush=True)


def lbb_objective(x):
    """Objective for L-BFGS-B: max of self-convolution (scale invariant)."""
    ax = np.maximum(x, 0.0)
    s = ax.sum()
    if s < 1e-8:
        return 1e10
    ax = ax / s
    b = fast_conv(ax)
    return float(b.max())
