def coordinate_update(a, i, b, S, n):
    """Find optimal value for coordinate i using golden section search with incremental updates."""
    ai = a[i]

    # Precompute unchanged max
    # b[k] for k not in [i, i+n-1] is unchanged
    mask = np.zeros(2 * n - 1, dtype=bool)
    lo_k = max(0, i)
    hi_k = min(2 * n - 2, i + n - 1)
    mask[lo_k:hi_k + 1] = True
    unchanged_max = b[~mask].max() if np.any(~mask) else 0.0

    def eval_J(x):
        new_S = S - ai + x
        if new_S < 1e-10:
            return 1e10
        max_c = unchanged_max
        for k in range(lo_k, hi_k + 1):
            ki = k - i
            if 0 <= ki < n:
                if k == 2 * i:
                    ck = b[k] - ai * ai + x * x
                else:
                    ck = b[k] + 2.0 * (x - ai) * a[ki]
            else:
                ck = b[k]
            if ck > max_c:
                max_c = ck
        return 2.0 * n * max_c / (new_S * new_S)

    # Golden section search
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    lo = 0.0
    hi = max(2.0 * ai + 0.01, S / n * 2.0, 0.05)

    c_pt = hi - gr * (hi - lo)
    d_pt = lo + gr * (hi - lo)
    fc = eval_J(c_pt)
    fd = eval_J(d_pt)

    for _ in range(30):
        if fc < fd:
            hi = d_pt
            d_pt = c_pt
            fd = fc
            c_pt = hi - gr * (hi - lo)
            fc = eval_J(c_pt)
        else:
            lo = c_pt
            c_pt = d_pt
            fc = fd
            d_pt = lo + gr * (hi - lo)
            fd = eval_J(d_pt)

    x_opt = (lo + hi) / 2.0
    return max(0.0, x_opt)


def coordinate_descent_pass(a, rng):
    """One full pass of coordinate descent with incremental updates."""
    a = a.astype(np.float64).copy()
    S = a.sum()
    if S < 1e-10:

            S = S - old_val + new_val

    a = a / a.sum()
    return a


def coordinate_descent(a, n_passes, deadline, tracker):

def resample_seq(src, dest_n):
    """Resample 1D sequence to new length via linear interpolation."""
    src = np.asarray(src, dtype=np.float64)

def fast_conv(a):
    """Full self-convolution via FFT."""
    n = len(a)
    if n == 0:
        return np.array([])
    s = 1
    while s < 2 * n - 1:
        s <<= 1
    fa = np.fft.rfft(a, s)
    return np.fft.irfft(fa * fa, s)[:2 * n - 1]


def compute_J(a):
    """Compute J(a) = 2*n*max(conv(a,a))/(sum(a))^2."""
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))


def project_simplex(v):
    """Project v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v.copy()
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho_candidates = np.nonzero(u * np.arange(1, n + 1) > (cssv - 1))[0]
    if len(rho_candidates) == 0:
        return np.ones(n) / n
    rho = rho_candidates[-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0)
