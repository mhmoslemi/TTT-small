import numpy as np
from scipy.optimize import linprog, minimize
from scipy.sparse import lil_matrix, csr_matrix
import time


def fast_conv(a):
    """Full self-convolution via FFT."""
    n = len(a)
    s = 1
    while s < 2 * n - 1:
        s <<= 1
    fa = np.fft.rfft(a, s)
    return np.fft.irfft(fa * fa, s)[:2 * n - 1]


def cross_corr_fft(p, a):
    """Compute cross-correlation (p cross a)[i] = sum_m p[i+m]*a[m] for i=0..n-1."""
    n = len(a)
    N = 1
    while N < len(p) + n - 1:
        N <<= 1
    P = np.fft.rfft(p, N)
    A = np.fft.rfft(a, N)
    result = np.fft.irfft(P * np.conj(A), N)
    return result[:n]


def compute_J(a):
    """Compute J(a) = 2*n*max(conv(a,a))/(sum(a))^2."""
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))

    """
    Compute smooth approximation of J and its gradient.
    J_smooth = 2n * (1/beta) * log(sum exp(beta * c_k)) / S^2

    J = 2.0 * n * F / (S * S)
    grad_J = 2.0 * n * (grad_F - 2.0 * F / S) / (S * S)
    
    return J, grad_J


def build_A_sparse(a):
    """Build sparse constraint matrix for LP."""
    n = len(a)
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        jl = max(0, k - n + 1)
        jh = min(n - 1, k)
        if jl <= jh:
            cnt = jh - jl + 1
            rows.extend([k] * cnt)
            cols.extend(range(jl, jh + 1))
            seg = a[k - jh:k - jl + 1][::-1]
            vals.extend(seg.tolist())
    A = lil_matrix((2 * n - 1, n))
    for idx in range(len(rows)):
        A[rows[idx], cols[idx]] = vals[idx]
    return A.tocsr()

    tracker.update(a)
    
    tighten = tighten_base
    stall = 0
    
    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

                tighten *= 0.5
                if tighten < 1e-7:
                    tighten = tighten_base
                stall += 1
                if stall > 30:
                    break
                continue
        
        g = np.maximum(g, 0.0)
        g_sum = g.sum()
        g /= g_sum
        
        # Try multiple step sizes
        progress = min(1.0, it / max(1, max_iters))
        base_t = max(0.02, 0.4 * (1.0 - 0.5 * progress))
        candidates_t = [base_t, base_t * 0.5, base_t * 1.5, 0.1, 0.3]
        
        best_a = None
        best_J = current_J

                break

            continue

        current_J = best_J
        stall = 0
        tighten = min(0.05, tighten * 1.02)

def softmax_anneal(a, deadline, tracker, betas=None):
    """Optimize using softmax smooth approximation with L-BFGS-B and beta annealing."""
    if betas is None:
        betas = [5, 10, 20, 50, 100, 200, 400]
    
    a = np.maximum(a.astype(np.float64), 0.001)
    a = a / a.sum()
    n = len(a)
    
    for beta in betas:
        if time.monotonic() > deadline:
            break
        
        def obj_grad(x):
            x = np.maximum(x, 0.0)
            J, g = softmax_J_grad(x, beta)
            g = np.clip(g, -1e8, 1e8)
            return J, g
        
        # Run L-BFGS-B
        maxiter = max(50, min(300, int(60 * n / 400)))
        result = minimize(obj_grad, a, method='L-BFGS-B',
                         bounds=[(0.0, 10.0)] * n,
                         options={'maxiter': maxiter, 'ftol': 1e-14, 
                                  'gtol': 1e-10, 'maxfun': maxiter * n})
        
        a = np.maximum(result.x, 0.0)
        s = a.sum()
        if s > 1e-10:
            a = a / s
        
        J = compute_J(a)
        tracker.update(a)
        print(f"    Softmax beta={beta}: J={J:.8f} (iter={result.nit})", flush=True)
    
    return a


def cd_pass(a):
    """One pass of coordinate descent with incremental updates."""
    a = a.astype(np.float64).copy()
    a = np.maximum(a, 0.0)
    S = a.sum()
    if S < 1e-10:
        return a
    n = len(a)
    b = fast_conv(a)
    current_max = b.max()
    
    indices = list(range(n))
    np.random.shuffle(indices)
    
    for i in indices:
        ai = a[i]
        if ai < 1e-15:
            continue
        
        # Find optimal value for a[i] using golden section
        def f_trial(x):
            new_S = S - ai + x
            if new_S < 1e-10:
                return 1e10
            # Compute new max
            max_c = 0.0
            # Only need to check k in [i, i+n-1] where c_k changes
            for k in range(max(0, i), min(2 * n, i + n)):
                ki = k - i
                if 0 <= ki < n:
                    if k == 2 * i:
                        ck = b[k] - ai * ai + x * x
                    else:
                        ck = b[k] - 2.0 * ai * a[ki] + 2.0 * x * a[ki]
                else:
                    ck = b[k]
                if ck > max_c:
                    max_c = ck
            # Also check unchanged values
            unchanged_max = 0.0
            for k in range(2 * n - 1):
                if k < i or k >= i + n:
                    if b[k] > unchanged_max:
                        unchanged_max = b[k]
            max_c = max(max_c, unchanged_max)
            return 2.0 * n * max_c / (new_S * new_S)
        
        # Golden section search
        gr = (np.sqrt(5.0) - 1.0) / 2.0
        lo = 0.0
        hi = max(2.0 * ai, S / n * 3.0)
        
        c_pt = hi - gr * (hi - lo)
        d_pt = lo + gr * (hi - lo)
        fc = f_trial(c_pt)
        fd = f_trial(d_pt)
        
        for _ in range(30):
            if fc < fd:
                hi = d_pt
                d_pt = c_pt
                fd = fc
                c_pt = hi - gr * (hi - lo)
                fc = f_trial(c_pt)
            else:
                lo = c_pt
                c_pt = d_pt
                fc = fd
                d_pt = lo + gr * (hi - lo)
                fd = f_trial(d_pt)
        
        x_opt = max(0.0, (lo + hi) / 2.0)
        
        # Only update if improvement
        J_new = f_trial(x_opt)
        J_old = 2.0 * n * current_max / (S * S)
        
        if J_new < J_old - 1e-12:
            old_val = a[i]
            a[i] = x_opt
            S = S - old_val + x_opt
            # Update convolution
            for k in range(max(0, i), min(2 * n - 1, i + n)):
                ki = k - i
                if 0 <= ki < n:
                    if k == 2 * i:
                        b[k] = b[k] - old_val * old_val + x_opt * x_opt
                    else:
                        b[k] = b[k] - 2.0 * old_val * a[ki] + 2.0 * x_opt * a[ki]
            current_max = max(b.max(), 0)
    
    return a


def resample_seq(src, dest_n):
    """Resample 1D sequence to new length."""
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)

"""
Minimizes J(a) = 2*n*max(convolve(a,a))/(sum(a))^2 using:
1. LP mirror descent with adaptive tightening
2. Softmax-smoothed L-BFGS-B optimization with beta annealing (KEY INNOVATION)
3. Fast coordinate descent with incremental convolution updates
4. Multi-length exploration
5. Perturbation restarts and final polish
The softmax smoothing transforms the non-smooth max into a differentiable
function enabling powerful quasi-Newton optimization.
"""

import numpy as np
from scipy.optimize import linprog, minimize
from scipy.sparse import lil_matrix, csr_matrix
import time


def fast_conv(a):
    """Full self-convolution via FFT."""
    n = len(a)
    s = 1
    while s < 2 * n - 1:
        s <<= 1
    fa = np.fft.rfft(a, s)
    return np.fft.irfft(fa * fa, s)[:2 * n - 1]


def cross_corr_fft(p, a):
    """Compute (p cross a)[i] = sum_m p[i+m]*a[m] for i=0..n-1 using FFT."""
    n = len(a)
    N = 1
    while N < len(p) + n - 1:
        N <<= 1
    P = np.fft.rfft(p, N)
    A = np.fft.rfft(a, N)
    result = np.fft.irfft(P * np.conj(A), N)
    return result[:n]


def compute_J(a):
    """Compute J(a) = 2*n*max(conv(a,a))/(sum(a))^2."""
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))


def softmax_J_grad(a, beta):
    """
    Compute smooth approximation of J and its gradient.
    Uses log-sum-exp for numerical stability.
    J_smooth = 2n * logsumexp(beta*c) / (beta * S^2)
    """
    n = len(a)
    c = fast_conv(a)
    S = a.sum()
    if S < 1e-10:
        return 1e10, np.zeros(n)

    bc = beta * c
    bc_max = bc.max()
    exp_bc = np.exp(bc - bc_max)
    sum_exp = exp_bc.sum()
    if sum_exp < 1e-300:
        return 1e10, np.zeros(n)

    p = exp_bc / sum_exp
    F = (bc_max + np.log(sum_exp)) / beta

    # grad_F[i] = 2 * sum_{m=0}^{n-1} p[i+m] * a[m]
    grad_F = 2.0 * cross_corr_fft(p, a)

    J = 2.0 * n * F / (S * S)
    grad_J = 2.0 * n * (grad_F - 2.0 * F / S) / (S * S)

    return J, grad_J


def build_A_sparse(a):
    """Build sparse constraint matrix for LP: (a*g)_k for all k."""
    n = len(a)
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        jl = max(0, k - n + 1)
        jh = min(n - 1, k)
        if jl <= jh:
            cnt = jh - jl + 1
            rows.extend([k] * cnt)
            cols.extend(range(jl, jh + 1))
            seg = a[k - jh:k - jl + 1][::-1]
            vals.extend(seg.tolist())
    A = lil_matrix((2 * n - 1, n))
    for idx in range(len(rows)):
        A[rows[idx], cols[idx]] = vals[idx]
    return A.tocsr()


def lp_best_response(a, M):
    """LP: max sum(g) s.t. (a*g)_k <= M for all k, g >= 0."""
    n = len(a)
    c_obj = -np.ones(n)
    A = build_A_sparse(a)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    try:
        r = linprog(c_obj, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
        if r.success and r.x is not None:
            g = np.maximum(r.x, 0.0)
            if g.sum() > 1e-10:
                return g
    except Exception:
        pass
    return None


def run_lp(a, max_iters, deadline, tracker, tighten_base=0.004):
    """LP mirror descent iterations with adaptive tightening."""
    a = np.maximum(a.astype(np.float64), 0.001)
    a /= a.sum()
    n = len(a)
    current_J = compute_J(a)
    tracker.update(a)

    tighten = tighten_base
    stall = 0

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        b = fast_conv(a)
        M = float(b.max())
        M_target = M * (1.0 - tighten)

        g = lp_best_response(a, M_target)
        if g is None:
            M_target = M * (1.0 - tighten * 0.5)
            g = lp_best_response(a, M_target)
            if g is None:
                tighten *= 0.5
                if tighten < 1e-7:
                    tighten = tighten_base
                stall += 1
                if stall > 30:
                    break
                continue

        g = np.maximum(g, 0.0)
        g_sum = g.sum()
        if g_sum < 1e-10:
            break
        g /= g_sum

        # Try multiple step sizes
        progress = min(1.0, it / max(1, max_iters))
        base_t = max(0.02, 0.4 * (1.0 - 0.5 * progress))
        candidates_t = [base_t, base_t * 0.5, base_t * 1.5, 0.1, 0.25]

        best_a = None
        best_J = current_J

        for t in candidates_t:
            a_trial = (1.0 - t) * a + t * g
            a_trial = np.maximum(a_trial, 0.0)
            s = a_trial.sum()
            if s < 1e-10:
                continue
            a_trial /= s
            J_trial = compute_J(a_trial)
            if J_trial < best_J - 1e-14:
                best_J = J_trial
                best_a = a_trial

        if best_a is None:
            stall += 1
            if stall > 30:
                break
            tighten *= 0.8
            continue

        a = best_a
        current_J = best_J
        stall = 0
        tighten = min(0.05, tighten * 1.03)

        if it % 25 == 0:
            tracker.update(a)

    tracker.update(a)
    return a


def softmax_anneal(a, deadline, tracker, betas=None):
    """Optimize using softmax smooth approximation with L-BFGS-B and beta annealing."""
    if betas is None:
        betas = [5, 10, 20, 50, 100, 200, 400]

    a = np.maximum(a.astype(np.float64), 0.001)
    a = a / a.sum()
    n = len(a)

    for beta in betas:
        if time.monotonic() > deadline:
            break

        def obj_grad(x):
            x = np.maximum(x, 0.0)
            J, g = softmax_J_grad(x, beta)
            g = np.clip(g, -1e6, 1e6)
            return J, g

        # Scale-aware max iterations
        maxiter = 150
        result = minimize(obj_grad, a, method='L-BFGS-B',
                         bounds=[(0.0, 5.0)] * n,
                         options={'maxiter': maxiter, 'ftol': 1e-14,
                                  'gtol': 1e-9, 'maxfun': maxiter * 3})

        a = np.maximum(result.x, 0.0)
        s = a.sum()
        if s > 1e-10:
            a = a / s

        J = compute_J(a)
        tracker.update(a)
        print(f"    Softmax beta={beta}: J={J:.8f} (nit={result.nit})", flush=True)

    return a


def cd_pass(a):
    """One efficient pass of coordinate descent with precomputed coefficients."""
    a = a.astype(np.float64).copy()
    a = np.maximum(a, 0.0)
    S = a.sum()
    if S < 1e-10:
        return a
    n = len(a)
    b = fast_conv(a)

    indices = list(range(n))
    np.random.shuffle(indices)

    for i in indices:
        ai = a[i]
        if ai < 1e-15:
            continue

        k_lo = max(0, i)
        k_hi = min(2 * n - 1, i + n - 1)
        k_range = range(k_lo, k_hi + 1)
        nk = k_hi - k_lo + 1

        # Precompute unchanged max
        if k_lo > 0:
            unchanged_left = b[:k_lo].max()
        else:
            unchanged_left = 0.0
        if k_hi < 2 * n - 1:
            unchanged_right = b[k_hi + 1:].max()
        else:
            unchanged_right = 0.0
        unchanged_max = max(unchanged_left, unchanged_right)

        # Precompute coefficients for changed convolution values
        # c_k(x) = base_k + coeff_k * x + quad_k * x^2
        base = np.zeros(nk)
        coeff = np.zeros(nk)
        quad = np.zeros(nk)
        for idx, k in enumerate(k_range):
            ki = k - i
            if ki == i:
                quad[idx] = 1.0
                base[idx] = b[k] - ai * ai
            elif 0 <= ki < n:
                coeff[idx] = 2.0 * a[ki]
                base[idx] = b[k] - 2.0 * ai * a[ki]
            else:
                base[idx] = b[k]

        def f_trial(x):
            new_S = S - ai + x
            if new_S < 1e-10:
                return 1e10
            changed = base + coeff * x + quad * x * x
            max_c = max(changed.max(), unchanged_max)
            return 2.0 * n * max_c / (new_S * new_S)

        J_cur = 2.0 * n * b.max() / (S * S)

        # Golden section search
        gr = (np.sqrt(5.0) - 1.0) / 2.0
        lo = 0.0
        hi = max(3.0 * ai, 3.0 * S / n)

        c_pt = hi - gr * (hi - lo)
        d_pt = lo + gr * (hi - lo)
        fc = f_trial(c_pt)
        fd = f_trial(d_pt)

        for _ in range(30):
            if fc < fd:
                hi = d_pt
                d_pt = c_pt
                fd = fc
                c_pt = hi - gr * (hi - lo)
                fc = f_trial(c_pt)
            else:
                lo = c_pt
                c_pt = d_pt
                fc = fd
                d_pt = lo + gr * (hi - lo)
                fd = f_trial(d_pt)

        x_opt = max(0.0, (lo + hi) / 2.0)

        # Verify improvement
        J_new = f_trial(x_opt)
        if J_new < J_cur - 1e-13:
            a[i] = x_opt
            S = S - ai + x_opt
            for idx, k in enumerate(k_range):
                b[k] = base[idx] + coeff[idx] * x_opt + quad[idx] * x_opt * x_opt

    return a


def resample_seq(src, dest_n):
    """Resample 1D sequence to new length."""
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)


class ScoreTracker:
    def __init__(self):
        self.best = None
        self.best_score = float('inf')
        self.t_start = time.monotonic()

    def update(self, a):
        a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
        if a.size == 0 or a.sum() < 0.01:
            return
        score = float(evaluate_sequence(a.tolist()))
        if np.isfinite(score) and score < self.best_score:
            self.best = a.copy()
            self.best_score = score
            elapsed = time.monotonic() - self.t_start
            print(f"  * NEW BEST: J={score:.8f} n={a.size} t={elapsed:.0f}s", flush=True)

    def report(self, label=""):
        if self.best is not None:
            print(f"  {label} best: J={self.best_score:.8f} n={self.best.size} "
                  f"t={time.monotonic()-self.t_start:.0f}s", flush=True)


def propose_candidate(seed=42, budget_s=1500, **kwargs):
    """
    Main search: LP + softmax L-BFGS-B + coordinate descent, alternating.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + float(budget_s) - 25.0
    tracker = ScoreTracker()

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        if p.ndim == 1 and p.size > 0 and np.all(np.isfinite(p)):
            parent = np.maximum(p, 0.0)
            tracker.update(parent)
            print(f"Parent loaded: n={parent.size}, J={tracker.best_score:.8f}", flush=True)
    except (NameError, AttributeError):
        pass

    if parent is None:
        parent = np.ones(384)
        tracker.update(parent)
        print(f"No parent found, using uniform n=384", flush=True)

    # === Phase 1: LP optimization ===
    print(f"\n{'='*50} Phase 1: LP {'='*50}", flush=True)
    t0 = time.monotonic()
    a = parent.copy()
    a = run_lp(a, 120, deadline - 600, tracker, tighten_base=0.004)
    tracker.report(f"Phase 1: {time.monotonic()-t0:.1f}s")

    # === Phase 2: Softmax L-BFGS-B (main innovation) ===
    if time.monotonic() < deadline - 400:
        print(f"\n{'='*50} Phase 2: Softmax L-BFGS-B {'='*50}", flush=True)
        t0 = time.monotonic()
        a = softmax_anneal(tracker.best, deadline - 400, tracker,
                           betas=[5, 10, 20, 50, 100, 200])
        tracker.report(f"Phase 2: {time.monotonic()-t0:.1f}s")

    # === Phase 3: LP again from improved point ===
    if time.monotonic() < deadline - 300:
        print(f"\n{'='*50} Phase 3: LP round 2 {'='*50}", flush=True)
        t0 = time.monotonic()
        a = run_lp(tracker.best, 80, deadline - 300, tracker, tighten_base=0.003)
        tracker.report(f"Phase 3: {time.monotonic()-t0:.1f}s")

    # === Phase 4: Softmax with higher betas ===
    if time.monotonic() < deadline - 200:
        print(f"\n{'='*50} Phase 4: Softmax high-beta {'='*50}", flush=True)
        t0 = time.monotonic()
        a = softmax_anneal(tracker.best, deadline - 200, tracker,
                           betas=[100, 200, 400, 600])
        tracker.report(f"Phase 4: {time.monotonic()-t0:.1f}s")

    # === Phase 5: Coordinate descent ===
    if time.monotonic() < deadline - 120:
        print(f"\n{'='*50} Phase 5: Coordinate Descent {'='*50}", flush=True)
        t0 = time.monotonic()
        a_cd = tracker.best.copy()
        for p in range(6):
            if time.monotonic() > deadline - 100:
                break
            a_cd = cd_pass(a_cd)
            a_cd = np.maximum(a_cd, 0.0)
            if a_cd.sum() > 1e-10:
                a_cd /= a_cd.sum()
            tracker.update(a_cd)
        tracker.report(f"Phase 5: {time.monotonic()-t0:.1f}s")

    # === Phase 6: LP + Softmax on CD result ===
    if time.monotonic() < deadline - 100:
        print(f"\n{'='*50} Phase 6: LP+Softmax polish {'='*50}", flush=True)
        t0 = time.monotonic()
        a = run_lp(tracker.best, 40, time.monotonic() + 60, tracker, tighten_base=0.002)
        if time.monotonic() < deadline - 60:
            a = softmax_anneal(tracker.best, time.monotonic() + min(40, deadline - time.monotonic() - 50),
                              tracker, betas=[200, 400])
        tracker.report(f"Phase 6: {time.monotonic()-t0:.1f}s")

    # === Phase 7: Nearby lengths ===
    if time.monotonic() < deadline - 60:
        print(f"\n{'='*50} Phase 7: Nearby lengths {'='*50}", flush=True)
        t0 = time.monotonic()
        n_best = tracker.best.size
        for nl in [n_best - 2, n_best + 2, n_best - 4, n_best + 4, n_best + 6, n_best - 6]:
            if time.monotonic() > deadline - 50:
                break
            if nl < 200 or nl > 600:
                continue
            a_res = resample_seq(tracker.best, nl)
            a_res = np.maximum(a_res, 0.001)
            a_res /= a_res.sum()
            a_res = run_lp(a_res, 25, time.monotonic() + 30, tracker)
            if time.monotonic() < deadline - 50:
                a_res = cd_pass(a_res)
                a_res = np.maximum(a_res, 0.0)
                if a_res.sum() > 1e-10:
                    a_res /= a_res.sum()
                tracker.update(a_res)
            print(f"    n={nl}: J={tracker.best_score:.8f}", flush=True)
        tracker.report(f"Phase 7: {time.monotonic()-t0:.1f}s")

    # === Phase 8: Perturbation restarts ===
    if time.monotonic() < deadline - 30:
        print(f"\n{'='*50} Phase 8: Perturbations {'='*50}", flush=True)
        t0 = time.monotonic()
        n = tracker.best.size
        for temp in [0.04, 0.02, 0.01]:
            if time.monotonic() > deadline - 25:
                break
            a_pert = tracker.best + rng.normal(0, temp, n)
            a_pert = np.maximum(a_pert, 0.001)
            a_pert /= a_pert.sum()
            a_pert = run_lp(a_pert, 15, time.monotonic() + min(20, deadline - time.monotonic() - 20), tracker)
            if time.monotonic() < deadline - 20 and a_pert.sum() > 0:
                a_pert = cd_pass(a_pert)
                a_pert = np.maximum(a_pert, 0.0)
                if a_pert.sum() > 1e-10:
                    a_pert /= a_pert.sum()
                tracker.update(a_pert)
        tracker.report(f"Phase 8: {time.monotonic()-t0:.1f}s")

    # === Phase 9: Final CD + Softmax ===
    if time.monotonic() < deadline - 5:
        print(f"\n{'='*50} Phase 9: Final Polish {'='*50}", flush=True)
        t0 = time.monotonic()
        a_final = cd_pass(tracker.best)
        a_final = np.maximum(a_final, 0.0)
        if a_final.sum() > 1e-10:
            a_final /= a_final.sum()
        tracker.update(a_final)
        if time.monotonic() < deadline - 3:
            _ = softmax_anneal(tracker.best, deadline - 2, tracker, betas=[300])
        tracker.report(f"Phase 9: {time.monotonic()-t0:.1f}s")

    # Fallback
    if tracker.best is None:
        tracker.best = np.ones(256)
        tracker.best_score = compute_J(tracker.best)

    print(f"\n{'='*50} FINAL RESULT {'='*50}", flush=True)
    print(f"J = {tracker.best_score:.10f}, n = {tracker.best.size}", flush=True)
    print(f"Total time: {time.monotonic()-tracker.t_start:.1f}s", flush=True)

    return [float(x) for x in tracker.best]
