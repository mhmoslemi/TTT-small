Improved optimizer for minimizing J(a) = 2*n*max(conv(a,a))/(sum(a))^2.

import numpy as np
from scipy.optimize import linprog
from scipy.sparse import csr_matrix
import time


def fast_conv(a):
    """Full self-convolution via FFT."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]


def fast_cross_conv(a, g):
    """Compute a*g (cross-convolution) via FFT."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    fg = np.fft.rfft(g, size)
    return np.fft.irfft(fa * fg, size)[:2 * n - 1]


def compute_J(a):
    """Compute J(a) = 2*n*max(conv(a,a))/(sum(a))^2."""
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))


def build_conv_structure(n):
    """Build sparsity structure for convolution matrix."""
    rows = []
    cols = []
    for k in range(2 * n - 1):
        j_min = max(0, k - n + 1)
        j_max = min(n - 1, k)
        for j in range(j_min, j_max + 1):
            rows.append(k)
            cols.append(j)
    return np.array(rows), np.array(cols)


def solve_lp(a, M, srows, scols):
    """Solve LP: max sum(g) s.t. (a*g)[k] <= M for all k, g >= 0."""
    n = len(a)
    num_constraints = 2 * n - 1
    vals = a[srows - scols]
    A = csr_matrix((vals, (srows, scols)), shape=(num_constraints, n))
    c = -np.ones(n)
    b_ub = np.full(num_constraints, M)
    bounds = [(0, None)] * n
    try:
        r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                    options={'presolve': True, 'dual_feasibility_tolerance': 1e-9})
        if r.success and r.x is not None:
            return np.maximum(r.x, 0)
    except Exception:
        pass
    return None


def lp_mirror_descent(a, max_iters, deadline, verbose=False):
    """LP mirror descent for minimizing max(conv(a,a)) with sum(a)=1, a>=0."""
    a = np.maximum(a.astype(np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        return a, float('inf')
    a = a / s
    n = len(a)

    srows, scols = build_conv_structure(n)

    current_M = float(fast_conv(a).max())
    best_M = current_M
    best_a = a.copy()
    no_improve = 0

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        b = fast_conv(a)
        M = float(b.max())

        g = solve_lp(a, M, srows, scols)
        if g is None:
            break
        g_sum = g.sum()
        if g_sum < 1e-12:
            break
        g = g / g_sum

        best_trial_M = M
        best_trial_a = None
        t = 1.0

        for _ in range(15):
            a_trial = (1.0 - t) * a + t * g
            a_trial = np.maximum(a_trial, 0.0)
            s_trial = a_trial.sum()
            if s_trial < 1e-10:
                t *= 0.5
                continue
            a_trial = a_trial / s_trial
            M_trial = float(fast_conv(a_trial).max())
            if M_trial < best_trial_M - 1e-16:
                best_trial_M = M_trial

                break
            t *= 0.55
            if t < 0.005:
                break

        if best_trial_a is not None:
            a = best_trial_a
            current_M = best_trial_M
            if current_M < best_M:
                best_M = current_M
                best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1
            if no_improve > 60:

            # Try a smaller perturbation to escape

            a_pert = np.maximum(a_pert, 0)
            sp = a_pert.sum()
            if sp > 1e-10:
                a = a_pert / sp

    return best_a, best_M


def subgradient_polish(a, max_iters, deadline):
    """FFT-accelerated subgradient method."""
    a = np.maximum(a.astype(np.float64), 0)
    s = a.sum()

        return a
    a = a / s
    n = len(a)

    size = 1

    best_a = a.copy()

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        # FFT-based convolution of y with itself
        fy = np.fft.rfft(y, size)
        b = np.fft.irfft(fy * fy, size)[:2 * n - 1]

        tol = 1e-9 * M + 1e-15
        active = (b >= M - tol).astype(np.float64)

        # Gradient: grad[i] = 2 * sum_{k in active} y[k-i]
        # This is the cross-correlation of y with the active indicator
        fa = np.fft.rfft(active, size)
        grad_full = np.fft.irfft(fy * fa, size)
        # The gradient for index i is: sum_k active[k] * y[k-i]
        # which is the circular cross-correlation, but we need linear
        # Actually: sum_k active[k] * y[k-i] = (active * y_flipped)[i] ... 
        # Let me think again.
        # grad[i] = 2 * sum_{k=0}^{2n-2} active[k] * y[k-i] (where y[j]=0 for j outside [0,n-1])
        # = 2 * sum_{k=i}^{i+n-1} active[k] * y[k-i] (only valid y indices)
        # This is like the cross-correlation of active with y, evaluated at lag i
        # cross_corr[i] = sum_k active[k] * y[k-i]
        # = (active * reverse(y))[n-1-i] ... this is getting complicated
        # Let me just use the FFT approach:
        # grad[i] = 2 * sum_{k} active[k] * y[k-i]
        # = 2 * sum_{k} active[k] * y[k-i]
        # If we define z[j] = y[j] for j in [0,n-1], 0 otherwise (length 2n-1)
        # Then grad[i] = 2 * sum_k active[k] * z[k-i]
        # = 2 * (active * z)[2n-2-i] ... no
        # Actually grad[i] = 2 * sum_k active[k] * z[k-i] = 2 * (z * active_flipped)[i]
        # where active_flipped[k] = active[2n-2-k]
        # Or more simply: this is the convolution of z and flipped active
        # Let me just compute it directly for the active set (usually small)
        
        active_idx = np.where(b >= M - tol)[0]
        grad = np.zeros(n)
        for k in active_idx:
            lo = max(0, k - n + 1)
            hi = min(n - 1, k)
            for i in range(lo, hi + 1):
                grad[i] += 2.0 * y[k - i]

        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-18:
            break

        # Nesterov accelerated step
        t_k = 2.0 / (it + 2.0)
        x_new = x - t_k * grad / gnorm
        x_new = np.maximum(x_new, 0)
        s_new = x_new.sum()
        if s_new < 1e-10:

            s_new = 1.0

            y = x_new + beta * (x_new - x)
            y = np.maximum(y, 0)
            sy = y.sum()
            if sy > 1e-10:
                y = y / sy
        else:
            y = x_new.copy()

        x = x_new

        M_x = float(fast_conv(x).max())
        if M_x < best_M:
            best_M = M_x
            best_a = x.copy()

def resample_seq(src, dest_n):
    """Resample 1D sequence to new length."""
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)


def generate_inits(n, rng, parent=None):

    inits.append(np.ones(n))
    # Cosine
    inits.append(0.5 * (1 + np.cos(2 * np.pi * t)))
    # Two-hump
    inits.append(np.maximum(0.4 - 0.5 * np.cos(2 * np.pi * t) + 0.1 * np.cos(4 * np.pi * t), 0.01))
    # Gaussian

    # Double Gaussian

    # Alternating (based on [1,r] insight)
    r_opt = 2 + np.sqrt(2)
    alt = np.where(np.arange(n) % 2 == 0, 1.0, r_opt)
    inits.append(alt)
    # Various frequencies

        inits.append(np.maximum(1.0 + 0.3 * np.cos(2 * np.pi * f * t), 0.01))
    # Triangle
    inits.append(np.maximum(1.0 - 0.6 * np.abs(2 * t - 1), 0.05))
    # Power law
    inits.append((t + 0.1) ** 2)
    inits.append((t + 0.1) ** 0.5)
    # Smooth random
    k = max(3, n // 16)
    kernel = np.ones(k) / k
    smooth = np.convolve(rng.uniform(0, 1, n), kernel, mode='same')
    inits.append(np.maximum(smooth + 0.2, 0.01))
    # More structured
    inits.append(np.maximum(1.0 + 0.4 * np.sin(2 * np.pi * 3 * t + 0.5), 0.01))
    # Tapered
    window = np.hanning(n)
    inits.append(np.maximum(0.5 + 0.5 * window, 0.01))

    result = []
    for a in inits:
        a = np.maximum(a, 0.001)
        s = a.sum()
        if s > 1e-6:
            result.append(a / s)
    return result

def dual_guided_update(a, max_iters, deadline):
    """Use LP dual information to guide updates more effectively."""
    a = np.maximum(a.astype(np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        return a, float('inf')
    a = a / s
    n = len(a)
    srows, scols = build_conv_structure(n)

    best_M = float(fast_conv(a).max())
    best_a = a.copy()

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        b = fast_conv(a)
        M = float(b.max())

        # Solve LP and get dual variables
        num_constraints = 2 * n - 1
        vals = a[srows - scols]
        A = csr_matrix((vals, (srows, scols)), shape=(num_constraints, n))
        c = -np.ones(n)
        b_ub = np.full(num_constraints, M)
        bounds = [(0, None)] * n

        try:
            r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                        options={'presolve': True})
            if not r.success or r.x is None:
                break
            g = np.maximum(r.x, 0)
        except Exception:
            break

        g_sum = g.sum()
        if g_sum < 1e-12:
            break

        # Check convergence: if g is proportional to a, we're at a fixed point
        ratio = g / np.maximum(a, 1e-15)
        if np.std(ratio[np.isfinite(ratio)]) < 1e-8:
            break

        # Water-filling inspired update: push more mass to where constraints are less tight
        a_trial = (1.0 - 0.3) * a + 0.3 * g / g_sum
        a_trial = np.maximum(a_trial, 0)
        st = a_trial.sum()
        if st < 1e-10:
            break
        a_trial = a_trial / st

        M_trial = float(fast_conv(a_trial).max())
        if M_trial < best_M:
            a = a_trial
            best_M = M_trial
            best_a = a.copy()

    return best_a, best_M

    if J7 < current_best_J:
        current_best = a7.copy()
        current_best_J = J7

    # Final decision
    if tracker.best is not None:
        final_a = tracker.best
    else:
        final_a = current_best

    # One final subgradient polish if time allows
    if time.monotonic() < total_deadline - 5:
        a_final = subgradient_polish(final_a, 10000, total_deadline)
        tracker.update(a_final)

    print(f"\n{'='*50}", flush=True)
    print(f"FINAL: J={tracker.best_score:.8f} n={tracker.best.size} time={time.monotonic()-tracker.t_start:.0f}s", flush=True)
    print(f"{'='*50}", flush=True)

    if tracker.best is not None:
        return [float(x) for x in tracker.best]
    return [float(x) for x in final_a]
