import numpy as np
import time
from scipy.optimize import linprog

try:
    import cvxpy as cp
    HAS_CVXPY = True
except ImportError:
    HAS_CVXPY = False


def compute_conv(a):
    """Compute linear convolution conv(a,a) using FFT."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    conv = np.fft.irfft(fa * fa, size)
    return conv[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    conv = compute_conv(a)
    max_b = float(np.max(conv))
    sum_a = float(np.sum(a))
    if sum_a < 0.01:
        return np.inf
    return 2.0 * n * max_b / (sum_a * sum_a)


def build_toeplitz(a):
    """Build Toeplitz matrix A where (A@g)[k] = sum_j a[j]*g[k-j] for k=0..2n-2."""
    n = len(a)
    A = np.zeros((2 * n - 1, n), dtype=np.float64)
    for j in range(n):
        k_end = min(j + n, 2 * n - 1)
        A[j:k_end, j] = a[:k_end - j]
    return A


def solve_lp_minpeak(a, time_limit=5):
    """Solve LP: min M s.t. (a*g)[k] <= M for all k, sum(g)=1, g>=0.
    Returns (g, M) or (None, None)."""
    n = len(a)
    A = build_toeplitz(a)
    # Variables: g[0..n-1], M
    # Minimize M
    c = np.zeros(n + 1)
    c[n] = 1.0
    A_ub = np.zeros((2 * n - 1, n + 1))
    A_ub[:, :n] = A
    A_ub[:, n] = -1.0
    b_ub = np.zeros(2 * n - 1)
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])
    bounds = [(0.0, None)] * n + [(0.0, None)]
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': time_limit})
        if res.status == 0:
            g = np.maximum(res.x[:n], 0)
            s = np.sum(g)
            if s < 1e-10:
                return None, None
            g = g / s
            return g, float(res.x[n])
    except Exception:
        pass
    return None, None


def solve_lp_maxsum(a, M, time_limit=5):
    """Solve LP: max sum(g) s.t. (a*g)[k] <= M for all k, g >= 0."""
    n = len(a)
    A = build_toeplitz(a)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': time_limit})
        if res.status == 0:
            g = np.maximum(res.x, 0)
            return g, float(np.sum(g))
    except Exception:
        pass
    return None, None


def smooth_gradient(a, tau):
    """Compute smooth gradient of logsumexp(conv/tau)*tau with respect to a."""
    n = len(a)
    conv = compute_conv(a)
    max_conv = np.max(conv)
    shifted = (conv - max_conv) / tau
    shifted = np.clip(shifted, -500, 0)
    w = np.exp(shifted)
    w_sum = np.sum(w)
    if w_sum < 1e-300:
        # Degenerate: gradient points at the argmax
        idx = int(np.argmax(conv))
        w = np.zeros(2 * n - 1)
        w[idx] = 1.0
    else:
        w /= w_sum
    # grad[i] = 2 * sum_k w[k] * a[k-i] (correlation)
    # This is correlate(w, a) evaluated at the right offset
    c = np.correlate(w, a, mode='full')
    grad = 2.0 * c[n - 1:n - 1 + n]
    return grad, conv, max_conv


def nesterov_smooth_optimize(a, time_budget=30.0, tau_start=3.0, tau_end=50.0,
                              eta_start=0.1, eta_end=0.01, momentum_start=0.3,
                              patience=200):
    """Nesterov-accelerated smooth optimization with adaptive parameters."""
    n = len(a)
    a = a.copy().astype(np.float64)
    s = np.sum(a)
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    # Nesterov momentum variables
    a_prev = a.copy()
    y = a.copy()  # extrapolated point
    momentum = momentum_start

    stall = 0
    it = 0
    total_iters = int(time_budget * 200)  # estimate

    while it < total_iters and time.monotonic() - start < time_budget:
        it += 1
        progress = it / total_iters

        # Linearly interpolate parameters
        tau = tau_start + (tau_end - tau_start) * progress
        eta = eta_start + (eta_end - eta_start) * progress
        momentum = momentum_start + (0.9 - momentum_start) * min(progress * 2, 1.0)

        # Nesterov extrapolation
        a_curr = a.copy()
        y = a_curr + momentum * (a_curr - a_prev)
        y = np.maximum(y, 1e-15)
        y_sum = np.sum(y)
        if y_sum > 1e-10:
            y = y / y_sum

        # Compute gradient at y
        grad, conv, max_conv = smooth_gradient(y, tau)

        # Center gradient to preserve sum approximately
        grad_centered = grad - np.mean(grad)

        # Multiplicative update (naturally maintains positivity)
        log_update = -eta * grad_centered
        log_update = np.clip(log_update, -10, 10)
        a_new = a_curr * np.exp(log_update)
        a_new = np.maximum(a_new, 1e-15)
        s_new = np.sum(a_new)
        if s_new > 1e-10:
            a_new = a_new / s_new

        a_prev = a_curr
        a = a_new

        obj = compute_obj(a)
        if obj < best_obj - 1e-14:
            best_obj = obj
            best_a = a.copy()
            stall = 0
        else:
            stall += 1
            if stall > patience:
                # Perturb and reset
                rng = np.random.default_rng(it * 104729)
                a = a * np.exp(rng.normal(0, 0.05, n))
                a = np.maximum(a, 1e-10)
                s = np.sum(a)
                if s > 1e-10:
                    a = a / s
                a_prev = a.copy()
                stall = 0
                patience = max(patience // 2, 50)
                momentum = momentum_start

    return best_a


def peak_targeting_search(a, time_budget=20.0):
    """Local search that specifically targets the peak lag of the autocorrelation."""
    n = len(a)
    a = a.copy().astype(np.float64)
    s = np.sum(a)
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    for attempt in range(200):
        if time.monotonic() - start > time_budget:
            break

        conv = compute_conv(best_a)
        peak_lag = int(np.argmax(conv))

        # The contribution of a[i] to conv[peak_lag] is:
        # 2 * a[peak_lag - i] if 0 <= peak_lag - i < n
        # So elements a[j] where j is near peak_lag contribute more
        # To reduce the peak, decrease a at positions that contribute to the peak,
        # and increase a at positions that don't

        # Compute sensitivity vector
        sensitivity = np.zeros(n)
        for j in range(n):
            k = peak_lag - j
            if 0 <= k < n:
                sensitivity[j] = 2.0 * best_a[k]

        # Normalize sensitivity
        sens_max = np.max(sensitivity)
        if sens_max > 1e-15:
            sensitivity /= sens_max

        # Perturbation: decrease high-sensitivity elements, increase low-sensitivity
        rng = np.random.default_rng(attempt * 31 + 7)
        scale = 0.03 + 0.04 * (attempt % 3)

        # Create perturbation direction
        direction = -sensitivity + 0.5 * np.ones(n)  # negative for high sensitivity
        direction += rng.normal(0, 0.1, n)  # add noise for exploration

        # Apply multiplicative perturbation
        a_new = best_a * np.exp(scale * direction)
        a_new = np.maximum(a_new, 1e-10)
        s_new = np.sum(a_new)
        if s_new > 1e-10:
            a_new = a_new / s_new

        obj_new = compute_obj(a_new)
        if obj_new < best_obj:
            best_obj = obj_new
            best_a = a_new.copy()

        # Also try a direct "spreading" perturbation
        # Move mass from peak-contributing positions to non-contributing
        if attempt % 3 == 0:
            top_k = max(5, n // 10)
            sens_sorted = np.argsort(sensitivity)
            top_positions = sens_sorted[-top_k:]
            bottom_positions = sens_sorted[:top_k]

            a_spread = best_a.copy()
            transfer = 0.02 * np.mean(best_a[top_positions])
            a_spread[top_positions] -= transfer
            a_spread[bottom_positions] += transfer
            a_spread = np.maximum(a_spread, 0)
            s_sp = np.sum(a_spread)
            if s_sp > 1e-10:
                a_spread = a_spread / s_sp
                obj_spread = compute_obj(a_spread)
                if obj_spread < best_obj:
                    best_obj = obj_spread
                    best_a = a_spread.copy()

    return best_a


def lp_smooth_hybrid(a, time_budget=60.0, lp_every=15):
    """Hybrid of LP and smooth optimization with smart switching."""
    n = len(a)
    a = a.copy().astype(np.float64)
    s = np.sum(a)
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    it = 0
    stall = 0

    while time.monotonic() - start < time_budget:
        remaining = time_budget - (time.monotonic() - start)
        if remaining < 5:
            break

        if it % lp_every == 0 and remaining > 10:
            # LP step
            g, M_opt = solve_lp_minpeak(a, time_limit=min(5, remaining * 0.3))
            if g is not None:
                # Line search
                best_t = 0.0
                best_obj_ls = best_obj
                for t in [0.3, 0.5, 0.7, 1.0]:
                    a_new = (1.0 - t) * a + t * g
                    a_new = np.maximum(a_new, 0)
                    s_new = np.sum(a_new)
                    if s_new < 1e-10:
                        continue
                    a_new = a_new / s_new
                    obj_new = compute_obj(a_new)
                    if obj_new < best_obj_ls:
                        best_obj_ls = obj_new
                        best_t = t

                if best_t > 0 and best_obj_ls < best_obj - 1e-13:
                    a_new = (1.0 - best_t) * a + best_t * g
                    a_new = np.maximum(a_new, 0)
                    s_new = np.sum(a_new)
                    if s_new > 1e-10:
                        a_new = a_new / s_new
                    best_obj = best_obj_ls
                    best_a = a_new.copy()
                    a = a_new
                    stall = 0
                else:
                    stall += 1
                    if stall > 3 * lp_every:
                        break
                    # Forced update
                    a = 0.85 * a + 0.15 * g
                    a = np.maximum(a, 0)
                    s = np.sum(a)
                    if s > 1e-10:
                        a = a / s
            else:
                stall += lp_every
                if stall > 3 * lp_every:
                    break
        else:
            # Smooth step
            short_budget = min(2.0, remaining * 0.1)
            a_new = nesterov_smooth_optimize(a, time_budget=short_budget,
                                              tau_start=5.0, tau_end=30.0,
                                              eta_start=0.08, eta_end=0.02,
                                              momentum_start=0.5,
                                              patience=50)
            obj_new = compute_obj(a_new)
            if obj_new < best_obj - 1e-13:
                best_obj = obj_new
                best_a = a_new.copy()
                a = a_new
                stall = 0
            else:
                stall += 1
                if stall > 5 * lp_every:
                    break

        it += 1

    return best_a


def random_init(n, seed, style='diverse'):
    """Generate diverse random initializations."""
    rng = np.random.default_rng(seed)
    k = np.arange(n)
    choice = rng.integers(0, 8)

    if choice == 0:
        f = rng.uniform(0.5, 3.0)
        phase = rng.uniform(0, 2 * np.pi)
        a = 1.0 + 0.7 * np.cos(2 * np.pi * f * k / n + phase)
    elif choice == 1:
        f1 = rng.uniform(0.3, 2.0)
        f2 = rng.uniform(2.0, 5.0)
        a = 1.0 + 0.3 * np.cos(2 * np.pi * f1 * k / n) + 0.3 * np.cos(2 * np.pi * f2 * k / n)
    elif choice == 2:
        a = rng.random(n)
        win = max(3, n // 15)
        kernel = np.ones(win) / win
        a = np.convolve(a, kernel, mode='same')
    elif choice == 3:
        a = np.ones(n)
        for _ in range(rng.integers(1, 4)):
            f = rng.uniform(0.5, 5.0)
            amp = rng.uniform(0.1, 0.4)
            a += amp * np.cos(2 * np.pi * f * k / n + rng.uniform(0, 2 * np.pi))
    elif choice == 4:
        # Two-bump
        c1 = n * rng.uniform(0.2, 0.4)
        c2 = n * rng.uniform(0.6, 0.8)
        w = n * rng.uniform(0.05, 0.15)
        a = 1.0 + 0.8 * np.exp(-0.5 * ((k - c1) / w) ** 2) + 0.8 * np.exp(-0.5 * ((k - c2) / w) ** 2)
    elif choice == 5:
        # Chirp
        f0 = rng.uniform(0.3, 1.0)
        f1 = rng.uniform(3.0, 7.0)
        t = k / n
        f = f0 + (f1 - f0) * t
        a = 1.0 + 0.5 * np.cos(2 * np.pi * np.cumsum(f))
    elif choice == 6:
        # Random smooth via interpolation
        n_ctrl = max(5, n // 8)
        ctrl = rng.random(n_ctrl)
        a = np.interp(k, np.arange(n_ctrl), ctrl)
    else:
        # Symmetric random
        half = n // 2
        a_half = 1.0 + 0.5 * rng.random(half + 1)
        a = np.zeros(n)
        a[:half + 1] = a_half
        a[half + 1:] = a_half[:max(0, n - half - 1)][::-1]

    a = np.maximum(a, 0)
    s = np.sum(a)
    if s < 1e-10:
        a = np.ones(n)
        s = n
    return a / s


def resample_sequence(a, new_n):
    """Resample a sequence to a new length."""
    n = len(a)
    if n == new_n:
        return a.copy()
    old_k = np.arange(n)
    new_k = np.linspace(0, n - 1, new_n)
    a_new = np.interp(new_k, old_k, a)
    a_new = np.maximum(a_new, 0)
    s = np.sum(a_new)
    if s > 1e-10:
        a_new = a_new / s
    return a_new

def intensive_refine(a, time_budget=60.0):
    """Multi-stage intensive refinement."""
    n = len(a)
    a = a.copy().astype(np.float64)
    s = np.sum(a)
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    # Stage 1: Nesterov smooth
    remaining = time_budget - (time.monotonic() - start)
    if remaining > 10:
        t = min(remaining * 0.35, 30.0)
        a_new = nesterov_smooth_optimize(best_a, time_budget=t,
                                         tau_start=3.0, tau_end=80.0,
                                         eta_start=0.12, eta_end=0.005,
                                         momentum_start=0.4, patience=300)
        obj = compute_obj(a_new)
        if obj < best_obj:
            best_obj = obj
            best_a = a_new.copy()

    # Stage 2: Peak targeting
    remaining = time_budget - (time.monotonic() - start)
    if remaining > 8:
        t = min(remaining * 0.25, 20.0)
        a_new = peak_targeting_search(best_a, time_budget=t)
        obj = compute_obj(a_new)
        if obj < best_obj:
            best_obj = obj
            best_a = a_new.copy()

    # Stage 3: LP hybrid
    remaining = time_budget - (time.monotonic() - start)
    if remaining > 10:
        t = min(remaining * 0.3, 30.0)
        a_new = lp_smooth_hybrid(best_a, time_budget=t, lp_every=10)
        obj = compute_obj(a_new)
        if obj < best_obj:
            best_obj = obj
            best_a = a_new.copy()

    # Stage 4: Final Nesterov polish
    remaining = time_budget - (time.monotonic() - start)
    if remaining > 5:
        t = min(remaining * 0.25, 15.0)
        a_new = nesterov_smooth_optimize(best_a, time_budget=t,
                                         tau_start=20.0, tau_end=200.0,
                                         eta_start=0.03, eta_end=0.002,
                                         momentum_start=0.7, patience=200)
        obj = compute_obj(a_new)
        if obj < best_obj:
            best_obj = obj
            best_a = a_new.copy()

    # Stage 5: Final peak targeting
    remaining = time_budget - (time.monotonic() - start)
    if remaining > 5:
        t = min(remaining * 0.3, 10.0)
        a_new = peak_targeting_search(best_a, time_budget=t)
        obj = compute_obj(a_new)
        if obj < best_obj:
            best_obj = obj
            best_a = a_new.copy()

    return best_a


def propose_candidate(seed=42, budget_s=1500):
    """
    Main search: combines Nesterov smooth optimization, peak-targeting search,
    LP alternation, and systematic n-exploration.
    """
    start_time = time.monotonic()
    deadline = start_time + float(budget_s) - 25.0

    best_seq = None
    best_score = np.inf

    # Load parent
    parent, parent_obj = try_parent()
    if parent is not None and parent_obj < best_score:
        best_score = parent_obj
        best_seq = parent.copy()
        n_parent = len(parent)
    else:
        n_parent = 330

    obj_parent = best_score
    print(f"Parent: n={n_parent}, obj={obj_parent:.6f}")
    print(f"Time budget: {budget_s}s, deadline in {deadline - time.monotonic():.0f}s")

    # Phase 1: Quick n-screening (find best n in a narrow range)
    print("\n=== Phase 1: Quick n-screening ===")
    n_candidates = [n_parent]
    for delta in [-5, -10, -15, -20, -25, -30, 5, 10, 15, 20, 25, 30]:
        nv = n_parent + delta
        if 150 <= nv <= 600 and nv != n_parent:
            n_candidates.append(nv)

    remaining = deadline - time.monotonic()
    screen_budget = min(remaining * 0.12, 120.0)
    per_screen = screen_budget / len(n_candidates)
    best_n = n_parent
    best_n_score = best_score
    best_n_seq = best_seq.copy() if best_seq is not None else None

    for nv in sorted(n_candidates):
        if time.monotonic() > deadline - 60:
            break
        if nv == n_parent:
            continue

        # Quick optimization: just Nesterov smooth with short budget
        parent_resampled = resample_sequence(parent, nv) if parent is not None else random_init(nv, seed + nv)
        t_scr = max(3.0, min(per_screen, 10.0))
        a_opt = nesterov_smooth_optimize(parent_resampled, time_budget=t_scr,
                                         tau_start=5.0, tau_end=40.0,
                                         eta_start=0.1, eta_end=0.02,
                                         momentum_start=0.5, patience=100)
        obj = compute_obj(a_opt)
        if obj < best_n_score:
            best_n_score = obj
            best_n = nv
            best_n_seq = a_opt.copy()
            print(f"  n={nv}: {obj:.6f} (improved)")
        else:
            print(f"  n={nv}: {obj:.6f}")

    if best_n_seq is not None and best_n_score < best_score:
        best_score = best_n_score
        best_seq = best_n_seq.copy()
        print(f"  Best n from screening: {best_n} with obj={best_score:.6f}")

    # Phase 2: Intensive optimization on best sequence
    print(f"\n=== Phase 2: Intensive optimization (n={len(best_seq)}) ===")
    remaining = deadline - time.monotonic()
    p2_budget = min(remaining * 0.45, 600.0)

    a_opt = intensive_refine(best_seq, time_budget=p2_budget)
    obj = compute_obj(a_opt)
    if obj < best_score:
        best_score = obj
        best_seq = a_opt.copy()
        print(f"  After intensive_refine: {best_score:.6f}")

    # Phase 3: LP-focused optimization
    print(f"\n=== Phase 3: LP-focused optimization ===")
    remaining = deadline - time.monotonic()
    if remaining > 30:
        p3_budget = min(remaining * 0.25, 300.0)
        a_opt = lp_smooth_hybrid(best_seq, time_budget=p3_budget, lp_every=8)
        obj = compute_obj(a_opt)
        if obj < best_score:
            best_score = obj
            best_seq = a_opt.copy()
            print(f"  After LP hybrid: {best_score:.6f}")

    # Phase 4: Peak-targeting intensive local search
    print(f"\n=== Phase 4: Peak-targeting local search ===")
    remaining = deadline - time.monotonic()
    if remaining > 20:
        p4_budget = min(remaining * 0.2, 200.0)
        a_opt = peak_targeting_search(best_seq, time_budget=p4_budget)
        obj = compute_obj(a_opt)
        if obj < best_score:
            best_score = obj
            best_seq = a_opt.copy()
            print(f"  After peak search: {best_score:.6f}")

    # Phase 5: Random restarts from best (to escape local minima)
    print(f"\n=== Phase 5: Informed restarts ===")
    remaining = deadline - time.monotonic()
    if remaining > 30:
        n = len(best_seq)
        n_restarts = max(2, int(remaining / 40))
        n_restarts = min(n_restarts, 8)
        per_restart = remaining / n_restarts * 0.7

        for r in range(n_restarts):
            if time.monotonic() > deadline - 20:
                break
            remaining = deadline - time.monotonic()
            if remaining < 20:
                break

            rng = np.random.default_rng(seed + r * 999983 + n)
            scale = 0.05 + 0.05 * (r % 3)
            a_pert = best_seq * np.exp(rng.normal(0, scale, n))
            a_pert = np.maximum(a_pert, 1e-10)
            s_pert = np.sum(a_pert)
            if s_pert > 1e-10:
                a_pert = a_pert / s_pert

            t_r = min(per_restart, remaining * 0.4)
            a_opt = intensive_refine(a_pert, time_budget=max(5.0, t_r))
            obj = compute_obj(a_opt)
            if obj < best_score:
                best_score = obj
                best_seq = a_opt.copy()
                print(f"  Restart {r}: {best_score:.6f} (improved!)")
            else:
                print(f"  Restart {r}: {obj:.6f}")

    # Phase 6: Final micro-refinement
    print(f"\n=== Phase 6: Final micro-refinement ===")
    remaining = deadline - time.monotonic()
    if remaining > 10:
        p6_budget = min(remaining * 0.4, remaining - 15.0)
        a_opt = nesterov_smooth_optimize(best_seq, time_budget=p6_budget,
                                         tau_start=50.0, tau_end=500.0,
                                         eta_start=0.01, eta_end=0.001,
                                         momentum_start=0.8, patience=500)
        obj = compute_obj(a_opt)
        if obj < best_score:
            best_score = obj
            best_seq = a_opt.copy()
            print(f"  After micro-refine: {best_score:.6f}")

        # One more peak targeting
        remaining = deadline - time.monotonic()
        if remaining > 5:
            a_opt = peak_targeting_search(best_seq, time_budget=min(5.0, remaining - 10.0))
            obj = compute_obj(a_opt)
            if obj < best_score:
                best_score = obj
                best_seq = a_opt.copy()
                print(f"  After final peak: {best_score:.6f}")

    # Final validation
    if best_seq is None:
        best_seq = np.ones(50)
        best_score = compute_obj(best_seq)

    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    val = evaluate_sequence(result)
    if val < best_score:
        best_score = val

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(best_seq)}, score={best_score:.6f}, time={elapsed:.1f}s ===")

    return result
