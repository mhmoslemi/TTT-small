"""
Minimizes J(a) = 2*n*max(convolve(a,a))/(sum(a))^2 using:
1. Adaptive LP best-response with binary-search tightening (core algorithm)
2. FFT-vectorized subgradient polish
3. Multi-length search (256, 512) with diverse initializations
4. Long LP polish phase with progressively tighter constraints
5. Simulated annealing with LP refinement for escape
"""

import numpy as np
from scipy.optimize import linprog
from scipy.sparse import csr_matrix
import time

# Precomputed sparsity cache
_spar_cache = {}

def get_sparsity(n):
    if n not in _spar_cache:
        rows = []
        cols = []
        for k in range(2 * n - 1):
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            rows.extend([k] * (je - js + 1))
            cols.extend(range(js, je + 1))
        _spar_cache[n] = (np.array(rows), np.array(cols), 2 * n - 1, n)
    return _spar_cache[n]

def fast_conv(a):
    n = len(a)
    s = 1
    while s < 2 * n - 1:
        s <<= 1
    fa = np.fft.rfft(a, s)
    return np.fft.irfft(fa * fa, s)[:2 * n - 1]

def compute_J(a):
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))

def build_lp_matrix(a, n):
    """Build the sparse constraint matrix for the LP."""
    rows, cols, nrows, ncols = get_sparsity(n)
    nnz = len(rows)
    vals = np.empty(nnz)
    for i in range(nnz):
        k = rows[i]
        j = cols[i]
        idx = k - j
        if 0 <= idx < n:
            vals[i] = a[idx]
        else:
            vals[i] = 0.0
    A = csr_matrix((vals, (rows, cols)), shape=(nrows, ncols))
    return A

def solve_lp(a, M, n, time_limit=15):
    """Solve the LP best-response: max sum(g) s.t. A*g <= M, g >= 0."""
    A = build_lp_matrix(a, n)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                options={'presolve': True, 'time_limit': time_limit})
    if r.success and r.x is not None:
        return np.maximum(r.x, 0.0)
    return None

def lp_best_response_adaptive(a, M, n, min_tighten=0.001, max_tighten=0.15, time_limit=15):
    """Adaptive LP: binary search on tightening factor to find best feasible M_tight."""
    # First try current M (should always be feasible since a itself is feasible)
    g = solve_lp(a, M, n, time_limit)
    if g is None or g.sum() < 1e-10:
        return None, 0.0

    base_sum = g.sum()

    # Try increasingly tighter M
    best_g = g.copy()
    best_tighten = 0.0
    lo = 0.0
    hi = max_tighten

    for _ in range(8):
        mid = (lo + hi) / 2.0
        if mid <= min_tighten:
            break
        M_tight = M * (1.0 - mid)
        g_tight = solve_lp(a, M_tight, n, time_limit)
        if g_tight is not None and g_tight.sum() > base_sum * 0.5:
            best_g = g_tight
            best_tighten = mid
            lo = mid
        else:
            hi = mid

    return best_g, best_tighten

def run_lp_opt(a, max_iters, deadline, tracker, time_limit=12):
    """Iterative LP best-response with adaptive tightening."""
    a = np.maximum(a.astype(np.float64), 1e-8)
    a /= a.sum()
    n = len(a)
    current_J = compute_J(a)
    tracker.update(a)

    tighten_history = []
    stall_count = 0

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        b = fast_conv(a)
        M = float(b.max())

        g, tighten_used = lp_best_response_adaptive(a, M, n, time_limit=time_limit)
        if g is None:
            break
        gs = g.sum()
        if gs < 1e-10:
            break
        g /= gs

        # Try multiple mixing ratios
        best_a = None
        best_J = current_J
        progress = it / max(1, max_iters)
        # Start with larger steps, decrease over time
        base_t = max(0.1, 0.8 * (1.0 - progress) ** 0.5)
        for t in [base_t, base_t * 0.5, base_t * 1.5, 0.3, 0.5, 0.7]:
            if t > 1.0 or t < 0.01:
                continue
            a_trial = (1.0 - t) * a + t * g
            a_trial = np.maximum(a_trial, 0.0)
            ss = a_trial.sum()
            if ss < 1e-10:
                continue
            a_trial /= ss
            J_trial = compute_J(a_trial)
            if J_trial < best_J - 1e-14:
                best_J = J_trial
                best_a = a_trial

        if best_a is None:
            stall_count += 1
            if stall_count > 15:
                break
            continue

        improvement = current_J - best_J
        if improvement < 1e-8:
            stall_count += 1
            if stall_count > 20:
                break
        else:
            stall_count = 0

        a = best_a
        current_J = best_J
        tighten_history.append(tighten_used)
        if it % 10 == 0:
            tracker.update(a)

    tracker.update(a)
    return a

def compute_subgrad_fft(a, active_indices):
    """Compute subgradient of max_k (a*a)_k w.r.t. a using FFT."""
    n = len(a)
    m = len(active_indices)
    if m == 0:
        return np.zeros(n)

    c = np.zeros(2 * n - 1)
    c[active_indices] = 1.0

    # We need: grad[m] = (2/|S|) * sum_{l=0}^{n-1} c[m+l]*a[l]
    # This equals (2/|S|) * (a_rev * c)[m + n - 1] where a_rev = a[::-1]
    a_rev = a[::-1]
    size = 1
    while size < len(a_rev) + len(c) - 1:
        size <<= 1
    f_ar = np.fft.rfft(a_rev, size)
    f_c = np.fft.rfft(c, size)
    conv = np.fft.irfft(f_ar * f_c, size)
    grad = (2.0 / m) * conv[n - 1:n - 1 + n]
    return grad

def subgradient_descent(a, max_iters, deadline, tracker):
    """Subgradient descent on the exact max of convolution."""
    a = np.maximum(a.astype(np.float64), 1e-8)
    a /= a.sum()
    n = len(a)
    current_J = compute_J(a)
    tracker.update(a)
    lr = 2.0 / n

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        b = fast_conv(a)
        M = float(b.max())
        active = np.where(b >= M - 1e-10 * M)[0]
        grad = compute_subgrad_fft(a, active)

        # Direction: minimize max(b), which means decreasing grad
        # Also account for the normalization: dJ/da_j = 2n*grad_j/s^2 - 4n*M*a_j/s^3
        # With s=1: direction = grad - 2*M*a
        direction = grad - 2.0 * M * a
        a_new = a - lr * direction
        a_new = np.maximum(a_new, 1e-8)
        a_new /= a_new.sum()
        J_new = compute_J(a_new)
        if J_new < current_J - 1e-15:
            a = a_new
            current_J = J_new
            lr = min(lr * 1.02, 10.0 / n)
        else:
            lr *= 0.7
            if lr < 1e-10:
                break
        if it % 20 == 0:
            tracker.update(a)
    tracker.update(a)
    return a

def flat_spectrum_init(n, rng):
    """Generate sequence with approximately flat power spectrum."""
    half = n // 2 + 1
    angles = rng.uniform(0, 2 * np.pi, half)
    A = np.zeros(n, dtype=complex)
    A[:half] = np.exp(1j * angles[:half])
    for k in range(1, min(n - n // 2, n)):
        A[k] = np.conj(A[n - k])
    a = np.fft.ifft(A).real
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-10:
        return np.ones(n) / n
    return a / s

def resample_seq(src, dest_n):
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)

def generate_inits(n, rng, parent):
    """Generate diverse initializations."""
    inits = []
    if parent is not None and parent.size > 0:
        inits.append(resample_seq(parent, n))
        inits.append(resample_seq(parent[::-1], n))

    inits.append(np.ones(n) / n)
    inits.append(rng.uniform(0.1, 1.0, n) / (n * 0.55))
    inits.append(flat_spectrum_init(n, rng))
    inits.append(flat_spectrum_init(n, rng))
    inits.append(flat_spectrum_init(n, rng))

    t = np.linspace(0, 1, n, endpoint=False)
    # Cosine modulations
    for freq, amp in [(2, 0.3), (3, 0.35), (5, 0.25), (7, 0.2), (1, 0.4)]:
        inits.append(1.0 + amp * np.cos(2 * np.pi * freq * t + rng.uniform(0, 6.28)))

    # Multi-frequency
    inits.append(1.0 + 0.3 * np.sin(2 * np.pi * t) + 0.2 * np.sin(4 * np.pi * t)
                + 0.15 * np.sin(7 * np.pi * t) + 0.1 * np.sin(11 * np.pi * t))

    # Gaussian bumps
    inits.append(0.5 * np.exp(-8 * (t - 0.3) ** 2) + 0.5 * np.exp(-8 * (t - 0.7) ** 2) + 0.05)
    inits.append(0.4 * np.exp(-5 * (t - 0.25) ** 2) + 0.4 * np.exp(-5 * (t - 0.5) ** 2)
                + 0.3 * np.exp(-5 * (t - 0.75) ** 2) + 0.03)

    # Triangle-like
    inits.append(0.3 + 0.7 * (1.0 - np.abs(2 * t - 1.0)) ** 0.5)

    # Power-law
    inits.append((t + 0.1) ** 0.5)
    inits.append((1.0 - t + 0.1) ** 0.5)
    inits.append(1.0 / (t + 0.5))

    # Sparse + smooth
    sparse = np.zeros(n)
    n_spikes = max(5, n // 15)
    positions = rng.choice(n, size=n_spikes, replace=False)
    sparse[positions] = 1.0
    kernel_size = max(3, n // 25)
    kernel = np.ones(kernel_size) / kernel_size
    inits.append(np.convolve(sparse, kernel, mode='same') + 0.05)

    # Random limited-weight
    n_coeff = min(n // 2 + 1, 15)
    coeffs = rng.normal(0, 1, n_coeff)
    sig = np.zeros(n)
    for i in range(n_coeff):
        sig += coeffs[i] * np.cos(2 * np.pi * i * t / n + rng.uniform(0, 6.28))
    inits.append(np.abs(sig) + 0.5)

    for x in inits:
        x = np.maximum(x, 0.001).astype(np.float64)
        s = x.sum()
        if s > 0:
            x /= s
    return inits

class ScoreTracker:
    def __init__(self):
        self.best = None
        self.best_score = float('inf')
        self.t_start = time.monotonic()

    def update(self, a):

    def report(self, label=""):

def simulated_annealing_lp(a, deadline, tracker, n_restarts=30, t0=0.12, time_limit=8):
    """Simulated annealing where each step is an LP refinement."""
    rng = np.random.default_rng(777)
    n = a.size
    a = a.copy().astype(np.float64)
    current_J = compute_J(a)
    tracker.update(a)
    temp = t0

    for r in range(n_restarts):
        if time.monotonic() > deadline:
            break
        a_pert = a + rng.normal(0, temp, n)
        a_pert = np.maximum(a_pert, 1e-6)
        a_pert /= a_pert.sum()

        # Short LP refinement
        a_refined = run_lp_opt(a_pert, max(3, 15 - r // 5), deadline, tracker, time_limit=time_limit)
        J = compute_J(a_refined)
        delta = J - current_J
        if delta < 0 or rng.random() < np.exp(-delta / max(temp, 1e-10)):
            a = a_refined
            current_J = J
            tracker.update(a)
        temp *= 0.93

    tracker.update(a)
    return a
