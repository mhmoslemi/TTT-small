import numpy as np
from scipy.optimize import minimize, linprog
from scipy.fft import rfft, irfft
import time


def compute_conv(a):
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    conv = irfft(fa * fa, size)
    return conv[:2 * n - 1]


def compute_obj(a):
    n = len(a)
    if n == 0:
        return np.inf
    conv = compute_conv(a)
    max_b = float(np.max(conv))
    sum_a = float(np.sum(a))
    if sum_a < 0.01:
        return np.inf
    return 2.0 * n * max_b / (sum_a * sum_a)


def fft_convolve(a, b):
    n, m = len(a), len(b)
    size = 1
    while size < n + m - 1:
        size <<= 1
    fa = rfft(a, size)
    fb = rfft(b, size)
    return irfft(fa * fb, size)[:n + m - 1]


def softmax(z):
    z = z - np.max(z)
    e = np.exp(z)
    return e / np.sum(e)

    """Project v onto {x >= 0, sum(x) = target_sum}."""
    n = len(v)
    # Scale to work on sum=1 simplex, then rescale
    u = v / max(v.sum(), 1e-30)

    w = np.maximum(u - theta, 0)
    return w * target_sum

"""
Improved hybrid optimizer for minimizing 2*n*max(conv(a,a))/(sum(a))^2.
Uses Nesterov momentum subgradient with Polyak step size, LP alternating polishing,
and L-BFGS-B refinement with softmax parametrization. Explores sequence lengths
400-1500 with window-based and parent-based initializations.

import numpy as np
from scipy.optimize import minimize, linprog
from scipy.fft import rfft, irfft
import time


def compute_conv(a):
    """Compute linear convolution a*a using FFT."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    conv = irfft(fa * fa, size)
    return conv[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    conv = compute_conv(a)
    max_b = float(np.max(conv))
    sum_a = float(np.sum(a))
    if sum_a < 0.01:
        return np.inf
    return 2.0 * n * max_b / (sum_a * sum_a)


def fft_convolve(a, b):
    """Linear convolution of a and b using FFT."""
    n, m = len(a), len(b)
    size = 1
    while size < n + m - 1:
        size <<= 1
    fa = rfft(a, size)
    fb = rfft(b, size)
    return irfft(fa * fb, size)[:n + m - 1]


def softmax_fn(z):
    """Numerically stable softmax."""
    z = z - np.max(z)
    e = np.exp(z)
    return e / np.sum(e)


def proj_simplex(v, target_sum=1.0):
    """Project v onto {x >= 0, sum(x) = target_sum}."""
    n = len(v)
    u = v / max(float(np.sum(v)), 1e-30)
    d = np.sort(u)[::-1]
    cssv = np.cumsum(d)
    cond = d * np.arange(1, n + 1) > (cssv - 1)
    idx = np.nonzero(cond)[0]
    if len(idx) == 0:
        w = np.ones(n) / n
    else:
        rho = idx[-1]
        theta = (cssv[rho] - 1) / (rho + 1.0)
        w = np.maximum(u - theta, 0)
    s = np.sum(w)
    if s < 1e-30:
        w = np.ones(n) / n
    else:
        w = w / s
    return w * target_sum


def compute_subgradient(a, tau):
    """
    Compute smoothed subgradient of max_k conv(a,a)[k] w.r.t. a.
    Returns (gradient, max_conv_value, weights).
    """
    n = len(a)
    conv = compute_conv(a)
    cmax = float(np.max(conv))

    if tau < 1e-10:
        # Exact subgradient: uniform on argmax
        w = np.zeros(len(conv))
        active = np.where(conv >= cmax - 1e-12)[0]
        w[active] = 1.0 / len(active)
    else:
        shifted = np.clip((conv - cmax) / tau, -500.0, 500.0)
        exp_conv = np.exp(shifted)
        Z = float(np.sum(exp_conv))
        w = exp_conv / Z

    # grad_a[i] = 2 * sum_j w[i+j] * a[j]
    w_rev = w[::-1]
    conv_fw = fft_convolve(a, w_rev)
    grad_a = 2.0 * conv_fw[n - 1:2 * n - 1][::-1]

    return grad_a, cmax, w


def nesterov_subgradient(a_init, n_steps, time_budget, tau_init, tau_end,
                         step_init, lower_bound=0.0, momentum_init=0.7):
    """
    Nesterov-accelerated subgradient descent on the probability simplex.
    Uses Polyak step size and annealed tau.
    """
    n = len(a_init)
    a = a_init.copy()
    y = a.copy()  # Nesterov lookahead
    velocity = np.zeros(n)

    best_a = a.copy()
    best_max = float(np.max(compute_conv(a)))
    start = time.monotonic()

    for it in range(n_steps):
        if time.monotonic() - start > time_budget:
            break

        frac = it / max(1, n_steps)
        tau = tau_init + (tau_end - tau_init) * frac
        beta = momentum_init + (1.0 - momentum_init) * frac  # momentum increases

        # Evaluate at Nesterov point
        grad_y, cmax_y, w_y = compute_subgradient(y, tau)

        # Gradient of the smoothed objective at y
        # For Polyak step, we need f(y) - f*
        current_val = cmax_y  # approx smoothed max
        gap = max(current_val - lower_bound, 1e-10)

        # Gradient norm
        grad_norm_sq = float(np.sum(grad_y * grad_y))
        if grad_norm_sq < 1e-30:
            break
        step = step_init * gap / np.sqrt(grad_norm_sq)
        step = min(step, 2.0 * gap)  # cap step size

        # Update velocity
        velocity = beta * velocity - step * grad_y

        # Update y (Nesterov lookahead)
        y_new = y + velocity

        # Project to simplex
        a_new = proj_simplex(y_new, 1.0)

        # Nesterov: y = a_new + (beta_new/beta_old) * (a_new - a)
        a = a_new

        # Evaluate and track best
        if it % 20 == 0 or it == n_steps - 1:
            true_max = float(np.max(compute_conv(a)))
            if true_max < best_max:
                best_max = true_max
                best_a = a.copy()

    return best_a


def build_toeplitz_sparse_info(a):
    """Return the Toeplitz matrix for conv(a, g)."""
    n = len(a)
    A = np.zeros((2 * n - 1, n), dtype=np.float64)
    for j in range(n):
        k_end = min(n, 2 * n - 1 - j)
        A[j:j + k_end, j] = a[:k_end]
    return A


def solve_primal_lp(a, M, time_limit=10):
    """Solve: max sum(g) s.t. T(a)*g <= M, g >= 0. Returns normalized g or None."""
    n = len(a)
    A = np.zeros((2 * n - 1, n), dtype=np.float64)
    for j in range(n):
        k_end = min(n, 2 * n - 1 - j)
        A[j:j + k_end, j] = a[:k_end]

    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': time_limit})
        if res.status == 0:
            g = np.maximum(res.x, 0)
            s = float(np.sum(g))
            if s > 1e-10:
                return g / s
    except Exception:
        pass
    return None


def solve_dual_lp(a, M, time_limit=10):
    """
    Solve dual: min M*sum(y) s.t. T(a)^T y >= 1, y >= 0.
    Returns optimal y or None.
    """
    n = len(a)
    A = np.zeros((2 * n - 1, n), dtype=np.float64)
    for j in range(n):
        k_end = min(n, 2 * n - 1 - j)
        A[j:j + k_end, j] = a[:k_end]

    # Dual: min M * sum(y) s.t. A^T y >= 1, y >= 0
    c = M * np.ones(2 * n - 1)
    A_ub = -A.T  # -A^T y <= -1
    b_ub = -np.ones(n)
    bounds = [(0.0, None)] * (2 * n - 1)
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': time_limit})
        if res.status == 0:
            y = np.maximum(res.x, 0)
            if np.sum(y) > 1e-15:
                return y
    except Exception:
        pass
    return None


def lp_alternating(a_init, max_iters=30, time_budget=30.0):
    """
    Alternating LP: fix a, solve for g, blend. Repeat.
    """
    n = len(a_init)
    a = a_init.copy()
    s = float(np.sum(a))
    if s < 1e-10:
        return a_init
    a = a / s

    best_a = a.copy()
    best_max = float(np.max(compute_conv(a)))
    start = time.monotonic()
    stall = 0

    for it in range(max_iters):
        if time.monotonic() - start > time_budget:
            break

        conv = compute_conv(a)
        M = float(np.max(conv))

        g = solve_primal_lp(a, M, time_limit=8)
        if g is None:
            break

        # Line search for blend factor
        best_t_obj = best_max
        best_a_new = best_a.copy()
        for t in [0.1, 0.2, 0.3, 0.5, 0.7, 1.0]:
            a_try = (1.0 - t) * a + t * g
            a_try = np.maximum(a_try, 0)
            s_try = float(np.sum(a_try))
            if s_try < 1e-10:
                continue
            a_try = a_try / s_try
            max_try = float(np.max(compute_conv(a_try)))
            if max_try < best_t_obj:
                best_t_obj = max_try
                best_a_new = a_try.copy()

        if best_t_obj < best_max - 1e-10:
            best_max = best_t_obj
            best_a = best_a_new.copy()
            a = best_a_new
            stall = 0
        else:
            stall += 1
            if stall > 15:
                # Perturb and restart
                rng = np.random.default_rng(it * 631 + 42)
                a = a * np.exp(rng.normal(0, 0.05, n))
                a = np.maximum(a, 0)
                s = float(np.sum(a))
                if s > 1e-10:
                    a = a / s
                stall = 0
            else:
                a = 0.8 * a + 0.2 * g
                a = np.maximum(a, 0)
                s = float(np.sum(a))
                if s > 1e-10:
                    a = a / s

    return best_a


def lfbs_refine(a_init, time_budget=20.0, tau_list=None):
    """
    L-BFGS-B refinement using log-parametrization and smoothed max.
    """
    n = len(a_init)
    a = a_init.copy()
    s = float(np.sum(a))
    if s < 1e-10:
        return a_init
    a = np.maximum(a, 1e-15)
    a = a / np.sum(a)
    z = np.log(a)

    if tau_list is None:
        tau_list = [5.0 / n, 2.0 / n, 1.0 / n, 0.5 / n, 0.25 / n, 0.1 / n]

    best_a = a.copy()
    best_max = float(np.max(compute_conv(a)))
    start = time.monotonic()

    for tau in tau_list:
        if time.monotonic() - start > time_budget:
            break

        def obj_and_grad(z_val):
            a_val = softmax_fn(z_val)
            conv = compute_conv(a_val)
            cmax = float(np.max(conv))
            shifted = np.clip((conv - cmax) / tau, -500.0, 500.0)
            exp_conv = np.exp(shifted)
            Z = float(np.sum(exp_conv))
            smax = cmax + tau * np.log(Z)
            w = exp_conv / Z

            # Gradient w.r.t. a
            w_rev = w[::-1]
            conv_fw = fft_convolve(a_val, w_rev)
            grad_a = 2.0 * conv_fw[n - 1:2 * n - 1][::-1]

            # Gradient w.r.t. z (softmax Jacobian)
            mean_g = float(np.dot(a_val, grad_a))
            grad_z = a_val * (grad_a - mean_g)

            return smax, grad_z

        try:
            result = minimize(obj_and_grad, z, jac=True, method='L-BFGS-B',
                              options={'maxiter': 200, 'ftol': 1e-14, 'gtol': 1e-9})
            z_new = result.x
        except Exception:
            continue

        a_new = softmax_fn(z_new)
        a_new = np.maximum(a_new, 0)
        s_new = float(np.sum(a_new))
        if s_new < 1e-10:
            continue
        a_new = a_new / s_new
        max_new = float(np.max(compute_conv(a_new)))

        if max_new < best_max:
            best_max = max_new
            best_a = a_new.copy()
            z = z_new

    return best_a


def init_window(n, window_type, seed=0):
    """Generate window-based initializations."""
    rng = np.random.default_rng(seed)
    k = np.arange(n)
    x = k / (n - 1) if n > 1 else np.array([0.5])

    if window_type == 0:
        # Hamming
        a = 0.54 - 0.46 * np.cos(2 * np.pi * x)
    elif window_type == 1:
        # Hann
        a = 0.5 - 0.5 * np.cos(2 * np.pi * x)
    elif window_type == 2:
        # Raised cosine with random params
        f = rng.uniform(0.5, 3.0)
        phase = rng.uniform(-np.pi, np.pi)
        a = 1.0 + rng.uniform(0.3, 0.8) * np.cos(2 * np.pi * f * x + phase)
    elif window_type == 3:
        # Multi-frequency
        a = np.ones(n)
        for _ in range(rng.integers(2, 5)):
            f = rng.uniform(0.3, 5.0)
            amp = rng.uniform(0.1, 0.5)
            a += amp * np.cos(2 * np.pi * f * x + rng.uniform(-np.pi, np.pi))
    elif window_type == 4:
        # Asymmetric bump
        center = rng.uniform(0.3, 0.7)
        width = rng.uniform(0.1, 0.4)
        a = 1.0 + rng.uniform(0.3, 1.0) * np.exp(-0.5 * ((x - center) / width) ** 2)
    elif window_type == 5:
        # Linear ramp with cosine
        a = (1.0 - x) + 0.5 * np.cos(2 * np.pi * rng.uniform(1, 3) * x)
    elif window_type == 6:
        # Quadratic
        c = rng.uniform(0.2, 0.8)
        w = rng.uniform(0.1, 0.4)
        a = 1.0 + rng.uniform(0.3, 0.8) * np.exp(-((x - c) / w) ** 2)
    elif window_type == 7:
        # Sinc-like (truncated)
        f = rng.uniform(1.0, 4.0)
        a = np.sinc(f * (x - 0.5))
        a = np.maximum(a, 0) + 0.1
    elif window_type == 8:
        # Polynomial
        coeff = rng.uniform(0.2, 1.0, 4)
        a = np.ones(n)
        for i, c in enumerate(coeff):
            a += c * (x - 0.5) ** (i + 1)
        a = np.maximum(a, 0.05)
    else:
        a = 1.0 + 0.3 * np.sin(2 * np.pi * rng.uniform(0.5, 3.0) * x + rng.uniform(0, 2 * np.pi))

    a = np.maximum(a, 0.01)
    return a / np.sum(a)


def init_random(n, seed):
    """Random initialization with smoothing."""
    rng = np.random.default_rng(seed)
    a = rng.random(n) + 0.1
    # Light smoothing
    win = max(3, n // 30)
    kernel = np.ones(win) / win
    a = np.convolve(a, kernel, mode='same')
    a = np.maximum(a, 0.01)
    return a / np.sum(a)


def resample_seq(a, new_n):
    """Resample to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    return np.interp(x_new, x_old, a)


def try_load_parent():
    """Try to load the parent sequence."""
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64)
        parent = np.clip(parent, 0, 1000.0)
        if parent.size > 0 and np.isfinite(parent).all():
            obj = compute_obj(parent)
            print(f"Parent loaded: n={parent.size}, obj={obj:.6f}")
            return parent.copy(), obj
    except Exception:
        pass
    return None, np.inf


def full_optimize(a_init, time_budget, use_lp=True, n_lp_max=800):
    """
    Run the full optimization pipeline on a single sequence.
    Returns the best sequence found.
    """
    n = len(a_init)
    a = a_init.copy()
    s = float(np.sum(a))
    if s < 1e-10:
        return a_init
    a = a / s

    best_a = a.copy()
    best_max = float(np.max(compute_conv(a)))
    start = time.monotonic()
    lower_bound = 1.0 / (2 * n - 1)

    # Phase 1: Nesterov subgradient (main exploration)
    remaining = time_budget - (time.monotonic() - start)
    if remaining > 5:
        sg_budget = min(remaining * 0.4, 40.0)
        tau_init = 10.0 / n
        tau_end = 0.2 / n
        a_sg = nesterov_subgradient(a, n_steps=2000, time_budget=sg_budget,
                                    tau_init=tau_init, tau_end=tau_end,
                                    step_init=0.02, lower_bound=lower_bound,
                                    momentum_init=0.5)
        max_sg = float(np.max(compute_conv(a_sg)))
        if max_sg < best_max:
            best_max = max_sg
            best_a = a_sg.copy()

    # Phase 2: Second subgradient pass with different params
    remaining = time_budget - (time.monotonic() - start)
    if remaining > 10:
        sg_budget = min(remaining * 0.25, 20.0)
        a_sg2 = nesterov_subgradient(best_a, n_steps=2000, time_budget=sg_budget,
                                     tau_init=5.0 / n, tau_end=0.1 / n,
                                     step_init=0.01, lower_bound=lower_bound,
                                     momentum_init=0.7)
        max_sg2 = float(np.max(compute_conv(a_sg2)))
        if max_sg2 < best_max:
            best_max = max_sg2
            best_a = a_sg2.copy()

    # Phase 3: LP alternating (if n is small enough)
    remaining = time_budget - (time.monotonic() - start)
    if use_lp and n <= n_lp_max and remaining > 15:
        lp_budget = min(remaining * 0.3, 30.0)
        a_lp = lp_alternating(best_a, max_iters=25, time_budget=lp_budget)
        max_lp = float(np.max(compute_conv(a_lp)))
        if max_lp < best_max:
            best_max = max_lp
            best_a = a_lp.copy()

    # Phase 4: L-BFGS-B refinement
    remaining = time_budget - (time.monotonic() - start)
    if remaining > 10:
        lbfgs_budget = min(remaining * 0.3, 20.0)
        tau_list = [3.0 / n, 1.0 / n, 0.5 / n, 0.2 / n, 0.1 / n]
        a_lfbs = lfbs_refine(best_a, time_budget=lbfgs_budget, tau_list=tau_list)
        max_lfbs = float(np.max(compute_conv(a_lfbs)))
        if max_lfbs < best_max:
            best_max = max_lfbs
            best_a = a_lfbs.copy()

    # Phase 5: Quick subgradient polish
    remaining = time_budget - (time.monotonic() - start)
    if remaining > 5:
        a_pol = nesterov_subgradient(best_a, n_steps=int(remaining * 30),
                                     time_budget=min(remaining * 0.5, 10.0),
                                     tau_init=2.0 / n, tau_end=0.05 / n,
                                     step_init=0.005, lower_bound=lower_bound,
                                     momentum_init=0.8)
        max_pol = float(np.max(compute_conv(a_pol)))
        if max_pol < best_max:
            best_max = max_pol
            best_a = a_pol.copy()

    return best_a


def propose_candidate():
    """
    Main search function. Explores multiple sequence lengths and initializations,
    using Nesterov momentum subgradient, LP alternating, and L-BFGS-B refinement.
    Returns the best sequence found.
    """
    start_time = time.monotonic()
    deadline = start_time + 1470.0  # Leave 30s buffer

    best_seq = None
    best_score = np.inf

    # Try to load parent
    parent, parent_obj = try_load_parent()
    if parent is not None and parent_obj < best_score:
        best_score = parent_obj
        best_seq = parent.copy()
        n_parent = len(parent)
    else:
        n_parent = 600

    print(f"Initial best: {best_score:.6f} (n={n_parent if best_seq is not None else 'N/A'})")

    # Determine n values to explore
    n_values = sorted(set([
        n_parent,
        max(300, n_parent - 100),
        n_parent + 100,
        n_parent + 200,
        max(400, n_parent // 2),
        n_parent + 400,
        n_parent + 500,
        1500,
        2000,
    ]))
    # Filter to reasonable range
    n_values = [n for n in n_values if 200 <= n <= 2500]
    n_values = sorted(set(n_values))

    print(f"Exploring n values: {n_values}")

    results = []  # (a, obj, n)

    # Phase 1: Explore different n values
    for n in n_values:
        if time.monotonic() > deadline - 120:
            break
        remaining = deadline - time.monotonic()
        if remaining < 30:
            break

        # Time allocation based on n
        if n <= 600:
            alloc = min(60.0, remaining * 0.12)
        elif n <= 1000:
            alloc = min(50.0, remaining * 0.1)
        else:
            alloc = min(35.0, remaining * 0.08)
        alloc = max(15.0, alloc)

        print(f"\nN={n}: {remaining:.0f}s left, alloc={alloc:.0f}s")

        # Generate multiple initializations
        n_inits = max(2, min(4, int(alloc / 18)))
        best_for_n = None
        best_for_n_max = np.inf

        for init_i in range(n_inits):
            if time.monotonic() > deadline - 100:
                break
            init_remaining = deadline - time.monotonic()
            if init_remaining < 15:
                break

            seed = n * 7919 + init_i * 31

            # Choose initialization
            if init_i == 0 and best_seq is not None:
                # Start from best known (resampled)
                a_init = resample_seq(best_seq, n)
            elif init_i == 1:
                a_init = init_window(n, 0, seed)  # Hamming
            elif init_i == 2:
                a_init = init_window(n, 2, seed)  # Cosine with random params
            elif init_i == 3:
                a_init = init_random(n, seed)
            else:
                a_init = init_window(n, init_i % 9, seed)

            # Optimize
            single_budget = alloc / n_inits
            a_opt = full_optimize(a_init, time_budget=single_budget,
                                  use_lp=True, n_lp_max=800)

            obj = compute_obj(a_opt)
            max_val = float(np.max(compute_conv(a_opt)))
            print(f"  init{init_i}: n={n}, obj={obj:.6f}, max_conv={max_val:.8f}")

            if obj < best_for_n_max:
                best_for_n_max = obj
                best_for_n = a_opt.copy()

            if obj < best_score:
                best_score = obj
                best_seq = a_opt.copy()
                print(f"  ** NEW BEST: n={n}, obj={best_score:.6f} **")

        if best_for_n is not None:
            results.append((best_for_n, best_for_n_max, n))

    # Phase 2: Polish the top candidates
    print(f"\n=== POLISHING PHASE ===")
    print(f"Best before polish: {best_score:.6f}")

    results.sort(key=lambda x: x[1])
    n_polish = min(4, len(results))

    for pi in range(n_polish):
        if time.monotonic() > deadline - 60:
            break
        remaining = deadline - time.monotonic()
        if remaining < 30:
            break

        a_cand, obj_cand, n_cand = results[pi]
        print(f"  Polishing cand {pi} (n={n_cand}, obj={obj_cand:.6f}), {remaining:.0f}s left")

        polish_budget = min(remaining * 0.3, 60.0)

        # Intensive LP + subgradient
        a_pol = full_optimize(a_cand, time_budget=polish_budget,
                              use_lp=True, n_lp_max=900)
        obj_pol = compute_obj(a_pol)
        if obj_pol < best_score:
            best_score = obj_pol
            best_seq = a_pol.copy()
            print(f"  ** AFTER POLISH: obj={best_score:.6f} **")

        # Try perturbation + re-optimize
        remaining = deadline - time.monotonic()
        if remaining > 30:
            rng = np.random.default_rng(pi * 997 + 13)
            for ptry in range(2):
                if time.monotonic() > deadline - 40:
                    break
                scale = rng.uniform(0.02, 0.08)
                a_pert = a_cand * np.exp(rng.normal(0, scale, n_cand))
                a_pert = np.maximum(a_pert, 0)
                s = float(np.sum(a_pert))
                if s < 1e-10:
                    continue
                a_pert = a_pert / s

                pert_budget = min(remaining * 0.2, 20.0)
                a_pert_opt = full_optimize(a_pert, time_budget=pert_budget,
                                           use_lp=True, n_lp_max=800)
                obj_pert = compute_obj(a_pert_opt)
                if obj_pert < best_score:
                    best_score = obj_pert
                    best_seq = a_pert_opt.copy()
                    print(f"  ** PERTURB {ptry}: obj={best_score:.6f} **")

    # Phase 3: Final intensive polish on global best
    if best_seq is not None and time.monotonic() < deadline - 30:
        remaining = deadline - time.monotonic()
        n_best = len(best_seq)
        print(f"\n=== FINAL POLISH: n={n_best}, {remaining:.0f}s ===")

        a_cur = best_seq.copy()
        for fr in range(3):
            if time.monotonic() > deadline - 15:
                break
            remaining = deadline - time.monotonic()
            if remaining < 15:
                break

            # LP round
            if n_best <= 900:
                a_lp = lp_alternating(a_cur, max_iters=20, time_budget=min(remaining * 0.3, 25.0))
                obj_lp = compute_obj(a_lp)
                if obj_lp < best_score:
                    best_score = obj_lp
                    best_seq = a_lp.copy()
                    a_cur = a_lp.copy()
                    print(f"  Final LP {fr}: {best_score:.6f}")

            # L-BFGS-B with very small tau
            remaining = deadline - time.monotonic()
            if remaining > 10:
                tau_final = [1.0 / n_best, 0.5 / n_best, 0.2 / n_best, 0.1 / n_best]
                a_lfbs = lfbs_refine(a_cur, time_budget=min(remaining * 0.3, 15.0), tau_list=tau_final)
                obj_lfbs = compute_obj(a_lfbs)
                if obj_lfbs < best_score:
                    best_score = obj_lfbs
                    best_seq = a_lfbs.copy()
                    a_cur = a_lfbs.copy()
                    print(f"  Final L-BFGS {fr}: {best_score:.6f}")

            # Subgradient with very small tau
            remaining = deadline - time.monotonic()
            if remaining > 8:
                a_sg = nesterov_subgradient(a_cur, n_steps=int(remaining * 20),
                                            time_budget=min(remaining * 0.3, 12.0),
                                            tau_init=0.5 / n_best, tau_end=0.01 / n_best,
                                            step_init=0.003,
                                            lower_bound=1.0 / (2 * n_best - 1),
                                            momentum_init=0.9)
                obj_sg = compute_obj(a_sg)
                if obj_sg < best_score:
                    best_score = obj_sg
                    best_seq = a_sg.copy()
                    a_cur = a_sg.copy()
                    print(f"  Final SG {fr}: {best_score:.6f}")

    # Fallback
    if best_seq is None:
        best_seq = np.ones(200)
        best_score = compute_obj(best_seq)

    # Final validation
    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    val = evaluate_sequence(result)
    if val < np.inf:
        if val < best_score:
            best_score = val
    else:
        result = [1.0] * 100
        best_score = evaluate_sequence(result)

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_score:.6f}, time={elapsed:.1f}s ===")

    return result
