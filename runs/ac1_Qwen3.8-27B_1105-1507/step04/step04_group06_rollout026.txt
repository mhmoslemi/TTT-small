"""
Minimizes J(a) = 2n*max(convolve(a,a))/(sum(a))² using:
1. Fast FISTA with adaptive softmax smoothing (β annealing)
2. Reduced-constraint LP mirror descent (top-K active constraints only)
3. Multi-length exploration with diverse initializations
4. Nesterov acceleration with adaptive restarts
5. Aggressive perturbation-based restarts at stall
"""

import numpy as np
from scipy.optimize import linprog
from scipy.sparse import lil_matrix, csr_matrix
import time

# ============================================================
# Core computation functions
# ============================================================

def fast_conv(a):
    """Full self-convolution via FFT."""
    n = len(a)
    s = 1
    while s < 2 * n - 1:
        s <<= 1
    fa = np.fft.rfft(a, s)
    return np.fft.irfft(fa * fa, s)[:2 * n - 1]


def compute_J(a):
    """Compute J(a) = 2*n*max(conv(a,a))/(sum(a))^2."""
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))


def smooth_max_and_grad(a, beta):
    """
    Compute smooth approximation of max(conv(a,a)) and its gradient.
    Uses softmax with temperature 1/beta.
    Returns (smooth_value, gradient_array).
    """
    n = len(a)
    b = fast_conv(a)
    b_max = float(b.max())
    shifted = beta * (b - b_max)
    shifted = np.clip(shifted, -500, 500)
    exp_shifted = np.exp(shifted)
    Z = float(exp_shifted.sum())
    if Z < 1e-300:
        p = np.zeros(2 * n - 1)
        p[np.argmax(b)] = 1.0
    else:
        p = exp_shifted / Z
    m_val = b_max + np.log(max(Z, 1e-300)) / beta
    
    # Gradient: ∇m[i] = 2 * Σ_d a[d] * p[i+d]
    # = 2 * np.correlate(p, a, 'valid')
    grad_m = 2.0 * np.correlate(p, a, 'valid')
    
    return m_val, grad_m, p, b


def F_and_grad(a, beta):
    """Compute F(a) = 2n*m_β(a)/(sum(a))^2 and gradient."""
    n = len(a)
    s = float(a.sum())
    if s < 1e-10:
        return 1e10, np.zeros(n)
    m_val, grad_m, p, b = smooth_max_and_grad(a, beta)
    F_val = 2.0 * n * m_val / (s * s)
    # ∇F = (2n/s^3) * (s * grad_m - 2*m_val * 1)
    grad_F = (2.0 * n / (s ** 3)) * (s * grad_m - 2.0 * m_val)
    return F_val, grad_F

# FISTA optimizer with adaptive restarts
# ============================================================

def fista_optimize(a0, beta_schedule, deadline, n_restarts=3):
    """
    FISTA (Fast Iterative Shrinkage-Thresholding Algorithm) with:
    - Backtracking line search
    - Nesterov momentum with restarts
    - Beta annealing schedule
    """
    a = np.maximum(a0.astype(np.float64), 1e-8)
    n = len(a)
    
    for beta, max_iters in beta_schedule:
        if time.monotonic() > deadline:
            break
        
        x = a.copy()
        y = a.copy()
        t_k = 1.0
        L = 50.0  # Initial Lipschitz estimate
        best_F = compute_J(a)
        best_a = a.copy()
        
        last_improve_iter = 0
        restart_count = 0
        
        for it in range(max_iters):
            if time.monotonic() > deadline:
                break
            
            # Compute gradient at y
            F_y, grad_y = F_and_grad(np.maximum(y, 0), beta)
            if not np.isfinite(F_y):
                break
            
            # Gradient step
            x_new = y - grad_y / L
            x_new = np.maximum(x_new, 0)  # Project onto non-negative orthant
            
            # Backtracking line search
            c1 = 1e-4
            while True:
                F_x, _ = F_and_grad(x_new, beta)
                if not np.isfinite(F_x):
                    L *= 2
                    x_new = y - grad_y / L
                    x_new = np.maximum(x_new, 0)
                    continue
                F_pred = F_y + grad_y @ (x_new - y) + 0.5 * L * np.sum((x_new - y)**2)
                if F_x <= F_pred + 1e-12:
                    break
                L *= 2
                x_new = y - grad_y / L
                x_new = np.maximum(x_new, 0)
            
            # Nesterov update
            t_new = (1.0 + np.sqrt(1.0 + 4.0 * t_k**2)) / 2.0
            
            # Restart check: if distance from y to x_new is large, reset momentum
            if restart_count >= 10 and it - last_improve_iter > 50:
                t_new = 1.0
                y = x_new.copy()
                restart_count += 1
            else:
                y = x_new + (t_k / t_new - 1.0) * (x_new - x)
            
            t_k = t_new
            
            # Check improvement
            F_true = compute_J(x_new)
            if F_true < best_F - 1e-12:
                best_F = F_true
                best_a = x_new.copy()
                last_improve_iter = it
            
            a = x_new
        
        a = best_a
    
    return a

def lp_best_response_reduced(a, M, K=30):
    """
    LP: max sum(g) s.t. conv(a,g)[k] <= M for top-K k, g >= 0.
    Returns normalized g or None.
    """
    n = len(a)
    b = fast_conv(a)
    top_k = np.argsort(b)[-K:]
    
    # Build reduced constraint matrix
    rows_list = []
    cols_list = []
    vals_list = []
    for idx, k in enumerate(top_k):
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        for j in range(j_start, j_end + 1):
            rows_list.append(idx)
            cols_list.append(j)
            vals_list.append(a[k - j])
    
    A_red = lil_matrix((K, n))
    for r, c, v in zip(rows_list, cols_list, vals_list):
        A_red[r, c] = v
    A_red = A_red.tocsr()
    
    c_obj = -np.ones(n)
    b_ub = np.full(K, M)
    bounds = [(0.0, None)] * n
    
    try:
        res = linprog(c_obj, A_ub=A_red, b_ub=b_ub, bounds=bounds, method='highs')
        if res.success and res.x is not None:
            g = np.maximum(res.x, 0.0)
            g_sum = g.sum()
            if g_sum > 1e-10:
                return g / g_sum
    except Exception:
        pass
    return None


def lp_mirror_descent_fast(a, max_iters, deadline, K=30, n_line_evals=12,
                            stall_patience=20, verbose=False):
    """
    LP-based mirror descent with reduced constraints and line search.
    """
    a = np.maximum(a.astype(np.float64), 1e-8)
    a = a / a.sum()
    current_J = compute_J(a)
    best_J = current_J
    best_a = a.copy()
    stall_count = 0
    
    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        
        b = fast_conv(a)
        M = float(b.max())
        
        g = lp_best_response_reduced(a, M, K=K)
        if g is None:
            break
        
        # Golden section line search
        best_t = 0.0
        gr = (np.sqrt(5.0) - 1.0) / 2.0
        lo = 0.0
        hi = 1.0
        
        J_at_1 = compute_J(g)
        if J_at_1 < current_J:
            best_J = min(best_J, J_at_1)
            best_a = g.copy()
            best_t = 1.0
        
        for _ in range(n_line_evals):
            t1 = hi - gr * (hi - lo)
            t2 = lo + gr * (hi - lo)
            a1 = (1.0 - t1) * a + t1 * g
            a2 = (1.0 - t2) * a + t2 * g
            J1 = compute_J(a1)
            J2 = compute_J(a2)
            if J1 < J2:
                hi = t2
            else:
                lo = t1
            
            t_mid = 0.5 * (lo + hi)
            a_mid = (1.0 - t_mid) * a + t_mid * g
            J_mid = compute_J(a_mid)
            if J_mid < best_J:
                best_J = J_mid
                best_a = a_mid.copy()
                best_t = t_mid
        
        a_new = (1.0 - best_t) * a + best_t * g
        J_new = compute_J(a_new)
        
        if J_new < current_J - 1e-13:
            a = a_new
            current_J = J_new
            stall_count = 0
        else:
            stall_count += 1
            if stall_count >= stall_patience:
                break
    
    return best_a if best_J < current_J else a

   To make |A[k]| approximately constant, we need a to be "band-limited" in a certain sense. This is related to the theory of "flat" signals.

   **Practical approach: Try sequences of the form a[k] = max(0, c0 + c1*cos(2πf1*k/n) + c2*cos(2πf2*k/n) + ...)**

   The max(0, ...) ensures non-negativity. By tuning the frequencies and amplitudes, we can get a good approximation of a "flat autocorrelation" sequence.

**Approach: Direct quadratic programming with active set method.**

Since J(a) = 2n * max_k (a*a)_k / (sum a)², and the max is the maximum of quadratic forms, we can formulate:

This is a non-convex QP (because (a*a)[k] is not convex in a). But we can use an active set method: at each iteration, only a few constraints are active (the ones where (a*a)[k] is close to t). We can solve the QP with just those active constraints iteratively.

Actually, this is essentially what the LP mirror descent does, but in a different way. The LP approach is:
- Fix a, find g that maximizes sum(g) subject to (a*g)[k] ≤ M
- This is equivalent to finding the "best response" in a game

The QP approach would be:
- Fix the active set S
- Solve: minimize t subject to (a*a)[k] ≤ t for k in S, sum(a) = 1, a ≥ 0
- This is a convex QP! (because we fix which constraints are active)

For k = 0: (a*a)[0] = a[0]², which is convex. So minimizing t subject to a[0]² ≤ t is convex.

For k = 1: (a*a)[1] = 2*a[0]*a[1], which is indefinite (not convex or concave).

```python
"""
Minimizes J(a) = 2n*max(convolve(a,a))/(sum(a))² using:
1. Fast FISTA with adaptive softmax smoothing for basin finding
2. Reduced-constraint LP mirror descent (top-K active constraints)
3. Multi-length exploration with diverse spectral initializations
4. Nesterov acceleration with adaptive momentum restarts
5. Aggressive perturbation-based exploration at stall
6. Fine-grained line search with golden section
"""

import numpy as np
from scipy.optimize import linprog
from scipy.sparse import lil_matrix, csr_matrix
import time


def fast_conv(a):
    """Full self-convolution via FFT."""
    n = len(a)
    s = 1
    while s < 2 * n - 1:
        s <<= 1
    fa = np.fft.rfft(a, s)
    return np.fft.irfft(fa * fa, s)[:2 * n - 1]


def compute_J(a):
    """Compute J(a) = 2*n*max(conv(a,a))/(sum(a))^2."""
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))


def smooth_max_and_grad(a, beta):
    """Compute smooth max of conv(a,a) and gradient."""
    n = len(a)
    b = fast_conv(a)
    b_max = float(b.max())
    shifted = beta * (b - b_max)
    shifted = np.clip(shifted, -500, 500)
    exp_shifted = np.exp(shifted)
    Z = float(exp_shifted.sum())
    if Z < 1e-300:
        p = np.zeros(2 * n - 1)
        p[np.argmax(b)] = 1.0
    else:
        p = exp_shifted / Z
    m_val = b_max + np.log(max(Z, 1e-300)) / beta
    grad_m = 2.0 * np.correlate(p, a, 'valid')
    return m_val, grad_m


def F_and_grad(a, beta):
    """F(a) = 2n*m_beta(a)/(sum(a))^2 and its gradient."""
    n = len(a)
    s = float(a.sum())
    if s < 1e-10:
        return 1e10, np.zeros(n)
    m_val, grad_m = smooth_max_and_grad(a, beta)
    F_val = 2.0 * n * m_val / (s * s)
    grad_F = (2.0 * n / (s ** 3)) * (s * grad_m - 2.0 * m_val)
    return F_val, grad_F


def fista_optimize(a0, beta_schedule, deadline):
    """FISTA with backtracking and restarts."""
    a = np.maximum(a0.astype(np.float64), 1e-8)
    n = len(a)
    
    for beta, max_iters in beta_schedule:
        if time.monotonic() > deadline:
            break
        
        x = a.copy()
        y = a.copy()
        t_k = 1.0
        L = 10.0
        best_F = compute_J(a)
        best_a = a.copy()
        no_improve = 0
        
        for it in range(max_iters):
            if time.monotonic() > deadline:
                break
            
            y_clipped = np.maximum(y, 0)
            F_y, grad_y = F_and_grad(y_clipped, beta)
            if not np.isfinite(F_y):
                L *= 2
                continue
            
            # Gradient step with backtracking
            c1 = 1e-4
            for _ in range(30):
                x_new = y - grad_y / L
                x_new = np.maximum(x_new, 0)
                F_x, _ = F_and_grad(x_new, beta)
                if not np.isfinite(F_x):
                    L *= 2
                    continue
                F_pred = F_y + grad_y @ (x_new - y_clipped) + 0.5 * L * np.sum((x_new - y_clipped)**2)
                if F_x <= F_pred + 1e-10:
                    break
                L *= 2
            else:
                break
            
            # Nesterov momentum
            t_new = (1.0 + np.sqrt(1.0 + 4.0 * t_k**2)) / 2.0
            y = x_new + (t_k / t_new - 1.0) * (x_new - x)
            t_k = t_new
            
            # Check improvement (use true objective periodically)
            if it % 10 == 0:
                F_true = compute_J(x_new)
                if F_true < best_F - 1e-12:
                    best_F = F_true
                    best_a = x_new.copy()
                    no_improve = 0
                else:
                    no_improve += 1
                    if no_improve > 15:
                        # Restart momentum
                        y = x_new.copy()
                        t_k = 1.0
                        L = max(1.0, L * 0.5)
                        no_improve = 0
            
            a = x_new
        
        a = best_a
    
    return a


def lp_best_response_reduced(a, M, K=30):
    """LP with top-K constraints: max sum(g) s.t. conv(a,g)[k] <= M, g >= 0."""
    n = len(a)
    b = fast_conv(a)
    top_k = np.argsort(b)[-K:]
    
    rows_list = []
    cols_list = []
    vals_list = []
    for idx, k in enumerate(top_k):
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        for j in range(j_start, j_end + 1):
            rows_list.append(idx)
            cols_list.append(j)
            vals_list.append(a[k - j])
    
    A_red = lil_matrix((K, n))
    for r, c, v in zip(rows_list, cols_list, vals_list):
        A_red[r, c] = v
    A_red = A_red.tocsr()
    
    c_obj = -np.ones(n)
    b_ub = np.full(K, M)
    bounds = [(0.0, None)] * n
    
    try:
        res = linprog(c_obj, A_ub=A_red, b_ub=b_ub, bounds=bounds, method='highs')
        if res.success and res.x is not None:
            g = np.maximum(res.x, 0.0)
            g_sum = g.sum()
            if g_sum > 1e-10:
                return g / g_sum
    except Exception:
        pass
    return None
