import numpy as np
from scipy.optimize import minimize, linprog
from scipy.sparse import csr_matrix
import time

def fast_conv(a):
    """Full self-convolution via FFT."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]

def compute_J(a):
    """Compute J(a) = 2*n*max(conv(a,a))/(sum(a))^2."""
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))

def smooth_grad(a, tau, top_k=None):
    """
    Compute gradient of (1/tau)*log(sum_k exp(tau*(a*a)[k])) w.r.t. a.
    Optionally only use top_k lags for efficiency.
    Returns (grad, b, M).
    """
    n = len(a)
    b = fast_conv(a)
    b_max = b.max()
    
    if top_k is not None and top_k < len(b):
        # Only use top_k lags
        idx = np.argsort(b)[-top_k:]
        b_use = b[idx]
        active_mask = np.zeros(len(b), dtype=bool)
        active_mask[idx] = True
    else:
        b_use = b
        active_mask = np.ones(len(b), dtype=bool)
    
    # Numerically stable softmax over active lags
    exp_b = np.exp(tau * (b_use - b_use.max()))
    p = exp_b / exp_b.sum()
    
    grad = np.zeros(n)
    if top_k is not None:
        for i, k in enumerate(idx):
            pk = p[i]
            if pk < 1e-14:
                continue
            lo = max(0, k - n + 1)
            hi = min(n - 1, k)
            grad[lo:hi+1] += 2.0 * pk * a[k-hi:k-lo+1][::-1]
    else:
        for k in range(2 * n - 1):
            pk = p[k]
            if pk < 1e-14:
                continue
            lo = max(0, k - n + 1)
            hi = min(n - 1, k)
            grad[lo:hi+1] += 2.0 * pk * a[k-hi:k-lo+1][::-1]
    
    return grad, b, b_max

def multiplicative_descent(a, max_iters, deadline, tau_init=5.0, tau_end=200.0,
                           lr_init=0.5, lr_end=0.01, top_k=None):
    """
    Multiplicative (mirror) descent with negative entropy.
    Update: a[i] <- a[i] * exp(-lr * grad[i]) / normalization
    """
    a = np.maximum(a.astype(np.float64), 1e-10)
    a = a / a.sum()
    n = len(a)
    
    best_a = a.copy()
    best_M = fast_conv(a).max()
    
    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        
        frac = it / max(1, max_iters - 1)
        tau = tau_init + (tau_end - tau_init) * frac
        lr = lr_init * (lr_end / lr_init) ** frac
        
        if top_k is None:
            top_k_eff = max(8, n // 4)
            # Adapt top_k: start broad, narrow down
            top_k_eff = max(4, int(n * 0.5 * (1 - 0.7 * frac)))
        else:
            top_k_eff = top_k
        
        grad, b, M = smooth_grad(a, tau, top_k=top_k_eff)
        
        if M < best_M * (1 - 1e-10):
            best_M = M
            best_a = a.copy()
        
        # Multiplicative update
        log_update = -lr * grad
        # Clip to avoid overflow
        log_update = np.clip(log_update, -5.0, 5.0)
        a_new = a * np.exp(log_update)
        s = a_new.sum()
        if s < 1e-10:
            break
        a_new = a_new / s
        a = a_new
    
    return best_a if fast_conv(best_a).max() <= fast_conv(a).max() + 1e-12 else a

def additive_descent(a, max_iters, deadline, tau_init=5.0, tau_end=200.0,
                     lr_init=0.3, lr_end=0.005, top_k=None, momentum=0.85):
    """

    """
    a = np.maximum(a.astype(np.float64), 1e-10)
    a = a / a.sum()

    v = np.zeros(n)

            break

        lr = lr_init * (lr_end / lr_init) ** frac

            top_k_eff = max(4, int(n * 0.5 * (1 - 0.7 * frac)))
        else:
            top_k_eff = top_k

        if M < best_M * (1 - 1e-10):

            best_a = a.copy()

        a_new = a - v
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s < 1e-10:

        a = a_new

def active_set_lp(a, M=None, top_lags=20, time_limit=10):
    """
    LP: max sum(g) s.t. (a*g)[k] <= M for top_k lags, g >= 0.
    Only considers the top_lags convolution values.
    """
    n = len(a)
    b = fast_conv(a)
    if M is None:
        M = b.max()
    
    # Get top lags
    top_idx = np.argsort(b)[-top_lags:]
    top_idx = np.sort(top_idx)
    
    # Build constraint matrix for these lags
    rows = []
    cols = []
    vals = []
    for k in top_idx:
        j_min = max(0, k - n + 1)
        j_max = min(n - 1, k)
        for j in range(j_min, j_max + 1):
            rows.append(k - top_idx[0])
            cols.append(j)
            vals.append(a[k - j])
    
    row_map = {k: i for i, k in enumerate(top_idx)}
    rows = [row_map[k] for k in rows]
    
    A = csr_matrix((vals, (rows, cols)), shape=(len(top_idx), n))
    c = -np.ones(n)
    b_ub = np.full(len(top_idx), M)
    bounds = [(0, None)] * n
    
    try:
        r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                    options={'time_limit': time_limit})
        if r.success and r.x is not None:
            g = np.maximum(r.x, 0)
            if g.sum() > 1e-15:
                return g / g.sum()
    except Exception:
        pass
    return None

def lp_mirror_descent(a, max_iters, deadline, top_lags=30):
    """LP mirror descent using reduced LP (top lags only)."""
    a = np.maximum(a.astype(np.float64), 1e-10)
    a = a / a.sum()
    
    best_a = a.copy()
    best_M = fast_conv(a).max()
    no_improve = 0
    
    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        
        b = fast_conv(a)
        M = float(b.max())
        
        g = active_set_lp(a, M, top_lags=top_lags, time_limit=5)
        if g is None:
            break
        
        # Line search
        improved = False
        base_t = 0.4 * (1 - 0.5 * it / max(1, max_iters))
        base_t = max(0.05, base_t)
        
        for t in [base_t, base_t * 0.5, base_t * 0.25, 0.02, 0.01]:
            a_trial = np.maximum((1 - t) * a + t * g, 0)
            s = a_trial.sum()
            if s < 1e-10:
                continue
            a_trial /= s
            M_trial = fast_conv(a_trial).max()
            if M_trial < best_M - 1e-15:
                a = a_trial.copy()
                best_M = M_trial
                best_a = a_trial.copy()
                improved = True
                break
        
        if not improved:
            no_improve += 1
            if no_improve > 30:
                break
        else:
            no_improve = 0
    
    return best_a

def resample_seq(src, dest_n):
    """Resample 1D sequence to new length via linear interpolation."""
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)

def structural_perturb(a, rng, mode='random'):
    """Apply structural perturbations to create diversity."""
    n = len(a)
    t = np.linspace(0, 1, n, endpoint=False)
    
    if mode == 'ramp':
        # Multiply by a smooth ramp
        alpha = rng.uniform(0.3, 2.0)
        a_new = a * (1 + (alpha - 1) * t)
    elif mode == 'ramp_rev':
        alpha = rng.uniform(0.3, 2.0)
        a_new = a * (1 + (alpha - 1) * (1 - t))
    elif mode == 'bump':
        # Multiply by a Gaussian bump
        center = rng.uniform(0.2, 0.8)
        width = rng.uniform(0.1, 0.4)
        a_new = a * (0.5 + 0.5 * np.exp(-((t - center) / width) ** 2))
    elif mode == 'cosine_mod':
        # Multiply by a cosine modulation
        freq = rng.uniform(1, 5)
        amp = rng.uniform(0.2, 0.6)
        phase = rng.uniform(0, 2 * np.pi)
        a_new = a * (1 + amp * np.cos(2 * np.pi * freq * t + phase))
    elif mode == 'warp':
        # Non-linear index warp
        alpha = rng.uniform(0.7, 1.3)

def generate_initializations(n, rng, parent=None):
    """Generate diverse initial sequences of length n."""
    inits = []
    t_arr = np.linspace(0, 1, n, endpoint=False)
    
    if parent is not None:
        p = resample_seq(parent, n)
        inits.append(p)
        inits.append(np.maximum(p + rng.normal(0, 0.03, n), 0.01))
        inits.append(np.maximum(p + rng.normal(0, 0.1, n), 0.01))
        inits.append(p[::-1])  # reversed
    
    inits.append(np.ones(n))
    inits.append(rng.uniform(0.5, 1.0, n))
    
    # Ramps (important for asymmetry!)
    inits.append(np.maximum(0.5 + t_arr, 0.01))
    inits.append(np.maximum(0.5 + (1 - t_arr), 0.01))
    inits.append(np.maximum(0.3 + 0.7 * t_arr ** 2, 0.01))
    inits.append(np.maximum(0.3 + 0.7 * (1 - t_arr) ** 2, 0.01))
    
    # Various frequency structures
    for freq in [1, 2, 3, 5]:
        inits.append(np.maximum(1.0 + 0.3 * np.cos(2 * np.pi * freq * t_arr), 0.01))
    
    # Exponential ramps
    inits.append(np.exp(1.5 * t_arr) + 0.1)
    inits.append(np.exp(1.5 * (t_arr - 1)) + 0.1)
    
    # Gaussian
    inits.append(np.exp(-3.0 * (t_arr - 0.5) ** 2) + 0.1)
    
    return inits

"""
Minimizes J(a) = 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Multiplicative (mirror) descent with negative entropy - naturally preserves positivity
2. Adaptive top-k smooth gradients (broad to focused)
3. Reduced-constraint LP mirror descent for precise refinement
4. Coordinate descent on high-gradient components
5. Structural perturbations for diversity escaping
6. Multi-length exploration with coarse-to-fine strategy
Key innovation: combines multiplicative updates (different geometry than additive GD)
with active-set LP and coordinate descent for a more thorough local search.

import numpy as np
from scipy.optimize import linprog, minimize
from scipy.sparse import csr_matrix
import time


def fast_conv(a):
    """Full self-convolution via FFT."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]


def compute_M(a):
    """Compute max of self-convolution (assumes a is normalized)."""
    return float(fast_conv(a).max())


def compute_J(a):
    """Compute J(a) = 2*n*max(conv(a,a))/(sum(a))^2."""
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))


def smooth_grad_topk(a, tau, top_k):
    """
    Compute gradient of smooth max using only top_k lags.
    Returns (grad, b, M).
    """
    n = len(a)
    b = fast_conv(a)
    M = b.max()
    n_lags = len(b)
    
    if top_k >= n_lags:
        idx = np.arange(n_lags)
    else:
        idx = np.argsort(b)[-top_k:]
    
    b_use = b[idx]
    b_shift = b_use - b_use.max()
    exp_b = np.exp(tau * b_shift)
    p = exp_b / exp_b.sum()
    
    grad = np.zeros(n)
    for i, k in enumerate(idx):
        pk = p[i]
        if pk < 1e-14:
            continue
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        grad[lo:hi + 1] += 2.0 * pk * a[k - hi:k - lo + 1][::-1]
    
    return grad, b, M


def multiplicative_descent(a, max_iters, deadline, tau_init=5.0, tau_end=200.0,
                           lr_init=0.5, lr_end=0.01, top_k_frac=0.5):
    """
    Multiplicative (mirror) descent with negative entropy.
    Update: a[i] <- a[i] * exp(-lr * grad[i]), then renormalize.
    """
    a = np.maximum(a.astype(np.float64), 1e-12)
    a = a / a.sum()
    n = len(a)
    
    best_a = a.copy()
    best_M = compute_M(a)
    
    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        
        frac = it / max(1, max_iters - 1)
        tau = tau_init + (tau_end - tau_init) * frac
        lr = lr_init * (lr_end / lr_init) ** frac
        top_k = max(4, int(n * top_k_frac * (1.0 - 0.6 * frac)))
        
        grad, b, M = smooth_grad_topk(a, tau, top_k)
        
        if M < best_M * (1 - 1e-12):
            best_M = M
            best_a = a.copy()
        
        log_update = -lr * grad
        log_update = np.clip(log_update, -4.0, 4.0)
        a_new = a * np.exp(log_update)
        s = a_new.sum()
        if s < 1e-10:
            break
        a = a_new / s
    
    final_M = compute_M(a)
    if final_M < best_M:
        best_a = a.copy()
        best_M = final_M
    return best_a


def additive_descent(a, max_iters, deadline, tau_init=5.0, tau_end=200.0,
                     lr_init=0.3, lr_end=0.005, top_k_frac=0.5, momentum=0.85):
    """Additive gradient descent with momentum."""
    a = np.maximum(a.astype(np.float64), 1e-12)
    a = a / a.sum()
    n = len(a)
    v = np.zeros(n)
    
    best_a = a.copy()
    best_M = compute_M(a)
    
    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        
        frac = it / max(1, max_iters - 1)
        tau = tau_init + (tau_end - tau_init) * frac
        lr = lr_init * (lr_end / lr_init) ** frac
        top_k = max(4, int(n * top_k_frac * (1.0 - 0.6 * frac)))
        
        grad, b, M = smooth_grad_topk(a, tau, top_k)
        
        if M < best_M * (1 - 1e-12):
            best_M = M
            best_a = a.copy()
        
        v = momentum * v + lr * grad
        a_new = a - v
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s < 1e-10:
            a_new = a.copy()
        else:
            a = a_new / s
    
    final_M = compute_M(a)
    if final_M < best_M:
        best_a = a.copy()
    return best_a


def active_set_lp(a, M, top_lags=30, time_limit=8):
    """
    LP: max sum(g) s.t. (a*g)[k] <= M for top_lags lags, g >= 0.
    Returns normalized g or None.
    """
    n = len(a)
    b = fast_conv(a)
    
    # Get top lags by value
    top_idx = np.argsort(b)[-top_lags:]
    top_idx_sorted = np.sort(top_idx)
    
    # Build sparse constraint matrix
    rows = []
    cols = []
    vals = []
    row_map = {k: i for i, k in enumerate(top_idx_sorted)}
    
    for k in top_idx_sorted:
        j_min = max(0, k - n + 1)
        j_max = min(n - 1, k)
        for j in range(j_min, j_max + 1):
            rows.append(row_map[k])
            cols.append(j)
            vals.append(a[k - j])
    
    A = csr_matrix((vals, (rows, cols)), shape=(len(top_idx_sorted), n))
    c = -np.ones(n)
    b_ub = np.full(len(top_idx_sorted), M)
    bounds = [(0, None)] * n
    
    try:
        r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                    options={'time_limit': time_limit})
        if r.success and r.x is not None:
            g = np.maximum(r.x, 0)
            s = g.sum()
            if s > 1e-15:
                return g / s
    except Exception:
        pass
    return None


def lp_mirror_descent(a, max_iters, deadline, top_lags=30):
    """LP mirror descent with geometric line search using reduced LP."""
    a = np.maximum(a.astype(np.float64), 1e-12)
    a = a / a.sum()
    
    best_a = a.copy()
    best_M = compute_M(a)
    no_improve = 0
    
    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        
        b = fast_conv(a)
        M = float(b.max())
        
        g = active_set_lp(a, M, top_lags=top_lags, time_limit=5)
        if g is None:
            break
        
        improved = False
        base_t = max(0.05, 0.5 * (1 - 0.4 * it / max(1, max_iters)))
        
        for t in [base_t, base_t * 0.5, base_t * 0.25, 0.03, 0.01]:
            a_trial = np.maximum((1 - t) * a + t * g, 0)
            s = a_trial.sum()
            if s < 1e-10:
                continue
            a_trial /= s
            M_trial = fast_conv(a_trial).max()
            if M_trial < best_M - 1e-14:
                a = a_trial.copy()
                best_M = M_trial
                best_a = a_trial.copy()
                improved = True
                break
        
        if not improved:
            no_improve += 1
            if no_improve > 40:
                break
        else:
            no_improve = 0
    
    return best_a


def coordinate_descent(a, n_steps=50, deadline=None):
    """
    Coordinate descent: optimize individual components using golden section.
    Focus on components with largest marginal impact on max conv.
    """
    n = len(a)
    a = a.copy()
    best_M = compute_M(a)
    
    # Precompute which lags each coordinate affects
    for step in range(n_steps):
        if deadline is not None and time.monotonic() > deadline:
            break
        
        # Compute gradient to find which coordinates to update
        grad, b, M = smooth_grad_topk(a, 50.0, max(10, n // 4))
        
        # Pick top coordinates by gradient magnitude
        top_coords = np.argsort(-np.abs(grad))[:min(20, n)]
        
        for i in top_coords:
            if deadline is not None and time.monotonic() > deadline:
                break
            
            # Golden section search for a[i]
            t_lo = 0.0
            t_hi = 3.0 / n  # reasonable upper bound
            ai_old = a[i]
            
            # Inner golden section
            gr = (np.sqrt(5.0) - 1.0) / 2.0
            c1 = t_lo + (1 - gr) * (t_hi - t_lo)
            c2 = t_lo + gr * (t_hi - t_lo)
            
            a[i] = c1
            f1 = compute_M(a)
            a[i] = c2
            f2 = compute_M(a)
            
            for _ in range(25):
                if f1 < f2:
                    t_hi = c2
                    c2 = c1
                    f2 = f1
                    c1 = t_lo + (1 - gr) * (t_hi - t_lo)
                    a[i] = c1
                    f1 = compute_M(a)
                else:
                    t_lo = c1
                    c1 = c2
                    f1 = f2
                    c2 = t_lo + gr * (t_hi - t_lo)
                    a[i] = c2
                    f2 = compute_M(a)
            
            a[i] = (c1 + c2) / 2.0
            f_opt = compute_M(a)
            
            if f_opt > best_M + 1e-10:
                a[i] = ai_old  # revert if worse
            
            if f_opt < best_M:
                best_M = f_opt
    
    s = a.sum()
    if s > 1e-10:
        a = a / s
    return a


def resample_seq(src, dest_n):
    """Resample 1D sequence to new length via linear interpolation."""
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)


def structural_perturb(a, rng, mode='random'):
    """Apply structural perturbations to create diversity."""
    n = len(a)
    t = np.linspace(0, 1, n, endpoint=False)
    
    if mode == 'ramp_up':
        alpha = rng.uniform(0.3, 2.5)
        a_new = a * (1 + (alpha - 1) * t)
    elif mode == 'ramp_down':
        alpha = rng.uniform(0.3, 2.5)
        a_new = a * (1 + (alpha - 1) * (1 - t))
    elif mode == 'quad_ramp':
        alpha = rng.uniform(0.5, 3.0)
        a_new = a * (1 + (alpha - 1) * t ** 2)
    elif mode == 'bump':
        center = rng.uniform(0.2, 0.8)
        width = rng.uniform(0.1, 0.4)
        a_new = a * (0.3 + 0.7 * np.exp(-((t - center) / width) ** 2))
    elif mode == 'cosine_mod':
        freq = rng.uniform(1, 6)
        amp = rng.uniform(0.2, 0.7)
        phase = rng.uniform(0, 2 * np.pi)
        a_new = a * (1 + amp * np.cos(2 * np.pi * freq * t + phase))
    elif mode == 'warp':
        alpha = rng.uniform(0.6, 1.5)
        new_t = t ** alpha
        a_new = np.interp(new_t, t, a)
    elif mode == 'mirror_mix':
        t_mix = rng.uniform(0.1, 0.5)
        a_new = (1 - t_mix) * a + t_mix * a[::-1]
    elif mode == 'block_shuffle':
        n_blocks = rng.integers(4, min(12, n // 16 + 1))
        block_size = n // n_blocks
        indices = np.arange(n_blocks)
        rng.shuffle(indices)
        a_new = np.zeros(n)
        for i, j in enumerate(indices):
            si = i * block_size
            sj = j * block_size
            a_new[si:si + block_size] = a[sj:sj + block_size]
        # Handle remainder
        rem = n - n_blocks * block_size
        if rem > 0:
            a_new[-rem:] = a[-rem:]
    elif mode == 'exp_mod':
        rate = rng.uniform(-2, 2)
        a_new = a * np.exp(rate * (t - 0.5))
    else:  # gaussian noise
        scale = rng.uniform(0.03, 0.15)
        a_new = a + rng.normal(0, scale, n)
    
    a_new = np.maximum(a_new, 0.005)
    s = a_new.sum()
    if s > 1e-10:
        a_new = a_new / s
    return a_new


def generate_initializations(n, rng, parent=None):
    """Generate diverse initial sequences of length n."""
    inits = []
    t_arr = np.linspace(0, 1, n, endpoint=False)
    
    if parent is not None:
        p = resample_seq(parent, n)
        inits.append(p)
        inits.append(np.maximum(p + rng.normal(0, 0.02, n), 0.01))
        inits.append(np.maximum(p + rng.normal(0, 0.08, n), 0.01))
        inits.append(p[::-1])
    
    # Ramps (important for asymmetry)
    inits.append(0.3 + 0.7 * t_arr)
    inits.append(0.3 + 0.7 * (1 - t_arr))
    inits.append(0.2 + 0.8 * t_arr ** 1.5)
    inits.append(0.2 + 0.8 * (1 - t_arr) ** 1.5)
    
    # Exponential
    inits.append(np.exp(2.0 * t_arr) + 0.05)
    inits.append(np.exp(2.0 * (t_arr - 1)) + 0.05)
    
    # Cosine
    inits.append(0.5 + 0.5 * np.cos(np.pi * t_arr))
    inits.append(0.5 + 0.3 * np.cos(2 * np.pi * t_arr) + 0.2 * np.cos(4 * np.pi * t_arr))
    
    # Random smooth
    ksize = max(3, n // 30)
    kernel = np.ones(ksize) / ksize
    smooth = np.convolve(rng.uniform(0, 1, n), kernel, mode='same')
    inits.append(smooth + 0.1)
    
    for a in inits:
        a = np.maximum(a, 0.01)
        inits[-1] = a / a.sum()
    
    return inits[:10]  # Limit to 10

            if parent.size > 10:
                tracker.update(parent)
                print(f"Parent loaded: n={parent.size}, J={tracker.best_score:.6f}", flush=True)
    except (NameError, AttributeError):
        pass
    
    if parent is None:
        parent = np.ones(256)
        tracker.update(parent)
    
    current_best = parent.copy()
    
    # === PHASE 1: Multi-length exploration ===
    print(f"\n{'='*60}", flush=True)
    print(f"PHASE 1: Multi-length Exploration", flush=True)
    print(f"{'='*60}", flush=True)
    
    # Focus near parent length and try a spread
    parent_n = parent.size
    lengths = sorted(set([
        max(128, parent_n - 128),
        max(128, parent_n - 64),
        parent_n,
        parent_n + 64,
        parent_n + 128,
        parent_n + 256,
        max(128, parent_n // 2 + 64),
        parent_n * 2 if parent_n * 2 <= 2048 else parent_n + 512,
    ]))
    lengths = [n for n in lengths if 128 <= n <= 2048]
    
    p1_deadline = min(time.monotonic() + 350, deadline - 400)
    time_per_n = max(20, (p1_deadline - time.monotonic()) / len(lengths))
    
    for n in lengths:
        if time.monotonic() > p1_deadline:
            break
        
        phase_dl = min(time.monotonic() + time_per_n, p1_deadline)
        t0 = time.monotonic()
        
        inits = generate_initializations(n, rng, current_best)
        best_for_n = None
        best_M_for_n = float('inf')
        
        for a0 in inits[:6]:  # Limit initializations
            if time.monotonic() > phase_dl:
                break
            a0 = np.maximum(a0.astype(np.float64), 1e-8)
            a0 = a0 / a0.sum()
            
            # Multiplicative descent (primary)
            a = multiplicative_descent(a0, max_iters=300, deadline=phase_dl,
                                       tau_init=5.0, tau_end=150.0,
                                       lr_init=0.4, lr_end=0.02)
            
            M_val = compute_M(a)
            if M_val < best_M_for_n:
                best_M_for_n = M_val
                best_for_n = a.copy()
        
        if best_for_n is not None:
            J_val = compute_J(best_for_n)
            elapsed = time.monotonic() - t0
            print(f"  n={n:4d}: M={best_M_for_n:.8f} J={J_val:.6f} ({elapsed:.1f}s)",
                  flush=True)
            if tracker.best is not None and J_val < tracker.best_score:
                tracker.update(best_for_n)
            elif tracker.best is None:
                tracker.update(best_for_n)
            if J_val < compute_J(current_best):
                current_best = best_for_n.copy()
    
    tracker.report("Phase 1 complete")
    
    # === PHASE 2: Intensive optimization on best ===
    print(f"\n{'='*60}", flush=True)
    print(f"PHASE 2: Intensive Local Optimization", flush=True)
    print(f"{'='*60}", flush=True)
    
    p2_deadline = min(time.monotonic() + 400, deadline - 250)
    
    if tracker.best is not None:
        a = tracker.best.copy()
        a = intensive_local_opt(a, p2_deadline)
        tracker.update(a)
    
    # Also try optimizing the reversed parent at same length
    if tracker.best is not None and time.monotonic() < p2_deadline - 60:
        a_rev = tracker.best[::-1].copy()
        a_rev = a_rev / a_rev.sum()
        remaining = p2_deadline - time.monotonic()
        if remaining > 60:
            a = multiplicative_descent(a_rev, 200, time.monotonic() + remaining * 0.5,
                                       tau_init=10, tau_end=200,
                                       lr_init=0.3, lr_end=0.01)
            tracker.update(a)
    
    tracker.report("Phase 2 complete")
    
    # === PHASE 3: Structural perturbation exploration ===
    print(f"\n{'='*60}", flush=True)
    print(f"PHASE 3: Structural Perturbation", flush=True)
    print(f"{'='*60}", flush=True)
    
    p3_deadline = min(time.monotonic() + 350, deadline - 150)
    
    if tracker.best is not None:
        n_best = tracker.best.size
        modes = ['ramp_up', 'ramp_down', 'quad_ramp', 'bump', 'cosine_mod',
                 'warp', 'mirror_mix', 'block_shuffle', 'exp_mod', 'ramp_up',
                 'ramp_down', 'cosine_mod', 'warp', 'exp_mod', 'bump']
        
        for r in range(len(modes)):
            if time.monotonic() > p3_deadline:
                break
            
            mode = modes[r]
            a_pert = structural_perturb(tracker.best, rng, mode=mode)
            
            # Quick multiplicative + additive
            phase_t = min(time.monotonic() + 18, p3_deadline)
            a = multiplicative_descent(a_pert, 150, phase_t,
                                       tau_init=5, tau_end=100,
                                       lr_init=0.3, lr_end=0.02)
            tracker.update(a)
            
            if time.monotonic() < p3_deadline - 10:
                phase_t2 = min(time.monotonic() + 8, p3_deadline)
                a = additive_descent(a, 100, phase_t2,
                                     tau_init=30, tau_end=300,
                                     lr_init=0.1, lr_end=0.01, momentum=0.9)
                tracker.update(a)
    
    tracker.report("Phase 3 complete")
    
    # === PHASE 4: Final intensive polish ===
    print(f"\n{'='*60}", flush=True)
    print(f"PHASE 4: Final Polish", flush=True)
    print(f"{'='*60}", flush=True)
    
    if tracker.best is not None and time.monotonic() < deadline - 10:
        # One more intensive pass
        a = tracker.best.copy()
        a = intensive_local_opt(a, min(time.monotonic() + 150, deadline - 5))
        tracker.update(a)
        
        # If we improved, try one more short LP
        if time.monotonic() < deadline - 10:
            n_best = tracker.best.size
            est_lp_time = 0.004 * (n_best / 128) ** 2
            max_iters = max(10, int((deadline - 5 - time.monotonic()) / max(0.01, est_lp_time * 1.5)))
            max_iters = min(max_iters, 80)
            a = lp_mirror_descent(tracker.best, max_iters, deadline - 3,
                                  top_lags=max(40, n_best // 6))
            tracker.update(a)
    
    # Fallback
    if tracker.best is None:
        tracker.best = np.ones(128)
        tracker.best_score = compute_J(tracker.best)
    
    print(f"\n{'='*60}", flush=True)
    print(f"FINAL: J={tracker.best_score:.8f} n={tracker.best.size} "
          f"time={time.monotonic()-tracker.t_start:.0f}s", flush=True)
    print(f"{'='*60}", flush=True)
    
    return [float(x) for x in tracker.best]
