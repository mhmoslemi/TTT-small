    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-8:
        return 1e10
    a = a / s
    b = fast_conv(a)

    b_max = b.max()
    smooth_max = b_max + np.log(np.exp(beta * (b - b_max)).sum()) / beta
    return float(2 * len(a) * smooth_max)

def homotopy_lbbfs(a, deadline):
    a = a / a.sum()
    for beta in [1, 2, 4, 8, 16, 32, 64, 128, 256, 512]:
        if time.monotonic() > deadline:
            break
        res = minimize(
            lambda x: smoothed_objective(x, beta),
            a,
            method='L-BFGS-B',
            bounds=[(0, None)] * len(a),
            options={'maxiter': 50, 'max_fun': 5000}
        )
        a = np.maximum(res.x, 0)
        if a.sum() > 1e-8:
            a = a / a.sum()
    return a

def fast_conv(a):
    """Full self-convolution via FFT."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]


def compute_J(a):
    """Compute J(a) = 2*n*max(conv(a,a))/(sum(a))^2."""
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))


def build_A_sparse(n, a):
    """Build sparse matrix A where A[k,j] = a[k-j] for valid j."""
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):

        j_max = min(n - 1, k)
        for j in range(j_min, j_max + 1):
            rows.append(k)
            cols.append(j)
            vals.append(a[k - j])
    return csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))


def lp_best_response(a, M):
    """Solve LP: max sum(g) s.t. A@g <= M*1, g >= 0."""
    n = len(a)
    A = build_A_sparse(n, a)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    try:
        r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
        if r.success and r.x is not None:
            g = np.maximum(r.x, 0)
            if g.sum() > 1e-15:
                return g
    except Exception:
        pass
    return None

from scipy.sparse import csr_matrix
import time


def fast_conv(a):
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]


def compute_J(a):
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))


def build_A_sparse(n, a):
    """Build sparse matrix A where A[k,j] = a[k-j] for valid (k,j). Vectorized."""
    rows_parts = []
    cols_parts = []
    for k in range(2 * n - 1):
        j_min = max(0, k - n + 1)
        j_max = min(n - 1, k)
        if j_max >= j_min:
            rows_parts.append(np.full(j_max - j_min + 1, k, dtype=np.int32))
            cols_parts.append(np.arange(j_min, j_max + 1, dtype=np.int32))
    rows = np.concatenate(rows_parts)
    cols = np.concatenate(cols_parts)
    vals = a[rows - cols].astype(np.float64)
    return csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))


def lp_best_response(a, M):
    n = len(a)
    A = build_A_sparse(n, a)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    try:
        r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
        if r.success and r.x is not None:
            g = np.maximum(r.x, 0)
            if g.sum() > 1e-15:
                return g
    except Exception:
        pass
    return None


def lp_mirror_descent(a, max_iters, deadline, use_multiplicative=False, aggressive=False):
    """LP mirror descent with adaptive line search on M (not J)."""
    a = np.maximum(a.astype(np.float64), 1e-10)
    a = a / a.sum()
    current_M = fast_conv(a).max()
    best_J = compute_J(a)
    best_a = a.copy()
    no_improve = 0

    if aggressive:
        base_rate = 0.25
        min_t = 0.08
    else:
        base_rate = 0.15
        min_t = 0.05

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        b = fast_conv(a)
        M = float(b.max())
        g = lp_best_response(a, M)
        if g is None:
            break
        g = np.maximum(g, 0.0)
        g_sum = g.sum()
        if g_sum < 1e-10:
            break
        g = g / g_sum

        # Adaptive step size
        decay = 0.995 if it > 100 else 0.99
        base_t = min(1.0, base_rate * np.exp(-0.1 * it / 50.0) + min_t)
        t = base_t

        best_trial_M = current_M
        best_trial_a = None

        for ls in range(15):
            if use_multiplicative:
                ratio = g / np.maximum(a, 1e-15)
                a_trial = a * np.power(ratio, t)
            else:
                a_trial = (1.0 - t) * a + t * g
            a_trial = np.maximum(a_trial, 0.0)
            s = a_trial.sum()
            if s < 1e-10:
                t *= 0.5
                continue
            a_trial = a_trial / s
            M_trial = fast_conv(a_trial).max()
            if M_trial < best_trial_M - 1e-14:
                best_trial_M = M_trial
                best_trial_a = a_trial.copy()
                break
            t *= 0.55
            if t < 0.001:
                break

        if best_trial_a is not None:
            a = best_trial_a
            current_M = best_trial_M
            J_trial = compute_J(a)
            if J_trial < best_J:
                best_J = J_trial
                best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1
            if no_improve > 25:
                break

    return best_a, best_J


def smoothed_max_and_grad(a, beta):
    """Compute smoothed max of autocorrelation and gradient."""
    n = len(a)
    s = a.sum()
    a_norm = a / s
    b = fast_conv(a_norm)

    b_max = b.max()
    exp_b = np.exp(beta * (b - b_max))
    Z = exp_b.sum()
    w = exp_b / Z

    smooth_max = b_max + np.log(Z) / beta
    J_val = 2.0 * n * smooth_max

    # Gradient via direct computation: grad[j] = (4*n*beta/s^2) * dot(w[j:j+n], a_norm)
    grad = np.zeros(n)
    for j in range(n):
        grad[j] = np.dot(w[j:j + n], a_norm)
    grad *= (4.0 * n * beta) / (s * s)

    return J_val, grad


def lbbfsm_func(x, beta):
    ax = np.maximum(x, 1e-15)
    val, _ = smoothed_max_and_grad(ax, beta)
    return val


def lbbfsm_jac(x, beta):
    ax = np.maximum(x, 1e-15)
    _, g = smoothed_max_and_grad(ax, beta)
    return g


def homotopy_lbfgsb(a, deadline):
    """Homotopy L-BFGS-B with log-sum-exp smoothing."""
    a = np.maximum(a, 1e-10)
    a = a / a.sum()
    best_J = compute_J(a)
    best_a = a.copy()
    n = len(a)

    betas = [1, 2, 4, 8, 16, 32, 64, 128, 256, 512, 1024, 2048]

    for beta in betas:
        if time.monotonic() > deadline - 5:
            break
        n_iter = max(30, min(80, 1500 // int(beta)))
        max_fun = n_iter * 15

        try:
            res = minimize(
                lbbfsm_func, a, args=(beta,),
                jac=lbbfsm_jac,
                method='L-BFGS-B',
                bounds=[(0.0, None)] * n,
                options={'maxiter': n_iter, 'max_fun': max_fun, 'ftol': 1e-16, 'gtol': 1e-12}
            )
            a_new = np.maximum(res.x, 1e-10)
            s = a_new.sum()
            if s > 1e-8:
                a_new = a_new / s
                J_new = compute_J(a_new)
                if J_new < best_J:
                    best_J = J_new
                    best_a = a_new.copy()
                a = a_new
        except Exception:
            pass

    return best_a, best_J


def es_one_plus_one(a, deadline, rng):
    """(1+1)-ES with adaptive step size."""
    a = np.maximum(a.astype(np.float64), 1e-10)
    a = a / a.sum()
    best_J = compute_J(a)
    best_a = a.copy()
    n = len(a)

    sigma = 0.02
    success_count = 0
    total_count = 0
    no_improve = 0

    while time.monotonic() < deadline:
        total_count += 1
        perturbation = rng.normal(0, sigma, n)
        a_trial = a + perturbation
        a_trial = np.maximum(a_trial, 0.0)
        s = a_trial.sum()
        if s < 1e-8:
            sigma *= 0.8
            continue
        a_trial = a_trial / s
        J_trial = compute_J(a_trial)

        if J_trial < best_J:
            best_J = J_trial
            best_a = a_trial.copy()
            success_count += 1
            no_improve = 0
        elif J_trial < compute_J(a):
            a = a_trial
            success_count += 1
            no_improve = 0
        else:
            no_improve += 1

        # 1/5 rule for step size adaptation
        if total_count % 50 == 0:
            rate = success_count / total_count
            if rate > 0.3:
                sigma *= 1.3
            elif rate < 0.1:
                sigma *= 0.7
            success_count = 0
            total_count = 0

        if no_improve > 200:
            break

    return best_a, best_J


def resample_seq(src, dest_n):
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)


def generate_diverse_inits(n, rng, parent=None):
    inits = []
    t = np.arange(n, dtype=np.float64) / n

    if parent is not None:
        p = resample_seq(parent, n)
        inits.append(p)
        inits.append(np.maximum(p + rng.normal(0, 0.03, n), 0.01))
        inits.append(np.maximum(p * (1 + 0.05 * rng.normal(0, 1, n)), 0.01))

    inits.append(np.ones(n))
    inits.append(0.5 * (1 + np.cos(2 * np.pi * t)))
    inits.append(np.maximum(0.42 - 0.5 * np.cos(2 * np.pi * t) + 0.08 * np.cos(4 * np.pi * t), 0.01))
    inits.append(np.exp(-4 * (t - 0.5) ** 2) + 0.1)
    inits.append(np.exp(-8 * (t - 0.3) ** 2) + np.exp(-8 * (t - 0.7) ** 2) + 0.05)

    # Quadratic residue-inspired (for prime n)
    prime_candidates = []
    for p in [n - 1, n + 1, n - 2, n + 2, n - 4, n + 4, n - 6, n + 6]:
        if p > 2 and p % 2 == 1:
            # Quick primality test
            is_prime = True
            for q in range(2, int(np.sqrt(p)) + 1):
                if p % q == 0:
                    is_prime = False
                    break
            if is_prime:
                prime_candidates.append(p)
                break

    if prime_candidates:
        p = prime_candidates[0]
        qr = np.zeros(p)
        for x in range(p):
            if (x * x) % p == 1 or x == 1:
                pass
        # Simpler: use Legendre symbol
        for x in range(1, p):
            qr[x] = 1.0 if pow(x, (p - 1) // 2, p) == 1 else 0.0
        if p != n:
            qr_n = resample_seq(qr, n)
        else:
            qr_n = qr
        inits.append(np.maximum(qr_n + 0.1, 0.05))

    for f in [1, 2, 3, 5]:
        inits.append(np.maximum(1.0 + 0.3 * np.cos(2 * np.pi * f * t), 0.01))

    k = max(3, n // 16)
    kernel = np.ones(k) / k
    smooth = np.convolve(rng.uniform(0, 1, n), kernel, mode='same')
    inits.append(np.maximum(smooth + 0.2, 0.01))

    result = []
    for a in inits:
        a = np.maximum(a, 0.001)
        s = a.sum()
        if s > 1e-6:
            result.append(a / s)
    return result


def recombine(a1, a2, rng):
    """Create a blend of two sequences."""
    for _ in range(10):
        alpha = rng.uniform(0.2, 0.8)
        blend = alpha * a1 + (1 - alpha) * a2
        # Add small perturbation
        blend = blend + rng.normal(0, 0.01, len(blend))
        blend = np.maximum(blend, 0.001)
        s = blend.sum()
        if s > 1e-8:
            blend = blend / s
            J = compute_J(blend)
            if J < max(compute_J(a1), compute_J(a2)):
                return blend
    return None


class ScoreTracker:
    def __init__(self):
        self.best = None
        self.best_score = float('inf')
        self.t_start = time.monotonic()

    def update(self, a):
        a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
        if a.size == 0 or a.sum() < 0.01:
            return
        score = float(evaluate_sequence(a.tolist()))
        if np.isfinite(score) and score < self.best_score:
            self.best = a.copy()
            self.best_score = score
            elapsed = time.monotonic() - self.t_start
            print(f"  [NEW BEST] J={score:.8f} n={a.size} t={elapsed:.0f}s", flush=True)

    def report(self, label=""):
        if self.best is not None:
            print(f"  {label} best: J={self.best_score:.8f} n={self.best.size}", flush=True)
