"""
Optimized step-function search minimizing 2*n*max(conv(a,a))/(sum(a))^2.
Uses: Nesterov-accelerated subgradient, improved LP alternating, spectral 
diagnosis for targeted peak reduction, coordinate descent polishing, and
multi-start with diverse initializations including parent interpolation.
Key improvements: accelerated methods, better LP warm-starting, adaptive 
target relaxation, and coordinate-wise refinement for final polish.

import numpy as np
from scipy.optimize import linprog
from scipy.interpolate import interp1d
import time


def compute_conv(a):
    """Compute linear convolution a*a via FFT."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    conv = np.fft.irfft(fa * fa, size)
    return conv[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    S = float(np.sum(a))
    if S < 1e-10:
        return np.inf
    conv = compute_conv(a)
    M = float(np.max(conv))
    return 2.0 * n * M / (S * S)


def build_toeplitz_sparse(a):
    """Build sparse Toeplitz matrix for conv(a, g) where g is variable."""
    n = len(a)
    rows = []
    cols = []
    vals = []
    for j in range(n):
        start = j
        end = min(j + n, 2 * n - 1)
        length = end - start
        rows.extend(range(start, end))
        cols.extend([j] * length)
        vals.extend(list(a[:length]))
    from scipy.sparse import coo_matrix
    A = coo_matrix((vals, (rows, cols)), shape=(2 * n - 1, n)).tocsr()
    return A


def solve_lp_maxsum(a, M, time_limit=8, warm_start=None):
    """Solve LP: max sum(g) s.t. conv(a,g) <= M, g >= 0. Returns g or None."""
    n = len(a)
    A = build_toeplitz_sparse(a)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    try:
        options = {'presolve': True, 'time_limit': time_limit}
        if warm_start is not None:
            options['warm_start'] = True
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                      options=options)
        if res.status == 0:
            return np.maximum(res.x, 0)
    except Exception:
        pass
    return None


def nesterov_subgradient(a, max_iters, time_budget, topk=8, verbose=False):
    """Nesterov-accelerated subgradient descent with projection onto simplex."""
    n = len(a)
    a = a.copy().astype(np.float64)
    S = np.sum(a)
    if S < 1e-10:
        return a
    a = a / S

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    y = a.copy()  # extrapolated point
    v = np.zeros(n)  # previous iterate for momentum
    t_momentum = 1.0

    for it in range(1, max_iters + 1):
        if time.monotonic() - start > time_budget:
            break

        # Compute subgradient at y
        conv_y = compute_conv(y)
        M_y = float(np.max(conv_y))
        active = np.argsort(conv_y)[-topk:]
        # Use weights proportional to how close to max
        weights = conv_y[active] / max(np.sum(conv_y[active]), 1e-15)

        grad = np.zeros(n)
        for k in active:
            for i in range(n):
                j = k - i
                if 0 <= j < n:
                    grad[i] += 2.0 * a[j] * weights[list(active).index(k)]

        grad /= np.sum(weights) if np.sum(weights) > 0 else 1.0
        # Center to preserve sum
        grad_centered = grad - np.sum(grad) / n

        # Nesterov step size
        step = 2.0 / (2.0 + t_momentum + it)
        # Additional scaling
        step *= 0.3

        # Update
        y = a - step * grad_centered
        y = np.maximum(y, 0)
        S_y = np.sum(y)
        if S_y < 1e-10:
            break
        y = y / S_y

        # Nesterov extrapolation
        beta = (t_momentum - 1.0) / (t_momentum + 2.0)
        t_momentum += 1.0
        new_a = y + beta * (y - a)
        new_a = np.maximum(new_a, 0)
        S_new = np.sum(new_a)
        if S_new < 1e-10:
            new_a = y.copy()
            S_new = S_y
        new_a = new_a / S_new

        a = new_a

        obj = compute_obj(a)
        if obj < best_obj - 1e-14:
            best_obj = obj
            best_a = a.copy()
            if verbose and (it % 500 == 0):
                print(f"      Nesterov it={it}: {best_obj:.6f}")

        # Periodic perturbation to escape local minima
        if it % 2000 == 0:
            rng = np.random.default_rng(it * 13 + 7)
            a = a * np.exp(rng.normal(0, 0.03, n))
            a = np.maximum(a, 0)
            a = a / np.sum(a)
            y = a.copy()
            t_momentum = 1.0

    return best_a


def subgradient_basic(a, max_iters, time_budget, topk=8, step_decay=0.999, verbose=False):
    """Basic subgradient with momentum and adaptive step size."""
    n = len(a)
    a = a.copy().astype(np.float64)
    S = np.sum(a)
    if S < 1e-10:
        return a
    a = a / S

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    v = np.zeros(n)
    momentum = 0.75
    step = 0.15
    stall = 0

    for it in range(max_iters):
        if time.monotonic() - start > time_budget:
            break

        conv = compute_conv(a)
        M = float(np.max(conv))
        active = np.argsort(conv)[-topk:]
        w = conv[active]
        w = w / max(np.sum(w), 1e-15)

        grad = np.zeros(n)
        for idx, k in enumerate(active):
            w_k = w[idx]
            lo = max(0, k - n + 1)
            hi = min(n - 1, k)
            for i in range(lo, hi + 1):
                j = k - i
                if 0 <= j < n:
                    grad[i] += 2.0 * a[j] * w_k

        grad_centered = grad - np.mean(grad)
        v = momentum * v + (1.0 - momentum) * grad_centered
        a_new = a - step * v
        a_new = np.maximum(a_new, 0)
        S_new = np.sum(a_new)
        if S_new < 1e-10:
            break
        a_new = a_new / S_new

        obj = compute_obj(a_new)
        if obj < best_obj - 1e-14:
            best_obj = obj
            best_a = a_new.copy()
            a = a_new
            stall = 0
            if verbose and (it % 500 == 0):
                print(f"      SG it={it}: {best_obj:.6f}")
        else:
            stall += 1
            step *= step_decay
            if stall > 500:
                rng = np.random.default_rng(it * 31 + 7)
                a = a * np.exp(rng.normal(0, 0.05, n))
                a = np.maximum(a, 0)
                a = a / np.sum(a)
                v = np.zeros(n)
                stall = 0
                step = 0.15 * 0.8
                if step < 1e-10:
                    break

    return best_a

def coordinate_descent(a, max_sweeps=5, time_budget=20.0, verbose=False):
    """Coordinate descent: optimize each coordinate one at a time."""
    n = len(a)
    a = a.copy().astype(np.float64)
    S = np.sum(a)
    if S < 1e-10:
        return a
    a = a / S

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    for sweep in range(max_sweeps):
        if time.monotonic() - start > time_budget:
            break
        improved = False
        rng = np.random.default_rng(sweep * 1000 + 42)
        order = rng.permutation(n)

        for idx in order:
            if time.monotonic() - start > time_budget:
                break

            old_val = a[idx]
            if old_val < 1e-15:
                continue

            # Try several candidate values for a[idx]
            # The effect on conv: conv[k] changes by (new-old)*a[k-idx] for valid k-idx
            # Try reducing it (most likely to help if it contributes to peaks)
            candidates = [old_val * 0.5, old_val * 0.7, old_val * 0.85,
                          old_val * 1.2, old_val * 1.5]

            best_cand = old_val
            best_cand_obj = compute_obj(a)  # current

            for c in candidates:
                a[idx] = c
                S_tmp = np.sum(a)
                if S_tmp < 1e-10:
                    a[idx] = old_val
                    continue
                a_norm = a / S_tmp
                obj_c = compute_obj(a_norm)
                if obj_c < best_cand_obj:
                    best_cand_obj = obj_c
                    best_cand = c
                a[idx] = old_val

            if best_cand != old_val:
                a[idx] = best_cand
                S_a = np.sum(a)
                if S_a > 1e-10:
                    a = a / S_a
                    improved = True
                    if compute_obj(a) < best_obj:
                        best_obj = compute_obj(a)
                        best_a = a.copy()
                        if verbose and sweep % 2 == 0:
                            print(f"      CD sweep={sweep}: {best_obj:.6f}")

        if not improved:
            break

    return best_a


def cosine_init(n, n_modes=8, seed=0):

    rng = np.random.default_rng(seed)
    k = np.arange(n)
    a = np.ones(n)
    for m in range(1, n_modes + 1):
        c = rng.uniform(-0.5, 0.5) / m
        phase = rng.uniform(0, 2 * np.pi)
        a += c * np.cos(2 * np.pi * m * k / n + phase)
    a = np.maximum(a, 0)
    S = np.sum(a)
    if S < 1e-10:
        return np.ones(n) / n
    return a / S


def smooth_random_init(n, seed=0, smoothness=3):
    """Generate a smooth random initialization."""
    rng = np.random.default_rng(seed)
    # Start with random values and smooth them
    a = rng.uniform(0.5, 1.5, n)
    # Apply Gaussian smoothing
    kernel_size = min(n, smoothness * 10 + 1)
    if kernel_size > 1:
        sigma = smoothness
        x = np.arange(kernel_size) - kernel_size // 2
        kernel = np.exp(-0.5 * (x / sigma) ** 2)
        kernel /= np.sum(kernel)
        a = np.convolve(a, kernel, mode='same')
    a = np.maximum(a, 1e-10)
    S = np.sum(a)
    return a / S


def endpoint_suppressed_init(n, seed=0):
    """Initialize with suppressed endpoints to reduce edge peaks."""
    k = np.arange(n)
    # Smooth profile with low edges
    x = 2.0 * k / (n - 1) - 1.0
    a = 1.0 - 0.5 * x ** 2
    # Add some random smooth perturbation
    rng = np.random.default_rng(seed)
    noise = rng.uniform(-0.1, 0.1, n)
    # Smooth noise
    if n > 5:
        x_k = np.arange(5) - 2
        kernel = np.exp(-0.5 * x_k ** 2)
        kernel /= np.sum(kernel)
        noise = np.convolve(noise, kernel, mode='same')
    a += noise
    a = np.maximum(a, 0)
    S = np.sum(a)
    if S < 1e-10:
        return np.ones(n) / n
    return a / S


def spectral_shape(a, n_iters=100, time_budget=15.0, verbose=False):
    """
    Spectral shaping: iteratively modify the phase/amplitude to reduce
    autocorrelation peaks. Uses the fact that we can adjust the signal
    in frequency domain to flatten its autocorrelation.
    """
    n = len(a)
    a = a.copy().astype(np.float64)
    S = np.sum(a)
    if S < 1e-10:
        return a
    a = a / S

    N = 1
    while N < 4 * n:
        N <<= 1

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    for it in range(n_iters):
        if time.monotonic() - start > time_budget:
            break

        # Compute autocorrelation via circular FFT
        A = np.fft.rfft(a, N)
        power = np.abs(A) ** 2
        r = np.fft.irfft(power, N)
        # Take the relevant part (linear conv)
        r_lin = r[:2 * n - 1]
        r_max = np.max(r_lin)

        # We want to reduce the peak. Strategy: where r is high,
        # we want to reduce the corresponding frequency content.
        # The relationship: r[k] = sum_f power[f] * cos(2*pi*f*k/N)
        # To reduce r at specific lags, we can reduce power at frequencies
        # that contribute positively to those lags.

        # Simple approach: identify the peak lags and try to reduce
        # their contribution by adjusting the amplitude spectrum
        peak_lags = np.where(r_lin > 0.95 * r_max)[0]

        # For each peak lag, find which frequencies contribute most positively
        freqs = np.fft.rfftfreq(N) * N  # 0, 1, 2, ...
        reduction = np.zeros(len(power))
        for k in peak_lags[:5]:  # Limit to top 5 peaks
            cos_vals = np.cos(2 * np.pi * freqs * k / N)
            reduction += np.maximum(cos_vals, 0)

        reduction /= max(np.max(reduction), 1e-10)

        # Reduce power at problematic frequencies
        alpha = 0.05  # Mixing parameter
        new_power = power * (1.0 - alpha * reduction)
        new_power = np.maximum(new_power, 0)

        # Get new signal (keep original phase)
        phase = np.angle(A)
        new_A = np.sqrt(new_power) * np.exp(1j * phase)
        a_new = np.fft.irfft(new_A, N)[:n]

        # Project: clip and normalize
        a_new = np.maximum(a_new, 0)
        S_new = np.sum(a_new)
        if S_new < 1e-10:
            break
        a_new = a_new / S_new

        obj_new = compute_obj(a_new)
        if obj_new < best_obj:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new
            if verbose and it % 20 == 0:
                print(f"      Spectral it={it}: {best_obj:.6f}")
        else:
            # Reduce alpha if not improving
            alpha *= 0.9
            if alpha < 1e-6:
                # Restart with fresh alpha
                alpha = 0.1

    return best_a


def refine_sequence(a, time_budget, verbose=False):
    """Full refinement pipeline combining multiple methods."""
    n = len(a)
    a = a.copy().astype(np.float64)
    S = np.sum(a)
    if S < 1e-10:
        return a, np.inf
    a = a / S

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    round_num = 0

    while time.monotonic() - start < time_budget - 3:
        round_num += 1
        remaining = time_budget - (time.monotonic() - start)
        if remaining < 5:
            break

        # 1. Nesterov subgradient
        sg_t = min(remaining * 0.25, 20.0)
        if sg_t > 2:
            a_sg = nesterov_subgradient(best_a, max_iters=int(sg_t * 50),
                                        time_budget=sg_t, topk=8)
            obj_sg = compute_obj(a_sg)
            if obj_sg < best_obj:
                best_obj = obj_sg
                best_a = a_sg.copy()
                if verbose:
                    print(f"    R{round_num} Nesterov: {best_obj:.6f}")

        # 2. Basic subgradient (different dynamics)
        remaining = time.monotonic() and (time_budget - (time.monotonic() - start))
        if remaining > 8:
            sg2_t = min(remaining * 0.2, 15.0)
            a_sg2 = subgradient_basic(best_a, max_iters=int(sg2_t * 60),
                                       time_budget=sg2_t, topk=6)
            obj_sg2 = compute_obj(a_sg2)
            if obj_sg2 < best_obj:
                best_obj = obj_sg2
                best_a = a_sg2.copy()
                if verbose:
                    print(f"    R{round_num} SG: {best_obj:.6f}")

        # 3. Spectral shaping
        remaining = time_budget - (time.monotonic() - start)
        if remaining > 8 and n <= 1000:
            spec_t = min(remaining * 0.15, 12.0)
            a_spec = spectral_shape(best_a, n_iters=200, time_budget=spec_t)
            obj_spec = compute_obj(a_spec)
            if obj_spec < best_obj:
                best_obj = obj_spec
                best_a = a_spec.copy()
                if verbose:
                    print(f"    R{round_num} Spectral: {best_obj:.6f}")

        # 4. LP-based alternating (for smaller n)
        remaining = time_budget - (time.monotonic() - start)
        if remaining > 15 and n <= 600:
            lp_t = min(remaining * 0.25, 30.0)
            a_lp = alternating_lp_improved(best_a, max_iters=40, time_budget=lp_t)
            obj_lp = compute_obj(a_lp)
            if obj_lp < best_obj:
                best_obj = obj_lp
                best_a = a_lp.copy()
                if verbose:
                    print(f"    R{round_num} LP: {best_obj:.6f}")

        # 5. Coordinate descent polish
        remaining = time_budget - (time.monotonic() - start)
        if remaining > 10 and n <= 800:
            cd_t = min(remaining * 0.1, 15.0)
            a_cd = coordinate_descent(best_a, max_sweeps=3, time_budget=cd_t)
            obj_cd = compute_obj(a_cd)
            if obj_cd < best_obj:
                best_obj = obj_cd
                best_a = a_cd.copy()
                if verbose:
                    print(f"    R{round_num} CD: {best_obj:.6f}")

        # 6. Perturbation escape
        remaining = time_budget - (time.monotonic() - start)
        if remaining > 10:
            rng = np.random.default_rng(round_num * 42 + 17)
            for pert_idx in range(2):
                if time.monotonic() - start > time_budget - 5:
                    break
                scale = rng.uniform(0.04, 0.1)
                a_pert = best_a * np.exp(rng.normal(0, scale, n))
                a_pert = np.maximum(a_pert, 0)
                S_p = np.sum(a_pert)
                if S_p < 1e-10:
                    continue
                a_pert = a_pert / S_p
                # Quick refinement
                a_pert = nesterov_subgradient(a_pert, max_iters=500,
                                              time_budget=min(3.0, remaining * 0.05),
                                              topk=6)
                obj_pert = compute_obj(a_pert)
                if obj_pert < best_obj:
                    best_obj = obj_pert
                    best_a = a_pert.copy()
                    if verbose:
                        print(f"    R{round_num} Pert{pert_idx}: {best_obj:.6f}")

    return best_a, best_obj


def try_parent():
    """Try to load parent sequence from global variable."""
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64)
        parent = np.clip(parent, 0, 1000.0)
        if parent.size > 0 and np.isfinite(parent).all():
            obj = compute_obj(parent)
            print(f"Parent (n={parent.size}): obj={obj:.6f}")
            return parent.copy(), obj
    except Exception:
        pass
    return None, np.inf


def propose_candidate(seed=42, budget_s=1500):
    """
    Main search function. Combines Nesterov subgradient, spectral shaping,
    LP alternating, coordinate descent, and multi-start exploration to
    minimize 2*n*max(conv(a,a))/(sum(a))^2.
    """
    start_time = time.monotonic()
    deadline = start_time + float(budget_s) - 30.0

    best_seq = None
    best_score = np.inf

    # Load parent
    parent, parent_obj = try_parent()
    if parent is not None and np.isfinite(parent_obj):
        best_score = parent_obj
        best_seq = parent.copy()
        n_parent = len(parent)
    else:
        n_parent = 500
        best_seq = None

    print(f"Initial best: {best_score:.6f} (n={n_parent})")

    # === Phase 1: Diverse initialization and quick optimization ===
    print("\n=== Phase 1: Diverse initializations ===")
    n_candidates = sorted(set([
        n_parent,
        max(100, n_parent - 100),
        min(800, n_parent + 100),
        min(1200, n_parent + 200),
        300,
        400,
    ]))
    n_candidates = [n for n in n_candidates if 100 <= n <= 1500]

    candidate_pool = []

    for n in n_candidates:
        if time.monotonic() > deadline:
            break
        remaining = deadline - time.monotonic()
        if remaining < 15:
            break

        inits = []

        # Cosine init
        inits.append((f"cos_{n}", cosine_init(n, n_modes=6, seed=seed + n)))
        # Endpoint suppressed
        inits.append((f"eps_{n}", endpoint_suppressed_init(n, seed=seed + n)))
        # Smooth random
        inits.append((f"smr_{n}", smooth_random_init(n, seed=seed + n, smoothness=3)))
        # Smooth random (more smooth)
        inits.append((f"smr2_{n}", smooth_random_init(n, seed=seed + n + 777, smoothness=5)))

        # If we have a parent, also try interpolation to this n
        if parent is not None and n != n_parent:
            x_old = np.linspace(0, 1, n_parent)
            x_new = np.linspace(0, 1, n)
            try:
                f_interp = interp1d(x_old, parent, kind='cubic',
                                    fill_value='extrapolate')
                a_interp = np.maximum(f_interp(x_new), 0)
                S_i = np.sum(a_interp)
                if S_i > 1e-10:
                    a_interp = a_interp / S_i
                    inits.append((f"interp_{n}", a_interp))
            except Exception:
                pass

        for name, a_init in inits:
            if time.monotonic() > deadline:
                break
            obj_init = compute_obj(a_init)

            # Quick Nesterov
            a_quick = nesterov_subgradient(a_init, max_iters=2000,
                                           time_budget=3.0, topk=6)
            obj_quick = compute_obj(a_quick)

            # Quick basic SG
            a_quick2 = subgradient_basic(a_init, max_iters=3000,
                                         time_budget=2.0, topk=5)
            obj_quick2 = compute_obj(a_quick2)

            # Take the best
            best_local = min(a_quick, a_quick2, a_init,
                             key=lambda x: compute_obj(x))
            best_local_obj = compute_obj(best_local)

            candidate_pool.append((best_local, best_local_obj, n))
            if best_local_obj < best_score:
                best_score = best_local_obj
                best_seq = best_local.copy()
                print(f"  {name} (n={n}): {best_local_obj:.6f} ** NEW BEST **")

    # Sort candidates
    candidate_pool.sort(key=lambda x: x[1])
    print(f"\n  Top candidate: obj={candidate_pool[0][1]:.6f} (n={candidate_pool[0][2]})")

    # === Phase 2: Refine top candidates ===
    print("\n=== Phase 2: Refine top candidates ===")
    n_refine = min(4, len(candidate_pool))
    for idx in range(n_refine):
        if time.monotonic() > deadline:
            break
        remaining = deadline - time.monotonic()
        if remaining < 30:
            break
        a_cand, obj_cand, n_cand = candidate_pool[idx]
        if obj_cand > best_score + 0.03:
            break

        alloc = min(remaining * 0.3, 80.0)
        print(f"  Refining candidate {idx} (n={n_cand}, obj={obj_cand:.6f}), alloc={alloc:.0f}s")

        a_ref, obj_ref = refine_sequence(a_cand, time_budget=alloc, verbose=True)
        if obj_ref < best_score:
            best_score = obj_ref
            best_seq = a_ref.copy()
            print(f"    ** NEW BEST: n={n_cand}, obj={best_score:.6f} **")

    # === Phase 3: Intensive refinement of global best ===
    print("\n=== Phase 3: Intensive refinement ===")
    a_cur = best_seq.copy()
    n_best = len(a_cur)

    for round_num in range(10):
        if time.monotonic() > deadline - 40:
            break
        remaining = deadline - time.monotonic()
        if remaining < 25:
            break

        alloc = min(remaining * 0.25, 60.0)
        a_new, obj_new = refine_sequence(a_cur, time_budget=alloc, verbose=False)

        if obj_new < best_score:
            best_score = obj_new
            best_seq = a_new.copy()
            a_cur = a_new.copy()
            print(f"  R{round_num}: {best_score:.6f} ** NEW BEST **")
        else:
            print(f"  R{round_num}: {obj_new:.6f} (best {best_score:.6f})")
            # Stronger perturbation
            if remaining > 30:
                rng = np.random.default_rng(round_num * 999 + 111)
                for pert_scale in [0.08, 0.15, 0.25]:
                    if time.monotonic() > deadline - 20:
                        break
                    a_pert = a_cur * np.exp(rng.normal(0, pert_scale, n_best))
                    a_pert = np.maximum(a_pert, 0)
                    S_p = np.sum(a_pert)
                    if S_p < 1e-10:
                        continue
                    a_pert = a_pert / S_p
                    a_pert = nesterov_subgradient(a_pert, max_iters=3000,
                                                  time_budget=min(5.0, remaining * 0.08),
                                                  topk=8)
                    obj_pert = compute_obj(a_pert)
                    if obj_pert < best_score:
                        best_score = obj_pert
                        best_seq = a_pert.copy()
                        a_cur = a_pert.copy()
                        print(f"  R{round_num} pert({pert_scale}): {best_score:.6f} ** NEW BEST **")

    # === Phase 4: Multi-n exploration ===
    print("\n=== Phase 4: Multi-n exploration ===")
    if time.monotonic() < deadline - 40:
        n_base = len(best_seq)
        n_tries = sorted(set([
            max(100, n_base - 200),
            max(100, n_base - 100),
            n_base + 100,
            n_base + 200,
            n_base + 400,
            max(100, n_base + 600),
        ]))
        n_tries = [n for n in n_tries if 100 <= n <= 2000]

        for n_try in n_tries:
            if time.monotonic() > deadline - 20:
                break
            remaining = deadline - time.monotonic()
            if remaining < 15:
                break

            x_old = np.linspace(0, 1, n_base)
            x_new = np.linspace(0, 1, n_try)
            try:
                f_interp = interp1d(x_old, best_seq, kind='cubic',
                                    fill_value='extrapolate')
                a_interp = np.maximum(f_interp(x_new), 0)
            except Exception:
                a_interp = np.ones(n_try) / n_try

            S_i = np.sum(a_interp)
            if S_i < 1e-10:
                continue
            a_interp = a_interp / S_i

            alloc = min(remaining * 0.3, 40.0)
            a_new, obj_new = refine_sequence(a_interp, time_budget=alloc, verbose=False)
            if obj_new < best_score:
                best_score = obj_new
                best_seq = a_new.copy()
                print(f"  n={n_try}: {best_score:.6f} ** NEW BEST **")
            else:
                print(f"  n={n_try}: {obj_new:.6f}")

    # === Phase 5: Final intensive polish ===
    print("\n=== Phase 5: Final polish ===")
    a_cur = best_seq.copy()
    n_final = len(a_cur)

    if time.monotonic() < deadline - 15:
        remaining = deadline - time.monotonic()

        # Extended Nesterov
        a_sg = nesterov_subgradient(a_cur, max_iters=int(remaining * 30),
                                     time_budget=remaining * 0.3, topk=10)
        obj_sg = compute_obj(a_sg)
        if obj_sg < best_score:
            best_score = obj_sg
            best_seq = a_sg.copy()
            a_cur = a_sg.copy()
            print(f"  Final Nesterov: {best_score:.6f} ** NEW BEST **")

        # Extended basic SG with different params
        remaining = deadline - time.monotonic()
        if remaining > 10:
            a_sg2 = subgradient_basic(a_cur, max_iters=int(remaining * 40),
                                       time_budget=remaining * 0.3, topk=5,
                                       step_decay=0.9995)
            obj_sg2 = compute_obj(a_sg2)
            if obj_sg2 < best_score:
                best_score = obj_sg2
                best_seq = a_sg2.copy()
                a_cur = a_sg2.copy()
                print(f"  Final SG: {best_score:.6f} ** NEW BEST **")

        # LP if small enough
        remaining = deadline - time.monotonic()
        if remaining > 15 and n_final <= 500:
            a_lp = alternating_lp_improved(a_cur, max_iters=60,
                                           time_budget=min(remaining * 0.4, 40.0),
                                           verbose=True)
            obj_lp = compute_obj(a_lp)
            if obj_lp < best_score:
                best_score = obj_lp
                best_seq = a_lp.copy()
                a_cur = a_lp.copy()
                print(f"  Final LP: {best_score:.6f} ** NEW BEST **")

        # Coordinate descent
        remaining = deadline - time.monotonic()
        if remaining > 8 and n_final <= 1000:
            a_cd = coordinate_descent(a_cur, max_sweeps=5,
                                      time_budget=min(remaining * 0.2, 15.0),
                                      verbose=True)
            obj_cd = compute_obj(a_cd)
            if obj_cd < best_score:
                best_score = obj_cd
                best_seq = a_cd.copy()
                print(f"  Final CD: {best_score:.6f} ** NEW BEST **")

        # Final perturbation attempts
        remaining = deadline - time.monotonic()
        if remaining > 10:
            rng = np.random.default_rng(12345)
            for i in range(4):
                if time.monotonic() > deadline - 5:
                    break
                scale = rng.uniform(0.02, 0.08)
                a_p = a_cur * np.exp(rng.normal(0, scale, n_final))
                a_p = np.maximum(a_p, 0)
                S_p = np.sum(a_p)
                if S_p > 1e-10:
                    a_p = a_p / S_p
                    a_p = nesterov_subgradient(a_p, max_iters=int(remaining),
                                               time_budget=min(remaining * 0.15, 10.0),
                                               topk=8)
                    a_p = coordinate_descent(a_p, max_sweeps=2,
                                             time_budget=min(remaining * 0.05, 5.0))
                    obj_p = compute_obj(a_p)
                    if obj_p < best_score:
                        best_score = obj_p
                        best_seq = a_p.copy()
                        print(f"  Final pert{i}: {best_score:.6f} ** NEW BEST **")

    # Final validation
    if best_seq is None:
        best_seq = np.ones(300) / 300
        best_score = compute_obj(best_seq)

    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    val = evaluate_sequence(result)
    if val < np.inf and val < best_score:
        best_score = val
    elif val == np.inf:
        result = [1.0] * 300
        best_score = 2.0

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_score:.6f}, time={elapsed:.1f}s ===")

    return result
