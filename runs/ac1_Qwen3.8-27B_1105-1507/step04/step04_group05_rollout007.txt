import numpy as np
from scipy.optimize import linprog
from scipy.sparse import lil_matrix
from scipy.optimize import minimize_scalar
import time


def fast_conv(a):
    """Full self-convolution via FFT."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]


def compute_J(a):
    """Compute J(a) = 2*n*max(conv(a,a))/(sum(a))^2."""
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))


def build_conv_matrix(a):
    """Build the convolution constraint matrix A where (A*g)[k] = (a*g)[k]."""
    n = len(a)
    A = lil_matrix((2 * n - 1, n))
    for k in range(2 * n - 1):
        jl = max(0, k - n + 1)
        jh = min(n - 1, k)
        for j in range(jl, jh + 1):
            A[k, j] = a[k - j]
    return A.tocsr()


def lp_best_response(a, M):
    """LP: max sum(g) s.t. (a*g)_k <= M for all k, g >= 0."""
    n = len(a)
    c = -np.ones(n)
    A = build_conv_matrix(a)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
    if r.success and r.x is not None:
        g = np.maximum(r.x, 0.0)
        if g.sum() > 1e-10:
            return g
    return None

def coordinate_descent_fast(a, n_passes=30, deadline=None, tracker=None):
    """Coordinate descent with greedy factor search per coordinate."""
    a = np.maximum(a.astype(np.float64), 0.0)
    S = a.sum()
    if S < 1e-10:
        return a
    n = len(a)
    b = fast_conv(a).copy()
    current_J = 2.0 * n * b.max() / (S * S)
    if tracker:
        tracker.update(a)

    for p in range(n_passes):
        if deadline and time.monotonic() > deadline:
            break
        improvement = 0.0
        for i in range(n):
            ai = a[i]
            if ai < 1e-14:
                continue
            min_k = i
            max_k = min(2 * n - 2, i + n - 1)
            best_trial = ai
            best_J = current_J

            # Try a range of factors
            factors = [0.3, 0.5, 0.7, 0.85, 0.95, 1.05, 1.15, 1.3, 1.5, 2.0]
            for f in factors:
                trial = ai * f
                delta = trial - ai
                if abs(delta) < 1e-14:
                    continue
                # Update affected b values
                for k in range(min_k, max_k + 1):
                    ki = k - i
                    if 0 <= ki < n:
                        if k == 2 * i:
                            b[k] += delta * (trial + ai)
                        else:
                            b[k] += 2.0 * delta * a[ki]

                new_S = S + delta
                if new_S < 1e-10:
                    # revert
                    for k in range(min_k, max_k + 1):
                        ki = k - i
                        if 0 <= ki < n:
                            if k == 2 * i:
                                b[k] -= delta * (trial + ai)
                            else:
                                b[k] -= 2.0 * delta * a[ki]
                    continue
                new_max = b.max()
                new_J = 2.0 * n * new_max / (new_S * new_S)

                if new_J < best_J - 1e-15:
                    best_J = new_J
                    best_trial = trial
                    # Keep the current b state (don't revert)
                else:
                    # Revert
                    for k in range(min_k, max_k + 1):
                        ki = k - i
                        if 0 <= ki < n:
                            if k == 2 * i:
                                b[k] -= delta * (trial + ai)
                            else:
                                b[k] -= 2.0 * delta * a[ki]

            if best_trial != ai:
                a[i] = best_trial
                S += best_trial - ai
                current_J = best_J
                improvement += 1.0

            break

def nesterov_subgradient(a, n_iters=500, deadline=None, tracker=None):
    """Nesterov-accelerated subgradient descent with adaptive step size."""
    a = np.maximum(a.astype(np.float64), 0.0)
    S = a.sum()
    if S < 1e-10:
        return a
    a = a / S
    n = len(a)

    t_k = 1.0
    t_prev = 1.0
    eta = 2.0  # initial step size
    a_prev = a.copy()
    current_J = compute_J(a)
    if tracker:
        tracker.update(a)

    for it in range(n_iters):
        if deadline and time.monotonic() > deadline:
            break

        # Nesterov extrapolation
        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_k * t_k))
        y = a + (t_prev / t_new) * (a - a_prev)
        y = np.maximum(y, 0.0)
        sy = y.sum()
        if sy < 1e-10:
            y = a.copy()
            sy = 1.0
        y = y / sy

        # Compute subgradient at y
        b = fast_conv(y)
        b_max = b.max()
        tol = 1e-6 * b_max
        active = np.where(b >= b_max - tol)[0]

        grad = np.zeros(n)
        n_active = len(active)
        for k in active:
            jl = max(0, k - n + 1)
            jh = min(n - 1, k)
            for j in range(jl, jh + 1):
                ki = k - j
                if 0 <= ki < n:
                    grad[j] += 2.0 * y[ki]
        grad /= n_active

        # Proximal gradient step with backtracking
        best_a = a
        best_J = current_J
        step = eta
        for ls in range(15):
            a_trial = y - step * grad
            a_trial = np.maximum(a_trial, 0.0)
            ts = a_trial.sum()
            if ts < 1e-10:
                step *= 0.5
                continue
            a_trial = a_trial / ts
            J_trial = compute_J(a_trial)
            if J_trial < best_J - 1e-15:
                best_J = J_trial
                best_a = a_trial
                break
            step *= 0.5

        a_prev = a
        a = best_a
        t_prev = t_k
        t_k = t_new
        current_J = best_J

        # Adaptively increase step size
        eta = min(eta * 1.05, 20.0)

        if tracker and it % 50 == 0:
            tracker.update(a)

    return a

            tighten = max(tighten * 0.5, tighten_min * 0.5)
            stall_count += 1
            if stall_count > 30:
                break
            continue

        g = np.maximum(g, 0.0)
        g_sum = g.sum()
        if g_sum < 1e-10:
            break
        g = g / g_sum

        # Backtracking line search for step size
        best_a = a
        best_J = current_J
        t_candidates = [0.8, 0.6, 0.4, 0.3, 0.2, 0.15, 0.1, 0.05, 0.03]

        for t in t_candidates:
            a_trial = (1.0 - t) * a + t * g
            a_trial = np.maximum(a_trial, 0.0)
            ts = a_trial.sum()
            if ts < 1e-10:
                continue
            a_trial = a_trial / ts
            J_trial = compute_J(a_trial)
            if J_trial < best_J - 1e-14:
                best_J = J_trial
                best_a = a_trial
                break
            # Keep going to try smaller steps

        if best_a is a:
            # No improvement, try very small step
            a_trial = (1.0 - 0.01) * a + 0.01 * g

                J_trial = compute_J(a_trial)

                    best_a = a_trial
                    best_J = J_trial
                else:
                    stall_count += 1
                    tighten *= 0.7
                    if stall_count > 40:
                        break
                    continue
            else:
                stall_count += 1
                if stall_count > 40:
                    break
                continue

        a = best_a
        current_J = best_J
        stall_count = 0

        # Adapt tightening

            tighten = min(tighten * 1.1, 0.08)
        else:

    return a


def block_coordinate_descent(a, block_size=3, n_passes=10, deadline=None, tracker=None):

def resample_seq(src, dest_n):
    """Resample 1D sequence to new length via linear interpolation."""
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)


def structured_init(n, rng):
    """Generate a structured initial sequence."""
    t = np.linspace(0, 1, n, endpoint=False)
    inits = []
    inits.append(np.ones(n))
    inits.append(1.0 + 0.3 * np.cos(2 * np.pi * 3 * t))
    inits.append(1.0 + 0.2 * np.cos(2 * np.pi * 5 * t) + 0.1 * np.cos(2 * np.pi * 11 * t))
    inits.append(np.abs(np.sin(np.pi * t)) + 0.5)
    inits.append(1.0 - 0.4 * np.cos(2 * np.pi * 2 * t))
    # Fejer-like kernel
    inits.append(np.abs(np.sinc(t - 0.5) ** 2) + 0.1)
    # Random
    inits.append(rng.uniform(0.3, 1.0, n))
    return [np.maximum(x, 0.01) for x in inits]

    remaining = deadline - time.monotonic()
    if remaining > 40:
        print(f"\n{'='*50} Phase 5: Nearby Lengths {'='*50}", flush=True)
        n_best = tracker.best.size
        nearby = [n_best - 2, n_best + 2, n_best - 4, n_best + 4, n_best + 6]
        seen = set()
        for nl in nearby:
            if nl in seen or nl < 200 or nl > 600:
                continue
            seen.add(nl)
            if time.monotonic() > deadline - 20:
                break
            a_res = resample_seq(tracker.best, nl)
            a_res = np.maximum(a_res, 0.001)
            a_res = a_res / a_res.sum()
            a_res = coordinate_descent_fast(a_res, n_passes=5,
                                           deadline=time.monotonic() + 20, tracker=tracker)
            # Quick LP
            if time.monotonic() < deadline - 15:
                a_res = lp_mirror_descent(a_res, max_iters=15,
                                          deadline=time.monotonic() + 15,
                                          tracker=tracker, tighten_start=0.02)
            print(f"    n={nl}: best J={tracker.best_score:.8f}", flush=True)

    # === Phase 6: Perturbation restarts ===
    remaining = deadline - time.monotonic()
    if remaining > 30:
        print(f"\n{'='*50} Phase 6: Perturbation Restarts {'='*50}", flush=True)
        n_best = tracker.best.size
        for temp in [0.03, 0.02, 0.01]:
            if time.monotonic() > deadline - 15:
                break
            a_pert = tracker.best + rng.normal(0, temp, n_best)
            a_pert = np.maximum(a_pert, 0.001)
            a_pert = a_pert / a_pert.sum()
            a_pert = coordinate_descent_fast(a_pert, n_passes=5,
                                           deadline=time.monotonic() + 15, tracker=tracker)
            if time.monotonic() < deadline - 10:
                a_pert = lp_mirror_descent(a_pert, max_iters=10,
                                          deadline=time.monotonic() + 10,
                                          tracker=tracker, tighten_start=0.015)

    # Final verification
    if tracker.best is None:
        tracker.best = np.ones(256)
        tracker.best_score = compute_J(tracker.best)

    # One final CD pass on the best
    a_final = tracker.best.copy()
    a_final = coordinate_descent_fast(a_final, n_passes=5, deadline=deadline, tracker=tracker)
    tracker.update(a_final)

    print(f"\n{'='*50} FINAL RESULT {'='*50}", flush=True)
    print(f"J = {tracker.best_score:.10f}, n = {tracker.best.size}", flush=True)
    print(f"Total time: {time.monotonic()-tracker.t_start:.1f}s", flush=True)

    return [float(x) for x in tracker.best]
