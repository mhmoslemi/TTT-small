    while s < 2 * n - 1:

def compute_J(a):

    s = a.sum()

def multiplicative_weights_step(a, eta, tol=0.01):
    """One multiplicative weights step."""
    n = len(a)
    b = fast_conv(a)
    M = b.max()
    active = np.where(b >= M * (1 - tol))[0]
    
    grad = np.zeros(n)
    for k in active:
        jl = max(0, k - n + 1)
        jh = min(n - 1, k)
        for j in range(jl, jh + 1):
            ki = k - j
            if 0 <= ki < n:
                grad[j] += a[ki]
    
    # Multiplicative update
    a_new = a * np.exp(-eta * grad)
    a_new = np.maximum(a_new, 1e-15)
    a_new /= a_new.sum()

    a /= a.sum()
    current_J = compute_J(a)
    tracker.update(a)
    
    stall = 0
    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        best_a = a

            eta = base_eta * eta_mult

                best_a = a_trial

            if stall > 30:

        else:
            a = best_a

Minimizes J(a) = 2n*max(a*a)/(sum(a))^2 using:
1. Multiplicative weights (exponentiated gradient) with adaptive step size
2. LP best response with continuation method
3. Coordinate descent with incremental updates
4. Multi-length exploration

import numpy as np
from scipy.optimize import linprog
from scipy.sparse import csr_matrix
import time

def fast_conv(a):
    n = len(a)
    s = 1
    while s < 2 * n - 1:
        s <<= 1
    fa = np.fft.rfft(a, s)
    return np.fft.irfft(fa * fa, s)[:2 * n - 1]

def compute_J(a):
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))

def build_A_sparse(a):
    n = len(a)
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        l_min = max(0, k - n + 1)
        l_max = min(n - 1, k)
        for l in range(l_min, l_max + 1):
            rows.append(k)
            cols.append(l)
            vals.append(a[k - l])
    A = csr_matrix((np.array(vals), (np.array(rows, dtype=int), np.array(cols, dtype=int))),
                   shape=(2 * n - 1, n))
    return A

def lp_best_response(a, M):
    n = len(a)
    c = -np.ones(n)
    A = build_A_sparse(a)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                options={'presolve': True})
    if r.success and r.x is not None:
        g = np.maximum(r.x, 0.0)
        if g.sum() > 1e-10:
            return g
    return None

def mw_grad(a, tol=0.01):
    """Compute subgradient of max(a*a) at a."""
    n = len(a)
    b = fast_conv(a)
    M = b.max()
    active = np.where(b >= M * (1 - tol))[0]
    grad = np.zeros(n)
    for k in active:
        l_min = max(0, k - n + 1)
        l_max = min(n - 1, k)
        for l in range(l_min, l_max + 1):
            grad[l] += a[k - l]
    return grad, M, active

def resample_seq(src, dest_n):
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)

"""
Minimizes J(a) = 2n*max(convolve(a,a))/(sum(a))^2 using:
1. Multiplicative weights (exponentiated gradient) with adaptive step size and line search
2. LP best response with continuation method and warm-starting
3. Coordinate descent with incremental convolution updates
4. Alternating MW+LP for escaping local minima
5. Multi-length exploration and perturbation restarts
Key innovations: faster MW iterations, adaptive LP tightening via duality,
and a continuation strategy that gradually reduces the LP bound.
"""

import numpy as np
from scipy.optimize import linprog
from scipy.sparse import csr_matrix
import time


def fast_conv(a):
    """Full self-convolution via FFT."""
    n = len(a)
    s = 1
    while s < 2 * n - 1:
        s <<= 1
    fa = np.fft.rfft(a, s)
    return np.fft.irfft(fa * fa, s)[:2 * n - 1]


def compute_J(a):
    """Compute J(a) = 2*n*max(conv(a,a))/(sum(a))^2."""
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))


def build_A_sparse(a):
    """Build sparse constraint matrix for LP: (a*g)_k <= M for all k."""
    n = len(a)
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        l_min = max(0, k - n + 1)
        l_max = min(n - 1, k)
        for l in range(l_min, l_max + 1):
            rows.append(k)
            cols.append(l)
            vals.append(a[k - l])
    A = csr_matrix((np.array(vals), (np.array(rows, dtype=np.int32), np.array(cols, dtype=np.int32))),
                   shape=(2 * n - 1, n))
    return A


def lp_best_response(a, M):
    """LP: max sum(g) s.t. (a*g)_k <= M for all k, g >= 0."""
    n = len(a)
    c = -np.ones(n)
    A = build_A_sparse(a)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                options={'presolve': True, 'time_limit': 5.0})
    if r.success and r.x is not None:
        g = np.maximum(r.x, 0.0)
        if g.sum() > 1e-10:
            return g
    elif r.status == 8:  # time limit reached, try to use best bound
        if hasattr(r, 'x') and r.x is not None:
            g = np.maximum(r.x, 0.0)
            if g.sum() > 1e-10:
                return g
    return None


def mw_compute_grad(a, tol=0.02):
    """Compute subgradient of max(a*a) at a, along with active set size."""
    n = len(a)
    b = fast_conv(a)
    M = b.max()
    threshold = M * (1.0 - tol)
    active = np.where(b >= threshold)[0]
    grad = np.zeros(n)
    for k in active:
        l_min = max(0, k - n + 1)
        l_max = min(n - 1, k)
        for l in range(l_min, l_max + 1):
            grad[l] += a[k - l]
    return grad, M, len(active)


def resample_seq(src, dest_n):
    """Resample 1D sequence to new length via linear interpolation."""
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)


class ScoreTracker:
    def __init__(self):
        self.best = None
        self.best_score = float('inf')
        self.t_start = time.monotonic()

    def update(self, a):

def run_mw_optimization(a, deadline, tracker, max_iters=50000, base_eta=0.03, tol=0.02):
    """Multiplicative weights optimization with adaptive step size and line search."""
    a = np.maximum(a.astype(np.float64), 1e-10)
    a /= a.sum()
    current_J = compute_J(a)
    tracker.update(a)

    eta = base_eta
    stall = 0
    it = 0
    n = len(a)

    while time.monotonic() < deadline and it < max_iters:
        grad, M, n_active = mw_compute_grad(a, tol)
        gn = np.linalg.norm(grad)
        if gn < 1e-15:
            break

        # Line search over step sizes
        best_J = current_J
        best_a = None
        for mult in [0.5, 1.0, 1.5, 2.0, 3.0]:
            eta_t = eta * mult
            log_a = np.log(a) - eta_t * grad
            log_a -= log_a.max()
            a_trial = np.exp(log_a)
            a_trial = np.maximum(a_trial, 1e-15)
            a_trial /= a_trial.sum()
            J_trial = compute_J(a_trial)
            if J_trial < best_J - 1e-13:
                best_J = J_trial
                best_a = a_trial

        if best_a is None:
            stall += 1
            if stall > 80:
                break
            eta *= 0.7
            if eta < 1e-5:
                break
        else:
            a = best_a
            current_J = best_J
            stall = 0
            eta = min(eta * 1.05, 3.0)

        it += 1
        if it % 200 == 0:
            print(f"    MW iter {it}: J={current_J:.8f} eta={eta:.4f} |act|={n_active}", flush=True)
            tracker.update(a)

    return a


def run_lp_optimization(a, deadline, tracker, max_iters=500, tighten_base=0.005):
    """LP best response with adaptive tightening."""
    a = np.maximum(a.astype(np.float64), 1e-10)
    a /= a.sum()
    n = len(a)
    current_J = compute_J(a)
    tracker.update(a)

    tighten = tighten_base
    stall = 0
    it = 0

    while time.monotonic() < deadline and it < max_iters:
        b = fast_conv(a)
        M_current = float(b.max())
        M_target = M_current * (1.0 - tighten)

        g = lp_best_response(a, M_target)
        if g is None:
            # Try less aggressive
            M_target = M_current * (1.0 - tighten * 0.3)
            g = lp_best_response(a, M_target)
            if g is None:
                tighten *= 0.7
                stall += 1
                if stall > 40:
                    break
                continue

        g = np.maximum(g, 0.0)
        g_sum = g.sum()
        if g_sum < 1e-10:
            break
        g /= g_sum

        # Line search on step size
        progress = min(1.0, it / max(1, max_iters))
        base_t = max(0.02, 0.6 * (1.0 - progress * 0.5))
        candidates_t = [base_t * 0.5, base_t, base_t * 1.5, base_t * 2.0, 0.1, 0.05]

        best_a = None
        best_J = current_J
        for t in candidates_t:
            a_trial = (1.0 - t) * a + t * g
            a_trial = np.maximum(a_trial, 0.0)
            s = a_trial.sum()
            if s < 1e-10:
                continue
            a_trial /= s
            J_trial = compute_J(a_trial)
            if J_trial < best_J - 1e-13:
                best_J = J_trial
                best_a = a_trial

        if best_a is None:
            stall += 1
            if stall > 50:
                break
            tighten *= 0.8
        else:
            a = best_a
            current_J = best_J
            stall = 0
            # Adapt tightening
            tighten = min(tighten * 1.02, 0.08)

        it += 1
        if it % 20 == 0:
            print(f"    LP iter {it}: J={current_J:.8f} tighten={tighten:.5f}", flush=True)
            tracker.update(a)

    tracker.update(a)
    return a


def cd_compute_f(a, i, b, S):
    """Objective function value when a[i]=x, others fixed (for CD)."""
    n = len(a)
    ai = a[i]

    def f(x):
        new_S = S - ai + x
        if new_S < 1e-10:
            return 1e10
        max_c = 0.0
        for k in range(2 * n - 1):
            if i <= k <= i + n - 1:
                ki = k - i
                if 0 <= ki < n:
                    if k == 2 * i:
                        ck = b[k] - ai * ai + x * x
                    else:
                        ck = b[k] - 2.0 * ai * a[ki] + 2.0 * x * a[ki]
                else:
                    ck = b[k]
            else:
                ck = b[k]
            if ck > max_c:
                max_c = ck
        return 2.0 * n * max_c / (new_S * new_S)

    return f


def coordinate_descent_pass(a, max_coords=None):
    """One full pass of coordinate descent with golden section search."""
    a = np.maximum(a.astype(np.float64), 0.0)
    S = a.sum()
    if S < 1e-10:
        return a, compute_J(a)
    n = len(a)
    b = fast_conv(a)
    current_J = 2.0 * n * b.max() / (S * S)

    n_coords = n if max_coords is None else min(max_coords, n)
    indices = list(range(n))
    np.random.shuffle(indices)
    indices = indices[:n_coords]

    gr = (np.sqrt(5.0) - 1.0) / 2.0

    for i in indices:
        ai = a[i]
        S = a.sum()
        b = fast_conv(a)  # Recompute for safety (slower but correct)

        def f(x):
            new_S = S - ai + x
            if new_S < 1e-10:
                return 1e10
            a[i] = x
            new_b = fast_conv(a)
            a[i] = ai
            return 2.0 * n * new_b.max() / (new_S * new_S)

        lo = 0.0
        hi = max(3.0 * ai, S / n * 3.0, 0.1)
        c_pt = hi - gr * (hi - lo)
        d_pt = lo + gr * (hi - lo)
        fc = f(c_pt)
        fd = f(d_pt)

        for _ in range(30):
            if fc < fd:
                hi = d_pt
                d_pt = c_pt
                fd = fc
                c_pt = hi - gr * (hi - lo)
                fc = f(c_pt)
            else:
                lo = c_pt
                c_pt = d_pt
                fc = fd
                d_pt = lo + gr * (hi - lo)
                fd = f(d_pt)

        x_opt = max(0.0, (lo + hi) / 2.0)
        a[i] = x_opt

    b_final = fast_conv(a)
    S_final = a.sum()
    new_J = 2.0 * n * b_final.max() / (S_final * S_final)
    return a, new_J


def run_cd(a, n_passes, deadline, tracker):
    """Multiple passes of coordinate descent."""
    a = np.maximum(a.astype(np.float64), 0.0)
    a /= a.sum()
    current_J = compute_J(a)
    tracker.update(a)

    for p in range(n_passes):
        if time.monotonic() > deadline:
            break
        a, new_J = coordinate_descent_pass(a)
        a /= a.sum()
        if new_J < current_J - 1e-12:
            current_J = new_J
            tracker.update(a)
            if p % 3 == 0:
                print(f"    CD pass {p}: J={current_J:.8f}", flush=True)
        else:
            if p > 4:
                break

    tracker.update(a)
    return a


def run_mw_lp_alternating(a, deadline, tracker, n_cycles=20):
    """Alternate between MW and LP for better convergence."""
    a = np.maximum(a.astype(np.float64), 1e-10)
    a /= a.sum()
    current_J = compute_J(a)
    tracker.update(a)
    n = len(a)

    for cycle in range(n_cycles):
        if time.monotonic() > deadline:
            break

        # MW phase (short)
        mw_deadline = min(deadline, time.monotonic() + max(2, (deadline - time.monotonic()) * 0.3))
        a = run_mw_optimization(a, mw_deadline, tracker, max_iters=5000, base_eta=0.02, tol=0.03)
        a /= a.sum()

        # LP phase (short)
        lp_deadline = min(deadline, time.monotonic() + max(3, (deadline - time.monotonic()) * 0.5))
        a = run_lp_optimization(a, lp_deadline, tracker, max_iters=30, tighten_base=0.004)
        a /= a.sum()

        J_now = compute_J(a)
        if cycle % 5 == 0:
            print(f"    MLP cycle {cycle}: J={J_now:.8f}", flush=True)

    tracker.update(a)
    return a


def generate_inits(n, rng):
    """Generate diverse initial sequences of length n."""
    inits = []
    t = np.linspace(0, 1, n, endpoint=False)

    inits.append(np.ones(n))
    inits.append(rng.uniform(0.3, 1.0, n))

    for freq in [2, 3, 5, 7, 11]:
        inits.append(1.0 + 0.3 * np.cos(2.0 * np.pi * freq * t + rng.uniform(0, 2 * np.pi)))

    inits.append(1.0 + 0.2 * np.cos(2 * np.pi * 3 * t) + 0.15 * np.cos(2 * np.pi * 8 * t))
    inits.append(np.abs(np.sin(np.pi * t)) + 0.3)
    inits.append(0.5 + 0.5 * np.cos(2 * np.pi * t) ** 2)
    inits.append(1.0 - 0.4 * np.abs(2 * t - 1.0))
    inits.append(0.4 * np.exp(-8 * (t - 0.3) ** 2) + 0.4 * np.exp(-8 * (t - 0.7) ** 2) + 0.2)

    # Random smooth
    smooth = np.convolve(rng.standard_normal(n), np.ones(20) / 20, mode='same')
    inits.append(np.abs(smooth - smooth.min()) + 0.1)

    # Log-uniform random
    inits.append(10.0 ** rng.uniform(-1, 1, n))

    return inits


def propose_candidate(seed=42, budget_s=1500, **kwargs):
    """
    Main search function combining MW, LP, coordinate descent,
    multi-length exploration, and perturbation restarts.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + float(budget_s) - 30.0
    tracker = ScoreTracker()

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        if p.ndim == 1 and p.size > 0 and np.all(np.isfinite(p)):
            parent = np.maximum(p, 0.0)
            tracker.update(parent)
            print(f"Parent loaded: n={parent.size}, J={tracker.best_score:.8f}", flush=True)
    except (NameError, AttributeError):
        pass

    if parent is None:
        parent = np.ones(384)
        tracker.update(parent)
        print(f"No parent found, using uniform n={parent.size}", flush=True)

    n_parent = parent.size
    print(f"Starting optimization, target < 1.5027, current best = {tracker.best_score:.8f}", flush=True)

    # === Phase 1: MW on parent (bulk optimization) ===
    print(f"\n{'='*50} Phase 1: MW on parent (n={n_parent}) {'='*50}", flush=True)
    t0 = time.monotonic()
    phase1_deadline = time.monotonic() + (deadline - time.monotonic()) * 0.25

    a = parent.copy()
    a = run_mw_optimization(a, phase1_deadline, tracker, max_iters=100000, base_eta=0.03, tol=0.02)
    tracker.report(f"Phase 1 MW: {time.monotonic()-t0:.1f}s")

    # === Phase 2: LP refinement ===
    if time.monotonic() < deadline - 200:
        print(f"\n{'='*50} Phase 2: LP Refinement {'='*50}", flush=True)
        t0 = time.monotonic()
        phase2_deadline = time.monotonic() + (deadline - time.monotonic()) * 0.25

        a = run_lp_optimization(tracker.best, phase2_deadline, tracker, max_iters=300, tighten_base=0.005)
        tracker.report(f"Phase 2 LP: {time.monotonic()-t0:.1f}s")

    # === Phase 3: Alternating MW+LP ===
    if time.monotonic() < deadline - 150:
        print(f"\n{'='*50} Phase 3: Alternating MW+LP {'='*50}", flush=True)
        t0 = time.monotonic()
        phase3_deadline = time.monotonic() + (deadline - time.monotonic()) * 0.20

        a = run_mw_lp_alternating(tracker.best, phase3_deadline, tracker, n_cycles=30)
        tracker.report(f"Phase 3 MLP: {time.monotonic()-t0:.1f}s")

    # === Phase 4: Coordinate descent polish ===
    if time.monotonic() < deadline - 100:
        print(f"\n{'='*50} Phase 4: Coordinate Descent {'='*50}", flush=True)
        t0 = time.monotonic()
        phase4_deadline = time.monotonic() + (deadline - time.monotonic()) * 0.15

        a = run_cd(tracker.best, n_passes=10, deadline=phase4_deadline, tracker=tracker)
        tracker.report(f"Phase 4 CD: {time.monotonic()-t0:.1f}s")

    # === Phase 5: Multi-length exploration ===
    if time.monotonic() < deadline - 60:
        print(f"\n{'='*50} Phase 5: Multi-length Exploration {'='*50}", flush=True)
        t0 = time.monotonic()
        phase5_deadline = deadline - 40

        n = tracker.best.size
        nearby_lengths = []
        for delta in [-12, -8, -6, -4, -2, 2, 4, 6, 8, 12, 16, 20, -20, 32, -32]:
            nl = n + delta
            if 150 <= nl <= 800:
                nearby_lengths.append(nl)
        # Remove duplicates
        seen = set()
        unique_lengths = []
        for l in nearby_lengths:
            if l not in seen:
                seen.add(l)
                unique_lengths.append(l)

        for nl in unique_lengths[:12]:
            if time.monotonic() > phase5_deadline:
                break

            a_resampled = resample_seq(tracker.best, nl)
            a_resampled = np.maximum(a_resampled, 0.001)
            a_resampled /= a_resampled.sum()

            # Quick MW
            a_trial = run_mw_optimization(a_resampled, min(phase5_deadline, time.monotonic() + 15),
                                          tracker, max_iters=5000, base_eta=0.03, tol=0.02)
            a_trial /= a_trial.sum()

            # Quick LP
            if time.monotonic() < phase5_deadline - 3:
                a_trial = run_lp_optimization(a_trial, min(phase5_deadline, time.monotonic() + 10),
                                              tracker, max_iters=20, tighten_base=0.004)
                a_trial /= a_trial.sum()
                tracker.update(a_trial)

            print(f"    Length {nl}: best J={tracker.best_score:.8f}", flush=True)

        tracker.report(f"Phase 5 Lengths: {time.monotonic()-t0:.1f}s")

    # === Phase 6: Perturbation restarts ===
    if time.monotonic() < deadline - 30:
        print(f"\n{'='*50} Phase 6: Perturbation Restarts {'='*50}", flush=True)
        t0 = time.monotonic()
        phase6_deadline = deadline - 10

        n = tracker.best.size
        for temp in [0.08, 0.05, 0.03, 0.02, 0.01]:
            if time.monotonic() > phase6_deadline:
                break

            a_pert = tracker.best.copy() + rng.normal(0, temp, n)
            a_pert = np.maximum(a_pert, 1e-6)
            a_pert /= a_pert.sum()

            # Short MW
            a_pert = run_mw_optimization(a_pert, min(phase6_deadline, time.monotonic() + 10),
                                         tracker, max_iters=3000, base_eta=0.02, tol=0.03)
            a_pert /= a_pert.sum()

            # Short LP
            if time.monotonic() < phase6_deadline - 2:
                a_pert = run_lp_optimization(a_pert, min(phase6_deadline, time.monotonic() + 8),
                                             tracker, max_iters=15, tighten_base=0.003)
                a_pert /= a_pert.sum()
                tracker.update(a_pert)

        tracker.report(f"Phase 6 Restarts: {time.monotonic()-t0:.1f}s")

    # === Phase 7: Final intensive polish on best ===
    if time.monotonic() < deadline - 5:
        print(f"\n{'='*50} Phase 7: Final Polish {'='*50}", flush=True)
        t0 = time.monotonic()

        # Final LP
        a_final = run_lp_optimization(tracker.best, deadline - 3, tracker,
                                       max_iters=30, tighten_base=0.002)
        a_final /= a_final.sum()
        tracker.update(a_final)

        # Final CD (1 pass)
        if time.monotonic() < deadline - 2:
            a_final, _ = coordinate_descent_pass(a_final)
            a_final /= a_final.sum()
            tracker.update(a_final)

        tracker.report(f"Phase 7 Final: {time.monotonic()-t0:.1f}s")

    # Fallback
    if tracker.best is None:
        tracker.best = np.ones(256)
        tracker.best_score = compute_J(tracker.best)

    print(f"\n{'='*50} FINAL RESULT {'='*50}", flush=True)
    print(f"J = {tracker.best_score:.10f}, n = {tracker.best.size}", flush=True)
    print(f"Total time: {time.monotonic()-tracker.t_start:.1f}s", flush=True)

    return [float(x) for x in tracker.best]
