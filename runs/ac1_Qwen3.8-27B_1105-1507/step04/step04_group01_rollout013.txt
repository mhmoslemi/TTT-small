"""
Spectral Hankel iteration + LP polishing for minimizing 2n*max(conv(a,a))/(sum(a))^2.
Uses the structure that the weighted sum of convolution values defines a Hankel matrix,
whose minimum eigenvector gives the optimal direction. Combined with LP steps and
multiple restarts for global optimization.

import numpy as np
from scipy.optimize import linprog
from scipy.fft import rfft, irfft
from scipy.sparse.linalg import eigsh
from scipy.sparse import diags
import time

def compute_conv(a):
    n = len(a)
    size = 1
    while size < 2*n - 1:
        size <<= 1
    fa = rfft(a, size)
    conv = irfft(fa * fa, size)
    return conv[:2*n-1]

def compute_obj(a):
    n = len(a)
    if n == 0:
        return np.inf
    conv = compute_conv(a)
    max_b = float(np.max(conv))
    sum_a = float(np.sum(a))
    if sum_a < 0.01:
        return np.inf
    return 2.0 * n * max_b / (sum_a * sum_a)

def build_hankel(w, n):
    """Build n x n Hankel matrix from weight vector w of length >= 2n-1."""
    # W[i,j] = w[i+j]
    # Use the fact that Hankel matrix is determined by first row and first column
    first_col = w[:n]  # W[0,0], W[1,0], ..., W[n-1,0] = w[0], w[1], ..., w[n-1]
    first_row = w[1:2*n-1]  # W[0,1], W[0,2], ..., W[0,n-1] = w[1], w[2], ..., w[n-1]
    # Actually first_row[k] = W[0,k] = w[0+k] = w[k] for k=1,...,n-1
    # So first_row = w[1:n]
    # And first_col[k] = W[k,0] = w[k+0] = w[k] for k=0,...,n-1
    # So first_col = w[0:n]
    
    # Efficient construction using the anti-diagonal structure
    # W[i,j] = w[i+j]
    idx_i, idx_j = np.meshgrid(np.arange(n), np.arange(n), indexing='ij')
    W = w[idx_i + idx_j]
    return W

def spectral_step(a, tau=1.0, beta=2.0):
    """One spectral Hankel iteration step. Returns new a."""
    n = len(a)
    a = a / np.sum(a)
    c = compute_conv(a)
    c_max = float(np.max(c))
    if c_max < 1e-30:
        return a
    
    # Compute weights
    if tau > 0:
        shifted = (c - c_max) / tau
        w = np.exp(shifted)
        w_sum = np.sum(w)
        if w_sum < 1e-300:
            w = np.zeros(len(c))
            w[int(np.argmax(c))] = 1.0
        else:
            w = w / w_sum
    else:
        # Power-based weights
        w = (c / c_max) ** beta
        w_sum = np.sum(w)
        if w_sum < 1e-300:
            w = np.zeros(len(c))
            w[int(np.argmax(c))] = 1.0
        else:
            w = w / w_sum
    
    # Build Hankel matrix
    W = build_hankel(w, n)
    
    # Find minimum eigenvector
    try:
        # Use dense eigensolver for n up to ~2000
        if n <= 2000:
            eigvals, eigvecs = np.linalg.eigh(W)
            v = eigvecs[:, 0]  # minimum eigenvalue eigenvector
        else:
            W_sparse = eigsh(W, k=1, which='SM')
            v = W_sparse[1]
    except Exception:
        return a
    
    # Project to non-negative
    v = np.maximum(v, 0)
    v_sum = np.sum(v)
    if v_sum < 1e-10:
        # Try the absolute value
        v = np.abs(eigvecs[:, 0]) if n <= 2000 else np.abs(v)
        v_sum = np.sum(v)
        if v_sum < 1e-10:
            return a
    v = v / v_sum
    
    return v

def spectral_iterate(a, n_steps=100, tau_start=None, tau_end=None, beta=2.0,

def lp_direction(a, M, time_limit=15):
    """Solve LP: max sum(g) s.t. (a*g)[k] <= M, g >= 0. Normalize g to sum=1."""
    n = len(a)
    # Build constraint matrix: (a*g)[k] <= M for k=0,...,2n-2
    # (a*g)[k] = sum_{i+j=k} a[i]*g[j]... wait no.
    # convolve(a, g)[k] = sum_i a[i] * g[k-i]
    # So the constraint is: for each k, sum_{i: 0<=i<n, 0<=k-i<n} a[i] * g[k-i] <= M
    # Let j = k-i, so: sum_{j: max(0,k-n+1)<=j<=min(k,n-1)} a[k-j] * g[j] <= M
    
    # For each k, the coefficient of g[j] is a[k-j] (if 0<=k-j<n)
    A = np.zeros((2*n-1, n))
    for k in range(2*n-1):
        for j in range(n):
            i = k - j
            if 0 <= i < n:
                A[k, j] = a[i]
    
    c = -np.ones(n)
    b_ub = np.full(2*n-1, M)
    bounds = [(0.0, None)] * n
    
    try:
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': time_limit})
        if res.status == 0:
            g = np.maximum(res.x, 0)
            s = np.sum(g)
            if s > 1e-10:
                return g / s
    except Exception:
        pass
    return None

def lp_step(a, time_budget=20.0):
    """LP-based step: find g from LP, blend with a."""
    n = len(a)
    a = a / np.sum(a)
    c = compute_conv(a)
    M = float(np.max(c))
    
    g = lp_direction(a, M, time_limit=min(15, time_budget - 2))
    if g is None:
        return a
    
    # Try different blend factors
    best_a = a.copy()
    best_obj = compute_obj(a)
    for t in [0.2, 0.3, 0.5, 0.7, 1.0]:
        a_try = (1-t) * a + t * g
        a_try = np.maximum(a_try, 0)
        s = np.sum(a_try)
        if s < 1e-10:
            continue
        a_try = a_try / s
        obj = compute_obj(a_try)
        if obj < best_obj:
            best_obj = obj
            best_a = a_try.copy()
    
    return best_a
