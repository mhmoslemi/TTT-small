"""
Minimizes J(a) = 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Nesterov-accelerated smooth gradient descent with adaptive log-sum-exp

import numpy as np
from scipy.optimize import minimize
import time

def fast_conv(a):
    """Full self-convolution via FFT."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]

def compute_J(a):
    """Compute J(a) = 2*n*max(conv(a,a))/(sum(a))^2."""
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))

def smooth_max_and_grad(a, tau):
    """
    Compute smooth max of conv(a,a) and its gradient w.r.t. a.
    Returns (smooth_max_value, gradient, conv_values).
    """
    n = len(a)
    b = fast_conv(a)
    b_max = b.max()
    exp_b = np.exp(tau * (b - b_max))
    p = exp_b / exp_b.sum()
    smooth_val = (b_max + np.log(exp_b.sum())) / tau
    
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        pk = p[k]
        if pk < 1e-14:
            continue
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        grad[lo:hi + 1] += 2.0 * pk * a[k - hi:k - lo + 1][::-1]
    
    return smooth_val, grad, b

def nesterov_smooth_descent(a, max_iters, deadline, tau_init=5.0, tau_final=200.0,
                            lr_init=0.5, lr_decay=0.999):
    """
    Nesterov-accelerated gradient descent on smooth max proxy.
    """
    a = np.maximum(a.astype(np.float64), 1e-10)
    a = a / a.sum()
    n = len(a)
    
    best_a = a.copy()
    best_J = compute_J(a)
    
    # Nesterov variables
    y = a.copy()
    v = np.zeros(n)
    t_counter = 1.0
    
    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        
        frac = min(1.0, it / max(1, max_iters))
        tau = tau_init * (tau_final / tau_init) ** frac
        lr = lr_init * (lr_decay ** it)
        
        # Compute gradient at y
        _, grad_y, _ = smooth_max_and_grad(y, tau)
        
        # Nesterov update
        t_new = 1.0
        if it > 0:
            t_new = (1 + np.sqrt(1 + 4 * t_counter ** 2)) / 2
        beta = (t_counter - 1) / t_new
        
        y = y - lr * grad_y
        a_new = y + beta * (y - a)
        
        # Project to simplex (non-negative, sum=1)
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s < 1e-12:
            a_new = a.copy()
            y = a.new = a.copy()
        else:
            a_new = a_new / s
        
        t_counter = t_new
        a = a_new
        
        # Check improvement
        if it % 10 == 0:
            J_val = compute_J(a)
            if J_val < best_J:
                best_J = J_val
                best_a = a.copy()
    
    return best_a

def multiplicative_descent(a, max_iters, deadline, tau_init=10.0, tau_final=150.0):
    """
    Multiplicative weight updates (mirror descent with entropy).
    a_i <- a_i * exp(-eta * grad_i) / Z
    """
    a = np.maximum(a.astype(np.float64), 1e-10)
    a = a / a.sum()
    n = len(a)
    
    best_a = a.copy()
    best_J = compute_J(a)
    
    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        
        frac = min(1.0, it / max(1, max_iters))
        tau = tau_init * (tau_final / tau_init) ** (frac ** 0.7)
        
        _, grad, b = smooth_max_and_grad(a, tau)
        
        # Multiplicative update
        eta = 0.1 * (1 + 0.5 * (1 - frac))
        log_update = -eta * grad / max(1.0, np.abs(grad).max())
        a_new = a * np.exp(log_update)
        
        s = a_new.sum()
        if s < 1e-12:
            break
        a_new = a_new / s
        a_new = np.maximum(a_new, 1e-10)
        a_new = a_new / a_new.sum()
        
        if it % 5 == 0:
            J_val = compute_J(a_new)
            if J_val < best_J:
                best_J = J_val
                best_a = a_new.copy()
        
        a = a_new
    
    return best_a

    t_counter = 1.0
    
    for it in range(max_iters):

        tau = tau_init * (tau_final / tau_init) ** frac

        grad_half = grad_full[:half]

        beta = (t_counter - 1) / t_new
        
        y_new = y - lr * grad_half
        y_new = y_new + beta * (y_new - y)

            full_sum = 2 * y_new.sum()
        else:

            J_val = compute_J(a_full)
            if J_val < best_J:
                best_J = J_val

def peak_deflation(a, iterations=100, deadline=None):
    """
    Local search: identify the peak of conv, and try to reduce it

    """
    a = np.maximum(a.astype(np.float64), 1e-10)
    a = a / a.sum()
    n = len(a)
    
    best_a = a.copy()

        if deadline and time.monotonic() > deadline:

        peak_idx = b.argmax()
        peak_val = b[peak_idx]
        
        # Find which elements contribute most to the peak
        lo = max(0, peak_idx - n + 1)
        hi = min(n - 1, peak_idx)
        contributions = a[lo:hi+1] * a[peak_idx - hi : peak_idx - lo + 1][::-1]
        
        # Try to reduce the peak by decreasing large contributors and increasing others
        # But we need to be careful not to increase other convolution values
        
        # Strategy: decrease the element with the largest contribution at the peak,
        # and distribute the mass to elements that don't contribute to the peak
        top_contributors = np.argsort(contributions)[-3:]
        
        improved = False
        for trial in range(5):
            a_trial = a.copy()
            # Pick a random element to decrease
            dec_idx = lo + top_contributors[trial % len(top_contributors)]
            # Pick elements to increase (away from the peak)
            # Elements that don't appear in the peak's sum range
            peak_range = set(range(lo, hi+1))
            all_indices = set(range(n))
            safe_indices = all_indices - peak_range
            
            if not safe_indices:
                # All elements contribute to the peak; try different approach
                # Decrease peak contributors and increase the least-contributing ones
                min_contributors = np.argsort(contributions)[:2]
                inc_indices = lo + min_contributors
            else:
                safe_list = list(safe_indices)
                inc_idx = safe_list[np.random.randint(len(safe_list))]
                inc_indices = [inc_idx]
            
            # Move mass
            delta = a_trial[dec_idx] * 0.01 * (1 + 0.5 * trial)
            a_trial[dec_idx] -= delta
            for inc_i in inc_indices:
                a_trial[inc_i] += delta / len(inc_indices)
            
            a_trial = np.maximum(a_trial, 1e-10)
            a_trial = a_trial / a_trial.sum()

def resample_seq(src, dest_n):
    """Resample 1D sequence to new length via linear interpolation."""
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)

def lbb_smooth_objective(x, tau=100.0):
    """Objective for L-BFGS-B."""
    ax = np.maximum(x, 1e-15)
    s = ax.sum()
    if s < 1e-10:
        return 1e10
    ax = ax / s
    b = fast_conv(ax)
    b_shifted = b - b.max()
    return float(np.log(np.sum(np.exp(tau * b_shifted))) / tau + b.max() / tau)

def lbb_smooth_grad(x, tau=100.0):

    b_max = b.max()
    exp_b = np.exp(tau * (b - b_max))

        gmax = np.abs(grad).max()
        if gmax < 1e-15:
            break
        log_update = -eta * grad / gmax
        a_new = a * np.exp(log_update)

        s = a_new.sum()
        if s < 1e-12:
            break
        a_new = np.maximum(a_new, 1e-10)
        a_new = a_new / a_new.sum()
        a = a_new

        if it % 10 == 0:
            J_val = compute_J(a)
            if J_val < best_J:
                best_J = J_val
                best_a = a.copy()

    return best_a


def sym_optimize(a, max_iters, deadline, tau_init=5.0, tau_final=200.0):
    a = np.maximum(a.astype(np.float64), 1e-10)
    n = len(a)

    half = (n + 1) // 2
    a_sym = np.zeros(n)
    if n % 2 == 0:
        a_sym[:half] = (a[:half] + a[n - 1:half - 1:-1]) / 2.0
        a_sym[half:] = a_sym[:half][::-1]
    else:
        a_sym[:half] = (a[:half] + a[n - 1:half - 1:-1]) / 2.0
        a_sym[half:] = a_sym[half - 1:][::-1]

    a_sym = np.maximum(a_sym, 1e-10)
    a_sym = a_sym / a_sym.sum()

    best_a = a_sym.copy()
    best_J = compute_J(a_sym)

    y = a_sym[:half].copy()
    t_counter = 1.0

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        frac = min(1.0, it / max(1, max_iters))
        tau = tau_init * (tau_final / tau_init) ** frac
        lr = 0.3 * (0.9995 ** it)

        a_full = np.zeros(n)
        a_full[:half] = y
        if n % 2 == 0:
            a_full[half:] = y[::-1]
        else:
            a_full[half:] = y[half - 1:][::-1]
        a_full = np.maximum(a_full, 1e-15)
        a_full = a_full / a_full.sum()

        grad_full, _ = smooth_max_grad(a_full, tau)
        grad_half = grad_full[:half]

        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t_counter * t_counter)) / 2.0
        beta = (t_counter - 1.0) / t_new

        y_new = y - lr * grad_half
        y_new = y_new + beta * (y_new - y)
        y_new = np.maximum(y_new, 1e-12)

        if n % 2 == 0:
            full_sum = 2.0 * y_new.sum()
        else:
            full_sum = 2.0 * y_new.sum() - y_new[half - 1]
        if full_sum < 1e-12:
            y_new = y.copy()
        else:
            y_new = y_new / full_sum

        t_counter = t_new
        y = y_new

        if it % 20 == 0:
            a_full2 = np.zeros(n)
            a_full2[:half] = y
            if n % 2 == 0:
                a_full2[half:] = y[::-1]
            else:
                a_full2[half:] = y[half - 1:][::-1]
            a_full2 = np.maximum(a_full2, 1e-15)
            a_full2 = a_full2 / a_full2.sum()
            J_val = compute_J(a_full2)
            if J_val < best_J:
                best_J = J_val
                best_a = a_full2.copy()

    return best_a


def peak_deflation(a, max_iters, deadline, rng):
    a = np.maximum(a.astype(np.float64), 1e-10)
    a = a / a.sum()
    n = len(a)

    best_a = a.copy()
    best_J = compute_J(a)

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        b = fast_conv(a)
        peak_idx = int(b.argmax())

        lo = max(0, peak_idx - n + 1)
        hi = min(n - 1, peak_idx)

        contributions = a[lo:hi + 1] * a[peak_idx - hi:peak_idx - lo + 1][::-1]

        improved = False
        for trial in range(10):
            a_trial = a.copy()

            top_k = np.argsort(contributions)[-1]
            dec_idx = lo + top_k

            peak_set = set(range(lo, hi + 1))
            safe = [i for i in range(n) if i not in peak_set]

            if safe:
                inc_idx = safe[int(rng.integers(len(safe)))]
            else:
                min_k = np.argsort(contributions)[0]
                inc_idx = lo + min_k

            delta = a_trial[dec_idx] * (0.003 + 0.003 * trial * 0.4)
            a_trial[dec_idx] -= delta
            a_trial[inc_idx] += delta

            a_trial = np.maximum(a_trial, 1e-10)
            a_trial = a_trial / a_trial.sum()

            J_trial = compute_J(a_trial)
            if J_trial < best_J - 1e-15:
                a = a_trial.copy()
                best_J = J_trial
                best_a = a_trial.copy()
                improved = True
                break

        if not improved:
            a_trial = a + rng.normal(0, 0.002, n)
            a_trial = np.maximum(a_trial, 1e-10)
            a_trial = a_trial / a_trial.sum()
            J_trial = compute_J(a_trial)
            if J_trial < best_J:
                a = a_trial.copy()
                best_J = J_trial
                best_a = a_trial.copy()
            elif it > 80:
                break

    return best_a


def resample_seq(src, dest_n):
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)


def lbb_obj(x, tau=80.0):
    ax = np.maximum(x, 1e-15)
    s = ax.sum()
    if s < 1e-10:
        return 1e10
    ax = ax / s
    b = fast_conv(ax)
    b_shifted = b - b.max()
    return float(b.max() / tau + np.log(np.sum(np.exp(tau * b_shifted))) / tau)


def lbb_grad(x, tau=80.0):
    ax = np.maximum(x, 1e-15)
    s = ax.sum()
    if s < 1e-10:
        return np.zeros_like(x)
    ax = ax / s
    n = len(ax)
    b = fast_conv(ax)
    b_max = b.max()
    exp_b = np.exp(tau * (b - b_max))
    p = exp_b / exp_b.sum()

    grad = np.zeros(n)
    for k in range(2 * n - 1):
        pk = p[k]
        if pk < 1e-14:
            continue
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        grad[lo:hi + 1] += 2.0 * pk * ax[k - hi:k - lo + 1][::-1]
    return grad


def try_update(state, a, label, t_start):
    a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
    if a.size == 0 or a.sum() < 0.01:
        return state
    score = float(evaluate_sequence(a.tolist()))
    if np.isfinite(score) and score < state['best_score']:
        state['best_seq'] = a.copy()
        state['best_score'] = score
        elapsed = time.monotonic() - t_start
        print(f"  ** NEW BEST: J={score:.8f} n={a.size} t={elapsed:.0f}s {label}", flush=True)
    return state


def nesterov_descent_v2(a, max_iters, deadline, tau_init=5.0, tau_final=300.0,
                        lr_init=0.4, lr_decay=0.9993, check_interval=5):
    """Nesterov with backtracking line search for guaranteed progress."""
    a = np.maximum(a.astype(np.float64), 1e-10)
    a = a / a.sum()
    n = len(a)

    best_a = a.copy()
    best_J = compute_J(a)

    y = a.copy()
    t_counter = 1.0

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        frac = min(1.0, it / max(1, max_iters))
        tau = tau_init * (tau_final / tau_init) ** frac
        lr = lr_init * (lr_decay ** it)

        grad_y, b_y = smooth_max_grad(y, tau)
        smooth_val = (b_y.max() + np.log(np.sum(np.exp(tau * (b_y - b_y.max()))))) / tau

        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t_counter * t_counter)) / 2.0
        beta = (t_counter - 1.0) / t_new

        # Backtracking: try decreasing lr until we get improvement in smooth objective
        y_trial = y - lr * grad_y
        a_trial = y_trial + beta * (y_trial - a)
        a_trial = np.maximum(a_trial, 0.0)
        s = a_trial.sum()
        if s < 1e-12:
            a_trial = a.copy()
        else:
            a_trial = a_trial / s

        # Evaluate smooth objective at trial
        b_trial = fast_conv(a_trial)
        smooth_trial = (b_trial.max() + np.log(np.sum(np.exp(tau * (b_trial - b_trial.max()))))) / tau

        if smooth_trial > smooth_val + 1e-14:
            # Backtrack
            lr = lr * 0.5
            y_trial = y - lr * grad_y
            a_trial = y_trial + beta * (y_trial - a)
            a_trial = np.maximum(a_trial, 0.0)
            s = a_trial.sum()
            if s < 1e-12:
                a_trial = a.copy()
            else:
                a_trial = a_trial / s

        t_counter = t_new
        a = a_trial
        y = y_trial

        if it % check_interval == 0:
            J_val = compute_J(a)
            if J_val < best_J:
                best_J = J_val
                best_a = a.copy()

    return best_a


def propose_candidate(seed=42, budget_s=1500, **kwargs):
    """
    Multi-phase search: Nesterov smooth descent (with backtracking), multiplicative
    updates, symmetric optimization, peak deflation, multi-length exploration,
    and L-BFGS-B polishing.
    """
    rng = np.random.default_rng(seed)
    t_start = time.monotonic()
    deadline = t_start + float(budget_s) - 20.0

    state = {'best_seq': None, 'best_score': float('inf')}

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        if p.ndim == 1 and p.size > 0 and np.all(np.isfinite(p)):
            parent = np.maximum(p, 0.0)
            state = try_update(state, parent, "parent", t_start)
            print(f"Parent: n={parent.size}, J={state['best_score']:.8f}", flush=True)
    except (NameError, AttributeError):
        pass

    if parent is None:
        parent = np.ones(256)
        state = try_update(state, parent, "default", t_start)

    n_parent = parent.size

    # === PHASE 1: Nesterov smooth descent with backtracking (main workhorse) ===
    print(f"\n{'=' * 60}", flush=True)
    print(f"PHASE 1: Nesterov Descent with Backtracking (n={n_parent})", flush=True)
    print(f"{'=' * 60}", flush=True)

    phase_deadline = min(t_start + 350, deadline - 350)

    configs = [
        {'tau_init': 3.0, 'tau_final': 150.0, 'lr_init': 0.3, 'lr_decay': 0.9995},
        {'tau_init': 10.0, 'tau_final': 400.0, 'lr_init': 0.2, 'lr_decay': 0.999},
        {'tau_init': 5.0, 'tau_final': 250.0, 'lr_init': 0.5, 'lr_decay': 0.9993},
    ]

    for cfg_idx, cfg in enumerate(configs):
        if time.monotonic() > phase_deadline:
            break

        a0 = parent.copy().astype(np.float64)
        if cfg_idx > 0:
            noise_scale = 0.01 * cfg_idx
            a0 = a0 + rng.normal(0, noise_scale, n_parent)
            a0 = np.maximum(a0, 1e-6)
            a0 = a0 / a0.sum()

        a = nesterov_descent_v2(a0, 5000, phase_deadline, **cfg)
        state = try_update(state, a, f"nesterov-bt-{cfg_idx}", t_start)

    # Standard Nesterov (no backtracking, faster per iteration)
    if time.monotonic() < phase_deadline - 30:
        a = nesterov_descent(parent, 5000, phase_deadline,
                            tau_init=5.0, tau_final=200.0,
                            lr_init=0.3, lr_decay=0.9995)
        state = try_update(state, a, "nesterov-std", t_start)

    # === PHASE 2: Multiplicative descent ===
    print(f"\n{'=' * 60}", flush=True)
    print(f"PHASE 2: Multiplicative Descent", flush=True)
    print(f"{'=' * 60}", flush=True)

    phase_deadline2 = min(t_start + 500, deadline - 300)
    current_best = state['best_seq'] if state['best_seq'] is not None else parent

    a = mult_descent(current_best, 3000, phase_deadline2, tau_init=10.0, tau_final=250.0)
    state = try_update(state, a, "multiplicative", t_start)

    # Second multiplicative pass with different params
    if time.monotonic() < phase_deadline2:
        a = mult_descent(current_best, 2000, phase_deadline2, tau_init=30.0, tau_final=500.0)
        state = try_update(state, a, "multiplicative-2", t_start)

    # === PHASE 3: Symmetric optimization ===
    print(f"\n{'=' * 60}", flush=True)
    print(f"PHASE 3: Symmetric Optimization", flush=True)
    print(f"{'=' * 60}", flush=True)

    phase_deadline3 = min(t_start + 650, deadline - 250)
    current_best = state['best_seq'] if state['best_seq'] is not None else parent

    a = sym_optimize(current_best, 3000, phase_deadline3, tau_init=5.0, tau_final=250.0)
    state = try_update(state, a, "symmetric", t_start)

    # Higher tau symmetric
    if time.monotonic() < phase_deadline3:
        a = sym_optimize(current_best, 2000, phase_deadline3, tau_init=20.0, tau_final=500.0)
        state = try_update(state, a, "symmetric-hi", t_start)

    # === PHASE 4: L-BFGS-B refinement ===
    print(f"\n{'=' * 60}", flush=True)
    print(f"PHASE 4: L-BFGS-B Refinement", flush=True)
    print(f"{'=' * 60}", flush=True)

    phase_deadline4 = min(t_start + 800, deadline - 200)
    current_best = state['best_seq'] if state['best_seq'] is not None else parent

    if current_best is not None:
        x0 = current_best.astype(np.float64)
        bounds_list = [(0.0, None)] * len(x0)
        try:
            res = minimize(lbb_obj, x0, jac=lbb_grad,
                           method='L-BFGS-B', bounds=bounds_list,
                           options={'maxiter': 300, 'max_fun': 50000, 'ftol': 1e-15,
                                    'gtol': 1e-12, 'iprint': -1})
            if res.fun < 1e8:
                state = try_update(state, res.x, "L-BFGS-B", t_start)
        except Exception as e:
            print(f"  L-BFGS-B error: {e}", flush=True)

    # === PHASE 5: Peak deflation local search ===
    print(f"\n{'=' * 60}", flush=True)
    print(f"PHASE 5: Peak Deflation", flush=True)
    print(f"{'=' * 60}", flush=True)

    phase_deadline5 = min(t_start + 1000, deadline - 150)
    current_best = state['best_seq'] if state['best_seq'] is not None else parent

    for round_idx in range(3):
        if time.monotonic() > phase_deadline5:
            break
        a = peak_deflation(current_best, 300, phase_deadline5, rng)
        state = try_update(state, a, f"peak-deflate-{round_idx}", t_start)
        current_best = state['best_seq']

    # === PHASE 6: Multi-length exploration ===
    print(f"\n{'=' * 60}", flush=True)
    print(f"PHASE 6: Multi-length Exploration", flush=True)
    print(f"{'=' * 60}", flush=True)

    phase_deadline6 = min(t_start + 1250, deadline - 100)
    current_best = state['best_seq'] if state['best_seq'] is not None else parent

    if current_best is not None:
        base_n = current_best.size
        candidate_lengths = set()
        for mult in [0.75, 1.25, 1.5, 2.0, 2.5, 3.0]:
            new_n = int(base_n * mult)
            if 128 <= new_n <= 4096:
                candidate_lengths.add(new_n)
        for n_fix in [256, 384, 512, 640, 768, 1024, 1280, 1536, 2048, 2560, 3072]:
            if 128 <= n_fix <= 4096:
                candidate_lengths.add(n_fix)

        # Sort by distance from base_n (prefer nearby)
        candidate_lengths = sorted(candidate_lengths, key=lambda x: abs(x - base_n))

        for new_n in candidate_lengths:
            if time.monotonic() > phase_deadline6:
                break

            a_ext = resample_seq(current_best, new_n)
            a_ext = np.maximum(a_ext, 1e-8)
            a_ext = a_ext / a_ext.sum()

            # Quick Nesterov on new length
            a_opt = nesterov_descent(a_ext, 800, phase_deadline6,
                                     tau_init=5.0, tau_final=150.0,
                                     lr_init=0.3, lr_decay=0.9995)
            state = try_update(state, a_opt, f"len-{new_n}", t_start)

    # === PHASE 7: Final fine-tuning on best ===
    print(f"\n{'=' * 60}", flush=True)
    print(f"PHASE 7: Final Fine-tuning", flush=True)
    print(f"{'=' * 60}", flush=True)

    current_best = state['best_seq'] if state['best_seq'] is not None else parent

    if current_best is not None and time.monotonic() < deadline - 5:
        # Very high tau Nesterov for precise convergence
        a = nesterov_descent_v2(current_best, 3000, deadline,
                                tau_init=50.0, tau_final=800.0,
                                lr_init=0.05, lr_decay=0.9999,
                                check_interval=5)
        state = try_update(state, a, "final-nesterov", t_start)

        # Peak deflation
        if time.monotonic() < deadline - 5:
            a = peak_deflation(state['best_seq'] if state['best_seq'] is not None else current_best,
                               100, deadline, rng)
            state = try_update(state, a, "final-peak", t_start)

        # One more L-BFGS-B
        if time.monotonic() < deadline - 3 and state['best_seq'] is not None:
            x0 = state['best_seq'].astype(np.float64)
            bounds_list = [(0.0, None)] * len(x0)
            try:
                res = minimize(lbb_obj, x0, jac=lbb_grad,
                               method='L-BFGS-B', bounds=bounds_list,
                               options={'maxiter': 100, 'max_fun': 20000, 'ftol': 1e-16,
                                        'gtol': 1e-13, 'iprint': -1})
                if res.fun < 1e8:
                    state = try_update(state, res.x, "final-LBFGS", t_start)
            except Exception:
                pass

    # Fallback
    if state['best_seq'] is None:
        state['best_seq'] = np.ones(128)
        state['best_score'] = compute_J(state['best_seq'])

    print(f"\n{'=' * 60}", flush=True)
    print(f"FINAL: J={state['best_score']:.8f} n={state['best_seq'].size} "
          f"time={time.monotonic() - t_start:.0f}s", flush=True)
    print(f"{'=' * 60}", flush=True)

    return [float(x) for x in state['best_seq']]
