Wait, I need to be more careful. `b[k] = sum_j a[j]*a[k-j]` where `0 <= j < n` and `0 <= k-j < n`, i.e., `max(0, k-n+1) <= j <= min(n-1, k)`.

The terms involving `a[i]` are those where `j = i` (requires `k-i` in `[0,n-1]`) or `j = k-i` (requires `i` in `[0,n-1]`, which is always true since `0 <= i < n`).

So for a given `k`:
- If `k-i` is in `[0, n-1]`: both `a[i]*a[k-i]` and `a[k-i]*a[i]` appear (unless `i = k-i`, in which case it's `a[i]^2`)
  - If `i ≠ k-i` (i.e., `k ≠ 2i`): contribution to `b[k]` is `2*a[i]*a[k-i]`
  - If `i = k-i` (i.e., `k = 2i`): contribution is `a[i]^2`

So the update when `a[i]` changes from `a_i` to `x`:
- For `k = 2i`: `b[k]_new = b[k] - a_i^2 + x^2`
- For `k ≠ 2i` with `0 <= k-i < n`: `b[k]_new = b[k] - 2*a_i*a[k-i] + 2*x*a[k-i]`
- For other `k`: `b[k]_new = b[k]`

Now, to minimize `max_k b[k]_new`, we need `x` such that the maximum of all these (piecewise linear/quadratic) functions is minimized. This is a 1D convex optimization problem (the max of convex functions is convex).

We can solve this with bisection on the target value `M`: for a given `M`, check if there exists `x >= 0` such that all `b[k]_new(x) <= M`. This is possible iff:
- `x^2 + (b[2i] - a_i^2) <= M` → `x <= sqrt(M - (b[2i] - a_i^2))` (if the RHS is non-negative)
- For each `k ≠ 2i` with `0 <= k-i < n`: `2*x*a[k-i] + (b[k] - 2*a_i*a[k-i]) <= M`
  - If `a[k-i] > 0`: `x <= (M - b[k] + 2*a_i*a[k-i]) / (2*a[k-i])`
  - If `a[k-i] = 0`: need `b[k] <= M` (independent of x)
  - Also `x >= 0`

So the feasible `x` is in `[0, min over all upper bounds]`. If this interval is non-empty, then `M` is feasible. We can binary search on `M` to find the minimum.

This is much more efficient than golden section search! The complexity per coordinate is O(n) for the bisection (since we need to check O(n) constraints each time), and we do O(log(precision)) bisection steps.

Actually, we can do even better. The minimum `M` is the minimum over `x >= 0` of `max_k b[k]_new(x)`. Since the functions are:

The max of these is a convex piecewise-linear-quadratic function. The minimum is where the "active" constraints intersect.

Actually, for a cleaner approach: the optimal `x` is where the maximum is minimized. We can find this by:
1. For a given `x`, compute `max_k b[k]_new(x)`

"""
Minimizes J(a) = 2*n*max(convolve(a,a))/(sum(a))^2 using:
1. Fast coordinate descent with bisection on target M
2. LP-based mirror descent for global improvements
3. Multi-length exploration
4. Random perturbation restarts
Key innovation: coordinate descent using bisection on M (O(n log(1/eps)) per coordinate)
instead of golden section search (O(n * iterations) per coordinate).
"""

import numpy as np
from scipy.optimize import linprog
from scipy.sparse import lil_matrix
import time


def fast_conv(a):
    """Full self-convolution via FFT."""
    n = len(a)
    s = 1
    while s < 2 * n - 1:
        s <<= 1
    fa = np.fft.rfft(a, s)
    return np.fft.irfft(fa * fa, s)[:2 * n - 1]


def compute_J(a):
    """Compute J(a) = 2*n*max(conv(a,a))/(sum(a))^2."""
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))


def find_optimal_x(a, i, b, S, n_bisect=50):
    """Find optimal value for coordinate a[i] using bisection on M.
    For a given x, b[k]_new(x) is:
      - x^2 + (b[2i] - a[i]^2) for k = 2i
      - 2*x*a[k-i] + (b[k] - 2*a[i]*a[k-i]) for k != 2i, 0 <= k-i < n
      - b[k] for other k
    We find the smallest M such that there exists x >= 0 with all b[k]_new(x) <= M.
    """
    n = len(a)
    ai = a[i]
    
    # Precompute the linear coefficients and constants
    # For k in [i, i+n-1] (where both i and k-i are valid):
    #   if k == 2i: b_new = x^2 + (b[2i] - ai^2)
    #   else: b_new = 2*x*a[k-i] + (b[k] - 2*ai*a[k-i])
    
    # For k not in [i, i+n-1]: b_new = b[k]
    
    # The "unaffected" max:
    # k ranges from 0 to 2n-2
    # Affected k: max(0, i) to min(2n-2, i+n-1) = i to i+n-1 (since 0<=i<n)
    
    k_lo = i
    k_hi = min(2 * n - 1, i + n - 1)
    
    # Unaffected range
    max_unaffected = 0.0
    if k_lo > 0:
        max_unaffected = max(max_unaffected, b[:k_lo].max() if k_lo > 0 else 0.0)
    if k_hi < 2 * n - 1:
        max_unaffected = max(max_unaffected, b[k_hi + 1:].max() if k_hi + 1 < 2 * n else 0.0)
    
    # Affected coefficients
    # For k in [k_lo, k_hi], k != 2i:
    #   b_new(k, x) = 2*x*a[k-i] + (b[k] - 2*ai*a[k-i])
    # For k = 2i:
    #   b_new(k, x) = x^2 + (b[2i] - ai^2)
    
    # Collect linear constraints: c*x + d <= M
    # and quadratic: x^2 + e <= M
    
    c_list = []
    d_list = []
    e_quad = None
    k_2i = 2 * i
    
    for k in range(k_lo, k_hi + 1):
        ki = k - i  # this is in [0, n-1]
        if k == k_2i:
            e_quad = b[k] - ai * ai
        else:
            c_list.append(2.0 * a[ki])
            d_list.append(b[k] - 2.0 * ai * a[ki])
    
    c_arr = np.array(c_list) if c_list else np.zeros(0)
    d_arr = np.array(d_list) if d_list else np.zeros(0)
    
    # For a given M, find if there exists x >= 0 with:
    #   x^2 + e_quad <= M  →  x <= sqrt(M - e_quad) (if M >= e_quad)
    #   c_j * x + d_j <= M for all j  →  x <= (M - d_j) / c_j (if c_j > 0)
    #   x >= 0
    #   max_unaffected <= M
    
    def feasible(M):
        if max_unaffected > M:
            return False, 0.0
        x_max = float('inf')
        if e_quad is not None:
            if M < e_quad:
                return False, 0.0
            x_max = min(x_max, np.sqrt(M - e_quad))
        if len(c_arr) > 0:
            # For c_j > 0: x <= (M - d_j) / c_j
            pos_mask = c_arr > 1e-15
            if np.any(pos_mask):
                upper = (M - d_arr[pos_mask]) / c_arr[pos_mask]
                x_max = min(x_max, upper.min())
            # For c_j <= 0 (shouldn't happen since a >= 0, c = 2*a[ki] >= 0)
            # but if a[ki] = 0, then c_j = 0, constraint is d_j <= M
            zero_mask = c_arr <= 1e-15
            if np.any(zero_mask):
                if d_arr[zero_mask].max() > M:
                    return False, 0.0
        if x_max < 0:
            return False, 0.0
        return True, min(x_max, float('inf'))
    
    # Binary search on M
    # Lower bound: max(max_unaffected, e_quad if not None, max(d_arr))
    M_lo = max_unaffected
    if e_quad is not None:
        M_lo = max(M_lo, e_quad)
    if len(d_arr) > 0:
        M_lo = max(M_lo, d_arr.max())
    
    # Upper bound: current max_b (M at x = ai)
    M_hi = float(b.max())
    
    # Actually, let's set M_hi to be the max at x = ai (current value)
    # b_new(ai) should equal b (since we're just replacing ai with ai)
    # So M_hi = b.max() is valid.
    
    # But we want to potentially reduce M, so let's also try M_hi = b.max()
    # The bisection will find the minimum M.
    
    for _ in range(n_bisect):
        M_mid = (M_lo + M_hi) / 2.0
        ok, x_val = feasible(M_mid)
        if ok:
            M_hi = M_mid
        else:
            M_lo = M_mid
    
    # Now M_hi is approximately the minimum M
    ok, x_opt = feasible(M_hi)
    if not ok:
        # Fallback: use current value
        x_opt = ai
    return max(0.0, x_opt), M_hi


def coordinate_descent_pass_fast(a, b, S):
    """One full pass of coordinate descent using bisection method.
    Returns updated a, b, S, and the max_b."""
    n = len(a)
    a = a.copy()
    
    # Process coordinates in random order
    indices = np.arange(n)
    np.random.shuffle(indices)
    
    for i in indices:
        x_opt, _ = find_optimal_x(a, i, b, S)
        old_val = a[i]
        if abs(x_opt - old_val) > 1e-12:
            a[i] = x_opt
            S = S - old_val + x_opt
            # Update b incrementally
            k_lo = i
            k_hi = min(2 * n - 1, i + n - 1)
            k_2i = 2 * i
            for k in range(k_lo, k_hi + 1):
                ki = k - i
                if k == k_2i:
                    b[k] = b[k] - old_val * old_val + x_opt * x_opt
                else:
                    b[k] = b[k] - 2.0 * old_val * a[ki] + 2.0 * x_opt * a[ki]
    
    return a, b, S, b.max()


def coordinate_descent_multi_pass(a, n_passes, deadline, tracker):
    """Multiple passes of fast coordinate descent."""
    a = np.maximum(a.astype(np.float64), 0.0)
    S = a.sum()
    if S < 1e-10:
        return a
    a = a / S
    S = 1.0
    n = len(a)
    b = fast_conv(a)
    current_J = 2.0 * n * b.max() / (S * S)
    tracker.update(a)
    
    for p in range(n_passes):
        if time.monotonic() > deadline:
            break
        t0 = time.monotonic()
        a, b, S, max_b = coordinate_descent_pass_fast(a, b, S)
        new_J = 2.0 * n * max_b / (S * S)
        if new_J < current_J - 1e-10:
            current_J = new_J
            tracker.update(a)
        elif p > 5:
            break
        if p % 3 == 0:
            print(f"    CD pass {p}: J={current_J:.10f} ({time.monotonic()-t0:.2f}s)", flush=True)
    
    tracker.update(a)
    return a


def build_A_sparse(a):
    """Build sparse constraint matrix for LP: (a*g)[k] <= M for all k."""
    n = len(a)
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        jl = max(0, k - n + 1)
        jh = min(n - 1, k)
        if jl <= jh:
            for j in range(jl, jh + 1):
                rows.append(k)
                cols.append(j)
                vals.append(a[k - j])
    A = lil_matrix((2 * n - 1, n))
    A[rows, cols] = vals
    return A.tocsr()


def lp_best_response(a, M):
    """LP: max sum(g) s.t. (a*g)[k] <= M for all k, g >= 0."""
    n = len(a)
    c = -np.ones(n)
    A = build_A_sparse(a)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    try:
        r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
        if r.success and r.x is not None:
            g = np.maximum(r.x, 0.0)
            if g.sum() > 1e-10:
                return g
    except:
        pass
    return None


def lp_mirror_descent(a, max_iters, deadline, tracker, tighten=0.005):
    """LP mirror descent iterations."""
    a = np.maximum(a.astype(np.float64), 1e-6)
    S = a.sum()
    a = a / S
    S = 1.0
    n = len(a)
    current_J = 2.0 * n * fast_conv(a).max() / (S * S)
    tracker.update(a)
    
    stall = 0
    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        
        b = fast_conv(a)
        M = float(b.max())
        M_target = M * (1.0 - tighten)
        
        g = lp_best_response(a, M_target)
        if g is None:
            M_target = M * (1.0 - tighten * 0.3)
            g = lp_best_response(a, M_target)
        if g is None:
            tighten *= 0.5
            stall += 1
            if stall > 30:
                break
            continue
        
        g = np.maximum(g, 0.0)
        gs = g.sum()
        if gs < 1e-10:
            break
        g = g / gs
        
        # Line search
        progress = it / max(1, max_iters)
        base_t = 0.3 * (1.0 - 0.5 * progress)
        candidates = [base_t, base_t * 0.5, base_t * 0.7, base_t * 1.5]
        
        best_a = None
        best_J = current_J
        for t in candidates:
            a_try = (1.0 - t) * a + t * g
            a_try = np.maximum(a_try, 0.0)
            s = a_try.sum()
            if s < 1e-10:
                continue
            a_try = a_try / s
            J_try = 2.0 * n * fast_conv(a_try).max() / s**2  # s should be 1
            J_try = 2.0 * n * fast_conv(a_try).max()
            if J_try < best_J - 1e-12:
                best_J = J_try
                best_a = a_try
        
        if best_a is None:
            stall += 1
            if stall > 30:
                break
            tighten *= 0.7
            continue
        
        a = best_a
        current_J = best_J
        stall = 0
        tighten = min(0.02, tighten * 1.01)
        
        if it % 10 == 0:
            tracker.update(a)
    
    tracker.update(a)
    return a


def resample_seq(src, dest_n):
    """Resample 1D sequence to new length via linear interpolation."""
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)

def smooth_and_reoptimize(a, smooth_sigma, deadline, tracker):
    """Apply Gaussian smoothing and re-optimize."""
    n = len(a)
    # Gentle smoothing
    kernel_size = max(3, int(smooth_sigma * 6) | 1)  # odd
    x = np.arange(kernel_size) - kernel_size // 2
    kernel = np.exp(-0.5 * (x / smooth_sigma) ** 2)
    kernel /= kernel.sum()
    
    a_smooth = np.convolve(a, kernel, mode='same')
    a_smooth = np.maximum(a_smooth, 0.001)
    a_smooth /= a_smooth.sum()
    
    # Re-optimize with coordinate descent
    a_opt = coordinate_descent_multi_pass(a_smooth, 10, deadline, tracker)
    return a_opt

"""
Minimizes J(a) = 2*n*max(convolve(a,a))/(sum(a))^2 using:
1. Fast coordinate descent with bisection on target M (O(n*log(1/eps)) per coordinate)
2. LP-based mirror descent for global improvements
3. Multi-length exploration via resampling

5. Intensive final polish with many CD passes
Key innovation: bisection-based coordinate descent that exactly solves the 1D subproblem

import numpy as np
from scipy.optimize import linprog
from scipy.sparse import lil_matrix
import time


def fast_conv(a):
    """Full self-convolution via FFT."""
    n = len(a)
    s = 1
    while s < 2 * n - 1:
        s <<= 1
    fa = np.fft.rfft(a, s)
    return np.fft.irfft(fa * fa, s)[:2 * n - 1]


def compute_J(a):
    """Compute J(a) = 2*n*max(conv(a,a))/(sum(a))^2."""
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))


def find_optimal_x(a, i, b):
    """Find optimal value for coordinate a[i] using bisection on M.
    
    For a given x (new value of a[i]), the convolution values change as:
      - b_new[2i] = x^2 + (b[2i] - a[i]^2)   [quadratic in x]
      - b_new[k] = 2*x*a[k-i] + (b[k] - 2*a[i]*a[k-i]) for k in [i, i+n-1], k != 2i [linear in x]
      - b_new[k] = b[k] for unaffected k
    
    We binary search on M to find the minimum achievable max convolution.
    """
    n = len(a)
    ai = a[i]
    k_2i = 2 * i
    k_lo = i
    k_hi = min(2 * n - 1, i + n - 1)
    
    # Unaffected max
    if k_lo > 0:
        max_unaffected = b[:k_lo].max()
    else:
        max_unaffected = 0.0
    if k_hi + 1 < 2 * n:
        tmp = b[k_hi + 1:].max()
        if tmp > max_unaffected:
            max_unaffected = tmp
    
    # Collect linear constraints: c*x + d <= M for k != 2i
    # and quadratic: x^2 + e <= M for k = 2i
    c_list = []
    d_list = []
    e_quad = b[k_2i] - ai * ai
    
    for k in range(k_lo, k_hi + 1):
        if k == k_2i:
            continue
        ki = k - i
        c_list.append(2.0 * a[ki])
        d_list.append(b[k] - 2.0 * ai * a[ki])
    
    c_arr = np.array(c_list) if c_list else np.zeros(0)
    d_arr = np.array(d_list) if d_list else np.zeros(0)
    
    # Precompute masks
    pos_mask = c_arr > 1e-15
    zero_mask = ~(pos_mask)
    
    d_zero_max = d_arr[zero_mask].max() if np.any(zero_mask) else 0.0
    
    def feasible(M):
        """Check if there exists x >= 0 with all b_new[k](x) <= M. Return x_max or -1."""
        if max_unaffected > M + 1e-15:
            return -1.0
        if M < e_quad - 1e-15:
            return -1.0
        if M < d_zero_max - 1e-15:
            return -1.0
        x_max = np.sqrt(max(0.0, M - e_quad))
        if np.any(pos_mask):
            upper = (M - d_arr[pos_mask]) / c_arr[pos_mask]
            x_max = min(x_max, upper.min())
        if x_max < -1e-10:
            return -1.0
        return max(0.0, x_max)
    
    # Binary search on M
    M_lo = max(max_unaffected, e_quad, d_zero_max)
    M_hi = float(b.max())
    
    # Ensure M_lo is feasible check
    if M_hi - M_lo < 1e-14:
        x_opt = feasible(M_hi)
        if x_opt < 0:
            return ai
        return max(0.0, x_opt)
    
    for _ in range(45):
        M_mid = (M_lo + M_hi) * 0.5
        x_test = feasible(M_mid)
        if x_test >= 0:
            M_hi = M_mid
        else:
            M_lo = M_mid
    
    x_opt = feasible(M_hi)
    if x_opt < 0:
        # Fallback
        x_opt = feasible(M_hi + 1e-10)
        if x_opt < 0:
            return ai
    return max(0.0, x_opt)


def coordinate_descent_pass_fast(a, b):
    """One full pass of coordinate descent using bisection method.
    Returns updated a, b, and max_b."""
    n = len(a)
    
    # Process coordinates in random order
    indices = np.random.permutation(n)
    
    for i in indices:
        x_opt = find_optimal_x(a, i, b)
        old_val = a[i]
        if abs(x_opt - old_val) > 1e-14:
            a[i] = x_opt
            # Update b incrementally
            k_2i = 2 * i
            k_lo = i
            k_hi = min(2 * n - 1, i + n - 1)

                if k == k_2i:

                else:
                    ki = k - i
                    b[k] = b[k] - 2.0 * old_val * a[ki] + 2.0 * x_opt * a[ki]
    
    return a, b, b.max()

def build_A_sparse(a):
    """Build sparse constraint matrix: (a*g)[k] for all k."""
    n = len(a)
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        jl = max(0, k - n + 1)
        jh = min(n - 1, k)
        for j in range(jl, jh + 1):
            rows.append(k)
            cols.append(j)
            vals.append(a[k - j])
    A = lil_matrix((2 * n - 1, n))
    A[rows, cols] = vals
    return A.tocsr()


def lp_best_response(a, M):
    """LP: max sum(g) s.t. (a*g)[k] <= M for all k, g >= 0."""
    n = len(a)
    c = -np.ones(n)
    A = build_A_sparse(a)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    try:
        r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                    options={'presolve': True, 'time_limit': 30})
        if r.success and r.x is not None:
            g = np.maximum(r.x, 0.0)
            if g.sum() > 1e-10:
                return g
    except Exception:
        pass
    return None

def resample_seq(src, dest_n):
    """Resample 1D sequence to new length via linear interpolation."""
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)


def smooth_and_reoptimize(a, smooth_sigma, deadline, tracker):

    n = len(a)

    kernel = np.exp(-0.5 * (x / smooth_sigma) ** 2)
    kernel /= kernel.sum()
    
    a_smooth = np.convolve(a, kernel, mode='same')

    a_smooth /= a_smooth.sum()

        print(f"Phase 3: CD after LP")
        print(f"{'='*60}", flush=True)
        t0 = time.monotonic()
        
        a = best_a.copy()
        a = coordinate_descent_multi_pass(a, 30, deadline - 180, tracker, verbose_every=5)
        tracker.report(f"Phase 3 CD: {time.monotonic()-t0:.1f}s")
    
    # === Phase 4: Try nearby lengths ===
    if time.monotonic() < deadline - 120:
        print(f"\n{'='*60}")
        print(f"Phase 4: Nearby Lengths")
        print(f"{'='*60}", flush=True)
        t0 = time.monotonic()
        
        length_candidates = sorted(set([
            n_parent - 6, n_parent - 4, n_parent - 2,
            n_parent + 2, n_parent + 4, n_parent + 6,
            n_parent + 10, n_parent + 14, n_parent - 10
        ]))
        length_candidates = [l for l in length_candidates if 150 <= l <= 1000]
        
        for nl in length_candidates:
            if time.monotonic() > deadline - 60:
                break
            
            a_res = resample_seq(best_a, nl)
            a_res = np.maximum(a_res, 0.001)
            a_res /= a_res.sum()
            
            a_opt = coordinate_descent_multi_pass(a_res, 12, time.monotonic() + 35, tracker, verbose_every=12)
        
        tracker.report(f"Phase 4 Lengths: {time.monotonic()-t0:.1f}s")
    
    # === Phase 5: Smoothing perturbations ===
    if time.monotonic() < deadline - 80:
        print(f"\n{'='*60}")
        print(f"Phase 5: Smoothing Perturbations")
        print(f"{'='*60}", flush=True)
        t0 = time.monotonic()
        
        for sigma in [1.5, 2.5, 4.0, 6.0]:
            if time.monotonic() > deadline - 40:
                break
            a_smooth = smooth_and_reoptimize(best_a, sigma, time.monotonic() + 25, tracker)
        
        tracker.report(f"Phase 5 Smooth: {time.monotonic()-t0:.1f}s")
    
    # === Phase 6: Multi-scale perturbation restarts ===
    if time.monotonic() < deadline - 50:
        print(f"\n{'='*60}")
        print(f"Phase 6: Perturbation Restarts")
        print(f"{'='*60}", flush=True)
        t0 = time.monotonic()
        
        n = best_a.size
        for temp in [0.02, 0.04, 0.01, 0.06, 0.03]:
            if time.monotonic() > deadline - 30:
                break
            
            a_pert = best_a.copy() + rng.normal(0, temp, n)
            a_pert = np.maximum(a_pert, 0.001)
            a_pert /= a_pert.sum()
            
            a_opt = coordinate_descent_multi_pass(a_pert, 10, time.monotonic() + 20, tracker, verbose_every=10)
        
        tracker.report(f"Phase 6 Perturb: {time.monotonic()-t0:.1f}s")
    
    # === Phase 7: Final intensive CD polish ===
    print(f"\n{'='*60}")
    print(f"Phase 7: Final CD Polish")
    print(f"{'='*60}", flush=True)
    t0 = time.monotonic()
    
    a_final = best_a.copy()
    a_final = coordinate_descent_multi_pass(a_final, 50, deadline - 5, tracker, verbose_every=10)
    tracker.report(f"Phase 7 Final: {time.monotonic()-t0:.1f}s")
    
    # Fallback
    if tracker.best is None:
        tracker.best = np.ones(256)
        tracker.best_score = compute_J(tracker.best)
    
    print(f"\n{'='*60}")
    print(f"FINAL RESULT")
    print(f"{'='*60}")
    print(f"J = {tracker.best_score:.10f}, n = {tracker.best.size}")
    print(f"Total time: {time.monotonic()-tracker.t_start:.1f}s", flush=True)
    
    return [float(x) for x in tracker.best]
