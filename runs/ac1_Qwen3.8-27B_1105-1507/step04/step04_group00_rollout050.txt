```python
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]
```

```python
"""
Minimizes J(a) = 2*n*max(conv(a,a))/(sum(a))^2 using:
1. L-BFGS-B with softmax parameterization and adaptive log-sum-exp smoothing
2. LP mirror descent for active-set refinement  
3. Coordinate fixing to reduce effective dimensionality
4. Multi-restart perturbation search
5. Multi-length exploration

Key innovation: softmax parameterization removes the need for projection onto
the simplex, enabling efficient second-order optimization via L-BFGS-B. 
The log-sum-exp temperature is adaptively increased to tighten the smoothing.
LP mirror descent provides exact best-response directions for local refinement.
"""

import numpy as np
from scipy.optimize import minimize, linprog
from scipy.sparse import csr_matrix
import time


def fast_conv(a):
    """Full linear self-convolution via FFT."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]


def compute_J(a):
    """Compute J(a) = 2*n*max(conv(a,a))/(sum(a))^2."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))


def softmax(x):
    """Numerically stable softmax."""
    x_shifted = x - x.max()
    exp_x = np.exp(x_shifted)
    s = exp_x.sum()
    if s < 1e-300:
        return np.ones(len(x)) / len(x)
    return exp_x / s


def compute_grad_a(a, p):
    """Compute gradient of sum_k p[k]*q_k(a) w.r.t. a.
    q_k(a) = (a*a)[k], so grad_a[j] = 2*sum_k p[k]*a[k-j] (valid k).
    """
    n = len(a)
    grad_a = np.zeros(n)
    for k in range(2 * n - 1):
        pk = p[k]
        if pk < 1e-14:
            continue
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        grad_a[lo:hi + 1] += 2.0 * pk * a[k - hi:k - lo + 1][::-1]
    return grad_a


def smoothed_func_and_grad_x(x, tau):
    """Compute smoothed objective and gradient w.r.t. x (softmax params)."""
    a = softmax(x)
    n = len(a)
    b = fast_conv(a)
    b_max = b.max()
    
    # Softmax weights
    exp_b = np.exp(tau * (b - b_max))
    exp_sum = exp_b.sum()
    p = exp_b / exp_sum
    
    # Objective
    obj = (1.0 / tau) * (b_max + np.log(exp_sum))  # = (1/tau)*log(sum exp(tau*b))
    # Actually: (1/tau)*log(sum exp(tau*b)) = (1/tau)*log(exp(tau*b_max) * sum exp(tau*(b-b_max)))
    # = b_max + (1/tau)*log(sum exp(tau*(b-b_max)))
    
    # Gradient w.r.t. a
    grad_a = compute_grad_a(a, p)
    
    # Gradient w.r.t. x (softmax Jacobian)
    dot = a @ grad_a
    grad_x = a * (grad_a - dot)
    
    return obj, grad_x


def run_lbb_softmax(a0, deadline, tau_values=None, max_iter_per_run=200):
    """Run L-BFGS-B with softmax parameterization for multiple tau values."""
    if tau_values is None:
        tau_values = [10, 20, 50, 100, 200]
    
    n = len(a0)
    best_a = a0.copy()
    best_M = fast_conv(a0).max()
    
    # Initialize x from a0
    a_clipped = np.maximum(a0, 1e-15)
    x = np.log(a_clipped / a_clipped.sum())
    
    for tau in tau_values:
        if time.monotonic() > deadline:
            break
        
        def obj(x_):
            val, _ = smoothed_func_and_grad_x(x_, tau)
            return val
        
        def grad(x_):
            _, g = smoothed_func_and_grad_x(x_, tau)
            return g
        
        t0 = time.monotonic()
        try:
            res = minimize(obj, x, jac=grad, method='L-BFGS-B',
                          options={'maxiter': max_iter_per_run, 'max_fun': 50000, 
                                   'ftol': 1e-16, 'gtol': 1e-10})
            a_new = softmax(res.x)
            M_new = fast_conv(a_new).max()
            elapsed = time.monotonic() - t0
            if M_new < best_M:
                best_M = M_new
                best_a = a_new.copy()
                x = res.x.copy()
                print(f"    tau={tau:.0f}: M={M_new:.10f} (improved) [{elapsed:.1f}s]", flush=True)
            else:
                print(f"    tau={tau:.0f}: M={M_new:.10f} (no improvement) [{elapsed:.1f}s]", flush=True)
        except Exception as e:
            print(f"    tau={tau:.0f}: error {e}", flush=True)
    
    return best_a


def build_lp_matrix(n, a):
    """Build the LP constraint matrix A where A[k,j] = a[k-j] if valid."""
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        for j in range(lo, hi + 1):
            rows.append(k)
            cols.append(j)
            vals.append(a[k - j])
    A = csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))
    return A


def lp_best_response(a, M, time_limit=10):
    """Solve LP: max sum(g) s.t. (a*g)[k] <= M, g >= 0."""
    n = len(a)
    A = build_lp_matrix(n, a)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, 2.0)] * n  # Upper bound of 2 for safety
    
    try:
        r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                    options={'time_limit': time_limit, 'presolve': True})
        if r.success and r.x is not None:
            g = np.maximum(r.x, 0)
            if g.sum() > 1e-15:
                return g / g.sum()  # Normalize to simplex
    except Exception:
        pass
    return None


def run_lp_mirror_descent(a, deadline, max_iters=500, t_init=0.5):
    """LP mirror descent with adaptive step size."""
    a = np.maximum(a.astype(np.float64), 0).copy()
    s = a.sum()
    if s < 1e-10:
        return a
    a /= s
    
    n = len(a)
    best_a = a.copy()
    best_M = fast_conv(a).max()
    no_improve = 0
    
    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        
        b = fast_conv(a)
        M = float(b.max())
        
        g = lp_best_response(a, M, time_limit=5)
        if g is None:
            break
        
        # Adaptive step size
        t = t_init * (0.95 ** (it // 10))
        
        improved = False
        for trial_t in [t, t * 0.5, t * 0.25, t * 0.1]:
            a_trial = np.maximum((1 - trial_t) * a + trial_t * g, 0)
            s_trial = a_trial.sum()
            if s_trial < 1e-10:
                continue
            a_trial /= s_trial
            M_trial = fast_conv(a_trial).max()
            if M_trial < best_M - 1e-15:
                a = a_trial
                best_M = M_trial
                best_a = a.copy()
                improved = True
                break
        
        if improved:
            no_improve = 0
        else:
            no_improve += 1
            if no_improve > 30:
                break
    
    return best_a


def fix_coordinates(a, threshold=1e-7):
    """Identify and zero out small coordinates. Returns (a_fixed, active_mask)."""
    n = len(a)
    mask = a > threshold
    a_fixed = a.copy()
    a_fixed[~mask] = 0
    s = a_fixed.sum()
    if s < 1e-10:
        return a, np.ones(n, dtype=bool)
    a_fixed /= s
    return a_fixed, mask


def resample_seq(src, dest_n):
    """Resample 1D sequence to new length."""
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)


class ScoreTracker:
    def __init__(self):
        self.best = None
        self.best_score = float('inf')
        self.t_start = time.monotonic()
    
    def update(self, a):
        a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
        if a.size == 0 or a.sum() < 0.01:
            return
        try:
            score = float(evaluate_sequence(a.tolist()))
        except Exception:
            return
        if np.isfinite(score) and score < self.best_score:
            self.best = a.copy()
            self.best_score = score
            elapsed = time.monotonic() - self.t_start
            print(f"  ** NEW BEST: J={score:.10f} n={a.size} t={elapsed:.0f}s", flush=True)
    
    def report(self, label=""):
        if self.best is not None:
            print(f"  {label} best: J={self.best_score:.10f} n={self.best.size}", flush=True)
```
