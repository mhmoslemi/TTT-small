    return np.fft.irfft(fa * fa, size)[:2 * n - 1]


def compute_J(a):
    """Compute J(a) = 2*n*max(conv(a,a))/(sum(a))^2."""
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))


def smooth_grad_fft(a, tau, size):
    """
    Compute gradient of smooth max (log-sum-exp) of conv(a,a) w.r.t. a.
    Returns (gradient, softmax_weights, conv_result).
    Uses FFT for O(n log n) computation.
    """
    n = len(a)
    b = fast_conv(a)
    b_max = b.max()
    exp_b = np.exp(tau * (b - b_max))
    p = exp_b / exp_b.sum()

    # grad_i = 2 * sum_k p[k] * a[k-i] * 1(0<=k-i<n)
    # This is 2 * cross_correlation(p, a)[i] where (p⊛a)[i] = sum_j p[i+j]*a[j]
    Fp = np.fft.rfft(p, size)
    Fa = np.fft.rfft(a, size)
    # Cross-correlation: IDFT(conj(FFT(p)) * FFT(a))
    corr = np.fft.irfft(np.conj(Fp) * Fa, size)[:n]
    grad = 2.0 * corr

    return grad, p, b


def mirror_descent_fft(a, max_iters, deadline, tau_init=5.0, tau_final=200.0,
                       lr_init=0.1, lr_min=0.005):
    """
    Mirror descent (multiplicative weights) with FFT-accelerated smooth gradient.
    Update: a_new[i] = a[i] * exp(-lr * grad[i]) / sum(...)
    This is geometrically natural for the probability simplex.
    """
    a = np.maximum(a.astype(np.float64), 1e-10)
    a = a / a.sum()
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1

    best_a = a.copy()
    best_M = fast_conv(a).max()
    no_improve = 0

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        frac = it / max(1, max_iters)
        tau = tau_init + (tau_final - tau_init) * (frac ** 0.7)
        lr = lr_init * (lr_min / lr_init) ** frac

        grad, p, b = smooth_grad_fft(a, tau, size)
        M = b.max()

        if M < best_M * (1 - 1e-9):
            best_M = M
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        # Mirror descent update: multiplicative weights
        # a_new[i] prop to a[i] * exp(-lr * grad[i])
        log_update = -lr * grad
        # Clip for numerical stability
        log_update = np.clip(log_update, -50, 50)
        weights = a * np.exp(log_update)
        s = weights.sum()
        if s < 1e-10:
            break
        a = weights / s

        if no_improve > 200:
            # Perturb to escape
            a += 0.01 * np.random.randn(n)
            a = np.maximum(a, 1e-10)
            a = a / a.sum()
            no_improve = 0

    return best_a


def gradient_descent_fft(a, max_iters, deadline, tau_init=3.0, tau_final=150.0,
                         lr_init=0.3, lr_min=0.02, momentum_beta=0.8):
    """
    Smooth gradient descent with momentum and FFT-accelerated gradients.
    """
    a = np.maximum(a.astype(np.float64), 1e-10)
    a = a / a.sum()
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1

    best_a = a.copy()
    best_M = fast_conv(a).max()
    v = np.zeros(n)
    no_improve = 0

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        frac = it / max(1, max_iters)
        tau = tau_init + (tau_final - tau_init) * (frac ** 0.5)
        lr = lr_init * (lr_min / lr_init) ** (frac ** 0.8)

        grad, p, b = smooth_grad_fft(a, tau, size)
        M = b.max()

        if M < best_M * (1 - 1e-9):
            best_M = M
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        # Momentum update
        v = momentum_beta * v + lr * grad
        a_new = a - v
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s < 1e-10:
            a_new = a.copy()
        else:
            a_new = a_new / s
        a = a_new

        if no_improve > 300:
            a += 0.02 * np.random.randn(n)
            a = np.maximum(a, 1e-10)
            a = a / a.sum()
            v = np.zeros(n)
            no_improve = 0

    return best_a


def freq_domain_project(a, max_iters, deadline, relax=0.4, target_bias=0.0):
    """
    Gerchberg-Saxton style alternating projection:
    - Frequency domain: flatten |FFT(a)|^2
    - Time domain: project to non-negative simplex
    """
    a = np.maximum(a.astype(np.float64), 0)
    s = a.sum()
    if s < 1e-10:
        return a
    a = a / s
    n = len(a)

    size = 1
    while size < 2 * n - 1:
        size <<= 1

    best_a = a.copy()
    best_J = compute_J(a)
    no_improve = 0

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        # Frequency domain step: make |A|^2 flat
        A = np.fft.rfft(a, size)
        power = np.abs(A) ** 2
        total_power = power.sum()
        # Target: flat power spectrum (with optional bias toward low frequencies)
        target_amp = np.sqrt(total_power / (size // 2 + 1))
        if target_bias != 0.0:
            freqs = np.arange(size // 2 + 1)
            bias = 1.0 + target_bias * np.exp(-10.0 * freqs / size)
            target_amp = target_amp * np.sqrt(bias)
            target_amp = target_amp * np.sqrt(total_power / ((target_amp**2).sum() * (size//2+1)))

        phase = np.angle(A)
        # Handle zero amplitude (phase undefined)
        mask = np.abs(A) > 1e-15
        A_new = np.zeros_like(A)
        A_new[mask] = target_amp[mask] * np.exp(1j * phase[mask])
        A_new[~mask] = target_amp[~mask]  # arbitrary phase for zero amp

        # Back to time domain
        a_freq = np.fft.irfft(A_new, size)[:n]

        # Project to non-negative simplex
        a_time = np.maximum(a_freq, 0)
        s_new = a_time.sum()
        if s_new < 1e-10:
            break
        a_time = a_time / s_new

        # Over-relaxation
        r = relax if it < max_iters // 3 else min(0.8, relax + 0.3 * it / max_iters)
        a_new = (1 - r) * a + r * a_time
        a_new = np.maximum(a_new, 0)
        s_new = a_new.sum()
        if s_new < 1e-10:
            break
        a = a_new / s_new

        J = compute_J(a)
        if J < best_J:
            best_J = J
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        if no_improve > 500:
            break

    return best_a


def admm_step(a, b, t, rho, lr, size):
    """One ADMM iteration step. Returns (a_new, b_new, t_new)."""
    n = len(a)
    c = fast_conv(a)
    d = b - c

    # Gradient of (rho/2)||b - conv(a,a)||^2 w.r.t. a
    # = -rho * 2 * cross_corr(d, a) [by our formula]
    Fd = np.fft.rfft(d, size)
    Fa = np.fft.rfft(a, size)
    corr = np.fft.irfft(np.conj(Fd) * Fa, size)[:n]
    grad = -2.0 * rho * corr

    # Projected gradient for a
    a_new = a - lr * grad
    a_new = np.maximum(a_new, 0)
    s = a_new.sum()
    if s < 1e-10:
        a_new = a.copy()
    else:
        a_new = a_new / s

    # b-update: b_k = min(conv(a_new, a_new)_k, t)
    c_new = fast_conv(a_new)
    b_new = np.minimum(c_new, t)

    # t-update: t = min(t, max(c_new))
    t_new = min(t, float(c_new.max()))

    return a_new, b_new, t_new


def admm_optimize(a, max_iters, deadline, rho_init=5.0):
    """
    ADMM: minimize t s.t. b_k <= t, b ~= conv(a,a), sum(a)=1, a>=0.
    """
    a = np.maximum(a.astype(np.float64), 0)
    s = a.sum()
    if s < 1e-10:
        return a
    a = a / s
    n = len(a)

    size = 1
    while size < 2 * n - 1:
        size <<= 1

    c = fast_conv(a)
    b = c.copy()
    t = float(c.max())
    rho = rho_init

    best_a = a.copy()
    best_J = compute_J(a)

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        lr = 0.5 / (1.0 + it * 0.005)
        a, b, t = admm_step(a, b, t, rho, lr, size)

        J = compute_J(a)
        if J < best_J:
            best_J = J
            best_a = a.copy()

        # Adapt rho periodically
        if it % 100 == 99:
            rho *= 1.3
            # Reset b to current conv to avoid drift
            c = fast_conv(a)
            b = np.minimum(c, t)

    return best_a


def active_set_refine(a, max_iters=100):
    """
    Refine using active set identification + KKT conditions.
    At optimum: for all i with a_i > 0,
    sum_{k active} lam_k * 2*a[k-i] = alpha (constant)
    We solve this iteratively via coordinate updates.
    """
    a = np.maximum(a.astype(np.float64), 0)
    s = a.sum()
    if s < 1e-10:
        return a
    a = a / s
    n = len(a)

    best_a = a.copy()
    best_M = fast_conv(a).max()

    for it in range(max_iters):
        b = fast_conv(a)
        M = b.max()

        if M < best_M * (1 - 1e-10):
            best_M = M
            best_a = a.copy()

        # Identify active set (with tolerance)
        tol = 1e-6 * M
        active = np.where(b >= M - tol)[0]
        if len(active) == 0:
            active = np.array([np.argmax(b)])

        # Compute uniform subgradient: grad_i = 2 * sum_{k in active} a[k-i] / len(active)
        # (Using uniform weights as a simple choice)
        w = np.zeros(2 * n - 1)
        w[active] = 1.0 / len(active)

        size = 1
        while size < 2 * n - 1:
            size <<= 1

        Fw = np.fft.rfft(w, size)
        Fa = np.fft.rfft(a, size)
        corr = np.fft.irfft(np.conj(Fw) * Fa, size)[:n]
        grad = 2.0 * corr

        # Take a small step in the negative subgradient direction
        lr = 0.1 * (1 - it / max_iters) + 0.01
        a_new = a - lr * grad
        a_new = np.maximum(a_new, 0)
        s_new = a_new.sum()
        if s_new < 1e-10:
            break
        a = a_new / s_new

        # Periodically re-identify active set with tighter tolerance
        if it % 20 == 19:
            b = fast_conv(a)
            M = b.max()
            tol_tight = 1e-8 * M
            active_tight = np.where(b >= M - tol_tight)[0]
            if len(active_tight) > 0:
                # Penalize non-active peaks by reducing components that contribute to them
                pass

    return best_a


def generate_chirp_init(n, rng):
    """Generate a chirp-based initialization (good autocorrelation properties)."""
    t = np.arange(n)
    # quadratic chirp
    freq_chirp = np.cos(2 * np.pi * (0.5 * t**2 / n + rng.uniform(0, 1)))
    a = 1.0 + 0.7 * freq_chirp
    a = np.maximum(a, 0.01)
    return a / a.sum()


def generate_barker_like_init(n, rng):
    """Generate a Barker-code-inspired initialization."""
    # Use a long Golay or complementary code structure
    # For simplicity, use a sum of cosines with specific frequencies
    t = np.arange(n)
    a = np.ones(n)
    n_harmonics = min(5, n // 32)
    for h in range(1, n_harmonics + 1):
        freq = rng.uniform(0.5, 2.0) * h / n
        phase = rng.uniform(0, 2 * np.pi)
        a += 0.3 / h * np.cos(2 * np.pi * freq * t + phase)
    a = np.maximum(a, 0.05)
    return a / a.sum()


def generate_m_seq_init(n, rng):
    """Generate an m-sequence-like initialization using PRBS pattern mapped to [0,1]."""
    # Simple PRBS-like pattern
    x = rng.integers(0, 2, n)
    # Smooth it a bit
    kernel = np.ones(3) / 3
    x_smooth = np.convolve(x, kernel, mode='same')
    a = 0.3 + 0.7 * x_smooth
    a = np.maximum(a, 0.01)
    return a / a.sum()


def generate_smooth_random(n, rng, smoothness=10):
    """Generate a smooth random initialization."""
    n_coarse = max(4, n // smoothness)
    coarse = rng.uniform(0.2, 1.0, n_coarse)
    # Interpolate
    t_coarse = np.linspace(0, 1, n_coarse)
    t_fine = np.linspace(0, 1, n)
    a = np.interp(t_fine, t_coarse, coarse)
    a = np.maximum(a, 0.01)
    return a / a.sum()


def generate_structured_inits(n, rng):
    """Generate diverse structured initializations."""
    inits = []
    t = np.linspace(0, 1, n, endpoint=False)

    # Chirp variants
    inits.append(generate_chirp_init(n, rng))
    inits.append(generate_chirp_init(n, rng))

    # Barker-like
    inits.append(generate_barker_like_init(n, rng))
    inits.append(generate_barker_like_init(n, rng))

    # M-sequence-like
    inits.append(generate_m_seq_init(n, rng))

    # Smooth random
    for sm in [5, 10, 20, 40]:
        inits.append(generate_smooth_random(n, rng, sm))

    # Cosine modulated
    for freq in [1, 2, 3, 5, 8, 13]:
        a = np.maximum(1.0 + 0.5 * np.cos(2 * np.pi * freq * t + rng.uniform(0, 2)), 0.05)
        inits.append(a / a.sum())

    # Double chirp
    a1 = np.cos(2 * np.pi * 0.3 * t**2)
    a2 = np.cos(2 * np.pi * 0.7 * t**2 + 1)
    a = np.maximum(1.0 + 0.4 * a1 + 0.3 * a2, 0.05)
    inits.append(a / a.sum())

    # Gaussian-modulated chirp
    env = np.exp(-4 * (t - 0.5)**2)
    a = np.maximum(0.3 + 0.7 * env * np.cos(2 * np.pi * 5 * t**2), 0.01)
    inits.append(a / a.sum())

    # Piecewise constant (block)
    n_blocks = rng.integers(4, max(5, n // 32))
    block_vals = rng.uniform(0.3, 1.0, n_blocks)
    block_seq = np.repeat(block_vals, n // n_blocks + 1)[:n]
    # Smooth the transitions
    a = np.convolve(block_seq, np.ones(5)/5, mode='same')
    a = np.maximum(a, 0.05)
    inits.append(a / a.sum())

    # Pure chirp (strong)
    a = np.maximum(1.0 + 0.9 * np.cos(2 * np.pi * 2.0 * t**2), 0.01)
    inits.append(a / a.sum())

    # Random phase sum of sinusoids (Wiener process in frequency)
    n_f = max(3, n // 16)
    freqs = np.sort(rng.uniform(0, 5, n_f))
    phases = rng.uniform(0, 2 * np.pi, n_f)
    amps = rng.uniform(0.1, 0.5, n_f) / np.sqrt(n_f)
    a = 1.0
    for f, ph, am in zip(freqs, phases, amps):
        a += am * np.cos(2 * np.pi * f * t + ph)
    a = np.maximum(a, 0.05)
    inits.append(a / a.sum())

    return inits


def resample_seq(src, dest_n):
    """Resample 1D sequence to new length via linear interpolation."""
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)


class ScoreTracker:
    def __init__(self):
        self.best = None
        self.best_score = float('inf')
        self.t_start = time.monotonic()

    def update(self, a):
        a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
        if a.size == 0 or a.sum() < 0.01:
            return
        score = float(evaluate_sequence(a.tolist()))
        if np.isfinite(score) and score < self.best_score:
            self.best = a.copy()
            self.best_score = score
            elapsed = time.monotonic() - self.t_start
            print(f"  ** J={score:.8f} n={a.size} t={elapsed:.0f}s", flush=True)

    def report(self, label=""):
        if self.best is not None:
            print(f"  {label} best: J={self.best_score:.8f} n={self.best.size}", flush=True)


def hybrid_optimize(a0, deadline, n_max_iters=50000):
    """
    Run multiple optimization methods and return the best result.
    Alternates between: gradient descent, mirror descent, freq projection.
    """
    a = np.maximum(a0.astype(np.float64), 1e-8)
    a = a / a.sum()
    n = len(a)

    best_a = a.copy()
    best_J = compute_J(a)

    phase_time = 30.0  # seconds per phase segment

    # Phase A: Grid descent
    dl = min(time.monotonic() + phase_time, deadline)
    a1 = gradient_descent_fft(a, n_max_iters, dl, tau_init=3, tau_final=100,
                              lr_init=0.3, lr_min=0.02)
    J1 = compute_J(a1)
    if J1 < best_J:
        best_J, best_a = J1, a1.copy()

    # Phase B: Mirror descent
    dl = min(time.monotonic() + phase_time, deadline)
    a2 = mirror_descent_fft(a1, n_max_iters, dl, tau_init=5, tau_final=150,
                            lr_init=0.1, lr_min=0.005)
    J2 = compute_J(a2)
    if J2 < best_J:
        best_J, best_a = J2, a2.copy()

    # Phase C: Frequency domain projection
    dl = min(time.monotonic() + phase_time, deadline)
    a3 = freq_domain_project(a2, n_max_iters, dl, relax=0.4)
    J3 = compute_J(a3)
    if J3 < best_J:
        best_J, best_a = J3, a3.copy()

    # Phase D: ADMM
    dl = min(time.monotonic() + phase_time, deadline)
    a4 = admm_optimize(a3, n_max_iters // 2, dl, rho_init=5.0)
    J4 = compute_J(a4)
    if J4 < best_J:
        best_J, best_a = J4, a4.copy()

    # Phase E: Another round of gradient + freq (on the best so far)
    if time.monotonic() < deadline:
        dl = min(time.monotonic() + phase_time * 0.5, deadline)
        a5 = gradient_descent_fft(best_a, 10000, dl, tau_init=10, tau_final=200,
                                  lr_init=0.1, lr_min=0.01)
        J5 = compute_J(a5)
        if J5 < best_J:
            best_J, best_a = J5, a5.copy()

        dl = min(time.monotonic() + phase_time * 0.5, deadline)
        a6 = freq_domain_project(best_a, 10000, dl, relax=0.3)
        J6 = compute_J(a6)
        if J6 < best_J:
            best_J, best_a = J6, a6.copy()

    return best_a


def propose_candidate(seed=42, budget_s=1500, **kwargs):
    """
    Multi-phase search combining frequency-domain projection, mirror descent,
    ADMM, and active-set refinement. Explores multiple lengths and
    diverse initializations including chirps and structured codes.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + float(budget_s) - 20.0
    tracker = ScoreTracker()

    # Try to load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        if p.ndim == 1 and p.size > 0 and np.all(np.isfinite(p)):
            parent = np.maximum(p, 0.0)
            tracker.update(parent)
            print(f"Parent loaded: n={parent.size}, J={tracker.best_score:.6f}", flush=True)
    except (NameError, AttributeError):
        pass

    if parent is None:
        parent = np.ones(256)
        tracker.update(parent)

    # === PHASE 1: Explore multiple lengths with hybrid optimizer ===
    p1_start = time.monotonic()
    total_p1_time = 400.0
    print(f"\n{'='*60}", flush=True)
    print(f"PHASE 1: Multi-length exploration (budget={total_p1_time:.0f}s)", flush=True)
    print(f"{'='*60}", flush=True)

    lengths = [192, 256, 320, 384, 512, 640, 768]
    # Prioritize lengths near the parent length
    if parent is not None:
        pn = parent.size
        lengths.sort(key=lambda n: abs(n - pn))

    # Allocate time: more time for fewer, more promising lengths
    time_per_length = total_p1_time / len(lengths)

    best_overall = parent.copy()
    best_overall_J = tracker.best_score

    for n in lengths:
        if time.monotonic() > deadline - 300:
            break

        phase_dl = min(time.monotonic() + time_per_length, deadline - 250)
        if time.monotonic() >= phase_dl:
            continue

        t0 = time.monotonic()
        inits = generate_structured_inits(n, rng)

        # Also include parent resampled
        if parent is not None:
            inits.append(resample_seq(parent, n))
            # Perturbed parent
            p_res = resample_seq(parent, n)
            inits.append(np.maximum(p_res + rng.normal(0, 0.05, n), 0.01))

        best_for_n = None
        best_J_n = float('inf')

        for idx, a0 in enumerate(inits):
            if time.monotonic() > phase_dl:
                break
            a0 = np.maximum(a0, 1e-6)
            a0 = a0 / a0.sum()

            # Quick hybrid optimization
            a_opt = hybrid_optimize(a0, phase_dl)
            J_opt = compute_J(a_opt)

            if J_opt < best_J_n:
                best_J_n = J_opt
                best_for_n = a_opt.copy()

        if best_for_n is not None:
            J_val = compute_J(best_for_n)
            print(f"  n={n:4d}: J={J_val:.6f} ({time.monotonic()-t0:.1f}s)", flush=True)
            if J_val < best_overall_J:
                best_overall_J = J_val
                best_overall = best_for_n.copy()
            tracker.update(best_for_n)

    tracker.report("Phase 1")
    print(f"Phase 1 time: {time.monotonic()-p1_start:.1f}s", flush=True)

    # === PHASE 2: Deep optimization on best candidate ===
    p2_start = time.monotonic()
    total_p2_time = 250.0
    print(f"\n{'='*60}", flush=True)
    print(f"PHASE 2: Deep optimization (budget={total_p2_time:.0f}s)", flush=True)
    print(f"{'='*60}", flush=True)

    n_best = tracker.best.size
    # Run longer optimization on the best
    a_deep = hybrid_optimize(tracker.best, min(time.monotonic() + total_p2_time * 0.6, deadline - 150))
    tracker.update(a_deep)

    # Active set refinement
    if time.monotonic() < deadline - 100:
        a_ref = active_set_refine(tracker.best, max_iters=200)
        tracker.update(a_ref)

    # ADMM polishing
    if time.monotonic() < deadline - 80:
        a_admm = admm_optimize(tracker.best, 30000, deadline - 80, rho_init=10.0)
        tracker.update(a_admm)

    tracker.report("Phase 2")
    print(f"Phase 2 time: {time.monotonic()-p2_start:.1f}s", flush=True)

    # === PHASE 3: Diversified restarts ===
    p3_start = time.monotonic()
    total_p3_time = 250.0
    print(f"\n{'='*60}", flush=True)
    print(f"PHASE 3: Diversified restarts (budget={total_p3_time:.0f}s)", flush=True)
    print(f"{'='*60}", flush=True)

    n_best = tracker.best.size
    n_restarts = 15
    time_per_restart = total_p3_time / n_restarts

    for r in range(n_restarts):
        if time.monotonic() > deadline - 150:
            break
        restart_dl = min(time.monotonic() + time_per_restart, deadline - 130)

        # Different perturbation strategies
        a = tracker.best.copy().astype(np.float64)

        if r % 5 == 0:
            # Gaussian noise
            noise_scale = 0.05 * (1 + 0.2 * r)
            a = a + rng.normal(0, noise_scale, n_best)
        elif r % 5 == 1:
            # Block flip
            block_size = n_best // 4
            start = rng.integers(0, n_best - block_size)
            a[start:start+block_size] *= 0.5
        elif r % 5 == 2:
            # Frequency perturbation
            A = np.fft.rfft(a, 2 * n_best)
            A *= (1 + 0.1 * rng.normal(0, 1, A.size))
            a = np.fft.irfft(A, 2 * n_best)[:n_best]
        elif r % 5 == 3:
            # Rescale different halves
            half = n_best // 2
            a[:half] *= rng.uniform(0.6, 1.3)
            a[half:] *= rng.uniform(0.6, 1.3)
        else:
            # Morph with a new random init
            a_new = generate_smooth_random(n_best, rng, smoothness=8)
            a = 0.7 * a + 0.3 * a_new

        a = np.maximum(a, 1e-6)
        a = a / a.sum()

        # Quick optimization
        a_opt = hybrid_optimize(a, restart_dl)
        tracker.update(a_opt)

    tracker.report("Phase 3")
    print(f"Phase 3 time: {time.monotonic()-p3_start:.1f}s", flush=True)

    # === PHASE 4: Length extension ===
    p4_start = time.monotonic()
    print(f"\n{'='*60}", flush=True)
    print(f"PHASE 4: Length extension", flush=True)
    print(f"{'='*60}", flush=True)

    n_best = tracker.best.size
    for mult in [2, 3]:
        if time.monotonic() > deadline - 60:
            break
        new_n = n_best * mult
        if new_n > 3072 or new_n < 128:
            continue

        a_ext = resample_seq(tracker.best, new_n)
        a_ext = np.maximum(a_ext, 1e-6)
        a_ext = a_ext / a_ext.sum()

        a_opt = hybrid_optimize(a_ext, min(time.monotonic() + 60, deadline - 40))
        tracker.update(a_opt)

    tracker.report("Phase 4")
    print(f"Phase 4 time: {time.monotonic()-p4_start:.1f}s", flush=True)

    # === PHASE 5: Final intensive polishing ===
    p5_start = time.monotonic()
    print(f"\n{'='*60}", flush=True)
    print(f"PHASE 5: Final polishing", flush=True)
    print(f"{'='*60}", flush=True)

    if tracker.best is not None and time.monotonic() < deadline - 10:
        # Long gradient descent with very fine tau
        a_final = gradient_descent_fft(tracker.best, 100000, deadline - 5,
                                       tau_init=50, tau_final=500,
                                       lr_init=0.05, lr_min=0.001)
        tracker.update(a_final)

        # One more freq projection pass
        if time.monotonic() < deadline - 5:
            a_fp = freq_domain_project(tracker.best, 50000, deadline - 3, relax=0.2)
            tracker.update(a_fp)

        # Active set
        if time.monotonic() < deadline - 3:
            a_as = active_set_refine(tracker.best, max_iters=500)
            tracker.update(a_as)

    tracker.report("Phase 5")
    print(f"Phase 5 time: {time.monotonic()-p5_start:.1f}s", flush=True)

    # Fallback
    if tracker.best is None:
        tracker.best = np.ones(128)
        tracker.best_score = compute_J(tracker.best)

    print(f"\n{'='*60}", flush=True)
    print(f"FINAL: J={tracker.best_score:.8f} n={tracker.best.size} "
          f"time={time.monotonic()-tracker.t_start:.0f}s", flush=True)
    print(f"{'='*60}", flush=True)

    return [float(x) for x in tracker.best]
