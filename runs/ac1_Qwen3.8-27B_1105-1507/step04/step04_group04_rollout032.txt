"""
Advanced step function optimization combining adaptive multiplicative gradient descent,
alternating LP, targeted mass-shifting, and SLSQP polish. Improvements over previous
approach: (1) Adam-style per-coordinate adaptive step sizes, (2) annealed softmax
smoothing for better gradient quality, (3) targeted mass-shifting based on worst/best
lag analysis, (4) SLSQP final polish with analytical gradient, (5) more aggressive
exploration of nearby n values with careful time budgeting.
"""

import numpy as np
from scipy.optimize import linprog, minimize
import time


def compute_conv(a):
    """Compute linear convolution conv(a,a) using FFT. Returns array of length 2n-1."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    conv = np.fft.irfft(fa * fa, size)
    return conv[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    conv = compute_conv(a)
    max_b = float(np.max(conv))
    sum_a = float(np.sum(a))
    if sum_a < 0.01:
        return np.inf
    return 2.0 * n * max_b / (sum_a * sum_a)


def softmax_weights(conv, tau):
    """Compute softmax weights over convolution values."""
    shifted = conv / tau
    shifted -= np.max(shifted)
    w = np.exp(shifted)
    s = np.sum(w)
    if s < 1e-300:
        idx = int(np.argmax(conv))
        w = np.zeros(len(conv))
        w[idx] = 1.0
    else:
        w /= s
    return w


def compute_grad_weighted(p, a):
    """
    Compute gradient of sum_k p[k]*conv(a,a)[k] with respect to a.
    d(conv(a,a)[k])/da[m] = 2*a[k-m] if 0 <= k-m <= n-1, else 0.
    So grad[m] = 2 * sum_k p[k]*a[k-m] (with valid indices).
    Equivalent: grad[m] = 2 * sum_j p[m+j]*a[j] where j in [0,n-1] and m+j in [0,2n-2].
    """
    n = len(a)
    # Build Hankel matrix: H[m, j] = p[m+j] if m+j < 2n-1 else 0
    # grad = 2 * H @ a
    grad = np.zeros(n)
    for m in range(n):
        # j ranges from 0 to n-1, m+j ranges from m to m+n-1
        # p is valid for indices 0 to 2n-2
        j_max = min(n - 1, 2 * n - 2 - m)
        if j_max >= 0:
            grad[m] = 2.0 * np.dot(p[m:m + j_max + 1], a[:j_max + 1])
    return grad


def compute_grad_direct(p, a):
    """Faster gradient computation using FFT-based cross-correlation."""
    n = len(a)
    conv_len = 2 * n - 1
    size = 1
    while size < conv_len + n - 1:
        size <<= 1
    # grad[m] = 2 * sum_j p[m+j] * a[j]
    # This is the cross-correlation of p (length 2n-1) with a (length n) at lag m
    # Use: grad/2 = ifft(fft(p, size) * conj(fft(a, size)))
    # But need to be careful with the direction
    # Actually: sum_j p[m+j]*a[j] = (p reversed cross-correlate a) at some lag
    # Let me just use the direct approach for clarity with n <= 2000
    grad = np.zeros(n)
    for m in range(n):
        end = min(n, 2 * n - 1 - m)
        if end > 0:
            grad[m] = 2.0 * np.dot(p[m:m + end], a[:end])
    return grad


def multiplicative_adam(a, time_budget=100.0, tau_start=10.0, tau_end=1.0,
                        lr_start=0.15, lr_min=0.005, beta1=0.9, beta2=0.999):
    """
    Multiplicative Adam optimization with annealed softmax smoothing.
    a should be normalized (sum=1).
    """
    n = len(a)
    a = a.copy().astype(np.float64)
    s = np.sum(a)
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    m = np.zeros(n)  # first moment
    v = np.zeros(n)  # second moment
    t = 0

    while time.monotonic() - start < time_budget:
        t += 1
        progress = min(1.0, t / max(1, int(time_budget * 20)))
        tau = tau_start + (tau_end - tau_start) * progress
        lr = max(lr_start * (0.98 ** t), lr_min)

        conv = compute_conv(a)
        p = softmax_weights(conv, tau)
        grad = compute_grad_weighted(p, a)

        # Center gradient to preserve sum approximately
        grad = grad - np.mean(grad)

        # Adam update
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * grad * grad
        m_hat = m / (1 - beta1 ** t)
        v_hat = v / (1 - beta2 ** t)
        update = m_hat / (np.sqrt(v_hat) + 1e-8)

        # Multiplicative update in log-space
        log_step = -lr * update
        log_step = np.clip(log_step, -5.0, 5.0)
        a_new = a * np.exp(log_step)
        a_new = np.maximum(a_new, 1e-15)
        s_new = np.sum(a_new)
        if s_new > 1e-10:
            a_new = a_new / s_new
        a = a_new

        obj = compute_obj(a)
        if obj < best_obj - 1e-14:
            best_obj = obj
            best_a = a.copy()

    return best_a


def build_toeplitz(a):
    """Build matrix A where (A@g)[k] = conv(a,g)[k] for k=0..2n-2."""
    n = len(a)
    A = np.zeros((2 * n - 1, n), dtype=np.float64)
    for j in range(n):
        k_start = j
        k_end = min(j + n, 2 * n - 1)
        A[k_start:k_end, j] = a[:k_end - k_start]
    return A


def solve_lp_maxsum(a, M, time_limit=6):
    """Solve LP: max sum(g) s.t. conv(a,g)[k] <= M for all k, g >= 0."""
    n = len(a)
    A = build_toeplitz(a)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': time_limit})
        if res.status == 0:
            g = np.maximum(res.x, 0)
            return g, float(np.sum(g))
    except Exception:
        pass
    return None, None


def solve_lp_minpeak(a, time_limit=6):
    """Solve LP: min M s.t. conv(a,g)[k] <= M, sum(g)=1, g>=0."""
    n = len(a)
    A = build_toeplitz(a)
    c = np.zeros(n + 1)
    c[-1] = 1.0
    A_ub = np.zeros((2 * n - 1, n + 1))
    A_ub[:, :n] = A
    A_ub[:, n] = -1.0
    b_ub = np.zeros(2 * n - 1)
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])
    bounds = [(0.0, None)] * n + [(0.0, None)]
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': time_limit})
        if res.status == 0:
            g = np.maximum(res.x[:n], 0)
            s = np.sum(g)
            if s < 1e-10:
                return None, None
            g = g / s
            return g, float(res.x[n])
    except Exception:
        pass
    return None, None

def targeted_mass_shift(a, time_budget=100.0):
    """
    Targeted mass shifting: identify worst lag (max conv) and best lag (min conv),
    compute optimal direction to shift mass, and apply.
    """
    n = len(a)
    a = a.copy().astype(np.float64)
    s = np.sum(a)
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    step_size = 0.03

    while time.monotonic() - start < time_budget - 5:
        remaining = time_budget - (time.monotonic() - start)
        if remaining < 10:
            break

        conv = compute_conv(a)
        k_max = int(np.argmax(conv))
        k_min = int(np.argmin(conv))

        # Gradient at max lag: grad_max[m] = 2*a[k_max - m] (valid)
        # Gradient at min lag: grad_min[m] = 2*a[k_min - m] (valid)
        # We want to decrease conv[k_max] and increase conv[k_min]
        # Direction: reduce a where it contributes to k_max, increase where it contributes to k_min
        # Score[m] = a[k_min - m] - a[k_max - m] (positive means good to increase a[m])

        score = np.zeros(n)
        for m in range(n):
            # Contribution to k_max: a[k_max - m] if valid
            idx_max = k_max - m
            contrib_max = a[idx_max] if 0 <= idx_max < n else 0.0
            # Contribution to k_min: a[k_min - m] if valid
            idx_min = k_min - m
            contrib_min = a[idx_min] if 0 <= idx_min < n else 0.0
            score[m] = contrib_min - contrib_max

        # Also consider the second-worst and second-best lags
        conv_sorted = np.sort(conv)[::-1]
        threshold = conv_sorted[min(5, len(conv_sorted) - 1)]
        worst_lags = np.where(conv >= threshold * 0.9)[0]
        best_lags = np.where(conv <= np.sort(conv)[min(5, len(conv) - 1)] * 1.1)[0]

        # Multi-lag score
        multi_score = np.zeros(n)
        for m in range(n):
            s_worst = 0.0
            s_best = 0.0
            for k in worst_lags:
                idx = int(k) - m
                if 0 <= idx < n:
                    s_worst += a[idx]
            for k in best_lags:
                idx = int(k) - m
                if 0 <= idx < n:
                    s_best += a[idx]
            multi_score[m] = s_best / max(len(best_lags), 1) - s_worst / max(len(worst_lags), 1)

        # Combine scores
        direction = 0.5 * score + 0.5 * multi_score
        # Center to preserve sum
        direction = direction - np.mean(direction)

        # Apply multiplicative shift
        a_new = a * np.exp(step_size * direction)
        a_new = np.maximum(a_new, 1e-15)
        s_new = np.sum(a_new)
        if s_new > 1e-10:
            a_new = a_new / s_new

        obj_new = compute_obj(a_new)
        if obj_new < best_obj - 1e-14:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new
            step_size = max(step_size * 0.99, 0.005)
        else:
            step_size = max(step_size * 0.7, 0.003)
            if step_size < 0.004:
                # Try a random restart of the step size
                step_size = 0.02
                # Small random perturbation
                rng = np.random.default_rng(int(time.monotonic() * 100) % 99999)
                a = a * np.exp(rng.normal(0, 0.01, n))
                a = np.maximum(a, 1e-15)
                a = a / np.sum(a)

    return best_a


def slsqp_polish(a, time_budget=60.0):
    """
    Use SLSQP to polish the solution. Minimize max(conv(a,a)) subject to sum(a)=1, a>=0.
    Uses analytical gradient of the max function.
    """
    n = len(a)
    a = a.copy().astype(np.float64)
    s = np.sum(a)
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    def objective(x):
        x = np.maximum(x, 0)
        sx = np.sum(x)
        if sx < 1e-10:
            return 1e10
        x = x / sx
        conv = compute_conv(x)
        return np.max(conv)

    def gradient(x):
        x = np.maximum(x, 0)
        sx = np.sum(x)
        if sx < 1e-10:
            return np.zeros(len(x))
        x = x / sx
        conv = compute_conv(x)
        k_max = int(np.argmax(conv))
        # Gradient of conv[k_max] with respect to x[m]
        grad = np.zeros(n)
        for m in range(n):
            idx = k_max - m
            if 0 <= idx < n:
                grad[m] = 2.0 * x[idx]
        # Adjust for normalization (sum(x)=1 constraint handled by SLSQP)
        return grad

    constraints = [
        {'type': 'eq', 'fun': lambda x: np.sum(x) - 1.0, 'jac': lambda x: np.ones(len(x))},
    ]
    bounds = [(0.0, 5.0)] * n

    max_iter = max(10, int((time_budget - 5) * 3))
    for attempt in range(3):
        if time.monotonic() - start > time_budget - 5:
            break
        try:
            res = minimize(objective, best_a, jac=gradient, method='SLSQP',
                           bounds=bounds, constraints=constraints,
                           options={'maxiter': max_iter, 'ftol': 1e-12, 'disp': False})
            if res.x is not None:
                x_new = np.maximum(res.x, 0)
                sx = np.sum(x_new)
                if sx > 1e-10:
                    x_new = x_new / sx
                    obj_new = compute_obj(x_new)
                    if obj_new < best_obj - 1e-14:
                        best_obj = obj_new
                        best_a = x_new.copy()
        except Exception:
            break

    return best_a

def resample_sequence(a, new_n):
    """Resample a sequence to a new length using cubic-like interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    old_k = np.arange(n, dtype=np.float64)
    new_k = np.linspace(0, n - 1, new_n)
    a_new = np.interp(new_k, old_k, a)
    a_new = np.maximum(a_new, 0)
    s = np.sum(a_new)
    if s > 1e-10:
        a_new = a_new / s
    return a_new


def try_parent():
    """Try to load parent sequence."""
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64)
        parent = np.clip(parent, 0, 1000.0)
        if parent.size > 0 and np.isfinite(parent).all():
            obj = compute_obj(parent)
            print(f"Parent (n={parent.size}): obj={obj:.6f}")
            return parent.copy(), obj
    except Exception:
        pass
    return None, np.inf


def optimize_pipeline(a, time_budget=120.0, verbose=False):
    """Run full optimization pipeline on a sequence."""
    n = len(a)
    a = a.copy().astype(np.float64)
    s = np.sum(a)
    if s < 1e-10:
        a = np.ones(n)
        s = n
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    # Phase 1: Multiplicative Adam
    remaining = time_budget - (time.monotonic() - start)
    if remaining > 10:
        t1 = min(remaining * 0.25, 30.0)
        a_new = multiplicative_adam(best_a, time_budget=t1)
        obj = compute_obj(a_new)
        if obj < best_obj:
            best_obj = obj
            best_a = a_new.copy()
            if verbose:
                print(f"    After Adam: {best_obj:.8f}")

    # Phase 2: Alternating LP (max-sum)
    remaining = time_budget - (time.monotonic() - start)
    if remaining > 20:
        t2 = min(remaining * 0.3, 40.0)
        a_new = alternating_lp(best_a, time_budget=t2, use_minpeak=False)
        obj = compute_obj(a_new)
        if obj < best_obj:
            best_obj = obj
            best_a = a_new.copy()
            if verbose:
                print(f"    After LP: {best_obj:.8f}")

    # Phase 3: Targeted mass shift
    remaining = time_budget - (time.monotonic() - start)
    if remaining > 15:
        t3 = min(remaining * 0.2, 25.0)
        a_new = targeted_mass_shift(best_a, time_budget=t3)
        obj = compute_obj(a_new)
        if obj < best_obj:
            best_obj = obj
            best_a = a_new.copy()
            if verbose:
                print(f"    After TMS: {best_obj:.8f}")

    # Phase 4: Alternating LP (min-peak)
    remaining = time_budget - (time.monotonic() - start)
    if remaining > 20:
        t4 = min(remaining * 0.2, 30.0)
        a_new = alternating_lp(best_a, time_budget=t4, use_minpeak=True)
        obj = compute_obj(a_new)
        if obj < best_obj:
            best_obj = obj
            best_a = a_new.copy()
            if verbose:
                print(f"    After LP-MP: {best_obj:.8f}")

    # Phase 5: Multiplicative Adam (second pass, higher tau)
    remaining = time_budget - (time.monotonic() - start)
    if remaining > 10:
        t5 = min(remaining * 0.2, 20.0)
        a_new = multiplicative_adam(best_a, time_budget=t5, tau_start=20.0, tau_end=5.0,
                                    lr_start=0.08, lr_min=0.003)
        obj = compute_obj(a_new)
        if obj < best_obj:
            best_obj = obj
            best_a = a_new.copy()
            if verbose:
                print(f"    After Adam2: {best_obj:.8f}")

    # Phase 6: Fine tune
    remaining = time_budget - (time.monotonic() - start)
    if remaining > 10:
        t6 = min(remaining * 0.3, 25.0)
        a_new = fine_tune(best_a, time_budget=t6)
        obj = compute_obj(a_new)
        if obj < best_obj:
            best_obj = obj
            best_a = a_new.copy()
            if verbose:
                print(f"    After FT: {best_obj:.8f}")

    return best_a, best_obj


def intensive_local_search(a, time_budget=100.0, verbose=False):
    """Intensive local search with targeted and random perturbations."""
    n = len(a)
    a = a.copy().astype(np.float64)
    s = np.sum(a)
    if s < 1e-10:
        return a, compute_obj(a)
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    if verbose:
        print(f"  Local search start: obj={best_obj:.8f}")

    attempt = 0
    while time.monotonic() - start < time_budget - 8:
        attempt += 1
        remaining = time_budget - (time.monotonic() - start)
        if remaining < 12:
            break

        rng = np.random.default_rng(attempt * 7919 + 31337)

        # Different perturbation strategies
        strategy = attempt % 5
        if strategy == 0:
            # Small multiplicative noise
            a_pert = best_a * np.exp(rng.normal(0, 0.015, n))
        elif strategy == 1:
            # Targeted: boost best lag positions, suppress worst
            conv = compute_conv(best_a)
            k_max = int(np.argmax(conv))
            k_min = int(np.argmin(conv))
            a_pert = best_a.copy()
            for _ in range(3):
                # Suppress positions contributing to k_max
                idx = rng.integers(0, n)
                a_pert[idx] *= 0.8
                # Boost positions contributing to k_min
                idx2 = rng.integers(0, n)
                a_pert[idx2] *= 1.2
        elif strategy == 2:
            # Swap-like: reduce max element, increase min
            a_pert = best_a.copy()
            max_idx = int(np.argmax(a_pert))
            min_idx = int(np.argmin(a_pert))
            delta = (a_pert[max_idx] - a_pert[min_idx]) * rng.uniform(0.05, 0.15)
            a_pert[max_idx] -= delta
            a_pert[min_idx] += delta
        elif strategy == 3:
            # Frequency-domain perturbation
            size = 1
            while size < 2 * n - 1:
                size <<= 1
            fa = np.fft.rfft(a_pert, size)
            # Perturb the phase of some frequencies
            for fi in range(rng.integers(1, min(10, fa.size))):
                fa[fi + rng.integers(0, 5)] *= np.exp(1j * rng.normal(0, 0.1))
            a_pert = np.fft.irfft(fa, size)[:n]
            a_pert = np.maximum(a_pert, 0)
        else:
            # Localized perturbation
            center = rng.integers(0, n)
            width = rng.integers(3, max(4, n // 10))
            mask = np.exp(-0.5 * ((np.arange(n) - center) / width) ** 2)
            a_pert = best_a * (1.0 + 0.1 * mask * rng.normal(0, 1, n))

        a_pert = np.maximum(a_pert, 1e-10)
        s_pert = np.sum(a_pert)
        if s_pert < 1e-10:
            continue
        a_pert = a_pert / s_pert

        # Quick optimization
        sub_time = min(15.0, remaining * 0.25)
        a_opt, obj = optimize_pipeline(a_pert, time_budget=sub_time, verbose=False)

        if obj < best_obj:
            best_obj = obj
            best_a = a_opt.copy()
            if verbose and attempt % 10 == 0:
                print(f"    Attempt {attempt}: {best_obj:.8f}")

    return best_a, best_obj


def propose_candidate(seed=42, budget_s=1500):
    """
    Main search: multi-phase optimization focusing on the parent sequence,
    with targeted improvements using Adam-style multiplicative updates,
    alternating LP, mass-shifting, and SLSQP polish.
    """
    start_time = time.monotonic()
    deadline = start_time + float(budget_s) - 30.0  # 30s buffer

    best_seq = None
    best_score = np.inf

    # Load parent
    parent, parent_obj = try_parent()
    if parent is not None and parent_obj < best_score:
        best_score = parent_obj
        best_seq = parent.copy()
        n_parent = len(parent)
    else:
        n_parent = 330

    print(f"Initial best: {best_score:.6f} (n={n_parent if best_seq is not None else 'N/A'})")
    print(f"Time budget: {budget_s}s, deadline in {deadline - start_time:.0f}s")

    # Phase 1: Intensive optimization of parent with full pipeline
    if best_seq is not None:
        print("\n=== Phase 1: Full pipeline on parent ===")
        remaining = deadline - time.monotonic()
        p1_budget = min(remaining * 0.2, 250.0)
        a_opt, obj = optimize_pipeline(best_seq, time_budget=p1_budget, verbose=True)
        if obj < best_score:
            best_score = obj
            best_seq = a_opt.copy()
            print(f"  After pipeline: {best_score:.8f}")

    # Phase 2: Targeted mass shifting (dedicated)
    if best_seq is not None:
        remaining = deadline - time.monotonic()
        if remaining > 50:
            print(f"\n=== Phase 2: Targeted mass shift ({min(remaining*0.15, 150.0):.0f}s) ===")
            t2 = min(remaining * 0.15, 150.0)
            a_opt, obj = targeted_mass_shift(best_seq, time_budget=t2)
            if obj < best_score:
                best_score = obj
                best_seq = a_opt.copy()
                print(f"  After TMS: {best_score:.8f}")

    # Phase 3: Alternating LP intensive
    if best_seq is not None:
        remaining = deadline - time.monotonic()
        if remaining > 50:
            print(f"\n=== Phase 3: Intensive alternating LP ===")
            t3 = min(remaining * 0.2, 200.0)
            a_opt = alternating_lp(best_seq, time_budget=t3, use_minpeak=False)
            obj = compute_obj(a_opt)
            if obj < best_score:
                best_score = obj
                best_seq = a_opt.copy()
                print(f"  After LP: {best_score:.8f}")

            remaining = deadline - time.monotonic()
            if remaining > 30:
                t3b = min(remaining * 0.1, 100.0)
                a_opt = alternating_lp(best_seq, time_budget=t3b, use_minpeak=True)
                obj = compute_obj(a_opt)
                if obj < best_score:
                    best_score = obj
                    best_seq = a_opt.copy()
                    print(f"  After LP-MP: {best_score:.8f}")

    # Phase 4: Multiplicative Adam with different parameters
    if best_seq is not None:
        remaining = deadline - time.monotonic()
        if remaining > 30:
            print(f"\n=== Phase 4: Adam variants ===")
            t4 = min(remaining * 0.1, 80.0)
            a_opt = multiplicative_adam(best_seq, time_budget=t4, tau_start=5.0, tau_end=0.5,
                                        lr_start=0.2, lr_min=0.01)
            obj = compute_obj(a_opt)
            if obj < best_score:
                best_score = obj
                best_seq = a_opt.copy()
                print(f"  After Adam-fast: {best_score:.8f}")

            remaining = deadline - time.monotonic()
            if remaining > 30:
                t4b = min(remaining * 0.1, 80.0)
                a_opt = multiplicative_adam(best_seq, time_budget=t4b, tau_start=50.0, tau_end=10.0,
                                            lr_start=0.05, lr_min=0.002)
                obj = compute_obj(a_opt)
                if obj < best_score:
                    best_score = obj
                    best_seq = a_opt.copy()
                    print(f"  After Adam-slow: {best_score:.8f}")

    # Phase 5: Intensive local search
    if best_seq is not None:
        remaining = deadline - time.monotonic()
        if remaining > 50:
            print(f"\n=== Phase 5: Intensive local search ({min(remaining*0.2, 200.0):.0f}s) ===")
            t5 = min(remaining * 0.2, 200.0)
            a_opt, obj = intensive_local_search(best_seq, time_budget=t5, verbose=True)
            if obj < best_score:
                best_score = obj
                best_seq = a_opt.copy()
                print(f"  After local search: {best_score:.8f}")

    # Phase 6: SLSQP polish
    if best_seq is not None:
        remaining = deadline - time.monotonic()
        if remaining > 30:
            print(f"\n=== Phase 6: SLSQP polish ===")
            t6 = min(remaining * 0.1, 80.0)
            a_opt = slsqp_polish(best_seq, time_budget=t6)
            obj = compute_obj(a_opt)
            if obj < best_score:
                best_score = obj
                best_seq = a_opt.copy()
                print(f"  After SLSQP: {best_score:.8f}")

    # Phase 7: Try nearby n values
    if best_seq is not None:
        remaining = deadline - time.monotonic()
        if remaining > 40:
            print(f"\n=== Phase 7: Nearby n values ===")
            n_best = len(best_seq)
            n_candidates = sorted(set([n_best + d for d in [-5, -3, -2, -1, 1, 2, 3, 5, -10, 10, -15, 15]]
                                     if 100 <= n_best + d <= 2000))
            per_n_time = min(25.0, remaining / max(len(n_candidates), 1) * 1.5)

            for n_val in n_candidates:
                if time.monotonic() > deadline - 30:
                    break
                remaining = deadline - time.monotonic()
                if remaining < 20:
                    break
                a_resampled = resample_sequence(best_seq, n_val)
                a_opt, obj = optimize_pipeline(a_resampled, time_budget=per_n_time, verbose=False)
                if obj < best_score:
                    best_score = obj
                    best_seq = a_opt.copy()
                    print(f"  ** UPDATE: n={n_val}, obj={best_score:.8f} **")

    # Phase 8: Final fine-tuning
    if best_seq is not None:
        remaining = deadline - time.monotonic()
        if remaining > 20:
            print(f"\n=== Phase 8: Final fine-tune ({min(remaining*0.3, 80.0):.0f}s) ===")
            t8 = min(remaining * 0.3, 80.0)
            a_opt = fine_tune(best_seq, time_budget=t8)
            obj = compute_obj(a_opt)
            if obj < best_score:
                best_score = obj
                best_seq = a_opt.copy()
                print(f"  After final FT: {best_score:.8f}")

            # One more targeted mass shift
            remaining = deadline - time.monotonic()
            if remaining > 15:
                t8b = min(remaining * 0.2, 50.0)
                a_opt, obj = targeted_mass_shift(best_seq, time_budget=t8b)
                if obj < best_score:
                    best_score = obj
                    best_seq = a_opt.copy()
                    print(f"  After final TMS: {best_score:.8f}")

    # Phase 9: Try completely different initializations if time allows
    if best_seq is not None:
        remaining = deadline - time.monotonic()
        if remaining > 60:
            print(f"\n=== Phase 9: Alternative initializations ===")
            n_curr = len(best_seq)
            for alt_seed in [seed + 1001, seed + 2002, seed + 3003]:
                if time.monotonic() > deadline - 30:
                    break
                remaining = deadline - time.monotonic()
                if remaining < 40:
                    break
                a_alt = random_init(n_curr, alt_seed)
                sub_budget = min(40.0, remaining * 0.4)
                a_opt, obj = optimize_pipeline(a_alt, time_budget=sub_budget, verbose=False)
                if obj < best_score:
                    best_score = obj
                    best_seq = a_opt.copy()
                    print(f"  ** UPDATE from alt init: obj={best_score:.8f} **")

    # Final validation
    if best_seq is None:
        best_seq = np.ones(330) / 330
        best_score = compute_obj(best_seq)

    # Clip and validate
    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    val = evaluate_sequence(result)
    if val < best_score:
        best_score = val
        # Re-evaluate to make sure
        result = [float(max(0.0, min(1000.0, x))) for x in best_seq]

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(best_seq)}, score={best_score:.8f}, time={elapsed:.1f}s ===")

    return result
