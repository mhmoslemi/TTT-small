import numpy as np
from scipy.optimize import linprog, minimize
from scipy.sparse import csr_matrix
import time


def fast_conv(a):
    """Full self-convolution via FFT."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]


def compute_J(a):
    """Compute J(a) = 2*n*max(conv(a,a))/(sum(a))^2."""
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))


def build_A_sparse(n, a):
    """Build sparse convolution matrix A[k,j] = a[k-j] (valid j)."""
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        j_min = max(0, k - n + 1)
        j_max = min(n - 1, k)
        for j in range(j_min, j_max + 1):
            rows.append(k)
            cols.append(j)
            vals.append(a[k - j])
    return csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))


def lp_best_response(a, M):
    """Solve LP: max sum(g) s.t. A@g <= M*1, g >= 0."""
    n = len(a)
    A = build_A_sparse(n, a)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    try:
        r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
        if r.success and r.x is not None:
            g = np.maximum(r.x, 0)
            if g.sum() > 1e-15:
                return g
    except Exception:
        pass
    return None


def fft_subgradient(a, tol_frac=1e-8):
    """Compute subgradient of max(conv(a,a)) using FFT. Returns (grad, active_count)."""
    n = len(a)
    b = fast_conv(a)
    M = b.max()
    if M < 1e-30:
        return np.zeros(n), 1, M
    tol = tol_frac * M
    active = b >= (M - tol)
    active_count = max(1, int(active.sum()))
    
    # Build active mask and compute gradient via FFT
    # grad[j] = (2/|A|) * sum_l a[l] * m[j+l] for l=0..n-1
    # This is cross-correlation of a with active mask m
    size = 1
    while size < 2 * n:
        size <<= 1
    
    # a reversed for correlation
    a_rev = np.zeros(size)
    a_rev[:n] = a[::-1]
    m_ext = np.zeros(size)
    m_ext[:2*n-1] = active.astype(np.float64)
    
    fa = np.fft.rfft(a_rev)
    fm = np.fft.rfft(m_ext)
    corr = np.fft.irfft(fa * fm, size)
    
    # corr[n-1+j] = sum_l a[n-1-l] * m[n-1+j-l] = sum_l' a[l'] * m[j+l']
    grad = (2.0 / active_count) * corr[n-1:n-1+n]
    return grad, active_count, M


def fft_subgradient_polish(a, max_iters=5000, deadline=None, lr_scale=1.0):
    """Accelerated subgradient descent with momentum for -max(conv(a,a))."""
    a = np.maximum(a.astype(np.float64), 1e-10)
    a = a / a.sum()
    n = len(a)
    best_M = fast_conv(a).max()
    best_a = a.copy()
    
    # Nesterov momentum
    a_prev = a.copy()
    a_prev2 = a.copy()
    momentum = np.zeros(n)
    beta = 0.7
    
    for it in range(max_iters):
        if deadline is not None and time.monotonic() > deadline:
            break
        
        grad, act_count, M = fft_subgradient(a)
        
        if M < best_M * (1 + 1e-12):
            best_M = M
            best_a = a.copy()
        
        # Adaptive learning rate
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-15:
            break
        t = lr_scale / (gnorm * np.sqrt(it + 1.0))
        t = min(t, 0.5)
        
        # Momentum update
        momentum = beta * momentum + (1 - beta) * grad / gnorm
        a_new = a - t * momentum
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s < 1e-10:
            break
        a_new = a_new / s
        
        # Nesterov step
        y = a_new + (it + 1.0) / (it + 3.0) * (a_new - a_prev)
        y = np.maximum(y, 0.0)
        s_y = y.sum()
        if s_y > 1e-10:
            y = y / s_y
            # Check if y is better
            M_y = fast_conv(y).max()
            if M_y <= M:
                a_prev2 = a_prev
                a_prev = a
                a = y
            else:
                a_prev2 = a_prev
                a_prev = a
                a = a_new
        else:
            a_prev2 = a_prev
            a_prev = a
            a = a_new
    
    # Final: use best_a
    return best_a, best_M


def lp_mirror_descent_v2(a, max_iters, deadline, use_multiplicative=False, momentum_coef=0.5):
    """LP mirror descent with momentum and adaptive step size."""
    a = np.maximum(a.astype(np.float64), 1e-10)
    a = a / a.sum()

    best_J = current_J
    best_a = a.copy()

    momentum_dir = np.zeros(n)

        if time.monotonic() > deadline:
            break
        
        b = fast_conv(a)
        M = float(b.max())
        g = lp_best_response(a, M)
        if g is None:
            break
        g = np.maximum(g, 0.0)
        g_sum = g.sum()
        if g_sum < 1e-10:
            break
        g = g / g_sum
        
        # Momentum on direction
        direction = g - a
        if momentum_coef > 0:
            momentum_dir = momentum_coef * momentum_dir + (1 - momentum_coef) * direction
            direction = momentum_dir
        
        # Adaptive step size with backtracking
        base_t = 0.3 * np.exp(-0.2 * it / 50.0) + 0.03
        t = min(1.0, base_t)
        
        best_trial_J = current_J
        best_trial_a = None
        for ls in range(20):
            if use_multiplicative:
                ratio = g / np.maximum(a, 1e-15)
                a_trial = a * np.power(ratio, t)
            else:
                a_trial = a + t * direction
            a_trial = np.maximum(a_trial, 0.0)
            s = a_trial.sum()
            if s < 1e-10:
                t *= 0.5
                continue
            a_trial = a_trial / s
            J_trial = compute_J(a_trial)
            if J_trial < best_trial_J - 1e-14:
                best_trial_J = J_trial
                best_trial_a = a_trial.copy()
                break

                best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1
            momentum_dir *= 0.5
            if no_improve > 50:
                break

def fourier_to_seq(params, n, n_harm):
    """Convert Fourier parameters to non-negative sequence (normalized)."""
    t = np.arange(n, dtype=np.float64) / n
    a = np.full(n, params[0], dtype=np.float64)
    for k in range(1, n_harm + 1):
        a += params[2 * k - 1] * np.cos(2.0 * np.pi * k * t)
        a += params[2 * k] * np.sin(2.0 * np.pi * k * t)
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-8:
        a = np.ones(n) / n
    else:
        a = a / s
    return a


def seq_to_fourier(a, n_harm):
    """Extract Fourier parameters from a sequence."""
    n = len(a)
    a_norm = a / a.sum()
    t = np.arange(n, dtype=np.float64) / n
    params = np.zeros(2 * n_harm + 1)
    params[0] = np.mean(a_norm)
    for k in range(1, n_harm + 1):
        params[2 * k - 1] = 2.0 * np.mean(a_norm * np.cos(2.0 * np.pi * k * t))
        params[2 * k] = 2.0 * np.mean(a_norm * np.sin(2.0 * np.pi * k * t))
    return params


def run_cmaes_fourier(n, n_harm, params0, sigma0, max_gens, deadline, tracker):

    dim = 2 * n_harm + 1
    lam = 4 + int(3 * np.log(max(dim, 2)))
    mu = lam // 2
    
    weights = np.log(mu + 0.5) - np.log(np.arange(1, mu + 1))
    weights = weights / weights.sum()
    mu_eff = 1.0 / np.sum(weights ** 2)
    c_sigma = (mu_eff + 2) / (dim + mu_eff + 5)
    c_c = (4 + mu_eff / dim) / (dim + 4 + 2 * mu_eff / dim)
    d_sigma = 1 + 2 * max(0, np.sqrt((mu_eff - 1) / (dim + 1) - 1) - 1) + c_sigma
    c_1 = 2 / ((dim + 1.3) ** 2 + mu_eff)
    c_mu = min(1 - c_1, 2 * (mu_eff - 2 + 1.0 / mu_eff) / ((dim + 2) ** 2 + mu_eff))
    chi_N = np.sqrt(dim) * (1 - 1 / (4 * dim) + 1 / (21 * dim ** 2))
    
    C = np.eye(dim)
    Cs = np.zeros((dim, dim))
    p_sigma = np.zeros(dim)
    p_c = np.zeros(dim)
    x = params0.copy()
    sigma = sigma0
    
    best_J = compute_J(fourier_to_seq(params0, n, n_harm))
    best_x = params0.copy()
    tracker.update(fourier_to_seq(params0, n, n_harm))
    
    for gen in range(max_gens):
        if time.monotonic() > deadline:
            break
        
        offspring = []
        offspring_J = []
        for i in range(lam):
            z = np.random.standard_normal(dim)
            eigvals, eigvecs = np.linalg.eigh(C)
            eigvals = np.maximum(eigvals, 1e-12)
            y = (eigvecs * np.sqrt(eigvals)) @ (eigvecs.T @ z)
            x_trial = x + sigma * y
            a_trial = fourier_to_seq(x_trial, n, n_harm)
            J_trial = compute_J(a_trial)
            offspring.append(x_trial)
            offspring_J.append(J_trial)
        
        idx = np.argsort(offspring_J)
        if offspring_J[idx[0]] < best_J:
            best_J = offspring_J[idx[0]]
            best_x = offspring[idx[0]].copy()
            tracker.update(fourier_to_seq(best_x, n, n_harm))
        
        sorted_x = [offspring[i] for i in idx[:mu]]
        x_new = np.zeros(dim)
        for i in range(mu):
            x_new += weights[i] * sorted_x[i]
        
        diff = (x_new - x) / (sigma * chi_N) if sigma > 1e-30 else np.zeros(dim)
        p_sigma = (1 - c_sigma) * p_sigma + np.sqrt(c_sigma * (2 - c_sigma) * mu_eff) * diff
        sigma *= np.exp(c_sigma / d_sigma * (np.linalg.norm(p_sigma) / chi_N - 1))
        sigma = np.clip(sigma, 1e-12, 1.0)
        
        h_sigma = 1.0
        if np.linalg.norm(p_sigma) / chi_N > (1.4 + c_c) * np.sqrt((1 + c_sigma) / (1 - c_sigma)):
            h_sigma = 0.0
        p_c = (1 - c_c) * p_c + h_sigma * np.sqrt(c_c * (2 - c_c) * mu_eff) * (x_new - x) / max(sigma, 1e-30)

        rank_mu = np.zeros((dim, dim))

        C = (1 - c_1 - c_mu) * C + c_1 * (rank_1 + h_sigma * (1 - h_sigma) * np.outer(p_c, p_c) * c_c) + c_mu * rank_mu

def structured_perturb(a, rng, perturb_type):
    """Apply a structured perturbation."""
    n = len(a)
    t = np.arange(n, dtype=np.float64) / n
    
    if perturb_type == 'shift':
        s = rng.integers(1, max(2, n // 10))
        a_p = np.roll(a, s)
    elif perturb_type == 'cosine':
        freq = rng.uniform(1, 10)
        amp = rng.uniform(0.01, 0.15)
        phase = rng.uniform(0, 2 * np.pi)
        a_p = a * (1 + amp * np.cos(2 * np.pi * freq * t + phase))
    elif perturb_type == 'smooth':
        k = rng.integers(3, max(4, n // 8))
        kernel = np.ones(k) / k
        a_p = np.convolve(a, kernel, mode='same')
    elif perturb_type == 'taper':
        alpha = rng.uniform(0.4, 0.9)
        edge = max(1, int(n * (1 - alpha) / 2))
        window = np.ones(n)
        for i in range(edge):
            frac = (i + 1) / (edge + 1)
            window[i] = frac
            window[n - 1 - i] = frac
        a_p = a * window
    elif perturb_type == 'antiphase':
        freq = rng.choice([2, 3, 4, 5, 6])
        amp = rng.uniform(0.05, 0.2)
        a_p = a * (1 + amp * np.cos(2 * np.pi * freq * t + np.pi))
    else:
        a_p = a + rng.normal(0, 0.03, n)
    
    a_p = np.maximum(a_p, 0.001)
    s = a_p.sum()
    if s > 1e-8:
        a_p = a_p / s
    return a_p


def resample_seq(src, dest_n):
    """Resample 1D sequence to new length."""
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)


def lbb_objective(x):
    """Objective for L-BFGS-B: max of self-convolution (scale invariant)."""
    ax = np.maximum(x, 0.0)
    s = ax.sum()
    if s < 1e-8:
        return 1e10
    ax = ax / s
    b = fast_conv(ax)
    return float(b.max())

        current_best_J = J2
    
    # === PHASE 3: LP multiplicative (150s) ===
    if time.monotonic() < deadline - 550:
        t0 = time.monotonic()
        print(f"\n{'='*50}", flush=True)
        print(f"Phase 3: LP multiplicative (n={current_best_n})", flush=True)
        print(f"{'='*50}", flush=True)
        
        a3 = current_best.copy()
        a3 = np.maximum(a3, 1e-10)
        a3 = a3 / a3.sum()
        dl3 = min(deadline - 500, t0 + 150)
        a3, J3 = lp_mirror_descent_v2(a3, 3000, dl3, use_multiplicative=True, momentum_coef=0.0)
        tracker.update(a3)
        print(f"  J={J3:.8f} ({time.monotonic()-t0:.1f}s)", flush=True)
        
        if J3 < current_best_J:
            current_best = a3.copy()
            current_best_n = current_best.size
            current_best_J = J3
    
    # === PHASE 4: CMA-ES on Fourier (200s) ===
    if time.monotonic() < deadline - 400:
        t0 = time.monotonic()
        print(f"\n{'='*50}", flush=True)
        print(f"Phase 4: CMA-ES Fourier (n={current_best_n})", flush=True)
        print(f"{'='*50}", flush=True)
        
        n_cma = current_best_n
        n_harm = min(20, n_cma // 6)
        params0 = seq_to_fourier(current_best, n_harm)
        dl4 = min(deadline - 350, t0 + 150)
        best_params, J_cma = run_cmaes_fourier(n_cma, n_harm, params0, 0.03, 200, dl4, tracker)
        
        a_cma = fourier_to_seq(best_params, n_cma, n_harm)
        J_cma = compute_J(a_cma)
        tracker.update(a_cma)
        print(f"  J={J_cma:.8f} ({time.monotonic()-t0:.1f}s)", flush=True)
        
        if J_cma < current_best_J:
            current_best = a_cma.copy()
            current_best_n = n_cma
            current_best_J = J_cma
        
        # LP from CMA result
        if time.monotonic() < deadline - 300:
            t0b = time.monotonic()
            a_lp = current_best.copy()
            a_lp = np.maximum(a_lp, 1e-10)
            a_lp = a_lp / a_lp.sum()
            dl_lp = min(deadline - 250, t0b + 100)
            a_lp, J_lp = lp_mirror_descent_v2(a_lp, 2000, dl_lp, use_multiplicative=False, momentum_coef=0.2)
            tracker.update(a_lp)
            print(f"  LP from CMA: J={J_lp:.8f} ({time.monotonic()-t0b:.1f}s)", flush=True)
            if J_lp < current_best_J:
                current_best = a_lp.copy()
                current_best_n = a_lp.size
                current_best_J = J_lp
    
    # === PHASE 5: Multi-length exploration (150s) ===
    if time.monotonic() < deadline - 250:
        t0 = time.monotonic()
        print(f"\n{'='*50}", flush=True)
        print(f"Phase 5: Multi-length (base n={current_best_n})", flush=True)
        print(f"{'='*50}", flush=True)
        
        base_n = current_best_n
        candidates = [base_n - 64, base_n + 32, base_n + 64, base_n + 128]
        candidates = [n for n in candidates if 128 <= n <= 2048 and n != base_n]
        # Add some very different lengths
        for extra in [384, 512, 768, 896, 1024]:
            if extra != base_n and 128 <= extra <= 2048:
                candidates.append(extra)
        candidates = sorted(set(candidates))[:5]
        
        per_n_time = max(15, min(40, (deadline - 200 - time.monotonic()) / max(1, len(candidates))))
        
        for n_try in candidates:
            if time.monotonic() > deadline - 180:
                break
            a_try = resample_seq(current_best, n_try)
            a_try = np.maximum(a_try, 1e-10)
            a_try = a_try / a_try.sum()
            # Quick subgradient first
            a_try, _ = fft_subgradient_polish(a_try, max_iters=200, deadline=time.monotonic() + per_n_time * 0.3, lr_scale=1.5)
            # Then LP
            a_try, J_try = lp_mirror_descent_v2(a_try, 1000, time.monotonic() + per_n_time, use_multiplicative=False, momentum_coef=0.3)
            tracker.update(a_try)
            print(f"  n={n_try}: J={J_try:.8f}", flush=True)
            
            if J_try < current_best_J:
                current_best = a_try.copy()
                current_best_n = n_try
                current_best_J = J_try
        
        print(f"Phase 5 done ({time.monotonic()-t0:.1f}s)", flush=True)
    
    # === PHASE 6: Structured perturbations + LP (150s) ===
    if time.monotonic() < deadline - 150:
        t0 = time.monotonic()
        print(f"\n{'='*50}", flush=True)
        print(f"Phase 6: Perturb + LP (n={current_best_n})", flush=True)
        print(f"{'='*50}", flush=True)
        
        perturb_types = ['shift', 'cosine', 'smooth', 'taper', 'antiphase', 'shift', 'cosine', 'smooth']
        n_perts = len(perturb_types)
        per_pert_time = max(10, min(30, (deadline - 120 - time.monotonic()) / n_perts))
        
        for i, ptype in enumerate(perturb_types):
            if time.monotonic() > deadline - 110:
                break
            a_pert = structured_perturb(current_best, rng, ptype)
            a_pert, J_pert = lp_mirror_descent_v2(a_pert, 500, time.monotonic() + per_pert_time,
                                                   use_multiplicative=False, momentum_coef=0.2)
            tracker.update(a_pert)
            if J_pert < current_best_J:
                current_best = a_pert.copy()
                current_best_n = a_pert.size
                current_best_J = J_pert
                print(f"  [{ptype}] J={J_pert:.8f}", flush=True)
        
        print(f"Phase 6 done ({time.monotonic()-t0:.1f}s)", flush=True)
    
    # === PHASE 7: Long LP on best (200s) ===
    if time.monotonic() < deadline - 80:
        t0 = time.monotonic()
        print(f"\n{'='*50}", flush=True)
        print(f"Phase 7: Long LP (n={current_best_n})", flush=True)
        print(f"{'='*50}", flush=True)
        
        a_long = current_best.copy()
        a_long = np.maximum(a_long, 1e-10)
        a_long = a_long / a_long.sum()
        dl7 = deadline - 50
        a_long, J_long = lp_mirror_descent_v2(a_long, 10000, dl7, use_multiplicative=False, momentum_coef=0.4)
        tracker.update(a_long)
        print(f"  J={J_long:.8f} ({time.monotonic()-t0:.1f}s)", flush=True)
        
        if J_long < current_best_J:
            current_best = a_long.copy()
            current_best_n = a_long.size
            current_best_J = J_long
        
        # Quick FFT subgradient polish on result
        if time.monotonic() < deadline - 30:
            a_sf = current_best.copy()
            a_sf, M_sf = fft_subgradient_polish(a_sf, max_iters=5000, deadline=deadline - 10, lr_scale=1.0)
            J_sf = compute_J(a_sf)
            tracker.update(a_sf)
            if J_sf < current_best_J:
                current_best = a_sf.copy()
                current_best_J = J_sf
            print(f"  FFT polish: J={J_sf:.8f}", flush=True)
    
    # === PHASE 8: L-BFGS-B final polish ===
    if time.monotonic() < deadline - 5 and tracker.best is not None:
        t0 = time.monotonic()
        try:
            res = minimize(lbb_objective, tracker.best.astype(np.float64),
                           method='L-BFGS-B', bounds=[(0.0, None)] * tracker.best.size,
                           options={'maxiter': 200, 'max_fun': 50000, 'ftol': 1e-16})
            if res.fun < 1e10:
                tracker.update(res.x)
        except Exception as e:
            print(f"  L-BFGS-B error: {e}", flush=True)
        print(f"  L-BFGS-B done ({time.monotonic()-t0:.1f}s)", flush=True)
    
    # Final
    if tracker.best is not None:
        final_a = tracker.best
    else:
        final_a = current_best
    
    print(f"\n{'='*50}", flush=True)
    print(f"FINAL: J={tracker.best_score:.8f} n={final_a.size} time={time.monotonic()-t_start:.0f}s", flush=True)
    print(f"{'='*50}", flush=True)
    
    return [float(x) for x in final_a]
