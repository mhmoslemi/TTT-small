import numpy as np
from scipy.optimize import linprog
from scipy.sparse import lil_matrix
import time


def fast_conv(a):
    """Full self-convolution via FFT."""
    n = len(a)
    s = 1
    while s < 2 * n - 1:
        s <<= 1
    fa = np.fft.rfft(a, s)
    return np.fft.irfft(fa * fa, s)[:2 * n - 1]


def compute_J(a):
    """Compute J(a) = 2*n*max(conv(a,a))/(sum(a))^2."""
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))


def build_A_sparse_active(a, active_indices):

    n = len(a)

    cols = []

            if 0 <= ki < n:

def cd_find_optimal(a, i, b, S):
    """Find optimal value for coordinate i given current b and S."""
    n = len(a)
    ai = a[i]
    
    # The affected convolution indices
    k_lo = i
    k_hi = i + n - 1
    k_lo = max(0, k_lo)
    k_hi = min(2 * n - 2, k_hi)
    
    # Precompute the "unaffected max"
    unaffected = np.delete(b, range(k_lo, k_hi + 1))
    if unaffected.size > 0:
        max_unaffected = float(unaffected.max())
    else:
        max_unaffected = 0.0
    
    # Precompute the coefficients for affected b[k]
    # b[k] = b[k]_base + 2*new*a[k-i] - 2*old*a[k-i] for k != 2*i
    # b[2*i] = b[2*i]_base + new^2 - old^2
    coeffs = np.zeros(k_hi - k_lo + 1)
    for idx, k in enumerate(range(k_lo, k_hi + 1)):
        ki = k - i
        if 0 <= ki < n:
            if k == 2 * i:
                coeffs[idx] = -1.0  # will handle x^2 term separately
            else:
                coeffs[idx] = 2.0 * a[ki]
    
    # Base values of affected b
    base = b[k_lo:k_hi + 1].copy()
    
    def obj(x):
        new_S = S - ai + x
        if new_S < 1e-10:
            return 1e10
        # Compute new b values for affected indices
        new_b_affected = base.copy()
        for idx in range(len(new_b_affected)):
            k = k_lo + idx
            ki = k - i
            if 0 <= ki < n:
                if k == 2 * i:
                    new_b_affected[idx] = base[idx] - ai * ai + x * x
                else:
                    new_b_affected[idx] = base[idx] - 2.0 * ai * a[ki] + 2.0 * x * a[ki]
        max_c = max(max_unaffected, float(new_b_affected.max()))
        if max_c < 0:
            max_c = 0.0
        return 2.0 * n * max_c / (new_S * new_S)
    
    # Golden section search
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    lo = 0.0
    hi = max(3.0 * ai, 3.0 * S / n, 0.5)
    
    c_pt = hi - gr * (hi - lo)
    d_pt = lo + gr * (hi - lo)
    fc = obj(c_pt)
    fd = obj(d_pt)
    
    for _ in range(30):
        if fc < fd:
            hi = d_pt
            d_pt = c_pt
            fd = fc
            c_pt = hi - gr * (hi - lo)
            fc = obj(c_pt)
        else:
            lo = c_pt
            c_pt = d_pt
            fc = fd
            d_pt = lo + gr * (hi - lo)
            fd = obj(d_pt)
    
    x_opt = (lo + hi) / 2.0
    return max(0.0, x_opt)


def cd_pass(a, n_passes_max=3, deadline=None, tracker=None, verbose=False):
    """Multiple passes of coordinate descent with incremental updates."""
    a = np.maximum(a.astype(np.float64), 0.0)
    S = a.sum()
    if S < 1e-10:
        return a
    a = a / S
    S = a.sum()
    n = len(a)
    b = fast_conv(a)
    current_J = 2.0 * n * b.max() / (S * S)
    if tracker:
        tracker.update(a)
    
    for p in range(n_passes_max):
        if deadline and time.monotonic() > deadline:
            break
        t0 = time.monotonic()
        
        indices = np.arange(n)
        np.random.shuffle(indices)
        
        improved = 0
        for i in indices:
            new_val = cd_find_optimal(a, i, b, S)
            if abs(new_val - a[i]) > 1e-12:
                old_val = a[i]
                a[i] = new_val
                S = S - old_val + new_val
                # Update convolution incrementally
                k_lo = max(0, i)
                k_hi = min(2 * n - 1, i + n) - 1
                for k in range(k_lo, k_hi + 1):
                    ki = k - i
                    if 0 <= ki < n:
                        if k == 2 * i:
                            b[k] = b[k] - old_val * old_val + new_val * new_val
                        else:
                            b[k] = b[k] - 2.0 * old_val * a[ki] + 2.0 * new_val * a[ki]
                improved += 1
        
        # Verify and compute new J
        b_check = fast_conv(a)
        new_J = 2.0 * n * b_check.max() / (S * S)
        b = b_check  # Use verified convolution
        
        if verbose:
            print(f"    CD pass {p}: J={new_J:.8f}, improved={improved}/{n} ({time.monotonic()-t0:.2f}s)", flush=True)
        
        if tracker and new_J < current_J - 1e-12:
            tracker.update(a)
            current_J = new_J
        
        if improved < n * 0.01:
            break
    
    if tracker:
        tracker.update(a)
    return a


def lp_mirror_descent(a, n_iters, deadline, tracker, k_top=30, tighten=0.005):

        M = float(b.max())
        M_target = M * (1.0 - tighten)
        
        g = lp_active_set(a, M_target, k_top)
        if g is None:
            M_target = M * (1.0 - tighten * 0.5)
            g = lp_active_set(a, M_target, k_top)
        if g is None:
            tighten = max(0.001, tighten * 0.7)
            stall += 1
            if stall > 30:
                break
            continue
        
        g = np.maximum(g, 0.0)
        g_sum = g.sum()
        if g_sum < 1e-10:
            break
        g = g / g_sum
        
        # Line search on step size
        best_a = None
        best_J = current_J
        base_t = 0.3
        for t in [base_t, base_t * 0.5, base_t * 1.5, base_t * 0.7, 0.8, 0.5]:
            a_trial = (1.0 - t) * a + t * g
            a_trial = np.maximum(a_trial, 0.0)
            s = a_trial.sum()
            if s < 1e-10:
                continue
            a_trial = a_trial / s
            J_trial = compute_J(a_trial)
            if J_trial < best_J - 1e-14:
                best_J = J_trial
                best_a = a_trial
        
        if best_a is None:
            stall += 1
            tighten = max(0.001, tighten * 0.8)
            if stall > 40:
                break
            continue
        
        a = best_a
        current_J = best_J
        stall = 0
        
        # Adapt tighten
        if it % 20 == 0 and it > 0:
            tighten = min(0.02, tighten * 1.1)
        if it % 50 == 0 and it > 0:
            tighten = max(0.002, tighten * 0.9)
        
        if it % 20 == 0:
            tracker.update(a)
    
    tracker.update(a)
    return a


def resample_seq(src, dest_n):
    """Resample 1D sequence to new length via linear interpolation."""
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)


class ScoreTracker:
    def __init__(self):
        self.best = None
        self.best_score = float('inf')
        self.t_start = time.monotonic()
    
    def update(self, a):
        a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
        if a.size == 0 or a.sum() < 0.01:
            return
        score = float(evaluate_sequence(a.tolist()))
        if np.isfinite(score) and score < self.best_score:
            self.best = a.copy()
            self.best_score = score
            elapsed = time.monotonic() - self.t_start
            print(f"  * NEW BEST: J={score:.8f} n={a.size} t={elapsed:.0f}s", flush=True)
    
    def report(self, label=""):
        if self.best is not None:
            print(f"  {label} best: J={self.best_score:.8f} n={self.best.size}", flush=True)


def generate_inits(n, rng):
    """Generate diverse initial sequences of length n."""
    inits = []
    t = np.linspace(0, 1, n, endpoint=False)
    
    inits.append(np.ones(n))
    inits.append(rng.uniform(0.3, 1.0, n))
    
    for freq in [2, 3, 5, 7]:
        inits.append(1.0 + 0.3 * np.cos(2.0 * np.pi * freq * t + rng.uniform(0, 6.28)))
    
    inits.append(1.0 + 0.2 * np.cos(2*np.pi*2*t) + 0.2 * np.cos(2*np.pi*5*t))
    inits.append(np.abs(np.sin(np.pi * t)) + 0.2)
    
    # Tapered
    taper = 1.0 - 0.3 * (2*t - 1)**2
    inits.append(taper)
    
    # Two peaks
    inits.append(0.5 * np.exp(-8*(t-0.3)**2) + 0.5 * np.exp(-8*(t-0.7)**2) + 0.1)
    
    return inits


def propose_candidate(seed=42, budget_s=1500, **kwargs):
    """
    Main search function combining active-set LP mirror descent,
    fast coordinate descent, and multi-length exploration.
    """

    deadline = time.monotonic() + float(budget_s) - 15.0
    tracker = ScoreTracker()
    
    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        if p.ndim == 1 and p.size > 0 and np.all(np.isfinite(p)):
            parent = np.maximum(p, 0.0)
            tracker.update(parent)
            print(f"Parent loaded: n={parent.size}, J={tracker.best_score:.8f}", flush=True)
    except (NameError, AttributeError):
        pass
    
    if parent is None:
        parent = np.ones(384)
        tracker.update(parent)
        print(f"No parent found, using uniform n={parent.size}", flush=True)
    
    n_parent = parent.size
    
    # === Phase 1: Intensive LP on parent ===
    print(f"\n{'='*60}", flush=True)
    print(f"Phase 1: Active-set LP on parent (n={n_parent})", flush=True)
    print(f"{'='*60}", flush=True)
    t0 = time.monotonic()
    
    a = parent.copy()
    a = a / a.sum()
    phase1_deadline = min(deadline - 300, t0 + 400)
    a = lp_mirror_descent(a, n_iters=500, deadline=phase1_deadline, tracker=tracker,
                          k_top=40, tighten=0.005)
    tracker.report(f"Phase1 LP [{time.monotonic()-t0:.1f}s]")
    
    # === Phase 2: Coordinate descent polish ===
    if time.monotonic() < deadline - 250:
        print(f"\n{'='*60}", flush=True)
        print(f"Phase 2: Coordinate Descent", flush=True)
        print(f"{'='*60}", flush=True)
        t0 = time.monotonic()
        cd_deadline = min(deadline - 250, t0 + 200)
        a = cd_pass(a, n_passes_max=5, deadline=cd_deadline, tracker=tracker, verbose=True)
        a = a / a.sum()
        tracker.report(f"Phase2 CD [{time.monotonic()-t0:.1f}s]")
    
    # === Phase 3: Alternating LP + CD ===
    if time.monotonic() < deadline - 150:
        print(f"\n{'='*60}", flush=True)
        print(f"Phase 3: Alternating LP + CD", flush=True)
        print(f"{'='*60}", flush=True)
        t0 = time.monotonic()
        
        alt_deadline = deadline - 150
        for alt_it in range(20):
            if time.monotonic() > alt_deadline:
                break
            
            # LP phase
            lp_budget = min(30, (alt_deadline - time.monotonic()) * 0.6)
            if lp_budget > 5:
                a = lp_mirror_descent(a, n_iters=50, deadline=time.monotonic() + lp_budget,
                                       tracker=tracker, k_top=30, tighten=0.003)
            
            if time.monotonic() > alt_deadline:
                break
            
            # CD phase
            cd_budget = alt_deadline - time.monotonic()
            if cd_budget > 10:
                n_cd = max(1, min(3, int(cd_budget / (a.size * 0.02))))
                a = cd_pass(a, n_passes_max=n_cd, deadline=alt_deadline, tracker=tracker)
                a = a / a.sum()
        
        tracker.report(f"Phase3 Alt [{time.monotonic()-t0:.1f}s]")
    
    # === Phase 4: Length sweep ===
    if time.monotonic() < deadline - 80:
        print(f"\n{'='*60}", flush=True)
        print(f"Phase 4: Length Sweep", flush=True)
        print(f"{'='*60}", flush=True)
        t0 = time.monotonic()
        
        # Generate candidate lengths
        candidate_lengths = set()
        for delta in range(-20, 21, 2):
            l = n_parent + delta
            if 100 <= l <= 800:
                candidate_lengths.add(l)
        # Also try some "round" lengths
        for l in [256, 300, 320, 350, 384, 400, 420, 450, 500, 512, 600]:
            candidate_lengths.add(l)
        
        candidate_lengths = sorted(candidate_lengths)
        
        # Try each length with quick optimization
        length_deadline = deadline - 80
        per_length_time = max(3.0, (length_deadline - time.monotonic()) / len(candidate_lengths))
        
        trial_results = []
        for nl in candidate_lengths:
            if time.monotonic() > length_deadline:
                break
            
            t_start = time.monotonic()
            a_res = resample_seq(tracker.best, nl)
            a_res = np.maximum(a_res, 0.001)
            a_res = a_res / a_res.sum()
            
            # Quick LP
            a_res = lp_mirror_descent(a_res, n_iters=30, deadline=t_start + per_length_time * 0.6,
                                       tracker=tracker, k_top=25, tighten=0.004)
            
            # Quick CD
            if time.monotonic() < t_start + per_length_time:
                a_res = cd_pass(a_res, n_passes_max=2, deadline=t_start + per_length_time,
                                tracker=tracker)
                a_res = a_res / a_res.sum()
                tracker.update(a_res)
            
            elapsed = time.monotonic() - t_start
            print(f"    n={nl}: J={tracker.best_score:.8f} ({elapsed:.1f}s)", flush=True)
        
        tracker.report(f"Phase4 Lengths [{time.monotonic()-t0:.1f}s]")
    
    # === Phase 5: Perturbation restarts on best ===
    if time.monotonic() < deadline - 40:
        print(f"\n{'='*60}", flush=True)
        print(f"Phase 5: Perturbation Restarts", flush=True)
        print(f"{'='*60}", flush=True)
        t0 = time.monotonic()
        
        best = tracker.best.copy()
        n = best.size
        rest_deadline = deadline - 40
        
        for temp in [0.05, 0.03, 0.02, 0.01, 0.015]:
            if time.monotonic() > rest_deadline:
                break
            
            a_pert = best.copy() + rng.normal(0, temp, n)
            a_pert = np.maximum(a_pert, 0.001)
            a_pert = a_pert / a_pert.sum()
            
            # Short LP
            t_pert = time.monotonic()
            budget_here = min(30, rest_deadline - time.monotonic())
            if budget_here > 5:
                a_pert = lp_mirror_descent(a_pert, n_iters=30, deadline=t_pert + budget_here * 0.6,
                                            tracker=tracker, k_top=25, tighten=0.003)
            
            # Short CD
            if time.monotonic() < rest_deadline:
                a_pert = cd_pass(a_pert, n_passes_max=2, deadline=rest_deadline, tracker=tracker)
                a_pert = a_pert / a_pert.sum()
                tracker.update(a_pert)
        
        tracker.report(f"Phase5 Restarts [{time.monotonic()-t0:.1f}s]")
    
    # === Phase 6: Final intensive polish ===
    if time.monotonic() < deadline - 10:
        print(f"\n{'='*60}", flush=True)
        print(f"Phase 6: Final Polish", flush=True)
        print(f"{'='*60}", flush=True)
        t0 = time.monotonic()
        
        a_final = tracker.best.copy()
        a_final = a_final / a_final.sum()
        
        # Final LP with tighter constraints
        a_final = lp_mirror_descent(a_final, n_iters=100, deadline=deadline - 10,
                                     tracker=tracker, k_top=50, tighten=0.002)
        
        # Final CD
        if time.monotonic() < deadline - 5:
            a_final = cd_pass(a_final, n_passes_max=3, deadline=deadline - 5, tracker=tracker)
            a_final = a_final / a_final.sum()
            tracker.update(a_final)
        
        tracker.report(f"Phase6 Final [{time.monotonic()-t0:.1f}s]")
    
    # Fallback
    if tracker.best is None:
        tracker.best = np.ones(256)
        tracker.best_score = compute_J(tracker.best)
    
    print(f"\n{'='*60}", flush=True)
    print(f"FINAL RESULT", flush=True)
    print(f"J = {tracker.best_score:.10f}, n = {tracker.best.size}", flush=True)
    print(f"Total time: {time.monotonic()-tracker.t_start:.1f}s", flush=True)
    print(f"{'='*60}", flush=True)
    
    return [float(x) for x in tracker.best]
