    n = len(a0)
    a = a0.copy()
    S = np.sum(a)
    if S < 1e-10:
        return a
    a = a / S
    z = a.copy()
    k = 1
    t_k = 1.0
    best_a = a.copy()
    best_obj = true_obj(a)
    lr = lr_init
    start = time.monotonic()
    
    for it in range(max_iters):
        if time.monotonic() - start > time_budget:
            break
        
        # Gradient at z
        _, grad = smooth_obj_and_grad(z, t)
        
        # Backtracking line search
        # Try: z - lr * grad, clip to non-negative
        while True:
            a_new = z - lr * grad
            a_new = np.maximum(a_new, 0)
            S_new = np.sum(a_new)
            if S_new < 1e-10:
                lr *= 0.5
                continue
            a_new = a_new / S_new
            obj_new = smooth_obj(a_new, t)[0]
            obj_z = smooth_obj(z, t)[0]
            # Check sufficient decrease
            if obj_new <= obj_z - 0.5 * lr * np.sum(grad * (a_new - z)):
                break
            lr *= 0.5
            if lr < 1e-15:
                break
        
        # FISTA update
        t_new = 0.5 * (1 + np.sqrt(1 + 4 * t_k * t_k))
        z_new = a_new + ((t_k - 1) / t_new) * (a_new - a)
        z_new = np.maximum(z_new, 0)

        if S_z > 1e-10:

        a = a_new
        z = z_new
        t_k = t_new
        k += 1
        
        # Track best

        if obj_true < best_obj:

"""
Optimized step-function coefficient search minimizing 2*n*max(conv(a,a))/(sum(a))^2.
Uses softmax-parameterized L-BFGS with log-sum-exp smoothing, temperature annealing,
FFT-based gradient computation, LP refinement, and multi-start diversification.
Key innovation: smooth gradient via softmax (replacing subgradient), enabling quasi-Newton
convergence on the simplex via exponential reparameterization.

import numpy as np
from scipy.optimize import minimize, linprog
from scipy.sparse import coo_matrix
import time


def full_conv(a):
    """Full linear convolution of a with itself via FFT."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    conv = np.fft.irfft(fa * fa, size)
    return conv[:2 * n - 1]


def true_obj(a):
    """Compute true objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    S = float(np.sum(a))
    if S < 1e-10:
        return np.inf
    conv = full_conv(a)
    M = float(np.max(conv))
    return 2.0 * n * M / (S * S)


def smooth_max_grad(a, t):
    """
    Compute smoothed max of conv(a,a) via log-sum-exp and its gradient w.r.t. a.
    Returns (smoothed_value, gradient_array).
    """
    n = len(a)
    conv = full_conv(a)
    c_max = np.max(conv)
    exp_t = np.exp(t * (conv - c_max))
    sum_exp = np.sum(exp_t)
    log_se = c_max + np.log(sum_exp) / t
    p = exp_t / sum_exp

    # Gradient: grad[i] = 2 * sum_j a[j] * p[i+j]
    p_rev = p[::-1]
    conv_ap = np.convolve(a, p_rev)
    h = conv_ap[n - 1:2 * n - 1][::-1]
    grad = 2.0 * h
    return log_se, grad


def softmax_obj_grad(b, t):
    """
    Objective and gradient in b-space (a = softmax(b)).
    Objective: 2n * smoothed_max(conv(a,a))
    Gradient: 2n * a * (grad_a - dot(grad_a, a))
    """
    n = len(b)
    # Stable softmax
    b_shift = b - np.max(b)
    exp_b = np.exp(b_shift)
    a = exp_b / np.sum(exp_b)

    log_se, grad_a = smooth_max_grad(a, t)
    obj = 2.0 * n * log_se

    # Gradient in b-space
    dot_ga_a = np.dot(grad_a, a)
    grad_b = 2.0 * n * a * (grad_a - dot_ga_a)

    return obj, grad_b


def fista_simplex(a0, t, time_budget, lr=None, max_iters=5000):

    FISTA (accelerated projected gradient) on the simplex for the smoothed objective.

    best_a = a.copy()
    best_obj = true_obj(a)
    start = time.monotonic()

    if lr is None:
        # Estimate step size
        lr = 0.5 / (t * np.max(a) + 1e-6)

    for it in range(max_iters):
        if time.monotonic() - start > time_budget:
            break

        _, grad_z = smooth_max_grad(z, t)
        # Project gradient onto tangent space of simplex
        grad_proj = grad_z - np.mean(grad_z)

        # Backtracking for step size
        obj_z = 2.0 * n * smooth_max_grad(z, t)[0]
        a_new = z - lr * grad_proj
        a_new = np.maximum(a_new, 0)
        S_new = np.sum(a_new)
        if S_new < 1e-10:
            lr *= 0.5
            continue
        a_new = a_new / S_new
        obj_new = 2.0 * n * smooth_max_grad(a_new, t)[0]

            lr *= 0.8
            a_new = z - lr * grad_proj
            a_new = np.maximum(a_new, 0)
            S_new = np.sum(a_new)
            if S_new < 1e-10:

            a_new = a_new / S_new

        z_new = np.maximum(z_new, 0)
        S_z = np.sum(z_new)

        else:
            z_new = a_new.copy()

        a = a_new
        z = z_new
        t_k = t_new

        obj_true = true_obj(a)

def lbfgs_softmax(a0, t, time_budget, max_iter=300):
    """
    L-BFGS optimization in b-space (a = softmax(b)) for the smoothed objective.
    """
    n = len(a0)
    S = np.sum(a0)
    a0 = a0 / S
    # Initialize b from a0
    b0 = np.log(a0 + 1e-30)

    start = time.monotonic()

    def obj_func(b):
        if time.monotonic() - start > time_budget:
            return 1e10, np.zeros(len(b))
        val, grad = softmax_obj_grad(b, t)
        return val, grad

    try:
        res = minimize(obj_func, b0, jac=True, method='L-BFGS-B',
                       options={'maxiter': max_iter, 'ftol': 1e-14, 'gtol': 1e-10,
                                'maxfun': max_iter * 3})
        b_opt = res.x
        b_opt_shift = b_opt - np.max(b_opt)
        a_opt = np.exp(b_opt_shift)
        a_opt = a_opt / np.sum(a_opt)
        return a_opt, true_obj(a_opt)
    except Exception:
        return a0, true_obj(a0)


def build_toeplitz_sparse(a):
    """Build sparse matrix T such that (T @ g) = conv(a, g)."""
    n = len(a)
    rows = []
    cols = []
    vals = []
    for j in range(n):
        start = j
        end = min(j + n, 2 * n - 1)
        length = end - start
        rows.extend(range(start, end))
        cols.extend([j] * length)
        vals.extend(list(a[:length]))
    A = coo_matrix((vals, (rows, cols)), shape=(2 * n - 1, n)).tocsr()
    return A


def solve_lp(a, M, time_limit=5):
    """Solve LP: max sum(g) s.t. conv(a,g) <= M, g >= 0."""
    n = len(a)
    A = build_toeplitz_sparse(a)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': time_limit})
        if res.status == 0:
            return np.maximum(res.x, 0)
    except Exception:
        pass
    return None

    n = len(a)

            a_try = np.maximum(a_try, 0)

        else:

            a = a * np.exp(rng.normal(0, 0.05, n))

            a = a / np.sum(a)

def diverse_init(n, seed):
    """Generate diverse initializations."""
    rng = np.random.default_rng(seed)
    inits = []

    # Uniform
    inits.append(np.ones(n) / n)

    # Cosine series
    for trial in range(6):
        k = np.arange(n)
        a = np.ones(n)
        n_modes = rng.integers(3, 12)
        for m in range(1, n_modes + 1):
            c = rng.uniform(-0.5, 0.5) / m
            phase = rng.uniform(0, 2 * np.pi)
            a += c * np.cos(2 * np.pi * m * k / n + phase)
        a = np.maximum(a, 1e-6)
        inits.append(a / np.sum(a))

    # Gaussian bumps at random locations
    for trial in range(4):
        a = np.zeros(n)
        n_bumps = rng.integers(2, 6)
        for _ in range(n_bumps):
            center = rng.uniform(0, n)
            sigma = rng.uniform(n * 0.05, n * 0.3)
            a += np.exp(-0.5 * ((np.arange(n) - center) / sigma) ** 2)
        a = np.maximum(a, 1e-6)
        inits.append(a / np.sum(a))

    # Exponential decay from random start
    for trial in range(3):
        start_pos = rng.integers(0, n)
        rate = rng.uniform(0.5, 3.0) / n
        k = np.arange(n)
        dist = np.minimum(np.abs(k - start_pos), n - np.abs(k - start_pos))
        a = np.exp(-rate * dist)
        inits.append(a / np.sum(a))

    # Sinc-like (zero-crossing pattern)
    for trial in range(3):
        k = np.arange(n)
        center = n / 2.0
        width = rng.uniform(0.5, 2.0)
        a = np.sinc(width * (k - center) / n)
        a = np.maximum(a, 1e-6)
        inits.append(a / np.sum(a))

    return inits

def refine_pipeline(a, time_budget):

    n = len(a)
    best_a = a.copy()
    best_obj = true_obj(a)
    start = time.monotonic()

    # Stage 1: Temperature-annealed FISTA
    remaining = time_budget - (time.monotonic() - start)
    if remaining > 10:
        alloc = min(remaining * 0.3, 60)
        a1, obj1 = temperature_anneal(best_a, time_budget=alloc, method='fista')
        if obj1 < best_obj:
            best_obj = obj1
            best_a = a1.copy()
            print(f"    FISTA: {best_obj:.6f}")

    # Stage 2: L-BFGS with final temperature
    remaining = time_budget - (time.monotonic() - start)
    if remaining > 10:
        alloc = min(remaining * 0.25, 40)
        a2, obj2 = lbfgs_softmax(best_a, 100, time_budget=alloc, max_iter=200)
        if obj2 < best_obj:
            best_obj = obj2
            best_a = a2.copy()
            print(f"    L-BFGS: {best_obj:.6f}")

    # Stage 3: LP refinement (only for n <= 600)
    remaining = time_budget - (time.monotonic() - start)
    if remaining > 10 and n <= 600:
        alloc = min(remaining * 0.25, 40)
        a3, obj3 = lp_refine(best_a, max_iters=int(alloc / 2), time_budget=alloc)
        if obj3 < best_obj:
            best_obj = obj3
            best_a = a3.copy()
            print(f"    LP: {best_obj:.6f}")

    # Stage 4: Perturbation + re-optimize
    remaining = time_budget - (time.monotonic() - start)
    if remaining > 8:
        rng = np.random.default_rng(int(time.monotonic() * 1000) % 100000)
        for scale in [0.03, 0.08, 0.15]:
            if time.monotonic() - start > time_budget - 5:
                break
            a_pert = best_a * np.exp(rng.normal(0, scale, n))
            a_pert = np.maximum(a_pert, 0)
            S_p = np.sum(a_pert)
            if S_p < 1e-10:
                continue
            a_pert = a_pert / S_p
            alloc = min(remaining * 0.2, 15)
            a4, obj4 = fista_simplex(a_pert, 50, time_budget=alloc, max_iters=int(alloc * 40))
            if obj4 < best_obj:
                best_obj = obj4
                best_a = a4.copy()
                print(f"    Pert({scale}): {best_obj:.6f}")

    return best_a, best_obj


def propose_candidate(seed=42, budget_s=1500):
    """
    Main search function combining smoothed gradient optimization (FISTA + L-BFGS
    in softmax space), LP-based refinement, temperature annealing, and multi-start
    diversification to minimize 2*n*max(conv(a,a))/(sum(a))^2.
    """
    start_time = time.monotonic()
    deadline = start_time + float(budget_s) - 25.0

    best_seq = None
    best_score = np.inf

    # Load parent
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64)
        parent = np.clip(parent, 0, 1000.0)
        if parent.size > 0 and np.isfinite(parent).all():
            S_p = np.sum(parent)
            if S_p > 1e-10:
                parent_norm = parent / S_p
                parent_obj = true_obj(parent_norm)
                print(f"Parent (n={parent.size}): obj={parent_obj:.6f}")
                best_seq = parent_norm.copy()
                best_score = parent_obj
    except Exception:
        pass

    if best_seq is None:
        best_seq = np.ones(500) / 500
        best_score = true_obj(best_seq)
        print(f"Default init (n=500): obj={best_score:.6f}")

    n_best = len(best_seq)

    # === Phase 1: Refine the parent with smoothed methods ===
    print("\n=== Phase 1: Refine parent ===")
    remaining = deadline - time.monotonic()
    if remaining > 20:
        alloc = min(remaining * 0.3, 120)
        a_ref, obj_ref = refine_pipeline(best_seq, time_budget=alloc)
        if obj_ref < best_score:
            best_score = obj_ref
            best_seq = a_ref.copy()
            print(f"  After Phase 1: {best_score:.6f}")

    # === Phase 2: Multi-start with diverse initializations ===
    print("\n=== Phase 2: Multi-start ===")
    n_list = sorted(set([
        n_best,
        max(100, n_best - 100),
        min(800, n_best + 100),
        min(1000, n_best + 200),
    ]))

    candidate_pool = [(best_seq.copy(), best_score)]

    for n in n_list:
        if time.monotonic() > deadline - 30:
            break
        remaining = deadline - time.monotonic()
        if remaining < 20:
            break

        inits = diverse_init(n, seed=seed + n)
        for idx, a_init in enumerate(inits):
            if time.monotonic() > deadline - 20:
                break
            obj_init = true_obj(a_init)
            # Quick FISTA to improve
            alloc = min(remaining * 0.05, 8.0)
            a_quick, obj_quick = fista_simplex(a_init, t=10, time_budget=alloc,
                                               max_iters=int(alloc * 30))
            best_local = a_quick if obj_quick < obj_init else a_init
            best_local_obj = min(obj_quick, obj_init)
            candidate_pool.append((best_local, best_local_obj))

            if best_local_obj < best_score:
                best_score = best_local_obj
                best_seq = best_local.copy()
                print(f"  n={n} init{idx}: {best_local_obj:.6f} ** NEW BEST **")

    # === Phase 3: Refine top candidates ===
    print("\n=== Phase 3: Refine top candidates ===")
    candidate_pool.sort(key=lambda x: x[1])
    n_refine = min(4, len(candidate_pool))
    for idx in range(n_refine):
        if time.monotonic() > deadline - 30:
            break
        remaining = deadline - time.monotonic()
        if remaining < 20:
            break
        a_cand, obj_cand = candidate_pool[idx]
        if obj_cand > best_score + 0.02:
            break  # Not competitive

        n_cand = len(a_cand)
        alloc = min(remaining * 0.2, 60.0)
        print(f"  Candidate {idx} (n={n_cand}, obj={obj_cand:.6f})")

        a_ref, obj_ref = refine_pipeline(a_cand, time_budget=alloc)
        if obj_ref < best_score:
            best_score = obj_ref
            best_seq = a_ref.copy()
            print(f"    ** NEW BEST: n={n_cand}, obj={best_score:.6f} **")

    # === Phase 4: Intensive refinement with varying temperatures ===
    print("\n=== Phase 4: Intensive refinement ===")
    a_cur = best_seq.copy()
    n_cur = len(a_cur)

    for round_num in range(8):
        if time.monotonic() > deadline - 20:
            break
        remaining = deadline - time.monotonic()
        if remaining < 15:
            break

        # L-BFGS with increasing temperature
        t_val = min(100 * (round_num + 1), 1000)
        alloc = min(remaining * 0.15, 30.0)
        a_new, obj_new = lbfgs_softmax(a_cur, t_val, time_budget=alloc,
                                       max_iter=int(alloc * 5))

        if obj_new < best_score:
            best_score = obj_new
            best_seq = a_new.copy()
            a_cur = a_new.copy()
            print(f"  R{round_num} (t={t_val}): {best_score:.6f} ** NEW BEST **")
        else:
            # Try perturbation escape
            rng = np.random.default_rng(round_num * 7777 + 42)
            for pert_scale in [0.05, 0.1]:
                if time.monotonic() > deadline - 15:
                    break
                a_pert = a_cur * np.exp(rng.normal(0, pert_scale, n_cur))
                a_pert = np.maximum(a_pert, 0)
                S_p = np.sum(a_pert)
                if S_p < 1e-10:
                    continue
                a_pert = a_pert / S_p
                alloc_p = min(remaining * 0.08, 15.0)
                a_p2, obj_p2 = fista_simplex(a_pert, 30, time_budget=alloc_p,
                                             max_iters=int(alloc_p * 40))
                if obj_p2 < best_score:
                    best_score = obj_p2
                    best_seq = a_p2.copy()
                    a_cur = a_p2.copy()
                    print(f"  R{round_num} pert({pert_scale}): {best_score:.6f} ** NEW BEST **")

    # === Phase 5: LP polish ===
    print("\n=== Phase 5: LP polish ===")
    if time.monotonic() < deadline - 15 and len(best_seq) <= 600:
        alloc = min(deadline - time.monotonic() - 10, 50.0)
        a_lp, obj_lp = lp_refine(best_seq, max_iters=int(alloc / 2), time_budget=alloc)
        if obj_lp < best_score:
            best_score = obj_lp
            best_seq = a_lp.copy()
            print(f"  LP: {best_score:.6f} ** NEW BEST **")

    # === Phase 6: Final FISTA polish with high temperature ===
    print("\n=== Phase 6: Final polish ===")
    if time.monotonic() < deadline - 10:
        alloc = min(deadline - time.monotonic() - 5, 30.0)
        a_final, obj_final = fista_simplex(best_seq, t=500, time_budget=alloc,
                                           max_iters=int(alloc * 60))
        if obj_final < best_score:
            best_score = obj_final
            best_seq = a_final.copy()
            print(f"  Final FISTA: {best_score:.6f} ** NEW BEST **")

    # Final validation
    if best_seq is None:
        best_seq = np.ones(500) / 500
        best_score = true_obj(best_seq)

    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    val = evaluate_sequence(result)
    if val < np.inf and val < best_score:
        best_score = val
    elif val == np.inf:
        result = [1.0] * 300
        best_score = 2.0

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_score:.6f}, time={elapsed:.1f}s ===")

    return result
