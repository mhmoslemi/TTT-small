    fa = np.fft.rfft(a, 2*size)
    conv = np.fft.irfft(fa * fa, 2*size)

    n = len(a)
    S = float(np.sum(a))

    M = float(np.max(conv))

def softmax_obj(a, t):
    """Smooth approximation of max(conv(a,a)) using softmax."""
    n = len(a)
    conv = compute_conv(a)
    # Shift to avoid overflow
    conv_shifted = conv - np.max(conv)
    weights = t * np.exp(t * conv_shifted)
    log_sum = np.log(np.sum(weights)) / t
    return np.max(conv) + log_sum  # This is the softmax approximation

For the LP, I need to be careful about the matrix construction. The constraint is:
`conv(a, g)[k] <= M for all k`
`g[i] >= 0 for all i`
maximize `sum(g)`

`conv(a, g)[k] = sum_j a[j] * g[k-j]` (for valid j, k-j)

So the constraint matrix A has A[k, i] = sum over valid j of a[j] where k-j = i, i.e., j = k-i.
So A[k, i] = a[k-i] if 0 <= k-i < n, else 0.

This is a Toeplitz-like matrix. For n=500, it's 999 x 500. Building it explicitly and passing to linprog should be fine.

Let me write the complete solution:

"""
Optimized step-function coefficient search minimizing 2*n*max(conv(a,a))/(sum(a))^2.
Uses: proximal gradient with Barzilai-Borwein step, softmax-smoothed subgradient,
alternating LP polishing, and multi-n exploration.
"""

import numpy as np
from scipy.optimize import linprog
from scipy.interpolate import interp1d
import time


def compute_conv(a):
    """Compute full linear convolution a*a via FFT."""
    n = len(a)
    size = 2 * n - 1
    fa = np.fft.rfft(a, size)
    conv = np.fft.irfft(fa * fa, size)
    return conv[:size]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    S = float(np.sum(a))
    if S < 1e-10:
        return np.inf
    conv = compute_conv(a)
    M = float(np.max(conv))
    return 2.0 * n * M / (S * S)


def compute_smoothed_grad(a, temperature=100.0):
    """
    Compute smoothed gradient of max(conv(a,a)) using softmax weighting.
    Returns (gradient, max_value, weights).
    """
    n = len(a)
    conv = compute_conv(a)
    M = float(np.max(conv))
    
    # Softmax weights
    conv_shifted = (conv - M) * temperature
    # Clip to avoid underflow
    conv_shifted = np.clip(conv_shifted, -50, 0)
    weights = np.exp(conv_shifted)
    total_w = np.sum(weights)
    if total_w < 1e-15:
        # Only argmax is active
        active = np.where(conv >= M - 1e-12)[0]
        weights = np.zeros(len(conv))
        weights[active] = 1.0 / len(active)
    else:
        weights /= total_w
    
    # Compute gradient: grad[i] = 2 * sum_k weights[k] * a[k-i] (for valid k-i)
    grad = np.zeros(n)
    for k in range(len(conv)):
        w = weights[k]
        if w < 1e-14:
            continue
        if k < n:
            # i ranges from 0 to k
            grad[:k+1] += 2.0 * w * a[k::-1]
        else:
            # i ranges from k-n+1 to n-1
            grad[k-n+1:] += 2.0 * w * a[n-1:k-n-1:-1]
    
    return grad, M, weights


def project_simplex(v):
    """Project vector v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - 1))[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    w = np.maximum(v - theta, 0)
    return w


def bb_step(alpha_prev, x_prev, x_curr, f_prev, f_curr, obj_prev, obj_curr):
    """Barzilai-Borwein step size computation."""
    s = x_curr - x_prev
    y = obj_curr - obj_prev  # This isn't quite right for BB...
    # Actually BB uses gradient differences:
    # alpha_BB = (s^T s) / (s^T y) for minimization
    # But we don't have gradient differences readily.
    # Let's use a simpler adaptive scheme.
    return alpha_curr


def proximal_gradient_paint(a, time_budget, verbose=False):
    """
    Proximal gradient descent with:
    - Softmax-smoothed subgradient
    - Adaptive step size (backtracking + BB-like)
    - Adaptive temperature
    """
    n = len(a)
    a = a.copy().astype(np.float64)
    S = np.sum(a)
    if S < 1e-10:
        return a, np.inf
    a = a / S
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    
    step = 0.05
    temperature = 50.0
    patience = 0
    max_patience = 200
    
    prev_a = a.copy()
    prev_obj = best_obj
    
    while time.monotonic() - start < time_budget:
        # Compute smoothed gradient
        grad, M, weights = compute_smoothed_grad(a, temperature)
        
        # Full objective gradient (accounting for 1/S^2 factor)
        # f(a) = 2*n*M/S^2, df/da[i] = 2*n*grad[i]/S^2 - 4*n*M/(S^3)
        # = (2*n/S^2) * (grad[i] - 2*M/S)
        S = np.sum(a)
        full_grad = (2.0 * n / (S * S)) * (grad - 2.0 * M / S)
        
        # Take step
        a_new = a - step * full_grad
        
        # Project onto simplex (a >= 0, sum = 1)
        a_new = project_simplex(a_new)
        
        # Evaluate
        obj_new = compute_obj(a_new)
        
        if obj_new < best_obj:
            best_obj = obj_new
            best_a = a_new.copy()
            prev_a = a.copy()
            prev_obj = obj_new
            a = a_new
            patience = 0
            # Decrease temperature slightly to sharpen
            temperature = max(10.0, temperature * 0.995)
        else:
            patience += 1
            # Backtracking
            step *= 0.8
            if patience > max_patience:
                # Reset with perturbation
                rng = np.random.default_rng(int(time.monotonic() * 1000) % 100000)
                a = project_simplex(best_a + 0.01 * rng.normal(0, 1, n))
                step = 0.05
                temperature = max(50.0, temperature * 1.5)
                patience = 0
                best_a = a.copy()
                best_obj = compute_obj(a)
        
        if verbose and patience % 500 == 0:
            elapsed = time.monotonic() - start
            print(f"    PG: t={elapsed:.1f}s, obj={best_obj:.6f}, step={step:.4f}, temp={temperature:.1f}")
    
    return best_a, best_obj


def solve_lp(a, M=None, time_limit=5):
    """
    Solve LP: max sum(g) s.t. conv(a,g) <= M, g >= 0.
    M defaults to max(conv(a,a)).
    """
    n = len(a)
    if M is None:
        conv = compute_conv(a)
        M = float(np.max(conv))
    
    # Build constraint matrix: A[k, i] = a[k-i] if 0 <= k-i < n
    # Shape: (2n-1) x n
    rows = []
    cols = []
    vals = []
    for i in range(n):
        # For column i, A[k, i] = a[k-i] for valid k
        # k-i must be in [0, n-1], so k in [i, i+n-1]
        k_start = i
        k_end = min(i + n, 2 * n - 1)
        for k in range(k_start, k_end):
            rows.append(k)
            cols.append(i)
            vals.append(a[k - i])
    
    from scipy.sparse import coo_matrix
    A = coo_matrix((vals, (rows, cols)), shape=(2 * n - 1, n)).tocsr()
    
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    
    try:
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': time_limit})
        if res.status == 0:
            return np.maximum(res.x, 0)
    except Exception:
        pass
    return None


def alternating_lp(a, max_iters=30, time_budget=60.0, verbose=False):
    """Alternating LP: fix a, optimize g; then update a = (1-t)*a + t*g."""
    n = len(a)
    a = a.copy().astype(np.float64)
    S = np.sum(a)
    if S < 1e-10:
        return a, np.inf
    a = a / S
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    
    for it in range(max_iters):
        if time.monotonic() - start > time_budget:
            break
        
        conv = compute_conv(a)
        M = float(np.max(conv))
        
        # Try slightly relaxed M for exploration
        M_try = M * 1.001
        
        g = solve_lp(a, M_try, time_limit=3)
        if g is None:
            break
        
        S_g = np.sum(g)
        if S_g < 1e-10:
            break
        g = g / S_g
        
        # Line search
        best_t = 0.0
        best_obj_ls = best_obj
        for t in [0.1, 0.2, 0.3, 0.5, 0.7, 0.9, 1.0]:
            a_try = (1.0 - t) * a + t * g
            a_try = np.maximum(a_try, 0)
            S_try = np.sum(a_try)
            if S_try < 1e-10:
                continue
            a_try = a_try / S_try
            obj_try = compute_obj(a_try)
            if obj_try < best_obj_ls:
                best_obj_ls = obj_try
                best_t = t
        
        if best_t > 0:
            a = (1.0 - best_t) * a + best_t * g
            a = np.maximum(a, 0)
            a = a / np.sum(a)
            best_obj = best_obj_ls
            best_a = a.copy()
            if verbose:
                print(f"    LP it{it}: t={best_t:.2f}, obj={best_obj:.6f}")
        else:
            # Small step even if no improvement found in line search
            a = 0.95 * a + 0.05 * g
            a = np.maximum(a, 0)
            a = a / np.sum(a)
    
    return best_a, best_obj


def cosine_init(n, n_modes=10, seed=0):
    """Generate cosine-series initialization."""
    rng = np.random.default_rng(seed)
    k = np.arange(n)
    a = np.ones(n)
    for m in range(1, n_modes + 1):
        c = rng.uniform(-0.5, 0.5) / m
        phase = rng.uniform(0, 2 * np.pi)
        a += c * np.cos(2 * np.pi * m * k / n + phase)
    a = np.maximum(a, 0)
    S = np.sum(a)
    if S < 1e-10:
        return np.ones(n) / n
    return a / S


def raised_cosine_init(n, alpha=1.0):
    """Raised cosine initialization."""
    k = np.arange(n)
    x = 2.0 * k / (n - 1) - 1.0
    a = 0.5 * (1.0 + np.cos(np.pi * x)) ** alpha
    a = np.maximum(a, 1e-10)
    return a / np.sum(a)


def gaussian_init(n, sigma_frac=0.3):
    """Gaussian initialization."""
    k = np.arange(n)
    center = n / 2.0
    sigma = sigma_frac * n
    a = np.exp(-0.5 * ((k - center) / sigma) ** 2)
    return a / np.sum(a)


def adam_proximal_gradient(a, time_budget, verbose=False):
    """
    Adam-accelerated proximal gradient with softmax smoothing.
    More robust than vanilla PG due to adaptive per-coordinate steps.
    """
    n = len(a)
    a = a.copy().astype(np.float64)
    S = np.sum(a)
    if S < 1e-10:
        return a, np.inf
    a = a / S
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    
    lr = 0.02
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    temperature = 80.0
    
    m = np.zeros(n)  # First moment
    v = np.zeros(n)  # Second moment
    t = 0
    patience = 0
    
    while time.monotonic() - start < time_budget:
        t += 1
        
        # Compute smoothed gradient of M = max(conv(a,a))
        grad_M, M_val, _ = compute_smoothed_grad(a, temperature)
        
        # Full gradient of f = 2*n*M/S^2
        S = np.sum(a)
        grad_full = (2.0 * n / (S * S)) * (grad_M - 2.0 * M_val / S)
        
        # Adam update
        m = beta1 * m + (1 - beta1) * grad_full
        v = beta2 * v + (1 - beta2) * grad_full ** 2
        m_hat = m / (1 - beta1 ** t)
        v_hat = v / (1 - beta2 ** t)
        
        a_new = a - lr * m_hat / (np.sqrt(v_hat) + eps)
        
        # Project onto simplex
        a_new = project_simplex(a_new)
        
        obj_new = compute_obj(a_new)
        
        if obj_new < best_obj:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new
            patience = 0
            temperature = max(20.0, temperature * 0.998)
        else:
            patience += 1
            if patience % 100 == 0:
                lr *= 0.9
                temperature = min(200.0, temperature * 1.1)
            if patience > 500:
                # Restart with perturbation
                rng = np.random.default_rng(t)
                a = project_simplex(best_a + 0.005 * rng.normal(0, 1, n))
                m = np.zeros(n)
                v = np.zeros(n)
                t = 0
                lr = 0.02
                temperature = 80.0
                patience = 0
    
    return best_a, best_obj
