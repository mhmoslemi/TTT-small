def fast_conv(a):

    s = 1
    while s < 2*n-1:
        s <<= 1
    fa = np.fft.rfft(a, s)
    return np.fft.irfft(fa*fa, s)[:2*n-1]

    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2*len(a)*b.max()/(s*s))

    """Compute smooth max of conv(a,a) and its gradient w.r.t. a."""
    n = len(a)
    c = fast_conv(a)
    c_max = c.max()
    shifted = beta * (c - c_max)
    shifted = np.clip(shifted, -100, 100)
    exp_shifted = np.exp(shifted)
    Z = exp_shifted.sum()
    p = exp_shifted / Z

"""
Minimizes J(a) = 2n*max(conv(a,a))/(sum(a))^2 using:
1. Fast exact coordinate descent (O(n) per coordinate via precomputed deltas)
2. Smooth mirror descent with FFT-based gradients for basin exploration
3. LP mirror descent with golden-section line search for precise refinement
4. Multi-length exploration (512-2048) with diverse initializations
5. Adaptive restarts with structured perturbations
6. Dual-guided direction from LP for informed coordinate updates
"""

import numpy as np
from scipy.optimize import linprog, minimize
from scipy.sparse import csr_matrix
from scipy.signal import correlate
import time


def fast_conv(a):
    """Full self-convolution via FFT, O(n log n)."""
    n = len(a)
    s = 1
    while s < 2 * n - 1:
        s <<= 1
    fa = np.fft.rfft(a, s)
    return np.fft.irfft(fa * fa, s)[:2 * n - 1]


def compute_J(a):
    """Compute J(a) = 2*n*max(conv(a,a))/(sum(a))^2."""
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))


def smooth_conv_and_grad(a, beta):
    """Compute smooth max of conv(a,a) and its gradient w.r.t. a."""
    n = len(a)
    c = fast_conv(a)
    c_max = float(c.max())
    shifted = beta * (c - c_max)
    shifted = np.clip(shifted, -200, 200)
    exp_shifted = np.exp(shifted)
    Z = float(exp_shifted.sum())
    p = exp_shifted / Z
    g_val = (c_max + np.log(max(Z, 1e-300))) / beta
    # Gradient: grad_i = 2 * sum_j a[j] * p[i+j]
    # = 2 * correlate(p, a, mode='valid')[i]
    grad = 2.0 * correlate(p.astype(np.float64), a, mode='valid')
    return g_val, grad, p


def F_and_grad(a, beta):
    """Compute J_beta(a) = 2n*smooth_max(conv(a,a))/(sum(a))^2 and gradient."""
    n = len(a)
    s = float(a.sum())
    if s < 1e-10:
        return 1e10, np.zeros(n)
    g_val, grad_g, _ = smooth_conv_and_grad(a, beta)
    F_val = 2.0 * n * g_val / (s * s)
    grad_F = 2.0 * n * (grad_g * s - 2.0 * g_val) / (s ** 3)
    return F_val, grad_F


def mirror_descent_smooth(a, beta_schedule, deadline, tracker=None):
    """
    Mirror descent with smooth objective on the simplex.
    beta_schedule: list of (beta, num_iters)
    """
    a = np.maximum(a.astype(np.float64), 1e-10)
    a = a / a.sum()
    n = len(a)
    current_J = compute_J(a)
    if tracker is not None:
        tracker.update(a)

    for beta, num_iters in beta_schedule:
        eta = 1.0 / (n * max(beta, 1.0))
        base_eta = eta

        for it in range(num_iters):
            if time.monotonic() > deadline:
                break

            F_val, grad_F = F_and_grad(a, beta)
            if not np.isfinite(F_val):
                break

            scaled = -eta * grad_F
            scaled = np.clip(scaled, -30, 30)
            a_new = a * np.exp(scaled)
            a_new_sum = a_new.sum()
            if a_new_sum < 1e-300:
                break
            a_new = a_new / a_new_sum

            J_new = compute_J(a_new)

            if J_new < current_J - 1e-14:
                a = a_new
                current_J = J_new
                if tracker is not None:
                    tracker.update(a)
                if it % 20 == 0:
                    eta = min(eta * 1.05, 5.0 / (n * max(beta, 1.0)))
            else:
                eta = base_eta * (0.7 ** (it // 50 + 1))
                if eta < 1e-12:
