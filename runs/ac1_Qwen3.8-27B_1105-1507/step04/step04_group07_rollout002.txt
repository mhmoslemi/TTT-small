# Precomputed sparsity cache
_spar_cache = {}

def get_sparsity(n):
    if n not in _spar_cache:
        rows = []
        cols = []
        for k in range(2 * n - 1):
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            rows.extend([k] * (je - js + 1))
            cols.extend(range(js, je + 1))
        _spar_cache[n] = (np.array(rows, dtype=np.int32), np.array(cols, dtype=np.int32), len(rows))
    return _spar_cache[n]

def fast_conv(a):
    n = len(a)
    s = 1
    while s < 2 * n - 1:
        s <<= 1
    fa = np.fft.rfft(a, s)
    return np.fft.irfft(fa * fa, s)[:2 * n - 1]

def compute_J(a):
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))

def lp_best_response(a, M, time_limit=8):
    n = len(a)
    rows, cols, nnz = get_sparsity(n)
    vals = np.empty(nnz)
    for i in range(nnz):
        k = rows[i]
        j = cols[i]
        kj = k - j
        if 0 <= kj < n:
            vals[i] = a[kj]
        else:
            vals[i] = 0.0
    A = csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                options={'presolve': True, 'time_limit': time_limit})
    if r.success and r.x is not None:
        g = np.maximum(r.x, 0.0)
        if g.sum() > 1e-10:
            # Try to extract dual variables
            dual = None
            try:

                    if dual.ndim == 1 and len(dual) == 2 * n - 1:
                        dual = np.maximum(dual, 0.0)
            except:
                pass
            return g, dual, r.slack
    return None, None, None

        if len(active) == 0:
            active = np.array([int(np.argmax(b))])
        
        # Compute subgradient
        grad = np.zeros(n)
        for k in active:
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            for j in range(js, je + 1):
                grad[j] += a[k - j]
        grad /= len(active)
        
        # Add negative component to push down large a_j
        direction = grad + 2.0 * M * a
        v = momentum * v + (1.0 - momentum) * direction
        a_new = a - lr * v
        a_new = np.maximum(a_new, 1e-8)
        ss = a_new.sum()
        a_new = a_new / ss
        
        J_new = compute_J(a_new)
        if J_new < current_J - 1e-15:
            a = a_new
            current_J = J_new
            lr = min(lr * 1.005, 10.0 / n)
        else:
            lr *= 0.6
            v *= 0.5
            if lr < 1e-18:
                break
        if it % 20 == 0:
            tracker.update(a)
    tracker.update(a)
    return a

def lse_gradient_descent(a, max_iters, deadline, tracker):
    a = np.maximum(a.astype(np.float64), 1e-6)
    a = a / a.sum()
    n = len(a)
    current_J = compute_J(a)
    tracker.update(a)

    tau = 30.0
    lr = 2.0 / (n * tau)
    momentum = 0.8
    v = np.zeros(n)

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        b = fast_conv(a)
        b_max = b.max()
        exp_b = np.exp(tau * (b - b_max))
        p = exp_b / exp_b.sum()

        # Gradient of LSE: sum_k p_k * d(b_k)/d(a_j)
        # d(b_k)/d(a_j) = a[k-j] (for valid k-j)
        s = 1
        while s < 3 * n:
            s <<= 1
        fp = np.fft.rfft(p, s)
        fa = np.fft.rfft(a, s)
        corr = np.fft.irfft(fp * np.conj(fa), s)
        grad = 2.0 * corr[:n]

        v = momentum * v + (1.0 - momentum) * grad
        a_new = a - lr * v
        a_new = np.maximum(a_new, 1e-6)
        a_new = a_new / a_new.sum()
        J_new = compute_J(a_new)
        if J_new < current_J - 1e-15:
            a = a_new
            current_J = J_new
            tau = min(200.0, tau * 1.008)
            lr = 2.0 / (n * tau)
        else:
            lr *= 0.5
            v *= 0.3
            if lr < 1e-18:
                break
        if it % 50 == 0:
            tracker.update(a)
    tracker.update(a)
    return a

def dct_parameterize(n, K, coeffs):
    """Reconstruct a from DCT coefficients."""
    c = np.zeros(n)
    for k in range(min(K, n)):
        if k < len(coeffs):
            c += coeffs[k]
    # DCT reconstruction: a[j] = sum_k c_k * cos(pi*(2j+1)*k/(2n))
    j = np.arange(n)
    a = np.zeros(n)
    for k in range(min(K, n)):
        if k < len(coeffs):
            a += coeffs[k] * np.cos(np.pi * (2 * j + 1) * k / (2 * n))
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-10:
        return np.ones(n) / n
    return a / s

def flat_spectrum_init(n, rng):
    half = n // 2 + 1
    angles = rng.uniform(0, 2 * np.pi, half)
    A = np.zeros(n, dtype=complex)
    A[:half] = np.exp(1j * angles[:half])
    for k in range(1, n - n // 2):
        A[k] = np.conj(A[n - k])
    a = np.fft.ifft(A).real
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-10:
        return np.ones(n) / n
    return a / s

def resample_seq(src, dest_n):
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    if src.size == 0:
        return np.ones(dest_n) / dest_n
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)

def generate_inits(n, rng, parent):
    inits = []
    if parent is not None and parent.size > 0:
        inits.append(resample_seq(parent, n))
        inits.append(resample_seq(parent[::-1], n))

    inits.append(np.ones(n) / n)
    inits.append(rng.uniform(0.1, 1.0, n))
    inits.append(flat_spectrum_init(n, rng))
    inits.append(flat_spectrum_init(n, rng))
    inits.append(flat_spectrum_init(n, rng))

    t = np.linspace(0, 1, n, endpoint=False)
    for freq in [2, 3, 5, 7, 11, 13]:
        inits.append(1.0 + 0.4 * np.cos(2 * np.pi * freq * t + rng.uniform(0, 6.28)))
    inits.append(1.0 + 0.3 * np.cos(2 * np.pi * 3.7 * t) + 0.2 * np.cos(2 * np.pi * 7.1 * t))
    inits.append(0.5 + 0.5 * np.cos(np.pi * t) ** 2)
    inits.append(0.3 + 0.7 * (1.0 - np.abs(2 * t - 1.0)) ** 0.5)
    inits.append(0.5 * np.exp(-6 * (t - 0.3) ** 2) + 0.5 * np.exp(-6 * (t - 0.7) ** 2) + 0.05)
    inits.append(1.0 + 0.25 * np.sin(2 * np.pi * t) + 0.2 * np.sin(4 * np.pi * t)
                + 0.15 * np.sin(7 * np.pi * t) + 0.1 * np.sin(11 * np.pi * t))
    inits.append((t + 0.1) ** rng.uniform(-1.0, 1.0))
    inits.append((1.0 - t + 0.1) ** rng.uniform(-1.0, 1.0))

    # Triangular wave
    inits.append(1.0 - 0.5 * np.abs(2 * (t - 0.5)))
    # Double bump
    inits.append(np.exp(-10 * (t - 0.25) ** 2) + np.exp(-10 * (t - 0.75) ** 2) + 0.1)
    # Sawtooth-like
    inits.append(np.abs(np.sin(np.pi * t)) ** 0.5 + 0.2)
    
    for x in inits:
        x = np.maximum(x, 0.001).astype(np.float64)
        x = x / x.sum()
    return inits

class ScoreTracker:
    def __init__(self):
        self.best = None
        self.best_score = float('inf')
        self.t_start = time.monotonic()

    def update(self, a):
        a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
        if a.size == 0 or a.sum() < 0.01:
            return
        score = float(evaluate_sequence(a.tolist()))
        if np.isfinite(score) and score < self.best_score:
            self.best = a.copy()
            self.best_score = score
            elapsed = time.monotonic() - self.t_start
            print(f"  * J={score:.8f} n={a.size} t={elapsed:.1f}s", flush=True)

    def report(self, label=""):
        if self.best is not None:
            print(f"  {label} best: J={self.best_score:.8f} n={self.best.size}", flush=True)

def simulated_annealing(a, deadline, tracker, n_restarts=30, t0=0.1):
    rng = np.random.default_rng(777)
    n = a.size
    a = a.copy().astype(np.float64)
    current_J = compute_J(a)
    tracker.update(a)
    temp = t0

    for r in range(n_restarts):
        if time.monotonic() > deadline:
            break
        a_pert = a + rng.normal(0, temp, n)
        a_pert = np.maximum(a_pert, 1e-6)
        a_pert = a_pert / a_pert.sum()

        # Short LP refinement
        a_refined = run_lp_opt(a_pert, max(3, 15 - r // 4), deadline, tracker, tighten=0.02)
        J = compute_J(a_refined)
        delta = J - current_J
        if delta < 0 or (temp > 1e-10 and rng.random() < np.exp(-max(0, delta) / max(temp, 1e-10))):
            a = a_refined
            current_J = J
        temp *= 0.93
    tracker.update(a)
    return a

def coordinate_descent(a, max_steps, deadline, tracker):
    a = a.copy().astype(np.float64)
    a = a / a.sum()
    n = len(a)
    current_J = compute_J(a)
    tracker.update(a)
    rng = np.random.default_rng(42)

    for step in range(max_steps):
        if time.monotonic() > deadline:
            break
        j = int(rng.integers(0, n))
        improved = False
        for scale in [0.5, 1.5, 0.2, 2.0, 0.8, 1.2]:
            a_trial = a.copy()
            a_trial[j] *= scale
            a_trial = np.maximum(a_trial, 1e-8)
            ss = a_trial.sum()
            if ss < 1e-10:
                continue
            a_trial = a_trial / ss
            J_trial = compute_J(a_trial)
            if J_trial < current_J - 1e-14:
                a = a_trial
                current_J = J_trial
                improved = True
                break
        if step % 200 == 0:
            tracker.update(a)
    tracker.update(a)
    return a

def propose_candidate(seed=42, budget_s=1500, **kwargs):
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + float(budget_s) - 30.0
    tracker = ScoreTracker()

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        if p.ndim == 1 and p.size > 0 and np.all(np.isfinite(p)):
            parent = np.maximum(p, 0.0)
            tracker.update(parent)
            print(f"Parent: n={parent.size}, J={tracker.best_score:.8f}", flush=True)
    except (NameError, AttributeError):
        pass

    # === Phase 1: Multi-length LP optimization (300s) ===
    t_phase1_end = time.monotonic() + min(300.0, (deadline - time.monotonic()) * 0.2)
    lengths = [256, 512]
    time_per_length = (t_phase1_end - time.monotonic()) / len(lengths)

    for n in lengths:
        t_start = time.monotonic()
        t_end = min(t_phase1_end, t_start + time_per_length)
        if time.monotonic() >= t_end - 30:
            continue
        print(f"\n{'='*50} Phase1 n={n} budget={time_per_length:.0f}s {'='*50}", flush=True)

        inits = generate_inits(n, rng, parent)
        best_for_n = None
        best_J_for_n = float('inf')
        
        # Sort inits by initial J (quick eval)
        init_Js = []
        for a0 in inits:
            J0 = compute_J(a0)
            init_Js.append((J0, a0))
        init_Js.sort(key=lambda x: x[0])
        
        n_inits = len(inits)
        init_budget = max(8.0, (t_end - time.monotonic()) / n_inits)

        for idx, (J0, a0) in enumerate(init_Js):
            if time.monotonic() > t_end - 20:
                break
            init_budget_actual = max(5.0, (t_end - time.monotonic()) / max(1, n_inits - idx))
            
            # LP phase
            n_lp_iters = max(8, int(init_budget_actual / 3.0))
            a = run_lp_opt(a0, n_lp_iters, t_end - 30, tracker, tighten=0.01)

            # Subgradient polish
            if time.monotonic() < t_end - 20:
                a = momentum_subgradient(a, max(30, int(20000 / n)), t_end - 20, tracker)

            J = compute_J(a)
            if J < best_J_for_n:
                best_J_for_n = J
                best_for_n = a.copy()

        if best_for_n is not None:
            tracker.update(best_for_n)
            if parent is None or best_J_for_n < (compute_J(resample_seq(parent, best_for_n.size)) if parent is not None else float('inf')):
                parent = best_for_n
            tracker.report(f"Phase1 n={n} ({time.monotonic()-t_start:.1f}s): J={best_J_for_n:.8f}")

    # === Phase 2: DCT parameterization (200s) ===
    t_phase2_end = time.monotonic() + min(200.0, (deadline - time.monotonic()) * 0.15)
    print(f"\n{'='*50} Phase2 DCT {'='*50}", flush=True)
    
    dct_lengths = [256, 512, 768]
    dct_time_each = (t_phase2_end - time.monotonic()) / len(dct_lengths)
    
    for n in dct_lengths:
        if time.monotonic() > t_phase2_end - 20:
            break
        t_start_dct = time.monotonic()
        K = min(30, n // 2)
        a_dct = dct_optimize(n, K, min(t_phase2_end, t_start_dct + dct_time_each), tracker, parent)
        J_dct = compute_J(a_dct)
        print(f"  DCT n={n} K={K}: J={J_dct:.8f}", flush=True)
        if parent is None or J_dct < compute_J(resample_seq(parent, a_dct.size)):
            parent = a_dct
            tracker.update(a_dct)

    tracker.report("Phase2 DCT")

    # === Phase 3: Focused optimization (400s) ===
    if tracker.best is not None:
        t_phase3_end = time.monotonic() + min(400.0, (deadline - time.monotonic()) * 0.3)
        n_focus = tracker.best.size
        print(f"\n{'='*50} Phase3 Focus n={n_focus} {'='*50}", flush=True)

        a = tracker.best.copy().astype(np.float64)
        
        # Extended LP with decreasing tightening
        tighten_sched = [0.02, 0.015, 0.01, 0.008, 0.005, 0.004, 0.003, 0.002]
        for tight in tighten_sched:
            if time.monotonic() > t_phase3_end - 100:
                break
            a = run_lp_opt(a, 25, t_phase3_end - 100, tracker, tighten=tight)

        # Momentum subgradient
        if time.monotonic() < t_phase3_end - 50:
            a = momentum_subgradient(a, max(200, int(50000 / n_focus)), t_phase3_end - 50, tracker)

        # LSE polish
        if time.monotonic() < t_phase3_end - 10:
            a = lse_gradient_descent(a, max(200, int(30000 / n_focus)), t_phase3_end - 10, tracker)

        tracker.report("Phase3 Focus")
        if a is not None:
            tracker.update(a)

    # === Phase 4: Simulated annealing (300s) ===
    if tracker.best is not None:
        t_phase4_end = time.monotonic() + min(300.0, (deadline - time.monotonic()) * 0.2)
        print(f"\n{'='*50} Phase4 SA {'='*50}", flush=True)
        a = simulated_annealing(tracker.best, t_phase4_end, tracker, n_restarts=30, t0=0.08)
        if time.monotonic() < deadline - 60:
            a = run_lp_opt(a, 15, deadline - 60, tracker, tighten=0.01)
            if time.monotonic() < deadline - 30:
                a = lse_gradient_descent(a, 500, deadline - 30, tracker)
        tracker.report("Phase4 SA")

    # === Phase 5: Final polish (100s) ===
    if tracker.best is not None:
        t_phase5_end = deadline - 5
        print(f"\n{'='*50} Phase5 Final {'='*50}", flush=True)
        a = coordinate_descent(tracker.best, 2000, t_phase5_end - 20, tracker)
        if time.monotonic() < t_phase5_end - 10:
            a = run_lp_opt(a, 10, t_phase5_end - 10, tracker, tighten=0.005)
        if time.monotonic() < t_phase5_end - 5:
            a = lse_gradient_descent(a, 300, t_phase5_end - 5, tracker)
        tracker.report("Phase5 Final")

    # Fallback
    if tracker.best is None:
        tracker.best = np.ones(256)
        tracker.update(tracker.best)

    print(f"\n{'='*50} FINAL {'='*50}", flush=True)
    print(f"J={tracker.best_score:.8f} n={tracker.best.size} time={time.monotonic()-tracker.t_start:.1f}s", flush=True)
    return [float(x) for x in tracker.best]
