"""
Polishing optimizer for minimizing 2*n*max(conv(a,a))/(sum(a))^2.
Uses intensive LP-based alternating with adaptive M, gradient refinement,
simulated annealing restarts, and targeted peak-shaving.
"""

import numpy as np
from scipy.optimize import minimize, linprog
from scipy.fft import rfft, irfft
import time


def compute_conv(a):
    """Compute linear convolution a*a using FFT."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    conv = irfft(fa * fa, size)
    return conv[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    conv = compute_conv(a)
    max_b = float(np.max(conv))
    sum_a = float(np.sum(a))
    if sum_a < 0.01:
        return np.inf
    return 2.0 * n * max_b / (sum_a * sum_a)


def fft_convolve(a, b):
    """Linear convolution of a and b using FFT."""
    n, m = len(a), len(b)
    size = 1
    while size < n + m - 1:
        size <<= 1
    fa = rfft(a, size)
    fb = rfft(b, size)
    return irfft(fa * fb, size)[:n + m - 1]


def softmax_fn(z):
    """Numerically stable softmax."""
    z = z - np.max(z)
    e = np.exp(z)
    return e / np.sum(e)


def build_toeplitz(a):
    """Build Toeplitz matrix where (A@x)[k] = (conv(a,x))[k]."""
    n = len(a)
    A = np.zeros((2 * n - 1, n), dtype=np.float64)
    for j in range(n):
        k_start = j
        k_end = min(j + n, 2 * n - 1)
        A[k_start:k_end, j] = a[:k_end - k_start]
    return A


def soft_max_obj_and_grad(a, tau):
    """
    Compute smoothed max of conv(a,a) and gradient w.r.t. a.
    """
    n = len(a)
    conv = compute_conv(a)
    cmax = float(np.max(conv))
    shifted = np.clip((conv - cmax) / tau, -500.0, 500.0)
    exp_conv = np.exp(shifted)
    Z = float(np.sum(exp_conv))
    soft_max_val = cmax + tau * np.log(Z)
    w = exp_conv / Z

    a_flipped = a[::-1]
    conv_wa = fft_convolve(w, a_flipped)
    corr = conv_wa[n - 1:2 * n - 1]
    grad_a = 2.0 * corr

    return soft_max_val, grad_a


def gradient_step(a, tau, lr):

    n = len(a)
    _, grad = soft_max_obj_and_grad(a, tau)
    grad_centered = grad - np.mean(grad)
    a_new = a - lr * grad_centered
    a_new = np.maximum(a_new, 0)
    s = np.sum(a_new)
    if s > 1e-15:
        a_new = a_new / s
    return a_new


def lp_solve_direction(a, M, time_limit=15):
    """Solve LP: max sum(g) s.t. (a*g)[k] <= M for all k, g >= 0. Returns normalized g or None."""
    n = len(a)
    A = build_toeplitz(a)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': time_limit})
        if res.status == 0:
            g = np.maximum(res.x, 0)
            s = float(np.sum(g))
            if s > 1e-10:
                return g / s
    except Exception:
        pass
    return None

            lr = 0.3

    return best_a, best_obj


def targeted_peak_shave(a, time_budget, verbose=False):
    """
    Targeted optimization: identify peak lag(s) and do focused gradient steps.
    """
    n = len(a)
    a = a.astype(np.float64).copy()
    s = np.sum(a)
    if s < 1e-10:
        return a, compute_obj(a)
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    while time.monotonic() - start < time_budget - 3:
        conv = compute_conv(a)
        max_val = float(np.max(conv))
        # Find all lags within 1% of the max
        threshold = max_val * 0.999
        peak_lags = np.where(conv >= threshold)[0]

        # Gradient of max at peak lag m: grad_a[i] = 2*a[m-i] (if valid)
        # Average over peak lags
        grad = np.zeros(n)
        for m in peak_lags:
            for i in range(n):
                k = m - i
                if 0 <= k < n:
                    grad[i] += 2.0 * a[k]
        grad /= len(peak_lags)
        grad_centered = grad - np.mean(grad)

        # Try several step sizes
        best_step_obj = best_obj
        best_step_a = best_a
        for lr in [0.01, 0.02, 0.05, 0.1, 0.2, 0.3]:
            a_try = a - lr * grad_centered
            a_try = np.maximum(a_try, 0)
            s_try = np.sum(a_try)
            if s_try < 1e-10:
                continue
            a_try = a_try / s_try
            obj_try = compute_obj(a_try)
            if obj_try < best_step_obj:
                best_step_obj = obj_try
                best_step_a = a_try.copy()

        if best_step_obj < best_obj - 1e-15:
            best_obj = best_step_obj
            best_a = best_step_a.copy()
            a = a_new if False else best_step_a
            if verbose:
                print(f"    PeakShave: obj={best_obj:.8f}")
        else:
            # No improvement, small perturbation
            rng = np.random.default_rng(int(time.monotonic() * 10000) % 99999)
            a = best_a * np.exp(rng.normal(0, 0.02, n))
            a = np.maximum(a, 0)
            s = np.sum(a)
            a = a / s

    return best_a, best_obj


def resample_sequence(a, new_n):
    """Resample sequence to new length via interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return a_new


def load_parent():
    """Try to load parent sequence."""
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64)
        parent = np.clip(parent, 0, 1000.0)
        if parent.size > 0 and np.isfinite(parent).all():
            obj = compute_obj(parent)
            s = np.sum(parent)
            if s > 0.01:
                parent_norm = parent / s
                print(f"Loaded parent: n={parent.size}, obj={obj:.8f}")
                return parent_norm.copy(), obj
    except Exception as e:
        print(f"No parent available: {e}")
    return None, np.inf


def combined_polish(a, time_budget, verbose=False):
    """
    Combined polishing: alternates LP, gradient, and peak shaving.
    """
    n = len(a)
    a = a.astype(np.float64).copy()
    s = np.sum(a)
    if s < 1e-10:
        return a, compute_obj(a)
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    while time.monotonic() - start < time_budget - 10:
        remaining = time_budget - (time.monotonic() - start)
        if remaining < 15:
            break

        round_obj = best_obj

        # Step 1: LP alternating (most powerful)
        if remaining > 30:
            a_new, obj_new = lp_alternate(best_a, min(remaining * 0.4, 120.0), verbose=verbose)
            if obj_new < best_obj:
                best_obj = obj_new
                best_a = a_new.copy()
                if verbose:
                    print(f"    Combined LP: {best_obj:.8f}")
            round_obj = best_obj

        # Step 2: Gradient polish
        remaining = time_budget - (time.monotonic() - start)
        if remaining > 20:
            a_new, obj_new = gradient_polish(best_a, min(remaining * 0.2, 40.0))
            if obj_new < best_obj:
                best_obj = obj_new
                best_a = a_new.copy()
                if verbose:
                    print(f"    Combined Grad: {best_obj:.8f}")

        # Step 3: Targeted peak shaving
        remaining = time_budget - (time.monotonic() - start)
        if remaining > 15:
            a_new, obj_new = targeted_peak_shave(best_a, min(remaining * 0.15, 25.0))
            if obj_new < best_obj:
                best_obj = obj_new
                best_a = a_new.copy()
                if verbose:
                    print(f"    Combined Peak: {best_obj:.8f}")

        if best_obj == round_obj:
            # No progress this round, add perturbation
            rng = np.random.default_rng(int(time.monotonic() * 7) % 50000)
            scale = 0.05
            a_pert = best_a * np.exp(rng.normal(0, scale, n))
            a_pert = np.maximum(a_pert, 0)
            s_pert = np.sum(a_pert)
            if s_pert > 1e-10:
                a_pert = a_pert / s_pert
                # Quick gradient polish on perturbed
                a_quick, obj_quick = gradient_polish(a_pert, 5.0)
                if obj_quick < best_obj:
                    best_obj = obj_quick
                    best_a = a_quick.copy()
                    if verbose:
                        print(f"    Combined Perturb: {best_obj:.8f}")

    return best_a, best_obj


def propose_candidate(seed=42, budget_s=1500):
    """
    Main search: starts from parent, performs intensive LP-based polishing,
    gradient refinement, and targeted peak shaving. Tries nearby lengths
    and multiple perturbation-based restarts.
    """
    start_time = time.monotonic()
    deadline = start_time + float(budget_s) - 35.0

    # Load parent
    best_a, best_obj = load_parent()

    if best_a is None:
        # Fallback: create a reasonable initial sequence
        n = 600
        rng = np.random.default_rng(seed)
        k = np.arange(n)
        best_a = 1.0 + 0.3 * np.cos(2 * np.pi * 1.5 * k / n) + 0.2 * np.cos(2 * np.pi * 3.7 * k / n)
        best_a = np.maximum(best_a, 0.1)
        best_a = best_a / np.sum(best_a)
        best_obj = compute_obj(best_a)
        print(f"Generated initial: n={n}, obj={best_obj:.8f}")

    print(f"Starting: n={len(best_a)}, obj={best_obj:.8f}")

    # Phase 1: Intensive combined polish on the parent
    n_cur = len(best_a)
    phase1_budget = min(500.0, (deadline - time.monotonic()) * 0.35)
    print(f"\n=== Phase 1: Combined Polish (budget={phase1_budget:.0f}s) ===")
    a_pol, obj_pol = combined_polish(best_a, phase1_budget, verbose=True)
    if obj_pol < best_obj:
        best_obj = obj_pol
        best_a = a_pol.copy()
        print(f"  Phase 1 result: obj={best_obj:.8f}")

    # Phase 2: Try perturbation-based restarts
    print(f"\n=== Phase 2: Perturbation Restarts ===")
    n_restarts = 5
    restart_budget_each = min(80.0, (deadline - time.monotonic()) / (n_restarts + 1))

    for r in range(n_restarts):
        if time.monotonic() > deadline - 60:
            break
        remaining = deadline - time.monotonic()
        if remaining < 20:
            break

        rng = np.random.default_rng(seed + r * 7919 + 12345)
        scale = 0.03 + 0.02 * r
        a_pert = best_a * np.exp(rng.normal(0, scale, n_cur))
        a_pert = np.maximum(a_pert, 0)
        s_pert = np.sum(a_pert)
        if s_pert < 1e-10:
            continue
        a_pert = a_pert / s_pert

        # Quick LP polish on perturbed
        budget_r = min(restart_budget_each, (deadline - time.monotonic()) - 30)
        if budget_r < 15:
            break
        a_r, obj_r = lp_alternate(a_pert, budget_r, verbose=False)

        if obj_r < best_obj:
            best_obj = obj_r
            best_a = a_r.copy()
            print(f"  Restart {r}: obj={best_obj:.8f} ** NEW BEST **")
        else:
            print(f"  Restart {r}: obj={obj_r:.8f} (no improvement)")

    # Phase 3: Try nearby lengths
    print(f"\n=== Phase 3: Nearby Lengths ===")
    nearby_lengths = []
    for delta in [5, 10, 15, 20, -10, -20, 30]:
        nl = n_cur + delta
        if 100 <= nl <= 1000:
            nearby_lengths.append(nl)
    nearby_lengths = sorted(set(nearby_lengths))

    for nl in nearby_lengths[:4]:
        if time.monotonic() > deadline - 60:
            break
        remaining = deadline - time.monotonic()
        if remaining < 30:
            break

        a_resampled = resample_sequence(best_a, nl)
        budget_nl = min(60.0, (deadline - time.monotonic()) - 20)
        if budget_nl < 15:
            break

        a_nl, obj_nl = lp_alternate(a_resampled, budget_nl, verbose=False)
        if obj_nl < best_obj:
            best_obj = obj_nl
            best_a = a_nl.copy()
            n_cur = nl
            print(f"  Length {nl}: obj={best_obj:.8f} ** NEW BEST **")
        else:
            print(f"  Length {nl}: obj={obj_nl:.8f}")

    # Phase 4: Final intensive polish
    print(f"\n=== Phase 4: Final Polish ===")
    remaining = deadline - time.monotonic()
    if remaining > 30:
        final_budget = min(remaining - 10, 400.0)
        a_final, obj_final = combined_polish(best_a, final_budget, verbose=True)
        if obj_final < best_obj:
            best_obj = obj_final
            best_a = a_final.copy()
            print(f"  Final polish: obj={best_obj:.8f}")

    # Phase 5: Very final LP grind
    remaining = deadline - time.monotonic()
    if remaining > 20:
        a_grind, obj_grind = lp_alternate(best_a, remaining - 5, verbose=False)
        if obj_grind < best_obj:
            best_obj = obj_grind
            best_a = a_grind.copy()
            print(f"  Final grind: obj={best_obj:.8f}")

    # Final validation
    if best_a is None:
        best_a = np.ones(200) / 200
        best_obj = compute_obj(best_a)

    result = [float(max(0.0, min(1000.0, x))) for x in best_a]
    val = evaluate_sequence(result)
    if val < np.inf:
        final_obj = val
    else:
        result = [1.0] * 100
        final_obj = 2.0

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={final_obj:.8f}, time={elapsed:.1f}s ===")

    return result
