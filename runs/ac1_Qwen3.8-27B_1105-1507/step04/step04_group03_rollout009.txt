def fast_conv(a):
    """Full self-convolution via FFT."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]


def compute_J(a):
    """Compute J(a) = 2*n*max(conv(a,a))/(sum(a))^2."""
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))


def build_A_sparse(n, a):
    """Build sparse matrix A where A[k,j] = a[k-j] for valid indices."""
    rows = []

            rows.append(k)
            cols.append(j)
            vals.append(a[k - j])
    return csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))

                if y is None:
                    try:
                        y = r.ineqlin
                    except:
                        y = None
                return g, y
    except Exception:
        pass
    return None, None


def lp_mirror_descent_momentum(a, max_iters, deadline, use_momentum=True, initial_t=0.4):
    """LP mirror descent with Nesterov momentum and adaptive step size."""
    a = np.maximum(a.astype(np.float64), 1e-10)
    a = a / a.sum()
    current_J = compute_J(a)
    best_J = current_J
    best_a = a.copy()
    no_improve = 0
    momentum = np.zeros_like(a)
    beta = 0.7  # momentum coefficient
    t = initial_t
    n = len(a)

    # Track previous for Nesterov
    a_prev = a.copy()

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        b = fast_conv(a)
        M = float(b.max())

        # Solve LP for best response
        g, _ = lp_best_response(a, M)
        if g is None:
            break
        g = np.maximum(g, 0.0)
        g_sum = g.sum()
        if g_sum < 1e-10:
            break
        g = g / g_sum

        # Compute direction with momentum
        direction = g - a
        if use_momentum:
            momentum = beta * momentum + direction
            direction = momentum

        # Adaptive line search with backtracking
        best_trial_J = current_J
        best_trial_a = None
        current_t = t

        for ls in range(30):
            a_trial = a + current_t * direction
            a_trial = np.maximum(a_trial, 0.0)
            s = a_trial.sum()
            if s < 1e-10:
                current_t *= 0.5
                continue
            a_trial = a_trial / s
            J_trial = compute_J(a_trial)
            if J_trial < best_trial_J - 1e-15:
                best_trial_J = J_trial
                best_trial_a = a_trial.copy()
                break
            current_t *= 0.6

        if best_trial_a is not None:
            # Update step size: increase if full step worked, decrease otherwise
            if current_t >= t * 0.5:
                t = min(1.0, t * 1.1)
            else:
                t = max(0.01, current_t)

            a_new = best_trial_a
            if use_momentum:
                momentum = beta * momentum
            a = a_new
            current_J = best_trial_J
            if current_J < best_J:
                best_J = current_J
                best_a = a.copy()
                no_improve = 0
            else:
                no_improve += 1
        else:
            no_improve += 1
            t = max(0.005, t * 0.7)
            if use_momentum:
                momentum = beta * momentum

        if no_improve > 60:
            # Reset and try with smaller step
            t = max(0.005, t * 0.5)
            momentum = np.zeros_like(a)
            no_improve = 0
            if t < 0.002:
                break

    return best_a, best_J


def gs_flattening(a, n_gs_iters=5):
    """Gerchberg-Saxton style frequency-domain flattening."""
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0)
    s = a.sum()
    if s < 1e-10:
        return a
    a = a / s

    for _ in range(n_gs_iters):
        size = 1
        while size < 2 * n - 1:
            size <<= 1
        # Zero-pad and compute FFT
        a_padded = np.zeros(size)
        a_padded[:n] = a
        F = np.fft.fft(a_padded).astype(np.complex128)

        # Compute target magnitude: flat spectrum
        # |F(0)| = sum(a) = 1
        # Target: |F(k)| = C for k != 0
        # Parseval: sum |F(k)|^2 = size * sum(a[i]^2)
        energy = size * np.sum(a ** 2)
        residue = energy - abs(F[0]) ** 2
        n_freqs = size - 1
        if residue <= 0:
            break
        C = np.sqrt(residue / n_freqs)

        # Apply target magnitude, keep phase
        F_new = F.copy()
        for k in range(1, size):
            if k >= size - n + 1:
                # Nyquist region - keep as is
                continue
            mag = abs(F[k])
            if mag > 1e-15:
                F_new[k] = C * F[k] / mag
            else:
                F_new[k] = C

        # Return to time domain
        a_new = np.fft.ifft(F_new).real[:n]
        a_new = np.maximum(a_new, 0.0)
        s_new = a_new.sum()
        if s_new < 1e-10:
            break
        a = a_new / s_new

    return a


def generate_diverse_inits(n, rng, parent=None):
    """Generate diverse initial sequences with spectral awareness."""
    inits = []
    t = np.arange(n, dtype=np.float64) / n

    if parent is not None:
        p = resample_seq(parent, n)
        inits.append(p)
        inits.append(np.maximum(p + rng.normal(0, 0.03, n), 0.01))
        inits.append(np.maximum(p + rng.normal(0, 0.08, n), 0.01))

    # Cosine windows
    inits.append(0.5 * (1 + np.cos(2 * np.pi * t)))
    inits.append(0.54 + 0.46 * np.cos(2 * np.pi * t))  # Hamming
    inits.append(0.42 - 0.5 * np.cos(2 * np.pi * t) + 0.08 * np.cos(4 * np.pi * t))  # Hann

    # Multi-frequency
    for f in [2, 3, 5, 7, 11, 13]:
        inits.append(np.maximum(1.0 + 0.25 * np.cos(2 * np.pi * f * t + rng.uniform(0, 2 * np.pi)), 0.05))

    # M-sequence inspired (pseudo-random with smoothing)
    k = max(3, n // 32)
    kernel = np.ones(k) / k
    raw = rng.choice([0.5, 1.5], size=n)
    inits.append(np.maximum(np.convolve(raw, kernel, mode='same'), 0.05))

    # Smooth random
    k2 = max(5, n // 16)
    kernel2 = np.ones(k2) / k2
    smooth = np.convolve(rng.uniform(0.3, 1.0, n), kernel2, mode='same')
    inits.append(np.maximum(smooth, 0.05))

    # Gaussian mixtures
    centers = rng.uniform(0.1, 0.9, 3)
    widths = rng.uniform(0.05, 0.15, 3)
    g_mix = np.zeros(n)
    for c, w in zip(centers, widths):
        g_mix += np.exp(-0.5 * ((t - c) / w) ** 2)
    inits.append(g_mix + 0.1)

    # Flat-spectrum target (try to make DFT flat)
    a_flat = np.ones(n)
    for _ in range(10):
        size = 1
        while size < 2 * n - 1:
            size <<= 1
        a_padded = np.zeros(size)
        a_padded[:n] = a_flat
        F = np.fft.fft(a_padded).astype(np.complex128)
        energy = size * np.sum(a_flat ** 2)
        residue = energy - abs(F[0]) ** 2
        if residue <= 0:
            break
        C = np.sqrt(residue / (size - 1))
        F_new = F.copy()
        for k in range(1, size):
            mag = abs(F[k])
            if mag > 1e-15:
                F_new[k] = C * F[k] / mag
            else:
                F_new[k] = C
        a_flat = np.fft.ifft(F_new).real[:n]
        a_flat = np.maximum(a_flat, 0.01)
        a_flat = a_flat / a_flat.sum()
    inits.append(a_flat)

    result = []
    for a in inits:
        a = np.maximum(a, 0.001)
        s = a.sum()
        if s > 1e-6:
            result.append(a / s)
    return result


def resample_seq(src, dest_n):
    """Resample 1D sequence to new length."""
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)


class ScoreTracker:
    def __init__(self):
        self.best = None
        self.best_score = float('inf')
        self.t_start = time.monotonic()

    def update(self, a):
        a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
        if a.size == 0 or a.sum() < 0.01:
            return
        score = float(evaluate_sequence(a.tolist()))
        if np.isfinite(score) and score < self.best_score:
            self.best = a.copy()
            self.best_score = score
            elapsed = time.monotonic() - self.t_start
            print(f"  [NEW BEST] J={score:.8f} n={a.size} t={elapsed:.0f}s", flush=True)


def lbb_objective(x):
    """Objective for L-BFGS-B: max conv (scale invariant)."""
    ax = np.maximum(x, 0.0)
    s = ax.sum()
    if s < 1e-8:
        return 1e10
    ax = ax / s
    b = fast_conv(ax)
    return float(b.max())


def dual_weighted_subgradient_step(a, max_iters, deadline):
    """Subgradient step using LP dual weights for better direction."""
    a = np.maximum(a.astype(np.float64), 1e-10)
    a = a / a.sum()
    best_M = fast_conv(a).max()
    best_a = a.copy()
    n = len(a)
    momentum = np.zeros_like(a)
    beta = 0.8

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        b = fast_conv(a)
        M = b.max()
        tol = 1e-10 * M
        active = np.where(b >= M - tol)[0]

        if len(active) == 0:
            break

        # Build gradient from active lags
        grad = np.zeros_like(a)
        for k in active:
            lo = max(0, k - n + 1)
            hi = min(n - 1, k)
            for i in range(lo, hi + 1):
                grad[i] += 2.0 * a[k - i]

        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-15:
            break

        # Polyak-like step size
        # Lower bound on optimal M: use 1/(2n) as rough lower bound
        M_lb = 1.0 / (2 * n)
        gap = M - M_lb
        if gap > 1e-12:
            t = min(0.5, gap / (gnorm * gnorm))
        else:
            t = 1.0 / (gnorm * np.sqrt(it + 1.0))

        momentum = beta * momentum + (1 - beta) * grad / gnorm
        a_new = a - t * momentum
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s < 1e-10:
            break
        a_new = a_new / s

        new_M = fast_conv(a_new).max()
        if new_M < best_M:
            best_M = new_M
            best_a = a_new.copy()
        a = a_new

    return best_a


def lp_mirror_descent_v2(a, max_iters, deadline, mode='momentum'):
    """
    Improved LP mirror descent.
    mode: 'momentum' (Nesterov), 'basic' (standard), 'aggressive' (larger steps)
    """
    a = np.maximum(a.astype(np.float64), 1e-10)
    a = a / a.sum()
    current_J = compute_J(a)
    best_J = current_J
    best_a = a.copy()
    no_improve = 0

    if mode == 'aggressive':
        beta = 0.5
        t = 0.6
        max_t = 0.8
    elif mode == 'momentum':
        beta = 0.7
        t = 0.35
        max_t = 0.5
    else:
        beta = 0.0
        t = 0.3
        max_t = 0.4

    momentum = np.zeros_like(a)
    n = len(a)
    step_decay = 0.9995  # very slow decay

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        b = fast_conv(a)
        M = float(b.max())

        # Solve LP
        g, _ = lp_best_response(a, M)
        if g is None:
            break
        g = np.maximum(g, 0.0)
        g_sum = g.sum()
        if g_sum < 1e-10:
            break
        g = g / g_sum

        # Direction
        direction = g - a
        if beta > 0:
            momentum = beta * momentum + direction
            direction = momentum

        # Line search
        best_trial_J = current_J
        best_trial_a = None
        current_t = t

        for ls in range(20):
            a_trial = a + current_t * direction
            a_trial = np.maximum(a_trial, 0.0)
            s = a_trial.sum()
            if s < 1e-10:
                current_t *= 0.5
                continue
            a_trial = a_trial / s
            J_trial = compute_J(a_trial)
            if J_trial < best_trial_J - 1e-15:
                best_trial_J = J_trial
                best_trial_a = a_trial.copy()
                break
            current_t *= 0.55

        if best_trial_a is not None:
            # Adapt step size
            if current_t > t * 0.8:
                t = min(max_t, t * 1.05)
            else:
                t = max(0.01, current_t * 1.1)

            a = best_trial_a
            current_J = best_trial_J
            if beta > 0:
                momentum = beta * momentum

            if current_J < best_J:
                best_J = current_J
                best_a = a.copy()
                no_improve = 0
            else:
                no_improve += 1
        else:
            no_improve += 1
            t = max(0.005, t * 0.7)
            if beta > 0:
                momentum *= beta

        # Very slow decay
        t = max(0.005, t * step_decay)

        if no_improve > 80:
            no_improve = 0
            if beta > 0:
                momentum = np.zeros_like(a)
            t = max(0.01, t * 0.8)
            if t < 0.003:
                break

    return best_a, best_J


def propose_candidate(seed=42, budget_s=1500, **kwargs):
    """
    Improved hybrid optimizer:
    - Phase 1: Multi-start diverse initializations + short LP
    - Phase 2: LP mirror descent with momentum (long)
    - Phase 3: GS flattening + LP refinement
    - Phase 4: Multi-length exploration
    - Phase 5: Very long LP refinement + subgradient polish
    - Phase 6: L-BFGS-B final polish
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + float(budget_s) - 30.0
    tracker = ScoreTracker()

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        if p.ndim == 1 and p.size > 0 and np.all(np.isfinite(p)):
            parent = np.maximum(p, 0.0)
            tracker.update(parent)
            print(f"Parent: n={parent.size}, J={tracker.best_score:.6f}", flush=True)
    except (NameError, AttributeError):
        pass

    if parent is None:
        parent = np.ones(640)
        tracker.update(parent)

    current_best = parent.copy()
    current_best_n = current_best.size

    # === Phase 1: Multi-start (300s) ===
    t0 = time.monotonic()
    print(f"\n{'='*50}", flush=True)
    print(f"Phase 1: Multi-start diverse inits (n={current_best_n})", flush=True)
    print(f"{'='*50}", flush=True)

    n = current_best_n
    inits = generate_diverse_inits(n, rng, parent)
    print(f"  {len(inits)} diverse initializations", flush=True)

    phase1_time_per = max(5, min(30, 250.0 / len(inits)))
    for i, a_init in enumerate(inits):
        if time.monotonic() > deadline - 1100:
            break
        a_init = np.maximum(a_init, 1e-10)
        a_init = a_init / a_init.sum()
        J_init = compute_J(a_init)
        dl = time.monotonic() + phase1_time_per
        a_ref, J_ref = lp_mirror_descent_v2(a_init, 200, dl, mode='momentum')
        tracker.update(a_ref)
        print(f"  init {i}: J_init={J_init:.6f} -> J_ref={J_ref:.6f}", flush=True)
        if J_ref < compute_J(current_best):
            current_best = a_ref.copy()

    print(f"Phase 1 done ({time.monotonic()-t0:.1f}s)", flush=True)

    # === Phase 2: Long LP with momentum (400s) ===
    t0 = time.monotonic()
    print(f"\n{'='*50}", flush=True)
    print(f"Phase 2: Long LP momentum (n={current_best_n})", flush=True)
    print(f"{'='*50}", flush=True)

    a2 = current_best.copy()
    a2 = np.maximum(a2, 1e-10)
    a2 = a2 / a2.sum()
    phase2_dl = min(deadline - 700, time.monotonic() + 400)
    a2, J2 = lp_mirror_descent_v2(a2, 10000, phase2_dl, mode='momentum')
    tracker.update(a2)
    print(f"  J={J2:.8f} ({time.monotonic()-t0:.1f}s)", flush=True)

    if J2 < compute_J(current_best):
        current_best = a2.copy()
        current_best_n = a2.size

    # Try aggressive mode
    if time.monotonic() < deadline - 650:
        t0b = time.monotonic()
        a2b = current_best.copy()
        a2b = np.maximum(a2b, 1e-10)
        a2b = a2b / a2b.sum()
        phase2b_dl = min(deadline - 650, time.monotonic() + 100)
        a2b, J2b = lp_mirror_descent_v2(a2b, 5000, phase2b_dl, mode='aggressive')
        tracker.update(a2b)
        print(f"  Aggressive: J={J2b:.8f} ({time.monotonic()-t0b:.1f}s)", flush=True)
        if J2b < compute_J(current_best):
            current_best = a2b.copy()

    # === Phase 3: GS flattening + LP (200s) ===
    t0 = time.monotonic()
    print(f"\n{'='*50}", flush=True)
    print(f"Phase 3: GS flattening + LP (n={current_best_n})", flush=True)
    print(f"{'='*50}", flush=True)

    phase3_budget = min(200, deadline - time.monotonic() - 450)
    n_gs_rounds = max(3, int(phase3_budget / 40))
    for round_i in range(n_gs_rounds):
        if time.monotonic() > deadline - 450:
            break
        # Apply GS flattening
        a_gs = gs_flattening(current_best, n_gs_iters=3)
        tracker.update(a_gs)

        # Refine with LP
        a_gs = np.maximum(a_gs, 1e-10)
        a_gs = a_gs / a_gs.sum()
        dl_gs = time.monotonic() + max(10, min(30, phase3_budget / (n_gs_rounds - round_i)))
        a_gs, J_gs = lp_mirror_descent_v2(a_gs, 500, dl_gs, mode='momentum')
        tracker.update(a_gs)
        J_cur = compute_J(current_best)
        if J_gs < J_cur:
            current_best = a_gs.copy()
            print(f"  GS round {round_i}: J={J_gs:.8f} (improved)", flush=True)
        else:
            print(f"  GS round {round_i}: J={J_gs:.8f} (no improve, best={J_cur:.8f})", flush=True)

    print(f"Phase 3 done ({time.monotonic()-t0:.1f}s)", flush=True)

    # === Phase 4: Multi-length exploration (250s) ===
    t0 = time.monotonic()
    print(f"\n{'='*50}", flush=True)
    print(f"Phase 4: Multi-length (base n={current_best_n})", flush=True)
    print(f"{'='*50}", flush=True)

    base_n = current_best_n
    candidate_lengths = sorted(set([
        max(256, base_n - 128),
        base_n,
        base_n + 128,
        base_n + 256,
        max(256, base_n // 2 + 128),
        min(2048, base_n * 2),
    ]))
    candidate_lengths = [n for n in candidate_lengths if 256 <= n <= 2560]

    phase4_budget = min(250, deadline - time.monotonic() - 200)
    per_len_time = max(15, phase4_budget / len(candidate_lengths))

    for n_try in candidate_lengths:
        if time.monotonic() > deadline - 200:
            break
        if n_try == base_n:
            continue
        a_try = resample_seq(current_best, n_try)
        a_try = np.maximum(a_try, 1e-10)
        a_try = a_try / a_try.sum()
        dl_try = time.monotonic() + per_len_time
        a_try, J_try = lp_mirror_descent_v2(a_try, 3000, dl_try, mode='momentum')
        tracker.update(a_try)
        print(f"  n={n_try}: J={J_try:.8f}", flush=True)

        if J_try < compute_J(current_best):
            current_best = a_try.copy()
            current_best_n = n_try

    print(f"Phase 4 done ({time.monotonic()-t0:.1f}s)", flush=True)

    # === Phase 5: Very long LP on best (remaining time - 100s) ===
    t0 = time.monotonic()
    print(f"\n{'='*50}", flush=True)
    print(f"Phase 5: Long LP refinement (n={current_best_n})", flush=True)
    print(f"{'='*50}", flush=True)

    a5 = current_best.copy()
    a5 = np.maximum(a5, 1e-10)
    a5 = a5 / a5.sum()
    phase5_dl = min(deadline - 100, time.monotonic() + 400)
    a5, J5 = lp_mirror_descent_v2(a5, 50000, phase5_dl, mode='momentum')
    tracker.update(a5)
    print(f"  J={J5:.8f} ({time.monotonic()-t0:.1f}s)", flush=True)

    if J5 < compute_J(current_best):
        current_best = a5.copy()
        current_best_n = a5.size

    # Subgradient polish
    if time.monotonic() < deadline - 50:
        t0s = time.monotonic()
        a_sub = dual_weighted_subgradient_step(current_best, 200, min(deadline - 30, time.monotonic() + 60))
        tracker.update(a_sub)
        J_sub = compute_J(a_sub)
        if J_sub < compute_J(current_best):
            current_best = a_sub.copy()
        print(f"  Subgradient: J={J_sub:.8f} ({time.monotonic()-t0s:.1f}s)", flush=True)

    # === Phase 6: L-BFGS-B final ===
    if time.monotonic() < deadline - 10 and tracker.best is not None:
        t0 = time.monotonic()
        n_lb = tracker.best.size
        print(f"\n{'='*50}", flush=True)
        print(f"Phase 6: L-BFGS-B (n={n_lb})", flush=True)
        print(f"{'='*50}", flush=True)
        try:
            res = minimize(lbb_objective, tracker.best.astype(np.float64),
                           method='L-BFGS-B', bounds=[(0.0, None)] * n_lb,
                           options={'maxiter': 200, 'max_fun': 50000, 'ftol': 1e-16})
            tracker.update(res.x)
            print(f"  L-BFGS-B: J={2 * n_lb * lbb_objective(res.x) / (np.maximum(res.x,0).sum())**2:.8f}"
                  if np.maximum(res.x, 0).sum() > 0.01 else "  L-BFGS-B: invalid", flush=True)
        except Exception as e:
            print(f"  L-BFGS-B error: {e}", flush=True)
        print(f"  Done ({time.monotonic()-t0:.1f}s)", flush=True)

    # Final short LP on the absolute best
    if time.monotonic() < deadline - 5 and tracker.best is not None:
        a_final = tracker.best.copy()
        a_final = np.maximum(a_final, 1e-10)
        a_final = a_final / a_final.sum()
        a_final, J_final = lp_mirror_descent_v2(a_final, 2000, deadline - 2, mode='momentum')
        tracker.update(a_final)

    # Final decision
    if tracker.best is not None:
        final_a = tracker.best
    else:
        final_a = current_best

    print(f"\n{'='*50}", flush=True)
    print(f"FINAL: J={tracker.best_score:.8f} n={final_a.size} time={time.monotonic()-tracker.t_start:.0f}s", flush=True)
    print(f"{'='*50}", flush=True)

    return [float(x) for x in final_a]
