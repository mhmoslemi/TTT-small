"""
Minimizes J(a) = 2*n*max(convolve(a,a))/(sum(a))² using:
1. Exponentiated Gradient (Mirror Descent on simplex) with Polyak step size

import numpy as np
from scipy.optimize import linprog, minimize
from scipy.sparse import csr_matrix
import time

_A_structure_cache = {}


def fast_conv(a):
    """Full self-convolution via FFT, O(n log n)."""
    n = len(a)
    s = 1
    while s < 2 * n - 1:
        s <<= 1
    fa = np.fft.rfft(a, s)
    return np.fft.irfft(fa * fa, s)[:2 * n - 1]


def compute_J(a):
    """Compute J(a) = 2*n*max(conv(a,a))/(sum(a))^2."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))


def smooth_max_and_weights(b, beta):
    """Compute soft-max value and probability weights."""
    b_max = float(b.max())
    shifted = beta * (b - b_max)
    shifted = np.clip(shifted, -500, 500)
    exp_shifted = np.exp(shifted)
    Z = float(exp_shifted.sum())
    if Z < 1e-300:
        p = np.zeros(len(b))
        p[np.argmax(b)] = 1.0
        return b_max, p
    p = exp_shifted / Z
    g_val = (b_max + np.log(Z)) / beta
    return g_val, p


def J_and_grad(a, beta):
    """Compute J(a) and gradient using smooth max approximation."""
    n = len(a)
    s = float(a.sum())
    if s < 1e-10:
        return 1e10, np.zeros(n)
    b = fast_conv(a)
    g_val, p = smooth_max_and_weights(b, beta)
    J_val = 2.0 * n * g_val / (s * s)
    # grad_m_i = 2 * sum_j a[j] * p[i+j]
    # = 2 * (a_flipped * p)[n-1+i]
    a_flipped = a[::-1]
    c = np.convolve(a_flipped, p, mode='full')
    grad_m = 2.0 * c[n - 1:2 * n - 1]
    grad_J = 2.0 * n * (s * grad_m - 2.0 * g_val) / (s ** 3)
    return J_val, grad_J


def project_simplex(y):
    """Project y onto {a >= 0, sum(a) = 1}."""
    n = len(y)
    if n == 0:
        return y
    u = np.sort(y)[::-1]
    cssv = np.cumsum(u)
    rho = np.where(u * np.arange(1, n + 1) > (cssv - 1))[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(y - theta, 0)


def resample_seq(src, dest_n):
    """Resample 1D sequence to new length via linear interpolation."""
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    result = np.interp(t_new, t_old, src)
    return np.maximum(result, 0.0)


def get_A_structure(n):
    """Get or pre-compute the (row, col) arrays for the constraint matrix."""
    if n not in _A_structure_cache:
        rows = []
        cols = []
        for k in range(2 * n - 1):
            jl = max(0, k - n + 1)
            jh = min(n - 1, k)
            for j in range(jl, jh + 1):
                rows.append(k)
                cols.append(j)
        _A_structure_cache[n] = (np.array(rows, dtype=np.int32),
                                  np.array(cols, dtype=np.int32))
    return _A_structure_cache[n]


def build_A_sparse(a, active_k=None):
    """Build sparse constraint matrix, optionally only for active_k rows."""
    n = len(a)
    rows_arr, cols_arr = get_A_structure(n)
    if active_k is not None:
        mask = np.isin(rows_arr, active_k)
        rows_arr = rows_arr[mask]
        cols_arr = cols_arr[mask]
    vals = a[rows_arr - cols_arr]
    if active_k is not None:
        shape = (len(active_k), n)
        row_idx = np.searchsorted(sorted(active_k), rows_arr)
    else:
        shape = (2 * n - 1, n)
        row_idx = rows_arr
    return csr_matrix((vals, (row_idx, cols_arr)), shape=shape)


def active_set_lp(a, M, top_k=50):
    """
    LP: max sum(g) s.t. (a*g)_k <= M for top_k most binding k, g >= 0.
    Returns normalized g or None.
    """
    n = len(a)
    b = fast_conv(a)
    # Find the top_k positions where (a*a)_k is closest to M
    idx = np.argsort(b)[::-1][:top_k]
    idx = np.sort(idx)
    
    c = -np.ones(n)
    A = build_A_sparse(a, active_k=idx)
    b_ub = np.full(len(idx), M)
    bounds = [(0.0, None)] * n
    try:
        r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
        if r.success and r.x is not None:
            g = np.maximum(r.x, 0.0)
            g_sum = g.sum()
            if g_sum > 1e-10:
                return g / g_sum
    except Exception:
        pass
    return None


def exponentiated_gradient(a, deadline, beta0=5.0, beta1=50.0,
                           max_iters=5000, tracker=None):
    """
    Mirror descent (Exponentiated Gradient) on the simplex.
    Uses Polyak step size based on best-known gap.
    """
    a = np.maximum(a.astype(np.float64), 1e-10)
    a = a / a.sum()
    n = len(a)
    
    J_cur = compute_J(a)
    best_J = J_cur
    best_a = a.copy()
    if tracker is not None:
        tracker.update(best_a)
    
    # Polyak step: eta = (J_cur - J_best) / (||grad||^2)
    # We use a smoothed version with upper bound
    eta_max = 2.0 / n
    eta = eta_max
    
    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        if it % 200 == 0:
            # Anneal beta
            progress = min(1.0, it / (0.5 * max_iters))
            beta = beta0 + (beta1 - beta0) * progress
        else:
            beta = beta0 + (beta1 - beta0) * min(1.0, it / (0.5 * max_iters))
        
        J_val, grad = J_and_grad(a, beta)
        
        # Gradient step in mirror (exponentiated) form
        # a_new_i ∝ a_i * exp(-eta * grad_i)
        grad_norm_sq = float(np.dot(grad, grad))
        if grad_norm_sq < 1e-30:
            break
        
        # Polyak-type step size
        gap = J_cur - best_J
        if gap > 1e-10:
            eta = min(eta_max, max(1e-6, gap / (grad_norm_sq + 1e-10)))
        else:
            eta = eta_max * 0.1
        
        # Exponentiated update
        log_ratio = -eta * grad
        log_ratio = np.clip(log_ratio, -50, 50)
        a_new = a * np.exp(log_ratio)
        a_new = np.maximum(a_new, 1e-15)
        s_new = a_new.sum()
        a_new = a_new / s_new
        
        J_new = compute_J(a_new)
        
        if J_new < J_cur - 1e-14:
            a = a_new
            J_cur = J_new
            if J_new < best_J - 1e-12:
                best_J = J_new
                best_a = a.copy()
                if tracker is not None:
                    tracker.update(best_a)
        else:
            # Halve step size and retry
            for _ in range(5):
                eta *= 0.5
                log_ratio = -eta * grad
                log_ratio = np.clip(log_ratio, -50, 50)
                a_try = a * np.exp(log_ratio)
                a_try = np.maximum(a_try, 1e-15)
                a_try = a_try / a_try.sum()
                J_try = compute_J(a_try)
                if J_try < J_cur - 1e-14:
                    a = a_try
                    J_cur = J_try
                    if J_try < best_J - 1e-12:
                        best_J = J_try
                        best_a = a.copy()
                        if tracker is not None:
                            tracker.update(best_a)
                    break
            else:
                # No progress, slightly perturb
                if it > 500:
                    a = a + 0.01 * np.random.default_rng(it).normal(0, 1, n) * a
                    a = np.maximum(a, 1e-10)
                    a = a / a.sum()

    if tracker is not None:
        tracker.update(best_a)
    return best_a


def nesterov_log_space(a, deadline, betas=None, max_iters=3000):
    """
    Nesterov-accelerated gradient descent in log-space.
    Works with x = log(a), so a = exp(x) / sum(exp(x)) (softmax).
    """
    if betas is None:
        betas = [(1.0, 500), (5.0, 500), (20.0, 500), (50.0, 500)]
    
    a = np.maximum(a.astype(np.float64), 1e-10)
    a = a / a.sum()
    n = len(a)
    
    # Start in log-space
    x = np.log(a)
    x = x - x.mean()  # center for numerical stability
    y = x.copy()  # Nesterov momentum variable
    v = np.zeros(n)
    
    best_J = compute_J(a)
    best_a = a.copy()
    
    for beta, n_iters in betas:
        if time.monotonic() > deadline:
            break
        t = 0
        for it in range(n_iters):
            if time.monotonic() > deadline:
                    break
            t += 1
            # Nesterov step
            x = y
            J_val, grad = J_and_grad(a, beta)
            
            # Convert gradient to log-space: grad_x_i = grad_a_i * a_i * n (for softmax)
            # Actually for a = softmax(x): da_i/dx_j = a_i(δ_ij - a_j)
            # grad_x_i = sum_j grad_a_j * da_j/dx_i = grad_a_i * a_i * (1 - a_i) + ... 
            # Simpler: use chain rule grad_x_i = grad_a_i * a_i - a_i * sum_j grad_a_j * a_j
            grad_x = grad * a - a * np.dot(grad, a)
            grad_x = np.nan_to_num(grad_x, nan=0.0, posinf=0.0, neginf=0.0)
            
            # Nesterov momentum
            mu = 2.0 / (t + 3)  # standard Nesterov momentum
            v = mu * v + (1 - mu) * grad_x
            step = 1.0 / (t + 1)
            y = x - step * v
            
            # Convert back: a = softmax(y)
            y_shifted = y - y.max()
            a_new = np.exp(y_shifted)
            a_new = a_new / a_new.sum()
            
            J_new = compute_J(a_new)
            if J_new < best_J - 1e-12:
                best_J = J_new
                best_a = a_new.copy()
            
            a = a_new
    
    return best_a


def coordinate_descent(a, deadline, max_sweeps=50):
    """
    Coordinate descent: for each coordinate i, find optimal value
    given all others fixed, by 1D minimization.
    """
    a = np.maximum(a.astype(np.float64), 1e-10)
    a = a / a.sum()
    n = len(a)
    best_J = compute_J(a)
    
    for sweep in range(max_sweps):
        if time.monotonic() > deadline:
            break
        improved = False
        for i in range(n):
            if time.monotonic() > deadline:
                break
            a_old_i = a[i]
            # Try a range of values for a[i]
            # The optimal a[i] should be in [0, 2*a_old_i] typically
            # Do a golden section search on a[i]
            lo = 0.0
            hi = 2.5 * a_old_i + 0.001
            gr = (np.sqrt(5.0) - 1.0) / 2.0
            best_val = a_old_i
            best_j = best_J
            
            for _ in range(15):
                t1 = hi - gr * (hi - lo)
                t2 = lo + gr * (hi - lo)
                a[i] = t1
                s = a.sum()
                if s < 1e-10:
                    break
                a_norm = a / s
                J1 = compute_J(a_norm)
                a[i] = t2
                s = a.sum()
                if s < 1e-10:
                    break
                a_norm = a / s
                J2 = compute_J(a_norm)
                if J1 < J2:
                    hi = t2
                else:
                    lo = t1
                a_mid = (lo + hi) / 2.0
                a[i] = a_mid
                s = a.sum()
                if s < 1e-10:
                    break
                a_norm = a / s
                Jm = compute_J(a_norm)
                if Jm < best_j:
                    best_j = Jm
                    best_val = a_mid
            
            a[i] = best_val
            a = a / a.sum()
            if best_j < best_J - 1e-12:
                best_J = best_j
                improved = True
        
        if not improved:
            break
    
    return a


def generate_initializations(n, rng, parent=None):
    """Generate diverse initial sequences of length n."""
    inits = []
    t_arr = np.linspace(0, 1, n, endpoint=False)
    
    if parent is not None and parent.size > 0:
        p = resample_seq(parent, n)
        inits.append(p)
        p2 = p + rng.normal(0, 0.05, n) * np.maximum(p, 0.01)
        inits.append(np.maximum(p2, 0.01))
    
    # Uniform
    inits.append(np.ones(n))
    
    # Random smooth
    noise = rng.normal(0, 1, n)
    from scipy.ndimage import gaussian_filter1d
    smooth_noise = gaussian_filter1d(noise, sigma=n / 32)
    inits.append(1.0 + 0.3 * smooth_noise / (np.std(smooth_noise) + 1e-10))
    
    # Cosine families with varied frequencies
    for freq in [2, 3, 4, 5, 7, 11]:
        phase = rng.uniform(0, 2 * np.pi)
        inits.append(1.0 + 0.3 * np.cos(2.0 * np.pi * freq * t_arr + phase))
    
    # Sum of cosines (irrationally related)
    inits.append(1.0 + 0.2 * np.cos(2 * np.pi * 3.7 * t_arr) + 0.15 * np.cos(2 * np.pi * 7.3 * t_arr + 1.0))
    
    # Gaussian bump
    inits.append(np.exp(-3.0 * (t_arr - 0.5) ** 2) + 0.1)
    
    # Two-peak
    inits.append(0.5 + 0.4 * (np.exp(-8*(t_arr-0.25)**2) + np.exp(-8*(t_arr-0.75)**2)))
    
    # Sine power
    inits.append(np.abs(np.sin(2 * np.pi * 3 * t_arr)) ** 0.5 + 0.2)
    
    # Spectrally flat attempt: use pseudo-random with smoothing
    spec_target = np.ones(n // 2 + 1)
    phases = rng.uniform(0, 2 * np.pi, n // 2 + 1)
    spec = spec_target * np.exp(1j * phases)
    sig = np.fft.irfft(spec * n // 2 + 1, n)
    inits.append(np.maximum(1.0 + 0.2 * sig / (np.std(sig) + 1e-10), 0.05))
    
    # Trapezoidal
    inits.append(np.minimum(4 * (t_arr + 0.25), 1.0) * np.minimum(4 * (1.75 - t_arr), 1.0) + 0.3)
    
    # Number-theoretic: based on quadratic residues (if n is prime-ish)
    qr = np.zeros(n)
    for j in range(n):
        qr[j] = 1.0 if (j * j % (n + 1)) < n // 2 else 0.3
    from scipy.ndimage import gaussian_filter1d
    qr_smooth = gaussian_filter1d(qr, sigma=n / 32)
    inits.append(qr_smooth / qr_smooth.mean())
    
    # Power law
    inits.append((t_arr + 0.1) ** 0.7 + 0.1 * (1 - t_arr) ** 0.7)
    
    return [np.maximum(x, 0.01) for x in inits]

def lbfgs_refine(a, deadline, beta=20.0, max_iter=200):
    """Quick L-BFGS-B refinement using smooth objective."""
    a = np.maximum(a.astype(np.float64), 1e-8)
    n = len(a)
    
    def func(x):
        x2 = np.maximum(x, 0.0)
        f, _ = J_and_grad(x2, beta)
        if not np.isfinite(f):
            return 1e10
        return f
    
    def jac(x):
        x2 = np.maximum(x, 0.0)
        _, g = J_and_grad(x2, beta)
        g = np.nan_to_num(g, nan=0.0, posinf=0.0, neginf=0.0)
        g[x2 < 1e-12] = 0.0
        return g
    
    bounds = [(0.0, None)] * n
    try:
        res = minimize(func, a.copy(), jac=jac, method='L-BFGS-B',
                       bounds=bounds,
                       options={'maxiter': max_iter, 'gtol': 1e-8, 'ftol': 1e-14})
        if res.fun < 1e10:
            new_a = np.maximum(res.x, 1e-10)
            if new_a.sum() > 1e-6:
                return new_a
    except Exception:
        pass
    return a

            a_opt = lbfgs_refine(a0, deadline_p1, beta=10.0, max_iter=max_iter)
            tracker.update(a_opt)
        
        print(f"    n={n} done in {time.monotonic()-t_start:.1f}s, best so far: {tracker.best_score:.8f}", flush=True)
    
    tracker.report("Phase 1 complete")
    
    # === Phase 2: Exponentiated Gradient / Mirror Descent (500s) ===
    if tracker.best is not None and time.monotonic() < deadline:
        t_phase2 = time.monotonic()
        n_best = tracker.best.size
        phase2_budget = min(500.0, deadline - t_phase2)
        deadline_p2 = t_phase2 + phase2_budget
        print(f"\n{'='*50} Phase 2: Exp-Gradient n={n_best} ({phase2_budget:.0f}s) {'='*50}", flush=True)
        
        # Run on best
        a = exponentiated_gradient(tracker.best, deadline_p2,
                                   beta0=5.0, beta1=80.0,
                                   max_iters=10000, tracker=tracker)
        
        # Try at other lengths
        for n_try in [512, 1024, 2048]:
            if time.monotonic() > deadline_p2:
                break
            if n_try == n_best:
                continue
            a_res = resample_seq(tracker.best, n_try)
            a_res = a_res / max(a_res.sum(), 1e-10)
            budget_here = min(120.0, deadline_p2 - time.monotonic())
            deadline_here = time.monotonic() + budget_here
            a_res = exponentiated_gradient(a_res, deadline_here,
                                           beta0=5.0, beta1=50.0,
                                           max_iters=5000, tracker=tracker)
        
        tracker.report(f"Phase 2: {time.monotonic()-t_phase2:.1f}s")
    
    # === Phase 3: Nesterov in log-space (300s) ===
    if tracker.best is None:
        tracker.best = parent if parent is not None else np.ones(512)
        tracker.best_score = compute_J(tracker.best)
    
    if tracker.best is not None and time.monotonic() < deadline:
        t_phase3 = time.monotonic()
        n_best = tracker.best.size
        phase3_budget = min(300.0, deadline - t_phase3)
        deadline_p3 = t_phase3 + phase3_budget
        print(f"\n{'='*50} Phase 3: Nesterov log-space n={n_best} ({phase3_budget:.0f}s) {'='*50}", flush=True)
        
        for restart in range(3):
            if time.monotonic() > deadline_p3:
                break
            if restart == 0:
                a_start = tracker.best.copy()
            else:
                a_start = tracker.best + rng.normal(0, 0.03, n_best) * tracker.best
                a_start = np.maximum(a_start, 1e-6)
                a_start = a_start / a_start.sum()
            
            a_opt = nesterov_log_space(a_start, time.monotonic() + min(100.0, deadline_p3 - time.monotonic()),
                                       betas=[(5.0, 300), (20.0, 300), (50.0, 300)])
            tracker.update(a_opt)
        
        tracker.report(f"Phase 3: {time.monotonic()-t_phase3:.1f}s")
    
    # === Phase 4: Active-set LP refinement (350s) ===
    if tracker.best is not None and time.monotonic() < deadline:
        t_phase4 = time.monotonic()
        n_best = tracker.best.size
        phase4_budget = min(350.0, deadline - t_phase4)
        deadline_p4 = t_phase4 + phase4_budget
        print(f"\n{'='*50} Phase 4: Active-set LP n={n_best} ({phase4_budget:.0f}s) {'='*50}", flush=True)
        
        a = tracker.best.copy().astype(np.float64)
        a = a / a.sum()
        current_J = compute_J(a)
        
        stall_count = 0
        for it in range(5000):
            if time.monotonic() > deadline_p4:
                break
            
            b = fast_conv(a)
            M = float(b.max())
            
            # Solve active-set LP
            g = active_set_lp(a, M, top_k=64)
            if g is None:
                break
            
            # Line search
            best_t = 0.0
            best_J = current_J
            for t_try in [0.3, 0.5, 0.7, 1.0, 0.1, 0.2, 0.8]:
                a_try = (1 - t_try) * a + t_try * g
                a_try = np.maximum(a_try, 0)
                s = a_try.sum()
                if s < 1e-10:
                    continue
                a_try = a_try / s
                J_try = compute_J(a_try)
                if J_try < best_J:
                    best_J = J_try
                    best_t = t_try
            
            if best_t > 0 and best_J < current_J - 1e-12:
                a = (1 - best_t) * a + best_t * g
                a = np.maximum(a, 0)
                a = a / a.sum()
                current_J = best_J
                stall_count = 0
                if current_J < tracker.best_score - 1e-10:
                    tracker.update(a)
            else:
                stall_count += 1
                if stall_count >= 30:
                    # Perturb and continue
                    a = a + rng.normal(0, 0.02, n_best) * a
                    a = np.maximum(a, 1e-6)
                    a = a / a.sum()
                    current_J = compute_J(a)
                    stall_count = 0
        
        tracker.update(a)
        tracker.report(f"Phase 4: {time.monotonic()-t_phase4:.1f}s")
    
    # === Phase 5: Coordinate descent + final polish (200s) ===
    if tracker.best is not None and time.monotonic() < deadline:
        t_phase5 = time.monotonic()
        n_best = tracker.best.size
        phase5_budget = min(200.0, deadline - t_phase5)
        deadline_p5 = t_phase5 + phase5_budget
        print(f"\n{'='*50} Phase 5: Coord Descent + Polish n={n_best} ({phase5_budget:.0f}s) {'='*50}", flush=True)
        
        # Coordinate descent
        a = coordinate_descent(tracker.best, deadline_p5, max_sweeps=30)
        tracker.update(a)
        
        # Final L-BFGS polish with high beta
        remaining = deadline_p5 - time.monotonic()
        if remaining > 20:
            a = lbfgs_refine(tracker.best, time.monotonic() + min(60.0, remaining),
                             beta=100.0, max_iter=300)
            tracker.update(a)
        
        # One more LP pass
        remaining = deadline_p5 - time.monotonic()
        if remaining > 30:
            a = tracker.best.copy().astype(np.float64)
            a = a / a.sum()
            for it in range(50):
                if time.monotonic() > deadline_p5:
                    break
                b = fast_conv(a)
                M = float(b.max())
                g = active_set_lp(a, M, top_k=32)
                if g is None:
                    break
                best_t = 0.0
                best_J = compute_J(a)
                for t_try in [0.3, 0.5, 0.7, 1.0]:
                    a_try = (1 - t_try) * a + t_try * g
                    a_try = np.maximum(a_try, 0)
                    s = a_try.sum()
                    if s < 1e-10:
                        continue
                    a_try = a_try / s
                    J_try = compute_J(a_try)
                    if J_try < best_J:
                        best_J = J_try
                        best_t = t_try
                if best_t > 0:
                    a = (1 - best_t) * a + best_t * g
                    a = np.maximum(a, 0)
                    a = a / a.sum()
                    tracker.update(a)
                else:
                    break
        
        tracker.report(f"Phase 5: {time.monotonic()-t_phase5:.1f}s")
    
    # === Phase 6: Length exploration (remaining time) ===
    if tracker.best is not None and time.monotonic() < deadline:
        t_phase6 = time.monotonic()
        n_best = tracker.best.size
        remaining = deadline - t_phase6
        if remaining > 40:
            print(f"\n{'='*50} Phase 6: Length Exploration ({remaining:.0f}s) {'='*50}", flush=True)
            
            candidate_lengths = sorted(set([
                n_best - 64, n_best - 32, n_best + 32, n_best + 64,
                384, 512, 640, 768, 896, 1024, 1152, 1280, 1536
            ]))
            
            for n_try in candidate_lengths:
                if n_try < 128 or n_try > 2560:
                    continue
                if abs(n_try - n_best) < 16:
                    continue
                if time.monotonic() > deadline - 10:
                    break
                
                a_res = resample_seq(tracker.best, n_try)
                a_res = a_res / max(a_res.sum(), 1e-10)
                budget_here = min(35.0, deadline - time.monotonic() - 5)
                if budget_here < 10:
                    continue
                
                # Quick mirror descent
                a_res = exponentiated_gradient(a_res, time.monotonic() + budget_here,
                                               beta0=5.0, beta1=40.0,
                                               max_iters=2000, tracker=tracker)
            
            tracker.report(f"Phase 6: {time.monotonic()-t_phase6:.1f}s")
    
    # Final fallback
    if tracker.best is None:
        tracker.best = np.ones(512)
        tracker.best_score = compute_J(tracker.best)
        print(f"  Using fallback: J={tracker.best_score:.8f}", flush=True)
    
    # One final evaluation to confirm
    final_score = float(evaluate_sequence(tracker.best.tolist()))
    if not np.isfinite(final_score):
        tracker.best = np.ones(256)
        final_score = float(evaluate_sequence(tracker.best.tolist()))
    
    print(f"\n{'='*50} FINAL RESULT {'='*50}", flush=True)
    print(f"J={final_score:.8f} n={tracker.best.size} "
          f"time={time.monotonic()-tracker.t_start:.1f}s", flush=True)
    
    result = np.clip(tracker.best, 0.0, 1000.0)
    if result.sum() < 0.01:
        result = np.ones(128)
    
    return [float(x) for x in result]
