Hybrid optimizer: Smoothed objective (log-sum-exp) with L-BFGS-B,
LP mirror descent, subgradient polishing, multi-length search.

import numpy as np
from scipy.optimize import minimize, linprog
from scipy.sparse import csr_matrix
from scipy.signal import fftconvolve
import time

def fast_conv(a):
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]

def compute_J(a):
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))

def smooth_grad_fft(a, T):
    """Compute smoothed objective and gradient using FFT for h computation."""
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-10:
        return 1e10, np.zeros_like(a)
    b = a / s
    n = len(b)
    
    # Autocorrelation
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(b, size)
    c = np.fft.irfft(fa * fa, size)[:2*n-1]
    
    # Smoothed objective
    c_max = c.max()
    c_shifted = c - c_max
    log_S = c_max / T + np.log(np.sum(np.exp(c_shifted / T)))
    f = T * log_S
    
    # Softmax weights
    w = np.exp(c_shifted / T) / np.exp(log_S - c_max / T)
    # Simpler: w = np.exp(c_shifted / T) / np.sum(np.exp(c_shifted / T))
    S_val = np.sum(np.exp(c_shifted / T))
    w = np.exp(c_shifted / T) / S_val
    
    # Compute h_j = sum_{i=0}^{n-1} w_{j+i} * b_i using FFT
    # This is the cross-correlation of w and b at lag j
    # h = ifft(fft(w, L) * conj(fft(b, L)))[:n]
    L = 1
    while L < 3 * n - 1:
        L <<= 1
    W = np.fft.rfft(w, L)
    B = np.fft.rfft(b, L)
    h_full = np.fft.irfft(W * np.conj(B), L)
    h = h_full[:n]
    
    # Gradient: 2 * (h - E_c) / s where E_c = dot(w, c)
    E_c = np.dot(w, c)
    grad = 2.0 * (h - E_c) / s
    
    return f, grad

"""
Hybrid optimizer for minimizing J(a) = 2*n*max(conv(a,a))/sum(a)^2.
Uses smoothed log-sum-exp objective with gradient descent + LP mirror descent,
multi-length search, and subgradient polishing.
"""

import numpy as np
from scipy.optimize import minimize, linprog
from scipy.sparse import csr_matrix
import time


def fast_conv(a):
    """Full self-convolution via FFT."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]


def compute_J(a):
    """Compute J(a) = 2*n*max(conv(a,a))/(sum(a))^2."""
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))


def smooth_obj_grad(x, T):
    """Compute smoothed objective f_T(x) = T*log(sum exp(c_k/T)) and its gradient.
    Uses FFT for the cross-correlation h_j = sum_i w_{j+i} * a_i."""
    a = np.maximum(x, 0.0)
    s = a.sum()
    if s < 1e-10:
        return 1e10, np.zeros_like(x)
    b = a / s
    n = len(b)

    # Autocorrelation
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(b, size)
    c = np.fft.irfft(fa * fa, size)[:2 * n - 1]

    # Numerically stable softmax
    c_max = c.max()
    c_shifted = c - c_max
    exp_vals = np.exp(c_shifted / T)
    S_val = exp_vals.sum()
    w = exp_vals / S_val
    f = c_max + T * np.log(S_val)

    # Cross-correlation h_j = sum_i w_{j+i} * b_i via FFT
    L = 1
    while L < 3 * n - 1:
        L <<= 1
    W = np.fft.rfft(w, L)
    B = np.fft.rfft(b, L)
    h_full = np.fft.irfft(W * np.conj(B), L)
    h = h_full[:n]

    # Gradient: 2*(h - E_c)/s where E_c = dot(w, c)
    E_c = np.dot(w, c)
    grad = 2.0 * (h - E_c) / s

    return f, grad


def smooth_gradient_descent(a0, deadline, max_time=300.0):

    a = np.maximum(a0.astype(np.float64), 1e-10)

    momentum = np.zeros_like(a)
    beta = 0.9
    it_global = 0

def build_A_sparse(n, a):
    """Build sparse matrix A where A[k,j] = a[k-j] for valid j."""
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        j_min = max(0, k - n + 1)
        j_max = min(n - 1, k)
        for j in range(j_min, j_max + 1):
            rows.append(k)
            cols.append(j)
            vals.append(a[k - j])
    return csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))


def lp_best_response(a, M):
    """Solve LP: max sum(g) s.t. A@g <= M*1, g >= 0."""
    n = len(a)
    A = build_A_sparse(n, a)
    c_obj = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    try:
        r = linprog(c_obj, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
        if r.success and r.x is not None:
            g = np.maximum(r.x, 0)
            if g.sum() > 1e-15:
                return g
    except Exception:
        pass
    return None

def subgradient_polish(a, max_iters, deadline):
    """Subgradient descent with momentum on the max-conv objective."""
    a = np.maximum(a.astype(np.float64), 1e-10)
    a = a / a.sum()

    best_a = a.copy()

        active = np.where(b >= M - tol)[0]

        grad = np.zeros_like(a)
        n = len(a)
        for k in active:
            lo = max(0, k - n + 1)
            hi = min(n - 1, k)
            for i in range(lo, hi + 1):
                grad[i] += 2.0 * a[k - i]

        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-15:

def resample_seq(src, dest_n):
    """Resample 1D sequence to new length."""
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)


def lbb_on_smooth(a0, T, maxiter=500):
    """Run L-BFGS-B on the smoothed objective at fixed T."""
    n = len(a0)

    def obj_fn(x):
        f, _ = smooth_obj_grad(x, T)
        return f

    def grad_fn(x):
        _, g = smooth_obj_grad(x, T)
        return g

    try:
        res = minimize(obj_fn, a0, jac=grad_fn, method='L-BFGS-B',
                       bounds=[(0.0, None)] * n,
                       options={'maxiter': maxiter, 'max_fun': 10000,
                                'ftol': 1e-16, 'gtol': 1e-12})
        if res.fun < 1e10:
            a_new = np.maximum(res.x, 0.0)
            s = a_new.sum()
            if s > 1e-10:
                return a_new / s
    except Exception:
        pass
    return a0

        print(f"Phase 4: Alt length n={alt_n}", flush=True)
        print(f"{'='*50}", flush=True)

        if alt_n != parent_n:
            a4 = resample_seq(current_best, alt_n)
            a4 = np.maximum(a4, 1e-10)
            a4 = a4 / a4.sum()

            p4a_dl = min(deadline - 300, t_phase + 100)
            a4, J4a = smooth_gradient_descent(a4, p4a_dl, max_time=90)
            update_best(a4)
            print(f"  Smooth GD n={alt_n}: J={J4a:.8f}", flush=True)

            if time.monotonic() < deadline - 280:
                c4 = fast_conv(a4)
                max_c4 = c4.max()
                for T_frac in [0.1, 0.03, 0.01]:
                    if time.monotonic() > deadline - 260:
                        break
                    T4 = T_frac * max_c4
                    a4 = lbb_on_smooth(a4, T4, maxiter=200)
                    J4 = compute_J(a4)
                    update_best(a4)
                    print(f"  L-BFGS T={T_frac}: J={J4:.8f}", flush=True)
                    if J4 < current_best_J:
                        current_best = a4.copy()
                        current_best_J = J4
                    c4 = fast_conv(a4)
                    max_c4 = max(c4.max(), 1e-10)

            if time.monotonic() < deadline - 240:
                p4b_dl = min(deadline - 220, time.monotonic() + 80)
                a4, J4b = lp_mirror_descent(a4, 300, p4b_dl, use_multiplicative=False)
                update_best(a4)
                print(f"  LP n={alt_n}: J={J4b:.8f}", flush=True)
                if J4b < current_best_J:
                    current_best = a4.copy()
                    current_best_J = J4b

        print(f"  Phase 4 done ({time.monotonic()-t_phase:.1f}s)", flush=True)

    # === Phase 5: LP multiplicative on best ===
    if time.monotonic() < deadline - 200:
        t_phase = time.monotonic()
        print(f"\n{'='*50}", flush=True)
        print(f"Phase 5: LP Mirror (multiplicative, n={current_best.size})", flush=True)
        print(f"{'='*50}", flush=True)

        p5_dl = min(deadline - 180, t_phase + 150)
        a5 = current_best.copy()
        a5, J5 = lp_mirror_descent(a5, 400, p5_dl, use_multiplicative=True)
        update_best(a5)
        print(f"  LP Mult: J={J5:.8f} ({time.monotonic()-t_phase:.1f}s)", flush=True)
        if J5 < current_best_J:
            current_best = a5.copy()
            current_best_J = J5

    # === Phase 6: Long LP refinement ===
    if time.monotonic() < deadline - 80:
        t_phase = time.monotonic()
        print(f"\n{'='*50}", flush=True)
        print(f"Phase 6: Long LP (n={current_best.size})", flush=True)
        print(f"{'='*50}", flush=True)

        a6 = current_best.copy()
        a6, J6 = lp_mirror_descent(a6, 10000, deadline - 60, use_multiplicative=False)
        update_best(a6)
        print(f"  Long LP: J={J6:.8f} ({time.monotonic()-t_phase:.1f}s)", flush=True)
        if J6 < current_best_J:
            current_best = a6.copy()
            current_best_J = J6

    # === Phase 7: Final smooth gradient polish ===
    if time.monotonic() < deadline - 20:
        t_phase = time.monotonic()
        print(f"\n{'='*50}", flush=True)
        print(f"Phase 7: Final polish (n={current_best.size})", flush=True)
        print(f"{'='*50}", flush=True)

        a7 = current_best.copy()
        a7, J7 = smooth_gradient_descent(a7, deadline - 5, max_time=60)
        update_best(a7)
        J7 = compute_J(a7)
        print(f"  Final polish: J={J7:.8f} ({time.monotonic()-t_phase:.1f}s)", flush=True)

    # Final
    if best_seq is not None:
        result = best_seq
    else:
        result = current_best

    # Scale to reasonable range
    result = np.maximum(result, 0.0)
    s = result.sum()
    if s > 0:
        result = result / s * 1.0  # normalize to sum=1

    print(f"\n{'='*50}", flush=True)
    print(f"FINAL: score={best_score:.8f} n={result.size} "
          f"J_internal={compute_J(result):.8f} time={time.monotonic()-t_start:.0f}s",
          flush=True)
    print(f"{'='*50}", flush=True)

    return [float(x) for x in result]
