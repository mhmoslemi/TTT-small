"""
Optimized LP mirror descent with precomputed sparsity, adaptive annealing step sizes,
multi-length exploration, and periodic perturbation escapes to minimize
J(a) = 2*n*max(conv(a,a))/sum(a)^2 for non-negative sequences.

import numpy as np
from scipy.optimize import linprog
from scipy.sparse import csr_matrix
import time
import math


def fast_conv(a):
    """Full self-convolution via FFT."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]


def compute_J(a):
    """Compute J(a) = 2*n*max(conv(a,a))/(sum(a))^2."""
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))


def build_conv_pattern(n):
    """Precompute sparsity pattern indices for convolution matrix."""
    rows = []
    cols = []
    for k in range(2 * n - 1):
        j_min = max(0, k - n + 1)
        j_max = min(n - 1, k)
        for j in range(j_min, j_max + 1):
            rows.append(k)
            cols.append(j)
    return np.array(rows, dtype=np.int32), np.array(cols, dtype=np.int32)


def make_A(n, a, pattern):
    """Build sparse convolution matrix A where A[k,j] = a[k-j]."""
    row_idx, col_idx = pattern
    vals = a[row_idx - col_idx]
    return csr_matrix((vals, (row_idx, col_idx)), shape=(2 * n - 1, n))


def lp_solve(a, M, pattern):
    """Solve LP: max sum(g) s.t. A@g <= M, g >= 0."""
    n = len(a)
    A = make_A(n, a, pattern)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    try:
        r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
        if r.success and r.x is not None:
            g = np.maximum(r.x, 0)
            if g.sum() > 1e-15:
                return g
    except Exception:
        pass
    return None


def lp_mirror_descent_v2(a, pattern, max_iters, deadline, mode='linear', base_t=0.3):
    """
    LP mirror descent with adaptive step size and annealing.
    mode: 'linear' for convex combo, 'multiplicative' for geometric mean.

    current_J = compute_J(a)

def resample_seq(src, dest_n):
    """Resample 1D sequence to new length."""
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)


def perturb_sequence(a, rng, strength=0.05):
    """Apply a structured perturbation and renormalize."""
    n = len(a)
    t = np.arange(n, dtype=np.float64) / n
    ptype = rng.integers(0, 5)

    if ptype == 0:  # Cosine modulation
        freq = rng.uniform(1, 6)
        amp = rng.uniform(0.02, strength)
        phase = rng.uniform(0, 2 * np.pi)
        a_p = a * (1 + amp * np.cos(2 * np.pi * freq * t + phase))
    elif ptype == 1:  # Smooth random
        k = max(3, n // 20)
        noise = np.convolve(rng.normal(0, strength, n), np.ones(k)/k, mode='same')
        a_p = a + noise
    elif ptype == 2:  # Shift
        s = rng.integers(1, max(2, n // 15))
        a_p = np.roll(a, s)
    elif ptype == 3:  # Taper
        alpha = rng.uniform(0.6, 0.95)
        edge = max(1, int(n * (1 - alpha) / 2))
        window = np.ones(n)
        for i in range(edge):
            frac = (i + 1) / (edge + 1)
            window[i] = frac
            window[n - 1 - i] = frac
        a_p = a * window
    else:  # Gaussian blend
        center = rng.uniform(0.2, 0.8)
        width = rng.uniform(0.05, 0.2)
        gauss = np.exp(-0.5 * ((t - center) / width) ** 2)
        a_p = a * (1 - strength) + gauss * strength

    a_p = np.maximum(a_p, 0.001)
    s = a_p.sum()
    if s > 1e-8:
        a_p = a_p / s
    return a_p

    pat = get_pattern(best_n)
    a1 = best_a.copy()
    a1 = np.maximum(a1, 1e-10)
    a1 = a1 / a1.sum()
    a1, J1 = lp_mirror_descent_v2(a1, pat, 150, time.monotonic() + 60, mode='linear', base_t=0.3)
    score_tracker_update(tracker, a1)
    print(f"  Linear: J={J1:.8f} ({time.monotonic()-t0:.1f}s)", flush=True)

    if J1 < best_J:
        best_a, best_J = a1.copy(), J1

    # Try multiplicative mode too
    if time.monotonic() < deadline - 1000:
        a1m = best_a.copy()
        a1m = np.maximum(a1m, 1e-10)
        a1m = a1m / a1m.sum()
        a1m, J1m = lp_mirror_descent_v2(a1m, pat, 100, time.monotonic() + 40, mode='multiplicative', base_t=0.2)
        score_tracker_update(tracker, a1m)
        print(f"  Multiplicative: J={J1m:.8f}", flush=True)
        if J1m < best_J:
            best_a, best_J = a1m.copy(), J1m

    # === Phase 2: Multi-length sweep ===
    t0 = time.monotonic()
    print(f"\n{'='*60}", flush=True)
    print(f"Phase 2: Multi-length sweep (base n={best_n})", flush=True)
    print(f"{'='*60}", flush=True)

    # Generate candidate lengths
    base_n = best_n
    length_candidates = sorted(set([
        max(128, base_n - 128),
        max(128, base_n - 64),
        base_n + 64,
        base_n + 128,
        max(128, int(base_n * 0.75)),
        int(base_n * 1.25),
        int(base_n * 1.5) if base_n * 1.5 <= 2048 else base_n + 192,
    ]))
    length_candidates = [n for n in length_candidates if 128 <= n <= 2048 and n != base_n]

    time_per_len = max(20, min(60, (deadline - 800 - time.monotonic()) / max(1, len(length_candidates))))

    for n_try in length_candidates:
        if time.monotonic() > deadline - 700:
            break
        a_try = resample_seq(best_a, n_try)
        a_try = np.maximum(a_try, 1e-10)
        a_try = a_try / a_try.sum()
        pat_try = get_pattern(n_try)
        a_try, J_try = lp_mirror_descent_v2(a_try, pat_try, 100, time.monotonic() + time_per_len,
                                             mode='linear', base_t=0.3)
        score_tracker_update(tracker, a_try)
        print(f"  n={n_try}: J={J_try:.8f}", flush=True)
        if J_try < best_J:
            best_a, best_n, best_J = a_try.copy(), n_try, J_try
            print(f"    -> New best!", flush=True)

    print(f"Phase 2 done ({time.monotonic()-t0:.1f}s), best n={best_n} J={best_J:.8f}", flush=True)

    # === Phase 3: Extended LP with annealing on best ===
    t0 = time.monotonic()
    print(f"\n{'='*60}", flush=True)
    print(f"Phase 3: Extended LP (n={best_n})", flush=True)
    print(f"{'='*60}", flush=True)

    pat_best = get_pattern(best_n)
    a3 = best_a.copy()
    a3 = np.maximum(a3, 1e-10)
    a3 = a3 / a3.sum()

    # Run with linear mode
    a3, J3 = lp_mirror_descent_v2(a3, pat_best, 5000, min(deadline - 300, time.monotonic() + 500),
                                    mode='linear', base_t=0.35)
    score_tracker_update(tracker, a3)
    print(f"  Linear extended: J={J3:.8f} ({time.monotonic()-t0:.1f}s)", flush=True)
    if J3 < best_J:
        best_a, best_J = a3.copy(), J3

    # Run with multiplicative mode from same starting point
    if time.monotonic() < deadline - 250:
        a3m = best_a.copy()
        a3m = np.maximum(a3m, 1e-10)
        a3m = a3m / a3m.sum()
        a3m, J3m = lp_mirror_descent_v2(a3m, pat_best, 2000, min(deadline - 200, time.monotonic() + 200),
                                         mode='multiplicative', base_t=0.25)
        score_tracker_update(tracker, a3m)
        print(f"  Mult extended: J={J3m:.8f}", flush=True)
        if J3m < best_J:
            best_a, best_J = a3m.copy(), J3m

    # === Phase 4: Perturbation escape attempts ===
    t0 = time.monotonic()
    print(f"\n{'='*60}", flush=True)
    print(f"Phase 4: Perturbation escapes (n={best_n})", flush=True)
    print(f"{'='*60}", flush=True)

    n_escapes = 12
    per_escape = max(15, min(50, (deadline - 200 - time.monotonic()) / n_escapes))
    strengths = [0.03, 0.05, 0.08, 0.12]

    for i in range(n_escapes):
        if time.monotonic() > deadline - 150:
            break
        strength = strengths[i % len(strengths)] * (1 + i * 0.1)
        a_pert = perturb_sequence(best_a, rng, strength=min(strength, 0.2))
        a_pert = np.maximum(a_pert, 1e-10)
        a_pert = a_pert / a_pert.sum()
        a_pert, J_pert = lp_mirror_descent_v2(a_pert, pat_best, 80, time.monotonic() + per_escape,
                                               mode='linear', base_t=0.3)
        score_tracker_update(tracker, a_pert)
        if J_pert < best_J:
            best_a, best_J = a_pert.copy(), J_pert
            print(f"  Escape {i}: J={J_pert:.8f} (improved!)", flush=True)
        else:
            print(f"  Escape {i}: J={J_pert:.8f} (no improvement)", flush=True)

    # === Phase 5: Try different n from best for final refinement ===
    if time.monotonic() < deadline - 100:
        t0 = time.monotonic()
        print(f"\n{'='*60}", flush=True)
        print(f"Phase 5: Final length refinement", flush=True)
        print(f"{'='*60}", flush=True)

        # Try n near best_n
        for delta in [-32, 32, 64]:
            n_try = best_n + delta
            if n_try < 128 or n_try > 2048 or time.monotonic() > deadline - 80:
                continue
            a_try = resample_seq(best_a, n_try)
            a_try = np.maximum(a_try, 1e-10)
            a_try = a_try / a_try.sum()
            pat_try = get_pattern(n_try)
            budget = min(60, max(20, (deadline - 80 - time.monotonic()) / 2))
            a_try, J_try = lp_mirror_descent_v2(a_try, pat_try, 500, time.monotonic() + budget,
                                                 mode='linear', base_t=0.3)
            score_tracker_update(tracker, a_try)
            print(f"  n={n_try}: J={J_try:.8f}", flush=True)
            if J_try < best_J:
                best_a, best_n, best_J = a_try.copy(), n_try, J_try
                pat_best = get_pattern(best_n)

        print(f"Phase 5 done ({time.monotonic()-t0:.1f}s)", flush=True)

    # === Phase 6: Final long polish ===
    if time.monotonic() < deadline - 30:
        t0 = time.monotonic()
        print(f"\n{'='*60}", flush=True)
        print(f"Phase 6: Final polish (n={best_n})", flush=True)
        print(f"{'='*60}", flush=True)

        pat_final = get_pattern(best_n)
        a_final = best_a.copy()
        a_final = np.maximum(a_final, 1e-10)
        a_final = a_final / a_final.sum()
        a_final, J_final = lp_mirror_descent_v2(a_final, pat_final, 10000, deadline - 15,
                                                  mode='linear', base_t=0.25)
        score_tracker_update(tracker, a_final)
        print(f"  Linear polish: J={J_final:.8f} ({time.monotonic()-t0:.1f}s)", flush=True)
        if J_final < best_J:
            best_a, best_J = a_final.copy(), J_final

        # One more multiplicative pass
        if time.monotonic() < deadline - 10:
            a_final2 = best_a.copy()
            a_final2 = np.maximum(a_final2, 1e-10)
            a_final2 = a_final2 / a_final2.sum()
            a_final2, J_final2 = lp_mirror_descent_v2(a_final2, pat_final, 2000, deadline - 5,
                                                       mode='multiplicative', base_t=0.2)
            score_tracker_update(tracker, a_final2)
            if J_final2 < best_J:
                best_a, best_J = a_final2.copy(), J_final2
                print(f"  Mult polish: J={J_final2:.8f}", flush=True)

    # Final decision
    if tracker['best'] is not None:
        final_a = tracker['best']
        final_score = tracker['best_score']
    else:
        final_a = best_a
        final_score = best_J
        score_tracker_update(tracker, final_a)
        final_score = tracker['best_score']

    print(f"\n{'='*60}", flush=True)
    print(f"FINAL: J={final_score:.8f} n={final_a.size} time={time.monotonic()-tracker['t_start']:.0f}s", flush=True)
    print(f"{'='*60}", flush=True)

    # Ensure valid output
    final_a = np.clip(final_a, 0.0, 1000.0)
    return [float(x) for x in final_a]
