"""
Minimizes J(a) = 2*n*max(convolve(a,a))/(sum(a))^2 using:
1. Top-k active-set LP mirror descent (faster LP with only peak constraints)
2. Vectorized coordinate descent with precomputed prefix/suffix max
3. Multi-length exploration with resampling
4. Perturbation restarts and adaptive step sizes
5. Momentum-augmented LP iterations for faster convergence
"""

import numpy as np
from scipy.optimize import linprog
from scipy.sparse import coo_matrix
import time


def fast_conv(a):
    """Full linear self-convolution via FFT."""
    n = len(a)
    s = 1
    while s < 2 * n - 1:
        s <<= 1
    fa = np.fft.rfft(a, s)
    return np.fft.irfft(fa * fa, s)[:2 * n - 1]


def compute_J_local(a):
    """Compute J locally (assume a is non-negative)."""
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))


def build_lp_matrix(active, a):
    """Build sparse matrix for LP constraints: (a*g)[m] for m in active."""
    n = len(a)
    rows = []
    cols = []
    vals = []
    for idx, m in enumerate(active):
        i_lo = max(0, m - n + 1)
        i_hi = min(n - 1, m)
        for i in range(i_lo, i_hi + 1):
            rows.append(idx)
            cols.append(i)
            vals.append(a[m - i])
    if not rows:
        return coo_matrix((len(active), n), dtype=np.float64)
    return coo_matrix((vals, (rows, cols)), shape=(len(active), n)).tocsr()


def lp_direction(a, active, M):
    """Solve max sum(g) s.t. (a*g)[k] <= M for k in active, g >= 0."""
    n = len(a)
    A = build_lp_matrix(active, a)
    c = -np.ones(n)
    b_ub = np.full(len(active), M)
    bounds = [(0.0, None)] * n
    res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                  options={'presolve': True, 'time_limit': 5.0})
    if res.success and res.x is not None:
        return np.maximum(res.x, 0.0)
    return None


def cd_one_coord(a, i, b, prefix_max, suffix_max):
    """Optimize coordinate i via golden section. Returns optimal value."""
    n = len(a)
    ai = a[i]
    S = a.sum()

    k_lo = max(0, i)
    k_hi = min(2 * n - 2, i + n - 1)

    # Precompute b_max_unchanged using prefix/suffix
    if k_lo == 0 and k_hi == 2 * n - 2:
        b_max_unch = 0.0
    elif k_lo == 0:
        b_max_unch = suffix_max[k_hi + 1]
    elif k_hi == 2 * n - 2:
        b_max_unch = prefix_max[k_lo - 1]
    else:
        b_max_unch = max(prefix_max[k_lo - 1], suffix_max[k_hi + 1])

    # Build vectors for changed indices
    ks = np.arange(k_lo, k_hi + 1)
    kis = ks - i
    valid = (kis >= 0) & (kis < n)

    if not valid.any():
        # Nothing changes, return current
        return ai

    ks_v = ks[valid]
    kis_v = kis[valid]

    quad_mask = (kis_v == i)
    lin_mask = (kis_v != i)

    const_l = np.array([])
    coef_l = np.array([])
    quad_const = None

    if lin_mask.any():
        const_l = b[ks_v[lin_mask]] - 2.0 * ai * a[kis_v[lin_mask]]
        coef_l = 2.0 * a[kis_v[lin_mask]]

    if quad_mask.any():
        kq = ks_v[quad_mask][0]
        quad_const = b[kq] - ai * ai

    def obj_func(x):
        S_new = S - ai + x
        if S_new < 1e-15:
            return 1e10
        max_c = b_max_unch
        if len(const_l) > 0:
            v = (const_l + coef_l * x).max()
            if v > max_c:
                max_c = v
        if quad_const is not None:
            vq = x * x + quad_const
            if vq > max_c:
                max_c = vq
        return 2.0 * n * max_c / (S_new * S_new)

    # Golden section search
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    lo = 0.0
    hi = max(3.0 * ai + 0.01, 3.0 * S / n + 0.01)

    cm = hi - gr * (hi - lo)
    dm = lo + gr * (hi - lo)
    fc = obj_func(cm)
    fd = obj_func(dm)

    for _ in range(45):
        if fc < fd:
            hi = dm
            dm = cm
            fd = fc
            cm = hi - gr * (hi - lo)
            fc = obj_func(cm)
        else:
            lo = cm
            cm = dm
            fc = fd
            dm = lo + gr * (hi - lo)
            fd = obj_func(dm)

    return max(0.0, (lo + hi) / 2.0)


def cd_pass(a, max_coords=None):
    """One full pass of coordinate descent. Returns (a, new_J)."""
    a = np.maximum(a.astype(np.float64), 0.0)
    n = len(a)
    S = a.sum()
    if S < 1e-10:
        return a

    b = fast_conv(a)
    prefix_max = np.maximum.accumulate(b)
    suffix_max = np.maximum.accumulate(b[::-1])[::-1]

    indices = list(range(n))
    np.random.shuffle(indices)
    if max_coords is not None:
        indices = indices[:min(max_coords, n)]

    for i in indices:
        new_val = cd_one_coord(a, i, b, prefix_max, suffix_max)
        if abs(new_val - a[i]) > 1e-12:
            old_val = a[i]
            a[i] = new_val
            S = S - old_val + new_val
            # Update b incrementally
            k_lo = max(0, i)
            k_hi = min(2 * n - 2, i + n - 1)
            for k in range(k_lo, k_hi + 1):
                ki = k - i
                if 0 <= ki < n:
                    if ki == i:
                        b[k] = b[k] - old_val * old_val + new_val * new_val
                    else:
                        b[k] = b[k] - 2.0 * old_val * a[ki] + 2.0 * new_val * a[ki]
            # Update prefix/suffix (expensive, do at end)

    # Recompute prefix/suffix after all updates
    prefix_max = np.maximum.accumulate(b)
    suffix_max = np.maximum.accumulate(b[::-1])[::-1]

    # Verify
    b_check = fast_conv(a)
    new_J = 2.0 * n * b_check.max() / (S * S) if S > 0 else float('inf')
    return a, new_J


def resample_seq(src, dest_n):
    """Resample 1D sequence to new length via linear interpolation."""
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)


def generate_smooth_init(n, rng, style=0):
    """Generate a smooth initial sequence of length n."""
    t = np.linspace(0, 1, n, endpoint=False)
    if style == 0:
        return np.ones(n)
    elif style == 1:
        return 0.5 + 0.5 * np.cos(2 * np.pi * t) + 0.01
    elif style == 2:
        return np.exp(-4.0 * (t - 0.5) ** 2) + 0.01
    elif style == 3:
        return 1.0 - 0.5 * np.abs(2 * t - 1.0) + 0.01
    elif style == 4:
        return 1.0 + 0.3 * np.cos(4 * np.pi * t + rng.uniform(0, 6.28)) + 0.01
    elif style == 5:
        return np.maximum(0.5 * np.exp(-8 * (t - 0.3) ** 2) + 0.5 * np.exp(-8 * (t - 0.7) ** 2) + 0.01, 0.0)
    elif style == 6:
        return np.abs(np.sin(np.pi * t)) + 0.01
    elif style == 7:
        return 0.5 + 0.3 * np.cos(2 * np.pi * 3 * t) + 0.2 * np.cos(2 * np.pi * 7 * t) + 0.01
    else:
        return np.ones(n) * (0.5 + rng.uniform(0, 0.5))

def run_lp_optimization(a, deadline, tracker, max_lp_iters=5000, top_k=5,
                        tighten_rate=0.999, step_decay=0.9995, momentum=0.0):
    """LP-based mirror descent optimization with top-k active constraints."""
    a = np.maximum(a.astype(np.float64), 1e-8)
    a = a / a.sum()
    n = len(a)

    tracker.update(a)
    current_J = compute_J_local(a)
    a_prev = a.copy()
    stall = 0
    t = 0.3  # initial step size

    for it in range(max_lp_iters):
        if time.monotonic() > deadline:
            break

        b = fast_conv(a)
        M = b.max()
        # Get top-k active indices
        k_use = min(top_k, 2 * n - 1)
        top_idx = np.argsort(b)[-k_use:]
        top_idx = top_idx[np.argsort(b[top_idx])]  # sort by value
        M_active = b[top_idx].max()

        # Solve LP with slightly tightened bound
        M_bound = M * tighten_rate
        g = lp_direction(a, top_idx, M_bound)

        if g is None:
            # Infeasible, try looser bound
            M_bound = M * 0.9999
            g = lp_direction(a, top_idx, M_bound)

        if g is None or g.sum() < 1e-10:
            stall += 1
            if stall > 50:
                break
            continue

        g = g / g.sum()

        # Try multiple step sizes
        base_t = max(0.01, t)
        candidates = [base_t, base_t * 0.5, base_t * 1.5, base_t * 0.25]
        if momentum > 0.0:
            directed = a + momentum * (a - a_prev)
            directed = np.maximum(directed, 0.0)
            if directed.sum() > 1e-10:
                directed = directed / directed.sum()
                candidates.append(momentum)  # special case handled below

        best_a = None
        best_J = current_J

        for ct in candidates:
            if ct > 1.0:
                continue
            if momentum > 0.0 and abs(ct - momentum) < 1e-12:
                a_trial = directed
            else:
                a_trial = (1.0 - ct) * a + ct * g
            a_trial = np.maximum(a_trial, 0.0)
            s = a_trial.sum()
            if s < 1e-10:
                continue
            a_trial = a_trial / s
            J_trial = compute_J_local(a_trial)
            if J_trial < best_J - 1e-12:
                best_J = J_trial
                best_a = a_trial.copy()

        if best_a is not None:
            a_prev = a.copy()
            a = best_a
            current_J = best_J
            stall = 0
            t = max(0.01, t * step_decay)
            if it % 100 == 0:
                tracker.update(a, verbose=False)
        else:
            stall += 1
            t *= 0.9
            if stall > 100:
                break

        if it % 500 == 0:
            tracker.update(a, verbose=False)

    tracker.update(a)
    return a


def run_cd_polish(a, n_passes, deadline, tracker):
    """Polish with coordinate descent."""
    a = np.maximum(a.astype(np.float64), 0.0)
    a = a / a.sum()
    current_J = compute_J_local(a)
    tracker.update(a)

    for p in range(n_passes):
        if time.monotonic() > deadline:
            break
        a, new_J = cd_pass(a)
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s < 1e-10:
            break
        a = a / s
        if new_J < current_J - 1e-10:
            current_J = new_J
            tracker.update(a, verbose=False)
        else:
            if p > 2:
                break

    tracker.update(a)
    return a
