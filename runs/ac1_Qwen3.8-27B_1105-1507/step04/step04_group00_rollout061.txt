"""
Minimizes J(a) = 2*n*max(conv(a,a))/(sum(a))^2 using an improved hybrid approach:
1. Structured initialization from low-autocorrelation sequence theory
2. Smooth gradient descent with adaptive log-sum-exp
3. LP mirror descent with active-set tracking and improved line search
4. Active-set Newton refinement (solves KKT system directly)
5. Multi-length exploration with cross-pollination
6. Final polish
"""

import numpy as np
from scipy.optimize import linprog, minimize
from scipy.sparse import csr_matrix
import time


def fast_conv(a):
    """Full self-convolution via FFT."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]


def compute_J(a):
    """Compute J(a) = 2*n*max(conv(a,a))/(sum(a))^2."""
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))


def smooth_max_and_grad(a, tau):
    """Compute smooth max and its gradient."""
    n = len(a)
    b = fast_conv(a)
    b_max = b.max()
    if tau * b_max > 500:  # numerical stability
        tau = 500.0 / max(b_max, 1e-10)
    exp_b = np.exp(tau * (b - b_max))
    p = exp_b / exp_b.sum()
    smooth_val = b_max + np.log(exp_b.sum()) / tau
    
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        pk = p[k]
        if pk < 1e-14:
            continue
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        grad[lo:hi + 1] += 2.0 * pk * a[k - hi:k - lo + 1][::-1]
    
    return smooth_val, grad, p, b


def build_A_sparse(n, a):
    """Build sparse constraint matrix for LP."""
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        j_min = max(0, k - n + 1)
        j_max = min(n - 1, k)
        for j in range(j_min, j_max + 1):
            rows.append(k)
            cols.append(j)
            vals.append(a[k - j])
    A = csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))
    return A


def lp_best_response(a, M):
    """LP: max sum(g) s.t. (a*g)[k] <= M for all k, g >= 0."""
    n = len(a)
    A = build_A_sparse(n, a)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    try:
        r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                    options={'time_limit': 60, 'presolve': True})
        if r.success and r.x is not None:
            g = np.maximum(r.x, 0)
            if g.sum() > 1e-15:
                return g
    except Exception:
        pass
    return None


def active_set_solve(a, active_set, max_iters=100, tol=1e-10):
    """
    Given a fixed active set, solve the KKT system.
    KKT: sum_{k in active} lambda_k * 2*a[k-i] = mu for all i with a_i > 0
    Plus: sum(a) = 1, a >= 0, sum(lambda) = 1, lambda >= 0
    
    We iterate: update a from the linear system, then update active set.
    """
    n = len(a)
    a = a.copy()
    
    for it in range(max_iters):
        # Compute current M
        b = fast_conv(a)
        M = b.max()
        
        # Update active set: positions where b is close to M
        tol_act = max(1e-8, 0.01 * M)
        active = set(np.where(b >= M - tol_act)[0].tolist())
        
        if len(active) == 0:
            break
        
        # Build the system: for each i, sum_{k in active, i in S_k} a[k-i] = c_i
        # This is a linear system in a: A_sys @ a = c
        # Where (A_sys)_{i,j} = |{k in active : i+j=k}| = 1 if i+j in active else 0
        
        # Actually the equation is: sum_{k in active} 2*a[k-i]*1(i in S_k) = mu
        # => 2 * sum_{k in active} a[k-i]*1(0<=k-i<n, max(0,k-n+1)<=i<=min(n-1,k)) = mu
        # => 2 * sum_{j: i+j in active, 0<=i<n, 0<=j<n} a[j] = mu
        # => sum_{j: i+j in active} a[j] = mu/2
        
        # So (A_sys)_{i,j} = 1 if (i+j) in active_set else 0
        # And we want A_sys @ a = (mu/2) * 1
        
        # Plus sum(a) = 1
        
        # This is: [A_sys; 1^T] @ [a; mu/2] = [mu/2 * 1; 1]
        # Which is linear in (a, mu/2)!
        
        # Let's build A_sys (n x n matrix)
        # (A_sys)_{i,j} = 1 if i+j in active else 0
        # This is a "sum-structured" matrix
        
        # For efficiency, let's think about it differently.
        # A_sys @ a = (mu/2) * 1
        # 1^T @ a = 1
        # => [A_sys - (1/2)*1*1^T] @ a = 0  ... from first eq: A_sys @ a = (mu/2)*1
        #    1^T @ a = 1
        # From first: (A_sys @ a)_i = (mu/2) for all i
        # So A_sys @ a is a constant vector!
        # => (A_sys - c*J) @ a = 0 where J is all-ones matrix and c = mu/2
        # But we don't know c.
        
        # Alternative: A_sys @ a = (mu/2) * 1, and 1^T a = 1
        # Let v = A_sys @ a. We need v_i = const for all i, and 1^T a = 1.
        # This means a is in the null space of (A_sys - alpha * 1*1^T) for some alpha,
        # and 1^T a = 1.
        
        # Simpler approach: directly solve the (n+1) x (n+1) system
        # [A_sys  -(1/2)*1] @ [a; mu] = 0  ... wait this isn't quite right
        # Actually: A_sys @ a - (1/2)*mu*1 = 0 => A_sys @ a = (mu/2)*1
        # 1^T @ a = 1
        # => [A_sys  -(1/2)*1] @ [a; mu] = 0
        #    [1^T    0       ] @ [a; mu] = 1
        
        # This is an (n+1) x (n+1) system. Let's build it.
        # But A_sys is n x n, so the full system is (n+1) x (n+1).
        
        # For n up to ~512, this is a 513 x 513 dense system, solvable in ms.
        
        # Build A_sys: (A_sys)_{i,j} = 1 if (i+j) in active else 0
        # This can be done efficiently using the structure.
        
        active_arr = np.zeros(2*n-1, dtype=bool)
        for k in active:
            if 0 <= k < 2*n-1:
                active_arr[k] = True
        
        # (A_sys)_{i,j} = active_arr[i+j]
        # This is a "Hankel-like" matrix (constant on anti-diagonals)
        # We can build it as: for each i, (A_sys)_{i,:} = active_arr[i : i+n]
        
        A_sys = np.zeros((n, n))
        for i in range(n):
            A_sys[i, :] = active_arr[i:i+n]
        
        # Build the augmented system
        # [A_sys  -(1/2)*1] @ [a; mu] = 0
        # [1^T    0       ] @ [a; mu] = 1
        
        # Actually, let's just use the fact that at the solution, A_sys @ a = const * 1
        # So we want to find a such that A_sys @ a is as constant as possible,
        # with 1^T a = 1 and a >= 0.
        
        # Simpler: solve [A_sys; 1^T] @ a = [c*1; 1] for unknowns a and c.
        # This gives n+1 equations in n+1 unknowns (a has n components, c is 1).
        
        # Let's augment:
        # A_aug = [A_sys  -0.5*1]  (n x (n+1))
        # b_aug = 0
        # A_aug = [A_aug; 1^T  0]  ((n+1) x (n+1))
        # b_aug = [0; 1]
        
        A_aug = np.zeros((n+1, n+1))
        A_aug[:n, :n] = A_sys
        A_aug[:n, n] = -0.5
        A_aug[n, :n] = 1.0
        A_aug[n, n] = 0.0
        b_aug = np.zeros(n+1)
        b_aug[n] = 1.0
        
        try:
            sol = np.linalg.solve(A_aug, b_aug)
            a_new = sol[:n]
            mu = sol[n]
        except np.linalg.LinAlgError:
            # System is singular, try least squares
            try:
                sol = np.linalg.lstsq(A_aug, b_aug, rcond=None)[0]
                a_new = sol[:n]
                mu = sol[n]
            except:
                break
        
        a_new = np.maximum(a_new, 0)
        s = a_new.sum()
        if s < 1e-10:
            break
        a_new /= s
        
        # Check convergence
        if np.linalg.norm(a_new - a) < tol:
            a = a_new
            break
        
        a = a_new
    
    return a

This is the "overlap" of f with its reflection. To minimize this, we want f and its reflection to have minimal overlap, i.e., f should be concentrated on one side.

If f is concentrated near x=0 (say f(x) ≈ δ(x)), then:
(f*f)(t) ≈ δ(t) (a spike at t=0)
But we also have (f*f)(t) ≈ 0 for t > 0.
The max would be at t=0, which in the discrete case is a[0]^2.

So there's a trade-off: concentrating f reduces the center peak but increases the boundary peak.

The optimal f balances these two effects. This is similar to the uncertainty principle!
