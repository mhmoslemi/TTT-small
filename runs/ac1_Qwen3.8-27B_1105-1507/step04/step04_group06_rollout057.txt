    # Let's just use: S = np.correlate(p, a, mode='full') and extract the right part
    # Actually the cleanest way:
    # sum_k p[k] * a[k-i] for k-i in [0,n-1] means k in [i, i+n-1]
    # = dot(p[i:i+n], a) -- but p has length 2n-1, a has length n
    # For i=0: dot(p[0:n], a)
    # For i=1: dot(p[1:n+1], a)
    # etc.
    # This is a sliding window dot product = a form of correlation
    # Use: np.correlate(p, a, mode='valid') gives n-1+... hmm
    # Actually: define the cross-correlation as:
    # R[m] = sum_j a[j] * p[j+m] for m = 0, 1, ..., (2n-2) - (n-1) = n-1
    # R[m] = sum_{j=0}^{n-1} a[j] * p[j+m]
    # We want S[i] = R[i] for i = 0, ..., n-1
    # R[m] can be computed as: flip a, convolve with p, extract
    # R[m] = (flip(a) * p)[(n-1) + m]  ... let me verify
    # flip(a)[k] = a[n-1-k]
    # (flip(a) * p)[t] = sum_k flip(a)[k] * p[t-k] = sum_k a[n-1-k] * p[t-k]
    # Let j = n-1-k, so k = n-1-j:
    # = sum_j a[j] * p[t-(n-1-j)] = sum_j a[j] * p[t-n+1+j]
    # We want sum_j a[j] * p[j+m], so t-n+1 = m, i.e., t = n-1+m
    # So R[m] = (flip(a) * p)[n-1+m]
    # And S[i] = R[i]
    # grad[i] ≈ 2 * S[i] (ignoring diagonal)
    a_flip = a[::-1]
    full_conv_ap = np.convolve(a_flip, p, mode='full')
    S = full_conv_ap[n - 1: 2 * n - 1]  # S[i] = R[i] for i=0..n-1
    # Add diagonal correction: for k=2i, we have a[i] instead of 2*a[i]
    # So subtract p[2i]*a[i] for each i where 2i < 2n-1
    diag_mask = np.arange(n) * 2 < (2 * n - 1)
    diag_correction = np.zeros(n)
    idx2 = np.arange(n) * 2
    valid_diag = idx2 < (2 * n - 1)
    diag_correction[valid_diag] = p[idx2[valid_diag]] * a[valid_diag]
    grad = 2.0 * S - diag_correction
    return g_val, grad


def F_and_grad(a, beta):
    """Compute F(a) = 2n*smooth_max(conv(a,a))/(sum(a))^2 and gradient."""
    n = len(a)
    s = float(a.sum())
    if s < 1e-10:
        return 1e10, np.zeros(n)
    g_val, grad_g = smooth_max_and_grad(a, beta)
    F_val = 2.0 * n * g_val / (s * s)
    grad_F = 2.0 * n * (s * grad_g - 2.0 * g_val) / (s ** 3)
    return F_val, grad_F

def build_A_sparse(a, active_idx):
    """Build sparse constraint matrix A where rows are active_idx only."""
    n = len(a)
    n_active = len(active_idx)
    rows_list = []
    cols_list = []
    vals_list = []
    for row_i, k in enumerate(active_idx):
        jl = max(0, k - n + 1)
        jh = min(n - 1, k)
        for j in range(jl, jh + 1):
            rows_list.append(row_i)
            cols_list.append(j)
            vals_list.append(a[j])
    if len(vals_list) == 0:
        return csr_matrix((n_active, n))
    return csr_matrix((vals_list, (rows_list, cols_list)), shape=(n_active, n))


def lp_best_response(a, M, warm_start=None, max_constraints=300, constraint_gen_iters=10):
    """
    LP: max sum(g) s.t. (a*g)_k <= M for k in active set, g >= 0.
    Uses constraint generation: start with top constraints, add violated ones.
    Returns normalized g or None.
    """
    n = len(a)
    b = fast_conv(a)

    # Initial active set: top K by b_k value
    K = min(max_constraints, 2 * n - 1)
    sorted_idx = np.argsort(b)[::-1]
    active_idx = sorted_idx[:K].copy()
    active_set = set(active_idx.tolist())

    g_current = None

    for cg_iter in range(constraint_gen_iters):
        # Build LP
        n_act = len(active_idx)
        c = -np.ones(n)
        A = build_A_sparse(a, active_idx)
        b_ub = np.full(n_act, M)
        bounds = [(0.0, None)] * n

        # Warm start
        options = {'presolve': True}
        if warm_start is not None and warm_start.shape[0] == n:
            # Can't easily warm-start HiGHS via linprog, but presolve helps
            pass

        r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs', options=options)
        if not r.success or r.x is None:
            break

        g_candidate = np.maximum(r.x, 0.0)
        g_sum = g_candidate.sum()
        if g_sum < 1e-10:
            break

        # Check if g_candidate violates any non-active constraint
        ag = fast_conv(a) * 0 + np.convolve(a, g_candidate, mode='full')
        # Only check non-active constraints
        all_satisfied = True
        most_violated = -1
        max_violation = 0.0
        for k in range(2 * n - 1):
            if k not in active_set:
                if ag[k] > M + 1e-10 * M:
                    all_satisfied = False
                    viol = ag[k] - M
                    if viol > max_violation:
                        max_violation = viol
                        most_violated = k

        if all_satisfied:
            g_current = g_candidate / g_sum
            break

        # Add the most violated constraint
        if most_violated >= 0:
            active_idx = np.append(active_idx, most_violated)
            active_set.add(most_violated)
            # Keep size bounded
            if len(active_idx) > max_constraints:
                # Remove the least important (lowest b_k value among active)
                b_active = b[active_idx]
                remove_pos = np.argmin(b_active)
                removed_k = active_idx[remove_pos]
                active_idx = np.delete(active_idx, remove_pos)
                active_set.remove(removed_k)
                # But make sure we don't remove the one we just added
                if removed_k == most_violated:
                    active_idx = np.append(active_idx, most_violated)
                    active_set.add(most_violated)
        else:
            g_current = g_candidate / g_sum
            break

    if g_current is None and r is not None and r.success and r.x is not None:
        g_candidate = np.maximum(r.x, 0.0)
        g_sum = g_candidate.sum()
        if g_sum > 1e-10:
            g_current = g_candidate / g_sum

    return g_current


def backtracking_line_search(a, g, J0, alpha=1e-4, rho=0.5, max_steps=20):
    """
    Backtracking line search: find t in (0,1] such that
    J((1-t)a + t*g) <= J0 - alpha * t * |direction|
    where direction is approximated by (J0 - J(g))/1 (descent estimate).
    """
    Jg = compute_J(g)
    if Jg < J0:
        desc = (J0 - Jg)  # Descent estimate
    else:
        desc = J0 * 0.01  # Fallback
    if desc < 1e-15:
        desc = 1e-6

    best_t = 0.0
    best_J = J0
    t = 1.0
    for _ in range(max_steps):
        a_try = (1.0 - t) * a + t * g
        a_try = np.maximum(a_try, 0.0)
        s_try = a_try.sum()
        if s_try < 1e-10:
            t *= rho
            continue
        a_try = a_try / s_try
        J_try = compute_J(a_try)
        if J_try < best_J:
            best_J = J_try
            best_t = t
        if J_try <= J0 - alpha * t * desc:
            return a_try, J_try, t
        t *= rho
        if t < 1e-8:
            break

    if best_t > 0:
        best_a = (1.0 - best_t) * a + best_t * g
        best_a = np.maximum(best_a, 0.0)
        s = best_a.sum()
        if s > 1e-10:
            best_a = best_a / s
        return best_a, best_J, best_t
    return a, J0, 0.0


def lp_mirror_descent(a, max_iters, deadline, tracker, n_line_evals=15,
                      stall_patience=40, max_constraints=200):
    """
    LP-based mirror descent with backtracking line search and active-set LP.
    """
    a = np.maximum(a.astype(np.float64), 1e-8)
    a = a / a.sum()
    current_J = compute_J(a)
    if tracker is not None:
        tracker.update(a)

    stall_count = 0
    best_J_so_far = current_J
    best_a_so_far = a.copy()
    n = len(a)

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        b = fast_conv(a)
        M = float(b.max())

        # Solve LP for best response
        g = lp_best_response(a, M, max_constraints=max_constraints,
                             constraint_gen_iters=5)
        if g is None:
            break

        # Line search
        a_new, J_new, t_used = backtracking_line_search(a, g, current_J)

        if J_new < current_J - 1e-14:
            a = a_new
            current_J = J_new
            stall_count = 0
            if J_new < best_J_so_far:
                best_J_so_far = J_new
                best_a_so_far = a.copy()
                if tracker is not None:
                    tracker.update(a)
        else:
            # Try smaller fixed steps
            improved = False
            for t_try in [0.05, 0.02, 0.01, 0.1]:
                a_try = (1.0 - t_try) * a + t_try * g
                a_try = np.maximum(a_try, 0.0)
                s_try = a_try.sum()
                if s_try < 1e-10:
                    continue
                a_try = a_try / s_try
                J_try = compute_J(a_try)
                if J_try < current_J - 1e-14:
                    a = a_try
                    current_J = J_try
                    stall_count = 0
                    improved = True
                    if J_try < best_J_so_far:
                        best_J_so_far = J_try
                        best_a_so_far = a.copy()
                        if tracker is not None:
                            tracker.update(a)
                    break
            if not improved:
                stall_count += 1
                if stall_count >= stall_patience:
                    break

    if tracker is not None:
        tracker.update(best_a_so_far)
    return best_a_so_far, best_J_so_far


def lbfgs_optimize(a0, deadline, tracker, n=1024):
    """L-BFGS-B optimization with adaptive beta annealing."""
    a = np.maximum(a0.astype(np.float64), 1e-8)
    a = a / a.sum()

    beta_schedule = [(1.0, 50), (3.0, 50), (10.0, 60), (30.0, 80), (100.0, 100)]

    for beta, max_iters in beta_schedule:
        if time.monotonic() > deadline:
            break

        def func(x):
            x2 = np.maximum(x, 0.0)
            f, _ = F_and_grad(x2, beta)
            if not np.isfinite(f):
                return 1e10
            return f

        def jac(x):
            x2 = np.maximum(x, 0.0)
            _, g = F_and_grad(x2, beta)
            g = np.nan_to_num(g, nan=0.0, posinf=0.0, neginf=0.0)
            g[x2 < 1e-12] = 0.0
            return g

        bounds = [(0.0, None)] * n
        try:
            res = minimize(func, a.copy(), jac=jac, method='L-BFGS-B',
                           bounds=bounds,
                           options={'maxiter': max_iters, 'max_fun': max_iters * 30,
                                    'gtol': 1e-10, 'ftol': 1e-16})
            if res.fun < 1e10:
                new_a = np.maximum(res.x, 1e-10)
                if new_a.sum() > 1e-6:
                    new_a = new_a / new_a.sum()
                    J_new = compute_J(new_a)
                    if J_new < 1e10:
                        a = new_a
                        if tracker is not None:
                            tracker.update(a)
        except Exception:
            pass

    return a


def resample_seq(src, dest_n):
    """Resample 1D sequence to new length via linear interpolation."""
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    result = np.interp(t_new, t_old, src)
    return np.maximum(result, 0.0)


def generate_inits(n, rng, parent=None):
    """Generate diverse initial sequences of length n."""
    inits = []
    t_arr = np.linspace(0, 1, n, endpoint=False)

    if parent is not None and parent.size > 0:
        p = resample_seq(parent, n)
        inits.append(p.copy())
        p_noisy = p + rng.normal(0, 0.05, n) * p
        inits.append(np.maximum(p_noisy, 0.01))
        p_noisy2 = p + rng.normal(0, 0.15, n) * p
        inits.append(np.maximum(p_noisy2, 0.01))

    # Uniform
    inits.append(np.ones(n).astype(np.float64))

    # Random
    inits.append(rng.uniform(0.3, 1.0, n))

    # Cosine families - different frequencies
    for freq in [3, 5, 7, 11]:
        inits.append(1.0 + 0.4 * np.cos(2.0 * np.pi * freq * t_arr + rng.uniform(0, 2*np.pi)))

    # Multi-frequency
    inits.append(1.0 + 0.25 * np.cos(2*np.pi*3*t_arr) + 0.15 * np.cos(2*np.pi*7*t_arr))
    inits.append(1.0 + 0.2 * np.cos(2*np.pi*5*t_arr) - 0.1 * np.cos(2*np.pi*13*t_arr))

    # Asymmetric patterns (key for J < 2)
    inits.append(1.0 + 0.3 * np.sin(2*np.pi*2*t_arr))
    inits.append(np.exp(-2.0*(t_arr-0.3)**2) + 0.3)
    inits.append(np.exp(-2.0*(t_arr-0.7)**2) + 0.3)

    # Asymmetric multi-scale
    inits.append(1.0 + 0.3*np.cos(2*np.pi*4*t_arr) + 0.2*np.cos(2*np.pi*9*t_arr + 0.5))

    # Power / exponential
    inits.append((t_arr + 0.1)**0.3 + 0.2)
    inits.append(np.exp(-1.0*t_arr) + 0.4)

    # Sparse peaks
    peak_positions = rng.choice(n, size=n//32, replace=False)
    sparse = np.ones(n) * 0.2
    sparse[peak_positions] = 1.0 + rng.uniform(0, 0.5, len(peak_positions))
    inits.append(sparse)

    return inits


class ScoreTracker:
    """Tracks the best sequence found so far."""

    def __init__(self):
        self.best = None
        self.best_score = float('inf')
        self.t_start = time.monotonic()

    def update(self, a):
        a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
        if a.size == 0 or a.sum() < 0.01:
            return
        score = float(evaluate_sequence(a.tolist()))
        if np.isfinite(score) and score < self.best_score:
            self.best = a.copy()
            self.best_score = score
            elapsed = time.monotonic() - self.t_start
            print(f"  * NEW BEST: J={score:.8f} n={a.size} t={elapsed:.1f}s", flush=True)

    def report(self, label=""):
        if self.best is not None:
            print(f"  {label} best: J={self.best_score:.8f} n={self.best.size}", flush=True)


def coordinate_descent_polish(a, max_coord_steps, deadline, tracker):
    """
    Try to improve by adjusting individual coordinates.
    For each coordinate i, do a 1D search over a[i].
    """
    a = a.copy()
    n = len(a)
    current_J = compute_J(a)
    improved_any = False

    # Shuffle coordinates for diversity
    rng_local = np.random.default_rng(42)
    order = rng_local.permutation(n)

    coord_iter = 0
    for i in order:
        if coord_iter >= max_coord_steps or time.monotonic() > deadline:
            break
        coord_iter += 1

        a_old = a[i]
        # Try a few scale factors
        for factor in [1.2, 0.8, 1.5, 0.5, 1.1, 0.9]:
            a[i] = max(0, a_old * factor)
            s = a.sum()
            if s < 0.01:
                a[i] = a_old
                continue
            J_try = compute_J(a)
            if J_try < current_J - 1e-12:
                current_J = J_try
                improved_any = True
                break
        a[i] = a  # keep the improved value

    if improved_any and tracker is not None:
        tracker.update(a)
    return a


def propose_candidate(seed=42, budget_s=1500, **kwargs):
    """
    Multi-phase optimization for minimizing J(a) = 2n*max(conv(a,a))/(sum(a))^2.

    Phase 1: Multi-start L-BFGS-B with smooth objective (fast basin finding)
    Phase 2: LP mirror descent with active-set and backtracking (main refinement)
    Phase 3: Multi-restart with diverse perturbations
    Phase 4: Final polish with extended LP + coordinate descent
    Phase 5: Length exploration
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + float(budget_s) - 15.0
    tracker = ScoreTracker()

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        if p.ndim == 1 and p.size > 0 and np.all(np.isfinite(p)):
            parent = np.maximum(p, 0.0)
            tracker.update(parent)
            print(f"Parent loaded: n={parent.size}, J={tracker.best_score:.8f}", flush=True)
    except (NameError, AttributeError):
        pass

    if parent is None:
        parent = np.ones(512)
        tracker.update(parent)
        print(f"Using default parent: n=512, J={tracker.best_score:.8f}", flush=True)

    # === Phase 1: L-BFGS-B with smooth objective ===
    t0 = time.monotonic()
    phase1_budget = 200.0
    deadline_p1 = time.monotonic() + phase1_budget
    print(f"\n{'='*50} Phase 1: L-BFGS-B (budget {phase1_budget:.0f}s) {'='*50}", flush=True)

    n_main = 1024
    for n in [n_main]:
        if time.monotonic() > deadline_p1:
            break
        t_start = time.monotonic()
        print(f"  L-BFGS-B n={n}...", flush=True)

        inits = generate_inits(n, rng, parent)
        # Limit number of initializations based on time
        max_inits = min(len(inits), 8)

        for idx, a0 in enumerate(inits[:max_inits]):
            if time.monotonic() > deadline_p1:
                break
            a0 = np.maximum(a0.astype(np.float64), 0.01)
            a0 = a0 / a0.sum()
            remaining = deadline_p1 - time.monotonic()
            if remaining < 5:
                break
            a_opt = lbfgs_optimize(a0, deadline_p1, tracker)
            if a_opt is not None and a_opt.sum() > 0.01:
                tracker.update(a_opt)
            print(f"    init {idx+1}/{max_inits} done, best={tracker.best_score:.8f}", flush=True)

        print(f"    n={n} done in {time.monotonic()-t_start:.1f}s", flush=True)

    tracker.report("Phase 1 complete")

    # === Phase 2: LP mirror descent with active-set (MAIN) ===
    if tracker.best is not None and time.monotonic() < deadline:
        t0 = time.monotonic()
        n_best = tracker.best.size
        phase2_budget = 600.0
        deadline_p2 = min(deadline, time.monotonic() + phase2_budget)
        print(f"\n{'='*50} Phase 2: LP MD (active-set) n={n_best} "
              f"(budget {phase2_budget:.0f}s) {'='*50}", flush=True)

        # Primary LP run
        remaining = deadline_p2 - time.monotonic()
        max_iters = int(remaining * 100)  # Estimate ~100 LP iters/sec with active set
        max_iters = min(max_iters, 20000)
        a, J_lp = lp_mirror_descent(tracker.best, max_iters, deadline_p2, tracker=tracker,
                                    max_constraints=200)
        print(f"  Primary LP done: J={J_lp:.8f}", flush=True)

        # Try a few other lengths
        for n_try in [768, 1280, 2048]:
            if time.monotonic() > deadline_p2:
                break
            a_res = resample_seq(tracker.best, n_try)
            a_res = a_res / max(a_res.sum(), 1e-10)
            budget_here = min(120.0, deadline_p2 - time.monotonic())
            max_iters_here = int(budget_here * 80)
            max_iters_here = min(max_iters_here, 10000)
            a_res, J_res = lp_mirror_descent(a_res, max_iters_here, deadline_p2,
                                             tracker=tracker, max_constraints=200)
            print(f"  LP n={n_try}: J={J_res:.8f}", flush=True)

        tracker.report(f"Phase 2: {time.monotonic()-t0:.1f}s")

    # === Phase 3: Multi-restart with diverse perturbations ===
    if tracker.best is not None and time.monotonic() < deadline:
        t0 = time.monotonic()
        n_best = tracker.best.size
        phase3_budget = 350.0
        deadline_p3 = min(deadline, time.monotonic() + phase3_budget)
        print(f"\n{'='*50} Phase 3: Restarts (budget {phase3_budget:.0f}s) {'='*50}", flush=True)

        strategies = [
            ('gaussian', 0.05),
            ('gaussian', 0.10),
            ('gaussian', 0.20),
            ('smooth_noise', 0.10),
            ('spike', 16),
            ('gaussian', 0.03),
            ('block', 0.15),
            ('gaussian', 0.07),
            ('uniform_scale', 0.2),
        ]

        attempt = 0
        for pert_type, param in strategies:
            for trial in range(5):
                if time.monotonic() > deadline_p3:
                    break
                attempt += 1
                a = tracker.best.copy().astype(np.float64)
                n = len(a)

                if pert_type == 'gaussian':
                    a = a + rng.normal(0, param, n) * a
                elif pert_type == 'smooth_noise':
                    noise = rng.normal(0, param, n)
                    kernel_size = max(5, n // 32)
                    kernel = np.ones(kernel_size) / kernel_size
                    noise = np.convolve(noise, kernel, mode='same')
                    a = a + noise * a
                elif pert_type == 'spike':
                    k = min(param, n // 8)
                    idx = rng.choice(n, size=k, replace=False)
                    a[idx] *= rng.uniform(0.2, 3.0, k)
                elif pert_type == 'block':
                    # Multiply a random block by a factor
                    block_len = max(8, n // 16)
                    start = rng.integers(0, n - block_len)
                    a[start:start+block_len] *= rng.uniform(0.5, 1.5)
                elif pert_type == 'uniform_scale':
                    a = a * rng.uniform(1 - param, 1 + param, n)

                a = np.maximum(a, 1e-4)
                a = a / a.sum()

                J_pert = compute_J(a)
                if J_pert < tracker.best_score:
                    tracker.update(a)

                # Run LP refinement
                budget_here = min(40.0, deadline_p3 - time.monotonic())
                if budget_here > 2:
                    max_iters = int(budget_here * 80)
                    max_iters = min(max_iters, 5000)
                    a, _ = lp_mirror_descent(a, max_iters, deadline_p3, tracker=tracker,
                                             max_constraints=150)

        tracker.report(f"Phase 3: {time.monotonic()-t0:.1f}s, {attempt} attempts")

    # === Phase 4: Final LP polish ===
    if tracker.best is not None and time.monotonic() < deadline:
        t0 = time.monotonic()
        n_best = tracker.best.size
        remaining = deadline - time.monotonic()
        phase4_budget = min(200.0, remaining)
        deadline_p4 = time.monotonic() + phase4_budget
        print(f"\n{'='*50} Phase 4: Final Polish n={n_best} "
              f"(budget {phase4_budget:.0f}s) {'='*50}", flush=True)

        # Long LP run with more constraints for precision
        max_iters = int(phase4_budget * 80)
        max_iters = min(max_iters, 15000)
        a, J_final = lp_mirror_descent(tracker.best, max_iters, deadline_p4, tracker=tracker,
                                       max_constraints=400)
        print(f"  Final LP: J={J_final:.8f}", flush=True)

        # Coordinate descent polish
        if tracker.best is not None and time.monotonic() < deadline_p4:
            remaining_cd = deadline_p4 - time.monotonic()
            if remaining_cd > 5:
                print(f"  Coordinate descent ({remaining_cd:.0f}s remaining)...", flush=True)
                a_cd = coordinate_descent_polish(tracker.best, int(remaining_cd * 200),
                                                 deadline_p4, tracker)

        tracker.report(f"Phase 4: {time.monotonic()-t0:.1f}s")

    # === Phase 5: Length exploration ===
    if tracker.best is not None and time.monotonic() < deadline:
        t0 = time.monotonic()
        n_best = tracker.best.size
        remaining = deadline - time.monotonic()
        if remaining > 20:
            print(f"\n{'='*50} Phase 5: Length Exploration ({remaining:.0f}s) {'='*50}", flush=True)

            for n_try in [n_best - 64, n_best + 64, n_best - 128, n_best + 128,
                          384, 640, 768, 1536, 2048]:
                if n_try < 64 or n_try > 3000 or time.monotonic() > deadline:
                    continue
                if abs(n_try - n_best) < 32:
                    continue
                a_res = resample_seq(tracker.best, n_try)
                a_res = a_res / max(a_res.sum(), 1e-10)
                budget_here = min(30.0, deadline - time.monotonic())
                if budget_here < 3:
                    continue
                max_iters = int(budget_here * 80)
                max_iters = min(max_iters, 3000)
                a_res, _ = lp_mirror_descent(a_res, max_iters, deadline, tracker=tracker,
                                             max_constraints=150)

            tracker.report(f"Phase 5: {time.monotonic()-t0:.1f}s")

    # Fallback
    if tracker.best is None:
        tracker.best = np.ones(512)
        tracker.best_score = compute_J(tracker.best)
        print(f"  Using fallback: J={tracker.best_score:.8f}", flush=True)

    print(f"\n{'='*50} FINAL RESULT {'='*50}", flush=True)
    print(f"J={tracker.best_score:.8f} n={tracker.best.size} "
          f"time={time.monotonic()-tracker.t_start:.1f}s", flush=True)

    # Ensure output is valid
    result = np.clip(tracker.best, 0.0, 1000.0)
    if result.sum() < 0.01:
        result = np.ones(128)

    return [float(x) for x in result]
