Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using Adam optimizer with soft-max

import numpy as np
from scipy.fft import rfft, irfft
import time

def compute_conv(a):
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    conv = irfft(fa * fa, size)
    return conv[:2 * n - 1]

def compute_obj(a):
    n = len(a)
    if n == 0:
        return np.inf
    conv = compute_conv(a)
    max_b = float(np.max(conv))
    sum_a = float(np.sum(a))
    if sum_a < 0.01:
        return np.inf
    return 2.0 * n * max_b / (sum_a * sum_a)

def softmax_params(z):
    z = z - np.max(z)
    e = np.exp(z)
    return e / e.sum()

def softmax_obj_and_grad(a, tau):
    """Compute soft-max of conv(a,a) and gradient w.r.t. a."""
    n = len(a)
    conv = compute_conv(a)
    cmax = float(np.max(conv))
    shifted = np.clip((conv - cmax) / tau, -500.0, 500.0)
    exp_conv = np.exp(shifted)
    Z = float(np.sum(exp_conv))
    soft_max_val = cmax + tau * np.log(Z)
    w = exp_conv / Z
    
    # grad[i] = 2 * sum_{j} w[i+j] * a[j] (convolution-based)
    w_rev = w[::-1]
    size = 1
    while size < n + len(w) - 1:
        size <<= 1
    fa = rfft(a, size)
    fw = rfft(w_rev, size)
    conv_aw = irfft(fa * fw, size)[:n + len(w) - 1]
    grad = 2.0 * conv_aw[n-1:2*n-1][::-1]
    
    return soft_max_val, grad

def adam_optimize(a_init, tau_list, time_budget=30.0, lr=0.01, beta1=0.9, beta2=0.999, epsilon=1e-8):
    """Adam optimizer on the simplex (sum a = 1, a >= 0) using log-parametrization."""
    n = len(a_init)
    a = a_init.copy().astype(np.float64)
    s = np.sum(a)
    if s < 1e-10:
        a = np.ones(n) / n
    else:
        a = a / s
    a = np.maximum(a, 1e-30)
    
    # Work in log-space: z = log(a), a = softmax-like (but we use exp(z) / sum(exp(z)))
    z = np.log(a)
    
    m = np.zeros(n)  # first moment
    v = np.zeros(n)  # second moment
    t = 0
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    
    for tau in tau_list:
        if time.monotonic() - start > time_budget:
            break
        
        iters = 0
        while time.monotonic() - start < time_budget:
            t += 1
            iters += 1
            
            a = softmax_params(z)
            soft_max_val, grad_a = softmax_obj_and_grad(a, tau)
            
            # Convert gradient from a-space to z-space
            # a = exp(z) / sum(exp(z)), so da_i/dz_j = a_i * (δ_ij - a_j)
            # grad_z = a * (grad_a - sum(a * grad_a))
            mean_grad = float(np.dot(a, grad_a))
            grad_z = a * (grad_a - mean_grad)
            
            # Adam update
            m = beta1 * m + (1 - beta1) * grad_z

            # Evaluate actual objective

                best_a = a.copy()
            
            # Periodic check
            if iters % 100 == 0:
                pass
    
    return best_a, best_obj

Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using Adam optimizer with soft-max
parametrization, spectral flattening initialization, multi-resolution search,
and coordinate descent + LP polishing. Key innovations over previous approach:

import numpy as np
from scipy.fft import rfft, irfft
from scipy.optimize import linprog
import time


def compute_conv(a):
    """Compute linear convolution a*a using FFT."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    conv = irfft(fa * fa, size)
    return conv[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    conv = compute_conv(a)
    max_b = float(np.max(conv))
    sum_a = float(np.sum(a))
    if sum_a < 0.01:
        return np.inf
    return 2.0 * n * max_b / (sum_a * sum_a)


def softmax_params(z):
    """Numerically stable softmax."""
    z = z - np.max(z)
    e = np.exp(z)
    return e / e.sum()


def fft_convolve(a, b):
    """Linear convolution using FFT."""
    n, m = len(a), len(b)
    size = 1
    while size < n + m - 1:
        size <<= 1
    fa = rfft(a, size)
    fb = rfft(b, size)
    return irfft(fa * fb, size)[:n + m - 1]


def softmax_conv_grad(a, tau):
    """Compute soft-max of conv(a,a) and gradient w.r.t. a. Returns (val, grad)."""
    n = len(a)
    conv = compute_conv(a)
    cmax = float(np.max(conv))
    shifted = np.clip((conv - cmax) / tau, -500.0, 500.0)
    exp_conv = np.exp(shifted)
    Z = float(np.sum(exp_conv))
    soft_max_val = cmax + tau * np.log(Z)
    w = exp_conv / Z

    # grad[i] = 2 * sum_j w[i+j] * a[j] via FFT
    w_rev = w[::-1]
    grad = 2.0 * fft_convolve(a, w_rev)[n - 1:2 * n - 1][::-1]

    return soft_max_val, grad


def grad_z_from_grad_a(a, grad_a):
    """Convert gradient from a-space to z-space (log-softmax parameterization)."""
    mean_grad = float(np.dot(a, grad_a))
    return a * (grad_a - mean_grad)


def adam_optimize(a_init, tau_list, time_budget=60.0, lr=0.05, beta1=0.9, beta2=0.999, epsilon=1e-8):
    """
    Adam optimizer on log-softmax parametrization with tau annealing.
    Minimizes 2n * soft_max_tau(conv(a,a)) subject to sum(a)=1, a>=0.
    """
    n = len(a_init)
    a = a_init.copy().astype(np.float64)
    s = np.sum(a)
    if s < 1e-10:
        a = np.ones(n) / n
    else:
        a = a / s
    a = np.maximum(a, 1e-30)
    z = np.log(a).copy()

    m = np.zeros(n)
    v = np.zeros(n)
    t = 0

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    best_tau_obj = best_obj

    for tau in tau_list:
        if time.monotonic() - start > time_budget:
            break

        tau_start_time = time.monotonic()
        no_improve = 0
        improve_threshold = 1e-10

        while time.monotonic() - start < time_budget:
            t += 1

            a = softmax_params(z)
            soft_max_val, grad_a = softmax_conv_grad(a, tau)
            grad_z = grad_z_from_grad_a(a, grad_a)

            # Adam update
            m = beta1 * m + (1 - beta1) * grad_z
            v = beta2 * v + (1 - beta2) * grad_z ** 2
            m_hat = m / (1 - beta1 ** t)
            v_hat = v / (1 - beta2 ** t)
            z = z - lr * m_hat / (np.sqrt(v_hat) + epsilon)

            # Evaluate every 10 iterations
            if t % 10 == 0:
                a = softmax_params(z)
                obj = compute_obj(a)
                if obj < best_obj - improve_threshold:
                    best_obj = obj
                    best_a = a.copy()
                    no_improve = 0
                else:
                    no_improve += 1

                if no_improve > 50:
                    break

    return best_a, best_obj


def spectral_flatten_init(n, seed, n_iters=50, alpha=0.5):
    """
    Initialize using spectral flattening (Gerchberg-Saxton-like approach).
    Iteratively flattens the power spectrum while projecting to non-negativity.
    """
    rng = np.random.default_rng(seed)
    a = rng.random(n) + 0.1
    a = a / a.sum()

    for it in range(n_iters):
        # Compute DFT
        A = rfft(a)
        power = A.real ** 2 + A.imag ** 2
        power = np.maximum(power, 1e-30)

        # Flattening: new amplitude = 1 / power^(alpha)
        new_amp = 1.0 / (power ** alpha)

        # Keep original phases
        phase = np.arctan2(A.imag, A.real)
        A_new = new_amp * np.cos(phase) + 1j * new_amp * np.sin(phase)

        # Inverse DFT
        a_new = irfft(A_new, n)

        # Project to non-negative (soft projection)
        a_new = np.maximum(a_new, 0)
        s = a_new.sum()
        if s < 1e-10:
            a = rng.random(n)
            a = a / a.sum()
        else:
            a = a_new / s

        # Blend with previous to avoid oscillation
        a = 0.5 * a + 0.5 * (np.maximum(a_new, 0) / max(s, 1e-10))

    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-10:
        a = np.ones(n) / n
    else:
        a = a / s
    return a


def coordinate_descent(a, max_iters=5000, time_budget=30.0, step_scale=0.1):
    """
    Coordinate descent: iterate over elements, optimize each one individually.
    For element i, the optimal value minimizes the max convolution peak.
    Uses line search within each coordinate.
    """
    n = len(a)
    a = a.copy().astype(np.float64)
    s = np.sum(a)
    if s < 1e-10:
        return a
    a = a / s

    best_obj = compute_obj(a)
    start = time.monotonic()
    it_count = 0
    rng = np.random.default_rng(12345)

    while time.monotonic() - start < time_budget:
        # Shuffle coordinates
        perm = rng.permutation(n)

        for i in perm:
            if time.monotonic() - start > time_budget:
                break
            it_count += 1

            # Try adjusting a[i]
            a_old = a[i]
            if a_old < 1e-15:
                continue

            # Try multiplicative updates
            for factor in [0.7, 0.85, 1.15, 1.3, 0.5, 1.5]:

                obj_try = compute_obj(a_try)

                    best_obj = obj_try
                    break

    return a


def build_toeplitz(a):
    """Build Toeplitz matrix for convolution with a."""
    n = len(a)
    A = np.zeros((2 * n - 1, n), dtype=np.float64)
    for j in range(n):
        k_end = min(j + n, 2 * n - 1)
        A[j:k_end, j] = a[:k_end - j]
    return A


def solve_lp_g(a, M, time_limit=15):
    """Solve LP: max sum(g) s.t. (a*g)[k] <= M, g >= 0."""
    n = len(a)
    A = build_toeplitz(a)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': time_limit})
        if res.status == 0:
            g = np.maximum(res.x, 0)
            s = float(np.sum(g))
            if s > 1e-10:
                return g / s
    except Exception:
        pass
    return None

def resample_sequence(a, new_n):
    """Resample sequence to new length via interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, n, endpoint=False)
    x_new = np.linspace(0, 1, new_n, endpoint=False)
    a_new = np.interp(x_new, x_old, a, period=1)
    return np.maximum(a_new, 0)

    for ci in range(n_polish):
        if time.monotonic() > deadline - 60:
            break
        rem = deadline - time.monotonic()
        if rem < 30:
            break

        a_cand = candidates[ci][0]
        n_cand = candidates[ci][2]
        obj_cand = candidates[ci][1]
        print(f"\n  Polishing candidate {ci} (n={n_cand}, obj={obj_cand:.6f}), {rem:.0f}s left")

        polish_budget = min(rem * 0.35, 80.0)
        a_pol, obj_pol = polish_sequence(a_cand, time_budget=polish_budget,
                                         n_max_lp=800)

        if obj_pol < best_score:
            best_score = obj_pol
            best_seq = a_pol.copy()
            print(f"  ** NEW BEST after polish: obj={best_score:.6f} **")

    # Phase 3: Final intensive polish on global best
    if best_seq is not None and time.monotonic() < deadline - 30:
        rem = deadline - time.monotonic()
        n_best = len(best_seq)
        print(f"\n=== FINAL POLISH: n={n_best}, {rem:.0f}s remaining ===")

        a_cur = best_seq.copy()

        for round_num in range(3):
            if time.monotonic() > deadline - 15:
                break
            rem = deadline - time.monotonic()
            if rem < 10:
                break

            # Very fine Adam
            tau_final = [0.1 / n_best, 0.05 / n_best, 0.02 / n_best]
            a_new, obj_new = adam_optimize(a_cur, tau_final,
                                           time_budget=min(rem * 0.25, 20.0), lr=0.02)
            if obj_new < best_score:
                best_score = obj_new
                best_seq = a_new.copy()
                a_cur = a_new.copy()
                print(f"  Final Adam r{round_num}: {best_score:.6f}")

            # Coordinate descent
            rem = deadline - time.monotonic()
            if rem > 10:
                a_new = coordinate_descent(a_cur, time_budget=min(rem * 0.25, 15.0))
                obj_new = compute_obj(a_new)
                if obj_new < best_score:
                    best_score = obj_new
                    best_seq = a_new.copy()
                    a_cur = a_new.copy()
                    print(f"  Final CD r{round_num}: {best_score:.6f}")

            # LP
            rem = deadline - time.monotonic()
            if rem > 15 and n_best <= 700:
                a_new = lp_alternating(a_cur, max_iters=20, time_budget=min(rem * 0.3, 25.0))
                obj_new = compute_obj(a_new)
                if obj_new < best_score:
                    best_score = obj_new
                    best_seq = a_new.copy()
                    a_cur = a_new.copy()
                    print(f"  Final LP r{round_num}: {best_score:.6f}")

    # Final validation
    if best_seq is None:
        best_seq = np.ones(200)
        best_score = compute_obj(best_seq)

    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    val = evaluate_sequence(result)
    if val < np.inf:
        best_score = val
    else:
        result = [1.0] * 100
        best_score = 2.0

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_score:.6f}, time={elapsed:.1f}s ===")

    return result
