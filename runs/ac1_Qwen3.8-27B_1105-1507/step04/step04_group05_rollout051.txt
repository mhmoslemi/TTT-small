To speed up, I can:
1. Only check the top ~20 lags (those near the current max)
2. Use numpy vectorization for the max computation

Let me use numpy: for a given delta, the affected b values are:
b_affected = b[affected_indices] + 2*delta*a[corresponding_indices]
plus the special case for k=2i.

I can precompute the indices and use vectorized operations.

def cd_1d_optimize(a, i, b, S, n_passes=30):
    n = len(a)
    old = a[i]
    
    # Affected lags: k in [i, i+n-1] intersect [0, 2n-2]
    k_start = max(0, i)
    k_end = min(2*n-2, i+n-1)
    affected_k = np.arange(k_start, k_end+1)
    
    # For each affected k, the coefficient of delta is 2*a[k-i]
    # except for k=2i where it's different
    coeffs = 2.0 * a[k_start:k_start+len(affected_k) - (affected_k[0]-k_start):]
    # Wait, this is getting confusing. Let me just do:
    
    delta_coeffs = np.zeros(len(affected_k))
    for idx, k in enumerate(affected_k):
        delta_coeffs[idx] = 2.0 * a[k - i]
    
    # But for k=2i, the formula is different: b[k] + 2*delta*old + delta^2
    # vs the general: b[k] + 2*delta*a[k-i]
    # For k=2i: a[k-i] = a[i] = old, so 2*delta*old matches the linear part
    # We just need to add delta^2 for k=2i
    
    has_2i = (2*i in affected_k)
    
    # Unaffected max
    mask = np.ones(2*n-1, dtype=bool)
    mask[affected_k] = False
    C = b[mask].max()
    
    def J_of_delta(d):
        new_b_affected = b[affected_k] + 2*d*delta_coeffs
        if has_2i:
            idx_2i = np.where(affected_k == 2*i)[0][0]
            new_b_affected[idx_2i] += d*d
        new_max = max(new_b_affected.max(), C)
        new_S = S + d
        if new_S <= 0:
            return 1e10
        return 2*n*new_max/(new_S*new_S)
    
    # Golden section search over x = old + delta, delta in [-(old), some upper bound]
    # x in [0, hi] where hi is some reasonable bound
    lo = 0.0
    hi = max(3*old, S/n*3, 0.1)
    
    gr = (np.sqrt(5)-1)/2
    c_pt = hi - gr*(hi-lo)
    d_pt = lo + gr*(hi-lo)
    fc = J_of_delta(c_pt - old)
    fd = J_of_delta(d_pt - old)
    
    for _ in range(n_passes):
        if fc < fd:
            hi = d_pt
            d_pt = c_pt
            fd = fc
            c_pt = hi - gr*(hi-lo)
            fc = J_of_delta(c_pt - old)
        else:
            lo = c_pt
            c_pt = d_pt
            fc = fd
            d_pt = lo + gr*(hi-lo)
            fd = J_of_delta(d_pt - old)
    
    x_opt = (lo+hi)/2
    return max(0.0, x_opt)

"""
Minimizes J(a) = 2*n*max(convolve(a,a))/(sum(a))^2 using:
1. Fast LP mirror descent with precomputed sparse structure and adaptive tightening
2. Vectorized coordinate descent with active-set acceleration
3. Multi-length exploration
4. Aggressive final polish alternating LP and CD
Key improvements: vectorized CD, better step-size selection, more thorough optimization.
"""

import numpy as np
from scipy.optimize import linprog
from scipy.sparse import csr_array, coo_array
import time


def fast_conv(a):
    """Full self-convolution via FFT."""
    n = len(a)
    s = 1
    while s < 2 * n - 1:
        s <<= 1
    fa = np.fft.rfft(a, s)
    return np.fft.irfft(fa * fa, s)[:2 * n - 1]


def compute_J(a):
    """Compute J(a) = 2*n*max(conv(a,a))/(sum(a))^2."""
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))


class ScoreTracker:
    def __init__(self):
        self.best = None
        self.best_score = float('inf')
        self.t_start = time.monotonic()

    def update(self, a):
        a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
        if a.size == 0 or a.sum() < 0.01:

    def report(self, label=""):
        if self.best is not None:
            print(f"  {label} best: J={self.best_score:.10f} n={self.best.size}", flush=True)


def build_lp_structure(n):
    """Precompute sparsity structure for the LP constraint matrix."""
    rows_l = []
    cols_l = []
    for k in range(2 * n - 1):
        jl = max(0, k - n + 1)
        jh = min(n - 1, k)
        for j in range(jl, jh + 1):
            rows_l.append(k)
            cols_l.append(j)
    return np.array(rows_l, dtype=np.int32), np.array(cols_l, dtype=np.int32)


def build_A_dense(n, a):
    """Build dense constraint matrix (for small n)."""
    n_constraints = 2 * n - 1
    A = np.zeros((n_constraints, n))
    for k in range(n_constraints):
        jl = max(0, k - n + 1)
        jh = min(n - 1, k)
        for j in range(jl, jh + 1):
            A[k, j] = a[k - j]
    return A


def lp_best_response(a, M):
    """LP: max sum(g) s.t. (a*g)_k <= M for all k, g >= 0."""
    n = len(a)
    c = -np.ones(n)
    A = build_A_dense(n, a)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    try:
        r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                    options={'presolve': True, 'dual_feasibility_tolerance': 1e-9,
                             'primal_feasibility_tolerance': 1e-9})
        if r.success and r.x is not None:
            g = np.maximum(r.x, 0.0)
            if g.sum() > 1e-10:
                return g
    except Exception:
        pass
    return None


def run_lp_iterations(a, max_iters, deadline, tracker, tighten_base=0.003):
    """LP mirror descent with adaptive tightening and line search."""
    a = np.maximum(a.astype(np.float64), 1e-6)
    a = a / a.sum()
    n = len(a)
    current_J = compute_J(a)
    tracker.update(a)

    tighten = tighten_base
    stall_count = 0

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        b = fast_conv(a)
        M = float(b.max())

        # Try the target
        M_target = M * (1.0 - tighten)
        g = lp_best_response(a, M_target)

        if g is None:
            # Back off
            tighten *= 0.5
            if tighten < 1e-7:
                stall_count += 1
                if stall_count > 30:
                    break
                tighten = tighten_base
            continue

        g = np.maximum(g, 0.0)
        g_sum = g.sum()
        if g_sum < 1e-10:
            break
        g = g / g_sum

        # Line search over step sizes
        progress = min(1.0, it / max(1, max_iters * 0.8))
        base_t = max(0.02, 0.6 * (1.0 - 0.7 * progress))
        candidates_t = [base_t, base_t * 0.5, base_t * 0.25, base_t * 1.5, 0.9, 0.7, 0.5]
        candidates_t = [t for t in candidates_t if 0 < t <= 1.0]

        best_a = None
        best_J = current_J

        for t in candidates_t:
            a_trial = (1.0 - t) * a + t * g
            a_trial = np.maximum(a_trial, 0.0)
            s = a_trial.sum()
            if s < 1e-10:
                continue
            a_trial = a_trial / s
            J_trial = compute_J(a_trial)
            if J_trial < best_J - 1e-12:
                best_J = J_trial
                best_a = a_trial

        if best_a is None:
            stall_count += 1
            tighten *= 0.7
            if stall_count > 40:
                break
            continue

        a = best_a
        current_J = best_J
        stall_count = 0

        # Adapt tightening
        tighten = min(0.08, tighten * 1.02)

        if it % 50 == 0:
            tracker.update(a)

    tracker.update(a)
    return a


def cd_pass_vectorized(a, max_iter_per_coord=35):
    """One pass of vectorized coordinate descent."""
    a = np.maximum(a.astype(np.float64), 0.0)
    S = a.sum()
    if S < 1e-10:
        return a, float('inf')
    n = len(a)
    b = fast_conv(a)
    current_J = 2.0 * n * b.max() / (S * S)

    indices = list(range(n))
    np.random.shuffle(indices)

    improved = False
    for i in indices:
        old = a[i]

        # Affected lags
        k_start = max(0, i)
        k_end = min(2 * n - 2, i + n - 1)
        affected_k = np.arange(k_start, k_end + 1)

        # Coefficients for each affected lag
        idx_in_affected = affected_k - i  # These are k-i values, all in [0, n-1]
        coeffs = 2.0 * a[idx_in_affected]

        # Check if k=2i is in affected range
        has_2i = (2 * i >= k_start) and (2 * i <= k_end)
        if has_2i:
            idx_2i = np.where(affected_k == 2 * i)[0][0]

        # Unaffected max
        mask = np.ones(2 * n - 1, dtype=bool)
        mask[affected_k] = False
        if mask.any():
            C = b[mask].max()
        else:
            C = 0.0

        b_affected_base = b[affected_k].copy()

        gr = (np.sqrt(5.0) - 1.0) / 2.0
        lo = 0.0
        hi = max(3.0 * old, S / n * 4.0, 0.05)

        def J_of_x(x):
            d = x - old
            new_b_aff = b_affected_base + 2.0 * d * coeffs
            if has_2i:
                new_b_aff[idx_2i] = b_affected_base[idx_2i] + 2.0 * d * old + d * d
            new_max = max(new_b_aff.max(), C)
            new_S = S - old + x
            if new_S <= 1e-10:
                return 1e10
            return 2.0 * n * new_max / (new_S * new_S)

        c_pt = hi - gr * (hi - lo)
        d_pt = lo + gr * (hi - lo)
        fc = J_of_x(c_pt)
        fd = J_of_x(d_pt)

        for _ in range(max_iter_per_coord):
            if fc < fd:
                hi = d_pt
                d_pt = c_pt
                fd = fc
                c_pt = hi - gr * (hi - lo)
                fc = J_of_x(c_pt)
            else:
                lo = c_pt
                c_pt = d_pt
                fc = fd
                d_pt = lo + gr * (hi - lo)
                fd = J_of_x(d_pt)

        x_opt = (lo + hi) / 2.0
        x_opt = max(0.0, x_opt)

        if abs(x_opt - old) > 1e-10:
            a[i] = x_opt
            S = S - old + x_opt
            # Update b incrementally
            d = x_opt - old
            for idx_k, k in enumerate(affected_k):
                if k == 2 * i:
                    b[k] = b[k] + 2.0 * d * old + d * d
                else:
                    b[k] = b[k] + 2.0 * d * a[k - i] + (x_opt - old) * 0  # already accounted
            # Actually let me redo this more carefully
            # b[k] new = b[k] old + (x_opt - old) * (sum of neighbors)
            # For k in affected range, k != 2i: b[k] += 2*d*a[k-i] (using OLD a[k-i], but a didn't change except at i)
            # Wait, a[k-i] might be a[i] if k-i == i, i.e., k=2i. We handle that separately.
            # For k != 2i: a[k-i] is unchanged (since we only changed a[i], and k-i != i)
            # So the formula is correct.
            improved = True

    # Verify
    b_final = fast_conv(a)
    final_J = 2.0 * n * b_final.max() / (S * S)
    return a, final_J


def coordinate_descent(a, n_passes, deadline, tracker):
    """Multiple passes of coordinate descent."""
    a = np.maximum(a.astype(np.float64), 0.0)
    a = a / a.sum()
    current_J = compute_J(a)
    tracker.update(a)

    for p in range(n_passes):
        if time.monotonic() > deadline:
            break
        t0 = time.monotonic()
        a, new_J = cd_pass_vectorized(a)
        a = a / a.sum()
        if new_J < current_J - 1e-14:
            current_J = new_J
            tracker.update(a)
            if p % 5 == 0:
                print(f"    CD pass {p}: J={current_J:.10f} ({time.monotonic()-t0:.1f}s)", flush=True)
        elif p > 5:
            break

    tracker.update(a)
    return a


def resample_seq(src, dest_n):
    """Resample 1D sequence to new length via linear interpolation."""
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)


def run_alternating(a, deadline, tracker):
    """Alternate LP and CD until deadline."""
    a = np.maximum(a.astype(np.float64), 1e-6)
    a = a / a.sum()
    current_J = compute_J(a)
    tracker.update(a)

    iteration = 0
    while time.monotonic() < deadline:
        remaining = deadline - time.monotonic()
        if remaining < 5:
            break

        # LP phase
        lp_time = min(remaining * 0.5, 60.0)
        a = run_lp_iterations(a, max_iters=50, deadline=time.monotonic() + lp_time,
                              tracker=tracker, tighten_base=0.002 + 0.001 * (1 - iteration / 100.0))

        if time.monotonic() > deadline:
            break

        # CD phase
        cd_time = min(deadline - time.monotonic(), 60.0)
        if cd_time > 3:
            n_passes = max(1, min(5, int(cd_time / 10.0)))
            a = coordinate_descent(a, n_passes, time.monotonic() + cd_time, tracker)

        iteration += 1
        if iteration % 5 == 0:
            print(f"    Alt iter {iteration}: J={compute_J(a):.10f}", flush=True)

    return a
