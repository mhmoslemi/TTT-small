    fa = np.fft.rfft(a, size)

    if HAS_CVXPY:
        try:
            g_var = cp.Variable(n, nonneg=True)
            A = build_toeplitz(a)
            constraints = [A @ g_var <= M]
            prob = cp.Problem(cp.Maximize(cp.sum(g_var)), constraints)
            for solver in [cp.CLARABEL, cp.SCS, cp.MOSEK]:
                try:
                    prob.solve(solver=solver, time_limit=int(time_limit))
                    break
                except Exception:
                    continue
            if g_var.value is not None:
                g_val = np.maximum(g_var.value, 0)
                return g_val, float(np.sum(g_val))
        except Exception:
            pass

    A = build_toeplitz(a)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': time_limit})
        if res.status == 0:
            g = np.maximum(res.x, 0)
            return g, float(np.sum(g))
    except Exception:
        pass
    return None, None

def pocs_spectral(a, max_iters=500, time_budget=20.0):
    """
    POCS-style spectral flattening.
    Projects between:
    - Set S1: {a: sum(a)=1, a>=0}
    - Set S2: {a: |FFT(a)| is approximately constant}
    """
    n = len(a)
    a = a.copy().astype(np.float64)
    s = np.sum(a)
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    # Zero-pad for FFT
    N = 1
    while N < 2 * n - 1:
        N <<= 1

    for it in range(max_iters):
        if time.monotonic() - start > time_budget:
            break

        # Forward FFT
        A = np.fft.fft(a, N)

        # We want |A| to be as flat as possible
        # Target magnitude: geometric mean or mean
        mag = np.abs(A)
        target = np.mean(mag[mag > 1e-10])  # average of non-zero magnitudes

        # Update: set magnitude to target, keep phase
        A_new = np.zeros(N, dtype=complex)
        for k in range(N):
            if mag[k] > 1e-15:
                A_new[k] = A[k] / mag[k] * target
            else:
                A_new[k] = target  # small random to avoid zero

        # Inverse FFT
        a_new = np.real(np.fft.ifft(A_new))[:n]

        # Project onto S1: non-negative, sum = 1
        a_new = np.maximum(a_new, 0)
        s_new = np.sum(a_new)
        if s_new < 1e-10:
            break
        a_new = a_new / s_new
        a = a_new

        obj = compute_obj(a)
        if obj < best_obj - 1e-14:
            best_obj = obj
            best_a = a.copy()

    return best_a

def pocs_conv_direct(a, max_iters=500, time_budget=20.0):
    """
    Direct POCS on the convolution: try to make conv(a,a) flat.
    """
    n = len(a)
    a = a.copy().astype(np.float64)
    s = np.sum(a)
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    N = 1
    while N < 2 * n - 1:
        N <<= 1

    for it in range(max_iters):
        if time.monotonic() - start > time_budget:
            break

        # Compute current convolution
        conv = compute_conv(a)

        # We want conv to be flat: conv[k] = 1/(2n-1) for all k
        # In frequency domain: A^2 should have ifft = flat on [0,2n-2]
        # This means A^2[k] = (2n-1)/N * (sum of exp(2*pi*i*k*j/N) for j=0 to 2n-2) / (2n-1)
        # = (1/N) * sum_{j=0}^{2n-2} exp(2*pi*i*k*j/N)
        # This is a sinc-like function in frequency

        # Simpler approach: use the gradient of the max
        # Find peak position
        peak_k = int(np.argmax(conv))
        peak_val = conv[peak_k]

        # Subgradient direction: reduce a[m] where a[peak_k - m] is large
        # (since grad of conv[peak_k] w.r.t. a[m] = 2*a[peak_k - m] for valid m)
        grad = np.zeros(n)
        for m in range(n):
            idx = peak_k - m
            if 0 <= idx < n:
                grad[m] += 2.0 * a[idx]
            if peak_k == 2 * m:
                # Already counted above, but the factor of 2 accounts for it

        # Actually, let's use the weighted subgradient over all active peaks
        active = conv > 0.9 * peak_val  # positions close to max
        active_k = np.where(active)[0]

        grad = np.zeros(n)
        for k in active_k:
            w = (conv[k] - 0.9 * peak_val + 1e-10) / (peak_val + 1e-10)
            for m in range(n):
                idx = k - m
                if 0 <= idx < n:
                    grad[m] += w * 2.0 * a[idx]

        # Normalize gradient
        g_norm = np.linalg.norm(grad)
        if g_norm < 1e-15:
            break
        grad = grad / g_norm

        # Multiplicative update
        step = 0.05 * (1 - it / max_iters)
        a_new = a * np.exp(-step * grad)
        a_new = np.maximum(a_new, 1e-15)
        s_new = np.sum(a_new)
        if s_new > 1e-10:
            a_new = a_new / s_new
        a = a_new

        obj = compute_obj(a)
        if obj < best_obj - 1e-14:
            best_obj = obj
            best_a = a.copy()

    return best_a

def multiplicative_subgradient(a, max_iters=3000, time_budget=15.0, tau=30.0, step=0.05):
    """
    Multiplicative subgradient descent with adaptive parameters.
    """
    n = len(a)
    a = a.copy().astype(np.float64)
    s = np.sum(a)
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    stall = 0
    patience = 100

    for it in range(max_iters):
        if time.monotonic() - start > time_budget:
            break

        conv = compute_conv(a)
        max_conv = float(np.max(conv))

        # Compute weights (softmax-like)
        shifted = tau * (conv - max_conv)
        shifted = np.clip(shifted, -500, 500)
        weights = np.exp(shifted)
        w_sum = float(np.sum(weights))
        if w_sum < 1e-300:
            break
        weights /= w_sum

        # Gradient: for each m, grad[m] = sum_k weights[k] * 2 * a[k-m] (valid k-m)
        # Use FFT/correlation for efficiency
        # grad[m] = 2 * sum_k weights[k] * a[k-m] = 2 * (weights * a_padded reversed)[m]
        # = 2 * cross-correlation of weights and a
        # Specifically: grad[m] = 2 * sum_{k: 0<=k-m<n} weights[k] * a[k-m]
        # Let j = k-m, then k = j+m: grad[m] = 2 * sum_{j=0}^{n-1} weights[j+m] * a[j]
        # This is the cross-correlation of weights and a, shifted

        # Use np.correlate: np.correlate(weights, a, 'full')[n-1 + m] = sum_j weights[j+m] * a[j]
        # Wait, let me be more careful.
        # np.correlate(v, w, 'full')[k] = sum_i v[i+k] * conj(w[i]) (for the 'full' mode)
        # Actually, np.correlate(v, w, 'full') gives:
        # result[k] = sum_i v[i] * w[i - (k - (len(w)-1))]
        # This is confusing. Let me just use the direct formula or FFT.

        # Better: use the fact that grad[m] = 2 * (weights convolved with reversed a) at appropriate position
        # Or just: grad = 2 * ifft(fft(weights, 2N) * conj(fft(a, 2N))) ... this is getting complicated.

        # Simplest correct approach:
        # grad[m] = 2 * sum_{k=m}^{m+n-1} weights[k] * a[k-m]
        # (where weights has length 2n-1, so k ranges over valid indices)
        # Let's use np.correlate(weights, a, 'full')
        # np.correlate(weights, a, 'full')[m + (n-1)] = sum_j weights[j] * a[j - (m + n - 1) + (n-1)]
        # Hmm, let me just do it directly with a matrix or use the FFT trick.

        # FFT approach:
        # We want for each m: sum_{j=0}^{n-1} weights[j+m] * a[j]
        # This is the cross-correlation. Let's use:
        # correlate(weights, a, 'full')
        # In numpy, correlate(v, w, 'full')[k] = sum_n v[n+k] * conj(w[n]) for the part where it makes sense
        # Actually for real signals: correlate(v, w, 'full')[k] = sum_n v[n+k-(len(w)-1)] * w[n]
        # This is getting confusing. Let me just use the matrix approach for correctness.

        # For n up to ~1000, we can afford O(n^2) per step if needed, but let's use FFT.
        # The vector grad[m] = sum_{j=0}^{n-1} weights[j+m] * a[j]
        # = sum_{j} a[j] * weights[j+m]
        # This is like: pad weights to length 2n, take the "cross-correlation" with a.
        # np.correlate(weights, a, 'full') gives a vector of length len(weights)+len(a)-1 = 3n-2
        # The element we want is at index (n-1) + m... let me verify with a small example.
        # weights = [w0, w1, w2], a = [a0, a1]
        # We want: grad[0] = w0*a0 + w1*a1, grad[1] = w1*a0 + w2*a1
        # np.correlate([w0,w1,w2], [a0,a1], 'full') = ?
        # According to numpy docs: correlate(v, w, mode='full')[k] = sum_n v[n + (len(w)-1-k)] * conj(w[n])
        # Hmm, this is the "cross-correlation" which is different from what I need.
        # Let me just use: correlate(a, weights, 'full')
        # correlate(a, weights, 'full')[k] = sum_n a[n + (len(weights)-1-k)] * weights[n]
        # For a = [a0,a1], weights = [w0,w1,w2]:
        # k=0: a[0+2] * w0 = a2 * w0 = 0 (out of bounds)
        # ...
        # This is getting too confusing. Let me just use the direct FFT method.

        # The cleanest way: use the Toeplitz matrix approach but with FFT
        # grad = 2 * T @ weights where T is the appropriate matrix
        # T[m, k] = a[k-m] if 0 <= k-m < n, else 0
        # This is a Toeplitz matrix! And we can compute T @ weights using FFT.

        # Actually, the simplest efficient approach for moderate n:
        # grad[m] = 2 * sum_{j=0}^{n-1} a[j] * weights[j+m]
        # = 2 * (a cross-correlated with weights, reversed and shifted)

        # Let me just use np.correlate correctly:
        # np.correlate(weights, a, 'full')[n-1-m] for m=0,...,n-1
        # Wait, let me verify:
        # np.correlate(v, w, 'full') where v has length M and w has length M:
        # result[k] = sum_n v[n] * w[n - (k - M + 1)]  ... I'll just test empirically.

        # For now, let's use the O(n^2) approach since n is at most ~1000:
        # Actually, for n=330, O(n^2) = 109000 operations per step, and we can do thousands of steps in 15 seconds. That's fine.

        grad = np.zeros(n)
        for m in range(n):
            # grad[m] = 2 * sum_{j=0}^{n-1} a[j] * weights[j+m]
            # weights has length 2n-1, so j+m ranges from m to m+n-1
            # which is valid as long as m+n-1 <= 2n-2, i.e., m <= n-1. Always true.
            grad[m] = 2.0 * np.dot(a, weights[m:m+n])

        # Normalize and update
        a_new = a * np.exp(-step * grad)
        a_new = np.maximum(a_new, 1e-15)
        s_new = np.sum(a_new)
        if s_new > 1e-10:
            a_new = a_new / s_new
        a = a_new

        obj = compute_obj(a)
        if obj < best_obj - 1e-13:
            best_obj = obj
            best_a = a.copy()
            stall = 0
        else:
            stall += 1
            if stall > patience:
                step = max(step * 0.7, 0.001)
                tau *= 1.1
                if step < 0.002:
                    break
                # Perturb to escape
                rng = np.random.default_rng(it * 31 + 7)
                a = a * np.exp(rng.normal(0, 0.05, n))
                a = np.maximum(a, 1e-10)
                s = np.sum(a)
                if s > 1e-10:
                    a = a / s
                stall = 0
                patience = max(patience // 2, 50)

        if it > 500:
            step *= 0.999

    return best_a

def alternating_lp(a, max_iters=60, time_budget=30.0):
    """
    Alternating LP: alternate between optimizing a (given g) and g (given a).
    At each step, solve the min-peak LP and do a line search.
    """
    n = len(a)
    a = a.copy().astype(np.float64)
    s = np.sum(a)
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    stall = 0

    for it in range(max_iters):
        if time.monotonic() - start > time_budget - 3:
            break

        # Solve LP: find best g given a
        g, M_opt = solve_lp_minpeak(a, time_limit=8)
        if g is None:
            break

        # Line search: blend a and g
        best_t = 0.0
        best_obj_ls = best_obj
        for t in [0.3, 0.5, 0.7, 0.85, 1.0]:
            a_new = (1.0 - t) * a + t * g
            a_new = np.maximum(a_new, 0)
            s_new = float(np.sum(a_new))
            if s_new < 1e-10:
                continue
            a_new = a_new / s_new
            obj_new = compute_obj(a_new)
            if obj_new < best_obj_ls:
                best_obj_ls = obj_new
                best_t = t

        if best_t > 0 and best_obj_ls < best_obj - 1e-12:
            a_new = (1.0 - best_t) * a + best_t * g
            a_new = np.maximum(a_new, 0)
            s_new = float(np.sum(a_new))
            if s_new > 1e-10:
                a_new = a_new / s_new
            best_obj = best_obj_ls
            best_a = a_new.copy()
            a = a_new
            stall = 0
        else:
            stall += 1
            if stall > 20:
                break
            # Forced blend
            a = 0.9 * a + 0.1 * g
            a = np.maximum(a, 0)
            s = float(np.sum(a))
            if s > 1e-10:
                a = a / s

    return best_a

def alternating_lp_maxsum(a, max_iters=60, time_budget=30.0):
    """
    Alternating LP using max-sum formulation.
    """
    n = len(a)
    a = a.copy().astype(np.float64)
    s = np.sum(a)
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    stall = 0

    for it in range(max_iters):
        if time.monotonic() - start > time_budget - 3:
            break

        conv = compute_conv(a)
        M = float(np.max(conv))

        g, sum_g = solve_lp_maxsum(a, M, time_limit=8)
        if g is None or sum_g < 1e-10:
            break

        g_norm = g / sum_g

        best_t = 0.0
        best_obj_ls = best_obj
        for t in [0.3, 0.5, 0.7, 0.85, 1.0]:
            a_new = (1.0 - t) * a + t * g_norm
            a_new = np.maximum(a_new, 0)
            s_new = float(np.sum(a_new))
            if s_new < 1e-10:
                continue
            a_new = a_new / s_new
            obj_new = compute_obj(a_new)
            if obj_new < best_obj_ls:
                best_obj_ls = obj_new
                best_t = t

        if best_t > 0 and best_obj_ls < best_obj - 1e-12:
            a_new = (1.0 - best_t) * a + best_t * g_norm
            a_new = np.maximum(a_new, 0)
            s_new = float(np.sum(a_new))
            if s_new > 1e-10:
                a_new = a_new / s_new
            best_obj = best_obj_ls
            best_a = a_new.copy()
            a = a_new
            stall = 0
        else:
            stall += 1
            if stall > 20:
                break
            a = 0.9 * a + 0.1 * g_norm
            a = np.maximum(a, 0)
            s = float(np.sum(a))
            if s > 1e-10:
                a = a / s

    return best_a

def fine_tune(a, time_budget=20.0):
    """
    Fine tuning with very small multiplicative steps.
    """
    n = len(a)
    a = a.copy().astype(np.float64)
    s = np.sum(a)
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    step = 0.005
    tau = 80.0
    stall = 0

    while time.monotonic() - start < time_budget:
        conv = compute_conv(a)
        max_conv = float(np.max(conv))

        shifted = tau * (conv - max_conv)
        shifted = np.clip(shifted, -500, 500)
        weights = np.exp(shifted)
        w_sum = float(np.sum(weights))
        if w_sum < 1e-300:
            break
        weights /= w_sum

        grad = np.zeros(n)
        for m in range(n):
            grad[m] = 2.0 * np.dot(a, weights[m:m+n])

        a_new = a * np.exp(-step * grad)
        a_new = np.maximum(a_new, 1e-15)
        s_new = float(np.sum(a_new))
        if s_new > 1e-10:
            a_new = a_new / s_new
        a = a_new

        obj = compute_obj(a)
        if obj < best_obj - 1e-14:
            best_obj = obj
            best_a = a.copy()
            stall = 0
        else:
            stall += 1
            if stall > 200:
                step *= 0.7
                tau *= 1.3
                if step < 0.0005:
                    break
                stall = 0

    return best_a

def random_init(n, seed):
    """Generate diverse random initializations."""
    rng = np.random.default_rng(seed)
    k = np.arange(n)
    choice = rng.integers(0, 12)

    if choice == 0:
        f = rng.uniform(0.5, 4.0)
        phase = rng.uniform(0, 2 * np.pi)
        a = 1.0 + 0.8 * np.cos(2 * np.pi * f * k / n + phase)
    elif choice == 1:
        f1 = rng.uniform(0.3, 2.0)
        f2 = rng.uniform(2.0, 6.0)
        a = 1.0 + 0.4 * np.cos(2 * np.pi * f1 * k / n) + 0.3 * np.cos(2 * np.pi * f2 * k / n)
    elif choice == 2:
        center = n * rng.uniform(0.3, 0.7)
        alpha = rng.uniform(0.5, 3.0)
        a = 1.0 + 0.5 * np.exp(-alpha * np.abs(k - center) / (n * 0.3))
    elif choice == 3:
        a = rng.random(n)
        win = max(3, n // 20)
        kernel = np.ones(win) / win
        a = np.convolve(a, kernel, mode='same')
    elif choice == 4:
        a = np.ones(n)
        for _ in range(rng.integers(1, 4)):
            f = rng.uniform(0.5, 5.0)
            amp = rng.uniform(0.1, 0.4)
            a += amp * np.cos(2 * np.pi * f * k / n + rng.uniform(0, 2 * np.pi))
    elif choice == 5:
        a = np.ones(n)
        for scale in range(3):
            f = rng.uniform(0.3 * (2 ** scale), 1.5 * (2 ** scale))
            amp = 0.3 / (scale + 1)
            a += amp * np.cos(2 * np.pi * f * k / n + rng.uniform(0, 2 * np.pi))
    elif choice == 6:
        # Triangular wave
        period = rng.uniform(n/4, n)
        a = 1.0 + 0.5 * np.abs(2 * ((k % period) / period - 0.5))
    elif choice == 7:
        # Chirp
        f0 = rng.uniform(0.3, 1.0)
        f1 = rng.uniform(3.0, 8.0)
        t = k / n
        f = f0 + (f1 - f0) * t
        a = 1.0 + 0.5 * np.cos(2 * np.pi * np.cumsum(f))
        a = np.maximum(a, 0)
    elif choice == 8:
        # Random smooth (interpolated noise)
        a = rng.random(max(5, n // 10))
        a = np.interp(k, np.arange(len(a)), a)
        a = 1.0 + a
    elif choice == 9:
        # Bimodal
        c1 = n * rng.uniform(0.2, 0.4)
        c2 = n * rng.uniform(0.6, 0.8)
        w = n * rng.uniform(0.05, 0.15)
        a = 1.0 + 0.8 * np.exp(-0.5*((k-c1)/w)**2) + 0.8 * np.exp(-0.5*((k-c2)/w)**2)
    elif choice == 10:
        # Power law
        gamma = rng.uniform(0.5, 2.0)
        a = (k + 1) ** (-gamma)
    else:
        # Symmetric random
        half = n // 2
        a_half = rng.random(half + 1)
        a = np.zeros(n)
        a[:half+1] = a_half
        a[half:] = a_half[:half][::-1] if n % 2 == 0 else a_half[half-1::-1]
        a = 0.5 + a

    a = np.maximum(a, 0)
    s = float(np.sum(a))
    if s < 1e-10:
        a = np.ones(n)
        s = n
    return a / s

def optimize_single(a, time_budget=60.0, verbose=False):
    """Run full optimization pipeline on a single sequence."""
    n = len(a)
    a = a.copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        a = np.ones(n) / n
        s = 1.0
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    if verbose:
        print(f"  Start: obj={best_obj:.6f} (n={n})")

    # Phase 1: Quick subgradient
    remaining = time_budget - (time.monotonic() - start)
    if remaining > 5:
        sg_time = min(10.0, remaining * 0.15)
        a_new = multiplicative_subgradient(best_a, max_iters=int(sg_time * 30),
                                            time_budget=sg_time, tau=20.0, step=0.06)
        obj = compute_obj(a_new)
        if obj < best_obj:
            best_obj = obj
            best_a = a_new.copy()
            if verbose:
                print(f"  After SG1: {best_obj:.6f}")

    # Phase 2: POCS spectral
    remaining = time_budget - (time.monotonic() - start)
    if remaining > 10:
        pocs_time = min(15.0, remaining * 0.2)
        a_new = pocs_spectral(best_a, max_iters=300, time_budget=pocs_time)
        obj = compute_obj(a_new)
        if obj < best_obj:
            best_obj = obj
            best_a = a_new.copy()
            if verbose:
                print(f"  After POCS: {best_obj:.6f}")

    # Phase 3: Alternating LP (min-peak)
    remaining = time_budget - (time.monotonic() - start)
    if remaining > 15:
        lp_time = min(25.0, remaining * 0.35)
        a_new = alternating_lp(best_a, max_iters=40, time_budget=lp_time)
        obj = compute_obj(a_new)
        if obj < best_obj:
            best_obj = obj
            best_a = a_new.copy()
            if verbose:
