    # A[k,j] = a[k-j] for 0 <= k-j < n
    rows = []

"""
Optimized search minimizing J(a) = 2*n*max(conv(a,a))/sum(a)^2.
Strategy: improved LP mirror descent with adaptive step sizes, multi-length
sweep, CMA-ES on Fourier parameters for global exploration, and L-BFGS-B polish.
Key improvements: aggressive backtracking line search, no premature step-size
decay, periodic restarts with perturbation, and broader length exploration.

import numpy as np
from scipy.optimize import linprog, minimize
from scipy.sparse import csr_matrix
import time


def fast_conv(a):
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]


def compute_J(a):
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))


def build_A_sparse(n, a):
    nnz = n * n
    rows = np.empty(nnz, dtype=np.int32)
    cols = np.empty(nnz, dtype=np.int32)
    vals = np.empty(nnz, dtype=np.float64)
    idx = 0
    for k in range(2 * n - 1):
        j_min = max(0, k - n + 1)
        j_max = min(n - 1, k)
        count = j_max - j_min + 1
        rows[idx:idx + count] = k
        cols[idx:idx + count] = np.arange(j_min, j_max + 1)
        vals[idx:idx + count] = a[k - np.arange(j_min, j_max + 1)]
        idx += count
    return csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))


def solve_lp(a, M):
    n = len(a)
    A = build_A_sparse(n, a)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    try:
        r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
        if r.success and r.x is not None:
            g = np.maximum(r.x, 0)
            if g.sum() > 1e-15:
                return g
    except Exception:
        pass
    return None


def lp_mirror_descent(a, deadline, max_no_improve=60, mode='linear'):
    """LP mirror descent with adaptive step size and restarts."""
    a = np.maximum(a.astype(np.float64), 1e-10)
    a = a / a.sum()
    current_J = compute_J(a)
    best_J = current_J
    best_a = a.copy()
    no_improve = 0
    n = len(a)
    it = 0

    while time.monotonic() < deadline:
        it += 1
        b = fast_conv(a)
        M = float(b.max())
        g = solve_lp(a, M)
        if g is None or g.sum() < 1e-10:
            break
        g = np.maximum(g, 0.0)
        g = g / g.sum()

        # Adaptive initial step size based on recent improvement
        gap = current_J - best_J
        if gap > 1e-4:
            t_start = 0.5
        elif gap > 1e-6:
            t_start = 0.3
        elif gap > 1e-8:
            t_start = 0.15
        else:
            t_start = 0.08

        t = t_start
        best_trial_J = current_J
        best_trial_a = None

        for _ in range(25):
            if mode == 'linear':
                a_trial = (1.0 - t) * a + t * g
            else:
                ratio = g / np.maximum(a, 1e-15)
                a_trial = a * np.power(ratio, t)
            a_trial = np.maximum(a_trial, 0.0)
            s = a_trial.sum()
            if s < 1e-10:
                t *= 0.5
                continue
            a_trial = a_trial / s
            J_trial = compute_J(a_trial)
            if J_trial < best_trial_J - 1e-16:
                best_trial_J = J_trial
                best_trial_a = a_trial.copy()
                break
            t *= 0.6
            if t < 0.002:
                break

        if best_trial_a is not None:
            a = best_trial_a
            current_J = best_trial_J
            if current_J < best_J:
                best_J = current_J
                best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1
            if no_improve > max_no_improve:
                # Try a perturbation before giving up
                if no_improve == max_no_improve + 10:
                    a = np.maximum(a + np.random.default_rng(it).normal(0, 0.02, n), 0.01)
                    a = a / a.sum()
                    current_J = compute_J(a)
                    no_improve = 0
                elif no_improve > max_no_improve + 30:
                    break

    return best_a, best_J


def resample_seq(src, dest_n):
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)


def fourier_to_seq(params, n, n_harm):
    t = np.arange(n, dtype=np.float64) / n
    a = np.full(n, params[0], dtype=np.float64)
    for k in range(1, n_harm + 1):
        a += params[2 * k - 1] * np.cos(2.0 * np.pi * k * t)
        a += params[2 * k] * np.sin(2.0 * np.pi * k * t)
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-8:
        a = np.ones(n) / n
    else:
        a = a / s
    return a


def seq_to_fourier(a, n_harm):
    n = len(a)
    a_norm = a / a.sum()
    t = np.arange(n, dtype=np.float64) / n
    params = np.zeros(2 * n_harm + 1)
    params[0] = np.mean(a_norm)
    for k in range(1, n_harm + 1):
        params[2 * k - 1] = 2.0 * np.mean(a_norm * np.cos(2.0 * np.pi * k * t))
        params[2 * k] = 2.0 * np.mean(a_norm * np.sin(2.0 * np.pi * k * t))
    return params


def cmaes_eigen(C, z, dim):
    eigvals, eigvecs = np.linalg.eigh(C)
    eigvals = np.maximum(eigvals, 1e-12)
    root = np.sqrt(eigvals)
    return (eigvecs * root) @ (eigvecs.T @ z)


def run_cmaes_fourier(n, n_harm, params0, sigma0, deadline):

    lam = 4 + int(3 * np.log(dim))
    mu = lam // 2

    weights = weights / weights.sum()
    mu_eff = 1.0 / np.sum(weights ** 2)
    c_sigma = (mu_eff + 2) / (dim + mu_eff + 5)
    c_c = (4 + mu_eff / dim) / (dim + 4 + 2 * mu_eff / dim)
    d_sigma = 1 + 2 * max(0, np.sqrt((mu_eff - 1) / (dim + 1) - 1) - 1) + c_sigma
    c_1 = 2 / ((dim + 1.3) ** 2 + mu_eff)
    c_mu = min(1 - c_1, 2 * (mu_eff - 2 + 1.0 / mu_eff) / ((dim + 2) ** 2 + mu_eff))
    chi_N = np.sqrt(dim) * (1 - 1 / (4 * dim) + 1 / (21 * dim ** 2))

    C = np.eye(dim)
    Cs = np.zeros((dim, dim))
    p_sigma = np.zeros(dim)
    p_c = np.zeros(dim)

    x = params0.copy()
    sigma = sigma0

    a_init = fourier_to_seq(params0, n, n_harm)
    best_J = compute_J(a_init)
    best_x = params0.copy()

    rng = np.random.default_rng(12345)

    for gen in range(5000):
        if time.monotonic() > deadline:
            break

        offspring_J = []
        for i in range(lam):

            a_trial = fourier_to_seq(x_trial, n, n_harm)

        for i in range(mu):

        else:
            diff = np.zeros(dim)
        p_sigma = (1 - c_sigma) * p_sigma + np.sqrt(c_sigma * (2 - c_sigma) * mu_eff) * diff
        sigma *= np.exp(c_sigma / d_sigma * (np.linalg.norm(p_sigma) / chi_N - 1))

        sigma = min(sigma, 1.0)

            best_x = offspring[idx[0]].copy()

def subgradient_polish(a, deadline, max_iters=50000):
    a = np.maximum(a.astype(np.float64), 1e-10)
    a = a / a.sum()
    best_M = fast_conv(a).max()
    best_a = a.copy()
    n = len(a)
    momentum = np.zeros_like(a)
    beta = 0.8

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        b = fast_conv(a)
        M = b.max()
        tol = 1e-10 * M
        active = np.where(b >= M - tol)[0]

        grad = np.zeros_like(a)
        for k in active:
            lo = max(0, k - n + 1)
            hi = min(n - 1, k)
            for i in range(lo, hi + 1):
                grad[i] += 2.0 * a[k - i]

        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-15:
            break

        t = 2.0 / (gnorm * (it + 1.0) ** 0.55)
        momentum = beta * momentum + (1 - beta) * grad / gnorm
        a_new = a - t * momentum
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s < 1e-10:
            break
        a_new = a_new / s

        new_M = fast_conv(a_new).max()
        if new_M < best_M:
            best_M = new_M
            best_a = a_new.copy()
        a = a_new

    return best_a

    candidate_lengths = sorted(set([
        max(256, base_n - 128),
        max(256, base_n - 64),
        base_n,
        min(1200, base_n + 64),
        min(1200, base_n + 128),
        min(1200, base_n + 256),
    ]))
    candidate_lengths = [n for n in candidate_lengths if 256 <= n <= 1200]

    n_lengths = len(candidate_lengths)
    time_per_length = max(30, min(120, (deadline - (t_global_start + float(budget_s) - 20.0)) - 1000) / n_lengths))
    # Actually compute based on remaining time after reserved later phases
    reserved_later = 600  # seconds reserved for later phases
    time_for_phase1 = max(60, min(400, float(budget_s) - 20 - reserved_later))
    time_per_length = time_for_phase1 / n_lengths

    for i, n_try in enumerate(candidate_lengths):
        if time.monotonic() > deadline - reserved_later:
            break
        phase1_dl = time.monotonic() + time_per_length

        # Generate a few diverse inits for this length
        inits = generate_diverse_inits(n_try, rng, parent)

        best_a_len = None
        best_J_len = float('inf')

        # Run LP on top 3 diverse inits
        for j, a_init in enumerate(inits[:4]):
            if time.monotonic() > phase1_dl - 5:
                break
            sub_dl = min(phase1_dl - 2, time.monotonic() + time_per_length / 4)

            a_result, J_result = lp_mirror_descent(a_init, sub_dl, max_no_improve=40, mode='linear')
            if J_result < best_J_len:
                best_J_len = J_result
                best_a_len = a_result.copy()
            if j == 0:
                # Also try multiplicative on the best
                if time.monotonic() < phase1_dl - 10:
                    a_mult, J_mult = lp_mirror_descent(a_init, min(phase1_dl - 2, time.monotonic() + time_per_length / 4), max_no_improve=40, mode='mult')
                    if J_mult < best_J_len:
                        best_J_len = J_mult
                        best_a_len = a_mult.copy()

        if best_a_len is not None:
            update_tracker(best_a_len)
            print(f"  n={n_try}: J={best_J_len:.8f}", flush=True)

    print(f"Phase 1 done. Best so far: J={best_score:.8f}", flush=True)

    # === Phase 2: Intensive LP on best candidate ===
    print(f"\n{'='*60}", flush=True)
    print(f"Phase 2: Intensive LP refinement", flush=True)
    print(f"{'='*60}", flush=True)

    if best_overall is not None:
        a_refine = best_overall.copy()
        phase2_dl = min(deadline - 300, time.monotonic() + 350)

        # Linear mode
        a2, J2 = lp_mirror_descent(a_refine, min(phase2_dl, time.monotonic() + 180), max_no_improve=80, mode='linear')
        update_tracker(a2)
        print(f"  Linear: J={J2:.8f}", flush=True)

        # Multiplicative mode from the best
        if time.monotonic() < phase2_dl - 30:
            a_refine2 = best_overall.copy()
            a2m, J2m = lp_mirror_descent(a_refine2, min(phase2_dl - 5, time.monotonic() + 120), max_no_improve=80, mode='mult')
            update_tracker(a2m)
            print(f"  Mult: J={J2m:.8f}", flush=True)

    # === Phase 3: CMA-ES global exploration ===
    print(f"\n{'='*60}", flush=True)
    print(f"Phase 3: CMA-ES Fourier exploration", flush=True)
    print(f"{'='*60}", flush=True)

    if best_overall is not None and time.monotonic() < deadline - 250:
        n_cma = best_overall.size
        n_harm = min(30, n_cma // 4)
        params0 = seq_to_fourier(best_overall, n_harm)
        phase3_dl = min(deadline - 200, time.monotonic() + 200)

        a_cma, J_cma = run_cmaes_fourier(n_cma, n_harm, params0, 0.03, phase3_dl)
        update_tracker(a_cma)
        print(f"  CMA-ES: J={J_cma:.8f}", flush=True)

        # LP polish from CMA result
        if time.monotonic() < deadline - 150:
            a_cma_lp, J_cma_lp = lp_mirror_descent(a_cma, min(deadline - 130, time.monotonic() + 100), max_no_improve=60, mode='linear')
            update_tracker(a_cma_lp)
            print(f"  CMA+LP: J={J_cma_lp:.8f}", flush=True)

    # === Phase 4: Subgradient polish ===
    print(f"\n{'='*60}", flush=True)
    print(f"Phase 4: Subgradient polish", flush=True)
    print(f"{'='*60}", flush=True)

    if best_overall is not None and time.monotonic() < deadline - 100:
        a_sub = subgradient_polish(best_overall, min(deadline - 80, time.monotonic() + 60))
        update_tracker(a_sub)
        J_sub = compute_J(a_sub)
        print(f"  Subgrad: J={J_sub:.8f}", flush=True)

        # LP from subgradient result
        if time.monotonic() < deadline - 60 and J_sub < best_score + 1e-9:
            a_sub_lp, J_sub_lp = lp_mirror_descent(a_sub, min(deadline - 50, time.monotonic() + 40), max_no_improve=40, mode='linear')
            update_tracker(a_sub_lp)
            print(f"  Subgrad+LP: J={J_sub_lp:.8f}", flush=True)

    # === Phase 5: Final LP push ===
    print(f"\n{'='*60}", flush=True)
    print(f"Phase 5: Final LP push", flush=True)
    print(f"{'='*60}", flush=True)

    if best_overall is not None and time.monotonic() < deadline - 30:
        a_final = best_overall.copy()
        a_final, J_final = lp_mirror_descent(a_final, min(deadline - 25, time.monotonic() + 80), max_no_improve=50, mode='linear')
        update_tracker(a_final)
        print(f"  Final LP: J={J_final:.8f}", flush=True)

    # === Phase 6: L-BFGS-B polish ===
    print(f"\n{'='*60}", flush=True)
    print(f"Phase 6: L-BFGS-B", flush=True)
    print(f"{'='*60}", flush=True)

    if best_overall is not None and time.monotonic() < deadline - 5:
        n_final = best_overall.size
        try:
            res = minimize(lbb_objective, best_overall.astype(np.float64),
                           method='L-BFGS-B', bounds=[(0.0, None)] * n_final,
                           options={'maxiter': 200, 'max_fun': 50000, 'ftol': 1e-16})
            if res.fun < 1e10:
                a_lbb = np.maximum(res.x, 0.0)
                a_lbb = a_lbb / a_lbb.sum()
                J_lbb = compute_J(a_lbb)
                if J_lbb < best_score:
                    best_overall = a_lbb.copy()
                    best_score = J_lbb
                    print(f"  L-BFGS-B: J={J_lbb:.8f}", flush=True)
                else:
                    print(f"  L-BFGS-B: J={J_lbb:.8f} (no improvement)", flush=True)
        except Exception as e:
            print(f"  L-BFGS-B error: {e}", flush=True)

    # Final decision
    if best_overall is not None:
        final_a = np.clip(best_overall, 0.0, 1000.0)
        final_score = evaluate_sequence(final_a.tolist())
        print(f"\n{'='*60}", flush=True)
        print(f"FINAL: J={final_score:.8f} n={final_a.size} "
              f"compute_J={best_score:.8f} time={time.monotonic()-t_global_start:.0f}s", flush=True)
        print(f"{'='*60}", flush=True)
        return [float(x) for x in final_a]
    else:
        # Fallback
        fallback = np.ones(640) / 640
        return [float(x) for x in fallback]
