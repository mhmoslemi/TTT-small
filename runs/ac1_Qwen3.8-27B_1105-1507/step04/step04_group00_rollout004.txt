import numpy as np
from scipy.optimize import linprog, minimize
from scipy.sparse import csr_matrix
import time

def fast_conv(a):
    """Full self-convolution via FFT."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]

def compute_J(a):
    """Compute J(a) = 2*n*max(conv(a,a))/(sum(a))^2."""
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))

def coordinate_update(a, i):
    """Find optimal value for a[i] given all other coordinates fixed.
    Returns the optimal value x >= 0 that minimizes max(conv)."""
    n = len(a)
    x = a[i]
    
    # Compute b = conv(a,a)
    b = fast_conv(a)
    
    # Determine affected range: k in [i, min(2n-2, i+n-1)]
    k_lo = i
    k_hi = min(2*n-2, i+n-1)
    
    # Compute c_k = b_k - delta_k for affected k
    # delta_k = 2*x*a[k-i] if k != 2i, or x^2 if k == 2i
    
    # Unaffected max
    if k_lo > 0 or k_hi < 2*n-2:
        mask = np.ones(2*n-1, dtype=bool)
        mask[k_lo:k_hi+1] = False
        if mask.any():
            base_max = b[mask].max()
        else:
            base_max = 0.0
    else:
        base_max = 0.0
    
    # For affected k, compute c_k and the linear coefficients
    affected_ks = np.arange(k_lo, k_hi+1)
    c_vals = np.zeros(len(affected_ks))
    slopes = np.zeros(len(affected_ks))  # coefficient of x
    quad_indices = []  # index in affected_ks where k == 2i

            slopes[idx] = 0.0

                c_vals[idx] = b[k] - 2*x*a[j]
                slopes[idx] = 2*a[j]
            else:

"""
Minimizes J(a) = 2*n*max(conv(a,a))/(sum(a))^2 using coordinate descent as the primary
optimization engine, combined with smooth gradient descent (Adam) and LP mirror descent
for final polish. Key innovation: exact 1D coordinate updates via golden-section search
exploit the structure of self-convolution, achieving O(n^2 log n) per full sweep with
no approximation error in the subproblem solution.
"""

import numpy as np
from scipy.optimize import linprog, minimize
from scipy.sparse import csr_matrix
import time


def fast_conv(a):
    """Full self-convolution via FFT."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]


def compute_J(a):
    """Compute J(a) = 2*n*max(conv(a,a))/(sum(a))^2."""
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))


def coordinate_update(a, i):
    """
    Find optimal value for a[i] (with normalization) given all other coordinates.
    Minimizes max(conv(a_new, a_new)) where a_new = a with a[i]=x, then normalized.
    Returns optimal x >= 0.
    """
    n = len(a)
    x_cur = a[i]
    S = a.sum()
    
    # Compute current convolution
    b = fast_conv(a)
    
    # Affected range: k where a[i] contributes to b[k]
    k_lo = i
    k_hi = min(2 * n - 2, i + n - 1)
    
    # Unaffected max (k outside [k_lo, k_hi])
    if k_lo > 0 or k_hi < 2 * n - 2:
        mask = np.ones(2 * n - 1, dtype=bool)
        mask[k_lo:k_hi + 1] = False
        if mask.any():
            base_max = b[mask].max()
        else:
            base_max = 0.0
    else:
        base_max = 0.0
    
    # For affected k, precompute c_k and slope
    affected_arange = np.arange(k_lo, k_hi + 1)
    c_vals = np.zeros(len(affected_arange))
    slopes = np.zeros(len(affected_arange))
    quad_idx = -1
    
    for idx, k in enumerate(affected_arange):
        if k == 2 * i:
            c_vals[idx] = b[k] - x_cur * x_cur
            slopes[idx] = 0.0
            quad_idx = idx
        else:
            j = k - i
            if 0 <= j < n:
                c_vals[idx] = b[k] - 2 * x_cur * a[j]
                slopes[idx] = 2 * a[j]
            else:
                c_vals[idx] = b[k]
                slopes[idx] = 0.0
    
    def obj(x_test):
        """Objective: max(conv) after setting a[i]=x_test and normalizing to sum=1."""
        new_sum = S - x_cur + x_test
        if new_sum <= 0:
            return 1e10
        scale = 1.0 / new_sum
        # b'_k = (c_k + delta_k(x_test)) * scale^2
        # But we want max of the normalized conv, which is max(b') / new_sum^2
        # Actually: a_new = a_new_raw / new_sum, so conv(a_new) = conv(a_new_raw) / new_sum^2
        # max(conv(a_new)) = max(conv(a_new_raw)) / new_sum^2
        # conv(a_new_raw)_k = c_k + delta_k(x_test) for affected k, b_k for others
        vals = c_vals + slopes * x_test
        m = base_max
        for idx2 in range(len(vals)):
            v = vals[idx2]
            if v > m:
                m = v
        if quad_idx >= 0:
            qv = c_vals[quad_idx] + x_test * x_test
            if qv > m:
                m = qv
        return m / (new_sum * new_sum)
    
    # Golden section search
    lo = 0.0
    hi = max(x_cur * 3.0, np.sqrt(max(b.max(), 0)) + 0.5, 0.1)
    
    # Expand hi if needed
    f_mid = obj(hi * 0.5)
    f_hi = obj(hi)
    if f_hi < f_mid and hi < 10.0:
        hi *= 2.0
    
    gr = (np.sqrt(5) - 1) / 2
    a_pt = hi - gr * (hi - lo)
    b_pt = lo + gr * (hi - lo)
    fa = obj(a_pt)
    fb = obj(b_pt)
    
    for _ in range(50):
        if fa < fb:
            hi = b_pt
            b_pt = a_pt
            fb = fa
            a_pt = hi - gr * (hi - lo)
            fa = obj(a_pt)
        else:
            lo = a_pt
            a_pt = b_pt
            fa = fb
            b_pt = lo + gr * (hi - lo)
            fb = obj(b_pt)
    
    x_opt = (lo + hi) / 2.0
    # Check boundary
    if obj(0.0) < obj(x_opt):
        x_opt = 0.0
    
    return max(0.0, x_opt)


def run_coordinate_descent(a, max_sweeps, deadline, shuffle=True, seed=0):
    """Run coordinate descent sweeps to minimize max(conv) with sum(a)=1."""
    rng_c = np.random.default_rng(seed)
    a = np.maximum(a.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-10:
        a = np.ones_like(a)
    else:
        a = a / s
    n = len(a)
    
    best_a = a.copy()
    best_M = fast_conv(a).max()
    no_improve = 0
    
    order = np.arange(n)
    
    for sweep in range(max_sweeps):
        if time.monotonic() > deadline:
            break
        if shuffle:
            rng_c.shuffle(order)
        
        sweep_improved = False
        for i in order:
            x_opt = coordinate_update(a, i)
            if abs(x_opt - a[i]) > 1e-12:
                a[i] = x_opt
                sweep_improved = True
        
        # Renormalize
        s = a.sum()
        if s < 1e-10:
            break
        a = a / s
        
        M = fast_conv(a).max()
        if M < best_M * (1 - 1e-10):
            best_M = M
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        if not sweep_improved:
            no_improve += 5
        if no_improve > 30:
            break
    
    return best_a


def smooth_max_gradient_adaptive(a, tau):
    """Compute gradient of smooth max (log-sum-exp) of convolution."""
    n = len(a)
    b = fast_conv(a)
    b_max = b.max()
    exp_b = np.exp(tau * (b - b_max))
    p = exp_b / exp_b.sum()
    
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        pk = p[k]
        if pk < 1e-13:
            continue
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        grad[lo:hi + 1] += 2.0 * pk * a[k - hi:k - lo + 1][::-1]
    
    return grad


def run_adam_smooth(a, max_iters, deadline, tau_start=5.0, tau_end=100.0,
                    lr=0.05, beta1=0.9, beta2=0.999):
    """Adam optimizer on smooth max objective."""
    a = np.maximum(a.astype(np.float64), 1e-8)
    a = a / a.sum()
    n = len(a)
    
    best_a = a.copy()
    best_M = fast_conv(a).max()
    
    m = np.zeros(n)  # first moment
    v = np.zeros(n)  # second moment
    
    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        
        frac = it / max(1, max_iters)
        tau = tau_start + (tau_end - tau_start) * (frac ** 0.7)
        lr_t = lr / (1 + 0.01 * it)
        
        grad = smooth_max_gradient_adaptive(a, tau)
        
        # Renormalize: the constraint sum(a)=1 means we should project gradient
        # onto the tangent space
        grad_mean = grad.mean()
        grad = grad - grad_mean  # project to sum-zero subspace
        
        # Adam update
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * grad * grad
        m_hat = m / (1 - beta1 ** (it + 1))
        v_hat = v / (1 - beta2 ** (it + 1))
        
        a_new = a - lr_t * m_hat / (np.sqrt(v_hat) + 1e-8)
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s < 1e-10:
            a_new = a.copy()
        else:
            a_new = a_new / s
        a = a_new
        
        M = fast_conv(a).max()
        if M < best_M * (1 - 1e-10):
            best_M = M
            best_a = a.copy()
    
    return best_a


def build_A_sparse(n, a):
    """Build sparse constraint matrix for LP."""
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        j_min = max(0, k - n + 1)
        j_max = min(n - 1, k)
        for j in range(j_min, j_max + 1):
            rows.append(k)
            cols.append(j)
            vals.append(a[k - j])
    A = csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))
    return A


def lp_best_response(a, M):
    """LP: max sum(g) s.t. (a*g)[k] <= M for all k, g >= 0."""
    n = len(a)
    A = build_A_sparse(n, a)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    try:
        r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                    options={'time_limit': 30})
        if r.success and r.x is not None:
            g = np.maximum(r.x, 0)
            if g.sum() > 1e-15:
                return g
    except Exception:
        pass
    return None


def run_lp_mirror_descent(a, max_iters, deadline):
    """LP mirror descent with geometric line search."""
    a = np.maximum(a.astype(np.float64), 1e-8)
    a = a / a.sum()
    best_a = a.copy()
    best_M = fast_conv(a).max()
    no_improve = 0
    
    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        
        b = fast_conv(a)
        M = float(b.max())
        g = lp_best_response(a, M)
        if g is None:
            break
        g = np.maximum(g, 0.0)
        g_sum = g.sum()
        if g_sum < 1e-10:
            break
        g /= g_sum
        
        improved = False
        base_t = max(0.05, 0.6 * (1 - 0.3 * it / max(1, max_iters)))
        for t in [base_t, base_t * 0.5, base_t * 0.25, 0.02, 0.01]:
            a_trial = np.maximum((1 - t) * a + t * g, 0)
            s = a_trial.sum()
            if s < 1e-10:
                continue
            a_trial /= s
            M_trial = fast_conv(a_trial).max()
            if M_trial < best_M - 1e-14:
                a = a_trial
                best_M = M_trial
                best_a = a.copy()
                improved = True
                break
        
        if not improved:
            no_improve += 1
            if no_improve > 15:
                break
        else:
            no_improve = 0
    
    return best_a


def resample_seq(src, dest_n):
    """Resample 1D sequence to new length via linear interpolation."""
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)


def diverse_inits(n, rng, parent=None):
    """Generate diverse initial sequences of length n."""
    inits = []
    t = np.linspace(0, 1, n, endpoint=False)
    
    if parent is not None:
        p = resample_seq(parent, n)
        inits.append(p.copy())
        inits.append(np.maximum(p + rng.normal(0, 0.05, n), 0.01))
        inits.append(np.maximum(p + rng.normal(0, 0.15, n), 0.01))
    
    inits.append(np.ones(n))
    inits.append(rng.uniform(0.5, 1.0, n))
    
    for freq in [1, 2, 3, 5]:
        inits.append(np.maximum(1.0 + 0.4 * np.cos(2 * np.pi * freq * t), 0.01))
    
    inits.append(np.exp(-4 * (t - 0.5) ** 2) + 0.1)
    inits.append(0.3 + 0.7 * np.abs(2 * t - 1))
    
    # Smooth random
    ksize = max(3, n // 30)
    kernel = np.ones(ksize) / ksize
    smooth = np.convolve(rng.uniform(0, 1, n), kernel, mode='same')
    inits.append(np.maximum(smooth + 0.1, 0.01))
    
    return inits

    print(f"{'='*60}", flush=True)
    
    if tracker.best is not None and time.monotonic() < deadline - 30:
        n_best = tracker.best.size
        est_lp_time = 0.005 * (n_best / 128) ** 2
        max_iters = max(20, min(300, int(180 / max(0.01, est_lp_time))))
        print(f"  n={n_best}, est LP time={est_lp_time*1000:.0f}ms, max_iters={max_iters}", flush=True)
        
        a_lp = run_lp_mirror_descent(tracker.best, max_iters, min(time.monotonic() + 180, deadline - 20))
        tracker.update(a_lp)
        
        # Second LP pass
        if time.monotonic() < deadline - 20:
            a_lp2 = run_lp_mirror_descent(a_lp, 50, deadline - 10)
            tracker.update(a_lp2)
    
    tracker.report("Phase 4 complete")
    print(f"Phase 4 total: {time.monotonic()-p4_start:.1f}s", flush=True)
    
    # === PHASE 5: Final combined polish ===
    p5_start = time.monotonic()
    print(f"\n{'='*60}", flush=True)
    print(f"PHASE 5: Final Polish (remaining time)", flush=True)
    print(f"{'='*60}", flush=True)
    
    if tracker.best is not None and time.monotonic() < deadline - 5:
        # Quick coordinate descent
        a_final = run_coordinate_descent(tracker.best, 50, deadline - 3, seed=77777)
        tracker.update(a_final)
        
        # Quick LP if time
        if time.monotonic() < deadline - 3:
            a_final2 = run_lp_mirror_descent(a_final, 20, deadline)
            tracker.update(a_final2)
    
    tracker.report("Phase 5 complete")
    
    # Try a couple of extended lengths if time remains
    if tracker.best is not None and time.monotonic() < deadline - 10:
        for mult in [2]:
            new_n = tracker.best.size * mult
            if new_n > 2048:
                continue
            a_ext = resample_seq(tracker.best, new_n)
            a_ext = np.maximum(a_ext, 1e-6)
            a_ext = a_ext / a_ext.sum()
            a_ext_opt = run_coordinate_descent(a_ext, 50, deadline - 5, seed=55555)
            tracker.update(a_ext_opt)
    
    # Fallback
    if tracker.best is None:
        tracker.best = np.ones(512)
        tracker.best_score = compute_J(tracker.best)
    
    print(f"\n{'='*60}", flush=True)
    print(f"FINAL: J={tracker.best_score:.8f} n={tracker.best.size} "
          f"time={time.monotonic()-tracker.t_start:.0f}s", flush=True)
    print(f"{'='*60}", flush=True)
    
    return [float(x) for x in tracker.best]
