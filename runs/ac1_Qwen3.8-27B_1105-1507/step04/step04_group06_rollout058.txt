import numpy as np
from scipy.optimize import linprog, minimize
from scipy.sparse import csr_matrix

# Cache for convolution sizes
_fft_cache = {}

def fast_conv(a):
    """Full self-convolution via FFT."""
    n = len(a)
    s = 1
    while s < 2 * n - 1:
        s <<= 1
    fa = np.fft.rfft(a, s)
    return np.fft.irfft(fa * fa, s)[:2 * n - 1]

def compute_J(a):
    """Compute J(a) = 2*n*max(conv(a,a))/(sum(a))^2."""
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))

def softmax_grad(b, beta):
    """Compute softmax weights and gradient of smooth max."""
    b_max = b.max()
    shifted = beta * (b - b_max)
    shifted = np.clip(shifted, -500, 500)
    exp_shifted = np.exp(shifted)
    Z = exp_shifted.sum()
    if Z < 1e-300:
        p = np.zeros_like(b)
        p[np.argmax(b)] = 1.0
    else:
        p = exp_shifted / Z
    smooth_max = b_max + np.log(max(Z, 1e-300)) / beta
    return smooth_max, p

def J_smooth_and_grad(a, beta):
    """Compute J_smooth(a) and its gradient for nonnegative a."""
    n = len(a)
    s = float(a.sum())
    if s < 1e-10:
        return 1e10, np.zeros(n)
    b = fast_conv(a)
    L, p = softmax_grad(b, beta)
    # Gradient of L w.r.t. a: 2 * correlate(p, a, 'valid')
    g_L = 2.0 * np.correlate(p.astype(np.float64), a.astype(np.float64), mode='valid')
    # Gradient of J = 2n * L / s^2
    J_val = 2.0 * n * L / (s * s)
    grad_J = 2.0 * n * (s * g_L - 2.0 * L) / (s ** 3)
    return J_val, grad_J

        L_est = max(L_est, 1.0)
        
        best_x = a_curr.copy()
        best_J = J_smooth_and_grad(a_curr, beta)[0]
        
        for it in range(max_iters):
            if time.monotonic() > deadline:
                break
            
            # Backtracking line search
            step = 1.0 / L_est
            while True:
                # Gradient step from z
                J_z, grad_z = J_smooth_and_grad(z, beta)
                x_new = np.maximum(z - step * grad_z, 0.0)
                J_new, _ = J_smooth_and_grad(x_new, beta)
                
                # Check sufficient decrease
                if J_new <= J_z + np.dot(grad_z, x_new - z) + (1.0 / (2.0 * step)) * np.dot(x_new - z, x_new - z) + 1e-12:
                    break
                step *= 0.5
                if step < 1e-15:
                    break
            
            # Update Nesterov momentum
            t_new = (1.0 + np.sqrt(1.0 + 4.0 * t * t)) / 2.0
            z_new = x_new + (t - 1.0) / t_new * (x_new - a_curr)
            z_new = np.maximum(z_new, 0.0)
            
            a_curr = x_new
            z = z_new
            t = t_new
            
            # Update Lipschitz estimate
            L_est = max(L_est * 0.9, 2.0 * n * beta * a_curr.max() / max(a_curr.sum()**2, 1e-10) + 1.0)
            
            # Track best
            J_curr, _ = J_smooth_and_grad(a_curr, beta)
            if J_curr < best_J:
                best_J = J_curr
                best_x = a_curr.copy()
                if tracker is not None:
                    tracker.update(a_curr)
        
        a = best_x
    
    return a

def reduced_lp_direction(a, M, top_k=None):
    """
    Solve reduced LP: max sum(g) s.t. (a*g)_k <= M for top_k indices, g >= 0.
    Returns normalized g or None.
    """
    n = len(a)
    b = fast_conv(a)
    
    if top_k is None:
        top_k = min(2 * n - 1, max(32, n // 4))
    
    # Select top_k indices where b is largest
    # Use the indices where b is closest to M
    threshold = M * (1 - 0.05)  # Within 5% of max
    candidates = np.where(b >= threshold)[0]
    if len(candidates) < top_k:
        # Just take top K
        idx = np.argsort(b)[::-1][:top_k]
    else:
        idx = candidates[:top_k]
    
    # Build sparse matrix for selected constraints
    rows = []
    cols = []
    for k in idx:
        jl = max(0, k - n + 1)
        jh = min(n - 1, k)
        for j in range(jl, jh + 1):
            rows.append(k)
            cols.append(j)
    
    if len(rows) == 0:
        return None
    
    vals = a[np.array(rows) - np.array(cols)]
    A = csr_matrix((vals, (np.array(rows, dtype=np.int32), np.array(cols, dtype=np.int32))),
                   shape=(len(idx), n))
    
    c = -np.ones(n)
    b_ub = np.full(len(idx), M)
    bounds = [(0.0, None)] * n
    
    r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                options={'presolve': True, 'time_limit': 10})
    
    if r.success and r.x is not None:
        g = np.maximum(r.x, 0.0)
        g_sum = g.sum()
        if g_sum > 1e-10:
            return g / g_sum
    return None

def full_lp_direction(a, M):
    """Solve full LP for best response direction."""
    n = len(a)
    rows = []
    cols = []
    for k in range(2 * n - 1):
        jl = max(0, k - n + 1)
        jh = min(n - 1, k)
        for j in range(jl, jh + 1):
            rows.append(k)
            cols.append(j)
    
    vals = a[np.array(rows) - np.array(cols)]
    A = csr_matrix((vals, (np.array(rows, dtype=np.int32), np.array(cols, dtype=np.int32))),
                   shape=(2 * n - 1, n))
    
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    
    r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                options={'presolve': True, 'time_limit': 30})
    
    if r.success and r.x is not None:
        g = np.maximum(r.x, 0.0)
        g_sum = g.sum()
        if g_sum > 1e-10:
            return g / g_sum
    return None

def line_search(a, g, n_evals=20):
    """Line search for best t in [0, 1] minimizing J((1-t)*a + t*g)."""
    best_t = 0.0
    best_J = compute_J(a)
    
    # Try t=1 first
    J1 = compute_J(g)
    if J1 < best_J:
        best_J = J1
        best_t = 1.0
    
    # Bisection / golden section
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    lo, hi = 0.0, 1.0
    
    for _ in range(n_evals):
        t1 = hi - gr * (hi - lo)
        t2 = lo + gr * (hi - lo)
        a1 = (1.0 - t1) * a + t1 * g
        a2 = (1.0 - t2) * a + t2 * g
        J1 = compute_J(a1)
        J2 = compute_J(a2)
        if J1 < J2:
            hi = t2
        else:
            lo = t1
        
        t_mid = (lo + hi) / 2.0
        a_mid = (1.0 - t_mid) * a + t_mid * g
        J_mid = compute_J(a_mid)
        if J_mid < best_J:
            best_J = J_mid
            best_t = t_mid
    
    return (1.0 - best_t) * a + best_t * g, best_J

def lp_mirror_descent_v2(a, max_iters, deadline, tracker=None, n_line_evals=20,
                          stall_patience=30, use_reduced_lp=True, top_k_ratio=0.25):
    """LP mirror descent with constraint reduction and adaptive line search."""
    a = np.maximum(a.astype(np.float64), 1e-6)
    a = a / a.sum()
    current_J = compute_J(a)
    best_a = a.copy()
    best_J = current_J
    stall_count = 0
    
    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        
        b = fast_conv(a)
        M = float(b.max())
        
        # Get LP direction
        if use_reduced_lp:
            g = reduced_lp_direction(a, M, top_k=max(int(len(a) * top_k_ratio), 32))
        else:
            g = full_lp_direction(a, M)
        
        if g is None:
            break
        
        # Line search
        a_new, J_new = line_search(a, g, n_evals=n_line_evals)
        
        if J_new < current_J - 1e-13:
            a = a_new
            current_J = J_new
            stall_count = 0
            if J_new < best_J:
                best_J = J_new
                best_a = a.copy()
                if tracker is not None:
                    tracker.update(a)
        else:
            # Try smaller steps
            improved = False
            for t_try in [0.3, 0.1, 0.05, 0.02, 0.01]:
                a_try = (1.0 - t_try) * a + t_try * g
                a_try = np.maximum(a_try, 0.0)
                s = a_try.sum()
                if s < 1e-10:
                    continue
                a_try = a_try / s
                J_try = compute_J(a_try)
                if J_try < current_J - 1e-14:
                    a = a_try
                    current_J = J_try
                    stall_count = 0
                    if J_try < best_J:
                        best_J = J_try
                        best_a = a.copy()
                        if tracker is not None:
                            tracker.update(a)
                    improved = True
                    break
            if not improved:
                stall_count += 1
                if stall_count >= stall_patience:
                    break
    
    if tracker is not None:
        tracker.update(best_a)
    return best_a

def resample_seq(src, dest_n):
    """Resample 1D sequence to new length."""
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)

class ScoreTracker:
    def __init__(self):
        self.best = None
        self.best_score = float('inf')
        self.t_start = time.monotonic()
    
    def update(self, a):
        a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)

"""
Minimizes J(a) = 2*n*max(conv(a,a))/(sum(a))² using:
1. FISTA with adaptive log-sum-exp smoothing for fast basin finding
2. Pre-computed sparse LP structure for rapid best-response computation
3. Adaptive constraint reduction (few → many as convergence improves)
4. Nesterov-momentum LP mirror descent with aggressive line search
5. Peak-targeted local perturbations and multi-length exploration

import numpy as np
from scipy.optimize import linprog
from scipy.sparse import csr_matrix
import time

# Pre-computed constraint structure cache
_structure_cache = {}


def fast_conv(a):
    """Full self-convolution via FFT, O(n log n)."""
    n = len(a)
    s = 1
    while s < 2 * n - 1:
        s <<= 1
    fa = np.fft.rfft(a, s)
    return np.fft.irfft(fa * fa, s)[:2 * n - 1]


def compute_J(a):
    """Compute J(a) = 2*n*max(conv(a,a))/(sum(a))^2."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))


def get_structure(n):
    """Pre-compute (row, col) index arrays for the convolution constraint matrix."""
    if n not in _structure_cache:
        rows = []
        cols = []
        for k in range(2 * n - 1):
            jl = max(0, k - n + 1)
            jh = min(n - 1, k)
            for j in range(jl, jh + 1):
                rows.append(k)
                cols.append(j)
        _structure_cache[n] = (np.array(rows, dtype=np.int32),
                               np.array(cols, dtype=np.int32))
    return _structure_cache[n]


def build_A(a, k_indices):
    """Build sparse constraint matrix for selected lag indices."""
    n = len(a)
    all_rows, all_cols = get_structure(n)
    # Filter to only selected k indices
    mask = np.isin(all_rows, np.asarray(k_indices, dtype=np.int32))
    rows = all_rows[mask]
    cols = all_cols[mask]
    vals = a[rows - cols]
    return csr_matrix((vals, (rows, cols)), shape=(len(k_indices), n))


def lp_direction(a, M, k_indices):
    """
    Solve LP: max sum(g) s.t. (a*g)_k <= M for k in k_indices, g >= 0.
    Returns normalized g or None.
    """
    n = len(a)
    A = build_A(a, k_indices)
    c = -np.ones(n)
    b_ub = np.full(len(k_indices), M)
    bounds = [(0.0, None)] * n
    try:
        r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                    options={'presolve': True, 'time_limit': 15})
        if r.success and r.x is not None:
            g = np.maximum(r.x, 0.0)
            g_sum = g.sum()
            if g_sum > 1e-10:
                return g / g_sum
    except Exception:
        pass
    return None


def select_constraints(b, M, n, mode='adaptive'):
    """Select which convolution lag indices to include in the LP."""
    if mode == 'full':
        return list(range(2 * n - 1))
    elif mode == 'top_percent':
        # Include constraints where b >= 0.95 * M
        threshold = 0.95 * M
        candidates = np.where(b >= threshold)[0]
        max_k = max(64, 2 * n // 4)
        if len(candidates) > max_k:
            # Take the top max_k
            order = np.argsort(b)[::-1][:max_k]
            return sorted(order.tolist())
        return sorted(candidates.tolist())
    else:
        # Fixed count
        k_count = max(32, min(2 * n - 1, n // 2))
        order = np.argsort(b)[::-1][:k_count]
        return sorted(order.tolist())


def line_search(a, g, n_evals=20):
    """Line search: find best t in [0,1] minimizing J((1-t)*a + t*g)."""
    best_t = 0.0
    best_J = compute_J(a)

    # Try t=1 first
    J1 = compute_J(g)
    if J1 < best_J:
        best_J = J1
        best_t = 1.0

    # Golden section
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    lo, hi = 0.0, 1.0

    for _ in range(n_evals):
        t1 = hi - gr * (hi - lo)
        t2 = lo + gr * (hi - lo)
        a1 = (1.0 - t1) * a + t1 * g
        a2 = (1.0 - t2) * a + t2 * g
        J1 = compute_J(a1)
        J2 = compute_J(a2)
        if J1 < J2:
            hi = t2
        else:
            lo = t1
        t_mid = (lo + hi) / 2.0
        a_mid = (1.0 - t_mid) * a + t_mid * g
        J_mid = compute_J(a_mid)
        if J_mid < best_J:
            best_J = J_mid
            best_t = t_mid

    return (1.0 - best_t) * a + best_t * g, best_J


def softmax_weights(b, beta):
    """Compute softmax weights for smoothing the max."""
    b_max = b.max()
    shifted = beta * (b - b_max)
    shifted = np.clip(shifted, -500, 500)
    exp_shifted = np.exp(shifted)
    Z = exp_shifted.sum()
    if Z < 1e-300:
        p = np.zeros_like(b)
        p[np.argmax(b)] = 1.0
    else:
        p = exp_shifted / Z
    smooth_max = b_max + np.log(max(Z, 1e-300)) / beta
    return smooth_max, p


def J_smooth_grad(a, beta):
    """Compute smoothed J(a) and gradient."""
    n = len(a)
    s = float(a.sum())
    if s < 1e-10:
        return 1e10, np.zeros(n)
    b = fast_conv(a)
    L, p = softmax_weights(b, beta)
    # grad_L[i] = 2 * sum_j a[j] * p[i+j]
    # = 2 * np.correlate(p, a, mode='valid')[i]
    g_L = 2.0 * np.correlate(p.astype(np.float64), a.astype(np.float64), mode='valid')
    J_val = 2.0 * n * L / (s * s)
    grad_J = 2.0 * n * (s * g_L - 2.0 * L) / (s ** 3)
    return J_val, grad_J


def fista_optimize(a0, betas, deadline, tracker=None):
    """FISTA (Nesterov accelerated projected gradient) on smoothed objective."""
    a = np.maximum(a0.astype(np.float64), 1e-8).copy()
    n = len(a)

    for beta, max_iters in betas:
        if time.monotonic() > deadline:
            break

        # Estimate initial Lipschitz constant
        L_est = max(2.0 * n * (beta + 4.0), 1.0)

        a_curr = a.copy()
        z = a.copy()
        t_mom = 1.0

        for it in range(max_iters):
            if time.monotonic() > deadline:
                break

            J_z, grad_z = J_smooth_grad(z, beta)
            if not np.isfinite(J_z):
                break

            # Backtracking line search
            step = 1.0 / L_est
            x_new = None
            for _bt in range(20):
                candidate = np.maximum(z - step * grad_z, 0.0)
                J_cand, _ = J_smooth_grad(candidate, beta)
                if J_cand <= J_z + np.dot(grad_z, candidate - z) + \
                        0.5 / step * np.dot(candidate - z, candidate - z) + 1e-10:
                    x_new = candidate
                    break
                step *= 0.5
            if x_new is None:
                x_new = np.maximum(z - (1.0 / L_est) * grad_z, 0.0)

            # Nesterov momentum update
            t_new = (1.0 + np.sqrt(1.0 + 4.0 * t_mom ** 2)) / 2.0
            z = x_new + (t_mom - 1.0) / t_new * (x_new - a_curr)
            z = np.maximum(z, 0.0)

            a_curr = x_new
            t_mom = t_new

            # Adaptive L update
            L_est = max(L_est * 1.05, 2.0 * n * (beta + 4.0))
            if it % 20 == 0:
                J_curr, _ = J_smooth_grad(a_curr, beta)
                if tracker is not None and np.isfinite(J_curr):
                    tracker.update(a_curr)

        a = a_curr

    return a


def lp_mirror_descent(a, max_iters, deadline, tracker=None, n_line_evals=18,
                      stall_patience=35, constraint_mode='adaptive'):
    """
    LP mirror descent with adaptive constraint selection and line search.
    """
    a = np.maximum(a.astype(np.float64), 1e-6)
    a = a / a.sum()
    current_J = compute_J(a)
    best_a = a.copy()
    best_J = current_J
    stall_count = 0
    iteration = 0

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        b = fast_conv(a)
        M = float(b.max())
        n = len(a)

        # Select constraints
        if constraint_mode == 'adaptive':
            # Start with fewer constraints, add more as we converge
            progress = 1.0 - (current_J - 1.0) / max(1.0, best_J - 1.0)
            progress = np.clip(progress, 0, 1)
            max_k = int(64 + progress * (2 * n - 1 - 64))
            max_k = max(64, min(2 * n - 1, max_k))
            k_indices = select_constraints(b, M, n, mode='top_percent')
            if len(k_indices) > max_k:
                order = np.argsort(b)[::-1][:max_k]
                k_indices = sorted(order.tolist())
        elif constraint_mode == 'full':
            k_indices = select_constraints(b, M, n, mode='full')
        else:
            k_indices = select_constraints(b, M, n, mode='top_percent')

        # Solve LP
        g = lp_direction(a, M, k_indices)
        if g is None:
            # Fallback: try full
            g = lp_direction(a, M, select_constraints(b, M, n, mode='full'))
            if g is None:
                break

        # Line search
        a_new, J_new = line_search(a, g, n_evals=n_line_evals)

        if J_new < current_J - 1e-13:
            a = a_new
            current_J = J_new
            stall_count = 0
            if J_new < best_J:
                best_J = J_new
                best_a = a.copy()
                if tracker is not None:
                    tracker.update(a)
        else:
            # Try smaller explicit steps
            improved = False
            for t_try in [0.5, 0.3, 0.15, 0.07, 0.03]:
                a_try = (1.0 - t_try) * a + t_try * g
                a_try = np.maximum(a_try, 0.0)
                s = a_try.sum()
                if s < 1e-10:
                    continue
                a_try = a_try / s
                J_try = compute_J(a_try)
                if J_try < current_J - 1e-14:
                    a = a_try
                    current_J = J_try
                    stall_count = 0
                    if J_try < best_J:
                        best_J = J_try
                        best_a = a.copy()
                        if tracker is not None:
                            tracker.update(a)
                    improved = True
                    break
            if not improved:
                stall_count += 1
                if stall_count >= stall_patience:
                    break

        iteration += 1

    if tracker is not None:
        tracker.update(best_a)
    return best_a


def resample_seq(src, dest_n):
    """Resample 1D sequence to new length via linear interpolation."""
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)


def generate_diverse_inits(n, parent, rng):
    """Generate diverse initial sequences of length n."""
    inits = []
    t_arr = np.linspace(0, 1, n, endpoint=False)

    if parent is not None and parent.size > 0:
        inits.append(resample_seq(parent, n))
        # Perturbed parent
        p = resample_seq(parent, n)
        inits.append(np.maximum(p + rng.normal(0, 0.05, n) * p, 0.01))

    inits.append(np.ones(n))
    inits.append(rng.uniform(0.3, 1.0, n))

    # Cosine families
    for freq in [2, 3, 4, 5, 7, 8]:
        inits.append(1.0 + 0.3 * np.cos(2.0 * np.pi * freq * t_arr))

    # Product of cosines
    inits.append(1.0 + 0.2 * np.cos(2.0 * np.pi * 3 * t_arr) +
                 0.15 * np.cos(2.0 * np.pi * 7 * t_arr))

    # Gaussian
    inits.append(np.exp(-4.0 * (t_arr - 0.5) ** 2) + 0.1)

    # Sine
    inits.append(np.abs(np.sin(np.pi * t_arr)) + 0.2)

    # Power law
    inits.append((t_arr + 0.1) ** 0.5)

    # Exponential
    inits.append(np.exp(-1.5 * t_arr) + 0.2)

    return inits

            a = lp_mirror_descent(tracker.best, max_iters2, deadline_p3, tracker=tracker,
                                  n_line_evals=22, stall_patience=55, constraint_mode='full')

        tracker.report(f"Phase 3: {time.monotonic()-t0:.1f}s")

    # === Phase 4: Peak-targeted perturbations ===
    if tracker.best is not None and time.monotonic() < deadline:
        t0 = time.monotonic()
        remaining = deadline - time.monotonic()
        phase4_budget = min(150.0, remaining * 0.2)
        deadline_p4 = time.monotonic() + phase4_budget
        a = tracker.best.copy().astype(np.float64)
        n = len(a)
        print(f"\n{'='*50} Phase 4: Peak Targeting (budget {phase4_budget:.0f}s) {'='*50}", flush=True)

        for round_num in range(8):
            if time.monotonic() > deadline_p4:
                break

            b = fast_conv(a)
            M = b.max()
            # Find peak positions
            peak_thresh = M * (0.90 + 0.05 * round_num)  # Expanding threshold
            peak_indices = np.where(b >= peak_thresh)[0]

            for peak_idx in peak_indices[:4]:
                if time.monotonic() > deadline_p4:
                    break

                # Create a direction that reduces contribution at this peak
                # (a*g)_peak = sum_j a_j * g_{peak-j}
                # To reduce this, make g small where a is large at the peak
                # Build a "repulsion" direction
                direction = a.copy()
                pi = int(peak_idx)
                jl = max(0, pi - n + 1)
                jh = min(n - 1, pi)
                # Elements contributing to this peak: a[j] for j in [jl, jh]
                # We want to decrease them, so subtract from direction
                for j in range(jl, jh + 1):
                    direction[j] -= 0.5 * a[j]

                # Combine with random perturbation
                a_try = a + 0.1 * direction + rng.normal(0, 0.02, n) * a
                a_try = np.maximum(a_try, 1e-4)
                a_try = a_try / a_try.sum()

                J_try = compute_J(a_try)
                if J_try < compute_J(a) - 1e-8:
                    a = a_try
                    tracker.update(a)

                # Short LP refinement
                if time.monotonic() < deadline_p4:
                    sub_deadline = time.monotonic() + min(15.0, deadline_p4 - time.monotonic())
                    a = lp_mirror_descent(a, 100, sub_deadline, tracker=tracker,
                                          n_line_evals=10, stall_patience=15,
                                          constraint_mode='adaptive')

        tracker.report(f"Phase 4: {time.monotonic()-t0:.1f}s")

    # === Phase 5: Multi-restart ===
    if tracker.best is not None and time.monotonic() < deadline:
        t0 = time.monotonic()
        remaining = deadline - time.monotonic()
        phase5_budget = min(200.0, remaining * 0.25)
        deadline_p5 = time.monotonic() + phase5_budget
        print(f"\n{'='*50} Phase 5: Multi-Restart (budget {phase5_budget:.0f}s) {'='*50}", flush=True)

        strategies = [
            ('gaussian', 0.04),
            ('gaussian', 0.08),
            ('gaussian', 0.12),
            ('gaussian', 0.02),
            ('uniform', 0.06),
            ('uniform', 0.12),
            ('spike', 5),
            ('spike', 12),
            ('smooth', 0.05),
            ('smooth', 0.1),
            ('directional', 0.06),
            ('directional', 0.1),
        ]

        for pert_type, param in strategies:
            if time.monotonic() > deadline_p5:
                break

            a = tracker.best.copy().astype(np.float64)
            n = len(a)

            if pert_type == 'gaussian':
                a = a + rng.normal(0, param, n) * a
            elif pert_type == 'uniform':
                a = a + rng.uniform(-param, param, n) * a
            elif pert_type == 'spike':
                k = min(int(param), n // 4)
                idx = rng.choice(n, size=k, replace=False)
                a[idx] *= rng.uniform(0.2, 4.0, k)
            elif pert_type == 'smooth':
                noise = rng.normal(0, param, n)
                kernel_size = max(3, n // 32)
                kernel = np.exp(-0.5 * (np.arange(kernel_size) - kernel_size // 2) ** 2 / (kernel_size // 4 + 1))
                kernel /= kernel.sum()
                noise = np.convolve(noise, kernel, mode='same')
                a = a + noise * a
            elif pert_type == 'directional':
                # Perturb in the direction of reducing peak autocorrelation
                b = fast_conv(a)
                M = b.max()
                peak_idx = int(np.argmax(b))
                direction = np.zeros(n)
                jl = max(0, peak_idx - n + 1)
                jh = min(n - 1, peak_idx)
                for j in range(jl, jh + 1):
                    direction[j] = -a[peak_idx - j]
                direction = direction - direction.mean()
                a = a + param * direction * a / max(a.max(), 1e-10)

            a = np.maximum(a, 1e-4)
            s = a.sum()
            if s < 1e-6:
                continue
            a = a / s
            tracker.update(a)

            # LP refinement
            budget_here = min(25.0, deadline_p5 - time.monotonic())
            if budget_here > 5:
                max_iters = int(budget_here * 60)
                a = lp_mirror_descent(a, max_iters, time.monotonic() + budget_here,
                                      tracker=tracker, n_line_evals=12, stall_patience=20,
                                      constraint_mode='adaptive')

        tracker.report(f"Phase 5: {time.monotonic()-t0:.1f}s")

    # === Phase 6: Final LP polish (use remaining time) ===
    if tracker.best is not None and time.monotonic() < deadline:
        t0 = time.monotonic()
        remaining = deadline - time.monotonic()
        if remaining > 20:
            n_best = tracker.best.size
            print(f"\n{'='*50} Phase 6: Final Polish (remaining {remaining:.0f}s, n={n_best}) {'='*50}", flush=True)

            # Alternate between adaptive and full
            sub_deadline = time.monotonic() + remaining * 0.5
            max_iters = int(remaining * 0.5 * 50)
            a = lp_mirror_descent(tracker.best, max_iters, sub_deadline, tracker=tracker,
                                  n_line_evals=22, stall_patience=60, constraint_mode='adaptive')

            if time.monotonic() < deadline:
                max_iters2 = int((deadline - time.monotonic()) * 40)
                a = lp_mirror_descent(tracker.best, max_iters2, deadline, tracker=tracker,
                                      n_line_evals=25, stall_patience=70, constraint_mode='full')

            tracker.report(f"Phase 6: {time.monotonic()-t0:.1f}s")

    # === Phase 7: Quick length exploration ===
    if tracker.best is not None and time.monotonic() < deadline:
        t0 = time.monotonic()
        remaining = deadline - time.monotonic()
        if remaining > 20:
            n_best = tracker.best.size
            print(f"\n{'='*50} Phase 7: Length Exploration (remaining {remaining:.0f}s) {'='*50}", flush=True)

            candidates = set()
            for delta in [-128, -64, -32, 32, 64, 128]:
                candidates.add(n_best + delta)
            candidates.update([512, 768, 1024, 1536, 2048, 2560])
            candidates = sorted([c for c in candidates if 128 <= c <= 4096 and abs(c - n_best) >= 32])

            for n_try in candidates[:6]:
                if time.monotonic() > deadline:
                    break
                a_res = resample_seq(tracker.best, n_try)
                a_res = a_res / max(a_res.sum(), 1e-10)
                _ = get_structure(n_try)
                budget_here = min(25.0, deadline - time.monotonic())
                if budget_here < 5:
                    continue
                max_iters = int(budget_here * 60)
                a_res = lp_mirror_descent(a_res, max_iters, time.monotonic() + budget_here,
                                          tracker=tracker, n_line_evals=12, stall_patience=20,
                                          constraint_mode='adaptive')
                tracker.update(a_res)

            tracker.report(f"Phase 7: {time.monotonic()-t0:.1f}s")

    # Fallback
    if tracker.best is None:
        tracker.best = np.ones(1024)
        tracker.best_score = compute_J(tracker.best)
        print(f"  Using fallback: J={tracker.best_score:.8f}", flush=True)

    print(f"\n{'='*50} FINAL RESULT {'='*50}", flush=True)
    print(f"J={tracker.best_score:.8f} n={tracker.best.size} "
          f"time={time.monotonic()-tracker.t_start:.1f}s", flush=True)

    # Ensure valid output
    result = np.clip(tracker.best, 0.0, 1000.0)
    if result.sum() < 0.01:
        result = np.ones(128)

    return [float(x) for x in result]
