"""
Spectral flattening + LP mirror descent hybrid for minimizing
J(a) = 2*n*max(conv(a,a))/(sum(a))^2 over non-negative sequences.

Key innovation: Gerchberg-Saxton-like spectral flattening to reduce
the peak of the autocorrelation, followed by LP-based local refinement.
"""

import numpy as np
from scipy.optimize import linprog
from scipy.sparse import csr_matrix
import time


def fast_conv(a):
    """Full self-convolution via FFT."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]


def compute_J(a):
    """Compute J(a) = 2*n*max(conv(a,a))/(sum(a))^2."""
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))


def spectral_flatten(a, alpha=0.5, n_iters=500, deadline=None):
    """
    Spectral flattening: iteratively flatten the power spectrum to
    reduce the peak of the autocorrelation.
    
    Parameters:
    - a: input sequence (non-negative)
    - alpha: flattening parameter (0 < alpha < 1, smaller = more aggressive)
    - n_iters: maximum iterations
    - deadline: time deadline
    
    Returns: (flattened sequence, best J found)
    """
    a = np.maximum(a.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-10:
        return a, float('inf')
    a = a / s
    
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    
    best_J = compute_J(a)
    best_a = a.copy()
    no_improve = 0
    
    for it in range(n_iters):
        if deadline is not None and time.monotonic() > deadline:
            break
        
        # Compute spectrum
        A = np.fft.rfft(a, size)
        mag = np.abs(A)
        phase = A / (mag + 1e-30)
        
        # Flatten: replace |A| with |A|^alpha
        mag_new = np.power(mag + 1e-30, alpha)
        A_new = mag_new * phase
        
        # Back to time domain
        a_new = np.fft.irfft(A_new, size)[:n]
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s < 1e-10:
            break
        a_new = a_new / s
        
        J_new = compute_J(a_new)
        if J_new < best_J:
            best_J = J_new
            best_a = a_new.copy()
            no_improve = 0
        else:
            no_improve += 1
            if no_improve > 50:
                break
        
        a = a_new
    
    return best_a, best_J


def build_A_sparse(n, a):
    """Build sparse matrix A where A[k,j] = a[k-j] for valid j."""
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        j_min = max(0, k - n + 1)
        j_max = min(n - 1, k)
        for j in range(j_min, j_max + 1):
            rows.append(k)
            cols.append(j)
            vals.append(a[k - j])
    return csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))


def lp_best_response(a, M):
    """Solve LP: max sum(g) s.t. A@g <= M*1, g >= 0."""
    n = len(a)
    A = build_A_sparse(n, a)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    try:
        r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
        if r.success and r.x is not None:
            g = np.maximum(r.x, 0)
            if g.sum() > 1e-15:
                return g
    except Exception:
        pass
    return None


def lp_mirror_descent(a, max_iters, deadline, use_multiplicative=False):
    """LP mirror descent with adaptive line search."""
    a = np.maximum(a.astype(np.float64), 1e-10)
    a = a / a.sum()
    current_J = compute_J(a)
    best_J = current_J
    best_a = a.copy()
    no_improve = 0

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        b = fast_conv(a)
        M = float(b.max())
        g = lp_best_response(a, M)
        if g is None:
            break
        g = np.maximum(g, 0.0)
        g_sum = g.sum()
        if g_sum < 1e-10:
            break
        g = g / g_sum

        base_t = 0.25 * np.exp(-0.3 * it / 100.0) + 0.05
        t = min(1.0, base_t)

        best_trial_J = current_J
        best_trial_a = None
        best_t = None

        for ls in range(25):
            if use_multiplicative:
                ratio = g / np.maximum(a, 1e-15)
                a_trial = a * np.power(ratio, t)
            else:
                a_trial = (1.0 - t) * a + t * g
            a_trial = np.maximum(a_trial, 0.0)
            s = a_trial.sum()
            if s < 1e-10:
                t *= 0.5
                continue
            a_trial = a_trial / s
            J_trial = compute_J(a_trial)
            if J_trial < best_trial_J - 1e-14:
                best_trial_J = J_trial
                best_trial_a = a_trial.copy()
                best_t = t
                break
            t *= 0.65
            if t < 0.002:
                break

        if best_trial_a is not None:
            a = best_trial_a
            current_J = best_trial_J
            if current_J < best_J:
                best_J = current_J
                best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1
            if no_improve > 40:
                break

    return best_a, best_J


def peak_shaving(a, n_iters=1000, deadline=None):
    """
    Simple peak-shaving: iteratively reduce the peak of the convolution
    by moving mass from high-contribution coordinates to low-contribution ones.
    """
    a = np.maximum(a.astype(np.float64), 1e-10)
    a = a / a.sum()
    n = len(a)
    best_J = compute_J(a)
    best_a = a.copy()
    
    for it in range(n_iters):
        if deadline is not None and time.monotonic() > deadline:
            break
        
        b = fast_conv(a)
        k_star = int(np.argmax(b))
        M = b[k_star]
        
        # Gradient: grad_i = 2*a[k_star - i] for valid k_star - i
        grad = np.zeros(n)
        for i in range(n):
            j = k_star - i
            if 0 <= j < n:
                grad[i] = 2.0 * a[j]
        
        # Move mass from high-grad to low-grad
        mean_grad = grad.mean()
        # Perturbation: decrease high-grad coords, increase low-grad coords
        delta = grad - mean_grad
        step = 0.1 / (np.max(np.abs(delta)) + 1e-10)
        
        a_new = a - step * delta
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s < 1e-10:
            break
        a_new = a_new / s
        
        J_new = compute_J(a_new)
        if J_new < best_J:
            best_J = J_new
            best_a = a_new.copy()
        a = a_new
    
    return best_a, best_J


def subgradient_polish(a, max_iters, deadline):
    """Subgradient descent with momentum."""
    a = np.maximum(a.astype(np.float64), 1e-10)
    a = a / a.sum()
    best_M = fast_conv(a).max()
    best_a = a.copy()
    momentum = np.zeros_like(a)
    beta = 0.85

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        b = fast_conv(a)
        M = b.max()
        tol = 1e-11 * M
        active = np.where(b >= M - tol)[0]

        grad = np.zeros_like(a)
        n = len(a)
        for k in active:
            lo = max(0, k - n + 1)
            hi = min(n - 1, k)
            for i in range(lo, hi + 1):
                grad[i] += 2.0 * a[k - i]

        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-15:
            break

        t = 1.5 / (gnorm * np.sqrt(it + 1.0))
        momentum = beta * momentum + (1 - beta) * grad / gnorm
        a_new = a - t * momentum
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s < 1e-10:
            break
        a_new = a_new / s

        new_M = fast_conv(a_new).max()
        if new_M < best_M:
            best_M = new_M
            best_a = a_new.copy()
        a = a_new

    return best_a


def resample_seq(src, dest_n):
    """Resample 1D sequence to new length."""
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)

Wait, I need to think about this more carefully. The spectral flattening approach - will it actually work?

Let me trace through what happens:
1. Start with a (non-negative, normalized)
2. A = FFT(a, N)
3. |A| = magnitude, phase = A/|A|
4. |A_new| = |A|^alpha (for alpha < 1, this compresses the dynamic range)
5. A_new = |A_new| * phase
6. a_new = IFFT(A_new)[:n]
7. Clip to non-negative, normalize

import numpy as np
from scipy.optimize import linprog
from scipy.sparse import csr_matrix
import time


def fast_conv(a):
    """Full self-convolution via FFT."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]


def compute_J(a):
    """Compute J(a) = 2*n*max(conv(a,a))/(sum(a))^2."""
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))


def spectral_flatten(a, alpha=0.5, n_iters=500, deadline=None, adaptive=False):
    """
    Spectral flattening via Gerchberg-Saxton-like iteration.
    Flattens the power spectrum to reduce autocorrelation peak.
    """
    a = np.maximum(a.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-10:
        return a.copy(), float('inf')
    a = a / s

    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1

    best_J = compute_J(a)
    best_a = a.copy()
    no_improve = 0
    current_alpha = alpha

    for it in range(n_iters):
        if deadline is not None and time.monotonic() > deadline:
            break

        A = np.fft.rfft(a, size)
        mag = np.abs(A)
        mag_safe = np.maximum(mag, 1e-30)
        phase = A / mag_safe

        if adaptive:
            current_alpha = alpha * (1.0 + 0.5 * np.cos(np.pi * it / max(1, n_iters)))
            current_alpha = max(0.1, min(0.9, current_alpha))

        mag_new = np.power(mag_safe, current_alpha)
        A_new = mag_new * phase

        a_new = np.fft.irfft(A_new, size)[:n]
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s < 1e-10:
            break
        a_new = a_new / s

        J_new = compute_J(a_new)
        if J_new < best_J - 1e-12:
            best_J = J_new
            best_a = a_new.copy()
            no_improve = 0
        else:
            no_improve += 1
            if no_improve > 100:
                break

        a = a_new

    return best_a, best_J


def _build_A_structure(n):
    """Precompute sparsity pattern for convolution constraint matrix."""
    rows = []
    cols = []
    for k in range(2 * n - 1):
        j_min = max(0, k - n + 1)
        j_max = min(n - 1, k)
        for j in range(j_min, j_max + 1):
            rows.append(k)
            cols.append(j)
    return np.array(rows, dtype=np.int32), np.array(cols, dtype=np.int32)


def lp_best_response_fast(a, M, row_idx, col_idx):
    """Solve LP: max sum(g) s.t. A@g <= M*1, g >= 0, using precomputed structure."""
    n = len(a)
    k = row_idx.astype(np.int64)
    j = col_idx.astype(np.int64)
    vals = a[k - j]
    A = csr_matrix((vals, (row_idx, col_idx)), shape=(2 * n - 1, n))
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    try:
        r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                    options={'presolve': True, 'dual_feasibility_tolerance': 1e-9})
        if r.success and r.x is not None:
            g = np.maximum(r.x, 0)
            if g.sum() > 1e-15:
                return g
    except Exception:
        pass
    return None


def lp_mirror_descent(a, max_iters, deadline, use_multiplicative=False, row_idx=None, col_idx=None):

    current_J = compute_J(a)
    best_J = current_J
    best_a = a.copy()
    no_improve = 0

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        b = fast_conv(a)
        M = float(b.max())
        g = lp_best_response_fast(a, M, row_idx, col_idx)
        if g is None:
            break
        g = np.maximum(g, 0.0)
        g_sum = g.sum()
        if g_sum < 1e-10:
            break
        g = g / g_sum

        base_t = 0.3 * np.exp(-0.2 * it / 150.0) + 0.05
        t = min(1.0, base_t)

        best_trial_J = current_J
        best_trial_a = None

        for ls in range(30):

                best_a = a.copy()
            no_improve = 0
        else:

def subgradient_polish(a, max_iters, deadline):
    """Subgradient descent with momentum on the max-conv objective."""
    a = np.maximum(a.astype(np.float64), 1e-10)
    a = a / a.sum()
    n = len(a)
    best_M = fast_conv(a).max()
    best_a = a.copy()
    momentum = np.zeros_like(a)
    beta = 0.8

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        b = fast_conv(a)
        M = b.max()
        tol = 1e-10 * M
        active = np.where(b >= M - tol)[0]

        grad = np.zeros_like(a)
        for k in active:
            lo = max(0, k - n + 1)
            hi = min(n - 1, k)
            for i in range(lo, hi + 1):
                grad[i] += 2.0 * a[k - i]

        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-15:
            break

        step = 2.0 / (gnorm * (it + 1.0) ** 0.6)
        momentum = beta * momentum + (1 - beta) * grad / gnorm
        a_new = a - step * momentum
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s < 1e-10:
            break
        a_new = a_new / s

        new_M = fast_conv(a_new).max()
        if new_M < best_M:
            best_M = new_M
            best_a = a_new.copy()
        a = a_new

    return best_a


def resample_seq(src, dest_n):
    """Resample 1D sequence to new length via interpolation."""
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)

def fast_conv(a):
    """Full self-convolution via FFT."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:

    return np.fft.irfft(fa * fa, size)[:2 * n - 1]


def compute_J(a):
