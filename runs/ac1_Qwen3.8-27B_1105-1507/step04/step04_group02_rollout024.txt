    """Build Hankel matrix from p (length 2n-1)."""
    n = (len(p) + 1) // 2
    P = np.zeros((n, n))
    for i in range(n):
        for j in range(n):
            if i + j < len(p):
                P[i, j] = p[i + j]
    return P

def solve_qp_simplex(P, max_iters=1000, tol=1e-10):
    """Solve min a^T P a subject to sum(a)=1, a>=0 using projected gradient."""
    n = P.shape[0]
    a = np.ones(n) / n
    best_a = a.copy()
    best_val = a @ P @ a
    step = 1.0 / (2.0 * np.max(np.abs(np.linalg.eigvals(P)))) + 1e-10
    
    for it in range(max_iters):
        grad = 2.0 * P @ a
        # Projected gradient step on simplex
        a_new = a - step * grad
        # Project onto simplex
        a_new = project_simplex(a_new)
        val = a_new @ P @ a_new
        if val < best_val:
            best_val = val
            best_a = a_new.copy()
        if np.max(np.abs(a_new - a)) < tol:
            break
        a = a_new
    return best_a

def project_simplex(v):
    """Project v onto the probability simplex {a >= 0, sum(a) = 1}."""
    n = len(v)
    if np.sum(v) <= 0:
        return np.ones(n) / n
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n+1) > (cssv - 1))[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0)
