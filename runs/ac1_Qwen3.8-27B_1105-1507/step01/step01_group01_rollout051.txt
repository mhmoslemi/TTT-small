"""
Search for low-score non-negative step-function coefficients.
Strategy: FFT-accelerated subgradient descent with adaptive top-K peak selection,
LP-based direction finding, multiscale initialization, and directed peak shaving.
Focuses on the n=512 regime where previous best was found, with careful

import time
import numpy as np
import scipy.fft as sp_fft

try:
    from scipy.optimize import linprog
    HAVE_SCIPY = True
except Exception:
    HAVE_SCIPY = False


def fft_self_convolve(a):
    """Compute full self-convolution of a using FFT."""
    n = a.size
    if n == 0:
        return np.array([], dtype=np.float64)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = sp_fft.rfft(a, fsize)
    return sp_fft.irfft(A * A, fsize)[:size]


def local_score(a):
    """Compute 2*n*max(convolve(a,a))/sum(a)^2."""
    a = np.asarray(a, dtype=np.float64)
    n = a.size
    if n == 0:
        return float('inf')
    a = np.clip(a, 0.0, 1000.0)
    s = float(np.sum(a))
    if s < 1e-12:
        return float('inf')
    c = fft_self_convolve(a)
    m = float(np.max(c))
    if not np.isfinite(m):
        return float('inf')
    return float(2.0 * n * m / (s * s))


def normalize(a):
    """Normalize to sum=1."""
    a = np.asarray(a, dtype=np.float64)
    a = np.clip(a, 0.0, 1000.0)
    s = float(np.sum(a))
    if s < 1e-300 or not np.isfinite(s):
        return None
    return a / s


def compute_gradient(a, c, K):
    """Compute subgradient from top-K peaks. d(c[k])/d(a[j]) = 2*a[k-j] if 0<=k-j<n."""
    n = a.size
    if K >= c.size:
        active = np.nonzero(c > 1e-15)[0]
        if active.size == 0:
            active = np.array([n - 1], dtype=np.int64)
    else:
        active = np.argpartition(c, -K)[-K:]

    w = c[active]
    w_max = float(np.max(w)) if w.size > 0 else 1.0
    if w_max > 0:
        w = w / w_max

    grad = np.zeros(n, dtype=np.float64)
    for idx in range(active.size):
        k = int(active[idx])
        wi = float(w[idx])
        lo = max(0, k - (n - 1))
        hi = min(n - 1, k)
        if lo <= hi:
            j = np.arange(lo, hi + 1)
            grad[j] += 2.0 * wi * a[k - j]

    return grad


def subgradient_step(a, eta, K, smooth_width=0):
    """One multiplicative subgradient step."""
    n = a.size
    c = fft_self_convolve(a)
    grad = compute_gradient(a, c, K)

    if smooth_width > 0:
        kern = np.ones(smooth_width) / smooth_width
        grad = np.convolve(grad, kern, mode='same')

    g_max = float(np.max(grad))
    if not np.isfinite(g_max) or g_max <= 0:
        return a.copy(), c

    mean_g = float(np.mean(grad))
    fac = eta * (mean_g - grad) / g_max
    fac = np.clip(fac, -3.0, 3.0)
    a_new = a * np.exp(fac)
    a_new = np.clip(a_new, 0.0, 1000.0)
    a_new = normalize(a_new)
    if a_new is None:
        return a.copy(), c
    return a_new, c


def subgradient_optimize(a, max_time, seed, K_init=30, eta_init=0.4):
    """Run subgradient optimization with adaptive parameters and momentum."""
    a = normalize(a)
    if a is None:
        return None, float('inf')
    n = a.size
    best = a.copy()
    bestsc = local_score(best)
    if max_time <= 0.5:
        return best, bestsc

    t0 = time.monotonic()
    deadline = t0 + float(max_time)
    rng = np.random.default_rng(seed)

    eta = float(eta_init)
    K = int(K_init)
    stuck = 0
    momentum = np.zeros(n, dtype=np.float64)
    beta = 0.7

    it = 0
    while time.monotonic() < deadline:
        # Adapt K: start small, grow as we converge
        if it % 200 == 0:
            K = int(np.clip(K * (0.9 + 0.3 * rng.random()), 10, min(200, n)))
            if stuck > 200 and eta > 0.1:
                eta *= 0.8

        c = fft_self_convolve(a)
        grad = compute_gradient(a, c, K)

        g_max = float(np.max(grad)) if grad.size > 0 else 1.0
        if g_max > 0 and np.isfinite(g_max):
            # Momentum update
            momentum = beta * momentum + (1 - beta) * grad
            mean_m = float(np.mean(momentum))
            fac = eta * (mean_m - momentum) / g_max
            fac = np.clip(fac, -3.0, 3.0)
            a_new = a * np.exp(fac)
        else:
            a_new = a.copy()

        a_new = np.clip(a_new, 0.0, 1000.0)
        a_new = normalize(a_new)
        if a_new is None:
            a_new = a.copy()

        sc = local_score(a_new)
        if np.isfinite(sc) and sc < bestsc:
            best = a_new.copy()
            bestsc = float(sc)
            stuck = 0
            # Slightly increase eta when making progress
            if eta < 0.6:
                eta = min(eta * 1.005, 0.6)
        else:
            stuck += 1
            if stuck % 50 == 0:
                a = best.copy()

        if stuck > 300:
            # Random perturbation to escape
            noise = np.exp(rng.normal(0.0, 0.15, n))
            a_new = best * noise
            a_new = normalize(a_new)
            if a_new is not None:
                a = a_new
            else:
                a = best.copy()
            momentum *= 0.5
            eta = max(0.2, eta * 0.7)
            stuck = 0

        a = a_new
        it += 1

    return best, bestsc


def build_lp_matrix(f, n):
    """Build constraint matrix for LP: (f*b)[k] for k=0..2n-2."""
    A = np.zeros((2 * n - 1, n), dtype=np.float64)
    for k in range(2 * n - 1):
        lo = max(0, k - (n - 1))
        hi = min(n - 1, k)
        if lo <= hi:
            j = np.arange(lo, hi + 1)
            A[k, j] = f[k - j]
    return A


def lp_direction(f, n, seed):
    """Find LP direction: maximize sum(b) s.t. (f*b)[k] <= M, b in [0,1]."""
    if not HAVE_SCIPY:
        return None

    f = normalize(f)
    if f is None:
        return None

    c = fft_self_convolve(f)
    M = float(np.max(c))
    if M <= 0 or not np.isfinite(M):
        return None

    rng = np.random.default_rng(seed + 777001)

    try:
        A = build_lp_matrix(f, n)
        cvec = -np.ones(n, dtype=np.float64)
        cvec += 1e-9 * rng.uniform(-1, 1, n)
        bvec = np.full(2 * n - 1, M * (1.0 + 1e-6), dtype=np.float64)
        bounds = [(0.0, 1.0)] * n

        res = linprog(cvec, A_ub=A, b_ub=bvec, bounds=bounds,
                      method='highs', options={'presolve': True})
    except Exception:
        return None

    if res is None or not res.success:
        return None

    b = np.asarray(res.x, dtype=np.float64)
    b = np.clip(b, 0.0, 1.0)
    s = float(np.sum(b))
    if s < 1e-10:
        return None
    b = b / s
    return b


def lp_optimize(a, max_time, seed):
    """Iterative LP-based optimization."""
    a = normalize(a)
    if a is None:
        return None, float('inf')
    n = a.size
    best = a.copy()
    bestsc = local_score(best)
    if max_time < 2.0 or not HAVE_SCIPY:
        return best, bestsc

    t0 = time.monotonic()
    deadline = t0 + float(max_time)
    ts = (0.03, 0.05, 0.08, 0.12, 0.18, 0.25, 0.35, 0.5)
    stall = 0

    while time.monotonic() < deadline:
        rem = deadline - time.monotonic()
        if rem < 3.0:
            break

        b = lp_direction(best, n, seed + int(stall * 13))
        if b is None:
            break

        new_best = best.copy()
        new_sc = bestsc

        for t in ts:
            cand = (1.0 - t) * best + t * b
            cand = normalize(cand)
            if cand is None:
                continue
            sc = local_score(cand)
            if np.isfinite(sc) and sc < new_sc:
                new_sc = float(sc)
                new_best = cand.copy()

        if new_sc < bestsc:
            best = new_best.copy()
            bestsc = float(new_sc)
            stall = 0
        else:
            stall += 1
            if stall > 15:
                break
            # Small subgradient polish between LP calls
            for _ in range(10):
                if time.monotonic() >= deadline:
                    break
                cand, _ = subgradient_step(best, 0.15, max(15, n // 20))
                sc = local_score(cand)
                if np.isfinite(sc) and sc < bestsc:
                    best = cand.copy()
                    bestsc = float(sc)
                    break

    return best, bestsc


def peak_shave(a, max_time, seed):
    """Targeted peak shaving: iteratively reduce the single largest peak."""
    a = normalize(a)
    if a is None:
        return None, float('inf')
    n = a.size
    best = a.copy()
    bestsc = local_score(best)
    if max_time < 1.0:
        return best, bestsc

    t0 = time.monotonic()
    deadline = t0 + float(max_time)
    rng = np.random.default_rng(seed)

    while time.monotonic() < deadline:
        c = fft_self_convolve(best)
        k_max = int(np.argmax(c))
        M = float(c[k_max])

        # Compute which pairs contribute to this peak
        lo = max(0, k_max - (n - 1))
        hi = min(n - 1, k_max)
        if lo > hi:
            break

        j_idx = np.arange(lo, hi + 1)
        contributions = best[j_idx] * best[k_max - j_idx]
        total_contrib = float(np.sum(contributions))
        if total_contrib < 1e-15:
            break

        # Reduce the largest contributors
        top_m = max(3, len(contributions) // 4)
        top_pos = np.argpartition(contributions, -top_m)[-top_m:]

        reduction = 0.05 + 0.03 * rng.random()
        for p in top_pos:
            j = int(j_idx[p])
            partner = k_max - j
            # Reduce both j and partner proportionally
            best[j] *= (1.0 - reduction * float(contributions[p]) / total_contrib)
            if partner != j and 0 <= partner < n:
                best[partner] *= (1.0 - reduction * 0.5 * float(contributions[p]) / total_contrib)

        best = np.clip(best, 0.0, 1000.0)
        best = normalize(best)
        if best is None:
            break

        sc = local_score(best)
        if np.isfinite(sc) and sc < bestsc:
            bestsc = float(sc)
        else:
            # Revert and try a different approach
            break

    return best, bestsc


def make_initialization(n, seed, kind=0, parent=None):
    """Generate a random or parent-based initialization."""
    rng = np.random.default_rng(seed + 999 * kind)

    if kind == 0 or parent is None:
        a = rng.uniform(0.1, 1.0, n)
    elif kind == 1:
        a = rng.beta(2.0, 5.0, n)
    elif kind == 2:
        a = 1.0 / (np.arange(n) + 1.0)
        a = a / np.max(a)
        a = a * rng.uniform(0.5, 1.5, n)
    elif kind == 3:
        # Smooth random
        a = rng.normal(size=n)
        k = np.ones(max(3, n // 16)) / (n // 16)
        a = np.convolve(a, k, mode='same')
        a = np.clip(a, 0.01, None)
    elif kind == 4:
        # Sine-based
        freq = rng.uniform(0.5, 3.0)
        phase = rng.uniform(0, 2 * np.pi)
        a = 0.5 + 0.5 * np.sin(2 * np.pi * freq * np.arange(n) / n + phase)
        a = np.clip(a, 0.01, None)
    elif kind == 5:
        # Parent perturbation
        if parent is not None and parent.size == n:
            a = parent * np.exp(rng.normal(0, 0.3, n))
        elif parent is not None:
            old = np.linspace(0, 1, parent.size)
            new = np.linspace(0, 1, n)
            a = np.interp(new, old, parent)
            a = a * np.exp(rng.normal(0, 0.3, n))
        else:
            a = rng.uniform(0.1, 1.0, n)
    else:
        a = rng.uniform(0.05, 1.0, n)

    a = np.clip(a, 0.0, 1000.0)
    a = normalize(a)
    if a is None:
        a = np.ones(n, dtype=np.float64) / n
    return a

                        print(f"Polish 4: {best_score:.6f}", flush=True)
            except Exception:
                pass

    # Fallback
    if best_seq is None:
        if parent_arr is not None:
            best_seq = normalize(parent_arr)
        if best_seq is None:
            best_seq = np.ones(64, dtype=np.float64) / 64.0

    final = normalize(best_seq)
    if final is None:
        final = np.ones(64, dtype=np.float64) / 64.0

    # Validate
    try:
        official = float(evaluate_sequence([float(x) for x in final]))
        if np.isfinite(official):
            best_score = official
        else:
            if parent_arr is not None:
                fp = normalize(parent_arr)
                if fp is not None:
                    try:
                        off2 = float(evaluate_sequence([float(x) for x in fp]))
                        if np.isfinite(off2):
                            final = fp
                            best_score = off2
                    except Exception:
                        pass
    except Exception:
        pass

    print(f"\nFINAL: score={best_score:.6f}, length={len(final)}", flush=True)
    return [float(x) for x in np.clip(np.asarray(final, dtype=np.float64), 0.0, 1000.0)]
