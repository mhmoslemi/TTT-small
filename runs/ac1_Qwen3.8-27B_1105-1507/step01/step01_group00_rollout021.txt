def build_conv_matrix_sparse(f):
    """Build sparse matrix A where (A@g)[k] = convolve(f,g)[k] for linear conv."""
    n = len(f)
    # For linear convolution, (conv(f,g))[k] = sum_j f[j] * g[k-j]
    # where 0 <= j < n and 0 <= k-j < n, i.e., max(0,k-n+1) <= j <= min(k, n-1)
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):

        j_max = min(k, n - 1)
        for j in range(j_min, j_max + 1):
            rows.append(k)
            cols.append(j)
            vals.append(f[k - j])
    return csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))


def solve_lp_direction(f, M, time_limit=30.0):

"""
Multi-stage optimization for minimizing 2n*max(conv(a,a))/sum(a)^2.
Combines LP-based direction finding, multi-peak subgradient polishing with
momentum, and diverse spectral initializations across multiple sequence lengths.
"""
import numpy as np
import time
from scipy.optimize import linprog
from scipy.sparse import csr_matrix
from scipy.interpolate import interp1d


def build_conv_matrix_sparse(f):
    """Build sparse matrix A where (A@g)[k] = convolve(f,g)[k] for linear conv."""
    n = len(f)
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        j_min = max(0, k - (n - 1))
        j_max = min(k, n - 1)
        for j in range(j_min, j_max + 1):
            rows.append(k)
            cols.append(j)
            vals.append(f[k - j])
    return csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))


def solve_lp_direction(f, M, time_limit=25.0):
    """Solve: max sum(g) s.t. g >= 0, conv(f,g)_k <= M for all k."""
    n = len(f)
    A = build_conv_matrix_sparse(f)
    c_obj = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    try:
        result = linprog(c_obj, A_ub=A, b_ub=b_ub, bounds=bounds,
                         method='highs', options={'time_limit': time_limit})
        if result.success:
            return result.x
    except Exception:
        pass
    return None


def compute_score(a):
    """Fast score: 2n*max(conv(a,a))/sum(a)^2."""
    n = len(a)
    s = float(np.sum(a))
    if s < 0.01:
        return float('inf')
    c = np.convolve(a, a)
    return float(2.0 * n * np.max(c) / (s * s))


def lp_optimize(f, deadline, max_iters=200, step_init=0.15, verbose=False, prefix=""):
    """LP-based iterative optimization."""
    f = f.astype(np.float64).copy()
    best_f = f.copy()
    best_score = compute_score(f)
    step = step_init
    no_improve = 0

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        c = np.convolve(f, f)
        M = float(np.max(c))
        if M <= 0:
            break
        g = solve_lp_direction(f, M, time_limit=20.0)
        if g is None:
            no_improve += 1
            if no_improve > 30:
                break
            continue
        sum_g = float(np.sum(g))
        if sum_g < 1e-15:
            no_improve += 1
            continue
        sum_f = float(np.sum(f))
        g = g * (sum_f / sum_g)
        new_f = (1.0 - step) * f + step * g
        new_f = np.clip(new_f, 0.0, 1e6)
        new_score = compute_score(new_f)
        if np.isfinite(new_score) and new_score < best_score:
            f = new_f
            best_f = f.copy()
            best_score = new_score
            no_improve = 0
            if verbose and it % 25 == 0:
                print(f"  {prefix}LP iter {it}: score={best_score:.6f} step={step:.3f}")
        else:
            no_improve += 1
            step *= 0.75
            if step < 0.003 or no_improve > 40:
                break
    return best_f, best_score


def multi_peak_subgradient(f, deadline, lr_init=0.015, momentum_beta=0.85,
                           threshold_frac=0.92, verbose=False, prefix=""):
    """Multi-peak subgradient with momentum. Targets all lags near the max."""
    f = f.astype(np.float64).copy()
    best_f = f.copy()
    best_score = compute_score(f)
    n = len(f)
    lr = lr_init
    momentum = np.zeros(n)
    no_improve = 0

    for it in range(5000):
        if time.monotonic() > deadline:
            break
        c = np.convolve(f, f)
        M = float(np.max(c))
        S = float(np.sum(f))
        if M <= 0 or S <= 0:
            break

        threshold = threshold_frac * M
        active = np.where(c >= threshold)[0]

        grad = np.zeros(n)
        for k in active:
            j_min = max(0, k - (n - 1))
            j_max = min(k, n - 1)
            grad[j_min:j_max + 1] += 2.0 * f[k - j_max:k - j_min + 1][::-1]

        # Normalize gradient
        gmax = np.max(grad)
        if gmax > 0:
            grad = grad / gmax * S

        momentum = momentum_beta * momentum + (1.0 - momentum_beta) * grad
        new_f = f - lr * momentum
        new_f = np.clip(new_f, 0.0, 1e6)
        new_S = float(np.sum(new_f))
        if new_S > 0:
            new_f = new_f * (S / new_S)

        new_score = compute_score(new_f)
        if np.isfinite(new_score) and new_score < best_score:
            f = new_f
            best_f = f.copy()
            best_score = new_score
            no_improve = 0
            if verbose and it % 200 == 0:
                print(f"  {prefix}SG iter {it}: score={best_score:.6f} lr={lr:.5f}")
        else:
            no_improve += 1
            lr *= 0.97
            if lr < 1e-9 or no_improve > 500:
                break
    return best_f, best_score


def make_inits(n, rng):
    """Generate diverse initializations of length n."""
    inits = []
    idx = np.arange(n, dtype=np.float64)

    # Uniform
    inits.append(('uniform', np.ones(n)))

    # Cosine window
    inits.append(('cosine', 0.5 * (1.0 + np.cos(2.0 * np.pi * idx / (n - 1)))))

    # Sum of random cosines
    for trial in range(3):
        a = np.ones(n)
        n_cos = rng.integers(2, 6)
        for _ in range(n_cos):
            freq = rng.uniform(0.5, 4.0)
            amp = rng.uniform(0.05, 0.4)
            ph = rng.uniform(0, 2.0 * np.pi)
            a += amp * np.cos(2.0 * np.pi * freq * idx / n + ph)
        a = np.clip(a, 0.01, None)
        inits.append((f'cossum{trial}', a))

    # Raised cosine / bell
    for beta in [1.0, 3.0, 8.0]:
        a = np.exp(-beta * (idx / (n - 1) - 0.5) ** 2)
        inits.append((f'bell{beta}', a))

    # Bimodal
    c1 = n * rng.uniform(0.15, 0.35)
    c2 = n * rng.uniform(0.65, 0.85)
    w = n * rng.uniform(0.05, 0.15)
    a = np.exp(-0.5 * ((idx - c1) / w) ** 2) + np.exp(-0.5 * ((idx - c2) / w) ** 2)
    inits.append(('bimodal', a))

    # Linear ramp
    inits.append(('ramp', np.linspace(0.1, 1.0, n)))

    # Quadratic (inverted parabola)
    t = idx / (n - 1)
    inits.append(('quadratic', 1.0 - (2.0 * t - 1.0) ** 2 + 0.1))

    # Random positive
    inits.append(('random', np.abs(rng.standard_normal(n)) + 0.1))

    # Alternating (high-low pattern)
    a_alt = np.ones(n)
    a_alt[1::2] = rng.uniform(0.3, 0.8)
    inits.append(('alternating', a_alt))

    # Step function (two levels)
    split = n * rng.uniform(0.3, 0.7)
    a_step = np.ones(n)
    a_step[int(split):] = rng.uniform(0.2, 0.6)
    inits.append(('step', a_step))

    return inits


def reshape_to_length(a, n):
    """Reshape sequence a to length n via interpolation."""
    if len(a) == n:
        return a.copy()
    old_x = np.linspace(0.0, 1.0, len(a))
    new_x = np.linspace(0.0, 1.0, n)
    f_interp = interp1d(old_x, a, kind='cubic', fill_value='extrapolate', assume_sorted=True)
    return np.clip(f_interp(new_x), 0.0, 1000.0)
