```python
            cols.append(j)
            vals.append(f[k-j])
    A = scipy.sparse.csr_matrix((vals, (rows, cols)), shape=(2*n-1, n))
```

```python
Sparse LP-based iterative optimization with coordinate descent polishing
for minimizing 2*n*max(convolve(a,a))/sum(a)^2 over non-negative sequences.
"""

import numpy as np
from scipy.optimize import linprog
from scipy import sparse
import time


def compute_score_fast(f):
    """Compute score = 2*n*max(conv)/sum^2 for sequence f."""
    n = len(f)
    S = f.sum()
    if S < 1e-15:
        return np.inf
    conv = np.convolve(f, f)
    M = conv.max()
    return 2.0 * n * M / (S * S)


def compute_score_normalized(f):
    """Compute score assuming f is already normalized to sum=1."""
    n = len(f)
    conv = np.convolve(f, f)
    M = conv.max()
    return 2.0 * n * M


def build_sparse_conv_matrix(f):
    """Build sparse matrix A such that A @ b = np.convolve(f, b, mode='full')."""
    n = len(f)
    # Column j: A[k, j] = f[k-j] for k in [j, j+n-1]
    rows = np.concatenate([np.arange(j, j + n) for j in range(n)])
    cols = np.repeat(np.arange(n), n)
    vals = np.tile(f, n)
    A = sparse.coo_matrix((vals, (rows, cols)), shape=(2 * n - 1, n)).tocsr()
    return A


def solve_lp_direction(f, M):
    """Solve LP: maximize sum(b) s.t. conv(f,b) <= M, b >= 0.
    Returns the optimal b (unnormalized)."""
    n = len(f)
    A = build_sparse_conv_matrix(f)
    c = -np.ones(n)

                  options={'presolve': True, 'time_limit': 30})

    return np.clip(res.x, 0, None)

def coordinate_descent_polish(f, rng, n_iter=5000, time_limit=None):
    """Polish a sequence using coordinate descent with random coordinate selection."""
    n = len(f)
    f = f.astype(np.float64).copy()
    f = f / f.sum()
    best_f = f.copy()
    best_score = compute_score_normalized(f)

    it = 0
    
    while it < n_iter:
        if time_limit is not None and time.monotonic() - start > time_limit:
            break

        i = rng.integers(0, n)
        
        # Try several values for f[i]
        # Current value
        val = f[i]
        S_excl = f.sum() - val
        
        # Try: val * factor for several factors
        candidates = [val]
        for factor in [0.5, 0.7, 1.3, 2.0, 0.3, 3.0]:
            candidates.append(val * factor)
        
        best_val = val
        best_local_score = best_score
        
        for new_val in candidates:
            if new_val < 0:
                continue
            f[i] = new_val
            S = S_excl + new_val
            if S < 1e-15:
                f[i] = val
                continue
            # Compute score
            conv = np.convolve(f, f)
            score = 2.0 * n * conv.max() / (S * S)
            if score < best_local_score:
                best_local_score = score
                best_val = new_val
        
        f[i] = best_val
        f = f / f.sum()
        
        if best_local_score < best_score - 1e-12:
            best_score = compute_score_normalized(f)
            best_f = f.copy()
        
        it += 1
    
    return best_f, best_score
```

```python
def project_simplex(v):
    """Project vector v onto the simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if v.sum() <= 0:
        return np.ones(n) / n
    # Algorithm from Duchi et al. (2012)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - 1))[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    w = np.maximum(v - theta, 0)
    return w
```

```python
"""
Sparse LP-based iterative optimization with active-set restriction,
coordinate descent polishing, and subgradient-assisted refinement
for minimizing 2*n*max(convolve(a,a))/sum(a)^2 over non-negative sequences.
"""

import numpy as np
from scipy.optimize import linprog
from scipy import sparse
import time


def compute_score_normalized(f):
    """Compute score assuming f is normalized to sum=1: 2*n*max(conv)."""
    n = len(f)
    conv = np.convolve(f, f)
    return 2.0 * n * conv.max()


def compute_score_general(f):
    """Compute the full score = 2*n*max(conv)/sum^2."""
    n = len(f)
    S = f.sum()
    if S < 1e-15:
        return np.inf
    conv = np.convolve(f, f)
    return 2.0 * n * conv.max() / (S * S)


def build_sparse_conv_rows(f, lags):
    """Build sparse matrix rows for given lags only.
    Returns a sparse matrix A such that A[k, j] corresponds to the constraint
    for lag lags[k]: sum_j f[k'-j] * b[j] <= M where k' = lags[k]."""
    n = len(f)
    rows = []
    cols = []
    vals = []
    for row_idx, k in enumerate(lags):
        k = int(k)
        j_start = max(0, k - n + 1)
        j_end = min(n, k + 1)
        for j in range(j_start, j_end):
            rows.append(row_idx)
            cols.append(j)
            vals.append(f[k - j])
    if not rows:
        return sparse.csr_matrix((len(lags), n))
    A = sparse.coo_matrix((vals, (rows, cols)), shape=(len(lags), n)).tocsr()
    return A


def solve_lp_full(f, M, time_limit=10.0):
    """Solve LP: maximize sum(b) s.t. conv(f,b) <= M (all lags), b >= 0."""
    n = len(f)
    lags = list(range(2 * n - 1))
    A = build_sparse_conv_rows(f, lags)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    try:
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': time_limit})
        if not res.success:
            return None
        return np.clip(res.x, 0, None)
    except Exception:
        return None
```

```python
            S = f.sum()
            if S < 1e-15:
                f[i] = val
                continue
            fn = f / S
            s = compute_score_normalized(fn)
            if s < best_local_s:
                best_local_s = s
                best_local = nv

        f[i] = best_local
        f = f / f.sum()
        if best_local_s < best_score - 1e-12:
            best_score = best_local_s
            best_f = f.copy()
            no_imp = 0
        else:
            no_imp += 1

    return best_f, best_score


def block_search(f, rng, time_limit=15.0):
    """Block perturbation local search."""
    n = len(f)
    f = f.astype(np.float64).copy()
    f = f / f.sum()
    best_f = f.copy()
    best_score = compute_score_normalized(f)
    start_time = time.monotonic()

    while time.monotonic() - start_time < time_limit:
        fc = f.copy()
        w = max(1, int(rng.integers(1, n // 6 + 2)))
        s = int(rng.integers(0, n - w + 1))
        fc[s:s + w] *= np.exp(rng.normal(0, 0.12, w))
        fc = np.clip(fc, 0, None)
        S = fc.sum()
        if S < 1e-15:
            continue
        fc = fc / S
        sc = compute_score_normalized(fc)
        if sc < best_score:
            best_score = sc
            best_f = fc.copy()
            f = fc

    return best_f, best_score


def propose_candidate(seed=42, budget_s=1500, **kwargs):
    """
    Multi-start sparse LP + subgradient optimization with coordinate descent
    polishing to minimize the step function evaluation score.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + budget_s - 25.0

    best_overall = None
    best_score_overall = np.inf

    # Parent sequence
    parent = None
    try:
        parent = np.array(height_sequence_1, dtype=np.float64)
        if parent.ndim != 1 or len(parent) == 0 or not np.isfinite(parent).all():
            parent = None
        else:
            parent = np.clip(parent, 0, None)
            if parent.sum() < 0.01:
                parent = None
    except Exception:
        parent = None

    print(f"Parent: {'n=' + str(len(parent)) if parent is not None else 'unavailable'}")
    print(f"Target: 1.5027 | Budget: {budget_s}s")

    # Starting points
    starts = []
    if parent is not None:
        starts.append((('parent', 3.0), parent.copy()))
        pp = parent * (1 + 0.15 * rng.standard_normal(len(parent)))
        pp = np.clip(pp, 0, None)
        if pp.sum() > 1e-10:
            starts.append((('parent_p', 1.5), pp))

    for n in [128, 256, 512, 768]:
        f0 = np.ones(n) * (1 + 0.12 * rng.standard_normal(n))
        f0 = np.clip(f0, 0.05, None)
        w = 1.5 if n <= 128 else (1.2 if n <= 256 else (1.0 if n <= 512 else 0.7))
        starts.append(((f'uniform_{n}', w), f0))

        x = np.linspace(0, 1, n)
        f0b = np.sin(np.pi * x) + 0.05 * rng.standard_normal(n)
        f0b = np.clip(f0b, 0.01, None)
        starts.append(((f'bell_{n}', w * 0.7), f0b))

    # Triangular
    n = 256
    x = np.arange(n)
    f0t = 1.0 - 2.0 * np.abs(x - n / 2.0) / (n / 2.0)
    f0t = f0t * (1 + 0.1 * rng.standard_normal(n))
    f0t = np.clip(f0t, 0.02, None)
    starts.append((('tri_256', 1.0), f0t))

    print(f"Starting points: {len(starts)}")

    # Time allocation
    total_avail = max(60, deadline - time.monotonic() - 40)
    w_sum = sum(w for (_, w) in starts)
    times = [total_avail * w / w_sum for (_, w) in starts]
    times = [min(t, total_avail * 0.35) for t in times]

    # Sort by weight descending
    order = np.argsort([-w for (_, w) in starts])

    for idx in order:
        if time.monotonic() > deadline:
            break
        (name, w), f0 = starts[idx]
        t_alloc = min(times[idx], deadline - time.monotonic() - 25)
        if t_alloc < 12:
            break

        f0 = f0.astype(np.float64)
        f0 = np.clip(f0, 0, None)
        if f0.sum() < 1e-10:
            continue
        f0 = f0 / f0.sum()
        init_score = compute_score_normalized(f0)

        print(f"\n[{name}] n={len(f0)} t={t_alloc:.0f}s init={init_score:.6f}")

        # Phase 1: Main optimization (72%)
        t1 = t_alloc * 0.72
        f_opt, score = optimize_combined(f0, t1, rng, verbose=True, top_k=25)

        # Phase 2: Coord descent (18%)
        t2 = min(t_alloc * 0.18, max(4, (deadline - time.monotonic()) * 0.15))
        if t2 > 3:
            f2, s2 = coord_descent(f_opt, rng, time_limit=t2)
            if s2 < score:
                f_opt, score = f2, s2
                print(f"  CD: {score:.6f}")

        # Phase 3: Block search (10%)
        t3 = min(t_alloc * 0.10, max(3, (deadline - time.monotonic()) * 0.08))
        if t3 > 2:
            f3, s3 = block_search(f_opt, rng, time_limit=t3)
            if s3 < score:
                f_opt, score = f3, s3
                print(f"  Block: {score:.6f}")

        if np.isfinite(score) and score < best_score_overall:
            best_score_overall = score
            best_overall = f_opt.copy()
            print(f"  >>> NEW BEST: {score:.6f}")

    # Final deep pass on best
    remaining = deadline - time.monotonic()
    if best_overall is not None and remaining > 50:
        print(f"\nFinal deep pass (remaining={remaining:.0f}s)")
        nb = len(best_overall)
        n_trials = min(5, int(remaining / 35))
        for tr in range(n_trials):
            t_left = deadline - time.monotonic() - 12
            if t_left < 18:
                break
            sig = 0.1 + 0.08 * tr
            f0 = best_overall * np.exp(sig * rng.standard_normal(nb))
            f0 = np.clip(f0, 0, None)
            if f0.sum() < 1e-10:
                continue
            f0 = f0 / f0.sum()
            t_opt = t_left * 0.65
            fr, sr = optimize_combined(f0, t_opt, rng, verbose=False, top_k=30)
            # Quick polish
            t_pol = deadline - time.monotonic() - 5
            t_pol = min(t_pol, t_left * 0.3)
            if t_pol > 3:
                fp, sp = coord_descent(fr, rng, time_limit=t_pol)
                if sp < sr:
                    fr, sr = fp, sp
            if np.isfinite(sr) and sr < best_score_overall:
                best_score_overall = sr
                best_overall = fr.copy()
                print(f"  Trial {tr}: >>> {sr:.6f}")

    # Final output
    if best_overall is not None:
        best_overall = np.clip(best_overall, 0, 1000)
        fl = best_overall.tolist()
        if sum(fl) < 0.01:
            fl = [1.0] * 256
        fs = evaluate_sequence(fl)
        print(f"\nFinal score (verified): {fs:.6f}")
        print(f"Length: {len(fl)}")
        return fl

    print("Fallback")
    return [1.0] * 128
```
