```python
def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    N = 1 << (size - 1).bit_length()
    fa = np.fft.rfft(a, N)
    result = np.fft.irfft(fa * fa, N)[:size]
    result[result < 0] = 0
    return result

def compute_score(c, n, sum_a):
    return 2.0 * n * np.max(c) / (sum_a * sum_a)

    sweep = 0

        sweep += 1
        improved = False
        # Random order
        order = rng.permutation(n)
        for i in order:
            if time.time() > deadline:
```

```python
"""
Optimizes non-negative sequence to minimize 2*n*max(conv(a,a))/sum(a)^2.
Uses SCA-LP for basin identification, fast coordinate descent with incremental
convolution updates for local refinement, spectral initialization, and
multi-length multi-restart search with time-budgeted phases.
"""

import numpy as np
import time
from scipy.optimize import linprog


def fast_convolve(a):
    """Compute full linear convolution of a with itself using FFT."""
    n = len(a)
    size = 2 * n - 1
    N = 1 << (size - 1).bit_length()
    fa = np.fft.rfft(a, N)
    result = np.fft.irfft(fa * fa, N)[:size]
    result[result < 0] = 0.0
    return result


def ternary_minimize(func, a, b, tol=1e-12, max_iter=80):
    """Ternary search for minimum of func on [a, b]."""
    for _ in range(max_iter):
        m1 = a + (b - a) / 3.0
        m2 = b - (b - a) / 3.0
        f1 = func(m1)
        f2 = func(m2)
        if f1 < f2:
            b = m2
        else:
            a = m1
        if b - a < tol:
            break
    return (a + b) / 2.0


def coord_obj(c, a, i, ai, x, S, n):
    """Evaluate 1D objective for coordinate i at value x."""
    # c_k(x) = c[k] + 2*(x-ai)*a[k-i] for 0<=k-i<n, k != 2i
    #         c[k] + x^2 - ai^2 for k = 2i
    #         c[k] for k-i<0 or k-i>=n
    delta = x - ai
    max_c = 0.0
    # Vectorized: affected k range is [i, i+n-1] intersected with [0, 2n-2]
    lo = i
    hi = min(2 * n - 1, i + n - 1) + 1
    if hi > lo:
        idx = np.arange(lo, hi)
        vals = c[idx].copy()
        vals += 2.0 * delta * a[idx - i]
        # Fix k=2i
        ki2 = 2 * i
        if lo <= ki2 < hi:
            vals[idx[0] == ki2] if False else None
            pos = ki2 - lo
            vals[pos] = c[ki2] + x * x - ai * ai
        local_max = np.max(vals)
        if local_max > max_c:
            max_c = local_max
    # Unaffected range
    if lo > 0:
        m = np.max(c[:lo])
        if m > max_c:
            max_c = m
    if hi < 2 * n - 1:
        m = np.max(c[hi:])
        if m > max_c:
            max_c = m
    denom = S + x
    if denom < 1e-15:
        return 1e15
    return max_c / (denom * denom)


def coordinate_descent(a, deadline, rng, max_sweeps=20000):
    """Local optimization via coordinate descent with incremental convolution updates."""
    n = len(a)
    a = a.copy().astype(np.float64)
    sum_a = a.sum()
    a = a / sum_a
    sum_a = 1.0
    c = fast_convolve(a)
    best_score = 2.0 * n * np.max(c) / (sum_a * sum_a)
    best_a = a.copy()

    sweep = 0
    stall = 0
    while time.time() < deadline and sweep < max_sweeps:
        sweep += 1
        order = rng.permutation(n)
        improved = False
        for i in order:
            if time.time() > deadline:
                break
            ai = a[i]
            S = sum_a - ai

            # 1D minimization: minimize max_k c_k(x) / (S+x)^2
            # Use ternary search on [0, 1.0] (since sum(a)=1, a_i <= 1)
            def obj_1d(x):
                return coord_obj(c, a, i, ai, x, S, n)

            x_best = ai  # default: no change
            # Coarse grid to find good bracket
            grid = np.linspace(0.0, 1.0, 20)
            best_x = ai
            best_val = obj_1d(ai)
            for gx in grid:
                v = obj_1d(gx)
                if v < best_val:
                    best_val = v
                    best_x = gx
            # Refine with ternary search around best_x
            lo_b = max(0.0, best_x - 0.1)
            hi_b = min(1.0, best_x + 0.1)
            if hi_b - lo_b > 1e-10:
                x_refined = ternary_minimize(obj_1d, lo_b, hi_b)
                v_ref = obj_1d(x_refined)
                if v_ref < best_val:
                    best_x = x_refined
                    best_val = v_ref
            x_best = best_x

            delta = x_best - ai
            if abs(delta) < 1e-14:
                continue

            # Update a and convolution incrementally
            a[i] = x_best
            sum_a += delta

            # Update c incrementally
            lo = i
            hi = min(2 * n - 1, i + n - 1) + 1
            if hi > lo:
                idx = np.arange(lo, hi)
                c[idx] += 2.0 * delta * a[idx - i]
                # Fix k = 2i (was double-counted in the linear update)
                ki2 = 2 * i
                if lo <= ki2 < hi:
                    # The linear update added 2*delta*a[i] = 2*delta*x_best
                    # But the correct update is x_best^2 - ai^2
                    # So subtract the incorrect and add the correct
                    c[ki2] -= 2.0 * delta * x_best
                    c[ki2] += x_best * x_best - ai * ai

        # End of sweep
        score = 2.0 * n * np.max(c) / (sum_a * sum_a)
        if score < best_score - 1e-12:
            best_score = score
            best_a = a.copy()
            stall = 0
            improved = True
        else:
            stall += 1

        # Renormalize periodically
        if sweep % 100 == 0:
            a = a / sum_a
            sum_a = 1.0
            c = fast_convolve(a)

        # Perturbation to escape stagnation
        if stall > 50:
            pert_size = 0.005 * min(1.0, float(stall) / 200.0)
            a = a + pert_size * rng.standard_normal(n)
            a = np.maximum(a, 0.0)
            s = a.sum()
            if s < 1e-10:
                a = np.ones(n) / n
                sum_a = 1.0
            else:
                a = a / s
                sum_a = 1.0
            c = fast_convolve(a)
            stall = 0

    return best_a, best_score


def sca_lp_one_step(a, c, n):
    """One SCA iteration: solve LP with linearized convolution constraints."""
    # LP: minimize t
    # s.t. 2 * sum_m a_{k-m}^0 * a_m <= t + c_k^0  for all k
    #      sum(a) = 1, a >= 0
    # Variables: a_0,...,a_{n-1}, t => n+1 variables

    # Build constraint matrix for the 2n-1 convolution constraints
    # Row k: [2*a_{k-0}^0, 2*a_{k-1}^0, ..., 2*a_{k-(n-1)}^0, -1] <= c_k^0
    A = np.zeros((2 * n - 1, n + 1))
    b_ub = np.empty(2 * n - 1)

    for k in range(2 * n - 1):
        for m in range(n):
            ki = k - m
            if 0 <= ki < n:
                A[k, m] = 2.0 * a[ki]
        A[k, n] = -1.0
        b_ub[k] = c[k]

    # Objective: minimize t => c_obj = [0,...,0,1]
    c_obj = np.zeros(n + 1)
    c_obj[n] = 1.0

    # Bounds: a_i >= 0, t unbounded below (but will be positive)
    bounds = [(0.0, None)] * n + [(None, None)]

    # Equality: sum(a) = 1
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])

    try:
        res = linprog(c_obj, A_ub=A, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs',
                      options={'time_limit': 10.0, 'presolve': True})
        if res.success:
            a_new = res.x[:n]
            a_new = np.maximum(a_new, 0.0)
            s = a_new.sum()
            if s > 1e-10:
                a_new = a_new / s
                return a_new
    except Exception:
        pass
    return None


def optimize_sca_cd(a, deadline, rng, use_lp=True, max_lp_iters=50):
    """Combined SCA-LP + coordinate descent optimizer."""
    n = len(a)
    a = a.copy().astype(np.float64)
    s = a.sum()
    if s < 1e-10:
        a = np.ones(n) / n
    else:
        a = a / s
    c = fast_convolve(a)
    best_score = 2.0 * n * np.max(c)
    best_a = a.copy()

    # Phase 1: SCA-LP (if enabled and n is reasonable)
    if use_lp and n <= 2048:
        lp_deadline = min(deadline - 60, time.time() + 60.0)
        for it in range(max_lp_iters):
            if time.time() > lp_deadline:
                break
            a_new = sca_lp_one_step(a, c, n)
            if a_new is None:
                break
            c_new = fast_convolve(a_new)
            score_new = 2.0 * n * np.max(c_new)
            if score_new < best_score - 1e-10:
                best_score = score_new
                best_a = a_new.copy()
                a = a_new
                c = c_new
            else:
                # Try mixing
                for alpha in [0.5, 0.3, 0.1]:
                    a_mix = (1.0 - alpha) * a + alpha * a_new
                    a_mix = np.maximum(a_mix, 0)
                    sm = a_mix.sum()
                    if sm < 1e-10:
                        continue
                    a_mix = a_mix / sm
                    c_mix = fast_convolve(a_mix)
                    score_mix = 2.0 * n * np.max(c_mix)
                    if score_mix < best_score - 1e-10:
                        best_score = score_mix
                        best_a = a_mix.copy()
                        a = a_mix
                        c = c_mix
                        break

    # Phase 2: Coordinate descent
    cd_deadline = deadline
    a_cd, score_cd = coordinate_descent(a, cd_deadline, rng)
    if score_cd < best_score:
        best_score = score_cd
        best_a = a_cd

    return best_a, best_score


def spectral_init(target_n, reference, rng):
    """Initialize a sequence of target_n using spectral information from reference."""
    N_fft = 1 << (2 * target_n - 2).bit_length()
    ref_len = min(len(reference), target_n)
    if target_n > ref_len:
        ref_padded = np.zeros(target_n)
        ref_padded[:ref_len] = reference[:ref_len]
    else:
        ref_padded = reference[:target_n].copy()

    F = np.fft.rfft(ref_padded, N_fft)
    mag = np.abs(F)

    # Generate new phases
    phases = rng.uniform(0.0, 2.0 * np.pi, len(mag))
    # Keep low-frequency phases from reference (they carry structure)
    n_keep = min(len(mag), target_n // 4)
    phases[:n_keep] = np.angle(F[:n_keep])

    F_new = mag * np.exp(1j * phases)
    a = np.fft.irfft(F_new, N_fft)[:target_n]
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-10:
        a = rng.random(target_n)
        a = a / a.sum()
    else:
        a = a / s
    return a


def template_init(n, template_type, rng):
    """Initialize from a parametric template."""
    x = np.linspace(0.0, 1.0, n)
    if template_type == 'cosine':
        a = 0.5 * (1.0 + np.cos(2.0 * np.pi * x * rng.uniform(1, 5))) + 0.05
    elif template_type == 'gaussian':
        center = rng.uniform(0.3, 0.7)
        width = rng.uniform(0.1, 0.4)
        a = np.exp(-((x - center) ** 2) / (2.0 * width ** 2)) + 0.01
    elif template_type == 'riemann':
        # Based on Riemann-von Mangoldt style weights
        a = 1.0 / (np.abs(np.sin(np.pi * x)) + 0.1)
    elif template_type == 'sinc':
        freq = rng.uniform(2, 10)
        a = np.abs(np.sinc(freq * (x - 0.5))) + 0.05
    elif template_type == 'bimodal':
        c1, c2 = sorted(rng.uniform(0.2, 0.8, 2))
        w = rng.uniform(0.05, 0.2)
        a = np.exp(-((x - c1) ** 2) / (2 * w ** 2)) + np.exp(-((x - c2) ** 2) / (2 * w ** 2)) + 0.01
    elif template_type == 'triangle':
        a = 1.0 - 2.0 * np.abs(x - 0.5) + 0.05
    else:
        a = 1.0 + 0.1 * rng.standard_normal(n)
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-10:
        a = np.ones(n) / n
    else:
        a = a / s
    return a

        best_seq = a_opt.copy()
    else:
        # Use the internal score (scale-invariant)
        if score_opt < best_score:
            best_score = score_opt
            best_seq = a_opt.copy()

    # ============================================================
    # Phase 2: Try other lengths with spectral/template init
    # ============================================================
    print("Phase 2: Exploring other lengths")
    other_lengths = [384, 640, 768, 896, 1024]
    templates = ['cosine', 'sinc', 'bimodal', 'gaussian', 'triangle']

    remaining_time = deadline - time.time()
    if remaining_time < 60:
        print("  Not enough time for Phase 2")
    else:
        time_per_config = remaining_time / (len(other_lengths) + 2)

        # For each length, try spectral init from parent + one template
        for n_try in other_lengths:
            if time.time() > deadline - 60:
                break

            # Spectral init from parent
            a_spec = spectral_init(n_try, parent, rng)
            config_deadline = min(deadline - 30, time.time() + time_per_config * 1.5)

            a_opt2, score2 = optimize_sca_cd(a_spec, config_deadline, rng,
                                             use_lp=(n_try <= 1536),
                                             max_lp_iters=20)
            score2_eval = evaluate_sequence(a_opt2.tolist())
            print(f"  n={n_try} spectral: score={score2_eval:.6f}")
            if np.isfinite(score2_eval) and score2_eval < best_score:
                best_score = score2_eval
                best_seq = a_opt2.copy()
                print(f"  >>> New best: {best_score:.6f}")

        # Try a few templates at the best length
        best_n = len(best_seq) if best_seq is not None else 512
        for tmpl in templates[:3]:
            if time.time() > deadline - 60:
                break
            a_tmpl = template_init(best_n, tmpl, rng)
            config_deadline = min(deadline - 30, time.time() + time_per_config)
            a_opt3, score3 = optimize_sca_cd(a_tmpl, config_deadline, rng,
                                             use_lp=(best_n <= 1536),
                                             max_lp_iters=15)
            score3_eval = evaluate_sequence(a_opt3.tolist())
            print(f"  n={best_n} {tmpl}: score={score3_eval:.6f}")
            if np.isfinite(score3_eval) and score3_eval < best_score:
                best_score = score3_eval
                best_seq = a_opt3.copy()
                print(f"  >>> New best: {best_score:.6f}")

    # ============================================================
    # Phase 3: Final refinement with coordinate descent on best
    # ============================================================
    if best_seq is not None:
        print(f"Phase 3: Final refinement (n={len(best_seq)}, current best={best_score:.6f})")
        final_deadline = deadline
        a_final, score_final = coordinate_descent(best_seq, final_deadline, rng, max_sweeps=50000)
        score_final_eval = evaluate_sequence(a_final.tolist())
        print(f"  After final CD: score={score_final_eval:.6f}")
        if np.isfinite(score_final_eval) and score_final_eval < best_score:
            best_score = score_final_eval
            best_seq = a_final.copy()

    # ============================================================
    # Phase 4: One more SCA refinement if time allows
    # ============================================================
    if best_seq is not None and time.time() < deadline - 30:
        print("Phase 4: Final SCA polish")
        polish_deadline = deadline
        a_polish, score_polish = optimize_sca_cd(best_seq, polish_deadline, rng,
                                                 use_lp=(len(best_seq) <= 2048),
                                                 max_lp_iters=20)
        score_polish_eval = evaluate_sequence(a_polish.tolist())
        print(f"  After SCA polish: score={score_polish_eval:.6f}")
        if np.isfinite(score_polish_eval) and score_polish_eval < best_score:
            best_score = score_polish_eval
            best_seq = a_polish.copy()

    # Final validation
    if best_seq is None:
        print("Falling back to uniform sequence")
        best_seq = np.ones(512) / 512

    result = [float(x) for x in best_seq]
    final_score = evaluate_sequence(result)
    print(f"FINAL: score={final_score:.6f}, length={len(result)}")
    return result
```
