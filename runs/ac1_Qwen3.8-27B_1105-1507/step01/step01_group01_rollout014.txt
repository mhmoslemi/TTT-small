```python
def lp_min_peak(f, seed):
    """Find b with sum(b)=1, b>=0 minimizing max_k (f*b)(k)."""
    n = len(f)
    # (f*b)(k) = sum_j f[j] * b[k-j] for valid indices
    # Build constraint matrix A where row k, col j gives coefficient of b_j in (f*b)(k)
    # (f*b)(k) = sum_j b_j * f[k-j] (where f[k-j] = 0 if k-j out of [0,n-1])
    
    # A[k, j] = f[k-j] if 0 <= k-j < n, else 0
    # For k in [0, 2n-2], j in [0, n-1]
    
    A = np.zeros((2*n-1, n))
    for k in range(2*n-1):
        for j in range(n):
            idx = k - j
            if 0 <= idx < n:
                A[k, j] = f[idx]
    
    # LP: minimize t
    # subject to A @ b - t <= 0 (i.e., (f*b)(k) <= t for all k)
    #          sum(b) = 1
    #          b >= 0
    #          t free
    
    # In linprog form: minimize c^T x
    # x = [b_0, ..., b_{n-1}, t]
    # c = [0, ..., 0, 1]
    # A_ub @ x <= b_ub: A @ [b, t] <= 0 => A_b @ b + A_t * t <= 0
    #   A_b = A, A_t = -1 (column of -1s)
    # A_eq @ x = b_eq: sum(b) = 1 => [1,...,1, 0] @ x = 1
    # bounds: b >= 0, t free
    
    c = np.concatenate([np.zeros(n), [1.0]])
    A_ub = np.hstack([A, -np.ones((2*n-1, 1))])
    b_ub = np.zeros(2*n-1)
    A_eq = np.hstack([np.ones(n), [0.0]])
    b_eq = np.array([1.0])
    bounds = [(0, None)] * n + [(None, None)]
    
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq, 
                  bounds=bounds, method='highs')
    if res is not None and res.success:
        b_opt = res.x[:n]
        b_opt = np.clip(b_opt, 0, None)
        s = np.sum(b_opt)
        if s > 0:
            b_opt /= s
        return b_opt
    return None
```

```python
"""
Hybrid optimization for minimizing 2*n*max(convolve(a,a))/sum(a)^2.

Uses smooth-max gradient descent with momentum, LP-based cross-correlation
minimization for periodic refinement, simulated-annealing-style restarts,
and multi-length exploration. The key innovation over previous approaches:
(1) exact gradient computation via smooth-max + FFT-free correlate,
(2) LP that directly minimizes peak cross-correlation (more principled
   than the previous maximize-sum-under-bound formulation),
(3) adaptive multi-scale alpha schedule for the smooth-max approximation.

import time
import numpy as np
from scipy.optimize import linprog


def normalize_sum1(a):
    """Normalize non-negative array to sum=1."""
    a = np.asarray(a, dtype=np.float64)
    a = np.clip(a, 0.0, 1000.0)
    s = float(np.sum(a))
    if s < 1e-300:
        return np.ones_like(a, dtype=np.float64) / max(1, len(a))
    return a / s


def compute_score(a):
    """Compute the objective: 2*n*max(conv(a,a)) / sum(a)^2."""
    a = np.asarray(a, dtype=np.float64)
    n = len(a)
    if n == 0:
        return float('inf')
    a = np.clip(a, 0.0, 1000.0)
    s = float(np.sum(a))
    if s < 1e-12:
        return float('inf')
    c = np.convolve(a, a)
    m = float(np.max(c))
    if not np.isfinite(m):
        return float('inf')
    return float(2.0 * n * m / (s * s))


def smooth_max_gradient(a, alpha):
    """
    Compute gradient of (1/alpha)*log(sum_k exp(alpha * c_k)) where c = a*a.
    Returns (gradient, convolution, c_max).
    """
    n = len(a)
    c = np.convolve(a, a)
    c_max = float(np.max(c))
    # Stable softmax weights
    exp_c = np.exp(alpha * (c - c_max))
    Z = float(np.sum(exp_c))
    if Z < 1e-300:
        return np.zeros(n), c, c_max
    w = exp_c / Z
    # Gradient: 2 * correlate(w, a, 'valid')
    grad = 2.0 * np.correlate(w, a, mode='valid')
    return grad, c, c_max


def smooth_max_value(c, alpha):
    """Compute smooth max value from convolution."""
    c_max = float(np.max(c))
    exp_c = np.exp(alpha * (c - c_max))
    return c_max + float(np.log(np.sum(exp_c))) / alpha

    """One momentum gradient descent step on smooth max."""
    grad, c, c_max = smooth_max_gradient(a, alpha)
    return grad, c, c_max


class SmoothMaxOptimizer:
    """Stateful optimizer with momentum."""
    def __init__(self, alpha=10.0, lr=0.05, momentum=0.9):
        self.alpha = alpha
        self.lr = lr
        self.momentum = momentum
        self.v = None

    def step(self, a):
        n = len(a)
        if self.v is None or len(self.v) != n:
            self.v = np.zeros(n)
        grad, c, c_max = smooth_max_gradient(a, self.alpha)
        self.v = self.momentum * self.v + (1.0 - self.momentum) * grad
        a_new = a - self.lr * self.v
        a_new = np.clip(a_new, 0.0, None)
        a_new = normalize_sum1(a_new)
        return a_new, grad, c, c_max

    def reset_momentum(self):
        if self.v is not None:
            self.v *= 0.0


def build_lp_matrix(f, n):
    """Build constraint matrix A where (f*b)(k) = (A @ b)_k."""
    A = np.zeros((2 * n - 1, n), dtype=np.float64)
    for k in range(2 * n - 1):
        j_min = max(0, k - (n - 1))
        j_max = min(n - 1, k)
        if j_min <= j_max:
            A[k, j_min:j_max + 1] = f[k - j_max:k - j_min + 1][::-1]
    return A


def lp_min_crosscorr(f, n, timeout=10.0):

        c_vec = np.concatenate([np.zeros(n), [1.0]])

        else:
            alpha_idx = 4

        alpha = alphas[alpha_idx]

        # Batch of gradient steps
        batch = 100
        for _ in range(batch):
            if time.monotonic() >= deadline:
                break
            grad, c, c_max = smooth_max_gradient(a, alpha)
            v = momentum * v + (1.0 - momentum) * grad
            a_new = a - lr * v
            a_new = np.clip(a_new, 0.0, None)
            a_new = normalize_sum1(a_new)
            sc = compute_score(a_new)
            if sc < best_score:
                best = a_new.copy()
                best_score = sc
                stagnant = 0
                last_improve_time = time.monotonic()
            else:
                stagnant += 1
            a = a_new

        # Adaptive learning rate
        if stagnant > 5 * batch:
            lr *= 0.7
            v *= 0.5
            if lr < 0.001:
                # Restart from best with noise
                noise = np.exp(rng.normal(0, 0.15, n))
                a = normalize_sum1(best * noise)
                v = np.zeros(n)
                lr = 0.03
                stagnant = 0
        elif stagnant > 2 * batch:
            lr *= 0.9
            momentum = max(0.7, momentum * 0.99)
        elif stagnant == 0:
            # Allow slightly larger lr when making progress
            lr = min(lr * 1.01, 0.1)

        # Periodic LP refinement
        if time.monotonic() - last_improve_time > 8.0 and phase_rem > 15.0:
            lp_timeout = min(8.0, 0.3 * phase_rem)
            b_opt, lp_val = lp_min_crosscorr(best, n, timeout=lp_timeout)
            if b_opt is not None:
                improved = False
                for tau in [0.02, 0.05, 0.1, 0.15, 0.2, 0.3, 0.5]:
                    blend = (1.0 - tau) * best + tau * b_opt
                    blend = normalize_sum1(blend)
                    sc = compute_score(blend)
                    if sc < best_score:
                        best = blend.copy()
                        best_score = sc
                        a = blend.copy()
                        stagnant = 0
                        last_improve_time = time.monotonic()
                        improved = True
                if improved:
                    print(f"    [{n}] LP improved to {best_score:.6f}", flush=True)

        # Check if we should try a different initialization
        if time.monotonic() - last_improve_time > 60.0 and frac < 0.8:
            # Try blending with a fresh random
            a_new = random_init(n, seed + int(time.monotonic() * 1000))
            for tau in [0.2, 0.4, 0.6]:
                blend = (1.0 - tau) * best + tau * a_new
                blend = normalize_sum1(blend)
                sc = compute_score(blend)
                if sc < best_score:
                    best = blend.copy()
                    best_score = sc
                    a = blend.copy()
                    stagnant = 0
                    last_improve_time = time.monotonic()
                    break
            else:
                # No improvement, perturb current
                noise = np.exp(rng.normal(0, 0.3, n))
                a = normalize_sum1(best * noise)
                v = np.zeros(n)
                stagnant = 0
                last_improve_time = time.monotonic()

    return best, best_score


def multi_alpha_optimize(n, time_budget, seed, parent_arr=None):
    """
    Run optimization with multiple parallel alpha values,
    keeping the best trajectory.
    """
    t0 = time.monotonic()
    deadline = t0 + time_budget
    rng = np.random.default_rng(seed)

    # Start from parent or random
    if parent_arr is not None and len(parent_arr) == n:
        a = normalize_sum1(parent_arr.copy())
    else:
        a = random_init(n, seed)
    best = a.copy()
    best_score = compute_score(best)

    alphas = [3.0, 8.0, 20.0, 50.0, 100.0, 250.0, 600.0]
    # For each alpha, maintain a separate state
    states = []
    for alpha in alphas:
        s = {
            'a': best.copy(),
            'v': np.zeros(n),
            'lr': 0.02,
            'momentum': 0.9,
            'score': compute_score(best.copy()),
            'stagnant': 0,
        }
        states.append(s)

    iteration = 0
    while time.monotonic() < deadline:
        if deadline - time.monotonic() < 3:
            break
        elapsed = time.monotonic() - t0
        frac = elapsed / time_budget

        # Higher alphas get more weight as we progress
        if frac < 0.2:
            active_alphas = alphas[:3]
        elif frac < 0.4:
            active_alphas = alphas[:4]
        elif frac < 0.7:
            active_alphas = alphas[2:]
        else:
            active_alphas = alphas[3:]

        for i, alpha in enumerate(alphas):
            if alpha not in active_alphas:
                continue
            state = states[i]
            a_i = state['a']

            # Gradient step
            grad, c, c_max = smooth_max_gradient(a_i, alpha)
            state['v'] = state['momentum'] * state['v'] + (1.0 - state['momentum']) * grad
            a_new = a_i - state['lr'] * state['v']
            a_new = np.clip(a_new, 0.0, None)
            a_new = normalize_sum1(a_new)
            sc = compute_score(a_new)

            if sc < state['score']:
                state['a'] = a_new
                state['score'] = sc
                state['stagnant'] = 0
                if sc < best_score:
                    best = a_new.copy()
                    best_score = sc
            else:
                state['stagnant'] += 1
                state['a'] = a_new

            # Adapt
            if state['stagnant'] > 300:
                state['lr'] *= 0.7
                if state['lr'] < 0.0005:
                    state['v'] *= 0.1
                    state['lr'] = 0.01

        # Periodic LP
        if iteration % 20 == 0 and deadline - time.monotonic() > 20:
            lp_tmo = min(6.0, 0.2 * (deadline - time.monotonic()))
            b_opt, _ = lp_min_crosscorr(best, n, timeout=lp_tmo)
            if b_opt is not None:
                for tau in [0.03, 0.08, 0.15, 0.25, 0.4]:
                    blend = (1.0 - tau) * best + tau * b_opt
                    blend = normalize_sum1(blend)
                    sc = compute_score(blend)
                    if sc < best_score:
                        best = blend.copy()
                        best_score = sc
                        # Update all states to share the improvement
                        for i in range(len(states)):
                            states[i]['a'] = best.copy()
                            states[i]['v'] *= 0.5
                            states[i]['stagnant'] = 0
                            states[i]['score'] = best_score
                        print(f"    [{n}] LP blend improved to {best_score:.6f}", flush=True)

        iteration += 1

    return best, best_score


def propose_candidate(seed=42, budget_s=1500, **kwargs):
    start_time = time.monotonic()
    try:
        budget = float(budget_s)
    except (TypeError, ValueError):
        budget = 1500.0
    if not np.isfinite(budget) or budget <= 0:
        budget = 1500.0
    deadline = start_time + budget - 15.0  # Safety margin

    # Get parent
    parent_arr = None
    try:
        p = globals().get('height_sequence_1', None)
        if p is not None:
            parent_arr = np.asarray(p, dtype=np.float64)
            parent_arr = np.clip(parent_arr, 0.0, 1000.0)
            if parent_arr.size == 0 or not np.isfinite(parent_arr).all():
                parent_arr = None
    except Exception:
        parent_arr = None

    best_seq = None
    best_score = float('inf')

    if parent_arr is not None:
        pn = len(parent_arr)
        p_norm = normalize_sum1(parent_arr)
        best_seq = p_norm.copy()
        best_score = compute_score(p_norm)
        print(f"Parent: n={pn}, score={best_score:.6f}", flush=True)

    # Length plan: focus on lengths near the parent, but also explore
    if parent_arr is not None:
        pn = len(parent_arr)
        lengths = sorted(set([
            max(64, pn // 2),
            max(64, int(pn * 0.75)),
            pn,
            min(2048, pn + pn // 4),
            min(2048, pn + pn // 2),
            min(2048, int(pn * 1.5)),
        ]))
    else:
        lengths = [256, 384, 512, 640, 768, 1024]

    # Allocate time proportionally, giving more to the parent length
    total_alloc = sum(1.0 + (1.0 if (parent_arr is not None and l == len(parent_arr)) else 0.0) for l in lengths)
    time_per_length = {}
    for l in lengths:
        weight = 1.0 + (1.0 if (parent_arr is not None and l == len(parent_arr)) else 0.0)
        time_per_length[l] = (weight / total_alloc) * (budget - 30.0)

    seed_cur = int(seed)
    for n in lengths:
        rem = deadline - time.monotonic()
        if rem < 15:
            break
        alloc = min(time_per_length[n], rem - 10.0)
        if alloc < 10:
            alloc = max(5.0, 0.4 * rem)

        print(f"\n--- Optimizing length {n} (budget {alloc:.0f}s) ---", flush=True)

        try:
            # Run with multi-alpha approach
            cand, sc = multi_alpha_optimize(n, alloc, seed_cur, parent_arr)
            if cand is not None and sc < best_score:
                best_seq = cand.copy()
                best_score = float(sc)
                print(f"  [{n}] improved to {best_score:.6f}", flush=True)
            else:
                print(f"  [{n}] best here: {sc:.6f}", flush=True)
        except Exception as e:
            print(f"  [{n}] error: {e}", flush=True)

        seed_cur += 100003

    # Final polishing: use remaining time on the best sequence
    if best_seq is not None:
        rem = deadline - time.monotonic()
        if rem > 10:
            n = len(best_seq)
            print(f"\n--- Final polish (n={n}, budget {rem:.0f}s) ---", flush=True)
            try:
                # High-alpha gradient descent for fine-tuning
                a = best_seq.copy()
                best_p = a.copy()
                best_ps = best_score
                v = np.zeros(n)
                lr = 0.005
                mom = 0.95
                alpha = 500.0
                stagnant = 0

                while time.monotonic() < deadline:
                    if deadline - time.monotonic() < 2:
                        break
                    for _ in range(50):
                        if time.monotonic() >= deadline:
                            break
                        grad, c, c_max = smooth_max_gradient(a, alpha)
                        v = mom * v + (1.0 - mom) * grad
                        a_new = a - lr * v
                        a_new = np.clip(a_new, 0.0, None)
                        a_new = normalize_sum1(a_new)
                        sc = compute_score(a_new)
                        if sc < best_ps:
                            best_p = a_new.copy()
                            best_ps = sc
                            stagnant = 0
                        else:
                            stagnant += 1
                        a = a_new

                    if stagnant > 2000:
                        lr *= 0.5
                        if lr < 0.0001:
                            # Try LP one more time
                            b_opt, _ = lp_min_crosscorr(best_p, n, timeout=5.0)
                            if b_opt is not None:
                                for tau in [0.01, 0.02, 0.05]:
                                    blend = (1.0 - tau) * best_p + tau * b_opt
                                    blend = normalize_sum1(blend)
                                    sc = compute_score(blend)
                                    if sc < best_ps:
                                        best_p = blend.copy()
                                        best_ps = sc
                            break

                    if stagnant > 0 and stagnant % 500 == 0:
                        print(f"    polish: lr={lr:.6f} score={best_ps:.6f} stagnant={stagnant}", flush=True)

                if best_ps < best_score:
                    best_seq = best_p.copy()
                    best_score = best_ps
                    print(f"  Polish improved to {best_score:.6f}", flush=True)
            except Exception as e:
                print(f"  Polish error: {e}", flush=True)

    # Fallback
    if best_seq is None or not np.isfinite(best_score):
        if parent_arr is not None:
            best_seq = normalize_sum1(parent_arr)
        else:
            best_seq = np.ones(128, dtype=np.float64) / 128.0
        best_score = compute_score(best_seq)

    # Final validation
    final = normalize_sum1(best_seq)
    final = np.clip(final, 0.0, 1000.0)
    try:
        official = float(evaluate_sequence([float(x) for x in final]))
        if np.isfinite(official):
            best_score = official
        else:
            if parent_arr is not None:
                fp = normalize_sum1(parent_arr)
                fp = np.clip(fp, 0.0, 1000.0)
                try:
                    official2 = float(evaluate_sequence([float(x) for x in fp]))
                    if np.isfinite(official2):
                        final = fp
                        best_score = official2
                except Exception:
                    pass
    except Exception:
        pass

    elapsed = time.monotonic() - start_time
    print(f"\nFINAL: score={best_score:.6f}, n={len(final)}, time={elapsed:.1f}s", flush=True)
    return [float(x) for x in final]
```
