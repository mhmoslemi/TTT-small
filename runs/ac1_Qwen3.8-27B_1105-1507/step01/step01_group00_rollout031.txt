import numpy as np
import time
from scipy.optimize import linprog
from scipy.sparse import csr_matrix

def build_conv_matrix_sparse(f):
    """Build sparse matrix A where (A@g)[k] = convolve(f,g)[k]."""
    n = len(f)
    # For each k (0 to 2n-2), the valid j range is max(0, k-n+1) to min(k, n-1)
    # A[k, j] = f[k-j] for valid j
    rows = np.empty(n * n, dtype=np.int32)
    cols = np.empty(n * n, dtype=np.int32)
    vals = np.empty(n * n, dtype=np.float64)
    idx = 0
    for k in range(2 * n - 1):
        j_min = max(0, k - (n - 1))
        j_max = min(k, n - 1)
        for j in range(j_min, j_max + 1):
            rows[idx] = k
            cols[idx] = j
            vals[idx] = f[k - j]
            idx += 1
    rows = rows[:idx]
    cols = cols[:idx]
    vals = vals[:idx]
    return csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))

def lp_direction(f, M):
    """Solve LP: max sum(g) s.t. A@g <= M, g >= 0"""
    n = len(f)
    A = build_conv_matrix_sparse(f)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    try:
        result = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
        if result.success:
            return result.x
    except:
        pass
    return None

def subgradient_step(f, lr):

    n = len(f)

    if S <= 0:
        return f
    c = np.convolve(f, f)
    M = np.max(c)
    k_star = int(np.argmax(c))
    
    # Subgradient of M w.r.t. f
    grad_M = np.zeros(n)
    j_min = max(0, k_star - (n - 1))
    j_max = min(k_star, n - 1)

    if S_new > 0:

    else:

import numpy as np
import time
from scipy.optimize import linprog
from scipy.sparse import csr_matrix


def build_conv_matrix_sparse(f):
    """Build sparse matrix A where (A@g)[k] = (convolve(f,g))[k]."""
    n = len(f)
    total = 0
    for k in range(2 * n - 1):
        total += min(k, n - 1) - max(0, k - n + 1) + 1
    rows = np.zeros(total, dtype=np.int32)
    cols = np.zeros(total, dtype=np.int32)
    vals = np.zeros(total, dtype=np.float64)
    idx = 0
    for k in range(2 * n - 1):
        j_lo = max(0, k - n + 1)
        j_hi = min(k, n - 1)
        for j in range(j_lo, j_hi + 1):
            rows[idx] = k
            cols[idx] = j
            vals[idx] = f[k - j]
            idx += 1
    return csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))


def solve_lp(f, M):
    """Solve LP: max sum(g) s.t. conv(f,g) <= M, g >= 0."""
    n = len(f)
    A = build_conv_matrix_sparse(f)
    c = -np.ones(n)
    b = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    try:
        res = linprog(c, A_ub=A, b_ub=b, bounds=bounds, method='highs')
        if res.success:
            return res.x
    except Exception:
        pass
    return None


def project_onto_simplex(v, s):
    """Project v onto {x >= 0, sum(x) = s}."""

    rho = rho[-1]

import numpy as np

def build_conv_matrix_sparse(f):
    n = len(f)
    total = 0
    for k in range(2 * n - 1):
        total += min(k, n - 1) - max(0, k - n + 1) + 1
    rows = np.zeros(total, dtype=np.int32)
    cols = np.zeros(total, dtype=np.int32)
    vals = np.zeros(total, dtype=np.float64)
    idx = 0
    for k in range(2 * n - 1):
        j_lo = max(0, k - n + 1)
        j_hi = min(k, n - 1)
        for j in range(j_lo, j_hi + 1):
            rows[idx] = k
            cols[idx] = j
            vals[idx] = f[k - j]
            idx += 1
    return csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))


def solve_lp_direction(f, M):
    n = len(f)
    A = build_conv_matrix_sparse(f)
    c = -np.ones(n)
    b = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    try:
        res = linprog(c, A_ub=A, b_ub=b, bounds=bounds, method='highs')
        if res.success:
            return res.x
    except Exception:
        pass
    return None


def project_simplex(v, s):
    n = len(v)
    if s <= 0:
        return np.zeros(n)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - s)
    rho = np.where(cond)[0]
    if len(rho) == 0:
        return np.full(n, s / n)
    rho = rho[-1]
    theta = (cssv[rho] - s) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def compute_subgrad(f, k_star, n):
    grad = np.zeros(n)
    j_lo = max(0, k_star - n + 1)
    j_hi = min(k_star, n - 1)
    if j_lo <= j_hi:

        grad[j_lo:j_hi + 1] = 2.0 * f[k_star - j_hi:k_star - j_lo + 1][::-1]
    return grad

        if rs < best_score:
            best_score = rs
            best_sequence = rf.copy()

        remaining_configs = len(configs) - i - 1
        if i % 5 == 0 or rs < best_score + 0.01:
            print(f"  [{name}] n={n}: score={rs:.6f} (best={best_score:.6f}, "
                  f"{remaining_configs} configs left, time_left={p1_end - time.monotonic():.0f}s)")

    # Phase 2: LP refinement (40% of time)
    if best_sequence is not None and time.monotonic() < deadline - 30:
        p2_budget = 0.40 * (deadline - time.monotonic())
        p2_end = time.monotonic() + p2_budget
        print(f"\n=== Phase 2: LP refinement (n={len(best_sequence)}, "
              f"score={best_score:.6f}, budget={p2_budget:.0f}s) ===")

        rf, rs = lp_refine(best_sequence, p2_end, max_iters=400, verbose_prefix="[LP]")
        if rs < best_score:
            best_score = rs
            best_sequence = rf.copy()
            print(f"  LP result: {best_score:.6f}")
        else:
            print(f"  LP: no improvement (best remains {best_score:.6f})")

    # Phase 3: Final momentum polish + second LP pass (10% of time)
    if best_sequence is not None and time.monotonic() < deadline - 15:
        p3_end = deadline - 5
        print(f"\n=== Phase 3: Final polish (budget={p3_end - time.monotonic():.0f}s) ===")

        # Short momentum pass
        m1_end = time.monotonic() + 0.4 * (p3_end - time.monotonic())
        rf, rs = momentum_subgrad(best_sequence, m1_end,
                                   lr_scale=0.04 * np.sqrt(len(best_sequence)),
                                   momentum=0.3)
        if rs < best_score:
            best_score = rs
            best_sequence = rf.copy()
            print(f"  Momentum polish: {best_score:.6f}")

        # Short LP pass
        if time.monotonic() < p3_end - 5:
            rf, rs = lp_refine(best_sequence, p3_end, max_iters=100, verbose_prefix="[LP2]")
            if rs < best_score:
                best_score = rs
                best_sequence = rf.copy()
                print(f"  Final LP: {best_score:.6f}")

    # Sanity check and finalize
    if best_sequence is None:
        best_sequence = np.ones(128, dtype=np.float64)
        best_score = evaluate_sequence(best_sequence.tolist())

    best_sequence = np.clip(best_sequence, 0.0, 1000.0)
    final_score = evaluate_sequence(best_sequence.tolist())
    elapsed = time.monotonic() - t0
    print(f"\nFINAL: n={len(best_sequence)}, score={final_score:.6f}, time={elapsed:.1f}s")

    return [float(x) for x in best_sequence]
