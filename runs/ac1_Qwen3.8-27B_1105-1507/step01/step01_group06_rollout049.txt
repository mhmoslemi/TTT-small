```python
def soft_obj_and_grad(x, t):
    """Objective and gradient for L-BFGS-B. x is the unnormalized sequence."""
    n = len(x)
    a = np.maximum(x, 1e-15)
    S = a.sum()
    a = a / S
    b = fast_conv(a)
    c = b / t
    mc = c.max()
    exp_c = np.exp(c - mc)
    soft = (mc + np.log(exp_c.sum())) / t
    w = exp_c / exp_c.sum()
    
    # Gradient of soft w.r.t. a (normalized):
    # grad_a_j = (2/t) * sum_i w_{j+i} * a_i
    size = 2*n - 1
    s2 = 1
    while s2 < size:
        s2 <<= 1
    A = np.fft.rfft(a, s2)
    W = np.fft.rfft(w, s2)
    grad_a = (2.0/t) * np.fft.irfft(np.conj(A) * W, s2)[:n]
    
    # Now we need gradient w.r.t. x (unnormalized):
    # a = x/S (where x = max(original, eps))
    # d(a_j)/d(x_i) = (δ_{ij} * S - x_i) / S² = (δ_{ij} - a_i) / S
    # grad_x_i = sum_j grad_a_j * (δ_{ij} - a_i) / S
    # = (grad_a_i - a_i * sum_j grad_a_j) / S
    
    grad_x = (grad_a - a * grad_a.sum()) / S
    return soft, grad_x
```

```python
"""
Minimizes J(a) = 2*n*max(convolve(a,a))/(sum(a))^2 using:
1. Log-sum-exp smooth gradient descent with temperature annealing
2. LP-based best-response polishing for smaller n
3. Multi-scale approach: optimize at multiple n values with resampling
4. Multiple initializations for robustness
"""

import numpy as np
from scipy.optimize import linprog, minimize
import time


def fast_conv(a):
    """Full self-convolution via FFT."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]


def soft_conv_obj_grad(x, t):
    """
    Compute the log-sum-exp smooth approximation to max(conv(x,x))
    and its gradient w.r.t. x.
    
    x: unnormalized non-negative sequence
    t: temperature parameter
    Returns (objective_value, gradient)
    """
    n = len(x)
    a = np.maximum(x, 1e-15)
    S = a.sum()
    if S < 1e-12:
        return 1e10, np.zeros(n)
    a = a / S
    b = fast_conv(a)
    c = b / t
    mc = c.max()
    exp_c = np.exp(c - mc)
    soft = (mc + np.log(exp_c.sum())) / t
    w = exp_c / exp_c.sum()
    
    # Gradient of soft w.r.t. a (normalized sequence)
    # grad_a_j = (2/t) * sum_i w_{j+i} * a_i  (cross-correlation)
    size = 2 * n - 1
    s2 = 1
    while s2 < size:
        s2 <<= 1
    A = np.fft.rfft(a, s2)
    W = np.fft.rfft(w, s2)
    grad_a = (2.0 / t) * np.fft.irfft(np.conj(A) * W, s2)[:n]
    
    # Gradient w.r.t. x (unnormalized): chain rule through a = x/S
    grad_x = (grad_a - a * grad_a.sum()) / S
    return soft, grad_x


def lp_best_response(a, M):
    """Solve LP: max sum(g) s.t. (a*g)_k <= M, g >= 0."""
    n = len(a)
    c = -np.ones(n)
    # Build constraint matrix
    A_ub = np.zeros((2 * n - 1, n))
    for k in range(2 * n - 1):
        jl = max(0, k - n + 1)
        jh = min(n - 1, k)
        if jl <= jh:
            A_ub[k, jl:jh + 1] = a[k - jh:k - jl + 1][::-1]
    r = linprog(c, A_ub=A_ub, b_ub=np.full(2 * n - 1, M),
                bounds=[(0, None)] * n, method='highs',
                options={'presolve': True})
    if r.success:
        g = r.x
        if g.sum() > 1e-10:
            return g
    return None


def resample_sequence(a, new_n):
    """Resample a sequence to a new length using linear interpolation."""
    old_n = len(a)
    if old_n == new_n:
        return a.copy()
    t_old = np.linspace(0, 1, old_n)
    t_new = np.linspace(0, 1, new_n)
    new_a = np.interp(t_new, t_old, a)
```
