def project_simplex(v):
    """Project v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n+1) > (cssv - 1.0))[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho = rho[-1]

    c_norm = (c - c_max) / tau
    p = np.exp(c_norm)
    p /= np.sum(p)
    M = float(np.dot(p, c))
    r = np.correlate(p, a, mode='valid')
    grad_b = 2.0 * a * (r - M)
    return M, grad_b

def optimize_b(a_init, time_budget=30.0, seed=0, base_lr=1.0, verbose=False):
    n = len(a_init)
    a_init = np.maximum(a_init, 1e-30)
    tau_init = 3.0 / n
    tau_end = 1.0 / (200.0 * n)
    b = np.log(a_init).copy()
    rng = np.random.default_rng(seed)
    
    m = np.zeros(n)
    v = np.zeros(n)
    beta1, beta2, eps = 0.85, 0.999, 1e-8
    
    best_obj = np.inf
    best_a = a_init.copy()
    stall = 0
    patience = 200
    
    t0 = time.monotonic()
    it = 0
    while time.monotonic() - t0 < time_budget:
        frac = min(1.0, it / 5000.0)
        tau = tau_init * (

tau_end / tau_init) ** frac
        
        a = a_from_b(b)
        M, grad_b = smooth_max_grad(a, tau)
        
        m = beta1 * m + (1 - beta1) * grad_b
        v = beta2 * v + (1 - beta2) * (grad_b ** 2)
        m_hat = m / (1 - beta1 ** (it + 1))
        v_hat = v / (1 - beta2 ** (it + 1))
        lr = base_lr / (1.0 + 0.0005 * it)
        b = b - lr * m_hat / (np.sqrt(v_hat) + eps)
        
        a_curr = a_from_b(b)
        obj = float(2.0 * n * np.max(np.convolve(a_curr, a_curr)))
        if obj < best_obj:
            best_obj = obj
            best_a = a_curr.copy()
            stall = 0
        else:
            stall += 1
        
        if stall > patience:
            noise_scale = 0.3 * (1.0 - frac * 0.5)
            b = b + rng.normal(0, noise_scale, n)
            stall = 0
        
        if verbose and it % 500 == 0:
            print(f"        it={it}, tau={tau:.2e}, obj={best_obj:.6f}, elapsed={time.monotonic()-t0:.1f}s")
        
        it += 1
    
    return best_a, best_obj

def random_refine(a, n_iter=5000, seed=0):
    """Random perturbation local search."""
    n = len(a)
    best_a = a.copy()
    best_obj = compute_obj(best_a)
    rng = np.random.default_rng(seed)
    
    for it in range(n_iter):
        # Perturb a random block
        width = max(1, n // 50)
        start = int(rng.integers(0, n - width + 1))
        candidate = best_a.copy()
        factor = np.exp(rng.normal(0, 0.1, width))
        candidate[start:start+width] *= factor
        candidate = np.maximum(candidate, 0)
        s = np.sum(candidate)
        if s < 1e-10:
            continue
        obj = compute_obj(candidate)
