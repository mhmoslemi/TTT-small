def subgradient_step(a, conv, top_k=5, lr=0.01):
    """
    Perform a subgradient step using the top-k active lags.
    Returns (new_a, active_lags).
    """
    n = len(a)
    S = np.sum(a)
    
    # Find top k lags
    top_lags = np.argsort(conv)[-top_k:][::-1]
    max_val = conv[top_lags[0]]
    if max_val <= 0:
        return a.copy(), top_lags
    
    # Compute weighted gradient (weight by how close each lag is to max)
    weights = np.zeros(len(top_lags))
    for i, k in enumerate(top_lags):
        weights[i] = 1.0 if abs(conv[k] - max_val) < 1e-10 * max_val else (conv[k] / max_val) ** 10
    
    # Normalize weights
    w_sum = np.sum(weights)
    if w_sum > 0:
        weights = weights / w_sum
    
    # Compute combined gradient
    grad = np.zeros(n)
    for i, k in enumerate(top_lags):
        for j in range(n):
            if 0 <= k - j < n:
                grad[j] += weights[i] * 2.0 * a[k - j]
    
    # Step
    new_a = a - lr * grad
    new_a = np.clip(new_a, 0.0, None)
    
    # Preserve sum
    new_sum = np.sum(new_a)
    if new_sum > 1e-15:
        new_a = new_a * (S / new_sum)
    
    return new_a, top_lags

    n = len(a)
    top_indices = np.argsort(conv)[-top_k:][::-1]
    max_val = conv[top_indices[0]]
    
    # Power weights: (q_k / q_max)^power
    weights = (conv[top_indices] / max_val) ** power
    weights = weights / weights.sum()
    
    grad = np.zeros(n)
    j_vals = np.arange(n)
    
    for i, k in enumerate(top_indices):
        if weights[i] < 1e-12:
            continue
        valid = (k - j_vals >= 0) & (k - j_vals < n)
        grad[valid] += weights[i] * 2.0 * a[k - j_vals[valid]]
    
    return grad

This should work well. The power parameter controls how selective we are: power=1 gives equal weight to top-k, power=10 gives near-exclusive weight to the max.

Now let me think about the overall algorithm flow:

1. Initialize from parent or parameterized shapes
2. Main loop:
   a. Compute conv = np.convolve(a, a)
   b. Compute gradient using top-k lags
   c. Take a step: a -= lr * grad
   d. Project: a = max(a, 0), then normalize (a /= sum(a))
   e. If score improved, accept; else reduce lr
   f. Periodically (every K iterations), try a spectral step or random perturbation

The key question is what learning rate to use. Since the problem is scale-invariant and we normalize after each step, the effective step size depends on the curvature of the objective.

Let me think about this differently. After normalization (sum = 1), the feasible set is the (n-1)-dimensional simplex. The objective is max_k q_k(a) where q_k is a quadratic form. The "curvature" of q_k at a point a is related to the eigenvalues of M_k (the matrix defining the quadratic form).

For a practical approach, I'll use a backtracking line search:
1. Try a step with lr
2. If the new score is better, accept (and optionally increase lr)
3. If worse, reduce lr and retry

This is robust and doesn't require tuning.

Let me also consider: should I normalize to sum=1 at each step, or let the sum vary? Since the objective is scale-invariant, it doesn't matter for the final answer. But for numerical stability, normalizing at each step is good.

One more thought: the problem has a lot of symmetry. If I flip the sequence (a[i] -> a[n-1-i]), the convolution is the same (just reversed). So I could restrict to, say, a[0] <= a[n-1] to break the symmetry. But this might not help much in practice.

def fast_conv(a):
    """Compute linear convolution efficiently."""
    return np.convolve(a, a)


def compute_score(a):
    """Compute the evaluation score."""
    n = len(a)
    conv = np.convolve(a, a)
    max_b = np.max(conv)
    sum_a = np.sum(a)
    if sum_a < 0.01:
        return np.inf, conv, max_b
    score = 2.0 * n * max_b / (sum_a ** 2)
    return score, conv, max_b


def compute_gradient(a, conv, top_k=5, power=10):
    """
    Compute gradient of max_k q_k(a) using power-weighted top-k lags.
    
    q_k(a) = sum_i a[i]*a[k-i] (where valid)
    d q_k / d a[j] = 2*a[k-j] if 0 <= k-j < n, else 0
    """
    n = len(a)
    num_lags = len(conv)
    
    if num_lags > top_k:
        top_idx = np.argsort(conv)[-top_k:][::-1]
    else:
        top_idx = np.arange(num_lags)
    
    max_val = conv[top_idx[0]]
    if max_val <= 0:
        return np.zeros(n)
    
    weights = (conv[top_idx] / max_val) ** power
    w_sum = np.sum(weights)
    if w_sum < 1e-30:
        return np.zeros(n)
    weights = weights / w_sum
    
    grad = np.zeros(n)
    for i, k in enumerate(top_idx):
        w = weights[i]
        if w < 1e-12:
            continue
        # grad[j] += w * 2 * a[k-j] for valid j
        j_min = max(0, k - n + 1)
        j_max = min(k, n - 1)
        if j_min <= j_max:
            grad[j_min:j_max+1] += w * 2.0 * a[k - j_max : k - j_min + 1][::-1]
    
    return grad


def optimize_subgradient(a, deadline, lr_init=0.01, top_k=3, power=10, 
                          verbose_prefix="", check_interval=50):
    """
    Main subgradient optimization loop with backtracking.
    """
    a = a.copy().astype(np.float64)
    S = np.sum(a)
    if S < 1e-15:
        return a, float('inf')
    a = a / S  # Normalize to sum=1
    
    conv = np.convolve(a, a)
    score = 2.0 * len(a) * np.max(conv)  # Since sum=1, score = 2n * max(conv)
    best_a = a.copy()
    best_score = score
    lr = lr_init
    no_improve = 0
    it = 0
    momentum = np.zeros_like(a)
    beta = 0.9
    
    while time.monotonic() < deadline:
        # Compute gradient
        grad = compute_gradient(a, conv, top_k=top_k, power=power)
        
        # Momentum
        momentum = beta * momentum + (1 - beta) * grad
        
        # Try a step
        new_a = a - lr * momentum
        new_a = np.clip(new_a, 0.0, None)
        new_sum = np.sum(new_a)
        if new_sum < 1e-15:
            lr *= 0.5
            continue
        new_a = new_a / new_sum  # Re-normalize
        
        # Evaluate
        new_conv = np.convolve(new_a, new_a)
        new_score = 2.0 * len(a) * np.max(new_conv)
        
        it += 1
        
        if new_score < best_score:
            a = new_a
            conv = new_conv
            best_a = a.copy()
            best_score = new_score
            no_improve = 0
            lr = min(lr * 1.05, lr_init)  # Slightly increase, capped
        else:
            no_improve += 1
            lr *= 0.7
            if no_improve > 300 or lr < 1e-10:
                # Perturb and restart
                a = best_a + np.random.default_rng(it).normal(0, 0.01, len(a))
                a = np.clip(a, 0.0, None)
                a = a / np.sum(a)
                conv = np.convolve(a, a)
                lr = lr_init * 0.3
                no_improve = 0
                momentum = np.zeros_like(a)

def spectral_step(a, strength=0.1):
    """
    Attempt to reduce autocorrelation peaks using spectral domain manipulation.
    """
    n = len(a)
    N = 1
    while N < 2 * n - 1:
        N *= 2
    
    A = np.fft.rfft(a, N)
    power = np.abs(A) ** 2
    
    # Compute current autocorrelation
    autocorr = np.fft.irfft(power, N)
    lin_part = autocorr[:2*n-1]
    
    # Find the peak
    peak_val = np.max(lin_part)
    mean_val = np.sum(lin_part) / (2*n-1)
    
    if peak_val < 1.01 * mean_val:
        return a.copy()
    
    # Strategy: reduce the power at frequencies that contribute most to the peak
    # The contribution of frequency m to autocorrelation lag k is |A_m|^2 * exp(i*2*pi*m*k/N)
    # To reduce the peak at lag k*, we want to reduce |A_m|^2 for m where
    # the real part of exp(i*2*pi*m*k*/N) is large (i.e., where they constructively interfere at k*)
    
    peak_lag = np.argmax(lin_part)
    # Reduce power at frequencies that align with the peak
    m = np.arange(len(power))
    phase = 2 * np.pi * m * peak_lag / N
    alignment = np.cos(phase)  # Positive when constructive
    
    # Damp constructive frequencies, boost destructive ones
    damping = 1.0 - strength * alignment
    damping = np.clip(damping, 0.5, 1.5)
    
    A_new = A * np.sqrt(damping)
    a_new = np.fft.irfft(A_new, N)[:n]
    a_new = np.clip(a_new, 0.0, None)
    a_new_sum = np.sum(a_new)
    if a_new_sum > 1e-15:
        a_new = a_new * (np.sum(a) / a_new_sum)
    
    return a_new


def parameterized_sequence(n, shape_type, **params):
    """Generate a parameterized sequence."""
    idx = np.arange(n, dtype=np.float64)
    
    if shape_type == 'bilateral':
        alpha1 = params.get('alpha1', 3.0)
        alpha2 = params.get('alpha2', 5.0)
        beta = params.get('beta', 0.3)
        a = np.exp(-alpha1 * np.maximum(beta * (n-1) - idx, 0) ** 2 / (n*0.1) - 
                   alpha2 * np.maximum(idx - beta * (n-1), 0) ** 2 / (n*0.1))
    elif shape_type == 'sigmoid':
        gamma = params.get('gamma', 3.0)
        delta = params.get('delta', 0.3)
        a = 1.0 / (1.0 + np.exp(gamma * (idx / (n-1) - delta)))
    elif shape_type == 'power':
        p = params.get('p', 0.5)
        a = (idx / (n-1) + 0.01) ** p
    elif shape_type == 'skew_gaussian':
        mu = params.get('mu', 0.3)
        sigma = params.get('sigma', 0.15)
        skew = params.get('skew', 0.3)
        x = (idx / (n-1) - mu) / sigma
        a = np.exp(-0.5 * x * (x + skew))  # Skewed Gaussian
    elif shape_type == 'two_bump':
        mu1 = params.get('mu1', 0.2)
        mu2 = params.get('mu2', 0.8)
        sigma = params.get('sigma', 0.1)
        a = np.exp(-0.5 * ((idx / (n-1) - mu1) / sigma) ** 2) + \
            np.exp(-0.5 * ((idx / (n-1) - mu2) / sigma) ** 2)
    else:
        return None
    
    a = np.clip(a, 0.0, None)
    if np.sum(a) < 1e-10:
        return None
    return a / np.sum(a)


def interpolate_to_length(a, new_n):
    """Interpolate sequence to new length."""
    n = len(a)
    if n == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.clip(a_new, 0.0, None)
    s = np.sum(a_new)
    if s > 1e-15:
        a_new = a_new / s
    return a_new

"""
Subgradient optimization with backtracking for minimizing 2n*max(conv(a,a))/sum(a)^2.
Uses power-weighted multi-lag subgradients, adaptive step sizes, spectral assistance,
multi-scale search (n=512 parent + interpolation to 1024/2048), and fine-tuning.
"""
import numpy as np
import time


def compute_score_normalized(a):
    """Compute score assuming sum(a)=1. Returns (score, conv, max_b)."""
    n = len(a)
    conv = np.convolve(a, a)
    max_b = float(np.max(conv))
    score = 2.0 * n * max_b
    return score, conv, max_b


def gradient_for_lag(a, k):
    """Compute gradient of q_k(a) = sum_i a[i]*a[k-i] w.r.t. a."""
    n = len(a)
    grad = np.zeros(n)
    j_min = max(0, k - n + 1)
    j_max = min(k, n - 1)
    if j_min <= j_max:
        grad[j_min:j_max + 1] = 2.0 * a[k - j_max:k - j_min + 1][::-1]
    return grad


def multi_lag_gradient(a, conv, top_k=3, power=20):
    """Compute power-weighted combination of top-k lag gradients."""
    n = len(a)
    num = len(conv)
    if num > top_k:
        idx = np.argpartition(conv, -top_k)[-top_k:]
        idx = idx[np.argsort(conv[idx])[::-1]]
    else:
        idx = np.arange(num)
    
    max_val = conv[idx[0]]
    if max_val <= 0:
        return np.zeros(n)
    
    weights = (conv[idx] / max_val) ** power
    ws = weights.sum()
    if ws < 1e-30:
        return np.zeros(n)
    weights = weights / ws
    
    grad = np.zeros(n)
    for i, k in enumerate(idx):
        if weights[i] < 1e-12:
            continue
        n_add = gradient_for_lag(a, int(k))
        grad += weights[i] * n_add
    return grad


def optimize_sg(a, deadline, lr_init=0.01, top_k=3, power=20,
                verbose_prefix="", report_every=200):
    """Subgradient optimization with backtracking and restarts."""
    a = a.copy().astype(np.float64)
    S = float(np.sum(a))
    if S < 1e-15:
        return a, float('inf')
    a = a / S
    n = len(a)

    score, conv, max_b = compute_score_normalized(a)
    best_a = a.copy()
    best_score = score
    lr = lr_init
    no_improve = 0
    it = 0
    rng = np.random.default_rng(42 + n)

    while time.monotonic() < deadline:
        grad = multi_lag_gradient(a, conv, top_k=top_k, power=power)
        grad_norm = np.linalg.norm(grad)
        if grad_norm < 1e-15:
            break

        # Normalize gradient to unit norm for consistent step sizing
        grad_dir = grad / grad_norm

        # Try step
        new_a = a - lr * grad_dir
        new_a = np.clip(new_a, 0.0, None)
        ns = float(np.sum(new_a))
        if ns < 1e-15:
            lr *= 0.5
            continue
        new_a = new_a / ns

        new_score, new_conv, new_max_b = compute_score_normalized(new_a)
        it += 1

        if new_score < best_score:
            a = new_a
            conv = new_conv
            max_b = new_max_b
            best_a = a.copy()
            best_score = new_score
            no_improve = 0
            lr = min(lr * 1.03, lr_init)
        else:
            no_improve += 1
            lr *= 0.75
            if no_improve > 500 or lr < 1e-12:
                # Perturb and restart
                a = best_a + rng.normal(0, 0.008, n)
                a = np.clip(a, 0.0, None)
                ns2 = float(np.sum(a))
                if ns2 > 1e-15:
                    a = a / ns2
                    conv = np.convolve(a, a)
                lr = lr_init * 0.4
                no_improve = 0

        if verbose_prefix and it % report_every == 0:
            print(f"    {verbose_prefix}it={it}, score={best_score:.7f}, lr={lr:.2e}")

    return best_a, best_score


def spectral_modify(a, strength=0.08):
    """Spectral domain peak reduction step."""
    n = len(a)
    N = 1
    while N < 2 * n - 1:
        N *= 2

    A = np.fft.rfft(a, N)
    power = np.abs(A) ** 2
    autocorr = np.fft.irfft(power, N)
    lin = autocorr[:2 * n - 1]

    peak_val = float(np.max(lin))
    mean_val = float(np.sum(lin)) / (2 * n - 1)

    if peak_val < 1.005 * mean_val:
        return a.copy()

    peak_lag = int(np.argmax(lin))
    m = np.arange(len(power), dtype=np.float64)
    phase = 2.0 * np.pi * m * peak_lag / N
    alignment = np.cos(phase)

    damping = 1.0 - strength * alignment
    damping = np.clip(damping, 0.6, 1.4)

    A_new = A * np.sqrt(damping)
    a_new = np.fft.irfft(A_new, N)[:n]
    a_new = np.clip(a_new, 0.0, None)
    s = float(np.sum(a_new))
    if s > 1e-15:
        a_new = a_new * (float(np.sum(a)) / s)
    return a_new


def param_seq(n, stype, **kw):
    """Generate parameterized sequence."""
    idx = np.arange(n, dtype=np.float64)
    if stype == 'bili':
        a1 = kw.get('a1', 3.0)
        a2 = kw.get('a2', 6.0)
        b = kw.get('b', 0.3)
        center = b * (n - 1)
        left = np.maximum(center - idx, 0.0) / (0.1 * n)
        right = np.maximum(idx - center, 0.0) / (0.1 * n)
        a = np.exp(-a1 * left**2 - a2 * right**2)
    elif stype == 'sig':
        g = kw.get('g', 3.0)
        d = kw.get('d', 0.3)
        a = 1.0 / (1.0 + np.exp(g * (idx / max(n-1, 1) - d)))
    elif stype == 'pow':
        p = kw.get('p', 0.3)
        a = (idx / max(n-1, 1) + 0.01) ** p
    elif stype == 'skg':
        mu = kw.get('mu', 0.3)
        sigma = kw.get('sigma', 0.12)
        skew = kw.get('skew', 0.5)
        x = (idx / max(n-1, 1) - mu) / sigma
        a = np.exp(-0.5 * x * (x + skew))
    else:
        return None
    a = np.clip(a, 0.0, None)
    s = float(np.sum(a))
    if s < 1e-10:
        return None
    return a / s


def interp_seq(a, new_n):
    """Interpolate to new length."""
    n = len(a)
    if n == new_n:
        return a.copy()
    x_old = np.linspace(0.0, 1.0, n)
    x_new = np.linspace(0.0, 1.0, new_n)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.clip(a_new, 0.0, None)
    s = float(np.sum(a_new))
    if s > 1e-15:
        a_new = a_new / s
    return a_new
