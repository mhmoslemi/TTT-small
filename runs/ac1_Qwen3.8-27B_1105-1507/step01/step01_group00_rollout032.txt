**Approach 4: Iterative reweighted / water-filling**

The key insight is: we want to "spread out" the convolution peaks. If the max is at lag k*, we want to reduce the contribution to that lag.

The contribution to conv[k*] from f[j] is f[j]*f[k*-j] (if valid). So to reduce conv[k*], we should reduce f[j] where f[k*-j] is large.

- Reduce f[j] proportionally to its pressure
- Renormalize

This is similar to the LP approach but faster.

"""
Multi-strategy optimization for minimizing 2n*max(conv(f,f))/(sum f)^2.
Uses subgradient descent with momentum, LP direction finding,
and coordinate descent on multiple sequence lengths and initializations.
"""
import numpy as np
import time
from scipy.optimize import linprog
from scipy.sparse import csr_matrix


def compute_objective(f):
    """Compute 2n*max(conv(f,f))/(sum f)^2 efficiently."""
    n = len(f)
    conv = np.convolve(f, f)
    S = f.sum()
    if S < 0.01:
        return np.inf
    return 2.0 * n * conv.max() / (S * S)


def subgradient_of_max_conv(f, max_lag):
    """
    Compute subgradient of max(conv(f,f)) w.r.t. f.
    At the maximizing lag k*, the gradient is:
    d(conv[k*])/df[j] = 2*f[k*-j] if 0 <= k*-j < n, else 0.
    """
    n = len(f)
    k = max_lag
    grad = np.zeros(n)
    # valid j: 0 <= k-j < n => k-n+1 <= j <= k, intersect [0, n-1]
    j_min = max(0, k - n + 1)
    j_max = min(k, n - 1)
    grad[j_min:j_max+1] = 2.0 * f[k - j_max : k - j_min : -1]
    return grad


def subgradient_descent(f, deadline, lr_init=0.01, momentum=0.9, name=""):
    """
    Subgradient descent with momentum to minimize max(conv(f,f))
    (equivalently the full objective, since it's homogeneous).
    """
    f = f.astype(np.float64).copy()
    n = len(f)
    best_f = f.copy()
    best_obj = compute_objective(f)
    lr = lr_init
    v = np.zeros(n)
    no_improve = 0
    it = 0
    
    while time.monotonic() < deadline:
        conv = np.convolve(f, f)
        max_lag = int(np.argmax(conv))
        M = conv[max_lag]
        S = f.sum()
        
        if M < 1e-15:
            break
        
        # Subgradient
        grad = subgradient_of_max_conv(f, max_lag)
        
        # Normalize gradient to have same scale as f
        # The objective is 2n*M/S^2, gradient of this w.r.t f is:
        # 2n/S^2 * dM/df - 2n*M*2*S*dS/df / S^4
        # = 2n/S^2 * grad - 4n*M/S^3 * ones
        # But since we normalize after each step, the second term doesn't matter
        # (it just changes the scale, which is fixed by normalization)
        
        # Momentum update
        v = momentum * v - lr * grad / S  # normalize by S to make step scale-invariant
        f_new = f + v
        f_new = np.clip(f_new, 0.0, None)
        
        # Renormalize to keep sum = S (or any constant)
        new_S = f_new.sum()
        if new_S < 1e-10:
            f_new = f.copy()
            new_S = S
        else:
            f_new = f_new * (S / new_S)
        
        obj = compute_objective(f_new)
        it += 1
        
        if obj < best_obj:
            best_obj = obj
            best_f = f_new.copy()
            f = f_new
            no_improve = 0
            if it % 500 == 0:
                print(f"  {name} subgrad iter {it}: obj={best_obj:.6f}")
        else:
            no_improve += 1
            if no_improve % 10 == 0:
                lr *= 0.85
            if no_improve > 200:
                # Try restarting with smaller lr
                lr = max(lr_init * 0.1, lr * 0.5)
                v = np.zeros(n)
                no_improve = 0
                if lr < lr_init * 0.01:
                    break
    
    return best_f, best_obj


def lp_direction(f, M, timeout=10.0):
    """
    Solve: max sum(g) s.t. conv(f,g) <= M, g >= 0.
    Returns the solution g or None.
    """
    n = len(f)
    # Build constraint matrix: (2n-1) x n
    # A[k, j] = f[k-j] if 0 <= k-j < n, else 0
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        j_min = max(0, k - n + 1)
        j_max = min(k, n - 1)
        for j in range(j_min, j_max + 1):
            rows.append(k)
            cols.append(j)
            vals.append(f[k - j])
    
    A = csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    
    try:
        result = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                         options={'time_limit': timeout})
        if result.success:
            return result.x
    except Exception:
        pass
    return None


def lp_iterate(f, deadline, name="", max_lp_iters=50):
    """LP-based iteration."""
    f = f.astype(np.float64).copy()
    best_f = f.copy()
    best_obj = compute_objective(f)
    it = 0
    step_size = 0.1
    
    while time.monotonic() < deadline and it < max_lp_iters:
        conv = np.convolve(f, f)
        M = float(conv.max())
        S = float(f.sum())
        
        if M < 1e-15:
            break
        
        t_start = time.monotonic()
        g = lp_direction(f, M, timeout=5.0)
        lp_time = time.monotonic() - t_start
        
        if g is None:
            break
        
        sum_g = g.sum()
        if sum_g < 1e-15:
            break
        
        # Rescale g to have same sum as f
        g = g * (S / sum_g)
        
        # Try different step sizes
        best_new = None
        best_new_obj = np.inf
        for alpha in [step_size, step_size * 0.5, step_size * 0.25]:
            f_new = (1 - alpha) * f + alpha * g
            f_new = np.clip(f_new, 0.0, None)
            obj = compute_objective(f_new)
            if obj < best_new_obj:
                best_new_obj = obj
                best_new = f_new
        
        if best_new is not None and best_new_obj < best_obj:
            best_obj = best_new_obj
            best_f = best_new.copy()
            f = best_new
            it += 1
            if it % 5 == 0:
                print(f"  {name} LP iter {it}: obj={best_obj:.6f} (lp_time={lp_time:.1f}s)")
        else:
            step_size *= 0.7
            if step_size < 0.01 or it >= max_lp_iters:
                break
    
    return best_f, best_obj


def coordinate_descent(f, deadline, name="", passes=10):
    """
    Coordinate descent: for each j, try to optimize f[j].
    Since the objective as a function of f[j] is max of ratios of linear/quadratic
    to quadratic, we use a simple approach: try several candidate values.
    """
    f = f.astype(np.float64).copy()
    n = len(f)
    best_f = f.copy()
    best_obj = compute_objective(f)
    S = f.sum()
    
    for p in range(passes):
        if time.monotonic() > deadline:
            break
        improved = False
        for j in range(n):
            if time.monotonic() > deadline:
                break
            old_val = f[j]
            # Try: old_val * factor for various factors, and 0
            candidates = [old_val * 0.5, old_val * 0.7, old_val * 1.3, old_val * 1.5, 0.0]
            # Also try values based on neighbors
            if j > 0:
                candidates.append(f[j-1])
            if j < n-1:
                candidates.append(f[j+1])
            
            best_cand = old_val
            best_cand_obj = best_obj
            
            for cand in candidates:
                f[j] = cand
                new_S = f.sum()
                if new_S < 0.01:
                    f[j] = old_val
                    continue
                f_norm = f * (S / new_S)  # renormalize
                obj = compute_objective(f_norm)
                if obj < best_cand_obj:
                    best_cand_obj = obj
                    best_cand = cand
            
            if best_cand != old_val:
                f[j] = best_cand
                new_S = f.sum()
                if new_S > 0:
                    f = f * (S / new_S)
                improved = True
                if compute_objective(f) < best_obj:
                    best_obj = compute_objective(f)
                    best_f = f.copy()
            else:
                f[j] = old_val
        
        if not improved:
            break
    
    return best_f, best_obj

"""
Multi-strategy optimization for minimizing 2n*max(conv(f,f))/(sum f)^2.
Key insight: asymmetry in f reduces the central convolution peak below sum(f)^2/n.
Uses subgradient descent with Nesterov momentum, LP direction finding,
coordinate descent, and simulated annealing across multiple lengths.
"""
import numpy as np
import time
from scipy.optimize import linprog
from scipy.sparse import csr_matrix


def compute_objective(f):
    """Compute 2n*max(conv(f,f))/(sum f)^2."""
    n = len(f)
    conv = np.convolve(f, f)
    S = f.sum()
    if S < 0.01:
        return np.inf
    return 2.0 * n * conv.max() / (S * S)


def get_subgradient_at_lag(f, k):
    """Subgradient of conv[k] w.r.t. f[:, 0<=k-j<n]: d/d f[j] = 2*f[k-j]."""
    n = len(f)
    grad = np.zeros(n)
    j_min = max(0, k - n + 1)
    j_max = min(k, n - 1)
    if j_max >= j_min:
        grad[j_min:j_max + 1] = 2.0 * f[k - j_max:k - j_min + 1][::-1]
    return grad


def subgradient_step(f, conv, top_frac=0.05, n_min_active=3):
    """Compute smoothed subgradient using top lags within top_frac of max."""
    n = len(f)
    max_val = conv.max()
    if max_val < 1e-15:
        return np.zeros(n)
    threshold = max_val * (1.0 - top_frac)
    active = np.where(conv >= threshold)[0]
    if len(active) < n_min_active:
        # Use top n_min_active lags
        active = np.argsort(conv)[-n_min_active:]
    # Weight by how close to max
    weights = conv[active] / conv[active].sum()
    grad = np.zeros(n)
    for w, k in zip(weights, active):
        grad += w * get_subgradient_at_lag(f, int(k))
    return grad


def nesterov_subgradient(f, deadline, lr_init=0.02, momentum=0.9, top_frac=0.03,
                         name="", anneal_rate=0.999):
    """Nesterov accelerated subgradient with simulated annealing."""
    f = f.astype(np.float64).copy()
    n = len(f)
    S = f.sum()
    best_f = f.copy()

    lr = lr_init
    v = np.zeros(n)
    y = f.copy()  # Nesterov lookahead
    no_improve = 0
    it = 0
    temp = 1.0
    
    while time.monotonic() < deadline:
        conv_y = np.convolve(y, y)
        max_val = conv_y.max()
        if max_val < 1e-15:
            break
        
        # Subgradient at y
        grad = subgradient_step(y, conv_y, top_frac=top_frac)
        
        # Normalize gradient (remove constant offset for scale invariance)
        grad_mean = grad.mean()
        grad = grad - grad_mean
        
        # Nesterov update
        v = momentum * v - lr * grad / S
        y = f + v
        y = np.clip(y, 0.0, None)
        
        # Renormalize
        y_S = y.sum()
        if y_S < 1e-10:
            y = f.copy()
            y_S = S
        else:
            y = y * (S / y_S)
        
        obj = compute_objective(y)
        it += 1

            if obj < best_obj:

            v *= 0.5
            no_improve = 0
            if lr < lr_init * 0.005:
                break
    
    return best_f, best_obj


def build_conv_matrix(f):
    """Build sparse matrix A where A@g gives convolve(f, g)."""
    n = len(f)
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        j_min = max(0, k - n + 1)
        j_max = min(k, n - 1)
        if j_max >= j_min:
            js = np.arange(j_min, j_max + 1)
            rows.extend([k] * len(js))
            cols.extend(js.tolist())
            vals.extend(f[js].tolist())
    A = csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))
    return A


def lp_solve_direction(f, M, timeout=5.0):
    """Solve max sum(g) s.t. conv(f,g) <= M, g >= 0."""
    n = len(f)
    A = build_conv_matrix(f)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    try:
        result = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                         options={'time_limit': timeout, 'presolve': True})
        if result.success:
            return result.x
    except Exception:
        pass
    return None

def coordinate_descent(f, deadline, name="", max_passes=5):
    """Coordinate descent with candidate evaluation."""
    f = f.astype(np.float64).copy()
    n = len(f)
    S = f.sum()
    best_f = f.copy()
    best_obj = compute_objective(f)
    
    for p in range(max_passes):
        if time.monotonic() > deadline:
            break
        any_improve = False
        # Random order
        order = np.random.permutation(n)
        for j in order:
            if time.monotonic() > deadline:
                break
            old = f[j]
            if old < 1e-10:
                continue
            # Generate candidates
            cands = [old * 0.3, old * 0.6, old * 1.4, old * 2.0, 0.0]
            if j > 0:
                cands.append(f[j - 1] * 0.8)
            if j < n - 1:
                cands.append(f[j + 1] * 1.2)
            
            best_c = old
            best_co = best_obj
            for c in cands:
                f[j] = max(0.0, c)
                ns = f.sum()
                if ns < 0.01:
                    f[j] = old
                    continue
                fn = f * (S / ns)
                obj = compute_objective(fn)
                if obj < best_co:
                    best_co = obj
                    best_c = max(0.0, c)
            
            if best_c != old:
                f[j] = best_c
                ns = f.sum()
                if ns > 0:
                    f = f * (S / ns)
                any_improve = True
                if compute_objective(f) < best_obj:
                    best_obj = compute_objective(f)
                    best_f = f.copy()
            else:
                f[j] = old
        
        if not any_improve:
            break

            print(f"  Optimizing {label} (init obj={obj_init:.4f})...")
            f_opt, obj_opt = optimize_single(f_init, per_init_deadline,
                                             name=f"[{n}/{label}]", rng=rng)
            try_update(f_opt)
            print(f"    -> {obj_opt:.6f}")
    
    # Final polishing
    if len(best_sequence) > 0 and time.monotonic() < deadline - 5:
        print(f"\n{'='*50}")
        print(f"Final polish: n={len(best_sequence)}, current best={best_score:.6f}")
        print(f"{'='*50}")
        
        remaining = deadline - time.monotonic() - 3
        bf = np.asarray(best_sequence, dtype=np.float64).copy()
        
        # LP polish
        if remaining > 30:
            t_lp = time.monotonic() + remaining * 0.3
            f_lp, obj_lp = lp_iterate(bf, t_lp, name="[final LP]")
            try_update(f_lp)
            bf = np.asarray(best_sequence, dtype=np.float64)
            remaining = deadline - time.monotonic() - 3
        
        # Subgradient polish
        if remaining > 15:
            t_sg = time.monotonic() + remaining * 0.5
            f_sg, obj_sg = nesterov_subgradient(bf, t_sg, lr_init=0.005, momentum=0.95,
                                                top_frac=0.01, name="[final SG]")
            try_update(f_sg)
            bf = np.asarray(best_sequence, dtype=np.float64)
            remaining = deadline - time.monotonic() - 3
        
        # Coordinate descent
        if remaining > 10:
            f_cd, obj_cd = coordinate_descent(bf, deadline - 3, name="[final CD]",
                                              max_passes=3)
            try_update(f_cd)
    
    # Verify final answer
    final_score = evaluate_sequence([float(x) for x in best_sequence])
    print(f"\n{'='*50}")
    print(f"FINAL ANSWER: n={len(best_sequence)}, evaluator score={final_score:.6f}")
    print(f"Total time: {time.monotonic()-t0:.1f}s")
    print(f"{'='*50}")
    
    return [float(max(0.0, min(1000.0, x))) for x in best_sequence]
