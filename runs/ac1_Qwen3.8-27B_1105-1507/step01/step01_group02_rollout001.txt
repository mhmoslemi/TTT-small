def local_refine(a, deadline, rng, max_iters=100000):

    n = len(a)

    c = fast_convolve(a)
    best_peak = np.max(c)
    best_a = a.copy()

    stall = 0

            stall += 1

            lr = max(lr * 0.7, lr_min)
            if lr < lr_min * 5:
                # Restart with perturbation
                if time.time() < deadline - 5:

                    a_sp = project_simplex(np.maximum(a_sp, 0))
                    s_sp = a_sp.sum()
                    if s_sp > 1e-10:
                        a_sp = a_sp / s_sp
                        if compute_score(a_sp) < compute_score(a):
                            a = a_sp
            stall = 0
        
        # Get active peaks
        threshold = peak * (1.0 - 1e-5)
        peak_indices = np.where(c >= threshold)[0]
        if len(peak_indices) > 20:
            top = np.argsort(c[peak_indices])[::-1][:10]
            peak_indices = peak_indices[top]
        
        # Compute subgradient
        grad = np.zeros(n)
        for k in peak_indices:
            lo = max(0, k - n + 1)
            hi = min(n - 1, k)
            if lo <= hi:
                grad[lo:hi + 1] += 2.0 * a[k - hi:k - lo + 1][::-1]
        if len(peak_indices) > 0:
            grad /= len(peak_indices)
        
        a = project_simplex(a - lr * grad)
    
    return best_a

"""
Multi-strategy optimizer for minimizing 2*n*max(conv(a,a))/sum(a)^2.
Combines differential evolution (global search), alternating projections
(spectral shaping), spectral-phase mutations, and adaptive subgradient
descent (local refinement) across multiple sequence lengths.
"""

import numpy as np
import time


def fast_convolve(a):
    """Compute full linear convolution a*a using FFT."""
    n = len(a)
    size = 2 * n - 1
    N = 1 << (size - 1).bit_length()
    fa = np.fft.rfft(a, N)
    fc = fa * fa
    result = np.fft.irfft(fc, N)[:size]
    result[result < 0] = 0.0
    return result


def compute_score(a):
    """Compute the objective: 2*n*max(conv(a,a))/sum(a)^2."""
    n = len(a)
    c = fast_convolve(a)
    max_b = np.max(c)
    sum_a = np.sum(a)
    if sum_a < 0.01:
        return np.inf
    return 2.0 * n * max_b / (sum_a * sum_a)


def project_simplex(v):
    """Project vector v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cv = np.cumsum(u) - 1.0
    rho_candidates = np.where(u * np.arange(1, n + 1) > cv)[0]
    if len(rho_candidates) == 0:
        return np.ones(n) / n
    rho = rho_candidates[-1]
    theta = cv[rho] / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def alternating_projection(a, n_iter=50, rng=None, flatness=1.0):
    """
    Alternating projections between flat-spectrum constraint (frequency domain)
    and non-negativity constraint (time domain).
    """
    n = len(a)
    N = 1 << (2 * n - 2).bit_length()
    a = a.copy()
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-10:
        a = np.ones(n) / n
    else:
        a = a / s

    for it in range(n_iter):
        # Time domain: enforce non-negativity and normalization
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s < 1e-10:
            a = np.ones(n) / n
            s = 1.0
        a = a / s

        # Frequency domain: enforce flat magnitude
        F = np.fft.rfft(a, N)
        mag = np.abs(F)
        if len(mag) > 1:
            mean_mag = np.mean(mag[1:])
        else:
            mean_mag = 1.0
        if mean_mag < 1e-15:
            mean_mag = 1.0
        target_mag = np.full(len(F), mean_mag)
        target_mag[0] = F[0].real
        # Optional taper for better non-negativity
        if flatness < 1.0:
            weights = np.linspace(1.0, flatness, len(F))
            target_mag[1:] *= weights
        phase = np.angle(F)
        F_new = target_mag * np.exp(1j * phase)
        F_new[0] = F[0].real + 0j
        a = np.fft.irfft(F_new, N)[:n]

    a = np.maximum(a, 0.0)
    s = a.sum()
    if s > 1e-10:
        a = a / s
    return a


def spectral_perturb(a, rng, strength=0.15):
    """Perturb the phase of the DFT to create a new candidate."""
    n = len(a)
    N = 1 << (2 * n - 2).bit_length()
    F = np.fft.rfft(a, N)
    phase = np.angle(F)
    mag = np.abs(F)
    perturbation = strength * rng.normal(0, 1, len(phase))
    phase = phase + perturbation
    F_new = mag * np.exp(1j * phase)
    F_new[0] = F[0].real + 0j
    a_new = np.fft.irfft(F_new, N)[:n]
    return a_new


def spectral_magnitude_perturb(a, rng, strength=0.1):
    """Perturb the magnitude of the DFT and recover a non-negative sequence."""
    n = len(a)
    N = 1 << (2 * n - 2).bit_length()
    F = np.fft.rfft(a, N)
    phase = np.angle(F)
    mag = np.abs(F)
    mag_perturb = mag * (1.0 + strength * rng.normal(0, 1, len(mag)))
    mag_perturb = np.maximum(mag_perturb, 0.0)
    F_new = mag_perturb * np.exp(1j * phase)
    F_new[0] = F[0].real + 0j
    a_new = np.fft.irfft(F_new, N)[:n]
    return a_new


def local_refine(a, deadline, rng, max_iters=200000):
    """
    Adaptive subgradient descent with restarts and spectral perturbations.
    Minimizes max(conv(a,a)) subject to a >= 0, sum(a) = 1.
    """
    n = len(a)
    a = project_simplex(a.copy().astype(np.float64))
    c = fast_convolve(a)
    best_peak = np.max(c)
    best_a = a.copy()

    lr = 0.5 / np.sqrt(n)
    lr_min = 1e-7
    stall = 0
    max_stall = 200
    it = 0

    while time.time() < deadline and it < max_iters:
        it += 1

        c = fast_convolve(a)
        peak = np.max(c)

        if peak < best_peak * (1.0 - 1e-10):
            best_peak = peak
            best_a = a.copy()
            stall = 0
        else:
            stall += 1

        if stall > max_stall:
            lr = max(lr * 0.75, lr_min)
            if lr < lr_min * 10 and time.time() < deadline - 3:
                # Try multiple restart strategies
                candidates = []
                # Additive noise
                a_noise = project_simplex(a + rng.normal(0, 0.04 / np.sqrt(n), n))
                candidates.append(a_noise)
                # Spectral phase perturbation
                a_spec = spectral_perturb(a, rng, strength=0.25)
                a_spec = np.maximum(a_spec, 0)
                s = a_spec.sum()
                if s > 1e-10:
                    candidates.append(a_spec / s)
                # Small alt-projection
                a_alt = alternating_projection(a, n_iter=10, rng=rng)
                candidates.append(a_alt)

                best_restart = a
                best_restart_score = compute_score(a)
                for cand in candidates:
                    sc = compute_score(cand)
                    if np.isfinite(sc) and sc < best_restart_score:
                        best_restart_score = sc
                        best_restart = cand
                a = best_restart
            stall = 0

        # Get active peak indices
        threshold = peak * (1.0 - 1e-4)
        peak_indices = np.where(c >= threshold)[0]
        if len(peak_indices) > 15:
            top = np.argsort(c[peak_indices])[::-1][:8]
            peak_indices = peak_indices[top]

        # Compute subgradient
        grad = np.zeros(n)
        for k in peak_indices:
            lo = max(0, k - n + 1)
            hi = min(n - 1, k)
            if lo <= hi:
                grad[lo:hi + 1] += 2.0 * a[k - hi:k - lo + 1][::-1]
        if len(peak_indices) > 0:
            grad /= len(peak_indices)

        a = project_simplex(a - lr * grad)

    return best_a

        for i in range(pop_size):
            if time.time() > deadline:

                    best_a = trial.copy()
                    best_idx = i
                    no_improve = 0
                    if gen % 10 == 1:

        best_score = compute_score(best_seq)

    # Prepare result
    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    eval_score = evaluate_sequence(result)
    print(f"\nFINAL: evaluate_score={eval_score:.6f}, compute_score={best_score:.6f}, "
          f"length={len(result)}, time_used={time.time()-t_start:.0f}s")

    if not np.isfinite(eval_score):
        result = [1.0 / 256.0] * 256
        eval_score = evaluate_sequence(result)
        print(f"Fallback evaluate_score: {eval_score:.6f}")

    return result
