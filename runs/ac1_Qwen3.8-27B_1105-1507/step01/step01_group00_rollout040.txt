Multi-phase optimization for minimizing 2n*max(conv(a,a))/sum(a)^2.
Phase 1: L-BFGS-B with Lp-norm smoothing (p=16) using softmax parameterization.
Phase 2: LP-based directional refinement (keeping conv(f,g) <= M).
Phase 3: Subgradient polish on the exact max objective.
Multiple lengths and initializations explored within the time budget.
"""
import numpy as np
import time
from scipy.optimize import minimize, linprog
from scipy.sparse import csr_matrix


def fast_convolve(a):
    """Fast linear convolution using FFT."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size *= 2
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2*n-1]


def lp_smooth_objective(b, p=16.0):
    """
    Objective in terms of unconstrained b, where a = softmax(b).
    Returns the Lp norm of conv(a,a) (which is the max to optimize since sum(a)=1).
    """
    b_shift = b - np.max(b)
    a = np.exp(b_shift)
    s = np.sum(a)
    if s < 1e-30:
        return 1e10
    a = a / s
    conv = fast_convolve(a)
    return np.sum(conv**p)**(1.0/p)


def lp_smooth_gradient(b, p=16.0):
    """
    Gradient of the Lp-smoothed objective w.r.t. b.
    a = softmax(b), objective = ||conv(a,a)||_p.
    """
    n = len(b)
    b_shift = b - np.max(b)
    a = np.exp(b_shift)
    s = np.sum(a)
    if s < 1e-30:
        return np.zeros(n)
    a = a / s  # a is now softmax(b)
    
    conv = fast_convolve(a)
    F = np.sum(conv**p)
    if F <= 0:
        return np.zeros(n)
    
    # d(objective)/da[j] where objective = F^(1/p)
    # = (1/p) * F^(1/p-1) * p * sum_k conv[k]^(p-1) * d(conv[k])/da[j]
    # = F^(1/p-1) * sum_k conv[k]^(p-1) * 2*a[k-j] * I(0<=k-j<n)
    
    v = conv**(p-1)  # length 2n-1
    factor = F**(1.0/p - 1.0)
    
    # grad_a[j] = factor * 2 * sum_m v[m+j] * a[m] for m=0..n-1, m+j < 2n-1
    # This is 2*factor * cross_correlation(v, a) evaluated at appropriate shifts
    
    # Using FFT for efficiency:
    # sum_m v[m+j]*a[m] = (v * a_reversed)[j + n - 1] ... 
    # Actually, let me think about this differently.
    # sum_m v[m+j]*a[m] for m = 0..n-1
    # Let's define the cross-correlation: (v ⋆ a)[τ] = sum_m v[m+τ]*a[m]
    # We want (v ⋆ a)[j] for j = 0..n-1.
    # 
    # Using FFT: cross-correlation can be computed as ifft(conj(fft(v)) * fft(a))
    # But v has length 2n-1 and a has length n. We need to be careful with padding.
    
    # Alternative: use the relationship to convolution.
    # sum_m v[m+j]*a[m] = sum_m a[m]*v[m+j]
    # This is the cross-correlation of a and v at lag j.
    # np.correlate(a, v, mode='full')[j + len(v) - 1] ... hmm, let me just do it with FFT.
    
    # Actually, for n <= 2048, direct computation is fast enough (n^2 = 4M for n=2048).
    # Let me use a vectorized approach:
    
    # grad_a[j] = 2 * factor * sum_{m=0}^{n-1} a[m] * v[m+j]
    # where v has length 2n-1, so m+j < 2n-1 means m < 2n-1-j.
    # For j >= 1: m can go up to n-1 and m+j <= n-1+j <= 2n-2 < 2n-1. Always valid.
    # For j = 0: m+j = m <= n-1 < 2n-1. Always valid.
    # So all terms are valid!
    
    # This is: 2*factor * (a * v) in the cross-correlation sense.
    # Let's compute it as: for each j, sum a[m]*v[m+j].
    # This is equivalent to: convolve(a_padded, v_flipped)... 
    
    # Simplest: use matrix multiplication or direct loop (n=512 to 2048, should be OK)
    # Or use the FFT trick:
    # sum_m a[m]*v[m+j] = ifft(fft(a, N) * fft(v, N)) evaluated at index j
    # where N >= n + len(v) - 1 = n + 2n-2 = 3n-2. Use next power of 2.
    
    N = 1
    while N < 3 * n:
        N *= 2
    fa = np.fft.rfft(a, N)
    fv = np.fft.rfft(v, N)
    # Product gives circular cross-correlation
    corr = np.fft.irfft(fa * fv, N)
    # corr[j] = sum_m a[m]*v[m+j] (circular, but since we padded enough, it's linear for j < n)
    grad_a = 2.0 * factor * corr[:n]
    
    # Now chain rule: da/db = a * (1 - a) (for softmax)
    # Actually, for softmax: da_i/db_j = a_i * (delta_ij - a_j)
    # So db objective = sum_i (d obj / da_i) * (da_i / db_j)
    # = sum_i grad_a[i] * a[i] * (delta_ij - a_j)
    # = a[j] * grad_a[j] - a[j] * sum_i grad_a[i] * a[i]
    
    weighted_sum = np.sum(grad_a * a)
    grad_b = a * (grad_a - weighted_sum)
    
    return grad_b


def build_conv_matrix_sparse(f):
    """Build sparse matrix A where (A@g)[k] = convolve(f,g)[k]."""
    n = len(f)
    rows = np.empty(n * n, dtype=np.int32)
    cols = np.empty(n * n, dtype=np.int32)
    vals = np.empty(n * n, dtype=np.float64)
    idx = 0
    for k in range(2 * n - 1):
        j_min = max(0, k - (n - 1))
        j_max = min(k, n - 1)
        for j in range(j_min, j_max + 1):
            rows[idx] = k
            cols[idx] = j
            vals[idx] = f[k - j]
            idx += 1
    rows = rows[:idx]
    cols = cols[:idx]
    vals = vals[:idx]
    return csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))


def solve_lp_direction(f, M):
    """Solve: max sum(g) s.t. g >= 0, convolve(f,g)_k <= M for all k."""
    n = len(f)
    A = build_conv_matrix_sparse(f)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    try:
        result = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
        if result.success:
            return result.x
    except Exception:
        pass
    return None


def lp_iterate(f, deadline, step_size_init=0.15, max_iters=200, verbose_prefix=""):

    best_f = f.copy()
    best_score = evaluate_sequence(f.tolist())
    step_size = step_size_init
    no_improve = 0
    it = 0

    while time.monotonic() < deadline and it < max_iters:
        it += 1
        conv_f = fast_convolve(f)
        M = float(np.max(conv_f))
        if M <= 0:
            break

        start_t = time.monotonic()
        g = solve_lp_direction(f, M)
        lp_time = time.monotonic() - start_t

            no_improve += 1
            if no_improve > 20:
                break
            continue

        sum_g = float(np.sum(g))
        if sum_g <= 1e-15:
            no_improve += 1
            continue

        # Rescale g to have same sum as f
        S_f = float(np.sum(f))
        g = g * (S_f / sum_g)

        # Blend
        new_f = (1.0 - step_size) * f + step_size * g
        new_f = np.clip(new_f, 0.0, 1000.0)

        if np.isfinite(new_score) and new_score < best_score:

            step_size *= 0.7

def subgradient_polish(f, deadline, lr_init=0.01, verbose_prefix=""):
    """
    Subgradient method on the exact max objective.
    At each step, find the lag with max convolution, compute subgradient, update.
    """
    f = f.copy().astype(np.float64)
    best_f = f.copy()
    best_score = evaluate_sequence(f.tolist())
    n = len(f)
    lr = lr_init
    no_improve = 0

    for it in range(2000):
        if time.monotonic() > deadline:
            break

        conv_f = fast_convolve(f)
        max_idx = int(np.argmax(conv_f))
        M = conv_f[max_idx]
        S = float(np.sum(f))

        if M <= 0 or S <= 0:
            break

        # Subgradient of max_k conv(a,a)_k w.r.t. a[j]:
        # = 2 * a[max_idx - j] if 0 <= max_idx - j < n, else 0
        grad = np.zeros(n)
        j_min = max(0, max_idx - (n - 1))
        j_max = min(max_idx, n - 1)
        for j in range(j_min, j_max + 1):
            grad[j] = 2.0 * f[max_idx - j]
        
        # Update: a -> a - lr * grad (to decrease the max)
        new_f = f - lr * grad
        new_f = np.clip(new_f, 0.0, 1000.0)

        # Renormalize to maintain sum
        new_sum = float(np.sum(new_f))
        if new_sum > 1e-10:
            new_f = new_f * (S / new_sum)

        new_score = evaluate_sequence(new_f.tolist())
        if np.isfinite(new_score) and new_score < best_score:
            f = new_f
            best_f = f.copy()
            best_score = new_score
            no_improve = 0
        else:
            no_improve += 1
            lr *= 0.8
            if lr < 1e-8 or no_improve > 50:
                break

    return best_f, best_score


def lfbgbs_optimize(f_init, deadline, p=16.0, verbose_prefix=""):
    """
    Optimize using L-BFGS-B on the Lp-smoothed objective.
    f_init is the initial sequence (will be converted to softmax parameterization).
    Returns (best_f, best_score).
    """
    n = len(f_init)
    f_init = f_init.copy()
    S = float(np.sum(f_init))
    if S < 1e-10:
        return f_init, evaluate_sequence(f_init.tolist())
    
    # Normalize to sum = 1 for the parameterization
    a0 = f_init / S
    
    # Convert to log-space for softmax parameterization
    # a = softmax(b) = exp(b) / sum(exp(b))
    # If a > 0, then b = log(a) + constant (any constant works)
    # To avoid issues with very small a, add epsilon
    eps = 1e-10
    b0 = np.log(a0 + eps)
    
    best_f = f_init.copy()
    best_score = evaluate_sequence(f_init.tolist())
    
    # Run L-BFGS-B
    try:
        # Use the objective and gradient in b-space
        result = minimize(
            lp_smooth_objective,
            b0,
            method='L-BFGS-B',
            jac=lp_smooth_gradient,
            args=(p,),
            options={
                'maxiter': 500,
                'maxfun': 2000,
                'ftol': 1e-10,
                'gtol': 1e-8,
                'factr': 1e8,
            }
        )
        
        if result.x is not None:
            b_opt = result.x
            b_shift = b_opt - np.max(b_opt)
            a_opt = np.exp(b_shift)
            a_opt = a_opt / np.sum(a_opt)
            # Scale back to original sum
            a_opt = a_opt * S
            score = evaluate_sequence(a_opt.tolist())
            if np.isfinite(score) and score < best_score:
                best_f = a_opt
                best_score = score
    except Exception as e:
        print(f"    {verbose_prefix}L-BFGS-B failed: {e}")
    
    return best_f, best_score

"""
Multi-phase optimization for minimizing 2n*max(conv(a,a))/sum(a)^2.
Phase 1: L-BFGS-B with Lp-norm smoothing (p=8,16,32) using softmax parameterization.
Phase 2: LP-based directional refinement (max sum(g) s.t. conv(f,g)<=M, g>=0).
Phase 3: Subgradient polish + random restarts on the best sequence.
Explores multiple lengths and diverse initializations within the time budget.
"""
import numpy as np
import time
from scipy.optimize import minimize, linprog
from scipy.sparse import csr_matrix


def fast_convolve(a):
    """Fast linear convolution using FFT."""
    n = len(a)
    if n == 1:
        return np.array([a[0] * a[0]])
    size = 1
    while size < 2 * n - 1:
        size *= 2
    fa = np.fft.rfft(a, size)
    result = np.fft.irfft(fa * fa, size)
    return result[:2 * n - 1]


def lp_objective(b, p):
    """Lp-smoothed objective in softmax parameter space."""
    b_shift = b - np.max(b)
    a = np.exp(b_shift)
    s = np.sum(a)
    if s < 1e-30:
        return 1e10
    a = a / s
    conv = fast_convolve(a)
    return np.sum(conv ** p) ** (1.0 / p)


def lp_gradient(b, p):
    """Gradient of Lp-smoothed objective w.r.t. softmax parameters."""
    n = len(b)
    b_shift = b - np.max(b)
    a = np.exp(b_shift)
    s = np.sum(a)
    if s < 1e-30:
        return np.zeros(n)
    a = a / s

    conv = fast_convolve(a)
    F = np.sum(conv ** p)
    if F <= 1e-30:
        return np.zeros(n)

    v = conv ** (p - 1)
    factor = F ** (1.0 / p - 1.0)

    # grad_a[j] = 2 * factor * sum_{m=0}^{n-1} a[m] * v[m+j]
    # Use FFT cross-correlation
    N = 1
    while N < 3 * n:
        N *= 2
    fa = np.fft.rfft(a, N)
    fv = np.fft.rfft(v, N)
    corr = np.fft.irfft(fa * fv, N)
    grad_a = 2.0 * factor * corr[:n]

    # Chain rule for softmax
    weighted_sum = np.sum(grad_a * a)
    grad_b = a * (grad_a - weighted_sum)
    return grad_b


def build_conv_matrix_sparse(f):
    """Build sparse matrix A where (A@g)[k] = convolve(f,g)[k]."""
    n = len(f)
    rows = np.empty(n * n, dtype=np.int32)
    cols = np.empty(n * n, dtype=np.int32)
    vals = np.empty(n * n, dtype=np.float64)
    idx = 0
    for k in range(2 * n - 1):
        j_min = max(0, k - (n - 1))
        j_max = min(k, n - 1)
        for j in range(j_min, j_max + 1):
            rows[idx] = k
            cols[idx] = j
            vals[idx] = f[k - j]
            idx += 1
    rows = rows[:idx]
    cols = cols[:idx]
    vals = vals[:idx]
    return csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))


def solve_lp_direction(f, M):
    """Solve: max sum(g) s.t. g >= 0, convolve(f,g)_k <= M for all k."""
    n = len(f)
    A = build_conv_matrix_sparse(f)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    try:
        result = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
        if result.success:
            return result.x
    except Exception:
        pass
    return None

    f = f.copy().astype(np.float64)

    step_size = step_size_init
    no_improve = 0

    """Subgradient method on the exact max objective."""
    f = f.copy().astype(np.float64)
    best_f = f.copy()
    best_score = evaluate_sequence(f.tolist())
    n = len(f)
    lr = lr_init
    no_improve = 0

    for it in range(3000):
        if time.monotonic() > deadline:
            break

        conv_f = fast_convolve(f)
        max_idx = int(np.argmax(conv_f))
        M = conv_f[max_idx]
        S = float(np.sum(f))

        if M <= 0 or S <= 1e-10:
            break

        grad = np.zeros(n)
        j_min = max(0, max_idx - (n - 1))
        j_max = min(max_idx, n - 1)
        for j in range(j_min, j_max + 1):
            grad[j] = 2.0 * f[max_idx - j]

        new_f = f - lr * grad
        new_f = np.clip(new_f, 0.0, 1000.0)

        new_sum = float(np.sum(new_f))
        if new_sum > 1e-10:
            new_f = new_f * (S / new_sum)

        new_score = evaluate_sequence(new_f.tolist())
        if np.isfinite(new_score) and new_score < best_score:
            f = new_f
            best_f = f.copy()
            best_score = new_score
            no_improve = 0
            if (it + 1) % 500 == 0:
                print(f"    {verbose_prefix}subgrad {it+1}: {best_score:.6f}")
        else:
            no_improve += 1
            lr *= 0.85
            if lr < 1e-9 or no_improve > 60:
                break

    return best_f, best_score


def run_lfbgbs(f_init, deadline, p=16.0, max_iter=300, verbose_prefix=""):
    """Run L-BFGS-B with Lp smoothing on a given initial sequence."""
    n = len(f_init)
    f_init = f_init.copy().astype(np.float64)
    S = float(np.sum(f_init))
    if S < 1e-10:
        return f_init, evaluate_sequence(f_init.tolist())

    a0 = f_init / S
    eps = 1e-12
    b0 = np.log(np.maximum(a0, eps))
    # Center b0 to avoid numerical issues
    b0 = b0 - np.mean(b0)

    best_f = f_init.copy()
    best_score = evaluate_sequence(f_init.tolist())

    try:
        result = minimize(
            lp_objective,
            b0,
            method='L-BFGS-B',
            jac=lp_gradient,
            args=(p,),
            options={'maxiter': max_iter, 'maxfun': max_iter * 4, 'ftol': 1e-12, 'gtol': 1e-8}
        )
        if result.x is not None:
            b_opt = result.x
            b_shift = b_opt - np.max(b_opt)
            a_opt = np.exp(b_shift)
            a_sum = np.sum(a_opt)
            if a_sum > 1e-10:
                a_opt = a_opt / a_sum * S
                score = evaluate_sequence(a_opt.tolist())
                if np.isfinite(score) and score < best_score:
                    best_f = a_opt.copy()
                    best_score = score
    except Exception as e:
        print(f"    {verbose_prefix}L-BFGS-B error: {e}")

    return best_f, best_score


def generate_initializations(n, parent_seq, rng):
    """Generate diverse initializations for a given length n."""
    inits = []
    idx = np.arange(n, dtype=np.float64)
    center = (n - 1) / 2.0

    # Uniform
    inits.append(('uniform', np.ones(n)))

    # Hann window (tapers to zero at edges)
    hann = 0.5 * (1.0 - np.cos(2.0 * np.pi * idx / (n - 1)))
    inits.append(('hann', np.maximum(hann, 1e-6)))

    # Raised cosine
    cos_t = 0.5 * (1.0 + np.cos(2.0 * np.pi * idx / (n - 1)))
    inits.append(('cosine', np.maximum(cos_t, 1e-6)))

    # Blackman
    blackman = (0.42 - 0.5 * np.cos(2.0 * np.pi * idx / (n - 1))
                + 0.08 * np.cos(4.0 * np.pi * idx / (n - 1)))
    inits.append(('blackman', np.maximum(blackman, 1e-4)))

    # Flat top with cosine-rolled edges
    edge = max(2, n // 12)
    flat = np.ones(n)
    for i in range(edge):
        t = (i + 1) / (edge + 1)
        flat[i] = 0.5 * (1 - np.cos(np.pi * t))
        flat[n - 1 - i] = 0.5 * (1 - np.cos(np.pi * t))
    inits.append(('flat_top', flat))

    # Gaussian
    gauss = np.exp(-0.5 * ((idx - center) / (n / 4.0)) ** 2)
    inits.append(('gauss', gauss))

    # Bimodal
    bim = (np.exp(-0.5 * ((idx - n * 0.25) / (n / 8.0)) ** 2)
           + np.exp(-0.5 * ((idx - n * 0.75) / (n / 8.0)) ** 2))
    inits.append(('bimodal', bim))

    # Sine (abs)
    sine = np.abs(np.sin(np.pi * idx / (n - 1)))
    inits.append(('sine', np.maximum(sine, 1e-6)))

    # Interpolated parent
    if parent_seq is not None and len(parent_seq) > 0:
        if len(parent_seq) == n:
            inits.append(('parent', parent_seq.copy()))
        else:
            parent_interp = np.interp(idx, np.linspace(0, n - 1, len(parent_seq)), parent_seq)
            inits.append(('parent_interp', np.maximum(parent_interp, 1e-6)))

    # Random smooth
    r = rng.uniform(0.3, 1.5, size=n)
    # Smooth it
    kernel = np.ones(5) / 5.0
    r_smooth = np.convolve(r, kernel, mode='same')
    inits.append(('random_smooth', r_smooth))

    return inits


def propose_candidate(budget_s=1500, **kwargs):
    t0 = time.monotonic()
    deadline = t0 + max(1.0, float(budget_s) - 5.0)
    total_budget = float(budget_s) - 5.0

    best_sequence = None
    best_score = float('inf')
    rng = np.random.default_rng(777)

    # Load parent
    parent_seq = None
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64).copy()
        parent = np.clip(parent, 0.0, 1000.0)
        if parent.ndim == 1 and parent.size > 0 and np.all(np.isfinite(parent)):
            parent_seq = parent
            score = evaluate_sequence(parent.tolist())
            if np.isfinite(score):
                best_score = score
                best_sequence = parent.copy()
                print(f"Parent n={parent.size}: score={score:.6f}")
    except (NameError, Exception) as e:
        print(f"No parent: {e}")

    # Phase 1: L-BFGS-B + LP across multiple lengths
    # Time allocation: 45% for Phase 1
    phase1_deadline = t0 + 0.45 * total_budget

    lengths_to_try = [512, 256, 768, 1024]
    # If parent exists, prioritize its length
    if parent_seq is not None:
        parent_n = len(parent_seq)
        if parent_n in lengths_to_try:
            lengths_to_try.remove(parent_n)
        lengths_to_try.insert(0, parent_n)

    for n in lengths_to_try:
        if time.monotonic() > phase1_deadline - 15:
            print("Phase 1 time up")
            break

        print(f"\n=== Phase 1: n={n} ===")
        inits = generate_initializations(n, parent_seq, rng)

        for label, f_init in inits:
            if time.monotonic() > phase1_deadline - 10:
                break

            f_init = np.clip(f_init.astype(np.float64), 0.0, 1000.0)
            if np.sum(f_init) < 0.01:
                continue

            score = evaluate_sequence(f_init.tolist())
            if not np.isfinite(score):
                continue

            if score < best_score:
                best_score = score
                best_sequence = f_init.copy()
                tag = " [BEST]"
            else:
                tag = ""
            print(f"  n={n} {label}: {score:.6f}{tag}", end="")

            # L-BFGS-B (p=16)
            lf, ls = run_lfbgbs(f_init, phase1_deadline, p=16.0, max_iter=200,
                                verbose_prefix=f"[n={n} {label}]")
            if ls < best_score:
                best_score = ls
                best_sequence = lf.copy()
                print(f" -> LBF: {ls:.6f} [BEST]")
            else:
                print(f" -> LBF: {ls:.6f}", end="")

            # LP refinement
            if time.monotonic() < phase1_deadline - 8:
                lf2, ls2 = lp_iterate(lf, phase1_deadline - 5, step_size_init=0.12,
                                       max_iters=30, verbose_prefix=f"[n={n} {label}]")
                if ls2 < best_score:
                    best_score = ls2
                    best_sequence = lf2.copy()
                    print(f" LP: {ls2:.6f} [BEST]")
                else:
                    print(f" LP: {ls2:.6f}")

    print(f"\nAfter Phase 1: best_score={best_score:.6f}, n={len(best_sequence) if best_sequence is not None else 0}")

    # Phase 2: Focused refinement on the best sequence
    # Time allocation: 45% for Phase 2
    phase2_deadline = t0 + 0.90 * total_budget

    if best_sequence is not None and time.monotonic() < phase2_deadline:
        n_best = len(best_sequence)
        print(f"\n=== Phase 2: Focused refinement n={n_best} ===")

        f_curr = best_sequence.copy()

        # Multiple rounds of LP with varying step sizes
        step_sizes = [0.15, 0.08, 0.04, 0.02, 0.1, 0.05, 0.03]
        for si, ss in enumerate(step_sizes):
            if time.monotonic() > phase2_deadline - 10:
                break
            sub_dl = min(phase2_deadline - 5, time.monotonic() + (phase2_deadline - time.monotonic()) / (len(step_sizes) - si))
            f_curr, s_curr = lp_iterate(f_curr, sub_dl, step_size_init=ss,
                                         max_iters=40, verbose_prefix=f"[P2 ss={ss}]")
            if s_curr < best_score:
                best_score = s_curr
                best_sequence = f_curr.copy()
                print(f"  LP ss={ss}: {best_score:.6f} [BEST]")
            else:
                print(f"  LP ss={ss}: no improvement ({s_curr:.6f})")

        # Subgradient polish
        if time.monotonic() < phase2_deadline - 5:
            gf, gs = subgradient_polish(best_sequence, phase2_deadline - 3, lr_init=0.006)
            if gs < best_score:
                best_score = gs
                best_sequence = gf.copy()
                print(f"  Subgrad: {best_score:.6f} [BEST]")
            else:
                print(f"  Subgrad: no improvement ({gs:.6f})")

    # Phase 3: Diversity - random restarts and multi-p L-BFGS-B
    # Time allocation: 10% for Phase 3
    if best_sequence is not None and time.monotonic() < deadline - 10:
        n_best = len(best_sequence)
        print(f"\n=== Phase 3: Diversity n={n_best} ===")

        # Try different p values with L-BFGS-B
        for p in [8.0, 32.0, 64.0]:
            if time.monotonic() > deadline - 8:
                break
            lf, ls = run_lfbgbs(best_sequence, deadline - 5, p=p, max_iter=150,
                                verbose_prefix=f"[p={p}]")
            if ls < best_score:
                best_score = ls
                best_sequence = lf.copy()
                print(f"  p={p} LBF: {best_score:.6f} [BEST]")
            else:
                print(f"  p={p} LBF: {ls:.6f}")

            # Quick LP after each
            if time.monotonic() < deadline - 5 and ls < best_score + 0.01:
                lf2, ls2 = lp_iterate(lf, deadline - 3, step_size_init=0.08, max_iters=20)
                if ls2 < best_score:
                    best_score = ls2
                    best_sequence = lf2.copy()
                    print(f"  p={p} LP: {best_score:.6f} [BEST]")

        # Random perturbations
        for trial in range(5):
            if time.monotonic() > deadline - 8:
                break
            pert = rng.normal(0, 0.1, size=n_best)
            f_pert = np.clip(best_sequence + pert, 0.0, 1000.0)
            f_pert = f_pert / np.sum(f_pert) * np.sum(best_sequence)
            s_pert = evaluate_sequence(f_pert.tolist())

            if np.isfinite(s_pert):
                # Quick LP
                lf, ls = lp_iterate(f_pert, deadline - 5, step_size_init=0.1, max_iters=20)
                if ls < best_score:
                    best_score = ls
                    best_sequence = lf.copy()
                    print(f"  Perturb {trial}: {best_score:.6f} [BEST]")
                else:
                    print(f"  Perturb {trial}: {ls:.6f}")

    # Final LP polish
    if best_sequence is not None and time.monotonic() < deadline - 3:
        lf, ls = lp_iterate(best_sequence, deadline - 1, step_size_init=0.03, max_iters=20)
        if ls < best_score:
            best_score = ls
            best_sequence = lf.copy()
            print(f"\nFinal LP: {best_score:.6f} [BEST]")

    if best_sequence is None:
        return [1.0] * 128

    elapsed = time.monotonic() - t0
    print(f"\nFINAL: n={len(best_sequence)}, score={best_score:.6f}, time={elapsed:.1f}s")
    return [float(x) for x in best_sequence]
