The LP is:
- Maximize: `sum_j g_j`
- Subject to: `sum_j f_{k-j} * g_j <= M` for all k (i.e., `(f*g)[k] <= M`)
- And: `g_j >= 0`

This finds the "largest" `g` (in sum) that, when convolved with `f`, doesn't exceed `M`. The idea is that `g` is a direction that, when blended with `f`, reduces the peak of `f*f`.

Why does this work? If we blend `f` and `g`, the new convolution is:
`(f_new * f_new)[k] = (1-t)^2 (f*f)[k] + t(1-t)(f*g)[k] + t(1-t)(g*f)[k] + t^2 (g*g)[k]`
`= (1-t)^2 (f*f)[k] + 2t(1-t)(f*g)[k] + t^2 (g*g)[k]`

For small t, this is approximately:
`≈ (f*f)[k] - 2t((f*f)[k] - (f*g)[k]) + O(t^2)`

So if `(f*g)[k] < (f*f)[k]` for the peak k, the peak decreases. The LP ensures `(f*g)[k] <= M = max(f*f)`, and at the peak, `(f*f)[k] = M`, so the peak doesn't increase (it can only stay the same or decrease if the LP is strict at the peak).

This is a nice property. The LP approach is essentially the "Frank-Wolfe" or "conditional gradient" method applied to the problem.

def build_conv_matrix(f):
    """Build matrix A such that A @ b gives convolve(f, b, mode='full')."""
    n = len(f)
    A = np.zeros((2 * n - 1, n))
    for k in range(2 * n - 1):
        for j in range(max(0, k - n + 1), min(n, k + 1)):
            A[k, j] = f[k - j]
    return A

def compute_score(a):
    """Compute 2*n*max(conv(a,a))/sum(a)^2."""
    n = len(a)
    s = a.sum()
    if s < 0.01:
        return np.inf
    conv = np.convolve(a, a)
    return 2.0 * n * conv.max() / (s * s)

"""
Hybrid optimization for minimizing 2*n*max(conv(a,a))/sum(a)^2.
Combines LP-based Frank-Wolfe, smooth-max gradient descent with adaptive
temperature, simplex projection, and coordinate descent polishing.
Uses multiple sequence lengths and diverse starting points.

import numpy as np
from scipy.optimize import linprog
import time


def build_conv_matrix(f):
    """Build matrix A such that A @ b = np.convolve(f, b, mode='full')."""
    n = len(f)
    A = np.zeros((2 * n - 1, n))
    for k in range(2 * n - 1):
        j_start = max(0, k - n + 1)
        j_end = min(n, k + 1)
        for j in range(j_start, j_end):
            A[k, j] = f[k - j]
    return A


def score_normalized(a):
    """Compute 2*n*max(conv(a,a)) assuming sum(a)=1."""
    n = len(a)
    conv = np.convolve(a, a)
    return 2.0 * n * conv.max()


def full_score(a):
    """Compute full score = 2*n*max(conv(a,a))/sum(a)^2."""
    s = a.sum()
    if s < 0.01:
        return np.inf
    n = len(a)
    conv = np.convolve(a, a)
    return 2.0 * n * conv.max() / (s * s)


def project_simplex(a):
    """Project vector onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(a)
    if a.sum() <= 1e-15:
        return np.ones(n) / n
    u = np.sort(a)[::-1]
    cssv = np.cumsum(u)
    idx = np.arange(1, n + 1)
    rho_mask = u * idx > (cssv - 1)
    if not rho_mask.any():
        return np.ones(n) / n
    rho = np.where(rho_mask)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1)
    return np.maximum(a - theta, 0.0)


def smooth_max_grad(a, T):
    """
    Compute the smooth-max approximation and its gradient.
    Returns (smooth_max_value, gradient).
    The smooth max approximates max_k (a*a)[k].
    """
    n = len(a)
    conv = np.convolve(a, a)
    c_max_val = conv.max()
    shifted = conv - c_max_val
    # Avoid overflow
    max_shifted = shifted.max()
    if max_shifted > 500 / T:
        shifted = shifted - max_shifted / T
    exp_vals = np.exp(T * shifted)
    p_sum = exp_vals.sum()
    if p_sum < 1e-300:
        p = np.zeros(len(conv))
        p[np.argmax(conv)] = 1.0
    else:
        p = exp_vals / p_sum
    
    smooth_val = c_max_val + np.log(p_sum) / T
    
    # Gradient: grad_j = 2 * sum_k p_k * a[k-j] (where 0 <= k-j < n)
    grad = np.zeros(n)
    for j in range(n):
        k_start = j
        k_end = min(j + n, 2 * n - 1)
        grad[j] = 2.0 * np.dot(a[:k_end - k_start], p[k_start:k_end])
    
    return smooth_val, grad


def optimize_smooth_max(a, time_budget, T_start=5.0, T_end=200.0,
                        lr_init=0.005, verbose=False):
    """
    Smooth-max gradient descent with adaptive learning rate, momentum,
    and simplex projection. a should be non-negative.
    """
    n = len(a)
    a = a.astype(np.float64).copy()
    if a.sum() < 1e-10:
        a = np.ones(n)
    a = project_simplex(a)
    best_a = a.copy()
    best_score = score_normalized(a)
    
    start = time.monotonic()
    it = 0
    lr = lr_init
    momentum = np.zeros(n)
    beta = 0.9
    stall = 0
    
    while time.monotonic() - start < time_budget:
        progress = (time.monotonic() - start) / time_budget
        T = T_start * (T_end / T_start) ** (progress ** 0.5)
        
        _, grad = smooth_max_grad(a, T)
        
        # Momentum update
        momentum = beta * momentum + lr * grad
        a_new = a - momentum
        a_new = project_simplex(a_new)
        
        new_score = score_normalized(a_new)
        
        if new_score < best_score - 1e-15:
            best_a = a_new.copy()
            best_score = new_score
            a = a_new
            stall = 0
            # Increase lr slightly
            lr = min(lr * 1.02, lr_init * 5)
            # Decay momentum
            if it % 50 == 0:
                beta = max(0.5, beta * 0.999)
        else:
            lr *= 0.8
            stall += 1
            if stall > 100:
                # Reset with a random perturbation
                a = project_simplex(a + 0.01 * np.random.standard_normal(n))
                lr = lr_init * 0.5
                stall = 0
                momentum = np.zeros(n)
        
        it += 1
        if verbose and it % 200 == 0:
            elapsed = time.monotonic() - start
            print(f"    SM it={it}, score={best_score:.6f}, T={T:.1f}, lr={lr:.6f}, el={elapsed:.1f}s")
    
    return best_a, best_score

def coordinate_descent(a, n_iter=200, n_restarts=3):
    """
    Coordinate descent: for each coordinate, try several multipliers.
    Multiple passes until no improvement or n_iter passes done.
    """
    n = len(a)
    a = a.astype(np.float64).copy()
    a = project_simplex(a)
    best_a = a.copy()
    best_score = score_normalized(a)
    
    factors = [0.3, 0.5, 0.7, 0.85, 1.0, 1.15, 1.3, 1.5, 2.0, 3.0]
    
    for it in range(n_iter):
        improved = False
        for j in range(n):
            current_val = a[j]
            if current_val < 1e-12:
                # Try small positive values
                for f_try in [0.001, 0.01, 0.05]:
                    a_try = a.copy()
                    a_try[j] = f_try
                    a_try = project_simplex(a_try)
                    sc = score_normalized(a_try)
                    if sc < best_score - 1e-14:
                        a = a_try
                        best_a = a.copy()
                        best_score = sc
                        improved = True
            else:
                for factor in factors:
                    a_try = a.copy()
                    a_try[j] = current_val * factor
                    a_try = np.clip(a_try, 0, None)
                    a_try = project_simplex(a_try)
                    sc = score_normalized(a_try)
                    if sc < best_score - 1e-14:
                        a = a_try
                        best_a = a.copy()
                        best_score = sc
                        improved = True
                        break
        if not improved:
            if n_restarts > 1 and it > 20:
                # Try a random restart from current best
                a = project_simplex(best_a * (1 + 0.1 * np.random.standard_normal(n)))
                a = np.clip(a, 0, None)
                a = project_simplex(a)
                if score_normalized(a) < best_score:
                    best_a = a.copy()
                    best_score = score_normalized(a)
                    n_restarts -= 1
                continue
            break
    
    return best_a, best_score


def random_local_search(a, n_iter=5000, rng=None):
    """Random block-perturbation local search."""
    if rng is None:

    n = len(a)
    a = a.astype(np.float64).copy()
    a = project_simplex(a)
    best_a = a.copy()
    best_score = score_normalized(a)
    
    for it in range(n_iter):
        a_try = a.copy()
        # Perturb a random block

        start_idx = int(rng.integers(0, n - width))
        scale = np.exp(rng.normal(0, 0.5))
        a_try[start_idx:start_idx + width] *= scale
        a_try = np.clip(a_try, 0, None)
        a_try = project_simplex(a_try)
        sc = score_normalized(a_try)

    return best_a, best_score

    except Exception:
        parent = None
    
    if parent is not None:
        parent = np.clip(parent, 0, None)
        if parent.sum() > 0.01:
            parent = parent / parent.sum()
        else:
            parent = None
    
    print(f"Parent available: {parent is not None}")
    if parent is not None:
        print(f"Parent length: {len(parent)}, initial score: {score_normalized(parent):.6f}")
    print(f"Total budget: {budget_s}s")
    
    def update_best(f, score, label=""):
        nonlocal best_overall, best_score_overall
        if np.isfinite(score) and score < best_score_overall:
            best_score_overall = score
            best_overall = f.copy()
            print(f"  *** New best: {score:.6f} ({label}) ***")
    
    # === Phase 1: Optimize parent with various methods ===
    if parent is not None:
        n_p = len(parent)
        t_remaining = deadline - time.monotonic()
        
        # 1a: Smooth-max gradient descent on parent
        t_alloc = min(300, t_remaining * 0.3)
        if t_alloc > 30:
            print(f"\n[Phase 1a: SmoothMax on parent, n={n_p}, t={t_alloc:.0f}s]")
            f_opt, score = optimize_smooth_max(parent, t_alloc, T_start=5.0, T_end=100.0, lr_init=0.005, verbose=True)
            update_best(f_opt, score, f"SM parent n={n_p}")
        
        # 1b: LP on parent (or on result of 1a if it improved)
        t_remaining = deadline - time.monotonic()
        t_alloc = min(300, t_remaining * 0.3)
        if t_alloc > 30:
            start_f = best_overall if best_overall is not None and len(best_overall) == n_p else parent
            print(f"\n[Phase 1b: LP on parent, n={n_p}, t={t_alloc:.0f}s]")
            f_opt, score = optimize_lp(start_f, t_alloc, t0=0.4, verbose=True, rng=rng)
            update_best(f_opt, score, f"LP parent n={n_p}")
        
        # 1c: Coordinate descent polish
        t_remaining = deadline - time.monotonic()
        if t_remaining > 15 and best_overall is not None and len(best_overall) == n_p:
            n_cd = min(500, int(t_remaining * 30))
            print(f"\n[Phase 1c: CoordDescent polish, n_cd={n_cd}]")
            f_opt, score = coordinate_descent(best_overall, n_iter=n_cd)
            update_best(f_opt, score, f"CD polish n={n_p}")
    
    # === Phase 2: Try different n values and starting points ===
    n_candidates = [256, 320, 384, 512]
    if parent is not None and len(parent) in n_candidates:
        n_candidates.remove(len(parent))
        n_candidates.insert(0, len(parent))
    
    for n in n_candidates:
        t_remaining = deadline - time.monotonic()
        if t_remaining < 80:
            break
        
        t_per_n = min(350, t_remaining * 0.35)
        if t_per_n < 50:
            break
        
        # Generate diverse starting points
        starts = []
        names = []
        
        s = np.ones(n) * (1 + 0.1 * rng.standard_normal(n))
        starts.append(np.clip(s, 0.01, None))
        names.append("perturbed_uniform")
        
        s = np.arange(1, n + 1, dtype=np.float64)
        starts.append(s)
        names.append("ramp")
        
        s = np.abs(rng.standard_normal(n)) + 0.1
        starts.append(s)
        names.append("random")
        
        x = np.arange(n, dtype=np.float64)
        s = np.exp(-0.5 * ((x - n / 2) / (n / 5)) ** 2)
        starts.append(s)
        names.append("gaussian")
        
        s = 1.0 / (1.0 + ((x - n / 2) / (n / 8)) ** 2)
        starts.append(s)
        names.append("lorentzian")
        
        # Chirp-like
        freq = 3.0 / n
        s = 0.5 + 0.5 * np.cos(2 * np.pi * freq * x + np.pi * freq * x * x / n)
        starts.append(np.abs(s) + 0.01)
        names.append("chirp")
        
        # If we have a best from another n, try resampling
        if best_overall is not None:
            s = np.interp(np.linspace(0, len(best_overall) - 1, n),
                         np.arange(len(best_overall)), best_overall)
            starts.append(s + 0.01)
            names.append("resampled_best")
        
        n_starts = len(starts)
        t_per_start = t_per_n / n_starts
        
        for i, (a0, name) in enumerate(zip(starts, names)):
            t_remaining = deadline - time.monotonic()
            if t_remaining < 30:
                break
            t_here = min(t_per_start, t_remaining * 0.4)
            if t_here < 20:
                continue
            
            a0 = np.clip(a0, 0.001, None).astype(np.float64)
            a0 = project_simplex(a0)
            init_score = score_normalized(a0)
            
            print(f"\n[n={n}, start={name}, t={t_here:.0f}s, init_score={init_score:.4f}]")
            
            # Smooth-max first (faster iterations)
            t_sm = t_here * 0.5
            f_opt, score = optimize_smooth_max(a0, t_sm, T_start=5.0, T_end=150.0, lr_init=0.005)
            label = f"SM n={n} {name}"
            update_best(f_opt, score, label)
            
            # LP refinement
            t_remaining = deadline - time.monotonic()
            t_lp = min(t_here * 0.3, t_remaining * 0.3)
            if t_lp > 15 and np.isfinite(score):
                f_opt, score = optimize_lp(f_opt, t_lp, t0=0.3, verbose=False, rng=rng)
                update_best(f_opt, score, f"LP n={n} {name}")
            
            # Coordinate descent
            t_remaining = deadline - time.monotonic()
            if t_remaining > 10 and np.isfinite(score):
                n_cd = min(300, int(t_remaining * 40))
                f_opt, score = coordinate_descent(f_opt, n_iter=n_cd)
                update_best(f_opt, score, f"CD n={n} {name}")
            
            # Random local search
            t_remaining = deadline - time.monotonic()
            if t_remaining > 10 and np.isfinite(score):
                n_iter_rs = min(3000, int(t_remaining * 500))
                f_opt, score = random_local_search(f_opt, n_iter=n_iter_rs, rng=rng)
                update_best(f_opt, score, f"RS n={n} {name}")
    
    # === Phase 3: Final polishing ===
    if best_overall is not None:
        t_remaining = deadline - time.monotonic()
        if t_remaining > 10:
            n_best = len(best_overall)
            print(f"\n[Phase 3: Final polishing, n={n_best}, t={t_remaining:.0f}s]")
            
            # Extended coordinate descent
            n_cd = min(2000, int(t_remaining * 50))
            f_opt, score = coordinate_descent(best_overall, n_iter=n_cd)
            update_best(f_opt, score, "final CD")
            
            # More smooth-max with high T
            t_remaining = deadline - time.monotonic()
            if t_remaining > 20:
                f_opt, score = optimize_smooth_max(f_opt, min(t_remaining * 0.5, 100),
                                                   T_start=50.0, T_end=500.0, lr_init=0.002)
                update_best(f_opt, score, "final SM high-T")
            
            # Final coordinate descent
            t_remaining = deadline - time.monotonic()
            if t_remaining > 5:
                n_cd = min(500, int(t_remaining * 40))
                f_opt, score = coordinate_descent(f_opt, n_iter=n_cd)
                update_best(f_opt, score, "final CD2")
    
    # === Return best ===
    if best_overall is not None:
        best_overall = np.clip(best_overall, 0, 1000)
        final_list = best_overall.tolist()
        final_score = evaluate_sequence(final_list)
        print(f"\n=== FINAL ===")
        print(f"Verified score: {final_score:.6f}")
        print(f"Sequence length: {len(final_list)}")
        print(f"Sum: {sum(final_list):.4f}")
        print(f"Max: {max(final_list):.4f}")
        return final_list
    
    # Fallback
    print("Fallback: returning parent or uniform")
    if parent is not None:
        return np.clip(parent, 0, 1000).tolist()
    return [1.0] * 256
