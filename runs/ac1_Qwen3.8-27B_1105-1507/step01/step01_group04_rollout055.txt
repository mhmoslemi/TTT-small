"""
Optimizes non-negative step function heights to minimize 2n*max(conv(f,f))/(sum(f))^2.
Uses multi-start mirror descent with FFT, coordinate descent refinement,
and adaptive time budgeting across different sequence lengths.
"""

import numpy as np
import time

def _fft_convolve(f):
    """Compute full linear convolution of f with itself using FFT."""
    n = len(f)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    F = np.fft.rfft(f, size)
    h = np.fft.irfft(F * F, size)
    return h[:2 * n - 1]

def _score(f):
    """Compute score 2n*max(conv(f,f))/(sum f)^2. Assumes f >= 0, sum > 0."""
    n = len(f)
    h = _fft_convolve(f)
    S = float(np.sum(f))
    if S < 0.01:
        return float('inf')
    M = float(np.max(h))
    return 2.0 * n * M / (S * S)

def _normalize(f):
    """Normalize to sum=1, clip negatives."""
    f = np.maximum(f, 0.0)
    s = float(np.sum(f))
    if s < 1e-30:
        return None
    return f / s

def _init_uniform(n):
    return np.ones(n) / n

def _init_cosine(n):
    j = np.arange(n, dtype=np.float64)
    f = np.cos(np.pi * (j + 0.5) / n) ** 2
    f = np.maximum(f, 0.0)
    return _normalize(f)

def _init_gaussian(n, sigma_frac=0.3):
    j = np.arange(n, dtype=np.float64) - (n - 1) / 2.0
    sigma = sigma_frac * n
    f = np.exp(-j * j / (2.0 * sigma * sigma))
    return _normalize(f)

def _init_tapered(n, p=1.0):
    j = np.arange(n, dtype=np.float64)
    f = np.sin(np.pi * (j + 0.5) / n) ** p
    f = np.maximum(f, 0.0)
    return _normalize(f)

def _init_random(n, rng):
    f = rng.random(n, dtype=np.float64)
    return _normalize(f)

def _mirror_descent(f_init, n_iter, eta_scale=0.5, thresh_frac=0.7, time_budget=30.0):

    n = len(f)

    while size < 2 * n - 1:
        size <<= 1

    h = np.fft.irfft(F * F, size)
    return h[:2 * n - 1]

    n = len(f)
    h = _fft_convolve(f)
    S = float(np.sum(f))

        return float('inf')
