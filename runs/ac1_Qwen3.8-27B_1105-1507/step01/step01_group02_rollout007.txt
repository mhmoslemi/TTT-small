"""
Multi-strategy optimizer for minimizing 2*n*max(convolve(a,a))/sum(a)^2.
Uses: (1) L2-smoothed objective with Nesterov momentum, (2) logsumexp max approximation,
(3) LP-based direction finding, (4) adaptive step sizing, (5) diverse initializations,
(6) multiple sequence lengths. Combines smooth and non-smooth phases for robustness.

import numpy as np
import time
from scipy.optimize import linprog, minimize
from scipy.signal import correlate as sig_correlate


def project_simplex(v):
    """Project vector v onto the probability simplex."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cv = np.cumsum(u) - 1.0
    rho_candidates = np.where(u * np.arange(1, n + 1) > cv)[0]
    if len(rho_candidates) == 0:
        return np.ones(n) / n
    rho = rho_candidates[-1]
    theta = cv[rho] / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def fast_convolve(a):
    """Compute full linear convolution of a with itself using FFT."""
    n = len(a)
    size = 2 * n - 1
    N = 1 << (size - 1).bit_length()
    fa = np.fft.rfft(a, N)
    fc = fa * fa
    result = np.fft.irfft(fc, N)[:size]
    result[result < 0] = 0.0
    return result


def compute_l2_grad(a):
    """Compute gradient of ||convolve(a,a)||^2 w.r.t. a. Returns (grad, conv)."""
    n = len(a)
    c = fast_convolve(a)
    # g[i] = 2 * sum_k c[k] * a[k-i] for valid indices
    # = 2 * scipy.signal.correlate(c, a, 'valid')[i]
    g = 2.0 * sig_correlate(c, a, mode='valid')
    return g, c


def compute_logsumexp_grad(a, beta):
    """Compute gradient of logsumexp(beta * convolve(a,a)) w.r.t. a."""
    n = len(a)
    c = fast_convolve(a)
    # Softmax weights
    max_c = np.max(c)
    exp_c = np.exp(beta * (c - max_c))
    weights = exp_c / exp_c.sum()
    # grad[i] = beta * sum_k weights[k] * a[k-i]
    # = beta * correlate(weights, a, 'valid')
    g = beta * sig_correlate(weights, a, mode='valid')
    return g, c


def compute_subgradient_max(a, c, top_k=20):
    """Subgradient of max(c) w.r.t. a, averaged over top peaks."""
    n = len(a)
    peak = np.max(c)
    threshold = peak * (1.0 - 1e-4)
    peak_indices = np.where(c >= threshold)[0]
    if len(peak_indices) > top_k:
        top_idx = np.argsort(c[peak_indices])[::-1][:top_k]
        peak_indices = peak_indices[top_idx]
    grad = np.zeros(n)
    for k in peak_indices:
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        if lo <= hi:
            grad[lo:hi + 1] += 2.0 * a[k - hi:k - lo + 1][::-1]
    grad /= max(1, len(peak_indices))
    return grad


def alt_projection_init(n, rng, n_iter=80):
    """Alternating projection for flat-spectrum nonnegative initialization."""
    N = 1 << (2 * n - 2).bit_length()
    phases = rng.uniform(0.0, 2.0 * np.pi, N // 2 + 1)
    F = np.exp(1j * phases)
    F[0] = 1.0 + 0j

    a = np.ones(n) / n
    for it in range(n_iter):
        a_full = np.fft.irfft(F, N)
        a = a_full[:n].copy()
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s < 1e-10:
            a = np.ones(n) / n
        else:
            a = a / s
        F_full = np.fft.rfft(a, N)
        weights = np.linspace(1.0, 0.7, len(F_full))
        mag_target = weights ** 0.1
        F = mag_target * np.exp(1j * np.angle(F_full))
        F[0] = F[0].real + 0j
    return a

        a_new = project_simplex(a + v)

def lp_improve(a, deadline):
    """LP-based improvement: find direction b that reduces peak while maintaining sum."""
    n = len(a)
    if n > 3000:
        return None

    c = fast_convolve(a)
    P = np.max(c)
    a_sum = a.sum()

    # Build convolution matrix
    A = np.zeros((2 * n - 1, n), dtype=np.float64)
    for k in range(2 * n - 1):
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        A[k, lo:hi + 1] = a[k - hi:k - lo + 1][::-1]

    c_obj = -np.ones(n)
    b_ub = np.full(2 * n - 1, P)

    try:
        res = linprog(c_obj, A_ub=A, b_ub=b_ub,
                      bounds=[(0.0, None)] * n,
                      method='highs',
                      options={'time_limit': 10.0})
        if res.success:
            b = res.x
            b_sum = b.sum()
            if b_sum > 1e-8:
                b = b * (a_sum / b_sum)
                # Verify improvement
                c_b = fast_convolve(b)
                if np.max(c_b) < P * 0.99:
                    return b
    except Exception:
        pass
    return None

def weighted_l2_optimize(a, deadline, rng):

            print(f"n={n}, init={init_type}, peak={peak:.6f}, sum={s:.4f}, "
                  f"score={score:.6f}, time_left={overall_deadline - time.time():.0f}s")

            if np.isfinite(score) and score < best_score:
                best_score = score
                best_seq = a_opt.copy()
                print(f"  >>> New best: {best_score:.6f} (n={n})")
        except Exception as e:
            print(f"  Error in config ({n}, {init_type}): {e}")
            continue

    # Final polish on best sequence with remaining time
    if best_seq is not None and time.time() < overall_deadline - 10:
        print(f"Polishing best sequence (n={len(best_seq)}, score={best_score:.6f})...")
        polish_deadline = overall_deadline
        n_best = len(best_seq)

        # Multiple polish passes
        a_polished = best_seq.copy()
        for polish_round in range(3):
            if time.time() > polish_deadline - 10:
                break
            phase_deadline = min(polish_deadline - 10, time.time() + 30)
            obj = 'l2' if polish_round % 2 == 0 else 'logsumexp'
            a_p = momentum_optimize(a_polished, phase_deadline, rng, objective=obj,
                                    beta=8.0 if obj == 'logsumexp' else 5.0)
            c_p = fast_convolve(a_p)
            peak_p = np.max(c_p)
            if peak_p < np.max(fast_convolve(a_polished)):
                a_polished = a_p

        if n_best <= 2048 and time.time() < polish_deadline - 15:
            b = lp_improve(a_polished, polish_deadline)
            if b is not None:
                for t_mix in [0.2, 0.4, 0.6]:
                    a_mix = project_simplex((1 - t_mix) * a_polished + t_mix * b)
                    c_mix = fast_convolve(a_mix)
                    if np.max(c_mix) < np.max(fast_convolve(a_polished)):
                        a_polished = a_mix
                        break

        if time.time() < polish_deadline - 5:
            a_polished = coordinate_perturbation(a_polished, polish_deadline - 5, rng, n_tries=8000)

        score = evaluate_sequence(a_polished.tolist())
        print(f"After final polish: score={score:.6f}")
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_seq = a_polished

    # Fallback
    if best_seq is None:
        print("Falling back to alt_proj initialization")
        best_seq = alt_projection_init(512, rng, n_iter=80)

    result = [float(x) for x in best_seq]
    final_score = evaluate_sequence(result)
    print(f"FINAL: score={final_score:.6f}, length={len(result)}")
    return result
