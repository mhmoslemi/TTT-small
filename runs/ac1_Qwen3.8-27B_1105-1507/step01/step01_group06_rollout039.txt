**Spectral approach:**
The idea is to make the power spectrum |A(ω)|^2 as concentrated at ω=0 as possible (which corresponds to a flat autocorrelation).

Algorithm:
1. Compute A = FFT(a, N)
2. Compute |A|^2
3. The "target" is to have all the power at ω=0: |A_target[0]| = sum(a), |A_target[k]| = 0 for k > 0
4. But this is achievable only if a is a constant (which gives a triangular autocorrelation, not flat)
5. Instead, use a "soft" target: |A_target[k]| = max(|A[k]| * (1 - t), |A[0]| / N * t) for some mixing parameter t
6. Update: A_new = |A_target| * A / |A| (preserve phase)
7. a_new = IFFT(A_new).real
8. Project: a_new = max(a_new, 0), normalize

"""
Minimizes J(a) = 2*n*max(convolve(a,a))/(sum(a))^2 using:
1. Fast subgradient descent with multiplicative weights (mirror descent)
   - O(n log n) per iteration via FFT-accelerated subgradient computation
   - Adaptive step size with restarts
2. L-BFGS-B gradient refinement for local optimization
3. Multiple sequence lengths and diverse initializations
The subgradient uses the active set of near-maximum lags and computes
the gradient vector via a single FFT-based convolution.
"""

import numpy as np
from scipy.optimize import minimize
import time

def fast_conv(a):
    """Full self-convolution via FFT, O(n log n)."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]

def fast_conv_pair(a, b):
    """Full convolution of a and b via FFT."""
    n = len(a)
    m = len(b)
    size = 1
    while size < n + m - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    return np.fft.irfft(fa * fb, size)[:n + m - 1]

def objective_normalized(a):
    """Compute max(conv(a,a)) assuming sum(a)=1."""
    b = fast_conv(a)
    return float(b.max())

def subgradient_update(a, lr, n_active=10):
    """
    One subgradient step with multiplicative weights.
    Uses only the top n_active lags for the active set.
    Returns updated a.
    """
    n = len(a)
    b = fast_conv(a)
    max_b = float(b.max())
    
    # Select top n_active lags
    if n_active < len(b):
        # Get indices of top n_active
        top_idx = np.argsort(b)[-n_active:]
        S = top_idx
    else:
        S = np.arange(len(b))
    
    # Build indicator
    h = np.zeros(2 * n - 1)
    h[S] = 1.0
    
    # Compute subgradient: g[m] = (2/|S|) * (a_rev * h)[m+n-1]
    a_rev = a[::-1]
    conv_ah = fast_conv_pair(a_rev, h)
    g = (2.0 / len(S)) * conv_ah[n - 1 : 2 * n - 1]
    
    # Weighted mean of g (Lagrange multiplier for sum constraint)
    lam = float(np.dot(g, a))
    # (since sum(a) = 1)
    
    # Multiplicative update
    diff = g - lam
    # Clip to avoid overflow
    diff = np.clip(diff, -50.0 / max(lr, 1e-10), 50.0 / max(lr, 1e-10))
    a_new = a * np.exp(-lr * diff)
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s < 1e-15:
        return a
    a_new /= s
    return a_new

def subgradient_descent(a0, max_iters=5000, lr0=0.1, decay=0.9995, n_active=20, tol=1e-12):
    """
    Run subgradient descent with multiplicative weights.
    """
    n = len(a0)
    a = a0.copy()
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return a
    a /= s
    
    best_a = a.copy()
    best_val = objective_normalized(a)
    
    for it in range(max_iters):
        a_new = subgradient_update(a, lr0 * (decay ** it), n_active)
        
        if not np.all(np.isfinite(a_new)):
            break
        
        val = objective_normalized(a_new)
        
        if val < best_val - tol:
            best_val = val
            best_a = a_new.copy()
        
        a = a_new
    
    return best_a

"""
Minimizes J(a) = 2*n*max(convolve(a,a))/(sum(a))^2 using:
1. Fast subgradient descent with multiplicative weights (mirror descent)
   - O(n log n) per iteration via FFT-accelerated subgradient
   - Adaptive step size with backtracking
2. Smooth log-sum-exp objective with analytical gradient for L-BFGS-B refinement
3. Multiple sequence lengths (256-8192) and diverse initializations
The subgradient uses the active set of near-maximum lags; the smooth version
uses temperature-weighted lags for a differentiable surrogate.
"""

import numpy as np
from scipy.optimize import minimize
import time


def fast_conv(a):
    """Full self-convolution via FFT, O(n log n)."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]


def fast_conv_pair(a, b):
    """Full linear convolution of a and b via FFT."""
    n = len(a)
    m = len(b)
    size = 1
    while size < n + m - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    return np.fft.irfft(fa * fb, size)[:n + m - 1]


def compute_subgradient(a, S):
    """
    Compute subgradient of max_{k in S} (a*a)_k.
    g[m] = (2/|S|) * sum_j a[j] * 1[m+j in S]
    Uses FFT: g[(2/|S|)] * (a_rev * h)[m+n-1]
    """
    n = len(a)
    h = np.zeros(2 * n - 1)
    h[S] = 1.0
    a_rev = a[::-1]
    conv_ah = fast_conv_pair(a_rev, h)
    g = (2.0 / len(S)) * conv_ah[n - 1:2 * n - 1]
    return g


def compute_smooth_gradient(a, b, temperature):
    """
    Compute gradient of log-sum-exp(b / temperature) / (sum a)^2.
    Returns gradient wrt a (unnormalized), assuming a > 0.
    """
    n = len(a)
    # Soft weights: w_k = exp(b_k / T) / sum(exp(b_k / T))
    shifted = b / temperature - (b.max() / temperature)  # numerical stability
    exp_vals = np.exp(shifted)
    w = exp_vals / exp_vals.sum()
    
    # Gradient of log-sum-exp(b/T) wrt a_m:
    # = (1/T) * sum_k w_k * 2 * a_{k-m} * 1[0<=k-m<n]
    # = (2/T) * (a_rev * w)[m+n-1]
    a_rev = a[::-1]
    conv_aw = fast_conv_pair(a_rev, w)
    grad_lse = (2.0 / temperature) * conv_aw[n - 1:2 * n - 1]
    
    return grad_lse


def smooth_objective(a, temperature):
    """Compute log-sum-exp(b/T) - log(2n-1) as smooth surrogate for max(b)."""
    n = len(a)
    b = fast_conv(a)
    shifted = b / temperature - (b.max() / temperature)
    return float(np.log(np.exp(shifted).sum()) + b.max() / temperature - np.log(2 * n - 1))


def subgradient_descent(a0, max_iters, lr0, decay, n_active, deadline):
    """
    Subgradient descent with multiplicative weights and adaptive step size.
    """
    n = len(a0)
    a = a0.copy()
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return a
    a /= s
    
    best_a = a.copy()
    best_val = float(fast_conv(a).max())
    lr = lr0
    
    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        
        b = fast_conv(a)
        max_b = float(b.max())
        
        # Select active set (top n_active lags)
        if n_active < len(b):
            S = np.argsort(b)[-n_active:]
        else:
            S = np.arange(len(b))
        
        g = compute_subgradient(a, S)
        lam = float(np.dot(g, a))  # since sum(a)=1
        diff = g - lam
        
        # Adaptive step: scale by inverse magnitude of diff
        max_diff = float(np.abs(diff).max())
        if max_diff < 1e-15:
            break
        effective_lr = lr / max_diff
        effective_lr = min(effective_lr, 2.0)  # cap to avoid explosion
        
        a_new = a * np.exp(-effective_lr * diff)
        a_new = np.maximum(a_new, 0.0)
        sn = a_new.sum()
        if sn < 1e-15:
            break
        a_new /= sn
        
        new_val = float(fast_conv(a_new).max())
        
        if new_val < best_val:
            best_val = new_val
            best_a = a_new.copy()
        
        a = a_new
        lr *= decay
    
    return best_a


def smooth_lbfgs(a0, temperature, maxiter, deadline):
    """L-BFGS-B on the smooth log-sum-exp surrogate."""
    n = len(a0)
    
    def obj_and_grad(x):
        ax = np.maximum(x, 0.0)
        s = ax.sum()
        if s < 1e-8:
            return 1e10, np.zeros(n)
        ax_n = ax / s
        b = fast_conv(ax_n)
        
        # Value: smooth approximation of max(b)
        shifted = b / temperature - b.max() / temperature
        lse = np.log(np.exp(shifted).sum()) + b.max() / temperature
        val = float(lse)
        
        # Gradient of val wrt x
        exp_vals = np.exp(shifted)
        w = exp_vals / exp_vals.sum()
        a_rev = ax_n[::-1]
        conv_aw = fast_conv_pair(a_rev, w)
        grad_lse = (2.0 / temperature) * conv_aw[n - 1:2 * n - 1]
        
        # Chain rule for normalization: d/d(ax) [f(ax/s)] = (grad_f - f'/s * s) / s
        # More carefully: let u = ax, f(u) = lse(conv(u/s, u/s))
        # df/du_m = (1/s) * d/da_m lse(conv(a_n, a_n)) - (2*lse/s^2) * s ... 
        # Actually: f(ax) = g(ax / sum(ax)) where g(a) = lse(conv(a,a))
        # df/d(ax)_m = sum_j (dg/da_j) * d(a_j/s)/d(ax_m)
        # = sum_j (dg/da_j) * (delta_{jm} * s - a_j) / s^2
        # = (dg/da_m)/s - (sum_j dg/da_j * a_j) / s^2
        # = (dg/da_m - dot(dg/da, a)) / s
        # where dg/da is the gradient of lse wrt a (the unnormalized variable)
        
        # dg/da_m = (2/T) * sum_k w_k * a_{k-m} = grad_lse[m] (computed above with a_n = ax/s)
        # But we need it wrt the unnormalized a... 
        # Actually, since g(a) = lse(conv(a/s, a/s)) and we want d/d(ax):
        # g(ax) = lse(conv(ax/s, ax/s))
        # Let's just compute: d/d(ax_m) g(ax)
        # = sum_j (d/d(a_j)) lse(conv(a,s)) |_{a=ax/s} * d(a_j)/d(ax_m)
        # where a_j = ax_j / s
        # d(a_j)/d(ax_m) = (delta_{jm} * s - ax_j) / s^2 = (delta_{jm} - a_j) / s
        # So d/d(ax_m) g = sum_j grad_g_j * (delta_{jm} - a_j) / s
        # = (grad_g_m - sum_j grad_g_j * a_j) / s
        
        # grad_g is the gradient of lse(conv(a_n, a_n)) wrt a_n, which is grad_lse
        grad_g = grad_lse
        sum_grad_a = float(np.dot(grad_g, ax_n))
        grad = (grad_g - sum_grad_a * ax_n) / s
        
        # Project: zero out gradient where x <= 0
        grad = np.where(x > 0, grad, 0.0)
        
        return val, grad
    
    res = minimize(obj_and_grad, a0, method='L-BFGS-B',
                   bounds=[(0, None)] * n,
                   options={'maxiter': maxiter, 'max_fun': maxiter * 5,
                            'ftol': 1e-16, 'gtol': 1e-12, 'disp': False})
    if res.fun < 1e10:
        return res.x
    return a0

        print(f"  L-BFGS-B final refinement: n={n}, budget={remaining:.0f}s", flush=True)
        print(f"{'='*50}", flush=True)
        
        # Try multiple temperatures (annealing)
        temperatures = [0.1 / n, 0.05 / n, 0.02 / n, 0.01 / n, 0.005 / n]
        
        for temp in temperatures:
            if time.monotonic() > deadline:
                break
            a_curr = best / best.sum()
            for restart in range(3):
                if time.monotonic() > deadline:
                    break
                a_start = a_curr.copy()
                if restart > 0:
                    a_start = a_start * (1 + 0.05 * rng.standard_normal(n))
                    a_start = np.maximum(a_start, 0)
                    a_start /= a_start.sum()
                
                refined = smooth_lbfgs(a_start, temp, maxiter=500, deadline=deadline)
                try_update(refined)
                if best is not None:
                    a_curr = best / best.sum()
        
        # Final subgradient polish
        if time.monotonic() < deadline:
            a = best / best.sum()
            for it in range(500):
                if time.monotonic() > deadline:
                    break
                b = fast_conv(a)
                max_b = float(b.max())
                n_act = max(5, n // 50)
                S = np.argsort(b)[-n_act:]
                g = compute_subgradient(a, S)
                lam = float(np.dot(g, a))
                diff = g - lam
                max_d = float(np.abs(diff).max())
                if max_d < 1e-15:
                    break
                lr_eff = min(0.5 / max_d, 1.0)
                a_new = a * np.exp(-lr_eff * diff)
                a_new = np.maximum(a_new, 0)
                sn = a_new.sum()
                if sn < 1e-15:
                    break
                a = a_new / sn
            try_update(a)

    if best is None:
        best = np.ones(256)

    print(f"\n{'='*50}", flush=True)
    print(f"  FINAL RESULT: J={best_score:.6f}, n={best.size}", flush=True)
    print(f"  Total time: {time.monotonic()-t_start:.1f}s", flush=True)
    print(f"{'='*50}", flush=True)
    
    return [float(x) for x in best]
