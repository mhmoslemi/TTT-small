import numpy as np
import time
from scipy.optimize import linprog, minimize

def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    N = 1 << (size - 1).bit_length()
    fa = np.fft.rfft(a, N)
    fc = fa * fa
    result = np.fft.irfft(fc, N)[:size]
    result[result < 0] = 0.0
    return result

def project_simplex(v):
    n = len(v)
    u = np.sort(v)[::-1]
    cv = np.cumsum(u) - 1.0
    rho_candidates = np.where(u * np.arange(1, n + 1) > cv)[0]
    if len(rho_candidates) == 0:
        return np.ones(n) / n
    rho = rho_candidates[-1]
    theta = cv[rho] / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def nesterov_subgradient(a, deadline, rng, max_iters=200000):
    """Nesterov accelerated projected subgradient descent."""
    n = len(a)
    a = project_simplex(a.copy().astype(np.float64))
    y = a.copy()  # Nesterov momentum variable
    alpha = 1.0  # momentum
    
    c = fast_convolve(a)
    best_peak = np.max(c)
    best_a = a.copy()
    
    lr = 1.0 / np.sqrt(n)
    lr_schedule = 'nesterov'
    
    stall = 0
    t = 0
    t_start = time.time()
    
    while time.time() < deadline and t < max_iters:
        t += 1
        
        # Compute gradient at y
        c_y = fast_convolve(y)
        peak = np.max(c_y)
        
        # Active set
        threshold = peak * (1 - 1e-6)
        active = np.where(c_y >= threshold)[0]
        if len(active) > 20:
            # Keep only top 10
            top = np.argsort(c_y[active])[::-1][:10]
            active = active[top]
        
        # Subgradient
        grad = np.zeros(n)
        for k in active:
            lo = max(0, k - n + 1)
            hi = min(n - 1, k)
            if lo <= hi:
                grad[lo:hi+1] += 2.0 * y[k-hi:k-lo+1][::-1]
        grad /= max(1, len(active))
        
        # Nesterov update
        t_curr = t
        alpha_new = (1 + np.sqrt(1 + 4*alpha**2)) / 2
        theta = (1 - alpha) / alpha_new
        alpha = alpha_new
        
        # Step
        step = lr * grad
        a_new = y - step
        a_new = project_simplex(a_new)
        
        # Nesterov extrapolation
        y = a_new + theta * (a_new - a)
        y = project_simplex(y)
        
        a = a_new
        
        # Check improvement
        c_a = fast_convolve(a)
        peak_a = np.max(c_a)
        if peak_a < best_peak - 1e-10:
            best_peak = peak_a
            best_a = a.copy()
            stall = 0
        else:
            stall += 1
        
        # Adaptive lr
        if stall > 500:
            lr *= 0.8
            stall = 0
            
    return best_a
