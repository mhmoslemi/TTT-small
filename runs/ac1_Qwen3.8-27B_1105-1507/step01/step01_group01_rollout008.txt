            cvec = -np.ones(n)
            bvec = np.full(2 * n - 1, M * 1.001)
            bounds = [(0, 1)] * n
            
            res = linprog(cvec, A_ub=A, b_ub=bvec, bounds=bounds,
                         method='highs', options={'presolve': True})
            
            if not res.success:
                stall += 1
                continue
            
            g = np.clip(res.x, 0, 1)
            if g.sum() < 1e-10:
                stall += 1
                continue
            
            if sym:
                g = sym_normalize(g)
            else:
                g = normalize(g)
            
            # Try interpolation
            new_best = best
            new_sc = best_sc
            for t in ts_list:
                cand = (1 - t) * best + t * g
                if sym:
                    cand = sym_normalize(cand)
                else:
                    cand = normalize(cand)
                sc = local_score(cand)
                if np.isfinite(sc) and sc < new_sc:
                    new_sc = sc
                    new_best = cand
