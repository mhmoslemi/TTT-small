Improved optimization for minimizing 2*n*max(conv(a,a))/(sum(a))^2.
Combines LP-based alternating minimization with subgradient local search,
FFT-accelerated objective computation, and diverse initializations.
"""

import numpy as np
from scipy.optimize import linprog
import time

def compute_obj(a):
    """Compute objective using FFT for speed."""
    n = len(a)
    if n == 0:
        return np.inf
    sum_a = float(np.sum(a))
    if sum_a < 0.01:
        return np.inf
    # FFT-based linear convolution
    size = 2 * n - 1
    fa = np.fft.rfft(a, size)
    conv = np.fft.irfft(fa * fa, size)[:n + n - 1]  # Actually we need all 2n-1 elements
    # Wait, irfft with size=2n-1 gives exactly 2n-1 elements
    conv = np.fft.irfft(fa * fa, 2 * n - 1)
    max_b = float(np.max(conv))
    return 2.0 * n * max_b / (sum_a * sum_a)

def compute_obj_fft(a):
    """Use FFT to compute the objective."""
    n = len(a)
    if n == 0:
        return np.inf
    sum_a = float(np.sum(a))
    if sum_a < 0.01:
        return np.inf
    size = 2 * n - 1
    fa = np.fft.rfft(a, size)
    power = fa * fa.conj()  # For real a, this is just fa**2
    conv = np.fft.irfft(power, size)
    max_b = float(np.max(conv))
    return 2.0 * n * max_b / (sum_a * sum_a)

def build_A(a):
    """Build constraint matrix for LP."""
    n = len(a)
    A = np.zeros((2 * n - 1, n), dtype=np.float64)
    for k in range(2 * n - 1):
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        A[k, j_start:j_end + 1] = a[k - j_end:k - j_start + 1][::-1]
    return A

def solve_lp(a, upper_bound=10.0):
    """Solve LP: max sum(g) s.t. A@g <= M, 0 <= g_i <= ub."""
    n = len(a)
    conv = np.convolve(a, a)
    M = float(np.max(conv))
    A = build_A(a)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, upper_bound)] * n
    res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
    if res.status == 0:
        return res.x, float(-res.fun), M
    return None, None, M

def project_simplex(v):
    """Project v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - 1))[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0)

def subgradient_step(a, step):
    """One subgradient step for minimizing max(conv(a,a)) subject to a on simplex."""
    n = len(a)
    conv = np.convolve(a, a)
    k_star = int(np.argmax(conv))
    
    # Compute subgradient
    g = np.zeros(n)
    if k_star == 0:
        g = 2.0 * a
    else:
        # ∂r_k/∂a_j = a[j+k] (if j+k<n) + a[j-k] (if j-k>=0)
        for j in range(n):
            if j + k_star < n:
                g[j] += a[j + k_star]
            if j - k_star >= 0:
                g[j] += a[j - k_star]
    
    a_new = a - step * g
    a_new = project_simplex(a_new)
    return a_new, k_star

def run_subgradient(a, max_iter=20000, verbose=False):
    """Run subgradient descent with decreasing step size."""
    n = len(a)
    a_curr = a.copy()
    a_curr = project_simplex(a_curr)
    best_a = a_curr.copy()
    best_obj = compute_obj(a_curr)
    
    # Running average
    a_avg = a_curr.copy()
    
    # Polyak step: step_t = c / sqrt(t)
    # We need to estimate c. The Lipschitz constant of the subgradient is bounded by 2*max(a) * 2 = 4*max(a)
    # For a on the simplex, max(a) <= 1, so L <= 4.
    # Use c = 1.0 as starting point
    
    for t in range(1, max_iter + 1):
        step = 1.0 / np.sqrt(t)
        a_curr, k_star = subgradient_step(a_curr, step)
        a_avg = a_avg + (a_curr - a_avg) / t
        
        if t % 1000 == 0 or t <= 5:
            obj_curr = compute_obj(a_curr)
            obj_avg = compute_obj(a_avg)
            if obj_avg < best_obj:
                best_obj = obj_avg
                best_a = a_avg.copy()
            if obj_curr < best_obj:
                best_obj = obj_curr
                best_a = a_curr.copy()
            if verbose:
                print(f"    subgrad t={t}: obj_curr={obj_curr:.6f} obj_avg={obj_avg:.6f} best={best_obj:.6f}")
    
    return best_a, best_obj

def structured_init(n, rng):
    """Generate diverse structured initializations."""
    choice = rng.integers(0, 6)
    if choice == 0:
        # Cosine
        f = rng.uniform(0.5, 5.0)
        phase = rng.uniform(0, 2 * np.pi)
        a = np.maximum(0, 1.0 + np.cos(2 * np.pi * f * np.arange(n) / n + phase))
    elif choice == 1:
        # Two-level periodic
        period = int(rng.integers(2, max(3, n // 10)))
        pattern = rng.random(period)
        a = np.tile(pattern, n // period + 1)[:n]
    elif choice == 2:
        # Exponential decay
        rate = rng.uniform(0.01, 0.1)
        a = np.exp(-rate * np.arange(n))
    elif choice == 3:
        # Random
        a = rng.random(n)
    elif choice == 4:
        # Bimodal
        center1 = int(rng.integers(0, n // 2))
        center2 = int(rng.integers(n // 2, n))
        width = int(rng.integers(n // 10, n // 3))
        a = np.exp(-0.5 * ((np.arange(n) - center1) / width) ** 2) + np.exp(-0.5 * ((np.arange(n) - center2) / width) ** 2)
    else:
        # Triangle wave
        f = rng.uniform(0.5, 3.0)
        a = 1.0 - np.abs((np.arange(n) * f / n) % 1.0 * 2 - 1.0)
    
    s = np.sum(a)
    if s < 1e-10:
        a = rng.random(n)
        s = np.sum(a)
    return a / s

def random_refine(a, rng, max_iter=5000):
    """Random perturbation local search."""
    best_a = a.copy()
    best_obj = compute_obj(a)
    n = len(best_a)
    
    for it in range(max_iter):
        width = max(1, int(n / rng.integers(10, 50)))
        start = int(rng.integers(0, max(1, n - width + 1)))
        candidate = best_a.copy()
        # Multiplicative perturbation
        candidate[start:start + width] *= np.exp(rng.normal(0, 0.1, width))
        candidate = np.clip(candidate, 0, None)
        s = np.sum(candidate)
        if s < 1e-10:
            continue
        candidate = candidate / s
        obj_new = compute_obj(candidate)
        if obj_new < best_obj:
            best_a = candidate
            best_obj = obj_new
    
    return best_a, best_obj
