```python
def simplex_project(v):
    """Project v onto the simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v
    # Sort v in decreasing order
    u = np.sort(v)[::-1]
    # Find rho
    cssv = np.cumsum(u)
    rho = np.where(u * (np.arange(1, n+1) - 1) > cssv - 1)[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho = rho[-1]
    theta = (cssv[rho] - 1) / (rho + 1)
    return np.maximum(v - theta, 0)
```

    f = simplex_project(f.astype(float))
    best_f = f.copy()
    best_score = 2 * n * np.max(np.convolve(f, f))
    start_time = time.monotonic()
    
    for it in range(max_iters):
        if time.monotonic() - start_time > time_budget:
            break
        
        # Compute convolution
        C = np.convolve(f, f)
        k_star = np.argmax(C)
        M = C[k_star]

        sg = np.zeros(n)
        for j in range(n):

                sg[j] += f[k_star - j]
            if 0 <= j - k_star < n:
                sg[j] += f[j - k_star]
        
        # Normalize subgradient
        sg_norm = np.linalg.norm(sg)
        if sg_norm < 1e-15:

        sg /= sg_norm

            best_f = f_new.copy()
        
        f = f_new
    
    return best_f, best_score
