```
def line_search_with_fft(a,q,fa,cp,alphas,n,cur_score,size):
    L=2*n-1
    fq=np.fft.rfft(q,size)
    cq=np.fft.irfft(fq*fq,size)[:L]
    pq=np.fft.irfft(fa*fq,size)[:L]
    cq=np.maximum(cq,0.0); pq=np.maximum(pq,0.0)
    best_s=cur_score; best_alpha=None
    for alpha in alphas:
        oa=1.0-alpha
        c = cp*oa*oa + 2.0*alpha*oa*pq + cq*alpha*alpha
        mx=float(np.max(c))
        if mx > best_s/(2.0*n): # maybe compare
        s=2.0*n*mx
        if s < best_s - 1e-13:
             best_s=s; best_alpha=alpha
    if best_alpha is None: return None
    p=(1.0-best_alpha)*a + best_alpha*q
    p=np.maximum(p,0.0)
    ps=float(p.sum())
    if ps <= 0: return None
    p=p/ps
    return p,best_s
```

```python
"""
Minimizes J(a) = 2*n*max(conv(a,a))/(sum(a))^2 for nonnegative sequences.

Main search idea:
- Work with probability-like sequences sum(a) = 1.
- Use an annealed mirroring multiplicative-weights (mirror descent) step on a
  smoothed maximum convolution objective:
      w_k = softmax(c_k / T),   c = a*a
      cost_i = sum_k w_k a_{k-i}
      q_i  proportional exp(log a_i - beta * cost_i / max(c))
- Mix q into a, with exact line search using FFT convolutions.
- Include an L-BFGS polish on a normalized log-sum-exp smooth surrogate for
  smaller/mid lengths.
- Search multiple lengths, restarting from the parent construction when
  available.
"""

import numpy as np
import time

from scipy.optimize import minimize


def fft_size_for_conv(n):
    n = int(n)
    L = 2 * n - 1
    s = 1
    while s < L:
        s <<= 1
    return s


def conv_full(a):
    a = np.asarray(a, dtype=np.float64)
    n = int(a.size)
    if n == 0:
        return np.zeros(0, dtype=np.float64)
    size = fft_size_for_conv(n)
    fa = np.fft.rfft(a, size)
    c = np.fft.irfft(fa * fa, size)[: 2 * n - 1]
    return np.maximum(c, 0.0)


def quick_score(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    a = np.clip(a, 0.0, 1000.0)
    if a.size == 0:
        return float("inf")
    s = float(a.sum())
    if s < 0.01:
        return float("inf")
    c = conv_full(a)
    mx = float(np.max(c)) if c.size else 0.0
    return float(2.0 * a.size * mx / (s * s))


def normalize_pos(a, min_val=1e-15):
    a = np.asarray(a, dtype=np.float64).ravel()
    a = np.maximum(a, 0.0)
    s = float(a.sum())
    if s <= 0.0:
        a = np.full(a.size, min_val, dtype=np.float64)
        s = float(a.sum())
    a = a / s
    a = np.maximum(a, min_val)
    s2 = float(a.sum())
    if s2 > 0.0:
        a = a / s2
    return a


def softmax_probs(y):
    y = np.asarray(y, dtype=np.float64)
    if y.size == 0:
        return y
    y = y - float(np.max(y))
    e = np.exp(y)
    s = float(e.sum())
    if s <= 0.0 or not np.isfinite(s):
        return np.full(y.shape, 1.0 / y.size, dtype=np.float64)
    return e / s


def upsample_seq(a, n):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = int(n)
    if a.size == 0:
        return np.ones(n, dtype=np.float64) / n
    if a.size == n:
        out = a.copy()
    else:
        x_old = np.linspace(0.0, 1.0, a.size)
        x_new = np.linspace(0.0, 1.0, n)
        out = np.interp(x_new, x_old, a)
    out = np.maximum(out, 1e-12)
    return normalize_pos(out, 1e-15)


def shape_base(n, rng=None):
    x = np.linspace(0.0, 1.0, int(n), endpoint=False)
    v = 1.0 + 0.7 * (x - 0.5) ** 2 + 0.15 * np.cos(6.0 * np.pi * x)
    if rng is not None:
        v = v * np.maximum(rng.normal(1.0, 0.05, n), 0.5)
    return np.maximum(v, 1e-8)


def make_start(n, kind, parent, rng):
    n = int(n)
    if kind == "parent":
        if parent is not None:
            return upsample_seq(parent, n)
        base = np.ones(n, dtype=np.float64)
    elif kind == "jitter":
        if parent is not None:
            base = upsample_seq(parent, n)
        else:
            base = shape_base(n, rng)
        noise = rng.normal(1.0, 0.20, n)
        base = base * np.maximum(noise, 0.20)
    elif kind == "shape":
        base = shape_base(n, rng)
    else:
        base = np.ones(n, dtype=np.float64)
    return normalize_pos(base, 1e-15)


def softmax_weights(vals, temp):
    vals = np.asarray(vals, dtype=np.float64)
    if temp <= 0.0:
        w = np.zeros(vals.shape, dtype=np.float64)
        if vals.size:
            w[int(np.argmax(vals))] = 1.0
        return w
    z = vals / float(temp)
    m = float(np.max(z))
    e = np.exp(z - m)
    s = float(e.sum())
    if s <= 0.0 or not np.isfinite(s):
        return np.full(vals.shape, 1.0 / vals.size, dtype=np.float64)
    return e / s


def correlation_weights(w, p):
    """
    Returns g_i = sum_k w_k p_{k-i} with p_j = 0 outside [0, n-1].
    """
    p = np.asarray(p, dtype=np.float64)
    w = np.asarray(w, dtype=np.float64)
    n = int(p.size)
    L = 2 * n - 1
    size = fft_size_for_conv(n)
    fa = np.fft.rfft(p, size)
    fw = np.fft.rfft(w[::-1], size)
    conv = np.fft.irfft(fa * fw, size)
    if n == 1:
        g = conv[:1]
    else:
        g = conv[L - 1 : L - n - 1 : -1]
    return np.maximum(g, 0.0)


def correlation_from_fft(fa, w, n, size):
    """
    Same as correlation_weights, but reuses fft of p stored in fa.
    """
    n = int(n)
    L = 2 * n - 1
    w = np.asarray(w, dtype=np.float64)
    fw = np.fft.rfft(w[::-1], size)
    conv = np.fft.irfft(fa * fw, size)
    if n == 1:
        g = conv[:1]
    else:
        g = conv[L - 1 : L - n - 1 : -1]
    return np.maximum(g, 0.0)


def line_search_with_fft(a, q, fa, cp, alphas, n, cur_score, size):
    """
    Exact line search for p = (1-alpha)*a + alpha*q, using precomputed FFTs.
    cp = a*a. Returns (p, fixed_score) if any alpha improves cur_score.
    """
    n = int(n)
    L = 2 * n - 1
    fq = np.fft.rfft(q, size)
    cq = np.fft.irfft(fq * fq, size)[:L]
    pq = np.fft.irfft(fa * fq, size)[:L]
    cq = np.maximum(cq, 0.0)
    pq = np.maximum(pq, 0.0)

    best_score = cur_score
    best_alpha = None

    for alpha in alphas:
        one = 1.0 - alpha
        c = cp * one * one + 2.0 * alpha * one * pq + cq * alpha * alpha
        mx = float(np.max(c)) if c.size else 0.0
        s = 2.0 * n * mx
        if s < best_score - 1e-12:
            best_score = s
            best_alpha = alpha

    if best_alpha is None:
        return None

    p = (1.0 - best_alpha) * a + best_alpha * q
    p = np.maximum(p, 0.0)
    ps = float(p.sum())
    if ps <= 0.0 or not np.isfinite(ps):
        return None
    p = p / ps
    return p, best_score


def br_optimize(a0, n, time_limit, rng=None):
    """
    Annealed mirror-descent / best-response optimization.
    """
    n = int(n)
    if rng is None:
        rng = np.random.default_rng(20260601)

    a = normalize_pos(a0, 1e-15)
    size = fft_size_for_conv(n)
    L = 2 * n - 1

    fa = np.fft.rfft(a, size)
    cp = np.fft.irfft(fa * fa, size)[:L]
    cp = np.maximum(cp, 0.0)

    cur = float(2.0 * n * np.max(cp) if cp.size else 0.0)
    best = a.copy()
    best_score = cur

    alphas = (0.5, 0.3, 0.2, 0.1, 0.05, 0.02, 0.01)
    temp_ratios = (0.5, 0.25, 0.1, 0.05, 0.02, 0.01, 0.005, 0.002, 0.001)
    betas = (1.0, 0.25, 3.0)

    if n <= 1024:
        max_it_per_temp = 900
        max_stag = 80
    elif n <= 4096:
        max_it_per_temp = 600
        max_stag = 60
    else:
        max_it_per_temp = 350
        max_stag = 45

    stuck_temps = 0

    for tr in temp_ratios:
        if time.monotonic() > time_limit:
            break

        no_ok = 0
        it = 0
        stage_improved = False

        while it < max_it_per_temp and time.monotonic() < time_limit:
            it += 1

            M = float(np.max(cp)) if cp.size else 0.0
            if M <= 1e-300 or not np.isfinite(M):
                break

            T = tr * M
            w = softmax_weights(cp, T)
            corr = correlation_from_fft(fa, w, n, size)

            loga = np.log(np.maximum(a, 1e-300))
            best_res = None

            for beta in betas:
                rate = beta / M
                logits = loga - rate * corr
                logits = logits - float(np.max(logits))
                logits = np.clip(logits, -50.0, 0.0)
                q = np.exp(logits)
                qs = float(q.sum())
                if qs <= 0.0 or not np.isfinite(qs):
                    continue
                q = q / qs

                # Small uniform support prevents stalling at exact zeros.
                q = 0.90 * q + 0.10 * (1.0 / n)
                q = q / float(q.sum())

                res = line_search_with_fft(a, q, fa, cp, alphas, n, cur, size)
                if res is not None:
                    if best_res is None or res[1] < best_res[1]:
                        best_res = res

            if best_res is not None:
                a, fixed_s = best_res

                # Recompute consistently.
                fa = np.fft.rfft(a, size)
                cp = np.fft.irfft(fa * fa, size)[:L]
                cp = np.maximum(cp, 0.0)
                cur = float(2.0 * n * np.max(cp) if cp.size else 0.0)

                no_ok = 0
                if cur < best_score - 1e-12:
                    best = a.copy()
                    best_score = cur
                    stage_improved = True
            else:
                no_ok += 1
                if no_ok >= max_stag:
                    # Reset to best point and try a few small perturbations.
                    a = best.copy()
                    fa = np.fft.rfft(a, size)
                    cp = np.fft.irfft(fa * fa, size)[:L]
                    cp = np.maximum(cp, 0.0)
                    cur = float(2.0 * n * np.max(cp) if cp.size else 0.0)

                    pert = rng.normal(0.0, 1.0, n)
                    pert -= float(pert.mean())

                    for scale in (0.03, 0.08, 0.15):
                        cand = a + scale * pert
                        cand = np.maximum(cand, 0.0)
                        cs = float(cand.sum())
                        if cs <= 0.0:
                            continue
                        cand = cand / cs
                        fa_c = np.fft.rfft(cand, size)
                        cp_c = np.fft.irfft(fa_c * fa_c, size)[:L]
                        cp_c = np.maximum(cp_c, 0.0)
                        sc = float(2.0 * n * np.max(cp_c) if cp_c.size else 0.0)
                        if np.isfinite(sc) and sc < best_score * 1.02:
                            a = cand
                            fa = fa_c
                            cp = cp_c
                            cur = sc
                            break

                    no_ok = 0
                    break

        if stage_improved:
            stuck_temps = 0
        else:
            stuck_temps += 1
            if stuck_temps >= 3:
                break

    return best, best_score


def smooth_data(y, n, temp):
    p = softmax_probs(y)
    c = conv_full(p)
    L = 2 * int(n) - 1

    if temp <= 0.0:
        w = np.zeros(L, dtype=np.float64)
        if L > 0:
            w[int(np.argmax(c))] = 1.0
        f = float(np.max(c) if c.size else 0.0)
        return p, c, w, f

    z = c / float(temp)
    m = float(np.max(z))
    e = np.exp(z - m)
    s = float(e.sum())
    if s <= 0.0 or not np.isfinite(s):
        w = np.full(L, 1.0 / L, dtype=np.float64)
        f = float(np.max(c) if c.size else 0.0)
    else:
        w = e / s
        # Normalized log-sum-exp: T log( mean exp(c/T) ). This is more favorable
        # to many near-active lags, matching the desired flattening behavior.
        f = float(temp * (m + np.log(s) - np.log(L)))

    return p, c, w, f


def smooth_obj(y, n, temp):
    _, _, _, f = smooth_data(y, n, temp)
    return f


def smooth_grad(y, n, temp):
    p, c, w, _ = smooth_data(y, n, temp)
    g = 2.0 * correlation_weights(w, p)
    mean_g = float(np.dot(p, g))
    grad = p * (g - mean_g)
    return grad


def smooth_lbfgs_optimize(a0, n, time_limit, maxfun=None):
    n = int(n)
    p0 = normalize_pos(a0, 1e-15)
    best = p0.copy()
    best_score = quick_score(p0)

    if maxfun is None:
        if n <= 512:
            maxfun = 140
        elif n <= 1024:
            maxfun = 90
        elif n <= 2048:
            maxfun = 55
        else:
            maxfun = 30

    y = np.log(np.maximum(best, 1e-15))
    y -= float(y.mean())

    if n <= 512:
        stages = (0.2, 0.1, 0.05, 0.02, 0.01, 0.005, 0.001)
    elif n <= 1024:
        stages = (0.15, 0.08, 0.04, 0.02, 0.01, 0.005, 0.001)
    else:
        stages = (0.15, 0.08, 0.04, 0.02, 0.01, 0.005)

    bounds = [(-30.0, 30.0)] * n

    for tr in stages:
        if time.monotonic() > time_limit - 3.0:
            break

        cmax = float(np.max(conv_full(best)) if best.size else 0.0)
        temp = max(tr * cmax, 1e-15)

        res = minimize(
            smooth_obj,
            y.copy(),
            jac=smooth_grad,
            args=(n, temp),
            method="L-BFGS-B",
            bounds=bounds,
            options={
                "maxiter": 90,
                "maxfun": maxfun,
                "ftol": 1e-15,
                "gtol": 1e-10,
                "disp": False,
            },
        )

        if res.x.size == n and np.all(np.isfinite(res.x)):
            pnew = softmax_probs(res.x)
            pnew = np.maximum(pnew, 1e-15)
            pnew = pnew / float(pnew.sum())
            s = quick_score(pnew)
            if np.isfinite(s) and s < best_score:
                best = pnew
                best_score = s
            if np.isfinite(s) and s < best_score * 1.05:
                y = res.x - float(np.max(res.x))

    return best, best_score

            # Extremely defensive fallback.
            result = [1.0 / 512] * 512
    except Exception:
        pass

    return result
```
