```python
"""
Search for low-score non-negative step-function coefficients using:
  - L-BFGS-B optimization with smooth max approximation (log-sum-exp style)
  - Homotopy in the power parameter q (smooth → sharp)
  - Subgradient descent with projection for peak reduction
  - Multiple lengths and random restarts
  - Spectral-inspired initializations
"""

import time
import numpy as np
from scipy.optimize import minimize


def _normalize(a):
    a = np.asarray(a, dtype=np.float64)
    a = np.clip(a, 0.0, 1000.0)
    if not np.isfinite(a).all():
        return None
    s = float(np.sum(a))
    if s < 1e-300:
        return None
    return a / s


def _score(a):
    a = np.asarray(a, dtype=np.float64)
    if a.size == 0:
        return float('inf')
    a = np.clip(a, 0.0, 1000.0)
    s = float(np.sum(a))
    if s < 1e-12:
        return float('inf')
    a = a / s
    c = np.convolve(a, a)
    m = float(np.max(c))
    return 2.0 * len(a) * m


def _convolve_fft(a, b):
    n = a.size + b.size - 1
    sz = 1
    while sz < n:
        sz <<= 1
    fa = np.fft.rfft(a, sz)
    fb = np.fft.rfft(b, sz)
    return np.fft.irfft(fa * fb, sz)[:n]


def _correlate_fft(w, a):
    """Compute np.correlate(w, a, mode='valid') using FFT. w has length >= len(a)."""
    n_w = w.size
    n_a = a.size
    n_out = n_w - n_a + 1
    sz = 1
    while sz < n_w + n_a - 1:
        sz <<= 1
    fw = np.fft.rfft(w, sz)
    # correlate(w, a) = convolve(w, flip(a))
    fa = np.fft.rfft(a[::-1], sz)
    result = np.fft.irfft(fw * fa, sz)
    # The valid part starts at index n_a - 1
    return result[n_a - 1:n_a - 1 + n_out]


def _smooth_obj_grad(x, q):
    """
    Objective and gradient for L-BFGS-B.
    x: non-negative raw vector, a = x/sum(x)
    q: power parameter for smooth max
    """
    n = x.size
    s = float(np.sum(x))
    if s < 1e-150:
        return 1e10, np.zeros(n)
    a = x / s

    c = np.convolve(a, a)
    c_plus = np.maximum(c, 1e-300)

    # T = sum c^q, H = correlate(c^(q-1), a, 'valid')
    cq = c_plus ** q
    T = float(np.sum(cq))
    if T < 1e-300:
        return 1e10, np.zeros(n)

    cqm1 = c_plus ** (q - 1)
    H = _correlate_fft(cqm1, a)

    # grad = (4n * T^(1/q - 1) / s) * (H - T)
    exponent = 1.0 / q - 1.0
    factor = 4.0 * n * (T ** exponent) / s
    grad = factor * (H - T)

    # Clip gradient for numerical stability
    grad = np.clip(grad, -1e6, 1e6)

    obj = 2.0 * n * (T ** (1.0 / q))
    return obj, grad


def _smooth_obj_only(x, q):
    n = x.size
    s = float(np.sum(x))
    if s < 1e-150:
        return 1e10
    a = x / s
    c = np.convolve(a, a)
    c_plus = np.maximum(c, 1e-300)
    T = float(np.sum(c_plus ** q))
    return 2.0 * n * (T ** (1.0 / q))


def _subgradient_step(a, eta, top_k=None):
    """One subgradient descent step on max(conv(a,a))."""
    n = a.size
    c = np.convolve(a, a)
    M = float(np.max(c))
    if M <= 0:
        return a.copy()

    # Find active lags (those near the max)
    if top_k is not None:
        threshold = np.sort(c)[-top_k][0]
        active = np.where(c >= threshold)[0]
    else:
        active = np.where(c >= 0.99 * M)[0]

    # Subgradient: g[j] = 2 * sum_{k in active} a[k-j] for 0<=k-j<n
    g = np.zeros(n)
    for k in active:
        j_start = max(0, k - n + 1)
        j_end = min(k, n - 1)
        g[j_start:j_end + 1] += 2.0 * a[k - j_start:k - j_end + 1][::-1]

    # Normalize subgradient
    gn = float(np.sum(g))
    if gn < 1e-150:
        return a.copy()
    g = g / gn  # normalize so sum = 1 (like a probability)

    # Update: move mass from high-gradient positions to low-gradient positions
    # a_new = (1-eta)*a + eta*(1 - g) ... no, let's do:
    # Step in direction -g on the simplex
    a_new = a - eta * (g - 1.0 / n)
    a_new = np.maximum(a_new, 0.0)
    s_new = float(np.sum(a_new))
    if s_new < 1e-150:
        return a.copy()
    a_new = a_new / s_new
    return a_new


def _subgradient_optimize(a, max_time, seed, eta_init=0.01):
    """Subgradient descent with adaptive step size."""
    t0 = time.monotonic()
    deadline = t0 + max_time
    best = a.copy()
    best_sc = _score(best)
    rng = np.random.default_rng(seed)
    n = best.size
    eta = eta_init
    stuck = 0

    while time.monotonic() < deadline:
        cand = _subgradient_step(best, eta, top_k=max(3, n // 16))
        sc = _score(cand)
        if sc < best_sc:
            best = cand
            best_sc = sc
            stuck = 0
            eta = min(eta * 1.05, 0.1)
        else:
            stuck += 1
            if stuck % 20 == 0:
                eta *= 0.7
                if eta < 1e-5:
                    # Restart with noise
                    noise = np.exp(rng.normal(0, 0.1, n))
                    cand2 = best * noise
                    best = _normalize(cand2)
                    best_sc = _score(best)
                    eta = eta_init
                    stuck = 0

    return best, best_sc


def _lbfgsb_optimize(a, max_time, seed, q_values=None):
    """L-BFGS-B with homotopy in q."""
    t0 = time.monotonic()
    deadline = t0 + max_time
    best = a.copy()
    best_sc = _score(best)
    n = best.size

    if q_values is None:
        q_values = [4, 8, 16, 32, 64, 128, 256, 512]

    # Convert to raw parameter for L-BFGS-B (x = a, with x >= 0)
    x0 = best.copy()
    # Scale so sum is 1 (L-BFGS-B will work on this scale)

    for q in q_values:
        if time.monotonic() >= deadline - 5:
            break

        try:
            res = minimize(
                _smooth_obj-only_wrapper,
                x0,
                args=(q,),
                method='L-BFGS-B',
                jac=True,
                bounds=[(0.0, None)] * n,
                options={'maxiter': 500, 'maxfun': 2000,
                         'ftol': 1e-12, 'gtol': 1e-8,
                         'maxls': 50}
            )
        except Exception:
            break

        a_new = _normalize(res.x)
        if a_new is not None:
            sc = _score(a_new)
            if sc < best_sc:
                best = a_new
                best_sc = sc
                x0 = best.copy()
                print(f"  L-BFGS q={q}: score={sc:.6f}", flush=True)

    return best, best_sc


def _smooth_obj_only_wrapper(x, q):
    """Wrapper that returns both objective and gradient for L-BFGS-B."""
    obj, grad = _smooth_obj_grad(x, q)
    return obj, grad


def _make_trie_tensor(n, seed):
    """Create a random non-negative sequence."""
    rng = np.random.default_rng(seed)
    a = rng.uniform(0.1, 1.0, n)
    return _normalize(a)


def _make_symmetric(n, seed):
    """Create a symmetric random sequence."""
    rng = np.random.default_rng(seed)
    m = (n + 1) // 2
    h = rng.uniform(0.1, 1.0, m)
    if n % 2 == 0:
        a = np.concatenate([h, h[::-1]])
    else:
        a = np.concatenate([h, h[-2::-1]])
    return _normalize(a)


def _spectral_init(n, seed, iters=50):
    """
    Initialize by trying to make the autocorrelation flat.
    Start from uniform and apply gradient steps to flatten c.
    """
    rng = np.random.default_rng(seed)
    a = np.ones(n) / n

    for _ in range(iters):
        c = np.convolve(a, a)
        c_mean = np.mean(c)
        # Gradient of sum (c[k] - c_mean)^2 w.r.t. a
        # = 2 * sum_k (c[k] - c_mean) * dc[k]/da[j]
        # dc[k]/da[j] = 2*a[k-j] for 0<=k-j<n
        residual = c - c_mean
        # grad[j] = 2 * sum_k residual[k] * 2*a[k-j] = 4 * correlate(residual, a)[j]
        # Using correlate: grad[j] = 4 * sum_{l=0}^{n-1} residual[j+l] * a[l]
        grad = _correlate_fft(residual, a) * 4.0
        # Normalize
        gn = np.linalg.norm(grad)
        if gn < 1e-15:
            break
        grad = grad / gn
        eta = 0.01
        a_new = a - eta * grad
        a_new = np.maximum(a_new, 0.0)
        s = np.sum(a_new)
        if s < 1e-15:
            break
        a_new = a_new / s
        a = a_new

    return a


def _peak_targeted_search(a, max_time, seed):
    """
    Targeted search: identify peak lags and try to reduce them
    by adjusting specific pairs of coefficients.
    """
    t0 = time.monotonic()
    deadline = t0 + max_time
    best = a.copy()
    best_sc = _score(best)
    n = best.size
    rng = np.random.default_rng(seed)

    it = 0
    while time.monotonic() < deadline:
        c = np.convolve(best, best)
        # Find the peak lag
        k_peak = int(np.argmax(c))
        peak_val = c[k_peak]

        # Find the dominant contributing pair
        j_start = max(0, k_peak - n + 1)
        j_end = min(k_peak, n - 1)
        js = np.arange(j_start, j_end + 1)
        products = best[js] * best[k_peak - js]
        max_prod_idx = int(np.argmax(products))
        i1 = int(js[max_prod_idx])
        i2 = k_peak - i1

        if i1 == i2:
            # Only one coefficient to adjust
            # Reduce it and add to a random other position
            delta = 0.1 * best[i1]
            idx_add = int(rng.integers(0, n))
            while idx_add == i1:
                idx_add = int(rng.integers(0, n))
            cand = best.copy()
            cand[i1] -= delta
            cand[idx_add] += delta
        else:
            # Reduce the larger one
            if best[i1] >= best[i2]:
                src = i1
                dst = i2
            else:
                src = i2
                dst = i1
            delta = 0.05 * best[src]
            # Add to a position that doesn't contribute to the peak
            # Try a few random positions
            best_cand = None
            best_cand_sc = float('inf')
            for _ in range(8):
                idx_add = int(rng.integers(0, n))
                if idx_add == src or idx_add == dst:
                    continue
                cand = best.copy()
                cand[src] -= delta
                cand[idx_add] += delta
                cand = np.maximum(cand, 0.0)
                s = cand.sum()
                if s < 1e-15:
                    continue
                cand = cand / s
                sc = _score(cand)
                if sc < best_cand_sc:
                    best_cand_sc = sc
                    best_cand = cand

            if best_cand is not None and best_cand_sc < best_sc:
                best = best_cand
                best_sc = best_cand_sc
        it += 1

        if it % 1000 == 0:
            print(f"    peak_search it={it} sc={best_sc:.6f} k_peak={k_peak} peak={peak_val:.6f}", flush=True)

    return best, best_sc


def _crossover(a1, a2, seed):
    """Blend two sequences."""
    rng = np.random.default_rng(seed)
    t = float(rng.uniform(0.2, 0.8))
    a = t * a1 + (1 - t) * a2
    return _normalize(a)


def _refine_with_lbfgsb(a, time_budget, seed):
    """Run L-BFGS-B with increasing q values."""
    t0 = time.monotonic()
    deadline = t0 + time_budget
    n = a.size
    best = a.copy()
    best_sc = _score(best)

    q_values = [4, 8, 16, 32, 64, 128, 256]
    # Distribute time among q values, more time for larger q
    weights = [0.5, 0.5, 1.0, 1.0, 1.5, 2.0, 3.0]
    total_w = sum(weights)
    time_per_q = [time_budget * w / total_w for w in weights]

    x0 = best.copy()

    for i, q in enumerate(q_values):
        rem = deadline - time.monotonic()
        if rem < 2:
            break
        tb = min(time_per_q[i], rem - 1)
        if tb < 0.5:
            break

        maxit = max(50, int(tb * 200))

        try:
            res = minimize(
                _smooth_obj_only_wrapper,
                x0,
                args=(q,),
                method='L-BFGS-B',
                jac=True,
                bounds=[(0.0, None)] * n,
                options={'maxiter': maxit, 'maxfun': maxit * 4,
                         'ftol': 1e-14, 'gtol': 1e-10,
                         'maxls': 40}
            )
            a_new = _normalize(res.x)
            if a_new is not None:
                sc = _score(a_new)
                if sc < best_sc:
                    best = a_new
                    best_sc = sc
                    x0 = best.copy()
                    print(f"    LBFGS q={q}: {sc:.6f}", flush=True)
        except Exception as e:
            print(f"    LBFGS q={q} failed: {e}", flush=True)
            break

    return best, best_sc


def propose_candidate(seed=42, budget_s=1500, **kwargs):
    """
    Main search function. Uses L-BFGS-B with smooth max homotopy,
    subgradient descent, and peak-targeted local search.
    """
    t_start = time.monotonic()
    deadline = t_start + float(budget_s) - 15.0  # 15s safety margin

    best_seq = None
    best_score = float('inf')

    # Try to get parent
    parent = None
    try:
        parent = globals().get('height_sequence_1', None)
    except Exception:
        pass

    if parent is not None:
        try:
            parent_arr = np.asarray(parent, dtype=np.float64)
            parent_arr = np.clip(parent_arr, 0.0, 1000.0)
            if parent_arr.size > 0 and np.isfinite(parent_arr).all():
                parent = _normalize(parent_arr)
                if parent is not None:
                    best_seq = parent.copy()
                    best_score = _score(best_seq)
                    print(f"Parent: len={len(parent)} score={best_score:.6f}", flush=True)
        except Exception:
            parent = None

    if parent is None:
        parent = _normalize(np.ones(512))
        best_seq = parent.copy()
        best_score = _score(best_seq)
        print(f"No parent, using uniform len=512 score={best_score:.6f}", flush=True)

    n_parent = len(parent)

    # === Phase 1: L-BFGS-B on parent length ===
    print(f"\n=== Phase 1: L-BFGS-B on n={n_parent} ===", flush=True)
    t1 = time.monotonic()
    alloc1 = min(400.0, 0.35 * (deadline - t1))
    if alloc1 > 10:
        a0 = parent.copy()
        c1, s1 = _refine_with_lbfgsb(a0, alloc1, seed + 1)
        if s1 < best_score:
            best_seq = c1.copy()
            best_score = s1
            print(f"Phase 1 result: {best_score:.6f}", flush=True)
        else:
            print(f"Phase 1 no improvement (best_here={s1:.6f})", flush=True)

    # === Phase 2: Subgradient on best ===
    print(f"\n=== Phase 2: Subgradient on n={len(best_seq)} ===", flush=True)
    t2 = time.monotonic()
    alloc2 = min(150.0, 0.12 * (deadline - t2))
    if alloc2 > 5:
        c2, s2 = _subgradient_optimize(best_seq, alloc2, seed + 2, eta_init=0.005)
        if s2 < best_score:
            best_seq = c2.copy()
            best_score = s2
            print(f"Phase 2 result: {best_score:.6f}", flush=True)
        else:
            print(f"Phase 2 no improvement (best_here={s2:.6f})", flush=True)

    # === Phase 3: Refine best with L-BFGS-B again (after subgradient) ===
    print(f"\n=== Phase 3: L-BFGS-B refinement ===", flush=True)
    t3 = time.monotonic()
    alloc3 = min(300.0, 0.22 * (deadline - t3))
    if alloc3 > 10:
        c3, s3 = _refine_with_lbfgsb(best_seq, alloc3, seed + 3)
        if s3 < best_score:
            best_seq = c3.copy()
            best_score = s3
            print(f"Phase 3 result: {best_score:.6f}", flush=True)
        else:
            print(f"Phase 3 no improvement (best_here={s3:.6f})", flush=True)

    # === Phase 4: Try different lengths ===
    print(f"\n=== Phase 4: Different lengths ===", flush=True)
    t4 = time.monotonic()
    rem4 = deadline - t4
    if rem4 > 30:
        # Try a length different from parent
        alt_lengths = [n_parent * 2, n_parent // 2, n_parent + 128]
        alt_lengths = [l for l in alt_lengths if 128 <= l <= 4096 and l != n_parent]
        alt_lengths = list(set(alt_lengths))[:3]

        for alt_n in alt_lengths:
            rem = deadline - time.monotonic()
            if rem < 30:
                break

            # Create initial guess by interpolating parent
            old = np.linspace(0, 1, n_parent)
            new_pts = np.linspace(0, 1, alt_n)
            a_init = np.interp(new_pts, old, parent)
            a_init = _normalize(a_init)

            # Also try a fresh random start
            a_rand = _make_trie_tensor(alt_n, seed + 100 + alt_n)

            # Pick the better starting point
            sc_init = _score(a_init)
            sc_rand = _score(a_rand)
            a_start = a_init if sc_init < sc_rand else a_rand
            sc_start = min(sc_init, sc_rand)

            alloc_alt = min(120.0, 0.5 * rem - 10)
            if alloc_alt > 10:
                c_alt, s_alt = _refine_with_lbfgsb(a_start, alloc_alt, seed + 200 + alt_n)
                if s_alt < best_score:
                    best_seq = c_alt.copy()
                    best_score = s_alt
                    print(f"  n={alt_n}: {best_score:.6f} (improved!)", flush=True)
                else:
                    print(f"  n={alt_n}: best_here={s_alt:.6f} (no improvement)", flush=True)

    # === Phase 5: Peak-targeted local search ===
    print(f"\n=== Phase 5: Peak-targeted search ===", flush=True)
    t5 = time.monotonic()
    rem5 = deadline - t5
    if rem5 > 20:
        alloc5 = min(120.0, 0.6 * rem5 - 10)
        if alloc5 > 5:
            c5, s5 = _peak_targeted_search(best_seq, alloc5, seed + 5)
            if s5 < best_score:
                best_seq = c5.copy()
                best_score = s5
                print(f"Phase 5 result: {best_score:.6f}", flush=True)
            else:
                print(f"Phase 5 no improvement (best_here={s5:.6f})", flush=True)

    # === Phase 6: Final L-BFGS-B polish ===
    print(f"\n=== Phase 6: Final L-BFGS-B polish ===", flush=True)
    t6 = time.monotonic()
    rem6 = deadline - t6
    if rem6 > 15:
        alloc6 = min(rem6 - 5, 100.0)
        if alloc6 > 10:
            c6, s6 = _refine_with_lbfgsb(best_seq, alloc6, seed + 999)
            if s6 < best_score:
                best_seq = c6.copy()
                best_score = s6
                print(f"Phase 6 result: {best_score:.6f}", flush=True)

    # === Phase 7: Subgradient polish ===
    t7 = time.monotonic()
    rem7 = deadline - t7
    if rem7 > 10:
        alloc7 = min(rem7 - 5, 60.0)
        if alloc7 > 3:
            c7, s7 = _subgradient_optimize(best_seq, alloc7, seed + 777, eta_init=0.003)
            if s7 < best_score:
                best_seq = c7.copy()
                best_score = s7
                print(f"Phase 7 result: {best_score:.6f}", flush=True)

    # Final validation
    final = _normalize(best_seq)
    if final is None:
        final = np.ones(64) / 64

    try:
        official = float(evaluate_sequence([float(x) for x in final]))
        if np.isfinite(official):
            best_score = official
    except Exception:
        pass

    print(f"\nFINAL: score={best_score:.6f} len={len(final)} time={time.monotonic()-t_start:.1f}s", flush=True)
    return [float(x) for x in np.clip(final, 0.0, 1000.0)]
```
