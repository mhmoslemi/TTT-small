The idea: we want a non-negative sequence a such that the autocorrelation b = a*a is as flat as possible. In the frequency domain, B(ω) = |A(ω)|². We want B to be as flat as possible, which means |A(ω)| should be as constant as possible.

- Find a ≥ 0 that minimizes max(IDFT(|A|²)) where A = DFT(a).

This is a non-convex problem. One approach is the "splitting" method:
1. Start with a random a ≥ 0
2. Compute A = DFT(a)
3. Set A_new = |A| * exp(i*phase(A)) but modify |A| to be flatter (e.g., water-filling)
4. Compute a_new = IDFT(A_new)
5. Project: a_new = max(a_new, 0)
6. Repeat

This is just a scaled version of the "phase-only" inverse DFT, which will generally be complex. After projecting to non-negative reals, we get a new starting point.

2. Compute the corresponding amplitude (from the data)
3. Inverse transform to get a "time domain" signal
4. Enforce the constraint (non-negativity) in the time domain
5. Forward transform to get new phases
6. Repeat

def build_toeplitz(a):
    """Build the convolution matrix A where b = A @ g gives convolve(a, g)."""
    n = len(a)
    m = 2 * n - 1
    A = np.zeros((m, n))
    for k in range(m):
        # b[k] = sum_j a[|k-j| direction] * g[j]
        # Actually: b[k] = sum_j a[k-j] * g[j] where 0<=k-j<n and 0<=j<n
        # So j ranges from max(0, k-(n-1)) to min(k, n-1)
        # And a[k-j] must be valid: 0 <= k-j <= n-1
        j_lo = max(0, k - (n - 1))

        if j_lo <= j_hi:

            A[k, j_lo:j_hi+1] = a[k - j_hi : k - j_lo + 1][::-1]
    return A

1. FFT-accelerated L-BFGS-B gradient optimization
2. LP-based mirror descent (best-response) for structured updates
3. Spectral flat-top initialization for diverse starting points
4. Multi-scale and multi-start strategy with adaptive time budgeting
5. Coordinate descent polish for final refinement
"""

import numpy as np
from scipy.optimize import linprog, minimize
import time


def fast_conv(a):
    """Full self-convolution via FFT, O(n log n)."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]


def obj_maxconv(x):
    """Objective for L-BFGS-B: minimize max(conv(x,x)) with x>=0, normalized."""
    x = np.maximum(x, 0.0)
    s = x.sum()
    if s < 1e-8:
        return 1e10
    x = x / s
    b = fast_conv(x)
    return float(b.max())


def build_toeplitz(a):
    """Build matrix A where A @ g = convolve(a, g) (full convolution)."""
    n = len(a)
    m = 2 * n - 1
    A = np.zeros((m, n))
    for k in range(m):
        j_lo = max(0, k - (n - 1))
        j_hi = min(n - 1, k)
        if j_lo <= j_hi:
            A[k, j_lo:j_hi + 1] = a[k - j_hi:k - j_lo + 1][::-1]
    return A


def lp_best_response(a, M):
    """Solve LP: max sum(g) s.t. (a*g)[k] <= M for all k, g >= 0."""
    n = len(a)
    m = 2 * n - 1
    c = -np.ones(n)
    A = build_toeplitz(a)
    b_ub = np.full(m, M)
    bounds = [(0, None)] * n
    res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
    if res.success and res.x is not None:
        g = res.x
        if g.sum() > 1e-10:
            return g
    return None


def spectral_flat_top(n, n_iters=60, seed=42):
    """Generate non-negative sequence with near-flat power spectrum."""
    rng = np.random.default_rng(seed)
    a = rng.uniform(0.1, 1.0, n)
    for _ in range(n_iters):
        A = np.fft.fft(a)
        A_flat = np.exp(1j * np.angle(A))
        a_new = np.real(np.fft.ifft(A_flat))
        a = np.maximum(a_new, 1e-10)
    a = a / a.sum()
    return a


def gradient_refine(a0, n_max_iter=300, max_fun=30000):
    """Refine sequence using L-BFGS-B to minimize max convolution peak."""
    n = len(a0)
    a0 = np.maximum(a0, 0.0)
    if a0.sum() < 1e-10:
        return a0
    a0 = a0 / a0.sum()
    res = minimize(obj_maxconv, a0, method='L-BFGS-B',
                   bounds=[(0, None)] * n,
                   options={'maxiter': n_max_iter, 'max_fun': max_fun,
                            'ftol': 1e-16, 'gtol': 1e-12, 'disp': False})
    if res.fun < 1e10 and res.x is not None:
        return np.maximum(res.x, 0.0)
    return a0

def coordinate_descent_polish(a, n_trials=5, deadline=None):
    """Local coordinate descent: try small perturbations of each coordinate."""
    a = a.copy()
    n = len(a)
    b = fast_conv(a)
    M = b.max()
    s = a.sum()
    best_M = M
    step = s / n * 0.1
    for i in range(n):
        if deadline is not None and time.monotonic() > deadline:
            break
        orig = a[i]
        for trial in range(n_trials):
            frac = (trial + 1) / (n_trials + 1)
            a[i] = orig * frac
            if a[i] < 0:
                a[i] = 0
            b = fast_conv(a)
            if b.max() < best_M:
                best_M = b.max()
            else:
                break
        # Also try increasing
        for trial in range(n_trials):
            frac = 1.0 + (trial + 1) / (n_trials + 1) * 0.5
            a[i] = orig * frac
            b = fast_conv(a)
            if b.max() < best_M:
                best_M = b.max()
            else:
                a[i] = orig
                break
    return a, best_M

    - Spectral initialization

    print(f"\nPhase 2: Gradient refinement on parent (n={n_parent})", flush=True)
    t0 = time.monotonic()

    perturbations = [parent.copy()]
    for scale in [0.15, 0.08, 0.03]:
        perturbations.append(parent * (1 + scale * rng.standard_normal(n_parent)))
    # Power perturbations
    perturbations.append(parent ** 0.7)
    perturbations.append(parent ** 1.3)

    t_arr = np.linspace(0, 1, n_parent, endpoint=False)
    perturbations.append(parent * (1 + 0.3 * np.cos(4 * np.pi * t_arr)))
    perturbations.append(parent * (1 + 0.2 * np.sin(6 * np.pi * t_arr)))

    for i, a0 in enumerate(perturbations):
        if time.monotonic() > deadline:
            break
        a0 = np.maximum(a0, 0)
        if a0.sum() < 0.01:
            continue
        a_refined = gradient_refine(a0, n_max_iter=400, max_fun=40000)
        best, best_score = evaluate_and_track(a_refined, best, best_score)

    if n_parent <= 1024 and time.monotonic() < deadline:

        t0 = time.monotonic()

        a = parent.copy()
        a = np.maximum(a, 0.01)

            b = fast_conv(a)
            M = float(b.max())
            g = lp_best_response(a, M)
            if g is None:
                break
            g = g / g.sum()
            # Adaptive mixing: quadratic decay
            t_mix = max(0.01, 0.6 * (1.0 - it / max_lp_iters) ** 2)
            a_new = (1.0 - t_mix) * a + t_mix * g
