import numpy as np
import time
from scipy.signal import correlate

def soft_max_objective(a, tau, n):
    """Compute f(a) = 2n * logsumexp(conv(a,a)/tau) / sum(a)^2"""
    S = np.sum(a)
    if S < 1e-10:
        return 1e10, np.zeros_like(a)
    conv = np.convolve(a, a)
    max_conv = np.max(conv)
    shifted = (conv - max_conv) / tau
    log_sum = max_conv / tau + np.log(np.sum(np.exp(shifted)))
    f = 2 * n * log_sum / (S * S)
    
    # Gradient
    exp_shifted = np.exp(shifted)
    softmax = exp_shifted / np.sum(exp_shifted)
    dL_da = (2.0 / tau) * correlate(softmax, a, mode='valid')
    grad = (2.0 * n / (S**3)) * (S * dL_da - 2.0 * log_sum)
    return f, grad

    from scipy.optimize import minimize

For a U-shaped distribution: a[i] = 1 + ε * ((2i/(n-1)) - 1)^2

Let me compute c[n-1] for this:
c[n-1] = sum_i a[i]*a[n-1-i]

If a[i] = 1 + ε * x_i^2 where x_i = 2i/(n-1) - 1 ∈ [-1, 1]:
a[n-1-i] = 1 + ε * x_{n-1-i}^2 = 1 + ε * (-x_i)^2 = 1 + ε * x_i^2 = a[i]

So a is symmetric, and a[i]*a[n-1-i] = a[i]^2 = (1 + ε*x_i^2)^2 = 1 + 2ε*x_i^2 + ε^2*x_i^4

c[n-1] = sum_i (1 + 2ε*x_i^2 + ε^2*x_i^4) ≈ n + 2ε*n/3 + ε^2*n/5

For the edges: c[0] = a[0]^2 = (1 + ε)^2 = 1 + 2ε + ε^2

So the ratio c[n-1]/c[0] ≈ (n + 2nε/3) / (1 + 2ε) for small ε, which is still ≈ n/1 = n for large n. This doesn't help much.

The issue is that the central lag involves n terms while the edge lags involve only 1 term. To flatten, we need the central terms to be small.

What if a[i] ≈ 0 in the middle? Like a "two-peak" distribution:
a[i] = 1 for i < n/4 or i > 3n/4, and a[i] = 0.3 for n/4 ≤ i ≤ 3n/4.

Then c[n-1] = sum of a[i]*a[n-1-i]. Since a is symmetric, a[n-1-i] = a[i]. So c[n-1] = sum a[i]^2.

For the two-peak: sum a[i]^2 ≈ (n/2)*1^2 + (n/2)*0.3^2 = n/2 + 0.045n ≈ 0.545n

c[0] = a[0]^2 = 1
c[n/2] ≈ a[n/2]^2 + ... hmm, this is getting complicated.

I think the L-BFGS-B approach will find these kinds of structures automatically. Let me just implement it and let the optimizer do its thing.

"""
Optimization of non-negative step function to minimize 2n*max(conv(a,a))/sum(a)^2.
Uses L-BFGS-B with soft-max surrogate (decreasing temperature) for smooth optimization,
combined with LP-based direction finding for refinement.
Explores multiple sequence lengths and initializations.
"""
import numpy as np
import time
from scipy.optimize import minimize, linprog
from scipy.signal import correlate
from scipy.sparse import csr_matrix


def soft_max_obj_grad(x, tau, n):
    """Compute soft-max objective value and gradient."""
    S = float(np.sum(x))
    if S < 1e-10:
        return 1e10, np.zeros_like(x)
    c = np.convolve(x, x)
    max_c = np.max(c)
    shifted = (c - max_c) / tau
    exp_shifted = np.exp(shifted)
    sum_exp = np.sum(exp_shifted)
    w = exp_shifted / sum_exp
    L = max_c / tau + np.log(sum_exp)
    f = 2.0 * n * L / (S * S)
    dL_da = (2.0 / tau) * correlate(w, x, mode='valid')
    grad = 2.0 * n * (S * dL_da - 2.0 * L) / (S * S * S)
    return f, grad


def soft_max_obj(x, tau, n):
    """Objective only."""
    return soft_max_obj_grad(x, tau, n)[0]


def soft_max_grad(x, tau, n):
    """Gradient only."""
    return soft_max_obj_grad(x, tau, n)[1]


def flatness_obj_grad(x, n):
    """Minimize sum(c^2)/S^4 (flatness of convolution)."""
    S = float(np.sum(x))
    if S < 1e-10:
        return 1e10, np.zeros_like(x)
    c = np.convolve(x, x)
    sum_c2 = float(np.sum(c * c))
    f = sum_c2 / (S * S * S * S)
    cross = correlate(c, x, mode='valid')
    grad = 4.0 * (S * cross - sum_c2) / (S * S * S * S * S)
    return f, grad


def flatness_obj(x, n):
    return flatness_obj_grad(x, n)[0]


def flatness_grad(x, n):
    return flatness_obj_grad(x, n)[1]


def run_lbfgsb(a_init, n, time_budget, use_flat_first=True, tau_values=None):
    """Run L-BFGS-B optimization with optional flatness pre-phase and soft-max phases."""
    a = a_init.copy().astype(np.float64)
    best_a = a.copy()
    best_score = evaluate_sequence(a.tolist())
    t_start = time.monotonic()

    # Phase 0: Flatness objective (smooth, guides toward flat convolution)
    if use_flat_first and (time.monotonic() - t_start) < time_budget * 0.3:
        try:
            result = minimize(
                fun=flatness_obj, x0=a, jac=flatness_grad,
                args=(n,), method='L-BFGS-B',
                bounds=[(0, None)] * n,
                options={'maxiter': 500, 'ftol': 1e-14, 'gtol': 1e-12}
            )
            a = np.maximum(result.x, 0.0)
            score = evaluate_sequence(a.tolist())
            if score < best_score:
                best_score = score
                best_a = a.copy()
                print(f"    flatness phase: score={best_score:.6f}")
        except Exception as e:
            print(f"    flatness phase error: {e}")

    # Phase 1: Soft-max with decreasing temperature
    if tau_values is None:
        tau_values = [1.0, 0.5, 0.25, 0.1, 0.05, 0.02, 0.01, 0.005]

    for tau in tau_values:
        if (time.monotonic() - t_start) > time_budget:
            break
        try:
            result = minimize(
                fun=soft_max_obj, x0=a, jac=soft_max_grad,
                args=(tau, n), method='L-BFGS-B',
                bounds=[(0, None)] * n,
                options={'maxiter': 500, 'ftol': 1e-14, 'gtol': 1e-12}
            )
            a = np.maximum(result.x, 0.0)
            score = evaluate_sequence(a.tolist())
            if score < best_score:
                best_score = score
                best_a = a.copy()
                print(f"    soft-max tau={tau}: score={best_score:.6f}")
            else:
                # Still update a to the solution (might help next tau)
                a = np.maximum(result.x, 0.0)
        except Exception as e:
            print(f"    soft-max tau={tau} error: {e}")
            break

    return best_a, best_score


def build_conv_matrix_sparse(f):
    """Build sparse matrix A where (A@g)[k] = convolve(f,g)[k]."""
    n = len(f)
    rows_list = []
    cols_list = []
    vals_list = []
    for k in range(2 * n - 1):
        j_min = max(0, k - (n - 1))
        j_max = min(k, n - 1)
        for j in range(j_min, j_max + 1):
            rows_list.append(k)
            cols_list.append(j)
            vals_list.append(f[k - j])
    return csr_matrix((vals_list, (rows_list, cols_list)), shape=(2 * n - 1, n))


def solve_lp_direction(f, M):
    """Solve LP: max sum(g) s.t. convolve(f,g) <= M, g >= 0."""
    n = len(f)
    A = build_conv_matrix_sparse(f)
    c_vec = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    try:
        result = linprog(c_vec, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
        if result.success:
            return result.x
    except Exception:
        pass
    return None


def lp_refine(a, n, deadline, max_iters=200):
    """LP-based iterative refinement."""
    a = a.copy().astype(np.float64)
    best_a = a.copy()
    best_score = evaluate_sequence(a.tolist())
    step_size = 0.2
    no_improve = 0
    it = 0

    while time.monotonic() < deadline and it < max_iters:
        conv_f = np.convolve(a, a)
        M = float(np.max(conv_f))
        if M <= 0:
            break

        start_t = time.monotonic()
        g = solve_lp_direction(a, M)
        lp_time = time.monotonic() - start_t

        if g is None:
            no_improve += 1
            if no_improve > 20:
                break
            continue

        sum_g = float(np.sum(g))
        if sum_g <= 1e-15:
            no_improve += 1
            continue

        sum_a = float(np.sum(a))
        g = g * (sum_a / sum_g)

        new_a = (1.0 - step_size) * a + step_size * g
        new_a = np.clip(new_a, 0.0, 1000.0)
        it += 1

        new_score = evaluate_sequence(new_a.tolist())
        if np.isfinite(new_score) and new_score < best_score:
            a = new_a
            best_a = a.copy()
            best_score = new_score
            no_improve = 0
            if it % 20 == 0:
                print(f"    LP iter {it}: score={best_score:.6f} ({lp_time:.2f}s)")
        else:
            no_improve += 1
            step_size *= 0.7
            if step_size < 0.01 or no_improve > 30:
                break

    return best_a, best_score


def subgradient_polish(a, n, deadline, lr_init=0.05, max_iters=2000):
    """Subgradient method with momentum, exploiting the simple gradient structure."""
    a = a.copy().astype(np.float64)
    best_a = a.copy()
    best_score = evaluate_sequence(a.tolist())
    lr = lr_init
    momentum = np.zeros(n)
    beta = 0.9
    no_improve = 0
    S = float(np.sum(a))

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        conv_f = np.convolve(a, a)
        k_star = int(np.argmax(conv_f))
        max_c = conv_f[k_star]

        # Subgradient of max(conv) w.r.t. a
        dmax = np.zeros(n)
        j_valid_start = max(0, k_star - (n - 1))
        j_valid_end = min(k_star, n - 1)
        for j in range(j_valid_start, j_valid_end + 1):
            dmax[j] = 2.0 * a[k_star - j]

        # Gradient of f = 2n * max_c / S^2
        # df/da[j] = 2n * (S * dmax[j] - 2*max_c) / S^3
        grad = 2.0 * n * (S * dmax - 2.0 * max_c) / (S * S * S)
        # Add regularization to keep sum constant
        grad -= np.sum(grad) / n

        # Momentum update
        momentum = beta * momentum + (1 - beta) * grad
        new_a = a - lr * momentum
        new_a = np.clip(new_a, 0.0, 1000.0)

        # Renormalize
        new_sum = float(np.sum(new_a))
        if new_sum > 1e-10:
            new_a = new_a * (S / new_sum)

        new_score = evaluate_sequence(new_a.tolist())
        if np.isfinite(new_score) and new_score < best_score:
            a = new_a
            best_a = a.copy()
            best_score = new_score
            no_improve = 0
            if (it + 1) % 200 == 0:
                print(f"    subgrad iter {it+1}: score={best_score:.6f}")
        else:
            no_improve += 1
            lr *= 0.95
            if lr < 1e-6 or no_improve > 300:
                break

    return best_a, best_score


def make_interpolated(parent, n_target):
    """Interpolate parent sequence to target length."""
    n_parent = len(parent)
    if n_parent == n_target:
        return parent.copy()
    x_old = np.linspace(0, 1, n_parent)
    x_new = np.linspace(0, 1, n_target)
    return np.interp(x_new, x_old, parent)


def make_cosine_taper(n):
    """Cosine-tapered initialization (high in middle, low at edges)."""
    idx = np.arange(n)
    return 0.5 + 0.5 * np.cos(np.pi * idx / (n - 1)) ** 2


def make_bimodal(n, sharpness=0.3):
    """Bimodal: high at edges, low in middle."""
    idx = np.arange(n, dtype=np.float64)
    t = 2.0 * idx / (n - 1) - 1.0
    return 1.0 + sharpness * t ** 2


def make_random_structured(n, rng, freq=None):
    """Random structured initialization."""
    idx = np.arange(n, dtype=np.float64)
    if freq is None:
        freq = rng.uniform(1, 5)
    a = 1.0 + 0.3 * np.cos(2 * np.pi * freq * idx / n + rng.uniform(0, 2 * np.pi))
    a = np.clip(a, 0.1, None)
    return a
