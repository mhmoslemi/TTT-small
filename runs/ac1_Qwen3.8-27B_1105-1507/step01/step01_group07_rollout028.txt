def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    conv = np.convolve(a, a)
    max_b = float(np.max(conv))
    sum_a = float(np.sum(a))
    if sum_a < 0.01:
        return np.inf
    return 2.0 * n * max_b / (sum_a * sum_a)


def build_A(a):
    """Build constraint matrix for LP. A[k,j] such that (A @ g)[k] = (a*g)[k]."""
    n = len(a)
    A = np.zeros((2 * n - 1, n), dtype=np.float64)
    for k in range(2 * n - 1):

        j_end = min(n - 1, k)
        A[k, j_start:j_end + 1] = a[k - j_end:k - j_start + 1][::-1]
    return A

"""
Hybrid LP alternating minimization + smoothed gradient descent for minimizing
2*n*max(conv(a,a))/(sum(a))^2. Uses adaptive step sizes, diverse structured
initializations, multiple sequence lengths, and gradient-based refinement.
"""

import numpy as np
from scipy.optimize import linprog
import time


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    conv = np.convolve(a, a)
    max_b = float(np.max(conv))
    sum_a = float(np.sum(a))
    if sum_a < 0.01:
        return np.inf
    return 2.0 * n * max_b / (sum_a * sum_a)


def build_A(a):
    """Build constraint matrix for LP. (A @ g)[k] = (a*g)[k]."""
    n = len(a)
    A = np.zeros((2 * n - 1, n), dtype=np.float64)
    for k in range(2 * n - 1):
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        A[k, j_start:j_end + 1] = a[k - j_end:k - j_start + 1][::-1]
    return A


def solve_lp_step(a):
    """Solve LP: max sum(g) s.t. (a*g)_k <= M for all k, g >= 0."""
    n = len(a)
    conv = np.convolve(a, a)
    M = float(np.max(conv))
    A = build_A(a)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
    if res.status == 0:
        return res.x, float(-res.fun), M
    return None, None, M


def run_lp_optimize(a_init, seed=0, max_iter=300, verbose=False):
    """Run LP-based alternating minimization with adaptive step size."""
    rng = np.random.default_rng(seed)
    n = len(a_init)

    if s < 1e-10:

    best_a = a.copy()

        g, g_sum, M = solve_lp_step(a)
        if g is None:
            break

        if g_sum <= 1.0 + 1e-6:
            stall += 1

                noise = np.exp(rng.normal(0, scale, n))
                a = best_a * noise

                s = np.sum(a)
                if s < 1e-10:
                    a = rng.random(n)
                    s = np.sum(a)
                a = a / s
                stall = 0
                t = 0.5
            continue
        stall = 0

        g = g / g_sum
        best_t = 0.0

def smoothed_objective_and_grad(a, tau):
    """Compute smoothed objective and gradient using log-sum-exp."""
    n = len(a)
    conv = np.convolve(a, a)
    sum_a = float(np.sum(a))
    if sum_a < 0.01:
        return np.inf, None

    max_conv = float(np.max(conv))
    exp_vals = np.exp((conv - max_conv) / tau)
    sum_exp = float(np.sum(exp_vals))
    weights = exp_vals / sum_exp
    lse = max_conv + tau * np.log(sum_exp)

    # Gradient of lse w.r.t. a_i = 2 * sum_k weights[k] * a[k-i] (a[x]=0 outside [0,n-1])
    # Compute via direct summation (O(n^2) but fine for n < 2000)
    grad_lse = np.zeros(n)
    for i in range(n):
        k_min = max(0, i)
        k_max = min(2 * n - 2, i + n - 1)
        if k_min <= k_max:
            grad_lse[i] = 2.0 * float(np.dot(weights[k_min:k_max + 1],
                                              a[k_min - i:k_max + 1 - i]))

    obj = 2.0 * n * lse / (sum_a * sum_a)
    grad = 2.0 * n * (grad_lse * sum_a - 2.0 * lse) / (sum_a ** 3)
    return obj, grad

        if grad is None:
            break

        # Gradient step
        a_new = a - lr * grad
        a_new = np.clip(a_new, 0, None)
        s = np.sum(a_new)
        if s < 1e-10:
            break
        a_new = a_new / s

        obj_new = compute_obj(a_new)
        if obj_new < best_obj:
            best_a = a_new.copy()
            best_obj = obj_new
            a = a_new
            if verbose and (it % 200 == 0):

            else:
                lr = max(0.001, lr * 0.9)

            if obj < best_score:
                best_score = obj
                best_seq = a.copy()
                print(f"  LP n={n} r={restart}: obj={obj:.6f}")

    # Phase 2: Gradient refinement on best
    if best_seq is not None and time.monotonic() < deadline - 60:
        remaining = deadline - time.monotonic()
        print(f"\nPhase 2: Gradient refinement (n={len(best_seq)}), {remaining:.0f}s left")
        a_g, obj_g = gradient_refine(best_seq, max_iter=2000, tau_start=0.5, tau_end=0.005)
        if obj_g < best_score:
            best_score = obj_g
            best_seq = a_g.copy()
            print(f"  Gradient: obj={best_score:.6f}")

    # Phase 3: More LP on best with smaller step
    if best_seq is not None and time.monotonic() < deadline - 40:
        remaining = deadline - time.monotonic()
        n = len(best_seq)
        print(f"\nPhase 3: More LP (n={n}), {remaining:.0f}s left")
        max_iter = int(min(400, max(50, remaining / 1.5)))
        a_lp, obj_lp = run_lp_optimize(best_seq, seed=seed + 99983, max_iter=max_iter)
        if obj_lp < best_score:
            best_score = obj_lp
            best_seq = a_lp.copy()
            print(f"  More LP: obj={best_score:.6f}")

    # Phase 4: Random refinement
    if best_seq is not None and time.monotonic() < deadline - 20:
        remaining = deadline - time.monotonic()
        print(f"\nPhase 4: Random refinement, {remaining:.0f}s left")
        rls_iters = int(remaining / 0.015)
        a_r, obj_r = random_refine(best_seq, rng, max_iter=rls_iters)
        if obj_r < best_score:
            best_score = obj_r
            best_seq = a_r.copy()
            print(f"  Random: obj={best_score:.6f}")

    # Phase 5: Final gradient polish
    if best_seq is not None and time.monotonic() < deadline - 5:
        remaining = deadline - time.monotonic()
        n = len(best_seq)
        if n <= 600:
            a_g2, obj_g2 = gradient_refine(best_seq, max_iter=int(remaining / 0.05),
                                           tau_start=0.1, tau_end=0.003)
            if obj_g2 < best_score:
                best_score = obj_g2
                best_seq = a_g2.copy()
                print(f"  Final gradient: obj={best_score:.6f}")

    # Fallback
    if best_seq is None:
        best_seq = np.ones(100)
        best_score = compute_obj(best_seq)

    elapsed = time.monotonic() - start_time
    print(f"\nDONE: n={len(best_seq)}, score={best_score:.6f}, time={elapsed:.1f}s")
    return [float(x) for x in best_seq]
