"""
Convex optimization approach for minimizing 2n*max(conv(a,a))/sum(a)^2.
Uses FISTA (accelerated gradient) on the simplex, followed by LP-based refinement.
The normalized problem (sum(a)=1, a>=0) is convex, so we find the global optimum.
"""
import numpy as np
import time
from scipy.optimize import linprog
from scipy.sparse import csr_matrix

def convolve_fft(a):
    """Compute conv(a, a) using FFT for efficiency."""
    n = len(a)
    size = 2 * n - 1
    f_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, f_size)
    return np.fft.irfft(A * A, f_size)[:size]

def project_simplex(v, s=1.0):
    """Project v onto {x: x >= 0, sum(x) = s} in O(n) expected time."""
    n = len(v)
    if n == 0:
        return v.copy()
    x = np.sort(v)[::-1]
    cssv = np.cumsum(x)
    ind = np.arange(1, n + 1, dtype=np.float64)
    cond = x - (cssv - s) / ind > 0
    rho = np.max(ind[cond]) if np.any(cond) else 1.0
    theta = (cssv[int(rho) - 1] - s) / rho
    return np.maximum(v - theta, 0)

def grad_conv_k(a, k):
    """Gradient of conv(a,a)[k] w.r.t. a. Vectorized."""
    n = len(a)
    j = k - np.arange(n)
    valid = (j >= 0) & (j < n)
    g = np.zeros(n)
    g[valid] = 2.0 * a[j[valid]]
    return g

def fista_min(a0, n_iters=50000, deadline=None, lr=None, verbose_prefix=""):
    """
    FISTA to minimize max(conv(a,a)[k]) on the simplex {a>=0, sum(a)=1}.
    Returns (best_a_normalized, best_objective).
    """
    a = a0.copy().astype(np.float64)
    a = np.maximum(a, 0)
    s = np.sum(a)
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
        s = 1.0
    else:
        a = a / s
    n = len(a)
    
    if lr is None:
        lr = 0.4 / n  # Conservative step size
    
    a_prev = a.copy()
    t = 1.0
    best_a = a.copy()
    best_conv_max = float(np.max(convolve_fft(a)))
    
    for it in range(n_iters):
        if deadline is not None and time.monotonic() > deadline:
            break
        
        # FISTA extrapolation
        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t * t))
        momentum = (t - 1.0) / t_new
        z = a + momentum * (a - a_prev)
        z = np.maximum(z, 0)
        zs = np.sum(z)
        if zs < 1e-15:
            z = a
        else:
            z = z / zs
        
        # Compute convolution and find active constraint(s)
        conv_z = convolve_fft(z)
        k_star = int(np.argmax(conv_z))
        
        # Gradient of active constraint
        g = grad_conv_k(z, k_star)
        
        # Project gradient onto tangent space of simplex
        g_tan = g - g.mean()
        
        # Gradient step
        a_candidate = z - lr * g_tan
        
        # Project onto simplex
        a_new = project_simplex(a_candidate, s=1.0)
        
        # Verify improvement (optional, for safety)
        a_prev = a
        a = a_new
        t = t_new
        
        # Track best every 500 iterations
        if (it + 1) % 500 == 0:
            cm = float(np.max(convolve_fft(a)))
            if cm < best_conv_max:
                best_conv_max = cm
                best_a = a.copy()
            if verbose_prefix and (it + 1) % 5000 == 0:
                print(f"    {verbose_prefix}iter {it+1}: max_conv={cm:.8f}, obj={2*n*cm:.6f}")
    
    # Final evaluation
    cm = float(np.max(convolve_fft(a)))
    if cm < best_conv_max:
        best_conv_max = cm
        best_a = a.copy()
    
    return best_a, 2.0 * n * best_conv_max

def build_lp_matrix(f):
    """Build sparse matrix for conv(f, g) as linear in g."""
    n = len(f)
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        j_min = max(0, k - n + 1)
        j_max = min(k, n - 1)
        for j in range(j_min, j_max + 1):
            rows.append(k)
            cols.append(j)
            vals.append(f[k - j])
    return csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))

def lp_refine_step(f_norm, max_lp_solves=30, deadline=None):
    """
    LP-based refinement: at each step, solve LP to find direction g
    that maximizes sum(g) subject to conv(f,g) <= M, g >= 0.
    Then blend f toward g.
    """
    n = len(f_norm)
    f = f_norm.copy().astype(np.float64)
    best_f = f.copy()
    best_cm = float(np.max(convolve_fft(f)))
    step = 0.15
    
    A = build_lp_matrix(f)  # Will rebuild if f changes
    
    for it in range(max_lp_solves):
        if deadline is not None and time.monotonic() > deadline:
            break
        
        M = float(np.max(convolve_fft(f)))
        
        # Solve LP
        c = -np.ones(n)
        b_ub = np.full(2 * n - 1, M)
        bounds = [(0, None)] * n
        try:
            res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                         options={'ipm': {'maxiter': 200}})
            if not res.success:
                break
            g = res.x
        except Exception:
            break
        
        sum_g = np.sum(g)
        if sum_g < 1e-15:
            break
        
        g = g / sum_g  # Normalize to sum 1
        
        # Blend
        new_f = (1 - step) * f + step * g
        new_f = np.maximum(new_f, 0)
        ns = np.sum(new_f)
        if ns < 1e-15:
            break
        new_f = new_f / ns
        
        new_cm = float(np.max(convolve_fft(new_f)))
        
        if new_cm < best_cm:
            best_cm = new_cm
            best_f = new_f.copy()
            f = new_f
            A = build_lp_matrix(f)  # Rebuild
            step = min(step * 1.05, 0.4)
        else:
            step *= 0.7
            if step < 0.005:
                break
    
    return best_f, 2.0 * n * best_cm
