```python
"""
Optimizes non-negative step function heights to minimize 2n*max(conv(f,f))/(sum(f))^2.
Uses continuation method with smooth max approximation (log-sum-exp) and L-BFGS
in log-softmax parameterization, combined with mirror descent and coordinate descent.
"""

import numpy as np
import time
from scipy.optimize import minimize
from scipy.special import logsumexp

def _fft_convolve_full(f):
    n = len(f)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    F = np.fft.rfft(f, size)
    h = np.fft.irfft(F * F, size)
    return h[:2 * n - 1]

def _softmax(g):
    g = g - np.max(g)
    e = np.exp(g)
    return e / np.sum(e)

def _score_raw(f):
    n = len(f)
    h = _fft_convolve_full(f)
    M = float(np.max(h))
    S = float(np.sum(f))
    if S < 0.01:
        return float('inf')
    return 2.0 * n * M / (S * S)

def _normalize(f):
    f = np.maximum(f, 0.0)
    s = float(np.sum(f))
    if s < 1e-30:
        return None
    return f / s

def _smooth_max_and_grad(g, tau):
    """
    Compute the smooth max objective and its gradient w.r.t. g.
    Objective = tau * logsumexp(conv(softmax(g), softmax(g)) / tau)
    Gradient = sum_k w_k * d(conv)_k/dg
    """
    n = len(g)
    f = _softmax(g)
    h = _fft_convolve_full(f)
    
    # Objective
    obj = tau * float(logsumexp(h / tau))
    
    # Softmax weights
    h_shifted = h - np.max(h)
    w_unnorm = np.exp(h_shifted / tau)
    w = w_unnorm / np.sum(w_unnorm)
    
    # Compute grad_f[a] = sum_k w_k * dh_k/df[a]
    # = sum_{k=a}^{a+n-1} w_k * f[k-a] + f[a] * sum_{k=a}^{a+n-1} w_k
    # (clipped to valid range [0, 2n-2])
    
    # First term: cross-correlation C[a] = sum_i w[a+i]*f[i]
    size = 1
    while size < 2 * n:
        size <<= 1
    W = np.fft.rfft(w, size)
    Ff = np.fft.rfft(f, size)
    # Cross-correlation R_wf[a] = sum_x w[x+a]*f[x] = ifft(conj(W)*Ff)
    C = np.fft.irfft(np.conj(W) * Ff, size)[:n]
    
    # Second term: f[a] * partial_sum_w[a]
    # partial_sum_w[a] = sum of w[k] for k in [max(0,a), min(2n-2, a+n-1)]
    w_cumsum = np.concatenate([[0.0], np.cumsum(w)])
    lo = np.arange(n, dtype=np.int64)  # max(0, a) = a
    hi = np.minimum(2 * n - 2, np.arange(n) + n - 1).astype(np.int64)
    partial_sum_w = w_cumsum[hi + 1] - w_cumsum[lo]
    
    grad_f = C + f * partial_sum_w
    
    # Convert to gradient w.r.t. g: dObj/dg = f * (grad_f - dot(f, grad_f))
    dot_f_gradf = float(np.dot(f, grad_f))
    grad_g = f * (grad_f - dot_f_gradf)
    
    return obj, grad_g

def _continuation_lbfgs(f_init, tau_start, tau_end, n_steps, max_iter_per_tau, time_budget):
    """
    Continuation method: decrease tau from tau_start to tau_end in n_steps,
    using L-BFGS at each step.
    """
    f = _normalize(np.asarray(f_init, dtype=np.float64).copy())
    if f is None:
        return f_init, float('inf')
    
    n = len(f)
    g = np.log(f + 1e-30)  # Initialize g from f
    best_f = f.copy()
    best_score = _score_raw(f)
    start_time = time.monotonic()
    
    # Geometric sequence of tau values
    taus = np.geomspace(tau_start, tau_end, n_steps)
    
    for i, tau in enumerate(taus):
        if time.monotonic() - start_time > time_budget:
            break
        
        def obj_fn(g):
            return _smooth_max_and_grad(g, tau)[0]
        
        def grad_fn(g):
            return _smooth_max_and_grad(g, tau)[1]
        
        res = minimize(obj_fn, g, method='L-BFGS-B', jac=grad_fn,
                       options={'maxiter': max_iter_per_tau, 'ftol': 1e-10, 'gtol': 1e-8})
        
        f_new = _softmax(res.x)
        f_new = _normalize(f_new)
        if f_new is None:
            continue
        
        score = _score_raw(f_new)
        if score < best_score:
            best_score = score
            best_f = f_new.copy()
        
        g = res.x
        f = f_new
        
        if (i + 1) % 3 == 0:
            print(f"    L-BFGS tau={tau:.6f}: score={best_score:.6f}")
    
    return best_f, best_score

def _mirror_descent(f_init, n_iter, eta_scale, threshold_frac, label=""):
    """Mirror descent with multiplicative updates."""
    f = _normalize(np.asarray(f_init, dtype=np.float64).copy())
    if f is None:
        return f_init.copy(), float('inf')
    n = len(f)
    best_f = f.copy()
    best_score = _score_raw(f)
    
    for it in range(n_iter):
        h = _fft_convolve_full(f)
        M = float(np.max(h))
        if M <= 0:
            break
        
        threshold = threshold_frac * M + 1e-15
        active = np.where(h >= threshold)[0]
        if len(active) == 0:
            active = np.array([int(np.argmax(h))])
        
        # Pressure vector
        p = np.zeros(n)
        for k in active:
            j_start = max(0, k - n + 1)
            j_end = min(n - 1, k)
            if j_end >= j_start:
                lo = k - j_end
                hi = k - j_start
                p[j_start:j_end + 1] += 2.0 * f[lo:hi + 1][::-1]
        p /= float(len(active))
        
        eta = eta_scale / M
        f = f * np.exp(np.clip(-eta * p, -25, 25))
        f_norm = _normalize(f)
        f = f_norm if f_norm is not None else np.ones(n) / n
        
        score = _score_raw(f)
        if score < best_score:
            best_score = score
            best_f = f.copy()
    
    return best_f, best_score

def _coordinate_descent(f_init, n_sweeps, time_budget):
    """Coordinate descent: optimize one coordinate at a time."""
    f = _normalize(np.asarray(f_init, dtype=np.float64).copy())
    if f is None:
        return f_init.copy(), float('inf')
    n = len(f)
    best_f = f.copy()
    best_score = _score_raw(f)
    start_time = time.monotonic()
    
    for sweep in range(n_sweeps):
        if time.monotonic() - start_time > time_budget:
            break
        
        improved = False
        for j in range(n):
            # Binary search for optimal f[j]
            # Fix all other coordinates, find f[j] that minimizes max(conv)
            # subject to f[j] >= 0 and sum constraint (we'll handle sum by rescaling)
            
            f_old = f[j]
            
            # The convolution as a function of f[j] = x:
            # For each lag k, (f*f)_k = a_k * x + b_k where:
            #   if k = 2j: (f*f)_k = x^2 + (other terms)... this is quadratic
            #   otherwise: (f*f)_k = 2*x*f[k-j] + (other terms) [linear in x]
            
            # For simplicity, use ternary search on x in [0, 2*f_old + 0.1]
            lo_x = 0.0
            hi_x = 2.0 * f_old + 0.01 / n
            
            # Evaluate score at a few points to bracket
            def eval_at_x(x):
                f[j] = x
                h = _fft_convolve_full(f)
                f[j] = f_old
                return np.max(h)
            
            # Ternary search (10 iterations)
            for _ in range(10):
                m1 = lo_x + (hi_x - lo_x) / 3
                m2 = hi_x - (hi_x - lo_x) / 3
                v1 = eval_at_x(m1)
                v2 = eval_at_x(m2)
                if v1 < v2:
                    hi_x = m2
                else:
                    lo_x = m1
            
            f_new_j = (lo_x + hi_x) / 2
            f[j] = f_new_j
            improved = True
        
        # Renormalize
        f_norm = _normalize(f)
        if f_norm is not None:
            f = f_norm
        
        score = _score_raw(f)
        if score < best_score:
            best_score = score
            best_f = f.copy()
    
    return best_f, best_score
```

```python
"""
Optimizes non-negative step function heights to minimize 2n*max(conv(f,f))/(sum(f))^2.
Uses continuation method with smooth max (log-sum-exp) approximation and L-BFGS in
log-softmax parameterization, combined with mirror descent for robustness.
FFT-based gradient computation enables efficient optimization for sequences up to n=4096.
"""

import numpy as np
import time
from scipy.optimize import minimize
from scipy.special import logsumexp


def _fft_convolve_full(f):
    """Compute full linear convolution of f with itself using FFT."""
    n = len(f)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    F = np.fft.rfft(f, size)
    h = np.fft.irfft(F * F, size)
    return h[:2 * n - 1]


def _softmax_stable(g):
    """Numerically stable softmax."""
    g = g - np.max(g)
    e = np.exp(g)
    s = e.sum()
    if s == 0:
        return np.ones(len(g)) / len(g)
    return e / s


def _score_raw(f):
    """Fast score: 2n*max(conv(f,f))/sum(f)^2."""
    n = len(f)
    h = _fft_convolve_full(f)
    M = float(np.max(h))
    S = float(np.sum(f))
    if S < 0.01:
        return float('inf')
    return 2.0 * n * M / (S * S)


def _normalize(f):
    """Normalize to sum=1, clip negatives."""
    f = np.maximum(f, 0.0)
    s = float(np.sum(f))
    if s < 1e-30:
        return None
    return f / s


def _smooth_max_obj_grad(g, tau):
    """
    Compute smooth max objective and gradient w.r.t. g.
    Obj = tau * logsumexp(conv(softmax(g), softmax(g)) / tau)
    Grad = sum_k w_k * d(conv)_k / dg  (standard softmax chain rule applied)
    """
    n = len(g)
    f = _softmax_stable(g)
    h = _fft_convolve_full(f)

    # Objective value
    obj = tau * float(logsumexp(h / tau))

    # Softmax weights over lags
    h_shifted = h - np.max(h)
    w_unnorm = np.exp(h_shifted / tau)
    w = w_unnorm / w_unnorm.sum()

    # grad_f[a] = sum_k w_k * dh_k/df[a]
    # = sum_{k in [a, a+n-1] ∩ [0,2n-2]} w_k * f[k-a]  +  f[a] * sum_{k in [a, a+n-1] ∩ [0,2n-2]} w_k
    # First term: cross-correlation C[a] = sum_i w[a+i]*f[i]
    size = 1
    while size < 2 * n:
        size <<= 1
    W = np.fft.rfft(w, size)
    Ff = np.fft.rfft(f, size)
    # R_wf[a] = sum_x w[x+a]*f[x] = ifft(conj(FFT(w)) * FFT(f))
    C = np.fft.irfft(np.conj(W) * Ff, size)[:n]

    # Second term: partial sums of w
    w_cumsum = np.concatenate([[0.0], np.cumsum(w)])
    lo = np.arange(n, dtype=np.int64)
    hi = np.minimum(2 * n - 2, np.arange(n) + n - 1).astype(np.int64)
    partial_sum_w = w_cumsum[hi + 1] - w_cumsum[lo]

    grad_f = C + f * partial_sum_w

    # Chain rule for softmax parameterization
    dot_fg = float(np.dot(f, grad_f))
    grad_g = f * (grad_f - dot_fg)

    return obj, grad_g


def _continuation_lbfgs(f_init, tau_start, tau_end, n_steps, max_iter_per_tau, time_budget, verbose=True):
    """
    Continuation method: geometrically decrease tau, run L-BFGS at each level.
    """
    f = _normalize(np.asarray(f_init, dtype=np.float64).copy())
    if f is None:
        return np.asarray(f_init, dtype=np.float64).copy(), float('inf')

    n = len(f)
    g = np.log(f + 1e-30)
    best_f = f.copy()
    best_score = _score_raw(f)
    start_time = time.monotonic()

    if tau_start <= 0 or tau_end <= 0:
        return best_f, best_score

    taus = np.geomspace(tau_start, tau_end, n_steps)

    for i, tau in enumerate(taus):
        elapsed = time.monotonic() - start_time
        if elapsed > time_budget:
            break

        def obj_fn(g_inner):
            return _smooth_max_obj_grad(g_inner, tau)[0]

        def grad_fn(g_inner):
            return _smooth_max_obj_grad(g_inner, tau)[1]

        remaining = time_budget - (time.monotonic() - start_time)
        max_it = max(10, min(max_iter_per_tau, int(remaining * 50)))

        try:
            res = minimize(obj_fn, g, method='L-BFGS-B', jac=grad_fn,
                           options={'maxiter': max_it, 'ftol': 1e-12, 'gtol': 1e-10,
                                    'maxls': 50})
        except Exception:
            break

        f_new = _softmax_stable(res.x)
        f_new = _normalize(f_new)
        if f_new is None:
            break

        score = _score_raw(f_new)
        if score < best_score:
            best_score = score
            best_f = f_new.copy()

        g = res.x
        f = f_new

        if verbose and (i + 1) % 2 == 0:
            print(f"      cont tau={tau:.6g}: score={best_score:.6f} (iters={res.nit})")

    return best_f, best_score


def _mirror_descent(f_init, n_iter, eta_scale, threshold_frac):
    """Mirror descent with multiplicative updates and active-set subgradient."""
    f = _normalize(np.asarray(f_init, dtype=np.float64).copy())
    if f is None:
        return np.asarray(f_init, dtype=np.float64).copy(), float('inf')
    n = len(f)
    best_f = f.copy()
    best_score = _score_raw(f)

    for it in range(n_iter):
        h = _fft_convolve_full(f)
        M = float(np.max(h))
        if M <= 0:
            break

        threshold = threshold_frac * M + 1e-15
        active = np.where(h >= threshold)[0]
        if len(active) == 0:
            active = np.array([int(np.argmax(h))])

        p = np.zeros(n)
        for k in active:
            j_start = max(0, k - n + 1)
            j_end = min(n - 1, k)
            if j_end >= j_start:
                lo = k - j_end
                hi = k - j_start
                p[j_start:j_end + 1] += 2.0 * f[lo:hi + 1][::-1]
        p /= float(len(active))

        eta = eta_scale / M
        f = f * np.exp(np.clip(-eta * p, -25.0, 25.0))
        f_norm = _normalize(f)
        f = f_norm if f_norm is not None else np.ones(n) / n

        score = _score_raw(f)
        if score < best_score:
            best_score = score
            best_f = f.copy()

    return best_f, best_score


def _cosine_init(n):
    j = np.arange(n, dtype=np.float64)
    f = np.cos(np.pi * (j + 0.5) / float(n)) ** 2
    f = np.maximum(f, 0.0)
    s = np.sum(f)
    if s > 0:
        f /= s
    return f


def _gaussian_init(n, sigma_frac):
    j = np.arange(n, dtype=np.float64) - (n - 1) / 2.0
    sigma = sigma_frac * float(n)
    f = np.exp(-j * j / (2.0 * sigma * sigma))
    s = np.sum(f)
    if s > 0:
        f /= s
    return f


def _recurrence_init(n):
    """
    Initialize from the 'flat convolution' recurrence:
    f[0] = 1, f[k] = (1 - sum_{i=1}^{k-1} 2*f[i]*f[k-i]) / (2*f[0]) for k < n
    (ignoring boundary effects, giving a good approximate flat-autocorrelation sequence)
    """
    f = np.zeros(n)
    f[0] = 1.0
    for k in range(1, n):
        s = 0.0
        for i in range(1, k):
            if i <= k - 1 and (k - i) <= n - 1:
                s += 2.0 * f[i] * f[k - i]
        # Also account for boundary: f[k] = (C - cross_terms) / (2*f[0])
        # For the "infinite" recurrence: f[k] = (1 - s) / (2*f[0])
        f[k] = max(0.0, (1.0 - s) / (2.0 * f[0]))
    f = np.maximum(f, 0.0)
    s = np.sum(f)
    if s > 0:
        f /= s
    return f


def _tapered_init(n, power):
    j = np.arange(n, dtype=np.float64)
    x = np.sin(np.pi * (j + 0.5) / float(n))
    f = x ** power
    f = np.maximum(f, 0.0)
    s = np.sum(f)
    if s > 0:
        f /= s
    return f


def _smooth_descent(f_init, n_iter, tau_scale, time_budget):
    """
    Gradient descent on the smooth max with adaptive step size (backtracking).
    Works on the simplex using projected gradient or softmax parameterization.
    """
    f = _normalize(np.asarray(f_init, dtype=np.float64).copy())
    if f is None:
        return np.asarray(f_init, dtype=np.float64).copy(), float('inf')
    n = len(f)
    g = np.log(f + 1e-30)
    best_f = f.copy()
    best_score = _score_raw(f)
    start_time = time.monotonic()

    h0 = _fft_convolve_full(f)
    tau = tau_scale * float(np.max(h0) - np.mean(h0) + 1e-10)
    tau = max(tau, 1e-4)

    step = 1.0
    for it in range(n_iter):
        if time.monotonic() - start_time > time_budget:
            break

        obj, grad = _smooth_max_obj_grad(g, tau)

        # Backtracking line search
        alpha = step
        for _ in range(20):
            g_trial = g - alpha * grad
            f_trial = _softmax_stable(g_trial)
            h_trial = _fft_convolve_full(f_trial)
            obj_trial = tau * float(logsumexp(h_trial / tau))
            if obj_trial <= obj - 1e-4 * alpha * np.dot(grad, grad):
                break
            alpha *= 0.5
        else:
            break

        g = g - alpha * grad
        f_new = _softmax_stable(g)
        f_new = _normalize(f_new)
        if f_new is None:
            break

        score = _score_raw(f_new)
        if score < best_score:
            best_score = score
            best_f = f_new.copy()

        f = f_new
        # Slowly decrease tau
        tau *= 0.995

    return best_f, best_score


def propose_candidate(seed=42, budget_s=1500, **kwargs):
    """
    Search for non-negative step function heights minimizing the evaluation score.
    Strategy:
      Phase 1: Multi-start with mirror descent for quick exploration.
      Phase 2: Continuation L-BFGS (smooth max with decreasing temperature) for refinement.
      Phase 3: Gradient descent with adaptive step for final polish.
      Phase 4: Try different sequence lengths and restarts.
    """
    deadline = time.monotonic() + float(budget_s) - 5.0
    rng = np.random.default_rng(seed)

    # Get parent sequence
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64).copy()
        parent = np.clip(parent, 0.0, 1000.0)
    except Exception:
        parent = np.ones(512, dtype=np.float64)

    best_f = parent.copy()
    best_score = float(evaluate_sequence(best_f.tolist()))
    print(f"Parent: n={len(best_f)}, score={best_score:.6f}")

    # Build candidate initializations for n=512
    n_default = 512
    candidates = []

    p = _normalize(parent)
    if p is not None:
        candidates.append(("parent", p))

    candidates.append(("uniform", np.ones(n_default) / n_default))
    candidates.append(("cosine", _cosine_init(n_default)))
    candidates.append(("recurrence", _recurrence_init(n_default)))

    for sf in [0.2, 0.3, 0.4]:
        candidates.append((f"gauss_{sf}", _gaussian_init(n_default, sf)))

    for pw in [0.5, 1.0, 2.0]:
        candidates.append((f"taper_{pw}", _tapered_init(n_default, pw)))

    # Random initializations
    for ri in range(3):
        f_r = rng.random(n_default)
        f_rn = _normalize(f_r)
        if f_rn is not None:
            candidates.append((f"rand_{ri}", f_rn))

    # Perturbed parent
    f_pert = parent + rng.random(n_default) * 0.1 * np.max(parent)
    f_pert_n = _normalize(f_pert)
    if f_pert_n is not None:
        candidates.append(("parent_pert", f_pert_n))

    print(f"Phase 1: {len(candidates)} candidates, MD + continuation L-BFGS")

    # Phase 1: Quick screen with mirror descent, then continuation on best
    md_results = []
    time_per_candidate = min(15.0, max(5.0, (deadline - time.monotonic()) / len(candidates) * 0.4))

    for idx, (name, f0) in enumerate(candidates):
        if time.monotonic() > deadline - 120:
            break

        n_it = int(max(300, min(3000, time_per_candidate * 200)))
        t0 = time.monotonic()
        f_md, s_md = _mirror_descent(f0, n_it, 0.5, 0.7)
        elapsed = time.monotonic() - t0
        md_results.append((s_md, name, f_md))
        print(f"    [{idx+1}/{len(candidates)}] {name}: MD score={s_md:.6f} ({elapsed:.1f}s)")

        if s_md < best_score:
            best_score = s_md
            best_f = f_md.copy()
            print(f"    ** New best: {best_score:.6f} from {name} **")

    md_results.sort(key=lambda x: x[0])

    # Phase 2: Continuation L-BFGS on top candidates
    print(f"\nPhase 2: Continuation L-BFGS on top {min(5, len(md_results))} candidates")
    top_k = min(5, len(md_results))
    cont_results = []

    for r in range(top_k):
        if time.monotonic() > deadline - 60:
            break
        s_cur, name, f_cur = md_results[r]
        n_eff = len(f_cur)

        # Estimate tau range from current h
        h_cur = _fft_convolve_full(f_cur)
        h_max = float(np.max(h_cur))
        h_mean = float(np.mean(h_cur))
        tau_start = max(h_max, 0.1 * h_max + 0.01)
        tau_end = max(0.005 * h_max, 1e-5)

        remaining = deadline - time.monotonic() - 30
        time_budget = min(remaining * 0.15, 120.0)
        if time_budget < 5:
            continue

        n_steps = 10
        max_iter = 200

        t0 = time.monotonic()
        f_cont, s_cont = _continuation_lbfgs(
            f_cur, tau_start, tau_end, n_steps, max_iter,
            time_budget=time_budget, verbose=True
        )
        elapsed = time.monotonic() - t0
        cont_results.append((s_cont, name + "_cont", f_cont))
        print(f"    {name}: cont score={s_cont:.6f} ({elapsed:.1f}s)")

        if s_cont < best_score:
            best_score = s_cont
            best_f = f_cont.copy()
            print(f"    ** New best: {best_score:.6f} **")

    cont_results.sort(key=lambda x: x[0])

    # Phase 3: Further refinement on the single best
    print(f"\nPhase 3: Deep refinement")
    if cont_results:
        s_best, name_best, f_best = cont_results[0]
        # More continuation steps with smaller tau end
        h_best = _fft_convolve_full(f_best)
        h_max_best = float(np.max(h_best))
        remaining = deadline - time.monotonic() - 20
        time_budget = min(remaining * 0.3, 180.0)
        if time_budget > 10:
            t0 = time.monotonic()
            f_deep, s_deep = _continuation_lbfgs(
                f_best, 0.05 * h_max_best, 0.001 * h_max_best,
                12, 300, time_budget=time_budget, verbose=True
            )
            elapsed = time.monotonic() - t0
            print(f"    Deep cont: score={s_deep:.6f} ({elapsed:.1f}s)")
            if s_deep < best_score:
                best_score = s_deep
                best_f = f_deep.copy()
                print(f"    ** New best: {best_score:.6f} **")

    # Phase 4: Smooth gradient descent polish on best
    print(f"\nPhase 4: Smooth gradient descent polish")
    remaining = deadline - time.monotonic() - 15
    if remaining > 20:
        h_best = _fft_convolve_full(best_f)
        h_max_best = float(np.max(h_best))
        tau_scale = 0.1 * h_max_best
        n_iter = int(min(5000, remaining * 100))
        t0 = time.monotonic()
        f_polish, s_polish = _smooth_descent(best_f, n_iter, tau_scale,
                                            time_budget=remaining - 10)
        elapsed = time.monotonic() - t0
        print(f"    Polish: score={s_polish:.6f} ({elapsed:.1f}s)")
        if s_polish < best_score:
            best_score = s_polish
            best_f = f_polish.copy()
            print(f"    ** New best: {best_score:.6f} **")

    # Phase 5: Try a few different sequence lengths
    print(f"\nPhase 5: Different lengths")
    remaining = deadline - time.monotonic() - 10
    if remaining > 30:
        for n_try in [256, 1024]:
            if remaining < 20:
                break
            f_try = _recurrence_init(n_try)
            # Quick MD
            f_try2, s_try = _mirror_descent(f_try, 2000, 0.5, 0.7)
            # Quick continuation
            h_try = _fft_convolve_full(f_try2)
            hm = float(np.max(h_try))
            time_b = min(remaining * 0.3, 60.0)
            if time_b > 5:
                f_try3, s_try2 = _continuation_lbfgs(
                    f_try2, hm, 0.005 * hm, 8, 150,
                    time_budget=time_b, verbose=False
                )
            else:
                f_try3, s_try2 = f_try2, s_try

            print(f"    n={n_try}: score={s_try2:.6f}")
            if s_try2 < best_score:
                best_score = s_try2
                best_f = f_try3.copy()
                print(f"    ** New best: {best_score:.6f} (n={n_try}) **")

            remaining = deadline - time.monotonic() - 10

    # Phase 6: Final mirror descent polish on the best
    remaining = deadline - time.monotonic() - 5
    if remaining > 10:
        n_it_final = int(min(5000, remaining * 200))
        f_final, s_final = _mirror_descent(best_f, n_it_final, 0.15, 0.85)
        if s_final < best_score:
            best_score = s_final
            best_f = f_final.copy()
            print(f"    Final MD polish: {best_score:.6f}")

    # Final verification
    final_list = [float(x) for x in best_f]
    final_score = float(evaluate_sequence(final_list))
    print(f"\nFinal verified score: {final_score:.6f} (n={len(final_list)})")

    if not np.isfinite(final_score) or final_score > 2.0:
        print("  Falling back to parent.")
        return [float(x) for x in parent]

    return final_list
```
