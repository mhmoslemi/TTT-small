"""
Multi-method search for nonnegative step function minimizing 2*n*max(conv(f,f))/(sum(f))^2.
Combines LP-based direction finding, spectral flattening, and coordinate descent polishing.
Uses multi-scale approach: optimize at coarser lengths first, then upscale and refine.

import time
import numpy as np
from scipy.optimize import linprog
from scipy.sparse import csr_matrix
from scipy.fft import fft, ifft


def normalize_sequence(arr):
    """Return a nonnegative floating vector normalized to sum 1, or None."""
    try:
        x = np.asarray(arr, dtype=np.float64).ravel()
    except Exception:
        return None
    if x.size == 0:
        return None
    x = np.where(np.isfinite(x), x, 0.0)
    x = np.maximum(x, 0.0)
    s = float(x.sum())
    if s <= 1e-300:
        return None
    x /= s
    return x


def score_normalized(x, n):
    """Score assuming x is nonnegative and sum(x) = 1."""
    conv = np.convolve(x, x)
    m = float(np.max(conv))
    if not np.isfinite(m):
        return float("inf")
    return float(2.0 * n * m)


def make_conv_template(n):
    """Precompute the CSR matrix structure for convolve(f, g)."""
    m = 2 * n - 1
    k64 = np.arange(m, dtype=np.int64)
    js = np.maximum(0, k64 - n + 1).astype(np.int32)
    je = np.minimum(n - 1, k64).astype(np.int32)
    counts = (je.astype(np.int64) - js.astype(np.int64) + 1).astype(np.int64)
    total = int(counts.sum())

    cum = np.cumsum(counts)
    starts = np.repeat(cum - counts, counts)
    local = np.arange(total, dtype=np.int64) - starts

    row = np.repeat(k64, counts).astype(np.int32)
    col_base = js[row.astype(np.int64)]
    col = (col_base + local).astype(np.int32)
    offsets = (row.astype(np.int64) - col.astype(np.int64)).astype(np.int32)

    A_off = csr_matrix((offsets, (row, col)), shape=(m, n), dtype=np.int32)
    edge_offsets = A_off.data.astype(np.int32, copy=True)
    A_template = A_off.astype(np.float64).copy()
    A_template.data.fill(0.0)

    c_obj = -np.ones(n, dtype=np.float64)
    bounds = [(0.0, None)] * n
    return A_template, edge_offsets, c_obj, bounds, m


def solve_direction_lp(A_template, edge_offsets, c_obj, bounds, n, f, M):
    """Solve the LP direction problem."""
    lp_scale = float(n)
    A = A_template.copy()
    A.data[:] = f[edge_offsets]
    A.data *= lp_scale
    b = np.full(2*n-1, M * lp_scale * lp_scale, dtype=np.float64)
    return linprog(c_obj, A_ub=A, b_ub=b, bounds=bounds, method="highs",
                   options={"presolve": True, "time_limit": 60})

    best_f = f.copy()

        if s < 1e-10:

            best_score = sc
            best_f = f_new.copy()
            f = f_new.copy()
        else:
            # Try with original f
            pass
    
    return best_f

"""
Multi-method search for nonnegative step function minimizing 2*n*max(conv(f,f))/(sum(f))^2.
Combines LP-based direction finding (improved), spectral flattening, coordinate descent polishing,
and multi-scale upscaling. Uses diverse parametric initializations and periodic perturbations.
"""

import time
import numpy as np
from scipy.optimize import linprog
from scipy.sparse import csr_matrix
from scipy.fft import fft, ifft


def normalize_sequence(arr):
    """Return a nonnegative floating vector normalized to sum 1, or None."""
    try:
        x = np.asarray(arr, dtype=np.float64).ravel()
    except Exception:
        return None
    if x.size == 0:
        return None
    x = np.where(np.isfinite(x), x, 0.0)
    x = np.maximum(x, 0.0)
    s = float(x.sum())
    if s <= 1e-300:
        return None
    x /= s
    return x


def score_normalized(x, n):
    """Score assuming x is nonnegative and sum(x) = 1."""
    conv = np.convolve(x, x)
    m = float(np.max(conv))
    if not np.isfinite(m):
        return float("inf")
    return float(2.0 * n * m)


def make_conv_template(n):
    """Precompute the CSR matrix structure for convolve(f, g)."""
    m = 2 * n - 1
    k64 = np.arange(m, dtype=np.int64)
    js = np.maximum(0, k64 - n + 1).astype(np.int32)
    je = np.minimum(n - 1, k64).astype(np.int32)
    counts = (je.astype(np.int64) - js.astype(np.int64) + 1).astype(np.int64)
    total = int(counts.sum())

    cum = np.cumsum(counts)
    starts = np.repeat(cum - counts, counts)
    local = np.arange(total, dtype=np.int64) - starts

    row = np.repeat(k64, counts).astype(np.int32)
    col_base = js[row.astype(np.int64)]
    col = (col_base + local).astype(np.int32)
    offsets = (row.astype(np.int64) - col.astype(np.int64)).astype(np.int32)

    A_off = csr_matrix((offsets, (row, col)), shape=(m, n), dtype=np.int32)
    edge_offsets = A_off.data.astype(np.int32, copy=True)
    A_template = A_off.astype(np.float64).copy()
    A_template.data.fill(0.0)

    c_obj = -np.ones(n, dtype=np.float64)
    bounds = [(0.0, None)] * n
    return A_template, edge_offsets, c_obj, bounds, m


def solve_direction_lp(A_template, edge_offsets, c_obj, bounds, n, f, M, time_limit=30):
    """Solve the LP direction problem."""
    lp_scale = float(n)
    A = A_template.copy()
    A.data[:] = f[edge_offsets]
    A.data *= lp_scale
    b = np.full(2 * n - 1, M * lp_scale * lp_scale, dtype=np.float64)
    return linprog(c_obj, A_ub=A, b_ub=b, bounds=bounds, method="highs",
                   options={"presolve": True, "time_limit": time_limit})


def parametric_family(n, family="raised_cosine", **kwargs):
    """Generate parametric families of non-negative sequences."""
    j = np.arange(n, dtype=np.float64)
    if family == "uniform":
        f = np.ones(n)
    elif family == "raised_cosine":
        f = 0.5 * (1 + np.cos(2 * np.pi * (j + 0.5) / n))
        f = np.maximum(f, 0)
    elif family == "sine_sq":
        f = np.sin(np.pi * (j + 0.5) / n) ** 2
    elif family == "gaussian":
        sigma = kwargs.get("sigma", 0.25)
        center = (n - 1) / 2.0
        f = np.exp(-((j - center) / (sigma * n)) ** 2)
    elif family == "triangular":
        center = (n - 1) / 2.0
        f = np.maximum(0, 1 - 2 * np.abs(j - center) / (n - 1))
    elif family == "polynomial":
        p = kwargs.get("p", 2)
        x = 2 * j / (n - 1) - 1
        f = np.maximum(0, 1 - x ** 2) ** p
    elif family == "sine_power":
        p = kwargs.get("p", 1.5)
        f = np.sin(np.pi * (j + 0.5) / n) ** p
    elif family == "waterline":
        # Flat top with smooth edges
        w = kwargs.get("w", 0.1)
        x = j / (n - 1.0)
        f = np.ones(n)
        edge = int(w * n)
        if edge > 0:
            for i in range(edge):
                t = i / edge
                f[i] = 0.5 * (1 + np.cos(np.pi * t))
                f[n - 1 - i] = f[i]
    else:
        f = np.ones(n)
    return normalize_sequence(f)


def interpolate_sequence(x, n_new):
    """Resample a nonnegative sequence to new length and normalize."""
    x = normalize_sequence(x)
    if x is None or n_new <= 0:
        return None
    n_old = x.size
    if n_old == n_new:
        return x
    if n_old == 1:
        return np.ones(n_new, dtype=np.float64) / float(n_new)
    old_pos = np.linspace(0.0, 1.0, n_old)
    new_pos = np.linspace(0.0, 1.0, n_new)
    out = np.interp(new_pos, old_pos, x)
    return normalize_sequence(out)


def mix_noisy(base, rng, eps):
    """Mix a base vector with a random Dirichlet vector."""
    f = normalize_sequence(base)
    if f is None:
        return None
    n = f.size
    if eps <= 1e-12:
        return f
    try:
        mix = rng.dirichlet(np.ones(n, dtype=np.float64))
    except Exception:
        mix = np.ones(n, dtype=np.float64) / float(n)
    if eps >= 1.0:
        f = mix
    else:
        f = (1.0 - eps) * f + eps * mix
    return normalize_sequence(f)


def spectral_shape(f, n_iter=100, alpha=0.7, blend=0.3, rng=None):
    """
    Spectral shaping: iteratively flatten the magnitude spectrum to reduce
    peak autocorrelation. Uses FFT-based approach with zero-padding for linear conv.
    """
    f = normalize_sequence(f)
    if f is None:
        return None
    n = f.size
    best_f = f.copy()
    best_score = score_normalized(f, n)
    n_fft = 2 * n  # zero-pad for linear convolution

    for it in range(n_iter):
        f_pad = np.zeros(n_fft, dtype=np.float64)
        f_pad[:n] = f
        F = fft(f_pad)
        mag = np.abs(F)
        phase = np.angle(F)

        # Compress dynamic range of magnitude
        mag_compressed = mag ** alpha
        # Blend with mean to push toward flat
        mean_mag = np.mean(mag_compressed)
        mag_new = (1 - blend) * mag_compressed + blend * mean_mag
        mag_new = np.maximum(mag_new, 1e-12)

        F_new = mag_new * np.exp(1j * phase)
        f_new = np.real(ifft(F_new))[:n]
        f_new = np.maximum(f_new, 0.0)
        s = f_new.sum()
        if s < 1e-10:
            break
        f_new /= s

        sc = score_normalized(f_new, n)
        if sc < best_score:
            best_score = sc
            best_f = f_new.copy()
            f = f_new.copy()
        else:
            # Degrade: try slightly different parameters
            if it % 10 == 0 and it > 0:
                alpha = max(0.3, alpha - 0.05)
                blend = min(0.7, blend + 0.05)

    return best_f


def coordinate_descent_fast(f, n_max_sweeps=50):
    """
    Efficient coordinate descent. When changing f[j] by delta, only O(n)
    convolution coefficients change, allowing fast max computation.
    """
    f = normalize_sequence(f)
    if f is None:
        return None
    n = f.size
    best_score = score_normalized(f, n)
    conv = np.convolve(f, f).copy()
    current_max = float(np.max(conv))

    for sweep in range(n_max_sweeps):
        improved = False
        for j in range(n):
            old_val = f[j]
            if old_val < 1e-15:
                # Try small positive values
                candidates = [1e-4, 1e-3, 1e-2]
            else:
                candidates = [old_val * 0.3, old_val * 0.6, old_val * 1.0,
                              old_val * 1.4, old_val * 1.8]

            best_sc = best_score
            best_v = old_val

            for v in candidates:
                delta = v - old_val
                if abs(delta) < 1e-14:
                    continue
                # Compute new conv for affected positions
                k_range = np.arange(max(0, j), min(n - 1 + j, 2 * n - 1))
                # For each k in range, (f*f)[k] changes
                # delta * f[k-j] appears twice (once from f[j] in first factor, once in second)
                # except when k = 2j (then it's delta*2*f[j] + delta^2)
                k_minus_j = k_range - j  # these are valid indices 0..n-1
                changes = 2.0 * delta * f[k_minus_j]
                if j in k_range:
                    # k = j means k-j = 0, and also the second term
                    # Actually k=2j: the diagonal term
                    pass
                # Handle the diagonal: when k = 2*j
                if 2 * j < 2 * n - 1:
                    idx_in_range = np.where(k_range == 2 * j)[0]
                    if len(idx_in_range) > 0:
                        changes[idx_in_range[0]] = 2.0 * delta * f[j] + delta * delta

                new_conv = conv.copy()
                new_conv[k_range] += changes
                new_max = float(np.max(new_conv))
                # Renormalize: sum changes by delta
                new_sum = f.sum() + delta
                sc = 2.0 * n * new_max / (new_sum ** 2)

                if sc < best_sc:
                    best_sc = sc
                    best_v = v
                    best_new_conv = new_conv
                    best_new_sum = new_sum

            if best_v != old_val and best_sc < best_score:
                f[j] = best_v
                conv = best_new_conv
                f_sum = best_new_sum
                # Renormalize
                f = f / f_sum
                conv = conv / (f_sum ** 2)
                best_score = best_sc
                improved = True

        if not improved:
            break

    return f


def optimize_lp(n, f_init, time_budget, rng, template, perturb_every=15, max_iter=500):
    """
    LP-based optimization with improved step sizes and periodic perturbations.
    """
    f = normalize_sequence(f_init)
    if f is None:
        f = np.ones(n, dtype=np.float64) / float(n)
    n = f.size

    A_template, edge_offsets, c_obj, bounds = template[0], template[1], template[2], template[3]

    best_f = f.copy()
    best_score = score_normalized(f, n)
    start_time = time.monotonic()
    deadline = start_time + max(1.0, float(time_budget))
    stall = 0
    iteration = 0

    while time.monotonic() < deadline and iteration < max_iter:
        iteration += 1
        t0 = time.monotonic()

        conv_ff = np.convolve(f, f)
        M = float(np.max(conv_ff))
        if not np.isfinite(M) or M <= 0.0:
            break

        try:
            res = solve_direction_lp(A_template, edge_offsets, c_obj, bounds, n, f, M,
                                     time_limit=min(30, max(5, (deadline - time.monotonic()) * 0.3)))
        except Exception:
            break

        x = getattr(res, "x", None)
        if x is None:
            stall += 1
            if stall > 15:
                break
            f = mix_noisy(best_f, rng, 0.3)
            continue

        x = np.asarray(x, dtype=np.float64).ravel()
        x = np.where(np.isfinite(x), x, 0.0)
        x = np.maximum(x, 0.0)
        s_star = float(x.sum())
        if s_star <= 1e-300:
            stall += 1
            if stall > 15:
                break
            f = mix_noisy(best_f, rng, 0.3)
            continue

        g = x / s_star

        r = np.convolve(f, g)
        r_max = float(np.max(r))
        if not np.isfinite(r_max) or r_max <= 0.0:
            stall += 1
            if stall > 15:
                break
            f = mix_noisy(best_f, rng, 0.3)
            continue

        if r_max >= M * (1.0 - 1e-9):
            stall += 1
            if stall > 15:
                break
            f = mix_noisy(best_f, rng, 0.3)
            continue

        G = float(np.max(np.convolve(g, g)))
        if not np.isfinite(G):
            stall += 1
            if stall > 15:
                break
            f = mix_noisy(best_f, rng, 0.3)
            continue

        # Safe step: H(t) = (1-t)^2*M + 2t(1-t)*r_max + t^2*G <= M
        a_coef = 2.0 * (r_max - M)
        b_coef = G + M - 2.0 * r_max
        if b_coef > 1e-14:
            t_max = -a_coef / b_coef
        else:
            t_max = 1.0

        if not np.isfinite(t_max) or t_max <= 0.0:
            t_max = 1e-4
        t_max = min(1.0, t_max)

        # Try multiple step sizes, prefer larger ones for faster convergence
        ts_candidates = [t_max, 0.8 * t_max, 0.5 * t_max, 0.3 * t_max, 0.1 * t_max, 1e-3, 1e-5]
        ts_candidates = sorted(set(max(1e-10, min(1.0, t)) for t in ts_candidates), reverse=True)

        best_cand_f = None
        best_cand_score = float("inf")

        for tv in ts_candidates:
            f_new = (1.0 - tv) * f + tv * g
            f_new = np.maximum(f_new, 0.0)
            ssum = float(f_new.sum())
            if ssum < 1e-300:
                continue
            f_new /= ssum
            sc = score_normalized(f_new, n)
            if np.isfinite(sc) and sc < best_cand_score:
                best_cand_score = sc
                best_cand_f = f_new.copy()

        if best_cand_f is not None and best_cand_score < best_score * (1 - 1e-10):
            f = best_cand_f
            best_f = f.copy()
            best_score = best_cand_score
            stall = 0
            if iteration <= 3 or iteration % 20 == 0:
                print(f"    n={n} iter={iteration} score={best_score:.6f} t={tv:.4f} time={time.monotonic()-t0:.1f}s",
                      flush=True)
        else:
            stall += 1
            # Periodic perturbation to escape local minima
            if stall >= perturb_every or stall % 5 == 0:
                eps = 0.15 + 0.1 * (stall / perturb_every)
                f = mix_noisy(best_f, rng, min(eps, 0.6))
            if stall > 30:
                break

    return best_f, float(best_score)


def try_parametric_starts(n, rng, time_budget):

            f = parametric_family(n, name, **params)
            if f is None:
                continue
            sc = score_normalized(f, n)
            if sc < best_score:
                best_score = sc
                best_f = f.copy()
        except Exception:
            continue

    if best_f is None:
        best_f = np.ones(n) / n
        best_score = 2.0

    # Try some random Dirichlet starts too
    for _ in range(5):
        try:
            f = rng.dirichlet(np.ones(n) * 0.5)
            sc = score_normalized(f, n)
            if sc < best_score:
                best_score = sc
                best_f = f.copy()
        except Exception:
            pass

    return best_f, best_score


def propose_candidate(seed=42, budget_s=1500, **kwargs):
    """
    Multi-method search: parametric initialization, LP optimization, spectral shaping,
    coordinate descent polishing, and multi-scale upscaling.
    """
    rng = np.random.default_rng(int(seed) if seed is not None else 42)
    total_budget = max(10.0, float(budget_s))
    reserve = 20.0
    start_time = time.monotonic()
    deadline = start_time + max(1.0, total_budget - reserve)

    best_f = None
    best_score = float("inf")

    # Phase 0: Score the given sequence if available
    given_f = None
    try:
        raw = np.asarray(height_sequence_1, dtype=np.float64)
        given_f = normalize_sequence(raw)
        if given_f is not None:
            best_f = given_f.copy()
            best_score = score_normalized(given_f, given_f.size)
            print(f"Given sequence: n={given_f.size}, score={best_score:.6f}", flush=True)
    except Exception:
        given_f = None

    if best_f is None:
        best_f = np.ones(128) / 128
        best_score = 2.0

    # Phase 1: Quick parametric search at several lengths
    print("\n=== Phase 1: Parametric initialization ===", flush=True)
    param_budget = min(60.0, (deadline - time.monotonic()) * 0.1)
    for n_try in [128, 256, 512, 1024]:
        rem = deadline - time.monotonic()
        if rem < 5.0:
            break
        t0 = time.monotonic()
        f_try, s_try = try_parametric_starts(n_try, rng, 5.0)
        if s_try < best_score:
            best_score = s_try
            best_f = f_try.copy()
            print(f"  Parametric n={n_try}: score={s_try:.6f} (new best)", flush=True)
        else:
            print(f"  Parametric n={n_try}: score={s_try:.6f}", flush=True)
        if time.monotonic() - t0 > param_budget:
            break

    # Phase 2: LP optimization at multiple scales
    print("\n=== Phase 2: LP optimization ===", flush=True)
    n_current = best_f.size

    # Determine the optimal n to work at
    lp_lengths = []
    if n_current <= 256:
        lp_lengths = [256, 512, 1024, 2048]
    elif n_current <= 512:
        lp_lengths = [512, 1024, 2048]
    elif n_current <= 1024:
        lp_lengths = [1024, 2048]
    else:
        lp_lengths = [n_current, 2048]

    # Budget allocation for LP phases
    total_lp_budget = max(10.0, (deadline - time.monotonic()) * 0.55)
    n_phases = len(lp_lengths)
    # Weight larger n more heavily
    weights = [1.0 + 0.3 * i for i in range(n_phases)]
    total_w = sum(weights)
    weights = [w / total_w for w in weights]

    for idx, n_lp in enumerate(lp_lengths):
        rem = deadline - time.monotonic()
        if rem < 15.0:
            break
        tb = total_lp_budget * weights[idx]
        tb = min(tb, rem - 10.0)
        if tb < 10.0:
            continue

        print(f"\n  LP at n={n_lp}, budget={tb:.1f}s", flush=True)

        try:
            template = make_conv_template(n_lp)
        except MemoryError:
            print(f"  Memory error for n={n_lp}, skipping", flush=True)
            continue
        except Exception as e:
            print(f"  Template error for n={n_lp}: {e}", flush=True)
            continue

        # Create multiple starts
        starts = []
        # Upsample current best
        f_up = interpolate_sequence(best_f, n_lp)
        starts.append(f_up)
        # Perturbed version
        starts.append(mix_noisy(f_up, rng, 0.1))
        # Parametric best at this length
        f_param, _ = try_parametric_starts(n_lp, rng, 3.0)
        starts.append(f_param)
        # Another perturbation
        starts.append(mix_noisy(f_up, rng, 0.2))

        # Split budget among starts (more to first, which is the upsampled best)
        m = len(starts)
        start_weights = [1.0 / (i + 1) for i in range(m)]
        sw_total = sum(start_weights)
        start_weights = [w / sw_total for w in start_weights]

        for s_idx, init in enumerate(starts):
            rem = deadline - time.monotonic()
            if rem < 5.0:
                break
            s_tb = tb * start_weights[s_idx]
            s_tb = min(s_tb, rem - 3.0)
            if s_tb < 3.0:
                continue

            f_res, s_res = optimize_lp(n_lp, init, s_tb, rng, template)
            if np.isfinite(s_res) and s_res < best_score:
                best_score = s_res
                best_f = f_res.copy()
                print(f"    New best: n={best_f.size}, score={best_score:.6f}", flush=True)

        try:
            del template
        except Exception:
            pass

    # Phase 3: Spectral shaping refinement
    print("\n=== Phase 3: Spectral shaping ===", flush=True)
    rem = deadline - time.monotonic()
    if rem > 30.0:
        n_spec = best_f.size
        spec_budget = min(rem * 0.25, 300.0)

        # Try different alpha and blend parameters
        for alpha in [0.5, 0.6, 0.7, 0.8]:
            for blend in [0.2, 0.3, 0.4]:
                rem = deadline - time.monotonic()
                if rem < 5.0:
                    break
                n_iter = max(20, int(spec_budget / 6 / 2 / 0.05))
                f_spec = spectral_shape(best_f, n_iter=n_iter, alpha=alpha, blend=blend, rng=rng)
                if f_spec is not None:
                    s_spec = score_normalized(f_spec, n_spec)
                    if s_spec < best_score:
                        best_score = s_spec
                        best_f = f_spec.copy()
                        print(f"  Spectral (alpha={alpha}, blend={blend}): score={best_score:.6f} (new best)",
                              flush=True)
                    else:
                        print(f"  Spectral (alpha={alpha}, blend={blend}): score={s_spec:.6f}", flush=True)

    # Phase 4: Coordinate descent polishing
    print("\n=== Phase 4: Coordinate descent polishing ===", flush=True)
    rem = deadline - time.monotonic()
    if rem > 20.0 and best_f.size <= 2048:
        cd_budget = min(rem * 0.3, 200.0)
        n_sweeps = int(cd_budget / 0.5)  # rough estimate
        n_sweeps = max(10, min(n_sweeps, 100))
        f_cd = coordinate_descent_fast(best_f, n_max_sweeps=n_sweeps)
        if f_cd is not None:
            s_cd = score_normalized(f_cd, best_f.size)
            if s_cd < best_score:
                best_score = s_cd
                best_f = f_cd.copy()
                print(f"  After CD: score={best_score:.6f} (new best)", flush=True)
            else:
                print(f"  After CD: score={s_cd:.6f}", flush=True)

    # Phase 5: Final random perturbation + short re-optimization
    print("\n=== Phase 5: Final refinement ===", flush=True)
    rem = deadline - time.monotonic()
    if rem > 30.0 and best_f.size <= 2048:
        n_final = best_f.size
        try:
            template = make_conv_template(n_final)
        except Exception:
            template = None

        if template is not None:
            # Try a few random perturbations + short LP
            for attempt in range(5):
                rem = deadline - time.monotonic()
                if rem < 10.0:
                    break
                eps = 0.05 + 0.05 * attempt
                f_pert = mix_noisy(best_f, rng, eps)
                f_res, s_res = optimize_lp(n_final, f_pert, min(rem * 0.2, 30.0), rng, template)
                if np.isfinite(s_res) and s_res < best_score:
                    best_score = s_res
                    best_f = f_res.copy()
                    print(f"  Final attempt {attempt}: score={best_score:.6f} (new best)", flush=True)

    # Phase 6: Try upscaling the best to a larger n
    print("\n=== Phase 6: Upscale attempt ===", flush=True)
    rem = deadline - time.monotonic()
    if rem > 60.0:
        n_up = max(2048, best_f.size * 2)
        if n_up <= 4096:
            print(f"  Upscaling to n={n_up}", flush=True)
            f_up = interpolate_sequence(best_f, n_up)
            try:
                template = make_conv_template(n_up)
                tb = min(rem * 0.4, 180.0)
                f_res, s_res = optimize_lp(n_up, f_up, tb, rng, template)
                if np.isfinite(s_res) and s_res < best_score:
                    best_score = s_res
                    best_f = f_res.copy()
                    print(f"  Upscaled: n={n_up}, score={best_score:.6f} (new best)", flush=True)
                # Spectral shaping on upscaled
                rem2 = deadline - time.monotonic()
                if rem2 > 30.0:
                    f_spec = spectral_shape(f_res if np.isfinite(s_res) else f_up,
                                            n_iter=50, alpha=0.7, blend=0.3)
                    if f_spec is not None:
                        s_spec = score_normalized(f_spec, n_up)
                        if s_spec < best_score:
                            best_score = s_spec
                            best_f = f_spec.copy()
                            print(f"  Upscaled+spectral: score={best_score:.6f} (new best)", flush=True)
            except Exception as e:
                print(f"  Upscale failed: {e}", flush=True)

    # Final output
    if best_f is None:
        best_f = np.ones(64) / 64

    seq = [float(x) for x in best_f]
    print(f"\n=== FINAL: n={len(seq)}, local_best_score={best_score:.6f} ===", flush=True)
    print(f"Total time used: {time.monotonic() - start_time:.1f}s", flush=True)

    # Verify with actual evaluator
    try:
        final_score = float(evaluate_sequence(seq))
        print(f"evaluate_sequence score: {final_score:.6f}", flush=True)
    except Exception as e:
        print(f"evaluate_sequence failed: {e}", flush=True)
        # Fallback
        if given_f is not None:
            seq = [float(x) for x in given_f]
        else:
            seq = [1.0 / 64.0] * 64

    return seq
