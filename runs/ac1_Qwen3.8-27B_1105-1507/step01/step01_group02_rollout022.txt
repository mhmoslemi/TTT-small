"""
Hybrid DE + smooth gradient optimization for minimizing autocorrelation peak.
"""

import numpy as np
import time

def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    N = 1 << (size - 1).bit_length()
    fa = np.fft.rfft(a, N)
    fc = fa * fa
    result = np.fft.irfft(fc, N)[:size]
    result[result < 0] = 0.0
    return result

def compute_score(a):
    n = len(a)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    c = fast_convolve(a)
    return 2.0 * n * np.max(c) / (s * s)

def smooth_objective_and_grad(a, t):
    """Returns (smooth_score, gradient) for the smooth approximation."""
    n = len(a)
    s = a.sum()
    c = fast_convolve(a)
    c_max = np.max(c)
    exp_c = np.exp((c - c_max) / t)
    sum_exp = np.sum(exp_c)
    smooth_max = c_max + t * np.log(sum_exp)
    weights = exp_c / sum_exp
    
    grad_max = np.zeros(n)
    active_k = np.where(weights > 1e-4 / sum_exp)[0]
    for k in active_k:
        w = weights[k]
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        grad_max[lo:hi+1] += w * 2.0 * a[k-hi:k-lo+1][::-1]
    
    score = 2.0 * n * smooth_max / (s * s)
    grad = (2.0 * n / (s * s)) * grad_max - (4.0 * n * smooth_max / (s**3)) * np.ones(n)
    return score, grad

def project_simplex(v):
    n = len(v)
    u = np.sort(v)[::-1]
    cv = np.cumsum(u) - 1.0
    rho_candidates = np.where(u * np.arange(1, n + 1) > cv)[0]
    if len(rho_candidates) == 0:
        return np.ones(n) / n
    rho = rho_candidates[-1]
    theta = cv[rho] / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def smooth_descent(a, deadline, t_init=0.001, t_min=1e-5, rng=None):
    """Smooth gradient descent with annealing temperature."""
    if rng is None:
        rng = np.random.default_rng()
    n = len(a)
    a = project_simplex(a.copy())
    a = a / a.sum()
    
    best_score = compute_score(a)
    best_a = a.copy()
    
    t = t_init
    lr = 1.0 / n
    stall = 0
    max_stall = 200
    
    while time.time() < deadline:
        score, grad = smooth_objective_and_grad(a, t)
        
        # Gradient step with projection
        a_new = a - lr * grad
        a = project_simplex(a_new)
        a = a / a.sum()
        
        actual_score = compute_score(a)
        if actual_score < best_score - 1e-10:
            best_score = actual_score
            best_a = a.copy()
            stall = 0
        else:
            stall += 1
        
        # Anneal temperature
        t = max(t * 0.995, t_min)
        
        # Adapt learning rate

            stall = 0
            # Perturbation
            if lr > 1e-7 and time.time() < deadline - 10:
                a = project_simplex(a + rng.normal(0, 0.01/np.sqrt(n), n))
                a = a / a.sum()
        
        # Reset stall periodically
        if stall > max_stall * 3:

    return best_a, best_score

Hybrid optimization: DE for global search + smooth gradient descent (log-sum-exp 
smoothing + Adam) for local refinement. Targets sequences with minimized 
autocorrelation peak under non-negativity constraint.

import numpy as np
import time

def fast_convolve(a):
    """Compute full linear convolution a * a using FFT."""
    n = len(a)
    size = 2 * n - 1
    N = 1 << (size - 1).bit_length()
    fa = np.fft.rfft(a, N)
    fc = fa * fa
    result = np.fft.irfft(fc, N)[:size]
    result[result < 0] = 0.0
    return result

def compute_score(a):
    """Compute 2*n*max(convolve(a,a))/sum(a)^2."""
    n = len(a)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    c = fast_convolve(a)
    return float(2.0 * n * np.max(c) / (s * s))

def project_simplex(v):
    """Project v onto {x >= 0, sum(x) = 1}."""
    n = len(v)
    u = np.sort(v)[::-1]
    cv = np.cumsum(u) - 1.0
    rho_candidates = np.where(u * np.arange(1, n + 1) > cv)[0]
    if len(rho_candidates) == 0:
        return np.ones(n) / n
    rho = rho_candidates[-1]
    theta = cv[rho] / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def smooth_grad(a, t):
    """
    Compute smooth approximation of max(convolve(a,a)) and its gradient.
    Uses log-sum-exp smoothing with temperature t.
    Returns (smooth_max, gradient w.r.t. a).
    """
    n = len(a)
    c = fast_convolve(a)
    c_max = np.max(c)
    shifted = (c - c_max) / t
    # Clip for numerical stability
    shifted = np.clip(shifted, -50, 50)
    exp_c = np.exp(shifted)
    sum_exp = np.sum(exp_c)
    weights = exp_c / sum_exp
    smooth_max = c_max + t * np.log(sum_exp)
    
    # Gradient: d(smooth_max)/d(a[i]) = 2 * sum_k weights[k] * a[k-i] (valid indices)
    grad = np.zeros(n)
    # Use direct computation for active lags
    active = weights > 1e-6
    for k in np.where(active)[0]:
        w = weights[k] * 2.0
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        grad[lo:hi+1] += w * a[k-hi:k-lo+1][::-1]
    
    return smooth_max, grad

    """
    Adam optimizer on the smooth objective with simplex projection.
    """
    n = len(a0)
    a = project_simplex(a0.astype(np.float64).copy())
    best_score = compute_score(a)
    best_a = a.copy()
    
    m = np.zeros(n)  # First moment
    v = np.zeros(n)  # Second moment
    t = t_start
    step = 0
    lr = 0.5 / np.sqrt(n)
    stall = 0
    max_stall = 300
    
    base_lr = 0.5 / np.sqrt(n)
    lr_decay = 0.995
    
    while time.time() < deadline:
        step += 1
        smooth_max, grad_max = smooth_grad(a, t)
        s = a.sum()
        # Full gradient of score = 2n*smooth_max/s^2
        grad = (2.0 * n / (s * s)) * grad_max - (4.0 * n * smooth_max / (s**3)) * np.ones(n)
        
        # Adam update
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad * grad)
        m_hat = m / (1 - beta1**step)
        v_hat = v / (1 - beta2**step)
        
        a_new = a - lr * m_hat / (np.sqrt(v_hat) + eps)
        a = project_simplex(a_new)
        
        # Evaluate
        score = compute_score(a)
        if score < best_score - 1e-12:
            best_score = score
            best_a = a.copy()
            stall = 0
        else:
            stall += 1

        progress = min(1.0, step / 10000.0)

        if step % 500 == 0:

    return best_a, best_score

"""
Hybrid optimization for minimizing 2*n*max(convolve(a,a))/sum(a)^2 over non-negative sequences.
Uses DE for global search, smooth gradient descent (log-sum-exp + Adam) for local refinement,
and subgradient polish. Exploits multiple sequence lengths and the parent construction.
"""

import numpy as np
import time


def fast_convolve(a):
    """Compute full linear convolution a * a using FFT."""
    n = len(a)
    size = 2 * n - 1
    N = 1 << (size - 1).bit_length()
    fa = np.fft.rfft(a, N)
    fc = fa * fa
    result = np.fft.irfft(fc, N)[:size]
    result[result < 0] = 0.0
    return result


def compute_score(a):
    """Compute 2*n*max(convolve(a,a))/sum(a)^2."""
    n = len(a)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    c = fast_convolve(a)
    return float(2.0 * n * np.max(c) / (s * s))


def project_simplex(v):
    """Project v onto {x >= 0, sum(x) = 1}."""
    n = len(v)
    u = np.sort(v)[::-1]
    cv = np.cumsum(u) - 1.0
    rho_candidates = np.where(u * np.arange(1, n + 1) > cv)[0]
    if len(rho_candidates) == 0:
        return np.ones(n) / n
    rho = rho_candidates[-1]
    theta = cv[rho] / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def smooth_max_and_grad(a, t):
    """
    Compute log-sum-exp smoothing of max(convolve(a,a)) and its gradient w.r.t. a.
    Returns (smooth_max_value, gradient_vector).
    """
    n = len(a)
    c = fast_convolve(a)
    c_max = np.max(c)
    shifted = (c - c_max) / t
    shifted = np.clip(shifted, -100, 100)
    exp_c = np.exp(shifted)
    sum_exp = np.sum(exp_c)
    weights = exp_c / sum_exp
    smooth_max = c_max + t * np.log(sum_exp)

    # Gradient: d(smooth_max)/d(a[i]) = 2 * sum_k weights[k] * a[k-i] for valid indices
    grad = np.zeros(n)
    active = weights > 1e-8
    active_indices = np.where(active)[0]
    for k in active_indices:
        w = weights[k] * 2.0
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        grad[lo:hi + 1] += w * a[k - hi:k - lo + 1][::-1]

    return smooth_max, grad


def adam_smooth_optimize(a0, deadline, rng, t_start=0.002, t_end=1e-5,
                         beta1=0.9, beta2=0.999, eps=1e-8):
    """
    Adam optimizer on smooth objective with simplex projection and temperature annealing.
    """
    n = len(a0)
    a = project_simplex(a0.astype(np.float64).copy())
    best_score = compute_score(a)
    best_a = a.copy()

    m = np.zeros(n)
    v = np.zeros(n)
    step = 0
    base_lr = 0.3 / np.sqrt(n)
    lr = base_lr
    stall = 0
    max_stall = 400

    while time.time() < deadline:
        step += 1
        smooth_max, grad_max = smooth_max_and_grad(a, max(t_start * (t_end / t_start) ** (step / 20000.0), t_end))
        s = a.sum()
        if s < 1e-10:
            break

        # Full gradient of 2n*smooth_max/s^2
        grad = (2.0 * n / (s * s)) * grad_max - (4.0 * n * smooth_max / (s ** 3)) * np.ones(n)

        # Adam
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad * grad)
        m_hat = m / (1 - beta1 ** step)
        v_hat = v / (1 - beta2 ** step)

        a_new = a - lr * m_hat / (np.sqrt(v_hat) + eps)
        a = project_simplex(a_new)

        score = compute_score(a)
        if score < best_score - 1e-12:
            best_score = score
            best_a = a.copy()
            stall = 0
        else:
            stall += 1

        if stall > max_stall:
            if step > 5000:
                break
            lr = max(lr * 0.5, base_lr * 1e-5)
            stall = 0
            if time.time() < deadline - 5:
                noise_scale = 0.03 / np.sqrt(n)
                a = project_simplex(a + rng.normal(0, noise_scale, n))

    return best_a, best_score


def subgradient_polish(a0, deadline, rng):
    """
    Subgradient descent with multiple active peaks for final polish.
    """
    n = len(a0)
    a = project_simplex(a0.astype(np.float64).copy())
    best_score = compute_score(a)
    best_a = a.copy()
    lr = 0.4 / n
    stall = 0

    for it in range(1000000):
        if time.time() > deadline:
            break

        c = fast_convolve(a)
        peak = np.max(c)
        threshold = peak * (1.0 - 1e-4)
        peak_indices = np.where(c >= threshold)[0]
        if len(peak_indices) > 20:
            top_k = np.argsort(c[peak_indices])[::-1][:10]
            peak_indices = peak_indices[top_k]

        grad = np.zeros(n)
        for k in peak_indices:
            lo = max(0, k - n + 1)
            hi = min(n - 1, k)
            grad[lo:hi + 1] += 2.0 * a[k - hi:k - lo + 1][::-1]
        grad /= len(peak_indices)

        a_new = a - lr * grad
        a = project_simplex(a_new)

        if it % 500 == 0:
            score = compute_score(a)
            if score < best_score - 1e-12:
                best_score = score
                best_a = a.copy()
                stall = 0
            else:
                stall += 1

            if stall > 20:
                lr = max(lr * 0.7, 1e-10)
                stall = 0
                if lr > 1e-9:
                    a = project_simplex(a + rng.normal(0, 0.02 / np.sqrt(n), n))

    return best_a, best_score


def de_optimize(n, deadline, rng, init=None, pop_size=25, max_gen=150):
    """
    Differential evolution (best/1/bin) on the probability simplex.
    """
    if init is not None:
        pop = [project_simplex(init.astype(np.float64).copy())]
        for i in range(pop_size - 1):
            blend = 0.3 + 0.5 * rng.random()
            a = blend * init + (1 - blend) * rng.random(n)
            a = project_simplex(np.maximum(a, 0))
            pop.append(a)
    else:
        pop = []
        for i in range(pop_size):
            a = rng.random(n)
            a = project_simplex(a)
            pop.append(a)

    scores = [compute_score(p) for p in pop]
    best_idx = int(np.argmin(scores))
    best_score = scores[best_idx]
    best_a = pop[best_idx].copy()

    F = 0.7
    CR = 0.8
    gen = 0

    while gen < max_gen and time.time() < deadline:
        gen += 1
        F = 0.8 - 0.5 * (gen / max(1, max_gen))
        F = max(F, 0.3)

        for i in range(pop_size):
            if time.time() > deadline:
                break

            r1 = int(rng.integers(pop_size))
            while r1 == i:
                r1 = int(rng.integers(pop_size))
            r2 = int(rng.integers(pop_size))
            while r2 == i or r2 == r1:
                r2 = int(rng.integers(pop_size))

            v = pop[best_idx] + F * (pop[r1] - pop[r2])

            mask = rng.random(n) < CR
            if not mask.any():
                mask[int(rng.integers(n))] = True
            trial = np.where(mask, v, pop[i])
            trial = project_simplex(np.maximum(trial, 0))

            trial_score = compute_score(trial)
            if trial_score < scores[i]:
                pop[i] = trial
                scores[i] = trial_score
                if trial_score < best_score:
                    best_score = trial_score
                    best_a = trial.copy()
                    best_idx = i

        if gen % 10 == 0:
            best_idx = int(np.argmin(scores))
            best_score = scores[best_idx]
            best_a = pop[best_idx].copy()

    return best_a, best_score


def alt_projection_init(n, rng, n_iter=40):
    """Alternating projection for flat-spectrum nonnegative initialization."""
    N = 1 << (2 * n - 2).bit_length()
    phases = rng.uniform(0.0, 2.0 * np.pi, N // 2 + 1)
    F = np.exp(1j * phases)
    F[0] = 1.0 + 0j

    a = np.ones(n) / n
    for it in range(n_iter):
        a_full = np.fft.irfft(F, N)
        a = np.maximum(a_full[:n], 0.0)
        s = a.sum()
        a = a / s if s > 1e-10 else np.ones(n) / n
        F_full = np.fft.rfft(a, N)
        mag = np.abs(F_full)
        phase = np.angle(F_full)
        mag_target = np.linspace(1.0, 0.8, len(mag)) ** 0.15
        F = mag_target * np.exp(1j * phase)
        F[0] = F[0].real + 0j

    return a
