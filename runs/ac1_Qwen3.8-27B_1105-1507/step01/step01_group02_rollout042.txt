def active_off_indices(c, n, max_active=25, rel_tol=1e-4):

```python
"""
Hybrid search for low-peak non-negative sequences.

The submitted evaluator is treated as a black box.  A quick diagnostic checks
whether the zero-lag term is included in the maximized convolution.  If it is
included, the uniform sequence is optimal and is returned.  Otherwise the
search minimizes the peak off-zero-lag autocorrelation.

The main optimizer is a smoothed max-likelihood style L-BFGS run on the
simplex via a softmax parameterization.  The smoothed objective is a
log-sum-exp of the off-zero convolution coefficients.  Exact peak values are
tracked throughout.  The result is polished with FFT-based subgradient descent
on the simplex.  Several initializations are tried: the supplied parent
construction, perturbed parent, spectral-flattening alternating projections,
absolute-value FFT initializations, and random candidates.
"""

import time
import math

import numpy as np
from scipy.optimize import minimize


# ---------------------------------------------------------------------------
# Global state used by the smooth L-BFGS objective.
# ---------------------------------------------------------------------------
G_SMOOTH_N = 0
G_SMOOTH_BETA = 1.0
G_SMOOTH_BEST_A = None
G_SMOOTH_BEST_F = float("inf")
G_SMOOTH_ITER = 0


# ---------------------------------------------------------------------------
# Basic helpers.
# ---------------------------------------------------------------------------
def normalize_nonnegative(a):
    """Return a finite non-negative float array with sum 1."""
    a = np.asarray(a, dtype=np.float64)
    if a.size == 0:
        return np.ones(1, dtype=np.float64)
    a = np.where(np.isfinite(a), a, 0.0)
    a = np.maximum(a, 0.0)
    a = np.minimum(a, 1000.0)
    s = float(np.sum(a))
    if not np.isfinite(s) or s <= 0.0:
        return np.ones(a.size, dtype=np.float64) / float(a.size)
    a = a / s
    a[np.isnan(a) | np.isinf(a)] = 0.0
    s2 = float(np.sum(a))
    if s2 <= 0.0:
        return np.ones(a.size, dtype=np.float64) / float(a.size)
    return a / s2


def fast_autocorr(a):
    """Full linear autocorrelation / convolution a * a using FFT."""
    n = int(a.size)
    if n == 0:
        return np.array([0.0], dtype=np.float64)
    size = 2 * n - 1
    N = 1 << max(1, (size - 1).bit_length())
    fa = np.fft.rfft(a, N)
    out = np.fft.irfft(fa * fa, N)[:size]
    out = np.ascontiguousarray(out, dtype=np.float64)
    out[out < 0.0] = 0.0
    return out


def peak_off_value(c, n):
    """Maximum convolution value excluding the central zero-lag coefficient."""
    if n <= 1:
        return 0.0
    off = c.copy()
    off[n - 1] = -np.inf
    m = float(np.max(off))
    if not np.isfinite(m):
        return 0.0
    return m


def proxy_off_score(a):
    """
    Proxy for the evaluator when the central coefficient is ignored:
    2 * n * max_off(conv(a, a)) / sum(a)^2.
    """
    a = np.asarray(a, dtype=np.float64)
    n = int(a.size)
    s = float(np.sum(a))
    if n <= 1 or s <= 0.0 or not np.isfinite(s):
        return float("inf")
    c = fast_autocorr(a)
    m = peak_off_value(c, n)
    return float(2.0 * n * m / (s * s))


def evaluate_array(a):
    """Call the external evaluate_sequence safely."""
    try:
        vals = []
        for x in a:
            vals.append(float(x))
        s = evaluate_sequence(vals)
        if not np.isfinite(s):
            return float("inf")
        return float(s)
    except Exception:
        return float("inf")


def softmax(x):
    """Stable softmax."""
    x = np.asarray(x, dtype=np.float64)
    if x.size == 0:
        return np.ones(1, dtype=np.float64)
    x = x - np.max(x)
    e = np.exp(x)
    s = float(np.sum(e))
    if not np.isfinite(s) or s <= 0.0:
        return np.ones(x.size, dtype=np.float64) / x.size
    return e / s


def active_off_indices(c, n, max_active=18, rel_tol=1e-4):
    """
    Indices of near-maximum off-zero convolution coefficients.
    Symmetric counterparts are included.
    """
    if n <= 1:
        return np.array([], dtype=int), 0.0
    off = c.copy()
    off[n - 1] = -np.inf
    peak = float(np.max(off))
    if not np.isfinite(peak) or peak <= 0.0:
        return np.array([], dtype=int), peak
    threshold = peak * (1.0 - rel_tol)
    idx = np.where(off >= threshold)[0]
    if idx.size == 0:
        return np.array([], dtype=int), peak
    if idx.size > max_active:
        vals = off[idx]
        order = np.argsort(vals)[::-1][:max_active]
        idx = idx[order]

    sym = set()
    total_lags = 2 * n - 2
    for k in idx.tolist():
        sym.add(int(k))
        sym.add(total_lags - int(k))
    sym.discard(n - 1)
    idx2 = np.array(sorted(sym), dtype=int)
    if idx2.size > 2 * max_active:
        vals = off[idx2]
        order = np.argsort(vals)[::-1][:2 * max_active]
        idx2 = idx2[order]
    return idx2, peak


def weighted_grad_fft(a, p):
    """
    Gradient of sum_k p_k * (a*a)_k with respect to a.
    p is length 2n-1.
    """
    n = int(a.size)
    if n <= 1:
        return np.zeros(n, dtype=np.float64)
    p = np.asarray(p, dtype=np.float64)
    if p.size != 2 * n - 1:
        q = np.zeros(2 * n - 1, dtype=np.float64)
        m = min(p.size, q.size)
        q[:m] = p[:m]
        p = q
    L = 3 * n - 2
    N = 1 << max(1, (L - 1).bit_length())
    ar = np.fft.rfft(a, N)
    pr = np.fft.rfft(p[::-1], N)
    conv = np.fft.irfft(ar * pr, N)[:L]
    v = conv[2 * n - 2 :: -1][:n]
    return 2.0 * v


# ---------------------------------------------------------------------------
# Smooth L-BFGS objective on softmax parameters.
# ---------------------------------------------------------------------------
def smooth_fun(x):
    """
    Objective and gradient for minimizing a log-sum-exp of off-zero
    autocorrelation coefficients over a = softmax(x - mean(x)).
    The exact peak off-zero proxy score is also tracked globally.
    """
    x = np.asarray(x, dtype=np.float64)
    n = G_SMOOTH_N
    if n <= 1:
        return (0.0, np.zeros_like(x))

    z = x - np.mean(x)
    a = softmax(z)
    c = fast_autocorr(a)
    off = c.copy()
    off[n - 1] = -np.inf
    m = float(np.max(off))
    if not np.isfinite(m):
        return (float("inf"), np.zeros_like(x))

    s_a = float(np.sum(a))
    if s_a > 0.0 and np.isfinite(s_a):
        exact_score = float(2.0 * n * m / (s_a * s_a))
    else:
        exact_score = float("inf")

    if exact_score < G_SMOOTH_BEST_F:
        G_SMOOTH_BEST_F = exact_score
        G_SMOOTH_BEST_A = a.copy()

    beta = G_SMOOTH_BETA
    scaled = beta * (off - m)
    expv = np.exp(scaled)
    s_exp = float(np.sum(expv))
    if (not np.isfinite(s_exp)) or s_exp <= 0.0:
        finite = np.isfinite(off)
        cnt = int(np.sum(finite))
        if cnt <= 0:
            return (float("inf"), np.zeros_like(x))
        p = np.zeros(2 * n - 1, dtype=np.float64)
        p[finite] = 1.0 / cnt
        F = float(np.mean(off[finite]))
    else:
        p = expv / s_exp
        F = float(m + math.log(s_exp) / beta)

    p[n - 1] = 0.0
    grad_a = weighted_grad_fft(a, p)
    dot = float(np.dot(a, grad_a))
    gz = a * (grad_a - dot)
    gx = gz - np.mean(gz)
    gx += 1e-10 * (x - np.mean(x))
    G_SMOOTH_ITER += 1
    return (F, gx)


def lbfgs_polish(a, deadline, rng):
    """
    Run L-BFGS on softmax parameters with increasing log-sum-exp temperature.
    Returns the best sequence found by exact peak off-zero proxy.
    """
    a = normalize_nonnegative(a)
    n = int(a.size)
    if n <= 1:
        return a

    best_a = a.copy()
    best_s = proxy_off_score(best_a)

    def z_from_array(arr):
        arr = np.asarray(arr, dtype=np.float64)
        z = np.log(np.maximum(arr, 1e-300))
        return z - np.mean(z)

    z = z_from_array(a)
    mults = [0.25, 1.0, 4.0, 16.0, 64.0, 256.0]
    max_attempts = 9

    for attempt in range(max_attempts):
        if time.time() >= deadline - 4.0:
            break

        if attempt < len(mults):
            beta = max(10.0, float(n) * float(mults[attempt]))
        else:
            beta = max(10.0, float(n) * float(mults[-1]))

        if n <= 512:
            maxiter = 90
        elif n <= 900:
            maxiter = 60
        else:
            maxiter = 35

        G_SMOOTH_N = n
        G_SMOOTH_BETA = float(beta)
        G_SMOOTH_BEST_A = None
        G_SMOOTH_BEST_F = float(best_s)
        G_SMOOTH_ITER = 0

        try:
            res = minimize(
                smooth_fun,
                z,
                method="L-BFGS-B",
                jac=True,
                options={
                    "maxiter": maxiter,
                    "ftol": 1e-13,
                    "gtol": 1e-6,
                    "maxcor": 10,
                    "maxls": 20,
                    "disp": 0,
                },
            )
        except Exception as exc:
            print(f"    L-BFGS attempt {attempt} failed: {exc}")
            res = None

        if res is not None and getattr(res, "x", None) is not None:
            a_res = normalize_nonnegative(softmax(np.asarray(res.x, dtype=np.float64) - np.mean(np.asarray(res.x, dtype=np.float64))))
            s_res = proxy_off_score(a_res)
            if np.isfinite(s_res) and s_res < best_s:
                best_s = s_res
                best_a = a_res.copy()

        if G_SMOOTH_BEST_A is not None and G_SMOOTH_BEST_F < best_s:
            best_s = float(G_SMOOTH_BEST_F)
            best_a = np.asarray(G_SMOOTH_BEST_A, dtype=np.float64).copy()

        z = z_from_array(best_a)

        if attempt >= 1 and time.time() < deadline - 8.0:
            sigma = 0.45 * math.exp(-0.18 * float(attempt))
            z = z + rng.normal(0.0, sigma / math.sqrt(float(n)), n)

    return best_a

# Projected subgradient descent for exact max off-zero peak.
# ---------------------------------------------------------------------------
def subgradient_polish(a, deadline, rng, max_iters=25000):
    """
    Short projected subgradient polishing using exact active peaks.
    Operates on the simplex.
    """
    a = normalize_nonnegative(a)
    n = int(a.size)
    if n <= 1:
        return a

    best_a = a.copy()
    best_s = proxy_off_score(best_a)
    base_norm = 0.035
    no_improve = 0

    for it in range(max_iters):
        if time.time() >= deadline:
            break

        c = fast_autocorr(a)
        peak, idx = active_off_indices(c, n, max_active=18)
        s_cur = float(2.0 * n * peak) if np.isfinite(peak) else 0.0

        if s_cur < best_s * (1.0 - 1e-9):
            best_s = s_cur
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        if no_improve > 160:
            base_norm *= 0.6
            if base_norm < 1e-6:
                break
            if time.time() < deadline - 1.0:
                noise = rng.normal(0.0, 6e-4 / math.sqrt(float(n)), n)
                a = normalize_nonnegative(a + noise)
            no_improve = 0

        if idx.size == 0:
            break

        p = np.zeros(2 * n - 1, dtype=np.float64)
        p[idx] = 1.0 / float(idx.size)
        grad = weighted_grad_fft(a, p)
        gnorm = float(np.linalg.norm(grad))
        if not np.isfinite(gnorm) or gnorm < 1e-15:
            break

        step_norm = base_norm / (math.sqrt(float(it + 1)) + 1.0)
        eta = step_norm / gnorm

        trial = normalize_nonnegative(a - eta * grad)
        s_trial = proxy_off_score(trial)
        if np.isfinite(s_trial) and s_trial < s_cur:
            a = trial
            continue

        eta_small = 0.22 * eta
        trial2 = normalize_nonnegative(a - eta_small * grad)
        s_trial2 = proxy_off_score(trial2)
        if np.isfinite(s_trial2) and s_trial2 < s_cur:
            a = trial2
            continue

        a = normalize_nonnegative(a - 0.05 * eta_small * grad)

    return best_a


# ---------------------------------------------------------------------------
# Initializations.
# ---------------------------------------------------------------------------
def spectral_flatten_init(n, rng, n_iter=50, start_a=None, exponent=0.70):
    """
    Alternating projection / spectral leveling initializer.
    Projects from frequency domain onto non-negative normalized vectors.
    """
    n = int(max(2, n))
    size = 2 * n - 1
    N = 1 << max(1, (size - 1).bit_length())

    if start_a is not None:
        a0 = normalize_nonnegative(start_a)
        F = np.fft.rfft(a0, N)
        a = a0.copy()
    else:
        phases = rng.uniform(0.0, 2.0 * np.pi, N // 2 + 1)
        F = np.exp(1j * phases)
        F[0] = 1.0 + 0.0j
        a = np.ones(n, dtype=np.float64) / float(n)

    for _ in range(n_iter):
        afull = np.fft.irfft(F, N)
        a = np.maximum(afull[:n], 0.0)
        a = normalize_nonnegative(a)

        F = np.fft.rfft(a, N)
        mag = np.abs(F)
        mag = np.power(np.maximum(mag, 1e-12), exponent)
        mean_mag = float(np.mean(mag))
        if mean_mag <= 0.0 or not np.isfinite(mean_mag):
            mag = np.ones_like(mag)
        else:
            mag = mag / mean_mag

        phase = np.exp(1j * np.angle(F))
        F = mag * phase
        F[0] = abs(F[0])
        F[-1] = abs(F[-1])

    return a


def abs_ifft_init(n, rng):
    """Non-negative initializer from absolute value of inverse FFT."""
    n = int(max(2, n))
    size = 2 * n - 1
    N = 1 << max(1, (size - 1).bit_length())
    phases = rng.uniform(0.0, 2.0 * np.pi, N // 2 + 1)
    F = np.exp(1j * phases)
    F[0] = 0.0 + 0.0j
    F[-1] = 0.0 + 0.0j
    a = np.abs(np.fft.irfft(F, N)[:n])
    return normalize_nonnegative(a)


def resize_sequence(a, n):
    """Linearly resize a sequence to length n."""
    a = np.asarray(a, dtype=np.float64)
    n = int(max(2, n))
    if a.size == n:
        return normalize_nonnegative(a)
    if a.size == 0:
        return np.ones(n, dtype=np.float64) / float(n)
    x0 = np.linspace(0.0, 1.0, a.size)
    x1 = np.linspace(0.0, 1.0, n)
    new = np.interp(x1, x0, a)
    return normalize_nonnegative(new)


# ---------------------------------------------------------------------------
# Diagnostic and center-mode fallback.
# ---------------------------------------------------------------------------
def calibrate_mode():
    """
    Infer whether the evaluator's max includes the zero-lag coefficient.
    For non-negative sequences the full max is always the zero-lag.
    With a centered one-hot, full max gives ~2n, off-zero gives 0.
    """
    n = 17
    one = [0.0] * n
    one[n // 2] = 1.0
    uni = [1.0] * n
    try:
        s_one = evaluate_sequence(one)
        s_uni = evaluate_sequence(uni)
    except Exception:
        return "center"

    print(f"    diagnostic one-hot score={s_one:.6f}, uniform score={s_uni:.6f}")
    if np.isfinite(s_one) and np.isfinite(s_uni):
        if s_one < 0.5 * max(1.0, float(s_uni)):
            return "off"
        if s_one < 1.0 and s_uni > 1.0:
            return "off"
    return "center"


def best_uniform_sequence(deadline):
    """If zero-lag is included, uniform sequences are optimal. Choose length safely."""
    best_score = float("inf")
    best_seq = None
    for n in [1, 2, 4, 8, 16, 32, 64, 128, 256, 512, 1024, 2048]:
        if time.time() >= deadline:
            break
        try:
            s = evaluate_sequence([1.0] * n)
        except Exception:
            s = float("inf")
        if np.isfinite(s) and s < best_score:
            best_score = float(s)
            best_seq = [1.0] * n
            print(f"    uniform length {n}: score={best_score:.6f}")
    if best_seq is None:
        best_seq = [1.0] * 512
    return best_seq


# ---------------------------------------------------------------------------
# Candidate polishing pipeline.
# ---------------------------------------------------------------------------
def polish_candidate(a, deadline, rng, label):
    """
    Apply L-BFGS, subgradient polishing, and a second L-BFGS pass.
    Returns best sequence and actual evaluator score for this candidate.
    """
    a = normalize_nonnegative(a)
    best_a = a.copy()
    best_s = evaluate_array(best_a)
    if not np.isfinite(best_s):
        best_s = proxy_off_score(best_a)
    print(f"  {label}: initial score={best_s:.6f}, n={len(best_a)}")

    if time.time() < deadline - 6.0:
        rem = float(deadline - time.time())
        dl = min(float(deadline), time.time() + max(8.0, 0.48 * rem))
        a1 = lbfgs_polish(best_a, dl, rng)
        s1 = evaluate_array(a1)
        if np.isfinite(s1) and s1 < best_s:
            best_s = s1
            best_a = a1.copy()
            print(f"    {label}: after L-BFGS score={best_s:.6f}")

    if time.time() < deadline - 6.0:
        rem = float(deadline - time.time())
        dl = min(float(deadline), time.time() + max(5.0, 0.25 * rem))
        a2 = subgradient_polish(best_a, dl, rng, max_iters=22000)
        s2 = evaluate_array(a2)
        if np.isfinite(s2) and s2 < best_s:
            best_s = s2
            best_a = a2.copy()
            print(f"    {label}: after subgradient score={best_s:.6f}")

    if time.time() < deadline - 10.0:
        rem = float(deadline - time.time())
        if rem > 18.0:
            dl = min(float(deadline), time.time() + 0.75 * rem)
            a3 = lbfgs_polish(best_a, dl, rng)
            s3 = evaluate_array(a3)
            if np.isfinite(s3) and s3 < best_s:
                best_s = s3
                best_a = a3.copy()
                print(f"    {label}: after second L-BFGS score={best_s:.6f}")

    if time.time() < deadline - 3.0:
        a4 = subgradient_polish(best_a, deadline, rng, max_iters=8000)
        s4 = evaluate_array(a4)
        if np.isfinite(s4) and s4 < best_s:
            best_s = s4
            best_a = a4.copy()
            print(f"    {label}: after final subgradient score={best_s:.6f}")

    return best_a, best_s


# ---------------------------------------------------------------------------
# Main entry point.
# ---------------------------------------------------------------------------
def propose_candidate(budget_s=1500.0):
    """Search for the best non-negative coefficient sequence."""
    t0 = time.time()
    total_budget = max(60.0, float(budget_s) - 30.0)
    hard_deadline = t0 + total_budget

    print("Starting search.")
    print(f"  budget={budget_s:.1f}s, hard_deadline_offset={total_budget:.1f}s")

    mode = calibrate_mode()
    print(f"  evaluator mode: {mode}")

    if mode == "center":
        seq = best_uniform_sequence(hard_deadline - 10.0)
        final = []
        for x in seq:
            final.append(float(min(1000.0, max(0.0, x))))
        if not final:
            final = [1.0] * 512
        s = evaluate_sequence(final)
        print(f"FINAL center-mode: n={len(final)}, score={s:.6f}")
        return final

    rng = np.random.default_rng(20250514)
    best_array = None
    best_score = float("inf")

    hs = None
    hs_len = 512
    try:
        raw = height_sequence_1
        arr = np.asarray(raw, dtype=np.float64)
        if arr.ndim != 1:
            arr = arr.reshape(-1)
        arr = np.where(np.isfinite(arr), arr, 0.0)
        arr = np.maximum(arr, 0.0)
        if arr.size >= 2 and arr.sum() > 0.0:
            hs = normalize_nonnegative(arr)
            hs_len = int(hs.size)
            print(f"  loaded parent sequence length={hs_len}")
    except Exception:
        pass

    span = max(60.0, total_budget - 30.0)

    def consider(a, s, label):
        nonlocal_best = None
        nonlocal_score = float("inf")
        return None

    # Python top-level no nested? The inline function above would be a closure.
    # Avoid it by updating manually below.

    # Candidate 1: parent sequence.
    if hs is not None:
        dl = t0 + 0.36 * span
        if time.time() < dl - 12.0:
            a, s = polish_candidate(hs, min(dl, hard_deadline - 5.0), rng, "parent")
            if np.isfinite(s) and s < best_score:
                best_score = s
                best_array = a.copy()
                print(f"  >>> best: {best_score:.6f}, n={len(best_array)}")

    # Candidate 2: perturbed parent.
    if hs is not None:
        dl = t0 + 0.55 * span
        if time.time() < dl - 12.0:
            noise = rng.normal(0.0, 0.22 / math.sqrt(float(hs.size)), hs.size)
            a0 = normalize_nonnegative(hs + noise)
            a, s = polish_candidate(a0, min(dl, hard_deadline - 5.0), rng, "parent-noise")
            if np.isfinite(s) and s < best_score:
                best_score = s
                best_array = a.copy()
                print(f"  >>> best: {best_score:.6f}, n={len(best_array)}")

    # Candidate 3: spectral flattening at parent length (or 512).
    L_main = hs_len if hs is not None else 512
    dl = t0 + 0.70 * span
    if time.time() < dl - 12.0:
        start = hs if hs is not None and hs.size == L_main else None
        a0 = spectral_flatten_init(L_main, rng, n_iter=55, start_a=start, exponent=0.68)
        a, s = polish_candidate(a0, min(dl, hard_deadline - 5.0), rng, f"alt-len{L_main}")
        if np.isfinite(s) and s < best_score:
            best_score = s
            best_array = a.copy()
            print(f"  >>> best: {best_score:.6f}, n={len(best_array)}")

    # Candidate 4: second length, usually 384.
    L2 = 384 if L_main != 384 else 640
    dl = t0 + 0.80 * span
    if time.time() < dl - 12.0:
        start = resize_sequence(hs, L2) if hs is not None else None
        a0 = spectral_flatten_init(L2, rng, n_iter=45, start_a=start, exponent=0.70)
        a, s = polish_candidate(a0, min(dl, hard_deadline - 5.0), rng, f"alt-len{L2}")
        if np.isfinite(s) and s < best_score:
            best_score = s
            best_array = a.copy()
            print(f"  >>> best: {best_score:.6f}, n={len(best_array)}")

    # Candidate 5: absolute IFFT initializer, often gives diverse shapes.
    L3 = L_main if L_main != 512 else 512
    dl = t0 + 0.87 * span
    if time.time() < dl - 12.0:
        a0 = abs_ifft_init(L3, rng)
        a, s = polish_candidate(a0, min(dl, hard_deadline - 5.0), rng, f"abs-ifft-{L3}")
        if np.isfinite(s) and s < best_score:
            best_score = s
            best_array = a.copy()
            print(f"  >>> best: {best_score:.6f}, n={len(best_array)}")

    # Candidate 6: try a longer length if time permits.
    L4 = 768 if L_main != 768 else 1024
    dl = hard_deadline - 45.0
    if time.time() < dl - 15.0:
        start = resize_sequence(hs, L4) if hs is not None else None
        a0 = spectral_flatten_init(L4, rng, n_iter=35, start_a=start, exponent=0.72)
        a, s = polish_candidate(a0, min(dl, hard_deadline - 5.0), rng, f"alt-len{L4}")
        if np.isfinite(s) and s < best_score:
            best_score = s
            best_array = a.copy()
            print(f"  >>> best: {best_score:.6f}, n={len(best_array)}")

    # If nothing valid was found, use a safe random/off initializer.
    if best_array is None:
        L = 512
        best_array = spectral_flatten_init(L, rng, n_iter=40, start_a=None)
        best_score = evaluate_array(best_array)
        if not np.isfinite(best_score):
            best_array = np.ones(L, dtype=np.float64) / float(L)
            best_score = evaluate_array(best_array)

    # Final polish on the globally best sequence.
    if time.time() < hard_deadline - 25.0:
        rem = hard_deadline - time.time()
        if rem > 60.0:
            a, s = polish_candidate(best_array, hard_deadline - 5.0, rng, "final")
            if np.isfinite(s) and s < best_score:
                best_score = s
                best_array = a.copy()
                print(f"  >>> final best: {best_score:.6f}, n={len(best_array)}")

    # Build final list safely.
    result = []
    for x in best_array:
        y = float(x)
        if not np.isfinite(y):
            y = 0.0
        if y < 0.0:
            y = 0.0
        if y > 1000.0:
            y = 1000.0
        result.append(y)

    if not result:
        result = [1.0] * 512

    # If final score is somehow invalid, fall back to uniform.
    final_score = evaluate_array(np.asarray(result, dtype=np.float64))
    if not np.isfinite(final_score):
        result = [1.0] * 512
        final_score = evaluate_array(np.asarray(result, dtype=np.float64))

    print(f"FINAL: n={len(result)}, score={final_score:.6f}, time={time.time()-t0:.1f}s")
    return result
```
