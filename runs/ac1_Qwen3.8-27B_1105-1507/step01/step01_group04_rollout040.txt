    # For real a, b: this gives the circular cross-correlation, which for zero-padded sequences
    # gives the linear cross-correlation.
    
    # Let's use: result = ifft(fft(w_norm, size) * conj(fft(f, size)))
    # The first n elements of result give us cross_correlation(w_norm, f)[0:n]
    
    W_full = np.fft.rfft(w_norm, size)
    F_full = np.fft.rfft(f, size)
    corr = np.fft.irfft(W_full * np.conj(F_full), size)
    grad_lse = 2.0 * corr[:n]
    
    # Now the full objective is 2n * lse / S^2
    # d/di (2n * lse / S^2) = (2n / S^2) * grad_lse[i] - (2n * lse / S^3) * S' [i]
    # where S' [i] = 1 (derivative of sum f w.r.t. f[i])
    # Actually: d(S^2)/df[i] = 2S
    # d(lse/S^2)/df[i] = (grad_lse[i] * S^2 - lse * 2S) / S^4 = grad_lse[i]/S^2 - 2*lse/(S^3)
    # d(2n * lse/S^2)/df[i] = 2n * (grad_lse[i]/S^2 - 2*lse/S^3)
    
    lse = max_shifted / alpha + np.log(w_sum) / alpha
    grad = 2.0 * n * (grad_lse / (S * S) - 2.0 * lse / (S * S * S))
    
    return grad

def _cosine_init(n, power=2):
    j = np.arange(n, dtype=np.float64)
    f = np.cos(np.pi * (j + 0.5) / float(n)) ** power
    f = np.maximum(f, 0.0)
    s = np.sum(f)
    if s > 0:
        f /= s
    return f

def _tapered_init(n, power=1.0):
    j = np.arange(n, dtype=np.float64)
    x = np.sin(np.pi * (j + 0.5) / float(n))
    f = x ** power
    f = np.maximum(f, 0.0)
    s = np.sum(f)
    if s > 0:
        f /= s
    return f

def _gaussian_init(n, sigma_frac=0.3):
    j = np.arange(n, dtype=np.float64) - (n - 1) / 2.0
    sigma = sigma_frac * float(n)
    f = np.exp(-j * j / (2.0 * sigma * sigma))
    s = np.sum(f)
    if s > 0:
        f /= s
    return f

using L-BFGS-B with log-sum-exp continuation method, multi-start strategies,
and adaptive sequence length exploration.

def _fft_convolve_full(f):
    n = len(f)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    F = np.fft.rfft(f, size)
    h = np.fft.irfft(F * F, size)
    return h[:2 * n - 1]

def _score_raw(f):
    n = len(f)
    h = _fft_convolve_full(f)
    M = float(np.max(h))
    S = float(np.sum(f))
    if S < 0.01:
        return float('inf')
    return 2.0 * n * M / (S * S)

def _lse_and_grad(f, alpha):
    """Compute LSE-approximated objective and its gradient."""
    n = len(f)
    S = float(np.sum(f))
    if S < 0.01:
        return 1e10, np.zeros(n)
    
    h = _fft_convolve_full(f)
    ah = alpha * h
    ah_max = np.max(ah)
    exp_ah = np.exp(ah - ah_max)
    w_sum = np.sum(exp_ah)
    w = exp_ah / w_sum  # softmax weights
    lse = (ah_max + np.log(w_sum)) / alpha
    
    # Gradient of LSE w.r.t. f[i] = 2 * sum_k w[k] * f[k-i] (for valid k-i)
    # This is 2 * cross_correlation(w, f)[i]
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    W = np.fft.rfft(w, size)
    F = np.fft.rfft(f, size)
    corr = np.fft.irfft(W * np.conj(F), size)
    grad_lse = 2.0 * corr[:n]
    
    # Full objective: J = 2n * lse / S^2
    obj = 2.0 * n * lse / (S * S)
    # dJ/df[i] = 2n * (grad_lse[i]/S^2 - 2*lse/S^3)
    grad = 2.0 * n * (grad_lse / (S * S) - 2.0 * lse / (S**3))
    
    return obj, grad

def _optimize_continuation(f_init, alphas, max_iter_per_alpha=200, time_budget=30):
    """L-BFGS-B with continuation over alpha values."""
    f = np.asarray(f_init, dtype=np.float64).copy()
    S = np.sum(f)
    if S < 0.01:
        return f, _score_raw(f)
    f = f / S  # normalize
    
    n = len(f)
    best_f = f.copy()
    best_score = _score_raw(f)
    start_time = time.monotonic()
    
    for alpha in alphas:
        if time.monotonic() - start_time > time_budget:
            break
        
        def obj(x):
            val, _ = _lse_and_grad(x, alpha)
            return val
        
        def grad(x):
            _, g = _lse_and_grad(x, alpha)
            return g
        
        try:
            res = minimize(obj, f, method='L-BFGS-B', jac=grad,
                          bounds=[(0, None)] * n,
                          options={'maxiter': max_iter_per_alpha, 'gtol': 1e-7})
            f_new = np.maximum(res.x, 0.0)
            S_new = np.sum(f_new)
            if S_new < 0.01:
                break
            f = f_new / S_new
        except Exception:
            break
        
        score = _score_raw(f)
        if score < best_score:
            best_score = score
            best_f = f.copy()
    
    return best_f, best_score

def _cosine_init(n, power=2):
    j = np.arange(n, dtype=np.float64)
    f = np.cos(np.pi * (j + 0.5) / float(n)) ** power
    f = np.maximum(f, 0.0)
    s = np.sum(f)
    if s > 0:
        f /= s
    return f

def _tapered_init(n, power=1.0):
    j = np.arange(n, dtype=np.float64)
    x = np.sin(np.pi * (j + 0.5) / float(n))
    f = x ** power
    f = np.maximum(f, 0.0)
    s = np.sum(f)
    if s > 0:
        f /= s
    return f

def _gaussian_init(n, sigma_frac=0.3):
    j = np.arange(n, dtype=np.float64) - (n - 1) / 2.0
    sigma = sigma_frac * float(n)
    f = np.exp(-j * j / (2.0 * sigma * sigma))
    s = np.sum(f)
    if s > 0:
        f /= s
    return f

def _bilateral_init(n, inner_power=1.0):
    """Initialize with mass at both ends, tapering towards center."""
    j = np.arange(n, dtype=np.float64)
    center = (n - 1) / 2.0
    dist = np.abs(j - center) / center  # 0 at center, 1 at edges
    f = dist ** inner_power
    f = np.maximum(f, 0.0)
    s = np.sum(f)
    if s > 0:
        f /= s
    return f

"""
Optimizes non-negative step function heights to minimize 2n*max(conv(f,f))/(sum(f))^2
using L-BFGS-B with log-sum-exp continuation, multi-start with diverse initializations,
adaptive alpha annealing, perturbation restarts, and sequence length exploration.
"""

import numpy as np
import time
from scipy.optimize import minimize

def _fft_convolve_full(f):
    """Compute full linear convolution of f with itself using FFT."""
    n = len(f)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    F = np.fft.rfft(f, size)
    h = np.fft.irfft(F * F, size)
    return h[:2 * n - 1]

def _score_raw(f):
    """Fast score computation assuming f >= 0, sum(f) > 0.01."""
    n = len(f)
    h = _fft_convolve_full(f)
    M = float(np.max(h))
    S = float(np.sum(f))
    if S < 0.01:
        return float('inf')
    return 2.0 * n * M / (S * S)

def _lse_obj_and_grad(f, alpha):
    """Compute LSE-approximated objective and gradient for normalized problem."""
    n = len(f)
    S = float(np.sum(f))
    if S < 0.01:
        return 1e10, np.zeros(n)

    h = _fft_convolve_full(f)
    ah = alpha * h
    ah_max = float(np.max(ah))
    exp_ah = np.exp(ah - ah_max)
    w_sum = float(np.sum(exp_ah))
    w = exp_ah / w_sum
    lse = (ah_max + np.log(w_sum)) / alpha

    # Gradient of LSE w.r.t. f: 2 * cross_correlation(w, f)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    W = np.fft.rfft(w, size)
    F = np.fft.rfft(f, size)
    corr = np.fft.irfft(W * np.conj(F), size)
    grad_lse = 2.0 * corr[:n]

    # J = 2n * lse / S^2
    obj = 2.0 * n * lse / (S * S)
    grad = 2.0 * n * (grad_lse / (S * S) - 2.0 * lse / (S * S * S))

    return obj, grad

def _optimize_lse(f_init, alphas, max_iter_per_alpha=200, time_budget=30.0):
    """L-BFGS-B with continuation over increasing alpha values."""
    f = np.asarray(f_init, dtype=np.float64).copy()
    S = float(np.sum(f))
    if S < 0.01:
        return f.copy(), _score_raw(f)
    f = f / S

    n = len(f)
    best_f = f.copy()
    best_score = _score_raw(f)
    start_time = time.monotonic()

    for alpha in alphas:
        if time.monotonic() - start_time > time_budget:
            break

        def obj_func(x):
            val, _ = _lse_obj_and_grad(x, alpha)
            return val

        def grad_func(x):
            _, g = _lse_obj_and_grad(x, alpha)
            return g

        try:
            res = minimize(obj_func, f, method='L-BFGS-B', jac=grad_func,
                          bounds=[(0.0, None)] * n,
                          options={'maxiter': max_iter_per_alpha, 'gtol': 1e-8})
            f_new = np.maximum(res.x, 0.0)
            S_new = float(np.sum(f_new))
            if S_new < 0.01:
                break
            f = f_new / S_new
        except Exception:
            break

        score = _score_raw(f)
        if score < best_score:
            best_score = score
            best_f = f.copy()

    return best_f, best_score

def _subgradient_step(f, n_steps=100, eta0=0.1, time_budget=10.0):
    """Projected subgradient method for the non-smooth objective."""
    f = np.asarray(f, dtype=np.float64).copy()
    S = float(np.sum(f))
    if S < 0.01:
        return f, _score_raw(f)
    f = f / S
    n = len(f)
    best_f = f.copy()
    best_score = _score_raw(f)
    start_time = time.monotonic()

    for it in range(n_steps):
        if time.monotonic() - start_time > time_budget:
            break

        h = _fft_convolve_full(f)
        M = float(np.max(h))
        active = np.where(h >= 0.8 * M)[0]
        if len(active) == 0:
            active = np.array([int(np.argmax(h))])

        # Average subgradient over active set
        grad = np.zeros(n)
        for k in active:
            for j in range(n):
                idx = k - j
                if 0 <= idx < n:
                    grad[j] += 2.0 * f[idx]
        grad /= float(len(active))

        eta = eta0 / (it + 1)
        f_new = f - eta * grad
        f_new = np.maximum(f_new, 0.0)
        S_new = float(np.sum(f_new))
        if S_new < 0.01:
            break
        f = f_new / S_new

        score = _score_raw(f)
        if score < best_score:
            best_score = score
            best_f = f.copy()

    return best_f, best_score

def _cosine_init(n, power=2):
    j = np.arange(n, dtype=np.float64)
    f = np.cos(np.pi * (j + 0.5) / float(n)) ** power
    f = np.maximum(f, 0.0)
    s = np.sum(f)
    if s > 0:
        f /= s
    return f

def _tapered_init(n, power=1.0):
    j = np.arange(n, dtype=np.float64)
    x = np.sin(np.pi * (j + 0.5) / float(n))
    f = x ** power
    f = np.maximum(f, 0.0)
    s = np.sum(f)
    if s > 0:
        f /= s
    return f

def _gaussian_init(n, sigma_frac=0.3):
    j = np.arange(n, dtype=np.float64) - (n - 1) / 2.0
    sigma = sigma_frac * float(n)
    f = np.exp(-j * j / (2.0 * sigma * sigma))
    s = np.sum(f)
    if s > 0:
        f /= s
    return f

def _bilateral_init(n, power=1.5):
    """Mass concentrated at both ends."""
    j = np.arange(n, dtype=np.float64)
    center = (n - 1) / 2.0
    dist = np.abs(j - center) / max(center, 1.0)
    f = dist ** power
    f = np.maximum(f, 0.0)
    s = np.sum(f)
    if s > 0:
        f /= s
    return f

def _ramp_init(n, side=1):
    """Linear ramp: increasing or decreasing."""
    j = np.arange(n, dtype=np.float64)
    if side == 1:
        f = j + 1.0
    else:
        f = n - j
    f = np.maximum(f, 0.0)
    s = np.sum(f)
    if s > 0:
        f /= s
    return f
