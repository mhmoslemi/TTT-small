"""
Optimizes non-negative sequences to minimize 2*n*max(convolve(a,a))/sum(a)^2.
Uses log-sum-exp smooth gradient with Nesterov momentum, adaptive tau annealing,
backtracking line search, and LP-based polish. Tries multiple sequence lengths
and initialization strategies (flat-spectrum alternating projection, random, etc.)
"""

import numpy as np
import time
from scipy.optimize import linprog


def project_simplex(v):
    """Project vector v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cv = np.cumsum(u) - 1.0
    rho_mask = u * np.arange(1, n + 1) > cv
    rho_candidates = np.where(rho_mask)[0]
    if len(rho_candidates) == 0:
        return np.ones(n) / n
    rho = rho_candidates[-1]
    theta = cv[rho] / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def fast_convolve(a):
    """Compute full linear convolution of a with itself using FFT."""
    n = len(a)
    size = 2 * n - 1
    N = 1 << (size - 1).bit_length()
    fa = np.fft.rfft(a, N)
    fc = fa * fa
    result = np.fft.irfft(fc, N)[:size]
    result[result < -1e-15] = 0.0
    return result


def compute_smooth_grad_fft(a, w):
    """
    Compute gradient of log-sum-exp(tau*c) using FFT cross-correlation.
    grad[i] = 2 * sum_d a[d] * w[i+d] (valid indices)
    where w = softmax(tau * c) and c = convolve(a, a).
    """
    n = len(a)
    w_len = len(w)
    N = 1 << (n + w_len - 1).bit_length()
    # Cross-correlation: (a ⊛ w)[i] = sum_d a[d] * w[d+i]
    # = IFFT(FFT(a) * conj(FFT(w)))
    fa = np.fft.rfft(a, N)
    fw = np.fft.rfft(w, N)
    xcorr = np.fft.irfft(fa * np.conj(fw), N)[:n]
    return 2.0 * xcorr


def compute_smooth_grad_direct(a, w, top_m=200):
    """
    Direct computation of smooth gradient using top-m weights only.
    More robust for verification.
    """
    n = len(a)
    grad = np.zeros(n)
    # Sort indices by weight, take top m
    if len(w) > top_m:
        top_idx = np.argsort(w)[::-1][:top_m]
    else:
        top_idx = np.arange(len(w))
    for k in top_idx:
        wk = w[k]
        if wk < 1e-15:
            continue
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        if lo <= hi:
            grad[lo:hi + 1] += wk * 2.0 * a[k - hi:k - lo + 1][::-1]
    return grad


def compute_hard_grad(a, c, tol=1e-6):
    """Compute exact subgradient of max(c), averaged over active peaks."""
    n = len(a)
    peak = np.max(c)
    peak_indices = np.where(c >= peak * (1.0 - tol))[0]
    if len(peak_indices) > 60:
        top_idx = np.argsort(c[peak_indices])[::-1][:30]
        peak_indices = peak_indices[top_idx]
    grad = np.zeros(n)
    for k in peak_indices:
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        if lo <= hi:
            grad[lo:hi + 1] += 2.0 * a[k - hi:k - lo + 1][::-1]
    grad /= max(1, len(peak_indices))
    return grad


def alt_projection_init(n, rng, n_iter=100):
    """
    Alternating projection to get a nonnegative sequence with near-flat
    magnitude spectrum.
    """
    N = 1 << (2 * n - 2).bit_length()
    phases = rng.uniform(0.0, 2.0 * np.pi, N // 2 + 1)
    F = np.exp(1j * phases)
    F[0] = 1.0 + 0j

    a = np.ones(n) / n
    for it in range(n_iter):
        a_full = np.fft.irfft(F, N)
        a = np.maximum(a_full[:n], 0.0)
        s = a.sum()
        if s < 1e-10:
            a = np.ones(n) / n
        else:
            a = a / s
        F_full = np.fft.rfft(a, N)
        # Flat target with slight low-frequency emphasis
        mags = np.abs(F_full)
        mean_mag = mags.mean()
        if mean_mag > 1e-10:
            F = (mean_mag / mags) * F_full
            F[0] = F[0].real + 0j
        else:
            break

    return a


def lp_polish(a, P, time_limit=10.0):
    """
    LP step: maximize sum(b) subject to convolve(a,b) <= P, b >= 0.
    Returns rescaled solution or None.
    """
    n = len(a)
    if n > 1000:
        return None

    A = np.zeros((2 * n - 1, n), dtype=np.float64)
    for k in range(2 * n - 1):
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        A[k, lo:hi + 1] = a[k - hi:k - lo + 1][::-1]

    c_obj = -np.ones(n)
    b_ub = np.full(2 * n - 1, P)

    try:
        res = linprog(c_obj, A_ub=A, b_ub=b_ub,
                      bounds=[(0.0, None)] * n,
                      method='highs',
                      options={'time_limit': time_limit})
        if res.success:
            b = res.x
            b_sum = b.sum()
            if b_sum > 1e-8:
                a_sum = a.sum()
                b = b * (a_sum / b_sum)
                b = project_simplex(b)
                return b
    except Exception:
        pass
    return None


def optimize_sequence(a, deadline, rng, use_lp=True, verbose=False):
    """
    Main optimizer: log-sum-exp smooth gradient + Nesterov momentum + backtracking.
    Phase 1: Smooth optimization with annealed tau
    Phase 2: Hard-max subgradient refinement
    Phase 3: Optional LP polish
    """
    n = len(a)
    a = project_simplex(a.copy().astype(np.float64))
    a_sum = a.sum()
    a = a / a_sum

    c = fast_convolve(a)
    best_peak = np.max(c)
    best_a = a.copy()

    # Nesterov momentum
    v = np.zeros(n)
    beta = 0.9  # momentum coefficient
    lr = 0.5 / np.sqrt(n)

    # Tau annealing: start smooth, end sharp
    tau_start = 2.0 / best_peak
    tau_end = 200.0 / best_peak
    total_iters = 30000

    stall_count = 0
    max_stall = 200

    while time.time() < deadline - 5:
        it = 0
        # Determine current tau based on progress
        c = fast_convolve(a)
        cur_peak = np.max(c)

        # Update best
        if cur_peak < best_peak * (1.0 - 1e-10):
            best_peak = cur_peak
            best_a = a.copy()
            stall_count = 0
            if verbose and stall_count == 0 and it % 1000 == 0:
                print(f"    peak improved to {cur_peak:.8f} at iter {it}")
        else:
            stall_count += 1

        # Adaptive tau: increase tau as we converge
        progress = min(1.0, stall_count / 5000.0)
        tau = tau_start * (tau_end / tau_start) ** progress
        # Cap tau to avoid numerical issues
        tau = min(tau, 500.0 / cur_peak)

        # Compute gradient
        scaled_c = tau * c
        # Numerically stable softmax
        scaled_c_shifted = scaled_c - np.max(scaled_c)
        w = np.exp(scaled_c_shifted)
        w_sum = w.sum()
        w = w / w_sum

        # Use FFT for gradient when n is large, direct for small
        if n <= 512:
            grad = compute_smooth_grad_fft(a, w)
        else:
            grad = compute_smooth_grad_fft(a, w)

        # Nesterov look-ahead
        a_lookahead = a + beta * v
        a_lookahead = project_simplex(a_lookahead)

        # Recompute gradient at lookahead point (Nesterov)
        c_la = fast_convolve(a_lookahead)
        tau_la = tau  # same tau
        scaled_la = tau_la * c_la
        scaled_la_shifted = scaled_la - np.max(scaled_la)
        w_la = np.exp(scaled_la_shifted)
        w_la = w_la / w_la.sum()
        grad_la = compute_smooth_grad_fft(a_lookahead, w_la)

        # Momentum update
        v = beta * v - lr * grad_la
        a_new = a + v
        a_new = project_simplex(a_new)

        # Backtracking line search
        c_new = fast_convolve(a_new)
        peak_new = np.max(c_new)
        c_old = fast_convolve(a)
        peak_old = np.max(c_old)

        steps = 0
        while peak_new > peak_old * (1.0 + 1e-8) and steps < 8 and time.time() < deadline - 5:
            lr *= 0.5
            v = beta * v - lr * grad_la
            a_new = a + v
            a_new = project_simplex(a_new)
            c_new = fast_convolve(a_new)
            peak_new = np.max(c_new)
            steps += 1

        a = a_new

        # Occasionally switch to hard gradient for precise peak reduction
        if stall_count > max_stall and stall_count % 200 == 0:
            hard_grad = compute_hard_grad(a, c, tol=1e-5)
            v = 0.8 * v - lr * hard_grad
            a = project_simplex(a + v)
            lr = max(lr * 0.5, 0.01 / n)

        # Anneal momentum slightly
        beta = max(0.5, beta * (1.0 - 1e-6))

        # Add small perturbation if very stuck
        if stall_count > 2000 and time.time() < deadline - 30:
            perturb = rng.normal(0, 0.01 / np.sqrt(n), n)
            a = project_simplex(a + perturb)
            v = np.zeros(n)
            stall_count = 0

        it += 1

        # LP polish every 2000 iterations if time permits
        if use_lp and n <= 512 and it % 2000 == 0 and time.time() < deadline - 30:
            b = lp_polish(a, cur_peak, time_limit=8.0)
            if b is not None:
                t_mix = 0.3 + 0.3 * rng.random()
                a_mixed = project_simplex((1.0 - t_mix) * a + t_mix * b)
                c_mixed = fast_convolve(a_mixed)
                if np.max(c_mixed) < np.max(c):
                    a = a_mixed
                    v = np.zeros(n)

        # Check time
        if time.time() >= deadline - 10:
            break

    # Final: ensure we return the best
    c_final = fast_convolve(best_a)
    final_peak = np.max(c_final)
    if final_peak < best_peak:
        best_peak = final_peak

    return best_a

    elif init_type == 'alt_proj_long':
        a = alt_projection_init(n, rng, n_iter=200)
    elif init_type == 'random':
        a = rng.random(n)
        a = a / a.sum()
    elif init_type == 'gaussian':
        a = np.abs(rng.normal(0, 1, n))
        a = a / a.sum()
    elif init_type == 'cosine_taper':
        x = np.linspace(0.0, 1.0, n)
        a = 0.5 * (1.0 + np.cos(2.0 * np.pi * x)) + 0.05
        a = a / a.sum()
    elif init_type == 'broadband':
        a = alt_projection_init(n, rng, n_iter=50)
        a = a + 0.1 * rng.random(n)
        a = np.maximum(a, 0.0)
        a = a / a.sum()
    elif init_type == 'parent':
        try:
            parent = np.array(height_sequence_1, dtype=np.float64)
            parent = np.clip(parent, 0.0, 1000.0)
            if len(parent) >= n:
                a = parent[:n].copy()
            else:
                a = np.zeros(n)
                a[:len(parent)] = parent
            a = np.maximum(a, 0.0)
            s = a.sum()
            a = a / s if s > 1e-10 else np.ones(n) / n
        except Exception:
            a = rng.random(n)
            a = a / a.sum()
    else:
        a = rng.random(n)
        a = a / a.sum()

    a_opt = optimize_sequence(a, deadline, rng, use_lp=use_lp)
    score = evaluate_sequence(a_opt.tolist())
    return a_opt, score


def propose_candidate(budget_s=1500):
    """
    Multi-phase search: tries multiple lengths and initializations using
    log-sum-exp smooth gradient with Nesterov momentum and LP polish.
    """
    overall_deadline = time.time() + budget_s - 30.0
    rng = np.random.default_rng(777)

    best_score = float('inf')
    best_seq = None

    print(f"Starting search with budget {budget_s}s, deadline at t={overall_deadline - time.time():.0f}s remaining")

    # Priority configs: (length, init_type)
    # Focus on lengths that previously worked well, plus exploration
    configs = [
        (512, 'alt_proj_long'),
        (512, 'parent'),
        (384, 'alt_proj_long'),
        (512, 'alt_proj'),
        (640, 'alt_proj_long'),
        (256, 'alt_proj_long'),
        (768, 'alt_proj'),
        (512, 'broadband'),
        (384, 'alt_proj'),
        (1024, 'alt_proj'),
        (448, 'alt_proj_long'),
        (512, 'random'),
    ]

    num_configs = len(configs)
    # Allocate time more aggressively to early configs
    # First 3 get 25% each of remaining, rest share 25%
    first_count = 3
    first_frac = 0.75 / first_count
    rest_frac = 0.25 / max(1, num_configs - first_count)

    for i, (n, init_type) in enumerate(configs):
        if time.time() > overall_deadline - 60:
            break

        if i < first_count:
            time_alloc = (overall_deadline - time.time()) * first_frac
        else:
            time_alloc = (overall_deadline - time.time()) * rest_frac

        config_deadline = min(overall_deadline - 20, time.time() + time_alloc)
        if config_deadline < time.time() + 20:
            continue

        use_lp = (n <= 512)
        print(f"\nConfig {i+1}/{num_configs}: n={n}, init={init_type}, time_alloc={time_alloc:.0f}s")
        t_start = time.time()

        try:
            a_opt, score = try_length(n, init_type, rng, config_deadline, use_lp=use_lp)
        except Exception as e:
            print(f"  Error: {e}")
            continue

        elapsed = time.time() - t_start
        print(f"  n={n}, init={init_type}, score={score:.6f}, time={elapsed:.0f}s")

        if np.isfinite(score) and score < best_score:
            best_score = score
            best_seq = a_opt.copy()
            print(f"  >>> NEW BEST: {best_score:.6f} (n={n}, init={init_type})")

    # Final polish on best sequence with all remaining time
    if best_seq is not None and time.time() < overall_deadline - 15:
        remaining = overall_deadline - time.time()
        print(f"\nFinal polish with {remaining:.0f}s remaining, n={len(best_seq)}")
        a_polished = optimize_sequence(best_seq, overall_deadline, rng,
                                       use_lp=len(best_seq) <= 512, verbose=True)
        score = evaluate_sequence(a_polished.tolist())
        print(f"After polish: score={score:.6f}")
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_seq = a_polished
            print(f"  >>> NEW BEST: {best_score:.6f}")

    # Fallback
    if best_seq is None:
        print("FALLBACK: using uniform sequence")
        best_seq = np.ones(256)

    result = [float(x) for x in best_seq]
    final_score = evaluate_sequence(result)
    print(f"\nFINAL: score={final_score:.6f}, length={len(result)}")
    return result
