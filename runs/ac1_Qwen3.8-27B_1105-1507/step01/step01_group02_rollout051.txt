"""
Optimizes non-negative sequence to minimize 2*n*max(conv(a,a))/sum(a)^2.
Combines accelerated subgradient descent, L-BFGS with log-sum-exp smoothing,
LP-based direction finding, and spectral initialization.
"""

import numpy as np
import time
from scipy.optimize import minimize, linprog

def project_simplex(v):
    """Project vector v onto {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cv = np.cumsum(u) - 1.0
    rho_candidates = np.where(u * np.arange(1, n + 1) > cv)[0]
    if len(rho_candidates) == 0:
        return np.ones(n) / n
    rho = rho_candidates[-1]
    theta = cv[rho] / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def fast_convolve(a):
    """Compute full linear convolution of a with itself using FFT."""
    n = len(a)
    size = 2 * n - 1
    N = 1 << (size - 1).bit_length()
    fa = np.fft.rfft(a, N)
    fc = fa * fa
    result = np.fft.irfft(fc, N)[:size]
    result[result < -1e-12] = 0.0
    return result

def compute_subgradient(a, peak_indices):
    """Compute subgradient of max_k c[k] at point a."""
    n = len(a)
    grad = np.zeros(n)
    m = len(peak_indices)
    for k in peak_indices:
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        if lo <= hi:
            grad[lo:hi+1] += a[k-hi:k-lo+1][::-1]
    grad *= 2.0 / m
    return grad

def score_from_conv(n, c):
    """Compute score from convolution and n."""
    return 2.0 * n * c.max()

    """
    LP step: find direction d that, when added to a, reduces the peak.
    Maximize sum(d) subject to:
      a + d >= 0
      sum(d) = 0 (maintain normalization)
      dc[k] <= 0 for k in active set (first-order: don't increase active peaks)

    threshold = current_peak * 0.999

    #    i.e., 2 * sum_{j: k-j in [0,n-1]} a[k-j] * d[j] <= 0
    # 2. sum(d) = 0

    # Objective: minimize -sum(d) (or just minimize 0, since we want any feasible direction)

    # Then it mixed: a_new = (1-t)*a + t*b
    # This is a different but valid approach.

    # Then mix a and b.
    
    if n > 2048:
        return None

    # This is a convolution matrix of a with b

            b_sum = b.sum()
            if b_sum > 1e-10:

def flat_spectrum_init(n, rng, n_iter=50):
    """Initialize using flat-spectrum design (iterative amplitude/phase)."""
    # We want a length-n sequence with flat aperiodic autocorrelation
    # Use iterative method:
    # 1. Start with random phases
    # 2. FFT, set magnitudes to target, IFFT
    # 3. Truncate to length n, set non-positive to 0, normalize
    # 4. Repeat
    
    # Use zero-padded FFT of length >= 2n-1
    N = 1 << (2*n - 2).bit_length()
    
    # Target: flat magnitude spectrum
    target_mag = np.ones(N // 2 + 1)
    # Slightly boost low frequencies to help non-negativity
    freq = np.linspace(0, 1, N // 2 + 1)
    target_mag = 1.0 + 0.3 * (1 - freq)
    
    phases = rng.uniform(0, 2*np.pi, N // 2 + 1)
    F = target_mag * np.exp(1j * phases)
    F[0] = abs(F[0])  # Make DC real and positive
    
    a = np.ones(n) / n
    for it in range(n_iter):
        a_full = np.fft.irfft(F, N)
        a = a_full[:n].copy()
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s < 1e-10:
            a = np.ones(n) / n
        else:
            a = a / s
        # Update phases
        F_full = np.fft.rfft(a, N)
        phases_new = np.angle(F_full)
        F = target_mag * np.exp(1j * phases_new)
        F[0] = abs(F[0])
    
    return a

        grad_norm_sq = np.sum(grad**2)
        if grad_norm_sq < 1e-20:
            break
        # Polyak step
        polyak_lr = (peak - lower_bound) / grad_norm_sq
        lr = min(lr * 1.005, polyak_lr, 5.0 / n)
        if lr < 1e-8:
            lr = 1e-8
        
        # Gradient step
        x_new = project_simplex(y - lr * grad)
        
        # Nesterov update
        t_new = (1 + np.sqrt(1 + 4 * t**2)) / 2
        beta = (t - 1) / t_new
        y = x_new + beta * (x_new - a)
        y = project_simplex(y)
        a = x_new
        t = t_new
        
        # LP step
        if use_lp and n <= 2048 and it % 300 == 0 and time.time() < deadline - 20:
            b = lp_direction(a, best_peak, deadline)
            if b is not None:
                t_mix = 0.3 + 0.3 * rng.random()
                a_new = project_simplex((1 - t_mix) * a + t_mix * b)
                c_new = fast_convolve(a_new)
                if c_new.max() < best_peak:
                    a = a_new
                    best_peak = c_new.max()
                    best_a = a.copy()
                # Reset momentum
                y = a.copy()
                t = 1.0
        
        # Perturbation if stalled
        if stall > max_stall:
            stall = 0
            if time.time() < deadline - 30:
                perturb = rng.normal(0, 0.03 / np.sqrt(n), n)
                a = project_simplex(a + perturb)
                y = a.copy()
                t = 1.0
                lr = 1.0 / n
    
    return best_a

def optimize_lse(a_init, deadline, n, beta_max=100):
    """
    Optimize using L-BFGS on log-sum-exp smoothed objective with softmax parameterization.
    """
    n = len(a_init)
    
    def get_objective_and_grad(x, beta):
        # Softmax
        x_shifted = x - x.max()
        exp_x = np.exp(x_shifted)
        p = exp_x / exp_x.sum()
        
        # Convolution
        c = fast_convolve(p)
        
        # Log-sum-exp
        beta_c = beta * c
        beta_c_shifted = beta_c - beta_c.max()
        lse = c.max() + np.log(np.sum(np.exp(beta_c_shifted))) / beta
        
        # Gradient of lse w.r.t. p
        w = np.exp(beta_c_shifted)
        w = w / w.sum()  # w[k] = exp(beta*c[k]) / sum(exp(beta*c[l]))
        
        grad_p = np.zeros(n)
        for k in range(len(c)):
            wk = w[k]
            if wk < 1e-10:
                continue
            lo = max(0, k - n + 1)
            hi = min(n - 1, k)
            if lo <= hi:
                grad_p[lo:hi+1] += 2.0 * wk * p[k-hi:k-lo+1][::-1]
        
        # Chain rule for softmax: d(f)/dx = p * (grad_p - dot(grad_p, p))
        correction = np.dot(grad_p, p)
        grad_x = p * (grad_p - correction)
        
        return 2.0 * n * lse, 2.0 * n * grad_x, p
    
    # Start with high beta (close to max) and decrease... or start low and increase
    # Actually, for L-BFGS, it's better to start with moderate beta
    
    best_p = a_init.copy()
    best_score = 2.0 * n * fast_convolve(a_init).max()
    
    # Anneal beta from low to high
    betas = [10, 20, 40, 60, 80, beta_max]
    
    for beta in betas:
        if time.time() > deadline - 10:
            break
        
        x0 = np.log(a_init + 1e-15)
        
        def obj(x):
            val, grad_x, p = get_objective_and_grad(x, beta)
            return val
        
        def grad_fn(x):
            val, grad_x, p = get_objective_and_grad(x, beta)
            return grad_x
        
        try:
            result = minimize(obj, x0, jac=grad_fn, method='L-BFGS-B',
                            options={'maxiter': 3000, 'maxls': 50,
                                   'ftol': 1e-12, 'gtol': 1e-8})
            x_opt = result.x
            p_opt = np.exp(x_opt - x_opt.max())
            p_opt = p_opt / p_opt.sum()
            p_opt = np.maximum(p_opt, 0)
            p_opt = p_opt / p_opt.sum()
            
            score = 2.0 * n * fast_convolve(p_opt).max()
            if score < best_score:
                best_score = score
                best_p = p_opt.copy()
                a_init = p_opt.copy()  # Continue from this point
        except Exception as e:
            print(f"  L-BFGS error at beta={beta}: {e}")
    
    return best_p
