"""
Minimize score = 2*n*max(convolve(a,a))/sum(a)^2 using Successive Linear Programming (SLP)
with active-set constraints, combined with multiplicative subgradient updates and
population-based restarts. Key innovation: efficient LP-based direction finding via
linearized convolution constraints at active peaks, with adaptive step control.

import time
import numpy as np
from scipy.optimize import linprog


def _normalize(a):
    """Normalize to sum=1, clip to [0, 1000]."""
    a = np.asarray(a, dtype=np.float64)
    a = np.clip(a, 0.0, 1000.0)
    s = float(np.sum(a))
    if not np.isfinite(s) or s <= 1e-300:
        return None
    a = a / s
    if not np.isfinite(a).all():
        return None
    return a


def _local_score(a):
    """Compute the objective score."""
    a = np.asarray(a, dtype=np.float64)
    n = a.size
    if n == 0:
        return float('inf')
    a = np.clip(a, 0.0, 1000.0)
    s = float(np.sum(a))
    if not np.isfinite(s) or s <= 1e-12:
        return float('inf')
    c = np.convolve(a, a)
    m = float(np.max(c))
    if not np.isfinite(m):
        return float('inf')
    return float(2.0 * n * m / (s * s))


def _convolve_with_refixed(a_fixed, a_var, k):
    """Compute (a_fixed * a_var)[k] efficiently."""
    n = a_fixed.size
    j_min = max(0, k - n + 1)
    j_max = min(k, n - 1)
    if j_min > j_max:
        return 0.0
    return float(np.dot(a_fixed[k - np.arange(j_min, j_max + 1)], a_var[j_min:j_max + 1]))


def _build_lp_matrix(a_cur, active, n):
    """Build the LP constraint matrix for SLP step. Returns A_ub, b_ub."""
    m = len(active)
    A = np.zeros((m, n + 1), dtype=np.float64)
    b = np.zeros(m, dtype=np.float64)
    c_cur = np.convolve(a_cur, a_cur)

    for idx, k in enumerate(active):
        j_min = max(0, k - n + 1)
        j_max = min(k, n - 1)
        if j_min <= j_max:
            A[idx, j_min:j_max + 1] = 2.0 * a_cur[k - np.arange(j_min, j_max + 1)]
        A[idx, n] = -1.0
        b[idx] = c_cur[k]

    return A, b, c_cur


def _slp_solve(a_cur, active, n):
    """Solve one SLP subproblem. Returns solution vector a or None."""
    M = float(np.max(np.convolve(a_cur, a_cur)))
    A, b, c_cur = _build_lp_matrix(a_cur, active, n)

    c_obj = np.zeros(n + 1, dtype=np.float64)
    c_obj[n] = 1.0

    bounds = [(0.0, None)] * n + [(-M * 1.1, None)]

    A_eq = np.zeros((1, n + 1), dtype=np.float64)
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])

    try:
        res = linprog(c_obj, A_ub=A, b_ub=b, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': 30})
    except Exception:
        return None

    if res is None or not res.success:
        return None

    a_sol = np.asarray(res.x[:n], dtype=np.float64)
    a_sol = np.clip(a_sol, 0.0, None)
    s = float(np.sum(a_sol))
    if s <= 1e-300:
        return None
    return a_sol / s


def _multiplicative_update(a, eta, K):
    """Multiplicative subgradient update based on active convolution peaks."""
    n = a.size
    c = np.convolve(a, a)
    M = float(np.max(c))
    if M <= 0:
        return a.copy()

    if K is None or K <= 0:
        K = max(16, n // 4)
    K = min(K, 2 * n - 1)

    active = np.argpartition(c, -K)[-K:]
    w = c[active] / M

    h = np.zeros(n, dtype=np.float64)
    for idx in range(active.size):
        k = int(active[idx])
        wk = float(w[idx])
        j_min = max(0, k - n + 1)
        j_max = min(k, n - 1)
        if j_min > j_max:
            continue
        j = np.arange(j_min, j_max + 1, dtype=np.int64)
        h[j] += wk * a[k - j]

    maxh = float(np.max(h))
    if maxh <= 0:
        return a.copy()

    meanh = float(np.mean(h))
    fac = eta * (meanh - h) / maxh
    fac = np.clip(fac, -8.0, 8.0)
    a2 = a * np.exp(fac)
    a2 = np.clip(a2, 0.0, 1000.0)
    s = float(np.sum(a2))
    if s <= 1e-300:
        return a.copy()
    return a2 / s

def _random_init(n, seed, kind=0):

    elif kind == 2:

        a = np.repeat(rng.uniform(0.1, 1.0, nb), n // nb + 1)[:n]
        if n > 4:

        a = np.clip(a, 0.01, None)

        a = 0.3 + 0.7 * (2 * x - 1)**2

    elif kind == 4:

        if a is not None:
            starts.append(a)

    # Quick screen: evaluate each start
    for i, a in enumerate(starts):
        sc = _local_score(a)
        if sc < best_sc:
            best = a.copy()
            best_sc = sc
        if time.monotonic() > deadline - 20:
            break

    if best is None:
        best = _random_init(n, seed, 0)
        best_sc = _local_score(best)

    print(f"  length={n} initial best score={best_sc:.6f}", flush=True)

    # Main SLP optimization
    slp_budget = min(budget * 0.7, deadline - time.monotonic() - 10)
    if slp_budget > 10:
        a_slp, sc_slp = _slp_optimize(best, slp_budget, seed + 777, verbose=True)
        if a_slp is not None and sc_slp < best_sc:
            best = a_slp.copy()
            best_sc = sc_slp
            print(f"  length={n} after SLP score={best_sc:.6f}", flush=True)

    # Local polish
    rem = deadline - time.monotonic()
    polish_budget = min(rem * 0.5, 60.0)
    if polish_budget > 5 and best is not None:
        a_pol, sc_pol = _local_polish(best, polish_budget, seed + 999)
        if a_pol is not None and sc_pol < best_sc:
            best = a_pol.copy()
            best_sc = sc_pol
            print(f"  length={n} after polish score={best_sc:.6f}", flush=True)

    # Second SLP pass with fresh active sets
    rem = deadline - time.monotonic()
    if rem > 30 and best is not None:
        a_slp2, sc_slp2 = _slp_optimize(best, rem * 0.6, seed + 55555)
        if a_slp2 is not None and sc_slp2 < best_sc:
            best = a_slp2.copy()
            best_sc = sc_slp2
            print(f"  length={n} after SLP2 score={best_sc:.6f}", flush=True)

    return best, best_sc


def propose_candidate(seed=42, budget_s=1500, **kwargs):
    """
    Search for non-negative step function coefficients minimizing:
        score = 2 * n * max(convolve(a,a)) / sum(a)^2

    Strategy:
      1. Start from parent sequence if available
      2. Run SLP (Successive Linear Programming) at multiple lengths
      3. Polish with multiplicative subgradient updates
      4. Return the best sequence found
    """
    start_time = time.monotonic()
    try:
        budget = float(budget_s)
    except Exception:
        budget = 1500.0
    if not np.isfinite(budget) or budget <= 0:
        budget = 1500.0

    deadline = start_time + max(10.0, budget) - 15.0  # 15s safety margin

    # Get parent
    parent = None
    try:
        p = globals().get('height_sequence_1', None)
        if p is not None:
            parent = np.asarray(p, dtype=np.float64)
            parent = np.clip(parent, 0.0, 1000.0)
            if parent.size == 0 or not np.isfinite(parent).all():
                parent = None
    except Exception:
        parent = None

    best_seq = None
    best_score = float('inf')

    if parent is not None:
        pn = _normalize(parent)
        if pn is not None:
            best_seq = pn.copy()
            best_score = _local_score(pn)
            print(f"Parent: length={pn.size} score={best_score:.6f}", flush=True)

    # Length plan: focus on 512 (parent length) and explore nearby
    # Time allocation weighted toward the parent length
    if parent is not None and 400 <= parent.size <= 700:
        plan = [
            (parent.size, 0.45, seed),       # Main effort on parent length
            (parent.size * 2, 0.25, seed + 1),  # Try larger
            (max(64, parent.size // 2), 0.10, seed + 2),  # Try smaller
            (parent.size * 4, 0.20, seed + 3),  # Try much larger
        ]
    elif parent is not None:
        plan = [
            (parent.size, 0.40, seed),
            (max(128, parent.size * 2), 0.25, seed + 1),
            (max(64, parent.size // 2), 0.15, seed + 2),
            (parent.size * 3, 0.20, seed + 3),
        ]
    else:
        plan = [
            (256, 0.20, seed),
            (512, 0.35, seed + 1),
            (1024, 0.30, seed + 2),
            (2048, 0.15, seed + 3),
        ]

    seed_cur = int(seed)
    for n, frac, s in plan:
        rem = deadline - time.monotonic()
        if rem < 30:
            break

        alloc = min(frac * budget, rem - 20)
        if alloc < 15:
            alloc = max(10, rem * 0.3)

        print(f"\n=== Trying length {n} with budget {alloc:.0f}s ===", flush=True)
        try:
            cand, sc = _try_length(n, alloc, s, parent)
            if cand is not None and np.isfinite(sc) and sc < best_score:
                best_seq = cand.copy()
                best_score = float(sc)
                print(f"  NEW BEST: length={n} score={best_score:.6f}", flush=True)
            else:
                print(f"  length={n} best_here={sc:.6f} (not better than {best_score:.6f})", flush=True)
        except Exception as exc:
            print(f"  length={n} failed: {exc}", flush=True)

        seed_cur += 100997

    # Final intensive SLP on the best sequence found
    if best_seq is not None:
        rem = deadline - time.monotonic()
        if rem > 30:
            print(f"\n=== Final SLP polish (budget {rem:.0f}s) ===", flush=True)
            try:
                a_final, sc_final = _slp_optimize(best_seq, min(rem * 0.6, 300), seed_cur + 1, verbose=True)
                if a_final is not None and sc_final < best_score:
                    best_seq = a_final.copy()
                    best_score = float(sc_final)
                    print(f"  Final SLP: score={best_score:.6f}", flush=True)
            except Exception as exc:
                print(f"  Final SLP failed: {exc}", flush=True)

        rem2 = deadline - time.monotonic()
        if rem2 > 15:
            try:
                a_pol, sc_pol = _local_polish(best_seq, min(rem2 * 0.7, 100), seed_cur + 2)
                if a_pol is not None and sc_pol < best_score:
                    best_seq = a_pol.copy()
                    best_score = float(sc_pol)
                    print(f"  Final polish: score={best_score:.6f}", flush=True)
            except Exception as exc:
                print(f"  Final polish failed: {exc}", flush=True)

    # Fallback
    if best_seq is None:
        best_seq = np.ones(128, dtype=np.float64) / 128.0

    # Final normalization and validation
    final = _normalize(best_seq)
    if final is None:
        final = np.ones(128, dtype=np.float64) / 128.0

    # Validate with official evaluator
    try:
        official = float(evaluate_sequence([float(x) for x in final]))
        if np.isfinite(official):
            best_score = official
        else:
            # Try parent as fallback
            if parent is not None:
                pf = _normalize(parent)
                if pf is not None:
                    official2 = float(evaluate_sequence([float(x) for x in pf]))
                    if np.isfinite(official2):
                        final = pf.copy()
                        best_score = official2
    except Exception:
        pass

    elapsed = time.monotonic() - start_time
    print(f"\nFINAL: score={best_score:.6f} length={len(final)} time={elapsed:.1f}s", flush=True)

    return [float(x) for x in np.clip(final, 0.0, 1000.0)]
