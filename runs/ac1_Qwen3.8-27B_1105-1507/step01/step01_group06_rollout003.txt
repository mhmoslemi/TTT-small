Minimizes J(a) = 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Fast analytic gradient descent with FFT convolution
2. Momentum and adaptive step sizes
3. Multiple sequence lengths (256 to 8192) and diverse initializations
4. LP-based best response refinement for medium n
5. Parent sequence interpolation
"""

import numpy as np
from scipy.optimize import linprog, minimize
import time
import math

def fast_conv(a):
    """Full self-convolution via FFT, O(n log n)."""
    n = len(a)
    s = 1
    while s < 2 * n - 1:
        s <<= 1
    fa = np.fft.rfft(a, s)
    return np.fft.irfft(fa * fa, s)[:2 * n - 1]

def objective_value(a):
    """Compute J(a) = 2*n*max(conv)/(sum)^2."""
    n = len(a)
    s = a.sum()
    if s < 1e-10:
        return float('inf')
    conv = fast_conv(a)
    return 2.0 * n * conv.max() / (s * s)

def gradient_step(a, lr, rng):
    """One gradient descent step. Returns (a_new, J, peak_val)."""
    n = len(a)
    conv = fast_conv(a)
    peak_val = float(conv.max())
    s = float(a.sum())
    
    # Random argmax for subgradient
    argmaxes = np.where(conv >= peak_val * (1 - 1e-10))[0]
    k_star = int(argmaxes[rng.integers(len(argmaxes))])
    
    # Gradient of max(conv) wrt a[i]
    # = 2*a[k_star - i] if 0 <= k_star - i < n, else 0
    grad = np.zeros(n)
    lo = max(0, k_star - n + 1)
    hi = min(n - 1, k_star)
    if lo <= hi:
        grad[lo:hi+1] = 2.0 * a[k_star - hi:k_star - lo + 1][::-1]
    
    # Full gradient of J
    full_grad = (2.0 * n / (s * s)) * (grad - 2.0 * peak_val / s)
    
    a_new = a - lr * full_grad
    a_new = np.maximum(a_new, 0.0)
    
    return a_new, peak_val

def gradient_descent(a0, n_steps, lr, rng, momentum=0.9, deadline=None):
    """Run gradient descent with momentum. Returns best a found."""
    a = a0.copy()
    s = a.sum()
    if s < 1e-10:
        return a0
    a = a / s  # normalize
    v = np.zeros_like(a)
    best_a = a.copy()
    best_J = objective_value(a)
    
    for i in range(n_steps):
        if deadline is not None and time.monotonic() > deadline:
            break
        
        # Backtracking line search
        lr_i = lr * (0.95 ** (i // 100))  # slow decay
        for backtrack in range(20):
            a_new, _ = gradient_step(a, lr_i, rng)
            s_new = a_new.sum()
            if s_new < 1e-10:
                break
            a_norm = a_new / s_new
            J_new = objective_value(a_norm)
            if J_new < best_J or J_new < objective_value(a) * 1.01:
                break
            lr_i *= 0.5
        
        if s_new < 1e-10:
            break
        
        # Momentum update
        v = momentum * v + (1 - momentum) * (a - a_norm)
        a_candidate = a - lr_i * v
        a_candidate = np.maximum(a_candidate, 0.0)
        s_cand = a_candidate.sum()
        if s_cand > 1e-10:
            a_candidate = a_candidate / s_cand
            J_cand = objective_value(a_candidate)
            if J_cand < best_J:
                best_J = J_cand
                best_a = a_candidate.copy()
            a = a_candidate
    
    return best_a, best_J

def fast_conv(a):
    """Full self-convolution via FFT, O(n log n)."""
    n = len(a)

            t_old = np.linspace(0, 1, len(p))
            interp = np.interp(t, t_old, p)
            interp = np.maximum(interp, 0.001)
            inits.append(interp / interp.sum())

    return inits


def propose_candidate(seed=42, budget_s=1500, **kwargs):
    """
    Search for sequence minimizing J(a) = 2n*max(conv(a,a))/(sum(a))^2.
    Uses subgradient descent + LP refinement + multiple scales.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + float(budget_s) - 15.0
    best_seq = None
    best_score = float('inf')

    # Try parent
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        if p.ndim == 1 and p.size > 0 and np.all(np.isfinite(p)):
            p = np.maximum(p, 0.0)
            sc = j_value(p)
            print(f"Parent: n={p.size}, J={sc:.6f}", flush=True)
            if np.isfinite(sc) and sc < best_score:
                best_score = sc
                best_seq = p.copy()
    except (NameError, AttributeError):
        p = None

    # Phase 1: Subgradient descent across lengths
    lengths = [256, 512, 1024, 2048, 4096, 8192]
    
    # Allocate time proportionally to length (longer n needs more iterations)
    # But also cap: don't spend too long on any single n
    time_per_n_base = 80.0  # seconds

    for n in lengths:
        if time.monotonic() > deadline:
            break
        # Time budget for this n: more for moderate n, less for very large
        if n <= 512:
            t_budget = time_per_n_base * 1.5
        elif n <= 2048:
            t_budget = time_per_n_base * 1.0
        else:
            t_budget = time_per_n_base * 0.7

        local_deadline = min(time.monotonic() + t_budget, deadline)
        print(f"\n=== Subgrad n={n}, budget={t_budget:.0f}s ===", flush=True)
        t0 = time.monotonic()

        inits = generate_initializations(n, rng, best_seq)

        # Prioritize: parent-based init first if available, then diverse
        if best_seq is not None:
            inits = [inits[-1]] + inits[:-1]

        max_iters = max(5000, int(4.0 * (local_deadline - time.monotonic())) * 1000 // n)
        max_iters = min(max_iters, 500000)

        for idx, a0 in enumerate(inits):
            if time.monotonic() > local_deadline:
                break
            a_best, J_best = grad_descent(a0, max_iters, local_deadline, rng)
            if np.isfinite(J_best) and J_best < best_score:
                best_score = J_best
                best_seq = a_best.copy()
                print(f"  n={n} init[{idx}]: J={J_best:.6f} (t={time.monotonic()-t0:.1f}s)", flush=True)

        print(f"  n={n} done: {time.monotonic()-t0:.1f}s, best={best_score:.6f}", flush=True)

    # Phase 2: LP refinement on best candidate's length
    if best_seq is not None and time.monotonic() < deadline - 60:
        n_best = len(best_seq)
        if n_best <= 2048:
            n_lp = min(n_best, 1024)
            print(f"\n=== LP refine n={n_lp} ===", flush=True)
            t0 = time.monotonic()
            
            # Interpolate best to n_lp if needed
            if n_best != n_lp:
                t_old = np.linspace(0, 1, n_best)
                t_new = np.linspace(0, 1, n_lp)
                a_lp = np.interp(t_new, t_old, best_seq)
            else:
                a_lp = best_seq.copy()
            a_lp = np.maximum(a_lp, 0.0)
            a_lp = a_lp / a_lp.sum()

            a_best_lp, J_best_lp = lp_refine(a_lp, max_iters=300, deadline=deadline)
            if J_best_lp < best_score:
                best_score = J_best_lp
                best_seq = a_best_lp.copy()
                print(f"  LP: J={J_best_lp:.6f}", flush=True)

            # Also try LP with a few different starting points
            inits_lp = generate_initializations(n_lp, rng, best_seq)
            for a0 in inits_lp[:4]:
                if time.monotonic() > deadline - 30:
                    break
                a_b, J_b = lp_refine(a0, max_iters=150, deadline=deadline)
                if J_b < best_score:
                    best_score = J_b
                    best_seq = a_b.copy()
                    print(f"  LP (alt init): J={J_b:.6f}", flush=True)

            print(f"  LP done: {time.monotonic()-t0:.1f}s", flush=True)

    # Phase 3: Final gradient polish on best sequence
    if best_seq is not None and time.monotonic() < deadline:
        n_polish = len(best_seq)
        print(f"\n=== Polish n={n_polish} ===", flush=True)
        t0 = time.monotonic()
        
        a_polish = best_seq.copy()
        a_polish = np.maximum(a_polish, 0.0)
        a_polish = a_polish / a_polish.sum()
        
        # Run gradient descent with smaller lr for fine-tuning
        a_final, J_final = grad_descent(a_polish, 200000, deadline, rng,
                                         lr_init=0.1 / n_polish, momentum=0.9)
        if J_final < best_score:
            best_score = J_final
            best_seq = a_final.copy()
            print(f"  Polish: J={J_final:.6f}", flush=True)
        
        # Try a few restarts from perturbed best
        for restart in range(5):
            if time.monotonic() > deadline:
                break
            a_pert = best_seq + rng.normal(0, 0.01 / n_polish, n_polish)
            a_pert = np.maximum(a_pert, 0.0)
            a_pert = a_pert / a_pert.sum()
            a_r, J_r = grad_descent(a_pert, 100000, deadline, rng,
                                     lr_init=0.3 / n_polish, momentum=0.85)
            if J_r < best_score:
                best_score = J_r
                best_seq = a_r.copy()
                print(f"  Restart {restart}: J={J_r:.6f}", flush=True)

        print(f"  Polish done: {time.monotonic()-t0:.1f}s", flush=True)

    # Phase 4: Try one more larger n if time permits
    if time.monotonic() < deadline - 120:
        n_large = 16384
        print(f"\n=== Large n={n_large} ===", flush=True)
        t0 = time.monotonic()
        
        inits_large = generate_initializations(n_large, rng, best_seq)
        # Only try a few good inits
        for a0 in inits_large[:3]:
            if time.monotonic() > deadline - 30:
                break
            a_l, J_l = grad_descent(a0, 500000, deadline, rng, lr_init=0.3 / n_large)
            if J_l < best_score:
                best_score = J_l
                best_seq = a_l.copy()
                print(f"  n={n_large}: J={J_l:.6f}", flush=True)
        print(f"  Large done: {time.monotonic()-t0:.1f}s", flush=True)

    # Final evaluation
    if best_seq is None:
        best_seq = np.ones(256) / 256.0
        best_score = 2.0

    final_score = j_value(best_seq)
    if final_score > best_score:
        final_score = best_score

    print(f"\nFINAL: J={final_score:.6f}, n={len(best_seq)}", flush=True)
    return [float(x) for x in best_seq]
