def fast_conv(a):
    n = len(a)
    size = 1
    while size < 2 * n:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):

    a = np.clip(a, 0.0, 1000.0)
    s = a.sum()
    if s < 0.01:
        return np.inf
    b = fast_conv(a)
    return 2.0 * len(a) * b.max() / (s * s)

def fast_conv(a):
    """Full self-convolution via FFT, O(n log n)."""
    n = len(a)
    size = 1
    while size < 2 * n:

    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]

    """Interpolate sequence a to length new_n using linear interpolation."""

    t_new = np.linspace(0, 1, new_n)
