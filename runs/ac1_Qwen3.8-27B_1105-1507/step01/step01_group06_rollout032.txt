import numpy as np
from scipy.optimize import linprog, minimize

def fast_conv(a):
    """Full self-convolution via FFT, O(n log n)."""
    n = len(a)
    s = 1
    while s < 2 * n - 1:
        s <<= 1
    fa = np.fft.rfft(a, s)
    return np.fft.irfft(fa * fa, s)[:2 * n - 1]


def build_A(a):
    """Build Toeplitz-like constraint matrix A where A[k,j] = a[k-j]."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for k in range(2 * n - 1):
        jl = max(0, k - n + 1)
        jh = min(n - 1, k)
        if jl <= jh:
            A[k, jl:jh + 1] = a[k - jh:k - jl + 1][::-1]
    return A


def lp_best_response(a, M):
    """LP: max sum(g) s.t. (a*g)_k <= M, g >= 0. Returns g or None."""
    n = len(a)
    c = -np.ones(n)
    A = build_A(a)
    r = linprog(c, A_ub=A, b_ub=np.full(2 * n - 1, M),
               bounds=[(0, None)] * n, method='highs')
    if r.success:
        g = r.x
        if g.sum() > 1e-10:
            return g
    return None


def smooth_obj(x, beta, n):
    """Smooth max objective: 2n * log-sum-exp(beta * conv(a,a)) where a = x/sum(x)."""
    ax = np.maximum(x, 0.0)
    s = ax.sum()
    if s < 1e-8:

    ax_n = ax / s
    b = fast_conv(ax_n)
    bm = b.max()
    logsumexp_val = bm + np.log(np.sum(np.exp(beta * (b - bm)))) / beta
    return float(2 * n * logsumexp_val)


def smooth_grad(x, beta, n):
    """Gradient of smooth max objective w.r.t. x (unnormalized)."""
    ax = np.maximum(x, 0.0)

    c = c[:n]
    dot_cp = np.dot(c, ax)
    return (4.0 * n / s) * c - (4.0 * n / s ** 2) * ax * dot_cp

    a_new = np.maximum(np.interp(t_new, t_old, a_from), 0.01)
    return a_new / a_new.sum()

                best_score = score
                print(f"  LP n={n}: J={score:.6f}", flush=True)

        print(f"  Phase1 LP n={n}: {time.monotonic()-t0:.1f}s, best={best_score:.6f}", flush=True)

    # === Phase 2: Smooth gradient descent (multiple betas) ===
    for n in [512, 1024, 256, 768]:
        if time.monotonic() > deadline:
            break
        print(f"\n--- SmoothGrad: n={n} ---", flush=True)
        t0 = time.monotonic()

        starts = []
        if best is not None:
            starts.append(resample_sequence(best, n))
        if p is not None and p.size != n:
            starts.append(resample_sequence(p, n))
        starts.append(rng.uniform(0.2, 1.0, n))
        starts.append(make_cosine_init(n, [2, 4, 6], [0.3, 0.15, 0.08]))
        starts.append(make_cosine_init(n, [1, 3], [0.25, 0.2]))
        starts.append(make_cosine_init(n, [4], [0.35], base=1.0))

        for beta in [100.0, 200.0, 50.0, 300.0]:
            if time.monotonic() > deadline:
                break
            for a0 in starts:
                if time.monotonic() > deadline:
                    break
                a0 = np.clip(a0, 0.01, None)
                a0 = a0 / a0.sum()
                res = minimize(smooth_obj, a0, args=(beta, n), jac=smooth_grad,
                              method='L-BFGS-B',
                              bounds=[(0, None)] * n,
                              options={'maxiter': 300, 'max_fun': 5000,
                                       'ftol': 1e-15, 'gtol': 1e-12, 'disp': False})
                if res.fun < 1e10:
                    score, valid = try_update_score(res.x)
                    if valid and score < best_score:
                        best = res.x.copy()
                        best_score = score
                        print(f"  SmoothGrad n={n} beta={beta:.0f}: J={score:.6f}", flush=True)

        print(f"  Phase2 SmoothGrad n={n}: {time.monotonic()-t0:.1f}s, best={best_score:.6f}", flush=True)

    # === Phase 3: Hard max L-BFGS-B polish on best ===
    if best is not None and time.monotonic() < deadline:
        n = best.size
        print(f"\n--- HardMax polish: n={n} ---", flush=True)
        t0 = time.monotonic()
        a0 = np.clip(best.copy(), 0.01, None)
        a0 = a0 / a0.sum()
        res = minimize(hard_obj, a0, args=(n,), method='L-BFGS-B',
                      bounds=[(0, None)] * n,
                      options={'maxiter': 1000, 'max_fun': 50000,
                               'ftol': 1e-15, 'disp': False})
        if res.fun < 1e10:
            score, valid = try_update_score(res.x)
            if valid and score < best_score:
                best = res.x.copy()
                best_score = score
                print(f"  HardMax: J={score:.6f}", flush=True)
        print(f"  Phase3 HardMax: {time.monotonic()-t0:.1f}s, best={best_score:.6f}", flush=True)

    # === Phase 4: LP polish on best ===
    if best is not None and time.monotonic() < deadline:
        n = best.size
        if n <= 1024:
            print(f"\n--- LP polish: n={n} ---", flush=True)
            t0 = time.monotonic()
            a = np.clip(best.copy(), 0.01, None)
            a = a / a.sum()
            a = lp_polish(a, 200, deadline, n, t_scale=0.05)
            score, valid = try_update_score(a)
            if valid and score < best_score:
                best = a.copy()
                best_score = score
                print(f"  LP polish: J={score:.6f}", flush=True)
            print(f"  Phase4 LP polish: {time.monotonic()-t0:.1f}s, best={best_score:.6f}", flush=True)

    # === Phase 5: Try nearby lengths ===
    if best is not None and time.monotonic() < deadline:
        n_best = best.size
        candidates = []
        for mult in [2, 0.5, 1.5]:
            n_try = int(n_best * mult)
            if 64 <= n_try <= 4096:
                candidates.append(n_try)
        for n_try in candidates:
            if time.monotonic() > deadline:
                break
            print(f"\n--- Resample: n={n_try} ---", flush=True)
            a_res = resample_sequence(best, n_try)
            if n_try <= 1024:
                a_res = lp_polish(a_res, 80, deadline, n_try, t_scale=0.2)
            # Quick smooth gradient
            if time.monotonic() < deadline:
                res = minimize(smooth_obj, a_res, args=(150.0, n_try), jac=smooth_grad,
                              method='L-BFGS-B',
                              bounds=[(0, None)] * n_try,
                              options={'maxiter': 200, 'max_fun': 3000,
                                       'ftol': 1e-15, 'disp': False})
                if res.fun < 1e10:
                    a_res = np.clip(res.x, 0.01, None)
                    a_res = a_res / a_res.sum()
            score, valid = try_update_score(a_res)
            if valid and score < best_score:
                best = a_res.copy()
                best_score = score
                print(f"  Resample n={n_try}: J={score:.6f}", flush=True)

    # === Phase 6: Final combined polish ===
    if best is not None and time.monotonic() < deadline:
        n = best.size
        print(f"\n--- Final polish: n={n} ---", flush=True)
        t0 = time.monotonic()
        a = np.clip(best.copy(), 0.01, None)
        a = a / a.sum()

        # High-temperature smooth gradient
        for beta in [500.0, 200.0]:
            if time.monotonic() > deadline:
                break
            res = minimize(smooth_obj, a, args=(beta, n), jac=smooth_grad,
                          method='L-BFGS-B',
                          bounds=[(0, None)] * n,
                          options={'maxiter': 200, 'max_fun': 3000,
                                   'ftol': 1e-16, 'gtol': 1e-14, 'disp': False})
            if res.fun < 1e10:
                a = np.clip(res.x, 0.01, None)
                a = a / a.sum()

        # Final LP polish
        if time.monotonic() < deadline and n <= 1024:
            a = lp_polish(a, 100, deadline, n, t_scale=0.02)

        score, valid = try_update_score(a)
        if valid and score < best_score:
            best = a.copy()
            best_score = score
            print(f"  Final: J={score:.6f}", flush=True)
        print(f"  Phase6 Final: {time.monotonic()-t0:.1f}s, best={best_score:.6f}", flush=True)

    # === Phase 7: One more round of diverse exploration if time remains ===
    if time.monotonic() < deadline:
        print(f"\n--- Extra exploration ---", flush=True)
        t0 = time.monotonic()
        for n in [512, 384, 640]:
            if time.monotonic() > deadline:
                break
            for trial in range(3):
                if time.monotonic() > deadline:
                    break
                a0 = make_cosine_init(n, [2 + trial, 5 + trial * 2],
                                      [0.2 + 0.1 * trial, 0.1], base=1.0)
                a0 = a0 / a0.sum()
                res = minimize(smooth_obj, a0, args=(150.0, n), jac=smooth_grad,
                              method='L-BFGS-B',
                              bounds=[(0, None)] * n,
                              options={'maxiter': 300, 'max_fun': 4000,
                                       'ftol': 1e-15, 'disp': False})
                if res.fun < 1e10:
                    score, valid = try_update_score(res.x)
                    if valid and score < best_score:
                        best = res.x.copy()
                        best_score = score
                        print(f"  Extra n={n} trial={trial}: J={score:.6f}", flush=True)
        print(f"  Phase7: {time.monotonic()-t0:.1f}s, best={best_score:.6f}", flush=True)

    if best is None:
        best = np.ones(256)

    print(f"\nFINAL: J={best_score:.6f}, n={best.size}", flush=True)
    return [float(x) for x in best]
