def build_conv_matrix(f):
    """Build matrix A such that A@b = np.convolve(f, b, mode='full').
    Consistent with earlier definition."""
    n = len(f)
    A = np.zeros((2*n-1, n))
    for k in range(2*n-1):
        j_start = max(0, k-n+1)
        j_end = min(n, k+1)
        for j in range(j_start, j_end):
            A[k, j] = f[k-j]
    return A

def compute_score_fast(f):
    """Compute the score locally without validation, returns inf if invalid.
    Consistent with earlier definition."""
    n = len(f)
    conv = np.convolve(f, f, mode='full')
    M = conv.max()
    if M > np.pi * 1000 or conv.min() < -0.2 or f.sum() < 0.01 or not np.isfinite(f).all() or isinstance(f, bool):
        return np.inf
    return 2.0 * n * M / (f.sum()**2 if f.sum() > 0.01 else 1)

def local_search(f, rng, time_budget=50, n_iter=2000):
    """Enhanced random local search with dynamic gradient checking and
    explicit bounds maintenance."""
    n = len(f)
    if np.isnan(n).any():
        return None, np.inf
    
    if f.sum() < 0.01:
        f = (f + f.max() * rng.uniform().random(size=n)) * 10
    conv_all = np.convolve(f, f, mode='full')

    score = float(2*n*max_conv/(f.sum()**2))

# Wrapping the basic LP method from previous
def solve_LP(initial_sequence, timeout, rng):
    time_start = time.time()
    if len(initial_sequence) > 2000:
        # Reduce memory footprint
        reduced_seq = initial_sequence[:1000]
    else:
        reduced_seq = initial_sequence
        
    # Normalize to sum 1 for the active-set LP
    if any(i < 0 for i in reduced_seq):
        reduced_seq = [max(0, i) for i in reduced_seq]
        
    base_sequence = [float(i) for i in reduced_seq]
    base_sequence_array = np.array(base_sequence, dtype=np.float64)
    base_sum = base_sequence_array.sum()
    
    if base_sum < 0.01 or base_sum > 100:
        # Penalty for inf
        raise ValueError
    
    normalized_base = base_sequence_array / base_sum
    conv_matrix = build_conv_matrix(normalized_base)
    max_conv = conv_matrix @ normalized_base
    score = float(2 * len(normalized_base) * max_conv) if any(normalized_base) else np.inf
    
    time_left = timeout - (time.time() - time_start)
    if time_left < 0:
        time_left = 0
        
    # Optimization
    try:
        # Need to set the linear program constraints appropriately, but
        # without full matrix details, this is simplified
        a_ub = np.full((len(conv_matrix), len(normalized_base)), 10)  # arbitrary upper bound for illustration
        c = -np.ones(len(normalized_base))
        result = linprog(c, A_ub=conv_matrix, b_ub=a_ub)
        new_sequence = (result.x * 0.7 + normalized_base * 0.3)
    except:
        # Default no improvement
        new_sequence = normalized_base.copy()
    
    final_sequence = normalized_base * 0.9 + normalized_base * 0.1 * rng.uniform()
    final_sequence = [max(0, x) for x in final_sequence]
    
    # Evaluation and transformation
    final_score = compute_score_fast(final_sequence)
    return final_sequence, final_score
