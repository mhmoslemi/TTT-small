Minimizes J(a) = 2*n*max(convolve(a,a))/(sum(a))^2 using:
1. LP-based best response (mirror descent) with symmetry optimization (halved constraints)
2. Multi-scale warm-starting across sequence lengths
3. L-BFGS-B gradient polishing with FFT-accelerated convolution
4. Adaptive time budget allocation and random restarts
"""

import numpy as np
from scipy.optimize import linprog, minimize
from scipy.fft import rfft, irfft
import time


def fast_conv(a):
    """Full self-convolution via FFT, O(n log n)."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def build_A_half(a):
    """Build constraint matrix A (n x n) where A[k,j] = a[k-j] if 0<=k-j<n else 0.
    Uses symmetry c[k]=c[2n-2-k] to halve constraints from 2n-1 to n."""
    n = len(a)
    A = np.zeros((n, n))
    for k in range(n):
        A[k, :k + 1] = a[k::-1]
    return A


def lp_solve(a, M):
    """Solve LP: max sum(g) s.t. (a*g)[k] <= M for k=0..n-1, g >= 0.
    Returns g or None."""
    n = len(a)
    A = build_A_half(a)
    c = -np.ones(n)
    b_ub = np.full(n, M)
    bounds = [(0, None)] * n
    try:
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'presolve': True})
        if res.success and res.x is not None:
            g = res.x
            if g.sum() > 1e-10:
                return g
    except Exception:
        pass
    return None

    """Objective for L-BFGS-B: 2n * max_conv / sum^2 (with normalization)."""
    ax = np.maximum(x, 0.0)
    s = ax.sum()

    ax = ax / s
    b = fast_conv(ax)

def run_lp_iterations(a, n, phase_deadline, max_iters=500):
    """Run LP-based best response iterations on sequence a."""
    for it in range(max_iters):
        if time.monotonic() > phase_deadline:
            break
        b = fast_conv(a)
        M = float(b.max())
        g = lp_solve(a, M)
        if g is None:

        g = g / g.sum()

        a_new = np.clip(a_new, 0.0, None)
        s = a_new.sum()
        if s < 1e-10:
            break
        a_new = a_new / s
        a = a_new
    return a

    """Interpolate/resize a sequence to target length."""
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, n_target)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)
