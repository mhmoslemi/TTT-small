    n = len(a)
    if n == 0:

    max_b = float(np.max(conv))
    sum_a = float(np.sum(a))
    if sum_a < 0.01:
        return np.inf
    return 2.0 * n * max_b / (sum_a * sum_a)

def build_conv_matrix(a):
    """Build the (2n-1) x n matrix A where A[k,j] = a[k-j] if valid."""
    n = len(a)
    A = np.zeros((2*n-1, n), dtype=np.float64)
    for k in range(2*n-1):
        j_lo = max(0, k - n + 1)

            # When j = j_hi: a[k - j_hi]
            # So the values are a[k-j_hi], a[k-j_hi+1], ..., a[k-j_lo]
            A[k, j_lo:j_hi+1] = a[k-j_hi : k-j_lo+1]
    return A

    return None

One more idea: **what if we formulate this as a semidefinite program (SDP)?**

The constraint `a^T T_k a <= M` is a quadratic constraint. The set of `(a, M)` satisfying all constraints is non-convex. However, we can use the S-lemma or other techniques to get a convex relaxation.

Actually, for a single quadratic constraint `a^T Q a <= t` with `Q` symmetric, the set `{(a,t) : a^T Q a <= t}` is convex if `Q >= 0` (positive semidefinite). Let's check if `T_k` is PSD.

`T_k` is the matrix such that `a^T T_k a = conv(a,a)_k = sum_j a_j * a_{k-j}`.

For k=0: `T_0 = I` (PSD ✓)
For k=1: `T_1` has 1s on the first super- and sub-diagonal. This is the adjacency matrix of a path graph, which is PSD? Let's check: eigenvalues of the path adjacency matrix are `2*cos(j*pi/(n+1))` for j=1,...,n. These can be negative (for j > (n+1)/2). So `T_1` is NOT PSD in general.

So the quadratic constraints are not convex, and we can't directly use SDP relaxation.

However, we can use the following trick: `a^T T_k a = (T_k * a)^T a`. If we introduce a variable `b = T_k * a`, then the constraint becomes `b^T a <= M` with `b = T_k * a`. The feasible set in `(a, b)` is non-convex (bilinear constraint `b = T_k * a` is linear, but the constraint `b^T a <= M` is bilinear).

This doesn't help much. Let me stick with the LP-based approach.

Let me also consider **using CVXPY** with a non-convex solver (like SCS with non-convex mode, or ECOS). But the problem is non-convex, so these might not work well.

Actually, there's another approach: **alternating direction method of multipliers (ADMM)**.

Minimize `max_k conv(a,a)_k` s.t. `sum(a)=1, a>=0`.

Introduce variable `t`: minimize `t` s.t. `conv(a,a)_k <= t` for all k, `sum(a)=1, a>=0`.

ADMM splitting:
- x-update: minimize `t + (rho/2)||conv(a,a) - u + z||^2` s.t. `sum(a)=1, a>=0`
- z-update: `z = max(t, 0)`... this doesn't quite work.

This is getting too complex. Let me just implement the LP-based approach with improvements.

"""
Optimization of non-negative sequence to minimize 2*n*max(conv(a,a))/(sum(a))^2.
Uses alternating LP with line search, subgradient refinement, and multi-start strategy.
"""

import numpy as np
from scipy.optimize import linprog
import time

def compute_obj(a):
    n = len(a)
    if n == 0:
        return np.inf
    conv = np.convolve(a, a)
    max_b = float(np.max(conv))
    sum_a = float(np.sum(a))
    if sum_a < 0.01:
        return np.inf
    return 2.0 * n * max_b / (sum_a * sum_a)

def build_conv_matrix(a):
    n = len(a)
    A = np.zeros((2 * n - 1, n), dtype=np.float64)
    for k in range(2 * n - 1):
        j_lo = max(0, k - n + 1)
        j_hi = min(n - 1, k)
        A[k, j_lo:j_hi + 1] = a[k - j_hi:k - j_lo + 1][::-1]
    return A

def solve_lp(a, M):
    n = len(a)
    A = build_conv_matrix(a)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, 1000.0)] * n
    res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
    if res.status == 0:
        return res.x
    return None

def normalize(a):
    a = np.maximum(a, 0)
    s = np.sum(a)
    if s < 1e-15:
        return None
    return a / s

def lp_iteration(a, rng):
    """One LP iteration: solve LP, find best blend, return new a and improvement."""
    n = len(a)
    conv = np.convolve(a, a)
    M = float(np.max(conv))
    g = solve_lp(a, M)
    if g is None:
        return a, False
    g_sum = float(np.sum(g))
    if g_sum <= 1.0 + 1e-8:
        return a, False
    g = g / g_sum  # Normalize to sum=1
    
    # Line search: try multiple blend factors
    best_t = 0.0
    best_obj = compute_obj(a)
    best_a = a
    for t in [0.3, 0.5, 0.7, 0.9, 1.0]:
        a_new = normalize((1 - t) * a + t * g)
        if a_new is None:
            continue
        obj = compute_obj(a_new)
        if obj < best_obj:
            best_obj = obj
            best_a = a_new
            best_t = t
    return best_a, (best_t > 0)

def subgradient_iteration(a, step_size):
    """One subgradient step."""
    n = len(a)
    conv = np.convolve(a, a)
    max_val = np.max(conv)
    # Active constraints (top few)
    num_active = min(5, len(conv))
    active_idx = np.argsort(conv)[-num_active:]
    
    grad = np.zeros(n)
    for k in active_idx:
        j_lo = max(0, k - n + 1)
        j_hi = min(n - 1, k)
        grad[j_lo:j_hi + 1] += 2.0 * a[k - j_hi:k - j_lo + 1][::-1]
    grad /= num_active
    
    a_new = a - step_size * grad
    a_new = normalize(a_new)
    if a_new is None:
        return a
    return a_new
