Uses flat-spectrum initialization, mirror descent with adaptive steps, and peak-reduction heuristics.

import numpy as np
import time

def _pad_size(n):
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    return size

def _fft_convolve_full(f):
    n = len(f)
    size = _pad_size(n)
    F = np.fft.rfft(f, size)
    h = np.fft.irfft(F * F, size)
    return h[:2 * n - 1]

def _score_raw(f):
    n = len(f)
    h = _fft_convolve_full(f)
    M = float(np.max(h))
    S = float(np.sum(f))
    if S < 0.01:
        return float('inf')

    f = np.maximum(f, 0.0)

def _flat_spectrum_init(f, n_iters=2000, damping=0.5):
    """Flat-spectrum iteration to find a sequence with flat autocorrelation."""
    f = np.asarray(f, dtype=np.float64).copy()
    f = _normalize(f)
    if f is None:
        return np.ones(len(f)) / len(f), float('inf')
    n = len(f)
    size = _pad_size(n)
    best_f = f.copy()
    best_score = _score_raw(f)
    
    for it in range(n_iters):
        F = np.fft.rfft(f, size)
        
        # Target: flat magnitude
        # F[0] is the DC (sum), keep it
        # For k > 0, target |F[k]| = |F[0]| / sqrt(size)
        target_mag = abs(F[0]) / np.sqrt(size)

        new_mags = (1.0 - damping) * mags + damping * target_mag

"""
Optimizes non-negative step function to minimize 2n*max(conv(f,f))/(sum(f))^2.
Combines flat-spectrum initialization, momentum-based mirror descent with
backtracking, and targeted peak reduction.
"""

import numpy as np
import time

def _pad_size(n):
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    return size

def _fft_convolve_full(f):
    n = len(f)
    size = _pad_size(n)
    F = np.fft.rfft(f, size)
    h = np.fft.irfft(F * F, size)
    return h[:2 * n - 1]

def _score_raw(f):
    n = len(f)
    h = _fft_convolve_full(f)
    M = float(np.max(h))
    S = float(np.sum(f))
    if S < 0.01:
        return float('inf')
    return 2.0 * n * M / (S * S)

def _normalize(f):
    f = np.maximum(f, 0.0)
    s = float(np.sum(f))
    if s < 1e-30:
        return None
    return f / s

def _flat_spectrum_init(f0, n_iters=1000, damping=0.3):
    """
    Iterative flat-spectrum method: alternate between flattening the
    frequency-domain magnitude and projecting to non-negativity in time domain.
    """
    f = np.asarray(f0, dtype=np.float64).copy()
    f = _normalize(f)
    if f is None:
        return f0.copy(), float('inf')
    n = len(f)
    size = _pad_size(n)
    best_f = f.copy()
    best_score = _score_raw(f)
    
    for it in range(n_iters):
        F = np.fft.rfft(f, size)
        
        # Target magnitude: geometric mean of current and flat
        target_mag = np.abs(F[0]) / np.sqrt(size)
        mags = np.abs(F)
        phases = np.angle(F)
        
        # Damped update
        new_mags = (1.0 - damping) * mags + damping * target_mag
        new_mags[0] = mags[0]
        
        # Handle the Nyquist bin (last element of rfft) - no conjugate pair
        F_new = new_mags * np.exp(1j * phases)
        f_new = np.fft.irfft(F_new, size)[:n]
        
        # Soft projection: use max(f, 0) but with a floor
        f_new = np.maximum(f_new, 0.0)
        fn = _normalize(f_new)
        if fn is None:
            f = np.ones(n) / n
        else:
            f = fn
        
        score = _score_raw(f)
        if score < best_score:
            best_score = score
            best_f = f.copy()
        
        if (it + 1) % 500 == 0:
            print(f"    FS it={it+1}: best={best_score:.6f}")
    
    return best_f, best_score

def _mirror_descent_momentum(f_init, n_iter, eta_scale, threshold_frac, 
                              momentum=0.5, label=""):
    """
    Mirror descent with momentum (EMA of pressure vector) and 
    backtracking line search.
    """
    f = np.asarray(f_init, dtype=np.float64).copy()
    f = _normalize(f)
    if f is None:
        return f_init.copy(), float('inf')
    n = len(f)
    best_f = f.copy()
    best_score = _score_raw(f)
    
    # Momentum buffer
    p_ema = np.zeros(n, dtype=np.float64)
    
    for it in range(n_iter):
        h = _fft_convolve_full(f)
        M = float(np.max(h))
        if M <= 0:
            break
        
        threshold = threshold_frac * M + 1e-15
        active = np.where(h >= threshold)[0]
        if len(active) == 0:
            active = np.array([int(np.argmax(h))], dtype=np.int64)
        
        # Compute pressure vector
        p = np.zeros(n, dtype=np.float64)
        for k in active:
            j_start = max(0, k - n + 1)
            j_end = min(n - 1, k)
            if j_end >= j_start:
                lo = k - j_end
                hi = k - j_start
                p[j_start:j_end + 1] += 2.0 * f[lo:hi + 1][::-1]
        p /= float(len(active))
        
        # EMA
        p_ema = momentum * p_ema + (1.0 - momentum) * p
        
        # Backtracking line search
        eta = eta_scale / M
        improved = False
        for bt in range(5):
            log_ratio = -eta * p_ema
            log_ratio = np.clip(log_ratio, -30.0, 30.0)
            f_new = f * np.exp(log_ratio)
            f_new = _normalize(f_new)
            if f_new is None:
                break
            score_new = _score_raw(f_new)
            if score_new < best_score + 1e-10:
                f = f_new
                if score_new < best_score:
                    best_score = score_new
                    best_f = f.copy()
                improved = True
                break
            eta *= 0.5
        
        if not improved:
            # If no improvement even with tiny steps, try a random perturbation
            if (it + 1) % 500 == 0:
                f = f + 0.01 * np.random.default_rng(it).random(n)
                f = _normalize(f)
                if f is None:
                    break
        
        if (it + 1) % 2000 == 0:
            print(f"    MD+ it={it+1}: best={best_score:.6f}, M={M:.10f}, |K|={len(active)}")
    
    return best_f, best_score

def _peak_reduction(f_init, n_iters, alpha=0.05):
    """
    Targeted peak reduction: reduce elements that most contribute to the
    highest autocorrelation lags.
    """
    f = np.asarray(f_init, dtype=np.float64).copy()
    f = _normalize(f)
    if f is None:
        return f_init.copy(), float('inf')
    n = len(f)
    best_f = f.copy()
    best_score = _score_raw(f)
    
    for it in range(n_iters):
        h = _fft_convolve_full(f)
        M = float(np.max(h))
        # Focus on the top few lags
        top_k = max(1, min(5, len(h)))
        peak_lags = np.argsort(h)[-top_k:]
        
        # Compute contribution of each element to peak lags
        reduction = np.zeros(n)
        for k in peak_lags:
            j_start = max(0, k - n + 1)
            j_end = min(n - 1, k)
            if j_end >= j_start:
                lo = k - j_end
                hi = k - j_start
                # Contribution of f[j] to h[k] is f[j] * f[k-j]
                contribs = f[j_start:j_end+1] * f[lo:hi+1][::-1]
                reduction[j_start:j_end+1] += contribs
        
        # Normalize reduction
        max_red = np.max(reduction)
        if max_red > 0:
            reduction /= max_red
        
        # Apply with adaptive alpha
        f = f - alpha * reduction
        f = _normalize(f)
        if f is None:
            break
        
        score = _score_raw(f)
        if score < best_score:
            best_score = score
            best_f = f.copy()
        
        if (it + 1) % 1000 == 0:
            print(f"    PR it={it+1}: best={best_score:.6f}")
    
    return best_f, best_score

def _cosine_init(n):
    j = np.arange(n, dtype=np.float64)
    f = np.cos(np.pi * (j + 0.5) / float(n)) ** 2
    f = np.maximum(f, 0.0)
    s = np.sum(f)
    if s > 0:
        f /= s
    return f

def _gaussian_init(n, sigma_frac):
    j = np.arange(n, dtype=np.float64) - (n - 1) / 2.0
    sigma = sigma_frac * float(n)
    f = np.exp(-j * j / (2.0 * sigma * sigma))
    s = np.sum(f)
    if s > 0:
        f /= s
    return f
