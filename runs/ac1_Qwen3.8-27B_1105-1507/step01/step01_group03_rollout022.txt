import numpy as np
from scipy.optimize import linprog
from scipy.sparse import csr_matrix
from scipy.special import logsumexp

def normalize_sequence(arr):
    x = np.asarray(arr, dtype=np.float64).ravel()
    if x.size == 0:
        return None
    x = np.where(np.isfinite(x), x, 0.0)
    x = np.maximum(x, 0.0)
    s = float(x.sum())
    if s <= 1e-300:
        return None
    return x / s

def score_normalized(x, n):
    conv = np.convolve(x, x)
    m = float(np.max(conv))
    if not np.isfinite(m):
        return float("inf")
    return float(2.0 * n * m)

def project_simplex(v, s=1.0):
    """Project v onto {x : sum(x)=s, x>=0}."""
    n = len(v)
    v = np.asarray(v, dtype=np.float64)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.where(u * (np.arange(1, n+1) + 1) > (cssv - s))[0][-1]
    theta = (cssv[rho] - s) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def smooth_objective(f, n, beta):
    """Compute smooth approximation of max(conv(f,f)) using logsumexp."""
    conv = np.convolve(f, f)
    return float(logsumexp(beta * conv) / beta)

    """Compute gradient of smooth objective w.r.t. f."""

"""
Hybrid LP + subgradient search for minimizing 2*n*max(conv(f,f))/(sum(f))^2.
Uses LP-based direction finding with diverse asymmetric initializations,
subgradient polishing with momentum, and multi-length exploration.
Implements a three-phase strategy: diverse exploration, focused LP optimization,
and subgradient refinement.
"""

import time
import numpy as np
from scipy.optimize import linprog
from scipy.sparse import csr_matrix


def normalize_sequence(arr):
    """Return a nonnegative floating vector normalized to sum 1, or None."""
    try:
        x = np.asarray(arr, dtype=np.float64).ravel()
    except Exception:
        return None
    if x.size == 0:
        return None
    x = np.where(np.isfinite(x), x, 0.0)
    x = np.maximum(x, 0.0)
    s = float(x.sum())
    if s <= 1e-300:
        return None
    return x / s


def score_normalized(x, n):
    """Score assuming x is nonnegative and sum(x) = 1."""
    conv = np.convolve(x, x)
    m = float(np.max(conv))
    if not np.isfinite(m):
        return float("inf")
    return float(2.0 * n * m)


def project_simplex(v, target_sum=1.0):
    """Project vector v onto the simplex {x : sum(x)=target_sum, x>=0}."""
    v = np.asarray(v, dtype=np.float64)
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho_candidates = np.where(u * (np.arange(1, n + 1) + 1.0) > (cssv - target_sum))[0]
    if len(rho_candidates) == 0:
        return np.full(n, target_sum / n, dtype=np.float64)
    rho = rho_candidates[-1]
    theta = (cssv[rho] - target_sum) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def make_conv_template(n):
    """Precompute the CSR matrix structure for convolve(f, g)."""
    m = 2 * n - 1
    k64 = np.arange(m, dtype=np.int64)
    js = np.maximum(0, k64 - n + 1).astype(np.int32)
    je = np.minimum(n - 1, k64).astype(np.int32)
    counts = (je.astype(np.int64) - js.astype(np.int64) + 1).astype(np.int64)
    total = int(counts.sum())

    cum = np.cumsum(counts)
    starts = np.repeat(cum - counts, counts)
    local = np.arange(total, dtype=np.int64) - starts

    row = np.repeat(k64, counts).astype(np.int32)
    col_base = js[row.astype(np.int64)]
    col = (col_base + local).astype(np.int32)
    offsets = (row.astype(np.int64) - col.astype(np.int64)).astype(np.int32)

    A_off = csr_matrix((offsets, (row, col)), shape=(m, n), dtype=np.int32)
    edge_offsets = A_off.data.astype(np.int32, copy=True)
    A_template = A_off.astype(np.float64).copy()
    A_template.data.fill(0.0)

    c_obj = -np.ones(n, dtype=np.float64)
    bounds = [(0.0, None)] * n
    return A_template, edge_offsets, c_obj, bounds, m


def solve_direction_lp(template, f, n, M):
    """Solve the LP direction problem for the given f and M."""
    A_template, edge_offsets, c_obj, bounds, m = template
    lp_scale = float(n)

    A = A_template.copy()
    A.data[:] = f[edge_offsets]
    A.data *= lp_scale

    b = np.full(m, M * lp_scale * lp_scale, dtype=np.float64)
    return linprog(c_obj, A_ub=A, b_ub=b, bounds=bounds, method="highs")


def subgradient_at(f, k_star):
    """Compute subgradient of max_k (f*f)_k at f, using active constraint k_star."""
    n = len(f)
    sg = np.zeros(n, dtype=np.float64)
    i_lo = max(0, k_star - n + 1)
    i_hi = min(n - 1, k_star)
    idx = np.arange(i_lo, i_hi + 1)
    sg[idx] = 2.0 * f[k_star - idx]
    return sg

    """
    Refine f using projected subgradient method with momentum.
    Returns best f found and its score.
    """
    f = normalize_sequence(f)
    if f is None:
        return None, float("inf")

    best_f = f.copy()
    best_score = score_normalized(f, n)
    deadline = time.monotonic() + max(1.0, time_budget)

    y = f.copy()
    v = np.zeros(n, dtype=np.float64)
    momentum = 0.6
    alpha = 0.05
    no_improve = 0
    iteration = 0

    while time.monotonic() < deadline:
        iteration += 1
        t0 = time.monotonic()

        conv = np.convolve(y, y)
        k_star = int(np.argmax(conv))
        sg = subgradient_at(y, k_star)

        # Momentum update
        v = momentum * v - alpha * sg
        f_new = project_simplex(y + v)

        sc_new = score_normalized(f_new, n)

        if np.isfinite(sc_new) and sc_new < best_score - 1e-12:
            f = f_new
            best_f = f.copy()
            best_score = sc_new
            no_improve = 0
            y = f_new
            # Slightly increase alpha on success
            alpha = min(alpha * 1.02, 0.2)
        else:
            no_improve += 1
            alpha *= 0.92
            momentum = min(momentum * 1.01, 0.85)
            y = best_f  # Reset y to best
            v *= 0.5

        if no_improve > 200:
            # Try a kick: random perturbation
            noise = rng.dirichlet(np.ones(n) * 0.5)
            f_new = (1 - 0.1) * best_f + 0.1 * noise
            f_new = np.maximum(f_new, 0)
            s = f_new.sum()
            if s > 1e-300:
                f_new /= s
                sc_new = score_normalized(f_new, n)
                if sc_new < best_score:
                    best_f = f_new
                    best_score = sc_new
                    f = f_new
                    y = f_new
                    no_improve = 0
                    alpha = 0.05
                    momentum = 0.6
                else:
                    no_improve += 50

        if no_improve > 500:
            break

        if iteration % 50 == 0 and time.monotonic() - t0 > 0:
            pass  # Could add logging here

    return best_f, best_score


def get_asymmetric_starts(n, rng, given_f=None, best_f=None):
    """Generate diverse asymmetric initializations."""
    starts = []

    # Given sequence (up-sampled if needed)
    if given_f is not None:
        g = _interp_to_length(given_f, n)
        if g is not None:
            starts.append(g)

    if best_f is not None and best_f.size != n:
        b = _interp_to_length(best_f, n)
        if b is not None:
            starts.append(b)
    elif best_f is not None:
        starts.append(best_f.copy())

    # Exponential decay (left-heavy)
    for rate in [0.7, 0.85, 0.95]:
        x = np.array([rate ** i for i in range(n)])
        x = normalize_sequence(x)
        if x is not None:
            starts.append(x)

    # Exponential decay (right-heavy)
    for rate in [0.85, 0.95]:
        x = np.array([rate ** (n - 1 - i) for i in range(n)])
        x = normalize_sequence(x)
        if x is not None:
            starts.append(x)

    # Linear ramp (increasing)
    x = np.arange(1, n + 1, dtype=np.float64)
    x = normalize_sequence(x)
    if x is not None:
        starts.append(x)

    # Linear ramp (decreasing)
    x = np.arange(n, 0, -1, dtype=np.float64)
    x = normalize_sequence(x)
    if x is not None:
        starts.append(x)

    # Half-filled (left)
    x = np.zeros(n)
    x[:n // 2] = 1.0
    x = normalize_sequence(x)
    if x is not None:
        starts.append(x)

    # Gaussian off-center
    center = n * 0.3
    sigma = n * 0.15
    x = np.exp(-0.5 * ((np.arange(n) - center) / sigma) ** 2)
    x = normalize_sequence(x)
    if x is not None:
        starts.append(x)

    # Random Dirichlet
    x = rng.dirichlet(np.ones(n) * 0.3)
    starts.append(x)

    # Uniform with noise
    x = np.ones(n) * (1 + 0.3 * rng.standard_normal(n))
    x = normalize_sequence(x)
    if x is not None:
        starts.append(x)

    # Deduplicate
    unique = []
    for s in starts:
        is_dup = False
        for u in unique:
            if np.linalg.norm(s - u) < 1e-4:
                is_dup = True
                break
        if not is_dup:
            unique.append(s)

    return unique[:8]


def _interp_to_length(x, n):
    """Resample a nonnegative sequence to length n and normalize."""
    x = normalize_sequence(x)
    if x is None or n <= 0:
        return None
    m = x.size
    if m == n:
        return x
    if m == 1:
        return np.ones(n, dtype=np.float64) / float(n)
    old = np.linspace(0.0, 1.0, m, dtype=np.float64)
    new = np.linspace(0.0, 1.0, n, dtype=np.float64)
    out = np.interp(new, old, x)
    return normalize_sequence(out)


def optimize_lp(n, f_init, time_budget, rng, template):
    """
    Run the LP-based iteration from one initial vector and return the best
    normalized vector and its normalized score.
    """
    f = normalize_sequence(f_init)
    if f is None:
        f = np.ones(n, dtype=np.float64) / float(n)

    if template is None:
        try:
            template = make_conv_template(n)
        except Exception:
            return f, score_normalized(f, n)

    best_f = f.copy()
    best_score = score_normalized(f, n)
    start_time = time.monotonic()
    deadline = start_time + max(1.0, time_budget)
    stall = 0
    iteration = 0

    while time.monotonic() < deadline:
        iteration += 1
        t0 = time.monotonic()

        conv_ff = np.convolve(f, f)
        M = float(np.max(conv_ff))
        if not np.isfinite(M) or M <= 0.0:
            break

        try:
            res = solve_direction_lp(template, f, n, M)
        except Exception:
            stall += 1
            if stall > 15:
                break
            # Perturb and continue
            noise = rng.dirichlet(np.ones(n) * 0.5)
            f = (1 - 0.2) * best_f + 0.2 * noise
            f = normalize_sequence(f)
            if f is None:
                f = best_f.copy()
            continue

        x = getattr(res, "x", None)
        if x is None:
            stall += 1
            if stall > 15:
                break
            noise = rng.dirichlet(np.ones(n) * 0.5)
            f = (1 - 0.2) * best_f + 0.2 * noise
            f = normalize_sequence(f)
            if f is None:
                f = best_f.copy()
            continue

        x = np.asarray(x, dtype=np.float64).ravel()
        if x.size != n:
            break
        x = np.where(np.isfinite(x), x, 0.0)
        x = np.maximum(x, 0.0)
        s_star = float(x.sum())
        if s_star <= 1e-300:
            stall += 1
            if stall > 15:
                break
            noise = rng.dirichlet(np.ones(n) * 0.5)
            f = (1 - 0.2) * best_f + 0.2 * noise
            f = normalize_sequence(f)
            if f is None:
                f = best_f.copy()
            continue

        s_ratio = s_star / float(n)
        if s_ratio <= 1.0 + 1e-8:
            stall += 1
            if stall > 15:
                break
            noise = rng.dirichlet(np.ones(n) * 0.5)
            f = (1 - 0.3) * best_f + 0.3 * noise
            f = normalize_sequence(f)
            if f is None:
                f = best_f.copy()
            continue

        g = x / s_star

        r = np.convolve(f, g)
        r_max = float(np.max(r))
        if (not np.isfinite(r_max)) or r_max <= 0.0:
            stall += 1
            if stall > 15:
                break
            continue

        if r_max >= M * (1.0 - 1e-10):
            stall += 1
            if stall > 15:
                break
            continue

        S_eff = M / r_max
        if S_eff <= 1.0 + 1e-8:
            stall += 1
            if stall > 15:
                break
            continue

        G = float(np.max(np.convolve(g, g)))
        if not np.isfinite(G):
            stall += 1
            if stall > 15:
                break
            continue

        # Safe step computation
        a_coef = 2.0 * (r_max - M)
        b_coef = G + M - 2.0 * r_max
        if b_coef > 1e-14:
            t_max = -a_coef / b_coef
        else:
            t_max = 1.0

        if (not np.isfinite(t_max)) or t_max <= 0.0:
            t_max = 1e-8
        base = min(1.0, 0.75 * t_max)
        if base < 1e-12:
            base = 1e-12

        # Try multiple step sizes
        raw_ts = [base, 0.5 * base, 0.25 * base, 0.1 * base, 0.05 * base, 0.01 * base]
        ts = []
        for tv in sorted(raw_ts, reverse=True):
            tv = max(1e-14, min(1.0, float(tv)))
            if tv not in ts:
                ts.append(tv)
            if len(ts) >= 6:
                break

        best_cand_f = None
        best_cand_score = best_score
        best_t = None

        for tv in ts:
            f_new = (1.0 - tv) * f + tv * g
            f_new = np.maximum(f_new, 0.0)
            ssum = float(f_new.sum())
            if ssum <= 1e-300:
                continue
            f_new /= ssum
            sc = score_normalized(f_new, n)
            if np.isfinite(sc) and sc < best_cand_score:
                best_cand_score = sc
                best_cand_f = f_new
                best_t = tv

        if best_cand_f is not None and best_cand_score < best_score:
            f = best_cand_f
            best_f = f.copy()
            best_score = best_cand_score
            stall = 0
            if iteration <= 5 or iteration % 20 == 0:
                elapsed = time.monotonic() - t0
                print(f"      n={n} iter={iteration} score={best_score:.6f} M={M:.6g} t={best_t:.4f} dt={elapsed:.2f}s", flush=True)
        else:
            stall += 1
            # Perturb to escape local optimum
            eps = min(0.5, 0.1 + 0.05 * stall)
            noise = rng.dirichlet(np.ones(n) * (1.0 / eps + 1.0))
            f = (1.0 - eps) * best_f + eps * noise
            f = normalize_sequence(f)
            if f is None:
                f = best_f.copy()
            if stall > 20:
                break

    return best_f, float(best_score)


def propose_candidate(seed=42, budget_s=1500, **kwargs):
    """
    Search for a low-scoring nonnegative sequence using hybrid LP + subgradient.
    Three phases: diverse exploration, focused LP, subgradient polish.
    """
    rng = np.random.default_rng(int(seed) if seed is not None else 42)
    total_budget = max(1.0, float(budget_s))
    start_wall = time.monotonic()
    reserve = 20.0
    deadline = start_wall + max(1.0, total_budget - reserve)

    best_f = None
    best_score = float("inf")
    given_f = None

    # Load given sequence if available
    if total_budget >= 30.0:
        try:
            raw = np.asarray(height_sequence_1, dtype=np.float64)
            given_f = normalize_sequence(raw)
            if given_f is not None:
                best_f = given_f.copy()
                best_score = score_normalized(given_f, given_f.size)
                print(f"given sequence: n={given_f.size}, score={best_score:.6f}", flush=True)
        except Exception:
            given_f = None

    if best_f is None:
        best_f = np.ones(64, dtype=np.float64) / 64.0
        best_score = 2.0

    # Phase 1: Diverse exploration across lengths (30% of time)
    phase1_budget = 0.30 * (total_budget - reserve)
    phase1_deadline = start_wall + phase1_budget

    plan_phase1 = [
        (256, 0.20),
        (512, 0.25),
        (1024, 0.30),
        (2048, 0.25),
    ]

    print(f"\n=== Phase 1: Diverse exploration (budget={phase1_budget:.0f}s) ===", flush=True)

    for n, frac in plan_phase1:
        rem = phase1_deadline - time.monotonic()
        if rem < 10.0:
            break
        alloc = min(frac * phase1_budget, rem - 5.0)
        if alloc < 8.0:
            continue

        print(f"\n  n={n}, alloc={alloc:.1f}s, current_best={best_score:.6f}", flush=True)

        try:
            template = make_conv_template(n)
        except MemoryError:
            print(f"  memory error for n={n}", flush=True)
            continue
        except Exception as e:
            print(f"  template error n={n}: {e}", flush=True)
            continue

        starts = get_asymmetric_starts(n, rng, given_f, best_f)

        # Allocate time to starts
        m = len(starts)
        per_start = alloc / m
        if per_start < 5.0 and m > 2:
            # Only use best 2 starts
            starts = starts[:2]
            per_start = alloc / 2

        for idx, init in enumerate(starts):
            rem = phase1_deadline - time.monotonic()
            if rem < 5.0:
                break
            tb = min(per_start, rem - 2.0)
            if tb < 4.0:
                continue

            f_res, s_res = optimize_lp(n, init, tb, rng, template)
            if np.isfinite(s_res) and s_res < best_score:
                best_score = float(s_res)
                best_f = np.asarray(f_res, dtype=np.float64).copy()
                print(f"  NEW BEST: n={best_f.size}, score={best_score:.6f}", flush=True)

        try:
            del template
        except Exception:
            pass

    # Phase 2: Focused LP optimization on best length (50% of time)
    phase2_budget = 0.50 * (total_budget - reserve)
    phase2_deadline = start_wall + phase1_budget + phase2_budget

    print(f"\n=== Phase 2: Focused LP (budget={phase2_budget:.0f}s) ===", flush=True)
    print(f"  Best so far: n={best_f.size}, score={best_score:.6f}", flush=True)

    # Focus on the best length and a couple neighbors
    n_best = best_f.size
    n_candidates = sorted(set([n_best, n_best // 2, n_best * 2, 1024, 2048]))
    n_candidates = [n for n in n_candidates if 128 <= n <= 4096]
    n_candidates = sorted(n_candidates)[:4]

    for n in n_candidates:
        rem = phase2_deadline - time.monotonic()
        if rem < 15.0:
            break
        alloc = rem * 0.4  # Use 40% of remaining for each n
        alloc = min(alloc, phase2_deadline - time.monotonic() - 5.0)
        if alloc < 10.0:
            continue

        print(f"\n  Focused n={n}, alloc={alloc:.1f}s", flush=True)

        try:
            template = make_conv_template(n)
        except MemoryError:
            continue
        except Exception:
            continue

        # Generate multiple diverse starts from best
        starts = []
        if best_f is not None:
            b = _interp_to_length(best_f, n)
            if b is not None:
                starts.append(b)
                # Perturbed versions
                for eps in [0.05, 0.1, 0.2, 0.3]:
                    noise = rng.dirichlet(np.ones(n) * (1.0 / eps))
                    ps = (1 - eps) * b + eps * noise
                    ps = normalize_sequence(ps)
                    if ps is not None:
                        starts.append(ps)

        # Add some fresh asymmetric starts
        asym = get_asymmetric_starts(n, rng, given_f, None)
        for a in asym[:3]:
            if len(starts) < 8:
                starts.append(a)

        m = len(starts)
        per_start = alloc / m
        if per_start < 5.0:
            starts = starts[:3]
            per_start = alloc / max(1, len(starts))

        for idx, init in enumerate(starts):
            rem = phase2_deadline - time.monotonic()
            if rem < 8.0:
                break
            tb = min(per_start, rem - 3.0)
            if tb < 5.0:
                continue

            f_res, s_res = optimize_lp(n, init, tb, rng, template)
            if np.isfinite(s_res) and s_res < best_score:
                best_score = float(s_res)
                best_f = np.asarray(f_res, dtype=np.float64).copy()
                print(f"  NEW BEST: n={best_f.size}, score={best_score:.6f}", flush=True)

        try:
            del template
        except Exception:
            pass

    # Phase 3: Subgradient polish + final LP (20% of time)
    phase3_budget = 0.20 * (total_budget - reserve)
    phase3_deadline = start_wall + total_budget - reserve

    print(f"\n=== Phase 3: Subgradient polish (budget={phase3_budget:.0f}s) ===", flush=True)
    print(f"  Best so far: n={best_f.size}, score={best_score:.6f}", flush=True)

    n_final = best_f.size
    sg_budget = phase3_budget * 0.5
    lp_budget = phase3_budget * 0.5

    # Subgradient polish
    if sg_budget > 5.0:
        print(f"  Subgradient polish on n={n_final}, budget={sg_budget:.1f}s", flush=True)
        f_sg, s_sg = subgradient_polish(best_f, n_final, sg_budget, rng)
        if f_sg is not None and np.isfinite(s_sg) and s_sg < best_score:
            best_score = float(s_sg)
            best_f = np.asarray(f_sg, dtype=np.float64).copy()
            print(f"  SG NEW BEST: n={best_f.size}, score={best_score:.6f}", flush=True)

    # Final LP polish
    if lp_budget > 5.0:
        rem = phase3_deadline - time.monotonic()
        if rem > 5.0:
            print(f"  Final LP polish on n={n_final}, budget={min(lp_budget, rem):.1f}s", flush=True)
            try:
                template = make_conv_template(n_final)
                f_final, s_final = optimize_lp(n_final, best_f, min(lp_budget, rem - 2.0), rng, template)
                if np.isfinite(s_final) and s_final < best_score:
                    best_score = float(s_final)
                    best_f = np.asarray(f_final, dtype=np.float64).copy()
                    print(f"  LP FINAL BEST: n={best_f.size}, score={best_score:.6f}", flush=True)
            except Exception as e:
                print(f"  Final LP error: {e}", flush=True)
            try:
                del template
            except Exception:
                pass

    # Also try subgradient + LP on a slightly different length if time permits
    rem = phase3_deadline - time.monotonic()
    if rem > 30.0 and best_f is not None:
        n_alt = n_final * 2 if n_final <= 2048 else n_final // 2
        if 128 <= n_alt <= 4096:
            print(f"\n  Bonus: trying n={n_alt}, budget={rem-5:.1f}s", flush=True)
            try:
                template = make_conv_template(n_alt)
                init = _interp_to_length(best_f, n_alt)
                if init is not None:
                    tb_sg = 0.4 * (rem - 5)
                    tb_lp = 0.5 * (rem - 5)
                    f_sg2, s_sg2 = subgradient_polish(init, n_alt, tb_sg, rng)
                    if f_sg2 is not None:
                        init = f_sg2
                    f_res, s_res = optimize_lp(n_alt, init, tb_lp, rng, template)
                    if np.isfinite(s_res) and s_res < best_score:
                        best_score = float(s_res)
                        best_f = np.asarray(f_res, dtype=np.float64).copy()
                        print(f"  BONUS NEW BEST: n={best_f.size}, score={best_score:.6f}", flush=True)
            except Exception as e:
                print(f"  Bonus error: {e}", flush=True)
            try:
                del template
            except Exception:
                pass

    # Final evaluation and return
    if best_f is None:
        best_f = np.ones(64, dtype=np.float64) / 64.0

    seq = [float(x) for x in best_f]
    try:
        final_score = float(evaluate_sequence(seq))
    except Exception:
        final_score = float("inf")

    if not np.isfinite(final_score) and given_f is not None:
        try:
            seq = [float(x) for x in given_f]
            final_score = float(evaluate_sequence(seq))
        except Exception:
            pass

    if not np.isfinite(final_score):
        seq = [1.0 / 64.0] * 64

    elapsed = time.monotonic() - start_wall
    print(f"\nFINAL: n={len(seq)}, local_best={best_score:.6f}, evaluate={final_score:.6f}, time={elapsed:.1f}s", flush=True)
    return seq
