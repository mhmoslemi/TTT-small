            no_improve += 1
            step *= 0.99
            if no_improve > 20000:
                break
        if step < 1e-10:
            break

    return best_f, best_score


def build_conv_matrix_sparse(f):
    """Build sparse matrix A where (A@g)[k] = convolve(f,g)[k]."""
    n = len(f)
    rows_list = []
    cols_list = []
    vals_list = []
    for k in range(2 * n - 1):
        j_min = max(0, k - (n - 1))
        j_max = min(k, n - 1)
        for j in range(j_min, j_max + 1):
            rows_list.append(k)
            cols_list.append(j)
            vals_list.append(f[k - j])
    rows = np.array(rows_list, dtype=np.int32)
    cols = np.array(cols_list, dtype=np.int32)
    vals = np.array(vals_list, dtype=np.float64)
    return csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))


def lp_direction(f, M):
    """Solve LP: max sum(g) s.t. conv(f,g)_k <= M, g >= 0."""
    n = len(f)
    A = build_conv_matrix_sparse(f)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    try:
        result = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                         options={'time_limit': 30})
        if result.success:
            return result.x
    except Exception:
        pass
    return None


def lp_polish(f, deadline, step_size_init=0.1, max_iters=50):
    """LP-based polish. f normalized to sum=1."""
    n = len(f)
    f = f.copy().astype(np.float64)
    f = f / f.sum()
    conv0 = np.convolve(f, f)
    best_score = 2.0 * n * conv0.max()
    best_f = f.copy()
    step_size = step_size_init
    no_improve = 0

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        conv_f = np.convolve(f, f)
        M = float(np.max(conv_f))
        if M <= 0:
            break

        g = lp_direction(f, M)
        if g is None:
            no_improve += 1
            if no_improve > 5:
                break
            continue

        sum_g = float(np.sum(g))
        if sum_g <= 1e-15:
            no_improve += 1
            continue

        # Rescale g to same sum as f
        g = g * (float(np.sum(f)) / sum_g)

        # Blend
        new_f = (1.0 - step_size) * f + step_size * g
        new_f = np.maximum(new_f, 0.0)
        s = new_f.sum()
        if s < 1e-12:
            step_size *= 0.5
            continue
        new_f = new_f / s

        conv_new = np.convolve(new_f, new_f)
        score = 2.0 * n * conv_new.max()

        if score < best_score:
            best_score = score
            best_f = new_f.copy()
            f = new_f
            no_improve = 0
            print(f"    LP iter {it}: score={score:.6f} (step={step_size:.3f})")
        else:
            no_improve += 1
            step_size *= 0.7
            if step_size < 0.005 or no_improve > 15:
                break

    return best_f, best_score


def make_initialization(n, kind, rng=None):
    """Create various initializations of length n."""
    if rng is None:
        rng = np.random.default_rng(42)
    idx = np.arange(n, dtype=np.float64)

    if kind == 'uniform':
        return np.ones(n)
    elif kind == 'cosine':
        return 0.5 * (1.0 + np.cos(2.0 * np.pi * idx / (n - 1)))
    elif kind == 'hann':
        return np.hanning(n) + 0.1
    elif kind == 'gauss_center':
        center = (n - 1) / 2.0
        sigma = n / 6.0
        return np.exp(-0.5 * ((idx - center) / sigma) ** 2)
    elif kind == 'gauss_wide':
        center = (n - 1) / 2.0
        sigma = n / 3.0
        return np.exp(-0.5 * ((idx - center) / sigma) ** 2)
    elif kind == 'bimodal':
        s1 = np.exp(-0.5 * ((idx - n * 0.2) / (n / 12.0)) ** 2)
        s2 = np.exp(-0.5 * ((idx - n * 0.8) / (n / 12.0)) ** 2)
        return s1 + s2
    elif kind == 'random_smooth':
        r = rng.uniform(0.3, 2.0, size=n)
        # Smooth with a small kernel
        kernel_size = max(3, n // 100)
        if kernel_size % 2 == 0:
            kernel_size += 1
        kernel = np.exp(-0.5 * (np.arange(-kernel_size // 2, kernel_size // 2 + 1) / (kernel_size // 4 + 1)) ** 2)
        kernel /= kernel.sum()
        return np.convolve(r, kernel, mode='same')
    elif kind == 'lowfreq':
        # Low-frequency random: random coefficients for first few FFT modes
        coeffs = rng.normal(0, 1, size=n // 4 + 1)
        f = np.zeros(n)
        for i in range(len(coeffs)):
            f += coeffs[i] * np.cos(2 * np.pi * i * idx / n)
        return np.abs(f) + 0.1
    elif kind == 'plateau':
        result = np.ones(n)
        result[:n // 6] = 0.4
        result[-(n // 6):] = 0.4
        return result
    else:
        return np.ones(n)


def propose_candidate(budget_s=1500, **kwargs):
    """
    Main search function. Tries multiple lengths and initializations,
    using soft-max gradient descent, subgradient, and LP methods.
    """
    t0 = time.monotonic()
    deadline = t0 + max(0.0, float(budget_s) - 8.0)

    best_score = float('inf')
    best_seq = None

    rng = np.random.default_rng(123)

    # Try parent
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64).copy()
        parent = np.clip(parent, 0.0, 1000.0)
        if parent.ndim == 1 and parent.size > 0 and np.all(np.isfinite(parent)):
            score = evaluate_sequence(parent.tolist())
            if np.isfinite(score):
                best_score = score
                best_seq = parent.copy()
                print(f"Parent n={parent.size}: score={score:.6f}")
    except (NameError, Exception) as e:
        print(f"Could not load parent: {e}")

    # Define search plan: (length, time_budget_fraction)
    # Focus more time on promising lengths
    search_plan = [
        (512, 0.10),
        (768, 0.10),
        (1024, 0.30),
        (1536, 0.25),
        (2048, 0.25),
    ]

    initializations = ['uniform', 'cosine', 'hann', 'gauss_center',
                       'gauss_wide', 'bimodal', 'random_smooth', 'lowfreq']

    for n, frac in search_plan:
        if time.monotonic() > deadline - 20:
            print("Time running out, stopping search.")
            break

        time_budget = int(frac * (budget_s - 20))
        n_deadline = t0 + time_budget

        print(f"\n=== n={n}, budget={time_budget}s ===")

        inits_to_try = initializations[:4]  # Try subset for larger n
        if n <= 768:
            inits_to_try = initializations  # Try all for small n

        for init_kind in inits_to_try:
            if time.monotonic() > n_deadline:
                break

            f_init = make_initialization(n, init_kind, rng)
            f_init = np.clip(f_init, 0.0, 1000.0)
            if f_init.sum() < 1e-10:
                continue
            f_init = f_init / f_init.sum()

            # Quick score
            conv0 = np.convolve(f_init, f_init)
            init_score = 2.0 * n * conv0.max()

            # Start with soft-max gradient descent (fast)
            sm_deadline = min(n_deadline, time.monotonic() + time_budget * 0.5)
            f_best, s_best = softmax_momentum_descent(
                f_init, sm_deadline, tau_frac=0.12, tau_decay=0.9997,
                step_init=0.08, verbose_prefix=f"[n={n},{init_kind}] ")

            if s_best < best_score:
                best_score = s_best
                best_seq = f_best.copy()
                print(f"  n={n} {init_kind}: softmax -> {s_best:.6f}")

            # Subgradient continuation
            if time.monotonic() < n_deadline - 5:
                sg_deadline = min(n_deadline, time.monotonic() + time_budget * 0.3)
                f_sg, s_sg = subgradient_optimize(
                    f_best, sg_deadline, step_init=0.01,
                    verbose_prefix=f"[n={n},{init_kind}] ")
                if s_sg < best_score:
                    best_score = s_sg
                    best_seq = f_sg.copy()
                    print(f"  n={n} {init_kind}: subgrad -> {s_sg:.6f}")
                    f_best = f_sg

            # LP polish if time permits
            if time.monotonic() < n_deadline - 10 and s_best < 1.6:
                f_lp, s_lp = lp_polish(f_best, n_deadline, step_size_init=0.08,
                                       max_iters=20)
                if s_lp < best_score:
                    best_score = s_lp
                    best_seq = f_lp.copy()
                    print(f"  n={n} {init_kind}: LP -> {s_lp:.6f}")

    # Final refinement on the best sequence found
    if best_seq is not None and time.monotonic() < deadline - 10:
        print(f"\n=== Final refinement (n={len(best_seq)}, score={best_score:.6f}) ===")
        n_best = len(best_seq)

        # Soft-max continues
        f_final, s_final = softmax_momentum_descent(
            best_seq, deadline - 5, tau_frac=0.05, tau_decay=0.9998,
            step_init=0.03, verbose_prefix="[final] ")
        if s_final < best_score:
            best_score = s_final
            best_seq = f_final.copy()

        # Subgradient
        if time.monotonic() < deadline - 5:
            f_sg, s_sg = subgradient_optimize(best_seq, deadline - 3,
                                              step_init=0.005,
                                              verbose_prefix="[final] ")
            if s_sg < best_score:
                best_score = s_sg
                best_seq = f_sg.copy()

        # LP polish
        if time.monotonic() < deadline - 5 and n_best <= 1536:
            f_lp, s_lp = lp_polish(best_seq, deadline, step_size_init=0.05,
                                   max_iters=15)
            if s_lp < best_score:
                best_score = s_lp
                best_seq = f_lp.copy()

    if best_seq is None:
        # Fallback
        n_fb = 512
        best_seq = make_initialization(n_fb, 'cosine', rng)
        best_score = evaluate_sequence(best_seq.tolist())
        print(f"Fallback: n={n_fb}, score={best_score:.6f}")

    elapsed = time.monotonic() - t0
    final_score = evaluate_sequence(best_seq.tolist())
    print(f"\nFINAL: n={len(best_seq)}, eval_score={final_score:.6f}, time={elapsed:.1f}s")

    return [float(x) for x in best_seq]
