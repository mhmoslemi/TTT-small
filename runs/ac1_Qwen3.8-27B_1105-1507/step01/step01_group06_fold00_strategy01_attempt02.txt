def evaluate_sequence(sequence: list[float] or tuple[float]) -> float:
    # Defensive checks as per the function definition
    try:
        sequence = list(float(x) for x in sequence)
    except TypeError:
        return np.inf
    n_val = len(sequence)
    if n_val == 0:
        return np.inf
    
    # Clip values to [0, 1000], compute sum and convolution
    sequence = [max(0.0, x) for x in sequence]
    sequence = [min(1000.0, x) for x in sequence]
    total_sum = np.sum(sequence)
    if total_sum < 0.01:  # Check normalization threshold

    conv = np.convolve(sequence, sequence, mode='full')

def tournament_selection(parents: list[np.ndarray], tournament_size: int = 5) -> np.ndarray:
    # Select a parent uniformly at random from the fitness list
    # And returns one with the best fitness (objective) among randomly sampled parents
    indices = np.random.choice(len(parents), tournament_size, replace=False)
    candidates = [parents[i][0] for i in indices]
    fitnesses = [parents[i][1] for i in indices]
    
    # Return candidate with min objective
    best_idx = np.argmin([fit for fit in fitnesses if np.isfinite(fit)])
    candidate_val = candidates[best_idx]
    return candidate_val

def crossover(a: np.ndarray, b: np.ndarray, alpha: float = 0.5) -> np.ndarray:
    # Perform blending crossover on two sequences
    # We require both sequences to be in [0,1000]^n, finite
    a = a.astype(np.float64)
    b = b.astype(np.float64)
