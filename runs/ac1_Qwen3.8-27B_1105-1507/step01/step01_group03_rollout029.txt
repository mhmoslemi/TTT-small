```python
Hybrid optimizer: coordinate descent + smooth-max L-BFGS for minimizing
2*n*max(convolve(f,f))/(sum(f))^2 over nonnegative sequences.

Key ideas:
1. Fast coordinate descent with incremental convolution updates
2. Smooth max (log-sum-exp) + L-BFGS-B for refinement
3. Multiple sequence lengths and random restarts
4. Time-managed scheduling

import time
import numpy as np
from scipy.optimize import minimize

def normalize_sequence(arr):
    """Normalize to sum 1, nonnegative."""
    x = np.asarray(arr, dtype=np.float64).ravel()
    x = np.where(np.isfinite(x), x, 0.0)
    x = np.maximum(x, 0.0)
    s = x.sum()
    if s < 1e-300:
        return None
    return x / s

def score_normalized(x, n):
    """Score for normalized sequence (sum=1): 2*n*max(f*f)."""
    c = np.convolve(x, x)
    m = np.max(c)
    return 2.0 * n * float(m)

def coord_descent(f, n, max_sweeps=100, time_budget=30.0, rng=None, verbose=False):
    """
    Coordinate descent to minimize max(f*f) with f >= 0.
    Returns the best normalized f and its score.
    """
    if rng is None:
        rng = np.random.default_rng()
    
    f = f.copy().astype(np.float64)
    c = np.convolve(f, f)  # length 2n-1
    best_f = f / f.sum()
    best_score = score_normalized(best_f, n)
    start_time = time.monotonic()
    
    for sweep in range(max_sweeps):
        if time.monotonic() - start_time > time_budget:
            break
        
        order = rng.permutation(n)
        for i in order:
            v = f[i]
            # Find x in [0, upper] minimizing max_k c_new[k]
            # c_new[k] = c[k] + delta_k(x)
            # delta_{2i}(x) = x^2 - v^2
            # delta_k(x) = 2*(x-v)*f[k-i] for k in [i, i+n-1], k != 2i
            # delta_k(x) = 0 otherwise
            
            # Ternary search for x in [0, upper]
            # upper: use max(v * 2, 1.0) or something reasonable
            lo = 0.0
            hi = max(v * 3.0, 1.0)
            
            # Define phi(x) = max_k (c[k] + delta_k(x))
            def phi(x):
                # Quadratic term at k=2i
                quad_val = c[2*i] + x*x - v*v
                # Linear terms for k in [i, i+n-1], k != 2i
                # c[k] + 2*(x-v)*f[k-i]
                k_range = np.arange(i, min(i+n, 2*n-1))
                mask = k_range != 2*i
                if np.any(mask):
                    k_sel = k_range[mask]
                    linear_vals = c[k_sel] + 2.0*(x-v)*f[k_sel - i]
                    lin_max = np.max(linear_vals)
                else:
                    lin_max = -np.inf
                # Constant terms for k not in [i, i+n-1]
                # and also not k = 2i (but 2i might be in or out of range)
                # Actually, let me just compute the max of everything
                # c_new = c.copy()
                # c_new[2*i] += x*x - v*v
                # k_range_all = np.arange(max(0, i), min(2*n-1, i+n-1))
                # for k in k_range_all:
                #     if k != 2*i:
                #         c_new[k] += 2*(x-v)*f[k-i]
                # return np.max(c_new)
                
                # Vectorized:
                c_new = c.copy()
                c_new[2*i] = c[2*i] + x*x - v*v
                k_lo = max(0, i)
                k_hi = min(2*n-2, i+n-1)
                if k_lo <= k_hi:
                    ks = np.arange(k_lo, k_hi+1)
                    delta = 2.0*(x-v)*f[ks - i]
                    # Don't double-count k=2i
                    delta[ks == 2*i] = 0.0
                    c_new[ks] += delta
                return np.max(c_new)
            
            # Ternary search
            for _ in range(40):
                m1 = lo + (hi - lo) / 3.0
                m2 = hi - (hi - lo) / 3.0
                if phi(m1) < phi(m2):
                    hi = m2
                else:

            x_opt = (lo + hi) / 2.0

            delta_v = x_opt - v
            if abs(delta_v) < 1e-15:
                continue
            f[i] = x_opt
            c[2*i] += x_opt*x_opt - v*v
            k_lo = max(0, i)
            k_hi = min(2*n-2, i+n-1)
            if k_lo <= k_hi:
```

```python
            # What about c[k] for k in [k_lo, k_hi] that are in neither ks_lin nor {2i}?
            # ks_lin = ks with k != 2i removed, so ks = ks_lin ∪ {2i}

        if s < 1e-300:

        f /= s
        c = np.convolve(f, f)
        sc = 2.0 * n * float(np.max(c))

            best_f = f.copy()
        
        sweep += 1
        if verbose and sweep % 10 == 0:
            print(f"    coord_descent sweep {sweep}: score={best_score:.6f}")
    
    return best_f, best_score
```

```python
"""
Hybrid optimizer combining coordinate descent (with incremental convolution
updates) and smooth-max L-BFGS-B to minimize 2*n*max(convolve(f,f))/(sum(f))^2.

Key innovations over the LP-based approach:
1. Fast coordinate descent exploiting the 1D convex structure of each
   coordinate update, with O(n) incremental convolution maintenance.
2. Smooth log-sum-exp approximation of max + L-BFGS-B for polish.
3. Multi-scale search across sequence lengths with adaptive time allocation.
4. Random kicks to escape local minima.
"""

import time
import numpy as np
from scipy.optimize import minimize


def normalize_sequence(arr):
    """Normalize nonnegative array to sum 1. Return None if all zeros."""
    x = np.asarray(arr, dtype=np.float64).ravel()
    x = np.where(np.isfinite(x), x, 0.0)
    x = np.maximum(x, 0.0)
    s = float(x.sum())
    if s < 1e-300:
        return None
    return x / s


def score_normalized(x, n):
    """Compute 2*n*max(f*f) assuming sum(x)=1."""
    c = np.convolve(x, x)
    return float(2.0 * n * np.max(c))


def score_raw(x):
    """Compute full score 2*n*max(f*f)/(sum(f))^2."""
    n = len(x)
    s = float(np.sum(x))
    if s < 1e-300:
        return float('inf')
    c = np.convolve(x, x)
    return float(2.0 * n * np.max(c) / (s * s))


def coord_descent(f_init, n, time_budget=60.0, rng=None, verbose=False,
                  max_coord=2.0):
    """
    Coordinate descent minimizing max(f*f) with f>=0.
    Normalizes after each sweep. Returns (best_f_normalized, best_score).
    """
    if rng is None:
        rng = np.random.default_rng()

    f = np.asarray(f_init, dtype=np.float64).copy()
    s = f.sum()
    if s < 1e-300:
        f = np.ones(n) / n
        s = 1.0
    f /= s
    c = np.convolve(f, f)  # length 2n-1

    best_f = f.copy()
    best_score = 2.0 * n * float(np.max(c))
    start_time = time.monotonic()
    sweep = 0
    no_improve = 0

    while time.monotonic() - start_time < time_budget:
        order = rng.permutation(n)
        sweep_start_score = best_score

        for i in order:
            v = f[i]
            k_lo = i
            k_hi = min(i + n - 1, 2 * n - 2)
            ks = np.arange(k_lo, k_hi + 1)
            mask = ks != 2 * i
            ks_lin = ks[mask]

            if len(ks_lin) > 0:
                slopes = 2.0 * f[ks_lin - i]
                intercepts = c[ks_lin] - 2.0 * v * f[ks_lin - i]
            else:
                slopes = np.empty(0)
                intercepts = np.empty(0)

            quad_intercept = c[2 * i] - v * v

            # Constant max for k outside [k_lo, k_hi]
            if k_lo > 0:
                left_max = float(np.max(c[:k_lo]))
            else:
                left_max = -np.inf
            if k_hi < 2 * n - 2:
                right_max = float(np.max(c[k_hi + 1:]))
            else:
                right_max = -np.inf
            const_max = max(left_max, right_max)

            # Ternary search
            lo = 0.0
            hi_bound = max(v * 3.0, max_coord)

            for _ in range(45):
                m1 = lo + (hi_bound - lo) / 3.0
                m2 = hi_bound - (hi_bound - lo) / 3.0
                q1 = m1 * m1 + quad_intercept
                l1 = float(np.max(slopes * m1 + intercepts)) if len(slopes) > 0 else -np.inf
                p1 = max(q1, l1, const_max)
                q2 = m2 * m2 + quad_intercept
                l2 = float(np.max(slopes * m2 + intercepts)) if len(slopes) > 0 else -np.inf
                p2 = max(q2, l2, const_max)
                if p1 < p2:
                    hi_bound = m2
                else:
                    lo = m1

            x_opt = (lo + hi_bound) / 2.0
            delta_v = x_opt - v
            if abs(delta_v) < 1e-14:
                continue

            f[i] = x_opt
            c[2 * i] += x_opt * x_opt - v * v
            if len(ks_lin) > 0:
                c[ks_lin] += 2.0 * delta_v * f[ks_lin - i]

        # End of sweep: normalize
        s = float(f.sum())
        if s < 1e-300:
            break
        f /= s
        c = np.convolve(f, f)
        sc = 2.0 * n * float(np.max(c))
        if sc < best_score - 1e-12:
            best_score = sc
            best_f = f.copy()
            no_improve = 0
        else:
            no_improve += 1
            if no_improve >= 15:
                # Random kick
                kick_size = max(1, n // 16)
                kick_idx = rng.choice(n, size=kick_size, replace=False)
                f[kick_idx] += rng.uniform(0, 0.5, size=kick_size)
                s2 = f.sum()
                if s2 > 1e-300:
                    f /= s2
                    c = np.convolve(f, f)
                no_improve = 0

        sweep += 1
        if verbose and sweep % 20 == 0:
            print(f"      CD sweep {sweep}: best={best_score:.6f}, "
                  f"t={time.monotonic()-start_time:.1f}s", flush=True)

    return best_f, best_score


def smooth_max_score(f, tau):
    """Compute smoothed score and gradient for L-BFGS."""
    n = len(f)
    c = np.convolve(f, f)  # length 2n-1
    s = float(np.sum(f))
    if s < 1e-300:
        return float('inf'), np.zeros(n)

    # Log-sum-exp of c with temperature tau
    c_max = float(np.max(c))
    c_shifted = (c - c_max) / tau
    exp_c = np.exp(c_shifted)
    sum_exp = float(np.sum(exp_c))
    if sum_exp < 1e-300:
        return float('inf'), np.zeros(n)
    lse = c_max + tau * np.log(sum_exp)
    weights = exp_c / sum_exp  # softmax weights

    # Gradient of LSE w.r.t. f:
    # dLSE/df[i] = sum_k weights[k] * d(c[k])/df[i]
    # d(c[k])/df[i] = 2*f[k-i] if 0 <= k-i <= n-1
    # = 2 * (cross-correlation of weights and f at position i)
    grad_lse = 2.0 * np.correlate(weights, f, mode='valid')  # length n

    # J = lse / s^2
    # dJ/df = (grad_lse * s^2 - lse * 2s) / s^4 = (grad_lse * s - 2*lse) / s^3
    score = lse / (s * s)
    grad = (grad_lse * s - 2.0 * lse) / (s * s * s)
    return score, grad


def lbfgs_smooth(f_init, n, time_budget=120.0, verbose=False):
    """
    L-BFGS-B optimization using smooth max approximation.
    Starts with large tau and anneals it down.
    """
    f = np.asarray(f_init, dtype=np.float64).copy()
    s = f.sum()
    if s < 1e-300:
        return f, 2.0
    f /= s

    best_f = f.copy()
    best_score = score_raw(f)
    start_time = time.monotonic()

    # Anneal tau from large to small
    tau_schedule = [1.0, 0.5, 0.2, 0.1, 0.05, 0.02, 0.01, 0.005, 0.002]
    tau_fractions = np.linspace(0, 1, len(tau_schedule))

    elapsed_max = time_budget
    for idx, tau in enumerate(tau_schedule):
        # Allocate time proportional to fraction
        frac_end = tau_fractions[idx + 1] if idx + 1 < len(tau_fractions) else 1.0
        frac_start = tau_fractions[idx]
        tau_time = elapsed_max * (frac_end - frac_start)
        tau_time = max(tau_time, 5.0)
        tau_deadline = time.monotonic() + tau_time

        def fun_grad(x):
            sc, gr = smooth_max_score(x, tau)
            if not np.isfinite(sc):
                return 1e10, np.zeros(n)
            # Clip gradient for stability
            gr = np.clip(gr, -1e6, 1e6)
            return sc, gr

        try:
            res = minimize(fun_grad, f, method='L-BFGS-B',
                           jac=True, bounds=[(0.0, None)] * n,
                           options={'maxiter': 200, 'maxfun': 500,
                                   'gtol': 1e-8, 'iprint': -1})
            if res.success or res.fun < best_score:
                f_try = res.x
                f_try = np.maximum(f_try, 0.0)
                s_try = f_try.sum()
                if s_try > 1e-300:
                    f_try /= s_try
                    sc_try = score_raw(f_try)
                    if np.isfinite(sc_try) and sc_try < best_score:
                        best_score = sc_try
                        best_f = f_try.copy()
                        f = f_try.copy()
        except Exception as e:
            if verbose:
                print(f"      L-BFGS error at tau={tau}: {e}", flush=True)
            break

        if time.monotonic() > tau_deadline:
            break

    # Final: do a few coord descent sweeps from the L-BFGS result
    rem = time_budget - (time.monotonic() - start_time)
    if rem > 5.0:
        best_f, best_score = coord_descent(best_f, n, time_budget=rem * 0.7,
                                           rng=np.random.default_rng(123),
                                           verbose=False)

    return best_f, best_score


def try_length(n, time_budget, rng, best_f_global, given_f, verbose=True):
    """Try optimizing at a specific length n, return (best_f, best_score)."""
    local_best_f = None
    local_best_score = float('inf')

    # Generate diverse starting points
    starts = []

    # Start 1: upsample global best
    if best_f_global is not None:
        m = best_f_global.size
        old = np.linspace(0, 1, m)
        new = np.linspace(0, 1, n)
        up = np.interp(new, old, best_f_global)
        up = np.maximum(up, 0)
        starts.append(up)

    # Start 2: given sequence (if different length, resample)
    if given_f is not None:
        m = given_f.size
        if m != n:
            old = np.linspace(0, 1, m)
            new = np.linspace(0, 1, n)
            gf = np.interp(new, old, given_f)
        else:
            gf = given_f.copy()
        gf = np.maximum(gf, 0)
        starts.append(gf)

    # Start 3: uniform with noise
    uni = 1.0 + rng.normal(0, 0.3, size=n)
    starts.append(np.maximum(uni, 0))

    # Start 4: random Dirichlet
    d = rng.dirichlet(np.ones(n) * 0.5)
    starts.append(d)

    # Start 5: sinusoidal
    x = np.linspace(0, 2 * np.pi * 3, n)
    sinf = np.maximum(np.sin(x) ** 2 + 0.1, 0)
    starts.append(sinf)

    # Deduplicate and normalize
    unique_starts = []
    for st in starts:
        st = np.asarray(st, dtype=np.float64)
        st = np.where(np.isfinite(st), st, 0.0)
        st = np.maximum(st, 0.0)
        s = st.sum()
        if s < 1e-10:
            continue
        st /= s
        if len(unique_starts) == 0:
            unique_starts.append(st)
        else:
            # Check similarity
            if min(np.linalg.norm(st - u) for u in unique_starts) > 0.01:
                unique_starts.append(st)

    if not unique_starts:
        unique_starts = [np.ones(n) / n]

    # Allocate time: 60% coord descent, 40% L-BFGS
    cd_budget = time_budget * 0.6
    lbfgs_budget = time_budget * 0.4
    n_starts = len(unique_starts)
    cd_per_start = cd_budget / n_starts
    lbfgs_per_start = lbfgs_budget / max(1, n_starts - 1)  # skip uniform for L-BFGS

    for idx, init in enumerate(unique_starts):
        rem = time_budget - 0  # we track externally
        # Coordinate descent
        f_cd, s_cd = coord_descent(init, n, time_budget=cd_per_start,
                                   rng=rng, verbose=verbose)
        if s_cd < local_best_score:
            local_best_score = s_cd
            local_best_f = f_cd
            if verbose:
                print(f"    n={n} start{idx} CD: {s_cd:.6f}", flush=True)

        # L-BFGS from best so far (skip for uniform start to save time)
        if idx == 0 and lbfgs_budget > 10:
            f_lb, s_lb = lbfgs_smooth(local_best_f, n, time_budget=lbfgs_per_start,
                                      verbose=verbose)
            if s_lb < local_best_score:
                local_best_score = s_lb
                local_best_f = f_lb
                if verbose:
                    print(f"    n={n} L-BFGS: {s_lb:.6f}", flush=True)

    return local_best_f, local_best_score

            kicked[kick_idx] += rng.uniform(0, 0.3, size=kick_size)
            # Also try a smooth perturbation
            pert = global_best_f.copy()
            noise = np.cumsum(rng.normal(0, 0.01, size=n_best))
            noise -= noise.mean()
            pert = np.maximum(pert + noise, 0)
            s_pert = pert.sum()
            if s_pert > 1e-10:
                pert /= s_pert

            f1, s1 = coord_descent(kicked, n_best, time_budget=rem * 0.4,
                                   rng=rng, verbose=False)
            if s1 < global_best_score:
                global_best_score = s1
                global_best_f = f1
                print(f"    Final CD round {round_idx} (kick): {s1:.6f}", flush=True)

            rem = deadline - time.monotonic()
            if rem < 10.0:
                break
            f2, s2 = coord_descent(pert, n_best, time_budget=rem * 0.4,
                                   rng=rng, verbose=False)
            if s2 < global_best_score:
                global_best_score = s2
                global_best_f = f2
                print(f"    Final CD round {round_idx} (perturb): {s2:.6f}",
                      flush=True)

    # Final: convert to list and evaluate
    seq = [float(max(0.0, x)) for x in global_best_f]

    # Verify with evaluate_sequence
    try:
        final_score = float(evaluate_sequence(seq))
    except Exception:
        final_score = float('inf')

    if not np.isfinite(final_score):
        # Fallback to given sequence
        seq = [float(max(0.0, x)) for x in given_f]
        try:
            final_score = float(evaluate_sequence(seq))
        except Exception:
            seq = [1.0 / 1024.0] * 1024
            final_score = 2.0

    print(f"\nFINAL: n={len(seq)}, local_best={global_best_score:.6f}, "
          f"evaluate_sequence={final_score:.6f}, "
          f"time={time.monotonic()-start_time:.1f}s", flush=True)

    return seq
```
