But wait, this double-counts when j = k-j, i.e., when 2j = k, i.e., j = k/2. In that case, the single term a[j]*a[k-j] = a[j]^2 has derivative 2*a[j] with respect to a[j], not a[j] + a[j] = 2*a[j]. So actually the formula 2*a[k-i] * [0<=k-i<n] is correct!

Let me verify: C[k] = sum_j a[j]*a[k-j]. If k is even, the term j=k/2 gives a[k/2]^2, whose derivative w.r.t. a[k/2] is 2*a[k/2]. The formula gives: for i=k/2, dC[k]/da[k/2] = 2*a[k-(k/2)] = 2*a[k/2]. Correct.

For i != k/2: the terms involving a[i] are j=i (giving a[i]*a[k-i]) and j=k-i (giving a[k-i]*a[i]). Both contribute a[k-i] to the derivative. So total is 2*a[k-i]. Correct.

So: dC[k]/da[i] = 2*a[k-i] if 0 <= k-i < n, else 0.

This means: for a fixed i, dC[k]/da[i] = 2*a[k-i] for k in [i, i+n-1] (where k-i ranges from 0 to n-1).

So the gradient of the log-sum-exp approximation:
dL/da[i] = tau * sum_{k=i}^{i+n-1} w[k] * 2*a[k-i] / Z

where w[k] = exp(C[k]/tau) and Z = sum_k w[k].

This can be computed efficiently! For each i, we need a weighted sum of a[0], a[1], ..., a[n-1] with weights 2*w[i], 2*w[i+1], ..., 2*w[i+n-1].

This is a correlation/convolution operation, which can be done with FFT.

This is a cross-correlation of a with w (where w is the length 2n-1 vector of exp(C[k]/tau)).

def compute_score_normalized(a):
    """Compute score for a normalized sequence (sum=1). Returns 2*n*max(conv)."""
    n = len(a)
    conv = np.convolve(a, a)
    return 2.0 * n * conv.max()

def compute_convffta(a):
    """Compute full convolution of a with itself using FFT."""
    n = len(a)
    size = 2 * n
    fa = np.fft.rfft(a, size=size)
    conv = np.fft.irfft(fa * fa, size=size)[:2*n-1]
    return conv

def compute_score_ffta(a):
    """Compute score using FFT-based convolution."""
    n = len(a)
    conv = compute_convffta(a)
    return 2.0 * n * conv.max()

def smooth_objective(a, tau):

    n = len(a)
    conv = compute_convffta(a)
    # log-sum-exp
    max_c = conv.max()
    lse = max_c + tau * np.log(np.sum(np.exp((conv - max_c) / tau)))
    return 2.0 * n * lse

    if rng is None:
        rng = np.random.default_rng(42)
    
    a = a.astype(np.float64).copy()
    S = a.sum()
    if S < 1e-10:
        return a, np.inf
    a = a / S
    
    best_a = a.copy()
    best_score = compute_score_ffta(a)
    
    for step in range(n_steps):
        # Adaptive tau
        progress = step / max(1, n_steps - 1)
        tau = tau_start * (tau_end / tau_start) ** progress
        
        # Compute gradient
        grad = smooth_gradient(a, tau)
        
        # Gradient descent step
        a_new = a - lr * grad
        a_new = np.clip(a_new, 0, None)
        S_new = a_new.sum()
        if S_new < 1e-10:
            continue
        a_new = a_new / S_new
        
        # Compute actual score
        score = compute_score_ffta(a_new)
        
        if score < best_score:
            best_a = a_new.copy()
            best_score = score
            a = a_new
        else:

            a_new2 = a - 0.5 * lr * grad
            a_new2 = np.clip(a_new2, 0, None)
            S_new2 = a_new2.sum()
            if S_new2 > 1e-10:

                    best_a = a_new2.copy()
                    best_score = score2
                    a = a_new2

    return best_a, best_score

def build_conv_matrix(f):

        j_end = min(n, k + 1)

    f = f.astype(np.float64).copy()

def cross_correlate_valid(w, a):
    """Compute sum_j a[j]*w[i+j] for i=0,...,len(a)-1 using FFT."""
    n = len(a)
    size = 2 * n  # FFT size
    fa = np.fft.rfft(a, size=size)
    fw = np.fft.rfft(w, size=size)
    # cross-correlation: ifft(conj(fa) * fw)
    result = np.fft.irfft(np.conj(fa) * fw, size=size)

Hybrid optimization for minimizing 2*n*max(convolve(a,a))/sum(a)^2.
Uses FFT-based smoothed gradient descent (log-sum-exp approximation),
LP direction refinement, and diverse local search across multiple sequence lengths.

import numpy as np
from scipy.optimize import linprog
import time


def compute_conv_ffta(a):
    """Compute full self-convolution using FFT."""
    n = len(a)
    size = 2 * n
    fa = np.fft.rfft(a, size=size)
    conv = np.fft.irfft(fa * fa, size=size)[:2 * n - 1]
    return conv


def compute_score_ffta(a):
    """Compute score = 2*n*max(conv)/sum^2 for normalized a (sum=1)."""
    n = len(a)
    conv = compute_conv_ffta(a)
    return 2.0 * n * conv.max()


def compute_score_direct(a):
    """Compute score using direct convolution (for verification)."""
    n = len(a)
    conv = np.convolve(a, a)
    return 2.0 * n * conv.max()


def smooth_gradient(a, tau):
    """Compute gradient of log-sum-exp smoothed max(conv) * 2n."""
    n = len(a)
    conv = compute_conv_ffta(a)
    max_c = conv.max()
    w = np.exp((conv - max_c) / tau)
    Z = w.sum()
    # g[i] = sum_j a[j] * w[i+j], computed via cross-correlation
    g = np.correlate(w, a, mode='valid')
    grad = 2.0 * n * tau * g / Z
    return grad

def build_conv_matrix(f):
    """Build matrix A such that A @ b = np.convolve(f, b, 'full')."""
    n = len(f)
    A = np.zeros((2 * n - 1, n))
    for k in range(2 * n - 1):
        lo = max(0, k - n + 1)
        hi = min(n, k + 1)
        for j in range(lo, hi):
            A[k, j] = f[k - j]
    return A

        n_iter = int(min(remaining * 2000, 80000))
        n_iter = max(5000, n_iter)
        a_final, s_final = local_search(best_overall, rng, n_iter=n_iter)
        print(f"  Final local: {s_final:.6f}")

        if s_final < best_score_overall:
            best_score_overall = s_final
            best_overall = a_final.copy()

        # Try a short LP pass if time permits
        rem2 = deadline - time.monotonic()
        if rem2 > 15 and len(best_overall) <= 1024:
            a_lp, s_lp = optimize_lp(best_overall, rem2 - 5, t0=0.3, verbose=False, rng=rng)
            print(f"  Final LP: {s_lp:.6f}")
            if s_lp < best_score_overall:
                best_score_overall = s_lp
                best_overall = a_lp.copy()

    # Verify and return
    if best_overall is not None:
        best_overall = np.clip(best_overall, 0, 1000)
        final_list = best_overall.tolist()
        final_score = evaluate_sequence(final_list)
        print(f"\nFinal verified score: {final_score:.6f}")
        print(f"Sequence length: {len(final_list)}")
        return final_list

    print("Fallback")
    return [1.0] * 256
