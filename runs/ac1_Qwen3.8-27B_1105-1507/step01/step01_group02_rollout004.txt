import numpy as np
import time

def project_simplex(v):
    """Project vector v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cv = np.cumsum(u) - 1.0
    rho_candidates = np.where(u * np.arange(1, n + 1) > cv)[0]
    if len(rho_candidates) == 0:
        return np.ones(n) / n
    rho = rho_candidates[-1]
    theta = cv[rho] / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def fast_convolve(a):
    """Compute full linear convolution of a with itself using FFT."""
    n = len(a)
    size = 2 * n - 1
    N = 1 << (size - 1).bit_length()
    fa = np.fft.rfft(a, N)
    fc = fa * fa
    result = np.fft.irfft(fc, N)[:size]
    result[result < 0] = 0.0
    return result

def logsumexp_convolve(a, T):
    """
    Compute log-sum-exp of convolution values with temperature T.
    Returns (lse_value, gradient).
    """
    n = len(a)
    c = fast_convolve(a)
    # Scale for numerical stability
    c_max = np.max(c)
    scaled = T * (c - c_max)
    # Clip for numerical stability
    scaled = np.clip(scaled, -100, 100)
    exp_scaled = np.exp(scaled)
    lse = c_max + np.log(np.sum(exp_scaled)) / T
    
    # Gradient: (1/T) * sum_k exp(T*c_k) * grad(c_k) / sum_k exp(T*c_k)
    weights = exp_scaled / np.sum(exp_scaled)
    # Only use top peaks for efficiency
    top_k = max(1, int(0.05 * len(c)))
    top_idx = np.argsort(weights)[::-1][:top_k]
    weights = np.zeros(len(c))
    weights[top_idx] = np.exp(T * (c[top_idx] - c_max))
    weights /= np.sum(weights)
    
    grad = np.zeros(n)
    for k in top_idx:
        w = weights[k]
        if w < 1e-10:

        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        if lo <= hi:

    return lse, grad

def peak_deflation(a, n_steps=50, rng=None):

        # To reduce this, we want to decrease a[i] where the product is large

        # Simple approach: reduce the largest contributor and add to a random other position

            j = peak_idx - i

        a[max_contrib_idx] -= transfer
        a[min_contrib_idx] += transfer

def alt_projection_init(n, rng, n_iter=100):
    """
    Alternating projection to get a nonnegative sequence with near-flat
    magnitude spectrum.
    """
    N = 1 << (2 * n - 2).bit_length()
    phases = rng.uniform(0.0, 2.0 * np.pi, N // 2 + 1)
    F = np.exp(1j * phases)
    F[0] = 1.0 + 0j

    a = np.ones(n) / n
    for it in range(n_iter):
        a_full = np.fft.irfft(F, N)
        a = a_full[:n].copy()
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s < 1e-10:
            a = np.ones(n) / n
        else:
            a = a / s
        F_full = np.fft.rfft(a, N)
        mag_target = np.ones(len(F_full))
        # Flat spectrum target
        F = mag_target * np.exp(1j * np.angle(F_full))
        F[0] = F[0].real + 0j

    return a

"""
Optimizes a nonnegative sequence to minimize 2*n*max(convolve(a,a))/sum(a)^2.
Uses FISTA with log-sum-exp smoothing, adaptive temperature, peak deflation,
and multi-phase optimization starting from the parent sequence.

import numpy as np
import time


def project_simplex(v):
    """Project vector v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cv = np.cumsum(u) - 1.0
    rho_candidates = np.where(u * np.arange(1, n + 1) > cv)[0]
    if len(rho_candidates) == 0:
        return np.ones(n) / n
    rho = rho_candidates[-1]
    theta = cv[rho] / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def fast_convolve(a):
    """Compute full linear convolution of a with itself using FFT."""
    n = len(a)
    size = 2 * n - 1
    N = 1 << (size - 1).bit_length()
    fa = np.fft.rfft(a, N)
    fc = fa * fa
    result = np.fft.irfft(fc, N)[:size]
    result[result < 0] = 0.0
    return result


def compute_conv_gradient(a, c, top_k=20):
    """
    Compute gradient of max(convolve(a,a)) using top-k active peaks.
    """
    n = len(a)
    peak = np.max(c)
    threshold = peak * (1.0 - 1e-4)
    peak_indices = np.where(c >= threshold)[0]
    if len(peak_indices) > top_k:
        scores = c[peak_indices]
        top_idx = np.argsort(scores)[::-1][:top_k]
        peak_indices = peak_indices[top_idx]

    # Use soft weights based on how close to peak
    weights = c[peak_indices] / np.sum(c[peak_indices])

    grad = np.zeros(n)
    for k, w in zip(peak_indices, weights):
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        if lo <= hi:
            seg = a[k - hi:k - lo + 1][::-1]
            grad[lo:hi + 1] += 2.0 * w * seg

    return grad


def lse_convolve(a, T, top_k=30):
    """
    Compute log-sum-exp of convolution values and its gradient.
    Returns (lse_value, gradient).
    """
    n = len(a)
    c = fast_convolve(a)
    c_max = np.max(c)

    # Numerically stable log-sum-exp
    scaled = T * (c - c_max)
    scaled = np.clip(scaled, -200, 200)
    exp_scaled = np.exp(scaled)
    lse = c_max + np.log(np.sum(exp_scaled)) / T

    # Softmax weights (concentrated on peaks for large T)
    weights = exp_scaled / np.sum(exp_scaled)

    # Only use top contributing peaks
    top_idx = np.argsort(weights)[::-1][:top_k]
    w_sum = weights[top_idx].sum()
    if w_sum < 1e-15:
        top_idx = np.array([np.argmax(c)])
        weights = np.zeros(len(c))
        weights[top_idx] = 1.0
        w_sum = 1.0

    grad = np.zeros(n)
    for k in top_idx:
        w = weights[k] / w_sum
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        if lo <= hi:
            seg = a[k - hi:k - lo + 1][::-1]
            grad[lo:hi + 1] += 2.0 * w * seg

    return lse, grad


def fista_smooth(a, deadline, rng, T_init=2.0, T_end=100.0, max_iters=100000):
    """
    FISTA optimization using log-sum-exp smoothed objective.
    Adapts temperature from T_init to T_end over the course of optimization.
    """
    n = len(a)
    a = project_simplex(a.copy().astype(np.float64))

    c = fast_convolve(a)
    best_peak = np.max(c)
    best_a = a.copy()

    y = a.copy()
    t_k = 1.0
    step = 0.05 / max(1, n ** 0.5)

    it = 0
    last_improve = 0
    patience = 3000
    stall_lr = 1.0

    while time.time() < deadline and it < max_iters:
        it += 1

        # Adaptive temperature: logarithmic schedule
        progress = min(1.0, it / (max_iters * 0.7))
        T = T_init * (T_end / T_init) ** progress

        # Compute gradient at y using LSE
        _, grad = lse_convolve(y, T, top_k=max(10, min(40, n // 10)))

        # Adaptive step size with backtracking
        # Estimate gradient norm for step sizing
        g_norm = np.linalg.norm(grad)
        if g_norm > 1e-12:
            step = min(0.5 / g_norm, 1.0) * stall_lr
        else:
            step = 0.1 * stall_lr

        # FISTA update
        x_new = project_simplex(y - step * grad)

        # Evaluate actual peak
        c_x = fast_convolve(x_new)
        peak_x = np.max(c_x)

        if peak_x < best_peak * (1.0 - 1e-9):
            best_peak = peak_x
            best_a = x_new.copy()
            last_improve = it
            # Slightly increase step on improvement

        # Momentum update
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t_k * t_k)) / 2.0
        momentum = (t_k - 1.0) / t_new
        # Clip momentum to avoid oscillation

        y = x_new + momentum * (x_new - a)
        a = x_new
        t_k = t_new

        # Patience: reduce step and add perturbation

                    a = project_simplex(a + perturb)
                    y = a.copy()
                    t_k = 1.0
                    last_improve = it

def peak_deflation(a, n_steps=100, rng=None):
    """
    Directly reduce peak convolution values by redistributing mass
    from high-contribution positions to low-contribution ones.
    """
    n = len(a)
    a = a.copy()

    for step in range(n_steps):
        c = fast_convolve(a)
        peak = np.max(c)

        # Get top few peaks
        top_peaks = np.argsort(c)[::-1][:5]

        for pk in top_peaks:
            lo = max(0, pk - n + 1)
            hi = min(n - 1, pk)

            # Compute contribution of each a[i] to this peak
            contrib = np.zeros(n)
            for i in range(lo, hi + 1):
                j = pk - i
                if 0 <= j < n:
                    val = a[i] * a[j]
                    contrib[i] += val
                    if j != i:
                        contrib[j] += val

            # Find highest and lowest contribution indices
            max_i = np.argmax(contrib)
            # Find an index with zero contribution
            zero_mask = contrib < 1e-12
            if np.any(zero_mask):
                min_i = np.where(zero_mask)[0][0]
            else:
                min_i = np.argmin(contrib)

            if max_i == min_i:
                continue

            # Transfer mass
            amount = 0.005 * a[max_i]
            if amount < 1e-12:
                continue
            a[max_i] -= amount
            a[min_i] += amount

        a = np.maximum(a, 0.0)
        total = a.sum()
        if total < 1e-10:
            break
        a = a / total

    return a


def alt_projection_init(n, rng, n_iter=100):
    """
    Alternating projection initialization for flat-spectrum nonnegative sequences.
    """
    N = 1 << (2 * n - 2).bit_length()
    phases = rng.uniform(0.0, 2.0 * np.pi, N // 2 + 1)
    F = np.exp(1j * phases)
    F[0] = 1.0 + 0j

    a = np.ones(n) / n
    for it in range(n_iter):
        a_full = np.fft.irfft(F, N)
        a = a_full[:n].copy()
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s < 1e-10:
            a = np.ones(n) / n
        else:
            a = a / s
        F_full = np.fft.rfft(a, N)
        # Target: flat magnitude spectrum
        mag = np.ones(len(F_full))
        F = mag * np.exp(1j * np.angle(F_full))
        F[0] = F[0].real + 0j

    return a


def granville_init(n, rng, n_iter=200):
    """
    Granville-style initialization: iteratively enforce non-negativity
    and flat power spectrum.
    """
    a = np.ones(n) / n
    for it in range(n_iter):
        # FFT
        F = np.fft.rfft(a, 2 * n - 1)
        # Enforce flat magnitude (with slight low-freq boost for non-negativity)
        freq = np.arange(len(F))
        weight = 1.0 + 0.1 * np.exp(-freq / (n * 0.1))
        F_flat = weight * np.exp(1j * np.angle(F))
        # IFFT
        a_full = np.fft.irfft(F_flat, 2 * n - 1)
        a = a_full[:n]
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s < 1e-10:
            a = np.ones(n) / n
        else:
            a = a / s

    return a

        (512, 'random'),
        (640, 'alt_proj'),
        (512, 'bimodal'),
        (384, 'alt_proj'),
        (1024, 'granville'),
    ]

    remaining_time = overall_deadline - time.time()
    time_per = remaining_time / len(configs)

    for i, (n, init_type) in enumerate(configs):
        if time.time() > overall_deadline - 20:
            break
        deadline_c = min(overall_deadline - 10, time.time() + time_per)
        if deadline_c < time.time() + 10:
            continue

        print(f"Phase 2 [{i+1}/{len(configs)}]: n={n}, init={init_type}")
        try:
            a_opt, score = try_configuration(n, init_type, rng, deadline_c, parent_seq)
            print(f"  Result: score={score:.6f}, time_left={overall_deadline - time.time():.0f}s")
            if np.isfinite(score) and score < best_score:
                best_score = score
                best_seq = a_opt.copy()
                print(f"  >>> New best: {best_score:.6f} (n={n}, init={init_type})")
        except Exception as e:
            print(f"  Error: {e}")

    # Phase 3: Final intensive polish on best sequence
    if best_seq is not None and time.time() < overall_deadline - 15:
        polish_deadline = overall_deadline
        print(f"Phase 3: Final polish, time_left={overall_deadline - time.time():.0f}s")
        # Use subgradient momentum for final refinement
        a_final = subgradient_momentum(best_seq, polish_deadline, rng, max_iters=1000000)
        # Peak deflation
        a_final = peak_deflation(a_final, n_steps=100, rng=rng)
        score_final = evaluate_sequence(a_final.tolist())
        print(f"  Final polish score: {score_final:.6f}")
        if np.isfinite(score_final) and score_final < best_score:
            best_score = score_final
            best_seq = a_final
            print(f"  >>> New best: {best_score:.6f} (after polish)")

    # Fallback
    if best_seq is None:
        print("Falling back to uniform sequence")
        best_seq = np.ones(256) / 256

    result = [float(x) for x in best_seq]
    final_score = evaluate_sequence(result)
    print(f"FINAL: score={final_score:.6f}, length={len(result)}")
    return result
