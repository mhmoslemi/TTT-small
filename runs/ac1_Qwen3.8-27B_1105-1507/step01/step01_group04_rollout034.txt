```python
import numpy as np
import time
from scipy.optimize import linprog
from scipy.sparse import coo_matrix

def fft_convolve(f):
    n = len(f)
    size = 1
    while size < 2*n - 1:
        size <<= 1
    F = np.fft.rfft(f, size)
    h = np.fft.irfft(F * F, size)
    return h[:2*n-1]

def score(f):
    n = len(f)
    S = f.sum()
    if S < 0.01:
        return np.inf
    h = fft_convolve(f)
    return 2.0 * n * h.max() / (S * S)

def normalize(f):
    f = np.maximum(f, 0)
    S = f.sum()
    if S < 1e-30:
        return None
    return f / S

def multiplicative_update(f, n_iter=5000, eta0=1.0, threshold=0.7):
    n = len(f)
    f = normalize(f)
    best_f = f.copy()
    best_score = score(f)
    
    for it in range(n_iter):
        h = fft_convolve(f)
        M = h.max()
        if M < 1e-30:
            break
        active = np.where(h >= threshold * M)[0]
        
        # Compute pressure
        p = np.zeros(n)
        for k in active:
            lo = max(0, k - n + 1)
            hi = min(n - 1, k)
            idx = k - np.arange(lo, hi + 1)
            p[lo:hi+1] += 2.0 * f[idx]
        p /= len(active)
        
        eta = eta0 / (M * n)  # Scale by n to be size-independent
        f = f * np.exp(-eta * p)
        f = normalize(f)
        if f is None:
            break
        
        s = score(f)
        if s < best_score:
            best_score = s
            best_f = f.copy()
        
        if (it+1) % 1000 == 0:
            print(f"      MU it={it+1}: score={best_score:.6f}")
    
    return best_f, best_score

def lp_frank_wolfe(f, time_budget=60, max_iters=50):
    """LP-based Frank-Wolfe for min max_k (f*f)(k) s.t. f>=0, sum f=1."""
    n = len(f)
    f = normalize(f)
    if f is None:
        return f, np.inf
    best_f = f.copy()
    best_score = score(f)
    t0 = time.monotonic()
    
    for it in range(max_iters):
        if time.monotonic() - t0 > time_budget:
            break
        
        h = np.convolve(f, f)  # Use direct for accuracy
        M = h.max()
        
        # Build constraint matrix for LP:
        # Maximize sum y_j
        # Subject to: sum_j f(k-j) * y_j <= M for each k
        #              y_j >= 0
        
        # The constraint matrix A has shape (2n-1, n)
        # A[k, j] = f(k-j) if 0 <= k-j < n, else 0
        # This is a Toeplitz-like matrix (actually it's the convolution matrix)
        
        # Build sparse matrix
        rows = []
        cols = []
        vals = []
        for k in range(2*n - 1):
            lo = max(0, k - n + 1)
            hi = min(n - 1, k)
            for j in range(lo, hi+1):
                rows.append(k)
                cols.append(j)
                vals.append(f[k-j])
        
        A = coo_matrix((vals, (rows, cols)), shape=(2*n-1, n))
        
        # Solve: max 1^T y s.t. A y <= M, y >= 0
        res = linprog(-np.ones(n), A_ub=A, b_ub=np.full(2*n-1, M),
                     bounds=[(0, None)]*n, method='highs')
        
        if not res.success:
            break
        
        y = res.x
        Sy = y.sum()
        if Sy < 1e-15:
            break
        y = y / Sy
        
        # FW update
        t = 2.0 / (it + 2)
        f_new = (1 - t) * f + t * y
        f_new = normalize(f_new)
        if f_new is None:
            break
        f = f_new
        
        s = score(f)
        if s < best_score:
            best_score = s
            best_f = f.copy()
        
        if (it+1) % 5 == 0:
            print(f"      LP it={it+1}: score={best_score:.6f}, M={M:.8f}")
```

```python
            best_f = f.copy()
        if verbose and (it + 1) % 2000 == 0:
            print(f"      MD it={it+1}: score={best_s:.6f}, M={M:.8f}, |K|={len(active)}")

    return best_f, best_s


def _lp_frank_wolfe(f0, time_budget, max_iters=30, verbose=True):
    f = _norm1(np.asarray(f0, dtype=np.float64))
    if f is None:
        return np.asarray(f0, dtype=np.float64).copy(), float('inf')
    n = len(f)
    best_f = f.copy()
    best_s = _score(f)
    t0 = time.monotonic()

    for it in range(max_iters):
        if time.monotonic() - t0 > time_budget:
            break
        h = np.convolve(f, f)
        M = float(h.max())

        rows = []
        cols = []
        vals = []
        for k in range(2 * n - 1):
            lo = max(0, k - n + 1)
            hi = min(n - 1, k)
            if hi >= lo:
                rows.extend(range(k, k + 1) * (hi - lo + 1))
                cols.extend(range(lo, hi + 1))
                vals.extend(f[k - hi:k - lo + 1].tolist())

        if len(rows) == 0:
            break
        A = coo_matrix((vals, (rows, cols)), shape=(2 * n - 1, n)).tocsr()
        res = linprog(-np.ones(n), A_ub=A, b_ub=np.full(2 * n - 1, M),
                      bounds=[(0, None)] * n, method='highs')
        if not res.success or res.x is None:
            break

        y = res.x
        Sy = float(y.sum())
        if Sy < 1e-15:
            break
        y = y / Sy
        t = 2.0 / (it + 2)
        f = _norm1((1.0 - t) * f + t * y)
        if f is None:
            break
        s = _score(f)
        if s < best_s:
            best_s = s
            best_f = f.copy()
        if verbose:
            print(f"      LP it={it+1}: score={best_s:.6f}, M={M:.8f}, t={t:.4f}")

    return best_f, best_s


def _perturb_restart(f, rng, frac):
    n = len(f)
    noise = rng.random(n) * frac
    return _norm1(f + noise)


def propose_candidate(seed=42, budget_s=1500):
    deadline = time.monotonic() + float(budget_s) - 10.0
    rng = np.random.default_rng(seed)

    # Get parent
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64).copy()
        parent = np.clip(parent, 0.0, 1000.0)
    except Exception:
        parent = None

    best_f = None
    best_score = float('inf')
    if parent is not None and len(parent) > 0:
        ps = float(evaluate_sequence(parent.tolist()))
        if np.isfinite(ps):
            best_f = parent.copy()
            best_score = ps
            print(f"Parent: n={len(parent)}, score={best_score:.6f}")

    # Build candidate pool: (name, n, init_func)
    candidates = []
    if parent is not None:
        pn = len(parent)
        candidates.append(("parent", pn, _norm1(parent)))

    inits = [
        (1024, "cos", _init_cosine),
        (2048, "cos", _init_cosine),
        (4096, "cos", _init_cosine),
        (1024, "powcos2", lambda n: _init_powcos(n, 2.0)),
        (2048, "powcos2", lambda n: _init_powcos(n, 2.0)),
        (2048, "powcos3", lambda n: _init_powcos(n, 3.0)),
        (4096, "powcos2", lambda n: _init_powcos(n, 2.0)),
        (1024, "sine", _init_sine),
        (2048, "sine", _init_sine),
        (2048, "u2", lambda n: _init_u_shape(n, 2.0)),
        (2048, "u4", lambda n: _init_u_shape(n, 4.0)),
        (4096, "u2", lambda n: _init_u_shape(n, 2.0)),
        (2048, "gauss30", lambda n: _init_gaussian(n, 0.30)),
        (2048, "gauss25", lambda n: _init_gaussian(n, 0.25)),
        (2048, "beta22", lambda n: _init_beta(n, 2.0, 2.0)),
        (2048, "beta33", lambda n: _init_beta(n, 3.0, 3.0)),
        (4096, "beta22", lambda n: _init_beta(n, 2.0, 2.0)),
    ]

    for n, name, func in inits:
        try:
            f = func(n)
            if f is not None:
                candidates.append((f"{name}_{n}", n, f))
        except Exception:
            pass

    # Add a random one
    f_rand = _norm1(rng.random(2048))
    candidates.append(("rand_2048", 2048, f_rand))

    print(f"Total candidates: {len(candidates)}")

    # Phase 1: Quick mirror descent on all candidates
    results = []
    time_per = max(6.0, min(30.0, (deadline - time.monotonic()) * 0.45 / max(1, len(candidates))))

    for idx, (name, n, f0) in enumerate(candidates):
        if time.monotonic() > deadline - 120:
            break
        s0 = _score(f0)
        # Try a quick run
        f_md, s_md = _mirror_descent(f0, n_iter=3000, eta_scale=0.08,
                                     thresh=0.7, deadline=deadline - 60)
        # Also try with different threshold
        f_md2, s_md2 = _mirror_descent(f0, n_iter=2000, eta_scale=0.05,
                                       thresh=0.9, deadline=deadline - 60)
        s = min(s_md, s_md2)
        f = f_md if s_md <= s_md2 else f_md2
        results.append((s, name, n, f))
        print(f"  [{idx+1}/{len(candidates)}] {name}: init={s0:.6f} -> {s:.6f}")
        if s < best_score:
            best_score = s
            best_f = f.copy()
            print(f"  ** New best: {best_score:.6f} from {name} **")

    # Sort results
    results.sort(key=lambda x: x[0])

    # Phase 2: Extended optimization on top candidates
    top_k = min(5, len(results))
    for r in range(top_k):
        if time.monotonic() > deadline - 60:
            break
        s_cur, name, n, f_cur = results[r]
        print(f"\n  Refining {name} (n={n}, score={s_cur:.6f})")

        # Extended MD with smaller eta
        remaining = deadline - time.monotonic() - 30
        n_it = int(min(8000, max(2000, remaining * 100)))
        f_ext, s_ext = _mirror_descent(f_cur, n_it, eta_scale=0.03,
                                       thresh=0.85, deadline=deadline - 30, verbose=True)
        if s_ext < best_score:
            best_score = s_ext
            best_f = f_ext.copy()
            print(f"  ** New best after ext MD: {best_score:.6f} **")
        f_cur = f_ext

        # Try different thresholds
        for tf in [0.6, 0.95]:
            if time.monotonic() > deadline - 30:
                break
            f_t, s_t = _mirror_descent(f_cur, 3000, eta_scale=0.04,
                                       thresh=tf, deadline=deadline - 30)
            if s_t < best_score:
                best_score = s_t
                best_f = f_t.copy()
                print(f"  ** New best (thresh={tf}): {best_score:.6f} **")
            f_cur = f_t if s_t < _score(f_cur) else f_cur

        # Perturbation restart
        if time.monotonic() > deadline - 40:
            break
        for pf in [0.02, 0.05, 0.1]:
            if time.monotonic() > deadline - 20:
                break
            f_p = _perturb_restart(f_cur, rng, pf)
            if f_p is None:
                continue
            f_pp, s_pp = _mirror_descent(f_p, 3000, eta_scale=0.05,
                                         thresh=0.8, deadline=deadline - 20)
            if s_pp < best_score:
                best_score = s_pp
                best_f = f_pp.copy()
                print(f"  ** New best (perturb={pf}): {best_score:.6f} **")
            f_cur = f_pp if s_pp < _score(f_cur) else f_cur

    # Phase 3: LP-Frank-Wolfe on best 1-2 candidates (only for feasible n)
    # Re-sort to get current best
    if best_f is not None:
        n_best = len(best_f)
        if n_best <= 3000 and (deadline - time.monotonic()) > 30:
            lp_budget = min(90.0, deadline - time.monotonic() - 10)
            print(f"\n  LP-FW on best (n={n_best}, score={best_score:.6f}, budget={lp_budget:.0f}s)")
            f_lp, s_lp = _lp_frank_wolfe(best_f, time_budget=lp_budget, max_iters=20)
            if s_lp < best_score:
                best_score = s_lp
                best_f = f_lp.copy()
                print(f"  ** New best from LP: {best_score:.6f} **")

        # Also try LP on second best if available
        if len(results) >= 2 and (deadline - time.monotonic()) > 30:
            s2, name2, n2, f2 = results[1]
            if n2 <= 2500 and s2 < best_score + 0.01:
                lp_budget2 = min(60.0, deadline - time.monotonic() - 10)
                print(f"\n  LP-FW on {name2} (n={n2}, score={s2:.6f})")
                f_lp2, s_lp2 = _lp_frank_wolfe(f2, time_budget=lp_budget2, max_iters=15)
                if s_lp2 < best_score:
                    best_score = s_lp2
                    best_f = f_lp2.copy()
                    print(f"  ** New best from LP2: {best_score:.6f} **")

    # Phase 4: Final MD polish on the absolute best
    if best_f is not None and (deadline - time.monotonic()) > 10:
        n_polish = int(min(5000, max(1000, (deadline - time.monotonic() - 2) * 80)))
        f_fin, s_fin = _mirror_descent(best_f, n_polish, eta_scale=0.02,
                                       thresh=0.9, deadline=deadline - 2)
        if s_fin < best_score:
            best_score = s_fin
            best_f = f_fin.copy()
            print(f"  ** Final polish: {best_score:.6f} **")

    # Final verification
    if best_f is None:
        # Fallback: return a known-good sequence
        best_f = _init_cosine(512)
        best_score = _score(best_f)

    final_list = [float(x) for x in best_f]
    final_score = float(evaluate_sequence(final_list))
    print(f"\nFinal: n={len(final_list)}, raw_score={best_score:.6f}, verified={final_score:.6f}")

    if not np.isfinite(final_score) or final_score > 2.5:
        print("  Fallback to cosine 512.")
        fb = _init_cosine(512)
        return [float(x) for x in fb]

    return final_list
```
