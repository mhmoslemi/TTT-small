def solve_lp_step(a, upper_bound=10.0):
    """Solve LP: max sum(g) s.t. conv(a,g)[k] <= M for all k, 0 <= g_i <= ub."""
    n = len(a)
    conv_aa = np.convolve(a, a)
    M = float(np.max(conv_aa))
    A = build_A_lp(a)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, upper_bound)] * n
    try:
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': 30})
        if res.status == 0:
            return res.x, float(-res.fun), M
    except Exception:
        pass
    return None, None, M


def lp_optimize(a, seed=0, max_iter=150, t=0.5, verbose=False):
    """LP-based alternating optimization."""
    rng = np.random.default_rng(seed)
    a = a.copy()
    s = a.sum()
    if s < 1e-10:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    best_a = a.copy()
    best_obj = compute_obj(best_a)

    stall = 0
    for it in range(max_iter):
        g, g_sum, M = solve_lp_step(a)
        if g is None:
            break
        if g_sum <= 1.0 + 1e-5:
            stall += 1
            if stall > 8:
                noise = np.exp(rng.normal(0, 0.4, len(a)))
                a = best_a * noise
                a = np.clip(a, 1e-10, None)
                a = a / a.sum()
                stall = 0
            continue
        stall = 0

        g = g / g.sum()
        a_new = (1.0 - t) * a + t * g
        a_new = np.clip(a_new, 0, None)
        s_new = a_new.sum()
        if s_new < 1e-10:
            break
        a_new = a_new / s_new

        obj_new = compute_obj(a_new)
        if obj_new < best_obj:
            best_a = a_new.copy()
            best_obj = obj_new
            if verbose and (it % 40 == 0 or it < 5):
                print(f"    LP it={it}: obj={best_obj:.6f} gsum={g_sum:.4f}")
        a = a_new

    return best_a, best_obj


def peak_deflation(a, n_iter=200, alpha_start=0.3, verbose=False):
    """Directly reduce convolution peaks using deflation method."""
    a = a.copy()
    s = a.sum()
    if s < 1e-10:
        a = np.ones(len(a)) / len(a)
        s = 1.0
    else:
        a = a / s
    best_a = a.copy()
    best_obj = compute_obj(best_a)
    n = len(a)
    alpha = alpha_start

    for it in range(n_iter):
        c = np.convolve(a, a)
        M = c.max()
        k_star = int(c.argmax())

        # Deflation: reduce a[j] proportionally to a[k_star - j]
        j_start = max(0, k_star - n + 1)
        j_end = min(n - 1, k_star)
        contributions = np.zeros(n)
        for j in range(j_start, j_end + 1):
            contributions[j] = a[j] * a[k_star - j]

        total_contrib = contributions.sum()
        if total_contrib < 1e-15:
            break

        # Reduce proportionally
        reduction = alpha * contributions / total_contrib
        a_new = a - reduction * a
        a_new = np.clip(a_new, 0, None)
        s_new = a_new.sum()
        if s_new < 1e-10:
            break
        a_new = a_new / s_new

        obj_new = compute_obj(a_new)
        if obj_new < best_obj:
            best_a = a_new.copy()
            best_obj = obj_new
            if verbose and it % 30 == 0:
                print(f"    Deflate it={it}: obj={best_obj:.6f} alpha={alpha:.4f}")
        a = a_new
        alpha *= 0.98

    return best_a, best_obj


def multi_peak_deflation(a, n_iter=150, top_k=3, alpha=0.2, verbose=False):
    """Reduce multiple peaks simultaneously."""
    a = a.copy()
    s = a.sum()
    if s < 1e-10:
        a = np.ones(len(a)) / len(a)
        s = 1.0
    else:
        a = a / s
    best_a = a.copy()
    best_obj = compute_obj(best_a)
    n = len(a)

    for it in range(n_iter):
        c = np.convolve(a, a)
        # Find top peaks
        peak_vals = np.sort(c)[::-1][:top_k]
        threshold = peak_vals[-1] * 0.9
        peak_indices = np.where(c >= threshold)[0]

        total_reduction = np.zeros(n)
        for k in peak_indices:
            j_start = max(0, k - n + 1)
            j_end = min(n - 1, k)
            for j in range(j_start, j_end + 1):
                total_reduction[j] += a[j] * a[k - j]

        sr = total_reduction.sum()
        if sr < 1e-15:
            break

        a_new = a - alpha * total_reduction / sr * a
        a_new = np.clip(a_new, 0, None)
        s_new = a_new.sum()
        if s_new < 1e-10:
            break
        a_new = a_new / s_new

        obj_new = compute_obj(a_new)
        if obj_new < best_obj:
            best_a = a_new.copy()
            best_obj = obj_new
            if verbose and it % 30 == 0:
                print(f"    MultiDeflate it={it}: obj={best_obj:.6f}")
        a = a_new

    return best_a, best_obj


def random_refine(a, rng, max_iter=3000, noise_scale=0.12):
    """Random block perturbation local search."""
    best_a = a.copy()
    best_obj = compute_obj(best_a)
    n = len(best_a)

    for it in range(max_iter):
        width = max(1, n // 60)
        start = int(rng.integers(0, n - width + 1))
        candidate = best_a.copy()
        candidate[start:start + width] *= np.exp(rng.normal(0, noise_scale, width))
        candidate = np.clip(candidate, 0, None)
        s = candidate.sum()
        if s < 1e-10:
            continue
        candidate = candidate / s
        obj_new = compute_obj(candidate)
        if obj_new < best_obj:
            best_a = candidate
            best_obj = obj_new

    return best_a, best_obj


def structured_init(n, rng):
    """Generate structured initialization."""
    choice = rng.integers(0, 6)
    if choice == 0:
        phase = rng.uniform(0, 2 * np.pi)
        freq = rng.uniform(0.1, 0.6)
        a = np.maximum(0, np.cos(2 * np.pi * freq * np.arange(n) + phase))
    elif choice == 1:
        a = np.ones(n)
        fraction = rng.uniform(0.3, 0.7)
        cut = int(n * fraction)
        a[cut:] = rng.uniform(0.4, 0.9)
    elif choice == 2:
        period = int(rng.integers(max(2, n // 30), max(3, n // 5)))
        pattern = rng.random(period)
        a = np.tile(pattern, n // period + 1)[:n]
    elif choice == 3:
        r = rng.uniform(0.6, 0.95)
        a = r ** np.arange(n)
    elif choice == 4:
        center = n / 2
        sigma = n * rng.uniform(0.15, 0.4)
        a = np.exp(-0.5 * ((np.arange(n) - center) / sigma) ** 2)
    else:
        a = rng.random(n)

    s = a.sum()
    if s < 1e-10:
        a = rng.random(n)
        s = a.sum()
    return a / s


def refine_sequence(a, seed, deadline, verbose=False):
    """Apply a sequence of refinement methods to a candidate."""
    rng_local = np.random.default_rng(seed)
    best_a = a.copy()
    best_obj = compute_obj(best_a)
    n = len(best_a)

    methods = [
        ("grad", lambda: gradient_optimize(best_a, tau_start=0.03, tau_end=0.0005,
                                           n_schedules=4, max_iter_per_sched=200,
                                           verbose=verbose)),
        ("lp", lambda: lp_optimize(best_a, seed=seed, max_iter=100, t=0.4,
                                   verbose=verbose)),
        ("deflate", lambda: peak_deflation(best_a, n_iter=100, alpha_start=0.25,
                                           verbose=verbose)),
        ("multideflate", lambda: multi_peak_deflation(best_a, n_iter=80, top_k=3,
                                                      alpha=0.15, verbose=verbose)),
        ("grad2", lambda: gradient_optimize(best_a, tau_start=0.01, tau_end=0.0002,
                                            n_schedules=3, max_iter_per_sched=150,
                                            verbose=verbose)),
    ]

    for method_name, method_func in methods:
        if time.monotonic() > deadline:
            break
        try:
            a_new, obj_new = method_func()
            if obj_new < best_obj:
                best_a = a_new.copy()
                best_obj = obj_new
                if verbose:
                    print(f"      {method_name}: obj={best_obj:.6f}")
        except Exception:
            pass

    return best_a, best_obj


def propose_candidate(seed=42, budget_s=1500, **kwargs):
    """
    Hybrid optimization combining smooth gradient-based methods (L-BFGS with
    log-sum-exp tau annealing), LP-based alternating optimization, peak deflation,
    and random local search. Explores multiple sequence lengths and initializations.
    """
    rng = np.random.default_rng(seed)
    start_time = time.monotonic()
    deadline = start_time + float(budget_s) - 5.0

    best_seq = None
    best_score = np.inf

    # Load parent sequence
    parent_a = None
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64)
        parent = np.clip(parent, 0, 1000.0)
        if parent.size > 0 and np.isfinite(parent).all():
            obj = compute_obj(parent)
            if obj < best_score:
                best_score = obj
                best_seq = parent.copy()
                parent_a = parent.copy()
            print(f"Parent (n={parent.size}): obj={obj:.6f}")
    except Exception as e:
        print(f"Parent loading failed: {e}")

    # Phase 1: Refine parent extensively
    if parent_a is not None and time.monotonic() < deadline - 60:
        print(f"\nPhase 1: Refining parent, n={len(parent_a)}")
        a_ref, obj_ref = refine_sequence(parent_a, seed + 1, deadline - 60, verbose=True)
        if obj_ref < best_score:
            best_score = obj_ref
            best_seq = a_ref.copy()
            print(f"  Parent refined: obj={best_score:.6f}")

    # Phase 2: Explore different n values and initializations
    n_values = [20, 30, 40, 50, 70, 100, 150, 200, 300, 400]
    if parent_a is not None:
        # Add parent's n if not already in list
        n_parent = len(parent_a)
        if n_parent not in n_values:
            n_values.append(n_parent)
            n_values.sort()

    print(f"\nPhase 2: Exploring n values, {deadline - time.monotonic():.0f}s remaining")

    for n in n_values:
        if time.monotonic() > deadline - 30:
            break
        remaining = deadline - time.monotonic()
        if remaining < 20:
            break

        # Determine how many restarts we can afford
        time_per_restart = 20  # rough estimate
        n_restarts = max(1, int(remaining / (time_per_restart * 2)))
        n_restarts = min(n_restarts, 3)

        for restart in range(n_restarts):
            if time.monotonic() > deadline - 10:
                break

            a_init = structured_init(n, rng)

            # Quick gradient optimization
            try:
                a_g, obj_g = gradient_optimize(
                    a_init, tau_start=0.05, tau_end=0.001,
                    n_schedules=3, max_iter_per_sched=150)

                if obj_g < best_score:
                    best_score = obj_g
                    best_seq = a_g.copy()
                    print(f"  n={n} r={restart} grad: obj={best_score:.6f}")

                # Further refinement
                if time.monotonic() < deadline - 15:
                    a_ref2, obj_ref2 = refine_sequence(
                        a_g, seed + n * 100 + restart, deadline - 10, verbose=False)
                    if obj_ref2 < best_score:
                        best_score = obj_ref2
                        best_seq = a_ref2.copy()
                        print(f"  n={n} r={restart} refined: obj={best_score:.6f}")
            except Exception:
                pass

    # Phase 3: Final intensive refinement of the best
    if best_seq is not None and time.monotonic() < deadline - 10:
        remaining = deadline - time.monotonic()
        print(f"\nPhase 3: Final refinement, n={len(best_seq)}, {remaining:.0f}s remaining")

        # Gradient with very fine tau
        try:
            a_gf, obj_gf = gradient_optimize(
                best_seq, tau_start=0.005, tau_end=0.0001,
                n_schedules=4, max_iter_per_sched=int(min(500, remaining / 0.05)))
            if obj_gf < best_score:
                best_score = obj_gf
                best_seq = a_gf.copy()
                print(f"  Fine grad: obj={best_score:.6f}")
        except Exception:
            pass

        # LP polish
        if time.monotonic() < deadline - 5:
            try:
                n_final = len(best_seq)
                max_lp_iter = int(min(100, (deadline - time.monotonic()) / 1.0))
                a_lp, obj_lp = lp_optimize(best_seq, seed=seed + 7777,
                                           max_iter=max_lp_iter, t=0.3)
                if obj_lp < best_score:
                    best_score = obj_lp
                    best_seq = a_lp.copy()
                    print(f"  LP polish: obj={best_score:.6f}")
            except Exception:
                pass

        # Peak deflation
        if time.monotonic() < deadline - 3:
            a_def, obj_def = peak_deflation(best_seq, n_iter=100, alpha_start=0.15)
            if obj_def < best_score:
                best_score = obj_def
                best_seq = a_def.copy()
                print(f"  Deflation: obj={best_score:.6f}")

        # Random local search
        if time.monotonic() < deadline - 2:
            rls_iters = int((deadline - time.monotonic()) / 0.005)
            a_rls, obj_rls = random_refine(best_seq, rng, max_iter=rls_iters,
                                           noise_scale=0.08)
            if obj_rls < best_score:
                best_score = obj_rls
                best_seq = a_rls.copy()
                print(f"  RLS: obj={best_score:.6f}")

    # Fallback
    if best_seq is None:
        best_seq = np.ones(100)
        best_score = compute_obj(best_seq)

    elapsed = time.monotonic() - start_time
    print(f"\nDONE: n={len(best_seq)}, score={best_score:.6f}, time={elapsed:.1f}s")
    return [float(x) for x in best_seq]
