Hybrid optimization for minimizing 2*n*max(convolve(f,f))/(sum(f))^2.

import time

def normalize_sequence(arr):
    """Return a nonnegative floating vector normalized to sum 1, or None."""
    try:
        x = np.asarray(arr, dtype=np.float64).ravel()
    except Exception:
        return None
    if x.size == 0:
        return None
    x = np.where(np.isfinite(x), x, 0.0)
    x = np.maximum(x, 0.0)
    s = float(x.sum())
    if s <= 1e-300:
        return None
    x /= s
    return x


def score_normalized(x, n):
    """Score assuming x is nonnegative and sum(x) = 1. Score = 2*n*max(conv)."""
    conv = np.convolve(x, x)
    m = float(np.max(conv))
    if not np.isfinite(m):
        return float("inf")
    return float(2.0 * n * m)


def mixture_entropy(f):
    """Compute -sum(f * log(f)) as a measure of spread."""
    f = np.maximum(f, 1e-30)
    return -float(np.sum(f * np.log(f)))


def multiplicative_weights_step(f, conv, eta, n_hot=10):
    """
    One step of multiplicative weights (mirror descent) to reduce max conv.
    Decreases f at positions contributing to the hottest convolution bins.
    """
    n = f.size
    m = float(np.max(conv))
    if m <= 0:
        return f.copy()
    
    # Find hot bins (top n_hot)
    threshold = m * (1.0 - 0.1)  # bins within 10% of max
    hot_mask = conv >= threshold
    n_hot_actual = int(np.sum(hot_mask))
    if n_hot_actual == 0:
        return f.copy()
    
    # Subgradient: for each position j, sum over hot k of 2*f[k-j]
    # (f*f)_k = sum_j f[j] * f[k-j]
    # d(f*f)_k / d f[j] = 2 * f[k-j] (when valid)
    
    # Instead of computing the full subgradient (expensive),
    # approximate: for each hot k, the positions j that contribute most
    # are those with large f[k-j].
    
    # Build the subgradient
    subgrad = np.zeros(n, dtype=np.float64)
    for k in np.where(hot_mask)[0]:
        # (f*f)_k = sum_j f[j]*f[k-j] where 0<=j<n and 0<=k-j<n
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        # d/d f[j] of (f*f)_k = 2*f[k-j] for j in [lo, hi]
        # f[k-j] for j in [lo, hi] is f[k-lo] down to f[k-hi]
        indices = np.arange(lo, hi + 1)
        partner = k - indices
        subgrad[indices] += 2.0 * f[partner]
    
    # Normalize subgradient
    sg_max = float(np.max(subgrad))
    if sg_max > 0:
        subgrad /= sg_max
    
    # Multiplicative update: f_new[j] = f[j] * exp(-eta * subgrad[j])
    f_new = f * np.exp(-eta * subgrad)
    f_new = np.maximum(f_new, 0.0)
    s = float(f_new.sum())
    if s <= 1e-300:
        return f.copy()
    f_new /= s
    return f_new

def make_conv_template(n):
    """Precompute the sparse matrix structure for convolution LP."""
    m = 2 * n - 1
    k64 = np.arange(m, dtype=np.int64)
    js = np.maximum(0, k64 - n + 1).astype(np.int32)
    je = np.minimum(n - 1, k64).astype(np.int32)
    counts = (je.astype(np.int64) - js.astype(np.int64) + 1).astype(np.int64)
    total = int(counts.sum())
    
    cum = np.cumsum(counts)
    starts = np.repeat(cum - counts, counts)
    local = np.arange(total, dtype=np.int64) - starts
    
    row = np.repeat(k64, counts).astype(np.int32)
    col_base = js[row.astype(np.int64)]
    col = (col_base + local).astype(np.int32)
    offsets = (row.astype(np.int64) - col.astype(np.int64)).astype(np.int32)
    
    A_off = csr_matrix((offsets, (row, col)), shape=(m, n), dtype=np.int32)
    edge_offsets = A_off.data.astype(np.int32, copy=True)
    A_template = A_off.astype(np.float64).copy()
    A_template.data.fill(0.0)
    
    c_obj = -np.ones(n, dtype=np.float64)
    bounds = [(0.0, None)] * n
    return A_template, edge_offsets, c_obj, bounds, m


def solve_lp_direction(template, f, n, M):
    """Solve the LP: max sum(g) s.t. (f*g)_k <= M, g >= 0."""
    A_template, edge_offsets, c_obj, bounds, m = template
    lp_scale = float(n)
    
    A = A_template.copy()
    A.data[:] = f[edge_offsets]
    A.data *= lp_scale
    
    b = np.full(m, M * lp_scale * lp_scale, dtype=np.float64)
    return linprog(c_obj, A_ub=A, b_ub=b, bounds=bounds, method="highs")

        target = np.full(2*n-1, conv_sum / (2*n-1))

            break
        
        # Reduce f where it contributes to over-target bins
        # subgrad approach
        hot = ratio > 1.0
        subgrad = np.zeros(n, dtype=np.float64)
        for k_idx in np.where(hot)[0]:
            lo = max(0, int(k_idx) - n + 1)
            hi = min(n - 1, int(k_idx))
            indices = np.arange(lo, hi + 1)
            partner = int(k_idx) - indices
            weight = ratio[int(k_idx)] - 1.0
            subgrad[indices] += weight * f[partner]

            eta = 0.01
            f_new = f * np.exp(-eta * subgrad / np.max(subgrad))

            s = float(f_new.sum())
            if s > 1e-300:
                f_new /= s
                sc = score_normalized(f_new, n)
                if sc < best_score:
                    best_score = sc
                    best_f = f_new.copy()
                    f = f_new
        
    return best_f, best_score

"""
Hybrid optimization for minimizing 2*n*max(convolve(f,f))/(sum(f))^2.
Combines fast vectorized multiplicative weights (mirror descent), spectral
flattening, LP-based direction finding, and multi-scale random restarts.
Uses the given sequence as warm start.

import time
import numpy as np
from scipy.optimize import linprog
from scipy.sparse import csr_matrix


def normalize_sequence(arr):
    """Return a nonnegative floating vector normalized to sum 1, or None."""
    try:
        x = np.asarray(arr, dtype=np.float64).ravel()
    except Exception:
        return None
    if x.size == 0:
        return None
    x = np.where(np.isfinite(x), x, 0.0)
    x = np.maximum(x, 0.0)
    s = float(x.sum())
    if s <= 1e-300:
        return None
    x /= s
    return x


def score_normalized(x, n):
    """Score assuming x is nonnegative and sum(x)=1. Returns 2*n*max(conv)."""
    conv = np.convolve(x, x)
    m = float(np.max(conv))
    if not np.isfinite(m):
        return float("inf")
    return float(2.0 * n * m)


def compute_subgradient_vectorized(f, conv, frac=0.15):
    """
    Compute subgradient of max(conv) using vectorized convolution.
    Uses top fraction of bins as 'hot' set.
    Returns (subgrad, M) where subgrad has same length as f.
    """
    n = f.size
    m = float(np.max(conv))
    if m <= 0 or not np.isfinite(m):
        return np.zeros(n), m
    threshold = m * (1.0 - frac)
    h = (conv >= threshold).astype(np.float64)
    if np.all(h == 0):
        return np.zeros(n), m
    # subgrad[j] = 2 * sum_k h[k] * f[k-j]
    # = 2 * (h * f_flip)[j + n - 1]
    f_flip = f[::-1]
    corr = np.convolve(h, f_flip)
    subgrad = 2.0 * corr[n - 1: n - 1 + n]
    return subgrad, m


def mw_step(f, conv, eta, frac=0.15):
    """One multiplicative weights step. Returns new f (normalized)."""
    subgrad, m = compute_subgradient_vectorized(f, conv, frac)
    sg_max = float(np.max(subgrad))
    if sg_max <= 1e-30:
        return f.copy()
    subgrad = subgrad / sg_max
    f_new = f * np.exp(-eta * subgrad)
    f_new = np.maximum(f_new, 0.0)
    s = float(f_new.sum())
    if s <= 1e-300:
        return f.copy()
    f_new /= s
    return f_new


def mw_optimize(f_init, time_budget, eta_init, frac, rng, verbose_interval=5000):

    Run multiplicative weights optimization with adaptive step size.
    Returns (best_f, best_score).
    """

    if f is None:

    n = f.size

    best_score = score_normalized(f, n)
    eta = eta_init
    deadline = time.monotonic() + time_budget
    it = 0
    no_improve = 0

    while time.monotonic() < deadline:
        conv = np.convolve(f, f)
        cur_score = score_normalized(f, n)
        if cur_score < best_score:
            best_score = cur_score
            best_f = f.copy()
            no_improve = 0
        else:
            no_improve += 1

        # Adaptive step: try current eta, if no improvement reduce
        f_new = mw_step(f, conv, eta, frac)
        new_score = score_normalized(f_new, n)
        if new_score < best_score * (1.0 + 1e-12):

def spectral_optimize(f_init, time_budget, rng, verbose_interval=2000):
    """
    Optimize by flattening the power spectrum |FFT(f)|^2.
    Reduces peak of conv by making spectral energy more uniform.
    """

    if f is None:
        return None, float('inf')

    best_f = f.copy()

    deadline = time.monotonic() + time_budget

    fft_len = max(2 * n, 2048)
    while fft_len % 2 == 0 and fft_len < 16384:
        fft_len *= 2
        break
    fft_len = max(fft_len, 4096)

    eta = 0.05
    it = 0
    no_improve = 0

    while time.monotonic() < deadline:
        # Compute spectrum
        F = np.fft.rfft(f, fft_len)
        P = np.abs(F) ** 2
        P_max = float(np.max(P))
        if P_max <= 0:
            break

        # Target: flat spectrum
        target = np.full(P.shape, float(np.sum(P)) / P.size)
        # Emphasize reducing peaks
        excess = np.maximum(P - target, 0.0)
        excess_max = float(np.max(excess))
        if excess_max < 1e-15:
            break

        # Gradient of P_max with respect to f:
        # P[k] = |F[k]|^2, dF[k]/df[j] = exp(-2*pi*i*k*j/fft_len)
        # dP[k]/df[j] = 2*Re(F[k] * conj(exp(-2*pi*i*k*j/fft_len)))
        # For the hot frequencies, accumulate gradient
        hot_freqs = np.where(P > target * 1.01)[0]
        if len(hot_freqs) == 0:
            break

        # Subgradient: sum over hot freqs of weight * dP/d f
        # dP[k]/df[j] = 2*Re(F[k] * exp(2*pi*i*k*j/fft_len))
        # (using the fact that d|F[k]|^2/df = 2*Re(F[k]*conj(dF[k]/df)))
        # dF[k]/df[j] = exp(-2*pi*i*k*j/fft_len)
        # So dP[k]/df[j] = 2*Re(F[k] * exp(2*pi*i*k*j/fft_len))

        # Vectorize: grad[j] = 2 * sum_k excess[k]/P_max * Re(F[k] * exp(2*pi*i*k*j/fft_len))
        # = 2 * IFFT(excess_normalized * F) ... let me think
        # Actually: sum_k w[k] * F[k] * exp(2*pi*i*k*j/fft_len)
        # = IFFT(w * F) evaluated at j (with appropriate scaling)
        # np.fft.ifft(w * F)[j] = (1/fft_len) * sum_k w[k]*F[k]*exp(2*pi*i*k*j/fft_len)
        # So sum_k w[k]*F[k]*exp(2*pi*i*k*j/fft_len) = fft_len * ifft(w*F)[j]

        w = excess / (P_max * fft_len)
        grad_complex = np.fft.ifft(w * F)
        grad = 2.0 * np.real(grad_complex)[:n]

        grad_max = float(np.max(np.abs(grad)))
        if grad_max < 1e-30:
            break

        # Multiplicative update to reduce gradient
        f_new = f * np.exp(-eta * grad / grad_max)
        f_new = np.maximum(f_new, 0.0)
        s = float(f_new.sum())
        if s > 1e-300:
            f_new /= s
            new_score = score_normalized(f_new, n)
            if new_score < best_score:
                best_score = new_score
                best_f = f_new.copy()
                f = f_new
                no_improve = 0
            else:
                no_improve += 1
                eta *= 0.8

        it += 1
        if verbose_interval and it % verbose_interval == 0:
            print(f"    Spectral iter={it}, score={best_score:.8f}, eta={eta:.6f}",
                  flush=True)

    return best_f, best_score


def make_conv_template(n):
    """Precompute sparse matrix for convolution LP constraints."""
    m = 2 * n - 1
    k64 = np.arange(m, dtype=np.int64)
    js = np.maximum(0, k64 - n + 1).astype(np.int32)
    je = np.minimum(n - 1, k64).astype(np.int32)
    counts = (je.astype(np.int64) - js.astype(np.int64) + 1).astype(np.int64)
    total = int(counts.sum())

    cum = np.cumsum(counts)
    starts = np.repeat(cum - counts, counts)
    local = np.arange(total, dtype=np.int64) - starts

    row = np.repeat(k64, counts).astype(np.int32)
    col_base = js[row.astype(np.int64)]
    col = (col_base + local).astype(np.int32)
    offsets = (row.astype(np.int64) - col.astype(np.int64)).astype(np.int32)

    A_off = csr_matrix((offsets, (row, col)), shape=(m, n), dtype=np.int32)
    edge_offsets = A_off.data.astype(np.int32, copy=True)
    A_template = A_off.astype(np.float64).copy()
    A_template.data.fill(0.0)

    c_obj = -np.ones(n, dtype=np.float64)
    bounds = [(0.0, None)] * n
    return A_template, edge_offsets, c_obj, bounds, m


def solve_lp_direction(template, f, n, M):
    """Solve LP: max sum(g) s.t. (f*g)_k <= M for all k, g >= 0."""
    A_template, edge_offsets, c_obj, bounds, m = template
    A = A_template.copy()
    A.data[:] = f[edge_offsets]
    b = np.full(m, M, dtype=np.float64)
    return linprog(c_obj, A_ub=A, b_ub=b, bounds=bounds, method="highs")


def lp_optimize(f_init, time_budget, rng, template, max_iters=300):
    """LP-based optimization with safe step sizes."""
    f = normalize_sequence(f_init)
    if f is None:
        f = np.ones(max(2, f_init.size), dtype=np.float64) / max(2, f_init.size)
    n = f.size
    best_f = f.copy()
    best_score = score_normalized(f, n)
    deadline = time.monotonic() + time_budget
    stall = 0
    it = 0

    while time.monotonic() < deadline and it < max_iters:
        conv_ff = np.convolve(f, f)
        M = float(np.max(conv_ff))
        if not np.isfinite(M) or M <= 0:
            break

        try:
            res = solve_lp_direction(template, f, n, M)
        except Exception:
            break

        x = getattr(res, "x", None)
        if x is None:
            stall += 1
            if stall > 8:
                break
            f = best_f + 0.1 * rng.standard_normal(n)
            f = np.maximum(f, 0.0)
            s = f.sum()
            if s > 1e-300:
                f /= s
            continue

        x = np.asarray(x, dtype=np.float64).ravel()
        x = np.where(np.isfinite(x), x, 0.0)
        x = np.maximum(x, 0.0)
        s_star = float(x.sum())
        if s_star <= 1e-300 or s_star <= float(n) * (1.0 + 1e-8):
            stall += 1
            if stall > 8:
                break
            f = best_f + 0.1 * rng.standard_normal(n)
            f = np.maximum(f, 0.0)
            s = f.sum()
            if s > 1e-300:
                f /= s
            continue

        g = x / s_star
        r = np.convolve(f, g)
        r_max = float(np.max(r))

        if r_max >= M * (1.0 - 1e-10):
            stall += 1
            if stall > 8:
                break
            continue

        G = float(np.max(np.convolve(g, g)))
        a_coef = 2.0 * (r_max - M)
        b_coef = G + M - 2.0 * r_max
        if b_coef > 1e-14:
            t_max = -a_coef / b_coef
        else:
            t_max = 1.0
        if not np.isfinite(t_max) or t_max <= 0:
            t_max = 1e-8
        base = min(1.0, 0.8 * t_max)

        best_cand = None
        best_cand_score = best_score
        for tv in [base, 0.5 * base, 0.2 * base, 0.05 * base, min(base, 1e-3)]:
            if tv <= 0 or tv > 1:
                tv = min(max(tv, 1e-12), 1.0)
            f_new = (1.0 - tv) * f + tv * g
            f_new = np.maximum(f_new, 0.0)
            s = float(f_new.sum())
            if s <= 1e-300:
                continue
            f_new /= s
            sc = score_normalized(f_new, n)
            if sc < best_cand_score:
                best_cand_score = sc
                best_cand = f_new.copy()

        if best_cand is not None:
            f = best_cand
            best_f = f.copy()
            best_score = best_cand_score
            stall = 0
        else:
            stall += 1
            if stall > 8:
                break

        it += 1
        if it % 20 == 0:
            print(f"    LP iter={it}, score={best_score:.8f}", flush=True)

    return best_f, best_score


def variance_optimize(f_init, time_budget, rng):
    """
    Minimize variance of the convolution as a smooth surrogate for max.
    Reduces both the peak and the spread simultaneously.
    """
    f = normalize_sequence(f_init)
    if f is None:
        return None, float('inf')
    n = f.size
    best_f = f.copy()
    best_score = score_normalized(f, n)
    deadline = time.monotonic() + time_budget

    eta = 0.02
    it = 0
    no_improve = 0

    while time.monotonic() < deadline:
        conv = np.convolve(f, f)
        conv_mean = float(np.mean(conv))
        conv_var = float(np.mean((conv - conv_mean) ** 2))

        # Gradient of conv_var w.r.t. f[j]:
        # conv_var = mean(conv^2) - mean(conv)^2
        # d mean(conv^2) / d f[j] = 2 * sum_k conv[k] * d(conv[k])/d f[j] / (2n-1)
        # d(conv[k])/d f[j] = 2*f[k-j]
        # So d mean(conv^2) / d f[j] = (4/(2n-1)) * sum_k conv[k] * f[k-j]
        # = (4/(2n-1)) * (conv_of_conv * f_flip)[j+n-1]... 
        # Actually: sum_k conv[k] * f[k-j] = (conv * f_flip)[j + n - 1]
        # But conv has length 2n-1 and f has length n, so conv * f_flip has length 3n-2
        # We want indices n-1 to 2n-2.

        conv_flip = conv[::-1]
        # sum_k conv[k]*f[k-j]: let's compute as cross-correlation
        # = (conv * f[::-1])[j + (2n-2)] ... hmm let me just use the direct formula
        # Actually: sum_k conv[k] * f[k-j] for k in [0,2n-2] and k-j in [0,n-1]
        # This is (conv * f_rev)[m] where m = j + n - 1... let me verify
        # (conv * f_rev)[m] = sum_k conv[k] * f_rev[m-k] = sum_k conv[k] * f[n-1-m+k]
        # For m = j + n - 1: = sum_k conv[k] * f[n-1-j-n+1+k] = sum_k conv[k] * f[k-j] ✓

        f_rev = f[::-1]
        prod = np.convolve(conv, f_rev)
        grad_conv2 = (4.0 / (2 * n - 1)) * prod[n - 1: n - 1 + n]

        # d mean(conv)^2 / d f[j] = 2*mean(conv) * d mean(conv)/d f[j]
        # mean(conv) = (sum f)^2 / (2n-1) = 1/(2n-1) (normalized)
        # d mean(conv) / d f[j] = 2*f[j] * (sum f) / (2n-1) = 2*f[j]/(2n-1)
        # (since sum f = 1)
        grad_mean2 = 2.0 * conv_mean * 2.0 * f / (2 * n - 1)

        grad = grad_conv2 - grad_mean2

        # Combine with max-reduction: also penalize positions contributing to max
        m = float(np.max(conv))
        subgrad_max, _ = compute_subgradient_vectorized(f, conv, 0.1)
        combined_grad = grad + 0.5 * subgrad_max / max(1e-30, np.max(subgrad_max))

        gm = float(np.max(np.abs(combined_grad)))
        if gm < 1e-30:
            break

        f_new = f * np.exp(-eta * combined_grad / gm)
        f_new = np.maximum(f_new, 0.0)
        s = float(f_new.sum())
        if s > 1e-300:
            f_new /= s
            new_score = score_normalized(f_new, n)
            if new_score < best_score:
                best_score = new_score
                best_f = f_new.copy()
                f = f_new
                no_improve = 0
            else:
                no_improve += 1
                eta *= 0.9

        it += 1
        if it % 5000 == 0:
            print(f"    Var iter={it}, score={best_score:.8f}, eta={eta:.6f}", flush=True)

    return best_f, best_score


def interpolate_sequence(x, n):
    """Resample a sequence to length n and normalize."""
    x = normalize_sequence(x)
    if x is None or n <= 0:
        return None
    m = x.size
    if m == n:
        return x
    if m == 1:
        return np.ones(n, dtype=np.float64) / float(n)
    old = np.linspace(0.0, 1.0, m)
    new = np.linspace(0.0, 1.0, n)
    out = np.interp(new, old, x)
    return normalize_sequence(out)


def propose_candidate(seed=42, budget_s=1500, **kwargs):
    """
    Hybrid search: MW + spectral + variance + LP + restarts.
    Targets n=1024 primarily, with optional exploration of nearby lengths.
    """
    rng = np.random.default_rng(int(seed) if seed is not None else 42)
    total_budget = max(1.0, float(budget_s))
    start_time = time.monotonic()
    deadline = start_time + total_budget - 20.0

    best_f = None
    best_score = float('inf')
    given_f = None

    # Load given sequence
    try:
        raw = np.asarray(height_sequence_1, dtype=np.float64)
        given_f = normalize_sequence(raw)
        if given_f is not None:
            best_f = given_f.copy()
            best_score = score_normalized(given_f, given_f.size)
            print(f"Given: n={given_f.size}, score={best_score:.8f}", flush=True)
    except Exception as e:
        print(f"Could not load given: {e}", flush=True)

    if best_f is None:
        best_f = np.ones(1024, dtype=np.float64) / 1024.0
        best_score = 2.0

    n = best_f.size
    print(f"Working with n={n}, target < 1.5027", flush=True)

    # === Phase 1: Fast MW refinement ===
    print(f"\n=== Phase 1: MW (eta=0.1, frac=0.15) ===", flush=True)
    t1 = min(150.0, (deadline - time.monotonic()) * 0.15)
    if t1 > 5:
        f1, s1 = mw_optimize(best_f, t1, 0.1, 0.15, rng)
        if s1 < best_score:
            best_score = s1
            best_f = f1.copy()
            print(f"  Phase 1 result: {best_score:.8f}", flush=True)

    # === Phase 2: Spectral flattening ===
    print(f"\n=== Phase 2: Spectral ===", flush=True)
    t2 = min(150.0, (deadline - time.monotonic()) * 0.15)
    if t2 > 5:
        f2, s2 = spectral_optimize(best_f, t2, rng)
        if f2 is not None and s2 < best_score:
            best_score = s2
            best_f = f2.copy()
            print(f"  Phase 2 result: {best_score:.8f}", flush=True)

    # === Phase 3: Variance-based optimization ===
    print(f"\n=== Phase 3: Variance ===", flush=True)
    t3 = min(150.0, (deadline - time.monotonic()) * 0.12)
    if t3 > 5:
        f3, s3 = variance_optimize(best_f, t3, rng)
        if f3 is not None and s3 < best_score:
            best_score = s3
            best_f = f3.copy()
            print(f"  Phase 3 result: {best_score:.8f}", flush=True)

    # === Phase 4: LP refinement ===
    print(f"\n=== Phase 4: LP ===", flush=True)
    template = None
    if n <= 2048:
        try:
            template = make_conv_template(n)
        except Exception as e:
            print(f"  Template error: {e}", flush=True)

    if template is not None:
        t4 = min(400.0, (deadline - time.monotonic()) * 0.3)
        if t4 > 30:
            f4, s4 = lp_optimize(best_f, t4, rng, template)
            if s4 < best_score:
                best_score = s4
                best_f = f4.copy()
                print(f"  Phase 4 result: {best_score:.8f}", flush=True)

    # === Phase 5: Fine MW polishing ===
    print(f"\n=== Phase 5: Fine MW polish ===", flush=True)
    t5 = min(200.0, (deadline - time.monotonic()) * 0.2)
    if t5 > 5:
        f5, s5 = mw_optimize(best_f, t5, 0.01, 0.05, rng, verbose_interval=10000)
        if s5 < best_score:
            best_score = s5
            best_f = f5.copy()
            print(f"  Phase 5 result: {best_score:.8f}", flush=True)

    # === Phase 6: Random restarts with different strategies ===
    print(f"\n=== Phase 6: Restart exploration ===", flush=True)
    n_restarts = 4
    rem = deadline - time.monotonic()
    per_restart = rem / max(1, n_restarts + 1)

    for r in range(n_restarts):
        rem = deadline - time.monotonic()
        if rem < 15:
            break

        # Different perturbation strategies
        if r == 0:
            # Small Gaussian noise
            noisy = best_f + 0.03 * rng.standard_normal(n)
        elif r == 1:
            # Dirichlet mixture
            d = rng.dirichlet(np.full(n, 0.5))
            noisy = 0.7 * best_f + 0.3 * d
        elif r == 2:
            # Anti-correlated perturbation: reduce where conv is high
            conv = np.convolve(best_f, best_f)
            hot = (conv >= np.max(conv) * 0.9).astype(np.float64)
            subgrad, _ = compute_subgradient_vectorized(best_f, conv, 0.2)
            noisy = best_f * (1.0 - 0.3 * subgrad / max(1e-30, np.max(subgrad)))
        else:
            # Frequency-domain perturbation
            F = np.fft.rfft(best_f, 4096)
            F *= (1.0 + 0.1 * rng.standard_normal(F.size))
            noisy = np.real(np.fft.irfft(F, 4096))[:n]

        noisy = np.maximum(noisy, 0.0)
        s_noisy = float(noisy.sum())
        if s_noisy > 1e-300:
            noisy /= s_noisy

        # Optimize the perturbed sequence
        tb = min(per_restart, 100.0)
        if tb < 10:
            break
        fr, sr = mw_optimize(noisy, tb, 0.05, 0.1, rng, verbose_interval=10000)
        if sr < best_score:
            best_score = sr
            best_f = fr.copy()
            print(f"  Restart {r}: {best_score:.8f}", flush=True)

    # === Phase 7: Final ultra-fine polish ===
    print(f"\n=== Phase 7: Final polish ===", flush=True)
    t7 = max(0.0, deadline - time.monotonic())
    if t7 > 3:
        f7, s7 = mw_optimize(best_f, t7, 0.003, 0.03, rng, verbose_interval=20000)
        if s7 < best_score:
            best_score = s7
            best_f = f7.copy()
            print(f"  Final: {best_score:.8f}", flush=True)

    # Final answer
    seq = [float(max(0.0, x)) for x in best_f]

    # Verify
    try:
        final_score = float(evaluate_sequence(seq))
        print(f"\nFINAL: n={len(seq)}, internal={best_score:.8f}, "
              f"evaluate={final_score:.8f}", flush=True)
    except Exception as e:
        print(f"Eval error: {e}, falling back to given", flush=True)
        if given_f is not None:
            seq = [float(max(0.0, x)) for x in given_f]

    if not np.isfinite(best_score) and given_f is None:
        seq = [1.0 / 64.0] * 64

    return seq
