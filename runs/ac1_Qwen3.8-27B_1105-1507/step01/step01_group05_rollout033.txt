```python
def compute_grad(f, p):
    n = len(f)
    # p has length 2n-1
    # grad[j] = sum_k p[k] * (f[k-j] if 0<=k-j<n else 0 + f[j-k] if 0<=j-k<n else 0)
    
    # First part: sum_k p[k] * f[k-j] (valid)
    # For each j, this is sum_{m=0}^{n-1} p[m+j] * f[m] (where p[m+j]=0 if out of range)
    # This is: (sliding dot product of p[j:] with f)
    
    # Second part: sum_k p[k] * f[j-k] (valid)
    # For each j, this is sum_{m=0}^{min(j,2n-2-j)} p[j-m] * f[m]
    # This is: (sliding dot product of p[:j+1] reversed with f[:j+1])
    
    # Efficient computation using FFT:
    # First part = cross_correlation(p, f) evaluated at positions 0, 1, ..., n-1
    # Second part = convolution(f, p) evaluated at positions 0, 1, ..., n-1
    
    N = 1
    while N < len(p) + n:
        N *= 2
    
    P = np.fft.rfft(p, N)
    F = np.fft.rfft(f, N)
    
    # Cross-correlation: ifft(conj(P) * F)
    corr = np.fft.irfft(np.conj(P) * F, N)
    
    # Convolution: ifft(F * P)
    conv = np.fft.irfft(F * P, N)
    
    return corr[:n] + conv[:n]
```

```python
"""
Hybrid optimization for minimizing 2*n*max(convolve(a,a))/sum(a)^2.
Combines soft-max gradient descent, LP-based optimization, and multiple
diverse starting points to find low-scoring non-negative sequences.
"""

import numpy as np
from scipy.optimize import linprog
import time


def compute_score_normalized(f):
    """Compute score for a normalized (sum=1) non-negative sequence."""
    n = len(f)
    conv = np.convolve(f, f)
    return 2.0 * n * conv.max()


def simplex_project(v):
    """Project v onto the probability simplex {x >= 0, sum(x) = 1}. O(n log n)."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - 1))[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho = rho[-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0)


def build_conv_matrix(f):
    """Build matrix A such that A @ b = np.convolve(f, b, mode='full')."""
    n = len(f)
    A = np.zeros((2 * n - 1, n))
    for k in range(2 * n - 1):
        j_start = max(0, k - n + 1)
        j_end = min(n, k + 1)
        for j in range(j_start, j_end):
            A[k, j] = f[k - j]
    return A


def softmax_gradient_descent(f, tau_start=0.5, tau_end=0.01, n_iter=2000, time_budget=60.0, rng=None):
    """
    Soft-max gradient descent with temperature annealing.
    Minimizes soft_max(convolve(f,f), tau) with tau decreasing over time.
    """
    if rng is None:
        rng = np.random.default_rng()
    
    n = len(f)
    f = f.astype(np.float64).copy()
    S = f.sum()
    f = f / S
    
    best_f = f.copy()
    best_score = compute_score_normalized(f)
    
    start = time.monotonic()
    N = 1
    while N < 3 * n:
        N *= 2
    
    stall = 0
    
    for it in range(n_iter):
        if time.monotonic() - start > time_budget:
            break
        
        # Anneal temperature
        progress = min(1.0, it / max(1, n_iter))
        tau = tau_start * (tau_end / tau_start) ** progress
        
        # Compute soft-max weights
        conv = np.convolve(f, f)
        conv_shifted = conv - conv.max()
        p = np.exp(conv_shifted / tau)
        p_sum = p.sum()
        if p_sum < 1e-300:
            p = np.ones(len(conv)) / len(conv)
        else:
            p = p / p_sum
        
        # Compute gradient using FFT
        P = np.fft.rfft(p, N)
        F = np.fft.rfft(f, N)
        
        # Cross-correlation: corr[j] = sum_k p[k] * f[k-j] (valid)
        corr = np.fft.irfft(np.conj(P) * F, N)
        # Convolution: conv_fp[j] = sum_m f[m] * p[j-m]
        conv_fp = np.fft.irfft(F * P, N)
        
        grad = (corr[:n] + conv_fp[:n]) / tau
        
        # Gradient descent with backtracking line search
        alpha = 1.0
        improved = False
        for _ in range(10):
            f_new = f - alpha * grad
            f_new = simplex_project(f_new)
            score = compute_score_normalized(f_new)
            if score < best_score - 1e-15:
                f = f_new
                if score < best_score:
                    best_score = score
                    best_f = f.copy()
                improved = True
                stall = 0
                break
            alpha *= 0.5
        
        if not improved:
            stall += 1
            if stall > 50:
                # Random kick
                f = f + rng.normal(0, 0.01, n)
                f = simplex_project(f)
                stall = 0
    
    return best_f, best_score

def spectral_init(n, rng):
    """Generate a sequence with approximately flat spectrum."""
    # Random phases in frequency domain
    N = n
    phases = rng.uniform(0, 2 * np.pi, N // 2 + 1)
    # Amplitudes: try to be flat
    amps = np.ones(N // 2 + 1)
    # Build spectrum
    spec = np.zeros(N // 2 + 1, dtype=np.complex128)
    spec[0] = amps[0]  # DC
    for k in range(1, N // 2 + 1):
        spec[k] = amps[k] * np.exp(1j * phases[k])
    
    # Inverse FFT
    f = np.fft.irfft(spec, N)
    f = np.abs(f)  # Make non-negative
    f = np.clip(f, 0, None)
    S = f.sum()
    if S < 1e-10:
        return np.ones(n) / n
    return f / S
```
