def smooth_gradient_step(f, tau=200.0, lr=0.005, rng=None):
    """One step of projected gradient descent on smooth approximation of max(conv)."""

    w_sum = w.sum()
    if w_sum < 1e-300:

    # = 2 * correlate(w, f, 'full')[n-1 : 2n-1]

    grad = 2.0 * corr[n - 1:2 * n - 1]

    f_new = project_simplex(f - lr * grad, 1.0)
    return f_new

def crossover(p1, p2, rng):
    """Uniform crossover with noise."""
    mask = rng.random(len(p1)) < 0.5
    child = np.where(mask, p1, p2)
    child = child * (1 + 0.05 * rng.standard_normal(len(child)))
    child = np.clip(child, 0, None)
    s = child.sum()
    if s < 1e-10:
        child = (p1 + p2) / 2
        s = child.sum()
    return child / s


def mutate(f, rng, n_mut=3):
    """Apply multiple mutation operations."""
    f = f.copy()
    n = len(f)
    for _ in range(n_mut):
        mtype = rng.integers(0, 5)
        if mtype == 0:
            # Block scaling
            w = max(1, int(rng.integers(1, n // 8)))
            start = int(rng.integers(0, n - w))
            f[start:start + w] *= np.exp(rng.normal(0, 0.4))
        elif mtype == 1:
            # Block swap
            w = max(1, int(rng.integers(1, n // 6)))
            s1 = int(rng.integers(0, n - w))
            s2 = int(rng.integers(0, n - w))
            f[s1:s1 + w], f[s2:s2 + w] = f[s2:s2 + w].copy(), f[s1:s1 + w].copy()
        elif mtype == 2:
            # Gaussian noise
            f += rng.normal(0, 0.05, n)
        elif mtype == 3:
            # Sinusoidal perturbation
            freq = rng.uniform(1.0, n / 4.0)
            phase = rng.uniform(0, 2 * np.pi)
            f += 0.1 * np.sin(2 * np.pi * np.arange(n) / freq + phase)
        elif mtype == 4:
            # Invert a block

            start = int(rng.integers(0, n - w))
            f[start:start + w] = f[start:start + w][::-1]

    s = f.sum()
    if s < 1e-10:

        s = n

    start_types.append(('ramp', make_ramp_init(n, rng)))

    # Quick LP for each
    quick_time = min(8.0, time_budget / (len(start_types) + 2))
    for name, f0 in start_types:
        if time.monotonic() > deadline - 30:
            break
        f0 = project_simplex(f0, 1.0)
        t_left = min(quick_time, deadline - time.monotonic() - 30)
        if t_left < 3:
            break
        f_opt, sc = lp_local(f0, t_left, init_t=0.4, rng=rng)
        if sc < best_score:
            best_score = sc
            best_f = f_opt
            print(f"      [{name}] n={n}: {sc:.6f}")

    # Phase 2: GA search
    if time.monotonic() < deadline - 60:
        ga_time = deadline - time.monotonic() - 60
        ga_pop = max(30, min(100, int(ga_time / 2)))
        f_ga, sc_ga = ga_search(best_f, ga_time, pop_size=ga_pop, rng=rng)
        if sc_ga < best_score:
            best_score = sc_ga
            best_f = f_ga
            print(f"      [GA] n={n}: {sc_ga:.6f}")

    # Phase 3: Final polishing
    if time.monotonic() < deadline - 10:
        final_time = deadline - time.monotonic() - 5
        f_final, sc_final = lp_local(best_f, final_time, init_t=0.3, rng=rng)
        if sc_final < best_score:
            best_score = sc_final
            best_f = f_final
            print(f"      [Final LP] n={n}: {sc_final:.6f}")

        # Also try gradient polish
        if time.monotonic() < deadline - 5:
            g_time = deadline - time.monotonic() - 3
            f_grad, sc_grad = gradient_polish(best_f, g_time, tau=300, lr=0.003, rng=rng)
            if sc_grad < best_score:
                best_score = sc_grad
                best_f = f_grad
                print(f"      [Grad] n={n}: {sc_grad:.6f}")

    return best_f, best_score


def propose_candidate(seed=42, budget_s=1500, **kwargs):
    """Main search: try multiple lengths, GA + LP hybrid."""
    rng = np.random.default_rng(seed)
    t_start = time.monotonic()
    deadline = t_start + budget_s - 15.0

    best_overall = None
    best_score_overall = np.inf

    # Try to get parent
    parent = None
    try:
        parent = np.array(height_sequence_1, dtype=np.float64)
        if parent.ndim != 1 or len(parent) == 0 or not np.isfinite(parent).all():
            parent = None
        else:
            parent = np.clip(parent, 0, 1000)
            if parent.sum() < 0.01:
                parent = None
    except Exception:
        parent = None

    print(f"Parent available: {parent is not None}")
    if parent is not None:
        print(f"  Parent length: {len(parent)}, score: {score_norm(project_simplex(parent, 1.0)):.6f}")

    # Define lengths to try
    lengths = [256, 384, 512, 768, 1024]

    # If parent exists, include its length
    if parent is not None:
        pn = len(parent)
        if pn not in lengths and 64 <= pn <= 2048:
            lengths.insert(0, pn)

    # Allocate time: weight earlier lengths more (they're faster)
    # Total time for search phases
    total_search_time = budget_s - 30.0
    n_lengths = len(lengths)

    # Give more time to smaller lengths initially, then distribute
    weights = [1.0 / (l / 256) for l in lengths]
    w_sum = sum(weights)
    time_alloc = [total_search_time * w / w_sum for w in weights]

    # Cap individual allocations
    for i in range(n_lengths):
        time_alloc[i] = min(time_alloc[i], total_search_time * 0.4)

    print(f"Lengths to try: {lengths}")
    print(f"Time allocation: {[f'{t:.0f}s' for t in time_alloc]}")

    for i, n in enumerate(lengths):
        if time.monotonic() > deadline:
            break

        # Get parent for this length if possible
        p_for_n = None
        if parent is not None:
            if len(parent) == n:
                p_for_n = parent.copy()
            elif len(parent) > n:
                p_for_n = parent[:n].copy()
            else:
                # Pad or interpolate
                p_for_n = np.interp(np.linspace(0, len(parent) - 1, n),
                                   np.arange(len(parent)), parent)
                p_for_n = np.clip(p_for_n, 0, None)

        remaining = deadline - time.monotonic()
        t_here = min(time_alloc[i], remaining - 20)
        if t_here < 15:
            break

        print(f"\n  Searching n={n}, budget={t_here:.0f}s")
        f_best, sc_best = search_length(n, t_here, rng, parent=p_for_n)

        if sc_best < best_score_overall:
            best_score_overall = sc_best
            best_overall = f_best.copy()
            print(f"  ** New overall best: {sc_best:.6f} (n={n}) **")

    # Final phase: try to improve the best found
    if best_overall is not None and time.monotonic() < deadline - 20:
        n_best = len(best_overall)
        remaining = deadline - time.monotonic()
        print(f"\n  Final improvement phase: n={n_best}, remaining={remaining:.0f}s")

        # Try multiple perturbations of the best
        for attempt in range(5):
            if time.monotonic() > deadline - 15:
                break
            f_pert = best_overall * (1 + 0.15 * rng.standard_normal(n_best))
            f_pert = project_simplex(f_pert, 1.0)
            t_here = (deadline - time.monotonic() - 15) / (5 - attempt)
            t_here = max(5, min(t_here, 60))
            f_opt, sc = lp_local(f_pert, t_here, init_t=0.3, rng=rng)
            if sc < best_score_overall:
                best_score_overall = sc
                best_overall = f_opt.copy()
                print(f"    Perturb {attempt}: {sc:.6f}")

        # One more gradient polish
        if time.monotonic() < deadline - 10:
            g_time = deadline - time.monotonic() - 5
            f_grad, sc_grad = gradient_polish(best_overall, g_time, tau=500, lr=0.002, rng=rng)
            if sc_grad < best_score_overall:
                best_score_overall = sc_grad
                best_overall = f_grad.copy()
                print(f"    Final gradient: {sc_grad:.6f}")

    # Verify and return
    if best_overall is not None:
        best_overall = np.clip(best_overall, 0, 1000)
        # Scale to avoid issues with very small or very large values
        s = best_overall.sum()
        if s > 0:
            best_overall = best_overall / s  # normalize to sum=1
            # Actually, let's keep it at sum=1 for now and let evaluate_sequence handle it
            # The score is scale-invariant, so sum=1 is fine
        final_list = best_overall.tolist()
        final_score = evaluate_sequence(final_list)
        print(f"\nFinal verified score: {final_score:.6f}")
        print(f"Sequence length: {len(final_list)}")
        print(f"Sum: {sum(final_list):.4f}")
        return final_list

    # Fallback
    print("Fallback: uniform 256")
    return [1.0] * 256
