```python
import time
import numpy as np
from scipy.optimize import linprog

def compute_score(a):
    a = np.asarray(a, dtype=np.float64)
    n = len(a)
    c = np.convolve(a, a)
    s = np.sum(a)
    if s < 1e-12:
        return float('inf')
    return float(2.0 * n * np.max(c) / (s * s))

def normalize(a):
    a = np.asarray(a, dtype=np.float64)
    a = np.clip(a, 0, None)
    s = np.sum(a)
    if s < 1e-300:
        return a
    return a / s

def build_conv_matrix(b, n, sym=False):
    """Build matrix A such that (A @ a)[k] = sum_j a[j] * b[k-j] for valid k."""
    if sym:
        m = (n + 1) // 2
        A = np.zeros((n, m), dtype=np.float64)
        for v in range(m):
            # a[v] contributes to c[k] via b[k-v] for k = v, ..., v+n-1
            # But we only keep k = 0, ..., n-1 (by symmetry of a)
            idx = np.arange(v, n)
            A[idx, v] = b[:n-v]
            # Also a[n-1-v] contributes (by symmetry a[n-1-v] = a[v])
            j2 = n - 1 - v
            if j2 != v:
                idx2 = np.arange(j2, n)  # k = j2, ..., j2+n-1, but we only want k < n
                # c[k] gets contribution a[j2]*b[k-j2] = a[v]*b[k-j2] for k >= j2
                # k ranges from j2 to j2+n-1, but we only keep k < n
                if j2 < n:
                    end = min(n, j2 + n)
                    start = j2
                    A[start:end, v] += b[:end-start]
        return A
    else:
        A = np.zeros((2*n-1, n), dtype=np.float64)
        for k in range(2*n-1):
            j_start = max(0, k - n + 1)
            j_end = min(n - 1, k)
            if j_start <= j_end:
                A[k, j_start:j_end+1] = b[k-j_end:k-j_start+1]
        return A
```

```python
"""
Minimize score = 2*n*max(convolve(a,a)) / sum(a)^2 using alternating LP 
with multiple restarts, varied lengths, and perturbation-based diversification.

import time
import numpy as np
from scipy.optimize import linprog

def compute_score(a):
    a = np.asarray(a, dtype=np.float64)
    n = len(a)
    if n == 0:
        return float('inf')
    s = np.sum(a)
    if s < 1e-12:
        return float('inf')
    c = np.convolve(a, a)
    return float(2.0 * n * np.max(c) / (s * s))

def normalize_sum1(a):
    a = np.asarray(a, dtype=np.float64).copy()
    a = np.clip(a, 0, None)
    s = np.sum(a)
    if s < 1e-300:
        return np.ones_like(a) / len(a)
    return a / s

def solve_lp_fix_b(b, n):
    """Given b (fixed), solve LP: min t s.t. conv(a,b)[k] <= t, sum(a)=1, a>=0.
    Returns optimal a or None."""
    # Build A_conv[2n-1, n]
    A_conv = np.zeros((2*n-1, n), dtype=np.float64)
    for k in range(2*n-1):
        js = max(0, k - n + 1)
        je = min(n-1, k)
        if js <= je:
            idx = np.arange(js, je+1)
            A_conv[k, idx] = b[k - idx]
    
    A_ub = np.column_stack([A_conv, -np.ones(2*n-1)])
    b_ub = np.zeros(2*n-1)
    A_eq = np.array([[np.ones(n), 0.0]])
    b_eq = np.array([1.0])
    c_obj = np.zeros(n+1)
    c_obj[-1] = 1.0
    bounds = [(0, None)]*(n+1)
    
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs', options={'presolve': True})
        if res.success:
            a = np.clip(res.x[:n], 0, None)
            s = np.sum(a)
            if s > 1e-12:
                return a / s
    except:
        pass
    return None

def alternating_lp(a_init, time_budget, seed, blend_factor=0.5):
    """Run alternating LP optimization."""
    a = normalize_sum1(a_init)
    n = len(a)
    b = a.copy()
    best_a = a.copy()
    best_score = compute_score(a)
    
    rng = np.random.default_rng(seed)
    t0 = time.monotonic()
    deadline = t0 + time_budget
    iterations = 0
    no_improve = 0
    
    while time.monotonic() < deadline:
        # Step 1: optimize a given b
        new_a = solve_lp_fix_b(b, n)
        if new_a is not None:
            # Blend with current a for smoother convergence
            a_blend = (1 - blend_factor) * a + blend_factor * new_a
            a = normalize_sum1(a_blend)
        else:
            a = new_a if new_a is not None else a
        
        sc = compute_score(a)
        if sc < best_score:
            best_score = sc
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        # Step 2: optimize b given a
        new_b = solve_lp_fix_b(a, n)
        if new_b is not None:
            b = normalize_sum1((1 - blend_factor) * b + blend_factor * new_b)
        else:
            b = a.copy()
        
        iterations += 1
        
        # Periodic perturbation to escape local minima
        if no_improve > 5:
            noise = np.exp(rng.normal(0, 0.1, n))
            a = normalize_sum1(best_a * noise)
            b = a.copy()
            no_improve = 0
        
        if no_improve > 15:
            # Bigger restart
            a = normalize_sum1(best_a * np.exp(rng.normal(0, 0.3, n)))
            b = a.copy()
            no_improve = 0
    
    return best_a, best_score
```

```python
    n = len(a)
    c = np.convolve(a, a)
```

```python
    base = c.copy()
    coef = np.zeros(2*n-1)
    
    a_i = a[i]
    for k in range(2*n-1):
        kmi = k - i
        if 0 <= kmi < n:
            if kmi == i:
                base[k] -= a_i * a_i
                # quadratic term: x^2
            else:
                base[k] -= 2 * a_i * a[kmi]
                coef[k] = 2 * a[kmi]
    
    def objective(x):
        vals = base + coef * x
        if 2*i < len(vals):
            vals[2*i] += x * x
        return np.max(vals)
    
    # Ternary search on [0, upper]
    lo = 0.0
    hi = max(10.0 * a_i, np.sum(a))
    
    # First, find a good upper bound
    while objective(hi) > objective(hi * 0.5) and hi < 100:
        hi *= 0.5
    while objective(hi) < objective(hi * 2) and hi < 1000:
        hi *= 2
    
    for _ in range(n_iters):
        m1 = lo + (hi - lo) / 3
        m2 = hi - (hi - lo) / 3
        if objective(m1) < objective(m2):
            hi = m2
        else:
            lo = m1

    a[i] = max(0, x_opt)
    return a
```

```python
"""
Minimize score = 2*n*max(convolve(a,a))/sum(a)^2 using coordinate descent
with incremental convolution updates, LP polishing steps, and multiple
diverse restarts across several sequence lengths.

from scipy.optimize import linprog


def compute_score(a):
    """Compute the score: 2*n*max(conv)/sum^2."""
    a = np.asarray(a, dtype=np.float64)
    n = len(a)
    if n == 0:
        return float('inf')
    s = np.sum(a)
    if s < 1e-12:
        return float('inf')
    c = np.convolve(a, a)
    return float(2.0 * n * np.max(c) / (s * s))


def normalize_sum1(a):
    """Normalize to sum 1, clip negatives."""
    a = np.asarray(a, dtype=np.float64).copy()
    a = np.clip(a, 0.0, None)
    s = np.sum(a)
    if s < 1e-300:
        return np.ones_like(a) / len(a)
    return a / s


def solve_lp_fix_b(b, n):
    """Given fixed b, solve LP: min t s.t. conv(a,b)[k]<=t, sum(a)=1, a>=0."""
    A_conv = np.zeros((2 * n - 1, n), dtype=np.float64)
    for k in range(2 * n - 1):
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        if js <= je:
            idx = np.arange(js, je + 1)
            A_conv[k, idx] = b[k - idx]
    A_ub = np.column_stack([A_conv, -np.ones(2 * n - 1)])
    b_ub = np.zeros(2 * n - 1)
    A_eq = np.array([[np.ones(n), 0.0]])
    b_eq = np.array([1.0])
    c_obj = np.zeros(n + 1)
    c_obj[-1] = 1.0
    bounds = [(0, None)] * (n + 1)
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs',
                      options={'presolve': True, 'primal_feasibility_tolerance': 1e-9})
        if res.success:
            a = np.clip(res.x[:n], 0, None)
            s = np.sum(a)
            if s > 1e-12:
                return a / s
    except Exception:
        pass
    return None

        x_new = optimize_single_coord(c, a, i, n)
        delta = x_new - old
        if abs(delta) > 1e-10:

        # Normalize periodically
        if passes % 100 == 0:
            a = normalize_sum1(a)
            c = np.convolve(a, a)
        
        sc = float(2.0 * n * np.max(c) / (np.sum(a) ** 2))
        if sc < best_score:
            if best_score - sc > 1e-10:
                best_score = sc
                best_a = a.copy()
                no_improve = 0
            else:
                no_improve += 1
        else:
            no_improve += 1
        
        # Perturbation if stuck
        if no_improve > 200:
            noise = np.exp(rng.normal(0, 0.15, n))
            a = normalize_sum1(best_a * noise)
            c = np.convolve(a, a)
            no_improve = 0
    
    return best_a, best_score


def lp_polish(a_init, time_budget, seed):
    """Alternating LP polishing."""
    a = normalize_sum1(a_init)
    n = len(a)
    b = a.copy()
    best_a = a.copy()
    best_score = compute_score(a)
    
    t0 = time.monotonic()
    deadline = t0 + time_budget
    rng = np.random.default_rng(seed)
    no_improve = 0
    
    while time.monotonic() < deadline:
        # Optimize a given b
        new_a = solve_lp_fix_b(b, n)
        if new_a is not None:
            a = normalize_sum1(0.3 * a + 0.7 * new_a)
        sc = compute_score(a)
        if sc < best_score:
            best_score = sc
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        if time.monotonic() >= deadline:
            break
        
        # Optimize b given a
        new_b = solve_lp_fix_a = solve_lp_fix_b(a, n)
        if new_b is not None:
            b = normalize_sum1(0.3 * b + 0.7 * new_b)
        
        if no_improve > 8:
            noise = np.exp(rng.normal(0, 0.2, n))
            a = normalize_sum1(best_a * noise)
            b = a.copy()
            no_improve = 0
    
    return best_a, best_score


def make_init(n, kind, rng, parent=None):
    """Generate an initial sequence."""
    if kind == 0:
        return np.ones(n) / n
    elif kind == 1:
        x = np.linspace(0, 1, n)
        return 0.5 * (1 + np.cos(2 * np.pi * (x - 0.5)))
    elif kind == 2:
        x = np.linspace(0, 1, n)
        return 1.0 - np.abs(2 * x - 1)
    elif kind == 3:
        return rng.beta(2, 2, n)
    elif kind == 4:
        v = rng.normal(0, 1, n)
        # Smooth
        k = max(3, n // 32)
        kernel = np.ones(k) / k
        v = np.convolve(v, kernel, mode='same')
        v = np.clip(v, 0, None)
        if v.sum() < 1e-10:
            v = np.abs(v)
        return v / v.sum()
    elif kind == 5:
        if parent is not None and len(parent) == n:
            p = parent * np.exp(rng.normal(0, 0.3, n))
            return normalize_sum1(p)
        return rng.uniform(0.1, 1.0, n)
    elif kind == 6:
        # Two-lobe
        x = np.linspace(0, 1, n)
        c1, c2 = 0.3, 0.7
        w = 0.15
        v = np.exp(-((x - c1) / w) ** 2) + np.exp(-((x - c2) / w) ** 2)
        return v / v.sum()
    elif kind == 7:
        # Random blocks
        nblocks = rng.integers(3, max(4, n // 50))
        v = np.zeros(n)
        for _ in range(nblocks):
            start = rng.integers(0, n)
            end = min(start + rng.integers(10, n // 10), n)
            v[start:end] += rng.uniform(0.5, 2.0)
        return normalize_sum1(v)
    else:
        return np.ones(n) / n


def propose_candidate(seed=42, budget_s=1500, **kwargs):
    start_time = time.monotonic()
    total_budget = float(budget_s) if budget_s else 1500.0
    if not np.isfinite(total_budget) or total_budget <= 0:
        total_budget = 1500.0
    deadline = start_time + total_budget - 15.0  # Leave 15s margin
    
    # Try to get parent
    parent_arr = None
    try:
        p = globals().get('height_sequence_1', None)
        if p is not None:
            parent_arr = np.asarray(p, dtype=np.float64)
            parent_arr = np.clip(parent_arr, 0, 1000)
            if parent_arr.size > 0 and np.isfinite(parent_arr).all():
                pass
            else:
                parent_arr = None
    except Exception:
        parent_arr = None
    
    best_seq = None
    best_score = float('inf')
    
    if parent_arr is not None:
        p_norm = normalize_sum1(parent_arr)
        best_seq = p_norm.copy()
        best_score = compute_score(p_norm)
        print(f"Parent: len={len(p_norm)}, score={best_score:.6f}", flush=True)
    
    rng = np.random.default_rng(seed)
    
    # Define search plan: (length, time_alloc, use_sym)
    # Focus on lengths where we can make progress
    if total_budget >= 1200:
        plan = [
            (512, 250, True),
            (256, 100, True),
            (1024, 350, True),
            (512, 200, True),
            (2048, 250, True),
            (1024, 200, True),
        ]
    elif total_budget >= 600:
        plan = [
            (512, 150, True),
            (256, 80, True),
            (1024, 200, True),
            (512, 100, True),
        ]
    else:
        plan = [
            (256, 80, True),
            (512, 120, True),
        ]
    
    init_kinds = [0, 1, 2, 3, 4, 5, 6, 7]
    trial = 0
    
    for n, alloc, use_sym in plan:
        remaining = deadline - time.monotonic()
        if remaining < 30:
            break
        alloc = min(alloc, remaining - 20)
        if alloc < 20:
            alloc = max(10, remaining * 0.4)
        
        # Determine how many initializations to try
        n_inits = max(2, int(alloc / 25))
        time_per_init = alloc / n_inits
        
        for ii in range(n_inits):
            rem = deadline - time.monotonic()
            if rem < 10:
                break
            kind = init_kinds[(trial + ii) % len(init_kinds)]
            # For kind 5 (parent-based), only use if parent exists and matches length
            if kind == 5 and parent_arr is not None and len(parent_arr) != n:
                # Resize parent
                from numpy import interp as np_interp
                old_t = np.linspace(0, 1, len(parent_arr))
                new_t = np.linspace(0, 1, n)
                parent_resized = np_interp(new_t, old_t, parent_arr)
                init = make_init(n, 5, rng, parent_resized)
            else:
                init = make_init(n, kind, rng, parent_arr if parent_arr is not None and len(parent_arr) == n else None)
            
            init = normalize_sum1(init)
            sc_init = compute_score(init)
            if np.isfinite(sc_init) and sc_init < best_score:
                best_score = sc_init
                best_seq = init.copy()
            
            # Choose method: coordinate descent (fast) or LP (slower but potentially better)
            time_cd = time_per_init * 0.6
            time_lp = time_per_init * 0.35
            
            # Run coordinate descent
            if time.monotonic() < deadline - 10:
                try:
                    cand, sc = coordinate_descent(init, time_cd, seed + 1000 * trial + 100 * ii)
                    if np.isfinite(sc) and sc < best_score:
                        best_score = sc
                        best_seq = cand.copy()
                        print(f"  CD: n={n} kind={kind} score={sc:.6f}", flush=True)
                except Exception as e:
                    pass
            
            # Run LP polish on the result
            if time.monotonic() < deadline - 10 and time_lp > 5:
                try:
                    start_lp = best_seq.copy() if (best_seq is not None and len(best_seq) == n) else init
                    cand2, sc2 = lp_polish(start_lp, time_lp, seed + 2000 * trial + 50 * ii)
                    if np.isfinite(sc2) and sc2 < best_score:
                        best_score = sc2
                        best_seq = cand2.copy()
                        print(f"  LP: n={n} kind={kind} score={sc2:.6f}", flush=True)
                except Exception as e:
                    pass
            
            trial += 1
    
    # Final deep optimization on the best sequence found
    rem = deadline - time.monotonic()
    if best_seq is not None and rem > 30:
        n_best = len(best_seq)
        print(f"\nFinal polish: n={n_best}, current best={best_score:.6f}, time_left={rem:.1f}s", flush=True)
        
        # Coordinate descent polish
        cd_time = min(rem * 0.5, 200.0)
        if cd_time > 10:
            try:
                cand, sc = coordinate_descent(best_seq, cd_time, seed + 99999)
                if np.isfinite(sc) and sc < best_score:
                    best_score = sc
                    best_seq = cand.copy()
                    print(f"  Final CD: score={sc:.6f}", flush=True)
            except Exception:
                pass
        
        # LP polish
        rem2 = deadline - time.monotonic()
        if rem2 > 15:
            try:
                cand, sc = lp_polish(best_seq, min(rem2 * 0.7, 100.0), seed + 88888)
                if np.isfinite(sc) and sc < best_score:
                    best_score = sc
                    best_seq = cand.copy()
                    print(f"  Final LP: score={sc:.6f}", flush=True)
            except Exception:
                pass
        
        # One more CD round
        rem3 = deadline - time.monotonic()
        if rem3 > 10:
            try:
                cand, sc = coordinate_descent(best_seq, min(rem3 * 0.8, 60.0), seed + 77777)
                if np.isfinite(sc) and sc < best_score:
                    best_score = sc
                    best_seq = cand.copy()
                    print(f"  Final CD2: score={sc:.6f}", flush=True)
            except Exception:
                pass
    
    # Ensure we have a valid sequence
    if best_seq is None:
        best_seq = np.ones(256) / 256.0
        best_score = compute_score(best_seq)
    
    # Final evaluation
    final = normalize_sum1(best_seq)
    try:
        official = float(evaluate_sequence([float(x) for x in final]))
        if np.isfinite(official):
            best_score = official
        else:
            # Fallback to parent
            if parent_arr is not None:
                fp = normalize_sum1(parent_arr)
                try:
                    off2 = float(evaluate_sequence([float(x) for x in fp]))
                    if np.isfinite(off2):
                        final = fp
                        best_score = off2
                except Exception:
                    pass
    except Exception:
        pass
    
    print(f"\nFINAL: score={best_score:.6f}, length={len(final)}", flush=True)
    return [float(x) for x in np.clip(final, 0.0, 1000.0)]
```
