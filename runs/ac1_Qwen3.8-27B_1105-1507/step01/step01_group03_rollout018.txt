def normalize_sequence(arr):
    """Return a nonnegative floating vector normalized to sum 1, or None."""
    try:
        x = np.asarray(arr, dtype=np.float64).ravel()
    except Exception:
        return None
    if x.size == 0:
        return None
    x = np.where(np.isfinite(x), x, 0.0)
    x = np.maximum(x, 0.0)
    s = float(x.sum())
    if s <= 1e-300:
        return None
    x /= s
    return x


def score_normalized(x, n):
    """Score assuming x is nonnegative and sum(x) = 1. Score = 2*n*max(conv)."""
    conv = np.convolve(x, x)
    m = float(np.max(conv))
    if not np.isfinite(m):
        return float("inf")
    return float(2.0 * n * m)


def fft_full_convolve(a, b):
    """Full linear convolution using FFT."""
    n = len(a) + len(b) - 1
    size = 1
    while size < n:
        size *= 2
    A = np.fft.rfft(a, size)
    B = np.fft.rfft(b, size)
    result = np.fft.irfft(A * B, size)

def project_to_simplex(y, target_sum=1.0):
    """Euclidean projection onto {x >= 0, sum(x) = target_sum}."""
    n = len(y)
    if n == 0:
        return y.copy()
    u = np.sort(y)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - target_sum)
    rho_arr = np.nonzero(cond)[0]
    if len(rho_arr) == 0:
        return np.full(n, target_sum / n)
    rho = rho_arr[-1]
    theta = (cssv[rho] - target_sum) / (rho + 1.0)
    return np.maximum(y - theta, 0.0)


def compute_lse_gradient(f, alpha):
    """
    Compute gradient of (1/alpha)*log(sum_k exp(alpha*C_k)) where C = f*f.
    Returns (grad, w) where grad has length len(f) and w has the softmax weights.
    """
    n = len(f)
    C = fft_full_convolve(f, f)  # length 2n-1

    # Numerically stable softmax
    alpha_C = alpha * C
    alpha_C -= alpha_C.max()
    exp_alpha_C = np.exp(alpha_C)
    w = exp_alpha_C / exp_alpha_C.sum()

    wf = fft_full_convolve(w, f_rev)  # length (2n-1)+(n-1) = 3n-2
    grad = 2.0 * wf[n - 1:2 * n - 1]

def make_conv_template(n):
    """Precompute CSR matrix structure for convolve constraints."""
    m = 2 * n - 1
    k64 = np.arange(m, dtype=np.int64)
    js = np.maximum(0, k64 - n + 1).astype(np.int32)
    je = np.minimum(n - 1, k64).astype(np.int32)
    counts = (je.astype(np.int64) - js.astype(np.int64) + 1).astype(np.int64)
    total = int(counts.sum())

    cum = np.cumsum(counts)
    starts = np.repeat(cum - counts, counts)
    local = np.arange(total, dtype=np.int64) - starts

    row = np.repeat(k64, counts).astype(np.int32)
    col_base = js[row.astype(np.int64)]
    col = (col_base + local).astype(np.int32)
    offsets = (row.astype(np.int64) - col.astype(np.int64)).astype(np.int32)

    A_template = csr_matrix((offsets.astype(np.float64), (row, col)), shape=(m, n), dtype=np.float64)
    A_template.data.fill(0.0)
    edge_offsets = A_template.data.copy().astype(np.int32)

    c_obj = -np.ones(n, dtype=np.float64)
    bounds = [(0.0, None)] * n
    return A_template, edge_offsets, c_obj, bounds, m


def solve_direction_lp(template, f, n, M):
    """Solve the LP direction problem."""
    A_template, edge_offsets, c_obj, bounds, m = template
    lp_scale = float(n)

    A = A_template.copy()
    A.data[:] = f[edge_offsets]
    A.data *= lp_scale

    b = np.full(m, M * lp_scale * lp_scale, dtype=np.float64)
    return linprog(c_obj, A_ub=A, b_ub=b, bounds=bounds, method="highs")

                break
            continue

        x = np.asarray(x, dtype=np.float64).ravel()
        if x.size != n:
            break
        x = np.where(np.isfinite(x), x, 0.0)
        x = np.maximum(x, 0.0)
        s_star = float(x.sum())
        if s_star <= 1e-300:
            stall += 1
            if stall > 8:
                break
            if rng is not None:
                f = best_f + rng.normal(0, 0.005, n)
                f = np.maximum(f, 0)
                f /= f.sum()
            else:
                break
            continue

        s_ratio = s_star / float(n)
        if s_ratio <= 1.0 + 1e-8:
            stall += 1
            if stall > 8:
                break
            if rng is not None:
                f = best_f + rng.normal(0, 0.005, n)
                f = np.maximum(f, 0)
                f /= f.sum()
            else:
                break
            continue

        g = x / s_star

        r = np.convolve(f, g)
        r_max = float(np.max(r))
        if (not np.isfinite(r_max)) or r_max <= 0.0:
            stall += 1
            if stall > 8:
                break
            continue

        if r_max >= M * (1.0 - 1e-10):
            stall += 1
            if stall > 8:
                break
            continue

        S_eff = M / r_max
        if S_eff <= 1.0 + 1e-8:
            stall += 1
            if stall > 8:
                break
            continue

        G = float(np.max(np.convolve(g, g)))
        if not np.isfinite(G):
            stall += 1
            if stall > 8:
                break
            continue

        a_coef = 2.0 * (r_max - M)
        b_coef = G + M - 2.0 * r_max
        if b_coef > 1e-14:
            t_max = -a_coef / b_coef
        else:
            t_max = 1.0

        if (not np.isfinite(t_max)) or t_max <= 0.0:
            t_max = 1e-8
        base = min(1.0, 0.8 * t_max)
        if base < 1e-12:
            base = 1e-12

        raw_ts = [base, 0.5 * base, 0.2 * base, 0.05 * base, 0.01 * base, 1e-4, 1e-6]
        ts = []
        last = 2.0
        for tv in sorted(set(float(v) for v in raw_ts), reverse=True):
            if tv < 1e-14 or tv > 1.0:
                continue
            if last - tv < 1e-13:
                continue
            last = tv
            ts.append(tv)
            if len(ts) >= 6:
                break

        best_cand_f = None
        best_cand_score = best_score
        for tv in ts:
            f_new = (1.0 - tv) * f + tv * g
            f_new = np.maximum(f_new, 0.0)
            ss = float(f_new.sum())
            if ss <= 1e-300:
                continue
            f_new /= ss
            sc = score_normalized(f_new, n)
            if np.isfinite(sc) and sc < best_cand_score:
                best_cand_score = sc
                best_cand_f = f_new

        if best_cand_f is not None and best_cand_score < best_score:
            f = best_cand_f
            best_f = f.copy()
            best_score = best_cand_score
            stall = 0
        else:
            stall += 1
            if rng is not None:
                f = best_f + rng.normal(0, 0.01, n)
                f = np.maximum(f, 0)
                f /= f.sum()
            else:
                if stall > 8:
                    break

    return best_f, float(best_score)


def interpolate_sequence(x, n):
    """Resample a nonnegative sequence to length n and normalize."""
    x = normalize_sequence(x)
    if x is None or n <= 0:
        return None
    m = x.size
    if m == n:
        return x
    if m == 1:
        return np.ones(n, dtype=np.float64) / float(n)
    old = np.linspace(0.0, 1.0, m, dtype=np.float64)
    new = np.linspace(0.0, 1.0, n, dtype=np.float64)
    out = np.interp(new, old, x)
    return normalize_sequence(out)


def mix_noisy(base, rng, eps):
    """Mix a base vector with random Dirichlet noise."""
    f = normalize_sequence(base)
    if f is None:
        return None
    n = f.size
    if eps <= 1e-12:
        return f
    mix = rng.dirichlet(np.ones(n, dtype=np.float64))
    if eps >= 1.0:
        f = mix
    else:
        f = (1.0 - eps) * f + eps * mix
    return normalize_sequence(f)


def propose_candidate(seed=42, budget_s=1500, **kwargs):
    """
    Search for a low-scoring nonnegative sequence using multi-phase optimization:
    Phase 1: Fast LSE gradient descent (FFT-based) for exploration
    Phase 2: LP polishing for precise convergence
    """
    rng = np.random.default_rng(int(seed) if seed is not None else 42)
    total_budget = max(1.0, float(budget_s))
    start_time = time.monotonic()
    reserve = 20.0
    deadline = start_time + max(1.0, total_budget - reserve)

    best_f = None
    best_score = float("inf")
    given_f = None

    # Load given sequence if available
    if total_budget >= 30.0:
        try:
            raw = np.asarray(height_sequence_1, dtype=np.float64)
            given_f = normalize_sequence(raw)
            if given_f is not None:
                best_f = given_f.copy()
                best_score = score_normalized(given_f, given_f.size)
                print(f"Given sequence: n={given_f.size}, score={best_score:.6f}", flush=True)
        except Exception:
            given_f = None

    if best_f is None:
        n0 = 1024
        best_f = np.ones(n0, dtype=np.float64) / n0
        best_score = 2.0

    # ===================== PHASE 1: LSE Gradient Descent =====================
    # Allocate ~55% of time to exploration
    phase1_budget = (deadline - time.monotonic()) * 0.55
    print(f"\n=== Phase 1: LSE Gradient Descent (budget={phase1_budget:.0f}s) ===", flush=True)

    # Try multiple lengths
    lengths = [512, 1024, 2048]
    if total_budget > 1200:
        lengths.append(3072)

    # Allocate time proportional to length (larger = more potential)
    total_len_weight = sum(l ** 0.7 for l in lengths)
    lse_results = []  # list of (f, score, n)

    for n in lengths:
        rem = deadline - time.monotonic()
        if rem < 30.0:
            break

        weight = (n ** 0.7) / total_len_weight
        alloc = phase1_budget * weight
        alloc = min(alloc, rem - 10.0)
        if alloc < 15.0:
            continue

        print(f"\n  LSE n={n}, alloc={alloc:.0f}s", flush=True)

        # Generate diverse initial points
        inits = []

        # From best found so far (interpolated)
        if best_f is not None:
            up = interpolate_sequence(best_f, n)
            if up is not None:
                inits.append(up)
                inits.append(mix_noisy(up, rng, 0.05))
                inits.append(mix_noisy(up, rng, 0.15))

        # From given sequence
        if given_f is not None:
            g = interpolate_sequence(given_f, n)
            if g is not None:
                inits.append(g)
                inits.append(mix_noisy(g, rng, 0.1))

        # Random starts
        inits.append(mix_noisy(np.ones(n, dtype=np.float64), rng, 1.0))
        inits.append(mix_noisy(rng.dirichlet(np.ones(n, dtype=np.float64)), rng, 0.0))

        # Structured: cosine taper (high in middle)
        t = np.linspace(0, np.pi, n)
        cos_shape = np.maximum(np.cos(t), 0.1)
        inits.append(normalize_sequence(cos_shape))

        # Structured: U-shape (high at edges)
        t = np.linspace(0, 1, n)
        u_shape = 0.3 + 0.7 * (2 * t - 1) ** 2
        inits.append(normalize_sequence(u_shape))

        # Remove duplicates and limit
        unique_inits = []
        for init in inits:
            if init is None:
                continue
            is_dup = False
            for u in unique_inits:
                if np.linalg.norm(init - u) < 1e-5:
                    is_dup = True
                    break
            if not is_dup:
                unique_inits.append(init)

        # Limit to max 5 starts per length
        unique_inits = unique_inits[:5]

        # Run LSE from each start with equal time split
        per_start_budget = alloc / len(unique_inits)

        for idx, init in enumerate(unique_inits):
            rem = deadline - time.monotonic()
            if rem < 5.0:
                break
            tb = min(per_start_budget, rem - 2.0)
            if tb < 3.0:
                continue

            print(f"    start {idx+1}/{len(unique_inits)}, budget={tb:.1f}s", flush=True)

            try:
                f_res, s_res = lse_optimize(
                    init, n, tb,
                    alpha_start=5.0,
                    alpha_end=300.0,
                    lr_init=0.02,
                    momentum_init=0.9,
                    rng=rng
                )
            except Exception as e:
                print(f"    LSE error: {e}", flush=True)
                continue

            if np.isfinite(s_res):
                lse_results.append((f_res, s_res, n))
                if s_res < best_score:
                    best_score = s_res
                    best_f = f_res.copy()
                    print(f"    NEW BEST: n={n}, score={s_res:.6f}", flush=True)

    # Sort LSE results by score
    lse_results.sort(key=lambda x: x[1])
    print(f"\n  LSE Phase complete. Top results:", flush=True)
    for i, (f, s, n) in enumerate(lse_results[:5]):
        print(f"    #{i+1}: n={n}, score={s:.6f}", flush=True)

    # ===================== PHASE 2: LP Polishing =====================
    phase2_budget = deadline - time.monotonic()
    print(f"\n=== Phase 2: LP Polishing (budget={phase2_budget:.0f}s) ===", flush=True)

    # Take top unique results for LP polishing
    lp_candidates = []
    seen_ns = set()
    for f, s, n in lse_results:
        if n not in seen_ns or s < 1.52:
            lp_candidates.append((f, n))
            seen_ns.add(n)
        if len(lp_candidates) >= 4:
            break

    # Also always include the current best
    if best_f is not None:
        best_n = best_f.size
        if not any(n == best_n for _, n in lp_candidates):
            lp_candidates.append((best_f.copy(), best_n))

    # Also try from the given sequence directly
    if given_f is not None:
        gf_n = given_f.size
        if not any(n == gf_n for _, n in lp_candidates):
            lp_candidates.append((given_f.copy(), gf_n))

    # Allocate time to LP runs based on length
    total_lp_weight = sum(n ** 0.5 for _, n in lp_candidates)
    for f_init, n in lp_candidates:
        rem = deadline - time.monotonic()
        if rem < 10.0:
            break

        weight = (n ** 0.5) / total_lp_weight
        alloc = phase2_budget * weight
        alloc = min(alloc, rem - 5.0)
        if alloc < 5.0:
            continue

        print(f"\n  LP n={n}, alloc={alloc:.0f}s, init_score={score_normalized(f_init, n):.6f}", flush=True)

        try:
            template = make_conv_template(n)
        except MemoryError:
            print(f"  MemoryError for n={n}, skipping", flush=True)
            continue
        except Exception as e:
            print(f"  Template error: {e}", flush=True)
            continue

        try:
            f_res, s_res = lp_polish(f_init, n, alloc, template, rng=rng)
        except Exception as e:
            print(f"  LP error: {e}", flush=True)
            f_res, s_res = f_init, score_normalized(f_init, n)

        if np.isfinite(s_res) and s_res < best_score:
            best_score = s_res
            best_f = f_res.copy()
            print(f"  LP NEW BEST: n={n}, score={s_res:.6f}", flush=True)

        try:
            del template
        except Exception:
            pass

    # ===================== PHASE 3: Final LSE refinement on best =====================
    rem = deadline - time.monotonic()
    if rem > 30.0 and best_f is not None:
        print(f"\n=== Phase 3: Final LSE refinement (budget={rem:.0f}s) ===", flush=True)
        n_best = best_f.size
        try:
            f_res, s_res = lse_optimize(
                best_f, n_best, min(rem - 5.0, 120.0),
                alpha_start=50.0, alpha_end=500.0,
                lr_init=0.005, momentum_init=0.95, rng=rng
            )
            if np.isfinite(s_res) and s_res < best_score:
                best_score = s_res
                best_f = f_res.copy()
                print(f"  Phase 3 NEW BEST: n={n_best}, score={s_res:.6f}", flush=True)
        except Exception as e:
            print(f"  Phase 3 error: {e}", flush=True)

    # Final LP polish if time remains
    rem = deadline - time.monotonic()
    if rem > 30.0 and best_f is not None:
        n_best = best_f.size
        try:
            template = make_conv_template(n_best)
            f_res, s_res = lp_polish(best_f, n_best, min(rem - 10.0, 300.0), template, rng=rng)
            if np.isfinite(s_res) and s_res < best_score:
                best_score = s_res
                best_f = f_res.copy()
                print(f"  Final LP NEW BEST: score={s_res:.6f}", flush=True)
            del template
        except Exception as e:
            print(f"  Final LP error: {e}", flush=True)

    # ===================== Return best =====================
    if best_f is None:
        best_f = np.ones(64, dtype=np.float64) / 64.0

    seq = [float(max(0, min(1000.0, x))) for x in best_f]
    final_score = float(evaluate_sequence(seq))

    if not np.isfinite(final_score):
        if given_f is not None:
            seq = [float(max(0, min(1000.0, x))) for x in given_f]
            final_score = float(evaluate_sequence(seq))

    if not np.isfinite(final_score):
        seq = [1.0 / 64.0] * 64
        final_score = float(evaluate_sequence(seq))

    elapsed = time.monotonic() - start_time
    print(f"\nFINAL: n={len(seq)}, local_best={best_score:.6f}, "
          f"evaluate={final_score:.6f}, time={elapsed:.1f}s", flush=True)
    return seq
