"""
Optimizes non-negative step function heights to minimize 2n*max(conv(f,f))/(sum(f))^2.
Uses asymmetric-focused multi-start mirror descent with adaptive cooling,
LP-based Frank-Wolfe refinement, and coordinate-wise local search.
Key insight: symmetric sequences cannot beat score 2, so we focus on asymmetric
initializations and preserve asymmetry during optimization.

import numpy as np
import time
from scipy.optimize import linprog
from scipy.sparse import coo_matrix


def _fft_convolve_full(f):
    """Compute full linear convolution of f with itself using FFT."""
    n = len(f)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    F = np.fft.rfft(f, size)
    h = np.fft.irfft(F * F, size)
    return h[:2 * n - 1]


def _score_fast(f):
    """Fast score assuming f >= 0, sum(f) > 0."""
    n = len(f)
    h = _fft_convolve_full(f)
    M = float(np.max(h))
    S = float(np.sum(f))
    if S < 0.01:
        return float('inf')
    return 2.0 * n * M / (S * S)


def _normalize(f):
    """Normalize f to sum 1, clip negatives."""
    f = np.maximum(f, 0.0)
    s = float(np.sum(f))
    if s < 1e-30:
        return None
    return f / s


def _pressure(f, active):
    """Compute average pressure vector from active lags."""
    n = len(f)
    p = np.zeros(n, dtype=np.float64)
    for k in active:
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        if j_end >= j_start:
            lo = k - j_end
            hi = k - j_start
            p[j_start:j_end + 1] += 2.0 * f[lo:hi + 1][::-1]
    p /= float(len(active))
    return p


def _mirror_descent(f_init, n_iter, eta_scale, thresh_start, thresh_end, label=""):
    """
    Mirror descent with multiplicative updates, adaptive threshold cooling,
    and step-size annealing. Returns (best_f, best_score).
    """
    f = np.asarray(f_init, dtype=np.float64).copy()
    f = _normalize(f)
    if f is None:
        return f_init.copy(), float('inf')
    n = len(f)
    best_f = f.copy()
    best_score = _score_fast(f)

    for it in range(n_iter):
        h = _fft_convolve_full(f)
        M = float(np.max(h))
        if M <= 1e-30:
            break

        # Linearly interpolate threshold fraction
        frac = thresh_start + (thresh_end - thresh_start) * (it / max(1, n_iter - 1))
        threshold = frac * M
        active = np.where(h >= threshold)[0]
        if len(active) == 0:
            active = np.array([int(np.argmax(h))], dtype=np.int64)

        p = _pressure(f, active)

        # Step size with cooling
        eta = eta_scale / M
        eta *= (1.0 - 0.6 * (it / max(1, n_iter - 1)))

        log_ratio = np.clip(-eta * p, -20.0, 20.0)
        f_new = f * np.exp(log_ratio)
        f_norm = _normalize(f_new)
        if f_norm is None:
            f_norm = f.copy()
        else:
            f_new = f_norm

        score_new = _score_fast(f_new)
        if score_new < best_score:
            best_score = score_new
            best_f = f_new.copy()

        # Accept new iterate (always, to allow exploration)
        f = f_new

        if (it + 1) % 2000 == 0:
            print(f"    MD[{label}] it={it+1}: best={best_score:.8f}, M={M:.10f}, |K|={len(active)}")

    return best_f, best_score


def _lp_refine(f_init, max_iters, time_budget, label=""):
    """LP-based Frank-Wolfe refinement."""
    f = np.asarray(f_init, dtype=np.float64).copy()
    f = _normalize(f)
    if f is None:
        return f_init.copy(), float('inf')
    n = len(f)
    best_f = f.copy()
    best_score = _score_fast(f)
    start_time = time.monotonic()

    for it in range(max_iters):
        if time.monotonic() - start_time > time_budget:
            break

        h = np.convolve(f, f)
        M = float(np.max(h))

        # Build sparse constraint matrix
        row_idx = []
        col_idx = []
        vals = []
        for j in range(n):
            rows_j = range(j, j + n)
            row_idx.extend(rows_j)
            col_idx.extend([j] * n)
            vals.extend([f[j]] * n)

        C = coo_matrix((vals, (row_idx, col_idx)), shape=(2 * n - 1, n))
        c_obj = -np.ones(n)
        res = linprog(c_obj, A_ub=C, b_ub=np.full(2 * n - 1, M),
                      bounds=[(0, None)] * n, method='highs')

        if not res.success or res.x is None:
            break

        g = np.maximum(res.x, 0.0)
        Sg = float(np.sum(g))
        if Sg < 1e-10:
            break
        g = g / Sg

        # Frank-Wolfe step with decay
        t = 2.0 / (it + 2)
        f_new = (1.0 - t) * f + t * g
        f_n = _normalize(f_new)
        if f_n is None:
            break
        f = f_n

        score = _score_fast(f)
        if score < best_score:
            best_score = score
            best_f = f.copy()

        if (it + 1) % 5 == 0:
            rem = time_budget - (time.monotonic() - start_time)
            print(f"    LP[{label}] it={it+1}: score={best_score:.8f}, budget_left={rem:.1f}s")

    return best_f, best_score


def _local_search(f_init, n_rounds, rng, label=""):
    """Coordinate-wise and pairwise local search."""
    f = np.asarray(f_init, dtype=np.float64).copy()
    n = len(f)
    best_f = f.copy()
    best_score = _score_fast(f)

    for rnd in range(n_rounds):
        improved = False

        # Coordinate moves: try shifting mass from one position to another
        for _ in range(n * 3):
            i = rng.integers(0, n)
            j = rng.integers(0, n)
            if i == j:
                continue
            frac = rng.choice([0.01, 0.02, 0.05, 0.1])
            amount = f[i] * frac
            if amount < 1e-10:
                continue

            f_trial = f.copy()
            f_trial[i] -= amount
            f_trial[j] += amount
            f_trial = np.maximum(f_trial, 0.0)
            s = np.sum(f_trial)
            if s < 1e-10:
                continue
            f_trial = f_trial / s

            score = _score_fast(f_trial)
            if score < best_score - 1e-10:
                f = f_trial
                best_f = f_trial.copy()
                best_score = score
                improved = True

        if not improved and rnd > 0:
            break

        if (rnd + 1) % 5 == 0:
            print(f"    LS[{label}] round={rnd+1}: score={best_score:.8f}")

    return best_f, best_score

    """Asymmetric Gaussian centered at center_frac*n."""
    j = np.arange(n, dtype=np.float64)
    c = center_frac * n
    sigma = sigma_frac * n
    f = np.exp(-(j - c) ** 2 / (2.0 * sigma * sigma))
    return _normalize(f)


def _asym_ramp(n, power, direction=1):
    """Asymmetric power ramp."""
    j = np.arange(n, dtype=np.float64)

    else:
        f = (n - j) ** power
    return _normalize(f)

            f_ref2, s_ref2 = _mirror_descent(f_cur, n_it3, 0.1, 0.85, 0.99, label=name + "_fine")
            if s_ref2 < best_score:
                best_score = s_ref2
                best_f = f_ref2.copy()
                print(f"  ** New best: {best_score:.8f} **")
            f_cur = f_ref2 if s_ref2 < s_cur else f_cur
            s_cur = min(s_cur, s_ref2)

        # LP refinement (only if n is small enough)
        n_eff = len(f_cur)
        if n_eff <= 1500 and (deadline - time.monotonic()) > 30:
            lp_budget = min(90.0, deadline - time.monotonic() - 15)
            if lp_budget > 10:
                f_lp, s_lp = _lp_refine(f_cur, max_iters=30, time_budget=lp_budget,
                                        label=name + "_lp")
                if s_lp < best_score:
                    best_score = s_lp
                    best_f = f_lp.copy()
                    print(f"  ** New best from LP: {best_score:.8f} **")
                f_cur = f_lp if s_lp < s_cur else f_cur
                s_cur = min(s_cur, s_lp)

        # Local search on the best version
        if (deadline - time.monotonic()) > 20:
            ls_budget_rounds = int(min(20, (deadline - time.monotonic() - 10) / 2.0))
            if ls_budget_rounds > 2:
                f_ls, s_ls = _local_search(f_cur, ls_budget_rounds, rng, label=name + "_ls")
                if s_ls < best_score:
                    best_score = s_ls
                    best_f = f_ls.copy()
                    print(f"  ** New best from LS: {best_score:.8f} **")

    # Phase 3: Try different sequence lengths with the best structure
    if time.monotonic() < deadline - 100:
        print("\n  Phase 3: Trying different lengths...")
        best_n = len(best_f)
        for n_try in [best_n * 2, best_n // 2, best_n + 128, best_n - 128]:
            if n_try < 64 or n_try > 8192:
                continue
            if time.monotonic() > deadline - 80:
                break

            # Upsample or downsample best_f to n_try
            if n_try > best_n:
                # Upsample: insert interpolated values
                x_old = np.linspace(0, 1, best_n)
                x_new = np.linspace(0, 1, n_try)
                f_try = np.interp(x_new, x_old, best_f)
            else:
                # Downsample: average blocks
                blocks = best_n // n_try
                f_try = best_f[:blocks * n_try].reshape(n_try, blocks).mean(axis=1)

            f_try = _normalize(f_try)
            if f_try is None:
                continue

            n_it = int(min(5000, (deadline - time.monotonic() - 30) * 300))
            f_up, s_up = _mirror_descent(f_try, n_it, 0.4, 0.6, 0.93, label=f"len_{n_try}")
            if s_up < best_score:
                best_score = s_up
                best_f = f_up.copy()
                print(f"  ** New best (n={n_try}): {best_score:.8f} **")

    # Phase 4: Final fine-tuning on the global best
    if time.monotonic() < deadline - 20:
        print("\n  Phase 4: Final fine-tuning...")
        n_it_final = int(min(15000, (deadline - time.monotonic() - 10) * 600))
        f_final, s_final = _mirror_descent(best_f, n_it_final, 0.05, 0.9, 0.999, label="final")
        if s_final < best_score:
            best_score = s_final
            best_f = f_final.copy()
            print(f"  ** Final best: {best_score:.8f} **")

        # One more LP pass if time permits
        if len(best_f) <= 1500 and (deadline - time.monotonic()) > 30:
            f_final_lp, s_final_lp = _lp_refine(best_f, max_iters=20,
                                                time_budget=deadline - time.monotonic() - 10,
                                                label="final_lp")
            if s_final_lp < best_score:
                best_score = s_final_lp
                best_f = f_final_lp.copy()
                print(f"  ** Final LP best: {best_score:.8f} **")

        # Final local search
        if (deadline - time.monotonic()) > 15:
            f_final_ls, s_final_ls = _local_search(best_f, 10, rng, label="final_ls")
            if s_final_ls < best_score:
                best_score = s_final_ls
                best_f = f_final_ls.copy()
                print(f"  ** Final LS best: {best_score:.8f} **")

    # Final verification
    final_list = [float(x) for x in best_f]
    final_score = float(evaluate_sequence(final_list))
    print(f"\nFinal verified score: {final_score:.8f} (n={len(final_list)})")

    if not np.isfinite(final_score) or final_score > best_score + 0.01:
        print("  Score verification failed, using best_f directly")
        final_score = best_score

    if not np.isfinite(final_score) or final_score > 2.0:
        print("  Falling back to parent.")
        try:
            return [float(x) for x in np.clip(parent, 0, 1000)]
        except Exception:
            return [1.0] * 512

    return final_list
