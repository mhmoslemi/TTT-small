def objective_value(a):
    """Compute the objective 2n*max(conv(a,a))/sum(a)^2 without calling evaluate_sequence."""
    n = len(a)
    s = np.sum(a)
    if s < 0.01:
        return np.inf
    conv_a = np.convolve(a, a)
    max_b = np.max(conv_a)
    return float(2 * n * max_b / (s ** 2))


def project_simplex(a, S):
    """Project a onto {x: x >= 0, sum(x) = S}."""
    a = np.clip(a, 0, None)
    s = np.sum(a)
    if s <= 0:
        return np.full(len(a), S / len(a))
    return a * (S / s)


def subgradient_step(a, S, eta, momentum=0.9):

    n = len(a)

    threshold = 0.9 * max_val

        indices = k - j_range
        grad[j_range] += 2.0 * a[indices]
    
    grad /= len(active_lags)
    
    return grad

Multi-phase optimization for minimizing 2n*max(conv(a,a))/sum(a)^2.
Phase 1: Soft-max subgradient with momentum (fast, many iterations)
Phase 2: Hard-max subgradient with careful step size (fine-tuning)
Phase 3: LP refinement (polish)
Tries multiple sequence lengths and initializations.

import numpy as np
import time
from scipy.optimize import linprog
from scipy.sparse import csr_matrix, coo_matrix


def fast_objective(a):
    """Compute 2n*max(conv(a,a))/sum(a)^2 quickly."""
    n = len(a)
    s = np.sum(a)
    if s < 1e-10:
        return np.inf
    conv_a = np.convolve(a, a)
    return float(2 * n * np.max(conv_a) / (s * s))


def project_simplex(a, S):
    """Project onto {x >= 0, sum(x) = S}."""
    a = np.maximum(a, 0)
    s = np.sum(a)
    if s <= 1e-15:
        return np.full(len(a), S / len(a))
    return a * (S / s)


def soft_max_grad(a, weights, idx_matrix):
    """Compute soft-max gradient using precomputed index matrix."""
    n = len(a)
    W = weights[idx_matrix]
    return 2.0 * (a @ W.T)


def run_soft_max_opt(f, S, deadline, T_init=5.0, T_final=200.0, momentum=0.85,
                     lr_init=1.0, verbose_prefix=""):
    """Optimize using soft-max subgradient with momentum."""
    f = f.astype(np.float64).copy()
    f = project_simplex(f, S)
    n = len(f)
    
    # Precompute index matrix
    idx = np.arange(n)[:, None] + np.arange(n)[None, :]
    idx_matrix = idx  # idx_matrix[j, m] = j + m
    
    best_f = f.copy()
    best_score = fast_objective(f)
    velocity = np.zeros(n)
    
    total_iters = 500000
    T_current = T_init
    lr = lr_init
    no_improve = 0
    
    for it in range(total_iters):
        if time.monotonic() > deadline:
            break
        
        # Anneal T
        frac = min(1.0, it / (total_iters * 0.7))
        T_current = T_init + (T_final - T_init) * frac
        
        # Compute conv and soft-max weights
        conv_a = np.convolve(f, f)
        max_conv = np.max(conv_a)
        shifted = T_current * (conv_a - max_conv)
        exp_shifted = np.exp(shifted)
        weights = exp_shifted / np.sum(exp_shifted)
        
        # Gradient
        grad = soft_max_grad(f, weights, idx_matrix)
        
        # Momentum update
        velocity = momentum * velocity - lr * grad
        f_new = f + velocity
        f_new = project_simplex(f_new, S)
        
        # Evaluate
        score = fast_objective(f_new)
        
        if score < best_score:
            best_score = score
            best_f = f_new.copy()
            f = f_new
            no_improve = 0
            if (it + 1) % 10000 == 0:
                print(f"    {verbose_prefix}iter {it+1}: score={best_score:.6f}, T={T_current:.1f}, lr={lr:.4f}")
        else:
            no_improve += 1
            if no_improve % 100 == 0:

            if no_improve > 2000 or lr < 1e-10:

    return best_f, best_score


def run_hard_max_opt(f, S, deadline, verbose_prefix=""):

    f = f.astype(np.float64).copy()

    n = len(f)

def build_conv_matrix(f):
    """Build sparse convolution matrix."""
    n = len(f)
    rows, cols, vals = [], [], []
    for k in range(2 * n - 1):
        j_min = max(0, k - n + 1)
        j_max = min(k + 1, n)
        for j in range(j_min, j_max):
            rows.append(k)
            cols.append(j)
            vals.append(f[k - j])
    return csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))


def lp_refine(f, deadline, max_iters=50, verbose_prefix=""):
    """LP-based refinement."""
    f = f.astype(np.float64).copy()
    n = len(f)
    S = np.sum(f)
    f = project_simplex(f, S)
    
    best_f = f.copy()
    best_score = fast_objective(f)
    
    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        
        conv_f = np.convolve(f, f)
        M = np.max(conv_f)
        
        if M <= 0:
            break
        
        # Build LP: max sum(g) s.t. conv(f, g) <= M, g >= 0
        A = build_conv_matrix(f)
        c = -np.ones(n)
        b_ub = np.full(2 * n - 1, M)
        bounds = [(0, None)] * n
        
        try:
            result = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                            options={'time_limit': max(1, (deadline - time.monotonic()) / max(1, max_iters - it) - 0.5)})
            if result.success:
                g = result.x
                sum_g = np.sum(g)
                if sum_g > 1e-15:
                    g = g * (S / sum_g)
                    # Try multiple step sizes
                    for t in [0.3, 0.15, 0.05, 0.02]:
                        f_new = (1 - t) * f + t * g
                        f_new = project_simplex(f_new, S)
                        score = fast_objective(f_new)
                        if score < best_score:
                            best_score = score
                            best_f = f_new.copy()
                            f = f_new
                            break
        except Exception:
            pass
    
    return best_f, best_score
