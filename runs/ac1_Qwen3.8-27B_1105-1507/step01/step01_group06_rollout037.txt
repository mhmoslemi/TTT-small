def project_simplex(v, s=1.0):
    """Project v onto the simplex {x >= 0, sum(x) = s}."""
    n = len(v)
    if n == 0:
        return v

    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n+1) > (cssv - s))[0][-1]
    theta = (cssv[rho] - s) / (rho + 1.0)

3. Diverse asymmetric initializations (exponential, off-center, power-law)
4. L-BFGS-B gradient refinement at multiple lengths
5. Final LP polish with tighter tolerances
"""

import numpy as np
from scipy.optimize import linprog, minimize
import time


def fast_conv(a):
    """Full self-convolution via FFT, O(n log n)."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size *= 2
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]


def build_constraint_matrix(a):
    """Build dense matrix A[2n-1, n] where A[k,j] = a[k-j] if 0<=k-j<n, else 0."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        length = min(n, 2 * n - 1 - j)
        A[j:j + length, j] = a[:length]
    return A


def solve_lp(a, M):
    """Solve LP: max sum(g) s.t. (a*g)[k] <= M for all k, g >= 0. Returns g or None."""
    n = len(a)
    A = build_constraint_matrix(a)
    c = -np.ones(n)
    b = np.full(2 * n - 1, M)
    res = linprog(c, A_ub=A, b_ub=b, bounds=[(0, None)] * n, method='highs')
    if res.success and res.x is not None:
        g = np.asarray(res.x, dtype=np.float64)
        if g.sum() > 1e-15:
            return g
    return None


def normalized_objective(a):
    """Compute max(convolve(a,a)) assuming sum(a)=1."""
    s = a.sum()
    if s < 1e-15:
        return 1e10
    return float(fast_conv(a / s).max())


def gradient_objective(x):
    """Objective for L-BFGS-B: max convolution peak of normalized non-negative x."""
    ax = np.maximum(x, 0.0)
    s = ax.sum()
    if s < 1e-10:
        return 1e10
    return float(fast_conv(ax / s).max())


def interpolate_sequence(src, target_n):
    """Interpolate a 1D sequence to a new length."""
    if src.size == target_n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, target_n)
    return np.maximum(np.interp(t_new, t_old, src), 0)


def make_inits(n, rng, previous_best=None):
    """Generate diverse initializations emphasizing asymmetric shapes."""
    inits = []
    i = np.arange(n, dtype=np.float64)
    # Uniform
    inits.append(np.ones(n))
    # Exponential decays (strongly asymmetric)
    for rate in [0.005, 0.01, 0.02, 0.05, 0.1, 0.2, 0.4]:
        inits.append(np.exp(-rate * i))
        inits.append(np.exp(-rate * (n - 1 - i)))
    # Off-center Gaussians
    for mu_frac in [0.2, 0.3, 0.5, 0.7, 0.8]:
        for sigma_frac in [0.05, 0.1, 0.15, 0.25]:
            mu = mu_frac * (n - 1)
            sigma = max(1.0, sigma_frac * n)
            inits.append(np.exp(-((i - mu) / sigma) ** 2))
    # Ramps
    inits.append(i + 1.0)
    inits.append(n - i)
    # Power laws
    for alpha in [0.3, 0.5, 1.0, 1.5, 2.0]:
        inits.append((i + 1.0) ** (-alpha))
        inits.append((n - i + 1.0) ** (-alpha))
    # Sine taper (symmetric)
    inits.append(np.sin(np.pi * (i + 0.5) / n))
    # Product of ramp and exponential
    inits.append((i + 1.0) * np.exp(-0.05 * i))
    inits.append((n - i) * np.exp(-0.05 * (n - 1 - i)))
    # Random
    inits.append(rng.uniform(0.05, 1.0, n))
    # Triangular (asymmetric)
    peak = int(0.3 * n)
    inits.append(np.minimum(i + 1, n - i + 1))
    peak_pos = int(0.6 * n)
    tri = np.where(i < peak_pos, i + 1, n - i)
    inits.append(np.maximum(tri, 0.1))
    # From previous best
    if previous_best is not None:
        inits.append(interpolate_sequence(previous_best, n))
    return inits


def lp_refine(a, deadline, max_iters, use_momentum=True):
    """Run LP best-response iterations with optional momentum. Returns refined a."""
    a = np.asarray(a, dtype=np.float64).copy()
    n = len(a)
    a = np.clip(a, 1e-8, None)
    a = a / a.sum()
    v = np.zeros(n) if use_momentum else None
    prev_obj = normalized_objective(a)
    best_local = a.copy()
    best_local_obj = prev_obj
    beta = 0.7 if use_momentum else 0.0

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        b = fast_conv(a)
        M = float(b.max())
        g = solve_lp(a, M)
        if g is None:
            break
        g = g / g.sum()

        # Adaptive step: larger early, smaller late
        progress = it / max(max_iters - 1, 1)
        t_mix = 0.5 * (1.0 - 0.5 * progress)

        if use_momentum:
            direction = g - a
            v = beta * v + direction
            a_new = a + t_mix * v
        else:
            a_new = (1.0 - t_mix) * a + t_mix * g

        a_new = np.clip(a_new, 0.0, None)
        s = a_new.sum()
        if s < 1e-10:
            a_new = (1.0 - t_mix) * a + t_mix * g
            a_new = np.clip(a_new, 0.0, None)
            s = a_new.sum()
            if s < 1e-10:
                break
        a = a_new / s

        # Track local best
        obj = normalized_objective(a)
        if obj < best_local_obj - 1e-15:
            best_local_obj = obj
            best_local = a.copy()

        # Early stop if no improvement for many iterations
        if it > 50 and abs(obj - prev_obj) < 1e-14:
            break
        prev_obj = obj

    return best_local


def update_state(state, a):
    """Evaluate candidate and update state dict if improved."""
    a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
    if a.size == 0 or a.sum() < 0.01:
        return
    score = float(evaluate_sequence(a.tolist()))
    if np.isfinite(score) and score < state['best_score']:
        state['best'] = a.copy()
        state['best_score'] = score
        print(f"  NEW BEST: J={score:.6f} (n={a.size})", flush=True)


def propose_candidate(seed=42, budget_s=1500, **kwargs):
    """
    Search for the best non-negative sequence minimizing J(a) = 2n*max(conv(a,a))/sum(a)^2.
    Uses LP waterfall iterations, multi-scale growth, and gradient refinement.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + float(budget_s) - 15.0
    state = {'best': None, 'best_score': float('inf')}

    # Phase 0: Try parent
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        if p.ndim == 1 and p.size > 0 and np.all(np.isfinite(p)):
            update_state(state, p)
    except (NameError, AttributeError):
        pass

    print(f"Starting search. Deadline in {deadline - time.monotonic():.0f}s", flush=True)
    if state['best'] is not None:
        print(f"  Parent score: {state['best_score']:.6f}", flush=True)

    # Phase 1: Multi-scale LP refinement
    scales = [256, 512, 1024, 2048]
    for target_n in scales:
        if time.monotonic() > deadline:
            break
        elapsed = time.monotonic() - (deadline - float(budget_s) + 15.0)
        remaining = float(budget_s) - 15.0 - elapsed
        print(f"\n=== LP Scale n={target_n} (remaining ~{remaining:.0f}s) ===", flush=True)

        inits = make_inits(target_n, rng, previous_best=state['best'])

        # Determine how many inits to try based on remaining time
        max_iters = max(50, int(min(300, remaining / max(len(inits), 1) / 0.3)))
        # For larger n, fewer inits but more iterations each
        if target_n <= 512:
            num_inits = min(len(inits), 20)
            max_iters = min(max_iters, 400)
        elif target_n <= 1024:
            num_inits = min(len(inits), 12)
            max_iters = min(max_iters, 250)
        else:
            num_inits = min(len(inits), 8)
            max_iters = min(max_iters, 150)

        for idx in range(num_inits):
            if time.monotonic() > deadline:
                break
            a0 = inits[idx]
            a0 = np.clip(a0, 1e-8, None)
            s0 = a0.sum()
            if s0 < 1e-10:
                continue
            a0 = a0 / s0
            update_state(state, a0)

            t_start = time.monotonic()
            a_refined = lp_refine(a0, deadline, max_iters, use_momentum=True)
            elapsed_this = time.monotonic() - t_start
            update_state(state, a_refined)

            # Also try without momentum (pure convex combination)
            if time.monotonic() < deadline and elapsed_this < remaining * 0.1:
                a_refined2 = lp_refine(a0, deadline, max(50, max_iters // 2), use_momentum=False)
                update_state(state, a_refined2)

    # Phase 2: Gradient refinement
    if state['best'] is not None and time.monotonic() < deadline:
        print(f"\n=== Gradient Refinement ===", flush=True)
        grad_lengths = [512, 1024, 2048, 4096]
        for n in grad_lengths:
            if time.monotonic() > deadline:
                break
            print(f"  Grad n={n} ...", flush=True)
            # Start from best (interpolated) and a few random starts
            a0 = interpolate_sequence(state['best'], n)
            a0 = np.clip(a0, 0.0, None)
            if a0.sum() < 0.01:
                continue
            a0 = a0 / a0.sum()

            res = minimize(gradient_objective, a0, method='L-BFGS-B',
                          bounds=[(0, None)] * n,
                          options={'maxiter': 400, 'max_fun': 8000, 'ftol': 1e-16, 'gtol': 1e-14})
            if res.fun < 1e10:
                update_state(state, res.x)

            # Also try a random start
            if time.monotonic() < deadline:
                a_rand = rng.uniform(0.05, 1.0, n)
                a_rand = a_rand / a_rand.sum()
                res2 = minimize(gradient_objective, a_rand, method='L-BFGS-B',
                               bounds=[(0, None)] * n,
                               options={'maxiter': 300, 'max_fun': 5000, 'ftol': 1e-16, 'gtol': 1e-14})
                if res2.fun < 1e10:
                    update_state(state, res2.x)

    # Phase 3: Final LP polish on best candidate
    if state['best'] is not None and time.monotonic() < deadline:
        n_best = state['best'].size
        if n_best <= 2048:
            print(f"\n=== Final LP Polish (n={n_best}) ===", flush=True)
            a = state['best'].copy()
            a = np.clip(a, 1e-8, None)
            a = a / a.sum()
            # Use smaller momentum for polishing
            polish_deadline = min(deadline, time.monotonic() + (deadline - time.monotonic()) * 0.6)
            a = lp_refine(a, polish_deadline, 200, use_momentum=True)
            update_state(state, a)

            # Try with no momentum
            if time.monotonic() < deadline:
                a2 = state['best'].copy()
                a2 = np.clip(a2, 1e-8, None)
                a2 = a2 / a2.sum()
                a2 = lp_refine(a2, deadline, 100, use_momentum=False)
                update_state(state, a2)

    # Phase 4: Quick random perturbations around best
    if state['best'] is not None and time.monotonic() < deadline:
        print(f"\n=== Perturbation Search ===", flush=True)
        n_best = state['best'].size
        a_base = state['best'].copy()
        a_base = a_base / a_base.sum()

        for pert in [0.01, 0.05, 0.1, 0.2]:
            if time.monotonic() > deadline:
                break
            for trial in range(5):
                if time.monotonic() > deadline:
                    break
                noise = rng.normal(0, pert, n_best)
                a_pert = a_base + noise
                a_pert = np.clip(a_pert, 0.0, None)
                if a_pert.sum() < 1e-10:
                    continue
                a_pert = a_pert / a_pert.sum()
                # Quick LP refinement
                a_ref = lp_refine(a_pert, deadline, 30, use_momentum=False)
                update_state(state, a_ref)

    if state['best'] is None:
        state['best'] = np.ones(256)
        state['best_score'] = float(evaluate_sequence(state['best'].tolist()))

    print(f"\nFINAL: J={state['best_score']:.6f}, n={state['best'].size}", flush=True)
    return [float(x) for x in state['best']]
