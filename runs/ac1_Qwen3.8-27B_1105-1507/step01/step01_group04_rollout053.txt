```python
"""
Optimizes non-negative step function heights to minimize 2n*max(conv(f,f))/(sum(f))^2.
Uses Lp-norm homotopy method with L-BFGS-B optimization in softmax parameterization.
Starts with small p (smooth objective) and gradually increases to large p (approaches true max).
"""

import numpy as np
import time
from scipy.optimize import minimize

# We need to make these top-level functions, no closures

def _fft_convolve(f):
    """Full linear convolution using FFT."""
    n = len(f)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    F = np.fft.rfft(f, size)
    result = np.fft.irfft(F * F, size)
    return result[:2 * n - 1]

def _fft_cross_corr_grad(f, g):
    """
    Compute grad[j] = sum_i f[i] * g[i+j] for j=0,...,n-1.
    Uses FFT for efficiency.
    """
    n = len(f)
    m = len(g)  # m = 2n-1
    # fr = reverse(f)
    fr = f[::-1]
    # h = conv(fr, g), length n + m - 1 = 3n - 2
    size = 1
    while size < n + m - 1:
        size <<= 1
    Fr = np.fft.rfft(fr, size)
    G = np.fft.rfft(g, size)
    h = np.fft.irfft(Fr * G, size)
    # h has length 3n-2 (padded to size)
    # grad[j] = h[n-1+j] for j=0,...,n-1
    return h[n-1 : 2*n-1]

def _score_and_grad_soft(alpha, p, n):
    """
    Compute the Lp-norm objective and its gradient with respect to alpha.
    f = softmax(alpha)
    Objective: 2n * (sum C[k]^p)^(1/p)
    """
    # Compute f from alpha
    alpha_shifted = alpha - np.max(alpha)
    exp_a = np.exp(alpha_shifted)
    f = exp_a / np.sum(exp_a)
    
    # Compute convolution
    C = _fft_convolve(f)
    
    # Compute Lp norm
    if p == 2:
        lp = np.sqrt(np.sum(C**2))
        # gradient of lp w.r.t. f
        # d(lp)/dC[k] = C[k] / lp
        # d(lp)/df[j] = sum_k (C[k]/lp) * 2*f[k-j] (valid)
        h = C / lp
        grad_f = 2.0 * _fft_cross_corr_grad(f, h)
    else:
        Cp = np.sum(C**p)
        lp = Cp**(1.0/p)
        # d(lp)/dC[k] = (1/p) * Cp^(1/p - 1) * p * C[k]^(p-1) = Cp^(1/p-1) * C[k]^(p-1)
        # d(lp)/df[j] = Cp^(1/p-1) * sum_k C[k]^(p-1) * 2*f[k-j]
        scalar = Cp**(1.0/p - 1.0)
        h = C**(p-1)
        grad_f = 2.0 * scalar * _fft_cross_corr_grad(f, h)
    
    objective = 2.0 * n * lp
    
    # Gradient w.r.t. alpha: f * (grad_f - f . grad_f)
    dot_f = np.dot(f, grad_f)
    grad_alpha = f * (grad_f - dot_f)
    
    return objective, grad_alpha

def _score_actual(f):
    """Compute the actual score (with true max)."""
    n = len(f)
    C = _fft_convolve(f)
    S = np.sum(f)
    if S < 0.01:
        return float('inf')
    return 2.0 * n * np.max(C) / (S * S)

def _cosine_init(n):
    """Cosine (Hann) window."""
    j = np.arange(n, dtype=np.float64)
    f = 0.5 * (1.0 + np.cos(2.0 * np.pi * (j + 0.5) / n))
    f = np.maximum(f, 1e-10)
    return f / np.sum(f)

def _gaussian_init(n, sigma_frac=0.3):
    """Gaussian window."""
    j = np.arange(n, dtype=np.float64) - (n - 1) / 2.0
    sigma = sigma_frac * n
    f = np.exp(-j**2 / (2.0 * sigma**2))
    return f / np.sum(f)

def _triangle_init(n):
    """Triangular (Bartlett) window."""
    j = np.arange(n, dtype=np.float64)
    f = 1.0 - np.abs(2.0 * j / (n - 1) - 1.0)
    f = np.maximum(f, 1e-10)
    return f / np.sum(f)

def _blackman_harris_init(n):
    """Blackman-Harris window."""
    j = np.arange(n, dtype=np.float64)
    t = 2.0 * np.pi * j / (n - 1)
    f = 0.35875 - 0.48829 * np.cos(t) + 0.14128 * np.cos(2*t) - 0.01168 * np.cos(3*t)
    f = np.maximum(f, 1e-10)
    return f / np.sum(f)

def _tapered_cosine_init(n, power=4):
    """Tapered cosine window: sin^power(pi*x/2) at edges."""
    j = np.arange(n, dtype=np.float64)
    x = np.sin(np.pi * (j + 0.5) / n)
    f = x**power
    f = np.maximum(f, 1e-10)
    return f / np.sum(f)

def _optimize_lp_homotopy(f_init, p_schedule, max_iter_per_p, tol):
    """
    Optimize using Lp-norm homotopy.
    f_init: initial f (will be converted to alpha)
    p_schedule: list of p values to try (increasing)
    max_iter_per_p: max L-BFGS-B iterations for each p
    Returns: best f found
    """
    n = len(f_init)
    f = f_init.copy()
    f = np.maximum(f, 1e-15)
    f = f / np.sum(f)
    
    best_f = f.copy()
    best_score = _score_actual(f)
    
    for p in p_schedule:
        # Convert f to alpha
        f_safe = np.maximum(f, 1e-30)
        alpha = np.log(f_safe)
        
        def obj_grad(alpha):
            obj, grad = _score_and_grad_soft(alpha, p, n)
            return obj, grad
        
        result = minimize(obj_grad, alpha, jac=True, method='L-BFGS-B',
                          options={'maxiter': max_iter_per_p, 'gtol': tol,
                                   'ftol': 1e-12, 'maxls': 50})
        
        if result.success or result.fun < 2.0 * n:
            # Convert back to f
            alpha_final = result.x - np.max(result.x)
            f_new = np.exp(alpha_final) / np.sum(np.exp(alpha_final))
            f = f_new
        
        score = _score_actual(f)
        if score < best_score:
            best_score = score
            best_f = f.copy()
        
        print(f"      p={p}: score={score:.6f}, best={best_score:.6f}, "
              f"iters={result.nit}, status={result.status}")
    
    return best_f, best_score

def _subgradient_polish(f, n_iter=100, eta=0.01):
    """
    Polish using subgradient of the actual max objective.
    min 2n * max_k C[k] s.t. sum(f)=1, f>=0
    Subgradient: 2n * 2 * f[k* - j] for each j, where k* = argmax C
    """
    n = len(f)
    f = f.copy()
    f = np.maximum(f, 0)
    f = f / np.sum(f)
    
    best_f = f.copy()
    best_score = _score_actual(f)
    
    for it in range(n_iter):
        C = _fft_convolve(f)
        k_star = np.argmax(C)
        # Active set: all k where C[k] is close to max
        threshold = C[k_star] * 0.95
        active = np.where(C >= threshold)[0]
        
        # Average subgradient
        grad_f = np.zeros(n)
        for k in active:
            for j in range(n):
                if 0 <= k - j < n:
                    grad_f[j] += 2.0 * f[k - j]
        grad_f /= len(active)
        
        # Projected gradient step: f_new = f - eta * grad_f, then project to simplex
        f_new = f - eta * grad_f
        f_new = np.maximum(f_new, 0)
        s = np.sum(f_new)
        if s < 1e-30:
            f_new = np.ones(n) / n
        else:
            f_new = f_new / s
        
        score = _score_actual(f_new)
        if score < best_score:
            best_score = score
            best_f = f_new.copy()
        f = f_new
        
        # Adaptive step size
        if it % 20 == 19:
            eta *= 0.7
    
    return best_f, best_score
```

```python
"""
Optimizes non-negative step function to minimize 2n*max(conv(f,f))/(sum(f))^2.
Uses Lp-norm homotopy with L-BFGS-B in softmax space, combined with
frequency-domain shaping and subgradient polishing.
"""

import numpy as np
import time
from scipy.optimize import minimize, linprog
from scipy.sparse import coo_matrix


def fft_convolve(f):
    """Full linear convolution using FFT."""
    n = len(f)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    F = np.fft.rfft(f, size)
    result = np.fft.irfft(F * F, size)
    return result[:2 * n - 1]


def cross_corr_f_g(f, g):
    """
    Compute out[j] = sum_i f[i] * g[i+j] for j=0,...,n-1.
    where n=len(f), len(g)>=n.
    Uses FFT-based convolution.
    """
    n = len(f)
    m = len(g)
    fr = f[::-1]
    size = 1
    while size < n + m - 1:
        size <<= 1
    Fr = np.fft.rfft(fr, size)
    G = np.fft.rfft(g, size)
    h = np.fft.irfft(Fr * G, size)
    return h[n - 1: 2 * n - 1]


def lp_objective_and_grad(alpha, p, n):
    """
    Compute Lp-norm of conv(f,f) and gradient w.r.t. alpha.
    f = softmax(alpha).
    Returns (objective_value, gradient_vector).
    """
    # softmax
    a = alpha - np.max(alpha)
    ea = np.exp(a)
    f = ea / np.sum(ea)
    
    # convolution
    C = fft_convolve(f)
    
    if p <= 1.001:
        # L1 norm (not differentiable, use L2 instead)
        p = 2.0
    
    if abs(p - 2.0) < 0.001:
        C2 = np.sum(C * C)
        lp = np.sqrt(C2)
        h = C / lp
        grad_f = 2.0 * cross_corr_f_g(f, h)
    else:
        Cp = np.sum(C ** p)
        lp = Cp ** (1.0 / p)
        scalar = Cp ** (1.0 / p - 1.0)
        h = C ** (p - 1)
        grad_f = 2.0 * scalar * cross_corr_f_g(f, h)
    
    objective = 2.0 * n * lp
    
    # gradient w.r.t. alpha = f * (grad_f - f.dot(grad_f))
    dot_f = np.dot(f, grad_f)
    grad_alpha = f * (grad_f - dot_f)
    
    return objective, grad_alpha


def score_actual(f):
    """Compute the true score."""
    n = len(f)
    C = fft_convolve(f)
    S = np.sum(f)
    if S < 0.01:
        return float('inf')
    return 2.0 * n * np.max(C) / (S * S)


def init_cosine(n):
    j = np.arange(n, dtype=np.float64)
    f = 0.5 * (1.0 + np.cos(2.0 * np.pi * (j + 0.5) / n))
    f = np.maximum(f, 1e-10)
    return f / np.sum(f)


def init_gaussian(n, sigma_frac=0.3):
    j = np.arange(n, dtype=np.float64) - (n - 1) / 2.0
    sigma = sigma_frac * n
    f = np.exp(-j * j / (2.0 * sigma * sigma))
    return f / np.sum(f)


def init_triangle(n):
    j = np.arange(n, dtype=np.float64)
    f = 1.0 - np.abs(2.0 * (j + 0.5) / n - 1.0)
    f = np.maximum(f, 1e-10)
    return f / np.sum(f)


def init_blackman_harris(n):

    t = 2.0 * np.pi * j / denom
    f = (0.35875 - 0.48829 * np.cos(t) + 0.14128 * np.cos(2 * t) - 0.01168 * np.cos(3 * t))
    f = np.maximum(f, 1e-10)
    return f / np.sum(f)

    f = np.maximum(f, 1e-10)

def init_uniform(n):

def freq_shaping(f, n_iter=50, peak_factor=2.0):
    """
    Iterative frequency-domain shaping: concentrate spectral energy at low frequencies.
    This makes the time-domain convolution flatter.
    """
    n = len(f)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    
    f_work = f.copy()
    
    for it in range(n_iter):
        F = np.fft.rfft(f_work, size)
        mag = np.abs(F)
        phase = np.angle(F)
        
        # Shape: boost low frequencies, suppress high
        freq_idx = np.arange(len(mag))
        # Create a smoothing filter in frequency domain
        # We want to make mag more peaked at DC
        target_mag = mag ** peak_factor  # This peaks the spectrum
        # Or: apply a low-pass shaping
        # Actually, let's just reweight: boost DC, keep rest
        # A simpler approach: apply a Gaussian lowpass in frequency
        freq_norm = freq_idx / len(mag)
        lp_filter = np.exp(-8.0 * freq_norm ** 2)
        
        # Blend: new_mag = mag * (alpha * lp_filter + (1-alpha))
        # Or simply: new_mag = mag * lp_filter (pure low-pass)
        # Let's try: sharpen the magnitude
        new_mag = mag * (0.5 + 0.5 * lp_filter)
        
        new_F = new_mag * np.exp(1j * phase)
        f_work = np.fft.irfft(new_F, size)[:n]
        
        # Enforce non-negativity
        f_work = np.maximum(f_work, 0)
        s = np.sum(f_work)
        if s < 1e-30:
            f_work = np.ones(n) / n
        else:
            f_work = f_work / s
    
    return f_work

    best_f = f.copy()
    best_score = score_actual(f)
    
    for p in p_values:
        # Convert to log parameterization

        def obj_grad(a):
            return lp_objective_and_grad(a, p, n)
        
        t0 = time.monotonic()
        result = minimize(obj_grad, alpha, jac=True, method='L-BFGS-B',

        elapsed = time.monotonic() - t0
        
        if result.status in [0, 1, 2, 3, 4, 5, 6, 7, 8]:  # various success/tolerance

def subgradient_polish(f, total_time=30.0):
    """
    Subgradient method for the true max objective.
    min 2n * max_k C[k] s.t. sum(f)=1, f>=0
    """
    n = len(f)
    f = f.copy()
    f = np.maximum(f, 0)
    f = f / np.sum(f)
    
    best_f = f.copy()
    best_score = score_actual(f)
    start = time.monotonic()
    
    eta = 0.05
    it = 0
    
    while time.monotonic() - start < total_time:
        C = fft_convolve(f)
        max_C = np.max(C)
        threshold = max_C * (1.0 - 0.01 * it * 0.01)  # tighten active set over time
        threshold = max(threshold, 0.99 * max_C)
        active = np.where(C >= threshold)[0]
        
        # Compute subgradient (average over active set)
        # subgrad[j] = (1/|K|) * sum_{k in K} 2*f[k-j] for valid k-j
        grad_f = np.zeros(n)
        for k in active:
            j_min = max(0, k - n + 1)
            j_max = min(n - 1, k)
            if j_max >= j_min:
                indices = np.arange(j_min, j_max + 1)
                src = k - indices
                grad_f[j_min:j_max + 1] += 2.0 * f[src]
        grad_f /= len(active)
        
        # Gradient step
        f_new = f - eta * grad_f
        f_new = np.maximum(f_new, 0)
        s = np.sum(f_new)
        if s < 1e-30:
            f_new = np.ones(n) / n
        else:
            f_new = f_new / s
        
        sc = score_actual(f_new)
        if sc < best_score:
            best_score = sc
            best_f = f_new.copy()
        f = f_new
        
        it += 1
        if it % 50 == 0:
            eta *= 0.5
        
        # Early stop if converged
        if it > 100 and best_score - sc < 1e-10:
            break
    
    return best_f, best_score


def fw_lp_step(f, M):
    """
    Single Frank-Wolfe step using LP:
    maximize sum(b) s.t. (f*b)[k] <= M for all k, b >= 0
    Returns normalized b.
    """
    n = len(f)
    
    # Build constraint matrix: (f*b)[k] = sum_j f[j] * b[k-j] for valid indices
    # (f*b)[k] = sum_{j: 0<=j<n, 0<=k-j<n} b[j] * f[k-j]
    # For each k (0 to 2n-2), the constraint is:
    #   sum_{j in valid range} f[k-j] * b[j] <= M
    
    row_idx = []
    col_idx =
    vals = []
    
    for j in range(n):
        # b[j] appears in (f*b)[k] for k = j, j+1, ..., j+n-1
        # (f*b)[k] includes term b[j] * f[k-j]
        for offset in range(n):
            k = j + offset
            if k < 2 * n - 1:
                row_idx.append(k)
                col_idx.
                vals.append(f[offset])
    
    C_mat = coo_matrix((vals, (row_idx, col_idx)), shape=(2 * n - 1, n))
    
    c_obj = -np.ones(n)  # minimize -sum(b) = maximize sum(b)
    
    res = linprog(c_obj, A_ub=C_mat, b_ub=np.full(2 * n - 1, M),
                  bounds=[(0, None)] * n, method='highs',
                  options={'presolve': True, 'dual_obj_bound': 1e6})
    
    if not res.success or res.x is None:
        return f.copy()
    
    b = np.maximum(res.x, 0)
    s = np.sum(b)
    if s < 1e-10:
        return f.copy()
    return b / s
```
