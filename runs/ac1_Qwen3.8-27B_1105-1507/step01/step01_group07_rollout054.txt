"""
Hybrid optimization: subgradient descent + LP refinement + coordinate descent
for minimizing 2*n*max(conv(a,a))/(sum(a))^2.
"""

import numpy as np
from scipy.optimize import linprog
import time


def compute_obj(a):
    n = len(a)
    if n == 0:
        return np.inf
    conv = np.convolve(a, a)
    max_b = float(np.max(conv))
    sum_a = float(np.sum(a))
    if sum_a < 0.01:
        return np.inf
    return 2.0 * n * max_b / (sum_a * sum_a)


def build_A(a):
    n = len(a)
    A = np.zeros((2 * n - 1, n), dtype=np.float64)
    for k in range(2 * n - 1):
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        A[k, j_start:j_end + 1] = a[k - j_end:k - j_start + 1][::-1]
    return A


def solve_lp(a, M=None):
    if M is None:
        conv = np.convolve(a, a)
        M = float(np.max(conv))
    n = len(a)
    A = build_A(a)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, 1000.0)] * n
    res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
    if res.status == 0:
        return res.x, float(-res.fun)
    return None, None


def subgradient_step(a, eta, k_star=None):
    """One subgradient step to reduce max conv(a,a)."""
    n = len(a)
    conv = np.convolve(a, a)
    if k_star is None:
        k_star = int(np.argmax(conv))
    
    # Gradient of conv[a=k_star] w.r.t. a
    # grad[i] = 2*a[k_star - i] if 0 <= k_star - i < n else 0
    grad = np.zeros(n)
    for i in range(n):
        j = k_star - i
        if 0 <= j < n:
            grad[i] = 2.0 * a[j]
    
    # Subtract mean to maintain sum constraint (approximately)
    grad = grad - np.mean(grad)
    
    a_new = a - eta * grad
    a_new = np.maximum(a_new, 0)
    s = np.sum(a_new)
    if s < 1e-10:
        return a
    return a_new / s


def coordinate_descent(a, max_iter=1000, seed=0):
    """Optimize one coordinate at a time."""
    rng = np.random.default_rng(seed)
    n = len(a)
    best_a = a.copy() / np.sum(a)
    best_obj = compute_obj(best_a)
    
    order = rng.permutation(n)
    for it in range(max_iter):
        i = order[it % n]
        # Try optimizing a[i]
        # Objective as function of a[i]: 2n * max_k conv[a=k] / (sum)^2
        # conv[a=k] is quadratic in a[i] (for k where i and k-i are both in range)
        # This is complex, so just try a few values
        current_val = best_a[i]
        for factor in [0.5, 0.7, 1.0, 1.3, 1.5, 2.0, 0.0]:
            candidate = best_a.copy()
            candidate[i] = current_val * factor
            candidate = np.maximum(candidate, 0)
            s = np.sum(candidate)
            if s < 1e-10:
                continue
            obj = compute_obj(candidate)
            if obj < best_obj:

Hybrid optimization combining subgradient descent, alternating LP,

def compute_obj(a):

    n = len(a)
    if n == 0:
        return np.inf
    conv = np.convolve(a, a)
    max_b = float(np.max(conv))
    sum_a = float(np.sum(a))
    if sum_a < 0.01:
        return np.inf
    return 2.0 * n * max_b / (sum_a * sum_a)


def compute_obj_conv(a):
    """Compute objective and convolution together."""
    n = len(a)
    conv = np.convolve(a, a)
    max_b = float(np.max(conv))
    sum_a = float(np.sum(a))
    if sum_a < 0.01:
        return np.inf, conv
    return 2.0 * n * max_b / (sum_a * sum_a), conv


def build_A_matrix(a):
    """Build constraint matrix for LP: A[k,j] such that (A*g)[k] = conv(a,g)[k]."""
    n = len(a)
    A = np.zeros((2 * n - 1, n), dtype=np.float64)
    for k in range(2 * n - 1):
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        A[k, js:je + 1] = a[k - je:k - js + 1][::-1]
    return A


def lp_step(a, M=None, max_val=1000.0):
    """Solve LP: max sum(g) s.t. conv(a,g) <= M, 0 <= g <= max_val."""
    n = len(a)

        conv = np.convolve(a, a)
        M = float(np.max(conv))
    A = build_A_matrix(a)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, max_val)] * n
    res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
    if res.status == 0:
        return res.x, float(-res.fun), M
    return None, None, M

def structured_init(n, rng):
    """Generate diverse structured initializations."""
    choice = rng.integers(0, 8)
    x = np.arange(n)

    if choice == 0:
        # Cosine single bump
        phase = rng.uniform(0, 2 * np.pi)
        freq = rng.uniform(0.5, 2.0)
        a = np.abs(np.cos(np.pi * (x + 0.5) / n * freq + phase))
    elif choice == 1:
        # Two bumps
        center1 = n * rng.uniform(0.2, 0.4)
        center2 = n * rng.uniform(0.6, 0.8)
        sigma = n * rng.uniform(0.05, 0.15)
        a = np.exp(-((x - center1) ** 2) / (2 * sigma ** 2)) + np.exp(-((x - center2) ** 2) / (2 * sigma ** 2))
    elif choice == 2:
        # U-shaped (high at edges)
        a = np.abs(x - n / 2) / (n / 2) + 0.1

        # Inverted U (high in center)
        a = 1.0 - np.abs(x - n / 2) / (n / 2)
        a = np.maximum(a, 0.01)
    elif choice == 4:
        # Periodic
        period = int(max(2, n * rng.uniform(0.1, 0.3)))
        pattern = rng.random(period) + 0.1
        a = np.tile(pattern, n // period + 1)[:n]
    elif choice == 5:
        # Fibonacci-like
        phi = (1 + np.sqrt(5)) / 2
        a = np.abs(np.cos(2 * np.pi * phi * x / n)) + 0.1
    elif choice == 6:
        # Random smoothed
        a = rng.random(n) + 0.01
        # Box filter smoothing
        k = max(1, n // 50)
        a = np.convolve(a, np.ones(k) / k, mode='same')
    else:
        # Pure random
        a = rng.random(n) + 0.01

    a = np.maximum(a, 0)
    s = np.sum(a)
    if s < 1e-10:
        a = rng.random(n) + 0.01
        s = np.sum(a)
    return a / s


def propose_candidate(seed=42, budget_s=1500, **kwargs):
    """
    Multi-phase optimization:
    Phase 1: Diverse initialization and quick evaluation.
    Phase 2: Moderate refinement of top candidates.
    Phase 3: Intensive refinement of the best.
    Phase 4: Final polishing.
    """
    rng = np.random.default_rng(seed)
    start_time = time.monotonic()
    deadline = start_time + float(budget_s) - 5.0

    best_seq = None
    best_score = np.inf

    def try_update(seq, score, label=""):
        nonlocal best_seq, best_score
        if score < best_score:
            best_score = score
            best_seq = seq.copy()
            elapsed = time.monotonic() - start_time
            print(f"  {label}: score={score:.6f} (n={len(seq)}, t={elapsed:.0f}s)")
            return True
        return False

    # Load parent
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64)
        parent = np.clip(parent, 0, 1000.0)
        if parent.size > 0 and np.isfinite(parent).all():
            obj = compute_obj(parent)
            try_update(parent, obj, "parent")
            print(f"Loaded parent: n={parent.size}, obj={obj:.6f}")
    except Exception as e:
        print(f"Could not load parent: {e}")

    # Phase 1: Generate and evaluate diverse candidates (60s)
    print("Phase 1: Diverse initialization...")
    phase1_deadline = min(deadline, start_time + 60)
    candidates = []

    # Try various lengths
    n_list = [80, 100, 120, 150, 200, 250, 300, 400, 500]

    for n in n_list:
        if time.monotonic() > phase1_deadline:
            break
        for init_idx in range(4):
            if time.monotonic() > phase1_deadline:
                break
            a = structured_init(n, rng)
            # Quick subgradient (200 iterations)
            a_ref, obj = subgradient_refine(a, 200, seed=seed + n * 100 + init_idx)
            candidates.append((obj, a_ref, n))
            try_update(a_ref, obj, f"init n={n} i={init_idx}")
            if time.monotonic() > phase1_deadline:
                break

    # Sort by score and keep top candidates
    candidates.sort(key=lambda x: x[0])
    print(f"Phase 1 done: {len(candidates)} candidates, best={candidates[0][0]:.6f}")

    # Phase 2: Refine top candidates (600s)
    print("Phase 2: Moderate refinement...")
    phase2_deadline = min(deadline, start_time + 660)
    top_candidates = candidates[:min(8, len(candidates))]

    for idx, (obj_c, a_c, n_c) in enumerate(top_candidates):
        if time.monotonic() > phase2_deadline:
            break
        remaining = phase2_deadline - time.monotonic()
        time_per = remaining / (len(top_candidates) - idx)
        print(f"  Refining candidate {idx} (n={n_c}, obj={obj_c:.6f}), budget={time_per:.0f}s")

        a_work = a_c.copy()

        # Alternating LP
        n_lp = int(min(80, time_per / 1.5))
        if n_lp > 5:
            a_work, obj_lp = alternating_lp(a_work, n_lp, seed=seed + 5000 + idx, t=0.4)
            try_update(a_work, obj_lp, f"LP n={n_c} c={idx}")

        # Subgradient
        n_sg = int(min(500, time_per / 0.1))
        if n_sg > 20:
            a_work, obj_sg = subgradient_refine(a_work, n_sg, seed=seed + 10000 + idx)
            try_update(a_work, obj_sg, f"SG n={n_c} c={idx}")

        # Multiplicative
        n_mp = int(min(300, time_per / 0.1))
        if n_mp > 20:
            a_work, obj_mp = multiplicative_refine(a_work, n_mp, seed=seed + 20000 + idx)
            try_update(a_work, obj_mp, f"MP n={n_c} c={idx}")

        # Another LP round
        if time.monotonic() < phase2_deadline - 10:
            n_lp2 = int(min(40, (phase2_deadline - time.monotonic()) / 2.0))
            if n_lp2 > 5:
                a_work, obj_lp2 = alternating_lp(a_work, n_lp2, seed=seed + 30000 + idx, t=0.3)
                try_update(a_work, obj_lp2, f"LP2 n={n_c} c={idx}")

    # Phase 3: Intensive refinement of the best (500s)
    print("Phase 3: Intensive refinement...")
    phase3_deadline = min(deadline, start_time + 1160)
    if best_seq is not None:
        a_best = best_seq.copy()
        n_best = len(a_best)
        print(f"  Best so far: n={n_best}, obj={best_score:.6f}")

        while time.monotonic() < phase3_deadline:
            remaining = phase3_deadline - time.monotonic()
            if remaining < 5:
                break

            # Cycle through methods
            iter_count = int(remaining / 30)
            for cycle in range(max(1, iter_count)):
                if time.monotonic() > phase3_deadline:
                    break

                # Subgradient with smaller steps
                n_sg = int(min(300, remaining / 0.15))
                if n_sg > 10:
                    a_best, obj = subgradient_refine(a_best, n_sg,
                                                     seed=seed + 40000 + cycle,
                                                     eta_init=0.05)
                    try_update(a_best, obj, f"P3 SG c={cycle}")
                    remaining = phase3_deadline - time.monotonic()

                # LP
                if time.monotonic() > phase3_deadline:
                    break
                n_lp = int(min(30, remaining / 2.0))
                if n_lp > 3:
                    a_best, obj = alternating_lp(a_best, n_lp, seed=seed + 50000 + cycle, t=0.3)
                    try_update(a_best, obj, f"P3 LP c={cycle}")
                    remaining = phase3_deadline - time.monotonic()

                # Multiplicative
                if time.monotonic() > phase3_deadline:
                    break
                n_mp = int(min(200, remaining / 0.1))
                if n_mp > 10:
                    a_best, obj = multiplicative_refine(a_best, n_mp, seed=seed + 60000 + cycle)
                    try_update(a_best, obj, f"P3 MP c={cycle}")
                    remaining = phase3_deadline - time.monotonic()

                # Fine local search
                if time.monotonic() > phase3_deadline:
                    break
                n_fs = int(min(2000, remaining / 0.05))
                if n_fs > 10:
                    a_best, obj = fine_local_search(a_best, n_fs, seed=seed + 70000 + cycle)
                    try_update(a_best, obj, f"P3 FS c={cycle}")
                    remaining = phase3_deadline - time.monotonic()

    # Phase 4: Final polishing
    print("Phase 4: Final polishing...")
    if best_seq is not None and time.monotonic() < deadline - 10:
        remaining = deadline - time.monotonic()
        n_best = len(best_seq)
        print(f"  Polishing n={n_best}, obj={best_score:.6f}, time={remaining:.0f}s")

        # Very fine local search
        n_fs = int(remaining / 0.03)
        a_fin, obj_fin = fine_local_search(best_seq, n_fs, seed=seed + 99999)
        try_update(a_fin, obj_fin, "final FS")

        # Another round of subgradient with tiny steps
        if time.monotonic() < deadline - 5:
            n_sg = int(min(500, (deadline - time.monotonic()) / 0.1))
            if n_sg > 10:
                a_fin2, obj_fin2 = subgradient_refine(best_seq, n_sg,
                                                      seed=seed + 88888, eta_init=0.02)
                try_update(a_fin2, obj_fin2, "final SG")

    # Fallback
    if best_seq is None:
        print("FALLBACK: using uniform sequence")
        best_seq = np.ones(100)
        best_score = compute_obj(best_seq)

    elapsed = time.monotonic() - start_time
    print(f"\nDONE: n={len(best_seq)}, score={best_score:.6f}, time={elapsed:.1f}s")

    # Ensure valid output
    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    return result
