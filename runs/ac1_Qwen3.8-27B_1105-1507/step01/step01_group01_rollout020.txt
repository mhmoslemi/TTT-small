"""
Optimize non-negative step-function coefficients to minimize 2*n*max(conv(a,a))/sum(a)^2.

Uses:
- Gaussian/cosine initializations for better starting points
- Subgradient descent with backtracking line search (leveraging convexity of max autocorrelation)
- LP-based direction finding (maximize sum(g) s.t. f*g <= M, g>=0)
- Proper line search after LP steps
- Multiple lengths and restarts with time management
"""

import time
import numpy as np

try:
    from scipy.optimize import linprog
    HAVE_SCIPY = True
except Exception:
    HAVE_SCIPY = False


def local_score(a):
    """Compute the objective: 2*n*max(convolve(a,a))/sum(a)^2"""
    a = np.asarray(a, dtype=np.float64)
    if a.size == 0:
        return float('inf')
    a = np.clip(a, 0.0, 1000.0)
    if not np.isfinite(a).all():
        return float('inf')
    s = float(np.sum(a))
    if not np.isfinite(s) or s <= 1e-12:
        return float('inf')
    c = np.convolve(a, a)
    m = float(np.max(c))
    if not np.isfinite(m):
        return float('inf')
    return float(2.0 * len(a) * m / (s * s))


def normalize(a):
    """Normalize to sum 1, clip to non-negative."""
    a = np.asarray(a, dtype=np.float64)
    a = np.clip(a, 0.0, 1000.0)
    s = float(np.sum(a))
    if s <= 1e-300 or not np.isfinite(s):
        return None
    return a / s


def subgradient_direction(f, top_k=10, tol_frac=0.01):
    """
    Compute subgradient direction for max_k (f*f)(k).
    Returns a direction vector d such that f - d decreases the peak.
    """
    n = len(f)
    c = np.convolve(f, f)
    M = float(np.max(c))
    if M <= 0 or not np.isfinite(M):
        return None

    # Find active set: indices where c[k] is within tol of max
    threshold = M * (1.0 - tol_frac)
    active = np.where(c >= threshold)[0]

    # Limit to top_k most active
    if len(active) > top_k:
        vals = c[active]
        idx = np.argsort(vals)[-top_k:]
        active = active[idx]

    # Compute subgradient: average of 2*f[k-j] over active k
    d = np.zeros(n, dtype=np.float64)
    for k in active:
        for j in range(n):
            idx2 = k - j
            if 0 <= idx2 < n:
                d[j] += 2.0 * f[idx2]
    d /= len(active)

    # Subtract mean to maintain sum constraint direction
    d -= np.mean(d)

    # Ensure we're moving in a direction that reduces the peak
    # (the gradient points uphill, so we want negative gradient)
    return -d  # negative gradient = descent direction


def subgradient_step(f, eta, tol_frac=0.01, top_k=10):
    """One subgradient descent step with clipping and renormalization."""
    d = subgradient_direction(f, top_k, tol_frac)
    if d is None:
        return f.copy()
    f_new = f + eta * d
    f_new = np.clip(f_new, 0.0, None)
    f_new = normalize(f_new)
    if f_new is None:
        return f.copy()
    return f_new


def subgradient_optimize(f, max_time, seed, tol_frac=0.01, top_k=10):
    """Subgradient descent with backtracking line search."""
    f = normalize(f)
    if f is None:
        return None, float('inf')
    n = len(f)
    best = f.copy()
    best_sc = local_score(best)
    t0 = time.monotonic()
    deadline = t0 + max_time
    rng = np.random.default_rng(seed)

    eta = 0.1
    stall = 0

    while time.monotonic() < deadline:
        # Try multiple step sizes (backtracking)
        improved = False
        for trial in range(8):
            eta_t = eta * (0.7 ** trial)
            cand = subgradient_step(best, eta_t, tol_frac, top_k)
            sc = local_score(cand)
            if np.isfinite(sc) and sc < best_sc:
                best = cand
                best_sc = sc
                improved = True
                # Increase step size slightly if we keep improving
                if trial == 0:
                    eta = min(eta * 1.1, 1.0)
                break

        if not improved:
            stall += 1
            eta *= 0.8
            if stall > 50:
                # Random perturbation to escape local issues
                noise = np.exp(rng.normal(0, 0.1, n))
                cand = best * noise
                cand = normalize(cand)
                if cand is not None:
                    sc = local_score(cand)
                    if sc < best_sc:
                        best = cand
                        best_sc = sc
                eta = 0.3
                stall = 0

        # Periodically adjust parameters
        if stall % 20 == 0:
            top_k = int(np.clip(top_k * rng.uniform(0.7, 1.5), 3, 30))
            tol_frac = float(np.clip(tol_frac * rng.uniform(0.7, 1.5), 0.001, 0.1))

    return best, best_sc


def lp_direction(f, seed=0):
    """
    Solve LP: maximize sum(g) subject to (f*g)(k) <= M for all k, g >= 0.
    Returns the optimal g (normalized to sum 1) or None.
    """
    if not HAVE_SCIPY:
        return None
    f = normalize(f)
    if f is None:
        return None
    n = len(f)
    c = np.convolve(f, f)
    M = float(np.max(c))
    if M <= 0 or not np.isfinite(M):
        return None

    # Build constraint matrix A: A[k, j] = f[k-j] if 0<=k-j<n else 0
    # Shape: (2n-1) x n
    A = np.zeros((2 * n - 1, n), dtype=np.float64)
    for j in range(n):
        # A[k, j] = f[k-j] for k-j in [0, n-1], i.e., k in [j, j+n-1]
        k_start = j
        k_end = min(j + n, 2 * n - 1)
        f_idx = np.arange(0, k_end - j)
        A[k_start:k_end, j] = f[f_idx]

    # LP: max 1^T g s.t. A @ g <= M, g >= 0
    # scipy linprog minimizes, so minimize -1^T g
    c_vec = -np.ones(n, dtype=np.float64)
    b_vec = np.full(2 * n - 1, M * (1 + 1e-9), dtype=np.float64)
    bounds = [(0.0, None)] * n

    try:
        res = linprog(c_vec, A_ub=A, b_ub=b_vec, bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': 60})
    except Exception:
        return None

    if res is None or not res.success:
        return None

    g = np.asarray(res.x, dtype=np.float64)
    g = np.clip(g, 0.0, None)
    s = float(np.sum(g))
    if s <= 1e-12:
        return None
    g = g / s
    return g

def make_gaussian(n, width_frac, center_frac=0.5):
    """Create a Gaussian-shaped sequence of length n."""
    x = np.arange(n, dtype=np.float64)
    center = center_frac * (n - 1)
    sigma = width_frac * n
    a = np.exp(-0.5 * ((x - center) / max(sigma, 1.0)) ** 2)
    return normalize(a)


def make_cosine(n, power=1.0):
    """Create a raised-cosine window."""
    x = np.arange(n, dtype=np.float64) / float(n)
    a = 0.5 * (1.0 + np.cos(2.0 * np.pi * (x - 0.5)))
    if power != 1.0:
        a = a ** power
    return normalize(a)


def make_exponential(n, decay, center_frac=0.5):
    """Create an exponential (Laplacian) shape."""
    x = np.arange(n, dtype=np.float64)
    center = center_frac * (n - 1)
    a = np.exp(-decay * np.abs(x - center))
    return normalize(a)


def make_bell_narrow(n, frac):
    """Narrow bell in the center."""
    a = np.zeros(n, dtype=np.float64)
    w = max(1, int(frac * n))
    start = (n - w) // 2
    x = np.arange(w, dtype=np.float64) / w
    a[start:start + w] = 0.5 * (1 + np.cos(2 * np.pi * (x - 0.5)))
    return normalize(a)

        if not improved_epoch:
            stall += 1
            if stall > 3:
                break
        else:
            stall = 0

    return best, best_sc


def optimize_length(n, budget, seed):
    """Optimize a sequence of given length with time budget."""
    t0 = time.monotonic()
    deadline = t0 + budget
    best = None
    best_sc = float('inf')
    rng = np.random.default_rng(seed)

    # Generate diverse initializations
    inits = []
    inits.append(make_gaussian(n, 0.25))
    inits.append(make_gaussian(n, 0.35))
    inits.append(make_gaussian(n, 0.15))
    inits.append(make_cosine(n, 1.0))
    inits.append(make_cosine(n, 2.0))
    inits.append(make_exponential(n, 2.0 / n))
    inits.append(make_bell_narrow(n, 0.5))

    # Add random inits
    for _ in range(3):
        w = int(rng.integers(3, min(20, n)))
        kern = np.ones(w) / w
        a = np.clip(np.convolve(rng.normal(size=n), kern, 'same'), 0, None)
        a = normalize(a)
        if a is not None:
            inits.append(a)

    # Try parent if available
    parent = globals().get('height_sequence_1', None)
    if parent is not None:
        try:
            parr = np.asarray(parent, dtype=np.float64)
            parr = np.clip(parr, 0, 1000)
            if parr.size == n:
                pnorm = normalize(parr)
                if pnorm is not None:
                    inits.append(pnorm)
            elif parr.size > 1:
                # Interpolate
                old_x = np.linspace(0, 1, parr.size)
                new_x = np.linspace(0, 1, n)
                a_interp = np.interp(new_x, old_x, parr)
                a_interp = normalize(a_interp)
                if a_interp is not None:
                    inits.append(a_interp)
        except Exception:
            pass

    # Deduplicate inits (roughly)
    unique_inits = []
    for a in inits:
        if a is None:
            continue
        is_dup = False
        for b in unique_inits:
            if np.linalg.norm(a - b) < 0.01:
                is_dup = True
                break
        if not is_dup:
            unique_inits.append(a)
    inits = unique_inits

    # Phase 1: Quick evaluation and initial optimization of each init
    per_init_time = min(30.0, budget / (len(inits) + 2))
    for idx, init in enumerate(inits):
        if time.monotonic() >= deadline:
            break
        rem = deadline - time.monotonic()
        if rem < 5.0:
            break
        alloc = min(per_init_time, rem - 5.0)
        if alloc < 2.0:
            break

        # Quick subgradient optimization
        cand, sc = subgradient_optimize(init, alloc, seed + idx * 101)
        if cand is not None and np.isfinite(sc) and sc < best_sc:
            best = cand
            best_sc = sc
            print(f"  [n={n}] init {idx}: sc={sc:.6f} (t={time.monotonic()-t0:.1f}s)", flush=True)

    if best is None:
        best = make_gaussian(n, 0.25)
        best_sc = local_score(best)

    # Phase 2: LP-based optimization on the best
    rem = deadline - time.monotonic()
    if rem > 20.0:
        lp_b = min(rem * 0.6, 300.0)
        if lp_b > 15.0:
            cand, sc = lp_optimize(best, lp_b, seed + 5000)
            if cand is not None and np.isfinite(sc) and sc < best_sc:
                best = cand
                best_sc = sc
                print(f"  [n={n}] LP: sc={sc:.6f} (t={time.monotonic()-t0:.1f}s)", flush=True)

    # Phase 3: More subgradient polishing
    rem = deadline - time.monotonic()
    if rem > 10.0:
        sg_b = min(rem * 0.5, 120.0)
        if sg_b > 5.0:
            cand, sc = subgradient_optimize(best, sg_b, seed + 9000, tol_frac=0.005, top_k=15)
            if cand is not None and np.isfinite(sc) and sc < best_sc:
                best = cand
                best_sc = sc
                print(f"  [n={n}] SG polish: sc={sc:.6f} (t={time.monotonic()-t0:.1f}s)", flush=True)

    # Phase 4: Try a few more LP rounds
    rem = deadline - time.monotonic()
    if rem > 15.0 and HAVE_SCIPY:
        cand, sc = lp_optimize(best, min(rem * 0.7, 100.0), seed + 15000)
        if cand is not None and np.isfinite(sc) and sc < best_sc:
            best = cand
            best_sc = sc
            print(f"  [n={n}] LP round2: sc={sc:.6f} (t={time.monotonic()-t0:.1f}s)", flush=True)

    return best, best_sc


def multi_start_optimize(n, budget, seed):
    """Optimize with multiple restarts to avoid local minima."""
    t0 = time.monotonic()
    deadline = t0 + budget - 10.0  # Leave 10s for final operations
    best = None
    best_sc = float('inf')
    rng = np.random.default_rng(seed)

    # First, get a good initial solution
    cand, sc = optimize_length(n, min(budget * 0.4, 200.0), seed)
    if cand is not None:
        best = cand
        best_sc = sc
        print(f"  [n={n}] initial: sc={sc:.6f}", flush=True)

    # Multiple restarts with perturbations of the best
    n_restarts = 5
    restart_budget = (deadline - time.monotonic()) / max(1, n_restarts)
    for r in range(n_restarts):
        if time.monotonic() >= deadline:
            break
        rb = min(restart_budget, deadline - time.monotonic() - 5.0)
        if rb < 10.0:
            break

        # Perturb the best solution
        sigma = 0.15 / (r + 1)  # Decreasing perturbation
        perturbed = best.copy()
        perturbed *= np.exp(rng.normal(0, sigma, n))
        perturbed = normalize(perturbed)
        if perturbed is None:
            continue

        cand, sc = optimize_length(n, rb, seed + 10000 * (r + 1))
        if cand is not None and np.isfinite(sc) and sc < best_sc:
            best = cand
            best_sc = sc
            print(f"  [n={n}] restart {r}: sc={sc:.6f}", flush=True)

    return best, best_sc


def propose_candidate(seed=42, budget_s=1500.0, **kwargs):
    """Main entry point: search for the best sequence within the time budget."""
    start_time = time.monotonic()
    try:
        budget = float(budget_s)
    except Exception:
        budget = 1500.0
    if not np.isfinite(budget) or budget <= 0:
        budget = 1500.0
    deadline = start_time + budget - 15.0  # Leave 15s margin

    print(f"Starting optimization with budget={budget:.0f}s, HAVE_SCIPY={HAVE_SCIPY}", flush=True)

    best_seq = None
    best_score = float('inf')

    # Try parent first
    parent = globals().get('height_sequence_1', None)
    if parent is not None:
        try:
            parr = np.asarray(parent, dtype=np.float64)
            parr = np.clip(parr, 0, 1000)
            if parr.size > 0 and np.isfinite(parr).all():
                best_seq = parr.copy()
                best_score = local_score(parr)
                print(f"Parent: len={parr.size}, score={best_score:.6f}", flush=True)
        except Exception:
            pass

    # Length plan: focus on lengths where improvement is most likely
    # Allocate time based on expected difficulty
    if budget >= 1200:
        lengths = [512, 768, 1024, 1536, 2048]
        allocs = [0.12, 0.12, 0.25, 0.25, 0.20]
    elif budget >= 600:
        lengths = [512, 1024, 2048]
        allocs = [0.15, 0.45, 0.35]
    else:
        lengths = [512, 1024]
        allocs = [0.3, 0.65]

    seed_cur = int(seed)
    for i, (n, frac) in enumerate(zip(lengths, allocs)):
        if time.monotonic() >= deadline:
            break
        rem = deadline - time.monotonic()
        alloc = min(frac * budget, rem - 15.0)
        if alloc < 30.0:
            continue

        print(f"\n=== Length {n}, budget {alloc:.0f}s ===", flush=True)
        try:
            cand, sc = multi_start_optimize(n, alloc, seed_cur)
            if cand is not None and np.isfinite(sc) and sc < best_score:
                best_seq = cand
                best_score = sc
                print(f"  NEW BEST: n={n}, score={sc:.6f}", flush=True)
            else:
                print(f"  No improvement (best here: {sc:.6f})", flush=True)
        except Exception as e:
            print(f"  Length {n} failed: {e}", flush=True)

        seed_cur += 7919

    # Final: also try optimizing the parent length specifically with more time
    if parent is not None and best_seq is not None:
        try:
            parent_arr = np.asarray(parent, dtype=np.float64)
            pn = len(parent_arr)
            if pn not in lengths and pn >= 256:
                rem = deadline - time.monotonic()
                if rem > 60.0:
                    print(f"\n=== Parent length {pn}, budget {rem:.0f}s ===", flush=True)
                    cand, sc = multi_start_optimize(pn, min(rem - 10, 200.0), seed_cur + 31337)
                    if cand is not None and np.isfinite(sc) and sc < best_score:
                        best_seq = cand
                        best_score = sc
                        print(f"  NEW BEST from parent: n={pn}, score={sc:.6f}", flush=True)
        except Exception:
            pass

    # Final polishing on the best sequence found
    if best_seq is not None:
        rem = deadline - time.monotonic()
        if rem > 20.0:
            print(f"\n=== Final polish (budget {rem:.0f}s) ===", flush=True)
            n = len(best_seq)
            # Quick subgradient
            if rem > 10:
                cand, sc = subgradient_optimize(best_seq, min(rem * 0.3, 60.0), seed_cur + 55555,
                                                tol_frac=0.003, top_k=20)
                if cand is not None and np.isfinite(sc) and sc < best_score:
                    best_seq = cand
                    best_score = sc
                    print(f"  Final SG: {sc:.6f}", flush=True)

            # Quick LP
            rem2 = deadline - time.monotonic()
            if rem2 > 15.0 and HAVE_SCIPY:
                cand, sc = lp_optimize(best_seq, min(rem2 * 0.5, 60.0), seed_cur + 66666)
                if cand is not None and np.isfinite(sc) and sc < best_score:
                    best_seq = cand
                    best_score = sc
                    print(f"  Final LP: {sc:.6f}", flush=True)

    # Fallback
    if best_seq is None:
        best_seq = make_gaussian(512, 0.25)
        best_score = local_score(best_seq)

    # Validate with official evaluator
    final = normalize(best_seq)
    if final is None:
        final = np.ones(64, dtype=np.float64) / 64.0

    try:
        official = float(evaluate_sequence([float(x) for x in final]))
        if np.isfinite(official):
            best_score = official
        else:
            # Fallback to parent
            if parent is not None:
                parr = np.asarray(parent, dtype=np.float64)
                parr = np.clip(parr, 0, 1000)
                pn = normalize(parr)
                if pn is not None:
                    off2 = float(evaluate_sequence([float(x) for x in pn]))
                    if np.isfinite(off2):
                        final = pn
                        best_score = off2
    except Exception:
        pass

    elapsed = time.monotonic() - start_time
    print(f"\nFINAL: score={best_score:.6f}, length={len(final)}, elapsed={elapsed:.1f}s", flush=True)
    return [float(x) for x in np.clip(np.asarray(final, dtype=np.float64), 0.0, 1000.0)]
