"""
Hybrid LP-alternation + subgradient optimization for minimizing
2*n*max(conv(a,a))/(sum(a))^2 over non-negative sequences.
Uses 'minimize peak cross-correlation' LP formulation, projected
subgradient descent, structured initializations, and random restarts.
"""

import numpy as np
from scipy.optimize import linprog
import time


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    conv = np.convolve(a, a)
    max_b = float(np.max(conv))
    sum_a = float(np.sum(a))
    if sum_a < 0.01:
        return np.inf
    return 2.0 * n * max_b / (sum_a * sum_a)


def build_conv_matrix(a):
    """Build matrix A such that A @ g = convolve(a, g)."""
    n = len(a)
    A = np.zeros((2 * n - 1, n), dtype=np.float64)
    for k in range(2 * n - 1):
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        A[k, j_start:j_end + 1] = a[k - j_end:k - j_start + 1][::-1]
    return A


def lp_min_peak(a, max_iter_inner=100):
    """
    Solve LP: minimize t s.t. (a*g)[k] <= t for all k, sum(g)=1, g>=0.
    Returns optimal g and t.
    """
    n = len(a)
    A_conv = build_conv_matrix(a)
    # Variables: g_0,...,g_{n-1}, t
    # Minimize t -> c = [0,...,0, 1]
    c = np.zeros(n + 1)
    c[-1] = 1.0
    # Inequality constraints: (a*g)[k] - t <= 0 for each k
    # A_ub has shape (2n-1, n+1)
    A_ub = np.zeros((2 * n - 1, n + 1))
    A_ub[:, :n] = A_conv
    A_ub[:, n] = -1.0
    b_ub = np.zeros(2 * n - 1)
    # Equality: sum(g) = 1
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])
    # Bounds: g_i >= 0, t >= 0
    bounds = [(0, None)] * n + [(0, None)]
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                  bounds=bounds, method='highs')
    if res.status == 0:
        g = res.x[:n]
        t = res.x[n]
        return g, t
    return None, None


def lp_max_sum(a, M):
    """
    Solve LP: maximize sum(g) s.t. (a*g)[k] <= M for all k, g>=0, g_i <= 1.
    Returns optimal g and its sum.
    """
    n = len(a)
    A_conv = build_conv_matrix(a)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, 1.0)] * n
    res = linprog(c, A_ub=A_conv, b_ub=b_ub, bounds=bounds, method='highs')
    if res.status == 0:
        g = res.x
        g_sum = float(-res.fun)
        return g, g_sum
    return None, None


def project_sum1_nonneg(x):
    """Project x onto {a >= 0, sum(a) = 1} using water-filling."""
    # Find tau such that sum(max(x - tau, 0)) = 1
    x_sorted = np.sort(x)[::-1]
    n = len(x)
    # Try each possible support size
    css = np.cumsum(x_sorted)
    # rho: largest k such that x_sorted[k-1] > (css[k-1] - 1) / k
    rho = np.where(x_sorted * np.arange(1, n + 1) > (css - 1.0))[0]
    if len(rho) == 0:
        return np.ones(n) / n
    k = rho[-1] + 1
    tau = (css[k - 1] - 1.0) / k
    result = np.maximum(x - tau, 0.0)
    result_sum = result.sum()
    if result_sum < 1e-15:
        return np.ones(n) / n
    return result / result_sum


def subgradient_refine(a, max_iter=5000, deadline=None, verbose=False):
    """
    Projected subgradient descent on f(a) = max_k (a*a)[k]
    subject to sum(a)=1, a>=0.
    """
    n = len(a)
    a = a.copy()
    a = a / a.sum()
    best_a = a.copy()
    best_obj = compute_obj(best_a)
    lr = 1.0 / np.sqrt(n)
    stall = 0
    patience = 500

    for it in range(max_iter):
        if deadline is not None and time.monotonic() > deadline:
            break

        conv = np.convolve(a, a)
        peak_val = conv.max()
        # Find near-peak indices (within 5% of peak)
        threshold = 0.95 * peak_val
        peak_indices = np.where(conv >= threshold)[0]

        # Compute subgradient (average over peak indices)
        grad = np.zeros(n)
        for k in peak_indices:
            for m in range(n):
                km = k - m
                mk = m + k
                contribution = 0.0
                if 0 <= km < n:
                    contribution += a[km]
                if 0 <= mk < n:
                    contribution += a[mk]
                if 2 * m == k and 0 <= km < n:
                    contribution -= a[km]  # avoid double counting
                grad[m] += contribution
        grad /= len(peak_indices)

        # Step
        a_new = a - lr * grad
        a_new = project_sum1_nonneg(a_new)

        obj_new = compute_obj(a_new)
        if obj_new < best_obj:
            best_obj = obj_new
            best_a = a_new.copy()
            stall = 0
        else:
            stall += 1

        a = a_new
        # Decay learning rate
        lr = lr * 0.999
        if lr < 1e-6:
            lr = 1.0 / np.sqrt(n) * 0.5

        if stall > patience:
            # Reset with perturbation
            stall = 0
            noise = np.exp(np.random.default_rng(it).normal(0, 0.1, n))
            a = best_a * noise
            a = project_sum1_nonneg(a)
            lr = 1.0 / np.sqrt(n)

    return best_a, best_obj


def vectorized_subgradient_step(a, conv, peak_indices):
    """Compute subgradient at peak indices efficiently."""
    n = len(a)
    grad = np.zeros(n)
    for k in peak_indices:
        # grad[m] += a[k-m] + a[m+k] (with bounds), - a[k-m] if 2m=k
        for m in range(n):
            val = 0.0
            if 0 <= k - m < n:
                val += a[k - m]
            if 0 <= m + k < n:
                val += a[m + k]
            if 2 * m == k and 0 <= k - m < n:
                val -= a[k - m]
            grad[m] += val
    return grad


def fast_subgradient_refine(a, max_iter=10000, deadline=None):
    """Faster subgradient with vectorized operations for the peak index."""
    n = len(a)
    a = a.copy().astype(np.float64)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
        return a, compute_obj(a)
    a = a / s
    best_a = a.copy()
    best_obj = compute_obj(best_a)
    base_lr = 2.0 / (n ** 0.7)
    lr = base_lr
    stall = 0
    rng = np.random.default_rng(12345)

    for it in range(max_iter):
        if deadline is not None and time.monotonic() > deadline:
            break

        conv = np.convolve(a, a)
        peak_val = conv.max()
        # Use top few peak indices
        top_k = min(5, len(conv))
        peak_indices = np.argsort(conv)[::-1][:top_k]
        # Weight by proximity to peak
        weights = np.exp(5.0 * (conv[peak_indices] - peak_val) / peak_val)
        weights /= weights.sum()

        grad = np.zeros(n)
        for idx, w in zip(peak_indices, weights):
            k = idx
            # Vectorized: for each m, contribution is a[k-m] + a[m+k]
            # Valid m range: max(0, k-(n-1)) <= m <= min(n-1, k) for a[k-m]
            #                  0 <= m <= min(n-1, 2n-2-k) for a[m+k]
            for m in range(n):
                v = 0.0
                if 0 <= k - m < n:
                    v += a[k - m]
                if 0 <= m + k < n:
                    v += a[m + k]
                if 2 * m == k and 0 <= k - m < n:
                    v -= a[k - m]
                grad[m] += w * v

        a_new = a - lr * grad
        a_new = project_sum1_nonneg(a_new)

        obj_new = compute_obj(a_new)
        if obj_new < best_obj:
            best_obj = obj_new
            best_a = a_new.copy()
            stall = 0
        else:
            stall += 1
            a = a  # don't update if worse (keep best)
            # Actually update a to the new point for exploration
            a = a_new

        if stall % 200 == 199 and stall > 0:
            # Geometric restart
            pert = 1.0 + 0.1 * rng.standard_normal(n)
            a = best_a * pert
            a = np.clip(a, 0, None)
            s2 = a.sum()
            if s2 < 1e-10:
                a = best_a.copy()
            else:
                a = a / s2
            lr = base_lr
            stall = 0

        lr = max(lr * 0.9995, 1e-4)

    return best_a, best_obj


def lp_alternate(a, max_iter=200, deadline=None, t=0.5, verbose=False):
    """LP alternation using 'minimize peak' formulation."""
    n = len(a)
    a = a.copy().astype(np.float64)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
        return a, compute_obj(a)
    a = a / s
    best_a = a.copy()
    best_obj = compute_obj(best_a)
    stall = 0

    for it in range(max_iter):
        if deadline is not None and time.monotonic() > deadline:
            break

        g, t_val = lp_min_peak(a)
        if g is None:
            break

        g = project_sum1_nonneg(g)

        # Check if fixed point
        if np.allclose(g, a, atol=1e-8):
            stall += 5
        else:
            stall = 0

        # Blend
        a_new = (1.0 - t) * a + t * g
        a_new = project_sum1_nonneg(a_new)

        obj_new = compute_obj(a_new)
        if obj_new < best_obj:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new
            if verbose and (it % 20 == 0):
                print(f"    LP it={it}: obj={best_obj:.6f} t_val={t_val:.6f}")
        else:
            # Try smaller step
            t_smaller = t * 0.3
            a_new2 = (1.0 - t_smaller) * a + t_smaller * g
            a_new2 = project_sum1_nonneg(a_new2)
            obj2 = compute_obj(a_new2)
            if obj2 < best_obj:
                best_obj = obj2
                best_a = a_new2.copy()
            a = a_new  # advance regardless

        if stall > 15:
            break

    return best_a, best_obj


def block_refine(a, max_iter=2000, deadline=None):
    """Optimize blocks of consecutive coordinates at a time."""
    n = len(a)
    a = a.copy().astype(np.float64)
    s = a.sum()
    a = a / s
    best_a = a.copy()
    best_obj = compute_obj(best_a)
    rng = np.random.default_rng(777)

    block_sizes = [3, 5, 10, 20]

    for it in range(max_iter):
        if deadline is not None and time.monotonic() > deadline:
            break

        bs = block_sizes[rng.integers(0, len(block_sizes))]
        if bs >= n:
            continue
        start = rng.integers(0, n - bs + 1)

        # Optimize a[start:start+bs] while keeping rest fixed
        # Use a few gradient steps on the sub-vector
        for _ in range(5):
            conv = np.convolve(a, a)
            peak_idx = conv.argmax()
            # Compute gradient only for the block
            grad_block = np.zeros(bs)
            for m in range(bs):
                mm = start + m
                v = 0.0
                if 0 <= peak_idx - mm < n:
                    v += a[peak_idx - mm]
                if 0 <= mm + peak_idx < n:
                    v += a[mm + peak_idx]
                if 2 * mm == peak_idx and 0 <= peak_idx - mm < n:
                    v -= a[peak_idx - mm]
                grad_block[m] = v

            # Gradient step on block
            lr_block = 0.1 / (bs + 1)
            a[start:start + bs] -= lr_block * grad_block
            a[start:start + bs] = np.clip(a[start:start + bs], 0, None)
            # Renormalize
            s2 = a.sum()
            if s2 < 1e-10:
                a[start:start + bs] = 0.1
                s2 = a.sum()
            a = a / s2

        obj_new = compute_obj(a)
        if obj_new < best_obj:
            best_obj = obj_new
            best_a = a.copy()
        else:
            # Revert block
            a[start:start + bs] = best_a[start:start + bs] * np.exp(rng.normal(0, 0.1, bs))
            a[start:start + bs] = np.clip(a[start:start + bs], 0, None)
            s2 = a.sum()
            if s2 > 1e-10:
                a = a / s2

    return best_a, best_obj


def structured_init(n, rng, init_type='random'):
    """Generate structured initialization."""
    if init_type == 'random':
        a = rng.random(n)
    elif init_type == 'cosine':
        x = np.arange(n) / n
        freq = rng.uniform(0.5, 3.0)
        phase = rng.uniform(0, 2 * np.pi)
        a = np.maximum(0, np.cos(2 * np.pi * freq * x + phase)) ** rng.uniform(0.5, 3.0)
    elif init_type == 'triangular':
        x = np.arange(n)
        center = (n - 1) / 2.0
        a = np.maximum(0, 1.0 - np.abs(x - center) / center)
    elif init_type == 'bimodal':
        x = np.arange(n)
        center = (n - 1) / 2.0
        a = np.maximum(0, np.exp(-((x - center / 2) ** 2) / (n * 0.05)) +
                        np.exp(-((x - 3 * center / 2) ** 2) / (n * 0.05)))
    elif init_type == 'smoothed_random':
        a = rng.random(n)
        # Smooth with a small kernel
        kernel_size = max(3, n // 20)
        kernel = np.ones(kernel_size) / kernel_size
        a = np.convolve(a, kernel, mode='same')
        a = np.maximum(a, 0)
    else:
        a = rng.random(n)

    s = a.sum()
    if s < 1e-10:
        a = rng.random(n)
        s = a.sum()
    return a / s


def full_optimize(n, seed, deadline, verbose=False):
    """Run full optimization pipeline for a given sequence length."""
    rng = np.random.default_rng(seed)

    # Try multiple initializations
    init_types = ['random', 'cosine', 'triangular', 'bimodal', 'smoothed_random']
    best_a = None
    best_obj = np.inf

    for init_type in init_types:
        if time.monotonic() > deadline:
            break

        a = structured_init(n, rng, init_type)
        obj_init = compute_obj(a)

        # LP alternation
        remaining_lp = max(10, int((deadline - time.monotonic()) / 5))
        a_lp, obj_lp = lp_alternate(a, max_iter=min(100, remaining_lp),
                                    deadline=deadline, t=0.5)
        if obj_lp < best_obj:
            best_obj = obj_lp
            best_a = a_lp.copy()
            if verbose:
                print(f"      LP ({init_type}): obj={obj_lp:.6f}")

        # Subgradient refinement
        if time.monotonic() < deadline - 5:
            a_sg, obj_sg = fast_subgradient_refine(a_lp, max_iter=3000,
                                                   deadline=deadline)
            if obj_sg < best_obj:
                best_obj = obj_sg
                best_a = a_sg.copy()
                if verbose:
                    print(f"      SG ({init_type}): obj={obj_sg:.6f}")

    # Block refinement on best
    if best_a is not None and time.monotonic() < deadline - 5:
        a_br, obj_br = block_refine(best_a, max_iter=1000, deadline=deadline)
        if obj_br < best_obj:
            best_obj = obj_br
            best_a = a_br.copy()
            if verbose:
                print(f"      Block: obj={obj_br:.6f}")

    # Final subgradient polish
    if best_a is not None and time.monotonic() < deadline - 3:
        a_fin, obj_fin = fast_subgradient_refine(best_a, max_iter=5000,
                                                 deadline=deadline)
        if obj_fin < best_obj:
            best_obj = obj_fin
            best_a = a_fin.copy()

    return best_a, best_obj
