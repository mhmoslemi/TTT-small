"""
Hybrid optimization for nonnegative step functions minimizing
2*n*max(convolve(f,f))/(sum(f))^2.

Uses LP-based direction finding for strong initial descent,

import time

from scipy.optimize import linprog
from scipy.sparse import csr_matrix


def normalize_sequence(arr):
    """Return a nonnegative floating vector normalized to sum 1, or None."""
    try:
        x = np.asarray(arr, dtype=np.float64).ravel()
    except Exception:
        return None
    if x.size == 0:
        return None
    x = np.where(np.isfinite(x), x, 0.0)
    x = np.maximum(x, 0.0)
    s = float(x.sum())
    if s <= 1e-300:
        return None
    x /= s
    return x


def score_normalized(x, n):
    """Score assuming x is nonnegative and sum(x) = 1."""
    conv = np.convolve(x, x)
    m = float(np.max(conv))
    if not np.isfinite(m):
        return float("inf")
    return float(2.0 * n * m)


def simplex_project(v):
    """Project v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = v.size
    if np.all(v <= 0):
        return np.ones(n) / n
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1.0)
    if not np.any(cond):
        return np.ones(n) / n
    rho = np.where(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def make_conv_template(n):
    """Precompute CSR matrix for convolve(f, g)."""
    m = 2 * n - 1
    k64 = np.arange(m, dtype=np.int64)
    js = np.maximum(0, k64 - n + 1).astype(np.int64)
    je = np.minimum(n - 1, k64).astype(np.int64)
    counts = (je - js + 1).astype(np.int64)
    total = int(counts.sum())
    
    cum = np.cumsum(counts)
    starts = np.repeat(cum - counts, counts)
    local = np.arange(total, dtype=np.int64) - starts
    
    row = np.repeat(k64, counts).astype(np.int32)
    col_base = js[row.astype(np.int64)]
    col = (col_base + local).astype(np.int32)
    
    A_data = np.ones(total, dtype=np.float64)
    A = csr_matrix((A_data, (row, col)), shape=(m, n), dtype=np.float64)
    
    c_obj = -np.ones(n, dtype=np.float64)
    bounds = [(0.0, None)] * n
    return A, c_obj, bounds, m


def solve_direction_lp(template, f, n, M):
    """Solve the LP direction problem."""
    A, c_obj, bounds, m = template
    # Constraint: (f * g)_k <= M for all k
    # A @ g = (f * g) where A is sparse
    # We need to fill in the f values
    # The template A has 1s in the right positions
    # We need to multiply each entry by f[col_index]
    
    A_copy = A.tocoo()
    A_copy.data = f[A_copy.col].astype(np.float64)
    A_sparse = csr_matrix((A_copy.data, (A_copy.row, A_copy.col)), shape=(m, n))
    
    b = np.full(m, M, dtype=np.float64)
    res = linprog(c_obj, A_ub=A_sparse, b_ub=b, bounds=bounds, method="highs")
    return res


def compute_gradient(f, beta):
    """
    Compute the gradient of the smoothed objective (1/beta)*log(sum exp(beta*(f*f))).
    Returns grad such that grad_j = 2 * sum_k w_k * f_{k-j}
    where w_k = exp(beta*(f*f)_k) / sum_m exp(beta*(f*f)_m)
    """
    n = f.size
    conv = np.convolve(f, f)
    # Softmax weights (numerically stable)
    max_conv = np.max(conv)
    exp_vals = np.exp(beta * (conv - max_conv))
    w = exp_vals / np.sum(exp_vals)
    
    # grad_j = 2 * sum_{m=0}^{n-1} f_m * w_{m+j}
    # This is the cross-correlation of f and w
    # np.correlate(w, f, 'full')[j + n - 1] = sum_m f[m] * w[m + j]
    
    cross = np.correlate(w, f, mode='full')
    grad = 2.0 * cross[n-1:2*n-1]
    
    return grad, conv
