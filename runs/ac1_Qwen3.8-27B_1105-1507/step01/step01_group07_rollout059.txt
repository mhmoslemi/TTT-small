```python
import numpy as np
from scipy.optimize import linprog
import time

def compute_obj(a):
    n = len(a)
    if n == 0:
        return np.inf
    conv = np.convolve(a, a)
    max_b = float(np.max(conv))
    sum_a = float(np.sum(a))
    if sum_a < 0.01:
        return np.inf
    return 2.0 * n * max_b / (sum_a * sum_a)

def build_A(a):
    n = len(a)
    A = np.zeros((2 * n - 1, n), dtype=np.float64)
    for k in range(2 * n - 1):
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        A[k, js:je+1] = a[k-je:k-js+1][::-1]
    return A

def solve_lp(a, ub=1.0):
    n = len(a)
    conv = np.convolve(a, a)
    M = float(np.max(conv))
    A = build_A(a)
    c = -np.ones(n)
    res = linprog(c, A_ub=A, b_ub=np.full(2*n-1, M), bounds=[(0,ub)]*n, method='highs')
    if res.status == 0:
        return res.x, M
    return None, M
```

```python
    """Compute the objective function."""

    conv = np.convolve(a, a)
    max_b = float(np.max(conv))
    sum_a = float(np.sum(a))

    return 2.0 * n * max_b / (sum_a * sum_a)

def build_A(a):
    """Build the constraint matrix for (a*g)_k for all k."""
    n = len(a)
    A = np.zeros((2*n-1, n))
    for k in range(2*n-1):
        js = max(0, k-n+1)
        je = min(n-1, k)
        # A[k, j] = a[k-j] for j in [js, je]
        A[k, js:je+1] = a[k-je:k-js+1][::-1]
    return A

def lp_step(a, ub=None):
    """One LP step: fix a, find best g."""
    n = len(a)
    conv = np.convolve(a, a)
    M = float(np.max(conv))

                  bounds=[(0, ub)]*n, method='highs')
    if res.status != 0:
        return None, M
    return res.x, M
```

```python
Optimized LP-alternation and multiplicative weights for minimizing
2*n*max(conv(a,a))/(sum(a))^2. Uses adaptive step sizes, natural LP bounds,
multiple sequence lengths, and structured initializations.

import numpy as np
from scipy.optimize import linprog
import time


def compute_obj(a):
    n = len(a)
    if n == 0:
        return np.inf
    conv = np.convolve(a, a)
    max_b = float(np.max(conv))
    sum_a = float(np.sum(a))
    if sum_a < 0.01:
        return np.inf
    return 2.0 * n * max_b / (sum_a * sum_a)


def build_A(a):
    n = len(a)
    A = np.zeros((2 * n - 1, n), dtype=np.float64)
    for k in range(2 * n - 1):
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        A[k, js:je + 1] = a[k - je:k - js + 1][::-1]
    return A


def lp_step(a, verbose=False):
    n = len(a)
    conv = np.convolve(a, a)
    M = float(np.max(conv))
    # Natural upper bound
    max_a = float(np.max(a))
    if max_a < 1e-15:
        return None, M
    ub = M / max_a * 1.1

    c = -np.ones(n)

                  bounds=[(0.0, ub)] * n, method='highs')
    if res.status != 0:
        return None, M
    return res.x, M

    best_obj = compute_obj(best_a)
```

```python
"""
LP-alternation with adaptive step sizes, multiplicative weights initialization,
structured parameterization, and diverse restarts to minimize
2*n*max(conv(a,a))/(sum(a))^2. Targets improvement below 1.5027.
"""

import numpy as np
from scipy.optimize import linprog
import time


def compute_obj(a):
    """Compute the objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    conv = np.convolve(a, a)
    max_b = float(np.max(conv))
    sum_a = float(np.sum(a))
    if sum_a < 0.01:
        return np.inf
    return 2.0 * n * max_b / (sum_a * sum_a)


def build_A(a):
    """Build constraint matrix where A[k,j] = a[k-j] if valid, else 0."""
    n = len(a)
    A = np.zeros((2 * n - 1, n), dtype=np.float64)
    for k in range(2 * n - 1):
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        A[k, js:je + 1] = a[k - je:k - js + 1][::-1]
    return A


def lp_step(a):
    """Solve LP: max sum(g) s.t. (a*g)_k <= M for all k, 0 <= g <= ub."""
    n = len(a)
    conv = np.convolve(a, a)
    M = float(np.max(conv))
    max_a = float(np.max(a))
    if max_a < 1e-15:
        return None, M
    ub = M / max_a * 1.2
    A = build_A(a)
    c = -np.ones(n)
    res = linprog(c, A_ub=A, b_ub=np.full(2 * n - 1, M),
                  bounds=[(0.0, ub)] * n, method='highs')
    if res.status != 0:
        return None, M
    return res.x, M

def mul_weights(a, n_iter=80, verbose=False):

def structured_init(n, rng):

    elif choice == 2:

    elif choice == 3:

    elif choice == 4:

        freq = rng.uniform(2, 8)
        a = (1.0 + 0.5 * np.cos(2 * np.pi * freq * x)) * np.sin(np.pi * x) ** 0.5
    elif choice == 5:

        a = 1.0
        mid = n // 4
        a[mid:n - mid] = rng.uniform(0.2, 0.5)
    elif choice == 6:
        # B-spline-like (smooth bump)
        a = np.maximum(0, 1.0 - np.abs(2 * x - 1) ** rng.uniform(2, 6))
    else:
        # Random
        a = rng.random(n)

    a = np.clip(a, 1e-10, None)
    s = np.sum(a)
    if s < 1e-10:
        a = rng.random(n)
        s = np.sum(a)

        candidate = candidate / s
        obj = compute_obj(candidate)
```
