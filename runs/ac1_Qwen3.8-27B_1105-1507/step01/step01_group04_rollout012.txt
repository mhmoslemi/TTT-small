def _stochastic_local_search(f0, max_iters, rng, step_size=0.01):
    """Random coordinate perturbations with acceptance criterion."""
    f = np.asarray(f0, dtype=np.float64).copy()
    f = _normalize(f)
    if f is None:
        return f0.copy(), float('inf')
    n = len(f)
    best_f = f.copy()
    best_score = _score_fast(f)
    
    for it in range(max_iters):
        # Perturb a random subset of coordinates
        num_perturb = max(1, n // 32)
        idx = rng.integers(0, n, size=num_perturb)
        f_new = f.copy()
        for i in idx:
            f_new[i] *= (1.0 + rng.uniform(-step_size, step_size))
        
        f_new = _normalize(f_new)
        if f_new is None:
            continue
        
        score = _score_fast(f_new)
        if score < best_score:
            best_score = score
            best_f = f_new
            f = f_new
        else:
            # Sometimes accept worse solutions (simulated annealing)
            if rng.random() < 0.01:
                f = f_new
    
    return best_f, best_score

def _cosine_init(n):
    j = np.arange(n, dtype=np.float64)
    f = np.cos(np.pi * (j + 0.5) / n) ** 2
    f = np.maximum(f, 0.0)
    s = np.sum(f)
    if s > 0:
        f /= s
    return f

def _gaussian_init(n, sigma_frac=0.3):
    j = np.arange(n, dtype=np.float64) - (n - 1) / 2.0
    sigma = sigma_frac * n
    f = np.exp(-j * j / (2.0 * sigma * sigma))
    s = np.sum(f)
    if s > 0:
        f /= s
    return f

def _tapered_init(n, power=2.0):
    j = np.arange(n, dtype=np.float64)
    x = np.sin(np.pi * (j + 0.5) / n)
    f = x ** power
    f = np.maximum(f, 0.0)
    s = np.sum(f)
    if s > 0:
        f /= s
    return f

def _top_hat_init(n, frac=0.7):
    f = np.zeros(n)
    center = n // 2
    half = int(n * frac / 2)
    f[center - half:center + half] = 1.0
    # Add cosine taper at edges
    taper = min(10, half)
    for i in range(taper):
        f[center - half + i] *= 0.5 * (1 + np.cos(np.pi * i / taper))
        f[center + half - 1 - i] *= 0.5 * (1 + np.cos(np.pi * i / taper))
    s = np.sum(f)
    if s > 0:
        f /= s
    return f

def _upsample(f, factor):
    """Upsample by inserting zeros and interpolating."""
    n = len(f)
    new_n = n * factor
    # Linear interpolation
    old_x = np.linspace(0, 1, n)
    new_x = np.linspace(0, 1, new_n)
    f_new = np.interp(new_x, old_x, f)
    return f_new

def _downsample(f, factor):
    """Downsample by averaging blocks."""
    n = len(f)
    new_n = n // factor
    f = f[:new_n * factor]
    f_new = f.reshape(new_n, factor).mean(axis=1)
    s = np.sum(f_new)
    if s > 0:
        f_new /= s
    return f_new

def _fw_step(f, M):
    """One Frank-Wolfe step using linearization of the convex constraints."""
    n = len(f)
    # Constraints: for each k, sum_j f[j] * f_new[k-j] <= M
    # In matrix form: A @ f_new <= M * ones
    # A[k, j] = f[k-j] if 0 <= k-j < n, else 0
    
    # This is a horizon matrix (convolution matrix)
    from scipy.sparse import coo_matrix, csr_matrix
    
    # Build A sparsely
    row_idx = []
    col_idx = []
    vals = []
    for j in range(n):
        # A[k, j] = f[k-j] for k-j in [0, n-1], i.e., k in [j, j+n-1]
        for k in range(j, j + n):
            row_idx.append(k)
            col_idx.append(j)
            vals.append(f[k - j])
    
    A = coo_matrix((vals, (row_idx, col_idx)), shape=(2*n-1, n))
    
    # LP: maximize sum f_new s.t. A @ f_new <= M, f_new >= 0, sum f_new = 1
    # Actually, let's just maximize sum f_new s.t. A @ f_new <= M, f_new >= 0
    # (the sum constraint is implicit in the normalization)
    
    from scipy.optimize import linprog
    
    c = -np.ones(n)
    res = linprog(c, A_ub=A, b_ub=np.full(2*n-1, M),
                  bounds=[(0, 1)] * n, method='highs')
    
    if not res.success:
        return f.copy()
    
    f_new = res.x
    s = np.sum(f_new)
    if s < 1e-10:
        return f.copy()
    return f_new / s

from scipy.sparse import coo_matrix
from scipy.optimize import linprog

def _fft_convolve(f):
    """Full linear convolution via FFT."""
    n = len(f)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    F = np.fft.rfft(f, size)
    result = np.fft.irfft(F * F, size)
    return result[:2 * n - 1]

def _score_fast(f):
    """Fast score computation."""
    n = len(f)
    h = _fft_convolve(f)
    M = float(np.max(h))
    S = float(np.sum(f))
    if S < 1e-10:
        return float('inf')
    return 2.0 * n * M / (S * S)

def _normalize(f):
    f = np.maximum(f, 0.0)
    s = float(np.sum(f))
    if s < 1e-30:
        return None
    return f / s

def _pressure_vectorized(f, active_lags):
    """Compute average subgradient of max_k conv(f,f)[k] over active lags."""
    n = len(f)
    grad = np.zeros(n, dtype=np.float64)
    for k in active_lags:
        j_min = max(0, k - n + 1)
        j_max = min(n - 1, k)
        if j_max >= j_min:
            idx = k - np.arange(j_min, j_max + 1)
            grad[j_min:j_max + 1] += 2.0 * f[idx]
    grad /= float(len(active_lags))
    return grad

def _build_conv_matrix(f):
    """Build the convolution matrix A where A[k,j] = f[k-j] if valid."""
    n = len(f)
    row_idx = np.empty(n * n, dtype=np.int32)
    col_idx = np.empty(n * n, dtype=np.int32)
    vals = np.empty(n * n, dtype=np.float64)
    pos = 0
    for j in range(n):
        k_start = j
        k_end = j + n
        row_idx[pos:pos + n] = np.arange(k_start, k_end)
        col_idx[pos:pos + n] = j
        vals[pos:pos + n] = f
        pos += n
    A = coo_matrix((vals, (row_idx, col_idx)), shape=(2 * n - 1, n))
    return A.tocsr()

def _fw_optimize(f0, max_iters, time_budget, restart_patience=200):
    """Frank-Wolfe optimization using LP subproblems."""
    f = np.asarray(f0, dtype=np.float64).copy()
    f = _normalize(f)
    if f is None:
        return f0.copy(), float('inf')
    n = len(f)
    best_f = f.copy()
    best_score = _score_fast(f)
    start_time = time.monotonic()
    no_improve = 0
    t_schedule_counter = 0

    for it in range(max_iters):
        if time.monotonic() - start_time > time_budget:
            break

        h = _fft_convolve(f)
        M = float(np.max(h))
        if M <= 0:
            break

        # Build constraint matrix
        A = _build_conv_matrix(f)

        # LP: maximize sum(f_new) s.t. A @ f_new <= M, f_new >= 0
        c = -np.ones(n)
        b = np.full(2 * n - 1, M)
        res = linprog(c, A_ub=A, b_ub=b, bounds=[(0, None)] * n, method='highs')

        if not res.success or res.x is None:
            break

        g = res.x
        Sg = float(np.sum(g))
        if Sg < 1e-10:
            break
        g = g / Sg

        # FW step with adaptive step size
        t_schedule_counter += 1
        t = 2.0 / (it + 2)

        f_new = (1.0 - t) * f + t * g
        f_norm = _normalize(f_new)
        if f_norm is None:
            break
        f = f_norm

        score = _score_fast(f)
        if score < best_score:
            best_score = score
            best_f = f.copy()
            no_improve = 0
        else:
            no_improve += 1

        # Restart with perturbation if stuck
        if no_improve >= restart_patience:
            rng_r = np.random.default_rng(it * 1000 + 7)
            f_pert = f + rng_r.normal(0, 0.01, n)
            f_pert = _normalize(f_pert)
            if f_pert is not None:
                f = f_pert
            no_improve = 0

    return best_f, best_score

def _mirror_descent(f0, max_iters, eta_init, eta_min, thresh_init, thresh_min):
    """Mirror descent with geometric decay on step size and threshold."""
    f = np.asarray(f0, dtype=np.float64).copy()
    f = _normalize(f)
    if f is None:
        return f0.copy(), float('inf')
    n = len(f)
    best_f = f.copy()
    best_score = _score_fast(f)

    if max_iters <= 1:
        return best_f, best_score

    eta_decay = (eta_min / eta_init) ** (1.0 / max(1, max_iters - 1))
    th_decay = (thresh_min / thresh_init) ** (1.0 / max(1, max_iters - 1))

    for it in range(max_iters):
        h = _fft_convolve(f)
        M = float(np.max(h))
        if M <= 0:
            break

        eta = max(eta_init * (eta_decay ** it), eta_min)
        thresh = max(thresh_init * (th_decay ** it), thresh_min)

        active = np.where(h >= thresh * M)[0]
        if len(active) == 0:
            active = np.array([int(np.argmax(h))], dtype=np.int64)

        p = _pressure_vectorized(f, active)
        log_ratio = -eta * p / M
        log_ratio = np.clip(log_ratio, -20.0, 20.0)
        f = f * np.exp(log_ratio)

        f_norm = _normalize(f)
        if f_norm is None:
            f = np.ones(n) / n
        else:
            f = f_norm

        score = _score_fast(f)
        if score < best_score:
            best_score = score
            best_f = f.copy()

    return best_f, best_score

def _stochastic_search(f0, max_iters, rng, step_frac=0.02):
    """Random multiplicative perturbations with acceptance."""
    f = np.asarray(f0, dtype=np.float64).copy()
    f = _normalize(f)
    if f is None:
        return f0.copy(), float('inf')
    n = len(f)
    best_f = f.copy()
    best_score = _score_fast(f)
    n_perturb = max(3, n // 256)

    for it in range(max_iters):
        f_new = f.copy()
        idx = rng.integers(0, n, size=n_perturb)
        for i in idx:
            if f_new[i] > 1e-10:
                f_new[i] *= (1.0 + rng.uniform(-step_frac, step_frac))
        f_new = _normalize(f_new)
        if f_new is None:
            continue
        score = _score_fast(f_new)
        if score < best_score:
            best_score = score
            best_f = f_new.copy()
            f = f_new
        elif rng.random() < 0.02:
            f = f_new  # occasional acceptance of worse (annealing)

    return best_f, best_score

def _cosine_init(n):
    j = np.arange(n, dtype=np.float64)
    f = np.cos(np.pi * (j + 0.5) / n) ** 2
    f = np.maximum(f, 0.0)
    s = np.sum(f)
    return f / s if s > 0 else np.ones(n) / n

def _gaussian_init(n, sigma_frac=0.3):
    j = np.arange(n, dtype=np.float64) - (n - 1) / 2.0
    sigma = sigma_frac * n
    f = np.exp(-j * j / (2.0 * sigma * sigma))
    s = np.sum(f)
    return f / s if s > 0 else np.ones(n) / n

def _tapered_init(n, power=2.0):
    j = np.arange(n, dtype=np.float64)
    x = np.sin(np.pi * (j + 0.5) / n)
    f = x ** power
    f = np.maximum(f, 0.0)
    s = np.sum(f)
    return f / s if s > 0 else np.ones(n) / n

def _top_hat_init(n, frac=0.75):
    f = np.zeros(n)
    center = n // 2
    half = max(1, int(n * frac / 2))
    f[center - half:center + half] = 1.0
    taper = min(10, half)
    if taper > 0:
        for i in range(taper):
            w = 0.5 * (1 + np.cos(np.pi * i / taper))
            f[center - half + i] *= w
            f[center + half - 1 - i] *= w
    s = np.sum(f)
    return f / s if s > 0 else np.ones(n) / n

def _downsample(f, factor):
    n = len(f)
    new_n = n // factor
    if new_n < 8:
        return f.copy()
    f = f[:new_n * factor]
    f_new = f.reshape(new_n, factor).mean(axis=1)
    s = np.sum(f_new)
    return f_new / s if s > 0 else f_new

def _upsample(f, factor):
    n = len(f)
    new_n = n * factor
    old_x = np.linspace(0.0, 1.0, n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_new = np.interp(new_x, old_x, f)
    s = np.sum(f_new)
    return f_new / s if s > 0 else f_new

    Main search function combining FW optimization, mirror descent,
    stochastic search, and multi-scale refinement.
    """
    deadline = time.monotonic() + float(budget_s) - 5.0
    rng = np.random.default_rng(seed)

    try:

        parent = np.clip(parent, 0.0, 1000.0)
    except (NameError, Exception):
        parent = np.ones(512)

    best_f = parent.copy()
    best_score = float(evaluate_sequence(best_f.tolist()))
    print(f"Parent: n={len(best_f)}, score={best_score:.6f}")

    # Generate diverse initializations
    inits = {}
    pn = _normalize(parent)
    if pn is not None:
        inits['parent'] = pn

    for n in [512, 1024]:
        inits[f'cos{nc}' if False else f'cos_{n}'] = _cosine_init(n)
        inits[f'gauss30_{n}'] = _gaussian_init(n, 0.30)
        inits[f'gauss50_{n}'] = _gaussian_init(n, 0.50)
        inits[f'tap2_{n}'] = _tapered_init(n, 2.0)
        inits[f'tap3_{n}'] = _tapered_init(n, 3.0)
        inits[f'that_{n}'] = _top_hat_init(n, 0.80)
        fr = rng.random(n)
        inits[f'rand_{n}'] = _normalize(fr)

    print(f"Total initializations: {len(inits)}")

    # Phase 1: FW optimization on each initialization
    remaining = deadline - time.monotonic()
    n_inits = len(inits)
    time_per_fw = max(5.0, min(40.0, remaining * 0.50 / n_inits))

    results = []
    for name, f0 in inits.items():
        if time.monotonic() > deadline - 180:
            break
        n = len(f0)
        t0 = time.monotonic()
        f_fw, s_fw = _fw_optimize(f0, max_iters=5000, time_budget=time_per_fw)
        elapsed = time.monotonic() - t0
        print(f"  FW {name} (n={n}): {s_fw:.6f} ({elapsed:.1f}s)")
        results.append((s_fw, name, f_fw))
        if s_fw < best_score:
            best_score = s_fw
            best_f = f_fw.copy()
            print(f"    ** New best: {best_score:.6f}")

    # Sort by score
    results.sort(key=lambda x: x[0])

    # Phase 2: Mirror descent polishing on top results
    print(f"\nPhase 2: MD polishing")
    top_k = min(4, len(results))
    for r in range(top_k):
        if time.monotonic() > deadline - 80:
            break
        s_cur, name, f_cur = results[r]
        n = len(f_cur)
        remaining = deadline - time.monotonic() - 30
        # Estimate iterations: each ~0.5ms for n=512, ~2ms for n=1024
        time_per_iter = 0.0005 * (n / 512.0)
        n_iters = int(max(2000, min(20000, remaining * 0.25 / time_per_iter)))

        f_md, s_md = _mirror_descent(f_cur, n_iters,
                                     eta_init=0.5, eta_min=0.01,
                                     thresh_init=0.6, thresh_min=0.98)
        print(f"  MD {name}: {s_md:.6f}")
        if s_md < best_score:
            best_score = s_md
            best_f = f_md.copy()
            print(f"    ** New best: {best_score:.6f}")
            results[r] = (s_md, name, f_md)
        else:
            results[r] = (s_md, name, f_md)

    # Phase 3: More FW on the best with longer budget
    print(f"\nPhase 3: Extended FW on best")
    if time.monotonic() < deadline - 60:
        s_best, name_best, f_best_cur = results[0]
        remaining = deadline - time.monotonic() - 20
        fw_time = min(remaining * 0.4, 200.0)
        f_fw2, s_fw2 = _fw_optimize(f_best_cur, max_iters=10000, time_budget=fw_time)
        print(f"  Extended FW: {s_fw2:.6f}")
        if s_fw2 < best_score:
            best_score = s_fw2
            best_f = f_fw2.copy()
            print(f"    ** New best: {best_score:.6f}")

    # Phase 4: Stochastic local search
    print(f"\nPhase 4: Stochastic search")
    if time.monotonic() < deadline - 30:
        remaining = deadline - time.monotonic() - 10
        n = len(best_f)
        time_per_iter = 0.001 * (n / 512.0)
        n_sls = int(max(500, min(10000, remaining * 0.4 / time_per_iter)))
        f_sls, s_sls = _stochastic_search(best_f, n_sls, rng, step_frac=0.03)
        print(f"  SLS: {s_sls:.6f}")
        if s_sls < best_score:
            best_score = s_sls
            best_f = f_sls.copy()
            print(f"    ** New best: {best_score:.6f}")

    # Phase 5: Multi-scale refinement
    print(f"\nPhase 5: Multi-scale")
    if time.monotonic() < deadline - 40:
        n_best = len(best_f)
        for factor in [2, 4]:
            if time.monotonic() > deadline - 20:
                break
            n_low = n_best // factor
            if n_low < 64:
                continue
            f_low = _downsample(best_f, factor)

            remaining = deadline - time.monotonic() - 15
            # Optimize at low res with FW
            fw_time = min(remaining * 0.2, 60.0)
            f_low_opt, s_low = _fw_optimize(f_low, max_iters=3000, time_budget=fw_time)

            # Upsample and polish
            f_up = _upsample(f_low_opt, factor)
            f_up = _normalize(f_up)
            if f_up is None:
                continue

            remaining = deadline - time.monotonic() - 10
            time_per_iter = 0.0005 * (n_best / 512.0)
            n_iters = int(max(1000, min(10000, remaining * 0.3 / time_per_iter)))
            f_ms, s_ms = _mirror_descent(f_up, n_iters,
                                         eta_init=0.3, eta_min=0.005,
                                         thresh_init=0.8, thresh_min=0.99)
            print(f"  Multi-scale f={factor}: {s_ms:.6f}")
            if s_ms < best_score:
                best_score = s_ms
                best_f = f_ms.copy()
                print(f"    ** New best: {best_score:.6f}")

    # Phase 6: Final fine polishing
    print(f"\nPhase 6: Final polish")
    if time.monotonic() < deadline - 8:
        remaining = deadline - time.monotonic() - 3
        n = len(best_f)
        time_per_iter = 0.0005 * (n / 512.0)
        n_iters = int(max(2000, min(30000, remaining * 0.7 / time_per_iter)))
        f_fin, s_fin = _mirror_descent(best_f, n_iters,
                                       eta_init=0.15, eta_min=0.001,
                                       thresh_init=0.85, thresh_min=0.999)
        print(f"  Final MD: {s_fin:.6f}")
        if s_fin < best_score:
            best_score = s_fin
            best_f = f_fin.copy()
            print(f"    ** New best: {best_score:.6f}")

        # Tiny step SLS
        if time.monotonic() < deadline - 5:
            n_sls2 = int(max(200, min(3000, (deadline - time.monotonic() - 2) / (0.001 * n / 512.0))))
            f_sls2, s_sls2 = _stochastic_search(best_f, n_sls2, rng, step_frac=0.005)
            print(f"  Final SLS: {s_sls2:.6f}")
            if s_sls2 < best_score:
                best_score = s_sls2
                best_f = f_sls2.copy()

    # Final verification
    final_list = [float(x) for x in best_f]
    final_score = float(evaluate_sequence(final_list))
    print(f"\nFinal verified score: {final_score:.6f} (n={len(final_list)})")

    if not np.isfinite(final_score) or final_score > 2.0:
        print("  Falling back to parent.")
        return [float(x) for x in parent]

    return final_list
