Uses a multi-phase approach: frequency-domain flattening, soft-max L-BFGS optimization,

import numpy as np

def fast_convolve(a):
    """Compute full linear convolution of a with itself using FFT."""
    n = len(a)
    size = 2 * n - 1
    N = 1 << (size - 1).bit_length()
    fa = np.fft.rfft(a, N)
    fc = fa * fa
    result = np.fft.irfft(fc, N)[:size]
    result[result < 0] = 0.0
    return result

def project_simplex(v):
    """Project vector v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cv = np.cumsum(u) - 1.0
    rho_candidates = np.where(u * np.arange(1, n + 1) > cv)[0]
    if len(rho_candidates) == 0:
        return np.ones(n) / n
    rho = rho_candidates[-1]
    theta = cv[rho] / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def softmax(x):
    """Numerically stable softmax."""
    x_shifted = x - np.max(x)
    e = np.exp(x_shifted)
    return e / np.sum(e)

def compute_convolve_gradient(a, weights):

    """

def soft_max_objective_and_grad(x, tau):
    """
    Compute soft-max objective and gradient for the softmax parameterization.
    x: unconstrained parameters
    Returns: (value, gradient)
    """
    a = softmax(x)
    n = len(a)
    c = fast_convolve(a)
    
    # Soft-max value
    tc = tau * c
    tc_max = np.max(tc)
    log_sum_exp = tc_max + np.log(np.sum(np.exp(tc - tc_max)))
    value = log_sum_exp / tau
    
    # Weights for soft-max
    w = np.exp(tc - tc_max)
    w /= np.sum(w)
    
    # Gradient of value w.r.t. a: grad_a = 2 * (w cross-correlate a)
    grad_a = compute_convolve_gradient_fast(a, w)
    
    # Gradient w.r.t. x: grad_x[j] = sum_i grad_a[i] * da_i/dx_j
    # da_i/dx_j = a_i * (delta_ij - a_j)
    # grad_x[j] = sum_i grad_a[i] * a_i * (delta_ij - a_j)
    #            = grad_a[j] * a_j - a_j * sum_i grad_a[i] * a_i
    
    weighted_sum = np.dot(grad_a, a)
    grad_x = grad_a * a - a * weighted_sum
    
    return value, grad_x

def soft_max_obj(x, tau):
    """Objective only."""
    a = softmax(x)
    c = fast_convolve(a)
    tc = tau * c
    tc_max = np.max(tc)
    log_sum_exp = tc_max + np.log(np.sum(np.exp(tc - tc_max)))
    return log_sum_exp / tau

def coordinate_descent_step(a, j, max_iter=50):
    """
    Optimize coordinate j of a to minimize the score.
    Returns the optimized value of a[j].
    """
    n = len(a)
    c = fast_convolve(a)
    S = np.sum(a)
    a_j = a[j]
    
    # For a new value v, the convolution changes as:
    # c_new[k] = c[k] + 2*(v - a_j)*a[k-j] + (v-a_j)^2 * I(k=2j)
    # (with a[m] = 0 for invalid m)
    
    # Precompute b[k] = a[k-j] (valid)
    b = np.zeros(2*n-1)
    for k in range(2*n-1):
        idx = k - j
        if 0 <= idx < n:
            b[k] = a[idx]
    
    # I(k=2j)
    has_diag = (2*j < 2*n-1)
    
    def score_of_v(v):
        d = v - a_j
        c_new = c + 2*d*b
        if has_diag:
            c_new[2*j] += d*d
        c_new = np.maximum(c_new, 0)  # shouldn't be negative but just in case
        S_new = S - a_j + v
        if S_new < 0.01:
            return np.inf
        return 2.0 * n * np.max(c_new) / (S_new * S_new)
    
    # Golden section search on [0, upper]
    # Upper bound: try up to 3*max(a_j, 0.01)
    upper = max(3.0 * a_j, 0.05)
    
    result = minimize_scalar(score_of_v, bounds=(0.0, upper), method='bounded',
                             options={'xatol': 1e-10})
    
    if result.success:
        return result.x
    else:
        return a_j

def lp_refinement(a, deadline, max_iters=50):
    """
    LP-based refinement using linearized quadratic constraints.
    """
    n = len(a)
    a = a.copy().astype(np.float64)
    S = a.sum()
    a = a / S
    
    c = fast_convolve(a)
    best_score = evaluate_sequence(a.tolist())
    best_a = a.copy()
    
    for it in range(max_iters):
        if time.time() > deadline:
            break
        
        c = fast_convolve(a)
        P = np.max(c)
        
        # Build the linearized LP:
        # minimize t
        # subject to: grad_k(a) @ (a_new - a) + c[k] <= t for all k
        #             a_new >= 0
        #             sum(a_new) = 1
        #
        # Variables: a_new (n) + t (1) = n+1
        # grad_k(a)[j] = 2*a[k-j] (valid)
        
        # Build constraint matrix
        # For each k: sum_j 2*a[k-j] * a_new[j] - t <= c[k]
        # i.e., sum_j 2*a[k-j] * a_new[j] - t <= c[k]
        
        # c_vec for linprog: minimize t, so c_obj = [0, 0, ..., 0, 1]
        c_obj = np.zeros(n + 1)
        c_obj[-1] = 1.0
        
        A_ub = np.zeros((2*n-1, n+1))
        b_ub = np.zeros(2*n-1)
        
        for k in range(2*n-1):
            # grad_k(a)[j] = 2*a[k-j] if 0<=k-j<n else 0
            for j in range(n):
                idx = k - j
                if 0 <= idx < n:
                    A_ub[k, j] = 2.0 * a[idx]
            A_ub[k, n] = -1.0
            b_ub[k] = c[k]
        
        # Add sum constraint: sum(a_new) = 1 -> -sum(a_new) <= -1 and sum(a_new) <= 1
        # Actually, equality: use A_eq
        A_eq = np.zeros((1, n+1))
        A_eq[0, :n] = 1.0
        b_eq = [1.0]
        
        bounds = [(0.0, None)] * n + [(0.0, None)]
        
        try:
            res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                         bounds=bounds, method='highs',
                         options={'time_limit': 5.0, 'presolve': True})
            if res.success:
                a_new = res.x[:n]
                a_new = np.maximum(a_new, 0)
                s = a_new.sum()
                if s > 0.01:
                    a_new = a_new / s
                    score = evaluate_sequence(a_new.tolist())
                    if np.isfinite(score) and score < best_score:
                        best_score = score
                        best_a = a_new.copy()
                        # Mix with current to avoid jumping
                        a = 0.7 * a + 0.3 * a_new
                        a = project_simplex(a)
                    else:
                        # No improvement, try smaller step
                        a = 0.95 * a + 0.05 * a_new
                        a = project_simplex(a)
        except Exception:
            pass
    
    return best_a, best_score

def alt_projection_init(n, rng, n_iter=80):
    """Alternating projection for flat-spectrum initialization."""
    N = 1 << (2 * n - 2).bit_length()
    a = np.ones(n) / n
    
    for it in range(n_iter):
        F = np.fft.rfft(a, N)
        # Target: flat magnitude
        mag = np.abs(F)
        target_mag = np.ones(len(F))
        # Slight taper for better non-negativity
        weights = np.linspace(1.0, 0.8, len(F))
        target_mag = weights ** 0.05
        F = target_mag * np.exp(1j * np.angle(F))
        F[0] = F[0].real + 0j
        a = np.fft.irfft(F, N)[:n]
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s < 1e-10:
            a = np.ones(n) / n
        else:
            a = a / s
    
    return a

"""
Multi-phase optimizer for minimizing 2*n*max(convolve(a,a))/sum(a)^2.
Uses soft-max L-BFGS with tau annealing, LP linearized refinement,
and fast coordinate descent. Tries multiple lengths and initializations
to escape local minima.

import numpy as np
import time
from scipy.optimize import minimize, minimize_scalar, linprog


def fast_convolve(a):
    """Compute full linear convolution a*a using FFT."""
    n = len(a)
    size = 2 * n - 1
    N = 1 << (size - 1).bit_length()
    fa = np.fft.rfft(a, N)
    result = np.fft.irfft(fa * fa, N)[:size]
    result[result < 0] = 0.0
    return result


def project_simplex(v):
    """Project vector onto the probability simplex."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cv = np.cumsum(u) - 1.0
    rho_candidates = np.where(u * np.arange(1, n + 1) > cv)[0]
    if len(rho_candidates) == 0:
        return np.ones(n) / n
    rho = rho_candidates[-1]
    theta = cv[rho] / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def softmax_func(x):
    """Numerically stable softmax."""
    x_shifted = x - np.max(x)
    e = np.exp(x_shifted)
    return e / np.sum(e)


def compute_grad_conv_weights(a, weights):
    """
    Compute g[j] = 2 * sum_k weights[k] * a[k-j] for valid (0<=k-j<n).
    Uses FFT cross-correlation for efficiency.
    """
    n = len(a)
    size = 2 * n - 1
    N = 1 << (size - 1).bit_length()

    b = np.zeros(N)
    b[:size] = weights

    a_rev_padded = np.zeros(N)
    a_rev_padded[:n] = a[::-1]

    B = np.fft.rfft(b, N)
    A = np.fft.rfft(a_rev_padded, N)
    conv = np.fft.irfft(B * A, N)

    grad = 2.0 * conv[n-1:2*n-1]
    return grad


def soft_max_value_grad(x, tau):
    """Compute soft-max objective value and gradient w.r.t. x (log-softmax params)."""
    a = softmax_func(x)
    c = fast_convolve(a)

    tc = tau * c
    tc_max = np.max(tc)
    log_sum_exp = tc_max + np.log(np.sum(np.exp(tc - tc_max)))
    value = log_sum_exp / tau

    w = np.exp(tc - tc_max)
    w = w / np.sum(w)

    g = compute_grad_conv_weights(a, w)
    W = np.dot(w, c)

    grad_x = (a / tau) * (g - 2.0 * W)
    return value, grad_x


def soft_max_fun_scipy(x, tau):
    """Wrapper for scipy minimize: objective."""
    return soft_max_value_grad(x, tau)[0]


def soft_max_jac_scipy(x, tau):
    """Wrapper for scipy minimize: gradient."""
    return soft_max_value_grad(x, tau)[1]

    if tau_schedule is None:
        tau_schedule = [10, 30, 80, 200, 500, 1000]
    if rng is None:
        rng = np.random.default_rng(0)

    a = a_init.copy().astype(np.float64)
    s = a.sum()
    a = np.maximum(a, 0) / s

def lp_refinement(a, deadline, max_iters=20):
    """LP-based refinement using linearized quadratic constraints."""
    n = len(a)
    if n > 2048:
        return a, evaluate_sequence(a.tolist())

    a = a.copy().astype(np.float64)
    s = a.sum()
    a = a / s

    best_a = a.copy()
    best_score = evaluate_sequence(a.tolist())

    for it in range(max_iters):
        if time.time() > deadline - 2:
            break

        c = fast_convolve(a)

        # Build LP: minimize t s.t. grad_k(a_new) @ a_new - t <= c[k], sum(a_new)=1, a_new>=0
        # grad_k(a)[j] = 2*a[k-j] (valid), so constraint is:
        # sum_j 2*a[k-j]*a_new[j] - t <= c[k]

        c_obj = np.zeros(n + 1)
        c_obj[-1] = 1.0

        A_ub = np.zeros((2 * n - 1, n + 1))
        b_ub = c.copy()

        for k in range(2 * n - 1):
            A_ub[k, n] = -1.0
            lo = max(0, k - n + 1)
            hi = min(n - 1, k)
            for j in range(lo, hi + 1):
                A_ub[k, j] = 2.0 * a[k - j]

        A_eq = np.zeros((1, n + 1))
        A_eq[0, :n] = 1.0
        b_eq = np.array([1.0])

        bounds = [(0.0, None)] * (n + 1)

        try:
            res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                          bounds=bounds, method='highs',
                          options={'time_limit': 5.0})
            if res.success:
                a_new = np.maximum(res.x[:n], 0)
                sn = a_new.sum()
                if sn > 0.01:
                    a_new = a_new / sn
                    sc = evaluate_sequence(a_new.tolist())
                    if np.isfinite(sc) and sc < best_score:
                        best_score = sc
                        best_a = a_new.copy()
                        a = 0.6 * a + 0.4 * a_new
                        a = project_simplex(a)
                    else:
                        a = 0.9 * a + 0.1 * a_new
                        a = project_simplex(a)
        except Exception:
            pass

    return best_a, best_score


def coord_descent_step(a, j, c, S):
    """Optimize single coordinate j. Returns new value."""
    n = len(a)
    a_j = a[j]

    b = np.zeros(2 * n - 1)
    k_start = j
    k_end = min(j + n, 2 * n - 1)
    b[k_start:k_end] = a[:k_end - k_start]

    has_diag = (2 * j < 2 * n - 1)
    diag_pos = 2 * j

    upper = max(5.0 * a_j + 0.01, 0.05)

    def score_fn(v):
        d = v - a_j
        c_new = c + 2.0 * d * b
        if has_diag:
            c_new[diag_pos] += d * d
        S_new = S - a_j + v
        if S_new < 0.01:
            return 1e10
        return 2.0 * n * np.max(c_new) / (S_new * S_new)

    result = minimize_scalar(score_fn, bounds=(0.0, upper), method='bounded',
                             options={'xatol': 1e-12})
    if result.success and result.fun < score_fn(a_j):
        return result.x
    return a_j


def coord_descent_full(a, deadline, max_rounds=500):
    """Full cyclic coordinate descent."""
    n = len(a)
    a = a.copy().astype(np.float64)
    S = a.sum()
    a = a / S

    best_a = a.copy()
    best_score = evaluate_sequence(a.tolist())

    c = fast_convolve(a)

    for rnd in range(max_rounds):
        if time.time() > deadline - 2:
            break

        improved = False
        for j in range(n):
            if time.time() > deadline - 2:
                return best_a, best_score

            v_new = coord_descent_step(a, j, c, S)
            if abs(v_new - a[j]) > 1e-14:
                d = v_new - a[j]
                # Update c in-place
                k_start = j
                k_end = min(j + n, 2 * n - 1)
                c[k_start:k_end] += 2.0 * d * a[:k_end - k_start]
                if 2 * j < 2 * n - 1:
                    c[2 * j] += d * d  # The extra d^2 term
                a[j] = v_new
                S += d
                improved = True

        if improved:
            score = evaluate_sequence(a.tolist())
            if np.isfinite(score) and score < best_score:
                best_score = score
                best_a = a.copy()
        else:
            break

    return best_a, best_score


def alt_projection_init(n, rng, n_iter=100):
    """Alternating projection for near-flat-spectrum nonnegative sequence."""
    N = 1 << (2 * n - 2).bit_length()
    a = np.ones(n) / n

        F = np.fft.rfft(a, N)
        mag_target = np.ones(len(F))
        weights = np.linspace(1.0, 0.7, len(F))
        mag_target = weights ** 0.08
        F = mag_target * np.exp(1j * np.angle(F))
        F[0] = np.abs(F[0])
        a = np.fft.irfft(F, N)[:n]
        a = np.maximum(a, 0.0)
        s = a.sum()
        a = a / s if s > 1e-10 else np.ones(n) / n

    return a

                    best_seq = a_opt.copy()
        except Exception as e:
            print(f"n={n_try} error: {e}")

    # --- Random init attempts ---
    for trial in range(3):
        if time.time() > overall_deadline - 60:
            break
        try:
            n_try = int(rng.integers(350, 700))
            a_init = rng.random(n_try) + 0.2
            t_start = time.time()
            alloc = min(100, overall_deadline - time.time() - 30)
            if alloc < 20:
                break
            a_opt, sc = full_optimize_phase(a_init, time.time() + alloc, rng)
            if a_opt is not None and np.isfinite(sc):
                print(f"Random n={n_try}: score={sc:.6f} [{time.time()-t_start:.0f}s]")
                if sc < best_score:
                    best_score = sc
                    best_seq = a_opt.copy()
        except Exception:
            pass

    # --- Final polish: coordinate descent on best ---
    if best_seq is not None and time.time() < overall_deadline - 10:
        print("Final coordinate descent polish...")
        a_final, sc_final = coord_descent_full(best_seq, overall_deadline, max_rounds=1000)
        if np.isfinite(sc_final) and sc_final < best_score:
            best_score = sc_final
            best_seq = a_final.copy()
            print(f"  After polish: {best_score:.6f}")

        # One more round of soft-max at high tau
        if time.time() < overall_deadline - 20:
            a_sm, sc_sm = optimize_soft_max(best_seq, overall_deadline - 5,
                                            tau_schedule=[500, 1000, 2000], rng=rng)
            if np.isfinite(sc_sm) and sc_sm < best_score:
                best_score = sc_sm
                best_seq = a_sm.copy()
                print(f"  After high-tau soft-max: {best_score:.6f}")

            # Final coord descent
            a_cd, sc_cd = coord_descent_full(best_seq, overall_deadline, max_rounds=500)
            if np.isfinite(sc_cd) and sc_cd < best_score:
                best_score = sc_cd
                best_seq = a_cd.copy()
                print(f"  Final: {best_score:.6f}")

    # Fallback
    if best_seq is None:
        print("Falling back to uniform sequence")
        best_seq = np.ones(256, dtype=np.float64)
        best_score = evaluate_sequence(best_seq.tolist())

    result = [float(x) for x in best_seq]
    final_score = evaluate_sequence(result)
    print(f"FINAL: score={final_score:.6f}, length={len(result)}")
    return result
