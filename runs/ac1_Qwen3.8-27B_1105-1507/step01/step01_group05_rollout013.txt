    """Solve LP: maximize sum(g) s.t. f*g <= M, g >= 0. Returns g."""
    n = len(f)

            A[k, j] = f[k - j]

        return None

        return None

def local_search(f, rng, n_iter=5000, verbose=False):
    """Random perturbation local search."""
    n = len(f)
    f = f.copy().astype(np.float64)
    f = f / f.sum()
    best = f.copy()
    best_score = compute_score_normalized(f)
    
    for i in range(n_iter):
        f_cand = f.copy()
        
        # Choose perturbation type
        r = rng.random()
        if r < 0.3:
            # Block scaling
            width = max(1, int(rng.integers(1, max(2, n // 16))))
            start = int(rng.integers(0, n - width))
            f_cand[start:start+width] *= np.exp(rng.normal(0, 0.2))
        elif r < 0.6:
            # Single element
            idx = int(rng.integers(0, n))
            f_cand[idx] *= np.exp(rng.normal(0, 0.3))
        elif r < 0.8:
            # Swap-and-scale two blocks
            w1 = max(1, int(rng.integers(1, max(2, n // 32))))
            w2 = max(1, int(rng.integers(1, max(2, n // 32))))
            s1 = int(rng.integers(0, n - w1))
            s2 = int(rng.integers(0, n - w2))
            f_cand[s1:s1+w1] *= np.exp(rng.normal(0, 0.2))
            f_cand[s2:s2+w2] *= np.exp(rng.normal(0, 0.2))
        else:
            # Gaussian noise on subset
            subset = rng.choice(n, size=n//4, replace=False)
            f_cand[subset] *= np.exp(rng.normal(0, 0.15, size=len(subset)))
        
        f_cand = np.clip(f_cand, 0, None)
        S = f_cand.sum()
        if S < 1e-12:
            continue
        f_cand = f_cand / S
        
        s = compute_score_normalized(f_cand)
        if s < best_score:
            best_score = s
            best = f_cand.copy()
            f = f_cand
    
    return best, best_score

        W = np.fft.rfft(w_ext)
        F = np.fft.rfft(f_ext)
        corr = np.fft.irfft(W * np.conj(F), N_fft)

        # corr[j] = sum_m w_ext[(j+m) mod N_fft] * f[m]
        # We want corr[j] for j=0..n-1 (valid range since j+m < 2n-1 <= N_fft)
        grad = 2.0 * corr[:n]

        # Gradient step with projection
        f_new = f - lr * grad
        f_new = np.clip(f_new, 0, None)
        S = f_new.sum()
        if S < 1e-12:
            lr *= 0.5
            continue
        f_new = f_new / S

        s = compute_score_normalized(f_new)
        if s < best_score - 1e-12:
            best_score = s
            best_f = f_new.copy()
            f = f_new
            stall = 0
            lr = min(lr * 1.02, 0.05)
        else:
            stall += 1
            lr *= 0.85
            if stall > 30:
                # Adaptive tau reduction and perturbation
                tau = max(tau * 0.8, 0.0005)
                lr = max(lr, 0.001)
                f_pert = f + rng.normal(0, 0.01, n)
                f_pert = np.clip(f_pert, 0, None)
                S_p = f_pert.sum()
                if S_p > 1e-10:
                    f_pert /= S_p
                    s_p = compute_score_normalized(f_pert)
                    if s_p < best_score:
                        f = f_pert
                        best_f = f.copy()
                        best_score = s_p
                        stall = 0
                        lr = lr_init * 0.5
                if time.monotonic() - start > time_budget * 0.85:
                    break

        # Slowly decrease tau for sharper max approximation
        tau = max(tau * 0.999, 0.0005)

        if verbose and it % 50 == 0:
            print(f"    GD it={it}, score={best_score:.8f}, lr={lr:.6f}, tau={tau:.6f}, elapsed={time.monotonic()-start:.1f}s")

        it += 1

    return best_f, best_score


def local_search(f, rng, n_iter=10000, verbose=False):
    """Diverse local search with multiple perturbation strategies."""
    n = len(f)
    f = f.astype(np.float64).copy()
    f = f / f.sum()
    best = f.copy()
    best_score = compute_score_normalized(f)
    accept_ratio = 0.5

    for i in range(n_iter):
        f_cand = f.copy()
        r = rng.random()

        if r < 0.25:
            # Block scaling
            width = max(1, int(rng.integers(1, max(2, n // 16))))
            start = int(rng.integers(0, n - width))
            f_cand[start:start + width] *= np.exp(rng.normal(0, 0.15))
        elif r < 0.45:
            # Single element
            idx = int(rng.integers(0, n))
            f_cand[idx] *= np.exp(rng.normal(0, 0.25))
        elif r < 0.60:
            # Two blocks
            w1 = max(1, int(rng.integers(1, max(2, n // 32))))
            w2 = max(1, int(rng.integers(1, max(2, n // 32))))
            s1 = int(rng.integers(0, n - w1))
            s2 = int(rng.integers(0, n - w2))
            f_cand[s1:s1 + w1] *= np.exp(rng.normal(0, 0.2))
            f_cand[s2:s2 + w2] *= np.exp(rng.normal(0, 0.2))
        elif r < 0.80:
            # Gaussian noise on random subset
            n_sub = max(1, n // 8)
            subset = rng.choice(n, size=n_sub, replace=False)
            f_cand[subset] *= np.exp(rng.normal(0, 0.1, size=n_sub))
        else:
            # Smooth perturbation (low-frequency)
            perm = rng.permutation(n)
            perm[:n // 4] = rng.choice(n, size=n // 4, replace=False)
            f_cand[perm] *= np.exp(rng.normal(0, 0.1, n))

        f_cand = np.clip(f_cand, 0, None)
        S = f_cand.sum()
        if S < 1e-12:
            continue
        f_cand = f_cand / S

        s = compute_score_normalized(f_cand)
        if s < best_score:
            best_score = s
            best = f_cand.copy()
            f = f_cand
        elif rng.random() < accept_ratio * np.exp(-(s - best_score) / 0.001):
            f = f_cand

        # Decrease acceptance over time
        accept_ratio = max(0.01, accept_ratio * 0.999)

        if verbose and i % 5000 == 0:
            print(f"    LS i={i}, best={best_score:.8f}")

    return best, best_score


def iterative_reweighting(f, time_budget, rng, verbose=True):
    """Iteratively reduce elements contributing to autocorrelation peaks."""
    n = len(f)
    f = f.astype(np.float64).copy()
    f = f / f.sum()

    best_f = f.copy()
    best_score = compute_score_normalized(f)
    t = 0.3

    start = time.monotonic()
    it = 0

    while time.monotonic() - start < time_budget:
        conv = np.convolve(f, f)
        M = conv.max()

        # Compute how much each f[j] contributes to peak values
        # Reduce f[j] proportionally to its contribution to high conv values
        contribution = np.zeros(n)
        threshold = 0.85 * M

        for k in range(2 * n - 1):
            if conv[k] > threshold:
                w = (conv[k] - threshold) / (M - threshold + 1e-15)
                j_start = max(0, k - n + 1)
                j_end = min(n, k + 1)
                for j in range(j_start, j_end):
                    contribution[j] += w * f[k - j]

        if contribution.max() < 1e-15:
            break

        # Multiplicative update: reduce high-contribution elements
        scale = 1.0 - t * contribution / (contribution.max() + 1e-15)
        scale = np.clip(scale, 0.01, None)
        f_new = f * scale
        f_new = np.clip(f_new, 0, None)
        S = f_new.sum()
        if S < 1e-12:
            t *= 0.5
            continue
        f_new = f_new / S

        s = compute_score_normalized(f_new)
        if s < best_score - 1e-10:
            best_score = s
            best_f = f_new.copy()
            f = f_new
            t = min(t * 1.05, 0.5)
        else:
            t *= 0.8

        if verbose and it % 20 == 0:
            print(f"    IW it={it}, score={best_score:.8f}, t={t:.4f}, elapsed={time.monotonic()-start:.1f}s")

        it += 1

    return best_f, best_score


def propose_candidate(seed=42, budget_s=1500, **kwargs):
    """Main search function combining multiple optimization strategies."""
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + budget_s - 10.0

    best_overall = None
    best_score_overall = np.inf

    # Get parent sequence if available
    parent = None
    try:
        parent = np.array(height_sequence_1, dtype=np.float64)
        if parent.ndim != 1 or len(parent) == 0 or not np.isfinite(parent).all():
            parent = None
    except Exception:
        parent = None

    if parent is not None:
        parent = np.clip(parent, 0, 1000)
        S_p = parent.sum()
        if S_p < 0.01:
            parent = None
        else:
            parent = parent / S_p
            parent_score = compute_score_normalized(parent)
            print(f"Parent: n={len(parent)}, score={parent_score:.6f}")

    results = []  # (f, score, n, name)

    def add_result(f, score, n, name):
        nonlocal best_overall, best_score_overall
        if np.isfinite(score) and score < best_score_overall:
            best_score_overall = score
            best_overall = f.copy()
            print(f"  ** NEW BEST: {score:.8f} ({name}) **")
        results.append((f, score, n, name))

    # === Phase 1: LP optimization on multiple starts ===
    print("\n===== PHASE 1: LP Optimization =====")

    # 1a: Parent (if available)
    if parent is not None:
        n_p = len(parent)
        budget = min(150, max(40, (deadline - time.monotonic()) * 0.25))
        print(f"\n[LP parent] n={n_p}, budget={budget:.0f}s")
        f_opt, score = lp_optimize(parent, budget, rng, t0=0.5, verbose=True)
        add_result(f_opt, score, n_p, f"lp_parent_{n_p}")

    # 1b: Uniform with perturbation at various lengths
    for n in [256, 512, 1024, 2048]:
        if time.monotonic() > deadline - 60:
            break
        budget = min(80, max(25, (deadline - time.monotonic()) * 0.15))
        f0 = np.ones(n) * (1 + 0.02 * rng.normal(0, 1, n))
        f0 = np.clip(f0, 0.05, None)
        f0 = f0 / f0.sum()
        print(f"\n[LP uniform] n={n}, budget={budget:.0f}s")
        f_opt, score = lp_optimize(f0, budget, rng, t0=0.4, verbose=(n <= 512))
        add_result(f_opt, score, n, f"lp_uniform_{n}")

    # 1c: Ramp initialization
    for n in [256, 512]:
        if time.monotonic() > deadline - 60:
            break
        budget = min(60, max(20, (deadline - time.monotonic()) * 0.1))
        f0 = np.arange(1, n + 1, dtype=np.float64)
        f0 = f0 / f0.sum()
        print(f"\n[LP ramp] n={n}, budget={budget:.0f}s")
        f_opt, score = lp_optimize(f0, budget, rng, t0=0.4, verbose=False)
        add_result(f_opt, score, n, f"lp_ramp_{n}")

    # === Phase 2: FFT-based gradient descent on best results ===
    print("\n===== PHASE 2: Gradient Descent (FFT) =====")

    if results:
        results.sort(key=lambda x: x[1])
        for f_best, s_best, n_best, name in results[:4]:
            if time.monotonic() > deadline - 40:
                break
            budget = min(150, max(30, (deadline - time.monotonic()) * 0.2))
            print(f"\n[GD] from {name}, n={n_best}, budget={budget:.0f}s")
            f_opt, score = gradient_descent_fft(f_best, budget, rng,
                                                 tau_init=0.005, lr_init=0.003,
                                                 verbose=(n_best <= 512))
            add_result(f_opt, score, n_best, f"gd_{name}")

    # === Phase 3: Local search polishing ===
    print("\n===== PHASE 3: Local Search =====")

    if results:
        results.sort(key=lambda x: x[1])
        for f_best, s_best, n_best, name in results[:4]:
            if time.monotonic() > deadline - 20:
                break
            n_iter = min(20000, max(2000, int((deadline - time.monotonic()) * 100)))
            print(f"\n[LS] from {name}, n={n_best}, iters={n_iter}")
            f_opt, score = local_search(f_best, rng, n_iter=n_iter, verbose=False)
            add_result(f_opt, score, n_best, f"ls_{name}")

    # === Phase 4: Iterative reweighting + second LP round ===
    print("\n===== PHASE 4: Refinement =====")

    if results:
        results.sort(key=lambda x: x[1])
        if time.monotonic() < deadline - 40:
            f_best, s_best, n_best, name = results[0]
            budget = min(80, max(20, (deadline - time.monotonic()) * 0.3))
            print(f"\n[IW] from {name}, n={n_best}, budget={budget:.0f}s")
            f_opt, score = iterative_reweighting(f_best, budget, rng, verbose=True)
            add_result(f_opt, score, n_best, f"iw_{name}")

            # Second LP round with perturbation
            if time.monotonic() < deadline - 30:
                perturbed = f_opt * (1 + 0.15 * rng.normal(0, 1, n_best))
                perturbed = np.clip(perturbed, 0, None)
                S_p = perturbed.sum()
                if S_p > 1e-10:
                    perturbed /= S_p
                    budget2 = min(60, max(15, deadline - time.monotonic() - 10))
                    print(f"\n[LP2] perturbed {name}, budget={budget2:.0f}s")
                    f_opt2, score2 = lp_optimize(perturbed, budget2, rng, t0=0.3, verbose=True)
                    add_result(f_opt2, score2, n_best, f"lp2_{name}")

    # === Phase 5: Final local search on overall best ===
    if results:
        results.sort(key=lambda x: x[1])
        if time.monotonic() < deadline - 10:
            f_best, s_best, n_best, name = results[0]
            n_iter = min(15000, max(1000, int((deadline - time.monotonic() - 5) * 200)))
            print(f"\n[Final LS] from {name}, n={n_best}, iters={n_iter}")
            f_opt, score = local_search(f_best, rng, n_iter=n_iter, verbose=False)
            add_result(f_opt, score, n_best, f"final_ls_{name}")

    # === Final verification ===
    if best_overall is not None:
        best_overall = np.clip(best_overall, 0, 1000)
        final_list = best_overall.tolist()
        final_score = evaluate_sequence(final_list)
        print(f"\n{'='*50}")
        print(f"FINAL SCORE: {final_score:.8f}")
        print(f"Sequence length: {len(final_list)}")
        print(f"Target: 1.5027")
        print(f"{'='*50}")
        return final_list

    # Fallback
    print("Fallback: uniform 256")
    return [1.0] * 256
