def normalize(a):
    """Normalize to sum 1, clip to [0, 1000]."""
    a = np.asarray(a, dtype=np.float64)
    a = np.clip(a, 0, 1000)
    s = a.sum()
    if s <= 1e-300 or not np.isfinite(s):
        return None
    return a / s

def score(a):
    """Compute the score (assuming sum(a) = 1)."""
    a = np.asarray(a, dtype=np.float64)
    n = len(a)
    c = np.convolve(a, a)
    return 2.0 * n * c.max()

def sym_expand(h, n):
    """Expand half-sequence to full symmetric sequence of length n."""
    m = (n + 1) // 2
    if n % 2 == 0:
        return np.concatenate([h[:m], h[:m][::-1]])
    else:
        return np.concatenate([h[:m], h[1:m][::-1]])

def sym_fold(a, n):
    """Fold symmetric sequence to half."""
    m = (n + 1) // 2
    if n % 2 == 0:
        return (a[:m] + a[m:][::-1]) / 2.0
    else:
        h = a[:m].copy()
        h[-1] = a[-1]  # center element
        return h

def lp_direction(f, n, M, max_constraints=None):
    """Solve LP to find direction g that maximizes sum(g) s.t. conv(f,g) <= M, g >= 0."""
    # Build constraint matrix A where A[k,j] = f[k-j] if valid
    # For symmetry: only need k = 0, ..., n-1 (by symmetry of convolution)
    rows = n if max_constraints is None else min(n, max_constraints)
    
    # Actually, let's use all 2n-1 constraints for correctness
    # But for efficiency, we can use just n (the first half + center)
    # since conv(f,g)[k] = conv(f,g)[2n-2-k] when f is symmetric
    
    if f is None or n < 2:
        return None
    
    # For non-symmetric f, use all constraints
    num_rows = 2 * n - 1
    A = np.zeros((num_rows, n), dtype=np.float64)
    for k in range(num_rows):
        for j in range(n):
            i = k - j
            if 0 <= i < n:
                A[k, j] = f[i]
    
    # This is O(n^2) to build, which for n=512 is ~524K operations. Fine.
    
    b = np.full(num_rows, M, dtype=np.float64)
    c_obj = -np.ones(n, dtype=np.float64)  # maximize sum = minimize -sum
    
    res = linprog(c_obj, A_ub=A, b_ub=b, bounds=[(0, None)] * n, method='highs')
    
    if not res.success:
        return None
    
    g = res.x
    s = g.sum()
    if s <= 1e-300:
        return None
    return g / s

def lp_direction_fast(f, n, M):
    """Faster LP direction using structured matrix."""
    if f is None or n < 2:
        return None
    
    num_rows = 2 * n - 1
    # Instead of explicit loop, use the structure
    # A[k, j] = f[k-j] if 0 <= k-j < n else 0
    # This is a "convolution matrix" (also called a Toeplitz-like matrix)
    
    A = np.zeros((num_rows, n), dtype=np.float64)
    for k in range(num_rows):
        # j ranges from max(0, k-n+1) to min(k, n-1)
        j_start = max(0, k - n + 1)
        j_end = min(k, n - 1)
        if j_start <= j_end:
            i_vals = k - np.arange(j_start, j_end + 1)
            A[k, j_start:j_end+1] = f[i_vals]
    
    b = np.full(num_rows, M, dtype=np.float64)
    c_obj = -np.ones(n, dtype=np.float64)
    
    res = linprog(c_obj, A_ub=A, b_ub=b, bounds=[(0, None)] * n, method='highs',
                  options={'presolve': True})
    
    if not res.success:
        return None
    
    g = res.x
    s = g.sum()
    if s <= 1e-300:
        return None
    return g / s

    c_shifted = c - c_max
    smooth_max = c_max + np.log(np.sum(np.exp(tau * c_shifted))) / tau
    return 2.0 * n * smooth_max
