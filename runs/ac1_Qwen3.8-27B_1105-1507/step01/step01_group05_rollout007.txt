```python
"""
Gradient-based optimization with softmax relaxation for minimizing
2*n*max(convolve(a,a))/sum(a)^2 over non-negative sequences.
Uses Adam optimizer with temperature annealing, multiple asymmetric initializations,
and zeroing-based polishing to beat the previous best of 1.5156.
"""

import numpy as np
import time


def compute_score(a):
    """Compute 2*n*max(conv(a,a))/sum(a)^2 for non-negative sequence a."""
    n = len(a)
    S = a.sum()
    if S < 1e-10:
        return np.inf
    conv = np.convolve(a, a)
    M = conv.max()
    return 2.0 * n * M / (S * S)


def smooth_gradient(a, tau, n):
    """Compute gradient of 2n*tau*log(sum exp(conv/tau)) w.r.t. a."""
    conv = np.convolve(a, a)
    if tau < 1e-8:
        k_max = int(np.argmax(conv))
        grad = np.zeros(n)
        for i in range(n):
            idx = k_max - i
            if 0 <= idx < n:
                grad[i] = 4.0 * n * a[idx]
        return grad
    shifted = conv / tau - np.max(conv / tau)
    exp_vals = np.exp(shifted)
    p = exp_vals / np.sum(exp_vals)
    grad = 4.0 * n * np.correlate(p, a, mode='valid')
    return grad


def adam_optimize(a0, tau_schedule, lr=0.05, beta1=0.9, beta2=0.999, eps=1e-8,
                  max_steps_per_tau=800, time_budget=60.0, rng=None, verbose=False):
    """Adam optimization with temperature annealing on the smooth objective."""
    if rng is None:
        rng = np.random.default_rng(0)
    n = len(a0)
    a = a0.copy().astype(np.float64)
    S = a.sum()
    if S < 1e-10:
        return a, np.inf
    a = a / S

    best_a = a.copy()
    best_score = compute_score(a)

    m = np.zeros(n)
    v = np.zeros(n)
    t = 0

    start_time = time.monotonic()
    current_lr = lr

    for tau_idx, tau in enumerate(tau_schedule):
        if time.monotonic() - start_time > time_budget:
            break
        # Decay learning rate somewhat as tau decreases
        current_lr = lr * (0.7 ** tau_idx)

        for step in range(max_steps_per_tau):
            if time.monotonic() - start_time > time_budget:
                break
            grad = smooth_gradient(a, tau, n)
            t += 1
            m = beta1 * m + (1 - beta1) * grad
            v = beta2 * v + (1 - beta2) * grad * grad
            m_hat = m / (1 - beta1 ** t)
            v_hat = v / (1 - beta2 ** t)
            a_new = a - current_lr * m_hat / (np.sqrt(v_hat) + eps)
            a_new = np.clip(a_new, 0, None)
            S_new = a_new.sum()
            if S_new < 1e-10:
                a_new = a.copy()
            else:
                a_new = a_new / S_new

            # Evaluate actual score periodically
            if step % 5 == 0 or step == max_steps_per_tau - 1:
                score = compute_score(a_new)
                if score < best_score:
                    best_score = score
                    best_a = a_new.copy()
                    if verbose:
                        elapsed = time.monotonic() - start_time
                        print(f"    tau={tau:.4f}, step={step}, score={score:.6f}, elapsed={elapsed:.1f}s")
            a = a_new

    return best_a, best_score

def block_local_search(a, rng, budget=5.0):
    """Local search via random block mutations."""
    n = len(a)
    a = a.copy()
    best_score = compute_score(a)
    best_a = a.copy()
    start = time.monotonic()
    temp = 0.5

    while time.monotonic() - start < budget:
        width = max(1, int(rng.integers(1, max(2, n // 20))))
        start_idx = int(rng.integers(0, n - width))
        a_new = a.copy()
        scale = np.exp(rng.normal(0, temp))
        a_new[start_idx:start_idx + width] *= scale
        a_new = np.clip(a_new, 0, None)
        S = a_new.sum()
        if S < 1e-10:
            continue
        a_new = a_new / S
        score = compute_score(a_new)
        if score < best_score:
            best_score = score
            best_a = a_new.copy()
            a = a_new
        else:
            # Sometimes accept worse solutions (simulated annealing)
            if rng.random() < np.exp(-(score - best_score) / max(temp * 0.1, 1e-6)):
                a = a_new
        temp *= 0.999

    return best_a, best_score

        for alpha in [2.0, 4.0, 6.0]:
            a = np.exp(-alpha * np.arange(n) / n)
            configs.append((f'exp{alpha:.0f}_{n}', a))
        # Power law decay
        a = (np.arange(n) + 1.0) ** (-1.0)
        configs.append((f'power_{n}', a))
        # Linear with floor
        a = np.linspace(1.0, 0.05, n)
        configs.append((f'linear_{n}', a))

    # Print plan
    print(f"Total configs: {len(configs)}")
    print(f"Time budget: {budget_s}s, deadline in {deadline - time.monotonic():.0f}s")
    if parent is not None:
        print(f"Parent score: {compute_score(parent):.6f}, length: {len(parent)}")
    print(f"Target: 1.5027")

    # Allocate time per config (more for promising ones)
    total_configs = len(configs)

    for idx, (name, a0) in enumerate(configs):
        if time.monotonic() > deadline:
            break
        remaining = deadline - time.monotonic()
        configs_left = total_configs - idx
        # Allocate proportionally but with minimum
        t_alloc = max(8.0, remaining * 0.7 / configs_left)
        t_alloc = min(t_alloc, remaining * 0.4)
        if t_alloc < 5:
            break

        n = len(a0)
        a0 = np.clip(a0, 0, None).astype(np.float64)
        if a0.sum() < 0.01:
            continue

        print(f"\n[{name}] n={n}, alloc_time={t_alloc:.1f}s, remaining={remaining:.0f}s")

        # Run Adam with temperature annealing
        tau_schedule = [2.0, 1.0, 0.5, 0.25, 0.1, 0.05, 0.02, 0.01, 0.005]
        # Adjust max steps based on time
        max_steps = max(100, int(t_alloc * 100 / len(tau_schedule)))

        a_opt, score = adam_optimize(
            a0, tau_schedule, lr=0.05, max_steps_per_tau=max_steps,
            time_budget=t_alloc * 0.7, rng=rng, verbose=(idx < 3)
        )

        if not np.isfinite(score):
            continue

        # Polish with local search if time permits
        t_left = deadline - time.monotonic()
        if t_left > 5:
            polish_time = min(5.0, t_left * 0.3)
            a_opt, score_ls = block_local_search(a_opt, rng, budget=polish_time * 0.5)
            if score_ls < score:
                score = score_ls
            a_opt, score_z = zeroing_polish(a_opt, rng, budget=polish_time * 0.5)
            if score_z < score:
                score = score_z

        if score < best_score_overall:
            best_score_overall = score
            best_overall = a_opt.copy()
            print(f"  ** New best: {score:.6f} (n={n}) **")
        else:
            print(f"  Score: {score:.6f} (best so far: {best_score_overall:.6f})")

    # Final polishing phase on the best solution
    if best_overall is not None:
        t_left = deadline - time.monotonic()
        print(f"\nFinal polishing phase, time left: {t_left:.0f}s")

        if t_left > 10:
            # More intensive local search
            a_pol, s_pol = block_local_search(best_overall, rng, budget=min(30, t_left * 0.3))
            if s_pol < best_score_overall:
                best_score_overall = s_pol
                best_overall = a_pol
                print(f"  After LS polish: {best_score_overall:.6f}")

        if t_left > 15:
            a_pol, s_pol = zeroing_polish(best_overall, rng, budget=min(30, t_left * 0.3))
            if s_pol < best_score_overall:
                best_score_overall = s_pol
                best_overall = a_pol
                print(f"  After zeroing polish: {best_score_overall:.6f}")

        # Try restarting from best with perturbation
        t_left = deadline - time.monotonic()
        if t_left > 20:
            n_best = len(best_overall)
            n_restarts = min(3, int(t_left / 20))
            for r in range(n_restarts):
                if time.monotonic() > deadline:
                    break
                a0 = best_overall * (1 + 0.15 * rng.normal(0, 1, n_best))
                a0 = np.clip(a0, 0, None)
                if a0.sum() < 0.01:
                    continue
                a0 = a0 / a0.sum()
                t_here = min(20.0, (deadline - time.monotonic()) * 0.5)
                if t_here < 5:
                    break
                tau_sched = [0.5, 0.2, 0.1, 0.05, 0.02]
                a_opt, score = adam_optimize(a0, tau_sched, lr=0.03,
                                             max_steps_per_tau=300,
                                             time_budget=t_here, rng=rng)
                if np.isfinite(score) and score < best_score_overall:
                    best_score_overall = score
                    best_overall = a_opt
                    print(f"  Restart {r}: new best {score:.6f}")

    # Final verification
    if best_overall is not None:
        best_overall = np.clip(best_overall, 0, 1000)
        final_list = best_overall.tolist()
        final_score = evaluate_sequence(final_list)
        print(f"\n=== FINAL RESULT ===")
        print(f"Score: {final_score:.6f}")
        print(f"Length: {len(final_list)}")
        print(f"Target: 1.5027")
        print(f"Previous best: 1.515592")
        return final_list

    # Fallback
    print("Fallback: uniform sequence")
    return [1.0] * 256
```
