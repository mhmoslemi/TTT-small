import numpy as np
import time
from scipy.optimize import linprog
from scipy.sparse import csr_matrix


def build_conv_matrix_sparse(f):
    """Build sparse matrix A where (A@g)[k] = convolve(f,g)[k]."""
    n = len(f)
    rows_list = []
    cols_list = []
    vals_list = []
    for k in range(2 * n - 1):
        j_min = max(0, k - (n - 1))
        j_max = min(k, n - 1)
        for j in range(j_min, j_max + 1):
            rows_list.append(k)
            cols_list.append(j)
            vals_list.append(f[k - j])
    rows = np.array(rows_list, dtype=np.int32)
    cols = np.array(cols_list, dtype=np.int32)
    vals = np.array(vals_list, dtype=np.float64)
    return csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))


def evaluate_fast(sequence):
    """Fast internal evaluation (no security checks)."""
    a = np.asarray(sequence, dtype=np.float64)
    a = np.clip(a, 0, 1000)
    n = len(a)
    if n == 0:
        return np.inf
    conv = np.convolve(a, a)
    max_b = np.max(conv)
    sum_a = np.sum(a)
    if sum_a < 0.01:
        return np.inf
    return float(2 * n * max_b / (sum_a ** 2))


def solve_lp_direction(f, M, top_k=1):
    """Solve LP: max sum(g) s.t. g >= 0, top-k conv(f,g) values <= M."""
    n = len(f)
    A = build_conv_matrix_sparse(f)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    try:
        result = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
        if result.success:
            return result.x
    except Exception:
        pass
    return None

def gradient_polish_smoothed(f, deadline, lr_init=0.01, temperature=0.5):
    """
    Gradient polish using a smoothed (log-sum-exp) objective over the

        k = min(5, len(conv_f))
        top_vals = np.sort(conv_f)[-k:]
        scaled = top_vals / (temperature * S)  # scale for numerical stability

        obj = log_sum_exp

        # For the log-sum-exp, gradient is weighted sum of individual gradients
        weights = np.exp(scaled - np.max(scaled)) / np.sum(np.exp(scaled - np.max(scaled)))
        
        # Get the indices of the top peaks

        new_f = f - lr * grad / S  # normalize gradient by S
        new_f = np.clip(new_f, 0.0, 1000.0)
        
        # Renormalize sum
        new_sum = float(np.sum(new_f))
        if new_sum > 1e-10:
            new_f = new_f * (S / new_sum)

def freq_domain_iterate(f, deadline, alpha=0.5):
    """
    Frequency-domain based iteration.
    In the Fourier domain, conv(f,f) corresponds to |FFT(f)|^2.
    We try to flatten the periodogram to reduce time-domain peaks.
    """
    f = f.copy().astype(np.float64)
    best_f = f.copy()
    best_score = evaluate_fast(f)
    n = len(f)
    S = float(np.sum(f))
    
    # Use a larger FFT size to avoid circular convolution effects
    fft_size = 4 * n
    
    for it in range(5000):
        if time.monotonic() > deadline:
            break
        
        F = np.fft.fft(f, n=fft_size)
        power = np.abs(F) ** 2
        
        # Flatten: compress the dynamic range of the power spectrum
        # power_new = power^alpha with alpha < 1 compresses
        power_max = np.max(power)
        if power_max <= 0:
            break
        power_norm = power / power_max
        power_new = power_norm ** alpha * power_max
        
        # Reconstruct: we need F_new such that |F_new|^2 = power_new
        # Keep the phase of F, adjust magnitude
        phase = np.angle(F)
        mag_new = np.sqrt(power_new)
        F_new = mag_new * np.exp(1j * phase)
        
        f_new = np.fft.ifft(F_new).real[:n]
        f_new = np.clip(f_new, 0.0, 1000.0)
        
        # Renormalize
        f_new_sum = float(np.sum(f_new))
        if f_new_sum > 1e-10:
            f_new = f_new * (S / f_new_sum)
        
        new_score = evaluate_fast(f_new)
        if np.isfinite(new_score) and new_score < best_score:
            f = f_new
            best_f = f.copy()
            best_score = new_score
        else:
            # Try less aggressive flattening
            alpha = max(0.1, alpha * 0.95)
    
    return best_f, best_score


def perturbation_restart(f, rng, amplitude=0.1):
    """Add random perturbation to escape local minima."""
    n = len(f)
    S = float(np.sum(f))
    perturbation = rng.uniform(-amplitude, amplitude, size=n)
    f_new = f + perturbation
    f_new = np.clip(f_new, 0.0, 1000.0)
    new_sum = float(np.sum(f_new))
    if new_sum > 1e-10:
        f_new = f_new * (S / new_sum)
    return f_new

def build_conv_matrix_sparse(f):
    """Build sparse matrix A where (A@g)[k] = convolve(f,g)[k]."""
    n = len(f)
    rows_list = []
    cols_list = []
    vals_list = []
    for k in range(2 * n - 1):
        j_min = max(0, k - (n - 1))
        j_max = min(k, n - 1)
        for j in range(j_min, j_max + 1):
            rows_list.append(k)
            cols_list.append(j)
            vals_list.append(f[k - j])
    rows = np.array(rows_list, dtype=np.int32)
    cols = np.array(cols_list, dtype=np.int32)
    vals = np.array(vals_list, dtype=np.float64)
    return csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))


def evaluate_fast(sequence):

    max_b = float(np.max(conv))
    sum_a = float(np.sum(a))
    if sum_a < 0.01:
        return np.inf
    return float(2 * n * max_b / (sum_a ** 2))


def solve_lp_tight(f):

def grad_multi_peak(f, deadline, lr_init=0.003, top_k=5, temperature=0.3):
    """Gradient descent using smoothed multi-peak objective."""
    f = f.copy().astype(np.float64)
    best_f = f.copy()
    best_score = evaluate_fast(f)
    n = len(f)
    S = float(np.sum(f))
    lr = lr_init
    no_improve = 0

    for it in range(20000):
        if time.monotonic() > deadline:
            break

        conv_f = np.convolve(f, f)
        k = min(top_k, len(conv_f))
        top_indices = np.argsort(conv_f)[-k:]
        top_vals = conv_f[top_indices]

        # Log-sum-exp weights
        max_val = np.max(top_vals)
        if max_val <= 0:
            break
        scaled = (top_vals - max_val) / (temperature * S)
        weights = np.exp(scaled)

        for idx in range(k):
            k_val = top_indices[idx]
            w = weights[idx]
            j_min = max(0, k_val - (n - 1))
            j_max = min(k_val, n - 1)
            for j in range(j_min, j_max + 1):
                grad[j] += 2.0 * w * f[k_val - j]

        new_f = f - lr * grad / S
        new_f = np.clip(new_f, 0.0, 1000.0)

            no_improve += 1
            lr *= 0.93
            if no_improve > 300 or lr < 1e-9:
                break

def freq_smooth(f, deadline, alpha_init=0.8):
    """Frequency-domain flattening iteration."""
    f = f.copy().astype(np.float64)
    best_f = f.copy()
    best_score = evaluate_fast(f)
    n = len(f)
    S = float(np.sum(f))
    fft_size = 4 * n
    alpha = alpha_init

    for it in range(3000):
        if time.monotonic() > deadline:
            break

        F = np.fft.fft(f, n=fft_size)
        power = np.abs(F) ** 2
        power_max = np.max(power)
        if power_max <= 0:
            break

        power_norm = power / power_max
        power_new = (power_norm ** alpha) * power_max

        phase = np.angle(F)
        mag_new = np.sqrt(power_new)
        F_new = mag_new * np.exp(1j * phase)

        f_new = np.fft.ifft(F_new).real[:n]
        f_new = np.clip(f_new, 0.0, 1000.0)
        f_new_sum = float(np.sum(f_new))
        if f_new_sum > 1e-10:
            f_new = f_new * (S / f_new_sum)

        new_score = evaluate_fast(f_new)
        if np.isfinite(new_score) and new_score < best_score:
            f = f_new
            best_f = f.copy()
            best_score = new_score
        else:
            alpha = max(0.3, alpha * 0.97)
            if alpha <= 0.31:
                break

    return best_f, best_score

        print(f"No parent: {e}")

    # Phase 1: Tight LP on parent
    if best_sequence is not None and time.monotonic() < deadline - 200:
        print("\n[Phase 1] Tight LP optimization")
        d1 = min(deadline, t0 + 400)
        bf, bs = lp_iterate_tight(best_sequence, d1)
        if bs < best_score:
            best_score = bs
            best_sequence = bf.copy()
            print(f"  Phase 1 result: {best_score:.6f}")

    # Phase 2: Single-peak gradient polish
    if best_sequence is not None and time.monotonic() < deadline - 150:
        print("\n[Phase 2] Single-peak gradient")
        d2 = min(deadline, time.monotonic() + 180)
        gf, gs = grad_single_peak(best_sequence, d2, lr_init=0.004)
        if gs < best_score:
            best_score = gs
            best_sequence = gf.copy()
            print(f"  Phase 2 result: {best_score:.6f}")

    # Phase 3: Multi-peak gradient
    if best_sequence is not None and time.monotonic() < deadline - 120:
        print("\n[Phase 3] Multi-peak gradient")
        d3 = min(deadline, time.monotonic() + 150)
        gf, gs = grad_multi_peak(best_sequence, d3, lr_init=0.002, top_k=5, temperature=0.2)
        if gs < best_score:
            best_score = gs
            best_sequence = gf.copy()
            print(f"  Phase 3 result: {best_score:.6f}")

    # Phase 4: LP refinement after gradients
    if best_sequence is not None and time.monotonic() < deadline - 100:
        print("\n[Phase 4] LP refinement")
        d4 = min(deadline, time.monotonic() + 200)
        bf, bs = lp_iterate_tight(best_sequence, d4)
        if bs < best_score:
            best_score = bs
            best_sequence = bf.copy()
            print(f"  Phase 4 result: {best_score:.6f}")

    # Phase 5: Frequency domain smoothing
    if best_sequence is not None and time.monotonic() < deadline - 60:
        print("\n[Phase 5] Frequency domain")
        d5 = min(deadline, time.monotonic() + 80)
        ff, fs = freq_smooth(best_sequence, d5, alpha_init=0.85)
        if fs < best_score:
            best_score = fs
            best_sequence = ff.copy()
            print(f"  Phase 5 result: {best_score:.6f}")

    # Phase 6: Gradient after freq domain
    if best_sequence is not None and time.monotonic() < deadline - 40:
        print("\n[Phase 6] Post-freq gradient")
        d6 = min(deadline, time.monotonic() + 60)
        gf, gs = grad_single_peak(best_sequence, d6, lr_init=0.003)
        if gs < best_score:
            best_score = gs
            best_sequence = gf.copy()
            print(f"  Phase 6 result: {best_score:.6f}")

    # Phase 7: Perturbation restarts
    if best_sequence is not None and time.monotonic() < deadline - 30:
        print("\n[Phase 7] Perturbation restarts")
        for i in range(8):
            if time.monotonic() > deadline - 5:
                break
            perturbed = make_perturbed(best_sequence, rng, 0.03)
            ps = evaluate_fast(perturbed)
            if ps < best_score + 0.1:
                d7 = min(deadline - 5, time.monotonic() + 25)
                gf, gs = grad_single_peak(perturbed, d7, lr_init=0.003)
                if gs < best_score:
                    best_score = gs
                    best_sequence = gf.copy()
                    print(f"  Restart {i}: {best_score:.6f}")

    # Phase 8: Try longer length (1024) via interpolation
    if parent is not None and time.monotonic() < deadline - 60:
        print("\n[Phase 8] Try n=1024")
        n2 = 1024
        x_old = np.linspace(0, 1, len(parent))
        x_new = np.linspace(0, 1, n2)
        f_interp = np.interp(x_new, x_old, parent)
        f_interp = np.clip(f_interp, 0.0, 1000.0)
        s_interp = evaluate_fast(f_interp)
        print(f"  Interpolated n=1024: {s_interp:.6f}", end="")

        d8 = min(deadline - 40, time.monotonic() + 80)
        bf2, bs2 = lp_iterate_tight(f_interp, d8)
        if bs2 < best_score:
            best_score = bs2
            best_sequence = bf2.copy()
            print(f" -> improved: {bs2:.6f}")
        else:
            print(f" -> no improvement (best={s_interp:.6f})")

        # Quick gradient on the 1024 result
        if time.monotonic() < deadline - 20:
            d8b = min(deadline - 15, time.monotonic() + 30)
            gf2, gs2 = grad_single_peak(bf2, d8b, lr_init=0.003)
            if gs2 < best_score:
                best_score = gs2
                best_sequence = gf2.copy()
                print(f"  n=1024 grad: {gs2:.6f}")

    # Phase 9: Final polish
    if best_sequence is not None and time.monotonic() < deadline - 3:
        print("\n[Phase 9] Final polish")
        gf, gs = grad_single_peak(best_sequence, deadline, lr_init=0.002)
        if gs < best_score:
            best_score = gs
            best_sequence = gf.copy()

    if best_sequence is None:
        best_sequence = np.ones(128)

    # Verify with actual evaluate_sequence
    final_list = [float(x) for x in best_sequence]
    final_score = evaluate_sequence(final_list)
    if not np.isfinite(final_score):
        final_score = best_score

    elapsed = time.monotonic() - t0
    print(f"\nFINAL: n={len(best_sequence)}, score={final_score:.6f}, time={elapsed:.1f}s")
    return final_list
