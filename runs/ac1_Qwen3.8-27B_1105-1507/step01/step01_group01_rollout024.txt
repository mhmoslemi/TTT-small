    theta = (cssv[int(rho_star) - 1] - 1.0) / rho_star
    return np.maximum(v - theta, 0.0)


def _projected_grad_descent(a_init, tau_list, lr_list, iterations_per_stage):
    """Projected gradient descent on the simplex with smooth max."""
    a = np.asarray(a_init, dtype=np.float64)
    a = np.clip(a, 1e-10, None)
    a = a / np.sum(a)
    n = a.size
    best_a = a.copy()
    best_score = _compute_score(a)

    for tau, lr in zip(tau_list, lr_list):
        v = np.zeros(n)  # momentum
        beta = 0.9
        for it in range(iterations_per_stage):
            c = np.convolve(a, a)
            tc = tau * c
            m = np.max(tc)
            exp_tc = np.exp(tc - m)
            p = exp_tc / np.sum(exp_tc)
            grad = 2.0 * correlate(p, a, mode='valid')
            # Project gradient to tangent space
            grad = grad - np.mean(grad)
            v = beta * v + grad
            a_new = _project_simplex(a - lr * v)
            score = _compute_score(a_new)
            if np.isfinite(score) and score < best_score:
                best_a = a_new.copy()
                best_score = score
                a = a_new.copy()
            # Decay learning rate
            lr *= 0.9995

    return best_a, best_score


def _spectral_init(n, seed):
    """Initialize with near-flat spectrum (random phases, flat magnitude)."""
    rng = np.random.default_rng(seed)
    N = 2 * n - 1
    half = N // 2 + 1
    phases = rng.uniform(0, 2 * np.pi, half)
    X = np.ones(half, dtype=complex) * np.exp(1j * phases)
    X_full = np.zeros(N, dtype=complex)
    X_full[:half] = X
    if N % 2 == 1:
        X_full[half:] = np.conj(X[1:half][::-1])
    else:
        X_full[half:] = np.conj(X[1:half - 1][::-1])
    a = np.real(np.fft.ifft(X_full))
    a = a[:n]
    a = np.clip(a, 0, None)
    s = np.sum(a)
    if s < 1e-10:
        # Retry with different seed
        return _spectral_init(n, seed + 1)
    return a / s


def _structured_init(n, seed, kind=0):
    """Various structured initializations."""
    rng = np.random.default_rng(seed)

    if kind == 0:
        # Cosine bump
        x = np.linspace(0, np.pi, n)
        a = np.cos(x) ** 2
        a = np.clip(a, 0.01, None)
        a = a * np.exp(rng.normal(0, 0.1, n))
    elif kind == 1:
        # Gaussian bump
        x = np.linspace(-2, 2, n)
        a = np.exp(-x ** 2)
        a = a * np.exp(rng.normal(0, 0.15, n))
    elif kind == 2:
        # Beta-like shape
        x = np.linspace(0.01, 0.99, n)
        alpha = float(rng.uniform(0.5, 3.0))
        beta = float(rng.uniform(0.5, 3.0))
        a = x ** (alpha - 1) * (1 - x) ** (beta - 1)
    elif kind == 3:
        # Random smooth
        w = int(rng.integers(3, min(11, n + 1)))
        a = rng.normal(size=n)
        kern = np.ones(w) / w
        a = np.convolve(a, kern, mode='same')
        a = np.clip(a, 0.01, None)
    elif kind == 4:
        # Power law
        x = np.arange(1, n + 1, dtype=np.float64)
        p = float(rng.uniform(0.5, 2.5))
        a = x ** (-p)
        a = a * np.exp(rng.normal(0, 0.1, n))
    else:
        # Flat top with random edges
        frac = float(rng.uniform(0.3, 0.8))
        a = np.ones(n)
        edge = int(n * (1 - frac) / 2)
        if edge > 0:
            fade = np.linspace(0, 1, edge)
            a[:edge] = fade
            a[-edge:] = fade[::-1]
        a = a * np.exp(rng.normal(0, 0.1, n))

    a = np.clip(a, 0.01, None)
    a = a / np.sum(a)
    return a


def _lp_step(f, n, M, seed):
    """LP step: find g that maximizes sum(g) subject to convolve(f,g) <= M, g in [0,1]."""
    try:
        from scipy.optimize import linprog
    except ImportError:
        return None

    try:
        # Build constraint matrix: A[k, j] = f[k-j] if valid
        A = np.zeros((2 * n - 1, n), dtype=np.float64)
        for k in range(2 * n - 1):
            s = max(0, k - (n - 1))
            e = min(n - 1, k)
            if s <= e:
                j = np.arange(s, e + 1)
                A[k, j] = f[k - j]

        cvec = -np.ones(n)
        bvec = np.full(2 * n - 1, M * (1 + 1e-9))
        bounds = [(0.0, 0.1)] * n

        res = linprog(cvec, A_ub=A, b_ub=bvec, bounds=bounds, method='highs',
                     options={'presolve': True, 'time_limit': 30})
        if res is None or not res.success:
            return None
        g = np.clip(res.x, 0, None)
        if np.sum(g) < 1e-10:
            return None
        return g
    except Exception:
        return None


def _refine_with_lp(a_init, time_budget, seed):
    """Iteratively apply LP blending steps."""
    a = np.asarray(a_init, dtype=np.float64)
    a = np.clip(a, 1e-10, None)
    a = a / np.sum(a)
    n = a.size
    best_a = a.copy()
    best_score = _compute_score(a)

    t0 = time.monotonic()
    stall = 0

    while (time.monotonic() - t0) < time_budget and stall < 15:
        c = np.convolve(a, a)
        M = float(np.max(c))

        g = _lp_step(a, n, M, seed)
        if g is None:
            break

        # Try different blend ratios
        improved = False
        for t in [0.1, 0.2, 0.3, 0.5, 0.7]:
            new = (1 - t) * a + t * g
            new = np.clip(new, 0, None)
            s = np.sum(new)
            if s < 1e-10:
                continue
            new = new / s
            sc = _compute_score(new)
            if np.isfinite(sc) and sc < best_score:
                best_a = new.copy()
                best_score = sc
                a = new.copy()
                improved = True
                break

        if not improved:
            stall += 1
        else:
            stall = 0
            seed += 1

    return best_a, best_score


def _optimize_single(a_init, time_budget, seed):
    """Full optimization pipeline for a single starting sequence."""
    t0 = time.monotonic()
    a = np.asarray(a_init, dtype=np.float64)
    a = np.clip(a, 1e-10, None)
    a = a / np.sum(a)
    n = a.size
    best_a = a.copy()
    best_score = _compute_score(a)

    def elapsed():
        return time.monotonic() - t0

    def remaining():
        return time_budget - elapsed()

    # Phase 1: Smooth-max L-BFGS with low temperature
    if remaining() > 30:
        tau_sched = [5, 20, 50, 100]
        alloc = min(60, remaining() * 0.25)
        c, s = _smooth_max_lBFGS(a, tau_sched, max_iter=150, time_limit=alloc, t_start=t0)
        if s < best_score:
            best_a, best_score = c, s
            a = c
            print(f"  Phase 1 (low tau L-BFGS): {best_score:.6f}", flush=True)

    # Phase 2: Medium temperature L-BFGS
    if remaining() > 30:
        tau_sched = [100, 300, 500, 1000]
        alloc = min(80, remaining() * 0.3)
        c, s = _smooth_max_lBFGS(a, tau_sched, max_iter=200, time_limit=alloc, t_start=t0)
        if s < best_score:
            best_a, best_score = c, s
            a = c
            print(f"  Phase 2 (med tau L-BFGS): {best_score:.6f}", flush=True)

    # Phase 3: High temperature L-BFGS
    if remaining() > 20:
        tau_sched = [1000, 2000, 5000]
        alloc = min(60, remaining() * 0.2)
        c, s = _smooth_max_lBFGS(a, tau_sched, max_iter=300, time_limit=alloc, t_start=t0)
        if s < best_score:
            best_a, best_score = c, s
            a = c
            print(f"  Phase 3 (high tau L-BFGS): {best_score:.6f}", flush=True)

    # Phase 4: Projected gradient refinement
    if remaining() > 20:
        tau_list = [2000, 5000, 10000]
        lr_list = [0.5 / n, 0.3 / n, 0.1 / n]
        alloc = min(40, remaining() * 0.15)
        iters = max(100, int(alloc * 50))
        c, s = _projected_grad_descent(a, tau_list, lr_list, iters)
        if s < best_score:
            best_a, best_score = c, s
            a = c
            print(f"  Phase 4 (proj grad): {best_score:.6f}", flush=True)

    # Phase 5: LP refinement
    if remaining() > 30:
        alloc = min(60, remaining() * 0.3)
        c, s = _refine_with_lp(a, alloc, seed)
        if s < best_score:
            best_a, best_score = c, s
            a = c
            print(f"  Phase 5 (LP): {best_score:.6f}", flush=True)

    # Phase 6: Final high-tau L-BFGS polish
    if remaining() > 15:
        tau_sched = [5000, 10000, 20000, 50000]
        alloc = min(60, remaining() * 0.5)
        c, s = _smooth_max_lBFGS(a, tau_sched, max_iter=500, time_limit=alloc, t_start=t0)
        if s < best_score:
            best_a, best_score = c, s
            print(f"  Phase 6 (final polish): {best_score:.6f}", flush=True)

    return best_a, best_score


def _try_length(n, budget, seed, parent_arr=None):
    """Try to find a good sequence of length n."""
    candidates = []
    scores = []

    # Candidate 1: From parent (interpolated to length n)
    if parent_arr is not None and parent_arr.size > 1:
        old = np.linspace(0, 1, parent_arr.size)
        new = np.linspace(0, 1, n)
        c = np.interp(new, old, parent_arr)
        c = np.clip(c, 0.01, None)
        c = c / np.sum(c)
        candidates.append(c)
        scores.append(_compute_score(c))

    # Candidate 2: Spectral init
    try:
        c = _spectral_init(n, seed)
        candidates.append(c)
        scores.append(_compute_score(c))
    except Exception:
        pass

    # Candidate 3-N: Structured inits
    for kind in range(5):
        try:
            c = _structured_init(n, seed + 100 * kind, kind)
            candidates.append(c)
            scores.append(_compute_score(c))
        except Exception:
            pass

    if not candidates:
        c = np.ones(n) / n
        return c, _compute_score(c)

    # Pick the best few to optimize (up to 3)
    order = np.argsort(scores)
    top_k = min(3, len(order))
    top_indices = order[:top_k]

    best_overall = candidates[top_indices[0]].copy()
    best_score = scores[top_indices[0]]

    per_candidate_budget = budget / top_k

    for i, idx in enumerate(top_indices):
        rem = per_candidate_budget - i * 0
        if rem < 10:
            break
        try:
            c, s = _optimize_single(candidates[idx], rem, seed + 1000 * i)
            if s < best_score:
                best_overall = c
                best_score = s
        except Exception as e:
            print(f"    optimize failed: {e}", flush=True)

    return best_overall, best_score


def propose_candidate(seed=42, budget_s=1500, **kwargs):
    """Main search function. Returns the best sequence found."""
    global _best_global

    start = time.monotonic()
    budget = float(budget_s)
    if not np.isfinite(budget) or budget <= 0:
        budget = 1500.0
    deadline = start + budget - 15.0  # Leave 15s buffer

    # Try to load parent
    parent_arr = None
    try:
        parent = globals().get('height_sequence_1', None)
        if parent is not None:
            parent_arr = np.asarray(parent, dtype=np.float64)
            parent_arr = np.clip(parent_arr, 0.0, 1000.0)
            if parent_arr.size == 0 or not np.isfinite(parent_arr).all():
                parent_arr = None
    except Exception:
        parent_arr = None

    best_seq = None
    best_score = float('inf')

    if parent_arr is not None:
        p = np.clip(parent_arr, 1e-10, None)
        p = p / np.sum(p)
        best_seq = p.copy()
        best_score = _compute_score(p)
        print(f"Parent: length={p.size}, score={best_score:.6f}", flush=True)

    # Length plan based on available time
    if budget >= 1200:
        plan = [
            (512, 500, 0),    # Main: refine parent length
            (512, 300, 1),    # Second attempt at 512 with different starts
            (384, 180, 2),    # Shorter
            (640, 200, 3),    # Longer
            (768, 150, 4),    # Even longer
        ]
    elif budget >= 600:
        plan = [
            (512, 400, 0),
            (384, 120, 1),
            (640, 150, 2),
        ]
    else:
        plan = [
            (512, min(300, budget - 30), 0),
            (384, max(30, budget // 4), 1),
        ]

    seed_cur = int(seed)

    for n, alloc, trial_idx in plan:
        rem = deadline - time.monotonic()
        if rem < 30:
            break
        b = min(float(alloc), rem - 20.0)
        if b < 20:
            break

        print(f"\nTrying length={n}, budget={b:.0f}s, trial={trial_idx}", flush=True)
        t0 = time.monotonic()

        try:
            # For the first trial at parent length, use parent directly
            if trial_idx == 0 and parent_arr is not None and parent_arr.size == n:
                # Extra multi-start: perturb parent
                rng = np.random.default_rng(seed_cur)
                for pert in range(3):
                    rem2 = deadline - time.monotonic()
                    if rem2 < 30:
                        break
                    if pert == 0:
                        init = parent_arr.copy()
                    else:
                        noise = np.exp(rng.normal(0, 0.2 * pert, n))
                        init = parent_arr * noise
                        init = np.clip(init, 1e-8, None)
                        init = init / np.sum(init)

                    alloc_this = min(b / 3, rem2 - 10)
                    if alloc_this < 15:
                        break
                    c, s = _optimize_single(init, alloc_this, seed_cur + 5000 * pert)
                    if s < best_score:
                        best_seq = c
                        best_score = s
                        print(f"  Perturb {pert}: score={best_score:.6f}", flush=True)
                    seed_cur += 7
            else:
                c, s = _try_length(n, b, seed_cur + 10000 * trial_idx, parent_arr)
                if s < best_score:
                    best_seq = c
                    best_score = s
                    print(f"  Length {n} trial {trial_idx}: score={best_score:.6f}", flush=True)
                else:
                    print(f"  Length {n} trial {trial_idx}: score={s:.6f} (not improved)", flush=True)
        except Exception as e:
            print(f"  Length {n} trial {trial_idx} failed: {e}", flush=True)

        seed_cur += 101237
        elapsed = time.monotonic() - t0
        print(f"  Length {n} took {elapsed:.1f}s", flush=True)

    # Final verification
    if best_seq is None:
        best_seq = np.ones(512) / 512

    # Ensure valid output
    final = np.asarray(best_seq, dtype=np.float64)
    final = np.clip(final, 0.0, 1000.0)
    s = np.sum(final)
    if s < 1e-6:
        final = np.ones(512) / 512
    else:
        final = final / s
        # Rescale so that evaluate_sequence gives the right score
        # evaluate_sequence uses the raw values, so we need sum >= 0.01
        # The score is scale-invariant: 2n*max(c/a^2) where c = convolve(a,a)
        # If we scale by k, c scales by k^2, sum scales by k, so score is unchanged.
        # But we need sum >= 0.01 to avoid the inf check.
        pass

    # Verify with official evaluator
    try:
        official = float(evaluate_sequence([float(x) for x in final]))
        if np.isfinite(official):
            if official < best_score:
                best_score = official
            print(f"\nOfficial score: {best_score:.6f}, length={len(final)}", flush=True)
        else:
            # Fallback to parent
            if parent_arr is not None:
                final = np.clip(parent_arr, 0, 1000).astype(np.float64)
                s = np.sum(final)
                if s > 0:
                    final = final / s
                try:
                    official2 = float(evaluate_sequence([float(x) for x in final]))
                    print(f"Fallback parent score: {official2:.6f}", flush=True)
                except Exception:
                    pass
            else:
                final = np.ones(512) / 512
    except Exception as e:
        print(f"Verification failed: {e}", flush=True)

    elapsed = time.monotonic() - start
    print(f"Total time: {elapsed:.1f}s", flush=True)

    return [float(x) for x in final]
