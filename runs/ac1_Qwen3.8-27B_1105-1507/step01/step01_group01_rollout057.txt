```python
"""
Search for low-score non-negative step-function coefficients.

Key ideas:
  - Work with sequences normalized to sum 1.  The score is then
        2 * n * max(conv(a, a)).
  - Use FFT for all convolution / gradient computations, which allows
    longer sequences and many more local-search steps.
  - Several complementary local updates are used:
      * peak-shaving multiplicative updates driven by the largest
        convolution coefficients;
      * smooth mean-square flattening updates toward the flat target
        (1/(2n-1), ..., 1/(2n-1));
      * a spectral Newton / Tikhonov division update A <- (A + F/A)/2,
        where F is the flat polynomial of length 2n-1;
      * an LP direction solving min t subject to sum(g)=1, g>=0 and
        conv(a,g) <= t, inspired by the previous state-of-the-art
        but with proper normalization;
      * random block/noise perturbations and restarts.
  - Start from the parent sequence if available, and also from cheap
    analytic/random starts, including a low-end formal square root of
    the flat polynomial and a principal spectral square root.
  - Search several lengths, focusing on the parent length and larger
    non-symmetric lengths.
"""

import time
import numpy as np

try:
    from scipy.optimize import linprog
    HAVE_SCIPY = True
except Exception:
    HAVE_SCIPY = False

np.seterr(over='ignore', invalid='ignore', divide='ignore')

_FFT_LEN_CACHE = {}
_TARGET_CACHE = {}
_FHAT_CACHE = {}
_PARENT_ARR = None


def _safe_normalize(a):
    """Clip to [0, 1000], remove non-finite values, force sum 1."""
    try:
        arr = np.asarray(a, dtype=np.float64).copy()
    except Exception:
        return None
    if arr.size == 0:
        return None
    bad = ~np.isfinite(arr)
    if bad.any():
        arr[bad] = 0.0
    arr = np.clip(arr, 0.0, 1000.0)
    s = float(np.sum(arr))
    if not np.isfinite(s) or s <= 1e-12:
        return None
    arr = arr / s
    # tiny floor avoids division by zero in algebraic updates
    arr = np.maximum(arr, 1e-15)
    s2 = float(np.sum(arr))
    if not np.isfinite(s2) or s2 <= 1e-12:
        return np.ones(arr.size, dtype=np.float64) / float(arr.size)
    arr = arr / s2
    return arr


def _fft_len(n):
    """Next power of two at least 2n-1."""
    n = int(n)
    if n in _FFT_LEN_CACHE:
        return _FFT_LEN_CACHE[n]
    m = 2 * n - 1
    p = 1
    while p < m:
        p <<= 1
    _FFT_LEN_CACHE[n] = p
    return p


def _target_vec(n):
    n = int(n)
    if n not in _TARGET_CACHE:
        _TARGET_CACHE[n] = np.full(2 * n - 1, 1.0 / float(2 * n - 1), dtype=np.float64)
    return _TARGET_CACHE[n]


def _flat_fhat(n):
    n = int(n)
    if n in _FHAT_CACHE:
        return _FHAT_CACHE[n]
    N = _fft_len(n)
    F = np.zeros(N, dtype=np.float64)
    m = 2 * n - 1
    F[:m] = 1.0 / float(m)
    Fhat = np.fft.rfft(F, N)
    _FHAT_CACHE[n] = (N, Fhat)
    return _FHAT_CACHE[n]


def _full_convolution(a):
    """Full linear convolution a*a using FFT (a is real)."""
    a = np.asarray(a, dtype=np.float64)
    n = a.size
    if n == 0:
        return np.array([], dtype=np.float64)
    if n == 1:
        return np.array([a[0] * a[0]], dtype=np.float64)
    N = _fft_len(n)
    fa = np.fft.rfft(a, N)
    out = np.fft.irfft(fa * fa, n=N)
    return out[:2 * n - 1]


def _score_from_conv(c, n):
    if c.size == 0:
        return float('inf')
    m = float(np.max(c))
    if not np.isfinite(m):
        return float('inf')
    return float(2.0 * int(n) * m)


def _local_score(a):
    """Local surrogate score for normalized or unnormalized input."""
    a = _safe_normalize(a)
    if a is None:
        return float('inf')
    c = _full_convolution(a)
    return _score_from_conv(c, a.size)


def _j_t_apply(r, a):
    """
    Apply the adjoint of the derivative of conv(a,a).

    For r of length 2n-1, returns h of length n where
        h[j] = 2 * sum_k r[k] * a[k-j]
    with valid indices.
    """
    a = np.asarray(a, dtype=np.float64)
    r = np.asarray(r, dtype=np.float64)
    n = a.size
    m = 2 * n - 1
    if r.size < m:
        tmp = np.zeros(m, dtype=np.float64)
        tmp[:r.size] = r[:r.size]
        r = tmp
    else:
        r = r[:m]
    N = _fft_len(n)
    fa = np.fft.rfft(a, N)
    fr = np.fft.rfft(r, N)
    h = np.fft.irfft(np.conjugate(fa) * fr, n=N)
    return 2.0 * h[:n]


def _exp_update_candidates(a, gradient, sign, etas):
    """Multiplicative updates in the direction of a centered gradient."""
    if gradient.size == 0:
        return []
    d = gradient - float(np.mean(gradient))
    scale = float(np.max(np.abs(d)))
    if not np.isfinite(scale) or scale <= 1e-300:
        return []
    out = []
    for eta in etas:
        x = sign * float(eta) * d / scale
        x = np.clip(x, -50.0, 50.0)
        b = a * np.exp(x)
        out.append(b)
    return out


def _peak_shave_candidates(a, c, rng):
    """Reduce coordinates contributing most to large convolution peaks."""
    a = np.asarray(a, dtype=np.float64)
    c = np.asarray(c, dtype=np.float64)
    n = a.size
    m = 2 * n - 1
    maxc = float(np.max(c))
    if not np.isfinite(maxc) or maxc <= 0.0:
        return []

    cands = []
    nv = 1 if n > 2048 else 2
    if n > 2048:
        etas = (0.1, 0.5, 2.0)
    else:
        etas = (0.05, 0.2, 0.8, 2.5)

    ks = [max(4, m // 64), max(8, m // 16), max(16, m // 8), max(32, m // 4)]
    for v in range(nv):
        K = int(rng.choice(ks))
        K = max(1, min(K, m))
        if K >= m:
            idx = np.arange(m)
        else:
            try:
                idx = np.argpartition(c, -K)[-K:]
            except Exception:
                idx = np.argsort(c)[-K:]
        r = np.zeros(m, dtype=np.float64)
        if v == 0:
            r[idx] = c[idx] / maxc
        else:
            rr = c[idx] / maxc
            r[idx] = rr * rr
        s = float(np.sum(r))
        if not np.isfinite(s) or s <= 0.0:
            continue
        r /= s
        h = _j_t_apply(r, a)
        cands.extend(_exp_update_candidates(a, h, -1.0, etas))
    return cands


def _mse_candidates(a, c, rng):
    """Direction toward making all convolution coefficients flat."""
    a = np.asarray(a, dtype=np.float64)
    c = np.asarray(c, dtype=np.float64)
    n = a.size
    r = _target_vec(n) - c
    h = _j_t_apply(r, a)
    if n > 2048:
        etas = (0.05, 0.25)
        neg_eta = (0.05,)
    else:
        etas = (0.05, 0.25, 0.9)
        neg_eta = (0.1,)
    cands = _exp_update_candidates(a, h, +1.0, etas)
    cands.extend(_exp_update_candidates(a, h, -1.0, neg_eta))
    return cands


def _spectral_candidates(a, rng):
    """Spectral Newton / Tikhonov step for A^2 ~= flat polynomial."""
    a = np.asarray(a, dtype=np.float64)
    n = a.size
    try:
        N, Fhat = _flat_fhat(n)
        A = np.fft.rfft(a, N)
        abs2 = np.abs(A) ** 2
        maxabs = float(np.max(abs2)) if abs2.size else 0.0
        if not np.isfinite(maxabs) or maxabs <= 0.0:
            maxabs = 1.0
        eps = 1e-8 * maxabs + 1e-15
        ratio = np.conjugate(A) * Fhat / (abs2 + eps)

        B = np.fft.irfft(ratio, n=N)[:n]
        Avg = np.fft.irfft(0.5 * (A + ratio), n=N)[:n]

        bases = []
        for arr in (B, Avg):
            b1 = arr.copy()
            b1[b1 < 0.0] = 0.0
            bases.append(b1)
            bases.append(np.abs(arr))

        if n > 2048:
            bases = bases[:2]
            ts = (0.4, 1.0)
        else:
            ts = (0.25, 0.55, 1.0)

        cands = []
        for b in bases:
            bn = _safe_normalize(b)
            if bn is None:
                continue
            for t in ts:
                cands.append((1.0 - t) * a + t * bn)
        return cands
    except Exception:
        return []


def _lp_candidates(a, time_left):
    """
    LP direction: min t s.t. sum(g)=1, g>=0, conv(a,g) <= t.
    """
    if not HAVE_SCIPY:
        return []
    a = np.asarray(a, dtype=np.float64)
    n = a.size
    if n > 1200 or time_left < 6.0:
        return []

    rows = 2 * n - 1
    cols = n + 1
    try:
        A = np.zeros((rows, cols), dtype=np.float64)
        A[:, 0] = -1.0
        for j in range(n):
            A[j:j + n, j + 1] = a[j]
    except MemoryError:
        return []
    except Exception:
        return []

    cvec = np.zeros(cols, dtype=np.float64)
    cvec[0] = 1.0
    Aeq = np.zeros((1, cols), dtype=np.float64)
    Aeq[0, 1:] = 1.0
    b_ub = np.zeros(rows, dtype=np.float64)
    bounds = [(0.0, None)] * cols

    try:
        res = linprog(
            cvec,
            A_ub=A,
            b_ub=b_ub,
            A_eq=Aeq,
            b_eq=[1.0],
            bounds=bounds,
            method='highs',
            options={'presolve': True}
        )
    except Exception:
        return []

    if res is None or not getattr(res, 'success', False):
        return []

    g = np.asarray(res.x[1:], dtype=np.float64)
    g = np.clip(g, 0.0, None)
    s = float(np.sum(g))
    if not np.isfinite(s) or s <= 0.0:
        return []
    g /= s
    return [(1.0 - t) * a + t * g for t in (0.1, 0.3, 0.7)]


def _block_candidates(a, rng):
    a = np.asarray(a, dtype=np.float64)
    n = a.size
    cands = []
    if n <= 1:
        return cands

    cands.append(a * np.exp(rng.normal(0.0, 0.05, n)))
    cands.append(a * np.exp(rng.normal(0.0, 0.15, n)))

    wmin = max(2, n // 128)
    wmax = max(wmin + 1, n // 8)
    wmax = min(wmax, n)
    w = int(rng.integers(wmin, wmax + 1))
    start = int(rng.integers(0, n - w + 1))

    b = a.copy()
    b[start:start + w] *= np.exp(rng.normal(0.0, 0.3, w))
    cands.append(b)

    b = a.copy()
    b[start:start + w] = a[start:start + w][::-1]
    cands.append(b)

    w2 = max(2, n // 64)
    w2 = min(w2, n)
    start1 = int(rng.integers(0, n - w2 + 1))
    start2 = int(rng.integers(0, n - w2 + 1))
    b = a.copy()
    if abs(start1 - start2) >= w2:
        tmp = b[start1:start1 + w2].copy()
        b[start1:start1 + w2] = b[start2:start2 + w2]
        b[start2:start2 + w2] = tmp
        cands.append(b)

    return cands


def _low_root_candidate(n):
    """
    One-sided formal square root of the flat polynomial, truncated to n terms.
    Coefficients are positive and decreasing.
    """
    n = int(n)
    if n <= 0:
        return None
    r = np.empty(n, dtype=np.float64)
    val = 1.0
    r[0] = val
    for k in range(n - 1):
        val *= (2.0 * k + 1.0) / (2.0 * k + 2.0)
        r[k + 1] = val
    a = np.sqrt(1.0 / float(2 * n - 1)) * r
    return _safe_normalize(a)


def _spectral_sqrt_candidate(n):
    """Principal square root of flat polynomial in Fourier domain."""
    n = int(n)
    if n <= 0:
        return None
    try:
        N, Fhat = _flat_fhat(n)
        Ahat = np.sqrt(Fhat.astype(np.complex128))
        inv = np.fft.irfft(Ahat, n=N)[:n]

        a = _safe_normalize(np.clip(inv, 0.0, None))
        if a is None:
            a = _safe_normalize(np.abs(inv))
        return a
    except Exception:
        return None


def _random_smooth_candidate(n, rng):
    n = int(n)
    if n <= 0:
        return None
    x = rng.normal(size=n)
    low = 1
    high = min(21, max(2, n + 1))
    w = int(rng.integers(low, high + 1))
    if w > 1:
        kern = np.ones(w, dtype=np.float64) / float(w)
        x = np.convolve(x, kern, mode='same')
    f = rng.uniform(0.5, 3.0)
    ph = rng.uniform(0.0, 2.0 * np.pi)
    amp = rng.uniform(0.1, 0.5)
    x = x + amp * np.sin(2.0 * np.pi * f * np.arange(n, dtype=np.float64) / float(n) + ph)
    x = np.maximum(x, 0.0) + 1e-3
    return _safe_normalize(x)


def _beta_candidate(n, rng):
    n = int(n)
    if n <= 0:
        return None
    a = rng.beta(2.0, 2.0, n)
    f = rng.uniform(0.3, 4.0)
    ph = rng.uniform(0.0, 2.0 * np.pi)
    amp = rng.uniform(0.2, 0.8)
    a = a + amp * np.sin(2.0 * np.pi * f * np.arange(n, dtype=np.float64) / float(n) + ph)
    a = np.maximum(a, 1e-4)
    return _safe_normalize(a)


def _binary_candidate(n, rng):
    n = int(n)
    if n <= 0:
        return None
    x = rng.integers(0, 2, n).astype(np.float64)
    x = 2.0 * x - 1.0
    alpha = rng.uniform(0.2, 1.0)
    a = 0.5 * (1.0 + alpha * x)
    w = int(rng.integers(1, 4))
    if w > 1:
        a = np.convolve(a, np.ones(w) / float(w), mode='same')
    a = np.maximum(a, 1e-4)
    return _safe_normalize(a)


def _interpolate_parent(parent, n, rng):
    parent = np.asarray(parent, dtype=np.float64)
    n = int(n)
    if n <= 0 or parent.size <= 1:
        return _safe_normalize(np.ones(n, dtype=np.float64))
    old = np.linspace(0.0, 1.0, parent.size, dtype=np.float64)
    new = np.linspace(0.0, 1.0, n, dtype=np.float64)
    a = np.interp(new, old, parent)
    a = a + 1e-4
    if rng is not None and n > 1:
        a = a * np.exp(rng.normal(0.0, 0.1, n))
    return _safe_normalize(a)


def _make_initial_set(n, seed):
    rng = np.random.default_rng(seed)
    n = int(n)
    cands = []
    parent = _PARENT_ARR
    if parent is not None and parent.size > 1:
        cands.append(_interpolate_parent(parent, n, rng))
    cands.append(np.full(n, 1.0 / float(n), dtype=np.float64))
    cands.append(_random_smooth_candidate(n, rng))
    cands.append(_low_root_candidate(n))
    cands.append(_spectral_sqrt_candidate(n))
    cands.append(_beta_candidate(n, rng))
    cands.append(_binary_candidate(n, rng))
    return cands


def _make_single_random(n, seed, parent):
    rng = np.random.default_rng(seed)
    n = int(n)
    kind = int(rng.integers(0, 8))
    if kind == 0:
        a = np.full(n, 1.0 / float(n), dtype=np.float64)
    elif kind == 1:
        a = _random_smooth_candidate(n, rng)
    elif kind == 2:
        a = _low_root_candidate(n)
    elif kind == 3:
        a = _low_root_candidate(n)
        if a is not None:
            a = a[::-1]
    elif kind == 4:
        a = _spectral_sqrt_candidate(n)
        if a is None:
            a = _random_smooth_candidate(n, rng)
    elif kind == 5:

    if a is None:

    return _safe_normalize(a)


def _optimize_sequence(a, budget, seed):
    a = _safe_normalize(a)
    if a is None:
        return None, float('inf')

    n = a.size
    best = a.copy()
    bestsc = _local_score(best)
    cur = best.copy()
    cursc = bestsc

    rng = np.random.default_rng(seed)
    t0 = time.monotonic()
    deadline = t0 + max(0.0, float(budget))

    step = 0
    stall = 0
    last_lp = t0
    lp_interval = 25.0 if n <= 512 else 50.0 if n <= 1200 else 1e12

    while time.monotonic() < deadline:
        rem = deadline - time.monotonic()
        if rem < 0.5:
            break

        c = _full_convolution(cur)
        cands = []

        if (HAVE_SCIPY and n <= 1200 and rem > 10.0 and
                (time.monotonic() - last_lp) > lp_interval):
            cands = _lp_candidates(cur, rem)
            if cands:
                last_lp = time.monotonic()

        if not cands:
            r = rng.random()
            if r < 0.45:
                cands = _peak_shave_candidates(cur, c, rng)
            elif r < 0.70:
                cands = _mse_candidates(cur, c, rng)
            elif r < 0.88:
                cands = _spectral_candidates(cur, rng)
            else:
                cands = _block_candidates(cur, rng)

        if not cands:
            cands = _block_candidates(cur, rng)

        best_cand = None
        best_cand_sc = cursc
        improved = False

        for cand in cands:
            if time.monotonic() >= deadline:
                break
            cn = _safe_normalize(cand)
            if cn is None:
                continue
            sc = _local_score(cn)
            if np.isfinite(sc) and sc < bestsc - 1e-12:
                best = cn.copy()
                bestsc = float(sc)
                improved = True
            if np.isfinite(sc) and sc < best_cand_sc:
                best_cand = cn.copy()
                best_cand_sc = float(sc)

        if best_cand is not None and best_cand_sc <= cursc + 1e-12:
            cur = best_cand
            cursc = best_cand_sc
        elif stall > 25 and best_cand is not None:
            temp = max(1e-5, 0.05 * max(0.1, bestsc - 1.001) * float(np.exp(-step / 150.0)))
            if best_cand_sc < max(cursc, bestsc) + temp:
                cur = best_cand
                cursc = best_cand_sc

        if improved:
            stall = 0
        else:
            stall += 1
            if stall > 60:
                if rng.random() < 0.6:
                    sigma = 0.2 + 0.6 * min(1.0, stall / 200.0)
                    newa = best * np.exp(rng.normal(0.0, sigma, n))
                else:
                    newa = _make_single_random(
                        n,
                        seed + step * 7919 + int(rng.integers(0, 1000000)),
                        _PARENT_ARR
                    )
                newn = _safe_normalize(newa)
                if newn is not None:
                    sc = _local_score(newn)
                    if np.isfinite(sc) and sc < bestsc + 0.2:
                        cur = newn
                        cursc = sc
                    else:
                        cur = best.copy()
                        cursc = bestsc
                stall = 0

        step += 1

    return best, float(bestsc)


def _explore_length(n, alloc, seed):
    n = int(n)
    if n < 8 or alloc <= 3.0:
        return None, float('inf')

    t0 = time.monotonic()
    deadline = t0 + max(5.0, float(alloc))

    raw = _make_initial_set(n, seed)
    scored = []
    for x in raw:
        xn = _safe_normalize(x)
        if xn is None:
            continue
        sc = _local_score(xn)
        if np.isfinite(sc):
            scored.append((sc, xn))
        if len(scored) >= 8 or time.monotonic() > deadline - 2.0:
            break

    if not scored:
        return None, float('inf')

    # Select the up-to-3 best initial candidates without using lambdas.
    remaining = scored[:]
    tops = []
    for _ in range(3):
        if not remaining:
            break
        mi = 0
        ms = remaining[0][0]
        for i in range(1, len(remaining)):
            if remaining[i][0] < ms:
                ms = remaining[i][0]
                mi = i
        tops.append(remaining[mi])
        remaining.pop(mi)

    if len(tops) == 1:
        fracs = [1.0]
    elif len(tops) == 2:
        fracs = [0.70, 0.30]
    else:
        fracs = [0.65, 0.25, 0.10]

    best = None
    bestsc = float('inf')

    for i, (_, start) in enumerate(tops):
        rem = deadline - time.monotonic()
        if rem < 5.0:
            break
        b = rem * fracs[i]
        if b < 2.0:
            b = max(2.0, 0.4 * rem)
        b = min(b, rem - 2.0)
        if b <= 0.0:
            continue
        a, sc = _optimize_sequence(start, b, seed + 1000003 * (i + 1))
        if a is not None and np.isfinite(sc) and sc < bestsc:
            bestsc = float(sc)
            best = a.copy()

    return best, float(bestsc) if best is not None else float('inf')


def _official_score(a):
    try:
        vals = [float(x) for x in np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)]
        if not vals:
            return float('inf')
        val = float(evaluate_sequence(vals))
        return val
    except Exception:
        return float('inf')


def propose_candidate(seed=42, budget_s=1500, **kwargs):
    global _PARENT_ARR

    t_start = time.monotonic()
    try:
        budget = float(budget_s)
    except Exception:
        budget = 1500.0
    if not np.isfinite(budget) or budget <= 10.0:
        budget = 1500.0
    deadline = t_start + max(10.0, budget) - 5.0

    parent = globals().get('height_sequence_1', None)
    parent_arr = None
    if parent is not None:
        parent_arr = _safe_normalize(parent)
    _PARENT_ARR = parent_arr

    best_seq = None
    best_local = float('inf')
    best_off_seq = None
    best_off = float('inf')

    if parent_arr is not None:
        psc = _local_score(parent_arr)
        if np.isfinite(psc) and psc < best_local:
            best_local = float(psc)
            best_seq = parent_arr.copy()
        poff = _official_score(parent_arr)
        if np.isfinite(poff) and poff < best_off:
            best_off = float(poff)
            best_off_seq = parent_arr.copy()
        print(f"parent length={parent_arr.size} local={psc:.6f} official={poff:.6f}", flush=True)

    length_set = {512, 1024, 2048, 4096}
    if parent_arr is not None:
        pl = int(parent_arr.size)
        if 16 <= pl <= 4096:
            length_set.add(pl)

    # Avoid too many odd / tiny lengths.
    length_set = {n for n in length_set if 16 <= n <= 4096}
    if not length_set:
        length_set = {512}

    budgets = {
        512: 430.0,
        1024: 330.0,
        2048: 240.0,
        4096: 160.0,
    }
    if parent_arr is not None:
        pl = int(parent_arr.size)
        if pl in length_set:
            budgets[pl] = max(budgets.get(pl, 250.0), 470.0)

    ordered = []
    if parent_arr is not None:
        pl = int(parent_arr.size)
        if pl in length_set:
            ordered.append(pl)
    for n in sorted(length_set):
        if n not in ordered:
            ordered.append(n)

    for idx, n in enumerate(ordered):
        rem = deadline - time.monotonic()
        if rem < 30.0:
            break
        alloc = min(budgets.get(n, 100.0), rem - 10.0)
        if n == int(parent_arr.size if parent_arr is not None else -1):
            alloc = min(max(alloc, 260.0), rem - 10.0)
        if alloc < 20.0:
            continue

        a, sc = _explore_length(n, alloc, int(seed) + 999983 * (idx + 1))
        print(f"length={n} alloc={alloc:.1f}s best_local_here={sc:.6f}", flush=True)
        if a is not None and np.isfinite(sc) and sc < best_local:
            best_local = float(sc)
            best_seq = a.copy()
            # Update official if affordable.
            if rem > 15.0 and n <= 4096:
                off = _official_score(a)
                print(f"official for length={n}: {off:.6f}", flush=True)
                if np.isfinite(off) and off < best_off:
                    best_off = float(off)
                    best_off_seq = a.copy()
                rem = deadline - time.monotonic()

    # Final polish on the best sequence found.
    rem = deadline - time.monotonic()
    if best_seq is not None and rem > 20.0:
        try:
            a, sc = _optimize_sequence(best_seq, min(rem - 5.0, 260.0), int(seed) + 123456789)
            if a is not None and np.isfinite(sc) and sc < best_local:
                best_local = float(sc)
                best_seq = a.copy()
                print(f"final polish local={best_local:.6f}", flush=True)
        except Exception as exc:
            print(f"final polish failed: {exc}", flush=True)

    # Choose the best by official score among leading candidates.
    pool = []
    if best_seq is not None:
        pool.append(best_seq)
    if best_off_seq is not None:
        # avoid duplicate if possible by identity-ish
        pool.append(best_off_seq)
    if parent_arr is not None:
        pool.append(parent_arr)
    if not pool:
        pool.append(np.ones(64, dtype=np.float64) / 64.0)

    chosen = None
    chosen_off = float('inf')
    chosen_local = float('inf')

    seen = 0
    for cand in pool:
        cn = _safe_normalize(cand)
        if cn is None:
            continue
        # crude duplicate skip
        dup = False
        for other in pool[:seen]:
            if other is not None and np.array_equal(cn, _safe_normalize(other)):
                dup = True
                break
        if dup:
            seen += 1
            continue
        seen += 1
        loc = _local_score(cn)
        off = _official_score(cn)
        if np.isfinite(off) and (
            off < chosen_off - 1e-12 or
            (abs(off - chosen_off) <= 1e-9 and np.isfinite(loc) and loc < chosen_local)
        ):
            chosen = cn
            chosen_off = float(off)
            chosen_local = float(loc)
            print(f"final candidate official={off:.6f} local={loc:.6f} length={cn.size}", flush=True)

    if chosen is None:
        if parent_arr is not None:
            chosen = _safe_normalize(parent_arr)
        if chosen is None:
            chosen = np.ones(64, dtype=np.float64) / 64.0

    chosen = np.clip(np.asarray(chosen, dtype=np.float64), 0.0, 1000.0)
    return [float(x) for x in chosen]
```
