Hybrid LP + gradient-based search for minimizing 2*n*max(conv(f,f))/(sum(f))^2.
Uses LP direction finding combined with projected gradient descent on a
smoothed max objective, symmetric sequence optimization, and diverse restarts.

from scipy.optimize import linprog
from scipy.sparse import csr_matrix


def normalize_sequence(arr):
    """Return a nonnegative floating vector normalized to sum 1, or None."""
    try:
        x = np.asarray(arr, dtype=np.float64).ravel()
    except Exception:
        return None
    if x.size == 0:
        return None
    x = np.where(np.isfinite(x), x, 0.0)
    x = np.maximum(x, 0.0)
    s = float(x.sum())
    if s <= 1e-300:
        return None
    x /= s
    return x


def score_normalized(x, n):
    """Score assuming x is nonnegative and sum(x) = 1."""
    conv = np.convolve(x, x)
    m = float(np.max(conv))
    if not np.isfinite(m):
        return float("inf")
    return float(2.0 * n * m)


def smoothed_max_score(x, beta):
    """Compute the smoothed max of the autocorrelation."""
    conv = np.convolve(x, x)
    # Shift to avoid overflow
    conv_shifted = conv - np.max(conv)
    log_sum_exp = np.max(conv_shifted) + np.log(np.sum(np.exp(beta * conv_shifted)))
    return (1.0 / beta) * log_sum_exp


def smoothed_max_gradient(x, beta):
    """Compute the gradient of the smoothed max autocorrelation."""
    n = x.size
    conv = np.convolve(x, x)
    conv_shifted = conv - np.max(conv)
    exp_vals = np.exp(beta * conv_shifted)
    p = exp_vals / exp_vals.sum()  # softmax weights, length 2n-1
    # Gradient: (2/beta) * correlate(p, x)
    grad = (2.0 / beta) * np.correlate(p, x, mode='valid')
    return grad


def project_simplex(v):
    """Project v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = v.size
    if n == 0:
        return v.copy()
    # Water-filling algorithm
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - 1))[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0)


def gradient_descent_simplex(x, beta, max_iters, step_size=None, rtol=1e-8):
    """Run projected gradient descent on the smoothed max autocorrelation."""
    x = x.copy().astype(np.float64)
    n = x.size
    best_x = x.copy()
    best_score = score_normalized(x, n)
    
    if step_size is None:
        step_size = 1.0 / (beta * np.max(x) + 1e-10)
    
    for it in range(max_iters):
        grad = smoothed_max_gradient(x, beta)
        if not np.all(np.isfinite(grad)):
            break
        
        # Backtracking line search
        eta = step_size
        accepted = False
        for _ in range(20):
            x_new = x - eta * grad
            x_proj = project_simplex(x_new)
            sc = score_normalized(x_proj, n)
            if np.isfinite(sc) and sc < best_score:
                x = x_proj
                best_x = x.copy()
                best_score = sc
                accepted = True
                break
            eta *= 0.5
        
        if not accepted:
            break
        
        # Check convergence
        if it > 0 and abs(prev_score - best_score) < rtol * max(1.0, abs(best_score)):
            break
        prev_score = best_score
    
    return best_x, best_score

"""
Hybrid LP + projected gradient + simulated annealing search for minimizing
2*n*max(conv(f,f))/(sum(f))^2. Combines LP direction finding with smoothed-max
gradient polishing, parametric initializations, diverse restarts, and
simulated annealing to escape local minima. Targets sequences of lengths 256-4096.
"""

import time
import numpy as np
from scipy.optimize import linprog
from scipy.sparse import csr_matrix


def normalize_sequence(arr):
    """Return a nonnegative floating vector normalized to sum 1, or None."""
    try:
        x = np.asarray(arr, dtype=np.float64).ravel()
    except Exception:
        return None
    if x.size == 0:
        return None
    x = np.where(np.isfinite(x), x, 0.0)
    x = np.maximum(x, 0.0)
    s = float(x.sum())
    if s <= 1e-300:
        return None
    x /= s
    return x


def score_normalized(x):
    """Score assuming x is nonnegative and sum(x) = 1. Returns 2*n*max(conv(x,x))."""
    n = x.size
    conv = np.convolve(x, x)
    m = float(np.max(conv))
    if not np.isfinite(m):
        return float("inf")
    return float(2.0 * n * m)


def project_simplex(v):
    """Project v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = v.size
    if n == 0:
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not np.any(cond):
        return np.ones(n, dtype=np.float64) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def smoothed_max_gradient(x, beta):
    """Compute gradient of (1/beta)*log(sum exp(beta*c_k)) where c=conv(x,x)."""
    n = x.size
    conv = np.convolve(x, x)
    conv_shifted = conv - np.max(conv)
    exp_vals = np.exp(beta * conv_shifted)
    total = exp_vals.sum()
    if total < 1e-300:
        return np.zeros(n, dtype=np.float64)
    p = exp_vals / total
    grad = (2.0 / beta) * np.correlate(p, x, mode='valid')
    return grad


def gradient_polish(x, beta_start, beta_end, max_iters, time_limit, rng):
    """Run projected gradient descent with animated beta on smoothed max."""
    x = x.copy().astype(np.float64)
    n = x.size
    best_x = x.copy()
    best_score = score_normalized(x, n)
    start_time = time.monotonic()

    for it in range(max_iters):
        if time.monotonic() - start_time > time_limit:
            break
        progress = float(it) / float(max(max_iters, 1))
        beta = beta_start * (beta_end / beta_start) ** progress
        beta = max(50.0, beta)

        grad = smoothed_max_gradient(x, beta)
        if not np.all(np.isfinite(grad)):
            break

        # Backtracking line search
        eta = 1.0 / (beta * float(np.max(x)) + 1.0)
        eta = min(eta, 0.5)
        improved = False
        for _ls in range(15):
            x_new = x - eta * grad
            x_proj = project_simplex(x_new)
            sc = score_normalized(x_proj)
            if np.isfinite(sc) and sc < best_score:
                x = x_proj
                best_x = x.copy()
                best_score = sc
                improved = True
                break
            eta *= 0.5

        if not improved:
            # Try a small random perturbation to escape
            if rng is not None and it < max_iters - 5:
                noise = rng.standard_normal(n) * 0.05 * np.max(x)
                x_new = x + noise
                x = project_simplex(x_new)
            else:
                break

    return best_x, best_score


def make_conv_template(n):
    """Precompute sparse matrix structure for the LP constraint matrix A where A@g = convolve(f, g)."""
    m = 2 * n - 1
    k64 = np.arange(m, dtype=np.int64)
    js = np.maximum(0, k64 - n + 1).astype(np.int64)
    je = np.minimum(n - 1, k64).astype(np.int64)
    counts = (je - js + 1).astype(np.int64)
    total = int(counts.sum())

    cum = np.cumsum(counts)
    starts = np.repeat(cum - counts, counts)
    local = np.arange(total, dtype=np.int64) - starts

    row = np.repeat(k64, counts).astype(np.int32)
    col_base = js[row.astype(np.int64)]
    col = (col_base + local).astype(np.int32)
    offsets = (row.astype(np.int64) - col.astype(np.int64)).astype(np.int32)

    A_off = csr_matrix((offsets, (row, col)), shape=(m, n), dtype=np.float64)
    edge_offsets = A_off.data.astype(np.int32, copy=True)
    A_template = A_off.astype(np.float64).copy()
    A_template.data.fill(0.0)

    c_obj = -np.ones(n, dtype=np.float64)
    bounds = [(0.0, None)] * n
    return A_template, edge_offsets, c_obj, bounds, m


def solve_lp_direction(template, f, M, n):
    """Solve the LP: max sum(g) s.t. (f*g)_k <= M, g >= 0."""
    A_template, edge_offsets, c_obj, bounds, m = template

    A = A_template.copy()
    A.data[:] = f[edge_offsets]
    b = np.full(m, M, dtype=np.float64)

    try:
        res = linprog(c_obj, A_ub=A, b_ub=b, bounds=bounds, method='highs')
    except Exception:
        return None
    return getattr(res, 'x', None)

def mix_noisy(base, rng, eps):
    """Mix base vector with random Dirichlet for diversity."""
    f = normalize_sequence(base)
    if f is None:
        return None
    n = f.size
    if eps <= 1e-12:
        return f
    try:
        mix = rng.dirichlet(np.ones(n, dtype=np.float64))
    except Exception:
        mix = np.ones(n, dtype=np.float64) / n
    if eps >= 1.0:
        f = mix
    else:
        f = (1.0 - eps) * f + eps * mix
    return normalize_sequence(f)


def parametric_starts(n, rng):
    """Generate diverse parametric initial sequences of length n."""
    starts = []
    idx = np.arange(n, dtype=np.float64)
    center = (n - 1) / 2.0
    half = n / 2.0

    # Uniform
    starts.append(np.ones(n, dtype=np.float64) / n)

    # Gaussian shapes
    for sigma_frac in [0.15, 0.2, 0.25, 0.3, 0.35, 0.4]:
        sigma = sigma_frac * n
        f = np.exp(-0.5 * ((idx - center) / sigma) ** 2)
        f = np.maximum(f, 0.0)
        s = f.sum()
        if s > 1e-10:
            starts.append(f / s)

    # Parabolic
    for power in [1.0, 1.5, 2.0, 3.0, 4.0]:
        t = 1.0 - np.abs(idx - center) / half
        f = np.maximum(t, 0.0) ** power
        s = f.sum()
        if s > 1e-10:
            starts.append(f / s)

    # Raised cosine
    for power in [1.0, 2.0, 3.0]:
        t = np.cos(np.pi * (idx - center) / n)
        f = np.maximum(t, 0.0) ** (2 * power)
        s = f.sum()
        if s > 1e-10:
            starts.append(f / s)

    # Asymmetric: concentrated on one side
    for frac in [0.3, 0.4, 0.5, 0.6, 0.7]:
        edge = int(frac * n)
        f = np.zeros(n, dtype=np.float64)
        f[:edge] = 1.0 - np.arange(edge) / max(edge, 1)
        f = np.maximum(f, 0.0)
        s = f.sum()
        if s > 1e-10:
            starts.append(f / s)

    # Bimodal
    for gap_frac in [0.1, 0.2, 0.3]:
        gap = int(gap_frac * n)
        f = np.zeros(n, dtype=np.float64)
        mid = n // 2
        f[:mid - gap // 2] = 1.0
        f[mid + gap // 2:] = 1.0
        s = f.sum()
        if s > 1e-10:
            starts.append(f / s)

    # Random Dirichlet
    for _ in range(3):
        f = rng.dirichlet(np.ones(n, dtype=np.float64))
        starts.append(f)

    # Random with structure
    for _ in range(3):
        alpha = rng.uniform(0.5, 3.0)
        f = rng.dirichlet(np.full(n, alpha, dtype=np.float64))
        starts.append(f)

    # Normalize all
    out = []
    for f in starts:
        g = normalize_sequence(f)
        if g is not None and g.size == n:
            out.append(g)
    return out


def simulate_anneal(x, time_limit, rng, n_perturb=5):
    """Simulated annealing local search on the sequence."""
    n = x.size
    x = x.copy().astype(np.float64)
    best_x = x.copy()
    best_score = score_normalized(x)
    start_time = time.monotonic()
    current_score = best_score
    T = 0.01
    T_min = 1e-6
    cooling = 0.995

    while time.monotonic() - start_time < time_limit:
        # Perturb a random subset of entries
        mask = rng.random(n) < (n_perturb / n)
        if mask.sum() == 0:
            mask[rng.integers(n)] = True
        perturb = 1.0 + rng.standard_normal(int(mask.sum())) * 0.3
        x_new = x.copy()
        x_new[mask] = x_new[mask] * perturb
        x_new = np.maximum(x_new, 0.0)
        s = x_new.sum()
        if s < 1e-10:
            T *= cooling
            continue
        x_new /= s

        new_score = score_normalized(x_new)
        delta = new_score - current_score

        if delta < 0 or rng.random() < np.exp(-delta / max(T, 1e-12)):
            x = x_new
            current_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_x = x.copy()
        else:
            # Don't accept, revert
            pass

        T *= cooling
        if T < T_min:
            # Reheat
            T = 0.005
            x = mix_noisy(best_x, rng, 0.1)

    return best_x, best_score


def interpolate_sequence(x, n):
    """Resample a nonnegative sequence to length n and normalize."""
    x = normalize_sequence(x)
    if x is None or n <= 0:
        return None
    m = x.size
    if m == n:
        return x
    if m == 1:
        return np.ones(n, dtype=np.float64) / n
    old = np.linspace(0.0, 1.0, m, dtype=np.float64)
    new = np.linspace(0.0, 1.0, n, dtype=np.float64)
    out = np.interp(new, old, x)
    return normalize_sequence(out)


def propose_candidate(seed=42, budget_s=1500, **kwargs):
    """
    Search for a low-scoring nonnegative sequence using hybrid LP + gradient + SA.
    Returns the best sequence found within the time budget.
    """
    rng = np.random.default_rng(int(seed) if seed is not None else 42)
    total_budget = max(1.0, float(budget_s))
    reserve = 20.0
    start_time = time.monotonic()
    deadline = start_time + total_budget - reserve

    best_f = None
    best_score = float("inf")
    given_f = None

    # Load given sequence if available
    try:
        raw = np.asarray(height_sequence_1, dtype=np.float64)
        given_f = normalize_sequence(raw)
        if given_f is not None:
            best_f = given_f.copy()
            best_score = score_normalized(given_f)
            print(f"Given sequence: n={given_f.size}, score={best_score:.6f}", flush=True)
    except Exception:
        given_f = None

    if best_f is None:
        best_f = np.ones(1024, dtype=np.float64) / 1024.0
        best_score = 2.0

    # Phase 1: Quick exploration with parametric starts and short LP runs
    print("\n=== Phase 1: Quick exploration ===", flush=True)
    p1_budget = min(80.0, (deadline - time.monotonic()) * 0.15)
    p1_deadline = start_time + p1_budget

    n_trials = [256, 512, 1024, 2048]
    for n in n_trials:
        if time.monotonic() > p1_deadline:
            break
        rem = p1_deadline - time.monotonic()
        if rem < 5.0:
            break

        print(f"  n={n}, time_remaining={rem:.1f}s", flush=True)
        starts = parametric_starts(n, rng)

        # Also include upsampled best
        if best_f is not None:
            up = interpolate_sequence(best_f, n)
            if up is not None:
                starts.append(up)
                starts.append(mix_noisy(up, rng, 0.1))

        # Try a few starts with very short LP
        template = None
        try:
            template = make_conv_template(n)
        except Exception:
            pass

        for idx, s in enumerate(starts[:6]):
            if time.monotonic() > p1_deadline:
                break
            rem = p1_deadline - time.monotonic()
            if rem < 1.0:
                break
            sc = score_normalized(s)
            if sc < best_score:
                best_score = sc
                best_f = s.copy()
                print(f"    parametric start {idx}: score={sc:.6f}", flush=True)
            if template is not None and rem > 3.0:
                f_res, s_res = lp_optimize(n, s, min(3.0, rem - 0.5), rng, template)
                if s_res < best_score:
                    best_score = s_res
                    best_f = f_res.copy()
                    print(f"    quick LP from start {idx}: score={s_res:.6f}", flush=True)

        if template is not None:
            del template

    print(f"  After Phase 1: best={best_score:.6f}, n={best_f.size}", flush=True)

    # Phase 2: Main LP optimization
    print("\n=== Phase 2: Main LP optimization ===", flush=True)
    p2_deadline = start_time + total_budget * 0.65

    # Choose lengths to optimize based on current best
    best_n = best_f.size
    lengths_to_try = []
    for n in [256, 512, 1024, 2048, 3072, 4096]:
        if n >= best_n // 2 and n <= best_n * 3:
            lengths_to_try.append(n)
    if not lengths_to_try:
        lengths_to_try = [512, 1024, 2048]

    # Sort by likelihood of improvement
    lengths_to_try.sort(key=lambda n: abs(n - best_n) / n)

    for n in lengths_to_try[:5]:
        rem = p2_deadline - time.monotonic()
        if rem < 10.0:
            break

        alloc = min(rem * 0.4, 200.0)
        if alloc < 8.0:
            alloc = rem * 0.3
        if alloc < 5.0:
            continue

        print(f"\n  n={n}, alloc={alloc:.1f}s, current_best={best_score:.6f}", flush=True)

        try:
            template = make_conv_template(n)
        except Exception as e:
            print(f"    Template failed: {e}", flush=True)
            continue

        # Generate diverse starts for this length
        starts = []
        if best_f is not None:
            up = interpolate_sequence(best_f, n)
            if up is not None:
                starts.append(up)
                starts.append(mix_noisy(up, rng, 0.05))
                starts.append(mix_noisy(up, rng, 0.15))
        starts.extend(parametric_starts(n, rng)[:4])

        # Run LP from each start with allocated time
        m = len(starts)
        for idx, s in enumerate(starts):
            rem = p2_deadline - time.monotonic()
            if rem < 3.0:
                break
            tb = alloc / (m - idx) * 0.7
            tb = min(tb, rem - 1.0)
            if tb < 2.0:
                continue

            print(f"    start {idx+1}/{m}, budget={tb:.1f}s", flush=True)
            f_res, s_res = lp_optimize(n, s, tb, rng, template)
            if s_res < best_score:
                best_score = s_res
                best_f = np.asarray(f_res, dtype=np.float64).copy()
                print(f"    NEW BEST: n={n}, score={best_score:.6f}", flush=True)

        del template

    print(f"\n  After Phase 2: best={best_score:.6f}, n={best_f.size}", flush=True)

    # Phase 3: Gradient polishing + simulated annealing
    print("\n=== Phase 3: Gradient polish + SA ===", flush=True)
    p3_deadline = start_time + total_budget * 0.9

    if best_f is not None:
        rem = p3_deadline - time.monotonic()
        if rem > 5.0:
            n = best_f.size
            print(f"  Polishing n={n}, budget={rem:.1f}s", flush=True)

            # Gradient polish with increasing beta
            g_budget = min(rem * 0.5, 120.0)
            x_poly, s_poly = gradient_polish(
                best_f, beta_start=100.0, beta_end=5000.0,
                max_iters=500, time_limit=g_budget, rng=rng
            )
            if s_poly < best_score:
                best_score = s_poly
                best_f = x_poly.copy()
                print(f"  Gradient polish improved: {best_score:.6f}", flush=True)

            # Simulated annealing
            sa_budget = min(p3_deadline - time.monotonic(), 150.0)
            if sa_budget > 5.0:
                print(f"  SA budget={sa_budget:.1f}s", flush=True)
                x_sa, s_sa = simulate_anneal(best_f, sa_budget, rng, n_perturb=max(3, n // 100))
                if s_sa < best_score:
                    best_score = s_sa
                    best_f = x_sa.copy()
                    print(f"  SA improved: {best_score:.6f}", flush=True)

            # Final gradient polish after SA
            rem = p3_deadline - time.monotonic()
            if rem > 5.0:
                x_final, s_final = gradient_polish(
                    best_f, beta_start=200.0, beta_end=10000.0,
                    max_iters=300, time_limit=rem, rng=rng
                )
                if s_final < best_score:
                    best_score = s_final
                    best_f = x_final.copy()
                    print(f"  Final gradient polish: {best_score:.6f}", flush=True)

    # Phase 4: Try longer sequences if time permits
    rem = deadline - time.monotonic()
    if rem > 60.0 and best_f is not None:
        print(f"\n=== Phase 4: Longer sequence attempt (rem={rem:.1f}s) ===", flush=True)
        for n_ext in [2048, 4096]:
            if n_ext <= best_f.size:
                continue
            rem = deadline - time.monotonic()
            if rem < 30.0:
                break
            up = interpolate_sequence(best_f, n_ext)
            if up is None:
                continue
            try:
                template = make_conv_template(n_ext)
                f_res, s_res = lp_optimize(n_ext, up, min(rem - 10.0, 60.0), rng, template)
                if s_res < best_score:
                    best_score = s_res
                    best_f = f_res.copy()
                    print(f"  Extension to n={n_ext}: score={best_score:.6f}", flush=True)
                del template
            except Exception:
                pass

    # Final: evaluate with the actual evaluator
    if best_f is None:
        best_f = np.ones(64, dtype=np.float64) / 64.0

    seq = [float(x) for x in best_f]
    try:
        final_score = float(evaluate_sequence(seq))
        print(f"\nFINAL: n={len(seq)}, local_best={best_score:.6f}, evaluate={final_score:.6f}", flush=True)
    except Exception as e:
        print(f"Evaluation failed: {e}", flush=True)
        final_score = float("inf")

    if not np.isfinite(final_score):
        # Fallback
        seq = [float(x) for x in given_f] if given_f is not None else [1.0 / 64.0] * 64
        try:
            final_score = float(evaluate_sequence(seq))
            print(f"Fallback evaluate={final_score:.6f}", flush=True)
        except Exception:
            seq = [1.0 / 64.0] * 64

    return seq
