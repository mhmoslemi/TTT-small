The direction that most reduces conv(f,f)[k*] (with sum constraint) is:
d[j] = -(f[k*-j] + f[j-k*]) + mean(f)  (for the projection onto sum=1)

Wait, more carefully: we want to minimize (f*f)[k*] = sum_i f[i]f[k*-i] subject to sum(f)=1, f>=0.
The gradient of (f*f)[k*] w.r.t. f[j] is f[k*-j] + f[j-k*] (treating out-of-range as 0).
So the steepest descent direction (projected onto sum=1, f>=0) would be:
d[j] = -(f[k*-j] + f[j-k*]) + lambda
where lambda = mean(f[k*-j] + f[j-k*]) (to ensure sum(d)=0).

Then we take a step f <- f + t*d (with t chosen by line search) and clip to f>=0.

This is essentially what the mirror descent was doing, but with a different parameterization. The advantage is that we can use a proper line search.

"""
Optimizes non-negative step function heights to minimize 2n*max(conv(f,f))/(sum(f))^2.
Uses adaptive mirror descent with backtracking line search, non-symmetric
initializations, LP-based refinement with reduced active sets, and multi-resolution
exploration. Improves on previous SOTA by exploiting dual information from LPs
and using diverse (including non-symmetric) starting points.
"""

import numpy as np
import time
import operator


def _fft_conv(f):
    """Full linear convolution of f with itself via FFT."""
    n = len(f)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    F = np.fft.rfft(f, size)
    h = np.fft.irfft(F * F, size)
    return h[:2 * n - 1]


def _score(f):
    """Compute the objective value. Assumes f >= 0, sum(f) > 0."""
    n = len(f)
    h = _fft_conv(f)
    M = float(np.max(h))
    S = float(np.sum(f))
    if S < 0.01:
        return float('inf')
    return 2.0 * n * M / (S * S)


def _normalize(f):
    """Normalize to sum=1, clip negatives."""
    f = np.maximum(f, 0.0)
    s = float(np.sum(f))
    if s < 1e-30:
        return None
    return f / s


def _pressure(f, active_lags):
    """Compute average subgradient over active lags."""
    n = len(f)
    p = np.zeros(n, dtype=np.float64)
    for k in active_lags:
        k = int(k)
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        if j_end >= j_start:
            lo = k - j_end
            hi = k - j_start
            p[j_start:j_end + 1] += 2.0 * f[lo:hi + 1][::-1]
    p /= float(len(active_lags))
    return p


def _pressure_weighted(f, active_lags, weights):
    """Compute weighted subgradient over active lags."""
    n = len(f)
    p = np.zeros(n, dtype=np.float64)
    wsum = 0.0
    for idx, k in enumerate(active_lags):
        k = int(k)
        w = weights[idx]
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        if j_end >= j_start:
            lo = k - j_end
            hi = k - j_start
            p[j_start:j_end + 1] += w * 2.0 * f[lo:hi + 1][::-1]
            wsum += w
    if wsum > 0:
        p /= wsum
    return p


def _mirror_descent_adaptive(f_init, n_iter, eta0, threshold_frac, deadline, label=""):
    """
    Mirror descent with backtracking line search and adaptive active set.
    Uses multiplicative updates: f <- f * exp(-eta * p), then normalize.
    """
    f = np.asarray(f_init, dtype=np.float64).copy()
    f = _normalize(f)
    if f is None:
        return f_init.copy(), float('inf')
    n = len(f)
    best_f = f.copy()
    best_score = _score(f)
    M_prev = float(np.max(_fft_conv(f)))

    for it in range(n_iter):
        if time.monotonic() > deadline:
            break

        h = _fft_conv(f)
        M = float(np.max(h))
        if M <= 0:
            break

        # Adaptive threshold: start loose, tighten over iterations
        progress = min(1.0, it / max(1, n_iter * 0.5))
        tf = threshold_frac * (1.0 - 0.3 * progress)
        threshold = tf * M + 1e-15
        active = np.where(h >= threshold)[0]
        if len(active) == 0:
            active = np.array([int(np.argmax(h))], dtype=np.int64)

        p = _pressure(f, active)

        # Backtracking line search for eta
        eta = eta0 / max(M, 1e-10)
        score_cur = best_score

        for ls in range(12):
            f_trial = f * np.exp(-eta * p)
            f_trial = _normalize(f_trial)
            if f_trial is None:
                eta *= 0.5
                continue
            s_trial = _score(f_trial)
            if s_trial < score_cur - 1e-12:
                f = f_trial
                score_cur = s_trial
                break
            else:
                eta *= 0.5

        if s_trial < best_score:
            best_score = s_trial
            best_f = f.copy()
            M_prev = M

        if (it + 1) % 2000 == 0:
            print(f"    MD[{label}] it={it+1}: best={best_score:.6f}, M={M:.8f}, |K|={len(active)}, eta={eta:.4e}")

    return best_f, best_score


def _lp_refine(f_init, deadline, max_iters=30, label=""):
    """
    LP-based refinement. Solves: max sum(g) s.t. C@g <= M * (1+eps), g >= 0.
    Uses the LP solution as a Frank-Wolfe direction.
    """
    from scipy.optimize import linprog
    from scipy.sparse import coo_matrix

    f = np.asarray(f_init, dtype=np.float64).copy()
    f = _normalize(f)
    if f is None:
        return f_init.copy(), float('inf')
    n = len(f)
    best_f = f.copy()
    best_score = _score(f)

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        h = np.convolve(f, f)
        M = float(np.max(h))

        # Build sparse matrix C: C[k,j] = f[k-j] if 0<=k-j<n
        row_idx = []
        col_idx = []
        vals = []
        for j in range(n):
            rows_j = np.arange(j, j + n)
            row_idx.extend(rows_j.tolist())
            col_idx.extend([j] * n)
            vals.extend(f.tolist())

        C = coo_matrix((vals, (row_idx, col_idx)), shape=(2 * n - 1, n))

        c_obj = -np.ones(n)
        b_ub = np.full(2 * n - 1, M * (1.0 + 1e-6))
        res = linprog(c_obj, A_ub=C, b_ub=b_ub,
                      bounds=[(0, None)] * n, method='highs')

        if not res.success or res.x is None:
            break

        g = np.maximum(res.x, 0.0)
        Sg = float(np.sum(g))
        if Sg < 1e-10:
            break
        g = g / Sg

        # Frank-Wolfe step with adaptive gamma
        gamma = 2.0 / (it + 2)
        f_new = (1.0 - gamma) * f + gamma * g
        f_n = _normalize(f_new)
        if f_n is None:
            break
        f = f_n

        score = _score(f)
        if score < best_score:
            best_score = score
            best_f = f.copy()

        if (it + 1) % 5 == 0:
            print(f"    LP[{label}] it={it+1}: score={best_score:.6f}")

    return best_f, best_score


def _lp_refine_active(f_init, deadline, max_iters=30, n_active=50, label=""):
    """
    LP refinement using only the top n_active constraints (faster).
    """
    from scipy.optimize import linprog
    from scipy.sparse import coo_matrix

    f = np.asarray(f_init, dtype=np.float64).copy()
    f = _normalize(f)
    if f is None:
        return f_init.copy(), float('inf')
    n = len(f)
    best_f = f.copy()
    best_score = _score(f)

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        h = np.convolve(f, f)
        M = float(np.max(h))

        # Select top active lags
        idx = np.argsort(h)[::-1][:n_active]

        # Build reduced constraint matrix
        row_idx = []
        col_idx = []
        vals = []
        b_ub = []
        for k in idx:
            for j in range(n):
                diff = k - j
                if 0 <= diff < n:
                    row_idx.append(len(b_ub))
                    col_idx.append(j)
                    vals.append(f[diff])
            b_ub.append(M * (1.0 + 1e-6))

        C = coo_matrix((vals, (row_idx, col_idx)), shape=(len(b_ub), n))
        c_obj = -np.ones(n)
        res = linprog(c_obj, A_ub=C, b_ub=np.array(b_ub),
                      bounds=[(0, None)] * n, method='highs')

        if not res.success or res.x is None:
            break

        g = np.maximum(res.x, 0.0)
        Sg = float(np.sum(g))
        if Sg < 1e-10:
            break
        g = g / Sg

        gamma = 2.0 / (it + 2)
        f_new = (1.0 - gamma) * f + gamma * g
        f_n = _normalize(f_new)
        if f_n is None:
            break
        f = f_n

        score = _score(f)
        if score < best_score:
            best_score = score
            best_f = f.copy()

    return best_f, best_score


def _init_uniform(n):
    return np.ones(n, dtype=np.float64) / n


def _init_cosine(n):
    j = np.arange(n, dtype=np.float64)
    f = np.cos(np.pi * (j + 0.5) / n) ** 2
    f = np.maximum(f, 0.0)
    s = np.sum(f)
    if s > 0:
        f /= s
    return f


def _init_exponential(n, rate=1.0):
    """Non-symmetric: exponential decay."""
    j = np.arange(n, dtype=np.float64)
    f = np.exp(-rate * j / n)
    s = np.sum(f)
    if s > 0:
        f /= s
    return f


def _init_ramp_asymmetric(n, alpha=1.0):
    """Non-symmetric: smooth ramp."""
    j = np.arange(n, dtype=np.float64) / (n - 1)
    f = (1.0 - j) ** alpha + 0.1
    f = np.maximum(f, 0.0)
    s = np.sum(f)
    if s > 0:
        f /= s
    return f


def _init_bimodal(n, sep=0.3):
    """Non-symmetric: two peaks at different heights."""
    j = np.arange(n, dtype=np.float64)
    center1 = n * (0.5 - sep)
    center2 = n * (0.5 + sep)
    width = n * 0.15
    f = np.exp(-(j - center1) ** 2 / (2 * width ** 2)) + 0.6 * np.exp(-(j - center2) ** 2 / (2 * width ** 2))
    s = np.sum(f)
    if s > 0:
        f /= s
    return f


def _init_random(n, rng):
    f = rng.random(n, dtype=np.float64)
    s = np.sum(f)
    if s > 0:
        f /= s
    return f


def _init_remapped(parent, n):
    """Resample parent to length n via interpolation."""
    m = len(parent)
    if m == n:
        f = parent.copy()
    else:
        x_old = np.linspace(0, 1, m)
        x_new = np.linspace(0, 1, n)
        f = np.interp(x_new, x_old, parent)
    f = _normalize(f)
    return f if f is not None else _init_uniform(n)


def _perturb_and_optimize(f_base, n_iter, eta_scale, threshold, deadline, rng, label=""):
    """Add random perturbation then optimize."""
    n = len(f_base)
    noise = rng.random(n, dtype=np.float64) * 0.05
    f_pert = _normalize(f_base + noise)
    if f_pert is None:
        return f_base.copy(), _score(f_base)
    return _mirror_descent_adaptive(f_pert, n_iter, eta_scale, threshold, deadline, label)
