            best_candidate = old_val
            best_c_score = best_score
            
            for val in candidates:
                f[i] = val
                S_new = f.sum()
                if S_new < 1e-10:
                    f[i] = old_val
                    continue
                f_norm = f / S_new

                    best_candidate = val

                S_new = f.sum()
                f = f / S_new
                best_score = best_c_score
                improved = True
        
        if not improved:
            break
    
    return f, best_score

        for j in range(n):
            idx = k - j
            if 0 <= idx < n:
                A[row_i, j] = f[idx]
    b_ub = conv_f[indices]
    return A, b_ub


def build_full_conv_matrix(f):
    """Build full constraint matrix A such that A @ g = np.convolve(f, g)."""
    n = len(f)
    A = np.zeros((2 * n - 1, n))
    for k in range(2 * n - 1):
        j_lo = max(0, k - n + 1)
        j_hi = min(n, k + 1)
        for j in range(j_lo, j_hi):
            A[k, j] = f[k - j]
    return A


def optimize_lp(f, time_budget, rng, verbose=True, use_active_set=True):
    """LP-based optimization with line search and active-set reduction."""
    n = len(f)
    f = f.astype(np.float64).copy()
    S = f.sum()
    if S < 1e-10:
        return f, np.inf
    f = f / S

    best_f = f.copy()
    best_score = compute_score_normalized(f)

    start = time.monotonic()
    it = 0
    t = 0.3
    stall_count = 0
    max_iter = 2000

    while time.monotonic() - start < time_budget and it < max_iter:
        conv_f = np.convolve(f, f)
        M = conv_f.max()

        # Build LP
        if use_active_set and n > 256:
            A, b_ub = build_conv_matrix_active(f, conv_f, threshold=0.85)
        else:
            A = build_full_conv_matrix(f)
            b_ub = np.full(2 * n - 1, M)

        c_obj = -np.ones(n)
        bounds = [(0, None)] * n

        try:
            res = linprog(c_obj, A_ub=A, b_ub=b_ub, bounds=bounds,
                         method='highs', options={'presolve': True})
        except Exception:
            break

        if not res.success or res.x is None:
            break

        g = np.clip(res.x, 0, None)
        g_sum = g.sum()
        if g_sum < 1e-15:
            break
        g = g / g_sum

        # Line search: try multiple step sizes
        best_t = 0
        best_f_try = None
        best_score_try = best_score

        t_candidates = [t, t * 0.5, t * 1.5, t * 0.25, 0.1, 0.05]
        for t_c in t_candidates:
            f_try = (1 - t_c) * f + t_c * g
            f_try = np.clip(f_try, 0, None)
            S_try = f_try.sum()
            if S_try < 1e-10:
                continue
            f_try = f_try / S_try
            score_try = compute_score_normalized(f_try)
            if score_try < best_score_try - 1e-12:
                best_score_try = score_try
                best_f_try = f_try.copy()
                best_t = t_c

        if best_f_try is not None:
            best_f = best_f_try
            best_score = best_score_try
            f = best_f_try
            t = min(0.8, max(0.05, best_t * 1.05))
            stall_count = 0
            if verbose and (it % 15 == 0 or it < 3):
                elapsed = time.monotonic() - start
                print(f"    LP it={it}, score={best_score:.8f}, t={t:.3f}, elapsed={elapsed:.1f}s")
        else:
            t *= 0.6
            stall_count += 1
            if stall_count > 20:
                # Escapation: directed perturbation at peak
                conv_cur = np.convolve(f, f)
                peak_idx = np.argmax(conv_cur)
                # Perturb elements that contribute to the peak
                f_pert = f.copy()
                for j in range(n):
                    k_idx = peak_idx - j
                    if 0 <= k_idx < n:
                        f_pert[j] *= (1 - 0.05 * rng.random())
                f_pert = np.clip(f_pert, 0, None)
                Sp = f_pert.sum()
                if Sp > 1e-10:
                    f_pert /= Sp
                    score_pert = compute_score_normalized(f_pert)
                    if score_pert < best_score:
                        f = f_pert
                        best_f = f.copy()
                        best_score = score_pert
                        t = 0.2
                        stall_count = 0
                if time.monotonic() - start > time_budget * 0.85:
                    break

        it += 1

    return best_f, best_score


def coordinate_descent(f, time_budget=60, rng=None, n_sweeps=50):
    """Coordinate descent with adaptive candidate values."""
    if rng is None:
        rng = np.random.default_rng(0)

    n = len(f)
    f = f.astype(np.float64).copy()
    S = f.sum()
    if S < 1e-10:
        return f, np.inf
    f = f / S
    best_score = compute_score_normalized(f)

    start = time.monotonic()
    prev_improvement = np.inf

    for sweep in range(n_sweeps):
        if time.monotonic() - start > time_budget:
            break

        sweep_start_score = best_score
        order = rng.permutation(n)

        for i in order:
            if time.monotonic() - start > time_budget:
                break

            old_val = f[i]
            # Generate candidate values
            if old_val > 1e-8:
                candidates = [0.0, old_val * 0.3, old_val * 0.6, old_val * 1.4,
                              old_val * 2.0, 1.0 / n]
            else:
                candidates = [1.0 / n, 2.0 / n, 5.0 / n]

            best_cand = old_val
            best_cand_score = best_score

            for val in candidates:
                f[i] = val
                S_new = f.sum()
                if S_new < 1e-10:
                    f[i] = old_val
                    continue
                score_c = compute_score_normalized(f / S_new)
                if score_c < best_cand_score - 1e-12:
                    best_cand_score = score_c
                    best_cand = val

            if best_cand != old_val:
                f[i] = best_cand
                f = f / f.sum()
                best_score = best_cand_score

        # Check convergence
        improvement = sweep_start_score - best_score
        if improvement < 1e-10:
            break
        if improvement < prev_improvement * 0.1 and improvement < 1e-7:
            break
        prev_improvement = improvement

    return f, best_score


def pair_exchange_search(f, time_budget=30, rng=None):
    """Try exchanging mass between pairs of elements."""
    if rng is None:
        rng = np.random.default_rng(0)

    n = len(f)
    f = f.astype(np.float64).copy()
    S = f.sum()
    f = f / S
    best_score = compute_score_normalized(f)

    start = time.monotonic()
    it = 0

    while time.monotonic() - start < time_budget:
        batch = 2000 if n <= 512 else 500
        for _ in range(batch):
            if time.monotonic() - start > time_budget:
                break
            i = int(rng.integers(0, n))
            j = int(rng.integers(0, n))
            if i == j:
                continue
            # Move some mass from i to j or vice versa
            frac = 0.1 + 0.4 * rng.random()
            f_try = f.copy()
            if rng.random() < 0.5:
                amt = f_try[i] * frac
                f_try[i] -= amt
                f_try[j] += amt
            else:
                amt = f_try[j] * frac
                f_try[j] -= amt
                f_try[i] += amt
            f_try = np.clip(f_try, 0, None)
            S_try = f_try.sum()
            if S_try < 1e-10:
                continue
            f_try = f_try / S_try
            score_try = compute_score_normalized(f_try)
            if score_try < best_score - 1e-12:
                best_score = score_try
                f = f_try
        it += 1

    return f, best_score


def propose_candidate(seed=42, budget_s=1500, **kwargs):
    """
    Main search function. Uses LP optimization with active sets, coordinate descent
    polishing, and multi-start strategy across different sequence lengths.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + budget_s - 20.0

    best_overall = None
    best_score_overall = np.inf

    # Get parent sequence
    parent = None
    try:
        parent = np.array(height_sequence_1, dtype=np.float64)
        if parent.ndim != 1 or len(parent) == 0 or not np.isfinite(parent).all():
            parent = None
        else:
            parent = np.clip(parent, 0, 1000)
    except Exception:
        parent = None

    if parent is not None:
        parent_score = compute_score(parent)
        print(f"Parent: n={len(parent)}, score={parent_score:.8f}")
        best_overall = parent.copy()
        best_score_overall = parent_score
    else:
        print("No parent available, starting fresh.")

    # Configuration list: (name, n, start_type, time_weight)
    configs = []

    if parent is not None:
        n_p = len(parent)
        configs.append((f"parent_{n_p}", n_p, 'parent', 0.28))
        if n_p == 256:
            configs.append((f"parent_pert_{n_p}", n_p, 'parent_pert', 0.12))
        # Upsampled parent
        if n_p <= 512:
            configs.append((f"upsampled_{n_p*2}", n_p * 2, 'upsample', 0.12))

    configs.append(("uniform_256", 256, 'uniform', 0.10))
    configs.append(("sine_512", 512, 'sine', 0.10))
    configs.append(("uniform_512", 512, 'uniform', 0.08))
    configs.append(("sine_1024", 1024, 'sine', 0.08))
    configs.append(("bimodal_512", 512, 'bimodal', 0.08))
    configs.append(("uniform_1024", 1024, 'uniform', 0.04))

    total_weight = sum(c[3] for c in configs)
    available_time = budget_s - 20.0
    print(f"Total configs: {len(configs)}, available time: {available_time:.0f}s")
    print(f"Target: 1.5027")

    for name, n, start_type, weight in configs:
        if time.monotonic() > deadline:
            break

        time_here = weight * available_time
        remaining = deadline - time.monotonic()
        time_here = min(time_here, max(20.0, remaining))

        if time_here < 20:
            break

        print(f"\n[{name}] n={n}, time={time_here:.0f}s")

        # Create initial sequence
        if start_type == 'parent':
            f0 = parent.copy()
        elif start_type == 'parent_pert':
            f0 = parent * (1 + 0.3 * rng.standard_normal(len(parent)))
            f0 = np.clip(f0, 0, None)
        elif start_type == 'upsample':
            f0 = np.zeros(n)
            p = parent if parent is not None else np.ones(n_p)
            ratio = n // len(p)
            for i in range(len(p)):
                f0[i * ratio:(i + 1) * ratio] = p[i]
            f0 = f0 + 0.01 * rng.standard_normal(n).clip(0, None)
        elif start_type == 'uniform':
            f0 = np.ones(n) * (1 + 0.15 * rng.standard_normal(n))
            f0 = np.clip(f0, 0.1, None)
        elif start_type == 'sine':
            j = np.arange(n, dtype=np.float64)
            f0 = 1.0 + 0.4 * np.sin(2 * np.pi * j / (n / 3.0))
            f0 = f0 + 0.1 * np.sin(2 * np.pi * j / (n / 7.0))
            f0 = np.clip(f0, 0.05, None)
            f0 = f0 * (1 + 0.05 * rng.standard_normal(n))
            f0 = np.clip(f0, 0.01, None)
        elif start_type == 'bimodal':
            j = np.arange(n, dtype=np.float64)
            center = n / 4
            width = n / 8
            f0 = np.exp(-((j - center) ** 2) / (2 * width ** 2))
            f0 += np.exp(-((j - 3 * center) ** 2) / (2 * width ** 2))
            f0 = np.clip(f0, 0.01, None)
        else:
            f0 = np.ones(n)

        f0 = np.clip(f0, 0, 1000).astype(np.float64)
        if f0.sum() < 0.01:
            continue
        f0 = f0 / f0.sum()

        # Time allocation: LP (55%), CD (25%), pair exchange (20%)
        lp_time = time_here * 0.55
        cd_time = time_here * 0.25
        pe_time = time_here * 0.20

        # Phase 1: LP optimization
        f_opt, score = optimize_lp(f0, lp_time, rng, verbose=(n <= 512))
        print(f"  After LP: {score:.8f}")

        # Phase 2: Coordinate descent
        if np.isfinite(score):
            f_cd, score_cd = coordinate_descent(f_opt, time_budget=cd_time, rng=rng)
            if score_cd < score:
                f_opt = f_cd
                score = score_cd
                print(f"  After CD: {score:.8f}")

        # Phase 3: Pair exchange
        if np.isfinite(score) and time.monotonic() < deadline - 5:
            f_pe, score_pe = pair_exchange_search(f_opt, time_budget=pe_time, rng=rng)
            if score_pe < score:
                f_opt = f_pe
                score = score_pe
                print(f"  After PE: {score:.8f}")

        if np.isfinite(score) and score < best_score_overall:
            best_score_overall = score
            best_overall = f_opt.copy()
            print(f"  ** New best: {score:.8f} (n={n}, config={name}) **")

    # Final intensive polishing phase
    remaining = deadline - time.monotonic()
    if best_overall is not None and remaining > 30:
        print(f"\n=== Final polishing phase ({remaining:.0f}s remaining) ===")
        n_best = len(best_overall)

        # Multiple CD + LP cycles
        cycle_time = min(remaining * 0.15, 60)
        for cycle in range(min(5, int(remaining // max(cycle_time, 10)))):
            if time.monotonic() > deadline - 10:
                break

            # CD pass
            t_cd = min(cycle_time * 0.4, deadline - time.monotonic() - 5)
            if t_cd > 5:
                f_cd, s_cd = coordinate_descent(best_overall, time_budget=t_cd, rng=rng)
                if s_cd < best_score_overall:
                    best_score_overall = s_cd
                    best_overall = f_cd
                    print(f"  Cycle {cycle} CD: {s_cd:.8f}")

            # LP pass
            t_lp = min(cycle_time * 0.5, deadline - time.monotonic() - 5)
            if t_lp > 10:
                f_lp, s_lp = optimize_lp(best_overall, t_lp, rng, verbose=False)
                if s_lp < best_score_overall:
                    best_score_overall = s_lp
                    best_overall = f_lp
                    print(f"  Cycle {cycle} LP: {s_lp:.8f}")

            # Pair exchange
            t_pe = min(cycle_time * 0.3, deadline - time.monotonic() - 5)
            if t_pe > 5:
                f_pe, s_pe = pair_exchange_search(best_overall, time_budget=t_pe, rng=rng)
                if s_pe < best_score_overall:
                    best_score_overall = s_pe
                    best_overall = f_pe
                    print(f"  Cycle {cycle} PE: {s_pe:.8f}")

        # Last CD sweep with all remaining time
        t_final = deadline - time.monotonic() - 3
        if t_final > 10:
            f_final, s_final = coordinate_descent(best_overall, time_budget=t_final, rng=rng)
            if s_final < best_score_overall:
                best_score_overall = s_final
                best_overall = f_final
                print(f"  Final CD: {s_final:.8f}")

    # Verify and return
    if best_overall is not None:
        best_overall = np.clip(best_overall, 0, 1000)
        # Ensure sum > 0.01
        if best_overall.sum() < 0.01:
            best_overall = best_overall * 100
        final_list = [float(x) for x in best_overall]
        final_score = evaluate_sequence(final_list)
        print(f"\nFinal verified score: {final_score:.8f}")
        print(f"Sequence length: {len(final_list)}")
        return final_list

    # Fallback
    if parent is not None:
        return [float(x) for x in parent]
    return [1.0] * 256
