```python
def solve_lp_min_M(other, n):
    """
    Solve: min M s.t. conv(x, other)[k] <= M for all k, x >= 0, sum(x) = 1.
    Returns (x_opt, M_opt).
    """
    # Variables: x[0], ..., x[n-1], M
    # Total: n + 1 variables
    # Objective: minimize M => c = [0, 0, ..., 0, 1]
    c = np.zeros(n + 1)
    c[-1] = 1.0  # minimize M
    
    # Constraints:
    # conv(x, other)[k] = sum_i x[i] * other[k-i] <= M for k = 0, ..., 2n-2
    # => sum_i x[i] * other[k-i] - M <= 0
    # For each k, the coefficient of x[i] is other[k-i] (0 if out of range)
    # The coefficient of M is -1.
    
    A_ub = np.zeros((2*n-1, n+1))
    b_ub = np.zeros(2*n-1)
    
    for k in range(2*n-1):
        for i in range(n):
            j = k - i
            if 0 <= j < n:
                A_ub[k, i] = other[j]
        A_ub[k, n] = -1.0  # coefficient of M
    
    # Equality constraint: sum(x) = 1
    A_eq = np.zeros((1, n+1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])
    
    # Bounds: x[i] >= 0, M >= 0
    bounds = [(0, None)] * n + [(0, None)]
    
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq, bounds=bounds, method='highs')
    
    if res.status == 0:
        return res.x[:n], res.x[n]
    return None, None
```

```python
    """Build matrix A where A[k,i] = other[k-i] if valid, else 0."""
    A = np.zeros((2*n-1, n))
    for k in range(2*n-1):
        # i ranges from max(0, k-n+1) to min(n-1, k)
        i_start = max(0, k - n + 1)
        i_end = min(n-1, k)
        for i in range(i_start, i_end + 1):
            A[k, i] = other[k - i]
```

```python
def build_conv_matrix(other, n):
    """Build matrix A where A[k,i] = other[k-i] if valid, else 0."""
    # k ranges from 0 to 2n-2, i ranges from 0 to n-1
    k = np.arange(2*n-1)[:, None]  # (2n-1, 1)
    i = np.arange(n)[None, :]  # (1, n)
    idx = k - i  # (2n-1, n)
    A = np.where((idx >= 0) & (idx < n), other[idx * ((idx >= 0) & (idx < n))], 0.0)

    valid = (idx >= 0) & (idx < n)
```

```python
def subgradient_step(a, tau=0.01, eta=0.1):
    """One step of smoothed subgradient method."""
    n = len(a)
    conv = np.convolve(a, a)
    # Softmax weights
    conv_scaled = conv / tau
    conv_scaled -= conv_scaled.max()  # for numerical stability
    weights = np.exp(conv_scaled)
    weights /= weights.sum()
    
    # Compute weighted gradient
    # d(conv[k])/da[m] = 2*a[k-m] if 0 <= k-m < n, else 0
    # grad[m] = sum_k weights[k] * 2*a[k-m] (with 0 for out-of-range)
    grad = np.zeros(n)
    for m in range(n):
        for k in range(2*n-1):
            j = k - m
            if 0 <= j < n:
                grad[m] += weights[k] * 2 * a[j]
    
    # Multiplicative update
    a = a * np.exp(-eta * grad)
    a = np.clip(a, 0, None)
    s = a.sum()
    if s < 1e-10:
        a = np.ones(n) / n
    else:
        a /= s
    return a
```

    sum_a = float(np.sum(a))

def build_conv_matrix(other, n):
    """Build matrix A where A[k,i] = other[k-i] if 0<=k-i<n else 0."""
    k = np.arange(2 * n - 1)[:, None]
    i = np.arange(n)[None, :]
    idx = k - i
    valid = (idx >= 0) & (idx < n)
    A = np.where(valid, other[np.clip(idx, 0, n - 1)], 0.0)
    return A

    """Solve: min M s.t. conv(x, other)[k] <= M for all k, x >= 0, sum(x) = 1."""
    A_conv = build_conv_matrix(other, n)
