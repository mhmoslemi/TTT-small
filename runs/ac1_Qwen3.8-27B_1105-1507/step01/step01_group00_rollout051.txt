"""
Multi-phase optimization for minimizing 2n*max(conv(f,f))/sum(f)^2.
Uses LP direction finding, soft-max gradient descent with Adam,
spectral initialization, and perturbation restarts.

import numpy as np
import time
from scipy.optimize import linprog
from scipy.sparse import csr_matrix


def build_conv_matrix_sparse(f):
    """Build sparse matrix A where (A@g)[k] = convolve(f,g)[k]."""
    n = len(f)
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        j_min = max(0, k - (n - 1))
        j_max = min(k, n - 1)
        for j in range(j_min, j_max + 1):
            rows.append(k)
            cols.append(j)
            vals.append(f[k - j])
    rows = np.array(rows, dtype=np.int32)
    cols = np.array(cols, dtype=np.int32)
    vals = np.array(vals, dtype=np.float64)
    return csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))


def solve_lp_direction(f, M, A_cache=None):
    """Solve: max sum(g) s.t. g >= 0, convolve(f,g)_k <= M for all k."""
    n = len(f)
    if A_cache is None:
        A = build_conv_matrix_sparse(f)
    else:
        A = A_cache
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    try:
        result = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
        if result.success:
            return result.x
    except Exception:
        pass
    return None

def compute_soft_max_gradient(f, beta):
    """
    Compute gradient of L_beta(f) = (1/beta) * log(sum_k exp(beta * C[k]))
    where C = conv(f,f). Returns (gradient, soft_max_value).

    n = len(f)
    C = np.convolve(f, f)
    max_C = C.max()
    exp_C = np.exp(beta * (C - max_C))
    total = exp_C.sum()
    if total < 1e-300:

    S = float(np.sum(f))

    best_score = evaluate_sequence(f.tolist())

        update = m_hat / (np.sqrt(v_hat) + eps)

            best_f = f.copy()

def tukey_window(n, r=0.3):
    """Generate a Tukey (tapered cosine) window."""
    w = np.ones(n)
    m = int(np.floor(r * n / 2))
    if m > 0:
        for i in range(m):
            w[i] = 0.5 * (1 + np.cos(np.pi * (2 * i / (2 * m) - 1)))
            w[n - 1 - i] = w[i]
    return w


def hann_window(n):
    """Generate a Hann window."""
    n = max(n, 2)
    w = 0.5 * (1 + np.cos(2 * np.pi * np.arange(n) / (n - 1)))
    return w


def psf_like_window(n, alpha=2.0):
    """Generate a super-Gaussian (PSF-like) window."""
    idx = np.arange(n, dtype=np.float64)
    center = (n - 1) / 2.0
    w = np.exp(-((idx - center) / (n / 2.0)) ** (2 * alpha))
    return w


def perturb(f, rng, strength=0.05):

    noise = 1.0 + strength * rng.standard_normal(len(f))
    new_f = f * np.clip(noise, 0.1, 2.0)
    new_f = np.clip(new_f, 0.0, 1000.0)
    S = float(np.sum(new_f))
    if S < 1e-10:
        return f.copy()
    return new_f / S

    if time.monotonic() < deadline - 2:
        f4, s4 = softmax_gradient_optimize(best_f, deadline - 2, beta_start=100.0,
                                            beta_end=500.0, lr_start=0.01,
                                            max_iters=2000, verbose_prefix=verbose_prefix + "P4")
        if s4 < best_score:
            best_f, best_score = f4, s4

    return best_f, best_score


def propose_candidate(budget_s=1500, **kwargs):
    t0 = time.monotonic()
    deadline = t0 + max(0.0, float(budget_s) - 10.0)

    best_sequence = None
    best_score = float('inf')
    rng = np.random.default_rng(123)

    # Try parent sequence
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64).copy()
        parent = np.clip(parent, 0.0, 1000.0)
        if parent.ndim == 1 and parent.size > 0 and np.all(np.isfinite(parent)):
            S = float(np.sum(parent))
            if S > 0.01:
                score = evaluate_sequence(parent.tolist())
                if np.isfinite(score):
                    best_score = score
                    best_sequence = parent.copy()
                    print(f"Parent n={parent.size}: score={score:.6f}")
    except (NameError, Exception) as e:
        print(f"Could not load parent: {e}")

    # Lengths to try - focus on promising range
    # Score = 2n * max_conv / sum^2. Longer n allows flatter autocorrelation.
    # But computation cost grows. Try: 256, 512, 768, 1024, 1536
    lengths_to_try = [512, 1024, 768, 2048, 256]

    # Time budget allocation (rough):
    # 512: 400s (best known length, refine heavily)
    # 1024: 350s (longer, potentially better)
    # 768: 200s
    # 2048: 200s
    # 256: 100s (quick check)
    # Remaining: perturbation restarts on best found

    time_budgets = {512: 0.30, 1024: 0.25, 768: 0.12, 2048: 0.13, 256: 0.06}
    total_budget = deadline - t0

    for n in lengths_to_try:
        if time.monotonic() > deadline - 30:
            print("Time running out, stopping length exploration")
            break

        n_budget = total_budget * time_budgets.get(n, 0.1)
        n_deadline = time.monotonic() + n_budget
        print(f"\n=== n={n} (budget {n_budget:.0f}s) ===")

        # Generate initializations
        inits = []

        # Tukey windows with different r
        for r in [0.2, 0.3, 0.4, 0.5]:
            inits.append((f'tukey_r{r}', tukey_window(n, r)))

        # Hann
        inits.append(('hann', hann_window(n)))

        # Super-Gaussian
        for alpha in [1.5, 2.0, 3.0]:
            inits.append((f'psf_a{alpha}', psf_like_window(n, alpha)))

        # Random smooth
        for seed in [1, 2]:
            r_rng = np.random.default_rng(seed + n)
            x = r_rng.standard_normal(n)
            # Smooth it
            kernel = np.exp(-0.5 * (np.arange(-10, 11) / 3.0) ** 2)
            smooth_x = np.convolve(x, kernel, mode='same')
            smooth_x = np.abs(smooth_x)
            smooth_x = smooth_x + 0.1
            inits.append((f'rand_smooth_{seed}', smooth_x))

        # If we have a parent of similar length, use it as init
        if best_sequence is not None:
            if len(best_sequence) == n:
                inits.insert(0, ('parent', best_sequence.copy()))
            else:
                # Interpolate parent to length n
                from numpy import interp
                parent_interp = interp(np.linspace(0, 1, n),
                                       np.linspace(0, 1, len(best_sequence)),
                                       best_sequence)
                parent_interp = np.clip(parent_interp, 0, None)
                inits.insert(0, ('parent_interp', parent_interp))

        # Optimize each initialization (with time sharing)
        n_inits = len(inits)
        per_init_time = n_budget / max(n_inits, 1) * 0.8  # leave 20% for restarts

        for label, f_init in inits:
            if time.monotonic() > n_deadline:
                break

            f_init = np.clip(f_init.astype(np.float64), 0.0, 1000.0)
            S = float(np.sum(f_init))
            if S < 0.01:
                continue

            init_score = evaluate_sequence(f_init.tolist())
            if not np.isfinite(init_score):
                continue

            if init_score < best_score:
                best_score = init_score
                best_sequence = f_init.copy()

            # Only do full optimization if this init has promise
            # (score not too far from best)
            if init_score < best_score * 1.05 or len(inits) <= 4:
                init_deadline = min(n_deadline, time.monotonic() + per_init_time)
                if time.monotonic() < init_deadline - 5:
                    try:
                        bf, bs = full_optimize(f_init, init_deadline,
                                               verbose_prefix=f"[n={n} {label}] ")
                        if bs < best_score:
                            best_score = bs
                            best_sequence = bf.copy()
                            print(f"  [n={n} {label}] improved to {bs:.6f}")
                    except Exception as e:
                        print(f"  [n={n} {label}] error: {e}")

    # Perturbation restarts on the best sequence found
    if best_sequence is not None and time.monotonic() < deadline - 20:
        print(f"\n=== Perturbation restarts (current best: {best_score:.6f}) ===")
        n_best = len(best_sequence)
        restart_budget = deadline - time.monotonic() - 10
        n_restarts = min(10, max(2, int(restart_budget / 30)))
        per_restart = restart_budget / n_restarts

        for r in range(n_restarts):
            if time.monotonic() > deadline - 10:
                break

            strength = 0.03 + 0.03 * r
            f_pert = perturb(best_sequence, rng, strength=strength)
            f_pert_deadline = time.monotonic() + per_restart * 0.7

            try:
                bf, bs = full_optimize(f_pert, f_pert_deadline,
                                       phase1_time_frac=0.4,
                                       verbose_prefix=f"[restart {r}] ")
                if bs < best_score:
                    best_score = bs
                    best_sequence = bf.copy()
                    print(f"  Restart {r} (str={strength:.3f}): {bs:.6f}")
                else:
                    print(f"  Restart {r} (str={strength:.3f}): no improvement ({bs:.6f})")
            except Exception as e:
                print(f"  Restart {r} error: {e}")

    # Final local refinement on the absolute best
    if best_sequence is not None and time.monotonic() < deadline - 5:
        print(f"\n=== Final refinement (n={len(best_sequence)}, score={best_score:.6f}) ===")
        try:
            bf, bs = lp_iterate(best_sequence, deadline - 5, step_size_init=0.03,
                                max_iters=30, verbose_prefix="[final] ")
            if bs < best_score:
                best_score = bs
                best_sequence = bf.copy()

            bf, bs = softmax_gradient_optimize(best_sequence, deadline - 3,
                                                beta_start=200.0, beta_end=1000.0,
                                                lr_start=0.005, max_iters=2000,
                                                verbose_prefix="[final_SM] ")
            if bs < best_score:
                best_score = bs
                best_sequence = bf.copy()
            print(f"  Final: {best_score:.6f}")
        except Exception as e:
            print(f"  Final refinement error: {e}")

    if best_sequence is None:
        # Fallback
        best_sequence = tukey_window(512, 0.3)
        best_score = evaluate_sequence(best_sequence.tolist())

    # Ensure all values are valid
    best_sequence = np.clip(best_sequence, 0.0, 1000.0)
    S = float(np.sum(best_sequence))
    if S < 0.01:
        best_sequence = np.ones(256)
        best_score = evaluate_sequence(best_sequence.tolist())

    elapsed = time.monotonic() - t0
    print(f"\nFINAL: n={len(best_sequence)}, score={best_score:.6f}, time={elapsed:.1f}s")
    return [float(x) for x in best_sequence]
