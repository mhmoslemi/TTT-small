    n = len(a)

    best_j = -1
    best_val = 0

def fast_conv(a):
    """Full linear self-convolution via FFT, O(n log n)."""
    n = len(a)

    while s < 2 * n - 1:

    n = len(a)

    for k in range(m):
        jl = max(0, k - n + 1)

        parent_len = best.size
        lengths = [parent_len] + [l for l in lengths if l != parent_len]

    lp_budget = total_budget * 0.45
    lp_deadline = time.monotonic() + lp_budget

    for n in lengths:
        if time.monotonic() > lp_deadline:
            break
        # Skip very long sequences if not enough time
        if n >= 1024 and (lp_deadline - time.monotonic()) < 120:
            continue
        if n >= 768 and (lp_deadline - time.monotonic()) < 60:
            continue

        t0 = time.monotonic()
        print(f"\n=== LP Phase: n={n} ===", flush=True)

        inits = generate_initializations(n, rng, best)

        # Limit number of initializations based on remaining time
        max_inits = min(len(inits), max(2, int((lp_deadline - time.monotonic()) / 30)))

        for i, a0 in enumerate(inits[:max_inits]):
            if time.monotonic() > lp_deadline:
                break
            a0 = np.clip(a0, 0.001, None)
            a0 = a0 / a0.sum()

            remaining = lp_deadline - time.monotonic()
            # Estimate: ~20-50ms per LP iteration for n<=512, ~100-200ms for n=1024
            est_time_per_iter = 0.03 if n <= 512 else 0.05 if n <= 768 else 0.12
            max_iters = max(100, int(remaining * 0.8 / est_time_per_iter))
            max_iters = min(max_iters, 5000)

            a_result, score = lp_iteration(a0, max_iters, lp_deadline)
            if score < best_score:
                best = a_result.copy()
                best_score = score
                print(f"  NEW BEST: J={score:.6f} (n={n}, init={i})", flush=True)

        print(f"  LP n={n} done in {time.monotonic()-t0:.1f}s, best={best_score:.6f}", flush=True)

    # === PHASE 2: Smooth refinement ===
    smooth_budget = total_budget * 0.25
    smooth_deadline = min(time.monotonic() + smooth_budget, deadline)

    if best is not None and time.monotonic() < smooth_deadline:
        t0 = time.monotonic()
        print(f"\n=== Smooth Refinement ===", flush=True)

        # Refine best at its current length
        a_result, score = smooth_refine(best, smooth_deadline)
        if score < best_score:
            best = a_result.copy()
            best_score = score
            print(f"  NEW BEST (smooth): J={score:.6f} (n={best.size})", flush=True)

        # Try at neighboring lengths
        for n2 in [max(128, best.size // 2), min(2048, best.size * 2)]:
            if time.monotonic() > smooth_deadline:
                break
            if n2 == best.size:
                continue
            t_old = np.linspace(0, 1, best.size)
            t_new = np.linspace(0, 1, n2)
            a2 = np.maximum(np.interp(t_new, t_old, best), 0)
            if a2.sum() > 0.01:
                a2 = a2 / a2.sum()
                a_result2, score2 = smooth_refine(a2, smooth_deadline)
                if score2 < best_score:
                    best = a_result2.copy()
                    best_score = score2
                    print(f"  NEW BEST (smooth n={n2}): J={score2:.6f}", flush=True)

        print(f"  Smooth done in {time.monotonic()-t0:.1f}s, best={best_score:.6f}", flush=True)

    # === PHASE 3: Perturbation + quick LP ===
    if best is not None and time.monotonic() < deadline:
        t0 = time.monotonic()
        print(f"\n=== Perturbation Search ===", flush=True)
        n = best.size
        remaining = deadline - time.monotonic()

        # Number of perturbation tries
        n_tries = int(remaining / 8)  # ~8s per try
        n_tries = max(5, min(100, n_tries))

        noise_scales = [0.05, 0.1, 0.02, 0.15, 0.03]

        for trial in range(n_tries):
            if time.monotonic() > deadline:
                    break
            scale = noise_scales[trial % len(noise_scales)]
            noise = rng.normal(0, scale, n)
            a_pert = best + noise
            a_pert = np.clip(a_pert, 0, None)
            if a_pert.sum() < 0.01:
                continue
            a_pert = a_pert / a_pert.sum()

            # Quick LP refinement
            remaining = deadline - time.monotonic()
            est_time_per_iter = 0.03 if n <= 512 else 0.06
            max_iters = max(50, int(remaining / (n_tries - trial) / est_time_per_iter))
            max_iters = min(max_iters, 2000)

            a_result, score = lp_iteration(a_pert, max_iters, deadline)
            if score < best_score:
                best = a_result.copy()
                best_score = score
                print(f"  NEW BEST (pert): J={score:.6f} (trial={trial})", flush=True)

        print(f"  Perturb done in {time.monotonic()-t0:.1f}s, best={best_score:.6f}", flush=True)

    # === PHASE 4: Final long LP polish ===
    if best is not None and time.monotonic() < deadline:
        t0 = time.monotonic()
        print(f"\n=== Final Polish ===", flush=True)
        a = best.copy()
        remaining = deadline - time.monotonic()
        n = a.size
        est_time_per_iter = 0.03 if n <= 512 else 0.05 if n <= 768 else 0.1
        max_iters = max(200, int(remaining * 0.9 / est_time_per_iter))
        max_iters = min(max_iters, 10000)

        a_result, score = lp_iteration(a, max_iters, deadline)
        if score < best_score:
            best = a_result.copy()
            best_score = score

        # One more smooth pass
        if time.monotonic() < deadline:
            a_result, score = smooth_refine(best, deadline)
            if score < best_score:
                best = a_result.copy()
                best_score = score

        print(f"  Polish done in {time.monotonic()-t0:.1f}s, best={best_score:.6f}", flush=True)

    # Final validation using the actual evaluator
    if best is None:
        best = np.ones(256) / 256
        best_score = compute_score(best)

    final_score = float(evaluate_sequence([float(x) for x in best]))
    if np.isfinite(final_score) and final_score < best_score:
        best_score = final_score

    print(f"\nFINAL: J={best_score:.6f}, n={best.size}", flush=True)
    return [float(x) for x in best]
