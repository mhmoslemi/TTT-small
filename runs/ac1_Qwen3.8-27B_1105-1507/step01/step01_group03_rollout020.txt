```python
def smooth_grad_descent(f, n, beta_start, beta_end, n_iter, lr, use_fista=True):
    """Smooth gradient descent with adaptive beta."""
    f = f.copy()
    beta = beta_start
    fista_v = f.copy() if use_fista else None
    best_f = f.copy()
    best_score = score_normalized(f, n)
    
    for it in range(n_iter):
        # Animate beta
        progress = it / max(1, n_iter - 1)
        beta = beta_start * (beta_end / beta_start) ** progress
        
        f_cur = fista_v if use_fista else f
        
        # Compute gradient
        c = np.convolve(f_cur, f_cur)
        # Softmax with numerical stability
        c_shifted = c - c.max()
        p = np.exp(beta * c_shifted)
        p /= p.sum()
        
        # Gradient
        p_rev = p[::-1]
        f_pad = np.zeros(2*n-1)
        f_pad[:n] = f_cur
        grad = 2 * np.convolve(p_rev, f_pad, mode='full')[n-1:2*n-1][::-1]
        
        # Gradient step
        f_new = f_cur - lr * grad
        
        if use_fista:
            # FISTA update
            t_k = (1 + np.sqrt(1 + 4 * fista_t**2)) / 2
            fista_v = f_new + ((fista_t - 1) / t_k) * (f_new - f)
            fista_t = t_k
            f = f_new
        else:
            f = f_new
        
        # Project onto simplex
        f = project_simplex(f)
        
        # Evaluate
        score = score_normalized(f, n)
        if score < best_score:
            best_score = score
            best_f = f.copy()
    
    return best_f, best_score
```

```python
    best_f = f.copy()
    best_score = score_normalized(f, n)
    
    for it in range(n_iter):
        c = np.convolve(f, f)
        c_shifted = c - c.max()
        p = np.exp(beta * c_shifted)
        p /= p.sum()
        
        p_rev = p[::-1]
        f_pad = np.zeros(2*n-1)
        f_pad[:n] = f
        grad = 2 * np.convolve(p_rev, f_pad, mode='full')[n-1:2*n-1][::-1]
        
        # Multiplicative update
        f = f * np.exp(-lr * grad)
        f = np.maximum(f, 0)
        f /= f.sum()
        
        score = score_normalized(f, n)
        if score < best_score:
            best_score = score
            best_f = f.copy()
    
    return best_f, best_score
```

"""
Hybrid spectral-gradient optimization for minimizing 2*n*max(conv(f,f))/(sum(f))^2.
Combines Gerchberg-Saxton spectral flattening, smooth gradient descent with
adaptive softmax, multiplicative weight updates, and LP-based direction finding.
"""

import time
import numpy as np
from scipy.optimize import linprog
from scipy.sparse import csr_matrix


def normalize_sequence(arr):
    """Normalize to a nonnegative probability vector."""
    x = np.asarray(arr, dtype=np.float64).ravel()
    x = np.where(np.isfinite(x), x, 0.0)
    x = np.maximum(x, 0.0)
    s = float(x.sum())
    if s <= 1e-300:
        return None
    return x / s


def score_norm(x, n):
    """Score assuming x >= 0 and sum(x) = 1."""
    c = np.convolve(x, x)
    m = float(c.max())
    return 2.0 * n * m if np.isfinite(m) else float('inf')


def project_simplex(v):
    """Euclidean projection onto {x >= 0, sum(x) = 1}."""
    n = v.size
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    rho = np.where(cond)[0]
    if len(rho) == 0:
        return np.ones(n) / n
    theta = (cssv[rho[-1]] - 1.0) / (rho[-1] + 1.0)
    return np.maximum(v - theta, 0.0)


def fft_convolve(a, b):
    """Linear convolution via FFT."""
    n = len(a) + len(b) - 1
    N = 1 << (n - 1).bit_length()
    A = np.fft.rfft(a, N)
    B = np.fft.rfft(b, N)
    return np.fft.irfft(A * B, N)[:n]


def compute_smooth_grad(f, p, n):
    """Gradient of (1/beta)*log(sum exp(beta * c_k)) w.r.t. f."""
    # grad_i = 2 * sum_k p[k] * f[k-i]
    p_rev = p[::-1]
    f_pad = np.zeros(2 * n - 1)
    f_pad[:n] = f
    conv = fft_convolve(p_rev, f_pad)
    # conv[m] = sum_k p_rev[k] * f_pad[m-k]
    # We showed: grad_i = 2 * conv[2n-2-i]
    # For i=0..n-1: indices 2n-2, 2n-3, ..., n-1
    return 2.0 * conv[n-1:2*n-1][::-1]


def smooth_grad_optimize(f, n, beta_min, beta_max, n_iter, lr_init, deadline):
    """Smooth gradient descent with adaptive beta and momentum."""
    f = f.copy().astype(np.float64)
    v = np.zeros_like(f)
    momentum = 0.85
    best_f = f.copy()
    best_score = score_norm(f, n)
    lr = lr_init
    no_improve = 0
    
    for it in range(n_iter):
        if time.monotonic() > deadline:
            break
        
        # Animate beta
        frac = (it + 1) / n_iter
        beta = beta_min * (beta_max / beta_min) ** frac
        
        # Compute gradient
        c = np.convolve(f, f)
        c_shift = c - c.max()
        p = np.exp(beta * c_shift)
        p_sum = p.sum()
        if p_sum < 1e-300:
            p = np.zeros_like(c)
            p[np.argmax(c)] = 1.0
        else:
            p /= p_sum
        
        grad = compute_smooth_grad(f, p, n)
        
        # Momentum step
        v = momentum * v + grad
        f_new = f - lr * v
        f_new = project_simplex(f_new)
        
        # Check improvement
        sc = score_norm(f_new, n)
        if sc < best_score:
            best_score = sc
            best_f = f_new.copy()
            no_improve = 0
            f = f_new
        else:
            no_improve += 1
            f = f_new  # Still update for momentum
            if no_improve > 50:
                lr *= 0.7
                v *= 0.5
                no_improve = 0
                if lr < 1e-10:
                    break
    
    return best_f, best_score


def multiplicative_optimize(f, n, beta, n_iter, lr, deadline):
    """Multiplicative weight / exponential gradient updates."""
    f = f.copy().astype(np.float64)
    best_f = f.copy()
    best_score = score_norm(f, n)
    
    for it in range(n_iter):
        if time.monotonic() > deadline:
            break
        
        c = np.convolve(f, f)
        c_shift = c - c.max()
        p = np.exp(beta * c_shift)
        p_sum = p.sum()
        if p_sum < 1e-300:
            break
        p /= p_sum
        
        grad = compute_smooth_grad(f, p, n)
        
        # Multiplicative update
        f = f * np.exp(-lr * grad)
        f = np.maximum(f, 1e-300)
        f /= f.sum()
        
        sc = score_norm(f, n)
        if sc < best_score:
            best_score = sc
            best_f = f.copy()
    
    return best_f, best_score


def gs_flatten(f, n, n_iter, target_power, deadline):
    """Gerchberg-Saxton style spectral flattening."""
    f = f.copy().astype(np.float64)
    best_f = f.copy()
    best_score = score_norm(f, n)
    
    N = 1 << (2 * n - 1).bit_length()  # FFT size
    
    for it in range(n_iter):
        if time.monotonic() > deadline:
            break
        
        # Go to frequency domain
        F = np.fft.rfft(f, N)
        mag = np.abs(F)
        
        # Set target magnitude (flat spectrum)
        # |F|^2 should be as flat as possible
        # target_power is the desired |F|^2
        target_mag = np.sqrt(target_power)
        
        # Update phase only, set magnitude to target
        # But we need to preserve F[0] (DC component) = sum(f) = 1
        new_mag = target_mag.copy()
        new_mag[0] = mag[0]  # Preserve DC
        
        # Update F
        phase = np.angle(F)
        F_new = new_mag * np.exp(1j * phase)
        
        # Back to time domain
        f_new = np.fft.irfft(F_new, N)[:n]
        
        # Project onto simplex
        f_new = project_simplex(f_new)
        f = f_new
        
        sc = score_norm(f, n)
        if sc < best_score:
            best_score = sc
            best_f = f.copy()
    
    return best_f, best_score
