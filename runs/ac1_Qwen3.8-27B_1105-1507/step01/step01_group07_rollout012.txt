def compute_obj(a):
    n = len(a)
    if n == 0:
        return np.inf
    conv = np.convolve(a, a)
    max_b = float(np.max(conv))
    sum_a = float(np.sum(a))
    if sum_a < 0.1:  # Larger threshold to avoid numerical issues
        return np.inf
    return 2.0 * n * max_b / (sum_a * sum_a)

def build_A(a):
    n = len(a)
    A = np.zeros((2*n-1, n), dtype=np.float64)
    for k in range(2*n-1):
        j_start = max(0, k - n + 1)
        j_end = min(n-1, k)
        if j_start <= j_end:
            A[k, j_start:j_end+1] = a[k-j_end:k-j_start+1][::-1]
    return A

def lp_step(a, M):
    """Solve LP: max sum(g) s.t. (a*g)[k] <= M, g >= 0."""
    n = len(a)
    A = build_A(a)
    c = -np.ones(n)
    b_ub = np.full(2*n-1, M)
    bounds = [(0, None)] * n
    res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
    if res.status == 0:
        return res.x
    return None

    n = len(a)
    a = a / np.sum(a)
    best_a = a.copy()
    best_obj = compute_obj(best_a)
    
    for it in range(max_iter):
        conv = np.convolve(a, a)
        M = np.max(conv)
        
        g = lp_step(a, M)
        if g is None:
            break
        g_sum = np.sum(g)

def gradient_refine(a, max_iter=10000, lr=0.01, verbose=False):
    """Projected subgradient method on the simplex."""
    n = len(a)
    a = a / np.sum(a)
    best_a = a.copy()
    best_obj = compute_obj(best_a)
    momentum = np.zeros(n)

        conv = np.convolve(a, a)
        max_val = np.max(conv)
        # Find all lags that achieve (nearly) the max
        active = np.where(conv >= max_val * 0.999)[0]
        
        # Compute subgradient (average over active lags)
        grad = np.zeros(n)
        for k in active:
            for i in range(n):
                val = 0.0
                if 0 <= k-i < n:
                    val += a[k-i]
                if 0 <= k-i < n and k-i != i:
                    val += a[i]
                grad[i] += val
        grad /= len(active)
        
        # Project gradient onto tangent space of simplex
        grad = grad - np.mean(grad)
        
        # Momentum
        momentum = mu * momentum + (1-mu) * grad
        a_new = a - lr * momentum
        a_new = np.maximum(a_new, 0)
        s = np.sum(a_new)
        if s < 1e-10:

        a_new = a_new / s
        
        obj = compute_obj(a_new)
        if obj < best_obj:
            best_a = a_new.copy()
            best_obj = obj
            a = a_new
        else:
            # Decay learning rate
            lr *= 0.999
    
    return best_a, best_obj

        conv = np.convolve(a, a)
        M = float(np.max(conv))
        
        g, g_sum = solve_lp(a, M)
        if g is None:
            break
        if g_sum <= 1.0 + 1e-9:
            break
        
        g = g / g_sum
        
        # Try step sizes from large to small
        improved = False
        for t in [1.0, 0.8, 0.6, 0.4, 0.2, 0.1]:
            a_new = (1.0 - t) * a + t * g
            a_new = np.maximum(a_new, 0.0)
            s = np.sum(a_new)
            if s < 1e-10:
                continue
            a_new = a_new / s
            obj = compute_obj(a_new)
            if obj < best_obj - 1e-12:
                best_a = a_new.copy()
                best_obj = obj
                a = a_new
                improved = True
                break
        
        if not improved:
            # Try smaller step
            a_new = (1.0 - 0.05) * a + 0.05 * g
            a_new = np.maximum(a_new, 0.0)
            s = np.sum(a_new)
            if s > 1e-10:
                a_new = a_new / s
                a = a_new
    
    return best_a, best_obj

def gradient_step(a, lr=0.001):
    """One projected subgradient step on the simplex."""
    n = len(a)
    conv = np.convolve(a, a)
    max_val = float(np.max(conv))
    active = np.where(conv >= max_val - 1e-10)[0]
    
    grad = np.zeros(n)
    for k in active:
        for i in range(n):
            ki = k - i
            if 0 <= ki < n:
                if ki == i:
                    grad[i] += 2.0 * a[i]
                else:
                    grad[i] += a[ki] + a[i]
    grad /= len(active)
    
    # Project to tangent space
    grad = grad - np.mean(grad)
    
    a_new = a - lr * grad
    a_new = np.maximum(a_new, 0.0)
    s = np.sum(a_new)
    if s < 1e-10:
        return a
    return a_new / s

def gradient_descent(a_init, max_iter=20000, time_budget=60):
    """Projected subgradient descent with adaptive lr."""
    n = len(a_init)
    a = a_init.astype(np.float64).copy()
    a = np.maximum(a, 0)
    s = np.sum(a)
    if s < 1e-10:
        a = np.random.dirichlet(np.ones(n))
    else:
        a = a / s
    
    best_a = a.copy()
    best_obj = compute_obj(best_a)
    lr = 1.0 / np.sqrt(n)
    start = time.monotonic()
    stall = 0
    
    for it in range(max_iter):
        if time.monotonic() - start > time_budget:
            break
        
        a_new = gradient_step(a, lr)
        obj = compute_obj(a_new)
        
        if obj < best_obj:
            best_a = a_new.copy()
            best_obj = obj
            a = a_new
            stall = 0
        else:
            a = a_new  # Still move, but don't update best
            stall += 1
            if stall > 500:
                lr *= 0.5
                stall = 0
                if lr < 1e-8:
                    break
    
    return best_a, best_obj

def random_refine(a_init, max_iter=50000, time_budget=30):
    """Random perturbation local search."""
    n = len(a_init)
    rng = np.random.default_rng(int(time.monotonic() * 1e6) % 2**31)
    a = a_init.astype(np.float64).copy()
    a = np.maximum(a, 0)
    s = np.sum(a)
    a = a / s
    
    best_a = a.copy()
    best_obj = compute_obj(best_a)
    start = time.monotonic()
    
    for it in range(max_iter):
        if time.monotonic() - start > time_budget:
            break
        
        # Perturb a random block
        block_size = max(1, n // 20)
        start_idx = int(rng.integers(0, n - block_size + 1))
        a_new = a.copy()
        a_new[start_idx:start_idx+block_size] *= np.exp(rng.normal(0, 0.1, block_size))
        a_new = np.maximum(a_new, 0)
        s = np.sum(a_new)
        if s < 1e-10:
            continue
        a_new = a_new / s
        obj = compute_obj(a_new)
        if obj < best_obj:
            best_a = a_new.copy()
            best_obj = obj
            a = a_new
    
    return best_a, best_obj
