Optimizes non-negative step function heights to minimize 2n*max(conv(f,f))/(sum(f))^2.
Uses LP-based Frank-Wolfe with active-set constraints, combined with mirror descent
initialization and multi-restart strategy.
"""

import numpy as np
import time
from scipy.optimize import linprog
from scipy.sparse import coo_matrix

def _fft_convolve_full(f):
    n = len(f)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    F = np.fft.rfft(f, size)
    h = np.fft.irfft(F * F, size)
    return h[:2 * n - 1]

def _score_raw(f):
    n = len(f)
    h = _fft_convolve_full(f)
    M = float(np.max(h))
    S = float(np.sum(f))
    if S < 0.01:
        return float('inf')
    return 2.0 * n * M / (S * S)

def _normalize(f):
    f = np.maximum(f, 0.0)
    s = float(np.sum(f))
    if s < 1e-30:
        return None
    return f / s

def _build_constraint_matrix(f, active_lags):
    """Build the constraint matrix C where C[k_idx, j] = f[k-j] for active lag k."""
    n = len(f)
    row_idx = []
    col_idx = []
    vals = []
    for idx, k in enumerate(active_lags):
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        for j in range(j_start, j_end + 1):
            row_idx.append(idx)
            col_idx.append(j)
            vals.append(f[k - j])
    C = coo_matrix((vals, (row_idx, col_idx)), shape=(len(active_lags), n))
    return C

def _mirror_descent(f_init, n_iter, eta_scale, threshold_frac):
    """Mirror descent with multiplicative updates."""
    f = np.asarray(f_init, dtype=np.float64).copy()
    f = _normalize(f)
    if f is None:
        return f_init.copy(), float('inf')
    n = len(f)
    best_f = f.copy()
    best_score = _score_raw(f)
    
    for it in range(n_iter):
        h = _fft_convolve_full(f)
        M = float(np.max(h))
        if M <= 0:
            break
        
        threshold = threshold_frac * M + 1e-15
        active = np.where(h >= threshold)[0]
        if len(active) == 0:
            active = np.array([int(np.argmax(h))])
        
        # Compute pressure vector
        p = np.zeros(n)
        for k in active:
            j_start = max(0, k - n + 1)
            j_end = min(n - 1, k)
            if j_end >= j_start:
                lo = k - j_end
                hi = k - j_start
                p[j_start:j_end + 1] += 2.0 * f[lo:hi + 1][::-1]
        p /= float(len(active))
        
        eta = eta_scale / M
        log_ratio = np.clip(-eta * p, -20.0, 20.0)
        f = f * np.exp(log_ratio)
        f_norm = _normalize(f)
        if f_norm is None:
            f = np.ones(n) / n
        else:
            f = f_norm
        
        score = _score_raw(f)
        if score < best_score:
            best_score = score
            best_f = f.copy()
    
    return best_f, best_score

def _cosine_init(n):
    j = np.arange(n, dtype=np.float64)
    f = np.cos(np.pi * (j + 0.5) / float(n)) ** 2
    f = np.maximum(f, 0.0)
    s = np.sum(f)
    if s > 0:
        f /= s
    return f

def _gaussian_init(n, sigma_frac):
    j = np.arange(n, dtype=np.float64) - (n - 1) / 2.0
    sigma = sigma_frac * float(n)
    f = np.exp(-j * j / (2.0 * sigma * sigma))

"""
Optimizes non-negative step function heights to minimize 2n*max(conv(f,f))/(sum(f))^2.
Uses LP-based Frank-Wolfe with adaptive active sets as the primary optimizer,
combined with mirror descent initialization, multi-start strategy, and
perturbation-based restarts to escape local minima.

import numpy as np
import time
from scipy.optimize import linprog
from scipy.sparse import coo_matrix, csr_matrix


def _fft_convolve_full(f):
    """Compute full linear convolution of f with itself using FFT."""
    n = len(f)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    F = np.fft.rfft(f, size)
    h = np.fft.irfft(F * F, size)
    return h[:2 * n - 1]


def _score_raw(f):
    """Fast score computation assuming f >= 0, sum(f) > 0.01."""
    n = len(f)
    h = _fft_convolve_full(f)
    M = float(np.max(h))
    S = float(np.sum(f))
    if S < 0.01:
        return float('inf')
    return 2.0 * n * M / (S * S)


def _normalize(f):
    """Normalize f to sum to 1, clip negatives. Returns None if degenerate."""
    f = np.maximum(f, 0.0)
    s = float(np.sum(f))
    if s < 1e-30:
        return None
    return f / s


def _pressure_vector(f, active_lags):
    """Compute average subgradient over active lags."""
    n = len(f)
    p = np.zeros(n, dtype=np.float64)
    for k in active_lags:
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        if j_end >= j_start:
            lo = k - j_end
            hi = k - j_start
            p[j_start:j_end + 1] += 2.0 * f[lo:hi + 1][::-1]
    p /= float(len(active_lags))
    return p


def _mirror_descent(f_init, n_iter, eta_scale, threshold_frac):

def _lp_oracle(f, M, active_lags, time_limit=3.0):
    """Solve the LP oracle for Frank-Wolfe direction.
    max sum(g) s.t. (f*g)_k <= M for k in active_lags, g >= 0
    """
    n = len(f)
    row_idx = []
    col_idx = []
    vals = []
    for idx, k in enumerate(active_lags):
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        for j in range(j_start, j_end + 1):
            row_idx.append(idx)
            col_idx.append(j)
            vals.append(f[k - j])
    if not row_idx:
        return None
    C = coo_matrix((vals, (row_idx, col_idx)), shape=(len(active_lags), n))
    C = C.tocsr()
    c_obj = -np.ones(n)
    b_ub = np.full(len(active_lags), M + 1e-12)
    try:
        res = linprog(c_obj, A_ub=C, b_ub=b_ub,
                      bounds=[(0, None)] * n, method='highs',
                      options={'time_limit': time_limit, 'presolve': True})
        if not res.success or res.x is None:
            return None
        return np.maximum(res.x, 0.0)
    except Exception:
        return None


def _frank_wolfe(f_init, max_iters, time_budget, active_frac=0.9, verbose=True):
    """LP-based Frank-Wolfe with adaptive active set and backtracking."""
    f = np.asarray(f_init, dtype=np.float64).copy()
    nf = _normalize(f)
    if nf is None:
        return f_init.copy(), float('inf')
    f = nf
    n = len(f)
    best_f = f.copy()
    best_score = _score_raw(f)
    start_time = time.monotonic()

    if verbose:
        print(f"    FW start: n={n}, score={best_score:.6f}")

    for it in range(max_iters):
        if time.monotonic() - start_time > time_budget:
            break

        h = _fft_convolve_full(f)
        M = float(np.max(h))

        # Adaptive active set: start small, grow
        if it < 5:
            frac = 0.7
        elif it < 20:
            frac = 0.8
        else:
            frac = active_frac
        threshold = frac * M + 1e-15
        active = np.where(h >= threshold)[0]
        if len(active) == 0:
            active = np.array([int(np.argmax(h))])
        # Limit active set size for speed
        if len(active) > 300:
            top_idx = np.argsort(h[active])[-300:]
            active = active[top_idx]

        # Solve LP
        t0 = time.monotonic()
        g = _lp_oracle(f, M, active, time_limit=3.0)
        lp_time = time.monotonic() - t0

        if g is None:
            # Fallback: use subgradient direction
            p = _pressure_vector(f, active)
            g = np.maximum(1.0 - 0.1 * p / (np.max(p) + 1e-30), 0.0)
            Sg = float(np.sum(g))
            if Sg < 1e-10:
                break
            g = g / Sg
        else:
            Sg = float(np.sum(g))
            if Sg < 1e-10:
                break
            g = g / Sg

        # Backtracking line search
        t_base = 2.0 / (it + 2)
        best_t = t_base
        best_f_t = (1.0 - t_base) * f + t_base * g
        best_f_t = _normalize(best_f_t)
        if best_f_t is None:
            break
        best_s_t = _score_raw(best_f_t)

        # Try a few other step sizes
        for t_try in [t_base * 0.5, t_base * 1.5, t_base * 2.0]:
            t_try = min(t_try, 1.0)
            f_try = (1.0 - t_try) * f + t_try * g
            f_try = _normalize(f_try)
            if f_try is None:
                continue
            s_try = _score_raw(f_try)
            if s_try < best_s_t:
                best_s_t = s_try
                best_t = t_try
                best_f_t = f_try

        f = best_f_t

        if best_s_t < best_score:
            best_score = best_s_t
            best_f = f.copy()

        if verbose and (it + 1) % 10 == 0:
            elapsed = time.monotonic() - start_time
            print(f"    FW it={it+1}: best={best_score:.6f}, M={M:.6e}, |K|={len(active)}, "
                  f"t={best_t:.3f}, lp={lp_time*1000:.0f}ms, elapsed={elapsed:.1f}s")

    return best_f, best_score


def _cosine_init(n):
    j = np.arange(n, dtype=np.float64)
    f = np.cos(np.pi * (j + 0.5) / float(n)) ** 2
    f = np.maximum(f, 0.0)
    s = np.sum(f)
    if s > 0:
        f /= s
    return f


def _gaussian_init(n, sigma_frac):
    j = np.arange(n, dtype=np.float64) - (n - 1) / 2.0
    sigma = sigma_frac * float(n)
    f = np.exp(-j * j / (2.0 * sigma * sigma))
    s = np.sum(f)
    if s > 0:
        f /= s
    return f


def _tapered_init(n, power):
    j = np.arange(n, dtype=np.float64)
    x = np.sin(np.pi * (j + 0.5) / float(n))
    f = x ** power
    f = np.maximum(f, 0.0)
    s = np.sum(f)
    if s > 0:
        f /= s
    return f


def _two_hump_init(n, sep_frac):
    """Two Gaussian bumps separated by a gap."""
    center1 = (1.0 - sep_frac) * (n - 1) / 2.0
    center2 = (1.0 + sep_frac) * (n - 1) / 2.0
    j = np.arange(n, dtype=np.float64)
    sigma = 0.15 * n
    f = np.exp(-(j - center1)**2 / (2*sigma*sigma)) + np.exp(-(j - center2)**2 / (2*sigma*sigma))
    s = np.sum(f)
    if s > 0:
        f /= s
    return f


def _uniform_taper_init(n, taper_frac):
    """Uniform in the middle, taper at edges."""
    j = np.arange(n, dtype=np.float64)
    f = np.ones(n)
    m = int(n * taper_frac)
    if m > 0:
        ramp = np.linspace(0, 1, m)
        f[:m] = ramp
        f[n-m:] = ramp[::-1]
    s = np.sum(f)
    if s > 0:
        f /= s
    return f


def propose_candidate(seed=42, budget_s=1500):
    """
    Search for non-negative step function heights minimizing the evaluation score.
    Strategy: multi-start with mirror descent initialization, then LP Frank-Wolfe
    refinement with adaptive active sets, plus perturbation restarts.
    """
    deadline = time.monotonic() + float(budget_s) - 10.0
    rng = np.random.default_rng(seed)

    # Get parent sequence
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64).copy()
        parent = np.clip(parent, 0.0, 1000.0)
    except Exception:
        parent = np.ones(512, dtype=np.float64)

    best_f = parent.copy()
    best_score = float(evaluate_sequence(best_f.tolist()))
    print(f"Parent: n={len(best_f)}, score={best_score:.6f}")

    # Build diverse candidate initializations
    candidates = []
    p = _normalize(parent)
    if p is not None:
        candidates.append(("parent", p))

    n_parent = len(parent)
    for n in [512, 768, 1024, 1536]:
        candidates.append(("cos_%d" % n, _cosine_init(n)))
        for sf in [0.15, 0.25, 0.35, 0.5]:
            candidates.append(("gauss_%d_%d" % (n, int(sf * 100)), _gaussian_init(n, sf)))
        for pw in [0.5, 1.0, 2.0, 4.0]:
            candidates.append(("tap_%d_%d" % (n, int(pw * 2)), _tapered_init(n, pw)))
        for tf in [0.1, 0.2, 0.3]:
            candidates.append(("utaper_%d_%d" % (n, int(tf * 100)), _uniform_taper_init(n, tf)))

    # Two-hump initializations
    for n in [512, 1024]:
        for sep in [0.3, 0.5, 0.7]:
            candidates.append(("2hump_%d_%d" % (n, int(sep * 10)), _two_hump_init(n, sep)))

    # Random initializations
    for n in [512, 1024]:
        f_r = rng.random(n)
        f_rn = _normalize(f_r)
        if f_rn is not None:
            candidates.append(("rand_%d" % n, f_rn))

    print(f"Total candidates: {len(candidates)}")

    # Phase 1: Quick mirror descent on all candidates to get good starting points
    print("\n=== Phase 1: Mirror Descent Initialization ===")
    results = []
    n_cand = len(candidates)
    remaining = deadline - time.monotonic()
    time_per = max(3.0, min(15.0, remaining / max(1, n_cand) * 0.4))

    for idx, (name, f0) in enumerate(candidates):
        if time.monotonic() > deadline - 90:
            break

        n_it = int(max(100, min(1500, time_per * 250)))
        t0 = time.monotonic()
        f_md, s_md = _mirror_descent(f0, n_it, 0.4, 0.8)
        elapsed = time.monotonic() - t0
        results.append((s_md, name, f_md))
        if idx % 5 == 0 or s_md < best_score:
            print(f"  [{idx+1}/{n_cand}] {name}: score={s_md:.6f} ({elapsed:.1f}s)")

        if s_md < best_score:
            best_score = s_md
            best_f = f_md.copy()
            print(f"  ** New best: {best_score:.6f} from {name} **")

    # Sort by score
    results.sort(key=lambda x: x[0])
    print(f"\nTop 5 after Phase 1:")
    for i in range(min(5, len(results))):
        print(f"  {results[i][1]}: {results[i][0]:.6f}")

    # Phase 2: LP Frank-Wolfe refinement on top candidates
    print("\n=== Phase 2: LP Frank-Wolfe Refinement ===")
    top_k = min(6, len(results))
    used_names = set()

    for r in range(top_k):
        if time.monotonic() > deadline - 40:
            break
        s_cur, name, f_cur = results[r]
        n = len(f_cur)
        remaining = deadline - time.monotonic() - 20
        if remaining < 30:
            break

        print(f"\n  Refining {name} (n={n}, score={s_cur:.6f}), budget={remaining:.0f}s")

        # First round of FW
        max_iters = int(min(300, remaining / 1.5))
        fw_budget = remaining * 0.5
        f_fw, s_fw = _frank_wolfe(f_cur, max_iters, fw_budget, active_frac=0.85)

        if s_fw < best_score:
            best_score = s_fw
            best_f = f_fw.copy()
            print(f"  ** New best: {best_score:.6f} from FW on {name} **")

        # Second round: MD then FW again
        if time.monotonic() < deadline - 40:
            remaining2 = deadline - time.monotonic() - 20
            if remaining2 > 30:
                n_it2 = int(min(2000, remaining2 * 100))
                f_md2, s_md2 = _mirror_descent(f_fw, n_it2, 0.2, 0.9)
                s_cur2 = min(s_fw, s_md2)
                f_cur2 = f_fw if s_fw <= s_md2 else f_md2

                if s_cur2 < best_score:
                    best_score = s_cur2
                    best_f = f_cur2.copy()
                    print(f"  ** New best: {best_score:.6f} from MD2 on {name} **")

                # Another FW round
                if time.monotonic() < deadline - 25:
                    remaining3 = deadline - time.monotonic() - 15
                    if remaining3 > 20:
                        max_iters2 = int(min(200, remaining3 / 2.0))
                        f_fw2, s_fw2 = _frank_wolfe(f_cur2, max_iters2, remaining3 * 0.8,
                                                    active_frac=0.9)
                        if s_fw2 < best_score:
                            best_score = s_fw2
                            best_f = f_fw2.copy()
                            print(f"  ** New best: {best_score:.6f} from FW2 on {name} **")
                        f_cur2 = f_fw2
                        s_cur2 = s_fw2

    # Phase 3: Also run FW directly on the parent (might be in a different basin)
    print("\n=== Phase 3: Direct FW on Parent ===")
    if time.monotonic() < deadline - 30:
        remaining = deadline - time.monotonic() - 15
        if remaining > 20:
            p_norm = _normalize(parent)
            if p_norm is not None:
                max_iters = int(min(200, remaining / 2.0))
                f_pfw, s_pfw = _frank_wolfe(p_norm, max_iters, remaining * 0.8, active_frac=0.8)
                if s_pfw < best_score:
                    best_score = s_pfw
                    best_f = f_pfw.copy()
                    print(f"  ** New best: {best_score:.6f} from FW on parent **")

    # Phase 4: Perturbation restarts to escape local minima
    print("\n=== Phase 4: Perturbation Restarts ===")
    if time.monotonic() < deadline - 30:
        perturbations = [0.01, 0.03, 0.05, 0.1, 0.2]
        for pf in perturbations:
            if time.monotonic() > deadline - 30:
                break
            remaining = deadline - time.monotonic() - 20
            if remaining < 20:
                break

            # Create perturbed version
            f_pert = np.asarray(best_f, dtype=np.float64).copy()
            noise = rng.random(len(f_pert)) * pf
            f_pert = _normalize(f_pert + noise)
            if f_pert is None:
                continue

            # Quick MD then FW
            n_it = int(min(1000, remaining * 100))
            f_pmd, s_pmd = _mirror_descent(f_pert, n_it, 0.3, 0.85)

            if time.monotonic() < deadline - 20:
                remaining2 = deadline - time.monotonic() - 15
                if remaining2 > 15:
                    max_it = int(min(100, remaining2 / 2.5))
                    f_pfw, s_pfw = _frank_wolfe(f_pmd, max_it, remaining2 * 0.7, active_frac=0.85)
                    s_best_pert = min(s_pmd, s_pfw)
                    f_best_pert = f_pmd if s_pmd <= s_pfw else f_pfw
                else:
                    s_best_pert = s_pmd
                    f_best_pert = f_pmd
            else:
                s_best_pert = s_pmd
                f_best_pert = f_pmd

            print(f"  Perturb({pf}): score={s_best_pert:.6f}")
            if s_best_pert < best_score:
                best_score = s_best_pert
                best_f = f_best_pert.copy()
                print(f"  ** New best: {best_score:.6f} from perturb({pf}) **")

    # Phase 5: Final polish with small-step FW
    print("\n=== Phase 5: Final Polish ===")
    if time.monotonic() < deadline - 10:
        remaining = deadline - time.monotonic() - 5
        if remaining > 10:
            max_it = int(min(100, remaining / 2.0))
            f_final, s_final = _frank_wolfe(best_f, max_it, remaining * 0.9,
                                            active_frac=0.95)
            if s_final < best_score:
                best_score = s_final
                best_f = f_final.copy()
                print(f"  ** Final polish improved to: {best_score:.6f} **")

    # Final verification
    final_list = [float(max(0.0, min(1000.0, x))) for x in best_f]
    final_score = float(evaluate_sequence(final_list))
    print(f"\nFinal verified score: {final_score:.6f} (n={len(final_list)})")

    if not np.isfinite(final_score) or final_score > 2.0:
        print("  Falling back to parent.")
        return [float(x) for x in parent]

    if final_score > best_score + 0.001:
        print(f"  Warning: verified score {final_score:.6f} > optimized {best_score:.6f}")

    return final_list
