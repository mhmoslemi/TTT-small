```python
def project_simplex(v):
    """Project v onto the simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    rho = np.where(u * np.arange(1, n+1) > cssv)[0][-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0)
    return w
```

```python
import numpy as np
from scipy.optimize import linprog, minimize
import time

def compute_obj(a):
    n = len(a)
    if n == 0:
        return np.inf
    conv = np.convolve(a, a)
    max_b = float(np.max(conv))
    sum_a = float(np.sum(a))
    if sum_a < 0.01:
        return np.inf
    return 2.0 * n * max_b / (sum_a * sum_a)

def project_simplex(v):
    """Project onto {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    idx = np.arange(1, n+1)
    rho = np.where(u * idx > cssv)[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho = rho[-1]
    theta = cssv[rho] / (rho + 1)
    return np.maximum(v - theta, 0)

def build_conv_matrix(A, n):
    """Build matrix such that (M * a) gives conv(A, a)."""
    # M[k, j] = A[k-j] if 0 <= k-j < len(A), else 0
    # But for our case, A is the fixed sequence
    m = len(A)
    M = np.zeros((2*m-1, n))
    for k in range(2*m-1):
        j_start = max(0, k - n + 1)
        j_end = min(n-1, k)
        if j_start <= j_end:
            M[k, j_start:j_end+1] = A[k-j_end:k-j_start+1][::-1]
    return M

def solve_lp(a, M_val, ub=1000.0):
    """Solve: max sum(g) s.t. conv(a,g) <= M_val, 0 <= g <= ub."""
    n = len(a)
    A_mat = build_conv_matrix(a, n)
    c = -np.ones(n)
    b_ub = np.full(2*n-1, M_val)
    bounds = [(0.0, ub)] * n
    res = linprog(c, A_ub=A_mat, b_ub=b_ub, bounds=bounds, method='highs')
    if res.status == 0:
        return res.x
    return None

def smooth_max_obj(a, beta):
    """Smooth approximation of 2*n*max(conv(a,a)) with sum(a)=1."""
    n = len(a)
    conv = np.convolve(a, a)
    # Shift for numerical stability
    conv_shifted = conv - np.max(conv)
    log_sum_exp = np.max(conv) + np.log(np.sum(np.exp(beta * conv_shifted))) / beta
    return 2.0 * n * log_sum_exp

def smooth_max_grad(a, beta):
    """Gradient of smooth max objective."""
    n = len(a)
    conv = np.convolve(a, a)
    conv_shifted = conv - np.max(conv)
    weights = np.exp(beta * conv_shifted)
    weights /= np.sum(weights)
    # Gradient of 2n * sum_k weights_k * conv_k
    # d(conv_k)/d(a_j) = 2*a[k-j] if 0 <= k-j < n
    grad = np.zeros(n)
    for k in range(2*n-1):
        w = weights[k]
        if w < 1e-15:
            continue
        j_start = max(0, k - n + 1)
        j_end = min(n-1, k)
        for j in range(j_start, j_end+1):
            grad[j] += 2.0 * w * a[k-j]
    grad *= 2.0 * n
    return grad

def lmbfgs_optimize(a_init, beta=100.0, maxiter=5000):
    """Optimize using L-BFGS-B with smooth max."""
    n = len(a_init)
    a = a_init.copy()
    
    def obj(a_flat):
        return smooth_max_obj(a_flat, beta)
    
    def grad(a_flat):
        return smooth_max_grad(a_flat, beta)
    
    res = minimize(obj, a, method='L-BFGS-B', jac=grad,
                   options={'maxiter': maxiter, 'ftol': 1e-12, 'gtol': 1e-10})
    # Project to simplex
    a_opt = project_simplex(res.x)
    return a_opt
```
