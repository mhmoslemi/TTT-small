Hybrid LP + subgradient optimization for minimizing 2*n*max(conv(a,a))/(sum(a))^2.
Uses alternating LP, subgradient descent on the max, L2 smoothing, and diverse
initializations across multiple sequence lengths.
"""

import numpy as np
from scipy.optimize import linprog
import time

# Global best tracking
_global_best_score = np.inf
_global_best_seq = None

def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    sum_a = float(np.sum(a))
    if sum_a < 0.01:
        return np.inf
    conv = np.convolve(a, a)
    max_b = float(np.max(conv))
    return 2.0 * n * max_b / (sum_a * sum_a)

def build_A(a):
    """Build constraint matrix A for LP. A[k,j] = a[k-j] if 0<=k-j<n else 0.
    conv(a, b)[k] = sum_j A[k,j] * b[j] = sum_j a[k-j] * b[j] (valid j only)
    """
    n = len(a)
    A = np.zeros((2*n-1, n), dtype=np.float64)
    for k in range(2*n-1):
        j_lo = max(0, k - n + 1)
        j_hi = min(n - 1, k)
        # A[k, j] = a[k-j] for j in [j_lo, j_hi]
        js = np.arange(j_lo, j_hi + 1)
        A[k, js] = a[k - js]
    return A

def solve_lp_for_b(a, T):
    """Solve LP: max sum(b) s.t. conv(a,b)[k] <= T for all k, b >= 0.
    Returns (b, max_sum) or (None, None) if infeasible.
    """
    n = len(a)
    A = build_A(a)
    c = -np.ones(n)  # minimize -sum(b) = maximize sum(b)
    b_ub = np.full(2*n-1, T)
    bounds = [(0.0, None)] * n
    res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
    if res.status == 0:
        return res.x, float(-res.fun)
    return None, None

def lp_alternate(a, max_iter=200, t_init=0.5, t_min=0.05, verbose=False):

    a = a / s  # normalize to sum=1
    
    best_a = a.copy()
    best_obj = compute_obj(best_a)
    t = t_init
    stall = 0
    
    for it in range(max_iter):
        conv_aa = np.convolve(a, a)
        T = float(np.max(conv_aa))
        
        b, b_sum = solve_lp_for_b(a, T)

        sg = np.zeros(n)

            s2 = 0.0
            for k in range(2*n-1):

def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    sum_a = float(np.sum(a))
    if sum_a < 0.01:
        return np.inf
    conv = np.convolve(a, a)
    max_b = float(np.max(conv))
    return 2.0 * n * max_b / (sum_a * sum_a)


def build_A(a):
    """Build constraint matrix for LP: conv(a,b)[k] = sum_j A[k,j]*b[j]."""
    n = len(a)
    A = np.zeros((2 * n - 1, n), dtype=np.float64)
    for k in range(2 * n - 1):
        j_lo = max(0, k - n + 1)
        j_hi = min(n - 1, k)
        js = np.arange(j_lo, j_hi + 1)
        A[k, js] = a[k - js]
    return A
