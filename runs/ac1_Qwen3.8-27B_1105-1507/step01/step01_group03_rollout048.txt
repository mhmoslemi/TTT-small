- Renormalize.

This maintains the sum and non-negativity (approximately).

Let me estimate: if each evaluation takes ~0.5ms (for n=1024, the convolution is ~2M multiplications), then in 300 seconds we can do ~600K evaluations. That's a lot of local search steps.

But wait, the convolution for n=1024 is O(n^2) = ~1M operations. In numpy, this should be very fast (~0.1ms). So we can do millions of evaluations.

"""
Improved LP-based iterative search with polishing phase for minimizing
2*n*max(convolve(f,f))/(sum(f))^2.

Key improvements over parent:
1. Focused time allocation on n=1024.
2. Peak-aware restart perturbations.
3. Post-LP local search polishing phase.
4. Better line search with parabolic interpolation.
5. Multiple LP restarts with different perturbation strategies.

import time
import numpy as np
from scipy.optimize import linprog
from scipy.sparse import csr_matrix


def normalize_sequence(arr):
    """Return a nonnegative floating vector normalized to sum 1, or None."""
    try:
        x = np.asarray(arr, dtype=np.float64).ravel()
    except Exception:
        return None
    if x.size == 0:
        return None
    x = np.where(np.isfinite(x), x, 0.0)
    x = np.maximum(x, 0.0)
    s = float(x.sum())
    if s <= 1e-300:
        return None
    x /= s
    return x


def score_normalized(x, n):
    """Score assuming x is nonnegative and sum(x) = 1."""
    conv = np.convolve(x, x)
    m = float(np.max(conv))
    if not np.isfinite(m):
        return float("inf")
    return float(2.0 * n * m)


def make_conv_template(n):
    """Precompute the CSR matrix structure for convolve(f, g)."""
    m = 2 * n - 1
    k64 = np.arange(m, dtype=np.int64)
    js = np.maximum(0, k64 - n + 1).astype(np.int32)
    je = np.minimum(n - 1, k64).astype(np.int32)
    counts = (je.astype(np.int64) - js.astype(np.int64) + 1).astype(np.int64)
    total = int(counts.sum())

    cum = np.cumsum(counts)
    starts = np.repeat(cum - counts, counts)
    local = np.arange(total, dtype=np.int64) - starts

    row = np.repeat(k64, counts).astype(np.int32)
    col_base = js[row.astype(np.int64)]
    col = (col_base + local).astype(np.int32)
    offsets = (row.astype(np.int64) - col.astype(np.int64)).astype(np.int32)

    A_off = csr_matrix((offsets, (row, col)), shape=(m, n), dtype=np.int32)
    edge_offsets = A_off.data.astype(np.int32, copy=True)
    A_template = A_off.astype(np.float64).copy()
    A_template.data.fill(0.0)

    c_obj = -np.ones(n, dtype=np.float64)
    bounds = [(0.0, None)] * n
    return A_template, edge_offsets, c_obj, bounds, m


def solve_lp(template, f, n, M):
    """Solve the LP direction problem."""
    A_template, edge_offsets, c_obj, bounds, m = template
    lp_scale = float(n)

    A = A_template.copy()
    A.data[:] = f[edge_offsets]
    A.data *= lp_scale

    b = np.full(m, M * lp_scale * lp_scale, dtype=np.float64)
    return linprog(c_obj, A_ub=A, b_ub=b, bounds=bounds, method="highs")


def peak_aware_perturb(f, rng, strength=0.1):
    """Perturb f in a way that targets the autocorrelation peak."""
    n = f.size
    h = np.convolve(f, f)
    k_peak = int(np.argmax(h))
    k_min = int(np.argmin(h))
    
    # Create a perturbation that reduces alignment at k_peak
    # and increases it at k_min
    perturb = np.zeros(n)
    # Shifted f by k_peak (this is the "bad" direction)
    if k_peak < n:
        perturb[:n - k_peak] -= f[k_peak:]
        perturb[k_peak:] -= f[:n - k_peak]
    # Shifted f by k_min (this is the "good" direction)  
    if k_min < n:
        perturb[:n - k_min] += f[k_min:]
        perturb[k_min:] += f[:n - k_min]
    
    # Normalize the perturbation
    p_norm = np.linalg.norm(perturb)
    if p_norm < 1e-10:
        perturb = rng.standard_normal(n)
        p_norm = np.linalg.norm(perturb)
    perturb /= p_norm
    
    # Apply perturbation and project to simplex
    f_new = f + strength * perturb
    f_new = np.maximum(f_new, 0.0)
    s = f_new.sum()
    if s < 1e-10:
        return f.copy()
    f_new /= s
    return f_new


def local_search_polish(f, n, iterations=5000, rng=None, time_limit=None):
    """Local search with small perturbations to polish the solution."""
    if rng is None:
        rng = np.random.default_rng()
    
    best_f = f.copy()
    best_score = score_normalized(f, n)
    start_time = time.monotonic()
    
    # Adaptive step size
    alpha = 0.02
    best_alpha = alpha
    
    for it in range(iterations):
        if time_limit is not None and time.monotonic() - start_time > time_limit:
            break
        
        # Random perturbation: pick 2-5 pairs and transfer weight
        n_pairs = rng.integers(2, 6)
        f_new = best_f.copy()
        
        for _ in range(n_pairs):
            i = rng.integers(n)
            j = rng.integers(n)
            if i == j:
                continue
            # Transfer from larger to smaller (or random direction)
            if rng.random() < 0.7:
                # Move from i to j (if f[i] > f[j])
                if f_new[i] > f_new[j]:
                    delta = alpha * f_new[i] * rng.random()
                    delta = min(delta, f_new[i] * 0.5)
                    f_new[i] -= delta
                    f_new[j] += delta
                else:
                    delta = alpha * f_new[j] * rng.random()
                    delta = min(delta, f_new[j] * 0.5)
                    f_new[j] -= delta
                    f_new[i] += delta
            else:
                # Random transfer
                delta = alpha * max(f_new[i], f_new[j]) * rng.random() * 0.5
                if f_new[i] >= delta:
                    f_new[i] -= delta
                    f_new[j] += delta
                elif f_new[j] >= delta:
                    f_new[j] -= delta
                    f_new[i] += delta
        
        s = f_new.sum()
        if s < 1e-10:
            continue
        f_new /= s
        
        sc = score_normalized(f_new, n)
        if sc < best_score:
            best_score = sc
            best_f = f_new.copy()
            best_alpha = max(alpha * 1.1, 0.001)
        else:
            # Slowly decrease alpha
            alpha = max(alpha * 0.995, 0.001)
    
    return best_f, best_score


def line_search_parabolic(f, g, f_score, n, max_iters=20):
    """Parabolic line search for t in (0,1) minimizing score((1-t)f + t*g)."""
    best_t = 0.0
    best_sc = f_score
    
    # Initial points
    ts = [0.01, 0.05, 0.1, 0.3, 0.5]
    scores = []
    
    for t in ts:
        f_new = (1 - t) * f + t * g
        f_new = np.maximum(f_new, 0.0)
        s = f_new.sum()
        if s < 1e-10:
            continue
        f_new /= s
        sc = score_normalized(f_new, n)
        scores.append((t, sc))
        if sc < best_sc:
            best_sc = sc
            best_t = t
    
    if len(scores) < 3:
        return best_t, best_sc
    
    # Parabolic interpolation on the best 3 points
    scores.sort(key=lambda x: x[1])
    (t1, s1), (t2, s2), (t3, s3) = scores[0], scores[1], scores[2]
    
    # Fit parabola: s(t) = a*t^2 + b*t + c
    # Using the 3 points
    denom = (t1 - t2) * (t1 - t3) * (t2 - t3)
    if abs(denom) < 1e-20:
        return best_t, best_sc
    
    a = (s1 * (t2 - t3) + s2 * (t3 - t1) + s3 * (t1 - t2)) / denom
    if a <= 0:
        return best_t, best_sc
    
    t_par = -1.0 / (2.0 * a)
    
    if 0 < t_par < 1:
        f_new = (1 - t_par) * f + t_par * g
        f_new = np.maximum(f_new, 0.0)
        s = f_new.sum()
        if s > 1e-10:
            f_new /= s
            sc = score_normalized(f_new, n)
            if sc < best_sc:
                best_sc = sc
                best_t = t_par
    
    return best_t, best_sc
