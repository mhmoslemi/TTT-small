"""
Optimized subgradient + L-BFGS-B + LP alternating approach for
minimizing 2*n*max(conv(a,a))/(sum(a))^2.
Uses annealed soft-max smoothing, exact active-set subgradients,
and multiple initializations to escape local optima.
"""

import numpy as np
import time
from scipy.optimize import minimize, linprog


def compute_obj(a):
    n = len(a)
    if n == 0:
        return np.inf
    conv = np.convolve(a, a)
    max_b = float(np.max(conv))
    sum_a = float(np.sum(a))
    if sum_a < 0.01:
        return np.inf
    return 2.0 * n * max_b / (sum_a * sum_a)


def compute_conv(a):
    return np.convolve(a, a)


def project_simplex(v):
    """Project v onto the standard simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if np.sum(v) < 1e-15:
        return np.ones(n) / n
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(1, n + 1)
    cond = u * ind > cssv
    if not np.any(cond):
        return np.ones(n) / n
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    return np.maximum(v - theta, 0)


def subgradient_step(a, eta):
    """One subgradient step: compute subgradient of obj, update a, project to simplex."""
    n = len(a)
    conv = np.convolve(a, a)
    M = np.max(conv)
    active = (conv >= M - 1e-12)
    w = active.astype(float)
    w /= w.sum()

    # Efficient computation: v = correlate-like
    v = np.zeros(n)
    for i in range(n):
        v[i] = np.dot(a, w[i:i+n])
    
    S = np.sum(a)
    # grad[i] = 4n*v[i]/S^2 - 4nC/S^3, where C = M
    grad = (4.0 * n / (S * S)) * v - (4.0 * n * M / (S ** 3))
    
    a_new = a - eta * grad
    a_new = project_simplex(a_new)
    return a_new, M


def subgradient_descent(a, max_iter=5000, eta0=1.0, decay=0.999, 
                        patience=500, verbose=False):
    """Subgradient descent with decreasing step size."""
    n = len(a)
    best_a = a.copy()
    best_obj = compute_obj(best_a)
    a = best_a.copy()
    eta = eta0
    no_improve = 0
    
    for it in range(max_iter):
        a, M = subgradient_step(a, eta)
        obj = compute_obj(a)
        
        if obj < best_obj - 1e-12:
            best_a = a.copy()
            best_obj = obj
            no_improve = 0
        else:
            no_improve += 1
        
        eta *= decay
        
        if verbose and it % 500 == 0:
            print(f"    SGD it={it}: obj={compute_obj(a):.6f}, best={best_obj:.6f}, eta={eta:.6f}")
        
        if no_improve > patience:
            break
    
    return best_a, best_obj


def softmax_objective_and_grad(a, T):
    """Compute soft-smoothed objective and its gradient."""
    n = len(a)
    conv = np.convolve(a, a)
    S = float(np.sum(a))
    
    max_c = np.max(conv)
    shifted = (conv - max_c) / T
    exp_s = np.exp(shifted)
    sum_exp = np.sum(exp_s)
    w = exp_s / sum_exp
    C = max_c + T * np.log(sum_exp)
    
    # v[i] = sum_k w[k] * a[k-i]
    v = np.zeros(n)
    for i in range(n):
        v[i] = np.dot(a, w[i:i+n])
    
    f = 2.0 * n * C / (S * S)
    grad = (4.0 * n / (S * S)) * v - (4.0 * n * C / (S ** 3))
    
    return f, grad


def lbfgsb_optimize(a, T_seq, max_iter=500, verbose=False):
    """L-BFGS-B optimization with annealed temperature."""
    n = len(a)
    best_a = a.copy()
    best_obj = compute_obj(best_a)
    
    for T in T_seq:
        def obj_func(x):
            x = np.maximum(x, 0)
            s = x.sum()
            if s < 1e-10:
                return 1e10
            x = x / s
            return compute_obj(x)
        
        def grad_func(x):
            x = np.maximum(x, 0)
            s = x.sum()
            if s < 1e-10:
                return np.ones(n)
            x = x / s
            _, g = softmax_objective_and_grad(x, T)
            return g
        
        try:
            res = minimize(obj_func, best_a, method='L-BFGS-B', 
                         jac=grad_func, bounds=[(0, None)] * n,
                         options={'maxiter': max_iter, 'ftol': 1e-14, 'gtol': 1e-8})
            if res.success or res.fun < best_obj:
                x = np.maximum(res.x, 0)
                s = x.sum()
                if s > 1e-10:
                    x = x / s
                    obj = compute_obj(x)
                    if obj < best_obj:
                        best_obj = obj
                        best_a = x.copy()
        except Exception as e:
            pass
    
    return best_a, best_obj


def build_A_matrix(a):
    """Build constraint matrix for LP: A[k,j] = a[k-j] if valid."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for k in range(2 * n - 1):
        j_lo = max(0, k - n + 1)
        j_hi = min(n - 1, k)
        if j_lo <= j_hi:
            A[k, j_lo:j_hi+1] = a[k - j_hi : k - j_lo + 1][::-1]
    return A


def lp_alternate(a, max_iter=200, upper_bound=1.0, verbose=False):
    """LP alternating: fix a, solve LP for g, blend."""
    n = len(a)
    A = build_A_matrix(a)
    best_a = a.copy()
    best_obj = compute_obj(best_a)
    
    for it in range(max_iter):
        conv = np.convolve(a, a)
        M = float(np.max(conv))
        
        c = -np.ones(n)
        b_ub = np.full(2 * n - 1, M)
        bounds = [(0.0, upper_bound)] * n
        
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
        if res.status != 0:
            break
        
        g = res.x
        g_sum = float(-res.fun)
        
        if g_sum <= 1.0 + 1e-6:
            # At optimum (or very close), try to tighten M
            break
        
        g_norm = g / g_sum
        # Blend
        t = 0.5
        a_new = (1 - t) * a + t * g_norm
        a_new = np.maximum(a_new, 0)
        s = a_new.sum()
        if s < 1e-10:
            break
        a_new = a_new / s
        
        obj_new = compute_obj(a_new)
        if obj_new < best_obj:
            best_a = a_new.copy()
            best_obj = obj_new
        a = a_new
    
    return best_a, best_obj


def random_refine(a, rng, max_iter=5000, width_frac=1/30):
    """Random block perturbation local search."""
    best_a = a.copy()
    best_obj = compute_obj(best_a)
    n = len(best_a)
    
    for it in range(max_iter):
        width = max(1, int(n * width_frac))
        start = int(rng.integers(0, n - width + 1))
        candidate = best_a.copy()
        scale = np.exp(rng.normal(0, 0.2, width))
        candidate[start:start+width] *= scale
        s = candidate.sum()
        if s < 1e-10:
            continue
        candidate = candidate / s
        obj_new = compute_obj(candidate)
        if obj_new < best_obj:
            best_a = candidate
            best_obj = obj_new
    
    return best_a, best_obj

import numpy as np

def compute_obj(a):
    n = len(a)
    if n == 0:
        return np.inf
    conv = np.convolve(a, a)
    max_b = float(np.max(conv))
    sum_a = float(np.sum(a))
    if sum_a < 0.01:
        return np.inf
    return 2.0 * n * max_b / (sum_a * sum_a)


def compute_gradient_info(a):
    """Compute conv, active set weights, C value, and gradient components."""
    n = len(a)
    conv = np.convolve(a, a)
    M = float(np.max(conv))
    active = (conv >= M - 1e-14 * max(1.0, abs(M)))
    w = active.astype(float)
    w /= w.sum()
    S = float(np.sum(a))
    # v[i] = sum_k w[k] * a[k-i]
    v = np.zeros(n)
    for i in range(n):
        v[i] = np.dot(a, w[i:i+n])
    # grad[i] = 4n*v[i]/S^2 - 4nC/S^3
    grad = (4.0 * n / (S * S)) * v - (4.0 * n * M / (S ** 3))
    return conv, w, M, S, v, grad

    A = np.zeros((2 * n - 1, n))
    for k in range(2 * n - 1):
        j_lo = max(0, k - n + 1)
        j_hi = min(n - 1, k)
        if j_lo <= j_hi:
            A[k, j_lo:j_hi+1] = a[k - j_hi:k - j_lo + 1][::-1]
    return A


def lp_alternate(a, max_iter=150, verbose=False, time_limit=None):
    """LP alternating optimization."""
    n = len(a)
    A = build_A_matrix(a)
    best_a = a.copy()
    best_obj = compute_obj(best_a)
    t0 = time.monotonic()

    for it in range(max_iter):
        if time_limit is not None and (time.monotonic() - t0) > time_limit:
            break
        conv = np.convolve(a, a)
        M = float(np.max(conv))

        c = -np.ones(n)
        b_ub = np.full(2 * n - 1, M)
        bounds = [(0.0, 1.0)] * n

        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
        if res.status != 0:
            break

        g = res.x
        g_sum = float(-res.fun)

        if g_sum <= 1.0 + 1e-6:
            break

        g_norm = g / g_sum
        for t_blend in [0.5, 0.3, 0.7]:
            a_new = (1 - t_blend) * a + t_blend * g_norm
            a_new = np.maximum(a_new, 0)
            s = a_new.sum()
            if s < 1e-12:
                continue
            a_new = a_new / s
            obj_new = compute_obj(a_new)
            if obj_new < best_obj:
                best_obj = obj_new
                best_a = a_new.copy()
                a = a_new
                break
        else:
            a = a_new
            s = a.sum()
            if s > 1e-12:
                a = a / s

    return best_a, best_obj


def random_refine(a, rng, max_iter=5000, time_limit=None):
    """Random block perturbation local search."""
    best_a = a.copy()
    best_obj = compute_obj(best_a)
    n = len(best_a)
    t0 = time.monotonic()

    for it in range(max_iter):
        if time_limit is not None and (time.monotonic() - t0) > time_limit:
            break
        width = max(1, n // 25)
        start = int(rng.integers(0, n - width + 1))
        candidate = best_a.copy()
        scale = np.exp(rng.normal(0, 0.15, width))
        candidate[start:start+width] *= scale
        s = candidate.sum()
        if s < 1e-12:
            continue
        candidate = candidate / s
        obj = compute_obj(candidate)
        if obj < best_obj:
            best_a = candidate
            best_obj = obj

    return best_a, best_obj


def elementwise_refine(a, rng, max_iter=3000, time_limit=None):
    """Single-element perturbation for fine-tuning."""
    best_a = a.copy()
    best_obj = compute_obj(best_a)
    n = len(best_a)
    t0 = time.monotonic()

    for it in range(max_iter):
        if time_limit is not None and (time.monotonic() - t0) > time_limit:
            break
        idx = int(rng.integers(0, n))
        candidate = best_a.copy()
        candidate[idx] *= np.exp(rng.normal(0, 0.1))
        candidate[idx] = max(0.0, candidate[idx])
        s = candidate.sum()
        if s < 1e-12:
            continue
        candidate = candidate / s
        obj = compute_obj(candidate)
        if obj < best_obj:
            best_a = candidate
            best_obj = obj

    return best_a, best_obj


def propose_candidate(seed=42, budget_s=1500, **kwargs):
    """
    Multi-phase optimizer: mirror descent + L-BFGS-B (annealed soft-max)
    + LP alternating + random/elementwise refinement.
    Tries multiple sequence lengths and restarts.
    """
    rng = np.random.default_rng(seed)
    start_time = time.monotonic()
    deadline = start_time + float(budget_s) - 5.0

    best_seq = None
    best_score = np.inf

    # Try parent sequence
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64)
        parent = np.clip(parent, 0, 1000.0)
        if parent.size > 0 and np.isfinite(parent).all():
            obj = compute_obj(parent)
            if obj < best_score:
                best_score = obj
                best_seq = parent.copy()
            print(f"Parent (n={parent.size}): obj={obj:.6f}")
    except Exception:
        pass

    def time_left():
        return deadline - time.monotonic()

    # Focus on n values around the parent and some alternatives
    n_values = [150, 130, 170, 200, 100, 250]

    for n in n_values:
        if time_left() < 40:
            break
        print(f"\n=== n={n}, {time_left():.0f}s remaining ===")

        num_restarts = 4 if n <= 170 else 2

        for r in range(num_restarts):
            if time_left() < 25:
                break

            # Initializations
            if r == 0 and best_seq is not None and len(best_seq) == n:
                a = best_seq.copy()
            elif r == 0:
                a = np.ones(n) / n
            elif r == 1:
                # Cosine modulation
                freq = rng.uniform(0.5, 3.0)
                phase = rng.uniform(0, 2 * np.pi)
                a = 0.5 + 0.4 * np.cos(2 * np.pi * freq * np.arange(n) / n + phase)
                a = np.maximum(a, 0.05)
                a /= a.sum()
            elif r == 2:
                # Bimodal
                a = np.ones(n)
                mid = n // 2
                a[:mid] = rng.uniform(0.5, 1.0)
                a[mid:] = rng.uniform(0.2, 0.8)
                a /= a.sum()
            else:
                a = rng.random(n) + 0.1
                a /= a.sum()

            # Phase 1: Mirror descent
            t_lim = min(30.0, time_left() * 0.3)
            a1, obj1 = mirror_descent(a, max_iter=5000, eta0=0.8,
                                      decay=0.9997, patience=400,
                                      verbose=False, time_limit=t_lim)
            print(f"  MD r={r}: obj={obj1:.6f}")
            if obj1 < best_score:
                best_score = obj1
                best_seq = a1.copy()
                print(f"    NEW BEST: {best_score:.6f}")

            if time_left() < 20:
                continue

            # Phase 2: L-BFGS-B with annealed T
            T_seq = [0.05, 0.02, 0.01, 0.005, 0.002]
            t_lim = min(20.0, time_left() * 0.25)
            a2, obj2 = lbfgsb_optimize(a1, T_seq=T_seq,
                                       max_iter_per_T=80, verbose=False,
                                       time_limit=t_lim)
            print(f"  LBFGS r={r}: obj={obj2:.6f}")
            if obj2 < best_score:
                best_score = obj2
                best_seq = a2.copy()
                print(f"    NEW BEST: {best_score:.6f}")

            if time_left() < 15:
                continue

            # Phase 3: More mirror descent from L-BFGS result
            t_lim = min(20.0, time_left() * 0.2)
            a3, obj3 = mirror_descent(a2, max_iter=3000, eta0=0.4,
                                      decay=0.9997, patience=300,
                                      verbose=False, time_limit=t_lim)
            print(f"  MD2 r={r}: obj={obj3:.6f}")
            if obj3 < best_score:
                best_score = obj3
                best_seq = a3.copy()
                print(f"    NEW BEST: {best_score:.6f}")

            if time_left() < 10:
                continue

            # Phase 4: Random block refinement
            t_lim = min(15.0, time_left() * 0.2)
            a4, obj4 = random_refine(a3, rng, max_iter=8000, time_limit=t_lim)
            print(f"  RLS r={r}: obj={obj4:.6f}")
            if obj4 < best_score:
                best_score = obj4
                best_seq = a4.copy()
                print(f"    NEW BEST: {best_score:.6f}")

    # Final polish phase on the best sequence found
    if best_seq is not None and time_left() > 30:
        n_best = len(best_seq)
        print(f"\n=== Final polish (n={n_best}), {time_left():.0f}s remaining ===")

        # LP alternating
        if n_best <= 300:
            t_lim = min(60.0, time_left() * 0.3)
            a_lp, obj_lp = lp_alternate(best_seq, max_iter=200,
                                        verbose=False, time_limit=t_lim)
            print(f"  Final LP: obj={obj_lp:.6f}")
            if obj_lp < best_score:
                best_score = obj_lp
                best_seq = a_lp.copy()
                print(f"    NEW BEST: {best_score:.6f}")

        # Another round of mirror descent
        if time_left() > 20:
            t_lim = min(30.0, time_left() * 0.3)
            a_md, obj_md = mirror_descent(best_seq, max_iter=5000, eta0=0.3,
                                          decay=0.9998, patience=400,
                                          verbose=False, time_limit=t_lim)
            print(f"  Final MD: obj={obj_md:.6f}")
            if obj_md < best_score:
                best_score = obj_md
                best_seq = a_md.copy()
                print(f"    NEW BEST: {best_score:.6f}")

        # L-BFGS-B with very small T
        if time_left() > 15:
            T_seq = [0.003, 0.001, 0.0005]
            t_lim = min(20.0, time_left() * 0.3)
            a_lb, obj_lb = lbfgsb_optimize(best_seq, T_seq=T_seq,
                                           max_iter_per_T=100, verbose=True,
                                           time_limit=t_lim)
            if obj_lb < best_score:
                best_score = obj_lb
                best_seq = a_lb.copy()
                print(f"    NEW BEST: {best_score:.6f}")

        # Element-wise fine tuning
        if time_left() > 10:
            t_lim = time_left() - 5
            a_el, obj_el = elementwise_refine(best_seq, rng, max_iter=10000,
                                              time_limit=t_lim)
            print(f"  Element refine: obj={obj_el:.6f}")
            if obj_el < best_score:
                best_score = obj_el
                best_seq = a_el.copy()
                print(f"    NEW BEST: {best_score:.6f}")

        # Final random refinement on very small steps
        if time_left() > 5:
            t_lim = time_left() - 2
            a_rf, obj_rf = random_refine(best_seq, rng, max_iter=10000,
                                         time_limit=t_lim)
            if obj_rf < best_score:
                best_score = obj_rf
                best_seq = a_rf.copy()

    if best_seq is None:
        best_seq = np.ones(100) / 100
        best_score = compute_obj(best_seq)

    elapsed = time.monotonic() - start_time
    print(f"\nDONE: n={len(best_seq)}, score={best_score:.6f}, time={elapsed:.1f}s")
    return [float(x) for x in best_seq]
