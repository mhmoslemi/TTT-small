    """Project v onto {x >= 0, sum(x) = S}."""
    n = len(v)
    if n == 0:
        return v

Multi-phase optimization to minimize 2*n*max(conv(f,f))/(sum(f))^2.
Phases: LP direction finding -> QP equalization -> subgradient polish -> coordinate descent.

def normalize_sequence(arr):
    x = np.asarray(arr, dtype=np.float64).ravel()
    if x.size == 0:
        return None
    x = np.where(np.isfinite(x), x, 0.0)
    x = np.maximum(x, 0.0)
    s = float(x.sum())
    if s <= 1e-300:
        return None
    x /= s
    return x


def score_norm(x, n):
    conv = np.convolve(x, x)
    m = float(np.max(conv))
    if not np.isfinite(m):
        return float('inf')
    return float(2.0 * n * m)


def mix_noisy(base, rng, eps):
    f = normalize_sequence(base)
    if f is None:
        return None
    n = f.size
    if eps <= 1e-12:
        return f
    mix = rng.dirichlet(np.ones(n, dtype=np.float64))
    if eps >= 1.0:
        f = mix
    else:
        f = (1.0 - eps) * f + eps * mix
    return normalize_sequence(f)


def interpolate_sequence(x, n):
    x = normalize_sequence(x)
    if x is None or n <= 0:
        return None
    m = x.size
    if m == n:
        return x
    if m == 1:
        return np.ones(n) / n
    old = np.linspace(0.0, 1.0, m)
    new = np.linspace(0.0, 1.0, n)
    out = np.interp(new, old, x)
    return normalize_sequence(out)


def make_conv_template(n):
    m = 2 * n - 1
    k64 = np.arange(m, dtype=np.int64)
    js = np.maximum(0, k64 - n + 1).astype(np.int32)
    je = np.minimum(n - 1, k64).astype(np.int32)
    counts = (je.astype(np.int64) - js.astype(np.int64) + 1).astype(np.int64)
    total = int(counts.sum())
    cum = np.cumsum(counts)
    starts = np.repeat(cum - counts, counts)
    local = np.arange(total, dtype=np.int64) - starts
    row = np.repeat(k64, counts).astype(np.int32)
    col_base = js[row.astype(np.int64)]
    col = (col_base + local).astype(np.int32)
    offsets = (row.astype(np.int64) - col.astype(np.int64)).astype(np.int32)
    A_off = csr_matrix((offsets, (row, col)), shape=(m, n), dtype=np.int32)
    edge_offsets = A_off.data.astype(np.int32, copy=True)
    A_template = A_off.astype(np.float64).copy()
    A_template.data.fill(0.0)
    c_obj = -np.ones(n, dtype=np.float64)
    bounds = [(0.0, None)] * n
    return A_template, edge_offsets, c_obj, bounds, m

    b = np.full(m, M, dtype=np.float64)
    res = linprog(c_obj, A_ub=A, b_ub=b, bounds=bounds, method='highs')
    return res

def subgradient_polish(f, n, time_budget=30.0, max_iters=2000):
    """Polish using subgradient method with adaptive step size."""
    f = f.copy()
    best_f = f.copy()
    best_score = score_norm(f, n)
    deadline = time.monotonic() + time_budget
    
    # Initial step size
    alpha = 0.01
    f_avg = f.copy()
    
    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        
        conv = np.convolve(f, f)
        M = float(np.max(conv))
        # Subgradient: 2 * f reversed and shifted by argmax
        k_star = int(np.argmax(conv))
        grad = np.zeros(n)
        for i in range(n):
            j = k_star - i
            if 0 <= j < n:
                grad[i] = 2.0 * f[j]
        
        # Adaptive step: alpha_t = c / sqrt(t+1)
        alpha_t = 0.5 / np.sqrt(it + 1)
        
        f_new = f - alpha_t * grad
        # Project onto simplex
        f_new = project_simplex(f_new, 1.0)
        
        f_avg = (it * f_avg + f_new) / (it + 1)
        f = f_new
        
        sc = score_norm(f, n)
        if sc < best_score:
            best_score = sc
            best_f = f.copy()
        
        sc_avg = score_norm(f_avg, n)
        if sc_avg < best_score:
            best_score = sc_avg
            best_f = f_avg.copy()
    
    return best_f, float(best_score)


def project_simplex(v, S=1.0):
    """Project v onto {x >= 0, sum(x) = S}."""
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * (np.arange(1, n+1)) > (cssv - S)
    rho = np.where(cond)[0]
    if len(rho) == 0:
        return np.full(n, S / n)
    rho = rho[-1]
    theta = (cssv[rho] - S) / (rho + 1)
    return np.maximum(v - theta, 0.0)
