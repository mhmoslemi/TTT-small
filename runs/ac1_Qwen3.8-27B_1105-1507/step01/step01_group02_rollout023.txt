```python
"""
Hybrid optimizer for minimum peak autocorrelation sequences.
Combines differential evolution, projected subgradient descent,
LP-based improvement, and local coordinate descent.
"""

import numpy as np
import time
from scipy.optimize import linprog

def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    N = 1 << (size - 1).bit_length()
    fa = np.fft.rfft(a, N)
    fc = fa * fa
    result = np.fft.irfft(fc, N)[:size]
    result[result < 0] = 0.0
    return result

def project_simplex(v):
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cv = np.cumsum(u) - 1.0
    rho_candidates = np.where(u * np.arange(1, n + 1) > cv)[0]
    if len(rho_candidates) == 0:
        return np.ones(n) / n
    rho = rho_candidates[-1]
    theta = cv[rho] / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def compute_objective(a):
    """Compute 2*n*max(conv(a,a))/sum(a)^2 for non-negative a."""
    n = len(a)
    c = fast_convolve(a)
    max_b = np.max(c)
    s = np.sum(a)
    if s < 1e-10:
        return float('inf'), c
    return 2.0 * n * max_b / (s * s), c

def subgradient_max_conv(a, c, tol=1e-4):
    """Subgradient of max(c) where c = convolve(a,a). Averaged over active peaks."""
    n = len(a)
    peak = np.max(c)
    active = np.where(c >= peak * (1.0 - tol))[0]
    if len(active) > 50:
        # Keep only top peaks
        top = np.argsort(c[active])[::-1][:20]
        active = active[top]
    
    grad = np.zeros(n)
    for k in active:
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        if lo <= hi:
            grad[lo:hi+1] += 2.0 * a[k-hi:k-lo+1][::-1]
    grad /= max(1, len(active))
    return grad

def gradient_step_with_linesearch(a, grad, c_max_target, max_steps=20):
    """
    Take a gradient step with backtracking line search.
    Try to reduce max(convolve(a,a)) below c_max_target.
    """
    n = len(a)
    best_a = a.copy()
    best_c = fast_convolve(best_a)
    best_max = np.max(best_c)
    
    lr = 0.1 / n
    for step in range(max_steps):
        a_try = project_simplex(a - lr * grad)
        c_try = fast_convolve(a_try)
        max_try = np.max(c_try)
        
        if max_try < best_max * (1 - 1e-6):
            best_a = a_try.copy()
            best_max = max_try
            lr = min(lr * 1.5, 1.0 / np.sqrt(n))
        else:
            lr *= 0.5
            if lr < 1e-10 / n:
                break
    
    return best_a, best_max

def lp_improve(a, M, deadline):
    """
    LP step: find b >= 0 with convolve(a, b) <= M, maximizing sum(b).
    Returns blended result or None.
    """
    n = len(a)
    if n > 3000:
        return None
    
    A = np.zeros((2 * n - 1, n), dtype=np.float64)
    for k in range(2 * n - 1):
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        A[k, lo:hi + 1] = a[k - hi:k - lo + 1][::-1]
    
    c_obj = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    
    try:
        res = linprog(c_obj, A_ub=A, b_ub=b_ub, 
                      bounds=[(0.0, None)] * n, 
                      method='highs',
                      options={'time_limit': 10.0})

def de_optimize(n, deadline, rng, pop_size=30, max_gens=500):
    """
    Differential evolution optimization.
    """
    # Initialize population
    pop = np.zeros((pop_size, n))
    pop[0] = np.ones(n) / n  # Uniform
    for i in range(1, pop_size):
        if i == 1:
            # Cosine window
            x = np.linspace(0, 1, n)
            pop[i] = 0.5 * (1 + np.cos(2*np.pi*x)) + 0.01
        elif i == 2:
            # Triangle
            x = np.linspace(0, 1, n)
            pop[i] = 1 - np.abs(2*x - 1) + 0.01
        elif i == 3:
            # Gaussian-like
            x = np.linspace(0, 1, n)
            pop[i] = np.exp(-20*(x-0.5)**2) + 0.01
        else:
            pop[i] = rng.random(n) + 0.01
        pop[i] = pop[i] / pop[i].sum()
    
    # Evaluate initial population
    scores = np.zeros(pop_size)
    for i in range(pop_size):
        scores[i], _ = compute_objective(pop[i])
    
    best_idx = np.argmin(scores)
    best_score = scores[best_idx]
    best_a = pop[best_idx].copy()
    
    F = 0.8  # Mutation factor
    CR = 0.9  # Crossover rate
    gen = 0
    
    while gen < max_gens and time.time() < deadline:
        gen += 1
        # Adaptive mutation factor
        F = max(0.3, 0.8 - 0.3 * gen / max_gens)
        
        for i in range(pop_size):
            if time.time() > deadline:
                break
            
            # Select three distinct individuals
            mask = np.ones(pop_size, dtype=bool)
            mask[i] = False
            candidates = np.where(mask)[0]
            r1, r2, r3 = rng.choice(candidates, size=3, replace=False)
            
            # Mutation
            v = pop[r1] + F * (pop[r2] - pop[r3])
            
            # Crossover
            trial = np.zeros(n)
            jitter = rng.random(n)
            j0 = rng.randint(0, n)
            for j in range(n):
                if jitter[(j + j0) % n] < CR or i == (j + j0) % n:
                    trial[j] = v[j]
                else:
                    trial[j] = pop[i][j]

            trial = project_simplex(trial)
            
            # Evaluate
            trial_score, _ = compute_objective(trial)

                scores[i] = trial_score
                pop[i] = trial
                if trial_score < best_score:
                    best_score = trial_score
                    best_a = trial.copy()
        
        # Update best
        if gen % 50 == 0:
            print(f"  DE gen {gen}/{max_gens}, best_score={best_score:.6f}")
```

```python
"""
Hybrid optimizer combining LP-based alternating minimization,
differential evolution, and projected subgradient descent
for minimum peak linear autocorrelation sequences.
"""

import numpy as np
import time
from scipy.optimize import linprog


def fast_convolve(a):
    """Compute full linear convolution of a with itself using FFT."""
    n = len(a)
    size = 2 * n - 1
    N = 1 << (size - 1).bit_length()
    fa = np.fft.rfft(a, N)
    fc = fa * fa
    result = np.fft.irfft(fc, N)[:size]
    result[result < 0] = 0.0
    return result


def project_simplex(v):
    """Project vector v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cv = np.cumsum(u) - 1.0
    rho_candidates = np.where(u * np.arange(1, n + 1) > cv)[0]
    if len(rho_candidates) == 0:
        return np.ones(n) / n
    rho = rho_candidates[-1]
    theta = cv[rho] / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def compute_objective(a):
    """Compute 2*n*max(conv(a,a))/sum(a)^2 for non-negative a."""
    n = len(a)
    c = fast_convolve(a)
    max_b = np.max(c)
    s = np.sum(a)
    if s < 1e-10:
        return float('inf'), c
    return 2.0 * n * max_b / (s * s), c


def build_conv_matrix(a):
    """Build the convolution matrix A where (A @ b) = convolve(a, b)."""
    n = len(a)
    A = np.zeros((2 * n - 1, n), dtype=np.float64)
    for k in range(2 * n - 1):
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        A[k, lo:hi + 1] = a[k - hi:k - lo + 1][::-1]
    return A


def lp_min_peak(a_fixed):
    """
    Given fixed a, find b >= 0, sum(b)=1 that minimizes max(convolve(a, b)).
    Returns optimal b and the optimal peak value.
    """
    n = len(a_fixed)
    A = build_conv_matrix(a_fixed)
    
    # Variables: [b_0, ..., b_{n-1}, t]
    # Minimize t
    c_obj = np.zeros(n + 1)
    c_obj[-1] = 1.0
    
    # Inequality: A[k, :] @ b - t <= 0 for each k
    # Row k: [A[k,0], ..., A[k,n-1], -1]
    A_ub = np.hstack([A, -np.ones((2*n-1, 1))])
    b_ub = np.zeros(2*n - 1)
    
    # Equality: sum(b) = 1
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])
    
    # Bounds: b_j >= 0, t is free
    bounds = [(0, None)] * n + [(-np.inf, None)]
    
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub,
                      A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs',
                      options={'time_limit': 15.0, 'presolve': True})
        if res.success:
            b_opt = res.x[:n]
            t_opt = res.x[-1]
            b_opt = np.maximum(b_opt, 0)
            return b_opt, t_opt
    except Exception:
        pass
    return None, None


def lp_alternating(a, b, deadline, max_iters=100, tol=1e-8):
    """
    Alternating LP minimization.
    Alternate between fixing a and optimizing b, then fixing b and optimizing a.
    """
    n = len(a)
    best_score, _ = compute_objective(a)
    best_a = a.copy()
    best_b = b.copy()
    
    prev_max = np.inf
    for it in range(max_iters):
        if time.time() > deadline:
            break
        
        # Fix a, optimize b
        b_new, t_b = lp_min_peak(a)
        if b_new is not None:
            b = b_new
        
        # Fix b, optimize a
        a_new, t_a = lp_min_peak(b)
        if a_new is not None:
            a = a_new
        
        # Evaluate
        c = fast_convolve(a)
        cur_max = np.max(c)
        score, _ = compute_objective(a)
        
        if score < best_score:
            best_score = score
            best_a = a.copy()
        
        # Check convergence
        if abs(cur_max - prev_max) < tol:
            break
        prev_max = cur_max
    
    return best_a, best_score


def subgradient_optimize(a, deadline, rng, max_iters=200000):
    """
    Projected subgradient descent with adaptive step size.
    """
    n = len(a)
    a = project_simplex(a.copy().astype(np.float64))
    
    c = fast_convolve(a)
    best_peak = np.max(c)
    best_a = a.copy()
    best_score, _ = compute_objective(a)
    
    lr = 0.5 / np.sqrt(n)
    lr_min = lr * 1e-5
    no_improve = 0
    max_no_improve = 3000
    
    it = 0
    while time.time() < deadline and it < max_iters:
        it += 1
        
        c = fast_convolve(a)
        peak = np.max(c)
        
        if peak < best_peak * (1 - 1e-9):
            best_peak = peak
            best_a = a.copy()
            best_score, _ = compute_objective(best_a)
            no_improve = 0
        else:
            no_improve += 1
        
        if no_improve >= max_no_improve:
            lr = max(lr * 0.6, lr_min)
            no_improve = 0
            if lr < lr_min * 3:
                # Perturb to escape
                a = project_simplex(a + rng.normal(0, 0.05/np.sqrt(n), n))
                max_no_improve = int(max_no_improve * 1.5)
                if max_no_improve > 50000:
                    break
        
        # Get active peaks
        threshold = peak * (1.0 - 1e-5)
        peak_indices = np.where(c >= threshold)[0]
        if len(peak_indices) > 30:
            top = np.argsort(c[peak_indices])[::-1][:10]
            peak_indices = peak_indices[top]
        
        # Subgradient
        grad = np.zeros(n)
        for k in peak_indices:
            lo = max(0, k - n + 1)
            hi = min(n - 1, k)
            if lo <= hi:
                grad[lo:hi+1] += 2.0 * a[k-hi:k-lo+1][::-1]
        grad /= max(1, len(peak_indices))
        
        # Step
        a = project_simplex(a - lr * grad)
        
        # Adaptive lr
        if it % 500 == 0:
            lr = max(lr * 0.95, lr_min)
    
    return best_a, best_score


def de_optimize(n, deadline, rng, pop_size=30, max_gens=200):
    """
    Differential evolution for global exploration.
    """
    pop = np.zeros((pop_size, n))
    pop[0] = np.ones(n) / n
    x = np.linspace(0, 1, n)
    pop[1] = (0.5 * (1 + np.cos(2*np.pi*x)) + 0.01)
    pop[1] /= pop[1].sum()
    pop[2] = (1 - np.abs(2*x - 1) + 0.01)

    pop[3] = (np.exp(-20*(x-0.5)**2) + 0.01)
    pop[3] /= pop[3].sum()
    for i in range(4, pop_size):

        pop[i] /= pop[i].sum()
    
    scores = np.zeros(pop_size)
    for i in range(pop_size):
        scores[i], _ = compute_objective(pop[i])
    
    best_idx = np.argmin(scores)
    best_score = scores[best_idx]
    best_a = pop[best_idx].copy()
    
    gen = 0
    while gen < max_gens and time.time() < deadline:
        gen += 1
        F = max(0.3, 0.8 * (1 - gen / max_gens))
        CR = 0.8
        
        for i in range(pop_size):
            if time.time() > deadline:
                break
            
            mask = np.ones(pop_size, dtype=bool)
            mask[i] = False
            candidates = np.where(mask)[0]
            r1, r2, r3 = rng.choice(candidates, size=3, replace=False)
            
            v = pop[r1] + F * (pop[r2] - pop[r3])
            
            # Crossover
            trial = pop[i].copy()
            j0 = rng.randint(0, n)
            for j in range(n):
                if rng.random() < CR or j == j0:
                    trial[j] = v[j]
            
            trial = project_simplex(trial)
            trial_score, _ = compute_objective(trial)
            
            if trial_score < scores[i]:
                scores[i] = trial_score
                pop[i] = trial
                if trial_score < best_score:
                    best_score = trial_score
                    best_a = trial.copy()
    
    return best_a, best_score


def coordinate_descent(a, deadline, max_rounds=100):
    """
    Coordinate descent: try adjusting individual coordinates.
    """
    n = len(a)
    a = a.copy()
    c = fast_convolve(a)
    peak = np.max(c)
    best_score, _ = compute_objective(a)
    
    for rd in range(max_rounds):
        if time.time() > deadline:
            break
        
        improved = False
        c = fast_convolve(a)
        peak = np.max(c)
        
        # Try random subsets of coordinates
        num_tries = min(n, 50)
        indices = np.random.choice(n, num_tries, replace=False)
        
        for i in indices:
            if time.time() > deadline:
                break
            base = a[i]
            for frac in [0.5, 0.7, 1.3, 1.5, 0.0, 0.3]:
                new_val = max(0.0, base + (frac - 1.0) * base) if frac != 0 else 0.0
                if abs(new_val - base) < 1e-12:
                    continue
                trial = a.copy()
                trial[i] = new_val
                ts = trial.sum()
                if ts < 1e-10:
                    continue
                trial = trial / ts
                c_t = fast_convolve(trial)
                mt = np.max(c_t)
                if mt < peak * (1 - 1e-7):
                    a = trial
                    peak = mt
                    improved = True
                    break
        
        if not improved:
            # Broader perturbation
            trial = a.copy()
            idx = np.random.choice(n, min(10, n), replace=False)
            for i in idx:
                trial[i] = max(0, trial[i] * (0.3 + 1.4 * np.random.random()))
            ts = trial.sum()
            if ts > 1e-10:
                trial /= ts
                mt = np.max(fast_convolve(trial))
                if mt < peak * (1 - 1e-7):
                    a = trial
                    peak = mt
                    improved = True
        
        if not improved:
            break
    
    final_score, _ = compute_objective(a)
    return a, final_score
```

"""
LP-alternating optimizer for minimum peak linear autocorrelation.
Uses a block-coordinate LP method as the primary optimization engine,
with DE for global exploration and subgradient descent for local polish.
"""

import numpy as np
import time
from scipy.optimize import linprog


def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    N = 1 << (size - 1).bit_length()
    fa = np.fft.rfft(a, N)
    fc = fa * fa
    result = np.fft.irfft(fc, N)[:size]
    result[result < 0] = 0.0
    return result


def project_simplex(v):
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cv = np.cumsum(u) - 1.0
    rho_candidates = np.where(u * np.arange(1, n + 1) > cv)[0]
    if len(rho_candidates) == 0:
        return np.ones(n) / n
    rho = rho_candidates[-1]
    theta = cv[rho] / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def compute_objective(a):
    n = len(a)
    c = fast_convolve(a)
    max_b = np.max(c)
    s = np.sum(a)
    if s < 1e-10:
        return float('inf'), c
    return 2.0 * n * max_b / (s * s), c
