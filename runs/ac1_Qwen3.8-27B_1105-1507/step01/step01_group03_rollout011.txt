            continue

        G = float(np.max(np.convolve(g, g)))
        a_coef = 2.0 * (r_max - M)
        b_coef = G + M - 2.0 * r_max
        if b_coef > 1e-14:
            t_max = -a_coef / b_coef
        else:
            t_max = 1.0
        t_max = max(1e-12, min(1.0, 0.6 * t_max))

        # Backtracking line search
        candidates_f = None
        candidates_score = best_score
        for t in [t_max, 0.7 * t_max, 0.4 * t_max, 0.2 * t_max, 0.05 * t_max, 0.01 * t_max]:
            f_new = (1.0 - t) * f + t * g
            f_new = np.maximum(f_new, 0.0)
            ss = f_new.sum()
            if ss <= 1e-300:
                continue
            f_new /= ss
            sc = score_seq(f_new)
            if np.isfinite(sc) and sc < candidates_score:
                candidates_score = sc
                candidates_f = f_new

        if candidates_f is not None:
            f = candidates_f
            best_f = f.copy()
            best_score = candidates_score
            stall = 0
        else:
            stall += 1
            # Escalating noise for escape
            noise_scale = min(0.5, 0.03 * (1 + stall * 0.5))
            f = np.maximum(best_f + rng.normal(0, noise_scale, n), 0.0)
            ss = f.sum()
            f /= ss if ss > 1e-300 else 1.0
            if stall > stall_threshold:
                break

    return best_f, best_score


def generate_starts(n, rng, best_f, given_f):
    """Generate diverse starting points for LP optimization."""
    starts = []

    # Uniform
    starts.append(np.ones(n, dtype=np.float64) / n)

    # Upsampled best
    if best_f is not None:
        up = interpolate_to_length(best_f, n)
        if up is not None:
            starts.append(up)
        # Noisy upsampled
        noisy = np.maximum(up + rng.normal(0, 0.03, n), 0.0)
        starts.append(normalize_sequence(noisy))

    # Given sequence
    if given_f is not None:
        g = interpolate_to_length(given_f, n)
        if g is not None:
            starts.append(g)
            noisy_g = np.maximum(g + rng.normal(0, 0.02, n), 0.0)
            starts.append(normalize_sequence(noisy_g))

    # Random Dirichlet (concentrated)
    try:
        d = rng.dirichlet(np.full(n, 0.3, dtype=np.float64))
        starts.append(d)
    except Exception:
        pass

    # Random Dirichlet (spread)
    try:
        d2 = rng.dirichlet(np.ones(n, dtype=np.float64))
        starts.append(d2)
    except Exception:
        pass

    # Smooth random (low-pass filtered noise)
    try:
        raw = np.abs(rng.normal(0, 1, n))
        # Simple moving average smoothing
        kernel_size = max(3, n // 32)
        kernel = np.ones(kernel_size) / kernel_size
        smoothed = np.convolve(raw, kernel, mode='same')
        f_smooth = normalize_sequence(smoothed)
        if f_smooth is not None:
            starts.append(f_smooth)
    except Exception:
        pass

    # Cosine-weighted
    x = np.linspace(0, np.pi, n)
    starts.append(normalize_sequence(np.abs(np.cos(x))))

    # Trimmed: emphasize middle
    x = np.arange(n)
    center = (n - 1) / 2.0
    starts.append(normalize_sequence(np.exp(-0.5 * ((x - center) / (n * 0.2)) ** 2)))

    # Deduplicate and limit
    unique_starts = []
    for s in starts:
        s = normalize_sequence(s)
        if s is None:
            continue
        is_dup = False
        for u in unique_starts:
            if np.linalg.norm(s - u) < 1e-4:
                is_dup = True
                break
        if not is_dup:
            unique_starts.append(s)

    return unique_starts[:6]


def propose_candidate(seed=42, budget_s=1500, **kwargs):
    """
    Search for a low-scoring non-negative sequence using LP-based iteration,
    mirror descent, and diverse random restarts. Returns the best sequence
    found within the time budget.
    """
    rng = np.random.default_rng(int(seed) if seed is not None else 42)
    start_time = time.monotonic()
    total_budget = max(1.0, float(budget_s))
    deadline = start_time + total_budget - 15.0

    best_f = None
    best_score = float("inf")
    given_f = None

    # Evaluate given sequence if available
    if total_budget >= 20.0:
        try:
            raw = np.asarray(height_sequence_1, dtype=np.float64)
            given_f = normalize_sequence(raw)
            if given_f is not None:
                sc = score_seq(given_f)
                if np.isfinite(sc) and sc < best_score:
                    best_score = sc
                    best_f = given_f.copy()
                print(f"Given: n={given_f.size}, score={sc:.6f}", flush=True)
        except Exception:
            given_f = None

    # Try uniform at various lengths
    for n in [32, 64, 128, 256, 512, 1024, 2048]:
        f = np.ones(n, dtype=np.float64) / n
        sc = score_seq(f)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_f = f.copy()

    print(f"After uniform: best_score={best_score:.6f}, n={best_f.size if best_f is not None else 0}", flush=True)

    # Main LP optimization phase
    # Plan: try multiple sizes, allocate time based on potential
    lp_plan = [
        (256, 60),     # Quick warm-up
        (512, 150),    # Medium
        (1024, 400),   # Main
        (2048, 350),   # Large
        (1024, 200),   # Second pass at 1024 with better starts
    ]

    for pass_idx, (n, alloc) in enumerate(lp_plan):
        rem = deadline - time.monotonic()
        if rem < 20.0:
            break
        alloc = min(alloc, rem - 10.0)
        if alloc < 15.0:
            continue

        print(f"\n=== LP Phase {pass_idx}: n={n}, budget={alloc:.0f}s, current_best={best_score:.6f} ===", flush=True)

        try:
            template = make_conv_template(n)
        except MemoryError:
            print(f"  MemoryError for n={n}, skipping", flush=True)
            continue
        except Exception as e:
            print(f"  Template build failed: {e}", flush=True)
            continue

        starts = generate_starts(n, rng, best_f, given_f)
        if not starts:
            starts = [np.ones(n, dtype=np.float64) / n]

        m = len(starts)
        time_per_start = alloc / m

        for i, init in enumerate(starts):
            rem = deadline - time.monotonic()
            if rem < 5.0:
                break
            tb = min(time_per_start, rem - 3.0)
            if tb < 3.0:
                continue

            print(f"  Start {i+1}/{m}, budget={tb:.0f}s", flush=True)
            f_res, s_res = lp_optimize(n, init, tb, rng, template)

            if np.isfinite(s_res) and s_res < best_score:
                best_score = s_res
                best_f = np.asarray(f_res, dtype=np.float64).copy()
                print(f"  *** New global best: n={n}, score={best_score:.8f} ***", flush=True)

        # Mirror descent polish on the best found so far at this size
        rem = deadline - time.monotonic()
        if rem > 20.0 and best_f is not None:
            n_md = best_f.size
            if n_md == n or (n_md != n and rem > 40.0):
                print(f"  Mirror descent polish (n={n_md}, budget={rem - 5:.0f}s)", flush=True)
                n_steps = int((rem - 5.0) * 100)  # ~100 iterations/sec estimate
                n_steps = max(100, min(n_steps, 50000))
                f_md, s_md = mirror_descent_optimize(
                    best_f if n_md == n else interpolate_to_length(best_f, n),
                    n_steps, eta_init=0.5, decay=0.005, rng=rng
                )
                if np.isfinite(s_md) and s_md < best_score:
                    best_score = s_md
                    best_f = f_md.copy()
                    print(f"  *** MD improved: score={best_score:.8f} ***", flush=True)

        try:
            del template
        except Exception:
            pass

    # Final: try a few more random restarts if time permits
    rem = deadline - time.monotonic()
    if rem > 30.0 and best_f is not None:
        n = best_f.size
        print(f"\n=== Final random restarts: n={n}, budget={rem:.0f}s ===", flush=True)
        try:
            template = make_conv_template(n)
            n_attempts = max(1, int(rem / 20.0))
            for att in range(n_attempts):
                rem2 = deadline - time.monotonic()
                if rem2 < 10.0:
                    break
                tb = min(15.0, rem2 - 5.0)
                # Perturb the best
                noise_scale = 0.1 + 0.1 * (att % 5)
                perturbed = np.maximum(best_f + rng.normal(0, noise_scale, n), 0.0)
                perturbed = normalize_sequence(perturbed)
                if perturbed is None:
                    continue
                f_res, s_res = lp_optimize(n, perturbed, tb, rng, template)
                if np.isfinite(s_res) and s_res < best_score:
                    best_score = s_res
                    best_f = f_res.copy()
                    print(f"  Restart {att}: new best={best_score:.8f}", flush=True)
            del template
        except Exception as e:
            print(f"  Final restarts failed: {e}", flush=True)

    # Construct final output
    if best_f is None:
        best_f = np.ones(1024, dtype=np.float64) / 1024.0

    # Scale to avoid numerical issues with very small values
    # The score is scale-invariant, so we can use any scale
    # Use sum = sqrt(2n) for numerical stability
    n_final = best_f.size
    target_sum = np.sqrt(2.0 * n_final)
    f_out = best_f * target_sum

    seq = [float(x) for x in f_out]

    # Validate
    try:
        final_score = float(evaluate_sequence(seq))
        if not np.isfinite(final_score):
            raise ValueError("invalid")
    except Exception:
        # Fallback: uniform
        seq = [1.0 / 1024.0] * 1024
        try:
            final_score = float(evaluate_sequence(seq))
        except Exception:
            seq = [1.0] * 1024
            try:
                final_score = float(evaluate_sequence(seq))
            except Exception:
                final_score = float("inf")

    elapsed = time.monotonic() - start_time
    print(f"\nFINAL: n={len(seq)}, local_best={best_score:.8f}, "
          f"evaluate_sequence={final_score:.8f}, time={elapsed:.1f}s", flush=True)
    return seq
