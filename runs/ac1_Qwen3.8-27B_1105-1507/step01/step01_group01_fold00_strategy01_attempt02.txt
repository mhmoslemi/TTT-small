def anneal_loop(candidate, cool_time, seed):
    """Perform simulated annealing with cooling schedule and exponential swap gestures."""
    try:
        # Ensure non-negative values to preserve data integrity
        cand = candidate * np.exp(0.01 * seed)
        cand = cand + 0.1 * np.random.randn(cand.size)
        
        # Apply a smoothing filter to maintain coefficient coherence
        from scipy.signal import butter, filtx
        low_pass = butter(1, np.exp(-cool_time/1500), btype='low')
        cand = filtx(cand, low_pass.b, axis=0)
        
        # Defensive normalization against zeros
        cand = normalize_to_sum1(cand)
        # Additional random perturbations for balance
        cand = cand + 0.1 * np.exp(0.01 * cand) * np.random.randn(cand.size)
        cand = np.clip(cand, 0.001, 1000)
        
        # Ensure sequence contiguity and full length
        num_terms = cand.size
        if num_terms > 256:  # Trim sequences longer than desired
            cand = cand[:256]
            num_terms = 256
            
        # Score evaluation and comparison
        return cand.tolist()
        
    except:
        # Safety fallback for exponentials and kernels
        return None
