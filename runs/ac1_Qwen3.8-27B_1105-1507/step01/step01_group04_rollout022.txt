"""
Optimizes non-negative step function heights to minimize 2n*max(conv(f,f))/(sum(f))^2.
Strategy: Multi-start mirror descent (multiplicative updates) with adaptive step size,
followed by LP-based Frank-Wolfe refinement for top candidates.
Key improvements: better step size schedule, more diverse initializations, 
efficient LP construction, and dedicated polishing phase.
"""

import numpy as np
import time

def _fft_convolve_full(f):
    """Compute full linear convolution of f with itself using FFT."""
    n = len(f)
    if n <= 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    F = np.fft.rfft(f, size)
    h = np.fft.irfft(F * F, size)
    return h[:2 * n - 1]

def _score_raw(f):
    """Fast score computation assuming f >= 0, sum(f) > 0.01."""
    n = len(f)
    h = _fft_convolve_full(f)
    M = float(np.max(h))
    S = float(np.sum(f))
    if S < 0.01:
        return float('inf')
    return 2.0 * n * M / (S * S)

def _normalize(f):
    """Normalize f to sum to 1, clip negatives."""
    f = np.maximum(f, 0.0)
    s = float(np.sum(f))
    if s < 1e-30:
        return None
    return f / s

def _mirror_descent(f_init, max_iters, time_limit, eta_scale=0.8, threshold_frac=0.75):
    """
    Mirror descent with multiplicative updates.
    Minimizes max(conv(f,f)) subject to sum(f)=1, f>=0.
    Uses adaptive step size and active set subgradient.
    """
    f = np.asarray(f_init, dtype=np.float64).copy()
    f = _normalize(f)
    if f is None:
        return f_init.copy(), float('inf')
    n = len(f)
    best_f = f.copy()
    best_score = _score_raw(f)
    start = time.monotonic()
    
    for it in range(max_iters):
        if time.monotonic() - start > time_limit:
            break
        
        h = _fft_convolve_full(f)
        M = float(np.max(h))
        if M <= 1e-15:
            break
        
        # Active set
        threshold = threshold_frac * M
        active = np.where(h >= threshold)[0]
        if len(active) == 0:
            active = np.array([int(np.argmax(h))])
        n_active = len(active)
        
        # Subgradient
        p = np.zeros(n, dtype=np.float64)
        for k in active:
            i_lo = max(0, k - n + 1)
            i_hi = min(n - 1, k)
            if i_hi >= i_lo:
                # p[i] += 2 * f[k-i] for i in [i_lo, i_hi]
                # k-i ranges from k-i_lo to k-i_hi
                # We need f[k-i] for i in [i_lo, i_hi]
                # = f[k-i_hi], f[k-i_hi+1], ..., f[k-i_lo]
                # = f[i_lo] to f[i_hi] reversed... 
                # Actually: when i = i_lo, k-i = k-i_lo. When i = i_hi, k-i = k-i_hi.
                # So f[k-i] for i in [i_lo, i_hi] = f[k-i_lo], f[k-i_lo-1], ..., f[k-i_hi]
                # The indices k-i_lo, k-i_lo-1, ..., k-i_hi are decreasing.
                # Since i_lo = max(0, k-n+1) and i_hi = min(n-1, k):
                #   k - i_lo = k - max(0, k-n+1) = min(k, n-1)
                #   k - i_hi = k - min(n-1, k) = max(0, k-n+1)
                # So f[k-i] for i in [i_lo, i_hi] = f[min(k,n-1)], f[min(k,n-1)-1], ..., f[max(0,k-n+1)]
                # = f[max(0,k-n+1):min(k,n-1)+1][::-1]
                lo = max(0, k - n + 1)
                hi = min(n - 1, k)
                p[i_lo:i_hi+1] += 2.0 * f[lo:hi+1][::-1]
        p /= n_active
        
        # Step size: adaptive, decreasing
        eta = eta_scale / (M * (1.0 + 0.01 * it))
        # Clip to prevent overflow
        log_ratio = -eta * p
        log_ratio = np.clip(log_ratio, -25.0, 25.0)
        
        f = f * np.exp(log_ratio)
        f_new = _normalize(f)
        if f_new is None:
            f = np.ones(n) / n
            M = 1.0 / n
        else:
            f = f_new
        
        score = _score_raw(f)
        if score < best_score:
            best_score = score
            best_f = f.copy()
    
    return best_f, best_score

def _lp_frank_wolfe(f_init, max_iters, time_limit):
    """
    LP-based Frank-Wolfe refinement.
    At each iteration, solve LP: max sum(g) s.t. (f*g)_k <= M, g >= 0.
    Then move toward g.
    """
    from scipy.optimize import linprog
    from scipy.sparse import lil_matrix
    
    f = np.asarray(f_init, dtype=np.float64).copy()
    f = _normalize(f)
    if f is None:
        return f_init.copy(), float('inf')
    n = len(f)
    best_f = f.copy()
    best_score = _score_raw(f)
    start = time.monotonic()
    
    for it in range(max_iters):
        if time.monotonic() - start > time_limit:
            break
        
        h = np.convolve(f, f)
        M = float(np.max(h))
        
        # Build constraint matrix: A[k, j] = f[k-j] if 0 <= k-j < n
        # Shape: (2n-1) x n
        A = lil_matrix((2 * n - 1, n), dtype=np.float64)
        for j in range(n):
            # A[k, j] = f[k-j] for k in [j, j+n-1]
            for k in range(j, j + n):
                A[k, j] = f[k - j]
        
        A_coo = A.tocoo()
        c_obj = -np.ones(n)
        b_ub = np.full(2 * n - 1, M)
        
        res = linprog(c_obj, A_ub=A_coo, b_ub=b_ub,
                      bounds=[(0, None)] * n, method='highs')
        
        if not res.success or res.x is None:
            print(f"    LP failed at iter {it}: {res.message}")
            break
        
        g = res.x
        Sg = float(np.sum(g))
        if Sg < 1e-10:
            break
        g = g / Sg
        
        # Frank-Wolfe step
        t = 2.0 / (it + 2)
        f_new = (1.0 - t) * f + t * g
        f_norm = _normalize(f_new)
        if f_norm is None:
            break
        f = f_norm
        
        score = _score_raw(f)
        if score < best_score:
            best_score = score
            best_f = f.copy()
        
        if (it + 1) % 5 == 0:
            elapsed = time.monotonic() - start
            print(f"    FW it={it+1}: score={best_score:.6f}, M={M:.8f}, t={t:.4f}, elapsed={elapsed:.1f}s")
    
    return best_f, best_score

def _cosine_init(n):
    j = np.arange(n, dtype=np.float64)
    f = np.cos(np.pi * (j + 0.5) / float(n)) ** 2
    f = np.maximum(f, 0.0)
    s = np.sum(f)
    if s > 0:
        f /= s
    return f

def _gaussian_init(n, sigma_frac=0.3):
    j = np.arange(n, dtype=np.float64) - (n - 1) / 2.0
    sigma = sigma_frac * float(n)
    f = np.exp(-j * j / (2.0 * sigma * sigma))
    s = np.sum(f)
    if s > 0:
        f /= s
    return f

def _tapered_init(n, power=2.0):
    j = np.arange(n, dtype=np.float64)
    x = np.sin(np.pi * (j + 0.5) / float(n))
    f = x ** power
    f = np.maximum(f, 0.0)
    s = np.sum(f)
    if s > 0:
        f /= s
    return f
