        if g_sum < 1e-15 or g_sum <= best_score / (2*n) + 1e-12:
            # g is not better than f (or close)
            break
        g = g / g_sum
        
        # Find optimal step
        conv_fg = np.convolve(f, g)
        conv_gg = np.convolve(g, g)
        S = np.where(conv_ff >= M * (1 - 1e-8))[0]
        
        best_t = 0.5
        best_new_score = np.inf
        for k in S[:10]:  # Limit to top 10 peaks
            C = conv_fg[k]
            D = conv_gg[k]
            denom = M - 2*C + D
            if abs(denom) < 1e-15:
                t = 1.0 if C < M else 0.0
            else:
                t = (M - C) / denom
            t = max(0.01, min(0.99, t))
            
            f_new = (1-t)*f + t*g
            f_new = np.clip(f_new, 0, None)
            if f_new.sum() < 1e-10:
                continue
            f_new = f_new / f_new.sum()
            score = compute_score_fast(f_new)
            if score < best_new_score:
                best_new_score = score
                best_t = t
        
        if best_new_score < best_score - 1e-12:
            f = (1-best_t)*f + best_t*g
            f = np.clip(f, 0, None)
            f = f / f.sum()
            best_f = f.copy()
            best_score = best_new_score
        else:
            # Try smaller step
            for t_try in [0.1, 0.05, 0.01]:
                f_new = (1-t_try)*f + t_try*g
                f_new = np.clip(f_new, 0, None)
                if f_new.sum() < 1e-10:
                    continue
                f_new = f_new / f_new.sum()
                score = compute_score_fast(f_new)
                if score < best_score:
                    f = f_new
                    best_f = f.copy()
                    best_score = score
                    break
            else:
                # No improvement, try perturbation
                break
        
        it += 1
        if it % 10 == 0:
            print(f"  LP it={it}, score={best_score:.8f}, elapsed={time.monotonic()-start:.1f}s")
    
    return best_f, best_score
