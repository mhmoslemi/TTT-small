def build_compressed_matrix(A, n, B):
    """
    Compress the matrix A (m x n) into A_comp (m x n/B) by summing columns in blocks.
    If g_j = b[j//B], then (A @ g) = A_comp @ b.
    """
    n_blocks = n // B
    A_comp = np.zeros((A.shape[0], n_blocks), dtype=np.float64)
    for q in range(n_blocks):
        start = q * B
        end = min(start + B, n)
        A_comp[:, q] = A[:, start:end].sum(axis=1)
    return A_comp

    """Solve: max sum(x) s.t. A@x <= b, x >= bounds_lo."""
    c = -np.ones(n_vars)
    res = linprog(c, A_ub=A, b_ub=b, bounds=[(bounds_lo, None)] * n_vars, method='highs')
    return res

    res = solve_lp(A, b, n)
    if res is None or res.x is None:
        return None, 0.0
    g = np.maximum(res.x, 0.0)
    return g, float(g.sum())

        return None
    return x / s


def score_normalized(x, n):
    """Score assuming sum(x)=1, x>=0. Returns 2*n*max(f*f)."""
    conv = np.convolve(x, x)
    return float(2.0 * n * np.max(conv))


def build_convolve_A(f, n):
    """Build matrix A (m x n) where (A @ g)[k] = (f * g)[k] (full convolution)."""
    m = 2 * n - 1
    A = np.zeros((m, n), dtype=np.float64)
    for j in range(n):
        A[j:j + n, j] = f
    return A


def build_compressed_A(A, n, B):
    """Compress columns of A by summing in blocks of size B. Returns (m x n/B) matrix."""
    n_blocks = n // B
    A_comp = np.zeros((A.shape[0], n_blocks), dtype=np.float64)
    for q in range(n_blocks):
        s = q * B
        e = min(s + B, n)
        A_comp[:, q] = A[:, s:e].sum(axis=1)
    return A_comp


def solve_max_sum_lp(A, b, n_vars):
    """Solve: max sum(x) s.t. A@x <= b, x >= 0. Returns x or None."""
    c = -np.ones(n_vars, dtype=np.float64)
    bounds = [(0.0, None)] * n_vars
    try:
        res = linprog(c, A_ub=A, b_ub=b, bounds=bounds, method='highs',
                      options={'presolve': True, 'dual_feasibility_tolerance': 1e-9})
    except Exception:
        return None
    if res is None or not res.success or res.x is None:
        # Try without presolve
        try:
            res = linprog(c, A_ub=A, b_ub=b, bounds=bounds, method='highs',
                          options={'presolve': False})
        except Exception:
            return None
    if res is None or res.x is None:
        return None
    x = np.asarray(res.x, dtype=np.float64)
    x = np.maximum(x, 0.0)
    return x


def lp_direction(f, n, M, A=None, B=1):
    """
    Find g maximizing sum(g) subject to (f*g)_k <= M, g >= 0.
    If B>1, g is constrained to be block-constant with block size B.
    Returns (g_full, sum_g) or (None, 0).
    """
    if A is None:
        A = build_convolve_A(f, n)
    m = 2 * n - 1
    b_vec = np.full(m, M, dtype=np.float64)

    if B > 1:
        n_blocks = n // B
        A_comp = build_compressed_A(A, n, B)
        x = solve_max_sum_lp(A_comp, b_vec, n_blocks)
        if x is None:
            return None, 0.0
        g = np.repeat(x, B)[:n]
    else:
        x = solve_max_sum_lp(A, b_vec, n)
        if x is None:
            return None, 0.0
        g = x

    s = float(g.sum())
    if s < 1e-300:
        return None, 0.0
    return g, s


def line_search_scores(f, g, n, n_trials=12):
    """Find best t in (0,1] minimizing score of (1-t)f + t*g (normalized)."""
    best_t = 0.0
    best_sc = score_normalized(f, n)
    # Geometric + linear mix of step sizes
    ts = []
    for i in range(n_trials):
        t = (i + 1) / (n_trials + 1)
        ts.append(t)
    ts.append(1.0)
    ts.append(0.01)
    ts = sorted(set(min(1.0, max(0.001, t)) for t in ts))

    for t in ts:
        f_new = (1.0 - t) * f + t * g
        f_new = np.maximum(f_new, 0.0)
        s = f_new.sum()
        if s < 1e-300:
            continue
        f_new /= s
        sc = score_normalized(f_new, n)
        if sc < best_sc - 1e-12:
            best_sc = sc
            best_t = t
    return best_t, best_sc


def gradient_polish(f, n, n_iter=500, time_budget=30.0):
    """Subgradient polishing to reduce peak autocorrelation."""
    f = normalize_sequence(f)
    if f is None:
        return None, float('inf')
    best_f = f.copy()
    best_score = score_normalized(f, n)
    lr = 2.0 / n
    stall = 0
    start_time = time.monotonic()

    for it in range(n_iter):
        if time.monotonic() - start_time > time_budget:
            break
        conv = np.convolve(f, f)
        M = float(conv.max())
        peak_pos = np.argmax(conv)
        # Get top few peaks for a weighted gradient
        top_k = min(5, len(conv))
        thresh = M * 0.95
        peak_positions = np.where(conv >= thresh)[0]
        if len(peak_positions) == 0:
            peak_positions = [peak_pos]

        # Compute subgradient: d(f*f)_k / df_j = 2*f[k-j] if 0<=k-j<n
        grad = np.zeros(n, dtype=np.float64)
        for k in peak_positions:
            lo = max(0, k - (n - 1))
            hi = min(n - 1, k)
            # grad[j] += 2 * f[k-j] for j in [lo, hi]
            # f[k-hi : k-lo+1] reversed
            seg = f[k - hi:k - lo + 1][::-1]
            grad[lo:hi + 1] += 2.0 * seg

        # Normalize gradient
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-30:
            break
        grad /= gnorm

        f_new = f - lr * grad
        f_new = np.maximum(f_new, 0.0)
        s = f_new.sum()
        if s < 1e-300:
            lr *= 0.5
            continue
        f_new /= s
        sc = score_normalized(f_new, n)

        if sc < best_score - 1e-10:
            best_score = sc
            best_f = f_new.copy()
            f = f_new
            stall = 0
            lr = min(lr * 1.05, 5.0 / n)
        else:
            stall += 1
            lr *= 0.85
            if stall > 30:
                break
            # Try a smaller step
            f_try = f - lr * grad
            f_try = np.maximum(f_try, 0.0)
            s2 = f_try.sum()
            if s2 < 1e-300:
                break
            f_try /= s2
            sc2 = score_normalized(f_try, n)
            if sc2 < best_score:
                best_score = sc2
                best_f = f_try.copy()
                f = f_try
                stall = 0

    return best_f, best_score


def active_set_equalize(f, n, n_iter=200, time_budget=30.0):
    """
    Try to equalize the top autocorrelation peaks using a Newton-like approach.
    Identifies active peaks and adjusts f to make them equal.
    """
    f = normalize_sequence(f)
    if f is None:
        return None, float('inf')
    best_f = f.copy()
    best_score = score_normalized(f, n)
    start_time = time.monotonic()
    n_peaks = 3

    for it in range(n_iter):
        if time.monotonic() - start_time > time_budget:
            break
        conv = np.convolve(f, f)
        M = float(conv.max())
        # Find peak and near-peak positions
        sorted_idx = np.argsort(conv)[::-1]
        active = sorted_idx[:n_peaks]
        active_vals = conv[active]

        if float(active_vals[-1]) < 0.995 * M:
            # Only one dominant peak; use gradient method instead
            break

        # We want to reduce the max and increase the min of active peaks
        # Direction: decrease f where it contributes to the max peak,
        #            increase f where it contributes to the min peak
        k_max = active[0]
        k_min = active[-1]

        # Gradient components
        grad_max = np.zeros(n)
        grad_min = np.zeros(n)
        for k in [k_max]:
            lo = max(0, k - (n - 1))
            hi = min(n - 1, k)
            grad_max[lo:hi + 1] = 2.0 * f[k - hi:k - lo + 1][::-1]
        for k in [k_min]:
            lo = max(0, k - (n - 1))
            hi = min(n - 1, k)
            grad_min[lo:hi + 1] = 2.0 * f[k - hi:k - lo + 1][::-1]

        # Direction: reduce max peak, increase min peak
        direction = -grad_max + 0.5 * grad_min
        direction = np.maximum(direction, -f * 10)  # Don't decrease f too much

        # Try a few step sizes
        for t in [0.01, 0.005, 0.001, 0.02, 0.05]:
            f_try = f + t * direction
            f_try = np.maximum(f_try, 0.0)
            s = f_try.sum()
            if s < 1e-300:
                continue
            f_try /= s
            sc = score_normalized(f_try, n)
            if sc < best_score - 1e-10:
                best_score = sc
                best_f = f_try.copy()
                f = f_try
                break

    return best_f, best_score


def upsample_sequence(f, new_n):
    """Upsample a sequence to a new length using interpolation."""
    f = normalize_sequence(f)
    if f is None:
        return None
    old_n = f.size
    if old_n == new_n:
        return f
    old_t = np.linspace(0.0, 1.0, old_n)
    new_t = np.linspace(0.0, 1.0, new_n)
    out = np.interp(new_t, old_t, f)
    return normalize_sequence(out)


def downsample_sequence(f, new_n):
    """Downsample a sequence by averaging blocks."""
    f = normalize_sequence(f)
    if f is None:
        return None
    old_n = f.size
    if old_n == new_n:
        return f
    if old_n < new_n:
        return upsample_sequence(f, new_n)
    blocks = old_n // new_n
    out = np.zeros(new_n)
    for i in range(new_n):
        s = i * blocks
        e = min(s + blocks, old_n)
        out[i] = f[s:e].sum()
    return normalize_sequence(out)


def make_starts(n, budget, rng, best_f=None):
    """Generate diverse starting sequences."""
    starts = []

    # Given best (upsampled)
    if best_f is not None:
        up = upsample_sequence(best_f, n)
        if up is not None:
            starts.append(up)
            # Noisy version
            noise = rng.dirichlet(np.ones(n))
            for eps in [0.1, 0.3]:
                mixed = (1 - eps) * up + eps * noise
                starts.append(normalize_sequence(mixed))

    # Raised cosine
    t = np.arange(n) / (n - 1)
    rc = np.cos(np.pi * t) ** 2
    rc[0] = 0.0  # Avoid zero at edges for better performance
    rc[-1] = 0.0
    starts.append(normalize_sequence(rc))

    # Uniform with noise
    u = rng.dirichlet(np.ones(n) * 0.1)
    starts.append(normalize_sequence(u))

    # Triangular
    tri = 1.0 - np.abs(2.0 * t - 1.0)
    starts.append(normalize_sequence(tri))

    # Top-hat with smooth edges
    th = np.ones(n)
    edge = max(1, n // 20)
    for i in range(edge):
        frac = (i + 1) / (edge + 1)
        th[i] = frac
        th[n - 1 - i] = frac
    starts.append(normalize_sequence(th))

    # Random Dirichlet with concentration
    d = rng.dirichlet(np.ones(n) * 0.5)
    starts.append(normalize_sequence(d))

    # Filter and deduplicate
    valid = []
    for s in starts:
        if s is not None and s.size == n:
            valid.append(s)

    # Deduplicate
    unique = []
    for s in valid:
        is_dup = False
        for u in unique:
            if np.linalg.norm(s - u) < 0.05:
                is_dup = True
                break
        if not is_dup:
            unique.append(s)

    return unique[:6]


def run_lp_optimization(f_init, n, time_budget, B=1, verbose=True):
    """
    Run LP-based optimization from f_init for the given time budget.
    B=1 for full LP, B>1 for compressed LP.
    Returns (best_f, best_score).
    """
    f = normalize_sequence(f_init)
    if f is None:
        f = np.ones(n) / n
    best_f = f.copy()
    best_score = score_normalized(f, n)
    start_time = time.monotonic()
    stall = 0
    iteration = 0
    A = None  # Lazily built

    while time.monotonic() - start_time < time_budget:
        t0 = time.monotonic()
        iteration += 1

        conv = np.convolve(f, f)
        M = float(conv.max())
        if not np.isfinite(M) or M <= 0:
            break

        # Build or rebuild A (only needed when f changes significantly)
        if A is None:
            A = build_convolve_A(f, n)

        # Solve LP direction
        g, s_g = lp_direction(f, n, M, A=A, B=B)
        if g is None or s_g <= 1.0 + 1e-8:
            stall += 1
            if stall > 15:
                break
            # Perturb and try again
            noise = np.random.dirichlet(np.ones(n) * 0.5)
            f = (1 - 0.3) * f + 0.3 * noise
            f = normalize_sequence(f)
            A = None
            continue

        # Rescale g to sum=1
        g = g / s_g

        # Check improvement potential
        r = np.convolve(f, g)
        r_max = float(np.max(r))
        if r_max >= M * (1 - 1e-8):
            stall += 1
            if stall > 15:
                break
            noise = np.random.dirichlet(np.ones(n) * 0.5)
            f = (1 - 0.3) * f + 0.3 * noise
            f = normalize_sequence(f)
            A = None
            continue

        # Line search
        best_t, best_sc = line_search_scores(f, g, n)
        if best_t > 0 and best_sc < best_score - 1e-10:
            f_new = (1.0 - best_t) * f + best_t * g
            f_new = np.maximum(f_new, 0.0)
            s_sum = f_new.sum()
            f_new /= s_sum
            f = f_new
            best_f = f_new.copy()
            best_score = best_sc
            stall = 0
            A = None  # Rebuild A since f changed
            if verbose:
                dt = time.monotonic() - t0
                print(f"    n={n} B={B} it={iteration} score={best_score:.6f} "
                      f"M={M:.6g} t={best_t:.4f} [{dt:.2f}s]", flush=True)
        else:
            stall += 1
            if stall > 15:
                break
            # Try a different perturbation
            noise = np.random.dirichlet(np.ones(n) * 0.3)
            f = (1 - 0.2) * f + 0.2 * noise
            f = normalize_sequence(f)
            A = None

        # Periodic local refinement
        if iteration % 20 == 0 and time.monotonic() - start_time < time_budget * 0.9:
            f_pol, sc_pol = gradient_polish(f, n, n_iter=50, time_budget=5.0)
            if f_pol is not None and sc_pol < best_score:
                best_score = sc_pol
                best_f = f_pol.copy()
                f = f_pol.copy()
                A = None
                if verbose:
                    print(f"    n={n} B={B} it={iteration} polish -> {best_score:.6f}", flush=True)

    return best_f, best_score


def run_gradient_optimization(f_init, n, time_budget):
    """Run gradient-based optimization."""
    f = normalize_sequence(f_init)
    if f is None:
        return None, float('inf')
    best_f = f.copy()
    best_score = score_normalized(f, n)
    start_time = time.monotonic()
    lr = 1.0 / np.sqrt(n)
    stall = 0

    while time.monotonic() - start_time < time_budget:
        conv = np.convolve(f, f)
        M = float(conv.max())
        thresh = M * 0.90
        peaks = np.where(conv >= thresh)[0]
        if len(peaks) == 0:
            peaks = [np.argmax(conv)]

        grad = np.zeros(n)
        for k in peaks:
            lo = max(0, k - (n - 1))
            hi = min(n - 1, k)
            grad[lo:hi + 1] += 2.0 * f[k - hi:k - lo + 1][::-1]

        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-30:
            break
        grad /= gnorm

        # Try multiple step sizes
        improved = False
        for t in [lr, lr * 0.5, lr * 0.1, lr * 2.0]:
            f_try = f - t * grad
            f_try = np.maximum(f_try, 0.0)
            s = f_try.sum()
            if s < 1e-300:
                continue
            f_try /= s
            sc = score_normalized(f_try, n)
            if sc < best_score - 1e-12:
                best_score = sc
                best_f = f_try.copy()
                f = f_try
                improved = True
                break

        if improved:
            stall = 0
            lr = min(lr * 1.02, 5.0 / n)
        else:
            stall += 1
            lr *= 0.9
            if stall > 50:
                # Random restart
                noise = np.random.dirichlet(np.ones(n) * 0.5)
                f = 0.7 * best_f + 0.3 * noise
                f = normalize_sequence(f)
                stall = 0
                lr = 1.0 / np.sqrt(n)

    return best_f, best_score


def propose_candidate(seed=42, budget_s=1500, **kwargs):
    """
    Search for a low-scoring non-negative sequence minimizing
    2*n*max(convolve(f,f))/(sum(f))^2.
    Uses multi-phase approach: compressed LP -> full LP -> gradient polish.
    """
    rng = np.random.default_rng(int(seed) if seed is not None else 42)
    total_budget = max(1.0, float(budget_s))
    start_time = time.monotonic()
    deadline = start_time + total_budget - 5.0  # Reserve 5s for cleanup

    best_f = None
    best_score = float('inf')
    best_n = 0

    # Phase 0: Load given sequence
    given_f = None
    if total_budget >= 10.0:
        try:
            raw = np.asarray(height_sequence_1, dtype=np.float64)
            given_f = normalize_sequence(raw)
            if given_f is not None:
                best_f = given_f.copy()
                best_n = given_f.size
                best_score = score_normalized(given_f, given_f.size)
                print(f"Given sequence: n={best_n}, score={best_score:.6f}", flush=True)
        except NameError:
            pass
        except Exception as e:
            print(f"Failed to load given sequence: {e}", flush=True)

    if best_f is None:
        # Default start
        n0 = 256
        best_f = np.ones(n0) / n0
        best_n = n0
        best_score = 2.0
        print(f"Using default uniform start: n={n0}, score={best_score:.6f}", flush=True)

    def remaining():
        return deadline - time.monotonic()

    # Phase 1: Quick LP at the given length (or 512)
    n1 = best_n if best_n >= 128 else 512
    if n1 > 2048:
        n1 = 2048
    alloc1 = min(80.0, remaining() * 0.1)
    if alloc1 > 10.0:
        print(f"\n=== Phase 1: LP at n={n1}, budget={alloc1:.0f}s ===", flush=True)
        f1 = upsample_sequence(best_f, n1) if best_n != n1 else best_f.copy()
        f_res, s_res = run_lp_optimization(f1, n1, alloc1 * 0.6, B=1, verbose=True)
        if s_res < best_score:
            best_score = s_res
            best_f = f_res.copy()
            best_n = n1
            print(f"  Phase 1 result: n={n1}, score={best_score:.6f}", flush=True)

    # Phase 2: Compressed LP at same or larger n
    n2 = max(n1, 1024)
    if n2 > 2048:
        n2 = 2048
    alloc2 = min(150.0, remaining() * 0.15)
    if alloc2 > 20.0:
        print(f"\n=== Phase 2: Compressed LP at n={n2}, budget={alloc2:.0f}s ===", flush=True)
        f2 = upsample_sequence(best_f, n2) if best_n != n2 else best_f.copy()

        # Try different block sizes
        for B in [4, 8]:
            if remaining() < 10.0:
                break
            tb = alloc2 / 2.0
            tb = min(tb, remaining() - 5.0)
            if tb < 5.0:
                break
            print(f"  Block size B={B}, budget={tb:.0f}s", flush=True)
            f_res, s_res = run_lp_optimization(f2, n2, tb, B=B, verbose=False)
            if s_res < best_score:
                best_score = s_res
                best_f = f_res.copy()
                best_n = n2
                print(f"  Compressed LP (B={B}): n={n2}, score={best_score:.6f}", flush=True)

    # Phase 3: Full LP refinement
    n3 = best_n
    alloc3 = min(300.0, remaining() * 0.3)
    if alloc3 > 30.0:
        print(f"\n=== Phase 3: Full LP at n={n3}, budget={alloc3:.0f}s ===", flush=True)
        f3 = best_f.copy()
        f_res, s_res = run_lp_optimization(f3, n3, alloc3, B=1, verbose=True)
        if s_res < best_score:
            best_score = s_res
            best_f = f_res.copy()
            best_n = n3
            print(f"  Phase 3 result: n={n3}, score={best_score:.6f}", flush=True)

    # Phase 4: Gradient polishing
    alloc4 = min(150.0, remaining() * 0.15)
    if alloc4 > 10.0:
        print(f"\n=== Phase 4: Gradient polish at n={best_n}, budget={alloc4:.0f}s ===", flush=True)
        f_res, s_res = run_gradient_optimization(best_f, best_n, alloc4)
        if f_res is not None and s_res < best_score:
            best_score = s_res
            best_f = f_res.copy()
            print(f"  Gradient result: n={best_n}, score={best_score:.6f}", flush=True)

        # Active set equalization
        if remaining() > 15.0:
            f_res, s_res = active_set_equalize(best_f, best_n, n_iter=200,
                                                time_budget=min(30.0, remaining() - 5.0))
            if f_res is not None and s_res < best_score:
                best_score = s_res
                best_f = f_res.copy()
                print(f"  Active set result: n={best_n}, score={best_score:.6f}", flush=True)

    # Phase 5: Try larger n if time permits
    if remaining() > 120.0 and best_n <= 1024:
        n5 = best_n * 2
        if n5 <= 4096:
            print(f"\n=== Phase 5: Explore n={n5} ===", flush=True)
            f5 = upsample_sequence(best_f, n5)
            # Quick compressed LP
            alloc5 = min(100.0, remaining() * 0.3)
            f_res, s_res = run_lp_optimization(f5, n5, alloc5 * 0.5, B=8, verbose=False)
            if s_res < best_score:
                best_score = s_res
                best_f = f_res.copy()
                best_n = n5
                print(f"  Larger n compressed: n={n5}, score={best_score:.6f}", flush=True)
                # Follow up with full LP if improved
                if remaining() > 30.0:
                    f_res, s_res = run_lp_optimization(best_f, n5, min(60.0, remaining() - 10.0), B=1, verbose=False)
                    if s_res < best_score:
                        best_score = s_res
                        best_f = f_res.copy()
                        print(f"  Larger n full LP: n={n5}, score={best_score:.6f}", flush=True)

    # Phase 6: Final polishing
    if remaining() > 15.0:
        print(f"\n=== Phase 6: Final polish ===", flush=True)
        f_res, s_res = gradient_polish(best_f, best_n, n_iter=300,
                                        time_budget=min(30.0, remaining() - 5.0))
        if f_res is not None and s_res < best_score:
            best_score = s_res
            best_f = f_res.copy()
            print(f"  Final polish: n={best_n}, score={best_score:.6f}", flush=True)

    # Convert to list for evaluation
    seq = [float(x) for x in best_f]

    # Verify with actual evaluator
    try:
        final_score = float(evaluate_sequence(seq))
        print(f"\nFinal: n={len(seq)}, local_score={best_score:.6f}, "
              f"evaluate_sequence={final_score:.6f}", flush=True)
    except Exception as e:
        print(f"Error in final evaluation: {e}", flush=True)
        final_score = float('inf')

    # Fallback
    if not np.isfinite(final_score):
        try:
            if given_f is not None:
                seq = [float(x) for x in given_f]
                final_score = float(evaluate_sequence(seq))
        except Exception:
            pass
    if not np.isfinite(final_score):
        seq = [1.0 / 128.0] * 128

    elapsed = time.monotonic() - start_time
    print(f"Total time: {elapsed:.1f}s", flush=True)
    return seq
