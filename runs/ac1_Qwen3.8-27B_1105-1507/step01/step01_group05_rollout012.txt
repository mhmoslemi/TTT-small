"""
Hybrid LP + subgradient optimization for minimizing 2*n*max(convolve(a,a))/sum(a)^2.
Uses FFT-accelerated score computation, LP direction finding with golden-section
line search, subgradient warm-up, momentum, and diverse restarts.

import numpy as np
from scipy.optimize import linprog
import time


def fft_convolve(a):
    """Compute full convolution using FFT."""
    n = len(a)
    size = 2 * n
    A = np.fft.rfft(a, size=size)
    return np.fft.irfft(A * A, size=size)[:2 * n - 1]


def compute_score_normalized(a):
    """Compute 2*n*max(conv) for a normalized sequence (sum=1)."""
    n = len(a)
    conv = fft_convolve(a)
    return 2.0 * n * conv.max()


def build_conv_matrix(a):
    """Build matrix A such that A @ b = np.convolve(a, b, mode='full')."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for k in range(2 * n - 1):
        j_start = max(0, k - n + 1)
        j_end = min(n, k + 1)
        for j in range(j_start, j_end):
            A[k, j] = a[k - j]
    return A


def project_onto_simplex(v):
    """Project v onto {x >= 0, sum(x) = 1} using bisection on threshold."""
    n = len(v)
    if n == 0:
        return v
    lo = 0.0
    hi = float(v.max())
    for _ in range(80):
        mid = (lo + hi) * 0.5
        s = np.maximum(v - mid, 0.0).sum()
        if s > 1.0:
            lo = mid
        else:
            hi = mid
    tau = (lo + hi) * 0.5
    w = np.maximum(v - tau, 0.0)
    s = w.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return w / s


def subgradient_step(a, step_size):
    """One subgradient step to reduce max autocorrelation peak."""
    n = len(a)
    r = fft_convolve(a)
    M = r.max()
    argmax = np.where(np.isclose(r, M, atol=1e-12))[0]

    grad = np.zeros(n)
    for k in argmax:
        # d(conv)_k / d(a_j) = a[k-j] + a[j-k] (with bounds)
        j1s = max(0, k - n + 1)
        j1e = min(n, k + 1)
        if j1e > j1s:
            idx1 = np.arange(j1s, j1e)
            grad[j1s:j1e] += a[k - idx1]
        j2s = max(0, k)
        j2e = min(n, k + n)
        if j2e > j2s:
            idx2 = np.arange(j2s, j2e)
            grad[j2s:j2e] += a[idx2 - k]

    grad /= len(argmax)
    a_new = a - step_size * grad
    return project_onto_simplex(a_new)


def subgradient_optimize(a, n_iter, step_size_init, rng, verbose=False):
    """Fast subgradient optimization with adaptive step size."""
    n = len(a)
    a = a.astype(np.float64).copy()
    s = a.sum()
    if s < 1e-10:
        return a, np.inf
    a = a / s

    best_a = a.copy()
    best_score = compute_score_normalized(a)
    step_size = step_size_init
    patience = 0

    for it in range(n_iter):
        a = subgradient_step(a, step_size)
        score = compute_score_normalized(a)
        if score < best_score - 1e-12:
            best_a = a.copy()
            best_score = score
            patience = 0
            step_size *= 0.9995
        else:
            patience += 1
            step_size *= 0.995
            if patience > 200:
                a = best_a.copy()
                pert = rng.normal(0, max(0.01, step_size * 0.3), n)
                a = np.clip(a + pert, 0, None)
                sa = a.sum()
                if sa > 1e-10:
                    a = a / sa
                patience = 0

        if verbose and it % 1000 == 0:
            print(f"    SG it={it}, score={best_score:.6f}, step={step_size:.4f}")

    return best_a, best_score


def golden_section_min(func, lo, hi, max_iter=40):
    """Golden section minimization on [lo, hi]."""
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    c = hi - gr * (hi - lo)
    d = lo + gr * (hi - lo)
    fc = func(c)
    fd = func(d)
    for _ in range(max_iter):
        if fc <= fd:
            hi = d
            d = c
            fd = fc
            c = hi - gr * (hi - lo)
            fc = func(c)
        else:
            lo = c
            c = d
            fc = fd
            d = lo + gr * (hi - lo)
            fd = func(d)
        if hi - lo < 1e-8:
            break
    t_opt = (lo + hi) * 0.5
    return t_opt, func(t_opt)


def optimize_lp(a, time_budget, rng, verbose=True, momentum_window=3):
    """LP-based optimization with golden-section line search and momentum."""
    n = len(a)
    a = a.astype(np.float64).copy()
    s = a.sum()
    if s < 1e-10:
        return a, np.inf
    a = a / s

    best_a = a.copy()
    best_score = compute_score_normalized(a)

    start = time.monotonic()
    it = 0
    stall = 0
    dir_momentum = np.zeros(n)

    while time.monotonic() - start < time_budget:
        r = fft_convolve(a)
        M = r.max()

        A = build_conv_matrix(a)
        c_obj = -np.ones(n)
        b_ub = np.full(2 * n - 1, M)
        bounds = [(0, None)] * n

        try:
            res = linprog(c_obj, A_ub=A, b_ub=b_ub, bounds=bounds,
                         method='highs', options={'presolve': True})
        except Exception:
            break

        if not res.success or res.x is None:
            break

        b = np.clip(res.x, 0, None)
        b_sum = b.sum()
        if b_sum < 1e-15:
            break

        g = b / b_sum

        # Apply momentum
        dir_momentum = 0.7 * dir_momentum + 0.3 * g
        dir_momentum = np.clip(dir_momentum, 0, None)
        dm_sum = dir_momentum.sum()
        if dm_sum < 1e-10:
            dir_momentum = g
        else:
            dir_momentum = dir_momentum / dm_sum

        # Use both the raw LP direction and momentum direction
        for direction in [g, dir_momentum]:
            if time.monotonic() - start > time_budget:
                break

            def score_at_t(t):
                a_t = (1.0 - t) * a + t * direction
                a_t = np.clip(a_t, 0, None)
                s_t = a_t.sum()
                if s_t < 1e-10:
                    return np.inf
                a_t = a_t / s_t
                return compute_score_normalized(a_t)

            # Coarse grid to bracket
            grid = [0.01, 0.02, 0.05, 0.1, 0.2, 0.3, 0.5, 0.7, 1.0]
            best_grid_val = np.inf
            best_grid_t = 0.01
            for gt in grid:
                gv = score_at_t(gt)
                if gv < best_grid_val:
                    best_grid_val = gv
                    best_grid_t = gt

            if best_grid_val >= best_score - 1e-12:
                continue

            # Golden section around the best grid point
            lo_t = max(0.0, best_grid_t - 0.1)
            hi_t = min(1.0, best_grid_t + 0.1)
            t_opt, f_opt = golden_section_min(score_at_t, lo_t, hi_t, max_iter=25)

            if f_opt < best_score - 1e-12:
                a_new = (1.0 - t_opt) * a + t_opt * direction
                a_new = np.clip(a_new, 0, None)
                s_new = a_new.sum()

                    a_new = a_new / s_new
                    best_a = a_new.copy()
                    best_score = f_opt
                    a = a_new
                    stall = 0
                    if verbose and it % 3 == 0:
                        elapsed = time.monotonic() - start
                        print(f"    LP it={it}, score={best_score:.6f}, t={t_opt:.4f}, el={elapsed:.0f}s")

        if stall > 0 or True:
            current_score = compute_score_normalized(a)
            if current_score >= best_score - 1e-12:
                stall += 1
            else:
                stall = 0

        if stall > 40:
            # Aggressive random perturbation
            strength = 0.05 + 0.05 * min(stall / 100.0, 1.0)
            a_pert = a * (1.0 + strength * rng.normal(0, 1, n))
            a_pert = np.clip(a_pert, 0, None)
            sp = a_pert.sum()
            if sp > 1e-10:
                a_pert = a_pert / sp
                sp_score = compute_score_normalized(a_pert)
                if sp_score < best_score:
                    a = a_pert
                    best_a = a_pert.copy()
                    best_score = sp_score
                    stall = 0
                    dir_momentum *= 0.5
                    if verbose:
                        print(f"    Perturbation at it={it}: {best_score:.6f}")

            if stall > 100:
                if verbose:
                    print(f"    Stuck at it={it}, score={best_score:.6f}. Stopping.")
                break

        it += 1

    return best_a, best_score


def optimize_lp_dual(a, time_budget, rng, verbose=True):
    """LP with dual formulation (minimize M*sum(lambda) s.t. A^T@lambda >= 1, lambda >= 0)."""
    n = len(a)
    a = a.astype(np.float64).copy()
    s = a.sum()
    if s < 1e-10:
        return a, np.inf
    a = a / s

    best_a = a.copy()
    best_score = compute_score_normalized(a)
    start = time.monotonic()

    stall = 0

        r = fft_convolve(a)
        M = r.max()

        # Dual: min M * sum(lambda) s.t. A^T @ lambda >= 1, lambda >= 0
        # A^T @ lambda >= 1 means for each j: sum_k A[k,j] * lambda[k] >= 1
        A = build_conv_matrix(a)
        AT = A.T  # shape (n, 2n-1)
        c_dual = np.full(2 * n - 1, M)

            res = linprog(c_dual, A_ub=A_ub_dual, b_ub=b_ub_dual, bounds=bounds_dual,
                         method='highs', options={'presolve': True})

            break

        if not res.success or res.x is None:
            break

    # Fallback: just use primal LP
    return optimize_lp(a, time_budget, rng, verbose=verbose)


def propose_candidate(seed=42, budget_s=1500, **kwargs):
    """Main search function combining multiple optimization strategies."""
    rng = np.random.default_rng(seed)
    total_start = time.monotonic()
    deadline = total_start + budget_s - 10.0

    best_overall = None
    best_score_overall = np.inf

    # Get parent sequence
    parent = None
    try:
        parent = np.array(height_sequence_1, dtype=np.float64)
        if parent.ndim != 1 or len(parent) == 0 or not np.isfinite(parent).all():
            parent = None
    except Exception:
        parent = None

    if parent is not None:
        parent_norm = parent / parent.sum()
        parent_score = compute_score_normalized(parent_norm)
        print(f"Parent: length={len(parent)}, score={parent_score:.6f}")
        if parent_score < best_score_overall:
            best_score_overall = parent_score
            best_overall = parent_norm.copy()
    else:
        print("No parent available, starting from scratch")

    n_main = 256
    if parent is not None and 128 <= len(parent) <= 1024:
        n_main = len(parent)

    # Phase 1: Subgradient warm-up (fast, ~100s)
    if best_overall is not None:
        print(f"\n=== Phase 1: Subgradient warm-up (n={n_main}) ===")
        t0 = time.monotonic()
        sg_time = min(120, (deadline - time.monotonic()) * 0.1)
        a_sg, score_sg = subgradient_optimize(best_overall, n_iter=50000,
                                               step_size_init=0.8, rng=rng, verbose=True)
        if score_sg < best_score_overall:
            best_score_overall = score_sg
            best_overall = a_sg.copy()
            print(f"  Phase 1 result: {best_score_overall:.6f}")
        else:
            print(f"  Phase 1: no improvement ({score_sg:.6f} vs {best_score_overall:.6f})")
        print(f"  Time: {time.monotonic()-t0:.1f}s")

    # Phase 2: LP optimization with line search (main phase, ~600s)
    print(f"\n=== Phase 2: LP optimization (n={n_main}) ===")
    t0 = time.monotonic()
    lp_time = min(600, (deadline - time.monotonic()) * 0.45)
    a_lp, score_lp = optimize_lp(best_overall, lp_time, rng, verbose=True)
    if score_lp < best_score_overall:
        best_score_overall = score_lp
        best_overall = a_lp.copy()
        print(f"  Phase 2 result: {best_score_overall:.6f}")
    else:
        print(f"  Phase 2: no improvement ({score_lp:.6f} vs {best_score_overall:.6f})")
    print(f"  Time: {time.monotonic()-t0:.1f}s")

    # Phase 3: Diverse restarts with different initializations
    print(f"\n=== Phase 3: Diverse starts ===")
    remaining = deadline - time.monotonic()
    if remaining > 100:
        generators = [
            ("gaussian_noise", lambda n, r: np.abs(r.normal(0, 1, n)) + 0.1),
            ("exp_decay", lambda n, r: np.exp(-np.arange(n, dtype=float) / (n * 0.25)) + 0.1),
            ("bimodal", lambda n, r: np.concatenate([np.ones(n // 2) * 2.0, np.ones(n - n // 2) * 0.5]) + 0.05 * r.normal(0, 1, n)),
            ("low_fric", lambda n, r: 1.0 + 0.4 * np.sin(2 * np.pi * 7 * np.arange(n) / n) + 0.1 * r.normal(0, 1, n)),
            ("high_fric", lambda n, r: 1.0 + 0.3 * np.cos(2 * np.pi * 23 * np.arange(n) / n) + 0.1 * r.normal(0, 1, n)),
            ("random_walk", lambda n, r: np.maximum(np.cumsum(r.normal(0, 1, n)) + n * 0.5, 0.1)),
        ]

        n_configs = len(generators)
        time_per = max(20.0, (remaining * 0.6) / n_configs)
        print(f"  {n_configs} configs, ~{time_per:.0f}s each")

        for name, gen_func in generators:
            if time.monotonic() > deadline - 30:
                break

            f0 = gen_func(n_main, rng)
            f0 = np.clip(f0, 0, None)
            s0 = f0.sum()
            if s0 < 1e-10:
                continue
            f0 = f0 / s0

            t0 = time.monotonic()
            t_budget = min(time_per, deadline - time.monotonic() - 20)
            if t_budget < 15:
                break

            # Quick subgradient
            a_sg2, score_sg2 = subgradient_optimize(f0, n_iter=3000, step_size_init=0.5, rng=rng)
            # LP refinement
            a_opt, score_opt = optimize_lp(a_sg2, t_budget * 0.7, rng, verbose=False)

            if score_opt < best_score_overall:
                best_score_overall = score_opt
                best_overall = a_opt.copy()
                print(f"  [{name}] ** NEW BEST: {best_score_overall:.6f} **")
            else:
                print(f"  [{name}]: {score_opt:.6f}")

    # Phase 4: Final LP polish from best found
    remaining = deadline - time.monotonic()
    if best_overall is not None and remaining > 60:
        print(f"\n=== Phase 4: Final LP polish ({remaining:.0f}s) ===")
        t0 = time.monotonic()
        polish_time = remaining - 10
        a_pol, score_pol = optimize_lp(best_overall, polish_time, rng, verbose=True)
        if score_pol < best_score_overall:
            best_score_overall = score_pol
            best_overall = a_pol.copy()
            print(f"  Phase 4 result: {best_score_overall:.6f}")
        print(f"  Time: {time.monotonic()-t0:.1f}s")

    # Phase 5: Try slightly different length if we have time
    remaining = deadline - time.monotonic()
    if best_overall is not None and remaining > 120:
        print(f"\n=== Phase 5: Try n=512 ===")
        n512 = 512
        f512 = np.ones(n512) * (1 + 0.05 * rng.normal(0, 1, n512))
        f512 = np.clip(f512, 0, None)
        f512 = f512 / f512.sum()
        t_budget = remaining - 15
        a_512, score_512 = optimize_lp(f512, t_budget, rng, verbose=True)
        if score_512 < best_score_overall:
            best_score_overall = score_512
            best_overall = a_512.copy()
            print(f"  n=512 NEW BEST: {best_score_overall:.6f}")
        else:
            print(f"  n=512: {score_512:.6f} (not better)")

    # Final verification
    if best_overall is not None:
        best_overall = np.clip(best_overall, 0, 1000)
        final_list = best_overall.tolist()
        final_score = evaluate_sequence(final_list)
        print(f"\n{'='*50}")
        print(f"FINAL SCORE: {final_score:.6f}")
        print(f"Sequence length: {len(final_list)}")
        print(f"Target: 1.5027")
        print(f"{'='*50}")
        return final_list

    # Fallback
    print("FALLBACK: uniform sequence")
    return [1.0] * 256
