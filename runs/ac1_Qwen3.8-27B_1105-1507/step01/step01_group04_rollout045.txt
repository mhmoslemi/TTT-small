"""
Optimizes non-negative step function heights to minimize 2n*max(conv(f,f))/(sum(f))^2.
Uses coarse-to-fine smooth mirror descent with adaptive temperature,
FFT-based gradient computation, and LP-based refinement.
"""

import numpy as np
import time

def _fft_convolve_full(f):
    n = len(f)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    F = np.fft.rfft(f, size)
    h = np.fft.irfft(F * F, size)
    return h[:2 * n - 1]

def _score_raw(f):
    n = len(f)
    h = _fft_convolve_full(f)
    M = float(np.max(h))
    S = float(np.sum(f))
    if S < 0.01:
        return float('inf')
    return 2.0 * n * M / (S * S)

def _normalize(f):
    f = np.maximum(f, 0.0)
    s = float(np.sum(f))
    if s < 1e-30:
        return None
    return f / s

def _soft_max_value_and_grad(f, tau):
    """Compute soft max of f*f and its gradient w.r.t. f."""
    n = len(f)
    h = _fft_convolve_full(f)
    h_max = float(np.max(h))
    shifted = (h - h_max) / tau
    exp_shifted = np.exp(shifted)
    sum_exp = float(np.sum(exp_shifted))
    weights = exp_shifted / sum_exp
    value = h_max + tau * np.log(sum_exp)
    
    # Gradient: grad[j] = 2 * sum_m weights[m+j] * f[m]
    # This is 2 * cross_correlation(weights, f)[j]
    corr = np.correlate(weights, f, mode='full')
    grad = 2.0 * corr[:n]
    
    return value, grad

def _smooth_mirror_descent(f_init, max_iter, tau_init, tau_final, eta_init, eta_decay, 
                            l2_reg=0.0, time_limit=None):
    """Smooth mirror descent with adaptive temperature."""
    f = np.asarray(f_init, dtype=np.float64).copy()
    f = _normalize(f)
    if f is None:
        return f_init.copy(), float('inf')
    n = len(f)
    best_f = f.copy()
    best_score = _score_raw(f)
    start_time = time.monotonic()
    
    for it in range(max_iter):
        if time_limit is not None and time.monotonic() - start_time > time_limit:
            break
        
        # Temperature schedule (geometric decay)
        if tau_final > 1e-10:
            tau = tau_init * (tau_final / tau_init) ** (it / max(float(max_iter - 1), 1))
        else:
            tau = tau_init / (1.0 + it * 0.1)
        
        # Compute soft max and gradient
        value, grad = _soft_max_value_and_grad(f, tau)
        
        # Add L2 regularization gradient if needed
        if l2_reg > 0:
            # grad of l2_reg * sum(h^2) = l2_reg * 2 * sum_k h_k * d(h_k)/d(f_j)
            # = l2_reg * 4 * sum_k h_k * f(k-j) [0<=k-j<n]
            # = l2_reg * 4 * np.correlate(h, f, 'full')[:n]
            h = _fft_convolve_full(f)
            l2_grad = np.correlate(h, f, mode='full')[:n]
            grad = grad + l2_reg * 4.0 * l2_grad
        
        # Step size (geometric decay)
        eta = eta_init * (eta_decay ** it)
        
        # Mirror descent update
        # f_new = f * exp(-eta * grad) / Z
        log_ratio = -eta * grad
        log_ratio = np.clip(log_ratio, -30.0, 30.0)
        f_new = f * np.exp(log_ratio)
        f_new = _normalize(f_new)
        if f_new is None:

        f = f_new
        
        # Evaluate
        score = _score_raw(f)
        if score < best_score:
            best_score = score
            best_f = f.copy()
    
    return best_f, best_score

def _upsample_coarse_to_fine(f_coarse, n_fine):
    """Upsample a coarse solution to fine resolution by inserting zeros and interpolating."""
    n_coarse = len(f_coarse)
    factor = n_fine // n_coarse
    if factor < 1:
        # Downsample
        idx = np.linspace(0, n_coarse - 1, n_fine, dtype=int)
        f_fine = f_coarse[idx]
    elif factor == 1:
        f_fine = f_coarse.copy()
    else:
        # Upsample: for each coarse element, create 'factor' fine elements
        f_fine = np.zeros(n_fine)
        for i in range(n_coarse):
            start = i * factor
            end = min(start + factor, n_fine)
            f_fine[start:end] = f_coarse[i]
    
    f_fine = _normalize(f_fine)
    return f_fine if f_fine is not None else np.ones(n_fine) / n_fine

def _interpolate_upsample(f_coarse, n_fine):
    """Upsample by linear interpolation."""
    n_coarse = len(f_coarse)
    if n_coarse == n_fine:
        return f_coarse.copy()
    x_coarse = np.linspace(0, 1, n_coarse)
    x_fine = np.linspace(0, 1, n_fine)
    f_fine = np.interp(x_fine, x_coarse, f_coarse)
    return _normalize(f_fine)

def _cosine_init(n):

    f = np.cos(np.pi * (j + 0.5) / float(n)) ** 2

    j = np.arange(n, dtype=np.float64) - (n - 1) / 2.0
    sigma = sigma_frac * float(n)
    f = np.exp(-j * j / (2.0 * sigma * sigma))

def _modulated_init(n, freq, power):

def _lp_frank_wolfe(f_init, max_iters, time_budget):
    """LP-based refinement."""
    from scipy.optimize import linprog
    from scipy.sparse import coo_matrix

    f = _normalize(f)

        return f_init.copy(), float('inf')

            break
        
        h = np.convolve(f, f)
        M = float(np.max(h))

        row_idx = []
        col_idx = []
        vals = []
        for j in range(n):
            rows_j = np.arange(j, j + n)
            row_idx.extend(rows_j.tolist())
            col_idx.extend([j] * n)
            vals.extend(f.tolist())
        
        C = coo_matrix((vals, (row_idx, col_idx)), shape=(2 * n - 1, n))
        
        c_obj = -np.ones(n)
        res = linprog(c_obj, A_ub=C, b_ub=np.full(2 * n - 1, M),
                      bounds=[(0, None)] * n, method='highs')
        
        if not res.success or res.x is None:
            break
        
        g = res.x
        Sg = float(np.sum(g))
        if Sg < 1e-10:
            break
        g = g / Sg
        
        # Line search
        best_t = 0.0
        best_s = best_score
        for t in [0.1, 0.2, 0.3, 0.5, 0.7, 1.0]:
            f_trial = _normalize((1.0 - t) * f + t * g)
            if f_trial is None:
                continue
            s = _score_raw(f_trial)
            if s < best_s:
                best_s = s
                best_t = t
        
        if best_t == 0.0:
            f = g  # Just jump to the LP solution
        else:
            f = _normalize((1.0 - best_t) * f + best_t * g)
            if f is None:
                break
        
        score = _score_raw(f)
        if score < best_score:
            best_score = score
            best_f = f.copy()
    
    return best_f, best_score
