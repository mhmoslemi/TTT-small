import time
import numpy as np

def project_simplex(v):
    """Project vector v onto the simplex {x >= 0, sum x = 1}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n+1) > (cssv - 1))[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0)

def local_score(a):
    a = np.asarray(a, dtype=np.float64)
    n = a.size
    if n == 0:
        return float('inf')
    a = np.clip(a, 0, 1000)
    s = float(np.sum(a))
    if s <= 1e-12:
        return float('inf')
    c = np.convolve(a, a)
    m = float(np.max(c))
    return float(2.0 * n * m / (s * s))

def compute_grad_smoothed(a, tau):
    """Compute the smoothed gradient of max_k c_k using softmax weighting."""
    n = a.size
    c = np.convolve(a, a)  # length 2n-1
    # Softmax weights
    c_shifted = c - np.max(c)  # numerical stability
    weights = np.exp(tau * c_shifted)
    weights /= np.sum(weights)
    # grad_j = 2 * sum_k weights_k * a_{k-j} (valid indices)
    # = 2 * sum_m a_m * weights_{j+m}
    grad = np.zeros(n, dtype=np.float64)
    for j in range(n):
        # sum over m=0 to n-1 of a[m] * weights[j+m]
        # j+m ranges from j to j+n-1, which must be < 2n-1
        # So m ranges from max(0, j-...) ... actually j+m < 2n-1 means m < 2n-1-j
        # and m >= 0, m < n
        # So m ranges from 0 to min(n-1, 2n-2-j)
        m_max = min(n-1, 2*n-2-j)
        if m_max >= 0:
            grad[j] = 2.0 * np.dot(a[:m_max+1], weights[j:j+m_max+1])
    return grad

def optimize_smoothed_subgradient(a, tau_start, tau_end, max_time, seed):
    a = np.asarray(a, dtype=np.float64).copy()
    a = project_simplex(a)  # ensure on simplex
    n = a.size
    best = a.copy()
    best_score = 2.0 * n * np.max(np.convolve(a, a))  # sum is 1
    
    t0 = time.monotonic()
    deadline = t0 + max_time
    rng = np.random.default_rng(seed)
    
    eta = 0.5 / n  # initial learning rate
    stuck = 0
    tau = tau_start
    
    while time.monotonic() < deadline:
        # Update tau (anneal towards tau_end)
        progress = (time.monotonic() - t0) / max_time
        tau = tau_start + (tau_end - tau_start) * progress
        
        # Compute smoothed gradient
        c = np.convolve(a, a)
        c_max = np.max(c)
        weights = np.exp(tau * (c - c_max))
        weights /= np.sum(weights)
        
        grad = 2.0 * np.convolve(a[::-1], weights)[n-1:2*n-1]
        
        # Take step
        a_new = a - eta * grad
        a_new = project_simplex(a_new)
        
        # Evaluate
        c_new = np.convolve(a_new, a_new)
        score_new = 2.0 * n * np.max(c_new)
        
        if score_new < best_score:
            best = a_new.copy()
            best_score = score_new
            stuck = 0
            a = a_new
        else:
            a = a_new  # still take the step (stochastic-like)
            stuck += 1
        
        # Adjust learning rate
        if stuck > 50:
            eta *= 0.8
            stuck = 0
        
        # Occasionally add noise to escape local minima
        if stuck > 200:
            noise = np.exp(rng.normal(0, 0.1, n))
            a = project_simplex(best * noise)
            eta = 0.5 / n
            stuck = 0
    
    return best, best_score
