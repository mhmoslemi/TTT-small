    """Subgradient descent to minimize max(conv) subject to sum fixed, a >= 0."""
    f = f.copy().astype(np.float64)
    n = len(f)
    S = float(np.sum(f))
    best_f = f.copy()
    best_score = evaluate_sequence(f.tolist())
    
    # Initial step size
    max_conv = float(np.max(np.convolve(f, f)))
    eta = S * max_conv / (n * S)  # some heuristic

    eta = 0.1 * S / max_conv if max_conv > 0 else 0.1

        if time.monotonic() > deadline:
            break
        
        conv = np.convolve(f, f)
        M = float(np.max(conv))

        top_k = np.argsort(conv)[-K:]
        # Weight by how close to max
        weights = np.zeros(len(conv))
        for idx in top_k:
            weights[idx] = max(0, conv[idx] - 0.9 * M)  # or some soft weighting

def solve_lp_minimize_peak(f, S):
    """Solve: min M s.t. conv(g,f)_k <= M, sum(g) = S, g >= 0."""
    n = len(f)

    # Objective: minimize M => c = [0, 0, ..., 0, 1]
    # Constraints:
    #   conv(g,f)_k <= M => sum_j f[k-j]*g[j] - M <= 0 for all k
    #   sum(g) = S (equality constraint)
    #   g[j] >= 0, M >= 0

    b_ub = np.zeros(2*n-1)

        for j in range(n):
            if 0 <= k-j < n:
                A_ub[k, j] = f[k-j]
        A_ub[k, n] = -1  # -M coefficient
    
    A_eq = np.zeros((1, n+1))
    A_eq[0, :n] = 1.0
    b_eq = [S]
    
    c = np.zeros(n+1)
    c[n] = 1.0  # minimize M
    
    bounds = [(0, None)] * (n+1)
    
    result = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq, bounds=bounds, method='highs')

    return None, None

Wait, I realize there's an issue with my subgradient computation. Let me fix it.

For conv[k] = sum_{j: 0<=j<n, 0<=k-j<n} f[j] * f[k-j], the derivative w.r.t. f[i] is:
- f[i] appears in the term where j=i: f[i]*f[k-i] (if 0<=k-i<n), derivative = f[k-i]
- f[i] appears in the term where k-j=i, i.e., j=k-i: f[k-i]*f[i] (if 0<=k-i<n), derivative = f[k-i]
- These are the same term when i = k-i, i.e., i = k/2. In that case, it's f[i]^2, derivative = 2*f[i] = 2*f[k-i].

So in all cases: d conv[k]/df[i] = 2*f[k-i] if 0 <= k-i < n, else 0.

This is what I had. Let me vectorize it:

def compute_subgradient(f, top_k_lags):
    """Compute weighted subgradient at given lags."""
    n = len(f)
    grad = np.zeros(n)
    for k in top_k_lags:
        # For each j in [0, n-1], if 0 <= k-j < n, add 2*f[k-j] to grad[j]
        j_min = max(0, k - (n-1))
        j_max = min(k, n-1)
        if j_max >= j_min:
            # f[k-j] for j in [j_min, j_max]
            # k-j ranges from k-j_max to k-j_min
            # = max(k-(n-1), 0) to min(k, n-1) reversed
            indices = k - np.arange(j_min, j_max+1)
            grad[j_min:j_max+1] += 2.0 * f[indices]
    return grad

    """
    Solve LP: minimize M
    subject to: conv(g, f)[k] <= M for all k=0..2n-2
                sum(g) = S
                g >= 0, M >= 0
    Returns (g_opt, M_opt) or (None, None).
    """
    n = len(f)
    # Build constraint matrix for conv(g,f) <= M
    # Variables: g[0..n-1], M  (total n+1)
    A_ub_rows = []
    A_ub_cols = []
    A_ub_data = []
    for k in range(2 * n - 1):
        j_min = max(0, k - (n - 1))
        j_max = min(k, n - 1)
        for j in range(j_min, j_max + 1):
            A_ub_rows.append(k)
            A_ub_cols.append(j)
            A_ub_data.append(f[k - j])
        # -M coefficient
        A_ub_rows.append(k)
        A_ub_cols.append(n)
        A_ub_data.append(-1.0)

    A_ub = csr_matrix((A_ub_data, (A_ub_rows, A_ub_cols)), shape=(2 * n - 1, n + 1))
    b_ub = np.zeros(2 * n - 1)

    # Equality: sum(g) = S
    A_eq = csr_matrix((np.ones(n), (np.zeros(n, dtype=int), np.arange(n))), shape=(1, n + 1))
    b_eq = np.array([S])

    c = np.zeros(n + 1)
    c[n] = 1.0  # minimize M

    bounds = [(0, None)] * (n + 1)

    try:
        result = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                         bounds=bounds, method='highs',
                         options={"presolve": True, "dual_feasibility_tolerance": 1e-9})
        if result.success:
            return result.x[:n].copy(), float(result.x[n])
    except Exception:
        pass
    return None, None


def compute_subgradient(f, top_lags):
    """Compute subgradient of max_k conv_k at given lags (unweighted sum)."""
    n = len(f)
    grad = np.zeros(n)
    for k in top_lags:
        j_min = max(0, k - (n - 1))
        j_max = min(k, n - 1)
        if j_max >= j_min:
            indices = k - np.arange(j_min, j_max + 1)
            grad[j_min:j_max + 1] += 2.0 * f[indices]
    return grad


def lp_blend_optimize(f, deadline, max_iter=300, verbose_prefix=""):
    """
    Iteratively solve LP to minimize peak of conv(g,f), then blend f toward g.
    Uses line search over blend factor for robustness.
    """
    f = f.copy().astype(np.float64)
    n = len(f)
    S = float(np.sum(f))
    best_f = f.copy()
    best_score = evaluate_sequence(f.tolist())
    no_improve = 0

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break

        start_t = time.monotonic()
        g_opt, M_opt = solve_lp_min_peak(f, S)
        lp_time = time.monotonic() - start_t

        if g_opt is None:
            no_improve += 1
            if no_improve > 20:
                break
            continue

        # Line search over blend factors
        blend_factors = [0.3, 0.5, 0.7, 0.85, 1.0]
        best_blend_f = None
        best_blend_score = best_score

        for t in blend_factors:
            candidate = (1.0 - t) * f + t * g_opt
            candidate = np.clip(candidate, 0.0, 1000.0)
            s = float(np.sum(candidate))
            if s > 1e-10:
                candidate = candidate * (S / s)
            score = evaluate_sequence(candidate.tolist())
            if np.isfinite(score) and score < best_blend_score:
                best_blend_score = score
                best_blend_f = candidate.copy()

        if best_blend_f is not None and best_blend_score < best_score:
            f = best_blend_f
            best_f = f.copy()
            best_score = best_blend_score
            no_improve = 0
            if (it + 1) % 20 == 0 or it < 5:
                print(f"    {verbose_prefix}LP iter {it+1}: score={best_score:.6f} "
                      f"(M_opt={M_opt:.6f}, lp={lp_time*1000:.0f}ms)")
        else:
            no_improve += 1
            if no_improve > 15:
                break

    return best_f, best_score


def subgradient_optimize(f, deadline, K=8, verbose_prefix=""):
    """
    Subgradient descent with top-K active constraints and adaptive step size.
    """
    f = f.copy().astype(np.float64)
    n = len(f)
    S = float(np.sum(f))
    best_f = f.copy()
    best_score = evaluate_sequence(f.tolist())

    conv = np.convolve(f, f)
    M_init = float(np.max(conv))
    eta = 0.05 * S / max(M_init, 1e-10)
    best_eta = eta
    step_mult = 0.97

    for it in range(100000):
        if time.monotonic() > deadline:
            break

        conv = np.convolve(f, f)
        M = float(np.max(conv))

        # Top K lags
        if K < len(conv):
            top_k = np.argpartition(conv, -K)[-K:]
        else:
            top_k = np.arange(len(conv))

        grad = compute_subgradient(f, top_k)
        # Normalize gradient direction
        grad_norm = np.linalg.norm(grad)
        if grad_norm < 1e-15:
            break

        new_f = f - eta * grad
        new_f = np.clip(new_f, 0.0, 1000.0)
        s = float(np.sum(new_f))
        if s > 1e-10:
            new_f = new_f * (S / s)
        else:
            eta *= step_mult
            continue

        score = evaluate_sequence(new_f.tolist())
        if np.isfinite(score) and score < best_score:
            f = new_f
            best_f = f.copy()
            best_score = score
            best_eta = eta
            if (it + 1) % 500 == 0:
                print(f"    {verbose_prefix}SG iter {it+1}: score={best_score:.6f}")
        else:
            eta *= step_mult
            if eta < 1e-14:
                break

    return best_f, best_score


def perturb_and_optimize(f, deadline, n_perturbs=8, verbose_prefix=""):
    """Random perturbations followed by quick LP optimization."""
    f = f.copy().astype(np.float64)
    n = len(f)
    S = float(np.sum(f))
    best_f = f.copy()
    best_score = evaluate_sequence(f.tolist())
    rng = np.random.default_rng(99999)

    for i in range(n_perturbs):
        if time.monotonic() > deadline:
            break

        # Different perturbation strategies
        strategy = i % 4
        if strategy == 0:
            # Gaussian noise
            noise = rng.normal(0, 0.05 * S / n, size=n)
            candidate = f + noise
        elif strategy == 1:
            # Random spike removal/addition
            mask = rng.random(n) < 0.1
            noise = np.zeros(n)
            noise[mask] = rng.uniform(-0.1, 0.1) * S / n
            candidate = f + noise
        elif strategy == 2:
            # Smoothed random perturbation
            raw = rng.normal(0, 0.1 * S / n, size=n)
            kernel = np.ones(5) / 5.0
            noise = np.convolve(raw, kernel, mode='same')
            candidate = f + noise
        else:
            # Edge modification
            candidate = f.copy()
            edge_size = max(5, n // 20)
            candidate[:edge_size] *= rng.uniform(0.7, 1.3)
            candidate[-edge_size:] *= rng.uniform(0.7, 1.3)

        candidate = np.clip(candidate, 0.0, 1000.0)
        s = float(np.sum(candidate))
        if s > 1e-10:
            candidate = candidate * (S / s)

        score = evaluate_sequence(candidate.tolist())
        if not np.isfinite(score):
            continue

        # Quick LP optimization
        sub_deadline = min(deadline, time.monotonic() + 20)
        opt_f, opt_s = lp_blend_optimize(candidate, sub_deadline, max_iter=30,
                                         verbose_prefix=f"{verbose_prefix}p{i} ")
        if opt_s < best_score:
            best_score = opt_s
            best_f = opt_f.copy()
            print(f"    {verbose_prefix}perturb {i} (strat={strategy}): score={best_score:.6f}")

    return best_f, best_score


def coordinate_descent_polish(f, deadline, max_iter=5000, verbose_prefix=""):
    """
    Coordinate-descent style: pick random positions, try to adjust them.
    """
    f = f.copy().astype(np.float64)
    n = len(f)
    S = float(np.sum(f))
    best_f = f.copy()
    best_score = evaluate_sequence(f.tolist())
    rng = np.random.default_rng(77777)

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break

        # Pick 2-3 random positions
        positions = rng.choice(n, size=min(3, n), replace=False)
        candidate = f.copy()

        for pos in positions:
            # Try a few values
            for delta in [0.01, -0.01, 0.05, -0.05] * S / n:
                test = candidate.copy()
                test[pos] = max(0.0, test[pos] + delta)
                # Renormalize
                s = float(np.sum(test))
                if s > 1e-10:
                    test = test * (S / s)
                    score = evaluate_sequence(test.tolist())
                    if np.isfinite(score) and score < best_score:
                        candidate = test.copy()
                        best_score = score
                        best_f = candidate.copy()

        # Also try the candidate directly
        s = float(np.sum(candidate))
        if s > 1e-10:
            candidate = candidate * (S / s)
            score = evaluate_sequence(candidate.tolist())
            if np.isfinite(score) and score < best_score:
                best_score = score
                best_f = candidate.copy()
            f = candidate
        else:
            f = candidate

        if (it + 1) % 100 == 0:
            print(f"    {verbose_prefix}CD iter {it+1}: score={best_score:.6f}")

    return best_f, best_score


def propose_candidate(budget_s=1500, **kwargs):
    t0 = time.monotonic()
    deadline = t0 + max(0.0, float(budget_s) - 10.0)

    best_sequence = None
    best_score = float('inf')

    # Load parent
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64).copy()
        parent = np.clip(parent, 0.0, 1000.0)
        if parent.ndim == 1 and parent.size > 0 and np.all(np.isfinite(parent)):
            score = evaluate_sequence(parent.tolist())
            if np.isfinite(score):
                best_score = score
                best_sequence = parent.copy()
                print(f"Parent n={parent.size}: score={score:.6f}")
    except (NameError, Exception) as e:
        print(f"Could not load parent: {e}")

    if best_sequence is None:
        best_sequence = np.ones(512)
        best_score = evaluate_sequence(best_sequence.tolist())
        print(f"Default uniform n=512: score={best_score:.6f}")

    n = len(best_sequence)

    # Phase 1: Heavy LP optimization on parent (50% of time)
    phase1_deadline = t0 + (deadline - t0) * 0.50
    print(f"\n=== Phase 1: LP optimization on n={n} (budget={phase1_deadline-t0:.0f}s) ===")
    f1, s1 = lp_blend_optimize(best_sequence, phase1_deadline, max_iter=500)
    if s1 < best_score:
        best_score = s1
        best_sequence = f1.copy()
        print(f"Phase 1 best: {best_score:.6f}")

    # Phase 2: Subgradient polish (10%)
    phase2_deadline = t0 + (deadline - t0) * 0.60
    if time.monotonic() < phase2_deadline:
        print(f"\n=== Phase 2: Subgradient polish ===")
        f2, s2 = subgradient_optimize(best_sequence, phase2_deadline, K=10)
        if s2 < best_score:
            best_score = s2
            best_sequence = f2.copy()
            print(f"Phase 2 best: {best_score:.6f}")

    # Phase 3: Random perturbations + local opt (15%)
    phase3_deadline = t0 + (deadline - t0) * 0.75
    if time.monotonic() < phase3_deadline:
        print(f"\n=== Phase 3: Perturbations (budget={phase3_deadline-time.monotonic():.0f}s) ===")
        f3, s3 = perturb_and_optimize(best_sequence, phase3_deadline, n_perturbs=6)
        if s3 < best_score:
            best_score = s3
            best_sequence = f3.copy()
            print(f"Phase 3 best: {best_score:.6f}")

    # Phase 4: Coordinate descent polish (10%)
    phase4_deadline = t0 + (deadline - t0) * 0.85
    if time.monotonic() < phase4_deadline:
        print(f"\n=== Phase 4: Coordinate descent ===")
        f4, s4 = coordinate_descent_polish(best_sequence, phase4_deadline, max_iter=3000)
        if s4 < best_score:
            best_score = s4
            best_sequence = f4.copy()
            print(f"Phase 4 best: {best_score:.6f}")

    # Phase 5: Final LP refinement with remaining time (5%)
    if time.monotonic() < deadline - 5:
        print(f"\n=== Phase 5: Final LP refinement ===")
        f5, s5 = lp_blend_optimize(best_sequence, deadline, max_iter=100, verbose_prefix="[final] ")
        if s5 < best_score:
            best_score = s5
            best_sequence = f5.copy()
            print(f"Phase 5 best: {best_score:.6f}")

    # Phase 6: Try doubling length if time permits
    if time.monotonic() < deadline - 90:
        print(f"\n=== Phase 6: Try n={n*2} ===")
        new_n = n * 2
        idx_old = np.linspace(0, 1, n)
        idx_new = np.linspace(0, 1, new_n)
        new_f = np.interp(idx_new, idx_old, best_sequence)
        new_f = np.clip(new_f, 0.0, 1000.0)
        score = evaluate_sequence(new_f.tolist())
        if np.isfinite(score):
            print(f"  Interpolated n={new_n}: score={score:.6f}")
            if score < best_score:
                best_score = score
                best_sequence = new_f.copy()

            sub_dl = min(deadline, time.monotonic() + 60)
            f6, s6 = lp_blend_optimize(new_f, sub_dl, max_iter=50, verbose_prefix="[n2] ")
            if s6 < best_score:
                best_score = s6
                best_sequence = f6.copy()
                print(f"Phase 6 best: {best_score:.6f}")

    elapsed = time.monotonic() - t0
    print(f"\nFINAL: n={len(best_sequence)}, score={best_score:.6f}, time={elapsed:.1f}s")
    return [float(x) for x in best_sequence]
