"""
Hybrid optimization for minimizing 2n*max(conv(a,a))/sum(a)^2.
Uses smoothed gradient descent (log-sum-exp approximation of max),
LP-based hill climbing, and coordinate descent polish.
Explores multiple initializations and sequence lengths.

import numpy as np
import time
from scipy.optimize import linprog
from scipy.sparse import csr_matrix


def build_conv_matrix_sparse(f):
    """Build sparse matrix A where (A@g)[k] = convolve(f,g)[k]."""
    n = len(f)
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        j_min = max(0, k - (n - 1))
        j_max = min(k, n - 1)
        for j in range(j_min, j_max + 1):
            rows.append(k)
            cols.append(j)
            vals.append(f[k - j])
    return csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))


def solve_lp_direction(f, M):
    """Solve: max sum(g) s.t. g >= 0, convolve(f,g)_k <= M for all k."""
    n = len(f)
    A = build_conv_matrix_sparse(f)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    try:
        result = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
        if result.success:
            return result.x
    except Exception:
        pass
    return None


def compute_smooth_gradient(f, beta):
    """Compute gradient of smooth max approximation of conv(f,f).
    Returns gradient vector of length n and the smooth max value."""
    n = len(f)
    conv = np.convolve(f, f)
    max_conv = float(np.max(conv))
    
    # Numerically stable log-sum-exp
    shifted = beta * (conv - max_conv)
    weights = np.exp(shifted)
    weights /= np.sum(weights)
    
    smooth_max = max_conv + np.log(np.sum(np.exp(shifted))) / beta
    
    # grad_j = 2 * sum_{m=0}^{n-1} f[m] * weights[j+m]
    # = 2 * (reversed_f * weights)[n-1 + j] for j=0..n-1
    # Using the formula: conv(reversed_f, weights)[n-1 : 2n-1]
    rev_f = f[::-1]
    full_conv = np.convolve(rev_f, weights, mode='full')
    grad = 2.0 * full_conv[n-1:2*n-1]
    
    return grad, smooth_max

def coordinate_descent_polish(f, deadline, max_sweeps=50):
    """Coordinate descent: optimize one coordinate at a time."""
    f = f.copy().astype(np.float64)
    n = len(f)
    best_f = f.copy()
    best_score = evaluate_sequence(f.tolist())
    S = float(np.sum(f))
    
    for sweep in range(max_sweeps):
        if time.monotonic() > deadline:
            break
        
        improved = False
        for j in range(n):
            if time.monotonic() > deadline:
                    break
            
            # Compute d_k, e_k, f_k for this j
            # conv_k = d_k + a_j * e_k + a_j^2 (if k == 2j)
            old_val = f[j]
            f[j] = 0.0  # temporarily set to 0
            conv_base = np.convolve(f, f)  # conv with a_j = 0
            
            # Now conv_k with a_j restored = conv_base_k + a_j * e_k + a_j^2 * delta(k, 2j)
            # e_k = 2 * a_{k-j} if 0 <= k-j < n and k-j != j
            #      = a_{k-j} if k-j == j (but then it's a_j^2, handled separately)
            e = np.zeros(2*n-1)
            for k in range(2*n-1):
                m = k - j
                if 0 <= m < n:
                    if m == j:  # k = 2j
                        pass  # handled by quadratic term
                    else:
                        e[k] = 2.0 * f[m]

            # Subject to: sum(f) = S, so x = S - sum(f_{-j}) = S - (sum(f) - old_val)
            # Wait, we need to maintain the sum. If we change a_j, we need to adjust.
            # Actually, for coordinate descent, we fix the sum constraint:
            # x must satisfy: x + sum(f_{-j}) = S, so x = S - sum(f_{-j})
            # But sum(f_{-j}) = S - old_val (since we set f[j]=0, sum(f) = S - old_val)
            # So x = S - (S - old_val) = old_val. That means we can't change a_j while keeping the sum!

            # Or: we optimize a_j freely and then renormalize the whole sequence.
            # Let's do the latter: find optimal a_j (>= 0) to minimize max conv,

            # where q_k = 1 if k==2j, else 0.

            a_hi = x_max

                f[j] = x
                c = np.convolve(f, f)

            def obj_fast(x):

                vals = conv_base + x * e
                if 2*j < 2*n-1:
                    vals[2*j] += x * x
                return float(np.max(vals))

            x2 = a_lo + gr * (a_hi - a_lo)

                    x2 = x1

                    x1 = a_hi - gr * (a_hi - a_lo)
                    f1 = obj_fast(x1)
                else:
                    a_lo = x1
                    x1 = x2
                    f1 = f2
                    x2 = a_lo + gr * (a_hi - a_lo)
                    f2 = obj_fast(x2)
            
            x_opt = (a_lo + a_hi) / 2
            new_val = max(0.0, x_opt)
            
            f[j] = new_val
            improved = True
        
        # Renormalize
        new_S = float(np.sum(f))
        if new_S > 1e-10:
            f = f * (S / new_S)
        
        new_score = evaluate_sequence(f.tolist())
        if np.isfinite(new_score) and new_score < best_score:

            best_score = new_score

"""
Hybrid optimization for minimizing 2n*max(conv(a,a))/sum(a)^2.
Uses smoothed gradient descent (log-sum-exp approximation of max) with FFT-based
gradient computation, LP-based hill climbing, and asymmetric initializations.
Key insight: optimal sequences must be asymmetric (symmetric ones have score >= 2).
"""
import numpy as np
import time
from scipy.optimize import linprog
from scipy.sparse import csr_matrix


def build_conv_matrix_sparse(f):
    """Build sparse matrix A where (A@g)[k] = convolve(f,g)[k]."""
    n = len(f)
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        j_min = max(0, k - (n - 1))
        j_max = min(k, n - 1)
        for j in range(j_min, j_max + 1):
            rows.append(k)
            cols.append(j)
            vals.append(f[k - j])
    return csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))


def solve_lp_direction(f, M):
    """Solve: max sum(g) s.t. g >= 0, convolve(f,g)_k <= M for all k."""
    n = len(f)
    A = build_conv_matrix_sparse(f)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    try:
        result = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
        if result.success:
            return result.x
    except Exception:
        pass
    return None


def compute_smooth_gradient(f, beta):
    """Compute gradient of smooth max approximation of conv(f,f).
    Returns (gradient_vector, smooth_max_value, raw_conv)."""
    n = len(f)
    conv = np.convolve(f, f)
    max_conv = float(np.max(conv))

    # Numerically stable softmax weights
    shifted = beta * (conv - max_conv)
    # Clip for numerical stability
    shifted = np.clip(shifted, -500, 500)
    weights = np.exp(shifted)
    wsum = np.sum(weights)
    if wsum <= 0 or not np.isfinite(wsum):
        weights = np.zeros(2 * n - 1)
        weights[np.argmax(conv)] = 1.0
    else:
        weights /= wsum

    smooth_max = max_conv + np.log(max(wsum, 1e-300)) / beta

    # grad_j = 2 * sum_{m=0}^{n-1} f[m] * weights[j+m]
    # = 2 * (reversed_f * weights)[n-1 + j] for j=0..n-1
    rev_f = f[::-1]
    full_conv = np.convolve(rev_f, weights, mode='full')
    grad = 2.0 * full_conv[n - 1:2 * n - 1]

    return grad, smooth_max, conv


def hard_max_gradient(f, conv=None):
    """Compute subgradient using uniform average over active (max) lags."""
    n = len(f)
    if conv is None:
        conv = np.convolve(f, f)
    max_val = float(np.max(conv))
    # Find all lags within numerical tolerance of max
    tol = 1e-10 * max_val
    active = np.where(conv >= max_val - tol)[0]
    # Average the hard-max gradient over active lags
    grad = np.zeros(n)
    for k in active:
        for j in range(n):
            m = k - j
            if 0 <= m < n:
                grad[j] += 2.0 * f[m]
    grad /= len(active)
    return grad


def smoothed_grad_descent(f, deadline, beta_init=2.0, beta_max=200.0,

def lp_iterate(f, deadline, step_size_init=0.15, verbose_prefix=""):
    """LP-based hill climbing with multiple step sizes."""
    f = f.copy().astype(np.float64)
    best_f = f.copy()
    best_score = evaluate_sequence(f.tolist())
    step_size = step_size_init
    no_improve = 0
    it = 0

    while time.monotonic() < deadline and no_improve < 30:
        conv_f = np.convolve(f, f)
        M = float(np.max(conv_f))
        if M <= 0:
            break

        g = solve_lp_direction(f, M)
        if g is None:
            no_improve += 1
            continue

        sum_g = float(np.sum(g))
        if sum_g <= 1e-15:
            no_improve += 1
            continue

        sum_f = float(np.sum(f))
        g = g * (sum_f / sum_g)

        # Try multiple step sizes for robustness
        best_new_f = None
        best_new_score = best_score
        for t in [step_size, step_size * 0.5, step_size * 0.25, step_size * 1.5]:
            new_f = (1.0 - t) * f + t * g
            new_f = np.clip(new_f, 0.0, 1000.0)
            s = evaluate_sequence(new_f.tolist())
            if np.isfinite(s) and s < best_new_score:
                best_new_score = s
                best_new_f = new_f

        it += 1
        if best_new_f is not None:
            f = best_new_f
            best_f = f.copy()

            no_improve = 0
            if it % 5 == 0:

        else:
            no_improve += 1
            step_size *= 0.7
            if step_size < 0.005:
                break

    best_score = evaluate_sequence(f.tolist())
    S = float(np.sum(f))
    gr = (np.sqrt(5.0 + 1.0) / 2.0 - 1.0)  # golden ratio conjugate ~0.618

    for sweep in range(max_sweeps):
        if time.monotonic() > deadline:
            break

        for j in range(n):
            if time.monotonic() > deadline:
                break

            old_val = f[j]
            # Compute base convolution (with a_j = 0)
            f[j] = 0.0
            conv_base = np.convolve(f, f)

            # Compute e[k] = 2*f[k-j] for valid k-j
            e = np.zeros(2 * n - 1)
            m_min = 0
            m_max = n - 1
            k_min = m_min + j
            k_max = m_max + j
            e[k_min:k_max + 1] = 2.0 * f[m_min:m_max + 1]
            # Remove the self-term at k=2j (that's the quadratic part)
            if 2 * j < 2 * n - 1:
                e[2 * j] = 0.0  # will be handled by x^2 term

            # Golden section search for optimal x in [0, x_max]
            x_max = max(3.0 * old_val, S / n * 3.0)

            def obj_fast(x):
                # max_k (conv_base[k] + x*e[k] + x^2 * delta(k, 2j))
                vals = conv_base + x * e
                if 2 * j < 2 * n - 1:
                    vals = vals.copy()
                    vals[2 * j] += x * x
                return float(np.max(vals))

            a_lo = 0.0
            a_hi = x_max
            x1 = a_hi - gr * (a_hi - a_lo)
            x2 = a_lo + gr * (a_hi - a_lo)
            f1 = obj_fast(x1)
            f2 = obj_fast(x2)

            for _ in range(50):
                if f1 < f2:
                    a_hi = x2
                    x2 = x1
                    f2 = f1
                    x1 = a_hi - gr * (a_hi - a_lo)
                    f1 = obj_fast(x1)
                else:
                    a_lo = x1
                    x1 = x2
                    f1 = f2
                    x2 = a_lo + gr * (a_hi - a_lo)
                    f2 = obj_fast(x2)

            x_opt = max(0.0, (a_lo + a_hi) / 2.0)
            f[j] = x_opt

        # Renormalize
        new_S = float(np.sum(f))
        if new_S > 1e-10:
            f = f * (S / new_S)

        new_score = evaluate_sequence(f.tolist())
        if np.isfinite(new_score) and new_score < best_score:
            best_f = f.copy()
            best_score = new_score
        # If no improvement over multiple sweeps, stop
        if sweep > 5 and new_score >= best_score * 0.9999:
            if sweep > 10:
                break

    return best_f, best_score


def make_initializations(n, rng, parent=None):
    """Generate diverse initializations emphasizing asymmetry."""
    inits = []
    idx = np.arange(n, dtype=np.float64)
    center = (n - 1) / 2.0
    half = center if center > 0 else 1.0

    # Parent (if available)
    if parent is not None and parent.size == n:
        inits.append(('parent', parent.copy()))

    # Linear tilts (asymmetric - key insight: symmetric => score >= 2)
    for amp in [0.2, 0.4, 0.6, 0.8]:
        tilt = 1.0 + amp * (idx - center) / half
        tilt = np.clip(tilt, 0.05, None)
        inits.append((f'tilt_{amp}', tilt))

    # Exponential tilts
    for amp in [0.5, 1.0, 1.5]:
        exp_tilt = np.exp(amp * (idx - center) / half)
        inits.append((f'exp_{amp}', exp_tilt))

    # Skewed bell curves
    for frac in [0.3, 0.4, 0.6, 0.7]:
        skew_bell = np.exp(-0.5 * ((idx - n * frac) / (n / 4.0)) ** 2)
        inits.append((f'skew_{frac}', skew_bell))

    # Sine-based asymmetric
    for freq in [1, 2, 3]:
        sine = 1.0 + 0.3 * np.sin(2 * np.pi * freq * idx / n)
        sine = np.clip(sine, 0.1, None)
        inits.append((f'sine_{freq}', sine))

    # Uniform
    inits.append(('uniform', np.ones(n)))

    # Random smooth
    freq1 = rng.uniform(0.5, 3.0)
    freq2 = rng.uniform(2.0, 5.0)
    ph1 = rng.uniform(0, 2 * np.pi)
    ph2 = rng.uniform(0, 2 * np.pi)
    smooth = 1.0 + 0.3 * np.cos(2 * np.pi * freq1 * idx / n + ph1) + \
             0.2 * np.cos(2 * np.pi * freq2 * idx / n + ph2)
    smooth = np.clip(smooth, 0.1, None)
    inits.append(('smooth_rand', smooth))

    # Random
    r = rng.uniform(0.5, 1.5, size=n)
    inits.append(('random', r))

    return inits


def propose_candidate(budget_s=1500, **kwargs):
    """Main search function: tries multiple initializations and optimizers."""
    t0 = time.monotonic()
    deadline = t0 + max(0.0, float(budget_s) - 8.0)

    best_sequence = None
    best_score = float('inf')

    # Load parent
    parent = None
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64).copy()
        parent = np.clip(parent, 0.0, 1000.0)
        if parent.ndim == 1 and parent.size > 0 and np.all(np.isfinite(parent)):
            score = evaluate_sequence(parent.tolist())
            if np.isfinite(score):
                best_score = score
                best_sequence = parent.copy()
                print(f"Parent n={parent.size}: score={score:.6f}")
    except (NameError, Exception) as e:
        print(f"No parent available: {e}")

    rng = np.random.default_rng(42)
    primary_n = parent.size if parent is not None else 512

    # Generate and evaluate initializations
    inits = make_initializations(primary_n, rng, parent)

    print(f"\n--- Initial scores (n={primary_n}) ---")
    init_scores = []
    for label, f_init in inits:
        f_init = np.clip(f_init.astype(np.float64), 0.0, 1000.0)
        score = evaluate_sequence(f_init.tolist())
        init_scores.append((label, f_init, score))
        if np.isfinite(score):
            print(f"  {label:15s}: {score:.6f}")
            if score < best_score:
                best_score = score
                best_sequence = f_init.copy()

    # Sort by score (best first)
    init_scores.sort(key=lambda x: x[2] if np.isfinite(x[2]) else 1e9)

    remaining = deadline - time.monotonic()
    print(f"\n--- Optimization phase ({remaining:.0f}s remaining) ---")

    # Allocate time: top initializations get more time
    # Use inverse-score weighting
    valid_inits = [(l, f, s) for l, f, s in init_scores if np.isfinite(s)]
    n_valid = len(valid_inits)

    # Time allocation: proportional to 1/score, with minimum
    weights = [1.0 / (s + 0.5) for _, _, s in valid_inits]
    total_w = sum(weights)
    time_alloc = [remaining * 0.6 * w / total_w for w in weights]

    # Ensure minimum 20s for top 3
    for i in range(min(3, n_valid)):
        time_alloc[i] = max(time_alloc[i], 30.0)

    # Cap total
    total_alloc = sum(time_alloc)
    if total_alloc > remaining * 0.6:
        time_alloc = [t * remaining * 0.6 / total_alloc for t in time_alloc]

    # Optimize each initialization
    for i, (label, f_init, init_score) in enumerate(valid_inits):
        if time.monotonic() > deadline - 30:
            print("Time running out, stopping optimization loop")
            break

        alloc_time = time_alloc[i]
        if alloc_time < 15:
            continue

        sub_deadline = min(deadline - 20, time.monotonic() + alloc_time)
        print(f"\n  [{label}] init={init_score:.6f}, budget={alloc_time:.0f}s")

        # Phase A: Smoothed gradient descent (coarse)
        sgd_deadline = min(sub_deadline - 20, time.monotonic() + alloc_time * 0.4)
        f1, s1 = smoothed_grad_descent(
            f_init, sgd_deadline,
            beta_init=1.0, beta_max=80.0,
            lr_init=0.004, momentum=0.85, beta_growth=1.015,
            use_hard_max_after=60.0,
            verbose_prefix=f"[{label}] "
        )
        if s1 < best_score:
            best_score = s1
            best_sequence = f1.copy()
            print(f"    SGD coarse: {best_score:.6f}")

        # Phase B: Smoothed gradient descent (sharp, starting from Phase A result)
        if time.monotonic() < sub_deadline - 15:
            sgd_deadline2 = min(sub_deadline - 15, time.monotonic() + (sub_deadline - time.monotonic()) * 0.5)
            f2, s2 = smoothed_grad_descent(
                f1, sgd_deadline2,
                beta_init=30.0, beta_max=300.0,
                lr_init=0.001, momentum=0.8, beta_growth=1.008,
                use_hard_max_after=30.0,
                verbose_prefix=f"[{label}-sharp] "
            )
            if s2 < best_score:
                best_score = s2
                best_sequence = f2.copy()
                print(f"    SGD sharp: {best_score:.6f}")

        # Phase C: LP polish on the best from A/B
        if time.monotonic() < sub_deadline - 5:
            current = f1 if s1 <= s2 else f2
            f3, s3 = lp_iterate(
                current, sub_deadline,
                step_size_init=0.1,
                verbose_prefix=f"[{label}-LP] "
            )
            if s3 < best_score:
                best_score = s3
                best_sequence = f3.copy()
                print(f"    LP: {best_score:.6f}")

        # Phase D: Coordinate descent polish
        if time.monotonic() < sub_deadline - 3:
            f4, s4 = coordinate_polish(best_sequence, sub_deadline, max_sweeps=10)
            if s4 < best_score:
                best_score = s4
                best_sequence = f4.copy()
                print(f"    Coord: {best_score:.6f}")

    # Final refinement on the overall best
    if best_sequence is not None and time.monotonic() < deadline - 15:
        print(f"\n--- Final refinement (n={len(best_sequence)}, score={best_score:.6f}) ---")

        # Multiple SGD rounds with increasing sharpness
        n_refine = best_sequence.copy()
        for rnd in range(6):
            if time.monotonic() > deadline - 10:
                break
            beta_start = 20.0 * (rnd + 1)
            lr_start = 0.0015 / (rnd + 1)
            f_r, s_r = smoothed_grad_descent(
                n_refine, deadline - 5,
                beta_init=beta_start, beta_max=beta_start * 3,
                lr_init=lr_start, momentum=0.85,
                beta_growth=1.005, use_hard_max_after=beta_start,
                verbose_prefix=f"[refine{rnd}] "
            )
            if s_r < best_score:
                best_score = s_r
                best_sequence = f_r.copy()
                n_refine = f_r.copy()
                print(f"    Refine {rnd}: {best_score:.6f}")
            else:
                break

        # LP final
        if time.monotonic() < deadline - 5:
            f_lp, s_lp = lp_iterate(best_sequence, deadline, step_size_init=0.05,
                                    verbose_prefix="[final-LP] ")
            if s_lp < best_score:
                best_score = s_lp
                best_sequence = f_lp.copy()
                print(f"    Final LP: {best_score:.6f}")

        # Coordinate descent final
        if time.monotonic() < deadline - 3:
            f_cd, s_cd = coordinate_polish(best_sequence, deadline, max_sweeps=15)
            if s_cd < best_score:
                best_score = s_cd
                best_sequence = f_cd.copy()
                print(f"    Final Coord: {best_score:.6f}")

    if best_sequence is None:
        return [1.0] * 128

    elapsed = time.monotonic() - t0
    print(f"\nFINAL: n={len(best_sequence)}, score={best_score:.6f}, time={elapsed:.1f}s")
    return [float(x) for x in best_sequence]
