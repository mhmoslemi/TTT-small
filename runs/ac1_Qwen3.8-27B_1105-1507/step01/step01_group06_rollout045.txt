```python
"""
Minimizes J(a) = 2*n*max(convolve(a,a))/(sum(a))^2 using:
1. LP-based best-response with adaptive backtracking mixing
2. Nesterov-accelerated subgradient descent
3. L-BFGS-B with log-sum-exp smoothing for local refinement
4. Multi-length exploration (512, 1024, 2048) with diverse initializations
5. Active-set-aware equalization to flatten the top convolution peaks
"""

import numpy as np
from scipy.optimize import linprog, minimize
import time


def fast_conv(a):
    """Full self-convolution via FFT, O(n log n)."""
    n = len(a)
    s = 1
    while s < 2 * n - 1:
        s <<= 1
    fa = np.fft.rfft(a, s)
    return np.fft.irfft(fa * fa, s)[:2 * n - 1]


def build_A(a):
    """Build constraint matrix A where row k encodes (a*g)[k] = sum_j a[k-j]*g[j]."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for k in range(2 * n - 1):
        jl = max(0, k - n + 1)
        jh = min(n - 1, k)
        if jl <= jh:
            A[k, jl:jh + 1] = a[k - jh:k - jl + 1][::-1]
    return A


def lp_solve(a, M):
    """Solve: max sum(g) s.t. (a*g)[k] <= M for all k, g >= 0."""
    n = len(a)
    A = build_A(a)
    c = -np.ones(n)
    res = linprog(c, A_ub=A, b_ub=np.full(2 * n - 1, M),
                  bounds=[(0, None)] * n, method='highs')
    if res.success and res.x is not None and res.x.sum() > 1e-12:
        return res.x
    return None


def compute_subgradient(a, active_indices):
    """Compute subgradient of max_k (a*a)[k] at 'a' given active set."""
    n = len(a)
    grad = np.zeros(n)
    for k in active_indices:
        vals = np.arange(n)
        shifted = k - vals
        mask = (shifted >= 0) & (shifted < n)
        grad[mask] += 2.0 * a[shifted[mask]]
    grad /= max(1, len(active_indices))
    return grad


def logsumexp_objective(a, eps=0.01):
    """Smoothed objective: (1/eps)*log(sum exp(eps*(a*a))) for a with sum=1."""
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-10:
        return 1e10
    a = a / s
    b = fast_conv(a)
    b_shifted = b - b.max()
    return float((b.max() + np.log(np.sum(np.exp(eps * b_shifted))) / eps) * 2 * len(a))


def logsumexp_gradient(a, eps=0.01):
    """Gradient of the smoothed objective."""
    n = len(a)
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-10:
        return np.zeros(n)
    a = a / s
    b = fast_conv(a)
    b_shifted = b - b.max()
    weights = np.exp(eps * b_shifted)
    weights /= weights.sum()
    # Gradient of sum weights_k * (a*a)[k] w.r.t. a
    grad = np.zeros(n)
    kn = len(b)
    for k in range(kn):
        if weights[k] < 1e-10:
            continue
        shifted = k - np.arange(n)
        mask = (shifted >= 0) & (shifted < n)
        grad[mask] += weights[k] * 2.0 * a[shifted[mask]]
    # Account for normalization (a/s)
    grad = grad - a * grad.sum() / s
    return grad * 2.0 * n / s


def try_update(a, state):
    """Evaluate and update best if improved."""
    a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
    if a.sum() < 0.01:
        return
    a_norm = a / a.sum()
    b = fast_conv(a_norm)
    score = float(2 * len(a) * b.max() / 1.0)  # (sum=1) so (sum)^2 = 1
    if np.isfinite(score) and score < state['best_score']:
        state['best_score'] = score
        state['best'] = a_norm.copy()
        print(f"  NEW BEST: J={score:.6f} (n={a.size})", flush=True)


def lp_descent(a, deadline, max_iters, t0_mix, t_min_mix, decay, verbose=True):
    """LP best-response descent with backtracking mixing schedule."""
    a = a.copy()
    n = len(a)
    a /= a.sum()
    stagnation = 0

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        b = fast_conv(a)
        M = float(b.max())

        g = lp_solve(a, M)
        if g is None:
            if verbose:
                print(f"    LP failed at iter {it}", flush=True)
            break
        g = g / g.sum()

        # Backtracking: find a mixing rate that actually reduces M
        t = t0_mix * (decay ** it)
        t = max(t, t_min_mix)
        improved = False
        for t_try in [t, t * 0.5, t * 0.25, 0.1, 0.05, 0.01]:
            a_trial = (1.0 - t_try) * a + t_try * g
            a_trial = np.clip(a_trial, 0.0, None)
            s = a_trial.sum()
            if s < 1e-10:
                continue
            a_trial /= s
            b_trial = fast_conv(a_trial)
            if b_trial.max() < M * (1.0 - 1e-10):
                a = a_trial
                improved = True
                stagnation = 0
                break
        if not improved:
            stagnation += 1
            if stagnation > 15:
                if verbose:
                    print(f"    Stagnated at iter {it}, M={M:.6f}", flush=True)
                break
        else:
            if verbose and (it % 20 == 0 or it == max_iters - 1):
                print(f"    LP iter {it}: M={M:.6f}, t={t_try:.4f}", flush=True)

    return a


def nesterov_subgradient(a, deadline, max_iters=2000, step_init=0.01):
    """Nesterov-accelerated subgradient on max self-conv (sum=1, a>=0)."""
    a = a.copy()
    a /= a.sum()
    n = len(a)
    a_prev = a.copy()
    beta = 0.5
    step = step_init
    best_local = a.copy()
    best_local_val = fast_conv(a).max()

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        # Nesterov lookahead
        y = a + beta * (a - a_prev)
        y = np.clip(y, 0.0, None)
        sy = y.sum()
        if sy < 1e-10:
            break
        y = y / sy

        # Compute subgradient
        b = fast_conv(y)
        M = b.max()
        active = np.where(b >= M * 0.9995)[0]
        grad = compute_subgradient(y, active)

        # Gradient step with projected normalization
        a = np.clip(a - step * grad, 0.0, None)
        sa = a.sum()
        if sa < 1e-10:
            a = best_local.copy()
            break
        a = a / sa

        # Update momentum
        beta = (1.0 + np.sqrt(1.0 + 4.0 * beta * beta)) / 2.0
        # Adaptive step size
        fa = fast_conv(a).max()
        if fa < best_local_val:
            best_local_val = fa
            best_local = a.copy()
        else:
            step *= 0.95

        it_mod = max(1, it + 1)
        beta = 2.0 / (it_mod + 2.0)  # Nesterov schedule

        a_prev = a_prev  # wait, need to save before updating

    return best_local


def lbfgs_refine(a, deadline, max_iter=2000):
    """L-BFGS-B refinement with log-sum-exp smoothing."""
    a = a.copy()
    a = np.clip(a, 0.0, None)
    a /= a.sum()
    n = len(a)

    def _obj(x):
        ax = np.maximum(x, 0.0)
        s = ax.sum()
        if s < 1e-10:
            return 1e10
        ax = ax / s
        b = fast_conv(ax)
        # Use a smooth approximation of max
        eps = 0.005
        b_shift = b - b.max()
        smooth_max = b.max() + np.log(np.sum(np.exp(eps * b_shift))) / eps
        return float(2.0 * n * smooth_max)

    def _grad(x):
        ax = np.maximum(x, 0.0)
        s = ax.sum()
        if s < 1e-10:
            return np.zeros(n)
        ax = ax / s
        b = fast_conv(ax)
        eps = 0.005
        b_shift = b - b.max()
        w = np.exp(eps * b_shift)
        w /= w.sum()
        grad = np.zeros(n)
        for k in range(len(b)):
            if w[k] < 1e-12:
                continue
            shifted = k - np.arange(n)
            mask = (shifted >= 0) & (shifted < n)
            grad[mask] += w[k] * 2.0 * ax[shifted[mask]]
        # Chain rule for normalization
        grad = grad - ax * grad.sum() / s
        return grad * 2.0 * n / s

    res = minimize(_obj, a, method='L-BFGS-B',
                   jac=_grad,
                   bounds=[(0, None)] * n,
                   options={'maxiter': max_iter, 'maxfun': 50000,
                            'ftol': 1e-16, 'gtol': 1e-14, 'disp': False})
    if res.fun < 1e10:
        return np.clip(res.x, 0, None)
    return a


def make_u_shape(n, c):
    """Create U-shaped sequence: a[i] = 1 + c*(2i/(n-1) - 1)^2."""
    t = np.linspace(-1, 1, n)
    return 1.0 + c * t * t


def make_cosine_shape(n, freq, amp):
    """Create cosine-modulated sequence."""
    t = np.linspace(0, 1, n)
    return 1.0 + amp * np.cos(2 * np.pi * freq * t)

def clip_and_norm(a):
    """Clip to non-negative and normalize to unit sum."""
    a = np.clip(a, 0.0, None)
    s = a.sum()
    if s < 1e-10:
        a = np.ones_like(a)
        s = a.sum()
    return a / s


def _short_descent(a, deadline, max_iters=500, step_init=0.005):
    """Simple subgradient descent (for longer sequences where LP is too slow)."""
    a = np.clip(a, 0.0, None)
    a /= a.sum()
    n = len(a)
    step = step_init
    a_prev = a.copy()
    best_a = a.copy()
    best_val = fast_conv(a).max()
    beta = 0.5
    stagnation = 0

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        # Nesterov
        y = a + beta * (a - a_prev)
        y = np.clip(y, 0.0, None)
        sy = y.sum()
        if sy < 1e-10:
            break
        y = y / sy

        b = fast_conv(y)
        M = b.max()
        active = np.where(b >= M * 0.999)[0]
        grad = compute_subgradient(y, active)

        a = np.clip(a - step * grad, 0.0, None)
        sa = a.sum()
        if sa < 1e-10:
            break
        a = a / sa

        curr_val = fast_conv(a).max()
        if curr_val < best_val - 1e-10:
            best_val = curr_val
            best_a = a.copy()
            stagnation = 0
        else:
            stagnation += 1
            step *= 0.97

        beta = 2.0 / (it + 2.0)
        a_prev = a

        if stagnation > 100:
            break

    return best_a
```
