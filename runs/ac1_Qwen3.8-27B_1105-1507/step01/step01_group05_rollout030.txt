        # So k ranges from j to j+n-1
        A[j:j+n, j] = f
    return A


def compute_score_normalized(f):
    """Compute score for a sequence with sum=1: 2n*max(conv(f,f))."""
    n = len(f)
    conv = np.convolve(f, f)
    M = conv.max()
    return 2.0 * n * M


def compute_score(f):
    """Compute the full score: 2*n*max(conv(a,a))/sum(a)^2."""
    n = len(f)
    S = f.sum()
    if S < 1e-15:
        return np.inf
    conv = np.convolve(f, f)
    M = conv.max()
    return 2.0 * n * M / (S * S)


def soft_max_gradient(a, p):
    """
    Compute gradient of the soft-max approximation of the score.
    score_approx = 2n * (sum (conv^p))^(1/p) / (sum a)^2
    Returns: (score_approx, gradient)
    """
    n = len(a)
    conv = np.convolve(a, a)
    S = a.sum()
    
    if S < 1e-15:
        return np.inf, np.zeros_like(a)
    
    conv_p_sum = np.sum(conv ** p)
    if conv_p_sum < 1e-300:
        return 0.0, np.zeros_like(a)
    
    u = conv_p_sum ** (1.0 / p)  # soft max value
    score = 2.0 * n * u / (S * S)
    
    # Gradient computation
    # d(score)/d(a_i) = 2n/S^3 * (S * du/d(a_i) - 2u)
    # du/d(a_i) = u^((1-p)/p) * 2 * sum_j conv_{i+j}^(p-1) * a[j]
    
    conv_pow = conv ** (p - 1)
    
    # Compute 2 * sum_j conv_{i+j}^(p-1) * a[j] for all i
    # Using sliding window: windows[i] = conv_pow[i:i+n]
    # result[i] = windows[i] @ a
    windows = sliding_window_view(conv_pow, n)
    conv_grad = 2.0 * (windows @ a)
    
    du_da = (u ** ((1.0 - p) / p)) * conv_grad
    
    grad = (2.0 * n / (S ** 3)) * (S * du_da - 2.0 * u * np.ones_like(a))
    
    return score, grad


def adam_step(params, grads, t, lr=0.01, beta1=0.9, beta2=0.999, eps=1e-8):
    """Single Adam step. params and grads are dicts."""
    m = {}
    v = {}
    for key in params:
        if key not in m:
            m[key] = 0.0
            v[key] = 0.0
        m[key] = beta1 * m[key] + (1 - beta1) * grads[key]
        v[key] = beta2 * v[key] + (1 - beta2) * grads[key] ** 2
        m_hat = m[key] / (1 - beta1 ** t)
        v_hat = v[key] / (1 - beta2 ** t)
        params[key] -= lr * m_hat / (np.sqrt(v_hat) + eps)
    return params.copy(), m, v


def optimize_soft_max(a, time_budget, rng, p_start=20, p_end=200, n_iters=None, verbose=True):
    """
    Optimize using soft-max gradient descent with Adam.
    a: initial sequence (will be normalized to sum=1)
    Returns: (optimized_sequence, score)
    """
    n = len(a)
    a = a.astype(np.float64).copy()
    S = a.sum()
    if S < 1e-15:
        return a, np.inf
    a = a / S
    
    best_a = a.copy()
    best_score = compute_score_normalized(a)
    
    start = time.monotonic()
    if n_iters is None:
        n_iters = 100000
    
    # Adam state
    m = np.zeros(n)
    v = np.zeros(n)
    t = 0
    
    # Learning rate schedule
    base_lr = 0.1 / np.sqrt(n)
    
    patience = 500
    no_improve = 0
    
    for it in range(n_iters):
        elapsed = time.monotonic() - start
        if elapsed > time_budget * 0.9:
            break
        
        t += 1
        # Adaptive p parameter
        progress = min(1.0, elapsed / time_budget)
        p = p_start + (p_end - p_start) * progress
        
        score, grad = soft_max_gradient(a, p)
        
        if not np.isfinite(score):
            break
        
        # Adam update
        lr = base_lr / (1.0 + 0.1 * t)
        m = 0.9 * m + 0.1 * grad
        v = 0.999 * v + 0.001 * grad ** 2
        m_hat = m / (1 - 0.9 ** t)
        v_hat = v / (1 - 0.999 ** t)
        update = lr * m_hat / (np.sqrt(v_hat) + 1e-8)
        
        a = a - update
        a = np.maximum(a, 0.0)
        
        # Normalize
        S = a.sum()
        if S < 1e-15:
            a = np.ones(n) / n
            continue
        a = a / S
        
        # Evaluate true score periodically
        if it % 50 == 0 or it < 10:
            true_score = compute_score_normalized(a)
            if true_score < best_score:
                best_score = true_score
                best_a = a.copy()
                no_improve = 0
                if verbose and it % 500 == 0:
                    print(f"      it={it}, p={p:.0f}, score={best_score:.6f}, lr={lr:.6f}")
            else:
                no_improve += 1
        
        if no_improve > patience:
            # Reduce learning rate
            base_lr *= 0.5
            no_improve = 0
            if base_lr < 1e-8:
                break
    
    # Final evaluation
    final_score = compute_score_normalized(best_a)
    return best_a, final_score


def peak_shaving(a, n_iter=1000, rng=None):
    """
    Peak shaving: iteratively identify the peak of the convolution
    and reduce the contributions to it.
    """
    if rng is None:
        rng = np.random.default_rng(0)
    
    n = len(a)
    a = a.astype(np.float64).copy()
    S = a.sum()
    a = a / S
    
    best_a = a.copy()
    best_score = compute_score_normalized(a)
    
    for it in range(n_iter):
        conv = np.convolve(a, a)
        k_star = np.argmax(conv)
        M = conv.max()
        
        # The peak at k_star gets contributions from pairs (j, k_star - j)
        # Reduce a_j for j where a_j * a_{k_star-j} is large
        # More precisely, reduce the "pressure" at the peak
        
        # Compute the gradient of the peak w.r.t. a
        # d(conv_{k_star})/d(a_i) = 2 * a[k_star - i] (if 0 <= k_star - i < n)
        k_valid_start = max(0, k_star - n + 1)
        k_valid_end = min(n, k_star + 1)
        
        # The "pressure" on a_i is proportional to a[k_star - i]
        pressure = np.zeros(n)
        for i in range(n):
            j = k_star - i
            if 0 <= j < n:
                pressure[i] = 2.0 * a[j]
        
        # multiplicative update: a_i *= exp(-alpha * pressure_i)
        alpha = 0.1 / (n / 10.0)  # scale with n
        a = a * np.exp(-alpha * pressure)
        a = np.maximum(a, 0.0)
        
        S = a.sum()
        if S < 1e-15:
            break
        a = a / S
        
        score = compute_score_normalized(a)
        if score < best_score:
            best_score = score
            best_a = a.copy()
    
    return best_a, best_score


def local_search(f, rng, n_iter=2000):
    """Simple random local search to polish a sequence."""
    n = len(f)
    f = f.copy()
    S = f.sum()
    f = f / S
    best = f.copy()
    best_score = compute_score_normalized(f)
    
    for _ in range(n_iter):
        f_candidate = f.copy()
        # Perturb a random block
        width = max(1, int(rng.integers(1, max(2, n // 32))))
        start = int(rng.integers(0, n - width))
        scale = np.exp(rng.normal(0, 0.3))
        f_candidate[start:start + width] *= scale
        f_candidate = np.clip(f_candidate, 0, None)
        S_new = f_candidate.sum()
        if S_new < 1e-10:
            continue
        f_candidate = f_candidate / S_new
        score = compute_score_normalized(f_candidate)
        if score < best_score:
            best = f_candidate.copy()
            best_score = score
            f = f_candidate
    
    return best, best_score

"""
Combined optimization for minimizing 2*n*max(conv(a,a))/sum(a)^2.
Uses soft-max gradient descent (Adam), LP-based optimization, peak shaving,
and frequency-domain inspired initializations across multiple sequence lengths.
Different from prior approach by primarily using gradient-based optimization
with soft-max surrogates rather than LP direction finding.
"""

import numpy as np
from scipy.optimize import linprog
import time
from numpy.lib.stride_tricks import sliding_window_view


def build_conv_matrix_fast(f):
    """Build matrix A such that A @ a gives conv(f, a)结构和 gradient structure."""
    n = len(f)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        A[j:j + n, j] = f
    return A


def compute_score_normalized(f):
    """Compute score for a sequence with sum=1: 2n*max(conv(f,f))."""
    n = len(f)
    conv = np.convolve(f, f)
    M = conv.max()
    return 2.0 * n * M


def compute_score(f):
    """Compute the full score: 2*n*max(conv(a,a))/sum(a)^2."""
    n = len(f)
    S = f.sum()
    if S < 1e-15:
        return np.inf
    conv = np.convolve(f, f)
    M = conv.max()
    return 2.0 * n * M / (S * S)


def soft_max_score_and_grad(a, p):
    """
    Compute soft-max approximation of score and its gradient.
    score_approx = 2n * (sum (conv^p))^(1/p) / (sum a)^2
    """
    n = len(a)
    conv = np.convolve(a, a)
    S = a.sum()

    if S < 1e-15:
        return np.inf, np.zeros_like(a)

    conv_p_sum = np.sum(conv ** p)
    if conv_p_sum < 1e-300:
        return 0.0, np.zeros_like(a)

    u = conv_p_sum ** (1.0 / p)
    score = 2.0 * n * u / (S * S)

    # du/d(a_i) = u^((1-p)/p) * 2 * sum_j conv_{i+j}^(p-1) * a[j]
    conv_pow = conv ** (p - 1)
    windows = sliding_window_view(conv_pow, n)
    conv_grad = 2.0 * (windows @ a)
    du_da = (u ** ((1.0 - p) / p)) * conv_grad

    grad = (2.0 * n / (S ** 3)) * (S * du_da - 2.0 * u * np.ones(n))

    return score, grad


def optimize_soft_max(a, time_budget, rng, p_start=15, p_end=500, verbose=False):
    """
    Optimize using soft-max gradient descent with Adam and cosine LR schedule.
    """
    n = len(a)
    a = a.astype(np.float64).copy()
    S = a.sum()
    if S < 1e-15:
        return a, np.inf
    a = a / S

    best_a = a.copy()
    best_score = compute_score_normalized(a)

    start = time.monotonic()
    t = 0
    m = np.zeros(n)
    v = np.zeros(n)

    base_lr = 0.5 / np.sqrt(n)
    peak_lr = base_lr
    min_lr = base_lr * 0.01
    total_iters = max(1000, int(time_budget * 200))
    warmup_iters = min(200, total_iters // 10)

    no_improve = 0
    patience = 800

    for it in range(total_iters):
        elapsed = time.monotonic() - start
        if elapsed > time_budget * 0.92:
            break

        t += 1
        # Adaptive p parameter (anneal from low to high for sharper max approximation)
        progress = min(1.0, elapsed / (time_budget * 0.8))
        p = p_start + (p_end - p_start) * progress ** 1.5

        score, grad = soft_max_score_and_grad(a, p)

        if not np.isfinite(score) or not np.isfinite(grad).all():
            t = max(1, t - 1)
            continue

        # Cosine annealing with warmup
        if t < warmup_iters:
            lr = peak_lr * (t / warmup_iters)
        else:
            cos_factor = 0.5 * (1 + np.cos(np.pi * (t - warmup_iters) / (total_iters - warmup_iters)))
            lr = min_lr + (peak_lr - min_lr) * cos_factor

        # Adam update
        m = 0.9 * m + 0.1 * grad
        v = 0.999 * v + 0.001 * grad ** 2
        m_hat = m / (1 - 0.9 ** t)
        v_hat = v / (1 - 0.999 ** t)
        update = lr * m_hat / (np.sqrt(v_hat) + 1e-8)

        a = a - update
        a = np.maximum(a, 0.0)

        S = a.sum()
        if S < 1e-15:
            a = np.ones(n) / n
            continue
        a = a / S

        # Evaluate true score periodically
        if it % 100 == 0 or it < 5:
            true_score = compute_score_normalized(a)
            if true_score < best_score - 1e-10:
                best_score = true_score
                best_a = a.copy()
                no_improve = 0
                if verbose and it % 1000 == 0:
                    print(f"      it={it}, p={p:.0f}, score={best_score:.6f}, lr={lr:.6f}")
            else:
                no_improve += 1

        if no_improve > patience:
            # Restart with slight perturbation
            a = best_a * (1 + 0.05 * rng.normal(0, 1, n))
            a = np.maximum(a, 0.0)
            a = a / a.sum()
            m = np.zeros(n)
            v = np.zeros(n)
            t = 0
            no_improve = 0
            base_lr = base_lr * 0.5
            peak_lr = base_lr
            min_lr = base_lr * 0.01
            if base_lr < 1e-10:
                break

    final_score = compute_score_normalized(best_a)
    return best_a, final_score

def peak_shaving(a, time_budget, rng=None):
    """
    Peak shaving: iteratively target the maximum of the convolution.
    Uses adaptive step size based on the pressure distribution.
    """
    if rng is None:
        rng = np.random.default_rng(0)

    n = len(a)
    a = a.astype(np.float64).copy()
    S = a.sum()
    a = a / S

    best_a = a.copy()
    best_score = compute_score_normalized(a)

    start = time.monotonic()
    alpha = 0.5

    while time.monotonic() - start < time_budget:
        conv = np.convolve(a, a)
        k_star = int(np.argmax(conv))
        M = conv.max()

        # Compute pressure on each a_i from the peak
        pressure = np.zeros(n)
        for i in range(n):
            j = k_star - i
            if 0 <= j < n:
                pressure[i] = 2.0 * a[j]

        # Also add pressure from nearby peaks (smoothed)
        # Use top-3 peaks
        top_k = np.argsort(conv)[-min(3, len(conv)):]
        for k in top_k[1:]:
            weight = conv[k] / M
            for i in range(n):
                j = k - i
                if 0 <= j < n:
                    pressure[i] += weight * 2.0 * a[j]

        # Multiplicative update
        a = a * np.exp(-alpha * pressure / np.max(pressure + 1e-10))
        a = np.maximum(a, 0.0)

        S = a.sum()
        if S < 1e-15:
            break
        a = a / S

        score = compute_score_normalized(a)
        if score < best_score:
            best_score = score
            best_a = a.copy()
            alpha = min(alpha * 1.05, 2.0)
        else:
            alpha *= 0.9
            if alpha < 0.01:
                # Try random perturbation
                a = best_a * (1 + 0.05 * rng.normal(0, 1, n))
                a = np.maximum(a, 0.0)
                a = a / a.sum()
                alpha = 0.5

    return best_a, best_score


def local_search(f, rng, n_iter=3000):
    """Random local search to polish a sequence."""
    n = len(f)
    f = f.copy()
    S = f.sum()
    f = f / S
    best = f.copy()
    best_score = compute_score_normalized(f)

    for _ in range(n_iter):
        f_candidate = f.copy()
        width = max(1, int(rng.integers(1, max(2, n // 16))))
        start = int(rng.integers(0, n - width))
        scale = np.exp(rng.normal(0, 0.2))
        f_candidate[start:start + width] *= scale
        f_candidate = np.clip(f_candidate, 0, None)
        S_new = f_candidate.sum()
        if S_new < 1e-10:
            continue
        f_candidate = f_candidate / S_new
        score = compute_score_normalized(f_candidate)
        if score < best_score:
            best = f_candidate.copy()
            best_score = score
            f = f_candidate

    return best, best_score


def optimize_lp(f, time_budget, t0=0.5, rng=None):

        b_ub = np.full(2 * n - 1, M)
        bounds = [(0, None)] * n

                          method='highs', options={'presolve': True})

            break

        b = np.clip(res.x, 0, None)
        b_sum = b.sum()
        if b_sum < 1e-15:

        g = b / b_sum
        f_new = (1 - t) * f + t * g
        f_new = np.clip(f_new, 0, None)

        if S_new < 1e-10:

            continue

def freq_domain_init(n, rng, freqs=None):
    """
    Create initialization inspired by frequency domain flatness.
    Uses cosine modulation to create structured non-uniform sequences.
    """
    if freqs is None:
        # Choose random frequency components
        n_freqs = int(rng.integers(1, max(2, n // 32)))
        freqs = rng.integers(1, max(2, n // 4), size=n_freqs)

    a = np.ones(n)
    for k in freqs:
        amplitude = rng.uniform(0.1, 0.4)
        phase = rng.uniform(0, 2 * np.pi)
        a += amplitude * np.cos(2 * np.pi * k * np.arange(n) / n + phase)

    # Add some random perturbation
    a += 0.1 * rng.normal(0, 1, n)
    a = np.clip(a, 0.01, None)
    a = a / a.sum()
    return a


def optimize_combined(a, time_budget, rng, verbose=True):
    """

    a = a.astype(np.float64).copy()

    if S < 1e-15:
        return a, np.inf

        print(f"\n[{name}] n={n}, time={time_here:.0f}s, init={init_score:.6f}")

        # Run combined optimization
        f_opt, score = optimize_combined(f0, time_here, rng, verbose=True)

        if np.isfinite(score) and score < best_score_overall:
            best_score_overall = score
            best_overall = f_opt.copy()
            print(f"  ** New best: {score:.6f} **")

    # If still not at target, try LP on the best found
    remaining = deadline - time.monotonic()
    if best_overall is not None and remaining > 30 and best_score_overall > 1.5027:
        n_best = len(best_overall)
        if n_best <= 512:  # LP is too slow for large n
            print(f"\nTrying LP on best (n={n_best}, time={remaining:.0f}s)")
            t_left = min(remaining - 10, 120)
            f_lp, s_lp = optimize_lp(best_overall, t_left, t0=0.4, rng=rng)
            if np.isfinite(s_lp) and s_lp < best_score_overall:
                best_score_overall = s_lp
                best_overall = f_lp
                print(f"  LP improved: {best_score_overall:.6f}")

    # Final restarts
    remaining = deadline - time.monotonic()
    if best_overall is not None and remaining > 40:
        n_best = len(best_overall)
        n_restarts = min(3, int(remaining / 50))
        print(f"\n{ n_restarts } restarts (n={n_best})")
        for r in range(n_restarts):
            if time.monotonic() > deadline:
                break
            f0 = best_overall * (1 + 0.25 * rng.normal(0, 1, n_best))
            f0 = np.clip(f0, 0.01, None)
            f0 = f0 / f0.sum()
            t_left = deadline - time.monotonic() - 10
            if t_left < 15:
                break
            f_opt, score = optimize_combined(f0, t_left, rng, verbose=False)
            if np.isfinite(score) and score < best_score_overall:
                best_score_overall = score
                best_overall = f_opt.copy()
                print(f"  Restart {r}: {best_score_overall:.6f}")

    # Final verification
    if best_overall is not None:
        best_overall = np.clip(best_overall, 0, 1000)
        final_list = best_overall.tolist()
        final_score = evaluate_sequence(final_list)
        print(f"\nFinal verified score: {final_score:.6f}")
        print(f"Sequence length: {len(final_list)}")
        return final_list

    print("Fallback: uniform 256")
    return [1.0] * 256
