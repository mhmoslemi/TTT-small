"""
Hybrid optimization for minimizing 2*n*max(conv(a,a))/sum(a)^2.
Uses softmax-smoothed gradient descent with projection onto the simplex,
LP-based polishing, structured starting points, and multiple restarts.
"""

import numpy as np
from scipy.optimize import linprog
import time


def compute_score(f):
    """Compute score = 2*n*max(conv)/sum^2 for sequence f."""
    n = len(f)
    S = f.sum()
    if S < 1e-15:
        return np.inf
    conv = np.convolve(f, f)
    return 2.0 * n * conv.max() / (S * S)


def compute_score_normalized(f):
    """Compute score assuming sum(f) = 1: 2*n*max(conv)."""
    n = len(f)
    conv = np.convolve(f, f)
    return 2.0 * n * conv.max()


def project_simplex(v, target_sum=1.0):
    """Project v onto {x >= 0, sum(x) = target_sum}."""
    # We want to find the projection of v onto the simplex
    # This is: minimize ||x - v||^2 subject to x >= 0, sum(x) = target_sum
    # Solution: x_i = max(v_i - tau, 0) where tau is chosen so sum(x) = target_sum
    n = len(v)
    # Binary search for tau
    lo = v.min() - target_sum
    hi = v.max()
    for _ in range(100):
        tau = (lo + hi) / 2
        x = np.clip(v - tau, 0, None)
        s = x.sum()
        if s > target_sum:
            lo = tau
        else:
            hi = tau
    x = np.clip(v - tau, 0, None)
    # Adjust for numerical precision
    diff = target_sum - x.sum()
    if x.sum() > 0:
        x += diff / n
        x = np.clip(x, 0, None)
    return x

def softmax_gradient_descent(f, time_budget, T_start=50.0, T_end=1000.0, 
                               step_factor=0.5, rng=None, verbose=False):
    """
    Run softmax-smoothed gradient descent on f (sum=1).
    Returns (best_f, best_score).
    """
    if rng is None:
        rng = np.random.default_rng(0)
    
    n = len(f)
    f = f.astype(np.float64).copy()
    f = f / f.sum()
    
    best_f = f.copy()
    best_score = compute_score_normalized(f)
    
    start = time.monotonic()
    it = 0
    step_size = step_factor / (4.0 * n)  # initial step size
    patience = 0
    max_patience = 200
    
    while time.monotonic() - start < time_budget:
        elapsed = time.monotonic() - start
        # Anneal temperature
        progress = min(1.0, elapsed / time_budget)
        T = T_start * (T_end / T_start) ** progress
        
        f_new, grad = softmax_gradient_step(f, T, step_size)
        new_score = compute_score_normalized(f_new)
        
        if new_score < best_score:
            best_f = f_new.copy()
            best_score = new_score
            f = f_new
            patience = 0
            # Slightly increase step size on success
            step_size = min(step_size * 1.05, 2.0 / (4.0 * n))
        else:
            patience += 1
            step_size *= 0.8
            if patience > max_patience:
                # Try a random perturbation
                f_pert = f + rng.normal(0, 0.01, n)
                f_pert = np.clip(f_pert, 0, None)
                S = f_pert.sum()
                if S > 1e-10:
                    f_pert = f_pert / S
                    if compute_score_normalized(f_pert) < best_score:
                        f = f_pert
                        best_f = f.copy()
                        best_score = compute_score_normalized(f)
                        step_size = step_factor / (4.0 * n)
                        patience = 0
                    else:
                        break  # stuck, move on
        
        it += 1
        if verbose and it % 100 == 0:
            print(f"    SGD it={it}, T={T:.1f}, score={best_score:.6f}, elapsed={elapsed:.1f}s")
    
    return best_f, best_score


def build_conv_matrix(f):
    """Build matrix A such that A @ b = np.convolve(f, b, mode='full')."""
    n = len(f)
    A = np.zeros((2 * n - 1, n))
    for k in range(2 * n - 1):
        j_start = max(0, k - n + 1)
        j_end = min(n, k + 1)
        for j in range(j_start, j_end):
            A[k, j] = f[k - j]
    return A


def lp_polish(f, time_budget, t0=0.3, rng=None, verbose=False):
    """LP-based polishing. f should have sum=1."""
    if rng is None:
        rng = np.random.default_rng(0)
    
    n = len(f)
    f = f.astype(np.float64).copy()
    f = f / f.sum()
    
    best_f = f.copy()
    best_score = compute_score_normalized(f)
    t = t0
    
    start = time.monotonic()
    it = 0
    stall_count = 0
    
    while time.monotonic() - start < time_budget:
        conv = np.convolve(f, f)
        M = conv.max()
        
        A = build_conv_matrix(f)
        c_obj = -np.ones(n)
        b_ub = np.full(2 * n - 1, M)
        bounds = [(0, None)] * n
        
        try:
            res = linprog(c_obj, A_ub=A, b_ub=b_ub, bounds=bounds,
                         method='highs', options={'presolve': True})
        except Exception:
            break
        
        if not res.success or res.x is None:
            break
        
        b = np.clip(res.x, 0, None)
        b_sum = b.sum()
        if b_sum < 1e-15:
            break
        
        g = b / b_sum
        f_new = (1 - t) * f + t * g
        f_new = np.clip(f_new, 0, None)
        S_new = f_new.sum()
        if S_new < 1e-10:
            t *= 0.5
            continue
        f_new = f_new / S_new
        
        new_score = compute_score_normalized(f_new)
        
        if new_score < best_score - 1e-12:
            best_f = f_new.copy()
            best_score = new_score
            f = f_new
            stall_count = 0
            if verbose and it % 5 == 0:
                print(f"    LP it={it}, score={best_score:.6f}, t={t:.4f}")
        else:
            t *= 0.7
            stall_count += 1
            if stall_count > 30:
                f_cand = f + rng.normal(0, 0.02, n)
                f_cand = np.clip(f_cand, 0, None)
                S_c = f_cand.sum()
                if S_c > 1e-10:
                    f_cand /= S_c
                    if compute_score_normalized(f_cand) < best_score:
                        f = f_cand
                        best_f = f.copy()
                        best_score = compute_score_normalized(f)
                        t = t0 * 0.5
                        stall_count = 0
                if time.monotonic() - start > time_budget * 0.85:
                    break
        
        it += 1
    
    return best_f, best_score


def generate_structured_starts(n, rng):
    """Generate diverse structured starting sequences."""
    starts = []
    
    # Linear ramp (various slopes)
    for slope in [0.3, 0.5, 0.7, 1.0]:
        x = np.arange(n, dtype=np.float64)
        f = 1.0 + slope * (x / (n - 1) - 0.5) * 2
        f = np.clip(f, 0.01, None)
        f /= f.sum()
        starts.append(f.copy())
    
    # Exponential ramp
    for rate in [0.5, 1.0, 2.0]:
        x = np.arange(n, dtype=np.float64)
        f = np.exp(rate * (x / (n - 1) - 0.5) * 2)
        f /= f.sum()
        starts.append(f.copy())
    
    # Quadratic
    for width in [0.3, 0.5, 0.7]:
        x = np.arange(n, dtype=np.float64) / (n - 1) - 0.5
        f = 1.0 + (x / width) ** 2
        f = np.clip(f, 0.01, None)
        f /= f.sum()
        starts.append(f.copy())
    
    # Sine modulation
    for freq in [1, 2, 3]:

def coordinate_descent(f, rng, n_iter=5000):
    """Simple coordinate descent / random search polishing."""
    n = len(f)
    f = f.copy()
    f = f / f.sum()
    best = f.copy()
    best_score = compute_score_normalized(f)
    
    for i in range(n_iter):
        f_cand = f.copy()
        # Pick a random block to perturb
        width = max(1, int(rng.integers(1, max(2, n // 16))))
        start = int(rng.integers(0, n - width + 1))
        scale = np.exp(rng.normal(0, 0.2))
        f_cand[start:start + width] *= scale
        f_cand = np.clip(f_cand, 0, None)
        S = f_cand.sum()

        if score < best_score:

            best_score = score
            f = f_cand

"""
Hybrid optimization for minimizing 2*n*max(conv(a,a))/sum(a)^2.
Combines softmax-smoothed gradient descent with projected steps onto the simplex,
LP-based active-set polishing, structured starting points, and multi-scale search.
Key improvements over prior approach: adaptive softmax annealing, reduced LP constraints,
gradient-based line search, and diverse parametric initializations.
"""

import numpy as np
from scipy.optimize import linprog
import time


def compute_score_normalized(f):
    """Compute score = 2*n*max(conv) assuming sum(f)=1."""
    n = len(f)
    conv = np.convolve(f, f)
    return 2.0 * n * conv.max()


def compute_score_full(f):
    """Compute full score = 2*n*max(conv)/sum^2."""
    n = len(f)
    S = f.sum()
    if S < 1e-15:
        return np.inf
    conv = np.convolve(f, f)
    return 2.0 * n * conv.max() / (S * S)


def project_simplex(v, target_sum=1.0):
    """Project v onto {x >= 0, sum(x) = target_sum} using Duchi et al. algorithm."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.arange(1, n + 1, dtype=np.float64)
    cond = u - (cssv - target_sum) / rho > 0
    if not np.any(cond):
        return np.full(n, target_sum / n)
    rho_star = int(np.max(rho[cond]))
    theta = (cssv[rho_star - 1] - target_sum) / rho_star
    return np.clip(v - theta, 0, None)


def softmax_gradient(f, T):
    """
    Compute gradient of smoothed score: 2n * (1/T) * log(sum_k exp(T*conv_k)).
    Returns (gradient, smoothed_value).
    """
    n = len(f)
    conv = np.convolve(f, f)
    # Softmax weights for numerical stability
    conv_shifted = conv - conv.max()
    w = np.exp(T * conv_shifted)
    w_sum = w.sum()
    if w_sum < 1e-300:
        w = np.zeros(len(conv))
        w[np.argmax(conv)] = 1.0
    else:
        w /= w_sum
    # Smoothed value
    smoothed = (1.0 / T) * (np.log(w_sum) + T * conv.max())
    # Gradient: 4n * sum_k w[k] * f[k-j]
    # = 4n * (conv(w, f_rev))[n-1 : 2n-1]
    f_rev = f[::-1]
    wf = np.convolve(w, f_rev)
    grad = 4.0 * n * wf[n - 1:2 * n - 1]
    return grad, 2.0 * n * smoothed

    S = f.sum()
    if S < 1e-15:
        return f, np.inf
    f = f / S

    best_f = f.copy()
    best_score = compute_score_normalized(f)
    score = best_score

    start = time.monotonic()
    it = 0
    step_size = step_init / (4.0 * n)
    stall = 0
    max_stall = 150

    while time.monotonic() - start < time_budget:
        elapsed = time.monotonic() - start
        progress = min(1.0, elapsed / max(time_budget, 1e-6))
        # Exponential annealing of temperature
        T = T_start * (T_end / T_start) ** (progress ** 0.7)

        grad, smoothed_val = softmax_gradient(f, T)

        # Adaptive step size with backtracking
        alpha = step_size
        for _ in range(10):
            f_new = f - alpha * grad
            f_new = project_simplex(f_new, 1.0)
            new_score = compute_score_normalized(f_new)
            # Accept if better, or with small probability (annealing)
            if new_score < score or (rng.random() < accept_worse_prob * max(0, 1.0 - progress)):
                break
            alpha *= 0.5

        if new_score < best_score:
            best_f = f_new.copy()
            best_score = new_score
        f = f_new
        score = new_score

        if new_score < best_score + 1e-10:
            stall = 0
            step_size = min(step_size * 1.02, 2.0 / (4.0 * n))
        else:
            stall += 1
            step_size *= 0.85

        if stall > max_stall:
            # Perturb to escape
            noise_scale = 0.01 * (1.0 + progress)
            f_pert = f + rng.normal(0, noise_scale, n)
            f_pert = np.clip(f_pert, 0, None)
            S_p = f_pert.sum()
            if S_p > 1e-10:
                f_pert /= S_p
                s_pert = compute_score_normalized(f_pert)
                if s_pert < best_score:
                    f = f_pert
                    best_f = f.copy()
                    best_score = s_pert
                    score = s_pert
                    step_size = step_init / (4.0 * n)
                    stall = 0
                else:
                    break

        it += 1
        if verbose and it % 200 == 0:
            print(f"      SGD it={it}, T={T:.1f}, score={best_score:.6f}, t={elapsed:.1f}s")

    return best_f, best_score


def build_conv_matrix_sparse(f, M, threshold=0.7):
    """Build reduced LP matrix: only include constraints where conv(f,f) > threshold*M."""
    n = len(f)
    conv_f = np.convolve(f, f)
    active = conv_f > threshold * M
    active_idx = np.where(active)[0]

    A = np.zeros((len(active_idx), n))
    b_ub = np.array([M] * len(active_idx))

    for row, k in enumerate(active_idx):
        j_start = max(0, k - n + 1)
        j_end = min(n, k + 1)
        for j in range(j_start, j_end):
            A[row, j] = f[k - j]

    return A, b_ub


def lp_polish(f, time_budget, t0=0.25, rng=None, verbose=False, threshold=0.7):
    """LP-based polishing with reduced constraint set. f should have sum=1."""
    if rng is None:
        rng = np.random.default_rng(0)
    n = len(f)
    f = f.astype(np.float64).copy()
    f = f / f.sum()

    best_f = f.copy()
    best_score = compute_score_normalized(f)
    t = t0

    start = time.monotonic()
    it = 0
    stall = 0

    while time.monotonic() - start < time_budget:
        conv = np.convolve(f, f)
        M = conv.max()

        # Use reduced constraints for speed
        if n > 512:
            A, b_ub = build_conv_matrix_sparse(f, M, threshold=threshold)
        else:
            A = np.zeros((2 * n - 1, n))
            for k in range(2 * n - 1):
                j_start = max(0, k - n + 1)
                j_end = min(n, k + 1)
                for j in range(j_start, j_end):
                    A[k, j] = f[k - j]
            b_ub = np.full(2 * n - 1, M)

        if A.shape[0] == 0:
            break

        c_obj = -np.ones(n)
        bounds = [(0, None)] * n

        try:
            res = linprog(c_obj, A_ub=A, b_ub=b_ub, bounds=bounds,
                           method='highs', options={'presolve': True})
        except Exception:
            break

        if not res.success or res.x is None:
            break

        b = np.clip(res.x, 0, None)
        b_sum = b.sum()
        if b_sum < 1e-15:
            break

        g = b / b_sum
        f_new = (1 - t) * f + t * g
        f_new = np.clip(f_new, 0, None)
        S_new = f_new.sum()
        if S_new < 1e-10:
            t *= 0.5
            continue
        f_new = f_new / S_new

        new_score = compute_score_normalized(f_new)

        if new_score < best_score - 1e-13:
            best_f = f_new.copy()
            best_score = new_score
            f = f_new
            stall = 0
            threshold = max(0.5, threshold - 0.02)  # gradually include more constraints
            if verbose and it % 5 == 0:
                print(f"      LP it={it}, score={best_score:.6f}, t={t:.3f}, ncons={A.shape[0]}")
        else:
            t *= 0.7
            stall += 1
            if stall > 40:
                # Escape
                f_cand = f + rng.normal(0, 0.015, n)
                f_cand = np.clip(f_cand, 0, None)
                S_c = f_cand.sum()
                if S_c > 1e-10:
                    f_cand /= S_c
                    s_c = compute_score_normalized(f_cand)
                    if s_c < best_score:
                        f = f_cand
                        best_f = f.copy()
                        best_score = s_c
                        t = t0 * 0.4
                        stall = 0
                        threshold = 0.7
                if time.monotonic() - start > time_budget * 0.85:
                    break

        it += 1

    return best_f, best_score


def block_local_search(f, rng, n_iter=3000, max_width_frac=1/8):
    """Block-based local search for polishing."""
    n = len(f)
    f = f.copy()
    f /
    = f / f.sum()
    best = f.copy()
    best_score = compute_score_normalized(f)

    for i in range(n_iter):
        f_cand = f.copy()
        width = max(1, int(rng.integers(1, max(2, n * max_width_frac))))
        start = int(rng.integers(0, n - width + 1))
        scale = np.exp(rng.normal(0, 0.15))
        f_cand[start:start + width] *= scale
        f_cand = np.clip(f_cand, 0, None)
        S = f_cand.sum()
        if S < 1e-10:
            continue
        f_cand /= S
        score = compute_score_normalized(f_cand)
        if score < best_score:
            best = f_cand.copy()
            best_score = score
            f = f_cand

    return best, best_score


def generate_starts(n, rng, parent=None):
    """Generate diverse starting sequences of length n."""
    starts = []
    x = np.arange(n, dtype=np.float64)
    x_norm = x / (n - 1) if n > 1 else x

    # Linear ramps (asymmetric)
    for slope in [0.3, 0.5, 0.8, 1.2, 1.5, 2.0]:
        f = 1.0 + slope * (x_norm - 0.5) * 2
        f = np.clip(f, 0.005, None)
        f /= f.sum()
        starts.append(f.copy())

    # Exponential ramps
    for rate in [0.5, 1.0, 1.5, 2.5, 4.0]:
        f = np.exp(rate * (x_norm - 0.5) * 2)
        f /= f.sum()
        starts.append(f.copy())

    # Asymmetric Gaussian bump
    for center in [0.25, 0.4, 0.6]:
        for width in [0.1, 0.2, 0.3]:
            f = np.exp(-((x_norm - center) ** 2) / (2 * width ** 2))
            f = f + 0.01
            f /= f.sum()
            starts.append(f.copy())

    # Power law
    for p in [0.5, 1.0, 2.0, 3.0]:
        f = (x_norm + 0.01) ** p
        f /= f.sum()
        starts.append(f.copy())

    # Sine-modulated
    for freq in [1, 2, 3, 5]:
        for amp in [0.2, 0.4, 0.6]:
            f = 1.0 + amp * np.sin(2 * np.pi * freq * x_norm)
            f = np.clip(f, 0.01, None)
            f /= f.sum()
            starts.append(f.copy())

    # Smoothed random
    for _ in range(8):
        f = rng.random(n)
        # Gaussian smoothing
        kernel_size = max(3, n // 32)
        kernel = np.exp(-0.5 * (np.arange(kernel_size) - kernel_size // 2) ** 2 / (kernel_size // 4 + 1))
        kernel /= kernel.sum()
        f = np.convolve(f, kernel, mode='same')
        f = np.clip(f, 0.005, None)
        f /= f.sum()
        starts.append(f.copy())

    # Pure random
    for _ in range(4):
        f = rng.random(n) + 0.01
        f /= f.sum()
        starts.append(f.copy())

    # Uniform
    f = np.ones(n) / n
    starts.append(f.copy())

    # Parent if available
    if parent is not None and len(parent) == n:
        p = np.clip(parent, 0, None).astype(np.float64)
        S = p.sum()
        if S > 1e-10:
            p /= S
            starts.insert(0, p)
            # Also add perturbed versions
            for scale in [0.1, 0.2, 0.3, 0.5]:
                p_pert = p * (1 + scale * (rng.random(n) - 0.5))
                p_pert = np.clip(p_pert, 0.001, None)
                p_pert /= p_pert.sum()
                starts.append(p_pert)

    return starts


def optimize_one_start(f0, time_budget, rng, verbose=False):
    """Run the full optimization pipeline on a single starting point."""
    n = len(f0)
    T_total = time_budget

    # Phase 1: Softmax GD (exploration) - 35% of time
    t1 = T_total * 0.35
    t1_end = time.monotonic() + t1
    f, s = softmax_sgd(f0, t1, T_start=20, T_end=500,
                       step_init=0.8, rng=rng, verbose=verbose)

    # Phase 2: LP polishing - 35% of time
    t2 = T_total * 0.35
    t2_end = time.monotonic() + t2
    if time.monotonic() < t2_end:
        remaining_lp = min(t2, t2_end - time.monotonic())
        if remaining_lp > 8:
            f, s_lp = lp_polish(f, remaining_lp, t0=0.25, rng=rng, verbose=verbose)
            s = min(s, s_lp)

    # Phase 3: Fine-grained SGD (exploitation) - 20% of time
    t3 = T_total * 0.20
    t3_end = time.monotonic() + t3
    if time.monotonic() < t3_end:
        remaining_sgd = min(t3, t3_end - time.monotonic())
        if remaining_sgd > 5:
            f, s3 = softmax_sgd(f, remaining_sgd, T_start=300, T_end=3000,
                                step_init=0.15, rng=rng, verbose=verbose)
            s = min(s, s3)

    # Phase 4: Block local search - 10% of time
    t4 = T_total * 0.10
    if time.monotonic() < time.monotonic() + t4:
        n_iter = max(500, int(t4 * 1500))
        f, s4 = block_local_search(f, rng, n_iter=n_iter)
        s = min(s, s4)

    return f, s


def propose_candidate(seed=42, budget_s=1500, **kwargs):
    """
    Main search function. Multi-phase hybrid optimizer:
    - Structured + random diverse starts
    - Softmax-smoothed gradient descent with adaptive annealing
    - LP-based active-set polishing with reduced constraints
    - Block local search for final polishing
    Multiple sequence lengths with adaptive time allocation.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + budget_s - 20.0

    best_overall = None
    best_score_overall = np.inf

    # Get parent sequence
    parent = None
    try:
        parent = np.array(height_sequence_1, dtype=np.float64)
        if parent.ndim != 1 or len(parent) == 0 or not np.isfinite(parent).all():
            parent = None
        else:
            parent = np.clip(parent, 0, 1000)
    except Exception:
        parent = None

    print(f"Parent: {'yes' if parent is not None else 'no'}, "
          f"len={len(parent) if parent is not None else 0}")
    if parent is not None:
        print(f"  Parent score: {compute_score_full(parent):.6f}")
    print(f"Budget: {budget_s}s, Target: 1.5027")

    # Sequence lengths to try (larger = more DOF, slower per step)
    lengths = [256, 512, 1024, 2048]

    # If parent has a different length, add it
    if parent is not None:
        lp = len(parent)
        if lp not in lengths and 128 <= lp <= 4096:
            lengths.insert(0, lp)

    # Time allocation weights (favor larger n for more potential)
    n_lengths = len(lengths)
    weights = np.array([1.0, 1.3, 1.8, 2.2, 2.5][:n_lengths])
    weights /= weights.sum()
    available = budget_s - 80.0  # reserve 80s for final polishing

    # Also reserve time for the parent if available
    if parent is not None:
        parent_time = min(available * 0.3, 300)
        available -= parent_time
        # Optimize parent first
        print(f"\n=== Optimizing parent (n={len(parent)}, t={parent_time:.0f}s) ===")
        f0 = parent.copy()
        S = f0.sum()
        f0 = np.clip(f0, 0, None)
        f0 = f0 / f0.sum()
        t_end = time.monotonic() + parent_time
        f_opt, s_opt = optimize_one_start(f0, parent_time, rng, verbose=True)
        if np.isfinite(s_opt):
            best_score_overall = s_opt
            best_overall = f_opt.copy()
            print(f"  Parent optimized: {s_opt:.6f}")

    for i, n in enumerate(lengths):
        if time.monotonic() > deadline:
            break
        remaining_total = deadline - time.monotonic()
        if remaining_total < 30:
            break

        t_alloc = available * weights[i]
        t_alloc = min(t_alloc, remaining_total - 15)
        if t_alloc < 20:
            break

        print(f"\n=== n={n}, time={t_alloc:.0f}s ===")

        # Generate starts
        starts = generate_starts(n, rng, parent=parent)

        # Score all starts quickly
        scored = []
        for f0 in starts:
            s = compute_score_normalized(f0)
            scored.append((s, f0))
        scored.sort(key=lambda x: x[0])

        # Keep top starts
        n_keep = min(len(scored), max(4, len(scored) // 3))
        scored = scored[:n_keep]
        print(f"  {len(scored)} starts, top score: {scored[0][0]:.6f}")

        # If we already have a good solution, only keep starts that might beat it
        if best_score_overall < np.inf:
            scored = [(s, f) for s, f in scored if s < best_score_overall * 1.05]
            if not scored:
                scored = scored[:1]  # at least try the best one

        # Allocate time among kept starts (give more to better ones)
        n_this = len(scored)
        # Inverse score weighting for time allocation
        inv_scores = [1.0 / max(s, 0.001) for s, _ in scored]
        total_inv = sum(inv_scores)
        times = [t_alloc * iv / total_inv for iv in inv_scores]

        best_for_n = None
        best_score_n = best_score_overall

        for rank, (init_s, f0) in enumerate(scored):
            if time.monotonic() > deadline - 10:
                break
            t_this = min(times[rank], deadline - time.monotonic() - 10)
            if t_this < 8:
                break

            f_opt, s_opt = optimize_one_start(f0, t_this, rng, verbose=(rank == 0))

            if np.isfinite(s_opt) and s_opt < best_score_n:
                best_score_n = s_opt
                best_for_n = f_opt.copy()
                if s_opt < best_score_overall:
                    best_score_overall = s_opt
                    best_overall = f_opt.copy()
                    print(f"  ** New global best: {s_opt:.6f} (n={n}, start={rank}) **")

            if rank == 0:
                print(f"  [{n}] start={init_s:.6f} -> {s_opt:.6f}")

        # If we found something good for this n, do one more focused pass
        if best_for_n is not None and best_score_n < best_score_overall + 0.01:
            if time.monotonic() < deadline - 15:
                t_extra = min(60, deadline - time.monotonic() - 10)
                if t_extra > 10:
                    f_final, s_final = optimize_one_start(best_for_n, t_extra, rng, verbose=False)
                    if np.isfinite(s_final) and s_final < best_score_overall:
                        best_score_overall = s_final
                        best_overall = f_final.copy()
                        print(f"  ** Extra pass improved: {s_final:.6f} **")

    # Final polishing of the global best
    if best_overall is not None and time.monotonic() < deadline - 5:
        t_final = min(60, deadline - time.monotonic() - 5)
        print(f"\n=== Final polishing (t={t_final:.0f}s) ===")

        # LP polish
        t_lp = t_final * 0.5
        if t_lp > 8:
            best_overall, s = lp_polish(best_overall, t_lp, t0=0.15, rng=rng, verbose=True)

        # SGD
        t_sgd = t_final * 0.3
        if t_sgd > 3 and time.monotonic() < deadline - 3:
            best_overall, s = softmax_sgd(best_overall, t_sgd, T_start=1000,
                                           T_end=10000, step_init=0.05, rng=rng)

        # Local search
        t_cd = max(5, t_final * 0.2)
        n_iter = max(200, int(t_cd * 2000))
        best_overall, s = block_local_search(best_overall, rng, n_iter=n_iter)

        final_internal = compute_score_normalized(best_overall)
        print(f"  Final internal score: {final_internal:.6f}")

    # Verify with evaluate_sequence and return
    if best_overall is not None:
        best_overall = np.clip(best_overall, 0, 1000)
        final_list = best_overall.tolist()
        final_score = evaluate_sequence(final_list)
        print(f"\n{'='*50}")
        print(f"FINAL VERIFIED SCORE: {final_score:.6f}")
        print(f"Sequence length: {len(final_list)}")
        print(f"{'='*50}")
        return final_list

    # Fallback
    print("Fallback: uniform sequence n=256")
    return [1.0] * 256
