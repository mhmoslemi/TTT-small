    return None


def smooth_gradient_step(f, t):
    """
    Compute the smooth gradient of soft-max(conv(f,f)) w.r.t. f.
    Returns gradient vector of same shape as f.
    """
    n = len(f)
    conv = np.convolve(f, f)
    
    # Softmax weights (temperature t)
    conv_shifted = conv - np.max(conv)
    weights = np.exp(t * conv_shifted)
    w_sum = weights.sum()
    if w_sum < 1e-300:
        weights = np.zeros_like(conv)
        weights[conv.argmax()] = 1.0
    else:
        weights = weights / w_sum
    
    # Gradient: grad[i] = 2 * sum_k weights[k] * f[k-i] (for valid k-i)
    grad = np.zeros(n)
    # Only process lags with significant weight
    threshold = 1e-8
    active = np.where(weights > threshold)[0]
    
    for k in active:
        w_k = weights[k]
        i_min = max(0, k - n + 1)
        i_max = min(k, n - 1)
        # f[k-i] for i in [i_min, i_max]
        # When i = i_min: k-i = k-i_min
        # When i = i_max: k-i = k-i_max
        # So we need f[k-i_max : k-i_min+1] reversed
        segment = f[k - i_max: k - i_min + 1][::-1]
        grad[i_min:i_max + 1] += 2.0 * w_k * segment
    
    return grad, conv, weights


def project_normalize(f):
    """Project f to non-negative and normalize to sum 1."""
    f = np.clip(f, 0.0, None)
    S = np.sum(f)
    if S < 1e-15:
        return np.ones(len(f)) / len(f)
    return f / S


def smooth_grad_descent(f_init, deadline, t_init=50.0, t_min=1.0, eta_init=0.3,
                        momentum_init=0.3, anneal_every=200, verbose_prefix=""):
    """
    Smooth gradient descent with momentum and temperature annealing.
    f is maintained at sum=1 normalization.
    """
    f = project_normalize(f_init.copy().astype(np.float64))
    n = len(f)
    best_f = f.copy()
    best_score = compute_score(f)
    
    velocity = np.zeros(n)
    eta = eta_init
    momentum = momentum_init
    t = t_init
    no_improve = 0
    it = 0
    best_it = 0
    
    while time.monotonic() < deadline:
        grad, conv, weights = smooth_gradient_step(f, t)
        M = conv.max()
        
        if M < 1e-15:
            break
        
        # Momentum update
        velocity = momentum * velocity - eta * grad
        f_new = f + velocity
        
        # Project and normalize
        f_new = project_normalize(f_new)
        
        score = compute_score(f_new)
        it += 1
        
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_f = f_new.copy()
            best_it = it
            f = f_new
            no_improve = 0
            if best_it % 2000 == 0:
                elapsed_in = time.monotonic() - (deadline - (deadline - time.monotonic()))
                remaining = deadline - time.monotonic()
                print(f"  {verbose_prefix}it={it} score={best_score:.8f} t={t:.1f} eta={eta:.4f} rem={remaining:.0f}s")
        else:
            no_improve += 1
            eta *= 0.95
            if no_improve > 500:
                # Reset with annealing
                t = max(t_min, t * 0.6)
                eta = max(0.05, eta_init * 0.5)
                momentum = max(0.1, momentum_init * 0.7)
                velocity = np.zeros(n)
                no_improve = 0
                if t <= t_min and eta < 0.05:
                    break
        
        if it % anneal_every == 0:
            t = max(t_min, t * 0.97)
            momentum = min(0.9, momentum * 1.01)
    
    return best_f, best_score


def lp_single_step(f):
    """Perform one LP-based direction step. Returns (new_f, new_score)."""
    n = len(f)
    conv = np.convolve(f, f)
    M = float(conv.max())
    S = float(np.sum(f))
    if M < 1e-15 or S < 0.01:
        return f, compute_score(f)
    
    g = solve_lp_direction(f, M)
    if g is None:
        return f, compute_score(f)
    
    sum_g = float(np.sum(g))
    if sum_g < 1e-15:
        return f, compute_score(f)
    
    g = g * (S / sum_g)
    
    best_f = f.copy()
    best_score = compute_score(f)
    
    for step in [0.3, 0.5, 0.7, 1.0]:
        new_f = (1.0 - step) * f + step * g
        new_f = np.clip(new_f, 0.0, 1000.0)
        new_f = project_normalize(new_f)
        score = compute_score(new_f)
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_f = new_f.copy()
    
    return best_f, best_score


def lp_polish(f, deadline, max_iterations=20):
    """Run multiple LP-based steps."""
    f = project_normalize(f.copy())
    best_f = f.copy()
    best_score = compute_score(f)
    it = 0
    
    while time.monotonic() < deadline and it < max_iterations:
        new_f, new_score = lp_single_step(f)
        if new_score < best_score - 1e-10:
            f = new_f
            best_f = f.copy()
            best_score = new_score
            it += 1
        else:
            break
    
    return best_f, best_score


def random_perturb(f, strength=0.1):
    """Add random perturbation and reproject."""
    n = len(f)
    noise = np.random.default_rng().standard_normal(n) * strength
    f_new = f + noise
    return project_normalize(f_new)


def propose_candidate(budget_s=1500, **kwargs):
    t0 = time.monotonic()
    deadline = t0 + max(0.0, float(budget_s) - 5.0)
    rng = np.random.default_rng(42)
    
    best_sequence = None
    best_score = float('inf')
    
    # Load parent
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64).copy()
        parent = np.clip(parent, 0.0, 1000.0)
        if parent.ndim == 1 and parent.size > 0 and np.all(np.isfinite(parent)):
            score = evaluate_sequence(parent.tolist())
            if np.isfinite(score):
                best_score = score
                best_sequence = parent.copy()
                print(f"Parent n={parent.size}: score={score:.8f}")
    except (NameError, Exception) as e:
        print(f"Could not load parent: {e}")
    
    # === Phase 1: Polish parent (0-250s) ===
    if best_sequence is not None and time.monotonic() < deadline - 100:
        print("\n=== Phase 1: Gradient descent on parent ===")
        p1_deadline = min(deadline - 100, t0 + 250)
        
        # Multiple t values for diversity
        for t_init in [100.0, 50.0, 20.0]:
            if time.monotonic() > p1_deadline:
                break
            sub_deadline = min(p1_deadline, time.monotonic() + 60)
            f, s = smooth_grad_descent(best_sequence, sub_deadline,
                                       t_init=t_init, t_min=t_init * 0.05,
                                       eta_init=0.3, momentum_init=0.3,
                                       verbose_prefix=f"[P1 t={t_init}]")
            if s < best_score:
                best_score = s
                best_sequence = f.copy()
                print(f"  t_init={t_init}: {best_score:.8f}")
        
        # LP polish
        if time.monotonic() < p1_deadline - 5:
            f, s = lp_polish(best_sequence, p1_deadline, max_iterations=10)
            if s < best_score:
                best_score = s
                best_sequence = f.copy()
                print(f"  After LP: {best_score:.8f}")
    
    # === Phase 2: Try different lengths (250s - 700s) ===
    print("\n=== Phase 2: Multiple lengths ===")
    lengths_to_try = [256, 384, 512, 640, 768, 1024]
    
    for n in lengths_to_try:
        if time.monotonic() > deadline - 200:
            break
        
        print(f"\n  --- n={n} ---")
        n_deadline = min(deadline - 200, time.monotonic() + 60)
        
        # Generate initializations
        inits = []
        
        # Uniform
        inits.append(('uniform', np.ones(n)))
        
        # Cosine
        idx = np.arange(n, dtype=np.float64)
        cos_t = (1.0 + np.cos(np.pi * idx / max(n - 1, 1))) / 2.0
        inits.append(('cosine', cos_t))
        
        # Symmetric random
        half_n = n // 2
        half = rng.uniform(0.2, 1.0, size=half_n)
        sym = np.concatenate([half, half[::-1]])[:n]
        inits.append(('sym_rand', sym))
        
        # Bimodal
        center = (n - 1) / 2.0
        two_peak = (np.exp(-0.5 * ((idx - n * 0.25) / (n * 0.1)) ** 2) +
                    np.exp(-0.5 * ((idx - n * 0.75) / (n * 0.1)) ** 2))
        inits.append(('bimodal', two_peak))
        
        # Parent interpolation
        if best_sequence is not None:
            pn = len(best_sequence)
            if n != pn:
                x_old = np.linspace(0, 1, pn)
                x_new = np.linspace(0, 1, n)
                from scipy.interpolate import interp1d
                f_interp = interp1d(x_old, best_sequence, kind='cubic')(x_new)
                f_interp = np.clip(f_interp, 0, 1000)
                inits.append(('parent', f_interp))
            else:
                inits.append(('parent', best_sequence.copy()))
        
        # Random with structure (low-pass filtered noise)
        white = rng.standard_normal(n)
        # Simple moving average filter
        kernel_size = max(3, n // 64)
        kernel = np.ones(kernel_size) / kernel_size
        filtered = np.convolve(white, kernel, mode='same')
        filtered = np.clip(filtered, 0.01, None)
        inits.append(('filtered_noise', filtered))
        
        for label, f_init in inits:
            if time.monotonic() > n_deadline:
                break
            
            f_init = np.clip(f_init.astype(np.float64), 0, 1000)
            if np.sum(f_init) < 0.01:
                continue
            f_init = project_normalize(f_init)
            
            score = compute_score(f_init)
            if not np.isfinite(score):
                continue
            
            if score < best_score:
                best_score = score
                best_sequence = f_init.copy()
            
            print(f"    n={n} {label}: init={score:.6f}", end="")
            
            # Run gradient descent
            sub_dl = min(n_deadline, time.monotonic() + 25)
            f, s = smooth_grad_descent(f_init, sub_dl,
                                       t_init=30.0, t_min=1.0,
                                       eta_init=0.3, momentum_init=0.5,
                                       verbose_prefix="")
            if s < best_score:
                best_score = s
                best_sequence = f.copy()
                print(f" -> {s:.6f}")
            else:
                print(f" -> no impr (best={score:.6f})")
        
        # LP polish on best for this length
        if time.monotonic() < n_deadline - 3:
            # Use best sequence restricted/padded to this length
            if len(best_sequence) == n:
                f_lp, s_lp = lp_polish(best_sequence, n_deadline, max_iterations=5)
                if s_lp < best_score:
                    best_score = s_lp
                    best_sequence = f_lp.copy()
    
    # === Phase 3: Intensive refinement on best (700s - 1400s) ===
    if best_sequence is not None and time.monotonic() < deadline - 20:
        print(f"\n=== Phase 3: Intensive refinement (n={len(best_sequence)}, score={best_score:.8f}) ===")
        p3_deadline = min(deadline - 10, t0 + budget_s - 15)
        
        # Alternating: gradient descent + LP + random restarts
        cycle_time = 30  # seconds per cycle
        cycle = 0
        
        while time.monotonic() < p3_deadline:
            cycle += 1
            cycle_start = time.monotonic()
            
            # Step 1: Fine gradient descent
            sub_dl = min(p3_deadline, cycle_start + cycle_time * 0.5)
            t_val = max(5.0, 50.0 / cycle)
            f, s = smooth_grad_descent(best_sequence, sub_dl,
                                       t_init=t_val, t_min=max(1.0, t_val * 0.1),
                                       eta_init=0.15, momentum_init=0.5,
                                       verbose_prefix=f"[P3 cyc{cycle}]")
            if s < best_score:
                best_score = s
                best_sequence = f.copy()
            
            # Step 2: LP step
            if time.monotonic() < p3_deadline - 5:
                f_lp, s_lp = lp_single_step(best_sequence)
                if s_lp < best_score:
                    best_score = s_lp
                    best_sequence = f_lp.copy()
                    print(f"    cycle {cycle}: LP -> {best_score:.8f}")
            
            # Step 3: Random perturb + re-optimize (every 3 cycles)
            if cycle % 3 == 0 and time.monotonic() < p3_deadline - 10:
                f_pert = random_perturb(best_sequence, strength=0.05)
                sub_dl2 = min(p3_deadline - 5, time.monotonic() + 15)
                f_r, s_r = smooth_grad_descent(f_pert, sub_dl2,
                                               t_init=10.0, t_min=1.0,
                                               eta_init=0.2, momentum_init=0.3,
                                               verbose_prefix="")
                if s_r < best_score:
                    best_score = s_r
                    best_sequence = f_r.copy()
                    print(f"    cycle {cycle}: perturbed -> {best_score:.8f}")
            
            if cycle % 5 == 0:
                remaining = p3_deadline - time.monotonic()
                print(f"    cycle {cycle}: best={best_score:.8f}, remaining={remaining:.0f}s")
    
    # === Phase 4: Final polish ===
    if best_sequence is not None and time.monotonic() < deadline - 3:
        print(f"\n=== Phase 4: Final polish ===")
        # Very fine gradient
        f, s = smooth_grad_descent(best_sequence, deadline - 2,
                                   t_init=5.0, t_min=1.0,
                                   eta_init=0.05, momentum_init=0.7,
                                   verbose_prefix="[final]")
        if s < best_score:
            best_score = s
            best_sequence = f.copy()
        
        # Final LP
        if time.monotonic() < deadline - 1:
            f_lp, s_lp = lp_single_step(best_sequence)
            if s_lp < best_score:
                best_score = s_lp
                best_sequence = f_lp.copy()
    
    # Verify with evaluate_sequence
    if best_sequence is not None:
        final_score = evaluate_sequence(best_sequence.tolist())
        if not np.isfinite(final_score):
            print(f"WARNING: final sequence invalid (score={final_score}), using fallback")
            best_sequence = np.ones(128)
            best_score = 2.0
        else:
            print(f"\nFINAL VERIFIED: n={len(best_sequence)}, score={final_score:.8f}")
            best_score = final_score
    
    if best_sequence is None:
        return [1.0] * 128
    
    elapsed = time.monotonic() - t0
    print(f"DONE: n={len(best_sequence)}, score={best_score:.8f}, time={elapsed:.1f}s")
    return [float(max(0.0, min(1000.0, x))) for x in best_sequence]
