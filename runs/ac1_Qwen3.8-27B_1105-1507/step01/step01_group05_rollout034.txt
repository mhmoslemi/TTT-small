def simplex_project(v):
    """Project v onto the probability simplex."""
    n = len(v)
    if n == 1:

Wait, I need to double-check my gradient computation. Let me verify:

`conv(k) = sum_j a[j] * a[k-j]` for valid j (0≤j<n, 0≤k-j<n)

The gradient of conv(k) with respect to a[m]:
- If 0≤m<n and 0≤k-m<n: the term a[m]*a[k-m] is in the sum, contributing a[k-m] to d/d a[m]
- Also, if 0≤m<n and 0≤k-m<n: the term a[k-m]*a[m] is also in the sum (when j=k-m), contributing a[m] to d/d a[m]
- But wait, these are the SAME term! When j=m, we get a[m]*a[k-m]. When j=k-m, we get a[k-m]*a[m]. These are the same product, but they appear as two separate terms in the sum (unless m = k-m).

So: d/d a[m] of conv(k) = a[k-m] + a[k-m] = 2*a[k-m] (when m ≠ k-m and both valid)
Or: d/d a[m] of conv(k) = 2*a[m] (when m = k-m, i.e., k = 2m)

In both cases: 2*a[k-m] (since when k=2m, k-m=m).

So the gradient of conv(k) is a vector g where g[m] = 2*a[k-m] for 0≤m<n and 0≤k-m<n, and 0 otherwise.

Now, the soft-max gradient:
`d/d a[m] soft_max(conv, t) = sum_k p[k] * 2*a[k-m]` (for valid k-m)
= 2 * sum_{k: 0≤k-m<2n-1} p[k] * a[k-m]
Let j = k-m: = 2 * sum_{j: 0≤j<n} p[m+j] * a[j] (since k = m+j, and 0≤k<2n-1 means 0≤m+j<2n-1, and since m,j < n, m+j < 2n-1 always)

So: grad[m] = 2 * sum_{j=0}^{n-1} a[j] * p[m+j]

This is the cross-correlation of a (length n) and p (length 2n-1) evaluated at lag m.

Using convolution: if we reverse a to get a_rev[j] = a[n-1-j], then:
(a_rev * p)[k] = sum_j a_rev[j] * p[k-j] = sum_j a[n-1-j] * p[k-j]

Let i = n-1-j: = sum_i a[i] * p[k-n+1+i]

We want sum_j a[j] * p[m+j]. Set k-n+1 = m, so k = m+n-1:
(a_rev * p)[m+n-1] = sum_j a[j] * p[m+j] ✓

So: grad = 2 * convolve(a[::-1], p, mode='full')[n-1:2*n-1]

def simplex_project(v):
    """Project v onto the probability simplex {x: x>=0, sum(x)=1}."""
    n = len(v)
    if n == 1:
        return np.array([max(0.0, float(v[0]))])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1.0)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def build_conv_matrix(f, k_list):

        je = min(n, k + 1)

    # Phase 2: Gradient (15% of time)
    remaining = t_budget - (time.monotonic() - time.monotonic())  # placeholder
    t2 = min(15.0, (t_budget - t1) * 0.3)
    if t2 > 5:
        f2, s2 = phase_grad(f, t2, rng)
        if s2 < best_score:
            best_f = f2.copy()
            best_score = s2
            f = f2
            if verbose:
                print(f"    After grad: {best_score:.6f}")

    # Phase 3: LP refinement (55% of time)
    elapsed = t1 + t2
    t3 = max(10.0, t_budget - elapsed - 10)
    if t3 > 10:
        f3, s3 = phase_lp(f, t3, rng, verbose=verbose)
        if s3 < best_score:
            best_f = f3.copy()
            best_score = s3
            f = f3
            if verbose:
                print(f"    After LP: {best_score:.6f}")

    # Phase 4: Local search (remaining)
    elapsed_all = t1 + t2 + t3
    t4 = max(3.0, t_budget - elapsed_all)
    if t4 > 3:
        n_iter = min(8000, int(t4 * 300))
        f4, s4 = phase_local(f, rng, n_iter)
        if s4 < best_score:
            best_f = f4.copy()
            best_score = s4
            if verbose:
                print(f"    After local: {best_score:.6f}")

    return best_f, best_score


def make_init(n, init_type, rng):
    """Create an initial sequence of length n with given type."""
    if init_type == 'uniform':
        f0 = np.ones(n) * (1.0 + 0.08 * rng.standard_normal(n))
        f0 = np.clip(f0, 0.01, None)
    elif init_type == 'random':
        f0 = np.abs(rng.standard_normal(n)) + 0.05
    elif init_type == 'bimodal':
        x = np.linspace(0, 1, n)
        f0 = 1.0 / (1.0 + 8.0 * np.exp(-((x - 0.5) ** 2) / 0.03))
        f0 = f0 * (1.0 + 0.1 * rng.standard_normal(n))
        f0 = np.clip(f0, 0.01, None)
    elif init_type == 'triangle':
        x = np.linspace(0, 1, n)
        f0 = np.abs(1.0 - 2.0 * x)
        f0 = f0 * (1.0 + 0.1 * rng.standard_normal(n))
        f0 = np.clip(f0, 0.005, None)
    elif init_type == 'plateau':
        x = np.linspace(0, 1, n)
        f0 = np.where((x > 0.2) & (x < 0.8), 1.0, 0.3)
        f0 = f0 * (1.0 + 0.1 * rng.standard_normal(n))
        f0 = np.clip(f0, 0.01, None)
    elif init_type == 'alternating':
        f0 = np.ones(n)
        f0[::2] = 0.4
        f0 = f0 * (1.0 + 0.1 * rng.standard_normal(n))
        f0 = np.clip(f0, 0.01, None)
    else:
        f0 = np.ones(n)

    return f0


def propose_candidate(seed=42, budget_s=1500):
    """Main search function. Returns best sequence found."""
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + budget_s - 20.0

    best_overall = None
    best_score_overall = np.inf

    # Get parent sequence if available
    parent = None
    try:
        p = np.array(height_sequence_1, dtype=np.float64)
        if p.ndim == 1 and len(p) > 0 and np.isfinite(p).all() and (p >= 0).all():
            parent = p.copy()
    except Exception:
        pass

    print(f"Budget: {budget_s}s, deadline in {budget_s - 20}s")
    if parent is not None:
        print(f"Parent: n={len(parent)}, score={compute_score_full(parent):.6f}")

    # Build configuration list
    configs = []

    n_options = [512, 1024, 2048]
    if parent is not None and 128 <= len(parent) <= 4096:
        np_ = len(parent)
        n_options = [np_] + [x for x in n_options if x != np_]

    init_types = ['uniform', 'random', 'bimodal', 'triangle', 'plateau', 'alternating']

    for n in n_options:
        for it in init_types[:3]:  # Top 3 initializations per length
            configs.append((n, it))

    if parent is not None:
        configs.insert(0, (len(parent), 'parent'))

    n_configs = len(configs)
    # Allocate more time to smaller n (faster per-iteration)
    weights = []
    for n, _ in configs:
        w = 1.0 / np.sqrt(n)
        weights.append(w)
    total_w = sum(weights)
    weights = [w / total_w * (budget_s - 60) for w in weights]

    print(f"Total configs: {n_configs}")

    for idx, (n, init_type) in enumerate(configs):
        if time.monotonic() > deadline:
            break

        # Time allocation
        t_alloc = weights[idx]
        # Cap time for large n
        if n >= 2048:
            t_alloc = min(t_alloc, 120.0)
        t_alloc = min(t_alloc, max(30.0, deadline - time.monotonic() - 30))
        if t_alloc < 20:
            continue

        # Create initial sequence
        if init_type == 'parent':
            f0 = parent.copy()
        else:
            f0 = make_init(n, init_type, rng)

        f0 = np.clip(f0, 0, 1000)
        if f0.sum() < 0.01:
            continue
        f0 = f0 / f0.sum()

        print(f"[{idx + 1}/{n_configs}] n={n}, init={init_type}, time={t_alloc:.0f}s")

        # Run optimization pipeline
        f_opt, score = run_pipeline(f0, t_alloc, rng, verbose=False)

        if np.isfinite(score) and score < best_score_overall:
            best_score_overall = score
            best_overall = f_opt.copy()
            print(f"  ** New best: {score:.6f} **")

    # Final restarts with best configuration
    remaining = deadline - time.monotonic()
    if best_overall is not None and remaining > 40:
        n_best = len(best_overall)
        n_restarts = min(int(remaining / 50), 6)
        print(f"\n{n_restarts} restarts with n={n_best}")
        for r in range(n_restarts):
            if time.monotonic() > deadline - 10:
                break
            f0 = best_overall * (1.0 + 0.25 * rng.standard_normal(n_best))
            f0 = np.clip(f0, 0, None)
            S0 = f0.sum()
            if S0 < 1e-10:
                continue
            f0 = f0 / S0
            t_left = max(15, deadline - time.monotonic() - 10)
            f_opt, score = run_pipeline(f0, t_left, rng, verbose=False)
            if np.isfinite(score) and score < best_score_overall:
                best_score_overall = score
                best_overall = f_opt.copy()
                print(f"  Restart {r}: {score:.6f}")

    # Verify with evaluate_sequence
    if best_overall is not None:
        best_overall = np.clip(best_overall, 0, 1000)
        final_list = [float(x) for x in best_overall]
        final_score = evaluate_sequence(final_list)
        print(f"\nFinal verified score: {final_score:.6f}")
        print(f"Sequence length: {len(final_list)}")
        return final_list

    # Fallback
    print("Fallback to uniform 256")
    return [1.0] * 256
