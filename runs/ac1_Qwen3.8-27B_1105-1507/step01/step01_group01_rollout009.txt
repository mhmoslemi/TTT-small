    # For k in [0, 2n-2], the constraint is:
    # sum_{i=max(0,k-n+1)}^{min(k, n-1)} a[k-i] * g[i] <= M

    # A is (2n-1) x n

    for k in range(2*n-1):

        e = min(k, n - 1)
        for i in range(s, e+1):
            A_ub[k, i] = a[k - i]

            a = rng.uniform(0.1, 1.0, n)
    
    return normalize_to_sum1(a)


def symmetrize(a):
    """Force symmetry: a[i] = a[n-1-i]."""
    n = a.size
    a = 0.5 * (a + a[::-1])
    return normalize_to_sum1(a)


def propose_candidate(seed=42, budget_s=1500, **kwargs):
    start_time = time.monotonic()
    try:
        budget = float(budget_s)
    except Exception:
        budget = 1500.0
    if not np.isfinite(budget) or budget <= 0:
        budget = 1500.0
    deadline = start_time + budget - 20  # leave margin
    
    # Get parent
    parent = globals().get('height_sequence_1', None)
    parent_arr = None
    if parent is not None:
        try:
            parent_arr = np.asarray(parent, dtype=np.float64)
            parent_arr = np.clip(parent_arr, 0, 1000)
            if parent_arr.size == 0 or not np.isfinite(parent_arr).all():
                parent_arr = None
        except Exception:
            parent_arr = None
    
    best_seq = None
    best_score = float('inf')
    
    if parent_arr is not None:
        p = normalize_to_sum1(parent_arr)
        if p is not None:
            best_seq = p.copy()
            best_score = local_score(p)
            print(f"Parent: n={p.size}, score={best_score:.6f}", flush=True)
    
    # Length plan with time fractions
    # Focus on lengths likely to beat current best
    length_plan = [
        (256, 0.08),
        (384, 0.10),
        (512, 0.32),
        (640, 0.10),
        (768, 0.12),
        (1024, 0.20),
    ]
    
    total_time = budget - 40  # reserve 40s for final
    seed_cur = int(seed)
    
    for L, frac in length_plan:
        rem = deadline - time.monotonic()
        if rem < 15:
            break
        alloc = min(frac * total_time, rem - 10)
        if alloc < 8:
            continue
        
        print(f"\n=== n={L}, alloc={alloc:.1f}s ===", flush=True)
        
        # Generate starts
        starts = []
        # Start from parent if matching length
        if parent_arr is not None and parent_arr.size == L:
            starts.append(normalize_to_sum1(parent_arr))
        elif parent_arr is not None:
            # Interpolate parent to this length
            old_x = np.linspace(0, 1, parent_arr.size)
            new_x = np.linspace(0, 1, L)
            interp = np.interp(new_x, old_x, parent_arr)
            starts.append(normalize_to_sum1(interp))
        
        # Random starts
        n_random = max(2, int(alloc / 20))
        for i in range(n_random):
            s = make_start(L, seed_cur + i * 104729, parent_arr)
            if s is not None:
                starts.append(s)
        
        # For longer searches, add a symmetric variant
        if alloc > 30:
            s = make_start(L, seed_cur + 99999, parent_arr)
            if s is not None:
                starts.append(symmetrize(s))
        
        # Optimize each start
        per_start_budget = alloc / (len(starts) + 1)
        
        for si, start_a in enumerate(starts):
            rem = deadline - time.monotonic()
            if rem < 10:
                break
            
            sb = min(per_start_budget * (1.5 if si == 0 else 1.0), rem - 5)
            if sb < 3:
                continue
            
            # Phase 1: Smoothed mirror descent
            beta_i = 15.0 if L <= 512 else 10.0
            K_i = max(3, L // 80)
            c1, s1 = smoothed_optimize(start_a, sb * 0.55, seed_cur + si * 7 + 1,
                                        beta_init=beta_i, alpha_init=0.8, K_init=K_i)
            
            if c1 is None:
                continue
            
            # Phase 2: LP polish
            rem2 = deadline - time.monotonic()
            lp_b = min(sb * 0.45, rem2 - 5)
            if lp_b > 2:
                c2, s2 = lp_optimize(c1, lp_b, seed_cur + si * 7 + 500)
                if c2 is not None and s2 < s1:
                    c1, s1 = c2, s2
            
            # Phase 3: Additional smoothed optimization from LP result
            rem3 = deadline - time.monotonic()
            if rem3 > 5:
                c3, s3 = smoothed_optimize(c1, min(rem3 - 3, 10), seed_cur + si * 7 + 1000,
                                            beta_init=30.0, alpha_init=0.4,
                                            K_init=max(4, L // 32))
                if c3 is not None and s3 < s1:
                    c1, s1 = c3, s3
            
            if s1 < best_score:
                best_seq = c1.copy()
                best_score = s1
                print(f"  Start {si} (len={L}): NEW BEST {s1:.6f}", flush=True)
            else:
                print(f"  Start {si} (len={L}): {s1:.6f}", flush=True)
        
        seed_cur += 200003
    
    # Final polishing on global best
    if best_seq is not None:
        rem = deadline - time.monotonic()
        if rem > 20:
            print(f"\n=== Final polish, rem={rem:.1f}s ===", flush=True)
            # Multiple LP rounds
            c, s = lp_optimize(best_seq, min(rem * 0.4, 60), seed_cur + 888888)
            if c is not None and s < best_score:
                best_seq = c
                best_score = s
                print(f"  LP polish: {s:.6f}", flush=True)
            
            rem2 = deadline - time.monotonic()
            if rem2 > 10:
                c, s = smoothed_optimize(best_seq, min(rem2 * 0.5, 30), seed_cur + 777777,
                                          beta_init=50.0, alpha_init=0.2,
                                          K_init=max(4, len(best_seq) // 16))
                if c is not None and s < best_score:
                    best_seq = c
                    best_score = s
                    print(f"  Smooth finish: {s:.6f}", flush=True)
            
            rem3 = deadline - time.monotonic()
            if rem3 > 10:
                c, s = lp_optimize(best_seq, min(rem3 * 0.6, 30), seed_cur + 666666)
                if c is not None and s < best_score:
                    best_seq = c
                    best_score = s
                    print(f"  Final LP: {s:.6f}", flush=True)
    
    # Fallback
    if best_seq is None:
        best_seq = np.ones(64, dtype=np.float64) / 64.0
        best_score = local_score(best_seq)
    
    # Validate
    final = normalize_to_sum1(best_seq)
    if final is None:
        final = np.ones(64, dtype=np.float64) / 64.0
    
    try:
        official = float(evaluate_sequence([float(x) for x in final]))
        if np.isfinite(official):
            best_score = official
        else:
            # Fallback to parent
            if parent_arr is not None:
                p = normalize_to_sum1(parent_arr)
                if p is not None:
                    final = p
                    best_score = local_score(final)
            else:
                final = np.ones(64, dtype=np.float64) / 64.0
                best_score = 2.0
    except Exception:
        pass
    
    print(f"\nFINAL: n={len(final)}, score={best_score:.6f}", flush=True)
    return [float(x) for x in np.clip(np.asarray(final, dtype=np.float64), 0.0, 1000.0)]
