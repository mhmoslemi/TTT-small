import numpy as np
from scipy.optimize import linprog
import time

def compute_score_normalized(a):
    """Compute score assuming sum(a) = 1."""
    n = len(a)
    conv = np.convolve(a, a)
    return 2.0 * n * conv.max()

def simplex_project(v, target=1.0):
    """Project v onto {a >= 0, sum(a) = target}."""
    n = len(v)
    # If all non-negative and sum = target, return as-is
    if np.all(v >= 0) and abs(v.sum() - target) < 1e-12:
        return v.copy()
    
    # Sort in decreasing order
    u = np.sort(v)[::-1]
    # Find the threshold
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n+1) > (cssv - target))[0][-1]
    theta = (cssv[rho] - target) / (rho + 1.0)
    return np.maximum(v - theta, 0)

def gradient_smmax(a, t):
    """Compute gradient of t*log(sum(exp(conv/t))) which approximates max(conv)."""
    n = len(a)
    conv = np.convolve(a, a)
    # Softmax
    conv_shifted = conv - conv.max()
    w = np.exp(conv_shifted / t)
    w /= w.sum()
    
    # Compute c[i] = sum_k a[k]*w[i+k]
    a_rev = a[::-1]
    c_full = np.convolve(a_rev, w)
    c = c_full[n-1:2*n-1]
    
    # g[i] = 2*c[i] - w[2*i]*a[i]
    g = 2.0 * c - w[0:2*n-1:2] * a
    
    return g, conv.max()

def gradient_hardmax(a):
    """Compute subgradient at the argmax of conv."""
    n = len(a)
    conv = np.convolve(a, a)
    M = conv.max()
    # Find all lags that achieve the max
    active = np.where(conv >= M - 1e-10)[0]

    g = np.zeros(n)

                else:

"""
Gradient-based optimization for minimizing 2*n*max(convolve(a,a))/sum(a)^2.
Uses log-parameterized gradient descent with softmax-smoothed objective,
multiple initializations, and local search polishing.
"""

import numpy as np
import time

def compute_score_normalized(a):
    """Compute score for a sequence normalized to sum=1."""
    n = len(a)
    conv = np.convolve(a, a)
    return 2.0 * n * conv.max()

def softmax(x):
    x = x - x.max()
    e = np.exp(x)
    return e / e.sum()

def forward(a):
    """Compute conv and score for normalized a."""
    n = len(a)
    conv = np.convolve(a, a)
    M = conv.max()
    score = 2.0 * n * M
    return conv, M, score

def grad_conv_a(a, t):
    """Gradient of t*log(sum(exp(conv/t))) w.r.t. a."""
    n = len(a)
    conv = np.convolve(a, a)
    conv_shifted = conv - conv.max()
    w = np.exp(conv_shifted / t)
    w /= w.sum()
    a_rev = a[::-1]
    c_full = np.convolve(a_rev, w)
    c = c_full[n-1:2*n-1]
    g = 2.0 * c - w[0:2*n-1:2] * a
    return g

def simplex_project(v, target=1.0):
    """Project onto {x >= 0, sum(x) = target}."""
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n+1) > (cssv - target)
    rho = np.nonzero(cond)[0]
    if len(rho) == 0:
        return np.full(n, target/n)
    rho = rho[-1]
    theta = (cssv[rho] - target) / (rho + 1.0)
    return np.maximum(v - theta, 0)

def optimize_logspace(x, time_budget, t_init=None, lr_init=1.0, 
                      t_decay=0.9995, t_min=1e-4, verbose=True, rng=None):
    """Gradient descent in log-space."""
    if rng is None:
        rng = np.random.default_rng(0)
    
    n = len(x)
    x = x.astype(np.float64).copy()
    a = softmax(x)
    
    conv, M, best_score = forward(a)
    best_a = a.copy()
    best_x = x.copy()
    
    if t_init is None:
        t_init = max(conv.max() - conv.min(), 1e-3)
    t = t_init
    lr = lr_init
    
    start = time.monotonic()
    it = 0
    no_improve = 0
    
    while time.monotonic() - start < time_budget:
        # Compute gradient in a-space
        g = grad_conv_a(a, t)
        
        # Transform to x-space gradient
        # dx[j] = a[j] * (g[j] - sum_k g[k]*a[k])
        ag = a @ g
        dx = a * (g - ag)
        
        # Update
        x_new = x - lr * dx
        a_new = softmax(x_new)
        
        # Evaluate
        _, _, score = forward(a_new)
        
        if score < best_score - 1e-12:
            best_a = a_new.copy()
            best_x = x_new.copy()
            best_score = score
            no_improve = 0
            if verbose and it % 200 == 0:
                elapsed = time.monotonic() - start
                print(f"    it={it}, score={best_score:.6f}, t={t:.6f}, lr={lr:.4f}, elapsed={elapsed:.1f}s")
        else:
            no_improve += 1
            lr = max(lr * 0.8, 1e-6)
            if no_improve > 100:
                # Perturb
                x = best_x + rng.normal(0, 0.1, n)
                a = softmax(x)
                best_score = compute_score_normalized(a)
                best_a = a.copy()
                best_x = x.copy()
                lr = lr_init * 0.5
                no_improve = 0
                t = min(t * 2, t_init)
        
        x = x_new
        a = a_new
        t = max(t * t_decay, t_min)
        it += 1
    
    return best_a, best_score

def optimize_simplex(a, time_budget, t_init=None, lr_init=0.05,
                     t_decay=0.999, t_min=0.001, verbose=True, rng=None):
    """Projected gradient descent on the simplex."""
    if rng is None:
        rng = np.random.default_rng(0)
    
    n = len(a)
    a = a.astype(np.float64).copy()
    a = a / a.sum()
    
    conv, M, best_score = forward(a)
    best_a = a.copy()
    
    if t_init is None:
        t_init = max(conv.max() - conv.min(), 1e-3)
    t = t_init
    lr = lr_init
    
    start = time.monotonic()
    it = 0
    no_improve = 0
    
    while time.monotonic() - start < time_budget:
        g = grad_conv_a(a, t)
        
        a_new = a - lr * g
        a_new = simplex_project(a_new, 1.0)
        
        score = compute_score_normalized(a_new)
        
        if score < best_score - 1e-12:
            best_a = a_new.copy()
            best_score = score
            no_improve = 0
            if verbose and it % 200 == 0:
                elapsed = time.monotonic() - start
                print(f"    it={it}, score={best_score:.6f}, t={t:.6f}, lr={lr:.4f}, elapsed={elapsed:.1f}s")
        else:
            no_improve += 1
            lr = max(lr * 0.7, 1e-8)
            if no_improve > 100:
                a = best_a + rng.normal(0, 0.05, n)
                a = np.clip(a, 0, None)
                if a.sum() > 1e-10:
                    a = a / a.sum()
                    best_score = compute_score_normalized(a)
                    best_a = a.copy()
                lr = lr_init * 0.3
                no_improve = 0
                t = min(t * 1.5, t_init)
        
        a = a_new
        t = max(t * t_decay, t_min)
        it += 1
    
    return best_a, best_score

def local_search(a, rng, n_iter=5000):
    """Random local search to polish."""
    n = len(a)
    a = a.copy()
    a = a / a.sum()
    best = a.copy()
    best_score = compute_score_normalized(a)
    
    for _ in range(n_iter):
        candidate = best.copy()
        # Perturb a random block
        width = max(1, min(n, int(rng.integers(1, n // 16 + 1))))
        start = int(rng.integers(0, n - width + 1))
        scale = np.exp(rng.normal(0, 0.2))
        candidate[start:start+width] *= scale
        candidate = np.clip(candidate, 0, None)
        s = candidate.sum()
        if s < 1e-10:
            continue
        candidate = candidate / s
        score = compute_score_normalized(candidate)
        if score < best_score:
            best = candidate
            best_score = score
    
    return best, best_score
