import numpy as np
from scipy.optimize import linprog, minimize
import time


def fast_conv(a):
    """Full self-convolution via FFT."""
    n = len(a)
    s = 1
    while s < 2 * n - 1:
        s <<= 1
    fa = np.fft.rfft(a, s)
    return np.fft.irfft(fa * fa, s)[:2 * n - 1]


def build_A(a):
    """Build constraint matrix for LP: A[k,j] gives coefficient of g_j in (a*g)_k."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for k in range(2 * n - 1):
        jl = max(0, k - n + 1)
        jh = min(n - 1, k)
        if jl <= jh:
            A[k, jl:jh + 1] = a[k - jh:k - jl + 1][::-1]
    return A


def lp_best_response(a, M):
    """Find g maximizing sum(g) s.t. (a*g)_k <= M, g >= 0."""
    n = len(a)
    c = -np.ones(n)
    A = build_A(a)
    r = linprog(c, A_ub=A, b_ub=np.full(2 * n - 1, M),
               bounds=[(0, None)] * n, method='highs')
    if r.success:
        g = r.x
        if g.sum() > 1e-10:
            return g
    return None


def lse_objective(x, tau):
    """Log-sum-exp smoothed objective. Minimize this with L-BFGS-B."""
    ax = np.maximum(x, 0.0)
    s = ax.sum()
    if s < 1e-8:
        return 1e10
    ax = ax / s
    b = fast_conv(ax)
    b_scaled = b / tau
    m = b_scaled.max()
    if m > 500:
        return float(m * tau)
    lse_val = tau * (m + np.log(np.sum(np.exp(b_scaled - m))))
    n = len(ax)
    return float(2.0 * n * lse_val)

    """Wrapper for scipy minimize (single argument)."""
    return lse_objective(x, tau)

    """Create objective function for given tau."""
    def obj(x):
        return lse_objective(x, tau)
    return obj

def interpolate_sequence(src, n_target):
    """Interpolate a sequence to a new length."""
    if src.size == n_target:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, n_target)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)

def fast_conv(a):
    n = len(a)
    s = 1

    fa = np.fft.rfft(a, s)
    return np.fft.irfft(fa * fa, s)[:2 * n - 1]


def build_A(a):
    n = len(a)

    return A

    """LP: max sum(g) s.t. (a*g)_k <= M, g >= 0."""
    n = len(a)
    c = -np.ones(n)
    A = build_A(a)
    r = linprog(c, A_ub=A, b_ub=np.full(2 * n - 1, M),
               bounds=[(0, None)] * n, method='highs')
    if r.success:
        g = r.x
        if g.sum() > 1e-10:
            return g
    return None


def compute_J(a):
    """Compute objective J(a) directly."""
    a = np.asarray(a, dtype=np.float64)
    a = np.clip(a, 0.0, None)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    n = len(a)
    b = fast_conv(a)
    return float(2.0 * n * b.max() / (s * s))


def lse_obj(x, tau):
    """Log-sum-exp smoothed objective for L-BFGS-B. Uses args=(tau,) in minimize."""
    ax = np.maximum(x, 0.0)
    s = ax.sum()
    if s < 1e-8:
        return 1e10
    ax = ax / s
    b = fast_conv(ax)
    b_scaled = b / tau
    m = b_scaled.max()
    if m > 500.0:
        return float(m * tau)
    lse_val = tau * (m + np.log(np.sum(np.exp(b_scaled - m))))
    n = len(ax)
    return float(2.0 * n * lse_val)


def lp_phase(a, max_iters, deadline):
    """Run LP best-response iterations. Returns improved sequence."""
    n = len(a)
    a = np.clip(a.copy(), 0.0, None)
    s = a.sum()
    if s < 1e-10:
        return a
    a = a / s
    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        b = fast_conv(a)
        M = float(b.max())
        g = lp_best_response(a, M)
        if g is None:
            break
        gs = g.sum()
        if gs < 1e-10:
            break
        g = g / gs
        # Adaptive learning rate: start large, decay
        t_mix = max(0.03, 0.5 * (1.0 - it / max_iters))
        a_new = (1.0 - t_mix) * a + t_mix * g
        a_new = np.clip(a_new, 0.0, None)
        sn = a_new.sum()
        if sn < 1e-10:
            break
        a_new = a_new / sn
        # Accept only if objective didn't worsen significantly
        b_new = fast_conv(a_new)
        if b_new.max() <= M * (1.0 + 1e-6):
            a = a_new
        else:
            # Try half step
            t_half = t_mix * 0.5
            a_new2 = (1.0 - t_half) * a + t_half * g
            a_new2 = np.clip(a_new2, 0.0, None)
            sn2 = a_new2.sum()
            if sn2 > 1e-10:
                a_new2 = a_new2 / sn2
                b_new2 = fast_conv(a_new2)
                if b_new2.max() <= M * (1.0 + 1e-6):
                    a = a_new2
    return a


def lse_phase(a, deadline):
    """Polish using L-BFGS-B with decreasing log-sum-exp temperature."""
    n = len(a)
    a = np.clip(a.copy(), 0.0, None)
    s = a.sum()
    if s < 1e-10:
        return a
    a = a / s
    # Tau schedule: start smooth, end sharp
    taus = [1.0, 0.5, 0.25, 0.12, 0.06, 0.03, 0.015, 0.008, 0.004]
    current = a.copy()
    for tau in taus:
        if time.monotonic() > deadline:
            break
        res = minimize(lse_obj, current, args=(tau,), method='L-BFGS-B',
                      bounds=[(0, None)] * n,
                      options={'maxiter': 200, 'max_fun': 8000,
                               'ftol': 1e-14, 'disp': False})
        if res.fun < 1e10:
            current = np.clip(res.x, 0.0, None)
            sc = current.sum()
            if sc > 1e-10:
                current = current / sc
    return current


def interp_seq(src, n_target):
    """Interpolate sequence to target length."""
    if src.size == n_target:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, n_target)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)


def get_inits(n, rng, parent):
    """Generate diverse initializations."""
    inits = []
    # Uniform
    inits.append(np.ones(n))
    # Parent interpolated
    if parent is not None:
        p = interp_seq(parent, n)
        if p.sum() > 0.01:
            inits.append(p)
    # Raised cosine
    t = np.arange(n) / max(1, n - 1)
    rc = 0.5 * (1.0 + np.cos(2.0 * np.pi * t))
    inits.append(np.clip(rc, 1e-3, None))
    # Gaussian centered
    center = (n - 1) / 2.0
    gauss = np.exp(-0.5 * ((np.arange(n) - center) / (n / 5.0)) ** 2)
    inits.append(np.clip(gauss, 1e-4, None))
    # Random
    r = rng.uniform(0.1, 1.0, n)
    inits.append(r)
    # Cosine modulated (multiple frequencies)
    t2 = np.arange(n) / float(n)
    cos_mod = 1.0 + 0.3 * np.cos(4.0 * np.pi * t2) + 0.1 * np.cos(8.0 * np.pi * t2)
    inits.append(np.clip(cos_mod, 0.01, None))
    # Broad flat-ish (low peak autocorrelation target)
    tri = 1.0 - 0.3 * np.abs(np.arange(n) - center) / max(1, center)
    inits.append(np.clip(tri, 0.1, None))
    return inits


def propose_candidate(seed=42, budget_s=1500, **kwargs):
    """Search for best sequence minimizing J(a) = 2*n*max(conv(a,a))/(sum(a))^2."""
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + float(budget_s) - 20.0
    best = None
    best_score = float('inf')

    # Try parent sequence
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        if p.ndim == 1 and p.size > 0 and np.all(np.isfinite(p)):
            parent = p.copy()
            score = float(evaluate_sequence(p.tolist()))
            if np.isfinite(score):
                best = p.copy()
                best_score = score
                print(f"Parent: J={score:.6f} (n={p.size})", flush=True)
    except (NameError, AttributeError):
        pass

    # === Phase 1: LP Best-Response ===
    print("\n=== Phase 1: LP Best-Response ===", flush=True)
    for n in [256, 512]:
        if time.monotonic() > deadline - 500:
            break
        print(f"\n  LP: n={n}", flush=True)
        t0 = time.monotonic()
        inits = get_inits(n, rng, parent)
        for i, a0 in enumerate(inits):
            if time.monotonic() > deadline - 500:
                break
            a0 = np.clip(a0, 1e-4, None)
            a0 = a0 / a0.sum()
            r = compute_J(a0)
            if r < best_score:
                best = a0.copy()
                best_score = r
                print(f"    init{i}: J={r:.6f}", flush=True)
            a_out = lp_phase(a0, max_iters=100, deadline=deadline - 500)
            r = compute_J(a_out)
            if r < best_score:
                best = a_out.copy()
                best_score = r
                print(f"    LP init{i}: J={r:.6f}", flush=True)
        print(f"  LP n={n}: {time.monotonic()-t0:.1f}s, best={best_score:.6f}", flush=True)

    # Try n=1024 LP if time allows
    if time.monotonic() < deadline - 200:
        n = 1024
        print(f"\n  LP: n={n}", flush=True)
        t0 = time.monotonic()
        inits = get_inits(n, rng, best)
        for i, a0 in enumerate(inits[:3]):  # fewer inits for larger n
            if time.monotonic() > deadline - 200:
                break
            a0 = np.clip(a0, 1e-4, None)
            a0 = a0 / a0.sum()
            a_out = lp_phase(a0, max_iters=50, deadline=deadline - 200)
            r = compute_J(a_out)
            if r < best_score:
                best = a_out.copy()
                best_score = r
                print(f"    LP init{i}: J={r:.6f}", flush=True)
        print(f"  LP n={n}: {time.monotonic()-t0:.1f}s, best={best_score:.6f}", flush=True)

    # === Phase 2: LSE + L-BFGS-B ===
    print("\n=== Phase 2: LSE + L-BFGS-B ===", flush=True)
    for n in [512, 1024, 2048]:
        if time.monotonic() > deadline - 80:
            break
        print(f"\n  LSE: n={n}", flush=True)
        t0 = time.monotonic()
        inits = get_inits(n, rng, best)
        # Also add perturbed best
        if best is not None:
            bp = interp_seq(best, n)
            noise = rng.normal(0, 0.05, n)
            bp_pert = np.clip(bp + noise, 0.0, None)
            if bp_pert.sum() > 0.01:
                inits.append(bp_pert)
        for i, a0 in enumerate(inits):
            if time.monotonic() > deadline - 80:
                break
            a0 = np.clip(a0, 0.0, None)
            s = a0.sum()
            if s < 0.01:
                continue
            a0 = a0 / s
            a_out = lse_phase(a0, deadline - 80)
            r = compute_J(a_out)
            if r < best_score:
                best = a_out.copy()
                best_score = r
                print(f"    LSE init{i}: J={r:.6f} (n={n})", flush=True)
        print(f"  LSE n={n}: {time.monotonic()-t0:.1f}s, best={best_score:.6f}", flush=True)

    # === Phase 3: Combined Polish ===
    print("\n=== Phase 3: Polish ===", flush=True)
    if best is not None and time.monotonic() < deadline - 10:
        n = best.size
        t0 = time.monotonic()

        # LP polish (if n is manageable)
        if n <= 512:
            a_polished = lp_phase(best, max_iters=60, deadline=deadline - 50)
            r = compute_J(a_polished)
            if r < best_score:
                best = a_polished.copy()
                best_score = r
                print(f"  LP polish: J={r:.6f}", flush=True)

        # LSE polish on best
        a_polished = lse_phase(best, deadline - 10)
        r = compute_J(a_polished)
        if r < best_score:
            best = a_polished.copy()
            best_score = r
            print(f"  LSE polish: J={r:.6f}", flush=True)

        # Try random restart + quick polish
        if time.monotonic() < deadline - 30:
            for attempt in range(3):
                if time.monotonic() > deadline - 30:
                    break
                a_restart = best.copy()
                noise = rng.normal(0, 0.1, a_restart.size)
                a_restart = np.clip(a_restart + noise, 0.0, None)
                if a_restart.sum() < 0.01:
                    continue
                a_restart = a_restart / a_restart.sum()
                a_restart = lse_phase(a_restart, deadline - 30)
                r = compute_J(a_restart)
                if r < best_score:
                    best = a_restart.copy()
                    best_score = r
                    print(f"  Restart {attempt}: J={r:.6f}", flush=True)

        print(f"  Polish total: {time.monotonic()-t0:.1f}s", flush=True)

    # === Final: Verify and return ===
    if best is None:
        best = np.ones(256)

    best = np.clip(best, 0.0, 1000.0)
    final_score = float(evaluate_sequence([float(x) for x in best]))
    if np.isfinite(final_score):
        best_score = min(best_score, final_score)
    print(f"\nFINAL: J={final_score:.6f}, n={best.size}, tracked_best={best_score:.6f}", flush=True)
    return [float(x) for x in best]
