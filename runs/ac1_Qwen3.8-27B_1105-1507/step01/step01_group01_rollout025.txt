"""
Minimize score = 2*n*max(conv(a,a))/sum(a)^2 using smoothed L-BFGS-B
optimization with log-sum-exp cooling schedule, combined with
multiplicative subgradient polishing.
"""

import time
import numpy as np
from scipy.optimize import minimize as scipy_minimize

def local_score(a):
    a = np.asarray(a, dtype=np.float64)
    n = a.size
    if n == 0:
        return float('inf')
    a = np.clip(a, 0.0, 1000.0)
    S = float(np.sum(a))
    if S <= 1e-12:
        return float('inf')
    c = np.convolve(a, a)
    m = float(np.max(c))
    return float(2.0 * n * m / (S * S))

def normalize_to_sum1(a):
    a = np.asarray(a, dtype=np.float64)
    a = np.clip(a, 0.0, 1000.0)
    S = float(np.sum(a))
    if S <= 1e-300:
        return None
    return a / S

def smooth_obj_and_grad(x, t, n):
    """Softmax-parameterized smooth objective and gradient."""
    # x is the free variable, a = softmax(x)
    x_shifted = x - np.max(x)  # numerical stability
    exp_x = np.exp(x_shifted)
    a = exp_x / np.sum(exp_x)
    
    # Compute convolution
    c = np.convolve(a, a)
    c_max = float(np.max(c))
    
    # Log-sum-exp
    logits = (c - c_max) / t
    exp_logits = np.exp(logits)
    sum_exp = float(np.sum(exp_logits))
    log_sum = float(np.log(sum_exp))
    L = c_max + t * log_sum
    
    # Score = 2*n*L (since sum(a) = 1)
    score = 2.0 * n * L
    
    # Gradient w.r.t. a
    # p_k = exp(c_k/t) / sum_exp
    p = exp_logits / sum_exp
    
    # q_i = sum_k p_k * a[k-i]
    # Using correlate
    q = np.correlate(p, a, mode='full')[n-1:2*n-1]
    
    # grad_a_i = 4*n*(S*q_i - L*a_i) / S^3 with S=1
    # = 4*n*(q_i - L*a_i)
    grad_a = 4.0 * n * (q - L * a)
    
    # Convert to gradient w.r.t. x
    # a = softmax(x), so da_i/dx_j = a_i*(delta_ij - a_j)
    # grad_x_j = sum_i grad_a_i * a_i * (delta_ij - a_j)
    # = a_j * grad_a_j - a_j * sum_i(grad_a_i * a_i)
    dot = float(np.dot(grad_a, a))
    grad_x = a * (grad_a - dot)
    
    return score, grad_x

def optimize_smooth_from_x(x_init, max_time, t_start, t_end, t_decay, max_iter_per_phase):
    """Run L-BFGS-B with cooling schedule in x-space (softmax parameterization)."""
    n = x_init.size
    t = t_start
    best_x = x_init.copy()
    best_score = smooth_obj_and_grad(x_init, t_end, n)[0]  # evaluate at final t for consistency
    # Actually, let's track the score at the current t
    best_score = smooth_obj_and_grad(x_init, t_start, n)[0]
    best_x = x_init.copy()
    
    t0 = time.monotonic()
    deadline = t0 + max_time
    
    while t > t_end and time.monotonic() < deadline:
        remaining = deadline - time.monotonic()
        if remaining < 2.0:

            )
            if res.success and res.fun < best_score:
                best_score = float(res.fun)
                best_x = res.x.copy()
            x_init = res.x  # continue from here
        except Exception:
            break
        
        t *= t_decay
    
    return best_x, best_score

def x_to_a(x):
    x_shifted = x - np.max(x)
    exp_x = np.exp(x_shifted)
    return exp_x / np.sum(exp_x)

def multiplicative_polish(a, max_time, num_peaks=None):
    """Polish using multiplicative updates on the active peaks."""
    a = np.asarray(a, dtype=np.float64).copy()
    n = a.size
    best = a.copy()
    best_score = local_score(best)
    
    t0 = time.monotonic()
    deadline = t0 + max_time
    eta = 0.15
    
    while time.monotonic() < deadline:
        c = np.convolve(a, a)
        M = float(np.max(c))
        if M <= 0:
            break
        
        # Find top K peaks
        if num_peaks is None:
            K = max(10, n // 8)
        else:
            K = num_peaks
        K = min(K, 2*n-1)
        
        active = np.argpartition(c, -K)[-K:]
        w = c[active] / M
        
        # Compute the "pressure" on each a_i
        h = np.zeros(n)
        for idx in range(active.size):
            k = int(active[idx])
            wk = float(w[idx])
            s = max(0, k - (n-1))
            e = min(k, n-1)
            for i in range(s, e+1):
                j = k - i
                h[i] += wk * a[j]
                h[j] += wk * a[i]
        
        # Multiplicative update
        maxh = float(np.max(h))
        if maxh <= 0:
            break
        fac = eta * (h.mean() - h) / maxh
        fac = np.clip(fac, -3.0, 3.0)
        a_new = a * np.exp(fac)
        a_new = np.clip(a_new, 0.0, 1000.0)
        a_new = normalize_to_sum1(a_new)
        if a_new is None:
            break
        
        sc = local_score(a_new)
        if sc < best_score:
            best = a_new.copy()
            best_score = float(sc)
            a = a_new
