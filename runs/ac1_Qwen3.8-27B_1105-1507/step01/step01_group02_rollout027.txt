```python
import numpy as np
import time

def fast_convolve(a):
    """Compute full linear convolution of a with itself using FFT."""
    n = len(a)
    size = 2 * n - 1
    N = 1 << (size - 1).bit_length()
    fa = np.fft.rfft(a, N)
    fc = fa * fa
    result = np.fft.irfft(fc, N)[:size]
    result[result < 0] = 0.0
    return result

def project_simplex(v):
    """Project vector v onto the probability simplex."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cv = np.cumsum(u) - 1.0
    rho_candidates = np.where(u * np.arange(1, n + 1) > cv)[0]
    if len(rho_candidates) == 0:
        return np.ones(n) / n
    rho = rho_candidates[-1]
    theta = cv[rho] / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def smooth_max(c, tau):
    """Smooth approximation of max using logsumexp."""
    c_shifted = c - np.max(c)
    return np.max(c) + tau * np.log(np.sum(np.exp(c_shifted / tau)))
```

def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    N = 1 << (size - 1).bit_length()
    fa = np.fft.rfft(a, N)
    result = np.fft.irfft(fa * fa, N)[:size]
    result[result < -1e-15] = 0.0

    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cv = np.cumsum(u) - 1.0
    tmp = u * np.arange(1, n+1) > cv

        theta = cv[-1] / n
    else:
        rho = np.where(tmp)[0][-1]
        theta = cv[rho] / (rho + 1)

import numpy as np

def fast_convolve(a):
    """Full linear convolution a*a via FFT."""
    n = len(a)
    size = 2 * n - 1
    N = 1 << (size - 1).bit_length()
    fa = np.fft.rfft(a, N)
    result = np.fft.irfft(fa * fa, N)[:size]
    np.maximum(result, 0.0, out=result)
    return result

def project_simplex(v):
    """Project onto probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cv = np.cumsum(u) - 1.0
    tmp = u * np.arange(1, n + 1) > cv
    if tmp.all():
        theta = cv[-1] / n
    else:
        rho = np.where(tmp)[0][-1]
        theta = cv[rho] / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def compute_grad_fft(a, w):
    """
    Compute gradient of sum_k w[k]*c[k] where c=convolve(a,a).
    Uses FFT cross-correlation for O(n log n) complexity.
    """
    n = len(a)
    N = 1 << (3 * n).bit_length()
    A = np.fft.rfft(a, N)
    W = np.fft.rfft(w, N)
    corr = np.fft.irfft(W * np.conj(A), N)
    return 2.0 * corr[:n]

def alt_proj_init(n, rng, n_iter=100, mode='random'):
    """
    Alternating projection: project between flat-spectrum and non-negative simplex.
    """
    N = 1 << (2 * n).bit_length()
    
    if mode == 'random':
        phases = rng.uniform(0, 2 * np.pi, N // 2 + 1)
    elif mode == 'chirp':
        k = np.arange(N // 2 + 1, dtype=np.float64)
        phases = np.pi * k * k / (N * 0.5)
    elif mode == 'linear':
        k = np.arange(N // 2 + 1, dtype=np.float64)
        phases = 2 * np.pi * k / N
    else:
        phases = rng.uniform(0, 2 * np.pi, N // 2 + 1)
    
    F = np.exp(1j * phases).astype(np.complex128)
    F[0] = F[0].real + 0j
    if N % 2 == 0 and len(F) > 1:
        F[-1] = F[-1].real + 0j
    
    a = np.ones(n) / n
    for it in range(n_iter):
        a_full = np.fft.irfft(F, N)
        a = np.maximum(a_full[:n], 0.0)
        s = a.sum()
        if s < 1e-10:
            a = np.ones(n) / n
        else:
            a = a / s
        F_full = np.fft.rfft(a, N)
        # Target magnitude: slight taper for better non-negativity
        t = np.linspace(1.0, 0.85, len(F_full))
        F = (t ** 0.3) * np.exp(1j * np.angle(F_full)).astype(np.complex128)
        F[0] = F[0].real + 0j
        if N % 2 == 0 and len(F) > 1:
            F[-1] = F[-1].real + 0j
    
    return a

def smooth_optimize(a, deadline, rng, n_max_iter=200000):
    """
    Smooth-max projected gradient with momentum and cosine-annealed temperature.
    """
    n = len(a)
    a = project_simplex(np.maximum(a, 0).astype(np.float64))
    
    # Initial evaluation
    c = fast_convolve(a)
    peak0 = np.max(c)
    best_a = a.copy()
    best_peak = peak0
    
    # Momentum
    velocity = np.zeros(n)
    mu = 0.85  # momentum coefficient
    
    # Temperature schedule (cosine annealing from peak0*0.3 to peak0*0.001)
    tau_start = peak0 * 0.3
    tau_end = peak0 * 0.001
    
    # Learning rate schedule
    lr_start = 2.0 / n
    lr_end = lr_start * 0.01
    
    total_iters = min(n_max_iter, int((deadline - time.time()) / 0.005))  # estimate
    total_iters = max(total_iters, 100)
    
    it = 0
    stall = 0
    max_stall = 3000
    
    while time.time() < deadline - 5 and it < total_iters:
        it += 1
        progress = it / total_iters
        
        # Cosine annealing
        cos_term = 0.5 * (1 + np.cos(np.pi * progress))
        tau = tau_end + (tau_start - tau_end) * cos_term
        lr = lr_end + (lr_start - lr_end) * cos_term
        
        # Nesterov-style lookahead
        y = a + mu * velocity / lr  # This isn't quite right for fixed lr...
        # Let me use simple heavy-ball instead:
        y = a.copy()
        
        # Compute smooth max gradient
        c_y = fast_convolve(y)
        c_shifted = (c_y - np.max(c_y)) / tau
        exp_c = np.exp(c_shifted)
        weights = exp_c / exp_c.sum()
        
        # Gradient via FFT
        grad = compute_grad_fft(y, weights)
        
        # Momentum update
        velocity = mu * velocity - lr * grad
        a = project_simplex(a + velocity)

            peak = np.max(c_eval)

                best_a = a.copy()

                stall += 1

                if peak0 > 0:
                    a = project_simplex(a + rng.normal(0, 0.05 * peak0 / np.sqrt(n) / peak0, n))
                    a = project_simplex(a + rng.normal(0, 0.02 / np.sqrt(n), n))
                velocity *= 0.1
                stall = 0
                max_stall = max(1000, max_stall * 0.7)
        
    return best_a
