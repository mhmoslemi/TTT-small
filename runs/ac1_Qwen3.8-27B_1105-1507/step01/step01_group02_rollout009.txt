def project_simplex(v):
    """Project vector v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cv = np.cumsum(u) - 1.0
    css = 1.0 + np.arange(1, n + 1) * u - cv
    rho = np.sum(css > 0)
    if rho == 0:
        return np.ones(n) / n
    theta = cv[rho - 1] / rho
    return np.maximum(v - theta, 0.0)


def fast_convolve(a):
    """Full linear convolution of a with itself using FFT."""
    n = len(a)
    size = 2 * n - 1

    fa = np.fft.rfft(a, N)
    result = np.fft.irfft(fa * fa, N)[:size]
    result[result < -1e-15] = 0.0
    return result

    min_step = 1e-8

    L = 2.0 * np.max(a) * 1.2
    min_L = 1e-6

    stall_count = 0
    max_stall = 500
    it = 0

    while time.time() < deadline and it < max_iters:
        it += 1

        # Adapt temperature
        progress = min(1.0, it / T_ramp_iters)
        T = T_start + (T_end - T_start) * progress

        # Nesterov momentum
        momentum = (t - 1.0) / (t + 1.0)
        y = a + momentum * (a - a_prev)
        y = np.maximum(y, 0.0)
        s = y.sum()
        if s < 1e-15:
            y = a.copy()
        else:
            y = y / s

        # Compute gradient at y
        b_y = fast_convolve(y)
        b_max = np.max(b_y)

        # Soft weights
        if T < 1.0:
            threshold = b_max * (1.0 - 1e-10)
            w = (b_y >= threshold).astype(np.float64)
            w_sum = w.sum()
            if w_sum > 0:
                w = w / w_sum
            else:
                w = np.ones(2 * n - 1) / (2 * n - 1)

            w = exp_vals / np.sum(exp_vals)

def ga_mutation(a, rng, strength):
    """Apply Gaussian mutation to a sequence, maintaining simplex constraint."""
    n = len(a)
    noise = rng.normal(0, strength / np.sqrt(n), n)
    a_new = a + noise
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s < 1e-10:
        return np.ones(n) / n
    return a_new / s


def ga_crossover(a1, a2, rng):
    """Uniform crossover between two sequences of same length."""
    n = len(a1)

    child = np.where(mask, a1, a2)
    s = child.sum()
    if s < 1e-10:
        return np.ones(n) / n
    return child / s

                    if sc < eval_score_normalized(result):
                        result = candidate
        except Exception:
            pass

    return result


def propose_candidate(budget_s=1500, **kwargs):
    """
    Multi-phase hybrid search: FISTA with log-sum-exp, GA exploration,
    LP polishing, and multi-length optimization.
    """
    overall_deadline = time.time() + budget_s - 15.0
    rng = np.random.default_rng(12345)

    best_score = float('inf')
    best_seq = None

    print(f"Starting optimization at {time.time():.0f}, deadline {overall_deadline:.0f}")

    # Phase 1: Try various n values with FISTA
    n_values = [512, 768, 1024, 640, 896]
    init_types = ['alt_proj', 'broadband', 'parent', 'random', 'cosine']

    # Allocate time for Phase 1
    phase1_deadline = time.time() + (budget_s - 15.0) * 0.45

    for i, n in enumerate(n_values):
        if time.time() > phase1_deadline - 30:
            break

        config_time = min(phase1_deadline - 10, 120.0)
        config_deadline = time.time() + config_time

        for j, itype in enumerate(init_types[:3]):
            if time.time() > phase1_deadline - 10:
                break
            inner_deadline = min(config_deadline, time.time() + 40)
            try:
                a_init = init_structured(n, rng, itype)
                a_opt, score = fista_optimize(a_init, inner_deadline, rng)
                print(f"  FISTA n={n}, init={itype}: score={score:.6f}")
                if score < best_score:
                    best_score = score
                    best_seq = a_opt.copy()
                    print(f"  >>> New best: {best_score:.6f} (n={n})")
            except Exception as e:
                print(f"  Error: {e}")

    # Phase 2: GA exploration on best n
    phase2_deadline = time.time() + (budget_s - 15.0) * 0.25
    best_n = len(best_seq) if best_seq is not None else 512

    print(f"\nPhase 2: GA exploration (n={best_n})")
    try:
        ga_deadline = min(phase2_deadline, time.time() + (budget_s - 15.0) * 0.20)
        ga_pop = 40
        a_ga, ga_score = genetic_search(best_n, ga_pop, rng, ga_deadline)
        print(f"  GA: score={ga_score:.6f}")
        if ga_score < best_score:
            best_score = ga_score
            best_seq = a_ga.copy()
            print(f"  >>> New best from GA: {best_score:.6f}")
    except Exception as e:
        print(f"  GA error: {e}")

    # Also try different n with GA
    for n_try in [best_n // 2, best_n * 2]:
        if time.time() > phase2_deadline:
            break
        if n_try < 64 or n_try > 4096:
            continue
        try:
            ga_deadline2 = min(phase2_deadline, time.time() + 60)
            a_ga2, ga_score2 = genetic_search(n_try, 25, rng, ga_deadline2)
            print(f"  GA n={n_try}: score={ga_score2:.6f}")
            if ga_score2 < best_score:
                best_score = ga_score2
                best_seq = a_ga2.copy()
                print(f"  >>> New best from GA: {best_score:.6f}")
        except Exception as e:
            print(f"  GA n={n_try} error: {e}")

    # Phase 3: Deep FISTA on best found
    print(f"\nPhase 3: Deep FISTA on best (n={len(best_seq)}, score={best_score:.6f})")
    phase3_deadline = time.time() + (budget_s - 15.0) * 0.20
    if best_seq is not None:
        try:
            a_deep, deep_score = fista_optimize(best_seq, phase3_deadline, rng, max_iters=500000)
            print(f"  Deep FISTA: score={deep_score:.6f}")
            if deep_score < best_score:
                best_score = deep_score
                best_seq = a_deep.copy()
                print(f"  >>> New best: {best_score:.6f}")
        except Exception as e:
            print(f"  Deep FISTA error: {e}")

        # Try multiple restarts from best
        for restart in range(3):
            if time.time() > phase3_deadline:
                break
            a_restart = ga_mutation(best_seq, rng, 0.03)
            restart_deadline = min(phase3_deadline, time.time() + 30)
            try:
                a_r, r_score = fista_optimize(a_restart, restart_deadline, rng, max_iters=100000)
                if r_score < best_score:
                    best_score = r_score
                    best_seq = a_r.copy()
                    print(f"  Restart {restart}: new best {best_score:.6f}")
            except Exception:
                pass

    # Phase 4: LP polish
    print(f"\nPhase 4: LP polish")
    if best_seq is not None and time.time() < overall_deadline - 30:
        n_best = len(best_seq)
        if n_best <= 2048:
            try:
                polish_deadline = overall_deadline - 15
                a_polished = lp_polish(best_seq, polish_deadline)
                p_score = eval_score_normalized(a_polished)
                print(f"  LP polish: score={p_score:.6f}")
                if p_score < best_score:
                    best_score = p_score
                    best_seq = a_polished.copy()
            except Exception as e:
                print(f"  LP error: {e}")

    # Phase 5: Final FISTA with remaining time
    if best_seq is not None and time.time() < overall_deadline - 20:
        try:
            a_final, f_score = fista_optimize(best_seq, overall_deadline, rng, max_iters=1000000)
            if f_score < best_score:
                best_score = f_score
                best_seq = a_final.copy()
            print(f"  Final FISTA: score={best_score:.6f}")
        except Exception:
            pass

    # Verify and return
    if best_seq is None:
        print("Falling back to uniform sequence")
        best_seq = np.ones(512) / 512

    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    if sum(result) < 0.01:
        result = [1.0 / 512.0] * 512

    final_score = evaluate_sequence(result)
    print(f"\nFINAL: score={final_score:.6f}, length={len(result)}")
    return result
