def subgradient_step(f, k_star, lr):
    """Take a Frank-Wolfe away/toward step based on subgradient at k_star."""
    n = len(f)
    # Compute subgradient of (f*f)_{k_star}
    # grad[m] = f[k_star - m] if 0 <= k_star-m < n (times 2 if k_star != 2*m)
    grad = np.zeros(n)
    for m in range(n):
        idx = k_star - m
        if 0 <= idx < n:
            if k_star == 2 * m:
                grad[m] = f[idx]
            else:
                grad[m] = 2.0 * f[idx]
    
    # Frank-Wolfe: move toward argmin grad, away from argmax positive grad in support
    min_idx = int(np.argmin(grad))
    
    # Away step: find index with max positive grad in current support
    support = np.where(f > 1e-10)[0]
    if len(support) > 0:
        max_idx = support[np.argmax(grad[support])]
        # Check if away step is beneficial
        if grad[max_idx] > 0 and grad[max_idx] > -grad[min_idx]:
            # Away step
            f_new = f.copy()
            f_new[max_idx] -= lr * f[max_idx]
            f_new[max_idx] = max(0, f_new[max_idx])
            f_new[min_idx] += lr * f[max_idx]
            s = f_new.sum()
            if s > 0:
                f_new /= s
                return f_new
    
    # Standard toward step
    f_new = (1 - lr) * f + lr * np.zeros(n)
    f_new[min_idx] = lr
    s = f_new.sum()
    if s > 0:
        f_new /= s
    return f_new

        return None
    if res.x is None:
        return None
    x = np.asarray(res.x, dtype=np.float64)
    x = np.maximum(x, 0)
    s = float(x.sum())
    if s < 1e-300:
        return None
    return x / s


def subgradient_step(f, k_star, rng):
    """Frank-Wolfe style step using subgradient at k_star."""
    n = len(f)
    m_vals = np.arange(n)
    idx = k_star - m_vals
    valid = (idx >= 0) & (idx < n)
    m_lo = max(0, k_star - n + 1)
    m_hi = min(n - 1, k_star)
    valid &= (m_vals >= m_lo) & (m_vals <= m_hi)
    grad = np.zeros(n)
    grad[valid] = 2.0 * f[idx[valid]]
    center = k_star // 2
    if k_star % 2 == 0 and 0 <= center < n and valid[center]:
        grad[center] = f[center]

    min_idx = int(np.argmin(grad))
    support = np.where(f > 1e-8)[0]
    if len(support) > 1:
        g_support = grad[support]
        max_local = int(support[np.argmax(g_support)])
        if grad[max_local] > -grad[min_idx] * 0.5 and grad[max_local] > 0:
            lr = 0.3
            f_new = f.copy()
            amount = min(lr * f[max_local], f[max_local] * 0.5)
            f_new[max_idx] -= amount
            f_new[min_idx] += amount
            return normalize(f_new)

    lr = 0.3
    f_new = f * (1.0 - lr)
    f_new[min_idx] += lr
    return normalize(f_new)


def interpolate(x, n):
    """Resample normalized sequence to length n."""
    x = normalize(x)
    if x is None:
        return np.ones(n) / n
    m = x.size
    if m == n:
        return x
    old = np.linspace(0.0, 1.0, m)
    new = np.linspace(0.0, 1.0, n)
    out = np.interp(new, old, x)
    r = normalize(out)
    return r if r is not None else np.ones(n) / n


def optimize_lp(n, f_init, time_budget, seed=42):
    """Main LP-based optimization loop for a given length n."""
    f = normalize(f_init)
    if f is None:
        f = np.ones(n) / n

    try:
        A, offs, m = build_lp_structure(n)
    except MemoryError:
        return f, score(f)
    except Exception:
        return f, score(f)

    best_f = f.copy()
    best_sc = score(f)
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + time_budget
    stall = 0
    it = 0
    active_threshold = 0.85
    rng2 = np.random.default_rng(seed + 1000)

    while time.monotonic() < deadline:
        it += 1
        t0 = time.monotonic()
        remaining = deadline - time.monotonic()
        if remaining < 1.0:
            break

        ac = autocorr(f)
        M = float(np.max(ac))
        if M <= 0 or not np.isfinite(M):
            break

        # Adaptive active set
        threshold = active_threshold * M
        active = np.where(ac >= threshold)[0]
        if len(active) < 15:
            top_idx = np.argsort(ac)[::-1][:min(30, m)]
            active = top_idx
        # In early iterations, use more constraints for accuracy
        if it < 10 and len(active) < m // 2:
            threshold2 = 0.7 * M
            active = np.where(ac >= threshold2)[0]

        g = solve_lp(A, offs, f, M, active)

        if g is None or not np.isfinite(g).all():
            # Fallback: subgradient step
            k_star = int(np.argmax(ac))
            f_try = subgradient_step(f, k_star, rng)
            if f_try is not None:
                sc_try = score(f_try)
                if sc_try < best_sc:
                    f = f_try
                    best_f = f.copy()
                    best_sc = sc_try
                    stall = 0
                else:
                    stall += 1
            else:
                stall += 1
            if stall > 12:
                # Diverse restart
                f = normalize(0.6 * f + 0.4 * rng.dirichlet(np.ones(n)))
                stall = 0
            continue

        # Check improvement potential
        conv_fg = convolve_fft(f, g)
        r_max = float(np.max(conv_fg))
        G = float(np.max(autocorr(g)))

        if r_max >= M * (1.0 - 1e-10):
            stall += 1
            if stall > 10:
                f = subgradient_step(f, int(np.argmax(ac)), rng)
                if f is None:
                    f = normalize(0.5 * best_f + 0.5 * rng.dirichlet(np.ones(n)))
                stall = 0
                # Tighten active set to find new directions
                active_threshold = min(0.95, active_threshold + 0.02)
            else:
                f = normalize(f + 0.03 * rng.dirichlet(np.ones(n)))
            continue

        # Compute safe step size
        a_coef = 2.0 * (r_max - M)
        b_coef = G + M - 2.0 * r_max
        if b_coef > 1e-14:
            t_max = -a_coef / b_coef
        else:
            t_max = 1.0
        if not np.isfinite(t_max) or t_max <= 0:
            t_max = 1e-5
        t_max = min(1.0, t_max)

        # Backtracking: try decreasing step sizes
        ts = sorted(set([t_max, 0.7 * t_max, 0.5 * t_max, 0.3 * t_max,
                         0.15 * t_max, 0.08 * t_max, 0.04 * t_max,
                         0.02 * t_max, 0.01 * t_max, 0.005 * t_max]),
                    reverse=True)
        ts = [t for t in ts if t > 1e-8 and t <= 1.0]
        if not ts:
            ts = [1e-5]

        best_sc_local = best_sc
        best_f_local = None
        best_t = None

        for tv in ts:
            f_new = normalize((1.0 - tv) * f + tv * g)
            if f_new is None:
                continue
            sc = score(f_new)
            if np.isfinite(sc) and sc < best_sc_local:
                best_sc_local = sc
                best_f_local = f_new
                best_t = tv

        if best_f_local is not None:
            f = best_f_local
            best_f = f.copy()
            best_sc = best_sc_local
            stall = 0
            # Adapt threshold: if improving well, tighten
            if best_sc < M / (2.0 * n) - 0.001:
                active_threshold = max(0.9, active_threshold - 0.01)
            if it % 10 == 1:
                dt = time.monotonic() - t0
                print(f"  n={n} it={it} sc={best_sc:.6f} M={M:.5g} t={best_t:.4f} "
                      f"|active|={len(active)} dt={dt:.2f}s rem={remaining:.0f}s", flush=True)
        else:
            stall += 1
            if stall <= 3:
                f = normalize(f + 0.05 * rng.dirichlet(np.ones(n)))
            elif stall <= 6:
                f = subgradient_step(f, int(np.argmax(ac)), rng)
                if f is None:
                    f = normalize(0.8 * best_f + 0.2 * rng.dirichlet(np.ones(n)))
            else:
                f = normalize(0.7 * best_f + 0.3 * rng.dirichlet(np.ones(n)))
                active_threshold = 0.85
                stall = 0
            if stall > 18:
                break

    return best_f, best_sc


def coordinate_polish(f, time_budget, seed=42):
    """Final polish: cyclic coordinate updates."""
    f = normalize(f)
    if f is None:
        return f
    n = len(f)
    best_f = f.copy()
    best_sc = score(f)
    deadline = time.monotonic() + time_budget
    rng = np.random.default_rng(seed)

    while time.monotonic() < deadline:
        improved = False
        for i in range(n):
            if time.monotonic() > deadline:
                break
            # Try adjusting coordinate i
            val = f[i]
            if val < 1e-12:
                # Try adding a small amount
                candidates = [0.0, val * 1.5, 1.0 / n]
            else:
                candidates = [val * 0.5, val * 0.75, val * 1.25, val * 1.5, 0.0]

            for cv in candidates:
                f_try = f.copy()
                f_try[i] = max(0.0, cv)
                f_try = normalize(f_try)
                if f_try is None:
                    continue
                sc = score(f_try)
                if sc < best_sc - 1e-10:
                    f = f_try
                    best_f = f.copy()
                    best_sc = sc
                    improved = True
                    break

        if not improved:
            break

    return best_f


def propose_candidate(seed=42, budget_s=1500, **kwargs):
    """
    Search for a low-scoring nonnegative sequence using FFT-accelerated LP
    with active set selection, subgradient escape, and coordinate polishing.
    """
    rng = np.random.default_rng(int(seed) if seed is not None else 42)
    start_time = time.monotonic()
    total_deadline = start_time + 1480.0  # Leave 20s reserve

    best_f = None
    best_score = float("inf")
    given_f = None

    # Load given sequence
    try:
        raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        given_f = normalize(raw)
        if given_f is not None:
            best_f = given_f.copy()
            best_score = score(given_f)
            print(f"Given sequence: n={given_f.size}, score={best_score:.6f}", flush=True)
    except Exception:
        pass

    if best_f is None:
        best_f = np.ones(1024, dtype=np.float64) / 1024.0
        best_score = 2.0
        print("No given sequence; starting from uniform n=1024", flush=True)

    # Phase 1: Optimize at n=1024 (where the best is)
    n1 = 1024
    f1 = best_f if best_f.size == n1 else interpolate(best_f, n1)
    t1 = min(500.0, (total_deadline - time.monotonic()) - 300)

    if t1 > 10:
        f1_res, s1_res = optimize_lp(n1, f1, t1, seed=42)
        if s1_res < best_score:
            best_score = s1_res
            best_f = f1_res.copy()
            print(f"  Phase 1 result: score={best_score:.6f}", flush=True)

            best_score = s3_res
            best_f = f3_res.copy()
            print(f"  Phase 3 result: score={best_score:.6f}", flush=True)

    # Phase 4: Further optimize at the best length found
    n_best = best_f.size
    t4 = max(50.0, (total_deadline - time.monotonic()) - 30)
    if t4 > 20:
        print(f"\n=== Phase 4: Polish n={n_best}, budget={t4:.0f}s ===", flush=True)

    # Phase 5: Coordinate polish
    t5 = max(10.0, (total_deadline - time.monotonic()))

                print(f"  Coordinate polish: score={best_score:.6f}", flush=True)

    # Final evaluation
    seq = [float(max(0.0, min(1000.0, x))) for x in best_f]

            final_score = 2.0

    print(f"\nFINAL: n={len(seq)}, local_best={best_score:.6f}, "
          f"evaluate={final_score:.6f}, elapsed={elapsed:.1f}s", flush=True)
    return seq
