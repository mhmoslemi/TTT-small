    n = len(a)

    M = float(np.max(conv))

    return None, None, M

    conv = np.convolve(a, a)
    max_b = float(np.max(conv))
    sum_a = float(np.sum(a))
    if sum_a < 0.01:
        return np.inf
    return 2.0 * n * max_b / (sum_a * sum_a)


def build_A_subset(a, top_frac=0.6):
    """Build constraint matrix using only the top fraction of near-max constraints."""
    n = len(a)
    conv = np.convolve(a, a)
    M = float(np.max(conv))
    # Select top constraints (positions where conv is near max)
    n_constraints = 2 * n - 1
    k_select = max(n, int(n_constraints * top_frac))
    # Get indices of largest conv values
    idx = np.argpartition(conv, -k_select)[-k_select:]
    idx = idx[np.sort(idx)]
    
    A = np.zeros((len(idx), n), dtype=np.float64)
    for row, k in enumerate(idx):
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        if j_end >= j_start:
            A[row, j_start:j_end + 1] = a[k - j_end:k - j_start + 1][::-1]
    return A, conv[idx], M


def lp_step_active(a, top_frac=0.6, upper_bound=None):
    """LP: max sum(g) s.t. (a*g)[k] <= conv[k] for top-k positions, g >= 0."""
    n = len(a)
    if upper_bound is None:
        upper_bound = 10.0
    A, b_ub, M = build_A_subset(a, top_frac)
    c = -np.ones(n)
    bounds = [(0.0, upper_bound)] * n
    try:
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
        if res.status == 0:
            return res.x, float(-res.fun), M
    except Exception:
        pass
    return None, None, M


def subgradient_step(a, lr, threshold_frac=0.08):
    """One subgradient step to reduce max convolution."""
    n = len(a)
    conv = np.convolve(a, a)
    M = float(np.max(conv))
    S = float(np.sum(a))
    if S < 1e-10:
        return a
    
    # Select positions near the maximum
    threshold = M * (1.0 - threshold_frac)
    mask = (conv >= threshold).astype(np.float64)
    
    # Subgradient: SG[i] = sum_k mask[k] * a[k-i] for valid k-i
    SG = np.zeros(n)
    for i in range(n):
        lo = i
        hi = min(i + n, 2 * n - 1)
        if hi > lo:
            SG[i] = np.dot(mask[lo:hi], a[:hi - lo])
    
    # Update direction: increase where subgradient is low, decrease where high
    target = M / S
    a_new = a + lr * (target - SG)
    a_new = np.clip(a_new, 0.0, None)
    s_new = np.sum(a_new)
    if s_new < 1e-10:
        return a
    return a_new / s_new


def lp_optimize(a, max_iter=200, t=0.5, top_frac=0.6, verbose=False, rng=None):
    """LP-based optimization with active constraint subset."""
    if rng is None:
        rng = np.random.default_rng(42)
    
    n = len(a)
    best_a = a.copy()
    best_obj = compute_obj(best_a)
    stall = 0
    
    for it in range(max_iter):
        g, g_sum, M = lp_step_active(a, top_frac=top_frac)
        if g is None or g_sum <= 1.0 + 1e-5:
            stall += 1
            if stall > 12:
                # Perturb to escape
                noise_scale = 0.3 + 0.2 * (stall / 20.0)
                a = best_a * np.exp(rng.normal(0, noise_scale, n))
                a = np.clip(a, 0, None)
                s = np.sum(a)
                if s < 1e-10:
                    a = rng.random(n) + 0.1
                    s = np.sum(a)
                a = a / s
                best_a = a.copy()
                best_obj = compute_obj(best_a)
                stall = 0
            continue
        stall = 0
        
        g = g / g_sum
        # Adaptive blending: larger step when g_sum is much bigger than 1
        t_eff = min(0.9, t * min(1.0, g_sum / 2.0))
        a_new = (1.0 - t_eff) * a + t_eff * g
        a_new = np.clip(a_new, 0, None)
        s = np.sum(a_new)
        if s < 1e-10:
            break
        a_new = a_new / s
        
        obj = compute_obj(a_new)
        if obj < best_obj:
            best_a = a_new.copy()
            best_obj = obj
            if verbose and (it % 50 == 0 or it < 5):
                print(f"    LP it={it}: obj={best_obj:.6f} gsum={g_sum:.4f}")
        else:
            # Try smaller step
            a_new2 = (1.0 - t_eff * 0.3) * a + t_eff * 0.3 * g
            a_new2 = np.clip(a_new2, 0, None)
            s2 = np.sum(a_new2)
            if s2 > 1e-10:
                a_new2 = a_new2 / s2
                obj2 = compute_obj(a_new2)
                if obj2 < best_obj:
                    best_a = a_new2.copy()
                    best_obj = obj2
                    a = a_new2
                else:
                    a = a_new
            else:
                a = a_new
    
    return best_a, best_obj


def subgrad_polish(a, max_iter=5000, lr_init=0.08, lr_decay=0.9995, threshold_frac=0.05):
    """Subgradient polish after LP convergence."""
    n = len(a)
    best_a = a.copy()
    best_obj = compute_obj(best_a)
    lr = lr_init
    
    for it in range(max_iter):
        a_new = subgradient_step(a, lr, threshold_frac)
        obj = compute_obj(a_new)
        if obj < best_obj:
            best_a = a_new.copy()
            best_obj = obj
        a = a_new
        lr *= lr_decay
        if lr < 1e-6:
            break
    
    return best_a, best_obj


def random_block_refine(a, rng, max_iter=3000, block_scale=0.15):
    """Random block perturbation local search."""
    n = len(a)
    best_a = a.copy()
    best_obj = compute_obj(best_a)
    
    for it in range(max_iter):
        width = max(1, n // 50)
        width = min(width, n // 4 + 1)
        start = int(rng.integers(0, n - width + 1))
        candidate = best_a.copy()
        candidate[start:start + width] *= np.exp(rng.normal(0, block_scale, width))
        candidate = np.clip(candidate, 0, None)
        s = np.sum(candidate)
        if s < 1e-10:
            continue
        obj = compute_obj(candidate)
        if obj < best_obj:
            best_a = candidate
            best_obj = obj
    
    return best_a, best_obj


def init_random(n, rng):
    return rng.random(n) + 0.05


def init_cosine(n, rng):
    phase = rng.uniform(0, 2 * np.pi)
    freq = rng.uniform(0.3, 1.5)
    a = np.abs(np.cos(freq * np.pi * np.arange(n) / n + phase))
    a = a + 0.1 * rng.random(n)
    return a


def init_bimodal(n, rng):
    # High in middle, low at edges (or vice versa)
    center = np.arange(n) - n / 2.0
    sigma = n * rng.uniform(0.15, 0.4)
    a = np.exp(-center ** 2 / (2 * sigma ** 2))
    a = a + 0.1 * rng.random(n)
    return a


def init_two_level(n, rng):
    # Two-level: most values are 1, some are c
    a = np.ones(n)
    n_high = int(n * rng.uniform(0.1, 0.4))
    positions = rng.choice(n, size=n_high, replace=False)
    a[positions] = rng.uniform(0.3, 0.7, n_high)
    # Randomize slightly
    a = a + 0.05 * rng.random(n)
    return a


def init_periodic(n, rng):
    period = int(rng.integers(max(2, n // 15), max(3, n // 4)))
    pattern = np.abs(rng.normal(1, 0.4, period))
    a = np.tile(pattern, n // period + 1)[:n]
    return a + 0.05 * rng.random(n)


def init_from_parent(parent, n, rng):
    """Resize parent to length n via interpolation."""
    m = len(parent)
    if m == n:
        return parent.copy() * np.exp(rng.normal(0, 0.1, n)) + 0.01
    # Interpolate
    x_old = np.linspace(0, 1, m)
    x_new = np.linspace(0, 1, n)
    a = np.interp(x_new, x_old, parent)
    a = np.clip(a, 0.01, None) + 0.05 * rng.random(n)
    return a


def propose_candidate(seed=42, budget_s=1500, **kwargs):
    """
    Hybrid LP + subgradient optimization with constraint subset LP,
    diverse initializations, and multi-length search.
    """
    rng = np.random.default_rng(seed)
    start_time = time.monotonic()
    deadline = start_time + float(budget_s) - 3.0

    best_seq = None
    best_score = np.inf

    # Try parent sequence
    parent = None
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64)
        parent = np.clip(parent, 0, 1000.0)
        if parent.size > 0 and np.isfinite(parent).all():
            obj = compute_obj(parent)
            if obj < best_score:
                best_score = obj
                best_seq = parent.copy()
            print(f"Parent (n={parent.size}): obj={obj:.6f}")
    except Exception:
        pass

    # Sequence lengths to try
    n_values = [100, 130, 150, 180, 220, 300, 400]
    
    # Initialize functions
    inits = [init_random, init_cosine, init_bimodal, init_two_level, init_periodic]

    for n in n_values:
        if time.monotonic() > deadline:
            break
        remaining = deadline - time.monotonic()
        print(f"\n=== n={n}, remaining={remaining:.0f}s ===")
        
        n_restarts = max(2, int(remaining / 80))
        n_restarts = min(n_restarts, 6)
        
        for restart in range(n_restarts):
            if time.monotonic() > deadline:
                break
            
            # Choose initialization
            if restart == 0 and parent is not None:
                a_init = init_from_parent(parent, n, rng)
            else:
                init_fn = inits[restart % len(inits)]
                a_init = init_fn(n, rng)
            
            s = np.sum(a_init)
            if s > 0:
                a_init = a_init / s
            a_init = a_init * np.sqrt(2.0 * n)  # Scale for reasonable magnitude

            # LP optimization
            time_per_restart = (deadline - time.monotonic()) / max(1, n_restarts - restart)
            max_iter = int(min(300, max(30, time_per_restart / 0.3)))
            
            a_opt, obj = lp_optimize(a_init, max_iter=max_iter, t=0.5,
                                     top_frac=0.55, verbose=(restart == 0), rng=rng)
            if obj < best_score:
                best_score = obj
                best_seq = a_opt.copy()
                print(f"  LP n={n} r={restart}: obj={best_score:.6f}")
            
            # Subgradient polish
            if time.monotonic() < deadline - 10:
                sg_iters = int(min(5000, (deadline - time.monotonic()) / 0.02))
                a_sg, obj_sg = subgrad_polish(a_opt, max_iter=sg_iters,
                                              lr_init=0.06, lr_decay=0.9997,
                                              threshold_frac=0.05)
                if obj_sg < best_score:
                    best_score = obj_sg
                    best_seq = a_sg.copy()
                    print(f"  SG n={n} r={restart}: obj={best_score:.6f}")
                else:
                    a_opt = a_sg  # Still use the polished version for next steps
            
            # Random block refinement
            if time.monotonic() < deadline - 5:
                rb_iters = int(min(3000, (deadline - time.monotonic()) / 0.03))
                a_rb, obj_rb = random_block_refine(a_opt, rng, max_iter=rb_iters,
                                                   block_scale=0.12)
                if obj_rb < best_score:
                    best_score = obj_rb
                    best_seq = a_rb.copy()
                    print(f"  RB n={n} r={restart}: obj={best_score:.6f}")

    # Final intensive polish on the best result
    if best_seq is not None and time.monotonic() < deadline - 5:
        remaining = deadline - time.monotonic()
        print(f"\n=== Final polish, remaining={remaining:.0f}s ===")
        n_best = len(best_seq)
        
        # More LP with tighter constraint subset
        max_iter = int(min(200, remaining / 0.4))
        a_fin, obj_fin = lp_optimize(best_seq, max_iter=max_iter, t=0.4,
                                     top_frac=0.4, verbose=True, rng=rng)
        if obj_fin < best_score:
            best_score = obj_fin
            best_seq = a_fin.copy()
            print(f"  Final LP: obj={best_score:.6f}")
        else:
            a_fin = a_fin  # keep for next step
        
        # Subgradient with very small steps
        if time.monotonic() < deadline - 3:
            sg_iters = int((deadline - time.monotonic()) / 0.02)
            a_sg2, obj_sg2 = subgrad_polish(a_fin, max_iter=sg_iters,
                                            lr_init=0.03, lr_decay=0.9998,
                                            threshold_frac=0.03)
            if obj_sg2 < best_score:
                best_score = obj_sg2
                best_seq = a_sg2.copy()
                print(f"  Final SG: obj={best_score:.6f}")
        
        # Random refinement
        if time.monotonic() < deadline - 1:
            rb_iters = int((deadline - time.monotonic()) / 0.03)
            a_rb2, obj_rb2 = random_block_refine(best_seq, rng, max_iter=rb_iters,
                                                 block_scale=0.08)
            if obj_rb2 < best_score:
                best_score = obj_rb2
                best_seq = a_rb2.copy()
                print(f"  Final RB: obj={best_score:.6f}")

    # Fallback
    if best_seq is None:
        best_seq = np.ones(100)
        best_score = compute_obj(best_seq)

    elapsed = time.monotonic() - start_time
    print(f"\nDONE: n={len(best_seq)}, score={best_score:.6f}, time={elapsed:.1f}s")
    return [float(x) for x in best_seq]
