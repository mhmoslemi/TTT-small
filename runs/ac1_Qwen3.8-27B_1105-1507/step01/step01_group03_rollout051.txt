    # grad_j/2 = sum_{i=0}^{n-1} f[i] * active_arr[j+i]
    # This is the cross-correlation of f and active_arr at lag j.
    # np.correlate(active_arr, f, mode='valid') gives sum_i active_arr[i+j] * f[i] for j=0,...,len(active_arr)-n
    # Since len(active_arr) = 2n-1, we get n results for j=0,...,n-1. 
    
    # Wait: np.correlate(a, v, mode='valid') computes:
    # out[j] = sum_i a[i+j] * conj(v[i]) for j = 0 to len(a)-len(v)
    # So np.correlate(active_arr, f, mode='valid'):
    # out[j] = sum_i active_arr[i+j] * f[i] for i=0 to n-1 (since len(f)=n)
    # out has length (2n-1) - n + 1 = n. 
    # out[j] = sum_{i=0}^{n-1} active_arr[i+j] * f[i] = grad_j / 2
    
    # Perfect!
    grad_half = np.correlate(active_arr, f, mode='valid')
    grad = 2.0 * grad_half
    
    gnorm_sq = float(np.dot(grad, grad))
    if gnorm_sq <= 1e-300:
        return f, M, grad
    
    eta = eta_frac * M / gnorm_sq
    f_new = f - eta * grad
    f_new = project_simplex(f_new)
    return f_new, M, grad


def subgradient_phase(f, time_budget, rng, verbose=True):
    """Run projected subgradient descent for the given time budget."""
    n = len(f)
    f = normalize_sequence(f).copy()
    best_f = f.copy()
    best_score = score_normalized(f, n)
    start_time = time.monotonic()
    deadline = start_time + time_budget
    
    # Diminishing step factor
    base_eta = 0.5
    iterate = 0
    no_improve = 0
    
    while time.monotonic() < deadline:
        iterate += 1
        
        # Try several step sizes
        eta_fracs = [base_eta, base_eta * 0.5, base_eta * 0.25, base_eta * 0.1, base_eta * 2.0]
        
        best_cand = None
        best_cand_score = best_score
        
        for ef in eta_fracs:
            f_new, M, grad = subgradient_step(f, ef)
            if f_new is None:
                continue
            sc = score_normalized(f_new, n)
            if np.isfinite(sc) and sc < best_cand_score:
                best_cand_score = sc
                best_cand = f_new
        
        if best_cand is not None:
            f = best_cand
            if best_cand_score < best_score:
                best_f = f.copy()
                best_score = best_cand_score
                no_improve = 0
                if verbose and (iterate <= 5 or iterate % 100 == 0):
                    print(f"    subgrad iter={iterate}, score={best_score:.6f}", flush=True)
            else:
                no_improve += 1
        else:
            # No improvement, try perturbation
            no_improve += 1
            if no_improve > 50:
                # Restart with perturbation
                f = project_simplex(f + 0.01 * rng.standard_normal(n))
                no_improve = 0
                base_eta = max(0.05, base_eta * 0.7)
                continue
        
        # Adaptively decrease step
        if iterate % 100 == 0:
            base_eta *= 0.95
    
    return best_f, best_score


def make_conv_template(n):
    """Precompute the CSR matrix structure for convolve(f, g)."""
    m = 2 * n - 1
    k64 = np.arange(m, dtype=np.int64)
    js = np.maximum(0, k64 - n + 1).astype(np.int32)
    je = np.minimum(n - 1, k64).astype(np.int32)
    counts = (je.astype(np.int64) - js.astype(np.int64) + 1).astype(np.int64)
    total = int(counts.sum())
    
    cum = np.cumsum(counts)
    starts = np.repeat(cum - counts, counts)
    local = np.arange(total, dtype=np.int64) - starts
    
    row = np.repeat(k64, counts).astype(np.int32)
    col_base = js[row.astype(np.int64)]
    col = (col_base + local).astype(np.int32)
    offsets = (row.astype(np.int64) - col.astype(np.int64)).astype(np.int32)
    
    A_off = csr_matrix((offsets, (row, col)), shape=(m, n), dtype=np.int32)
    edge_offsets = A_off.data.astype(np.int32, copy=True)
    A_template = A_off.astype(np.float64).copy()
    A_template.data.fill(0.0)
    
    c_obj = -np.ones(n, dtype=np.float64)
    bounds = [(0.0, None)] * n
    return A_template, edge_offsets, c_obj, bounds, m


def solve_direction_lp(template, f, n, M):
    """Solve the LP direction problem."""
    A_template, edge_offsets, c_obj, bounds, m = template
    lp_scale = float(n)
    
    A = A_template.copy()
    A.data[:] = f[edge_offsets]
    A.data *= lp_scale
    
    b = np.full(m, M * lp_scale * lp_scale, dtype=np.float64)
    return linprog(c_obj, A_ub=A, b_ub=b, bounds=bounds, method="highs")


def lp_phase(f_init, time_budget, rng, template, verbose=True):
    """Run LP-based iteration."""
    f = normalize_sequence(f_init)
    n = len(f)
    
    if template is None:
        try:
            template = make_conv_template(n)
        except Exception:
            return f, score_normalized(f, n)
    
    best_f = f.copy()
    best_score = score_normalized(f, n)
    start_time = time.monotonic()
    deadline = start_time + time_budget
    stall = 0
    iteration = 0
    
    while time.monotonic() < deadline:
        iteration += 1
        t0 = time.monotonic()
        
        conv_ff = np.convolve(f, f)
        M = float(np.max(conv_ff))
        if not np.isfinite(M) or M <= 0:
            break
        
        try:
            res = solve_direction_lp(template, f, n, M)
        except Exception:
            break
        
        x = getattr(res, "x", None)
        if x is None:
            stall += 1
            if stall > 15:
                break
            f = project_simplex(f + 0.05 * rng.standard_normal(n))
            continue
        
        x = np.asarray(x, dtype=np.float64).ravel()
        if x.size != n:
            break
        x = np.where(np.isfinite(x), x, 0.0)
        x = np.maximum(x, 0.0)
        s_star = float(x.sum())
        if s_star <= 1e-300:
            stall += 1
            if stall > 15:
                break
            f = project_simplex(f + 0.05 * rng.standard_normal(n))
            continue
        
        s_ratio = s_star / float(n)
        if s_ratio <= 1.0 + 1e-8:
            stall += 1
            if stall > 15:
                break
            f = project_simplex(f + 0.05 * rng.standard_normal(n))
            continue
        
        g = x / s_star
        r = np.convolve(f, g)
        r_max = float(np.max(r))
        if not np.isfinite(r_max) or r_max <= 0:
            stall += 1
            if stall > 15:
                break
            f = project_simplex(f + 0.05 * rng.standard_normal(n))
            continue
        
        if r_max >= M * (1.0 - 1e-10):
            stall += 1
            if stall > 15:
                break
            f = project_simplex(f + 0.05 * rng.standard_normal(n))
            continue
        
        # Safe step
        G = float(np.max(np.convolve(g, g)))
        if not np.isfinite(G):
            stall += 1
            if stall > 15:
                break
            f = project_simplex(f + 0.05 * rng.standard_normal(n))
            continue
        
        a_coef = 2.0 * (r_max - M)
        b_coef = G + M - 2.0 * r_max
        if b_coef > 1e-14:
            t_max = -a_coef / b_coef
        else:
            t_max = 1.0
        
        if not np.isfinite(t_max) or t_max <= 0:
            t_max = 1e-8
        base = min(1.0, 0.9 * t_max)
        if base < 1e-12:
            base = 1e-12
        
        # Try multiple step sizes
        ts = []
        for frac in [1.0, 0.9, 0.8, 0.7, 0.5, 0.3, 0.1]:
            tv = frac * base
            if tv > 0 and tv <= 1.0:
                ts.append(tv)
        
        best_cand_f = None
        best_cand_score = best_score
        best_t = None
        
        for tv in ts:
            f_new = (1.0 - tv) * f + tv * g
            f_new = project_simplex(f_new)
            sc = score_normalized(f_new, n)
            if np.isfinite(sc) and sc < best_cand_score:
                best_cand_score = sc
                best_cand_f = f_new
                best_t = tv
        
        if best_cand_f is not None and best_cand_score < best_score:
            f = best_cand_f
            best_f = f.copy()
            best_score = best_cand_score
            stall = 0
            if verbose and (iteration <= 5 or iteration % 10 == 0):
                print(f"    LP iter={iteration}, score={best_score:.6f}, dt={time.monotonic()-t0:.2f}s", flush=True)
        else:
            stall += 1
            f = project_simplex(f + 0.05 * rng.standard_normal(n))
            if stall > 15:
                break
    
    return best_f, best_score


def coordinate_descent_phase(f_init, time_budget, rng, verbose=True):
    """
    Pairwise coordinate descent: pick pairs (j, l), vary the split of their
    combined mass, and update to minimize the max convolution term.
    """
    f = normalize_sequence(f_init).copy()
    n = len(f)
    best_f = f.copy()
    best_score = score_normalized(f, n)
    start_time = time.monotonic()
    deadline = start_time + time_budget
    
    # Precompute which k positions are affected by each pair (j, l)
    # For coordinate j: affects k = j (when i=0, k-i=j), k = j+1, ..., k = j+n-1
    # Actually (f*f)_k involves f_j when j <= k < j+n (i.e., k-j in [0, n-1]) or k-j < 0 and j-k in [0,n-1]
    # More precisely: (f*f)_k = sum_i f_i * f_{k-i}
    # f_j appears when i=j (needs 0<=k-j<n) or when k-i=j i.e. i=k-j (needs 0<=k-j<n)
    # So f_j affects conv[k] for k in [j, j+n-1] (from i=j) and k in [j, j+n-1] (from i=k-j)
    # Actually both give k-j in [0,n-1], so k in [j, j+n-1].
    # Wait: from i=j: k-j in [0,n-1] means k in [j, j+n-1]
    # From i=k-j: k-j in [0,n-1] means k in [j, j+n-1]
    # So f_j affects conv[k] for k in [j, j+n-1] (capped at [0, 2n-2])
    
    # For efficiency, I'll do a simple random pairwise CD:
    
    no_improve = 0
    iteration = 0
    
    while time.monotonic() < deadline:
        iteration += 1
        improved = False
        
        # Random pairs
        pairs = rng.integers(0, n, size=(200, 2))
        
        for idx in range(len(pairs)):
            if time.monotonic() > deadline:
                break
            j, l = int(pairs[idx][0]), int(pairs[idx][1])
            if j == l:
                continue
            
            s = f[j] + f[l]
            if s < 1e-15:
                continue
            
            # f[j] = t, f[l] = s - t, t in [0, s]
            # conv[k] for affected k:
            # For k affected by j: conv[k] = A_k + f[j]*B_k + f[l]*C_k (simplified)
            # Actually let me just evaluate directly for a few t values.
            
            # The affected k values are those where j or l appears
            k_affected_j = set(range(max(0, j), min(2*n-1, j + n)))
            k_affected_l = set(range(max(0, l), min(2*n-1, l + n)))
            k_affected = k_affected_j | k_affected_l
            
            # For efficiency, compute the "base" convolution (with f[j]=f[l]=0)
            # and the linear/quadratic terms.
            
            # (f*f)_k with f[j]=t, f[l]=s-t:
            # Base: all terms not involving f[j] or f[l]
            # + terms involving f[j]: f[j]*f[k-j] + f[k-j]*f[j] - (if k=2j: f[j]^2 counted once)
            # This gets complicated. Let me just do a simple ternary search.
            
            old_j, old_l = f[j], f[l]
            
            # Try a few t values
            best_t = old_j
            best_M = float('inf')
            
            # Sample t values
            t_candidates = np.linspace(0, s, 20)
            # Also include the current value
            t_candidates = np.unique(np.concatenate([t_candidates, [old_j]]))
            
            for t in t_candidates:
                tl = s - t
                if t < -1e-15 or tl < -1e-15:
                    continue
                f[j], f[l] = max(0, t), max(0, tl)
                # Only compute affected convolution terms
                conv_affected = np.convolve(f, f)  # Full computation (slow but simple)
                # Actually for n=1024 this is ~2M ops per evaluation, and we do 20 per pair, 200 pairs = 8 billion. Too slow.
                f[j], f[l] = old_j, old_l
                break  # This approach is too slow for full convolution
            
            # Better: compute only affected terms
            f[j], f[l] = max(0, old_j), max(0, old_l)
            
            # For each t in t_candidates, compute max over affected k only
            # (f*f)_k = sum_i f_i * f_{k-i}
            # With f[j]=t, f[l]=s-t:
            # = sum_{i not in {j,l}} f_i * f_{k-i} + contributions from j and l
            
            # Let me precompute the "rest" convolution (excluding j and l):
            f_rest = f.copy()
            f_rest[j] = 0
            f_rest[l] = 0
            conv_rest = np.convolve(f_rest, f_rest)
            
            # Now (f*f)_k = conv_rest[k] + t*f[k-j] + t*f[k-j] (from i=j)
            # Wait, I need to be more careful.
            # (f*f)_k = sum_i f_i * f_{k-i}
            # = sum_{i not j,l} f_i * f_{k-i} + f_j * f_{k-j} + f_l * f_{k-l}
            #   + [if k-j != l and k-j not = j: f_{k-j} * f_j]  -- wait this is double counting
            # Actually: the sum includes ALL pairs (i, k-i). f_j appears as f_i when i=j, and as f_{k-i} when k-i=j (i.e., i=k-j).
            # Similarly for f_l.
            
            # Let me just compute it properly:
            # (f*f)_k with f[j]=t, f[l]=s-t:
            # = conv_rest[k] + t * f[k-j]_{rest_or_l} + (s-t) * f[k-l]_{rest_or_j}
            #   + t * f_{k-j} (where f_{k-j} might be t or s-t if k-j is j or l)
            # This is getting complicated. Let me use a simpler but still fast approach.
            
            # Fast approach: for a given (t, s-t), compute conv at affected k only.
            # For a specific k: (f*f)_k = sum_{i=max(0,k-n+1)}^{min(n-1,k)} f_i * f_{k-i}
            # With f[j]=t and f[l]=s-t, and f_rest for others.
            # = sum_{i in range, i not j,l} f_rest[i] * f[k-i] + t * f[k-j] + (s-t) * f[k-l]
            #   - [corrections if k-j == j or k-j == l, etc.]
            
            # This is getting too complex for inline computation. Let me use a different strategy:
            # Instead of pairwise CD, use a simple "swap" approach:
            # Move mass from j to l (or vice versa) in small increments.
            
            delta = s * 0.1  # Move 10% of the mass
            for direction in [1, -1]:
                new_j = old_j + direction * delta
                new_l = old_l - direction * delta
                if new_j < 0 or new_l < 0:
                    continue
                f[j], f[l] = new_j, new_l
                conv_test = np.convolve(f, f)
                M_test = float(np.max(conv_test))
                f[j], f[l] = old_j, old_l
                if M_test < best_M:
                    best_M = M_test
                    best_t = old_j + direction * delta
            
            if best_t != old_j:
                f[j] = best_t
                f[l] = s - best_t
                sc = score_normalized(f, n)
                if sc < best_score:
                    best_score = sc
                    best_f = f.copy()
                    improved = True
                no_improve = 0
            else:
                no_improve += 1
        
        if no_improve > 5:
            # Add noise
            f = project_simplex(f + 0.01 * rng.standard_normal(n))
            no_improve = 0
    
    return best_f, best_score
