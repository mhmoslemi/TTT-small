def _fft_convolve(f):

    while size < 2 * n - 1:
        size <<= 1
    F = np.fft.rfft(f, size)
    h = np.fft.irfft(F * F, size)
    return h[:2 * n - 1]

def _score(f):

def _subgradient(f, active_lags):
    """Compute subgradient of max_k conv(f,f)[k] w.r.t. f."""
    n = len(f)
    grad = np.zeros(n)
    for k in active_lags:
        # d/dx[j] of sum_i x[i]*x[i+k] = x[j+k] + x[j-k] (when valid)
        if k == 0:
            grad += 2.0 * f
        else:
            # Forward: f[j+k] for j = 0..n-1-k
            if k < n:
                np.add.at(grad, np.arange(n - k), f[k:])
            # Backward: f[j-k] for j = k..n-1
            np.add.at(grad, np.arange(k, n), f[:n-k])
    grad /= float(len(active_lags))
    return grad

"""
Optimizes non-negative step function to minimize 2*n*max(conv(f,f))/(sum(f))^2.
Uses enhanced multi-start mirror descent with adaptive active sets,
LP-based Frank-Wolfe refinement, projected subgradient descent,
and spectral-domain equalization for final polishing.

import numpy as np
import time
from scipy.optimize import linprog
from scipy.sparse import coo_matrix


def _fft_convolve(f):
    """Compute full linear convolution f*f using FFT."""
    n = len(f)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    F = np.fft.rfft(f, size)
    h = np.fft.irfft(F * F, size)
    return h[:2 * n - 1]


def _score(f):
    """Fast evaluation of the objective (assumes f>=0, sum(f)>=0.01)."""
    n = len(f)
    S = float(np.sum(f))
    if S < 0.01:
        return float('inf')
    h = _fft_convolve(f)
    M = float(np.max(h))
    return 2.0 * n * M / (S * S)


def _normalize(f, target=1.0):
    """Clip to non-negative and normalize to target sum."""
    f = np.maximum(np.asarray(f, dtype=np.float64), 0.0)
    s = float(np.sum(f))
    if s < 1e-30:
        return None
    return f * target / s


def _subgradient_autocorr(f, active_lags):
    """Compute subgradient of max_k (f*f)[k] w.r.t. f at given active lags."""
    n = len(f)
    grad = np.zeros(n, dtype=np.float64)
    for k in active_lags:
        k = int(k)
        if k == 0:
            grad += 2.0 * f
        elif k < n:
            # Forward shift: contributes f[j+k] to grad[j] for j=0..n-1-k
            grad[:n - k] += f[k:]
            # Backward shift: contributes f[j-k] to grad[j] for j=k..n-1
            grad[k:] += f[:n - k]
    grad /= float(len(active_lags))
    return grad


def _project_simplex(v):
    """Project v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - 1))[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho = rho[-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0)


def _mirror_descent(f0, n_iter, eta0, decay_rate, thresh_frac, time_limit, label=""):
    """
    Mirror descent with multiplicative updates for min max(conv(f,f)) s.t. sum(f)=1, f>=0.
    Adaptive step size with geometric decay and dynamic active set.
    """
    f = _normalize(np.asarray(f0, dtype=np.float64))
    if f is None:
        return np.asarray(f0, dtype=np.float64).copy(), float('inf')
    n = len(f)
    best_f = f.copy()
    best_score = _score(f)
    start = time.monotonic()

    for it in range(n_iter):
        if time.monotonic() - start > time_limit:
            break

        h = _fft_convolve(f)
        M = float(np.max(h))
        if M <= 1e-15:
            break

        # Dynamic threshold: start broad, narrow over iterations
        progress = it / max(1, n_iter)
        tf = thresh_frac + 0.15 * progress  # gradually increase threshold
        tf = min(tf, 0.99)
        threshold = tf * M + 1e-15
        active = np.where(h >= threshold)[0]
        if len(active) == 0:
            active = np.array([int(np.argmax(h))])

        grad = _subgradient_autocorr(f, active)

        # Adaptive step size: geometric decay
        eta = eta0 / ((it + 1) ** decay_rate) / (M + 1e-10)
        eta = min(eta, 5.0)

        log_update = -eta * grad
        log_update = np.clip(log_update, -25.0, 25.0)

        f = f * np.exp(log_update)
        f = _normalize(f)
        if f is None:
            f = np.ones(n) / n
            break

        s = _score(f)
        if s < best_score:
            best_score = s
            best_f = f.copy()

    return best_f, best_score


def _projected_subgrad(f0, n_iter, eta0, decay_rate, thresh_frac, time_limit):
    """
    Projected subgradient descent on the simplex.
    Uses the subgradient of max(conv(f,f)) and projects onto the simplex.
    """
    f = _normalize(np.asarray(f0, dtype=np.float64))
    if f is None:
        return np.asarray(f0, dtype=np.float64).copy(), float('inf')
    n = len(f)
    best_f = f.copy()
    best_score = _score(f)
    start = time.monotonic()

    for it in range(n_iter):
        if time.monotonic() - start > time_limit:
            break

        h = _fft_convolve(f)
        M = float(np.max(h))
        if M <= 1e-15:
            break

        threshold = thresh_frac * M + 1e-15

        f_raw = f - eta * grad
        f = _project_simplex(f_raw)

        s = _score(f)
        if s < best_score:
            best_score = s

def _frank_wolfe_lp(f0, max_iters, time_limit):
    """
    LP-based Frank-Wolfe for: min max_k conv(f,g)[k] s.t. sum(g)=1, g>=0
    where f is fixed and we find the best direction g.
    """
    f = _normalize(np.asarray(f0, dtype=np.float64))
    if f is None:
        return np.asarray(f0, dtype=np.float64).copy(), float('inf')
    n = len(f)
    best_f = f.copy()
    best_score = _score(f)
    start = time.monotonic()

    for it in range(max_iters):
        if time.monotonic() - start > time_limit:
            break

        h = np.convolve(f, f)
        M = float(np.max(h))

        # Build sparse constraint matrix: A[k,j] = f[k-j] if 0<=k-j<n
        row_idx = []
        col_idx = []
        vals = []
        for j in range(n):
            lo = max(0, j - (n - 1))
            hi = min(2 * n - 2, j + (n - 1))
            for k in range(lo, hi + 1):
                idx = k - j
                if 0 <= idx < n:
                    row_idx.append(k)
                    col_idx.append(j)
                    vals.append(f[idx])

        C = coo_matrix((vals, (row_idx, col_idx)), shape=(2 * n - 1, n))

        c_obj = -np.ones(n)
        try:
            res = linprog(c_obj, A_ub=C, b_ub=np.full(2 * n - 1, M),
                          bounds=[(0, None)] * n, method='highs',
                          options={'presolve': True, 'time_limit': max(1.0, time_limit / (max_iters - it))})
        except Exception:
            break

        if not res.success or res.x is None:
            break

        g = np.maximum(res.x, 0)
        Sg = float(np.sum(g))
        if Sg < 1e-10:
            break
        g = g / Sg

        # Frank-Wolfe step
        t = 2.0 / (it + 2)
        f_new = (1.0 - t) * f + t * g
        f = _normalize(f_new)
        if f is None:
            break

        s = _score(f)
        if s < best_score:
            best_score = s
            best_f = f.copy()

        if (it + 1) % 5 == 0:
            print(f"      FW iter {it+1}: score={best_score:.6f}")

    return best_f, best_score


def _spectral_refine(f0, n_iter, time_limit):
    """
    Spectral-domain refinement: iterate between time-domain non-negativity
    projection and frequency-domain equalization to reduce autocorrelation peak.
    """
    f = _normalize(np.asarray(f0, dtype=np.float64))
    if f is None:
        return np.asarray(f0, dtype=np.float64).copy(), float('inf')
    n = len(f)
    best_f = f.copy()
    best_score = _score(f)
    start = time.monotonic()

    size = 1
    while size < 2 * n - 1:
        size <<= 1

    for it in range(n_iter):
        if time.monotonic() - start > time_limit:
            break

        h = _fft_convolve(f)
        M = float(np.max(h))

        # Compute the "excess" autocorrelation above the mean level
        mean_level = 1.0 / (2 * n - 1)  # target flat level for sum(f)=1
        excess = np.maximum(h - M * 0.3, 0.0)

        # Frequency domain: modify spectrum to reduce excess
        F = np.fft.rfft(f, size)
        mag = np.abs(F)
        phase = np.angle(F)

        # Compute which frequency bins contribute to the excess
        F_excess = np.fft.rfft(np.concatenate([excess, np.zeros(size - len(excess))]), size)
        # Reduce spectral components that correlate with the excess
        cross = np.real(F * np.conj(F_excess))
        # Damp frequencies that contribute positively to the excess
        damp = 1.0 - 0.1 * np.maximum(cross, 0) / (np.max(np.abs(cross)) + 1e-10)
        damp = np.clip(damp, 0.5, 1.0)

        new_mag = mag * damp
        new_F = new_mag * np.exp(1j * phase)
        f_new = np.fft.irfft(new_F, size)[:n]

        # Enforce non-negativity
        f_new = np.maximum(f_new, 0.0)
        f = _normalize(f_new)
        if f is None:
            f = best_f.copy()
            break

        s = _score(f)
        if s < best_score:
            best_score = s
            best_f = f.copy()

    return best_f, best_score


def _cosine_init(n):
    j = np.arange(n, dtype=np.float64)
    f = np.cos(np.pi * (j + 0.5) / float(n)) ** 2
    f = np.maximum(f, 0.0)
    s = float(np.sum(f))
    if s > 0:
        f /= s
    return f


def _gaussian_init(n, sigma_frac=0.3):
    j = np.arange(n, dtype=np.float64) - (n - 1) / 2.0
    sigma = sigma_frac * float(n)
    f = np.exp(-j * j / (2.0 * sigma * sigma))
    s = float(np.sum(f))
    if s > 0:
        f /= s
    return f


def _tapered_init(n, power=2.0):
    j = np.arange(n, dtype=np.float64)
    x = np.sin(np.pi * (j + 0.5) / float(n))
    f = x ** power
    f = np.maximum(f, 0.0)
    s = float(np.sum(f))
    if s > 0:
        f /= s
    return f


def propose_candidate(seed=42, budget_s=1500, **kwargs):
    """
    Multi-phase search for non-negative step function heights minimizing
    2*n*max(conv(f,f))/(sum(f))^2.
    
    Phase 1: Multi-start mirror descent with varied parameters.
    Phase 2: Projected subgradient descent from best Phase 1 solutions.
    Phase 3: LP Frank-Wolfe refinement.
    Phase 4: Spectral refinement and perturbation restarts.
    Phase 5: Try different sequence lengths.
    """
    deadline = time.monotonic() + float(budget_s) - 10.0
    rng = np.random.default_rng(seed)

    # Get parent sequence
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64).copy()
        parent = np.clip(parent, 0.0, 1000.0)
    except Exception:
        parent = np.ones(512, dtype=np.float64)

    best_f = parent.copy()
    best_score = float(evaluate_sequence(best_f.tolist()))
    print(f"Parent: n={len(best_f)}, score={best_score:.6f}")

    parent_norm = _normalize(parent)
    if parent_norm is None:
        parent_norm = np.ones(len(parent)) / len(parent)

    results = []  # list of (score, sequence, name)

    def _update_best(f, score, name):
        nonlocal best_f, best_score
        if score < best_score:
            best_score = score
            best_f = f.copy()
            print(f"  ** NEW BEST: {best_score:.6f} ({name}) **")
        results.append((score, f.copy(), name))

    # === PHASE 1: Multi-start mirror descent ===
    print("\n=== Phase 1: Multi-start Mirror Descent ===")
    p1_deadline = deadline - 600.0  # Leave 600s for later phases

    # Build initial candidates
    n_parent = len(parent_norm)
    inits = [("parent", parent_norm)]

    for n_try in [n_parent, 1024, 2048]:
        inits.append(("cos%d" % n_try, _cosine_init(n_try)))
        inits.append(("gauss%d" % n_try, _gaussian_init(n_try, 0.3)))
        inits.append(("taper%d" % n_try, _tapered_init(n_try, 2.0)))

    # Add perturbed parent
    for pf in [0.05, 0.15]:
        pert = _normalize(parent_norm + rng.random(n_parent) * pf)
        if pert is not None:
            inits.append(("pert%d" % int(pf * 100), pert))

    print(f"  {len(inits)} initial candidates")

    md_configs = [
        (1.0, 0.3, 0.5),
        (0.7, 0.4, 0.7),
        (0.5, 0.5, 0.85),
        (1.5, 0.2, 0.6),
        (0.8, 0.6, 0.9),
    ]

    for init_name, f0 in inits:
        if time.monotonic() > p1_deadline:
            break
        time_left = p1_deadline - time.monotonic()
        if time_left < 5:
            break

        n_iter = int(min(20000, time_left * 50))
        for eta0, decay, tf in md_configs[:3]:
            if time.monotonic() > p1_deadline:
                break
            t_left = p1_deadline - time.monotonic()
            if t_left < 5:
                break
            f_res, s_res = _mirror_descent(
                f0, n_iter=n_iter, eta0=eta0, decay_rate=decay,
                thresh_frac=tf, time_limit=t_left * 0.3,
                label=f"{init_name}_md{eta0}"
            )
            _update_best(f_res, s_res, f"{init_name}_md{eta0}")

    # Sort results so far
    results.sort(key=lambda x: x[0])
    print(f"  Phase 1 best: {results[0][0]:.6f}")

    # === PHASE 2: Projected subgradient from top solutions ===
    print("\n=== Phase 2: Projected Subgradient Descent ===")
    p2_deadline = deadline - 400.0

    top_k = min(3, len(results))
    for idx in range(top_k):
        if time.monotonic() > p2_deadline:
            break
        s_cur, f_cur, name = results[idx]
        time_left = p2_deadline - time.monotonic()
        if time_left < 5:
            break

        for eta0 in [0.01, 0.005, 0.02]:
            if time.monotonic() > p2_deadline:
                break
            t_left = p2_deadline - time.monotonic()
            if t_left < 5:
                break
            f_res, s_res = _projected_subgrad(
                f_cur, n_iter=10000, eta0=eta0, decay_rate=0.5,
                thresh_frac=0.8, time_limit=min(t_left * 0.3, 30.0)
            )
            _update_best(f_res, s_res, f"{name}_pgd{eta0}")

    results.sort(key=lambda x: x[0])
    print(f"  Phase 2 best: {results[0][0]:.6f}")

    # === PHASE 3: LP Frank-Wolfe ===
    print("\n=== Phase 3: LP Frank-Wolfe ===")
    p3_deadline = deadline - 200.0

    top_k = min(2, len(results))
    for idx in range(top_k):
        if time.monotonic() > p3_deadline:
            break
        s_cur, f_cur, name = results[idx]
        n = len(f_cur)
        if n > 1500:
            print(f"  Skipping LP for n={n} (too large)")
            continue

        time_left = min(120.0, p3_deadline - time.monotonic())
        if time_left < 15:
            break
        print(f"  LP FW on {name} (n={n}, score={s_cur:.6f}), budget={time_left:.0f}s")
        f_res, s_res = _frank_wolfe_lp(f_cur, max_iters=50, time_limit=time_left)
        _update_best(f_res, s_res, f"{name}_FW")

    results.sort(key=lambda x: x[0])
    print(f"  Phase 3 best: {results[0][0]:.6f}")

    # === PHASE 4: Spectral refinement + perturbation restarts ===
    print("\n=== Phase 4: Spectral + Perturbation ===")
    p4_deadline = deadline - 80.0

    # Spectral refinement on current best
    if time.monotonic() < p4_deadline - 10:
        time_left = min(60.0, p4_deadline - time.monotonic() - 5)
        f_res, s_res = _spectral_refine(best_f, n_iter=2000, time_limit=time_left)
        _update_best(f_res, s_res, "spectral")

    # Perturbation restarts on best
    for pert_frac in [0.02, 0.05, 0.1, 0.2]:
        if time.monotonic() > p4_deadline:
            break
        noise = rng.standard_normal(len(best_f)) * pert_frac
        f_pert = _normalize(best_f + noise)
        if f_pert is None:
            continue
        t_left = p4_deadline - time.monotonic()
        if t_left < 5:
            break
        f_res, s_res = _mirror_descent(
            f_pert, n_iter=8000, eta0=0.8, decay_rate=0.4,
            thresh_frac=0.75, time_limit=min(t_left * 0.25, 40.0)
        )
        _update_best(f_res, s_res, f"pert{pert_frac}")

    results.sort(key=lambda x: x[0])
    print(f"  Phase 4 best: {results[0][0]:.6f}")

    # === PHASE 5: Final refinement with remaining time ===
    print("\n=== Phase 5: Final Refinement ===")
    remaining = deadline - time.monotonic()
    print(f"  Time remaining: {remaining:.1f}s")

    if remaining > 20:
        # More mirror descent with very small step (polishing)
        n_iter = int(min(30000, remaining * 40))
        f_res, s_res = _mirror_descent(
            best_f, n_iter=n_iter, eta0=0.3, decay_rate=0.6,
            thresh_frac=0.9, time_limit=remaining * 0.5
        )
        _update_best(f_res, s_res, "final_md")

    if time.monotonic() < deadline - 20:
        # LP polish if n is small enough
        n = len(best_f)
        if n <= 1000:
            time_left = min(60.0, deadline - time.monotonic() - 5)
            if time_left > 10:
                f_res, s_res = _frank_wolfe_lp(best_f, max_iters=30, time_limit=time_left)
                _update_best(f_res, s_res, "final_FW")

    # Final round of perturbations if time remains
    if time.monotonic() < deadline - 10:
        for _ in range(5):
            if time.monotonic() > deadline - 5:
                break
            noise = rng.standard_normal(len(best_f)) * 0.03
            f_pert = _normalize(best_f + noise)
            if f_pert is None:
                continue
            t_left = deadline - time.monotonic() - 3
            if t_left < 3:
                break
            f_res, s_res = _mirror_descent(
                f_pert, n_iter=3000, eta0=0.5, decay_rate=0.5,
                thresh_frac=0.8, time_limit=t_left * 0.8
            )
            _update_best(f_res, s_res, "final_pert")

    # === Final verification ===
    results.sort(key=lambda x: x[0])
    best_final = results[0][1] if results else best_f
    final_list = [float(x) for x in best_final]
    final_score = float(evaluate_sequence(final_list))
    print(f"\nFinal verified score: {final_score:.6f} (n={len(final_list)})")

    if not np.isfinite(final_score) or final_score >= 2.0:
        print("  Falling back to parent.")
        return [float(x) for x in parent]

    return final_list
