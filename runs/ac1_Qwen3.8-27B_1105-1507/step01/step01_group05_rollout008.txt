"""
Multi-method optimization for minimizing 2*n*max(conv(a,a))/sum(a)^2.
Uses Adam on smoothed max-autocorrelation, LP-based inner loop, and local search.
"""

import numpy as np
from scipy.optimize import linprog
import time

def compute_score_normalized(a):
    """Assume sum(a)=1, return 2n * max(conv(a,a))."""
    n = len(a)
    conv = np.convolve(a, a)
    return 2.0 * n * conv.max()

def smooth_max_grad(a, T):
    """Compute smooth max of conv(a,a) and its gradient w.r.t. a."""
    n = len(a)
    conv = np.convolve(a, a)
    conv_shifted = conv - conv.max()
    exp_vals = np.exp(T * conv_shifted)
    Z = exp_vals.sum()
    weights = exp_vals / Z
    smooth_val = conv.max() + np.log(Z) / T
    # grad[j] = sum_{m=0}^{n-1} a[m] * weights[j+m]
    grad = np.correlate(weights, a, mode='valid')
    return smooth_val, grad

def adam_optimize(a0, time_budget, lr=0.01, T_init=50, T_decay=0.999, rng=None):
    """Adam optimization on smooth max autocorrelation with simplex projection."""
    if rng is None:
        rng = np.random.default_rng(0)
    
    n = len(a0)
    a = a0.astype(np.float64).copy()
    S = a.sum()
    if S < 1e-10:
        return a, np.inf
    a = a / S
    
    m = np.zeros(n)  # first moment
    v = np.zeros(n)  # second moment
    beta1, beta2, eps = 0.8, 0.999, 1e-8
    t = 0
    T = T_init
    
    best_a = a.copy()
    best_score = compute_score_normalized(a)
    
    start = time.monotonic()
    no_improve = 0
    
    while time.monotonic() - start < time_budget:
        t += 1
        smooth_val, grad = smooth_max_grad(a, T)
        
        # Adam update
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * grad**2
        m_hat = m / (1 - beta1**t)
        v_hat = v / (1 - beta2**t)
        
        a_new = a - lr * m_hat / (np.sqrt(v_hat) + eps)
        a_new = np.clip(a_new, 0, None)
        S_new = a_new.sum()
        if S_new < 1e-10:
            a_new = a
            S_new = 1.0
        a_new = a_new / S_new
        
        score = compute_score_normalized(a_new)
        a = a_new
        
        if score < best_score - 1e-10:
            best_a = a.copy()
            best_score = score
            no_improve = 0
        else:
            no_improve += 1
        
        # Adaptive temperature
        if no_improve > 100:
            T *= T_decay
            if T < 500:
                T = 500
            
            # Try random restart from best
            if no_improve > 500:
                a = best_a * (1 + 0.1 * rng.normal(0, 1, n))
                a = np.clip(a, 0, None)
                S = a.sum()
                if S > 1e-10:
                    a = a / S
                m = np.zeros(n)
                v = np.zeros(n)
                t = 0
                no_improve = 0
                lr = lr * 0.5
        
        # Periodic printing
        if t % 10000 == 0:
            elapsed = time.monotonic() - start
            print(f"      t={t}, score={best_score:.6f}, T={T:.1f}, elapsed={elapsed:.1f}s")
    
    return best_a, best_score

import numpy as np

def compute_score_normalized(a):
    """Assume sum(a)=1, return 2n * max(conv(a,a))."""
    n = len(a)
    conv = np.convolve(a, a)
    return 2.0 * n * conv.max()


def build_conv_matrix(f):
    """Build matrix A such that A @ b = np.convolve(f, b, mode='full')."""
    n = len(f)
    A = np.zeros((2 * n - 1, n))
    for k in range(2 * n - 1):
        j_start = max(0, k - n + 1)
        j_end = min(n, k + 1)
        for j in range(j_start, j_end):
            A[k, j] = f[k - j]
    return A


def grad_smooth_max(a, T):
    """Compute gradient of log-sum-exp(T * conv(a,a)) w.r.t. a."""
    conv = np.convolve(a, a)
    conv_max = conv.max()
    exp_vals = np.exp(T * (conv - conv_max))
    Z = exp_vals.sum()
    weights = exp_vals / Z
    grad = 2.0 * np.correlate(weights, a, mode='valid')
    return grad


def softmax(z):
    """Numerically stable softmax."""
    z_shifted = z - z.max()
    exp_z = np.exp(z_shifted)
    return exp_z / exp_z.sum()


def grad_z_via_softmax(z, T):
    """Gradient of smooth_max(softmax(z)) w.r.t. z."""
    a = softmax(z)
    grad_a = grad_smooth_max(a, T)
    return a * (grad_a - np.dot(a, grad_a))

def project_simplex(a):
    """Project onto probability simplex (sum=1, a>=0)."""
    n = len(a)
    if a.sum() < 1e-15:
        return np.ones(n) / n
    a = np.clip(a, 0, None)

def lp_step(f, M):
    """Solve LP: max sum(b) s.t. conv(f,b) <= M, b >= 0. Returns b or None."""
    n = len(f)
    A = build_conv_matrix(f)
    c_obj = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    try:
        res = linprog(c_obj, A_ub=A, b_ub=b_ub, bounds=bounds,
                      method='highs', options={'presolve': True, 'time_limit': 30})
        if res.success and res.x is not None:
            b = np.clip(res.x, 0, None)
            if b.sum() > 1e-15:
                return b
    except Exception:
        pass
    return None

    # Phase 1: Quick Adam runs on all configs
    print("\n=== Phase 1: Adam optimization ===")
    time_per_adam = max(15.0, (budget_s * 0.4) / len(configs))
    
    for name, f0, method in configs:
        if time.monotonic() > deadline:
            break
        if method != 'adam':
            continue
        
        remaining = deadline - time.monotonic()
        t_alloc = min(time_per_adam, max(10.0, remaining / 3))
        
        z0 = np.log(f0 + 1e-15)
        f_opt, score = adam_optimize_z(z0, t_alloc, lr=1.0, T_init=50.0, rng=rng, verbose=False)
        
        if np.isfinite(score) and score < best_score_overall:
            best_score_overall = score
            best_overall = f_opt.copy()
            print(f"  [{name}] n={len(f0)}: score={score:.6f} *** NEW BEST ***")
        else:
            print(f"  [{name}] n={len(f0)}: score={score:.6f}")
    
    # Phase 2: LP optimization on best candidates and LP configs
    print(f"\n=== Phase 2: LP optimization ===")
    print(f"Current best: {best_score_overall:.6f}")
    
    time_per_lp = max(30.0, (budget_s * 0.4) / 6)
    
    # LP on parent
    if parent is not None:
        remaining = deadline - time.monotonic()
        if remaining > 60:
            t_alloc = min(time_per_lp, remaining * 0.3)
            print(f"  LP on parent (n={len(parent)}), time={t_alloc:.0f}s")
            f_opt, score = optimize_lp(parent, t_alloc, t0=0.5, rng=rng, verbose=True)
            if np.isfinite(score) and score < best_score_overall:
                best_score_overall = score
                best_overall = f_opt.copy()
                print(f"  LP parent: {score:.6f} *** NEW BEST ***")
    
    # LP on best adam result
    if best_overall is not None:
        remaining = deadline - time.monotonic()
        if remaining > 60:
            n_best = len(best_overall)
            t_alloc = min(time_per_lp, remaining * 0.25)
            print(f"  LP on best adam (n={n_best}), time={t_alloc:.0f}s")
            f_opt, score = optimize_lp(best_overall, t_alloc, t0=0.4, rng=rng, verbose=True)
            if np.isfinite(score) and score < best_score_overall:
                best_score_overall = score
                best_overall = f_opt.copy()
                print(f"  LP on best: {score:.6f} *** NEW BEST ***")
    
    # LP on sparse configs
    for name, f0, method in configs:
        if time.monotonic() > deadline:
            break
        if method != 'lp':
            continue
        remaining = deadline - time.monotonic()
        if remaining < 40:
            break
        t_alloc = min(time_per_lp, remaining * 0.3)
        print(f"  [{name}] n={len(f0)}, time={t_alloc:.0f}s")
        f_opt, score = optimize_lp(f0, t_alloc, t0=0.5, rng=rng, verbose=False)
        if np.isfinite(score) and score < best_score_overall:
            best_score_overall = score
            best_overall = f_opt.copy()
            print(f"  [{name}]: {score:.6f} *** NEW BEST ***")
        else:
            print(f"  [{name}]: {score:.6f}")
    
    # Phase 3: Alternating Adam + LP on best
    print(f"\n=== Phase 3: Alternating refinement ===")
    print(f"Current best: {best_score_overall:.6f}")
    
    if best_overall is not None:
        remaining = deadline - time.monotonic()
        n_iter_ph3 = min(10, max(2, int(remaining / 60)))
        
        f_cur = best_overall.copy()
        score_cur = best_score_overall
        
        for i in range(n_iter_ph3):
            if time.monotonic() > deadline:
                break
            remaining = deadline - time.monotonic()
            if remaining < 30:
                break
            
            # Adam step
            t_adam = min(30.0, remaining * 0.4)
            z0 = np.log(f_cur + 1e-15)
            f_adam, score_adam = adam_optimize_z(z0, t_adam, lr=0.8, T_init=80.0, rng=rng, verbose=False)
            if score_adam < score_cur:
                f_cur = f_adam
                score_cur = score_adam
            
            # LP step
            remaining = deadline - time.monotonic()
            if remaining < 20:
                break
            t_lp = min(30.0, remaining * 0.4)
            f_lp, score_lp = optimize_lp(f_cur, t_lp, t0=0.3, rng=rng, verbose=False)
            if score_lp < score_cur:
                f_cur = f_lp
                score_cur = score_lp
            
            if score_cur < best_score_overall:
                best_score_overall = score_cur
                best_overall = f_cur.copy()
                print(f"  Phase3 iter {i}: {score_cur:.6f} *** NEW BEST ***")
            else:
                print(f"  Phase3 iter {i}: {score_cur:.6f}")
    
    # Phase 4: Polish with local search
    print(f"\n=== Phase 4: Local search polish ===")
    if best_overall is not None:
        remaining = deadline - time.monotonic()
        if remaining > 10:
            n_iter_ls = min(50000, int(remaining * 1000))
            f_pol, score_pol = local_search_polish(best_overall, rng, n_iter=n_iter_ls)
            if score_pol < best_score_overall:
                best_score_overall = score_pol
                best_overall = f_pol.copy()
                print(f"  After polish: {score_pol:.6f} *** NEW BEST ***")
            else:
                print(f"  After polish: {score_pol:.6f}")
    
    # Phase 5: Final Adam with low temperature on best
    if best_overall is not None:
        remaining = deadline - time.monotonic()
        if remaining > 20:
            z0 = np.log(best_overall + 1e-15)
            f_final, score_final = adam_optimize_z(z0, remaining - 10, lr=0.5, T_init=200.0, rng=rng, verbose=True)
            if score_final < best_score_overall:
                best_score_overall = score_final
                best_overall = f_final.copy()
                print(f"  Final adam: {score_final:.6f} *** NEW BEST ***")
    
    print(f"\n=== Final Result ===")
    print(f"Best score: {best_score_overall:.6f}")
    print(f"Target: 1.5027")
    
    if best_overall is not None:
        best_overall = np.clip(best_overall, 0, 1000)
        final_list = best_overall.tolist()
        final_score = evaluate_sequence(final_list)
        print(f"Verified score: {final_score:.6f}")
        print(f"Sequence length: {len(final_list)}")
        return final_list
    
    # Fallback
    return [1.0] * 256
