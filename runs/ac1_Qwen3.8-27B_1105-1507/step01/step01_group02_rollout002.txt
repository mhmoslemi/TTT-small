"""
Optimizes a nonnegative sequence to minimize 2*n*max(convolve(a,a))/sum(a)^2.
Uses a combined approach of alternating projection initialization,
Nesterov-accelerated subgradient descent, and frequent LP-based improvement steps.
Tries multiple sequence lengths with warm-starting from previous best.
"""

import numpy as np
import time
from scipy.optimize import linprog


def project_simplex(v):
    """Project vector v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho_candidates = np.where(u * np.arange(1, n + 1) > cssv)[0]
    if len(rho_candidates) == 0:
        return np.ones(n) / n
    rho = rho_candidates[-1]
    theta = cssv[rho] / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def fast_convolve(a):
    """Compute full linear convolution of a with itself using FFT."""
    n = len(a)
    size = 2 * n - 1
    N = 1 << (size - 1).bit_length()
    fa = np.fft.rfft(a, N)
    fc = fa * fa
    result = np.fft.irfft(fc, N)[:size]
    result[result < 0] = 0.0
    return result


def compute_subgradient(a, peak_indices):
    """Compute subgradient of max_k (a*a)[k] at point a, averaged over active peaks."""
    n = len(a)
    grad = np.zeros(n)
    for k in peak_indices:
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        if lo <= hi:
            # (a*a)[k] = sum_i a[i] * a[k-i]
            # d/dx_j (a*a)[k] = a[k-j] + a[k-j] if j is in range... 
            # Actually d/dx_j sum_i a[i]*a[k-i] = a[k-j] (from the first sum where i=j) 
            # + a[k-j] (from the second sum where k-i=j, i.e., i=k-j)
            # But we need to be careful about the range.
            # For the linear convolution (a*a)[k] = sum_{i: 0<=i<n, 0<=k-i<n} a[i]*a[k-i]
            # d/dx_j = a[k-j] (if 0 <= k-j < n) + a[k-j] (if 0 <= k-j < n) 
            # Wait, that's double counting. Let me be more careful.
            # (a*a)[k] = sum_i a[i] * a[k-i]
            # d/dx_j (a*a)[k] = sum_i (da[i]/dx_j * a[k-i] + a[i] * da[k-i]/dx_j)
            # = a[k-j] + a[k-j] = 2*a[k-j] (if 0 <= k-j < n)
            # So the subgradient contribution from peak k is 2*a[k-j] for j in [k-n+1, k] ∩ [0, n-1]
            grad[lo:hi + 1] += 2.0 * a[k - hi:k - lo + 1][::-1]
    grad /= max(1, len(peak_indices))
    return grad


def alt_projection_init(n, rng, n_iter=300):
    """
    Alternating projection to get a nonnegative sequence with near-flat
    magnitude spectrum.
    """
    N = 1 << (2 * n - 2).bit_length()
    phases = rng.uniform(0.0, 2.0 * np.pi, N // 2 + 1)
    F = np.exp(1j * phases)
    F[0] = 1.0 + 0j

    a = np.ones(n) / n
    for it in range(n_iter):
        a_full = np.fft.irfft(F, N)
        a = a_full[:n].copy()
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s < 1e-10:
            a = np.ones(n) / n
        else:
            a = a / s
        F_full = np.fft.rfft(a, N)
        mag_target = np.ones(len(F_full))
        # Slight taper to help with nonnegativity
        weights = np.linspace(1.0, 0.8, len(F_full))
        mag_target = weights ** 0.05
        F = mag_target * np.exp(1j * np.angle(F_full))
        F[0] = F[0].real + 0j

    return a


def lp_step(a, P, time_limit=5.0):
    """
    LP step: maximize sum(b) subject to conv(a,b) <= P, b >= 0.
    Returns solution b or None.
    """
    n = len(a)
    # Build convolution matrix A of shape (2n-1, x n)
    A = np.zeros((2 * n - 1, n), dtype=np.float64)
    for k in range(2 * n - 1):
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        A[k, lo:hi + 1] = a[k - hi:k - lo + 1][::-1]

    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, P)

    try:
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=[(0.0, None)] * n, method='highs',
                      options={'time_limit': time_limit, 'presolve': True})
        if res.success:
            b = res.x
            b_sum = b.sum()
            if b_sum > 1e-8:
                return b
    except Exception:
        pass
    return None


def optimize_sequence(a, deadline, rng, use_lp=True, lp_every=15, max_iters=1000000):
    """
    Optimized subgradient descent with Nesterov momentum and periodic LP steps.
    Minimizes max(conv(a,a)) with sum(a)=1, a>=0.
    """
    n = len(a)
    a = project_simplex(a.copy().astype(np.float64))

    c = fast_convolve(a)
    best_peak = np.max(c)
    best_a = a.copy()

    # Nesterov momentum
    y = a.copy()
    beta = 0.9
    # Adaptive learning rate
    lr = 1.0 / (2.0 * np.sqrt(n))
    lr_min = 1e-6
    stall_count = 0
    max_stall = 200
    it = 0
    prev_c = c
    prev_peak = best_peak

    while time.time() < deadline and it < max_iters:
        it += 1

        # Update momentum
        t_k = (1.0 + np.sqrt(1.0 + 4.0 * beta * beta * it)) / 2.0
        beta_new = (1.0 - 1.0 / t_k) / (1.0 + beta)
        
        # Compute gradient at y (momentum point)
        if it > 1:
            y = project_simplex(a + beta_new * (a - y))
        
        c_y = fast_convolve(y)
        peak_y = np.max(c_y)
        
        # Active set
        threshold = peak_y * (1.0 - 1e-4)
        peak_indices = np.where(c_y >= threshold)[0]
        if len(peak_indices) > 20:
            top_idx = np.argsort(c_y[peak_indices])[::-1][:10]
            peak_indices = peak_indices[top_idx]
        
        grad = compute_subgradient(y, peak_indices)
        
        # Gradient step
        a_new = y - lr * grad
        a = project_simplex(a_new)
        
        # Check improvement
        c = fast_convolve(a)
        peak = np.max(c)
        
        if peak < best_peak * (1.0 - 1e-8):
            best_peak = peak
            best_a = a.copy()
            stall_count = 0
        else:
            stall_count += 1
        
        # Adaptive learning rate
        if stall_count % 50 == 0 and stall_count > 0:
            lr *= 0.85
            lr = max(lr, lr_min)
        
        # LP step
        if use_lp and it % lp_every == 0 and time.time() < deadline - 30:
            # Use current peak as bound (or slightly tighter)
            P_lp = peak * 1.0  # Could try peak * 0.99 for tighter bound
            b_sol = lp_step(a, P_lp, time_limit=3.0)
            if b_sol is not None:
                b_sum = b_sol.sum()
                if b_sum > 1e-8:
                    # Mix with current
                    t_mix = 0.5
                    a_mixed = (1.0 - t_mix) * a + t_mix * (b_sol / b_sum)
                    a = project_simplex(a_mixed)
                    # Check if this improved
                    c_lp = fast_convolve(a)
                    peak_lp = np.max(c_lp)
                    if peak_lp < peak:
                        # Keep the LP result
                        pass
                    else:
                        # Revert
                        a = project_simplex(a_new)  # Revert to pre-LP state
        
        # Perturbation to escape local minima
        if stall_count > 500 and time.time() < deadline - 60:
            perturb = rng.normal(0, 0.03 / np.sqrt(n), n)
            a = project_simplex(a + perturb)
            stall_count = 0
            lr = max(lr * 1.5, lr_min * 10)
        
        prev_c = c
        prev_peak = peak

    return best_a, best_peak
