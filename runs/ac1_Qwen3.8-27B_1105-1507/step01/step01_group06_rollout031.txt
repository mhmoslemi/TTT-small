    return np.fft.irfft(fa * fa, s)[:2 * n - 1]

    """Build Toeplitz-like constraint matrix A where row k corresponds to (a*g)_k."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))

        jh = min(n - 1, k)
        if jl <= jh:
            A[k, jl:jh + 1] = a[k - jh:k - jl + 1][::-1]
    return A

    if r.success:
        g = r.x
        if g.sum() > 1e-10:
            return g
    return None

"""
Minimizes J(a) = 2*n*max(convolve(a,a))/(sum(a))^2 using:
1. LP-based mirror descent with two LP variants and adaptive step sizes
2. Log-sum-exp smoothed gradient refinement via L-BFGS-B
3. Peak-shaving coordinate descent
4. Multiple diverse initializations and sequence lengths
Key improvement: dual LP variants (max-sum and min-peak), backtracking step
sizes, smoothed objectives for gradient polish, and a peak-shaving heuristic.
"""

import numpy as np
from scipy.optimize import linprog, minimize
import time


def fast_conv(a):
    """Full self-convolution via FFT, O(n log n)."""
    n = len(a)
    s = 1
    while s < 2 * n - 1:
        s <<= 1
    fa = np.fft.rfft(a, s)
    return np.fft.irfft(fa * fa, s)[:2 * n - 1]


def build_A(a):
    """Build Toeplitz-like constraint matrix A where row k gives (a*g)_k."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for k in range(2 * n - 1):
        jl = max(0, k - n + 1)
        jh = min(n - 1, k)
        if jl <= jh:
            A[k, jl:jh + 1] = a[k - jh:k - jl + 1][::-1]
    return A


def lp_max_sum(a, M, A=None):
    """LP variant 1: max sum(g) s.t. (a*g)_k <= M, g >= 0."""
    n = len(a)
    if A is None:
        A = build_A(a)
    c = -np.ones(n)
    r = linprog(c, A_ub=A, b_ub=np.full(2 * n - 1, M),
               bounds=[(0, None)] * n, method='highs')
    if r.success and r.x.sum() > 1e-10:
        return r.x
    return None


def lp_min_peak(a, A=None):
    """LP variant 2: min t s.t. (a*g)_k <= t, sum(g)=1, g >= 0.
    Returns (g, t) or (None, inf)."""
    n = len(a)
    if A is None:
        A = build_A(a)
    # Variables: [g_0, ..., g_{n-1}, t]  (n+1 variables)
    c = np.zeros(n + 1)
    c[n] = 1.0  # minimize t
    # Inequality: A*g - t <= 0  =>  [A, -1] * [g; t] <= 0
    A_ub = np.hstack([A, -np.ones((2 * n - 1, 1))])
    b_ub = np.zeros(2 * n - 1)
    # Equality: sum(g) = 1
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])
    bounds = [(0, None)] * n + [(0, None)]
    r = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
               bounds=bounds, method='highs')
    if r.success:
        g = r.x[:n]
        t = r.x[n]
        if g.sum() > 1e-10:
            return g, t
    return None, float('inf')


def smoothed_obj(x, T):
    """Smoothed max of convolution using log-sum-exp with temperature T."""
    ax = np.maximum(x, 0.0)
    s = ax.sum()
    if s < 1e-8:
        return 1e10
    ax = ax / s
    b = fast_conv(ax)
    bm = b.max()
    if bm <= 0:
        return 0.0
    return float(bm + np.log(np.sum(np.exp(T * (b - bm)))) / T)


def raw_max_obj(x):
    """Raw max of convolution (non-smooth)."""
    ax = np.maximum(x, 0.0)
    s = ax.sum()
    if s < 1e-8:
        return 1e10
    ax = ax / s
    return float(fast_conv(ax).max())


def lp_mirror_descent(a, max_iters, deadline, use_min_peak=False):
    """Run LP mirror descent iterations with adaptive step size."""
    n = len(a)
    a = a.copy()
    a = np.clip(a, 0.01, None)
    s = a.sum()
    if s > 0:
        a /= s

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        b = fast_conv(a)
        M = float(b.max())
        A = build_A(a)

        if use_min_peak:
            g, t_new = lp_min_peak(a, A)
            if g is None:
                break
            g = g / g.sum()
        else:
            g = lp_max_sum(a, M, A)
            if g is None:
                break
            g = g / g.sum()

        # Backtracking: try several step sizes, pick the best
        best_a = None
        best_obj = M
        for t_try in [0.3, 0.2, 0.15, 0.1, 0.05]:
            a_try = (1.0 - t_try) * a + t_try * g
            a_try = np.clip(a_try, 0.0, None)
            ss = a_try.sum()
            if ss < 1e-10:
                continue
            a_try /= ss
            obj_try = fast_conv(a_try).max()
            if obj_try < best_obj:
                best_obj = obj_try
                best_a = a_try

        if best_a is None:
            break
        improvement = (M - best_obj) / max(M, 1e-15)
        a = best_a
        if improvement < 1e-7 and it > 5:
            break

    return a


def peak_shave(a, max_iters, deadline):
    """Peak shaving: iteratively reduce the dominant convolution peak."""
    n = len(a)
    a = a.copy()
    a = np.clip(a, 0.0, None)
    s = a.sum()
    if s < 0.01:
        return a
    a /= s

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        b = fast_conv(a)
        M = b.max()
        peak_idx = int(np.argmax(b))

        # Find contributing pairs for this peak
        jl = max(0, peak_idx - n + 1)
        jh = min(n - 1, peak_idx)
        if jl > jh:
            break
        products = np.zeros(n)
        valid = np.arange(jl, jh + 1)
        products[valid] = a[valid] * a[peak_idx - valid][::-1]

        # Find the largest contributing pair
        max_j = valid[np.argmax(products[valid])]
        partner = peak_idx - max_j

        if partner == max_j or partner < 0 or partner >= n:
            # Can't do a simple swap, try reducing the peak element
            # Find a quiet position (minimum contribution to any peak)
            contributions = np.zeros(n)
            for k in range(2 * n - 1):
                if b[k] > 0.5 * M:
                    jl2 = max(0, k - n + 1)
                    jh2 = min(n - 1, k)
                    v2 = np.arange(jl2, jh2 + 1)
                    contributions[v2] += a[v2] * a[k - v2][::-1]
            dst = int(np.argmin(contributions))
            if dst == max_j:
                break
            delta = 0.005 * a[max_j]
            a_new = a.copy()
            a_new[max_j] -= delta
            a_new[dst] += delta
        else:
            # Swap: move mass from larger to smaller
            if a[max_j] > a[partner]:
                src, dst = max_j, partner
            else:
                src, dst = partner, max_j
            delta = 0.01 * a[src]
            a_new = a.copy()
            a_new[src] -= delta
            a_new[dst] += delta

        a_new = np.clip(a_new, 0.0, None)
        ss = a_new.sum()
        if ss < 0.01:
            break
        a_new /= ss
        new_M = fast_conv(a_new).max()
        if new_M < M * (1 - 1e-8):
            a = a_new

    return a

                starts.append(np.maximum(np.interp(t_new, t_old, best), 0.0))
        starts.append(rng.uniform(0.05, 1.0, n))

        max_T = 2048
        for T_smooth in [50.0, 200.0, 1000.0]:
            if time.monotonic() > deadline:
                break
            for a0 in starts:
                if time.monotonic() > deadline:
                    break
                a0 = np.clip(a0, 0.0, None)
                if a0.sum() < 0.01:
                    continue
                a0 = a0 / a0.sum()
                cur_T = T_smooth

                def make_obj(_T, _n):
                    def obj_fn(x):
                        return smoothed_obj(x, _T)
                    return obj_fn

                obj_fn = make_obj(cur_T, n)
                maxiter = 200 if n <= 512 else 100
                maxfun = 4000 if n <= 512 else 2000
                res = minimize(obj_fn, a0, method='L-BFGS-B',
                              bounds=[(0, None)] * n,
                              options={'maxiter': maxiter, 'max_fun': maxfun,
                                       'ftol': 1e-15, 'gtol': 1e-12, 'disp': False})
                if res.fun < 1e10:
                    record(res.x)
                    # Also try raw max from this starting point
                    a_start = np.clip(res.x, 0.0, None)
                    if a_start.sum() > 0.01:
                        a_start /= a_start.sum()
                        res2 = minimize(raw_max_obj, a_start, method='L-BFGS-B',
                                       bounds=[(0, None)] * n,
                                       options={'maxiter': 100, 'max_fun': 2000,
                                                'ftol': 1e-15, 'disp': False})
                        if res2.fun < 1e10:
                            record(res2.x)

        print(f"  Smoothed Grad n={n}: {time.monotonic()-t0:.1f}s, best={best_score:.6f}", flush=True)

    # === Phase 3: Peak Shaving ===
    if best is not None and time.monotonic() < deadline:
        print(f"\n=== Peak Shaving ===", flush=True)
        t0 = time.monotonic()
        a = best.copy()
        a_shaved = peak_shave(a, max_iters=2000, deadline=deadline)
        record(a_shaved)
        print(f"  Peak Shaving: {time.monotonic()-t0:.1f}s, best={best_score:.6f}", flush=True)

    # === Phase 4: Final Combined Polish ===
    if best is not None and time.monotonic() < deadline:
        n = best.size
        print(f"\n=== Final Polish: n={n} ===", flush=True)
        t0 = time.monotonic()
        a = best.copy()
        a = np.clip(a, 0.01, None)
        a /= a.sum()

        for it in range(50):
            if time.monotonic() > deadline:
                break
            # LP step
            b = fast_conv(a)
            M = b.max()
            A = build_A(a)
            g = lp_max_sum(a, M, A)
            if g is not None:
                g /= g.sum()
                a_new = 0.7 * a + 0.3 * g
                a_new = np.clip(a_new, 0.0, None)
                if a_new.sum() > 0.01:
                    a_new /= a_new.sum()
                    if fast_conv(a_new).max() <= M * 1.001:
                        a = a_new

            # Short gradient step
            cur_T = 500.0

            def obj_polish(x):
                return smoothed_obj(x, cur_T)

            res = minimize(obj_polish, a, method='L-BFGS-B',
                          bounds=[(0, None)] * n,
                          options={'maxiter': 50, 'max_fun': 1000,
                                   'ftol': 1e-15, 'disp': False})
            if res.fun < 1e10:
                a_new = np.clip(res.x, 0.0, None)
                if a_new.sum() > 0.01:
                    a_new /= a_new.sum()
                    if fast_conv(a_new).max() <= fast_conv(a).max() * 1.001:
                        a = a_new

            record(a)

        # Peak shave the polished result
        a_shaved = peak_shave(a, max_iters=500, deadline=deadline)
        record(a_shaved)
        print(f"  Final Polish: {time.monotonic()-t0:.1f}s, best={best_score:.6f}", flush=True)

    # === Phase 5: Try slightly perturbed versions of best ===
    if best is not None and time.monotonic() < deadline:
        print(f"\n=== Perturbation Search ===", flush=True)
        t0 = time.monotonic()
        n = best.size
        for trial in range(20):
            if time.monotonic() > deadline:
                break
            a_pert = best.copy() * (1.0 + 0.01 * rng.standard_normal(n))
            a_pert = np.clip(a_pert, 0.0, None)
            if a_pert.sum() < 0.01:
                continue
            a_pert /= a_pert.sum()
            # Quick gradient refinement
            cur_T = 200.0

            def obj_pert(x):
                return smoothed_obj(x, cur_T)

            res = minimize(obj_pert, a_pert, method='L-BFGS-B',
                          bounds=[(0, None)] * n,
                          options={'maxiter': 100, 'max_fun': 2000,
                                   'ftol': 1e-15, 'disp': False})
            if res.fun < 1e10:
                record(res.x)

        print(f"  Perturbation: {time.monotonic()-t0:.1f}s, best={best_score:.6f}", flush=True)

    if best is None:
        best = np.ones(256)

    print(f"\nFINAL: J={best_score:.6f}, n={best.size}", flush=True)
    return [float(x) for x in best]
