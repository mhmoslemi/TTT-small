```python
def softmax_gradient_step(a, beta, lr):
    n = len(a)
    conv = np.convolve(a, a)
    # Softmax weights
    shifted = beta * (conv - conv.max())
    weights = np.exp(shifted)
    weights /= weights.sum()
    
    # Gradient of soft-max w.r.t. a[i]
    grad = np.zeros(n)
    for k in range(len(conv)):
        w = weights[k]
        if w < 1e-15:
            continue
        # d(conv[k])/da[i] = a[k-i] for valid i
        s = max(0, k - (n-1))
        e = min(n-1, k)
        for i in range(s, e+1):
            j = k - i
            grad[i] += w * a[j]
    
    # Project gradient to maintain sum=1: subtract mean
    grad -= grad.mean()
    # Also ensure non-negativity: use multiplicative update
    # a_new = a * exp(-lr * grad)
    a_new = a * np.exp(-lr * grad)
    a_new = np.maximum(a_new, 0)
    a_new /= a_new.sum()
    return a_new
```

```python
    v = np.zeros(n)  # momentum
    best_a = a.copy()
    best_score = score(a)

            break

        M = conv.max()
        shifted = beta * (conv - M)
        exp_vals = np.exp(shifted)

        # Gradient
        grad = np.zeros(n)

        significant = np.where(weights > 1e-10)[0]

            best_score = new_score

        # Decay learning rate

    return best_a, best_score
```

```python
import time
import numpy as np

try:
    from scipy.optimize import linprog

    HAVE_SCIPY = False

def local_score(a):
    a = np.asarray(a, dtype=np.float64)
    n = a.size
    if n == 0:
        return float('inf')
    a = np.clip(a, 0.0, 1000.0)
    s = a.sum()
    if s < 1e-12:
        return float('inf')
    conv = np.convolve(a, a)
    return float(2.0 * n * conv.max() / (s * s))

def normalize(a):
    a = np.asarray(a, dtype=np.float64)
    a = np.clip(a, 0.0, None)
    s = a.sum()
    if s < 1e-300:
        return np.ones_like(a) / a.size
    return a / s

def softmax_gradient(a, beta):
    """Compute gradient of soft-max approximation of score using FFT."""
    n = a.size
    conv = np.convolve(a, a)
    M = conv.max()
    shifted = beta * (conv - M)
    np.maximum(shifted, -30, out=shifted)  # numerical stability
    exp_vals = np.exp(shifted)
    weights = exp_vals / exp_vals.sum()
    
    # Gradient: G[i] = 2 * sum_k weights[k] * a[k-i] (for valid k-i)
    # = 2 * convolve(weights, a[::-1])[n-1 : 2n-1]
    rev_a = a[::-1]
    conv_wa = np.convolve(weights, rev_a)
    G = 2.0 * conv_wa[n-1:2*n-1]
    
    # Project to sum=0 (for the constraint sum(a)=1)
    G -= G.mean()
    # Scale by 2n (from the score formula) - actually the 2n factor is constant
    # so it doesn't affect the direction, only the magnitude
    return G, conv, weights, M

def multiplicative_step(a, G, lr):
    """Apply multiplicative gradient step."""
    # a_new = a * exp(-lr * G)
    # But we need to be careful: G is the gradient of the soft-max, 
    # which is already scaled. The score is 2n * max_conv, so the gradient
    # of the score is 2n * G. But since we're doing multiplicative updates,
    # the absolute scale doesn't matter as much as the direction.
    update = -lr * G
    np.clip(update, -2.0, 2.0, out=update)  # prevent explosion
    a_new = a * np.exp(update)
    a_new = np.maximum(a_new, 1e-300)
    return normalize(a_new)

def optimize_softmax(a, time_budget, seed=42):
    """Main optimization using soft-max gradient descent."""
    t0 = time.monotonic()
    deadline = t0 + time_budget
    n = a.size
    
    best = a.copy()
    best_sc = local_score(best)
    
    eta = 0.5  # learning rate
    beta = 50.0  # soft-max sharpness
    momentum = 0.7
    v = np.zeros(n)
    
    it = 0
    stuck = 0
    
    while time.monotonic() < deadline:
        # Every 100 iterations, try to anneal
        if it % 200 == 199 and stuck > 50:
            # Perturb
            rng = np.random.default_rng(seed + it)
            noise = np.exp(rng.normal(0, 0.3, n))
            a = normalize(best * noise)
            v = np.zeros(n)
            eta = 0.5
            beta = 50.0
            stuck = 0
            it += 1
            continue
        
        G, conv, weights, M = softmax_gradient(a, beta)
        
        # Momentum
        v = momentum * v - eta * G
        a_new = multiplicative_step(a, v, 1.0 / eta if eta < 0.01 else 1.0)
        
        # Actually, let me simplify: just use the velocity directly
        update = -v  # negative gradient direction with momentum
        np.clip(update, -1.5, 1.5, out=update)
        a_new = a * np.exp(update)
        a_new = np.maximum(a_new, 1e-300)
        a_new = normalize(a_new)
        
        sc = local_score(a_new)
        
        if sc < best_sc:
            best = a_new.copy()
            best_sc = sc
            stuck = 0
        else:
            stuck += 1
            a = a_new  # still move forward (momentum helps)
        
        # Adaptive parameters
        if it % 50 == 0:
            beta *= 1.05  # sharpen
            beta = min(beta, 500.0)
            eta *= 0.995  # decay learning rate
            eta = max(eta, 0.01)
        
        it += 1
    
    return best, best_sc
```
