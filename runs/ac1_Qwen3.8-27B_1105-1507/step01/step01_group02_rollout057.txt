"""
Optimizes non-negative sequences to minimize 2*n*max(conv(a,a))/sum(a)^2.
Uses mirror descent (exponential family updates) with FFT convolution,
periodic LP-based refinement, and multiple template initializations.
"""

import numpy as np
import time
from scipy.optimize import linprog


def fast_convolve(a):
    """Compute full linear convolution a*a via FFT."""
    n = len(a)
    size = 2 * n - 1
    N = 1 << (size - 1).bit_length()
    fa = np.fft.rfft(a, N)
    result = np.fft.irfft(fa * fa, N)[:size]
    result[result < 0] = 0.0
    return result


def compute_subgradient(a, c, tol=1e-4, max_active=15):
    """Compute subgradient of max(c) at point a, where c = conv(a,a)."""
    n = len(a)
    peak = np.max(c)
    active = np.where(c >= peak * (1.0 - tol))[0]
    if len(active) > max_active:
        top = np.argsort(c[active])[::-1][:max_active]
        active = active[top]
    
    grad = np.zeros(n)
    for k in active:
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        grad[lo:hi+1] += 2.0 * a[k-hi:k-lo+1][::-1]
    grad /= len(active)
    return grad


def mirror_step(a, grad, eta):
    """Exponential family (mirror descent) step with normalization."""
    a_safe = np.maximum(a, 1e-30)
    log_a = np.log(a_safe) - eta * grad
    log_a -= np.max(log_a)
    a_new = np.exp(log_a)
    s = a_new.sum()
    if s < 1e-30:
        a_new = np.ones(len(a)) / len(a)
    else:
        a_new /= s
    return a_new


def lp_refine(a, P, time_limit=5.0):
    """Solve LP: max sum(b) s.t. conv(a,b) <= P, b >= 0. Returns b/sum(b) or None."""
    n = len(a)
    if n > 4000:
        return None
    
    # Build convolution matrix
    A = np.zeros((2*n-1, n))
    for k in range(2*n-1):
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        A[k, lo:hi+1] = a[k-hi:k-lo+1][::-1]
    
    c_obj = -np.ones(n)
    b_ub = np.full(2*n-1, P)
    
    try:
        res = linprog(c_obj, A_ub=A, b_ub=b_ub, bounds=[(0.0, None)]*n,
                      method='highs', options={'time_limit': time_limit, 'presolve': True})
        if res.success:
            b = res.x
            s = b.sum()
            if s > 1e-10:
                return b / s
    except Exception:
        pass
    return None


def optimize(a_init, deadline, rng, use_lp=True, lp_interval=100):
    """Main optimization loop using mirror descent + optional LP refinement."""
    n = len(a_init)
    a = np.maximum(a_init, 0.0).astype(np.float64)
    s = a.sum()
    a = a / s if s > 1e-10 else np.ones(n) / n
    
    best_a = a.copy()
    best_peak = np.inf
    best_score = np.inf
    
    # Initial evaluation
    c = fast_convolve(a)
    peak = np.max(c)
    best_peak = peak
    best_a = a.copy()
    best_score = 2.0 * n * peak  # since sum(a) = 1
    
    eta = 1.0  # initial step size
    it = 0
    no_improve = 0
    max_no_improve = 200
    
    while time.time() < deadline and it < 500000:
        it += 1
        c = fast_convolve(a)
        peak = np.max(c)
        score = 2.0 * n * peak
        
        if score < best_score * (1 - 1e-10):
            best_score = score
            best_a = a.copy()
            best_peak = peak
            no_improve = 0
            # Slowly increase eta slightly when improving
            eta = min(eta * 1.001, 10.0)
        else:
            no_improve += 1
        
        # Adaptive step size
        if no_improve > max_no_improve:
            eta = max(eta * 0.5, 0.01)
            no_improve = 0
        
        # Compute subgradient
        grad = compute_subgradient(a, c)
        
        # Mirror descent step
        a = mirror_step(a, grad, eta)
        
        # LP refinement
        if use_lp and it % lp_interval == 0 and time.time() < deadline - 20:
            P = best_peak * (1 + 1e-6)  # allow slight tolerance
            b = lp_refine(a, P, time_limit=3.0)
            if b is not None:
                # Check if b is an improvement
                c_b = fast_convolve(b)
                peak_b = np.max(c_b)
                if peak_b < best_peak:
                    t_mix = 0.5
                    a = (1 - t_mix) * a + t_mix * b
                    a = np.maximum(a, 0)
                    s = a.sum()
                    a = a / s
                else:
                    # Mix a little to explore
                    t_mix = 0.1 + 0.2 * rng.random()
                    a = (1 - t_mix) * a + t_mix * b
                    a = np.maximum(a, 0)
                    s = a.sum()
                    a = a / s
        
        # Random restart on long stall
        if no_improve > max_no_improve * 5 and time.time() < deadline - 60:
            # Perturb and continue
            noise = rng.normal(0, 0.1 / np.sqrt(n), n)
            a = np.maximum(a + noise, 0)
            a = a / a.sum()
            eta = 1.0
            no_improve = 0
    
    return best_a, best_score

"""
Optimizes non-negative sequences to minimize 2*n*max(conv(a,a))/sum(a)^2.
Uses mirror descent (exponential family updates) with FFT-based convolution,
periodic LP refinement, smooth gradient (log-sum-exp), and diverse template
initializations. Includes coordinate-descent polishing for final refinement.
"""

import numpy as np
import time
from scipy.optimize import linprog


def fast_convolve(a):
    """Compute full linear convolution a*a via FFT."""
    n = len(a)
    size = 2 * n - 1
    N = 1 << (size - 1).bit_length()
    fa = np.fft.rfft(a, N)
    result = np.fft.irfft(fa * fa, N)[:size]
    result[result < 0] = 0.0
    return result


def compute_smooth_gradient(a, c, beta=200.0, max_terms=30):
    """Compute gradient of log-sum-exp approximation to max(c)."""
    n = len(a)
    c_max = np.max(c)
    shifted = beta * (c - c_max)
    # Clamp for numerical stability
    shifted = np.clip(shifted, -500, 0)
    weights = np.exp(shifted)
    wsum = weights.sum()
    if wsum < 1e-300:
        # Fallback to uniform over top entries
        weights = np.zeros_like(c)
        top_k = min(len(c), max_terms)
        top_idx = np.argsort(c)[::-1][:top_k]
        weights[top_idx] = 1.0
        wsum = top_k
    weights /= wsum

    # Only process significant weights
    threshold = 1.0 / max(10 * n, wsum * 0.01)
    significant = weights > threshold
    if significant.sum() == 0:
        significant = np.zeros(len(c), dtype=bool)
        top_idx = np.argsort(c)[::-1][:max_terms]
        significant[top_idx] = True

    grad = np.zeros(n)
    for k in np.where(significant)[0]:
        w = weights[k]
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        grad[lo:hi + 1] += w * 2.0 * a[k - hi:k - lo + 1][::-1]
    return grad


def compute_subgradient(a, c, tol=1e-4, max_active=12):
    """Compute subgradient of max(c) at point a (uniform over active set)."""
    n = len(a)
    peak = np.max(c)
    active = np.where(c >= peak * (1.0 - tol))[0]
    if len(active) > max_active:
        top = np.argsort(c[active])[::-1][:max_active]
        active = active[top]

    grad = np.zeros(n)
    for k in active:
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        grad[lo:hi + 1] += 2.0 * a[k - hi:k - lo + 1][::-1]
    grad /= len(active)
    return grad


def mirror_step(a, grad, eta):
    """Exponential family (mirror descent) step preserving non-negativity and normalization."""
    a_safe = np.maximum(a, 1e-30)
    # Normalize gradient for stable steps
    g_norm = np.max(np.abs(grad))
    if g_norm < 1e-30:
        return a.copy()
    grad_normed = grad / g_norm
    log_a = np.log(a_safe) - eta * grad_normed
    log_a -= np.max(log_a)
    a_new = np.exp(log_a)
    s = a_new.sum()
    if s < 1e-30:
        a_new = np.ones(len(a)) / len(a)
    else:
        a_new /= s
    return a_new


def lp_refine(a, P, time_limit=4.0):
    """Solve LP: max sum(b) s.t. conv(a,b)_k <= P for all k, b >= 0. Returns normalized b or None."""
    n = len(a)
    if n > 5000:
        return None

    # Build convolution matrix efficiently
    A = np.zeros((2 * n - 1, n), dtype=np.float64)
    for k in range(2 * n - 1):
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        A[k, lo:hi + 1] = a[k - hi:k - lo + 1][::-1]

    c_obj = -np.ones(n)
    b_ub = np.full(2 * n - 1, P)

    try:
        res = linprog(c_obj, A_ub=A, b_ub=b_ub, bounds=[(0.0, None)] * n,
                      method='highs',
                      options={'time_limit': time_limit, 'presolve': True})
        if res.success and res.x is not None:
            b = np.maximum(res.x, 0)
            s = b.sum()
            if s > 1e-10:
                return b / s
    except Exception:
        pass
    return None


def coordinate_descent_polish(a, n_iter=200, deadline=None):
    """Coordinate descent: optimize each component a_i to minimize max(conv(a,a))."""
    n = len(a)
    a = np.maximum(a, 0.0).astype(np.float64).copy()
    a /= a.sum()

    for it in range(n_iter):
        if deadline is not None and time.time() > deadline:
            break

        c = fast_convolve(a)
        peak = np.max(c)

        for i in range(n):
            # Save old value
            old_val = a[i]
            if old_val < 1e-15:
                continue

            # Determine upper bound on a[i] from each constraint c_k <= peak
            # c_k = 2*a[i]*a[k-i] + (other terms) for k != 2i
            # c_{2i} = a[i]^2 + (other terms)
            # We want to find the max a[i] such that all c_k <= current_peak
            # But actually we want to MINIMIZE the peak, so we want to REDUCE a[i]
            # if it's contributing to the peak. Let's try reducing it.
            
            # Try a few candidate values
            candidates = [0.0, old_val * 0.1, old_val * 0.3, old_val * 0.5, old_val * 0.7, old_val * 0.9]
            
            best_val = old_val
            best_obj = np.inf
            
            for val in candidates:
                a[i] = val
                s = a.sum()
                if s < 1e-10:
                    a[i] = old_val
                    break
                a_normed = a / s
                c_test = fast_convolve(a_normed)
                obj = np.max(c_test)
                if obj < best_obj:
                    best_obj = obj
                    best_val = val
                a[i] = old_val
            
            a[i] = best_val
            a /= a.sum()

    return a


def alt_projection_init(n, rng, n_iter=50):
    """Alternating projection for flat-spectrum initialization."""
    N = 1 << (2 * n - 2).bit_length()
    phases = rng.uniform(0.0, 2.0 * np.pi, N // 2 + 1)
    F = np.exp(1j * phases)
    F[0] = 1.0 + 0j

    a = np.ones(n) / n
    for it in range(n_iter):
        a_full = np.fft.irfft(F, N)
        a = np.maximum(a_full[:n].copy(), 0.0)
        s = a.sum()
        if s < 1e-10:
            a = np.ones(n) / n
        else:
            a = a / s
        F_full = np.fft.rfft(a, N)
        mag = np.ones(len(F_full))
        F = mag * np.exp(1j * np.angle(F_full))
        F[0] = F[0].real + 0j

    return a

    ]

    # Time allocation: high priority gets more time
    total_time = total_deadline - time.time()
    # First 3 configs: 35% total time
    # Next 4 configs: 35% total time
    # Last 3 configs: 20% total time
    # Reserve 10% for final polishing

    time_first3 = total_time * 0.30
    time_mid4 = total_time * 0.35
    time_last3 = total_time * 0.15
    time_polish = total_time * 0.20

    config_times = []
    for i in range(len(configs)):
        if i < 3:
            config_times.append(time_first3 / 3.0)
        elif i < 7:
            config_times.append(time_mid4 / 4.0)
        else:
            config_times.append(time_last3 / 3.0)

    for i, (n, init_type, _) in enumerate(configs):
        if time.time() > total_deadline - 60:
            break

        config_deadline = min(total_deadline - time_polish, time.time() + config_times[i])
        if config_deadline < time.time() + 12:
            continue

        print(f"Config {i}: n={n}, init={init_type}, time_alloc={config_times[i]:.0f}s, "
              f"remaining={total_deadline - time.time():.0f}s")

        try:
            a = generate_init(n, init_type, rng)
            a_opt, score = optimize(a, config_deadline, rng, use_lp=True, lp_interval=60)
        except Exception as e:
            print(f"  Error: {e}")
            continue

        actual_score = evaluate_sequence(a_opt.tolist())
        print(f"  Result: internal={score:.6f}, actual={actual_score:.6f}")

        if np.isfinite(actual_score) and actual_score < best_score:
            best_score = actual_score
            best_seq = a_opt.copy()
            print(f"  >>> New best: {best_score:.6f} (n={n})")

    # Final aggressive polishing on best sequence
    if best_seq is not None and time.time() < total_deadline - 10:
        print(f"\nFinal polishing phase, time remaining: {total_deadline - time.time():.0f}s")
        try:
            polished, pol_score = optimize_aggressive(best_seq, total_deadline, rng)
            actual_pol = evaluate_sequence(polished.tolist())
            print(f"  Polished: internal={pol_score:.6f}, actual={actual_pol:.6f}")
            if np.isfinite(actual_pol) and actual_pol < best_score:
                best_score = actual_pol
                best_seq = polished.copy()
                print(f"  >>> New best after polish: {best_score:.6f}")
        except Exception as e:
            print(f"  Polish error: {e}")

    # Additional quick tries with different seeds if time remains
    if best_seq is not None and time.time() < total_deadline - 30:
        n_best = len(best_seq)
        for seed_offset in range(3):
            if time.time() > total_deadline - 15:
                break
            rng2 = np.random.default_rng(100 + seed_offset)
            a_try = generate_init(n_best, 'alt_proj', rng2)
            # Mix with best
            t = 0.3
            a_try = (1 - t) * (best_seq / best_seq.sum()) + t * (a_try / a_try.sum())
            a_try = np.maximum(a_try, 0)
            a_try /= a_try.sum()
            try:
                a_opt, sc = optimize(a_try, total_deadline - 15, rng, use_lp=True, lp_interval=50)
                actual_sc = evaluate_sequence(a_opt.tolist())
                if np.isfinite(actual_sc) and actual_sc < best_score:
                    best_score = actual_sc
                    best_seq = a_opt.copy()
                    print(f"  Extra try {seed_offset}: {actual_sc:.6f}")
            except Exception:
                pass

    # Fallback
    if best_seq is None:
        print("Falling back to uniform sequence")
        best_seq = np.ones(256) / 256

    result = [float(x) for x in best_seq]
    final_score = evaluate_sequence(result)
    print(f"\nFINAL: score={final_score:.6f}, length={len(result)}")
    return result
