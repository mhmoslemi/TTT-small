Optimized subgradient + LP hybrid for minimizing 2*n*max(conv(a,a))/(sum(a))^2.
Uses active-set subgradient descent with adaptive step sizes, periodic LP refinement,
and multi-scale random perturbations. Explores sequence lengths 100-400.

import numpy as np
from scipy.optimize import linprog
import time

def compute_obj(a):
    n = len(a)
    if n == 0:
        return np.inf
    conv = np.convolve(a, a)
    max_b = float(np.max(conv))
    sum_a = float(np.sum(a))
    if sum_a < 0.01:
        return np.inf
    return 2.0 * n * max_b / (sum_a * sum_a)

def normalize(a):
    s = np.sum(a)
    if s < 1e-15:
        return a
    return a / s

def subgradient_direction(a, top_k=3):
    """Compute subgradient direction for the max-convolution objective."""
    n = len(a)
    conv = np.convolve(a, a)
    
    # Get top-k active lags
    top_indices = np.argsort(conv)[::-1][:top_k]
    
    # Compute weighted subgradient
    grad = np.zeros(n)
    for k in top_indices:
        # d(conv_k)/d(a_i) = a_{i+k} (if i+k < n) + a_{i-k} (if i-k >= 0)
        # Vectorized:
        if k > 0:
            # a_{i+k} for i = 0..n-1-k
            grad[:n-k] += a[k:]
            # a_{i-k} for i = k..n-1
            grad[k:] += a[:n-k]
        else:
            # k=0: d(sum a_i^2)/d(a_i) = 2*a_i
            grad += 2.0 * a
    
    return grad, top_indices

def lp_step(a, upper_bound=None):
    """LP step: find g maximizing sum(g) s.t. conv(a,g) <= M, g >= 0."""
    n = len(a)
    conv_aa = np.convolve(a, a)
    M = float(np.max(conv_aa))
    
    # Build constraint matrix
    A = np.zeros((2 * n - 1, n), dtype=np.float64)
    for k in range(2 * n - 1):
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        # A[k, j] = a[k - j] for j in range
        idx = k - np.arange(j_start, j_end + 1)
        A[k, j_start:j_end+1] = a[idx]
    
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    if upper_bound is None:
        upper_bound = max(a) * 2
    bounds = [(0.0, upper_bound)] * n
    
    res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
    if res.status == 0:
        return res.x, float(-res.fun), M
    return None, None, M

    This directly finds the best g to pair with a.
    """
    n = len(a)
    # Build constraint matrix
    A = np.zeros((2 * n - 1, n), dtype=np.float64)
    for k in range(2 * n - 1):
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        idx = k - np.arange(j_start, j_end + 1)
        A[k, j_start:j_end+1] = a[idx]
    
    # Variables: g_0, ..., g_{n-1}, t
    # Minimize t
    c = np.zeros(n + 1)
    c[n] = 1.0
    
    # Constraints: conv_k - t <= 0 for all k
    # A_ub: [A | -I_{2n-1}]
    A_full = np.hstack([A, -np.eye(2 * n - 1)])
    b_ub = np.zeros(2 * n - 1)
    
    # sum(g) = 1 -> as equality constraint
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])
    
    bounds = [(0, None)] * n + [(0, None)]
    
    res = linprog(c, A_ub=A_full, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,

    if res.status == 0:

Another angle: consider the problem in terms of the Toeplitz matrix. The autocorrelation matrix T has entries T[i,j] = conv_{|i-j|}(a). We want to minimize the maximum diagonal entry of T (which is conv_0 = sum a_i^2) while keeping sum(a) = 1.

This is essentially a "minimax" problem on the autocorrelation sequence.

Actually, `logsumexp(beta * conv_k) / beta` approximates `max(conv_k)` as beta -> infinity. For finite beta, it's a smooth upper bound.

The gradient of this is:
`d/d(a_i) [logsumexp(beta * conv_k) / beta] = sum_k softmax_k * d(conv_k)/d(a_i) / beta`

where `softmax_k = exp(beta * conv_k) / sum_j exp(beta * conv_j)`.

Uses smooth max-convolution gradient descent (log-sum-exp), LP refinement,
and multi-scale random perturbations. Explores multiple sequence lengths.

import numpy as np
from scipy.optimize import linprog

def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    conv = np.convolve(a, a)
    max_b = float(np.max(conv))
    sum_a = float(np.sum(a))
    if sum_a < 0.01:
        return np.inf
    return 2.0 * n * max_b / (sum_a * sum_a)


def normalize(a):
    """Normalize to unit sum, ensuring non-negativity."""
    a = np.clip(a, 0, None)
    s = np.sum(a)
    if s < 1e-15:
        return a
    return a / s


def smooth_max_conv_grad(a, beta=50.0):
    """
    Compute smooth approximation of max(conv(a,a)) and its gradient w.r.t. a.
    Uses log-sum-exp for numerical stability.
    Returns (smooth_value, gradient).
    """
    n = len(a)
    conv = np.convolve(a, a)
    max_conv = float(np.max(conv))
    shifted = beta * (conv - max_conv)
    # Numerical safety
    shifted = np.clip(shifted, -500, 500)
    weights = np.exp(shifted)
    wsum = np.sum(weights)
    if wsum < 1e-300:
        weights = np.zeros(len(conv))
        weights[np.argmax(conv)] = 1.0
        wsum = 1.0
    weights /= wsum

    smooth_val = max_conv + np.log(wsum) / beta

    grad = np.zeros(n)
    for k in range(2 * n - 1):
        w = weights[k]
        if w < 1e-12:
            continue
        if k == 0:
            grad += 2.0 * w * a
        else:
            grad[:n - k] += w * a[k:]
            grad[k:] += w * a[:n - k]

    return smooth_val, grad


def lp_step_minimize_t(a):
    """
    LP: minimize t s.t. conv(a,g)_k <= t for all k, sum(g)=1, g>=0.
    Returns (g, t_optimal).
    """
    n = len(a)
    A = np.zeros((2 * n - 1, n), dtype=np.float64)
    for k in range(2 * n - 1):
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        idx = k - np.arange(j_start, j_end + 1)
        A[k, j_start:j_end + 1] = a[idx]

    c = np.zeros(n + 1)
    c[n] = 1.0

    A_full = np.hstack([A, -np.eye(2 * n - 1)])
    b_ub = np.zeros(2 * n - 1)

    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])

    bounds = [(0, None)] * (n + 1)

    res = linprog(c, A_ub=A_full, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                  bounds=bounds, method='highs')
    if res.status == 0:
        return res.x[:n].copy(), float(res.x[n])
    return None, None
