"""
Search for low-score non-negative step-function coefficients using smoothed
gradient descent with simplex projection, LP-based refinement, and multiple
lengths/initializations.

The objective is score(r) = 2*n*max(convolve(r,r)) / sum(r)^2.
We minimize this over non-negative sequences.

Key ideas:
  - Smooth the max with a softmax approximation to get a differentiable objective.
  - Use projected gradient descent onto the probability simplex.
  - Exploit convolution structure for O(n) gradient computation via cross-correlation.
  - Use LP-based steps for larger moves (inherited from previous SOTA).
  - Try multiple lengths and initializations to avoid local minima.

import time
import numpy as np

try:
    from scipy.optimize import linprog
    HAVE_SCIPY = True
except Exception:
    HAVE_SCIPY = False


def local_score(a):
    """Compute the score exactly as evaluate_sequence would."""
    a = np.asarray(a, dtype=np.float64)
    n = a.size
    if n == 0:
        return float('inf')
    a = np.clip(a, 0.0, 1000.0)
    s = float(np.sum(a))
    if s <= 1e-12:
        return float('inf')
    c = np.convolve(a, a)
    m = float(np.max(c))
    return float(2.0 * n * m / (s * s))


def softmax_max_and_weights(c, t):
    """Compute smooth max of c and softmax weights for gradient."""
    cm = np.max(c)
    shifted = t * (c - cm)
    # Clip for numerical stability
    shifted = np.clip(shifted, -500, 500)
    exps = np.exp(shifted)
    s = np.sum(exps)
    smax = cm + np.log(s) / t
    p = exps / s
    return smax, p


def simplex_project(v):
    """Project v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = v.size
    if n == 0:
        return v
    mu = np.sort(v)[::-1]
    cv = (np.cumsum(mu) - 1.0) / np.arange(1, n + 1)
    cond = mu > cv
    if not np.any(cond):
        return np.ones(n) / n
    rho = np.max(np.nonzero(cond))[0]
    tau = cv[rho]
    return np.maximum(v - tau, 0.0)


def sym_expand(h, n):
    """Expand symmetric half h to full length n."""
    m = (n + 1) // 2
    h = np.asarray(h, dtype=np.float64)[:m]
    if n == 1:
        return h.copy()
    if n % 2 == 0:
        return np.concatenate([h, h[::-1]])
    else:
        return np.concatenate([h, h[-2::-1]])


def sym_compress(a, n):
    """Compress full a to symmetric half."""
    return a[:(n + 1) // 2].copy()


def normalize_sum1(a):
    a = np.asarray(a, dtype=np.float64)
    a = np.clip(a, 0.0, 1000.0)
    s = float(np.sum(a))
    if s <= 1e-300:
        return None
    return a / s


def force_sym(a, n):

    a = 0.5 * (a + a[::-1])

def compute_gradient(a, p):
    """
    Compute gradient of smooth_max(conv(a,a), t) w.r.t. a.
    grad[j] = 2 * sum_i p[i+j] * a[i] = 2 * cross_correlation(p, a)[j]

    grad = 2.0 * np.correlate(p, a, mode='full')[:n]

def smoothed_gradient_optimize(a, t_start, t_end, eta_start, eta_end, n_iter, force_sym, seed):
    """
    Optimize using smoothed max + projected gradient descent.
    t (sharpness) and eta (step size) anneal over iterations.
    """
    n = a.size
    best = a.copy()
    best_sc = local_score(best)
    rng = np.random.default_rng(seed)

    for it in range(n_iter):
        frac = it / max(1, n_iter - 1)
        t = t_start * (t_end / t_start) ** frac
        eta = eta_start * (eta_end / eta_start) ** frac

        c = np.convolve(a, a)
        smax, p = softmax_max_and_weights(c, t)

        if force_sym:
            grad_full = compute_gradient(a, p)
            # Symmetrize
            grad_sym = 0.5 * (grad_full + grad_full[::-1])
            v = a - eta * grad_sym
            v = 0.5 * (v + v[::-1])
            a_new = simplex_project(v)
            a = 0.5 * (a_new + a_new[::-1])
            s = a.sum()
            if s > 0:
                a = a / s
        else:
            grad = compute_gradient(a, p)
            v = a - eta * grad
            a = simplex_project(v)

        sc = local_score(a)
        if sc < best_sc:
            best = a.copy()
            best_sc = sc

        # Occasionally add noise to escape flat regions
        if it % 200 == 199 and frac > 0.3:

            a_noisy = normalize_sum1(a_noisy)
            if a_noisy is not None:
                sc_n = local_score(a_noisy)
                if sc_n < best_sc * 1.05:  # Accept if not too much worse
                    a = a_noisy

def build_A_for_lp(f, n):
    """Build constraint matrix for LP: (f*g)[k] <= M for all k."""
    # (f*g)[k] = sum_j f[j] * g[k-j]
    # We want: for each k, sum_j A[k,j] * g[j] <= M
    # A[k,j] = f[k-j] if 0 <= k-j < n, else 0
    A = np.zeros((n, n), dtype=np.float64)  # Only need n constraints (symmetry)
    for k in range(n):
        for j in range(n):
            idx = k - j
            if 0 <= idx < n:
                A[k, j] = f[idx]
    return A


def lp_direction(f, n, seed):
    """
    Find a direction g via LP:
    Maximize sum(g) subject to (f*g)[k] <= M for k=0..n-1, 0 <= g <= 1.
    Returns normalized g or None.
    """
    if not HAVE_SCIPY:
        return None

    f = np.asarray(f, dtype=np.float64)
    c = np.convolve(f, f)
    M = float(np.max(c))
    if not np.isfinite(M) or M <= 0:
        return None

    A = np.zeros((n, n), dtype=np.float64)
    for k in range(n):
        s = max(0, k - n + 1)
        e = min(n, k + 1)
        j = np.arange(s, e, dtype=np.int64)
        idx = k - j
        A[k, j] = f[idx]

    cvec = -np.ones(n, dtype=np.float64)
    bvec = np.full(n, M * (1.0 + 1e-9), dtype=np.float64)
    bounds = [(0.0, 1.0)] * n

    try:
        res = linprog(cvec, A_ub=A, b_ub=bvec, bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': 30})
        if res is None or not res.success:
            return None
        g = np.asarray(res.x, dtype=np.float64)
        g = np.clip(g, 0.0, 1.0)
        s = g.sum()
        if s <= 0:
            return None
        return g / s
    except Exception:
        return None


def lp_refine(a, max_time, seed):
    """Iteratively apply LP-based direction finding and blending."""
    n = a.size
    best = a.copy()
    best_sc = local_score(best)
    t0 = time.monotonic()
    deadline = t0 + max_time
    stall = 0
    ts = (0.05, 0.1, 0.15, 0.2, 0.3, 0.5, 0.7)

    while time.monotonic() < deadline:
        rem = deadline - time.monotonic()
        if rem < 5.0:
            break

        g = lp_direction(best, n, seed + stall)
        if g is None:
            break

        new_best = best
        new_sc = best_sc
        for t in ts:
            cand = (1.0 - t) * best + t * g
            cand = normalize_sum1(cand)
            if cand is None:
                continue
            sc = local_score(cand)
            if sc < new_sc:
                new_sc = sc
                new_best = cand.copy()

        if new_sc < best_sc:
            best = new_best
            best_sc = new_sc
            stall = 0
            seed += 12345
        else:
            stall += 1
            if stall > 15:
                break

    return best, best_sc


def make_initialization(n, kind, seed, parent=None):
    """Create an initial sequence of length n."""
    rng = np.random.default_rng(seed + 7919 * kind)

    if parent is not None and kind == 10:
        # Interpolate parent to length n
        p = np.asarray(parent, dtype=np.float64)
        old_t = np.linspace(0, 1, p.size)
        new_t = np.linspace(0, 1, n)
        val = np.interp(new_t, old_t, p)
        val = np.maximum(val, 0.01)
        a = normalize_sum1(val)
        if a is not None:
            return a

    if kind == 0:
        # Uniform
        val = np.ones(n) / n
    elif kind == 1:
        # Raised cosine
        x = np.arange(n)
        val = np.cos(np.pi * (x / (n - 1) - 0.5)) ** 2
    elif kind == 2:
        # Bimodal (cosine bump)
        x = np.arange(n) / (n - 1)
        alpha = rng.uniform(0.3, 0.9)
        val = 1.0 + alpha * np.cos(2 * np.pi * x)
    elif kind == 3:
        # V-shaped
        x = np.abs(np.arange(n) - (n - 1) / 2.0) / (n / 2.0)
        beta = rng.uniform(0.5, 2.0)
        val = 1.0 + beta * x
    elif kind == 4:
        # Gaussian
        x = (np.arange(n) - (n - 1) / 2.0) / (n / 4.0)
        sigma = rng.uniform(0.5, 1.5)
        val = np.exp(-x**2 / (2 * sigma**2))
    elif kind == 5:
        # Random smoothed
        val = rng.uniform(0, 1, n)
        w = max(3, n // 20)
        kern = np.ones(w) / w
        val = np.convolve(val, kern, mode='same')
    elif kind == 6:
        # Inverse cosine (peaked at edges)
        x = np.arange(n) / max(1, n - 1)
        val = 1.0 / (0.1 + np.cos(np.pi * x))
    elif kind == 7:
        # Power law
        x = np.abs(np.arange(n) - (n - 1) / 2.0) / max(1, (n - 1) / 2.0)
        p = rng.uniform(0.5, 2.0)
        val = 1.0 + x ** p
    elif kind == 8:
        # Two bumps
        c1 = rng.uniform(0.1, 0.4)
        c2 = 1.0 - c1
        s = rng.uniform(0.02, 0.1)
        x = np.arange(n) / max(1, n - 1)
        val = np.exp(-((x - c1) / s) ** 2) + np.exp(-((x - c2) / s) ** 2)
    else:
        # Sinc-like
        x = np.arange(n)
        xc = (n - 1) / 2.0
        val = np.abs(np.sinc((x - xc) / (n * rng.uniform(0.1, 0.3)))) + 0.1

    val = np.maximum(val, 0.0)
    a = normalize_sum1(val)
    if a is None:
        a = np.ones(n) / n
    return a


def optimize_length(n, budget, seed, force_sym, parent):
    """Optimize for a given length, returning best sequence and score."""
    t0 = time.monotonic()
    deadline = t0 + budget
    best = None
    best_sc = float('inf')

    # Try multiple initializations
    n_init = min(6, max(2, int(budget / 20)))
    for i in range(n_init):
        if time.monotonic() >= deadline - 10:
            break
        kind = i if i < 9 else 10
        a = make_initialization(n, kind, seed + i * 331, parent)
        if force_sym:
            a = 0.5 * (a + a[::-1])
            a = normalize_sum1(a)

        # Short gradient optimization
        rem = deadline - time.monotonic()
        t_b = min(30.0, max(5.0, 0.3 * rem))
        n_iter = int(2000 * t_b / 30.0)
        a2, sc2 = smoothed_gradient_optimize(a, 10, 80, 0.8, 0.1, max(100, n_iter), force_sym, seed + i * 557)
        if sc2 < best_sc:
            best = a2.copy()
            best_sc = sc2
            print(f"  init {i} (kind={kind}): score={sc2:.6f}", flush=True)

    if best is None:
        best = np.ones(n) / n
        best_sc = local_score(best)

    # Longer gradient optimization with annealing
    rem = deadline - time.monotonic()
    if rem > 10:
        t_b = min(rem * 0.4, 60.0)
        n_iter = int(5000 * t_b / 60.0)
        a2, sc2 = smoothed_gradient_optimize(best, 20, 200, 0.6, 0.05, max(200, n_iter), force_sym, seed + 4242)
        if sc2 < best_sc:
            best = a2.copy()
            best_sc = sc2
            print(f"  grad refine: score={sc2:.6f}", flush=True)

    # LP refinement
    rem = deadline - time.monotonic()
    if rem > 10 and HAVE_SCIPY:
        t_b = min(rem * 0.3, 40.0)
        a2, sc2 = lp_refine(best, t_b, seed + 55555)
        if sc2 < best_sc:
            best = a2.copy()
            best_sc = sc2
            print(f"  LP refine: score={sc2:.6f}", flush=True)

    # Final gradient polish
    rem = deadline - time.monotonic()
    if rem > 5:
        t_b = min(rem * 0.5, 30.0)
        n_iter = int(10000 * t_b / 30.0)
        a2, sc2 = smoothed_gradient_optimize(best, 50, 500, 0.3, 0.01, max(100, n_iter), force_sym, seed + 66666)
        if sc2 < best_sc:
            best = a2.copy()
            best_sc = sc2
            print(f"  final polish: score={sc2:.6f}", flush=True)

    return best, best_sc


def multiplicative_perturb(a, eta, K, force_sym, seed):
    """
    Multiplicative update based on active convolution peaks.
    Reduce elements that contribute to the maximum convolution values.
    """
    n = a.size
    c = np.convolve(a, a)
    M = float(np.max(c))
    if M <= 0:
        return a.copy()

    # Find top K active lags
    if K >= c.size:
        active = np.nonzero(c > 0.5 * M)[0]
    else:
        active = np.argpartition(c, -K)[-K:]

    # Compute contribution of each element to active lags
    harm = np.zeros(n, dtype=np.float64)
    for k in active:
        s = max(0, k - n + 1)
        e = min(n, k + 1)
        for j in range(s, e):
            harm[j] += c[k] * a[k - j] / M
            other = k - j
            if other != j:
                harm[other] += c[k] * a[j] / M

    max_h = np.max(harm)
    if max_h <= 0:
        return a.copy()

    # Multiplicative update: reduce heavily-harmed elements
    fac = np.exp(-eta * harm / max_h)
    a2 = a * fac
    if force_sym:
        a2 = 0.5 * (a2 + a2[::-1])
    a2 = normalize_sum1(a2)
    return a2


def multiplicative_optimize(a, max_time, seed, force_sym):
    """Iterative multiplicative perturbation optimization."""
    n = a.size
    best = a.copy()
    best_sc = local_score(best)
    t0 = time.monotonic()
    deadline = t0 + max_time
    rng = np.random.default_rng(seed)
    eta = 0.5
    K = max(10, n // 10)
    stuck = 0

    while time.monotonic() < deadline:
        a_new = multiplicative_perturb(best if stuck > 50 else a, eta, K, force_sym, seed)
        sc = local_score(a_new)
        if sc < best_sc:
            best = a_new.copy()
            best_sc = sc
            a = a_new.copy()
            stuck = 0
            eta = min(1.0, eta * 1.05)
        else:
            stuck += 1
            if stuck > 100:
                # Restart with noise
                noise = np.exp(rng.normal(0, 0.15, n))
                a = best * noise
                if force_sym:
                    a = 0.5 * (a + a[::-1])
                a = normalize_sum1(a)
                if a is None:
                    a = best.copy()
                eta = 0.5
                stuck = 0

    return best, best_sc


def propose_candidate(seed=42, budget_s=1500, **kwargs):
    """
    Main search function. Returns the best sequence found within budget_s seconds.
    """
    start_time = time.monotonic()
    try:
        budget = float(budget_s)
    except Exception:
        budget = 1500.0
    if not np.isfinite(budget) or budget <= 0:
        budget = 1500.0
    deadline = start_time + budget - 15.0  # Leave 15s buffer

    # Load parent
    parent = None
    try:
        parent = globals().get('height_sequence_1', None)
    except Exception:
        parent = None

    parent_arr = None
    if parent is not None:
        try:
            parent_arr = np.asarray(parent, dtype=np.float64)
            parent_arr = np.clip(parent_arr, 0.0, 1000.0)
            if parent_arr.size == 0 or not np.isfinite(parent_arr).all():
                parent_arr = None
        except Exception:
            parent_arr = None

    best_seq = None
    best_score = float('inf')

    if parent_arr is not None:
        p = normalize_sum1(parent_arr)
        if p is not None:
            best_seq = p.copy()
            best_score = local_score(p)
            print(f"Parent: length={p.size}, score={best_score:.6f}", flush=True)

    # Length plan: try several lengths
    # Score = 2n * max_conv / sum^2, so different n trade off differently
    lengths = [256, 384, 512, 640, 768, 1024]
    # Adjust based on budget
    if budget < 600:
        lengths = [256, 512, 768]
    elif budget < 1000:
        lengths = [384, 512, 768, 1024]

    seed_cur = int(seed)
    for n in lengths:
        rem = deadline - time.monotonic()
        if rem < 30:
            break

        # Allocate time proportionally, with more time for lengths near parent length
        alloc = min(rem * 0.4, 200.0)
        if parent_arr is not None and abs(n - parent_arr.size) < 64:
            alloc = min(rem * 0.5, 300.0)

        # Without symmetry
        try:
            cand, sc = optimize_length(n, alloc, seed_cur, False, parent_arr)
            if cand is not None and sc < best_score:
                best_seq = cand.copy()
                best_score = sc
                print(f"Length {n} (asym): score={sc:.6f}", flush=True)
            seed_cur += 1009
        except Exception as e:
            print(f"Length {n} asym failed: {e}", flush=True)

        # With symmetry (use remaining time for this length)
        rem = deadline - time.monotonic()
        if rem > 30:
            alloc_sym = min(rem * 0.3, 150.0)
            try:
                cand, sc = optimize_length(n, alloc_sym, seed_cur, True, parent_arr)
                if cand is not None and sc < best_score:
                    best_seq = cand.copy()
                    best_score = sc
                    print(f"Length {n} (sym): score={sc:.6f}", flush=True)
                seed_cur += 1009
            except Exception as e:
                print(f"Length {n} sym failed: {e}", flush=True)

    # Final refinement on the best sequence found
    if best_seq is not None:
        n_best = best_seq.size

        # Multiplicative optimization
        rem = deadline - time.monotonic()
        if rem > 15:
            t_b = min(rem * 0.3, 60.0)
            try:
                c2, s2 = multiplicative_optimize(best_seq, t_b, seed_cur + 7777, False)
                if s2 < best_score:
                    best_seq = c2.copy()
                    best_score = s2
                    print(f"Mult optimize: score={s2:.6f}", flush=True)
            except Exception as e:
                print(f"Mult opt failed: {e}", flush=True)

        # More gradient steps with high sharpness
        rem = deadline - time.monotonic()
        if rem > 15:
            t_b = min(rem * 0.4, 80.0)
            n_iter = int(20000 * t_b / 80.0)
            try:
                c2, s2 = smoothed_gradient_optimize(best_seq, 100, 1000, 0.2, 0.005,
                                                    max(500, n_iter), False, seed_cur + 8888)
                if s2 < best_score:
                    best_seq = c2.copy()
                    best_score = s2
                    print(f"High-t gradient: score={s2:.6f}", flush=True)
            except Exception as e:
                print(f"High-t grad failed: {e}", flush=True)

        # LP polish
        rem = deadline - time.monotonic()
        if rem > 10 and HAVE_SCIPY:
            t_b = min(rem * 0.5, 60.0)
            try:
                c2, s2 = lp_refine(best_seq, t_b, seed_cur + 9999)
                if s2 < best_score:
                    best_seq = c2.copy()
                    best_score = s2
                    print(f"Final LP: score={s2:.6f}", flush=True)
            except Exception as e:
                print(f"Final LP failed: {e}", flush=True)

        # Very fine gradient steps
        rem = deadline - time.monotonic()
        if rem > 10:
            n_iter = int(50000 * rem / 10.0)
            n_iter = min(n_iter, 200000)
            try:
                c2, s2 = smoothed_gradient_optimize(best_seq, 500, 2000, 0.1, 0.001,
                                                    max(200, n_iter), False, seed_cur + 11111)
                if s2 < best_score:
                    best_seq = c2.copy()
                    best_score = s2
                    print(f"Ultra-fine gradient: score={s2:.6f}", flush=True)
            except Exception as e:
                print(f"Ultra-fine grad failed: {e}", flush=True)

    # Fallback
    if best_seq is None:
        if parent_arr is not None:
            best_seq = normalize_sum1(parent_arr)
            if best_seq is None:
                best_seq = np.ones(128) / 128
        else:
            best_seq = np.ones(128) / 128
        best_score = local_score(best_seq)

    # Final validation
    final = normalize_sum1(best_seq)
    if final is None:
        final = np.ones(128) / 128

    try:
        official = float(evaluate_sequence([float(x) for x in final]))
        if np.isfinite(official):
            best_score = official
        else:
            if parent_arr is not None:
                fp = normalize_sum1(parent_arr)
                if fp is not None:
                    off2 = float(evaluate_sequence([float(x) for x in fp]))
                    if np.isfinite(off2):
                        final = fp
                        best_score = off2
    except Exception:
        pass

    elapsed = time.monotonic() - start_time
    print(f"\nFINAL: score={best_score:.6f}, length={len(final)}, time={elapsed:.1f}s", flush=True)
    return [float(x) for x in np.clip(np.asarray(final, dtype=np.float64), 0.0, 1000.0)]
