import numpy as np

from scipy.optimize import linprog

def project_simplex(v):
    """Project vector v onto {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v
    # Sort in decreasing order
    u = np.sort(v)[::-1]
    cv = np.cumsum(u) - 1.0
    rho_candidates = np.where(u * np.arange(1, n + 1) > cv)[0]
    if len(rho_candidates) == 0:
        return np.ones(n) / n
    rho = rho_candidates[-1]
    theta = cv[rho] / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def fast_convolve(a):
    """Compute full linear convolution of a with itself using FFT."""
    n = len(a)
    size = 2 * n - 1
    N = 1 << (size - 1).bit_length()
    fa = np.fft.rfft(a, N)
    fc = fa * fa
    result = np.fft.irfft(fc, N)[:size]
    result[result < 0] = 0.0
    return result

def compute_subgradient_mean(a, peak_indices):
    """Compute mean subgradient of max_k c[k] at a, over active peaks."""
    n = len(a)
    grad = np.zeros(n)
    for k in peak_indices:
        for i in range(n):
            idx = k - i
            if 0 <= idx < n:
                grad[i] += 2.0 * a[idx]
    grad /= max(1, len(peak_indices))
    return grad

"""
Optimizes non-negative sequence to minimize 2*n*max(convolve(a,a))/sum(a)^2.
Uses FISTA (accelerated projected subgradient), smooth log-sum-exp approximation
with L-BFGS-B, LP-based improvements, and coordinate descent polish.
"""

import numpy as np
import time
from scipy.optimize import linprog, minimize

# Top-level helper functions

def project_simplex(v):
    """Project onto {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cv = np.cumsum(u) - 1.0
    rho_candidates = np.where(u * np.arange(1, n + 1) > cv)[0]
    if len(rho_candidates) == 0:
        return np.ones(n) / n
    rho = rho_candidates[-1]
    theta = cv[rho] / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def fast_convolve(a):
    """Full linear convolution a*a via FFT."""
    n = len(a)
    size = 2 * n - 1
    N = 1 << (size - 1).bit_length()
    fa = np.fft.rfft(a, N)
    result = np.fft.irfft(fa * fa, N)[:size]
    result[result < 0] = 0
    return result

def active_peaks(b, n_max=20):
    """Get indices of top peaks in b."""
    threshold = b.max() * (1.0 - 1e-5)
    idx = np.where(b >= threshold)[0]
    if len(idx) > n_max:
        idx = idx[np.argsort(b[idx])[::-1][:n_max]]
    return idx

def subgrad_from_peaks(a, peaks):
    """Subgradient of max_k b_k at a, averaged over given peaks."""
    n = len(a)
    grad = np.zeros(n)
    for k in peaks:
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        if lo <= hi:
            # grad[lo:hi+1] += 2 * a[k-hi : k-lo+1][::-1]
            seg = a[k - hi : k - lo + 1][::-1]
            grad[lo:hi + 1] += 2.0 * seg
    grad /= max(1, len(peaks))
    return grad

def fista_optimize(a0, deadline, rng, L_start=5.0, max_iters=1000000):
    """FISTA with adaptive step size for minimizing max(convolve(a,a)) on simplex."""
    n = len(a0)
    a = project_simplex(a0.copy())
    y = a.copy()
    t = 1.0
    L = L_start
    best_a = a.copy()
    best_b = fast_convolve(a).max()
    
    for it in range(max_iters):
        if time.time() > deadline:
            break
        
        b = fast_convolve(y)
        peak = b.max()
        peaks = active_peaks(b)
        grad = subgrad_from_peaks(y, peaks)
        
        # FISTA step
        a_new = project_simplex(y - grad / L)
        
        # Check convergence / adapt L
        b_new = fast_convolve(a_new)
        val_new = b_new.max()
        
        # Simple adaptive: if value went up significantly, increase L
        if val_new > peak * 1.001:
            L *= 1.3
            a_new = project_simplex(y - grad / L)
            b_new = fast_convolve(a_new)
            val_new = b_new.max()
        elif it > 100 and val_new < peak * 0.999:
            L = max(2.0, L * 0.97)
        
        # Momentum update
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t * t)) / 2.0
        y = a_new + ((t - 1.0) / t_new) * (a_new - a)
        t = t_new
        a = a_new
        
        if val_new < best_b:
            best_b = val_new
            best_a = a_new.copy()
    
    return best_a

def smooth_logsumexp_objective(x, mu=5.0):
    """Smooth approximation of max(convolve(softmax(x), softmax(x)))."""
    n = len(x)
    x_shift = x - x.max()
    a = np.exp(x_shift)
    a /= a.sum()
    b = fast_convolve(a)
    mu_b = mu * b
    mu_b_max = mu_b.max()
    w = np.exp(mu_b - mu_b_max)
    w /= w.sum()
    obj = (mu_b_max + np.log(w.sum())) / mu
    return obj, a, b, w

def smooth_gradient(x, mu=5.0):
    """Gradient of smooth objective w.r.t. x."""
    n = len(x)
    x_shift = x - x.max()
    a = np.exp(x_shift)
    a /= a.sum()
    b = fast_convolve(a)
    mu_b = mu * b
    mu_b_max = mu_b.max()
    w = np.exp(mu_b - mu_b_max)
    w /= w.sum()
    
    # grad w.r.t. a: weighted average of individual gradients
    grad_a = np.zeros(n)
    for k, wk in enumerate(w):
        if wk < 1e-12:
            continue
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        if lo <= hi:
            seg = a[k - hi : k - lo + 1][::-1]
            grad_a[lo:hi + 1] += 2.0 * wk * seg
    
    # Chain rule: d(obj)/dx_j = a_j * (grad_a_j - sum_i grad_a_i * a_i)
    ga_a = np.dot(grad_a, a)
    grad_x = a * (grad_a - ga_a)
    return grad_x

def lbfgs_smooth_optimize(a0, deadline, mu=5.0, max_iters=5000):
    """Optimize using L-BFGS-B on softmax parameterization."""
    n = len(a0)
    # Initialize x from a0
    eps = 1e-10
    a0_safe = a0 + eps
    x0 = np.log(a0_safe / a0_safe.sum() + 1e-15)
    
    def obj_and_grad(x):
        obj, a, b, w = smooth_logsumexp_objective(x, mu)
        g = smooth_gradient(x, mu)
        return obj, g
    
    def callback(xk):
        return time.time() > deadline
    
    try:
        result = minimize(obj_and_grad, x0, jac=True, method='L-BFGS-B',
                          callback=callback,
                          options={'maxiter': max_iters, 'gtol': 1e-8,
                                   'maxls': 50, 'factr': 1e4})
        x_opt = result.x
        _, a_opt, _, _ = smooth_logsumexp_objective(x_opt, mu)
        return a_opt
    except Exception as e:
        print(f"  L-BFGS-B error: {e}")
        return a0

def lp_improve(a, P, deadline):
    """LP step: find b that satisfies conv(a,b) <= P and maximize sum(b)."""
    n = len(a)
    if n > 1024:
        return None
    
    # Build A matrix: A[k, j] = a[k-j] if 0 <= k-j < n
    A = np.zeros((2 * n - 1, n))
    for k in range(2 * n - 1):
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        A[k, lo:hi + 1] = a[k - hi : k - lo + 1][::-1]
    
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, P)
    
    try:
        res = linprog(c, A_ub=A, b_ub=b_ub,
                      bounds=[(0.0, None)] * n, method='highs',
                      options={'time_limit': 5.0})
        if res.success:
            b = res.x
            if b.sum() > 1e-8:
                b *= a.sum() / b.sum()
                return b
    except Exception:
        pass
    return None

def coordinate_descent_polish(a, deadline, n_rounds=100):
    """Coordinate descent: try moving mass between pairs."""
    n = len(a)
    a = project_simplex(a.copy())
    best_score = 2 * n * fast_convolve(a).max() / (a.sum() ** 2)
    b = fast_convolve(a)
    best_b_max = b.max()
    
    for round_idx in range(n_rounds):
        if time.time() > deadline:
            break
        
        improved = False
        # Try random pairs
        for _ in range(2 * n):
            if time.time() > deadline:
                break
            i = np.random.randint(n)
            j = np.random.randint(n)
            if i == j:
                continue
            if a[i] < 1e-10:
                continue
            
            # Try moving a small amount from i to j
            for frac in [0.01, 0.05, 0.1, 0.2, 0.5]:
                delta = frac * a[i]
                a[i] -= delta
                a[j] += delta
                b_new = fast_convolve(a)
                if b_new.max() < best_b_max * (1.0 - 1e-8):
                    best_b_max = b_new.max()
                    improved = True
                    break
                else:
                    a[i] += delta
                    a[j] -= delta
        
        if not improved:
            break
    
    return project_simplex(a)
