import time

HAVE_SCIPY = True
try:

except:
    HAVE_SCIPY = False


def local_score(a):
    a = np.asarray(a, dtype=np.float64)
    a = np.clip(a, 0, 1000)
    if a.size == 0:
        return float('inf')
    s = float(a.sum())
    if s < 1e-12:
        return float('inf')
    c = np.convolve(a, a)
    m = float(c.max())
    return float(2.0 * len(a) * m / (s * s))


def compute_grad(a, p):
    """
    Compute gradient of sum_k p[k] * c[k] where c = convolve(a,a).
    grad[j] = 2 * sum_k p[k] * a[k-j] (for 0 <= k-j < n)
    """

    for j in range(n):
        # k ranges from j to min(j+n-1, 2n-2)
        k_end = min(j + n, 2*n - 1)
        if j < k_end:
            grad[j] = 2.0 * np.dot(p[j:k_end], a[:k_end-j])
    return grad

    """
    Optimize a using log-sum-exp smoothed gradient descent.
    a is assumed to have sum 1.
    """
    n = len(a)
    a = a.copy()
    a = np.maximum(a, 1e-15)  # ensure positive for now
    a /= a.sum()

    best_score = local_score(best_a)

    T = T_min
    lr = 0.05
    patience = 0
    
    while time.monotonic() - t0 < max_time:
        c = np.convolve(a, a)
        M = float(c.max())
        if M < 1e-15:
            break
        
        # Softmax weights
        Tc = T * c / M
        Tc -= Tc.max()
        p = np.exp(Tc)
        p_sum = p.sum()
        if p_sum < 1e-15 or not np.isfinite(p_sum):
            T = T_max
            continue
        p /= p_sum
        
        # Gradient
        grad = compute_grad(a, p)
        
        # Normalize gradient for scale
        grad_norm = np.linalg.norm(grad)
        if grad_norm < 1e-15:
            break
        
        # Update
        a_new = a - lr * grad
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s < 1e-15:
            lr *= 0.5
            continue
        a_new /= s
        
        # Evaluate
        sc = local_score(a_new)

            a = a_new

            # Slightly increase lr if making good progress

```python
def fast_grad(a, w_full):
    """
    grad[j] = 2 * sum_k w_full[k] * a[k-j] (for 0 <= k-j < n)
    where w_full has length 2n-1.
    """
    n = len(a)
    # Use FFT cross-correlation
    N = 2 * n  # sufficient for no wrap-around
    W = np.fft.rfft(w_full, N)
    A = np.fft.rfft(a, N)
    # cross-correlation: sum_k w[k] * conj(a[k-j])
    # For real a: sum_k w[k] * a[k-j]
    corr = np.fft.irfft(W * np.conj(A), N)
    # corr[j] = sum_k w[k] * a[k-j] for j = 0, ..., N-1
    # We want j = 0, ..., n-1
    return 2.0 * corr[:n]
```
