def compute_obj(a):

"""
Multi-strategy optimizer combining coordinate descent, sedimentary (core+edge) 
initialization, LP iteration, and random perturbation for minimizing
2*n*max(conv(a,a))/(sum(a))^2. Uses FFT for large sequences and manages
time budget across exploration and exploitation phases.
"""

import numpy as np
from scipy.optimize import linprog
import time


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    conv = np.convolve(a, a)
    max_b = float(np.max(conv))
    sum_a = float(np.sum(a))
    if sum_a < 0.01:
        return np.inf
    return 2.0 * n * max_b / (sum_a * sum_a)


def fast_convolve(a, b):
    """FFT-based convolution for potentially faster large-n computations."""
    n = len(a) + len(b) - 1
    size = 1
    while size < n:
        size <<= 1
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    result = np.fft.irfft(fa * fb, size)[:n]
    return result


def cd_coordinate(a, conv, n, m, search_range=None):
    """
    Optimize single coordinate a[m] given all others fixed.
    Uses candidate generation + golden section refinement.
    Returns (new_a, new_conv, new_obj).
    """
    old_am = a[m]
    total_sum = float(np.sum(a))
    S = total_sum - old_am  # sum excluding a[m]

    # D_valid[k-m] for k in [m, m+n-1]
    # conv[k] = D[k-m] + 2*old_am*a[k-m] (for k-m != m) or D[k-m] + old_am^2 (for k-m == m)
    D_valid = conv[m:m+n].copy() - 2.0 * old_am * a
    D_valid[m] = conv[2 * m] - old_am ** 2

    # Max of D for k outside [m, m+n-1]
    D_invalid_max = 0.0
    if m > 0:
        D_invalid_max = max(D_invalid_max, float(np.max(conv[:m])))
    if m + n < 2 * n - 1:
        D_invalid_max = max(D_invalid_max, float(np.max(conv[m + n:])))

    current_obj = 2.0 * n * float(np.max(conv)) / (total_sum ** 2)

    # Determine search range
    if search_range is None:
        upper = max(5.0 * float(np.mean(a)), 2.0 * float(np.max(a)), 1.0)
    else:
        upper = search_range

    # Generate candidates
    candidates = [0.0, old_am, upper * 0.1, upper * 0.3, upper * 0.5]

    # From active constraints
    current_max = float(np.max(conv))
    active_threshold = current_max * (1 - 1e-8)
    active_ks = np.where(conv >= active_threshold)[0]

    for k in active_ks[:8]:
        if k < m or k > m + n - 1:
            continue
        km = k - m
        i = km  # index into D_valid
        D_val = D_valid[i]
        if km == m:
            if S > 1e-10:
                cand = D_val / S
                if 0 < cand < upper:
                    candidates.append(cand)
        else:
            s_k = a[km]
            if s_k > 1e-15:
                cand = S - D_val / s_k
                if 0 < cand < upper:
                    candidates.append(cand)
                # Also try fractional versions
                if cand > 0:
                    candidates.append(cand * 0.5)
                    candidates.append(cand * 1.5)

    def eval_am(cand):
        """Evaluate objective if a[m] = cand."""
        cand = max(0.0, min(cand, upper))
        sum_new = S + cand
        if sum_new < 0.01:
            return np.inf
        new_conv_valid = D_valid + 2.0 * cand * a
        new_conv_valid[m] = D_valid[m] + cand ** 2
        new_max = max(float(np.max(new_conv_valid)), D_invalid_max)
        return 2.0 * n * new_max / (sum_new * sum_new)

    # Evaluate candidates
    best_cand = old_am
    best_obj_cand = current_obj
    for cand in candidates:
        cand = max(0.0, min(cand, upper))
        if abs(cand - old_am) < 1e-12:
            continue
        obj_cand = eval_am(cand)
        if obj_cand < best_obj_cand - 1e-12:
            best_obj_cand = obj_cand
            best_cand = cand

    # Golden section refinement around best candidate
    if best_cand > 0:
        center = best_cand
        width = max(center * 0.5, 0.1 * upper)
        lo = max(0.0, center - width)
        hi = center + width
        if hi > upper:
            hi = upper
            lo = max(0.0, hi - 2 * width)
        # Golden section
        gr = (np.sqrt(5.0) - 1.0) / 2.0
        c1 = hi - gr * (hi - lo)
        c2 = lo + gr * (hi - lo)
        f1 = eval_am(c1)
        f2 = eval_am(c2)
        for _ in range(20):
            if f1 < f2:
                hi = c2
                c2 = c1
                f2 = f1
                c1 = hi - gr * (hi - lo)
                f1 = eval_am(c1)
            else:
                lo = c1
                c1 = c2
                f1 = f2
                c2 = lo + gr * (hi - lo)
                f2 = eval_am(c2)
        best_in_search = (lo + hi) / 2.0
        obj_search = eval_am(best_in_search)
        if obj_search < best_obj_cand - 1e-12:
            best_cand = best_in_search
            best_obj_cand = obj_search

    # Also check near 0
    if best_cand > upper * 0.05:
        obj_zero = eval_am(0.0)
        if obj_zero < best_obj_cand - 1e-12:
            best_cand = 0.0
            best_obj_cand = obj_zero

    # Apply update
    if abs(best_cand - old_am) > 1e-12:
        new_conv_valid = D_valid + 2.0 * best_cand * a
        new_conv_valid[m] = D_valid[m] + best_cand ** 2
        conv[m:m + n] = new_conv_valid
        a[m] = best_cand

    return a, conv, best_obj_cand


def coordinate_descent(a, max_sweeps=2000, verbose=False, start_time=None, deadline=None):
    """Full coordinate descent optimization."""
    n = len(a)
    a = a.copy().astype(np.float64)
    conv = np.convolve(a, a)
    best_obj = compute_obj(a)

    for sweep in range(max_sweeps):
        if start_time is not None and time.monotonic() > deadline:
            break
        for m in range(n):
            a, conv, obj = cd_coordinate(a, conv, n, m)
            if obj < best_obj:
                best_obj = obj
        if verbose and sweep % 100 == 0:
            actual = compute_obj(a)
            print(f"      CD sweep {sweep}: obj={actual:.6f}")
        # Early termination check
        if sweep > 0 and sweep % 200 == 0:
            actual = compute_obj(a)
            if actual - best_obj < 1e-10:
                break

    return a, compute_obj(a)


def edge_optimize(a, edge_len, max_iter=200):
    """Optimize only the edge elements (first and last edge_len elements)."""
    n = len(a)
    a = a.copy().astype(np.float64)
    conv = np.convolve(a, a)
    best_obj = compute_obj(a)

    indices = list(range(edge_len)) + list(range(n - edge_len, n))

    for it in range(max_iter):
        for m in indices:
            a, conv, obj = cd_coordinate(a, conv, n, m)
            if obj < best_obj:
                best_obj = obj

    return a, compute_obj(a)


def sedimentary_init(n, rng, core_type=None):
    """
    Generate a sedimentary initialization: core pattern + edge envelope.
    """
    if core_type is None:
        core_type = int(rng.integers(0, 5))

    edge_len = max(3, n // 15)
    core_len = n - 2 * edge_len

    if core_len <= 0:
        edge_len = max(1, n // 20)
        core_len = n - 2 * edge_len

    # Generate core
    if core_type == 0:
        # Random
        core = rng.random(core_len) + 0.1
    elif core_type == 1:
        # Sinusoidal modulation
        freq = rng.uniform(1, 4)
        phase = rng.uniform(0, 2 * np.pi)
        core = 0.5 + 0.5 * np.cos(2 * np.pi * freq * np.arange(core_len) / n + phase)
        core = np.maximum(core, 0.05)
    elif core_type == 2:
        # Two-level pattern
        period = int(rng.integers(max(2, core_len // 20), max(3, core_len // 5)))
        level = rng.random(period) + 0.2
        core = np.tile(level, core_len // period + 1)[:core_len]
    elif core_type == 3:
        # Smooth random (low-pass filtered)
        raw = rng.random(core_len)
        kernel = np.ones(5) / 5
        core = np.convolve(raw, kernel, mode='same') + 0.1
    else:
        # Exponential decay from center
        center = core_len / 2
        core = np.exp(-0.5 * ((np.arange(core_len) - center) / (core_len / 4)) ** 2)

    # Apply edge envelopes
    a = np.zeros(n)
    a[edge_len:edge_len + core_len] = core

    # Edge transition
    for i in range(edge_len):
        t = (i + 1) / (edge_len + 1)
        # Raised cosine edge
        w = 0.5 * (1 - np.cos(np.pi * t))
        a[i] = w * core[min(i, core_len - 1)] if i < core_len else w * core[min(edge_len - 1 - i, core_len - 1)]
        a[n - 1 - i] = w * core[min(i, core_len - 1)] if i < core_len else w * core[min(edge_len - 1 - i, core_len - 1)]

    # Ensure non-negative
    a = np.maximum(a, 0.0)
    s = np.sum(a)
    if s < 1e-10:
        a = rng.random(n) + 0.1
        s = np.sum(a)
    return a / s * n  # Normalize so mean = 1


def lp_step(a, M_override=None):
    """
    LP step: find g maximizing sum(g) subject to conv(a,g) <= M, g >= 0.
    Returns (g, g_sum, M) or (None, None, M) on failure.
    """
    n = len(a)
    conv_aa = np.convolve(a, a)
    M = float(np.max(conv_aa)) if M_override is None else M_override

    # Build constraint matrix: conv(a,g)_k <= M for k=0,...,2n-2
    # conv(a,g)_k = sum_j a_j * g_{k-j} (valid j)
    # For each k, the constraint is: sum_j a_{k-j} * g_j <= M
    A = np.zeros((2 * n - 1, n))
    for k in range(2 * n - 1):
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        for j in range(j_start, j_end + 1):
            A[k, j] = a[k - j]

    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n

    try:
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'max_iter': 5000})
        if res.status == 0:
            return res.x, float(-res.fun), M
    except Exception:
        pass
    return None, None, M


def lp_optimize(a, max_iter=100, t=0.5, verbose=False):
    """LP-based iterative optimization."""
    n = len(a)
    a = a.copy().astype(np.float64)
    s = np.sum(a)
    if s < 1e-10:
        return a, np.inf
    best_obj = compute_obj(a)

    for it in range(max_iter):
        g, g_sum, M = lp_step(a)
        if g is None:
            break
        if g_sum <= 1.0 + 1e-6:
            break

        # Scale g to have same sum as a
        g_scaled = g * (s / g_sum)
        a_new = (1 - t) * a + t * g_scaled
        a_new = np.maximum(a_new, 0)
        s_new = np.sum(a_new)
        if s_new < 1e-10:
            break
        obj_new = compute_obj(a_new)
        if obj_new < best_obj:
            a = a_new
            s = s_new
            best_obj = obj_new
            if verbose and it % 20 == 0:
                print(f"      LP it={it}: obj={best_obj:.6f}")
        else:
            # Try smaller step
            a_new2 = (1 - t * 0.3) * a + t * 0.3 * g_scaled
            a_new2 = np.maximum(a_new2, 0)
            s2 = np.sum(a_new2)
            if s2 > 1e-10:
                obj2 = compute_obj(a_new2)
                if obj2 < best_obj:
                    a = a_new2
                    s = s2
                    best_obj = obj2

    return a, best_obj


def random_block_refine(a, rng, max_iter=2000, block_frac=0.05):
    """Random block perturbation local search."""
    n = len(a)
    a = a.copy().astype(np.float64)
    best_obj = compute_obj(a)
    best_a = a.copy()

    for it in range(max_iter):
        width = max(1, int(n * block_frac))
        start = int(rng.integers(0, n - width + 1))
        candidate = best_a.copy()
        noise = np.exp(rng.normal(0, 0.2, width))
        candidate[start:start + width] *= noise
        candidate = np.maximum(candidate, 0)
        s = np.sum(candidate)
        if s < 0.01:
            continue
        obj = compute_obj(candidate)
        if obj < best_obj:
            best_obj = obj
            best_a = candidate

    return best_a, best_obj


def perturb_sequence(a, rng, strength=0.3):
    """Randomly perturb a sequence to escape local optima."""
    n = len(a)
    a_new = a.copy()
    # Add multiplicative noise
    noise = np.exp(rng.normal(0, strength, n))
    a_new *= noise
    # Randomly zero out a small fraction
    frac = rng.random()
    if frac < 0.3:
        zero_mask = rng.random(n) < 0.05
        a_new[zero_mask] = 0
    a_new = np.maximum(a_new, 0)
    s = np.sum(a_new)
    if s < 0.01:
        a_new = rng.random(n)
        s = np.sum(a_new)
    a_new = a_new / s * float(np.sum(a))
    return a_new

            print(f"  After CD: obj={obj:.6f}")

        # Random block refinement
        if time.monotonic() < deadline - 5:
            rls_iters = int((deadline - 5 - time.monotonic()) / 0.02)
            a_r, obj_r = random_block_refine(best_seq, rng, max_iter=rls_iters, block_frac=0.03)
            if obj_r < best_score:
                best_score = obj_r
                best_seq = a_r.copy()
                print(f"  After RLS: obj={obj_r:.6f}")

        # One more CD pass
        if time.monotonic() < deadline - 3:
            a_f, obj_f = coordinate_descent(best_seq, max_sweeps=int((deadline - 3 - time.monotonic()) / 1.0))
            if obj_f < best_score:
                best_score = obj_f
                best_seq = a_f.copy()
                print(f"  Final CD: obj={obj_f:.6f}")

    # Fallback
    if best_seq is None:
        best_seq = np.ones(100)
        best_score = compute_obj(best_seq)

    elapsed = time.monotonic() - start_time
    print(f"\nDONE: n={len(best_seq)}, score={best_score:.6f}, time={elapsed:.1f}s")

    # Verify with evaluate_sequence
    try:
        eval_score = evaluate_sequence([float(x) for x in best_seq])
        if eval_score != best_score:
            print(f"  NOTE: evaluate_sequence gives {eval_score:.6f} vs our {best_score:.6f}")
            if eval_score < np.inf:
                best_score = eval_score
    except Exception:
        pass

    return [float(x) for x in best_seq]
