Actually, for a normalized distribution (sum=1) on n points, the minimum possible max autocorrelation is at least 1/n (by averaging: sum_k (f*f)_k = (sum f)^2 = 1, and there are 2n-1 values, so the max is at least 1/(2n-1) ~ 1/(2n)). The score would then be at least 2n/(2n) = 1. But achieving this bound is likely impossible for non-negative sequences.

The uniform distribution gives max = 1/n (the center of the triangle), score = 2.
A "U-shaped" distribution might give a flatter convolution.

I think n=1024 is a good choice because it gives enough "degrees of freedom" to shape the distribution while keeping computation tractable.

"""
Combined LP + Gradient search for minimizing 2*n*max(conv(f,f))/(sum(f))^2.

Phase 1: LP-based iteration (like parent) with improved step size selection.
Phase 2: Projected gradient descent with softmax smoothing of the max.
Phase 3: Random restarts and local search.
"""

import time
import numpy as np
from scipy.optimize import linprog
from scipy.sparse import csr_matrix


def normalize_sequence(arr):
    """Return a nonnegative float64 vector normalized to sum 1, or None."""
    try:
        x = np.asarray(arr, dtype=np.float64).ravel()
    except Exception:
        return None
    if x.size == 0:
        return None
    x = np.where(np.isfinite(x), x, 0.0)
    x = np.maximum(x, 0.0)
    s = float(x.sum())
    if s <= 1e-300:
        return None
    x /= s
    return x


def score_normalized(x, n):
    """Score assuming x is nonnegative and sum(x) = 1."""
    conv = np.convolve(x, x)
    m = float(np.max(conv))
    if not np.isfinite(m):
        return float("inf")
    return float(2.0 * n * m)


def project_to_simplex(v):
    """Project vector v onto the non-negative simplex {x >= 0, sum(x) = 1}."""
    v = np.asarray(v, dtype=np.float64).copy()
    n = v.size
    if n == 0:
        return v
    # Standard algorithm
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - 1))[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    w = np.maximum(v - theta, 0)
    s = w.sum()
    if s > 0:
        w /= s
    else:
        w = np.ones(n) / n
    return w


def softmax_weights(conv, alpha):
    """Compute softmax weights over the convolution values."""
    c = np.asarray(conv, dtype=np.float64)
    c_max = c.max()
    shifted = alpha * (c - c_max)
    # Numerical stability
    exp_vals = np.exp(shifted)
    s = exp_vals.sum()
    if s <= 0 or not np.isfinite(s):
        # Fallback: concentrate on the max
        p = np.zeros_like(c)
        p[np.argmax(c)] = 1.0
        return p
    return exp_vals / s


def compute_gradient(f, alpha):
    """Compute gradient of the smoothed max-convolution objective."""
    n = f.size
    c = np.convolve(f, f)  # length 2n-1
    p = softmax_weights(c, alpha)
    # grad_j = 2 * sum_k p_k * f_{k-j} = 2 * (p * f_reversed)[j + n - 1]
    f_rev = f[::-1].copy()
    cross = np.convolve(p, f_rev)  # length (2n-1) + n - 1 = 3n-1
    grad = 2.0 * cross[n-1 : n-1+n]  # extract the right slice, length n
    return grad


def gradient_descent_phase(f, time_budget, rng, alpha_start=10.0, alpha_end=500.0, lr_start=0.01):
    """
    Run projected gradient descent with adaptive alpha.
    Returns the best f found.
    """
    n = f.size
    f = normalize_sequence(f)
    if f is None:
        return None
    
    best_f = f.copy()
    best_score = score_normalized(f, n)
    
    deadline = time.monotonic() + max(0.0, time_budget)
    alpha = alpha_start
    lr = lr_start
    num_iter = 0
    stall = 0
    
    while time.monotonic() < deadline:
        t0 = time.monotonic()
        num_iter += 1
        
        grad = compute_gradient(f, alpha)
        
        if not np.all(np.isfinite(grad)):
            break
        
        f_new = f - lr * grad
        f_new = project_to_simplex(f_new)
        
        sc = score_normalized(f_new, n)
        
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_f = f_new.copy()
            f = f_new
            stall = 0
        else:
            # Reduce learning rate
            lr *= 0.9
            stall += 1
            if stall > 50:

                alpha = min(alpha * 1.5, alpha_end)
                lr = lr_start * 0.1
                stall = 0
                if alpha >= alpha_end:

                    if score_normalized(f, n) > best_score * 1.01:

        if num_iter % 100 == 0:
            print(f"    GD: iter={num_iter}, alpha={alpha:.1f}, lr={lr:.2e}, score={best_score:.6f}", flush=True)
    
    return best_f


def make_conv_template(n):
    """Precompute CSR matrix structure for the convolution constraint."""
    m = 2 * n - 1
    k64 = np.arange(m, dtype=np.int64)
    js = np.maximum(0, k64 - n + 1).astype(np.int64)
    je = np.minimum(n - 1, k64).astype(np.int64)
    counts = (je - js + 1).astype(np.int64)
    total = int(counts.sum())
    
    cum = np.cumsum(counts)
    starts = np.repeat(cum - counts, counts)
    local = np.arange(total, dtype=np.int64) - starts
    
    row = np.repeat(k64, counts).astype(np.int32)
    col_base = js[row.astype(np.int64)]
    col = (col_base + local).astype(np.int32)
    
    A_off = csr_matrix((np.ones(total, dtype=np.float64), (row, col)), shape=(m, n), dtype=np.float64)
    edge_offsets = A_off.data.astype(np.int32, copy=True)
    # Actually we need to store the column indices to look up f values
    col_data = A_off.indices.astype(np.int32, copy=True)
    
    A_template_data = np.zeros(total, dtype=np.float64)
    c_obj = -np.ones(n, dtype=np.float64)
    bounds = [(0.0, None)] * n
    return m, total, row, col_data, c_obj, bounds


def solve_lp_direction(f, n, M, m, total, row, col_data, c_obj, bounds):
    """Solve the LP: max sum(g) s.t. (f*g)_k <= M, g >= 0."""
    # Build the constraint matrix data
    A_data = f[col_data] * 1.0  # Each entry A_{k,j} = f_{k-j}
    A = csr_matrix((A_data, (row, col_data)), shape=(m, n), dtype=np.float64)
    b = np.full(m, M, dtype=np.float64)
    
    res = linprog(c_obj, A_ub=A, b_ub=b, bounds=bounds, method='highs')
    return res


def lp_optimize(n, f_init, time_budget, rng):
    """
    Run LP-based optimization from initial vector f_init.
    Returns best normalized vector and its score.
    """
    f = normalize_sequence(f_init)
    if f is None:
        f = np.ones(n, dtype=np.float64) / n
    
    best_f = f.copy()
    best_score = score_normalized(f, n)
    
    # Build template
    m, total, row, col_data, c_obj, bounds = make_conv_template(n)
    
    deadline = time.monotonic() + max(0.0, time_budget)
    stall = 0
    iteration = 0
    prev_f = f.copy()
    
    while time.monotonic() < deadline:
        iteration += 1
        t0 = time.monotonic()
        
        conv_ff = np.convolve(f, f)
        M = float(np.max(conv_ff))
        if not np.isfinite(M) or M <= 0:
            break
        
        # Try multiple constraint levels
        found_improvement = False
        
        for M_frac in [1.0, 0.95, 0.9]:
            M_try = M * M_frac
            
            try:
                res = solve_lp_direction(f, n, M_try, m, total, row, col_data, c_obj, bounds)
            except Exception:
                continue
            
            x = getattr(res, 'x', None)
            if x is None:
                continue
            
            x = np.asarray(x, dtype=np.float64).ravel()
            x = np.where(np.isfinite(x), x, 0.0)
            x = np.maximum(x, 0.0)
            s_star = float(x.sum())
            if s_star <= 1e-300:

                    best_t = t2

                else:
                    t_lo = t1

            if best_t > 0 and best_sc < best_score:

                found_improvement = True

        if time.monotonic() - t0 > time_budget * 0.5 and iteration > 50:
            break
    
    return best_f, float(best_score)


def interpolate_sequence(x, n):
    """Resample a nonnegative sequence to length n and normalize."""
    x = normalize_sequence(x)
    if x is None or n <= 0:
        return None
    m = x.size
    if m == n:
        return x
    if m == 1:
        return np.ones(n) / n
    old = np.linspace(0.0, 1.0, m)
    new = np.linspace(0.0, 1.0, n)
    out = np.interp(new, old, x)
    return normalize_sequence(out)

"""
Combined LP + Projected Gradient search for minimizing 2*n*max(conv(f,f))/(sum(f))^2.

Strategy:
- Phase 1: LP-based iteration with golden-section step size and multiple constraint levels.
- Phase 2: Projected gradient descent with adaptive softmax smoothing of the max.
- Phase 3: Adaptive switching between LP and gradient, with structured restarts.
- The best sequence found across all phases is returned.
"""

import time
import numpy as np
from scipy.optimize import linprog
from scipy.sparse import csr_matrix


def normalize_sequence(arr):
    """Return a nonnegative float64 vector normalized to sum 1, or None."""
    try:
        x = np.asarray(arr, dtype=np.float64).ravel()
    except Exception:
        return None
    if x.size == 0:
        return None
    x = np.where(np.isfinite(x), x, 0.0)
    x = np.maximum(x, 0.0)
    s = float(x.sum())
    if s <= 1e-300:
        return None
    x /= s
    return x


def score_normalized(x, n):
    """Score assuming x is nonnegative and sum(x) = 1. Returns 2*n*max(conv(x,x))."""
    conv = np.convolve(x, x)
    m = float(np.max(conv))
    if not np.isfinite(m):
        return float("inf")
    return float(2.0 * n * m)


def project_to_simplex(v):
    """Project vector v onto the non-negative simplex {x >= 0, sum(x) = 1}."""
    v = np.asarray(v, dtype=np.float64).copy()
    n = v.size
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not np.any(cond):
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    s = w.sum()
    if s > 0:
        w /= s
    else:
        w = np.ones(n) / n
    return w


def softmax_weights(conv, alpha):
    """Compute softmax(alpha * conv) with numerical stability."""
    c = np.asarray(conv, dtype=np.float64)
    c_max = float(c.max())
    shifted = alpha * (c - c_max)
    # Clip to avoid underflow issues (values far below max get weight ~0 anyway)
    shifted = np.maximum(shifted, -700.0)
    exp_vals = np.exp(shifted)
    s = float(exp_vals.sum())
    if s <= 0 or not np.isfinite(s):
        p = np.zeros_like(c)
        p[np.argmax(c)] = 1.0
        return p
    return exp_vals / s


def compute_gradient(f, alpha):
    """
    Compute gradient of the smoothed max-convolution objective.
    J_alpha(f) = (1/alpha) * log(sum_k exp(alpha * (f*f)_k))
    grad_j = 2 * sum_k p_k * f_{k-j} where p = softmax(alpha * (f*f))
    """
    n = f.size
    c = np.convolve(f, f)  # length 2n-1
    p = softmax_weights(c, alpha)
    # grad_j = 2 * (p * f_rev)[j + n - 1] where f_rev = f[::-1]
    f_rev = f[::-1]
    cross = np.convolve(p, f_rev)  # length 3n-1
    grad = 2.0 * cross[n - 1:2 * n - 1]  # length n
    return grad


def gradient_step(f, alpha, lr):
    """One projected gradient descent step. Returns new f."""
    n = f.size
    grad = compute_gradient(f, alpha)
    if not np.all(np.isfinite(grad)):
        return f
    f_new = f - lr * grad
    return project_to_simplex(f_new)


def make_conv_template(n):
    """
    Precompute CSR structure for the convolution constraint matrix A.
    A_{k,j} = f_{k-j} (with f_m = 0 outside [0,n-1]).
    Returns (m, n_nonzeros, row_indices, col_indices) for building the sparse matrix.
    """
    m = 2 * n - 1
    row_all = []
    col_all = []
    for k in range(m):
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        for j in range(js, je + 1):
            row_all.append(k)
            col_all.append(j)
    row = np.array(row_all, dtype=np.int32)
    col = np.array(col_all, dtype=np.int32)
    n_nz = len(row)
    return m, n_nz, row, col


def solve_lp_direction(f, n, M, m, n_nz, row, col, c_obj, bounds):
    """Solve LP: max sum(g) s.t. (f*g)_k <= M for all k, g >= 0."""
    A_data = f[col]
    A = csr_matrix((A_data, (row, col)), shape=(m, n), dtype=np.float64)
    b = np.full(m, M, dtype=np.float64)
    res = linprog(c_obj, A_ub=A, b_ub=b, bounds=bounds, method='highs')
    return res


def golden_section_search(f, g, t_lo, t_hi, best_score, max_iter=40):
    """
    Golden section search for best t in [t_lo, t_hi] minimizing
    score of project_to_simplex((1-t)*f + t*g).
    Returns (best_t, best_score, best_f).
    """
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    best_t = 0.0
    best_sc = best_score
    best_f = None

    a = t_lo
    b = t_hi
    c = b - gr * (b - a)
    d = a + gr * (b - a)

    fc = score_normalized(project_to_simplex((1.0 - c) * f + c * g), f.size)
    fd = score_normalized(project_to_simplex((1.0 - d) * f + d * g), f.size)

    for _ in range(max_iter):
        if b - a < 1e-12:
            break
        if fc < fd:
            b = d
            d = c
            fd = fc
            c = b - gr * (b - a)
            fc = score_normalized(project_to_simplex((1.0 - c) * f + c * g), f.size)
        else:
            a = c
            c = d
            fc = fd
            d = a + gr * (b - a)
            fd = score_normalized(project_to_simplex((1.0 - d) * f + d * g), f.size)

        if np.isfinite(fc) and fc < best_sc:
            best_sc = fc
            best_t = c
            best_f = project_to_simplex((1.0 - c) * f + c * g)
        if np.isfinite(fd) and fd < best_sc:
            best_sc = fd
            best_t = d
            best_f = project_to_simplex((1.0 - d) * f + d * g)

    return best_t, best_sc, best_f

def interpolate_sequence(x, n):
    """Resample a nonnegative sequence to length n and normalize."""
    x = normalize_sequence(x)
    if x is None or n <= 0:
        return None
    m = x.size
    if m == n:
        return x
    if m == 1:
        return np.ones(n, dtype=np.float64) / n
    old = np.linspace(0.0, 1.0, m, dtype=np.float64)
    new = np.linspace(0.0, 1.0, n, dtype=np.float64)
    out = np.interp(new, old, x)
    return normalize_sequence(out)

                mode = 'lp'  # Stay in LP mode
            else:
                consecutive_no_improve += 1
                if consecutive_no_improve >= 3:
                    # Switch to gradient
                    mode = 'gd'
                    gd_alpha = 5.0
                    gd_lr = 0.02
                    print(f"  Switching to GD after {consecutive_no_improve} LP stalls", flush=True)
                    consecutive_no_improve = 0

        elif mode == 'gd':
            # Run a batch of gradient iterations
            gd_budget = min(30.0, deadline - time.monotonic())
            if gd_budget < 1.0:
                break

            f_res, s_res = gradient_optimize(n, f_current, gd_budget, rng,
                                             alpha_start=gd_alpha, alpha_end=2000.0)
            if f_res is not None and np.isfinite(s_res) and s_res < best_score:
                best_score = s_res
                best_f = f_res
                consecutive_no_improve = 0
                if time.monotonic() - last_print > 30:
                    print(f"  [GD] iter={total_iters} score={best_score:.6f} "
                          f"elapsed={time.monotonic()-start_time:.0f}s", flush=True)
                    last_print = time.monotonic()
                # Stay in GD or switch back to LP
                mode = 'lp' if s_res < best_score + 1e-8 else 'gd'
            else:
                consecutive_no_improve += 1
                if consecutive_no_improve >= 5:
                    # Try structured restart
                    restarts_tried = 0
                    for _ in range(5):
                        if time.monotonic() >= deadline:
                            break
                        restarts_tried += 1
                        if restarts_tried == 1:
                            # Small perturbation
                            f_try = project_to_simplex(best_f + rng.normal(0, 0.003, n))
                        elif restarts_tried == 2:
                            # Medium perturbation
                            f_try = project_to_simplex(best_f + rng.normal(0, 0.01, n))
                        elif restarts_tried == 3:
                            # Mix with Dirichlet
                            w = rng.dirichlet(np.ones(n) * 0.5)
                            f_try = project_to_simplex(0.9 * best_f + 0.1 * w)
                        elif restarts_tried == 4:
                            # Mix with uniform
                            f_try = project_to_simplex(0.85 * best_f + 0.15 * np.ones(n) / n)
                        else:
                            # Symmetric perturbation
                            half = n // 2
                            pert = rng.normal(0, 0.005, half)
                            f_try = best_f.copy()
                            f_try[:half] += pert
                            f_try[half:] += pert[::-1]
                            f_try = project_to_simplex(f_try)

                        sc_try = score_normalized(f_try, n)
                        if np.isfinite(sc_try) and sc_try < best_score:
                            best_score = sc_try
                            best_f = f_try
                            print(f"  Restart improved! score={best_score:.6f}", flush=True)
                            break

                    mode = 'lp'
                    consecutive_no_improve = 0

        # Safety: check time
        if time.monotonic() >= deadline:
            break

    # Final polish with gradient
    remaining = deadline - time.monotonic()
    if remaining > 5 and best_f is not None:
        print(f"\nFinal gradient polish ({remaining:.0f}s remaining)...", flush=True)
        f_gd, s_gd = gradient_optimize(n, best_f, remaining, rng,
                                       alpha_start=50.0, alpha_end=5000.0)
        if f_gd is not None and np.isfinite(s_gd) and s_gd < best_score:
            best_score = s_gd
            best_f = f_gd
            print(f"Final GD improved: {best_score:.6f}", flush=True)

    # Convert to output format
    seq = [float(max(0.0, min(1000.0, x))) for x in best_f]

    # Verify
    try:
        final_score = float(evaluate_sequence(seq))
        print(f"\nFINAL: n={len(seq)}, local_score={best_score:.6f}, "
              f"evaluate_sequence={final_score:.6f}", flush=True)
    except Exception as e:
        print(f"Evaluation error: {e}", flush=True)
        # Fallback to given sequence or uniform
        if given_f is not None:
            seq = [float(max(0.0, min(1000.0, x))) for x in given_f]
        else:
            seq = [1.0 / 1024.0] * 1024

    assert isinstance(seq, list)
    assert len(seq) > 0
    return seq
