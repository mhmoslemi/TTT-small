Uses LP direction finding, coordinate descent refinement, and multi-start strategy.

from scipy.optimize import linprog
from scipy.sparse import csr_matrix


def normalize(arr):
    """Normalize a non-negative array to sum 1."""
    x = np.asarray(arr, dtype=np.float64).ravel()
    if x.size == 0:
        return None
    x = np.where(np.isfinite(x), x, 0.0)
    x = np.maximum(x, 0.0)
    s = float(x.sum())
    if s < 1e-300:
        return None
    return x / s


def score_norm(x):
    """Score assuming x is non-negative and sum(x) = 1."""
    n = x.size
    conv = np.convolve(x, x)
    M = float(np.max(conv))
    if not np.isfinite(M):
        return float('inf')
    return 2.0 * n * M


def build_conv_template(n):
    """
    Build a sparse matrix template for computing f*g cross-correlation.
    Returns (A_structure, offsets, m) where:
      A_structure is a CSR matrix with the sparsity pattern
      offsets is an array: for each non-zero entry in A, the offset = k - col
      m is the number of rows (2n-1)
    To compute f*g, fill A.data with f[offsets] and multiply by g.
    """
    m = 2 * n - 1
    k_arr = np.arange(m, dtype=np.int64)
    i_start = np.maximum(0, k_arr - (n - 1)).astype(np.int64)
    i_end = np.minimum(n - 1, k_arr).astype(np.int64)
    counts = i_end - i_start + 1
    total = int(counts.sum())
    
    row_idx = np.repeat(k_arr, counts)
    # Generate column indices
    col_idx = np.empty(total, dtype=np.int64)
    pos = 0
    for k in range(m):
        col_idx[pos:pos + int(counts[k])] = np.arange(i_start[k], i_end[k] + 1)
        pos += int(counts[k])
    
    offsets = (row_idx - col_idx).astype(np.int32)
    A = csr_matrix((np.ones(total, dtype=np.float64), (row_idx.astype(np.int32), col_idx.astype(np.int32))), shape=(m, n))
    return A, offsets, m


def solve_lp_direction(template, f, M):
    """
    Solve: max sum(g) s.t. (f*g)_k <= M for all k, g >= 0.
    Returns optimal g and optimal value.
    """
    A_template, offsets, m = template
    n = f.size
    
    # Build A: A[k, i] = f[k-i] for valid k-i
    A = A_template.copy()
    A.data[:] = f[offsets]
    
    b = np.full(m, M, dtype=np.float64)
    c = -np.ones(n, dtype=np.float64)
    
    res = linprog(c, A_ub=A, b_ub=b, bounds=[(0, None)] * n, method='highs')
    
    if res.status != 0:
        return None, None
    return res.x, res.fun

def mix_with_random(f, rng, eps):
    """Mix f with a random Dirichlet vector."""
    n = f.size
    if eps >= 1.0:

    if eps <= 0.0:
        return f.copy()
    d = rng.dirichlet(np.ones(n))
    return (1 - eps) * f + eps * d

"""
Improved LP-based iterative search for minimizing 2*n*max(convolve(f,f))/(sum(f))^2.
Uses LP direction finding with multi-direction perturbation, fast coordinate descent,
adaptive M-shrinking, and diverse restarts to escape local optima.
"""

import time
import numpy as np
from scipy.optimize import linprog
from scipy.sparse import csr_matrix


def normalize(arr):
    """Normalize a non-negative array to sum 1."""
    x = np.asarray(arr, dtype=np.float64).ravel()
    if x.size == 0:
        return None
    x = np.where(np.isfinite(x), x, 0.0)
    x = np.maximum(x, 0.0)
    s = float(x.sum())
    if s < 1e-300:
        return None
    return x / s


def score_norm(x):
    """Score assuming x is non-negative and sum(x) = 1. Returns 2*n*max(conv)."""
    n = x.size
    conv = np.convolve(x, x)
    M = float(np.max(conv))
    if not np.isfinite(M):
        return float('inf')
    return 2.0 * n * M


def build_conv_template(n):
    """
    Build sparse template for computing f*g (full convolution / cross-correlation).
    Returns (A_template, offsets, m) where A_template has the sparsity pattern,
    offsets gives which element of f to use for each entry.
    """
    m = 2 * n - 1
    k_arr = np.arange(m, dtype=np.int64)
    i_start = np.maximum(0, k_arr - (n - 1)).astype(np.int64)
    i_end = np.minimum(n - 1, k_arr).astype(np.int64)
    counts = i_end - i_start + 1
    total = int(counts.sum())

    row_idx = np.repeat(k_arr, counts)
    col_idx = np.empty(total, dtype=np.int64)
    pos = 0
    for k in range(m):
        c = int(counts[k])
        col_idx[pos:pos + c] = np.arange(i_start[k], i_end[k] + 1)
        pos += c

    offsets = (row_idx - col_idx).astype(np.int32)
    A = csr_matrix(
        (np.ones(total, dtype=np.float64),
         (row_idx.astype(np.int32), col_idx.astype(np.int32))),
        shape=(m, n)
    )
    return A, offsets, m


def solve_lp(template, f, M, c_vec=None):
    """Solve: max c^T g s.t. (f*g)_k <= M for all k, g >= 0."""
    A_template, offsets, m = template
    n = f.size

    A = A_template.copy()
    A.data[:] = f[offsets]

    b = np.full(m, M, dtype=np.float64)
    if c_vec is None:
        c_vec = -np.ones(n, dtype=np.float64)

    res = linprog(c_vec, A_ub=A, b_ub=b, bounds=[(0, None)] * n, method='highs')

    if res.status != 0:
        return None
    return res.x


def fast_coord_descend(x, n_passes=2):
    """
    Fast coordinate descent: for each coordinate, try a few candidate values
    and keep the one that minimizes the score.
    """
    x = x.copy()
    n = x.size
    best_score = score_norm(x)

    for _p in range(n_passes):
        any_improved = False
        for i in range(n):
            old = x[i]
            if old < 1e-12:
                continue

            candidates = [0.0, old * 0.5, old * 0.75, old * 1.5]
            cur_best = x.copy()
            cur_best_score = best_score

            for cand in candidates:
                if cand < 0 or cand < 1e-15:
                    x[i] = 0.0
                else:
                    x[i] = cand
                s = float(x.sum())
                if s < 1e-300:
                    x[i] = old
                    continue
                xn = x / s
                sc = score_norm(xn)
                if sc < cur_best_score - 1e-12:
                    cur_best_score = sc
                    cur_best = xn.copy()

            x[i] = old
            if cur_best_score < best_score - 1e-12:
                x = cur_best
                best_score = cur_best_score
                any_improved = True

        if not any_improved:
            break

    return x, best_score


def mix_random(f, rng, eps):
    """Mix f with a random Dirichlet vector."""
    n = f.size
    if eps >= 1.0:
        return rng.dirichlet(np.ones(n))
    if eps <= 0.0:
        return f.copy()
    d = rng.dirichlet(np.ones(n))
    return (1.0 - eps) * f + eps * d


def interpolate_to_length(f, n):
    """Resample f to length n and normalize."""
    m = f.size
    if m == n:
        return f.copy()
    if m == 1:
        return np.ones(n) / n
    old_t = np.linspace(0, 1, m)
    new_t = np.linspace(0, 1, n)
    x = np.interp(new_t, old_t, f)
    return normalize(x)
