def project_simplex(v):

    """Alternating projection for flat-spectrum initialization."""
    N = 1 << (2 * n - 2).bit_length()
    phases = rng.uniform(0, 2 * np.pi, N // 2 + 1)
    F = np.exp(1j * phases)
    F[0] = 1.0 + 0j
    a = np.ones(n) / n
    for it in range(n_iter):
        a_full = np.fft.irfft(F, N)
        a = a_full[:n].copy()
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s < 1e-10:
            a = np.ones(n) / n
        else:
            a = a / s
        F_full = np.fft.rfft(a, N)
        F = np.ones(len(F_full)) * np.exp(1j * np.angle(F_full))
        F[0] = F[0].real + 0j
    return a


def nesterov_subgradient(a0, deadline, rng, lp_interval=120, use_lp=True):
    """Nesterov-accelerated projected subgradient descent with LP steps."""
    n = len(a0)
    a = project_simplex(a0.astype(np.float64).copy())
    a_prev = a.copy()
    momentum = 0.82

    c = fast_convolve(a)
    best_peak = c.max()
    best_a = a.copy()

    lr = 4.0 / n
    lr_min = lr * 1e-4
    stall = 0
    max_stall = 400
    it = 0

    while time.time() < deadline:
        it += 1

        # Nesterov extrapolation
        y = a + momentum * (a - a_prev)
        y = np.maximum(y, 0.0)
        sy = y.sum()
        if sy < 1e-10:
            y = a.copy()
        else:
            y = y / sy

        # Evaluate at y
        c = fast_convolve(y)
        peak = c.max()

        # Track best periodically
        if it % 5 == 1:
            c_a = fast_convolve(a)
            peak_a = c_a.max()
            if peak_a < best_peak - 1e-10:
                best_peak = peak_a
                best_a = a.copy()
                stall = 0
            else:
                stall += 1

        # Adaptive LR
        if stall > max_stall:
            lr = max(lr * 0.7, lr_min)
            stall = 0
            if lr < lr_min * 10 and time.time() < deadline - 20:
                noise = rng.normal(0, 0.03 / np.sqrt(n), n)
                a = project_simplex(a + noise)
                a_prev = a.copy()

        # Active set
        threshold = peak * (1.0 - 3e-4)
        peak_idx = np.where(c >= threshold)[0]
        if len(peak_idx) > 20:
            top = np.argsort(c[peak_idx])[::-1][:8]
            peak_idx = peak_idx[top]

        # Subgradient at y
        grad = compute_subgradient(y, peak_idx)

        # Update
        a_prev = a.copy()
        a = project_simplex(y - lr * grad)

        # LP step with line search
        if use_lp and n <= 800 and it % lp_interval == 0 and time.time() < deadline - 50:
            b = lp_step(a, best_peak, time_limit=3.0)
            if b is not None:
                best_t = 0.0
                best_p = best_peak
                for t_val in [0.15, 0.25, 0.35, 0.45, 0.55, 0.7]:
                    mix = project_simplex((1 - t_val) * a + t_val * b)
                    p = fast_convolve(mix).max()
                    if p < best_p:
                        best_p = p
                        best_t = t_val
                if best_t > 0:
                    a = project_simplex((1 - best_t) * a + best_t * b)
                    a_prev = a.copy()

        # Slowly decrease momentum
        if it % 1000 == 0:
            momentum = max(0.5, momentum - 0.02)

    return best_a


def coordinate_descent(a0, deadline, max_sweeps=200):
    """Fast coordinate descent using analytical O(n) convolution update per coordinate."""
    n = len(a0)
    a = a0.copy().astype(np.float64)
    c = fast_convolve(a)
    conv_size = 2 * n - 1

    for sweep in range(max_sweeps):
        if time.time() >= deadline:
            break
        any_improve = False

        for m in range(n):
            if time.time() > deadline:
                break

            current_am = a[m]
            if current_am < 1e-14:
                continue

            # Build shift vector: shift[k] = a[k-m] if 0 <= k-m < n else 0
            shift = np.zeros(conv_size)
            lo = m
            hi = min(m + n - 1, conv_size - 1)
            shift[lo:hi + 1] = a[:hi + 1 - lo]

            # Ternary search for optimal delta in [-current_am, current_am * 2]
            lo_d = -current_am
            hi_d = current_am * 2.0

            for _ in range(30):
                m1 = lo_d + (hi_d - lo_d) / 3.0
                m2 = hi_d - (hi_d - lo_d) / 3.0

                new_c1 = c + 2.0 * m1 * shift
                if 2 * m < conv_size:
                    new_c1[2 * m] += m1 * m1
                f1 = new_c1.max()

                new_c2 = c + 2.0 * m2 * shift
                if 2 * m < conv_size:
                    new_c2[2 * m] += m2 * m2
                f2 = new_c2.max()

                if f1 < f2:
                    hi_d = m2
                else:
                    lo_d = m1

            best_delta = (lo_d + hi_d) / 2.0
            new_am = current_am + best_delta
            if new_am < 0:
                new_am = 0.0

            if abs(best_delta) > 1e-10:
                a[m] = new_am
                c += 2.0 * best_delta * shift
                if 2 * m < conv_size:
                    c[2 * m] += best_delta * best_delta
                c = np.maximum(c, 0.0)
                any_improve = True

        if not any_improve:
            break

    return a


def propose_candidate(budget_s=1500):
    """
    Multi-phase optimizer: Nesterov subgradient + coordinate descent polish + LP refinement.
    Tries multiple lengths and diverse initializations (QR, chirp, cosine, alt-projection).
    """
    overall_deadline = time.time() + budget_s - 25.0
    rng = np.random.default_rng(77)

    best_score = float('inf')
    best_seq = None

    # Get parent sequence
    parent = None
    try:
        parent = np.array(height_sequence_1, dtype=np.float64)
        parent = np.clip(parent, 0.0, 1000.0)
        if parent.sum() > 1e-10:
            parent = parent / parent.sum()
    except Exception:
        parent = None

    # Phase 1: Subgradient optimization for various configs (60% of time)
    phase1_end = time.time() + (budget_s - 25) * 0.55

    # Define (n, init_type, time_fraction)
    configs = []
    if parent is not None:
        configs.append((len(parent), 'parent', 0.25))
    configs.append((511, 'qr', 0.15))
    configs.append((512, 'chirp', 0.12))
    configs.append((383, 'qr', 0.10))
    configs.append((255, 'qr', 0.08))
    configs.append((512, 'alt_proj', 0.10))
    configs.append((384, 'cosine', 0.08))
    configs.append((768, 'chirp', 0.07))
    configs.append((640, 'cosine', 0.05))

    for n_cfg, init_type, frac in configs:
        if time.time() > phase1_end - 30:
            break
        alloc_time = (budget_s - 25) * frac * 0.55
        cfg_deadline = min(phase1_end - 10, time.time() + alloc_time)
        if cfg_deadline < time.time() + 15:
            continue

        # Generate initialization
        try:
            if init_type == 'parent' and parent is not None:
                if len(parent) == n_cfg:
                    a_init = parent.copy()
                elif len(parent) > n_cfg:
                    a_init = parent[:n_cfg].copy()
                    a_init = a_init / max(a_init.sum(), 1e-10)
                else:
                    a_init = np.zeros(n_cfg)
                    a_init[:len(parent)] = parent
                    a_init = a_init / max(a_init.sum(), 1e-10)
            elif init_type == 'qr':
                a_init = qr_init(n_cfg, rng)
            elif init_type == 'chirp':
                a_init = chirp_init(n_cfg, rng)
            elif init_type == 'cosine':
                a_init = cosine_init(n_cfg, rng, freq=1.0 + 0.5 * rng.random())
            elif init_type == 'alt_proj':
                a_init = alt_projection_init(n_cfg, rng, n_iter=40)
            else:
                a_init = rng.random(n_cfg)
                a_init = a_init / a_init.sum()
        except Exception:
            a_init = rng.random(n_cfg)
            a_init = a_init / a_init.sum()

        use_lp = n_cfg <= 700
        a_opt = nesterov_subgradient(a_init, cfg_deadline, rng, lp_interval=100, use_lp=use_lp)

        # Evaluate
        try:
            score = evaluate_sequence(a_opt.tolist())
        except Exception:
            score = float('inf')

        print(f"Phase1: n={n_cfg}, init={init_type}, score={score:.6f}, time_left={overall_deadline - time.time():.0f}s")

        if np.isfinite(score) and score < best_score:
            best_score = score
            best_seq = a_opt.copy()
            print(f"  >>> New best: {best_score:.6f}")

    # Phase 2: Coordinate descent polish on best (25% of time)
    phase2_end = time.time() + (budget_s - 25) * 0.25
    if best_seq is not None and time.time() < phase2_end:
        print(f"Phase2: Coordinate descent polish on n={len(best_seq)}")
        a_cd = coordinate_descent(best_seq, phase2_end, max_sweeps=500)
        try:
            score_cd = evaluate_sequence(a_cd.tolist())
        except Exception:
            score_cd = float('inf')
        print(f"  After CD: score={score_cd:.6f}")
        if np.isfinite(score_cd) and score_cd < best_score:
            best_score = score_cd
            best_seq = a_cd.copy()

    # Phase 3: Further subgradient + LP on best (remaining time)
    if best_seq is not None and time.time() < overall_deadline - 30:
        print(f"Phase3: Final subgradient + LP polish, n={len(best_seq)}")
        n_best = len(best_seq)
        use_lp = n_best <= 700
        a_final = nesterov_subgradient(best_seq, overall_deadline - 10, rng,
                                       lp_interval=80, use_lp=use_lp)
        # Quick CD on top
        if time.time() < overall_deadline - 10:
            a_final = coordinate_descent(a_final, overall_deadline - 5, max_sweeps=200)

        try:
            score_final = evaluate_sequence(a_final.tolist())
        except Exception:
            score_final = float('inf')
        print(f"  Final: score={score_final:.6f}")
        if np.isfinite(score_final) and score_final < best_score:
            best_score = score_final
            best_seq = a_final.copy()

    # Fallback
    if best_seq is None:
        print("FALLBACK: uniform sequence")
        best_seq = np.ones(256)
        best_score = evaluate_sequence(best_seq.tolist())

    result = [float(x) for x in best_seq]
    final_check = evaluate_sequence(result)
    print(f"FINAL: score={final_check:.6f}, length={len(result)}")
    if np.isfinite(final_check) and final_check < best_score:
        best_score = final_check
    return result
