```python
"""
LP-based optimization with 'minimize M' formulation, gradient polishing,
and multiple restarts/lengths to minimize 2*n*max(conv(a,a))/sum(a)^2.
"""

import numpy as np
from scipy.optimize import linprog
from scipy.sparse import csr_matrix
import time


def compute_score(f):
    """Compute 2n * max(conv) for normalized sequence (sum=1)."""
    n = len(f)
    conv = np.convolve(f, f)
    return 2.0 * n * conv.max()


def build_constraint_matrix(f):
    """Build sparse matrix for LP: rows are (2n-1) conv constraints, cols are (n for b, 1 for M)."""
    n = len(f)
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        j_start = max(0, k - n + 1)
        j_end = min(n, k + 1)
        for j in range(j_start, j_end):
            rows.append(k)
            cols.append(j)
            vals.append(f[k - j])
        # M coefficient
        rows.append(k)
        cols.append(n)
        vals.append(-1.0)
    return csr_matrix((vals, (rows, cols)), shape=(2*n-1, n+1))


def solve_lp_minimize_M(f, M_bound=None):
    """
    Solve: min M s.t. f*b <= M (elementwise, all 2n-1 positions), sum(b)=1, b>=0.
    Returns (b_opt, M_opt).
    """
    n = len(f)
    A = build_constraint_matrix(f)
    
    # Objective: minimize M (last variable)
    c = np.zeros(n + 1)
    c[n] = 1.0
    
    # Equality: sum(b) = 1
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])
    
    # Bounds: b_i >= 0, M free (or M >= 0)
    bounds = [(0, None)] * n + [(0, None)]
    
    res = linprog(c, A_ub=A, b_ub=np.zeros(2*n-1), A_eq=A_eq, b_eq=b_eq,
                  bounds=bounds, method='highs')
    
    if not res.success:
        return None, None
    
    b_opt = np.clip(res.x[:n], 0, None)
    M_opt = res.x[n]
    return b_opt, M_opt


def gradient_step(f, lr):
    """One gradient descent step on the score (using current argmax)."""
    n = len(f)
    conv = np.convolve(f, f)
    k_max = conv.argmax()
    M = conv[k_max]
    S = f.sum()
    
    # Gradient: d(score)/da[m] = (4n/S^2) * (a[k_max - m] - M/S) (where valid)
    grad = np.zeros(n)
    for m in range(n):
        km = k_max - m
        if 0 <= km < n:
            grad[m] = (4.0 * n / S**2) * (f[km] - M / S)
    
    f_new = f - lr * grad
    f_new = np.clip(f_new, 0, None)
    S_new = f_new.sum()
    if S_new < 1e-10:
        return None
    f_new = f_new / S_new
    return f_new


def gradient_polish(f, n_steps=10000, lr0=0.01):
    """Polish with gradient descent, adaptive learning rate."""
    best_f = f.copy()
```

```python
LP-based optimization with 'minimize M' formulation, projected gradient with momentum,
symmetrization, and multi-length/multi-restart search to minimize 2*n*max(conv(a,a))/sum(a)^2.
"""

import numpy as np
from scipy.sparse import csr_matrix, coo_matrix
from scipy.optimize import linprog
import time


def compute_score(f):
    """Compute 2n * max(conv) for a sequence (assumed normalized to sum=1)."""
    n = len(f)
    conv = np.convolve(f, f)
    return 2.0 * n * conv.max()


def compute_score_general(f):
    """Compute the full score = 2*n*max(conv)/sum^2 for any non-negative sequence."""
    n = len(f)
    S = f.sum()
    if S < 1e-15:
        return np.inf
    conv = np.convolve(f, f)
    return 2.0 * n * conv.max() / (S * S)


def build_lp_matrix(f):
    """Build sparse constraint matrix for LP: min M s.t. f*b - M <= 0, sum(b)=1, b>=0.
    Matrix shape: (2n-1, n+1). Last column is -1 for the M variable."""
    n = len(f)
    rows_list = []
    cols_list = []
    vals_list = []
    for k in range(2 * n - 1):
        j_start = max(0, k - n + 1)
        j_end = min(n, k + 1)
        for j in range(j_start, j_end):
            rows_list.append(k)
            cols_list.append(j)
            vals_list.append(f[k - j])
        rows_list.append(k)
        cols_list.append(n)
        vals_list.append(-1.0)
    return csr_matrix((vals_list, (rows_list, cols_list)), shape=(2 * n - 1, n + 1))


def solve_lp(f):
    """Solve min M s.t. (f*b)(k) <= M for all k, sum(b)=1, b>=0.
    Returns (b_opt, M_opt) or (None, None) on failure."""
    n = len(f)
    A = build_lp_matrix(f)
    c = np.zeros(n + 1)
    c[n] = 1.0
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])
    bounds = [(0, None)] * n + [(0, None)]
    b_ub = np.zeros(2 * n - 1)
    try:
        res = linprog(c, A_ub=A, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs',
                      options={'presolve': True, 'dual_feasibility_tolerance': 1e-9})
    except Exception:
        return None, None
    if not res.success:
        return None, None
    b_opt = np.clip(res.x[:n], 0, None)
    if b_opt.sum() < 1e-15:
        return None, None
    b_opt = b_opt / b_opt.sum()
    return b_opt, res.x[n]

        if b_opt is None:
            break

        score_b = compute_score(b_opt)
        improved = False

        if score_b < best_score:
            f = b_opt.copy()
            best_f = f.copy()
            best_score = score_b
            stall = 0
            improved = True
        else:
            for t in [0.5, 0.3, 0.7, 0.2, 0.1, 0.8]:
                f_try = (1.0 - t) * f + t * b_opt
                f_try = np.clip(f_try, 0, None)
                S_try = f_try.sum()
                if S_try < 1e-10:
                    continue
                f_try = f_try / S_try
                s_try = compute_score(f_try)
                if s_try < best_score - 1e-12:
                    f = f_try
                    best_f = f.copy()
                    best_score = s_try
                    stall = 0
                    improved = True
                    break

        if not improved:
            stall += 1
            if stall > 30:
                rng = np.random.default_rng(it)
                f_pert = f + rng.normal(0, 0.03, n)
                f_pert = np.clip(f_pert, 0, None)
                S = f_pert.sum()
                if S > 1e-10:
                    f_pert /= S
                    if compute_score(f_pert) < best_score:
                        f = f_pert
                        best_f = f.copy()
                        best_score = compute_score(f)
                        stall = 0
                if stall > 80:
                    break

        it += 1
        if verbose and it % 5 == 0:
            elapsed = time.monotonic() - start
            print(f"      LP it={it}, score={best_score:.6f}, elapsed={elapsed:.1f}s")

    return best_f, best_score


def symmetrize(f):
    """Average with reversed version to encourage symmetry."""
    return (f + f[::-1].copy()) / 2.0


def coordinate_descent_polish(f, time_budget=30.0):
    """Polish via coordinate descent: optimize one coordinate at a time."""
    n = len(f)
    f = f.copy()
    f = f / f.sum()
    best_f = f.copy()
    best_score = compute_score(f)
    start = time.monotonic()

    while time.monotonic() - start < time_budget:
        improved = False
        for m in range(n):
            if time.monotonic() - start > time_budget:
                break
            old_val = f[m]
            if old_val < 1e-15:
                continue
            # Try a few values around old_val
            for scale in [0.5, 0.7, 1.3, 0.3, 1.5, 0.8, 1.2]:
                trial = f.copy()
                trial[m] = old_val * scale
                S = trial.sum()
                if S < 1e-10:
                    continue
                trial = trial / S
                s = compute_score(trial)
                if s < compute_score(f) - 1e-12:
                    f = trial
                    improved = True
                    break
        if not improved:
            break
        if compute_score(f) < best_score:
            best_f = f.copy()
            best_score = best_score if compute_score(best_f) < compute_score(f) else compute_score(f)
            best_f = f.copy()
            best_score = compute_score(f)

    return best_f, best_score


def propose_candidate(seed=42, budget_s=1500, **kwargs):
    """Main search: multi-length, multi-start LP + gradient optimization."""
    rng = np.random.default_rng(seed)
    start_time = time.monotonic()
    deadline = start_time + budget_s - 10.0

    best_overall = None
    best_score_overall = np.inf

    # Get parent sequence
    parent = None
    try:
        parent = np.array(height_sequence_1, dtype=np.float64)
        if parent.ndim != 1 or len(parent) == 0 or not np.isfinite(parent).all():
            parent = None
    except Exception:
        pass

    if parent is not None:
        parent = np.clip(parent, 0, None)
        if parent.sum() > 1e-10:
            parent = parent / parent.sum()
            parent_score = compute_score(parent)
            print(f"Parent: n={len(parent)}, score={parent_score:.6f}")
        else:
            parent = None

    # Build list of (name, f0) candidates
    candidates = []

    # Parent (highest priority)
    if parent is not None:
        candidates.append(('parent', parent.copy()))
        # Slightly perturbed parent
        f_pert = parent * (1 + 0.1 * rng.normal(0, 1, len(parent)))
        f_pert = np.clip(f_pert, 0, None)
        if f_pert.sum() > 1e-10:
            candidates.append(('parent_perturbed', f_pert / f_pert.sum()))
        # Symmetrized parent
        f_sym = symmetrize(parent)
        f_sym = np.clip(f_sym, 0, None)
        if f_sym.sum() > 1e-10:
            candidates.append(('parent_sym', f_sym / f_sym.sum()))

    # Various lengths with different initializations
    lengths_to_try = [128, 256, 384, 512, 1024]

    for n in lengths_to_try:
        # Uniform
        candidates.append((f'uniform_{n}', np.ones(n) * (1 + 0.05 * rng.normal(0, 1, n))))
        # Random
        candidates.append((f'random_{n}', rng.random(n) + 0.05))
        # Sine-based (to break symmetry in an interesting way)
        x = np.linspace(0, 2 * np.pi, n)
        candidates.append((f'sine_{n}', 1 + 0.3 * np.sin(x) + 0.1 * np.sin(2 * x)))

    # Clean up candidates
    clean_candidates = []
    for name, f0 in candidates:
        f0 = np.clip(f0, 0, None).astype(np.float64)
        if f0.sum() > 1e-10:
            f0 = f0 / f0.sum()
            clean_candidates.append((name, f0))

    n_cands = len(clean_candidates)
    print(f"\nTotal candidates: {n_cands}")
    print(f"Target: 1.5027")

    # Phase 1: Quick screen of all candidates with short LP
    quick_time = min(15.0, max(5.0, 60.0 / n_cands))
    print(f"\n=== Phase 1: Quick screen ({quick_time:.0f}s per candidate) ===")
    screened = []
    for name, f0 in clean_candidates:
        if time.monotonic() > deadline:
            break
        f_opt, score = lp_optimize(f0, time_budget=quick_time, verbose=False)
        screened.append((name, f0, f_opt, score, len(f0)))
        if score < best_score_overall:
            best_score_overall = score
            best_overall = f_opt.copy()
        print(f"  {name}: n={len(f0)}, score={score:.6f}")

    # Sort by score (best first)
    screened.sort(key=lambda x: x[3] if np.isfinite(x[3]) else np.inf)
    print("\nScreening results (best first):")
    for name, f0, f_opt, score, n in screened[:8]:
        print(f"  {name} (n={n}): {score:.6f}")

    # Phase 2: Focus on top candidates with more time
    print(f"\n=== Phase 2: Deep optimization ===")
    top_k = min(5, len(screened))
    remaining = deadline - time.monotonic()
    time_per_top = max(30.0, remaining / (top_k + 2) * 0.7)

    for i in range(top_k):
        if time.monotonic() > deadline - 20:
            break
        name, f0, f_opt, score, n = screened[i]
        t_here = min(time_per_top, deadline - time.monotonic() - 15)
        if t_here < 15:
            break

        print(f"\n  [{name}] n={n}, current={score:.6f}, time={t_here:.0f}s")

        # LP phase
        t_lp = t_here * 0.45
        f_lp, s_lp = lp_optimize(f_opt, time_budget=t_lp, verbose=True)

        # Gradient phase
        t_grad = min(t_here * 0.3, deadline - time.monotonic() - 10)
        if t_grad > 5 and np.isfinite(s_lp):
            f_grad, s_grad = gradient_polish(f_lp, time_budget=t_grad, lr=0.005, verbose=False)
            if s_grad < s_lp:
                f_lp = f_grad
                s_lp = s_grad
                print(f"    After gradient: {s_lp:.6f}")

        # Additional LP on polished result
        t_lp2 = deadline - time.monotonic() - 10
        if t_lp2 > 15:
            f_lp2, s_lp2 = lp_optimize(f_lp, time_budget=min(t_lp2, t_here * 0.25), verbose=False)
            if s_lp2 < s_lp:
                f_lp = f_lp2
                s_lp = s_lp2
                print(f"    After more LP: {s_lp:.6f}")

        # Symmetrization check
        f_sym = symmetrize(f_lp)
        f_sym = np.clip(f_sym, 0, None)
        if f_sym.sum() > 1e-10:
            f_sym = f_sym / f_sym.sum()
            s_sym = compute_score(f_sym)
            if s_sym < s_lp:
                f_lp = f_sym
                s_lp = s_sym
                print(f"    After symmetrization: {s_lp:.6f}")

        if np.isfinite(s_lp) and s_lp < best_score_overall:
            best_score_overall = s_lp
            best_overall = f_lp.copy()
            print(f"    ** New overall best: {s_lp:.6f} **")

    # Phase 3: Final intensive optimization on the best
    remaining = deadline - time.monotonic()
    if best_overall is not None and remaining > 30:
        print(f"\n=== Phase 3: Final intensive optimization (time={remaining:.0f}s) ===")
        n_best = len(best_overall)

        # Gradient with smaller lr
        t1 = min(remaining * 0.3, deadline - time.monotonic() - 10)
        f_g, s_g = gradient_polish(best_overall, time_budget=t1, lr=0.002, verbose=True)
        if s_g < best_score_overall:
            best_overall = f_g
            best_score_overall = s_g
            print(f"  After final gradient: {s_g:.6f}")

        # More LP
        t2 = min(remaining * 0.4, deadline - time.monotonic() - 10)
        if t2 > 15:
            f_l, s_l = lp_optimize(best_overall, time_budget=t2, verbose=True)
            if s_l < best_score_overall:
                best_overall = f_l
                best_score_overall = s_l
                print(f"  After final LP: {s_l:.6f}")

        # Final gradient
        t3 = deadline - time.monotonic() - 5
        if t3 > 10:
            f_g2, s_g2 = gradient_polish(best_overall, time_budget=t3, lr=0.001, verbose=False)
            if s_g2 < best_score_overall:
                best_overall = f_g2
                best_score_overall = s_g2
                print(f"  After last gradient: {s_g2:.6f}")

    # Phase 4: Try a few random restarts if time allows
    remaining = deadline - time.monotonic()
    if best_overall is not None and remaining > 40:
        print(f"\n=== Phase 4: Random restarts (time={remaining:.0f}s) ===")
        n_best = len(best_overall)
        n_restarts = min(4, int(remaining / 20))
        for r in range(n_restarts):
            if time.monotonic() > deadline - 10:
                break
            f0 = best_overall * (1 + 0.3 * rng.normal(0, 1, n_best))
            f0 = np.clip(f0, 0, None)
            if f0.sum() < 1e-10:
                continue
            f0 = f0 / f0.sum()
            t_here = deadline - time.monotonic() - 10
            if t_here < 15:
                break
            f_r, s_r = lp_optimize(f0, time_budget=t_here * 0.5, verbose=False)
            if np.isfinite(s_r):
                t_g = deadline - time.monotonic() - 10
                if t_g > 5:
                    f_r, s_r = gradient_polish(f_r, time_budget=t_g, lr=0.003, verbose=False)
            if np.isfinite(s_r) and s_r < best_score_overall:
                best_overall = f_r
                best_score_overall = s_r
                print(f"  Restart {r}: new best {s_r:.6f}")

    # Final verification
    if best_overall is not None:
        best_overall = np.clip(best_overall, 0, 1000)
        final_list = best_overall.tolist()
        final_score = evaluate_sequence(final_list)
        print(f"\n{'='*50}")
        print(f"FINAL VERIFIED SCORE: {final_score:.6f}")
        print(f"Sequence length: {len(final_list)}")
        print(f"Target: 1.5027")
        print(f"{'='*50}")
        return final_list

    # Fallback
    print("Falling back to uniform sequence")
    return [1.0] * 256
```
