    """Compute softmax weights for log-sum-exp smoothing of max."""
    max_c = np.max(conv_a)
    shifted = (conv_a - max_c) / tau
    shifted = np.clip(shifted, -50.0, 50.0)
    exp_c = np.exp(shifted)
    s = np.sum(exp_c)
    if s < 1e-300:
        return np.full(len(conv_a), 1.0 / len(conv_a))
    return exp_c / s


def smoothed_grad_descent(f_init, deadline, rng, max_iters=200000):
    """Main optimization: smoothed subgradient with annealing + backtracking + perturbation."""
    f = f_init.copy().astype(np.float64)
    n = len(f)
    S = np.sum(f)
    if S <= 0:
        f = np.ones(n)
        S = n
    f = f / S

    best_f = f.copy()
    best_quick = 2.0 * n * np.max(np.convolve(f, f))
    best_actual = evaluate_sequence(f.tolist())

    conv_f = np.convolve(f, f)
    max_c = np.max(conv_f)
    tau = max_c / 4.0
    tau_min = max_c / 200.0
    lr = 0.02
    no_improve = 0
    it = 0

    while time.monotonic() < deadline - 3 and it < max_iters:
        conv_f = np.convolve(f, f)
        max_c = np.max(conv_f)
        w = smooth_max_weights(conv_f, tau)
        grad = compute_gradient_fast(f, w)

        grad_norm = np.linalg.norm(grad)
        if grad_norm < 1e-15:
            break

        # Backtracking line search
        f_accepted = f.copy()
        score_accepted = best_quick
        step = lr
        improved = False
        for _bt in range(10):
            new_f = f - step * grad
            new_f = np.maximum(new_f, 0.0)
            new_sum = np.sum(new_f)
            if new_sum < 1e-12:
                step *= 0.4
                continue
            new_f = new_f / new_sum
            new_conv = np.convolve(new_f, new_f)
            new_quick = 2.0 * n * np.max(new_conv)
            if new_quick < score_accepted - 1e-15:
                f_accepted = new_f
                score_accepted = new_quick
                improved = True
                break
            step *= 0.4

        f = f_accepted
        it += 1

        if improved and score_accepted < best_quick:
            best_quick = score_accepted
            best_f = f.copy()
            no_improve = 0
        else:
            no_improve += 1

        # Perturbation when stuck
        if no_improve > 150 and it % 50 == 0:
            noise_scale = max_c * 0.3 / n
            f = f + noise_scale * rng.standard_normal(n)
            f = np.maximum(f, 0.0)
            fs = np.sum(f)
            if fs > 0:
                f = f / fs
            tau = np.max(np.convolve(f, f)) / 4.0
            lr = 0.02
            no_improve = 0

        # Anneal temperature
        tau *= 0.999
        if tau < tau_min:
            tau = tau_min

        # Adapt learning rate
        if no_improve > 80:
            lr *= 0.7
            lr = max(lr, 1e-8)
        elif no_improve == 0 and it > 100:
            lr = min(lr * 1.1, 0.1)

        # Periodic actual evaluation
        if it % 500 == 0:
            actual = evaluate_sequence(f.tolist())
            if np.isfinite(actual) and actual < best_actual:
                best_actual = actual
                best_f = f.copy()
            if it % 2000 == 0:
                print(f"      SGD it={it}: quick={best_quick:.6f}, actual={best_actual:.6f}, "
                      f"tau={tau:.2e}, lr={lr:.2e}, no_imp={no_improve}")

    return best_f, best_actual


def lp_direction(f, M):
    """Solve LP: max sum(g) s.t. conv(f,g)[k] <= M for all k, g >= 0."""
    n = len(f)
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        j_min = max(0, k - n + 1)
        j_max = min(k, n - 1)
        for j in range(j_min, j_max + 1):
            rows.append(k)
            cols.append(j)
            vals.append(f[k - j])
    A = csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))
    c_obj = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    try:
        result = linprog(c_obj, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                         options={'presolve': True, 'time_limit': 30})
        if result.success:
            return result.x
    except Exception:
        pass
    return None


def lp_refine(f_init, deadline, max_iters=100):
    """LP-based direction finding with line search."""
    f = f_init.copy().astype(np.float64)
    n = len(f)
    S = np.sum(f)
    if S <= 0:
        return f, evaluate_sequence(f.tolist())
    f = f / S

    best_f = f.copy()
    best_actual = evaluate_sequence(f.tolist())
    best_quick = 2.0 * n * np.max(np.convolve(f, f))

    step_size = 0.15
    no_improve = 0
    it = 0

    while time.monotonic() < deadline - 5 and it < max_iters:
        conv_f = np.convolve(f, f)
        M = np.max(conv_f)
        if M <= 0:
            break

        g = lp_direction(f, M)
        if g is None:
            no_improve += 1
            if no_improve > 30:
                break
            continue

        sum_g = np.sum(g)
        if sum_g < 1e-15:
            no_improve += 1
            continue

        g = g / sum_g  # normalize g to sum=1

        best_bt_f = f.copy()
        best_bt_score = best_quick
        for t_try in [step_size, step_size * 0.5, step_size * 0.25, step_size * 2.0, step_size * 0.1]:
            new_f = (1.0 - t_try) * f + t_try * g
            new_f = np.maximum(new_f, 0.0)
            ns = np.sum(new_f)
            if ns < 1e-12:
                continue
            new_f = new_f / ns
            new_quick = 2.0 * n * np.max(np.convolve(new_f, new_f))
            if new_quick < best_bt_score:
                best_bt_score = new_quick
                best_bt_f = new_f.copy()

        f = best_bt_f
        it += 1

        if best_bt_score < best_quick - 1e-10:
            best_quick = best_bt_score
            best_f = f.copy()
            no_improve = 0
        else:
            no_improve += 1
            step_size *= 0.6

        if it % 3 == 0:
            actual = evaluate_sequence(f.tolist())
            if np.isfinite(actual) and actual < best_actual:
                best_actual = actual
                best_f = f.copy()
            print(f"      LP it={it}: quick={best_quick:.6f}, actual={best_actual:.6f}, step={step_size:.4f}")

        if no_improve > 40:
            break

    return best_f, best_actual


def upsample_sequence(f, new_n):
    """Upsample sequence to new length using linear interpolation."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_idx = np.arange(old_n, dtype=np.float64)
    new_idx = np.linspace(0, old_n - 1, new_n)
    return np.interp(new_idx, old_idx, f)


def propose_candidate(budget_s=1500, **kwargs):
    t0 = time.monotonic()
    deadline = t0 + max(0.0, float(budget_s) - 5.0)

    best_sequence = None
    best_score = float('inf')
    rng = np.random.default_rng(2024)

    # Load parent
    parent = None
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64).copy()
        parent = np.clip(parent, 0.0, 1000.0)
        if parent.ndim == 1 and parent.size > 0 and np.all(np.isfinite(parent)):
            score = evaluate_sequence(parent.tolist())
            if np.isfinite(score):
                best_score = score
                best_sequence = parent.copy()
                print(f"Parent n={parent.size}: score={score:.6f}")
    except Exception as e:
        print(f"Could not load parent: {e}")

    # Define initializations for each length
    def get_initializations(n, seed_parent):
        inits = []
        # Uniform
        inits.append(('uniform', np.ones(n)))
        # Hann window
        idx = np.arange(n, dtype=np.float64)
        hann = 0.5 * (1.0 + np.cos(2.0 * np.pi * idx / (n - 1)))
        inits.append(('hann', hann))
        # Cosine (1+cos)/2
        cos_w = (1.0 + np.cos(np.pi * idx / (n - 1))) / 2.0
        inits.append(('cosine', cos_w))
        # Two-peak
        center1 = n * 0.25
        center2 = n * 0.75
        sigma = n / 8.0
        two_peak = (np.exp(-0.5 * ((idx - center1) / sigma) ** 2) +
                    np.exp(-0.5 * ((idx - center2) / sigma) ** 2))
        inits.append(('two_peak', two_peak))
        # Random smooth
        r_raw = rng.uniform(0, 1, size=n // 8 + 2)
        r_smooth = np.interp(np.linspace(0, len(r_raw) - 1, n), np.arange(len(r_raw)), r_raw)
        inits.append(('rand_smooth', 0.5 + r_smooth))
        # Modulated uniform: 1 + r*cos
        for freq in [1, 2, 3]:
            mod = 1.0 + 0.3 * np.cos(2.0 * np.pi * freq * idx / n)
            inits.append((f'mod_{freq}', np.maximum(mod, 0.01)))
        # Parent upsampled
        if seed_parent is not None and len(seed_parent) != n:
            up = upsample_sequence(seed_parent, n)
            inits.append(('parent_up', up))
        elif seed_parent is not None and len(seed_parent) == n:
            inits.append(('parent', seed_parent.copy()))
        return inits

    # Phase 1: Quick search across lengths and initializations
    print("\n=== Phase 1: Quick search ===")
    phase1_deadline = min(deadline, t0 + 300)
    lengths = [512, 768, 1024, 1536]
    seed = parent if parent is not None else best_sequence

    for n in lengths:
        if time.monotonic() > phase1_deadline - 30:
            break
        print(f"\n  --- n={n} ---")
        inits = get_initializations(n, seed)
        for label, f_init in inits:
            if time.monotonic() > phase1_deadline - 10:
                break
            f_init = np.clip(f_init.astype(np.float64), 0.0, 1000.0)
            if np.sum(f_init) < 1e-6:
                continue
            score = evaluate_sequence(f_init.tolist())
            if not np.isfinite(score):
                continue
            if score < best_score:
                best_score = score
                best_sequence = f_init.copy()
            # Quick gradient descent
            sub_dl = min(phase1_deadline - 5, time.monotonic() + 40)
            f_best, s_best = smoothed_grad_descent(f_init, sub_dl, rng, max_iters=50000)
            if s_best < best_score:
                best_score = s_best
                best_sequence = f_best.copy()
                print(f"    {label} -> improved to {best_score:.6f}")
            else:
                print(f"    {label}: init={score:.4f}, after SGD={s_best:.4f}")

    print(f"\n  Phase 1 best: n={len(best_sequence) if best_sequence is not None else 0}, "
          f"score={best_score:.6f}")

    # Phase 2: Extended optimization on the best found with LP injection
    print("\n=== Phase 2: Extended optimization ===")
    phase2_deadline = min(deadline, t0 + 1200)
    if best_sequence is not None:
        f = best_sequence.copy()
        n = len(f)
        S = np.sum(f)
        f_normalized = f / S if S > 0 else f

        # Extended smoothed gradient descent with LP injection every 2000 iters
        deadline_phase2 = phase2_deadline
        f_best, s_best = smoothed_grad_descent(f, deadline_phase2, rng, max_iters=500000)
        if s_best < best_score:
            best_score = s_best
            best_sequence = f_best.copy()
            print(f"  After extended SGD: {best_score:.6f}")

        # Try a perturbation + restart
        if time.monotonic() < phase2_deadline - 60:
            f_pert = best_sequence.copy()
            S_p = np.sum(f_pert)
            f_pert = f_pert / S_p
            # Add structured perturbation
            idx = np.arange(n, dtype=np.float64)
            for freq in [2, 5, 8]:
                pert = f_pert + 0.1 * np.cos(2.0 * np.pi * freq * idx / n)
                pert = np.maximum(pert, 0.0)
                ps = np.sum(pert)
                if ps > 0:
                    pert = pert / ps
                    sub_dl = min(phase2_deadline - 30, time.monotonic() + 60)
                    f_p, s_p = smoothed_grad_descent(pert, sub_dl, rng, max_iters=30000)
                    if s_p < best_score:
                        best_score = s_p
                        best_sequence = f_p.copy()
                        print(f"  Perturbation freq={freq}: improved to {best_score:.6f}")

    # Phase 3: LP refinement
    print("\n=== Phase 3: LP refinement ===")
    phase3_deadline = deadline
    if best_sequence is not None and time.monotonic() < phase3_deadline - 10:
        f_lp, s_lp = lp_refine(best_sequence, phase3_deadline, max_iters=80)
        if s_lp < best_score:
            best_score = s_lp
            best_sequence = f_lp.copy()
            print(f"  After LP: {best_score:.6f}")

        # One final smooth gradient pass
        if time.monotonic() < phase3_deadline - 5:
            f_fg, s_fg = smoothed_grad_descent(best_sequence, phase3_deadline, rng, max_iters=50000)
            if s_fg < best_score:
                best_score = s_fg
                best_sequence = f_fg.copy()
                print(f"  Final SGD: {best_score:.6f}")

    # Ensure output is valid
    if best_sequence is None:
        best_sequence = np.ones(128)
        best_score = evaluate_sequence(best_sequence.tolist())

    best_sequence = np.clip(best_sequence, 0.0, 1000.0)
    final_score = evaluate_sequence(best_sequence.tolist())
    if np.isfinite(final_score) and final_score < best_score:
        best_score = final_score

    elapsed = time.monotonic() - t0
    print(f"\n=== FINAL: n={len(best_sequence)}, score={best_score:.6f}, time={elapsed:.1f}s ===")
    return [float(x) for x in best_sequence]
