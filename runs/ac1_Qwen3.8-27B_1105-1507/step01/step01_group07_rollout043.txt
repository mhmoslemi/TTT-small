Hybrid optimization for minimizing 2*n*max(conv(a,a))/(sum(a))^2.
Combines LP alternation, subgradient methods, and sedimentary construction.
Uses FFT for fast convolution on large sequences.
"""

import numpy as np
from scipy.optimize import linprog, minimize
import time

def fast_conv(a):
    n = len(a)
    if n < 256:
        return np.convolve(a, a, mode='full')
    size = 1
    while size < 2 * n - 1:
        size *= 2
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa)[:2*n-1]

def compute_obj(a):
    n = len(a)
    conv = fast_conv(a)
    max_b = float(np.max(conv))
    sum_a = float(np.sum(a))
    if sum_a < 0.01:
        return np.inf
    return 2.0 * n * max_b / (sum_a * sum_a)

def simplex_project(v):
    """Project v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.where(u * np.arange(1, n+1) > (cssv - 1))[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0)

def build_conv_matrix(a):
    """Build matrix A such that A @ g = conv(a, g)."""
    n = len(a)
    A = np.zeros((2*n - 1, n))
    for k in range(2*n - 1):
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        A[k, j_start:j_end+1] = a[k - j_end:k - j_start + 1][::-1]
    return A

def lp_step(a):
    """LP: max sum(g) s.t. conv(a,g) <= max(conv(a,a)), g >= 0."""
    n = len(a)
    conv_aa = fast_conv(a)
    M = float(np.max(conv_aa))
    A = build_conv_matrix(a)
    c = -np.ones(n)
    b_ub = np.full(2*n - 1, M)
    bounds = [(0.0, None)] * n
    res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
    if res.status == 0:
        return res.x, float(-res.fun), M
    return None, None, M

def run_lp_optimize(a_init, max_iter=500, t=0.5, verbose=False):
    """LP alternation optimization."""
    n = len(a_init)
    a = a_init.copy()
    s = np.sum(a)
    if s < 1e-10:
        a = np.ones(n) / n
        s = 1.0
    a = a / s
    best_a = a.copy()
    best_obj = compute_obj(best_a)
    
    stall = 0
    for it in range(max_iter):
        g, g_sum, M = lp_step(a)
        if g is None:
            break
        if g_sum <= 1.0 + 1e-5:
            stall += 1
            if stall > 10:
                break
            continue
        stall = 0
        g = g / g_sum
        a_new = (1.0 - t) * a + t * g
        a_new = np.clip(a_new, 0, None)
        s = np.sum(a_new)
        if s < 1e-10:

        a_new = a_new / s
        obj_new = compute_obj(a_new)
        if obj_new < best_obj:
            best_a = a_new.copy()
            best_obj = obj_new
            if verbose and (it % 20 == 0 or it < 5):

        a = a_new
    return best_a, best_obj

def subgradient_step(a, step_size, active_set_size=5):
    """One subgradient step to reduce the peak convolution."""
    n = len(a)
    conv = fast_conv(a)
    # Find active set (top positions)
    top_k = np.argsort(conv)[::-1][:active_set_size]
    max_val = conv[top_k[0]]
    # Average gradient over active set
    g = np.zeros(n)
    for k in top_k:
        # d(conv_k)/da[i] = 2*a[k-i] if 0 <= k-i < n
        valid = np.arange(n)
        diff = k - valid
        mask = (diff >= 0) & (diff < n)
        g[mask] += 2 * a[diff[mask]]
    g /= len(top_k)
    # Update
    a_new = a - step_size * g
    a_new = simplex_project(a_new)
    return a_new

        a[i] = frac

    for i in range(edge_len):
        frac = (i + 1) / (edge_len + 1)
        a[n - 1 - i] = frac
    s = np.sum(a)
    return a / s

"""
Hybrid optimization combining LP alternation, smoothed-objective gradient
descent, sedimentary initialization, and random block refinement to minimize
2*n*max(conv(a,a))/(sum(a))^2 for non-negative sequences.
"""

import numpy as np
from scipy.optimize import linprog, minimize
import time

# --- Fast convolution ---
def fast_conv(a):
    n = len(a)
    if n < 512:
        return np.convolve(a, a, mode='full')
    size = 1
    while size < 2 * n - 1:
        size *= 2
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa)[:2*n-1]

# --- Objective ---
def compute_obj(a):
    n = len(a)
    conv = fast_conv(a)
    max_b = float(np.max(conv))
    sum_a = float(np.sum(a))
    if sum_a < 0.01:
        return np.inf
    return 2.0 * n * max_b / (sum_a * sum_a)

# --- Simplex projection ---
def simplex_project(v):

Hybrid optimization for minimizing 2*n*max(conv(a,a))/(sum(a))^2.
Combines LP alternation, smoothed-objective gradient descent (log-sum-exp),
sedimentary initialization with structured edge profiles, flat-convolution

import numpy as np
from scipy.optimize import linprog
import time


def fast_conv(a):
    """Compute a*a convolution using FFT for large n."""
    n = len(a)
    if n < 512:
        return np.convolve(a, a, mode='full')
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa)[:2 * n - 1]


def compute_obj(a):
    """Compute 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    conv = fast_conv(a)
    max_b = float(np.max(conv))
    sum_a = float(np.sum(a))
    if sum_a < 0.01:
        return np.inf
    return 2.0 * n * max_b / (sum_a * sum_a)


def simplex_project(v):
    """Project v onto {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho_candidates = np.where(u * np.arange(1, n + 1) > (cssv - 1))[0]
    if len(rho_candidates) == 0:
        return np.ones(n) / n
    rho = rho_candidates[-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0)


def build_conv_matrix(a):
    """Build matrix A such that A @ g = conv(a, g)."""
    n = len(a)
    A = np.zeros((2 * n - 1, n), dtype=np.float64)
    for k in range(2 * n - 1):
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        A[k, j_start:j_end + 1] = a[k - j_end:k - j_start + 1][::-1]
    return A


def lp_step(a):

def smooth_obj_grad(a, T):
    """Compute smoothed objective and gradient using log-sum-exp.
    Assumes sum(a) = 1. Returns (obj, grad)."""
    n = len(a)
    conv = fast_conv(a)
    conv = np.clip(conv, 0, None)
    max_c = float(np.max(conv))
    shifted = T * (conv - max_c)
    exp_shifted = np.exp(shifted)
    sum_exp = float(np.sum(exp_shifted))
    if sum_exp < 1e-300:
        w = np.ones(len(conv)) / len(conv)
    else:
        w = exp_shifted / sum_exp
    logsumexp = max_c + np.log(sum_exp) / T
    obj = 2.0 * n * logsumexp
    # Gradient: (4n/T) * (w cross-correlate a)
    # (w ⊛ a)[i] = sum_j w[j+i] * a[j]
    # = np.correlate(w, a, 'full')[n-1+i]
    corr = np.correlate(w, a, mode='full')

def flat_conv_init(n):
    """Compute the 'ideal' flat-convolution sequence via recurrence.
    May not be perfectly flat, but provides a good initialization."""
    if n < 2:
        return np.ones(n) / n
    c = 1.0 / (2 * n - 1)
    a = np.zeros(n)
    a[0] = np.sqrt(c)
    for k in range(1, n):
        # conv[k] = 2*a[0]*a[k] + sum_{j=1}^{k-1} a[j]*a[k-j] = c
        cross = 0.0
        for j in range(1, k):
            if j < n and (k - j) < n:
                cross += a[j] * a[k - j]
        a[k] = (c - cross) / (2.0 * a[0])
    # Enforce non-negativity
    a = np.maximum(a, 0.0)
    s = np.sum(a)
    if s < 1e-10:
        return np.ones(n) / n
    return a / s


def sedimentary_init(n, edge_len, rng):
    """Generate sedimentary initial sequence with structured edges."""
    a = np.ones(n)

        a[n - 1 - i] = 0.2 + 0.8 * frac
    s = np.sum(a)
    if s < 1e-10:

    return a / s


def raised_cosine_init(n, rng):
    """Raised cosine profile."""
    t = np.linspace(0, 1, n)
    a = 0.5 * (1 + np.cos(2 * np.pi * t))

    s = np.sum(a)

def linear_taper_init(n, rng, asymmetry=0.3):
    """Linear taper with asymmetry."""
    t = np.linspace(0, 1, n)

    a = np.maximum(a, 0.01)

    s = np.sum(a)

    if best_seq is not None and time.monotonic() < deadline - 10:
        remaining = deadline - time.monotonic()
        n_best = len(best_seq)

        # Random block refinement (longer)
        rls_iters = int(remaining * 4)
        a, obj = random_block_refine(best_seq, rng, max_iter=rls_iters)
        update_best(a, "Final RBL")

        # Single point refinement
        if time.monotonic() < deadline - 10:
            sp_iters = int((deadline - time.monotonic()) * 2)
            a, obj = single_point_refine(best_seq, rng, max_iter=sp_iters)
            update_best(a, "Final SP")

        # One more LP round on the refined best
        if time.monotonic() < deadline - 30 and n_best <= 500:
            max_iter = int((deadline - time.monotonic()) / 1.5)
            a, obj = run_lp_optimize(best_seq, max_iter=max(10, max_iter), t=0.3)
            update_best(a, "Final LP")

            # Follow with refinement
            if time.monotonic() < deadline - 10:
                rls_iters = int((deadline - time.monotonic()) * 4)
                a, obj = random_block_refine(a, rng, max_iter=rls_iters)
                update_best(a, "Final RBL2")

    # Fallback
    if best_seq is None:
        best_seq = np.ones(100)
        best_score = compute_obj(best_seq)

    # Final verification
    final_score = compute_obj(best_seq)
    if final_score != best_score:
        best_score = final_score

    elapsed = time.monotonic() - start_time
    print(f"DONE: n={len(best_seq)}, score={best_score:.6f}, time={elapsed:.1f}s")
    return [float(x) for x in best_seq]
