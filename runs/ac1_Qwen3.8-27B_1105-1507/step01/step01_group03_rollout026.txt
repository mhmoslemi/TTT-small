**Better polishing: "Peak shaving"**

1. Compute conv = f*f
2. Find the peak position k*
3. The peak is: conv[k*] = sum_i f_i * f[k*-i]
4. To reduce the peak, we should reduce the contributions from the largest terms
5. For each pair (i, k*-i), the contribution is f_i * f[k*-i]
6. We can reduce this by reducing either f_i or f[k*-i]
7. The freed mass should go to positions that contribute least to the peak

Algorithm:
1. Compute conv, find peak k*
2. Compute contributions: c_i = f_i * f[k*-i] (for valid i)
3. Sort by contribution (descending)
4. For the top contributor i:
   - Reduce f_i by some amount delta
   - The peak reduces by approximately delta * f[k*-i]
   - Add delta to a position j that minimizes the increase in conv
   - The increase in conv[k] from adding delta to f_j is delta * f[k-j]
   - We want to minimize the maximum increase over all k
   - A simple heuristic: add to j where f[j] is already large (since it won't create new peaks)
     Actually, we want to add where it creates the least new peak. This is complex.
   - Alternative: add uniformly to all positions (or to positions with f=0)
5. Repeat

This is getting complex. Let me just use a simpler approach:

**Polishing: Random perturbation + accept if better**

1. Start with current best f
2. For many iterations:
   a. Pick a random subset of coordinates (e.g., 10% of them)
   b. Perturb them by a small random amount
   c. Normalize
   d. If score improved, keep; else revert
3. Gradually decrease the perturbation size (annealing)

"""
Multi-resolution LP optimization with golden-section line search and
stochastic polishing for minimizing 2*n*max(conv(f,f))/(sum(f))^2.
Uses FFT for fast convolution, golden-section search for step sizes,
and a simulated-annealing-style polish phase for final improvements.
"""

import time
import numpy as np
from scipy.optimize import linprog
from scipy.sparse import csr_matrix


def normalize_sequence(arr):
    """Return a nonnegative floating vector normalized to sum 1."""
    x = np.asarray(arr, dtype=np.float64).ravel()
    if x.size == 0:
        return None
    x = np.where(np.isfinite(x), x, 0.0)
    x = np.maximum(x, 0.0)
    s = float(x.sum())
    if s <= 1e-300:
        return None
    return x / s


def score_normalized(x, n):
    """Score assuming x is nonnegative and sum(x)=1: 2*n*max(conv(x,x))."""
    conv = np.convolve(x, x)
    m = float(np.max(conv))
    if not np.isfinite(m):
        return float("inf")
    return float(2.0 * n * m)


def conv_fft(f):
    """Linear convolution via FFT (faster for large n)."""
    n = len(f)
    size = 1
    target = 2 * n - 1
    while size < target:
        size *= 2
    F = np.fft.rfft(f, size)
    result = np.fft.irfft(F * F, size)
    return result[:target]


def score_normalized_fft(x, n):
    """Score using FFT convolution."""
    conv = conv_fft(x)
    m = float(np.max(conv))
    if not np.isfinite(m):
        return float("inf")
    return float(2.0 * n * m)


def interp_to_length(x, n):
    """Resample a nonnegative sequence to length n and normalize."""
    x = normalize_sequence(x)
    if x is None or n <= 0:
        return None
    m = x.size
    if m == n:
        return x.copy()
    if m == 1:
        return np.ones(n) / n
    if n == 1:
        return np.array([1.0])
    old = np.linspace(0.0, 1.0, m)
    new = np.linspace(0.0, 1.0, n)
    out = np.interp(new, old, x)
    return normalize_sequence(out)


def build_conv_indices(n):
    """Build row/col index arrays for the convolution matrix structure."""
    m = 2 * n - 1
    ks = np.arange(m, dtype=np.int64)
    js = np.maximum(0, ks - n + 1)
    je = np.minimum(n - 1, ks)
    counts = je - js + 1
    rows = np.repeat(ks, counts)
    total = int(counts.sum())
    cum = np.cumsum(counts)
    starts = cum - counts
    local = np.arange(total, dtype=np.int64) - np.repeat(starts, counts)
    cols = np.repeat(js, counts) + local
    return rows, cols


def solve_lp_direction(f, n, rows, cols, M):
    """
    Solve: max sum(g)
           s.t. (f*g)_k <= M for all k
                g >= 0
    Returns g (LP solution) or None.
    """
    m = 2 * n - 1
    vals = f[rows - cols]
    A = csr_matrix((vals, (rows, cols)), shape=(m, n))
    c = -np.ones(n)
    b = np.full(m, M, dtype=np.float64)
    bounds = [(0, None)] * n
    try:
        res = linprog(c, A_ub=A, b_ub=b, bounds=bounds, method='highs')
    except Exception:
        return None
    if res.status != 0 or res.x is None:
        return None
    x = np.maximum(res.x, 0.0)
    s = x.sum()
    if s < 1e-300:
        return None
    return x


def golden_section_min(func, a, b, tol=1e-10, max_iter=60):
    """Minimize func on [a,b] using golden section search."""
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    c = b - gr * (b - a)
    d = a + gr * (b - a)
    fc = func(c)
    fd = func(d)
    for _ in range(max_iter):
        if b - a < tol:
            break
        if fc < fd:
            b = d
            d = c
            fd = fc
            c = b - gr * (b - a)
            fc = func(c)
        else:
            a = c
            c = d
            fc = fd
            d = a + gr * (b - a)
            fd = func(d)
    x_opt = (a + b) / 2.0
    return x_opt, min(func(x_opt), fc, fd)


def line_search(f, g, n, current_score):
    """
    Find the best step size s in [0, 1] along direction g from f.
    f_new(s) = normalize((1-s)*f + s*g)
    Returns (best_s, best_f, best_score).
    """
    def objective(s):
        f_new = (1.0 - s) * f + s * g
        f_new = np.maximum(f_new, 0.0)
        ss = f_new.sum()
        if ss < 1e-300:
            return float("inf")
        f_new = f_new / ss
        return score_normalized_fft(f_new, n)

    # Quick check: is s=1 better?
    val_0 = current_score
    val_1 = objective(1.0)
    val_half = objective(0.5)

    if min(val_0, val_1, val_half) == val_0 and val_half > val_0 and val_1 > val_0:
        return 0.0, f.copy(), val_0

    # Golden section on [0, 1]
    s_opt, sc_opt = golden_section_min(objective, 0.0, 1.0, tol=1e-12, max_iter=50)

    f_best = (1.0 - s_opt) * f + s_opt * g
    f_best = np.maximum(f_best, 0.0)
    ss = f_best.sum()
    if ss < 1e-300:
        return 0.0, f.copy(), val_0
    f_best = f_best / ss

    return s_opt, f_best, sc_opt


def lp_optimize(n, f_init, time_budget, rng, use_fft=True):
    """
    Run LP-based iteration from f_init for time_budget seconds.
    Returns (best_f, best_score).
    """
    f = normalize_sequence(f_init)
    if f is None:
        f = np.ones(n) / n

    rows, cols = build_conv_indices(n)

    def do_score(x):
        if use_fft and n >= 256:
            return score_normalized_fft(x, n)
        return score_normalized(x, n)

    best_f = f.copy()
    best_score = do_score(f)
    start_score = best_score
    deadline = time.monotonic() + max(0.1, float(time_budget))
    stall = 0
    iteration = 0

    while time.monotonic() < deadline:
        iteration += 1
        t0 = time.monotonic()

        # Compute current peak
        conv_ff = conv_fft(f) if (use_fft and n >= 256) else np.convolve(f, f)
        M = float(np.max(conv_ff))
        if not np.isfinite(M) or M <= 0:
            break

        # Solve LP
        g_raw = solve_lp_direction(f, n, rows, cols, M)
        if g_raw is None:
            stall += 1
            if stall > 15:
                break
            # Perturb and retry
            noise = rng.random(n) * 0.1
            f = f + noise
            f = normalize_sequence(f)
            continue

        s_star = float(g_raw.sum())
        if s_star <= 1.0 + 1e-10:
            stall += 1
            if stall > 15:
                break
            noise = rng.random(n) * 0.1
            f = f + noise
            f = normalize_sequence(f)
            continue

        g = g_raw / s_star  # Normalize g to sum 1

        # Check if g is truly a descent direction
        fg = conv_fft(f * 0 + g)  # dummy, will compute properly
        # Actually compute f*g
        fg = conv_fft(f) if False else None
        # Just use the step: if s_star > 1, the LP found a direction
        # that can carry more mass while staying under M.
        # The normalized g should reduce the peak when mixed with f.

        # Line search
        s_opt, f_new, sc_new = line_search(f, g, n, best_score)

        if np.isfinite(sc_new) and sc_new < best_score - 1e-14:
            f = f_new
            best_f = f.copy()
            best_score = sc_new
            stall = 0
            if iteration <= 3 or iteration % 20 == 0 or best_score < start_score - 0.05:
                print(f"      n={n} iter={iteration} score={best_score:.8f} "
                      f"s_star={s_star:.4f} s_opt={s_opt:.4f} "
                      f"dt={time.monotonic()-t0:.2f}s", flush=True)
        else:
            stall += 1
            if stall % 5 == 0:
                print(f"      n={n} iter={iteration} stall={stall} best={best_score:.8f} "
                      f"s_star={s_star:.4f}", flush=True)
            if stall > 15:
                # Try a perturbation to escape
                eps = 0.1 * stall / 15.0
                noise = rng.random(n) * eps
                f = best_f + noise
                f = normalize_sequence(f)
                if f is None:
                    f = best_f.copy()
                continue

    return best_f, float(best_score)


def polish_stochastic(f, n, time_budget, rng):
    """
    Stochastic local search / simulated annealing polish.
    Perturb random subsets of coordinates and keep improvements.
    """
    f = f.astype(np.float64).copy()
    best_f = f.copy()
    best_score = score_normalized_fft(f, n)
    deadline = time.monotonic() + max(0.1, float(time_budget))

    # Annealing schedule
    start_temp = 0.05
    end_temp = 0.001
    elapsed = 0.0

    iter_count = 0
    while time.monotonic() < deadline:
        iter_count += 1
        elapsed = time.monotonic() - (deadline - time_budget)
        frac = min(1.0, elapsed / max(0.1, time_budget))
        temp = start_temp * (end_temp / start_temp) ** frac

        # Choose number of coordinates to perturb
        n_perturb = max(1, int(n * temp * 2))
        n_perturb = min(n_perturb, n)

        # Pick random coordinates
        idx = rng.choice(n, size=min(n_perturb, n), replace=False)

        # Perturb: scale selected coordinates by random factors
        f_new = f.copy()
        for i in idx:
            factor = 1.0 + rng.uniform(-1.0, 1.0) * temp * 2
            f_new[i] = max(0.0, f[i] * factor)

        ss = f_new.sum()
        if ss < 1e-300:
            continue
        f_new = f_new / ss

        sc_new = score_normalized_fft(f_new, n)

        if sc_new < best_score:
            best_score = sc_new
            best_f = f_new.copy()
            f = f_new.copy()
        elif sc_new < best_score + temp * 0.01:
            # Accept with some probability (annealing)
            if rng.random() < 0.3:
                f = f_new.copy()

        if iter_count % 5000 == 0:
            print(f"      polish iter={iter_count} temp={temp:.5f} "
                  f"best_score={best_score:.8f}", flush=True)

    return best_f, float(best_score)


def polish_coordinate(f, n, time_budget, rng):
    """
    Coordinate-wise optimization: for each coordinate, try to find
    a better value via local search.
    """
    f = f.astype(np.float64).copy()
    best_f = f.copy()
    best_score = score_normalized_fft(f, n)
    deadline = time.monotonic() + max(0.1, float(time_budget))

    # Precompute the convolution for reuse
    while time.monotonic() < deadline:
        # Pick a batch of random coordinates
        batch_size = max(1, n // 200)
        for _ in range(batch_size):
            i = int(rng.integers(n))
            if not np.isfinite(f[i]) or f[i] < 0:
                continue

            # Try several multipliers
            fi = f[i]
            if fi < 1e-15:
                candidates = [0.0, 1e-6, 1e-4, 1e-3, 1e-2]
            else:
                candidates = [0.0, fi * 0.1, fi * 0.3, fi * 0.5,
                              fi * 0.7, fi * 1.3, fi * 1.7, fi * 3.0]

            best_local = fi
            best_local_sc = best_score

            for cv in candidates:
                f_test = f.copy()
                f_test[i] = cv
                ss = f_test.sum()
                if ss < 1e-300:
                    continue
                f_test = f_test / ss
                sc = score_normalized_fft(f_test, n)
                if sc < best_local_sc - 1e-13:
                    best_local_sc = sc
                    best_local = cv

            if abs(best_local - fi) > 1e-15:
                f[i] = best_local
                ss = f.sum()
                if ss > 1e-300:
                    f = f / ss
                    sc_check = score_normalized_fft(f, n)
                    if sc_check < best_score:
                        best_score = sc_check
                        best_f = f.copy()

        if (time.monotonic() - (deadline - time_budget)) % 10 < 0.1:
            print(f"      coord polish: best_score={best_score:.8f}", flush=True)

    return best_f, float(best_score)
