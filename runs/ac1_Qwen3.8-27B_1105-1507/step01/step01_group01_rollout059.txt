```python
"""
Search for low-score non-negative step-function coefficients.

Objective:
    score = 2 * n * max(convolve(a, a)) / sum(a)^2

This program combines:
  - many parametric and randomized initial densities,
  - mirror/subgradient updates using active convolution peaks,
  - an LP "best response" step minimizing max(p * q) over q on the simplex,
  - nonnegative least-squares dual/primal alternating steps,
  - a final L-BFGS-B polish on a smoothed softmax parameterization.

The best sequence found within the time budget is returned.
"""

import time
import math
import numpy as np
from operator import itemgetter

try:
    from scipy.optimize import linprog
    HAVE_LINPROG = True
except Exception:
    HAVE_LINPROG = False

try:
    from scipy.optimize import nnls
    HAVE_NNLS = True
except Exception:
    HAVE_NNLS = False

try:
    from scipy.optimize import minimize
    HAVE_MINIMIZE = True
except Exception:
    HAVE_MINIMIZE = False


def _normalize(a, floor=0.0):
    """Return a nonnegative finite array with positive finite sum, scaled to sum 1."""
    try:
        arr = np.asarray(a, dtype=np.float64).ravel()
    except Exception:
        return None
    if arr.size == 0:
        return None
    arr = np.clip(arr, 0.0, 1.0e6)
    if not np.all(np.isfinite(arr)):
        return None
    s = float(np.sum(arr))
    if not np.isfinite(s) or s <= 1.0e-300:
        return None
    arr = arr / s
    if floor > 0.0:
        arr = np.maximum(arr, float(floor))
        s = float(np.sum(arr))
        if not np.isfinite(s) or s <= 1.0e-300:
            return None
        arr = arr / s
    return arr


def _score_norm(a):
    """Exact objective for nonnegative finite a (scale invariant)."""
    p = np.asarray(a, dtype=np.float64).ravel()
    if p.size == 0:
        return float("inf")
    p = np.clip(p, 0.0, 1.0e6)
    if not np.all(np.isfinite(p)):
        return float("inf")
    s = float(np.sum(p))
    if not np.isfinite(s) or s <= 1.0e-300:
        return float("inf")
    c = np.convolve(p, p)
    m = float(np.max(c))
    if not np.isfinite(m):
        return float("inf")
    return float(2.0 * p.size * m / (s * s))


def _score_from_conv(p, c):
    """Exact objective from precomputed convolution c of p."""
    p = np.asarray(p, dtype=np.float64).ravel()
    c = np.asarray(c, dtype=np.float64).ravel()
    n = p.size
    if n == 0 or c.size == 0:
        return float("inf")
    s = float(np.sum(p))
    if not np.isfinite(s) or s <= 1.0e-300:
        return float("inf")
    m = float(np.max(c))
    if not np.isfinite(m):
        return float("inf")
    return float(2.0 * n * m / (s * s))


def _resize(seq, n):
    """Resize a sequence by linear interpolation, then normalize."""
    try:
        arr = np.asarray(seq, dtype=np.float64).ravel()
    except Exception:
        return None
    if arr.size == 0:
        return None
    arr = np.clip(arr, 0.0, 1.0e6)
    if not np.all(np.isfinite(arr)):
        return None
    if arr.size == n:
        return _normalize(arr, 0.0)
    if arr.size == 1:
        return _normalize(np.full(n, float(arr.flat[0]), dtype=np.float64), 0.0)
    old = np.linspace(0.0, 1.0, arr.size, dtype=np.float64)
    new = np.linspace(0.0, 1.0, n, dtype=np.float64)
    out = np.interp(new, old, arr)
    return _normalize(out, 0.0)


def _softmax_theta(x):
    """Softmax with overflow protection; p.sum() == 1."""
    x = np.asarray(x, dtype=np.float64).ravel()
    if x.size == 0:
        return None
    x = np.clip(x, -700.0, 700.0)
    m = float(np.max(x))
    p = np.exp(x - m)
    s = float(np.sum(p))
    if not np.isfinite(s) or s <= 1.0e-300:
        return None
    p = p / s
    p = np.clip(p, 0.0, None)
    return p


def _h_from_w(p, w):
    """
    For p length n and lag weights w length 2n-1, return
    h_i = 2 * sum_r p_r w_{i+r}.

    This is the gradient (scaled) of the weighted sum of convolution coefficients
    with respect to p.
    """
    p = np.asarray(p, dtype=np.float64)
    w = np.asarray(w, dtype=np.float64)
    n = p.size
    if n == 0:
        return np.zeros(0, dtype=np.float64)
    if n == 1:
        return np.array([2.0 * p[0] * w[0]], dtype=np.float64)
    conv = np.convolve(p, w[::-1], mode="full")
    idx = np.arange(2 * n - 2, n - 2, -1, dtype=np.int64)
    return 2.0 * np.asarray(conv[idx], dtype=np.float64)


def _soft_step_with_w(p, w, eta):
    """Mirror-descent step using a weight vector w over lags."""
    p = np.asarray(p, dtype=np.float64)
    w = np.asarray(w, dtype=np.float64)
    if p.size == 0:
        return None
    h = _h_from_w(p, w)
    if h.size == 0 or not np.all(np.isfinite(h)):
        return None
    med = float(np.median(h))
    scale = max(float(np.max(h)) - med, 1.0e-15)
    fac = float(eta) * (med - h) / scale
    fac = np.clip(fac, -5.0, 5.0)
    p2 = p * np.exp(fac)
    return _normalize(p2, 1.0e-18)


def _soft_step_from_c(p, c, K, T, eta):
    """Mirror-descent step using softmax weights over the K largest lags of c."""
    p = np.asarray(p, dtype=np.float64)
    c = np.asarray(c, dtype=np.float64)
    n = p.size
    rows = c.size
    if n == 0 or rows == 0:
        return None
    K = int(K)
    if K <= 0:
        K = 1
    if K >= rows:
        part = np.arange(rows, dtype=np.int64)
    else:
        part = np.argpartition(c, -K)[-K:]
    M = float(np.max(c[part]))
    if not np.isfinite(M):
        return None
    if T <= 0.0 or not np.isfinite(T):
        wv = np.full(part.size, 1.0 / max(1, part.size), dtype=np.float64)
    else:
        u = (c[part] - M) / T
        u = np.clip(u, -700.0, 0.0)
        wv = np.exp(u)
        sw = float(np.sum(wv))
        if not np.isfinite(sw) or sw <= 0.0:
            wv = np.full(part.size, 1.0 / max(1, part.size), dtype=np.float64)
        else:
            wv = wv / sw
    w = np.zeros(rows, dtype=np.float64)
    w[part] = wv
    return _soft_step_with_w(p, w, eta)


def _best_response_lp(p, time_limit=10.0):
    """
    Best-response LP for fixed p:
        min t
        s.t. (p*q)_k <= t for all k,
             sum q = 1,
             q >= 0.
    Returns q (sum 1) or None.
    """
    if not HAVE_LINPROG:
        return None
    p = _normalize(p, 0.0)
    if p is None:
        return None
    n = p.size
    if n == 0:
        return None
    rows = 2 * n - 1
    A = np.zeros((rows, n + 1), dtype=np.float64)
    A[:, n] = -1.0
    for k in range(rows):
        s = k - (n - 1)
        if s < 0:
            s = 0
        e = k
        if e >= n:
            e = n - 1
        if s <= e:
            j = np.arange(s, e + 1, dtype=np.int64)
            A[k, j] = p[k - j]
    Aeq = np.zeros((1, n + 1), dtype=np.float64)
    Aeq[0, :n] = 1.0
    b_eq = np.ones(1, dtype=np.float64)
    cvec = np.zeros(n + 1, dtype=np.float64)
    cvec[n] = 1.0
    bounds = [(0.0, None)] * (n + 1)
    b_ub = np.zeros(rows, dtype=np.float64)

    res = None
    try:
        opts = {"presolve": True}
        if time_limit > 0:
            opts["time_limit"] = float(time_limit)
        res = linprog(
            cvec,
            A_ub=A,
            b_ub=b_ub,
            A_eq=Aeq,
            b_eq=b_eq,
            bounds=bounds,
            method="highs",
            options=opts,
        )
    except Exception:
        try:
            res = linprog(
                cvec,
                A_ub=A,
                b_ub=b_ub,
                A_eq=Aeq,
                b_eq=b_eq,
                bounds=bounds,
                method="highs",
                options={"presolve": True},
            )
        except Exception:
            return None

    if res is None or not res.success:
        return None
    x = np.asarray(res.x, dtype=np.float64)
    if x.size != n + 1 or not np.all(np.isfinite(x)):
        return None
    q = np.clip(x[:n], 0.0, None)
    s = float(np.sum(q))
    if not np.isfinite(s) or s <= 1.0e-300:
        return None
    q = q / s
    return q


def _dual_weights_from_p(p):
    """
    Given p, find nonnegative lag weights w (normalized to sum 1) such that
    L_p(w) is close to constant.  This is a dual-certificate search.
    """
    if not HAVE_NNLS:
        return None
    p = _normalize(p, 0.0)
    if p is None:
        return None
    n = p.size
    if n == 0:
        return None
    Kmat = np.zeros((n, 2 * n - 1), dtype=np.float64)
    two_p = 2.0 * p
    for i in range(n):
        Kmat[i, i:i + n] = two_p
    target = np.ones(n, dtype=np.float64)
    try:
        w, _ = nnls(Kmat, target)
    except Exception:
        return None
    w = np.asarray(w, dtype=np.float64)
    if w.size != 2 * n - 1 or not np.all(np.isfinite(w)):
        return None
    s = float(np.sum(w))
    if not np.isfinite(s) or s <= 1.0e-300:
        return None
    return w / s


def _p_from_weights(w):
    """
    Given normalized lag weights w, find nonnegative p (normalized to sum 1)
    such that L_w(p) is close to constant.
    """
    if not HAVE_NNLS:
        return None
    w = np.asarray(w, dtype=np.float64).ravel()
    if w.size % 2 == 0:
        return None
    n = (w.size + 1) // 2
    if n <= 0:
        return None
    B = np.zeros((n, n), dtype=np.float64)
    two_w = 2.0 * w
    for i in range(n):
        B[i, :] = two_w[i:i + n]
    target = np.ones(n, dtype=np.float64)
    try:
        q, _ = nnls(B, target)
    except Exception:
        return None
    q = np.asarray(q, dtype=np.float64)
    if q.size != n or not np.all(np.isfinite(q)):
        return None
    return _normalize(q, 1.0e-18)


def _theta_func(x, tau):
    """
    Smooth objective and gradient for softmax variables x:
        p = softmax(x)
        J = logsumexp_k( (p*p)_k / tau ) + 1e-9 ||x||^2
    """
    x = np.asarray(x, dtype=np.float64).ravel()
    z = np.zeros_like(x, dtype=np.float64)
    p = _softmax_theta(x)
    if p is None:
        return 1.0e100, z
    c = np.convolve(p, p)
    scaled = c / float(tau)
    umax = float(np.max(scaled))
    expu = np.exp(scaled - umax)
    Z = float(np.sum(expu))
    if not np.isfinite(Z) or Z <= 0.0:
        return 1.0e100, z
    w = expu / Z
    h = _h_from_w(p, w)
    gp = (2.0 / float(tau)) * h
    mean = float(np.dot(p, gp))
    g = p * (gp - mean)
    pen = 1.0e-9
    val = umax + math.log(Z) + pen * float(np.dot(x, x))
    g = g + 2.0 * pen * x
    if not np.all(np.isfinite(g)):
        g = np.zeros_like(x, dtype=np.float64)
    if not np.isfinite(val):
        val = 1.0e100
    return float(val), g


def _linalg_polish(p, budget):
    """Polish a sequence using L-BFGS-B on a softmax/logsumexp surrogate."""
    if not HAVE_MINIMIZE:
        return None, float("inf")
    p = _normalize(p, 0.0)
    if p is None:
        return None, float("inf")
    n = p.size
    if n == 0 or n > 1024:
        return None, float("inf")
    best = p.copy()
    best_sc = _score_norm(p)
    if budget <= 0.0:
        return best, best_sc
    deadline = time.monotonic() + float(budget)

    c = np.convolve(p, p)
    M = float(np.max(c))
    med = float(np.median(c))
    base = max(M - med, M * 1.0e-5, 1.0e-300)

    factors = (0.25, 0.10, 0.04)
    for f in factors:
        now = time.monotonic()
        if now >= deadline - 5.0:
            break
        tau = max(base * float(f), 1.0e-9)
        x0 = np.log(np.clip(p, 1.0e-300, None))
        x0 = x0 - float(np.mean(x0))
        if n <= 512:
            maxiter = 220
        elif n <= 1024:
            maxiter = 120
        else:
            maxiter = 60
        options = {
            "maxiter": maxiter,
            "gtol": 1.0e-8,
            "ftol": 1.0e-16,
            "maxls": 50,
        }
        try:
            res = minimize(
                _theta_func,
                x0,
                args=(tau,),
                jac=True,
                method="L-BFGS-B",
                options=options,
            )
        except Exception:
            res = None
        if res is not None and getattr(res, "x", None) is not None:
            p2 = _softmax_theta(res.x)
            if p2 is not None:
                sc2 = _score_norm(p2)
                if np.isfinite(sc2) and sc2 < best_sc:
                    best = p2.copy()
                    best_sc = float(sc2)
    return best, best_sc


def _add_start(starts, f, max_starts):
    """Add a finite normalized start with its score to the list."""
    if len(starts) >= max_starts:
        return
    p = _normalize(f, 0.0)
    if p is None:
        return
    sc = _score_norm(p)
    if np.isfinite(sc):
        starts.append((p, float(sc)))

    with np.errstate(divide="ignore", over="ignore", under="ignore", invalid="ignore"):
        for a, b in beta_pairs:
            if len(starts) >= max_starts:
                break
            f = np.power(x, float(a)) * np.power(1.0 - x, float(b))
            f = np.nan_to_num(f, nan=0.0, posinf=1.0e6, neginf=0.0)
            _add_start(starts, f, max_starts)

    # Linear skew.
    for m in (-2.0, -1.0, -0.5, 0.5, 1.0, 2.0):
        if len(starts) >= max_starts:
            break
        f = 1.0 + m * x
        f = np.maximum(f, 1.0e-4)
        _add_start(starts, f, max_starts)

    # Sinusoidal envelope.
    for amp in (0.4, 0.8):
        if len(starts) >= max_starts:
            break
        for phi in (0.0, 0.5 * math.pi, math.pi, 1.5 * math.pi):
            if len(starts) >= max_starts:
                break
            f = 1.0 + amp * np.sin(2.0 * math.pi * x + phi)
            f = np.maximum(f, 1.0e-4)
            _add_start(starts, f, max_starts)

    # Random smoothed log-normal starts.
    rng = np.random.default_rng(int(seed))
    guard = 0
    while len(starts) < max_starts and guard < 8 * max_starts:
        guard += 1
        k = int(rng.integers(max(1, n // 10), max(2, n // 3 + 1)))
        sigma = max(1, k // 3)
        kern = np.exp(-0.5 * (np.arange(-k, k + 1) / float(sigma)) ** 2)
        logp = np.convolve(rng.normal(0.0, 1.0, n), kern, mode="same")
        f = np.exp(0.6 * logp)
        f = np.maximum(f, 1.0e-6)
        _add_start(starts, f, max_starts)

    if len(starts) == 0:
        _add_start(starts, np.ones(n, dtype=np.float64), max_starts)
    return starts[:max_starts]


def _combined_refine(p, budget, seed):
    """
    Multi-method local refinement:
      - soft mirror descent,
      - LP best response,
      - NNLS dual/primal alternating moves.
    """
    p = _normalize(p, 1.0e-18)
    if p is None:
        return None, float("inf")

    best = p.copy()
    best_sc = _score_norm(p)
    if budget <= 0.0:
        return best, best_sc

    deadline = time.monotonic() + float(budget)
    rng = np.random.default_rng(int(seed))
    n = p.size

    eta = 0.30
    K = int(max(8, min(300, int(0.12 * n))))
    Tfrac = 0.35
    stuck = 0
    it = 0

    last_lp = 0.0
    last_nnls = 0.0
    lp_gap = max(3.0, min(20.0, float(budget) / 25.0))
    nnls_gap = max(10.0, min(45.0, float(budget) / 10.0)) if (HAVE_NNLS and n <= 1024) else 1.0e30

    lp_fail = 0
    nnls_fail = 0

    while True:
        now = time.monotonic()
        rem = deadline - now
        if rem < 2.0:
            break

        action = "soft"
        lp_enabled = HAVE_LINPROG and n <= 1024
        nnls_enabled = HAVE_NNLS and n <= 1024

        if lp_enabled and (it < 2 or now - last_lp >= lp_gap or (stuck > 120 and lp_fail < 8)):
            action = "lp"

        if nnls_enabled and rem > 20.0:
            if it == 0:
                action = "nnls"
            elif (now - last_nnls >= nnls_gap) and (it < 4 or (action == "soft" and stuck > 60)):
                action = "nnls"

        # ---------------- NNLS action ----------------
        if action == "nnls":
            last_nnls = now
            w = _dual_weights_from_p(p)
            if w is None:
                nnls_fail += 1
                stuck += 2
                it += 1
                continue

            nnls_fail = 0
            cur_sc = _score_norm(p)
            improved = False

            # 1) mirror step with the learned dual weights
            for e in (eta, 0.5 * eta, 0.25 * eta):
                p2 = _soft_step_with_w(p, w, e)
                if p2 is None:
                    continue
                sc2 = _score_norm(p2)
                if np.isfinite(sc2) and sc2 < cur_sc:
                    p = p2
                    cur_sc = sc2
                    improved = True
                    break

            if improved:
                stuck = 0
            else:
                stuck += 1

            # 2) primal NNLS jump toward a p that is stationary for this w
            q = _p_from_weights(w)
            if q is not None:
                cur_sc = _score_norm(p)
                for alpha in (0.25, 0.5, 0.75, 1.0, 0.9, 0.15, 0.35, 0.65):
                    cand = (1.0 - alpha) * p + alpha * q
                    cand = _normalize(cand, 1.0e-18)
                    if cand is None:
                        continue
                    sc = _score_norm(cand)
                    if np.isfinite(sc) and sc < cur_sc:
                        p = cand
                        cur_sc = sc
                        improved = True
                        break

            cur_sc = _score_norm(p)
            if np.isfinite(cur_sc) and cur_sc < best_sc:
                best = p.copy()
                best_sc = float(cur_sc)
                stuck = 0

            it += 1
            continue

        # ---------------- LP action ----------------
        if action == "lp":
            last_lp = now
            tl = min(15.0, max(1.5, rem * 0.25))
            g = _best_response_lp(p, tl)
            if g is None:
                lp_fail += 1
                stuck += 1
                it += 1
                continue

            lp_fail = 0
            cur_sc = _score_norm(p)
            for alpha in (0.05, 0.10, 0.20, 0.35, 0.50, 0.70, 0.90, 1.00):
                cand = (1.0 - alpha) * p + alpha * g
                cand = _normalize(cand, 1.0e-18)
                if cand is None:
                    continue
                sc = _score_norm(cand)
                if np.isfinite(sc) and sc < cur_sc:
                    p = cand
                    cur_sc = sc
                    break

            cur_sc = _score_norm(p)
            if np.isfinite(cur_sc) and cur_sc < best_sc:
                best = p.copy()
                best_sc = float(cur_sc)
                stuck = 0
            else:
                stuck += 2

            it += 1
            continue

        # ---------------- soft chunk ----------------
        chunk_end = min(deadline, now + 2.0)
        attempts = 0

        while attempts < 120:
            now2 = time.monotonic()
            if now2 >= chunk_end or now2 >= deadline:
                break
            attempts += 1

            c = np.convolve(p, p)
            cur_sc = _score_from_conv(p, c)
            if not np.isfinite(cur_sc):
                p = best.copy()
                cur_sc = best_sc
                stuck = 10
                continue

            if cur_sc < best_sc:
                best = p.copy()
                best_sc = float(cur_sc)
                stuck = 0

            M = float(np.max(c))
            med = float(np.median(c))
            base = max(M - med, M * 1.0e-5, 1.0e-300)
            T = max(1.0e-300, Tfrac * base)

            improved = False
            for e in (eta, 0.5 * eta):
                p2 = _soft_step_from_c(p, c, K, T, e)
                if p2 is None:
                    continue
                c2 = np.convolve(p2, p2)
                sc2 = _score_from_conv(p2, c2)
                if np.isfinite(sc2) and sc2 < cur_sc:
                    p = p2
                    cur_sc = sc2
                    improved = True
                    if cur_sc < best_sc:
                        best = p.copy()
                        best_sc = float(cur_sc)
                    break

            if improved:
                eta = min(1.5, eta * 1.05)
            else:
                stuck += 1
                eta = max(0.04, eta * 0.75)

                if stuck > 70:
                    if rng.random() < 0.5:
                        Tfrac = max(0.02, Tfrac * 0.6)
                    else:
                        Tfrac = min(3.0, Tfrac * 1.6)

                    if rng.random() < 0.5:
                        K = int(max(4, min((n + 1) // 2, K * 1.3)))
                    else:
                        K = int(max(4, min((n + 1) // 2, K * 0.7)))

                    if rng.random() < 0.5:
                        noise = np.exp(rng.normal(0.0, 0.25, n))
                    else:
                        noise = np.ones(n, dtype=np.float64)
                        blen = int(rng.integers(max(2, n // 20), max(3, n // 5 + 1)))
                        if blen >= n:
                            blen = max(1, n // 10)
                        bstart = int(rng.integers(0, max(1, n - blen + 1)))
                        noise[bstart:bstart + blen] = float(np.exp(rng.normal(0.0, 0.4)))

                    cand = _normalize(p * noise, 1.0e-18)
                    if cand is not None:
                        sc_c = _score_norm(cand)
                        if np.isfinite(sc_c) and sc_c < cur_sc:
                            p = cand
                            cur_sc = sc_c
                            if cur_sc < best_sc:
                                best = p.copy()
                                best_sc = float(cur_sc)

                    stuck = 0
                    if eta < 0.15:
                        eta = 0.25

                if stuck > 250:
                    p = _normalize(best * np.exp(rng.normal(0.0, 0.35, n)), 1.0e-18)
                    if p is None:
                        p = best.copy()
                    cur_sc = _score_norm(p)
                    eta = 0.30
                    Tfrac = 0.35
                    K = int(max(8, min(300, int(0.12 * n))))
                    stuck = 0

            it += 1

        it += 1

    return best, best_sc


def _optimize_length(n, budget, seed, parent):
    """Generate starts for a fixed length and refine the best candidates."""
    if n <= 1:
        p = _normalize(np.ones(1, dtype=np.float64), 0.0)
        return p, _score_norm(p)

    deadline = time.monotonic() + max(1.0, float(budget))

    if n > 1024:
        max_starts = 10
    elif n > 512:
        max_starts = 16
    else:
        max_starts = 24
    if float(budget) < 100.0:
        max_starts = min(max_starts, 8)

    starts = _generate_starts(n, parent, seed, max_starts)
    if not starts:
        p = _normalize(np.ones(n, dtype=np.float64), 1.0e-18)
        return p, _score_norm(p)

    starts.sort(key=itemgetter(1))
    top = starts[:4]

    best_seq = top[0][0].copy()
    best_score = float(top[0][1])

    rem0 = deadline - time.monotonic()
    if rem0 > 20.0:
        refine_total = max(0.0, rem0 - 5.0)
        weights = (0.30, 0.24, 0.18, 0.12)
        for i in range(min(4, len(starts))):
            alloc = refine_total * weights[i]
            if alloc < 3.0:
                continue
            rem = deadline - time.monotonic()
            if rem < alloc + 3.0:
                alloc = max(3.0, rem - 3.0)
            cand = starts[i][0]
            try:
                c2, s2 = _combined_refine(cand, alloc, seed + 101 * i)
            except Exception as exc:
                print(f"length {n} refine candidate {i} failed: {exc}", flush=True)
                c2, s2 = None, float("inf")
            if c2 is not None and np.isfinite(s2) and s2 < best_score:
                best_seq = c2.copy()
                best_score = float(s2)

    rem = deadline - time.monotonic()
    if rem > 10.0:
        try:
            c2, s2 = _combined_refine(best_seq, rem - 3.0, seed + 777)
        except Exception as exc:
            print(f"length {n} final refine failed: {exc}", flush=True)
            c2, s2 = None, float("inf")
        if c2 is not None and np.isfinite(s2) and s2 < best_score:
            best_seq = c2.copy()
            best_score = float(s2)

    return best_seq, best_score


def propose_candidate(seed=42, budget_s=1500, **kwargs):
    """
    Entry point.  Runs multi-length search within the allotted time and returns
    the best list of nonnegative coefficients found.
    """
    start_time = time.monotonic()
    try:
        budget = float(budget_s)
    except Exception:
        budget = 1500.0
    if not np.isfinite(budget) or budget <= 0.0:
        budget = 1500.0

    deadline = start_time + max(20.0, budget) - 8.0

    # Load parent if available.
    parent = None
    try:
        if "height_sequence_1" in globals():
            parent = _normalize(globals()["height_sequence_1"], 0.0)
    except Exception:
        parent = None

    best_seq = None
    best_score = float("inf")

    if parent is not None:
        best_seq = parent.copy()
        best_score = _score_norm(parent)
        print(f"parent length={parent.size} score={best_score:.8f}", flush=True)
    else:
        print("no parent sequence available; starting from scratch", flush=True)

    # Allocate search across lengths.  The known-good parent is length 512, so
    # emphasize that basin while still probing larger lengths.
    if budget >= 1000.0:
        plan = (
            (512, 0.60),
            (1024, 0.26),
            (2048, 0.11),
        )
    elif budget >= 500.0:
        plan = (
            (512, 0.70),
            (1024, 0.25),
        )
    else:
        plan = (
            (512, 0.90),
            (1024, 0.08),
        )

    seedcur = int(seed)
    for n, frac in plan:
        rem = deadline - time.monotonic()
        if rem < 25.0:
            break
        alloc = budget * float(frac)
        alloc = min(alloc, rem - 5.0)
        if alloc < 10.0:
            alloc = max(5.0, rem * 0.4)

        try:
            cand, sc = _optimize_length(n, alloc, seedcur, best_seq)
            print(f"stage n={n} alloc={alloc:.1f}s best_here={sc:.8f} global={best_score:.8f}", flush=True)
            if cand is not None and np.isfinite(sc) and sc < best_score:
                best_seq = cand.copy()
                best_score = float(sc)
        except Exception as exc:
            print(f"stage n={n} failed: {exc}", flush=True)

        seedcur += 104729

        # If we are already below the stated target, stop probing larger lengths
        # and spend remaining time polishing the best sequence.
        if best_score < 1.5028:
            break

    # Final global polish.
    rem = deadline - time.monotonic()
    if best_seq is not None and rem > 15.0:
        try:
            c2, s2 = _combined_refine(best_seq, rem * 0.55 + 5.0, seedcur + 991)
            if c2 is not None and np.isfinite(s2) and s2 < best_score:
                best_seq = c2.copy()
                best_score = float(s2)
                print(f"final combined refine score={best_score:.8f}", flush=True)
        except Exception as exc:
            print(f"final combined refine failed: {exc}", flush=True)

        rem = deadline - time.monotonic()
        if rem > 10.0 and best_seq is not None and best_seq.size <= 1024:
            try:
                c3, s3 = _linalg_polish(best_seq, min(rem - 2.0, 120.0))
                if c3 is not None and np.isfinite(s3) and s3 < best_score:
                    best_seq = c3.copy()
                    best_score = float(s3)
                    print(f"final L-BFGS polish score={best_score:.8f}", flush=True)
            except Exception as exc:
                print(f"final L-BFGS polish failed: {exc}", flush=True)

    # Fallback to a valid sequence.
    if best_seq is None:
        if parent is not None:
            best_seq = parent.copy()
        else:
            best_seq = _normalize(np.ones(64, dtype=np.float64), 0.0)

    final = _normalize(best_seq, 0.0)
    if final is None:
        final = _normalize(np.ones(64, dtype=np.float64), 0.0)

    # Validate against the official evaluator when available.
    try:
        official = float(evaluate_sequence([float(x) for x in final]))
        if np.isfinite(official):
            best_score = official
        else:
            if parent is not None:
                p2 = _normalize(parent, 0.0)
                official2 = float(evaluate_sequence([float(x) for x in p2]))
                if np.isfinite(official2):
                    final = p2
                    best_score = official2
    except Exception:
        pass

    print(f"FINAL local_score={best_score:.8f} length={len(final)}", flush=True)
    return [float(x) for x in np.clip(np.asarray(final, dtype=np.float64), 0.0, 1000.0)]
```
