def fast_convolve(a):
    """Computes the full linear convolution of a with itself using FFT."""
    n = len(a)
    size = 2*n - 1
    N = 1 << (size-1).bit_length()
    fa = np.fft.rfft(a, N)
    fc = fa * fa
    result = np.fft.irfft(fc, N)[:size]
    # Ensure non-negative and cap values to 1000
    result[result < 0] = 0.0
    result = np.minimum(np.maximum(result, 0.0), 1000.0)
    return result

def project_simplex(v, tol=1e-10):
    """Projects a vector to the simplex {x>=0, sum(x)=1} using an efficient method."""
    n = len(v)
    if n == 0:
        return np.ones(1)/1.0
    u = np.sort(v)[::-1]  # Sort in descending order
    cv = np.cumsum(u) - 1.0
    rho = np.minimum(np.sum(cv>=0) - 1, n-1)
    if rho <= 0:
        return np.ones(n)/n
    if rho >= n:
        rho = n-1
    theta = (cv[rho] + tol) / (rho + 1.0)
    # v_new = v - theta * np.minimum(y, something) and then normalize
    v_new = np.maximum(v - theta, 0.0)
    s = np.sum(v_new)
    if s < tol:
        return np.ones(n)/n
    return v_new / s
