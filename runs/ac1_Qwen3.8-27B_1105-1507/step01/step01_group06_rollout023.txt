Minimizes J(a) = 2*n*max(convolve(a,a))/(sum(a))^2 using:
1. Softmax-smoothed L-BFGS-B with progressive temperature annealing
2. LP best-response mirror descent for structured updates (n <= 1024)
3. Multi-start with diverse asymmetric initializations across multiple lengths (512-4096)
4. High-temperature final polish on the best candidate found
"""

import numpy as np
from scipy.optimize import minimize, linprog
from scipy.signal import correlate
from functools import partial

def fast_conv(a):
    """Full self-convolution via FFT, O(n log n)."""
    n = len(a)
    s = 1
    while s < 2 * n - 1:
        s <<= 1
    fa = np.fft.rfft(a, s)
    return np.fft.irfft(fa * fa, s)[:2 * n - 1]


def compute_obj_grad(x, T):
    """Compute softmax-smoothed objective value and gradient."""
    x = np.maximum(x, 0.0)
    n = len(x)
    s = x.sum()
    if s < 1e-8:
        return 1e10, np.zeros(n)
    c = fast_conv(x)
    Tc = T * c
    max_Tc = Tc.max()
    exp_Tc = np.exp(Tc - max_Tc)
    S = exp_Tc.sum()
    soft_max_val = (max_Tc + np.log(S)) / T
    p = exp_Tc / S
    grad_sm = correlate(p, x.astype(np.float64), mode='valid')
    obj = 2.0 * n * soft_max_val / (s * s)
    grad = 2.0 * n * (grad_sm * s - 2.0 * soft_max_val) / (s * s * s)
    return obj, grad


def softmax_obj_func(x, T):
    """Objective wrapper for minimize."""
    val, _ = compute_obj_grad(x, T)
    return val


def softmax_grad_func(x, T):
    """Gradient wrapper for minimize."""
    _, g = compute_obj_grad(x, T)
    return g


def build_A(a):
    """Build constraint matrix for LP: A[k,j] gives (a*g)_k as linear in g."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for k in range(2 * n - 1):
        jl = max(0, k - n + 1)
        jh = min(n - 1, k)
        if jl <= jh:
            A[k, jl:jh + 1] = a[k - jh:k - jl + 1][::-1]
    return A


def lp_best_response(a, M):
    """LP: max sum(g) s.t. (a*g)_k <= M for all k, g >= 0."""
    n = len(a)
    c_vec = -np.ones(n)
    A = build_A(a)
    r = linprog(c_vec, A_ub=A, b_ub=np.full(2 * n - 1, M),
               bounds=[(0, None)] * n, method='highs')
    if r.success:
        g = r.x
        if g is not None and g.sum() > 1e-10:
            return g
    return None
