```python
"""
Minimizes 2n*max(conv(f,f))/(sum(f))^2 using adaptive mirror descent with
multi-scale initialization, spectral pre-conditioning, and multi-start strategy.
Key innovations: U-shaped and bimodal initializations, adaptive threshold scheduling,
geometric LR decay, periodic restarts, and progressive n enlargement.
"""

import numpy as np
import time


def _fft_conv(f):
    """Full linear convolution of f with itself via FFT."""
    n = len(f)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    F = np.fft.rfft(f, size)
    h = np.fft.irfft(F * F, size)
    return h[:2 * n - 1]


def _score(f):
    """Compute objective: 2n * max(conv(f,f)) / (sum(f))^2."""
    n = len(f)
    h = _fft_conv(f)
    M = float(np.max(h))
    S = float(np.sum(f))
    if S < 0.01:
        return float('inf')
    return 2.0 * n * M / (S * S)


def _normalize(f):
    """Clip to non-negative and normalize to sum 1."""
    f = np.maximum(f, 0.0)
    s = float(np.sum(f))
    if s < 1e-30:
        return None
    return f / s


def _subgradient(f, active):
    """Average subgradient of max_k conv(f,f)[k] over active set."""
    n = len(f)
    g = np.zeros(n, dtype=np.float64)
    for k in active:
        m_lo = max(0, k - n + 1)
        m_hi = min(n - 1, k)
        if m_hi >= m_lo:
            # d b[k]/d f[m] = 2*f[k-m] for m in [m_lo, m_hi]
            # k-m ranges from k-m_hi (>=0) to k-m_lo (<n)
            g[m_lo:m_hi + 1] += 2.0 * f[k - m_hi:k - m_lo + 1][::-1]
    g /= float(len(active))
    return g


def _mirror_descent(f_init, n_iter, eta0, eta_decay, thresh_start, thresh_end,
                    restart_patience=500, label=""):
    """
    Mirror descent for min max_k conv(f,f)[k] s.t. f>=0, sum(f)=1.
    Adaptive threshold and geometric LR decay.
    """
    f = _normalize(np.asarray(f_init, dtype=np.float64))
    if f is None:
        return np.asarray(f_init, dtype=np.float64).copy(), float('inf')
    n = len(f)
    best_f = f.copy()
    best_score = _score(f)
    stagnation = 0

    for it in range(n_iter):
        h = _fft_conv(f)
        M = float(np.max(h))
        if M <= 0:
            break

        # Adaptive threshold: interpolate from thresh_start to thresh_end
        frac = (it + 1) / float(n_iter)
        thresh = thresh_start + (thresh_end - thresh_start) * frac
        threshold = thresh * M + 1e-30
        active = np.where(h >= threshold)[0]
        if len(active) == 0:
            active = np.array([int(np.argmax(h))])

        g = _subgradient(f, active)

        eta = eta0 * (eta_decay ** it)
        log_ratio = -eta * g
        log_ratio = np.clip(log_ratio, -30.0, 30.0)

        f = f * np.exp(log_ratio)
        f_n = _normalize(f)
        if f_n is None:
            f = np.ones(n) / n
        else:
            f = f_n

        s = _score(f)
        if s < best_score - 1e-10:
            best_score = s
            best_f = f.copy()
            stagnation = 0
        else:
            stagnation += 1

        # Restart if stuck
        if stagnation >= restart_patience and it < n_iter - 100:
            rng = np.random.default_rng(hash((n, it)) % 2**31)
            pert = rng.random(n) * 0.1
            f_p = _normalize(best_f + pert)
            if f_p is not None:
                f = f_p
            stagnation = 0

    return best_f, best_score


def _spectral_precondition(f_init, n_iter, alpha):
    """
    Spectral pre-conditioning: iteratively flatten the power spectrum
    by compressing dynamic range of |F|, then project back to f>=0.
    """
    n = len(f_init)
    f = _normalize(np.asarray(f_init, dtype=np.float64))
    if f is None:
        return np.asarray(f_init, dtype=np.float64).copy()

    N = 1
    while N < 2 * n - 1:
        N <<= 1

    best_f = f.copy()
    best_score = _score(f)

    for it in range(n_iter):
        F = np.fft.rfft(f, N)
        amp = np.abs(F)
        # Compress dynamic range
        amp_max = np.max(amp)
        amp_min = np.min(amp[amp > 0]) if np.any(amp > 0) else 0
        if amp_max <= 0:
            break
        # Geometric compression: amp_new = amp^alpha * amp_max^(1-alpha)
        amp_new = amp ** alpha * amp_max ** (1.0 - alpha)
        # Ensure minimum
        amp_new = np.maximum(amp_new, amp_min * 0.1)
        phase = np.angle(F)
        F_new = amp_new * np.exp(1j * phase)
        f_new = np.fft.irfft(F_new, N)[:n]
        f_new = np.maximum(f_new, 0.0)
        s = float(np.sum(f_new))
        if s < 1e-10:
            break
        f_new = f_new / s
        sc = _score(f_new)
        if sc < best_score:
            best_score = sc
            best_f = f_new.copy()
        f = f_new

    return best_f


def _cos_init(n):
    """Cosine-squared (bell) initialization."""
    j = np.arange(n, dtype=np.float64)
    f = np.cos(np.pi * (j + 0.5) / n) ** 2
    return _normalize(f)


def _u_init(n):
    """U-shaped (raised cosine) initialization."""
    j = np.arange(n, dtype=np.float64)
    f = 1.0 + np.cos(2.0 * np.pi * (j + 0.5) / n)
    return _normalize(f)


def _bimodal_init(n, k_peaks=2):
    """Multi-modal initialization with k_peaks peaks."""
    j = np.arange(n, dtype=np.float64)
    f = np.sin(k_peaks * np.pi * (j + 0.5) / n) ** 2
    return _normalize(f)


def _raised_cosine_window(n):
    """Standard raised cosine window."""
    j = np.arange(n, dtype=np.float64)
    f = 0.5 * (1 + np.cos(2 * np.pi * j / n))
    f = np.maximum(f, 0.0)
    return _normalize(f)


def _smoothed_random(n, rng, sigma_frac=0.2):
    """Random signal smoothed by a Gaussian kernel."""
    f = rng.random(n)
    # Circular Gaussian smoothing
    kernel_size = max(3, int(2 * sigma_frac * n) | 1)
    kernel = np.exp(-0.5 * (np.arange(kernel_size) - kernel_size // 2) ** 2 / (sigma_frac * n / 3) ** 2)
    kernel /= kernel.sum()
    # Pad for circular convolution
    f_padded = np.concatenate([f[-kernel_size//2:], f, f[:kernel_size//2]])
    f_smooth = np.convolve(f_padded, kernel, mode='valid')[:n]
    return _normalize(f_smooth)


def _upsample(f, factor=2):
    """Upsample by zero insertion followed by simple smoothing."""
    n = len(f)
    new_n = n * factor
    f_new = np.zeros(new_n, dtype=np.float64)
    f_new[::factor] = f
    # Simple box filter smoothing
    if new_n > 2:
        f_new = np.convolve(f_new, np.ones(factor) / factor, mode='same')
    return _normalize(f_new)

        n = len(f)
        print(f"    MS n={n}: score={s:.6f}")

    # Get parent
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64).copy()
        parent = np.clip(parent, 0.0, 1000.0)
        pnorm = _normalize(parent)
    except Exception:
        pnorm = None
        parent = None

    best_f = None
    best_score = float('inf')

    if pnorm is not None:
        best_f = pnorm.copy()
        best_score = _score(pnorm)
        print(f"Parent: n={len(pnorm)}, score={best_score:.6f}")

    # --- Phase 1: Multi-scale optimization for various target sizes ---
    target_ns = [1024, 2048, 4096]
    iters_ms = [3000, 5000, 8000]

    for idx, (tn, ims) in enumerate(zip(target_ns, iters_ms)):
        if time.monotonic() > deadline - 300:
            break
        t0 = time.monotonic()
        f_ms, s_ms = _multiscale_optimize(tn, ims, 0.8, 0.999, 0.95, 0.6, rng, "ms")
        el = time.monotonic() - t0
        print(f"  Multiscale n={tn}: score={s_ms:.6f} ({el:.1f}s)")
        if s_ms < best_score:
            best_score = s_ms
            best_f = f_ms.copy()
            print(f"  ** New best: {best_score:.6f} **")

    # --- Phase 2: Multi-start direct optimization at best n ---
    n_best = len(best_f) if best_f is not None else 2048

    inits = []
    # Parent
    if pnorm is not None:
        inits.append(("parent", _normalize(pnorm) if len(pnorm) == n_best else _resize(pnorm, n_best)))
    # Cosine
    inits.append(("cos", _cos_init(n_best)))
    # U-shape
    inits.append(("u", _u_init(n_best)))
    # Bimodal
    inits.append(("bim2", _bimodal_init(n_best, 2)))
    inits.append(("bim4", _bimodal_init(n_best, 4)))
    # Raised cosine
    inits.append(("rc", _raised_cosine_window(n_best)))
    # Smoothed random (multiple)
    for ri in range(3):
        inits.append((f"sr{ri}", _smoothed_random(n_best, rng, 0.15 + 0.1 * ri)))

    print(f"\nPhase 2: Direct optimization at n={n_best}, {len(inits)} initializations")

    remaining = deadline - time.monotonic() - 120
    time_per_init = max(5.0, remaining / max(1, len(inits)))

    for idx, (name, f0) in enumerate(inits):
        if time.monotonic() > deadline - 120:
            break
        t0 = time.monotonic()
        alloc = min(time_per_init * 1.5, deadline - time.monotonic() - 120)
        n_iter = int(max(2000, min(20000, alloc * 800)))

        # Spectral pre-conditioning first
        f_pc = _spectral_precondition(f0, 100, 0.7)
        s_pc = _score(f_pc)

        # Then mirror descent
        f_md, s_md = _mirror_descent(f_pc, n_iter, 0.6, 0.9995, 0.95, 0.5,
                                     restart_patience=800, label=name)
        el = time.monotonic() - t0
        print(f"  [{idx+1}/{len(inits)}] {name}: pc={s_pc:.6f}, md={s_md:.6f} ({el:.1f}s)")

        if s_md < best_score:
            best_score = s_md
            best_f = f_md.copy()
            print(f"  ** New best: {best_score:.6f} from {name} **")

    # --- Phase 3: Refine best solution ---
    print(f"\nPhase 3: Refining best (score={best_score:.6f}, n={len(best_f)})")
    if time.monotonic() < deadline - 60:
        remaining = deadline - time.monotonic() - 60
        n_iter_ref = int(max(5000, min(30000, remaining * 500)))

        # Fine mirror descent with small eta and low threshold
        f_fin, s_fin = _mirror_descent(best_f, n_iter_ref, 0.1, 0.9999, 0.8, 0.3,
                                       restart_patience=1500, label="fine")
        if s_fin < best_score:
            best_score = s_fin
            best_f = f_fin.copy()
            print(f"  ** Fine refinement: {best_score:.6f} **")

        # Try different threshold strategies
        if time.monotonic() < deadline - 30:
            for tf in [0.5, 0.7, 0.9]:
                if time.monotonic() > deadline - 20:
                    break
                n_it = int(min(10000, (deadline - time.monotonic() - 10) * 200))
                f_t, s_t = _mirror_descent(best_f, n_it, 0.05, 0.99995, tf, tf - 0.2,
                                           restart_patience=2000, label=f"th{int(tf*10)}")
                if s_t < best_score:
                    best_score = s_t
                    best_f = f_t.copy()
                    print(f"  ** Threshold {tf}: {best_score:.6f} **")

    # --- Phase 4: Try a slightly different n ---
    if time.monotonic() < deadline - 30 and best_f is not None:
        n_alt = len(best_f) * 2
        if n_alt <= 8192:
            t0 = time.monotonic()
            f_alt = _upsample(best_f, 2)
            n_it = int(min(10000, (deadline - time.monotonic() - 10) * 400))
            f_a, s_a = _mirror_descent(f_alt, n_it, 0.3, 0.9997, 0.9, 0.5,
                                       restart_patience=1000, label="double_n")
            el = time.monotonic() - t0
            print(f"  Double n={n_alt}: score={s_a:.6f} ({el:.1f}s)")
            if s_a < best_score:
                best_score = s_a
                best_f = f_a.copy()
                print(f"  ** Doubling n: {best_score:.6f} **")

    # --- Final verification ---
    if best_f is None:
        # Fallback: return uniform
        best_f = np.ones(512) / 512
        best_score = _score(best_f)

    final_list = [float(x) for x in best_f]
    final_score = float(evaluate_sequence(final_list))
    print(f"\nFinal verified score: {final_score:.6f} (n={len(final_list)})")

    if not np.isfinite(final_score) or final_score > 2.5:
        print("  Fallback to parent.")
        if parent is not None:
            return [float(x) for x in parent]
        return [1.0] * 512

    return final_list


def _resize(f, target_n):
    """Resize f to target_n via interpolation."""
    from numpy import interp
    n = len(f)
    if n == target_n:
        return _normalize(f)
    old_x = np.linspace(0, 1, n)
    new_x = np.linspace(0, 1, target_n)
    f_new = np.interp(new_x, old_x, f)
    return _normalize(f_new)
```
