def normalize_sequence(arr):
    """Return a nonnegative floating vector normalized to sum 1, or None."""
    try:
        x = np.asarray(arr, dtype=np.float64).ravel()
    except Exception:
        return None
    if x.size == 0:
        return None
    x = np.where(np.isfinite(x), x, 0.0)
    x = np.maximum(x, 0.0)
    s = float(x.sum())
    if s <= 1e-300:
        return None
    x /= s
    return x


def score_normalized(x, n):
    """Score = 2*n*max(conv(x,x)) assuming sum(x)=1."""
    conv = np.convolve(x, x)
    m = float(np.max(conv))
    if not np.isfinite(m):
        return float("inf")
    return float(2.0 * n * m)


def lse_objective_and_grad(f, beta):
    """
    Compute the log-sum-exp of (f*f) and its gradient.
    Returns (value, gradient, max_conv).
    """
    c = np.convolve(f, f)
    c_max = float(c.max())
    if not np.isfinite(c_max) or c_max <= 0:
        return c_max, np.zeros_like(f), c_max
    
    # For numerical stability, shift by max
    shifted = c - c_max
    log_p = beta * shifted
    max_log_p = float(log_p.max())
    p = np.exp(log_p - max_log_p)
    p_sum = float(p.sum())
    if p_sum <= 0:
        return c_max, np.zeros_like(f), c_max
    p /= p_sum
    
    value = c_max + float(np.log(p_sum)) / beta  # = (1/beta)*log(sum(exp(beta*c)))
    
    # grad[j] = sum_k p[k] * f[k-j] = np.convolve(f, p, mode='valid')
    grad = np.convolve(f, p, mode='valid')
    
    return value, grad, c_max


def project_simplex(v, s=1.0):
    """Project v onto {x >= 0, sum(x) = s}."""
    v = np.asarray(v, dtype=np.float64).copy()
    n = v.size
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cv - s)
    if not cond.any():
        return np.full(n, s / n)
    rho = int(np.where(cond)[0][-1])
    theta = (cv[rho] - s) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def multiplicative_update(f, grad, eta):
    """Multiplicative (mirror descent) update: f * exp(-eta*grad), then normalize."""
    log_f = np.log(np.maximum(f, 1e-300)) - eta * grad
    # For numerical stability, shift
    log_f -= log_f.max()
    new_f = np.exp(log_f)
    s = float(new_f.sum())
    if s <= 0 or not np.isfinite(s):
        return f.copy()
    return new_f / s

            break

def make_conv_template(n):
    """Precompute CSR matrix for convolve(f, g)."""
    m = 2 * n - 1
    k64 = np.arange(m, dtype=np.int64)
    js = np.maximum(0, k64 - n + 1).astype(np.int32)
    je = np.minimum(n - 1, k64).astype(np.int32)
    counts = (je.astype(np.int64) - js.astype(np.int64) + 1).astype(np.int64)
    total = int(counts.sum())
    
    cum = np.cumsum(counts)
    starts = np.repeat(cum - counts, counts)
    local = np.arange(total, dtype=np.int64) - starts
    
    row = np.repeat(k64, counts).astype(np.int32)
    col_base = js[row.astype(np.int64)]
    col = (col_base + local).astype(np.int32)
    offsets = (row.astype(np.int64) - col.astype(np.int64)).astype(np.int32)
    
    A_off = csr_matrix((offsets, (row, col)), shape=(m, n), dtype=np.int32)
    edge_offsets = A_off.data.astype(np.int32, copy=True)
    A_template = A_off.astype(np.float64).copy()
    A_template.data.fill(0.0)
    
    c_obj = -np.ones(n, dtype=np.float64)
    bounds = [(0.0, None)] * n
    return A_template, edge_offsets, c_obj, bounds, m


def solve_lp_direction(template, f, n, M):
    """Solve the LP direction problem."""
    A_template, edge_offsets, c_obj, bounds, m = template
    
    # Scale for numerical stability
    lp_scale = float(n)
    A = A_template.copy()
    A.data[:] = f[edge_offsets]
    A.data *= lp_scale
    b = np.full(m, M * lp_scale * lp_scale, dtype=np.float64)
    
    res = linprog(c_obj, A_ub=A, b_ub=b, bounds=bounds, method="highs")
    return res

def interpolate_sequence(x, n_target):
    """Resample a normalized sequence to a new length."""
    x = normalize_sequence(x)
    if x is None or n_target <= 0:
        return None
    m = x.size
    if m == n_target:
        return x
    if m == 1:
        return np.ones(n_target) / n_target
    old = np.linspace(0.0, 1.0, m)
    new = np.linspace(0.0, 1.0, n_target)
    out = np.interp(new, old, x)
    return normalize_sequence(out)


def generate_starts(n, rng, best_f=None, given_f=None):

        if up is not None:
            noised = up + rng.normal(0, 0.02, n)
            noised = np.maximum(noised, 0)
            starts.append(normalize_sequence(noised))
    
    # From given
    if given_f is not None:
        up = interpolate_sequence(given_f, n)
        if up is not None:
            starts.append(up)
    
    # Uniform
    starts.append(np.ones(n, dtype=np.float64) / n)
    
    # Raised cosine
    t = np.linspace(0, 1, n)
    rc = 0.5 * (1 + np.cos(2 * np.pi * t))
    starts.append(normalize_sequence(rc))
    
    # Triangular
    tri = np.minimum(t, 1 - t) * 2
    starts.append(normalize_sequence(tri))
    
    # Random Dirichlet
    starts.append(rng.dirichlet(np.ones(n)))
    
    # Remove None entries
    starts = [s for s in starts if s is not None and np.all(np.isfinite(s))]
    
    return starts

"""
Hybrid LP + log-sum-exp gradient search for minimizing
2*n*max(convolve(f,f))/(sum(f))^2 over non-negative sequences.

Strategy:
- Primary: LP-based directional search (max sum(g) s.t. f*g <= M, g>=0)
- Secondary: Log-sum-exp gradient descent for fast local polishing
- Tertiary: Subgradient method for breaking symmetry at peak ties
- Multi-length exploration with adaptive time allocation
"""

import time
import numpy as np
from scipy.optimize import linprog
from scipy.sparse import csr_matrix


def normalize_sequence(arr):
    """Return a nonnegative floating vector normalized to sum 1, or None."""
    try:
        x = np.asarray(arr, dtype=np.float64).ravel()
    except Exception:
        return None
    if x.size == 0:
        return None
    x = np.where(np.isfinite(x), x, 0.0)
    x = np.maximum(x, 0.0)
    s = float(x.sum())
    if s <= 1e-300:
        return None
    x /= s
    return x


def score_normalized(x, n):
    """Score = 2*n*max(conv(x,x)) assuming sum(x)=1, x>=0."""
    conv = np.convolve(x, x)
    m = float(np.max(conv))
    if not np.isfinite(m):
        return float("inf")
    return float(2.0 * n * m)


def lse_objective_and_grad(f, beta):
    """
    Compute log-sum-exp of (f*f) coefficients and its gradient.
    Returns (lse_value, gradient_array, max_conv_value).
    """
    c = np.convolve(f, f)
    c_max = float(c.max())
    if not np.isfinite(c_max) or c_max <= 0:
        return c_max, np.zeros_like(f), c_max
    shifted = c - c_max
    log_p = beta * shifted
    max_log_p = float(log_p.max())
    p = np.exp(log_p - max_log_p)
    p_sum = float(p.sum())
    if p_sum <= 0 or not np.isfinite(p_sum):
        return c_max, np.zeros_like(f), c_max
    p /= p_sum
    value = c_max + np.log(p_sum) / beta
    grad = np.convolve(f, p, mode='valid')
    return value, grad, c_max


def project_simplex(v, s=1.0):
    """Project v onto probability simplex {x >= 0, sum(x) = s}."""
    v = np.asarray(v, dtype=np.float64).copy()
    n = v.size
    if n == 0:
        return v
    if n == 1:
        return np.array([s])
    u = np.sort(v)[::-1]
    cv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cv - s)
    if not cond.any():
        return np.full(n, s / n)
    rho = int(np.where(cond)[0][-1])
    theta = (cv[rho] - s) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def multiplicative_update(f, grad, eta):
    """Mirror descent step: f * exp(-eta*grad), renormalized."""
    log_f = np.log(np.maximum(f, 1e-300)) - eta * grad
    log_f -= log_f.max()
    new_f = np.exp(log_f)
    s = float(new_f.sum())
    if s <= 0 or not np.isfinite(s):
        return f.copy()
    return new_f / s


def lse_gradient_descent(f, time_budget, beta=80.0, eta_init=0.01, rng=None):
    """
    Optimize using gradient descent on log-sum-exp of convolution.
    Uses both projected gradient and multiplicative updates.
    Returns (best_f, best_score).
    """
    f = normalize_sequence(f)
    if f is None:
        return f, float('inf')
    n = f.size
    best_f = f.copy()
    best_score = score_normalized(f, n)
    deadline = time.monotonic() + time_budget
    eta = eta_init
    stall = 0
    it = 0

    while time.monotonic() < deadline:
        it += 1
        value, grad, c_max = lse_objective_and_grad(f, beta)
        if not np.isfinite(value) or not np.all(np.isfinite(grad)):
            break

        grad_norm = float(np.max(np.abs(grad)))
        if grad_norm < 1e-15:
            break

        best_step_f = None
        best_step_score = best_score

        for eta_try in [eta, 0.5 * eta, 2.0 * eta, 0.1 * eta, 5.0 * eta, 0.01 * eta]:
            if eta_try <= 0 or not np.isfinite(eta_try):
                continue
            f_pg = project_simplex(f - eta_try * grad)
            sc_pg = score_normalized(f_pg, n)
            if np.isfinite(sc_pg) and sc_pg < best_step_score:
                best_step_score = sc_pg
                best_step_f = f_pg

            f_mu = multiplicative_update(f, grad, eta_try)
            sc_mu = score_normalized(f_mu, n)
            if np.isfinite(sc_mu) and sc_mu < best_step_score:
                best_step_score = sc_mu
                best_step_f = f_mu

        if best_step_f is not None and best_step_score < best_score:
            f = best_step_f
            best_f = f.copy()
            best_score = best_step_score
            stall = 0
            eta = min(eta * 1.03, 0.05)
        else:
            stall += 1
            eta *= 0.7
            if stall > 50:
                break

    return best_f, float(best_score)


def subgradient_step(f, eta, rng=None):
    """
    One subgradient step on the max convolution coefficient.
    Uses the subgradient at active (near-max) constraints.
    """
    c = np.convolve(f, f)
    c_max = float(c.max())
    if c_max <= 0 or not np.isfinite(c_max):
        return f
    # Find active constraints (within 1% of max)
    threshold = c_max * 0.995
    active = np.where(c >= threshold)[0]
    # Subgradient: average of dc_k/df for active k
    # dc_k/df_j = f[k-j] (with f=0 outside [0,n-1])
    n = f.size
    grad = np.zeros(n)
    for k in active:
        # (dc_k/df)_j = f[k-j] valid when 0 <= k-j < n
        js = np.arange(max(0, k - n + 1), min(n, k + 1))
        if len(js) > 0:
            grad[js] += f[k - js]
    grad /= len(active)
    f_new = project_simplex(f - eta * grad)
    return f_new


def make_conv_template(n):
    """Precompute sparse CSR structure for convolution matrix."""
    m = 2 * n - 1
    k64 = np.arange(m, dtype=np.int64)
    js = np.maximum(0, k64 - n + 1).astype(np.int32)
    je = np.minimum(n - 1, k64).astype(np.int32)
    counts = (je.astype(np.int64) - js.astype(np.int64) + 1).astype(np.int64)
    total = int(counts.sum())
    cum = np.cumsum(counts)
    starts = np.repeat(cum - counts, counts)
    local = np.arange(total, dtype=np.int64) - starts
    row = np.repeat(k64, counts).astype(np.int32)
    col_base = js[row.astype(np.int64)]
    col = (col_base + local).astype(np.int32)
    offsets = (row.astype(np.int64) - col.astype(np.int64)).astype(np.int32)
    A_off = csr_matrix((offsets, (row, col)), shape=(m, n), dtype=np.int32)
    edge_offsets = A_off.data.astype(np.int32, copy=True)
    A_template = A_off.astype(np.float64).copy()
    A_template.data.fill(0.0)
    c_obj = -np.ones(n, dtype=np.float64)
    bounds = [(0.0, None)] * n
    return A_template, edge_offsets, c_obj, bounds, m


def solve_lp(template, f, n, M):
    """Solve max sum(g) s.t. (f*g)_k <= M*scale, g >= 0. Scaled by n."""
    A_template, edge_offsets, c_obj, bounds, m = template
    lp_scale = float(n)
    A = A_template.copy()
    A.data[:] = f[edge_offsets]
    A.data *= lp_scale
    b = np.full(m, M * lp_scale * lp_scale, dtype=np.float64)
    res = linprog(c_obj, A_ub=A, b_ub=b, bounds=bounds, method="highs")
    return res

                f = np.maximum(f, 0)

                phase = "lp"
                stall = 0

    return best_f, float(best_score)


def interpolate_sequence(x, n_target):
    """Resample normalized sequence to new length."""
    x = normalize_sequence(x)
    if x is None or n_target <= 0:
        return None
    m = x.size
    if m == n_target:
        return x
    if m == 1:
        return np.ones(n_target) / n_target
    old = np.linspace(0.0, 1.0, m)
    new = np.linspace(0.0, 1.0, n_target)
    out = np.interp(new, old, x)
    return normalize_sequence(out)


def generate_starts(n, rng, best_f=None, given_f=None):
    """Generate diverse starting sequences of length n."""
    starts = []
    if best_f is not None:
        up = interpolate_sequence(best_f, n)
        if up is not None:
            noised = up + rng.normal(0, 0.01, n)
            noised = np.maximum(noised, 0)
            starts.append(normalize_sequence(noised))
    if given_f is not None:
        up = interpolate_sequence(given_f, n)
        if up is not None:
            starts.append(up)
    starts.append(np.ones(n, dtype=np.float64) / n)
    t = np.linspace(0, 1, n)
    rc = 0.5 * (1.0 + np.cos(2.0 * np.pi * t))
    starts.append(normalize_sequence(rc))
    tri = np.minimum(t, 1.0 - t) * 2.0
    starts.append(normalize_sequence(tri))
    # Gaussian-like
    mu = n / 2.0
    sigma = n / 4.0
    gauss = np.exp(-0.5 * ((np.arange(n) - mu) / sigma) ** 2)
    starts.append(normalize_sequence(gauss))
    starts = [s for s in starts if s is not None and s.size == n and np.all(np.isfinite(s))]
    return starts


def propose_candidate(seed=42, budget_s=1500, **kwargs):
    """
    Search for low-scoring non-negative sequence using hybrid
    LP + LSE gradient + subgradient optimization.
    """
    rng = np.random.default_rng(seed)
    start_time = time.monotonic()
    total_budget = float(budget_s)
    deadline = start_time + total_budget - 10.0

    best_f = None
    best_score = float('inf')
    given_f = None

    # Load given sequence
    try:
        raw = np.asarray(height_sequence_1, dtype=np.float64)
        given_f = normalize_sequence(raw)
        if given_f is not None:
            best_f = given_f.copy()
            best_score = score_normalized(given_f, given_f.size)
            print(f"Given: n={given_f.size}, score={best_score:.6f}", flush=True)
    except Exception:
        pass

    if best_f is None:
        best_f = np.ones(256, dtype=np.float64) / 256.0
        best_score = 2.0

    # Determine which lengths to try
    # Focus on the given length and a few nearby
    given_n = given_f.size if given_f is not None else 1024
    candidate_lengths = [given_n]
    for alt_n in [512, 1024, 2048, 256]:
        if alt_n not in candidate_lengths:
            candidate_lengths.append(alt_n)
    # Sort: given_n first, then by size
    candidate_lengths = sorted(candidate_lengths, key=lambda x: (x != given_n, abs(x - given_n)))

    # Time allocation: more time for the given length
    time_alloc = {}
    remaining_total = total_budget - 30.0
    # 60% to given length, 40% split among others
    time_alloc[given_n] = 0.6 * remaining_total
    other_time = 0.4 * remaining_total
    others = [n for n in candidate_lengths if n != given_n]
    if others:
        per_other = other_time / len(others)
        for n in others:
            time_alloc[n] = per_other

    for n in candidate_lengths:
        remaining = deadline - time.monotonic()
        if remaining < 15.0:
            break
        alloc = min(time_alloc.get(n, 20.0), remaining - 10.0)
        if alloc < 10.0:
            continue

        print(f"\n{'='*50}", flush=True)
        print(f"=== n={n}, alloc={alloc:.0f}s, current_best={best_score:.6f} ===", flush=True)

        try:
            template = make_conv_template(n)
        except Exception as e:
            print(f"  Template failed for n={n}: {e}", flush=True)
            continue

        starts = generate_starts(n, rng, best_f, given_f)
        m = len(starts)
        per_start = alloc / max(1, m)

        if per_start < 8.0:
            # Not enough time per start, just use the best 2
            starts = starts[:2]
            per_start = alloc / 2

        for idx, init in enumerate(starts):
            remaining = deadline - time.monotonic()
            if remaining < 8.0:
                break
            tb = min(per_start, remaining - 5.0)
            if tb < 5.0:
                continue
            print(f"  Start {idx+1}/{len(starts)}, tb={tb:.0f}s, "
                  f"sum_check={float(init.sum()):.6f}", flush=True)
            f_res, s_res = optimize_combined(init, tb, template, rng, n)
            if np.isfinite(s_res) and s_res < best_score:
                best_score = s_res
                best_f = np.asarray(f_res, dtype=np.float64).copy()
                print(f"  *** NEW BEST: n={best_f.size}, score={best_score:.6f} ***", flush=True)

        del template

    # Final polish phase on the best sequence
    remaining = deadline - time.monotonic()
    if remaining > 10.0 and best_f is not None:
        n_best = best_f.size
        print(f"\n=== Final polish, n={n_best}, time={remaining:.0f}s ===", flush=True)
        try:
            template = make_conv_template(n_best)
            # LP polish
            lp_time = min(remaining * 0.5, 200.0)
            f_res, s_res = optimize_combined(best_f, lp_time, template, rng, n_best)
            if np.isfinite(s_res) and s_res < best_score:
                best_score = s_res
                best_f = f_res.copy()
                print(f"  After LP polish: score={best_score:.6f}", flush=True)
            del template
        except Exception as e:
            print(f"  Final LP polish error: {e}", flush=True)

        # LSE polish
        remaining = deadline - time.monotonic()
        if remaining > 5.0:
            lse_time = min(remaining - 5.0, 30.0)
            f_lse, s_lse = lse_gradient_descent(best_f, lse_time, beta=100.0, eta_init=0.005)
            if s_lse < best_score:
                best_score = s_lse
                best_f = f_lse.copy()
                print(f"  After LSE polish: score={best_score:.6f}", flush=True)

    # Convert to output
    if best_f is None:
        best_f = np.ones(64) / 64.0
    seq = [float(x) for x in best_f]

    # Verify
    try:
        final_score = float(evaluate_sequence(seq))
        print(f"\nFINAL: n={len(seq)}, evaluate_sequence score={final_score:.6f}", flush=True)
    except Exception as e:
        print(f"Verification failed: {e}", flush=True)
        if given_f is not None:
            seq = [float(x) for x in given_f]
        else:
            seq = [1.0 / 64.0] * 64

    total_time = time.monotonic() - start_time
    print(f"Total time used: {total_time:.1f}s", flush=True)
    return seq
