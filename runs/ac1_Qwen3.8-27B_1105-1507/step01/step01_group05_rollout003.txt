    # Using FFT: (p ⋆ f) = ifft(fft(p) * conj(fft(f)))
    # But we need to be careful about the length.
    
    fft_size = 1
    while fft_size < 2*n - 1:
        fft_size *= 2
    
    P = np.fft.rfft(p, fft_size)
    F = np.fft.rfft(f, fft_size)
    cross = np.fft.irfft(P * np.conj(F), fft_size)[:n]
    
    grad = 2.0 * cross
    
    # Multiplicative update
    # f_new[i] = f[i] * exp(-eta * grad[i]) / Z
    log_scale = -eta * grad
    log_scale = log_scale - log_scale.max()  # for stability
    weights = np.exp(log_scale)
    f_new = f * weights
    S = f_new.sum()
    if S < 1e-300:
        return f
    f_new = f_new / S
    
    return f_new

def softmax_gradient_descent(f0, time_budget, initial_tau=1.0, final_tau=0.01,

"""
Hybrid optimization combining softmax gradient descent (global phase) with 
LP-based active-set polishing (local phase) for minimizing 2*n*max(convolve(a,a))/sum(a)^2.
Uses FFT-accelerated convolutions, multiple sequence lengths, diverse initializations,

import numpy as np
from scipy.optimize import linprog
import time


def fast_convolve(f):
    """Compute full convolution using FFT."""
    n = len(f)
    size = 2 * n - 1
    fft_size = 1
    while fft_size < size:
        fft_size *= 2
    F = np.fft.rfft(f, fft_size)
    conv = np.fft.irfft(F * F, fft_size)[:size]
    return np.maximum(conv, 0)  # clip tiny negatives from FFT


def compute_score_fast(f):
    """Compute score = 2*n*max(conv)/sum^2 for arbitrary f."""
    n = len(f)
    S = f.sum()
    if S < 1e-300:
        return np.inf
    conv = fast_convolve(f)
    M = conv.max()
    return 2.0 * n * M / (S * S)


def compute_score_normalized(f):
    """Compute score assuming sum(f)=1: 2*n*max(conv)."""
    n = len(f)
    conv = fast_convolve(f)
    return 2.0 * n * conv.max()


def gradient_descent_step(f, tau, eta):
    """One step of smoothed (log-sum-exp) gradient descent on the simplex.
    
    Minimize 2n * logsumexp(conv(f,f)/tau) * tau  (smooth approximation to 2n*max(conv))
    subject to f >= 0, sum(f) = 1.
    
    Uses multiplicative (mirror descent with entropy) update.
    """
    n = len(f)
    c = fast_convolve(f)
    
    # Softmax weights over convolution values
    c_scaled = c / tau
    c_scaled -= c_scaled.max()
    p = np.exp(c_scaled)
    p_sum = p.sum()
    if p_sum < 1e-300:
        return f
    p = p / p_sum
    
    # Compute gradient: grad[i] = 4n * sum_j p[i+j] * f[j]
    # This is 4n * cross_correlate(p, f)[i]
    fft_size = 1
    while fft_size < 2 * n - 1:
        fft_size *= 2
    
    P = np.fft.rfft(p, fft_size)
    F = np.fft.rfft(f, fft_size)
    cross = np.fft.irfft(P * np.conj(F), fft_size)[:n]
    
    grad = 4.0 * n * cross
    
    # Multiplicative update with clipping for stability
    log_weights = -eta * grad
    log_weights -= log_weights.max()
    # Clip to avoid underflow
    log_weights = np.clip(log_weights, -50, 50)
    weights = np.exp(log_weights)
    
    f_new = f * weights
    S = f_new.sum()
    if S < 1e-300:
        return f
    f_new = f_new / S
    
    return f_new


def softmax_gd(f0, time_budget, tau_start=0.1, tau_end=0.005, eta=0.02,
               iters_per_tau=20, rng=None, verbose=False):
    """Run annealed softmax gradient descent."""
    if rng is None:
        rng = np.random.default_rng(0)
    
    f = f0.copy()
    f = np.clip(f, 0, None)
    S = f.sum()
    if S < 1e-300:
        return f, np.inf
    f = f / S
    
    best_f = f.copy()
    best_score = compute_score_normalized(f)
    
    start_time = time.monotonic()
    n = len(f)
    n_tau_steps = max(1, int(time_budget / (iters_per_tau * 0.001)))  # estimate
    n_tau_steps = min(n_tau_steps, 200)
    
    for i in range(n_tau_steps):
        if time.monotonic() - start_time > time_budget:
            break
        
        progress = i / max(1, n_tau_steps - 1)

        tau = tau_start * (tau_end / tau_start) ** progress

def build_conv_matrix(f):
    """Build matrix A such that A @ b = np.convolve(f, b, mode='full')."""
    n = len(f)
    A = np.zeros((2 * n - 1, n))
    for k in range(2 * n - 1):
        j_start = max(0, k - n + 1)
        j_end = min(n, k + 1)
        for j in range(j_start, j_end):
            A[k, j] = f[k - j]
    return A

            f_rs, score_rs = random_local_search(best_overall, rng, n_iter=min(2000, int(remaining * 200)))
            if np.isfinite(score_rs) and score_rs < best_score_overall:
                best_score_overall = score_rs
                best_overall = f_rs.copy()
                print(f"  After RS: {score_rs:.6f} (new best!)")
    
    # Final phase: deeper optimization on the best sequence found
    remaining = deadline - time.monotonic()
    if best_overall is not None and remaining > 20:
        n_best = len(best_overall)
        print(f"\nFinal deep optimization on best (n={n_best}), time={remaining:.0f}s")
        
        # LP polish with more time
        f_deep, score_deep = lp_polish(
            best_overall, remaining * 0.7,
            t0=0.3, rng=rng, verbose=True
        )
        if np.isfinite(score_deep) and score_deep < best_score_overall:
            best_score_overall = score_deep
            best_overall = f_deep.copy()
            print(f"  Deep LP: {score_deep:.6f}")
        
        # Random search
        remaining2 = deadline - time.monotonic()
        if remaining2 > 5:
            f_rs, score_rs = random_local_search(
                best_overall, rng, n_iter=min(5000, int(remaining2 * 300))
            )
            if np.isfinite(score_rs) and score_rs < best_score_overall:
                best_score_overall = score_rs
                best_overall = f_rs.copy()
                print(f"  Final RS: {score_rs:.6f}")
    
    # Verify and return
    if best_overall is not None:
        best_overall = np.clip(best_overall, 0, 1000)
        final_list = best_overall.tolist()
        final_score = evaluate_sequence(final_list)
        print(f"\n=== FINAL ===")
        print(f"Score: {final_score:.6f}")
        print(f"Length: {len(final_list)}")
        print(f"Target: 1.5027")
        print(f"Previous: 1.515592")
        if final_score < 1.5027:
            print("TARGET ACHIEVED!")
        elif final_score < 1.515592:
            print("IMPROVED over previous!")
        return final_list
    
    # Fallback
    print("Fallback to uniform sequence")
    return [1.0] * 256
