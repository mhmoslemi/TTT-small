"""
Minimizes 2*n*max(conv(a,a))/sum(a)^2 using smooth L-BFGS optimization
with softmax parameterization, multi-start strategy, and subgradient polishing.
"""

import numpy as np
from scipy.optimize import minimize, linprog
import time

def softmax(z):
    z = z - np.max(z)
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300:
        return np.ones(len(z)) / len(z)
    return e / s

def full_convolve_fft(a, b):
    """Compute full linear convolution using FFT."""
    n = len(a)
    m = len(b)
    size = n + m - 1
    N = 1 << (size - 1).bit_length()
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    result = np.fft.irfft(fa * fb, N)[:size]
    result[result < 0] = 0.0
    return result

def compute_obj_grad(z, temperature):
    """Compute smooth objective and gradient w.r.t. z (in softmax parameterization)."""
    n = len(z)
    p = softmax(z)
    conv = full_convolve_fft(p, p)
    
    # Smooth max via log-sum-exp
    v = conv / temperature
    v_max = np.max(v)
    log_sum_exp = v_max + np.log(np.sum(np.exp(v - v_max)))
    obj = temperature * log_sum_exp
    
    # Weights proportional to exp(conv[k]/T)
    weights = np.exp(v - v_max)
    weights /= weights.sum()
    
    # grad_p[i] = sum_k weights[k] * d(conv[k])/d(p[i]) = sum_k weights[k] * p[k-i] (if valid)
    # = sum_j weights[i+j] * p[j] for j in range where i+j < 2n-1
    grad_p = np.zeros(n)
    for i in range(n):
        # j ranges from 0 to n-1, and i+j ranges from i to i+n-1 (which is < 2n-1)
        grad_p[i] = np.dot(weights[i:i+n], p)
    
    # Chain rule: grad_z = p * (grad_p - <grad_p, p>)
    grad_z = p * (grad_p - np.dot(grad_p, p))
    
    return obj, grad_z

def optimize_with_lbfgs(n, init_z, deadline, rng):
    """Optimize using L-BFGS with adaptive temperature."""
    best_p = softmax(init_z)
    best_obj = np.max(full_convolve_fft(best_p, best_p))
    
    # Adaptive temperature: start high, decrease
    temps = [1.0, 0.5, 0.2, 0.1, 0.05, 0.02, 0.01, 0.005]
    
    z = init_z.copy()
    for T in temps:
        if time.time() > deadline - 20:
            break
        sub_deadline = min(deadline - 10, time.time() + max(5, (deadline - time.time()) / (len(temps) + 1)))
        
        def fun(z_inner):
            obj, grad = compute_obj_grad(z_inner, T)
            return obj, grad
        
        try:
            res = minimize(fun, z, method='L-BFGS-B', 
                          options={'maxiter': 500, 'gtol': 1e-8, 'maxfun': 2000})
            p_new = softmax(res.x)
            conv_new = full_convolve_fft(p_new, p_new)
            obj_new = np.max(conv_new)
            
            if obj_new < best_obj:
                best_obj = obj_new
                best_p = p_new.copy()
                z = res.x.copy()
        except Exception:
            pass
    
    return best_p, 2 * n * best_obj

def subgradient_polish(p, n, deadline, rng, max_iters=100000):
    """Polish using projected subgradient descent with momentum."""
    p = p.copy()
    p = np.maximum(p, 0)
    s = p.sum()
    if s < 1e-10:
        p = np.ones(n) / n
        s = 1.0
    p = p / s
    
    best_p = p.copy()
    best_max = np.max(full_convolve_fft(p, p))
    
    momentum = np.zeros(n)
    beta = 0.9
    lr = 0.1 / np.sqrt(n)
    stall = 0
    it = 0
    
    while time.time() < deadline and it < max_iters:
        it += 1
        conv = full_convolve_fft(p, p)
        peak = np.max(conv)
        
        if peak < best_max - 1e-12:
            best_max = peak
            best_p = p.copy()
            stall = 0
        else:
            stall += 1
        
        if stall > 500:
            lr = max(lr * 0.85, 1e-10)
            stall = 0
            if lr < 1e-8:
                break
        
        # Get active indices (top peaks)
        threshold = peak * (1 - 1e-4)
        active = np.where(conv >= threshold)[0]
        if len(active) > 20:
            top = np.argsort(conv[active])[::-1][:10]
            active = active[top]
        
        # Subgradient: grad_p[i] = mean over active k of p[k-i]
        grad_p = np.zeros(n)
        for k in active:
            lo = max(0, k - n + 1)
            hi = min(n - 1, k)
            grad_p[lo:hi+1] += p[k-hi:k-lo+1][::-1]
        grad_p /= len(active)
        
        # Nesterov-like momentum step
        momentum = beta * momentum + (1 - beta) * grad_p
        p_new = p - lr * momentum
        p = np.maximum(p_new, 0)
        s = p.sum()
        if s < 1e-10:
            p = best_p.copy()
        else:
            p = p / s
    
    return best_p

    inits = {}

    # Two-level (random set)
    mask = rng.random(n) > 0.5
    inits['two_level'] = np.where(mask, 1.0, 0.3)
    
    # Periodic structure
    period = max(2, n // 4)
    base = rng.uniform(0.5, 1.5, period)
    inits['periodic'] = np.tile(base, n // period + 1)[:n]
    
    return inits

"""
Minimizes 2*n*max(conv(a,a))/sum(a)^2 using smooth L-BFGS with softmax
parameterization, adaptive temperature annealing, FFT-accelerated gradients,
and subgradient polishing. Multi-start over sequence lengths and templates.
"""

import numpy as np
from scipy.optimize import minimize
import time


def softmax(z):
    """Numerically stable softmax."""
    z = z - np.max(z)
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300:
        return np.ones(len(z)) / len(z)
    return e / s


def full_convolve_fft(a, b):
    """Compute full linear convolution using FFT."""
    n = len(a)
    m = len(b)
    size = n + m - 1
    N = 1 << (size - 1).bit_length()
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    result = np.fft.irfft(fa * fb, N)[:size]
    result[result < 0] = 0.0
    return result


def compute_smooth_obj_grad(z, temperature):
    """
    Compute smooth log-sum-exp objective and gradient w.r.t. z.
    p = softmax(z), objective = T * logsumexp(conv(p,p)/T)
    Gradient computed via FFT-accelerated correlation.
    """
    n = len(z)
    p = softmax(z)
    conv = full_convolve_fft(p, p)

    # Log-sum-exp for smooth max
    v = conv / temperature
    v_max = np.max(v)
    log_sum_exp = v_max + np.log(np.sum(np.exp(v - v_max)))
    obj = temperature * log_sum_exp

    # Softmax weights over convolution values
    weights = np.exp(v - v_max)
    weights /= weights.sum()

    # grad_p[i] = sum_j weights[i+j] * p[j]
    # Computed via: w_rev = weights[::-1], conv(w_rev, p)[n-1:2n-1][::-1]
    w_rev = weights[::-1]
    conv_result = full_convolve_fft(w_rev, p)
    grad_p = conv_result[n - 1:2 * n - 1][::-1]

    # Chain rule through softmax: grad_z = p * (grad_p - <grad_p, p>)
    mean_grad = np.dot(grad_p, p)
    grad_z = p * (grad_p - mean_grad)

    return obj, grad_z


def optimize_lbfgs(n, z_init, deadline, max_temps=8):
    """
    L-BFGS optimization with geometric temperature decay.
    Returns best probability vector found.
    """
    best_p = softmax(z_init)
    best_max = np.max(full_convolve_fft(best_p, best_p))
    z = z_init.copy()

    # Geometric temperature schedule
    temps = 1.0 * np.array([1.0, 0.5, 0.25, 0.12, 0.06, 0.03, 0.015, 0.008, 0.004, 0.002])
    temps = temps[:max_temps]

    for idx, T in enumerate(temps):
        if time.time() > deadline - 15:
            break

        def fun(z_inner):
            obj, grad = compute_smooth_obj_grad(z_inner, T)
            # Add tiny regularization for numerical stability
            grad = grad + 1e-15 * z_inner
            return obj, grad

        try:
            max_iter = max(100, int(2000 * (1.0 - 0.5 * idx / max_temps)))
            res = minimize(fun, z, method='L-BFGS-B',
                           options={'maxiter': max_iter, 'gtol': 1e-10,
                                    'maxfun': max_iter * 3, 'ftol': 1e-12})
            p_new = softmax(res.x)
            conv_new = full_convolve_fft(p_new, p_new)
            obj_new = np.max(conv_new)

            if obj_new < best_max:
                best_max = obj_new
                best_p = p_new.copy()
                z = res.x.copy()
        except Exception:
            pass

    return best_p, 2.0 * n * best_max


def subgradient_polish(p, n, deadline, rng, max_iters=200000):
    """
    Projected subgradient descent with Nesterov momentum on the simplex.
    Minimizes max(conv(p,p)) subject to p >= 0, sum(p) = 1.
    """
    p = np.maximum(p, 0.0).astype(np.float64)
    s = p.sum()
    if s < 1e-10:
        p = np.ones(n) / n
    else:
        p = p / s

    best_p = p.copy()
    best_max = np.max(full_convolve_fft(p, p))

    beta = 0.85
    momentum = np.zeros(n)
    lr = 0.3 / np.sqrt(n)
    lr_min = 1e-12
    stall = 0
    stall_limit = 400
    it = 0

    while time.time() < deadline and it < max_iters:
        it += 1
        conv = full_convolve_fft(p, p)
        peak = np.max(conv)

        if peak < best_max - 1e-13:
            best_max = peak
            best_p = p.copy()
            stall = 0
        else:
            stall += 1

        if stall >= stall_limit:
            lr = max(lr * 0.8, lr_min)
            stall = 0
            if lr < lr_min * 10:
                break
            # Perturb to escape local minimum
            if time.time() < deadline - 30:
                p = p + rng.normal(0, 0.01 / np.sqrt(n), n)
                p = np.maximum(p, 0)
                p = p / p.sum()

        # Active set: top peaks
        threshold = peak * (1.0 - 5e-5)
        active = np.where(conv >= threshold)[0]
        if len(active) > 15:
            top_idx = np.argsort(conv[active])[::-1][:8]
            active = active[top_idx]

        # Subgradient: average of p[k-i] over active k
        grad_p = np.zeros(n)
        for k in active:
            lo = max(0, k - n + 1)
            hi = min(n - 1, k)
            grad_p[lo:hi + 1] += p[k - hi:k - lo + 1][::-1]
        grad_p /= max(1, len(active))

        # Momentum step
        momentum = beta * momentum + (1.0 - beta) * grad_p
        p_new = p - lr * momentum
        p = np.maximum(p_new, 0)
        s = p.sum()
        if s < 1e-10:
            p = best_p.copy()
        else:
            p = p / s

    return best_p


def local_search(p, n, deadline, rng, max_iters=50000):
    """
    Pair-swap local search: try swapping mass between pairs of positions.
    """
    p = p.copy()
    best_max = np.max(full_convolve_fft(p, p))
    it = 0
    no_improve = 0

    while time.time() < deadline - 5 and it < max_iters and no_improve < 500:
        it += 1
        i = rng.integers(0, n)
        j = rng.integers(0, n)
        if i == j:
            continue

        # Try moving delta from i to j
        delta = (p[i] - p[j]) * 0.1 * rng.choice([-1, 1])
        delta = max(-p[i] * 0.5, min(p[j] * 0.5, delta))
        if abs(delta) < 1e-15:
            continue

        p_new = p.copy()
        p_new[i] -= delta
        p_new[j] += delta

        if p_new[i] < 0 or p_new[j] < 0:
            continue

        conv_new = full_convolve_fft(p_new, p_new)
        max_new = np.max(conv_new)

        if max_new < best_max - 1e-14:
            p = p_new
            best_max = max_new
            no_improve = 0
        else:
            no_improve += 1

    return p


def generate_inits(n, rng, parent_seq=None):
    """Generate diverse initializations."""
    inits = {}

    # Uniform with small noise
    inits['uniform'] = np.ones(n) * (1 + rng.normal(0, 0.005, n))

    # Cosine window
    x = np.linspace(0, 1, n)
    inits['cosine'] = 0.5 * (1 + np.cos(2 * np.pi * x)) + 0.05

    # Triangular
    inits['triangular'] = 1.0 - np.abs(2 * x - 1) + 0.05

    # Gaussian
    inits['gaussian'] = np.exp(-2.0 * ((np.arange(n) - n / 2) / (n / 3.0)) ** 2) + 0.01

    # Two-level random
    mask = rng.random(n) > 0.5
    inits['two_level'] = np.where(mask, 1.0 + rng.uniform(0, 0.2, n),
                                  0.3 + rng.uniform(0, 0.1, n))

    # Periodic
    period = max(2, n // 5)
    base = 0.5 + 0.5 * rng.random(period)
    inits['periodic'] = np.tile(base, n // period + 1)[:n]

    # Sinc-like
    t = (np.arange(n) - n / 2) / (n / 6)
    with np.errstate(divide='ignore', invalid='ignore'):
        sinc = np.where(np.abs(t) < 1e-10, 1.0, np.sin(np.pi * t) / (np.pi * t))
    inits['sinc'] = np.abs(sinc) + 0.01

    # Parent sequence (if available and matches length)
    if parent_seq is not None:
        try:
            parent = np.array(parent_seq, dtype=np.float64)
            parent = np.clip(parent, 0, 1000)
            if len(parent) == n:
                inits['parent'] = parent + 1e-10
            elif len(parent) > n:
                inits['parent'] = parent[:n] + 1e-10
            else:
                ext = np.zeros(n)
                ext[:len(parent)] = parent
                inits['parent'] = ext + 1e-10
        except Exception:
            pass

    # Normalize all
    for key in inits:
        inits[key] = np.maximum(inits[key], 0.0)
        s = inits[key].sum()
        if s > 0:
            inits[key] = inits[key] / s
        else:
            inits[key] = np.ones(n) / n

    return inits

    try:

    # === PHASE 2: Explore other n values ===
