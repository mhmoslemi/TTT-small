    for k in range(2 * n - 1):

        j_end = min(n, k + 1)
        for j in range(j_start, j_end):
            A[k, j] = f[k - j]
    return A


def compute_score(f):
    """Compute score for normalized sequence (sum=1). Returns 2*n*max(conv)."""
    n = len(f)
    conv = np.convolve(f, f)
    return 2.0 * n * conv.max()


def proj_simplex(v, s=1.0):
    """Project v onto simplex {x : sum(x)=s, x>=0}."""
    n = len(v)
    # Use the algorithm from Duchi et al. 2011
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - s))[0][-1]
    theta = (cssv[rho] - s) / (rho + 1.0)
    return np.maximum(v - theta, 0)


def subgradient_step(f, lr):
    """One projected subgradient step for minimizing max(convolve(f,f)) subject to sum(f)=1, f>=0."""
    n = len(f)
    conv = np.convolve(f, f)
    k_star = int(np.argmax(conv))
    # Subgradient: grad[j] = f[k_star - j] if 0 <= k_star - j < n
    grad = np.zeros(n)
    for j in range(n):
        idx = k_star - j
        if 0 <= idx < n:
            grad[j] = f[idx]
        # Factor of 2 for the "diagonal" term
        if k_star == 2 * j:
            grad[j] += f[j]
    f_new = f - lr * grad
    f_new = proj_simplex(f_new, 1.0)
    return f_new


def subgradient_polish(f, rng, n_iter=500, time_budget=30.0):
    """Polish using projected subgradient descent with backtracking."""
    n = len(f)
    f = f.copy()
    best_f = f.copy()
    best_score = compute_score(f)
    lr = 0.1
    start = time.monotonic()
    
    for it in range(n_iter):
        if time.monotonic() - start > time_budget:
            break
        f_new = subgradient_step(f, lr)
        score = compute_score(f_new)
        if score < best_score:
            best_f = f_new.copy()
            best_score = score
            f = f_new
            lr = min(lr * 1.05, 0.5)  # increase step size slightly
        else:
            lr *= 0.8
            # Occasionally try a random perturbation
            if it % 50 == 0:
                f_pert = f + rng.normal(0, lr * 0.1, n)
                f_pert = proj_simplex(f_pert, 1.0)
                if compute_score(f_pert) < best_score:
                    f = f_pert
                    best_f = f.copy()
                    best_score = compute_score(f)
    return best_f, best_score


def local_search(f, rng, n_iter=2000, time_budget=20.0):
    """Block perturbation local search."""
    n = len(f)
    f = f.copy()
    best = f.copy()
    best_score = compute_score(f)
    start = time.monotonic()
    
    for it in range(n_iter):
        if time.monotonic() - start > time_budget:
            break
        f_candidate = f.copy()
        width = max(1, int(rng.integers(1, max(2, n // 16))))

"""
Improved LP-based iterative optimization with subgradient polishing,
adaptive constraint tightening, and multi-length search for minimizing
2*n*max(convolve(a,a))/sum(a)^2. Uses warm-starting, line search on blend
parameter, and projected subgradient descent to escape local minima.
"""

import numpy as np
from scipy.optimize import linprog
import time


def build_conv_matrix(f):
    """Build matrix A such that A @ b = np.convolve(f, b, mode='full')."""
    n = len(f)
    A = np.zeros((2 * n - 1, n))
    for k in range(2 * n - 1):
        j_min = max(0, k - n + 1)
        j_max = min(n - 1, k)
        if j_min <= j_max:
            A[k, j_min:j_max + 1] = f[k - j_min:k - j_max - 1:-1]
    return A


def compute_score(f):
    """Compute score for normalized sequence (sum=1). Returns 2*n*max(conv)."""
    n = len(f)
    conv = np.convolve(f, f)
    return 2.0 * n * conv.max()


def proj_simplex(v, s=1.0):
    """Project v onto simplex {x : sum(x)=s, x>=0}."""
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho_candidates = np.nonzero(u * np.arange(1, n + 1) > (cssv - s))[0]
    if len(rho_candidates) == 0:
        return np.full(n, s / n)
    rho = rho_candidates[-1]
    theta = (cssv[rho] - s) / (rho + 1.0)
    return np.maximum(v - theta, 0)


def subgradient_polish(f, rng, time_budget=30.0):
    """Polish using projected subgradient descent with adaptive step size."""
    n = len(f)
    f = f.copy()
    best_f = f.copy()
    best_score = compute_score(f)
    lr = 0.05
    start = time.monotonic()
    it = 0
    stall = 0

    while time.monotonic() - start < time_budget:
        conv = np.convolve(f, f)
        k_star = int(np.argmax(conv))
        grad = np.zeros(n)
        for j in range(n):
            idx = k_star - j
            if 0 <= idx < n:
                grad[j] = f[idx]
                if k_star == 2 * j:
                    grad[j] += f[j]
        f_new = f - lr * grad
        f_new = proj_simplex(f_new, 1.0)
        score = compute_score(f_new)
        if score < best_score - 1e-10:
            best_f = f_new.copy()
            best_score = score
            f = f_new
            lr = min(lr * 1.02, 0.3)
            stall = 0
        else:
            lr *= 0.85
            stall += 1
            if stall > 30:
                f_pert = f + rng.normal(0, lr * 0.5, n)
                f_pert = proj_simplex(f_pert, 1.0)
                if compute_score(f_pert) < best_score:
                    f = f_pert
                    best_f = f.copy()
                    best_score = compute_score(f)
                    lr = 0.05
                    stall = 0
                elif time.monotonic() - start > time_budget * 0.8:
                    break
        it += 1
    return best_f, best_score

        f_candidate = f.copy()
        width = max(1, int(rng.integers(1, max(2, n // 16))))
        idx = int(rng.integers(0, n - width))
        scale = np.exp(rng.normal(0, 0.15))
        f_candidate[idx:idx + width] *= scale
        f_candidate = np.clip(f_candidate, 0, None)
        S = f_candidate.sum()
        if S < 1e-10:
            it += 1
            continue
        f_candidate /= S
        score = compute_score(f_candidate)
        if score < best_score:
            best = f_candidate.copy()
            best_score = score
            f = f_candidate
        it += 1
    return best, best_score


def optimize_lp(f, time_budget, rng, verbose=False):
    """LP-based optimization with adaptive M tightening and line search."""
    n = len(f)
    f = f.astype(np.float64).copy()
    S = f.sum()
    if S < 1e-10:
        return f, np.inf
    f = f / S

    best_f = f.copy()
    best_score = compute_score(f)
    t = 0.3
    start = time.monotonic()
    it = 0
    stall_count = 0
    squeeze = 0.998

    while time.monotonic() - start < time_budget:
        conv = np.convolve(f, f)
        M_current = conv.max()

        # Try tighter constraint first
        M_target = M_current * squeeze

        A = build_conv_matrix(f)
        c_obj = -np.ones(n)
        bounds = [(0, None)] * n

        lp_done = False
        res = None

        # Try tight constraint
        try:
            res = linprog(c_obj, A_ub=A, b_ub=np.full(2 * n - 1, M_target),
                         bounds=bounds, method='highs',
                         options={'presolve': True, 'time_limit': max(5.0, time_budget * 0.05)})
            if res.success:
                lp_done = True
        except Exception:
            pass

        # Fall back to current M if tight was infeasible or failed
        if not lp_done:
            try:
                res = linprog(c_obj, A_ub=A, b_ub=np.full(2 * n - 1, M_current * 1.001),
                             bounds=bounds, method='highs',
                             options={'presolve': True, 'time_limit': max(5.0, time_budget * 0.05)})
                if res.success:
                    lp_done = True
                    # Relax squeeze since we couldn't tighten
                    squeeze = min(0.9995, squeeze * 1.001)
            except Exception:
                break

        if not lp_done or res.x is None:
            stall_count += 5
            if stall_count > 40:
                break
            continue

        b = np.clip(res.x, 0, None)
        b_sum = b.sum()
        if b_sum < 1e-15:
            break
        g = b / b_sum

        # Line search over t values
        best_t = 0.0
        best_t_score = best_score
        t_candidates = [t, t * 1.5, t * 0.5, 0.1, 0.2, 0.3, 0.5, 0.7, 0.9]
        for t_try in t_candidates:
            if t_try <= 0 or t_try > 0.95:
                continue
            f_new = (1 - t_try) * f + t_try * g
            f_new = np.clip(f_new, 0, None)
            S_new = f_new.sum()
            if S_new < 1e-10:
                continue
            f_new /= S_new
            sc = compute_score(f_new)
            if sc < best_t_score:
                best_t_score = sc
                best_t = t_try

        if best_t > 0 and best_t_score < best_score - 1e-12:
            f_new = (1 - best_t) * f + best_t * g
            f_new = np.clip(f_new, 0, None)
            f_new /= f_new.sum()
            best_f = f_new.copy()
            best_score = best_t_score
            f = f_new
            t = best_t
            stall_count = 0
            if verbose and it % 20 == 0:
                elapsed = time.monotonic() - start
                print(f"      it={it}, score={best_score:.6f}, t={t:.4f}, el={elapsed:.0f}s")
        else:
            t *= 0.75
            stall_count += 1
            if stall_count > 30:
                f_pert = f + rng.normal(0, 0.03, n)
                f_pert = np.clip(f_pert, 0, None)
                Sp = f_pert.sum()
                if Sp > 1e-10:
                    f_pert /= Sp
                    if compute_score(f_pert) < best_score:
                        f = f_pert
                        best_f = f.copy()
                        best_score = compute_score(f)
                        t = 0.3
                        stall_count = 0
                if time.monotonic() - start > time_budget * 0.85:
                    break

        it += 1

    return best_f, best_score


def make_init(n, stype, rng):
    """Create initial sequence."""
    if stype == 'uniform':
        f = np.ones(n) * (1.0 + 0.05 * rng.normal(0, 1, n))
    elif stype == 'ramp':
        f = np.arange(1, n + 1, dtype=np.float64)
    elif stype == 'sqrt_ramp':
        f = np.sqrt(np.arange(1, n + 1, dtype=np.float64))
    elif stype == 'gaussian':
        x = np.arange(n, dtype=np.float64)
        f = np.exp(-0.5 * ((x - n * 0.5) / (n * 0.25)) ** 2)
    elif stype == 'cosine':
        x = np.arange(n, dtype=np.float64)
        f = np.cos(np.pi * x / (2.0 * n)) ** 2 + 0.01
    elif stype == 'bimodal':
        x = np.arange(n, dtype=np.float64)
        f = (np.exp(-0.5 * ((x - n * 0.25) / (n * 0.08)) ** 2) +
             np.exp(-0.5 * ((x - n * 0.75) / (n * 0.08)) ** 2))
    elif stype == 'random':
        f = rng.random(n) + 0.01
    else:
        f = np.ones(n)
    return np.clip(f, 0, None)


def extend_sequence(f, n_new):
    """Extend sequence f to length n_new by interpolation + random perturbation."""
    n_old = len(f)
    if n_new <= n_old:
        return f[:n_new].copy()
    x_old = np.linspace(0, 1, n_old)
    x_new = np.linspace(0, 1, n_new)
    f_ext = np.interp(x_new, x_old, f)
    return f_ext


def propose_candidate(seed=42, budget_s=1500.0, **kwargs):
    """Main search: multi-length LP optimization with subgradient and local search polish."""
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + budget_s - 15.0

    best_overall = None
    best_score_overall = np.inf

    # Get parent sequence
    parent = None
    try:
        parent = np.array(height_sequence_1, dtype=np.float64)
        if parent.ndim != 1 or len(parent) == 0 or not np.isfinite(parent).all():
            parent = None
    except Exception:
        parent = None

    if parent is not None:
        parent = np.clip(parent, 0, 1000)
        if parent.sum() > 0.01:
            parent = parent / parent.sum()
            print(f"Parent: n={len(parent)}, score={compute_score(parent):.6f}")
            if compute_score(parent) < best_score_overall:
                best_score_overall = compute_score(parent)
                best_overall = parent.copy()

    # Build configuration list
    configs = []
    if parent is not None:
        configs.append((f"parent_{len(parent)}", len(parent), 'parent', 0.25))

    for n in [256, 512, 1024, 2048]:
        for st in ['gaussian', 'uniform', 'cosine']:
            configs.append((f"{st}_{n}", n, st, 0.00))

    for n in [512, 1024]:
        configs.append((f"bimodal_{n}", n, 'bimodal', 0.00))
        configs.append((f"ramp_{n}", n, 'ramp', 0.00))

    configs.append(("random_1024", 1024, 'random', 0.00))

    total_time = budget_s - 30.0
    print(f"Configs: {len(configs)}, total_time={total_time:.0f}s, target=1.5027")

    # Allocate time: parent gets priority, rest share equally
    parent_time = 0.25 * total_time if parent is not None else 0
    n_other = len([c for c in configs if c[2] != 'parent'])
    time_other = max(10.0, (total_time - parent_time) / max(1, n_other))

    for name, n, stype, _ in configs:
        if time.monotonic() > deadline - 20:
            break

        if stype == 'parent':
            f0 = parent.copy()
            t_alloc = min(parent_time, deadline - time.monotonic() - 15)
        else:
            f0 = make_init(n, stype, rng)
            S0 = f0.sum()
            if S0 < 1e-10:
                continue
            f0 = f0 / S0
            t_alloc = min(time_other, deadline - time.monotonic() - 15)

        if t_alloc < 8:
            continue

        init_score = compute_score(f0)
        print(f"\n[{name}] n={n}, t={t_alloc:.0f}s, init_score={init_score:.6f}")

        # LP phase (60% of allocated time)
        lp_time = t_alloc * 0.55
        f_opt, score = optimize_lp(f0, lp_time, rng, verbose=False)

        # Subgradient phase
        rem = deadline - time.monotonic() - 15
        if rem > 8 and np.isfinite(score):
            sg_t = min(15.0, t_alloc * 0.2, rem)
            if sg_t > 4:
                f_sg, sc_sg = subgradient_polish(f_opt, rng, time_budget=sg_t)
                if sc_sg < score:
                    f_opt = f_sg
                    score = sc_sg

        # Local search phase
        rem = deadline - time.monotonic() - 15
        if rem > 5 and np.isfinite(score):
            ls_t = min(12.0, rem)
            if ls_t > 3:
                f_ls, sc_ls = local_search(f_opt, rng, time_budget=ls_t)
                if sc_ls < score:
                    f_opt = f_ls
                    score = sc_ls

        if np.isfinite(score) and score < best_score_overall:
            best_score_overall = score
            best_overall = f_opt.copy()
            print(f"  ** New best: {score:.6f} (n={n}, {name}) **")

    # Extension strategy: try doubling the best sequence length
    rem = deadline - time.monotonic()
    if best_overall is not None and rem > 40:
        n_best = len(best_overall)
        n_ext = min(n_best * 2, 4096)
        if n_ext > n_best:
            print(f"\nExtending best from n={n_best} to n={n_ext}")
            f_ext = extend_sequence(best_overall, n_ext)
            f_ext = f_ext + rng.normal(0, 0.02, n_ext)
            f_ext = np.clip(f_ext, 0, None)
            Se = f_ext.sum()
            if Se > 1e-10:
                f_ext /= Se
                t_ext = min(rem * 0.5, deadline - time.monotonic() - 20)
                if t_ext > 10:
                    f_e, sc_e = optimize_lp(f_ext, t_ext * 0.6, rng, verbose=False)
                    rem2 = deadline - time.monotonic() - 15
                    if rem2 > 5:
                        f_e, sc_e2 = local_search(f_e, rng, time_budget=min(10, rem2))
                        sc_e = min(sc_e, sc_e2)
                    if np.isfinite(sc_e) and sc_e < best_score_overall:
                        best_score_overall = sc_e
                        best_overall = f_e
                        print(f"  ** Extended best: {sc_e:.6f} (n={n_ext}) **")

    # Final intensive polish on best
    rem = deadline - time.monotonic()
    if best_overall is not None and rem > 20:
        print(f"\nFinal polish, time={rem:.0f}s")
        n_f = len(best_overall)
        f_p, sc_p = optimize_lp(best_overall, rem * 0.4, rng, verbose=False)
        if np.isfinite(sc_p) and sc_p < best_score_overall:
            best_overall = f_p
            best_score_overall = sc_p
            print(f"  LP: {sc_p:.6f}")

        rem2 = deadline - time.monotonic()
        if rem2 > 10:
            f_s, sc_s = subgradient_polish(best_overall, rng, time_budget=min(15, rem2 * 0.4))
            if np.isfinite(sc_s) and sc_s < best_score_overall:
                best_overall = f_s
                best_score_overall = sc_s
                print(f"  SG: {sc_s:.6f}")

        rem3 = deadline - time.monotonic()
        if rem3 > 8:
            f_l, sc_l = local_search(best_overall, rng, time_budget=min(20, rem3 * 0.5))
            if np.isfinite(sc_l) and sc_l < best_score_overall:
                best_overall = f_l
                best_score_overall = sc_l
                print(f"  LS: {sc_l:.6f}")

    # Verify and return
    if best_overall is not None:
        final_list = np.clip(best_overall, 0, 1000).tolist()
        final_score = evaluate_sequence(final_list)
        print(f"\nFinal verified score: {final_score:.6f}")
        print(f"Sequence length: {len(final_list)}")
        return final_list

    print("Fallback to uniform")
    return [1.0] * 256
