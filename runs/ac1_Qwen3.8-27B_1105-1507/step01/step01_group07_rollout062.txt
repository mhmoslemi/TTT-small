"""
Coordinate descent + LP hybrid optimization for minimizing
2*n*max(conv(a,a))/(sum(a))^2 over non-negative sequences.
Uses exact 1D coordinate optimization with upper envelope computation,
combined with LP-based moves for escaping local optima.

import numpy as np
from scipy.optimize import linprog
import time


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    conv = np.convolve(a, a)
    max_b = float(np.max(conv))
    sum_a = float(np.sum(a))
    if sum_a < 0.01:
        return np.inf
    return 2.0 * n * max_b / (sum_a * sum_a)


def compute_upper_envelope(base_k, w):
    """
    Compute upper envelope of lines y = base_k[j] + w[j]*x.
    Returns breakpoints and active line indices.
    """
    # Filter out zero-slope lines (they're just constants)
    n_lines = len(w)
    if n_lines == 0:
        return np.array([]), np.array([], dtype=int)
    
    # Sort by slope, then by intercept (for same slope, keep highest intercept)
    order = np.lexsort((base_k, w))
    w_sorted = w[order]
    b_sorted = base_k[order]
    
    # Remove duplicates: for same slope, keep only the one with highest intercept
    unique_mask = np.concatenate([[True], np.diff(w_sorted) > 1e-15])
    # Actually, we need to keep the max intercept for each unique slope
    # Since sorted by slope then intercept ( ascending), the last one for each slope group is the max
    # Let me redo this
    unique_slopes = []
    unique_intercepts = []
    unique_indices = []
    
    i = 0
    while i < len(w_sorted):
        j = i
        while j < len(w_sorted) and abs(w_sorted[j] - w_sorted[i]) < 1e-15:
            j += 1
        # Group i to j-1 has same slope
        max_idx = i + np.argmax(b_sorted[i:j])
        unique_slopes.append(w_sorted[max_idx])
        unique_intercepts.append(b_sorted[max_idx])
        unique_indices.append(order[max_idx])
        i = j
    
    m = len(unique_slopes)
    if m == 0:
        return np.array([]), np.array([], dtype=int)
    
    if m == 1:
        return np.array([0.0]), np.array([unique_indices[0]], dtype=int)
    
    # Monotone chain for upper envelope
    # Lines sorted by slope ascending
    # A line j is on the upper envelope if its intersection with the previous
    # envelope line is to the right of the intersection of the previous two
    
    hull = []  # list of (breakpoint, line_idx)
    
    for idx in range(m):
        s = unique_slopes[idx]
        b = unique_intercepts[idx]
        
        if len(hull) == 0:
            hull.append((0.0, idx))
        else:
            # Find intersection with last line in hull
            s_prev = unique_slopes[hull[-1][1]]
            b_prev = unique_intercepts[hull[-1][1]]
            if abs(s - s_prev) < 1e-15:
                # Same slope, new line must have higher intercept to matter
                if b > b_prev:
                    hull[-1] = (hull[-1][0], idx)
                continue
            
            # Intersection: s_prev * x + b_prev = s * x + b
            # x = (b_prev - b) / (s - s_prev)
            x_int = (b_prev - b) / (s - s_prev)
            
            while len(hull) >= 2 and x_int <= hull[-1][0] + 1e-15:
                hull.pop()
                s_prev = unique_slopes[hull[-1][1]]
                b_prev = unique_intercepts[hull[-1][1]]
                if abs(s - s_prev) < 1e-15:
                    break
                x_int = (b_prev - b) / (s - s_prev)
            
            if len(hull) == 1:
                s_prev = unique_slopes[hull[-1][1]]
                b_prev = unique_intercepts[hull[-1][1]]
                if x_int > hull[-1][0]:
                    hull.append((x_int, idx))
            else:
                if x_int > hull[-1][0] + 1e-15:
                    hull.append((x_int, idx))
    
    # Build breakpoints array
    breakpoints = np.array([h[0] for h in hull])
    active = np.array([h[1] for h in hull], dtype=int)
    
    # Map back to original indices
    active_orig = np.array([unique_indices[a] for a in active], dtype=int)
    
    return breakpoints, active_orig


def coord_descent_pass(a, conv, coord_order=None):
    """One pass of coordinate descent. Returns updated a, conv, and improvement flag."""
    n = len(a)
    S = float(np.sum(a))
    a_new = a.copy()
    conv_new = conv.copy()
    S_new = S
    improved = False
    
    if coord_order is None:
        coord_order = list(range(n))
    
    for i in coord_order:
        old = a_new[i]
        S_other = S_new - old
        
        k_min = i
        k_max = min(i + n - 1, 2 * n - 2)
        k_range_len = k_max - k_min + 1
        
        j_vals = np.arange(k_range_len)  # 0 to k_range_len-1, representing k-i
        k_vals = k_min + j_vals
        
        # Compute base (conv without a[i]'s contribution)
        base_k = conv_new[k_vals].copy()
        
        # Contribution of a[i]:
        # For j != i: 2*old*a[j]
        # For j == i: old^2
        contrib = 2.0 * old * a_new[j_vals]
        if i < k_range_len:
            contrib[i] = old * old
        
        base_k -= contrib
        
        # Linear coefficients (w[j] = 2*a[j] for j != i, 0 for j == i)
        w = 2.0 * a_new[j_vals].copy()
        if i < k_range_len:
            quad_base = base_k[i]
            w[i] = 0.0
        else:
            quad_base = None
        
        # Max of unaffected parts
        max_const = 0.0
        if k_min > 0:
            max_const = max(max_const, float(np.max(conv_new[:k_min])))
        if k_max + 1 < len(conv_new):
            max_const = max(max_const, float(np.max(conv_new[k_max + 1:])))
        
        # Find optimal x
        candidates = [0.0, min(old, 1000.0), 1000.0]
        
        # From linear pieces
        valid = w > 1e-15
        if np.any(valid):
            x_cands = S_other - 2.0 * base_k[valid] / w[valid]
            x_cands = x_cands[(x_cands > 1e-10) & (x_cands < 999.99)]
            candidates.extend(x_cands.tolist())
        
        # From quadratic piece
        if quad_base is not None and S_other > 1e-15:
            xq = quad_base / S_other
            if xq > 1e-10 and xq < 999.99:
                candidates.append(xq)
        
        # Evaluate at candidates
        best_x = old
        best_f = 2.0 * n * float(np.max(conv_new)) / (S_new * S_new)
        
        for x in candidates:
            x = max(0.0, min(1000.0, x))
            new_sum = S_other + x
            if new_sum < 0.01:
                continue
            
            # max_c = max(max_const, max(base_k + w*x), quad_base + x^2)
            lin_vals = base_k + w * x
            max_aff = float(np.max(lin_vals))
            if quad_base is not None:
                max_aff = max(max_aff, quad_base + x * x)
            max_c = max(max_const, max_aff)
            
            f = 2.0 * n * max_c / (new_sum * new_sum)
            if f < best_f - 1e-15:
                best_f = f
                best_x = x
        
        if abs(best_x - old) > 1e-12:
            # Update a and conv
            delta = best_x - old
            a_new[i] = best_x
            S_new += delta
            
            # Update conv in the affected range
            for k_idx in range(k_range_len):
                k = k_vals[k_idx]
                j = j_vals[k_idx]
                if j == i:
                    conv_new[k] = base_k[k_idx] + best_x * best_x
                else:
                    conv_new[k] = base_k[k_idx] + 2.0 * best_x * a_new[j]
            
            improved = True
    
    return a_new, conv_new, improved

def solve_lp_for_g(a, M, upper_bound=1000.0):
    """Solve LP: max sum(g) s.t. (a*g)[k] <= M for all k, 0 <= g <= upper_bound."""
    n = len(a)
    # Build constraint matrix A where A[k, j] = a[k-j] if 0 <= k-j < n
    A = np.zeros((2 * n - 1, n))
    for k in range(2 * n - 1):
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        if j_start <= j_end:
            A[k, j_start:j_end+1] = a[k-j_end:k-j_start+1][::-1]
    
    c = -np.ones(n)
    b_ub = np.full(2*n-1, M)
    bounds = [(0.0, upper_bound)] * n
    
    res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
    if res.status == 0:
        return res.x
    return None


def lp_blend_step(a, t=0.5, upper_bound=1000.0):
    """One LP-based blend step."""
    n = len(a)
    conv = np.convolve(a, a)
    M = float(np.max(conv))
    
    g = solve_lp_for_g(a, M, upper_bound)
    if g is None:
        return a, False
    
    g_sum = float(np.sum(g))
    if g_sum <= 1.0 + 1e-5:
        return a, False
    
    # Scale g to same sum as a
    a_sum = float(np.sum(a))
    g_scaled = g * (a_sum / g_sum)
    
    a_new = (1.0 - t) * a + t * g_scaled
    a_new = np.clip(a_new, 0, None)
    
    s = np.sum(a_new)
    if s < 0.01:
        return a, False
    a_new = a_new / s * a_sum  # maintain same sum
    
    return a_new, True


def generate_init(n, method, rng):
    """Generate initial sequence of length n."""
    if method == 'uniform':
        a = np.ones(n)
    elif method == 'cosine':
        x = np.linspace(-1, 1, n)
        a = np.cos(np.pi * x / 2) ** 2
    elif method == 'parabola':
        x = np.linspace(-1, 1, n)
        a = 1.0 - x * x
        a = np.maximum(a, 0.01)
    elif method == 'gaussian':
        x = np.linspace(-3, 3, n)
        sigma = rng.uniform(0.5, 1.5)
        a = np.exp(-x**2 / (2 * sigma**2))
    elif method == 'random':
        a = rng.random(n)
    elif method == 'bump':
        x = np.linspace(0, 1, n)
        center = rng.uniform(0.3, 0.7)
        width = rng.uniform(0.1, 0.4)
        a = np.exp(-((x - center)**2) / (2 * width**2))
    elif method == 'sine_mod':
        x = np.arange(n)
        freq = rng.uniform(0.5, 3.0)
        phase = rng.uniform(0, 2*np.pi)
        a = np.maximum(0, np.sin(freq * x / n * 2 * np.pi + phase))
        if np.sum(a) < 1e-10:
            a = np.ones(n)
    elif method == 'two_hump':
        x = np.linspace(0, 1, n)
        c1 = rng.uniform(0.2, 0.4)
        c2 = rng.uniform(0.6, 0.8)
        w = rng.uniform(0.05, 0.15)
        a = np.exp(-((x-c1)**2)/(2*w**2)) + np.exp(-((x-c2)**2)/(2*w**2))
    else:
        a = np.ones(n)
    
    s = np.sum(a)
    if s < 1e-10:
        a = np.ones(n)
        s = n
    return a / s

"""
Hybrid coordinate descent + LP optimization for minimizing
2*n*max(conv(a,a))/(sum(a))^2 over non-negative sequences.
Uses exact 1D coordinate optimization (optimal step per coordinate),
LP-based moves for escaping local optima, and diverse initializations.

import numpy as np
from scipy.optimize import linprog
import time


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    conv = np.convolve(a, a)
    max_b = float(np.max(conv))
    sum_a = float(np.sum(a))
    if sum_a < 0.01:
        return np.inf
    return 2.0 * n * max_b / (sum_a * sum_a)


def cd_pass(a, conv, coord_order=None):
    """One full pass of coordinate descent with exact 1D optimization."""
    n = len(a)
    S = float(np.sum(a))
    improved = False

    if coord_order is None:
        coord_order = range(n)

    for i in coord_order:
        old = a[i]
        S_other = S - old

        k_min = i
        k_max = min(i + n - 1, 2 * n - 2)
        k_len = k_max - k_min + 1

        j_vals = np.arange(k_len)
        k_vals = k_min + j_vals

        # Base: convolution without a[i]'s contribution
        base_k = conv[k_vals].copy()
        contrib = 2.0 * old * a[j_vals]
        if i < k_len:
            contrib[i] = old * old
        base_k -= contrib

        # Linear coefficients
        w = 2.0 * a[j_vals].copy()
        if i < k_len:
            quad_base = base_k[i]
            w[i] = 0.0
        else:
            quad_base = None

        # Max of unaffected parts
        max_const = 0.0
        if k_min > 0:
            max_const = max(max_const, float(np.max(conv[:k_min])))
        if k_max + 1 < len(conv):
            max_const = max(max_const, float(np.max(conv[k_max + 1:])))

        # Build candidate x values
        candidates = [0.0, min(old, 1000.0), 1000.0]

        valid = w > 1e-15
        if np.any(valid):
            xc = S_other - 2.0 * base_k[valid] / w[valid]
            xc = xc[(xc > 1e-12) & (xc < 999.999)]
            candidates.extend(xc.tolist())

        if quad_base is not None and S_other > 1e-15:
            xq = quad_base / S_other
            if 1e-12 < xq < 999.999:
                candidates.append(xq)

        # Evaluate candidates
        best_x = old
        current_max_c = max(max_const, float(np.max(base_k + w * old)),
                            quad_base + old * old if quad_base is not None else 0)
        best_f = 2.0 * n * current_max_c / (S * S) if S > 0.01 else np.inf

        for x in candidates:
            x = max(0.0, min(1000.0, x))
            new_sum = S_other + x
            if new_sum < 0.01:
                continue

            lin_vals = base_k + w * x
            max_aff = float(np.max(lin_vals))
            if quad_base is not None:
                max_aff = max(max_aff, quad_base + x * x)
            max_c = max(max_const, max_aff)

            f = 2.0 * n * max_c / (new_sum * new_sum)
            if f < best_f - 1e-14:
                best_f = f
                best_x = x

        if abs(best_x - old) > 1e-13:
            delta = best_x - old
            a[i] = best_x
            S += delta
            for idx in range(k_len):

                j = j_vals[idx]

                    conv[k] = base_k[idx] + best_x * best_x
                else:
                    conv[k] = base_k[idx] + 2.0 * best_x * a[j]
            improved = True

    return improved

def solve_lp(a, M):
    """LP: max sum(g) s.t. (a*g)[k] <= M, g >= 0."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for k in range(2 * n - 1):
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        if js <= je:
            A[k, js:je + 1] = a[k - je:k - js + 1][::-1]
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
    if res.status == 0:
        return res.x
    return None


def lp_step(a, t=0.5):
    """One LP blend step. Returns (new_a, success)."""
    n = len(a)
    if n > 300:
        return a, False
    conv = np.convolve(a, a)
    M = float(np.max(conv))
    g = solve_lp(a, M)
    if g is None:
        return a, False
    g_sum = float(np.sum(g))
    a_sum = float(np.sum(a))
    if g_sum <= a_sum * (1.0 + 1e-6):
        return a, False
    g_scaled = g * (a_sum / g_sum)
    a_new = (1.0 - t) * a + t * g_scaled
    a_new = np.clip(a_new, 0, None)
    s = float(np.sum(a_new))
    if s < 0.01:
        return a, False
    return a_new * (a_sum / s), True


def gen_init(n, method, rng):
    """Generate initial sequence."""
    if method == 'uniform':
        a = np.ones(n)
    elif method == 'cosine':
        x = np.linspace(-1, 1, n)
        a = np.cos(np.pi * x / 2.0) ** 2
    elif method == 'parabola':
        x = np.linspace(-1, 1, n)
        a = np.maximum(1.0 - x * x, 0.01)
    elif method == 'gaussian':
        x = np.linspace(-3, 3, n)
        a = np.exp(-x ** 2 / (2 * rng.uniform(0.5, 1.5) ** 2))
    elif method == 'random':
        a = rng.random(n) + 0.01
    elif method == 'bump':
        x = np.linspace(0, 1, n)
        c = rng.uniform(0.3, 0.7)
        w = rng.uniform(0.1, 0.4)
        a = np.exp(-((x - c) ** 2) / (2 * w ** 2))
    elif method == 'sine_mod':
        x = np.arange(n)
        f = rng.uniform(0.5, 3.0)
        ph = rng.uniform(0, 2 * np.pi)
        a = np.maximum(0.0, np.sin(f * x / n * 2 * np.pi + ph)) + 0.01
    elif method == 'two_hump':
        x = np.linspace(0, 1, n)
        c1, c2 = rng.uniform(0.2, 0.4), rng.uniform(0.6, 0.8)
        w = rng.uniform(0.05, 0.15)
        a = np.exp(-((x - c1) ** 2) / (2 * w ** 2)) + np.exp(-((x - c2) ** 2) / (2 * w ** 2))
    elif method == 'triangular':
        x = np.linspace(-1, 1, n)
        a = np.maximum(1.0 - np.abs(x), 0.01)
    elif method == 'inverse_cos':
        x = np.linspace(0.01, 0.99, n)
        a = 1.0 / np.sqrt(x * (1 - x))
        a = a / np.max(a)
    else:
        a = np.ones(n)

    s = float(np.sum(a))
    if s < 1e-10:
        a = np.ones(n)
        s = n
    return a / s


def propose_candidate(seed=42, budget_s=1500, **kwargs):
    """
    Hybrid coordinate descent + LP optimization.
    Tries multiple n values and diverse initializations.
    """
    rng = np.random.default_rng(seed)
    start_time = time.monotonic()
    deadline = start_time + float(budget_s) - 5.0

    best_seq = None
    best_score = np.inf

    # Try parent
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64)
        parent = np.clip(parent, 0, 1000.0)
        if parent.size > 0 and np.isfinite(parent).all():
            obj = compute_obj(parent)
            if obj < best_score:
                best_score = obj
                best_seq = parent.copy()
            print(f"Parent (n={parent.size}): obj={ obj:.6f}")
    except Exception:
        pass

    n_values = [20, 30, 40, 50, 60, 80, 100, 120, 150, 200, 250]
    methods = ['uniform', 'cosine', 'parabola', 'gaussian', 'random',
               'bump', 'sine_mod', 'two_hump', 'triangular', 'inverse_cos']

    all_results = []

    for n in n_values:
        if time.monotonic() > deadline - 60:
            break
        remaining = deadline - time.monotonic()
        print(f"\nN={n}: {remaining:.0f}s remaining")

        for mi, method in enumerate(methods):
            if time.monotonic() > deadline - 15:
                break

            t0 = time.monotonic()
            a_init = gen_init(n, method, rng)
            s = float(np.sum(a_init))
            a_init = a_init * (np.sqrt(n) / s)  # normalize so values are O(1)

            # CD phase
            max_p = int(min(5000, max(200, remaining / 0.005)))
            a_cd, obj_cd = run_cd(a_init, max_passes=max_p,
                                  seed=seed * 31 + n * 7 + mi,
                                  verbose=(method == 'uniform' and n == 100))

            if obj_cd < best_score:
                best_score = obj_cd
                best_seq = a_cd.copy()
                print(f"  CD n={n} {method}: {obj_cd:.6f} ({time.monotonic()-t0:.1f}s)")

            all_results.append((obj_cd, a_cd.copy(), n, method))

            # LP step if time allows
            if time.monotonic() < deadline - 30 and n <= 250:
                a_lp = a_cd.copy()
                for li in range(30):
                    if time.monotonic() > deadline - 5:
                        break
                    a_new, ok = lp_step(a_lp, t=0.4)
                    if not ok:
                        break
                    a_lp = a_new
                    obj_lp = compute_obj(a_lp)
                    if obj_lp < best_score:
                        best_score = obj_lp
                        best_seq = a_lp.copy()
                        print(f"  LP n={n} {method} it={li}: {obj_lp:.6f}")
                    # Quick CD after LP
                    if time.monotonic() < deadline - 5:
                        a_c, obj_c = run_cd(a_lp, max_passes=2000,
                                             seed=seed + n + li * 13)
                        if obj_c < best_score:
                            best_score = obj_c
                            best_seq = a_c.copy()
                            a_lp = a_c

    # Refine top results
    all_results.sort(key=lambda x: x[0])

    if all_results and time.monotonic() < deadline - 20:
        print(f"\n--- Refining top {min(5, len(all_results))} results ---")
        for obj_r, a_r, n_r, meth_r in all_results[:5]:
            if time.monotonic() > deadline - 10:
                break
            print(f"  Refining n={n_r} {meth_r} (obj={obj_r:.6f})")

            a_ref, obj_ref = run_cd(a_r, max_passes=20000,
                                    seed=seed + 77777, verbose=True)
            if obj_ref < best_score:
                best_score = obj_ref
                best_seq = a_ref.copy()
                print(f"    -> {best_score:.8f}")

            if time.monotonic() < deadline - 10 and n_r <= 250:
                a_l = a_ref.copy()
                for _ in range(50):
                    if time.monotonic() > deadline - 5:
                        break
                    a_new, ok = lp_step(a_l, t=0.3)
                    if not ok:
                        break
                    a_l = a_new
                    obj_l = compute_obj(a_l)
                    if obj_l < best_score:
                        best_score = obj_l
                        best_seq = a_l.copy()
                    if time.monotonic() < deadline - 5:
                        a_c, obj_c = run_cd(a_l, max_passes=5000,
                                             seed=seed + 88888)
                        if obj_c < best_score:
                            best_score = obj_c
                            best_seq = a_c.copy()

    # Final intensive CD on best
    if best_seq is not None and time.monotonic() < deadline - 3:
        a_f, obj_f = run_cd(best_seq, max_passes=50000,
                            seed=seed + 111111, verbose=True)
        if obj_f < best_score:
            best_score = obj_f
            best_seq = a_f.copy()
            print(f"  Final: {best_score:.8f}")

    if best_seq is None:
        best_seq = np.ones(100)
        best_score = compute_obj(best_seq)

    elapsed = time.monotonic() - start_time
    print(f"\nDONE: n={len(best_seq)}, score={best_score:.8f}, time={elapsed:.1f}s")
    return [float(x) for x in best_seq]
