   """
   Smooth-max gradient descent with momentum, adaptive temperature,
   and LP polishing for minimizing 2n*max(conv(a,a))/sum(a)^2.
   """
   import numpy as np
   import time
   from scipy.optimize import linprog
   from scipy.sparse import csr_matrix

   def build_conv_matrix_sparse(f):
       n = len(f)
       # Pre-allocate
       total = 0
       for k in range(2 * n - 1):
           j_min = max(0, k - (n - 1))
           j_max = min(k, n - 1)
           total += j_max - j_min + 1
       rows = np.empty(total, dtype=np.int32)
       cols = np.empty(total, dtype=np.int32)
       vals = np.empty(total, dtype=np.float64)
       idx = 0
       for k in range(2 * n - 1):
           j_min = max(0, k - (n - 1))
           j_max = min(k, n - 1)
           for j in range(j_min, j_max + 1):
               rows[idx] = k
               cols[idx] = j
               vals[idx] = f[k - j]
               idx += 1
       return csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))

       A = build_conv_matrix_sparse(f)

           result = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
           if result.success:

"""
Smooth-max gradient descent with momentum, adaptive temperature annealing,
subgradient polishing, and LP refinement for minimizing 2n*max(conv(a,a))/sum(a)^2.
Uses vectorized Hankel matrix-vector products for O(n log n) gradient evaluation.
"""
import numpy as np
import time
from scipy.optimize import linprog
from scipy.sparse import csr_matrix


def build_conv_matrix_sparse(f):
    """Build sparse matrix A where (A@g)[k] = convolve(f,g)[k]."""
    n = len(f)
    total = 0
    for k in range(2 * n - 1):
        j_min = max(0, k - (n - 1))
        j_max = min(k, n - 1)
        total += j_max - j_min + 1
    rows = np.empty(total, dtype=np.int32)
    cols = np.empty(total, dtype=np.int32)
    vals = np.empty(total, dtype=np.float64)
    idx = 0
    for k in range(2 * n - 1):
        j_min = max(0, k - (n - 1))
        j_max = min(k, n - 1)
        for j in range(j_min, j_max + 1):
            rows[idx] = k
            cols[idx] = j
            vals[idx] = f[k - j]
            idx += 1
    return csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))


def solve_lp_direction(f, M):
    """Solve: max sum(g) s.t. g >= 0, convolve(f,g)_k <= M for all k."""
    n = len(f)
    A = build_conv_matrix_sparse(f)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    try:
        result = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
        if result.success:
            return result.x
    except Exception:
        pass
    return None


def hankel_matvec(p, a):
    """Compute (Hankel(p) @ a)[j] = sum_i p[i+j]*a[i] for j=0..n-1.
    Uses convolution: result = convolve(p, a[::-1])[n-1 : 2n-1]"""
    n = len(a)
    a_rev = a[::-1]
    c = np.convolve(p, a_rev)
    return c[n - 1:2 * n - 1]


def smooth_max_gradient(a, T):
    """Compute gradient of log(sum(exp(conv(a,a)/T))) w.r.t. a.
    Returns (gradient, convolution)."""
    n = len(a)
    conv_aa = np.convolve(a, a)
    x = conv_aa / T
    x -= x.max()
    exp_x = np.exp(x)
    p = exp_x / exp_x.sum()
    grad = (2.0 / T) * hankel_matvec(p, a)
    return grad, conv_aa


def subgradient_step(a, T_sub=0.0):
    """Compute subgradient of max(conv(a,a)) at a.
    Uses the active set (lags within T_sub of max)."""
    n = len(a)
    conv_aa = np.convolve(a, a)
    M = np.max(conv_aa)
    if T_sub > 0:
        active = conv_aa >= (M - T_sub)
    else:
        active = conv_aa == M
    if not np.any(active):
        active = np.zeros(len(conv_aa), dtype=bool)
        active[int(np.argmax(conv_aa))] = True
    # Subgradient: (2/|active|) * sum_{k in active} 2*a[k-j] (where valid)
    # = (4/|active|) * hankel_matvec(p_active, a) where p_active = 1/|active| on active
    p_active = np.zeros(len(conv_aa))
    p_active[active] = 1.0 / np.sum(active)
    grad = 2.0 * hankel_matvec(p_active, a)

            no_improve += 1
            it += 1
            continue
        a_new = a_new * (S / S_new)

        score = evaluate_sequence(a_new.tolist())
        it += 1

        if np.isfinite(score) and score < best_score:
            a = a_new
            best_a = a.copy()
            best_score = score
            no_improve = 0
            T_active = max(T_active * 0.9, M * 0.001)
            if it % 500 == 0:
                print(f"    {verbose_prefix}sg iter {it}: score={best_score:.6f} eta={eta:.4f}")
        else:
            no_improve += 1
            eta *= 0.9
            if eta < 1e-6:
                break

        if no_improve > 500:
            break

    return best_a, best_score


def lp_polish(f, deadline, max_iters=300, verbose_prefix=""):
    """LP-based polishing with adaptive step size."""
    f = f.copy().astype(np.float64)
    best_f = f.copy()
    best_score = evaluate_sequence(f.tolist())
    step_size = 0.15
    no_improve = 0

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        conv_f = np.convolve(f, f)
        M = float(np.max(conv_f))

        g = solve_lp_direction(f, M)
        if g is None or float(np.sum(g)) <= 1e-15:
            no_improve += 1
            if no_improve > 40:
                break
            continue

        g = g * (float(np.sum(f)) / float(np.sum(g)))
        new_f = (1.0 - step_size) * f + step_size * g
        new_f = np.clip(new_f, 0.0, 1000.0)

        new_score = evaluate_sequence(new_f.tolist())

        if np.isfinite(new_score) and new_score < best_score:
            f = new_f
            best_f = f.copy()
            best_score = new_score
            no_improve = 0
            if it % 50 == 0:
                print(f"    {verbose_prefix}lp iter {it}: score={best_score:.6f} step={step_size:.3f}")
        else:
            no_improve += 1
            step_size *= 0.75
            if step_size < 0.003:
                break

    return best_f, best_score


def random_perturb(a, strength, rng):
    """Add random perturbation and renormalize."""
    noise = rng.uniform(-strength, strength, size=len(a))
    a_new = a + noise
    a_new = np.clip(a_new, 0.0, 1000.0)
    S_new = float(np.sum(a_new))
    if S_new > 1e-10:
        a_new = a_new * (float(np.sum(a)) / S_new)
    return a_new


def propose_candidate(budget_s=1500, **kwargs):
    t0 = time.monotonic()
    deadline = t0 + max(0.0, float(budget_s) - 10.0)

    best_sequence = None
    best_score = float('inf')
    rng = np.random.default_rng(42)

    # Try parent
    parent = None
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64).copy()
        parent = np.clip(parent, 0.0, 1000.0)
        if parent.ndim == 1 and parent.size > 0 and np.all(np.isfinite(parent)):
            score = evaluate_sequence(parent.tolist())
            if np.isfinite(score):
                best_score = score
                best_sequence = parent.copy()
                print(f"Parent n={parent.size}: score={score:.6f}")
    except (NameError, Exception) as e:
        print(f"Could not load parent: {e}")

    # Strategy: focus computational budget on the most promising lengths
    # Use parent length if available, plus a couple of alternatives
    if parent is not None:
        n_parent = len(parent)
        # Try parent length first, then nearby lengths
        lengths_to_try = sorted(set([n_parent, n_parent * 2 if n_parent <= 1024 else n_parent // 2]))
    else:
        lengths_to_try = [512, 1024]

    # Add some diversity
    if 1024 not in lengths_to_try and len(lengths_to_try) < 3:
        lengths_to_try.append(1024)
    lengths_to_try = sorted(lengths_to_try)

    for n in lengths_to_try:
        if time.monotonic() > deadline - 90:
            print(f"Time running out, skipping n={n}")
            break

        print(f"\n=== Trying n={n} ===")

        # Build diverse initializations
        init_list = []
        idx = np.arange(n, dtype=np.float64)
        center = (n - 1) / 2.0

        # Cosine window
        init_list.append(('cosine', (1.0 + np.cos(2 * np.pi * idx / n)) / 2.0 + 0.01))

        # Raised cosine
        init_list.append(('raised_cos', (1.0 - np.cos(np.pi * (idx + 0.5) / n)) / 2.0 + 0.01))

        # Exponential decay (asymmetric)
        init_list.append(('exp_decay', np.exp(-2.0 * idx / n) + 0.01))

        # Two-sided exponential (asymmetric centers)
        init_list.append(('asym_exp',
                          np.exp(-3.0 * np.abs(idx - n * 0.35) / (n * 0.25)) +
                          0.3 * np.exp(-3.0 * np.abs(idx - n * 0.7) / (n * 0.2)) + 0.01))

        # Power-law
        init_list.append(('power', (1.0 - idx / (n + 1)) ** 1.5 + 0.01))

        # Piecewise: high middle, low edges
        envelope = np.ones(n)
        edge = max(1, n // 10)
        envelope[:edge] = np.linspace(0.1, 1.0, edge)
        envelope[-edge:] = np.linspace(1.0, 0.1, edge)
        init_list.append(('taper', envelope))

        # Random with structure
        r1 = rng.uniform(0.2, 1.0, size=n)
        smooth_r = np.convolve(r1, np.ones(max(1, n // 50)) / max(1, n // 50), mode='same')
        init_list.append(('smooth_rand', smooth_r + 0.01))

        # If we have a parent, try to interpolate it to length n
        if parent is not None and len(parent) != n:
            from numpy import interp
            x_old = np.linspace(0, 1, len(parent))
            x_new = np.linspace(0, 1, n)
            parent_interp = interp(x_new, x_old, parent)
            init_list.append(('parent_interp', parent_interp + 0.01))

        # If parent has same length, use it as init
        if parent is not None and len(parent) == n:
            init_list.insert(0, ('parent', parent.copy()))

        for label, a_init in init_list:
            if time.monotonic() > deadline - 45:
                break

            a_init = np.clip(a_init.astype(np.float64), 0.01, 1000.0)
            score = evaluate_sequence(a_init.tolist())
            if not np.isfinite(score):
                continue

            print(f"  {label}: init={score:.6f}", end="")

            if score < best_score:
                best_score = score
                best_sequence = a_init.copy()

            # Phase 1: Smooth-max optimization
            sm_deadline = min(deadline - 30, time.monotonic() + 90)
            a_sm, s_sm = smooth_max_optimize(a_init, sm_deadline, verbose_prefix=f"[{label}] ")
            if s_sm < best_score:
                best_score = s_sm
                best_sequence = a_sm.copy()
                print(f" -> smooth: {s_sm:.6f}", end="")
            else:
                print(f" -> smooth: {max(score, s_sm):.6f}", end="")

            # Phase 2: Subgradient polish
            if time.monotonic() < deadline - 25:
                start_p2 = a_sm if s_sm <= score else a_init
                sg_deadline = min(deadline - 20, time.monotonic() + 40)
                a_sg, s_sg = subgradient_polish(start_p2, sg_deadline, verbose_prefix=f"[{label}] ")
                if s_sg < best_score:
                    best_score = s_sg
                    best_sequence = a_sg.copy()
                    print(f" -> subgrad: {s_sg:.6f}", end="")

            # Phase 3: LP polish
            if time.monotonic() < deadline - 10:
                to_polish = best_sequence if best_score == s_sg or s_sg < score else (a_sm if s_sm < score else a_init)
                lp_deadline = min(deadline - 5, time.monotonic() + 45)
                a_lp, s_lp = lp_polish(to_polish, lp_deadline, max_iters=150, verbose_prefix=f"[{label}] ")
                if s_lp < best_score:
                    best_score = s_lp
                    best_sequence = a_lp.copy()
                    print(f" -> LP: {s_lp:.6f}", end="")

            print()

        # If best is from this length, do additional refinement with random restarts
        if best_sequence is not None and len(best_sequence) == n and time.monotonic() < deadline - 20:
            print(f"  Extra refinement on best (n={n})...")
            for restart in range(3):
                if time.monotonic() > deadline - 15:
                    break
                a_pert = random_perturb(best_sequence, 0.05 * np.max(best_sequence), rng)
                a_opt, s_opt = smooth_max_optimize(a_pert, min(deadline - 10, time.monotonic() + 30),
                                                    verbose_prefix=f"[restart{restart}] ")
                if s_opt < best_score:
                    best_score = s_opt
                    best_sequence = a_opt.copy()
                    print(f"    restart {restart}: {s_opt:.6f}")
                a_lp, s_lp = lp_polish(best_sequence, min(deadline - 5, time.monotonic() + 20), max_iters=80)
                if s_lp < best_score:
                    best_score = s_lp
                    best_sequence = a_lp.copy()
                    print(f"    restart {restart} LP: {s_lp:.6f}")

    # Final LP polish on overall best
    if best_sequence is not None and time.monotonic() < deadline:
        print(f"\n=== Final polish (n={len(best_sequence)}) ===")
        a_final, s_final = lp_polish(best_sequence, deadline, max_iters=100, verbose_prefix="[final] ")
        if s_final < best_score:
            best_score = s_final
            best_sequence = a_final.copy()

        # One more subgradient pass
        if time.monotonic() < deadline - 5:
            a_sg2, s_sg2 = subgradient_polish(best_sequence, deadline, verbose_prefix="[final_sg] ")
            if s_sg2 < best_score:
                best_score = s_sg2
                best_sequence = a_sg2.copy()

    if best_sequence is None:
        return [1.0] * 128

    elapsed = time.monotonic() - t0
    print(f"\nFINAL: n={len(best_sequence)}, score={best_score:.6f}, time={elapsed:.1f}s")
    return [float(x) for x in best_sequence]
