import numpy as np
from scipy.optimize import linprog
import time

def compute_obj(a):
    n = len(a)
    conv = np.convolve(a, a)
    max_b = float(np.max(conv))
    sum_a = float(np.sum(a))
    if sum_a < 0.01:
        return np.inf
    return 2.0 * n * max_b / (sum_a * sum_a)

def subgradient(a, n_peaks=1):
    """Compute subgradient of the objective."""
    n = len(a)
    conv = np.convolve(a, a)
    sum_a = np.sum(a)
    max_b = conv.max()
    
    # Find peak lags
    peak_threshold = max_b * (1 - 1e-6)
    peak_lags = np.where(conv >= peak_threshold)[0]
    
    # Use top n_peaks
    if len(peak_lags) > n_peaks:
        # Take the largest ones
        peak_vals = conv[peak_lags]
        top_idx = np.argsort(peak_vals)[-n_peaks:]
        peak_lags = peak_lags[top_idx]
    
    # Compute subgradient
    grad = np.zeros(n)
    for k in peak_lags:
        for j in range(n):
            idx = k - j
            if 0 <= idx < n:
                grad[j] += 2.0 * a[idx]
    grad /= len(peak_lags)
    
    # Scale by 2n/S^2 and subtract 4nM/S^3
    grad = (2.0 * n / (sum_a ** 2)) * grad - (4.0 * n * max_b / (sum_a ** 3))
    
    return grad

def gradient_refine(a, max_iter=5000, eta_init=0.01, verbose=False):
    """Projected subgradient descent on the simplex."""
    n = len(a)
    a = a.copy()
    a = np.clip(a, 0, None)
    s = a.sum()
    if s < 1e-10:
        return a, compute_obj(a)
    a = a / s  # Normalize to sum = 1
    
    best_a = a.copy()
    best_obj = compute_obj(best_a)
    eta = eta_init

        grad = subgradient(a)
        a_new = np.clip(a - eta * grad, 0, None)

            a = best_a  # Reset to best
        
        if verbose and it % 100 == 0:

def build_A(a):

        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        if j_start <= j_end:
            A[k, j_start:j_end + 1] = a[k - j_end:k - j_start + 1][::-1]
    return A

def solve_lp(a):
    """Solve LP: max sum(g) s.t. (a*g)_k <= M, g >= 0."""
    n = len(a)
    conv = np.convolve(a, a)
    M = float(np.max(conv))
    A = build_A(a)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)

    res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
    if res.status == 0:

    return None, None, M

Hybrid LP + projected subgradient optimization for minimizing
2*n*max(conv(a,a))/(sum(a))^2 over non-negative sequences.
"""

from scipy.optimize import linprog
import time


def compute_obj(a):
    n = len(a)
    if n == 0:
        return np.inf
    conv = np.convolve(a, a)
    max_b = float(np.max(conv))
    sum_a = float(np.sum(a))
    if sum_a < 0.01:
        return np.inf
    return 2.0 * n * max_b / (sum_a * sum_a)


def compute_subgrad(a, top_k=3):
    n = len(a)
    conv = np.convolve(a, a)
    sum_a = float(np.sum(a))
    M = float(np.max(conv))
    
    if top_k >= len(conv):
        peak_indices = np.arange(len(conv))
    else:
        peak_indices = np.argsort(conv)[-top_k:][::-1]
    
    grad = np.zeros(n)
    for k in peak_indices:
        j_min = max(0, k - n + 1)
        j_max = min(n - 1, k)
        if j_min <= j_max:
            js = np.arange(j_min, j_max + 1)
            grad[js] += 2.0 * a[k - js]
    
    grad /= len(peak_indices)
    grad = (2.0 * n / (sum_a ** 2)) * grad - (4.0 * n * M / (sum_a ** 3))
    
    return grad, M, sum_a


def build_A(a):
    n = len(a)
    A = np.zeros((2 * n - 1, n), dtype=np.float64)
    for k in range(2 * n - 1):
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        if j_start <= j_end:
            A[k, j_start:j_end + 1] = a[k - j_end:k - j_start + 1][::-1]
    return A


def solve_lp(a):
    n = len(a)
    conv = np.convolve(a, a)
    M = float(np.max(conv))
    A = build_A(a)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, 1000.0)] * n
    res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
    if res.status == 0:
        return res.x, float(-res.fun), M
    return None, None, M


def gradient_descend(a, max_iter=5000, eta_init=0.01, top_k=3, verbose=False):
    n = len(a)
    a = np.clip(a.copy(), 0, None)
    s = a.sum()
    if s < 1e-10:
        return a, compute_obj(a)
    a = a / s
    
    best_a = a.copy()
    best_obj = compute_obj(best_a)
    eta = eta_init
    no_improve = 0
    rng_local = np.random.default_rng(12345)
    
    for it in range(max_iter):
        grad, M, S = compute_subgrad(a, top_k=top_k)
        a_new = np.clip(a - eta * grad, 0, None)
        s_new = a_new.sum()
        if s_new < 1e-10:
            eta *= 0.5
            if eta < 1e-10:
                break
            continue
        a_new = a_new / s_new
        obj_new = compute_obj(a_new)
        
        if obj_new < best_obj - 1e-12:
            best_a = a_new.copy()
            best_obj = obj_new
            no_improve = 0
            eta = min(eta * 1.03, 0.1)
        else:
            no_improve += 1
            eta *= 0.85
            if no_improve > 200:
                a = best_a * np.exp(rng_local.normal(0, 0.1, n))
                a = np.clip(a, 0, None)
                s2 = a.sum()
                if s2 > 1e-10:
                    a = a / s2
                no_improve = 0
                eta = 0.005
        
        if verbose and it % 500 == 0:
            print(f"      GD it={it}: obj={best_obj:.6f}, eta={eta:.6f}")
    
    return best_a, best_obj


def lp_optimize(a, max_iter=200, t=0.5, verbose=False):
    n = len(a)
    a = np.clip(a.copy(), 0, None)
    s = a.sum()
    if s < 1e-10:
        return a, compute_obj(a)
    a = a / s
    
    best_a = a.copy()
    best_obj = compute_obj(best_a)
    stall = 0
    rng_local = np.random.default_rng(67890)
    
    for it in range(max_iter):
        g, g_sum, M = solve_lp(a)
        if g is None:
            break
        
        if g_sum <= 1.0 + 1e-5:
            stall += 1
            if stall > 15:
                # Perturb
                a = best_a * np.exp(rng_local.normal(0, 0.2, n))
                a = np.clip(a, 0, None)
                s2 = a.sum()
                if s2 > 1e-10:
                    a = a / s2
                stall = 0
            continue
        stall = 0
        
        g_norm = g / g_sum
        
        # Try multiple step sizes
        best_candidate = None
        best_candidate_obj = best_obj
        for t_candidate in [t, 0.3 * t, 0.7 * t]:
            a_try = (1.0 - t_candidate) * a + t_candidate * g_norm
            a_try = np.clip(a_try, 0, None)
            s_try = a_try.sum()
            if s_try < 1e-10:
                continue
            a_try = a_try / s_try
            obj_try = compute_obj(a_try)
            if obj_try < best_candidate_obj:
                best_candidate = a_try.copy()
                best_candidate_obj = obj_try
        
        if best_candidate is not None:
            a = best_candidate
            best_a = best_candidate.copy()
            best_obj = best_candidate_obj
        else:
            a = (1.0 - 0.1 * t) * a + 0.1 * t * g_norm
            a = np.clip(a, 0, None)
            s2 = a.sum()
            if s2 > 1e-10:
                a = a / s2
        
        if verbose and (it % 20 == 0 or it < 3):
            print(f"      LP it={it}: obj={best_obj:.6f}")
    
    return best_a, best_obj


def random_perturb(a, rng, max_iter=5000):
    best_a = a.copy()
    best_obj = compute_obj(best_a)
    n = len(best_a)
    
    for it in range(max_iter):
        width = max(1, n // 50)
        start = int(rng.integers(0, max(1, n - width + 1)))
        candidate = best_a.copy()
        candidate[start:start + width] *= np.exp(rng.normal(0, 0.1, width))
        candidate = np.clip(candidate, 0, None)
        s = candidate.sum()
        if s < 1e-10:
            continue
        obj_new = compute_obj(candidate)
        if obj_new < best_obj:
            best_a = candidate
            best_obj = obj_new
    
    return best_a, best_obj


def optimize_single(a_init, time_budget, seed, verbose=False):
    """Full optimization pipeline for a single initial sequence."""
    n = len(a_init)
    a = np.clip(a_init.copy(), 0, None)
    s = a.sum()
    if s < 1e-10:
        a = np.random.default_rng(seed).random(n)
        a = a / a.sum()
    else:
        a = a / s
    
    best_a = a.copy()
    best_obj = compute_obj(best_a)
    start = time.monotonic()
    
    if verbose:
        print(f"    n={n}, init obj={best_obj:.6f}")
    
    # Phase 1: LP iterations
    lp_iters = min(100, int(time_budget * 0.4 / 0.5))
    if lp_iters > 5:
        a, obj = lp_optimize(a, max_iter=lp_iters, t=0.5, verbose=verbose)
        if obj < best_obj:
            best_a, best_obj = a.copy(), obj
    
    # Phase 2: Gradient iterations
    if time.monotonic() - start < time_budget * 0.7:
        remaining = time_budget * 0.7 - (time.monotonic() - start)
        gd_iters = int(min(5000, remaining / 0.005))
        if gd_iters > 10:
            a, obj = gradient_descend(a, max_iter=gd_iters, eta_init=0.01, top_k=3, verbose=verbose)
            if obj < best_obj:
                best_a, best_obj = a.copy(), obj
    
    # Phase 3: More LP (with improved starting point)
    if time.monotonic() - start < time_budget * 0.85:
        remaining = time_budget * 0.85 - (time.monotonic() - start)
        lp_iters2 = int(min(80, remaining / 0.5))
        if lp_iters2 > 3:
            a, obj = lp_optimize(a, max_iter=lp_iters2, t=0.3)
            if obj < best_obj:
                best_a, best_obj = a.copy(), obj
    
    # Phase 4: Gradient polish
    if time.monotonic() - start < time_budget - 5:
        remaining = time_budget - 5 - (time.monotonic() - start)
        gd_iters2 = int(min(3000, remaining / 0.005))
        if gd_iters2 > 10:
            a, obj = gradient_descend(a, max_iter=gd_iters2, eta_init=0.005, top_k=5)
            if obj < best_obj:
                best_a, best_obj = a.copy(), obj
    
    return best_a, best_obj
