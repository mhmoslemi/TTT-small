def subgradient_step(a, K_active, eta):
    """One subgradient step to minimize max(convolve(a,a)) subject to a>=0, sum(a)=1."""
    n = len(a)
    # Compute subgradient
    g = np.zeros(n)
    for k in K_active:
        # g[j] += 2*a[k-j] for valid j
        for j in range(max(0, k-n+1), min(k+1, n)):
            g[j] += 2 * a[k-j]
    g /= len(K_active)
    
    # Update
    a_new = a - eta * g

    a_new = project_simplex(a_new)

def local_score(a):
    """Compute 2*n*max(conv(a,a))/sum(a)^2 for non-negative a."""
    a = np.asarray(a, dtype=np.float64)
    if a.size == 0:
        return float('inf')
    a = np.clip(a, 0.0, 1000.0)
    s = float(np.sum(a))
    if s < 1e-300:
        return float('inf')
    c = np.convolve(a, a)
    m = float(np.max(c))
    return float(2.0 * len(a) * m / (s * s))


def normalize_sum1(a):
    a = np.asarray(a, dtype=np.float64)
    a = np.clip(a, 0.0, None)
    s = float(np.sum(a))
    if s < 1e-300:
        return None
    return a / s


def project_simplex(v):
    """Project v onto {x >= 0, sum(x) = 1}."""
    v = np.asarray(v, dtype=np.float64)
    n = v.size
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    ind = np.arange(1, n + 1)
    cond = u * ind > (cssv - 1.0)
    if not np.any(cond):
        return np.ones(n, dtype=np.float64) / n
    rho = ind[cond][-1]
    theta = (cssv[rho - 1] - 1.0) / rho
    return np.clip(v - theta, 0.0, None)


def compute_subgradient(a, w):
    """
    Compute subgradient g[j] = sum_k w[k] * 2*a[k-j] (with bounds).
    w has length 2n-1, a has length n.
    Returns g of length n.
    """
    n = a.size
    # g[j] = sum_i w[i+j] * a[i] for i in [0, n-1], valid i+j in [0, 2n-2]
    # This is np.correlate(w, a, 'full')[0:n]
    g = np.correlate(w, a, mode='full')[:n]
    return 2.0 * g


def multiplicative_update_soft(a, eta, T, K_top):
    """
    Multiplicative update with soft active-set weighting.
    T: temperature for softmax weighting of convolution peaks.
    K_top: minimum number of lags to include (soft floor).
    """
    n = a.size
    c = np.convolve(a, a)
    M = float(np.max(c))
    if M <= 0:
        return a.copy()

    # Soft weights: exp(c/T) normalized
    if T > 1e-10:
        log_w = c / T - np.log(np.sum(np.exp(c / T)))
        w = np.exp(log_w)
    else:
        w = np.zeros(2 * n - 1, dtype=np.float64)
        w[np.argmax(c)] = 1.0

    # Ensure at least K_top lags have non-negligible weight
    if K_top > 1:
        top_idx = np.argpartition(c, -K_top)[-K_top:]
        w[top_idx] += 1.0 / K_top
        w /= np.sum(w)

    # Compute subgradient
    g = compute_subgradient(a, w)

    # Multiplicative update: a[j] *= exp(-eta * (g[j] - mean(g)) / max(g))
    max_g = float(np.max(g))
    if max_g < 1e-300:
        return a.copy()
    mean_g = float(np.mean(g))
    log_factor = eta * (mean_g - g) / max_g
    log_factor = np.clip(log_factor, -10.0, 10.0)
    a_new = a * np.exp(log_factor)
    a_new = np.clip(a_new, 0.0, None)
    a_new = normalize_sum1(a_new)
    if a_new is None:
        return a.copy()
    return a_new

    """Find LP direction: max sum(b) s.t. f*b <= M, 0 <= b <= 1."""

    M = float(np.max(c))
    if M <= 0:
        return None

    # Build constraint matrix A where A[k,j] = f[k-j] (with bounds)
    A = np.zeros((2 * n - 1, n), dtype=np.float64)
    for k in range(2 * n - 1):
        s = max(0, k - (n - 1))
        e = min(k, n - 1)
        if s <= e:
            j = np.arange(s, e + 1)
            A[k, j] = f[k - j]

    cvec = -np.ones(n, dtype=np.float64)
    bvec = np.full(2 * n - 1, M * (1.0 + 1e-10), dtype=np.float64)
    bounds = [(0.0, 1.0)] * n

    try:
        res = linprog(cvec, A_ub=A, b_ub=bvec, bounds=bounds, method='highs',
                      options={'presolve': True})
    except Exception:
        return None

    if res is None or not res.success:
        return None

    x = np.asarray(res.x, dtype=np.float64)
    x = np.clip(x, 0.0, 1.0)
    if float(np.sum(x)) < 1e-12:
        return None
    return normalize_sum1(x)


def lp_optimize(a, max_time, seed):
    """Iteratively apply LP direction with line search."""
    a = normalize_sum1(a)
    if a is None:
        return a, float('inf')
    best = a.copy()
    bestsc = local_score(best)
    if max_time <= 0:
        return best, bestsc

    t0 = time.monotonic()
    deadline = t0 + max_time
    n = best.size
    stall = 0

    while time.monotonic() < deadline:
        rem = deadline - time.monotonic()
        if rem < 5.0:
            break
        g = lp_direction(best, n, seed)
        if g is None:
            break

        new_best = best
        new_sc = bestsc
        for t in [0.05, 0.1, 0.15, 0.2, 0.3, 0.5, 0.7, 1.0]:
            candidate = (1 - t) * best + t * g
            candidate = normalize_sum1(candidate)
            if candidate is None:
                continue
            sc = local_score(candidate)
            if sc < new_sc:
                new_sc = sc
                new_best = candidate.copy()

        if new_sc < bestsc:
            best = new_best
            bestsc = new_sc
            stall = 0
        else:
            stall += 1
            if stall > 5:
                break
        seed += 1
    return best, bestsc


def make_random_init(n, seed, kind=0):
    """Generate a random non-negative sequence of length n, normalized to sum 1."""
    rng = np.random.default_rng(seed)
    if kind == 0:
        a = rng.uniform(0.1, 1.0, n)
    elif kind == 1:
        a = rng.beta(2.0, 2.0, n)
    elif kind == 2:
        # Smoothed random
        a = rng.normal(size=n)
        w = max(3, n // 30)
        kern = np.ones(w) / w
        a = np.convolve(a, kern, mode='same')
        a = np.clip(a, 0.0, None)
    elif kind == 3:
        # Block structure
        nb = max(2, n // 20)
        bv = rng.uniform(0.1, 1.0, nb)
        a = np.repeat(bv, n // nb + 1)[:n]
        a = 0.3 * a + 0.7 * a.mean()
    else:
        a = rng.uniform(0.0, 1.0, n)
    a = np.clip(a, 0.0, None)
    res = normalize_sum1(a)
    if res is None:
        res = np.ones(n, dtype=np.float64) / n
    return res


def annealing_optimize(a, max_time, seed):
    """
    Main optimization loop combining:
    - Soft multiplicative updates with adaptive temperature
    - Hard top-K multiplicative updates
    - LP refinement
    - Random restarts
    """
    a = normalize_sum1(a)
    if a is None:
        return None, float('inf')
    n = a.size
    best = a.copy()
    bestsc = local_score(best)
    if max_time <= 0:
        return best, bestsc

    t0 = time.monotonic()
    deadline = t0 + max_time
    rng = np.random.default_rng(seed)

    # Adaptive parameters
    T = 0.01 / bestsc  # Initial temperature (normalized)
    eta = 0.4
    K_soft = max(10, n // 20)
    K_hard = max(20, n // 10)
    stuck = 0
    it = 0
    last_lp = time.monotonic()
    lp_interval = 30.0  # seconds between LP attempts

    while time.monotonic() < deadline:
        rem = deadline - time.monotonic()
        if rem < 2.0:
            break

        # Choose update type
        r = rng.random()

        if r < 0.6:
            # Soft multiplicative update
            a_new = multiplicative_update_soft(best if stuck > 50 else a, eta, T, K_soft)
        elif r < 0.85:
            # Hard top-K multiplicative update
            a_new = multiplicative_update_topK(best if stuck > 50 else a, eta, K_hard)
        else:
            # Small random perturbation
            noise = np.exp(rng.normal(0, 0.15, n))
            a_new = normalize_sum1(a * noise)
            if a_new is None:
                a_new = a.copy()

        sc = local_score(a_new)
        if np.isfinite(sc) and sc < bestsc:
            best = a_new.copy()
            bestsc = float(sc)
            stuck = 0
            # Slightly decrease temperature (exploit more)
            T *= 0.98
            T = max(T, 0.001 / bestsc)
            # Slightly increase eta if making progress
            eta = min(eta * 1.01, 1.5)
        else:
            stuck += 1
            # Increase temperature (explore more)
            T *= 1.05
            T = min(T, 1.0 / bestsc)

        a = a_new

        # Periodic LP refinement
        if time.monotonic() - last_lp > lp_interval and HAVE_SCIPY:
            last_lp = time.monotonic()
            lp_rem = min(20.0, rem * 0.3)
            if lp_rem > 5.0:
                cand_lp, sc_lp = lp_optimize(best, lp_rem, seed + it)
                if sc_lp < bestsc:
                    best = cand_lp.copy()
                    bestsc = sc_lp
                    a = best.copy()
                    stuck = 0
                    print(f"  LP improved to {bestsc:.6f}", flush=True)

        # Restart if stuck too long
        if stuck > 200:
            # Try a blend with a fresh random initialization
            fresh = make_random_init(n, seed + stuck, kind=int(rng.integers(0, 5)))
            for blend in [0.3, 0.5, 0.7]:
                cand = normalize_sum1((1 - blend) * best + blend * fresh)
                if cand is not None:
                    sc = local_score(cand)
                    if sc < bestsc:
                        best = cand.copy()
                        bestsc = sc
                        a = best.copy()
                        break
            T = 0.05 / bestsc
            eta = 0.3
            stuck = 0

        it += 1

    return best, bestsc


def coordinate_polish(a, max_time, seed):
    """
    Fine-grained coordinate perturbation: perturb small random blocks
    and accept if score improves.
    """
    a = normalize_sum1(a)
    if a is None:
        return None, float('inf')
    n = a.size
    best = a.copy()
    bestsc = local_score(best)
    if max_time <= 0:
        return best, bestsc

    t0 = time.monotonic()
    deadline = t0 + max_time
    rng = np.random.default_rng(seed)
    stuck = 0

    while time.monotonic() < deadline:
        # Perturb a random block
        w = max(1, int(rng.integers(1, max(2, n // 30))))
        start = int(rng.integers(0, n - w + 1))
        factor = np.exp(rng.normal(0, 0.3, w))
        cand = best.copy()
        cand[start:start + w] *= factor
        cand = normalize_sum1(cand)
        if cand is None:
            continue
        sc = local_score(cand)
        if sc < bestsc:
            best = cand.copy()
            bestsc = sc
            stuck = 0
        else:
            stuck += 1
            if stuck > 500:
                # Larger perturbation
                w2 = max(2, n // 10)
                start2 = int(rng.integers(0, n - w2 + 1))
                cand[start2:start2 + w2] *= np.exp(rng.normal(0, 0.5, w2))
                cand = normalize_sum1(cand)
                if cand is not None:
                    sc2 = local_score(cand)
                    if sc2 < bestsc:
                        best = cand.copy()
                        bestsc = sc2
                stuck = 0

    return best, bestsc


def resize_sequence(a, new_n):
    """Resize sequence to new length via interpolation."""
    a = np.asarray(a, dtype=np.float64)
    old_n = a.size
    if old_n == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, old_n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    a_new = normalize_sum1(a_new)
    if a_new is None:
        a_new = np.ones(new_n, dtype=np.float64) / new_n
    return a_new


def propose_candidate(seed=42, budget_s=1500, **kwargs):
    """
    Search for the best non-negative sequence minimizing 2*n*max(conv(a,a))/sum(a)^2.
    Returns a list of floats.
    """
    start_time = time.monotonic()
    try:
        budget = float(budget_s)
    except Exception:
        budget = 1500.0
    if not np.isfinite(budget) or budget <= 0:
        budget = 1500.0
    deadline = start_time + budget - 15.0

    # Get parent sequence
    parent_arr = None
    try:
        p = globals().get('height_sequence_1', None)
        if p is not None:
            parent_arr = np.asarray(p, dtype=np.float64)
            parent_arr = np.clip(parent_arr, 0.0, 1000.0)
            if parent_arr.size == 0 or not np.isfinite(parent_arr).all():
                parent_arr = None
    except Exception:
        parent_arr = None

    best_seq = None
    best_score = float('inf')

    # Initialize from parent if available
    if parent_arr is not None:
        p = normalize_sum1(parent_arr)
        if p is not None:
            best_seq = p.copy()
            best_score = local_score(p)
            print(f"Parent: n={p.size}, score={best_score:.6f}", flush=True)

    # Plan: (length, time_budget, use_parent)
    if budget >= 1400:
        plan = [
            (512, 50, True),
            (1024, 150, True),
            (2048, 400, True),
            (4096, 500, False),
            (2048, 200, False),
        ]
    elif budget >= 800:
        plan = [
            (512, 40, True),
            (1024, 150, True),
            (2048, 400, True),
            (4096, 200, False),
        ]
    else:
        plan = [
            (512, 30, True),
            (1024, 100, True),
            (2048, 300, True),
        ]

    seed_cur = int(seed)

    for n_len, t_alloc, use_parent in plan:
        rem = deadline - time.monotonic()
        if rem < 30:
            break
        t_budget = min(float(t_alloc), rem - 10.0)
        if t_budget < 10:
            continue

        print(f"\n=== n={n_len}, budget={t_budget:.0f}s ===", flush=True)

        # Generate multiple initializations
        inits = []
        if use_parent and parent_arr is not None:
            p_resized = resize_sequence(parent_arr, n_len)
            inits.append(p_resized)
            best_score_init = local_score(p_resized)
        else:
            best_score_init = float('inf')

        # Add random inits
        for k in range(3):
            ri = make_random_init(n_len, seed_cur + k * 1000 + n_len, kind=k % 5)
            inits.append(ri)

        # Optimize each init with time split
        n_inits = len(inits)
        t_per_init = t_budget / (n_inits + 0.5)  # Reserve some for final polish

        local_best = None
        local_best_sc = float('inf')

        for i_init, init_a in enumerate(inits):
            rem_init = deadline - time.monotonic()
            if rem_init < 5:
                break
            t_here = min(t_per_init, rem_init - 5.0)
            if t_here < 5:
                break

            sc_init = local_score(init_a)
            print(f"  Init {i_init}: score={sc_init:.6f}", flush=True)

            # Phase 1: Annealing optimization (main work)
            t_anneal = t_here * 0.7
            if t_anneal > 5:
                cand, sc = annealing_optimize(init_a, t_anneal, seed_cur + i_init * 7919)
                if cand is not None and sc < local_best_sc:
                    local_best = cand.copy()
                    local_best_sc = sc
                    print(f"  After annealing: {sc:.6f}", flush=True)

            # Phase 2: Coordinate polish
            t_polish = t_here * 0.3
            if local_best is not None and t_polish > 5:
                cand2, sc2 = coordinate_polish(local_best, t_polish, seed_cur + i_init * 31 + 500)
                if cand2 is not None and sc2 < local_best_sc:
                    local_best = cand2.copy()
                    local_best_sc = sc2
                    print(f"  After polish: {sc2:.6f}", flush=True)

        # Update global best
        if local_best is not None and local_best_sc < best_score:
            best_seq = local_best.copy()
            best_score = local_best_sc
            print(f"  *** New global best: n={n_len}, score={best_score:.6f} ***", flush=True)

        seed_cur += 99991

    # Final polish on the overall best
    rem = deadline - time.monotonic()
    if best_seq is not None and rem > 10:
        print(f"\n=== Final polish, remaining={rem:.0f}s ===", flush=True)

        # Annealing
        t_an = min(rem * 0.4, 100.0)
        if t_an > 5:
            cand, sc = annealing_optimize(best_seq, t_an, seed_cur + 777)
            if cand is not None and sc < best_score:
                best_seq = cand.copy()
                best_score = sc
                print(f"  Final anneal: {best_score:.6f}", flush=True)

        # Coordinate polish
        rem2 = deadline - time.monotonic()
        t_cp = min(rem2 * 0.5, 80.0)
        if t_cp > 5:
            cand2, sc2 = coordinate_polish(best_seq, t_cp, seed_cur + 888)
            if cand2 is not None and sc2 < best_score:
                best_seq = cand2.copy()
                best_score = sc2
                print(f"  Final polish: {best_score:.6f}", flush=True)

        # LP on the best
        rem3 = deadline - time.monotonic()
        if rem3 > 10 and HAVE_SCIPY:
            t_lp = min(rem3 * 0.6, 60.0)
            cand3, sc3 = lp_optimize(best_seq, t_lp, seed_cur + 999)
            if cand3 is not None and sc3 < best_score:
                best_seq = cand3.copy()
                best_score = sc3
                print(f"  Final LP: {best_score:.6f}", flush=True)

    # Fallback
    if best_seq is None:
        best_seq = np.ones(256, dtype=np.float64) / 256.0

    # Final normalization and validation
    final = normalize_sum1(best_seq)
    if final is None:
        final = np.ones(256, dtype=np.float64) / 256.0

    # Validate with the actual evaluator
    try:
        official = evaluate_sequence([float(x) for x in final])
        if np.isfinite(official):
            best_score = official
        else:
            # Try parent
            if parent_arr is not None:
                p = normalize_sum1(parent_arr)
                if p is not None:
                    off2 = evaluate_sequence([float(x) for x in p])
                    if np.isfinite(off2):
                        final = p
                        best_score = off2
    except Exception:
        pass

    final = np.clip(np.asarray(final, dtype=np.float64), 0.0, 1000.0)
    print(f"\nFINAL: n={len(final)}, score={best_score:.6f}, time={time.monotonic()-start_time:.1f}s", flush=True)
    return [float(x) for x in final]
