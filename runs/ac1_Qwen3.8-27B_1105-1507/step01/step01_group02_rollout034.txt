```python
                candidates = [S * 0.001, S / n * 0.5, S / n]
            else:
                # Try scaling around current value
                candidates = [old_val * f for f in [0.3, 0.5, 0.7, 0.85, 1.0, 1.15, 1.3, 1.5, 2.0]]
                # Also try 0
                candidates.append(0.0)
            
            best_score = eval_score(a)
            best_val = old_val
            
            for cv in candidates:
                a[i] = cv
                s = eval_score(a)
                if s < best_score:
                    best_score = s
                    best_val = cv
            
            a[i] = best_val
            S = a.sum()
    
    return a
```

Wait, the coordinate descent with eval_score for each candidate is too slow. eval_score does a full FFT convolution each time. For n=512, that's 512 coordinates * 10 candidates * FFT(1024) ≈ 512 * 10 * 1024*log(1024) ≈ 5120 * 10000 ≈ 50 million operations per sweep. That's probably 1-2 seconds per sweep. Not terrible but not great.

Let me optimize: instead of computing the full convolution for each candidate, I can compute the change in the convolution when only one coordinate changes.

When a[i] changes from v to w (delta = w - v), the change in c_k is:
- For k in [i, i+n-1] (valid range where k-i is in [0, n-1]):
  - If k == 2i: c_k changes by 2*v*delta + delta^2 (since old contrib was v^2, new is w^2 = (v+delta)^2)
  - If k != 2i: c_k changes by 2*a[k-i]*delta (since old contrib was 2*v*a[k-i], new is 2*w*a[k-i])

Wait, that's not quite right either. Let me be more careful.

c_k = sum_p a_p * a_{k-p} for valid p (0 <= p < n, 0 <= k-p < n)

The terms involving a_i:
- p = i: a_i * a_{k-i} (valid if 0 <= k-i < n)
- p = k-i: a_{k-i} * a_i (valid if 0 <= k-i < n)

If k-i = i (i.e., k = 2i), these are the same term: a_i * a_i = a_i^2
If k-i ≠ i, they are different terms: a_i * a_{k-i} + a_{k-i} * a_i = 2*a_i*a_{k-i}

So the total contribution of a_i to c_k:
- If k = 2i: a_i^2
- If 0 <= k-i < n and k ≠ 2i: 2*a_i*a_{k-i}
- Otherwise: 0

When a_i changes by delta:
- If k = 2i: change = 2*a_i*delta + delta^2
- If 0 <= k-i < n and k ≠ 2i: change = 2*a_{k-i}*delta

So I can update c_k in O(n) time (only n values of k are affected), and then find the max. This makes each coordinate update O(n) instead of O(n log n).

For a full sweep: O(n^2) = O(512^2) = O(262144) operations. Very fast!

But actually, I realize I can make this even faster. Instead of trying a fixed set of multipliers, I can analytically find the optimal value.

For a given coordinate i, the objective as a function of v (the new value of a[i]) is:
f(v) = 2n * max_k [c_k_base + L_k*v + Q_k*v^2] / (S' + v)^2

where S' = S - old_val, and c_k_base, L_k, Q_k are computed from the current state with a[i] = 0.

This is a 1D optimization. Since it's a max of rational functions, it's generally not convex, but in practice it should be unimodal near the current value.

I'll use a simple approach: compute the subgradient at the current value, take a step, and project. This is essentially what the overall subgradient method does, but restricted to one coordinate.

Let me also consider: what if we work in the frequency domain? The autocorrelation c_k is the inverse DFT of |A(f)|^2 where A(f) is the DFT of a. So we want |A(f)|^2 to be as flat as possible. This is the "flat magnitude spectrum" problem, which is what the alternating projection method addresses.

The previous code already uses alternating projection for initialization. Let me try a different angle:

**Idea: Direct frequency-domain optimization**

We want to minimize max_k c_k = max_k IDFT(|A|^2)_k subject to sum(a) = 1, a >= 0.

In frequency domain:
- sum(a) = A(0) (the DC component) = 1 (normalization)
- a >= 0 is a time-domain constraint
- max_k c_k = max of the IDFT of |A|^2

This is still hard, but we can use the following approach:
1. Start with a flat |A(f)| (which gives near-delta autocorrelation, but a might have negative values)
2. Iterate: project to a >= 0 (in time domain), then adjust phases to make |A| flat (in frequency domain)

This is exactly the alternating projection / Gerchberg-Saxton algorithm! The previous code uses this for initialization. Let me use it for the main optimization too, with a twist: instead of making |A| perfectly flat, I'll make it flat except at a few "hot spots" where the autocorrelation peaks.

```python
Optimizes nonnegative sequence to minimize 2*n*max(conv(a,a))/sum(a)^2.
Uses: subgradient descent with momentum + Polyak averaging,
coordinate descent refinement, LP-based global moves,
and diverse initialization (parent, templates, alternating projection).

import numpy as np

def project_simplex(v):
    n = len(v)
    if n == 0:
        return v.copy()
    u = np.sort(v)[::-1]
    cv = np.cumsum(u) - 1.0
    rho_candidates = np.where(u * np.arange(1, n + 1) > cv)[0]
    if len(rho_candidates) == 0:
        return np.ones(n) / n
    rho = rho_candidates[-1]
    theta = cv[rho] / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    N = 1 << (size - 1).bit_length()
    fa = np.fft.rfft(a, N)
    fc = fa * fa
    result = np.fft.irfft(fc, N)[:size]
    result[result < 0] = 0.0
    return result


def eval_score(a):
    n = len(a)
    c = fast_convolve(a)
    max_b = np.max(c)
    sum_a = np.sum(a)
    if sum_a < 0.01:
        return float('inf')
    return float(2 * n * max_b / (sum_a ** 2))


def compute_subgradient(a, peak_indices):
    n = len(a)
    grad = np.zeros(n)
    for k in peak_indices:
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        if lo <= hi:
            grad[lo:hi + 1] += 2.0 * a[k - hi:k - lo + 1][::-1]
    grad /= max(1, len(peak_indices))
    return grad


def lp_improve(a, time_limit=8.0):
    n = len(a)
    if n > 5000:
        return None
    c = fast_convolve(a)
    P = np.max(c)
    A = np.zeros((2 * n - 1, n), dtype=np.float64)
    for k in range(2 * n - 1):
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        A[k, lo:hi + 1] = a[k - hi:k - lo + 1][::-1]
    c_obj = -np.ones(n)
    b_ub = np.full(2 * n - 1, P)
    try:
        res = linprog(c_obj, A_ub=A, b_ub=b_ub, bounds=[(0.0, None)] * n,
                      method='highs', options={'time_limit': time_limit})
        if res.success:
            b = res.x
            b_sum = b.sum()
            if b_sum > 1e-8:
                a_sum = a.sum()
                b = b * (a_sum / b_sum)
                return b
    except Exception:
        pass
    return None


def coordinate_descent(a, n_sweeps=5, verbose=False):
    n = len(a)
    a = a.copy().astype(np.float64)
    S = a.sum()
    if S < 1e-15:
        return a
    c = fast_convolve(a)
    best_score = 2 * n * np.max(c) / S**2
    
    for sweep in range(n_sweeps):
        for i in range(n):
            old_val = a[i]
            if old_val < 1e-14:
                candidates = np.array([S / n * 0.1, S / n * 0.5, S / n, S / n * 2])
            else:
                candidates = old_val * np.array([0.4, 0.6, 0.8, 1.0, 1.25, 1.5, 2.0, 0.0])
            
            best_local_score = best_score
            best_local_val = old_val
            best_local_c = c
            
            for cv in candidates:
                delta = cv - old_val
                if abs(delta) < 1e-15:
                    continue
                
                # Fast update of convolution
                c_new = c.copy()
                k_lo = i
                k_hi = min(i + n - 1, 2 * n - 2)
                
                # Vectorized update
                j_vals = np.arange(k_lo, k_hi + 1) - i  # = k - i values
                # All j_vals are in [0, n-1] by construction
                coeffs = 2.0 * a[j_vals] * delta
                # Handle k = 2i case (quadratic term)
                mask_2i = (np.arange(k_lo, k_hi + 1) == 2 * i)
                coeffs[mask_2i] = 2.0 * old_val * delta + delta * delta
                
                c_new[k_lo:k_hi + 1] += coeffs
                
                new_S = S - old_val + cv
                if new_S < 1e-10:
                    continue
                score = 2.0 * n * np.max(c_new) / (new_S * new_S)
                if score < best_local_score:
                    best_local_score = score
                    best_local_val = cv
                    best_local_c = c_new
            
            a[i] = best_local_val
            S = S - old_val + best_local_val
            c = best_local_c
            best_score = best_local_score
        
        if verbose:
            print(f"    CD sweep {sweep}: score={best_score:.8f}")
    
    return a
```

```python
"""
Optimizes nonnegative sequence to minimize 2*n*max(conv(a,a))/sum(a)^2.
Strategy: Multi-phase optimization combining:
1. Diverse initialization (parent, templates, alternating projection)
2. Projected subgradient descent with adaptive step size
3. Fast coordinate descent refinement (O(n^2) per sweep)
4. LP-based global improvement
"""

import numpy as np
import time
from scipy.optimize import linprog


def project_simplex(v):
    """Project vector onto probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v.copy()
    u = np.sort(v)[::-1]
    cv = np.cumsum(u) - 1.0
    rho_candidates = np.where(u * np.arange(1, n + 1) > cv)[0]
    if len(rho_candidates) == 0:
        return np.ones(n) / n
    rho = rho_candidates[-1]
    theta = cv[rho] / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def fast_convolve(a):
    """Full linear convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    N = 1 << (size - 1).bit_length()
    fa = np.fft.rfft(a, N)
    fc = fa * fa
    result = np.fft.irfft(fc, N)[:size]
    result[result < 0] = 0.0
    return result


def eval_score(a):
    """Compute score (avoids calling evaluate_sequence for speed)."""
    n = len(a)
    c = fast_convolve(a)
    max_b = np.max(c)
    sum_a = np.sum(a)
    if sum_a < 0.01:
        return float('inf')
    return float(2 * n * max_b / (sum_a ** 2))


def get_subgradient(a, c, n_top=10):
    """Compute subgradient of max(c) w.r.t. a, using top peaks."""
    n = len(a)
    # Find top n_top peaks
    top_k = np.argsort(c)[::-1][:n_top]
    # Weight by how close to max
    max_c = c[top_k[0]]
    weights = c[top_k] / max_c  # all close to 1
    weights = weights ** 10  # sharpen: emphasize the true max
    
    grad = np.zeros(n)
    for k in top_k:
        w = weights[list(top_k).index(k)]
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        if lo <= hi:
            grad[lo:hi + 1] += w * 2.0 * a[k - hi:k - lo + 1][::-1]
    grad /= np.sum(weights)
    return grad


def coordinate_descent(a, n_sweeps=3, rng=None):
    """
    Fast coordinate descent: for each coordinate, try candidate values
    and pick the one that minimizes the score. Uses O(n) update per candidate.
    """
    n = len(a)
    a = a.copy().astype(np.float64)
    S = a.sum()
    if S < 1e-15:
        return a
    c = fast_convolve(a)
    best_score = 2.0 * n * np.max(c) / (S * S)
    
    for sweep in range(n_sweeps):
        # Shuffle order for diversity
        order = np.arange(n)
        if rng is not None:
            rng.shuffle(order)
        
        for i in order:
            old_val = a[i]
            if old_val < 1e-14:
                candidates = np.array([S * 0.001, S / (2*n), S / n, 2*S/n])
            else:
                # Adaptive candidates based on local gradient
                candidates = np.concatenate([
                    old_val * np.array([0.5, 0.7, 0.85, 1.15, 1.3, 1.6]),
                    np.array([0.0])
                ])
            
            best_local_score = best_score
            best_local_val = old_val
            best_local_c = c
            
            for cv in candidates:
                delta = cv - old_val
                if abs(delta) < 1e-15 * max(1.0, S):
                    continue
                
                # Fast convolution update: only n positions change
                c_new = c
                # Save original values
                k_lo = i
                k_hi = min(i + n - 1, 2 * n - 2)
                saved = c[k_lo:k_hi+1].copy()
                
                # Apply update
                j_idx = np.arange(k_lo, k_hi + 1) - i
                update = 2.0 * a[j_idx] * delta
                # Fix k = 2i (quadratic term)
                mask_q = (np.arange(k_lo, k_hi + 1) == 2 * i)
                update[mask_q] = 2.0 * old_val * delta + delta * delta
                
                c[k_lo:k_hi+1] += update
                new_S = S - old_val + cv
                if new_S > 1e-10:
                    score = 2.0 * n * np.max(c) / (new_S * new_S)
                    if score < best_local_score:
                        best_local_score = score
                        best_local_val = cv
                        best_local_c = c.copy()
                
                # Restore
                c[k_lo:k_hi+1] = saved
            
            if best_local_val != old_val:
                a[i] = best_local_val
                S = S - old_val + best_local_val
                c = best_local_c
                best_score = best_local_score
    
    return a


def lp_improve(a, time_limit=6.0):
    """LP step: find b maximizing sum(b) s.t. conv(a,b) <= max(conv(a,a)), b>=0."""
    n = len(a)
    if n > 5000:
        return None
    c = fast_convolve(a)
    P = np.max(c)
    
    A = np.zeros((2*n - 1, n), dtype=np.float64)
    for k in range(2*n - 1):
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        A[k, lo:hi+1] = a[k-hi:k-lo+1][::-1]
    
    c_obj = -np.ones(n)
    b_ub = np.full(2*n - 1, P)
    
    try:
        res = linprog(c_obj, A_ub=A, b_ub=b_ub,
                      bounds=[(0.0, None)]*n,
                      method='highs',
                      options={'time_limit': time_limit})
        if res.success:
            b = res.x
            b_sum = b.sum()
            if b_sum > 1e-8:
                return b * (a.sum() / b_sum)
    except Exception:
        pass
    return None


def alt_proj_init(n, rng, n_iter=40):
    """Alternating projection for flat-spectrum initialization."""
    N = 1 << (2*n - 2).bit_length()
    phases = rng.uniform(0.0, 2.0*np.pi, N//2 + 1)
    F = np.exp(1j * phases)
    F[0] = 1.0 + 0j
```

```python
Optimizes nonnegative sequence to minimize 2*n*max(conv(a,a))/sum(a)^2.
Strategy: Multi-phase optimization combining:
1. Diverse initialization (parent, templates, alternating projection)
2. Projected subgradient descent with momentum and adaptive step size
3. Fast coordinate descent refinement (O(n) update per candidate per coordinate)
4. LP-based global improvement for structural changes

import numpy as np

from scipy.optimize import linprog


def project_simplex(v):
    """Project vector onto probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v.copy()
    u = np.sort(v)[::-1]
    cv = np.cumsum(u) - 1.0
    rho_candidates = np.where(u * np.arange(1, n + 1) > cv)[0]
    if len(rho_candidates) == 0:
        return np.ones(n) / n
    rho = rho_candidates[-1]
    theta = cv[rho] / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def fast_convolve(a):
    """Full linear convolution of a with itself via FFT."""
    n = len(a)
    size = 2 * n - 1
    N = 1 << (size - 1).bit_length()
    fa = np.fft.rfft(a, N)
    fc = fa * fa
    result = np.fft.irfft(fc, N)[:size]
    result[result < 0] = 0.0
    return result


def eval_score(a):
    """Compute score without calling evaluate_sequence (faster)."""
    n = len(a)
    c = fast_convolve(a)
    max_b = np.max(c)
    sum_a = np.sum(a)
    if sum_a < 0.01:
        return float('inf')
    return float(2 * n * max_b / (sum_a ** 2))


def get_subgradient(a, c, n_top=10):
    """Compute subgradient of max(c) w.r.t. a, weighted by peak prominence."""
    n = len(a)
    max_c = np.max(c)
    top_k = np.argsort(c)[::-1][:n_top]
    # Exponential weighting to emphasize true maximum
    weights = (c[top_k] / max_c) ** 20.0
    if np.sum(weights) < 1e-15:
        weights = np.ones(len(top_k))
    grad = np.zeros(n)
    for idx, k in enumerate(top_k):
        w = 2.0 * weights[idx]
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        if lo <= hi:
            grad[lo:hi + 1] += w * a[k - hi:k - lo + 1][::-1]
    grad /= np.sum(weights)
    return grad


def coordinate_descent(a, n_sweeps=3, rng=None):
    """
    Fast coordinate descent. For each coordinate i, try candidate values
    using O(n) convolution update per candidate.
    """
    n = len(a)
    a = a.copy().astype(np.float64)
    S = a.sum()
    if S < 1e-15:
        return a
    c = fast_convolve(a)
    best_score = 2.0 * n * np.max(c) / (S * S)

    for sweep in range(n_sweeps):
        order = np.arange(n)
        if rng is not None:
            perm = rng.permutation(n)
            order = perm

        for i in order:
            old_val = a[i]
            if old_val < 1e-14:
                candidates = np.array([S * 0.002, S / (2.0 * n), S / n, 2.0 * S / n])
            else:
                candidates = np.array([0.0, old_val * 0.4, old_val * 0.65,
                                       old_val * 0.85, old_val * 1.18,
                                       old_val * 1.45, old_val * 2.0])

            best_local_score = best_score
            best_local_val = old_val

            k_lo = i
            k_hi = min(i + n - 1, 2 * n - 2)
            k_range = np.arange(k_lo, k_hi + 1)
            j_idx = k_range - i

            for cv in candidates:
                delta = cv - old_val
                if abs(delta) < 1e-15 * max(1.0, S):
                    continue

                # Apply fast update
                update = 2.0 * a[j_idx] * delta
                mask_q = (k_range == 2 * i)
                update[mask_q] = 2.0 * old_val * delta + delta * delta
                c[k_lo:k_hi + 1] += update

                new_S = S - old_val + cv
                if new_S > 1e-10:
                    score = 2.0 * n * np.max(c) / (new_S * new_S)
                    if score < best_local_score:
                        best_local_score = score
                        best_local_val = cv

                # Revert
                c[k_lo:k_hi + 1] -= update

            if best_local_val != old_val:
                # Commit the change
                delta = best_local_val - old_val
                update = 2.0 * a[j_idx] * delta
                mask_q = (k_range == 2 * i)
                update[mask_q] = 2.0 * old_val * delta + delta * delta
                c[k_lo:k_hi + 1] += update
                a[i] = best_local_val
                S = S - old_val + best_local_val
                best_score = best_local_score

    return a


def lp_improve(a, time_limit=6.0):
    """LP: maximize sum(b) s.t. conv(a,b) <= max(conv(a,a)), b >= 0."""
    n = len(a)
    if n > 4000:
        return None
    c = fast_convolve(a)
    P = np.max(c)
    A = np.zeros((2 * n - 1, n), dtype=np.float64)
    for k in range(2 * n - 1):
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        A[k, lo:hi + 1] = a[k - hi:k - lo + 1][::-1]
    c_obj = -np.ones(n)
    b_ub = np.full(2 * n - 1, P)
    try:
        res = linprog(c_obj, A_ub=A, b_ub=b_ub,
                      bounds=[(0.0, None)] * n,
                      method='highs', options={'time_limit': time_limit})

            b = res.x
            b_sum = b.sum()

        pass
    return None

        t_end = time.time() + alloc
        use_lp = (n <= 1024 and rank < 5)
        a_opt = subgradient_stage(a, t_end, rng, use_lp=use_lp)
        # Quick CD polish
        a_opt = coordinate_descent(a_opt, n_sweeps=2, rng=rng)
        score = eval_score(a_opt)
        report(score, name, a_opt)
        print(f"  [{rank+1:2d}] {name:12s} n={n:4d}: init={s_init:.6f} -> {score:.6f} (t={time.time()-t_start:.0f}s)")

    # ---- Phase 2: Deep optimization of current best ----
    print(f"\n=== Phase 2: Deep optimization (best={best_score:.8f}) ===")
    if best_seq is not None and time.time() < deadline - 120:
        n_best = len(best_seq)
        phase2_end = min(deadline - 120, time.time() + (budget_s * 0.35))
        a_deep = subgradient_stage(best_seq, phase2_end, rng,
                                   use_lp=(n_best <= 2048), lp_interval=150)
        a_deep = coordinate_descent(a_deep, n_sweeps=3, rng=rng)
        score = eval_score(a_deep)
        report(score, 'deep', a_deep)
        print(f"  Deep: {score:.8f} (t={time.time()-t_start:.0f}s)")

    # Try LP + CD on best
    if best_seq is not None and len(best_seq) <= 3000 and time.time() < deadline - 60:
        b = lp_improve(best_seq, time_limit=10.0)
        if b is not None:
            for t_mix in [0.3, 0.5, 0.7]:
                a_mix = project_simplex((1.0 - t_mix) * best_seq + t_mix * b)
                a_mix = coordinate_descent(a_mix, n_sweeps=2, rng=rng)
                score = eval_score(a_mix)
                report(score, f'lp_{t_mix}', a_mix)

    # ---- Phase 3: Intensive coordinate descent polish ----
    print(f"\n=== Phase 3: CD polish (best={best_score:.8f}) ===")
    if best_seq is not None and time.time() < deadline - 15:
        a_cd = best_seq.copy()
        while time.time() < deadline - 15:
            a_cd = coordinate_descent(a_cd, n_sweeps=3, rng=rng)
            score = eval_score(a_cd)
            if score < best_score:
                best_score = score
                best_seq = a_cd.copy()
                print(f"  CD: {score:.10f} (t={time.time()-t_start:.0f}s)")
            else:
                break

        # Final LP attempt
        if len(best_seq) <= 3000 and time.time() < deadline - 10:
            b = lp_improve(best_seq, time_limit=8.0)
            if b is not None:
                a_mix = project_simplex(0.6 * best_seq + 0.4 * b)
                a_mix = coordinate_descent(a_mix, n_sweeps=2, rng=rng)
                score = eval_score(a_mix)
                report(score, 'final_lp', a_mix)

    # ---- Final output ----
    if best_seq is None:
        best_seq = np.ones(256) / 256.0

    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    final_score = evaluate_sequence(result)
    print(f"\nFINAL: score={final_score:.8f}, length={len(result)}, time={time.time()-t_start:.0f}s")
    return result
```
