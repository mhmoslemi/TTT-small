def solve_convex(n):
    a = cp.Variable(n, non_negative=True)
    t = cp.Variable()
    
    constraints = [cp.sum(a) == 1]
    
    # Add convolution constraints
    for k in range(2*n - 1):

def project_simplex(v):
    """Project v onto the probability simplex {x: x>=0, sum(x)=1}."""
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n+1) > (cssv - 1))[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0)

def gradient_max_conv(a, n):

                if k_j == j:
                    grad[j] += a[j]
                else:
                    grad[j] += a[k_j]
    grad /= len(active)

"""
Hybrid optimization for minimizing 2*n*max(conv(a,a))/(sum(a))^2.
Combines projected gradient with soft-max smoothing, coordinate descent,
LP-based dual updates, and multiple restarts with structured initializations.
"""

import numpy as np
from scipy.optimize import linprog
import time


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    conv = np.convolve(a, a)
    max_b = float(np.max(conv))
    sum_a = float(np.sum(a))
    if sum_a < 0.01:
        return np.inf
    return 2.0 * n * max_b / (sum_a * sum_a)


def project_simplex(v):
    """Project v onto the probability simplex {x: x>=0, sum(x)=1}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    rho_arr = np.nonzero(cond)[0]
    if len(rho_arr) == 0:
        return np.ones(n) / n
    rho = rho_arr[-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    w = np.maximum(v - theta, 0)
    # Ensure exact sum
    s = w.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return w / s


def conv_grad_at_k(a, k, n):
    """Compute gradient of conv_k w.r.t. a. grad[j] = 2*a[k-j] if 0<=k-j<n else 0."""
    grad = np.zeros(n)
    j_start = max(0, k - n + 1)
    j_end = min(n - 1, k)
    if j_start <= j_end:
        grad[j_start:j_end + 1] = 2.0 * a[k - j_end:k - j_start + 1][::-1]
    return grad


def soft_max_and_grad(a, kappa):
    """Compute soft max of conv(a,a) and its gradient w.r.t. a."""
    n = len(a)
    conv = np.convolve(a, a)
    max_conv = np.max(conv)
    shifted = conv - max_conv
    exp_vals = np.exp(shifted / kappa)
    sum_exp = np.sum(exp_vals)
    weights = exp_vals / sum_exp
    
    soft_max = max_conv + kappa * np.log(sum_exp)
    
    # Gradient: weighted sum of individual gradients
    grad = np.zeros(n)
    for k in range(len(conv)):
        w = weights[k]
        if w < 1e-12:
            continue
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        if j_start <= j_end:
            grad[j_start:j_end + 1] += w * 2.0 * a[k - j_end:k - j_start + 1][::-1]
    
    return soft_max, grad


def projected_gradient_step(a, kappa, eta, momentum=0.0):
    """One step of projected gradient with soft max."""
    n = len(a)
    fm, grad = soft_max_and_grad(a, kappa)
    # Project gradient onto tangent space of simplex (remove mean)
    grad_tangent = grad - np.mean(grad)
    return a - eta * grad_tangent, fm, grad


def backtracking_line_search(a, fm, grad_tangent, kappa, eta0, max_iter=50):
    """Backtracking line search for step size."""
    c1 = 1e-4
    rho = 0.5
    eta = eta0
    for _ in range(max_iter):
        a_new = project_simplex(a - eta * grad_tangent)
        fm_new, _ = soft_max_and_grad(a_new, kappa)
        if fm_new <= fm - c1 * eta * np.dot(grad_tangent, a - a_new):
            return a_new, eta
        eta *= rho
    return project_simplex(a - eta * grad_tangent), eta


def run_projected_gradient(n, a_init=None, max_iter=5000, kappa_start=1.0,
                           kappa_end=0.01, eta0=0.1, momentum_start=0.0,
                           momentum_end=0.5, seed=0):
    """Run projected gradient with soft max and annealing."""
    rng = np.random.default_rng(seed)
    
    if a_init is None:
        a = rng.random(n)
    else:
        a = a_init.copy()
    s = np.sum(a)
    if s < 1e-10:
        a = rng.random(n)
        s = np.sum(a)
    a = a / s
    
    best_a = a.copy()
    best_obj = 2.0 * n * np.max(np.convolve(a, a))  # since sum=1
    
    v = np.zeros(n)  # momentum
    eta = eta0
    
    for it in range(max_iter):
        # Anneal kappa
        progress = it / max_iter
        kappa = kappa_start * (kappa_end / kappa_start) ** progress
        mu = momentum_start + (momentum_end - momentum_start) * progress
        
        fm, grad = soft_max_and_grad(a, kappa)
        grad_tangent = grad - np.mean(grad)
        
        # Momentum update
        v = mu * v + (1 - mu) * grad_tangent
        
        # Line search
        a_new = None
        for ls_iter in range(30):
            a_trial = project_simplex(a - eta * v)
            fm_trial, _ = soft_max_and_grad(a_trial, kappa)
            if fm_trial <= fm - 1e-4 * eta * np.dot(v, a - a_trial):
                a_new = a_trial
                break
            eta *= 0.5
        if a_new is None:
            a_new = project_simplex(a - eta * v)
        
        # Check if we should increase eta
        if a_new is not None and np.linalg.norm(a_new - a) < 1e-10:
            eta = min(eta * 2.0, 1.0)
        
        a = a_new
        
        # Track best
        current_max_conv = np.max(np.convolve(a, a))
        current_obj = 2.0 * n * current_max_conv
        if current_obj < best_obj:
            best_obj = current_obj
            best_a = a.copy()
    
    return best_a, best_obj


def coordinate_descent(a, max_iter=200, seed=0):
    """Coordinate descent: optimize one coordinate at a time."""
    rng = np.random.default_rng(seed)
    n = len(a)
    a = a.copy()
    s = np.sum(a)
    a = a / s
    
    best_obj = compute_obj(a)
    order = rng.permutation(n)
    
    for it in range(max_iter):
        improved = False
        for idx in order:
            # Try different values for a[idx]
            # Keep others fixed, adjust a[idx] to maintain sum=1
            # Actually, let's just try perturbations
            for scale in [0.5, 0.7, 0.8, 1.2, 1.5, 2.0]:
                trial = a.copy()
                trial[idx] *= scale
                ts = np.sum(trial)
                if ts < 1e-10:
                    continue
                trial = trial / ts
                obj = compute_obj(trial)
                if obj < best_obj:
                    a = trial
                    best_obj = obj
                    improved = True
                    break
        if not improved:
            break
    
    return a, best_obj


def build_A_matrix(a):
    """Build constraint matrix for LP."""
    n = len(a)
    A = np.zeros((2 * n - 1, n), dtype=np.float64)
    for k in range(2 * n - 1):
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        if j_start <= j_end:
            A[k, j_start:j_end + 1] = a[k - j_end:k - j_start + 1][::-1]
    return A


def solve_lp_dual(a, upper_bound=10.0):
    """Solve LP: max sum(g) s.t. (a*g)_k <= M, 0 <= g_i <= ub."""
    n = len(a)
    conv = np.convolve(a, a)
    M = float(np.max(conv))
    A = build_A_matrix(a)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, upper_bound)] * n
    res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
    if res.status == 0:
        return res.x, float(-res.fun), M
    return None, None, M


def run_lp_iteration(n, a_init, max_iter=100, t=0.5, seed=0):
    """LP-based dual iteration."""
    rng = np.random.default_rng(seed)
    a = a_init.copy()
    s = np.sum(a)
    a = a / s
    best_a = a.copy()
    best_obj = compute_obj(best_a)
    
    stall = 0
    for it in range(max_iter):
        g, g_sum, M = solve_lp_dual(a)
        if g is None:
            break
        if g_sum <= 1.0 + 1e-5:
            stall += 1
            if stall > 5:
                # Perturb
                noise = np.exp(rng.normal(0, 0.5, n))
                a = np.clip(a * noise, 1e-10, None)
                a = a / np.sum(a)
                stall = 0
            continue
        stall = 0
        g = g / g_sum
        a_new = (1 - t) * a + t * g
        a_new = np.clip(a_new, 1e-10, None)
        a_new = a_new / np.sum(a_new)
        obj_new = compute_obj(a_new)
        if obj_new < best_obj:
            best_obj = obj_new
            best_a = a_new.copy()
        a = a_new
    
    return best_a, best_obj


def structured_init(n, rng, mode=0):
    """Generate structured initializations."""
    if mode == 0:
        # Random
        a = rng.random(n)
    elif mode == 1:
        # Cosine concentrated at edges
        x = np.linspace(0, 1, n)
        a = np.cos(np.pi * x) ** 2
        a = np.clip(a, 0, None)
    elif mode == 2:
        # Bimodal (two bumps at edges)
        centers = [n * 0.15, n * 0.85]
        a = np.zeros(n)
        for c in centers:
            a += np.exp(-((np.arange(n) - c) ** 2) / (2 * (n * 0.1) ** 2))
    elif mode == 3:
        # Power law (concentrated at one end)
        a = np.arange(1, n + 1) ** 2
    elif mode == 4:
        # Alternating
        a = np.ones(n)
        a[1::2] = 0.3
    else:
        a = rng.random(n)
    
    s = np.sum(a)
    if s < 1e-10:
        a = rng.random(n)
        s = np.sum(a)
    return a / s


def spectral_init(n, rng):
    """Initialize using the spectral (ideal) target, clipped to non-negative."""
    # The ideal P(x) satisfies P(x)^2 = 1/(2n-1) * (1 + x + ... + x^{2n-2})
    # The coefficients of P(x) are the "ideal" but may have negative values
    # We use the absolute values or clamp to 0 as initialization
    
    # Compute square root of the all-ones polynomial of degree 2n-2
    # Using the formula: if Q(x) = sum_{k=0}^{2n-2} c_k x^k with c_k = 1/(2n-1),
    # then P(x) = sqrt(Q(x)) can be computed via Newton's method on polynomials
    
    # For simplicity, use the "flat" initialization: uniform with slight perturbation
    a = np.ones(n)
    # Add slight structure to break symmetry
    a *= np.exp(rng.normal(0, 0.1, n))
    a = np.clip(a, 0, None)
    s = np.sum(a)
    a = a / s
    return a

Hybrid optimization for minimizing 2*n*max(conv(a,a))/(sum(a))^2.
Uses projected subgradient with soft-max smoothing, BB step sizes,
momentum, LP dual iterations, and coordinate descent.

import numpy as np
from scipy.optimize import linprog
import time


def compute_obj(a):
    n = len(a)
    if n == 0:
        return np.inf
    conv = np.convolve(a, a)
    max_b = float(np.max(conv))
    sum_a = float(np.sum(a))
    if sum_a < 0.01:
        return np.inf
    return 2.0 * n * max_b / (sum_a * sum_a)


def project_simplex(v):
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    rho_arr = np.nonzero(cond)[0]
    if len(rho_arr) == 0:
        return np.ones(n) / n
