Tries multiple lengths and initializations.

def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    N = 1 << (size - 1).bit_length()
    fa = np.fft.rfft(a, N)
    fc = fa * fa
    result = np.fft.irfft(fc, N)[:size]
    result[result < 0] = 0.0
    return result


def project_simplex(v):
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cv = np.cumsum(u) - 1.0
    rho_candidates = np.where(u * np.arange(1, n + 1) > cv)[0]
    if len(rho_candidates) == 0:
        return np.ones(n) / n
    rho = rho_candidates[-1]
    theta = cv[rho] / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def build_conv_matrix(a):
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for k in range(2 * n - 1):
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        A[k, lo:hi + 1] = a[k - hi:k - lo + 1][::-1]
    return A


def solve_lp(a, M, time_limit=5.0):
    n = len(a)
    A = build_conv_matrix(a)
    c_vec = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    try:
        res = linprog(c_vec, A_ub=A, b_ub=b_ub,
                      bounds=[(0.0, None)] * n,
                      method='highs',
                      options={'time_limit': time_limit, 'dual_feasibility_tolerance': 1e-8})
        if res.success and res.x is not None:
            b = np.maximum(res.x, 0.0)
            if b.sum() > 1e-10:
                return b
    except Exception:
        pass
    return None

    a_new = project_simplex(a - lr * grad)
    return a_new

    return a

**Let me try a different angle: what if I use the "cyclic" version of the problem as a guide?**

For cyclic convolution (i.e., FFT-based), the problem is:
`minimize max_k IFFT(|FFT(a)|^2)[k]` subject to `a >= 0, sum(a) = 1`.

The optimal solution for the cyclic version (without the non-negativity constraint) is a sequence whose FFT has constant magnitude (like a chirp or m-sequence). With the non-negativity constraint, it's harder.

But the cyclic version gives a lower bound on the linear version (since the linear convolution is a "truncated" version of the cyclic one, with zero-padding).

Actually, for zero-padded sequences, the linear convolution equals the first `2n-1` samples of the cyclic convolution (with period `>= 2n-1`). So if we can make the cyclic autocorrelation flat, the linear one will also be flat in the central region.

The issue is that the linear convolution has "edge effects" - the first and last few samples are smaller because the overlap is less. This actually works in our favor: the max of the linear convolution is likely in the central region, where it equals the cyclic autocorrelation.

So the strategy is:
1. Find a non-negative sequence whose *cyclic* autocorrelation is as flat as possible.
2. The linear autocorrelation will then be flat in the center, with lower values at the edges.
3. The max will be in the center, and will be approximately `1/(2n-1)` if perfectly flat.

To achieve a flat cyclic autocorrelation with a non-negative sequence, we need the FFT magnitude to be as flat as possible. This is the "minimum entropy spectral estimation" or "flat spectrum" problem.

"""
Optimizes non-negative sequence to minimize 2*n*max(conv(a,a))/sum(a)^2.
Uses LP-based alternating optimization with adaptive M tightening,
alternating projections initialization, and subgradient refinement.
Tries multiple sequence lengths and initialization strategies.
"""

import numpy as np
import time
from scipy.optimize import linprog


def fast_convolve(a):
    """FFT-based linear convolution of a with itself."""
    n = len(a)
    size = 2 * n - 1
    N = 1 << (size - 1).bit_length()
    fa = np.fft.rfft(a, N)
    fc = fa * fa
    result = np.fft.irfft(fc, N)[:size]
    result[result < 0] = 0.0
    return result


def project_simplex(v):
    """Project vector onto {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cv = np.cumsum(u) - 1.0
    rho_candidates = np.where(u * np.arange(1, n + 1) > cv)[0]
    if len(rho_candidates) == 0:
        return np.ones(n) / n
    rho = rho_candidates[-1]
    theta = cv[rho] / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def build_conv_matrix(a):
    """Build (2n-1) x n matrix A where A[k,j] = a[k-j] if valid."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for k in range(2 * n - 1):
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        A[k, lo:hi + 1] = a[k - hi:k - lo + 1][::-1]
    return A


def solve_lp(a, M, time_limit=4.0):
    """Solve: max sum(b) s.t. conv(a,b) <= M, b >= 0. Return b or None."""
    n = len(a)
    A = build_conv_matrix(a)
    c_vec = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    try:
        res = linprog(c_vec, A_ub=A, b_ub=b_ub,
                      bounds=[(0.0, None)] * n,
                      method='highs',
                      options={'time_limit': time_limit})
        if res.success and res.x is not None:
            b = np.maximum(res.x, 0.0)
            if b.sum() > 1e-10:
                return b
    except Exception:
        pass
    return None


def compute_subgradient(a, peak_indices):
    """Compute average subgradient over active peak indices."""
    n = len(a)
    grad = np.zeros(n)
    for k in peak_indices:
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        grad[lo:hi + 1] += 2.0 * a[k - hi:k - lo + 1][::-1]
    grad /= max(1, len(peak_indices))
    return grad


def alt_proj_init(n, n_iter=120, seed=None):
    """Alternating projections for flat-spectrum non-negative sequence."""
    rng = np.random.RandomState(seed if seed is not None else np.random.randint(1e6))
    N = 1 << (2 * n - 2).bit_length()
    phases = rng.uniform(0.0, 2.0 * np.pi, N // 2 + 1)
    F = np.exp(1j * phases)
    F[0] = 1.0 + 0j
    a = np.ones(n) / n
    for it in range(n_iter):
        a_full = np.fft.irfft(F, N)
        a = np.maximum(a_full[:n], 0.0)
        s = a.sum()
        if s < 1e-10:
            a = np.ones(n) / n
        else:
            a = a / s
        F_full = np.fft.rfft(a, N)
        mag = np.abs(F_full)
        # Adaptive flattening: stronger early, gentler late
        alpha = 0.15 * (1.0 - float(it) / n_iter) + 0.02
        mean_mag = mag.mean()
        mag_target = (1.0 - alpha) * mag + alpha * mean_mag
        F = mag_target * np.exp(1j * np.angle(F_full))
        F[0] = F[0].real + 0j
    return a


def optimize_lp_based(a, deadline, lp_time=4.0, verbose=False):
    """
    Pure LP-based alternating optimization.
    At each step: solve LP for b, then update a toward b.
    """
    n = len(a)
    a = project_simplex(a.copy().astype(np.float64))
    c = fast_convolve(a)
    best_peak = c.max()
    best_a = a.copy()
    lp_calls = 0
    fail_count = 0
    max_fail = 10

    while time.time() < deadline and lp_calls < 2000:
        c = fast_convolve(a)
        peak = c.max()

        if peak < best_peak - 1e-9:
            best_peak = peak
            best_a = a.copy()
            fail_count = 0
            if verbose and lp_calls % 50 == 0:
                print(f"      lp_iter={lp_calls}, peak={peak:.10f}, score={2*n*peak:.6f}")

        # Try LP with progressively tighter M
        M = peak * (1.0 - 1e-6)
        b = solve_lp(a, M, time_limit=lp_time)

        if b is None:
            fail_count += 1
            if fail_count >= max_fail:
                # Perturb and retry
                a = project_simplex(a + 0.01 / np.sqrt(n) * np.random.randn(n))
                fail_count = 0
            continue

        lp_calls += 1
        b_norm = b / b.sum()

        # Try different mixing ratios
        best_new_peak = peak
        best_new_a = a
        for t_mix in [0.7, 0.5, 0.3]:
            a_candidate = project_simplex((1.0 - t_mix) * a + t_mix * b_norm)
            c_cand = fast_convolve(a_candidate)
            if c_cand.max() < best_new_peak:
                best_new_peak = c_cand.max()
                best_new_a = a_candidate

        if best_new_peak < peak - 1e-12:
            a = best_new_a
        else:
            # Try with lower M
            M2 = peak * 0.998
            b2 = solve_lp(a, M2, time_limit=lp_time)
            if b2 is not None:
                lp_calls += 1
                b2_norm = b2 / b2.sum()
                for t_mix in [0.6, 0.4]:
                    a_candidate = project_simplex((1.0 - t_mix) * a + t_mix * b2_norm)
                    c_cand = fast_convolve(a_candidate)
                    if c_cand.max() < best_new_peak:
                        best_new_peak = c_cand.max()
                        best_new_a = a_candidate
                if best_new_peak < peak - 1e-12:
                    a = best_new_a

    return best_a


def optimize_mixed(a, deadline, lp_time=3.0, verbose=False):
    """
    Mixed LP + subgradient optimization.
    Uses LP every few steps for major improvements, subgradient for fine-tuning.
    """
    n = len(a)
    a = project_simplex(a.copy().astype(np.float64))
    c = fast_convolve(a)
    best_peak = c.max()
    best_a = a.copy()

    lr = 2.0 / n
    lr_decay = 0.95
    stall = 0
    max_stall = 30
    lp_interval = 5
    it = 0
    lp_calls = 0

    while time.time() < deadline and it < 300000:
        it += 1
        c = fast_convolve(a)
        peak = c.max()

        if peak < best_peak - 1e-8:
            best_peak = peak
            best_a = a.copy()
            stall = 0
            if verbose and it % 5000 == 0:
                print(f"      it={it}, peak={peak:.10f}, score={2*n*peak:.6f}")

        # LP step periodically
        if it % lp_interval == 0 and lp_calls < 1000 and time.time() < deadline - 15:
            M = peak
            b = solve_lp(a, M, time_limit=lp_time)
            if b is not None:
                lp_calls += 1
                b_norm = b / b.sum()
                a_candidate = project_simplex(0.5 * a + 0.5 * b_norm)
                c_cand = fast_convolve(a_candidate)
                if c_cand.max() < peak:
                    a = a_candidate
                    lr = max(lr, 1.0 / n)
                    stall = 0
                    continue

        # Subgradient step
        thresh = peak * (1.0 - 5e-5)
        active = np.where(c >= thresh)[0]
        if len(active) > 25:
            active = active[np.argsort(c[active])[::-1][:12]]
        grad = compute_subgradient(a, active)
        a = project_simplex(a - lr * grad)

        # Adaptive learning rate
        if stall > max_stall:
            lr *= lr_decay
            stall = 0
            if lr < 1e-6 / n:
                a = project_simplex(a + 0.03 / np.sqrt(n) * np.random.randn(n))
                lr = 2.0 / n

        stall += 1

    return best_a


def refine_sequence(a, deadline, verbose=False):
    """
    Full refinement: alternate between LP-based and mixed optimization,
    with periodic re-initialization to escape local minima.
    """
    n = len(a)
    a = project_simplex(a.copy().astype(np.float64))
    c = fast_convolve(a)
    best_peak = c.max()
    best_a = a.copy()

    use_lp = n <= 3000
    time_remaining_func = lambda: deadline - time.time()

    # Phase 1: LP-based optimization (bulk of the time)
    if use_lp:
        t1 = time.time() + time_remaining_func() * 0.5
        a1 = optimize_lp_based(a, t1, lp_time=3.0, verbose=verbose)
        c1 = fast_convolve(a1)
        if c1.max() < best_peak:
            best_peak = c1.max()
            best_a = a1.copy()

    # Phase 2: Mixed optimization on the best so far
    a2_init = best_a.copy()
    t2 = time.time() + (deadline - time.time()) * 0.6
    a2 = optimize_mixed(a2_init, t2, lp_time=2.0, verbose=verbose)
    c2 = fast_convolve(a2)
    if c2.max() < best_peak:
        best_peak = c2.max()
        best_a = a2.copy()

    # Phase 3: Final LP polish
    if use_lp and time.time() < deadline - 5:
        a3 = optimize_lp_based(best_a, deadline, lp_time=2.0, verbose=verbose)
        c3 = fast_convolve(a3)
        if c3.max() < best_peak:
            best_peak = c3.max()
            best_a = a3.copy()

    return best_a
