    """Compute softmax(x/tau) stably."""
    x = np.asarray(x, dtype=np.float64)
    x = x / tau
    x = x - np.max(x)
    e = np.exp(x)
    return e / np.sum(e)

    q = np.convolve(a, a)
    max_q = float(np.max(q))

def project_simplex(v):
    """Project v onto {x >= 0, sum(x) = 1}."""
    v = np.asarray(v, dtype=np.float64)
    n = len(v)
    if n == 0:
        return v.copy()
    # Algorithm from Duchi et al. (2012)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n+1) > (cssv - 1))[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0)

def normalize_sum1(a):
    a = np.asarray(a, dtype=np.float64)
    a = np.clip(a, 0, 1000)
    s = float(np.sum(a))
    if s <= 1e-12:
        return np.ones(len(a)) / len(a)
    return a / s

"""
Minimize score = 2n * max(conv(a,a)) / sum(a)^2 using:
  - Log-sum-exp smoothing with continuation (tau annealing)
  - FISTA (accelerated projected gradient) with backtracking line search
  - FFT-based convolution for efficiency
  - Multiplicative weight updates for fine-tuning
  - Multiple sequence lengths and restarts
  - Spectral flatness initialization
"""

import time
import numpy as np

# ============================================================
# Core utility functions
# ============================================================

def _fft_convolve(a, b):
    """Fast linear convolution using FFT."""
    n = len(a) + len(b) - 1
    size = 1 << (n - 1).bit_length()
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    return np.fft.irfft(fa * fb, size)[:n]


def _self_conv(a):
    """Compute a * a (linear autocorrelation) via FFT."""
    n = len(a)
    size = 1 << (2 * n - 2).bit_length()
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]


def local_score(a):
    """Compute the exact score for a sequence."""
    a = np.asarray(a, dtype=np.float64)
    if a.size == 0:
        return float('inf')
    a = np.clip(a, 0.0, 1000.0)
    s = float(np.sum(a))
    if s <= 1e-12:
        return float('inf')
    c = _self_conv(a)
    m = float(np.max(c))
    return float(2.0 * len(a) * m / (s * s))


def project_simplex(v):
    """Project v onto {x >= 0, sum(x) = 1} in O(n log n)."""
    v = np.asarray(v, dtype=np.float64)
    n = len(v)
    if n == 0:
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho_candidates = np.nonzero(u * np.arange(1, n + 1) > (cssv - 1.0))[0]
    if len(rho_candidates) == 0:
        theta = (v.sum() - 1.0) / n
        return np.maximum(v - theta, 0.0)
    rho = rho_candidates[-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def normalize_sum1(a):
    """Normalize a to have sum 1, clip to non-negative."""
    a = np.asarray(a, dtype=np.float64)
    a = np.clip(a, 0.0, 1000.0)
    s = float(np.sum(a))
    if s <= 1e-300:
        return np.ones(len(a), dtype=np.float64) / len(a)
    return a / s

# Smoothed objective and gradient
# ============================================================

def smooth_obj_and_p(a, tau):
    """Return (smoothed_obj, softmax_weights_p) of max(conv(a,a))."""
    q = _self_conv(a)
    max_q = float(np.max(q))
    if max_q <= 0:
        return 0.0, np.ones(len(q)) / len(q)
    shifted = tau * (q - max_q)
    e = np.exp(shifted)
    p = e / np.sum(e)
    obj = max_q + np.log(np.sum(e)) / tau
    return obj, p


def smooth_grad(a, tau, p=None):
    """Gradient of (1/tau)*log(sum exp(tau*q_k)) w.r.t. a.
    
    g[i] = 2 * sum_j a[j] * p[i+j]  (for valid i+j)
    Computed via FFT convolution.
    """
    n = len(a)
    if p is None:
        _, p = smooth_obj_and_p(a, tau)
    # g[i] = 2 * sum_{j=0}^{n-1} a[j] * p[i+j]
    # = 2 * conv(a, p_rev)[2n-2-i]  where p_rev = p[::-1]
    # Equivalently: g = 2 * conv(a, p[::-1], 'full')[2n-2 : n-2 : -1]
    conv = _fft_convolve(a, p[::-1])
    # conv has length n + (2n-1) - 1 = 3n-2
    # We want indices 2n-2, 2n-3, ..., n-2  (n elements)
    g = 2.0 * conv[2 * n - 2 : n - 2 : -1]
    return g

# FISTA optimization with backtracking

def fista_optimize(a, tau, n_iter, verbose=False):
    """FISTA for min f_tau(a) s.t. a in simplex.
    
    Returns (best_a, best_score).
    """
    a = project_simplex(a.copy())
    n = len(a)
    
    # Initial Lipschitz estimate
    q = _self_conv(a)
    L = 4.0 * tau * float(np.max(q)) * 2.0 + 1e-10
    L = max(L, 1e-10)
    
    alpha = 1.0 / L
    x = a.copy()
    y = a.copy()
    t = 1.0
    best_x = a.copy()
    _, best_obj = smooth_obj_and_p(a, tau)
    
    for k in range(n_iter):
        obj_y, p_y = smooth_obj_and_p(y, tau)
        g = smooth_grad(y, tau, p_y)
        
        # Backtracking line search
        z = x  # will be updated
        for bs in range(30):
            z = project_simplex(y - alpha * g)
            obj_z, _ = smooth_obj_and_p(z, tau)
            # Quadratic model check
            model = obj_y + g.dot(z - y) + 0.5 * (1.0 / alpha) * np.sum((z - y) ** 2)
            if obj_z <= model + 1e-15:
                break
            alpha *= 0.5
        L = max(L * 2.0, 1e-10)
        
        # FISTA momentum update
        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t * t))
        y = z + ((t - 1.0) / t_new) * (z - x)
        y = project_simplex(y)
        x = z
        t = t_new
        
        if obj_z < best_obj:
            best_obj = obj_z
            best_x = z.copy()
        elif (k + 1) % 500 == 0 and best_x is not x:
            # Periodically reset if stuck
            pass
    
    return best_x, best_obj


def fista_with_continuation(a, tau_schedule, iters_per_tau, total_time_limit, seed=0):
    """Run FISTA with continuation on tau. Returns best sequence found."""
    a = project_simplex(a.copy())
    best = a.copy()
    best_score = local_score(best)
    t0 = time.monotonic()
    
    for tau in tau_schedule:
        rem = total_time_limit - (time.monotonic() - t0)
        if rem < 1.0:
            break
        # Scale iterations by remaining time
        n_iter = max(50, int(iters_per_tau * min(1.0, rem / total_time_limit)))
        
        result_a, smooth_val = fista_optimize(a, tau, n_iter)
        result_score = local_score(result_a)
        
        if result_score < best_score:
            best = result_a.copy()
            best_score = result_score
            a = result_a.copy()
            if verbose:
                print(f"  tau={tau:.0f}: score={result_score:.6f}", flush=True)
    
    return best, best_score

def multiplicative_update(a, eta, K=None):
    """Multiplicative weight update targeting active convolution peaks."""
    a = np.asarray(a, dtype=np.float64)
    n = a.size
    if n == 0:
        return a.copy()
    
    q = _self_conv(a)
    M = float(np.max(q))
    if M <= 0 or not np.isfinite(M):
        return a.copy()
    
    if K is None:
        K = max(4, n // 8)
    K = min(K, len(q))
    
    active = np.argpartition(q, -K)[-K:]
    w = q[active] / M
    
    h = np.zeros(n, dtype=np.float64)
    for idx in range(active.size):
        k = int(active[idx])
        wk = float(w[idx])
        s = max(0, k - (n - 1))
        e = min(k, n - 1)
        if s > e:
            continue
        i = np.arange(s, e + 1)
        j = k - i
        h[i] += wk * a[j]
        h[j] += wk * a[i]
    
    maxh = float(np.max(h))
    if maxh <= 0:
        return a.copy()
    
    meanh = float(np.mean(h))
    fac = eta * (meanh - h) / maxh
    fac = np.clip(fac, -3.0, 3.0)
    a2 = a * np.exp(fac)
    return project_simplex(a2)


def multiplicative_optimize(a, max_time, seed=0):
    """Iterative multiplicative updates with adaptive parameters."""
    a = project_simplex(a.copy())
    best = a.copy()
    bestsc = local_score(best)
    t0 = time.monotonic()
    rng = np.random.default_rng(seed)
    n = best.size
    
    eta = 0.4
    K = max(8, n // 6)
    stuck = 0
    
    while time.monotonic() - t0 < max_time:
        if stuck > 50:
            eta = max(0.05, eta * 0.7)
            K = max(4, int(K * 0.8))
            if stuck > 150:
                # Restart with noise
                noise = np.exp(rng.normal(0, 0.1, n))
                a = project_simplex(best * noise)
                eta = 0.4
                K = max(8, n // 6)
                stuck = 0
        
        a = multiplicative_update(a, eta, K)
        sc = local_score(a)
        
        if sc < bestsc:
            best = a.copy()
            bestsc = sc
            stuck = 0
            if eta < 0.8:
                eta *= 1.02
        else:
            stuck += 1
    
    return best, bestsc

def lbfgs_optimize(a, tau, maxiter=200):
    """Use scipy L-BFGS-B on the smoothed problem (no explicit constraint,
    rely on projection happening implicitly via the smooth landscape)."""
    from scipy.optimize import minimize as sp_min
    
    n = len(a)
    
    def obj_and_grad(x):
        x = np.clip(x, 0, 1000)
        o, p = smooth_obj_and_p(x, tau)
        g = smooth_grad(x, tau, p)
        return o, g
    
    res = sp_min(
        fun=lambda x: obj_and_grad(x)[0],
        x0=a,
        jac=lambda x: obj_and_grad(x)[1],
        method='L-BFGS-B',
        bounds=[(0, 1000)] * n,
        options={'maxiter': maxiter, 'ftol': 1e-18, 'gtol': 1e-12, 'maxls': 50}
    )
    
    if res.success or res.fun < smooth_obj_and_p(a, tau)[0]:
        return project_simplex(res.x), local_score(project_simplex(res.x))
    return a, local_score(a)

def init_uniform(n, rng):
    return np.ones(n, dtype=np.float64) / n


def init_rand_nonneg(n, rng):

def init_flat_spectrum(n, rng, n_iter=200):
    """Initialize via iterative projection: flat |A|^2 in freq, non-negative in time."""
    a = project_simplex(rng.uniform(0.5, 1.0, n))
    for _ in range(n_iter):
        # Compute DFT
        A = np.fft.rfft(a, 2 * n)
        # Target: flat magnitude spectrum
        target_mag = np.ones(len(A))
        # Update spectrum to have flat magnitude
        phase = A / (np.abs(A) + 1e-30)
        A_new = target_mag * phase
        # Inverse DFT
        a_new = np.fft.irfft(A_new, 2 * n)[:n]
        # Project to non-negative, sum 1
        a = project_simplex(np.clip(a_new, 0, None))
    return a


def init_from_parent(parent, n, rng):
    """Resample/interpolate parent to length n."""
    if parent is None or len(parent) != n:
        if parent is not None and len(parent) > 0:
            old = np.linspace(0, 1, len(parent))
            new = np.linspace(0, 1, n)
            a = np.interp(new, old, parent)
            return project_simplex(np.clip(a, 0, None))
        return project_simplex(rng.uniform(0.1, 1.0, n))
    else:
        return project_simplex(parent.copy())

def coordinate_descent(a, n_sweeps=3, time_limit=30.0):
    """Exact coordinate descent: for each i, find optimal a[i] given others fixed."""
    a = project_simplex(a.copy())
    n = len(a)
    t0 = time.monotonic()
    best = a.copy()
    bestsc = local_score(best)
    
    for sweep in range(n_sweeps):
        if time.monotonic() - t0 > time_limit:
            break
        improved = False
        for i in range(n):
            if time.monotonic() - t0 > time_limit:
                    break
            old_val = a[i]
            if old_val <= 0:
                continue
            
            # Compute q with a[i] = 0
            a[i] = 0.0
            q_base = _self_conv(a)
            S_base = float(np.sum(a))  # = 1 - old_val (since we're on simplex)
            
            # q_k as function of x = a[i]:
            # q_k(x) = q_base[k] + x * (a[k-i] + a[i-k]) + x^2 * delta(k==2i)
            # (where a[j] = 0 if j out of range)
            # max_k q_k(x) / (S_base + x)^2  * 2n  = score(x)
            
            # Find optimal x by sampling + golden section
            # x is in [0, 1] (on simplex, max value is 1)
            lo = 0.0
            hi = min(1.0, S_base + old_val)  # can't exceed 1
            
            # Golden section search
            gr = (np.sqrt(5.0) - 1.0) / 2.0
            c1 = hi - gr * (hi - lo)
            c2 = lo + gr * (hi - lo)
            
            def score_1d(x):
                a[i] = x
                q = _self_conv(a)
                # Add back the cross terms (already included in _self_conv)
                s = S_base + x
                if s < 1e-15:
                    return float('inf')
                return 2.0 * n * float(np.max(q)) / (s * s)
            
            f1 = score_1d(c1)
            f2 = score_1d(c2)
            
            for _ in range(30):
                if f1 < f2:
                    hi = c2
                    c2 = c1
                    f2 = f1
                    c1 = hi - gr * (hi - lo)
                    f1 = score_1d(c1)
                else:
                    lo = c1
                    c1 = c2
                    f1 = f2
                    c2 = lo + gr * (hi - lo)
                    f2 = score_1d(c2)
            
            x_opt = (lo + hi) / 2.0
            a[i] = x_opt
            a = project_simplex(a)
            sc = local_score(a)
            if sc < bestsc:
                best = a.copy()
                bestsc = sc
                improved = True
            a[i] = old_val  # restore for next iteration (or use new if improved)
            a = project_simplex(a)
        
        if not improved and sweep > 0:
            break
    
    return best, bestsc

def search_length(n, budget, seed, parent=None, force_time=0.0):

        a = project_simplex(a)
        sc = local_score(a)
        if np.isfinite(sc) and sc < best_score:
            best = a.copy()
            best_score = sc
            if label:
                print(f"    [{label}] n={n} score={sc:.6f}", flush=True)
        return a, sc
    
    # --- Initialization and warm starts ---
    inits = []
    
    # From parent (resampled)
    if parent is not None and len(parent) > 0:
        p = init_from_parent(parent, n, rng)

        try_candidate(init_a, f"init_{i}")
    
    if best is None:
        best = np.ones(n) / n
        best_score = local_score(best)
    
    # --- Phase 1: FISTA with continuation ---
    rem = deadline - time.monotonic()
    if rem > 5.0:
        tau_schedule = [1.0, 3.0, 10.0, 30.0, 100.0, 300.0, 1000.0]
        iters_per_tau = max(100, min(3000, int(rem / len(tau_schedule) * 500)))
        
        a_start = best.copy()
        fista_budget = min(rem * 0.5, 120.0)
        try:
            fista_a, fista_sc = fista_with_continuation(
                a_start, tau_schedule, iters_per_tau, fista_budget
            )
            try_candidate(fista_a, "fista")
        except Exception as e:
            print(f"    FISTA error: {e}", flush=True)
    
    # --- Phase 2: L-BFGS (for smaller n) ---
    rem = deadline - time.monotonic()
    if rem > 5.0 and n <= 1024:
        for tau in [50.0, 200.0, 1000.0]:
            if time.monotonic() > deadline - 3.0:
                break
            try:
                lb_a, lb_sc = lbfgs_optimize(best, tau, maxiter=100)
                try_candidate(lb_a, f"lbfgs_tau{tau:.0f}")
            except Exception as e:
                pass
    
    # --- Phase 3: Multiplicative update ---
    rem = deadline - time.monotonic()
    if rem > 3.0:
        mul_budget = min(rem * 0.3, 60.0)
        try:
            mul_a, mul_sc = multiplicative_optimize(best, mul_budget, seed + 777)
            try_candidate(mul_a, "multiplicative")
        except Exception as e:
            print(f"    Multiplicative error: {e}", flush=True)
    
    # --- Phase 4: FISTA again with higher tau (polish) ---
    rem = deadline - time.monotonic()
    if rem > 5.0:
        tau_schedule2 = [500.0, 2000.0, 5000.0, 10000.0]
        fista_budget2 = min(rem * 0.4, 90.0)
        iters2 = max(200, min(5000, int(fista_budget2 * 200)))
        try:
            f2_a, f2_sc = fista_with_continuation(
                best, tau_schedule2, iters2, fista_budget2
            )
            try_candidate(f2_a, "fista_polish")
        except Exception as e:
            pass
    
    # --- Phase 5: Coordinate descent (for smaller n) ---
    rem = deadline - time.monotonic()
    if rem > 5.0 and n <= 512:
        try:
            cd_a, cd_sc = coordinate_descent(best, n_sweeps=2, time_limit=min(rem * 0.4, 30.0))
            try_candidate(cd_a, "coord_descent")
        except Exception as e:
            pass
    
    # --- Phase 6: Final multiplicative polish ---
    rem = deadline - time.monotonic()
    if rem > 3.0:
        mul_budget2 = min(rem * 0.5, 30.0)
        try:
            mul2_a, mul2_sc = multiplicative_optimize(best, mul_budget2, seed + 999)
            try_candidate(mul2_a, "final_mul")
        except Exception:
            pass
    
    return best, best_score

# Main propose_candidate function

def propose_candidate(seed=42, budget_s=1500, **kwargs):
    """
    Search for low-score non-negative step-function coefficients.
    Uses smooth optimization (log-sum-exp + FISTA), multiplicative updates,
    coordinate descent, and L-BFGS across multiple sequence lengths.
    """
    start_time = time.monotonic()
    budget = float(budget_s)
    if not np.isfinite(budget) or budget <= 0:
        budget = 1500.0
    deadline = start_time + budget - 5.0  # Leave 5s margin
    
    # Get parent
    parent = None
    try:
        parent = globals().get('height_sequence_1', None)
    except Exception:
        pass
    
    parent_arr = None
    if parent is not None:
        try:
            parent_arr = np.asarray(parent, dtype=np.float64)
            parent_arr = np.clip(parent_arr, 0.0, 1000.0)
            if parent_arr.size == 0 or not np.isfinite(parent_arr).all():
                parent_arr = None
        except Exception:
            parent_arr = None
    
    best_seq = None
    best_score = float('inf')
    
    if parent_arr is not None:
        p = normalize_sum1(parent_arr)
        if p is not None:
            best_seq = p.copy()
            best_score = local_score(p)
            print(f"Parent: n={p.size}, score={best_score:.6f}", flush=True)
    
    # Length plan: (length, time_fraction)
    # Allocate more time to lengths near the parent's length
    parent_len = len(parent_arr) if parent_arr is not None else 512
    
    plan = [
        (256, 0.10),
        (512, 0.35),
        (768, 0.10),
        (1024, 0.25),
        (2048, 0.20),
    ]
    
    # Adjust plan: give more time to the parent length
    adjusted_plan = []
    for n, frac in plan:
        if n == parent_len:
            adjusted_plan.append((n, frac + 0.10))
        else:
            adjusted_plan.append((n, max(0.05, frac)))
    
    total_frac = sum(f for _, f in adjusted_plan)
    adjusted_plan = [(n, f / total_frac) for n, f in adjusted_plan]
    
    search_budget = budget - 30.0  # Reserve 30s for final polish
    
    seed_cur = int(seed)
    for n, frac in adjusted_plan:
        rem = deadline - time.monotonic()
        if rem < 10.0:
            break
        alloc = min(frac * search_budget, rem - 10.0)
        if alloc < 5.0:
            continue
        
        print(f"\n=== Searching n={n}, budget={alloc:.1f}s ===", flush=True)
        try:
            cand, sc = search_length(n, alloc, seed_cur, parent_arr)
            if cand is not None and np.isfinite(sc) and sc < best_score:
                best_seq = cand.copy()
                best_score = sc
                print(f"  NEW BEST: n={n}, score={best_score:.6f}", flush=True)
            else:
                print(f"  Best for n={n}: {sc:.6f} (not improving global best {best_score:.6f})", flush=True)
        except Exception as e:
            print(f"  Error for n={n}: {e}", flush=True)
        
        seed_cur += 1013
    
    # --- Final polishing on the globally best sequence ---
    rem = deadline - time.monotonic()
    if best_seq is not None and rem > 5.0:
        print(f"\n=== Final polish (budget={rem:.1f}s) ===", flush=True)
        
        # FISTA polish with very high tau
        if rem > 10.0:
            try:
                tau_final = [2000.0, 5000.0, 10000.0, 20000.0]
                fista_b = min(rem * 0.4, 60.0)
                iters = max(500, int(fista_b * 300))
                fa, fs = fista_with_continuation(best_seq, tau_final, iters, fista_b)
                if fs < best_score:
                    best_seq = fa
                    best_score = fs
                    print(f"  FISTA polish: {best_score:.6f}", flush=True)
            except Exception as e:
                print(f"  FISTA polish error: {e}", flush=True)
        
        # Multiplicative polish
        rem2 = deadline - time.monotonic()
        if rem2 > 5.0:
            try:
                mb = min(rem2 * 0.4, 30.0)
                ma, ms = multiplicative_optimize(best_seq, mb, seed_cur + 4242)
                if ms < best_score:
                    best_seq = ma
                    best_score = ms
                    print(f"  Mul polish: {best_score:.6f}", flush=True)
            except Exception:
                pass
        
        # Try L-BFGS on best
        rem3 = deadline - time.monotonic()
        if rem3 > 5.0 and len(best_seq) <= 1024:
            try:
                la, ls = lbfgs_optimize(best_seq, tau=5000.0, maxiter=100)
                if ls < best_score:
                    best_seq = la
                    best_score = ls
                    print(f"  L-BFGS polish: {best_score:.6f}", flush=True)
            except Exception:
                pass
    
    # --- Validate with official evaluator ---
    if best_seq is None:
        best_seq = np.ones(512, dtype=np.float64) / 512.0
    
    final = normalize_sum1(best_seq)
    final = np.clip(final, 0.0, 1000.0)
    
    try:
        official = float(evaluate_sequence([float(x) for x in final]))
        print(f"\nFinal: n={len(final)}, local_score={local_score(final):.6f}, official={official:.6f}", flush=True)
    except Exception as e:
        print(f"Validation error: {e}", flush=True)
    
    return [float(x) for x in final]
