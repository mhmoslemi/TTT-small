"""
Mirror descent + LP hybrid optimizer for minimizing 2*n*max(conv(a,a))/(sum(a))^2.
Uses log-sum-exp smoothing of the max, multiplicative updates (mirror descent),
and FFT-based convolution for efficiency. Combines with LP for small n and
random restarts for escaping local optima.
"""

import numpy as np
from scipy.optimize import linprog
from scipy.signal import fftconvolve
import time

def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    conv = fftconvolve(a, a)
    max_b = float(np.max(conv))
    sum_a = float(np.sum(a))
    if sum_a < 0.01:
        return np.inf
    return 2.0 * n * max_b / (sum_a * sum_a)

def project_simplex(v):
    """Project v onto the probability simplex."""
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    rho = np.where(cond)[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1)
    return np.clip(v - theta, 0, None)

def mirror_descent(p, n_steps, eta=1.0, eps=10.0, eps_decay=0.999,
                   perturb_freq=500, perturb_strength=0.1, rng=None):
    """
    Mirror descent on the simplex to minimize 2n*max(p*p).
    """
    if rng is None:
        rng = np.random.default_rng()
    n = len(p)
    p = p.copy()
    p = np.clip(p, 0, None)
    s = p.sum()
    if s < 1e-10:
        p = np.ones(n) / n
    else:
        p /= s
    
    best_p = p.copy()
    best_obj = 2.0 * n * float(np.max(fftconvolve(p, p)))
    
    for t in range(n_steps):
        eps_t = eps * (eps_decay ** t)
        eta_t = eta / (1.0 + t / 2000.0)
        
        # Compute autocorrelation
        c = fftconvolve(p, p)
        c_max = c.max()
        
        # Softmax weights for smooth max
        c_shifted = c - c_max
        w = np.exp(eps_t * c_shifted)
        w_sum = w.sum()
        if w_sum < 1e-30:
            break
        w /= w_sum
        
        # Compute gradient g_j = sum_k w_k * p_{k-j}
        g = np.zeros(n)
        # Only process significant weights
        active = np.where(w > 1e-8)[0]
        for k in active:
            wk = w[k]
            j_start = max(0, k - n + 1)
            j_end = min(n - 1, k)
            if j_start <= j_end:
                g[j_start:j_end + 1] += wk * p[k - j_end:k - j_start + 1][::-1]
        
        # Mirror descent update (multiplicative)
        # Scale eta so that the update is reasonable
        g_scale = g.max()
        if g_scale < 1e-15:
            break
        log_update = -eta_t * g / g_scale
        # Clip to prevent overflow/underflow
        log_update = np.clip(log_update, -50, 50)
        p = p * np.exp(log_update)
        s = p.sum()
        if s < 1e-30:
            p = np.ones(n) / n
            s = 1.0
        p /= s
        
        # Evaluate
        c = fftconvolve(p, p)
        obj = 2.0 * n * float(c.max())
        if obj < best_obj:
            best_obj = obj
            best_p = p.copy()
        
        # Periodic perturbation to escape local minima
        if (t + 1) % perturb_freq == 0:
            noise = np.exp(rng.normal(0, perturb_strength, n))
            p = p * noise
            p /= p.sum()
    
    return best_p, best_obj

def lp_optimize(a, max_iter=200, verbose=False):
    """LP-based alternating optimization (from previous code)."""
    n = len(a)
    a = a.copy()
    a = np.clip(a, 0, None)
    s = a.sum()
    if s < 1e-10:
        return a, np.inf
    a /= s
    
    best_a = a.copy()
    best_obj = compute_obj(best_a)
    
    for it in range(max_iter):
        conv = fftconvolve(a, a)
        M = float(conv.max())
        
        # Build constraint matrix
        c_len = 2 * n - 1
        A = np.zeros((c_len, n))
        for k in range(c_len):
            j_start = max(0, k - n + 1)
            j_end = min(n - 1, k)
            if j_start <= j_end:
                A[k, j_start:j_end + 1] = a[k - j_end:k - j_start + 1][::-1]
        
        c_obj = -np.ones(n)
        b_ub = np.full(c_len, M)
        bounds = [(0, None)] * n
        
        res = linprog(c_obj, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
        if res.status != 0:
            break
        
        g = res.x
        g_sum = float(g.sum())
        if g_sum <= 1.0 + 1e-6:
            break
        
        # Blend
        g = g / g_sum
        t_blend = 0.5
        a_new = (1 - t_blend) * a + t_blend * g
        a_new = np.clip(a_new, 0, None)
        s = a_new.sum()
        if s < 1e-10:
            break
        a_new /= s
        
        obj = compute_obj(a_new)
        if obj < best_obj:
            best_obj = obj
            best_a = a_new.copy()
        a = a_new
    
    return best_a, best_obj

def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:

    conv = fftconvolve(a, a)
    max_b = float(np.max(conv))
    sum_a = float(np.sum(a))
    if sum_a < 0.01:

    return 2.0 * n * max_b / (sum_a * sum_a)

    elif method == 'bimodal':
        x = np.arange(n)
        c1 = n * rng.uniform(0.2, 0.4)
        c2 = n * rng.uniform(0.6, 0.8)
        sigma = n * rng.uniform(0.05, 0.15)
        p = np.exp(-0.5 * ((x - c1) / sigma) ** 2) + np.exp(-0.5 * ((x - c2) / sigma) ** 2)
    elif method == 'random_smooth':
        p = rng.random(n)
        # Smooth it
        kernel_size = max(1, n // 20)
        kernel = np.ones(kernel_size) / kernel_size
        p = np.convolve(p, kernel, mode='same')
    else:
        p = rng.random(n)

    p = np.clip(p, 1e-10, None)
    p /= p.sum()
    return p


def resample_sequence(p, new_n):
    """Resample a sequence to a new length using linear interpolation."""
    old_n = len(p)
    if old_n == new_n:
        return p.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    p_new = np.interp(new_x, old_x, p)
    p_new = np.clip(p_new, 1e-10, None)
    p_new /= p_new.sum()
    return p_new


def build_constraint_matrix(a):
    """Build the constraint matrix A for LP: A[k,j] = a[k-j] if valid."""
    n = len(a)
    c_len = 2 * n - 1
    A = np.zeros((c_len, n), dtype=np.float64)
    for k in range(c_len):
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        if j_start <= j_end:
            A[k, j_start:j_end + 1] = a[k - j_end:k - j_start + 1][::-1]
    return A


def lp_alternating(p, max_iter=150, t_blend=0.5):
    """LP-based alternating optimization. Returns (best_p, best_objective)."""
    n = len(p)
    p = p.copy()
    p = np.clip(p, 0, None)
    s = p.sum()
    if s < 1e-10:
        p = np.ones(n) / n
    else:
        p /= s

    best_p = p.copy()
    best_obj = compute_obj(best_p)

    for it in range(max_iter):
        c = fftconvolve(p, p)
        M = float(np.max(c))

        A = build_constraint_matrix(p)
        c_obj = -np.ones(n)
        b_ub = np.full(2 * n - 1, M)
        bounds = [(0, None)] * n

        res = linprog(c_obj, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
        if res.status != 0:
            break

        g = res.x
        g_sum = float(g.sum())
        if g_sum <= 1.0 + 1e-6:
            break

        g = g / g_sum
        p_new = (1.0 - t_blend) * p + t_blend * g
        p_new = np.clip(p_new, 0, None)
        s = p_new.sum()
        if s < 1e-10:
            break
        p_new /= s

        obj = compute_obj(p_new)
        if obj < best_obj:
            best_obj = obj
            best_p = p_new.copy()
        p = p_new

    return best_p, best_obj


def mirror_descent(p, n_steps, eta=2.0, eps_start=5.0, eps_end=50.0,
                   perturb_freq=300, perturb_start=0.3, rng=None):
    """
    Mirror descent on the simplex to minimize max autocorrelation.
    Uses log-sum-exp smoothing with increasing temperature.
    Returns (best_p, best_objective = 2*n*max_autocorr).
    """
    if rng is None:
        rng = np.random.default_rng()
    n = len(p)
    p = p.astype(np.float64).copy()
    p = np.clip(p, 0, None)
    s = p.sum()
    if s < 1e-10:
        p = np.ones(n) / n
    else:
        p /= s

    best_p = p.copy()
    best_val = max_autocorr(best_p)

    for t in range(n_steps):
        # Temperature schedule: increase eps over time
        frac = t / max(1, n_steps - 1)
        eps_t = eps_start + (eps_end - eps_start) * frac
        # Step size decay
        eta_t = eta / (1.0 + 0.002 * t)
        # Perturbation strength decay
        pstr = perturb_start / (1.0 + t / 500.0)

        # Compute autocorrelation
        c = fftconvolve(p, p)
        c_max = c.max()

        # Softmax weights for smooth max
        c_shifted = c - c_max
        w = np.exp(eps_t * c_shifted)
        w_sum = w.sum()
        if w_sum < 1e-30:
            break
        w /= w_sum

        # Compute gradient: grad_j = sum_k w_k * 2 * p_{k-j}
        g = np.zeros(n)
        active = np.where(w > 1e-8)[0]
        for k in active:
            wk = 2.0 * w[k]
            j_start = max(0, k - n + 1)
            j_end = min(n - 1, k)
            if j_start <= j_end:
                g[j_start:j_end + 1] += wk * p[k - j_end:k - j_start + 1][::-1]

        # Mirror descent update
        log_step = np.clip(-eta_t * g, -40.0, 40.0)
        p = p * np.exp(log_step)
        s = p.sum()
        if s < 1e-30:
            p = np.ones(n) / n
        else:
            p /= s

        # Evaluate
        val = float(np.max(fftconvolve(p, p)))
        if val < best_val:
            best_val = val
            best_p = p.copy()

        # Periodic perturbation
        if (t + 1) % perturb_freq == 0:
            noise = np.exp(rng.normal(0, pstr, n))
            p = p * noise
            p = np.clip(p, 0, None)
            s = p.sum()
            if s > 1e-10:
                p /= s

    obj = 2.0 * n * best_val
    return best_p, obj


def random_refine(p, rng, n_steps=5000, strength=0.05):
    """Random block perturbation local search."""
    n = len(p)
    best_p = p.copy()
    best_obj = compute_obj(best_p)
    width = max(1, n // 30)

    for it in range(n_steps):
        start = int(rng.integers(0, max(1, n - width + 1)))
        candidate = best_p.copy()
        candidate[start:start + width] *= np.exp(rng.normal(0, strength, width))
        candidate = np.clip(candidate, 0, None)
        s = candidate.sum()
        if s < 1e-10:
            continue
        candidate /= s
        obj = compute_obj(candidate)
        if obj < best_obj:
            best_p = candidate
            best_obj = obj

    return best_p, best_obj


def propose_candidate(seed=42, budget_s=1500, **kwargs):
    """
    Hybrid optimizer: LP for small n + mirror descent for medium/large n.
    Uses FFT, multiplicative updates, diverse initializations, and random restarts.
    """
    rng = np.random.default_rng(seed)
    start_time = time.monotonic()
    deadline = start_time + float(budget_s) - 15.0

    best_seq = None
    best_score = np.inf

    # Try parent sequence
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64)
        parent = np.clip(parent, 0, 1000.0)
        if parent.size > 0 and np.isfinite(parent).all():
            obj_p = compute_obj(parent)
            if obj_p < best_score:
                best_score = obj_p
                best_seq = parent.copy()
            print(f"Parent (n={parent.size}): obj={obj_p:.6f}")
    except Exception:
        parent = None

    # Phase 1: LP optimization for small n
    print("\n=== Phase 1: LP for small n ===")
    n_values_small = [50, 80, 100, 120, 150, 200]
    init_methods = ['uniform', 'cosine', 'random_smooth', 'gaussian']

    for n in n_values_small:
        if time.monotonic() > deadline - 400:
            break
        for restart in range(2):
            if time.monotonic() > deadline - 400:
                break
            method = init_methods[(restart + n) % len(init_methods)]
            p0 = make_initialization(n, method, rng)
            a, obj = lp_alternating(p0, max_iter=80, t_blend=0.5)
            if obj < best_score:
                best_score = obj
                best_seq = a.copy()
                print(f"  LP n={n} r={restart} [{method}]: obj={obj:.6f}")
            # Also try with parent resampled
            if parent is not None:
                p0r = resample_sequence(parent, n)
                ar, obr = lp_alternating(p0r, max_iter=50, t_blend=0.5)
                if obr < best_score:
                    best_score = obr
                    best_seq = ar.copy()
                    print(f"  LP n={n} parent-resample: obj={obr:.6f}")

    # Phase 2: Mirror descent for various n
    print("\n=== Phase 2: Mirror descent ===")
    n_values_md = [100, 150, 200, 300, 500, 800, 1200, 2000, 3000, 5000]
    md_methods = ['uniform', 'cosine', 'cosine2', 'gaussian', 'triangle',
                  'edge_emphasis', 'periodic', 'bimodal', 'random_smooth', 'random']

    for n in n_values_md:
        if time.monotonic() > deadline - 80:
            break
        remaining = deadline - time.monotonic()
        # Allocate time per n based on how many n's remain
        n_remaining = len(n_values_md) - n_values_md.index(n)
        time_per_n = min(120.0, remaining / max(1, n_remaining))
        n_restarts = max(1, int(time_per_n / 30.0))

        for restart in range(n_restarts):
            if time.monotonic() > deadline - 80:
                break
            if time.monotonic() > start_time + float(budget_s) - 60:
                break

            method = md_methods[(restart * 3 + n) % len(md_methods)]
            p0 = make_initialization(n, method, rng)

            # Estimate time per iteration (rough)
            est_time_per_iter = max(0.001, n * 0.00003)
            n_steps = int(min(30000, max(1000, (time_per_n - 5) / est_time_per_iter)))

            p_best, obj = mirror_descent(
                p0, n_steps=n_steps,
                eta=2.0, eps_start=5.0, eps_end=50.0,
                perturb_freq=max(100, n_steps // 10),
                perturb_start=0.3, rng=rng
            )
            if obj < best_score:
                best_score = obj
                best_seq = p_best.copy()
                print(f"  MD n={n} r={restart} [{method}] steps={n_steps}: obj={obj:.6f}")

            # Bootstrap: try resampling parent if available
            if parent is not None and restart == 0:
                p0r = resample_sequence(parent, n)
                n_steps2 = int(min(20000, max(500, (time_per_n - 10) / est_time_per_iter)))
                p_best2, obj2 = mirror_descent(
                    p0r, n_steps=n_steps2,
                    eta=1.5, eps_start=10.0, eps_end=50.0,
                    perturb_freq=max(100, n_steps2 // 10),
                    perturb_start=0.2, rng=rng
                )
                if obj2 < best_score:
                    best_score = obj2
                    best_seq = p_best2.copy()
                    print(f"  MD n={n} bootstrap: obj={obj2:.6f}")

    # Phase 3: Refine the best solution
    print("\n=== Phase 3: Refinement ===")
    if best_seq is not None:
        n_best = len(best_seq)
        if time.monotonic() < deadline - 30:
            remaining = deadline - time.monotonic()

            # Extended mirror descent on the best solution
            est_time_per_iter = max(0.001, n_best * 0.00003)
            n_steps_ref = int(min(50000, max(2000, (remaining * 0.4) / est_time_per_iter)))

            p = best_seq.copy()
            p = np.clip(p, 0, None)
            s = p.sum()
            if s > 0:
                p /= s

            p_ref, obj_ref = mirror_descent(
                p, n_steps=n_steps_ref,
                eta=1.0, eps_start=20.0, eps_end=80.0,
                perturb_freq=max(200, n_steps_ref // 8),
                perturb_start=0.1, rng=rng
            )
            if obj_ref < best_score:
                best_score = obj_ref
                best_seq = p_ref.copy()
                print(f"  Refine MD: obj={obj_ref:.6f}")
            else:
                p_ref = best_seq

            # Random refinement
            if time.monotonic() < deadline - 20:
                remaining2 = deadline - time.monotonic()
                rls_steps = int(max(1000, remaining2 / 0.02))
                p_r, obj_r = random_refine(p_ref, rng, n_steps=rls_steps, strength=0.03)
                if obj_r < best_score:
                    best_score = obj_r
                    best_seq = p_r.copy()
                    print(f"  Random refine: obj={obj_r:.6f}")

    # Phase 4: Final micro-perturbations
    if best_seq is not None and time.monotonic() < deadline - 10:
        remaining = deadline - time.monotonic()
        n_best = len(best_seq)
        p = best_seq.copy()
        p = np.clip(p, 0, None)
        s = p.sum()
        if s > 0:
            p /= s

        n_iters = int(max(100, remaining / 0.01))
        for it in range(n_iters):
            if time.monotonic() > deadline - 5:
                break
            width = max(1, n_best // 40)
            start = int(rng.integers(0, max(1, n_best - width + 1)))
            candidate = p.copy()
            candidate[start:start + width] *= np.exp(rng.normal(0, 0.02, width))
            candidate = np.clip(candidate, 0, None)
            s = candidate.sum()
            if s < 1e-10:
                continue
            candidate /= s
            obj = compute_obj(candidate)
            if obj < best_score:
                best_score = obj
                p = candidate
                best_seq = candidate.copy()

    # Fallback
    if best_seq is None:
        best_seq = np.ones(100)
        best_score = compute_obj(best_seq)

    final_score = compute_obj(best_seq)
    elapsed = time.monotonic() - start_time
    print(f"\nFINAL: n={len(best_seq)}, score={final_score:.6f}, time={elapsed:.1f}s")

    return [float(x) for x in best_seq]
