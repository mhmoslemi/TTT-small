```python
def frank_wolfe_iteration(x, M):
    """Solve the LP: max sum(b) s.t. (x*b)[k] <= M for all k, b >= 0"""
    n = len(x)
    # Build constraint matrix: A[k, l] = x[k-l] if valid
    # A has shape (2n-1, n)
    # For constraint k: sum_l A[k,l] * b[l] <= M
    # A[k,l] = x[k-l] if 0 <= k-l < n and 0 <= l < n
    
    # Use sparse representation
    row_idx = []
    col_idx = []
    vals = []
    for k in range(2*n-1):
        l_min = max(0, k-n+1)
        l_max = min(n-1, k)
        for l in range(l_min, l_max+1):
            row_idx.append(k)
            col_idx.append(l)
            vals.append(x[k-l])
    
    A = coo_matrix((vals, (row_idx, col_idx)), shape=(2*n-1, n))
    
    # Solve: max 1^T b s.t. A b <= M*1, b >= 0
    res = linprog(-np.ones(n), A_ub=A, b_ub=np.full(2*n-1, M), bounds=[(0, None)]*n, method='highs')
    return res.x
```
