"""
Optimized step function search using polynomial square-root initialization,
LP-based direction finding with adaptive targets, momentum subgradient polishing,
and multi-length exploration to minimize 2n*max(conv(a,a))/sum(a)^2.
"""
import numpy as np
import time
from scipy.optimize import linprog
from scipy.sparse import csr_matrix


def poly_sqrt_coeffs(n):
    """Compute coefficients of sqrt(1 + x + ... + x^{2n-2}) as a power series."""
    q = np.zeros(2 * n - 1)
    q[0] = 1.0
    for k in range(1, 2 * n - 1):
        s = 0.0
        for i in range(1, k):
            s += q[i] * q[k - i]
        q[k] = (1.0 - s) / 2.0
    return q


def init_poly_sqrt(n):
    """Create initial sequence from positive part of polynomial square root."""
    q = poly_sqrt_coeffs(n)
    # Take only the first n coefficients (the sequence has length n)
    a = q[:n].copy()
    a = np.clip(a, 0.0, None)
    if np.sum(a) < 1e-10:
        a = np.ones(n)
    a = a / np.sum(a)
    return a


def init_hann(n):
    """Hann window initialization."""
    idx = np.arange(n, dtype=np.float64)
    a = 0.5 * (1 + np.cos(2 * np.pi * idx / (n - 1)))
    a = np.clip(a, 1e-6, None)
    return a / np.sum(a)


def init_blackman(n):
    """Blackman window initialization."""
    idx = np.arange(n, dtype=np.float64)
    N = n - 1
    a = 0.42 - 0.5 * np.cos(2 * np.pi * idx / N) + 0.08 * np.cos(4 * np.pi * idx / N)
    a = np.clip(a, 1e-6, None)
    return a / np.sum(a)


def init_uniform(n):
    return np.ones(n) / n


def init_random(n, rng):
    a = rng.uniform(0.5, 1.5, size=n)
    return a / np.sum(a)


def build_conv_matrix_sparse(f):
    """Build sparse matrix A where (A@g)[k] = convolve(f,g)[k]."""
    n = len(f)
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        j_min = max(0, k - (n - 1))
        j_max = min(k, n - 1)
        for j in range(j_min, j_max + 1):
            rows.append(k)
            cols.append(j)
            vals.append(f[k - j])
    return csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))


def solve_lp_direction(f, M, matrix_cache=None):
    """Solve: max sum(g) s.t. g >= 0, convolve(f,g)_k <= M for all k."""
    n = len(f)
    if matrix_cache is not None and np.array_equal(matrix_cache.get('f'), f):
        A = matrix_cache['A']
    else:
        A = build_conv_matrix_sparse(f)
        if matrix_cache is not None:
            matrix_cache['f'] = f.copy()
            matrix_cache['A'] = A
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    try:
        result = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
        if result.success:
            return result.x
    except Exception:
        pass
    return None


def lp_optimize(f, deadline, step_size_init=0.2, verbose_prefix=""):
    """
    LP-based iterative optimization.
    At each step, solve LP to find g maximizing sum(g) subject to conv(f,g) <= M.
    Blend f toward g with adaptive step size.
    Uses multiple M targets for exploration.
    """
    f = f.copy().astype(np.float64)
    best_f = f.copy()
    best_score = evaluate_sequence(f.tolist())
    step_size = step_size_init
    no_improve = 0
    it = 0
    matrix_cache = {}

    # Track multiple M targets
    m_targets = [1.0, 0.95, 0.9]  # fractions of current max

    while time.monotonic() < deadline:
        conv_f = np.convolve(f, f)
        M = float(np.max(conv_f))
        if M <= 0:
            break

        best_g = None
        best_g_score = best_score

        # Try multiple M targets
        for mf in m_targets:
            M_try = M * mf
            g = solve_lp_direction(f, M_try, matrix_cache)
            if g is None:
                continue
            sum_g = float(np.sum(g))
            if sum_g < 1e-10:
                continue
            g = g * (float(np.sum(f)) / sum_g)

            # Try the blend
            new_f = (1.0 - step_size) * f + step_size * g
            new_f = np.clip(new_f, 0.0, 1000.0)
            new_score = evaluate_sequence(new_f.tolist())

            if np.isfinite(new_score) and new_score < best_g_score:
                best_g_score = new_score
                best_g = new_f

        if best_g is not None:
            f = best_g
            best_f = f.copy()
            best_score = best_g_score
            no_improve = 0
            it += 1
            if it % 50 == 1:
                print(f"    {verbose_prefix}LP iter {it}: score={best_score:.6f} (step={step_size:.3f})")
        else:
            no_improve += 1
            step_size *= 0.7
            if step_size < 0.005 or no_improve > 80:
                break

    return best_f, best_score


def subgradient_polish(f, deadline, lr_init=0.01, momentum=0.9, verbose_prefix=""):
    """
    Subgradient method with momentum for minimizing max(conv(f,f)).
    Fixes sum(f) = 1 by renormalization.
    """
    f = f.copy().astype(np.float64)
    best_f = f.copy()
    best_score = evaluate_sequence(f.tolist())
    n = len(f)
    lr = lr_init
    v = np.zeros(n)  # momentum
    no_improve = 0

    for it in range(20000):
        if time.monotonic() > deadline:
            break

        conv_f = np.convolve(f, f)
        k_star = int(np.argmax(conv_f))
        M = conv_f[k_star]
        S = float(np.sum(f))

        if M <= 0 or S <= 0:
            break

        # Subgradient: d/d f[i] of conv[k_star] = 2*f[k_star - i] if valid
        grad = np.zeros(n)
        for i in range(n):
            j = k_star - i
            if 0 <= j < n:
                grad[i] = 2.0 * f[j]

        # Momentum update
        v = momentum * v - lr * grad
        new_f = f + v
        new_f = np.clip(new_f, 0.0, 1000.0)

        # Renormalize to maintain sum
        new_sum = float(np.sum(new_f))
        if new_sum > 1e-15:
            new_f = new_f * (S / new_sum)

        new_score = evaluate_sequence(new_f.tolist())
        if np.isfinite(new_score) and new_score < best_score:
            f = new_f
            best_f = f.copy()
            best_score = new_score
            no_improve = 0
            if (it + 1) % 500 == 0:
                print(f"    {verbose_prefix}SG iter {it+1}: score={best_score:.6f} (lr={lr:.6f})")
        else:
            no_improve += 1
            lr *= 0.95
            if lr < 1e-8:
                break
            if no_improve > 200:
                # Reset momentum and try again with fresh lr
                v = np.zeros(n)
                lr = lr_init * 0.5
                no_improve = 0

    return best_f, best_score


def random_restart_polish(f, deadline, n_restarts=50, verbose_prefix=""):
    """Try small random perturbations to escape local minima."""
    f = f.copy().astype(np.float64)
    best_f = f.copy()
    best_score = evaluate_sequence(f.tolist())
    n = len(f)
    rng = np.random.default_rng(123)

    for r in range(n_restarts):
        if time.monotonic() > deadline:
            break

        # Random perturbation
        scale = 0.05 * (1 - r / n_restarts)
        pert = rng.normal(0, scale, size=n)
        new_f = f + pert
        new_f = np.clip(new_f, 0.0, 1000.0)
        new_sum = float(np.sum(new_f))
        if new_sum < 1e-10:
            continue
        new_f = new_f * (float(np.sum(f)) / new_sum)

        new_score = evaluate_sequence(new_f.tolist())
        if np.isfinite(new_score) and new_score < best_score:
            best_f = new_f.copy()
            best_score = new_score
            f = new_f
            print(f"    {verbose_prefix}Restart {r}: improved to {best_score:.6f}")

    return best_f, best_score
