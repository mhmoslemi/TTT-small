Actually, I think a cleaner approach would be:

1. Use the smooth max (log-sum-exp) to get a differentiable objective
2. Use gradient descent with momentum (like Adam)
3. Periodically project and renormalize
4. Every so often, do an LP step for a bigger jump

Let me think about the log-sum-exp approach:
Φ_β(f) = (1/β) * log(sum_k exp(β * (f*f)[k]))

As β → ∞, this approaches max_k (f*f)[k]. The gradient is:
∇Φ_β(f) = sum_k p_k * ∇(f*f)[k] where p_k = exp(β*(f*f)[k]) / sum_j exp(β*(f*f)[j])

This is exactly the weighted gradient I described above!

So the algorithm is:
1. Start with f (normalized)
2. Repeat:
   a. Compute conv = f*f
   b. Compute weights p_k = softmax(β * conv)
   c. Compute grad = weighted gradient using p
   d. Take Adam step
   e. Project onto simplex (f >= 0, sum(f) = 1)
   f. Track best

For the projection onto the simplex {x >= 0, sum(x) = 1}, the proper algorithm is:
1. Sort x in descending order: x_(1) >= x_(2) >= ... >= x_(n)
2. Find the largest m such that x_(m) + 1/m * (1 - sum_{i=1}^m x_(i)) > 0
3. ν = (sum_{i=1}^m x_(i) - 1) / m
4. x_i^proj = max(x_i - ν, 0)

This is O(n log n) per iteration.

```python
import time
import numpy as np
from scipy.optimize import linprog
from scipy.sparse import csr_matrix

def simplex_project(v):
    """Euclidean projection onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = v.size
    if n == 1:
        return np.array([max(v[0], 0.0)])
    x = np.sort(v)[::-1]
    cssv = np.cumsum(x)
    ind = np.arange(1, n + 1)
    cond = x - (cssv - 1.0) / ind > 0
    if not np.any(cond):
        # All would be zero, just return uniform
        return np.ones(n) / n
    m = np.max(np.where(cond)[0])
    theta = (cssv[m] - 1.0) / (m + 1)
    return np.maximum(v - theta, 0.0)

def normalize_sequence(arr):
    x = np.asarray(arr, dtype=np.float64).ravel()
    if x.size == 0:
        return None
    x = np.where(np.isfinite(x), x, 0.0)
    x = np.maximum(x, 0.0)
    s = float(x.sum())
    if s <= 1e-300:
        return None
    x /= s
    return x

def score_normalized(x, n):
    conv = np.convolve(x, x)
    m = float(np.max(conv))
    if not np.isfinite(m):
        return float("inf")
    return float(2.0 * n * m)

def compute_conv(f):
    return np.convolve(f, f)

def compute_grad_fft(f, weights):
    """Compute gradient using FFT. weights has length 2n-1."""
    n = len(f)
    # We need N >= (2n-1) + n - 1 = 3n - 2
    N = 1
    while N < 3 * n:
        N *= 2
    W = np.fft.rfft(weights, N)
    F = np.fft.rfft(f, N)
    cross = np.fft.irfft(W * np.conj(F), N)
    return cross[:n]

def smooth_max_and_grad(f, beta):
    """Compute smooth max value and its gradient."""
    n = len(f)
    conv = np.convolve(f, f)
    max_conv = np.max(conv)
    shifted = conv - max_conv
    exp_vals = np.exp(beta * shifted)
    total = exp_vals.sum()
    weights = exp_vals / total
    smooth_val = max_conv + np.log(total) / beta
    grad = compute_grad_fft(f, weights)
    return smooth_val, grad, conv, weights

def adam_step(params, grad, t):
    """Adam optimizer step."""
    beta1, beta2, eps = 0.9, 0.999, 1e-8
    m = params['m']
    v = params['v']
    m[:] = beta1 * m + (1 - beta1) * grad
    v[:] = beta2 * v + (1 - beta2) * grad * grad
    m_hat = m / (1 - beta1 ** t)
    v_hat = v / (1 - beta2 ** t)
    return m_hat / (np.sqrt(v_hat) + eps)

def make_conv_template(n):
    """Precompute the sparse matrix for the LP direction finding."""
    m = 2 * n - 1
    k64 = np.arange(m, dtype=np.int64)
    js = np.maximum(0, k64 - n + 1).astype(np.int32)
    je = np.minimum(n - 1, k64).astype(np.int32)
    counts = (je.astype(np.int64) - js.astype(np.int64) + 1).astype(np.int64)
    total = int(counts.sum())
    cum = np.cumsum(counts)
    starts = np.repeat(cum - counts, counts)
    local = np.arange(total, dtype=np.int64) - starts
    row = np.repeat(k64, counts).astype(np.int32)
    col_base = js[row.astype(np.int64)]
    col = (col_base + local).astype(np.int32)
    offsets = (row.astype(np.int64) - col.astype(np.int64)).astype(np.int32)
    A_off = csr_matrix((offsets, (row, col)), shape=(m, n), dtype=np.int32)
    edge_offsets = A_off.data.astype(np.int64, copy=True)
    A_template = A_off.astype(np.float64).copy()
    A_template.data.fill(0.0)
    c_obj = -np.ones(n, dtype=np.float64)
    bounds = [(0.0, None)] * n
    return A_template, edge_offsets, c_obj, bounds, m

def solve_lp_direction(template, f, n, M):
    """Solve the LP direction problem."""
    A_template, edge_offsets, c_obj, bounds, m = template
    A = A_template.copy()
    A.data[:] = f[edge_offsets]
    b = np.full(m, M, dtype=np.float64)
    return linprog(c_obj, A_ub=A, b_ub=b, bounds=bounds, method="highs")

def optimize_smooth(n, f_init, time_budget, rng, template, beta_init=20.0):
    """Optimize using smooth max gradient with Adam, plus periodic LP steps."""
    f = normalize_sequence(f_init)
    if f is None:
        f = np.ones(n) / n
    f = f.astype(np.float64)
    
    best_f = f.copy()
    best_score = score_normalized(f, n)
    
    # Adam state
    m_state = np.zeros(n)
    v_state = np.zeros(n)
    t = 0
    lr = 1.0 / np.sqrt(n)
    
    # Anneal beta (smoothness parameter)
    beta = beta_init
    beta_end = 200.0
    
    deadline = time.monotonic() + time_budget
    stall = 0
    lp_every = 50
    
    while time.monotonic() < deadline:
        t += 1
        t0 = time.monotonic()
        
        # Compute smooth max and gradient
        try:
            smooth_val, grad, conv, weights = smooth_max_and_grad(f, beta)
        except Exception:
            break
        
        if not np.isfinite(smooth_val):
            break
        
        # Anneal beta
        progress = min(1.0, t / max(1, int(3 * time_budget)))
        beta = beta_init + (beta_end - beta_init) * progress
        if beta > beta_end:
            beta = beta_end
        
        # Adam step
        grad_adam = adam_step({'m': m_state, 'v': v_state}, grad, t)
        f_new = f - lr * grad_adam
        
        # Project onto simplex
        f_new = simplex_project(f_new)
        
        # Check improvement
        sc = score_normalized(f_new, n)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_f = f_new.copy()
            stall = 0
        else:
            stall += 1
        
        f = f_new
        
        # Periodic LP step
        if stall >= lp_every and template is not None:
            stall = 0
            conv_ff = np.convolve(f, f)
            M = float(np.max(conv_ff))
            if M > 0 and np.isfinite(M):
                try:
                    res = solve_lp_direction(template, f, n, M)
                    x = getattr(res, 'x', None)
                    if x is not None:
                        x = np.asarray(x, dtype=np.float64).ravel()
                        x = np.maximum(x, 0.0)
                        s_star = float(x.sum())
                        if s_star > 1.0 + 1e-6:
                            g = x / s_star
                            # Try interpolation
                            for tv in [0.5, 0.3, 0.1, 0.05, 0.01]:
                                f_try = (1 - tv) * f + tv * g
                                f_try = simplex_project(f_try)
                                sc_try = score_normalized(f_try, n)
                                if np.isfinite(sc_try) and sc_try < best_score:
                                    best_score = sc_try
                                    best_f = f_try.copy()
                                    f = f_try.copy()
                                    break
                except Exception:
                    pass
        
        # Reset Adam if stuck
        if stall > 200:
            m_state[:] = 0
            v_state[:] = 0
            stall = 0
            # Small random perturbation
            f = f + 0.01 * rng.standard_normal(n)
            f = simplex_project(f)
        
        if t % 100 == 0:
            print(f"    t={t}, score={best_score:.6f}, beta={beta:.1f}, "
                  f"lr={lr:.4f}, dt={time.monotonic()-t0:.3f}s", flush=True)
        
        # Decay learning rate
        if t % 500 == 0:
            lr *= 0.8
    
    return best_f, float(best_score)
```
