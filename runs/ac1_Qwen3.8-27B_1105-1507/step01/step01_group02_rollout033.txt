```python
from scipy.signal import fftconvolve, correlate

def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    N = 1 << (size - 1).bit_length()
    fa = np.fft.rfft(a, N)
    fc = fa * fa
    result = np.fft.irfft(fc, N)[:size]
    result[result < 0] = 0.0
    return result

def gradient_logsumexp(a, beta):
    """Compute gradient of (1/beta)*log(sum exp(beta*c_k)) where c = convolve(a,a)"""
    n = len(a)
    c = fast_convolve(a)
    # Numerically stable softmax
    bc = beta * c
    bc_max = np.max(bc)
    weights = np.exp(bc - bc_max)
    weights = weights / np.sum(weights)
    # grad[j] = sum_k weights[k] * a[k-j] (valid k-j)
    # This is correlate(weights, a, mode='valid')
    grad = correlate(weights, a, mode='valid')[:n]
    return grad, c, weights

def project_simplex(v):
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cv = np.cumsum(u) - 1.0
    rho_mask = u * np.arange(1, n + 1) > cv
    rho_candidates = np.where(rho_mask)[0]
    if len(rho_candidates) == 0:
        return np.ones(n) / n
    rho = rho_candidates[-1]
    theta = cv[rho] / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

    best_peak = np.max(fast_convolve(a))

    y = a.copy()
    v = np.zeros(n)  # velocity

    stall = 0

        beta = beta_start + (beta_end - beta_start) * progress ** 2
```

```python
"""
Optimizes a non-negative sequence to minimize 2*n*max(convolve(a,a))/sum(a)^2.
Uses log-sum-exp smoothed gradient descent with Nesterov momentum,
adaptive beta scheduling, and multiple initialization strategies.
"""

import numpy as np
import time

def project_simplex(v):
    """Project v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cv = np.cumsum(u) - 1.0
    rho_mask = u * np.arange(1, n + 1) > cv
    idx = np.where(rho_mask)[0]
    if len(idx) == 0:
        return np.ones(n) / n
    rho = idx[-1]
    theta = cv[rho] / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def fast_convolve(a):
    """Compute full linear convolution of a with itself using FFT."""
    n = len(a)
    size = 2 * n - 1
    N = 1 << (size - 1).bit_length()
    fa = np.fft.rfft(a, N)
    fc = fa * fa
    result = np.fft.irfft(fc, N)[:size]
    result[result < 0] = 0.0
    return result

def grad_logsumexp(c, a, beta):
    """
    Compute gradient of (1/beta)*log(sum exp(beta*c_k)) with respect to a.
    c = convolve(a, a), length 2n-1.
    Returns gradient of length n.
    """
    n = len(a)
    # Softmax weights
    bc = beta * c
    bc -= np.max(bc)
    w = np.exp(bc)
    w_sum = np.sum(w)
    if w_sum < 1e-300:
        w = np.ones(len(c)) / len(c)
    else:
        w /= w_sum
    
    # grad[j] = sum_{k=j}^{j+n-1} w[k] * a[k-j] (capped at valid range)
    # = correlate(w, a, mode='valid')[:n]
```
