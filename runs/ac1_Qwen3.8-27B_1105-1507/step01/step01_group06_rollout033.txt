def fast_conv(a):

    return np.fft.irfft(fa * fa, s)[:2 * n - 1]

        kend = min(j + n - 1, 2 * n - 2)
        A[kstart:kend + 1, j] = a[0:kend - kstart + 1]
    return A

def fast_conv(a):

                # Perturb to escape local optimum

                    noise = rng.normal(0, 0.02 * prev_j, n)
                    a_pert = np.clip(a + noise, 0.0, None)
                    if a_pert.sum() > 0.01:
                        a = a_pert / a_pert.sum()
                        prev_j = compute_J(a)
                        stall = 0
                if stall > 50:
                    break
            else:
                stall = 0
                prev_j = new_j
                a = a_new

        tracker.update(a)

        if (init_idx + 1) % 3 == 0 or init_idx == len(inits) - 1:
            elapsed = time.monotonic() - (deadline - 1485) if False else 0
            print(f"  n={n}, init {init_idx+1}/{len(inits)}: "
                  f"J={compute_J(a):.6f}, best={tracker.best_score:.6f}",
                  flush=True)


def run_gradient_phase(n, a_start, tracker, deadline, n_restarts=4):
    """L-BFGS-B gradient refinement."""
    rng = np.random.default_rng(999 + n)

    def obj_func(x):
        ax = np.maximum(x, 0.0)
        s = ax.sum()
        if s < 1e-8:
            return 1e10
        ax = ax / s
        b = fast_conv(ax)
        return float(b.max())

    a0 = np.clip(a_start, 0.0, None)
    if a0.size != n:
        t_old = np.linspace(0, 1, a0.size)
        t_new = np.linspace(0, 1, n)
        a0 = np.interp(t_new, t_old, a0)
    a0 = a0 / a0.sum()

    for r in range(n_restarts):
        if time.monotonic() > deadline:
            break

        if r == 0:
            x0 = a0.copy()
        else:
            scale = 0.005 * (r + 1)
            noise = rng.normal(0, scale, n)
            x0 = np.clip(a0 + noise, 0.0, None)
            if x0.sum() < 1e-8:
                x0 = a0.copy()
            x0 = x0 / x0.sum()

        try:
            res = minimize(obj_func, x0, method='L-BFGS-B',
                           bounds=[(0, None)] * n,
                           options={'maxiter': 200, 'max_fun': 3000,
                                    'ftol': 1e-15, 'disp': False})
            if res.fun < 1e10:
                tracker.update(res.x)
        except Exception:
            pass


def propose_candidate(seed=42, budget_s=1500, **kwargs):
    """Main search: LP with line search + gradient refinement + perturbation restarts."""
    tracker = BestTracker()
    deadline = time.monotonic() + float(budget_s) - 12.0

    # Try parent
    p = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        if p.ndim == 1 and p.size > 0 and np.all(np.isfinite(p)):
            tracker.update(p)
    except (NameError, AttributeError):
        p = None

    rng = np.random.default_rng(seed)

    # === Phase 1: LP with line search (70% of time) ===
    phase1_deadline = time.monotonic() + 0.65 * (deadline - time.monotonic())
    for n in [512, 1024]:
        if time.monotonic() > phase1_deadline:
            break
        print(f"\n=== LP Phase: n={n} ===", flush=True)
        t0 = time.monotonic()

        inits = make_initializations(n, rng, p)
        # Put the best-so-far first (interpolated)
        if tracker.best is not None:
            b = tracker.best
            if b.size != n:
                t_old = np.linspace(0, 1, b.size)
                t_new = np.linspace(0, 1, n)
                interp = np.interp(t_new, t_old, b)
            else:
                interp = b.copy()
            interp = np.clip(interp, 0.001, None)
            inits = [interp / interp.sum()] + inits

        run_lp_phase(n, inits, tracker, phase1_deadline, max_iters_per_init=200)
        print(f"  LP n={n} done: {time.monotonic()-t0:.1f}s, best={tracker.best_score:.6f}",
              flush=True)

    # === Phase 2: Gradient refinement (20% of time) ===
    if tracker.best is not None and time.monotonic() < deadline:
        remaining = deadline - time.monotonic()
        phase2_deadline = time.monotonic() + 0.5 * remaining
        n_best = tracker.best.size
        print(f"\n=== Gradient Phase: n={n_best} ===", flush=True)
        t0 = time.monotonic()

        # Try the best length and one nearby
        lengths_to_try = [n_best]
        if n_best == 512:
            lengths_to_try.append(1024)
        elif n_best == 1024:
            lengths_to_try.append(512)

        for n_try in lengths_to_try:
            if time.monotonic() > phase2_deadline:
                break
            run_gradient_phase(n_try, tracker.best, tracker, phase2_deadline,
                               n_restarts=3)

        print(f"  Gradient done: {time.monotonic()-t0:.1f}s, best={tracker.best_score:.6f}",
              flush=True)

    # === Phase 3: Final LP polish (remaining time) ===
    if tracker.best is not None and time.monotonic() < deadline and tracker.best.size <= 1024:
        n_p = tracker.best.size
        print(f"\n=== LP Polish: n={n_p} ===", flush=True)
        t0 = time.monotonic()
        a = tracker.best.copy()
        a = a / a.sum()
        prev_j = compute_J(a)
        stall = 0
        rng2 = np.random.default_rng(777)

        for it in range(500):
            if time.monotonic() > deadline:
                break

            a_new, new_j = lp_one_iteration(a)

            if not np.isfinite(new_j) or new_j >= prev_j - 1e-12:
                stall += 1
                if stall > 15 and stall % 15 == 0:
                    noise = rng2.normal(0, 0.01, n_p)
                    a_p = np.clip(a + noise, 0.0, None)
                    if a_p.sum() > 0.01:
                        a = a_p / a_p.sum()
                        prev_j = compute_J(a)
                        stall = 0
                if stall > 80:
                    break
            else:
                stall = 0
                prev_j = new_j
                a = a_new
                tracker.update(a)

        tracker.update(a)
        print(f"  Polish done: {time.monotonic()-t0:.1f}s, best={tracker.best_score:.6f}",
              flush=True)

    # Final check with evaluate_sequence on the best
    if tracker.best is not None:
        final_score = float(evaluate_sequence(tracker.best.tolist()))
        print(f"\nFINAL VERIFIED: J={final_score:.8f}, n={tracker.best.size}", flush=True)
        if np.isfinite(final_score):
            return [float(x) for x in tracker.best]

    # Fallback
    fallback = np.ones(512) / 512
    return [float(x) for x in fallback]
