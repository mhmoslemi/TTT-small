import numpy as np

def compute_obj(a):
    n = len(a)
    if n == 0:
        return np.inf
    conv = np.convolve(a, a)
    max_b = float(np.max(conv))
    sum_a = float(np.sum(a))
    if sum_a < 0.01:
        return np.inf
    return 2.0 * n * max_b / (sum_a * sum_a)

def build_A(a):
    n = len(a)
    A = np.zeros((2 * n - 1, n), dtype=np.float64)
    for k in range(2 * n - 1):
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        A[k, j_start:j_end + 1] = a[k - j_end:k - j_start + 1][::-1]
    return A

def solve_lp(a, M):
    n = len(a)
    A = build_A(a)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, 1000.0)] * n
    res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
    if res.status == 0:
        return res.x, float(-res.fun)
    return None, None

def lp_alternate(a, max_iter=2000, tol=1e-8, verbose=False):
    """Run LP alternating minimization until convergence."""
    n = len(a)
    a = a / np.sum(a)  # normalize to sum=1
    best_a = a.copy()
    best_obj = compute_obj(best_a)
    
    for it in range(max_iter):
        conv = np.convolve(a, a)
        M = float(np.max(conv))
        
        g, g_sum = solve_lp(a, M)
        if g is None:
            break
        
        # gap = g_sum - 1 (since sum(a)=1, if g=a then g_sum=1)
        # If g_sum is close to 1, we're at a local optimum
        if g_sum <= 1.0 + tol:
            if verbose:
                print(f"    Converged at iter {it}, gap={g_sum-1:.2e}")
            break
        
        # Blend toward g
        # Use adaptive step: larger when gap is large, smaller when gap is small
        t = min(1.0, 0.3 * (g_sum - 1.0))
        t = max(t, 0.01)
        
        g_norm = g / g_sum  # normalize g to sum=1
        a_new = (1.0 - t) * a + t * g_norm
        a_new = np.maximum(a_new, 0)
        s = np.sum(a_new)
        if s < 1e-10:
            break
        a_new = a_new / s
        
        obj_new = compute_obj(a_new)
        if obj_new < best_obj:
            best_obj = obj_new
            best_a = a_new.copy()
        a = a_new
        
        if verbose and (it % 100 == 0):
            print(f"    iter={it}, obj={best_obj:.6f}, M={M:.6f}, gap={g_sum-1:.4f}")

    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf

    max_b = float(np.max(conv))

def build_A_from_a(a):
    """Build constraint matrix: (A@g)[k] = (a*g)[k]."""
    n = len(a)
    A = np.zeros((2 * n - 1, n), dtype=np.float64)
    for k in range(2 * n - 1):
        for j in range(n):
            idx = k - j
            if 0 <= idx < n:
                A[k, j] = a[idx]
    return A

    Returns (g*, t*)."""
    n = len(a)
    A = build_A_from_a(a)
    # Variables: g_0,...,g_{n-1}, t
    # Minimize t
    c = np.zeros(n + 1)
    c[-1] = 1.0  # minimize t

    A_ub = np.hstack([A, -np.eye(2*n-1)])
    b_ub = np.zeros(2*n-1)

    A_eq = np.zeros((1, n+1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])
    
    bounds = [(0, None)] * n + [(0, None)]  # g_i >= 0, t >= 0
    
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq, 
                  bounds=bounds, method='highs')

        return res.x[:n], float(res.x[n])
    return None, None

"""
Hybrid LP Alternating + Smoothed Gradient Descent optimization.
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 by:
1. LP alternating minimization (min-peak formulation) for fast convergence
2. Smoothed (log-sum-exp) gradient descent for escaping local optima
3. Block perturbation for final polishing
Explores multiple sequence lengths and initializations within time budget.
"""

import numpy as np
from scipy.optimize import linprog
import time


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    conv = np.convolve(a, a)
    max_b = float(np.max(conv))
    sum_a = float(np.sum(a))
    if sum_a < 0.01:
        return np.inf
    return 2.0 * n * max_b / (sum_a * sum_a)


def build_conv_matrix(a):
    """Build matrix A such that (A@g)_k = (a*g)[k] for k=0..2n-2."""
    n = len(a)
    A = np.zeros((2 * n - 1, n), dtype=np.float64)
    for k in range(2 * n - 1):
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        A[k, j_start:j_end + 1] = a[k - j_end:k - j_start + 1][::-1]
    return A


def solve_lp_min_peak(a, A):
    """Solve: min t s.t. (a*g)_k <= t for all k, sum(g)=1, g>=0.
    Returns (g_opt, t_opt) or (None, None)."""
    n = len(a)
    c = np.zeros(n + 1)
    c[-1] = 1.0
    A_ub = np.hstack([A, -np.eye(2 * n - 1)])
    b_ub = np.zeros(2 * n - 1)
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])
    bounds = [(0, None)] * n + [(0, None)]
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                  bounds=bounds, method='highs')
    if res.status == 0:
        return res.x[:n], float(res.x[n])
    return None, None


def lp_alternate(a, max_iter=500, tol=1e-7, verbose=False):
    """LP alternating minimization with min-peak LP. Returns (best_a, best_obj)."""
    n = len(a)
    a = a.copy().astype(np.float64)
    a = np.maximum(a, 0)
    s = np.sum(a)
    if s < 1e-15:
        a = np.ones(n)
        s = n
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(best_a)
    A = build_conv_matrix(a)

    for it in range(max_iter):
        g, t = solve_lp_min_peak(a, A)
        if g is None:
            break

        current_max = float(np.max(np.convolve(a, a)))
        # Convergence check: if t is close to current max, we're at local opt
        if t >= current_max * (1.0 - tol) or t <= 0:
            if verbose:
                print(f"      LP converged iter={it}, t={t:.8f}, max={current_max:.8f}")
            break

        # Adaptive blending step
        ratio = (current_max - t) / current_max if current_max > 1e-15 else 0
        step = min(1.0, 0.3 + 0.7 * ratio)

        a_new = (1.0 - step) * a + step * g
        a_new = np.maximum(a_new, 0)
        s = np.sum(a_new)
        if s < 1e-15:
            break
        a_new = a_new / s

        obj_new = compute_obj(a_new)
        if obj_new < best_obj:
            best_obj = obj_new
            best_a = a_new.copy()
        a = a_new
        A = build_conv_matrix(a)

        if verbose and (it % 20 == 0):
            print(f"      LP iter={it}, obj={best_obj:.6f}, t={t:.6f}, max={current_max:.6f}")

    return best_a, best_obj


def smoothed_grad_descent(a, max_iter=3000, beta_start=2.0, beta_end=50.0,
                          lr_start=0.05, lr_end=0.0005, verbose=False):
    """Smoothed gradient descent using log-sum-exp approximation to max."""
    n = len(a)
    a = a.copy().astype(np.float64)
    a = np.maximum(a, 0)
    s = np.sum(a)
    if s < 1e-15:
        a = np.ones(n)
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(best_a)

    for it in range(max_iter):
        # Anneal beta and learning rate
        frac = it / max(max_iter - 1, 1)
        beta = beta_start * (beta_end / beta_start) ** frac
        lr = lr_start * (lr_end / lr_start) ** frac

        conv = np.convolve(a, a)
        max_conv_val = float(np.max(conv))
        # Softmax weights
        shifted = beta * (conv - max_conv_val)
        exp_shifted = np.exp(shifted)
        weights = exp_shifted / np.sum(exp_shifted)

        # Gradient of smoothed max w.r.t. a
        grad = np.zeros(n)
        for k in range(2 * n - 1):
            w = weights[k] * beta
            if w < 1e-12:
                continue
            j_start = max(0, k - n + 1)
            j_end = min(n - 1, k)
            grad[j_start:j_end + 1] += w * 2.0 * a[k - j_end:k - j_start + 1][::-1]

        # Update with projection onto simplex {a>=0, sum(a)=1}
        a_new = a - lr * grad
        a_new = np.maximum(a_new, 0)
        s_new = np.sum(a_new)
        if s_new < 1e-15:
            a_new = a.copy()
            s_new = 1.0
        a_new = a_new / s_new

        obj_new = compute_obj(a_new)
        if obj_new < best_obj:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new
        else:
            # Stay at current a (don't accept worse moves)
            # But occasionally reset to best to avoid drift
            if it % 200 == 0 and it > 0:
                a = best_a.copy()
                lr *= 0.8

        if verbose and (it % 500 == 0):
            print(f"      GD iter={it}, obj={best_obj:.6f}, beta={beta:.2f}, lr={lr:.6f}")

    return best_a, best_obj


def block_perturb_refine(a, rng, max_iter=20000):
    """Random block perturbation local search."""
    n = len(a)
    best_a = a.copy()
    best_obj = compute_obj(best_a)
    count = 0

    for it in range(max_iter):
        # Variable block size
        r = rng.random()
        if r < 0.3:
            width = max(1, n // 50)
        elif r < 0.7:
            width = max(1, n // 20)
        else:
            width = max(1, n // 8)
        width = min(width, n)
        start = int(rng.integers(0, n - width + 1))

        candidate = best_a.copy()
        noise_scale = rng.uniform(0.05, 0.3)
        candidate[start:start + width] *= np.exp(rng.normal(0, noise_scale, width))
        candidate = np.maximum(candidate, 0)
        s = np.sum(candidate)
        if s < 1e-15:
            continue
        candidate = candidate / s

        obj = compute_obj(candidate)
        if obj < best_obj:
            best_obj = obj
            best_a = candidate
            count += 1

    return best_a, best_obj, count


def structured_init(n, rng, mode=0):
    """Generate structured initializations."""
    x = np.arange(n, dtype=np.float64)
    if mode == 0:
        a = rng.random(n) + 0.05
    elif mode == 1:
        # Gaussian bump
        center = rng.uniform(0.2 * n, 0.8 * n)
        sigma = rng.uniform(n / 6, n / 3)
        a = np.exp(-(x - center) ** 2 / (2 * sigma ** 2))
    elif mode == 2:
        # Bimodal
        c1, c2 = rng.uniform(0, n, 2)
        if abs(c1 - c2) < n * 0.2:
            c2 = (c1 + n * 0.5) % n
        sigma = n / 6
        a = np.exp(-(x - c1) ** 2 / (2 * sigma ** 2)) + np.exp(-(x - c2) ** 2 / (2 * sigma ** 2))
    elif mode == 3:
        # Periodic with noise
        period = int(max(3, n // rng.integers(8, 20)))
        pattern = rng.random(period) + 0.1
        a = np.tile(pattern, n // period + 1)[:n]
    elif mode == 4:
        # Cosine window
        a = np.maximum(0, np.cos(np.pi * x / (n - 1))) ** rng.uniform(0.5, 3.0)
    else:
        # Random walk (smooth)
        a = np.cumsum(rng.normal(0, 1, n)) ** 2 + 0.1

    a = np.maximum(a, 0.01)
    a = a / np.sum(a)
    return a

                print(f"  Final GD: obj={best_score:.6f}")

        # More BP
        if time.monotonic() < deadline - 5:
            remaining3 = deadline - time.monotonic()
            max_iter = int(min(30000, remaining3 / 0.002))
            a3, obj3, cnt = block_perturb_refine(a_cur, rng, max_iter=max_iter)
            if obj3 < best_score:
                best_score = obj3
                best_seq = a3.copy()
                print(f"  Final BP: obj={best_score:.6f} ({cnt} improvements)")

    # Fallback
    if best_seq is None:
        best_seq = np.ones(100)
        best_score = compute_obj(best_seq)

    elapsed = time.monotonic() - start_time
    print(f"\nDONE: n={len(best_seq)}, score={best_score:.6f}, time={elapsed:.1f}s")
    return [float(x) for x in best_seq]
