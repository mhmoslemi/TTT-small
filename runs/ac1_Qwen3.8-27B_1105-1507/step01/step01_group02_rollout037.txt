Optimizes non-negative sequences to minimize 2*n*max(conv(a,a))/sum(a)^2.
Uses log-sum-exp smoothed gradient descent with FFT-accelerated gradient

def fast_convolve(a):
    """Full linear convolution a*a via FFT."""
    n = len(a)
    size = 2 * n - 1
    N = 1 << (size - 1).bit_length()
    fa = np.fft.rfft(a, N)
    result = np.fft.irfft(fa * fa, N)[:size]
    return result


def project_simplex(v, target_sum=1.0):
    """Project v onto {x >= 0, sum(x) = target_sum}."""
    n = len(v)
    v = v.copy()
    # Scale to sum=1, then scale back
    u = np.sort(v)[::-1]
    cv = np.cumsum(u) - 1.0
    rho = np.where(u > cv / np.arange(1, n + 1))[0]
    if len(rho) == 0:
        return np.full(n, target_sum / n)
    rho = rho[-1]
    theta = cv[rho] / (rho + 1)
    proj = np.maximum(v - theta, 0.0)
    return proj * target_sum


def compute_gradient(a, c, t):
    """
    Compute gradient of F_t(a) = 2*n*t*log(sum exp(c/t)) / (sum a)^2
    where c = convolve(a, a).
    Returns (gradient, F_t).
    """
    n = len(a)
    sum_a = a.sum()
    
    # Softmax weights
    c_shift = c - c.max()
    exp_c = np.exp(c_shift / t)
    S = exp_c.sum()
    w = exp_c / S  # w[k] = exp(c[k]/t) / sum exp(c[j]/t)
    
    # L = t * log(S) + c.max()
    L = t * np.log(S) + c.max()
    F_t = 2.0 * n * L / (sum_a ** 2)
    
    # g[j] = 2 * sum_m a[m] * w[j+m]
    # Using FFT: g[j] = 2 * conv(a, w_rev)[2n-2-j]
    size = 2 * n - 1
    L_fft = 1 << (3 * n).bit_length()
    A = np.fft.rfft(a, L_fft)
    W_rev = np.fft.rfft(w[::-1], L_fft)
    conv_result = np.fft.irfft(A * W_rev, L_fft)
    # g[j] = 2 * conv_result[2n-2-j] for j = 0,...,n-1
    g = 2.0 * conv_result[2*n-2 : n-2 : -1]
    # Verify: g[0] = 2*conv[2n-2], g[n-1] = 2*conv[n-1]

    # So g[j] = 2*conv_result[2n-2-j]. ✓
    
    # Gradient
    grad = (2.0 * n / (sum_a ** 2)) * g - 2.0 * F_t / sum_a
    return grad, F_t


def compute_subgradient(a, c, tol=1e-6):
    """
    Compute subgradient of max(c) at point a.
    c = convolve(a, a).
    Returns the average of gradients at active (near-peak) lags.
    """
    n = len(a)
    sum_a = a.sum()
    peak = c.max()
    
    # Active set: lags where c[k] >= peak - tol * peak
    active = np.where(c >= peak * (1.0 - tol))[0]
    if len(active) == 0:
        active = np.array([np.argmax(c)])
    
    # Gradient of c[k] w.r.t. a is 2*a[k-j] (shifted)
    # Average over active lags
    grad = np.zeros(n)
    for k in active:
        # grad_k[j] = 2*a[k-j] for max(0,k-n+1) <= j <= min(n-1,k)
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        if lo <= hi:
            # a[k-hi : k-lo+1] reversed
            grad[lo:hi+1] += 2.0 * a[k-hi:k-lo+1][::-1]
    grad /= len(active)
    
    # Normalize: the subgradient of max_k c[k] / (sum a)^2
    # d/d a[j] [max_c / (sum_a)^2] = (subgrad_max_c[j] * (sum_a)^2 - max_c * 2 * sum_a) / (sum_a)^4
    # = subgrad_max_c[j] / (sum_a)^2 - 2 * max_c / (sum_a)^3
    obj = 2.0 * n * peak / (sum_a ** 2)
    grad_norm = grad / (sum_a ** 2) - 2.0 * peak / (sum_a ** 3)
    # Scale by 2n
    grad_final = 2.0 * n * grad_norm
    return grad_final, obj


def lp_improve(a, P, time_limit=5.0):
    """
    LP step: maximize sum(b) s.t. conv(a,b) <= P, b >= 0.
    Returns rescaled b-to-sum(a) solution or None.
    """
    n = len(a)
    if n > 1500:
        return None
    
    # Build matrix: conv(a,b)[k] = sum_j a[k-j]*b[j] for valid indices
    # A[k, j] = a[k-j] if 0 <= k-j < n, else 0
    A = np.zeros((2 * n - 1, n), dtype=np.float64)
    for k in range(2 * n - 1):
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        A[k, lo:hi+1] = a[k-hi:k-lo+1][::-1]
    
    c_obj = -np.ones(n)
    b_ub = np.full(2 * n - 1, P)
    bounds = [(0.0, None)] * n
    
    try:
        res = linprog(c_obj, A_ub=A, b_ub=b_ub, bounds=bounds,
                      method='highs', options={'time_limit': time_limit})
        if res.success:
            b = res.x
            b_sum = b.sum()
            if b_sum > 1e-10:
                a_sum = a.sum()
                b = b * (a_sum / b_sum)
                return b
    except Exception:
        pass
    return None

Optimizes non-negative sequences to minimize 2*n*max(conv(a,a))/sum(a)^2.
Uses log-sum-exp smoothed gradient descent with FFT-accelerated gradients,
temperature annealing, symmetry-breaking perturbations, and LP refinement.
Explores multiple sequence lengths and initializations.

from scipy.optimize import linprog


def fast_convolve(a):
    """Full linear convolution a*a via FFT. Returns array of length 2*len(a)-1."""
    n = len(a)
    size = 2 * n - 1
    N = 1 << (size - 1).bit_length()
    fa = np.fft.rfft(a, N)
    result = np.fft.irfft(fa * fa, N)[:size]
    return result


def project_simplex(v, target_sum=1.0):
    """Project v onto {x >= 0, sum(x) = target_sum}."""
    n = len(v)
    if n == 0:
        return v
    # First project onto sum=1, then scale
    v = v / max(v.max(), 1e-15)  # avoid overflow
    u = np.sort(v)[::-1]
    cumsum_u = np.cumsum(u)
    rho_candidates = np.where(u * np.arange(1, n + 1) > cumsum_u - 1.0)[0]
    if len(rho_candidates) == 0:
        return np.full(n, target_sum / n)
    rho = rho_candidates[-1]
    theta = (cumsum_u[rho] - 1.0) / (rho + 1)
    proj = np.maximum(v - theta, 0.0)
    actual_sum = proj.sum()
    if actual_sum < 1e-15:
        return np.full(n, target_sum / n)
    return proj * (target_sum / actual_sum)


def compute_smoothed_gradient(a, c, t):
    """
    Compute gradient of F_t(a) = 2*n*t*log(sum exp(c/t))/(sum a)^2.
    c = convolve(a, a) with length 2n-1.
    Returns (grad, F_t_value).
    """
    n = len(a)
    sum_a = a.sum()
    
    # Softmax weights (numerically stable)
    c_max = c.max()
    exp_c = np.exp((c - c_max) / t)
    S = exp_c.sum()
    w = exp_c / S  # w[k] proportional to exp(c[k]/t)
    
    # L = t * log(S) + c_max = log(sum exp(c/t))
    L = t * np.log(S) + c_max
    F_t = 2.0 * n * L / (sum_a ** 2)
    
    # g[j] = 2 * sum_{m=0}^{n-1} a[m] * w[j+m]
    # Use FFT: g[j] = 2 * conv(a, w_rev)[2n-2-j]
    # where w_rev = w[::-1]
    # conv length: n + (2n-1) - 1 = 3n-2
    L_fft = 1 << (3 * n - 1).bit_length()
    A = np.fft.rfft(a, L_fft)
    W_rev = np.fft.rfft(w[::-1], L_fft)
    conv_result = np.fft.irfft(A * W_rev, L_fft)
    
    # g[j] = 2 * conv_result[2n-2-j] for j = 0, ..., n-1
    # Index: 2n-2, 2n-3, ..., n-1
    g = 2.0 * conv_result[n-1:2*n-1][::-1]
    
    # Final gradient
    grad = (2.0 * n / (sum_a ** 2)) * g - 2.0 * F_t / sum_a
    return grad, F_t


def compute_exact_subgradient(a, c, rel_tol=1e-4):
    """
    Compute the (averaged) subgradient of 2*n*max(c)/(sum a)^2 at a.
    c = convolve(a, a).
    Returns (grad, objective_value).
    """
    n = len(a)
    sum_a = a.sum()
    peak = c.max()
    obj = 2.0 * n * peak / (sum_a ** 2)
    
    # Active set
    threshold = peak * (1.0 - rel_tol)
    active = np.where(c >= threshold)[0]
    if len(active) > 20:
        # Keep top 10
        top = np.argsort(c[active])[::-1][:10]
        active = active[top]
    
    # Subgradient of max(c) is average of gradients at active peaks
    # grad_c[k][j] = 2*a[k-j] for max(0,k-n+1) <= j <= min(n-1,k)
    subgrad = np.zeros(n)
    for k in active:
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        subgrad[lo:hi+1] += 2.0 * a[k-hi:k-lo+1][::-1]
    subgrad /= len(active)
    
    # Chain rule: d/d a[j] [2n * max_c / (sum_a)^2]
    # = 2n * [subgrad_c[j] / (sum_a)^2 - 2 * max_c / (sum_a)^3]
    grad = 2.0 * n * (subgrad / (sum_a ** 2) - 2.0 * peak / (sum_a ** 3))
    return grad, obj


def lp_improve(a, P, time_limit=5.0):
    """
    LP: maximize sum(b) s.t. conv(a,b) <= P (elementwise), b >= 0.
    Returns b rescaled to match sum(a), or None.
    """
    n = len(a)
    if n > 1200:
        return None
    
    A = np.zeros((2 * n - 1, n), dtype=np.float64)
    for k in range(2 * n - 1):
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        A[k, lo:hi+1] = a[k-hi:k-lo+1][::-1]
    
    c_obj = -np.ones(n)
    b_ub = np.full(2 * n - 1, P)
    bounds = [(0.0, None)] * n
    
    try:
        res = linprog(c_obj, A_ub=A, b_ub=b_ub, bounds=bounds,
                      method='highs', options={'time_limit': time_limit})
        if res.success:
            b = res.x
            b_sum = b.sum()
            if b_sum > 1e-10:
                b = b * (a.sum() / b_sum)
                return b
    except Exception:
        pass
    return None
