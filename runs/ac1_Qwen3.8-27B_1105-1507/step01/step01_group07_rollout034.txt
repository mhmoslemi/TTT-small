def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    conv = np.convolve(a, a)
    max_b = float(np.max(conv))
    return 2.0 * n * max_b / (s * s)


def compute_conv_and_lag(a):
    """Compute convolution, max value, and argmax lag."""
    conv = np.convolve(a, a)
    max_b = float(np.max(conv))
    argmax = int(np.argmax(conv))
    return conv, max_b, argmax


def build_A(a):
    """Build constraint matrix for LP: A[k,j] = max(0, a[k-j]) if valid."""
    n = len(a)
    m = 2 * n - 1
    A = np.zeros((m, n), dtype=np.float64)
    for k in range(m):
        # conv[k] = sum_j a[j] * a[k-j] where 0<=j<n and 0<=k-j<n
        # For fixed k, valid j: max(0, k-n+1) <= j <= min(n-1, k)
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        if j_start <= j_end:
            # A[k, j] = a[k-j] for j in [j_start, j_end]
            for j in range(j_start, j_end + 1):
                A[k, j] = a[k - j]
    return A

def build_A_fast(n):
    """Build the structural pattern (which indices are nonzero) for convolution."""
    m = 2 * n - 1
    # For each row k, the nonzero columns and their corresponding source indices
    rows = []
    for k in range(m):
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        cols = list(range(j_start, j_end + 1))
        src = [k - j for j in cols]  # A[k,j] = a[k-j]
        rows.append((cols, src))
    return rows

    for k, (cols, src) in enumerate(pattern):
        if cols:
            A[k, cols] = np.array([a[s] for s in src])
    return A

with adaptive blending, subgradient steps, and multiple restarts.

    """Build convolution matrix: (A @ x)[k] = sum_j a[k-j]*x[j]."""

        hi = min(n - 1, k)

    a_sym = 0.5 * (a + a[::-1])
    a_sym = np.clip(a_sym, 0, None)
    s = np.sum(a_sym)
    if s < 1e-10:
        return None
    a_sym /= s
    obj_sym = compute_obj(a_sym)
    obj_cur = compute_obj(a)
    if obj_sym < obj_cur:
        return a_sym
    return None


def optimize_seq(a, end_time, verbose=False):
    """Main optimization loop combining LP, subgradient, and perturbations."""
    n = len(a)
    rng = np.random.default_rng(7919 + n)
    a = np.clip(a.astype(np.float64), 0, None)
    sc = np.sum(a)
    a = a / sc if sc > 1e-10 else np.ones(n) / n

    best_a = a.copy()
    best_obj = compute_obj(a)
    if verbose:
        print(f"    n={n} init obj={best_obj:.6f}")

    stall = 0
    it = 0
    best_alpha = 0.3

    while time.monotonic() < end_time:
        it += 1
        improved = False

        # --- LP peak minimization ---
        a_lp = solve_lp_peak(a)
        if a_lp is not None:
            a_lp = np.clip(a_lp, 0, None)
            sl = np.sum(a_lp)
            if sl > 1e-10:
                a_lp /= sl
                # Try several blend factors, prefer larger steps when improving
                alphas = [1.0, 0.7, best_alpha, 0.3, 0.2, 0.1]
                for alpha in alphas:
                    if time.monotonic() > end_time:
                        break
                    cand = a_lp if alpha >= 0.99 else (1 - alpha) * a + alpha * a_lp
                    cand = np.clip(cand, 0, None)
                    sc2 = np.sum(cand)
                    if sc2 < 1e-10:
                        continue
                    cand /= sc2
                    oc = compute_obj(cand)
                    if oc < best_obj - 1e-12:
                        best_obj = oc
                        best_a = cand.copy()
                        a = cand
                        best_alpha = alpha
                        improved = True
                        break

        # --- Subgradient step (if LP didn't help) ---
        if not improved:
            sg, k_pk = subgradient_direction(a)
            mx_sg = np.max(sg)
            if mx_sg > 1e-12:
                for lr_factor in [0.5, 0.2, 0.1, 0.05]:
                    a_sg = a - (lr_factor / mx_sg) * sg
                    a_sg = np.clip(a_sg, 0, None)
                    ss = np.sum(a_sg)
                    if ss < 1e-10:
                        continue
                    a_sg /= ss
                    os = compute_obj(a_sg)
                    if os < best_obj - 1e-12:
                        best_obj = os
                        best_a = a_sg.copy()
                        a = a_sg
                        improved = True
                        break

        # --- Block swap perturbation (occasionally) ---
        if not improved and it % 8 == 0:
            bsz = max(1, n // 20)
            i1 = int(rng.integers(0, n - bsz))
            i2 = int(rng.integers(0, n - bsz))
            if i1 != i2 and abs(i1 - i2) >= bsz:
                a_sw = a.copy()
                a_sw[i1:i1 + bsz], a_sw[i2:i2 + bsz] = a_sw[i2:i2 + bsz].copy(), a_sw[i1:i1 + bsz].copy()
                a_sw = np.clip(a_sw, 0, None)
                sw = np.sum(a_sw)
                if sw > 1e-10:
                    a_sw /= sw
                    ow = compute_obj(a_sw)
                    if ow < best_obj - 1e-12:
                        best_obj = ow
                        best_a = a_sw.copy()
                        a = a_sw
                        improved = True

        # --- Symmetrization (occasionally) ---
        if not improved and it % 12 == 0:
            a_sym = try_symmetrize(a)
            if a_sym is not None:
                best_obj = compute_obj(a_sym)
                best_a = a_sym.copy()
                a = a_sym
                improved = True

        if improved:
            stall = 0
        else:
            stall += 1

        # --- Perturbation to escape local optima ---
        if stall >= 12:
            strength = min(1.0, 0.3 + 0.05 * stall)
            noise = np.exp(rng.normal(0, strength, n))
            a = best_a * noise
            a = np.clip(a, 0, None)
            sa = np.sum(a)
            a = a / sa if sa > 1e-10 else best_a.copy()
            stall = 0

        if verbose and it % 40 == 0:
            print(f"      it={it} obj={best_obj:.6f} stall={stall} alpha={best_alpha:.2f}")

    return best_a, best_obj


def make_init(n, rng, style):
    """Create a diverse initial sequence of length n."""
    if style == 0:
        a = np.ones(n)
    elif style == 1:
        freq = rng.integers(1, max(2, n // 20))
        phase = rng.uniform(0, 2 * np.pi)
        a = 1.0 + 0.4 * np.cos(2 * np.pi * freq * np.arange(n) / n + phase)
    elif style == 2:
        a = np.ones(n)
        mid_start = n // 3
        mid_end = 2 * n // 3
        a[mid_start:mid_end] = rng.uniform(0.15, 0.6)
    elif style == 3:
        p = int(rng.integers(2, max(3, n // 12)))
        pat = np.maximum(0.05, rng.random(p))
        pat /= np.sum(pat)
        a = np.tile(pat, n // p + 1)[:n]
    elif style == 4:
        c = (n - 1) / 2.0
        width = rng.uniform(0.05, 0.3)
        a = np.exp(-((np.arange(n) - c) / (n * width)) ** 2)
    elif style == 5:
        a = rng.random(n)
    elif style == 6:
        # Quadratic residue inspired
        gr = (1.0 + np.sqrt(5.0)) / 2.0
        a = 1.0 / (1.0 + 0.5 * np.sin(gr * np.arange(n) ** 2 + rng.uniform(0, 6)))
        a = np.clip(a, 0.05, None)
    elif style == 7:
        # Two peaks
        pos1 = int(rng.integers(0, n // 2))
        pos2 = n - 1 - pos1
        a = np.ones(n) * 0.3
        w = max(1, n // 10)
        for d in range(w):
            for p in [pos1, pos2]:
                if 0 <= p - d < n:
                    a[p - d] += 1.0
                if 0 <= p + d < n:
                    a[p + d] += 1.0
    else:
        a = np.ones(n)

    a = np.clip(a, 0.01, None)
    a /= np.sum(a)
    return a


def propose_candidate(seed=42, budget_s=1500, **kwargs):
    """
    Search for non-negative sequence minimizing 2*n*max(conv(a,a))/sum(a)^2.
    Target: below 1.5027. Uses LP peak minimization, subgradient descent,
    block swaps, symmetrization, and multi-length exploration.
    """
    start_time = time.monotonic()
    deadline = start_time + float(budget_s) - 8.0
    best_seq = None
    best_score = np.inf

    # Try parent sequence
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64)
        parent = np.clip(parent, 0, 1000.0)
        if parent.size > 0:
            o = compute_obj(parent)
            print(f"Parent: n={parent.size} obj={o:.6f}")
            if o < best_score:
                best_score = o
                best_seq = parent.copy()
    except Exception as e:
        print(f"Parent load error: {e}")

    rng = np.random.default_rng(seed)

    # Optimize the parent first (it's our best starting point)
    if best_seq is not None:
        t_end = min(deadline, start_time + 250.0)
        ao, oo = optimize_seq(best_seq, t_end, verbose=True)
        if oo < best_score:
            best_score = oo
            best_seq = ao.copy()
            print(f"  Parent refined: {oo:.6f}")

    # Explore different configurations
    configs = [
        (100, [0, 1, 2, 5, 6]),
        (120, [0, 1, 3, 5]),
        (140, [0, 1, 2, 4, 7]),
        (150, [0, 1, 2, 3, 4, 5, 6, 7]),
        (160, [0, 1, 4, 5]),
        (180, [0, 1, 2, 3, 5, 7]),
        (200, [0, 1, 3, 4, 6]),
        (220, [0, 1, 2, 5]),
        (250, [0, 1, 3, 4]),
        (300, [0, 1, 2, 5]),
        (350, [0, 1, 4]),
        (400, [0, 1]),
    ]

    for n, styles in configs:
        if time.monotonic() > deadline:
            break
        for style in styles:
            if time.monotonic() > deadline:
                break
            remaining = deadline - time.monotonic()
            if remaining < 25:
                break

            # Time per trial: scale with remaining time, cap at 80s
            n_trials = 1 if remaining > 200 else 1
            t_per = min(80.0, remaining * 0.12)
            t_end = time.monotonic() + t_per

            for trial in range(n_trials):
                if time.monotonic() > deadline:
                    break
                a0 = make_init(n, rng, style)
                ao, oo = optimize_seq(a0, min(t_end, deadline))
                if oo < best_score:
                    best_score = oo
                    best_seq = ao.copy()
                    print(f"  n={n} style={style} t={trial}: obj={oo:.6f}")

        print(f"  [n={n}] best={best_score:.6f} time_left={deadline - time.monotonic():.0f}s")

    # Final refinement on best found
    if best_seq is not None and time.monotonic() < deadline - 5:
        ao, oo = optimize_seq(best_seq, deadline, verbose=True)
        if oo < best_score:
            best_score = oo
            best_seq = ao.copy()
            print(f"  Final refinement: {oo:.6f}")

    # Fallback
    if best_seq is None:
        best_seq = np.ones(150)
        best_score = compute_obj(best_seq)

    best_seq = np.clip(best_seq, 0.0, None)
    elapsed = time.monotonic() - start_time
    print(f"DONE: n={len(best_seq)} score={best_score:.6f} elapsed={elapsed:.1f}s")
    return [float(x) for x in best_seq]
