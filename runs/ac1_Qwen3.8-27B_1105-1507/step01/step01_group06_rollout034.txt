    n = len(a)

def project_simplex(v):
    """Project v onto {x >= 0, sum(x) = 1}."""
    # Use the standard algorithm
    n = len(v)
    if v.sum() <= 0:
        return np.ones(n) / n
    # Sort in decreasing order
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.findlast((u - (cssv - 1) / np.arange(1, n + 1)) > 0)
    theta = (cssv[rho] - 1) / (rho + 1)
    return np.maximum(v - theta, 0)

    best_score = float('inf')

1. L-BFGS-B on soft-max smoothed objective with adaptive annealing
2. LP-based best-response mirror descent for structured updates
3. Multiple diverse initializations across sequence lengths 256-4096
4. Final coordinate descent polish
"""

import numpy as np
from scipy.signal import correlate as sig_correlate
from scipy.optimize import minimize, linprog
import time

def fast_conv(a):
    """Full linear self-convolution via FFT, O(n log n)."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]

def compute_score(a):
    """Compute J(a) = 2n * max(b) / S^2."""
    n = len(a)
    S = a.sum()
    if S < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2.0 * n * b.max() / (S * S))

def smooth_obj_and_grad(x, tau):
    """Smoothed objective and gradient for L-BFGS-B."""
    a = np.maximum(x, 0.0)
    n = len(a)
    S = a.sum()
    if S < 1e-10:
        return 1e10, np.zeros(n)
    b = fast_conv(a)
    # Soft-max
    tb = tau * b
    tb_max = tb.max()
    exp_tb = np.exp(tb - tb_max)
    sum_exp = exp_tb.sum()
    M_s = (tb_max + np.log(sum_exp)) / tau
    w = exp_tb / sum_exp
    # c[j] = sum_m w[j+m] * a[m]
    c = sig_correlate(w, a, mode='full')[:n]
    obj = 2.0 * n * M_s / (S * S)
    grad = (2.0 * n / (S ** 3)) * (S * c - 2.0 * M_s)
    return obj, grad

def build_A(a):
    """Build constraint matrix for LP: A[k,j] gives (a*g)(k) = sum_j A[k,j]*g[j]."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for k in range(2 * n - 1):
        jl = max(0, k - n + 1)
        jh = min(n - 1, k)
        if jl <= jh:
            A[k, jl:jh + 1] = a[k - jh:k - jl + 1][::-1]
    return A

def lp_best_response(a, M):
    """LP: max sum(g) s.t. (a*g)(k) <= M, g >= 0."""
    n = len(a)
    c = -np.ones(n)
    A = build_A(a)
    r = linprog(c, A_ub=A, b_ub=np.full(2 * n - 1, M),
               bounds=[(0, None)] * n, method='highs')
    if r.success:
        g = r.x
        if g.sum() > 1e-10:
            return g
    return None

def generate_initializations(n, rng):
    """Generate diverse initial sequences of length n."""
    inits = []
    t = np.linspace(0, 1, n, endpoint=False)
    
    # Uniform
    inits.append(np.ones(n) / n)
    
    # Cosine modulations at various frequencies
    for freq in [1, 2, 3, 4, 6]:
        a = 1.0 + 0.4 * np.cos(2 * np.pi * freq * t)
        a = np.maximum(a, 0.01)
        inits.append(a / a.sum())
    
    # Quadratic envelope
    a = 1.0 + 0.5 * (2 * t - 1) ** 2
    inits.append(a / a.sum())
    
    # Gaussian-like
    a = np.exp(-2 * (2 * t - 1) ** 2)
    inits.append(a / a.sum())
    
    # Random smooth
    for _ in range(3):
        a = rng.uniform(0.5, 1.5, n)
        # Smooth with a short kernel
        kernel = np.ones(5) / 5
        a = np.convolve(a, kernel, mode='same')
        a = np.maximum(a, 0.01)
        inits.append(a / a.sum())
    
    # Chirp
    a = 1.0 + 0.3 * np.cos(2 * np.pi * (2 * t ** 2))
    a = np.maximum(a, 0.01)
    inits.append(a / a.sum())
    
    return inits

def lbbfbs_refine(a0, tau, deadline, max_iter=500):
    """Refine a0 using L-BFGS-B on smoothed objective."""
    n = len(a0)
    
    def fun(x):
        val, _ = smooth_obj_and_grad(x, tau)
        return val
    
    def grad(x):
        _, g = smooth_obj_and_grad(x, tau)
        return g
    
    res = minimize(fun, a0, jac=grad, method='L-BFGS-B',
                   bounds=[(0, None)] * n,
                   options={'maxiter': max_iter, 'max_fun': 5000,
                            'ftol': 1e-16, 'gtol': 1e-10, 'disp': False})
    return res.x

def lp_iterate(a, n_iters, deadline):
    """LP-based best-response iterations."""
    n = len(a)
    a = a.copy()
    a = np.maximum(a, 0.01)
    a /= a.sum()
    
    for it in range(n_iters):
        if time.monotonic() > deadline:
            break
        b = fast_conv(a)
        M = float(b.max())
        g = lp_best_response(a, M)
        if g is None:
            break
        g /= g.sum()
        t_mix = max(0.05, 0.5 * (1.0 - it / n_iters))
        a_new = (1.0 - t_mix) * a + t_mix * g
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s < 1e-10:
            break
        a_new /= s
        a = a_new
    return a

"""
Minimizes J(a) = 2*n*max(convolve(a,a))/(sum(a))^2 using:
1. L-BFGS-B with analytically computed gradients on soft-max smoothed objective
2. LP-based best-response mirror descent for structured updates
3. Annealed tau (soft-max temperature) for progressive sharpening
4. Multiple diverse initializations across sequence lengths 256-4096
5. Perturbation-based restarts to escape local minima
"""

import numpy as np
from scipy.signal import correlate as sig_correlate
from scipy.optimize import minimize, linprog
import time


def fast_conv(a):
    """Full linear self-convolution via FFT, O(n log n)."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]


def compute_score(a):
    """Compute J(a) = 2n * max(b) / S^2."""
    n = len(a)
    S = a.sum()
    if S < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2.0 * n * b.max() / (S * S))


def smooth_obj_and_grad(x, tau):
    """
    Smoothed objective f(a) = 2n * M_s / S^2 and its gradient.
    M_s = (1/tau) * log(sum_k exp(tau * b(k)))  (soft-max of b)
    Gradient uses: dB(k)/da[j] = 2*a[k-j], so dMs/da[j] = 2*c[j]
    where c[j] = sum_k w[k]*a[k-j].
    """
    a = np.maximum(x, 0.0)
    n = len(a)
    S = a.sum()
    if S < 1e-12:
        return 1e10, np.zeros(n)
    b = fast_conv(a)
    # Soft-max
    tb = tau * b
    tb_max = tb.max()
    exp_tb = np.exp(tb - tb_max)
    sum_exp = exp_tb.sum()
    M_s = (tb_max + np.log(sum_exp)) / tau
    w = exp_tb / sum_exp
    # c[j] = sum_k w[k] * a[k-j] = correlate(w, a, 'full')[j]
    c = sig_correlate(w, a, mode='full')[:n]
    # dMs/da[j] = 2*c[j]
    # f = 2n * M_s / S^2
    # df/da[j] = (2n/S^3) * (S * dMs/da[j] - 2*M_s) = (4n/S^3)*(S*c[j] - M_s)
    obj = 2.0 * n * M_s / (S * S)
    grad = (4.0 * n / (S ** 3)) * (S * c - M_s)
    return obj, grad


def build_A_matrix(a):
    """Build constraint matrix for LP: (a*g)(k) = sum_j A[k,j]*g[j]."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for k in range(2 * n - 1):
        jl = max(0, k - n + 1)
        jh = min(n - 1, k)
        if jl <= jh:
            A[k, jl:jh + 1] = a[k - jh:k - jl + 1][::-1]
    return A


def lp_best_response(a, M):
    """LP: max sum(g) s.t. (a*g)(k) <= M for all k, g >= 0."""
    n = len(a)
    c_vec = -np.ones(n)
    A = build_A_matrix(a)
    r = linprog(c_vec, A_ub=A, b_ub=np.full(2 * n - 1, M),
               bounds=[(0, None)] * n, method='highs')
    if r.success:
        g = r.x
        if g.sum() > 1e-10:
            return g
    return None

    # Cosine modulations
    for freq in [1, 2, 3, 5, 8]:
        a = 1.0 + 0.4 * np.cos(2 * np.pi * freq * t)
        a = np.maximum(a, 0.01)
        inits.append(a / a.sum())

    # Quadratic
    a = 1.0 + 0.5 * (2 * t - 1) ** 2
    inits.append(a / a.sum())

    # Gaussian
    a = np.exp(-3.0 * (2 * t - 1) ** 2)
    inits.append(a / a.sum())

    # Chirp
    a = 1.0 + 0.3 * np.cos(4 * np.pi * t ** 2)
    a = np.maximum(a, 0.01)
    inits.append(a / a.sum())

    # Random smooth
    for _ in range(2):
        a = rng.uniform(0.5, 1.5, n)
        kernel = np.ones(3) / 3.0
        a = np.convolve(a, kernel, mode='same')
        a = np.maximum(a, 0.01)
        inits.append(a / a.sum())

    return inits


def lbbfbs_refine(a0, tau, deadline, max_iter=400, max_fun=4000):
    """Refine a0 using L-BFGS-B on the smoothed objective."""
    def fun(x):
        val, _ = smooth_obj_and_grad(x, tau)
        return val

    def grad(x):
        _, g = smooth_obj_and_grad(x, tau)
        return g

    try:
        res = minimize(fun, a0, jac=grad, method='L-BFGS-B',
                       bounds=[(0, None)] * len(a0),
                       options={'maxiter': max_iter, 'max_fun': max_fun,
                                'ftol': 1e-18, 'gtol': 1e-12, 'disp': False})
        return res.x
    except Exception:
        return a0


def lp_iterate(a, n_iters, deadline):
    """LP-based best-response iterations (mirror descent)."""
    n = len(a)
    a = np.maximum(a.copy(), 0.01)
    a /= a.sum()
    for it in range(n_iters):
        if time.monotonic() > deadline:
            break
        b = fast_conv(a)
        M = float(b.max())
        g = lp_best_response(a, M)
        if g is None:
            break
        g /= g.sum()
        t_mix = max(0.05, 0.5 * (1.0 - it / max(1, n_iters)))
        a_new = (1.0 - t_mix) * a + t_mix * g
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s < 1e-10:
            break
        a_new /= s
        a = a_new
    return a


def subgradient_polish(a, n_iters, tau, deadline):
    """Subgradient descent with fixed large tau (near-hard max)."""
    a = a.copy()
    a = np.maximum(a, 1e-6)
    a /= a.sum()
    for it in range(n_iters):
        if time.monotonic() > deadline:
            break
        obj, grad = smooth_obj_and_grad(a, tau)
        # Adaptive step size based on gradient magnitude
        g_norm = np.max(np.abs(grad))
        if g_norm < 1e-15:
            break
        lr = 0.5 / (g_norm + 1e-12)
        # Cap step to avoid oscillation
        lr = min(lr, 0.05 * a.mean() / (g_norm + 1e-12))
        a_new = a - lr * grad
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s < 1e-10:
            break
        a_new /= s
        a = a_new
    return a


def resample_sequence(a, new_n):
    """Resample sequence a to have length new_n using interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    t_old = np.linspace(0, 1, n)
    t_new = np.linspace(0, 1, new_n)
    return np.maximum(np.interp(t_new, t_old, a), 0.0)


def propose_candidate(seed=42, budget_s=1500, **kwargs):
    """
    Main search: minimize J(a) = 2n*max(b)/S^2 using multi-phase approach.
    Phase 1: L-BFGS-B with soft-max smoothing and tau annealing
    Phase 2: LP mirror descent for structured updates
    Phase 3: Subgradient polish on best candidate
    Phase 4: Explore larger sequence lengths
    """
    rng = np.random.default_rng(seed)
    start_time = time.monotonic()
    deadline = start_time + float(budget_s) - 20.0

    best = None
    best_score = float('inf')

    def try_update(a):
        nonlocal best, best_score
        a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
        if a.sum() < 0.01:
            return
        score = float(evaluate_sequence(a.tolist()))
        if np.isfinite(score) and score < best_score:
            best = a.copy()
            best_score = score
            elapsed = time.monotonic() - start_time
            print(f"  NEW BEST: J={score:.6f} (n={a.size}, t={elapsed:.0f}s)", flush=True)

    # Try parent sequence
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        if p.ndim == 1 and p.size > 0 and np.all(np.isfinite(p)):
            try_update(p)
    except (NameError, AttributeError):
        pass

    # === Phase 1: L-BFGS-B with tau annealing ===
    print("\n=== Phase 1: L-BFGS-B with annealing ===", flush=True)
    lengths = [512, 1024, 256]
    for n in lengths:
        if time.monotonic() > deadline:
            break
        print(f"\n  L-BFGS-B: n={n}", flush=True)
        t0 = time.monotonic()

        inits = generate_initializations(n, rng)

        # Add resampled parent
        try:
            p = np.asarray(height_sequence_1, dtype=np.float64)
            if p.ndim == 1 and p.size > 0 and np.all(np.isfinite(p)):
                if p.size == n:
                    inits.append(p / p.sum())
                else:
                    inits.append(resample_sequence(p, n))
                    inits[-1] = np.maximum(inits[-1], 0.01)
                    inits[-1] /= inits[-1].sum()
        except (NameError, AttributeError):
            pass

        # Add resampled best (if found)
        if best is not None and best.size != n:
            a_res = resample_sequence(best, n)
            a_res = np.maximum(a_res, 0.01)
            inits.append(a_res / a_res.sum())

        for a0 in inits:
            if time.monotonic() > deadline:
                break
            a0 = np.maximum(a0, 0.0)
            if a0.sum() < 0.01:
                continue
            a0 = a0 / a0.sum()

            # Multi-tau annealing: start smooth, get sharper
            for tau in [2000, 10000, 50000, 200000]:
                if time.monotonic() > deadline:
                    break
                a_refined = lbbfbs_refine(a0, tau, deadline, max_iter=300, max_fun=3000)
                a_refined = np.maximum(a_refined, 0.0)
                if a_refined.sum() > 0.01:
                    a_refined = a_refined / a_refined.sum()
                    try_update(a_refined)
                    a0 = a_refined

        print(f"  n={n} done: {time.monotonic()-t0:.1f}s, best={best_score:.6f}", flush=True)

    # === Phase 2: LP mirror descent ===
    print("\n=== Phase 2: LP Mirror Descent ===", flush=True)
    for n in [256, 512]:
        if time.monotonic() > deadline:
            break
        print(f"\n  LP: n={n}", flush=True)
        t0 = time.monotonic()

        starts = []
        if best is not None:
            if best.size == n:
                starts.append(best.copy())
            else:
                a_r = resample_sequence(best, n)
                a_r = np.maximum(a_r, 0.01)
                starts.append(a_r / a_r.sum())
        # Add a few diverse starts
        inits = generate_initializations(n, rng)
        starts.extend(inits[:3])

        for a0 in starts:
            if time.monotonic() > deadline:
                break
            a_lp = lp_iterate(a0, 60, deadline)
            try_update(a_lp)

        print(f"  n={n} done: {time.monotonic()-t0:.1f}s, best={best_score:.6f}", flush=True)

    # === Phase 3: Subgradient polish + perturbation restarts ===
    if best is not None and time.monotonic() < deadline:
        print("\n=== Phase 3: Subgradient polish ===", flush=True)
        t0 = time.monotonic()
        n = best.size

        # Direct subgradient polish
        a_pol = subgradient_polish(best, 500, 500000, deadline)
        try_update(a_pol)

        # L-BFGS-B final polish with very large tau
        a0 = best.copy()
        a0 = np.maximum(a0, 0.01)
        a0 /= a0.sum()
        a_fin = lbbfbs_refine(a0, 500000, deadline, max_iter=500, max_fun=5000)
        a_fin = np.maximum(a_fin, 0.0)
        if a_fin.sum() > 0.01:
            try_update(a_fin / a_fin.sum())

        # Perturbation restarts
        n_restarts = 5
        for r in range(n_restarts):
            if time.monotonic() > deadline:
                break
            a0 = best.copy().astype(float)
            noise_scale = 0.1 * a0.mean()
            a0 += rng.normal(0, noise_scale, n)
            a0 = np.maximum(a0, 0.0)
            if a0.sum() < 0.01:
                continue
            a0 /= a0.sum()
            for tau in [100000, 500000]:
                if time.monotonic() > deadline:
                    break
                a_ref = lbbfbs_refine(a0, tau, deadline, max_iter=200, max_fun=2000)
                a_ref = np.maximum(a_ref, 0.0)
                if a_ref.sum() > 0.01:
                    try_update(a_ref / a_ref.sum())
                    a0 = a_ref / a_ref.sum()

        print(f"  Phase 3 done: {time.monotonic()-t0:.1f}s, best={best_score:.6f}", flush=True)

    # === Phase 4: Try larger n if time permits ===
    for n in [2048, 4096]:
        if time.monotonic() > deadline:
            break
        remaining = deadline - time.monotonic()
        if remaining < 60:
            break
        print(f"\n=== Phase 4: n={n} (remaining {remaining:.0f}s) ===", flush=True)
        t0 = time.monotonic()

        starts = []
        if best is not None:
            a_r = resample_sequence(best, n)
            a_r = np.maximum(a_r, 0.0)
            starts.append(a_r / max(a_r.sum(), 1e-10))
        starts.append(rng.uniform(0.5, 1.0, n) / (n * 0.75))

        for a0 in starts:
            if time.monotonic() > deadline:
                break
            a0 = np.maximum(a0, 0.0)
            if a0.sum() < 0.01:
                continue
            a0 /= a0.sum()
            for tau in [2000, 20000]:
                if time.monotonic() > deadline:
                    break
                a_ref = lbbfbs_refine(a0, tau, deadline, max_iter=150, max_fun=1500)
                a_ref = np.maximum(a_ref, 0.0)
                if a_ref.sum() > 0.01:
                    try_update(a_ref / a_ref.sum())
                    a0 = a_ref / a_ref.sum()

        print(f"  n={n} done: {time.monotonic()-t0:.1f}s, best={best_score:.6f}", flush=True)

    # Final: ensure we return a valid sequence
    if best is None:
        best = np.ones(256)
        best_score = compute_score(best)

    print(f"\nFINAL: J={best_score:.6f}, n={best.size}", flush=True)
    return [float(x) for x in best]
