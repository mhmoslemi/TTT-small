```python
"""
LP-based iterative optimization with exact line search for minimizing
2n*max(conv(a,a))/sum(a)^2. Uses scipy linprog to find optimal descent
directions, then computes exact maximum step via quadratic constraint analysis.
"""
import numpy as np
import time
from scipy.optimize import linprog
from scipy.sparse import csr_matrix

def build_conv_matrix_sparse(f):
    """Build sparse matrix A where (A@g)[k] = convolve(f,g)[k]."""
    n = len(f)
    # Use COO format for efficient construction
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        j_min = max(0, k - (n - 1))
        j_max = min(k, n - 1)
        for j in range(j_min, j_max + 1):
            rows.append(k)
            cols.append(j)
            vals.append(f[k - j])
    return csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))

def exact_line_search(f, g, M):
    """
    Compute maximum t in [0, 1] such that convolve((1-t)f + tg, (1-t)f + tg) <= M.
    Uses the fact that conv of the blend is quadratic in t.
    """
    conv_f = np.convolve(f, f)
    conv_fg = np.convolve(f, g)
    conv_g = np.convolve(g, g)
    
    # For each k: conv_g[k]*t^2 + 2*conv_fg[k]*t + conv_f[k] - M <= 0
    t_max = 1.0  # Upper bound for blending
    
    for k in range(len(conv_f)):
        A = conv_g[k]
        B = 2.0 * conv_fg[k]
        C = conv_f[k] - M
        
        if C > 0:
            # Already violates at t=0, shouldn't happen
            return 0.0
        
        if A < 1e-15:
            # Linear: B*t + C <= 0
            if B > 1e-15:
                t_k = -C / B
                t_max = min(t_max, t_k)
            # If B <= 0, no constraint (LHS is non-increasing or constant)
        else:
            # Quadratic: A*t^2 + B*t + C <= 0
            # Since A > 0 and C <= 0, there's always a positive root
            disc = B * B - 4.0 * A * C
            if disc < 0:
                # No real roots, quadratic is always positive (since A > 0)
                # But C <= 0 means at t=0 it's <= 0, so there must be a root
                # This case shouldn't actually occur
                t_k = 0.0
            else:
                sqrt_disc = np.sqrt(disc)
                t1 = (-B - sqrt_disc) / (2.0 * A)
                t2 = (-B + sqrt_disc) / (2.0 * A)
                # We need t1 <= t <= t2, and t >= 0
                # The valid range is [max(0, t1), t2]
                t_max = min(t_max, t2)
    
    return max(0.0, min(1.0, t_max))

def lp_iterate_linesearch(f, deadline, verbose_prefix=""):
    """LP iterations with exact line search."""
    f = f.copy().astype(np.float64)
    n = len(f)
    
    # Normalize to sum = 1
    S = float(np.sum(f))
    if S <= 0:
        return f, float('inf')
    f = f / S
    
    best_f = f.copy()
    best_score = 2.0 * n * float(np.max(np.convolve(f, f)))  # sum=1, so denominator is 1
    it = 0
    no_improve = 0
    
    while time.monotonic() < deadline:
        conv_f = np.convolve(f, f)
        M = float(np.max(conv_f))
        
        if M <= 0:
            break
        
        # Build constraint matrix
        A = build_conv_matrix_sparse(f)
        c = -np.ones(n)
        b_ub = np.full(2 * n - 1, M)
        bounds = [(0, None)] * n
        
        try:
            result = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                            options={'presolve': True, 'dual_feasibility_tolerance': 1e-9})
        except Exception:
            no_improve += 1
            if no_improve > 20:
                break
            continue
        
        if not result.success or result.x is None:
            no_improve += 1
            if no_improve > 20:
                break
            continue
        
        g = result.x
        sum_g = float(np.sum(g))
        if sum_g <= 1e-15:
            no_improve += 1
            continue
        
        # Rescale g to have sum = 1 (same as f)
        g = g / sum_g
        
        # Exact line search
        t = exact_line_search(f, g, M)
        
        if t < 1e-10:
            no_improve += 1
            if no_improve > 30:
                break
            continue
        
        # Update
        f_new = (1.0 - t) * f + t * g
        f_new = np.clip(f_new, 0.0, None)
        
        # Renormalize (should be sum=1 already, but clip might affect it)
        S_new = float(np.sum(f_new))
        if S_new > 1e-15:
            f_new = f_new / S_new
        
        new_score = 2.0 * n * float(np.max(np.convolve(f_new, f_new)))
        it += 1
        
        if new_score < best_score:
            f = f_new
            best_f = f.copy()
            best_score = new_score
            no_improve = 0
            if it % 5 == 1:
                print(f"    {verbose_prefix}iter {it}: score={best_score:.6f} t={t:.6f}")
        else:
            no_improve += 1
            if no_improve > 30:
                break
    
    return best_f, best_score

def get_initializations(n, rng):
    """Generate diverse initializations."""
    inits = {}
    inits['uniform'] = np.ones(n)
    
    idx = np.arange(n, dtype=np.float64)
    inits['cosine'] = (1.0 + np.cos(np.pi * idx / max(n-1, 1))) / 2.0
    inits['random'] = rng.uniform(0.5, 1.5, size=n)
    
    center = (n - 1) / 2.0
    inits['bell'] = np.exp(-0.5 * ((idx - center) / (n / 4.0)) ** 2)
    
    two_peak = (np.exp(-0.5 * ((idx - n * 0.25) / (n / 8.0)) ** 2) +
                np.exp(-0.5 * ((idx - n * 0.75) / (n / 8.0)) ** 2))
    inits['bimodal'] = two_peak
    
    # Triangle
    inits['triangle'] = 1.0 - 2.0 * np.abs(idx - center) / n
    
    # Sine
    inits['sine'] = 0.5 + 0.5 * np.sin(np.pi * idx / n)
    
    return inits
```
