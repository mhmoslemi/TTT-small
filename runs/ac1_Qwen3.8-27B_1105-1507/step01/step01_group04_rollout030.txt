"""
Optimizes non-negative step function heights to minimize 2n*max(conv(f,f))/(sum(f))^2.
Uses L-BFGS with log-sum-exp smooth surrogate (temperature annealing),
combined with mirror descent refinement and coordinate descent polish.
Explores multiple sequence lengths for better solution quality.
"""

import numpy as np
import time
from scipy.optimize import minimize

def _fft_convolve_full(f):
    n = len(f)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    F = np.fft.rfft(f, size)
    h = np.fft.irfft(F * F, size)
    return h[:2 * n - 1]

def _score_raw(f):
    n = len(f)
    h = _fft_convolve_full(f)
    M = float(np.max(h))
    S = float(np.sum(f))
    if S < 0.01:
        return float('inf')
    return 2.0 * n * M / (S * S)

def _soft_objective_grad(f, temperature):
    """Compute soft objective and gradient using log-sum-exp."""
    n = len(f)
    h = _fft_convolve_full(f)
    S = float(np.sum(f))
    if S < 1e-10:
        return float('inf'), np.zeros(n)
    
    # Numerically stable log-sum-exp
    Th = temperature * h
    Th_max = np.max(Th)
    exp_Th = np.exp(Th - Th_max)
    sum_exp = np.sum(exp_Th)
    L = (Th_max + np.log(sum_exp)) / temperature  # soft max value
    
    # Softmax weights
    p = exp_Th / sum_exp
    
    # Score
    score = 2.0 * n * L / (S * S)
    
    # Gradient of L w.r.t. f_j = 2 * sum_k p_k * f_{k-j}
    # = 2 * (p corr f)[j]
    grad_L = 2.0 * np.correlate(p, f, mode='valid')[:n]
    
    # Gradient of score = 2n * L / S^2
    grad = 2.0 * n * (grad_L * S - 2.0 * L) / (S ** 3)
    
    return score, grad

def _normalize(f):
    f = np.maximum(f, 0.0)
    s = float(np.sum(f))
    if s < 1e-30:
        return None
    return f / s

def _cosine_init(n):
    j = np.arange(n, dtype=np.float64)
    f = np.cos(np.pi * (j + 0.5) / float(n)) ** 2
    f = np.maximum(f, 0.0)
    s = np.sum(f)
    if s > 0:
        f /= s
    return f

def _gaussian_init(n, sigma_frac):
    j = np.arange(n, dtype=np.float64) - (n - 1) / 2.0
    sigma = sigma_frac * float(n)
    f = np.exp(-j * j / (2.0 * sigma * sigma))
    s = np.sum(f)
    if s > 0:

    return f

def _interpolate_sequence(f, new_n):
    """Interpolate sequence to new length using FFT zero-padding."""
    n = len(f)
    if new_n <= n:
        return f[:new_n]
    size = 1
    while size < new_n:
        size <<= 1
    F = np.fft.rfft(f, size)
    f_interp = np.fft.irfft(F, size)[:new_n]
    f_interp = np.maximum(f_interp, 0.0)
    s = np.sum(f_interp)
    if s > 0:
        f_interp /= s
    else:
        f_interp = np.ones(new_n) / new_n
    return f_interp

def _mirror_descent(f_init, n_iter, eta_scale, threshold_frac):
    """Mirror descent with multiplicative updates."""
    f = _normalize(f_init.copy())
    if f is None:
        return f_init.copy(), float('inf')
    n = len(f)
    best_f = f.copy()
    best_score = _score_raw(f)
    
    for it in range(n_iter):
        h = _fft_convolve_full(f)
        M = float(np.max(h))
        if M <= 0:
            break
        
        threshold = threshold_frac * M + 1e-15
        active = np.where(h >= threshold)[0]
        if len(active) == 0:
            active = np.array([int(np.argmax(h))])
        
        # Compute pressure: p[j] = (1/|K|) * sum_{k in K} 2*f[k-j]
        p = np.zeros(n)
        for k in active:
            j_start = max(0, k - n + 1)
            j_end = min(n - 1, k)
            if j_end >= j_start:
                lo = k - j_end
                hi = k - j_start
                p[j_start:j_end + 1] += 2.0 * f[lo:hi + 1][::-1]
        p /= float(len(active))
        
        eta = eta_scale / M
        log_ratio = np.clip(-eta * p, -30.0, 30.0)
        f = f * np.exp(log_ratio)
        f_norm = _normalize(f)
        if f_norm is None:
            f = np.ones(n) / n
        else:
            f = f_norm
        
        score = _score_raw(f)
        if score < best_score:
            best_score = score
            best_f = f.copy()
    
    return best_f, best_score

"""
Optimizes non-negative step function heights to minimize 2n*max(conv(f,f))/(sum(f))^2.
Uses L-BFGS with log-sum-exp smooth surrogate (temperature annealing) combined with
mirror descent refinement and coordinate descent polish. Explores multiple sequence
lengths (512-2048) for better solution quality via increased degrees of freedom.
"""

import numpy as np
import time
from scipy.optimize import minimize


def _fft_convolve_full(f):
    """Compute full linear convolution of f with itself using FFT."""
    n = len(f)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    F = np.fft.rfft(f, size)
    h = np.fft.irfft(F * F, size)
    return h[:2 * n - 1]


def _score_raw(f):
    """Fast score computation assuming f >= 0, sum(f) > 0.01."""
    n = len(f)
    h = _fft_convolve_full(f)
    M = float(np.max(h))
    S = float(np.sum(f))
    if S < 0.01:
        return float('inf')
    return 2.0 * n * M / (S * S)


def _soft_objective_grad(f, temperature):
    """Compute soft objective (log-sum-exp) and its gradient w.r.t. f."""
    n = len(f)
    h = _fft_convolve_full(f)
    S = float(np.sum(f))
    if S < 1e-10:
        return float('inf'), np.zeros(n)

    # Numerically stable log-sum-exp of T*h
    Th = temperature * h
    Th_max = float(np.max(Th))
    exp_Th = np.exp(Th - Th_max)
    sum_exp = float(np.sum(exp_Th))
    if sum_exp < 1e-300:
        return 2.0 * n * Th_max / temperature / (S * S), np.zeros(n)

    # Soft-max value: L = (1/T) * log(sum exp(Th))
    L = (Th_max + np.log(sum_exp)) / temperature

    # Softmax weights p_k = exp(Th_k) / sum_exp
    p = exp_Th / sum_exp

    # Score = 2n * L / S^2
    score = 2.0 * n * L / (S * S)

    # Gradient of L w.r.t. f_j = sum_k p_k * dh_k/df_j = 2 * sum_k p_k * f_{k-j}
    # This is 2 * cross_correlation(p, f)[j]
    grad_L = 2.0 * np.correlate(p, f, mode='valid')[:n]

    # Gradient of score = 2n * L / S^2
    # d(score)/df_j = 2n * (grad_L_j * S - 2*L) / S^3
    grad = 2.0 * n * (grad_L * S - 2.0 * L) / (S ** 3)

    return score, grad


def _normalize(f):
    """Normalize f to sum=1, clip negatives."""
    f = np.maximum(f, 0.0)
    s = float(np.sum(f))
    if s < 1e-30:
        return None
    return f / s


def _cosine_init(n):
    """Cosine-squared window initialization."""
    j = np.arange(n, dtype=np.float64)
    f = np.cos(np.pi * (j + 0.5) / float(n)) ** 2
    f = np.maximum(f, 0.0)
    s = np.sum(f)
    if s > 0:
        f /= s
    return f


def _gaussian_init(n, sigma_frac):
    """Gaussian centered initialization."""
    j = np.arange(n, dtype=np.float64) - (n - 1) / 2.0
    sigma = sigma_frac * float(n)
    f = np.exp(-j * j / (2.0 * sigma * sigma))
    s = np.sum(f)
    if s > 0:
        f /= s
    return f


def _tapered_init(n, power):
    """Sine-tapered initialization."""
    j = np.arange(n, dtype=np.float64)
    x = np.sin(np.pi * (j + 0.5) / float(n))
    f = x ** power
    f = np.maximum(f, 0.0)
    s = np.sum(f)
    if s > 0:
        f /= s
    return f


def _interpolate_sequence(f, new_n):
    """Interpolate sequence to new length using FFT zero-padding (bandlimited)."""
    n = len(f)
    if new_n <= n:
        f_out = f[:new_n].copy()
        s = np.sum(f_out)
        if s > 0:
            f_out /= s
        return f_out
    size = 1
    while size < new_n:
        size <<= 1
    F = np.fft.rfft(f, size)
    f_interp = np.fft.irfft(F, size)[:new_n]
    f_interp = np.maximum(f_interp, 0.0)
    s = np.sum(f_interp)
    if s > 1e-10:
        f_interp /= s
    else:
        f_interp = np.ones(new_n) / new_n
    return f_interp


def _lbfgs_anneal(f0, n, time_budget):
    """Run L-BFGS with temperature annealing schedule."""
    f = f0.copy()
    start_time = time.monotonic()

    # Temperature schedule: from smooth to sharp
    # h values are ~1/n in magnitude for normalized f
    base = 1.0 / n
    temperatures = [base * 5.0, base * 2.0, base * 0.8, base * 0.3, base * 0.1, base * 0.04]

    n_stages = len(temperatures)
    time_per_stage = time_budget / n_stages

    for stage, T in enumerate(temperatures):
        if time.monotonic() - start_time > time_budget:
            break

        stage_deadline = time.monotonic() + time_per_stage * 1.5

        def _func(x):
            val, _ = _soft_objective_grad(x, T)
            return val

        def _func_grad(x):
            val, g = _soft_objective_grad(x, T)
            return val, g

        maxiter = int(max(50, time_per_stage * 30))
        maxfun = int(max(200, time_per_stage * 80))

        try:
            result = minimize(_func_grad, f, method='L-BFGS-B',
                              bounds=[(0.0, None)] * n,
                              options={'maxiter': maxiter, 'maxfun': maxfun,
                                       'ftol': 1e-14, 'gtol': 1e-10, 'iprint': -1})
            if result.x is not None:
                f_new = np.maximum(result.x, 0.0)
                S = np.sum(f_new)
                if S > 0.01:
                    f = f_new
        except Exception as e:
            print(f"        L-BFGS error: {e}")
            break

        score = _score_raw(f)
        elapsed = time.monotonic() - start_time
        print(f"        Stage {stage+1}/{n_stages} (T={T:.6f}): score={score:.6f} ({elapsed:.1f}s)")

        if time.monotonic() > stage_deadline:
            break

    return f, _score_raw(f)


def _mirror_descent(f_init, n_iter, eta_scale, threshold_frac):
    """Mirror descent with multiplicative updates and active-set subgradient."""
    f = _normalize(f_init.copy())
    if f is None:
        return f_init.copy(), float('inf')
    n = len(f)
    best_f = f.copy()
    best_score = _score_raw(f)

    for it in range(n_iter):
        h = _fft_convolve_full(f)
        M = float(np.max(h))
        if M <= 0:
            break

        threshold = threshold_frac * M + 1e-15
        active = np.where(h >= threshold)[0]
        if len(active) == 0:
            active = np.array([int(np.argmax(h))], dtype=np.int64)

        # Compute pressure vector
        p = np.zeros(n, dtype=np.float64)
        for k in active:
            j_start = max(0, k - n + 1)
            j_end = min(n - 1, k)
            if j_end >= j_start:
                lo = k - j_end
                hi = k - j_start
                p[j_start:j_end + 1] += 2.0 * f[lo:hi + 1][::-1]
        p /= float(len(active))

        eta = eta_scale / (M + 1e-15)
        log_ratio = np.clip(-eta * p, -30.0, 30.0)
        f = f * np.exp(log_ratio)
        f_norm = _normalize(f)
        if f_norm is None:
            f = np.ones(n) / n
        else:
            f = f_norm

        score = _score_raw(f)
        if score < best_score:
            best_score = score
            best_f = f.copy()

    return best_f, best_score


def _coordinate_polish(f, n_iter=50):
    """Coordinate descent: for each coordinate, find optimal value via golden section."""
    n = len(f)
    f = f.copy()
    best_score = _score_raw(f)

    for iteration in range(n_iter):
        improved = False
        for j in range(n):
            old_val = f[j]
            if old_val < 1e-15:
                continue

            # Try a few candidate values for f[j]
            # The optimal f[j] minimizes max_k h_k where h_k depends linearly on f[j]
            # (except at k=2j where it's quadratic)
            # Try: current, 0, and values around the current
            candidates = [0.0, old_val * 0.5, old_val * 0.8, old_val, old_val * 1.2, old_val * 1.5]
            best_cand_val = old_val
            best_cand_score = float('inf')

            for cv in candidates:
                f[j] = cv
                S = np.sum(f)
                if S < 0.01:
                    f[j] = old_val
                    continue
                sc = _score_raw(f)
                if sc < best_cand_score:
                    best_cand_score = sc
                    best_cand_val = cv

            f[j] = best_cand_val
            fn = _normalize(f)
            if fn is not None:
                f = fn
                sc = _score_raw(f)
                if sc < best_score - 1e-8:
                    best_score = sc
                    improved = True

        if not improved:
            break

    return f, _score_raw(f)


def _spectral_flatten(f, n_iter=100):
    """Iterative spectral flattening: try to make |FFT(f)|^2 as flat as possible."""
    n = len(f)
    f = f.copy()

    for it in range(n_iter):

        # Target: flat spectrum. Compute the ideal flat level

        # We want to reduce peaks in the time-domain autocorrelation
        # Equivalent to reducing variance of power spectrum

        # Iterative reweighting: boost frequencies where power is low

        target = np.full_like(power, np.mean(power))
        # Blend: new spectrum = sqrt(old * target) geometric mean

        # Preserve phase

            if s_spec < best_score:
                best_score = s_spec
                best_f = f_spec.copy()
                print(f"    ** NEW BEST (Spectral): {best_score:.6f} **")
            f_cur = f_spec
            s_cur = s_spec

        if time.monotonic() > phase2_deadline:
            break

        # 2d: L-BFGS with low temperature (final polish)
        remaining_d = phase2_deadline - time.monotonic()
        if remaining_d > 20:
            T_low = 0.03 / n
            time_d = min(remaining_d * 0.5, 30.0)
            f_lb, s_lb = _lbfgs_anneal(f_cur, n, time_d)
            # Only use low temperatures
            # Actually _lbfgs_anneal does full schedule, which is fine
            s_lb = _score_raw(f_lb)
            print(f"    L-BFGS polish: score={s_lb:.6f}")
            if s_lb < best_score:
                best_score = s_lb
                best_f = f_lb.copy()
                print(f"    ** NEW BEST (L-BFGS): {best_score:.6f} **")
            f_cur = f_lb
            s_cur = s_lb

    # Phase 3: Coordinate polish on the overall best
    print(f"\nPhase 3: Coordinate polish")
    if time.monotonic() < deadline - 20:
        n_best = len(best_f)
        n_coord_iter = min(20, int((deadline - time.monotonic() - 10) / max(1, n_best) * 50))
        print(f"  n={n_best}, max_coord_iters={n_coord_iter}")
        f_polish, s_polish = _coordinate_polish(best_f, n_coord_iter)
        print(f"  After polish: score={s_polish:.6f}")
        if s_polish < best_score:
            best_score = s_polish
            best_f = f_polish.copy()
            print(f"  ** NEW BEST (polish): {best_score:.6f} **")

    # Phase 4: Final short L-BFGS + MD on the best
    if time.monotonic() < deadline - 15:
        remaining = deadline - time.monotonic() - 10
        print(f"\nPhase 4: Final refinement (budget={remaining:.0f}s)")

        # Short MD burst
        n_iter_final = int(max(100, min(2000, remaining * 3)))
        t0 = time.monotonic()
        f_fin, s_fin = _mirror_descent(best_f, n_iter_final, 0.1, 0.9)
        fin_time = time.monotonic() - t0
        print(f"  Final MD: score={s_fin:.6f} ({fin_time:.1f}s)")
        if s_fin < best_score:
            best_score = s_fin
            best_f = f_fin.copy()
            print(f"  ** NEW BEST (final MD): {best_score:.6f} **")

        # One more L-BFGS pass with very low temperature
        if time.monotonic() < deadline - 10:
            n_best = len(best_f)
            T_very_low = 0.01 / n_best

            def _final_func(x):
                val, _ = _soft_objective_grad(x, T_very_low)
                return val

            def _final_grad(x):
                val, g = _soft_objective_grad(x, T_very_low)
                return val, g

            rem = deadline - time.monotonic() - 5
            try:
                res = minimize(_final_grad, best_f, method='L-BFGS-B',
                              bounds=[(0.0, None)] * n_best,
                              options={'maxiter': int(max(20, rem * 20)),
                                       'ftol': 1e-14, 'gtol': 1e-10, 'iprint': -1})
                if res.x is not None:
                    f_final = np.maximum(res.x, 0.0)
                    sf = np.sum(f_final)
                    if sf > 0.01:
                        s_final = _score_raw(f_final)
                        print(f"  Final L-BFGS: score={s_final:.6f}")
                        if s_final < best_score:
                            best_score = s_final
                            best_f = f_final.copy()
                            print(f"  ** NEW BEST (final L-BFGS): {best_score:.6f} **")
            except Exception as e:
                print(f"  Final L-BFGS error: {e}")

    # Final verification with the actual evaluator
    final_list = [float(max(0.0, min(1000.0, x))) for x in best_f]
    final_score = float(evaluate_sequence(final_list))
    print(f"\nFinal verified: n={len(final_list)}, score={final_score:.6f}")

    # Safety check
    if not np.isfinite(final_score) or final_score > 2.0:
        print("  Fallback to parent.")
        parent_list = [float(max(0.0, min(1000.0, x))) for x in parent]
        return parent_list

    # Also verify parent still works as fallback
    parent_score = float(evaluate_sequence([float(max(0.0, min(1000.0, x))) for x in parent]))
    if final_score > parent_score:
        print(f"  Parent is better ({parent_score:.6f}), using parent.")
        return [float(max(0.0, min(1000.0, x))) for x in parent]

    return final_list
