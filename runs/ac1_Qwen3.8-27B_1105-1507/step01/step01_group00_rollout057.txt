"""
Hybrid subgradient + LP optimization for minimizing 2n*max(conv(a,a))/sum(a)^2.
Uses Nesterov-accelerated subgradient with backtracking, multi-peak gradients,
and periodic LP corrections to escape local minima. Multiple restarts with
diverse initializations across several sequence lengths.

import numpy as np
import time
from scipy.optimize import linprog
from scipy.sparse import csr_matrix

def build_conv_matrix_sparse(f):
    """Build sparse matrix A where (A@g)[k] = convolve(f,g)[k]."""
    n = len(f)
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        j_min = max(0, k - (n - 1))
        j_max = min(k, n - 1)
        for j in range(j_min, j_max + 1):
            rows.append(k)
            cols.append(j)
            vals.append(f[k - j])
    return csr_matrix((np.array(vals), (np.array(rows), np.array(cols))), shape=(2 * n - 1, n))

def solve_lp_direction(f, M, top_k=50):
    """Solve LP: max sum(g) s.t. g >= 0, top conv(f,g)_k <= M."""
    n = len(f)
    c_conv = np.convolve(f, f)
    # Only use top_k constraints for speed
    top_indices = np.argsort(c_conv)[-top_k:]
    A = build_conv_matrix_sparse(f)
    A_top = A[top_indices, :]
    c = -np.ones(n)
    b_ub = np.full(len(top_indices), M)
    bounds = [(0, None)] * n
    try:
        result = linprog(c, A_ub=A_top, b_ub=b_ub, bounds=bounds, method='highs')
        if result.success:
            return result.x
    except Exception:
        pass
    # Fallback: full LP
    try:
        result = linprog(c, A_ub=A, b_ub=np.full(2*n-1, M), bounds=bounds, method='highs')
        if result.success:
            return result.x
    except Exception:
        pass
    return None

    # Get top k peak positions

    top_idx = top_idx[::-1]  # descending order
    weights = c[top_idx]
    total_w = sum(weights)
    if total_w < 1e-15:
        return np.zeros(n), c[top_idx[0]]
    
    grad = np.zeros(n)
    for k_star in top_idx:
        w = c[k_star] / total_w
        j_min = max(0, k_star - (n - 1))
        j_max = min(k_star, n - 1)
        idx = k_star - np.arange(j_min, j_max + 1)
        grad[j_min:j_max + 1] += w * 2.0 * y[idx]

def subgradient_optimize(f_init, deadline, n_sub_iters=5000, top_k=3, use_lp=False, lp_every=500, verbose_prefix=""):
    """
    Nesterov-accelerated subgradient with multi-peak gradient.
    """
    f = f_init.copy().astype(np.float64)
    n = len(f)
    S = float(np.sum(f))
    if S < 0.01:
        print(f"    {verbose_prefix}Sum too small, skipping")
        return f_init, float('inf')
    
    best_f = f.copy()
    best_score = evaluate_sequence(f.tolist())
    
    t = 1.0
    x = f.copy()
    x_prev = f.copy()
    alpha = 2.0 / (1.0 + np.sqrt(n))  # initial step size
    
    rng = np.random.default_rng(int(hash(tuple(f[:10])) % 100000))
    no_improve = 0
    
    for it in range(n_sub_iters):
        if time.monotonic() > deadline:
            break
        
        # Nesterov momentum
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t * t)) / 2.0
        beta = (t - 1.0) / t_new
        y = x + beta * (x - x_prev)
        y = np.clip(y, 0.0, None)
        Sy = float(np.sum(y))
        if Sy > 1e-10:
            y = y * (S / Sy)
        
        # Compute multi-peak gradient
        grad, M_y = compute_multi_peak_grad(y, n, top_k)
        
        # Step
        alpha_t = alpha / t_new
        x_new = y - alpha_t * grad
        x_new = np.clip(x_new, 0.0, None)
        
        # Renormalize
        S_new = float(np.sum(x_new))
        if S_new > 1e-10:
            x_new = x_new * (S / S_new)
        
        # Evaluate
        score_new = evaluate_sequence(x_new.tolist())
        
        if np.isfinite(score_new) and score_new < best_score:
            best_f = x_new.copy()
            best_score = score_new
            no_improve = 0
            if it % 200 == 0:
                print(f"    {verbose_prefix}iter {it}: score={best_score:.6f}")
        else:
            no_improve += 1
        
        # Reheat if stuck
        if no_improve > 50:
            alpha = 2.0 / (1.0 + np.sqrt(n))
            t = 1.0
            no_improve = 0
            perturbation = rng.normal(0, 0.1, size=n)
            x = x + perturbation
            x = np.clip(x, 0.0, None)
            Sx = float(np.sum(x))
            if Sx > 1e-10:
                x = x * (S / Sx)
            x_prev = x.copy()
        
        # Periodic LP correction
        if use_lp and it > 0 and it % lp_every == 0 and time.monotonic() < deadline - 30:
            c_f = np.convolve(x, x)
            M = float(np.max(c_f))
            g = solve_lp_direction(x, M, top_k=30)
            if g is not None:
                sum_g = float(np.sum(g))
                if sum_g > 1e-10:
                    g = g * (S / sum_g)
                    x_blended = 0.7 * x + 0.3 * g
                    x_blended = np.clip(x_blended, 0.0, None)
                    S_b = float(np.sum(x_blended))
                    if S_b > 1e-10:
                        x_blended = x_blended * (S / S_b)
                    score_b = evaluate_sequence(x_blended.tolist())
                    if np.isfinite(score_b) and score_b < best_score:
                        best_f = x_blended.copy()
                        best_score = score_b
                        print(f"    {verbose_prefix}LP correction at iter {it}: score={best_score:.6f}")
                    x = x_blended
                    x_prev = x.copy()
                    t = 1.0
                    no_improve = 0
        
        x_prev = x
        x = x_new
        t = t_new
    
    return best_f, best_score

def lp_iterate(f, deadline, step_size_init=0.1, verbose_prefix="", max_iters=200):
    """LP-based iteration (slower but powerful)."""
    f = f.copy().astype(np.float64)
    best_f = f.copy()
    best_score = evaluate_sequence(f.tolist())
    step_size = step_size_init
    no_improve = 0
    it = 0
    
    while time.monotonic() < deadline and it < max_iters:
        conv_f = np.convolve(f, f)
        M = float(np.max(conv_f))
        if M <= 0:
            break
        
        g = solve_lp_direction(f, M, top_k=50)
        if g is None:
            no_improve += 1
            if no_improve > 20:
                break
            continue
        
        sum_g = float(np.sum(g))
        if sum_g <= 1e-15:
            no_improve += 1
            continue
        
        g = g * (float(np.sum(f)) / sum_g)
        new_f = (1.0 - step_size) * f + step_size * g
        new_f = np.clip(new_f, 0.0, 1000.0)
        
        new_score = evaluate_sequence(new_f.tolist())
        it += 1
        
        if np.isfinite(new_score) and new_score < best_score:
            f = new_f
            best_f = f.copy()
            best_score = new_score
            no_improve = 0
            if it % 10 == 1:
                print(f"    {verbose_prefix}LP iter {it}: score={best_score:.6f}")
        else:
            no_improve += 1
            step_size *= 0.7
            if step_size < 0.003 or no_improve > 30:
                break
    
    return best_f, best_score

def compute_subgradient(y, k_star, n):
    """Compute subgradient of C[k_star] = sum_i y[i]*y[k_star-i] w.r.t. y."""
    grad = np.zeros(n)
    j_min = max(0, k_star - (n - 1))
    j_max = min(k_star, n - 1)
    if j_max >= j_min:
        idx = k_star - np.arange(j_min, j_max + 1)
        grad[j_min:j_max + 1] = 2.0 * y[idx]
    return grad


def compute_multi_peak_subgradient(y, n, top_k=3):
    """Compute weighted subgradient using top-k peaks of conv(y,y)."""
    c = np.convolve(y, y)
    top_idx = np.argsort(c)[-top_k:]
    top_idx = top_idx[::-1]
    weights = c[top_idx].copy()
    total_w = float(np.sum(weights))
    if total_w < 1e-15:
        return np.zeros(n), c[top_idx[0]], top_idx
    weights = weights / total_w
    grad = np.zeros(n)
    for i, k_star in enumerate(top_idx):
        j_min = max(0, k_star - (n - 1))
        j_max = min(k_star, n - 1)
        if j_max >= j_min:
            idx = k_star - np.arange(j_min, j_max + 1)
            grad[j_min:j_max + 1] += weights[i] * 2.0 * y[idx]
    return grad, c[top_idx[0]], top_idx

            # For each k where f[j] appears in C[k]:
            #   if k-j is valid and k != 2j: C[k] = fj*f[k-j]*2 + rest, need <= M
            #     => fj <= (M - rest) / (2*f[k-j])
            #   if k == 2j: C[k] = fj^2 + rest, need <= M
            #     => fj <= sqrt(M - rest)
            # We compute the tightest bound
            limit = float('inf')
            for k in range(2 * n - 1):
                kj = k - j
                if 0 <= kj < n:
                    if k == 2 * j:
                        rest = C[k] - fj * fj
                        if rest >= M:
                            limit = 0.0
                            break
                        val = np.sqrt(max(0.0, M - rest))
                        limit = min(limit, val)
                    else:
                        fkj = f[kj]
                        if fkj > 1e-15:
                            rest = C[k] - 2.0 * fj * fkj
                            if rest >= M:
                                limit = 0.0
                                break
                            val = (M - rest) / (2.0 * fkj)
                            limit = min(limit, val)
            f[j] = min(limit, 1000.0) if limit < float('inf') else fj

        # Renormalize to maintain sum
        S_new = float(np.sum(f))
        if S_new > 1e-10:
            f = f * (S / S_new)
            # After rescaling, some C[k] might exceed M again; that's ok for one pass

    return f


def build_conv_matrix_sparse(f):
    """Build sparse matrix A where (A@g)[k] = convolve(f,g)[k]."""
    n = len(f)
    total_entries = 0
    for k in range(2 * n - 1):
        j_min = max(0, k - (n - 1))
        j_max = min(k, n - 1)
        total_entries += (j_max - j_min + 1)

    rows = np.empty(total_entries, dtype=np.int32)
    cols = np.empty(total_entries, dtype=np.int32)
    vals = np.empty(total_entries, dtype=np.float64)
    idx = 0
    for k in range(2 * n - 1):
        j_min = max(0, k - (n - 1))
        j_max = min(k, n - 1)
        for j in range(j_min, j_max + 1):
            rows[idx] = k
            cols[idx] = j
            vals[idx] = f[k - j]
            idx += 1
    return csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))


def solve_lp_direction(f, M, top_k_constraints=None):
    """
    Solve: max sum(g) s.t. g >= 0, conv(f,g)[k] <= M for selected k.
    If top_k_constraints is given, only use that many tightest constraints.
    """
    n = len(f)
    A = build_conv_matrix_sparse(f)
    c_vec = -np.ones(n)
    bounds = [(0, None)] * n

    if top_k_constraints is not None and top_k_constraints < (2 * n - 1):
        c_conv = np.convolve(f, f)
        top_idx = np.argsort(c_conv)[-top_k_constraints:]
        A_sub = A[top_idx, :]
        b_ub = np.full(len(top_idx), M)
    else:
        A_sub = A
        b_ub = np.full(2 * n - 1, M)

    try:
        result = linprog(c_vec, A_ub=A_sub, b_ub=b_ub, bounds=bounds, method='highs')
        if result.success and result.fun < 0:
            return result.x
    except Exception:
        pass
    return None


def lp_polish(f, deadline, step_size=0.1, max_iters=50, top_k_constraints=50,
              verbose_prefix=""):
    """LP-based direction finding with blending."""
    f = f.copy().astype(np.float64)
    n = len(f)
    S = float(np.sum(f))
    best_f = f.copy()
    best_score = evaluate_sequence(f.tolist())
    ts = step_size
    no_improve = 0

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        c_f = np.convolve(f, f)
        M = float(np.max(c_f))
        if M <= 0:
            break

        g = solve_lp_direction(f, M, top_k_constraints)
        if g is None:
            no_improve += 1
            if no_improve > 10:
                break
            continue

        sum_g = float(np.sum(g))
        if sum_g < 1e-10:
            no_improve += 1
            continue

        g = g * (S / sum_g)
        new_f = (1.0 - ts) * f + ts * g
        new_f = np.clip(new_f, 0.0, 1000.0)
        new_f = project_nonneg_sum(new_f, S)

        new_score = evaluate_sequence(new_f.tolist())
        if np.isfinite(new_score) and new_score < best_score:
            f = new_f
            best_f = f.copy()
            best_score = new_score
            no_improve = 0
            print(f"    {verbose_prefix}LP iter {it}: score={best_score:.6f}")
        else:
            no_improve += 1
            ts *= 0.7
            if ts < 0.005 or no_improve > 15:
                break

    return best_f, best_score


def propose_candidate(budget_s=1500, **kwargs):
    t0 = time.monotonic()
    deadline = t0 + max(0.0, float(budget_s) - 5.0)

    best_sequence = None
    best_score = float('inf')
    rng = np.random.default_rng(777)

    # Load parent
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64).copy()
        parent = np.clip(parent, 0.0, 1000.0)
        if parent.ndim == 1 and parent.size > 0 and np.all(np.isfinite(parent)):
            score = evaluate_sequence(parent.tolist())
            if np.isfinite(score):
                best_score = score
                best_sequence = parent.copy()
                print(f"Parent n={parent.size}: score={score:.6f}")
    except (NameError, Exception) as e:
        print(f"Could not load parent: {e}")

    # Determine lengths to try
    if best_sequence is not None:
        p_len = len(best_sequence)
        lengths = sorted(set([p_len, max(256, p_len // 2), min(2048, p_len * 2)]))
    else:
        lengths = [256, 512, 1024]

    # Time allocation per length
    time_per_length = max(60.0, (deadline - time.monotonic()) / max(1, len(lengths)))

    for n in lengths:
        if time.monotonic() > deadline - 30:
            break

        print(f"\n=== Length n={n} (budget {time_per_length:.0f}s) ===")
        length_deadline = min(deadline - 10, time.monotonic() + time_per_length)

        # Generate diverse initializations
        inits = []

        # Parent (if matching length)
        if best_sequence is not None and len(best_sequence) == n:
            inits.append(('parent', best_sequence.copy()))

        # Parent interpolated (if different length)
        if best_sequence is not None and len(best_sequence) != n:
            x_old = np.linspace(0, 1, len(best_sequence))
            x_new = np.linspace(0, 1, n)
            inits.append(('parent_interp', np.interp(x_new, x_old, best_sequence)))

        # Uniform
        inits.append(('uniform', np.ones(n)))

        # Cosine
        idx = np.arange(n, dtype=np.float64)
        inits.append(('cosine', 0.5 + 0.5 * np.cos(np.pi * idx / max(1, n - 1))))

        # Sine
        inits.append(('sine', np.abs(np.sin(np.pi * (idx + 0.5) / n))))

        # Random
        inits.append(('random', rng.uniform(0.2, 1.0, size=n)))

        # Chirp-inspired: cos^2 of quadratic phase
        phase = np.pi * (idx / n) ** 2
        inits.append(('chirp', 0.5 + 0.5 * np.cos(2 * phase)))

        # Bipolar (two peaks)
        c1, c2 = n * 0.3, n * 0.7
        inits.append(('bimodal',
                      np.exp(-((idx - c1) ** 2) / (2 * (n / 6) ** 2)) +
                      np.exp(-((idx - c2) ** 2) / (2 * (n / 6) ** 2))))

        # Asymmetric (mass shifted left)
        inits.append(('asym_left', np.exp(-idx / (n * 0.4))))

        # Asymmetric (mass shifted right)
        inits.append(('asym_right', np.exp(-(n - 1 - idx) / (n * 0.4))))

        time_per_init = max(20.0, (length_deadline - time.monotonic()) / max(1, len(inits)))

        for label, f_init in inits:
            if time.monotonic() > length_deadline - 10:
                break

            f_init = np.clip(f_init.astype(np.float64), 0.0, 1000.0)
            S_init = float(np.sum(f_init))
            if S_init < 0.01:
                continue

            init_score = evaluate_sequence(f_init.tolist())
            if not np.isfinite(init_score):
                continue

            print(f"  [{label}] init={init_score:.6f}", end="")

            # Set target sum for optimization (scale-invariant, use sqrt(2n))
            S_target = np.sqrt(2.0 * n)
            f_norm = f_init * (S_target / S_init)

            sub_deadline = min(length_deadline - 5, time.monotonic() + time_per_init)

            # Run Nesterov proximal gradient
            best_f, best_s = nesterov_proximal_gradient(
                f_norm, sub_deadline, S=S_target,
                top_k=3, max_iters=50000, reheat_every=150,
                verbose_prefix=f"[{label}] ", verbose_interval=2000
            )

            # Water-filling polish
            if time.monotonic() < sub_deadline:
                c_best = np.convolve(best_f, best_f)
                M_best = float(np.max(c_best))
                # Try to reduce M by water-filling with a tighter target
                wf_target = M_best * 0.95
                wf_f = waterfill_sweep(best_f, wf_target, n_passes=2)
                wf_f = project_nonneg_sum(wf_f, S_target)
                wf_score = evaluate_sequence(wf_f.tolist())
                if np.isfinite(wf_score) and wf_score < best_s:
                    best_f = wf_f
                    best_s = wf_score
                    print(f" wf={wf_score:.6f}", end="")

            if best_s < best_score:
                best_score = best_s
                best_sequence = best_f.copy()
                print(f" -> {best_score:.6f}")
            else:
                print(f" -> no improvement (best={init_score:.6f})")
            print()

    # === Final refinement phase ===
    if best_sequence is not None and time.monotonic() < deadline - 20:
        print(f"\n=== Final refinement (n={len(best_sequence)}, score={best_score:.6f}) ===")
        n_final = len(best_sequence)
        S_final = np.sqrt(2.0 * n_final)
        f_work = best_sequence * (S_final / float(np.sum(best_sequence)))

        # More aggressive Nesterov with higher top_k
        refine_deadline = min(deadline, time.monotonic() + (deadline - time.monotonic()) * 0.4)
        fin_f, fin_s = nesterov_proximal_gradient(
            f_work, refine_deadline, S=S_final,
            top_k=5, max_iters=100000, reheat_every=100,
            verbose_prefix="[refine] ", verbose_interval=2000
        )
        if fin_s < best_score:
            best_score = fin_s
            best_sequence = fin_f.copy()
            print(f"  Refine Nesterov: {best_score:.6f}")

        # Water-filling
        if time.monotonic() < deadline - 10:
            c_fin = np.convolve(best_sequence, best_sequence)
            M_fin = float(np.max(c_fin))
            for reduction in [0.98, 0.96, 0.94]:
                wf_f = waterfill_sweep(best_sequence, M_fin * reduction, n_passes=3)
                wf_f = project_nonneg_sum(wf_f, S_final)
                wf_score = evaluate_sequence(wf_f.tolist())
                if np.isfinite(wf_score) and wf_score < best_score:
                    best_score = wf_score
                    best_sequence = wf_f.copy()
                    print(f"  Waterfill (red={reduction}): {best_score:.6f}")
                    M_fin = wf_score * S_final ** 2 / (2 * n_final)

        # LP polish (only for n <= 1024)
        if n_final <= 1024 and time.monotonic() < deadline - 30:
            lp_deadline = min(deadline, time.monotonic() + 120)
            lp_f, lp_s = lp_polish(best_sequence, lp_deadline, step_size=0.08,
                                   max_iters=40, top_k_constraints=60,
                                   verbose_prefix="[LP] ")
            if lp_s < best_score:
                best_score = lp_s
                best_sequence = lp_f.copy()
                print(f"  LP polish: {best_score:.6f}")

    if best_sequence is None:
        return [1.0] * 128

    # Final sanity check
    best_sequence = np.clip(best_sequence, 0.0, 1000.0)
    final_score = evaluate_sequence(best_sequence.tolist())
    if not np.isfinite(final_score):
        print("WARNING: final score is not finite, returning parent or fallback")
        return [1.0] * 128

    elapsed = time.monotonic() - t0
    print(f"\nFINAL: n={len(best_sequence)}, score={final_score:.6f}, time={elapsed:.1f}s")
    return [float(x) for x in best_sequence]
