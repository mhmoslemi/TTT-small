"""
Minimizes J(a) = 2*n*max(convolve(a,a))/(sum(a))^2 using:
1. LP-based best-response (fictitious play) iterations with adaptive steps
2. Analytical subgradient descent with projection onto simplex

import numpy as np
from scipy.optimize import linprog, minimize
import time


def fast_conv(a):
    """Full self-convolution via FFT."""
    n = len(a)
    s = 1
    while s < 2 * n - 1:
        s <<= 1
    fa = np.fft.rfft(a, s)
    return np.fft.irfft(fa * fa, s)[:2 * n - 1]


def compute_score(a):
    """Compute J(a) = 2*n*max(conv)/(sum)^2."""
    n = len(a)
    S = a.sum()
    if S < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2.0 * n * b.max() / (S * S))


def build_A(a):
    """Build constraint matrix A where (A @ g)[k] = (a*g)[k]."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for k in range(2 * n - 1):
        jl = max(0, k - n + 1)
        jh = min(n - 1, k)
        if jl <= jh:
            A[k, jl:jh + 1] = a[k - jh:k - jl + 1][::-1]
    return A


def lp_best_response(a, M):
    """LP: max sum(g) s.t. (a*g)[k] <= M, g >= 0."""
    n = len(a)
    c = -np.ones(n)
    A = build_A(a)
    r = linprog(c, A_ub=A, b_ub=np.full(2 * n - 1, M),
               bounds=[(0, None)] * n, method='highs')
    if r.success:
        g = r.x
        if g.sum() > 1e-10:
            return g
    return None


def lp_descent_direction(a, M):
    """Find direction d minimizing max_k (d*a)[k] s.t. sum(d)=0, -a <= d <= a."""
    n = len(a)
    # Variables: d (n vars), t (1 var)
    # Minimize t
    # (d*a)[k] - t <= 0 for all k
    # sum(d) = 0
    # -a <= d <= a (to keep a + eps*d >= 0 for small eps)
    
    c = np.zeros(n + 1)
    c[-1] = 1.0  # minimize t
    
    # Inequality: (d*a)[k] - t <= 0
    A = np.zeros((2 * n - 1, n + 1))
    A_mat = build_A(a)
    A[:, :n] = A_mat
    A[:, n] = -1.0  # -t
    b_ub = np.zeros(2 * n - 1)
    
    # Equality: sum(d) = 0
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    
    # Bounds: -a <= d <= a, t <= some_large
    bounds = [(-a[i], a[i]) for i in range(n)] + [(-1e10, 1e10)]
    
    r = linprog(c, A_ub=A, b_ub=b_ub, A_eq=A_eq, b_eq=[0.0],
               bounds=bounds, method='highs')
    if r.success:
        return r.x[:n], r.x[-1]
    return None, None


def softmax_obj(x, tau):
    """Smooth objective using softmax approximation of max."""
    a = np.maximum(x, 0.0)
    S = a.sum()
    if S < 1e-8:
        return 1e10
    a = a / S
    n = len(a)
    b = fast_conv(a)
    b_shift = b - b.max()
    smooth_max = b.max() + tau * np.log(np.exp(b_shift / tau).sum()) - tau * np.log(len(b))
    # Actually: tau*log(sum(exp(b/tau))) = b.max() + tau*log(sum(exp((b-b.max())/tau)))
    # We want this to approximate b.max() as tau -> 0
    # sum(exp((b-b.max())/tau)) is between 1 and len(b)
    # So smooth_max is between b.max() and b.max() + tau*log(len(b))
    # For our purposes, just use b.max() + tau*log(sum(exp((b-b.max())/tau)))
    # and accept the small bias
    smooth_max = b.max() + tau * np.log(np.exp(b_shift / tau).sum())
    return 2.0 * n * smooth_max


def project_simplex(v, target_sum):
    """Project v onto {x >= 0, sum(x) = target_sum}."""
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - target_sum))[0][-1]
    theta = (cssv[rho] - target_sum) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def lp_iteration(a, max_iters=500, deadline=None, verbose=True):
    """Run LP best-response iterations. Returns best sequence and score found."""
    n = len(a)
    a = a.copy()
    S = a.sum()
    a = a / S  # normalize
    
    best_a = a.copy()
    best_score = compute_score(a)
    if verbose:
        print(f"  LP start: score={best_score:.6f}", flush=True)
    
    no_improve = 0
    step = 0.5
    
    for it in range(max_iters):
        if deadline is not None and time.monotonic() > deadline:
            break
        
        b = fast_conv(a)
        M = float(b.max())
        
        g = lp_best_response(a, M)
        if g is None:
            break
        
        g = g / g.sum()
        
        # Adaptive step size
        a_new = (1.0 - step) * a + step * g
        a_new = np.maximum(a_new, 0.0)
        S_new = a_new.sum()
        if S_new < 1e-10:
            break
        a_new = a_new / S_new
        
        score_new = compute_score(a_new)
        
        if score_new < best_score - 1e-10:
            best_a = a_new.copy()
            best_score = score_new
            no_improve = 0
            # Increase step if improving fast
            if it < max_iters * 0.3:
                step = min(0.8, step * 1.05)
            else:
                step = min(0.6, step * 1.02)
            if verbose and (it % 50 == 0 or score_new < 1.51):
                print(f"  LP it={it}: score={best_score:.6f}, step={step:.3f}", flush=True)
        else:
            a_new = (1.0 - step * 0.5) * a + step * 0.5 * g
            a_new = np.maximum(a_new, 0.0)
            S_new = a_new.sum()
            if S_new < 1e-10:
                break
            a_new = a_new / S_new
            no_improve += 1
            if no_improve % 10 == 0:
                step = max(0.05, step * 0.85)
        
        a = a_new
    
    return best_a, best_score


def gradient_refinement(a, max_iters=300, deadline=None, verbose=True):
    """Analytical subgradient descent on the simplex."""
    n = len(a)
    a = a.copy()
    S = a.sum()
    a = a / S
    
    best_a = a.copy()
    best_score = compute_score(a)
    if verbose:
        print(f"  Grad start: score={best_score:.6f}", flush=True)
    
    step = 0.1
    momentum = np.zeros(n)
    beta = 0.9
    
    for it in range(max_iters):
        if deadline is not None and time.monotonic() > deadline:
            break
        
        b = fast_conv(a)
        k_star = int(np.argmax(b))
        M = b[k_star]
        S = a.sum()
        
        # Subgradient: df/da[j] = 2*n*(S*a[k-j] - 2*M*a[j]) / S^3
        grad = np.zeros(n)
        for j in range(n):
            kj = k_star - j
            if 0 <= kj < n:
                grad[j] = a[kj]
            grad[j] -= 2.0 * M * a[j] / S
        grad *= 2.0 * n / (S * S)
        # Actually: dJ/da[j] = 2n*(S*a[k*-j] - 2*M*a[j])/S^3
        # Let me recompute:
        # J = 2n*M/S^2
        # dJ/da[j] = 2n*(dM/da[j]*S^2 - M*2S*da_j)/S^4 = 2n*(dM/da[j]*S - 2M*a[j])/S^3
        # dM/da[j] = a[k*-j] if 0<=k*-j<n else 0
        
        grad = np.zeros(n)
        for j in range(n):
            kj = k_star - j
            val = 0.0
            if 0 <= kj < n:
                val = a[kj]
            grad[j] = val * S - 2.0 * M * a[j]
        grad *= 2.0 * n / (S ** 3)
        
        # Momentum update
        momentum = beta * momentum - step * grad
        a_new = a + momentum
        a_new = project_simplex(a_new, 1.0)
        
        score_new = compute_score(a_new)
        if score_new < best_score:
            best_a = a_new.copy()
            best_score = score_new
            if verbose and (it % 30 == 0 or score_new < 1.51):
                print(f"  Grad it={it}: score={best_score:.6f}", flush=True)
            # Slightly increase step
            step = min(0.3, step * 1.01)
        else:
            step *= 0.95
            a_new = a
        
        a = a_new
    
    return best_a, best_score


def lBFGS_refinement(a, deadline=None, verbose=True):
    """L-BFGS-B with smooth objective."""
    n = len(a)
    a0 = a.copy()
    S = a0.sum()
    a0 = a0 / S
    
    tau = 0.05  # softmax temperature
    
    res = minimize(
        lambda x: softmax_obj(x, tau),
        a0,
        method='L-BFGS-B',
        bounds=[(0, None)] * n,
        options={'maxiter': 200, 'max_fun': 5000, 'ftol': 1e-14, 'disp': False,
                 'maxls': 50, 'maxcor': 20, 'iprint': -1}
    )
    
    if res.fun < 1e10:
        a_opt = np.maximum(res.x, 0)
        S = a_opt.sum()
        if S > 1e-8:
            a_opt = a_opt / S
            return a_opt
    return a0


def local_search(a, max_iters=2000, deadline=None, verbose=True):
    """Random coordinate/local perturbation search."""
    n = len(a)
    rng = np.random.default_rng(12345)
    a = a.copy()
    S = a.sum()
    a = a / S
    
    best_a = a.copy()
    best_score = compute_score(a)
    if verbose:
        print(f"  Local start: score={best_score:.6f}", flush=True)
    
    perturb_scale = 0.1
    
    for it in range(max_iters):
        if deadline is not None and time.monotonic() > deadline:
            break
        
        # Random perturbation of a few coordinates
        n_perturb = rng.integers(1, max(2, n // 64))
        idx = rng.choice(n, size=n_perturb, replace=False)
        
        a_new = a.copy()
        for i in idx:
            delta = rng.uniform(-perturb_scale, perturb_scale) * a.mean()
            a_new[i] = max(0, a_new[i] + delta)
        
        S_new = a_new.sum()
        if S_new < 1e-8:
            continue
        a_new = a_new / S_new
        
        score_new = compute_score(a_new)
        if score_new < best_score:

            best_score = score_new
            a = a_new
            if verbose and (it % 100 == 0 or score_new < 1.51):
                print(f"  Local it={it}: score={best_score:.6f}", flush=True)
        else:
            # Occasionally reduce perturb scale
            if it % 200 == 0:
                perturb_scale *= 0.9
    
    return best_a, best_score


def generate_diverse_inits(n, rng):
    """Generate diverse initializations of length n."""
    inits = []
    
    # Uniform
    inits.append(np.ones(n))
    
    # Random
    inits.append(rng.uniform(0.1, 1.0, n))
    
    # Cosine patterns
    t = np.linspace(0, 1, n, endpoint=False)
    for freq in [2, 3, 4, 6, 8]:
        inits.append(np.maximum(1.0 + 0.5 * np.cos(2 * np.pi * freq * t), 0.01))
    
    # Chirp
    inits.append(np.maximum(1.0 + 0.4 * np.cos(2 * np.pi * t ** 2 * 4), 0.01))
    
    # Bimodal
    inits.append(np.exp(-4 * (t - 0.3) ** 2) + np.exp(-4 * (t - 0.7) ** 2) + 0.01)
    
    # Triangular
    inits.append(1.0 - np.abs(2 * t - 1) + 0.1)
    
    # S-shaped
    inits.append(1.0 + 0.3 * np.sin(np.pi * t) ** 3)
    
    # High freq oscillation
    inits.append(np.maximum(1.0 + 0.6 * np.cos(2 * np.pi * 16 * t), 0.01))
    
    # Random with smooth structure
    low_freq = rng.uniform(-0.3, 0.3, min(n, 16))
    smooth = np.interp(np.linspace(0, 1, n), np.linspace(0, 1, len(low_freq)), low_freq)
    inits.append(np.maximum(1.0 + smooth, 0.01))
    
    for a in inits:
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s > 1e-8:
            a = a / s
        inits[-1] = a
    
    return inits

        print(f"  n={n} done: {time.monotonic()-t0:.1f}s elapsed, best={best_score:.6f}", flush=True)

    print(f"\n=== Phase 3: Gradient refinement ===", flush=True)
    t0 = time.monotonic()
    
    if best_seq is not None:
        a = best_seq / best_seq.sum()
        
        # Subgradient descent
        a, sc = gradient_refinement(a, max_iters=200, deadline=deadline, verbose=True)
        try_update(a, "grad")
        
        # L-BFGS-B
        if time.monotonic() < deadline - 60:
            print(f"  L-BFGS-B: n={a.size}", flush=True)
            a_lbfgs = lBFGS_refinement(a, deadline=deadline, verbose=True)
            try_update(a_lbfgs, "LBFGS")
            
            # Another round of gradient after L-BFGS
            a, sc = gradient_refinement(a_lbfgs, max_iters=100, deadline=deadline, verbose=False)
            try_update(a, "grad-after-LBFGS")
        
        # More LP after gradient
        if time.monotonic() < deadline - 60:
            print(f"  LP after gradient: n={a.size}", flush=True)
            a, sc = lp_iteration(a, max_iters=150, deadline=deadline, verbose=True)
            try_update(a, "LP-after-grad")
    
    # Phase 4: Local search
    print(f"\n=== Phase 4: Local search ===", flush=True)
    t0 = time.monotonic()
    
    if best_seq is not None:
        a = best_seq / best_seq.sum()
        a, sc = local_search(a, max_iters=3000, deadline=deadline, verbose=True)
        try_update(a, "local")
        
        # LP after local
        if time.monotonic() < deadline - 30:
            a, sc = lp_iteration(a, max_iters=100, deadline=deadline, verbose=False)
            try_update(a, "LP-after-local")
            
            # Another local
            a, sc = local_search(a, max_iters=2000, deadline=deadline, verbose=False)
            try_update(a, "local-2")
    
    # Phase 5: Final attempt - try the descent direction LP
    print(f"\n=== Phase 5: Descent direction LP ===", flush=True)
    t0 = time.monotonic()
    
    if best_seq is not None and time.monotonic() < deadline - 30:
        a = best_seq / best_seq.sum()
        n = a.size
        
        for it in range(50):
            if time.monotonic() > deadline - 10:
                break
            
            b = fast_conv(a)
            M = float(b.max())
            
            d, t_val = lp_descent_direction(a, M)
            if d is None:
                break
            
            # Take a step in direction d
            for alpha in [0.1, 0.3, 0.5, 0.7]:
                a_new = a + alpha * d
                a_new = np.maximum(a_new, 0.0)
                S = a_new.sum()
                if S < 1e-8:
                    continue
                a_new = a_new / S
                sc = compute_score(a_new)
                if sc < compute_score(a):
                    a = a_new
                    break
            
            try_update(a, f"descent-{it}")
    
    # Phase 6: Final local polish
    print(f"\n=== Phase 6: Final polish ===", flush=True)
    if best_seq is not None and time.monotonic() < deadline - 10:
        a = best_seq / best_seq.sum()
        a, sc = local_search(a, max_iters=5000, deadline=deadline, verbose=False)
        try_update(a, "final-polish")
        
        # One last LP
        if time.monotonic() < deadline - 5:
            a, sc = lp_iteration(a, max_iters=50, deadline=deadline, verbose=False)
            try_update(a, "final-LP")
    
    # Ensure we return something valid
    if best_seq is None:
        # Fallback: uniform of length 256
        best_seq = np.ones(256)
        best_score = compute_score(best_seq)
        print(f"  FALLBACK: uniform, J={best_score:.6f}", flush=True)
    
    total_time = time.monotonic() - start_time
    print(f"\n{'='*60}", flush=True)
    print(f"FINAL: J={best_score:.6f}, n={best_seq.size}, time={total_time:.1f}s", flush=True)
    print(f"{'='*60}", flush=True)
    
    return [float(x) for x in best_seq]
