    else:

def subgradient_update(a, eta, K=50, force_sym=False):
    """Subgradient step focusing on top-K peaks."""
    n = a.size
    c = np.convolve(a, a)
    M = float(np.max(c))
    if M <= 0:
        return a.copy()
    
    # Find top-K active positions
    if K >= c.size:
        active = np.where(c > M * 0.01)[0]
    else:
        active = np.argpartition(c, -K)[-K:]
    
    # Compute pressure vector
    pressure = np.zeros(n)
    for k in active:
        weight = c[k] / M
        s = max(0, k - n + 1)
        e = min(n - 1, k)
        for i in range(s, e + 1):
            j = k - i
            pressure[i] += weight * a[j]
            if j != i:
                pressure[j] += weight * a[i]
    
    # Normalize pressure
    pmax = float(np.max(pressure))
    if pmax <= 0:
        return a.copy()
    
    # Multiplicative update
    # Reduce a[i] where pressure is high, increase where low
    mean_p = float(np.mean(pressure))
    fac = eta * (mean_p - pressure) / pmax
    fac = np.clip(fac, -3.0, 3.0)
    a_new = a * np.exp(fac)
    a_new = np.clip(a_new, 0.0, None)
    
    if force_sym:
        a_new = sym_normalize(a_new)
    else:
        a_new = normalize1(a_new)
    
    return a_new if a_new is not None else a.copy()

def spectral_shaping(a, n_iter=50, force_sym=False):
    """Gerchberg-Saxton style spectral flattening."""
    n = a.size
    a = a.copy()
    # Zero-pad to at least 2n for linear convolution via FFT
    L = 1
    while L < 2 * n - 1:
        L *= 2
    
    for it in range(n_iter):
        # Forward FFT
        A = np.fft.rfft(a, L)
        # Target: flat magnitude
        mag = np.abs(A)
        mmean = float(np.mean(mag[mag > 1e-15])) if np.any(mag > 1e-15) else 1.0
        # Set magnitude to mean (flatten spectrum)
        phase = np.angle(A)
        A_flat = mmean * np.exp(1j * phase)
        # Inverse FFT
        a_new = np.fft.irfft(A_flat, L)[:n]
        # Project to non-negative
        a_new = np.maximum(a_new, 0.0)
        # Normalize
        a_new = normalize1(a_new)
        if a_new is None:
            break
        a = a_new
    
    if force_sym:
        a = sym_normalize(a)
    return a

def lp_step(a, seed=0):
    """LP relaxation step."""
    n = a.size
    c = np.convolve(a, a)
    M = float(np.max(c))
    if M <= 0:
        return None
    
    # Build constraint matrix: A_ub @ x <= b
    # (a*x)[k] <= M for all k
    # (a*x)[k] = sum_i a[i] * x[k-i]
    # This is a Toeplitz-like matrix
    
    # Use symmetry to reduce size
    # x is the new sequence, we want to maximize sum(x) subject to:
    #   For all k: sum_i a[i]*x[k-i] <= M
    #   x >= 0
    
    # Build the matrix
    A_rows = []
    for k in range(2 * n - 1):
        row = np.zeros(n)
        s = max(0, k - n + 1)
        e = min(n - 1, k)
        for i in range(s, e + 1):
            row[i] = a[k - i]
        A_rows.append(row)
    
    A = np.array(A_rows)
    b = np.full(2 * n - 1, M * (1 + 1e-9))
    c_obj = -np.ones(n)  # minimize -sum(x) = maximize sum(x)
    bounds = [(0, 1)] * n
    
    try:
        res = linprog(c_obj, A_ub=A, b_ub=b, bounds=bounds, method='highs')
        if res.success:
            x = np.clip(res.x, 0, 1)
            x = normalize1(x)
            return x
    except:
        pass
    return None

def coordinate_polish(a, n_steps=200, force_sym=False):
    """Coordinate descent: try moving each element to reduce the max convolution."""
    n = a.size
    a = a.copy()
    best_score = local_score(a)
    
    for step in range(n_steps):
        improved = False
        # Try moving each coordinate
        order = np.random.permutation(n) if step % 5 == 0 else np.arange(n)
        for j in order:
            # Try different values for a[j]
            c = np.convolve(a, a)
            M = float(np.max(c))
            s = float(np.sum(a))
            
            # The contribution of a[j] to c[k] is a[j]*a[k-j]
            # We want to reduce the max, so try reducing a[j]
            # or moving mass from a[j] to a position that doesn't contribute to peaks
            
            # Simple: try a[j] * factor for several factors
            best_j = a[j]
            best_local = M
            for factor in [0.5, 0.7, 0.85, 1.0, 1.15, 1.3, 1.5]:
                trial = a.copy()
                trial[j] = a[j] * factor
                # Redistribute the difference
                diff = a[j] - trial[j]
                if abs(diff) > 1e-15:
                    # Add diff uniformly to other elements
                    mask = np.ones(n)
                    mask[j] = 0
                    trial[mask] += diff / (n - 1)
                    trial = np.maximum(trial, 0)
                    trial = normalize1(trial)
                if trial is None:
                    continue
                c_trial = np.convolve(trial, trial)
                M_trial = float(np.max(c_trial))
                if M_trial < best_local:
                    best_local = M_trial
                    best_j = trial.copy()
            
            if best_local < M * (1 - 1e-6):
                a = best_j
                if force_sym:
                    a = sym_normalize(a)
                if local_score(a) < best_score:
                    best_score = local_score(a)
                    improved = True
        
        if not improved and step > 50:
            break
    
    return a, best_score

def optimize_with_subgrad(a, max_time, seed=0, force_sym=False):
    """Main optimization loop using subgradient method."""
    a = normalize1(a)
    if force_sym:
        a = sym_normalize(a)
    if a is None:
        return None, float('inf')
    
    best = a.copy()
    best_sc = local_score(best)
    t0 = time.monotonic()
    deadline = t0 + max_time
    
    eta = 0.3
    K = max(10, a.size // 10)
    stuck = 0
    rng = np.random.default_rng(seed)
    
    while time.monotonic() < deadline:
        a_new = subgradient_update(a, eta, K, force_sym)
        sc = local_score(a_new)
        
        if sc < best_sc:
            best = a_new.copy()
            best_sc = sc
            a = a_new
            stuck = 0
            # Slightly increase eta if improving
            eta = min(eta * 1.01, 0.5)
        else:
            a = best.copy()  # Revert
            stuck += 1
            if stuck > 30:
                # Reduce step size
                eta *= 0.8
                if eta < 0.05:
                    # Random restart from best
                    noise = np.exp(rng.normal(0, 0.1, best.size))
                    a = best * noise
                    if force_sym:
                        a = sym_normalize(a)
                    else:
                        a = normalize1(a)
                    eta = 0.3
                    stuck = 0
        
        # Occasionally try spectral shaping
        if stuck > 50 and rng.random() < 0.3:
            a_spec = spectral_shaping(a, 20, force_sym)
            sc_spec = local_score(a_spec)
            if sc_spec < best_sc:
                best = a_spec.copy()
                best_sc = sc_spec
                a = a_spec
                stuck = 0
                eta = 0.3
        
        # Occasionally try LP
        if stuck > 80 and rng.random() < 0.2 and a.size <= 1024:
            a_lp = lp_step(best, seed)
            if a_lp is not None:
                for t in [0.1, 0.2, 0.3, 0.5, 0.7]:
                    blend = (1-t) * best + t * a_lp
                    if force_sym:
                        blend = sym_normalize(blend)
                    else:
                        blend = normalize1(blend)
                    if blend is not None:
                        sc_blend = local_score(blend)
                        if sc_blend < best_sc:
                            best = blend.copy()
                            best_sc = sc_blend
                            a = blend
                            stuck = 0
                            break
    
    return best, best_sc

def make_initial(n, seed, kind=0):
    """Generate an initial candidate sequence."""
    rng = np.random.default_rng(seed)
    
    if kind == 0:
        # Random uniform
        return rng.uniform(0.1, 1.0, n)
    elif kind == 1:
        # Sine wave
        x = np.linspace(0, 2*np.pi, n)
        freq = rng.uniform(0.5, 3.0)
        return np.maximum(0.5 + 0.5 * np.sin(freq * x), 0.01)
    elif kind == 2:
        # Raised cosine
        x = np.linspace(0, 1, n)
        return 0.5 * (1 + np.cos(2 * np.pi * x))
    elif kind == 3:
        # Gaussian bump
        x = np.linspace(0, 1, n)
        center = rng.uniform(0.3, 0.7)
        width = rng.uniform(0.1, 0.3)
        return np.exp(-((x - center) / width) ** 2)
    elif kind == 4:
        # M-sequence based
        # Generate a simple m-sequence using LFSR
        m = max(2, int(np.log2(n)) + 1)
        # Use a simple feedback polynomial
        # For simplicity, use a PRBS-like pattern
        state = rng.integers(1, 2**m)
        seq = []
        for _ in range(n):
            bit = (state ^ (state >> 1)) & 1
            seq.append(bit)
            state = (state >> 1) | ((bit << (m-1)) & (2**m - 1))
        # Map to non-negative with some smoothing
        arr = np.array(seq, dtype=float)
        arr = 0.3 + 0.7 * arr
        # Smooth
        if n > 3:
            kern = np.ones(3)/3
            arr = np.convolve(arr, kern, mode='same')
        return np.maximum(arr, 0.01)
    else:
        # Beta distribution
        return rng.beta(2, 2, n)
