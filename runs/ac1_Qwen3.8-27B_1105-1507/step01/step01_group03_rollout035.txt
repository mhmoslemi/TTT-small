"""
Hybrid optimization for minimizing 2*n*max(conv(f,f))/(sum(f))^2.
Uses L-BFGS on softmax parameterization for fast local optimization,
combined with LP-based refinement and multi-scale search.
"""

import time
import numpy as np
from scipy.optimize import minimize, linprog
from scipy.sparse import csr_matrix


def normalize_simplex(x):
    x = np.asarray(x, dtype=np.float64)
    x = np.maximum(x, 0.0)
    s = x.sum()
    if s < 1e-300:
        return np.ones_like(x) / len(x)
    return x / s


def score_of(x):
    """Score for normalized x (sum=1)."""
    n = len(x)
    conv = np.convolve(x, x)
    return 2.0 * n * float(np.max(conv))


def softmax(theta):
    theta = theta - theta.max()
    e = np.exp(theta)
    return e / e.sum()


def lse_objective_and_grad(theta, beta):
    """Compute log-sum-exp objective and gradient w.r.t. theta."""
    f = softmax(theta)
    n = len(f)
    g = np.convolve(f, f)  # length 2n-1
    bg = beta * g
    bg_shifted = bg - bg.max()
    log_sum = bg_shifted.max() + np.log(np.sum(np.exp(bg_shifted)))
    # Actually: log(sum exp(bg)) = bg.max() + log(sum exp(bg - bg.max()))
    obj = log_sum  # this is log(sum exp(beta * g[k]))
    
    # Gradient w.r.t. f: 2*beta*w[k]*f[k-j] where w = softmax(beta*g)
    w = np.exp(bg_shifted)
    w /= w.sum()
    
    # FFT-based correlation
    N = 1
    while N < 3 * n:
        N *= 2
    W = np.fft.rfft(w, N)
    F = np.fft.rfft(f, N)
    C = np.fft.irfft(W * np.conj(F), N)[:n]
    grad_f = 2.0 * beta * C
    
    # Chain rule: d(theta) -> d(f)
    # f = softmax(theta), df[l]/dtheta[j] = f[l]*(delta_{lj} - f[j])
    # dObj/dtheta[j] = sum_l grad_f[l] * f[l]*(delta_{lj} - f[j])
    # = grad_f[j]*f[j] - f[j]*sum_l grad_f[l]*f[l]
    dot = np.dot(grad_f, f)
    grad_theta = f * (grad_f - dot)
    
    return obj, grad_theta


def optimize_lse(f_init, beta, max_iter=500, time_limit=60.0):
    """Optimize using L-BFGS on softmax parameterization."""
    n = len(f_init)
    f0 = normalize_simplex(f_init)
    
    # Initialize theta
    theta0 = np.log(f0 + 1e-30)
    
    start_time = time.monotonic()
    
    def fun(theta):
        if time.monotonic() - start_time > time_limit:
            raise TimeoutError()
        obj, grad = lse_objective_and_grad(theta, beta)
        return obj, grad
    
    try:
        res = minimize(fun, theta0, method='L-BFGS-B', maxiter=max_iter)
        f_result = softmax(res.x)
    except TimeoutError:

    except Exception:

    f_result = normalize_simplex(f_result)
    return f_result, score_of(f_result)
