    return val, grad_x

"""

Combines: (1) L-BFGS gradient descent on log-sum-exp smoothed objective with
increasing temperature, (2) LP-based direction finding, (3) multi-scale

def normalize_sequence(arr):
    """Return a nonnegative floating vector normalized to sum 1, or None."""
    try:
        x = np.asarray(arr, dtype=np.float64).ravel()
    except Exception:
        return None
    if x.size == 0:
        return None
    x = np.where(np.isfinite(x), x, 0.0)
    x = np.maximum(x, 0.0)
    s = float(x.sum())
    if s <= 1e-300:
        return None
    x /= s
    return x


def score_normalized(x, n):
    """Score = 2*n*max(conv(x,x)) assuming sum(x)=1."""
    conv = np.convolve(x, x)
    m = float(np.max(conv))
    if not np.isfinite(m):
        return float("inf")
    return float(2.0 * n * m)


def make_window(n, kind='hann', rng=None):
    """Create a window function of length n normalized to sum 1."""
    t = np.linspace(0, 1, n)
    if kind == 'uniform':
        w = np.ones(n)
    elif kind == 'hann':
        w = 0.5 * (1.0 + np.cos(2.0 * np.pi * t))
    elif kind == 'hamming':
        w = 0.54 - 0.46 * np.cos(2.0 * np.pi * t)
    elif kind == 'blackman':
        w = 0.42 - 0.50 * np.cos(2.0 * np.pi * t) + 0.08 * np.cos(4.0 * np.pi * t)
    elif kind == 'gaussian':
        center = (n - 1) / 2.0
        sigma = n / 4.0
        w = np.exp(-0.5 * ((np.arange(n) - center) / sigma) ** 2)
    elif kind == 'triangle':
        w = 1.0 - np.abs(2.0 * t - 1.0)
    elif kind == 'raised_cos2':
        w = 0.5 * (1.0 + np.cos(2.0 * np.pi * t)) ** 2
    else:
        w = np.ones(n)
    if rng is not None and rng.random() < 0.3:
        w = w * (0.8 + 0.4 * rng.random(n))
    return normalize_sequence(w)


def interpolate_normalize(x, n):
    """Resample a normalized nonnegative sequence to length n."""
    x = normalize_sequence(x)
    if x is None or n <= 0:
        return None
    m = x.size
    if m == n:
        return x
    if m == 1:
        return np.ones(n) / float(n)
    old = np.linspace(0.0, 1.0, m)
    new = np.linspace(0.0, 1.0, n)
    out = np.interp(new, old, x)
    return normalize_sequence(out)


def softmax_stable(x):
    """Numerically stable softmax."""
    x = x - np.max(x)
    e = np.exp(x)
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(x) / x.size
    return e / s


def compute_smoothed_obj_and_grad(x_log, t):
    """
    Compute the log-sum-exp smoothed max of auto-conv and its gradient
    w.r.t. x_log where f = softmax(x_log).
    Returns (objective_value, gradient).
    """
    f = softmax_stable(x_log)
    n = f.size
    conv = np.convolve(f, f)  # length 2n-1
    tconv = t * conv
    max_tc = float(np.max(tconv))

    if max_tc > 500:
        k_star = int(np.argmax(conv))
        val = conv[k_star]
        grad_f = np.zeros(n)
        for i in range(n):
            j = k_star - i
            if 0 <= j < n:
                grad_f[i] = 2.0 * f[j]
    else:
        exp_shifted = np.exp(tconv - max_tc)
        sum_exp = float(np.sum(exp_shifted))
        p = exp_shifted / sum_exp
        val = (max_tc + np.log(sum_exp)) / t

        # grad_f[i] = 2 * sum_j f[j] * p[i+j]
        grad_f = np.zeros(n)
        for i in range(n):
            j_max = min(n - 1, 2 * n - 2 - i)
            if j_max >= 0:
                grad_f[i] = 2.0 * np.dot(f[:j_max + 1], p[i:i + j_max + 1])

    # Chain rule: d/dx_log_i = f_i * (grad_f_i - sum_j grad_f_j * f_j)
    dot_gf = float(np.dot(grad_f, f))
    grad_x = f * (grad_f - dot_gf)
    return val, grad_x


def optimize_lbgfs(n, f_init, time_budget=60.0):
    """
    Run L-BFGS-B optimization with increasing temperature schedule.
    Returns (best_f_normalized, best_score).
    """
    f_init = np.maximum(f_init, 1e-30)
    f_init = f_init / f_init.sum()
    x_log = np.log(f_init)

    best_f = f_init.copy()
    best_score = score_normalized(f_init, n)

    start_time = time.monotonic()
    temps = [5.0, 20.0, 50.0, 150.0, 400.0, 1000.0, 3000.0]

    for t_val in temps:
        elapsed = time.monotonic() - start_time
        if elapsed > time_budget * 0.85:
            break
        t_budget = min(time_budget - elapsed - 2.0, time_budget * 0.15)
        if t_budget < 1.0:
            break

        def obj_grad(x):
            val, grad = compute_smoothed_obj_and_grad(x, t_val)
            return val, grad

        try:
            res = minimize(obj_grad, x_log, method='L-BFGS-B',
                           options={'maxiter': 200, 'maxfun': 500,
                                    'gtol': 1e-8, 'ftol': 1e-12})
            f_new = softmax_stable(res.x)
            f_new = np.maximum(f_new, 0.0)
            s = f_new.sum()
            if s > 1e-300:
                f_new /= s
            sc = score_normalized(f_new, n)
            if np.isfinite(sc) and sc < best_score:
                best_score = sc
                best_f = f_new.copy()
                x_log = np.log(np.maximum(f_new, 1e-30))
        except Exception:
            pass

        elapsed = time.monotonic() - start_time
        if elapsed > time_budget:
            break

    return best_f, best_score


def make_conv_template(n):
    """Precompute structure for the LP direction problem."""
    m = 2 * n - 1
    k64 = np.arange(m)
    js = np.maximum(0, k64 - n + 1).astype(np.int32)
    je = np.minimum(n - 1, k64).astype(np.int32)
    counts = (je.astype(np.int64) - js.astype(np.int64) + 1).astype(np.int64)
    total = int(counts.sum())

    cum = np.cumsum(counts)
    starts = np.repeat(cum - counts, counts)
    local = np.arange(total) - starts

    row = np.repeat(k64, counts).astype(np.int32)
    col_base = js[row.astype(np.int64)]
    col = (col_base + local).astype(np.int32)
    offsets = (row.astype(np.int64) - col.astype(np.int64)).astype(np.int32)

    from scipy.sparse import csr_matrix
    A_off = csr_matrix((offsets.astype(np.float64), (row, col)), shape=(m, n))
    edge_offsets = A_off.data.astype(np.int32, copy=True)
    A_template = A_off.copy()
    A_template.data.fill(0.0)

    c_obj = -np.ones(n)
    bounds = [(0.0, None)] * n
    return A_template, edge_offsets, c_obj, bounds, m


def solve_lp_direction(template, f, n, M):
    """Solve the LP: max sum(g) s.t. (f*g)_k <= M, g >= 0."""
    A_template, edge_offsets, c_obj, bounds, m = template
    A = A_template.copy()
    A.data[:] = f[edge_offsets]
    b = np.full(m, M)
    try:
        res = linprog(c_obj, A_ub=A, b_ub=b, bounds=bounds, method='highs',
                      options={'time_limit': 30})
        return res
    except Exception:
        return None


def optimize_lp(n, f_init, time_budget=60.0, template=None):
    """Run LP-based iterative optimization. Returns (best_f, best_score)."""
    f = normalize_sequence(f_init)
    if f is None:
        f = np.ones(n) / n
    if template is None:
        try:
            template = make_conv_template(n)
        except Exception:
            return f, score_normalized(f, n)

    best_f = f.copy()
    best_score = score_normalized(f, n)
    start_time = time.monotonic()
    stall = 0
    rng = np.random.default_rng(int(time.monotonic() * 1e6) % 100000)

    while time.monotonic() - start_time < time_budget * 0.9:
        conv_ff = np.convolve(f, f)
        M = float(np.max(conv_ff))
        if not np.isfinite(M) or M <= 0:
            break

        res = solve_lp_direction(template, f, n, M)
        x = getattr(res, 'x', None)
        if x is None:
            stall += 1
            if stall > 10:
                break
            # Perturb and retry
            f = f + 0.1 * rng.normal(size=n)
            f = np.maximum(f, 0)
            f = f / f.sum()
            continue

        x = np.asarray(x, dtype=np.float64)
        x = np.maximum(x, 0)
        s_star = float(x.sum())
        if s_star <= 1e-300:
            stall += 1
            if stall > 10:
                break
            f = f + 0.1 * rng.normal(size=n)
            f = np.maximum(f, 0)
            f = f / f.sum()
            continue

        s_ratio = s_star / 1.0  # f is normalized to sum 1
        if s_ratio <= 1.0 + 1e-8:
            stall += 1
            if stall > 10:
                break
            f = f + 0.1 * rng.normal(size=n)
            f = np.maximum(f, 0)
            f = f / f.sum()
            continue

        g = x / s_star
        r = np.convolve(f, g)
        r_max = float(np.max(r))
        if not np.isfinite(r_max) or r_max <= 0:
            stall += 1
            if stall > 10:
                break
            continue

        if r_max >= M * (1.0 - 1e-10):
            stall += 1
            if stall > 10:
                break
            f = f + 0.05 * rng.normal(size=n)
            f = np.maximum(f, 0)
            f = f / f.sum()
            continue

        G = float(np.max(np.convolve(g, g)))
        if not np.isfinite(G):
            stall += 1
            if stall > 10:
                break
            continue

        a_coef = 2.0 * (r_max - M)
        b_coef = G + M - 2.0 * r_max
        if b_coef > 1e-14:
            t_max = -a_coef / b_coef
        else:
            t_max = 1.0
        if not np.isfinite(t_max) or t_max <= 0:
            t_max = 1e-8
        base = min(1.0, 0.7 * t_max)
        if base < 1e-12:
            base = 1e-12

        ts = [base, 0.5 * base, 0.25 * base, 0.1 * base, 0.01 * base]
        best_cand = None
        best_cand_sc = best_score

        for tv in ts:
            if tv <= 0 or tv > 1:
                continue
            f_new = (1.0 - tv) * f + tv * g
            f_new = np.maximum(f_new, 0)
            ss = f_new.sum()
            if ss < 1e-300:
                continue
            f_new /= ss
            sc = score_normalized(f_new, n)
            if np.isfinite(sc) and sc < best_cand_sc:
                best_cand_sc = sc
                best_cand = f_new.copy()

        if best_cand is not None:
            f = best_cand
            best_f = f.copy()
            best_score = best_cand_sc
            stall = 0
        else:
            stall += 1
            mix_eps = 0.2 + 0.1 * stall
            noise = rng.normal(size=n) * 0.05
            f = f + noise
            f = np.maximum(f, 0)
            f = f / f.sum()
            if stall > 15:
                break

    return best_f, best_score


def multiplicative_descent(n, f_init, time_budget=30.0, lr=0.1, n_iter=200):
    """
    Mirror descent (multiplicative weights) on the smoothed max objective.
    f_new = f * exp(-lr * grad) / sum(...)
    """
    f = normalize_sequence(f_init)
    if f is None:
        f = np.ones(n) / n
    best_f = f.copy()
    best_score = score_normalized(f, n)
    start_time = time.monotonic()
    t_val = 50.0

    for it in range(n_iter):
        if time.monotonic() - start_time > time_budget * 0.9:
            break
        # Increase temperature slowly
        t_val = 50.0 * (1.0 + it / 50.0)

        conv = np.convolve(f, f)
        tconv = t_val * conv
        max_tc = np.max(tconv)
        if max_tc > 500:
            k_star = int(np.argmax(conv))
            grad_f = np.zeros(n)
            for i in range(n):
                j = k_star - i
                if 0 <= j < n:
                    grad_f[i] = 2.0 * f[j]
        else:
            exp_s = np.exp(tconv - max_tc)
            p = exp_s / exp_s.sum()
            grad_f = np.zeros(n)
            for i in range(n):
                j_max = min(n - 1, 2 * n - 2 - i)
                if j_max >= 0:
                    grad_f[i] = 2.0 * np.dot(f[:j_max + 1], p[i:i + j_max + 1])

        # Multiplicative update
        f_new = f * np.exp(-lr * grad_f / max(np.max(np.abs(grad_f)), 1e-15))
        f_new = np.maximum(f_new, 1e-30)
        f_new /= f_new.sum()
        sc = score_normalized(f_new, n)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_f = f_new.copy()
            f = f_new
            lr = min(lr * 1.02, 1.0)
        else:
            lr *= 0.95
            if lr < 1e-6:
                break

    return best_f, best_score


def propose_candidate(seed=42, budget_s=1500, **kwargs):
    """
    Search for a low-scoring nonnegative sequence using multi-strategy approach:
    gradient-based (L-BFGS), LP-based, and multiplicative descent, combined with
    multi-scale upsampling and diverse initializations.
    """
    rng = np.random.default_rng(int(seed) if seed is not None else 42)
    total_budget = max(1.0, float(budget_s))
    start_time = time.monotonic()
    deadline = start_time + total_budget - 20.0  # reserve 20s for final eval

    best_f = None
    best_score = float('inf')
    given_f = None

    # Load provided construction
    try:
        raw = np.asarray(height_sequence_1, dtype=np.float64)
        given_f = normalize_sequence(raw)
        if given_f is not None:
            best_f = given_f.copy()
            best_score = score_normalized(given_f, given_f.size)
            print(f"Given: n={given_f.size}, score={best_score:.6f}", flush=True)
    except Exception:
        pass

    if best_f is None:
        best_f = make_window(64, 'hann', rng)
        best_score = score_normalized(best_f, best_f.size)
        print(f"Initialized: n={best_f.size}, score={best_score:.6f}", flush=True)

    # Multi-scale plan: (n, total_time_allocated)
    plan = [
        (32, 15),
        (64, 25),
        (128, 50),
        (256, 120),
        (512, 280),
        (1024, 550),
        (2048, 300),
    ]

    window_kinds = ['hann', 'blackman', 'gaussian', 'hamming', 'raised_cos2', 'triangle']

    for n, alloc in plan:
        rem = deadline - time.monotonic()
        if rem < 15:
            break
        alloc = min(alloc, rem - 5)
        if alloc < 10:
            continue

        print(f"\n=== n={n}, alloc={alloc:.0f}s, best_so_far={best_score:.6f} ===", flush=True)

        # Generate starts
        starts = []

        # 1. Upsampled best
        if best_f is not None:
            up = interpolate_normalize(best_f, n)
            if up is not None:
                starts.append(up)

        # 2. Window functions
        for wk in window_kinds[:3]:
            w = make_window(n, wk, rng)
            if w is not None:
                # Check diversity
                diverse = True
                for s in starts:
                    if np.linalg.norm(w - s) < 0.05:
                        diverse = False
                        break
                if diverse:
                    starts.append(w)

        # 3. Given sequence upscaled (if different from best)
        if given_f is not None and (best_f is None or given_f.size != n):
            g_up = interpolate_normalize(given_f, n)
            if g_up is not None:
                starts.append(g_up)

        if not starts:
            starts.append(np.ones(n) / n)

        # Cap number of starts
        n_starts = min(len(starts), 4)
        starts = starts[:n_starts]

        # Allocate time among starts
        # Give more time to the first start (upsampled best)
        if n_starts == 1:
            weights = [1.0]
        elif n_starts == 2:
            weights = [0.6, 0.4]
        elif n_starts == 3:
            weights = [0.45, 0.30, 0.25]
        else:
            weights = [0.35, 0.25, 0.20, 0.20]

        # Precompute LP template
        try:
            template = make_conv_template(n)
        except Exception:
            template = None

        for idx in range(n_starts):
            rem = deadline - time.monotonic()
            if rem < 5:
                break
            t_avail = alloc * weights[idx]
            t_avail = min(t_avail, rem - 3)
            if t_avail < 3:
                continue

            print(f"  Start {idx+1}/{n_starts}, budget={t_avail:.1f}s", flush=True)
            init = starts[idx]

            # Split time among methods
            t_grad = t_avail * 0.40
            t_lp = t_avail * 0.35
            t_mult = t_avail * 0.25

            # 1. L-BFGS gradient optimization
            if t_grad > 2.0:
                f_g, s_g = optimize_lbgfs(n, init, time_budget=t_grad)
                if s_g < best_score:
                    best_score = s_g
                    best_f = f_g.copy()
                    print(f"    L-BFGS: {s_g:.6f}", flush=True)
                init = f_g  # Use as start for next method

            # 2. LP-based optimization
            if t_lp > 2.0:
                f_l, s_l = optimize_lp(n, init, time_budget=t_lp, template=template)
                if s_l < best_score:
                    best_score = s_l
                    best_f = f_l.copy()
                    print(f"    LP: {s_l:.6f}", flush=True)
                init = f_l

            # 3. Multiplicative descent
            if t_mult > 2.0:
                f_m, s_m = multiplicative_descent(n, init, time_budget=t_mult)
                if s_m < best_score:
                    best_score = s_m
                    best_f = f_m.copy()
                    print(f"    MultDesc: {s_m:.6f}", flush=True)
                init = f_m

            # 4. One more round of L-BFGS from the best of the above
            rem2 = deadline - time.monotonic()
            if rem2 > 10 and t_avail * 0.15 > 3:
                t_final = min(t_avail * 0.15, rem2 - 5)
                f_fin, s_fin = optimize_lbgfs(n, init, time_budget=t_final)
                if s_fin < best_score:
                    best_score = s_fin
                    best_f = f_fin.copy()
                    print(f"    Final L-BFGS: {s_fin:.6f}", flush=True)

        try:
            del template
        except Exception:
            pass

    # Final polish: run L-BFGS + LP on the best solution found
    rem = deadline - time.monotonic()
    if rem > 10 and best_f is not None:
        n_best = best_f.size
        print(f"\n=== Final polish: n={n_best}, budget={rem:.0f}s ===", flush=True)
        t_polish = rem * 0.5
        f_p, s_p = optimize_lbgfs(n_best, best_f, time_budget=t_polish)
        if s_p < best_score:
            best_score = s_p
            best_f = f_p.copy()
            print(f"  Polish L-BFGS: {s_p:.6f}", flush=True)

        rem2 = deadline - time.monotonic()
        if rem2 > 5:
            try:
                template = make_conv_template(n_best)
                f_l, s_l = optimize_lp(n_best, best_f, time_budget=rem2 * 0.8, template=template)
                if s_l < best_score:
                    best_score = s_l
                    best_f = f_l.copy()
                    print(f"  Polish LP: {s_l:.6f}", flush=True)
                del template
            except Exception:
                pass

    # Construct final output
    if best_f is None:
        best_f = np.ones(64) / 64

    # The evaluate_sequence function is scale-invariant, so we can use any scale.
    # But to be safe, let's use a reasonable scale.
    seq = [float(x) for x in best_f]

    # Verify with evaluate_sequence
    try:
        final_score = float(evaluate_sequence(seq))
    except Exception:
        final_score = float('inf')

    if not np.isfinite(final_score):
        # Fallback
        seq = [1.0] * 64
        try:
            final_score = float(evaluate_sequence(seq))
        except Exception:
            pass

    print(f"\nFINAL: n={len(seq)}, local={best_score:.6f}, evaluate={final_score:.6f}", flush=True)
    return seq
