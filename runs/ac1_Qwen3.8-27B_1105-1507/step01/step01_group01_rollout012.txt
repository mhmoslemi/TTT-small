def score_of(a):

    a = np.clip(a, 0.0, 1000.0)

    n = a.size
    n_full = 2 * n - 1
    # Compute softmax weights
    c_max = c.max()
    scaled = T * (c - c_max)
    scaled = np.clip(scaled, -50, 50)
    exp_s = np.exp(scaled)
    p = exp_s / exp_s.sum()

    g -= g.mean()

except:
    HAVE_SCIPY = False


def score_of(a):
    a = np.asarray(a, dtype=np.float64)
    n = a.size
    if n == 0:
        return float('inf')
    a = np.clip(a, 0.0, 1000.0)
    s = a.sum()
    if s < 1e-12:
        return float('inf')
    c = np.convolve(a, a)
    return 2.0 * n * c.max() / (s * s)


def compute_smooth_grad(a, c, T):
    n = a.size
    n_full = 2 * n - 1
    c_max = c.max()
    scaled = T * (c - c_max)
    scaled = np.clip(scaled, -50.0, 50.0)
    exp_s = np.exp(scaled)
    p = exp_s / exp_s.sum()
    
    # g[i] = 2 * sum_{j=0}^{n-1} a[j] * p[i+j]
    # Use as_strided for zero-copy view
    try:
        from numpy.lib.stride_tricks import as_strided
        row_stride = p.strides[0]
        mat = as_strided(p, shape=(n, n), strides=(row_stride, row_stride))
        g = mat.T.dot(a)
    except:
        g = np.zeros(n)
        for i in range(n):
            g[i] = np.dot(a, p[i:i+n])
    g *= 2.0
    g -= g.mean()
    return g


def do_smooth_step(a, eta, T):
    n = a.size
    c = np.convolve(a, a)
    g = compute_smooth_grad(a, c, T)
    a_new = a * np.exp(-eta * g)
    a_new = np.clip(a_new, 1e-30, 1000.0)
    s = a_new.sum()
    if s < 1e-30:
        return a.copy(), score_of(a)
    a_new /= s
    return a_new, score_of(a_new)


def do_hard_step(a, eta, K=20):
    n = a.size
    c = np.convolve(a, a)
    M = c.max()
    if K >= c.size:
        active = np.where(c > 1e-15)[0]
    else:
        active = np.argpartition(c, -K)[-K:]
    g = np.zeros(n)
    for k in active:
        s = max(0, k - n + 1)
        e = min(k, n - 1)
        if s > e:
            continue
        i = np.arange(s, e + 1)
        g[i] += 2.0 * a[k - i]
    if active.size > 0:
        g /= active.size
    g -= g.mean()
    a_new = a * np.exp(-eta * g)
    a_new = np.clip(a_new, 1e-30, 1000.0)
    s = a_new.sum()
    if s < 1e-30:
        return a.copy(), score_of(a)
    a_new /= s
    return a_new, score_of(a_new)


def lp_jolt(a, seed):
    """LP-based direction finding (simplified)."""
    if not HAVE_SCIPY:
        return None
    a = np.asarray(a, dtype=np.float64)
    n = a.size
    c = np.convolve(a, a)
    M = c.max()
    if M <= 0:
        return None
    
    # Build constraint matrix: for each k in [0, 2n-2], conv(a, g)[k] <= M * sum(g)
    # Actually, we want: for all k, sum_j a[j]*g[k-j] <= M
    # With the constraint sum(g) = 1, g >= 0
    # This is: A @ g <= M * ones, where A[k, j] = a[k-j] (valid) or 0
    
    n_full = 2 * n - 1
    A = np.zeros((n_full, n))
    for k in range(n_full):
        for j in range(n):
            idx = k - j
            if 0 <= idx < n:
                A[k, j] = a[idx]
    
    cvec = -np.ones(n)
    bvec = np.full(n_full, M * 1.0001)
    bounds = [(0.0, None)] * n
    
    try:
        res = linprog(cvec, A_ub=A, b_ub=bvec, bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': 30})
    except:
        return None
    
    if res is None or not res.success:
        return None
    
    g = np.clip(res.x, 0, None)
    s = g.sum()
    if s < 1e-15:
        return None
    g /= s
    return g


def optimize_sequence(a, budget, seed):
    """Main optimization loop for a single sequence length."""
    a = np.asarray(a, dtype=np.float64)
    n = a.size
    a = np.clip(a, 0.0, 1000.0)
    s = a.sum()
    if s < 1e-12:
        a = np.ones(n) / n
    else:
        a /= s
    
    best = a.copy()
    best_sc = score_of(best)
    
    t0 = time.monotonic()
    deadline = t0 + budget
    rng = np.random.default_rng(seed)
    
    eta = 0.3
    T = 5.0 / c_max  # Initial temperature (will be adaptive)
    c = np.convolve(a, a)
    c_max = c.max()
    T = 5.0 / max(c_max, 1e-10)
    
    stuck = 0
    it = 0
    mode = 'smooth'  # 'smooth' or 'hard'
    hard_K = 20
    
    while time.monotonic() < deadline:
        rem = deadline - time.monotonic()
        if rem < 0.5:
            break
        
        # Choose mode
        if it % 200 == 0:
            # Periodically switch mode and adjust hyperparams
            if stuck > 50:
                mode = 'hard' if mode == 'smooth' else 'smooth'
                T *= 1.5
                hard_K = min(hard_K + 5, n // 4)
                eta = max(0.05, eta * 0.7)
            elif stuck > 20:
                T *= 0.7  # Smoother
                eta = min(eta * 1.2, 1.0)
        
        # Take a step
        if mode == 'smooth':
            a_new, sc = do_smooth_step(a, eta, T)
        else:
            a_new, sc = do_hard_step(a, eta, hard_K)
        
        if sc < best_sc:
            best = a_new.copy()
            best_sc = sc
            stuck = 0
            # Slightly increase eta when making progress
            eta = min(eta * 1.01, 1.5)
        else:
            a = a_new  # Still move (helps escape flat regions)
            stuck += 1
            if stuck > 100:
                # Reheat: add noise and reset
                noise = np.exp(rng.normal(0, 0.3, n))
                a = best * noise
                a = np.clip(a, 1e-30, 1000.0)
                a /= a.sum()
                eta = 0.5
                T = 5.0 / max(score_of(best) / (2*n), 1e-10)
                stuck = 0
                mode = 'smooth'
        
        a = a_new if sc < best_sc else a  # Keep best as reference
        # Actually, let me always move to a_new for exploration, but track best separately
        a = a_new
        it += 1
    
    return best, best_sc
