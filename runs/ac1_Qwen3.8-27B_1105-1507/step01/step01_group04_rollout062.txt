"""
Optimizes non-negative step function heights to minimize 2n*max(conv(f,f))/(sum(f))^2.
Uses Nesterov-accelerated gradient descent (FISTA) on a log-sum-exp smoothed
surrogate with adaptive sigma (homotopy), combined with LP-based Frank-Wolfe
refinement for polish.
"""

import numpy as np
import time

# --- Helper functions ---

def _normalize(f):
    """Normalize f to sum to 1, clip negatives. Returns None if degenerate."""
    f = np.maximum(f, 0.0)
    s = float(np.sum(f))
    if s < 1e-30:
        return None
    return f / s

def _project_simplex(v):
    """Project v onto the probability simplex {f: f>=0, sum(f)=1}."""
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n+1) > (cssv - 1.0))[0]
    if len(rho) == 0:
        return np.ones(n) / n
    hat = rho[-1]
    theta = (cssv[hat] - 1.0) / (hat + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w

def _fft_convolve_full(f):
    """Compute full linear convolution of f with itself using FFT."""
    n = len(f)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    F = np.fft.rfft(f, size)
    h = np.fft.irfft(F * F, size)
    return h[:2 * n - 1]

def _score_raw(f):
    """Fast score computation assuming f >= 0, sum(f) > 0.01."""
    n = len(f)
    h = _fft_convolve_full(f)
    M = float(np.max(h))
    S = float(np.sum(f))
    if S < 0.01:
        return float('inf')
    return 2.0 * n * M / (S * S)

def _lse_value(h, sigma):
    """Compute log-sum-exp(h/sigma) * sigma."""
    h_max = np.max(h)
    if h_max * sigma > 700:  # overflow protection
        return float('inf')
    return sigma * (h_max + np.log(np.sum(np.exp((h - h_max) / sigma))))

def _lse_gradient(f, p, n):
    """
    Compute gradient of LSE(conv(f,f), sigma) w.r.t. f.
    p: softmax weights (length 2n-1)
    Returns grad of length n.
    """
    # grad_m = 2 * sum_k p_k * f_{k-m} for 0 <= k-m < n
    # = 2 * (f_rev * p)[m + n - 1] where f_rev = f[::-1]
    size = 1
    while size < 3 * n:
        size <<= 1
    F = np.fft.rfft(f[::-1], size)
    P = np.fft.rfft(p, size)
    conv = np.fft.irfft(F * P, size)
    grad = 2.0 * conv[n-1:2*n-1]
    return grad

def _lse_fista(f_init, sigma_schedule, iters_per_phase, time_limit, verbose=True):
    """
    FISTA on LSE smoothed objective with homotopy (decreasing sigma).
    f_init: initial point on the simplex (sum=1, f>=0)
    sigma_schedule: list of sigma values (decreasing)
    iters_per_phase: iterations per sigma value
    """
    f = np.asarray(f_init, dtype=np.float64).copy()
    f = _normalize(f)
    if f is None:
        return f_init.copy(), float('inf')
    n = len(f)
    best_f = f.copy()
    best_score = _score_raw(f)
    start_time = time.monotonic()
    
    # FISTA variables
    x = f.copy()
    y = f.copy()
    t = 1.0
    
    for sigma_idx, sigma in enumerate(sigma_schedule):
        if time.monotonic() - start_time > time_limit:
            break
        
        for it in range(iters_per_phase):
            if time.monotonic() - start_time > time_limit:
                break
            
            # Compute h and p
            h = _fft_convolve_full(y)
            h_shifted = h - np.max(h)
            p = np.exp(h_shifted / sigma)
            p /= np.sum(p)
            
            # Compute gradient
            grad = _lse_gradient(y, p, n)
            
            # Lipschitz constant estimate (conservative)
            L = 3.0  # safe upper bound
            
            # Gradient step
            z = y - (1.0 / L) * grad
            x_new = _project_simplex(z)
            
            # Check improvement
            score = _score_raw(x_new)
            if score < best_score:
                best_score = score
                best_f = x_new.copy()
            
            # FISTA update
            t_new = (1.0 + np.sqrt(1.0 + 4.0 * t * t)) / 2.0
            y = x_new + ((t - 1.0) / t_new) * (x_new - x)
            # Clip y to be non-negative (helps with numerical stability)
            y = np.maximum(y, 0.0)
            y = y / np.sum(y)  # re-normalize for stability
            
            x = x_new
            t = t_new
            
            if verbose and (it + 1) % 5000 == 0:
                print(f"      sigma={sigma:.4f}, it={it+1}, score={best_score:.6f}")
        
        if verbose:
            print(f"    Phase sigma={sigma:.4f}: best={best_score:.6f}")
    
    return best_f, best_score

def _lp_frank_wolfe(f_init, max_iters, time_budget):
    """LP-based Frank-Wolfe for the normalized problem."""
    from scipy.optimize import linprog
    from scipy.sparse import coo_matrix
    
    f = np.asarray(f_init, dtype=np.float64).copy()
    f = _normalize(f)
    if f is None:
        return f_init.copy(), float('inf')
    n = len(f)
    best_f = f.copy()
    best_score = _score_raw(f)
    start_time = time.monotonic()
    
    for it in range(max_iters):
        if time.monotonic() - start_time > time_budget:
            break
        
        h = np.convolve(f, f)
        M = float(np.max(h))
        
        # Build sparse constraint matrix
        row_idx = []
        col_idx = []
        vals = []
        for j in range(n):
            rows_j = np.arange(j, j + n)
            row_idx.extend(rows_j.tolist())
            col_idx.extend([j] * n)
            vals.extend(f.tolist())
        
        C = coo_matrix((vals, (row_idx, col_idx)), shape=(2 * n - 1, n))
        c_obj = -np.ones(n)
        res = linprog(c_obj, A_ub=C, b_ub=np.full(2 * n - 1, M),
                      bounds=[(0, None)] * n, method='highs')
        
        if not res.success or res.x is None:
            break
        
        g = res.x
        Sg = float(np.sum(g))
        if Sg < 1e-10:
            break
        g = g / Sg
        
        t = 2.0 / (it + 2)
        f_new = (1.0 - t) * f + t * g
        f_n = _normalize(f_new)
        if f_n is None:
            break
        f = f_n
        
        score = _score_raw(f)
        if score < best_score:
            best_score = score
            best_f = f.copy()
    
    return best_f, best_score

def _cosine_init(n):
    j = np.arange(n, dtype=np.float64)
    f = np.cos(np.pi * (j + 0.5) / float(n)) ** 2
    f = np.maximum(f, 0.0)
    s = np.sum(f)
    if s > 0:
        f /= s
    return f

def _gaussian_init(n, sigma_frac=0.3):
    j = np.arange(n, dtype=np.float64) - (n - 1) / 2.0
    sigma = sigma_frac * float(n)
    f = np.exp(-j * j / (2.0 * sigma * sigma))
    s = np.sum(f)
    if s > 0:
        f /= s
    return f

def _tapered_init(n, power=2.0):
    j = np.arange(n, dtype=np.float64)
    x = np.sin(np.pi * (j + 0.5) / float(n))
    f = x ** power
    f = np.maximum(f, 0.0)
    s = np.sum(f)
    if s > 0:
        f /= s
    return f

def _random_init(n, seed):
    rng = np.random.default_rng(seed)
    f = rng.random(n)
    f = _normalize(f)
    return f

def _additive_perturb(f, rng, frac):
    n = len(f)
    noise = rng.random(n) * frac
    f_p = f + noise
    return _normalize(f_p)

"""
Optimizes non-negative step function heights to minimize 2n*max(conv(f,f))/(sum(f))^2.
Uses Nesterov-accelerated gradient descent (FISTA) on a log-sum-exp smoothed surrogate
with adaptive sigma (homotopy), spectral initialization, and LP-based Frank-Wolfe polish.
"""

import numpy as np
import time

# ===== Helper Functions =====

def _normalize(f):
    f = np.maximum(f, 0.0)
    s = float(np.sum(f))
    if s < 1e-30:
        return None
    return f / s

def _project_simplex(v):
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - 1.0))[0]
    if len(rho) == 0:
        return np.ones(n) / n
    hat = rho[-1]
    theta = (cssv[hat] - 1.0) / (hat + 1.0)
    return np.maximum(v - theta, 0.0)

def _fft_convolve_full(f):
    n = len(f)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    F = np.fft.rfft(f, size)
    h = np.fft.irfft(F * F, size)
    return h[:2 * n - 1]

def _score_raw(f):
    n = len(f)
    h = _fft_convolve_full(f)
    M = float(np.max(h))
    S = float(np.sum(f))
    if S < 0.01:
        return float('inf')
    return 2.0 * n * M / (S * S)

def _lse_gradient(f, p, n):
    size = 1
    while size < 3 * n:
        size <<= 1
    F = np.fft.rfft(f[::-1], size)
    P = np.fft.rfft(p, size)
    conv = np.fft.irfft(F * P, size)
    return 2.0 * conv[n - 1:2 * n - 1]

def _lse_fista(f_init, sigmas, time_limit, verbose=True):
    f = np.asarray(f_init, dtype=np.float64).copy()
    f = _normalize(f)
    if f is None:
        return f_init.copy(), float('inf')
    n = len(f)
    best_f = f.copy()
    best_score = _score_raw(f)
    start_time = time.monotonic()

    x = f.copy()
    y = f.copy()
    t = 1.0

    for sigma in sigmas:
        if time.monotonic() - start_time > time_limit:
            break
        phase_start = time.monotonic()
        L = 2.5

        while time.monotonic() - phase_start < time_limit and time.monotonic() - start_time < time_limit:
            h = _fft_convolve_full(y)
            h_max = float(np.max(h))
            h_shifted = h - h_max
            p = np.exp(h_shifted / sigma)
            p_sum = float(np.sum(p))
            if p_sum < 1e-300:
                break
            p /= p_sum

            grad = _lse_gradient(y, p, n)

            z = y - (1.0 / L) * grad
            x_new = _project_simplex(z)

            score = _score_raw(x_new)
            if score < best_score:
                best_score = score
                best_f = x_new.copy()

            t_new = (1.0 + np.sqrt(1.0 + 4.0 * t * t)) / 2.0
            y = x_new + ((t - 1.0) / t_new) * (x_new - x)
            y = np.maximum(y, 0.0)
            y_sum = float(np.sum(y))
            if y_sum < 1e-30:
                y = x_new.copy()
            else:
                y = y / y_sum
            x = x_new
            t = t_new

            if t > 1e6:
                break

        if verbose:
            print(f"      sigma={sigma:.5f}: best={best_score:.8f}")

    return best_f, best_score

def _subgrad_fista(f_init, time_limit, verbose=True):
    f = np.asarray(f_init, dtype=np.float64).copy()
    f = _normalize(f)
    if f is None:
        return f_init.copy(), float('inf')
    n = len(f)
    best_f = f.copy()
    best_score = _score_raw(f)
    start_time = time.monotonic()

    x = f.copy()
    y = f.copy()
    t = 1.0
    L = 2.0

    while time.monotonic() - start_time < time_limit:
        h = _fft_convolve_full(y)
        M = float(np.max(h))
        if M <= 0:
            break

        threshold = 0.85 * M
        active = np.where(h >= threshold)[0]
        if len(active) == 0:
            active = np.array([int(np.argmax(h))])

        # Subgradient: g_j = (2/|K|) * sum_{k in K} y_{k-j}
        grad = np.zeros(n)
        for k in active:
            j_start = max(0, k - n + 1)
            j_end = min(n - 1, k)
            if j_end >= j_start:
                lo = k - j_end
                hi = k - j_start
                grad[j_start:j_end + 1] += 2.0 * y[lo:hi + 1][::-1]
        grad /= float(len(active))

        eta = 2.0 / (L * (t + 1.0)) * n
        z = y - eta * grad
        x_new = _project_simplex(z)

        score = _score_raw(x_new)
        if score < best_score:
            best_score = score
            best_f = x_new.copy()

        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t * t)) / 2.0
        y = x_new + ((t - 1.0) / t_new) * (x_new - x)
        y = np.maximum(y, 0.0)
        y_sum = float(np.sum(y))
        if y_sum < 1e-30:
            y = x_new.copy()
        else:
            y = y / y_sum
        x = x_new
        t = t_new

        if t > 1e6:
            break

        if verbose and int(t) % 100000 < 100:
            print(f"      subgrad t={int(t)}: best={best_score:.8f}")

    return best_f, best_score

def _lp_frank_wolfe(f_init, max_iters, time_budget):
    try:
        from scipy.optimize import linprog
        from scipy.sparse import coo_matrix
    except ImportError:
        return f_init.copy(), _score_raw(f_init)

    f = np.asarray(f_init, dtype=np.float64).copy()
    f = _normalize(f)
    if f is None:
        return f_init.copy(), float('inf')
    n = len(f)
    best_f = f.copy()
    best_score = _score_raw(f)
    start_time = time.monotonic()

    for it in range(max_iters):
        if time.monotonic() - start_time > time_budget:
            break

        h = np.convolve(f, f)
        M = float(np.max(h))

        row_idx = []
        col_idx = []
        vals = []
        for j in range(n):
            rows_j = np.arange(j, j + n)
            row_idx.extend(rows_j.tolist())
            col_idx.extend([j] * n)
            vals.extend(f.tolist())

        C = coo_matrix((vals, (row_idx, col_idx)), shape=(2 * n - 1, n))
        c_obj = -np.ones(n)
        res = linprog(c_obj, A_ub=C, b_ub=np.full(2 * n - 1, M),
                      bounds=[(0, None)] * n, method='highs',
                      options={'time_limit': max(1, time_budget - (time.monotonic() - start_time) - 2)})

        if not res.success or res.x is None:
            break

        g = np.asarray(res.x, dtype=np.float64)
        g = np.maximum(g, 0.0)
        Sg = float(np.sum(g))
        if Sg < 1e-10:
            break
        g = g / Sg

        mu = 2.0 / (it + 2)
        f_new = (1.0 - mu) * f + mu * g
        f_n = _normalize(f_new)
        if f_n is None:
            break
        f = f_n

        score = _score_raw(f)
        if score < best_score:
            best_score = score
            best_f = f.copy()

    return best_f, best_score

def _cosine_init(n):
    j = np.arange(n, dtype=np.float64)
    f = np.cos(np.pi * (j + 0.5) / float(n)) ** 2
    f = np.maximum(f, 0.0)
    s = np.sum(f)
    if s > 0:
        f /= s
    return f

def _gaussian_init(n, sigma_frac=0.3):
    j = np.arange(n, dtype=np.float64) - (n - 1) / 2.0
    sigma = sigma_frac * float(n)
    f = np.exp(-j * j / (2.0 * sigma * sigma))
    s = np.sum(f)
    if s > 0:
        f /= s
    return f

def _tapered_init(n, power=2.0):
    j = np.arange(n, dtype=np.float64)
    x = np.sin(np.pi * (j + 0.5) / float(n))
    f = x ** power
    f = np.maximum(f, 0.0)
    s = np.sum(f)
    if s > 0:
        f /= s
    return f

def _spectral_init(n, rng):
    """Initialize via flat spectrum approach."""
    # Target: flat autocorrelation -> flat |F|^2
    # Start with random phase, take inverse FFT
    for _ in range(50):
        phase = rng.random(2 * n - 1) * 2 * np.pi
        # Create a "flat-ish" power spectrum
        N = 1
        while N < 2 * n - 1:
            N <<= 1
        H = np.ones(N)
        # Add some structure: slight peak at DC
        H[:n] = 1.0 + 0.5 * np.exp(-np.arange(n) / (n * 0.2))
        H[N - n + 1:] = 1.0 + 0.5 * np.exp(-np.arange(n - 1, 0, -1) / (n * 0.2))
        F_f = H * np.exp(1j * phase[:N])
        # Enforce Hermitian symmetry for real f
        F_f[:N//2 + 1] = (F_f[:N//2 + 1] + np.conj(F_f[N//2 + 1 - (N//2 + 1):][::-1])) / 2 if N % 2 == 0 else F_f[:N//2 + 1]
        f = np.fft.irfft(F_f, N)[:n]
        f = np.maximum(f, 0.0)
        s = np.sum(f)
        if s > 0.5 * n * 0.1:
            f /= s
            break
    return _normalize(f) if _normalize(f) is not None else _cosine_init(n)

def _random_init(n, seed):
    rng = np.random.default_rng(seed)
    f = rng.random(n)
    return _normalize(f)

def _asymmetric_init(n, rng):
    """Create an asymmetric initialization to break symmetry."""
    half = n // 2
    f = np.zeros(n)
    f[:half] = rng.random(half) ** 0.5
    f[half:] = rng.random(n - half) ** 1.5 * 0.3
    return _normalize(f)

            best_f = f_ref.copy()
            print(f"  ** Refinement new best: {best_score:.8f} **")

        # Subgrad on refined
        if time.monotonic() < deadline - 20 and s_ref < 1.55:
            t_rem2 = min(60.0, deadline - time.monotonic() - 15.0)
            f_sg2, s_sg2 = _subgrad_fista(f_ref, t_rem2, verbose=False)
            if s_sg2 < best_score:
                best_score = s_sg2
                best_f = f_sg2.copy()
                print(f"  ** Subgrad2 new best: {best_score:.8f} **")

    # Phase 4: Perturbation restarts on best
    if time.monotonic() < deadline - 30:
        for frac in [0.005, 0.02, 0.05]:
            if time.monotonic() > deadline - 20:
                break
            f_pert = _normalize(best_f + rng.random(len(best_f)) * frac)
            if f_pert is None:
                continue
            n_p = len(f_pert)
            sigma_max = max(0.005, 0.5 / n_p)
            sigmas_p = [sigma_max * (0.5 ** i) for i in range(6)]
            sigmas_p = [s for s in sigmas_p if s > 1e-6]
            t_rem = min(60.0, deadline - time.monotonic() - 20.0)
            if t_rem < 5:
                break
            f_p, s_p = _lse_fista(f_pert, sigmas_p, t_rem, verbose=False)
            if s_p < best_score:
                best_score = s_p
                best_f = f_p.copy()
                print(f"  ** Perturb({frac}) new best: {best_score:.8f} **")

    # Final verification
    print(f"\n{'='*50}")
    print(f"Best found: score={best_score:.8f}, n={len(best_f)}")
    final_list = [float(x) for x in best_f]
    final_score = float(evaluate_sequence(final_list))
    print(f"Verified score: {final_score:.8f}")

    if not np.isfinite(final_score) or final_score > 2.0:
        print("  Falling back to parent.")
        return [float(x) for x in parent]

    # If our best is worse than parent, return parent
    if final_score > float(evaluate_sequence([float(x) for x in parent])):
        print("  Returning parent (better than optimized).")
        return [float(x) for x in parent]

    return final_list
