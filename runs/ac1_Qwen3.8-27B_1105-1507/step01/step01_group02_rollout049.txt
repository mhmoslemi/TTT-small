Optimizes non-negative sequences to minimize 2*n*max(conv(a,a))/sum(a)^2.
Uses log-sum-exp smoothed objective with Nesterov accelerated projected gradient,
temperature annealing, alternating projections initialization, and LP polish.
"""

import numpy as np
import time
from scipy.optimize import linprog


def project_simplex(v):
    """Project v onto {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cv = np.cumsum(u) - 1.0
    rho = np.where(u * np.arange(1, n + 1) > cv)[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho = rho[-1]
    theta = cv[rho] / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def fast_convolve(a):
    """Full linear convolution of a with a using FFT."""
    n = len(a)
    size = 2 * n - 1
    N = 1 << (size - 1).bit_length()
    fa = np.fft.rfft(a, N)
    result = np.fft.irfft(fa * fa, N)[:size]
    result = np.maximum(result, 0.0)
    return result


def smooth_max_and_grad(a, c, tau):
    """
    Given c = conv(a,a), compute the smooth approximation to max(c)
    and its gradient w.r.t. a.
    Returns (value, grad).
    """
    n = len(a)
    c_max = np.max(c)
    shifted = (c - c_max) / tau
    # Clip for numerical stability
    shifted = np.clip(shifted, -50, 50)
    exp_shifted = np.exp(shifted)
    sum_exp = np.sum(exp_shifted)
    w = exp_shifted / sum_exp  # weights (softmax)
    
    value = c_max + tau * np.log(sum_exp)
    
    # grad[i] = 2 * sum_j w[i+j] * a[j]
    # = 2 * conv(a, w_rev)[2n-2-i]
    w_rev = w[::-1]
    # Compute conv(a, w_rev) via FFT
    size = n + len(w_rev) - 1  # = n + (2n-1) - 1 = 3n-2
    N = 1 << (size - 1).bit_length()
    fa = np.fft.rfft(a, N)
    fwr = np.fft.rfft(w_rev, N)
    conv_awr = np.fft.irfft(fa * fwr, N)[:size]
    
    # grad[i] = 2 * conv_awr[2n-2-i] for i = 0,...,n-1
    # conv_awr has length 3n-2, indices 0 to 3n-3
    # We need indices 2n-2, 2n-3, ..., n-1 (i.e., from 2n-2 down to n-1)
    grad = 2.0 * conv_awr[n-1:2*n-1][::-1]
    
    return value, grad


def alt_projection_init(n, rng, n_iter=100, target='flat_conv'):
    """
    Alternating projections to initialize a sequence.
    Target: flat magnitude spectrum (for 'flat_conv' mode) or 
    flat convolution (for 'flat_conv' mode).
    """
    N = 1 << (2 * n - 2).bit_length()
    
    if target == 'flat_conv':
        # We want conv(a,a) to be flat, which means A(f)^2 should be flat
        # (in the sense of having a flat inverse transform).
        # A good proxy: make |A(f)| flat AND make the phase of A(f) such that
        # the inverse of A(f)^2 is flat.
        # Simplest: make |A(f)| flat (random phases).
        a = np.ones(n) / n
        for it in range(n_iter):
            # Impose flat spectrum
            A = np.fft.rfft(a, N)
            mag = np.ones(len(A))
            # Slight high-frequency boost to help nonnegativity
            mag = 1.0 + 0.1 * np.linspace(0, 1, len(mag))
            phase = np.angle(A)
            A = mag * np.exp(1j * phase)
            A[0] = A[0].real + 0j  # Keep DC real and positive
            a = np.fft.irfft(A, N)[:n]
            # Project to non-negative simplex
            a = np.maximum(a, 0.0)
            s = a.sum()
            if s < 1e-10:
                a = np.ones(n) / n
            else:
                a = a / s
    elif target == 'random':
        a = rng.random(n)
        a = a / a.sum()
    elif target == 'cosine':
        x = np.linspace(0, 1, n)
        a = 0.5 * (1 + np.cos(2 * np.pi * x)) + 0.01
        a = a / a.sum()
    elif target == 'sinc':
        x = np.linspace(-1, 1, n)
        a = np.sinc(x) ** 2 + 0.01
        a = a / a.sum()
    elif target == 'gaussian':
        x = np.linspace(-1, 1, n)
        a = np.exp(-2 * x**2) + 0.01
        a = a / a.sum()
    
    return a
