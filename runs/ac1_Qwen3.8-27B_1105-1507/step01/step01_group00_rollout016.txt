"""
Multi-strategy optimization for minimizing 2n*max(conv(a,a))/sum(a)^2.
Uses LP-based direction search, subgradient descent with averaging,
Gerchberg-Saxton initialization, and backtracking line search.
"""
import numpy as np
import time
from scipy.optimize import linprog
from scipy.sparse import csr_matrix


def build_conv_matrix_sparse(f):
    """Build sparse matrix A where (A@g)[k] = convolve(f,g)[k]."""
    n = len(f)
    rows = np.empty(n * n, dtype=np.int32)
    cols = np.empty(n * n, dtype=np.int32)
    vals = np.empty(n * n, dtype=np.float64)
    idx = 0
    for k in range(2 * n - 1):
        j_min = max(0, k - (n - 1))
        j_max = min(k, n - 1)
        for j in range(j_min, j_max + 1):
            rows[idx] = k
            cols[idx] = j
            vals[idx] = f[k - j]
            idx += 1
    rows = rows[:idx]
    cols = cols[:idx]
    vals = vals[:idx]
    return csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))


def solve_lp_direction(f, M, f_prev=None, reg=0.0):
    """Solve: max sum(g) - reg*||g - f_prev||^2 s.t. g >= 0, convolve(f,g)_k <= M."""

    return None

def subgradient_descent(f, deadline, verbose_prefix=""):
    """Subgradient descent with Polyak step size and iterate averaging."""
    f = f.copy().astype(np.float64)
    best_f = f.copy()
    best_score = evaluate_sequence(f.tolist())
    n = len(f)
    S = float(np.sum(f))
    
    # Polyak step: know lower bound on objective
    # Lower bound: 2n/(2n-1) (from average of conv values)
    lower_bound = 2.0 * n / (2 * n - 1)
    
    average_f = f.copy()
    count = 0
    step = 0.1
    no_improve = 0

    for it in range(20000):
        if time.monotonic() > deadline:
            break

        conv_f = np.convolve(f, f)
        k_star = int(np.argmax(conv_f))
        M = conv_f[k_star]
        S = float(np.sum(f))
        current_obj = 2.0 * n * M / (S * S)

        # Subgradient of max conv at f: 2*a[k*-j] for valid j
        grad = np.zeros(n)
        for j in range(n):
            k_minus_j = k_star - j
            if 0 <= k_minus_j < n:
                grad[j] = 2.0 * f[k_minus_j]

        # Gradient of the full objective f(a) = 2n*M/S^2
        # df/da[j] = 2n*(grad[j]*S - 2*M)/S^3
        obj_grad = (2.0 * n / (S * S * S)) * (grad * S - 2.0 * M)
        
        # Polyak step size
        polyak_num = current_obj - lower_bound
        grad_sq = float(np.sum(obj_grad * obj_grad))
        if grad_sq > 1e-20:
            alpha = polyak_num / grad_sq
        else:
            alpha = step
        alpha = min(alpha, 1.0)  # Cap step size

        # Update
        new_f = f - alpha * obj_grad
        new_f = np.clip(new_f, 0.0, 1000.0)
        
        # Renormalize to keep sum constant
        new_sum = float(np.sum(new_f))
        if new_sum > 0.01:
            new_f = new_f * (S / new_sum)
        else:
            new_f = f  # Revert

        count += 1
        average_f = average_f + (new_f - average_f) / count

        new_score = evaluate_sequence(new_f.tolist())
        if np.isfinite(new_score) and new_score < best_score:
            f = new_f
            best_f = f.copy()
            best_score = new_score
            no_improve = 0
        else:
            f = new_f  # Continue anyway (Polyak method doesn't require monotone decrease)
            no_improve += 1

        # Also check the average
        avg_score = evaluate_sequence(average_f.tolist())
        if np.isfinite(avg_score) and avg_score < best_score:
            best_f = average_f.copy()
            best_score = avg_score

        if (it + 1) % 500 == 0:
            print(f"    {verbose_prefix}subgrad iter {it+1}: score={best_score:.6f}, obj={current_obj:.6f}")

        if no_improve > 2000:
            break

    return best_f, best_score


def gerchberg_saxton_init(n, iterations=100, seed=42):
    """Initialize using Gerchberg-Saxton for flat autocorrelation."""
    rng = np.random.default_rng(seed)
    a = rng.uniform(0.5, 1.5, n).astype(np.float64)
    
    for _ in range(iterations):
        # Forward transform
        A = np.fft.rfft(a)
        # Flatten the magnitude
        mag = np.abs(A)
        mag_target = np.full_like(mag, float(np.mean(mag)))
        # Update
        A_new = A * (mag_target / np.maximum(mag, 1e-10))
        # Inverse transform
        a_new = np.fft.irfft(A_new, n=n)
        # Project to non-negative
        a = np.maximum(a_new, 0.0)
        # Renormalize
        s = float(np.sum(a))
        if s > 0:
            a = a / s * np.sqrt(2 * n)  # Keep reasonable scale

    return a


def random_perturbation(f, scale=0.1):
    """Add random perturbation to f."""
    n = len(f)
    S = float(np.sum(f))
    rng = np.random.default_rng()
    perturbation = rng.normal(0, scale * S / n, n)
    new_f = f + perturbation
    new_f = np.clip(new_f, 0.0, 1000.0)
    new_sum = float(np.sum(new_f))
    if new_sum > 0.01:
        new_f = new_f * (S / new_sum)
    return new_f

- We need max_conv / S² = 1.5136/(2*512) = 1.5136/1024 ≈ 0.001478
- If S is normalized to 1: max_conv ≈ 0.001478
- Average conv = 1/1023 ≈ 0.000977
- So max_conv is about 1.51 times the average.

This means the convolution is NOT very flat - the max is 51% above average. For a truly flat convolution, the score would be 2n/(2n-1) ≈ 1.001.

The gap between 1.001 and 1.5136 is due to the non-negativity constraint. With non-negative entries, the convolution has a specific structure that prevents it from being perfectly flat.

Key constraint: conv[0] = a[0]² and conv[1] = 2*a[0]*a[1]. If a[0] is large, conv[0] is large. If a[0] is small, then a[1] can be relatively large (since conv[1] = 2*a[0]*a[1] is small). But then conv[2] = a[0]*a[2] + 2*a[1]² involves a[1]², which might be large.

This creates a "cascading" effect: to keep edge convolutions small, edge values must be small, but then the next values must compensate, leading to a specific optimal profile.

I suspect the optimal profile is something like a[0] < a[1] < a[2] < ... < a[n/2] > ... > a[2] > a[1] > a[0], i.e., an inverted triangle or bell shape. Let me verify with a small example.

For n=3, a = [a, b, a] (symmetric):

Case 1: 2ab >= 2a²+b², i.e., 2ab - 2a² - b² >= 0, i.e., (b-a)² <= a², i.e., 0 <= b <= 2a (and b-a is real, which it is).
Wait: 2ab >= 2a²+b² ⟺ 0 >= 2a²-2ab+b² = (a-b)² + a². This can only be 0 if a=b=0. So this case never happens (for positive a,b).

Score = 6(2a²+b²)/(2a+b)².

Minimize over b/a. Let t = b/a:
Score = 6(2+t²)/(2+t)²
d/dt = 6[2t(2+t)² - (2+t²)2(2+t)]/(2+t)⁴ = 12(2+t)[t(2+t) - (2+t²)]/(2+t)⁴ = 12(t²+2t-2-t²)/(2+t)³ = 24(t-1)/(2+t)³... 

Wait: t(2+t) - (2+t²) = 2t+t²-2-t² = 2t-2.
So d/dt = 12(2+t)(2t-2)/(2+t)⁴ = 24(t-1)/(2+t)³.

Zero at t=1, i.e., b=a. Score = 6(2+1)/(2+1)² = 18/9 = 2.

So for n=3, the minimum (among symmetric sequences) is 2, achieved at a=b (uniform). The non-symmetric case might do better.
