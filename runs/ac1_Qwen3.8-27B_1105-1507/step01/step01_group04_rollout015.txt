"""

import numpy as np
import time
from scipy.optimize import linprog

def _score_fast(f):
    """Compute score for numpy array f (non-negative, sum > 0.01)."""
    n = len(f)
    conv = np.convolve(f, f)
    M = float(np.max(conv))
    S = float(np.sum(f))
    if S < 0.01:
        return float('inf')
    return 2.0 * n * M / (S * S)

def _normalize(f):
    """Normalize f to sum to 1, clip negatives. Returns None if degenerate."""
    f = np.maximum(f, 0.0)
    s = float(np.sum(f))
    if s < 1e-30:
        return None
    return f / s

def _compute_pressure(f, active_idx, weights=None):
    """
    Compute gradient of weighted sum of active convolution terms.
    d conv[k]/df[j] = 2*f[k-j] if 0 <= k-j < n, else 0.
    Returns pressure vector p of length n.
    """
    n = len(f)
    p = np.zeros(n, dtype=np.float64)
    if weights is None:
        weights = np.ones(len(active_idx)) / len(active_idx)
    for i, k in enumerate(active_idx):
        w = weights[i]
        if w < 1e-15:
            continue
        # For each j in [max(0, k-n+1), min(n-1, k)]:
        #   p[j] += 2 * w * f[k-j]
        jstart = max(0, k - n + 1)
        jend = min(n - 1, k)
        if jend >= jstart:
            # f[k-j] for j in [jstart, jend] means f indices [k-jend, k-jstart]
            lo = k - jend
            hi = k - jstart
            p[jstart:jend+1] += 2.0 * w * f[lo:hi+1][::-1]
    return p

    n = len(f)

    best_score = _score_fast(f)

        M = float(np.max(conv))
        if M <= 0:

        # Progressive threshold: start wide, narrow down
        progress = min(1.0, it / (max_iters * 0.6))
        threshold_frac = 0.4 + 0.55 * progress  # 0.4 to 0.95

    by minimizing the max convolution. Uses ternary search on [0, current*3].
    """
    f = _normalize(np.asarray(f_init, dtype=np.float64).copy())
    if f is None:
        return np.asarray(f_init, dtype=np.float64).copy(), float('inf')
    
    n = len(f)
    best_f = f.copy()
    best_score = _score_fast(f)
    
    if label:
        print(f"    CD start: n={n}, score={best_score:.6f}")
    
    for round_idx in range(max_rounds):
        improved = False
        for j in range(n):
            # Try a range of values for f[j]
            lo = 0.0
            hi = f[j] * 4.0 + 1.0 / n
            if hi < 1e-10:
                hi = 1.0 / n
            
            # Ternary search
            for _ in range(15):
                m1 = lo + (hi - lo) / 3.0
                m2 = hi - (hi - lo) / 3.0
                
                f[j] = m1
                f = _normalize(f)
                s1 = _score_fast(f)
                
                f[j] = m2
                f = _normalize(f)
                s2 = _score_fast(f)
                
                if s1 < s2:
                    hi = m2
                else:
                    lo = m1
            
            f[j] = (lo + hi) / 2.0
            f = _normalize(f)
            score = _score_fast(f)
            
            if score < best_score:
                best_score = score
                best_f = f.copy()
                improved = True
        
        if not improved and round_idx > 3:
            break
    
    return best_f, best_score

def _lp_refine(f_init, max_iters=30, time_budget=60.0):
    """LP-based Frank-Wolfe refinement."""
    from scipy.sparse import coo_matrix
    
    f = _normalize(np.asarray(f_init, dtype=np.float64).copy())
    if f is None:
        return np.asarray(f_init, dtype=np.float64).copy(), float('inf')
    
    n = len(f)
    best_f = f.copy()
    best_score = _score_fast(f)
    start_time = time.monotonic()
    
    print(f"    LP start: n={n}, score={best_score:.6f}")
    
    for it in range(max_iters):
        if time.monotonic() - start_time > time_budget:
            break
        
        conv = np.convolve(f, f)
        M = float(np.max(conv))
        
        # Build constraint matrix: conv[k] <= M for all k
        # conv[k] = sum_j f[j] * f[k-j]
        # This is bilinear, not linear! LP won't work directly.
        # Instead, use the linear approximation: fix f and optimize g
        # s.t. f*g[k] <= M, sum(g)=1, g>=0 (Frank-Wolfe)
        
        # C[k, j] = f[k-j] if 0<=k-j<n else 0
        row_idx = []
        col_idx = []
        vals = []
        for j in range(n):
            rows_j = np.arange(j, j + n)
            row_idx.extend(rows_j.tolist())
            col_idx.extend([j] * n)
            vals.extend(f.tolist())
        
        C = coo_matrix((vals, (row_idx, col_idx)), shape=(2 * n - 1, n))
        
        c_obj = -np.ones(n)
        res = linprog(c_obj, A_ub=C, b_ub=np.full(2 * n - 1, M),
                      bounds=[(0, None)] * n, method='highs')
        
        if not res.success or res.x is None:
            break
        
        g = res.x
        Sg = float(np.sum(g))
        if Sg < 1e-10:
            break
        g = g / Sg
        
        t = 2.0 / (it + 2)
        f_new = (1.0 - t) * f + t * g
        f_n = _normalize(f_new)
        if f_n is None:
            break
        f = f_n
        
        score = _score_fast(f)
        if score < best_score:
            best_score = score
            best_f = f.copy()
        
        if (it + 1) % 5 == 0:
            print(f"    LP it={it+1}: score={best_score:.6f}")
    
    return best_f, best_score

def _cosine_init(n, power=2.0):
    """Cosine-tapered initialization."""
    j = np.arange(n, dtype=np.float64)
    x = np.cos(np.pi * (j + 0.5) / float(n))
    f = np.abs(x) ** power
    s = np.sum(f)
    if s > 0:
        f /= s
    return f

def _gaussian_init(n, sigma_frac=0.3):
    """Gaussian initialization centered."""
    j = np.arange(n, dtype=np.float64) - (n - 1) / 2.0
    sigma = sigma_frac * float(n)
    f = np.exp(-j * j / (2.0 * sigma * sigma))
    s = np.sum(f)
    if s > 0:
        f /= s
    return f

def _uniform_init(n):
    return np.ones(n, dtype=np.float64) / n

def _bilinear_init(n, rng):
    """Initialize as a smooth bilinear-interpolated random signal."""
    m = max(4, n // 32)
    coarse = rng.random(m)
    coarse = (coarse + coarse[::-1]) / 2.0  # make symmetric
    f = np.interp(np.linspace(0, m - 1, n), np.arange(m), coarse)
    f = np.maximum(f, 0.01)
    return _normalize(f)

def _spectral_flat_init(n, rng, num_iters=50):
    """
    Initialize by trying to make the power spectrum flat.
    Start with random phase, take real part, clip to non-negative.
    """
    f = np.ones(n) / n
    for _ in range(num_iters):
        # Take FFT, randomize phase, come back
        F = np.fft.rfft(f)
        angles = rng.random(len(F)) * 2 * np.pi
        F_new = np.abs(F) * np.exp(1j * angles)
        F_new[0] = np.abs(F_new[0])  # keep DC real and positive
        f_new = np.fft.irfft(F_new, n)
        f_new = np.maximum(f_new, 0.0)
        s = f_new.sum()
        if s < 1e-10:
            break
        f_new /= s
        # Check if it helps
        if _score_fast(f_new) < _score_fast(f):
            f = f_new
    return f

def propose_candidate(seed=42, budget_s=1500, **kwargs):
    """
    Multi-stage search for optimal non-negative step function.
    Stage 1: Multi-start mirror descent at various n values.
    Stage 2: LP Frank-Wolfe refinement on top candidates.
    Stage 3: Coordinate descent polishing.
    Stage 4: Final adaptive mirror descent.
    """
    deadline = time.monotonic() + float(budget_s) - 5.0
    rng = np.random.default_rng(seed)
    
    # Get parent sequence
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64).copy()
        parent = np.clip(parent, 0.0, 1000.0)
    except Exception:
        parent = np.ones(256, dtype=np.float64)
    
    parent_norm = _normalize(parent)
    best_f = parent.copy()
    best_score = float(evaluate_sequence(best_f.tolist()))
    print(f"Parent: n={len(best_f)}, score={best_score:.6f}")
    
    all_results = []  # (score, f, name)
    
    # Stage 1: Multi-start mirror descent
    print("\n=== Stage 1: Multi-start Mirror Descent ===")
    
    # Generate diverse initializations
    init_list = []
    
    # Parent
    if parent_norm is not None:
        init_list.append(("parent", parent_norm))
    
    # Various n values with different initializations
    n_values = [256, 384, 512, 640, 768, 1024]
    
    for n in n_values:
        init_list.append((f"cos_{n}", _cosine_init(n, 2.0)))
        init_list.append((f"cos3_{n}", _cosine_init(n, 3.0)))
        init_list.append((f"gauss_{n}", _gaussian_init(n, 0.3)))
        init_list.append((f"bilin_{n}", _bilinear_init(n, rng)))
    
    # Spectral flat init for a couple sizes
    for n in [512, 1024]:
        init_list.append((f"spec_{n}", _spectral_flat_init(n, rng, 30)))
    
    # Random
    for n in [512, 1024]:
        f_r = rng.random(n)
        f_rn = _normalize(f_r)
        if f_rn is not None:
            init_list.append((f"rand_{n}", f_rn))
    
    print(f"Total initializations: {len(init_list)}")
    
    # Time budget for Stage 1
    stage1_budget = (deadline - time.monotonic()) * 0.55
    time_per_init = max(3.0, stage1_budget / len(init_list) * 0.7)
    max_iters_per = int(min(3000, max(200, time_per_init * 150)))
    
    for idx, (name, f0) in enumerate(init_list):
        if time.monotonic() > deadline - 60:
            break
        
        n = len(f0)
        # Scale iterations based on n (larger n needs fewer iters per unit time)
        iters = int(max(100, min(max_iters_per, max_iters_per * 512.0 / n)))
        
        f_opt, s_opt = _mirror_descent_adaptive(f0, iters, base_eta=0.4, label=name)
        elapsed = time.monotonic() - (deadline - stage1_budget - time_per_init * idx)
        
        all_results.append((s_opt, f_opt.copy(), name))
        
        if s_opt < best_score:
            best_score = s_opt
            best_f = f_opt.copy()
            print(f"  ** New best: {best_score:.6f} from {name} (n={n}) **")
    
    # Sort results
    all_results.sort(key=lambda x: x[0])
    print(f"\n  Top 5 from Stage 1:")
    for i in range(min(5, len(all_results))):
        print(f"    {all_results[i][2]}: score={all_results[i][0]:.6f}, n={len(all_results[i][1])}")
    
    # Stage 2: LP refinement on top candidates
    print("\n=== Stage 2: LP Frank-Wolfe Refinement ===")
    
    stage2_budget = (deadline - time.monotonic()) * 0.25
    for i in range(min(3, len(all_results))):
        if time.monotonic() > deadline - 60:
            break
        s_cur, f_cur, name = all_results[i]
        n = len(f_cur)
        if n > 1500:
            continue
        
        lp_time = min(stage2_budget / max(1, 3 - i), 60.0)
        if lp_time < 5:
            continue
        
        f_lp, s_lp = _lp_refine(f_cur, max_iters=20, time_budget=lp_time)
        
        if s_lp < best_score:
            best_score = s_lp
            best_f = f_lp.copy()
            print(f"  ** New best from LP: {best_score:.6f} ({name}) **")
        else:
            print(f"  LP on {name}: {s_lp:.6f} (no improvement)")
    
    # Stage 3: Coordinate descent on current best
    print("\n=== Stage 3: Coordinate Descent ===")
    
    if time.monotonic() < deadline - 40:
        n_best = len(best_f)
        # CD is O(n * rounds * log_range) evaluations, can be slow for large n
        # Only do it if n is manageable
        if n_best <= 1024:
            # Create a smaller version if needed
            cd_budget = min(120.0, (deadline - time.monotonic()) * 0.2)
            n_rounds = min(15, max(3, int(cd_budget / max(1, n_best / 100))))
            f_cd, s_cd = _coordinate_descent(best_f, max_rounds=n_rounds, label="CD")
            if s_cd < best_score:
                best_score = s_cd
                best_f = f_cd.copy()
                print(f"  ** New best from CD: {best_score:.6f} **")
            else:
                print(f"  CD: {s_cd:.6f} (no improvement)")
    
    # Stage 4: Final refinement - adaptive mirror descent with many iterations
    print("\n=== Stage 4: Final Refinement ===")
    
    if time.monotonic() < deadline - 20:
        remaining = deadline - time.monotonic() - 5
        n_iters = int(min(10000, remaining * 200))
        
        # Try with different eta values
        for eta_try in [0.5, 0.3, 0.15]:
            if time.monotonic() > deadline - 10:
                break
            f_fin, s_fin = _mirror_descent_adaptive(best_f, n_iters, base_eta=eta_try, label=f"final_eta{eta_try}")
            if s_fin < best_score:
                best_score = s_fin
                best_f = f_fin.copy()
                print(f"  ** New best (eta={eta_try}): {best_score:.6f} **")
        
        # One more CD pass
        if time.monotonic() < deadline - 30 and len(best_f) <= 768:
            f_cd2, s_cd2 = _coordinate_descent(best_f, max_rounds=8, label="CD2")
            if s_cd2 < best_score:
                best_score = s_cd2
                best_f = f_cd2.copy()
                print(f"  ** New best from CD2: {best_score:.6f} **")
    
    # Final verification
    final_list = [float(x) for x in best_f]
    final_score = float(evaluate_sequence(final_list))
    print(f"\nFinal verified score: {final_score:.6f} (n={len(final_list)})")
    
    if not np.isfinite(final_score) or final_score > 2.0:
        print("Falling back to parent.")
        return [float(x) for x in parent]
    
    return final_list
