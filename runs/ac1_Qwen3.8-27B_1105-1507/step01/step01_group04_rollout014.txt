"""
Optimizes non-negative step function heights to minimize 2n*max(conv(f,f))/(sum(f))^2.
Key innovations over previous approach:
1. Momentum-enhanced mirror descent for faster convergence
2. Asymmetric parametric initializations (skewed Gaussians)
3. Simulated annealing restarts to escape local minima
4. Pairwise swap local search for final polishing
5. Multi-length exploration (256, 512, 1024)

import numpy as np
import time

def _fft_convolve_full(f):
    n = len(f)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    F = np.fft.rfft(f, size)
    h = np.fft.irfft(F * F, size)
    return h[:2 * n - 1]

def _score_raw(f):
    n = len(f)
    h = _fft_convolve_full(f)
    M = float(np.max(h))
    S = float(np.sum(f))
    if S < 0.01:
        return float('inf')
    return 2.0 * n * M / (S * S)

def _normalize(f):
    f = np.maximum(f, 0.0)
    s = float(np.sum(f))
    if s < 1e-30:
        return None
    return f / s

def _pressure_vector(f, active_lags):
    n = len(f)
    p = np.zeros(n, dtype=np.float64)
    for k in active_lags:
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        if j_end >= j_start:
            lo = k - j_end
            hi = k - j_start
            p[j_start:j_end + 1] += 2.0 * f[lo:hi + 1][::-1]
    p /= float(len(active_lags))
    return p

def _skew_gaussian_init(n, center, width, skew):
    """Generate a skewed Gaussian initialization."""
    j = np.arange(n, dtype=np.float64) / n
    x = (j - center) / width
    f = np.exp(-0.5 * x * x + skew * x)
    f = np.maximum(f, 0.0)
    s = np.sum(f)
    if s > 1e-30:
        f /= s
    else:
        f = np.ones(n) / n
    return f

def _ramp_init(n, sharpness):
    """Generate a ramp-like initialization (mass concentrated on one side)."""
    j = np.arange(n, dtype=np.float64) / n
    f = np.exp(-sharpness * (1.0 - j))
    f = np.maximum(f, 0.0)
    s = np.sum(f)
    if s > 1e-30:
        f /= s
    else:
        f = np.ones(n) / n
    return f

def _bump_init(n, center, width):
    """Generate a bump initialization (mass concentrated around center)."""
    j = np.arange(n, dtype=np.float64) / n
    f = np.exp(-0.5 * ((j - center) / width) ** 2)
    f = np.maximum(f, 0.0)
    s = np.sum(f)
    if s > 1e-30:
        f /= s
    return f

def _mirror_descent_momentum(f_init, n_iter, eta_scale, threshold_frac, 
                              momentum=0.7, anneal=False, label=""):
    """
    Mirror descent with momentum and optional simulated annealing restarts.
    """
    f = np.asarray(f_init, dtype=np.float64).copy()
    f = _normalize(f)
    if f is None:
        return f_init.copy(), float('inf')
    n = len(f)
    best_f = f.copy()
    best_score = _score_raw(f)
    v = np.zeros(n, dtype=np.float64)  # momentum buffer
    rng = np.random.default_rng(hash(label) % 2**32)

    for it in range(n_iter):
        h = _fft_convolve_full(f)
        M = float(np.max(h))
        if M <= 0:
            break

        threshold = threshold_frac * M + 1e-15
        active = np.where(h >= threshold)[0]
        if len(active) == 0:
            active = np.array([int(np.argmax(h))], dtype=np.int64)

        p = _pressure_vector(f, active)
        
        # Momentum update
        v = momentum * v + (1.0 - momentum) * p
        
        # Adaptive step size with cosine annealing
        if anneal:
            progress = it / max(1, n_iter - 1)
            eta = eta_scale * (0.5 + 0.5 * np.cos(np.pi * progress)) / M
        else:
            eta = eta_scale / M
        
        log_ratio = -eta * v
        log_ratio = np.clip(log_ratio, -30.0, 30.0)
        f = f * np.exp(log_ratio)
        
        # Simulated annealing: random perturbation
        if anneal and it % 200 == 100:
            temp = (1.0 - it / n_iter) * 0.05
            noise = rng.standard_normal(n) * temp * f
            f = f + noise
        
        f_norm = _normalize(f)
        if f_norm is None:
            f = np.ones(n) / n
        else:
            f = f_norm

        score = _score_raw(f)
        if score < best_score:
            best_score = score
            best_f = f.copy()

    return best_f, best_score

def _pairwise_swap_search(f_init, n_swaps, time_budget, label=""):
    """
    Local search using pairwise mass transfers between coordinates.
    """
    f = np.asarray(f_init, dtype=np.float64).copy()
    f = _normalize(f)
    if f is None:
        return f_init.copy(), float('inf')
    n = len(f)
    best_f = f.copy()
    best_score = _score_raw(f)
    start_time = time.monotonic()
    rng = np.random.default_rng(12345)

    for it in range(n_swaps):
        if time.monotonic() - start_time > time_budget:
            break
        
        # Pick two random coordinates
        i = rng.integers(0, n)
        j = rng.integers(0, n)
        if i == j:
            continue
        
        # Try moving a small fraction of mass from i to j (or vice versa)
        # Try multiple transfer amounts
        for frac in [0.01, 0.05, 0.1]:
            if f[i] < 1e-15:
                continue
            delta = frac * f[i]
            f_try = f.copy()
            f_try[i] -= delta
            f_try[j] += delta
            f_try = np.maximum(f_try, 0.0)
            s = float(np.sum(f_try))
            if s < 1e-15:
                continue
            f_try /= s
            
            score = _score_raw(f_try)
            if score < best_score - 1e-12:
                best_score = score
                best_f = f_try.copy()
                f = f_try
                break
    
    return best_f, best_score

def _coordinate_descent_polish(f_init, n_sweeps, time_budget):
    """
    Coordinate descent: for each coordinate, find the optimal value
    given all others fixed.
    """
    f = np.asarray(f_init, dtype=np.float64).copy()
    f = _normalize(f)
    if f is None:
        return f_init.copy(), float('inf')
    n = len(f)
    best_f = f.copy()
    best_score = _score_raw(f)
    start_time = time.monotonic()

    for sweep in range(n_sweeps):
        if time.monotonic() - start_time > time_budget:
            break
        
        for j in range(n):
            # Try perturbing coordinate j
            old_val = f[j]
            if old_val < 1e-18:
                # Try adding a small amount
                for frac in [1e-4, 1e-3, 1e-2]:
                    f_try = f.copy()
                    # Take from the maximum other coordinate
                    mask = np.ones(n, dtype=bool)
                    mask[j] = False
                    max_idx = np.argmax(f * mask)
                    delta = frac * f[max_idx]
                    f_try[max_idx] -= delta
                    f_try[j] += delta
                    if f_try[max_idx] < 0:
                        continue
                    s = np.sum(f_try)
                    f_try /= s
                    score = _score_raw(f_try)
                    if score < best_score - 1e-12:
                        best_score = score
                        best_f = f_try.copy()
                        f = f_try
                        break
            else:
                # Try reducing this coordinate
                for frac in [0.01, 0.05, 0.1]:
                    f_try = f.copy()
                    delta = frac * old_val
                    f_try[j] -= delta
                    # Distribute to a random other coordinate
                    mask = np.ones(n, dtype=bool)
                    mask[j] = False
                    candidates = np.where(f > 1e-15)[0]
                    if len(candidates) == 0:
                        continue
                    target = candidates[np.random.randint(len(candidates))]
                    f_try[target] += delta
                    s = np.sum(f_try)
                    f_try /= s
                    score = _score_raw(f_try)
                    if score < best_score - 1e-12:
                        best_score = score
                        best_f = f_try.copy()
                        f = f_try
                        break

    return best_f, best_score

"""
Optimizes non-negative step function heights to minimize 2n*max(conv(f,f))/(sum(f))^2.
Key innovations: momentum-enhanced mirror descent, asymmetric parametric initializations,
simulated annealing restarts, coordinate descent polish, and pairwise swap local search.
Explores multi-length sequences and diverse starting points.
"""

import numpy as np
import time

def _fft_convolve_full(f):
    n = len(f)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    F = np.fft.rfft(f, size)
    h = np.fft.irfft(F * F, size)
    return h[:2 * n - 1]

def _score_raw(f):
    n = len(f)
    h = _fft_convolve_full(f)
    M = float(np.max(h))
    S = float(np.sum(f))
    if S < 0.01:
        return float('inf')
    return 2.0 * n * M / (S * S)

def _normalize(f):
    f = np.maximum(f, 0.0)
    s = float(np.sum(f))
    if s < 1e-30:
        return None
    return f / s

def _pressure_vector(f, active_lags):
    n = len(f)
    p = np.zeros(n, dtype=np.float64)
    for k in active_lags:
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        if j_end >= j_start:
            lo = k - j_end
            hi = k - j_start
            p[j_start:j_end + 1] += 2.0 * f[lo:hi + 1][::-1]
    p /= float(len(active_lags))
    return p

def _skew_gaussian_init(n, center, width, skew):
    j = np.arange(n, dtype=np.float64) / float(n)
    x = (j - center) / max(width, 0.01)
    f = np.exp(-0.5 * x * x + skew * x)
    f = np.maximum(f, 0.0)
    s = np.sum(f)
    if s > 1e-30:
        f /= s
    else:
        f = np.ones(n) / n
    return f

def _ramp_init(n, sharpness):
    j = np.arange(n, dtype=np.float64) / float(n)
    f = np.exp(-abs(sharpness) * (1.0 - j))
    f = np.maximum(f, 0.0)
    s = np.sum(f)
    if s > 1e-30:
        f /= s
    else:
        f = np.ones(n) / n
    return f

def _bump_init(n, center, width):
    j = np.arange(n, dtype=np.float64) / float(n)
    f = np.exp(-0.5 * ((j - center) / max(width, 0.01)) ** 2)
    f = np.maximum(f, 0.0)
    s = np.sum(f)
    if s > 1e-30:
        f /= s
    return f

def _mirror_descent_momentum(f_init, n_iter, eta_scale, threshold_frac,
                              momentum=0.7, anneal=False, label=""):
    f = np.asarray(f_init, dtype=np.float64).copy()
    f = _normalize(f)
    if f is None:
        return f_init.copy(), float('inf')
    n = len(f)
    best_f = f.copy()
    best_score = _score_raw(f)
    v = np.zeros(n, dtype=np.float64)
    seed_val = hash(label) % (2**31)
    rng = np.random.default_rng(seed_val)

    for it in range(n_iter):
        h = _fft_convolve_full(f)
        M = float(np.max(h))
        if M <= 0:
            break

        threshold = threshold_frac * M + 1e-15
        active = np.where(h >= threshold)[0]
        if len(active) == 0:
            active = np.array([int(np.argmax(h))], dtype=np.int64)

        p = _pressure_vector(f, active)
        v = momentum * v + (1.0 - momentum) * p

        if anneal:
            progress = float(it) / max(1, n_iter - 1)
            eta = eta_scale * (0.3 + 0.7 * (1.0 - progress)) / M
        else:
            eta = eta_scale / M

        log_ratio = -eta * v
        log_ratio = np.clip(log_ratio, -30.0, 30.0)
        f = f * np.exp(log_ratio)

        if anneal and (it % 300 == 150) and it < n_iter * 0.8:
            temp = (1.0 - float(it) / n_iter) * 0.03
            noise = rng.standard_normal(n) * temp * (f + 1e-10)
            f = f + noise

        f_norm = _normalize(f)
        if f_norm is None:
            f = np.ones(n) / n
        else:
            f = f_norm

        score = _score_raw(f)
        if score < best_score:
            best_score = score
            best_f = f.copy()

    return best_f, best_score

def _pairwise_swap_search(f_init, n_swaps, time_budget, label=""):
    f = np.asarray(f_init, dtype=np.float64).copy()
    f = _normalize(f)
    if f is None:
        return f_init.copy(), float('inf')
    n = len(f)
    best_f = f.copy()
    best_score = _score_raw(f)
    start_time = time.monotonic()
    rng = np.random.default_rng(42)

    for it in range(n_swaps):
        if time.monotonic() - start_time > time_budget:
            break

        i = int(rng.integers(0, n))
        j = int(rng.integers(0, n))
        if i == j:
            continue

        for frac in [0.005, 0.02, 0.05, 0.1]:
            # Try i -> j
            if f[i] > 1e-14:
                delta = frac * f[i]
                f_try = f.copy()
                f_try[i] -= delta
                f_try[j] += delta
                s = float(np.sum(f_try))
                if s > 1e-14:
                    f_try /= s
                    score = _score_raw(f_try)
                    if score < best_score - 1e-11:
                        best_score = score
                        best_f = f_try.copy()
                        f = f_try
                        break

    return best_f, best_score

def _coord_descent_polish(f_init, n_sweeps, time_budget, seed=7):
    f = np.asarray(f_init, dtype=np.float64).copy()
    f = _normalize(f)
    if f is None:
        return f_init.copy(), float('inf')
    n = len(f)
    best_f = f.copy()
    best_score = _score_raw(f)
    start_time = time.monotonic()
    rng = np.random.default_rng(seed)

    for sweep in range(n_sweeps):
        if time.monotonic() - start_time > time_budget:
            break
        for j in range(n):
            if time.monotonic() - start_time > time_budget:
                break
            old_val = f[j]
            if old_val < 1e-16:
                continue
            for frac in [0.02, 0.05, 0.1]:
                delta = frac * old_val
                # Move mass to a random other position
                target = int(rng.integers(0, n))
                if target == j:
                    target = (j + 1) % n
                f_try = f.copy()
                f_try[j] -= delta
                f_try[target] += delta
                s = float(np.sum(f_try))
                if s > 1e-14:
                    f_try /= s
                    score = _score_raw(f_try)
                    if score < best_score - 1e-11:
                        best_score = score
                        best_f = f_try.copy()
                        f = f_try
                        break

    return best_f, best_score
