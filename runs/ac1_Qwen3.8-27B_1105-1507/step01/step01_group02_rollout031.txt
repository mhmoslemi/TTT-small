def project_simplex(v):
    """Project vector v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cv = np.cumsum(u) - 1.0
    rho_candidates = np.where(u * np.arange(1, n + 1) > cv)[0]
    if len(rho_candidates) == 0:
        return np.ones(n) / n
    rho = rho_candidates[-1]
    theta = cv[rho] / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

"""
Optimizes a nonnegative sequence to minimize 2*n*max(convolve(a,a))/sum(a)^2.
Uses FISTA with log-sum-exp smoothing for exploration, momentum subgradient
descent with harmonic diversification for refinement, and coordinate polish.

import numpy as np
import time
from scipy.optimize import linprog


def project_simplex(v):
    """Project vector v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cv = np.cumsum(u) - 1.0
    rho_candidates = np.where(u * np.arange(1, n + 1) > cv)[0]
    if len(rho_candidates) == 0:
        return np.ones(n) / n
    rho = rho_candidates[-1]
    theta = cv[rho] / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def fast_convolve(a):
    """Compute full linear convolution of a with itself using FFT."""
    n = len(a)
    size = 2 * n - 1
    N = 1 << (size - 1).bit_length()
    fa = np.fft.rfft(a, N)
    fc = fa * fa
    result = np.fft.irfft(fc, N)[:size]
    result[result < 0] = 0.0
    return result


def compute_smooth_grad(a, tau):
    """Compute gradient of tau*logsumexp(conv(a,a)/tau) using FFT cross-correlation."""
    n = len(a)
    c = fast_convolve(a)
    c_shift = c - c.max()
    weights = np.exp(c_shift / tau)
    wsum = weights.sum()
    if wsum < 1e-30:
        weights = np.ones(2 * n - 1) / (2 * n - 1)
    else:
        weights /= wsum
    # grad[m] = 2 * sum_j a[j] * weights[m+j]
    # = 2 * cross_correlation(a, weights)[m]
    # Using FFT: irfft(conj(rfft(a,N)) * rfft(weights,N), N)
    N = 1 << (3 * n - 2).bit_length()
    fa = np.fft.rfft(a, N)
    fw = np.fft.rfft(weights, N)
    xc = np.fft.irfft(np.conj(fa) * fw, N)
    return 2.0 * xc[:n], c


def fista_optimize(a0, deadline, tau_init=0.05, tau_min=1e-4, rng=None):
    """FISTA with log-sum-exp smoothing of max convolution."""
    if rng is None:
        rng = np.random.default_rng(0)
    n = len(a0)
    a = project_simplex(a0.astype(np.float64).copy())
    a_prev = a.copy()
    t = 1.0
    tau = tau_init
    
    c = fast_convolve(a)
    best_peak = c.max()
    best_a = a.copy()
    
    it = 0
    stall = 0
    N_fft = 1 << (3 * n - 2).bit_length()
    
    while time.time() < deadline:
        it += 1
        c = fast_convolve(a)
        c_shift = c - c.max()
        weights = np.exp(c_shift / tau)
        wsum = weights.sum()
        if wsum < 1e-30:
            weights = np.ones(2 * n - 1) / (2 * n - 1)
        else:
            weights /= wsum
        
        # Gradient via FFT cross-correlation
        fa = np.fft.rfft(a, N_fft)
        fw = np.fft.rfft(weights, N_fft)
        xc = np.fft.irfft(np.conj(fa) * fw, N_fft)
        grad = 2.0 * xc[:n]

def momentum_subgradient(a0, deadline, rng=None):
    """Momentum subgradient descent on max(convolve(a,a)) with sum(a)=1."""
    if rng is None:
        rng = np.random.default_rng(1)
    n = len(a0)
    a = project_simplex(a0.astype(np.float64).copy())
    v = np.zeros(n)  # momentum
    mu = 0.9  # momentum coefficient
    step = 1.0 / np.sqrt(n)
    
    c = fast_convolve(a)
    best_peak = c.max()
    best_a = a.copy()
    
    it = 0
    stall = 0
    step_min = step * 1e-4
    
    while time.time() < deadline:
        it += 1
        
        c = fast_convolve(a)
        peak = c.max()
        
        if peak < best_peak * (1.0 - 1e-12):
            best_peak = peak
            best_a = a.copy()
            stall = 0
        else:
            stall += 1
        
        # Active peaks
        threshold = peak * (1.0 - 1e-4)
        peak_indices = np.where(c >= threshold)[0]
        if len(peak_indices) > 20:
            top_idx = np.argsort(c[peak_indices])[::-1][:10]
            peak_indices = peak_indices[top_idx]
        
        # Subgradient (averaged over active peaks)
        grad = np.zeros(n)
        for k in peak_indices:
            lo = max(0, k - n + 1)
            hi = min(n - 1, k)
            if lo <= hi:
                grad[lo:hi + 1] += 2.0 * a[k - hi:k - lo + 1][::-1]
        grad /= max(1, len(peak_indices))
        
        # Momentum update
        v = mu * v - step * grad
        a_new = project_simplex(a + v)
        
        a = a_new
        
        # Decay step size
        if stall > 200:
            step = max(step * 0.8, step_min)
            stall = 0
        
        # Periodic perturbation
        if it % 1000 == 0 and time.time() < deadline - 10:
            temp = 0.03 * step_min ** 0.5
            a = project_simplex(a + temp * rng.standard_normal(n) / np.sqrt(n))
            v *= 0.5
    
    return best_a


def harmonic_perturb(a, rng, n_harmonics=5):
    """Try harmonic perturbations and return best."""
    n = len(a)
    x = np.linspace(0, 1, n, endpoint=False)
    best_a = a.copy()
    best_peak = fast_convolve(a).max()
    
    for h in range(n_harmonics):
        k = rng.integers(1, max(2, n // 2))
        phi = rng.uniform(0, 2 * np.pi)
        eps = rng.uniform(0.05, 0.3)
        
        for sign in [1.0, -1.0]:
            a_try = a + sign * eps * np.cos(2 * np.pi * k * x + phi)
            a_try = project_simplex(a_try)
            peak = fast_convolve(a_try).max()
            if peak < best_peak:
                best_peak = peak
                best_a = a_try.copy()
    
    return best_a


def lp_diversify(a, P, time_limit=5.0):
    """LP-based diversification: find b with high sum and conv(a,b) <= P."""
    n = len(a)
    if n > 2048:
        return None
    
    A = np.zeros((2 * n - 1, n), dtype=np.float64)
    for k in range(2 * n - 1):
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        if lo <= hi:
            A[k, lo:hi + 1] = a[k - hi:k - lo + 1][::-1]
    
    c_obj = -np.ones(n)
    b_ub = np.full(2 * n - 1, P)
    
    try:
        res = linprog(c_obj, A_ub=A, b_ub=b_ub,
                      bounds=[(0.0, None)] * n, method='highs',
                      options={'time_limit': time_limit})
        if res.success:
            b = res.x
            b_sum = b.sum()
            if b_sum > 1e-8:
                b = b / b_sum  # normalize to sum=1
                return b
    except Exception:
        pass
    return None


def alt_projection_init(n, rng, n_iter=60):
    """Alternating projection for flat-spectrum initialization."""
    N = 1 << (2 * n - 2).bit_length()
    phases = rng.uniform(0.0, 2.0 * np.pi, N // 2 + 1)
    F = np.exp(1j * phases)
    F[0] = 1.0 + 0j
    
    a = np.ones(n) / n
    for _ in range(n_iter):
        a_full = np.fft.irfft(F, N)
        a = a_full[:n].copy()
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s < 1e-10:
            a = np.ones(n) / n
        else:
            a = a / s
        F_full = np.fft.rfft(a, N)
        weights = np.linspace(1.0, 0.8, len(F_full))
        F = weights ** 0.1 * np.exp(1j * np.angle(F_full))
        F[0] = F[0].real + 0j
    
    return a


def cosine_init(n, rng):

        return a / s if s > 1e-10 else np.ones(n) / n
    except Exception:
        return random_init(n, rng)


def optimize_single(a0, deadline, rng, method='fista'):
    """Optimize a single sequence using the specified method."""
    if method == 'fista':
        return fista_optimize(a0, deadline, rng=rng)
    elif method == 'subgrad':
        return momentum_subgradient(a0, deadline, rng=rng)
    else:
        return a0
