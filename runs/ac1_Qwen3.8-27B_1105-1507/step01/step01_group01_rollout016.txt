def lbfgs_optimize(a, T, deadline, maxiter=200, seed=0):
    """L-BFGS optimization on the log-parameterized smoothed objective."""
    a = normalize_to_sum1(a.copy())
    if a is None:
        return None, float('inf')
    
    b0 = np.log(a + 1e-30)
    n = len(a)
    
    def obj(b):
        return smoothed_objective(b, T)
    
    def grad(b):
        return smoothed_gradient(b, T)
    
    try:
        result = minimize(obj, b0, jac=grad, method='L-BFGS-B',
                         options={'maxiter': maxiter, 'ftol': 1e-14, 'gtol': 1e-8})
        b_opt = result.x
        a_opt = np.exp(b_opt - np.max(b_opt))
        a_opt = a_opt / np.sum(a_opt)
    except Exception:

    sc = local_score(a_opt)

def resample_to_length(a, n_new, seed=0):
    """Resample a sequence to a new length using linear interpolation."""
    a = np.asarray(a, dtype=np.float64)
    n_old = len(a)
    if n_old == n_new:
        return a.copy()
    old_x = np.linspace(0, 1, n_old)
    new_x = np.linspace(0, 1, n_new)
    a_new = np.interp(new_x, old_x, a)
    return normalize_to_sum1(a_new)
