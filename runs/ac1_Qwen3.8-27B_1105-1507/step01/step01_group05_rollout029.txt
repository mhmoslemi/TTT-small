```python
def project_simplex(v):
    """Project v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n+1) > (cssv - 1))[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0)
```

```python
"""
Hybrid optimization for minimizing 2*n*max(convolve(a,a))/sum(a)^2.
Combines softmax gradient descent, LP-based direction finding,
and random search with multiple sequence lengths and restarts.
"""

import numpy as np
from scipy.optimize import linprog
import time

def project_simplex(v):
    """Project v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0)

def score_normalized(a):
    """Score for normalized sequence (sum=1)."""
    n = len(a)
    C = np.convolve(a, a)
    return 2.0 * n * C.max()

def score_and_grad(a, tau=None):
    """Compute score and gradient using softmax approximation."""
    n = len(a)
    C = np.convolve(a, a)
    M = C.max()
    
    if tau is None:
        tau = max(M * 0.01, 1e-6)
    
    # Softmax weights
    exp_C = np.exp((C - M) / tau)
    weights = exp_C / exp_C.sum()
    soft_max = (C * weights).sum()
    score = 2.0 * n * soft_max
    
    # Gradient computation
    # grad[m] = 2 * sum_{j=0}^{n-1} a[j] * weights[m+j] - weights[2m] * a[m]
    idx = np.arange(n)
    W = weights[idx[:, None] + idx[None, :]]  # W[m,j] = weights[m+j]
    grad = 2.0 * (W @ a)
    grad -= weights[2 * idx] * a  # diagonal correction
    
    return score, grad

def gradient_descent(a, n_iter=5000, verbose=False, rng=None):
    """Projected gradient descent on the simplex."""
    if rng is None:
        rng = np.random.default_rng(0)
    
    a = a.copy()
    S = a.sum()
    if S < 1e-10:
        return a, np.inf

    best = a.copy()
    best_score = score_normalized(a)
    
    # Adaptive tau
    C0 = np.convolve(a, a)
    M0 = C0.max()
    tau = M0 * 0.1
    
    eta = 0.01  # initial step size
    n_stall = 0

        score, grad = score_and_grad(a, tau)
        
        # Projected gradient step

        a_new = project_simplex(a_new)
        
        new_score = score_normalized(a_new)

            a = a_new
            n_stall = 0

                        best_score = s_p

def lp_direction(f, M):
    """Solve LP to find best direction b with conv(f,b) <= M, b >= 0."""
    n = len(f)
    A = np.zeros((2 * n - 1, n))
    for k in range(2 * n - 1):
        j_start = max(0, k - n + 1)
        j_end = min(n, k + 1)
        for j in range(j_start, j_end):
            A[k, j] = f[k - j]
    
    c_obj = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    
    try:
        res = linprog(c_obj, A_ub=A, b_ub=b_ub, bounds=bounds,
                     method='highs', options={'presolve': True})
    except Exception:
        return None
    
    if not res.success or res.x is None:
        return None
    
    b = np.clip(res.x, 0, None)
    if b.sum() < 1e-15:
        return None
    
    return b / b.sum()
```

```python
"""
Hybrid optimization for minimizing 2*n*max(convolve(a,a))/sum(a)^2.
Combines softmax gradient descent on the probability simplex, LP-based
direction finding, and randomized local search across multiple sequence
lengths with adaptive time management.
"""

import numpy as np
from scipy.optimize import linprog
import time


def project_simplex(v):
    """Project v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0)


def score_norm(a):
    """Compute score for a normalized sequence (sum=1, a>=0)."""
    n = len(a)
    C = np.convolve(a, a)
    return 2.0 * n * C.max()


def grad_softmax(a, tau):
    """Compute softmax-based score and gradient for normalized a."""
    n = len(a)
    C = np.convolve(a, a)
    M = C.max()
    if tau <= 0:
        tau = max(M * 0.01, 1e-8)
    exp_C = np.exp((C - M) / tau)
    weights = exp_C / exp_C.sum()
    soft_max = (C * weights).sum()
    score = 2.0 * n * soft_max
    idx = np.arange(n)
    W = weights[idx[:, None] + idx[None, :]]
    grad = 2.0 * (W @ a)
    grad -= weights[2 * idx] * a
    return score, grad


def grad_exact(a, rel_tol=1e-6):
    """Compute exact subgradient at near-argmax positions."""
    n = len(a)
    C = np.convolve(a, a)
    M = C.max()
    k_indices = np.where(C >= M * (1 - rel_tol))[0]
    score = 2.0 * n * M
    grad = np.zeros(n)
    for k in k_indices:
        for m in range(n):
            idx2 = k - m
            if 0 <= idx2 < n:
                if k == 2 * m:
                    grad[m] += a[idx2]
                else:
                    grad[m] += 2.0 * a[idx2]
    grad /= max(1, len(k_indices))
    return score, grad


def gd_optimize(a, time_budget, rng=None, verbose=False):
    """Projected gradient descent with adaptive step and tau."""
    if rng is None:
        rng = np.random.default_rng(0)
    a = a.astype(np.float64).copy()
    S = a.sum()
    if S < 1e-10:
        return a, np.inf
    a = a / S
    best = a.copy()
    best_score = score_norm(a)
    C0 = np.convolve(a, a)
    M0 = C0.max()
    tau = M0 * 0.1
    eta = 0.005
    momentum = np.zeros_like(a)
    beta = 0.9
    n_stall = 0
    start = time.monotonic()
    it = 0
    while time.monotonic() - start < time_budget:
        score, grad = grad_softmax(a, tau)
        momentum = beta * momentum + (1.0 - beta) * grad
        a_new = a - eta * momentum
        a_new = project_simplex(a_new)
        new_score = score_norm(a_new)
        if new_score < best_score - 1e-13:
            best = a_new.copy()
            best_score = new_score
            a = a_new
            n_stall = 0
            if it % 200 == 0 and it > 0:
                tau = max(tau * 0.97, M0 * 0.005)
            if verbose and it % 500 == 0:
                print(f"        GD it={it} score={best_score:.7f} tau={tau:.2e} eta={eta:.5f}")
        else:
            n_stall += 1
            eta *= 0.85
            if n_stall > 80:
                a_pert = a + rng.normal(0, 0.005, len(a))
                a_pert = np.clip(a_pert, 0, None)
                Sp = a_pert.sum()
                if Sp > 1e-10:
                    a_pert /= Sp
                    sp = score_norm(a_pert)
                    if sp < best_score:
                        a = a_pert
                        best = a.copy()
                        best_score = sp
                        eta = 0.005
                        momentum = np.zeros_like(a)
                        n_stall = 0
                if n_stall > 150:
                    break
        it += 1
    return best, best_score


def build_conv_matrix(f):
    """Build matrix A such that A @ b = np.convolve(f, b, mode='full')."""
    n = len(f)
    A = np.zeros((2 * n - 1, n))
    for k in range(2 * n - 1):
        j_start = max(0, k - n + 1)
        j_end = min(n, k + 1)
        for j in range(j_start, j_end):
            A[k, j] = f[k - j]
    return A


def lp_optimize(a, time_budget, t0=0.5, rng=None, verbose=False):
    """LP-based direction finding optimization."""
    if rng is None:
        rng = np.random.default_rng(0)
    n = len(a)
    a = a.astype(np.float64).copy()
    S = a.sum()
    if S < 1e-10:
        return a, np.inf
    a = a / S
    best = a.copy()
    best_score = score_norm(a)
    t = t0
    n_stall = 0
    start = time.monotonic()
    it = 0
    while time.monotonic() - start < time_budget:
        C = np.convolve(a, a)
        M = C.max()
        if n > 2048:
            break
        A = build_conv_matrix(a)
        c_obj = -np.ones(n)
        b_ub = np.full(2 * n - 1, M)
        bounds = [(0, None)] * n
        try:
            res = linprog(c_obj, A_ub=A, b_ub=b_ub, bounds=bounds,
                          method='highs', options={'presolve': True})
        except Exception:
            break
        if not res.success or res.x is None:
            break
        b = np.clip(res.x, 0, None)
        b_sum = b.sum()
        if b_sum < 1e-15:
            break
        g = b / b_sum
        a_new = (1 - t) * a + t * g
        a_new = np.clip(a_new, 0, None)
        Sn = a_new.sum()
        if Sn < 1e-10:
            t *= 0.5
            continue
        a_new = a_new / Sn
        new_score = score_norm(a_new)
        if new_score < best_score - 1e-13:
            best = a_new.copy()
            best_score = new_score
            a = a_new
            n_stall = 0
            if verbose and it % 10 == 0:
                print(f"        LP it={it} score={best_score:.7f} t={t:.4f}")
        else:
            t *= 0.7
            n_stall += 1
            if n_stall > 60:
                a_pert = a + rng.normal(0, 0.01, n)
                a_pert = np.clip(a_pert, 0, None)
                Sp = a_pert.sum()
                if Sp > 1e-10:
                    a_pert /= Sp
                    if score_norm(a_pert) < best_score:
                        a = a_pert
                        best = a.copy()
                        best_score = score_norm(a)
                        t = t0 * 0.4
                        n_stall = 0
                if n_stall > 120:

        it += 1
    return best, best_score


def random_search(a, time_budget, rng=None, verbose=False):
    """Randomized block perturbation local search."""
    if rng is None:
        rng = np.random.default_rng(0)
    n = len(a)
    a = a.astype(np.float64).copy()
    S = a.sum()
    if S < 1e-10:
        return a, np.inf
    a = a / S
    best = a.copy()
    best_score = score_norm(a)
    start = time.monotonic()
    it = 0
    while time.monotonic() - start < time_budget:
        a_c = a.copy()
        width = max(1, int(rng.integers(1, max(2, n // 16))))
        s_idx = int(rng.integers(0, n - width))
        scale = np.exp(rng.normal(0, 0.4))
        a_c[s_idx:s_idx + width] *= scale
        a_c = np.clip(a_c, 0, None)
        Sc = a_c.sum()
        if Sc < 1e-10:
            continue
        a_c /= Sc
        sc = score_norm(a_c)
        if sc < best_score:
            best = a_c.copy()
            best_score = sc
            a = a_c
        it += 1
    if verbose:
        print(f"        Random it={it} score={best_score:.7f}")
    return best, best_score

        final_list = [float(x) for x in best_overall.tolist()]
        final_score = evaluate_sequence(final_list)
        print(f"\nFinal score (verified): {final_score:.6f}")
        print(f"Sequence length: {len(final_list)}")
        if np.isfinite(final_score):
            return final_list

    # Fallback
    print("Using fallback uniform sequence")
    return [1.0] * 256
```
