def minimize_peak(a, max_iters=10000, tol=1e-6):
    n = len(a)
    a = np.maximum(a, 0)
    a = a / a.sum()
    
    for it in range(max_iters):
        b = fast_conv(a)
        M = b.max()
        
        # Active set: indices where b is close to max
        active = np.where(b > M * 0.999)[0]
        
        # Compute subgradient: average of 2*a[k*-j] over active set
        grad = np.zeros(n)
        for k in active:
            # grad[j] += 2*a[k-j] for valid j (where 0<=k-j<n)
            # j ranges from max(0, k-n+1) to min(n-1, k)
            j_lo = max(0, k - n + 1)
            j_hi = min(n - 1, k)
            # a[k-j] for j in [j_lo, j_hi] is a[k-j_hi : k-j_lo+1][::-1]
            # Let's verify: j=j_lo gives a[k-j_lo], j=j_hi gives a[k-j_hi]
            # So the slice is a[k-j_hi : k-j_lo+1] in reverse
            # k-j_lo = k - max(0, k-n+1) = min(k, n-1)
            # k-j_hi = k - min(n-1, k) = max(0, k-n+1)
            # So the slice is a[max(0,k-n+1) : min(k,n-1)+1][::-1]
            grad[j_lo:j_hi+1] += 2 * a[k-j_hi:k-j_lo+1][::-1]
        
        grad /= len(active)
        
        # Mirror descent update (multiplicative)
        # a_new[j] = a[j] * exp(-eta * grad[j])
        # Adaptive step size
        eta = 0.5 / (grad.max() + 1e-10)
        a = a * np.exp(-eta * grad)
        a = np.maximum(a, 0)
        s = a.sum()
        if s < 1e-10:
            break
        a = a / s
    
    return a

"""
Minimizes J(a) = 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Active-set mirror descent (multiplicative updates) - fast, no LP needed
2. Signal-processing-inspired initializations (chirps, modulated windows)
3. Multi-scale exploration (n=256 to 4096)
4. L-BFGS-B gradient polishing for final refinement
5. Perturbation-restart to escape local optima

import numpy as np
from scipy.optimize import minimize
import time


def fast_conv(a):
    """Full linear self-convolution via FFT."""
    n = len(a)
    s = 1
    while s < 2 * n - 1:
        s <<= 1
    fa = np.fft.rfft(a, s)
    return np.fft.irfft(fa * fa, s)[:2 * n - 1]


def compute_objective(a):
    """Compute J(a) = 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    s = a.sum()
    if s < 1e-10:
        return 1e10
    b = fast_conv(a)
    return float(2.0 * n * b.max() / (s * s))


def run_mirror_descent(a, max_iters, active_frac, eta0, eta_decay, seed, deadline):
    """Run active-set mirror descent to minimize peak autocorrelation."""
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-10:
        a = np.ones(n) / n
        return a
    a = a / s

    best_a = a.copy()
    best_M = fast_conv(a).max()
    eta = eta0

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        b = fast_conv(a)
        M = b.max()

        if M < best_M * (1.0 - 1e-10):
            best_M = M
            best_a = a.copy()

        threshold = M * active_frac + 1e-15
        active = np.where(b >= threshold)[0]

        grad = np.zeros(n)
        for k in active:
            j_lo = max(0, k - n + 1)
            j_hi = min(n - 1, k)
            grad[j_lo:j_hi + 1] += a[k - j_hi:k - j_lo + 1][::-1]
        grad *= 2.0 / len(active)

        g_max = grad.max()
        if g_max < 1e-15:
            break

        a_new = a * np.exp(-eta * grad / g_max)
        a_new = np.maximum(a_new, 1e-15)
        s_new = a_new.sum()
        if s_new < 1e-10:
            break
        a_new = a_new / s_new

        if np.max(np.abs(a_new - a)) < 1e-13:
            a = a_new
            break
        a = a_new
        eta *= eta_decay

    return best_a

def gradient_polish(a, max_iter):
    """Polish using L-BFGS-B to minimize peak autocorrelation."""

def resample_sequence(a, new_n):
    """Resample a sequence to a new length using linear interpolation."""
    old_n = len(a)
    t_old = np.linspace(0, 1, old_n)
    t_new = np.linspace(0, 1, new_n)
    new_a = np.interp(t_new, t_old, a)
    return np.maximum(new_a, 0.0)
