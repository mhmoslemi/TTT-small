from scipy.optimize import linprog

def normalize_seq(x):
    """Normalize a non-negative array to sum 1."""
    x = np.asarray(x, dtype=np.float64).ravel()
    x = np.clip(x, 0, None)
    s = x.sum()
    if s < 1e-300:
        return None
    return x / s

def compute_score_norm(f):
    """Compute 2*n*max(convolve(f,f)) assuming sum(f)=1, f>=0."""
    n = len(f)
    c = np.convolve(f, f, mode='full')
    return 2.0 * n * float(c.max())

def solve_minmax_lp(A_values, n):
    """
    Solve: min t s.t. (A @ x)_k <= t for k=0..m-1, sum(x)=1, x>=0, t>=0
    where A[k,j] = A_values[k,j] is given as a dense 2D array of shape (m, n).
    Returns (x_opt, t_opt) or (None, None) on failure.
    """
    m = A_values.shape[0]
    c = np.zeros(n + 1)
    c[n] = 1.0  # minimize t
    
    A_ub = np.zeros((m, n + 1))
    A_ub[:, :n] = A_values
    A_ub[:, n] = -1.0
    b_ub = np.zeros(m)

    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])

    """Build (2n-1) x n matrix where A[k,j] = f[k-j] if valid, else 0."""
    m = 2 * n - 1
    k_idx = np.arange(m, dtype=np.int64)[:, None]
    j_idx = np.arange(n, dtype=np.int64)[None, :]
    diff = k_idx - j_idx

def solve_minmax(A, n):
    """Solve min t s.t. A@x <= t*1, sum(x)=1, x>=0, t>=0."""
    m = A.shape[0]
    c = np.zeros(n + 1)
    c[n] = 1.0
    
    A_ub = np.hstack([A, -np.ones((m, 1))])
    b_ub = np.zeros(m)
    
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])
    
    bounds = [(0, None)] * (n + 1)
    
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                  bounds=bounds, method='highs')

        return None, None
    x = np.clip(res.x[:n], 0, None)

def run_subgradient(f_init, n, time_budget, rng):
    f = interpolate_to_length(f_init, n)
    best_f = f.copy()
    best_score = score_norm(f)

    lr = 0.01

    v = np.zeros(n)
    
    while time.monotonic() < deadline:
        # Compute autocorrelation
        c = np.convolve(f, f, mode='full')
        k_star = int(c.argmax())
        
        # Subgradient: grad[j] = 2 * f[k_star - j] (with zero extension)
        grad = np.zeros(n)
        for j in range(n):
            idx = k_star - j
            if 0 <= idx < n:
                grad[j] = 2.0 * f[idx]

        v = momentum * v - lr * grad
        f_new = f + v
        
        # Project onto simplex {x >= 0, sum(x) = 1}
        f_new = project_simplex(f_new)
        
        # Evaluate
        s = score_norm(f_new)
        if s < best_score:
            best_score = s
            best_f = f_new.copy()
            # Reduce learning rate
            lr *= 0.999
        else:
            # Keep old f, reduce learning rate
            lr *= 0.95
        
        f = f_new
        
        if lr < 1e-10:
            break
    
    return best_f, best_score

def project_simplex(v):
    """Project v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.where(u * np.arange(1, n+1) > (cssv - 1))[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1)
    return np.maximum(v - theta, 0)

            fidx_arr.append(k - j)
    return (np.array(rows_arr, dtype=np.intp),
            np.array(cols_arr, dtype=np.intp),
            np.array(fidx_arr, dtype=np.intp))


def build_conv_matrix(f, structure, n):
    """Build dense (2n-1) x n convolution matrix using precomputed structure."""
    rows, cols, fidx = structure
    m = 2 * n - 1
    A = np.zeros((m, n), dtype=np.float64)
    A[rows, cols] = f[fidx]
    return A


def solve_minmax_lp(A, n):
    """Solve min t s.t. A@x <= t, sum(x)=1, x>=0, t>=0. Returns (x, t)."""
    m = A.shape[0]
    c = np.zeros(n + 1)
    c[n] = 1.0
    A_ub = np.empty((m, n + 1), dtype=np.float64)
    A_ub[:, :n] = A
    A_ub[:, n] = -1.0
    b_ub = np.zeros(m)
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.ones(1)
    bounds = [(0.0, None)] * (n + 1)
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs',
                      options={'presolve': True})
    except Exception:
        return None, None
    if res.status != 0 or res.x is None:
        return None, None
    x = np.clip(res.x[:n], 0.0, None)
    s = float(x.sum())
    if s < 1e-300:
        return None, None
    x /= s
    return x, float(res.x[n])


def solve_maxsum_lp(A, b_val, n):
    """Solve max sum(x) s.t. A@x <= b_val, x>=0. Returns (x, sum_x)."""
    m = A.shape[0]
    c = -np.ones(n)
    b_ub = np.full(m, b_val)
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
    except Exception:
        return None, None
    if res.status != 0 or res.x is None:
        return None, None
    x = np.clip(res.x, 0.0, None)
    s = float(x.sum())
    return x, s


def interpolate_seq(f, n):
    """Interpolate normalized sequence to length n."""
    f = normalize_seq(f)
    if f is None:
        return np.ones(n) / n
    if f.size == n:
        return f.copy()
    if f.size == 1:
        return np.ones(n) / n
    old_t = np.linspace(0.0, 1.0, f.size)
    new_t = np.linspace(0.0, 1.0, n)
    out = np.interp(new_t, old_t, f)
    return normalize_seq(out)


def project

_simplex(v):
    """Project vector onto probability simplex."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    idx = np.arange(1, n + 1)
    cond = u * idx > (cssv - 1.0)
    if not np.any(cond):
        return np.ones(n) / n
    rho = int(np.where(cond)[0][-1])
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def run_alternating(f_init, n, structure, time_budget, rng):
    """Alternating LP: alternate min t s.t. (f*g)_k <= t and min t s.t. (g*f')_k <= t."""
    f = interpolate_seq(f_init, n)
    best_f = f.copy()
    best_score = score_norm(f)
    deadline = time.monotonic() + time_budget
    no_improve = 0

    while time.monotonic() < deadline:
        t0 = time.monotonic()

        # Phase 1: fix f, find g minimizing max(f*g)
        A_f = build_conv_matrix(f, structure, n)
        g, t1 = solve_minmax_lp(A_f, n)
        if g is None:
            break

        # Phase 2: fix g, find f' minimizing max(g*f')
        A_g = build_conv_matrix(g, structure, n)
        f_new, t2 = solve_minmax_lp(A_g, n)
        if f_new is None:
            break

        # Evaluate actual objective
        sc = score_norm(f_new)
        if sc < best_score - 1e-10:
            best_score = sc
            best_f = f_new.copy()
            no_improve = 0
        else:
            no_improve += 1

        f = f_new

        if no_improve > 30:
            # Perturb from best
            f = best_f + 0.03 * rng.standard_normal(n)
            f = normalize_seq(f)
            no_improve = 0

        if no_improve > 80:
            # Larger perturbation
            f = 0.7 * best_f + 0.3 * rng.dirichlet(np.ones(n))
            f = normalize_seq(f)
            no_improve = 0

        if no_improve > 150:
            break

        if time.monotonic() - t0 > time_budget:
            break

    return best_f, best_score


def run_lp_descent(f_init, n, structure, time_budget, rng):
    """LP direction method: find g maximizing sum(g) s.t. (f*g)_k <= M, step toward g."""
    f = interpolate_seq(f_init, n)
    best_f = f.copy()
    best_score = score_norm(f)
    deadline = time.monotonic() + time_budget
    stall = 0

    while time.monotonic() < deadline:
        t0 = time.monotonic()

        conv = np.convolve(f, f)
        M = float(conv.max())
        if M <= 0 or not np.isfinite(M):
            break

        # LP: max sum(g) s.t. (f*g)_k <= M, g >= 0
        A_f = build_conv_matrix(f, structure, n)
        g, s_g = solve_maxsum_lp(A_f, M, n)
        if g is None or s_g <= 1.0 + 1e-9:
            stall += 1
            if stall > 10:
                f = f + 0.1 * rng.standard_normal(n)
                f = normalize_seq(f)
            if stall > 30:
                break
            continue

        g_norm = g / s_g

        # Backtracking line search
        improved = False
        for tv in [1.0, 0.7, 0.5, 0.3, 0.1, 0.05, 0.01]:
            f_try = (1.0 - tv) * f + tv * g_norm
            f_try = normalize_seq(f_try)
            if f_try is None:
                continue
            sc = score_norm(f_try)
            if sc < best_score - 1e-10:
                best_score = sc
                best_f = f_try.copy()
                f = f_try
                improved = True
                break

        if improved:
            stall = 0
        else:
            stall += 1
            if stall % 5 == 0:
                f = best_f + 0.05 * rng.standard_normal(n)
                f = normalize_seq(f)
            if stall > 40:
                break

    return best_f, best_score
