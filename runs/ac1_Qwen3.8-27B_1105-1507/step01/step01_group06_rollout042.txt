import time

def fast_conv(a):
    """Full self-convolution via FFT."""
    n = len(a)
    s = 1
    while s < 2 * n - 1:
        s <<= 1
    fa = np.fft.rfft(a, s)
    return np.fft.irfft(fa * fa, s)[:2 * n - 1]

def soft_conv_objective(x, beta, n):
    """Objective: logsumexp(beta * b) where b = conv(softmax(x), softmax(x))"""
    # Compute softmax
    x_shift = x - np.max(x)
    exp_x = np.exp(x_shift)
    a = exp_x / exp_x.sum()
    b = fast_conv(a)
    # logsumexp
    max_b = np.max(beta * b)
    return float(max_b + np.log(np.sum(np.exp(beta * b - max_b))))

def soft_conv_gradient(x, beta, n):
    """Gradient of soft_conv_objective with respect to x."""
    x_shift = x - np.max(x)
    exp_x = np.exp(x_shift)
    a = exp_x / exp_x.sum()
    b = fast_conv(a)
    bb = beta * b
    max_bb = np.max(bb)
    exp_bb = np.exp(bb - max_bb)
    w = exp_bb / exp_bb.sum()
    # (w * a) truncated to length n

    # We need the first n elements
    # (w*a)_i = sum_k w_k * a_{i-k} for k such that both are valid
    # Use FFT for efficiency
    s = 1
    while s < len(w) + n - 1:

    fw = np.fft.rfft(w, s)
    fa = np.fft.rfft(a, s)
    conv_wa = np.fft.irfft(fw * fa, s)[:n]
    # grad_h_i = 2 * beta * conv_wa_i
    # dx_i = a_i * (grad_h_i - dot(a, grad_h))
    grad_h = 2.0 * beta * conv_wa
    dot = np.dot(a, grad_h)

"""
Minimizes J(a) = 2*n*max(convolve(a,a))/(sum(a))^2 using:
1. Efficient coordinate descent with incremental convolution updates (O(n^2) per cycle)
2. L-BFGS-B gradient refinement with softmax parameterization and log-sum-exp objective
3. LP-based mirror descent for structured updates
4. Multiple sequence lengths and diverse initializations

Key improvement: coordinate descent with O(n) incremental b updates allows
thousands of refinement cycles, achieving much better local optima than LP alone.
"""

import numpy as np
from scipy.optimize import minimize, linprog
import time


def fast_conv(a):
    """Full self-convolution via FFT, O(n log n)."""
    n = len(a)
    s = 1
    while s < 2 * n - 1:
        s <<= 1
    fa = np.fft.rfft(a, s)
    return np.fft.irfft(fa * fa, s)[:2 * n - 1]


def softmax_fn(x):
    """Numerically stable softmax."""
    x_shift = x - np.max(x)
    e = np.exp(x_shift)
    return e / e.sum()


def soft_obj_vals(x, beta):
    """Compute log-sum-exp objective value for softmax parameterization."""
    a = softmax_fn(x)
    n = len(a)
    b = fast_conv(a)
    bb = beta * b
    max_bb = np.max(bb)
    return float(max_bb + np.log(np.sum(np.exp(bb - max_bb))))


def soft_obj_grad(x, beta):
    """Compute gradient of log-sum-exp objective w.r.t. x (softmax param)."""
    a = softmax_fn(x)
    n = len(a)
    b = fast_conv(a)
    bb = beta * b
    max_bb = np.max(bb)
    exp_bb = np.exp(bb - max_bb)
    sum_exp = np.sum(exp_bb)
    w = exp_bb / sum_exp
    # (w * a) truncated to length n via FFT
    s = 1
    while s < len(w) + n - 1:
        s <<= 1
    fw = np.fft.rfft(w, s)
    fa = np.fft.rfft(a, s)
    conv_wa = np.fft.irfft(fw * fa, s)[:n]
    grad_h = 2.0 * beta * conv_wa
    dot = np.dot(a, grad_h)
    grad = a * (grad_h - dot)
    return grad


def soft_obj_wrapper(x, beta):
    """Wrapper for scipy minimize: returns (objective, gradient)."""
    a = softmax_fn(x)
    n = len(a)
    b = fast_conv(a)
    bb = beta * b
    max_bb = np.max(bb)
    obj = float(max_bb + np.log(np.sum(np.exp(bb - max_bb))))
    bb2 = bb - max_bb
    exp_bb = np.exp(bb2)
    sum_exp = np.sum(exp_bb)
    w = exp_bb / sum_exp
    s = 1
    while s < len(w) + n - 1:
        s <<= 1
    fw = np.fft.rfft(w, s)
    fa = np.fft.rfft(a, s)
    conv_wa = np.fft.irfft(fw * fa, s)[:n]
    grad_h = 2.0 * beta * conv_wa
    dot = np.dot(a, grad_h)
    grad = a * (grad_h - dot)
    return obj, grad


def build_A_matrix(a):
    """Build Toeplitz-like constraint matrix for LP."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for k in range(2 * n - 1):
        jl = max(0, k - n + 1)
        jh = min(n - 1, k)
        if jl <= jh:
            A[k, jl:jh + 1] = a[k - jh:k - jl + 1][::-1]
    return A


def lp_best_response(a, M):
    """LP: max sum(g) s.t. (a*g)_k <= M, g >= 0. Returns g or None."""
    n = len(a)
    c = -np.ones(n)
    A = build_A_matrix(a)
    r = linprog(c, A_ub=A, b_ub=np.full(2 * n - 1, M),
                bounds=[(0, None)] * n, method='highs')
    if r.success and r.x.sum() > 1e-10:
        return r.x
    return None

            # Try a few candidates
            candidates = [d_opt, 0.3 * d_opt, -0.001, 0.001, -0.01, 0.01]
            # Remove duplicates and invalid
            tried = set()
            for d in candidates:
                key = round(d, 8)
                if key in tried:
                    continue
                tried.add(key)
                if a[i] + d < 1e-15:
                    continue
                new_S = S + d
                if new_S < 1e-10:
                    continue

                # Update b incrementally: b[i:i+n] += 2*d*a, b[2*i] += d^2
                old_seg = b[i:i + n].copy()
                b[i:i + n] += 2.0 * d * a
                b[2 * i] += d * d

                new_M = float(np.max(b))
                new_J = 2.0 * n * new_M / (new_S * new_S)

                if new_J < J * (1.0 - 1e-12):
                    # Accept
                    a[i] += d
                    S = new_S
                    M = new_M
                    J = new_J
                    k_peak = int(np.argmax(b))
                    improved = True
                    if J < best_J:
                        best_J = J
                        best_a = a.copy()
                    break
                else:
                    # Rollback
                    b[i:i + n] = old_seg
            # End candidate loop
        # End coordinate loop

        if not improved:
            stall += 1
        else:
            stall = 0

        # Periodic recompute to avoid drift
        if cycle > 0 and cycle % 5000 == 0:
            b = fast_conv(a)
            M = float(np.max(b))
            k_peak = int(np.argmax(b))
            J = 2.0 * n * M / (S * S)

    return best_a


def gradient_refine(a, deadline=None):
    """L-BFGS-B refinement with softmax parameterization and beta annealing."""
    n = len(a)
    a = a.copy().astype(np.float64)
    S = a.sum()
    if S < 1e-10:
        return a
    a = a / S
    # Initialize x in log-space
    x = np.log(np.maximum(a, 1e-15))
    x = x - np.mean(x)  # center for numerical stability

    best_score = float(evaluate_sequence(a.tolist()))
    best_a = a.copy()

    for beta in [1.0, 3.0, 10.0, 30.0, 100.0, 300.0]:
        if deadline is not None and time.monotonic() > deadline:
            break

        def _obj(x_inner):
            a_inner = softmax_fn(x_inner)
            b_inner = fast_conv(a_inner)
            bb = beta * b_inner
            max_bb = np.max(bb)
            return float(max_bb + np.log(np.sum(np.exp(bb - max_bb))))

        def _grad(x_inner):
            a_inner = softmax_fn(x_inner)
            b_inner = fast_conv(a_inner)
            bb = beta * b_inner
            max_bb = np.max(bb)
            exp_bb = np.exp(bb - max_bb)
            w = exp_bb / np.sum(exp_bb)
            s = 1
            while s < (2 * n - 1) + n - 1:
                s <<= 1
            fw = np.fft.rfft(w, s)
            fa = np.fft.rfft(a_inner, s)
            conv_wa = np.fft.irfft(fw * fa, s)[:n]
            grad_h = 2.0 * beta * conv_wa
            dot = np.dot(a_inner, grad_h)
            return a_inner * (grad_h - dot)

        try:
            res = minimize(_obj, x, jac=_grad, method='L-BFGS-B',
                           options={'maxiter': 300, 'max_fun': 5000,
                                    'ftol': 1e-16, 'disp': False})
            if res.fun < 1e10:
                a_new = softmax_fn(res.x)
                score = float(evaluate_sequence(a_new.tolist()))
                if np.isfinite(score) and score < best_score:
                    best_score = score
                    best_a = a_new.copy()
                    x = res.x.copy()
        except Exception:
            pass

    return best_a


def lp_mirror_descent(a, max_iters=80, deadline=None, verbose=False):
    """LP-based mirror descent (best-response iterations)."""
    n = len(a)
    a = a.copy().astype(np.float64)
    a = np.clip(a, 1e-6, None)
    a = a / a.sum()
    best_score = float(evaluate_sequence(a.tolist()))
    best_a = a.copy()

    for it in range(max_iters):
        if deadline is not None and time.monotonic() > deadline:
            break
        b = fast_conv(a)
        M = float(b.max())
        g = lp_best_response(a, M)
        if g is None:
            break
        g = np.clip(g, 0, None)
        g_sum = g.sum()
        if g_sum < 1e-10:
            break
        g = g / g_sum
        t_mix = max(0.05, 0.5 * (1.0 - it / max_iters))
        a_new = (1.0 - t_mix) * a + t_mix * g
        a_new = np.clip(a_new, 1e-8, None)
        a_new = a_new / a_new.sum()
        a = a_new
        score = float(evaluate_sequence(a.tolist()))
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_a = a.copy()

    return best_a


def propose_candidate(seed=42, budget_s=1500, **kwargs):
    """
    Main search function. Combines coordinate descent, gradient refinement,
    and LP mirror descent to minimize the convolution peak ratio.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + float(budget_s) - 15.0
    best = None
    best_score = float('inf')

    def try_update(a):
        nonlocal best, best_score
        a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
        if a.sum() < 0.01:
            return
        score = float(evaluate_sequence(a.tolist()))
        if np.isfinite(score) and score < best_score:
            best_score = score
            best = a.copy()
            print(f"  NEW BEST: J={score:.6f} (n={a.size})", flush=True)

    # Try parent solution
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        if p.ndim == 1 and p.size > 0 and np.all(np.isfinite(p)):
            try_update(p)
            print(f"  Parent: J={best_score:.6f} (n={p.size})", flush=True)
    except (NameError, AttributeError):
        pass

    # Generate initializations for various lengths
    def make_inits(n, rng):
        inits = []
        # Uniform
        inits.append(np.ones(n) / n)
        # Cosine
        t = np.arange(n) / n
        inits.append(np.maximum(1.0 + 0.3 * np.cos(4 * np.pi * t), 0.01))
        # Quadratic
        inits.append(np.maximum(1.0 + 0.5 * (t - 0.5) ** 2 * 4, 0.01))
        # Chirp
        inits.append(1.0 + 0.3 * np.cos(2 * np.pi * t ** 2 * 3))
        # Random
        inits.append(rng.uniform(0.1, 1.0, n))
        # Asymmetric (inspired by small-n analysis: one small, rest equal)
        arr = np.full(n, 1.0)
        arr[0] = 0.1
        inits.append(arr)
        arr2 = np.full(n, 1.0)
        arr2[-1] = 0.1
        inits.append(arr2)
        # Normalize all
        inits = [np.maximum(x, 0.01) for x in inits]
        inits = [x / x.sum() for x in inits]
        return inits

    # === Phase 1: Coordinate descent on multiple inits and lengths ===
    lengths = [512]
    if time.monotonic() < deadline - 300:
        lengths.append(1024)
    if time.monotonic() < deadline - 500:
        lengths.append(256)

    for n in lengths:
        if time.monotonic() > deadline:
            break
        print(f"\n=== Coord descent: n={n} ===", flush=True)
        t0 = time.monotonic()

        inits = make_inits(n, rng)

        # If we have a best at this length, add it
        if best is not None and best.size == n:
            inits.insert(0, best.copy() / best.sum())
        elif best is not None:
            # Resample best to this length
            t_old = np.linspace(0, 1, best.size)
            t_new = np.linspace(0, 1, n)
            resampled = np.maximum(np.interp(t_new, t_old, best), 0.01)
            inits.insert(0, resampled / resampled.sum())

        # limit cycles based on time budget remaining
        time_per_n = min(300, max(60, (deadline - time.monotonic()) / len(lengths)))
        max_cycles = int(time_per_n / (n * n * 0.000005))  # rough estimate
        max_cycles = min(max(max_cycles, 1000), 8000)

        for init_i, a0 in enumerate(inits):
            if time.monotonic() > deadline:
                break
            if time.monotonic() > deadline - 30:
                break
            sub_deadline = min(deadline - 5, time.monotonic() + time_per_n / len(inits) - 2)
            a_result = coord_descent(a0, max_cycles=max_cycles, rng=rng,
                                     deadline=sub_deadline, verbose=False)
            try_update(a_result)

        print(f"  Coord n={n}: {time.monotonic()-t0:.1f}s, best={best_score:.6f}", flush=True)

    # === Phase 2: LP mirror descent on best solution ===
    if best is not None and time.monotonic() < deadline - 60:
        n_best = best.size
        if n_best <= 1024:
            print(f"\n=== LP mirror descent: n={n_best} ===", flush=True)
            t0 = time.monotonic()
            a_lp = lp_mirror_descent(best, max_iters=60, deadline=min(deadline - 10, time.monotonic() + 120))
            try_update(a_lp)
            # Also run coord descent on the LP result
            if time.monotonic() < deadline - 30:
                a_cd = coord_descent(a_lp, max_cycles=3000, rng=rng,
                                     deadline=min(deadline - 5, time.monotonic() + 60))
                try_update(a_cd)
            print(f"  LP: {time.monotonic()-t0:.1f}s, best={best_score:.6f}", flush=True)

    # === Phase 3: Gradient refinement on best ===
    if best is not None and time.monotonic() < deadline - 30:
        n_best = best.size
        print(f"\n=== Gradient refinement: n={n_best} ===", flush=True)
        t0 = time.monotonic()
        a_grad = gradient_refine(best, deadline=min(deadline - 5, time.monotonic() + 120))
        try_update(a_grad)
        # Follow with coord descent
        if time.monotonic() < deadline - 15:
            a_cd2 = coord_descent(a_grad, max_cycles=3000, rng=rng,
                                  deadline=min(deadline - 3, time.monotonic() + 60))
            try_update(a_cd2)
        print(f"  Grad: {time.monotonic()-t0:.1f}s, best={best_score:.6f}", flush=True)

    # === Phase 4: Extended coordinate descent (if time remains) ===
    if best is not None and time.monotonic() < deadline - 10:
        n_best = best.size
        remaining = deadline - time.monotonic() - 5
        if remaining > 30:
            print(f"\n=== Extended coord descent: n={n_best} (budget {remaining:.0f}s) ===", flush=True)
            t0 = time.monotonic()
            max_ext = int(remaining / (n_best * n_best * 0.000003))
            max_ext = min(max(max_ext, 2000), 15000)
            a_ext = coord_descent(best, max_cycles=max_ext, rng=rng,
                                  deadline=deadline - 5)
            try_update(a_ext)
            print(f"  Extended: {time.monotonic()-t0:.1f}s, best={best_score:.6f}", flush=True)

    # === Phase 5: Final polish - alternate coord descent and gradient ===
    if best is not None and time.monotonic() < deadline - 5:
        remaining = deadline - time.monotonic() - 3
        if remaining > 20:
            print(f"\n=== Final polish (budget {remaining:.0f}s) ===", flush=True)
            n_best = best.size
            a_cur = best.copy()
            for round_i in range(3):
                if time.monotonic() > deadline - 3:
                    break
                # Gradient step
                if time.monotonic() < deadline - 15 and n_best <= 1024:
                    a_cur = gradient_refine(a_cur, deadline=min(deadline - 3, time.monotonic() + remaining / 3))
                    try_update(a_cur)
                # Coord descent step
                if time.monotonic() < deadline - 3:
                    sub_dead = min(deadline - 2, time.monotonic() + remaining / 3)
                    a_cur = coord_descent(a_cur, max_cycles=5000, rng=rng, deadline=sub_dead)
                    try_update(a_cur)
            print(f"  Final: best={best_score:.6f}", flush=True)

    if best is None:
        best = np.ones(512) / 512.0

    print(f"\nFINAL: J={best_score:.6f}, n={best.size}", flush=True)
    return [float(x) for x in best]
