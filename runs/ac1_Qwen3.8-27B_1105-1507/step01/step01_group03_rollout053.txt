def project_simplex(v):
    """Project vector v onto the probability simplex (non-negative, sum=1)."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - 1))[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    w = np.maximum(v - theta, 0)
    return w

def make_conv_template(n):
    """
    Precompute the CSR matrix structure for convolve(f, g).
    """
    m = 2 * n - 1
    k64 = np.arange(m, dtype=np.int64)
    js = np.maximum(0, k64 - n + 1).astype(np.int64)
    je = np.minimum(n - 1, k64).astype(np.int64)
    counts = (je - js + 1).astype(np.int64)
    total = int(counts.sum())

    cum = np.cumsum(counts)
    starts = np.repeat(cum - counts, counts)
    local = np.arange(total, dtype=np.int64) - starts

    row = np.repeat(k64, counts).astype(np.int64)
    col_base = js[row]
    col = (col_base + local)
    offsets = (row - col).astype(np.int64)

    A_off = csr_matrix((offsets, (row, col)), shape=(m, n), dtype=np.int64)
    edge_offsets = A_off.data.astype(np.int64, copy=True)
    A_template = A_off.astype(np.float64).copy()
    A_template.data.fill(0.0)

    c_obj = -np.ones(n, dtype=np.float64)
    bounds = [(0.0, None)] * n
    return A_template, edge_offsets, c_obj, bounds, m


def solve_direction_lp(template, f, n, M):
    """Solve the LP direction problem."""
    A_template, edge_offsets, c_obj, bounds, m = template
    
    A = A_template.copy()
    A.data[:] = f[edge_offsets]
    # Scale for numerical stability
    A.data *= float(n)
    
    b = np.full(m, M * float(n) * float(n), dtype=np.float64)
    res = linprog(c_obj, A_ub=A, b_ub=b, bounds=bounds, method="highs")
    return res


def exact_line_search(f, g, M, n, max_iter=50):
    """
    Find the maximum step t in [0, 1] such that the score at (1-t)*f + t*g
    (normalized) is less than the score at f.
    """
    best_t = 0.0
    best_score = score_normalized(f, n)
    
    if best_score <= 0 or not np.isfinite(best_score):
        return 0.0, best_score
    
    lo = 0.0
    hi = 1.0
    
    for _ in range(max_iter):
        mid = (lo + hi) / 2.0
        f_new = (1.0 - mid) * f + mid * g
        f_new = np.maximum(f_new, 0.0)
        s = float(f_new.sum())
        if s <= 1e-300:
            hi = mid
            continue
        f_new /= s
        sc = score_normalized(f_new, n)
        if np.isfinite(sc) and sc < best_score:
            best_t = mid
            best_score = sc
            lo = mid
        else:
            hi = mid
    
    return best_t, best_score

def mix_noisy(base, rng, eps):
    """Mix a base vector with a random Dirichlet vector."""
    f = normalize_sequence(base)
    if f is None:
        return None
    n = f.size
    if eps <= 1e-12:
        return f
    try:
        mix = rng.dirichlet(np.ones(n, dtype=np.float64))
    except Exception:
        mix = np.ones(n, dtype=np.float64) / float(n)
    if eps >= 1.0:
        f = mix
    else:
        f = (1.0 - eps) * f + eps * mix
    return normalize_sequence(f)


def interpolate_sequence(x, n):
    """Resample a nonnegative sequence to length n and normalize it."""
    x = normalize_sequence(x)
    if x is None or n <= 0:
        return None
    m = x.size
    if m == n:
        return x
    if m == 1:
        return np.ones(n, dtype=np.float64) / float(n)
    old = np.linspace(0.0, 1.0, m, dtype=np.float64)
    new = np.linspace(0.0, 1.0, n, dtype=np.float64)
    out = np.interp(new, old, x)
    return normalize_sequence(out)


def choose_starts(n, rng, best_f, given_f):
    """Choose diverse starting sequences."""
    starts = []
    
    if best_f is not None:
        up = interpolate_sequence(best_f, n)
        if up is not None:
            up = mix_noisy(up, rng, 0.03)
            starts.append(up)
    
    uniform = np.ones(n, dtype=np.float64) / float(n)
    starts.append(uniform)
    
    d = rng.dirichlet(np.ones(n, dtype=np.float64))
    starts.append(d)
    
    if given_f is not None:
        g = interpolate_sequence(given_f, n)
        if g is not None:
            g = mix_noisy(g, rng, 0.05)
            starts.append(g)
    
    # Try a "bump" sequence (concentrated in the middle)
    bump = np.zeros(n)
    center = n // 2
    width = n // 4
    for i in range(max(0, center - width), min(n, center + width)):
        bump[i] = 1.0 - abs(i - center) / width
    bump = normalize_sequence(bump)
    if bump is not None:
        starts.append(bump)
    
    # Try a "U-shaped" sequence
    u_shape = np.zeros(n)
    for i in range(n):
        t = i / max(1, n - 1)
        u_shape[i] = t * t + (1 - t) * (1 - t)
    u_shape = normalize_sequence(u_shape)
    if u_shape is not None:
        starts.append(u_shape)
    
    # Deduplicate
    out = []
    for f in starts:
        f = normalize_sequence(f)
        if f is None or f.size != n:
            continue
        dup = False
        for g in out:
            if float(np.linalg.norm(f - g)) < 1e-6:
                dup = True
                break
        if not dup:
            out.append(f)
        if len(out) >= 5:
            break
    
    if not out:
        out.append(np.ones(n, dtype=np.float64) / float(n))
    return out
