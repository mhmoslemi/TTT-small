"""
Hybrid optimization for minimizing 2*n*max(conv(a,a))/(sum(a))^2.
Combines LP alternation with line search, smooth gradient descent,
and coordinate descent. Uses FFT for fast convolution.
"""

import numpy as np
from scipy.optimize import linprog
from scipy.fft import fft, ifft
import time


def convolve_fast(a, b=None):
    """Fast convolution using FFT."""
    if b is None:
        b = a
    n = len(a)
    m = len(b)
    size = n + m - 1
    fa = fft(a, n=size)
    fb = fft(b, n=size)
    return np.real(ifft(fa * fb))


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    conv = convolve_fast(a)
    max_b = float(np.max(conv))
    sum_a = float(np.sum(a))
    if sum_a < 0.01:
        return np.inf
    return 2.0 * n * max_b / (sum_a * sum_a)


def build_conv_matrix(a):
    """Build the convolution matrix A where (A @ g)_k = (a*g)_k."""
    n = len(a)
    A = np.zeros((2 * n - 1, n), dtype=np.float64)
    for k in range(2 * n - 1):
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        idx = np.arange(j_start, j_end + 1)
        A[k, idx] = a[k - idx]
    return A


def solve_lp(a, M=None, upper_bound=1000.0):
    """Solve LP: max sum(g) s.t. (a*g)_k <= M, 0 <= g <= upper_bound."""
    n = len(a)
    if M is None:
        conv = convolve_fast(a)
        M = float(np.max(conv))
    
    A = build_conv_matrix(a)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, upper_bound)] * n
    res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
    if res.status == 0:
        return res.x, M, res
    return None, M, res

"""
Hybrid optimization: LP alternation + subgradient + coordinate descent.
Uses FFT for fast convolution. Aims to beat 1.5027.
"""

import numpy as np
from scipy.optimize import linprog
from scipy.fft import fft, ifft
import time


def convolve_fast(a, b=None):
    if b is None:
        b = a
    n = len(a)
    m = len(b)
    size = n + m - 1
    fa = fft(a, n=size)
    fb = fft(b, n=size)
    return np.real(ifft(fa * fb))


def compute_obj(a):
    n = len(a)
    if n == 0:
        return np.inf
    conv = convolve_fast(a)
    max_b = float(np.max(conv))
    sum_a = float(np.sum(a))
    if sum_a < 0.01:
        return np.inf
    return 2.0 * n * max_b / (sum_a * sum_a)


def build_conv_matrix(a):
    n = len(a)
    A = np.zeros((2 * n - 1, n), dtype=np.float64)
    for k in range(2 * n - 1):
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        idx = np.arange(j_start, j_end + 1)
        A[k, idx] = a[k - idx]
    return A


def solve_lp(a, M=None, ub=1000.0):
    n = len(a)
    if M is None:
        conv = convolve_fast(a)
        M = float(np.max(conv))
    A = build_conv_matrix(a)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, ub)] * n
    res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
    if res.status == 0:
        return res.x, M
    return None, M


def subgradient_step(a, n_iter=1):
    """One step of subgradient descent on max(conv(a,a))."""
    n = len(a)
    conv = convolve_fast(a)
    max_val = np.max(conv)
    # Active set: positions where conv is close to max
    tolerance = 0.01 * max_val
    active = np.where(conv >= max_val - tolerance)[0]
    
    # Subgradient: for each j, grad[j] = 2 * sum_{k in active} a[k-j] (valid indices)
    grad = np.zeros(n)
    for k in active:
        for j in range(n):
            idx = k - j
            if 0 <= idx < n:
                grad[j] += 2.0 * a[idx]
    
    return grad, active, max_val


def coordinate_descent(a, max_iter=50000, lr=0.05, deadline=None):
    """Coordinate descent: optimize one element at a time."""
    n = len(a)
    a = a.copy()
    best_obj = compute_obj(a)
    best_a = a.copy()
    rng = np.random.default_rng(42)
    
    for it in range(max_iter):
        if deadline and time.monotonic() > deadline:
            break
        
        j = int(rng.integers(0, n))
        # Try different values for a[j]
        # The objective as a function of a[j] (with others fixed) is:
        # 2n * max_k (conv_k) / (sum)^2
        # conv_k = old_conv_k + a[j]*(a[k-j] + a[k-j] - a[j]) ... 
        # Actually: new_conv_k = old_conv_k - a[j]*a[k-j] + new_val*a[k-j] (for the cross terms)
        # Plus: if k = 2j, there's an a[j]^2 term
        # This is complex. Let's just try a few values.
        
        old_val = a[j]
        if old_val < 1e-15:
            # Try small positive values
            candidates = [0.001, 0.01, 0.05, 0.1, 0.5]
        else:
            candidates = [old_val * 0.5, old_val * 0.8, old_val * 1.2, old_val * 1.5, 0.0]
        
        best_cand = old_val
        best_cand_obj = best_obj
        
        for cv in candidates:
            cv = max(0.0, cv)
            a[j] = cv
            obj = compute_obj(a)
            if obj < best_cand_obj:
                best_cand_obj = obj
                best_cand = cv
        
        a[j] = best_cand
        
        if best_cand_obj < best_obj - 1e-12:
            best_obj = best_cand_obj
            best_a = a.copy()
    
    return best_a, best_obj


def lp_with_linesearch(a, max_iter=500, deadline=None, verbose=False):
    """LP alternation with adaptive line search."""
    n = len(a)
    a = a / np.sum(a)
    best_a = a.copy()
    best_obj = compute_obj(best_a)
    
    stall = 0
    step_scales = [1.0, 0.7, 0.5, 0.3, 0.2, 0.1, 0.05]
    
    for it in range(max_iter):
        if deadline and time.monotonic() > deadline:
            break
        
        g, M = solve_lp(a)
        if g is None:
            break
        
        g_sum = np.sum(g)
        if g_sum < 1e-10:
            break
        g = g / g_sum
        
        # Line search
        best_t = 0.0
        best_new_obj = best_obj
        a_new = a.copy()
        
        for scale in step_scales:
            a_t = (1 - scale) * a + scale * g
            a_t = np.clip(a_t, 0, None)
            s = np.sum(a_t)
            if s < 1e-10:

            a_t = a_t / s
            obj_t = compute_obj(a_t)
            if obj_t < best_new_obj:
                best_new_obj = obj_t
                best_t = scale
                a_new = a_t.copy()

            best_a = a.copy()
            best_obj = best_new_obj
            stall = 0

                if s < 1e-10:
                    a = best_a.copy()
                    s = np.sum(a)

    return best_a, best_obj


def subgradient_descent(a, max_iter=20000, deadline=None):
    """Subgradient descent on the (non-smooth) objective."""
    n = len(a)
    a = a / np.sum(a)
    best_a = a.copy()
    best_obj = compute_obj(best_a)
    
    # Adaptive learning rate
    step = 0.01
    patience = 0
    
    for it in range(max_iter):
        if deadline and time.monotonic() > deadline:
            break
        
        conv = convolve_fast(a)
        max_val = float(np.max(conv))
        
        # Active set (positions near max)
        tol = 0.05 * max_val
        active = np.where(conv >= max_val - tol)[0]
        
        # Subgradient
        grad = np.zeros(n)
        for k in active:
            j_min = max(0, k - n + 1)
            j_max = min(n - 1, k)
            idx = np.arange(j_min, j_max + 1)
            grad[idx] += 2.0 * a[k - idx]
        
        # Normalize gradient
        grad_norm = np.linalg.norm(grad)
        if grad_norm < 1e-15:

        # Projected gradient step on simplex
        a_new = a - step * grad / grad_norm
        a_new = np.clip(a_new, 0, None)
        s = np.sum(a_new)
        if s < 1e-10:
            a_new = a.copy()
            s = 1.0
        a_new = a_new / s
        
        obj_new = compute_obj(a_new)
        
        if obj_new < best_obj:
            best_a = a_new.copy()
            best_obj = obj_new
            a = a_new
            patience = 0
            # Decrease step slowly
            step = max(0.001, step * 0.999)
        else:
            patience += 1
            if patience > 500:
                step = max(0.0005, step * 0.5)
                patience = 0
        
        # Every 500 iterations, try to restart step size
        if it % 1000 == 999:
            step = max(0.005, step * 0.7)
    
    return best_a, best_obj

Hybrid optimization for minimizing 2*n*max(conv(a,a))/(sum(a))^2.
Combines LP alternation (with variations), FFT-accelerated subgradient

import numpy as np
from scipy.optimize import linprog
from scipy.fft import fft, ifft
import time


def convolve_fast(a, b=None):
    """Fast convolution using FFT."""
    if b is None:
        b = a
    n = len(a)
    m = len(b)
    size = n + m - 1
    fa = fft(a, n=size)
    fb = fft(b, n=size)
    return np.real(ifft(fa * fb))


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    conv = convolve_fast(a)
    max_b = float(np.max(conv))
    sum_a = float(np.sum(a))
    if sum_a < 0.01:
        return np.inf
    return 2.0 * n * max_b / (sum_a * sum_a)


def build_conv_matrix(a):
    """Build convolution matrix A where (A @ g)_k = (a*g)_k."""
    n = len(a)
    A = np.zeros((2 * n - 1, n), dtype=np.float64)
    for k in range(2 * n - 1):
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        idx = np.arange(j_start, j_end + 1)
        A[k, idx] = a[k - idx]
    return A
