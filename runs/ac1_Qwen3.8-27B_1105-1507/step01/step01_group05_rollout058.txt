from scipy.optimize import linprog

def compute_score(a):
    """Compute 2*n*max(conv(a,a))/sum(a)^2"""
    n = len(a)
    s = a.sum()
    if s < 1e-15:
        return np.inf
    conv = np.convolve(a, a)
    return 2.0 * n * conv.max() / (s * s)

"""
Hybrid coordinate descent + LP optimization for minimizing 2*n*max(conv(a,a))/sum(a)^2.
Uses exact per-coordinate optimization via subgradient on the 1D subproblem,
combined with occasional LP-based moves to escape local optima.
"""

import numpy as np
from scipy.optimize import linprog
import time


def compute_score(a):
    """Compute 2*n*max(conv(a,a))/sum(a)^2"""
    n = len(a)
    s = a.sum()
    if s < 1e-15:
        return np.inf
    conv = np.convolve(a, a)
    return 2.0 * n * conv.max() / (s * s)


def build_conv_matrix(f):
    """Build matrix A such that A @ b = np.convolve(f, b, mode='full')."""
    n = len(f)
    A = np.zeros((2 * n - 1, n))
    for k in range(2 * n - 1):
        j_start = max(0, k - n + 1)
        j_end = min(n, k + 1)
        for j in range(j_start, j_end):
            A[k, j] = f[k - j]
    return A


def coord_descend(a, n_sweeps=500, time_limit=None, t_start=None, verbose=False, subgrad_iters=40):
    """
    Coordinate descent: for each coordinate, minimize the 1D subproblem using
    subgradient descent on f(x) = max_k g_k(x) / h(x).
    """
    n = len(a)
    a = a.astype(np.float64).copy()
    best_a = a.copy()
    best_score = compute_score(a)

    if time_limit is not None:
        t_start = t_start or time.monotonic()

    for sweep in range(n_sweeps):
        if time_limit is not None and time.monotonic() - t_start > time_limit:
            break

        for i in range(n):
            x0 = a[i]
            S0 = a.sum() - x0

            # Compute quadratic coefficients A[k]*x^2 + B[k]*x + C[k] for conv[k]
            a[i] = 0.0
            conv0 = np.convolve(a, a)
            a[i] = 1.0
            conv1 = np.convolve(a, a)
            a[i] = 2.0
            conv2 = np.convolve(a, a)
            a[i] = x0

            Ak = (conv2 - 2.0 * conv1 + conv0) * 0.5
            Bk = conv1 - conv0 - Ak
            Ck = conv0

            # Subgradient descent on f(x) = max_k (Ak*x^2 + Bk*x + Ck) / (S0+x)^2
            x = x0 if x0 > 0 else 1.0 / n
            lr = 0.5

            for it in range(subgrad_iters):
                g = Ak * x * x + Bk * x + Ck
                h = (S0 + x) ** 2
                ratio = g / h
                k_star = np.argmax(ratio)
                f_val = ratio[k_star]

                # Subgradient
                g_k = g[k_star]
                gk_prime = 2.0 * Ak[k_star] * x + Bk[k_star]
                h_prime = 2.0 * (S0 + x)
                f_prime = (g_k * h_prime - h * gk_prime) / (h * h)

                # Ensure we're not going negative
                if f_prime > 0 and x < f_prime * lr:
                    x_new = 0.0
                else:
                    x_new = x - lr * f_prime
                    x_new = max(0.0, x_new)

                # Adaptive learning rate
                if x_new == x0 and it > 5:
                    lr *= 0.7
                if abs(x_new - x) < 1e-14 * max(1.0, x):
                    x = x_new
                    break
                x = x_new

            a[i] = x

        new_score = compute_score(a)
        if new_score < best_score:
            best_score = new_score
            best_a = a.copy()
        elif new_score > best_score * 1.01:
            # Revert if we got much worse
            a = best_a.copy()

        if verbose and sweep % 20 == 0:
            print(f"    CD sweep {sweep}: score = {best_score:.8f}")

    return best_a, best_score


def lp_step(f, M=None, verbose=False):
    """
    One LP step: find direction g that maximizes sum(b) subject to conv(f,b) <= M, b >= 0.
    Returns the direction g (normalized to sum=1).
    """
    n = len(f)
    if M is None:
        conv = np.convolve(f, f)
        M = conv.max()

    A = build_conv_matrix(f)
    c_obj = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n

    try:
        res = linprog(c_obj, A_ub=A, b_ub=b_ub, bounds=bounds,
                      method='highs', options={'presolve': True, 'time_limit': 10})
    except Exception:
        return None

    if not res.success or res.x is None:
        return None

    b = np.clip(res.x, 0, None)
    b_sum = b.sum()
    if b_sum < 1e-15:
        return None

    return b / b_sum


def lp_optimize(f, time_budget, t0=0.5, verbose=False, rng=None):
    """Run LP-based optimization within a time budget."""
    if rng is None:
        rng = np.random.default_rng(42)

    n = len(f)
    f = f.astype(np.float64).copy()
    S0 = f.sum()
    if S0 < 1e-10:
        return f, np.inf
    f = f / S0

    best_f = f.copy()
    best_score = compute_score(f)
    t = t0
    start = time.monotonic()
    it = 0
    stall = 0

    while time.monotonic() - start < time_budget:
        conv = np.convolve(f, f)
        M = conv.max()

        g = lp_step(f, M=M)
        if g is None:
            break

        f_new = (1 - t) * f + t * g
        f_new = np.clip(f_new, 0, None)
        S_new = f_new.sum()
        if S_new < 1e-10:
            t *= 0.5
            continue
        f_new = f_new / S_new

        new_score = compute_score(f_new)

        if new_score < best_score - 1e-12:
            best_f = f_new.copy()
            best_score = new_score
            f = f_new
            stall = 0
            if verbose and it % 10 == 0:
                print(f"    LP it={it}, score={best_score:.8f}")
        else:
            t *= 0.7
            stall += 1
            if stall > 30:
                # Random perturbation
                f_candidate = f + rng.normal(0, 0.02, n)
                f_candidate = np.clip(f_candidate, 0, None)
                S_c = f_candidate.sum()
                if S_c > 1e-10:
                    f_candidate /= S_c
                    sc = compute_score(f_candidate)
                    if sc < best_score:
                        f = f_candidate
                        best_f = f.copy()
                        best_score = sc
                        t = t0 * 0.5
                        stall = 0
                if time.monotonic() - start > time_budget * 0.85:
                    break

        it += 1

    return best_f, best_score


def local_polish(f, rng, n_iter=3000):
    """Random local search to polish."""
    n = len(f)
    f = f.copy()
    S = f.sum()
    f = f / S
    best = f.copy()
    best_score = compute_score(f)

    for _ in range(n_iter):
        f_c = f.copy()
        width = max(1, int(rng.integers(1, max(2, n // 16))))
        start = int(rng.integers(0, n - width + 1))
        scale = np.exp(rng.normal(0, 0.25))
        f_c[start:start + width] *= scale
        f_c = np.clip(f_c, 0, None)
        S_new = f_c.sum()
        if S_new < 1e-10:
            continue
        f_c = f_c / S_new
        score = compute_score(f_c)
        if score < best_score:
            best = f_c.copy()
            best_score = score
            f = f_c

    return best, best_score
