```python
"""
Optimizes non-negative sequence to minimize 2*n*max(conv(a,a))/sum(a)^2.
Uses Nesterov-accelerated projected subgradient descent with FFT convolution,
multiple structured initializations, dynamic active set, and coordinate descent polish.
"""

import numpy as np
import time


def project_simplex(v):
    """Project vector v onto {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cv = np.cumsum(u) - 1.0
    rho_candidates = np.where(u * np.arange(1, n + 1) > cv)[0]
    if len(rho_candidates) == 0:
        return np.ones(n) / n
    rho = rho_candidates[-1]
    theta = cv[rho] / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def fast_convolve(a):
    """Full linear convolution of a with itself using FFT."""
    n = len(a)
    size = 2 * n - 1
    N = 1 << (size - 1).bit_length()
    fa = np.fft.rfft(a, N)
    fc = fa * fa
    result = np.fft.irfft(fc, N)[:size]
    result[result < 0] = 0.0  # Clean numerical artifacts
    return result


def compute_subgradient(a, peak_indices):
    """Subgradient of max_k (a*a)[k] averaged over active peaks."""
    n = len(a)
    grad = np.zeros(n)
    for k in peak_indices:
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        if lo <= hi:
            # a[k-hi : k-lo+1] reversed gives a[k-j] for j in [lo, hi]
            grad[lo:hi+1] += a[k-hi:k-lo+1][::-1] + a[lo:hi+1]
    grad /= len(peak_indices)
    return grad


def initraised_cosine(n, rng):
    """Raised cosine initialization."""
    x = np.linspace(0, np.pi, n)
    a = 0.5 * (1 + np.cos(x))
    a = a + 1e-6  # Ensure positivity
    return a / a.sum()


def init_uniform(n):
    """Uniform initialization."""
    return np.ones(n) / n


def init_truncated_cosine(n, rng, alpha=1.0):
    """Raised cosine with tunable width."""
    x = np.linspace(-1, 1, n)
    a = np.cos(np.pi * x / 2 * alpha)**2
    a = a + 1e-6
    return a / a.sum()


def init_bell(n, rng, sigma=0.3):
    """Gaussian bell initialization."""
    x = np.linspace(0, 1, n)
    center = 0.5
    a = np.exp(-0.5 * ((x - center) / sigma)**2)
    a = a + 1e-6
    return a / a.sum()


def init_random_structured(n, rng):
    """Random but structured: sum of a few cosines."""
    a = np.ones(n) / n
    for _ in range(3):
        freq = rng.integers(1, n // 4 + 1)
        phase = rng.uniform(0, 2 * np.pi)
        amplitude = rng.uniform(0.1, 0.5)
        x = np.linspace(0, 1, n)
        a += amplitude * np.cos(2 * np.pi * freq * x + phase)
    a = np.maximum(a, 0)
    if a.sum() < 1e-10:
        return np.ones(n) / n
    return a / a.sum()


def init_from_parent(n, parent):
    """Initialize from a parent sequence (resize if needed)."""
    if len(parent) >= n:
        a = parent[:n].copy()
    else:
        # Upsample
        from scipy.interpolate import interp1d
        x_old = np.linspace(0, 1, len(parent))
        x_new = np.linspace(0, 1, n)
        f = interp1d(x_old, parent, kind='cubic', fill_value='extrapolate')
        a = f(x_new)
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-10:
        return np.ones(n) / n
    return a / s


def fista_optimize(a, deadline, rng, alpha=None, momentum_decay=0.99, 
                   active_set_size=10, restart_every=2000):
    """
    Nesterov-accelerated projected subgradient descent.
    Minimizes max(conv(a,a)) subject to a >= 0, sum(a) = 1.
    """
    n = len(a)
    a = project_simplex(a.copy().astype(np.float64))
    
    # Compute initial objective
    c = fast_convolve(a)
    best_peak = np.max(c)
    best_a = a.copy()
    
    if alpha is None:
        alpha = 1.0 / n  # Initial step size
    
    # Nesterov variables
    y = a.copy()
    t_momentum = 1.0
    
    it = 0
    stall_count = 0
    max_stall = 500
    alpha_min = 1e-5 / n
    alpha_max = 10.0 / n
    
    while time.time() < deadline:
        it += 1
        
        # Compute gradient at y (Nesterov point)
        c_y = fast_convolve(y)
        
        # Active set: top peaks
        peak = np.max(c_y)
        if active_set_size < len(c_y):
            top_k = np.argsort(c_y)[::-1][:active_set_size]
            # Only include those close to the peak
            threshold = peak * 0.95
            peak_indices = top_k[c_y[top_k] >= threshold]
            if len(peak_indices) == 0:
                peak_indices = top_k[:3]
        else:
            peak_indices = np.where(c_y >= peak * 0.95)[0]
        
        grad = compute_subgradient(y, peak_indices)
        
        # Gradient step
        a_new = y - alpha * grad
        a_new = project_simplex(a_new)
        
        # Nesterov update
        t_new = (1 + np.sqrt(1 + 4 * t_momentum**2)) / 2
        y = a_new + (t_momentum - 1) / t_new * (a_new - a)
        y = np.maximum(y, 0)  # Keep y in the feasible region roughly
        
        a = a_new
        t_momentum = t_new
        
        # Evaluate
        c = fast_convolve(a)
        peak_val = np.max(c)
        
        if peak_val < best_peak * (1 - 1e-10):
            best_peak = peak_val
            best_a = a.copy()
            stall_count = 0
        else:
            stall_count += 1
        
        # Adaptive step size
        if stall_count > max_stall:
            alpha = max(alpha * 0.5, alpha_min)
            stall_count = 0
            if alpha < alpha_min * 5:
                # Reset Nesterov and add perturbation
                y = best_a + rng.normal(0, 0.01 / np.sqrt(n), n)
                y = project_simplex(y)
                t_momentum = 1.0
                alpha = 1.0 / n
                stall_count = 0
        
        # Periodic restart
        if it % restart_every == 0 and time.time() < deadline - 60:
            # Try a few random restarts
            for _ in range(2):
                a_try = best_a + rng.normal(0, 0.05 / np.sqrt(n), n)
                a_try = project_simplex(a_try)
                c_try = fast_convolve(a_try)
                if np.max(c_try) < best_peak:
                    a = a_try
                    y = a_try.copy()
                    t_momentum = 1.0
                    break
    
    return best_a


def coordinate_descent_polish(a, deadline, max_sweeps=50):
    """
    Coordinate descent: optimize one coordinate at a time.
    For coordinate j, (a*a)[k] as a function of a[j] is:
    - Linear in a[j] for k != 2j
    - Quadratic in a[j] for k = 2j
    We minimize the max over k.
    """
    n = len(a)
    a = a.copy()
    
    for sweep in range(max_sweeps):
        if time.time() > deadline:
            break
        
        improved = False
        for j in range(n):
            # Save current a[j]
            old_val = a[j]
            a[j] = 0.0
            
            # Compute base convolution (without j's contribution)
            c_base = fast_convolve(a)
            
            # Now (a_full*a_full)[k] = c_base[k] + 2*a[j]*a_full[k-j] + a[j]^2*[k=2j]
            # where a_full[k-j] = a[k-j] if k-j != j, else 0 (since a[j]=0 in base)
            # Actually, a_full[k-j] = original a[k-j] for k-j != j
            
            # Let's recompute: with a[j] = v, 
            # conv[k] = c_base[k] + v * (a_orig[k-j] + a_orig[k-j]) + v^2 * [k=2j]
            # Wait, I need to be more careful.
            
            # Let a_orig be the sequence with a[j] = old_val.
            # a_current has a[j] = 0.
            # conv(a_current, a_current)[k] = c_base[k]
            # conv(a_orig, a_orig)[k] = c_base[k] + 2*v*a_orig[k-j] + v^2*[k=2j]
            # where a_orig[k-j] is the value at position k-j in a_orig (which is a[k-j] for k-j != j)
            
            # But we want to find the best v (and then re-sum to 1).
            # Actually, for coordinate descent on the simplex, we should:
            # 1. Set a[j] = 0
            # 2. Find the best value for a[j] to minimize max(conv)
            # 3. But this changes the sum! So we need to be more careful.
            
            # Simpler approach: just try a few values of a[j] and pick the best.
            # Try: 0, old_val * 0.5, old_val, old_val * 1.5, and the optimal for k=2j.
            
            a[j] = old_val  # Restore
            break  # This approach is too slow for large n. Let me use a simpler method.
        
        if not improved:
            break
    
    return a
```

```python
"""
Optimizes non-negative sequence to minimize 2*n*max(conv(a,a))/sum(a)^2.
Uses Nesterov-accelerated projected subgradient descent with FFT convolution,
structured initializations, dynamic active set management, multi-length search,
and multi-phase time allocation.
"""

import numpy as np
import time


def project_simplex(v):
    """Project vector v onto {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cv = np.cumsum(u) - 1.0
    rho_candidates = np.where(u * np.arange(1, n + 1) > cv)[0]
    if len(rho_candidates) == 0:
        return np.ones(n) / n
    rho = rho_candidates[-1]
    theta = cv[rho] / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def fast_convolve(a):
    """Full linear convolution of a with itself using FFT."""
    n = len(a)
    size = 2 * n - 1
    N = 1 << (size - 1).bit_length()
    fa = np.fft.rfft(a, N)
    fc = fa * fa
    result = np.fft.irfft(fc, N)[:size]
    result[result < 0] = 0.0
    return result


def compute_subgradient(a, peak_indices):
    """Subgradient of max_k (a*a)[k] averaged over active peak indices."""
    n = len(a)
    grad = np.zeros(n)
    for k in peak_indices:
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        if lo <= hi:
            grad[lo:hi + 1] += a[k - hi:k - lo + 1][::-1] + a[lo:hi + 1]
    grad /= len(peak_indices)
    return grad


def init_raised_cosine(n):
    """Raised cosine window initialization."""
    x = np.linspace(0, np.pi, n)
    a = 0.5 * (1.0 + np.cos(x))
    a = a + 1e-8
    return a / a.sum()


def init_uniform(n):
    """Uniform initialization."""
    return np.ones(n) / n


def init_gaussian(n, sigma=0.25):
    """Gaussian bell initialization."""
    x = np.linspace(0, 1, n)
    a = np.exp(-0.5 * ((x - 0.5) / sigma) ** 2)
    a = a + 1e-8
    return a / a.sum()


def init_triangle(n):
    """Triangular (Bartlett) window initialization."""
    x = np.arange(n)
    mid = (n - 1) / 2.0
    a = 1.0 - np.abs(x - mid) / (mid + 1)
    a = a + 1e-8
    return a / a.sum()


def init_random_structured(n, rng):
    """Random structured: sum of low-frequency cosines."""
    x = np.linspace(0, 1, n)
    a = np.ones(n)
    for _ in range(4):
        freq = rng.integers(1, max(2, n // 16))
        phase = rng.uniform(0, 2 * np.pi)
        amp = rng.uniform(0.2, 0.6)
        a += amp * np.cos(2 * np.pi * freq * x + phase)
    a = np.maximum(a, 0.0)
    if a.sum() < 1e-10:
        return np.ones(n) / n
    return a / a.sum()


def init_from_parent(n, parent):
    """Initialize from a parent sequence, resampling to length n."""
    parent = np.asarray(parent, dtype=np.float64)
    parent = np.clip(parent, 0.0, None)
    if parent.sum() < 1e-10:
        return np.ones(n) / n
    parent = parent / parent.sum()
    if len(parent) == n:
        return parent.copy()
    x_old = np.linspace(0.0, 1.0, len(parent))
    x_new = np.linspace(0.0, 1.0, n)
    a = np.interp(x_new, x_old, parent)
    a = np.maximum(a, 0.0)
    if a.sum() < 1e-10:
        return np.ones(n) / n
    return a / a.sum()


def gerchberg_saxton_init(n, rng, n_iter=40):
    """Gerchberg-Saxton flat-spectrum initialization."""
    N = 1 << (2 * n - 1).bit_length()
    a = init_raised_cosine(n)
    for _ in range(n_iter):
        F = np.fft.rfft(a, N)
        # Target: flat magnitude (with slight low-freq emphasis for nonneg)
        mag = np.ones(len(F))
        freqs = np.linspace(0, 1, len(F))
        mag *= 1.0 - 0.1 * freqs ** 0.5
        F_new = mag * np.exp(1j * np.angle(F))
        F_new[0] = F_new[0].real + 0j
        a_full = np.fft.irfft(F_new, N)
        a = np.maximum(a_full[:n], 0.0)
        if a.sum() < 1e-10:
            a = init_raised_cosine(n)
        else:
            a = a / a.sum()
    return a


def fista_optimize(a, deadline, rng, init_alpha=None, max_active=12,
                   stall_limit=400, verbose=False):
    """
    Nesterov-accelerated projected subgradient descent.
    Minimizes max(conv(a,a)) s.t. a >= 0, sum(a) = 1.
    """
    n = len(a)
    a = project_simplex(a.copy().astype(np.float64))

    c = fast_convolve(a)
    best_peak = np.max(c)
    best_a = a.copy()

    if init_alpha is None:
        init_alpha = 1.5 / np.sqrt(n)
    alpha = init_alpha
    alpha_min = 1e-6 / np.sqrt(n)
    alpha_max = 5.0 / np.sqrt(n)

    y = a.copy()
    t_mom = 1.0
    it = 0
    stall_count = 0
    best_stall_alpha = alpha

    while time.time() < deadline:
        it += 1

        # Gradient at Nesterov point y
        c_y = fast_convolve(y)
        peak_y = np.max(c_y)

        # Dynamic active set
        if max_active < len(c_y):
            sorted_idx = np.argsort(c_y)[::-1]
            threshold = peak_y * 0.9
            candidates = sorted_idx[:max_active * 3]
            peak_indices = candidates[c_y[candidates] >= threshold]
            if len(peak_indices) < 2:
                peak_indices = sorted_idx[:max(2, max_active // 3)]
        else:
            peak_indices = np.where(c_y >= peak_y * 0.9)[0]

        grad = compute_subgradient(y, peak_indices)

        # Gradient step with projection
        a_new = y - alpha * grad
        a_new = project_simplex(a_new)

        # Nesterov momentum update
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t_mom * t_mom)) / 2.0
        y = a_new + ((t_mom - 1.0) / t_new) * (a_new - a)
        y = np.maximum(y, 0.0)

        a = a_new
        t_mom = t_new

        # Evaluate
        c = fast_convolve(a)
        peak_val = np.max(c)

        if peak_val < best_peak * (1.0 - 1e-12):
            best_peak = peak_val
            best_a = a.copy()
            stall_count = 0
        else:
            stall_count += 1

        # Adaptive step size based on stalls
        if stall_count >= stall_limit:
            alpha *= 0.5
            if alpha < alpha_min:
                # Major restart
                y = best_a.copy()
                a = best_a.copy()
                t_mom = 1.0
                alpha = best_stall_alpha
                # Perturb
                noise_scale = 0.03 / np.sqrt(n)
                a = project_simplex(a + rng.normal(0, noise_scale, n))
                y = a.copy()
            stall_count = 0
        elif stall_count >= stall_limit // 2:
            alpha = max(alpha * 0.7, alpha_min)
            stall_count = 0

        # If we've been making progress, slowly try larger steps
        if stall_count == 0 and alpha < init_alpha:
            alpha = min(alpha * 1.02, init_alpha)

    return best_a


def multistart_optimize(n, rng, deadline, parent=None, n_starts=3):
    """Try multiple initializations and return the best."""
    best_a = None
    best_peak = float('inf')

    init_types = []
    if parent is not None:
        init_types.append(('parent', init_from_parent(n, parent)))
    init_types.append(('cosine', init_raised_cosine(n)))
    init_types.append(('gaussian', init_gaussian(n, sigma=0.25)))
    init_types.append(('gs', gerchberg_saxton_init(n, rng, n_iter=30)))
    init_types.append(('random', init_random_structured(n, rng)))

    # Shuffle non-parent inits
    fixed = [x for x in init_types if x[0] == 'parent']
    rest = [x for x in init_types if x[0] != 'parent']
    rng_shuffled = np.random.default_rng(rng.integers(0, 2**31))
    rng_shuffled.shuffle(rest)
    init_types = fixed + rest

    time_per = max(5.0, (deadline - time.time()) / (len(init_types) + 0.5))

    for name, a_init in init_types:
        if time.time() > deadline - 5:
            break
        sub_deadline = min(deadline - 5, time.time() + time_per)
        a_opt = fista_optimize(a_init, sub_deadline, rng)
        c = fast_convolve(a_opt)
        peak = np.max(c)
        if peak < best_peak:
            best_peak = peak
            best_a = a_opt.copy()
            if verbose:
                print(f"  [{name}] peak={peak:.8f}")

    return best_a
```
