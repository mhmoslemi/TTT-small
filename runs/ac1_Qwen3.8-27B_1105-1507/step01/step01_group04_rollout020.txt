Optimizes non-negative step function heights to minimize 2n*max(conv(f,f))/(sum(f))^2.
Uses multi-start multiplicative mirror descent with averaging, adaptive step sizes,
diverse initializations across multiple sequence lengths, and coordinate descent polishing.

import numpy as np
import time

def _fft_convolve_full(f):
    """Compute full linear convolution of f with itself using FFT."""
    n = len(f)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    F = np.fft.rfft(f, size)
    h = np.fft.irfft(F * F, size)
    return h[:2 * n - 1]

def _score_raw(f):
    """Fast score computation assuming f >= 0, sum(f) > 0.01."""
    n = len(f)
    h = _fft_convolve_full(f)
    M = float(np.max(h))
    S = float(np.sum(f))
    if S < 0.01:
        return float('inf')
    return 2.0 * n * M / (S * S)

def _normalize(f):
    """Normalize f to sum to 1, clip negatives."""
    f = np.maximum(f, 0.0)
    s = float(np.sum(f))
    if s < 1e-30:
        return None
    return f / s

def _pressure_vector(f, active_lags):
    """
    Compute p[j] = (1/|K|) * sum_{k in K} 2*f[k-j] for valid 0<=k-j<n.
    This is the subgradient of max_k (f*f)_k, normalized by |K|.
    """
    n = len(f)
    p = np.zeros(n, dtype=np.float64)
    for k in active_lags:
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        if j_end >= j_start:
            lo = k - j_end
            hi = k - j_start
            p[j_start:j_end + 1] += 2.0 * f[lo:hi + 1][::-1]
    p /= float(len(active_lags))
    return p

    Returns (best_f, best_score).

    if f is None:

    best_f = f.copy()
    best_score = _score_raw(f)
    start_time = time.monotonic()
    
    # For averaging
    avg_f = np.zeros(n, dtype=np.float64)
    avg_count = 0
    best_avg_score = float('inf')
    best_avg_f = f.copy()
    
    for it in range(n_iter):
        if time.monotonic() - start_time > time_limit:
            break
        
        h = _fft_convolve_full(f)
        M = float(np.max(h))
        if M <= 1e-30:
            break

        active = np.where(h >= threshold)[0]
        if len(active) == 0:
            active = np.array([int(np.argmax(h))], dtype=np.int64)
        
        p = _pressure_vector(f, active)

        decay = 0.9995
        eta = eta_scale * (decay ** (it // 100)) / (M + 1e-20)

            avg_count += 1
            if avg_count >= avg_window and avg_count % avg_window == 0:

def _coord_descent_polish(f_init, time_limit=10.0, step_size=0.01):
    """
    Coordinate descent: try increasing/decreasing each coordinate slightly.
    """
    f = np.asarray(f_init, dtype=np.float64).copy()
    f = _normalize(f)
    if f is None:
        return f_init.copy(), float('inf')
    n = len(f)
    best_score = _score_raw(f)
    start_time = time.monotonic()
    
    steps = [step_size * 0.25, step_size * 0.5, step_size, step_size * 2, step_size * 4]
    
    improved = True
    sweep = 0
    while improved and (time.monotonic() - start_time) < time_limit:
        improved = False
        sweep += 1
        # Decay step size
        scale = max(0.1, 1.0 / sweep)
        for j in range(n):
            if (time.monotonic() - start_time) > time_limit:
                break
            for s in steps:
                ds = s * scale
                # Try increasing
                f_new = f.copy()
                f_new[j] += ds
                f_new = _normalize(f_new)
                if f_new is not None:
                    sc = _score_raw(f_new)
                    if sc < best_score:
                        best_score = sc
                        f = f_new
                        improved = True
                # Try decreasing
                f_new = f.copy()
                f_new[j] = max(0, f_new[j] - ds)
                f_new = _normalize(f_new)
                if f_new is not None:
                    sc = _score_raw(f_new)
                    if sc < best_score:
                        best_score = sc
                        f = f_new
                        improved = True
    
    return f, best_score

def _additive_pgd(f_init, n_iter, eta0, time_limit=30.0):
    """Additive projected gradient descent with subgradient of max(f*f)."""
    f = np.asarray(f_init, dtype=np.float64).copy()
    f = _normalize(f)
    if f is None:
        return f_init.copy(), float('inf')
    n = len(f)
    best_f = f.copy()
    best_score = _score_raw(f)
    start_time = time.monotonic()
    
    # Averaging
    avg_f = np.zeros(n, dtype=np.float64)
    avg_count = 0
    best_avg_score = float('inf')
    best_avg_f = f.copy()
    
    for it in range(n_iter):
        if time.monotonic() - start_time > time_limit:
            break
        
        h = _fft_convolve_full(f)
        M = float(np.max(h))
        if M <= 1e-30:
            break
        
        threshold = 0.8 * M + 1e-15
        active = np.where(h >= threshold)[0]
        if len(active) == 0:
            active = np.array([int(np.argmax(h))], dtype=np.int64)
        
        p = _pressure_vector(f, active)
        
        eta = eta0 / (it + 10) ** 0.5
        f = f - eta * p
        f = np.maximum(f, 0)
        f_norm = _normalize(f)
        if f_norm is None:
            break
        f = f_norm
        
        score = _score_raw(f)
        if score < best_score:
            best_score = score
            best_f = f.copy()
        
        avg_f += f
        avg_count += 1
        if avg_count % 50 == 0:
            f_avg = avg_f / avg_count
            f_avg = _normalize(f_avg)
            if f_avg is not None:
                avg_score = _score_raw(f_avg)
                if avg_score < best_avg_score:
                    best_avg_score = avg_score
                    best_avg_f = f_avg.copy()
    
    if best_avg_score < best_score:
        best_f = best_avg_f
        best_score = best_avg_score
    
    return best_f, best_score

def _init_uniform(n):
    return np.ones(n) / n

def _init_cosine(n):
    j = np.arange(n, dtype=np.float64)
    f = np.cos(np.pi * (j + 0.5) / n) ** 2
    f = np.maximum(f, 0.0)
    return _normalize(f)

def _init_gaussian(n, center_frac, sigma_frac):
    j = np.arange(n, dtype=np.float64)
    center = center_frac * (n - 1)
    sigma = sigma_frac * n
    f = np.exp(-(j - center) ** 2 / (2 * sigma * sigma))
    return _normalize(f)

def _init_exponential(n, rate):
    j = np.arange(n, dtype=np.float64)
    f = np.exp(-rate * j / n)
    return _normalize(f)

def _init_triangle(n):
    j = np.arange(n, dtype=np.float64)
    f = 1.0 - np.abs(2.0 * (j + 0.5) / n - 1.0)
    f = np.maximum(f, 0.0)
    return _normalize(f)

def _init_two_hump(n, sep=0.25, width=0.2):
    j = np.arange(n, dtype=np.float64)
    c1 = (0.5 - sep) * n
    c2 = (0.5 + sep) * n
    sig = width * n
    f = np.exp(-(j - c1)**2 / (2*sig*sig)) + np.exp(-(j - c2)**2 / (2*sig*sig))
    return _normalize(f)

def _init_random_smooth(n, rng, n_frequencies=3):
    j = np.arange(n, dtype=np.float64)
    f = np.ones(n)
    for _ in range(n_frequencies):
        freq = rng.integers(1, 6)
        phase = rng.uniform(0, 2 * np.pi)
        amp = rng.uniform(0.2, 0.6)
        f += amp * np.sin(2 * np.pi * freq * j / n + phase)
    f = np.maximum(f, 0.0)
    s = np.sum(f)
    if s < 1e-10:
        return np.ones(n) / n
    return f / s

def _init_sqrt_tail(n):
    """Initialize with a square-root decay towards the edges."""
    j = np.arange(n, dtype=np.float64)
    x = (j + 0.5) / n
    f = np.sqrt(np.sin(np.pi * x))
    f = np.maximum(f, 0.0)
    return _normalize(f)

        mean_mag = np.mean(mag)
        if mean_mag > 0:
            mag_new = np.power(mag, 1.0 - beta) * np.power(mean_mag, beta)
        else:
            mag_new = mag
        
        # Reconstruct
        F_new = mag_new * np.exp(1j * phase)
        f_new = np.fft.irfft(F_new, n)
        
        # Project to non-negative, sum=1
        f_new = np.maximum(f_new.real, 0.0)
        f_norm = _normalize(f_new)
        if f_norm is None:
            break
        f = f_norm
        
        score = _score_raw(f)
        if score < best_score:
            best_score = score
            best_f = f.copy()
    
    return best_f, best_score

"""
Optimizes non-negative step function heights to minimize 2n*max(conv(f,f))/(sum(f))^2.
Uses multi-start multiplicative mirror descent with iterate averaging, flat-spectrum
alternating projection, additive PGD, diverse initializations across lengths,
and coordinate descent polishing.
"""

import numpy as np
import time
import operator


def _fft_convolve_full(f):
    """Compute full linear convolution of f with itself using FFT."""
    n = len(f)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    F = np.fft.rfft(f, size)
    h = np.fft.irfft(F * F, size)
    return h[:2 * n - 1]


def _score_raw(f):
    """Fast score computation assuming f >= 0, sum(f) > 0.01."""
    n = len(f)
    h = _fft_convolve_full(f)
    M = float(np.max(h))
    S = float(np.sum(f))
    if S < 0.01:
        return float('inf')
    return 2.0 * n * M / (S * S)


def _normalize(f):
    """Normalize f to sum to 1, clip negatives. Returns None if degenerate."""
    f = np.maximum(f, 0.0)
    s = float(np.sum(f))
    if s < 1e-30:
        return None
    return f / s


def _pressure_vector(f, active_lags):
    """Subgradient of max_k (f*f)_k averaged over active lags."""
    n = len(f)
    p = np.zeros(n, dtype=np.float64)
    for k in active_lags:
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        if j_end >= j_start:
            lo = k - j_end
            hi = k - j_start
            p[j_start:j_end + 1] += 2.0 * f[lo:hi + 1][::-1]
    p /= float(len(active_lags))
    return p


def _mirror_descent(f_init, n_iter, eta_scale, threshold_frac, time_limit=30.0,
                    use_averaging=True, avg_window=50, decay_rate=0.999):
    """Multiplicative mirror descent with adaptive step size and averaging."""
    f = np.asarray(f_init, dtype=np.float64).copy()
    f = _normalize(f)
    if f is None:
        return f_init.copy(), float('inf')
    n = len(f)
    best_f = f.copy()
    best_score = _score_raw(f)
    start_time = time.monotonic()

    avg_f = np.zeros(n, dtype=np.float64)
    avg_count = 0
    best_avg_score = float('inf')
    best_avg_f = f.copy()

    for it in range(n_iter):
        if time.monotonic() - start_time > time_limit:
            break

        h = _fft_convolve_full(f)
        M = float(np.max(h))
        if M <= 1e-30:
            break

        threshold = threshold_frac * M + 1e-15
        active = np.where(h >= threshold)[0]
        if len(active) == 0:
            active = np.array([int(np.argmax(h))], dtype=np.int64)

        p = _pressure_vector(f, active)

        eta = eta_scale * (decay_rate ** (it // 200)) / (M + 1e-20)
        log_ratio = -eta * p
        log_ratio = np.clip(log_ratio, -30.0, 30.0)

        f = f * np.exp(log_ratio)
        f_norm = _normalize(f)
        if f_norm is None:
            f = np.ones(n) / n
        else:
            f = f_norm

        score = _score_raw(f)
        if score < best_score:
            best_score = score
            best_f = f.copy()

        if use_averaging:
            avg_f += f
            avg_count += 1
            if avg_count >= avg_window and avg_count % avg_window == 0:
                f_avg = avg_f / avg_count
                f_avg = _normalize(f_avg)
                if f_avg is not None:
                    avg_score = _score_raw(f_avg)
                    if avg_score < best_avg_score:
                        best_avg_score = avg_score
                        best_avg_f = f_avg.copy()

        if (it + 1) % 5000 == 0:
            print("      MD it=%d: best=%.6f, avg_best=%.6f" % (it + 1, best_score, best_avg_score))

    if best_avg_score < best_score:
        best_f = best_avg_f
        best_score = best_avg_score

    return best_f, best_score


def _flat_spectrum(f_init, n_iter, beta, time_limit=30.0):
    """Alternating projection: enforce non-negativity in time, flatten spectrum in freq."""
    f = np.asarray(f_init, dtype=np.float64).copy()
    f = _normalize(f)
    if f is None:
        return f_init.copy(), float('inf')
    n = len(f)
    best_f = f.copy()
    best_score = _score_raw(f)
    start_time = time.monotonic()

    # Averaging
    avg_f = np.zeros(n, dtype=np.float64)
    avg_count = 0
    best_avg_score = float('inf')
    best_avg_f = f.copy()

    for it in range(n_iter):
        if time.monotonic() - start_time > time_limit:
            break

        F = np.fft.rfft(f, n)
        mag = np.abs(F)
        phase = np.angle(F)

        mean_mag = np.mean(mag)
        if mean_mag > 1e-20:
            mag_new = np.power(np.maximum(mag, 1e-20), 1.0 - beta) * np.power(mean_mag, beta)
        else:
            break

        F_new = mag_new * np.exp(1j * phase)
        f_new = np.fft.irfft(F_new, n)
        f_new = np.maximum(f_new.real, 0.0)
        f_norm = _normalize(f_new)
        if f_norm is None:
            break
        f = f_norm

        score = _score_raw(f)
        if score < best_score:
            best_score = score
            best_f = f.copy()

        avg_f += f
        avg_count += 1
        if avg_count % 50 == 0:
            f_avg = avg_f / avg_count
            f_avg = _normalize(f_avg)
            if f_avg is not None:
                avg_score = _score_raw(f_avg)
                if avg_score < best_avg_score:
                    best_avg_score = avg_score
                    best_avg_f = f_avg.copy()

    if best_avg_score < best_score:
        best_f = best_avg_f
        best_score = best_avg_score

    return best_f, best_score


def _additive_pgd(f_init, n_iter, eta0, time_limit=30.0):
    """Additive projected gradient descent with subgradient of max(f*f)."""
    f = np.asarray(f_init, dtype=np.float64).copy()
    f = _normalize(f)
    if f is None:
        return f_init.copy(), float('inf')
    n = len(f)
    best_f = f.copy()
    best_score = _score_raw(f)
    start_time = time.monotonic()

    avg_f = np.zeros(n, dtype=np.float64)
    avg_count = 0
    best_avg_score = float('inf')
    best_avg_f = f.copy()

    for it in range(n_iter):
        if time.monotonic() - start_time > time_limit:
            break

        h = _fft_convolve_full(f)
        M = float(np.max(h))
        if M <= 1e-30:
            break

        threshold = 0.8 * M + 1e-15
        active = np.where(h >= threshold)[0]
        if len(active) == 0:
            active = np.array([int(np.argmax(h))], dtype=np.int64)

        p = _pressure_vector(f, active)

        eta = eta0 / ((it + 10) ** 0.5)
        f = f - eta * p
        f = np.maximum(f, 0)
        f_norm = _normalize(f)
        if f_norm is None:
            break
        f = f_norm

        score = _score_raw(f)
        if score < best_score:
            best_score = score
            best_f = f.copy()

        avg_f += f
        avg_count += 1
        if avg_count % 50 == 0:
            f_avg = avg_f / avg_count
            f_avg = _normalize(f_avg)
            if f_avg is not None:
                avg_score = _score_raw(f_avg)
                if avg_score < best_avg_score:
                    best_avg_score = avg_score
                    best_avg_f = f_avg.copy()

    if best_avg_score < best_score:
        best_f = best_avg_f
        best_score = best_avg_score

    return best_f, best_score


def _coord_descent_polish(f_init, time_limit=10.0, base_step=0.015):
    """Coordinate descent: try small perturbations of individual coordinates."""
    f = np.asarray(f_init, dtype=np.float64).copy()
    f = _normalize(f)
    if f is None:
        return f_init.copy(), float('inf')
    n = len(f)
    best_score = _score_raw(f)
    start_time = time.monotonic()

    multipliers = [0.25, 0.5, 1.0, 2.0, 4.0]
    improved = True
    sweep = 0
    while improved and (time.monotonic() - start_time) < time_limit:
        if not improved:
            break
        improved = False
        sweep += 1
        scale = max(0.05, 1.0 / (1.0 + 0.3 * sweep))
        for j in range(n):
            if (time.monotonic() - start_time) > time_limit:
                break
            for mult in multipliers:
                ds = base_step * mult * scale
                # Try increasing
                f_new = f.copy()
                f_new[j] += ds
                f_new = _normalize(f_new)
                if f_new is not None:
                    sc = _score_raw(f_new)
                    if sc < best_score:
                        best_score = sc
                        f = f_new
                        improved = True
                # Try decreasing
                f_new = f.copy()
                f_new[j] = max(0.0, f_new[j] - ds)
                f_new = _normalize(f_new)
                if f_new is not None:
                    sc = _score_raw(f_new)
                    if sc < best_score:
                        best_score = sc
                        f = f_new
                        improved = True

    return f, best_score


def _init_uniform(n):
    return np.ones(n) / n


def _init_cosine(n):
    j = np.arange(n, dtype=np.float64)
    f = np.cos(np.pi * (j + 0.5) / n) ** 2
    f = np.maximum(f, 0.0)
    return _normalize(f)


def _init_gaussian(n, center_frac, sigma_frac):
    j = np.arange(n, dtype=np.float64)
    center = center_frac * (n - 1)
    sigma = sigma_frac * n
    f = np.exp(-(j - center) ** 2 / (2.0 * sigma * sigma))
    return _normalize(f)


def _init_exponential(n, rate):
    j = np.arange(n, dtype=np.float64)
    f = np.exp(-rate * j / n)
    return _normalize(f)


def _init_triangle(n):
    j = np.arange(n, dtype=np.float64)
    f = 1.0 - np.abs(2.0 * (j + 0.5) / n - 1.0)
    f = np.maximum(f, 0.0)
    return _normalize(f)


def _init_two_hump(n, sep, width):
    j = np.arange(n, dtype=np.float64)
    c1 = (0.5 - sep) * n
    c2 = (0.5 + sep) * n
    sig = width * n
    f = np.exp(-((j - c1) ** 2) / (2.0 * sig * sig)) + np.exp(-((j - c2) ** 2) / (2.0 * sig * sig))
    return _normalize(f)


def _init_random_smooth(n, rng, n_freq):
    j = np.arange(n, dtype=np.float64)
    f = np.ones(n)
    for _ in range(n_freq):
        freq = rng.integers(1, 7)
        phase = rng.uniform(0.0, 2.0 * np.pi)
        amp = rng.uniform(0.2, 0.6)
        f += amp * np.sin(2.0 * np.pi * freq * j / n + phase)
    f = np.maximum(f, 0.0)
    s = np.sum(f)
    if s < 1e-10:
        return np.ones(n) / n
    return f / s


def _init_sqrt_tail(n):
    j = np.arange(n, dtype=np.float64)
    x = (j + 0.5) / n
    f = np.sqrt(np.sin(np.pi * x))
    f = np.maximum(f, 0.0)
    return _normalize(f)


def _init_tilted(n, alpha):
    """Uniform with a linear tilt."""
    j = np.arange(n, dtype=np.float64)
    f = 1.0 + alpha * (2.0 * j / n - 1.0)
    f = np.maximum(f, 0.0)
    return _normalize(f)


def _sort_key_score(x):
    return x[0]
