def fast_convolve(a):

    size = 2 * n - 1

    fc = fa * fa

def project_simplex(v):
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cv = np.cumsum(u) - 1.0
    rho_candidates = np.where(u * np.arange(1, n + 1) > cv)[0]
    if len(rho_candidates) == 0:
        return np.ones(n) / n
    rho = rho_candidates[-1]
    theta = cv[rho] / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def smooth_objective_and_grad(a, alpha):
    """Compute smooth approximation of max(conv) and its gradient."""
    n = len(a)
    conv = fast_convolve(a)
    # Numerical stability: subtract max
    c_max = np.max(conv)
    shifted = conv - c_max
    log_sum_exp = c_max + np.log(np.sum(np.exp(alpha * shifted)))
    obj = log_sum_exp / alpha
    p = np.exp(alpha * shifted)
    p /= p.sum()
    # Gradient: grad[j] = 2 * sum_m a[m] * p[j+m]
    rows = np.arange(n)[:, None] + np.arange(n)[None, :]
    M = p[rows]
    grad = 2.0 * (M @ a)
    return obj, grad, p, conv

class Adam:
    def __init__(self, dim, lr=0.01, beta1=0.9, beta2=0.999, eps=1e-8):
        self.m = np.zeros(dim)
        self.v = np.zeros(dim)
        self.t = 0
        self.lr = lr
        self.beta1 = beta1
        self.beta2 = beta2
        self.eps = eps
    
    def step(self, grad):
        self.t += 1
        self.m = self.beta1 * self.m + (1 - self.beta1) * grad
        self.v = self.beta2 * self.v + (1 - self.beta2) * (grad ** 2)
        m_hat = self.m / (1 - self.beta1 ** self.t)
        v_hat = self.v / (1 - self.beta2 ** self.t)
        return self.lr * m_hat / (np.sqrt(v_hat) + self.eps)

"""
Optimizes non-negative sequences to minimize 2*n*max(conv(a,a))/sum(a)^2.
Uses smoothed-objective ADAM with adaptive alpha, LP refinement steps,
alternating projection initialization, and multi-scale search.
"""

import numpy as np
import time
from scipy.optimize import linprog


def fast_convolve(a):
    """Full linear self-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    N = 1 << (size - 1).bit_length()
    fa = np.fft.rfft(a, N)
    fc = fa * fa
    result = np.fft.irfft(fc, N)[:size]
    result[result < 0] = 0.0
    return result


def project_simplex(v):
    """Project onto {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cv = np.cumsum(u) - 1.0
    rho_candidates = np.where(u * np.arange(1, n + 1) > cv)[0]
    if len(rho_candidates) == 0:
        return np.ones(n) / n
    rho = rho_candidates[-1]
    theta = cv[rho] / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def smooth_grad(a, alpha, idx):
    """Compute gradient of smooth max approximation."""
    n = len(a)
    conv = fast_convolve(a)
    c_max = np.max(conv)
    shifted = alpha * (conv - c_max)
    # Clip for numerical stability
    shifted = np.clip(shifted, -50, 50)
    exp_shifted = np.exp(shifted)
    p = exp_shifted / np.sum(exp_shifted)
    # grad[j] = 2 * sum_m a[m] * p[j+m]
    P = p[idx]  # (n, n) indexed matrix
    grad = 2.0 * (P @ a)
    return grad, c_max, p


def lp_improve(a, peak_value, time_limit=5.0):
    """LP step: max sum(b) s.t. conv(a,b) <= peak_value, b >= 0."""
    n = len(a)
    if n > 2000:
        return None
    # Build A matrix: A[k, j] = a[k-j] if valid
    A = np.zeros((2 * n - 1, n), dtype=np.float64)
    for k in range(2 * n - 1):
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        A[k, lo:hi + 1] = a[k - hi:k - lo + 1][::-1]
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, peak_value)
    try:
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=[(0.0, None)] * n,
                      method='highs', options={'time_limit': time_limit})
        if res.success:
            b = res.x
            if b.sum() > 1e-8:
                return b * (a.sum() / b.sum())
    except Exception:
        pass
    return None


def alt_projection_init(n, rng, n_iter=100):
    """Alternating projection for flat-spectrum non-negative sequences."""
    N = 1 << (2 * n - 2).bit_length()
    phases = rng.uniform(0.0, 2.0 * np.pi, N // 2 + 1)
    F = np.exp(1j * phases)
    F[0] = 1.0 + 0j
    a = np.ones(n) / n
    for _ in range(n_iter):
        a_full = np.fft.irfft(F, N)
        a = np.maximum(a_full[:n], 0.0)
        s = a.sum()
        if s < 1e-10:
            a = np.ones(n) / n
        else:
            a = a / s
        F_full = np.fft.rfft(a, N)
        # Flat magnitude target
        F = np.ones(len(F_full)) * np.exp(1j * np.angle(F_full))
        F[0] = F[0].real + 0j
    return a

    a = project_simplex(a.copy())

    best_score = np.max(fast_convolve(a))  # with sum=1, score = 2n*max_conv

    v = np.zeros(n)

    alpha = 5.0

    it = 0
    stall = 0
    max_stall = 3000
    last_lp_time = 0

"""
Optimizes non-negative sequences to minimize 2*n*max(conv(a,a))/sum(a)^2.
Uses smoothed max-objective with ADAM, adaptive alpha annealing, LP refinement,
alternating projection initialization, and multi-start search with diverse configs.
"""

import numpy as np
import time
from scipy.optimize import linprog


def fast_convolve(a):
    """Full linear self-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    N = 1 << (size - 1).bit_length()
    fa = np.fft.rfft(a, N)
    fc = fa * fa
    result = np.fft.irfft(fc, N)[:size]
    result[result < 0] = 0.0
    return result


def project_simplex(v):
    """Project vector onto probability simplex {x>=0, sum(x)=1}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cv = np.cumsum(u) - 1.0
    rho_candidates = np.where(u * np.arange(1, n + 1) > cv)[0]
    if len(rho_candidates) == 0:
        return np.ones(n) / n
    rho = rho_candidates[-1]
    theta = cv[rho] / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def make_idx(n):
    """Create index matrix for gradient computation."""
    return np.arange(n)[:, None] + np.arange(n)[None, :]


def compute_smooth_grad(a, alpha, idx):
    """Compute gradient of smooth max approximation of conv."""
    n = len(a)
    conv = fast_convolve(a)
    c_max = conv.max()
    shifted = alpha * (conv - c_max)
    shifted = np.clip(shifted, -50.0, 50.0)
    exp_shifted = np.exp(shifted)
    p = exp_shifted / exp_shifted.sum()
    P = p[idx]
    grad = 2.0 * (P @ a)
    return grad, c_max, p


def lp_improve(a, peak_val, time_limit=5.0):
    """LP step to find improving direction."""
    n = len(a)
    if n > 1500:
        return None
    A = np.zeros((2 * n - 1, n), dtype=np.float64)
    for k in range(2 * n - 1):
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        A[k, lo:hi + 1] = a[k - hi:k - lo + 1][::-1]
    c_vec = -np.ones(n)
    b_ub = np.full(2 * n - 1, peak_val)
    try:
        res = linprog(c_vec, A_ub=A, b_ub=b_ub,
                      bounds=[(0.0, None)] * n,
                      method='highs', options={'time_limit': time_limit})
        if res.success and res.x is not None:
            b = res.x
            b_sum = b.sum()
            if b_sum > 1e-8:
                return b * (a.sum() / b_sum)
    except Exception:
        pass
    return None


def alt_projection_init(n, rng, n_iter=80):
    """Alternating projection for flat-spectrum non-negative sequences."""
    N = 1 << (2 * n - 2).bit_length()
    phases = rng.uniform(0.0, 2.0 * np.pi, N // 2 + 1)
    F = np.exp(1j * phases)
    F[0] = 1.0 + 0j
    a = np.ones(n) / n
    for _ in range(n_iter):
        a_full = np.fft.irfft(F, N)
        a = np.maximum(a_full[:n], 0.0)
        s = a.sum()
        if s < 1e-10:
            a = np.ones(n) / n
        else:
            a = a / s
        F_full = np.fft.rfft(a, N)
        F = np.ones(len(F_full)) * np.exp(1j * np.angle(F_full))
        F[0] = F[0].real + 0j
    return a

    return a / a.sum()

def optimize_adam_smooth(a, deadline, rng, idx):
    """
    Main optimization: minimize smooth max(conv) using ADAM on simplex.
    Adaptive alpha, periodic LP, restarts on stall.
    """
    n = len(a)
    a = project_simplex(a.copy())

    conv_init = fast_convolve(a)
    best_peak = conv_init.max()
    best_a = a.copy()

    # ADAM hyperparameters
    beta1, beta2, eps = 0.9, 0.999, 1e-8
    lr = 0.3 / np.sqrt(n)
    m = np.zeros(n)
    v = np.zeros(n)
    t = 0

    # Alpha schedule
    alpha = 5.0
    alpha_max = 300.0
    alpha_growth = 1.02

    it = 0
    stall = 0
    max_stall = 2500
    last_best_it = 0

    while time.time() < deadline:
        it += 1
        t += 1

        grad, c_max, p = compute_smooth_grad(a, alpha, idx)

        if c_max < best_peak * (1.0 - 1e-10):
            best_peak = c_max
            best_a = a.copy()
            stall = 0
            last_best_it = it
        else:
            stall += 1

        # ADAM update
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        step = lr * m_hat / (np.sqrt(v_hat) + eps)
        a_new = a - step
        a = project_simplex(a_new)

        # Alpha increase

                a = project_simplex((1.0 - mix) * a + mix * b)

                v = np.zeros(n)
                t = 0
                # Check improvement
                conv_check = fast_convolve(a)
                if conv_check.max() < best_peak:

                    best_a = a.copy()
                    stall = 0
