"""
Search for low-score non-negative step-function coefficients using
subgradient descent with adaptive active-set, LP refinement,
and multi-length exploration.

The objective is score = 2*n*max(convolve(a,a)) / sum(a)^2,
which is scale-invariant, so we work on the simplex sum(a)=1.
"""

import time
import numpy as np

try:
    from scipy.optimize import linprog
    HAVE_SCIPY = True
except Exception:
    HAVE_SCIPY = False


def normalize_sum1(a):
    a = np.asarray(a, dtype=np.float64)
    a = np.clip(a, 0.0, 1000.0)
    s = float(a.sum())
    if not np.isfinite(s) or s < 1e-300:
        return None
    return a / s


def local_score(a):
    a = np.asarray(a, dtype=np.float64)
    a = np.clip(a, 0.0, 1000.0)
    s = float(a.sum())
    if s < 1e-12:
        return float('inf')
    c = np.convolve(a, a)
    m = float(c.max())
    if not np.isfinite(m):
        return float('inf')
    return float(2.0 * len(a) * m / (s * s))


def compute_conv_and_gradient(a, top_k=50):
    """Compute convolution and subgradient of max convolution."""
    n = len(a)
    c = np.convolve(a, a)
    M = float(c.max())
    if not np.isfinite(M) or M <= 0:
        return c, np.zeros(n), M

    if top_k >= len(c):
        threshold = M * 0.99
        active = np.where(c >= threshold)[0]
        if len(active) == 0:
            active = np.array([int(np.argmax(c))])
    else:
        active = np.argpartition(c, -top_k)[-top_k:]

    g = np.zeros(n, dtype=np.float64)
    for k in active:
        lo = max(0, k - (n - 1))
        hi = min(int(k), n - 1)
        if lo <= hi:
            g[lo:hi + 1] += 2.0 * a[k - hi:k - lo:-1]

    g /= float(len(active))
    return c, g, M


def subgradient_step(a, eta, top_k=50):
    """One subgradient descent step with simplex projection."""
    n = len(a)
    c, g, M = compute_conv_and_gradient(a, top_k)
    # Project gradient onto tangent space of simplex
    g = g - g.mean()
    a_new = a - eta * g
    a_new = np.clip(a_new, 0.0, None)
    a_new = normalize_sum1(a_new)
    if a_new is None:
        return a.copy()
    return a_new


def subgradient_optimize(a, budget, seed, top_k_start=80, top_k_end=20):
    """Run subgradient optimization with adaptive step size and active set."""
    a = normalize_sum1(a)
    if a is None:
        return None, float('inf')

    best = a.copy()
    best_sc = local_score(best)
    n = len(a)

    t0 = time.monotonic()
    deadline = t0 + budget
    rng = np.random.default_rng(seed)

    eta = 0.1 / np.sqrt(n)
    top_k = top_k_start
    stall = 0
    it = 0

    while time.monotonic() < deadline:
        # Adaptive top_k: start large, shrink over time
        frac = (time.monotonic() - t0) / max(budget, 1e-6)
        current_top_k = int(top_k_start + frac * (top_k_end - top_k_start))
        current_top_k = max(10, min(current_top_k, 200))

        a_new = subgradient_step(a, eta, current_top_k)
        sc = local_score(a_new)

        if np.isfinite(sc) and sc < best_sc:
            best = a_new.copy()
            best_sc = float(sc)
            stall = 0
            # Slightly increase eta if improving
            if eta < 0.5 / np.sqrt(n):
                eta *= 1.01
        else:
            stall += 1
            a = a_new  # still move, but don't update best

        if stall > 60:
            # Decay step size and restart from best
            eta *= 0.7
            a = best.copy()
            # Occasionally add noise
            if rng.random() < 0.3 and eta < 0.01 / np.sqrt(n):
                noise = np.exp(rng.normal(0.0, 0.1, n))
                a = best * noise
                a = normalize_sum1(a)
                if a is None:
                    a = best.copy()
            stall = 0

        # Periodically shrink top_k to focus on the main peak
        if it % 200 == 0:
            current_top_k = max(10, current_top_k // 2)

        it += 1

    return best, best_sc


def lp_refine(a, budget, seed):
    """LP-based refinement: find direction that maximizes sum subject to conv constraints."""
    if not HAVE_SCIPY:
        return None, float('inf')

    a = normalize_sum1(a)
    if a is None:
        return None, float('inf')

    best = a.copy()
    best_sc = local_score(best)
    n = len(a)

    t0 = time.monotonic()
    deadline = t0 + budget
    rng = np.random.default_rng(seed)
    stall = 0

    while time.monotonic() < deadline:
        if deadline - time.monotonic() < 5.0:
            break

        c = np.convolve(best, best)
        M = float(c.max())
        if not np.isfinite(M) or M <= 0:
            break

        # Build constraint matrix A where A[k,j] = a[k-j] if valid
        # For memory efficiency with large n, subsample constraints
        num_constraints = 2 * n - 1
        if n > 3000:
            # Subsample: keep top constraints
            idx = np.argpartition(c, -min(n, num_constraints))[-min(n, num_constraints):]
            idx = np.sort(idx)
        else:
            idx = np.arange(num_constraints)

        m_var = n
        A = np.zeros((len(idx), m_var), dtype=np.float64)
        for row, k in enumerate(idx):
            lo = max(0, k - (n - 1))
            hi = min(int(k), n - 1)
            if lo <= hi:
                A[row, lo:hi + 1] = a[k - hi:k - lo:-1]

        cvec = -np.ones(m_var, dtype=np.float64)
        bvec = M * np.ones(len(idx), dtype=np.float64)
        bounds = [(0.0, None)] * m_var

        try:
            res = linprog(cvec, A_ub=A, b_ub=bvec, bounds=bounds,
                          method='highs', options={'presolve': True, 'max_iter': 100000})
        except Exception:
            break

        if res is None or not res.success:
            break

        x = np.asarray(res.x, dtype=np.float64)
        x = np.clip(x, 0.0, None)
        sx = float(x.sum())
        if sx < 1e-15:
            break

        x = x / sx  # normalize

        # Try blending
        improved = False
        for t in (0.3, 0.5, 0.7, 0.85, 1.0, 0.15):
            candidate = (1.0 - t) * best + t * x
            candidate = normalize_sum1(candidate)
            if candidate is None:
                continue
            sc = local_score(candidate)
            if np.isfinite(sc) and sc < best_sc:
                best = candidate.copy()
                best_sc = float(sc)
                improved = True
                break

        if not improved:
            stall += 1
            if stall > 5:
                break
        else:
            stall = 0

    return best, best_sc


def multiplicative_polish(a, budget, seed):
    """Quick multiplicative updates for final polish."""
    a = normalize_sum1(a)
    if a is None:
        return None, float('inf')

    best = a.copy()
    best_sc = local_score(best)
    n = len(a)

    t0 = time.monotonic()
    deadline = t0 + budget
    rng = np.random.default_rng(seed)

    eta = 0.2
    K = max(20, n // 8)

    while time.monotonic() < deadline:
        c, g, M = compute_conv_and_gradient(best, K)
        # Multiplicative update: decrease components with high gradient
        g_norm = g / (g.max() + 1e-15)
        a_new = best * np.exp(-eta * g_norm)
        a_new = normalize_sum1(a_new)
        if a_new is None:
            break
        sc = local_score(a_new)
        if np.isfinite(sc) and sc < best_sc:
            best = a_new.copy()
            best_sc = float(sc)
            eta = min(eta * 1.02, 1.0)
        else:
            eta *= 0.9
            if eta < 0.01:
                break

    return best, best_sc


def make_initial(n, seed, kind=0):
    """Create initial candidate sequences of various types."""
    rng = np.random.default_rng(seed)

    if kind == 0:

        a = np.ones(n) * rng.uniform(0.5, 1.5, n)
    elif kind == 1:
        # Smooth random
        a = np.abs(rng.normal(size=n))
        w = max(3, n // 20)
        kern = np.ones(w) / w
        a = np.convolve(a, kern, mode='same')
    elif kind == 2:
        # Two-level
        levels = rng.uniform(0.3, 1.0, n)
        a = np.where(rng.random(n) < 0.5, levels * 0.3, levels)
    elif kind == 3:
        # Cosine-windowed
        x = np.linspace(0, np.pi, n)
        a = 0.5 + 0.5 * np.cos(x) + rng.uniform(0.05, 0.2, n)
    elif kind == 4:

        a = np.abs(np.fft.irfft(rng.normal(size=n // 2 + 1)))[:n]
        a = a + 0.1
    else:
        a = rng.beta(2, 2, n)

    a = np.clip(a, 0.01, None)
    return normalize_sum1(a)
