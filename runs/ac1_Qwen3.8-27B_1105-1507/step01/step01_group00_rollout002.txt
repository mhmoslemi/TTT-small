```python
import numpy as np
import time

def compute_score_quick(f):
    """Quick score computation without calling evaluate_sequence."""
    n = len(f)
    S = np.sum(f)
    if S < 0.01:
        return float('inf')
    c = np.convolve(f, f)
    M = np.max(c)
    return 2.0 * n * M / (S * S)

def subgradient_descent(f, deadline, verbose=False):
    """
    Minimize max(conv(f,f)) subject to f >= 0, sum(f) = 1.
    Uses subgradient with adaptive step size and momentum.
    """
    f = f.copy().astype(np.float64)
    S = np.sum(f)
    f = f / S  # normalize to sum 1
    
    best_f = f.copy()
    best_score = compute_score_quick(f)
    
    # Adam parameters
    m = np.zeros_like(f)  # first moment
    v = np.zeros_like(f)  # second moment
    beta1, beta2, eps = 0.9, 0.999, 1e-8
    t = 0
    lr = 0.01
    no_improve = 0
    
    while time.monotonic() < deadline:
        t += 1
        c = np.convolve(f, f)
        M = np.max(c)
        
        # Subgradient of max conv at the argmax lag
        k_star = int(np.argmax(c))
        grad = np.zeros(len(f))
        n = len(f)
        for j in range(n):
            if 0 <= k_star - j < n:
                grad[j] = 2.0 * f[k_star - j]
        
        # Adam update
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * grad**2
        mhat = m / (1 - beta1**t)
        vhat = v / (1 - beta2**t)
        
        f_new = f - lr * mhat / (np.sqrt(vhat) + eps)
        f_new = np.clip(f_new, 0, None)
        
        # Renormalize
        S_new = np.sum(f_new)
        if S_new < 1e-10:
            # Reset
            f_new = np.ones(n) / n
        else:
            f_new = f_new / S_new
        
        score = compute_score_quick(f_new)
        if score < best_score:
            best_f = f_new.copy()
            best_score = score
            no_improve = 0
            # Decay lr slowly
            lr = max(0.001, lr * 0.9999)
        else:
            no_improve += 1
            lr *= 0.95
            if no_improve > 100 or lr < 1e-6:
                break
        
        f = f_new
    
    return best_f, best_score
```

```python
Multi-strategy optimizer for minimizing 2n*max(conv(f,f))/sum(f)^2.
Uses subgradient descent with Adam, frequency-domain spectral shaping,
and alternating projections. Tries multiple sequence lengths.
"""
import numpy as np
import time

def quick_score(f):
    n = len(f)
    S = np.sum(f)
    if S < 0.01:
        return float('inf')
    c = np.convolve(f, f)
    return 2.0 * n * float(np.max(c)) / (S * S)

def simplex_project(f):
    """Project onto {x >= 0, sum(x) = 1}."""
    f = np.clip(f, 0, None)
    S = np.sum(f)
    if S > 0:
        f = f / S
    else:
        f = np.ones_like(f) / len(f)
    return f

def subgradient_adam(f_init, deadline, lr0=0.01, verbose_tag=""):
    """Subgradient descent with Adam for minimizing max(conv(f,f)) on simplex."""
    f = simplex_project(f_init.copy())
    n = len(f)
    best_f = f.copy()
    best_score = quick_score(f)
    
    m = np.zeros(n)
    v = np.zeros(n)
    b1, b2, eps = 0.9, 0.999, 1e-8
    t = 0
    lr = lr0
    patience = 0
    
    while time.monotonic() < deadline:
        t += 1
        c = np.convolve(f, f)
        k_star = int(np.argmax(c))
        
        # Subgradient
        grad = np.zeros(n)
        for j in range(n):
            idx = k_star - j
            if 0 <= idx < n:
                grad[j] = 2.0 * f[idx]
        
        # Adam
        m = b1 * m + (1 - b1) * grad
        v = b2 * v + (1 - b2) * grad * grad
        m_hat = m / (1 - b1**t)
        v_hat = v / (1 - b2**t)
        
        f_new = f - lr * m_hat / (np.sqrt(v_hat) + eps)
        f_new = simplex_project(f_new)
        
        sc = quick_score(f_new)
        if sc < best_score:
            best_f = f_new.copy()
            best_score = sc
            patience = 0
            if verbose_tag and t % 500 == 0:
                print(f"  {verbose_tag} t={t} score={best_score:.6f}")
        else:
            patience += 1
            lr *= 0.98
            if patience > 200 or lr < 1e-8:
                break
        f = f_new
    
    return best_f, best_score

def spectral_shaping(f_init, deadline, beta=0.3, verbose_tag=""):
    """Frequency-domain spectral flattening via Gerchberg-Saxton style iteration."""
    f = simplex_project(f_init.copy())
    n = len(f)
    N = 4 * n  # FFT size
    best_f = f.copy()
    best_score = quick_score(f)
    
    for it in range(100000):
        if time.monotonic() > deadline:
            break
        
        F = np.fft.fft(f, N)
        mag = np.abs(F)
        mag2 = mag * mag
        
        # Target: flat spectrum
        # Scale to reduce dynamic range
        mean_mag2 = np.mean(mag2)
        # New magnitude: interpolate between current and flat
        new_mag2 = mag2 ** (1 - beta) * mean_mag2 ** beta
        new_mag = np.sqrt(new_mag2)
        
        # Preserve phase
        phase = F / (mag + 1e-30)
        F_new = phase * new_mag
        
        f_new = np.real(np.fft.ifft(F_new))[:n]
        f_new = simplex_project(f_new)
        
        sc = quick_score(f_new)
        if sc < best_score:
            best_f = f_new.copy()
            best_score = sc
            if verbose_tag and it % 1000 == 0:
                print(f"  {verbose_tag} iter={it} score={best_score:.6f}")
    
    return best_f, best_score

def alternating_projection(f_init, deadline, beta=0.3, verbose_tag=""):
    """Alternates between time-domain (simplex) and frequency-domain (spectral) constraints."""
    f = simplex_project(f_init.copy())
```

```python
"""
Multi-strategy optimizer for minimizing 2n*max(conv(f,f))/sum(f)^2.
Uses: (1) smoothed subgradient descent with top-k peak tracking,
(2) frequency-domain spectral flattening (Gerchberg-Saxton style),
(3) alternating time/frequency projections, (4) multiple lengths and
random restarts. Scale-invariant: works with sum(f)=1 WLOG.
"""
import numpy as np
import time


def quick_score(f):
    """Compute objective without calling evaluate_sequence (for speed)."""
    n = len(f)
    S = np.sum(f)
    if S < 0.01:
        return float('inf')
    c = np.convolve(f, f)
    return 2.0 * n * float(np.max(c)) / (S * S)


def simplex_project(f):
    """Project onto {x >= 0, sum(x) = 1}."""
    f = np.maximum(f, 0.0)
    S = np.sum(f)
    if S > 1e-15:
        return f / S
    return np.ones_like(f) / len(f)


def smoothed_subgradient_adam(f_init, deadline, lr0=0.005, top_k=5, verbose_tag=""):
    """
    Smoothed subgradient with Adam. Uses top-k autocorrelation peaks
    for a smoother gradient signal than single argmax.
    """
    f = simplex_project(f_init.copy().astype(np.float64))
    n = len(f)
    best_f = f.copy()
    best_score = quick_score(f)

    m = np.zeros(n)
    v = np.zeros(n)
    b1, b2, eps = 0.9, 0.999, 1e-8
    t = 0
    lr = lr0
    patience = 0
    best_patience = 0

    while time.monotonic() < deadline:
        t += 1
        c = np.convolve(f, f)

        # Top-k lag selection for smoothed gradient
        k_sel = min(top_k, len(c))
        top_indices = np.argsort(c)[::-1][:k_sel]
        weights = np.zeros(len(c))
        for idx in top_indices:
            weights[idx] = c[idx]
        w_sum = np.sum(weights)
        if w_sum < 1e-30:
            break
        weights = weights / w_sum

        # Compute weighted gradient: grad[j] = sum_k weights[k] * 2*f[k-j]
        grad = np.zeros(n)
        for k in top_indices:
            w = weights[k]
            if w < 1e-15:
                continue
            j_min = max(0, k - (n - 1))
            j_max = min(k, n - 1)
            for j in range(j_min, j_max + 1):
                grad[j] += w * 2.0 * f[k - j]

        # Adam update
        m = b1 * m + (1.0 - b1) * grad
        v = b2 * v + (1.0 - b2) * grad * grad
        m_hat = m / (1.0 - b1 ** t)
        v_hat = v / (1.0 - b2 ** t)

        f_new = f - lr * m_hat / (np.sqrt(v_hat) + eps)
        f_new = simplex_project(f_new)

        sc = quick_score(f_new)
        if sc < best_score:
            best_f = f_new.copy()
            best_score = sc
            patience = 0
            best_patience = patience
            if verbose_tag and t % 200 == 0:
                print(f"    {verbose_tag} t={t} score={best_score:.6f} lr={lr:.6f}")
        else:
            patience += 1
            if patience > 50:
                lr *= 0.9
            if patience > 300 or lr < 1e-9:
                break
        f = f_new

    return best_f, best_score


def spectral_flatten(f_init, deadline, beta=0.3, verbose_tag=""):
    """
    Gerchberg-Saxton style: iteratively flatten |FFT(f)|^2 while
    maintaining f >= 0 and sum(f) = 1.
    """
    f = simplex_project(f_init.copy().astype(np.float64))
    n = len(f)
    N = max(4 * n, 2048)
    best_f = f.copy()
    best_score = quick_score(f)
    patience = 0

    for it in range(200000):
        if time.monotonic() > deadline:
            break

        F = np.fft.rfft(f, N)
        mag2 = np.abs(F) ** 2
        mean_mag2 = np.mean(mag2)
        if mean_mag2 < 1e-30:
            break

        # Compress dynamic range of spectrum
        # new_mag2 = mag2^(1-beta) * mean_mag2^beta
        # This reduces peaks in spectrum -> reduces autocorrelation peaks
        new_mag2 = mag2 ** (1.0 - beta) * mean_mag2 ** beta
        new_mag = np.sqrt(new_mag2)

        # Preserve phase
        phase = F / (np.abs(F) + 1e-30)
        F_new = phase * new_mag

        f_new = np.fft.irfft(F_new, N)[:n]
        f_new = simplex_project(f_new)

        sc = quick_score(f_new)
        if sc < best_score:
            best_f = f_new.copy()
            best_score = sc
            patience = 0
            if verbose_tag and it % 2000 == 0:
                print(f"    {verbose_tag} iter={it} score={best_score:.6f}")
        else:
            patience += 1
            if patience > 5000:
                break
        f = f_new

    return best_f, best_score


def alternating_optimize(f_init, deadline, beta=0.3, lr0=0.003, verbose_tag=""):
    """
    Alternate between spectral flattening and subgradient steps.
    """
    f = simplex_project(f_init.copy().astype(np.float64))
    n = len(f)
    N = max(4 * n, 2048)
    best_f = f.copy()
    best_score = quick_score(f)

    m = np.zeros(n)
    v = np.zeros(n)
    b1, b2, eps = 0.9, 0.999, 1e-8
    t = 0
    lr = lr0
    patience = 0

    while time.monotonic() < deadline:
        # Frequency step
        F = np.fft.rfft(f, N)
        mag = np.abs(F)
        mag2 = mag * mag
        mean_mag2 = np.mean(mag2)
        if mean_mag2 > 1e-30:
            new_mag2 = mag2 ** (1.0 - beta) * mean_mag2 ** beta
            new_mag = np.sqrt(new_mag2)
            phase = F / (mag + 1e-30)
            F_new = phase * new_mag
            f = np.fft.irfft(F_new, N)[:n]
            f = simplex_project(f)

        # Time step (subgradient)
        t += 1
        c = np.convolve(f, f)
        k_sel = min(3, len(c))
        top_indices = np.argsort(c)[::-1][:k_sel]
        weights = np.zeros(len(c))
        for idx in top_indices:
            weights[idx] = c[idx]
        w_sum = np.sum(weights)
        if w_sum > 1e-30:
            weights /= w_sum
            grad = np.zeros(n)
            for k in top_indices:
                w = weights[k]
                if w < 1e-15:
                    continue
                j_min = max(0, k - (n - 1))

                    grad[j] += w * 2.0 * f[k - j]

            m = b1 * m + (1.0 - b1) * grad
            v = b2 * v + (1.0 - b2) * grad * grad
            m_hat = m / (1.0 - b1 ** t)
            v_hat = v / (1.0 - b2 ** t)
            f = f - lr * m_hat / (np.sqrt(v_hat) + eps)

        if sc < best_score:
            best_f = f.copy()
            best_score = sc
            patience = 0

        else:
            patience += 1

                lr *= 0.95
            if patience > 5000 or lr < 1e-10:
                break

    return best_f, best_score


def peak_shave(f_init, deadline, verbose_tag=""):
    """
    Targeted peak shaving: identify autocorrelation peaks and directly
    reduce the contributing f values.
    """
    f = simplex_project(f_init.copy().astype(np.float64))
    n = len(f)
    best_f = f.copy()
    best_score = quick_score(f)
    patience = 0

    while time.monotonic() < deadline:
        c = np.convolve(f, f)
        M = c.max()
        if M < 1e-15:
            break

        # Find top peaks
        num_peaks = min(5, len(c))
        peak_idxs = np.argsort(c)[::-1][:num_peaks]

        # For each peak, reduce contributing f values proportionally
        step = 0.01 / M
        f_new = f.copy()
        for k in peak_idxs:
            if c[k] / M < 0.8:
                continue
            j_min = max(0, k - (n - 1))
            j_max = min(k, n - 1)
            for j in range(j_min, j_max + 1):
                contrib = f[j] * f[k - j]
                if contrib > 0:
                    f_new[j] -= step * f[k - j]

        f_new = simplex_project(f_new)
        sc = quick_score(f_new)
        if sc < best_score:
            best_f = f_new.copy()
            best_score = sc
            patience = 0
            f = f_new
        else:
            patience += 1
            step *= 0.8
            if patience > 200 or step < 1e-10:
                break

    return best_f, best_score

            if sc_out < best_score:
                best_score = sc_out
                best_sequence = f_out.copy()
                print(f"  Final alternating: {sc_out:.6f}")

    # Verify with actual evaluate_sequence
    if best_sequence is not None:
        final_sc = evaluate_sequence(best_sequence.tolist())
        elapsed = time.monotonic() - t0
        print(f"\nFINAL: n={len(best_sequence)}, score={final_sc:.6f}, time={elapsed:.1f}s")
        if not np.isfinite(final_sc):
            print("WARNING: final score is not finite, using fallback")
            return [1.0] * 128
    else:
        print("No valid sequence found, using fallback")
        return [1.0] * 128

    return [float(x) for x in best_sequence]
```
