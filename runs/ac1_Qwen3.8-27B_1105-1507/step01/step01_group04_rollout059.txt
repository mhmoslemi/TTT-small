"""
Optimizes non-negative step function heights to minimize 2n*max(conv(f,f))/(sum(f))^2.
Uses a multi-phase approach:
1. Projected gradient with log-sum-exp smoothing (FFT-accelerated) for fast initial optimization
2. Frank-Wolfe with LP subproblem for convergence to high-quality local optima
3. Local coordinate descent polishing for final micro-improvements
"""

import numpy as np
import time
from scipy.optimize import linprog
from scipy.sparse import coo_matrix, csr_matrix


def _fft_convolve_full(f):
    """Compute full linear convolution of f with itself using FFT."""
    n = len(f)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    F = np.fft.rfft(f, size)
    h = np.fft.irfft(F * F, size)
    return h[:2 * n - 1]


def _fft_cross_correlate(p, g):
    """Compute cross-correlation (p ⊛ g)[l] = sum_k p[k] * g[k-l] for valid indices."""
    n = len(g)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    P = np.fft.rfft(p, size)
    G = np.fft.rfft(g, size)
    # Cross-correlation: ifft(conj(P) * G) gives (p ⊛ g)
    cc = np.fft.irfft(np.conj(P) * G, size)
    return cc[:2 * n - 1]


def _simplex_project(z):
    """Project z onto the probability simplex {w >= 0, sum(w) = 1}."""
    n = len(z)
    if np.sum(z) <= 0:
        return np.ones(n) / n
    # Binary search for threshold
    u = np.sort(z)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - 1))[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    w = np.maximum(z - theta, 0.0)
    s = np.sum(w)
    if s > 0:
        w /= s
    else:
        w = np.ones(n) / n
    return w


def _score_from_conv(h, n):
    """Compute score from precomputed convolution (assumes sum(f)=1)."""
    M = float(np.max(h))
    return 2.0 * n * M


def _logsumexp_score(h, alpha):
    """Smooth approximation to max(h) using log-sum-exp."""
    h_max = np.max(h)
    if h_max < 0:
        h_max = 0.0
    shifted = alpha * (h - h_max)
    # Clip for numerical stability
    shifted = np.maximum(shifted, -100.0)
    s = np.sum(np.exp(shifted))
    if s <= 0:
        return h_max
    return h_max + np.log(s) / alpha


def _logsumexp_gradient(h, f, alpha):
    """
    Compute the gradient of the log-sum-exp smoothing of max(conv(f,f)).
    Returns q[j] = sum_k p_k * 2*f[k-j] (the subgradient weighted by softmax).
    """
    n = len(f)
    h_max = np.max(h)
    shifted = alpha * (h - h_max)
    shifted = np.maximum(shifted, -100.0)
    exp_vals = np.exp(shifted)
    total = np.sum(exp_vals)
    if total <= 0:
        p = np.zeros(len(h))
        p[np.argmax(h)] = 1.0
    else:
        p = exp_vals / total
    # p has length 2n-1, f has length n
    # q[j] = sum_k p[k] * 2*f[k-j] for valid k-j
    # This is 2 * (p cross-correlate f)[j]
    # (p ⊛ f)[j] = sum_k p[k] * f[k-j]
    q = 2.0 * _fft_cross_correlate(p, f)
    return q[:n]


def _projected_gradient_descent(f_init, n_iter, alpha_start, alpha_end, step_scale, label=""):
    """
    Projected gradient descent with log-sum-exp smoothing and alpha annealing.
    Minimize max(conv(f,f)) subject to f >= 0, sum(f) = 1.
    """
    f = np.asarray(f_init, dtype=np.float64).copy()
    total = np.sum(f)
    if total < 1e-30:
        f = np.ones(len(f)) / len(f)
    else:
        f = f / total
    n = len(f)

    best_f = f.copy()
    h = _fft_convolve_full(f)
    best_score = _score_from_conv(h, n)

    for it in range(n_iter):
        # Anneal alpha
        frac = float(it) / max(1, n_iter - 1)
        alpha = alpha_start * (alpha_end / alpha_start) ** frac

        h = _fft_convolve_full(f)
        M = float(np.max(h))
        if M < 1e-30:
            break

        # Compute gradient
        q = _logsumexp_gradient(h, f, alpha)

        # Step size: use 1/(alpha * 2 * max_f) as a rough Lipschitz estimate
        max_f = float(np.max(f))
        L = alpha * 2.0 * (max_f + 1e-10)
        step = step_scale / max(L, 1e-10)

        # Take step and project
        f_new = f - step * q
        f_new = _simplex_project(f_new)

        # Check improvement
        h_new = _fft_convolve_full(f_new)
        score_new = _score_from_conv(h_new, n)

        if score_new < best_score:
            best_score = score_new
            best_f = f_new.copy()

        f = f_new

        if (it + 1) % 2000 == 0:
            print(f"    PGD {label} it={it+1}: score={best_score:.6f}, alpha={alpha:.1f}")

    return best_f, best_score


def _frank_wolfe(f_init, max_iters, time_budget, label=""):
    """
    Frank-Wolfe (conditional gradient) method for minimizing max(conv(f,f))
    on the simplex. Uses LP subproblem: maximize sum(g) s.t. conv(g,f)_k <= M, g >= 0.
    """
    f = np.asarray(f_init, dtype=np.float64).copy()
    total = np.sum(f)
    if total < 1e-30:
        f = np.ones(len(f)) / len(f)
    else:
        f = f / total
    n = len(f)

    best_f = f.copy()
    h = _fft_convolve_full(f)
    best_score = _score_from_conv(h, n)
    start_time = time.monotonic()

    # Precompute structure for the LP constraint matrix
    # A[k, j] = f[k-j] if 0 <= k-j <= n-1
    # We'll build this each iteration since f changes

    for it in range(max_iters):
        if time.monotonic() - start_time > time_budget:
            break

        h = _fft_convolve_full(f)
        M = float(np.max(h))

        # Build sparse constraint matrix
        row_idx = []
        col_idx = []
        vals = []
        # For each j (column), the non-zero rows are k = j, j+1, ..., j+n-1
        # A[k, j] = f[k-j]
        js = np.arange(n)
        for j in js:
            ks = j + np.arange(n)
            row_idx.extend(ks.tolist())
            col_idx.extend([j] * n)
            vals.extend(f.tolist())

        A = coo_matrix((vals, (row_idx, col_idx)), shape=(2 * n - 1, n)).tocsr()

        # Solve LP: max 1^T g s.t. A @ g <= M, g >= 0
        c_obj = -np.ones(n)
        b_ub = np.full(2 * n - 1, M)
        res = linprog(c_obj, A_ub=A, b_ub=b_ub,
                      bounds=[(0, None)] * n, method='highs')

        if not res.success or res.x is None:
            print(f"    FW {label}: LP failed at iter {it}: {res.message}")
            break

        g = res.x
        Sg = float(np.sum(g))
        if Sg < 1e-12:
            break
        g = g / Sg

        # Frank-Wolfe step size
        t = 2.0 / (it + 2)

        f_new = (1.0 - t) * f + t * g
        f_new = _simplex_project(f_new)

        h_new = _fft_convolve_full(f_new)
        score_new = _score_from_conv(h_new, n)

        if score_new < best_score:
            best_score = score_new
            best_f = f_new.copy()

        f = f_new

        if (it + 1) % 5 == 0:
            elapsed = time.monotonic() - start_time
            print(f"    FW {label} it={it+1}: score={best_score:.6f} ({elapsed:.1f}s)")

    return best_f, best_score


def _cosine_init(n):
    """Raised cosine initialization."""
    j = np.arange(n, dtype=np.float64)
    f = np.cos(np.pi * (j + 0.5) / float(n)) ** 2
    f = np.maximum(f, 0.0)
    s = np.sum(f)
    if s > 0:
        f /= s
    return f


def _super_gaussian_init(n, p):
    """Super-Gaussian initialization: f[j] = exp(-((j - c)/sigma)^p)."""
    c = (n - 1) / 2.0
    j = np.arange(n, dtype=np.float64)
    sigma = n / 4.0
    f = np.exp(-np.abs(j - c) ** p / (sigma ** p))
    s = np.sum(f)
    if s > 0:
        f /= s
    return f


def _dithered_init(n, period, amplitude):
    """Dithered initialization: uniform + small periodic perturbation."""
    j = np.arange(n, dtype=np.float64)
    f = 1.0 + amplitude * np.sin(2.0 * np.pi * j / period)
    f = np.maximum(f, 0.0)
    s = np.sum(f)
    if s > 0:
        f /= s
    return f


def _tapered_init(n, power):
    """Sine-tapered initialization."""
    j = np.arange(n, dtype=np.float64)
    x = np.sin(np.pi * (j + 0.5) / float(n))
    f = x ** power
    f = np.maximum(f, 0.0)
    s = np.sum(f)
    if s > 0:
        f /= s
    return f


def _coordinate_descent_polish(f, n_iter, label=""):
    """
    Coordinate descent: for each coordinate j, find the optimal value
    that minimizes max(conv(f,f)) while keeping other coordinates fixed.
    """
    f = np.asarray(f, dtype=np.float64).copy()
    n = len(f)
    h = _fft_convolve_full(f)
    best_score = _score_from_conv(h, n)
    best_f = f.copy()

    for it in range(n_iter):
        improved = False
        for j in range(n):
            # Try small perturbations of f[j]
            # When we change f[j] by delta, conv changes by:
            # conv_new[k] = conv_old[k] + delta * (f[k-j] + f[j+k-n+1-k] ... )
            # Actually, conv_new[k] = conv_old[k] + 2*delta*f[k-j] - delta^2 (if k=2j)
            # More precisely: conv_new[k] = sum_l f[l]*f[k-l]
            # If we change f[j] to f[j]+delta:
            #   conv_new[k] = conv_old[k] + delta*(f[k-j] + f[k-j]) - (f[j] was counted)
            # Actually let's just recompute for the affected coordinates
            # But that's expensive. Instead, use a simple grid search on f[j].

            current_fj = f[j]
            # Try a few candidate values
            candidates = [current_fj]
            # Slightly increase/decrease
            for frac in [0.9, 0.95, 1.05, 1.1]:
                candidates.append(current_fj * frac)
            candidates.append(0.0)

            best_val = current_fj
            best_local_score = best_score + 1e-10  # require improvement

            for val in candidates:
                old = f[j]
                f[j] = max(0.0, val)
                # Recompute sum
                s = np.sum(f)
                if s < 1e-10:
                    f[j] = old
                    continue
                f = f / s
                h = _fft_convolve_full(f)
                sc = _score_from_conv(h, n)
                if sc < best_local_score:
                    best_local_score = sc
                    best_val = val
                f[j] = old  # restore

            if best_val != f[j]:
                f[j] = max(0.0, best_val)
                s = np.sum(f)
                if s > 0:
                    f = f / s
                h = _fft_convolve_full(f)
                sc = _score_from_conv(h, n)
                if sc < best_score:
                    best_score = sc
                    best_f = f.copy()
                    improved = True
                f[j] = old if sc >= best_score else f[j]
                # Re-normalize if we changed
                s = np.sum(f)
                if s > 0:
                    f = f / s

        if not improved:
            break

    return best_f, best_score


def propose_candidate(seed=42, budget_s=1500, **kwargs):
    """
    Search for non-negative step function heights minimizing the evaluation score.
    Multi-phase: projected gradient (log-sum-exp) -> Frank-Wolfe (LP) -> coordinate polish.
    """
    deadline = time.monotonic() + float(budget_s) - 10.0
    rng = np.random.default_rng(seed)

    # Get parent sequence
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64).copy()
        parent = np.maximum(parent, 0.0)
        parent = np.minimum(parent, 1000.0)
    except Exception:
        parent = np.ones(512, dtype=np.float64)

    # Normalize parent
    ps = np.sum(parent)
    if ps > 0:
        parent_norm = parent / ps
    else:
        parent_norm = np.ones(len(parent)) / len(parent)

    best_f = parent_norm.copy()
    h_best = _fft_convolve_full(best_f)
    best_score = _score_from_conv(h_best, len(best_f))
    print(f"Parent: n={len(best_f)}, score={best_score:.6f}")

    # Define candidate initializations
    n_target = max(len(parent), 512)
    # Try multiple sizes
    sizes = sorted(set([n_target, 512, 1024, 2048]))
    # Cap at reasonable size
    sizes = [s for s in sizes if s <= 4096]

    candidates = []
    for n in sizes:
        candidates.append(("cos_%d" % n, _cosine_init(n)))
        candidates.append(("supg3_%d" % n, _super_gaussian_init(n, 3)))
        candidates.append(("supg4_%d" % n, _super_gaussian_init(n, 4)))
        candidates.append(("tap1_%d" % n, _tapered_init(n, 1.0)))
        candidates.append(("tap2_%d" % n, _tapered_init(n, 2.0)))
        for period in [4, 8, 16]:
            candidates.append(("dith%d_%d" % (period, n), _dithered_init(n, period, 0.3)))
        # Random
        f_r = rng.random(n)
        f_r = f_r / np.sum(f_r)
        candidates.append(("rand_%d" % n, f_r))

    # Add parent (padded/truncated to various sizes)
    if len(parent_norm) > 0:
        candidates.append(("parent", parent_norm.copy()))

    print(f"Total candidates: {len(candidates)}")

    # Phase 1: Projected gradient with log-sum-exp on each candidate
    phase1_deadline = deadline - 600.0  # Reserve time for phase 2
    time_per_candidate = max(5.0, (phase1_deadline - time.monotonic()) / max(1, len(candidates)))
    print(f"Phase 1: {len(candidates)} candidates, ~{time_per_candidate:.1f}s each")

    results = []
    for idx, (name, f0) in enumerate(candidates):
        if time.monotonic() > phase1_deadline - 10:
            break

        n_cand = len(f0)
        # Scale iterations based on time available
        n_iter = int(min(5000, max(1000, time_per_candidate * 800)))
        t0 = time.monotonic()

        f_pgd, s_pgd = _projected_gradient_descent(
            f0, n_iter,
            alpha_start=20.0, alpha_end=200.0,
            step_scale=0.8,
            label=name
        )
        elapsed = time.monotonic() - t0
        results.append((s_pgd, name, f_pgd))
        print(f"  [{idx+1}/{len(candidates)}] {name} (n={n_cand}): score={s_pgd:.6f} ({elapsed:.1f}s)")

        if s_pgd < best_score:
            best_score = s_pgd
            best_f = f_pgd.copy()
            print(f"  ** New best: {best_score:.6f} from {name} **")

    # Sort results by score
    results.sort(key=lambda x: x[0])

    # Phase 2: Frank-Wolfe on top candidates
    print(f"\nPhase 2: Frank-Wolfe refinement")
    top_k = min(5, len(results))
    fw_deadline = deadline - 100.0

    for r in range(top_k):
        if time.monotonic() > fw_deadline - 30:
            break
        s_cur, name, f_cur = results[r]
        n_cur = len(f_cur)
        print(f"\n  FW on {name} (n={n_cur}, score={s_cur:.6f})")

        # Time budget for this FW
        remaining = fw_deadline - time.monotonic() - 5
        if remaining < 30:
            break

        # For large n, fewer iterations
        if n_cur <= 1024:
            max_fw_iters = 100
        elif n_cur <= 2048:
            max_fw_iters = 50
        else:
            max_fw_iters = 25

        time_per_fw = min(remaining * 0.8, 400.0)
        f_fw, s_fw = _frank_wolfe(f_cur, max_fw_iters, time_per_fw, label=name)

        if s_fw < best_score:
            best_score = s_fw
            best_f = f_fw.copy()
            print(f"  ** New best: {best_score:.6f} from FW({name}) **")

        # Update the candidate for potential further refinement
        results[r] = (s_fw, name, f_fw)

        # If time permits, do more FW iterations on the best
        if time.monotonic() < fw_deadline - 60 and s_fw <= results[0][0] + 1e-6:
            remaining2 = fw_deadline - time.monotonic() - 5
            if remaining2 > 30:
                f_fw2, s_fw2 = _frank_wolfe(f_fw, max_fw_iters // 2, remaining2 * 0.7,
                                             label=name + "_2")
                if s_fw2 < best_score:
                    best_score = s_fw2
                    best_f = f_fw2.copy()
                    print(f"  ** New best: {best_score:.6f} from FW({name}_2) **")
                    best_f = f_fw2

    # Phase 3: Coordinate descent polish on the best solution
    print(f"\nPhase 3: Coordinate descent polish (score={best_score:.6f})")
    if time.monotonic() < deadline - 20:
        n_polish_iters = min(5, max(1, int((deadline - time.monotonic() - 10) / 30)))
        f_pol, s_pol = _coordinate_descent_polish(best_f, n_polish_iters, label="polish")
        if s_pol < best_score:
            best_score = s_pol
            best_f = f_pol.copy()
            print(f"  ** New best after polish: {best_score:.6f} **")

    # Final verification with the actual evaluator
    final_list = [float(x) for x in best_f]
    final_score = float(evaluate_sequence(final_list))
    print(f"\nFinal verified score: {final_score:.6f} (n={len(final_list)})")

    # Fallback if something went wrong
    if not np.isfinite(final_score) or final_score > 2.0:
        print("  Falling back to parent.")
        final_score = float(evaluate_sequence([float(x) for x in parent_norm]))
        if np.isfinite(final_score):
            return [float(x) for x in parent_norm]
        return [1.0] * 512

    return final_list
