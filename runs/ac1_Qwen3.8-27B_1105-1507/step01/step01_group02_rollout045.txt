import numpy as np
import time

def fast_convolve(a):
    """Compute linear convolution of a with itself using FFT."""
    n = len(a)
    size = 2 * n - 1
    N = 1 << (size - 1).bit_length()
    fa = np.fft.rfft(a, N)
    fc = fa * fa
    result = np.fft.irfft(fc, N)[:size]
    result[result < 0] = 0.0
    return result

def project_simplex(v):
    """Project onto probability simplex."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cv = np.cumsum(u) - 1.0
    rho_candidates = np.where(u * np.arange(1, n + 1) > cv)[0]
    if len(rho_candidates) == 0:
        return np.ones(n) / n
    rho = rho_candidates[-1]
    theta = cv[rho] / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def gradient_for_peak(a, k):
    """Compute ∂(a*a)_k / ∂a for a single peak index k."""
    n = len(a)
    grad = np.zeros(n)
    if k < n:
        grad[:n-k] += a[k:]
        grad[k:] += a[:n-k]
    return grad

def compute_subgradient(a, peak_indices):
    """Compute average subgradient over peak indices."""
    grad = np.zeros(len(a))
    for k in peak_indices:
        grad += gradient_for_peak(a, k)
    grad /= len(peak_indices)
    return grad

def multiplicative_update(a, grad, eta):
    """Multiplicative weight update: a <- a * exp(-eta * grad), renormalized."""
    a_new = a * np.exp(-eta * (grad - grad.mean()))
    a_new = np.maximum(a_new, 1e-15)
    s = a_new.sum()
    return a_new / s

def fast_convolve(a):
    """Compute linear convolution of a with itself using FFT."""
    n = len(a)
    size = 2 * n - 1
    N = 1 << (size - 1).bit_length()
    fa = np.fft.rfft(a, N)
    fc = fa * fa
    result = np.fft.irfft(fc, N)[:size]
    result[result < 0] = 0.0
    return result


def project_simplex(v):
    """Project vector onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cv = np.cumsum(u) - 1.0
    rho_candidates = np.where(u * np.arange(1, n + 1) > cv)[0]
    if len(rho_candidates) == 0:
        return np.ones(n) / n
    rho = rho_candidates[-1]
    theta = cv[rho] / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def gradient_for_peak(a, k):
    """Compute gradient of (a*a)_k with respect to a. O(n) per call."""
    n = len(a)
    grad = np.zeros(n)
    if k < n:
        grad[:n - k] += a[k:]
        grad[k:] += a[:n - k]
    return grad


def compute_gradient(a, c, t):
    """
    Compute smooth-max gradient: weighted average of per-peak gradients.
    Weights proportional to exp(t * c_k).
    Returns the gradient vector (length n).
    """
    n = len(a)
    # Soft-max weights
    c_max = np.max(c)
    log_weights = t * (c - c_max)
    log_weights -= np.max(log_weights)  # numerical stability
    weights = np.exp(log_weights)
    weights /= weights.sum()

    # Only use peaks with significant weight to save computation
    significant = weights > 1e-4
    if not np.any(significant):
        # Fall back to top peak
        k = int(np.argmax(c))
        return gradient_for_peak(a, k)

    peak_indices = np.where(significant)[0]
    # Limit to top 20 for efficiency
    if len(peak_indices) > 20:
        order = np.argsort(weights[peak_indices])[::-1][:20]
        peak_indices = peak_indices[order]
        w_sub = weights[peak_indices]
        w_sub /= w_sub.sum()
    else:
        w_sub = weights[peak_indices]

    grad = np.zeros(n)
    for idx, w in zip(peak_indices, w_sub):
        grad += w * gradient_for_peak(a, idx)
    return grad

def spectral_shaping_init(n, rng, n_iter=150):
    """
    Initialize using iterative spectral shaping (Fienup/Gerchberg-Saxton variant).
    Targets flat magnitude spectrum to flatten autocorrelation.
    """
    N = 1 << (2 * n - 2).bit_length()
    a = np.maximum(rng.random(n), 0.01)
    a = a / a.sum()

        A = np.fft.rfft(a, N)
        # Target: approximately flat magnitude
        mag = np.ones(len(A))
        # Slight DC emphasis (non-negativity forces positive mean)

        # Slight taper at high frequencies to help non-negativity

        mag = mag * taper

        A_new = mag * np.exp(1j * np.angle(A))
        a = np.fft.irfft(A_new, N)[:n]
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s < 1e-10:

        else:
            a = a / s

    n_configs = len(configs)
    # Allocate time: more time to early configs (which include parent)
    total_time = overall_deadline - time.time()
    # First 4 configs get 45% of time, rest share 50%, keep 5% for polish
    first_time = total_time * 0.45 // 4
    rest_time = total_time * 0.50 // max(1, n_configs - 4)

    for i, (n, init_type, method) in enumerate(configs):
        if time.time() > overall_deadline - 60:
            break

        if i < 4:
            config_deadline = min(overall_deadline - 30, time.time() + first_time)
        else:
            config_deadline = min(overall_deadline - 30, time.time() + rest_time)

        if config_deadline < time.time() + 15:
            continue

        try:
            # Generate initial sequence
            if init_type == 'parent':
                a = np.array(height_sequence_1, dtype=np.float64)
                a = np.clip(a, 0.0, 1000.0)
                a = a / a.sum()
            elif init_type == 'chirp':
                a = chirp_init(n, rng)
            elif init_type == 'spectral':
                a = spectral_shaping_init(n, rng, n_iter=100)
            elif init_type == 'cosine':
                a = cosine_init(n, rng)
            else:
                a = rng.random(n)
                a = a / a.sum()

            # Run optimization
            if method == 'smooth':
                a_opt = optimize_sequence(a, config_deadline, rng, t_init=5.0)
            else:
                a_opt = optimize_hard_peak(a, config_deadline, rng)

            score = evaluate_sequence(a_opt.tolist())

            print(f"n={n}, init={init_type}, method={method}, "
                  f"score={score:.6f}, time_left={overall_deadline - time.time():.0f}s")

            if np.isfinite(score) and score < best_score:
                best_score = score
                best_seq = a_opt.copy()
                print(f"  >>> New best: {best_score:.6f} (n={n})")

        except Exception as e:
            print(f"  Error in config ({n}, {init_type}, {method}): {e}")
            continue

    # If we found a good sequence, do local refinement
    if best_seq is not None and time.time() < overall_deadline - 20:
        polish_deadline = overall_deadline
        a_polished = local_refinement(best_seq, polish_deadline, rng)
        score = evaluate_sequence(a_polished.tolist())
        print(f"After local refinement: score={score:.6f}")
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_seq = a_polished

    # One more round: re-optimize best with remaining time using both methods
    if best_seq is not None and time.time() < overall_deadline - 30:
        mid = time.time() + (overall_deadline - 30 - time.time()) / 2
        a1 = optimize_sequence(best_seq, mid, rng, t_init=10.0)
        s1 = evaluate_sequence(a1.tolist())
        if np.isfinite(s1) and s1 < best_score:
            best_score = s1
            best_seq = a1
            print(f"  Re-optimize smooth: {s1:.6f}")

        if time.time() < overall_deadline - 10:
            a2 = optimize_hard_peak(best_seq, overall_deadline - 10, rng)
            s2 = evaluate_sequence(a2.tolist())
            if np.isfinite(s2) and s2 < best_score:
                best_score = s2
                best_seq = a2
                print(f"  Re-optimize hard: {s2:.6f}")

    # Final local polish
    if best_seq is not None and time.time() < overall_deadline - 3:
        a_final = local_refinement(best_seq, overall_deadline, rng)
        s_final = evaluate_sequence(a_final.tolist())
        if np.isfinite(s_final) and s_final < best_score:
            best_score = s_final
            best_seq = a_final

    if best_seq is None:
        print("Fallback: using uniform sequence")
        best_seq = np.ones(256) / 256

    result = [float(max(0.0, x)) for x in best_seq]
    final_score = evaluate_sequence(result)
    print(f"FINAL: score={final_score:.6f}, n={len(result)}")
    return result
