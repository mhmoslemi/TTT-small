from scipy.sparse import csr_matrix

def build_conv_matrix_sparse(f):
    """Build sparse matrix A where (A@g)[k] = convolve(f,g)[k]."""
    n = len(f)
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        j_min = max(0, k - (n - 1))
        j_max = min(k, n - 1)
        for j in range(j_min, j_max + 1):
            rows.append(k)
            cols.append(j)
            vals.append(f[k - j])
    return csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))

        j_max = min(k, n - 1)
        for j in range(j_min, j_max + 1):
            rows[idx] = k
            cols[idx] = j
            vals[idx] = f[k - j]
            idx += 1
    return csr_matrix((vals[:idx], (rows[:idx], cols[:idx])), shape=(2 * n - 1, n))

def solve_lp(f, M):
    n = len(f)
    A = build_conv_matrix_sparse(f)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    try:

        if result.success:

def subgradient_step(f, lr, top_fraction=0.3):
    """One subgradient step considering top lags."""
    n = len(f)
    conv_f = np.convolve(f, f)
    M = conv_f.max()
    S = f.sum()
    
    # Find active lags (top fraction)
    n_active = max(1, int(len(conv_f) * top_fraction))
    active_indices = np.argsort(conv_f)[-n_active:]
    
    # Compute subgradient
    grad = np.zeros(n)
    for k in active_indices:
        for j in range(n):
            if 0 <= k - j < n:
                grad[j] += 2.0 * f[k - j]
    grad /= len(active_indices)
    
    # Center gradient (handle sum constraint)
    grad -= grad.mean()
    
    # Update
    new_f = f - lr * grad
    new_f = np.clip(new_f, 0, 1000)
    new_S = new_f.sum()
    if new_S > 0:
        new_f = new_f * (S / new_S)
    return new_f

"""
Multi-strategy optimization for minimizing 2n*max(conv(a,a))/sum(a)^2.
Combines LP direction search, multi-lag subgradient descent, random perturbations,
and block coordinate descent across multiple sequence lengths.
"""
import numpy as np
import time
from scipy.optimize import linprog
from scipy.sparse import csr_matrix


def build_conv_matrix_sparse(f):
    """Build sparse matrix A where (A@g)[k] = convolve(f,g)[k]."""
    n = len(f)
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        j_min = max(0, k - (n - 1))
        j_max = min(k, n - 1)
        for j in range(j_min, j_max + 1):
            rows.append(k)
            cols.append(j)
            vals.append(f[k - j])
    return csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))


def solve_lp(f, M):
    """Solve: max sum(g) s.t. g >= 0, convolve(f,g)_k <= M for all k."""
    n = len(f)
    A = build_conv_matrix_sparse(f)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    try:
        result = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
        if result.success:
            return result.x
    except Exception:
        pass
    return None


def multi_lag_subgradient(f, lr, top_frac=0.25):
    """Subgradient step using top-k active lags."""
    n = len(f)
    conv_f = np.convolve(f, f)
    S = f.sum()
    n_active = max(1, int(len(conv_f) * top_frac))
    active = np.argsort(conv_f)[-n_active:]

    grad = np.zeros(n)
    for k in active:
        j_min = max(0, k - (n - 1))
        j_max = min(k, n - 1)
        for j in range(j_min, j_max + 1):
            grad[j] += 2.0 * f[k - j]
    grad /= n_active

    # Remove component along uniform direction (handles sum constraint)
    grad -= grad.mean()

    new_f = f - lr * grad
    new_f = np.clip(new_f, 0.0, 1000.0)
    new_S = new_f.sum()
    if new_S > 0:
        new_f = new_f * (S / new_S)
    return new_f

    plateau[-t:] = ramp[::-1]

    # Phase 1: LP optimization
    t1 = min(deadline, time.monotonic() + (deadline - time.monotonic()) * 0.4)
    f1, s1 = lp_optimize(f, t1, step_init=0.12, verbose_prefix=verbose_prefix)
    if s1 < best_score:
        best_score = s1
        best_f = f1.copy()

    # Phase 2: Subgradient polish
    t2 = min(deadline, time.monotonic() + (deadline - time.monotonic()) * 0.25)
    f2, s2 = subgrad_optimize(f1, t2, lr_init=0.006, verbose_prefix=verbose_prefix)
    if s2 < best_score:
        best_score = s2
        best_f = f2.copy()

    # Phase 3: Shake and re-optimize
    t3 = min(deadline, time.monotonic() + (deadline - time.monotonic()) * 0.2)
    f3, s3 = shake_optimize(f2, t3, n_shakes=8, verbose_prefix=verbose_prefix)
    if s3 < best_score:
        best_score = s3
        best_f = f3.copy()

    # Phase 4: Final subgradient + LP sweep
    if time.monotonic() < deadline - 10:
        f4, s4 = subgrad_optimize(best_f, min(deadline - 5, time.monotonic() + 30),
                                  lr_init=0.004, verbose_prefix=verbose_prefix + "final ")
        if s4 < best_score:
            best_score = s4
            best_f = f4.copy()

    return best_f, best_score


def propose_candidate(budget_s=1500, **kwargs):
    """Main search function: multi-length, multi-strategy optimization."""
    t0 = time.monotonic()
    deadline = t0 + max(0.0, float(budget_s) - 10.0)

    rng = np.random.default_rng(42)
    best_sequence = None
    best_score = float('inf')

    # Load parent
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64).copy()
        parent = np.clip(parent, 0.0, 1000.0)
        score = evaluate_sequence(parent.tolist())
        if np.isfinite(score):
            best_score = score
            best_sequence = parent.copy()
            print(f"Parent: n={parent.size}, score={score:.6f}")
    except (NameError, Exception) as e:
        print(f"No parent: {e}")

    # Time budget allocation
    if best_sequence is not None:
        n_parent = len(best_sequence)
    else:
        n_parent = 512

    # Phase 1: Aggressively optimize the parent (or n=512 default)
    if best_sequence is not None:
        print("\n=== Phase 1: Optimizing parent ===")
        p1_deadline = t0 + (budget_s - 10) * 0.35
        f_opt, s_opt = full_optimize(best_sequence, p1_deadline, verbose_prefix="[parent] ")
        if s_opt < best_score:
            best_score = s_opt
            best_sequence = f_opt.copy()
        print(f"Phase 1 done: best={best_score:.6f}, time={time.monotonic()-t0:.1f}s")

    # Phase 2: Try other lengths
    print("\n=== Phase 2: Exploring other lengths ===")
    other_lengths = [256, 384, 640, 768, 1024]
    if n_parent in other_lengths:
        other_lengths.remove(n_parent)

    remaining = deadline - time.monotonic()
    per_length = max(30.0, remaining / (len(other_lengths) + 1))

    for n in other_lengths:
        if time.monotonic() > deadline - 30:
            break

        print(f"\n--- Trying n={n} (budget {per_length:.0f}s) ---")
        inits = generate_inits(n, rng)
        n_inits = len(inits)
        per_init = max(5.0, per_length / n_inits)

        for label, f_init in inits:
            if time.monotonic() > deadline - 15:
                break

            f_init = np.clip(f_init.astype(np.float64), 0.0, 1000.0)
            if f_init.sum() < 0.01:
                continue

            s_init = evaluate_sequence(f_init.tolist())
            if not np.isfinite(s_init):
                continue

            if s_init < best_score:
                best_score = s_init
                best_sequence = f_init.copy()
                print(f"  n={n} {label}: init {s_init:.6f} (NEW BEST)")

            # Quick LP + subgrad
            init_dl = min(deadline - 10, time.monotonic() + per_init)
            if per_init > 40:
                f1, s1 = lp_optimize(f_init, init_dl * 0.5, step_init=0.15)
                f2, s2 = subgrad_optimize(f1, init_dl, lr_init=0.005)
            else:
                f2, s2 = lp_optimize(f_init, init_dl, step_init=0.15)

            if s2 < best_score:
                best_score = s2
                best_sequence = f2.copy()
                print(f"  n={n} {label}: optimized {s2:.6f} (NEW BEST)")
            else:
                print(f"  n={n} {label}: optimized {s2:.6f} (best={best_score:.6f})")

    # Phase 3: Final refinement on the best found
    if best_sequence is not None and time.monotonic() < deadline - 50:
        print(f"\n=== Phase 3: Final refinement (n={len(best_sequence)}) ===")
        f_final, s_final = full_optimize(best_sequence, deadline, verbose_prefix="[final] ")
        if s_final < best_score:
            best_score = s_final
            best_sequence = f_final.copy()

        # One more shake attempt
        if time.monotonic() < deadline - 30:
            f_shake, s_shake = shake_optimize(best_sequence, deadline, n_shakes=5,
                                              verbose_prefix="[shake] ")
            if s_shake < best_score:
                best_score = s_shake
                best_sequence = f_shake.copy()

    if best_sequence is None:
        return [1.0] * 512

    elapsed = time.monotonic() - t0
    print(f"\n===== FINAL: n={len(best_sequence)}, score={best_score:.6f}, time={elapsed:.1f}s =====")
    return [float(x) for x in best_sequence]
