"""
Hybrid subgradient + coordinate descent optimization for minimizing
2n*max(convolve(a,a))/(sum(a))^2. Uses incremental convolution updates
for efficient coordinate descent, and adaptive subgradient steps for
coarse optimization.
"""
import numpy as np
import time


def compute_score(a):
    """Compute the score 2*n*max(conv)/(sum)^2."""
    n = len(a)
    conv = np.convolve(a, a)
    max_conv = np.max(conv)
    sum_a = np.sum(a)
    if sum_a < 0.01:
        return float('inf')
    return float(2.0 * n * max_conv / (sum_a * sum_a))


def subgradient_descent(a, deadline, verbose_prefix=""):
    """
    Subgradient descent to minimize max(conv) subject to sum(a)=1, a>=0.
    Uses adaptive step size and momentum.
    """
    n = len(a)
    a = a.copy().astype(np.float64)
    sum_a = np.sum(a)
    if sum_a <= 0:
        return a
    a = a / sum_a  # normalize to sum=1
    
    conv = np.convolve(a, a)
    best_score = 2.0 * n * np.max(conv)  # since sum=1
    best_a = a.copy()
    
    alpha = 0.01  # initial step size
    alpha_min = 1e-7
    beta = 0.9  # momentum
    momentum = np.zeros(n)
    no_improve = 0
    max_no_improve = 200
    it = 0
    
    while time.monotonic() < deadline:
        conv = np.convolve(a, a)
        max_val = np.max(conv)
        score = 2.0 * n * max_val
        
        if score < best_score:
            best_score = score
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        if no_improve > max_no_improve:
            break
        
        # Find top-k lags (use top 3 for a smoother gradient)
        k_top = min(3, 2*n-1)
        top_indices = np.argsort(conv)[-k_top:]
        
        # Compute combined subgradient
        g = np.zeros(n)
        for m in top_indices:
            m = int(m)
            j_min = max(0, m - (n - 1))
            j_max = min(m, n - 1)
            for j in range(j_min, j_max + 1):
                g[j] += 2.0 * a[m - j]
        
        # Normalize gradient
        g_norm = np.linalg.norm(g)
        if g_norm < 1e-15:
            break
        g = g / g_norm
        
        # Momentum update
        momentum = beta * momentum - alpha * g
        a_new = a + momentum
        a_new = np.clip(a_new, 0.0, 1000.0)
        sum_new = np.sum(a_new)
        if sum_new < 1e-10:
            alpha *= 0.5
            continue
        a_new = a_new / sum_new
        
        # Check if this is an improvement
        conv_new = np.convolve(a_new, a_new)
        score_new = 2.0 * n * np.max(conv_new)
        
        if score_new <= score:
            a = a_new
            # Keep or slightly increase alpha
            if no_improve == 0:
                alpha = min(alpha * 1.05, 0.1)
        else:
            alpha *= 0.8
            if alpha < alpha_min:
                break
            # Revert
            a = a  # keep old a
            momentum = beta * momentum  # decay momentum
        
        it += 1
        if it % 500 == 0:
            print(f"    {verbose_prefix}SGD iter {it}: score={score:.6f}, best={best_score:.6f}, alpha={alpha:.6f}")
    
    return best_a

def interp_to_length(a, new_n):
    """Interpolate sequence a to new length new_n."""
    old_n = len(a)
    if old_n == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, old_n)
    x_new = np.linspace(0, 1, new_n)
    return np.interp(x_new, x_old, a)


def generate_inits(n, parent, rng):
    """Generate various initializations for length n."""
    inits = []
    
    # 1. Interpolated parent
    if parent is not None and len(parent) > 0:
        inits.append(('parent_interp', interp_to_length(parent, n)))
    
    # 2. Uniform
    inits.append(('uniform', np.ones(n)))
    
    # 3. Random uniform
    inits.append(('random', rng.uniform(0.1, 1.0, size=n)))
    
    # 4. Sparse random (50% zeros)
    sparse = rng.uniform(0.5, 2.0, size=n)
    mask = rng.random(n) > 0.5
    sparse[~mask] = 0
    if np.sum(sparse) > 0:
        inits.append(('sparse', sparse))
    
    # 5. Cosine tapered
    idx = np.arange(n, dtype=np.float64)
    cos_t = 0.5 * (1 + np.cos(np.pi * idx / (n - 1)))
    inits.append(('cosine', cos_t + 0.1))
    
    # 6. Random with cosine taper
    r = rng.uniform(0.3, 1.0, size=n)

    two_tone = np.where(rng.random(n) > 0.5, 1.0, 0.2)
    inits.append(('two_tone', two_tone))
    
    return inits

def compute_score(a):

    alpha = 0.02
    alpha_min = 1e-8
    beta = 0.85
    momentum = np.zeros(n)
    no_improve = 0
    max_no_improve = 300
    it = 0
    k_top = 3

    while time.monotonic() < deadline:
        conv = np.convolve(a, a)
        max_val = float(np.max(conv))
        score = 2.0 * n * max_val

        if score < best_score - 1e-12:
            best_score = score
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        if no_improve > max_no_improve:
            break

        # Find top-k lags
        k_use = min(k_top, 2 * n - 1)
        top_indices = np.argsort(conv)[-k_use:]

        # Compute combined subgradient
        g = np.zeros(n)
        for m in top_indices:
            m = int(m)
            j_min = max(0, m - (n - 1))
            j_max = min(m, n - 1)
            # g[j] += 2 * a[m-j] for j in [j_min, j_max]
            # a[m-j] for j in [j_min, j_max] is a[m-j_min : m-j_max+1] reversed
            indices = np.arange(j_min, j_max + 1)
            g[indices] += 2.0 * a[m - indices]

        g_norm = float(np.linalg.norm(g))
        if g_norm < 1e-15:
            break
        g = g / g_norm

        # Momentum update
        momentum = beta * momentum - alpha * g
        a_new = a + momentum
        a_new = np.clip(a_new, 0.0, 1000.0)
        sum_new = float(np.sum(a_new))
        if sum_new < 1e-10:
            alpha *= 0.5
            continue
        a_new = a_new / sum_new

        # Evaluate
        conv_new = np.convolve(a_new, a_new)
        score_new = 2.0 * n * float(np.max(conv_new))

        if score_new <= score + 1e-12:
            a = a_new
            if no_improve == 0:
                alpha = min(alpha * 1.03, 0.15)
        else:
            alpha *= 0.75
            if alpha < alpha_min:
                break
            momentum = beta * momentum

        it += 1
        if it % 1000 == 0:
            print(f"    {verbose_prefix}SGD iter {it}: score={score:.6f}, best={best_score:.6f}, alpha={alpha:.6f}")

    return best_a


def coord_descent(a, deadline, verbose_prefix="", n_candidates=5):
    """
    Coordinate descent with vectorized incremental convolution updates.
    Minimizes max(conv) subject to sum(a)=1, a>=0.
    """
    n = len(a)
    a = a.copy().astype(np.float64)
    sum_a = float(np.sum(a))
    if sum_a <= 0:
        return a
    a = a / sum_a

    conv = np.convolve(a, a)
    best_score = 2.0 * n * float(np.max(conv))
    best_a = a.copy()

    no_improve_sweeps = 0
    max_no_improve_sweeps = 50
    rng = np.random.default_rng(7)
    sweep = 0

    while time.monotonic() < deadline:
        sweep += 1
        sweep_improved = 0
        order = rng.permutation(n)
        current_max = float(np.max(conv))
        peak_lag = int(np.argmax(conv))

        for j in order:
            if time.monotonic() > deadline:
                break
            j = int(j)
            old_val = a[j]

            # Determine candidates based on whether j is in active set of peak
            if 0 <= peak_lag - j < n:
                # Active: try reducing
                candidates = np.array([0.0, old_val * 0.25, old_val * 0.5, old_val * 0.75, old_val])
            else:
                # Inactive: try increasing or keeping
                avg_val = 1.0 / n
                candidates = np.array([old_val, old_val * 1.5, old_val * 2.0, old_val * 3.0 + 0.1 * avg_val, avg_val])

            # Filter valid candidates
            candidates = np.clip(candidates, 0.0, 100.0)
            candidates = candidates[np.abs(candidates - old_val) > 1e-10]
            if len(candidates) == 0:
                continue

            # Precompute unaffected max
            k_min = j
            k_max = min(j + n - 1, 2 * n - 2)
            if k_min > 0:
                left_max = float(np.max(conv[:k_min]))
            else:
                left_max = 0.0
            if k_max < 2 * n - 2:
                right_max = float(np.max(conv[k_max + 1:]))
            else:
                right_max = 0.0
            unaffected_max = max(left_max, right_max)

            best_candidate = old_val
            best_max = current_max

            # Vectorized: a[k-j] for k in [j, j+n-1] is just a[0], a[1], ..., a[n-1] = a
            for cand in candidates:
                delta = cand - old_val
                if abs(delta) < 1e-12:
                    continue
                # new_affected = conv[j:j+n] + 2*delta*a, with diagonal fix
                new_affected = conv[j:j + n] + 2.0 * delta * a
                new_affected[j] += delta * delta  # fix diagonal: add delta^2
                new_max = max(unaffected_max, float(np.max(new_affected)))

                if new_max < best_max - 1e-12:
                    best_max = new_max
                    best_candidate = cand

            if best_candidate != old_val:
                delta = best_candidate - old_val
                conv[j:j + n] += 2.0 * delta * a
                conv[2 * j] += delta * delta
                a[j] = best_candidate
                if best_max < current_max:
                    current_max = best_max
                    peak_lag = int(np.argmax(conv))
                sweep_improved += 1

        # Update global best
        final_max = float(np.max(conv))
        if final_max * 2.0 * n < best_score - 1e-12:
            best_score = 2.0 * n * final_max
            best_a = a.copy()
            no_improve_sweeps = 0
        else:
            no_improve_sweeps += 1

        if no_improve_sweeps > max_no_improve_sweeps:
            break

        if sweep % 10 == 0:
            print(f"    {verbose_prefix}CD sweep {sweep}: score={best_score:.6f}, improved={sweep_improved}")

    return best_a


def mass_transfer(a, deadline, verbose_prefix=""):
    """
    Mass transfer heuristic: move mass from positions in the active set
    of the peak lag to positions outside it.
    """
    n = len(a)
    a = a.copy().astype(np.float64)
    sum_a = float(np.sum(a))
    if sum_a <= 0:
        return a
    a = a / sum_a

    conv = np.convolve(a, a)
    best_score = 2.0 * n * float(np.max(conv))
    best_a = a.copy()
    it = 0

    while time.monotonic() < deadline:
        conv = np.convolve(a, a)
        peak_val = float(np.max(conv))
        peak_lag = int(np.argmax(conv))
        score = 2.0 * n * peak_val

        if score < best_score - 1e-12:
            best_score = score
            best_a = a.copy()

        # Active set: positions j where 0 <= peak_lag - j < n
        j_min = max(0, peak_lag - (n - 1))
        j_max = min(peak_lag, n - 1)
        active = np.arange(j_min, j_max + 1)
        inactive = np.setdiff1d(np.arange(n), active, assume_unique=True)

        if len(active) == 0 or len(inactive) == 0:
            break

        # Transfer: reduce active positions proportional to a[peak_lag-j]
        # Increase inactive positions uniformly
        t = 0.1  # transfer fraction
        transfer = t * np.sum(a[active])
        if transfer < 1e-12:
            break

        # Reduce active positions proportionally
        weights = a[peak_lag - active].copy()
        w_sum = float(np.sum(weights))
        if w_sum < 1e-15:
            weights = np.ones(len(active))
            w_sum = float(len(active))
        a[active] -= t * a[active] * (weights / w_sum) * 2.0
        a[active] = np.clip(a[active], 0.0, None)

        # Distribute to inactive
        a[inactive] += transfer / len(inactive)
        a = np.clip(a, 0.0, None)

        # Renormalize
        s = float(np.sum(a))
        if s < 1e-10:
            break
        a = a / s

        it += 1
        if it > 2000:
            break

    return best_a


def unified_optimize(a, deadline, verbose_prefix=""):
    """
    Combined optimization pipeline: mass transfer -> subgradient -> coord descent.
    Multiple rounds if time permits.
    """
    n = len(a)
    a = a.copy().astype(np.float64)
    sum_a = float(np.sum(a))
    if sum_a <= 0:
        return a
    a = a / sum_a

    best_a = a.copy()
    best_score = 2.0 * n * float(np.max(np.convolve(a, a)))

    # Phase 1: Mass transfer (10% time)
    mt_dl = time.monotonic() + 0.1 * (deadline - time.monotonic())
    a_mt = mass_transfer(a, mt_dl, verbose_prefix + "MT")
    s = 2.0 * n * float(np.max(np.convolve(a_mt, a_mt)))
    if s < best_score:
        best_score = s
        best_a = a_mt.copy()

    # Phase 2: Subgradient (35% time)
    sg_dl = time.monotonic() + 0.35 * (deadline - time.monotonic())
    a_sg = subgradient_descent(best_a, sg_dl, verbose_prefix + "SGD")
    s = 2.0 * n * float(np.max(np.convolve(a_sg, a_sg)))
    if s < best_score:
        best_score = s
        best_a = a_sg.copy()

    # Phase 3: Coordinate descent (35% time)
    cd_dl = time.monotonic() + 0.35 * (deadline - time.monotonic())
    a_cd = coord_descent(best_a, cd_dl, verbose_prefix + "CD")
    s = 2.0 * n * float(np.max(np.convolve(a_cd, a_cd)))
    if s < best_score:
        best_score = s
        best_a = a_cd.copy()

    # Phase 4: Second round if time remains (20% time)
    if time.monotonic() < deadline - 10:
        remaining = deadline - time.monotonic()
        sg2_dl = time.monotonic() + 0.4 * remaining
        a2 = subgradient_descent(best_a, sg2_dl, verbose_prefix + "SGD2")
        s = 2.0 * n * float(np.max(np.convolve(a2, a2)))
        if s < best_score:
            best_score = s
            best_a = a2.copy()

        if time.monotonic() < deadline - 5:
            a3 = coord_descent(best_a, deadline, verbose_prefix + "CD2")
            s = 2.0 * n * float(np.max(np.convolve(a3, a3)))
            if s < best_score:
                best_score = s
                best_a = a3.copy()

    return best_a


def interp_to_length(a, new_n):
    """Interpolate sequence a to new length."""
    old_n = len(a)
    if old_n == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, old_n)
    x_new = np.linspace(0, 1, new_n)
    return np.interp(x_new, x_old, a)


def generate_inits(n, parent, rng):
    """Generate various initializations for length n."""
    inits = []

    if parent is not None and len(parent) > 0:
        inits.append(('parent', interp_to_length(parent, n)))

    inits.append(('uniform', np.ones(n)))

    inits.append(('rand', rng.uniform(0.2, 1.0, size=n)))

    # Sparse
    sparse = rng.uniform(0.5, 2.0, size=n)
    mask = rng.random(n) > 0.4
    sparse[~mask] = 0.0
    if float(np.sum(sparse)) > 0.01:
        inits.append(('sparse', sparse))

    # Cosine tapered random
    idx = np.arange(n, dtype=np.float64)
    cos_t = 0.5 * (1.0 + np.cos(np.pi * idx / max(1, n - 1)))
    inits.append(('rand_cos', rng.uniform(0.3, 1.0, size=n) * cos_t + 0.05))

    # Two-level
    two = np.where(rng.random(n) > 0.5, 1.0, 0.15)
    inits.append(('two_level', two))

    return inits


def propose_candidate(budget_s=1500, **kwargs):
    """
    Main entry point. Searches for the best sequence minimizing
    2*n*max(convolve(a,a))/(sum(a))^2 using hybrid subgradient +
    coordinate descent + mass transfer optimization across multiple
    lengths and initializations.
    """
    t0 = time.monotonic()
    deadline = t0 + max(1.0, float(budget_s) - 15.0)

    best_sequence = None
    best_score = float('inf')
    rng = np.random.default_rng(42)

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64).copy()
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.all(np.isfinite(p)):
            parent = p
            score = evaluate_sequence(parent.tolist())
            if np.isfinite(score):
                best_score = score
                best_sequence = parent.copy()
                print(f"Parent n={parent.size}: score={score:.6f}")
            else:
                parent = None
    except (NameError, Exception) as e:
        print(f"No parent: {e}")

    # Lengths to try (parent length first if available)
    if parent is not None:
        lengths = [len(parent), 256, 768, 1024, 384, 1536, 640, 2048]
    else:
        lengths = [512, 256, 768, 1024, 384, 1536, 640, 2048]

    # Remove duplicates
    seen = set()
    unique_lengths = []
    for ln in lengths:
        if ln not in seen:
            seen.add(ln)
            unique_lengths.append(ln)
    lengths = unique_lengths

    n_inits_per_length = 3  # Limit inits per length to manage time

    for li, n in enumerate(lengths):
        if time.monotonic() > deadline - 30:
            print("Time running out, stopping.")
            break

        remaining = deadline - time.monotonic()
        combos_left = (len(lengths) - li) * n_inits_per_length
        time_per_init = max(10.0, remaining / max(1, combos_left) * 0.85)

        print(f"\n--- n={n} (time_per_init~{time_per_init:.0f}s, remaining={remaining:.0f}s) ---")

        inits = generate_inits(n, parent, rng)
        if len(inits) > n_inits_per_length:
            # Prioritize parent, then random, then others
            inits = inits[:n_inits_per_length]

        for init_name, a_init in inits:
            if time.monotonic() > deadline - 20:
                break

            a_init = np.clip(a_init.astype(np.float64), 0.0, 1000.0)
            if float(np.sum(a_init)) < 0.01:
                continue

            init_score = compute_score(a_init)
            if not np.isfinite(init_score):
                continue

            if init_score < best_score:
                best_score = init_score
                best_sequence = a_init.copy()

            print(f"  n={n} {init_name}: init={init_score:.6f}", end="", flush=True)

            opt_deadline = min(deadline - 5, time.monotonic() + time_per_init)
            a_opt = unified_optimize(a_init, opt_deadline, verbose_prefix=f"[n={n} {init_name}] ")
            opt_score = compute_score(a_opt)

            if np.isfinite(opt_score) and opt_score < best_score:
                best_score = opt_score
                best_sequence = a_opt.copy()

            print(f" -> opt={opt_score:.6f} (best={best_score:.6f})")

    # Final validation
    if best_sequence is not None:
        best_seq = [float(max(0.0, min(1000.0, x))) for x in best_sequence]
        final_score = evaluate_sequence(best_seq)
        if np.isfinite(final_score):
            best_score = final_score
        print(f"\nFINAL: n={len(best_sequence)}, score={best_score:.6f}, time={time.monotonic()-t0:.1f}s")
    else:
        best_seq = [1.0] * 128
        print(f"\nFALLBACK: n=128, time={time.monotonic()-t0:.1f}s")

    return best_seq
