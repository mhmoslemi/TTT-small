                           top_k=10, soft_beta=5.0, deadline=None, verbose=False):
    """Mirror descent with multiplicative updates and soft top-k weighting."""

    a /= a.sum()

    a /= a.sum()

    t = t_init

    n = len(a0)

        s = ax.sum()
        ax = ax / s
        b = fast_conv(ax)
        # Smooth max using log-sum-exp
        beta = 50.0
        m = b.max()
        val = m + np.log(np.sum(np.exp(beta * (b - m)))) / beta
        return float(val)
    
    res = minimize(smooth_obj, a0, method='L-BFGS-B',
                  bounds=[(0, None)] * n,

        a_out /= a_out.sum()
    else:
        a_out = a0

Actually, coordinate descent is interesting! For a fixed set of all coordinates except a_i, the objective max_k (a*a)_k is a piecewise linear function of a_i (since (a*a)_k = a_i * a_{k-i} + (terms not involving a_i)). """
Minimizes J(a) = 2*n*max(convolve(a,a))/(sum(a))^2 using:
1. Adaptive multiplicative (mirror) descent with soft top-k weighting and adaptive learning rate
2. LP-based best response for structured updates
3. L-BFGS-B gradient refinement with log-sum-exp smoothing
4. Multi-scale exploration across sequence lengths (256 to 2048)
Key improvements: adaptive soft-attention on peaks, perturbation-restart to escape
local minima, better time allocation, coordinate-wise polish, and diverse n exploration.
"""

import numpy as np
from scipy.optimize import linprog, minimize
import time


def fast_conv(a):
    """Full self-convolution via FFT, O(n log n)."""
    n = len(a)
    s = 1
    while s < 2 * n - 1:
        s <<= 1
    fa = np.fft.rfft(a, s)
    return np.fft.irfft(fa * fa, s)[:2 * n - 1]


def obj_value(a):
    """Compute J(a) = 2*n*max(conv)/sum^2."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-10:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))


def multiplicative_descent(a0, n_iter=5000, eta_init=0.5, eta_decay=0.999,
                           top_k=10, soft_beta=5.0, deadline=None, verbose=False):
    """Mirror descent with multiplicative updates and soft top-k weighting."""
    n = len(a0)
    a = a0.copy().astype(np.float64)
    a = np.maximum(a, 1e-10)
    a /= a.sum()

    best_a = a.copy()
    best_max = float(fast_conv(a).max())
    eta = eta_init

    for it in range(n_iter):
        if deadline is not None and time.monotonic() > deadline:
            break
        if it > 0 and it % 300 == 0:
            eta = max(0.003, eta * eta_decay)

        b = fast_conv(a)
        M = float(b.max())

        k = min(top_k, 2 * n - 1)
        top_idx = np.argpartition(b, -k)[-k:].astype(int)
        top_vals = b[top_idx]

        # Adaptive soft weighting
        spread = max(M - top_vals.min(), 1e-15)
        adaptive_beta = soft_beta * max(1.0, 5.0 / (spread * n + 1e-10))
        weights = np.exp(-adaptive_beta * (M - top_vals))
        weights /= weights.sum()

        # Compute weighted gradient
        grad = np.zeros(n)
        for i in range(k):
            pos = int(top_idx[i])
            jl = max(0, pos - n + 1)
            jh = min(n - 1, pos)
            if jl <= jh:
                grad[jl:jh + 1] += weights[i] * a[pos - jh:pos - jl + 1][::-1]

        # Multiplicative update
        a = a * np.exp(-eta * grad)
        a = np.maximum(a, 1e-15)
        a /= a.sum()

        if verbose and (it % 200 == 0 or it == n_iter - 1):
            curr_max = float(fast_conv(a).max())
            if curr_max < best_max:
                best_max = curr_max
                best_a = a.copy()
                print(f"    MD it={it}, J={2 * n * best_max:.6f}, eta={eta:.4f}", flush=True)

    curr_max = float(fast_conv(a).max())
    if curr_max < best_max:
        best_max = curr_max
        best_a = a.copy()

    return best_a, best_max


def build_conv_matrix(a):
    """Build matrix A where A[k,j] = a[k-j] if 0<=k-j<=n-1, else 0."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for k in range(2 * n - 1):
        jl = max(0, k - n + 1)
        jh = min(n - 1, k)
        if jl <= jh:
            A[k, jl:jh + 1] = a[k - jh:k - jl + 1][::-1]
    return A


def lp_best_response(a, M=None):
    """LP: max sum(g) s.t. (a*g)_k <= M, g >= 0. Returns g or None."""
    n = len(a)
    if M is None:
        M = float(fast_conv(a).max())

    A = build_conv_matrix(a)
    c = -np.ones(n)
    r = linprog(c, A_ub=A, b_ub=np.full(2 * n - 1, M),
                bounds=[(0, None)] * n, method='highs')
    if r.success and r.x is not None and r.x.sum() > 1e-10:
        return r.x
    return None


def lp_alternation(a0, max_iters=80, t_init=0.5, t_decay=0.96, deadline=None, verbose=False):
    """LP-based alternating optimization."""
    n = len(a0)
    a = a0.copy().astype(np.float64)
    a = np.maximum(a, 1e-10)
    a /= a.sum()

    best_a = a.copy()
    best_max = float(fast_conv(a).max())
    t = t_init

    for it in range(max_iters):
        if deadline is not None and time.monotonic() > deadline:
            break

        b = fast_conv(a)
        M = float(b.max())

        g = lp_best_response(a, M)
        if g is None:
            break

        g = np.maximum(g, 0)
        g_sum = g.sum()
        if g_sum < 1e-10:
            break
        g /= g_sum

        a_new = (1.0 - t) * a + t * g
        a_new = np.maximum(a_new, 1e-10)
        a_new /= a_new.sum()
        a = a_new

        t = max(0.01, t * t_decay)

        curr_max = float(fast_conv(a).max())
        if curr_max < best_max:
            best_max = curr_max
            best_a = a.copy()
            if verbose:
                print(f"    LP it={it}, J={2 * n * best_max:.6f}, t={t:.3f}", flush=True)

    return best_a, best_max


def gradient_refine(a0, deadline=None, verbose=False):
    """L-BFGS-B refinement using log-sum-exp smoothed surrogate."""
    n = len(a0)
    a0 = np.maximum(a0, 0).astype(np.float64)
    if a0.sum() < 1e-10:
        return a0.copy(), float(fast_conv(a0).max())
    a0 = a0 / a0.sum()

    def smooth_obj(x):
        ax = np.maximum(x, 1e-12)
        s = ax.sum()
        if s < 1e-8:
            return 1e10
        ax = ax / s
        b = fast_conv(ax)
        # Log-sum-exp smoothing of max
        for beta in [30.0, 80.0]:
            m = b.max()
            val = m + np.log(np.sum(np.exp(beta * (b - m)))) / beta
            if beta > 50:
                break
        return float(val)

    try:
        res = minimize(smooth_obj, a0, method='L-BFGS-B',
                       bounds=[(0, None)] * n,
                       options={'maxiter': 300, 'max_fun': 3000, 'ftol': 1e-15, 'disp': False})
        a_out = np.maximum(res.x, 0)
    except Exception:
        a_out = a0.copy()

    if a_out.sum() < 1e-10:
        a_out = a0.copy()
    else:
        a_out /= a_out.sum()

    curr_max = float(fast_conv(a_out).max())
    return a_out, curr_max


def resample_sequence(a, new_n):
    """Resample a 1D sequence to a new length using linear interpolation."""
    old_n = len(a)
    if old_n == new_n:
        return a.copy()
    t_old = np.linspace(0, 1, old_n)
    t_new = np.linspace(0, 1, new_n)
    return np.maximum(np.interp(t_new, t_old, a), 0.01)


def update_best(a, container):
    """Evaluate and update best if improved. container = [best_array, best_score]."""
    a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
    if a.sum() < 0.01:
        return

    if np.isfinite(score) and score < container[1]:
        container[0] = a.copy()
        container[1] = score
        print(f"  *** NEW BEST: J={score:.6f} (n={a.size})", flush=True)


def generate_initializations(n, rng, best=None):
    """Generate diverse initializations for a given length n."""
    inits = []
    if best is not None:
        inits.append(resample_sequence(best, n))

    inits.append(np.ones(n) / n)

    t_arr = np.linspace(0, 1, n, endpoint=False)
    # U-shaped (more weight at edges)
    inits.append((1.0 + 0.8 * (2 * t_arr - 1) ** 2))
    # Cosine modulation
    inits.append(1.0 + 0.4 * np.cos(4 * np.pi * t_arr))
    # Chirp
    inits.append(1.0 + 0.3 * np.cos(2 * np.pi * t_arr ** 2 * 3))
    # Random
    inits.append(rng.uniform(0.3, 1.5, n))
    # Bimodal
    center = n // 2
    inits.append(np.exp(-2.0 * ((np.arange(n) - center) / (n / 4.0)) ** 2))

    # Normalize all
    result = []
    for init in inits:
        init = np.maximum(np.asarray(init, dtype=np.float64), 0.01)
        init /= init.sum()
        result.append(init)
    return result
