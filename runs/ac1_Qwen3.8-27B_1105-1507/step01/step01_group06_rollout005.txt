def fast_conv(a):
    """Full self-convolution via FFT, O(n log n)."""
    n = len(a)
    s = 1
    while s < 2 * n - 1:
        s <<= 1
    fa = np.fft.rfft(a, s)
    return np.fft.irfft(fa * fa, s)[:2 * n - 1]


def cross_conv(a, b):
    """Full cross-convolution a * b via FFT."""
    n, m = len(a), len(b)
    s = 1
    while s < n + m - 1:
        s <<= 1
    fa = np.fft.rfft(a, s)
    fb = np.fft.rfft(b, s)
    return np.fft.irfft(fa * fb, s)[:n + m - 1]


def build_A_matrix(a):
    """Build convolution matrix A where A[k,j] = a[k-j] if 0<=k-j<n."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):

def lp_solve_full(a, M):

    n = len(a)
    c = -np.ones(n)
    A = build_A_matrix(a)

            t_mix = max(step_min, step_init * (step_decay ** it))

        g = g / g.sum()
        a_new = (1.0 - t_mix) * a + t_mix * g
        a_new = np.maximum(a_new, 0.0)
        s_new = a_new.sum()
        if s_new < 1e-10:
            break
        a_new = a_new / s_new

        cur_obj = obj_value(a_new)
        if cur_obj < best_obj:
            best_obj = cur_obj
            best_a = a_new.copy()

        a = a_new

        # Periodic perturbation to escape local minima
        if (it + 1) % perturb_every == 0 and perturb_std > 0:
            noise = rng.normal(0, perturb_std, n)
            a = a + noise
            a = np.maximum(a, 0.0)
            s = a.sum()
            if s < 1e-10:
                a = best_a.copy()
            else:
                a = a / s

    return best_a


def grad_refine(a, deadline, max_iter=300):
    """L-BFGS-B gradient refinement."""
    n = len(a)
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-10:
        return a
    a = a / s

    def _obj(x):
        ax = np.maximum(x, 0.0)
        sx = ax.sum()
        if sx < 1e-8:
            return 1e10
        return float(fast_conv(ax / sx).max())

    try:
        res = minimize(_obj, a, method='L-BFGS-B',
                       bounds=[(0, None)] * n,
                       options={'maxiter': max_iter, 'max_fun': max_iter * 20,
                                'ftol': 1e-15, 'gtol': 1e-12, 'disp': False})
        if res.fun < 1e10:
            return np.maximum(res.x, 0.0)
    except Exception:
        pass
    return a


def resample(a, new_n):
    """Resample sequence a to length new_n using linear interpolation."""
    old_n = len(a)
    if old_n == new_n:
        return a.copy()
    t_old = np.linspace(0, 1, old_n)
    t_new = np.linspace(0, 1, new_n)
    return np.maximum(np.interp(t_new, t_old, a), 0.0)


def propose_candidate(seed=42, budget_s=1500, **kwargs):
    """Main search: LP mirror descent + perturbation + gradient refinement."""
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + float(budget_s) - 20.0
    best = None
    best_score = float('inf')

    def try_update(a):
        nonlocal best, best_score
        a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
        if a.sum() < 0.01:
            return
        score = float(evaluate_sequence(a.tolist()))
        if np.isfinite(score) and score < best_score:
            best = a.copy()
            best_score = score
            print(f"  ** J={score:.6f} (n={a.size})", flush=True)

    # Try parent sequence
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        if p.ndim == 1 and p.size > 0 and np.all(np.isfinite(p)):
            try_update(p)
    except (NameError, AttributeError, Exception):
        pass

    t_start = time.monotonic()

    # === Phase 1: LP mirror descent across multiple n values ===
    configs = [
        # (n, max_iters, use_block, perturb_every, perturb_std, step_init)
        (256, 150, False, 30, 0.10, 0.5),
        (512, 200, False, 40, 0.08, 0.4),
        (1024, 120, False, 50, 0.06, 0.35),
        (2048, 60, True, 40, 0.05, 0.3),
    ]

    for n, max_iters, use_blk, p_every, p_std, s_init in configs:
        if time.monotonic() > deadline:
            break
        print(f"\n=== LP n={n} (block={use_blk}) ===", flush=True)
        t0 = time.monotonic()

        # Generate diverse initializations
        inits = []
        t_arr = np.linspace(0, 1, n, endpoint=False)

        # 1. Uniform
        inits.append(np.ones(n))
        # 2. Random
        inits.append(rng.uniform(0.3, 1.0, n))
        # 3. Cosine
        inits.append(0.5 + 0.5 * np.cos(2 * np.pi * t_arr))
        # 4. Hamming window
        inits.append(0.54 + 0.46 * np.cos(2 * np.pi * t_arr))
        # 5. Inverted triangle
        inits.append(1.0 - np.abs(2.0 * t_arr - 1.0))
        # 6. Chirp (quadratic phase)
        inits.append(0.5 + 0.5 * np.cos(4 * np.pi * t_arr ** 2))
        # 7. From best (resampled)
        if best is not None:
            inits.append(np.maximum(resample(best, n), 0.01))
        # 8. From parent (resampled)
        try:
            p = np.asarray(height_sequence_1, dtype=np.float64)
            if p.ndim == 1 and p.size > 0:
                inits.append(np.maximum(resample(p, n), 0.01))
        except (NameError, AttributeError, Exception):
            pass

        for a0 in inits:
            if time.monotonic() > deadline:
                break
            a0 = np.maximum(a0, 0.01)
            a0 = a0 / a0.sum()

            a_result = lp_mirror_descent(
                a0, max_iters, deadline,
                use_block=use_blk,
                block_size=256 if use_blk else 0,
                perturb_every=p_every,
                perturb_std=p_std,
                step_init=s_init
            )
            try_update(a_result)

            # Gradient refinement
            if time.monotonic() < deadline - 60:
                a_grad = grad_refine(a_result, deadline)
                try_update(a_grad)

        elapsed = time.monotonic() - t0
        print(f"  Phase n={n}: {elapsed:.0f}s, current best={best_score:.6f}", flush=True)

    # === Phase 2: Perturbation + reoptimize on best ===
    if best is not None and time.monotonic() < deadline:
        n_best = best.size
        print(f"\n=== Perturbation & Reoptimize: n={n_best} ===", flush=True)
        t0 = time.monotonic()
        rng2 = np.random.default_rng(seed + 100)

        num_pert = 15
        for pi in range(num_pert):
            if time.monotonic() > deadline:
                break
            a = best.copy()
            # Different perturbation strategies
            strategy = pi % 4
            if strategy == 0:
                # Gaussian noise
                a = a + rng2.normal(0, 0.15, n_best)
            elif strategy == 1:
                # Multiplicative noise
                a = a * rng2.uniform(0.5, 1.5, n_best)
            elif strategy == 2:
                # Swap mass between random pairs
                for _ in range(n_best // 10):
                    i, j = rng2.choice(n_best, 2, replace=False)
                    delta = 0.3 * a[i]
                    a[i] -= delta
                    a[j] += delta
            else:
                # Block perturbation
                block_start = rng2.integers(0, n_best - 64)
                block_end = block_start + 64
                a[block_start:block_end] *= rng2.uniform(0.3, 2.0)

            a = np.maximum(a, 0.0)
            s = a.sum()
            if s < 1e-10:
                continue
            a = a / s

            # Short LP optimization
            use_blk = n_best > 1024
            a_result = lp_mirror_descent(
                a, 40, deadline,
                use_block=use_blk,
                block_size=256 if use_blk else 0,
                perturb_every=999,  # no perturbation in short runs
                perturb_std=0.0,
                step_init=0.3
            )
            try_update(a_result)

        print(f"  Perturb: {time.monotonic()-t0:.0f}s, best={best_score:.6f}", flush=True)

    # === Phase 3: Final intensive polish ===
    if best is not None and time.monotonic() < deadline:
        n_pol = best.size
        print(f"\n=== Final Polish: n={n_pol} ===", flush=True)
        t0 = time.monotonic()

        a = best.copy()
        a = a / a.sum()

        # Many small LP steps
        num_steps = 80 if n_pol <= 512 else 40
        for it in range(num_steps):
            if time.monotonic() > deadline:
                break
            b = fast_conv(a)
            M = float(b.max())
            g = lp_solve_full(a, M)
            if g is None:
                break
            g = g / g.sum()
            t_mix = 0.005
            a_new = (1 - t_mix) * a + t_mix * g
            a_new = np.maximum(a_new, 0.0)
            a = a_new / a_new.sum()

        try_update(a)

        # Gradient polish
        if time.monotonic() < deadline - 30:
            a_grad = grad_refine(a, deadline, max_iter=200)
            try_update(a_grad)

        # One more short LP round after gradient
        if time.monotonic() < deadline - 20:
            a2 = grad_refine(a, deadline, max_iter=100)
            for it in range(30):
                if time.monotonic() > deadline:
                    break
                b = fast_conv(a2)
                M = float(b.max())
                g = lp_solve_full(a2, M)
                if g is None:
                    break
                g = g / g.sum()
                a2 = (0.995 * a2 + 0.005 * g)
                a2 = np.maximum(a2, 0.0)
                a2 = a2 / a2.sum()
            try_update(a2)

        print(f"  Polish: {time.monotonic()-t0:.0f}s, best={best_score:.6f}", flush=True)

    # === Phase 4: Try once more at a different n if time remains ===
    if best is not None and time.monotonic() < deadline - 100:
        n_alt = best.size * 2 if best.size <= 1024 else best.size // 2
        if n_alt in [256, 512, 1024, 2048] and n_alt != best.size:
            print(f"\n=== Alt n={n_alt} ===", flush=True)
            a_init = np.maximum(resample(best, n_alt), 0.01)
            a_init = a_init / a_init.sum()
            a_res = lp_mirror_descent(a_init, 60, deadline,
                                      use_block=(n_alt > 1024),
                                      block_size=256,
                                      perturb_every=20, perturb_std=0.1,
                                      step_init=0.4)
            try_update(a_res)
            a_grad = grad_refine(a_res, deadline)
            try_update(a_grad)

    if best is None:
        best = np.ones(256)

    total_time = time.monotonic() - t_start
    print(f"\nFINAL: J={best_score:.6f}, n={best.size}, time={total_time:.0f}s", flush=True)
    return [float(x) for x in best]
