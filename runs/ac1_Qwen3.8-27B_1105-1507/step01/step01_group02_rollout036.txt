"""
Minimizes 2*n*max(convolve(a,a))/sum(a)^2 using softmax-parameterized
log-sum-exp smoothed optimization with Adam, followed by subgradient
refinement and LP-based improvements.
"""

import numpy as np
import time
from scipy.optimize import linprog

# Global for parent sequence
# height_sequence_1 will be available

def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    N = 1 << (size - 1).bit_length()
    fa = np.fft.rfft(a, N)
    fc = fa * fa
    result = np.fft.irfft(fc, N)[:size]
    result[np.abs(result) < 1e-15] = 0.0
    result[result < 0] = 0.0
    return result

def project_simplex(v):
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cv = np.cumsum(u) - 1.0
    rho_candidates = np.where(u * np.arange(1, n + 1) > cv)[0]
    if len(rho_candidates) == 0:
        return np.ones(n) / n
    rho = rho_candidates[-1]
    theta = cv[rho] / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def softmax(x):
    x_shifted = x - np.max(x)
    e = np.exp(x_shifted)
    return e / np.sum(e)

def compute_score(a):
    n = len(a)
    c = fast_convolve(a)
    s = np.sum(a)
    if s < 0.01:
        return float('inf')
    return float(2.0 * n * np.max(c) / (s * s))

def smooth_loss_and_grad(a, tau):
    n = len(a)
    c = fast_convolve(a)
    c_max = np.max(c)
    c_shifted = c - c_max
    exp_vals = np.exp(c_shifted / tau)
    sum_exp = np.sum(exp_vals)
    loss = c_max + tau * np.log(sum_exp)
    weights = exp_vals / sum_exp
    grad_a = 2.0 * np.correlate(weights, a, mode='valid')
    return loss, grad_a

def optimize_with_adam(x0, n, deadline, tau_start, tau_end, lr=0.1, total_iters=50000):
    """Adam optimization in softmax parameter space."""
    x = x0.copy()
    m = np.zeros(n)
    v = np.zeros(n)
    beta1, beta2, eps = 0.8, 0.999, 1e-8
    t = 0
    best_loss = float('inf')
    best_x = x.copy()
    best_actual_score = float('inf')
    
    stuck_count = 0
    max_stuck = 2000
    
    while time.time() < deadline and t < total_iters:
        t += 1
        frac = min(1.0, t / max(1, total_iters))
        tau = tau_start * (tau_end / tau_start) ** frac
        
        a = softmax(x)
        loss, grad_a = smooth_loss_and_grad(a, tau)
        
        # Chain rule through softmax
        dot = np.dot(grad_a, a)
        grad_x = a * (grad_a - dot)
        
        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad_x
        v = beta2 * v + (1.0 - beta2) * grad_x * grad_x
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        x = x - lr * m_hat / (np.sqrt(v_hat) + eps)
        
        # Track best
        if t % 100 == 0:
            a_current = softmax(x)
            actual = compute_score(a_current)
            if actual < best_actual_score:
                best_actual_score = actual
                best_x = x.copy()
        
        if loss < best_loss:
            best_loss = loss
            stuck_count = 0
        else:
            stuck_count += 1
        
        if stuck_count > max_stuck:
            # Increase tau temporarily (reheat)
            tau = tau_start * 0.5
            stuck_count = 0
            # Also perturb
            x += 0.5 * np.random.standard_normal(n)
    
    # Final evaluation
    a_final = softmax(best_x)
    final_score = compute_score(a_final)
    return a_final, final_score, best_x

        threshold = peak * 0.99
        peak_idx = np.where(c >= threshold)[0]
        if len(peak_idx) > 20:

            peak_idx = peak_idx[top]

        grad = np.zeros(n)

            grad[lo:hi+1] += 2.0 * a[k-hi:k-lo+1][::-1]
        grad /= max(1, len(peak_idx))
        
        a = project_simplex(a - lr * grad)
    
    return best_a, best_score

def lp_refine(a, deadline, max_iters=50):
    """LP-based refinement."""
    n = len(a)
    if n > 2000:
        return a, compute_score(a)
    
    a = a / max(a.sum(), 1e-10)
    best_a = a.copy()
    best_score = compute_score(a)
    
    for it in range(max_iters):
        if time.time() > deadline:
            break
        
        c = fast_convolve(a)
        P = np.max(c)
        
        # Build convolution matrix
        A = np.zeros((2 * n - 1, n))
        for k in range(2 * n - 1):
            lo = max(0, k - n + 1)
            hi = min(n - 1, k)
            A[k, lo:hi+1] = a[k-hi:k-lo+1][::-1]
        
        c_obj = -np.ones(n)
        b_ub = np.full(2 * n - 1, P)
        
        try:
            res = linprog(c_obj, A_ub=A, b_ub=b_ub, 
                         bounds=[(0.0, None)] * n,
                         method='highs',
                         options={'time_limit': 5.0})
            if res.success:
                b = res.x
                b_sum = b.sum()
                if b_sum > 1e-8:
                    b = b / b_sum
                    # Mix
                    for t_mix in [0.3, 0.5, 0.7]:
                        mixed = (1 - t_mix) * a + t_mix * b
                        mixed = project_simplex(mixed)
                        score = compute_score(mixed)
                        if score < best_score:
                            best_score = score
                            best_a = mixed.copy()
                            a = mixed
                    if a is not best_a:  # If we improved
                        continue
        except Exception:
            pass
        
        # If LP didn't help, break
        break
    
    return best_a, best_score

def alt_projection_init(n, rng, n_iter=50):
    """Alternating projection for flat spectrum initialization."""
    N = 1 << (2 * n - 2).bit_length()
    phases = rng.uniform(0, 2 * np.pi, N // 2 + 1)
    F = np.exp(1j * phases)
    F[0] = 1.0 + 0j
    
    a = np.ones(n) / n
    for _ in range(n_iter):

"""
Minimizes 2*n*max(convolve(a,a))/sum(a)^2 using a multi-phase approach:
1) Log-sum-exp smoothed optimization via projected gradient with momentum
2) Subgradient refinement on the true objective
3) LP-based improvement steps
Uses continuation (decreasing temperature), multiple initializations,
and adaptive step sizes. Tries several sequence lengths.

import numpy as np
import time
from scipy.optimize import linprog


def fast_convolve(a):
    """Compute full linear convolution of a with itself using FFT."""
    n = len(a)
    size = 2 * n - 1
    N = 1 << (size - 1).bit_length()
    fa = np.fft.rfft(a, N)
    fc = fa * fa
    result = np.fft.irfft(fc, N)[:size]
    result[result < 0] = 0.0
    return result


def project_simplex(v):
    """Project vector v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cv = np.cumsum(u) - 1.0
    rho_candidates = np.where(u * np.arange(1, n + 1) > cv)[0]
    if len(rho_candidates) == 0:
        return np.ones(n) / n
    rho = rho_candidates[-1]
    theta = cv[rho] / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def compute_score(a):
    """Compute the objective value for a normalized sequence."""
    n = len(a)
    c = fast_convolve(a)
    s = np.sum(a)
    if s < 0.01:
        return float('inf')
    return float(2.0 * n * np.max(c) / (s * s))


def smooth_grad(a, tau):
    """
    Compute the gradient of the log-sum-exp smoothed max of conv(a,a).
    Returns (smoothed_loss, gradient_wrt_a).
    """
    n = len(a)
    c = fast_convolve(a)
    c_max = np.max(c)
    c_shifted = c - c_max
    exp_vals = np.exp(c_shifted / tau)
    sum_exp = np.sum(exp_vals)
    loss = c_max + tau * np.log(sum_exp)
    weights = exp_vals / sum_exp
    # grad[m] = 2 * sum_l weights[m+l] * a[l]
    grad = 2.0 * np.correlate(weights, a, mode='valid')
    return loss, grad


def optimize_smooth_projected(a_init, n, deadline, tau_levels, iters_per_level=5000):

    best_score = compute_score(a)

            a_new = project_simplex(a - v)

            a = a_new

def subgradient_refine(a_init, deadline, max_iters=500000):
    """Subgradient descent on the true (unsmoothed) objective."""
    n = len(a_init)
    a = project_simplex(a_init.astype(np.float64) / max(a_init.sum(), 1e-10))
    best_a = a.copy()
    best_score = compute_score(a)
    rng_local = np.random.default_rng(777)

    lr = 2.0 / n
    lr_base = lr
    momentum = 0.7
    v = np.zeros(n)
    stall = 0

    for it in range(max_iters):
        if time.time() > deadline:
            break

        c = fast_convolve(a)
        peak = np.max(c)

            best_a = a.copy()

def lp_refine(a_init, deadline, max_attempts=30):
    """LP-based refinement: maximize sum(b) s.t. conv(a,b) <= P, b >= 0."""
    n = len(a_init)
    if n > 1500:
        return a_init, compute_score(a_init)

    a = project_simplex(a_init.astype(np.float64) / max(a_init.sum(), 1e-10))
    best_a = a.copy()
    best_score = compute_score(a)

    for attempt in range(max_attempts):
        if time.time() > deadline:
            break

        c = fast_convolve(a)
        P = np.max(c)

        # Build convolution matrix
        A = np.zeros((2 * n - 1, n))
        for k in range(2 * n - 1):
            lo = max(0, k - n + 1)
            hi = min(n - 1, k)
            A[k, lo:hi + 1] = a[k - hi:k - lo + 1][::-1]

        c_obj = -np.ones(n)
        b_ub = np.full(2 * n - 1, P)

        try:
            res = linprog(c_obj, A_ub=A, b_ub=b_ub,
                          bounds=[(0.0, None)] * n,
                          method='highs',
                          options={'time_limit': 4.0})
            if res.success:
                b = res.x
                b_sum = np.sum(b)
                if b_sum > 1e-8:
                    b = b / b_sum
                    # Try mixing
                    improved = False
                    for t_mix in [0.3, 0.5, 0.7, 0.9]:
                        mixed = (1.0 - t_mix) * a + t_mix * b

                            best_score = score
                            best_a = mixed.copy()
                            a = mixed.copy()
                            improved = True
                    if not improved:
                        break
        except Exception:
            break

    return best_a, best_score


def alt_projection_init(n, rng, n_iter=50):
    """Alternating projection for near-flat-spectrum nonnegative initialization."""
    N = 1 << (2 * n - 2).bit_length()
    phases = rng.uniform(0.0, 2.0 * np.pi, N // 2 + 1)
    F = np.exp(1j * phases)
    F[0] = 1.0 + 0j

    a = np.ones(n) / n
    for _ in range(n_iter):
        a_full = np.fft.irfft(F, N)
        a = np.maximum(a_full[:n], 0.0)
        s = a.sum()
        if s < 1e-10:
            a = np.ones(n) / n
        else:
            a = a / s
        F_full = np.fft.rfft(a, N)
        F = np.exp(1j * np.angle(F_full))
        F[0] = 1.0 + 0j

    return a


def cosine_init(n, rng):
    """Cosine-windowed initialization."""
    x = np.linspace(0.0, 1.0, n)
    a = 0.5 * (1.0 + np.cos(2.0 * np.pi * x)) + 0.05
    a = a + 0.05 * rng.random(n)
    a = np.maximum(a, 0.0)
    return a / a.sum()


def random_gaussian_init(n, rng):
    """Gaussian-shaped random initialization."""
    center = n // 2
    x = np.arange(n)
    sigma = n / 4.0
    a = np.exp(-0.5 * ((x - center) / sigma) ** 2)
    a = a + 0.1 * rng.random(n)
    a = np.maximum(a, 0.0)
    return a / a.sum()


def upsample_init(a_coarse, n_target):

        if time.time() > phase1_deadline - 10:
            break
        config_deadline = min(phase1_deadline - 2, time.time() + time_per_config)
        if config_deadline < time.time() + 10:
            continue

        # Generate initialization
        try:
            if init_type == 'alt_proj':
                a_init = alt_projection_init(n, rng, n_iter=40)
            elif init_type == 'parent':
                parent = np.array(height_sequence_1, dtype=np.float64)
                parent = np.clip(parent, 1e-10, 1000.0)
                if len(parent) >= n:
                    a_init = parent[:n].copy()
                else:
                    a_init = np.zeros(n)
                    a_init[:len(parent)] = parent
                a_init = np.maximum(a_init, 1e-10)
                a_init = a_init / a_init.sum()
            elif init_type == 'cosine':
                a_init = cosine_init(n, rng)
            elif init_type == 'gaussian':
                a_init = random_gaussian_init(n, rng)
            else:
                a_init = rng.random(n)
                a_init = a_init / a_init.sum()
        except Exception as e:
            print(f"  Init error ({n}, {init_type}): {e}")
            continue

        # Tau schedule: from smooth to sharp
        # Scale with 1/n since conv values are O(1/n)
        base_tau = 1.0 / n
        tau_levels = [base_tau * 0.1, base_tau * 0.04, base_tau * 0.015,
                      base_tau * 0.006, base_tau * 0.002, base_tau * 0.0008]

        try:
            a_opt, score = optimize_smooth_projected(a_init, n, config_deadline, tau_levels,
                                                     iters_per_level=3000)
        except Exception as e:
            print(f"  Optimize error ({n}, {init_type}): {e}")
            continue

        print(f"  n={n}, init={init_type}, score={score:.6f}, "
              f"remaining={overall_deadline - time.time():.0f}s")

        if score < best_score:
            best_score = score
            best_seq = a_opt.copy()
            print(f"  >>> New best: {best_score:.6f} (n={n})")

    # Coarse-to-fine: if we have a good short sequence, use it to init longer ones
    if best_seq is not None and len(best_seq) <= 256 and time.time() < phase1_deadline - 30:
        for n_target in [512, 1024]:
            if time.time() > phase1_deadline - 15:
                break
            a_init = upsample_init(best_seq, n_target)
            a_init = a_init + 0.02 * rng.random(n_target)
            a_init = np.maximum(a_init, 0)
            a_init = a_init / a_init.sum()

            base_tau = 1.0 / n_target
            tau_levels = [base_tau * 0.05, base_tau * 0.02, base_tau * 0.008, base_tau * 0.003]
            config_deadline = min(phase1_deadline - 2, time.time() + 60)

            try:
                a_opt, score = optimize_smooth_projected(a_init, n_target, config_deadline,
                                                         tau_levels, iters_per_level=2000)
                if score < best_score:
                    best_score = score
                    best_seq = a_opt.copy()
                    print(f"  >>> Coarse-to-fine new best: {best_score:.6f} (n={n_target})")
            except Exception:
                pass

    # === Phase 2: Subgradient refinement (30% of time) ===
    print(f"\n=== Phase 2: Subgradient refinement ===")
    if best_seq is not None:
        phase2_deadline = min(overall_deadline - 60, time.time() + (overall_deadline - time.time()) * 0.35)

        # Try subgradient from best and from a perturbed version
        for pert_level in [0.0, 0.01, 0.03]:
            if time.time() > phase2_deadline:
                break
            a_perturbed = best_seq + pert_level * rng.standard_normal(len(best_seq))
            a_perturbed = project_simplex(np.maximum(a_perturbed, 0) / max(np.sum(np.maximum(a_perturbed, 0)), 1e-10))

            sub_deadline = phase2_deadline - (phase2_deadline - time.time()) * 0.25
            a_ref, score = subgradient_refine(a_perturbed, sub_deadline)
            print(f"  Subgrad (pert={pert_level}): score={score:.6f}")

            if score < best_score:
                best_score = score
                best_seq = a_ref.copy()
                print(f"  >>> New best: {best_score:.6f}")

    # === Phase 3: LP refinement (15% of time) ===
    print(f"\n=== Phase 3: LP refinement ===")
    if best_seq is not None and len(best_seq) <= 1024:
        lp_deadline = min(overall_deadline - 10, time.time() + (overall_deadline - time.time()) * 0.4)
        a_lp, score = lp_refine(best_seq, lp_deadline, max_attempts=20)
        print(f"  LP result: score={score:.6f}")
        if score < best_score:
            best_score = score
            best_seq = a_lp.copy()
            print(f"  >>> New best: {best_score:.6f}")

    # === Final polish: quick subgradient on best ===
    if best_seq is not None and time.time() < overall_deadline - 15:
        a_final, score = subgradient_refine(best_seq, overall_deadline - 2, max_iters=50000)
        if score < best_score:
            best_score = score
            best_seq = a_final
            print(f"  Final polish: {best_score:.6f}")

    # Fallback
    if best_seq is None:
        print("Fallback to uniform sequence")
        best_seq = np.ones(256) / 256

    # Ensure valid output
    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    final_score = evaluate_sequence(result)
    print(f"\nFINAL: score={final_score:.6f}, length={len(result)}")
    return result
