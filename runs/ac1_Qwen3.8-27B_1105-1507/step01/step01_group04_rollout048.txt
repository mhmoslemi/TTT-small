"""
Optimizes non-negative step function heights to minimize 2n*max(conv(f,f))/(sum(f))^2.
Uses accelerated mirror descent with soft active sets, LP-based Frank-Wolfe refinement,
and diverse initializations including flat-spectrum construction.

import numpy as np
import time
import operator

def _fft_convolve_full(f):
    """Compute full linear convolution of f with itself using FFT."""
    n = len(f)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    F = np.fft.rfft(f, size)
    h = np.fft.irfft(F * F, size)
    return h[:2 * n - 1]

def _score_raw(f):
    """Fast score computation assuming f >= 0, sum(f) > 0.01."""
    n = len(f)
    h = _fft_convolve_full(f)
    M = float(np.max(h))
    S = float(np.sum(f))
    if S < 0.01:
        return float('inf')
    return 2.0 * n * M / (S * S)

def _normalize(f):
    """Normalize f to sum to 1, clip negatives. Returns None if degenerate."""
    f = np.maximum(f, 0.0)
    s = float(np.sum(f))
    if s < 1e-30:
        return None
    return f / s

def _pressure_vector_soft(f, h, temperature):
    """
    Compute soft pressure vector using softmax weighting of lags.
    p[j] = sum_k w[k] * 2 * f[k-j] / |K| where w[k] = softmax(h[k] / temperature)
    """
    n = len(f)
    # Softmax weights
    h_shifted = h - np.max(h)
    w = np.exp(h_shifted / temperature)
    w_sum = np.sum(w)
    if w_sum < 1e-30:
        w = np.zeros_like(h)
        w[np.argmax(h)] = 1.0
    else:
        w /= w_sum
    
    p = np.zeros(n, dtype=np.float64)
    for k in range(len(h)):
        if w[k] < 1e-10:
            continue
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        if j_end >= j_start:
            lo = k - j_end
            hi = k - j_start
            p[j_start:j_end + 1] += w[k] * 2.0 * f[lo:hi + 1][::-1]
    return p

def _pressure_vector_hard(f, active_lags):
    """
    Compute p[j] = (1/|K|) * sum_{k in K} 2*f[k-j] for valid 0<=k-j<n.
    """
    n = len(f)
    p = np.zeros(n, dtype=np.float64)
    for k in active_lags:
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        if j_end >= j_start:
            lo = k - j_end
            hi = k - j_start
            p[j_start:j_end + 1] += 2.0 * f[lo:hi + 1][::-1]
    p /= float(len(active_lags))
    return p

def _cosine_init(n):
    j = np.arange(n, dtype=np.float64)
    f = np.cos(np.pi * (j + 0.5) / float(n)) ** 2
    f = np.maximum(f, 0.0)
    s = np.sum(f)
    if s > 0:
        f /= s
    return f

def _gaussian_init(n, sigma_frac):
    j = np.arange(n, dtype=np.float64) - (n - 1) / 2.0
    sigma = sigma_frac * float(n)
    f = np.exp(-j * j / (2.0 * sigma * sigma))
    s = np.sum(f)
    if s > 0:
        f /= s
    return f

def _tapered_init(n, power):
    j = np.arange(n, dtype=np.float64)
    x = np.sin(np.pi * (j + 0.5) / float(n))
    f = x ** power
    f = np.maximum(f, 0.0)
    s = np.sum(f)
    if s > 0:
        f /= s
    return f

def _flat_spectrum_init(n, rng):
    """
    Generate a sequence with approximately flat circular autocorrelation
    by using random phases in the frequency domain.
    """
    N = 2 * n  # Pad to at least 2n for circular conv to approximate linear
    # Actually, let's use a size that's at least 2n-1
    N = 1
    while N < 2 * n - 1:
        N *= 2
    
    # Create random phases with conjugate symmetry
    phases = rng.random(N // 2 + 1) * 2 * np.pi
    # Ensure DC and Nyquist are real (phase 0 or pi)
    phases[0] = 0.0
    if N % 2 == 0:
        phases[N // 2] = 0.0
    
    # Build spectrum with unit magnitude
    F = np.zeros(N // 2 + 1, dtype=np.complex128)
    F = np.exp(1j * phases)
    
    # Inverse FFT to get real signal
    f_full = np.fft.irfft(F, N)
    
    # Take first n elements, make non-negative
    f = np.abs(f_full[:n])
    s = np.sum(f)
    if s > 0:
        f /= s
    return f

def _mirror_descent_accelerated(f_init, n_iter, eta0, eta_decay, threshold_frac, label=""):
    """
    Accelerated mirror descent with multiplicative updates.
    Uses momentum on the pressure vector and adaptive threshold.
    """
    f = np.asarray(f_init, dtype=np.float64).copy()
    f = _normalize(f)
    if f is None:
        return f_init.copy(), float('inf')
    n = len(f)
    best_f = f.copy()
    best_score = _score_raw(f)
    
    # Momentum
    p_momentum = np.zeros(n, dtype=np.float64)
    momentum_beta = 0.7
    
    print(f"    MD-acc start: n={n}, score={best_score:.6f}")

    stall_count = 0
    stall_threshold = 200

    for it in range(n_iter):
        h = _fft_convolve_full(f)
        M = float(np.max(h))
        if M <= 0:
            break

        # Adaptive threshold: start tight, relax over time
        progress = (it + 1) / float(n_iter)
        threshold = (threshold_frac + 0.3 * progress) * M + 1e-15
        active = np.where(h >= threshold)[0]
        if len(active) == 0:
            active = np.array([int(np.argmax(h))], dtype=np.int64)

        p = _pressure_vector_hard(f, active)
        
        # Momentum update
        p_momentum = momentum_beta * p_momentum + (1 - momentum_beta) * p
        p_use = p_momentum

        # Adaptive step size
        eta = eta0 / (1.0 + eta_decay * it)
        # Also scale by 1/M to be scale-invariant
        eta = eta / (M + 1e-30)
        
        log_ratio = -eta * p_use
        log_ratio = np.clip(log_ratio, -30.0, 30.0)

        f_new = f * np.exp(log_ratio)
        f_norm = _normalize(f_new)
        if f_norm is None:
            f = np.ones(n) / n
        else:
            f = f_norm

        score = _score_raw(f)
        if score < best_score - 1e-8:
            best_score = score
            best_f = f.copy()
            stall_count = 0
        else:
            stall_count += 1
        
        # If stalled for too long, reduce momentum and increase exploration
        if stall_count > stall_threshold:
            momentum_beta = max(0.3, momentum_beta - 0.05)
            p_momentum = np.zeros(n, dtype=np.float64)
            stall_count = 0

        if (it + 1) % 500 == 0:
            print(f"    MD-acc it={it + 1}: best={best_score:.6f}, M={M:.8f}, |K|={len(active)}, eta={eta:.6f}")

def _lp_frank_wolfe(f_init, max_iters, time_budget):
    """LP-based Frank-Wolfe for the normalized problem."""
    from scipy.optimize import linprog
    from scipy.sparse import coo_matrix

    f = np.asarray(f_init, dtype=np.float64).copy()
    f = _normalize(f)
    if f is None:
        return f_init.copy(), float('inf')
    n = len(f)
    best_f = f.copy()
    best_score = _score_raw(f)
    start_time = time.monotonic()

    print(f"    LP-FW start: n={n}, score={best_score:.6f}")

    for it in range(max_iters):
        if time.monotonic() - start_time > time_budget:
            break

        h = _fft_convolve_full(f)
        M = float(np.max(h))

        # Build sparse constraint matrix: C[k, j] = f[k-j] if 0<=k-j<n else 0
        # Shape: (2n-1, n)
        row_idx = []
        col_idx = []
        vals = []
        for j in range(n):
            rows_j = np.arange(j, j + n)
            row_idx.extend(rows_j.tolist())
            col_idx.extend([j] * n)
            vals.extend(f.tolist())

        C = coo_matrix((vals, (row_idx, col_idx)), shape=(2 * n - 1, n))

        c_obj = -np.ones(n)
        b_ub = np.full(2 * n - 1, M)
        res = linprog(c_obj, A_ub=C, b_ub=b_ub,
                      bounds=[(0, None)] * n, method='highs')

        if not res.success or res.x is None:
            print(f"    LP failed at iter {it}: {res.message}")
            break

        g = res.x
        Sg = float(np.sum(g))
        if Sg < 1e-10:
            break
        g = g / Sg

        # Frank-Wolfe step size
        t = 2.0 / (it + 2)
        f_new = (1.0 - t) * f + t * g
        f_n = _normalize(f_new)
        if f_n is None:
            break
        f = f_n

        score = _score_raw(f)
        if score < best_score:
            best_score = score
            best_f = f.copy()

        if (it + 1) % 5 == 0:
            print(f"    LP-FW it={it + 1}: score={best_score:.6f}")

    return best_f, best_score

"""

def _fft_convolve_full(f):
    """Compute full linear convolution of f with itself using FFT."""
    n = len(f)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    F = np.fft.rfft(f, size)
    h = np.fft.irfft(F * F, size)
    return h[:2 * n - 1]


def _score_raw(f):
    """Fast score computation assuming f >= 0, sum(f) > 0.01."""
    n = len(f)
    h = _fft_convolve_full(f)
    M = float(np.max(h))
    S = float(np.sum(f))
    if S < 0.01:
        return float('inf')
    return 2.0 * n * M / (S * S)


def _normalize(f):
    """Normalize f to sum to 1, clip negatives. Returns None if degenerate."""
    f = np.maximum(f, 0.0)
    s = float(np.sum(f))
    if s < 1e-30:

def _pressure_vector(f, active_lags):
    """
    Compute p[j] = (1/|K|) * sum_{k in K} 2*f[k-j] for valid 0<=k-j<n.
    """
    n = len(f)
    p = np.zeros(n, dtype=np.float64)
    for k in active_lags:
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        if j_end >= j_start:
            lo = k - j_end
            hi = k - j_start
            p[j_start:j_end + 1] += 2.0 * f[lo:hi + 1][::-1]
    p /= float(len(active_lags))
    return p

    best_score = _score_raw(f)
    start_time = time.monotonic()

    print(f"    LP-FW start: n={n}, score={best_score:.6f}")

    for it in range(max_iters):
        if time.monotonic() - start_time > time_budget:
            break

        h = _fft_convolve_full(f)
        M = float(np.max(h))

        row_idx = []
        col_idx = []
        vals = []
        for j in range(n):
            rows_j = np.arange(j, j + n)
            row_idx.extend(rows_j.tolist())
            col_idx.extend([j] * n)
            vals.extend(f.tolist())

        C = coo_matrix((vals, (row_idx, col_idx)), shape=(2 * n - 1, n))

        c_obj = -np.ones(n)
        b_ub = np.full(2 * n - 1, M)
        res = linprog(c_obj, A_ub=C, b_ub=b_ub,
                      bounds=[(0, None)] * n, method='highs')

        if not res.success or res.x is None:
            print(f"    LP failed at iter {it}: {res.message}")
            break

        g = res.x
        Sg = float(np.sum(g))
        if Sg < 1e-10:
            break
        g = g / Sg

        t = 2.0 / (it + 2)
        f_new = (1.0 - t) * f + t * g
        f_n = _normalize(f_new)
        if f_n is None:
            break
        f = f_n

        score = _score_raw(f)
        if score < best_score:
            best_score = score
            best_f = f.copy()

        if (it + 1) % 5 == 0:
            print(f"    LP-FW it={it + 1}: score={best_score:.6f}")

    return best_f, best_score


def _multiplicative_perturb(f, rng, frac):
    """Apply multiplicative random perturbation and re-normalize."""
    n = len(f)
    noise = np.exp(rng.standard_normal(n) * frac * 0.5)
    f_p = f * noise
    return _normalize(f_p)


def _additive_perturb(f, rng, frac):
    """Add a small random perturbation and re-normalize."""
    n = len(f)
    noise = rng.random(n, dtype=np.float64) * frac
    f_p = f + noise
    return _normalize(f_p)


def propose_candidate(seed=42, budget_s=1500, **kwargs):
    """
    Search for non-negative step function heights minimizing the evaluation score.
    Strategy: diverse initializations (GS flat-spectrum, cosine, Gaussian, tapered),
    accelerated mirror descent with momentum, LP Frank-Wolfe refinement,
    and perturbation restarts.
    """
    deadline = time.monotonic() + float(budget_s) - 10.0
    rng = np.random.default_rng(seed)

    # Get parent sequence
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64).copy()
        parent = np.clip(parent, 0.0, 1000.0)
    except Exception:
        parent = np.ones(512, dtype=np.float64)

    best_f = parent.copy()
    best_score = float(evaluate_sequence(best_f.tolist()))
    print(f"Parent: n={len(best_f)}, score={best_score:.6f}")

    # Build diverse candidate initializations
    candidates = []

    p = _normalize(parent)
    if p is not None:
        candidates.append(("parent", p))

    # Flat spectrum via Gerchberg-Saxton (key new initialization)
    for n in [512, 1024]:
        for trial in range(4):
            f_gs = _flat_spectrum_gs_init(n, 200, rng)
            f_gs = _normalize(f_gs)
            if f_gs is not None:
                candidates.append(("gs_%d_%d" % (n, trial), f_gs))

    # Cosine initializations
    for n in [512, 1024, 2048]:
        candidates.append(("cos_%d" % n, _cosine_init(n)))

    # Gaussian initializations
    for n in [512, 1024]:
        for sf in [0.2, 0.3, 0.35, 0.45]:
            candidates.append(("gauss_%d_%d" % (n, int(sf * 100)), _gaussian_init(n, sf)))

    # Tapered initializations
    for n in [512, 1024]:
        for pw in [0.5, 1.0, 2.0, 3.0, 5.0]:
            candidates.append(("tap_%d_%d" % (n, int(pw * 10)), _tapered_init(n, pw)))

    # Random
    for n in [512, 1024]:
        f_r = rng.random(n, dtype=np.float64)
        f_rn = _normalize(f_r)
        if f_rn is not None:
            candidates.append(("rand_%d" % n, f_rn))

    print(f"Total candidates: {len(candidates)}")

    # Phase 1: Mirror descent on each candidate
    seqs = []
    scores = []
    names = []

    remaining = deadline - time.monotonic()
    phase1_budget = remaining * 0.50
    time_per = max(3.0, min(25.0, phase1_budget / max(1, len(candidates))))

    for idx, (name, f0) in enumerate(candidates):
        if time.monotonic() > deadline - 90:
            break

        t0 = time.monotonic()
        n_it = int(max(400, min(4000, time_per * 250)))

        f_md, s_md = _mirror_descent(f0, n_it, 1.0, 0.03, 0.8, 0.6, label=name)
        elapsed = time.monotonic() - t0
        print(f"  [{idx + 1}/{len(candidates)}] {name}: score={s_md:.6f} ({elapsed:.1f}s)")

        seqs.append(f_md)
        scores.append(s_md)
        names.append(name)

        if s_md < best_score:
            best_score = s_md
            best_f = f_md.copy()
            print(f"  ** New best: {best_score:.6f} from {name} **")

    # Phase 2: Refine top candidates
    order = sorted(enumerate(scores), key=lambda x: x[1])
    top_k = min(6, len(order))
    phase2_deadline = deadline - 40

    for r in range(top_k):
        if time.monotonic() > phase2_deadline:
            break
        i_cur = order[r][0]
        s_cur = order[r][1]
        name = names[i_cur]
        f_cur = seqs[i_cur].copy()
        n = len(f_cur)
        print(f"\n  Refining {name} (n={n}, score={s_cur:.6f})")

        remaining2 = phase2_deadline - time.monotonic()
        if remaining2 < 15:
            break

        # 2a: More MD with smaller step, tighter threshold
        n_it2 = int(min(4000, remaining2 * 25))
        if n_it2 > 200:
            f_ref, s_ref = _mirror_descent(f_cur, n_it2, 0.4, 0.01, 0.9, 0.7, label=name + "_ref")
            if s_ref < best_score:
                best_score = s_ref
                best_f = f_ref.copy()
                print(f"  ** New best after refine: {best_score:.6f} **")
            if s_ref < s_cur:
                f_cur = f_ref
                s_cur = s_ref

        # 2b: LP refinement for small n
        n_eff = len(f_cur)
        remaining2 = phase2_deadline - time.monotonic()
        if n_eff <= 800 and remaining2 > 35:
            lp_budget = min(40.0, remaining2 - 5)
            if lp_budget > 10:
                f_lp, s_lp = _lp_frank_wolfe(f_cur, max_iters=20, time_budget=lp_budget)
                if s_lp < best_score:
                    best_score = s_lp
                    best_f = f_lp.copy()
                    print(f"  ** New best from LP: {best_score:.6f} **")
                if s_lp < s_cur:
                    f_cur = f_lp
                    s_cur = s_lp

        # 2c: More MD after LP
        remaining2 = phase2_deadline - time.monotonic()
        if remaining2 > 20:
            n_it3 = int(min(3000, remaining2 * 20))
            if n_it3 > 200:
                f_md2, s_md2 = _mirror_descent(f_cur, n_it3, 0.3, 0.008, 0.92, 0.75, label=name + "_md2")
                if s_md2 < best_score:
                    best_score = s_md2
                    best_f = f_md2.copy()
                    print(f"  ** New best after MD2: {best_score:.6f} **")
                if s_md2 < s_cur:
                    f_cur = f_md2
                    s_cur = s_md2

        # 2d: Perturbation restarts
        remaining2 = phase2_deadline - time.monotonic()
        if remaining2 > 25:
            for pert_frac in [0.03, 0.07, 0.15]:
                if time.monotonic() > phase2_deadline:
                    break
                f_pert = _multiplicative_perturb(f_cur, rng, pert_frac)
                if f_pert is None:
                    continue
                n_it4 = int(min(2000, (phase2_deadline - time.monotonic() - 5) * 12))
                if n_it4 < 100:
                    break
                f_p, s_p = _mirror_descent(f_pert, n_it4, 0.6, 0.02, 0.8, 0.5, label=name + "_pert")
                if s_p < best_score:
                    best_score = s_p
                    best_f = f_p.copy()
                    print(f"  ** New best from perturb({pert_frac}): {best_score:.6f} **")
                if s_p < s_cur:
                    f_cur = f_p
                    s_cur = s_p

            # Additive perturbation
            if time.monotonic() < phase2_deadline - 10:
                f_pert2 = _additive_perturb(f_cur, rng, 0.05)
                if f_pert2 is not None:
                    n_it5 = int(min(1500, (phase2_deadline - time.monotonic() - 3) * 15))
                    if n_it5 > 100:
                        f_p2, s_p2 = _mirror_descent(f_pert2, n_it5, 0.5, 0.02, 0.85, 0.6, label=name + "_apert")
                        if s_p2 < best_score:
                            best_score = s_p2
                            best_f = f_p2.copy()
                            print(f"  ** New best from additive perturb: {best_score:.6f} **")

    # Phase 3: Final polish on the best
    if (deadline - time.monotonic()) > 20:
        n_final = len(best_f)
        print(f"\n  Final polish: n={n_final}, score={best_score:.6f}")

        # Very small step MD
        n_it6 = int(min(4000, (deadline - time.monotonic() - 5) * 15))
        if n_it6 > 200:
            f_pol, s_pol = _mirror_descent(best_f, n_it6, 0.15, 0.005, 0.95, 0.8, label="polish_fine")
            if s_pol < best_score:
                best_score = s_pol
                best_f = f_pol.copy()
                print(f"  ** New best (polish fine): {best_score:.6f} **")

        # Try different thresholds
        if (deadline - time.monotonic()) > 10:
            n_it7 = int(min(2000, (deadline - time.monotonic() - 3) * 15))
            if n_it7 > 100:
                f_pol2, s_pol2 = _mirror_descent(best_f, n_it7, 0.1, 0.003, 0.6, 0.85, label="polish_wide")
                if s_pol2 < best_score:
                    best_score = s_pol2
                    best_f = f_pol2.copy()
                    print(f"  ** New best (polish wide): {best_score:.6f} **")

    # Final verification
    final_list = [float(x) for x in best_f]
    final_score = float(evaluate_sequence(final_list))
    print(f"\nFinal verified score: {final_score:.6f} (n={len(final_list)})")

    if not np.isfinite(final_score) or final_score > 2.0:
        print("  Falling back to parent.")
        return [float(x) for x in parent]

    return final_list
