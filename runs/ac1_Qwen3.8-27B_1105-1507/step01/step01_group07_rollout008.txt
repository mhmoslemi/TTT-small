    n = len(a)
    conv = np.convolve(a, a)

"""
Multi-strategy optimizer for minimizing 2*n*max(conv(a,a))/(sum(a))^2.
Combines subgradient descent, LP-based alternating minimization,
and random perturbation search across multiple sequence lengths.
"""

import numpy as np
from scipy.optimize import linprog
import time


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    conv = np.convolve(a, a)
    max_b = float(np.max(conv))
    sum_a = float(np.sum(a))
    if sum_a < 0.01:
        return np.inf
    return 2.0 * n * max_b / (sum_a * sum_a)


def compute_subgradient(a, tol=1e-10):
    """Compute a valid subgradient of the objective at point a."""
    n = len(a)
    conv = np.convolve(a, a)
    max_b = float(np.max(conv))
    sum_a = float(np.sum(a))
    if sum_a < 0.01:
        return np.zeros(n), np.inf

    # Find active set (convolution coefficients at or near the max)
    active = np.where(conv >= max_b - tol * max_b)[0]

    # Subgradient of max w.r.t. a[i]:
    # d(conv[k])/d(a[i]) = 2*a[k-i] if 0 <= k-i < n, else 0
    # Average over active set for a valid subgradient of the max
    grad_M = np.zeros(n)
    for k in active:
        # k-i ranges from k-(n-1) to k, valid when 0 <= k-i < n
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        # k-i for i in [lo, hi] gives values from k-lo down to k-hi
        # a[k-i] for i in range: a[k-lo], a[k-lo-1], ..., a[k-hi]
        # So a[k-i] for i=lo..hi is a[k-lo : k-hi-1 : -1]
        idx = k - np.arange(lo, hi + 1)
        grad_M[lo:hi + 1] += 2.0 * a[idx]

    grad_M /= len(active)

    # Full subgradient of f = 2n*M/S^2
    # df/da[i] = (2n/S^2)*grad_M[i] - (4nM/S^3)
    grad = (2.0 * n / (sum_a ** 2)) * grad_M - (4.0 * n * max_b / (sum_a ** 3))
    return grad, 2.0 * n * max_b / (sum_a ** 2)


def subgradient_descent(a, max_iter=500, lr_init=0.5):
    """Subgradient descent with backtracking line search."""

        a = np.ones(n) / n
        s = 1.0

    best_obj = compute_obj(best_a)

        grad, obj = compute_subgradient(a)
        if obj < best_obj - 1e-12:
            best_obj = obj
            best_a = a.copy()

        # Decaying learning rate
        lr = lr_init / (n * (1 + 0.01 * it))

        # Gradient step with projection
        a_new = a - lr * grad
        a_new = np.maximum(a_new, 0.0)
        s_new = np.sum(a_new)

            parent_obj = compute_obj(parent)
            if parent_obj < best_score:
                best_score = parent_obj
                best_seq = parent.copy()
            print(f"Parent (n={parent.size}): obj={parent_obj:.6f}")
    except Exception:
        pass

    # --- Quick uniform baseline ---
    for n in [50, 100, 150, 200, 300, 500, 1000]:
        a = np.ones(n)
        obj = compute_obj(a)
        if obj < best_score:
            best_score = obj
            best_seq = a.copy()
    print(f"After uniform baselines: best={best_score:.6f}")

    # --- Main optimization over multiple n values ---
    n_values = [20, 30, 40, 50, 60, 80, 100, 120, 150, 200, 250, 300]

    # Sort by expected time cost (shorter first for more restarts)
    for n in n_values:
        if time.monotonic() > deadline:
            break
        remaining = deadline - time.monotonic()
        if remaining < 15:
            break

        # Number of restarts based on remaining time
        est_per_restart = max(5, n * 0.05)
        n_restarts = max(1, min(8, int(remaining / est_per_restart)))

        for restart in range(n_restarts):
            if time.monotonic() > deadline:
                break

            t0 = time.monotonic()
            a_init = structured_init(n, rng)

            # Phase 1: Subgradient descent
            a_gd, obj_gd = subgradient_descent(a_init, max_iter=min(400, int(remaining / 0.1)))
            if obj_gd < best_score:
                best_score = obj_gd
                best_seq = a_gd.copy()
                print(f"SGD n={n} r={restart}: obj={obj_gd:.6f}")

            # Phase 2: LP alternating (if time permits)
            if time.monotonic() < deadline - 40 and n <= 200:
                a_lp, obj_lp = lp_alternating(a_gd, max_iter=80, t=0.5)
                if obj_lp < best_score:
                    best_score = obj_lp
                    best_seq = a_lp.copy()
                    print(f"LP n={n} r={restart}: obj={obj_lp:.6f}")
                a_cur = a_lp
                obj_cur = obj_lp
            else:
                a_cur = a_gd
                obj_cur = obj_gd

            # Phase 3: Random local search refinement
            if time.monotonic() < deadline - 10:
                iters = int(min(4000, max(500, (deadline - time.monotonic()) / 0.01)))
                a_rs, obj_rs = random_local_search(a_cur, rng, max_iter=iters, sigma=0.08)
                if obj_rs < best_score:
                    best_score = obj_rs
                    best_seq = a_rs.copy()
                    print(f"RLS n={n} r={restart}: obj={obj_rs:.6f}")

            elapsed_restart = time.monotonic() - t0
            if restart == 0:
                print(f"  n={n}: first restart took {elapsed_restart:.1f}s, best so far={best_score:.6f}")

    # --- Final intensive refinement on best solution ---
    if best_seq is not None and time.monotonic() < deadline - 10:
        remaining = deadline - time.monotonic()
        print(f"\nFinal refinement with {remaining:.0f}s remaining...")
        n_best = len(best_seq)

        # More subgradient iterations with smaller steps
        a_fin, obj_fin = subgradient_descent(best_seq, max_iter=int(remaining * 5), lr_init=0.2)
        if obj_fin < best_score:
            best_score = obj_fin
            best_seq = a_fin.copy()
            print(f"Final SGD: obj={best_score:.6f}")

        # Final random search
        if time.monotonic() < deadline - 5:
            iters = int((deadline - time.monotonic()) * 50)
            a_r, obj_r = random_local_search(best_seq, rng, max_iter=iters, sigma=0.03)
            if obj_r < best_score:
                best_score = obj_r
                best_seq = a_r.copy()
                print(f"Final RLS: obj={best_score:.6f}")

    # --- Fallback ---
    if best_seq is None:
        best_seq = np.ones(100)
        best_score = compute_obj(best_seq)

    # Verify with evaluate_sequence
    try:
        final_score = evaluate_sequence([float(x) for x in best_seq])
        if final_score < best_score:
            best_score = final_score
        print(f"Verified score: {final_score:.6f}")
    except Exception:
        pass

    elapsed = time.monotonic() - start_time
    print(f"\nDONE: n={len(best_seq)}, score={best_score:.6f}, time={elapsed:.1f}s")
    return [float(x) for x in best_seq]
