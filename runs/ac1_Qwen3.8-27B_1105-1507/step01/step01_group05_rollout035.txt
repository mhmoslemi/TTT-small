"""
Combined optimization for minimizing 2*n*max(conv(a,a))/sum(a)^2.
Uses gradient descent (Adam) with softmax max approximation, LP polishing,
block coordinate descent, and multiple diverse starting points/lengths.
"""

import numpy as np
from scipy.optimize import linprog
import time


def compute_score(a):
    """Compute score = 2*n*max(conv(a,a)) assuming sum(a)=1."""
    n = len(a)
    conv = np.convolve(a, a)
    return 2.0 * n * conv.max()


def build_conv_matrix(f):
    """Build matrix A such that (A @ b)[k] = sum_j f[j] * b[k-j] (full conv)."""
    n = len(f)
    m = 2 * n - 1
    A = np.zeros((m, n))
    for k in range(m):
        j_start = max(0, k - n + 1)
        j_end = min(n, k + 1)
        js = np.arange(j_start, j_end)
        A[k, js] = f[k - js]
    return A


def gradient_softmax_max(a, epsilon):
    """Gradient of softmax approximation of max(conv(a,a)) w.r.t. a."""
    n = len(a)
    conv = np.convolve(a, a)
    mval = conv.max()
    w = np.exp((conv - mval) / epsilon)
    w /= w.sum()
    # grad[i] = 2 * sum_j w[i+j] * a[j]
    grad = 2.0 * np.convolve(a[::-1], w, mode='full')[n - 1:2 * n - 1]
    return grad


def lp_direction(f, M):
    """Solve LP: max sum(b) s.t. conv(f,b) <= M, b >= 0. Returns b/sum(b)."""
    n = len(f)
    if n > 4096:
        return None
    try:
        A = build_conv_matrix(f)
        c_obj = -np.ones(n)
        b_ub = np.full(2 * n - 1, M)
        bounds = [(0, None)] * n
        res = linprog(c_obj, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': 10})
        if res.success and res.x is not None:
            b = np.clip(res.x, 0, None)
            if b.sum() > 1e-15:
                return b / b.sum()
    except Exception:
        pass
    return None

def block_cd_polish(f0, time_budget, rng, block_size=8, n_passes=50):
    """Polish using block coordinate descent."""
    n = len(f0)
    f = f0.astype(np.float64).copy() / f0.sum()
    best_f = f.copy()
    best_score = compute_score(f)

    start = time.monotonic()

    for _ in range(n_passes):
        if time.monotonic() - start > time_budget:
            break
        improved = False
        for _ in range(max(1, n // block_size)):
            if time.monotonic() - start > time_budget:
                break
            p = rng.integers(0, n - block_size + 1)
            block = f[p:p + block_size].copy()

            def block_obj(x):
                f_tmp = f.copy()
                f_tmp[p:p + block_size] = np.clip(x, 0, None)
                s = f_tmp.sum()
                if s < 1e-10:
                    return 1e10
                f_tmp = f_tmp / s
                return compute_score(f_tmp)

            try:
                from scipy.optimize import minimize as sci_min
                res = sci_min(block_obj, block, method='Nelder-Mead',
                              options={'maxiter': 200, 'xatol': 1e-10, 'fatol': 1e-12})
                new_block = np.clip(res.x, 0, None)
                f[p:p + block_size] = new_block
                f = f / f.sum()
                score = compute_score(f)
                if score < best_score - 1e-12:
                    best_f = f.copy()
                    best_score = score
                    improved = True
            except Exception:
                pass

        if not improved:
            break

    return best_f, best_score

    # Give more time to promising starts (smaller n for speed, parent first)
    time_per_start = (budget_s - 80.0) / n_starts
    time_per_start = max(20.0, time_per_start)

    print(f"Total starting points: {n_starts}")
    print(f"Time per start: {time_per_start:.0f}s")
    print(f"Target: 1.5027, Previous best: 1.515592")
    print(f"Total budget: {budget_s}s")

    # Prioritize: parent first, then smaller n
    start_times = []
    for i, (name, f0, n) in enumerate(starts):
        if 'parent' in name:
            t_alloc = time_per_start * 1.5
        elif n <= 512:
            t_alloc = time_per_start * 1.2
        elif n <= 1024:
            t_alloc = time_per_start * 1.0
        else:
            t_alloc = time_per_start * 0.8
        start_times.append(t_alloc)

    # Normalize total time
    total_alloc = sum(start_times)
    scale = min(1.0, (budget_s - 80.0) / total_alloc)
    start_times = [t * scale for t in start_times]

    for i, (name, f0, n) in enumerate(starts):
        if time.monotonic() > deadline:
            break

        t_here = min(start_times[i], deadline - time.monotonic() - 5)
        if t_here < 10:
            break

        print(f"\n[{name}] n={n}, time={t_here:.0f}s", flush=True)

        # Phase 1: Gradient optimization
        t_grad = t_here * 0.6
        f_opt, score = optimize_gradient(f0, t_grad, rng)

        # Phase 2: LP polish
        t_lp_remaining = t_here - (time.monotonic() - (deadline - t_here)) if False else None
        elapsed_so_far = 0  # not tracking exactly, just use remaining time
        t_remaining = t_here - (elapsed_so_far)
        # Just use a fraction of remaining time for LP
        t_lp = min(30, t_here * 0.2)
        if time.monotonic() < deadline - 5:
            M = compute_score(f_opt) / (2.0 * n)
            for _ in range(5):
                if time.monotonic() > deadline - 5:
                    break
                g = lp_direction(f_opt, M * 1.001)
                if g is not None:
                    for t_mix in [0.05, 0.1, 0.15, 0.2, 0.3]:
                        f_try = (1 - t_mix) * f_opt + t_mix * g
                        f_try = np.clip(f_try, 0, None)
                        if f_try.sum() > 1e-10:
                            f_try = f_try / f_try.sum()
                            s = compute_score(f_try)
                            if s < score:
                                score = s
                                f_opt = f_try
                                M = compute_score(f_opt) / (2.0 * n)
                                break

        # Phase 3: Block CD polish (only for smaller n)
        if n <= 1024 and time.monotonic() < deadline - 10:
            t_bc = min(30, t_here * 0.15)
            f_cd, score_cd = block_cd_polish(f_opt, t_bc, rng, block_size=6)
            if score_cd < score:
                f_opt = f_cd
                score = score_cd
                print(f"  Block CD improved to: {score:.6f}", flush=True)

        # Phase 4: Another gradient pass
        if time.monotonic() < deadline - 10 and t_here > 60:
            t_grad2 = min(30, t_here * 0.2)
            f_opt2, score2 = optimize_gradient(f_opt, t_grad2, rng)
            if score2 < score:
                f_opt = f_opt2
                score = score2

        if np.isfinite(score) and score < best_score_overall:
            best_score_overall = score
            best_overall = f_opt.copy()
            print(f"  ** New best: {score:.6f} (n={n}) **", flush=True)
        else:
            print(f"  Best for this start: {score:.6f}", flush=True)

    # Final intensive polishing on the best solution
    if best_overall is not None and time.monotonic() < deadline - 30:
        n_best = len(best_overall)
        t_left = deadline - time.monotonic() - 10
        print(f"\n=== Final polishing: n={n_best}, time={t_left:.0f}s ===", flush=True)

        # Multiple LP rounds
        for round_num in range(20):
            if time.monotonic() > deadline - 10:
                break
            M = compute_score(best_overall) / (2.0 * n_best)
            g = lp_direction(best_overall, M * 1.0005)
            if g is not None:
                best_improved = False
                for t_mix in [0.02, 0.05, 0.1, 0.15]:
                    f_try = (1 - t_mix) * best_overall + t_mix * g
                    f_try = np.clip(f_try, 0, None)
                    if f_try.sum() > 1e-10:
                        f_try = f_try / f_try.sum()
                        s = compute_score(f_try)
                        if s < best_score_overall - 1e-12:
                            best_score_overall = s
                            best_overall = f_try
                            best_improved = True
                            print(f"  LP round {round_num}: {s:.6f}", flush=True)
                            break
                if not best_improved and round_num > 5:
                    break

        # Final gradient pass
        if time.monotonic() < deadline - 15:
            t_final = min(60, deadline - time.monotonic() - 10)
            f_final, s_final = optimize_gradient(best_overall, t_final, rng)
            if s_final < best_score_overall:
                best_score_overall = s_final
                best_overall = f_final
                print(f"  Final gradient: {s_final:.6f}", flush=True)

        # Block CD on best
        if n_best <= 2048 and time.monotonic() < deadline - 10:
            t_bc = min(30, deadline - time.monotonic() - 5)
            f_bc, s_bc = block_cd_polish(best_overall, t_bc, rng, block_size=8)
            if s_bc < best_score_overall:
                best_score_overall = s_bc
                best_overall = f_bc
                print(f"  Final block CD: {s_bc:.6f}", flush=True)

    # Verify and return
    if best_overall is not None:
        final_list = np.clip(best_overall, 0, 1000).tolist()
        final_score = evaluate_sequence(final_list)
        print(f"\n=== FINAL: score={final_score:.6f}, length={len(final_list)} ===", flush=True)
        return final_list

    # Fallback
    print("Fallback to uniform sequence")
    return [1.0] * 256
