def subgradient_step(f, eta):
    """One subgradient step to reduce max autocorrelation."""
    n = len(f)

        return f

    conv = np.convolve(f_norm, f_norm)

    f_new = np.maximum(f_new, 0)
    # Normalize

    if S_new > 0:
        f_new = f_new / S_new
    return f_new

def solve_lp_direction(f, M):
    """Solve LP: max sum(g) s.t. conv(f,g) <= M, g >= 0."""
    n = len(f)

    from scipy.sparse import csr_matrix
    rows = []
    cols = []
    vals = []

        j_min = max(0, k - (n - 1))
        j_max = min(k, n - 1)
        for j in range(j_min, j_max + 1):
            rows.append(k)
            cols.append(j)
            vals.append(f[k - j])
    A = csr_matrix((vals, (rows, cols)), shape=(2*n-1, n))
    c = -np.ones(n)
    b_ub = np.full(2*n-1, M)
    bounds = [(0, None)] * n
    result = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
    if result.success:
        return result.x
    return None

    best_score = 2 * n * np.max(np.convolve(f, f)) / (S**2)

            break
        conv_f = np.convolve(f, f)
        M = np.max(conv_f)

        if g is None or np.sum(g) <= 0:

        g = g * (S / np.sum(g))
        # Blend

        new_f = np.maximum(new_f, 0)
        new_S = np.sum(new_f)
        if new_S > 0:

        # Evaluate

        new_score = 2 * n * np.max(new_conv) / (S**2)

            best_f = new_f.copy()
            f = new_f

        else:
            no_improve += 1
            step_size *= 0.8
            if step_size < 0.01 or no_improve > 30:
