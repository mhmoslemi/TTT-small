```python
    a = a.astype(float).copy()
    a = np.clip(a, 0, 1000)
    if a.sum() < 1e-10:
        return a, float('inf')
    a = a / a.sum()
    
    best = a.copy()
    best_score = local_score(best)
    rng = np.random.default_rng(seed)
    
    for alpha in alpha_schedule:
        eta = 0.3
        for it in range(max_iter):
            b = np.convolve(a, a)
            w = alpha * np.exp(alpha * b - alpha * b.max())  # for numerical stability
            # grad[j] = 2 * sum_m w[j+m] * a[m]
            corr = np.correlate(w, a, 'full')
            grad = 2.0 * corr[n-1:2*n-1]
            
            # Normalize gradient for relative update
            grad_mean = grad.mean()
            grad_std = grad.std() + 1e-15
            
            # Multiplicative update
            log_fac = -eta * (grad - grad_mean) / grad_std
            log_fac = np.clip(log_fac, -3, 3)
            a_new = a * np.exp(log_fac)
            a_new = np.clip(a_new, 0, 1000)
            if a_new.sum() < 1e-30:
                break
            a_new = a_new / a_new.sum()
            
            sc = local_score(a_new)
            if sc < best_score:
                best = a_new.copy()
                best_score = sc
            if sc < local_score(a):
                a = a_new
            else:
                eta *= 0.8
                if eta < 0.01:
                    break
        # Occasionally perturb
        if rng.random() < 0.3:
            a = best * np.exp(rng.normal(0, 0.1, n))
            a = np.clip(a, 0, None)
            a = a / a.sum()
    
    return best, best_score
```
