    """Smooth approximation of max using log-sum-exp."""
    c_max = c.max()
    return c_max + np.log(np.sum(np.exp(beta * (c - c_max)))) / beta

def smooth_max_grad(c, a, beta):
    """Gradient of smooth max of convolution w.r.t. a."""
    n = len(a)
    c_max = c.max()
    log_weights = beta * (c - c_max)
    log_weights -= log_weights.max()  # numerical stability
    weights = np.exp(log_weights)

    return grad

    while size < 3*n:

    W = np.fft.rfft(w, size)
    A = np.fft.rfft(a, size)
    corr = np.fft.irfft(W * np.conj(A), size)

    # Actually, the cross-correlation via FFT:
    # IFFT(FFT(w) * conj(FFT(a))) gives the correlation
    # corr[0] = sum_m w[m] * a[m]
    # corr[k] = sum_m w[m+k] * a[m] (for k >= 0, with appropriate wrapping)

    return grad

Minimizes J(a) = 2*n*max(convolve(a,a))/(sum(a))^2 using:
1. Softmax-smoothed (log-sum-exp) objective with FFT-accelerated L-BFGS-B
2. Subgradient mirror descent on the simplex for fine-tuning
3. LP best-response for structured large moves (reduced constraint set)
4. Adaptive temperature annealing for the smooth approximation
5. Diverse initializations including resampled parent and structured starts

Key innovation: log-sum-exp smoothing provides gradient access to the max
function, enabling efficient second-order optimization that the previous
unsmoothed approach could not fully exploit.
"""

import numpy as np
from scipy.optimize import linprog, minimize
import time


def fast_conv(a):
    """Full self-convolution via FFT, O(n log n)."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]


def softmax_val_and_grad(a, beta):
    """
    Compute smoothed objective (1/beta)*log(sum exp(beta*c_k)) and its gradient.
    a is assumed to be non-negative. Returns (value, gradient).
    """
    n = len(a)
    c = fast_conv(a)
    c_max = c.max()
    log_terms = beta * (c - c_max)
    log_terms -= log_terms.max()
    exp_terms = np.exp(log_terms)
    val = c_max + np.log(np.sum(exp_terms)) / beta
    p = exp_terms / np.sum(exp_terms)
    # grad[j] = sum_k p[k] * 2 * a[k-j] (if k-j in [0,n-1])
    # = 2 * cross_correlation(p, a)[j]
    size = 1
    while size < 3 * n:
        size <<= 1
    P = np.fft.rfft(p, size)
    A = np.fft.rfft(a, size)
    corr = np.fft.irfft(P * np.conj(A), size)
    grad = 2.0 * corr[:n]
    return val, grad


def subgradient_step(a, lr):
    """
    One subgradient mirror descent step on the simplex.
    Picks the argmax lag and moves in the negative gradient direction.
    """
    n = len(a)
    c = fast_conv(a)
    k_max = int(np.argmax(c))
    # Subgradient: g[j] = 2 * a[k_max - j] if 0 <= k_max - j < n
    g = np.zeros(n)
    j_min = max(0, k_max - n + 1)
    j_max = min(n - 1, k_max)
    if j_min <= j_max:
        g[j_min:j_max + 1] = 2.0 * a[k_max - j_max:k_max - j_min + 1][::-1]
    # Mirror descent on simplex: proj_simplex(a - lr * g)
    v = a - lr * g
    # Project onto simplex {x: sum(x)=1, x>=0}
    mu = np.sort(v)[::-1]
    cssv = np.cumsum(mu)
    rho = np.nonzero(np.arange(1, n + 1) * (mu - cssv / np.arange(1, n + 1)) > 0)[0]
    if len(rho) > 0:
        theta = (cssv[rho[-1]] - 1.0) / (rho[-1] + 1.0)
    else:
        theta = v.min()
    a_new = np.maximum(v - theta, 0.0)
    return a_new, k_max

    Subgradient step using top n_peaks lags for a smoother direction.
    """
    n = len(a)
    c = fast_conv(a)
    # Get top peaks
    top_indices = np.argsort(c)[::-1][:n_peaks]
    # Weight by how much they exceed the mean
    c_mean = c.mean()
    weights = np.maximum(c[top_indices] - c_mean, 0)
    if weights.sum() < 1e-15:
        weights = np.ones(n_peaks)
    weights /= weights.sum()

    g = np.zeros(n)
    for w, k in zip(weights, top_indices):
        j_min = max(0, k - n + 1)

def resample(parent, new_n):
    """Resample a sequence to a new length using linear interpolation."""
    old_n = len(parent)
    if old_n == new_n:
        return parent.copy()
    x_old = np.linspace(0, 1, old_n)
    x_new = np.linspace(0, 1, new_n)
    return np.maximum(np.interp(x_new, x_old, parent), 0.0)

Minimizes J(a)=2n*max(conv(a,a))/(sum a)^2 via softmax-smoothed L-BFGS-B
