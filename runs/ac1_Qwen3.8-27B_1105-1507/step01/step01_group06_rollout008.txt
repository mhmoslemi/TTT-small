Minimizes J(a) = 2*n*max(convolve(a,a))/(sum(a))^2 using:

import numpy as np

def fast_conv(a):
    """Full self-convolution via FFT."""
    n = len(a)
    s = 1
    while s < 2 * n - 1:
        s <<= 1
    fa = np.fft.rfft(a, s)
    return np.fft.irfft(fa * fa, s)[:2 * n - 1]

def build_A(a, active_set):
    """Build constraint matrix for active constraints only."""
    n = len(a)
    A = np.zeros((len(active_set), n))
    for idx, k in enumerate(active_set):
        jl = max(0, k - n + 1)
        jh = min(n - 1, k)
        if jl <= jh:
            A[idx, jl:jh + 1] = a[k - jh:k - jl + 1][::-1]
    return A

def lp_best_response(a, M, active_set=None, max_constraints=500):
    """LP: max sum(g) s.t. (a*g)_k <= M, g >= 0. Uses active set for speed."""
    n = len(a)
    if active_set is None:
        b = fast_conv(a)
        # Use only the top constraints
        threshold = np.quantile(b, 0.8)  # top 20%
        active_set = np.where(b >= threshold)[0]
        if len(active_set) > max_constraints:
            top_idx = np.argsort(b[active_set])[-max_constraints:]
            active_set = active_set[top_idx]
    
    c = -np.ones(n)
    A = build_A(a, active_set)
    b_ub = np.full(len(active_set), M)
    r = linprog(c, A_ub=A, b_ub=b_ub,
               bounds=[(0, None)] * n, method='highs')
    if r.success:
        g = r.x
        if g.sum() > 1e-10:
            return g
    return None

def objective_smoothed(x, n, tau=50.0):
    """Smoothed objective for gradient methods."""
    a = np.clip(x, 0.0, 1000.0)
    s = a.sum()
    if s < 0.01:
        return 1e10
    a = a / s
    c = fast_conv(a)
    c_max = c.max()
    # Softmax smoothing
    c_shifted = tau * (c - c_max)
    log_sum_exp = c_max + np.log(np.exp(c_shifted).sum()) / tau
    return 2.0 * n * log_sum_exp / 1.0  # Since sum(a)=1, denominator is 1

"""
Minimizes J(a) = 2*n*max(convolve(a,a))/(sum(a))^2 using:
1. LP-based mirror descent with adaptive active sets and geometric mixing
2. Exact-gradient L-BFGS-B refinement (analytical gradient via FFT/dot products)
3. Randomized restarts with structured perturbations
4. Multi-length exploration with interpolation between lengths
"""

import numpy as np
from scipy.optimize import linprog, minimize
import time


def fast_conv(a):
    """Full self-convolution via FFT, O(n log n)."""
    n = len(a)
    s = 1
    while s < 2 * n - 1:
        s <<= 1
    fa = np.fft.rfft(a, s)
    return np.fft.irfft(fa * fa, s)[:2 * n - 1]


def build_A(a, active_set):
    """Build Toeplitz constraint matrix A where A[k,j] = a[k-j] for row k."""
    n = len(a)
    m = len(active_set)
    A = np.zeros((m, n))
    for idx in range(m):
        k = active_set[idx]
        jl = max(0, k - n + 1)
        jh = min(n - 1, k)
        if jl <= jh:
            A[idx, jl:jh + 1] = a[k - jh:k - jl + 1][::-1]
    return A


def lp_best_response(a, M, active_set):
    """LP: max sum(g) s.t. (a*g)_k <= M for k in active_set, g >= 0."""
    n = len(a)
    c = -np.ones(n)
    A = build_A(a, active_set)
    b_ub = np.full(len(active_set), M)
    r = linprog(c, A_ub=A, b_ub=b_ub,
                bounds=[(0, None)] * n, method='highs',
                options={'presolve': True})
    if r.success:
        g = r.x
        if g.sum() > 1e-10:
            # Verify the solution
            check = A @ g
            if np.all(check <= M * (1 + 1e-6)):
                return g
    return None


def compute_gradient_smoothed(a, tau):
    """Compute gradient of 2n * smooth_max(c) where c = a*a, using softmax smoothing."""
    n = len(a)
    c = fast_conv(a)
    c_max = c.max()
    # Softmax weights
    shifted = tau * (c - c_max)
    shifted -= shifted.max()  # numerical stability
    weights = np.exp(shifted)
    weights /= weights.sum()
    
    # grad[i] = 4n * sum_j a[j] * w[j+i] for i = 0..n-1
    # = 4n * correlate(w, a, 'full')[:n]
    grad = np.array([np.dot(a, weights[i:i+n]) for i in range(n)])
    grad *= 4.0 * n
    
    # Smooth max value
    smooth_max = c_max + np.log(weights.sum() * np.exp(tau * (c_max - c_max))) / tau
    # Actually: smooth_max = (1/tau) * log(sum(exp(tau*c)))
    # = (1/tau) * (tau*c_max + log(sum(exp(tau*(c-c_max)))))
    # = c_max + (1/tau) * log(sum(exp(shifted)))
    # But we already normalized shifted, so:
    raw_shifted = tau * (c - c_max)
    log_sum = np.log(np.exp(raw_shifted).sum())
    smooth_max = c_max + log_sum / tau
    
    return smooth_max, grad


def objective_and_grad(x, n, tau=50.0):
    """Objective and gradient for smoothed problem (sum a = 1 implicitly)."""
    a = np.clip(x, 0.0, 1000.0)
    s = a.sum()
    if s < 1e-8:
        return 1e10, np.zeros(n)
    a_n = a / s
    c = fast_conv(a_n)
    c_max = c.max()
    raw_shifted = tau * (c - c_max)
    log_sum = np.log(np.exp(raw_shifted).sum())
    smooth_max = c_max + log_sum / tau
    f_val = 2.0 * n * smooth_max
    
    # Gradient
    shifted = raw_shifted - raw_shifted.max()
    weights = np.exp(shifted)
    weights /= weights.sum()
    grad = np.array([np.dot(a_n, weights[i:i+n]) for i in range(n)])
    grad *= 4.0 * n
    
    return f_val, grad


def lp_refine(a, max_iters, deadline, n_constraints_frac=0.3, max_constraints=500):
    """LP mirror descent refinement. Returns refined a."""
    n = len(a)
    a = np.clip(a, 0.01, None).astype(np.float64)
    a /= a.sum()
    
    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        b = fast_conv(a)
        M = float(b.max())
        
        # Active set: top fraction of constraints
        n_c = min(max_constraints, max(50, int(n_constraints_frac * (2 * n - 1))))
        top_idx = np.argsort(b)[-n_c:]
        
        g = lp_best_response(a, M, top_idx)
        if g is None:
            break
        g /= g.sum()
        
        # Geometric mixing schedule
        t_mix = 0.05 + 0.4 * (0.9 ** it)
        t_mix = min(t_mix, 0.5)
        
        a_new = (1.0 - t_mix) * a + t_mix * g
        a_new = np.clip(a_new, 0.0, None)
        s = a_new.sum()
        if s < 1e-10:
            break
        a_new /= s
        a = a_new
    
    return a


def gradient_refine(a0, n, deadline, tau=50.0):
    """L-BFGS-B with analytical gradient for smoothed objective."""
    a0 = np.clip(a0, 0.0, None)
    if a0.sum() < 0.01:
        return a0
    a0 = a0 / a0.sum()
    
    def _obj(x):
        ax = np.clip(x, 0.0, 1000.0)
        s = ax.sum()
        if s < 1e-8:
            return 1e10
        ax = ax / s
        c = fast_conv(ax)
        c_max = c.max()
        raw_shifted = tau * (c - c_max)
        log_sum = np.log(np.exp(raw_shifted).sum())
        smooth_max = c_max + log_sum / tau
        return 2.0 * n * smooth_max
    
    res = minimize(_obj, a0, method='L-BFGS-B',
                   bounds=[(0, None)] * n,
                   options={'maxiter': 1500, 'max_fun': 30000,
                            'ftol': 1e-15, 'gtol': 1e-14, 'disp': False})
    if res.fun < 1e10:
        return res.x
    return a0
