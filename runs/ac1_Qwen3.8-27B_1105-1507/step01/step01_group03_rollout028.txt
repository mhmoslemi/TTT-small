def line_search_1d(f, g, n, n_grid=30):
    """Minimize h(t)=score((1-t)f+tg) over t in [0,1] via grid + refinement."""
    cf = fft_conv(f, f)
    c_fg = fft_conv(f, g)
    cg = fft_conv(g, g)

    def h_of_t(t):
        a = (1.0 - t)
        c = a * a * cf + 2.0 * a * t * c_fg + t * t * cg
        return float(np.max(c))

    # Coarse grid
    ts = np.linspace(0.0, 1.0, n_grid)
    vals = np.array([h_of_t(t) for t in ts])
    idx = int(np.argmin(vals))

    # Local refinement around best
    lo = max(0.0, ts[max(0, idx - 1)])
    hi = min(1.0, ts[min(n_grid - 1, idx + 1)])
    # Golden section
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    best_t = ts[idx]
    best_v = vals[idx]
    for _ in range(40):
        if hi - lo < 1e-10:
            break
        t1 = hi - gr * (hi - lo)
        t2 = lo + gr * (hi - lo)
        v1 = h_of_t(t1)
        v2 = h_of_t(t2)
        if v1 < v2:
            hi = t2
        else:
            lo = t1
        mid_best = min(v1, v2)
        mid_t = t1 if v1 < v2 else t2
        if mid_best < best_v:
            best_v = mid_best
            best_t = mid_t

    return best_t, 2.0 * n * best_v


def solve_lp_primal(f, M, n, edge_offsets, row, col, m):
    """Solve primal LP: max sum(g) s.t. A@ g <= M*1, g >= 0."""
    A = build_A(f, edge_offsets, row, col, m, n)
    c_obj = -np.ones(n)
    b_ub = np.full(m, M)
    bounds = [(0.0, None)] * n
    res = linprog(c_obj, A_ub=A, b_ub=b_ub, bounds=bounds, method="highs")
    if res.status != 0:
        return None
    return res.x
