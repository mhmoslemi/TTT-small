"""
Hybrid coordinate descent + LP optimization for minimizing 2n*max(conv(a,a))/sum(a)^2.
Uses vectorized 1D coordinate search, KKT-inspired initializations, and LP refinement.

import numpy as np
import time
from scipy.optimize import linprog
from scipy.sparse import csr_matrix


def compute_conv_zero(a, j):
    """Compute conv(a,a) with a[j] set to 0."""
    a_zero = a.copy()
    old_val = a_zero[j]
    a_zero[j] = 0.0
    return a_zero, np.convolve(a_zero, a_zero), old_val


def coord_descent_1d(a, deadline, max_passes=200, verbose=False):
    """Coordinate descent with 1D grid search for each coordinate."""
    n = len(a)
    best_a = a.copy()
    best_score = evaluate_sequence(a.tolist())
    a = a.copy()
    
    for p in range(max_passes):
        if time.monotonic() > deadline:
            break
        
        old_score = best_score
        improved_count = 0
        
        for j in range(n):
            if time.monotonic() > deadline:
                break
            
            # Current value
            v_old = a[j]
            S = float(np.sum(a)) - v_old  # sum of all others
            
            if S < 1e-10:
                continue
            
            # Compute conv with a[j] = 0
            # Contribution of a[j] to conv[k]:
            # For k where max(0, k-n+1) <= j <= min(k, n-1):
            #   if k == 2*j: contribution = v_old^2
            #   else: contribution = 2 * v_old * a[k-j]
            
            conv_full = np.convolve(a, a)
            
            # Remove contribution of a[j]
            conv_zero = conv_full.copy()
            k_min = max(0, j - (n - 1))  # smallest k such that k-n+1 <= j
            # Actually: k-n+1 <= j => k <= j+n-1
            # And j <= k => k >= j
            # So k ranges from max(0, j) to min(2n-2, j+n-1)
            # Wait: k-n+1 <= j means k <= j + n - 1
            # j <= k means k >= j
            # Also 0 <= k < 2n-1
            k_lo = max(0, j)  # j <= k, k >= 0
            k_hi = min(2*n-2, j + n - 1)  # k <= j+n-1, k <= 2n-2
            
            for k in range(k_lo, k_hi + 1):
                if k == 2 * j:
                    conv_zero[k] -= v_old * v_old
                else:
                    conv_zero[k] -= 2.0 * v_old * a[k - j]
            
            # Now search for optimal v in [0, v_upper]
            # Upper bound: current best score gives a constraint
            # 2n * max_conv(v) / (S+v)^2 < best_score
            # If max_conv(v) <= C (some bound), then 2n*C/(S+v)^2 < best_score
            # => (S+v)^2 > 2n*C/best_score
            # This is always satisfied for large v, so no useful upper bound from this.
            # Use a practical bound: v_upper = max(3*v_old, 2*max(a), 1)
            v_upper = max(3.0 * v_old, 2.0 * float(np.max(a)), 1.0)
            
            # Grid search
            n_grid = 50
            v_grid = np.linspace(0.0, v_upper, n_grid)
            
            best_v = v_old
            best_v_score = best_score
            
            # Vectorized evaluation
            # For each v in v_grid, compute max_conv
            # conv_new[k] = conv_zero[k] + 2*v*a[k-j] (for valid k != 2j)
            #             = conv_zero[k] + v^2 (for k = 2j)
            
            # Precompute the linear part
            # For valid k (k_lo to k_hi), the coefficient of v is 2*a[k-j]
            # For k = 2j, the term is v^2 instead of 2*v*a[j] (which would be 2*v*a[j] = 2*v*a[2j-j] = 2*v*a[j])
            # Wait, at k=2j, k-j = j, so 2*a[k-j] = 2*a[j] = 2*v_old. But the actual term is v^2, not 2*v*v_old.
            # So I need to handle k=2j separately.
            
            valid_ks = np.arange(k_lo, k_hi + 1)
            # Linear coefficients for valid k
            lin_coefs = 2.0 * a[valid_ks - j]  # shape: (len(valid_ks),)
            # For k = 2j, replace the linear term with quadratic
            is_quad = (valid_ks == 2 * j)
            
            # base_max for k NOT in valid range
            if k_lo > 0 or k_hi < 2*n-2:
                mask_invalid = np.ones(2*n-1, dtype=bool)
                mask_invalid[k_lo:k_hi+1] = False
                base_max = float(np.max(conv_zero[mask_invalid])) if np.any(mask_invalid) else 0.0
            else:
                base_max = 0.0
            
            # For each v, max_conv = max(base_max, max over valid k of (conv_zero[k] + term(k,v)))
            # where term = 2*v*a[k-j] for k != 2j, and v^2 for k = 2j
            
            # Vectorize over v_grid:
            # For linear terms (k != 2j):
            #   values[k_idx, v_idx] = conv_zero[valid_ks[k_idx]] + v_grid[v_idx] * lin_coefs[k_idx]
            lin_ks = valid_ks[~is_quad]
            lin_coefs_ks = lin_coefs[~is_quad]
            conv_zero_lin = conv_zero[lin_ks]
            
            # Shape: (n_lin, n_grid)
            lin_values = conv_zero_lin[:, None] + v_grid[None, :] * lin_coefs_ks[:, None]
            lin_max = np.max(lin_values, axis=0)  # shape: (n_grid,)
            
            # For quadratic term (k = 2j):
            quad_values = np.full(n_grid, base_max)
            if np.any(is_quad):
                k2j = int(valid_ks[is_quad][0])
                quad_base = conv_zero[k2j]
                quad_values = np.maximum(quad_values, quad_base + v_grid**2)
            
            # Combine
            max_conv = np.maximum(np.maximum(lin_max, quad_values), base_max)
            
            # Objective: 2n * max_conv / (S + v)^2
            obj = 2.0 * n * max_conv / (S + v_grid)**2
            obj = np.where(S + v_grid > 0, obj, np.inf)
            
            idx_best = int(np.argmin(obj))
            best_v = float(v_grid[idx_best])
            best_v_score = float(obj[idx_best])
            
            # Refine around best_v
            if 0 < idx_best < n_grid - 1:
                v_lo = float(v_grid[idx_best - 1])
                v_hi = float(v_grid[idx_best + 1])
            else:
                delta = v_upper / n_grid
                v_lo = max(0.0, best_v - delta)
                v_hi = best_v + delta
            
            v_refine = np.linspace(v_lo, v_hi, 20)
            # Re-evaluate on refined grid
            lin_values_r = conv_zero_lin[:, None] + v_refine[None, :] * lin_coefs_ks[:, None]
            lin_max_r = np.max(lin_values_r, axis=0)
            quad_values_r = np.full(len(v_refine), base_max)
            if np.any(is_quad):
                quad_values_r = np.maximum(quad_values_r, quad_base + v_refine**2)
            max_conv_r = np.maximum(np.maximum(lin_max_r, quad_values_r), base_max)
            obj_r = 2.0 * n * max_conv_r / (S + v_refine)**2
            obj_r = np.where(S + v_refine > 0, obj_r, np.inf)
            
            idx_r = int(np.argmin(obj_r))
            if obj_r[idx_r] < best_v_score:
                best_v = float(v_refine[idx_r])
                best_v_score = float(obj_r[idx_r])
            
            a[j] = best_v
            if best_v_score < old_score - 1e-12:
                improved_count += 1
        
        # Check convergence
        score = evaluate_sequence(a.tolist())
        if score < best_score:
            best_score = score
            best_a = a.copy()
            if verbose and p % 10 == 0:
                print(f"  CD pass {p}: score={best_score:.6f}")
        else:
            # Revert
            a = best_a.copy()
            if improved_count == 0:
                break
    
    return best_a, best_score


def build_conv_matrix_sparse(f):
    """Build sparse matrix A where (A@g)[k] = convolve(f,g)[k]."""
    n = len(f)
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        j_min = max(0, k - (n - 1))
        j_max = min(k, n - 1)
        for j in range(j_min, j_max + 1):
            rows.append(k)
            cols.append(j)
            vals.append(f[k - j])
    return csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))


def lp_direction(f, M, tighten=1.0):
    """Find g >= 0 maximizing sum(g) s.t. conv(f,g) <= M*tighten."""
    n = len(f)
    A = build_conv_matrix_sparse(f)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M * tighten)
    bounds = [(0, None)] * n
    try:
        result = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
        if result.success:
            return result.x
    except Exception as e:
        pass
    return None


def lp_refine(f, deadline, max_iters=50, verbose=False):
    """LP-based refinement with tightening."""
    f = f.copy().astype(np.float64)
    best_f = f.copy()
    best_score = evaluate_sequence(f.tolist())
    
    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        
        conv_f = np.convolve(f, f)
        M = float(np.max(conv_f))
        
        # Try to tighten
        tighten = max(0.99, 1.0 - 0.01 * (it + 1) / max_iters)
        g = lp_direction(f, M, tighten)
        
        if g is None or np.sum(g) < 1e-15:
            break
        
        # Rescale g to same sum as f
        sum_f = float(np.sum(f))
        sum_g = float(np.sum(g))
        g = g * (sum_f / sum_g)
        
        # Try different step sizes
        best_step_f = f
        best_step_score = best_score
        for t in [0.3, 0.5, 0.7, 1.0]:
            new_f = (1.0 - t) * f + t * g
            new_f = np.clip(new_f, 0.0, 1000.0)
            new_score = evaluate_sequence(new_f.tolist())
            if np.isfinite(new_score) and new_score < best_step_score:
                best_step_score = new_score
                best_step_f = new_f.copy()
        
        if best_step_score < best_score:
            f = best_step_f
            best_f = f.copy()
            best_score = best_step_score
            if verbose and it % 5 == 0:
                print(f"  LP iter {it}: score={best_score:.6f}")
        else:
            break
    
    return best_f, best_score


def gradient_polish(f, deadline, lr=0.01, max_iters=2000):
    """Subgradient polishing."""
    f = f.copy().astype(np.float64)
    best_f = f.copy()
    best_score = evaluate_sequence(f.tolist())
    n = len(f)
    
    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        
        conv_f = np.convolve(f, f)
        max_val = float(np.max(conv_f))
        
        # Find all indices at (or near) max
        threshold = max_val * 0.999
        max_indices = np.where(conv_f >= threshold)[0]
        
        # Weighted subgradient
        grad = np.zeros(n)
        weights = conv_f[max_indices]
        weights = weights / weights.sum()
        
        for w, k in zip(weights, max_indices):
            # d(conv[k])/d(a[j]) = 2*a[k-j] if 0 <= k-j < n, else 0
            j_min = max(0, k - (n - 1))
            j_max = min(k, n - 1)
            for j in range(j_min, j_max + 1):
                grad[j] += w * 2.0 * f[k - j]
        
        # Update
        new_f = f - lr * grad
        new_f = np.clip(new_f, 0.0, 1000.0)
        
        # Preserve sum
        S = float(np.sum(f))
        new_S = float(np.sum(new_f))
        if new_S > 0:
            new_f = new_f * (S / new_S)
        
        new_score = evaluate_sequence(new_f.tolist())
        if np.isfinite(new_score) and new_score < best_score:
            f = new_f
            best_f = f.copy()
            best_score = new_score
            if (it + 1) % 500 == 0:
                print(f"  Grad iter {it+1}: score={best_score:.6f}, lr={lr:.6f}")
        else:
            lr *= 0.85
            if lr < 1e-8:
                break
    
    return best_f, best_score


def make_initializations(n, rng):
    """Generate diverse initializations of length n."""
    inits = []
    idx = np.arange(n, dtype=np.float64)
    
    # Uniform
    inits.append(('uniform', np.ones(n)))
    
    # Ramp (increasing)
    inits.append(('ramp', idx + 1.0))
    
    # Reverse ramp
    inits.append(('rev_ramp', n - idx))
    
    # Quadratic ramp
    inits.append(('quad_ramp', (idx + 1.0)**2))
    
    # Cosine window
    inits.append(('cosine', 0.5 * (1 + np.cos(2*np.pi*idx/n))))
    
    # KKT-inspired: plateau + ramp
    # Plateau on first half, ramp on second
    n_half = n // 2
    plateau_ramp = np.ones(n)
    plateau_ramp[n_half:] = np.linspace(1.0, 2.0, n - n_half)
    inits.append(('plat_ramp', plateau_ramp))
    
    # Reverse: ramp on first half, plateau on second
    ramp_plat = np.ones(n)
    ramp_plat[:n_half] = np.linspace(2.0, 1.0, n_half)
    inits.append(('ramp_plat', ramp_plat))
    
    # Exponential
    inits.append(('exp', np.exp(0.5 * idx / n)))
    inits.append(('rev_exp', np.exp(0.5 * (n - 1 - idx) / n)))
    
    # Random
    inits.append(('random', rng.uniform(0.5, 1.5, size=n)))
    
    # Bimodal
    center = (n - 1) / 2.0
    inits.append(('bimodal', np.exp(-0.5*((idx - n*0.25)/(n*0.1))**2) + np.exp(-0.5*((idx - n*0.75)/(n*0.1))**2)))
    
    return inits
