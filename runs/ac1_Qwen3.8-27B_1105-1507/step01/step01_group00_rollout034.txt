The objective is: min_p 2n * max_k (p^T A_k p) subject to sum(p) = 1, p >= 0.

For k = 0: (A_0)_{i,j} = 1 if i+j = 0, i.e., i=j=0. So A_0 = e_0 * e_0^T.

The constraints are: p^T A_k p <= t for all k, where t is the variable we're minimizing (times 2n).

This is a system of quadratic inequality constraints. The feasible set {p: p^T A_k p <= t, sum p = 1, p >= 0} is non-convex in (p, t) for k where A_k is indefinite.

However, for k = 0, A_0 = e_0 e_0^T is PSD, so the constraint p_0^2 <= t is convex in (p, t). Similarly for k = 2n-2.

For other k, the constraint is non-convex.

This explains why the problem is hard. The non-convexity comes from the cross terms in the autocorrelation.

Frank-Wolfe based optimization with strong step size for minimizing
2n*max(conv(a,a))/sum(a)^2. Uses FFT for fast convolution, LP oracle,
and projected gradient polish.
"""
import numpy as np
import time
from scipy.optimize import linprog
from scipy.sparse import csr_matrix

def fast_conv(a, b):
    n = len(a) + len(b) - 1
    fsize = 1
    while fsize < n:
        fsize <<= 1
    fa = np.fft.rfft(a, fsize)
    fb = np.fft.rfft(b, fsize)
    return np.fft.irfft(fa * fb, fsize)[:n]

def compute_score(a):
    n = len(a)
    s = np.sum(a)
    if s < 0.01:
        return np.inf
    b = fast_conv(a, a)
    max_b = np.max(b)
    return 2.0 * n * max_b / (s * s)

def build_conv_matrix_sparse(f):
    n = len(f)
    rows = np.empty((2*n-1) * n, dtype=np.int32)  # over-allocate
    cols = np.empty((2*n-1) * n, dtype=np.int32)
    vals = np.empty((2*n-1) * n, dtype=np.float64)
    idx = 0
    for k in range(2 * n - 1):
        j_min = max(0, k - (n - 1))
        j_max = min(k, n - 1)
        for j in range(j_min, j_max + 1):
            rows[idx] = k
            cols[idx] = j
            vals[idx] = f[k - j]
            idx += 1
    return csr_matrix((vals[:idx], (rows[:idx], cols[:idx])), shape=(2 * n - 1, n))

    n = len(f)
    A = build_conv_matrix_sparse(f)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    try:
        result = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
        if result.success:
            return result.x
    except Exception:
        pass
    return None

def project_simplex(v, s=1.0):
    """Project v onto {x: sum(x) = s, x >= 0}."""
    n = len(v)
    v = v.copy()
    if np.sum(v) <= 0:
        return np.full(n, s / n)
    # Scale to sum = s first... no, we need the actual projection
    # Standard algorithm:
    # Project onto {x: sum(x) = s, x >= 0}
    # = s * (project v/s onto {x: sum(x) = 1, x >= 0})
    mu = np.zeros(n)
    v_sorted = np.sort(v)[::-1]
    cssv = np.cumsum(v_sorted)
    rho = np.where(v_sorted > (cssv - s) / np.arange(1, n + 1))[0][-1]
    theta = (cssv[rho] - s) / (rho + 1)
    proj = np.maximum(v - theta, 0)
    # Ensure sum is exactly s
    diff = s - np.sum(proj)
    if diff > 1e-12:
        # Add to the largest elements
        idx = np.argsort(v)[::-1]
        proj[idx[0]] += diff
    return proj
