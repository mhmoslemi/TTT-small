import time
import numpy as np
from scipy.optimize import linprog
from scipy.sparse import csr_matrix


def normalize_sequence(arr):
    x = np.asarray(arr, dtype=np.float64).ravel()
    if x.size == 0:
        return None
    x = np.where(np.isfinite(x), x, 0.0)
    x = np.maximum(x, 0.0)
    s = float(x.sum())
    if s <= 1e-300:
        return None
    x /= s
    return x


def score_normalized(x, n):
    conv = np.convolve(x, x)
    m = float(np.max(conv))
    if not np.isfinite(m):
        return float("inf")
    return float(2.0 * n * m)


def make_conv_matrix(n):
    """Build sparse matrix A where A @ g = convolve(f_fixed, g)."""
    m = 2 * n - 1
    rows = []
    cols = []
    data = []
    for k in range(m):
        i_start = max(0, k - n + 1)
        i_end = min(n - 1, k)
        for i in range(i_start, i_end + 1):
            rows.append(k)
            cols.append(i)
            data.append(0.0)  # will be filled with f[k-i]
    
    A = csr_matrix((data, (rows, cols)), shape=(m, n))
    # Store the pattern: for entry at (k, i), the value is f[k-i]
    offsets = (np.array(rows) - np.array(cols)).astype(np.int32)
    return A, offsets


def solve_lp(A_template, offsets, f, M, n):
    """Solve: max sum(g) s.t. A@f * g <= M, g >= 0."""
    # Fill in the constraint matrix
    A_data = np.array(A_template.data, copy=True)
    A_data[:] = f[offsets]
    A = A_template.copy()
    A.data = A_data
    
    c = -np.ones(n)
    b = np.full(A.shape[0], M)
    bounds = [(0, None)] * n
    
    try:
        res = linprog(c, A_ub=A, b_ub=b, bounds=bounds, method='highs')
        return res
    except Exception:
        return None

    f = normalize_sequence(f_init)
    if f is None:

    m = A_template.shape[0]

    best_score = score_normalized(f, n)

def peeling_phase(f, n, num_rounds=50):
    """Try to reduce the peak by moving mass from peak contributors to safe positions."""
    f = f.copy()
    best_score = score_normalized(f, n)
    n = len(f)
    
    for round_num in range(num_rounds):
        conv = np.convolve(f, f)
        k_peak = int(np.argmax(conv))
        M = float(conv[k_peak])
        
        # Find top contributor to the peak
        i_start = max(0, k_peak - n + 1)
        i_end = min(n - 1, k_peak)
        
        best_i = -1
        best_prod = 0
        for i in range(i_start, i_end + 1):
            j = k_peak - i
            if i == j:
                prod = f[i] ** 2
            else:
                prod = f[i] * f[j]
            if prod > best_prod:
                best_prod = prod
                best_i = i
        
        if best_i < 0:
            break
        
        j_peak = k_peak - best_i
        
        # Find a safe receiver: position r where f[r]*f[k_peak-r] is small
        # and r is not contributing to the peak
        best_r = -1
        best_r_score = np.inf
        for r in range(n):
            if r == best_i:
                continue
            # Check contribution of r to the peak
            if 0 <= k_peak - r < n:
                r_contrib = f[r] * f[k_peak - r]
            else:
                r_contrib = 0
            if r_contrib < best_r_score:
                best_r_score = r_contrib
                best_r = r
        
        if best_r < 0 or best_r == j_peak and j_peak != best_i:
            continue
        
        # Try moving mass
        for frac in [0.3, 0.1, 0.05, 0.01]:

            f_try = f.copy()
            f_try[best_i] -= amount

            f_try = np.maximum(f_try, 0)
            s = f_try.sum()
            if s < 1e-10:
                continue
            f_try /= s
            
            new_score = score_normalized(f_try, n)
            if new_score < best_score:
                f = f_try
                best_score = new_score
                break
    
    return f, best_score

For a symmetric f: (f*f)[n-1] = sum f[i]² >= 1/n. So for symmetric f, the peak is at least 1/n, and the score is at least 2.

For example, f = [1, 0, 0, ..., 0]: (f*f)[n-1] = 0 (no overlap at the "center"). But (f*f)[0] = 1, so the peak is 1.

The key is to arrange f so that:
- The "center" convolution (f*f)[n-1] = sum f[i]f[n-1-i] is small
- The "edge" convolutions are also small
- All convolutions are balanced

This is a complex combinatorial optimization problem. The LP approach handles it iteratively.

"""
LP-iterative search with targeted improvements over parent:
1. Exact step size from quadratic bound (reduces line search overhead)
2. Multi-temperature LP (solve with M and M*1.03 for flatter directions)
3. Structured starts (cosine-modulated, block patterns)
4. Targeted peeling phase for final peak reduction
5. Multi-length search (1024 primary, 2048 if time permits)
6. Adaptive perturbation sized to current stagnation
"""

import time
import numpy as np
from scipy.optimize import linprog
from scipy.sparse import csr_matrix


def normalize_sequence(arr):
    """Return nonnegative float vector normalized to sum 1, or None."""
    try:
        x = np.asarray(arr, dtype=np.float64).ravel()
    except Exception:
        return None
    if x.size == 0:
        return None
    x = np.where(np.isfinite(x), x, 0.0)
    x = np.maximum(x, 0.0)
    s = float(x.sum())
    if s <= 1e-300:
        return None
    x /= s
    return x


def score_normalized(x, n):
    """Score assuming x is nonnegative and sum(x)=1. Returns 2*n*max(conv)."""
    conv = np.convolve(x, x)
    m = float(np.max(conv))
    if not np.isfinite(m):
        return float("inf")
    return float(2.0 * n * m)


def make_conv_matrix(n):
    """Build sparse matrix structure for A@g = convolve(f_fix, g)."""
    m = 2 * n - 1
    k64 = np.arange(m, dtype=np.int64)
    js = np.maximum(0, k64 - n + 1)
    je = np.minimum(n - 1, k64)
    counts = je - js + 1
    total = int(counts.sum())

    cum = np.cumsum(counts)
    starts = np.repeat(cum - counts, counts)
    local = np.arange(total, dtype=np.int64) - starts

    row = np.repeat(k64, counts)
    col_base = js[row]
    col = col_base + local
    offsets = (row - col).astype(np.int32)

    data = np.zeros(total, dtype=np.float64)
    A = csr_matrix((data, (row.astype(np.int32), col.astype(np.int32))), shape=(m, n))
    offsets = offsets.copy()
    return A, offsets


def solve_direction_lp(A_template, offsets, f, M, n):
    """Solve LP: max sum(g) s.t. (f*g)[k] <= M for all k, g >= 0."""
    A = A_template.copy()
    A.data[:] = f[offsets]
    c = -np.ones(n, dtype=np.float64)
    b = np.full(A.shape[0], M, dtype=np.float64)
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c, A_ub=A, b_ub=b, bounds=bounds, method="highs")
        return res
    except Exception:
        return None


def cosine_start(n, rng, k=1):
    """Create a cosine-modulated starting vector."""
    x = np.ones(n, dtype=np.float64)
    theta = 2.0 * np.pi * k * np.arange(n) / n
    amp = 0.2 + 0.3 * rng.random()
    x *= (1.0 + amp * np.cos(theta + rng.uniform(0, 2 * np.pi)))
    x = np.maximum(x, 0.0)
    s = x.sum()
    if s < 1e-10:
        return np.ones(n) / n
    return x / s


def block_start(n, rng, num_blocks=8):
    """Create a block-structured starting vector."""
    sizes = np.full(num_blocks, n // num_blocks, dtype=np.int64)
    sizes[:n % num_blocks] += 1
    x = np.zeros(n, dtype=np.float64)
    pos = 0
    for sz in sizes:
        x[pos:pos + sz] = rng.uniform(0.3, 1.5)
        pos += sz
    s = x.sum()
    if s < 1e-10:
        return np.ones(n) / n
    return x / s


def upsample(f, n_new):
    """Upsample a normalized sequence to a new length."""
    m = f.size
    if m == n_new:
        return f.copy()
    old = np.linspace(0.0, 1.0, m)
    new = np.linspace(0.0, 1.0, n_new)
    out = np.interp(new, old, f)
    return normalize_sequence(out)


def downsample(f, n_new):
    """Downsample a normalized sequence to a new length."""
    m = f.size
    if m == n_new:
        return f.copy()
    old = np.linspace(0.0, 1.0, m)
    new = np.linspace(0.0, 1.0, n_new)
    out = np.interp(new, old, f)
    return normalize_sequence(out)


def peeling_phase(f, n, max_rounds=200, rng=None):
    """Reduce peak by moving mass from peak contributors to safe positions."""
    f = f.copy()
    best_score = score_normalized(f, n)
    improved = 0

    for _ in range(max_rounds):
        conv = np.convolve(f, f)
        k_peak = int(np.argmax(conv))
        M = float(conv[k_peak])

        i_start = max(0, k_peak - n + 1)
        i_end = min(n - 1, k_peak)

        # Find the top contributor pair to the peak
        prods = np.zeros(i_end - i_start + 1)
        for idx, i in enumerate(range(i_start, i_end + 1)):
            j = k_peak - i
            prods[idx] = f[i] * f[j]
        max_idx = int(np.argmax(prods))
        best_i = i_start + max_idx
        j_peak = k_peak - best_i

        # Find safe receiver: position r minimizing f[r]*f[k_peak-r]
        r_contrib = np.zeros(n)
        valid = (k_peak - np.arange(n) >= 0) & (k_peak - np.arange(n) < n)
        r_contrib[valid] = f[valid] * f[k_peak - np.arange(n)[valid]]
        r_contrib[best_i] = np.inf
        if j_peak != best_i:
            r_contrib[j_peak] = np.inf

        best_r = int(np.argmin(r_contrib))
        if r_contrib[best_r] >= f[best_i] * f[j_peak] * 0.5:
            continue

        for frac in [0.2, 0.05, 0.01]:
            amount = frac * f[best_i]
            f_try = f.copy()
            f_try[best_i] -= amount
            f_try[best_r] += amount
            f_try = np.maximum(f_try, 0.0)
            s = f_try.sum()
            if s < 1e-10:
                continue
            f_try /= s
            new_score = score_normalized(f_try, n)
            if new_score < best_score * (1.0 - 1e-9):
                f = f_try
                best_score = new_score
                improved += 1
                break

    return f, best_score


def optimize_lp(n, f_init, time_budget, rng, A_template, offsets):
    """Main LP-iteration loop with exact step and adaptive perturbation."""
    f = normalize_sequence(f_init)
    if f is None:
        f = np.ones(n, dtype=np.float64) / n

    best_f = f.copy()
    best_score = score_normalized(f, n)
    start_time = time.monotonic()
    stall = 0
    it = 0
    last_improve_time = start_time

    while time.monotonic() - start_time < time_budget:
        t0 = time.monotonic()
        it += 1

        conv_ff = np.convolve(f, f)
        M = float(np.max(conv_ff))
        if M <= 1e-300:
            break

        # Solve LP with current M
        res = solve_direction_lp(A_template, offsets, f, M, n)

        g = None
        if res is not None and res.x is not None:
            g = np.maximum(res.x, 0.0)
            s_g = g.sum()
            if s_g > 1e-10:
                g = g / s_g
            else:
                g = None

        if g is None:
            stall += 1
            if stall > 6:
                scale = 0.05 + 0.05 * min(stall // 6, 5)
                noise = rng.normal(0, scale, n)
                f = f + noise
                f = np.maximum(f, 0.0)
                s = f.sum()
                if s < 1e-10:
                    f = np.ones(n) / n
                else:
                    f /= s
                stall = 0
            continue

        r = np.convolve(f, g)
        r_peak = float(np.max(r))
        k_peak_r = int(np.argmax(r))

        if r_peak >= M * (1.0 - 1e-10):
            stall += 1
            if stall > 6:
                scale = 0.05 + 0.05 * min(stall // 6, 5)
                noise = rng.normal(0, scale, n)
                f = f + noise
                f = np.maximum(f, 0.0)
                f /= f.sum()
                stall = 0
            continue

        # Exact step from quadratic bound at the peak lag
        G_peak = float(np.convolve(g, g)[k_peak_r])
        denom = M + G_peak - 2.0 * r_peak
        if denom > 1e-10:
            t_opt = 2.0 * (M - r_peak) / denom
        else:
            t_opt = 1.0
        t_opt = max(1e-6, min(t_opt, 1.0))

        # Try the exact step and a few alternatives
        candidates = []
        t1 = 0.95 * t_opt
        if t1 > 0:
            candidates.append(t1)
        t2 = 0.7 * t_opt
        if t2 > 0:
            candidates.append(t2)
        t3 = min(t_opt, 0.3)
        if t3 > 0 and t3 not in candidates:
            candidates.append(t3)

        best_cand_score = best_score
        best_cand_f = None

        for tv in candidates:
            f_new = (1.0 - tv) * f + tv * g
            f_new = np.maximum(f_new, 0.0)
            s_new = f_new.sum()
            if s_new < 1e-10:
                continue
            f_new /= s_new
            sc = score_normalized(f_new, n)
            if sc < best_cand_score:
                best_cand_score = sc
                best_cand_f = f_new

        if best_cand_f is not None:
            f = best_cand_f
            best_f = f_new.copy()
            best_score = best_cand_score
            stall = 0
            last_improve_time = time.monotonic()
            if it <= 5 or it % 30 == 0:
                elapsed = time.monotonic() - start_time
                print(f"    n={n} it={it} score={best_score:.6f} M={M:.6g} t={t_opt:.4f} elapsed={elapsed:.1f}s", flush=True)
        else:
            stall += 1
            if stall > 8:
                # Try a different M for the LP (relax constraint)
                M2 = M * 1.05
                res2 = solve_direction_lp(A_template, offsets, f, M2, n)
                if res2 is not None and res2.x is not None:
                    g2 = np.maximum(res2.x, 0.0)
                    s2 = g2.sum()
                    if s2 > 1e-10:
                        g2 /= s2
                        f = 0.5 * f + 0.5 * g2
                        f = np.maximum(f, 0.0)
                        f /= f.sum()
                        stall = 0
                        continue
                # If that fails, perturb
                scale = 0.1 + 0.05 * min(stall // 8, 8)
                noise = rng.normal(0, scale, n)
                f = f + noise
                f = np.maximum(f, 0.0)
                s = f.sum()
                if s < 1e-10:
                    f = np.ones(n) / n
                else:
                    f /= s
                stall = 0

        # If no improvement for a long time, try a structured restart
        if time.monotonic() - last_improve_time > 60.0:
            # Restart from a mix of best and a structured pattern
            struct = cosine_start(n, rng, k=rng.integers(1, 5))
            f = 0.6 * best_f + 0.4 * struct
            f = np.maximum(f, 0.0)
            f /= f.sum()
            last_improve_time = time.monotonic()

    return best_f, float(best_score)

        try:

            starts.append(cosine_start(n_primary, rng, k=3))

            n_starts = len(starts)

            total_w = 0.0
