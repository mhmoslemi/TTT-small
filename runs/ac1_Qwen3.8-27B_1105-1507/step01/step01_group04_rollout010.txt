"""
Optimizes non-negative step function heights to minimize 2n*max(conv(f,f))/(sum(f))^2.
Uses multi-start mirror descent with adaptive step size, successive zero insertion
for length exploration, and refined optimization passes.
"""

import numpy as np
import time

def _fft_convolve_full(f):
    """Compute full linear convolution of f with itself using FFT."""
    n = len(f)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    F = np.fft.rfft(f, size)
    h = np.fft.irfft(F * F, size)
    return h[:2 * n - 1]

def _score_raw(f):
    """Fast score computation assuming f >= 0, sum(f) > 0.01."""
    n = len(f)
    h = _fft_convolve_full(f)
    M = float(np.max(h))
    S = float(np.sum(f))
    if S < 0.01:
        return float('inf')
    return 2.0 * n * M / (S * S)

def _normalize(f):
    """Normalize f to sum to 1, clip negatives."""
    f = np.maximum(f, 0.0)
    s = float(np.sum(f))
    if s < 1e-30:
        return None
    return f / s

def _pressure_vector(f, active_lags):
    """
    Compute p[j] = (1/|K|) * sum_{k in K} 2*f[k-j] for valid 0<=k-j<n.
    This is the average gradient component of max_k (f*f)_k.
    """
    n = len(f)
    p = np.zeros(n, dtype=np.float64)
    for k in active_lags:
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        if j_end >= j_start:
            lo = k - j_end
            hi = k - j_start
            p[j_start:j_end + 1] += 2.0 * f[lo:hi + 1][::-1]
    p /= float(len(active_lags))
    return p

def _mirror_descent(f_init, n_iter, eta_scale, threshold_frac, label=""):
    """
    Mirror descent with multiplicative updates.
    Uses adaptive step size and active-set subgradient.
    """
    f = np.asarray(f_init, dtype=np.float64).copy()
    f = _normalize(f)
    if f is None:
        return f_init.copy(), float('inf')
    n = len(f)

    prev_score = best_score
    stuck_count = 0

        h = _fft_convolve_full(f)
        M = float(np.max(h))
        if M <= 0:
            break
        
        threshold = threshold_frac * M + 1e-15
        active = np.where(h >= threshold)[0]
        if len(active) == 0:
            active = np.array([int(np.argmax(h))], dtype=np.int64)
        
        p = _pressure_vector(f, active)
        
        eta = eta_scale / M
        log_ratio = -eta * p
        log_ratio = np.clip(log_ratio, -30.0, 30.0)
        
        f = f * np.exp(log_ratio)
        f_norm = _normalize(f)
        if f_norm is None:
            f = np.ones(n) / n
        else:
            f = f_norm
        
        score = _score_raw(f)
        if score < best_score:
            best_score = score
            best_f = f.copy()
            stuck_count = 0
        else:
            stuck_count += 1
            # If stuck, try reducing step size
            # (handled by the caller via different eta_scale)
        
        if (it + 1) % 2000 == 0:
            print(f"    MD it={it+1}: best={best_score:.6f}, M={M:.8f}, |K|={len(active)}")
    
    return best_f, best_score

def _mirror_descent_linesearch(f_init, n_iter, eta_base, threshold_frac, label=""):

def _sz_i_insert(f, pos):
    """Insert a zero at position pos in sequence f, returning new sequence of length n+1."""
    n = len(f)
    new_f = np.zeros(n + 1)
    new_f[:pos] = f[:pos]
    new_f[pos+1:] = f[pos:]
    return new_f

def _cosine_init(n):
    j = np.arange(n, dtype=np.float64)
    f = np.cos(np.pi * (j + 0.5) / float(n)) ** 2
    f = np.maximum(f, 0.0)
    s = np.sum(f)
    if s > 0:
        f /= s
    return f

def _tapered_init(n, power):
    j = np.arange(n, dtype=np.float64)
    x = np.sin(np.pi * (j + 0.5) / float(n))
    f = x ** power
    f = np.maximum(f, 0.0)
    s = np.sum(f)
    if s > 0:
        f /= s
    return f

def _bimodal_init(n, center_frac, width_frac):
    """Initialize with two peaks."""
    j = np.arange(n, dtype=np.float64)
    c1 = center_frac * (n - 1)
    c2 = (1 - center_frac) * (n - 1)
    sigma = width_frac * n
    f = np.exp(-(j - c1)**2 / (2*sigma*sigma)) + np.exp(-(j - c2)**2 / (2*sigma*sigma))
    s = np.sum(f)
    if s > 0:
        f /= s
    return f

def _flat_topped_init(n, edge_frac):
    """Flat in the middle, tapered at edges."""
    j = np.arange(n, dtype=np.float64)
    x = j / (n - 1) if n > 1 else np.array([0.5])
    # Taper at edges
    taper_left = np.minimum(x / edge_frac, 1.0)
    taper_right = np.minimum((1 - x) / edge_frac, 1.0)
    f = taper_left * taper_right
    f = np.maximum(f, 0.0)
    s = np.sum(f)
    if s > 0:
        f /= s
    return f

def _random_smooth_init(n, rng, n_freqs=5):

    f = np.maximum(f, 0.0)
    s = np.sum(f)
    if s > 0:

    return f

def _fft_convolve_full(f):
    """Compute full linear convolution of f with itself using FFT."""
    n = len(f)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    F = np.fft.rfft(f, size)
    h = np.fft.irfft(F * F, size)
    return h[:2 * n - 1]

def _score_raw(f):
    """Fast score computation assuming f >= 0, sum(f) > 0.01."""
    n = len(f)
    h = _fft_convolve_full(f)
    M = float(np.max(h))
    S = float(np.sum(f))
    if S < 0.01:
        return float('inf')
    return 2.0 * n * M / (S * S)

def _normalize(f):
    """Normalize f to sum to 1, clip negatives."""
    f = np.maximum(f, 0.0)

def _pressure_vector(f, active_lags):
    """
    Compute p[j] = (1/|K|) * sum_{k in K} 2*f[k-j] for valid 0<=k-j<n.
    Average subgradient of max_k (f*f)_k.
    """
    n = len(f)
    p = np.zeros(n, dtype=np.float64)
    for k in active_lags:
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        if j_end >= j_start:
            lo = k - j_end
            hi = k - j_start
            p[j_start:j_end + 1] += 2.0 * f[lo:hi + 1][::-1]
    p /= float(len(active_lags))
    return p

        parent = np.clip(parent, 0.0, 1000.0)
    except Exception:
        parent = np.ones(512, dtype=np.float64)

    best_f = parent.copy()
    best_score = float(evaluate_sequence(best_f.tolist()))
    print("Parent: n={}, score={:.6f}".format(len(best_f), best_score))

    # === Phase 1: Sinusoidal parameterization (global search) ===
    print("\nPhase 1: Sinusoidal parameterization")
    start_sin = time.monotonic()
    sin_budget = min(180.0, (deadline - time.monotonic()) * 0.15)

    for n_s in [384, 512, 640, 768]:
        if time.monotonic() > deadline - sin_budget:
            break
        for n_harm in [4, 6, 8]:
            if time.monotonic() > deadline - 600:
                break
            t_rem = deadline - time.monotonic() - 600
            alloc = max(10.0, min(40.0, t_rem / 12))
            f_sin, s_sin = _sinusoidal_optimize(n_s, n_harm, rng, alloc)
            if f_sin is not None and s_sin < best_score:
                best_score = s_sin
                best_f = f_sin.copy()
                print("  ** Sinusoidal n={}, harm={}: {:.6f}".format(n_s, n_harm, best_score))

    # === Phase 2: Multi-start mirror descent ===
    print("\nPhase 2: Multi-start mirror descent")

    candidates = []
    p = _normalize(parent)
    if p is not None:
        candidates.append(("parent", p))

    for n in [384, 512, 640, 768, 896, 1024]:
        candidates.append(("cos_{}".format(n), _cosine_init(n)))

    for n in [512, 640, 768]:
        for pw in [0.75, 1.0, 1.5, 2.0, 2.5]:
            candidates.append(("tap_{}_{}".format(n, int(pw * 10)), _tapered_init(n, pw)))

    for n in [512, 768]:
        for ef in [0.15, 0.25, 0.35]:
            candidates.append(("flat_{}_{}".format(n, int(ef * 100)), _flat_topped_init(n, ef)))

    for n in [512, 768]:
        for cf in [0.3, 0.35, 0.4]:
            for wf in [0.12, 0.2, 0.3]:
                candidates.append(("bi_{}_{}".format(n, int(cf * 100)),
                                   _bimodal_init(n, cf, wf)))

    for n in [512, 768, 1024]:
        candidates.append(("rs_{}".format(n), _random_smooth_init(n, rng)))

    for n in [512, 768]:
        f_r = rng.random(n)
        f_rn = _normalize(f_r)
        if f_rn is not None:
            candidates.append(("rand_{}".format(n), f_rn))

    # Include current best if it's good
    best_norm = _normalize(best_f)
    if best_norm is not None:
        candidates.insert(0, ("best_so_far", best_norm))

    print("  {} candidates".format(len(candidates)))

    phase2_budget = (deadline - time.monotonic()) * 0.3
    time_per = max(4.0, phase2_budget / max(1, len(candidates)))

    phase2_results = []

    for idx, (name, f0) in enumerate(candidates):
        if time.monotonic() > deadline - (budget_s * 0.45):
            break

        n = len(f0)
        n_iter = int(max(400, min(2500, time_per * 250 / max(1, n / 512))))

        t0 = time.monotonic()
        f_md, s_md = _mirror_descent(f0, n_iter, 0.5, 0.7, label=name)
        elapsed = time.monotonic() - t0

        phase2_results.append((s_md, name, f_md))

        if s_md < best_score:
            best_score = s_md
            best_f = f_md.copy()
            print("  ** New best: {:.6f} from {} ({:.1f}s)".format(best_score, name, elapsed))

    # Sort results
    phase2_results.sort(key=lambda x: x[0])

    # === Phase 3: Successive Zero Insertion ===
    print("\nPhase 3: Successive Zero Insertion")

    top_k = min(4, len(phase2_results))
    sz_start_f = best_f.copy()
    sz_start_s = best_score

    for r in range(top_k):
        if time.monotonic() > deadline - (budget_s * 0.25):
            break

        s_cur, name_cur, f_cur = phase2_results[r]
        n_orig = len(f_cur)
        if n_orig > 1500:
            continue
        print("  SZI from {} (score={:.6f}, n={})".format(name_cur, s_cur, n_orig))

        f_work = _normalize(f_cur)
        s_work = _score_raw(f_work) if f_work is not None else float('inf')

        max_insertions = 80
        no_improve_count = 0

        for ins_round in range(max_insertions):
            if time.monotonic() > deadline - (budget_s * 0.2):
                break

            n_cur = len(f_work)
            if n_cur > 2500:
                break

            # Try strategic + random positions
            positions = [0, n_cur // 4, n_cur // 2, 3 * n_cur // 4, n_cur]
            positions.extend(rng.integers(0, n_cur + 1, size=8).tolist())
            # Also try positions near the peaks of the current autocorrelation
            h_cur = _fft_convolve_full(f_work)
            peak_pos = int(np.argmax(h_cur))
            positions.extend([max(0, peak_pos // 2), min(n_cur, peak_pos - n_cur // 2)])

            best_ins_score = s_work
            best_ins_f = f_work

            for pos in set(int(x) for x in positions):
                if pos < 0 or pos > n_cur:
                    continue
                f_ins = _sz_i_insert(f_work, pos)
                f_ins_norm = _normalize(f_ins)
                if f_ins_norm is None:
                    continue
                s_ins = _score_raw(f_ins_norm)
                if s_ins < best_ins_score:
                    best_ins_score = s_ins
                    best_ins_f = f_ins_norm

            if best_ins_score < s_work * 0.999:
                f_work = best_ins_f
                s_work = best_ins_score
                no_improve_count = 0
                # Brief re-optimization
                n_iter = int(min(800, max(200, 80000.0 / max(1, len(f_work)))))
                if (deadline - time.monotonic()) > 5:
                    f_work, s_work = _mirror_descent(f_work, n_iter, 0.3, 0.75)
                if s_work < best_score:
                    best_score = s_work
                    best_f = f_work.copy()
                    if ins_round % 10 == 0:
                        print("    ** SZI improvement: {:.6f} (n={})".format(best_score, len(f_work)))
            else:
                no_improve_count += 1
                if no_improve_count > 3:
                    break

        if s_work < best_score:
            best_score = s_work
            best_f = f_work.copy()
            print("  ** Final SZI best: {:.6f} (n={})".format(best_score, len(f_work)))

    # === Phase 4: Refined optimization ===
    print("\nPhase 4: Refined optimization")
    f_ref = _normalize(best_f)
    if f_ref is None:
        f_ref = _normalize(parent)

    # 4a: Multiple threshold settings
    for tf in [0.55, 0.7, 0.8, 0.9, 0.95]:
        if time.monotonic() > deadline - 40:
            break
        n_iter = int(min(4000, max(500, (deadline - time.monotonic() - 40) * 200 / max(1, len(f_ref)))))
        if n_iter < 200:
            break
        f_t, s_t = _mirror_descent(f_ref, n_iter, 0.3, tf, label="ref_tf_{}".format(int(tf * 100)))
        if s_t < best_score:
            best_score = s_t
            best_f = f_t.copy()
            f_ref = f_t.copy()
            print("  ** New best (tf={}): {:.6f}".format(tf, best_score))

    # 4b: Line search mirror descent
    if time.monotonic() < deadline - 60:
        n_iter = int(min(4000, max(500, (deadline - time.monotonic() - 40) * 150 / max(1, len(f_ref)))))
        f_ls, s_ls = _mirror_descent_linesearch(f_ref, n_iter, 0.6, 0.8, label="linesearch")
        if s_ls < best_score:
            best_score = s_ls
            best_f = f_ls.copy()
            f_ref = f_ls.copy()
            print("  ** New best (line search): {:.6f}".format(best_score))

    # 4c: Perturbation restarts
    if time.monotonic() < deadline - 60:
        for pert_frac in [0.005, 0.02, 0.05, 0.08]:
            if time.monotonic() > deadline - 30:
                break
            noise = rng.random(len(f_ref)) * pert_frac
            f_pert = _normalize(f_ref + noise)
            if f_pert is None:
                continue
            n_iter = int(min(3000, max(300, (deadline - time.monotonic() - 10) * 150 / max(1, len(f_pert)))))
            if n_iter < 100:
                break
            f_p, s_p = _mirror_descent(f_pert, n_iter, 0.4, 0.8)
            if s_p < best_score:
                best_score = s_p
                best_f = f_p.copy()
                f_ref = f_p.copy()
                print("  ** New best (pert={:.3f}): {:.6f}".format(pert_frac, best_score))

    # 4d: SZI from refined best
    if time.monotonic() < deadline - 40:
        f_sz = _normalize(best_f)
        s_sz = _score_raw(f_sz) if f_sz is not None else float('inf')
        for _ in range(40):
            if time.monotonic() > deadline - 20:
                break
            n_cur = len(f_sz)
            if n_cur > 2500:
                break
            positions = [n_cur // 3, 2 * n_cur // 3, n_cur // 2,
                         rng.integers(0, n_cur + 1, size=3).tolist()]
            best_ins = (f_sz, s_sz)
            for pos in positions:
                pos = int(pos)
                if pos < 0 or pos > n_cur:
                    continue
                f_ins = _normalize(_sz_i_insert(f_sz, pos))
                if f_ins is None:
                    continue
                s_ins = _score_raw(f_ins)
                if s_ins < best_ins[1]:
                    best_ins = (f_ins, s_ins)
            if best_ins[1] < s_sz * 0.999:
                f_sz, s_sz = best_ins
                n_iter = int(min(600, max(100, (deadline - time.monotonic() - 10) * 80 / max(1, len(f_sz)))))
                if n_iter > 50:
                    f_sz, s_sz = _mirror_descent(f_sz, n_iter, 0.3, 0.8)
                if s_sz < best_score:
                    best_score = s_sz
                    best_f = f_sz.copy()
                    print("  ** New best (SZI2): {:.6f} (n={})".format(best_score, len(f_sz)))
            else:
                break

    # === Final verification ===
    final_list = [float(x) for x in best_f]
    final_score = float(evaluate_sequence(final_list))
    print("\nFinal verified score: {:.6f} (n={})".format(final_score, len(final_list)))

    if not np.isfinite(final_score) or final_score > 2.0:
        print("  Falling back to parent.")
        return [float(x) for x in parent]

    return final_list
