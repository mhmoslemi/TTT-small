"""
Improved optimization for minimizing 2n*max(conv(a,a))/sum(a)^2.
Combines fast LP-based direction finding, targeted deflation at peak lags,
adaptive step sizes, perturbation-based restarts, and multi-length exploration.

import numpy as np
import time
from scipy.optimize import linprog
from scipy.sparse import csr_matrix


def build_conv_matrix_fast(f):
    """Build sparse matrix A where (A@g)[k] = convolve(f,g)[k]."""
    n = len(f)
    all_rows = []
    all_cols = []
    all_vals = []
    for d in range(n):
        length = min(n, 2 * n - 1 - d)
        cols_d = np.arange(length, dtype=np.int32)
        rows_d = cols_d + d
        all_rows.append(rows_d)
        all_cols.append(cols_d)
        all_vals.append(np.full(length, f[d], dtype=np.float64))
    rows = np.concatenate(all_rows)
    cols = np.concatenate(all_cols)
    vals = np.concatenate(all_vals)
    return csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))


def solve_lp_direction(f, M):
    """Solve: max sum(g) s.t. g >= 0, A@g <= M."""
    n = len(f)
    A = build_conv_matrix_fast(f)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    try:
        result = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
        if result.success:
            return result.x
    except Exception:
        pass
    return None


def compute_gradient_at_peak(f, k_star):
    """Compute gradient of (f*f)[k_star] w.r.t. f."""
    n = len(f)
    grad = np.zeros(n)
    for j in range(n):
        idx = k_star - j
        if 0 <= idx < n:
            grad[j] = 2.0 * f[idx]
    return grad


def deflation_step(f, alpha, k_star=None):
    """Perform a deflation step at the peak lag."""
    n = len(f)
    if k_star is None:
        conv = np.convolve(f, f)
        k_star = int(np.argmax(conv))
    S = float(np.sum(f))
    grad = compute_gradient_at_peak(f, k_star)
    # Subtract alpha * grad, clip, renormalize
    new_f = f - alpha * grad
    new_f = np.clip(new_f, 0.0, 1000.0)
    new_sum = float(np.sum(new_f))
    if new_sum > 1e-15:
        new_f = new_f * (S / new_sum)
    return new_f

            gf, gs = deflation_polish(best_sequence, deadline - 2, alpha_init=0.002,
                                      max_iters=50000, verbose_prefix="[final_defl] ")
            if gs < best_score - 1e-10:
                best_score = gs
                best_sequence = gf.copy()
                print(f"  Final deflation: score={best_score:.7f}")

    if best_sequence is None:
        best_sequence = np.ones(128)

    elapsed = time.monotonic() - t0
    print(f"\nFINAL: n={len(best_sequence)}, score={best_score:.7f}, time={elapsed:.1f}s")
    return [float(x) for x in best_sequence]
