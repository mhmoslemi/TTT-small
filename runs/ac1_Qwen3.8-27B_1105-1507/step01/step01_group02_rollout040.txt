Combines Nesterov-accelerated subgradient descent with spectral flattening,
soft-max peak weighting, and simulated annealing restarts.

import numpy as np

def fast_convolve(a):
    """Full linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    N = 1 << (size - 1).bit_length()
    fa = np.fft.rfft(a, N)
    fc = fa * fa
    result = np.fft.irfft(fc, N)[:size]
    result[np.abs(result) < 1e-12] = 0.0
    result[result < 0] = 0.0
    return result

def project_simplex(v):
    """Euclidean projection onto probability simplex."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cv = np.cumsum(u) - 1.0
   .tmp = u * np.arange(1, n + 1) - cv
    rho_candidates = np.where(.tmp > 0)[0]
    if len(rho_candidates) == 0:
        return np.ones(n) / n
    rho = rho_candidates[-1]
    theta = cv[rho] / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def spectral_flatten(a, power=0.3):
    """Compress magnitude spectrum dynamic range."""
    n = len(a)
    N = 1 << (2 * n - 2).bit_length()
    A = np.fft.rfft(a, N)
    mag = np.abs(A)
    angle = np.angle(A)
    mmax = mag.max()
    if mmax < 1e-15:
        return a.copy()
    mag_norm = mag / mmax
    mag_new = (mag_norm ** power) * mmax
    A_new = mag_new * np.exp(1j * angle)
    # Ensure DC is real
    A_new[0] = A_new[0].real + 0j
    a_new = np.fft.irfft(A_new, N)[:n]
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s > 1e-10:
        a_new /= s
    else:
        a_new = a.copy()
    return a_new

def alt_projection_init(n, rng, n_iter=100):
    """Alternating projection for flat-spectrum initialization."""
    N = 1 << (2 * n - 2).bit_length()
    phases = rng.uniform(0, 2 * np.pi, N // 2 + 1)
    F = np.exp(1j * phases)
    F[0] = 1.0 + 0j
    a = np.ones(n) / n
    for _ in range(n_iter):
        a_full = np.fft.irfft(F, N)
        a = np.maximum(a_full[:n], 0.0)
        s = a.sum()
        if s < 1e-10:
            a = np.ones(n) / n
        else:
            a /= s
        F = np.fft.rfft(a, N)
        mag = np.abs(F)
        # Target: compress towards flat
        mmax = mag.max()
        if mmax > 0:
            mag_target = (mag / mmax) ** 0.5 * mmax
        else:
            mag_target = mag
        F = mag_target * np.exp(1j * np.angle(F))
        F[0] = F[0].real + 0j
    return a

Hybrid optimizer for min 2n*max(conv(a,a))/(sum a)^2 with a>=0.
Uses FFT-accelerated subgradient with soft-max weighting, Nesterov momentum,
spectral flattening, mirror descent steps, and genetic blending.

import numpy as np
import time


def fast_convolve(a):
    """Full auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    N = 1 << (size - 1).bit_length()
    fa = np.fft.rfft(a, N)
    fc = fa * fa
    result = np.fft.irfft(fc, N)[:size]
    result[result < 0] = 0.0
    return result


def project_simplex(v):
    """Project onto {x >= 0, sum(x) = 1}."""
    n = len(v)
    u = np.sort(v)[::-1]
    cv = np.cumsum(u) - 1.0
    rho = np.where(u * np.arange(1, n + 1) > cv)[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho = rho[-1]
    theta = cv[rho] / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def subgrad_fft(a, weights, n, N):
    """Weighted subgradient via FFT: 2*(weights ⋆ a)[:n]."""
    A = np.fft.rfft(a, N)
    W = np.fft.rfft(weights, N)
    return 2.0 * np.fft.irfft(np.conj(W) * A, N)[:n]

    """Compress magnitude spectrum to push towards flat auto-convolution."""
    n = len(a)
    N = 1 << (2 * n - 2).bit_length()
    A = np.fft.rfft(a, N)
    mag = np.abs(A)
    mmax = mag.max()
    if mmax < 1e-15:

def fast_convolve(a):

def project_simplex(v):

            cand_scored.append((c.max() * n_best, a))
        cand_scored.sort(key=lambda x: x[0])

        for i, (csc, a) in enumerate(cand_scored[:4]):
            if time.time() > deadline - 40:
                break
            t_alloc = min(150.0, deadline - time.time() - 30)
            if t_alloc < 20:
                break
            a_opt = optimize_seq(a, time.time() + t_alloc, rng)
            sc = evaluate_sequence(a_opt.tolist())
            if np.isfinite(sc) and sc < best_score:
                best_score = sc
                best_seq = a_opt.copy()
                print(f"    Blend {i}: NEW BEST {best_score:.6f}")

    # Phase 3: Final extended polish
    if best_seq is not None and time.time() < deadline - 15:
        print(f"\nPhase 3: Final polish (current best={best_score:.6f})")
        a_final = optimize_seq(best_seq, deadline, rng)
        sc_final = evaluate_sequence(a_final.tolist())
        if np.isfinite(sc_final) and sc_final < best_score:
            best_score = sc_final
            best_seq = a_final
        print(f"    After polish: {best_score:.6f}")

    # Fallback
    if best_seq is None:
        print("FALLBACK to uniform")
        best_seq = np.ones(n0) / n0

    result = [float(x) for x in best_seq]
    final_score = evaluate_sequence(result)
    print(f"\nFINAL RESULT: score={final_score:.6f}, length={len(result)}")
    return result
