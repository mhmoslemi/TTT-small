"""
Combined optimization for minimizing 2*n*max(conv(f,f))/sum(f)^2.
Uses mirror descent, LP refinement, and local search with multiple
initializations and sequence lengths.
"""

import numpy as np
from scipy.optimize import linprog
import time


def compute_score(f):
    """Compute score = 2*n*max(conv(f,f))/sum(f)^2."""
    n = len(f)
    S = f.sum()
    if S < 1e-30:
        return np.inf
    conv = np.convolve(f, f)
    return 2.0 * n * conv.max() / (S * S)


def compute_score_normalized(f):
    """Compute 2*n*max(conv(f,f)) assuming sum(f)=1."""
    n = len(f)
    conv = np.convolve(f, f)
    return 2.0 * n * conv.max()


def gradient_of_conv_entry(f, k):
    """Compute gradient of conv(f,f)[k] w.r.t. f.
    grad[j] = 2*f[k-j] if 0 <= k-j < n, else 0.
    """
    n = len(f)
    grad = np.zeros(n)
    # For each j, grad[j] = 2*f[k-j] if 0 <= k-j < n
    for j in range(n):
        idx = k - j
        if 0 <= idx < n:
            grad[j] = 2.0 * f[idx]
    return grad


def gradient_of_conv_entry_fast(f, k):
    """Fast version using array operations."""
    n = len(f)
    # grad[j] = 2*f[k-j] for j where 0 <= k-j < n
    # i.e., j in [max(0, k-n+1), min(n-1, k)]
    j_start = max(0, k - n + 1)
    j_end = min(n, k + 1)
    grad = np.zeros(n)
    # k-j ranges from k-j_start to k-j_end+1
    idx_start = k - (j_end - 1)
    idx_end = k - j_start + 1
    grad[j_start:j_end] = 2.0 * f[idx_start:idx_end]
    return grad


def subgradient_max_conv(f, top_k=3):
    """Compute a subgradient of max conv(f,f) using top-k peaks."""
    n = len(f)
    conv = np.convolve(f, f)
    
    # Get top-k indices
    top_indices = np.argsort(conv)[-top_k:]
    
    # Weight peaks by their value (emphasize higher peaks)
    weights = np.zeros(2 * n - 1)
    for idx in top_indices:
        weights[idx] = conv[idx]
    total = weights.sum()
    if total < 1e-30:
        weights[np.argmax(conv)] = 1.0
        return gradient_of_conv_entry(f, np.argmax(conv))
    weights /= total
    
    # Weighted sum of gradients
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        if weights[k] > 0:
            grad += weights[k] * gradient_of_conv_entry_fast(f, k)
    return grad


def mirror_descent_optimize(f, n_iter=5000, verbose=False, rng=None):
    """Optimize f using mirror descent (multiplicative updates)."""
    n = len(f)
    f = f.astype(np.float64).copy()
    S = f.sum()
    if S < 1e-10:
        return f, np.inf
    f = f / S
    
    best_f = f.copy()
    best_score = compute_score_normalized(f)
    
    alpha = 0.1  # base step size
    patience = 0
    patience_limit = 500
    
    for it in range(n_iter):
        # Compute subgradient
        grad = subgradient_max_conv(f, top_k=3)
        
        # Center the gradient (to maintain sum constraint)
        grad_centered = grad - grad.mean()
        
        # Multiplicative update
        f_new = f * np.exp(-alpha * grad_centered)
        f_new = np.clip(f_new, 0, None)
        S_new = f_new.sum()
        if S_new < 1e-30:
            alpha *= 0.5
            continue
        f_new = f_new / S_new
        
        score = compute_score_normalized(f_new)
        
        if score < best_score - 1e-12:
            best_f = f_new.copy()
            best_score = score
            f = f_new
            patience = 0
        else:
            f = f_new  # Still update f (it might help later)
            patience += 1
            if patience > patience_limit:
                alpha *= 0.5
                patience = 0
                if alpha < 1e-6:
                    break
        
        if verbose and it % 500 == 0:
            print(f"      MD it={it}, score={best_score:.6f}, alpha={alpha:.6f}")
    
    return best_f, best_score


def build_conv_matrix(f):
    """Build matrix A such that A @ b = np.convolve(f, b, mode='full')."""
    n = len(f)
    A = np.zeros((2 * n - 1, n))
    for k in range(2 * n - 1):
        j_start = max(0, k - n + 1)
        j_end = min(n, k + 1)
        for j in range(j_start, j_end):
            A[k, j] = f[k - j]
    return A


def lp_refine(f, time_budget, t0=0.3, verbose=False, rng=None):
    """LP-based refinement (similar to previous approach)."""
    if rng is None:
        rng = np.random.default_rng(0)
    
    n = len(f)
    f = f.astype(np.float64).copy()
    S0 = f.sum()
    if S0 < 1e-10:
        return f, np.inf
    f = f / S0
    
    best_f = f.copy()
    best_score = compute_score_normalized(f)
    t = t0
    
    start = time.monotonic()
    it = 0
    stall_count = 0
    
    while time.monotonic() - start < time_budget:
        conv = np.convolve(f, f)
        M = conv.max()
        
        A = build_conv_matrix(f)
        c_obj = -np.ones(n)
        b_ub = np.full(2 * n - 1, M)
        bounds = [(0, None)] * n
        
        try:
            res = linprog(c_obj, A_ub=A, b_ub=b_ub, bounds=bounds,
                         method='highs', options={'presolve': True})
        except Exception:
            break
        
        if not res.success or res.x is None:
            break
        
        b = np.clip(res.x, 0, None)
        b_sum = b.sum()
        if b_sum < 1e-15:
            break
        
        g = b / b_sum
        f_new = (1 - t) * f + t * g
        f_new = np.clip(f_new, 0, None)
        S_new = f_new.sum()
        if S_new < 1e-10:
            t *= 0.5
            continue
        f_new = f_new / S_new
        
        new_score = compute_score_normalized(f_new)
        
        if new_score < best_score - 1e-12:
            best_f = f_new.copy()
            best_score = new_score
            f = f_new
            stall_count = 0
            if verbose and it % 10 == 0:
                print(f"      LP it={it}, score={best_score:.6f}, t={t:.4f}")
        else:
            t *= 0.7
            stall_count += 1
            if stall_count > 30:
                # Random perturbation
                f_cand = f + rng.normal(0, 0.02, n)
                f_cand = np.clip(f_cand, 0, None)
                S_c = f_cand.sum()
                if S_c > 1e-10:
                    f_cand /= S_c
                    if compute_score_normalized(f_cand) < best_score:
                        f = f_cand
                        best_f = f.copy()
                        best_score = compute_score_normalized(f)
                        t = t0 * 0.5
                        stall_count = 0
                if time.monotonic() - start > time_budget * 0.85:
                    break
        
        it += 1
    
    return best_f, best_score


def fourier_init(n, rng):
    """Initialize using Fourier method: random phase with flat magnitude."""
    # Create a flat-magnitude spectrum with random phase
    freqs = np.fft.fftfreq(n)
    # Only use positive frequencies for a real signal
    N = 2 * n - 1  # Length of convolution
    # We want the convolution to be approximately uniform
    # In frequency domain: |F|^2 should be approximately constant
    # So |F| should be approximately constant
    
    # Use DFT of length N
    F_mag = np.ones(N)
    F_phase = rng.uniform(0, 2 * np.pi, N)
    F = F_mag * np.exp(1j * F_phase)
    
    # Enforce Hermitian symmetry for real f
    F[0] = F_mag[0]  # DC component is real
    for k in range(1, (N + 1) // 2):
        if k < N - k:
            F[N - k] = np.conj(F[k])
    
    f = np.fft.ifft(F).real
    f = np.clip(f, 0, None)
    f = f[:n]  # Take first n elements
    S = f.sum()
    if S < 1e-10:
        return np.ones(n) / n
    return f / S


def random_init(n, rng):
    """Random initialization with various distributions."""
    method = rng.integers(4)
    if method == 0:
        f = rng.uniform(0.1, 1.0, n)
    elif method == 1:
        f = rng.exponential(1.0, n) + 0.01
    elif method == 2:
        f = np.abs(rng.normal(0, 1, n)) + 0.01
    else:
        f = rng.chisquare(df=rng.integers(1, 10), size=n) + 0.01
    S = f.sum()
    return f / S


def local_search(f, rng, n_iter=2000):

    n = len(f)
    f = f.astype(np.float64).copy()
    S = f.sum()
    f = f / S
    best_f = f.copy()
    best_score = compute_score_normalized(f)
    
    for _ in range(n_iter):
        f_cand = f.copy()
        width = max(1, int(rng.integers(1, max(2, n // 32))))
        start = int(rng.integers(0, n - width))
        scale = np.exp(rng.normal(0, 0.3))
        f_cand[start:start + width] *= scale
        f_cand = np.clip(f_cand, 0, None)
        S_new = f_cand.sum()
        if S_new < 1e-10:

        f_cand = f_cand / S_new

        if score < best_score:

"""
Combined optimization for minimizing 2*n*max(conv(f,f))/sum(f)^2.
Uses mirror descent (multiplicative updates), LP refinement, and local search
with multiple initializations (random, Fourier-based, parent) and sequence lengths.

import numpy as np
from scipy.optimize import linprog
import time


def compute_score(f):
    """Compute score = 2*n*max(conv(f,f))/sum(f)^2."""
    n = len(f)
    S = f.sum()
    if S < 1e-30:
        return np.inf
    conv = np.convolve(f, f)
    return 2.0 * n * conv.max() / (S * S)


def compute_score_normalized(f):
    """Compute 2*n*max(conv(f,f)) assuming sum(f)=1."""
    n = len(f)
    conv = np.convolve(f, f)
    return 2.0 * n * conv.max()


def gradient_of_conv_entry_fast(f, k):
    """Fast gradient of conv(f,f)[k] w.r.t. f.
    grad[j] = 2*f[k-j] if 0 <= k-j < n, else 0.
    """
    n = len(f)
    j_start = max(0, k - n + 1)
    j_end = min(n, k + 1)
    grad = np.zeros(n)
    if j_end > j_start:
        idx_start = k - (j_end - 1)
        idx_end = k - j_start + 1
        grad[j_start:j_end] = 2.0 * f[idx_start:idx_end][::-1]
    return grad


def subgradient_max_conv(f, top_k=5):
    """Compute subgradient of max conv(f,f) using top-k peaks weighted by value."""
    n = len(f)
    conv = np.convolve(f, f)
    
    # Get top-k indices
    if top_k < len(conv):
        top_indices = np.argsort(conv)[-top_k:]
    else:
        top_indices = np.arange(len(conv))
    
    # Weight peaks by their value
    weights = np.zeros(len(conv))
    for idx in top_indices:
        if conv[idx] > 0:
            weights[idx] = conv[idx]
    total = weights.sum()
    if total < 1e-30:
        k_max = np.argmax(conv)
        return gradient_of_conv_entry_fast(f, k_max)
    weights /= total
    
    # Weighted sum of gradients
    grad = np.zeros(n)
    for k in range(len(conv)):
        if weights[k] > 1e-15:
            grad += weights[k] * gradient_of_conv_entry_fast(f, k)
    return grad


def mirror_descent_optimize(f, n_iter=5000, time_budget=None, verbose=False, rng=None):
    """Optimize f using mirror descent (multiplicative updates)."""
    n = len(f)
    f = f.astype(np.float64).copy()
    S = f.sum()
    if S < 1e-10:
        return f, np.inf
    f = f / S
    
    best_f = f.copy()
    best_score = compute_score_normalized(f)
    
    alpha = 0.15
    patience = 0
    patience_limit = 300
    start_time = time.monotonic()
    
    for it in range(n_iter):
        if time_budget is not None and time.monotonic() - start_time > time_budget:
            break
        
        grad = subgradient_max_conv(f, top_k=5)
        grad_centered = grad - grad.mean()
        
        # Multiplicative update
        f_new = f * np.exp(-alpha * grad_centered)
        f_new = np.clip(f_new, 0.0, None)
        S_new = f_new.sum()
        if S_new < 1e-30:
            alpha *= 0.5
            continue
        f_new = f_new / S_new
        
        score = compute_score_normalized(f_new)
        
        if score < best_score - 1e-12:
            best_f = f_new.copy()
            best_score = score
            f = f_new
            patience = 0
            if verbose and it % 500 == 0:
                print(f"      MD it={it}, score={best_score:.6f}, alpha={alpha:.4f}")
        else:
            f = f_new
            patience += 1
            if patience % 50 == 0:
                alpha *= 0.85
            if patience > patience_limit:
                alpha *= 0.5
                patience = 0
                if alpha < 1e-7:
                    if verbose:
                        print(f"      MD: alpha too small, stopping at it={it}")
                    break
        
        # Periodic restart to escape plateaus
        if patience == patience_limit // 2 and rng is not None:
            f_pert = f * (1 + 0.05 * rng.normal(0, 1, n))
            f_pert = np.clip(f_pert, 0, None)
            S_p = f_pert.sum()
            if S_p > 1e-10:
                f_pert /= S_p
                if compute_score_normalized(f_pert) < best_score:
                    f = f_pert
                    best_f = f.copy()
                    best_score = compute_score_normalized(f)
                    alpha = max(alpha, 0.05)
                    patience = 0
    
    return best_f, best_score


def build_conv_matrix(f):
    """Build matrix A such that A @ b = np.convolve(f, b, mode='full')."""
    n = len(f)
    A = np.zeros((2 * n - 1, n))
    for k in range(2 * n - 1):
        j_start = max(0, k - n + 1)
        j_end = min(n, k + 1)
        for j in range(j_start, j_end):
            A[k, j] = f[k - j]
    return A


def lp_refine(f, time_budget, t0=0.3, verbose=False, rng=None):
    """LP-based refinement."""
    if rng is None:
        rng = np.random.default_rng(0)
    
    n = len(f)
    f = f.astype(np.float64).copy()
    S0 = f.sum()
    if S0 < 1e-10:
        return f, np.inf
    f = f / S0
    
    best_f = f.copy()
    best_score = compute_score_normalized(f)
    t = t0
    
    start = time.monotonic()
    it = 0
    stall_count = 0
    
    while time.monotonic() - start < time_budget:
        conv = np.convolve(f, f)
        M = conv.max()
        
        A = build_conv_matrix(f)
        c_obj = -np.ones(n)
        b_ub = np.full(2 * n - 1, M)
        bounds = [(0, None)] * n
        
        try:
            res = linprog(c_obj, A_ub=A, b_ub=b_ub, bounds=bounds,
                         method='highs', options={'presolve': True})
        except Exception:
            break
        
        if not res.success or res.x is None:
            break
        
        b = np.clip(res.x, 0, None)
        b_sum = b.sum()
        if b_sum < 1e-15:
            break
        
        g = b / b_sum
        f_new = (1 - t) * f + t * g
        f_new = np.clip(f_new, 0, None)
        S_new = f_new.sum()
        if S_new < 1e-10:
            t *= 0.5
            continue
        f_new = f_new / S_new
        
        new_score = compute_score_normalized(f_new)
        
        if new_score < best_score - 1e-12:
            best_f = f_new.copy()
            best_score = new_score
            f = f_new
            stall_count = 0
            if verbose and it % 10 == 0:
                print(f"      LP it={it}, score={best_score:.6f}, t={t:.4f}")
        else:
            t *= 0.7
            stall_count += 1
            if stall_count > 30:
                f_cand = f + rng.normal(0, 0.02, n)
                f_cand = np.clip(f_cand, 0, None)
                S_c = f_cand.sum()
                if S_c > 1e-10:
                    f_cand /= S_c
                    if compute_score_normalized(f_cand) < best_score:
                        f = f_cand
                        best_f = f.copy()
                        best_score = compute_score_normalized(f)
                        t = t0 * 0.5
                        stall_count = 0
                if time.monotonic() - start > time_budget * 0.85:
                    break
        
        it += 1
    
    return best_f, best_score


def fourier_init(n, rng):
    """Initialize using Fourier method: random phase flat magnitude spectrum."""
    N = 2 * n - 1
    F_phase = rng.uniform(0, 2 * np.pi, N)
    F = np.exp(1j * F_phase)
    F[0] = 1.0
    for k in range(1, (N + 1) // 2):
        if k < N - k:
            F[N - k] = np.conj(F[k])
    f = np.fft.ifft(F).real
    f = np.clip(f, 0, None)
    f = f[:n]
    S = f.sum()
    if S < 1e-10:
        return np.ones(n) / n
    return f / S


def random_init(n, rng):
    """Random initialization with various distributions."""
    method = rng.integers(5)
    if method == 0:
        f = rng.uniform(0.1, 1.0, n)
    elif method == 1:
        f = rng.exponential(1.0, n) + 0.01
    elif method == 2:
        f = np.abs(rng.normal(0, 1, n)) + 0.01
    elif method == 3:
        f = rng.chisquare(df=rng.integers(2, 8), size=n) + 0.01
    else:
        # Bimodal
        f = np.abs(rng.normal(0, 1, n))
        f[f < 0.3] = 0.01
        f = f + 0.01
    S = f.sum()
    return f / S


def structured_init(n, rng):
    """Structured initialization: sinusoidal, polynomial, etc."""
    x = np.linspace(0, 2 * np.pi, n)
    method = rng.integers(4)
    if method == 0:
        f = 1.0 + 0.5 * np.sin(rng.uniform(0, 2*np.pi) + x)
    elif method == 1:
        f = np.cos(x * rng.uniform(1, 5)) ** 2 + 0.1
    elif method == 2:
        # Polynomial
        coeffs = rng.uniform(-1, 1, 4)
        f = np.polyval(coeffs, np.linspace(-1, 1, n))
        f = np.abs(f) + 0.1
    else:
        # Gaussian mixture
        centers = rng.uniform(0, n, 3)
        widths = rng.uniform(1, n//4, 3)
        f = np.zeros(n)
        for c, w in zip(centers, widths):
            f += np.exp(-0.5 * ((np.arange(n) - c) / w) ** 2)
        f += 0.05
    f = np.clip(f, 0.01, None)
    S = f.sum()
    return f / S
