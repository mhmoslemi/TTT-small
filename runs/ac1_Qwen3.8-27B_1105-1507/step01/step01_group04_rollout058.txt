"""
Optimizes non-negative step function heights to minimize 2n*max(conv(f,f))/(sum(f))^2.
Uses a combination of:
1. Soft-max smoothed gradient descent (FISTA) for the main optimization
2. Mirror descent (multiplicative updates) as an alternative
3. LP-based Frank-Wolfe for final polish
4. Multiple sequence lengths and initializations
"""

import numpy as np
import time

def fft_convolve_full(f):
    """Compute full linear convolution of f with itself using FFT."""
    n = len(f)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    F = np.fft.rfft(f, size)
    h = np.fft.irfft(F * F, size)
    return h[:2 * n - 1]

def score_raw(f):
    """Fast score computation assuming f >= 0, sum(f) > 0.01."""
    n = len(f)
    h = fft_convolve_full(f)
    M = float(np.max(h))
    S = float(np.sum(f))
    if S < 0.01:
        return float('inf')
    return 2.0 * n * M / (S * S)

def normalize(f):
    """Normalize f to sum to 1, clip negatives."""
    f = np.maximum(f, 0.0)
    s = float(np.sum(f))
    if s < 1e-30:
        return None
    return f / s

def grad_max_conv(f, threshold_frac=1.0):

    Compute gradient of max_k (f*f)(k) using active set (top fraction).
    Returns gradient vector of length n.
    """
    n = len(f)
    h = fft_convolve_full(f)
    M = float(np.max(h))
    if M <= 0:
        return np.zeros(n)
    
    threshold = threshold_frac * M
    active = np.where(h >= threshold)[0]
    if len(active) == 0:
        active = np.array([int(np.argmax(h))])
    
    # Gradient of max is average of gradients at active points
    # dh(k)/df(j) = 2*f(k-j) for valid k-j (when k != 2j)
    # Simplified: use 2*f(k-j) for all valid k-j
    grad = np.zeros(n, dtype=np.float64)

        j_lo = max(0, k - n + 1)

            # k-j_lo = k - max(0, k-n+1), k-j_hi = k - min(n-1, k)
            # The values f(k-j) for j from j_lo to j_hi are f[k-j_hi], f[k-j_hi+1], ..., f[k-j_lo]
            # So grad[j_lo:j_hi+1] += 2*f[k-j_hi : k-j_lo+1]
            idx = k - np.arange(j_hi + 1 - j_lo)  # k-j_hi, k-j_hi+1, ..., k-j_lo
            grad[j_lo:j_hi+1] += 2.0 * f[idx]
    
    grad /= float(len(active))
    return grad

def soft_max_conv(f, tau):
    """Compute log-sum-exp approximation of max of convolution."""
    h = fft_convolve_full(f)
    max_h = np.max(h)
    if max_h < 0:
        return max_h  # all negative, just return max
    # log(sum(exp(h/tau))) * tau ≈ tau * log(sum(exp((h-max_h)/tau))) + max_h
    shifted = (h - max_h) / tau
    log_sum = np.log(np.sum(np.exp(shifted)))
    return tau * log_sum + max_h

def gradient_soft_max(f, tau):
    """Gradient of the soft-max approximation of max convolution."""
    n = len(f)
    h = fft_convolve_full(f)
    max_h = np.max(h)
    shifted = (h - max_h) / tau
    weights = np.exp(shifted) / np.sum(np.exp(shifted))
    
    grad = np.zeros(n, dtype=np.float64)
    for k in range(len(h)):
        if weights[k] < 1e-15:

        j_lo = max(0, k - n + 1)
        j_hi = min(n - 1, k)
        if j_hi >= j_lo:
            idx = k - np.arange(j_hi + 1 - j_lo)
            grad[j_lo:j_hi+1] += 2.0 * weights[k] * f[idx]
    
    return grad

    f = normalize(f)

"""
Optimizes non-negative step function heights to minimize 2n*max(conv(f,f))/(sum(f))^2.
Uses temperature-annealed mirror descent with backtracking step size,
combined with LP Frank-Wolfe refinement and multi-resolution search.
"""

import numpy as np
import time

def fft_conv_full(f):
    n = len(f)
    size = 1
    while size < 2*n - 1:
        size <<= 1
    F = np.fft.rfft(f, size)
    return np.fft.irfft(F*F, size)[:2*n-1]

def score(f):
    n = len(f)
    h = fft_conv_full(f)
    M = np.max(h)
    S = np.sum(f)
    if S < 0.01:
        return float('inf')
    return 2.0*n*M/(S*S)

def normalize(f):
    f = np.maximum(f, 0.0)
    s = f.sum()
    if s < 1e-30:
        return None
    return f/s

def grad_active(f, thresh_frac):
    """Gradient using active set (fraction of max)."""
    n = len(f)
    h = fft_conv_full(f)
    M = np.max(h)
    if M <= 0:
        return np.zeros(n)
    thresh = thresh_frac * M
    active = np.where(h >= thresh)[0]
    if len(active) == 0:
        active = np.array([np.argmax(h)])
    
    grad = np.zeros(n)
    for k in active:
        j_lo = max(0, k-n+1)
        j_hi = min(n-1, k)
        if j_hi >= j_lo:
            # 2*f(k-j) for j in [j_lo, j_hi]
            # f(k-j) = f[k-j_lo : k-j_hi+1][::-1]
            vals = f[k-j_hi : k-j_lo+1][::-1]  # This gives f[k-j_hi], f[k-j_hi+1], ..., f[k-j_lo]
            # We want grad[j] += 2*f[k-j]
            # For j=j_lo: f[k-j_lo]
            # For j=j_hi: f[k-j_hi]
            # So grad[j_lo:j_hi+1] += 2 * f[k-j_hi : k-j_lo+1] reversed
            # f[k-j_hi], f[k-j_hi+1], ..., f[k-j_lo]
            # Index into f: k-j_hi, k-j_hi+1, ..., k-j_lo
            # That's f[k-j_hi : k-j_lo+1] (Python slice)
            grad[j_lo:j_hi+1] += 2.0 * f[k-j_hi : k-j_lo+1]
    grad /= len(active)
    return grad

def mirror_descent_anneal(f_init, n_iter, eta_scale, thresh_start, thresh_end, label=""):
    """Mirror descent with temperature annealing on threshold fraction."""
    f = np.asarray(f_init, dtype=np.float64).copy()
    f = normalize(f)
    if f is None:
        return f_init.copy(), float('inf')
    n = len(f)
    best_f = f.copy()
    best_score = score(f)
    
    # Backtracking: track consecutive failures
    consec_fail = 0
    eta = eta_scale
    
    for it in range(n_iter):
        # Anneal threshold
        progress = it / max(1, n_iter - 1)
        thresh = thresh_start + (thresh_end - thresh_start) * progress
        
        h = fft_conv_full(f)
        M = np.max(h)
        
        grad = grad_active(f, thresh)
        
        # Mirror descent update
        log_ratio = -eta * grad
        log_ratio = np.clip(log_ratio, -20.0, 20.0)
        f_new = f * np.exp(log_ratio)
        f_norm = normalize(f_new)
        if f_norm is None:
            f_new = np.ones(n) / n
        else:
            f_new = f_norm
        
        new_score = score(f_new)
        
        if new_score <= best_score * 1.0001:  # Allow tiny increase
            if new_score < best_score:
                best_score = new_score
                best_f = f_new.copy()
            f = f_new
            consec_fail = 0
            # If improved a lot, try increasing eta
            if best_score < score(f_init) * 0.99:
                eta = min(eta * 1.01, eta_scale * 3.0)
        else:
            f = f_new  # Still update (for diversity)
            consec_fail += 1
            if consec_fail >= 10:
                eta *= 0.5
                consec_fail = 0
        
        if (it+1) % 2000 == 0:
            print(f"    MD anneal {it+1}/{n_iter}: best={best_score:.6f}, M={M:.6f}, eta={eta:.4f}, thresh={thresh:.3f}")
    
    return best_f, best_score

def lp_frank_wolfe(f_init, max_iters, time_budget):
    """LP-based Frank-Wolfe for the normalized problem."""
    from scipy.optimize import linprog
    from scipy.sparse import coo_matrix
    
    f = np.asarray(f_init, dtype=np.float64).copy()
    f = normalize(f)
    if f is None:
        return f_init.copy(), float('inf')
    n = len(f)
    best_f = f.copy()
    best_score = score(f)
    start_time = time.monotonic()
    
    print(f"    LP-FW start: n={n}, score={best_score:.6f}")
    
    for it in range(max_iters):
        if time.monotonic() - start_time > time_budget:
            break
        
        h = np.convolve(f, f)
        M = float(np.max(h))
        
        # Build sparse constraint matrix
        row_idx = []
        col_idx = []
        vals = []
        for j in range(n):
            rows_j = np.arange(j, j+n)
            row_idx.extend(rows_j.tolist())
            col_idx.extend([j]*n)
            vals.extend(f.tolist())
        
        C = coo_matrix((vals, (row_idx, col_idx)), shape=(2*n-1, n))
        c_obj = -np.ones(n)
        
        res = linprog(c_obj, A_ub=C, b_ub=np.full(2*n-1, M),
                      bounds=[(0, None)]*n, method='highs')
        
        if not res.success or res.x is None:
            break
        
        g = np.maximum(res.x, 0)
        Sg = g.sum()
        if Sg < 1e-10:
            break
        g /= Sg
        
        t = 2.0 / (it + 2)
        f_new = (1.0-t)*f + t*g
        f_n = normalize(f_new)
        if f_n is None:
            break
        f = f_n
        
        s = score(f)
        if s < best_score:
            best_score = s
            best_f = f.copy()
        
        if (it+1) % 5 == 0:
            print(f"    LP-FW it={it+1}: score={best_score:.6f}")
    
    return best_f, best_score

    j = np.arange(n)
    f = np.cos(np.pi*(j+0.5)/n)**2
    f = np.maximum(f, 0)
    return normalize(f)

    j = np.arange(n) - (n-1)/2.0

def upsample(parent, target_n):
    """Upsample parent to target_n using linear interpolation."""
    n = len(parent)
    if n == target_n:
        return parent.astype(np.float64).copy()
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, target_n)
    new = np.interp(x_new, x_old, parent)
    return normalize(new)

def downsample(parent, target_n):
    """Downsample parent to target_n."""
    n = len(parent)
    if n == target_n:
        return parent.astype(np.float64).copy()

    x_new = np.linspace(0, 1, target_n)
    new = np.interp(x_new, x_old, parent)
    return normalize(new)

So to make h flat, we want |F|^2 to be concentrated at t=0, which means F is concentrated at t=0, which means f is spread out (close to uniform).

But wait, if f is uniform (all 1/n), then F(0) = 1, F(t) = 0 for t != 0. So |F|^2 = delta_{t,0}. Then h(k) should be constant = 1/(n^2) * ... hmm let me just compute directly.

For uniform f(j) = 1/n:
h(k) = sum_j f(j) f(k-j) = sum_{j=max(0,k-n+1)}^{min(n-1,k)} (1/n)(1/n) = (n-|k|)/n^2 for |k|<n.

So h(0) = n/n^2 = 1/n, h(1) = (n-1)/n^2, ..., h(n-1) = 1/n^2.
This is a triangle, NOT flat.

So my DFT analysis was wrong somewhere. Let me redo it.

The full linear convolution h = f * f has length 2n-1. Its DFT (of length 2n-1, or more) is H = F * F where F is the DFT of f (padded to length 2n-1).

For uniform f of length n (padded with n-1 zeros to length 2n-1):
F(t) = (1/n) * sum_{j=0}^{n-1} exp(-2pi i j t/(2n-1))

This is a Dirichlet kernel, not a delta. So H(t) = |F(t)|^2 is not a delta either.

I was confusing the circular convolution with the linear convolution. For circular convolution (cyclic), the DFT of the circular autocorrelation is |F|^2. For linear convolution, it's more complex because of the zero-padding.

"""
Optimizes non-negative step function heights to minimize 2n*max(conv(f,f))/(sum(f))^2.
Uses temperature-annealed mirror descent with adaptive step size, multi-resolution search,
LP Frank-Wolfe refinement, and spectral diversification to escape local minima.
"""

import numpy as np
import time


def fft_conv_full(f):
    """Compute full linear convolution of f with itself using FFT."""
    n = len(f)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    F = np.fft.rfft(f, size)
    return np.fft.irfft(F * F, size)[:2 * n - 1]


def score_raw(f):
    """Fast score computation assuming f >= 0, sum(f) > 0."""
    n = len(f)
    h = fft_conv_full(f)
    M = float(np.max(h))
    S = float(np.sum(f))
    if S < 0.01:
        return float('inf')
    return 2.0 * n * M / (S * S)


def normalize(f):
    """Normalize f to sum to 1, clip negatives. Returns None if degenerate."""
    f = np.maximum(f, 0.0)
    s = float(np.sum(f))
    if s < 1e-30:
        return None
    return f / s


def grad_active_set(f, thresh_frac):
    """
    Compute (sub)gradient of max_k (f*f)(k) using active set.
    Uses average gradient over all k where h(k) >= thresh_frac * max(h).
    """
    n = len(f)
    h = fft_conv_full(f)
    M = float(np.max(h))
    if M <= 0:
        return np.zeros(n)
    thresh = thresh_frac * M
    active = np.where(h >= thresh)[0]
    if len(active) == 0:
        active = np.array([int(np.argmax(h))])

    grad = np.zeros(n, dtype=np.float64)
    for k in active:
        j_lo = max(0, k - n + 1)
        j_hi = min(n - 1, k)
        if j_hi >= j_lo:
            # grad[j] += 2*f[k-j] for j in [j_lo, j_hi]
            # f[k-j] for j=j_lo..j_hi is f[k-j_lo], f[k-j_lo-1], ..., f[k-j_hi]
            # i.e., f[k-j_hi : k-j_lo+1]
            grad[j_lo:j_hi + 1] += 2.0 * f[k - j_hi:k - j_lo + 1]
    grad /= float(len(active))
    return grad


def mirror_descent(f_init, n_iter, eta_scale, thresh_start, thresh_end,
                   step_anneal=True, label=""):
    """
    Mirror descent with multiplicative updates, threshold annealing, and adaptive step size.
    """
    f = np.asarray(f_init, dtype=np.float64).copy()
    f = normalize(f)
    if f is None:
        return f_init.copy(), float('inf')
    n = len(f)
    init_score = score_raw(f)
    best_f = f.copy()
    best_score = init_score

    eta = eta_scale
    consec_fail = 0
    patience = 0

    for it in range(n_iter):
        progress = it / max(1, n_iter - 1)
        if step_anneal:
            thresh = thresh_start + (thresh_end - thresh_start) * (progress ** 0.7)
        else:
            thresh = thresh_end

        h = fft_conv_full(f)
        M = float(np.max(h))
        if M <= 0:
            break

        grad = grad_active_set(f, thresh)

        # Adaptive: scale eta by 1/M to normalize gradient magnitude
        eta_eff = eta / (M + 1e-20)

        log_ratio = -eta_eff * grad
        log_ratio = np.clip(log_ratio, -25.0, 25.0)

        f_new = f * np.exp(log_ratio)
        f_norm = normalize(f_new)
        if f_norm is None:
            f_new = np.ones(n) / n
        else:
            f_new = f_norm

        new_score = score_raw(f_new)

        if new_score <= best_score:
            if new_score < best_score:
                best_score = new_score
                best_f = f_new.copy()
                patience = 0
            f = f_new
            consec_fail = 0
            # Slightly increase eta on improvement
            eta = min(eta * 1.005, eta_scale * 4.0)
        else:
            f = f_new
            consec_fail += 1
            patience += 1
            if consec_fail >= 5:
                eta *= 0.5
                consec_fail = 0
            if patience > n_iter * 0.4:
                break

        if (it + 1) % 5000 == 0:
            print(f"      MD {it+1}/{n_iter}: best={best_score:.6f}, M={M:.8f}, eta={eta:.4f}, thresh={thresh:.3f}")

    return best_f, best_score


def lp_frank_wolfe(f_init, max_iters, time_budget):
    """LP-based Frank-Wolfe for the normalized problem."""
    from scipy.optimize import linprog
    from scipy.sparse import coo_matrix

    f = np.asarray(f_init, dtype=np.float64).copy()
    f = normalize(f)
    if f is None:
        return f_init.copy(), float('inf')
    n = len(f)
    best_f = f.copy()
    best_score = score_raw(f)
    start_time = time.monotonic()

    print(f"    LP-FW start: n={n}, score={best_score:.6f}, budget={time_budget:.0f}s")

    for it in range(max_iters):
        if time.monotonic() - start_time > time_budget:
            break

        h = np.convolve(f, f)
        M = float(np.max(h))

        # Build sparse constraint matrix: C[k, j] = f[k-j] if 0<=k-j<n
        row_idx = []
        col_idx = []
        vals = []
        for j in range(n):
            rows_j = np.arange(j, j + n)
            row_idx.extend(rows_j.tolist())
            col_idx.extend([j] * n)
            vals.extend(f.tolist())

        C = coo_matrix((vals, (row_idx, col_idx)), shape=(2 * n - 1, n))
        c_obj = -np.ones(n)

        res = linprog(c_obj, A_ub=C, b_ub=np.full(2 * n - 1, M),
                      bounds=[(0, None)] * n, method='highs')

        if not res.success or res.x is None:
            print(f"    LP failed at iter {it}: {res.message}")
            break

        g = np.maximum(res.x, 0.0)
        Sg = float(np.sum(g))
        if Sg < 1e-10:
            break
        g = g / Sg

        # Frank-Wolfe step size
        t = 2.0 / (it + 2)
        f_new = (1.0 - t) * f + t * g
        f_n = normalize(f_new)
        if f_n is None:
            break
        f = f_n

        s = score_raw(f)
        if s < best_score:
            best_score = s
            best_f = f.copy()

        if (it + 1) % 5 == 0:
            print(f"    LP-FW it={it+1}/{max_iters}: score={best_score:.6f}")

    return best_f, best_score


def init_cosine(n):
    """Cosine-squared initialization."""
    j = np.arange(n, dtype=np.float64)
    f = np.cos(np.pi * (j + 0.5) / float(n)) ** 2
    f = np.maximum(f, 0.0)
    return normalize(f)


def init_gaussian(n, sigma_frac):
    """Gaussian window initialization."""
    j = np.arange(n, dtype=np.float64) - (n - 1) / 2.0
    sigma = sigma_frac * float(n)
    f = np.exp(-j * j / (2.0 * sigma * sigma))
    return normalize(f)


def init_tapered(n, power):
    """Sinusoidal taper initialization."""
    j = np.arange(n, dtype=np.float64)
    x = np.sin(np.pi * (j + 0.5) / float(n))
    f = x ** power
    f = np.maximum(f, 0.0)
    return normalize(f)


def init_uniform(n):
    return np.ones(n, dtype=np.float64) / n


def init_random(n, rng):
    f = rng.random(n, dtype=np.float64)
    return normalize(f)


def resample(f, target_n):
    """Resample sequence f to target_n using linear interpolation."""
    n = len(f)
    if n == target_n:
        return f.astype(np.float64).copy()
    x_old = np.linspace(0.0, 1.0, n)
    x_new = np.linspace(0.0, 1.0, target_n)
    new = np.interp(x_new, x_old, f)
    return normalize(new)


def spectral_diversify(f, rng, n_harmonics=5):
    """
    Create a diversified version by modifying the spectrum.
    Add random phase perturbations to the DFT, then take IFT.
    """
    n = len(f)
    size = 1
    while size < n:
        size <<= 1
    F = np.fft.rfft(f, size)
    # Add small random phase perturbations
    phases = rng.random(F.shape) * 0.3
    F_perturbed = F * np.exp(1j * phases)
    f_new = np.fft.irfft(F_perturbed, size)[:n]
    return normalize(f_new)
