import numpy as np
import time
from scipy.optimize import linprog

def build_conv_matrix(f):
    """Build matrix A where (A @ g)[k] = convolve(f, g)[k]."""
    n = len(f)
    # For each k in 0..2n-2, and each j in 0..n-1:
    # (convolve(f,g))[k] = sum_{j: 0<=j<n, 0<=k-j<n} f[k-j] * g[j]
    # So A[k, j] = f[k-j] if 0<=k-j<n, else 0
    A = np.zeros((2*n-1, n))
    for k in range(2*n-1):
        for j in range(max(0, k-(n-1)), min(k+1, n)):
            A[k, j] = f[k-j]
    return A

def solve_lp(f, M, tol=1e-10):
    """Solve LP: max sum(g) s.t. conv(f,g)[k] <= (M - conv(f,f)[k])/2, g >= 0."""
    n = len(f)
    conv_ff = np.convolve(f, f)
    B = (M - conv_ff) / 2
    # If any B[k] < 0, the constraint is infeasible (but B >= 0 since M >= conv_ff[k])
    B = np.maximum(B, 0)
    
    A = build_conv_matrix(f)
    c = -np.ones(n)  # minimize -sum(g) = maximize sum(g)
    bounds = [(0, None)] * n
    
    result = linprog(c, A_ub=A, b_ub=B, bounds=bounds, method='highs', 
                     options={'time_limit': 30, 'presolve': True})
    if result.success:
        return result.x
    return None
