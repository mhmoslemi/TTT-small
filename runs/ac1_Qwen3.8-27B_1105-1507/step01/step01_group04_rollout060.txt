```python
"""
Minimizes 2n*max(conv(f,f))/(sum f)^2 over non-negative sequences.
Uses accelerated mirror descent with FFT-based subgradients, Nesterov momentum,
diverse initializations, and coordinate descent refinement.
"""

import numpy as np
import time


def _fft_conv(f):
    """Full linear convolution f*f via FFT."""
    n = len(f)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    F = np.fft.rfft(f, size)
    return np.fft.irfft(F * F, size)[:2*n-1]


def _score(f):
    """Compute 2n * max(conv(f,f)) / (sum f)^2."""
    n = len(f)
    h = _fft_conv(f)
    M = float(h.max())
    S = float(f.sum())
    if S < 0.01:
        return float('inf')
    return 2.0 * n * M / (S * S)


def _norm(f):
    """Clip non-negative and normalize to sum 1."""
    f = np.maximum(f, 0.0)
    s = f.sum()
    if s < 1e-30:
        return None
    return f / s


def _subgrad(f, active):
    """FFT-based subgradient of max_k (f*f)_k averaged over active set."""
    n = len(f)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    mask = np.zeros(size)
    mask[active] = 1.0
    F = np.fft.rfft(f, size)
    M = np.fft.rfft(mask, size)
    corr = np.fft.irfft(M * np.conj(F), size)
    return (2.0 / len(active)) * corr[:n]


def _md(f0, n_iter, eta0, decay, mu, af, reseed=0, seed=0, af_progressive=False):
    """
    Accelerated mirror descent.
    f0: initial (normalized)
    eta0: initial step size
    decay: step size decay (eta_t = eta0 / (1 + t*decay))
    mu: momentum (0-1)
    af: active fraction threshold
    reseed: reseed interval (0 = off)
    af_progressive: if True, start with af=0.5 and progress to af
    """
    rng = np.random.default_rng(seed)
    f = _norm(f0.astype(np.float64).copy())
    if f is None:
        return f0.copy(), float('inf')
    n = len(f)
    lf = np.log(f + 1e-30)
    v = np.zeros(n)
    bf = f.copy()
    bs = _score(f)

    for i in range(n_iter):
        # Progressive active fraction
        if af_progressive and i < n_iter // 4:
            af_cur = 0.5 + 0.5 * af * (4.0 * i / n_iter)
        else:
            af_cur = af

        h = _fft_conv(f)
        M = float(h.max())
        thr = af_cur * M
        act = np.where(h >= thr)[0]
        if act.size == 0:
            act = np.array([int(np.argmax(h))])

        g = _subgrad(f, act)
        eta = eta0 / (1.0 + i * decay)

        if mu > 0:
            v = mu * v - eta * g
            lf = lf + v
        else:
            lf = lf - eta * g

        # Numerical stability
        lf_max = lf.max()
        if lf_max > 50:
            lf = lf - lf_max
        f_new = np.exp(lf)
        f = _norm(f_new)
        if f is None:
            f = bf.copy()
            lf = np.log(f + 1e-30)
            v = np.zeros(n)
            continue

        # Periodic reseed to escape local minima
        if reseed > 0 and (i + 1) % reseed == 0:
            noise_scale = 0.003 * (1.0 + i / n_iter)
            fp = _norm(f + noise_scale * rng.random(n))
            if fp is not None:
                f = fp
                lf = np.log(f + 1e-30)
                v = np.zeros(n)

        s = _score(f)
        if s < bs:
            bs = s
            bf = f.copy()

        if (i + 1) % 20000 == 0:
            print(f"        MD i={i+1} score={bs:.6f} M={M:.8f} |K|={act.size}")

    return bf, bs


def _cd(f0, n_sweeps, seed=0):
    """Coordinate descent: random coordinate, try multipliers."""
    rng = np.random.default_rng(seed)
    f = _norm(f0.astype(np.float64).copy())
    if f is None:
        return f0.copy(), float('inf')
    n = len(f)
    bf = f.copy()
    bs = _score(f)
    multipliers = [0.2, 0.5, 0.75, 1.33, 1.5, 2.0, 4.0]

    for sweep in range(n_sweeps):
        for _ in range(min(n, 500)):  # visit at most 500 coords per sweep
            j = rng.integers(0, n)
            if f[j] < 1e-15:
                continue
            best_m = 1.0
            best_s = bs
            for m in multipliers:
                f_trial = f.copy()
                f_trial[j] *= m
                f_trial = _norm(f_trial)
                if f_trial is None:
                    continue
                s = _score(f_trial)
                if s < best_s:
                    best_s = s
                    best_m = m
            if best_m != 1.0:
                f[j] *= best_m
                f = _norm(f)
                if f is not None and best_s < bs:
                    bs = best_s
                    bf = f.copy()

        if (sweep + 1) % 5 == 0:
            print(f"      CD sweep={sweep+1} score={bs:.6f}")

    return bf, bs


def _init_cosine(n):
    j = np.arange(n, dtype=np.float64)
    f = np.cos(np.pi * (j + 0.5) / float(n)) ** 2
    return _norm(f)


def _init_gauss(n, sigma_frac):
    j = np.arange(n, dtype=np.float64) - (n - 1) / 2.0
    sigma = sigma_frac * n
    f = np.exp(-j * j / (2.0 * sigma * sigma))
    return _norm(f)


def _init_rand(n, seed):
    rng = np.random.default_rng(seed)
    return _norm(rng.random(n).astype(np.float64))


def _init_taper(n, power):
    """Raised cosine to a power."""
    j = np.arange(n, dtype=np.float64)
    x = np.sin(np.pi * (j + 0.5) / float(n))
    f = x ** power
    return _norm(f)


def _init_ripple(n, freq, seed):
    """Random-phase cosine with smoothing."""
    rng = np.random.default_rng(seed)
    j = np.arange(n, dtype=np.float64)
    phase = rng.uniform(0, 2 * np.pi)
    f = np.abs(np.cos(2 * np.pi * freq * j / n + phase)) ** 2
    return _norm(f)
```

```python
"""
Minimizes 2n*max(conv(f,f))/(sum f)^2 over non-negative sequences f.
Strategy: Accelerated mirror descent with FFT-based subgradients, Nesterov momentum,
diverse initializations at multiple resolutions, progressive active-set tightening,
and coordinate descent refinement for local polish.
"""

import numpy as np
import time


def _fft_conv(f):
    """Full linear convolution f*f via FFT."""
    n = len(f)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    F = np.fft.rfft(f, size)
    return np.fft.irfft(F * F, size)[:2 * n - 1]


def _score_normalized(f):
    """Score assuming sum(f)=1: returns 2n * max(conv(f,f))."""
    n = len(f)
    h = _fft_conv(f)
    return 2.0 * n * float(h.max())


def _norm(f):
    """Clip non-negative, normalize to sum=1. Returns None if degenerate."""
    f = np.maximum(f, 0.0)
    s = f.sum()
    if s < 1e-30:
        return None
    return f / s


def _subgrad(f, active):
    """FFT-based subgradient: g[j] = (2/|K|) * sum_{k in K} f[k-j]."""
    n = len(f)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    mask = np.zeros(size)
    mask[active] = 1.0
    F = np.fft.rfft(f, size)
    M = np.fft.rfft(mask, size)
    corr = np.fft.irfft(M * np.conj(F), size)
    return (2.0 / float(len(active))) * corr[:n]


def _md(f0, n_iter, eta0, decay, mu, af, reseed=0, seed=0, af_prog=False, time_limit=0.0):
    """
    Accelerated mirror descent (multiplicative weights) with Nesterov momentum.
    Operates on normalized f (sum=1). Score = 2n * max(f*f).
    """
    rng = np.random.default_rng(seed)
    f = _norm(f0.astype(np.float64).copy())
    if f is None:
        return f0.copy(), float('inf')
    n = len(f)
    lf = np.log(f + 1e-30)
    v = np.zeros(n)
    bf = f.copy()
    h = _fft_conv(f)
    bs = 2.0 * n * float(h.max())
    start = time.monotonic()

    for i in range(n_iter):
        if time_limit > 0 and (time.monotonic() - start) > time_limit:
            break

        # Progressive active fraction: start wide, tighten over first 25%
        if af_prog and i < max(1, n_iter // 4):
            frac = 4.0 * i / max(1, n_iter // 4)
            af_cur = 0.4 + 0.6 * (af - 0.4) * frac
        else:
            af_cur = af

        M = float(h.max())
        thr = af_cur * M
        act = np.where(h >= thr)[0]
        if act.size == 0:
            act = np.array([int(np.argmax(h))])

        g = _subgrad(f, act)
        eta = eta0 / (1.0 + i * decay)

        if mu > 0:
            v = mu * v - eta * g
            lf = lf + v
        else:
            lf = lf - eta * g

        # Numerical stability
        lfm = lf.max()
        if lfm > 50.0 or lfm < -50.0:
            lf = lf - lfm
        f_new = np.exp(lf)
        f = _norm(f_new)
        if f is None:
            f = bf.copy()
            lf = np.log(f + 1e-30)
            v = np.zeros(n)
            h = _fft_conv(f)
            continue

        # Periodic reseed to escape local minima
        if reseed > 0 and (i + 1) % reseed == 0:
            ns = 0.002 + 0.004 * min(1.0, i / max(1, n_iter))
            fp = _norm(f + ns * rng.random(n))
            if fp is not None:
                f = fp
                lf = np.log(f + 1e-30)
                v = np.zeros(n)

        h = _fft_conv(f)
        s_cur = 2.0 * n * float(h.max())
        if s_cur < bs:
            bs = s_cur
            bf = f.copy()

        if (i + 1) % 50000 == 0:
            el = time.monotonic() - start
            print(f"        MD i={i+1} score={bs:.6f} el={el:.1f}s")

    return bf, bs


def _cd(f0, n_sweeps, seed=0):
    """
    Coordinate descent: pick random coordinate, try multipliers, keep best.
    Operates on normalized f (sum=1).
    """
    rng = np.random.default_rng(seed)
    f = _norm(f0.astype(np.float64).copy())
    if f is None:
        return f0.copy(), float('inf')
    n = len(f)
    bf = f.copy()
    bs = _score_normalized(f)
    multipliers = [0.1, 0.3, 0.5, 0.7, 1.25, 1.5, 2.0, 3.0]

    for sweep in range(n_sweeps):
        n_trial = min(n, 800)
        for _ in range(n_trial):
            j = int(rng.integers(0, n))
            if f[j] < 1e-18:
                continue
            best_m = 1.0
            best_s = bs
            for m in multipliers:
                f[j] *= m
                f2 = _norm(f)
                if f2 is None:
                    f[j] /= m
                    continue
                s = _score_normalized(f2)
                if s < best_s:
                    best_s = s
                    best_m = m
                f[j] /= m  # restore
            if best_m != 1.0:
                f[j] *= best_m
                f = _norm(f)
                if f is not None and best_s < bs:
                    bs = best_s
                    bf = f.copy()

        if (sweep + 1) % 3 == 0:
            print(f"      CD sweep={sweep+1}/{n_sweeps} score={bs:.6f}")

    return bf, bs


def _init_cosine(n):
    j = np.arange(n, dtype=np.float64)
    f = np.cos(np.pi * (j + 0.5) / float(n)) ** 2
    return _norm(f)


def _init_gauss(n, sigma_frac):
    j = np.arange(n, dtype=np.float64) - (n - 1) / 2.0
    sigma = sigma_frac * float(n)
    f = np.exp(-j * j / (2.0 * sigma * sigma))
    return _norm(f)


def _init_rand(n, seed):
    rng = np.random.default_rng(seed)
    return _norm(rng.random(n, dtype=np.float64))


def _init_taper(n, power):
    j = np.arange(n, dtype=np.float64)
    x = np.sin(np.pi * (j + 0.5) / float(n))
    f = np.maximum(x, 0.0) ** power
    return _norm(f)


def _init_ripple(n, n_ripples, seed):
    """Superposition of cosine ripples with random phases."""
    rng = np.random.default_rng(seed)
    j = np.arange(n, dtype=np.float64)
    f = np.ones(n, dtype=np.float64)
    for k in range(1, n_ripples + 1):
        phase = rng.uniform(0, 2 * np.pi)
        f += 0.5 * np.cos(2.0 * np.pi * k * j / n + phase)
    f = np.maximum(f, 0.0)
    return _norm(f)
```
