        if j_hi >= j_lo:

def _pressure_topk(f, h, frac=0.8):
    """Compute pressure using lags where h >= frac * max(h)."""
    n = len(f)
    M = float(np.max(h))
    threshold = frac * M + 1e-15
    active = np.where(h >= threshold)[0]
    if len(active) == 0:
        active = np.array([int(np.argmax(h))])
    
    p = np.zeros(n)
    w_avg = 1.0 / len(active)
    for k in active:
        j_lo = max(0, k - n + 1)
        j_hi = min(n - 1, k)
        if j_hi >= j_lo:
            p[j_lo:j_hi+1] += w_avg * 2.0 * f[k - j_hi:k - j_lo + 1][::-1]
    
    return p

def _cosine_init(n):
    j = np.arange(n, dtype=np.float64)
    f = np.cos(np.pi * (j + 0.5) / n) ** 2
    f = np.maximum(f, 0.0)
    s = np.sum(f)
    if s > 0:
        f /= s
    return f

def _uniform_init(n):
    return np.ones(n, dtype=np.float64) / n

def _gaussian_init(n, sigma_frac=0.3):
    j = np.arange(n, dtype=np.float64) - (n - 1) / 2.0
    sigma = sigma_frac * n
    f = np.exp(-j * j / (2.0 * sigma * sigma))
    s = np.sum(f)
    if s > 0:
        f /= s
    return f

def _tapered_init(n, power=2.0):
    j = np.arange(n, dtype=np.float64)
    x = np.sin(np.pi * (j + 0.5) / n)
    f = x ** power
    f = np.maximum(f, 0.0)
    s = np.sum(f)
    if s > 0:
        f /= s
    return f

def _upsample(f, factor=2):
    """Upsample by factor using linear interpolation."""
    n = len(f)
    n_new = n * factor
    old_idx = np.arange(n)
    new_idx = np.linspace(0, n - 1, n_new)
    f_new = np.interp(new_idx, old_idx, f)
    s = np.sum(f_new)
    if s > 0:
        f_new /= s
    return f_new

def _perturb(f, rng, frac=0.05):
    """Add multiplicative perturbation."""
    n = len(f)
    noise = rng.random(n) * frac
    f_p = f * (1.0 + noise)
    return _normalize(f_p)

def _perturb_additive(f, rng, frac=0.02):
    """Add additive perturbation."""
    n = len(f)
    noise = rng.random(n) * frac
    f_p = f + noise
    return _normalize(f_p)

"""
Optimizes non-negative step function heights to minimize 2n*max(conv(f,f))/(sum(f))^2.
Uses a combined two-objective strategy: alternating between peak (max) minimization
and energy (variance) minimization of the autocorrelation, with adaptive step size,

import numpy as np
import time


def _fft_convolve_square(f):
    """Compute full linear convolution f*f using FFT."""
    n = len(f)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    F = np.fft.rfft(f, size)
    h = np.fft.irfft(F * F, size)
    return h[:2 * n - 1]


def _score_normalized(f, h=None):
    """Score assuming sum(f)=1."""
    n = len(f)
    if h is None:
        h = _fft_convolve_square(f)
    M = float(np.max(h))
    return 2.0 * n * M


def _normalize(f):
    f = np.maximum(f, 0.0)
    s = float(np.sum(f))
    if s < 1e-30:
        return None
    return f / s


def _pressure_max(f, h, top_k=200, temperature=0.0):
    """Compute subgradient of max(h) using top-k lags with optional soft weighting."""
    n = len(f)
    h_len = len(h)
    k_eff = min(top_k, h_len)

    if k_eff == h_len:
        part = np.arange(h_len)
    else:
        part = np.argpartition(h, -k_eff)[-k_eff:]

    if temperature > 1e-10:
        h_top = h[part] - np.max(h[part])
        w_top = np.exp(h_top / temperature)
        w_sum = w_top.sum()
        w = np.zeros(h_len)
        if w_sum > 1e-30:
            w[part] = w_top / w_sum
        else:
            w[part[0]] = 1.0
    else:
        w = np.zeros(h_len)
        w[part] = 1.0 / k_eff

    non_zero = np.where(w > 1e-14)[0]
    p = np.zeros(n)
    for k in non_zero:
        wk = w[k]
        j_lo = max(0, k - n + 1)
        j_hi = min(n - 1, k)
        if j_hi >= j_lo:
            p[j_lo:j_hi + 1] += wk * 2.0 * f[k - j_hi:k - j_lo + 1][::-1]
    return p


def _pressure_energy(f, h):
    """
    Compute gradient of sum(h_k^2) w.r.t. f.
    grad[i] = 4 * sum_k h[k] * f[k-i] (cross-correlation of h and f).
    """
    n = len(f)
    size = 1
    while size < 3 * n - 1:
        size <<= 1
    H = np.fft.fft(h, size)
    F = np.fft.fft(f, size)
    c = np.fft.ifft(H * np.conj(F)).real
    return 4.0 * c[:n]


def _mirror_step(f, pressure, step):
    """Apply multiplicative weight update."""
    log_ratio = -step * pressure
    log_ratio = np.clip(log_ratio, -15.0, 15.0)
    f_new = f * np.exp(log_ratio)
    return _normalize(f_new)


def _optimize_peak(f_init, time_budget, top_k=300, temperature=0.0,
                   step_scale=1.0, decay=0.9995, min_step=0.001):
    """Mirror descent minimizing max(conv(f,f)) with sum(f)=1."""
    f = _normalize(np.asarray(f_init, dtype=np.float64))
    if f is None:
        return np.ones(len(f_init)) / len(f_init), float('inf')

    h = _fft_convolve_square(f)
    best_f = f.copy()
    best_score = _score_normalized(f, h)
    start = time.monotonic()
    step = step_scale
    stall = 0
    patience = 100

    while time.monotonic() - start < time_budget:
        h = _fft_convolve_square(f)
        p = _pressure_max(f, h, top_k, temperature)
        p_max = float(np.max(p))
        if p_max < 1e-15:
            break

        s = step / p_max
        s = float(np.clip(s, min_step, 2.0))
        f_new = _mirror_step(f, p, s)
        if f_new is None:
            break
        f = f_new

        h = _fft_convolve_square(f)
        score = _score_normalized(f, h)
        if score < best_score - 1e-9:
            best_score = score
            best_f = f.copy()
            stall = 0
        else:
            stall += 1
            if stall >= patience:
                step *= 0.7
                stall = 0
                if step < min_step * 0.1:
                    break
        step *= decay

    return best_f, best_score


def _optimize_energy(f_init, time_budget, step_scale=0.5, decay=0.999, min_step=0.001):
    """Mirror descent minimizing sum(h_k^2) i.e. energy of autocorrelation."""
    f = _normalize(np.asarray(f_init, dtype=np.float64))
    if f is None:
        return np.ones(len(f_init)) / len(f_init), float('inf')

    h = _fft_convolve_square(f)
    best_f = f.copy()
    best_energy = float(np.sum(h * h))
    best_score = _score_normalized(f, h)
    start = time.monotonic()
    step = step_scale
    stall = 0

    while time.monotonic() - start < time_budget:
        h = _fft_convolve_square(f)
        p = _pressure_energy(f, h)
        p_max = float(np.max(p))
        if p_max < 1e-15:
            break

        s = step / p_max
        s = float(np.clip(s, min_step, 1.0))
        f_new = _mirror_step(f, p, s)
        if f_new is None:
            break
        f = f_new

        h = _fft_convolve_square(f)
        energy = float(np.sum(h * h))
        score = _score_normalized(f, h)
        if energy < best_energy - 1e-12 or score < best_score - 1e-9:
            best_energy = energy
            best_score = score
            best_f = f.copy()
            stall = 0
        else:
            stall += 1
            if stall >= 150:
                step *= 0.6
                stall = 0
                if step < min_step * 0.1:
                    break
        step *= decay

    return best_f, best_score


def _optimize_combined(f_init, time_budget, top_k=300, temperature=0.0,
                       peak_frac=0.6, step_scale=1.0):
    """Alternate between peak and energy minimization."""
    f = _normalize(np.asarray(f_init, dtype=np.float64))
    if f is None:
        return np.ones(len(f_init)) / len(f_init), float('inf')

    h = _fft_convolve_square(f)
    best_f = f.copy()
    best_score = _score_normalized(f, h)
    start = time.monotonic()
    step_peak = step_scale
    step_energy = step_scale * 0.5
    stall = 0

    while time.monotonic() - start < time_budget:
        phase = (time.monotonic() - start) / (time_budget * peak_frac)
        use_peak = (phase % 2.0) < 1.0

        h = _fft_convolve_square(f)
        if use_peak:
            p = _pressure_max(f, h, top_k, temperature)
            p_max = float(np.max(p))
            s = step_peak / p_max if p_max > 1e-15 else 0.01
            s = float(np.clip(s, 0.001, 1.5))
        else:
            p = _pressure_energy(f, h)
            p_max = float(np.max(p))
            s = step_energy / p_max if p_max > 1e-15 else 0.01
            s = float(np.clip(s, 0.001, 1.0))

        f_new = _mirror_step(f, p, s)
        if f_new is None:
            break
        f = f_new

        h = _fft_convolve_square(f)
        score = _score_normalized(f, h)
        if score < best_score - 1e-9:
            best_score = score
            best_f = f.copy()
            stall = 0
        else:
            stall += 1
            if stall >= 80:
                step_peak *= 0.8
                step_energy *= 0.8
                stall = 0
                if step_peak < 0.0005:
                    break
        step_peak *= 0.9997
        step_energy *= 0.9997

    return best_f, best_score


def _cosine_init(n):
    j = np.arange(n, dtype=np.float64)
    f = np.cos(np.pi * (j + 0.5) / n) ** 2
    f = np.maximum(f, 0.0)
    s = np.sum(f)
    return f / s if s > 0 else np.ones(n) / n


def _gaussian_init(n, sigma_frac=0.25):
    j = np.arange(n, dtype=np.float64) - (n - 1) / 2.0
    sigma = sigma_frac * n
    f = np.exp(-j * j / (2.0 * sigma * sigma))
    s = np.sum(f)
    return f / s if s > 0 else np.ones(n) / n


def _tapered_init(n, power=2.0):
    j = np.arange(n, dtype=np.float64)
    x = np.sin(np.pi * (j + 0.5) / n)
    f = x ** power
    f = np.maximum(f, 0.0)
    s = np.sum(f)
    return f / s if s > 0 else np.ones(n) / n


def _boxcar_tapered_init(n, inner_frac=0.6):
    """Box in the middle, tapered at edges."""
    j = np.arange(n, dtype=np.float64)
    center = (n - 1) / 2.0
    half = n / 2.0
    x = np.abs(j - center) / half
    f = np.ones(n)
    mask = x > inner_frac
    edge_x = (x[mask] - inner_frac) / (1.0 - inner_frac)
    f[mask] = np.cos(np.pi * edge_x / 2.0) ** 2
    f = np.maximum(f, 0.0)
    s = np.sum(f)
    return f / s if s > 0 else np.ones(n) / n


def _uplifted_cosine_init(n):
    """Cosine with a bump in the middle."""
    j = np.arange(n, dtype=np.float64)
    base = np.cos(np.pi * (j + 0.5) / n) ** 2
    center_bump = np.exp(-((j - n / 2.0) / (n * 0.15)) ** 2 / 2.0)
    f = base + 0.3 * center_bump
    s = np.sum(f)
    return f / s if s > 0 else np.ones(n) / n


def _upsample(f, target_len):
    """Upsample f to target_len using interpolation."""
    n = len(f)
    if target_len <= n:
        idx = np.linspace(0, n - 1, target_len)
        f_new = np.interp(idx, np.arange(n), f)
    else:
        idx = np.linspace(0, n - 1, target_len)
        f_new = np.interp(idx, np.arange(n), f)
    f_new = np.maximum(f_new, 0.0)
    s = np.sum(f_new)
    return f_new / s if s > 0 else np.ones(target_len) / target_len


def _perturb_mult(f, rng, frac=0.03):
    noise = rng.random(len(f)) * frac
    f_p = f * (1.0 + noise)
    return _normalize(f_p)


def _perturb_add(f, rng, frac=0.05):
    noise = rng.random(len(f)) * frac
    return _normalize(f + noise)


def _perturb_freq(f, rng, strength=0.1):
    """Perturb in frequency domain to induce structured changes."""
    n = len(f)
    size = 1
    while size < n:
        size <<= 1
    F = np.fft.rfft(f, size)
    magnitude = np.abs(F)
    phase = np.angle(F)
    mag_noise = 1.0 + strength * rng.standard_normal(len(F))
    mag_noise = np.maximum(mag_noise, 0.1)
    F_new = magnitude * mag_noise * np.exp(1j * phase)
    f_new = np.fft.irfft(F_new, size)[:n]
    return _normalize(f_new)


def _symmetrize(f):
    """Make f symmetric: f[i] = f[n-1-i]."""
    n = len(f)
    f_sym = 0.5 * (f + f[::-1])
    return _normalize(f_sym)

            f_e, s_e = _optimize_energy(f_zoom, t_alloc * 0.4, step_scale=0.4)

            f_p, s_p = _optimize_peak(f_e, t_alloc * 0.5, top_k=400, step_scale=1.0)
            # Then combined
            t_left = (deadline - time.monotonic() - 30) * 0.3
            if t_left > 5:
                f_c, s_c = _optimize_combined(f_p, t_left, top_k=300,
                                              peak_frac=0.5)
                if s_c < s_p:
                    f_p = f_c
                    s_p = s_c

            if s_p < best_score:
                best_score = s_p
                best_f = f_p.copy()
                print(f"  ** Zoom to {target}: {best_score:.6f} **")

    # ============ PHASE 5: Perturbation restarts ============
    print("\n=== Phase 5: Perturbation restarts ===")

    best_f = _normalize(best_f)
    max_pert_rounds = 15
    for rnd in range(max_pert_rounds):
        if time.monotonic() > deadline - 15:
            break

        pert_frac = 0.02 + 0.01 * rnd
        pert_methods = [
            ("mult", lambda: _perturb_mult(best_f, rng, pert_frac)),
            ("add", lambda: _perturb_add(best_f, rng, pert_frac * 0.5)),
            ("freq", lambda: _perturb_freq(best_f, rng, 0.05 + 0.01 * rnd)),
        ]

        for ptype, pfunc in pert_methods:
            if time.monotonic() > deadline - 10:
                break
            t_alloc = min(20.0, (deadline - time.monotonic() - 10) * 0.35)
            if t_alloc < 5:
                break

            f_pert = pfunc()
            if f_pert is None:
                continue

            # Try both peak and combined
            method = _optimize_peak if rnd % 2 == 0 else _optimize_combined
            kwargs_opt = {"top_k": 300, "step_scale": 1.0} if method == _optimize_peak \
                else {"top_k": 300, "peak_frac": 0.5, "step_scale": 1.0}
            f_opt, s_opt = method(f_pert, t_alloc, **kwargs_opt)

            if s_opt < best_score:
                best_score = s_opt
                best_f = f_opt.copy()
                print(f"  ** Pert {ptype} r{rnd}: {best_score:.6f} **")

    # ============ PHASE 6: Final fine-tuning ============
    print("\n=== Phase 6: Final fine-tuning ===")

    for eta_cfg, topk_cfg, temp_cfg in [
        (0.5, 500, 0.0), (0.3, 600, 0.02), (0.8, 300, 0.0), (0.2, 800, 0.05),
    ]:
        if time.monotonic() > deadline - 3:
            break
        t_alloc = min(25.0, (deadline - time.monotonic() - 3) * 0.5)
        if t_alloc < 3:
            break

        f_fin, s_fin = _optimize_peak(best_f, t_alloc, top_k=topk_cfg,
                                      temperature=temp_cfg, step_scale=eta_cfg,
                                      decay=0.9998)
        if s_fin < best_score:
            best_score = s_fin
            best_f = f_fin.copy()
            print(f"  ** Final (ss={eta_cfg}): {best_score:.6f} **")

    # Also try combined for final polish
    if time.monotonic() < deadline - 3:
        t_alloc = min(20.0, (deadline - time.monotonic() - 3) * 0.6)
        if t_alloc > 3:
            f_fin, s_fin = _optimize_combined(best_f, t_alloc, top_k=500,
                                              temperature=0.01, peak_frac=0.7,
                                              step_scale=0.5)
            if s_fin < best_score:
                best_score = s_fin
                best_f = f_fin.copy()
                print(f"  ** Final combined: {best_score:.6f} **")

    # Final verification
    best_f = np.maximum(best_f, 0.0)
    final_list = [float(x) for x in best_f]
    final_score = float(evaluate_sequence(final_list))
    print(f"\nFinal verified score: {final_score:.6f} (n={len(final_list)})")

    if not np.isfinite(final_score) or final_score > 2.5:
        print("  Falling back to parent.")
        return [float(x) for x in parent]

    return final_list
