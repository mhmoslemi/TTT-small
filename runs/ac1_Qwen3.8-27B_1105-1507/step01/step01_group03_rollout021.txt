    try:

        return None
    x = np.where(np.isfinite(x), x, 0.0)
    x = np.maximum(x, 0.0)
    s = float(x.sum())

    return x / s


def score_normalized(x, n):
    """Score = 2*n*max(conv(x,x)) for normalized x (sum=1)."""
    conv = np.convolve(x, x)
    m = float(np.max(conv))
    if not np.isfinite(m):
        return float('inf')
    return 2.0 * n * m


def project_simplex(y):
    """Project y onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = y.size
    if n == 1:
        return np.array([1.0])
    d = np.sort(y)[::-1]
    cssv = np.cumsum(d)
    rho = np.where(d * np.arange(1, n + 1) > (cssv - 1.0))[0]
    if len(rho) == 0:
        return np.full(n, 1.0 / n)
    rho = rho[-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(y - theta, 0.0)


def smoothed_grad_and_value(p, tau):
    """

def make_conv_template(n):
    """Precompute CSR template for LP: A @ g = conv(f, g)."""
    m = 2 * n - 1
    k64 = np.arange(m, dtype=np.int64)
    js = np.maximum(0, k64 - n + 1).astype(np.int32)
    je = np.minimum(n - 1, k64).astype(np.int32)
    counts = (je.astype(np.int64) - js.astype(np.int64) + 1).astype(np.int64)
    total = int(counts.sum())
    
    cum = np.cumsum(counts)
    starts = np.repeat(cum - counts, counts)
    local = np.arange(total, dtype=np.int64) - starts
    
    row = np.repeat(k64, counts).astype(np.int32)
    col_base = js[row.astype(np.int64)]
    col = (col_base + local).astype(np.int32)
    offsets = (row.astype(np.int64) - col.astype(np.int64)).astype(np.int32)
    
    A_off = csr_matrix((offsets, (row, col)), shape=(m, n), dtype=np.int32)
    edge_offsets = A_off.data.astype(np.int32, copy=True)
    A_template = A_off.astype(np.float64).copy()
    A_template.data.fill(0.0)
    
    c_obj = -np.ones(n, dtype=np.float64)
    bounds = [(0.0, None)] * n
    return A_template, edge_offsets, c_obj, bounds, m


def lp_polish(f, n, time_budget, template, rng):
    """
    Run LP-based direction finding to polish f.
    Returns best (f, score) found.
    """
    best_f = f.copy()
    best_score = score_normalized(f, n)
    deadline = time.monotonic() + max(0.1, float(time_budget))
    stall = 0
    
    if template is None:
        try:
            template = make_conv_template(n)
        except Exception:
            return best_f, best_score
    
    while time.monotonic() < deadline and stall < 15:
        f = normalize_sequence(f)
        if f is None:
            break
        conv_ff = np.convolve(f, f)
        M = float(np.max(conv_ff))
        if not np.isfinite(M) or M <= 0:
            break
        
        A_template, edge_offsets, c_obj, bounds, m = template
        A = A_template.copy()
        A.data[:] = f[edge_offsets]
        
        # LP: max sum(g) s.t. A@g <= M, g >= 0
        res = linprog(c_obj, A_ub=A, b_ub=np.full(m, M), bounds=bounds, method='highs')
        
        x = getattr(res, 'x', None)
        if x is None:
            stall += 1
            continue
        
        x = np.maximum(np.asarray(x, dtype=np.float64), 0.0)
        s_star = float(x.sum())
        if s_star <= 1e-12:
            stall += 1
            continue
        
        g = x / s_star
        
        # Safe step
        r = np.convolve(f, g)
        r_max = float(np.max(r))
        G = float(np.max(np.convolve(g, g)))
        
        a_coef = 2.0 * (r_max - M)
        b_coef = G + M - 2.0 * r_max
        t_max = -a_coef / b_coef if b_coef > 1e-14 else 1.0
        if not np.isfinite(t_max) or t_max <= 0:
            t_max = 1e-6
        t_max = min(1.0, 0.8 * t_max)
        
        # Line search
        best_t = 0
        best_sc = best_score
        for tv in [t_max, 0.5*t_max, 0.25*t_max, 0.1*t_max, 0.01*t_max, 0.001*t_max]:
            if tv <= 0:
                continue
            f_new = np.maximum((1-tv)*f + tv*g, 0)
            s = f_new.sum()
            if s <= 1e-12:
                continue
            f_new /= s
            sc = score_normalized(f_new, n)
            if sc < best_sc:
                best_sc = sc
                best_t = tv
        
        if best_t > 0 and best_sc < best_score:
            best_f = best_f - best_f + ((1-best_t)*f + best_t*g)  # Hmm this is wrong
            # Actually:
            new_f = np.maximum((1-best_t)*f + best_t*g, 0)
            new_f = normalize_sequence(new_f)
            if new_f is not None:
                best_f = new_f
                best_score = best_sc
            stall = 0
        else:
            stall += 1
            # Perturb
            f = normalize_sequence(f + 0.1 * np.maximum(rng.normal(0, 0.05, n), 0))
    
    return best_f, best_score


def generate_initializations(n, rng, given_f=None):
    """Generate diverse initializations for length n."""
    inits = []
    
    # Uniform
    inits.append(np.ones(n) / n)
    
    # Gaussian-like (centered)
    x = np.linspace(0, 1, n)
    g = np.exp(-8 * (x - 0.5)**2)
    inits.append(g / g.sum())
    
    # Bimodal (edges)
    g2 = np.exp(-20 * x**2) + np.exp(-20 * (x-1)**2)
    inits.append(g2 / g2.sum())
    
    # Cosine-squared chirp
    for freq in [1, 2, 3]:
        c = 1 + 0.3 * np.cos(2 * np.pi * freq * x)
        c = np.maximum(c, 0.01)
        inits.append(c / c.sum())
    
    # Random Dirichlet
    for alpha_val in [1.0, 0.3, 5.0]:
        d = rng.dirichlet(np.full(n, alpha_val))
        inits.append(d)
    
    # From given sequence (interpolated)
    if given_f is not None:
        old_m = given_f.size
        old_t = np.linspace(0, 1, old_m)
        new_t = np.linspace(0, 1, n)
        interp = np.interp(new_t, old_t, given_f)
        interp = np.maximum(interp, 0)
        s = interp.sum()
        if s > 0:
            inits.append(interp / s)
    
    return inits
