import numpy as np
import time
from scipy.optimize import linprog
from scipy.sparse import csr_matrix


def build_conv_matrix(f):
    """Build sparse matrix A where (A@g)[k] = convolve(f,g)[k]."""
    n = len(f)
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        j_min = max(0, k - (n - 1))
        j_max = min(k, n - 1)
        for j in range(j_min, j_max + 1):
            rows.append(k)
            cols.append(j)
            vals.append(f[k - j])
    return csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))


def solve_lp(f, M):
    """Solve: max sum(g) s.t. g >= 0, convolve(f,g) <= M."""
    n = len(f)
    A = build_conv_matrix(f)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    try:
        result = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
        if result.success:
            return result.x
    except Exception:
        pass
    return None


def lp_iterate(f, deadline, step_init=0.1, verbose=False, prefix=""):

def make_initializations(n, rng):

    inits.append(0.5 + 0.4 * np.sin(2 * np.pi * idx / n))

    pat = np.concatenate([np.ones(period // 2), np.zeros(period // 2)])
    inits.append(np.tile(pat, n // period + 1)[:n] + 0.1)

                break

    # Final deep optimization on best sequence found
    if best_seq is not None and time.monotonic() < deadline - 10:
        print(f"\n--- Final deep optimization (n={len(best_seq)}, score={best_score:.6f}) ---")

        # Multiple rounds of LP + gradient with perturbation
        round_deadline = deadline
        for rnd in range(5):
            if time.monotonic() > round_deadline - 5:
                break

            # LP refinement
            f_cur, s_cur = lp_iterate(best_seq, round_deadline, step_init=0.05)
            try_update(f_cur, f"refine LP r{rnd}")

            # Gradient polish
            if time.monotonic() > round_deadline - 3:
                break
            gf, gs = gradient_polish(best_seq, round_deadline, lr=0.002)
            try_update(gf, f"refine grad r{rnd}")

            # Try a perturbation to escape local minimum
            if time.monotonic() > round_deadline - 3:
                break
            perturbed = random_perturb(best_seq, 0.05 * (0.5 ** rnd), rng)
            try_update(perturbed, f"perturb r{rnd}")
            if time.monotonic() > round_deadline - 3:
                break
            f_p, s_p = lp_iterate(perturbed, round_deadline, step_init=0.05)
            try_update(f_p, f"perturb+LP r{rnd}")

    if best_seq is None:
        best_seq = np.ones(128)
        best_score = evaluate_sequence(best_seq.tolist())

    elapsed = time.monotonic() - t0
    print(f"\nFINAL: n={len(best_seq)}, score={best_score:.6f}, time={elapsed:.1f}s")
    return [float(x) for x in best_seq]
