def solve_best_response(a, n):
    """Solve: min M s.t. conv(a,b)_k <= M, sum(b)=1, b>=0"""
    # Variables: b_0, ..., b_{n-1}, M (total n+1 variables)
    # Objective: minimize M -> c = [0,...,0, 1]
    # Constraints:
    #   sum(b) = 1 -> equality
    #   conv(a,b)_k <= M for k=0..2n-2 -> (2n-1) inequalities
    #   b >= 0
    #   M >= 0 (implied by b >= 0 and sum(b) = 1, but include for safety)
    
    from scipy.optimize import linprog
    from scipy.sparse import csr_matrix, coo_matrix
    
    # Build constraint matrix for conv(a,b)_k - M <= 0
    # conv(a,b)_k = sum_{i: 0<=i<n, 0<=k-i<n} a_i * b_{k-i}
    # So conv(a,b)_k - M <= 0
    # In matrix form: [A_k, -e_k] * [b; M] <= 0
    # where A_k is a row with A_k[j] = a_{k-j} if 0<=k-j<n
    
    rows = []
    cols = []
    vals = []
    
    for k in range(2*n-1):
        for j in range(n):
            i = k - j
            if 0 <= i < n:
                rows.append(k)
                cols.append(j)  # coefficient of b_j
                vals.append(a[i])
        # Add -1 for M (variable at position n)
        rows.append(k)
        cols.append(n)
        vals.append(-1.0)
    
    A_ineq = csr_matrix((vals, (rows, cols)), shape=(2*n-1, n+1))
    b_ineq = np.zeros(2*n-1)
    
    # Equality: sum(b) = 1
    A_eq = csr_matrix(np.array([[1]*n + [0]]), shape=(1, n+1))
    b_eq = np.array([1.0])
    
    # Bounds: b_i >= 0, M >= 0
    bounds = [(0, None)] * n + [(0, None)]
    
    c = np.zeros(n+1)
    c[n] = 1.0  # minimize M
    
    result = linprog(c, A_ub=A_ineq, b_ub=b_ineq, A_eq=A_eq, b_eq=b_eq, 
                     bounds=bounds, method='highs')
    
    if result.success:
        return result.x[:n], result.x[n]  # b, M
    return None, None
```

def smooth_objective(a, alpha=100):
    """Compute sum_k exp(alpha * conv_k) / sum_k 1 as a smooth approximation to max."""
    conv = np.convolve(a, a)
    # Soft max approximation
    return np.sum(np.exp(alpha * (conv - conv.max()))) / len(conv) + conv.max()

def smooth_gradient(a, alpha=100):
    """Gradient of smooth objective w.r.t. a."""
    conv = np.convolve(a, a)
    weights = np.exp(alpha * (conv - conv.max()))
    weights = weights / weights.sum()
    # Gradient: sum_k weights[k] * d(conv_k)/d(a_i) = sum_k weights[k] * 2*a_{k-i}
    n = len(a)
    grad = np.zeros(n)
    for k in range(len(conv)):
        for i in range(n):
            j = k - i
            if 0 <= j < n:
                grad[i] += weights[k] * 2 * a[j]
    return grad
```

Multi-phase optimization for minimizing 2n*max(conv(a,a))/sum(a)^2.
Uses smooth gradient descent (L-BFGS-B) with adaptive temperature,
LP best-response iteration, and local perturbation for escaping local minima.
"""
import numpy as np
import time
from scipy.optimize import linprog, minimize
from scipy.sparse import csr_matrix


def soft_max_conv(a, alpha):
    """Compute soft-max approximation of max(conv(a,a))."""
    conv = np.convolve(a, a)
    m = conv.max()
    if m <= 0:
        return 0.0
    scaled = alpha * (conv - m)
    # For numerical stability
    max_scaled = scaled.max()
    exp_vals = np.exp(scaled - max_scaled)
    return m + (max_scaled + np.log(np.sum(exp_vals))) / alpha


def soft_max_conv_grad(a, alpha):
    """Gradient of soft-max approximation of max(conv(a,a))."""
    conv = np.convolve(a, a)
    m = conv.max()
    scaled = alpha * (conv - m)
    max_scaled = scaled.max()
    exp_vals = np.exp(scaled - max_scaled)
    w = exp_vals / exp_vals.sum()
    
    n = len(a)
    grad = np.zeros(n)
    # grad[m] = 2 * sum_{j} w[m+j] * a[j] (for valid m+j < 2n-1)
    # Use matrix form: grad = 2 * H @ a where H[m,j] = w[m+j]
    # For efficiency, use the fact that this is a sliding dot product
    for j in range(n):
        # w[m+j] for m=0..n-1 is w[j:2n-1] (truncated)
        valid = min(n, 2*n-1-j)
        grad[:valid] += a[j] * w[j:j+valid]
    grad *= 2.0
    return grad


def score(a):
    """Compute the evaluation score."""
    a_list = [float(max(0, x)) for x in a]
    a_list = [min(1000.0, x) for x in a_list]
    n = len(a_list)
    conv = np.convolve(a_list, a_list)
    max_b = np.max(conv)
    sum_a = np.sum(a_list)
    if sum_a < 0.01:
        return float('inf')
    return float(2 * n * max_b / (sum_a ** 2))


def lp_best_response(a, n):
    """Solve: min M s.t. conv(a,b)_k <= M, sum(b)=1, b>=0.
    Returns (b, M) or (None, None)."""
    # Variables: b_0,...,b_{n-1}, M => n+1 variables
    # Minimize M
    # s.t. conv(a,b)_k - M <= 0 for k=0..2n-2
    #      sum(b) = 1
    #      b >= 0, M >= 0
    
    # Build A_ub for inequality constraints: conv(a,b)_k - M <= 0
    rows = []
    cols = []
    vals = []
    
    for k in range(2*n-1):
        # conv(a,b)_k = sum_{j} a_{k-j} * b_j (for valid k-j)
        for j in range(n):
            i = k - j
            if 0 <= i < n:
                rows.append(k)
                cols.append(j)
                vals.append(a[i])
        # Subtract M
        rows.append(k)
        cols.append(n)  # M is variable index n
        vals.append(-1.0)
    
    A_ub = csr_matrix((vals, (rows, cols)), shape=(2*n-1, n+1))
    b_ub = np.zeros(2*n-1)
    
    # Equality: sum(b) = 1
    A_eq = np.zeros((1, n+1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])
    
    c = np.zeros(n+1)
    c[n] = 1.0  # minimize M
    
    bounds = [(0, None)] * (n+1)
    
    result = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                     bounds=bounds, method='highs', options={'presolve': True})
    
    if result.success:
        return result.x[:n], result.x[n]
    return None, None


def smooth_descent(a, deadline, alpha_start=50, alpha_end=5000, max_iter=200):
    """L-BFGS-B style smooth descent with increasing alpha."""
    a = a.copy().astype(np.float64)
    n = len(a)
    best_a = a.copy()
    best_score = score(a)
    
    # Normalize
    s = a.sum()
    if s > 0:
        a = a / s
    
    for restart in range(5):
        alpha = alpha_start * (alpha_end / alpha_start) ** (restart / 4.0)
        
        def obj(x):
            if x.sum() < 1e-10:
                return 1e10
            x_norm = x / x.sum()
            return soft_max_conv(x_norm, alpha)
        
        def obj_grad(x):
            if x.sum() < 1e-10:
                return np.zeros(n)
            x_norm = x / x.sum()
            g = soft_max_conv_grad(x_norm, alpha)
            return g
        
        x0 = a.copy()
        result = minimize(obj, x0, jac=obj_grad, method='L-BFGS-B',
                         bounds=[(0, None)]*n,
                         options={'maxiter': max_iter, 'ftol': 1e-12, 'gtol': 1e-8})
        
        if result.success or result.fun < obj(a):
            a = result.x
            a = np.clip(a, 0, 1000)
            new_score = score(a)
            if new_score < best_score:
                best_score = new_score
                best_a = a.copy()
                print(f"    smooth: alpha={alpha:.0f}, score={best_score:.6f}")
        
        if time.monotonic() > deadline:
            break
    
    return best_a, best_score


def lp_iterate(a, deadline, step_size=0.3, max_iters=200):
    """LP best-response iteration."""
    a = a.copy().astype(np.float64)
    n = len(a)
    best_a = a.copy()
    best_score = score(a)
    
    s = a.sum()
    if s > 0:
        a = a / s
    
    no_improve = 0
    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        
        b, M = lp_best_response(a, n)
        if b is None:
            no_improve += 1
            if no_improve > 10:
                break
            continue
        
        # Blend
        new_a = (1 - step_size) * a + step_size * b
        new_a = np.clip(new_a, 0, 1000)
        ns = new_a.sum()
        if ns > 0:
            new_a = new_a / ns
        
        new_score = score(new_a)
        if np.isfinite(new_score) and new_score < best_score:
            a = new_a
            best_a = a.copy()
            best_score = new_score
            no_improve = 0
            if it % 20 == 0:
                print(f"    LP iter {it}: score={best_score:.6f}, M={M:.6f}")
        else:
            no_improve += 1
            step_size *= 0.8
            if step_size < 0.01 or no_improve > 20:
                break
    
    return best_a, best_score


def perturb(a, rng, frac=0.1):
    """Randomly perturb a fraction of entries."""
    a = a.copy()
    n = len(a)
    n_perturb = max(1, int(frac * n))

    a = np.clip(a, 0, 1000)
    s = a.sum()
    if s > 0:
        a = a / s
    return a

Let me also add one more idea: **iterative reweighting**.

The idea: at each iteration, compute the convolution, find the "hot" lags (those with high values), and modify the sequence to reduce contributions to those lags while maintaining contributions to "cold" lags.

Specifically:
1. Compute conv = a*a
2. Compute weights w_k = exp(beta * conv_k) / sum(exp(beta * conv_j)) (concentrated on hot lags)
3. The "effective" convolution weighted by w is: sum_k w_k * conv_k = sum_k w_k * sum_i a_i * a_{k-i}
4. To reduce this, we want to reduce a_i * a_{k-i} for hot lags k
5. Update: a_i *= (1 - eta * sum_k w_k * a_{k-i}) for appropriate eta

This is essentially what the smooth gradient descent does, but with a multiplicative update instead of additive. Multiplicative updates can be better for non-negative variables.

"""
Multi-phase optimization for minimizing 2n*max(conv(a,a))/sum(a)^2.
Uses smooth L-BFGS-B descent with adaptive temperature, LP best-response iteration,
multiplicative updates, and local perturbation to escape local minima.
"""
import numpy as np
import time
from scipy.optimize import linprog, minimize
from scipy.sparse import csr_matrix


def compute_score(a):
    """Compute the evaluation score directly (fast, no function call overhead)."""
    a = np.asarray(a, dtype=np.float64)
    a = np.clip(a, 0.0, 1000.0)
    n = len(a)
    conv = np.convolve(a, a)
    max_b = float(np.max(conv))
    sum_a = float(np.sum(a))
    if sum_a < 0.01:
        return float('inf')
    return float(2.0 * n * max_b / (sum_a ** 2))


def soft_max_conv_value(conv, alpha):
    """Compute soft-max of conv array."""
    m = conv.max()
    if m <= 0:
        return 0.0
    scaled = alpha * (conv - m)
    max_s = scaled.max()
    return m + (max_s + np.log(np.sum(np.exp(scaled - max_s)))) / alpha


def soft_max_conv_weights(conv, alpha):
    """Compute normalized exponential weights for soft-max."""
    m = conv.max()
    if m <= 0:
        return np.ones(len(conv)) / len(conv)
    scaled = alpha * (conv - m)
    max_s = scaled.max()
    exp_vals = np.exp(scaled - max_s)
    return exp_vals / exp_vals.sum()


def soft_max_grad_from_weights(a, w):
    """Compute gradient given pre-computed weights w over lags."""
    n = len(a)
    grad = np.zeros(n)
    for j in range(n):
        grad += a[j] * w[j:j + n]
    grad *= 2.0
    return grad


def smooth_descent_lbfgs(a, deadline, alpha_start=30, alpha_end=10000, total_time=60.0):
    """L-BFGS-B smooth descent with increasing alpha schedule."""
    a = a.copy().astype(np.float64)
    n = len(a)
    s = a.sum()
    if s <= 0:
        return a, compute_score(a)
    a = a / s
    best_a = a.copy()
    best_score = compute_score(a)

    n_phases = 5
    for phase in range(n_phases):
        if time.monotonic() > deadline:
            break
        alpha = alpha_start * (alpha_end / alpha_start) ** (phase / max(1, n_phases - 1))
        if phase == n_phases - 1:
            alpha = alpha_end

        lambda_pen = 500.0

        def obj(x):
            conv = np.convolve(x, x)
            val = soft_max_conv_value(conv, alpha)
            s = np.sum(x)
            penalty = lambda_pen * (s - 1.0) ** 2
            return val + penalty

        def obj_grad(x):
            conv = np.convolve(x, x)
            w = soft_max_conv_weights(conv, alpha)
            g = soft_max_grad_from_weights(x, w)
            s = np.sum(x)
            g += 2.0 * lambda_pen * (s - 1.0) * np.ones(n)
            return g

        x0 = best_a.copy()
        try:
            result = minimize(obj, x0, jac=obj_grad, method='L-BFGS-B',
                              bounds=[(0, None)] * n,
                              options={'maxiter': 300, 'ftol': 1e-14, 'gtol': 1e-10,
                                       'maxls': 50})
            if result.x is not None:
                new_a = np.clip(result.x, 0, 1000)
                ns = new_a.sum()
                if ns > 0:
                    new_a = new_a / ns
                new_score = compute_score(new_a)
                if np.isfinite(new_score) and new_score < best_score:
                    best_score = new_score
                    best_a = new_a.copy()
        except Exception:
            pass

    return best_a, best_score


def lp_best_response(a, n):
    """Solve: min M s.t. conv(a,b)_k <= M for all k, sum(b)=1, b>=0.
    Returns (b, M) or (None, None)."""
    rows = []
    cols = []
    vals = []

    for k in range(2 * n - 1):
        for j in range(n):
            i = k - j
            if 0 <= i < n:
                rows.append(k)
                cols.append(j)
                vals.append(a[i])
        rows.append(k)
        cols.append(n)
        vals.append(-1.0)

    A_ub = csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n + 1))
    b_ub = np.zeros(2 * n - 1)

    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])

    c = np.zeros(n + 1)
    c[n] = 1.0

    bounds = [(0, None)] * (n + 1)

    try:
        result = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                         bounds=bounds, method='highs',
                         options={'presolve': True, 'time_limit': 10})
        if result.success:
            return result.x[:n], float(result.x[n])
    except Exception:
        pass
    return None, None


def lp_iterate(a, deadline, step_size_init=0.3, max_iters=500):
    """LP best-response iteration with adaptive step size."""
    a = a.copy().astype(np.float64)
    n = len(a)
    s = a.sum()
    if s > 0:
        a = a / s
    best_a = a.copy()
    best_score = compute_score(a)
    step_size = step_size_init
    no_improve = 0

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        b, M = lp_best_response(a, n)
        if b is None:
            no_improve += 1
            if no_improve > 15:
                break
            continue

        new_a = (1.0 - step_size) * a + step_size * b
        new_a = np.clip(new_a, 0.0, 1000.0)
        ns = new_a.sum()
        if ns > 0:
            new_a = new_a / ns

        new_score = compute_score(new_a)
        if np.isfinite(new_score) and new_score < best_score:
            a = new_a
            best_a = a.copy()
            best_score = new_score
            no_improve = 0
            if (it + 1) % 25 == 0:
                print(f"    LP iter {it+1}: score={best_score:.6f}, M={M:.8f}")
        else:
            no_improve += 1
            step_size *= 0.75
            if step_size < 0.005 or no_improve > 25:
                break

    return best_a, best_score


def multiplicative_iterate(a, deadline, beta_start=50, beta_end=2000, eta_init=0.15, max_iters=2000):
    """Multiplicative update iteration with increasing beta."""
    a = a.copy().astype(np.float64)
    n = len(a)
    s = a.sum()
    if s > 0:
        a = a / s
    best_a = a.copy()
    best_score = compute_score(a)
    eta = eta_init
    no_improve = 0

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        beta = min(beta_start + (beta_end - beta_start) * it / max(1, max_iters - 1), beta_end)
        conv = np.convolve(a, a)
        m = conv.max()
        scaled = beta * (conv - m)
        max_s = scaled.max()
        w = np.exp(scaled - max_s)
        w = w / w.sum()

        update = np.zeros(n)
        for j in range(n):
            update += a[j] * w[j:j + n]

        a_new = a * (1.0 - eta * update)
        a_new = np.clip(a_new, 0.0, 1000.0)
        ns = a_new.sum()
        if ns > 0:
            a_new = a_new / ns

        new_score = compute_score(a_new)
        if np.isfinite(new_score) and new_score < best_score:
            a = a_new
            best_a = a.copy()
            best_score = new_score
            no_improve = 0
            if (it + 1) % 200 == 0:
                print(f"    mult iter {it+1}: score={best_score:.6f}, beta={beta:.0f}")
        else:
            no_improve += 1
            eta *= 0.9
            if eta < 1e-5 or no_improve > 50:
                break

    return best_a, best_score


def perturb_sequence(a, rng, frac=0.08):
    """Randomly perturb a fraction of entries multiplicatively."""
    a = a.copy().astype(np.float64)
    n = len(a)
    n_perturb = max(1, int(frac * n))
    idx = rng.choice(n, size=n_perturb, replace=False)
    a[idx] *= rng.uniform(0.3, 3.0, size=n_perturb)
    a = np.clip(a, 0.0, 1000.0)
    s = a.sum()
    if s > 0:
        a = a / s
    return a
