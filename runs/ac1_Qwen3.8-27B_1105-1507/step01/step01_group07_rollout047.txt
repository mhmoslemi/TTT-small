```python
"""
Hybrid optimization for minimizing 2*n*max(conv(a,a))/(sum(a))^2.
Uses multiplicative weight descent (subgradient of max) combined with
LP polishing, multiple initializations, and various sequence lengths.
"""

import numpy as np
from scipy.optimize import linprog
import time


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    conv = np.convolve(a, a)
    max_b = float(np.max(conv))
    sum_a = float(np.sum(a))
    if sum_a < 0.01:
        return np.inf
    return 2.0 * n * max_b / (sum_a * sum_a)


def build_A(a):
    """Build constraint matrix for LP. A[k,j] = a[k-j] if 0<=k-j<n else 0."""
    n = len(a)
    A = np.zeros((2 * n - 1, n), dtype=np.float64)
    for k in range(2 * n - 1):
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        if j_start <= j_end:
            # For each j in [j_start, j_end], A[k,j] = a[k-j]
            j_range = np.arange(j_start, j_end + 1)
            A[k, j_range] = a[k - j_range]
    return A


def solve_lp_for_g(a, M=None):
    """Solve LP: max sum(g) s.t. conv(a,g) <= M, g >= 0."""
    n = len(a)
    if M is None:
        conv = np.convolve(a, a)
        M = float(np.max(conv))
    A = build_A(a)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
    if res.status == 0:
        return res.x, float(-res.fun)
    return None, None


def mwd_step(a, eta, active_fraction=0.1):
    """
    Multiplicative weight descent step.
    Computes subgradient of max(conv(a,a)) and applies multiplicative update.
    """
    n = len(a)
    conv = np.convolve(a, a)
    M = float(np.max(conv))
    
    # Find active set (top fraction of convolution values)
    threshold = np.sort(conv)[-max(1, int(active_fraction * len(conv))):][0]
    active = np.where(conv >= threshold)[0]
    
    # Compute subgradient: average gradient over active set
    # grad of (conv)_k w.r.t. a_i = a_{k-i} + a_{i-k} (if indices valid)
    s = np.zeros(n)
    for k in active:
        for i in range(n):
            val = 0.0
            if 0 <= k - i < n:
                val += a[k - i]
            if 0 <= i - k < n:
                val += a[i - k]
            s[i] += val
    s /= len(active)
    
    # Multiplicative update
    a_new = a * np.exp(-eta * s)
    a_new = np.clip(a_new, 1e-300, None)
    s_new = np.sum(a_new)
    if s_new < 1e-20:
        return a
    a_new /= s_new
    return a_new

def run_mwd(a, max_iter=500, eta0=0.1, eta_min=0.001, active_fraction=0.05):
    """Run multiplicative weight descent iterations."""
    n = len(a)
    a = a / np.sum(a)
    best_a = a.copy()
    best_obj = compute_obj(best_a)
    
    for it in range(max_iter):
        eta = max(eta_min, eta0 * 0.995 ** it)
        a_new = mwd_fast(a, eta, active_fraction)
        obj = compute_obj(a_new)
        if obj < best_obj:
            best_a = a_new.copy()
            best_obj = obj
        a = a_new
    
    return best_a, best_obj


def run_lp_optimize(a_init, max_iter=200, t=0.5):
    """Run LP-based alternating minimization."""
    n = len(a_init)
    a = a_init.copy()
    a = a / np.sum(a)
    best_a = a.copy()
    best_obj = compute_obj(best_a)
    
    stall = 0
    for it in range(max_iter):
        conv = np.convolve(a, a)
        M = float(np.max(conv))
        
        g, g_sum = solve_lp_for_g(a, M)
        if g is None:
            break
        
        if g_sum <= 1.0 + 1e-5:
            stall += 1
            if stall > 10:
                break
            continue
        stall = 0
        
        g = g / g_sum
        a_new = (1.0 - t) * a + t * g
        a_new = np.clip(a_new, 0, None)
        s = np.sum(a_new)
        if s < 1e-10:
            break
        a_new /= s
        
        obj_new = compute_obj(a_new)
        if obj_new < best_obj:
            best_a = a_new.copy()
            best_obj = obj_new
        a = a_new
    
    return best_a, best_obj


def random_refine(a, rng, max_iter=3000, noise_scale=0.1):
    """Random block perturbation local search."""
    best_a = a.copy()
    best_obj = compute_obj(best_a)
    n = len(best_a)
    
    for it in range(max_iter):
        width = max(1, n // 30)
        start = int(rng.integers(0, n - width + 1))
        candidate = best_a.copy()
        candidate[start:start + width] *= np.exp(rng.normal(0, noise_scale, width))
        candidate = np.clip(candidate, 0, None)
        s = np.sum(candidate)
        if s < 1e-10:
            continue
        obj_new = compute_obj(candidate)
        if obj_new < best_obj:
            best_a = candidate
            best_obj = obj_new
    
    return best_a, best_obj


def cosine_init(n):
    """Cosine tapered initialization."""
    x = np.arange(n) / max(1, n - 1)
    a = np.cos(np.pi * x) ** 2
    a = np.maximum(a, 0.01)
    return a / np.sum(a)


def gaussian_init(n, sigma_frac=0.4):
    """Gaussian initialization."""
    x = np.arange(n) / max(1, n - 1) - 0.5
    a = np.exp(-x**2 / (2 * sigma_frac**2))
    return a / np.sum(a)


def uniform_init(n):
    return np.ones(n) / n


def random_init(n, rng):
    a = rng.random(n)
    return a / np.sum(a)
```

```python
    n = len(a)
    a = a / np.sum(a)

def smooth_max_grad(a, beta):
```

```python
"""
Hybrid optimization for minimizing 2*n*max(conv(a,a))/(sum(a))^2.
Combines smooth gradient descent (log-sum-exp approximation of max),
multiplicative weight descent, LP polishing, and random refinement.
Uses adaptive beta annealing for the smooth approximation to balance
exploration and exploitation across multiple sequence lengths.
"""

import numpy as np
from scipy.optimize import linprog
import time


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    conv = np.convolve(a, a)
    max_b = float(np.max(conv))
    sum_a = float(np.sum(a))
    if sum_a < 0.01:
        return np.inf
    return 2.0 * n * max_b / (sum_a * sum_a)


def smooth_obj_and_grad(a, beta):
    """
    Compute smooth approximation of 2n*max(conv(a,a)) (assuming sum(a)=1)
    and its gradient using log-sum-exp.
    Returns (objective_value, gradient_wrt_a).
    """
    n = len(a)
    conv = np.convolve(a, a)
    max_c = np.max(conv)
    exps = np.exp(beta * (conv - max_c))
    sum_exps = np.sum(exps)
    weights = exps / sum_exps
    log_sum_exp = max_c + np.log(sum_exps)
    smooth_max = log_sum_exp / beta
    obj_val = 2.0 * n * smooth_max

    # Gradient: sum_k weights[k] * d(conv_k)/da_i
    # d(conv_k)/da_i = a[k-i] + a[i-k] (when indices valid)
    s = np.zeros(n)
    for k in range(2 * n - 1):
        w = weights[k]
        if w < 1e-16:
            continue
        # a[k-i] valid when 0 <= k-i < n => k-n+1 <= i <= k
        i_lo1 = max(0, k - n + 1)
        i_hi1 = min(n - 1, k)
        if i_lo1 <= i_hi1:
            s[i_lo1:i_hi1 + 1] += w * a[k - i_hi1:k - i_lo1 + 1][::-1]
        # a[i-k] valid when 0 <= i-k < n => k <= i <= k+n-1
        i_lo2 = max(0, k)
        i_hi2 = min(n - 1, k + n - 1)
        if i_lo2 <= i_hi2:
            s[i_lo2:i_hi2 + 1] += w * a[i_lo2 - k:i_hi2 - k + 1]

    grad = 2.0 * n * s
    return obj_val, grad


def project_simplex(v):
    """Project vector onto {x >= 0, sum(x) = 1}."""
    n = len(v)
    if np.sum(v) < 1e-15:
        return np.ones(n) / n
    # Defense: clip negatives
    v = np.maximum(v, 0.0)
    s = np.sum(v)
    if s < 1e-15:
        return np.ones(n) / n
    return v / s


def build_A(a):
    """Build constraint matrix for LP: A[k,j] = a[k-j] if valid."""
    n = len(a)
    A = np.zeros((2 * n - 1, n), dtype=np.float64)
    for k in range(2 * n - 1):
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        if j_start <= j_end:
            j_range = np.arange(j_start, j_end + 1)
            A[k, j_range] = a[k - j_range]
    return A


def solve_lp_for_g(a):
    """Solve LP: max sum(g) s.t. conv(a,g) <= M, g >= 0. Returns (g, g_sum, M)."""
    n = len(a)
    conv = np.convolve(a, a)
    M = float(np.max(conv))
    A = build_A(a)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
    if res.status == 0:
        return res.x, float(-res.fun), M
    return None, None, M


def run_smooth_gradient(a_init, beta_start=5.0, beta_end=200.0, max_iter=300,
                        lr0=0.1, lr_min=0.001, verbose=False):
    """
    Smooth gradient descent with annealing beta.
    Uses momentum (heavy ball) for faster convergence.
    """
    n = len(a_init)
    a = a_init.copy()
    a = project_simplex(a)
    best_a = a.copy()
    best_obj = compute_obj(best_a)

    momentum = np.zeros(n)
    gamma = 0.9  # momentum coefficient

    for it in range(max_iter):
        # Anneal beta
        frac = it / max(1, max_iter - 1)
        beta = beta_start * (beta_end / beta_start) ** frac
        lr = lr0 * (lr_min / lr0) ** frac

        obj_val, grad = smooth_obj_and_grad(a, beta)

        # Gradient step with momentum
        momentum = gamma * momentum - lr * grad
        a_new = a + momentum

        # Project to simplex
        a_new = project_simplex(a_new)

        # Check actual objective
        actual_obj = compute_obj(a_new)
        if actual_obj < best_obj:
            best_a = a_new.copy()
            best_obj = actual_obj
            if verbose and (it % 50 == 0):
                print(f"    SG it={it} beta={beta:.1f} obj={best_obj:.6f}")

        a = a_new

    return best_a, best_obj


def run_mwd(a_init, max_iter=500, eta0=0.1, eta_min=0.001, active_fraction=0.05,
            verbose=False):
    """
    Multiplicative weight descent for minimizing max(conv(a,a)) subject to sum=1.
    Uses subgradient of the true max (not smooth approximation).
    """
    n = len(a_init)
    a = a_init.copy()
    a = project_simplex(a)
    best_a = a.copy()
    best_obj = compute_obj(best_a)

    for it in range(max_iter):
        eta = max(eta_min, eta0 * (eta_min / eta0) ** (it / max(1, max_iter)))

        conv = np.convolve(a, a)
        M = float(np.max(conv))

        # Find active set (top fraction)
        n_active = max(1, int(active_fraction * (2 * n - 1)))
        sorted_idx = np.argsort(conv)[-n_active:]
        active = sorted_idx

        # Compute subgradient
        s = np.zeros(n)
        for k in active:
            i_lo1 = max(0, k - n + 1)
            i_hi1 = min(n - 1, k)
            if i_lo1 <= i_hi1:
                s[i_lo1:i_hi1 + 1] += a[k - i_hi1:k - i_lo1 + 1][::-1]
            i_lo2 = max(0, k)
            i_hi2 = min(n - 1, k + n - 1)
            if i_lo2 <= i_hi2:
                s[i_lo2:i_hi2 + 1] += a[i_lo2 - k:i_hi2 - k + 1]
        s /= len(active)

        # Multiplicative update
        a_new = a * np.exp(-eta * s)
        a_new = np.clip(a_new, 1e-300, None)
        total = np.sum(a_new)
        if total < 1e-20:
            break
        a_new /= total

        obj = compute_obj(a_new)
        if obj < best_obj:
            best_a = a_new.copy()
            best_obj = obj
            if verbose and (it % 100 == 0):
                print(f"    MWD it={it} obj={best_obj:.6f}")
        a = a_new

    return best_a, best_obj


def run_lp_optimize(a_init, max_iter=100, t=0.4, verbose=False):
    """LP-based alternating minimization."""
    n = len(a_init)
    a = a_init.copy()
    a = project_simplex(a)
    best_a = a.copy()
    best_obj = compute_obj(best_a)

    stall = 0
    for it in range(max_iter):
        g, g_sum, M = solve_lp_for_g(a)
        if g is None:
            break

        if g_sum <= 1.0 + 1e-6:
            stall += 1
            if stall > 8:
                break
            continue
        stall = 0

        g = g / g_sum
        a_new = (1.0 - t) * a + t * g
        a_new = project_simplex(a_new)

        obj_new = compute_obj(a_new)
        if obj_new < best_obj:
            best_a = a_new.copy()
            best_obj = obj_new
            if verbose and (it % 20 == 0):
                print(f"    LP it={it} obj={best_obj:.6f} gsum={g_sum:.4f}")
        a = a_new

    return best_a, best_obj


def random_refine(a, rng, max_iter=5000, noise_scale=0.08):
    """Random block perturbation local search."""
    best_a = a.copy()
    best_obj = compute_obj(best_a)
    n = len(best_a)

    for it in range(max_iter):
        width = max(1, n // 25)
        start = int(rng.integers(0, n - width + 1))
        candidate = best_a.copy()
        candidate[start:start + width] *= np.exp(rng.normal(0, noise_scale, width))
        candidate = np.clip(candidate, 0, None)
        s = np.sum(candidate)
        if s < 1e-10:
            continue
        candidate /= s
        obj_new = compute_obj(candidate)
        if obj_new < best_obj:
            best_a = candidate
            best_obj = obj_new

    return best_a, best_obj


def simulated_annealing(a, rng, max_iter=5000, T0=0.01, T_end=0.001, noise_scale=0.15):
    """Simulated annealing with block moves."""
    best_a = a.copy()
    best_obj = compute_obj(best_a)
    cur_a = a.copy()
    cur_obj = best_obj
    n = len(cur_a)

    for it in range(max_iter):
        T = T0 * (T_end / T0) ** (it / max(1, max_iter))
        width = max(1, n // 20)
        start = int(rng.integers(0, n - width + 1))
        candidate = cur_a.copy()
        candidate[start:start + width] *= np.exp(rng.normal(0, noise_scale, width))
        candidate = np.clip(candidate, 0, None)
        s = np.sum(candidate)
        if s < 1e-10:
            continue
        candidate /= s
        obj_new = compute_obj(candidate)

        delta = obj_new - cur_obj
        if delta < 0 or rng.random() < np.exp(-delta / max(T, 1e-10)):
            cur_a = candidate
            cur_obj = obj_new
            if cur_obj < best_obj:
                best_a = cur_a.copy()
                best_obj = cur_obj

    return best_a, best_obj


def cosine_init(n):
    """Cosine tapered initialization."""
    x = np.arange(n) / max(1, n - 1)
    a = np.cos(np.pi * x) ** 2
    a = np.maximum(a, 0.005)
    return a / np.sum(a)


def raised_cosine_init(n):
    """Raised cosine with wider main lobe."""
    x = np.arange(n) / max(1, n - 1) - 0.5
    a = np.maximum(0, np.cos(np.pi * x * 0.8)) ** 2
    a = np.maximum(a, 0.001)
    return a / np.sum(a)


def gaussian_init(n, sigma_frac=0.35):
    """Gaussian initialization."""
    x = np.arange(n) / max(1, n - 1) - 0.5
    a = np.exp(-x ** 2 / (2 * sigma_frac ** 2))
    return a / np.sum(a)


def uniform_init(n):
    return np.ones(n) / n


def two_hump_init(n):
    """Two-hump initialization."""
    x = np.arange(n) / max(1, n - 1) - 0.5
    a = np.exp(-(x - 0.25) ** 2 / 0.02) + np.exp(-(x + 0.25) ** 2 / 0.02)
    a = np.maximum(a, 0.001)
    return a / np.sum(a)


def random_init(n, rng):
    a = rng.random(n)
    a = np.maximum(a, 1e-6)
    return a / np.sum(a)


def smooth_noise_init(n, rng):
    """Smooth random initialization using sine components."""
    a = np.ones(n)
    for _ in range(5):
        freq = rng.uniform(0.5, 5.0)
        phase = rng.uniform(0, 2 * np.pi)
        amp = rng.uniform(0.3, 1.0)
        a += amp * np.sin(2 * np.pi * freq * np.arange(n) / n + phase)
    a = np.maximum(a, 0.001)
    return a / np.sum(a)


def propose_candidate(seed=42, budget_s=1500, **kwargs):
    """
    Hybrid optimization: smooth gradient + MWD + LP + SA + random refinement.
    Uses adaptive beta annealing and multiple initializations/lengths.
    """
    rng = np.random.default_rng(seed)
    start_time = time.monotonic()
    deadline = start_time + float(budget_s) - 5.0

    best_seq = None
    best_score = np.inf

    # Try parent sequence
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64)
        parent = np.clip(parent, 0, 1000.0)
        if parent.size > 0 and np.isfinite(parent).all():
            obj = compute_obj(parent)
            if obj < best_score:
                best_score = obj
                best_seq = parent.copy()
            print(f"Parent (n={parent.size}): obj={obj:.6f}")
    except Exception:
        pass

    # Sequence lengths to try
    n_values = [50, 80, 100, 130, 150, 200, 250, 350]

    # Add parent length if different
    if best_seq is not None and len(best_seq) not in n_values:
        n_values.append(len(best_seq))

    n_values = sorted(set(n_values))

    print(f"Trying sequence lengths: {n_values}")

    for n in n_values:
        if time.monotonic() > deadline:
            break
        remaining = deadline - time.monotonic()
        if remaining < 15:
            break
        print(f"\n{'='*40}\nN={n}: {remaining:.0f}s remaining")

        # Generate diverse initializations
        inits = [
            ("cosine", cosine_init(n)),
            ("raised_cos", raised_cosine_init(n)),
            ("gauss", gaussian_init(n)),
            ("two_hump", two_hump_init(n)),
            ("smooth_noise", smooth_noise_init(n, rng)),
            ("random", random_init(n, rng)),
            ("uniform", uniform_init(n)),
        ]

        # Add parent-based init if available
        if best_seq is not None:
            p = best_seq.copy()
            if len(p) == n:
                inits.insert(0, ("parent", p / np.sum(p)))
            else:
                # Interpolate parent to length n
                p_norm = p / np.sum(p)
                new_p = np.interp(np.linspace(0, 1, n), np.linspace(0, 1, len(p)), p_norm)
                new_p = np.maximum(new_p, 1e-6)
                inits.insert(0, ("parent_interp", new_p / np.sum(new_p)))

        for init_name, a_init in inits:
            if time.monotonic() > deadline:
                break
            remaining = deadline - time.monotonic()
            if remaining < 5:
                break

            a_curr = a_init.copy()
            init_obj = compute_obj(a_curr)

            # Step 1: Smooth gradient descent
            if remaining > 20:
                max_iter_sg = int(min(200, (remaining * 0.3) / 0.01))
                a_sg, obj_sg = run_smooth_gradient(
                    a_curr, beta_start=5.0, beta_end=150.0,
                    max_iter=max_iter_sg, lr0=0.05, lr_min=0.001
                )
                if obj_sg < best_score:
                    best_score = obj_sg
                    best_seq = a_sg.copy()
                    print(f"  [SG] {init_name}: obj={best_score:.6f}")
                a_curr = a_sg

            # Step 2: MWD
            if time.monotonic() > deadline:
                break
            remaining = deadline - time.monotonic()
            if remaining > 15:
                max_iter_mwd = int(min(300, (remaining * 0.2) / 0.005))
                a_mwd, obj_mwd = run_mwd(
                    a_curr, max_iter=max_iter_mwd,
                    eta0=0.08, eta_min=0.001, active_fraction=0.05
                )
                if obj_mwd < best_score:
                    best_score = obj_mwd
                    best_seq = a_mwd.copy()
                    print(f"  [MWD] {init_name}: obj={best_score:.6f}")
                a_curr = a_mwd

            # Step 3: LP polishing
            if time.monotonic() > deadline:
                break
            remaining = deadline - time.monotonic()
            if remaining > 30 and n <= 300:
                max_iter_lp = int(min(60, (remaining * 0.2) / 0.3))
                a_lp, obj_lp = run_lp_optimize(a_curr, max_iter=max_iter_lp, t=0.5)
                if obj_lp < best_score:
                    best_score = obj_lp
                    best_seq = a_lp.copy()
                    print(f"  [LP] {init_name}: obj={best_score:.6f}")
                a_curr = a_lp

            # Step 4: Random refinement
            if time.monotonic() > deadline:
                break
            remaining = deadline - time.monotonic()
            if remaining > 10:
                max_iter_r = int(min(2000, remaining / 0.005))
                a_r, obj_r = random_refine(a_curr, rng, max_iter=max_iter_r, noise_scale=0.06)
                if obj_r < best_score:
                    best_score = obj_r
                    best_seq = a_r.copy()
                    print(f"  [Rnd] {init_name}: obj={best_score:.6f}")

    # ---- Final refinement phase ----
    if best_seq is not None and time.monotonic() < deadline - 30:
        remaining = deadline - time.monotonic()
        n_best = len(best_seq)
        print(f"\n{'='*40}\nFINAL REFINEMENT: n={n_best}, best={best_score:.6f}, {remaining:.0f}s left")

        a_curr = best_seq.copy()

        # More aggressive smooth gradient with lower betas first
        if remaining > 40:
            max_iter = int(min(300, (remaining * 0.25) / 0.01))
            a1, obj1 = run_smooth_gradient(
                a_curr, beta_start=3.0, beta_end=100.0,
                max_iter=max_iter, lr0=0.03, lr_min=0.0005
            )
            if obj1 < best_score:
                best_score = obj1
                best_seq = a1.copy()
                print(f"  Final SG: obj={best_score:.6f}")
            a_curr = a1

        # MWD with varied active fraction
        if time.monotonic() < deadline - 30:
            remaining = deadline - time.monotonic()
            for af in [0.03, 0.08, 0.15]:
                if time.monotonic() > deadline:
                    break
                max_iter = int(min(300, (deadline - time.monotonic()) / 0.005))
                a2, obj2 = run_mwd(a_curr, max_iter=max_iter,
                                   eta0=0.04, eta_min=0.0005, active_fraction=af)
                if obj2 < best_score:
                    best_score = obj2
                    best_seq = a2.copy()
                    print(f"  Final MWD af={af}: obj={best_score:.6f}")
                a_curr = a2

        # LP polishing (if feasible)
        if time.monotonic() < deadline - 20 and n_best <= 300:
            remaining = deadline - time.monotonic()
            max_iter = int(min(50, remaining / 0.3))
            a3, obj3 = run_lp_optimize(a_curr, max_iter=max_iter, t=0.3)
            if obj3 < best_score:
                best_score = obj3
                best_seq = a3.copy()
                print(f"  Final LP: obj={best_score:.6f}")
            a_curr = a3

        # Simulated annealing
        if time.monotonic() < deadline - 15:
            remaining = deadline - time.monotonic()
            max_iter = int(min(3000, remaining / 0.005))
            a4, obj4 = simulated_annealing(a_curr, rng, max_iter=max_iter,
                                           T0=0.005, T_end=0.0005, noise_scale=0.12)
            if obj4 < best_score:
                best_score = obj4
                best_seq = a4.copy()
                print(f"  Final SA: obj={best_score:.6f}")

        # Final random refinement
        if time.monotonic() < deadline - 5:
            remaining = deadline - time.monotonic()
            max_iter = int(min(5000, remaining / 0.003))
            a5, obj5 = random_refine(best_seq, rng, max_iter=max_iter, noise_scale=0.04)
            if obj5 < best_score:
                best_score = obj5
                best_seq = a5.copy()
                print(f"  Final Rnd: obj={best_score:.6f}")

    # Fallback
    if best_seq is None:
        best_seq = cosine_init(150)
        best_score = compute_obj(best_seq)

    elapsed = time.monotonic() - start_time
    print(f"\n{'='*40}\nDONE: n={len(best_seq)}, score={best_score:.6f}, time={elapsed:.1f}s")
    return [float(x) for x in best_seq]
```
