def normalize_sequence(arr):
    """Return a nonnegative floating vector normalized to sum 1, or None."""
    try:
        x = np.asarray(arr, dtype=np.float64).ravel()
    except Exception:
        return None
    if x.size == 0:
        return None
    x = np.where(np.isfinite(x), x, 0.0)
    x = np.maximum(x, 0.0)
    s = float(x.sum())
    if s <= 1e-300:
        return None
    x /= s
    return x


def score_normalized(x, n):
    """Score assuming x is nonnegative and sum(x) = 1."""
    conv = np.convolve(x, x)
    m = float(np.max(conv))
    if not np.isfinite(m):
        return float("inf")
    return float(2.0 * n * m)


def make_conv_template(n):
    """Precompute the CSR matrix structure for convolve(f, g)."""
    m = 2 * n - 1
    k64 = np.arange(m, dtype=np.int64)
    js = np.maximum(0, k64 - n + 1).astype(np.int64)
    je = np.minimum(n - 1, k64).astype(np.int64)
    counts = (je - js + 1).astype(np.int64)
    total = int(counts.sum())

    cum = np.cumsum(counts)
    starts = np.repeat(cum - counts, counts)
    local = np.arange(total, dtype=np.int64) - starts

    row = np.repeat(k64, counts).astype(np.int32)
    col_base = js[row.astype(np.int64)]
    col = (col_base + local).astype(np.int32)
    offsets = (row.astype(np.int64) - col.astype(np.int64)).astype(np.int32)

    A_off = csr_matrix((offsets, (row, col)), shape=(m, n), dtype=np.int32)
    edge_offsets = A_off.data.astype(np.int32, copy=True)
    A_template = A_off.astype(np.float64).copy()
    A_template.data.fill(0.0)

    c_obj = -np.ones(n, dtype=np.float64)
    bounds = [(0.0, None)] * n
    return A_template, edge_offsets, c_obj, bounds, m


def solve_direction_lp(template, f, n, M):
    """Solve the LP direction problem for the given f and M."""
    A_template, edge_offsets, c_obj, bounds, m = template
    lp_scale = float(n)

    A = A_template.copy()
    A.data[:] = f[edge_offsets]
    A.data *= lp_scale

    b = np.full(m, M * lp_scale * lp_scale, dtype=np.float64)
    return linprog(c_obj, A_ub=A, b_ub=b, bounds=bounds, method="highs")

The interpolation argument: consider h(s) = f + s*(f_new - f) = (1-s)*f + s*f_new for s in [0,1].
- h(0) = f, h(1) = f_new
- max(h*h) at s=0 is M
- We want to show that for small s > 0, max(h*h) < M

(h*h)_k = ((1-s)f + s*f_new)_k * ((1-s)f + s*f_new)_k (convolution)
= (1-s)^2 (f*f)_k + 2s(1-s) (f*f_new)_k + s^2 (f_new*f_new)_k

At s=0: (h*h)_k = (f*f)_k, so max = M.
The derivative at s=0: d/ds (h*h)_k |_{s=0} = 2((f*f_new)_k - (f*f)_k)

For the active constraint k* where (f*f)_k* = M:
d/ds (h*h)_k* |_{s=0} = 2((f*f_new)_k* - M)

Since (f*f_new)_k* <= t < M (from the LP), the derivative is negative. So for small s > 0, (h*h)_k* < M.

But we also need to check other k values. For k where (f*f)_k < M, the increase might be bounded.

This gives a valid descent direction! The step size can be determined by the safe bound:
(h*h)_k <= (1-s)^2 (f*f)_k + 2s(1-s) t + s^2 * G where G = max(f_new*f_new)

We need: (1-s)^2 M + 2s(1-s) t + s^2 G <= M
=> -2sM + s^2 M + 2st - 2s^2 t + s^2 G <= 0
=> s(-2M + 2t) + s^2(M - 2t + G) <= 0
=> s[2(t-M) + s(M + G - 2t)] <= 0

For s > 0: 2(t-M) + s(M + G - 2t) <= 0
=> s <= 2(M-t) / (M + G - 2t) (if denominator > 0)

LP-based search for nonnegative step functions minimizing 2*n*max(convolve(f,f))/(sum(f))^2.
Uses direct minimax LP formulation with safe stepping, multiple lengths, and gradient polishing.
"""

import time
import numpy as np
from scipy.optimize import linprog
from scipy.sparse import csr_matrix, csc_matrix


def normalize_sequence(arr):
    x = np.asarray(arr, dtype=np.float64).ravel()
    if x.size == 0:
        return None
    x = np.where(np.isfinite(x), x, 0.0)
    x = np.maximum(x, 0.0)
    s = float(x.sum())
    if s <= 1e-300:
        return None
    return x / s


def score_normalized(x, n):
    conv = np.convolve(x, x)
    m = float(np.max(conv))
    if not np.isfinite(m):
        return float("inf")
    return float(2.0 * n * m)


def build_template(n):
    """Build sparse template for convolution matrix."""
    m = 2 * n - 1
    k = np.arange(m, dtype=np.int64)
    js = np.maximum(0, k - n + 1)
    je = np.minimum(n - 1, k)
    counts = je - js + 1
    total = int(counts.sum())
    
    cum = np.cumsum(counts)
    starts = np.repeat(cum - counts, counts)
    local = np.arange(total, dtype=np.int64) - starts
    
    row = np.repeat(k, counts)
    col_base = js[row]
    col = col_base + local
    offsets = (row - col).astype(np.int32)
    
    A = csr_matrix((offsets, (row, col)), shape=(m, n), dtype=np.int32)
    edge_offsets = A.data.astype(np.int32, copy=True)
    A_float = A.astype(np.float64).copy()
    A_float.data.fill(0.0)
    
    return A_float, edge_offsets, m


def solve_minimax_lp(f, n, A_template, edge_offsets, m):
    """
    Solve: min t
           s.t. (f_new * f)_k <= t for all k
                sum(f_new) = 1
                f_new >= 0
    """
    # Build constraint matrix
    A = A_template.copy()
    A.data[:] = f[edge_offsets]
    
    # For linprog with sparse A_ub, we need to add the -t column
    # Let's build a dense matrix for simplicity (n=1024 -> 2047 x 1025)
    A_dense = A.toarray()
    A_full = np.column_stack([A_dense, -np.ones(m)])
    
    c = np.zeros(n + 1)
    c[n] = 1.0
    
    bounds = [(0.0, None)] * n + [(None, None)]
    
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])
    
    res = linprog(c, A_ub=A_full, b_ub=np.zeros(m), A_eq=A_eq, b_eq=b_eq,
                  bounds=bounds, method='highs')
    
    if not res.success:
        return None, None
    return res.x[:n], res.x[n]


def solve_maxsum_lp(f, n, A_template, edge_offsets, m, M):
    """
    Parent's LP: max sum(g) s.t. (f*g)_k <= M, g >= 0
    """
    A = A_template.copy()
    A.data[:] = f[edge_offsets]
    
    c_obj = -np.ones(n)
    bounds = [(0.0, None)] * n
    b_ub = np.full(m, M)
    
    res = linprog(c_obj, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
    
    if not res.success:
        return None
    return res.x

    """Perturb f by mixing with a random Dirichlet vector."""
    n = f.size
    d = rng.dirichlet(np.ones(n))
    f_new = (1 - alpha) * f + alpha * d
    return normalize_sequence(f_new)

    if f is None:

"""
LP-based search minimizing 2*n*max(convolve(f,f))/(sum(f))^2.
Uses direct minimax LP, max-sum LP, safe stepping, and multi-length search.
"""

import time
import numpy as np
from scipy.optimize import linprog
from scipy.sparse import csr_matrix, csc_matrix, hstack


def normalize_sequence(arr):
    """Normalize to nonnegative, sum 1."""
    try:
        x = np.asarray(arr, dtype=np.float64).ravel()
    except Exception:
        return None
    if x.size == 0:
        return None
    x = np.where(np.isfinite(x), x, 0.0)
    x = np.maximum(x, 0.0)
    s = float(x.sum())
    if s <= 1e-300:
        return None
    return x / s


def score_normalized(x, n):
    """Score for normalized (sum=1) nonnegative sequence."""
    conv = np.convolve(x, x)
    m = float(np.max(conv))
    if not np.isfinite(m):
        return float("inf")
    return float(2.0 * n * m)


def build_template(n):
    """Build sparse convolution template."""
    m = 2 * n - 1
    k = np.arange(m, dtype=np.int64)
    js = np.maximum(0, k - n + 1)
    je = np.minimum(n - 1, k)
    counts = je - js + 1
    total = int(counts.sum())
    
    cum = np.cumsum(counts)
    starts = np.repeat(cum - counts, counts)
    local = np.arange(total, dtype=np.int64) - starts
    
    row = np.repeat(k, counts)
    col = js[row] + local
    offsets = (row - col).astype(np.int32)
    
    A = csr_matrix((offsets, (row, col)), shape=(m, n), dtype=np.int32)
    edge_offsets = A.data.astype(np.int32, copy=True)
    A_float = A.astype(np.float64).copy()
    A_float.data.fill(0.0)
    
    return A_float, edge_offsets, m


def solve_minimax_lp_sparse(f, n, A_template, edge_offsets, m):
    """Min t s.t. (f_new*f)_k <= t, sum(f_new)=1, f_new>=0."""
    A = A_template.copy()
    A.data[:] = f[edge_offsets]
    
    # Add -t column
    neg_col = csc_matrix(-np.ones((m, 1)))
    A_full = hstack([A, neg_col], format='csc')
    
    c = np.zeros(n + 1)
    c[n] = 1.0
    
    bounds = [(0.0, None)] * n + [(None, None)]
    
    A_eq_row = np.zeros(n + 1)
    A_eq_row[:n] = 1.0
    A_eq = A_eq_row.reshape(1, -1)
    b_eq = np.array([1.0])
    
    res = linprog(c, A_ub=A_full, b_ub=np.zeros(m), A_eq=A_eq, b_eq=b_eq,
                  bounds=bounds, method='highs')
    
    if not res.success:
        return None, None
    f_new = np.maximum(res.x[:n], 0)
    s = f_new.sum()
    if s < 1e-300:
        return None, None
    f_new /= s
    return f_new, float(res.x[n])


def solve_maxsum_lp(f, n, A_template, edge_offsets, m, M):
    """Max sum(g) s.t. (f*g)_k <= M, g>=0."""
    A = A_template.copy()
    A.data[:] = f[edge_offsets]
    
    c_obj = -np.ones(n)
    bounds = [(0.0, None)] * n
    b_ub = np.full(m, M)
    
    res = linprog(c_obj, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
    
    if not res.success:
        return None
    g = np.maximum(res.x, 0)
    return g


def perturb(f, rng, alpha):
    """Perturb f with random Dirichlet."""
    n = f.size
    try:
        d = rng.dirichlet(np.ones(n))
    except Exception:
        d = np.ones(n) / n
    f_new = (1 - alpha) * f + alpha * d
    return normalize_sequence(f_new)


def interpolate_seq(x, n_target):
    """Resample to target length."""
    x = normalize_sequence(x)
    if x is None or n_target <= 0:
        return None
    m = x.size
    if m == n_target:
        return x
    if m == 1:
        return np.ones(n_target) / n_target
    old = np.linspace(0, 1, m)
    new = np.linspace(0, 1, n_target)
    out = np.interp(new, old, x)
    return normalize_sequence(out)

        else:
            s_max = 1.0
        s_max = min(1.0, max(s_max, 1e-6))
        
        # Line search
        step_candidates = sorted(
            set(min(1.0, max(1e-6, v)) for v in
                [s_max, 0.9*s_max, 0.7*s_max, 0.5*s_max, 0.3*s_max, 0.1*s_max, 0.05*s_max]),
            reverse=True
        )
        
        improved = False
        for s in step_candidates:
            f_trial = (1 - s) * f + s * f_new
            f_trial = normalize_sequence(f_trial)
            if f_trial is None:

            sc = score_normalized(f_trial, n)
            if np.isfinite(sc) and sc < best_score * (1 - 1e-10):
                f = f_trial
                best_f = f.copy()
                best_score = sc
                improved = True
                break

            if stall > max_stall:
                break
            f = perturb(best_f, rng, min(0.1 + 0.03 * stall, 0.6))
