"""
Multi-stage optimization for minimizing 2n*max(conv(a,a))/sum(a)^2.
Combines proximal gradient, LP-based direction finding, and parameterized
initializations to escape local minima and refine the solution.
"""
import numpy as np
import time
from scipy.optimize import linprog
from scipy.sparse import csr_matrix


def evaluate_quick(a):
    """Fast evaluation without the security checks."""
    n = len(a)
    conv = np.convolve(a, a)
    max_b = np.max(conv)
    sum_a = np.sum(a)
    if sum_a < 0.01:
        return np.inf
    return 2.0 * n * max_b / (sum_a ** 2)


def build_conv_matrix_sparse(f):
    """Build sparse matrix A where (A@g)[k] = convolve(f,g)[k]."""
    n = len(f)
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        j_min = max(0, k - (n - 1))
        j_max = min(k, n - 1)
        for j in range(j_min, j_max + 1):
            rows.append(k)
            cols.append(j)
            vals.append(f[k - j])
    return csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))


def solve_lp_direction(f, M, time_limit=10.0):
    """Solve: max sum(g) s.t. g >= 0, convolve(f,g)_k <= M for all k."""
    n = len(f)
    A = build_conv_matrix_sparse(f)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    try:
        result = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                         options={'time_limit': time_limit, 'presolve': True})
        if result.success:
            return result.x
    except Exception:
        pass
    return None


def proximal_gradient(f, deadline, lr_init=0.01, patience=200):
    """
    Proximal gradient method to minimize max_k conv(f,f)_k subject to
    sum(f) = 1, f >= 0.
    """
    f = f.copy().astype(np.float64)
    # Normalize to sum = 1
    S = np.sum(f)
    if S <= 0:
        return f, np.inf
    f = f / S
    n = len(f)
    
    best_f = f.copy()
    best_max_b = np.max(np.convolve(f, f))
    lr = lr_init
    no_improve = 0
    it = 0
    
    while time.monotonic() < deadline and no_improve < patience:
        conv_f = np.convolve(f, f)
        max_idx = int(np.argmax(conv_f))
        M = conv_f[max_idx]
        
        # Subgradient of max_k conv_k at the maximizer:
        # d/d f[j] of conv[max_idx] = 2 * f[max_idx - j] if 0 <= max_idx-j < n
        grad = np.zeros(n)
        for j in range(n):
            k2 = max_idx - j
            if 0 <= k2 < n:
                grad[j] = 2.0 * f[k2]
        
        # Gradient step
        new_f = f - lr * grad
        
        # Project onto simplex {x >= 0, sum(x) = 1}
        new_f = project_simplex(new_f)
        
        new_conv = np.convolve(new_f, new_f)
        new_max = float(np.max(new_conv))
        
        it += 1
        if new_max < best_max_b:
            best_max_b = new_max
            best_f = new_f.copy()
            f = new_f
            no_improve = 0
        else:
            no_improve += 1
            if no_improve % 20 == 0:
                lr *= 0.8
                if lr < 1e-8:

    # Rescale to any convenient scale (objective is scale-invariant)
    return best_f, evaluate_quick(best_f.tolist())


def project_simplex(v, s=1.0):
    """Project v onto {x >= 0, sum(x) = s}."""
    n = len(v)
    # Binary search for the threshold
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.where(u * np.arange(1, n + 1) > (cssv - s))[0][-1]
    theta = (cssv[rho] - s) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def lp_iterate(f, deadline, step_size_init=0.1, max_lp_time=5.0):
    """Run LP-based iterations on f (normalized to sum=1)."""

    S = np.sum(f)

        return f, np.inf

    n = len(f)

    no_improve = 0
    it = 0

        conv_f = np.convolve(f, f)
        M = float(np.max(conv_f))
        if M <= 0:

        if g is None:
            no_improve += 1
            if no_improve > 20:

        sum_g = float(np.sum(g))
        if sum_g <= 1e-15:
            no_improve += 1

        # Rescale g to have same sum as f (which is 1)
        g = g / sum_g
        
        # Blend with line search
        found = False
        for t in [step_size, step_size*0.5, step_size*0.25, step_size*0.1, step_size*0.05]:

def generate_initializations(n, rng):
    """Generate diverse initial sequences of length n."""
    inits = []
    idx = np.arange(n, dtype=np.float64)
    
    # Uniform
    inits.append(('uniform', np.ones(n)))
    
    # Cosine taper
    cos_t = np.sin(np.pi * (idx + 0.5) / n) ** 2
    inits.append(('sin2', cos_t))
    
    # Raised cosine
    rc = 0.5 * (1 + np.cos(2 * np.pi * idx / n))
    inits.append(('raised_cos', rc))
    
    # Power law taper
    for p in [0.5, 1.0, 2.0, 3.0]:
        x = np.abs(idx - (n-1)/2.0) / ((n-1)/2.0)
        tl = (1 - x**p)
        tl = np.maximum(tl, 0.01)
        inits.append((f'taper_p{p}', tl))
    
    # Gaussian
    center = (n - 1) / 2.0
    for sigma_frac in [0.2, 0.3, 0.4, 0.5]:
        g = np.exp(-0.5 * ((idx - center) / (n * sigma_frac)) ** 2)
        inits.append((f'gauss_{sigma_frac}', g))
    
    # Bimodal
    two_peak = (np.exp(-0.5 * ((idx - n * 0.25) / (n * 0.1)) ** 2) +
                np.exp(-0.5 * ((idx - n * 0.75) / (n * 0.1)) ** 2))
    inits.append(('bimodal', two_peak))
    
    # Plateau
    plateau = np.ones(n)
    edge = max(1, n // 10)
    ramp = np.linspace(0.1, 1.0, edge)
    plateau[:edge] = ramp
    plateau[-edge:] = ramp[::-1]
    inits.append(('plateau', plateau))
    
    # Random
    for seed in [1, 2, 3]:
        r = rng.uniform(0.1, 1.0, size=n)
        inits.append((f'rand_{seed}', r))
    
    # Random binary
    for seed in [4, 5]:
        b = (rng.random(n) < 0.5).astype(float)
        # Ensure at least some are 1
        if b.sum() < n * 0.1:
            b = (rng.random(n) < 0.7).astype(float)
        inits.append((f'bin_{seed}', b))
    
    # Smoothed binary (binary convolved with small kernel)
    kernel = np.ones(5) / 5.0
    b = (rng.random(n) < 0.6).astype(float)
    sb = np.convolve(b, kernel, mode='same')
    inits.append(('smooth_bin', sb))
    
    return inits

"""
Multi-stage optimization for minimizing 2n*max(conv(a,a))/sum(a)^2.
Combines proximal gradient with simplex projection, LP-based direction finding,
diverse parameterized initializations, and perturbation search to escape local minima.
Key improvements over prior work: proper simplex projection in PG, adaptive
line search in LP blending, strategic time allocation, and perturbation-based
local search in the final phase.
"""
import numpy as np
import time
from scipy.optimize import linprog
from scipy.sparse import csr_matrix


def evaluate_quick(a):
    """Fast evaluation without security checks."""
    n = len(a)
    if n == 0:
        return np.inf
    conv = np.convolve(a, a)
    max_b = float(np.max(conv))
    sum_a = float(np.sum(a))
    if sum_a < 0.01:
        return np.inf
    return 2.0 * n * max_b / (sum_a ** 2)


def project_simplex(v, s=1.0):
    """Project v onto {x >= 0, sum(x) = s}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - s)
    if not np.any(cond):
        return np.full(n, s / n)
    rho = np.where(cond)[0][-1]
    theta = (cssv[rho] - s) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def build_conv_matrix_sparse(f):
    """Build sparse matrix A where (A@g)[k] = convolve(f,g)[k]."""
    n = len(f)
    rows = np.empty((2 * n - 1) * n, dtype=np.int32)
    cols = np.empty((2 * n - 1) * n, dtype=np.int32)
    vals = np.empty((2 * n - 1) * n, dtype=np.float64)
    idx = 0
    for k in range(2 * n - 1):
        j_min = max(0, k - (n - 1))
        j_max = min(k, n - 1)
        for j in range(j_min, j_max + 1):
            rows[idx] = k
            cols[idx] = j
            vals[idx] = f[k - j]
            idx += 1
    return csr_matrix((vals[:idx], (rows[:idx], cols[:idx])), shape=(2 * n - 1, n))


def solve_lp_direction(f, M, time_limit=3.0):
    """Solve: max sum(g) s.t. g >= 0, convolve(f,g)_k <= M for all k."""
    n = len(f)
    A = build_conv_matrix_sparse(f)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    try:
        result = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                         options={'time_limit': time_limit, 'presolve': True})
        if result.success:
            return result.x
    except Exception:
        pass
    return None


def proximal_gradient(f, deadline, lr_init=0.05, patience=300):
    """
    Proximal gradient to minimize max_k conv(f,f)_k s.t. sum(f)=1, f>=0.
    Returns (best_f_normalized_to_sum1, score).
    """
    f = f.copy().astype(np.float64)
    S = float(np.sum(f))
    if S <= 0:
        return f, np.inf
    f = f / S
    n = len(f)

    best_f = f.copy()
    best_max_b = float(np.max(np.convolve(f, f)))
    lr = lr_init
    no_improve = 0
    it = 0

    while time.monotonic() < deadline and no_improve < patience:
        conv_f = np.convolve(f, f)
        max_idx = int(np.argmax(conv_f))
        M = conv_f[max_idx]

        # Subgradient: d/d f[j] of conv[max_idx] = 2*f[max_idx-j] if valid
        grad = np.zeros(n)
        for j in range(n):
            k2 = max_idx - j
            if 0 <= k2 < n:
                grad[j] = 2.0 * f[k2]

        new_f = f - lr * grad
        new_f = project_simplex(new_f, 1.0)

        new_conv = np.convolve(new_f, new_f)
        new_max = float(np.max(new_conv))

        it += 1
        if new_max < best_max_b - 1e-12:
            best_max_b = new_max
            best_f = new_f.copy()
            f = new_f
            no_improve = 0
        else:
            no_improve += 1
            if no_improve % 30 == 0:
                lr *= 0.7
                if lr < 1e-10:
                    break

    return best_f, 2.0 * n * best_max_b


def lp_iterate(f, deadline, step_size_init=0.12, max_lp_time=3.0):
    """LP-based refinement on f (will normalize internally)."""
    f = f.copy().astype(np.float64)
    S = float(np.sum(f))
    if S <= 0:
        return f, np.inf
    f = f / S
    n = len(f)

    best_f = f.copy()
    best_score = 2.0 * n * float(np.max(np.convolve(f, f)))
    step_size = step_size_init
    no_improve = 0
    it = 0

    while time.monotonic() < deadline and no_improve < 100:
        conv_f = np.convolve(f, f)
        M = float(np.max(conv_f))
        if M <= 0:
            break

        g = solve_lp_direction(f, M, time_limit=max_lp_time)
        if g is None:
            no_improve += 5
            if no_improve > 30:
                break
            continue

        sum_g = float(np.sum(g))
        if sum_g <= 1e-15:
            no_improve += 1
            continue

        g = g / sum_g

        # Line search for optimal step
        found = False
        for t in [step_size, step_size * 0.6, step_size * 0.3, step_size * 0.1, step_size * 0.05]:
            new_f = (1.0 - t) * f + t * g
            new_f = np.clip(new_f, 0.0, None)
            ns = float(np.sum(new_f))
            if ns < 1e-10:
                continue
            new_f = new_f / ns
            new_score = 2.0 * n * float(np.max(np.convolve(new_f, new_f)))
            if new_score < best_score - 1e-10:
                f = new_f
                best_f = f.copy()
                best_score = new_score
                no_improve = 0
                it += 1
                found = True
                break

        if not found:
            no_improve += 1
            step_size *= 0.6
            if step_size < 0.003:
                break

        if it % 30 == 0:
            print(f"    LP iter {it}: score={best_score:.6f}, M={M:.8f}, step={step_size:.4f}")

    return best_f, best_score


def generate_initializations(n, rng):
    """Generate diverse initial sequences of length n."""
    inits = []
    idx = np.arange(n, dtype=np.float64)

    inits.append(('uniform', np.ones(n)))

    cos_t = np.sin(np.pi * (idx + 0.5) / n) ** 2
    inits.append(('sin2', cos_t))

    rc = 0.5 * (1.0 + np.cos(2.0 * np.pi * idx / n))
    inits.append(('raised_cos', rc))

    center = (n - 1) / 2.0
    for p in [1.0, 2.0, 3.0, 5.0]:
        x = np.abs(idx - center) / center
        tl = np.power(1.0 - x, p)
        tl = np.maximum(tl, 0.005)
        inits.append((f'taper_p{p}', tl))

    for sf in [0.15, 0.25, 0.35, 0.5]:
        g = np.exp(-0.5 * ((idx - center) / (n * sf)) ** 2)
        inits.append((f'gauss_{sf}', g))

    two_peak = (np.exp(-0.5 * ((idx - n * 0.25) / (n * 0.12)) ** 2) +
                np.exp(-0.5 * ((idx - n * 0.75) / (n * 0.12)) ** 2))
    inits.append(('bimodal', two_peak))

    plateau = np.ones(n)
    edge = max(2, n // 8)
    ramp = np.linspace(0.05, 1.0, edge)
    plateau[:edge] = ramp
    plateau[-edge:] = ramp[::-1]
    inits.append(('plateau', plateau))

    for seed in [0, 1, 2]:
        r = rng.uniform(0.05, 1.0, size=n)
        inits.append((f'rand_{seed}', r))

    for seed in [3, 4]:
        b = (rng.random(n) < 0.5).astype(np.float64)
        if b.sum() < n * 0.1:
            b = (rng.random(n) < 0.7).astype(np.float64)
        inits.append((f'bin_{seed}', b))

    return inits


def optimize_candidate(f_init, deadline, phase_time_frac):
    """Run full optimization pipeline on one candidate."""
    t_start = time.monotonic()
    alloc = (deadline - t_start) * phase_time_frac
    alloc = max(20.0, alloc)

    n = len(f_init)
    f = f_init.copy().astype(np.float64)
    best_f = f.copy()
    best_score = 2.0 * n * float(np.max(np.convolve(f, f)))

    # Stage 1: Proximal gradient (fast, many iterations)
    pg_deadline = min(deadline, time.monotonic() + alloc * 0.25)
    pg_f, pg_score = proximal_gradient(f, pg_deadline, lr_init=0.04)
    if pg_score < best_score:
        best_score = pg_score
        best_f = pg_f.copy()
        f = pg_f.copy()

    # Stage 2: LP refinement
    lp_deadline = min(deadline, time.monotonic() + alloc * 0.55)
    lp_f, lp_score = lp_iterate(f, lp_deadline, step_size_init=0.12, max_lp_time=3.0)
    if lp_score < best_score:
        best_score = lp_score
        best_f = lp_f.copy()
        f = lp_f.copy()

    # Stage 3: PG polish on LP result
    pg2_deadline = min(deadline, time.monotonic() + alloc * 0.2)
    pg2_f, pg2_score = proximal_gradient(f, pg2_deadline, lr_init=0.01)
    if pg2_score < best_score:
        best_score = pg2_score
        best_f = pg2_f.copy()

    return best_f, best_score
