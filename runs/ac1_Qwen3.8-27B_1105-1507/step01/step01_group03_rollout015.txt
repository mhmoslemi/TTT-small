"""
Hybrid LP + smooth-gradient optimization for minimizing the peak
self-convolution of a nonnegative sequence.

Algorithm:
1. LP-based block coordinate descent (similar to previous SOTA) for
   large-scale optimization.
2. Smooth-max projected gradient descent with beta annealing for
   fine-tuning.
3. Local perturbation search (frequency-domain and time-domain) to
   escape local minima.
"""

import time
import numpy as np
from scipy.optimize import linprog
from scipy.sparse import csr_matrix


def normalize_sequence(arr):
    """Normalize to nonnegative vector with sum 1."""
    try:
        x = np.asarray(arr, dtype=np.float64).ravel()
    except Exception:
        return None
    if x.size == 0:
        return None
    x = np.where(np.isfinite(x), x, 0.0)
    x = np.maximum(x, 0.0)
    s = float(x.sum())
    if s <= 1e-300:
        return None
    x /= s
    return x


def score_normalized(x, n):
    """Score = 2n * max(conv(x,x)) assuming sum(x)=1."""
    conv = np.convolve(x, x)
    m = float(np.max(conv))
    if not np.isfinite(m):
        return float("inf")
    return float(2.0 * n * m)


def mix_noisy(base, rng, eps):
    """Mix base with random Dirichlet."""
    f = normalize_sequence(base)
    if f is None:
        return None
    n = f.size
    if eps <= 1e-12:
        return f
    try:
        mix = rng.dirichlet(np.ones(n, dtype=np.float64))
    except Exception:
        mix = np.ones(n, dtype=np.float64) / float(n)
    if eps >= 1.0:
        f = mix
    else:
        f = (1.0 - eps) * f + eps * mix
    return normalize_sequence(f)


def freq_perturb(p, rng, strength):
    """Apply random phase perturbation in frequency domain."""
    n = len(p)
    F = np.fft.rfft(p)
    # Random phase
    phase = rng.uniform(0, 2 * np.pi, size=len(F))
    # Blend: preserve some of the original phase
    blended = (1 - strength) * F + strength * np.abs(F) * np.exp(1j * phase)
    p_new = np.fft.irfft(blended, n=n)
    p_new = np.maximum(p_new, 0.0)
    s = p_new.sum()
    if s < 1e-300:
        return None
    p_new /= s
    return p_new


def interpolate_sequence(x, n):
    """Resample to length n and normalize."""
    x = normalize_sequence(x)
    if x is None or n <= 0:
        return None
    m = x.size
    if m == n:
        return x
    if m == 1:
        return np.ones(n, dtype=np.float64) / float(n)
    old = np.linspace(0.0, 1.0, m, dtype=np.float64)
    new = np.linspace(0.0, 1.0, n, dtype=np.float64)
    out = np.interp(new, old, x)
    return normalize_sequence(out)


def make_conv_template(n):
    """Precompute CSR template for LP."""
    m = 2 * n - 1
    k64 = np.arange(m, dtype=np.int64)
    js = np.maximum(0, k64 - n + 1).astype(np.int32)
    je = np.minimum(n - 1, k64).astype(np.int32)
    counts = (je.astype(np.int64) - js.astype(np.int64) + 1).astype(np.int64)
    total = int(counts.sum())

    cum = np.cumsum(counts)
    starts = np.repeat(cum - counts, counts)
    local = np.arange(total, dtype=np.int64) - starts

    row = np.repeat(k64, counts).astype(np.int32)
    col_base = js[row.astype(np.int64)]
    col = (col_base + local).astype(np.int32)
    offsets = (row.astype(np.int64) - col.astype(np.int64)).astype(np.int32)

    A_off = csr_matrix((offsets, (row, col)), shape=(m, n), dtype=np.int32)
    edge_offsets = A_off.data.astype(np.int32, copy=True)
    A_template = A_off.astype(np.float64).copy()
    A_template.data.fill(0.0)

    c_obj = -np.ones(n, dtype=np.float64)
    bounds = [(0.0, None)] * n
    return A_template, edge_offsets, c_obj, bounds, m


def solve_direction_lp(template, f, n, M):
    """Solve the LP for the descent direction."""
    A_template, edge_offsets, c_obj, bounds, m = template
    lp_scale = float(n)

    A = A_template.copy()
    A.data[:] = f[edge_offsets]
    A.data *= lp_scale

    b = np.full(m, M * lp_scale * lp_scale, dtype=np.float64)
    return linprog(c_obj, A_ub=A, b_ub=b, bounds=bounds, method="highs")

        g = x / s_star

        r = np.convolve(f, g)
        r_max = float(np.max(r))
        if (not np.isfinite(r_max)) or r_max <= 0.0:
            stall += 1
            if stall > 10:
                break
            nf = freq_perturb(best_f, rng, 0.3)
            if nf is None:
                nf = mix_noisy(best_f, rng, 0.3)
            f = nf if nf is not None else best_f.copy()
            continue

        if r_max >= M * (1.0 - 1e-10):
            stall += 1
            if stall > 10:
                break
            nf = freq_perturb(best_f, rng, 0.3)
            if nf is None:
                nf = mix_noisy(best_f, rng, 0.3)
            f = nf if nf is not None else best_f.copy()
            continue

        G = float(np.max(np.convolve(g, g)))
        if not np.isfinite(G):
            stall += 1
            if stall > 10:
                break
            nf = freq_perturb(best_f, rng, 0.3)
            if nf is None:
                nf = mix_noisy(best_f, rng, 0.3)
            f = nf if nf is not None else best_f.copy()
            continue

        # Safe step bound
        a_coef = 2.0 * (r_max - M)
        b_coef = G + M - 2.0 * r_max
        if b_coef > 1e-14:
            t_max = -a_coef / b_coef
        else:
            t_max = 1.0

        if (not np.isfinite(t_max)) or t_max <= 0.0:
            t_max = 1e-8
        base = min(1.0, 0.75 * t_max)
        if base < 1e-12:
            base = 1e-12

        # Line search
        ts = sorted({base, 0.7 * base, 0.4 * base, 0.2 * base,
                      0.1 * base, 0.05 * base, base * 1.2, base * 1.5},
                    reverse=True)
        ts = [t for t in ts if 0 < t <= 1.0]
        if not ts:
            ts = [1e-6]

        best_cand_f = None
        best_cand_score = best_score
        best_t = None

        for tv in ts:
            f_new = (1.0 - tv) * f + tv * g
            f_new = np.maximum(f_new, 0.0)
            ssum = float(f_new.sum())
            if ssum <= 1e-300:
                continue
            f_new /= ssum
            sc = score_normalized(f_new, n)
            if np.isfinite(sc) and sc < best_cand_score:
                best_cand_score = sc
                best_cand_f = f_new
                best_t = tv

        if best_cand_f is not None and best_cand_score < best_score:
            f = best_cand_f
            best_f = f.copy()
            best_score = best_cand_score
            stall = 0
            if iteration <= 5 or iteration % 10 == 0:
                print(f"      n={n} iter={iteration} score={best_score:.6f} "
                      f"t={best_t:.6f} dt={time.monotonic()-t0:.2f}s", flush=True)
        else:
            stall += 1
            if stall % 3 == 0:
                # Try different perturbation strategies
                if stall % 9 == 0:
                    nf = freq_perturb(best_f, rng, 0.5)
                elif stall % 6 == 0:
                    nf = mix_noisy(best_f, rng, 0.5)
                else:
                    nf = freq_perturb(best_f, rng, 0.2)
                if nf is None:
                    nf = mix_noisy(best_f, rng, 0.2)
                f = nf if nf is not None else best_f.copy()
            if stall > 15:
                break

    return best_f, float(best_score)


def optimize_gradient(n, f_init, time_budget, rng):
    """Smooth-max gradient descent with beta annealing."""
    f = normalize_sequence(f_init)
    if f is None:
        f = np.ones(n, dtype=np.float64) / float(n)

    best_f = f.copy()
    best_score = score_normalized(f, n)

    deadline = time.monotonic() + max(0.0, float(time_budget))
    
    # Beta schedule: start moderate, increase
    total_iters_est = int(time_budget * 50)  # rough estimate
    current_iter = 0
    
    # Starting beta and step size
    beta = 5.0
    alpha = 1.0 / n  # initial step size
    
    stall = 0
    best_beta = beta
    best_alpha = alpha
    
    while time.monotonic() < deadline:
        current_iter += 1
        
        # Update beta (annealing): increase over time
        progress = min(1.0, current_iter / max(1, total_iters_est * 0.8))
        beta = 5.0 + 40.0 * progress * progress  # quadratic increase up to 45
        
        # Compute convolution
        conv = np.convolve(f, f)
        conv_max = float(np.max(conv))
        
        # Compute smooth max weights
        shifted = conv - conv_max
        # Clip to prevent overflow
        max_shift = shifted.max()
        if beta * max_shift > 500:
            shifted = shifted - max_shift * (beta * max_shift - 500) / (beta * max_shift)
        exp_vals = np.exp(beta * shifted)
        w = exp_vals / exp_vals.sum()
        
        # Compute gradient using correlate
        grad = 2.0 * np.correlate(w, f, mode='valid')
        
        # Adaptive step size (use gradient norm)
        grad_norm = np.linalg.norm(grad)
        if grad_norm < 1e-15:
            stall += 1
        else:
            # Target: reduce smooth max by a fixed fraction
            # Rough estimate: alpha * grad_norm * something
            alpha = min(1.0, 2.0 / (n * grad_norm + 1e-10))
            
            # Backtracking line search
            f_candidate = f.copy()
            alpha_try = alpha
            success = False
            for _ in range(10):
                f_candidate = f - alpha_try * grad
                f_candidate = np.maximum(f_candidate, 0.0)
                s = f_candidate.sum()
                if s > 1e-300:
                    f_candidate /= s
                
                # Evaluate
                conv_c = np.convolve(f_candidate, f_candidate)
                max_c = float(np.max(conv_c))
                
                if max_c < conv_max * (1.0 - 1e-6):
                    success = True
                    break
                
                alpha_try *= 0.5
            
            if success:
                f = f_candidate
                stall = 0
                sc = score_normalized(f, n)
                if sc < best_score:
                    best_score = sc
                    best_f = f.copy()
            else:
                stall += 1
        
        if stall > 20:
            # Restart with perturbation
            nf = freq_perturb(f, rng, 0.2)
            if nf is None:
                nf = mix_noisy(f, rng, 0.2)
            if nf is not None:
                f = nf
            stall = 0
            beta = max(5.0, beta * 0.5)  # reduce beta after restart
        
        if current_iter % 100 == 0:
            print(f"      grad n={n} iter={current_iter} beta={beta:.1f} "
                  f"score={best_score:.6f} stall={stall}", flush=True)
    
    return best_f, float(best_score)


def local_search(n, f_init, time_budget, rng):
    """Local perturbation search."""
    f = normalize_sequence(f_init)
    if f is None:
        f = np.ones(n, dtype=np.float64) / float(n)
    
    best_f = f.copy()
    best_score = score_normalized(f, n)
    
    deadline = time.monotonic() + max(0.0, float(time_budget))
    temperature = 0.01  # initial perturbation strength
    
    while time.monotonic() < deadline:
        # Generate perturbation
        r = rng.random()
        if r < 0.4:
            # Small Gaussian perturbation
            noise = rng.normal(0, temperature, n)
            f_try = f + noise
        elif r < 0.7:
            # Frequency domain perturbation
            f_try = freq_perturb(f, rng, temperature * 2)
        else:
            # Coordinate swap
            i = rng.integers(0, n)
            j = rng.integers(0, n)
            f_try = f.copy()
            delta = (f_try[i] - f_try[j]) * 0.5 * temperature * 10
            f_try[i] -= delta
            f_try[j] += delta
        
        f_try = np.maximum(f_try, 0.0)
        s = f_try.sum()
        if s < 1e-300:
            continue
        f_try /= s
        
        sc = score_normalized(f_try, n)
        
        if sc < best_score:
            best_score = sc
            best_f = f_try.copy()
            f = f_try.copy()
            temperature = max(0.001, temperature * 0.95)
        else:
            # Accept with some probability (simulated annealing)
            delta = sc - best_score
            if delta < 0.01 and rng.random() < np.exp(-delta / (temperature * 0.1)):
                f = f_try.copy()
        
        # Decrease temperature
        temperature = max(0.0005, temperature * 0.999)
    
    return best_f, float(best_score)
