def _fft_conv(f):

                print(f"  ** New best after zero-insert: {best_score:.6f} **")
            f_cur = f_zi
            s_cur = s_zi

            # Re-optimize after insertion
            if (deadline - time.monotonic()) > 25:
                n_it3 = int(min(3000, (deadline - time.monotonic() - 10) * 80))
                f_ri, s_ri = _mirror_descent_v2(
                    f_cur, n_it3, eta_scale=0.3, thresh_start=0.8, thresh_end=0.95,
                    momentum=0.3, label=name + "_zi_ref")
                if s_ri < best_score:
                    best_score = s_ri
                    best_f = f_ri.copy()
                    print(f"  ** New best after zi+refine: {best_score:.6f} **")

    # ===== Phase 3: Softmax-smoothed final polish =====
    if (deadline - time.monotonic()) > 20:
        print("\n  Phase 3: Softmax polish")
        f = _normalize(best_f)
        if f is not None:
            n = len(f)
            f_poly = f.copy()
            for beta in [20.0, 50.0, 100.0, 200.0]:
                if (deadline - time.monotonic()) < 5:
                    break
                # A few iterations with smooth gradient
                for it in range(200):
                    if (deadline - time.monotonic()) < 5:
                        break
                    p = _softmax_pressure(f_poly, beta)
                    M = float(np.max(_fft_conv(f_poly)))
                    eta = 0.4 / (M + 1e-10)
                    log_ratio = np.clip(-eta * p, -20.0, 20.0)
                    f_poly = f_poly * np.exp(log_ratio)
                    f_poly = _normalize(f_poly)
                    if f_poly is None:
                        f_poly = np.ones(n) / n
                sc = _score(f_poly)
                if sc < best_score:
                    best_score = sc
                    best_f = f_poly.copy()
                    print(f"  Softmax beta={beta}: score={best_score:.6f}")

    # ===== Phase 4: Try a few random restarts from best =====
    if (deadline - time.monotonic()) > 30:
        print("\n  Phase 4: Random restarts")
        for restart in range(3):
            if (deadline - time.monotonic()) < 15:
                break
            pert = rng.random(len(best_f)) * 0.02 * np.max(best_f)
            f_p = _normalize(best_f + pert)
            if f_p is None:
                continue
            n_it = int(min(3000, (deadline - time.monotonic() - 5) * 80))
            f_r, s_r = _mirror_descent_v2(
                f_p, n_it, eta_scale=0.4, thresh_start=0.7, thresh_end=0.95,
                momentum=0.4, label=f"restart_{restart}")
            if s_r < best_score:
                best_score = s_r
                best_f = f_r.copy()
                print(f"  Restart {restart}: new best={best_score:.6f}")

    # ===== Final verification =====
    final_list = [float(x) for x in best_f]
    final_score = float(evaluate_sequence(final_list))
    print(f"\nFinal verified: n={len(final_list)}, score={final_score:.6f}")

    if not np.isfinite(final_score) or final_score > 2.0:
        print("  Fallback to parent.")
        return [float(x) for x in parent]

    return final_list
