import numpy as np
import time


def _fft_conv(f):
    """Compute full linear convolution f*f using FFT."""
    n = len(f)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    F = np.fft.rfft(f, size)
    return np.fft.irfft(F * F, size)[:2 * n - 1]


def _score(f):
    """Fast score assuming f >= 0, sum(f) > 0."""
    n = len(f)
    h = _fft_conv(f)
    M = float(np.max(h))
    S = float(np.sum(f))
    if S < 0.01:
        return float('inf')
    return 2.0 * n * M / (S * S)


def _normalize(f):
    """Clip to non-negative and normalize to sum=1."""
    f = np.maximum(f, 0.0)
    s = float(np.sum(f))
    if s < 1e-30:
        return None
    return f / s


def _subgradient(f, active_lags):
    """Compute averaged subgradient of max_k (f*f)(k) over active lags."""
    n = len(f)
    grad = np.zeros(n, dtype=np.float64)
    for k in active_lags:
        j_lo = max(0, k - n + 1)
        j_hi = min(n - 1, k)
        if j_hi >= j_lo:
            # grad[j] += 2*f(k-j) for j in [j_lo, j_hi]
            # k-j ranges from k-j_hi to k-j_lo
            vals = f[k - j_hi : k - j_lo + 1][::-1]
            grad[j_lo : j_hi + 1] += 2.0 * vals
    grad /= float(len(active_lags))
    return grad


def _project_simplex(v):
    """Project v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - 1))[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    w = np.maximum(v - theta, 0)
    return w


def _mirror_descent_nesterov(f0, n_iter, eta_scale, top_k, momentum_beta, label=""):
    """
    Nesterov-accelerated mirror descent on the simplex.
    Minimizes max_k (f*f)(k) subject to f >= 0, sum(f) = 1.
    Uses multiplicative (exponentiated) updates with momentum on the subgradient.
    """
    f = np.asarray(f0, dtype=np.float64).copy()
    f = _normalize(f)
    if f is None:
        return f0.copy(), float('inf')
    n = len(f)
    best_f = f.copy()
    best_score = _score(f)
    m = np.zeros(n, dtype=np.float64)  # momentum buffer
    y = f.copy()  # Nesterov extrapolation point
    t_curr = 1.0  # Nesterov counter
    t_prev = 1.0

    for it in range(n_iter):
        # Compute subgradient at y (extrapolated point)
        h = _fft_conv(y)
        M = float(np.max(h))
        if M <= 1e-30:
            break

        # Top-k active set
        kk = min(top_k, len(h))
        active = np.argsort(h)[-kk:]

        p = _subgradient(y, active)

        # Momentum update
        m = momentum_beta * m + (1.0 - momentum_beta) * p

        # Nesterov step sizes
        t_curr = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_prev * t_prev))
        gamma = 1.0 - (t_prev / t_curr)

        # Mirror descent update: f_new = exp(-eta * m) * f, normalized
        eta = eta_scale / (M + 1e-30)
        eta = min(eta, 1.0)  # cap step size
        log_ratio = -eta * m
        log_ratio = np.clip(log_ratio, -30.0, 30.0)
        f_new = f * np.exp(log_ratio)
        f_new = _normalize(f_new)
        if f_new is None:
            f_new = np.ones(n) / n

        # Nesterov extrapolation
        y = f_new + gamma * (f_new - f)
        y = _normalize(y)
        if y is None:
            y = f_new.copy()

        f = f_new
        t_prev = t_curr

        score = _score(f)
        if score < best_score:
            best_score = score
            best_f = f.copy()

        if (it + 1) % 2000 == 0:
            print(f"    MD-N it={it+1}: best={best_score:.6f}, M={M:.8f}")

    return best_f, best_score


def _additive_gradient_descent(f0, n_iter, eta_init, label=""):

    Additive projected gradient descent on the simplex.
    Uses Euclidean projection (unlike mirror descent's multiplicative updates).
    This explores a different geometry and can escape different local minima.

    f = _normalize(f)
    if f is None:
        return f0.copy(), float('inf')
    n = len(f)
    best_f = f.copy()
    best_score = _score(f)

    # Backtracking line search parameters
    c = 0.5
    rho = 0.8

    for it in range(n_iter):
        h = _fft_conv(f)
        M = float(np.max(h))
        if M <= 1e-30:
            break

        kk = min(10, len(h))
        active = np.argsort(h)[-kk:]
        p = _subgradient(f, active)

        # Center the gradient (projection onto sum=0 hyperplane)
        p_centered = p - np.mean(p)

        # Backtracking line search
        eta = eta_init / (M + 1e-30)
        eta = min(eta, 1.0)
        for _ in range(10):
            f_trial = _project_simplex(f - eta * p_centered)

            if s_trial <= M:  # sufficient decrease in max

            eta *= rho
        else:
            # Accept anyway if no decrease found (helps escape stalls)
            if s_trial > M * 1.01:
                f_trial = f.copy()
                s_trial = M

        f = f_trial
        score = _score(f)
        if score < best_score:
            best_score = score
            best_f = f.copy()

        if (it + 1) % 2000 == 0:
            print(f"    AGD it={it+1}: best={best_score:.6f}")

def _coordinate_descent(f0, n_passes, label=""):
    """
    Coordinate descent: for each coordinate, try to improve by adjusting its value.
    Uses a local search: try the current value +/- small perturbations.
    """
    f = np.asarray(f0, dtype=np.float64).copy()
    f = _normalize(f)
    if f is None:
        return f0.copy(), float('inf')
    n = len(f)
    best_f = f.copy()
    best_score = _score(f)

    for p_ in range(n_passes):
        perm = np.arange(n)
        np.random.shuffle(perm)
        improved = False
        for j in perm:
            old_val = f[j]
            if old_val < 1e-10:
                # Try adding a small amount
                candidates = [old_val, 0.001 / n, 0.01 / n, 0.05 / n]
            else:
                # Try scaling up and down
                candidates = [old_val,
                              old_val * 0.5, old_val * 0.7, old_val * 0.85,
                              old_val * 1.15, old_val * 1.3, old_val * 2.0,
                              old_val * 0.1]
            best_cand_score = best_score
            best_cand_val = old_val
            for cv in candidates:
                if cv < 0:
                    continue
                f[j] = cv
                fn = _normalize(f)
                if fn is None:
                    continue
                s = _score(fn)
                if s < best_cand_score:
                    best_cand_score = s
                    best_cand_val = cv
            f[j] = best_cand_val
            fn = _normalize(f)
            if fn is not None:
                f = fn
            if best_cand_val != old_val:
                improved = True
            if best_cand_score < best_score:
                best_score = best_cand_score
                best_f = f.copy()

        if not improved and p_ > 0:
            break

        if (p_ + 1) % 5 == 0:
            print(f"    CD pass={p_+1}: best={best_score:.6f}")

    return best_f, best_score


def _cosine_init(n):
    """Raised cosine window initialization."""
    j = np.arange(n, dtype=np.float64)
    f = np.cos(np.pi * (j + 0.5) / n) ** 2
    f = np.maximum(f, 0.0)
    f = _normalize(f)
    return f if f is not None else np.ones(n) / n


def _gaussian_init(n, sigma_frac):
    j = np.arange(n, dtype=np.float64) - (n - 1) / 2.0
    sigma = sigma_frac * n
    f = np.exp(-j * j / (2.0 * sigma * sigma))
    f = _normalize(f)
    return f if f is not None else np.ones(n) / n


def _bimodal_init(n):
    """Bimodal (W-shaped) initialization."""
    j = np.arange(n, dtype=np.float64)
    center = (n - 1) / 2.0
    width = n * 0.2
    f = np.exp(-((j - center * 0.5) ** 2) / (2 * width ** 2)) + \
        np.exp(-((j - center * 1.5) ** 2) / (2 * width ** 2))
    f = _normalize(f)
    return f if f is not None else np.ones(n) / n

            best_f = p_norm.copy()
            print(f"Parent: n={len(parent)}, score={s:.6f}")

    # Define candidate initializations
    # Try multiple lengths
    lengths = [512, 1024, 2048, 4096, 8192]

    candidates = []
    for n in lengths:
        candidates.append(("cos_%d" % n, _cosine_init(n)))
        candidates.append(("gauss_%d" % n, _gaussian_init(n, 0.3)))
        candidates.append(("bimodal_%d" % n, _bimodal_init(n)))
        candidates.append(("binary_%d" % n, _optimal_binary_init(n)))

    # Add perturbed parent
    if parent is not None and len(parent) > 10:
        n_p = len(parent)
        p_norm = _normalize(parent)
        if p_norm is not None:
            for frac in [0.01, 0.05, 0.1]:
                noise = rng.random(n_p) * frac
                pert = _normalize(p_norm + noise)
                if pert is not None:
                    candidates.append(("pert_parent_%d" % int(frac * 100), pert))

    # Add random inits
    for n in [1024, 2048]:
        f_r = rng.random(n)
        f_rn = _normalize(f_r)
        if f_rn is not None:
            candidates.append(("rand_%d" % n, f_rn))

    print(f"Total candidates: {len(candidates)}")

    # Phase 1: Run Nesterov mirror descent on each candidate
    seqs = []
    scores = []
    names = []

    # Allocate time: give more time to larger sequences (they tend to be better)
    # and to the parent-based candidates
    n_cands = len(candidates)
    time_per_base = (deadline - time.monotonic() - 60.0) / n_cands
    time_per_base = max(3.0, min(30.0, time_per_base))

    for idx, (name, f0) in enumerate(candidates):
        if time.monotonic() > deadline - 30:
            break

        # Allocate more iterations for larger sequences
        n0 = len(f0)
        time_alloc = time_per_base * min(2.0, max(0.8, n0 / 1024.0))
        n_iter = int(min(8000, max(500, time_alloc * 300)))

        t0 = time.monotonic()
        f_md, s_md = _mirror_descent_nesterov(f0, n_iter, 0.4, 10, 0.7, label=name)
        elapsed = time.monotonic() - t0
        print(f"  [{idx+1}/{n_cands}] {name}: MD score={s_md:.6f} ({elapsed:.1f}s)")

        seqs.append(f_md)
        scores.append(s_md)
        names.append(name)

        if s_md < best_score:
            best_score = s_md
            best_f = f_md.copy()
            print(f"  *** New best after MD: {best_score:.6f}")

        # Quick additive gradient descent refinement
        if (deadline - time.monotonic()) > 15:
            n_agd = int(min(3000, max(200, (deadline - time.monotonic() - 15) * 50.0 / max(1, n_cands - idx))))
            f_agd, s_agd = _additive_gradient_descent(f_md, n_agd, 0.3, label=name + "_agd")
            if s_agd < best_score:
                best_score = s_agd
                best_f = f_agd.copy()
                print(f"  *** New best after AGD: {best_score:.6f}")
            # Keep the better one
            if s_agd < s_md:
                seqs[-1] = f_agd
                scores[-1] = s_agd

    # Phase 2: Refine top candidates with coordinate descent
    order = sorted(enumerate(scores), key=lambda x: x[1])
    top_k = min(5, len(order))

    print(f"\nPhase 2: Refining top {top_k} candidates")
    for r in range(top_k):
        if time.monotonic() > deadline - 20:
            break
        idx = order[r][0]
        name = names[idx]
        f_cur = seqs[idx].copy()
        s_cur = scores[idx]
        n = len(f_cur)
        print(f"\n  Refining {name} (n={n}, score={s_cur:.6f})")

        # More mirror descent with smaller step
        if (deadline - time.monotonic()) > 10:
            n_it2 = int(min(10000, (deadline - time.monotonic() - 10) * 80))
            f_ref, s_ref = _mirror_descent_nesterov(f_cur, n_it2, 0.15, 15, 0.8, label=name + "_ref")
            if s_ref < best_score:
                best_score = s_ref
                best_f = f_ref.copy()
                print(f"  *** New best after refinement MD: {best_score:.6f}")
            f_cur = f_ref
            s_cur = s_ref

        # Additive gradient descent
        if (deadline - time.monotonic()) > 10:
            n_it3 = int(min(5000, (deadline - time.monotonic() - 10) * 60))
            f_agd2, s_agd2 = _additive_gradient_descent(f_cur, n_it3, 0.15, label=name + "_agd2")
            if s_agd2 < best_score:
                best_score = s_agd2
                best_f = f_agd2.copy()
                print(f"  *** New best after refinement AGD: {best_score:.6f}")
            if s_agd2 < s_cur:
                f_cur = f_agd2
                s_cur = s_agd2

        # Coordinate descent (expensive, use only for smaller n)
        if n <= 4096 and (deadline - time.monotonic()) > 15:
            n_passes = int(min(30, (deadline - time.monotonic() - 15) / max(1.0, n * 0.001)))
            f_cd, s_cd = _coordinate_descent(f_cur, n_passes, label=name + "_cd")
            if s_cd < best_score:
                best_score = s_cd
                best_f = f_cd.copy()
                print(f"  *** New best after CD: {best_score:.6f}")
            f_cur = f_cd
            s_cur = s_cd

        # Perturbation restart
        if (deadline - time.monotonic()) > 15:
            for frac in [0.005, 0.02, 0.05]:
                if (deadline - time.monotonic()) < 10:
                    break
                noise = rng.random(len(f_cur)) * frac
                f_pert = _normalize(f_cur + noise)
                if f_pert is None:
                    continue
                n_it4 = int(min(3000, (deadline - time.monotonic() - 5) * 80))
                f_p, s_p = _mirror_descent_nesterov(f_pert, n_it4, 0.2, 10, 0.7, label=name + f"_pert{frac}")
                if s_p < best_score:
                    best_score = s_p
                    best_f = f_p.copy()
                    print(f"  *** New best after perturb({frac}): {best_score:.6f}")
                if s_p < s_cur:
                    f_cur = f_p
                    s_cur = s_p

    # Phase 3: Final multi-threshold sweep on the best solution
    if best_f is not None and (deadline - time.monotonic()) > 15:
        print(f"\nPhase 3: Final sweep on best (score={best_score:.6f})")
        for top_k_val in [5, 20, 50]:
            if (deadline - time.monotonic()) < 8:
                break
            n_it5 = int(min(5000, (deadline - time.monotonic() - 8) * 70))
            f_t, s_t = _mirror_descent_nesterov(best_f, n_it5, 0.1, top_k_val, 0.85, label=f"final_k{top_k_val}")
            if s_t < best_score:
                best_score = s_t
                best_f = f_t.copy()
                print(f"  *** New best (top_k={top_k_val}): {best_score:.6f}")

        # Additive GD as final touch
        if (deadline - time.monotonic()) > 8:
            n_it6 = int(min(5000, (deadline - time.monotonic() - 8) * 60))
            f_a, s_a = _additive_gradient_descent(best_f, n_it6, 0.1, label="final_agd")
            if s_a < best_score:
                best_score = s_a
                best_f = f_a.copy()
                print(f"  *** New best (final AGD): {best_score:.6f}")

    # Final verification
    if best_f is None:
        print("ERROR: No valid solution found, returning uniform.")
        best_f = np.ones(1024) / 1024

    final_list = [float(x) for x in best_f]
    final_score = float(evaluate_sequence(final_list))
    print(f"\nFinal verified score: {final_score:.6f} (n={len(final_list)})")

    # Fallback if something went wrong
    if not np.isfinite(final_score) or final_score > 2.5:
        print("  Falling back to parent or uniform.")
        if parent is not None and len(parent) > 0:
            return [float(max(0, x)) for x in parent]
        return [1.0] * 1024

    return final_list
