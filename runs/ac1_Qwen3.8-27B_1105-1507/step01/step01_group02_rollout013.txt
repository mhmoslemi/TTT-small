def project_simplex(v):
    """Project vector v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cv = np.cumsum(u) - 1.0
    rho_candidates = np.where(u * np.arange(1, n + 1) > cv)[0]
    if len(rho_candidates) == 0:
        return np.ones(n) / n
    rho = rho_candidates[-1]
    theta = cv[rho] / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

"""
Optimizes non-negative sequence to minimize 2*n*max(conv(a,a))/sum(a)^2.
Uses soft-max smoothed gradient descent followed by coordinate descent polish.
Multiple diverse initializations with adaptive temperature scheduling.
"""

import numpy as np
import time


def project_simplex(v):
    """Project vector v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho = np.where(u * np.arange(1, n + 1) > cssv)[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho = rho[-1]
    theta = cssv[rho] / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def fast_convolve(a):
    """Compute full linear convolution a*a using FFT."""
    n = len(a)
    size = 2 * n - 1
    N = 1 << (size - 1).bit_length()
    fa = np.fft.rfft(a, N)
    fc = fa * fa
    result = np.fft.irfft(fc, N)[:size]
    result[result < 0] = 0.0
    return result


def score_of(a):
    """Compute the objective value 2*n*max_c (assuming sum(a)=1)."""
    n = len(a)
    c = fast_convolve(a)
    return 2.0 * n * np.max(c)


def gradient_of_max_c(a, c, top_k=50):

    """
    Smoothed gradient descent to minimize max_k c[k] with sum(a)=1, a>=0.
    Uses soft-max weighting of lags for smooth gradient.
    """
    n = len(a)
    a = project_simplex(a.copy().astype(np.float64))
    
    best_score = score_of(a)
    best_a = a.copy()
    
    # Parameters
    base_lr = 1.0 / n
    lr = base_lr
    stall_count = 0
    max_stall = 300
    top_k_start = 20
    top_k_end = 500  # effectively all
    
    it = 0
    while time.time() < deadline and it < max_iters:
        it += 1
        
        c = fast_convolve(a)
        cur_score = 2.0 * n * np.max(c)
        
        if cur_score < best_score * (1 - 1e-10):
            best_score = cur_score
            best_a = a.copy()
            stall_count = 0
        else:
            stall_count += 1
        
        # Adaptive top-k: start with few (focus on peaks), increase to many (flatten all)
        progress = min(1.0, it / (max_iters * 0.7))
        top_k = int(top_k_start + (top_k_end - top_k_start) * progress ** 1.5)
        top_k = min(top_k, 2 * n - 1)
        
        # Adaptive learning rate
        if stall_count > max_stall:
            lr = max(lr * 0.8, base_lr * 1e-3)
            stall_count = 0
            if lr < base_lr * 1e-2:

        grad, peak = gradient_of_max_c(a, c, top_k)
        
        # Gradient step
        a_new = a - lr * grad
        a = project_simplex(a_new)
        
        if it % 5000 == 0:
            remaining = deadline - time.time()
            print(f"  [SGD] iter={it}, score={best_score:.6f}, lr={lr:.2e}, top_k={top_k}, remaining={remaining:.0f}s")
    
    return best_a, best_score


def coordinate_descent(a, deadline, rng, n_sweeps=50):
    """
    Coordinate descent: optimize one coordinate at a time using golden section search.
    """
    n = len(a)
    a = project_simplex(a.copy().astype(np.float64))
    best_score = score_of(a)
    best_a = a.copy()
    
    for sweep in range(n_sweeps):
        if time.time() > deadline - 5:
            break
        
        sweep_improved = False
        # Randomize coordinate order for diversity
        order = rng.permutation(n)
        
        for j in order:
            if time.time() > deadline - 2:
                break
            
            old_val = a[j]
            if old_val < 1e-12:
                continue
            
            S = 1.0 - old_val  # sum of other elements
            
            # Golden section search for optimal t = a[j]
            # Parameterization: a_new[j] = t, a_new[i] = a[i]*(1-t)/S for i≠j
            # t in [0, 1]
            
            # Evaluate at a few points to get initial bracket
            def eval_t(t):
                if t < 0 or t > 1:
                    return float('inf')
                if S < 1e-15:
                    # All mass is in j
                    return float('inf')
                a_new = np.empty(n)
                a_new[j] = t
                a_new[:] = 0
                mask = np.ones(n, dtype=bool)
                mask[j] = False
                a_new[mask] = a[mask] * (1.0 - t) / S
                c = fast_convolve(a_new)
                return 2.0 * n * np.max(c)
            
            # Quick grid to find approximate minimum
            grid_points = [0.0, 0.1, 0.25, 0.4, 0.5, 0.6, 0.75, 0.9, 1.0]
            # Include the current value
            grid_points.append(old_val)
            grid_points = sorted(set(max(0, min(1, p)) for p in grid_points))
            
            vals = [eval_t(t) for t in grid_points]
            best_local = min(zip(grid_points, vals), key=lambda x: x[1])
            
            # Golden section refinement around best grid point
            idx_best = grid_points.index(best_local[0])
            lo = grid_points[max(0, idx_best - 1)]
            hi = grid_points[min(len(grid_points) - 1, idx_best + 1)]
            
            if hi - lo > 1e-6:
                # Golden section search
                gr = (np.sqrt(5) - 1) / 2  # golden ratio conjugate
                c1 = hi - gr * (hi - lo)
                c2 = lo + gr * (hi - lo)
                f1 = eval_t(c1)
                f2 = eval_t(c2)
                
                for _ in range(30):
                    if hi - lo < 1e-7:
                        break
                    if f1 < f2:
                        hi = c2
                        c2 = c1
                        f2 = f1
                        c1 = hi - gr * (hi - lo)
                        f1 = eval_t(c1)
                    else:
                        lo = c1
                        c1 = c2
                        f1 = f2
                        c2 = lo + gr * (hi - lo)
                        f2 = eval_t(c2)

            else:
                t_opt = best_local[0]

                mask = np.ones(n, dtype=bool)
                mask[j] = False
                a[mask] = a[mask] * (1.0 - t_opt) / S
                # Re-normalize to handle floating point
                a = project_simplex(a)
            
            cur_score = score_of(a)

                best_score = cur_score
                best_a = a.copy()
                sweep_improved = True
        
        if sweep > 0 and not sweep_improved:
            # No improvement in this sweep, try a smaller perturbation
            if time.time() < deadline - 10:
                a = project_simplex(a + rng.normal(0, 0.005 / np.sqrt(n), n))
    
    return best_a, best_score
