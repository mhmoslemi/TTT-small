def _fft_convolve_full(f):
    """Compute full linear convolution of f with itself using FFT."""
    n = len(f)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    F = np.fft.rfft(f, size)
    h = np.fft.irfft(F * F, size)
    return h[:2 * n - 1]

def _score_raw(f):
    """Fast score computation assuming f >= 0, sum(f) > 0.01."""
    n = len(f)
    h = _fft_convolve_full(f)
    M = float(np.max(h))
    S = float(np.sum(f))
    if S < 0.01:
        return float('inf')
    return 2.0 * n * M / (S * S)

import time

def _fft_convolve_full(f):
    n = len(f)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    F = np.fft.rfft(f, size)
    h = np.fft.irfft(F * F, size)
    return h[:2 * n - 1]

def _score_raw(f):
    n = len(f)
    h = _fft_convolve_full(f)
    M = float(np.max(h))
    S = float(np.sum(f))
    if S < 0.01:
        return float('inf')
    return 2.0 * n * M / (S * S)

def _normalize(f):
    f = np.maximum(f, 0.0)
    s = float(np.sum(f))
    if s < 1e-30:
        return None
    return f / s

def _cosine_init(n, power=2.0):
    j = np.arange(n, dtype=np.float64)
    x = np.cos(np.pi * (j + 0.5) / float(n))
    f = np.abs(x) ** power
    s = np.sum(f)
    if s > 0:
        f /= s
    return f

def _tapered_init(n, power=2.0):
    j = np.arange(n, dtype=np.float64)
    x = np.sin(np.pi * (j + 0.5) / float(n))
    f = x ** power
    f = np.maximum(f, 0.0)
    s = np.sum(f)
    if s > 0:
        f /= s
    return f

    sigma = sigma_frac * float(n)
    f = np.exp(-j * j / (2.0 * sigma * sigma))
    s = np.sum(f)
    if s > 0:

    return f

def _pressure_topk(f, h, top_k):
    """Compute pressure vector using top-k lags."""
    n = len(f)
    # Get top-k lags
    if top_k >= 2 * n - 1:
        active = np.arange(2 * n - 1)
    else:

        else:
            top_indices = np.arange(len(h))
        active = np.sort(top_indices)
    
    p = np.zeros(n, dtype=np.float64)

        j_end = min(n - 1, k)
        if j_end >= j_start:
            lo = k - j_end
            hi = k - j_start
            p[j_start:j_end + 1] += 2.0 * f[lo:hi + 1][::-1]
    p /= float(len(active))
    return p

def _pressure_smooth(f, h, beta, top_k=None):
    """Compute pressure using softmax-weighted gradient."""
    n = len(f)
    max_h = float(np.max(h))

    # Clip for numerical stability
    shifted = np.clip(shifted, -100, 100)
    exp_shifted = np.exp(shifted)

        mask = softmax >= threshold

            hi = k - j_start
            p[j_start:j_end + 1] += w * f[lo:hi + 1][::-1]
    return p

def _mirror_descent(f_init, n_iter, eta0, decay, top_k_frac, label=""):
    """Mirror descent with adaptive step size and top-k active set."""
    f = np.asarray(f_init, dtype=np.float64).copy()

            break

            print(f"    MD it={it+1}: best={best_score:.6f}, M={M:.8f}, eta={eta:.4f}, |K|={top_k}")
