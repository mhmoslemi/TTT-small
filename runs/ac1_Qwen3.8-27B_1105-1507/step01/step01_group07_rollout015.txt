def compute_obj(a):
    n = len(a)
    if n == 0:
        return np.inf
    b = np.convolve(a, a)
    max_b = float(np.max(b))
    S = float(np.sum(a))
    if S < 0.01:
        return np.inf
    return 2.0 * n * max_b / (S * S)

def smooth_objective_and_grad(a, beta=50.0):
    """Compute smooth objective and gradient using log-sum-exp approximation of max."""
    n = len(a)
    b = np.convolve(a, a)
    S = np.sum(a)
    
    # Stable log-sum-exp
    b_max = np.max(b)
    scaled = beta * (b - b_max)
    exp_scaled = np.exp(scaled)
    log_sum_exp = b_max + np.log(np.sum(exp_scaled)) / beta
    # This is the smooth max
    
    # Weights p_k = exp(beta * b_k) / sum exp(beta * b_l)
    p = exp_scaled / np.sum(exp_scaled)
    
    # Gradient of smooth max w.r.t. a_j
    # dm/da_j = sum_k p_k * a_{k-j} for 0 <= k-j < n
    # = sum_{m=0}^{n-1} p_{j+m} * a_m (with p_k = 0 for k >= len(p))
    
    dm_da = np.zeros(n)
    for j in range(n):
        # sum over k from j to j+n-1 of p[k] * a[k-j]
        # k-j ranges from 0 to n-1
        k_vals = np.arange(j, min(j + n, len(p)))
        dm_da[j] = np.sum(p[k_vals] * a[k_vals - j])
    
    # J = 2n * m / S^2
    J = 2.0 * n * log_sum_exp / (S * S)

    grad = 2.0 * n * (dm_da * S - 2.0 * log_sum_exp) / (S * S * S)
    
    return J, grad

Combined gradient-based and LP-optimized search for minimizing

gradient computation, projected gradient descent, alternating LP,
and diverse initializations across multiple sequence lengths.

import numpy as np
from scipy.optimize import linprog

def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    b = np.convolve(a, a)
    max_b = float(np.max(b))
    S = float(np.sum(a))
    if S < 0.01:
        return np.inf
    return 2.0 * n * max_b / (S * S)


def smooth_obj_grad(a, beta=50.0):
    """Compute smooth objective (log-sum-exp approx of max) and gradient."""
    n = len(a)
    b = np.convolve(a, a)
    S = float(np.sum(a))
    
    b_max = float(np.max(b))
    scaled = beta * (b - b_max)
    exp_scaled = np.exp(scaled)
    sum_exp = float(np.sum(exp_scaled))
    log_sum_exp = b_max + np.log(sum_exp) / beta
    
    # Softmax weights
    p = exp_scaled / sum_exp
    
    # Gradient of smooth max w.r.t. a_j = sum_k p_k * a_{k-j} for 0<=k-j<n
    dm_da = np.zeros(n)
    for j in range(n):
        # sum over m=0 to n-1 of p[j+m] * a[m], where p has length 2n-1
        k_start = j
        k_end = min(j + n, 2 * n - 1)
        if k_start < k_end:
            dm_da[j] = float(np.dot(p[k_start:k_end], a[:k_end - k_start]))
    
    # J = 2n * m / S^2
    J = 2.0 * n * log_sum_exp / (S * S)
    
    # dJ/da_j = 2n * (dm_da_j * S - 2*m) / S^3
    grad = 2.0 * n * (dm_da * S - 2.0 * log_sum_exp) / (S ** 3)
    
    return J, grad

def build_A_sparse(a):
    """Build the constraint matrix for LP (dense for simplicity)."""
    n = len(a)
    A = np.zeros((2 * n - 1, n), dtype=np.float64)
    for k in range(2 * n - 1):
        j_lo = max(0, k - n + 1)
        j_hi = min(n - 1, k)
        # A[k, j] = a[k-j] for j in [j_lo, j_hi]
        # a[k-j_lo : k-j_hi+1] reversed
        A[k, j_lo:j_hi + 1] = a[k - j_hi:k - j_lo + 1][::-1]
    return A


def lp_step(a, upper_bound=100.0):
    """Solve LP: max sum(g) s.t. (a*g)_k <= M, 0 <= g <= ub."""
    n = len(a)
    b = np.convolve(a, a)
    M = float(np.max(b))
    A = build_A_sparse(a)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, upper_bound)] * n
    
    res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                  options={'presolve': True})
    if res.status == 0:
        return res.x, float(-res.fun), M
    return None, None, M

"""
Combined optimization for minimizing 2*n*max(conv(a,a))/(sum(a))^2.
Uses smooth-max gradient descent (log-sum-exp), alternating LP, simulated
annealing, and diverse structured initializations across multiple lengths.
"""

import numpy as np
from scipy.optimize import linprog
import time


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    b = np.convolve(a, a)
    max_b = float(np.max(b))
    S = float(np.sum(a))
    if S < 0.01:
        return np.inf
    return 2.0 * n * max_b / (S * S)


def smooth_obj_grad(a, beta=50.0):
    """Compute smooth objective (log-sum-exp approx of max) and its gradient.
    Returns (J, grad) where J approximates 2n*max(conv(a,a))/S^2.
    """
    n = len(a)
    b = np.convolve(a, a)
    S = float(np.sum(a))

    b_max = float(np.max(b))
    scaled = beta * (b - b_max)
    exp_scaled = np.exp(scaled)
    sum_exp = float(np.sum(exp_scaled))
    log_sum_exp = b_max + np.log(sum_exp) / beta

    # Softmax weights p_k
    p = exp_scaled / sum_exp

    # Gradient of smooth max w.r.t. a_j via FFT-based convolution
    # dm_da[j] = sum_{m=0}^{n-1} p[j+m] * a[m]
    # Computed as: np.convolve(a, p[::-1], 'full')[n-1:2n-1][::-1]
    dm_da = np.convolve(a, p[::-1], 'full')[n - 1:2 * n - 1][::-1]

    # J = 2n * m / S^2
    J = 2.0 * n * log_sum_exp / (S * S)

    # dJ/da_j = 2n * (dm_da_j * S - 2*m) / S^3
    grad = 2.0 * n * (dm_da * S - 2.0 * log_sum_exp) / (S ** 3)

    return J, grad


def gradient_optimize(a, max_time=60.0, initial_lr=0.1, beta_start=20.0,
                      beta_end=500.0, verbose=False):
    """Projected gradient descent with smooth max, adaptive beta and learning rate."""
    n = len(a)
    a = a.copy().astype(np.float64)
    S = np.sum(a)
    if S < 1e-10:
        a = np.ones(n) / n
        S = 1.0
    a = a / S

    best_a = a.copy()
    best_obj = compute_obj(a)

    start = time.monotonic()
    lr = initial_lr
    patience = 0
    max_patience = 300
    total_iters = 0

    for it in range(200000):
        if time.monotonic() - start > max_time:
            break

        # Adaptive beta: increase over time for sharper peak focus
        progress = min(1.0, (time.monotonic() - start) / max_time)
        beta = beta_start * (beta_end / beta_start) ** progress

        J, grad = smooth_obj_grad(a, beta)

        # Remove mean component (scale invariance direction)
        grad_mean = np.mean(grad)
        grad_centered = grad - grad_mean

        # Gradient step
        a_new = a - lr * grad_centered
        a_new = np.maximum(a_new, 0.0)
        S_new = np.sum(a_new)
        if S_new < 1e-10:
            lr *= 0.5
            patience += 1
            if patience > max_patience:
                break
            continue
        a_new = a_new / S_new

        total_iters += 1
        obj = compute_obj(a_new)
        if obj < best_obj:
            best_a = a_new.copy()
            best_obj = obj
            patience = 0
        else:
            patience += 1
            lr *= 0.95
            if patience > max_patience:
                break

        a = a_new

        if verbose and it % 1000 == 0:
            print(f"    GD it={it}: obj={best_obj:.6f} lr={lr:.5f} beta={beta:.1f}")

    return best_a, best_obj


def build_A_matrix(a):
    """Build the dense constraint matrix for LP: A[k,j] = a[k-j] if valid."""
    n = len(a)
    A = np.zeros((2 * n - 1, n), dtype=np.float64)
    for k in range(2 * n - 1):
        j_lo = max(0, k - n + 1)
        j_hi = min(n - 1, k)
        A[k, j_lo:j_hi + 1] = a[k - j_hi:k - j_lo + 1][::-1]
    return A


def lp_step(a, upper_bound=100.0):
    """Solve LP: max sum(g) s.t. (a*g)_k <= M for all k, 0 <= g <= ub."""
    n = len(a)
    b = np.convolve(a, a)
    M = float(np.max(b))
    A = build_A_matrix(a)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, upper_bound)] * n
    res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                  options={'presolve': True, 'dual_feasibility_tolerance': 1e-9})
    if res.status == 0:
        return res.x, float(-res.fun), M
    return None, None, M


def lp_optimize(a, max_iter=200, t=0.5, verbose=False):
    """Alternating LP optimization to minimize the objective."""
    n = len(a)
    a = a.copy().astype(np.float64)
    S = np.sum(a)
    if S < 1e-10:
        a = np.ones(n) / n
        S = 1.0
    a = a / S

    best_a = a.copy()
    best_obj = compute_obj(a)
    stall = 0

    for it in range(max_iter):
        g, g_sum, M = lp_step(a)
        if g is None:
            break

        if g_sum <= 1.0 + 1e-6:
            stall += 1
            if stall > 15:
                break
            continue
        stall = 0

        g = g / g_sum
        # Adaptive step size
        t_eff = t * min(1.0, g_sum - 1.0 + 0.1)
        a_new = (1.0 - t_eff) * a + t_eff * g
        a_new = np.maximum(a_new, 0.0)
        S_new = np.sum(a_new)
        if S_new < 1e-10:
            break
        a_new = a_new / S_new

        obj = compute_obj(a_new)
        if obj < best_obj:
            best_a = a_new.copy()
            best_obj = obj
            if verbose and it % 10 == 0:
                print(f"    LP it={it}: obj={best_obj:.6f} M={M:.6f}")
        a = a_new

    return best_a, best_obj


def initialize(n, init_type, rng):
    """Generate diverse structured initializations."""
    x = np.arange(n, dtype=np.float64)
    if init_type == 'random':
        a = rng.random(n)
    elif init_type == 'cosine':
        freq = rng.uniform(0.5, 3.0)
        phase = rng.uniform(0, 2 * np.pi)
        a = np.maximum(0, np.cos(freq * x / n * 2 * np.pi + phase))
        if np.sum(a) < 1e-10:
            a = rng.random(n)
    elif init_type == 'bump':
        center = n / 2.0
        width = rng.uniform(0.15, 0.4) * n
        a = np.exp(-(x - center) ** 2 / (2 * width ** 2))
    elif init_type == 'edge':
        center = n / 2.0
        width = rng.uniform(0.2, 0.5) * n
        a = 1.0 - np.exp(-(x - center) ** 2 / (2 * width ** 2))
        a = np.maximum(a, 0.01)
    elif init_type == 'split':
        split = int(n * rng.uniform(0.3, 0.7))
        a = np.ones(n)
        a[split:] = rng.uniform(0.2, 0.8)
    elif init_type == 'periodic':
        period = int(rng.integers(max(2, n // 30), max(3, n // 4)))
        pattern = rng.random(period)
        a = np.tile(pattern, n // period + 1)[:n]
    elif init_type == 'chirp':
        # Linear chirp (Zadoff-Chu like, but real part)
        mu = rng.choice([1, 2, 3])
        phi = 2.0 * np.pi * mu * x * x / n
        a = np.abs(np.exp(1j * phi)) + 0.5 * np.cos(phi)
        a = np.maximum(a, 0.01)
    elif init_type == 'triangle':
        # Triangular shape
        mid = n // 2
        a = np.minimum(x, n - 1 - x).astype(np.float64)
        a = a / max(np.max(a), 1e-10)
        a = 1.0 - a  # inverted triangle (peaks at edges)
        if rng.random() < 0.5:
            a = a[::-1]  # sometimes peaks in center
        a = np.maximum(a, 0.01)
    else:
        a = rng.random(n)

    a = np.maximum(a, 0.0)
    S = np.sum(a)
    if S < 1e-10:
        a = np.ones(n)
        S = float(n)
    return a / S


def simulated_annealing_refine(a, rng, max_time=30.0, initial_temp=0.15):
    """Simulated annealing local search with block perturbations."""
    n = len(a)
    best_a = a.copy()
    best_obj = compute_obj(a)
    current_a = a.copy()
    current_obj = best_obj
    temp = initial_temp

    start = time.monotonic()
    while time.monotonic() - start < max_time:
        # Variable-width perturbation
        r = rng.random()
        if r < 0.3:
            width = max(1, n // 50)
        elif r < 0.7:
            width = max(1, n // 15)
        else:
            width = max(1, n // 5)
        pos = int(rng.integers(0, max(1, n - width + 1)))
        noise = np.exp(rng.normal(0, temp, width))
        candidate = current_a.copy()
        candidate[pos:pos + width] *= noise
        candidate = np.maximum(candidate, 0.0)
        S = np.sum(candidate)
        if S < 1e-10:
            temp *= 0.95
            continue
        candidate = candidate / S

        new_obj = compute_obj(candidate)
        delta = new_obj - current_obj

        if delta < 0 or (temp > 1e-8 and rng.random() < np.exp(-delta / temp)):
            current_a = candidate
            current_obj = new_obj
            if new_obj < best_obj:
                best_a = candidate.copy()
                best_obj = new_obj

        temp *= 0.998
        if temp < 1e-7:
            temp = 1e-7

    return best_a, best_obj


def multi_scale_gradient(a, max_time=60.0, verbose=False):
    """Multi-scale gradient: start with coarse (low beta) then refine (high beta)."""
    n = len(a)
    stages = [
        (10.0, 20.0, 0.3),   # beta_start, beta_end, lr
        (20.0, 100.0, 0.1),
        (50.0, 300.0, 0.05),
        (100.0, 1000.0, 0.02),
    ]
    elapsed = 0.0
    for beta_s, beta_e, lr in stages:
        if elapsed >= max_time:
            break
        remaining = max_time - elapsed
        stage_time = min(remaining, 20.0)
        a, obj = gradient_optimize(a, max_time=stage_time, initial_lr=lr,
                                   beta_start=beta_s, beta_end=beta_e, verbose=verbose)
        elapsed += time.monotonic() - (time.monotonic() - elapsed)  # approximate
    return a, compute_obj(a)
