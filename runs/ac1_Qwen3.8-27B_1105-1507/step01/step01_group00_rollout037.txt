import numpy as np
import time

def project_simplex(v, target_sum=1.0):
    """Project v onto {x >= 0, sum(x) = target_sum}."""
    n = len(v)
    if n == 0:
        return v
    # Standard simplex projection
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - target_sum))[0][-1]
    theta = (cssv[rho] - target_sum) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

    while time.monotonic() < deadline:

        max_idx = int(np.argmax(conv_f))
        M = conv_f[max_idx]

        it += 1

            best_f = f.copy()
            best_score = new_score
            no_improve = 0
        else:

            lr_current *= 0.9

            best_f = f.copy()

        lr *= 0.999
    
    return best_f, best_score

Hybrid LP + smooth gradient optimization for minimizing 2n*max(conv(a,a))/sum(a)^2.
Combines LP-based direction finding with log-sum-exp smoothed gradient descent,

import numpy as np

def project_simplex(v, target_sum=1.0):
    """Euclidean projection onto the simplex {x >= 0, sum(x) = target_sum}."""
    n = len(v)
    if n == 0:
        return v.copy()
    if target_sum <= 0:
        return np.zeros(n)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - target_sum))[0]
    if len(rho) == 0:
        return np.full(n, target_sum / n)
    rho = rho[-1]
    theta = (cssv[rho] - target_sum) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def build_conv_matrix_sparse(f):
    """Build sparse matrix A where (A@g)[k] = convolve(f,g)[k]."""
    n = len(f)
    rows_list = []
    cols_list = []
    vals_list = []
    for k in range(2 * n - 1):
        j_min = max(0, k - (n - 1))
        j_max = min(k, n - 1)
        for j in range(j_min, j_max + 1):
            rows_list.append(k)
            cols_list.append(j)
            vals_list.append(f[k - j])
    rows = np.array(rows_list, dtype=np.int32)
    cols = np.array(cols_list, dtype=np.int32)
    vals = np.array(vals_list, dtype=np.float64)
    return csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))

def solve_lp_direction(f, M, top_k=None):
    """Solve: max sum(g) s.t. g >= 0, convolve(f,g)_k <= M for all k (or top_k constraints)."""
    n = len(f)
    A = build_conv_matrix_sparse(f)
    c = -np.ones(n)
    
    if top_k is not None and top_k < 2*n-1:
        # Only use the top_k constraints (largest diagonal of A, i.e., most active)
        # Actually, we should use the constraints that are most "tight"
        # For now, use all constraints but limit the number
        b_ub = np.full(2 * n - 1, M)
    else:
        b_ub = np.full(2 * n - 1, M)
    
    bounds = [(0, None)] * n
    try:
        result = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
        if result.success:
            return result.x
    except Exception:
        pass
    return None

def lp_iterate(f, deadline, step_size_init=0.15, verbose_prefix=""):
    """Run LP-based iterations on f."""
    f = f.copy().astype(np.float64)
    best_f = f.copy()
    best_score = evaluate_sequence(f.tolist())
    step_size = step_size_init
    no_improve = 0
    it = 0
    
    while time.monotonic() < deadline:
        conv_f = np.convolve(f, f)
        M = float(np.max(conv_f))
        if M <= 0:
            break
        
        g = solve_lp_direction(f, M)
        if g is None:
            no_improve += 1
            if no_improve > 20:
                break
            continue
        
        sum_g = float(np.sum(g))
        if sum_g <= 1e-15:
            no_improve += 1
            continue
        
        # Rescale g to have same sum as f
        sum_f = float(np.sum(f))
        g = g * (sum_f / sum_g)
        
        # Line search: try several step sizes
        best_local = f
        best_local_score = best_score
        for t in [step_size, step_size * 0.5, step_size * 0.7, step_size * 1.3, step_size * 0.3]:
            new_f = (1.0 - t) * f + t * g
            new_f = np.clip(new_f, 0.0, 1000.0)
            new_score = evaluate_sequence(new_f.tolist())
            if np.isfinite(new_score) and new_score < best_local_score:
                best_local = new_f
                best_local_score = new_score
        
        it += 1
        if best_local_score < best_score:
            f = best_local
            best_f = f.copy()
            best_score = best_local_score
            no_improve = 0
            # Increase step size slightly if we made good progress
            if step_size < 0.4:
                step_size = min(0.4, step_size * 1.1)
        else:
            no_improve += 1
            step_size *= 0.7
            if step_size < 0.003 or no_improve > 40:
                break
    
    return best_f, best_score

def lse_gradient_descent(f, deadline, T_init=0.1, T_min=0.005, T_decay=0.998, lr_init=0.05, verbose_prefix=""):
    """Log-sum-exp smoothed gradient descent with simplex projection."""
    n = len(f)
    f = f.copy().astype(np.float64)
    S = float(np.sum(f))
    if S <= 0:
        return f, evaluate_sequence(f.tolist())
    f = f / S  # normalize to sum = 1
    
    best_f = f.copy()
    best_score = evaluate_sequence(f.tolist())
    
    T = T_init
    lr = lr_init
    no_improve = 0
    it = 0
    
    while time.monotonic() < deadline and T > T_min:
        conv_f = np.convolve(f, f)
        
        # Softmax weights (numerically stable)
        max_conv = float(np.max(conv_f))
        exp_conv = np.exp((conv_f - max_conv) / T)
        w = exp_conv / float(np.sum(exp_conv))
        
        # Gradient computation (vectorized)
        # grad[j] = 2 * sum_k w[k] * f[k-j] for valid k-j
        # Build the weighted correlation
        # For efficiency, use the fact that this is a convolution-like operation
        # grad = 2 * correlate(w, f) at specific lags
        
        # Direct O(n^2) computation (fast for n <= 2048)
        grad = np.zeros(n)
        for k in range(2 * n - 1):
            wk = w[k]
            if wk < 1e-12:
                continue
            j_min = max(0, k - (n - 1))
            j_max = min(k, n - 1)
            grad[j_min:j_max+1] += wk * 2.0 * f[k - j_min:k - j_max + 1][::-1]
        
        # Gradient descent with simplex projection
        new_f = f - lr * grad
        new_f = project_simplex(new_f, 1.0)
        
        new_score = evaluate_sequence(new_f.tolist())
        it += 1
        
        if np.isfinite(new_score) and new_score < best_score:
            f = new_f.copy()
            best_f = f.copy()
            best_score = new_score
            no_improve = 0
        else:
            no_improve += 1
            lr *= 0.9
            if lr < 1e-6:
                break
            if no_improve > 200:
                break
        
        T *= T_decay
    
    return best_f, best_score

def randomized_lp(f, deadline, n_random=10, verbose_prefix=""):

grad[j] = 2 * sum_{k: 0<=k-j<n} w[k] * f[k-j]

Let m = k-j, so k = j+m:
grad[j] = 2 * sum_{m: 0<=m<n, 0<=j+m<2n-1} w[j+m] * f[m]

For j = 0: grad[0] = 2 * sum_{m=0}^{n-1} w[m] * f[m] (since j+m = m < 2n-1 always for m < n)
For j = n-1: grad[n-1] = 2 * sum_{m: 0<=m<n, 0<=n-1+m<2n-1} w[n-1+m] * f[m] = 2 * sum_{m=0}^{n-1} w[n-1+m] * f[m] (since n-1+m < 2n-1 for m < n)

So for all j in [0, n-1], j+m ranges from j to j+n-1, and we need j+m < 2n-1, which means m < 2n-1-j. Since m < n and j >= 0, we have m < n <= 2n-1 <= 2n-1-j (for j <= n-1). So the condition is always satisfied!

Therefore:
grad[j] = 2 * sum_{m=0}^{n-1} w[j+m] * f[m]

This is the cross-correlation of w (length 2n-1) with f (length n) at lag j.

In numpy, this can be computed as:
```python
# We want corr[j] = sum_{m=0}^{n-1} w[j+m] * f[m] for j = 0, ..., n-1
# This is the "valid" part of the correlation of w with f
# scipy.signal.correlate(w, f, mode='full') would give a length (2n-1)+n-1 = 3n-2 result
# But what we want is specifically: for each j, sum_m w[j+m] * f[m]
# This is equivalent to: correlate(w, f, mode='valid') if len(w) >= len(f)
# Actually, scipy.signal.correlate(in1, in2, mode='valid') gives:
#   out[k] = sum_n in1[n+k] * in2[n] for k = 0, ..., len(in1)-len(in2)
# So if in1 = w (length 2n-1) and in2 = f (length n):
#   out[k] = sum_n w[n+k] * f[n] for k = 0, ..., (2n-1)-n = n-1
# That's exactly what we want!
```

So: `corr = scipy.signal.correlate(w, f, mode='valid')` gives us `corr[j] = sum_m w[j+m] * f[m]` for j = 0, ..., n-1.

Then `grad = 2.0 * corr`.

This is O(n * (2n-1)) = O(n^2) but highly optimized in scipy.

def project_simplex(v, target_sum=1.0):

def build_conv_sparse(f):

        if M <= 0:
            break
        # Subgradient: grad[j] = 2*f[max_idx - j] if 0 <= max_idx-j < n
        grad = np.zeros(n)
        jmin = max(0, max_idx - n + 1)
        jmax = min(max_idx, n - 1)
        grad[jmin:jmax + 1] = 2.0 * f[max_idx - jmin:max_idx - jmax + 1][::-1]
        nf = f - lr * grad
        nf = project_simplex(nf, 1.0)
        ns = evaluate_sequence(nf.tolist())
        it += 1
        if np.isfinite(ns) and ns < best_s:
            f = nf.copy()
            best_f = f.copy()
            best_s = ns
            no_impr = 0
            if it % 500 == 0:
                print(f"  {prefix}it{it}: score={best_s:.6f} (lr={lr:.5f})")
        else:
            no_impr += 1
            lr *= 0.9
            if lr < 1e-8 or no_impr > 300:
                break
    return best_f, best_s


def try_other_length(n, deadline, prefix=""):
    """Try a different sequence length with multiple initializations and LP."""
    best_f = None
    best_s = float('inf')
    idx = np.arange(n, dtype=np.float64)
    
    inits = []
    inits.append(np.ones(n))
    inits.append((1.0 + np.cos(2 * np.pi * idx / n)) * 0.5 + 0.1)
    inits.append(np.exp(-0.5 * ((idx - n/2) / (n/6)) ** 2) + 0.05)
    # U-shaped
    inits.append(0.5 + 0.5 * np.cos(np.pi * idx / (n - 1)))
    
    per_init = max(10.0, (deadline - time.monotonic()) / max(1, len(inits)))
    
    for init in inits:
        if time.monotonic() > deadline:
            break
        init = np.clip(init, 0.01, None)
        s = evaluate_sequence(init.tolist())
        if s < best_s:
            best_s = s
            best_f = init.copy()
        
        d = min(deadline, time.monotonic() + per_init)
        if d - time.monotonic() > 15:
            f, s2 = run_lp(init.copy(), d, step_init=0.12, prefix=prefix)
            if s2 < best_s:
                best_s = s2
                best_f = f.copy()
    
    return best_f, best_s


def propose_candidate(budget_s=1500, **kwargs):
    t0 = time.monotonic()
    deadline = t0 + max(10.0, float(budget_s) - 10.0)
    
    best_seq = None
    best_score = float('inf')
    
    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.all(np.isfinite(p)) and np.sum(p) > 0.01:
            parent = p.copy()
    except Exception:
        pass
    
    if parent is not None:
        s = evaluate_sequence(parent.tolist())
        if np.isfinite(s):
            best_score = s
            best_seq = parent.copy()
            print(f"Parent n={parent.size}: score={s:.6f}")
    
    if best_seq is None:
        best_seq = np.ones(512, dtype=np.float64)
        best_score = evaluate_sequence(best_seq.tolist())
        print(f"Default n=512: score={best_score:.6f}")
    
    n_parent = len(best_seq)
    
    # Phase 1: LP on parent (40% of time)
    d1 = t0 + (deadline - t0) * 0.40
    if time.monotonic() < d1:
        print("\n=== Phase 1: LP on parent ===")
        f, s = run_lp(best_seq.copy(), d1, step_init=0.15, prefix="[P1] ")
        if s < best_score:
            best_score = s
            best_seq = f.copy()
            print(f"  P1 result: {best_score:.6f}")
    
    # Phase 2: LSE gradient (20%)
    d2 = t0 + (deadline - t0) * 0.60
    if time.monotonic() < d2:
        print("\n=== Phase 2: LSE gradient ===")
        f, s = run_lse(best_seq.copy(), d2, T0=0.05, Tmin=0.01, Tdec=0.997, lr0=0.03, beta=0.9, prefix="[P2] ")
        if s < best_score:
            best_score = s
            best_seq = f.copy()
            print(f"  P2 result: {best_score:.6f}")
    
    # Phase 3: Subgradient (10%)
    d3 = t0 + (deadline - t0) * 0.70
    if time.monotonic() < d3:
        print("\n=== Phase 3: Subgradient polish ===")
        f, s = run_lse_subgradient(best_seq.copy(), d3, lr0=0.005, prefix="[P3] ")
        if s < best_score:
            best_score = s
            best_seq = f.copy()
            print(f"  P3 result: {best_score:.6f}")
    
    # Phase 4: Try other lengths (15%) - only if beneficial
    d4 = t0 + (deadline - t0) * 0.85
    if time.monotonic() < d4 and n_parent <= 1024:
        print("\n=== Phase 4: Try n=1024 ===")
        f, s = try_other_length(1024, d4, prefix="[P4-1024] ")
        if f is not None and s < best_score:
            best_score = s
            best_seq = f.copy()
            print(f"  P4 result: {best_score:.6f}")
    
    # Phase 5: Final LP + LSE polish (remaining time)
    if time.monotonic() < deadline - 5:
        print("\n=== Phase 5: Final polish ===")
        d5a = min(deadline, time.monotonic() + max(10, (deadline - time.monotonic()) * 0.6))
        f, s = run_lp(best_seq.copy(), d5a, step_init=0.05, prefix="[P5a-LP] ")
        if s < best_score:
            best_score = s
            best_seq = f.copy()
        
        if time.monotonic() < deadline - 3:
            f, s = run_lse(best_seq.copy(), deadline - 3, T0=0.02, Tmin=0.005, Tdec=0.999, lr0=0.008, beta=0.95, prefix="[P5b-LSE] ")
            if s < best_score:
                best_score = s
                best_seq = f.copy()
    
    # Final subgradient micro-polish
    if time.monotonic() < deadline:
        f, s = run_lse_subgradient(best_seq.copy(), deadline, lr0=0.001, prefix="[Final-sub] ")
        if s < best_score:
            best_score = s
            best_seq = f.copy()
    
    elapsed = time.monotonic() - t0
    print(f"\nFINAL: n={len(best_seq)}, score={best_score:.6f}, time={elapsed:.1f}s")
    return [float(x) for x in best_seq]
