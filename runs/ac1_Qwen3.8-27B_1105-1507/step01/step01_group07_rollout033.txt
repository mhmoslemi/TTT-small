"""
Hybrid soft-max gradient + LP optimization for minimizing 2*n*max(conv(a,a))/(sum(a))^2.
Uses temperature annealing with L-BFGS-B, multiple structured initializations,
and LP-based refinement to escape local optima.
"""

import numpy as np
from scipy.optimize import minimize, linprog
import time


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    conv = np.convolve(a, a)
    max_b = float(np.max(conv))
    sum_a = float(np.sum(a))
    if sum_a < 0.01:
        return np.inf
    return 2.0 * n * max_b / (sum_a * sum_a)


def soft_max_objective(a, tau):
    """
    Compute the soft-max approximation of 2n*max(conv(a,a))/sum(a)^2
    and its gradient.
    
    Parameters:
    - a: array of shape (n,)
    - tau: temperature parameter
    
    Returns:
    - objective value
    - gradient
    """
    n = len(a)
    conv = np.convolve(a, a)
    sum_a = np.sum(a)
    
    if sum_a < 1e-10:
        return 1e10, np.zeros(n)
    
    # Soft max of conv
    conv_scaled = conv / tau
    conv_max = np.max(conv_scaled)
    exp_vals = np.exp(conv_scaled - conv_max)
    sum_exp = np.sum(exp_vals)
    log_sum_exp = conv_max + np.log(sum_exp)
    S = tau * log_sum_exp  # soft max value
    
    softmax = exp_vals / sum_exp
    
    # Objective
    obj = 2.0 * n * S / (sum_a * sum_a)
    
    # Gradient
    # dS/d(a[j]) = 2 * correlate(softmax, a, 'valid')[j]
    corr = np.correlate(softmax, a, mode='valid')
    # df/d(a[j]) = 2n * dS/d(a[j]) / sum_a^2 - 2n * S * 2 * a[j] / sum_a^3
    grad = 4.0 * n * corr / (sum_a * sum_a) - 4.0 * n * S * a / (sum_a ** 3)
    
    return obj, grad


def soft_max_minimize(a_init, tau_start=0.1, tau_end=0.001, n_anneal_steps=10, max_iter=500):
    """
    Minimize the soft-max objective with temperature annealing.
    """
    n = len(a_init)
    a = a_init.copy()
    s = np.sum(a)
    if s < 1e-10:
        a = np.ones(n)
        s = n
    a = a / s  # Normalize to sum=1
    
    best_a = a.copy()
    best_obj = compute_obj(best_a)
    
    for step in range(n_anneal_steps):
        # Interpolate temperature
        frac = step / max(1, n_anneal_steps - 1)
        tau = tau_start * (tau_end / tau_start) ** frac
        
        # Minimize at current temperature
        result = minimize(
            lambda x: soft_max_objective(x, tau)[0],
            a,
            jac=lambda x: soft_max_objective(x, tau)[1],
            method='L-BFGS-B',
            bounds=[(0, None)] * n,
            options={'maxiter': max_iter, 'ftol': 1e-12, 'gtol': 1e-8}
        )
        
        if result.success or result.fun < compute_obj(result.x):
            a_new = result.x
            s_new = np.sum(a_new)
            if s_new > 1e-10:
                a_new = a_new / s_new
                obj_new = compute_obj(a_new)
                if obj_new < best_obj:
                    best_a = a_new.copy()
                    best_obj = obj_new
                a = a_new
    
    return best_a, best_obj


def build_A_matrix(a):
    """Build constraint matrix for LP: A[k,j] such that A @ g = conv(a, g)."""
    n = len(a)
    A = np.zeros((2 * n - 1, n), dtype=np.float64)
    for k in range(2 * n - 1):
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        # A[k, j] = a[k-j] for j in [j_start, j_end]
        for j in range(j_start, j_end + 1):
            A[k, j] = a[k - j]
    return A


def solve_lp_for_g(a, M):
    """Solve LP: max sum(g) s.t. conv(a,g) <= M, g >= 0, g <= 1000."""
    n = len(a)
    A = build_A_matrix(a)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, 1000.0)] * n
    res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
    if res.status == 0:
        return res.x
    return None


def lp_refine(a, max_iter=200, t=0.3):
    """LP-based refinement (similar to previous approach but optimized)."""
    n = len(a)
    a = a.copy()
    s = np.sum(a)
    if s < 1e-10:
        return a, compute_obj(a)
    a = a / s
    best_a = a.copy()
    best_obj = compute_obj(best_a)
    
    for it in range(max_iter):
        conv = np.convolve(a, a)
        M = float(np.max(conv))
        
        g = solve_lp_for_g(a, M)
        if g is None:
            break
        
        g_sum = np.sum(g)
        if g_sum <= 1.0 + 1e-6:
            break
        
        # Scale and blend
        g = g / g_sum
        a_new = (1.0 - t) * a + t * g
        a_new = np.clip(a_new, 0, None)
        s_new = np.sum(a_new)
        if s_new < 1e-10:
            break
        a_new = a_new / s_new
        
        obj_new = compute_obj(a_new)
        if obj_new < best_obj:
            best_a = a_new.copy()
            best_obj = obj_new
        a = a_new
    
    return best_a, best_obj


def structured_init(n, rng, mode=0):
    """Generate structured initializations."""
    if mode == 0:
        # Linearly decreasing ramp
        a = np.arange(n, 0, -1, dtype=np.float64)
    elif mode == 1:
        # Exponential decay
        rate = rng.uniform(0.01, 0.1)
        a = np.exp(-rate * np.arange(n))
    elif mode == 2:
        # Cosine window shifted to one side
        phase = rng.uniform(0, 1)
        x = np.linspace(0, 2 * np.pi * (1 + phase), n)
        a = np.maximum(0, np.cos(x - np.pi / 2))
    elif mode == 3:
        # Two-hump: high at start, dip in middle, medium at end
        a = np.ones(n)
        a[:n//4] *= rng.uniform(1.5, 3.0)
        a[n//4:3*n//4] *= rng.uniform(0.3, 0.7)
        a[3*n//4:] *= rng.uniform(0.5, 1.5)
    elif mode == 4:
        # Random with smoothness (low-pass filtered)
        a = rng.random(n)
        # Simple moving average for smoothing
        kernel_size = max(3, n // 20)
        kernel = np.ones(kernel_size) / kernel_size
        a = np.convolve(a, kernel, mode='same')
    elif mode == 5:
        # Quadratic (parabola) shifted
        shift = rng.uniform(0.2, 0.8)
        x = np.linspace(0, 1, n)
        a = np.maximum(0, 1 - ((x - shift) / max(0.1, shift * 0.8))**2)
    else:
        a = rng.random(n)
    
    s = np.sum(a)
    if s < 1e-10:
        a = rng.random(n)
        s = np.sum(a)
    return a / s


def frequency_init(n, rng):
    """
    Generate initialization from frequency domain.
    Create a random power spectrum, take inverse FFT, clip to non-negative.
    """
    # Random magnitude spectrum
    freqs = np.fft.rfftfreq(n)
    # Low-pass filter to get smooth sequence
    cutoff = rng.uniform(0.1, 0.5)
    mask = (np.abs(freqs) < cutoff).astype(np.float64)
    # Smooth the mask
    mask = np.convolve(mask, np.ones(5)/5, mode='same')
    
    magnitudes = rng.random(len(freqs)) * mask
    phases = rng.uniform(0, 2*np.pi, len(freqs))
    spectrum = magnitudes * np.exp(1j * phases)
    
    a = np.real(np.fft.irfft(spectrum, n=n))
    a = np.maximum(0, a)
    s = np.sum(a)
    if s < 1e-10:
        a = rng.random(n)
        s = np.sum(a)
    return a / s

The objective is f(a) = 2n * max_k (a*a)[k] / (sum a)^2.

This is scale-invariant, so WLOG sum(a) = 1. Then f(a) = 2n * max_k (a*a)[k].

Now, (a*a)[k] for k = n-1 (the "center") is:
(a*a)[n-1] = sum_{j=0}^{n-1} a[j] * a[n-1-j]

If a is symmetric (a[j] = a[n-1-j]), this equals sum_j a[j]^2, which by Cauchy-Schwarz is >= 1/n.

If a is "anti-symmetric" in some sense (a[j] is large when a[n-1-j] is small), this can be much less than 1/n.

The extreme case: a[j] = 1 for j = 0, ..., m-1 and a[j] = 0 for j = m, ..., n-1 (a "step" function).
Then (a*a)[n-1] = sum_{j=0}^{m-1} a[j] * a[n-1-j] = 0 (since n-1-j >= n-m > m-1 for j < m, assuming n > 2m).

So the central autocorrelation is 0! But the autocorrelation at other lags might be large.
