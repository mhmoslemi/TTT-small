    # This needs f as input, so let's make it parameterized
    pass

def convolve_matrix(f):
    """Build matrix A (2n-1 x n) such that (A@g)[k] = sum_j A[k,j]*g[j] = conv(f,g)[k]."""
    n = len(f)
    A = np.zeros((2*n-1, n))
    for k in range(2*n-1):
        j_min = max(0, k - (n-1))
        j_max = min(k, n-1)
        for j in range(j_min, j_max+1):
            A[k, j] = f[k-j]
    return A

def solve_min_max_lp(f, M_current):
    """
    Solve: min t s.t. conv(f,g)_k <= t for all k, sum(g) = sum(f), g >= 0.
    Returns the optimal g.
    """
    n = len(f)
    S = sum(f)
    A = convolve_matrix(f)  # (2n-1) x n
    
    # Variables: [g_0, ..., g_{n-1}, t] (length n+1)
    # Objective: minimize t -> c = [0, ..., 0, 1]
    c = np.zeros(n+1)
    c[-1] = 1.0
    
    # Inequality constraints: A@g - t <= 0 for each k
    # [A, -1] @ [g; t] <= 0 (shape: (2n-1) x (n+1))
    A_ub = np.hstack([A, -np.ones((2*n-1, 1))])
    b_ub = np.zeros(2*n-1)
    
    # Equality: sum(g) = S
    # [1, ..., 1, 0] @ [g; t] = S
    A_eq = np.zeros((1, n+1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([S])
    
    # Bounds: g_i >= 0, t >= 0
    bounds = [(0, None)] * n + [(0, None)]
    
    try:
        result = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,

        if result.success:

def convolve_matrix(f):
    """Build matrix A (2n-1 x n) such that (A@g)[k] = convolve(f,g)[k]."""
    n = len(f)

    """

        #   If j == k-j (k=2j): f[j]^2
        #   Else: 0

        # Just try f[j] = 0, f[j] = current, f[j] = some other values, and pick the best.
        
        old_val = f[j]
        # Try a few candidate values
        candidates = [old_val * 0.5, old_val * 0.8, old_val, old_val * 1.2, old_val * 1.5, 0.0]
        # Also try the value that would equalize the max
        best_score = float('inf')
        best_val = old_val
        
        conv_f = np.convolve(f, f)
        max_b = float(np.max(conv_f))
        
        for cand in candidates:
            if cand < 0:
                continue
            f[j] = cand
            # Renormalize to keep sum = S
            current_sum = float(np.sum(f))
            if current_sum <= 0:
                f[j] = old_val

            f_scaled = f * (S / current_sum)
            conv_new = np.convolve(f_scaled, f_scaled)
            score = 2 * n * float(np.max(conv_new)) / (S * S)
            if score < best_score:
                best_score = score
                best_val = cand
        f[j] = best_val
    
    # Final renormalization
    current_sum = float(np.sum(f))
    if current_sum > 0:
        f = f * (S / current_sum)
    f = np.maximum(f, 0)
    return f

Optimizes sequence to minimize 2n*max(conv(a,a))/sum(a)^2.
Strategy: LP-based direction finding (minimize peak of conv(f,g)) 
with adaptive step size, combined with smoothed subgradient polishing.
Multiple restarts and length exploration.

def build_A(f):
    """Build (2n-1) x n matrix A where (A@g)[k] = convolve(f,g)[k]."""
    n = len(f)
    A = np.zeros((2*n - 1, n))
    for j in range(n):
        # f[j] contributes to conv[k] for k in [j, j+n-1] (capped at 2n-2)
        k_start = j
        k_end = min(j + n, 2*n - 1)
        for k in range(k_start, k_end):
            A[k, j] = f[k - j]
    return A


def eval_score(f):
    """Compute the objective value (fast, no security checks)."""
    n = len(f)
    S = float(np.sum(f))
    if S < 0.01:
        return float('inf')
    conv = np.convolve(f, f)
    max_b = float(np.max(conv))
    return 2.0 * n * max_b / (S * S)

    A_eq[0, :n] = 1.0
    b_eq = np.array([S])
    
    bounds = [(0, None)] * (n + 1)
    
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                     bounds=bounds, method='highs',
                     options={'time_limit': max_time, 'presolve': True})
        if res.success:
            return res.x[:n].copy(), float(res.x[-1])
    except Exception:
        pass
    return None, None


def solve_lp_combined(f, alpha=0.1, max_time=10.0):
    """
    Solve: min c s.t. 2*alpha*conv(f,g)_k + conv(f,f)_k - c <= 0 for all k,
           sum(g) = sum(f), g >= 0.
    This directly minimizes the predicted new peak after a step of size alpha.
    Returns (g, c_opt) or (None, None).
    """
    n = len(f)
    S = float(np.sum(f))
    A = build_A(f)
    M = np.convolve(f, f)
    
    # Variables: [g (n), c (1)]
    # Objective: minimize c
    c_obj = np.zeros(n + 1)
    c_obj[-1] = 1.0
    
    # Inequality: 2*alpha*A@g - c <= -M
    A_ub = np.zeros((2*n - 1, n + 1))
    A_ub[:, :n] = 2.0 * alpha * A
    A_ub[:, -1] = -1.0
    b_ub = -M.copy()
    
    # Equality: sum(g) = S
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([S])
    
    bounds = [(0, None)] * (n + 1)
    
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                     bounds=bounds, method='highs',
                     options={'time_limit': max_time, 'presolve': True})
        if res.success:
            return res.x[:n].copy(), float(res.x[-1])
    except Exception:
        pass
    return None, None


def project_onto_simplex(v, S):
    """Project vector v onto {x >= 0, sum(x) = S}."""
    n = len(v)
    if n == 0:
        return v
    if S <= 0:
        return np.zeros(n)
    
    # Sort in decreasing order
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - S))[0]
    if len(rho) == 0:
        return np.zeros(n)
    rho = rho[-1]
    theta = (cssv[rho] - S) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def smoothed_objective_gradient(f, temperature):
    """
    Compute gradient of the temperature-smoothed objective.
    Uses log-sum-exp to approximate max(conv(f,f)).
    """
    n = len(f)
    S = float(np.sum(f))
    conv_f = np.convolve(f, f)
    
    # Soft-max weights
    max_c = float(np.max(conv_f))
    exp_vals = np.exp(temperature * (conv_f - max_c))
    weights = exp_vals / float(np.sum(exp_vals))
    
    # Gradient: sum_k weights[k] * d(conv[k])/d(f[j])
    # d(conv[k])/d(f[j]) = 2*f[k-j] if 0 <= k-j < n, else 0
    grad = np.zeros(n)
    for k in range(2*n - 1):
        w = weights[k]
        if w < 1e-12:
            continue
        # f[j] appears in conv[k] when 0 <= k-j < n
        j_min = max(0, k - (n - 1))
        j_max = min(k, n - 1)
        for j in range(j_min, j_max + 1):
            grad[j] += 2.0 * w * f[k - j]
    
    return grad


def anneal_polish(f, deadline, t_start=10.0, t_end=100.0, lr_start=0.05, lr_end=0.001):
    """
    Smoothed gradient descent with annealing temperature.
    Temperature increases over time (sharper max approximation).
    Learning rate decreases over time.
    """
    n = len(f)
    S = float(np.sum(f))
    best_f = f.copy()
    best_score = eval_score(f)
    
    n_iter = 0
    start_time = time.monotonic()
    
    while time.monotonic() < deadline:
        # Compute progress in [0,1]
        elapsed = time.monotonic() - start_time
        # Adaptive: use time-based annealing
        progress = min(1.0, elapsed / 60.0)  # Full anneal in 60 seconds
        
        t = t_start + (t_end - t_start) * progress
        lr = lr_start * (lr_end / lr_start) ** progress
        
        grad = smoothed_objective_gradient(f, t)
        
        # Gradient step with projection
        v = f - lr * grad
        new_f = project_onto_simplex(v, S)
        
        new_score = eval_score(new_f)
        n_iter += 1
        
        if np.isfinite(new_score) and new_score < best_score:
            f = new_f
            best_f = f.copy()
            best_score = new_score
            if n_iter % 500 == 0:
                print(f"    anneal iter {n_iter}: score={best_score:.6f}, t={t:.1f}, lr={lr:.5f}")
        else:
            lr *= 0.99
            if lr < 1e-7:
                break
    
    return best_f, best_score


def main_optimize(f, deadline, method='lp_minmax'):
    """
    Main optimization loop using LP direction finding with backtracking.
    """
    n = len(f)
    S = float(np.sum(f))
    best_f = f.copy()
    best_score = eval_score(f)
    current_f = f.copy()
    
    alpha = 0.3  # Initial step size
    no_improve = 0
    it = 0
    t0 = time.monotonic()
    
    while time.monotonic() < deadline:
        # Solve LP for direction
        if method == 'lp_minmax':
            g, t_opt = solve_lp_minmax(current_f, max_time=15.0)
        elif method == 'lp_combined':
            g, t_opt = solve_lp_combined(current_f, alpha=alpha, max_time=15.0)
        else:
            break
        
        if g is None:
            no_improve += 1
            if no_improve > 20:
                break
            continue
        
        # Rescale g to have same sum as current_f
        sum_g = float(np.sum(g))
        if sum_g <= 1e-15:
            no_improve += 1
            break
        g = g * (S / sum_g)
        
        # Backtracking line search
        found = False
        for _ in range(10):
            new_f = (1.0 - alpha) * current_f + alpha * g
            new_f = np.maximum(new_f, 0.0)
            
            # Ensure sum is preserved (numerical fix)
            new_sum = float(np.sum(new_f))
            if new_sum > 0:
                new_f = new_f * (S / new_sum)
            
            new_score = eval_score(new_f)
            
            if np.isfinite(new_score) and new_score < best_score:
                current_f = new_f
                best_f = new_f.copy()
                best_score = new_score
                no_improve = 0
                it += 1
                found = True
                if it % 50 == 0:
                    elapsed = time.monotonic() - t0
                    print(f"    LP iter {it}: score={best_score:.6f}, alpha={alpha:.3f}, elapsed={elapsed:.0f}s")
                break
            else:
                alpha *= 0.5
        
        if not found:
            no_improve += 1
            if no_improve > 30:
                break
        
        # Occasionally increase alpha
        if no_improve == 0 and it % 20 == 0:
            alpha = min(0.5, alpha * 1.1)
    
    return best_f, best_score


def random_perturb(f, strength=0.1, rng=None):
    """Apply random perturbation to escape local minima."""
    if rng is None:
        rng = np.random.default_rng()
    n = len(f)
    S = float(np.sum(f))
    
    # Multiplicative perturbation
    noise = rng.uniform(1.0 - strength, 1.0 + strength, size=n)
    new_f = f * noise
    new_f = np.maximum(new_f, 0.0)
    new_sum = float(np.sum(new_f))
    if new_sum > 0:
        new_f = new_f * (S / new_sum)
    return new_f

"""
Optimizes sequence to minimize 2n*max(conv(a,a))/sum(a)^2.
Uses min-max LP direction finding (different from prior max-sum LP),
combined with annealed smoothed-gradient polishing and random restarts.
Key innovation: LP directly minimizes predicted new peak, enabling
more precise convergence than the previous approach.

import numpy as np
import time
from scipy.optimize import linprog


def build_A(f):
    """Build (2n-1) x n matrix A where (A@g)[k] = convolve(f,g)[k]."""
    n = len(f)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        k_start = j
        k_end = min(j + n, 2 * n - 1)
        for k in range(k_start, k_end):
            A[k, j] = f[k - j]
    return A


def eval_score(f):
    """Fast objective computation (no security checks)."""
    n = len(f)
    S = float(np.sum(f))
    if S < 0.01:
        return float('inf')
    conv = np.convolve(f, f)
    max_b = float(np.max(conv))
    return 2.0 * n * max_b / (S * S)


def solve_lp_minmax(f, max_time=12.0):
    """
    Solve: min t s.t. conv(f,g)_k <= t for all k, sum(g)=sum(f), g>=0.
    Returns (g, t_opt) or (None, None).
    """
    n = len(f)
    S = float(np.sum(f))
    if S <= 0:
        return None, None
    A = build_A(f)

    c = np.zeros(n + 1)
    c[-1] = 1.0

    A_ub = np.zeros((2 * n - 1, n + 1))
    A_ub[:, :n] = A
    A_ub[:, -1] = -1.0
    b_ub = np.zeros(2 * n - 1)

    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([S])

    bounds = [(0, None)] * (n + 1)

                      bounds=bounds, method='highs',

        return None, None
    A = build_A(f)
    M = np.convolve(f, f)

    c_obj = np.zeros(n + 1)
    c_obj[-1] = 1.0

    A_ub = np.zeros((2 * n - 1, n + 1))
    A_ub[:, :n] = 2.0 * alpha * A
    A_ub[:, -1] = -1.0
    b_ub = -M.copy()

    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([S])

    bounds = [(0, None)] * (n + 1)

    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs',
                      options={'time_limit': max_time, 'presolve': True})
        if res.success:
            return res.x[:n].copy(), float(res.x[-1])
    except Exception:
        pass
    return None, None


def project_simplex(v, S):
    """Project v onto {x >= 0, sum(x) = S}."""
    n = len(v)
    if n == 0:
        return v
    if S <= 0:
        return np.zeros(n)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - S)
    rho_arr = np.nonzero(cond)[0]
    if len(rho_arr) == 0:
        return np.zeros(n)
    rho = rho_arr[-1]
    theta = (cssv[rho] - S) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def smoothed_grad(f, temperature):
    """Gradient of temperature-smoothed max(conv(f,f)) w.r.t. f."""
    n = len(f)
    conv_f = np.convolve(f, f)
    max_c = float(np.max(conv_f))
    exp_vals = np.exp(temperature * (conv_f - max_c))
    weights = exp_vals / float(np.sum(exp_vals))

    grad = np.zeros(n)
    for k in range(2 * n - 1):
        w = weights[k]
        if w < 1e-12:
            continue
        j_min = max(0, k - (n - 1))
        j_max = min(k, n - 1)
        for j in range(j_min, j_max + 1):
            grad[j] += 2.0 * w * f[k - j]
    return grad


def anneal_polish(f, deadline, t_start=5.0, t_end=200.0,
                  lr_start=0.03, lr_end=0.0005, anneal_time=45.0):
    """Annealed smoothed gradient descent."""
    n = len(f)
    S = float(np.sum(f))
    best_f = f.copy()
    best_score = eval_score(f)
    start_time = time.monotonic()
    it = 0

    while time.monotonic() < deadline:
        elapsed = time.monotonic() - start_time
        progress = min(1.0, elapsed / anneal_time)
        t = t_start + (t_end - t_start) * progress
        lr = lr_start * (lr_end / lr_start) ** progress

        grad = smoothed_grad(f, t)
        v = f - lr * grad
        new_f = project_simplex(v, S)
        new_score = eval_score(new_f)
        it += 1

        if np.isfinite(new_score) and new_score < best_score:
            f = new_f
            best_f = f.copy()
            best_score = new_score
            if it % 1000 == 0:
                print(f"    anneal it={it}: score={best_score:.7f} t={t:.1f} lr={lr:.6f}")
        else:
            lr *= 0.995
            if lr < 1e-8:
                break
    return best_f, best_score


def lp_optimize(f, deadline, method='minmax', alpha_init=0.25):
    """Main LP optimization loop with backtracking."""
    n = len(f)
    S = float(np.sum(f))
    best_f = f.copy()
    best_score = eval_score(f)
    cur = f.copy()
    alpha = alpha_init
    no_improve = 0
    it = 0
    t0 = time.monotonic()

    while time.monotonic() < deadline:
        if method == 'minmax':
            g, _ = solve_lp_minmax(cur, max_time=12.0)
        else:
            g, _ = solve_lp_combined(cur, alpha=alpha, max_time=12.0)

        if g is None:
            no_improve += 1
            if no_improve > 25:
                break
            continue

        sum_g = float(np.sum(g))
        if sum_g < 1e-15:
            no_improve += 1
            break
        g = g * (S / sum_g)

        found = False
        for _ in range(12):
            new_f = (1.0 - alpha) * cur + alpha * g
            new_f = np.maximum(new_f, 0.0)
            ns = float(np.sum(new_f))
            if ns > 0:
                new_f = new_f * (S / ns)
            sc = eval_score(new_f)
            if np.isfinite(sc) and sc < best_score:
                cur = new_f
                best_f = new_f.copy()
                best_score = sc
                no_improve = 0
                it += 1
                found = True
                if it % 30 == 0:
                    el = time.monotonic() - t0
                    print(f"    LP[{method}] it={it}: score={best_score:.7f} a={alpha:.3f} t={el:.0f}s")
                break
            else:
                alpha *= 0.5

        if not found:
            no_improve += 1
            if no_improve > 35:
                break
        elif it % 15 == 0:
            alpha = min(0.45, alpha * 1.05)

    return best_f, best_score


def perturb(f, strength, rng):
    """Multiplicative random perturbation."""
    S = float(np.sum(f))
    noise = rng.uniform(1.0 - strength, 1.0 + strength, size=len(f))
    nf = f * noise
    nf = np.maximum(nf, 0.0)
    ns = float(np.sum(nf))
    if ns > 0:
        nf = nf * (S / ns)
    return nf


def propose_candidate(budget_s=1500, **kwargs):
    t0 = time.monotonic()
    deadline = t0 + max(0.0, float(budget_s) - 12.0)
    best_sequence = None
    best_score = float('inf')
    rng = np.random.default_rng(7)

    # Load parent
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64).copy()
        parent = np.clip(parent, 0.0, 1000.0)
        if parent.ndim == 1 and parent.size > 0 and np.all(np.isfinite(parent)):
            sc = eval_score(parent)
            if np.isfinite(sc):
                best_score = sc
                best_sequence = parent.copy()
                print(f"Parent n={parent.size}: score={best_score:.7f}")
    except Exception as e:
        print(f"Parent load failed: {e}")

    if best_sequence is None:
        best_sequence = np.ones(512)
        best_score = eval_score(best_sequence)
        print(f"Using default uniform n=512: score={best_score:.7f}")

    # Phase 1: LP minmax (35% of time)
    if time.monotonic() < deadline:
        d1 = t0 + (deadline - t0) * 0.35
        print(f"\n=== Phase 1: LP min-max (n={len(best_sequence)}) ===")
        bf, bs = lp_optimize(best_sequence, d1, method='minmax', alpha_init=0.25)
        if bs < best_score:
            best_score = bs
            best_sequence = bf.copy()
            print(f"  After LP minmax: {best_score:.7f}")

    # Phase 2: LP combined (20% of time)
    if time.monotonic() < deadline:
        d2 = t0 + (deadline - t0) * 0.55
        print(f"\n=== Phase 2: LP combined (n={len(best_sequence)}) ===")
        bf, bs = lp_optimize(best_sequence, d2, method='combined', alpha_init=0.15)
        if bs < best_score:
            best_score = bs
            best_sequence = bf.copy()
            print(f"  After LP combined: {best_score:.7f}")

    # Phase 3: Annealed gradient polish (15% of time)
    if time.monotonic() < deadline:
        d3 = t0 + (deadline - t0) * 0.70
        print(f"\n=== Phase 3: Anneal polish (n={len(best_sequence)}) ===")
        bf, bs = anneal_polish(best_sequence, d3)
        if bs < best_score:
            best_score = bs
            best_sequence = bf.copy()
            print(f"  After anneal: {best_score:.7f}")

    # Phase 4: Random restarts + short re-optimization (20% of time)
    if time.monotonic() < deadline:
        d4 = deadline - 25
        print(f"\n=== Phase 4: Random restarts (n={len(best_sequence)}) ===")
        n_restarts = 30
        for r in range(n_restarts):
            if time.monotonic() > d4:
                break
            strength = 0.03 + 0.12 * (r / n_restarts)
            fp = perturb(best_sequence, strength, rng)
            sp = eval_score(fp)
            if sp < best_score:
                best_score = sp
                best_sequence = fp.copy()

            sub_dl = min(d4, time.monotonic() + 20.0)
            bf, bs = lp_optimize(fp, sub_dl, method='minmax', alpha_init=0.2)
            if bs < best_score:
                best_score = bs
                best_sequence = bf.copy()
                print(f"  Restart {r} (str={strength:.3f}): {best_score:.7f}")

    # Phase 5: Final polish
    if time.monotonic() < deadline - 8:
        print(f"\n=== Phase 5: Final polish ===")
        bf, bs = lp_optimize(best_sequence, deadline - 5, method='minmax', alpha_init=0.1)
        if bs < best_score:
            best_score = bs
            best_sequence = bf.copy()
        if time.monotonic() < deadline:
            bf, bs = anneal_polish(best_sequence, deadline, t_start=50.0, t_end=500.0,
                                   lr_start=0.01, lr_end=0.0001, anneal_time=20.0)
            if bs < best_score:
                best_score = bs
                best_sequence = bf.copy()

    # Verify with actual evaluator
    try:
        final_check = evaluate_sequence([float(x) for x in best_sequence])
        if np.isfinite(final_check) and final_check < best_score:
            best_score = final_check
        elif not np.isfinite(final_check):
            print(f"WARNING: final eval returned {final_check}, using fallback")
            best_sequence = np.ones(128)
            best_score = eval_score(best_sequence)
    except Exception:
        pass

    elapsed = time.monotonic() - t0
    print(f"\nFINAL: n={len(best_sequence)}, score={best_score:.7f}, time={elapsed:.1f}s")
    return [float(x) for x in best_sequence]
