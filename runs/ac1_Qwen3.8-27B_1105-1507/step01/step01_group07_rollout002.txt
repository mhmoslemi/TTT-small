Multi-method optimization for minimizing 2*n*max(conv(a,a))/(sum(a))^2.
Combines subgradient descent, LP-based alternation, and structured random search.

import numpy as np
from scipy.optimize import linprog
import time

def compute_obj(a):
    n = len(a)
    if n == 0:
        return np.inf
    conv = np.convolve(a, a)
    max_b = float(np.max(conv))
    sum_a = float(np.sum(a))
    if sum_a < 0.01:
        return np.inf
    return 2.0 * n * max_b / (sum_a * sum_a)

def fft_convolve(a, b):
    """Fast convolution using FFT."""
    n = len(a) + len(b) - 1
    size = 1
    while size < n:
        size *= 2
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    return np.fft.irfft(fa * fb, size)[:n]

def subgradient_step(a, lr=0.01):
    """One subgradient descent step to minimize max(conv(a,a)) subject to sum(a)=1, a>=0."""
    n = len(a)
    conv = np.convolve(a, a)
    max_b = np.max(conv)
    # Find active set (within tolerance)
    tol = 0.05 * max_b
    active = conv >= (max_b - tol)
    
    # Compute subgradient
    # d(conv)_k / da_i = a_{k-i} + a_{i-k} (with bounds)
    subgrad = np.zeros(n)
    for k in range(2*n-1):
        if active[k]:
            for i in range(n):
                # a_{k-i} if 0<=k-i<n
                if 0 <= k-i < n:
                    subgrad[i] += a[k-i]
                # a_{i-k} if 0<=i-k<n
                if 0 <= i-k < n:
                    subgrad[i] += a[i-k]
    
    # Popov direction (to avoid oscillation)
    a = a - lr * subgrad
    a = np.clip(a, 0, None)
    s = np.sum(a)
    if s < 1e-10:
        a = np.ones(n) / n
    else:
        a = a / s
    return a, max_b

def run_subgradient(n, seed=0, max_iter=10000, lr_init=0.05, verbose=False):
    rng = np.random.default_rng(seed)
    a = rng.random(n)
    a = a / np.sum(a)
    best_a = a.copy()
    best_obj = compute_obj(best_a)
    
    lr = lr_init
    for it in range(max_iter):
        a, max_b = subgradient_step(a, lr)
        obj = compute_obj(a)
        if obj < best_obj:
            best_a = a.copy()
            best_obj = obj
            # Decrease learning rate slowly
            lr *= 0.999
        else:
            lr *= 0.95
    
    return best_a, best_obj

Multi-method optimizer combining smooth gradient descent, LP-based alternation,
and structured random search to minimize 2*n*max(conv(a,a))/(sum(a))^2.
"""

import numpy as np
from scipy.optimize import linprog
import time

def compute_obj(a):
    """Compute objective value."""
    n = len(a)
    if n == 0:
        return np.inf
    conv = np.convolve(a, a)
    max_b = float(np.max(conv))
    sum_a = float(np.sum(a))
    if sum_a < 0.01:
        return np.inf
    return 2.0 * n * max_b / (sum_a * sum_a)

def smooth_max_obj(a, tau=0.1):
    """Smooth approximation of the objective using log-sum-exp."""
    n = len(a)
    conv = np.convolve(a, a)
    c_max = np.max(conv)
    log_sum_exp = c_max + tau * np.log(np.sum(np.exp((conv - c_max) / tau)))
    sum_a = np.sum(a)
    return 2.0 * n * log_sum_exp / (sum_a * sum_a)

def smooth_grad(a, tau=0.1):
    """Compute gradient of smooth objective."""
    n = len(a)
    conv = np.convolve(a, a)
    c_max = np.max(conv)
    weights = np.exp((conv - c_max) / tau)
    weights /= np.sum(weights)
    sum_a = np.sum(a)
    log_sum_exp = c_max + tau * np.log(np.sum(weights))
    
    # grad of log_sum_exp w.r.t. a_i
    grad_lse = np.zeros(n)
    for i in range(n):
        # d(log_sum_exp)/da_i = sum_k weights[k] * (a[k-i] + a[i-k])
        s = 0.0
        for k in range(2*n-1):
            if weights[k] < 1e-10:
                continue
            if 0 <= k-i < n:
                s += weights[k] * a[k-i]
            if 0 <= i-k < n:
                s += weights[k] * a[i-k]
        grad_lse[i] = s
    
    # d(2n*lse/sum_a^2)/da_i = 2n*grad_lse[i]/sum_a^2 - 4n*lse/sum_a^3
    obj_grad = 2.0 * n * grad_lse / (sum_a * sum_a) - 4.0 * n * log_sum_exp / (sum_a**3)
    return obj_grad

"""
Multi-method optimizer combining LP-based alternation, smooth gradient descent,
and structured random search for minimizing 2*n*max(conv(a,a))/(sum(a))^2.
Uses core-edge decomposition and adaptive smoothing for efficient optimization.
"""

import numpy as np
from scipy.optimize import linprog
import time

# --- Core computation ---

def compute_obj(a):
    n = len(a)
    if n == 0:
        return np.inf
    conv = np.convolve(a, a)
    max_b = float(np.max(conv))
    sum_a = float(np.sum(a))
    if sum_a < 0.01:
        return np.inf
    return 2.0 * n * max_b / (sum_a * sum_a)

def normalize(a):
    s = np.sum(a)
    if s < 1e-10:
        return a
    return a / s

# --- Initialization ---

def init_random(n, rng):
    a = rng.random(n)
    return normalize(a)

def init_cosine(n, rng):
    phase = rng.uniform(0, 2*np.pi)
    freq = rng.uniform(0.5, 2.0)
    a = np.maximum(0.01, np.cos(2*np.pi*freq*np.arange(n)/n + phase))
    return normalize(a)

def init_bell(n, rng):
    center = n / 2
    width = rng.uniform(0.2, 0.5) * n
    x = np.arange(n)
    a = np.exp(-0.5 * ((x - center) / width) ** 2)
    a += 0.01 * rng.random(n)
    return normalize(a)

def init_u_shape(n, rng):
    x = np.arange(n)
    t = x / (n - 1)
    a = t**2 + (1-t)**2 + 0.1*(rng.random(n))
    return normalize(a)

def init_flat_with_noise(n, rng):
    a = np.ones(n) + 0.3 * rng.standard_normal(n)
    a = np.abs(a) + 0.01
    return normalize(a)

def init_parent_based(n, parent, rng):
    """Interpolate or resample parent to length n, add noise."""
    if len(parent) == n:
        a = parent.copy()
        a += 0.05 * rng.standard_normal(n) * np.mean(parent)
    else:
        # Linear interpolation
        old_x = np.linspace(0, 1, len(parent))
        new_x = np.linspace(0, 1, n)
        a = np.interp(new_x, old_x, parent)
        a += 0.05 * rng.standard_normal(n) * np.mean(a)
    a = np.clip(a, 0, None)
    return normalize(a)

# --- LP-based alternating ---

def build_A_sparse(n):
    """Build the (2n-1) x n convolution matrix."""
    A = np.zeros((2*n-1, n))
    for k in range(2*n-1):
        for j in range(n):
            idx = k - j
            if 0 <= idx < n:
                A[k, j] = 1.0  # placeholder, we'll multiply by a later
    return A

def build_A_for_a(a):
    """Build A such that A @ g = convolve(a, g)."""
    n = len(a)
    A = np.zeros((2*n-1, n))
    for k in range(2*n-1):
        for j in range(n):
            idx = k - j
            if 0 <= idx < n:
                A[k, j] = a[idx]
    return A

def solve_lp(a, ub=None):
    """Solve LP: max sum(g) s.t. conv(a,g) <= M, 0<=g<=ub."""
    n = len(a)
    conv = np.convolve(a, a)
    M = float(np.max(conv))
    
    if ub is None:
        ub = 2.0  # relative to sum(a)=1, so each a_i <= 2/n roughly
    
    A = build_A_for_a(a)
    c = -np.ones(n)
    b_ub = np.full(2*n-1, M)
    bounds = [(0.0, ub)] * n
    
    res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
    if res.status == 0:
        return res.x, float(-res.fun), M
    return None, None, M

def run_lp_alternating(a, max_iter=300, t=0.5, ub=None, verbose=False):
    """LP-based alternating optimization."""
    n = len(a)
    best_a = a.copy()
    best_obj = compute_obj(best_a)
    
    for it in range(max_iter):
        g, g_sum, M = solve_lp(a, ub)
        if g is None:
            break
        
        if g_sum <= 1.0 + 1e-4:
            # Near optimal for this LP
            break
        
        # Blend
        g_norm = g / g_sum
        a_new = (1 - t) * a + t * g_norm
        a_new = np.clip(a_new, 0, None)
        a_new = normalize(a_new)
        
        obj = compute_obj(a_new)
        if obj < best_obj:
            best_obj = obj
            best_a = a_new.copy()
            a = a_new
        else:
            # Try smaller step
            t2 = t * 0.3
            a_new2 = (1-t2) * a + t2 * g_norm
            a_new2 = np.clip(a_new2, 0, None)
            a_new2 = normalize(a_new2)
            obj2 = compute_obj(a_new2)
            if obj2 < best_obj:
                best_obj = obj2
                best_a = a_new2.copy()
            a = a_new2
    
    return best_a, best_obj

# --- Smooth gradient descent ---

def smooth_gd_step(a, tau=0.1, lr=0.1, momentum=None, beta=0.9):
    """One step of smooth gradient descent."""
    n = len(a)
    conv = np.convolve(a, a)
    c_max = np.max(conv)
    shifted = (conv - c_max) / tau
    weights = np.exp(shifted)
    w_sum = np.sum(weights)
    weights /= w_sum
    log_sum_exp = c_max + tau * np.log(w_sum)
    sum_a = np.sum(a)
    
    # Gradient of log_sum_exp w.r.t. a
    # Use correlate and convolve
    h = np.correlate(weights, a, mode='full')[:n]
    g = np.convolve(weights, a, mode='full')[:n]
    grad_lse = h + g
    
    # Objective gradient
    obj_grad = 2.0 * n * grad_lse / (sum_a * sum_a) - 4.0 * n * log_sum_exp / (sum_a**3)
    
    if momentum is None:
        momentum = np.zeros(n)
    momentum = beta * momentum + (1.0 - beta) * obj_grad
    
    a_new = a - lr * momentum
    a_new = np.clip(a_new, 0, None)
    a_new = normalize(a_new)
    
    return a_new, momentum, compute_obj(a_new)

def run_smooth_gd(a, seed=0, max_iter=50000, tau_init=1.0, lr_init=0.5):
    """Run smooth gradient descent with adaptive parameters."""
    n = len(a)
    best_a = a.copy()
    best_obj = compute_obj(best_a)
    momentum = np.zeros(n)
    beta = 0.9
    lr = lr_init
    no_improve = 0
    
    for it in range(max_iter):
        tau = max(0.01, tau_init * (0.9997 ** it))
        
        a_new, momentum, obj = smooth_gd_step(a, tau, lr, momentum, beta)
        
        if obj < best_obj:
            best_obj = obj
            best_a = a_new.copy()
            no_improve = 0
        else:
            no_improve += 1
            lr *= 0.95
            momentum *= 0.5
        
        a = a_new
        
        if no_improve > 500:
            break
    
    return best_a, best_obj

# --- Random perturbation ---

def random_perturb(a, seed=0, max_iter=10000, noise_scale=0.1):
    """Random block perturbation local search."""
    rng = np.random.default_rng(seed)
    n = len(a)
    best_a = a.copy()
    best_obj = compute_obj(best_a)
    
    for it in range(max_iter):
        # Random block
        width = max(1, int(rng.integers(1, max(2, n // 20))))
        start = int(rng.integers(0, n - width))
        
        candidate = best_a.copy()
        candidate[start:start+width] *= np.exp(rng.normal(0, noise_scale, width))
        candidate = np.clip(candidate, 0, None)
        candidate = normalize(candidate)
        
        obj = compute_obj(candidate)
        if obj < best_obj:
            best_obj = obj
            best_a = candidate
    
    return best_a, best_obj

def propose_candidate(seed=42, budget_s=1500, **kwargs):
    """
    Multi-method optimizer combining LP-based alternation, smooth gradient descent,
    and structured random search.
    """
    rng = np.random.default_rng(seed)
    start_time = time.monotonic()
    deadline = start_time + float(budget_s) - 10.0
    
    best_seq = None
    best_score = np.inf
    
    # Try parent
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64)
        parent = np.clip(parent, 0, 1000.0)
        if parent.size > 0 and np.isfinite(parent).all():
            obj = compute_obj(parent)
            if obj < best_score:
                best_score = obj
                best_seq = parent.copy()
            print(f"Parent (n={parent.size}): obj={obj:.6f}")
    except Exception:
        parent = None
    
    # List of n values to try
    n_values = [100, 130, 150, 180, 200, 250, 300, 400]
    
    n_iter = 0
    for n in n_values:
        if time.monotonic() > deadline:
            break
        remaining = deadline - time.monotonic()
        print(f"N={n}: {remaining:.0f}s remaining, best={best_score:.6f}")
        
        # Multiple restarts for this n
        num_restarts = 3 if n <= 200 else 2
        for restart in range(num_restarts):
            if time.monotonic() > deadline:
                break
            
            # Choose initialization
            init_method = rng.integers(0, 5)
            if parent is not None and init_method == 0:
                a = init_parent_based(n, parent, rng)
            elif init_method == 1:
                a = init_cosine(n, rng)
            elif init_method == 2:
                a = init_bell(n, rng)
            elif init_method == 3:
                a = init_u_shape(n, rng)
            elif init_method == 4:
                a = init_flat_with_noise(n, rng)
            else:
                a = init_random(n, rng)
            
            # LP-based alternating (main engine)
            lp_iters = int(min(300, max(20, remaining / 2.0)))
            a, obj = run_lp_alternating(a, max_iter=lp_iters, t=0.5, verbose=False)
            
            # Smooth GD refinement
            gd_iters = int(min(30000, max(5000, remaining / 1.0)))
            a, obj = run_smooth_gd(a, seed=seed+n*100+restart, max_iter=gd_iters,
                                    tau_init=0.5, lr_init=0.3)
            
            # Random perturbation
            rp_iters = int(min(5000, max(1000, remaining / 2.0)))
            a, obj = random_perturb(a, seed=seed+n*200+restart, max_iter=rp_iters,
                                     noise_scale=0.08)
            
            if obj < best_score:
                best_score = obj
                best_seq = a.copy()
                print(f"  n={n}, r={restart}: obj={best_score:.6f} (method={init_method})")
            
            n_iter += 1
    
    # Final intensive refinement on best solution
    if best_seq is not None and time.monotonic() < deadline - 30:
        remaining = deadline - time.monotonic()
        print(f"Final refinement: {remaining:.0f}s remaining")
        
        # More LP iterations
        lp_iters = int(min(200, remaining / 0.5))
        a, obj = run_lp_alternating(best_seq, max_iter=lp_iters, t=0.3)
        if obj < best_score:
            best_score = obj
            best_seq = a.copy()
            print(f"  After final LP: obj={best_score:.6f}")
        
        # More smooth GD
        if time.monotonic() < deadline - 10:
            gd_iters = int((deadline - time.monotonic()) / 0.005)
            a, obj = run_smooth_gd(best_seq, seed=seed+99999, max_iter=min(gd_iters, 100000),
                                    tau_init=0.1, lr_init=0.1)
            if obj < best_score:
                best_score = obj
                best_seq = a.copy()
                print(f"  After final GD: obj={best_score:.6f}")
        
        # Final random perturbation
        if time.monotonic() < deadline - 5:
            rp_iters = int((deadline - time.monotonic()) / 0.002)
            a, obj = random_perturb(best_seq, seed=seed+88888, max_iter=min(rp_iters, 50000),
                                     noise_scale=0.03)
            if obj < best_score:
                best_score = obj
                best_seq = a.copy()
                print(f"  After final RP: obj={best_score:.6f}")

        best_score = compute_obj(best_seq)
