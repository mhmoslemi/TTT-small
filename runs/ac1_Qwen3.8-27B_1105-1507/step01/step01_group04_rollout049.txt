def _subgrad_simplex(f, active):
    """Compute subgradient of max over active lags of (f*f)_k."""
    n = len(f)
    g = np.zeros(n)
    for k in active:
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        if j_end >= j_start:
            idx = np.arange(j_start, j_end + 1)
            src = k - idx
            g[idx] += 2.0 * f[src]
    g /= max(1, len(active))
    return g

def _lp_oracle(f, M):
    """LP oracle: find g maximizing sum(g) subject to (f*g)_k <= M, g >= 0."""
    n = len(f)
    row_idx = []
    col_idx = []
    vals = []
    for j in range(n):
        rows_j = np.arange(j, j + n)
        row_idx.extend(rows_j.tolist())
        col_idx.extend([j] * n)
        vals.extend(f.tolist())
    
    C = coo_matrix((vals, (row_idx, col_idx)), shape=(2 * n - 1, n))
    c_obj = -np.ones(n)
    res = linprog(c_obj, A_ub=C, b_ub=np.full(2 * n - 1, M),
                  bounds=[(0, None)] * n, method='highs')
    if not res.success:
        return None
    g = res.x
    Sg = float(np.sum(g))
    if Sg < 1e-10:
        return None
    return g / Sg

"""
Optimizes non-negative step function heights to minimize 2n*max(conv(f,f))/(sum(f))^2.
Strategy: Multi-phase optimization with diverse initializations, momentum-based
mirror descent, LP-based Frank-Wolfe refinement, and annealing-style restarts.
Key differences from previous approach: momentum smoothing, more diverse inits,
adaptive multi-phase time allocation, and annealing perturbations.
"""

import numpy as np
import time
from scipy.optimize import linprog
from scipy.sparse import coo_matrix

_global_deadline = 0.0

def _fft_convolve_full(f):
    n = len(f)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    F = np.fft.rfft(f, size)
    h = np.fft.irfft(F * F, size)
    return h[:2 * n - 1]

def _score_raw(f):
    n = len(f)
    h = _fft_convolve_full(f)
    M = float(np.max(h))
    S = float(np.sum(f))
    if S < 0.01:
        return float('inf')
    return 2.0 * n * M / (S * S)

def _normalize(f):
    f = np.maximum(f.astype(np.float64), 0.0)
    s = float(np.sum(f))
    if s < 1e-30:
        return np.ones(len(f), dtype=np.float64) / len(f)
    return f / s

def _subgrad_simplex(f, active):
    n = len(f)
    g = np.zeros(n, dtype=np.float64)
    for k in active:
        j_start = max(0, int(k) - n + 1)
        j_end = min(n - 1, int(k))
        if j_end >= j_start:
            idx = np.arange(j_start, j_end + 1)
            src = int(k) - idx
            g[idx] += 2.0 * f[src]
    g /= max(1, len(active))
    return g

def _mirror_descent_momentum(f_init, n_iter, eta_scale, threshold_frac, momentum, patience, restart_frac):
    global _global_deadline
    f = _normalize(f_init.copy())
    n = len(f)
    best_f = f.copy()
    best_score = _score_raw(f)
    v = np.zeros(n, dtype=np.float64)
    it_no_improve = 0

    for it in range(n_iter):
        if time.monotonic() > _global_deadline:
            break

        h = _fft_convolve_full(f)
        M = float(np.max(h))
        if M <= 0:
            break

        threshold = threshold_frac * M + 1e-15
        active = np.where(h >= threshold)[0]
        if len(active) == 0:
            active = np.array([int(np.argmax(h))], dtype=np.int64)

        g = _subgrad_simplex(f, active)

        # Adaptive step size: smaller as we go
        progress = it / max(1, n_iter)
        eta = eta_scale * (1.0 - 0.5 * progress) / (M * (1.0 + 0.1 * progress))

        # Momentum update
        v = momentum * v + (1.0 - momentum) * (-g)

        log_ratio = eta * v
        log_ratio = np.clip(log_ratio, -20.0, 20.0)

        f_new = f * np.exp(log_ratio)
        f_new = _normalize(f_new)

        score = _score_raw(f_new)
        if score < best_score:
            best_score = score
            best_f = f_new.copy()
            it_no_improve = 0
        else:
            it_no_improve += 1

        f = f_new

        # Annealing restart
        if patience > 0 and it_no_improve >= patience:
            rng = np.random.default_rng(hash((it, n)) & 0xFFFFFFFF)
            noise = rng.random(n, dtype=np.float64) * restart_frac
            f = _normalize(f * (1.0 + noise))
            v = np.zeros(n, dtype=np.float64)
            it_no_improve = 0

    return best_f, best_score

def _lp_oracle(f, M):
    n = len(f)
    row_idx = []
    col_idx = []
    vals = []
    f_list = f.tolist()
    for j in range(n):
        rows_j = list(range(j, j + n))
        row_idx.extend(rows_j)
        col_idx.extend([j] * n)
        vals.extend(f_list)
    C = coo_matrix((vals, (row_idx, col_idx)), shape=(2 * n - 1, n))
    c_obj = -np.ones(n, dtype=np.float64)
    b_ub = np.full(2 * n - 1, M, dtype=np.float64)
    bounds = [(0.0, None)] * n
    res = linprog(c_obj, A_ub=C, b_ub=b_ub, bounds=bounds, method='highs')
    if not res.success or res.x is None:
        return None
    g = np.asarray(res.x, dtype=np.float64)
    Sg = float(np.sum(g))
    if Sg < 1e-10:
        return None
    return g / Sg

            break

        M = float(np.max(h))

        g = _lp_oracle(f, M)
        if g is None:
            break

        # Strongly separating FW step size
        t = 2.0 / (it + 2.0)
        f_new = (1.0 - t) * f + t * g
        f = _normalize(f_new)

        score = _score_raw(f)
        if score < best_score:
            best_score = score
            best_f = f.copy()

    return best_f, best_score

                f_p, s_p = _mirror_descent_momentum(f_pert, n_it_p, 0.35, 0.85, 0.85, 120, 0.02)
                if s_p < best_score:
                    best_score = s_p
                    best_f = f_p.copy()
                    print("  ** New best (pert=%.2f): %.6f **" % (pert, best_score))
                f_cur = f_p
                s_cur = s_p

    # Phase 3: LP-based Frank-Wolfe on top candidates (only for moderate n)
    print("\nPhase 3: LP Frank-Wolfe refinement")
    # Re-sort current bests
    # Use the current best and one other top candidate
    fw_candidates = []
    if len(best_f) > 0:
        fw_candidates.append(("best", best_f.copy()))
    for i in range(min(3, len(results))):
        if len(results[i][2]) <= 2048:
            fw_candidates.append(("fp3_%s" % results[i][1], results[i][2].copy()))

    seen = set()
    for name, fc in fw_candidates:
        if fc.shape[0] in seen:
            continue
        seen.add(fc.shape[0])
        if time.monotonic() > _global_deadline - 30:
            break
        n = len(fc)
        if n > 2500:
            continue

        lp_budget = min(120.0, (_global_deadline - time.monotonic()) * 0.25)
        if lp_budget < 15:
            break

        print("  FW on %s (n=%d), budget=%.0fs" % (name, n, lp_budget))
        f_fw, s_fw = _frank_wolfe_lp(fc, max_iters=40, time_budget=lp_budget)
        if s_fw < best_score:
            best_score = s_fw
            best_f = f_fw.copy()
            print("  ** New best from FW: %.6f **" % best_score)

    # Phase 4: Final polish - more MD iterations on the best
    if time.monotonic() < _global_deadline - 10:
        remaining = _global_deadline - time.monotonic() - 5
        print("\nPhase 4: Final polish (remaining=%.0fs)" % remaining)
        n_it_final = int(max(1000, min(10000, remaining * 500)))

        for tf_final in [0.9, 0.95, 0.8]:
            if time.monotonic() > _global_deadline - 5:
                break
            f_fin, s_fin = _mirror_descent_momentum(best_f, n_it_final // 3, 0.15, tf_final, 0.9, 200, 0.0)
            if s_fin < best_score:
                best_score = s_fin
                best_f = f_fin.copy()
                print("  ** Final polish (tf=%.2f): %.6f **" % (tf_final, best_score))

    # Final verification
    final_list = [float(x) for x in best_f]
    final_score = float(evaluate_sequence(final_list))
    print("\nFinal verified score: %.6f (n=%d)" % (final_score, len(final_list)))

    if not np.isfinite(final_score) or final_score > 2.0:
        print("  Falling back to parent.")
        return [float(x) for x in parent]

    return final_list
