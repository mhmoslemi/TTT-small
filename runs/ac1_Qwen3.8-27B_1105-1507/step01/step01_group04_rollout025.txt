def _fft_convolve_full(f):

import numpy as np
import time


def _fft_convolve_full(f):
    """Compute full linear convolution of f with itself using FFT."""
    n = len(f)
    size = 2
    while size < 2 * n - 1:
        size <<= 1
    F = np.fft.rfft(f, size)
    h = np.fft.irfft(F * F, size)
    return h[:2 * n - 1]


def _score_normalized(f):
    """Score for normalized f (sum=1): 2n * max(conv(f,f))."""
    n = len(f)
    h = _fft_convolve_full(f)
    M = float(np.max(h))
    return 2.0 * n * M


def _subgradient_fft(f, active_set):
    """Compute subgradient of (2/|K|) * sum_{k in K} conv(f,f)[k] using FFT."""
    n = len(f)
    h = f[::-1].copy()
    v = np.zeros(2 * n - 1)
    v[active_set] = 1.0
    size = 2
    while size < (2 * n - 1) + n - 1:
        size <<= 1
    V = np.fft.rfft(v, size)
    H = np.fft.rfft(h, size)
    conv_result = np.fft.irfft(V * H, size)
    sg = conv_result[n - 1: n - 1 + n]
    sg *= (2.0 / float(len(active_set)))
    return sg


def _project_simplex(v):
    """Project vector v onto the probability simplex {x >= 0, sum x = 1}."""
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - 1))[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho = rho[-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def _active_set(h, threshold_frac):
    """Get active set: indices where h is within threshold_frac of max."""
    M = np.max(h)
    threshold = threshold_frac * M + 1e-15
    active = np.where(h >= threshold)[0]
    if len(active) == 0:
        active = np.array([np.argmax(h)], dtype=np.int64)
    return active


def _accelerated_pgd(f_init, max_iters, threshold_frac, init_step, time_limit, label=""):
    """
    Nesterov-accelerated projected gradient descent for min 2n*max(conv(f,f))
    subject to f in simplex.
    """
    f = _project_simplex(f_init.copy())
    n = len(f)
    best_f = f.copy()
    h = _fft_convolve_full(f)
    best_score = float(2.0 * n * np.max(h))
    
    f_prev = f.copy()
    t = 1.0
    eta = init_step
    start_time = time.monotonic()
    
    for it in range(max_iters):
        if time.monotonic() - start_time > time_limit:
            break
        
        h = _fft_convolve_full(f)
        M = float(np.max(h))
        score = 2.0 * n * M
        if score < best_score:
            best_score = score
            best_f = f.copy()
        
        active = _active_set(h, threshold_frac)
        sg = _subgradient_fft(f, active)
        
        # Nesterov momentum
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t * t)) / 2.0
        beta = (t - 1.0) / t_new
        y = f + beta * (f - f_prev)
        
        # Backtracking line search
        y_proj = _project_simplex(y)
        step = eta
        found = False
        for _ in range(30):
            f_new = _project_simplex(y_proj - step * sg)
            if np.sum(f_new) < 0.01:
                f_new = _project_simplex(y_proj)
            h_new = _fft_convolve_full(f_new)
            M_new = float(np.max(h_new))

            # Armijo condition (simplified: just check improvement)

                found = True
                break
            step *= 0.5
            if step < 1e-15:
                break
        
        if not found:

        f_prev = f
        f = f_new
        t = t_new
        
        # Adaptive step size: slowly increase if we're making progress
        if it % 100 == 0 and it > 0:
            eta *= 1.05
    
    return best_f, best_score

def _normalize(f):
    """Normalize to sum 1, clip negatives."""
    f = np.maximum(f, 0.0)
    s = float(np.sum(f))
    if s < 1e-30:
        return None
    return f / s


def _cosine_init(n):
    j = np.arange(n, dtype=np.float64)
    f = np.cos(np.pi * (j + 0.5) / float(n)) ** 2
    f = np.maximum(f, 0.0)
    s = np.sum(f)
    if s > 0:
        f /= s
    return f


def _gaussian_init(n, sigma_frac):
    j = np.arange(n, dtype=np.float64) - (n - 1) / 2.0
    sigma = sigma_frac * float(n)
    f = np.exp(-j * j / (2.0 * sigma * sigma))
    s = np.sum(f)
    if s > 0:
        f /= s
    return f


def _tapered_init(n, power):
    j = np.arange(n, dtype=np.float64)
    x = np.sin(np.pi * (j + 0.5) / float(n))
    f = x ** power
    f = np.maximum(f, 0.0)
    s = np.sum(f)
    if s > 0:
        f /= s
    return f


def _fejer_init(n):
    """Fejér-kernel-like initialization: |sin(n*x/2)/sin(x/2)|^2 normalized."""
    j = np.arange(n, dtype=np.float64)
    x = 2.0 * np.pi * (j + 0.5) / float(n)
    # Fejer kernel: (1/(n)) * (sin(n*x/2)/sin(x/2))^2
    sin_half = np.sin(x / 2.0)
    sin_n_half = np.sin(n * x / 2.0)
    # Avoid division by zero
    mask = np.abs(sin_half) > 1e-10
    f = np.zeros(n)
    f[mask] = (sin_n_half[mask] / sin_half[mask]) ** 2
    f[~mask] = float(n)
    f = np.maximum(f, 0.0)
    s = np.sum(f)
    if s > 0:
        f /= s
    return f


def _bump_oscillate_init(n, oscillations):
    """Bell-shaped with oscillations to flatten autocorrelation."""
    j = np.arange(n, dtype=np.float64) - (n - 1) / 2.0
    gaussian = np.exp(-j * j / (2.0 * (0.35 * n) ** 2))
    oscillation = 1.0 + 0.3 * np.cos(2.0 * np.pi * oscillations * j / float(n))
    f = gaussian * oscillation
    f = np.maximum(f, 0.0)
    s = np.sum(f)
    if s > 0:
        f /= s
    return f

            break
        
        print(f"\n  Phase 2: Refining {name} (score={s_cur:.6f}), budget={alloc:.0f}s")
        
        # Run with tighter threshold and more iterations
        f_opt2, s_opt2 = _accelerated_pgd(f_cur, max_iters=100000,
                                           threshold_frac=0.90,
                                           init_step=0.2,
                                           time_limit=alloc * 0.6,
                                           label=name + "_tight")
        if s_opt2 < best_score:
            best_score = s_opt2
            best_f = f_opt2.copy()
            print(f"  ** New best: {best_score:.6f} (tight active set) **")
        f_cur = f_opt2 if s_opt2 < s_cur else f_cur
        
        # Run simple PGD with different step sizes
        if remaining - (time.monotonic() - (deadline - alloc)) > 10:
            f_opt3, s_opt3 = _pgd_simple(f_cur, max_iters=50000,
                                          threshold_frac=0.85,
                                          init_step=0.15,
                                          time_limit=min(alloc * 0.25, remaining * 0.3),
                                          label=name + "_simple")
            if s_opt3 < best_score:
                best_score = s_opt3
                best_f = f_opt3.copy()
                print(f"  ** New best: {best_score:.6f} (simple PGD) **")
    
    # Phase 3: Coordinate descent polish
    remaining = deadline - time.monotonic()
    if remaining > 30:
        n = len(best_f)
        cd_budget = min(0.25 * (deadline - time.monotonic()), remaining * 0.4)
        print(f"\n  Phase 3: Coordinate descent on best (n={n}, score={best_score:.6f}), budget={cd_budget:.0f}s")
        
        f_cd, s_cd = _coordinate_descent(best_f, max_rounds=20, time_limit=cd_budget)
        if s_cd < best_score:
            best_score = s_cd
            best_f = f_cd.copy()
            print(f"  ** New best: {best_score:.6f} (coordinate descent) **")
    
    # Phase 4: Random perturbations and re-optimization
    remaining = deadline - time.monotonic()
    if remaining > 20:
        print(f"\n  Phase 4: Perturbation restarts, remaining={remaining:.0f}s")
        n_perturb = min(8, int(remaining / 15))
        
        for p_idx in range(n_perturb):
            if time.monotonic() > deadline - 5:
                break
            
            # Different perturbation strategies
            frac = 0.02 + 0.03 * p_idx
            noise = rng.random(len(best_f)) * frac
            f_pert = _normalize(best_f + noise)
            if f_pert is None:
                continue
            
            alloc_p = (deadline - time.monotonic() - 5) / max(1, n_perturb - p_idx)
            if alloc_p < 3:
                break
            
            f_p, s_p = _accelerated_pgd(f_pert, max_iters=5000,
                                         threshold_frac=0.8,
                                         init_step=0.25,
                                         time_limit=alloc_p,
                                         label="pert_%d" % p_idx)
            if s_p < best_score:
                best_score = s_p
                best_f = f_p.copy()
                print(f"  ** New best: {best_score:.6f} (perturbation {p_idx}) **")
    
    # Phase 5: Try a couple of different n values if time allows
    remaining = deadline - time.monotonic()
    if remaining > 60:
        print(f"\n  Phase 5: Try alternative n values, remaining={remaining:.0f}s")
        best_n = len(best_f)
        alt_ns = [n for n in [384, 448, 512, 640, 768] if n != best_n]
        alt_ns = alt_ns[:3]
        
        for n_alt in alt_ns:
            if time.monotonic() > deadline - 10:
                break
            f_alt = _cosine_init(n_alt)
            alloc_a = min(30.0, (deadline - time.monotonic() - 5) * 0.5)
            f_a, s_a = _accelerated_pgd(f_alt, max_iters=5000,
                                         threshold_frac=0.8,
                                         init_step=0.3,
                                         time_limit=alloc_a,
                                         label="alt_n%d" % n_alt)
            if s_a < best_score:
                best_score = s_a
                best_f = f_a.copy()
                print(f"  ** New best: {best_score:.6f} (n={n_alt}) **")
    
    # Final: one more round of PGD on the best
    remaining = deadline - time.monotonic()
    if remaining > 15:
        f_final, s_final = _accelerated_pgd(best_f, max_iters=10000,
                                            threshold_frac=0.95,
                                            init_step=0.1,
                                            time_limit=min(remaining - 5, 40.0),
                                            label="final_polish")
        if s_final < best_score:
            best_score = s_final
            best_f = f_final.copy()
            print(f"  ** Final polish: {best_score:.6f} **")
    
    # Verify with evaluate_sequence
    final_list = [float(x) for x in best_f]
    final_score = float(evaluate_sequence(final_list))
    print(f"\nFinal verified score: {final_score:.6f} (n={len(final_list)})")
    
    # Safety: if something went wrong, fall back to parent
    if not np.isfinite(final_score) or final_score > 2.0:
        print("  Score invalid, falling back to parent.")
        return [float(x) for x in parent]
    
    # Also verify parent score for reference
    parent_score = float(evaluate_sequence([float(x) for x in parent]))
    if parent_score < final_score:
        print(f"  Parent ({parent_score:.6f}) is better, returning parent.")
        return [float(x) for x in parent]
    
    return final_list
