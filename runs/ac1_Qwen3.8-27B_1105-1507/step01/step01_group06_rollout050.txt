"""
Minimizes J(a) = 2*n*max(convolve(a,a))/(sum(a))^2 using:
1. LP-based best-response mirror descent iterations for structured updates
2. Adam gradient descent with analytical FFT-based gradients on log-sum-exp smoothed objective
3. Temperature annealing, multi-length exploration, and diverse initializations
4. Line-search based adaptive step sizing for robustness
"""

import numpy as np
from scipy.optimize import linprog
import time

def fast_conv(a):
    """Full self-convolution via FFT, O(n log n)."""
    n = len(a)
    s = 1
    while s < 2 * n - 1:
        s <<= 1
    fa = np.fft.rfft(a, s)
    return np.fft.irfft(fa * fa, s)[:2 * n - 1]

def fast_cross_conv(a, b):
    """Full linear cross-convolution via FFT."""
    s = 1
    while s < len(a) + len(b) - 1:
        s <<= 1
    fa = np.fft.rfft(a, s)
    fb = np.fft.rfft(b, s)
    return np.fft.irfft(fa * fb, s)[:len(a) + len(b) - 1]

def build_A_matrix(a):
    """Build constraint matrix A where A[k,j] = a[k-j] for the LP."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for k in range(2 * n - 1):
        jl = max(0, k - n + 1)
        jh = min(n - 1, k)
        if jl <= jh:
            js = np.arange(jl, jh + 1)
            A[k, js] = a[k - js]
    return A

def lp_best_response(a, M):
    """LP: max sum(g) s.t. (a*g)[k] <= M for all k, g >= 0."""
    n = len(a)
    c = -np.ones(n)
    A = build_A_matrix(a)
    try:
        r = linprog(c, A_ub=A, b_ub=np.full(2 * n - 1, M),
                   bounds=[(0, None)] * n, method='highs')
        if r.success and r.x.sum() > 1e-10:
            return r.x
    except Exception:
        pass
    return None

def smooth_obj_and_grad(a, tau):
    """Compute log-sum-exp smoothed 2n*max(b) and gradient w.r.t. a (assumes sum(a)=1, a>=0)."""
    n = len(a)
    b = fast_conv(a)
    b_max = b.max()
    exp_b = np.exp((b - b_max) / tau)
    exp_sum = exp_b.sum()
    lse = b_max + tau * np.log(exp_sum)
    obj = 2.0 * n * lse
    weights = exp_b / exp_sum
    w = b * weights
    # grad[i] = sum_k w[k]*(a[k-i] + a[i-k])
    # Term 1: sum_k w[k]*a[k-i] = cross-correlation np.correlate(w,a,'full')[i]
    # Term 2: sum_k w[k]*a[i-k] = linear convolution (w*a)[i]
    term1 = np.correlate(w, a, 'full')[:n]
    term2 = fast_cross_conv(w, a)[:n]
    grad = term1 + term2
    return obj, grad

def run_lp_iterations(a, max_iters, deadline):
    """LP best-response mirror descent."""
    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        b = fast_conv(a)
        M = float(b.max())
        g = lp_best_response(a, M)

            break
        a = a_new / s
    return a

def run_adam_descent(a, deadline, max_iters=3000, lr=0.05):
    """Adam optimizer on smoothed objective with temperature annealing."""
    n = len(a)
    m = np.zeros(n)
    v = np.zeros(n)
    beta1, beta2, eps = 0.8, 0.999, 1e-8
    t = 0
    tau_start = 5.0 / n
    tau_end = 0.2 / n
    last_obj = float('inf')

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        t += 1
        progress = min(1.0, t / max_iters)
        tau = tau_start * (tau_end / tau_start) ** progress
        obj, grad = smooth_obj_and_grad(a, tau)
        if obj < last_obj - 1e-12:
            stuck_count = 0
        else:
            stuck_count += 1

            lr = max(0.001, lr * 0.5)
            stuck_count = 0
        last_obj = obj
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * grad * grad
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)

        a = a / s
