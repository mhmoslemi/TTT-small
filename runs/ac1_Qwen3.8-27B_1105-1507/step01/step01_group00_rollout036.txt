"""
Optimized step function sequence search using combined Adam smoothed descent,
LP-based refinement, and coordinate descent polishing.
Targets minimization of 2n*max(conv(a,a))/sum(a)^2.
"""
import numpy as np
import time
from scipy.optimize import linprog
from scipy.sparse import csr_matrix


def quick_score(f):
    """Fast score computation (equiv to evaluate_sequence but without safety checks)."""
    n = len(f)
    conv = np.convolve(f, f)
    max_b = float(np.max(conv))
    sum_a = float(np.sum(f))
    if sum_a < 0.01:
        return float('inf')
    return 2.0 * n * max_b / (sum_a * sum_a)


def softmax_weights(conv, tau):
    """Compute softmax weights over convolution values."""
    max_c = np.max(conv)
    log_w = tau * conv - tau * max_c
    # Use log-sum-exp for numerical stability
    max_log_w = np.max(log_w)
    log_sum = max_log_w + np.log(np.sum(np.exp(log_w - max_log_w)))
    return np.exp(log_w - log_sum)


def compute_smooth_gradient(f, tau):
    """
    Compute gradient of smoothed max of autocorrelation w.r.t. f.
    Returns gradient array of same shape as f.
    """
    n = len(f)
    conv = np.convolve(f, f)
    weights = softmax_weights(conv, tau)
    # grad[j] = 2 * sum_{m=0}^{n-1} weights[j+m] * f[m]
    # This is np.correlate(weights, f, 'valid')
    grad = 2.0 * np.correlate(weights, f, mode='valid')
    return grad, conv, weights


def project_simplex(v):
    """Project v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v
    if np.sum(v) < 1e-15:
        return np.ones(n) / n
    # Bisection for the offset
    lo = np.min(v) - 1.0
    hi = np.max(v)
    for _ in range(100):
        mid = (lo + hi) / 2.0
        proj = np.maximum(v - mid, 0.0)
        s = np.sum(proj)
        if abs(s - 1.0) < 1e-12:
            break
        if s > 1.0:
            lo = mid
        else:
            hi = mid
    return np.maximum(v - mid, 0.0)


def adam_smooth_descent(f, deadline, verbose_prefix=""):

    n = len(f)
    f = f.astype(np.float64).copy()
    S = float(np.sum(f))

    beta2 = 0.999
    eps = 1e-8

    best_f = f.copy()
    best_score = quick_score(f)
    no_improve = 0
    
    while time.monotonic() < deadline:
        t += 1
        tau = max(tau_min, tau * tau_decay)
        
        grad, conv, weights = compute_smooth_gradient(f, tau)
        
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        
        f_new = f - lr * m_hat / (np.sqrt(v_hat) + eps)

def coordinate_descent(f, deadline, verbose_prefix=""):
    """
    Coordinate descent: optimize one element at a time.
    For each element j, find the best value (given all others fixed)

    """

    f = f.astype(np.float64).copy()

    it = 0

                break

            # Current value
            old_val = f[j]
            rest_sum = S - old_val

            candidates = [old_val * 0.5, old_val * 0.8, old_val * 1.2, old_val * 2.0, 0.0, old_val]

def lp_refine(f, deadline, verbose_prefix=""):

def simulated_annealing(f, deadline, T_init=1.0, T_min=0.001, cooling=0.999, verbose_prefix=""):
    """Simulated annealing with structure-aware moves."""
    n = len(f)
    f = f.astype(np.float64).copy()
    S = float(np.sum(f))
    
    best_f = f.copy()
    best_score = quick_score(f)
    T = T_init
    rng = np.random.default_rng(123)
    no_improve = 0
    
    it = 0
    while time.monotonic() < deadline and T > T_min:
        it += 1
        T *= cooling
        
        # Random move: transfer mass from one position to another
        i = rng.integers(0, n)
        j = rng.integers(0, n)
        if i == j:
            continue
        
        # Amount to transfer
        epsilon = abs(f[i]) * rng.uniform(0.1, 0.5)
        direction = 1 if f[i] > f[j] else -1
        
        f_new = f.copy()
        f_new[i] = max(0.0, f[i] - direction * epsilon)
        f_new[j] = f[j] + direction * epsilon
        
        if np.sum(f_new) < 1e-10:
            continue
        
        score_new = quick_score(f_new)
        score_cur = quick_score(f)
        
        delta = score_new - score_cur
        if delta < 0 or rng.random() < np.exp(-delta / max(T, 1e-10)):
            f = f_new
            if score_new < best_score:
                best_f = f.copy()
                best_score = score_new
                no_improve = 0
            if no_improve > 5000:
                break
        else:
            no_improve += 1
        
        if no_improve > 20000:
            break
    
    return best_f, best_score

"""
Combined optimization for minimizing 2n*max(conv(a,a))/sum(a)^2.
Uses Adam smoothed descent, LP refinement, coordinate descent, and simulated annealing.
Explores multiple sequence lengths and initializations with adaptive time allocation.
"""
import numpy as np
import time
from scipy.optimize import linprog
from scipy.sparse import csr_matrix


def quick_score(f):
    """Fast score computation equivalent to evaluate_sequence."""
    n = len(f)
    if n == 0:
        return float('inf')
    conv = np.convolve(f, f)
    max_b = float(np.max(conv))
    sum_a = float(np.sum(f))
    if sum_a < 0.01:
        return float('inf')
    return 2.0 * n * max_b / (sum_a * sum_a)


def softmax_weights(conv, tau):
    """Compute numerically stable softmax weights over convolution values."""
    max_c = np.max(conv)
    log_w = tau * conv - tau * max_c
    max_log_w = np.max(log_w)
    log_sum = max_log_w + np.log(np.sum(np.exp(log_w - max_log_w)))
    return np.exp(log_w - log_sum)


def compute_smooth_gradient(f, tau):
    """Compute gradient of smoothed max of autocorrelation w.r.t. f."""
    n = len(f)
    conv = np.convolve(f, f)
    weights = softmax_weights(conv, tau)
    grad = 2.0 * np.correlate(weights, f, mode='valid')
    return grad, conv


def project_simplex(v):
    """Project v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v
    if np.sum(v) < 1e-15:
        return np.ones(n) / n
    lo = float(np.min(v)) - 2.0
    hi = float(np.max(v)) + 1.0
    mid = hi
    for _ in range(80):
        mid = (lo + hi) / 2.0
        proj = np.maximum(v - mid, 0.0)
        s = float(np.sum(proj))
        if abs(s - 1.0) < 1e-13:
            break
        if s > 1.0:
            lo = mid
        else:
            hi = mid
    return np.maximum(v - mid, 0.0)


def adam_smooth_descent(f, deadline, verbose_prefix=""):
    """Adam-based smoothed gradient descent to minimize the score."""
    n = len(f)
    f = f.astype(np.float64).copy()
    S = float(np.sum(f))
    if S <= 0:
        return f, quick_score(f)
    f = f / S

    m = np.zeros(n)
    v = np.zeros(n)
    t = 0
    lr = 0.08
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    tau = 200.0
    tau_min = 2.0
    tau_decay = 0.9997

    best_f = f.copy()
    best_score = quick_score(f)
    no_improve = 0
    kick_count = 0

    rng = np.random.default_rng(7)

    while time.monotonic() < deadline:
        t += 1
        tau = max(tau_min, tau * tau_decay)

        grad, conv = compute_smooth_gradient(f, tau)

        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)

        f_new = f - lr * m_hat / (np.sqrt(v_hat) + eps)
        f_new = project_simplex(f_new)

        score_new = quick_score(f_new)

        if score_new < best_score:
            f = f_new
            best_f = f.copy()
            best_score = score_new
            no_improve = 0
        else:

            if no_improve % 500 == 0:
                lr *= 0.92

            row_list.append(k)
            col_list.append(j)
            val_list.append(f[k - j])

    rows = np.array(row_list, dtype=np.int32)
    cols = np.array(col_list, dtype=np.int32)
    vals = np.array(val_list, dtype=np.float64)
    A = csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))

    bounds = [(0.0, None)] * n
    step_size = 0.12
    no_improve = 0

            break

            result = linprog(c_obj, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',

            no_improve += 1

        else:

            if time.monotonic() > deadline:
                break

            old_val = f[j]
            if old_val < 1e-10:
                # Try setting this to a small positive value
                candidates = [0.01, 0.05, 0.1]

                    pressure = f[max_k - j]

                    candidates.append(old_val + 0.05 * pressure)

        else:

def simulated_annealing(f, deadline, T_init=0.01, T_min=0.0005, cooling=0.9995, verbose_prefix=""):
    """Simulated annealing with mass-transfer moves."""
    n = len(f)
    f = f.astype(np.float64).copy()

    best_f = f.copy()
    best_score = quick_score(f)
    T = T_init
    rng = np.random.default_rng(200)
    no_improve = 0
    it = 0

    while time.monotonic() < deadline and T > T_min:
        it += 1
        T *= cooling

        # Choose move type
        move_type = rng.random()
        f_new = f.copy()

        if move_type < 0.6:
            # Transfer: move mass from i to j
            i = int(rng.integers(0, n))
            j = int(rng.integers(0, n))
            if i == j:
                continue
            epsilon = max(0.01, abs(f[i])) * rng.uniform(0.05, 0.4)
            if f[i] >= epsilon:
                f_new[i] -= epsilon
                f_new[j] += epsilon
            else:
                f_new[j] -= epsilon * f_new[j] / max(f_new[j], 0.01)
                f_new[i] += epsilon * f_new[j] / max(f_new[j], 0.01)
        elif move_type < 0.8:
            # Single coordinate scaling
            j = int(rng.integers(0, n))
            factor = rng.uniform(0.5, 1.5)
            f_new[j] *= factor
        else:
            # Pair swap
            idx = rng.choice(n, size=2, replace=False)
            f_new[idx[0]], f_new[idx[1]] = f_new[idx[1]], f_new[idx[0]]

        if np.any(f_new < 0):
            f_new = np.clip(f_new, 0, None)
        if np.sum(f_new) < 1e-10:
            continue

        score_new = quick_score(f_new)
        score_cur = quick_score(f)

        delta = score_new - score_cur
        if delta < 0 or (T > 1e-10 and rng.random() < np.exp(-abs(delta) / T)):
            f = f_new
            if score_new < best_score:
                best_f = f.copy()
                best_score = score_new
                no_improve = 0
            if no_improve > 3000:
                break
        else:
            no_improve += 1
            if no_improve > 5000:
                break

    return best_f, best_score

                f2, s2 = lp_refine(best_for_n, lp_deadline, verbose_prefix=f"[{label}]")
                if s2 < best_score_n:
                    best_score_n = s2
                    best_for_n = f2.copy()
                print(f" ->LP:{s2:.6f}", end="")

            # Phase 3: Coordinate descent
            if time.monotonic() < exploration_deadline - 8:
                cd_deadline = min(exploration_deadline - 5, time.monotonic() + max(5.0, time_per_init * 0.15))
                f3, s3 = coordinate_descent(best_for_n, cd_deadline, verbose_prefix=f"[{label}]")
                if s3 < best_score_n:
                    best_score_n = s3
                    best_for_n = f3.copy()
                print(f" ->CD:{s3:.6f}")
            else:
                print()

            # Phase 4: Short simulated annealing
            if time.monotonic() < exploration_deadline - 6:
                sa_deadline = min(exploration_deadline - 3, time.monotonic() + max(5.0, time_per_init * 0.1))
                f4, s4 = simulated_annealing(best_for_n, sa_deadline, T_init=0.005, verbose_prefix=f"[{label}]")
                if s4 < best_score_n:
                    best_score_n = s4
                    best_for_n = f4.copy()
                    print(f"  ->SA:{s4:.6f}")

        if best_for_n is not None:
            print(f"\n  Best for n={n}: {best_score_n:.6f}")
            if best_score_n < best_score:
                best_score = best_score_n
                best_sequence = best_for_n.copy()

    # Final refinement phase
    if best_sequence is not None and time.monotonic() < deadline:
        print(f"\n{'='*50}")
        print(f"  FINAL REFINEMENT (n={len(best_sequence)}, current={best_score:.6f})")
        print(f"{'='*50}")

        remaining = deadline - time.monotonic()

        # Adam with fresh momentum
        adam_tl = max(5.0, remaining * 0.3)
        f_fin, s_fin = adam_smooth_descent(best_sequence, t0 + adam_tl + (time.monotonic() - t0 - (remaining - adam_tl)), verbose_prefix="[fin-Adam]")
        # Simpler: just compute deadline properly
        adam_deadline = time.monotonic() + max(5.0, (deadline - time.monotonic()) * 0.35)
        f_fin, s_fin = adam_smooth_descent(best_sequence, adam_deadline, verbose_prefix="[fin-Adam]")
        if s_fin < best_score:
            best_score = s_fin
            best_sequence = f_fin.copy()
            print(f"  Adam: {best_score:.6f}")

        # LP
        if time.monotonic() < deadline - 10:
            lp_deadline = time.monotonic() + max(5.0, (deadline - time.monotonic()) * 0.3)
            f_lp, s_lp = lp_refine(best_sequence, lp_deadline, verbose_prefix="[fin-LP]")
            if s_lp < best_score:
                best_score = s_lp
                best_sequence = f_lp.copy()
                print(f"  LP: {best_score:.6f}")

        # Coordinate descent
        if time.monotonic() < deadline - 8:
            cd_deadline = time.monotonic() + max(5.0, (deadline - time.monotonic()) * 0.2)
            f_cd, s_cd = coordinate_descent(best_sequence, cd_deadline, verbose_prefix="[fin-CD]")
            if s_cd < best_score:
                best_score = s_cd
                best_sequence = f_cd.copy()
                print(f"  CD: {best_score:.6f}")

        # Final SA
        if time.monotonic() < deadline - 3:
            sa_deadline = deadline - 2
            f_sa, s_sa = simulated_annealing(best_sequence, sa_deadline, T_init=0.002, cooling=0.9998, verbose_prefix="[fin-SA]")
            if s_sa < best_score:
                best_score = s_sa
                best_sequence = f_sa.copy()
                print(f"  SA: {best_score:.6f}")

    if best_sequence is None:
        print("Falling back to uniform sequence")
        return [1.0] * 128

    final_list = [float(max(0.0, min(1000.0, x))) for x in best_sequence]
    final_score = evaluate_sequence(final_list)
    if not np.isfinite(final_score):
        final_score = quick_score(best_sequence)

    elapsed = time.monotonic() - t0
    print(f"\n{'='*50}")
    print(f"  RESULT: n={len(final_list)}, score={final_score:.6f}, time={elapsed:.1f}s")
    print(f"{'='*50}")

    return final_list
