"""
Hybrid optimization for minimizing 2*n*max(conv(a,a))/sum(a)^2.
Uses soft-max gradient descent with temperature annealing,
coordinate descent polishing, and LP-based refinement.
"""

import numpy as np
from scipy.optimize import linprog
from scipy.special import logsumexp
import time


def compute_score_fast(f):
    """Compute score for a sequence normalized to sum=1."""
    n = len(f)
    conv = np.convolve(f, f)
    M = conv.max()
    return 2.0 * n * M


def simplex_project(v):
    """Project vector v onto the simplex {x : sum(x)=1, x>=0}."""
    n = len(v)
    if n == 0:
        return v
    # Algorithm from Duchi et al. (2011)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.argmax(u * np.arange(1, n+1) - (cssv - 1) > 0)
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0)


def soft_gradient(a, T):
    """Compute gradient of soft-max objective for sequence a (sum=1)."""
    n = len(a)
    c = np.convolve(a, a)  # length 2n-1
    # Soft-max with temperature T
    c_scaled = c / T
    # Use logsumexp for numerical stability
    max_c = c_scaled.max()
    exp_c = np.exp(c_scaled - max_c)
    sum_exp = exp_c.sum()
    p = exp_c / sum_exp  # softmax probabilities
    # soft_max_val = T * log(sum_exp) + max_c * T
    soft_max_val = T * (np.log(sum_exp) + max_c)
    S = a.sum()  # should be 1
    # Gradient of 2n * soft_max_val / S^2
    # = 2n/S^2 * grad(soft_max_val) - 4n * soft_max_val / S^3
    # grad(soft_max_val) = sum_k p_k * grad(c_k) = 2 * q where q[i] = sum_k p_k * a[k-i]
    q = np.correlate(p, a, mode='valid')  # length n
    grad_svm = 2.0 * q
    grad_obj = 2.0 * n / S**2 * grad_svm - 4.0 * n * soft_max_val / S**3
    return grad_obj, soft_max_val, p


def grad_descent_soft(a, T_max, T_min, n_iter, lr_init, momentum=0.9, rng=None):
    """Run soft-max gradient descent with temperature annealing."""
    n = len(a)
    a = a.astype(np.float64).copy()
    S = a.sum()
    a = a / S  # normalize
    
    best_a = a.copy()
    best_score = compute_score_fast(a)
    
    lr = lr_init
    v = np.zeros(n)  # momentum
    
    for it in range(n_iter):
        # Anneal temperature
        progress = it / max(1, n_iter - 1)
        T = T_max * (T_min / T_max) ** progress
        
        grad, svm, p = soft_gradient(a, T)
        
        # Momentum update
        v = momentum * v - lr * grad
        a_new = a + v
        a_new = simplex_project(a_new)
        
        score = compute_score_fast(a_new)
        if score < best_score:
            best_a = a_new.copy()
            best_score = score
            a = a_new
        
        # Adaptive learning rate: reduce if no improvement for a while
        # (simplified: just use a fixed decay)
        lr = lr_init * (0.999 ** it)
    
    return best_a, best_score

def build_conv_matrix(f):
    """Build matrix A such that A @ b = np.convolve(f, b, mode='full')."""
    n = len(f)
    # Use scattering approach for efficiency
    A = np.zeros((2 * n - 1, n))
    for k in range(2 * n - 1):
        j_start = max(0, k - n + 1)
        j_end = min(n, k + 1)
        for j in range(j_start, j_end):
            A[k, j] = f[k - j]
    return A
