Optimizes nonnegative sequence to minimize 2*n*max(conv(a,a))/sum(a)^2.
Uses adaptive log-sum-exp smoothing with Nesterov accelerated proximal gradient,
combined with LP refinement steps and multiple initialization strategies.
"""

import time
from scipy.optimize import linprog


def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    N = 1 << (size - 1).bit_length()
    fa = np.fft.rfft(a, N)
    fc = fa * fa
    result = np.fft.irfft(fc, N)[:size]
    result[result < 0] = 0.0
    return result


def project_simplex(v):
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cv = np.cumsum(u) - 1.0
    rho_candidates = np.where(u * np.arange(1, n + 1) > cv)[0]
    if len(rho_candidates) == 0:
        return np.ones(n) / n
    rho = rho_candidates[-1]
    theta = cv[rho] / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def compute_smooth_gradient(a, c, tau):
    """Compute gradient of (1/tau)*log(sum(exp(tau*c))) w.r.t. a."""
    n = len(a)
    m = np.max(c)
    exp_vals = np.exp(tau * (c - m))
    weights = exp_vals / np.sum(exp_vals)

    A_f = np.fft.rfft(a, N)
    W_f = np.fft.rfft(weights, N)
    grad = 2.0 * np.fft.irfft(A_f * np.conj(W_f), N)[:n]
    return grad


def subgradient_step(a, c, n_top=10):
    """Subgradient using top n_top peaks."""
    n = len(a)
    peaks = np.argsort(c)[::-1][:n_top]
    grad = np.zeros(n)
    for k in peaks:
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        if lo <= hi:
            grad[lo:hi + 1] += 2.0 * a[k - hi:k - lo + 1][::-1]
    grad /= len(peaks)
    return grad


def alt_projection_init(n, rng, n_iter=100):
    """Alternating projection for flat magnitude spectrum initialization."""
    N = 1 << (2 * n - 2).bit_length()
    phases = rng.uniform(0.0, 2.0 * np.pi, N // 2 + 1)
    F = np.exp(1j * phases)
    F[0] = 1.0 + 0j
    
    for it in range(n_iter):
        a_full = np.fft.irfft(F, N)
        a = np.maximum(a_full[:n].copy(), 0.0)
        s = a.sum()
        if s < 1e-10:
            a = np.ones(n) / n
        else:
            a = a / s
        F_full = np.fft.rfft(a, N)
        F = np.ones(len(F_full)) * np.exp(1j * np.angle(F_full))
        F[0] = 1.0 + 0j
    
    return a


def lp_improve(a, P, time_limit=5.0):
    """LP step: maximize sum(b) s.t. convolve(a,b) <= P, b >= 0."""
    n = len(a)
    if n > 1536:
        return None
    
    A = np.zeros((2 * n - 1, n), dtype=np.float64)
    for k in range(2 * n - 1):
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        if lo <= hi:
            A[k, lo:hi + 1] = a[k - hi:k - lo + 1][::-1]
    
    c_obj = -np.ones(n)
    b_ub = np.full(2 * n - 1, P)
    
    try:
        res = linprog(c_obj, A_ub=A, b_ub=b_ub, bounds=[(0.0, None)] * n,
                      method='highs', options={'time_limit': time_limit})
        if res.success:
            b = res.x
            if b.sum() > 1e-8:
                return b * (a.sum() / b.sum())
    except Exception:
        pass
    return None

"""
Optimizes nonnegative sequence to minimize 2*n*max(conv(a,a))/sum(a)^2.
Uses adaptive log-sum-exp smoothing with Nesterov-accelerated proximal gradient (FISTA),
LP refinement steps, multiple diverse initializations, and aggressive time management.
Key improvements over previous: smoother gradient via log-sum-exp, FISTA acceleration,
adaptive tau scheduling, and more aggressive LP exploitation.
"""

import numpy as np
import time
from scipy.optimize import linprog


def fast_convolve(a):
    """Compute full linear convolution of a with itself using FFT."""
    n = len(a)
    size = 2 * n - 1
    N = 1 << (size - 1).bit_length()
    fa = np.fft.rfft(a, N)
    fc = fa * fa
    result = np.fft.irfft(fc, N)[:size]
    result[result < 0] = 0.0
    return result


def project_simplex(v):
    """Project vector v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cv = np.cumsum(u) - 1.0
    rho_candidates = np.where(u * np.arange(1, n + 1) > cv)[0]
    if len(rho_candidates) == 0:
        return np.ones(n) / n
    rho = rho_candidates[-1]
    theta = cv[rho] / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def compute_smooth_gradient(a, c, tau):
    """Compute gradient of (1/tau)*log(sum(exp(tau*c))) w.r.t. a using FFT."""
    n = len(a)
    m = np.max(c)
    exp_vals = np.exp(tau * (c - m))
    weights = exp_vals / np.sum(exp_vals)

    N = 1 << (2 * n - 2).bit_length()
    A_f = np.fft.rfft(a, N)
    W_f = np.fft.rfft(weights, N)
    # grad_a[j] = 2 * sum_k weights[k] * a[k-j] (for valid k-j)
    # = 2 * ifft(fft(a) * conj(fft(weights)))[:n]
    grad = 2.0 * np.fft.irfft(A_f * np.conj(W_f), N)[:n]
    return grad


def subgradient_top(a, c, n_top=15):
    """Subgradient using top n_top peak positions."""
    n = len(a)
    peaks = np.argsort(c)[::-1][:n_top]
    grad = np.zeros(n)
    for k in peaks:
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        if lo <= hi:
            grad[lo:hi + 1] += 2.0 * a[k - hi:k - lo + 1][::-1]
    grad /= len(peaks)
    return grad


def alt_projection_init(n, rng, n_iter=100):
    """Alternating projection for near-flat magnitude spectrum initialization."""
    N = 1 << (2 * n - 2).bit_length()
    phases = rng.uniform(0.0, 2.0 * np.pi, N // 2 + 1)
    F = np.exp(1j * phases)
    F[0] = 1.0 + 0j

    for it in range(n_iter):
        a_full = np.fft.irfft(F, N)
        a = np.maximum(a_full[:n].copy(), 0.0)
        s = a.sum()
        if s < 1e-10:
            a = np.ones(n) / n
            s = 1.0
        a = a / s
        F_full = np.fft.rfft(a, N)
        F = np.ones(len(F_full)) * np.exp(1j * np.angle(F_full))
        F[0] = 1.0 + 0j

    return a


def spectral_init(n, rng, n_iter=60, taper=0.3):
    """Spectral initialization with tapered flat target for better nonnegativity."""
    N = 1 << (2 * n - 2).bit_length()
    phases = rng.uniform(0.0, 2.0 * np.pi, N // 2 + 1)
    F = np.exp(1j * phases)
    F[0] = 1.0 + 0j

    for it in range(n_iter):
        a_full = np.fft.irfft(F, N)
        a = np.maximum(a_full[:n].copy(), 0.0)
        s = a.sum()
        if s < 1e-10:
            a = np.ones(n) / n
            s = 1.0
        a = a / s
        F_full = np.fft.rfft(a, N)
        # Tapered target: slightly emphasize low frequencies
        w = np.linspace(1.0, 1.0 - taper, len(F_full))
        target_mag = w ** 0.5
        F = target_mag * np.exp(1j * np.angle(F_full))
        F[0] = F[0].real + 0j

    return a


def lp_improve(a, P, time_limit=8.0):
    """LP: maximize sum(b) s.t. convolve(a,b) <= P, b >= 0. Returns rescaled b or None."""
    n = len(a)
    if n > 2048:
        return None

    A = np.zeros((2 * n - 1, n), dtype=np.float64)
    for k in range(2 * n - 1):
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        if lo <= hi:
            A[k, lo:hi + 1] = a[k - hi:k - lo + 1][::-1]

    c_obj = -np.ones(n)
    b_ub = np.full(2 * n - 1, P)

    try:
        res = linprog(c_obj, A_ub=A, b_ub=b_ub, bounds=[(0.0, None)] * n,
                      method='highs', options={'time_limit': time_limit})
        if res.success:
            b = res.x
            bsum = b.sum()
            if bsum > 1e-8:
                return b * (a.sum() / bsum)
    except Exception:
        pass
    return None


def fista_smooth(a0, deadline, rng, tau_init=20.0, tau_max=500.0,
                 n_top=15, lr_scale=1.0):
    """
    FISTA with adaptive log-sum-exp smoothing.
    Minimizes max(conv(a,a)) s.t. a >= 0, sum(a) = 1.
    """
    n = len(a0)
    a = project_simplex(a0.copy().astype(np.float64))
    y = a.copy()
    t_k = 1.0

    base_lr = lr_scale * 1.0 / np.sqrt(n)
    best_a = a.copy()
    best_val = np.inf

    c = fast_convolve(a)
    best_val = np.max(c)
    stall = 0
    it = 0
    tau = tau_init

    while time.time() < deadline:
        it += 1

        # FISTA momentum
        t_next = (1.0 + np.sqrt(1.0 + 4.0 * t_k * t_k)) / 2.0
        y_new = y + (t_k - 1.0) / t_next * (y - a)

        # Compute gradient at y
        c_y = fast_convolve(y)

        if tau < tau_max * 0.5:
            grad = compute_smooth_gradient(y, c_y, tau)
        else:
            grad = subgradient_top(y, c_y, n_top)

        # Adaptive learning rate based on gradient norm
        gnorm = np.linalg.norm(grad)
        if gnorm > 1e-15:
            lr = base_lr / max(gnorm, 0.1)
        else:
            lr = base_lr

        # Gradient step + projection
        a_new = project_simplex(y - lr * grad)

        # Evaluate
        c_a = fast_convolve(a_new)
        peak_a = np.max(c_a)

        if peak_a < best_val * (1.0 - 1e-10):
            best_val = peak_a
            best_a = a_new.copy()
            stall = 0
        else:
            stall += 1

        a = a_new
        y = y_new
        t_k = t_next

        # Adaptive tau: increase when stalled ( tighter approximation to max)
        if stall > 100:
            tau = min(tau * 1.3, tau_max)
        if stall > 300:
            base_lr = max(base_lr * 0.85, 1e-8)
        if stall > 600:
            # Perturbation to escape
            if time.time() < deadline - 15:
                eps = 0.03 / np.sqrt(n)
                a = project_simplex(a + rng.normal(0, eps, n))
                y = a.copy()
                t_k = 1.0
                tau = max(tau_init, tau * 0.5)
            stall = 0

    return best_a, best_val


def optimize_full(a0, deadline, rng, use_lp=True):
    """
    Full optimization pipeline: FISTA -> LP refinement -> FISTA polish.
    """
    n = len(a0)
    a = a0.copy()
    best_a = a.copy()

    c = fast_convolve(a)
    best_peak = np.max(c)
    a_sum = a.sum()

    # Phase 1: FISTA (60% of time)

    if peak1 < best_peak:

        best_a = a.copy()

    print(f"    FISTA phase: peak={peak1:.8f}, score~{2*n*peak1:.6f}")

    # Phase 2: LP refinement (25% of time)
    if use_lp and n <= 2048 and time.time() < deadline - 60:

        P = 2.0 * n * best_peak / (a_sum * a_sum)  # current peak with sum_a=1

                for t_mix in [0.6, 0.8, 0.9, 0.95]:
                    a_mix = project_simplex((1 - t_mix) * a + t_mix * b)

                    if peak_mix < best_peak * (1 - 1e-6):

                        a = a_mix.copy()
                        P = peak_mix

        print(f"    After LP: peak={best_peak:.8f}, score~{2*n*best_peak:.6f}")
