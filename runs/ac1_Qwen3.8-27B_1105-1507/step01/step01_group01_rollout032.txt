"""
Improved search for low-score non-negative step-function coefficients.
Uses subgradient descent with adaptive step size, LP-based moves,
coordinate descent, and simulated annealing.
"""

import time
import numpy as np
from scipy.optimize import linprog, minimize_scalar

def normalize_to_sum1(a):
    a = np.asarray(a, dtype=np.float64)
    if a.size == 0:
        return None
    a = np.clip(a, 0.0, 1000.0)
    s = float(np.sum(a))
    if not np.isfinite(s) or s <= 1e-300:
        return None
    return a / s

def local_score(a):
    a = np.asarray(a, dtype=np.float64)
    if a.size == 0:
        return float('inf')
    a = np.clip(a, 0.0, 1000.0)
    s = float(np.sum(a))
    if not np.isfinite(s) or s <= 1e-12:
        return float('inf')
    c = np.convolve(a, a)
    m = float(np.max(c))
    if not np.isfinite(m):
        return float('inf')
    return float(2.0 * len(a) * m / (s * s))

def compute_subgradient(a, frac=0.95):
    """Compute subgradient of max(conv(a,a)) w.r.t. a."""
    n = a.size
    c = np.convolve(a, a)
    M = float(np.max(c))
    if M <= 0:
        return np.zeros(n)
    
    # Find peaks above threshold
    threshold = M * frac
    peaks = np.where(c >= threshold)[0]
    
    # Compute subgradient
    g = np.zeros(n)
    for k in peaks:
        weight = c[k] / M  # normalize weight
        # c[k] = sum_i a[i]*a[k-i]
        # dc[k]/da[i] = a[k-i] (if valid) + a[i-k] (if valid and i != k-i)
        s_start = max(0, k - n + 1)
        s_end = min(n - 1, k)
        for i in range(s_start, s_end + 1):
            j = k - i
            if 0 <= j < n:
                g[i] += weight * a[j]
                if j != i:
                    g[j] += weight * a[i]
    
    return g / max(1.0, float(np.max(g)))

Zadoff-Chu sequences are complex sequences with constant amplitude. If I can find a non-negative real sequence whose FFT magnitude is approximately constant, that would be ideal.

But non-negative real sequences have a specific FFT structure: FFT(a)[k] = sum a[j] * exp(-2πijk/n), and since a is real and non-negative, the FFT has conjugate symmetry.

This is a useful perspective but not directly actionable. Let me focus on the optimization.

**My final plan:**

1. Start from the parent (score 1.514).
2. Use a "cooling" subgradient method:
   - Start with high temperature (smooth approximation) → find the "basin"
   - Gradually decrease temperature → sharpen the max
   - Use line search for step size
3. Periodically do LP-based moves with per-peak bounds.
4. Near the end, do coordinate descent and fine-tuning.
5. Also try a few different lengths (480, 512, 544, 576) to see if a different length helps.

Let me code this up now. I'll aim for clean, efficient code.

Actually, one more important thought: the previous code's `multiplicative_update` function reduces elements that contribute to peaks and increases others (via the `fac = eta * (meanh - h) / maxh` term). This is essentially a "contraction" of the high-convolution positions and "expansion" of low-convolution positions. My subgradient approach only contracts the high positions. The multiplicative approach might be better because it also spreads energy to low positions.

Let me combine both: use the subgradient direction for the "target" and the multiplicative update for the "mechanism."

Here's my refined approach:
1. Compute the soft subgradient g (points in the direction of increasing max conv).
2. Apply a multiplicative update: a[i] *= exp(-eta * g[i] + gamma) where gamma is chosen to preserve the mean (or normalization).
3. This is equivalent to: a[i] *= exp(-eta * (g[i] - mean(g)))
4. Renormalize.

This is very similar to what the previous code does! The difference is that the previous code uses a hard subgradient (only top-K peaks), while I'll use a soft subgradient (all peaks with exponential weighting).

    n = a.size
    c = np.convolve(a, a)
    max_c = float(c.max())

    if w_sum < 1e-300:

        wk = w[k]

        mask = jdx != idx
        if mask.any():
            g[idx[mask]] += wk * a[jdx[mask]]
            g[jdx[mask]] += wk * a[idx[mask]]
    return g

def lp_step(f, t, seed=0):
    """LP-based move: find b that minimizes max increase in convolution."""
    n = f.size
    c = np.convolve(f, f)
    M = float(c.max())
    
    # Per-peak bounds: conv(b, f)[k] <= (M - c[k]) / t + eps
    # This ensures conv((1-t)f + tb, (1-t)f + tb)[k] <= M + O(t^2)
    bounds_vec = (M - c) / max(t, 1e-6) + 1e-6
    bounds_vec = np.maximum(bounds_vec, 1e-6)  # ensure positive
    
    # Also bound b itself: 0 <= b[i] <= 1
    # Build constraint matrix: A b <= bounds_vec
    # A[k, i] = f[k-i] (if valid) + f[i-k] (if valid, i≠k-i)
    # Actually: conv(b, f)[k] = sum_i b[i] * f[k-i] (for valid k-i)
    # So A[k, i] = f[k-i] if 0 <= k-i < n, else 0
    
    A = np.zeros((2 * n - 1, n), dtype=np.float64)
    for k in range(2 * n - 1):
        s0 = max(0, k - n + 1)
        s1 = min(n - 1, k)
        for i in range(s0, s1 + 1):
            A[k, i] = f[k - i]
    
    cvec = -np.ones(n)  # maximize sum(b)
    res = linprog(cvec, A_ub=A, b_ub=bounds_vec, bounds=[(0, 1)] * n,
                  method='highs', options={'presolve': True})
    
    if res is None or not res.success:
        return None
    
    b = np.clip(res.x, 0, 1)
    b_sum = b.sum()
    if b_sum < 1e-10:
        return None
    
    return b / b_sum  # normalize

def coordinate_descent(a, max_iter=100):

def build_conv_matrix(f, n):
    A = np.zeros((2 * n - 1, n), dtype=np.float64)
    for k in range(2 * n - 1):
        s0 = max(0, k - n + 1)
        s1 = min(n - 1, k)
        if s0 <= s1:
            A[k, s0:s1 + 1] = f[k - np.arange(s0, s1 + 1)]
    return A

        small_b = min(10.0, max(2.0, 0.01 * total_budget))
        sa, sc = coordinate_descent(cc, max_iter=40, total_budget=small_b, rng=master_rng)
        if sa is not None and np.isfinite(sc) and sc < s0:
            cc, s0 = sa, sc
        if s0 < best_score:
            best = cc.copy()
            best_score = float(s0)

    if best is None:
        best = normalize(np.ones(256))
        best_score = get_score(best)
    print(f"Phase 0 best: {best_score:.6f} n={best.size}", flush=True)

    best_len = int(best.size)

    # Phase 1: main subgradient loop on parent length
    n_main = best_len
    b1 = min(900.0, 0.65 * total_budget)
    main_a = best.copy()
    main_sc = best_score
    temp = 0.10
    eta = 0.32
    sa = 0.0
    lp_cd = 0
    last_improve = time.monotonic()
    last_lp = time.monotonic()
    t0 = time.monotonic()
    d0 = t0 + b1
    it = 0

    while time.monotonic() < d0:
        rem = d0 - time.monotonic()
        if rem < 8.0:
            break
        g = soft_subgrad(main_a, temp)
        trial_ts = [(0.25, eta), (1.0, eta), (1.0, eta * 0.5), (2.5, eta * 0.5), (2.5, eta * 1.5)]
        acc_sc = None
        for mult, et in trial_ts:
            a2 = mul_update(main_a, g, et)
            if a2 is None:
                continue
            sc2 = get_score(a2)
            if not np.isfinite(sc2):
                continue
            sa = sa + mult
            if sc2 < main_sc - 1e-11:
                main_a = a2
                main_sc = float(sc2)
                best = a2.copy()
                best_score = float(sc2)
                last_improve = time.monotonic()
                if acc_sc is None or sc2 < acc_sc:
                    acc_sc = sc2
        if acc_sc is not None:
            sa = 0.0
            temp = max(0.015, temp * 0.995)
            eta = min(0.8, eta * 1.01)
        else:
            sa += 1.0
            if sa > 50.0:
                if temp > 0.25:
                    temp *= 1.3
                else:
                    eta = max(0.1, eta * 0.6)
            big_sa = 0.0
            if sa > 150.0:
                big_sa = 50.0
                nz = float(np.count_nonzero(main_a > 1e-6))
                if nz < n_main * 0.5:
                    fn = normalize(main_a + 0.05)
                    if fn is not None:
                        main_a = fn
                        big_sa = 10.0
            if sa > 400.0:
                jam = (time.monotonic() - last_improve) > 30.0
                if jam or main_sc > best_score + 0.02:
                    alt = best * np.exp(master_rng.normal(0, 0.4, n_main))
                    a2 = normalize(alt)
                    if a2 is not None:
                        sc2 = get_score(a2)
                        if np.isfinite(sc2) and sc2 < main_sc:
                            main_a = a2
                            main_sc = float(sc2)
                sa = 0.0
                temp = 0.12
                eta = 0.3
        big_sa = max(0.0, big_sa) + 0.0
        if big_sa > 0:
            pass
        if HAVE_SCIPY and (time.monotonic() - last_lp) > 45.0 and rem > 40.0:
            last_lp = time.monotonic()
            for mix_v in [0.25, 0.4, 0.6]:
                b = lp_step(main_a, mix_v, int(seed) + it)
                if b is None:
                    continue
                for mv in [0.15, 0.3, 0.5, 0.7]:
                    cand = normalize((1.0 - mv) * main_a + mv * b)
                    if cand is None:
                        continue
                    sc = get_score(cand)
                    if np.isfinite(sc) and sc < main_sc:
                        main_a = cand
                        main_sc = float(sc)
                        if sc < best_score:
                            best = cand.copy()
                            best_score = float(sc)
                        last_improve = time.monotonic()
                        sa = 0.0
                        break
        if (it % 800 == 0 and it > 0) and HAVE_SCIPY and rem > 30.0:
            sa_a, sa_sc = coordinate_descent(main_a, max_iter=12, total_budget=3.0, rng=master_rng)
            if sa_a is not None and np.isfinite(sa_sc) and sa_sc < main_sc:
                main_a = sa_a
                main_sc = float(sa_sc)
                if main_sc < best_score:
                    best = main_a.copy()
                    best_score = float(main_sc)
        it += 1
        if it % 500 == 0:
            print(f"  main it={it} best={best_score:.6f} main={main_sc:.6f} temp={temp:.4f}", flush=True)

    print(f"Phase 1 done: best={best_score:.6f}", flush=True)

    # Phase 2: secondary lengths
    rem2 = deadline - time.monotonic()
    if rem2 > 120.0:
        for sec_n in [best_len // 2, best_len * 2, 256, 768]:
            if sec_n <= 32 or sec_n > 2048:
                continue
            if sec_n == best_len:
                continue
            b2 = min(60.0, rem2 * 0.12)
            if b2 < 15.0:
                break
            rng2 = np.random.default_rng(int(seed) + 555 * sec_n)
            cands = [
                normalize(rng2.uniform(0.05, 1.0, sec_n)),
                make_fourier_candidate(sec_n, int(seed) + 77 * sec_n),
                make_tapered_candidate(sec_n, int(seed) + 88 * sec_n),
            ]
            if parent_arr is not None and sec_n != n_parent:
                old = np.linspace(0, 1, n_parent)
                new = np.linspace(0, 1, sec_n)
                cands.append(normalize(np.interp(new, old, parent_arr)))
            local_best = None
            local_sc = float('inf')
            for cc in cands:
                if cc is None:
                    continue
                sa_a, sa_sc = coordinate_descent(cc, max_iter=50, total_budget=3.0, rng=rng2)
                if sa_a is not None and np.isfinite(sa_sc) and sa_sc < local_sc:
                    local_best, local_sc = sa_a, sa_sc
                if np.isfinite(sa_sc) and local_best is None:
                    local_best, local_sc = cc, sa_sc
            if local_best is not None and local_sc < best_score:
                best = local_best
                best_score = local_sc
                print(f"  secondary n={sec_n} improved to {best_score:.6f}", flush=True)
            rem2 = deadline - time.monotonic()
            if rem2 < 60.0:
                break

    # Phase 3: final coordinate descent
    rem3 = deadline - time.monotonic()
    if rem3 > 15.0:
        cd_iter = min(300, max(20, int(rem3 // 2)))
        rng3 = np.random.default_rng(int(seed) + 999)
        sa_a, sa_sc = coordinate_descent(best, max_iter=cd_iter,
                                         total_budget=max(5.0, rem3 - 20.0), rng=rng3)
        if sa_a is not None and np.isfinite(sa_sc) and sa_sc < best_score:
            best = sa_a
            best_score = sa_sc
            print(f"  final CD: {best_score:.6f}", flush=True)

    # Phase 4: one more subgradient polish if time remains
    rem4 = deadline - time.monotonic()
    if rem4 > 30.0:
        main_a = best.copy()
        main_sc = best_score
        temp = 0.06
        eta = 0.2
        for _ in range(3000):
            if time.monotonic() > deadline:
                break
            g = soft_subgrad(main_a, temp)
            ok = False
            for et in [eta, eta * 0.5, eta * 2.0]:
                a2 = mul_update(main_a, g, et)
                if a2 is None:
                    continue
                sc2 = get_score(a2)
                if np.isfinite(sc2) and sc2 < main_sc:
                    main_a = a2
                    main_sc = float(sc2)
                    if main_sc < best_score:
                        best = a2.copy()
                        best_score = float(main_sc)
                    ok = True
                    break
            if ok:
                temp = max(0.02, temp * 0.99)
            else:
                eta = max(0.05, eta * 0.8)
                if eta < 0.06:
                    break
        print(f"  polish: {best_score:.6f}", flush=True)

    # Final validation
    final = normalize(best)
    if final is None:
        final = np.ones(64, dtype=np.float64) / 64.0
    final = np.clip(final, 0.0, 1000.0)
    try:
        official = float(evaluate_sequence([float(x) for x in final]))
        if np.isfinite(official):
            best_score = official
    except Exception:
        pass
    print(f"FINAL score={best_score:.6f} length={len(final)}", flush=True)
    return [float(x) for x in final]
