"""
Hybrid optimizer for minimizing 2*n*max(convolve(a,a))/sum(a)^2 over non-negative sequences.
Combines LP active-set direction finding, multiplicative weights (mirror descent),
smooth gradient polishing, and local search with multiple window-based initializations.
"""

import numpy as np
from scipy.optimize import linprog
import time


def _score(f):
    """Compute 2*n*max(conv)/sum^2 for a non-negative sequence."""
    n = len(f)
    S = f.sum()
    if S < 1e-15:
        return np.inf
    conv = np.convolve(f, f)
    return 2.0 * n * conv.max() / (S * S)


def _score_norm(f):
    """Score assuming f is already normalized to sum=1."""
    n = len(f)
    return 2.0 * n * np.convolve(f, f).max()


def _conv_rows(f, idx):
    """Build rows of the cross-convolution matrix for specified lag indices."""
    n = len(f)
    out = np.zeros((len(idx), n))
    for r, k in enumerate(idx):
        lo = max(0, k - n + 1)
        hi = min(n, k + 1)
        for j in range(lo, hi):
            out[r, j] = f[k - j]
    return out


def _lp_direction(f, M, idx_list):
    """Solve LP: maximize sum(b) s.t. (f*b)_k <= M for k in idx_list, b >= 0."""
    n = len(f)
    A = _conv_rows(f, idx_list)
    res = linprog(-np.ones(n), A_ub=A, b_ub=np.full(len(idx_list), M),
                  bounds=[(0, None)] * n, method='highs',
                  options={'presolve': True, 'time_limit': 15})
    if not res.success or res.x is None:
        return None
    return np.clip(res.x, 0, None)


def _lp_optimize(f, budget_s, t0=0.4, rng=None, verbose=False):
    """LP-based Frank-Wolfe style optimization with backtracking."""
    if rng is None:
        rng = np.random.default_rng(0)
    f = np.maximum(f, 0.0).astype(np.float64)
    S = f.sum()
    if S < 1e-15:
        f = np.ones(len(f)) / len(f)
    else:
        f /= S
    n = len(f)
    best = f.copy()
    bs = _score_norm(f)
    t = t0
    stall = 0
    it = 0
    t_start = time.monotonic()

    while time.monotonic() - t_start < budget_s:
        c = np.convolve(f, f)
        M = c.max()
        # Use top min(30, 2n-1) tightest constraints
        n_con = min(30, 2 * n - 1)
        idx = np.argsort(c)[::-1][:n_con].tolist()

        b = _lp_direction(f, M, idx)
        if b is None or b.sum() < 1e-15:
            t *= 0.5
            stall += 1
            if stall > 50:
                break
            continue

        g = b / b.sum()
        accepted = False
        for _ in range(20):
            f_new = np.clip((1.0 - t) * f + t * g, 0, None)
            S_new = f_new.sum()
            if S_new < 1e-15:
                t *= 0.5
                continue
            f_new /= S_new
            s = _score_norm(f_new)
            if s < bs - 1e-12:
                f = f_new
                best = f_new.copy()
                bs = s
                stall = 0
                accepted = True
                break
            t *= 0.65

        if not accepted:
            stall += 1
            if stall % 30 == 0:
                # Random perturbation to escape
                fp = f + rng.normal(0, 0.02, n)
                fp = np.clip(fp, 0, None)
                Sp = fp.sum()
                if Sp > 1e-10:
                    fp /= Sp
                    if _score_norm(fp) < bs:
                        f = fp
                        best = fp.copy()
                        bs = _score_norm(f)
                        t = t0 * 0.8
                        stall = 0
            if stall > 200:
                break

        if verbose and it % 50 == 0:
            print(f"    LP it={it}, score={bs:.6f}, t={t:.4f}, elapsed={time.monotonic()-t_start:.1f}s")
        it += 1

    return best, bs


def _mw_optimize(f, budget_s, eta0=0.5, topk=3, rng=None):
    """Multiplicative weights (mirror descent) optimization."""
    if rng is None:
        rng = np.random.default_rng(0)
    f = np.maximum(f, 0.0).astype(np.float64)
    f /= f.sum()
    n = len(f)
    best = f.copy()
    bs = _score_norm(f)
    eta = eta0
    t_start = time.monotonic()
    it = 0

    while time.monotonic() - t_start < budget_s:
        c = np.convolve(f, f)
        top = np.argsort(c)[::-1][:topk]
        g = np.zeros(n)
        for k in top:
            lo = max(0, k - n + 1)
            hi = min(n, k + 1)
            # g[lo:hi] += 2 * f[k-i] for i in [lo, hi)
            # f[k-lo], f[k-lo-1], ..., f[k-hi+1]
            g[lo:hi] += 2.0 * f[k - hi + 1: k - lo + 1][::-1]
        g /= topk

        # Multiplicative (exponential) update
        f = f * np.exp(-eta * g)
        f = np.clip(f, 0, None)
        S = f.sum()
        if S < 1e-15:
            f = np.ones(n) / n
        else:
            f /= S

        s = _score_norm(f)
        if s < bs:
            bs = s
            best = f.copy()
        else:
            # Anneal eta slightly
            if it % 100 == 0 and eta > 0.05:
                eta *= 0.95

        it += 1

    return best, bs


def _smooth_gradient_optimize(f, budget_s, lr=0.01, tau=0.005, rng=None):
    """Smooth LogSumExp-based gradient optimization with Adam."""
    if rng is None:
        rng = np.random.default_rng(0)
    f = np.maximum(f, 0.0).astype(np.float64)
    f /= f.sum()
    n = len(f)
    best = f.copy()
    bs = _score_norm(f)

    m = np.zeros(n)
    v = np.zeros(n)
    beta1, beta2, eps = 0.85, 0.999, 1e-8
    t_step = 0
    t_start = time.monotonic()

    while time.monotonic() - t_start < budget_s:
        c = np.convolve(f, f)
        # LogSumExp smoothing
        shifted = c / tau - (c / tau).max()
        w = np.exp(shifted)
        w /= w.sum()

        # Gradient: grad[i] = sum_k w[k] * 2 * f[k-i] (for valid k-i)
        grad = np.zeros(n)
        for k in range(2 * n - 1):
            if w[k] < 1e-12:
                continue
            lo = max(0, k - n + 1)
            hi = min(n, k + 1)
            grad[lo:hi] += w[k] * 2.0 * f[k - hi + 1: k - lo + 1][::-1]

        # Adam update
        t_step += 1
        lr_t = lr / (1.0 + t_step * 0.0005)
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad * grad)
        m_hat = m / (1 - beta1 ** t_step)
        v_hat = v / (1 - beta2 ** t_step)

        f_new = f - lr_t * m_hat / (np.sqrt(v_hat) + eps)
        f_new = np.clip(f_new, 0, None)
        S = f_new.sum()
        if S < 1e-15:
            f_new = np.ones(n) / n
        else:
            f_new /= S

        s = _score_norm(f_new)
        if s < bs:
            bs = s
            best = f_new.copy()
        f = f_new

    return best, bs


def _local_search(f, rng, budget_s=10.0):
    """Random local search for polishing."""
    f = f.astype(np.float64).copy()
    f = np.clip(f, 0, None)
    f /= f.sum()
    n = len(f)
    best = f.copy()
    bs = _score_norm(f)
    t_start = time.monotonic()

    while time.monotonic() - t_start < budget_s:
        fc = f.copy()
        # Perturb a random block
        width = max(1, int(rng.integers(1, max(2, n // 32))))
        start = int(rng.integers(0, n - width))
        scale = np.exp(rng.normal(0, 0.25))
        fc[start:start + width] *= scale
        fc = np.clip(fc, 0, None)
        S = fc.sum()
        if S < 1e-15:
            continue
        fc /= S
        s = _score_norm(fc)
        if s < bs:
            bs = s
            best = fc.copy()
            f = fc
        elif rng.random() < 0.01:
            # occasional larger perturbation
            fc2 = f * np.exp(rng.normal(0, 0.5, n))
            fc2 = np.clip(fc2, 0, None)
            S2 = fc2.sum()
            if S2 > 1e-10:
                fc2 /= S2
                if _score_norm(fc2) < bs:
                    bs = _score_norm(fc2)
                    best = fc2.copy()
                    f = fc2

    return best, bs

                continue
            f0 /= S0

            t_left = deadline - time.monotonic() - 3
            if t_left < 10:
                break
            # Quick LP + MW
            f_r, s_r = _lp_optimize(f0, t_left * 0.6, t0=0.3, rng=rng)
            f_r2, s_r2 = _mw_optimize(f_r, t_left * 0.3, eta0=0.3, topk=3, rng=rng)
            s_final = min(s_r, s_r2)
            f_final = f_r if s_r <= s_r2 else f_r2

            if s_final < best_score:
                best_score = s_final
                best_overall = f_final.copy()
                print(f"  restart {r}: {s_final:.6f} ** IMPROVED **", flush=True)
            else:
                print(f"  restart {r}: {s_final:.6f}", flush=True)

    # Final: verify with evaluate_sequence and convert to list
    if best_overall is not None:
        best_overall = np.clip(best_overall, 0, 1000)
        final_list = best_overall.tolist()
        final_score = evaluate_sequence(final_list)
        print(f"\n=== FINAL: score={final_score:.6f}, length={len(final_list)} ===", flush=True)
        return final_list

    # Fallback
    print("Fallback: uniform 256", flush=True)
    return [1.0] * 256
