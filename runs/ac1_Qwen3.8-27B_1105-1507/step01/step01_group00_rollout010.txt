import numpy as np
import time
from scipy.optimize import linprog
from scipy.sparse import csr_matrix

def build_conv_matrix_sparse(f):
    """Build sparse matrix A where (A@g)[k] = convolve(f,g)[k]."""
    n = len(f)
    # For each k in range(2n-1), the non-zero entries are at positions j where max(0, k-(n-1)) <= j <= min(k, n-1)
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        j_min = max(0, k - (n - 1))
        j_max = min(k, n - 1)
        for j in range(j_min, j_max + 1):
            rows.append(k)
            cols.append(j)
            vals.append(f[k - j])
    return csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))


def solve_lp_with_active_constraints(f, M, active_frac=1.0, deadline=None, max_iter_lp=200):
    """Solve LP with optional active constraint reduction."""
    n = len(f)
    A = build_conv_matrix_sparse(f)
    
    # Get current convolution
    conv_f = np.convolve(f, f)
    
    if active_frac < 1.0:
        # Only keep top active_frac fraction of constraints
        num_constraints = int(len(conv_f) * active_frac)
        sorted_idx = np.argsort(-conv_f)  # descending
        active_idx = sorted_idx[:num_constraints]
        A_active = A[active_idx, :]
        b_active = np.full(len(active_idx), M)
    else:
        A_active = A
        b_active = np.full(2 * n - 1, M)
    
    c = -np.ones(n)
    bounds = [(0, None)] * n
    
    try:
        result = linprog(c, A_ub=A_active, b_ub=b_active, bounds=bounds, 
                        method='highs', options={'maxiter': max_iter_lp})
        if result.success:
            return result.x
    except Exception:
        pass
    return None

        # (most impact per unit of mass removed)
        ratios = grad[candidates]
        # Move mass from high-grad coordinates to low-grad coordinates
        max_grad_idx = candidates[np.argmax(ratios)]
        min_grad_idx = candidates[np.argmin(ratios)]

            lr *= 0.9

        new_f[max_grad_idx] -= delta
        new_f[min_grad_idx] += delta
        
        if new_f[max_grad_idx] < 0:
            delta = f[max_grad_idx] * 0.5
            new_f[max_grad_idx] = f[max_grad_idx] - delta
            new_f[min_grad_idx] = f[min_grad_idx] + delta
        
        new_f = np.clip(new_f, 0.0, 1000.0)

def random_perturb(f, magnitude=0.1, rng=None):
    """Add random perturbation."""
    if rng is None:
        rng = np.random.default_rng()
    n = len(f)
    S = float(np.sum(f))
    # Add random noise and renormalize
    noise = rng.uniform(-magnitude, magnitude, size=n)
    new_f = f + noise
    new_f = np.clip(new_f, 0.0, None)
    new_sum = float(np.sum(new_f))
    if new_sum > 0:
        new_f = new_f * (S / new_sum)
    return new_f

"""
Multi-stage optimization for minimizing 2n*max(conv(a,a))/sum(a)^2.
Uses LP with active constraint reduction, coordinate descent polish,
and random restarts. Tries multiple sequence lengths with diverse initializations.
"""
import numpy as np
import time
from scipy.optimize import linprog
from scipy.sparse import csr_matrix


def build_conv_matrix_sparse(f):
    """Build sparse matrix A where (A@g)[k] = sum_j f[k-j]*g[j] = convolve(f,g)[k]."""
    n = len(f)
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        j_min = max(0, k - (n - 1))
        j_max = min(k, n - 1)
        for j in range(j_min, j_max + 1):
            rows.append(k)
            cols.append(j)
            vals.append(f[k - j])
    return csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))


def solve_lp_direction(f, M, active_frac=1.0, time_limit=30.0):
    """Solve: max sum(g) s.t. g >= 0, convolve(f,g)[k] <= M for (active) k."""
    n = len(f)
    A = build_conv_matrix_sparse(f)
    conv_f = np.convolve(f, f)
    
    if active_frac < 1.0:
        num_active = max(1, int(len(conv_f) * active_frac))
        sorted_idx = np.argsort(-conv_f)
        active_idx = sorted_idx[:num_active]
        A_use = A[active_idx, :]
        b_use = np.full(len(active_idx), M)
    else:
        A_use = A
        b_use = np.full(2 * n - 1, M)
    
    c = -np.ones(n)
    bounds = [(0, None)] * n
    
    try:
        result = linprog(c, A_ub=A_use, b_ub=b_use, bounds=bounds,
                         method='highs', options={'time_limit': time_limit})
        if result.success:
            return result.x
    except Exception:
        pass
    return None


def lp_iterate(f, deadline, step_size_init=0.15, active_frac=1.0,
               max_no_improve=50, verbose_prefix=""):

        conv_f = np.convolve(f, f)
        M = float(np.max(conv_f))
        if M <= 0:
            break
        
        start_t = time.monotonic()
        g = solve_lp_direction(f, M, active_frac=active_frac,
                               time_limit=max(1.0, deadline - time.monotonic() - 1))
        lp_time = time.monotonic() - start_t
        
        if g is None:
            no_improve += 1
            if no_improve > 5:
                break
            continue
        
        sum_g = float(np.sum(g))
        if sum_g <= 1e-15:
            no_improve += 1
            continue
        
        sum_f = float(np.sum(f))
        g = g * (sum_f / sum_g)
        
        new_f = (1.0 - step_size) * f + step_size * g
        new_f = np.clip(new_f, 0.0, 1000.0)
        
        new_score = evaluate_sequence(new_f.tolist())

        else:
            no_improve += 1
            step_size *= 0.75

def gradient_descent_polish(f, deadline, lr_init=0.01, max_iter=3000, verbose_prefix=""):
    """
    Subgradient method: decrease coordinates contributing to the peak convolution value.
    """
    f = f.copy().astype(np.float64)
    best_f = f.copy()
    best_score = evaluate_sequence(f.tolist())
    n = len(f)
    lr = lr_init
    
    for it in range(max_iter):
        if time.monotonic() > deadline:
            break
        
        conv_f = np.convolve(f, f)
        max_idx = int(np.argmax(conv_f))
        M = conv_f[max_idx]
        S = float(np.sum(f))
        
        if M <= 0 or S <= 0:
            break
        
        # Gradient of conv[max_idx] w.r.t. f[j] = 2*f[max_idx-j] if valid
        grad = np.zeros(n)
        for j in range(n):
            if 0 <= max_idx - j < n:
                grad[j] = 2.0 * f[max_idx - j]
        
        new_f = f - lr * grad
        new_f = np.clip(new_f, 0.0, 1000.0)
        
        # Renormalize to preserve sum
        new_sum = float(np.sum(new_f))
        if new_sum > 1e-10:
            new_f = new_f * (S / new_sum)
        
        new_score = evaluate_sequence(new_f.tolist())
        if np.isfinite(new_score) and new_score < best_score:
            f = new_f
            best_f = f.copy()
            best_score = new_score
            if (it + 1) % 1000 == 0:
                print(f"    {verbose_prefix}GD iter {it+1}: score={best_score:.6f} (lr={lr:.5f})")
        else:
            lr *= 0.85
            if lr < 1e-7:
                break
    
    return best_f, best_score


def coordinate_transfer_polish(f, deadline, max_iter=5000, delta_frac=0.05, verbose_prefix=""):
    """
    Transfer mass from coordinates with high autocorrelation contribution to
    coordinates with low contribution.
    """
    f = f.copy().astype(np.float64)
    best_f = f.copy()
    best_score = evaluate_sequence(f.tolist())
    n = len(f)
    
    for it in range(max_iter):
        if time.monotonic() > deadline:
            break
        
        conv_f = np.convolve(f, f)
        max_idx = int(np.argmax(conv_f))
        M = conv_f[max_idx]
        S = float(np.sum(f))
        
        if M <= 0 or S <= 0:
            break
        
        # Contribution of each coordinate to conv[max_idx]
        contrib = np.zeros(n)
        for j in range(n):
            if 0 <= max_idx - j < n:
                contrib[j] = 2.0 * f[j] * f[max_idx - j]
        
        # Find source (high contrib, high value) and target (low contrib, low value)
        pos = np.where(f > 0.01 * S / n)[0]
        if len(pos) < 2:
            break
        
        src_candidates = pos[contrib[pos] > np.median(contrib[pos])]
        dst_candidates = pos[contrib[pos] <= np.median(contrib[pos])]
        
        if len(src_candidates) == 0 or len(dst_candidates) == 0:
            break
        
        src = src_candidates[np.argmax(contrib[src_candidates])]
        dst = dst_candidates[np.argmin(contrib[dst_candidates])]
        
        if src == dst:
            break
        
        delta = delta_frac * f[src]
        new_f = f.copy()
        new_f[src] -= delta
        new_f[dst] += delta
        new_f = np.clip(new_f, 0.0, 1000.0)
        
        new_score = evaluate_sequence(new_f.tolist())
        if np.isfinite(new_score) and new_score < best_score:
            f = new_f
            best_f = f.copy()
            best_score = new_score
            if (it + 1) % 500 == 0:
                print(f"    {verbose_prefix}CT iter {it+1}: score={best_score:.6f}")
        else:
            delta_frac *= 0.9
            if delta_frac < 1e-5:
                break
    
    return best_f, best_score


def random_perturb(f, magnitude=0.05, rng=None):
    """Add random perturbation and renormalize."""
    if rng is None:
        rng = np.random.default_rng()
    n = len(f)
    S = float(np.sum(f))
    noise = rng.uniform(-magnitude, magnitude, size=n)
    new_f = f + noise
    new_f = np.clip(new_f, 0.0, None)
    new_sum = float(np.sum(new_f))
    if new_sum > 1e-10:
        new_f = new_f * (S / new_sum)
    return new_f

    inits.append(('ramp', 0.5 + np.linspace(0, 1, n)))

    p2 = np.exp(-((idx - n * 0.75) ** 2) / (2 * (n * 0.08) ** 2))
    inits.append(('bimodal', p1 + p2 + 0.02))
    
    # Triangular (peak in center)
    tri = 1.0 - 2.0 * np.abs(idx - (n - 1) / 2.0) / (n - 1)
    inits.append(('tri', np.clip(tri, 0.05, None)))
    
    # Inverse triangle (dip in center)
    inv_tri = 0.3 + 2.0 * np.abs(idx - (n - 1) / 2.0) / (n - 1)
    inits.append(('invtri', inv_tri))
    
    # Alternating
    alt = np.where(idx.astype(int) % 2 == 0, 1.3, 0.7)
    inits.append(('alt', alt))
    
    # Power law
    inits.append(('power', (idx + 1) ** 0.5 / np.sqrt(n)))
    
    # Logarithmic
    inits.append(('log', np.log(idx + 1) / np.log(n + 1)))
    
    # Random
    inits.append(('rand1', rng.uniform(0.1, 1.5, size=n)))
    inits.append(('rand2', rng.exponential(1.0, size=n) + 0.01))
    
    # Sine modulated
    sine_mod = 0.5 + 0.4 * np.sin(2 * np.pi * idx / (n / 3.0))
    inits.append(('sinmod', np.clip(sine_mod, 0.05, None)))
    
    # Smooth random (low-pass filtered random)
    raw = rng.standard_normal(n)
    kernel = np.exp(-0.5 * (np.arange(-5, 6) / 2.0) ** 2)

    smooth = np.convolve(raw, kernel, mode='same')
    smooth = np.clip(smooth, 0, None)
    if smooth.max() > 0:

        candidates_n.add(pn // 2)
        candidates_n.add(pn * 2)
        candidates_n.update([128, 256, 512, 1024])
        lengths = sorted(n for n in candidates_n if 64 <= n <= 2048)
    else:
        lengths = [256, 512, 1024]
    
    print(f"Lengths to try: {lengths}")
    
    # Phase 1: Explore different lengths with diverse initializations
    for n in lengths:
        if time.monotonic() > deadline - 60:
            break
        
        print(f"\n=== n={n} ===")
        
        inits = generate_initializations(n, rng)
        
        # If parent has this length, add it
        if best_sequence is not None and len(best_sequence) == n:
            inits.insert(0, ('parent', best_sequence.copy()))
        
        # We'll do fewer inits for larger n due to time constraints
        if n >= 1024:
            inits = inits[:8]
        
        for label, f_init in inits:
            if time.monotonic() > deadline - 30:
                break
            
            f_init = np.clip(f_init.astype(np.float64), 0.001, 1000.0)
            score = evaluate_sequence(f_init.tolist())
            if not np.isfinite(score):
                continue
            
            if score < best_score:
                best_score = score
                best_sequence = f_init.copy()
            
            # Quick LP pass to see if this init is promising
            quick_deadline = min(deadline, time.monotonic() + 30)
            best_f, best_s = lp_iterate(f_init, quick_deadline,
                                         step_size_init=0.2, active_frac=0.5,
                                         max_no_improve=20,
                                         verbose_prefix=f"[{n}/{label}] ")
            
            if best_s < best_score:
                best_score = best_s
                best_sequence = best_f.copy()
            
            print(f"  [{label}] init={score:.6f} -> {best_s:.6f}")
    
    # Phase 2: Deep optimization on the best sequence found
    if best_sequence is not None:
        n_best = len(best_sequence)
        print(f"\n=== Deep optimization n={n_best}, current best={best_score:.6f} ===")
        
        # Multiple LP passes with decreasing active fraction
        for af in [1.0, 0.7, 0.5, 0.35, 0.25]:
            if time.monotonic() > deadline - 20:
                break
            sub_deadline = min(deadline, time.monotonic() + 90)
            best_f, best_s = lp_iterate(best_sequence, sub_deadline,
                                         step_size_init=0.08 if af < 1.0 else 0.15,
                                         active_frac=af,
                                         max_no_improve=40,
                                         verbose_prefix=f"[deep/af={af}] ")
            if best_s < best_score:
                best_score = best_s
                best_sequence = best_f.copy()
        
        # Gradient descent polish
        if time.monotonic() < deadline - 15:
            gf, gs = gradient_descent_polish(best_sequence, min(deadline, time.monotonic() + 60),
                                             lr_init=0.005, verbose_prefix="[GD] ")
            if gs < best_score:
                best_score = gs
                best_sequence = gf.copy()
        
        # Coordinate transfer polish
        if time.monotonic() < deadline - 10:
            cf, cs = coordinate_transfer_polish(best_sequence, min(deadline, time.monotonic() + 40),
                                                max_iter=3000, verbose_prefix="[CT] ")
            if cs < best_score:
                best_score = cs
                best_sequence = cf.copy()
        
        # Random restarts
        for attempt in range(15):
            if time.monotonic() > deadline - 10:
                break
            mag = 0.03 + 0.02 * (attempt % 3)
            perturbed = random_perturb(best_sequence, magnitude=mag, rng=rng)
            sub_deadline = min(deadline, time.monotonic() + 40)
            best_f, best_s = lp_iterate(perturbed, sub_deadline,
                                         step_size_init=0.1, active_frac=0.5,
                                         max_no_improve=20,
                                         verbose_prefix=f"[restart/{attempt}] ")
            if best_s < best_score:
                best_score = best_s
                best_sequence = best_f.copy()
                print(f"  Restart {attempt}: improved to {best_s:.6f}")
        
        # Final gradient + coordinate transfer
        if time.monotonic() < deadline - 5:
            gf, gs = gradient_descent_polish(best_sequence, min(deadline, time.monotonic() + 30),
                                             lr_init=0.002, verbose_prefix="[GD2] ")
            if gs < best_score:
                best_score = gs
                best_sequence = gf.copy()
        
        if time.monotonic() < deadline - 3:
            cf, cs = coordinate_transfer_polish(best_sequence, deadline - 2,
                                                max_iter=2000, verbose_prefix="[CT2] ")
            if cs < best_score:
                best_score = cs
                best_sequence = cf.copy()
    
    # Phase 3: Try one more length if time remains
    if best_sequence is not None and time.monotonic() < deadline - 100:
        try_lengths = [n for n in [len(best_sequence) * 2, len(best_sequence) // 2]
                      if n != len(best_sequence) and 64 <= n <= 2048]
        for n in try_lengths:
            if time.monotonic() > deadline - 30:
                break
            print(f"\n=== Exploring n={n} ===")
            inits = generate_initializations(n, rng)[:6]
            for label, f_init in inits:
                if time.monotonic() > deadline - 15:
                    break
                f_init = np.clip(f_init.astype(np.float64), 0.001, 1000.0)
                score = evaluate_sequence(f_init.tolist())
                if not np.isfinite(score):
                    continue
                if score < best_score:
                    best_score = score
                    best_sequence = f_init.copy()
                    print(f"  [{label}] NEW BEST: {best_score:.6f}")
                else:
                    # Quick check if LP can improve
                    qd = min(deadline, time.monotonic() + 25)
                    bf, bs = lp_iterate(f_init, qd, step_size_init=0.15,
                                        active_frac=0.5, max_no_improve=15)
                    if bs < best_score:
                        best_score = bs
                        best_sequence = bf.copy()
                        print(f"  [{label}] LP improved to: {best_score:.6f}")
    
    if best_sequence is None:
        return [1.0] * 128
    
    elapsed = time.monotonic() - t0
    final_score = evaluate_sequence(best_sequence.tolist())
    print(f"\nFINAL: n={len(best_sequence)}, score={final_score:.6f}, time={elapsed:.1f}s")
    return [float(x) for x in best_sequence]
