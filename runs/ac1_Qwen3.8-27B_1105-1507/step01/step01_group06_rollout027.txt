    Returns (gradient, smoothed_max_value, b).

    n = len(a)
    c = -np.ones(n)

        jh = min(n - 1, k)

               bounds=[(0, None)] * n, method='highs')

        g = r.x
        if g.sum() > 1e-10:
            return g
    return None

    n = len(a)

            s[i] = a[j]

    a_new = np.maximum(a_new, 0)
    s_norm = a_new.sum()
    if s_norm > 1e-10:
        a_new /= s_norm
    return a_new

    """min t s.t. (a*g)_k <= t for all k, sum(g) = S, g >= 0. Returns (g, t)."""
    n = len(a)
    # Variables: g_0, ..., g_{n-1}, t
    # Objective: minimize t (variable index n)
    c = np.zeros(n + 1)
    c[n] = 1.0
    
    # Constraints:
    # (a*g)_k <= t for k = 0, ..., 2n-2
    # sum(g) = S
    # g_i >= 0

        jh = min(n-1, k)

        A_ub[k, n] = -1.0

    # Equality: sum(g) = S
    A_eq = np.zeros((1, n+1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([S])
    
    bounds = [(0, None)] * n + [(None, None)]

               bounds=bounds, method='highs')
    if r.success:

        t = r.x[n]
        return g, t
    return None, None

"""
Minimizes J(a) = 2*n*max(convolve(a,a))/(sum(a))^2 using:
1. Smoothed-max gradient descent with adaptive temperature and backtracking
2. LP best-response mirror descent with adaptive mixing
3. Min-peak LP for alternative updates
4. Subgradient polishing and structured restarts
5. Multi-scale search (n=512 primary, n=1024 secondary)

import numpy as np
from scipy.optimize import linprog
import time


def fast_conv(a):
    """Full self-convolution via FFT, O(n log n)."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]


def normalize_nonneg(a):
    """Clip to non-negative and normalize to unit sum."""
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-12:
        return np.ones_like(a) / len(a)
    return a / s


def compute_smooth_grad(a, tau):
    """
    Compute gradient of smoothed max of convolve(a,a).
    smoothed_max = (1/tau) * log(sum_k exp(tau * b_k))
    Gradient g_i = sum_k p_k * a_{k-i} (valid indices)
    Returns (gradient, smoothed_max_value, b).
    """
    n = len(a)
    b = fast_conv(a)
    b_max = b.max()
    tau_b = tau * (b - b_max)
    tau_b = np.clip(tau_b, -500.0, 500.0)
    e = np.exp(tau_b)
    p = e / e.sum()
    smoothed_max = b_max + np.log(e.sum()) / tau

    # g_i = sum_{j: 0<=j<n, 0<=i+j<2n-1} p[i+j] * a[j]
    g = np.zeros(n)
    for i in range(n):
        j_max = min(n, 2 * n - 1 - i)
        if j_max > 0:
            g[i] = np.dot(p[i:i + j_max], a[:j_max])
    return g, smoothed_max, b


def lp_best_response(a, M):
    """LP: max sum(g) s.t. (a*g)_k <= M, g >= 0. Returns g or None."""
    n = len(a)
    c = -np.ones(n)
    A = np.zeros((2 * n - 1, n))
    for k in range(2 * n - 1):
        jl = max(0, k - n + 1)
        jh = min(n - 1, k)
        if jl <= jh:
            A[k, jl:jh + 1] = a[k - jh:k - jl + 1][::-1]
    r = linprog(c, A_ub=A, b_ub=np.full(2 * n - 1, M),
               bounds=[(0, None)] * n, method='highs')
    if r.success:
        g = r.x
        if g.sum() > 1e-10:
            return g
    return None


def lp_min_peak(a, target_sum):
    """
    LP: min t s.t. (a*g)_k <= t for all k, sum(g) = target_sum, g >= 0.
    Returns (g, t) or (None, inf).
    """
    n = len(a)
    c = np.zeros(n + 1)
    c[n] = 1.0

    A_ub = np.zeros((2 * n - 1, n + 1))
    for k in range(2 * n - 1):
        jl = max(0, k - n + 1)
        jh = min(n - 1, k)
        if jl <= jh:
            A_ub[k, jl:jh + 1] = a[k - jh:k - jl + 1][::-1]
        A_ub[k, n] = -1.0

    b_ub = np.zeros(2 * n - 1)
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([target_sum])
    bounds = [(0, None)] * n + [(-np.inf, np.inf)]

    r = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
               bounds=bounds, method='highs')
    if r.success:
        g = r.x[:n]
        t = r.x[n]
        return g, t
    return None, np.inf


def subgradient_step(a, step):
    """One step of subgradient descent on max(convolve(a,a))."""
    n = len(a)
    b = fast_conv(a)
    k_star = int(b.argmax())
    s = np.zeros(n)
    for i in range(n):
        j = k_star - i
        if 0 <= j < n:
            s[i] = a[j]
    a_new = a - step * s
    return normalize_nonneg(a_new)

def smoothed_grad_descent(a, n_iters, deadline, base_tau, tau_growth, state, verbose=False):
    """Run smoothed-max gradient descent with backtracking line search."""
    for it in range(n_iters):
        if time.monotonic() > deadline:
            break

        tau = base_tau * (1.0 + it / tau_growth)
        step = 1.0 / (tau * a.sum() + 1e-10)
        b_curr = fast_conv(a)
        peak_curr = b_curr.max()

        improved = False
        for ls in range(25):
            g, sm, _ = compute_smooth_grad(a, tau)
            g_norm = np.linalg.norm(g)
            if g_norm < 1e-14:
                break
            g = g / g_norm
            a_trial = a - step * g
            a_trial = np.maximum(a_trial, 0.0)
            trial_sum = a_trial.sum()
            if trial_sum < 1e-10:
                step *= 0.5
                continue
            a_trial /= trial_sum
            peak_trial = fast_conv(a_trial).max()
            if peak_trial < peak_curr * (1.0 - 1e-5):
                a = a_trial
                improved = True
                break
            else:
                step *= 0.65

        if not improved and it > 5:
            # Try a random perturbation to escape
            if it % 20 == 0:
                noise = np.random.default_rng(it).normal(0, 0.05, len(a))
                a = normalize_nonneg(a + noise)
            elif it > 50:
                break

        if verbose and it % 50 == 0:
            b_check = fast_conv(a)
            print(f"    iter {it}: peak={b_check.max():.8f}, tau={tau:.1f}", flush=True)

    try_update(a, state)
    return a


def lp_mirror_descent(a, n_iters, deadline, state, verbose=False):
    """LP best-response mirror descent with adaptive step size."""
    prev_peak = fast_conv(a).max()
    stall_count = 0

    for it in range(n_iters):
        if time.monotonic() > deadline:
            break

        b = fast_conv(a)
        M = float(b.max())
        g = lp_best_response(a, M)
        if g is None:
            break
        g = normalize_nonneg(g)

        # Adaptive mixing coefficient
        t_mix = 0.4 * (1.0 - 0.5 * it / max(1, n_iters)) + 0.1
        a_new = (1.0 - t_mix) * a + t_mix * g
        a_new = np.maximum(a_new, 0.0)
        new_peak = fast_conv(normalize_nonneg(a_new)).max()

        if new_peak < prev_peak:
            a = normalize_nonneg(a_new)
            prev_peak = new_peak
            stall_count = 0
        else:
            # Try smaller step
            t_mix_small = t_mix * 0.3
            a_new = (1.0 - t_mix_small) * a + t_mix_small * g
            a_new = np.maximum(a_new, 0.0)
            new_peak = fast_conv(normalize_nonneg(a_new)).max()
            if new_peak < prev_peak:
                a = normalize_nonneg(a_new)
                prev_peak = new_peak
            else:
                stall_count += 1
                if stall_count > 20:
                    break

        if verbose and it % 30 == 0:
            print(f"    LP iter {it}: peak={prev_peak:.8f}", flush=True)

    try_update(a, state)
    return a


def peak_shave(a, n_steps, deadline, state):
    """Direct peak shaving: reduce elements contributing most to the peak."""
    for _ in range(n_steps):
        if time.monotonic() > deadline:
            break
        b = fast_conv(a)
        k_star = int(b.argmax())
        # Find elements contributing most to b[k_star]
        contribs = []
        for i in range(len(a)):
            j = k_star - i
            if 0 <= j < len(a):
                contribs.append((a[i] * a[j], i, j))
        contribs.sort(reverse=True)

        # Reduce top contributors slightly
        n_reduce = max(1, len(contribs) // 10)
        total_reduction = 0.0
        for _, i, j in contribs[:n_reduce]:
            reduction = 0.01 * a[i]
            a[i] -= reduction
            total_reduction += reduction
            if i != j:
                a[j] = a[j] + reduction * (a[j] / a[i] if a[i] > 0 else 0)

        a = normalize_nonneg(a)

    try_update(a, state)
    return a
