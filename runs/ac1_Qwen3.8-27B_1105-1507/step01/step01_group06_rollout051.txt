    exp_vals = np.exp(Tb_shifted)
    p = exp_vals / exp_vals.sum()
    
    h = cross_corr_fft(p, a)
    p_even = p[::2]  # length n
    grad = 2.0 * h - p_even * a
    return grad, b

def build_A_toeplitz(a):
    """Build the matrix A for LP: A[k,j] = a[k-j] if valid."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))

        hi = min(n - 1, k)
        if lo <= hi:
            A[k, lo:hi+1] = a[k - hi:k - lo + 1][::-1]
    return A

def lp_best_response(a, M):
    """Solve LP: max sum(g) s.t. (a*g)_k <= M, g >= 0."""
    n = len(a)
    c = -np.ones(n)

    b = np.full(2 * n - 1, M)
    res = linprog(c, A_ub=A, b_ub=b, bounds=[(0, None)] * n, method='highs')
    if res.success and res.x is not None:
        g = res.x
        if g.sum() > 1e-10:
            return g / g.sum()
    return None

            best_bmax = bmax
            best_a = a.copy()

"""
Minimizes J(a) = 2*n*max(convolve(a,a))/(sum(a))^2 using:
1. Temperature-annealed log-sum-exp mirror descent (multiplicative updates)
   with FFT-based gradient computation for O(n log n) per iteration.
2. LP-based best-response polish for final refinement.
3. Multiple sequence lengths and diverse structured initializations.
The log-sum-exp surrogate smoothly approximates the max, enabling
gradient-based optimization. Mirror descent with entropy mirror map
naturally maintains the simplex constraint (a>=0, sum(a)=1).
"""

import numpy as np
from scipy.optimize import linprog
import time


def fast_conv(a):
    """Full self-convolution via FFT, O(n log n)."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]


def cross_corr_fft(p, a):
    """Compute h[j] = sum_k p[k] * a[k-j] for j=0..n-1 via FFT."""
    n = len(a)
    size = 1
    while size < len(p) + n:
        size <<= 1
    Fp = np.fft.rfft(p, size)
    Fa = np.fft.rfft(a, size)
    return np.fft.irfft(Fp * np.conj(Fa), size)[:n]


def simplex_project(z):
    """Project z onto probability simplex {x >= 0, sum(x) = 1}."""
    n = len(z)
    if n == 0:
        return z
    u = np.sort(z)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    idx = np.nonzero(cond)[0]
    if len(idx) == 0:
        return np.full(n, 1.0 / n)
    rho = idx[-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(z - theta, 0.0)


def mirror_step(a, grad, lr):
    """Mirror descent step (multiplicative updates) on simplex."""
    log_ratio = -lr * grad
    log_ratio -= log_ratio.max()
    a_new = a * np.exp(log_ratio)
    s = a_new.sum()
    if s < 1e-300:
        return a.copy()
    return a_new / s


def compute_gradient_smooth(a, T):
    """Gradient of (1/T)*log(sum(exp(T*b))) where b = a*a."""
    n = len(a)
    b = fast_conv(a)
    Tb = T * b
    Tb_shifted = Tb - Tb.max()
    exp_vals = np.exp(Tb_shifted)
    p = exp_vals / exp_vals.sum()
    h = cross_corr_fft(p, a)
    p_even = p[::2]
    grad = 2.0 * h - p_even * a
    return grad, b


def softmax_mirror_descent(a0, time_budget, seed=0):
    """Annealed log-sum-exp mirror descent on the simplex."""
    rng = np.random.default_rng(seed)
    n = len(a0)
    a = simplex_project(np.maximum(a0.astype(np.float64), 1e-10).copy())
    a = a / a.sum()
    deadline = time.monotonic() + time_budget

    best_a = a.copy()
    best_bmax = float(fast_conv(a).max())

    phases = [
        (0.5, 3.0, 1.5, 0.12),
        (3.0, 15.0, 0.8, 0.20),
        (15.0, 80.0, 0.3, 0.30),
        (80.0, 400.0, 0.12, 0.25),
        (400.0, 2000.0, 0.05, 0.13),
    ]

    check_interval = max(50, n // 5)

    for T_s, T_e, lr_base, frac in phases:
        if time.monotonic() > deadline:
            break
        phase_time = time_budget * frac
        phase_end = time.monotonic() + phase_time
        # Estimate iterations: ~0.5ms per iter for n<=1024, ~2ms for n=2048
        est_time_per_iter = max(0.0005, n * 0.0000005)
        n_iters = max(50, int(phase_time / est_time_per_iter))

        for it in range(n_iters):
            if time.monotonic() > phase_end:
                break
            frac_done = it / max(1, n_iters - 1)
            T = T_s + (T_e - T_s) * frac_done
            # Learning rate decays within phase
            lr = lr_base * (1.0 - 0.5 * frac_done)

            grad, b = compute_gradient_smooth(a, T)
            a = mirror_step(a, grad, lr)

            if (it + 1) % check_interval == 0:
                bmax = b.max()
                if bmax < best_bmax:
                    best_bmax = bmax
                    best_a = a.copy()

            # Periodic perturbation to escape local minima
            if (it + 1) % 500 == 0 and frac_done < 0.7:
                noise = 0.02 * rng.standard_normal(n)
                a = simplex_project(a + noise)

    bmax = float(fast_conv(a).max())
    if bmax < best_bmax:
        best_bmax = bmax
        best_a = a.copy()

    return best_a, float(2.0 * n * best_bmax)

    """Build constraint matrix for LP: A[k,j] = a[k-j] if 0<=k-j<n."""

def lp_best_response(a, M):
    """LP: max sum(g) s.t. (a*g)_k <= M for all k, g >= 0. Returns normalized g."""
    n = len(a)

        n_best = best_a.size

            print(f"\n--- Phase 3: Coord descent n={n_best} ---", flush=True)
            t0 = time.monotonic()
            alloc = min(120.0, deadline - time.monotonic())
            if alloc > 10:
                res_a, res_score = coordinate_descent_polish(best_a, alloc)
                record_candidate(res_a)
                print(f"  Coord descent: {time.monotonic()-t0:.1f}s, best={best_score:.6f}", flush=True)

    # === Phase 4: LP polish on coordinate-descent result ===
    if best_a is not None and time.monotonic() < deadline:
        n_best = best_a.size
        if n_best <= 1024:
            print(f"\n--- Phase 4: LP polish v2 n={n_best} ---", flush=True)
            alloc = min(150.0, deadline - time.monotonic())
            if alloc > 15:
                res_a, res_score = lp_polish(best_a, n_iters=80, time_budget=alloc)
                record_candidate(res_a)
                print(f"  LP v2: best={best_score:.6f}", flush=True)

    # === Phase 5: Try different length with remaining time ===
    if best_a is not None and time.monotonic() < deadline:
        for n_try in [4096, 1024]:
            if time.monotonic() > deadline:
                break
            if n_try == best_a.size:
                continue
            time_left = deadline - time.monotonic()
            if time_left < 60:
                break
            t_old = np.linspace(0, 1, best_a.size)
            t_new = np.linspace(0, 1, n_try)
            a_interp = np.maximum(np.interp(t_new, t_old, best_a), 0.01)
            a_interp /= a_interp.sum()
            alloc = min(time_left * 0.4, 180.0)
            print(f"\n--- Phase 5: softmax n={n_try} ---", flush=True)
            res_a, _ = softmax_mirror_descent(a_interp, alloc, seed=seed + 500)
            record_candidate(res_a)
            if n_try <= 1024 and time.monotonic() < deadline:
                alloc2 = min(deadline - time.monotonic(), 100.0)
                if alloc2 > 15:
                    res_a2, _ = lp_polish(res_a, n_iters=50, time_budget=alloc2)
                    record_candidate(res_a2)

    # Final validation
    final_score = float(evaluate_sequence(best_a.tolist()))
    if not np.isfinite(final_score):
        best_a = np.ones(256) / 256.0
        final_score = float(evaluate_sequence(best_a.tolist()))

    print(f"\n{'='*50}", flush=True)
    print(f"FINAL RESULT: J={final_score:.6f}, n={best_a.size}", flush=True)
    print(f"{'='*50}", flush=True)

    return [float(x) for x in best_a]
