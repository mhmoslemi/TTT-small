"""
Multi-method optimization for minimizing 2n*max(conv(a,a))/sum(a)^2.
Uses a combination of LP-based direction finding with line search,
fast subgradient descent with momentum, and multiple structured
initializations. Focuses computational budget on the most promising
sequence lengths and initializations identified during a quick
exploration phase.
"""
import numpy as np
import time
from scipy.optimize import linprog
from scipy.sparse import csr_matrix


def build_conv_matrix_sparse(f):
    """Build sparse matrix A where (A@g)[k] = convolve(f,g)[k]."""
    n = len(f)
    rows = np.empty(n * n, dtype=np.int32)
    cols = np.empty(n * n, dtype=np.int32)
    vals = np.empty(n * n, dtype=np.float64)
    idx = 0
    for k in range(2 * n - 1):
        j_min = max(0, k - (n - 1))
        j_max = min(k, n - 1)
        for j in range(j_min, j_max + 1):
            rows[idx] = k
            cols[idx] = j
            vals[idx] = f[k - j]
            idx += 1
    rows = rows[:idx]
    cols = cols[:idx]
    vals = vals[:idx]
    return csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))


def solve_lp_direction(f, M):
    """Solve: max sum(g) s.t. g >= 0, convolve(f,g)_k <= M for all k."""
    n = len(f)
    A = build_conv_matrix_sparse(f)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    try:
        result = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
        if result.success:
            return result.x
    except Exception:
        pass
    return None


def compute_score(f):
    """Compute the objective value directly (fast, no type checks)."""
    n = len(f)
    conv = np.convolve(f, f)
    M = float(np.max(conv))
    S = float(np.sum(f))
    if S < 0.01:
        return float('inf')
    return float(2 * n * M / (S * S))


def subgradient_direction(f, top_k=3):
    """Compute a subgradient direction to reduce max convolution.
    Uses top-k lags weighted by their excess above the mean."""
    n = len(f)
    conv = np.convolve(f, f)
    S = float(np.sum(f))
    mean_conv = (S * S) / (2 * n - 1)

    # Get top-k lags
    top_indices = np.argsort(conv)[-top_k:]
    # Weights: how much above mean
    weights = np.maximum(conv[top_indices] - mean_conv, 0)
    if np.sum(weights) < 1e-15:
        # All lags are at or below mean, which shouldn't happen
        weights = np.ones(top_k)

    grad = np.zeros(n)
    for k, w in zip(top_indices, weights):
        if w <= 0:
            continue
        j_min = max(0, k - (n - 1))
        j_max = min(k, n - 1)
        for j in range(j_min, j_max + 1):
            grad[j] += w * 2.0 * f[k - j]

    return grad


def fast_subgradient_optimize(f, deadline, lr_init=0.005, momentum_decay=0.85,
                              top_k=5, verbose_prefix="", max_iter=50000):
    """Fast subgradient descent with momentum for the max convolution."""
    f = f.copy().astype(np.float64)
    best_f = f.copy()
    best_score = compute_score(f)
    n = len(f)
    S = float(np.sum(f))
    lr = lr_init
    momentum = np.zeros(n)
    no_improve = 0
    it = 0

    while time.monotonic() < deadline and it < max_iter:
        conv = np.convolve(f, f)
        M = float(np.max(conv))
        if M <= 0:
            break

        grad = subgradient_direction(f, top_k=top_k)
        norm_grad = np.linalg.norm(grad)
        if norm_grad < 1e-15:
            break

        # Normalize gradient to unit step size
        grad = grad / (norm_grad + 1e-15)

        # Momentum update
        momentum = momentum_decay * momentum - lr * grad
        new_f = f + momentum
        new_f = np.clip(new_f, 0.0, None)

        # Renormalize to maintain sum
        new_sum = float(np.sum(new_f))
        if new_sum < 1e-15:
            lr *= 0.5
            no_improve += 1
            if no_improve > 100:
                break
            it += 1
            continue
        new_f = new_f * (S / new_sum)

        new_score = compute_score(new_f)
        it += 1

        if np.isfinite(new_score) and new_score < best_score:
            f = new_f
            best_f = f.copy()
            best_score = new_score
            no_improve = 0
            if it % 1000 == 0:
                print(f"    {verbose_prefix}subgrad iter {it}: score={best_score:.6f} lr={lr:.5f}")
        else:
            no_improve += 1
            lr *= 0.95
            momentum *= 0.5
            if lr < 1e-7 or no_improve > 500:
                break

    return best_f, best_score


def lp_optimize(f, deadline, step_init=0.1, verbose_prefix=""):
    """LP-based optimization with adaptive step size and line search."""
    f = f.copy().astype(np.float64)
    best_f = f.copy()
    best_score = compute_score(f)
    n = len(f)
    S = float(np.sum(f))
    step = step_init
    no_improve = 0
    it = 0

    while time.monotonic() < deadline:
        conv_f = np.convolve(f, f)
        M = float(np.max(conv_f))
        if M <= 0:
            break

        # Solve LP
        t_start = time.monotonic()
        g = solve_lp_direction(f, M)
        if g is None:
            no_improve += 1
            if no_improve > 30:
                break
            continue
        lp_time = time.monotonic() - t_start

        sum_g = float(np.sum(g))
        if sum_g < 1e-15:
            no_improve += 1
            continue

        # Rescale g to match sum of f
        g = g * (S / sum_g)

        # Line search: try several step sizes
        best_step_score = best_score
        best_step_f = None
        steps_to_try = [step, step * 0.5, step * 1.5, step * 0.25]
        for alpha in steps_to_try:
            alpha = min(alpha, 1.0)
            if alpha < 0.001:
                continue
            candidate = (1.0 - alpha) * f + alpha * g
            candidate = np.clip(candidate, 0.0, None)
            cand_sum = float(np.sum(candidate))
            if cand_sum < 1e-15:
                continue
            candidate = candidate * (S / cand_sum)
            cand_score = compute_score(candidate)
            if np.isfinite(cand_score) and cand_score < best_step_score:
                best_step_score = cand_score
                best_step_f = candidate

        it += 1

        if best_step_f is not None:
            f = best_step_f
            best_f = f.copy()
            best_score = best_step_score
            no_improve = 0
            if it % 20 == 0:
                print(f"    {verbose_prefix}LP iter {it}: score={best_score:.6f} "
                      f"(step={step:.3f}, lp_time={lp_time:.2f}s)")
            # Decay step slowly
            step *= 0.995
            step = max(step, 0.01)
        else:
            no_improve += 1
            step *= 0.7
            if step < 0.003 or no_improve > 50:
                break

    return best_f, best_score


def mixed_optimize(f, deadline, verbose_prefix=""):
    """Alternate between LP steps and subgradient steps for robustness."""
    f = f.copy().astype(np.float64)
    best_f = f.copy()
    best_score = compute_score(f)
    n = len(f)
    S = float(np.sum(f))

    # Phase A: Subgradient warmup (first 20% of time)
    sub_deadline = time.monotonic() + (deadline - time.monotonic()) * 0.2
    f, s = fast_subgradient_optimize(f, sub_deadline, lr_init=0.01,
                                      top_k=5, verbose_prefix=verbose_prefix + "sub ")
    if s < best_score:
        best_score = s
        best_f = f.copy()

    # Phase B: LP optimization (next 50% of remaining time)

    lp_deadline = time.monotonic() + remaining * 0.5

                       verbose_prefix=verbose_prefix + "LP ")
    if s < best_score:
        best_score = s
        best_f = f.copy()

    # Phase C: Subgradient polish

                                      top_k=7, verbose_prefix=verbose_prefix + "polish ")
    if s < best_score:
        best_score = s
        best_f = f.copy()

    # Phase D: Final LP polish
    f, s = lp_optimize(f, deadline, step_init=0.03,
                       verbose_prefix=verbose_prefix + "finalLP ")
    if s < best_score:
        best_score = s
        best_f = f.copy()

    return best_f, best_score


def make_initializations(n, rng):
    """Create diverse initializations for length n."""
    inits = []
    idx = np.arange(n, dtype=np.float64)

    # Uniform
    inits.append(('uniform', np.ones(n)))

    # Cosine taper
    cos_t = (1.0 + np.cos(np.pi * idx / (n - 1))) / 2.0
    inits.append(('cosine', cos_t))

    # Linear ramp (decreasing)
    ramp = np.linspace(1.0, 0.1, n)
    inits.append(('ramp', ramp))

    # Linear ramp (increasing)
    inits.append(('ramp_inc', np.linspace(0.1, 1.0, n)))

    # Exponential decay
    exp_dec = np.exp(-2.0 * idx / n)
    inits.append(('exp_decay', exp_dec))

    # Exponential decay (other side)
    exp_dec2 = np.exp(-2.0 * (n - 1 - idx) / n)
    inits.append(('exp_decay2', exp_dec2))

    # Bell (centered)
    center = (n - 1) / 2.0
    bell = np.exp(-0.5 * ((idx - center) / (n / 4.0)) ** 2)
    inits.append(('bell', bell))

    # Bimodal
    two_peak = (np.exp(-0.5 * ((idx - n * 0.25) / (n / 8.0)) ** 2) +
                np.exp(-0.5 * ((idx - n * 0.75) / (n / 8.0)) ** 2))
    inits.append(('bimodal', two_peak))

    # Random
    r = rng.uniform(0.2, 2.0, size=n)
    inits.append(('random', r))

    # S-curve (sigmoid)
    s_curve = 1.0 / (1.0 + np.exp(-4.0 * (idx / (n - 1) - 0.5)))
    inits.append(('sigmoid', s_curve))

    # Quadratic
    quad = 1.0 - (2.0 * idx / (n - 1) - 1.0) ** 2
    quad = np.clip(quad, 0.05, None)
    inits.append(('quadratic', quad))

    return inits
