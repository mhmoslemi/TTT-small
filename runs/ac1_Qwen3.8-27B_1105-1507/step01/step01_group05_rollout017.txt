Multi-phase optimization for minimizing 2*n*max(conv(a,a))/sum(a)^2.
Uses subgradient descent with softmax smoothing as the primary method,
combined with LP-based direction finding and optimal step selection.
"""

import numpy as np
from scipy.optimize import linprog
import time


def normalize_sequence(a):
    """Normalize to sum=1, clip non-negative."""
    a = np.clip(a, 0, None)
    s = a.sum()
    if s < 1e-15:
        return a
    return a / s


def compute_max_conv(a):
    """Compute max of autocorrelation assuming sum(a)=1."""
    conv = np.convolve(a, a)
    return conv.max()


def compute_score(a):
    """Compute the actual score (scale-invariant)."""
    s = a.sum()
    if s < 1e-15:
        return np.inf
    n = len(a)
    conv = np.convolve(a, a)
    return 2.0 * n * conv.max() / (s * s)


def subgradient_step(a, tau):
    """
    Compute one subgradient step using softmax smoothing.
    Returns (gradient, conv, softmax_weights).
    """
    n = len(a)
    conv = np.convolve(a, a)  # length 2n-1
    # Softmax weights (avoid overflow)
    conv_max = conv.max()
    shifted = (conv - conv_max) / tau
    # Clip to avoid underflow issues
    shifted = np.clip(shifted, -50, 0)
    p = np.exp(shifted)
    p_sum = p.sum()
    if p_sum < 1e-30:
        # All mass on the argmax
        p = np.zeros(2 * n - 1)
        p[np.argmax(conv)] = 1.0
    else:
        p /= p_sum
    # Gradient: g[i] = 2 * sum_j p[i+j] * a[j]
    # This is 2 * correlate(p, a, 'valid') since len(p)=2n-1, len(a)=n, valid length = n
    g = 2.0 * np.correlate(p, a, mode='valid')
    return g, conv, p


def subgradient_optimize(a0, n_iter, tau_init, tau_final, eta_init,
                         decay_period, decay_factor, time_limit=None,
                         start_time=None, verbose=False, log_interval=50000):
    """
    Subgradient descent with softmax annealing and restart-to-best.
    Minimizes max(conv(a,a)) subject to sum(a)=1, a>=0.
    """
    if start_time is None:
        start_time = time.monotonic()
    if time_limit is None:
        time_limit = np.inf

    n = len(a0)
    a = normalize_sequence(a0.copy())
    if a.sum() < 1e-15:
        return a0.copy(), np.inf

    best_a = a.copy()
    best_val = compute_max_conv(a)

    eta = eta_init
    tau = tau_init

    for it in range(n_iter):
        # Check time
        if time.monotonic() - start_time > time_limit:
            break

        # Anneal tau (exponential)
        if n_iter > 1:
            progress = it / (n_iter - 1)
        else:
            progress = 1.0
        tau = tau_init * (tau_final / tau_init) ** progress
        tau = max(tau, best_val * 1e-6)  # Don't let tau get too small

        # Subgradient step
        g, conv, p = subgradient_step(a, tau)

        # Update with projection
        a_new = a - eta * g
        a_new = np.clip(a_new, 0, None)
        s_new = a_new.sum()
        if s_new < 1e-15:
            # Reset to best
            a = best_a.copy()
            eta *= 0.5
            continue

            best_a = a_new.copy()
            a = a_new

def build_conv_matrix(f):
    """Build matrix A such that A @ b = np.convolve(f, b, mode='full')."""
    n = len(f)
    A = np.zeros((2 * n - 1, n))
    for k in range(2 * n - 1):
        j_start = max(0, k - n + 1)
        j_end = min(n, k + 1)
        for j in range(j_start, j_end):
            A[k, j] = f[k - j]
    return A


def lp_direction(f, M):
    """
    Solve LP to find direction g maximizing sum(g)
    subject to conv(f,g) <= M, g >= 0.
    Returns g normalized to sum=1, or None.
    """
    n = len(f)
    A = build_conv_matrix(f)
    c_obj = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n

    try:
        res = linprog(c_obj, A_ub=A, b_ub=b_ub, bounds=bounds,
                      method='highs', options={'presolve': True, 'time_limit': 60})
    except Exception:
        return None

    if not res.success or res.x is None:
        return None

    g = np.clip(res.x, 0, None)
    s = g.sum()
    if s < 1e-15:
        return None
    return g / s


def optimal_step_size(f, g, M):
    """
    Find optimal step size t for blend (1-t)f + tg.
    Uses the formula t* = (M - conv(f,g)_{k*}) / conv(g,g)_{k*}
    where k* is the argmax of conv(f,f).
    """
    conv_fg = np.convolve(f, g)
    conv_gg = np.convolve(g, g)
    conv_ff = np.convolve(f, f)

    k_star = np.argmax(conv_ff)

    cvfg_k = conv_fg[k_star]
    cvgg_k = conv_gg[k_star]

    if cvgg_k <= 0:
        return 0.1

    # t* = (M - cvfg_k) / cvgg_k, clamped to [0, 1]
    t_star = (M - cvfg_k) / cvgg_k
    t_star = np.clip(t_star, 0.01, 1.0)

    # Also try a few nearby values and pick the best
    candidates = [t_star, t_star * 0.5, t_star * 1.5, 0.1, 0.3, 0.5]
    best_t = t_star
    best_val = np.inf

    for t in candidates:
        t = np.clip(t, 0.001, 1.0)
        blend = (1 - t) * f + t * g
        blend = np.clip(blend, 0, None)
        s = blend.sum()
        if s < 1e-15:
            continue
        blend /= s
        val = compute_max_conv(blend)
        if val < best_val:
            best_val = val
            best_t = t

    return best_t


def lp_optimize(f, n_iterations, time_limit, start_time=None, verbose=False):
    """
    LP-based iterative optimization with optimal step sizes.
    """
    if start_time is None:
        start_time = time.monotonic()
    if time_limit is None:
        time_limit = np.inf

    n = len(f)
    f = normalize_sequence(f.copy())
    best_f = f.copy()
    best_score = 2.0 * n * compute_max_conv(f)
    t0 = 0.5

    for it in range(n_iterations):
        if time.monotonic() - start_time > time_limit:
            break

        conv_ff = np.convolve(f, f)
        M = conv_ff.max()

        g = lp_direction(f, M)
        if g is None:
            break

        # Find optimal step
        t = optimal_step_size(f, g, M)

        # Blend
        f_new = (1 - t) * f + t * g
        f_new = np.clip(f_new, 0, None)
        s = f_new.sum()
        if s < 1e-15:
            t0 *= 0.5
            continue
        f_new /= s

        new_score = 2.0 * n * compute_max_conv(f_new)

        if new_score < best_score:
            best_score = new_score
            best_f = f_new.copy()
            f = f_new
            if verbose and it % 5 == 0:
                elapsed = time.monotonic() - start_time
                print(f"      lp: it={it}, score={best_score:.6f}, t={t:.4f}, elapsed={elapsed:.1f}s")
        else:
            t0 *= 0.7
            if t0 < 0.01:
                # Try perturbation
                break

    return best_f, best_score


def local_search_polish(a, n_iter, rng, noise_scale=0.05):
    """
    Simple local search with random perturbations for final polish.
    """
    n = len(a)
    a = normalize_sequence(a.copy())
    best_a = a.copy()
    best_score = compute_max_conv(a)

    for it in range(n_iter):
        # Random block perturbation
        width = max(1, int(rng.integers(1, max(2, n // 16))))
        start = int(rng.integers(0, n - width))
        scale = np.exp(rng.normal(0, noise_scale, width))
        a_new = a.copy()
        a_new[start:start + width] *= scale
        a_new = normalize_sequence(a_new)
        if a_new.sum() < 1e-15:
            continue

        val = compute_max_conv(a_new)
        if val < best_score:
            best_score = val
            best_a = a_new.copy()
            a = a_new

    return best_a, 2.0 * n * best_score


def propose_candidate(seed=42, budget_s=1500, **kwargs):
    """
    Main search function combining subgradient, LP, and local search.
    """
    rng = np.random.default_rng(seed)
    start = time.monotonic()
    deadline = start + budget_s - 20.0  # 20s buffer

    best_overall = None
    best_score_overall = np.inf

    # Try to get parent sequence
    parent = None
    try:
        p = np.array(height_sequence_1, dtype=np.float64)
        if p.ndim == 1 and len(p) > 0 and np.isfinite(p).all() and p.min() >= 0:
            parent = p
    except Exception:
        pass

    if parent is not None:
        print(f"Parent sequence available, length={len(parent)}, score={compute_score(parent):.6f}")
    else:
        print("No parent sequence available.")

    # Define configurations
    # (name, n, init_type, priority)
    configs = []

    if parent is not None:
        n_p = len(parent)
        if 64 <= n_p <= 4096:
            configs.append((f"parent_{n_p}", n_p, 'parent', 1))

    for n in [256, 512, 1024, 2048]:
        configs.append((f"uniform_{n}", n, 'uniform', 2))
        configs.append((f"random_{n}", n, 'random', 2))

    if parent is not None:
        # Also try extending parent to larger sizes
        n_p = len(parent)
        for n_ext in [2 * n_p, 4 * n_p]:
            if n_ext <= 4096:
                configs.append((f"ext_parent_{n_ext}", n_ext, 'extend_parent', 3))

    # Sort by priority, then by n
    configs.sort(key=lambda x: (x[3], x[1]))

    print(f"Total configs: {len(configs)}")
    print(f"Configs: {[c[0] for c in configs]}")
    print(f"Budget: {budget_s}s, Deadline in: {deadline - time.monotonic():.0f}s")

    # Time allocation: give more time to promising configs
    # Parent config gets most time, then smaller n, then larger n
    total_time = deadline - time.monotonic() - 60.0  # 60s reserved for final polish
    if total_time < 30:
        total_time = max(30, deadline - time.monotonic())

    # Allocate time: priority 1 gets 40%, priority 2 gets 30%, priority 3 gets remaining
    time_p1 = total_time * 0.40
    time_p2 = total_time * 0.35
    time_p3 = total_time * 0.15
    # Each priority group shares its time among its configs
    # But we process sequentially and use remaining time adaptively

    for cfg_idx, (name, n, init_type, priority) in enumerate(configs):
        if time.monotonic() > deadline:
            break

        remaining = deadline - time.monotonic()
        if remaining < 30:
            break

        # Time for this config: adaptive
        configs_remaining = len(configs) - cfg_idx
        time_this = min(remaining * 0.4, max(20, total_time / max(1, configs_remaining) * 1.5))
        if remaining < 60:
            time_this = remaining - 5
        if time_this < 10:
            break

        print(f"\n[{name}] n={n}, time={time_this:.0f}s, remaining={remaining:.0f}s")

        # Create initial sequence
        if init_type == 'parent' and parent is not None:
            a0 = parent.copy().astype(np.float64)
        elif init_type == 'extend_parent' and parent is not None:
            # Tile or interpolate parent to length n
            n_p = len(parent)
            if n_p >= n:
                a0 = parent[:n].copy()
            else:
                # Repeat parent to fill n
                reps = int(np.ceil(n / n_p))
                a0 = np.tile(parent, reps)[:n]
        elif init_type == 'uniform':
            a0 = np.ones(n)
            # Small perturbation to break symmetry
            a0 = a0 * (1.0 + 0.1 * rng.standard_normal(n))
            a0 = np.clip(a0, 0.01, None)
        elif init_type == 'random':
            # Exponential random (gives a natural non-negative distribution)
            a0 = rng.exponential(1.0, n)
            a0 = np.clip(a0, 0.001, None)
        else:
            a0 = np.ones(n)

        a0 = normalize_sequence(a0)
        initial_score = 2.0 * n * compute_max_conv(a0)
        print(f"  Initial score: {initial_score:.6f}, max_conv={compute_max_conv(a0):.8f}")

        if initial_score >= best_score_overall and best_score_overall < np.inf:
            # Skip if already worse (but still try a bit)
            time_this = min(time_this, 15)

        # Phase 1: Subgradient optimization
        sg_time = time_this * 0.6
        sg_start = time.monotonic()

        # Determine parameters based on n
        if n <= 512:
            n_iter_max = 500000
            tau_init = compute_max_conv(a0) * 0.5
            tau_final = compute_max_conv(a0) * 0.01
            eta_init = 0.05
            decay_period = 5000
            decay_factor = 0.9
        elif n <= 1024:
            n_iter_max = 200000
            tau_init = compute_max_conv(a0) * 0.5
            tau_final = compute_max_conv(a0) * 0.01
            eta_init = 0.05
            decay_period = 5000
            decay_factor = 0.9
        else:
            n_iter_max = 100000
            tau_init = compute_max_conv(a0) * 0.5
            tau_final = compute_max_conv(a0) * 0.01
            eta_init = 0.05
            decay_period = 5000
            decay_factor = 0.9

        a_sg, score_sg = subgradient_optimize(
            a0, n_iter_max, tau_init, tau_final, eta_init,
            decay_period, decay_factor,
            time_limit=sg_time, start_time=sg_start,
            verbose=True, log_interval=50000
        )
        print(f"  After subgradient: score={score_sg:.6f}")

        # Phase 2: LP refinement (if time permits)
        elapsed = time.monotonic() - (start)
        remaining_after_sg = deadline - time.monotonic()
        if remaining_after_sg > 20 and np.isfinite(score_sg):
            lp_time = min(remaining_after_sg * 0.3, time_this * 0.3)
            if lp_time > 10:
                a_lp, score_lp = lp_optimize(
                    a_sg, n_iterations=100,
                    time_limit=lp_time,
                    start_time=time.monotonic(),
                    verbose=True
                )
                print(f"  After LP: score={score_lp:.6f}")
                if score_lp < score_sg:
                    a_sg = a_lp
                    score_sg = score_lp

        # Phase 3: More subgradient with finer parameters (if time)
        remaining_after_lp = deadline - time.monotonic()
        if remaining_after_lp > 20 and np.isfinite(score_sg):
            sg2_time = min(remaining_after_lp * 0.4, time_this * 0.2)
            if sg2_time > 5:
                current_max = compute_max_conv(a_sg)
                a_sg2, score_sg2 = subgradient_optimize(
                    a_sg, n_iter_max,
                    tau_init=current_max * 0.1,
                    tau_final=current_max * 0.001,
                    eta_init=0.02,
                    decay_period=3000,
                    decay_factor=0.92,
                    time_limit=sg2_time,
                    start_time=time.monotonic(),
                    verbose=False
                )
                if score_sg2 < score_sg:
                    a_sg = a_sg2
                    score_sg = score_sg2
                    print(f"  After fine subgradient: score={score_sg:.6f}")

        # Phase 4: Local search polish
        remaining_final = deadline - time.monotonic()
        if remaining_final > 10 and np.isfinite(score_sg):
            ls_iters = min(20000, int(remaining_final * 1000))
            a_ls, score_ls = local_search_polish(a_sg, ls_iters, rng)
            if score_ls < score_sg:
                a_sg = a_ls
                score_sg = score_ls
                print(f"  After local search: score={score_sg:.6f}")

        # Update best
        if np.isfinite(score_sg) and score_sg < best_score_overall:
            best_score_overall = score_sg
            best_overall = a_sg.copy()
            print(f"  ** New best: {score_sg:.6f} (n={n}) **")

    # Final phase: intensive optimization on the best sequence found
    remaining = deadline - time.monotonic()
    print(f"\n=== Final phase: {remaining:.0f}s remaining ===")

    if best_overall is not None and remaining > 20:
        n_best = len(best_overall)
        current_max = compute_max_conv(best_overall)

        # Multiple rounds of subgradient with different parameters
        a_current = best_overall.copy()
        score_current = 2.0 * n_best * current_max

        # Round 1: Very fine tau
        r1_time = min(remaining * 0.3, 200)
        a_r1, score_r1 = subgradient_optimize(
            a_current, 500000,
            tau_init=current_max * 0.05,
            tau_final=current_max * 0.0005,
            eta_init=0.01,
            decay_period=3000,
            decay_factor=0.93,
            time_limit=r1_time,
            start_time=time.monotonic(),
            verbose=True, log_interval=50000
        )
        if score_r1 < score_current:
            score_current = score_r1
            a_current = a_r1
            print(f"  Final R1: {score_current:.6f}")

        # Round 2: Different eta
        remaining2 = deadline - time.monotonic()
        if remaining2 > 20:
            r2_time = min(remaining2 * 0.3, 150)
            a_r2, score_r2 = subgradient_optimize(
                a_current, 500000,
                tau_init=compute_max_conv(a_current) * 0.02,
                tau_final=compute_max_conv(a_current) * 0.0002,
                eta_init=0.005,
                decay_period=2000,
                decay_factor=0.95,
                time_limit=r2_time,
                start_time=time.monotonic(),
                verbose=False
            )
            if score_r2 < score_current:
                score_current = score_r2
                a_current = a_r2
                print(f"  Final R2: {score_current:.6f}")

        # Round 3: LP polish
        remaining3 = deadline - time.monotonic()
        if remaining3 > 15 and n_best <= 2048:
            a_lp, score_lp = lp_optimize(
                a_current, n_iterations=50,
                time_limit=min(remaining3 * 0.4, 100),
                start_time=time.monotonic(),
                verbose=True
            )
            if score_lp < score_current:
                score_current = score_lp
                a_current = a_lp
                print(f"  Final LP: {score_current:.6f}")

        # Final local search
        remaining4 = deadline - time.monotonic()
        if remaining4 > 10:
            ls_iters = min(50000, int(remaining4 * 500))
            a_ls, score_ls = local_search_polish(a_current, ls_iters, rng, noise_scale=0.02)
            if score_ls < score_current:
                score_current = score_ls
                a_current = a_ls
                print(f"  Final LS: {score_current:.6f}")

        if score_current < best_score_overall:
            best_score_overall = score_current
            best_overall = a_current.copy()

    # Verify and return
    if best_overall is not None:
        best_overall = np.clip(best_overall, 0, 1000)
        # Ensure sum is well above 0.01
        s = best_overall.sum()
        if s < 1.0:
            best_overall = best_overall * (1.0 / max(s, 0.01))
        final_list = best_overall.tolist()
        final_score = evaluate_sequence(final_list)
        print(f"\n=== FINAL ===")
        print(f"Score: {final_score:.8f}")
        print(f"Length: {len(final_list)}")
        print(f"Target: 1.5027")
        print(f"Previous best: 1.515592")
        print(f"Target met: {final_score < 1.5027}")
        return final_list
    else:
        # Fallback
        print("FALLBACK to uniform 256")
        return [1.0] * 256
