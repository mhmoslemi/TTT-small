Let me think about the line search more carefully. We want to minimize:
f(t) = score(normalize((1-t)*a + t*g))

where a and g are both normalized to sum 1.

normalize((1-t)*a + t*g) = ((1-t)*a + t*g) / (sum((1-t)*a + t*g)) = ((1-t)*a + t*g) / 1 = (1-t)*a + t*g

Since both a and g have sum 1, their convex combination also has sum 1! So no re-normalization needed.

So f(t) = 2*n * max(((1-t)*a + t*g) * ((1-t)*a + t*g)) / 1

The convolution of (1-t)*a + t*g with itself:
= (1-t)^2 * (a*a) + 2t(1-t) * (a*g) + t^2 * (g*g)

So f(t) = 2*n * max_k [(1-t)^2 * (a*a)(k) + 2t(1-t) * (a*g)(k) + t^2 * (g*g)(k)]
= 2*n * max_k [t^2 * (g*g)(k) + 2t(1-t) * (a*g)(k) + (1-t)^2 * (a*a)(k)]

Each term in the max is a quadratic in t (a parabola opening upward since (g*g)(k) >= 0). The max of convex functions is convex. So f(t) is convex in t!

This means we can use a simple approach: since f(t) is convex on [0,1], the minimum is at t=0 or at a point where the derivative changes sign. We can do a ternary search or just sample a fine grid.

Actually, since f(0) = 2*n*M (the current score) and f(1) = 2*n*max(g*g) (the score of g alone), and f is convex, the minimum is somewhere in [0,1]. If f(1) < f(0), then the minimum is in (0,1]. We can do a golden section search.

But actually, for small t, f(t) ≈ f(0) + t * f'(0) where f'(0) = 2*n * max_k [2*(a*g)(k) - 2*(a*a)(k)] = 4*n * max_k [(a*g)(k) - (a*a)(k)].

If (a*g)(k) < (a*a)(k) for all k (which is likely since the LP ensures (a*g)(k) <= M = max(a*a), and for most k, (a*a)(k) < M), then f'(0) < 0, meaning the function is decreasing at t=0. So the minimum is at t > 0.

```python
import numpy as np
from scipy.optimize import linprog

def normalize(a):

    s = a.sum()
    if s < 1e-300:
        return None
    return a / s

def score(a):
    a = np.asarray(a, dtype=np.float64)
    n = len(a)
    if n == 0:
        return np.inf
    c = np.convolve(a, a)
    m = c.max()
    s = a.sum()
    if s < 1e-12:
        return np.inf
    return 2.0 * n * m / (s * s)

def solve_lp(a, M, seed=0):
    """Solve LP: max sum(g) s.t. (a*g)(k) <= M for all k, g >= 0."""
    n = len(a)
    # Build constraint matrix A where A[k, i] = a[k-i] (valid range)
    # Using the structure of the convolution matrix
    A = np.zeros((2*n-1, n), dtype=np.float64)
    for k in range(2*n-1):
        i_start = max(0, k - n + 1)
        i_end = min(n - 1, k)
        for i in range(i_start, i_end + 1):

    res = linprog(c_vec, A_ub=A, b_ub=b_vec, bounds=bounds, method='highs')
    if not res.success:
        return None
    g = np.clip(res.x, 0, None)
    g = normalize(g)
    return g
```

```python
import time
import numpy as np
from scipy.optimize import linprog

def normalize_sum1(a):
    """Normalize a to sum 1, clip to non-negative."""
    a = np.asarray(a, dtype=np.float64)
    a = np.clip(a, 0.0, None)
    s = a.sum()
    if s < 1e-300 or not np.isfinite(s):
        return None
    return a / s

def compute_score(a):
    """Compute the objective score."""
    a = np.asarray(a, dtype=np.float64)
    n = len(a)
    if n == 0:
        return np.inf
    s = a.sum()
    if s < 1e-12 or not np.isfinite(s):
        return np.inf
    c = np.convolve(a, a)
    m = c.max()
    if not np.isfinite(m):
        return np.inf
    return float(2.0 * n * m / (s * s))

def build_A(a):
    """Build the convolution constraint matrix for the LP."""
    n = len(a)
    A = np.zeros((2 * n - 1, n), dtype=np.float64)
    for k in range(2 * n - 1):
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        if lo <= hi:
            # A[k, i] = a[k - i] for i in [lo, hi]
            # k - i goes from k - lo down to k - hi
            A[k, lo:hi+1] = a[k - hi : k - lo + 1][::-1]
    return A

def lp_step(a, M, seed=0):
    """
    Solve LP: maximize sum(g) subject to (a*g)(k) <= M for all k, g >= 0.
    Returns normalized g or None.
    """
    n = len(a)
    A = build_A(a)
    c = -np.ones(n, dtype=np.float64)
    b = np.full(2 * n - 1, M * (1 + 1e-10), dtype=np.float64)
    bounds = [(0.0, None)] * n
    
    try:
        res = linprog(c, A_ub=A, b_ub=b, bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': 30})
    except Exception:
        return None
    
    if not res.success:
        return None
    
    g = np.asarray(res.x, dtype=np.float64)
    g = np.clip(g, 0.0, None)
    g = normalize_sum1(g)
    return g

def line_search(a, g, max_evals=30):
    """
    Find t in [0,1] that minimizes score of (1-t)*a + t*g.
    Since both a and g have sum 1, the combination also has sum 1,
    so no renormalization needed.
    """
    n = len(a)
    c_a = np.convolve(a, a)
    c_ag = np.convolve(a, g)
    c_g = np.convolve(g, g)
    
    # f(t) = 2n * max_k [(1-t)^2 c_a(k) + 2t(1-t) c_ag(k) + t^2 c_g(k)]
    # Each bracket is a quadratic in t with positive leading coefficient (c_g(k) >= 0)
    # So f(t) is convex.
    
    def f(t):
        val = (1-t)**2 * c_a + 2*t*(1-t) * c_ag + t**2 * c_g
        return 2.0 * n * val.max()
    
    # Golden section search on [0, 1]
    invphi = 2.0 / (1.0 + np.sqrt(5.0))
    a_lo = 0.0
    a_hi = 1.0
    c_pt = a_hi - invphi * (a_hi - a_lo)
    d_pt = a_lo + invphi * (a_hi - a_lo)
    f_c = f(c_pt)
    f_d = f(d_pt)
    
    for _ in range(max_evals):
        if f_c < f_d:
            a_hi = d_pt
            d_pt = c_pt
            f_d = f_c
            c_pt = a_hi - invphi * (a_hi - a_lo)
            f_c = f(c_pt)
        else:
            a_lo = c_pt
            c_pt = d_pt
            f_c = f_d
            d_pt = a_lo + invphi * (a_hi - a_lo)
            f_d = f(d_pt)
    
    t_opt = (a_lo + a_hi) / 2.0
    return t_opt

def subgradient_step(a, eta=0.3, K=None):
    """
    Multiplicative subgradient step to reduce the peak of a*a.
    """
    n = len(a)
    c = np.convolve(a, a)
    M = c.max()
    if M <= 0:
        return a.copy()
    
    if K is None:
        K = max(10, n // 10)
    
    # Find active peaks
    threshold = M * 0.95
    active = np.nonzero(c >= threshold)[0]
    if len(active) == 0:
        active = np.array([np.argmax(c)])
    
    # Compute the "gradient" direction
    h = np.zeros(n, dtype=np.float64)
    for k in active:
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        for i in range(lo, hi + 1):
            j = k - i
            h[i] += a[j]
            if i != j:
                h[j] += a[i]
    
    # Scale
    h = h / max(1.0, h.max())
    
    # Multiplicative update: decrease components with high h
    fac = eta * (h.mean() - h)
    fac = np.clip(fac, -2.0, 2.0)
    a_new = a * np.exp(fac)
    a_new = np.clip(a_new, 1e-300, None)
    a_new = normalize_sum1(a_new)
    return a_new if a_new is not None else a.copy()

def lp_optimize(a, max_time, seed, polynomial_steps=5):
    """
    Main LP-based optimization loop.
    """
    a = normalize_sum1(a)
    if a is None:
        return None, np.inf
    n = len(a)
    
    best = a.copy()
    best_score = compute_score(best)
    t0 = time.monotonic()
    deadline = t0 + max_time
    
    it = 0
    stall = 0
    eta = 0.25
    
    while time.monotonic() < deadline - 2.0:
        # LP step
        c = np.convolve(a, a)
        M = c.max()
        g = lp_step(a, M, seed + it)
        
        if g is not None:
            # Line search
            t_opt = line_search(a, g, max_evals=25)
            a_new = (1 - t_opt) * a + t_opt * g
            a_new = normalize_sum1(a_new)
            
            if a_new is not None:
                sc = compute_score(a_new)
                if sc < best_score:
                    best = a_new.copy()
                    best_score = sc
                    stall = 0
                elif sc < compute_score(a):
                    a = a_new.copy()
                else:
                    a = a_new.copy()  # Still move, might help escape
                    stall += 1
            else:
                stall += 1
        else:
            stall += 1
        
        # Subgradient polish
        if stall > 3 or it % polynomial_steps == 0:
            for _ in range(polynomial_steps):
                a = subgradient_step(a, eta)
                sc = compute_score(a)
                if sc < best_score:
                    best = a.copy()
                    best_score = sc
                    stall = 0
                if time.monotonic() >= deadline - 1.0:
                    break
        
        # Adaptive step size
        if stall > 10:
            eta *= 0.7
            if eta < 0.05:
                # Restart

                a = best * noise
                a = normalize_sum1(a)
                eta = 0.3

        it += 1
    
    return best, best_score
```

```python
def build_A_fast(a):
    n = len(a)
    # The (k,i) entry is a[k-i] if 0 <= k-i < n and 0 <= i < n
    # i.e., k-n+1 <= i <= k, and 0 <= i <= n-1
    # So i ranges from max(0, k-n+1) to min(n-1, k)
    
    # We can construct this using the "toeplitz-like" structure
    # Actually, this is a convolution matrix. Each row k is a "cross-correlation" of a with a delta.
    
    # Simple approach: use the fact that A[k, i] = a[k-i] for valid (k,i)
    # Create index arrays
    k_idx = np.arange(2*n-1)[:, None]  # (2n-1, 1)
    i_idx = np.arange(n)[None, :]      # (1, n)
    diff = k_idx - i_idx               # (2n-1, n)
    valid = (diff >= 0) & (diff < n)
    
    A = np.zeros((2*n-1, n), dtype=np.float64)
    A[valid] = a[diff[valid]]
    return A
```

One more thought: the line search assumes both a and g have sum 1. After the LP, g is normalized to sum 1. And a is maintained at sum 1 throughout. So the convex combination (1-t)*a + t*g also has sum 1. Good.

But wait: is the score convex in t? Let me verify.

f(t) = 2n * max_k [(1-t)^2 * c_a(k) + 2t(1-t) * c_ag(k) + t^2 * c_g(k)]

Each term inside the max is a quadratic in t:
q_k(t) = c_g(k)*t^2 + 2(c_ag(k) - c_a(k))*t + c_a(k)

The leading coefficient is c_g(k) >= 0 (since c_g(k) = sum g_i * g_{k-i} >= 0 for non-negative g). So each q_k is convex (or linear if c_g(k) = 0).

The max of convex functions is convex. So f(t) is convex. Golden section search is valid.

```python
"""
LP-based iterative optimization for minimizing the autocorrelation peak ratio.
Combines repeated LP steps with golden-section line search, multiplicative
subgradient polishing, and multi-length exploration.

Core idea: Given sequence a (sum=1), solve LP to find g maximizing sum(g)
subject to (a*g)(k) <= M for all k. Blend with line search, iterate.
The convexity of the blend in t enables efficient golden-section search.
"""

import time
import numpy as np
from scipy.optimize import linprog


def normalize_sum1(a):
    a = np.asarray(a, dtype=np.float64)
    a = np.clip(a, 0.0, None)
    s = float(a.sum())
    if s < 1e-300 or not np.isfinite(s):
        return None
    return a / s


def compute_score(a):
    a = np.asarray(a, dtype=np.float64)
    n = len(a)
    if n == 0:
        return float('inf')
    s = float(a.sum())
    if s < 1e-12 or not np.isfinite(s):
        return float('inf')
    c = np.convolve(a, a)
    m = float(c.max())
    if not np.isfinite(m):
        return float('inf')
    return float(2.0 * n * m / (s * s))


def build_conv_matrix(a):
    n = len(a)
    k_idx = np.arange(2 * n - 1)[:, None].astype(np.int64)
    i_idx = np.arange(n)[None, :].astype(np.int64)
    diff = k_idx - i_idx
    valid = (diff >= 0) & (diff < n)
    A = np.zeros((2 * n - 1, n), dtype=np.float64)
    A[valid] = a[diff[valid]]
    return A


def lp_step(a, M, time_limit=20.0):
    n = len(a)
    A = build_conv_matrix(a)
    c = -np.ones(n, dtype=np.float64)
    b = np.full(2 * n - 1, M * (1.0 + 1e-9), dtype=np.float64)
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c, A_ub=A, b_ub=b, bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': time_limit})
    except Exception:
        return None
    if not res.success:
        return None
    g = np.asarray(res.x, dtype=np.float64)
    g = np.clip(g, 0.0, None)
    g = normalize_sum1(g)
    if g is None or float(g.sum()) < 1e-15:
        return None
    return g


def golden_line_search(c_a, c_ag, c_g, n, max_iter=40):
    invphi = 2.0 / (1.0 + np.sqrt(5.0))
    lo = 0.0
    hi = 1.0
    c_pt = hi - invphi * (hi - lo)
    d_pt = lo + invphi * (hi - lo)

    def f(t):
        val = (1.0 - t) * (1.0 - t) * c_a + 2.0 * t * (1.0 - t) * c_ag + t * t * c_g
        return float(2.0 * n * val.max())

    f_c = f(c_pt)
    f_d = f(d_pt)
    for _ in range(max_iter):
        if f_c < f_d:
            hi = d_pt
            d_pt = c_pt
            f_d = f_c
            c_pt = hi - invphi * (hi - lo)
            f_c = f(c_pt)
        else:
            lo = c_pt
            c_pt = d_pt
            f_c = f_d
            d_pt = lo + invphi * (hi - lo)
            f_d = f(d_pt)
    t_opt = 0.5 * (lo + hi)
    f_opt = f(t_opt)
    return t_opt, f_opt


def subgradient_step(a, eta=0.3, frac_active=0.1):
    n = len(a)
    c = np.convolve(a, a)
    M = float(c.max())
    if M <= 0 or not np.isfinite(M):
        return a.copy()
    K = max(5, int(n * frac_active))
    active = np.argpartition(c, -K)[-K:]
    w = c[active] / M
    h = np.zeros(n, dtype=np.float64)
    for idx in range(len(active)):
        k = int(active[idx])
        wk = float(w[idx])
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        if lo > hi:
            continue
        i = np.arange(lo, hi + 1)
        j = k - i
        h[i] += wk * a[j]
        h[j] += wk * a[i]
    h_max = float(h.max())
    if h_max <= 0 or not np.isfinite(h_max):
        return a.copy()
    h_norm = h / h_max
    h_mean = float(h_norm.mean())
    fac = eta * (h_mean - h_norm)
    fac = np.clip(fac, -3.0, 3.0)
    a_new = a * np.exp(fac)
    a_new = normalize_sum1(a_new)
    if a_new is None:
        return a.copy()
    return a_new


def lp_optimize(a, max_time, seed):
    a = normalize_sum1(a)
    if a is None:
        return None, float('inf')
    n = len(a)
    best = a.copy()
    best_score = compute_score(best)
    t0 = time.monotonic()
    deadline = t0 + max_time
    rng = np.random.default_rng(seed)
    stall = 0
    it = 0
    eta = 0.3

    while time.monotonic() < deadline - 3.0:
        c_a = np.convolve(a, a)
        M = float(c_a.max())
        if not np.isfinite(M) or M <= 0:
            break

        rem = deadline - time.monotonic()
        tl = min(15.0, max(2.0, rem * 0.1))
        g = lp_step(a, M, time_limit=tl)

        if g is not None:
            c_ag = np.convolve(a, g)
            c_g = np.convolve(g, g)
            t_opt, f_val = golden_line_search(c_a, c_ag, c_g, n, max_iter=35)

            if f_val < best_score * 1.001:
                a_new = (1.0 - t_opt) * a + t_opt * g
                a_new = normalize_sum1(a_new)
                if a_new is not None:
                    sc = compute_score(a_new)
                    if sc < best_score:
                        best = a_new.copy()
                        best_score = float(sc)
                        a = a_new.copy()
                        stall = 0
                    else:
                        a = a_new.copy()
                        stall += 1
                else:
                    stall += 1
            else:
                stall += 1
        else:
            stall += 1

                a = best.copy()
            eta = 0.35

def resample_to_length(a, new_len):
    if a is None or len(a) == 0:
        return None
    a = np.asarray(a, dtype=np.float64)
    old_len = len(a)
    if old_len == new_len:
        return a.copy()
    old_x = np.linspace(0.0, 1.0, old_len)
    new_x = np.linspace(0.0, 1.0, new_len)
    a_new = np.interp(new_x, old_x, a)
    a_new = np.clip(a_new, 0.0, None)
    return normalize_sum1(a_new)


def make_random_init(n, seed, kind=0):
    rng = np.random.default_rng(seed)
    if kind == 0:
        a = rng.uniform(0.1, 1.0, n)
    elif kind == 1:
        a = rng.beta(2.0, 2.0, n)
    elif kind == 2:
        x = rng.uniform(-1.0, 1.0, n)
        a = 0.5 * (1.0 + 0.7 * x)
    elif kind == 3:
        w = int(rng.integers(3, max(4, n // 8)))
        kern = np.exp(-0.5 * (np.arange(w) - w // 2) ** 2 / (w // 4 + 1))
        raw = rng.normal(0, 1, n)
        a = np.convolve(raw, kern, mode='same')
        a = np.clip(a, 0.01, None)
    else:
        nb = max(4, n // 20)
        blocks = rng.uniform(0.2, 1.0, nb)
        a = np.repeat(blocks, n // nb + 1)[:n]
        a = 0.6 * a + 0.4 * np.convolve(a, np.array([0.25, 0.5, 0.25]), mode='same')
        a = np.clip(a, 0.01, None)
    return normalize_sum1(a)
```
