"""
Optimizes non-negative step function to minimize 2n*max(conv(f,f))/(sum(f))^2.
Uses multi-phase approach: mirror descent for initial optimization,
active-set LP Frank-Wolfe for refinement.
"""

import numpy as np
import time
from scipy.optimize import linprog
from scipy.sparse import csr_matrix

def _fft_convolve(f):
    """Full linear convolution using FFT."""
    n = len(f)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    F = np.fft.rfft(f, size)
    h = np.fft.irfft(F * F, size)
    return h[:2 * n - 1]

def _score(f):
    """Compute score assuming f >= 0, sum(f) > 0."""
    n = len(f)
    h = np.convolve(f, f)
    M = float(np.max(h))
    S = float(np.sum(f))
    if S < 0.01:
        return float('inf')
    return 2.0 * n * M / (S * S)

def _normalize(f):
    f = np.maximum(f, 0.0)
    s = float(np.sum(f))
    if s < 1e-30:
        return None
    return f / s

def _pressure(f, active_lags):
    """Compute subgradient pressure for active lags."""
    n = len(f)
    p = np.zeros(n)
    for k in active_lags:
        # (f*f)[k] = sum_m f[m]*f[k-m]
        # d/dx[j] (f*f)[k] = f[k-j] if 0<=k-j<n, times 2 if j != k-j, times 1 if j == k-j
        # Simplified: p[j] += 2*f[k-j] for valid k-j (handles the factor of 2 for j != k-j)
        # For j == k-j (i.e., 2j == k), the factor should be 1, not 2.
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        if j_end >= j_start:
            # For j in [j_start, j_end], k-j is in [0, n-1]
            # p[j] += 2 * f[k-j]
            # But if 2j == k, p[j] should only get f[k-j] not 2*f[k-j]
            indices = np.arange(j_start, j_end + 1)
            p[indices] += 2.0 * f[k - indices]
            # Fix the diagonal: if 2j == k for some j in range, subtract f[k-j]
            if k % 2 == 0:
                j_diag = k // 2
                if j_start <= j_diag <= j_end:
                    p[j_diag] -= f[j_diag]  # because we added 2*f but should have added f
    p /= float(len(active_lags))
    return p

def _mirror_descent(f0, n_iter, eta_scale, threshold, time_limit=60.0):
    """Mirror descent with multiplicative updates."""
    f = _normalize(np.asarray(f0, dtype=np.float64))
    if f is None:
        return np.asarray(f0, dtype=np.float64).copy(), float('inf')
    n = len(f)
    best_f = f.copy()
    best_score = _score(f)
    start = time.monotonic()
    
    for it in range(n_iter):
        if time.monotonic() - start > time_limit:
            break
        h = _fft_convolve(f)
        M = float(np.max(h))
        if M <= 0:
            break
        threshold_val = threshold * M + 1e-15
        active = np.where(h >= threshold_val)[0]
        if len(active) == 0:
            active = np.array([int(np.argmax(h))])
        
        p = _pressure(f, active)
        eta = eta_scale / M
        f = f * np.exp(np.clip(-eta * p, -20.0, 20.0))
        f_new = _normalize(f)
        if f_new is None:
            f = np.ones(n) / n
        else:
            f = f_new
        
        s = _score(f)
        if s < best_score:
            best_score = s
            best_f = f.copy()
    
    return best_f, best_score

def _active_lp_fw(f, max_iters, time_limit, n_max_constraints=200):
    """Active-set LP Frank-Wolfe."""
    from scipy.optimize import linprog
    
    f = _normalize(np.asarray(f, dtype=np.float64))
    if f is None:
        return np.asarray(f, dtype=np.float64).copy(), float('inf')
    n = len(f)
    best_f = f.copy()
    best_score = _score(f)
    start = time.monotonic()
    
    for it in range(max_iters):
        if time.monotonic() - start > time_limit:
            break
        
        h = np.convolve(f, f)
        M = float(np.max(h))
        
        # Select active constraints
        threshold = M * 0.95
        active = np.where(h >= threshold)[0]
        if len(active) > n_max_constraints:
            # Keep only the top n_max_constraints
            top_idx = np.argsort(h[active])[-n_max_constraints:]
            active = active[top_idx]
        
        # Build constraint matrix A: A[i, j] such that sum_j A[i,j]*g[j] <= b[i]
        # Constraint: (f*g)[active[i]] <= M
        # (f*g)[k] = sum_j f[k-j]*g[j] (with f[x]=0 for x not in [0,n-1])
        # So A[i, j] = f[k-j] where k = active[i], valid if 0 <= k-j < n
        
        # Build sparse matrix
        row_data = []
        for i, k in enumerate(active):
            j_start = max(0, k - n + 1)
            j_end = min(n - 1, k)
            if j_end >= j_start:
                js = np.arange(j_start, j_end + 1)
                vals = f[k - js]
                row_data.append((i, js, vals))
        
        # Use COO format
        total_nnz = sum(len(js) for _, js, _ in row_data)
        rows = np.concatenate([np.full(len(js), i) for i, js, _ in row_data]) if row_data else np.array([])
        cols = np.concatenate([js for _, js, _ in row_data]) if row_data else np.array([])
        data = np.concatenate([vals for _, _, vals in row_data]) if row_data else np.array([])
        
        A = csr_matrix((data, (rows, cols)), shape=(len(active), n))
        b = np.full(len(active), M)
        
        # Solve LP: max 1^T g s.t. A*g <= b, g >= 0
        c = -np.ones(n)
        res = linprog(c, A_ub=A, b_ub=b, bounds=[(0, None)] * n, method='highs')
        
        if not res.success or res.x is None:
            break
        
        g = np.maximum(res.x, 0)
        Sg = float(np.sum(g))
        if Sg < 1e-10:
            break
        g = g / Sg
        
        # Frank-Wolfe step
        t = 2.0 / (it + 2)
        f_new = (1.0 - t) * f + t * g
        f_norm = _normalize(f_new)
        if f_norm is None:
            break
        f = f_norm
        
        s = _score(f)
        if s < best_score:
            best_score = s
            best_f = f.copy()
        
        if (it + 1) % 20 == 0:
            print(f"      LP-FW it={it+1}: score={best_score:.6f}, M={M:.8f}, |K|={len(active)}")
    
    return best_f, best_score

    j = np.arange(n, dtype=np.float64)

    f = np.maximum(f, 0.0)
    s = np.sum(f)

    """Initialize using random frequency components."""
    freqs = rng.integers(1, n, size=n_freq)

    f = np.maximum(f, 0.0)

def _init_abs_ifft(n, rng):
    """Initialize as |IFFT(random complex)|^2, normalized."""
    # Generate random complex numbers
    Z = (rng.random(n) + 1j * rng.random(n))

    f = np.fft.irfft(Z, n)
    f = np.abs(f) ** 2

    s = np.sum(f)

    f = np.exp(-j**2 / (2.0 * sigma**2))
    s = np.sum(f)
    return f / s if s > 0 else np.ones(n) / n

def _init_bipolar_smooth(n, rng, blur=3):
    """Generate a bipolar-like sequence and smooth it, then take absolute value."""

    f = np.convolve(f, kernel, mode='same')
    f = np.abs(f)
    s = np.sum(f)

"""
Optimizes non-negative step function heights to minimize 2n*max(conv(f,f))/(sum(f))^2.
Uses multi-phase approach: diverse initializations + mirror descent + active-set LP Frank-Wolfe.
Key improvements: larger n (up to 8192), better structured initializations,
longer mirror descent runs, and active-set LP refinement.
"""

import numpy as np
import time
from scipy.optimize import linprog
from scipy.sparse import csr_matrix


def _fft_convolve(f):
    """Full linear convolution using FFT."""
    n = len(f)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    F = np.fft.rfft(f, size)
    h = np.fft.irfft(F * F, size)
    return h[:2 * n - 1]


def _score_fast(f):
    """Compute score using FFT convolution. Assumes f >= 0, sum(f) > 0."""
    n = len(f)
    h = _fft_convolve(f)
    M = float(np.max(h))
    S = float(np.sum(f))
    if S < 0.01:
        return float('inf')
    return 2.0 * n * M / (S * S)


def _normalize(f):
    """Normalize to sum=1, clip negatives."""
    f = np.maximum(f, 0.0)
    s = float(np.sum(f))
    if s < 1e-30:
        return None
    return f / s


def _pressure(f, active_lags):
    """Compute subgradient: p[j] = (1/|K|) * sum_{k in K} 2*f[k-j] (if valid)."""
    n = len(f)
    p = np.zeros(n, dtype=np.float64)
    for k in active_lags:
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        if j_end >= j_start:
            indices = np.arange(j_start, j_end + 1)
            p[indices] += 2.0 * f[k - indices]
    p /= float(len(active_lags))
    return p


def _mirror_descent(f0, n_iter, eta_scale, threshold, time_limit=30.0):
    """Mirror descent with multiplicative updates and active-set subgradient."""
    f = _normalize(np.asarray(f0, dtype=np.float64))
    if f is None:
        return np.asarray(f0, dtype=np.float64).copy(), float('inf')
    n = len(f)
    best_f = f.copy()
    best_score = _score_fast(f)
    start = time.monotonic()

    for it in range(n_iter):
        if time.monotonic() - start > time_limit:
            break
        h = _fft_convolve(f)
        M = float(np.max(h))
        if M <= 1e-30:
            break
        thresh_val = threshold * M + 1e-15
        active = np.where(h >= thresh_val)[0]
        if len(active) == 0:
            active = np.array([int(np.argmax(h))], dtype=np.int64)

        p = _pressure(f, active)
        eta = eta_scale / M
        f = f * np.exp(np.clip(-eta * p, -20.0, 20.0))
        f_new = _normalize(f)
        if f_new is None:
            f = np.ones(n, dtype=np.float64) / n
        else:
            f = f_new

        s = _score_fast(f)
        if s < best_score:
            best_score = s
            best_f = f.copy()

        if (it + 1) % 2000 == 0:
            elapsed = time.monotonic() - start
            print(f"      MD it={it+1}: best={best_score:.6f}, M={M:.6f}, |K|={len(active)}, t={elapsed:.1f}s")

    return best_f, best_score


def _active_lp_frank_wolfe(f, max_iters, time_limit, max_constraints=200):
    """Active-set LP Frank-Wolfe refinement."""
    f = _normalize(np.asarray(f, dtype=np.float64))
    if f is None:
        return np.asarray(f, dtype=np.float64).copy(), float('inf')
    n = len(f)
    best_f = f.copy()
    best_score = _score_fast(f)
    start = time.monotonic()

    for it in range(max_iters):
        if time.monotonic() - start > time_limit:
            break

        h = np.convolve(f, f)
        M = float(np.max(h))

        # Select most active constraints
        threshold = M * 0.90
        active = np.where(h >= threshold)[0]
        if len(active) > max_constraints:
            top_idx = np.argsort(h[active])[-max_constraints:]
            active = active[top_idx]
        if len(active) == 0:
            active = np.array([int(np.argmax(h))])

        # Build sparse constraint matrix
        rows_parts = []
        cols_parts = []
        data_parts = []
        for i, k in enumerate(active):
            j_start = max(0, k - n + 1)
            j_end = min(n - 1, k)
            if j_end >= j_start:
                js = np.arange(j_start, j_end + 1)
                rows_parts.append(np.full(len(js), i, dtype=np.int32))
                cols_parts.append(js.astype(np.int32))
                data_parts.append(f[k - js].copy())

        if not rows_parts:
            break
        rows = np.concatenate(rows_parts)
        cols = np.concatenate(cols_parts)
        data = np.concatenate(data_parts)

        A = csr_matrix((data, (rows, cols)), shape=(len(active), n))
        b = np.full(len(active), M + 1e-12)

        c_obj = -np.ones(n)
        try:
            res = linprog(c_obj, A_ub=A, b_ub=b,
                          bounds=[(0, None)] * n, method='highs')
        except Exception:
            break

        if not res.success or res.x is None:
            break

        g = np.maximum(res.x, 0.0)
        Sg = float(np.sum(g))
        if Sg < 1e-10:
            break
        g = g / Sg

        # Frank-Wolfe step size
        t = 2.0 / (it + 2)
        f_new = (1.0 - t) * f + t * g
        f_norm = _normalize(f_new)
        if f_norm is None:
            break
        f = f_norm

        s = _score_fast(f)
        if s < best_score:
            best_score = s
            best_f = f.copy()

        if (it + 1) % 10 == 0:
            elapsed = time.monotonic() - start
            print(f"      LP-FW it={it+1}: score={best_score:.6f}, |K|={len(active)}, t={elapsed:.1f}s")

    return best_f, best_score


def _init_cosine(n):
    j = np.arange(n, dtype=np.float64)
    f = np.cos(np.pi * (j + 0.5) / n) ** 2
    f = np.maximum(f, 0.0)
    s = np.sum(f)
    return f / s if s > 0 else np.ones(n) / n


def _init_random(n, rng):
    f = rng.random(n)
    return _normalize(f)


def _init_random_freq(n, rng, n_freq=15):
    """Sum of random cosines, clipped to non-negative."""
    freqs = rng.integers(1, n // 2 + 1, size=n_freq)
    phases = rng.random(n_freq) * 2.0 * np.pi
    amps = rng.random(n_freq) * 0.4
    j = np.arange(n, dtype=np.float64)
    f = np.ones(n, dtype=np.float64)
    for k in range(n_freq):
        f += amps[k] * np.cos(2.0 * np.pi * freqs[k] * j / n + phases[k])
    f = np.maximum(f, 0.0)
    return _normalize(f) if np.sum(f) > 0 else np.ones(n) / n


def _init_abs_ifft(n, rng):
    """|IFFT(random complex)|^2 as initialization."""
    Z = rng.random(n) + 1j * rng.random(n)
    f = np.fft.irfft(Z, n)
    f = f * f
    f = np.maximum(f, 0.0)
    return _normalize(f) if np.sum(f) > 0 else np.ones(n) / n


def _init_tapered(n, power):
    j = np.arange(n, dtype=np.float64)
    x = np.sin(np.pi * (j + 0.5) / n)
    f = x ** power
    f = np.maximum(f, 0.0)
    return _normalize(f) if np.sum(f) > 0 else np.ones(n) / n


def _init_gaussian(n, sigma_frac):
    j = np.arange(n, dtype=np.float64) - (n - 1) / 2.0
    sigma = sigma_frac * n
    f = np.exp(-j * j / (2.0 * sigma * sigma))
    return _normalize(f) if np.sum(f) > 0 else np.ones(n) / n


def _multiscale_upgrade(f_small, n_large, rng):
    """Upscale a small optimized sequence to a larger one via interpolation."""
    n_small = len(f_small)
    # Upsample by inserting zeros, then smooth
    ratio = n_large // n_small
    f_up = np.zeros(n_large, dtype=np.float64)
    for i in range(n_small):
        f_up[i * ratio] = f_small[i] * ratio
    # Smooth with a window
    if ratio >= 2:
        kernel = np.hanning(min(ratio * 2, 16))
        kernel /= kernel.sum()
        f_up = np.convolve(f_up, kernel, mode='same')
    f_up = np.maximum(f_up, 0.0)
    # Add small random perturbation for diversity
    f_up += rng.random(n_large) * 0.01
    return _normalize(f_up)

    final_list = [float(x) for x in best_f]
    final_score = float(evaluate_sequence(final_list))
    print(f"Final verified score: {final_score:.6f} (n={len(final_list)})")

    if not np.isfinite(final_score) or final_score > 2.0:
        print("  Falling back to parent.")
        return [float(x) for x in parent]

    return final_list
