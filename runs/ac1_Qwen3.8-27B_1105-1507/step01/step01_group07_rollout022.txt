def linearized_iteration(a):
    """One iteration of the successive linearization method."""
    from scipy.optimize import linprog
    n = len(a)
    conv = np.convolve(a, a)
    
    # Build constraint matrix
    # 2 * A[k,j] * a[j] - t <= conv[k] for all k
    # A[k,j] = a[k-j] if 0<=k-j<n else 0
    A = np.zeros((2*n-1, n+1))
    for k in range(2*n-1):
        for j in range(n):
            idx = k - j
            if 0 <= idx < n:
                A[k, j] = 2.0 * a[idx]
        A[k, n] = -1.0  # coefficient of t
    
    b = conv.copy()  # RHS
    
    # Objective: minimize t, so c = [0, 0, ..., 0, 1]
    c = np.zeros(n+1)
    c[n] = 1.0
    
    # Equality: sum(a) = 1
    A_eq = np.zeros((1, n+1))
    A_eq[0, :n] = 1.0
    b_eq = [1.0]
    
    # Bounds: a[j] >= 0, t >= 0
    bounds = [(0, None)] * (n+1)
    
    res = linprog(c, A_ub=A, b_ub=b, A_eq=A_eq, b_eq=b_eq, bounds=bounds, method='highs')
    if res.status != 0:
        return None
    
    return res.x[:n]  # the new a (t is res.x[n])

    """
    rng = np.random.default_rng(seed)
    start_time = time.monotonic()
    deadline = start_time + float(budget_s) - 5.0

    best_seq = None
    best_score = np.inf

    # Try parent sequence
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64)
        parent = np.clip(parent, 0, 1000.0)
        if parent.size > 0 and np.isfinite(parent).all():
            obj = compute_obj(parent)
            if obj < best_score:
                best_score = obj
                best_seq = parent.copy()
            print(f"Parent (n={parent.size}): obj={obj:.6f}")
    except Exception:
        pass

    # Phase 1: Alternating LP across multiple n values
    n_values = [50, 75, 100, 130, 150, 200, 250, 300, 400]
    phase1_deadline = start_time + (deadline - start_time) * 0.45
    print(f"Phase 1: Alternating LP, deadline at t+{(phase1_deadline-start_time):.0f}s")

    for n in n_values:
        if time.monotonic() > phase1_deadline:
            break
        remaining = phase1_deadline - time.monotonic()
        n_restarts = max(1, int(remaining / 15))
        n_restarts = min(n_restarts, 4)
        print(f"  n={n}: {n_restarts} restarts, {remaining:.0f}s left")

        for r in range(n_restarts):
            if time.monotonic() > phase1_deadline:
                break
            a_init = structured_init(n, rng)
            max_iter = int(min(150, max(20, (phase1_deadline - time.monotonic()) / (n_restarts - r) / 3.0)))
            a, obj = alternating_lp(a_init, max_iter=max_iter, verbose=(r == 0))
            if obj < best_score:
                best_score = obj
                best_seq = a.copy()
                print(f"    ALT n={n} r={r}: obj={obj:.6f}")

    print(f"  After Phase 1: best={best_score:.6f} (n={len(best_seq) if best_seq is not None else 0})")

    # Phase 2: Successive linearization + coordinate descent on best
    phase2_deadline = start_time + (deadline - start_time) * 0.75
    print(f"Phase 2: SL + CD, deadline at t+{(phase2_deadline-start_time):.0f}s")

    if best_seq is not None:
        n_best = len(best_seq)
        # Run SL on the best
        a_sl, obj_sl = successive_linearization(best_seq, max_iter=100, verbose=True)
        if obj_sl < best_score:
            best_score = obj_sl
            best_seq = a_sl.copy()

        # CD polishing
        a_cd, obj_cd = coordinate_descent(best_seq, max_sweeps=30, verbose=True)
        if obj_cd < best_score:
            best_score = obj_cd
            best_seq = a_cd.copy()

        # Try SL from a few perturbed starts of the best
        for pert in range(3):
            if time.monotonic() > phase2_deadline:
                break
            a_pert = best_seq * np.exp(rng.normal(0, 0.1, n_best))
            a_pert = np.maximum(a_pert, 0)
            s = np.sum(a_pert)
            if s < 1e-10:
                continue
            a_pert = a_pert / s
            a_sl2, obj_sl2 = successive_linearization(a_pert, max_iter=60)
            if obj_sl2 < best_score:
                best_score = obj_sl2
                best_seq = a_sl2.copy()
                print(f"    SL-pert{pert}: obj={obj_sl2:.6f}")

    # Also try SL/CD on other n values if time permits
    if time.monotonic() < phase2_deadline - 30:
        for n in [best_seq.size if best_seq is not None else 150,
                  int((best_seq.size if best_seq is not None else 150) * 0.7),
                  int((best_seq.size if best_seq is not None else 150) * 1.3)]:
            if n < 10 or n > 500:
                continue
            if time.monotonic() > phase2_deadline - 10:
                break
            a_init = structured_init(n, rng)
            a_sl, obj_sl = successive_linearization(a_init, max_iter=80)
            a_cd, obj_cd = coordinate_descent(a_sl, max_sweeps=20)
            if obj_cd < best_score:
                best_score = obj_cd
                best_seq = a_cd.copy()
                print(f"    SL+CD n={n}: obj={obj_cd:.6f}")

    # Phase 3: Gradient polish + random refine
    print(f"Phase 3: Gradient polish, deadline at t+{(deadline-start_time):.0f}s")
    if best_seq is not None:
        n_best = len(best_seq)
        remaining = deadline - time.monotonic()

        # Gradient polish
        gd_iters = int(min(3000, max(500, remaining / 0.5)))
        a_gd, obj_gd = gradient_polish(best_seq, max_iter=gd_iters, verbose=True)
        if obj_gd < best_score:
            best_score = obj_gd
            best_seq = a_gd.copy()

        # Random refine
        if time.monotonic() < deadline - 10:
            rls_iters = int((deadline - time.monotonic()) / 0.02)
            rls_iters = min(rls_iters, 20000)
            a_r, obj_r = random_refine(best_seq, rng, max_iter=rls_iters)
            if obj_r < best_score:
                best_score = obj_r
                best_seq = a_r.copy()
                print(f"    Random refine: obj={obj_r:.6f}")

        # Final CD
        if time.monotonic() < deadline - 5:
            a_cd2, obj_cd2 = coordinate_descent(best_seq, max_sweeps=10)
            if obj_cd2 < best_score:
                best_score = obj_cd2
                best_seq = a_cd2.copy()
                print(f"    Final CD: obj={obj_cd2:.6f}")

    # Fallback
    if best_seq is None:
        best_seq = np.ones(100)
        best_score = compute_obj(best_seq)

    elapsed = time.monotonic() - start_time
    print(f"DONE: n={len(best_seq)}, score={best_score:.6f}, time={elapsed:.1f}s")
    return [float(x) for x in best_seq]
