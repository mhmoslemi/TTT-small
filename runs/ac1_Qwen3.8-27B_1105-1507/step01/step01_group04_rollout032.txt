    f = (f + f[::-1]) / 2.0
    return f

def _lse_optimize(f0, n_iter, T_start, T_end, lr, time_limit=None):
    """
    Log-sum-exp smoothed optimization with Adam.
    Minimizes (1/T)*log(sum exp(T*(f*f)_k)) as a proxy for max(f*f).
    """
    f = _normalize(np.asarray(f0, dtype=np.float64).copy())
    if f is None:
        return np.asarray(f0, dtype=np.float64).copy(), float('inf')
    
    n = len(f)
    f = _symmetrize(f)
    f = _normalize(f)
    if f is None:
        return np.asarray(f0, dtype=np.float64).copy(), float('inf')
    
    best_f = f.copy()
    best_score = _score(f)
    
    # Adam state
    m = np.zeros(n)
    v = np.zeros(n)
    beta1, beta2, eps = 0.9, 0.999, 1e-8
    t_adam = 0
    
    # Precompute FFT size
    fft_size = 1
    while fft_size < 2*n:
        fft_size <<= 1
    
    start_time = time.monotonic()
    
    for it in range(n_iter):
        if time_limit is not None and time.monotonic() - start_time > time_limit:
            break
        
        # Anneal temperature (geometric)
        progress = (it + 1) / max(1, n_iter)
        T = T_start * (T_end / T_start) ** progress
        
        # Compute softmax weights
        h = _fft_conv(f)
        hT = T * h
        hT = hT - np.max(hT)  # for numerical stability
        weights = np.exp(hT)
        weights /= np.sum(weights)
        
        # Gradient: (2/T) * cross_corr(f, weights)
        # cross_corr[j] = sum_m f[m] * weights[m+j]
        F_fft = np.fft.rfft(f, fft_size)
        W_fft = np.fft.rfft(weights, fft_size)
        cross = np.fft.irfft(F_fft * np.conj(W_fft), fft_size)
        grad = 2.0 * cross[:n] / T
        
        # Symmetrize gradient
        grad = _symmetrize(grad)
        
        # Adam update
        t_adam += 1
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * grad * grad
        m_hat = m / (1 - beta1**t_adam)
        v_hat = v / (1 - beta2**t_adam)
        
        f = f - lr * m_hat / (np.sqrt(v_hat) + eps)
        f = _symmetrize(f)
        f = np.maximum(f, 0.0)
        s = float(np.sum(f))
        if s < 1e-30:
            f = np.ones(n) / n
        else:
            f = f / s
        
        # Update best
        score = _score(f)
        if score < best_score:
            best_score = score
            best_f = f.copy()
        
        if (it+1) % 5000 == 0:
            print(f"    LSE it={it+1}, T={T:.3f}, score={best_score:.6f}")
    
    return best_f, best_score

Minimizes 2n*max(conv(f,f))/(sum(f))^2 using:
1. Log-sum-exp smoothed gradient descent with Adam optimizer
2. Symmetry enforcement (f[i] = f[n-1-i])
3. Multiple diverse initializations
4. Coordinate descent refinement
5. LP-based Frank-Wolfe for final polish

import numpy as np
import time

def _fft_conv(f):
    """Full linear self-convolution via FFT."""
    n = len(f)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    F = np.fft.rfft(f, size)
    return np.fft.irfft(F * F, size)[:2 * n - 1]

def _score(f):
    """Compute 2n*max(conv)/sum^2."""
    n = len(f)
    h = _fft_conv(f)
    M = float(np.max(h))
    S = float(np.sum(f))
    if S < 0.01:
        return float('inf')
    return 2.0 * n * M / (S * S)

def _clip_norm(f):
    """Clip to non-negative and normalize to sum=1."""
    f = np.maximum(f, 0.0)
    s = float(np.sum(f))
    if s < 1e-30:
        return None
    return f / s

def _sym(f):
    """Enforce palindromic symmetry."""
    return (f + f[::-1]) * 0.5

def _init_cosine(n):
    j = np.arange(n) + 0.5
    f = np.cos(np.pi * j / n) ** 2
    return _clip_norm(f)

        f = _clip_norm(f)
        if f is None:

        fft_size <<= 1

"""
Minimizes 2n*max(conv(f,f))/(sum(f))^2 using:
1. Log-sum-exp smoothed gradient descent with Adam optimizer and temperature annealing
2. Symmetry enforcement (palindromic constraint)
3. Diverse parametric initializations + random restarts
4. p-norm based gradient as alternative smooth objective
5. Coordinate descent refinement
6. LP-based Frank-Wolfe for final polish on smaller sequences

import numpy as np
import time

def _fft_conv(f):
    """Full linear self-convolution via FFT."""
    n = len(f)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    F = np.fft.rfft(f, size)
    return np.fft.irfft(F * F, size)[:2 * n - 1]

def _score(f):
    """Compute 2n*max(conv)/sum^2."""
    n = len(f)
    h = _fft_conv(f)
    M = float(np.max(h))
    S = float(np.sum(f))
    if S < 0.01:
        return float('inf')
    return 2.0 * n * M / (S * S)

def _clip_norm(f):
    """Clip to non-negative and normalize to sum=1."""
    f = np.maximum(f, 0.0)
    s = float(np.sum(f))
    if s < 1e-30:
        return None
    return f / s

def _sym(f):
    """Enforce palindromic symmetry."""
    return (f + f[::-1]) * 0.5

def _init_cosine(n):
    j = np.arange(n) + 0.5
    f = np.cos(np.pi * j / n) ** 2
    return _clip_norm(f)

def _init_triangle(n):
    j = np.arange(n, dtype=float)
    mid = (n - 1) / 2.0
    f = np.maximum(1.0 - np.abs(j - mid) / mid, 0.0)
    return _clip_norm(f)

def _init_gaussian(n, sigma):
    j = np.arange(n, dtype=float) - (n - 1) / 2.0
    s = sigma * n
    f = np.exp(-j * j / (2.0 * s * s))
    return _clip_norm(f)

def _init_poly(n, p):
    j = np.arange(n, dtype=float)
    x = 2.0 * j / (n - 1) - 1.0
    f = np.maximum(1.0 - x * x, 0.0) ** p
    return _clip_norm(f)

def _init_sinc(n):
    j = np.arange(n, dtype=float) - (n - 1) / 2.0
    x = j / (0.25 * n)
    with np.errstate(divide='ignore', invalid='ignore'):
        f = np.sin(np.pi * x) / (np.pi * x)
    f[~np.isfinite(f)] = 0.0
    f = np.maximum(f, 0.0)
    return _clip_norm(f)

def _init_bell(n, p):
    """(1-|2x-1|^p)^(1/p) - superellipse."""
    j = np.arange(n, dtype=float)
    x = np.abs(2.0 * j / (n - 1) - 1.0)
    f = np.maximum(1.0 - x ** p, 0.0) ** (1.0 / p)
    return _clip_norm(f)

def _lse_adam(f0, n_iter, T0, T1, lr, time_limit=None, use_sym=True):
    """
    Log-sum-exp smoothed optimization with Adam.
    Minimizes (1/T)*log(sum exp(T*c_k)) as smooth proxy for max(c_k).
    """
    f = np.asarray(f0, dtype=np.float64).copy()
    f = _clip_norm(f)
    if f is None:
        return np.asarray(f0, dtype=float).copy(), float('inf')
    if use_sym:
        f = _sym(f)
        f = _clip_norm(f)
        if f is None:
            return np.asarray(f0, dtype=float).copy(), float('inf')

    n = len(f)
    best_f = f.copy()
    best_score = _score(f)

    fft_size = 1
    while fft_size < 2 * n:
        fft_size <<= 1

    m = np.zeros(n)
    v = np.zeros(n)
    b1, b2, eps = 0.85, 0.999, 1e-8
    t_adam = 0
    t0 = time.monotonic()

    for it in range(n_iter):
        if time_limit is not None and time.monotonic() - t0 > time_limit:
            break

        frac = (it + 1) / max(1, n_iter)
        T = T0 * (T1 / T0) ** frac

        h = _fft_conv(f)
        hT = T * (h - np.max(h))
        w = np.exp(hT)
        w_sum = w.sum()
        w /= w_sum

        Ff = np.fft.rfft(f, fft_size)
        Fw = np.fft.rfft(w, fft_size)
        grad_raw = np.fft.irfft(Ff * np.conj(Fw), fft_size)[:n]
        grad = grad_raw * (2.0 / T)
        if use_sym:
            grad = _sym(grad)

        t_adam += 1
        m = b1 * m + (1.0 - b1) * grad
        v = b2 * v + (1.0 - b2) * grad * grad
        mh = m / (1.0 - b1 ** t_adam)
        vh = v / (1.0 - b2 ** t_adam)

        f = f - lr * mh / (np.sqrt(vh) + eps)
        if use_sym:
            f = _sym(f)
        f = _clip_norm(f)
        if f is None:
            f = np.ones(n) / n
            if use_sym:
                f = _sym(f)

        sc = _score(f)
        if sc < best_score:
            best_score = sc
            best_f = f.copy()

        if (it + 1) % 5000 == 0:
            print(f"      LSE it={it+1}/{n_iter} T={T:.1f} score={best_score:.6f}")

    return best_f, best_score

def _pnorm_adam(f0, n_iter, p, lr, time_limit=None, use_sym=True):
    """
    p-norm based optimization with Adam.
    Minimizes ||conv||_p^p = sum c_k^p (smooth for p >= 1).
    """

        fft_size <<= 1

    m = np.zeros(n)
    v = np.zeros(n)
    b1, b2, eps = 0.9, 0.999, 1e-8
    t_adam = 0

        w_sum = w.sum()
        if w_sum < 1e-30:

            grad = _sym(grad)

        t_adam += 1

        mh = m / (1.0 - b1 ** t_adam)
        vh = v / (1.0 - b2 ** t_adam)

        f = f - lr * mh / (np.sqrt(vh) + eps)

def _mirror_descent(f0, n_iter, eta, thresh_frac, time_limit=None, use_sym=True):
    """Multiplicative mirror descent with active set subgradient."""
    f = np.asarray(f0, dtype=np.float64).copy()
    f = _clip_norm(f)
    if f is None:
        return np.asarray(f0, dtype=float).copy(), float('inf')
    if use_sym:
        f = _sym(f)
        f = _clip_norm(f)
        if f is None:
            return np.asarray(f0, dtype=float).copy(), float('inf')

    n = len(f)
    best_f = f.copy()
    best_score = _score(f)
    t0 = time.monotonic()

    for it in range(n_iter):
        if time_limit is not None and time.monotonic() - t0 > time_limit:
            break

        h = _fft_conv(f)
        M = float(np.max(h))
        if M <= 0:
            break
        threshold = thresh_frac * M
        active = np.where(h >= threshold)[0]
        if len(active) == 0:
            active = np.array([int(np.argmax(h))])

        # Pressure vector
        p = np.zeros(n)
        for k in active:
            j_start = max(0, k - n + 1)
            j_end = min(n - 1, k)
            if j_end >= j_start:
                lo = k - j_end
                hi = k - j_start
                p[j_start:j_end + 1] += 2.0 * f[lo:hi + 1][::-1]
        p /= len(active)
        if use_sym:
            p = _sym(p)

        eta_eff = eta / M
        f = f * np.exp(np.clip(-eta_eff * p, -20.0, 20.0))
        if use_sym:
            f = _sym(f)
        f = _clip_norm(f)
        if f is None:
            f = np.ones(n) / n

        sc = _score(f)
        if sc < best_score:
            best_score = sc
            best_f = f.copy()

        if (it + 1) % 3000 == 0:
            print(f"      MD it={it+1}/{n_iter} score={best_score:.6f} |K|={len(active)}")

    return best_f, best_score

def _coord_descent_fast(f, n_sweeps, time_limit=None):
    """
    Coordinate descent with ternary search per coordinate.
    For each j, minimize max_k (h_k + 2*(x - f[j])*f[k-j]) over x >= 0.
    """
    f = np.asarray(f, dtype=np.float64).copy()
    n = len(f)
    t0 = time.monotonic()

    for sweep in range(n_sweeps):
        if time_limit is not None and time.monotonic() - t0 > time_limit:
            break

        h = _fft_conv(f)
        h_left = np.maximum.accumulate(h)  # max of h[0..k]
        # Precompute prefix max and suffix max for efficiency
        prefix_max = np.zeros(2 * n - 1)
        prefix_max[0] = h[0]
        for i in range(1, 2 * n - 1):
            prefix_max[i] = max(prefix_max[i - 1], h[i])
        suffix_max = np.zeros(2 * n - 1)
        suffix_max[-1] = h[-1]
        for i in range(2 * n - 2, -1, -1):
            suffix_max[i] = max(suffix_max[i + 1] if i + 1 < 2 * n - 1 else -np.inf, h[i])

        for j in range(n):
            fj = f[j]
            # M(x) = max( max(h[0:j]), max(h[j:j+n] + 2*(x-fj)*f), max(h[j+n:]) )
            left_max = prefix_max[j - 1] if j > 0 else -np.inf
            right_max = suffix_max[j + n] if j + n < 2 * n - 1 else -np.inf
            base = max(left_max, right_max)

            # Middle part: max_i (h[j+i] + 2*(x-fj)*f[i]) for i=0..n-1
            # This is max of n lines in x. Find minimum of max(base, middle(x)).
            # If base >= middle(x) for all x, M = base (constant).
            # Otherwise, find where middle(x) = base and check.

            # Simple approach: ternary search on [0, 3*fj]
            lo = 0.0
            hi = max(3.0 * fj, 1e-6)

            # Check if x=0 is already optimal
            mid_vals_at_0 = h[j:j + n]
            M0 = max(base, float(np.max(mid_vals_at_0)))

            # Try a few points to get a better bracket
            candidates = [0.0, fj * 0.5, fj, fj * 1.5, fj * 2.0, fj * 3.0]
            best_x = fj
            best_M = M0
            for x_c in candidates:
                d = 2.0 * (x_c - fj)
                mid_vals = h[j:j + n] + d * f
                M_c = max(base, float(np.max(mid_vals)))
                if M_c < best_M:
                    best_M = M_c
                    best_x = x_c

            # Refine with ternary search around best_x
            span = max(0.5 * fj, 1e-6)
            lo = max(0.0, best_x - span)
            hi = best_x + span
            for _ in range(15):
                m1 = lo + (hi - lo) / 3.0
                m2 = hi - (hi - lo) / 3.0
                d1 = 2.0 * (m1 - fj)
                d2 = 2.0 * (m2 - fj)
                M1 = max(base, float(np.max(h[j:j + n] + d1 * f)))
                M2 = max(base, float(np.max(h[j:j + n] + d2 * f)))
                if M1 > M2:
                    lo = m1
                else:
                    hi = m2

            x_opt = (lo + hi) / 2.0
            d_opt = 2.0 * (x_opt - fj)
            M_opt = max(base, float(np.max(h[j:j + n] + d_opt * f)))
            if M_opt < best_M:
                f[j] = max(0.0, x_opt)

    f = _clip_norm(f)
    if f is None:
        f = np.ones(n) / n
    return f

def _lp_frank_wolfe(f_init, max_iters, time_budget):
    """LP-based Frank-Wolfe: maximize sum(b) s.t. f*b <= M, b >= 0."""
    from scipy.optimize import linprog
    from scipy.sparse import coo_matrix

    f = np.asarray(f_init, dtype=np.float64).copy()
    f = _clip_norm(f)
    if f is None:
        return np.asarray(f_init, dtype=float).copy(), float('inf')
    n = len(f)
    best_f = f.copy()
    best_score = _score(f)
    t0 = time.monotonic()

    for it in range(max_iters):
        if time.monotonic() - t0 > time_budget:
            break

        h = np.convolve(f, f)
        M = float(np.max(h))

        # Build sparse C: C[k,j] = f[k-j] if 0 <= k-j < n
        row_idx = []
        col_idx = []
        vals = []
        for j in range(n):
            rows = np.arange(j, j + n)
            row_idx.extend(rows.tolist())
            col_idx.extend([j] * n)
            vals.extend(f.tolist())

        C = coo_matrix((vals, (row_idx, col_idx)), shape=(2 * n - 1, n))
        c_obj = -np.ones(n)
        res = linprog(c_obj, A_ub=C, b_ub=np.full(2 * n - 1, M),
                      bounds=[(0, None)] * n, method='highs')

        if not res.success or res.x is None:
            break

        g = res.x
        Sg = float(np.sum(g))
        if Sg < 1e-10:
            break
        g = g / Sg

        t = 2.0 / (it + 2)
        f_new = (1.0 - t) * f + t * g
        f = _clip_norm(f_new)
        if f is None:
            break

        sc = _score(f)
        if sc < best_score:
            best_score = sc
            best_f = f.copy()

        if (it + 1) % 5 == 0:
            print(f"      LP it={it+1} score={best_score:.6f}")

    return best_f, best_score
