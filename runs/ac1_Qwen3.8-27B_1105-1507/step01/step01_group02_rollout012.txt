            lr = max(lr * 0.85, lr_min)
        elif stall_count == 0:
            lr = min(lr * 1.02, base_lr)

        # Number of active peaks (increase as tau increases for stability)
        top_k = min(30 + int(tau / 20), 2 * n)

        # Compute smoothed gradient
        grad = compute_smooth_grad(a, c, tau, top_k=top_k)

        # Adam update
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * grad ** 2
        m_hat = m / (1 - beta1 ** t_adam)
        v_hat = v / (1 - beta2 ** t_adam)
        a_new = a - lr * m_hat / (np.sqrt(v_hat) + eps)
        a = project_simplex(a_new)

        # LP step periodically
        if use_lp and n <= 2000 and it % 250 == 0 and time.time() < deadline - 25:
            b = lp_step(a, best_peak, time_limit=3.0)
            if b is not None:
                t_mix = 0.25 + 0.35 * rng.random()
                a = project_simplex((1.0 - t_mix) * a + t_mix * b)
                # Reset Adam state after mix
                m *= 0.5
                v *= 0.5

        # Perturbation to escape local minima
        if stall_count > max_stall and time.time() < deadline - 15:
            noise_scale = 0.02 + 0.03 * rng.random()
            a = project_simplex(a + rng.normal(0, noise_scale / np.sqrt(n), n))
            stall_count = 0
            tau = max(tau * 0.6, tau_min)
            lr = min(lr * 1.3, base_lr)

    return best_a


def anneal_polish(a, deadline, rng, n_steps=5000):
    """
    Simulated annealing polish: swap small amounts between random pairs
    of elements, accepting improvements and occasional worsenings.
    """
    n = len(a)
    a = project_simplex(a.copy())
    c = fast_convolve(a)
    best_peak = np.max(c)
    best_a = a.copy()

    T = 0.001
    T_min = 1e-7

    for step in range(n_steps):
        if time.time() > deadline:
            break

        T = max(T * 0.9995, T_min)

        # Random swap: move epsilon from a[i] to a[j]
        i, j = rng.integers(0, n, 2)
        if i == j:
            continue
        epsilon = 0.01 / n * rng.random()

        a_trial = a.copy()
        a_trial[i] -= epsilon
        a_trial[j] += epsilon
        if a_trial[i] < 0:
            continue
        a_trial = project_simplex(a_trial)

        c_trial = fast_convolve(a_trial)
        peak_trial = np.max(c_trial)
        delta = peak_trial - best_peak

        if delta < 0 or rng.random() < np.exp(-abs(delta) / max(T, 1e-15)):
            a = a_trial
            if peak_trial < best_peak:
                best_peak = peak_trial
                best_a = a.copy()

    return best_a


def try_configuration(n, init_type, rng, deadline, use_lp=True):
    """Initialize and optimize a sequence of given length."""
    if init_type == 'alt_proj':
        a = alt_projection_init(n, rng, n_iter=min(100, max(40, n // 5)))
    elif init_type == 'random':
        a = rng.random(n)
        a = a / a.sum()
    elif init_type == 'cosine':
        x = np.linspace(0.0, 1.0, n)
        a = 0.5 * (1.0 + np.cos(2.0 * np.pi * x)) + 0.005
        a = a / a.sum()
    elif init_type == 'broadband':
        a = alt_projection_init(n, rng, n_iter=min(60, max(25, n // 8)))
        a = a + 0.08 * rng.random(n)
        a = np.maximum(a, 0.0)
        a = a / a.sum()
    elif init_type == 'gaussian':
        x = np.linspace(-3.5, 3.5, n)
        a = np.exp(-x ** 2 / 5.0)
        a = a / a.sum()
    elif init_type == 'arch':
        x = np.linspace(0.0, 1.0, n)
        a = 4.0 * x * (1.0 - x) + 0.01
        a = a / a.sum()
    elif init_type == 'parent':
        try:
            parent = np.array(height_sequence_1, dtype=np.float64)
            parent = np.clip(parent, 0.0, 1000.0)
            if len(parent) >= n:
                a = parent[:n].copy()
            else:
                a = np.zeros(n)
                a[:len(parent)] = parent
            a = np.maximum(a, 0.0)
            s = a.sum()
            a = a / s if s > 1e-10 else np.ones(n) / n
        except Exception:
            a = rng.random(n)
            a = a / a.sum()
    else:
        a = rng.random(n)
        a = a / a.sum()

    a_opt = optimize_sequence(a, deadline, rng, use_lp=use_lp)
    score = evaluate_sequence(a_opt.tolist())
    return a_opt, score


def propose_candidate(budget_s=1500):
    """
    Multi-phase search: Adam-optimized log-sum-exp with diverse inits,
    multiple lengths, LP refinement, and annealing polish.
    """
    overall_deadline = time.time() + budget_s - 40.0
    rng = np.random.default_rng(777)

    best_score = float('inf')
    best_seq = None

    # Prioritized configurations
    configs = [
        (1024, 'alt_proj'),
        (512, 'alt_proj'),
        (1024, 'parent'),
        (2048, 'alt_proj'),
        (1024, 'broadband'),
        (512, 'parent'),
        (2048, 'broadband'),
        (1024, 'cosine'),
        (4096, 'alt_proj'),
        (512, 'broadband'),
        (2048, 'parent'),
        (768, 'alt_proj'),
    ]

    for i, (n, init_type) in enumerate(configs):
        remaining = overall_deadline - time.time()
        if remaining < 40:
            break

        # Time allocation: front-load important configs
        if i < 3:
            alloc = min(remaining * 0.18, 220)
        elif i < 6:
            alloc = min(remaining * 0.14, 160)
        elif i < 9:
            alloc = min(remaining * 0.12, 120)
        else:
            alloc = min(remaining * 0.10, 90)

        if alloc < 20:
            continue

        config_deadline = time.time() + alloc
        use_lp = n <= 2048

        try:
            a_opt, score = try_configuration(n, init_type, rng, config_deadline, use_lp=use_lp)
        except Exception as e:
            print(f"  Error in config ({n}, {init_type}): {e}")
            continue

        print(f"n={n}, init={init_type}, score={score:.6f}, time_left={overall_deadline - time.time():.0f}s")

        if np.isfinite(score) and score < best_score:
            best_score = score
            best_seq = a_opt.copy()
            print(f"  >>> New best: {best_score:.6f} (n={n})")

    # Annealing polish on best
    if best_seq is not None and time.time() < overall_deadline - 20:
        print(f"Annealing polish (n={len(best_seq)}, current={best_score:.6f})...")
        a_annealed = anneal_polish(best_seq, overall_deadline - 10, rng, n_steps=3000)
        score = evaluate_sequence(a_annealed.tolist())
        print(f"After anneal: score={score:.6f}")
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_seq = a_annealed

    # Final short optimization polish
    if best_seq is not None and time.time() < overall_deadline:
        a_final = optimize_sequence(best_seq, overall_deadline, rng,
                                    use_lp=len(best_seq) <= 2048, max_iters=50000)
        score = evaluate_sequence(a_final.tolist())
        print(f"After final polish: score={score:.6f}")
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_seq = a_final

    if best_seq is None:
        print("Fallback to uniform")
        best_seq = np.ones(256)

    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    final_score = evaluate_sequence(result)
    print(f"FINAL: score={final_score:.6f}, length={len(result)}")
    return result
