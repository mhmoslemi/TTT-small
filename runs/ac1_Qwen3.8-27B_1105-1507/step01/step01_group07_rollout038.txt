    if s < 1e-10:
        return np.ones(len(a)) / len(a)
    return a / s

    n = len(a)
    conv = np.convolve(a, a)
    k_star = int(np.argmax(conv))
    
    s = np.zeros(n)

    j_end = min(n - 1, k_star)

"""
Multi-strategy optimization for minimizing 2*n*max(conv(a,a))/(sum(a))^2.
Uses subgradient descent on the max-convolution, LP alternating, adaptive
random perturbations, and greedy balancing across multiple sequence lengths.
Key innovation: fast subgradient method with multi-active-lag averaging
for handling non-unique maxima, combined with simulated annealing for
escaping local optima.
"""

import numpy as np
from scipy.optimize import linprog
import time


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    conv = np.convolve(a, a)
    max_b = float(np.max(conv))
    sum_a = float(np.sum(a))
    if sum_a < 0.01:
        return np.inf
    return 2.0 * n * max_b / (sum_a * sum_a)


def normalize(a):
    """Normalize a to sum=1, clip to non-negative."""
    a = np.clip(a, 0.0, None)
    s = np.sum(a)
    if s < 1e-12:
        return np.ones(len(a)) / len(a)
    return a / s


def subgradient_of_max_conv(a, n_active=1):
    """Compute averaged subgradient of max_k conv(a,a)[k] w.r.t. a.
    Averages over the top n_active lags for robustness to ties."""
    n = len(a)
    conv = np.convolve(a, a)
    peak = conv.max()
    
    # Find active lags (within 1% of peak)
    threshold = peak * 0.99
    active_lags = np.where(conv >= threshold)[0]
    
    if len(active_lags) > n_active:
        active_lags = active_lags[:n_active]
    
    s = np.zeros(n)
    for k in active_lags:
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        for i in range(j_start, j_end + 1):
            s[i] += 2.0 * a[k - i]
    
    s /= len(active_lags)
    return s, conv


def subgradient_descent(a, max_iter=50000, alpha_init=0.15, alpha_decay=0.99995,
                         n_active=1, verbose=False):
    """Minimize max(conv(a,a)) via projected subgradient descent with sum(a)=1."""
    a = normalize(a.copy())
    best_a = a.copy()
    best_obj = compute_obj(best_a)
    alpha = alpha_init
    stall_count = 0
    
    for it in range(max_iter):
        s, conv = subgradient_of_max_conv(a, n_active=n_active)
        
        # Project onto tangent space of sum(a)=1
        s_proj = s - np.mean(s)
        
        # Scale step by current peak for adaptive sizing
        peak = conv.max()
        scale = 1.0 / (peak + 1e-10)
        
        a_new = a - alpha * scale * s_proj
        a_new = normalize(a_new)
        
        obj_new = compute_obj(a_new)
        if obj_new < best_obj:
            best_a = a_new.copy()
            best_obj = obj_new
            stall_count = 0
        else:
            stall_count += 1
        
        a = a_new
        alpha *= alpha_decay
        
        if alpha < 1e-7:
            break
        if stall_count > 2000:
            # Restart with smaller alpha
            alpha = max(alpha, 0.01)
            a = best_a.copy()
            stall_count = 0
    
    return best_a, best_obj


def simulated_annealing_refine(a, rng, max_iter=30000, T_init=0.3, T_min=0.01,
                                width_frac=0.05):
    """Simulated annealing with block multiplicative perturbations."""
    best_a = a.copy()
    best_obj = compute_obj(best_a)
    cur_a = a.copy()
    cur_obj = best_obj
    n = len(cur_a)
    T = T_init
    cool_rate = (T_min / T_init) ** (1.0 / max(max_iter, 1))
    
    for it in range(max_iter):
        T *= cool_rate
        width = max(1, int(n * width_frac))
        start = int(rng.integers(0, n - width + 1))
        
        candidate = cur_a.copy()
        noise = rng.normal(0, 1.0, width)
        candidate[start:start + width] *= np.exp(noise)
        candidate = normalize(candidate)
        
        obj_new = compute_obj(candidate)
        delta = obj_new - cur_obj
        
        if delta < 0 or (T > T_min and rng.random() < np.exp(-delta / (T * best_obj + 1e-10))):
            cur_a = candidate
            cur_obj = obj_new
            if obj_new < best_obj:
                best_a = candidate.copy()
                best_obj = obj_new
    
    return best_a, best_obj

    return A

    best_a = a.copy()
    best_obj = compute_obj(best_a)
    n = len(best_a)
    improvements = 0
    
    for it in range(max_iter):
        # Adaptive width
        if it < max_iter // 3:
            width = max(1, n // 20)
        elif it < 2 * max_iter // 3:
            width = max(1, n // 50)
        else:
            width = max(1, n // 100)
        
        start = int(rng.integers(0, n - width + 1))
        candidate = best_a.copy()
        candidate[start:start + width] *= np.exp(rng.normal(0, noise_scale, width))
        candidate = normalize(candidate)
        
        obj_new = compute_obj(candidate)
        if obj_new < best_obj:
            best_a = candidate
            best_obj = obj_new
            improvements += 1
    
    return best_a, best_obj


def full_block_perturb(a, rng, max_iter=5000, noise_scale=0.3):
    """Perturb the entire sequence with multiplicative noise."""
    best_a = a.copy()
    best_obj = compute_obj(best_a)
    n = len(best_a)
    
    for it in range(max_iter):
        candidate = best_a * np.exp(rng.normal(0, noise_scale, n))
        candidate = normalize(candidate)
        obj_new = compute_obj(candidate)
        if obj_new < best_obj:
            best_a = candidate
            best_obj = obj_new
    
    return best_a, best_obj


def structured_init(n, rng, style=0):
    """Generate structured initialization."""
    if style == 0:
        # Linearly decreasing
        a = np.linspace(1.0, 0.2, n)
    elif style == 1:
        # Two-level alternating
        a = np.ones(n)
        a[::2] = rng.uniform(1.0, 2.0)
        a[1::2] = rng.uniform(0.3, 0.7)
    elif style == 2:
        # Cosine taper
        x = np.arange(n) / max(n - 1, 1)
        a = 0.5 + 0.5 * np.cos(2 * np.pi * x * rng.uniform(0.5, 1.5))
        a = a + 0.1
    elif style == 3:
        # Geometric decay
        r = rng.uniform(0.7, 0.95)
        a = r ** np.arange(n)
    elif style == 4:
        # Random
        a = rng.random(n)
    else:
        a = np.ones(n)
    
    return normalize(a)


def propose_candidate(seed=42, budget_s=1500, **kwargs):
    """
    Multi-strategy optimization: subgradient descent + simulated annealing
    + LP alternating + random perturbations across multiple sequence lengths.
    """
    rng = np.random.default_rng(seed)
    start_time = time.monotonic()
    deadline = start_time + float(budget_s) - 5.0

    best_seq = None
    best_score = np.inf

    # --- Try parent sequence ---
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64)
        parent = np.clip(parent, 0, 1000.0)
        if parent.size > 0 and np.isfinite(parent).all():
            obj = compute_obj(parent)
            if obj < best_score:
                best_score = obj
                best_seq = parent.copy()
            print(f"Parent (n={parent.size}): obj={obj:.6f}")
    except Exception:
        pass

    # --- Phase 1: Subgradient + SA on parent ---
    if best_seq is not None and time.monotonic() < deadline - 60:
        n_p = len(best_seq)
        a0 = normalize(best_seq.copy())
        
        t0 = time.monotonic()
        a1, obj1 = subgradient_descent(a0, max_iter=80000, alpha_init=0.2,
                                        alpha_decay=0.99997, n_active=1)
        print(f"  SG n={n_p}: obj={obj1:.6f} ({time.monotonic()-t0:.1f}s)")
        if obj1 < best_score:
            best_score = obj1
            best_seq = a1.copy()
        
        if time.monotonic() < deadline - 30:
            a2, obj2 = simulated_annealing_refine(a1, rng, max_iter=20000,
                                                    T_init=0.2, T_min=0.02,
                                                    width_frac=0.08)
            print(f"  SA n={n_p}: obj={obj2:.6f}")
            if obj2 < best_score:
                best_score = obj2
                best_seq = a2.copy()
        
        if time.monotonic() < deadline - 20:
            a3, obj3 = random_block_perturb(best_seq, rng, max_iter=15000,
                                             noise_scale=0.12)
            if obj3 < best_score:
                best_score = obj3
                best_seq = a3.copy()
                print(f"  RB n={n_p}: obj={obj3:.6f}")

    # --- Phase 2: Search multiple n values ---
    n_values = [10, 15, 20, 25, 30, 40, 50, 60, 80, 100, 120, 150,
                200, 250, 300, 400, 500, 600]
    
    # Prioritize n values close to parent if it exists
    if best_seq is not None:
        n_parent = len(best_seq)
        n_values.sort(key=lambda x: abs(x - n_parent))
    
    for n in n_values:
        if time.monotonic() > deadline - 15:
            break
        remaining = deadline - time.monotonic()
        if remaining < 10:
            break
        
        print(f"N={n}: {remaining:.0f}s left, best={best_score:.6f}")
        
        # Number of restarts based on remaining time
        n_restarts = 3 if remaining > 120 else 2
        
        for restart in range(n_restarts):
            if time.monotonic() > deadline - 10:
                break
            
            style = (restart + n) % 5
            a_init = structured_init(n, rng, style=style)
            
            # Subgradient descent
            max_iter_sg = 40000 if n <= 200 else 20000
            a_sg, obj_sg = subgradient_descent(a_init, max_iter=max_iter_sg,
                                                alpha_init=0.15, alpha_decay=0.99995)
            
            # SA refinement
            if time.monotonic() < deadline - 15:
                max_iter_sa = min(15000, int((deadline - time.monotonic() - 10) / 0.02))
                max_iter_sa = max(max_iter_sa, 2000)
                a_sa, obj_sa = simulated_annealing_refine(a_sg, rng,
                                                           max_iter=max_iter_sa,
                                                           T_init=0.15, T_min=0.02,
                                                           width_frac=0.06)
                if obj_sa < obj_sg:
                    a_sg, obj_sg = a_sa, obj_sa
            
            # Quick random perturb
            if time.monotonic() < deadline - 10:
                a_rp, obj_rp = random_block_perturb(a_sg, rng, max_iter=5000,
                                                     noise_scale=0.1)
                if obj_rp < obj_sg:
                    a_sg, obj_sg = a_rp, obj_rp
            
            if obj_sg < best_score:
                best_score = obj_sg
                best_seq = a_sg.copy()
                print(f"  n={n} r={restart} style={style}: obj={best_score:.6f}")

    # --- Phase 3: LP alternating on best (if n small enough) ---
    if best_seq is not None and len(best_seq) <= 250 and time.monotonic() < deadline - 40:
        a_lp, obj_lp = lp_alternating(normalize(best_seq.copy()),
                                       max_iter=40, t=0.4, upper_bound=50.0)
        if obj_lp < best_score:
            best_score = obj_lp
            best_seq = a_lp.copy()
            print(f"LP (n={len(best_seq)}): obj={best_score:.6f}")
        
        # Follow up
        if time.monotonic() < deadline - 15:
            a_fin, obj_fin = simulated_annealing_refine(best_seq, rng,
                                                         max_iter=10000,
                                                         T_init=0.1, T_min=0.01,
                                                         width_frac=0.05)
            if obj_fin < best_score:
                best_score = obj_fin
                best_seq = a_fin.copy()
                print(f"Post-LP SA: obj={best_score:.6f}")

    # --- Phase 4: Final intensive refinement ---
    if best_seq is not None and time.monotonic() < deadline - 10:
        remaining = deadline - time.monotonic()
        n_best = len(best_seq)
        
        # More subgradient with fine-tuning
        max_iter_final = int(remaining * 3000)
        max_iter_final = min(max_iter_final, 100000)
        a_final, obj_final = subgradient_descent(best_seq, max_iter=max_iter_final,
                                                  alpha_init=0.05, alpha_decay=0.99998,
                                                  n_active=3)
        if obj_final < best_score:
            best_score = obj_final
            best_seq = a_final.copy()
        
        # Final random perturb
        if time.monotonic() < deadline - 5:
            a_last, obj_last = random_block_perturb(best_seq, rng,
                                                     max_iter=int((deadline - time.monotonic() - 2) * 2000),
                                                     noise_scale=0.08)
            if obj_last < best_score:
                best_score = obj_last
                best_seq = a_last.copy()
        
        print(f"Final n={n_best}: obj={best_score:.6f}")

    # --- Fallback ---
    if best_seq is None:
        best_seq = np.ones(100) / 100.0
        best_score = compute_obj(best_seq)

    elapsed = time.monotonic() - start_time
    print(f"DONE: n={len(best_seq)}, score={best_score:.6f}, time={elapsed:.1f}s")
    return [float(x) for x in best_seq]
