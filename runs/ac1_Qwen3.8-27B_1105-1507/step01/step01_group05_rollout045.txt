def compute_score_normalized(f):
    """Compute score for a normalized sequence (sum=1): 2*n*max(conv)."""
    n = len(f)
    conv = np.convolve(f, f)
    return 2.0 * n * conv.max()


def normalize(f):
    """Normalize f to sum to 1."""
    s = f.sum()
    if s < 1e-15:
        return f
    return f / s


def spectral_flat(f, n_iter=500, rng=None, target_ratio=1.0):
    """Iterative spectral flattening: project onto flat-magnitude spectrum + non-negative simplex."""
    if rng is None:

def pressure_update(f, n_iter=200, eta=0.1, rng=None):
    """Fast pressure-based update: reduce elements that contribute to convolution peaks."""
    if rng is None:
        rng = np.random.default_rng(0)
    f = f.copy().astype(np.float64)
    f = normalize(f)
    n = len(f)

    best_f = f.copy()
    best_score = compute_score_normalized(f)
    stall_count = 0

    for it in range(n_iter):
        conv = np.convolve(f, f)
        M = conv.max()

        # Compute pressure: for each i, how much does increasing f[i] increase the max?
        # The convolution at position j contributed by f[i] is f[i] * f[j-i]
        # The max is at j*, so pressure on f[i] is f[j*-i]
        # But we should consider all positions near the max
        # Use: pressure_i = max over j of f[j] * f[|j-i|] weighted by proximity to argmax

        # Simpler: use the subgradient at the argmax
        j_star = np.argmax(conv)
        pressure = np.zeros(n)
        for i in range(n):
            idx = j_star - i
            if 0 <= idx < n:
                pressure[i] = f[idx]

        # Also consider the second-highest peak
        conv_sorted = np.sort(conv)[::-1]
        if len(conv_sorted) > 1:
            second_peak = conv_sorted[1]
            if second_peak > 0.9 * M:
                # Find second peak position
                j_star2 = np.argmax(np.where(np.isclose(conv, second_peak), conv, -1))
                if j_star2 != j_star:
                    for i in range(n):
                        idx = j_star2 - i
                        if 0 <= idx < n:
                            pressure[i] += 0.5 * f[idx]

        if pressure.max() > 0:
            pressure = pressure / pressure.max()

        # Multiplicative update
        eta_eff = eta * (1.0 / (1.0 + 0.01 * it))
        f_new = f * np.exp(-eta_eff * pressure)
        f_new = np.clip(f_new, 1e-15, None)
        f_new = normalize(f_new)

        score = compute_score_normalized(f_new)
        if score < best_score - 1e-12:
            best_score = score
            best_f = f_new.copy()
            f = f_new
            stall_count = 0

                    f = f_pert
                    best_f = f.copy()
                    best_score = compute_score_normalized(f)
                    stall_count = 0

def build_conv_matrix(f):
    """Build matrix A such that A @ b = np.convolve(f, b, mode='full')."""
    n = len(f)
    A = np.zeros((2 * n - 1, n))
    for k in range(2 * n - 1):
        j_start = max(0, k - n + 1)
        j_end = min(n, k + 1)
        for j in range(j_start, j_end):
            A[k, j] = f[k - j]
    return A


def lp_refine(f, time_budget, t0=0.5, rng=None, lp_time_limit=3):
    """LP-based refinement with adaptive step size."""
    if rng is None:
        rng = np.random.default_rng(0)

    n = len(f)
    f = f.astype(np.float64).copy()
    f = normalize(f)

    best_f = f.copy()
    best_score = compute_score_normalized(f)
    t = t0
    start = time.monotonic()

    while time.monotonic() - start < time_budget:
        conv = np.convolve(f, f)
        M = conv.max()

        A = build_conv_matrix(f)
        c_obj = -np.ones(n)
        b_ub = np.full(2 * n - 1, M)
        bounds = [(0, None)] * n

        try:
            res = linprog(c_obj, A_ub=A, b_ub=b_ub, bounds=bounds,
                          method='highs', options={'presolve': True, 'time_limit': lp_time_limit})
        except Exception:
            break

        if not res.success or res.x is None:
            break

        b = np.clip(res.x, 0, None)
        b_sum = b.sum()
        if b_sum < 1e-15:
            break

        g = b / b_sum

        # Try multiple step sizes (larger first)
        improved = False
        for step in [min(1.0, t * 1.5), t, t * 0.5, t * 0.25]:
            f_new = (1 - step) * f + step * g
            f_new = np.clip(f_new, 0, None)
            s = f_new.sum()
            if s < 1e-10:
                continue
            f_new = f_new / s

            new_score = compute_score_normalized(f_new)
            if new_score < best_score - 1e-12:
                best_f = f_new.copy()
                best_score = new_score
                f = f_new
                t = min(1.0, t * 1.1)
                improved = True
                break

        if not improved:
            t *= 0.6
            if t < 0.005:
                # Try perturbation
                f_pert = f + rng.normal(0, 0.02, n)
                f_pert = np.clip(f_pert, 0, None)
                f_pert = normalize(f_pert)
                if compute_score_normalized(f_pert) < best_score:
                    f = f_pert
                    best_f = f.copy()
                    best_score = compute_score_normalized(f)
                    t = 0.1
                else:
                    break

    return best_f, best_score


def coordinate_descent(f, n_sweeps=10, rng=None):
    """Coordinate descent polishing."""
    if rng is None:
        rng = np.random.default_rng(0)

    f = f.copy().astype(np.float64)
    f = normalize(f)
    n = len(f)
    best_score = compute_score_normalized(f)

    for sweep in range(n_sweeps):
        order = rng.permutation(n)
        for i in order:
            original = f[i]
            if original < 1e-10:
                continue
            best_val = original
            best_local = best_score

            # Try geometric spread of values
            for factor in [0.3, 0.5, 0.7, 0.85, 1.0/0.85, 1.0/0.7, 1.0/0.5, 1.0/0.3]:
                f[i] = max(0.0, original * factor)
                conv = np.convolve(f, f)
                s = f.sum()
                if s < 1e-10:
                    f[i] = original
                    continue
                score = 2.0 * n * conv.max() / (s * s)
                if score < best_local:
                    best_local = score
                    best_val = f[i]

            f[i] = best_val
            if best_local < best_score:
                best_score = best_local

        f = np.clip(f, 0, None)
        f = normalize(f)

    return f, best_score


def simulated_annealing(f, rng, n_iter=5000, T0=0.05, T_end=0.0001):
    """Simulated annealing with diverse moves."""
    f = f.copy().astype(np.float64)
    f = normalize(f)
    n = len(f)
    best_f = f.copy()
    best_score = compute_score_normalized(f)
    cur_score = best_score
    no_improve = 0

    for it in range(n_iter):
        T = T0 * (T_end / T0) ** (it / n_iter)
        f_new = f.copy()

        r = rng.random()
        if r < 0.3:
            # Block scaling
            width = max(1, int(rng.integers(1, max(2, n // 16))))
            start = int(rng.integers(0, n - width))
            f_new[start:start + width] *= np.exp(rng.normal(0, 0.4))
        elif r < 0.5:
            # Single element perturbation
            i = int(rng.integers(0, n))
            f_new[i] = max(0.0, f_new[i] + rng.normal(0, 0.05))
        elif r < 0.7:
            # Two-element swap
            i, j = rng.integers(0, n, 2)
            if i != j:
                f_new[i], f_new[j] = f_new[j], f_new[i]
        elif r < 0.85:
            # Smooth perturbation (Gaussian kernel)
            center = int(rng.integers(0, n))
            width = max(2, int(n // 32))
            kernel = np.exp(-0.5 * (np.arange(n) - center) ** 2 / (width ** 2))
            f_new += rng.normal(0, 0.1) * kernel
        else:
            # Large move: random reweight
            f_new *= np.exp(rng.normal(0, 0.3, n))

        f_new = np.clip(f_new, 0, None)
        s = f_new.sum()
        if s < 1e-10:
            continue
        f_new = f_new / s
        new_score = compute_score_normalized(f_new)

        if new_score < cur_score or rng.random() < np.exp(-(new_score - cur_score) / max(T, 1e-10)):
            f = f_new
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_f = f_new.copy()
                no_improve = 0
            else:
                no_improve += 1
        else:
            no_improve += 1

        if no_improve > 2000 and it > n_iter // 2:
            break

    return best_f, best_score

            f0 = diverse_init(n, rng)

            # Spectral flattening (fast)
            f1, s1 = spectral_flat(f0, n_iter=int(t_spec * 50), rng=rng)
            print(f"  After spectral: {s1:.6f}")
            if s1 < best_score_overall:
                best_score_overall = s1
                best_overall = f1.copy()

            # Pressure update
            f2, s2 = pressure_update(f1, n_iter=int(t_press * 50), eta=0.15, rng=rng)
            print(f"  After pressure: {s2:.6f}")
            if s2 < best_score_overall:
                best_score_overall = s2
                best_overall = f2.copy()

            # LP refinement
            remaining2 = deadline - time.monotonic()
            t_lp_actual = min(t_lp, max(20, remaining2 * 0.4))
            if t_lp_actual > 10:
                f3, s3 = lp_refine(f2, t_lp_actual, t0=0.5, rng=rng,
                                   lp_time_limit=max(1, t_lp_actual // 10))
                print(f"  After LP: {s3:.6f}")
                if s3 < best_score_overall:
                    best_score_overall = s3
                    best_overall = f3.copy()

        # Strategy B: Different init + spectral + pressure
        remaining = deadline - time.monotonic()
        if remaining > 60:
            t_spec2 = min(20, remaining * 0.15)
            t_press2 = min(20, remaining * 0.15)
            t_lp2 = min(remaining * 0.3, 60)

            f0b = multi_scale_init(n, n_start=max(16, n // 16), rng=rng)

            f1b, s1b = spectral_flat(f0b, n_iter=int(t_spec2 * 50), rng=rng)
            f2b, s2b = pressure_update(f1b, n_iter=int(t_press2 * 50), eta=0.1, rng=rng)

            if s2b < best_score_overall:
                best_score_overall = s2b
                best_overall = f2b.copy()
                print(f"  Strategy B: {s2b:.6f}")

            remaining3 = deadline - time.monotonic()
            if t_lp2 > 10 and remaining3 > t_lp2:
                f3b, s3b = lp_refine(f2b, t_lp2, t0=0.3, rng=rng,
                                     lp_time_limit=max(1, t_lp2 // 10))
                print(f"  Strategy B+LP: {s3b:.6f}")
                if s3b < best_score_overall:
                    best_score_overall = s3b
                    best_overall = f3b.copy()

        # Strategy C: If parent at this length, do focused refinement
        if parent is not None and len(parent) == n:
            remaining = deadline - time.monotonic()
            if remaining > 40:
                t_parent = min(remaining * 0.3, 60)
                # Perturb parent and refine
                fp = parent * (1 + 0.15 * rng.normal(0, 1, n))
                fp = np.clip(fp, 0, None)
                fp = normalize(fp)
                fp, sp = pressure_update(fp, n_iter=200, eta=0.1, rng=rng)
                if sp < best_score_overall and t_parent > 10:
                    fp, sp = lp_refine(fp, t_parent, t0=0.3, rng=rng)
                    print(f"  Parent refine: {sp:.6f}")
                    if sp < best_score_overall:
                        best_score_overall = sp
                        best_overall = fp.copy()

    # Final polishing phase
    if best_overall is not None:
        remaining = deadline - time.monotonic()
        if remaining > 20:
            print(f"\n--- Polishing (remaining: {remaining:.0f}s) ---")
            n_best = len(best_overall)

            # Coordinate descent
            if remaining > 15:
                n_sweeps = max(5, int(min(20, remaining * 0.3)))
                f_cd, s_cd = coordinate_descent(best_overall, n_sweeps=n_sweeps, rng=rng)
                if s_cd < best_score_overall:
                    best_score_overall = s_cd
                    best_overall = f_cd.copy()
                    print(f"  After CD: {best_score_overall:.6f}")

            # Simulated annealing
            remaining = deadline - time.monotonic()
            if remaining > 10:
                n_iter_sa = min(8000, int(remaining * 300))
                f_sa, s_sa = simulated_annealing(best_overall, rng, n_iter=n_iter_sa,
                                                 T0=0.03, T_end=0.00001)
                if s_sa < best_score_overall:
                    best_score_overall = s_sa
                    best_overall = f_sa.copy()
                    print(f"  After SA: {best_score_overall:.6f}")

            # One more CD pass
            remaining = deadline - time.monotonic()
            if remaining > 10:
                f_cd2, s_cd2 = coordinate_descent(best_overall, n_sweeps=5, rng=rng)
                if s_cd2 < best_score_overall:
                    best_score_overall = s_cd2
                    best_overall = f_cd2.copy()
                    print(f"  Final CD: {best_score_overall:.6f}")

    # Verify with evaluate_sequence
    if best_overall is not None:
        best_overall = np.clip(best_overall, 0, 1000)
        final_list = [float(x) for x in best_overall]
        final_score = evaluate_sequence(final_list)
        print(f"\nFinal: score={final_score:.6f}, n={len(final_list)}")
        return final_list

    # Fallback
    print("Fallback")
    return [1.0] * 256
