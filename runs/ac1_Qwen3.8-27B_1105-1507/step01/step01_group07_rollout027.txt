"""
Step function coefficient optimization using gradient descent with soft-max,
LP-based alternating, and multiple restarts.
Minimizes 2*n*max(conv(a,a))/(sum(a))^2.
"""

import numpy as np
from scipy.optimize import linprog, minimize
import time

def compute_obj(a):
    n = len(a)
    if n == 0:
        return np.inf
    conv = np.convolve(a, a)
    max_b = float(np.max(conv))
    sum_a = float(np.sum(a))
    if sum_a < 0.01:
        return np.inf
    return 2.0 * n * max_b / (sum_a * sum_a)

def build_A(a):
    n = len(a)
    A = np.zeros((2 * n - 1, n), dtype=np.float64)
    for k in range(2 * n - 1):
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        A[k, j_start:j_end + 1] = a[k - j_end:k - j_start + 1][::-1]
    return A

def solve_lp(a, upper_bound=100.0):
    n = len(a)
    conv = np.convolve(a, a)
    M = float(np.max(conv))
    A = build_A(a)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, upper_bound)] * n
    res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
    if res.status == 0:
        return res.x, float(-res.fun), M
    return None, None, M

def soft_max_obj_and_grad(a, T):
    """Compute soft-max of conv(a,a) and gradient."""
    n = len(a)
    conv = np.convolve(a, a)
    max_conv = np.max(conv)
    shifted = (conv - max_conv) / T
    exp_vals = np.exp(shifted)
    sum_exp = np.sum(exp_vals)
    p = exp_vals / sum_exp
    sm_val = max_conv + T * np.log(sum_exp)
    
    # grad[i] = (2/T) * sum_j p[i+j] * a[j]
    idx = np.arange(n)
    M = p[idx[:, None] + idx[None, :]]
    grad = (2.0 / T) * (M @ a)
    
    return sm_val, grad

def grad_descent(a, T=0.01, max_iter=5000, lr=0.001, verbose=False):
    """Gradient descent on soft-max of conv."""
    n = len(a)
    a = a.copy().astype(np.float64)
    a = np.clip(a, 0, None)
    s = np.sum(a)
    if s < 1e-10:
        a = np.ones(n) / n
    else:
        a = a / s
    
    best_a = a.copy()
    best_obj = compute_obj(best_a)
    lr_cur = lr
    
    for it in range(max_iter):
        sm_val, grad = soft_max_obj_and_grad(a, T)
        
        a_new = a - lr_cur * grad
        a_new = np.clip(a_new, 0, None)
        s = np.sum(a_new)
        if s < 1e-10:
            break
        a_new = a_new / s
        
        obj = compute_obj(a_new)
        if obj < best_obj:
            best_a = a_new.copy()
            best_obj = obj
            a = a_new
            # Slowly decrease T
            if it % 100 == 0 and T > 0.001:
                T *= 0.95
        else:
            lr_cur *= 0.9
            if lr_cur < 1e-10:
                break
    
    return best_a, best_obj

def lp_optimize(a, max_iter=200, t=0.5):
    """LP-based alternating optimization."""
    n = len(a)
    a = a.copy().astype(np.float64)
    a = np.clip(a, 0, None)
    s = np.sum(a)
    if s < 1e-10:
        a = np.ones(n) / n
    else:
        a = a / s
    
    best_a = a.copy()
    best_obj = compute_obj(best_a)
    stall = 0
    
    for it in range(max_iter):
        g, g_sum, M = solve_lp(a)
        if g is None:
            break
        
        if g_sum <= 1.0 + 1e-4:
            stall += 1
            if stall > 10:
                break
            continue
        stall = 0
        
        g = g / g_sum
        a_new = (1.0 - t) * a + t * g
        a_new = np.clip(a_new, 0, None)
        s = np.sum(a_new)
        if s < 1e-10:
            break
        a_new = a_new / s
        
        obj = compute_obj(a_new)
        a = a_new
        if obj < best_obj:
            best_a = a_new.copy()
            best_obj = obj
    
    return best_a, best_obj

def scipy_minimize_smooth(a, T=0.01, max_iter=1000):
    """Use scipy minimize with soft-max objective."""
    n = len(a)
    a0 = a.copy().astype(np.float64)
    a0 = np.clip(a0, 0, None)
    s = np.sum(a0)
    if s < 1e-10:
        a0 = np.ones(n) / n
    else:
        a0 = a0 / s
    
    def obj_func(x):
        x = np.clip(x, 0, None)
        s = np.sum(x)
        if s < 1e-10:
            return 1e6
        x = x / s
        conv = np.convolve(x, x)
        max_c = np.max(conv)
        shifted = (conv - max_c) / T
        return 2.0 * n * (max_c + T * np.log(np.sum(np.exp(shifted)))) / (1.0)
    
    # Actually, since we normalize to sum=1, the objective is just 2n * soft_max
    # But scipy doesn't handle the normalization well. Let me use a different approach.
    # Use trust-constr with equality constraint sum(x) = 1
    
    def obj_grad(x):
        x = np.clip(x, 0, None)
        s = np.sum(x)
        if s < 1e-10:
            return 1e6, np.zeros(n)
        x = x / s
        conv = np.convolve(x, x)
        max_c = np.max(conv)
        shifted = (conv - max_c) / T
        exp_vals = np.exp(shifted)
        sum_exp = np.sum(exp_vals)
        p = exp_vals / sum_exp
        sm_val = max_c + T * np.log(sum_exp)
        
        idx = np.arange(len(x))
        M = p[idx[:, None] + idx[None, :]]
        grad = (2.0 / T) * (M @ x)
        return 2.0 * len(x) * sm_val, 2.0 * len(x) * grad
    
    constraints = [{'type': 'eq', 'fun': lambda x: np.sum(x) - 1.0, 'jac': lambda x: np.ones(n)}]
    bounds = [(0, None)] * n
    
    res = minimize(obj_grad, a0, method='trust-constr', jac=True,
                   bounds=bounds, constraints=constraints,
                   options={'maxiter': max_iter, 'gtol': 1e-8})
    
    if res.success or res.fun < compute_obj(a0):
        x = np.clip(res.x, 0, None)
        s = np.sum(x)
        if s > 1e-10:
            x = x / s
            obj = compute_obj(x)
            if obj < compute_obj(a0):
                return x, obj
    
    return a0, compute_obj(a0)

def structured_init(n, rng):
    choice = rng.integers(0, 6)
    if choice == 0:
        phase = rng.uniform(0, 2 * np.pi)
        freq = rng.uniform(0.05, 0.5)
        a = np.maximum(0, np.cos(2 * np.pi * freq * np.arange(n) + phase))
    elif choice == 1:
        a = np.exp(-0.5 * ((np.arange(n) - n/2) / (n * rng.uniform(0.2, 0.5)))**2)
    elif choice == 2:
        period = int(rng.integers(max(2, n // 30), max(3, n // 5)))
        pattern = rng.random(period)
        pattern = np.maximum(0, pattern - 0.3)
        a = np.tile(pattern, n // period + 1)[:n]
    elif choice == 3:
        # Two peaks
        center1 = int(n * rng.uniform(0.2, 0.4))
        center2 = int(n * rng.uniform(0.6, 0.8))
        a = np.zeros(n)
        a += np.exp(-0.5 * ((np.arange(n) - center1) / (n * 0.1))**2)
        a += np.exp(-0.5 * ((np.arange(n) - center2) / (n * 0.1))**2)
    elif choice == 4:
        # Alternating high-low
        a = np.ones(n)
        a[::2] = rng.uniform(0.3, 0.7)
    else:
        a = rng.random(n)
        a = np.maximum(0, a - 0.2)
    
    s = np.sum(a)
    if s < 1e-10:
        a = rng.random(n)
        s = np.sum(a)
    return a / s

def random_refine(a, rng, max_iter=3000):

    n = len(a)

        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        A[k, j_start:j_end + 1] = a[k - j_end:k - j_start + 1][::-1]
    return A

        return res.x, float(-res.fun), M
    return None, None, M

def get_grad(a, T):
    """Compute gradient of soft-max approximation of max(conv(a,a))."""
    n = len(a)

                continue
            a_new = a_new / s
            obj = compute_obj(a_new)
            if obj < best_obj:
                best_a = a_new.copy()
                best_obj = obj
                a = a_new
                break
        else:
            a = (1.0 - t) * a + t * g
            a = np.clip(a, 0.0, None)
            s = np.sum(a)
            if s > 1e-10:
                a = a / s

    return best_a, best_obj


def gd_phase(a, max_time, T_init=None):
    n = len(a)
    a = a.copy()
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    if T_init is None:
        conv = np.convolve(a, a)
        T_init = 0.1 * float(np.max(conv))

    T = T_init
    lr = 0.01
    last_improve = 0

    for it in range(20000):
        if time.monotonic() - start > max_time:
            break
        sm_val, grad = get_soft_max_grad(a, T)
        grad_norm = np.linalg.norm(grad)
        if grad_norm < 1e-12:
            break

        a_new = a - lr * grad
        a_new = np.clip(a_new, 0.0, None)
        s = np.sum(a_new)
        if s < 1e-10:
            lr *= 0.5
            continue
        a_new = a_new / s
        obj = compute_obj(a_new)

        if obj < best_obj:
            best_a = a_new.copy()
            best_obj = obj
            a = a_new
            last_improve = it
            if T > 0.001:
                T = max(0.001, T * 0.995)
        else:
            lr *= 0.93
            if lr < 1e-10:
                break
            if it - last_improve > 200:
                T = max(0.001, T * 0.8)
                lr = 0.005
                last_improve = it

    return best_a, best_obj


def block_refine(a, rng, max_time, start_time=None):
    if start_time is None:
        start_time = time.monotonic()
    n = len(a)
    best_a = a.copy()
    best_obj = compute_obj(a)
    it = 0
    max_iters = 200000

    while it < max_iters and (time.monotonic() - start_time) < max_time:
        width = max(1, int(n * rng.uniform(0.01, 0.12)))
        pos = int(rng.integers(0, n - width + 1))
        candidate = best_a.copy()
        sigma = 0.03 + 0.08 * max(0.0, 1.0 - it / max_iters)
        candidate[pos:pos + width] *= np.exp(rng.normal(0, sigma, width))
        candidate = np.clip(candidate, 0.0, None)
        s = np.sum(candidate)
        if s < 1e-10:
            it += 1
            continue
        obj = compute_obj(candidate)
        if obj < best_obj:
            best_a = candidate
            best_obj = obj
        it += 1

    return best_a, best_obj


def full_pipeline(a_init, time_budget):
    n = len(a_init)
    a = a_init.copy().astype(np.float64)
    a = np.clip(a, 0.0, None)
    s = np.sum(a)
    if s < 1e-10:
        a = np.ones(n) / n
    else:
        a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    # Phase 1: LP
    lp_time = min(20.0, time_budget * 0.35)
    max_lp_iters = max(20, int(lp_time / 0.05))
    a, obj = lp_phase(a, max_lp_iters, lp_time)
    if obj < best_obj:
        best_a = a.copy()
        best_obj = obj

    # Phase 2: GD
    if time.monotonic() - start < time_budget * 0.75:
        gd_time = min(25.0, time_budget * 0.35)
        a, obj = gd_phase(a, gd_time)
        if obj < best_obj:
            best_a = a.copy()
            best_obj = obj

    # Phase 3: Second LP round (from GD result)
    if time.monotonic() - start < time_budget * 0.85:
        lp_time2 = min(10.0, time_budget * 0.1)
        a2, obj2 = lp_phase(a, max(10, int(lp_time2 / 0.05)), lp_time2)
        if obj2 < best_obj:
            best_a = a2.copy()
            best_obj = obj2
            a = a2

    # Phase 4: Fine GD
    if time.monotonic() - start < time_budget * 0.95:
        gd_time2 = min(10.0, time_budget * 0.08)
        a, obj = gd_phase(a, gd_time2, T_init=0.005)
        if obj < best_obj:
            best_a = a.copy()
            best_obj = obj

    return best_a, best_obj


def propose_candidate(seed=42, budget_s=1500, **kwargs):
    """
    Multi-phase optimization: structured init -> LP alternating -> gradient
    descent on soft-max -> second LP -> fine GD -> block perturbation.
    Tries multiple sequence lengths and initializations with random restarts.
    """
    rng = np.random.default_rng(seed)
    start_time = time.monotonic()
    deadline = start_time + float(budget_s) - 6.0

    best_seq = None
    best_score = np.inf

    # Try parent
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64)
        parent = np.clip(parent, 0.0, 1000.0)
        if parent.size > 0 and np.isfinite(parent).all():
            obj = compute_obj(parent)
            if obj < best_score:
                best_score = obj
                best_seq = parent.copy()
            print(f"Parent (n={parent.size}): obj={obj:.6f}")
    except Exception:
        pass

    # Also try to optimize the parent if it exists
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64)
        parent = np.clip(parent, 0.0, 1000.0)
        if parent.size > 0 and np.isfinite(parent).all() and parent.size <= 500:
            p_opt, p_obj = full_pipeline(parent.copy(), min(40.0, float(budget_s) * 0.1))
            if p_obj < best_score:
                best_score = p_obj
                best_seq = p_opt.copy()
                print(f"Parent optimized: obj={p_obj:.6f}")
    except Exception:
        pass

    # Main search over different n values
    n_values = [30, 40, 50, 60, 80, 100, 120, 150, 180, 200, 250, 300]

    for n in n_values:
        if time.monotonic() > deadline - 40:
            break
        remaining = deadline - time.monotonic()
        if remaining < 30:
            break

        # Estimate time per n
        num_remaining_n = len([x for x in n_values if x >= n])
        time_per_n = remaining / max(1, num_remaining_n)
        time_per_n = min(time_per_n, 100.0)

        num_restarts = max(2, min(4, int(time_per_n / 25)))
        time_per_restart = time_per_n / num_restarts

        print(f"\nN={n}: remaining={remaining:.0f}s, {num_restarts} restarts")

        for r in range(num_restarts):
            if time.monotonic() > deadline - 40:
                break
            a_init = structured_init(n, rng)
            a_opt, obj_opt = full_pipeline(a_init, time_per_restart)

            if obj_opt < best_score:
                best_score = obj_opt
                best_seq = a_opt.copy()
                print(f"  n={n} r={r}: obj={obj_opt:.6f} ** NEW BEST **")
            else:
                print(f"  n={n} r={r}: obj={obj_opt:.6f}")

    # Final refinement on best found
    if best_seq is not None and time.monotonic() < deadline - 10:
        remaining = deadline - time.monotonic()
        print(f"\nFinal refinement: {remaining:.0f}s")

        # GP phase
        a_ref, obj_ref = full_pipeline(best_seq.copy(), min(remaining * 0.5, 60.0))
        if obj_ref < best_score:
            best_score = obj_ref
            best_seq = a_ref.copy()
            print(f"  After GP: obj={best_score:.6f}")

        # Block perturbation
        if time.monotonic() < deadline - 5:
            bp_time = deadline - time.monotonic() - 3.0
            a_bp, obj_bp = block_refine(best_seq, rng, bp_time)
            if obj_bp < best_score:
                best_score = obj_bp
                best_seq = a_bp.copy()
                print(f"  After BP: obj={best_score:.6f}")

    if best_seq is None:
        best_seq = np.ones(100)
        best_score = compute_obj(best_seq)

    elapsed = time.monotonic() - start_time
    print(f"\nDONE: n={len(best_seq)}, score={best_score:.6f}, time={elapsed:.1f}s")
    return [float(x) for x in best_seq]
