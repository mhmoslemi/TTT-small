```python
def project_simplex(v, c):
    """Project v onto {x >= 0, sum(x) = c}."""
    n = len(v)
    if c <= 0:
        return np.full(n, c/n)  # degenerate
    # Standard algorithm
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n+1) > (cssv - c))[0][-1]
    theta = (cssv[rho] - c) / (rho + 1.0)
    return np.maximum(v - theta, 0)
```

```python
def logsumexp_smoothed_objective(a, t):
    """Compute 2n * (1/t) * log(sum exp(t * conv(a,a)))."""
    n = len(a)
    conv = np.convolve(a, a)
    tc = t * conv
    tc_shifted = tc - np.max(tc)
    result = 2 * n * (np.max(tc) + np.log(np.sum(np.exp(tc_shifted)))) / t
    return result

def logsumexp_smoothed_gradient(a, t):
    """Compute gradient of 2n * (1/t) * log(sum exp(t * conv(a,a)))."""
    n = len(a)
    conv = np.convolve(a, a)
    tc = t * conv
    tc_shifted = tc - np.max(tc)
    exp_vals = np.exp(tc_shifted)
    weights = exp_vals / np.sum(exp_vals)
    # Gradient = 2n * sum_k weights[k] * d(conv[k])/d(a)
    # = 2n * compute_grad(weights, a)
    grad = np.zeros(n)
    for j in range(n):
        end = min(n, 2*n-1-j)
        grad[j] = 2.0 * np.dot(weights[j:j+end], a[:end])
    return 2.0 * n * grad

def project_simplex(v, s):
    """Project v onto the simplex {x >= 0, sum(x) = s}."""
    n = len(v)
    if n == 0:
        return v
    if s <= 0:
        return np.zeros(n)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.arange(1, n+1)
    cond = u * rho > (cssv - s)
    if not np.any(cond):
        return np.full(n, s/n)
    rho_val = np.max(rho[cond])
    theta = (cssv[rho_val-1] - s) / rho_val
    return np.maximum(v - theta, 0)
```

```python
"""
Optimized step function search using log-sum-exp smoothed gradient descent (L-BFGS-B),
subgradient methods with simplex projection, and LP-based direction finding.
Uses multiple sequence lengths and initializations to escape local minima.
"""
import numpy as np
import time
from scipy.optimize import minimize, linprog
from scipy.sparse import csr_matrix


def _lse_objective(x, t):
    """Compute 2n * (1/t) * log(sum(exp(t * conv(x,x))))."""
    n = len(x)
    conv = np.convolve(x, x)
    tc = t * conv
    tc_max = np.max(tc)
    return 2.0 * n * (tc_max + np.log(np.sum(np.exp(tc - tc_max)))) / t


def _lse_gradient(x, t):
    """Compute gradient of 2n * (1/t) * log(sum(exp(t * conv(x,x)))) w.r.t. x."""
    n = len(x)
    conv = np.convolve(x, x)
    tc = t * conv
    tc_max = np.max(tc)
    exp_vals = np.exp(tc - tc_max)
    weights = exp_vals / np.sum(exp_vals)
    grad = np.zeros(n)
    for j in range(n):
        end = min(n, 2 * n - 1 - j)
        grad[j] = 2.0 * np.dot(weights[j:j + end], x[:end])
    return 2.0 * n * grad


class _SmoothedFun:
    """Wrapper to pass t to objective and gradient without closures."""
    def __init__(self, t):
        self.t = t
    def eval(self, x):
        return _lse_objective(x, self.t)
    def grad(self, x):
        return _lse_gradient(x, self.t)


def _project_simplex(v, s):
    """Project v onto {x >= 0, sum(x) = s}."""
    n = len(v)
    if n == 0:
        return v
    if s <= 0:
        return np.zeros(n)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.arange(1, n + 1)
    cond = u * rho > (cssv - s)
    if not np.any(cond):
        return np.full(n, s / n)
    rho_val = np.max(rho[cond])
    theta = (cssv[rho_val - 1] - s) / rho_val
    return np.maximum(v - theta, 0.0)


def _subgradient_optimize(a0, deadline, lr_init=0.005):
    """Subgradient method with soft active set and simplex projection."""
    a = a0.copy().astype(np.float64)
    S = float(np.sum(a))
    if S > 0:
        a = a / S
    best_a = a.copy()
    best_score = evaluate_sequence(best_a.tolist())
    lr = lr_init
    n = len(a)

    it = 0
    while time.monotonic() < deadline:
        conv = np.convolve(a, a)
        max_val = float(np.max(conv))
        if max_val <= 0:
            break

        # Soft active set: top-K lags with exponential weights
        K = min(20, 2 * n - 1)
        top_k = np.argsort(conv)[::-1][:K]
        diffs = max_val - conv[top_k]
        scale = max(0.005 * max_val, 1e-8)
        w = np.exp(-diffs / scale)
        w /= np.sum(w)

        grad = np.zeros(n)
        for idx in range(K):
            k = int(top_k[idx])
            for j in range(n):
                m = k - j
                if 0 <= m < n:
                    grad[j] += w[idx] * 2.0 * a[m]

        new_a = a - lr * grad
        new_a = _project_simplex(new_a, 1.0)

        score = evaluate_sequence(new_a.tolist())
        it += 1
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_a = new_a.copy()
            a = new_a.copy()
            if it % 500 == 0:
                print(f"      subgrad iter {it}: score={best_score:.6f}, lr={lr:.2e}")
        else:
            lr *= 0.97
            if lr < 1e-9:
                break

    return best_a, best_score


def _lp_direction_optimize(a0, deadline, step_init=0.12):
    """LP-based direction finding with adaptive step size."""
    a = a0.copy().astype(np.float64)
    S = float(np.sum(a))
    if S > 0:
        a = a / S
    best_a = a.copy()
    best_score = evaluate_sequence(best_a.tolist())
    step = step_init
    no_improve = 0
    n = len(a)

    it = 0
    while time.monotonic() < deadline:
        conv_a = np.convolve(a, a)
        M = float(np.max(conv_a))
        if M <= 1e-15:
            break

        # Build sparse constraint matrix: (A@g)[k] = convolve(a, g)[k] <= M
        rows = []
        cols = []
        vals = []
        for k in range(2 * n - 1):
            j_min = max(0, k - (n - 1))
            j_max = min(k, n - 1)
            for j in range(j_min, j_max + 1):
                rows.append(k)
                cols.append(j)
                vals.append(a[k - j])

        A = csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))
        c_vec = -np.ones(n)
        b_ub = np.full(2 * n - 1, M)
        bounds = [(0.0, None)] * n

        try:
            res = linprog(c_vec, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                          options={'presolve': True, 'time_limit': 30})
            if res.success:
                g = res.x
                sum_g = float(np.sum(g))
                if sum_g > 1e-12:
                    g = g / sum_g
                    new_a = (1.0 - step) * a + step * g
                    new_a = np.maximum(new_a, 0.0)
                    s = float(np.sum(new_a))
                    if s > 0:
                        new_a = new_a / s
                        score = evaluate_sequence(new_a.tolist())
                        it += 1
                        if np.isfinite(score) and score < best_score:
                            best_score = score
                            best_a = new_a.copy()
                            a = new_a.copy()
                            no_improve = 0
                            if it % 50 == 0:
                                print(f"      LP iter {it}: score={best_score:.6f}, step={step:.3f}")
                        else:
                            no_improve += 1
                            step *= 0.75
                            if step < 0.003 or no_improve > 100:
                                break
                    else:
                        no_improve += 1
                        if no_improve > 50:
                            break
                else:
                    no_improve += 1
                    if no_improve > 50:
                        break
            else:
                no_improve += 1
                if no_improve > 30:
                    break
        except Exception:
            no_improve += 1
            if no_improve > 30:
                break

    return best_a, best_score


def _lbgfgs_optimize(a0, deadline, t_schedule=None):
    """L-BFGS-B with log-sum-exp smoothing and decreasing temperature."""
    a = a0.copy().astype(np.float64)
    S = float(np.sum(a))
    if S > 0:
        a = a / S
    best_a = a.copy()
    best_score = evaluate_sequence(best_a.tolist())

    if t_schedule is None:
        t_schedule = [4.0, 2.0, 1.0, 0.5, 0.25, 0.1, 0.05, 0.02, 0.01]

    n = len(a)
    # Scale maxiter based on n
    max_iter = max(100, min(500, 200000 // n))

    for t in t_schedule:
        if time.monotonic() > deadline:
            break

        sm = _SmoothedFun(t)
        try:
            res = minimize(
                sm.eval, a, jac=sm.grad,
                method='L-BFGS-B',
                bounds=[(0.0, None)] * n,
                options={'maxiter': max_iter, 'ftol': 1e-14, 'gtol': 1e-10, 'maxfun': 5000}
            )
            if res.x is not None:
                a_new = np.maximum(res.x, 0.0)
                s = float(np.sum(a_new))
                if s > 0:
                    a_new = a_new / s
                    score = evaluate_sequence(a_new.tolist())
                    if np.isfinite(score) and score < best_score:
                        best_score = score
                        best_a = a_new.copy()
                        a = a_new.copy()
        except Exception:
            pass

    return best_a, best_score


def _random_restart_optimize(a0, deadline, rng, n_restarts=5):
    """Random perturbation + local optimization to escape local minima."""
    a = a0.copy().astype(np.float64)
    S = float(np.sum(a))
    if S > 0:
        a = a / S
    best_a = a.copy()
    best_score = evaluate_sequence(best_a.tolist())
    n = len(a)

    for r in range(n_restarts):
        if time.monotonic() > deadline:
            break
        # Perturb
        noise = rng.normal(0, 0.05, n)
        perturbed = a + noise
        perturbed = np.maximum(perturbed, 0.0)
        s = float(np.sum(perturbed))
        if s > 0:
            perturbed = perturbed / s

        # Quick local optimization
        score_pert = evaluate_sequence(perturbed.tolist())
        if np.isfinite(score_pert) and score_pert < best_score:
            best_score = score_pert
            best_a = perturbed.copy()
            a = perturbed.copy()
            # Continue from here with subgradient
            sub_deadline = min(deadline, time.monotonic() + (deadline - time.monotonic()) * 0.3)
            a_new, score_new = _subgradient_optimize(perturbed, sub_deadline, lr_init=0.003)
            if score_new < best_score:
                best_score = score_new
                best_a = a_new.copy()
                a = a_new.copy()

    return best_a, best_score


def _make_initializations(n, rng, parent_seq):
    """Generate diverse initializations for length n."""
    inits = []
    idx = np.arange(n, dtype=np.float64)

    # Uniform
    inits.append(('uniform', np.ones(n)))

    # Cosine taper
    inits.append(('cosine', 0.5 * (1.0 + np.cos(np.pi * idx / max(n - 1, 1)))))

    # Hann window
    if n > 2:
        inits.append(('hann', 0.5 * (1.0 - np.cos(2.0 * np.pi * idx / n))))

    # Slight taper at edges
    taper = np.ones(n)
    edge = max(1, n // 20)
    taper[:edge] = np.linspace(0.6, 1.0, edge)
    taper[-edge:] = np.linspace(1.0, 0.6, edge)[::-1]
    inits.append(('taper', taper))

    # Raised cosine (smooth)
    rc = 0.5 * (1.0 + np.cos(np.pi * (2.0 * idx / n - 1.0)))
    inits.append(('raised_cos', rc))

    # Random smooth
    r = rng.uniform(0.7, 1.3, n)
    # Smooth with a short moving average
    kernel = np.ones(5) / 5.0
    r_smooth = np.convolve(r, kernel, mode='same')
    inits.append(('rand_smooth', r_smooth))

    # Bimodal
    c1 = n * 0.25
    c2 = n * 0.75
    w = n * 0.15
    bim = np.exp(-0.5 * ((idx - c1) / w) ** 2) + np.exp(-0.5 * ((idx - c2) / w) ** 2) + 0.3
    inits.append(('bimodal', bim))

    # Parent interpolation
    if parent_seq is not None and len(parent_seq) > 1:
        from numpy import interp
        x_old = np.linspace(0, 1, len(parent_seq))
        x_new = np.linspace(0, 1, n)
        inits.append(('parent_interp', interp(x_new, x_old, parent_seq)))

    return inits
```
