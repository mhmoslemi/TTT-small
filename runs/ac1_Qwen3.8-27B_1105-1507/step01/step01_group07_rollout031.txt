            j_start = max(0, k - n + 1)
            j_end = min(n - 1, k)
            grad_m[j_start:j_end + 1] += a[k - j_end:k - j_start + 1][::-1]
        grad_m /= topk
        
        grad = (2.0 * n / S**2) * grad_m - (4.0 * n * m / S**3) * np.ones(n)
        
        # Adaptive line search
        found = False
        current_lr = lr
        for _ in range(15):
            a_new = a - current_lr * grad
            a_new = np.clip(a_new, 0, None)
            S_new = np.sum(a_new)
            if S_new < 1e-10:
                current_lr *= 0.5
                continue
            a_new = a_new / S_new
            b_new = fftconvolve(a_new, a_new)
            obj_new = 2.0 * n * np.max(b_new) / S_new**2
            
            if obj_new < obj - 1e-10:
                a = a_new
                lr = min(current_lr * 1.2, 5.0)
                found = True
                break
            else:
                current_lr *= 0.5
                
        if not found:
            lr = max(lr * 0.5, 0.01)
            
        it += 1
        
    return best_a, best_obj


def propose_candidate(seed=42, budget_s=1500, **kwargs):
    """
    Main search function. Generates core sequences, optimizes them, pads with 
    edge patterns, and optimizes the full sequences to find the best candidate.
    """
    rng = np.random.default_rng(seed)
    start_time = time.monotonic()
    deadline = start_time + float(budget_s) - 10.0
    
    best_seq = None
    best_score = np.inf
    parent = None
    
    # Try parent sequence
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64)
        parent = np.clip(parent, 0, None)
        if parent.size > 0 and np.isfinite(parent).all():
            obj = compute_obj(parent)
            if obj < best_score:
                best_score = obj
                best_seq = parent.copy()
            print(f"Parent (n={parent.size}): obj={obj:.6f}")
    except Exception:
        pass
    
    # Phase 1: Optimize core sequences
    core_lengths = [30, 50, 80]
    target_lengths = [100, 150, 200, 300, 500, 1000]
    
    print("Phase 1: Core Optimization")
    for n_c in core_lengths:
        if time.monotonic() > deadline:
            break
        remaining = deadline - time.monotonic()
        num_cores = int(min(5, max(2, remaining / 30)))
        
        for _ in range(num_cores):
            if time.monotonic() > deadline:
                break
            
            # Generate different core initializations
            choice = rng.integers(0, 4)
            if choice == 0:
                a_core = np.ones(n_c)
            elif choice == 1:
                a_core = np.sin(np.pi * (np.arange(n_c) + 0.5) / n_c)**2
            elif choice == 2:
                a_core = np.arange(n_c) + 1
            else:
                a_core = rng.random(n_c)
            
            a_core = np.clip(a_core, 0, None)
            S = np.sum(a_core)
            if S < 1e-10:
                a_core = rng.random(n_c)
                S = np.sum(a_core)
            a_core = a_core / S
            
            max_iter = int(min(500, max(50, (deadline - time.monotonic()) / 0.5)))
            a_opt, obj = optimize_sequence(a_core, max_iter, rng, deadline, topk=5)
            
            if obj < best_score:
                best_score = obj
                best_seq = a_opt.copy()
                print(f"  Core n={n_c}: obj={obj:.6f}")
    
    # Phase 2: Pad cores with edge patterns and optimize full sequences
    print("Phase 2: Edge Padding and Full Optimization")
    if best_seq is not None:
        core = best_seq
    else:
        core = np.ones(50)
    
    for n_target in target_lengths:
        if time.monotonic() > deadline:
            break
        
        remaining = deadline - time.monotonic()
        num_edges = int(min(5, max(2, remaining / 20)))
        
        for _ in range(num_edges):
            if time.monotonic() > deadline:
                break
            
            # Create edge patterns
            n_pad = n_target - len(core)
            if n_pad <= 0:
                a_full = core[:n_target]
            else:
                n_left = n_pad // 2
                n_right = n_pad - n_left
                
                # Left edge
                if rng.integers(0, 3) == 0:
                    left_edge = np.linspace(core[0], core[0] * 0.2, n_left) if n_left > 0 else []
                elif n_left > 0:
                    left_edge = rng.random(n_left) * core[0]
                else:
                    left_edge = []
                
                # Right edge
                if rng.integers(0, 3) == 0:
                    right_edge = np.linspace(core[-1], core[-1] * 0.2, n_right) if n_right > 0 else []
                elif n_right > 0:
                    right_edge = rng.random(n_right) * core[-1]
                else:
                    right_edge = []
                
                a_full = np.concatenate([left_edge, core, right_edge])
            
            # Add some noise and normalize
            a_full = a_full * np.exp(rng.normal(0, 0.2, len(a_full)))
            a_full = np.clip(a_full, 0, None)
            S = np.sum(a_full)
            if S < 1e-10:
                a_full = np.ones(n_target)
                S = n_target
            a_full = a_full / S
            
            max_iter = int(min(800, max(100, (deadline - time.monotonic()) / 0.2)))
            a_opt, obj = optimize_sequence(a_full, max_iter, rng, deadline, topk=10)
            
            if obj < best_score:
                best_score = obj
                best_seq = a_opt.copy()
                print(f"  Full n={n_target}: obj={obj:.6f}")
    
    # Phase 3: Final refinement on the best sequence found
    if best_seq is not None and time.monotonic() < deadline - 20:
        print("Phase 3: Final Refinement")
        remaining = deadline - time.monotonic()
        max_iter = int(min(2000, max(200, remaining / 0.1)))
        a_opt, obj = optimize_sequence(best_seq, max_iter, rng, deadline, topk=15)
        if obj < best_score:
            best_score = obj
            best_seq = a_opt.copy()
            print(f"  Final: obj={obj:.6f}")
    
    # Fallback
    if best_seq is None:
        best_seq = np.ones(100)
        best_score = compute_obj(best_seq)
    
    elapsed = time.monotonic() - start_time
    print(f"DONE: n={len(best_seq)}, score={best_score:.6f}, time={elapsed:.1f}s")
    return [float(x) for x in best_seq]
