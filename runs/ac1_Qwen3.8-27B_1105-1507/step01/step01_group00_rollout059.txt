Multi-stage optimization for minimizing 2n*max(conv(a,a))/sum(a)^2.
Uses LP-based direction finding combined with momentum subgradient polishing,
multiple initializations, and periodic random restarts to escape local minima.
"""
import numpy as np
import time
from scipy.optimize import linprog
from scipy.sparse import csr_matrix


def build_conv_matrix_sparse(f):
    """Build sparse matrix A where (A@g)[k] = convolve(f,g)[k]."""
    n = len(f)
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        j_min = max(0, k - (n - 1))
        j_max = min(k, n - 1)
        for j in range(j_min, j_max + 1):
            rows.append(k)
            cols.append(j)
            vals.append(f[k - j])
    return csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))


def solve_lp_direction(f, M):
    """Solve: max sum(g) s.t. g >= 0, convolve(f,g)_k <= M for all k."""
    n = len(f)
    A = build_conv_matrix_sparse(f)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    try:
        result = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                         options={'presolve': True, 'time_limit': 30})
        if result.success:
            return result.x
    except Exception:
        pass
    return None


def compute_multi_lag_gradient(f, threshold=0.9):
    """
    Compute a smoothed gradient using all lags that are within
    'threshold' fraction of the max convolution value.
    """
    n = len(f)
    conv_f = np.convolve(f, f)
    M = float(np.max(conv_f))
    if M <= 0:
        return np.zeros(n)
    
    # Find active lags
    active = conv_f >= threshold * M
    if not np.any(active):
        active = np.zeros(len(conv_f), dtype=bool)
        active[int(np.argmax(conv_f))] = True
    
    grad = np.zeros(n)
    for k in np.where(active)[0]:
        k = int(k)
        weight = conv_f[k] / conv_f[active].sum()
        for j in range(n):
            kj = k - j
            if 0 <= kj < n:
                grad[j] += weight * 2.0 * f[kj]
    
    return grad

def build_conv_matrix_sparse(f):
    """Build sparse matrix A where (A@g)[k] = convolve(f,g)[k]."""
    n = len(f)
    rows = np.empty(n * n, dtype=np.int32)
    cols = np.empty(n * n, dtype=np.int32)
    vals = np.empty(n * n, dtype=np.float64)
    idx = 0
    for k in range(2 * n - 1):
        j_min = max(0, k - (n - 1))
        j_max = min(k, n - 1)
        for j in range(j_min, j_max + 1):
            rows[idx] = k
            cols[idx] = j
            vals[idx] = f[k - j]
            idx += 1
    rows = rows[:idx]
    cols = cols[:idx]
    vals = vals[:idx]
    return csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))


def solve_lp_direction(f, M, time_limit=20):
    """Solve: max sum(g) s.t. g >= 0, convolve(f,g)_k <= M for all k."""
    n = len(f)
    A = build_conv_matrix_sparse(f)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    try:
        result = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                         options={'presolve': True, 'time_limit': time_limit})
        if result.success:
            return result.x
    except Exception:
        pass
    return None

    x = np.exp(np.clip(y, -30, 15))

def momentum_subgrad(f, deadline, lr_init=0.05, momentum=0.9, threshold=0.85,
                     verbose_prefix=""):
    """Momentum subgradient with multiple active lags."""
    f = f.copy().astype(np.float64)
    n = len(f)
    S = float(np.sum(f))
    best_f = f.copy()
    best_score = evaluate_sequence(f.tolist())
    v = np.zeros(n)
    lr = lr_init
    no_improve = 0
    it = 0
    
    while time.monotonic() < deadline - 5:
        if it % 500 == 0 and time.monotonic() > deadline - 10:
            break
        
        conv_f = np.convolve(f, f)
        M = float(np.max(conv_f))
        if M <= 0:
            break
        
        active = conv_f >= threshold * M
        if not np.any(active):
            active = np.zeros(len(conv_f), dtype=bool)
            active[int(np.argmax(conv_f))] = True
        
        weights = np.where(active, conv_f, 0.0)
        w_sum = float(np.sum(weights))
        if w_sum > 0:
            weights = weights / w_sum
            grad = 2.0 * np.correlate(weights, f, 'full')[:n]
        else:
            break
        
        v = momentum * v - lr * grad
        new_f = f + v
        new_f = np.clip(new_f, 0.0, 1000.0)
        new_sum = float(np.sum(new_f))
        if new_sum < 1e-10:
            break
        new_f = new_f * (S / new_sum)
        
        new_score = evaluate_sequence(new_f.tolist())
        it += 1
        
        if np.isfinite(new_score) and new_score < best_score:
            f = new_f
            best_f = f.copy()
            best_score = new_score
            no_improve = 0
            momentum = min(momentum * 1.001, 0.99)
            if (it + 1) % 2000 == 0:
                print(f"    {verbose_prefix}subgrad iter {it}: score={best_score:.6f} (lr={lr:.5f})")
        else:
            no_improve += 1
            lr *= 0.97
            if no_improve > 500:
                v = np.zeros(n)
                lr = lr_init * 0.3
                no_improve = 0
            if lr < 1e-7:
                break
        
        lr *= 0.9995
    
    return best_f, best_score

        print(f"\n=== n={n} ===")
        inits = get_initializations(n, rng)
        
        # Sort initializations by their initial score (try best first)
        init_scores = []
        for label, f_init in inits.items():
            f_init = np.clip(f_init.astype(np.float64), 1e-4, 1000.0)
            s = evaluate_sequence(f_init.tolist())
            if np.isfinite(s):
                init_scores.append((s, label, f_init))
        
        init_scores.sort(key=lambda x: x[0])
        
        # Allocate time per initialization
        time_per_init = min(120, (deadline - time.monotonic() - 60) / max(len(init_scores), 1))
        
        for score_init, label, f_init in init_scores:
            if time.monotonic() > deadline - 30:
                break
            
            if score_init < best_score:
                best_score = score_init
                best_sequence = f_init.copy()
            
            print(f"  [{label}] init={score_init:.6f}", end="")
            
            sub_dl = time.monotonic() + time_per_init
            best_f = f_init.copy()
            best_s = score_init
            
            # Strategy 1: Log-param L-BFGS-B
            if time.monotonic() < sub_dl - 20:
                lf, ls = log_param_optimize(best_f, time.monotonic() + min(60, sub_dl - time.monotonic() - 15),
                                            tau_init=5.0, tau_min=0.5,
                                            verbose_prefix=f"[n={n} {label} LBFGS] ")
                if ls < best_s:
                    best_s = ls
                    best_f = lf
                    print(f" -> LBFGS: {ls:.6f}", end="")
            
            # Strategy 2: LP optimization
            if time.monotonic() < sub_dl - 15:
                lf, ls = lp_optimize(best_f, time.monotonic() + min(45, sub_dl - time.monotonic() - 10),
                                     step_size_init=0.2,
                                     verbose_prefix=f"[n={n} {label} LP] ")
                if ls < best_s:
                    best_s = ls
                    best_f = lf
                    print(f" -> LP: {ls:.6f}", end="")
            
            # Strategy 3: Momentum subgradient
            if time.monotonic() < sub_dl - 5:
                gf, gs = momentum_subgrad(best_f, time.monotonic() + min(30, sub_dl - time.monotonic() - 5),
                                          lr_init=0.03, verbose_prefix=f"[n={n} {label} subg] ")
                if gs < best_s:
                    best_s = gs
                    best_f = gf
                    print(f" -> subg: {gs:.6f}", end="")
            
            if best_s < best_score:
                best_score = best_s
                best_sequence = best_f.copy()
            
            print(f" | best so far: {best_score:.6f}")
    
    # Final refinement phase
    if best_sequence is not None and time.monotonic() < deadline - 20:
        print(f"\n=== Final refinement (n={len(best_sequence)}, score={best_score:.6f}) ===")
        remaining = deadline - time.monotonic() - 10
        
        # L-BFGS polish
        if remaining > 30:
            lf, ls = log_param_optimize(best_sequence, time.monotonic() + min(remaining * 0.3, 120),
                                        tau_init=2.0, tau_min=0.3,
                                        verbose_prefix="[refine LBFGS] ")
            if ls < best_score:
                best_score = ls
                best_sequence = lf
                print(f"  After LBFGS: {best_score:.6f}")
        
        # LP polish
        if time.monotonic() < deadline - 20:
            lf, ls = lp_optimize(best_sequence, time.monotonic() + min(remaining * 0.25, 90),
                                 step_size_init=0.1,
                                 verbose_prefix="[refine LP] ")
            if ls < best_score:
                best_score = ls
                best_sequence = lf
                print(f"  After LP: {best_score:.6f}")
        
        # Subgradient polish
        if time.monotonic() < deadline - 10:
            gf, gs = momentum_subgrad(best_sequence, deadline,
                                      lr_init=0.01, momentum=0.95, threshold=0.9,
                                      verbose_prefix="[refine subg] ")
            if gs < best_score:
                best_score = gs
                best_sequence = gf
                print(f"  After subg: {best_score:.6f}")
    
    if best_sequence is None:
        return [1.0] * 128
    
    elapsed = time.monotonic() - t0
    print(f"\nFINAL: n={len(best_sequence)}, score={best_score:.6f}, time={elapsed:.1f}s")
    return [float(x) for x in best_sequence]
