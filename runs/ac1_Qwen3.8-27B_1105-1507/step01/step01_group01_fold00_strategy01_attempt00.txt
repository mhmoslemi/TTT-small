                                new_init = generate_candidate(new_length, seed)
                                new_score = evaluate_sequence(new_init)
                                evaluations_left -= 2
                                if np.isfinite(new_score) and new_score < best_score:
                                    best_score = new_score
                                    best_sequence = new_init
                                # Lower-priority length exploration
                                else:
                                    if evaluations_left > 1:
                                        sub_sequence = np.clip(mutate_sequence(new_init), 0, 1000)
                                        new_score = evaluate_sequence(sub_sequence)
                                        evaluations_left -= 2
                                        if np.isfinite(new_score) and new_score < best_score:
                                            best_score = new_score
                                            best_sequence = sub_sequence
                        except:
                            pass

                except KeyboardInterrupt:
                    break
            final_sequence = [float(x) for x in normalize_sequence(best_sequence) if x > 0]

        except np.inf in final_sequence:
            # Fallback to standard sequence
            final_sequence = np.ones(len(best_sequence) if best_sequence is not None else 256)
            final_sequence = normalize_sequence(final_sequence)

    else:
        # Fallback to local random search without scipy
        while time.monotonic() < deadline and evaluations_left > 0:
            candidate_seq = random_highpass_filter(generate_candidate(n_candidate, seed))
            candidate_score = evaluate_sequence(candidate_seq)
            if np.isfinite(candidate_score) and candidate_score < best_score:
                best_score = candidate_score
                best_sequence = candidate_seq

            evaluations_left -= 2
            best_sequence = find_minimum(best_sequence, candidate_score, evaluate_sequence)
            seed += 10

        final_sequence = best_sequence.tolist() if best_sequence is not None else None

    return final_sequence
