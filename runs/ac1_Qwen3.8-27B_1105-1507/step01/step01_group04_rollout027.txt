- Use a cheaper "direction finding" than full LP

For better initialization, let me try:
- **Binary phase sequences**: Start with a {0,1} sequence with good autocorrelation properties, then smooth it.
- **m-sequences**: Maximum-length sequences from LFSRs have excellent autocorrelation.
- **Barker-like**: Short sequences with near-perfect autocorrelation, then tiled or padded.

Actually, for convolution (not correlation), the relevant sequences are different. Let me think...

For convolution f*f to be flat, we need the DFT to satisfy |F(ω)|^2 ≈ constant. This is the "flat magnitude" or "equi-amplitude" condition on the frequency response.

Sequences with flat frequency response include:
- ZCZC (Zero Cross-Zero Correlation Zone) sequences
- Fürster sequences (for prime lengths)
- Barannikov sequences

For non-negative sequences with flat |DFT|, one approach is:
1. Start with a sequence that has flat |DFT| (like a Fourier matrix row, but that gives complex values)
2. Take the real part or magnitude
3. Project onto non-negativity

Actually, a simpler approach: use the fact that if f is non-negative and conv(f,f) is flat, then f is approximately a "continuous" version of a perfect sequence. For large n, a good approximation is:
f[j] = |sum_k exp(2πi * k * j / p)| for some prime p

But this gives period p sequences. Let me think differently.

Let me just focus on making the optimization faster and more effective. Here's my plan:

1. **Fast smoothed mirror descent with adaptive step size and momentum**
2. **Multiple diverse initializations**
3. **Periodic re-initialization from best + noise**
4. **For the final best, try LP refinement if n is small enough**
5. **Try both small (512-1024) and medium (2048-4096) sequences**

Let me also consider: the score is 2n * M / S^2. If I scale f by c, the score doesn't change (M scales by c^2, S^2 scales by c^2). So the scale doesn't matter, only the shape.

"""
Optimizes non-negative step function heights to minimize 2n*max(conv(f,f))/(sum(f))^2.
Uses accelerated mirror descent with log-sum-exp smoothing, diverse initializations,
and LP-based Frank-Wolfe refinement for the best candidates.
"""

import numpy as np
import time
import sys

def _fft_convolve(f):
    """Compute full linear convolution f*f using FFT."""
    n = len(f)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    F = np.fft.rfft(f, size)
    h = np.fft.irfft(F * F, size)
    return h[:2 * n - 1]

def _score(f):
    """Compute score = 2n * max(conv(f,f)) / (sum(f))^2."""
    n = len(f)
    h = _fft_convolve(f)
    M = float(np.max(h))
    S = float(np.sum(f))
    if S < 0.01:
        return float('inf')
    return 2.0 * n * M / (S * S)

def _normalize(f):
    """Normalize f to sum to 1, clip to non-negative."""
    f = np.maximum(f, 0.0)
    s = float(np.sum(f))
    if s < 1e-30:
        return None
    return f / s

def _init_uniform(n):
    return np.ones(n) / n

def _init_cosine(n):
    j = np.arange(n, dtype=np.float64)
    f = np.cos(np.pi * (j + 0.5) / n) ** 2
    f = np.maximum(f, 0.0)
    s = np.sum(f)
    return f / s

def _init_gaussian(n, sigma):
    j = np.arange(n, dtype=np.float64) - (n - 1) / 2.0
    f = np.exp(-j * j / (2.0 * sigma * sigma))
    s = np.sum(f)
    return f / s

def _init_rand(n, rng):
    f = rng.random(n)
    s = np.sum(f)
    return f / s

def _init_stripe(n, period, rng):
    """Create a striped pattern with some randomness."""
    j = np.arange(n)
    f = np.where((j % period) < period // 2, 1.0, 0.1)
    f = f + rng.random(n) * 0.1
    f = np.maximum(f, 0.0)
    s = np.sum(f)
    return f / s

def _init_msequence(n):
    """Generate an m-sequence (maximum length sequence) of length n (padded if needed)."""
    # Use a simple LFSR to generate a pseudo-random binary sequence
    # Then smooth it
    n_target = n
    # Find a primitive polynomial for the LFSR
    # For simplicity, use a fixed one and pad/truncate
    # m-sequence of length 2^m - 1
    m = max(1, int(np.ceil(np.log2(n_target + 1))))
    seq_len = 2**m - 1
    # Use a known primitive polynomial (x^8 + x^4 + x^3 + x^2 + 1 for m=8)
    if m == 4:
        taps = [4, 3]
    elif m == 5:
        taps = [5, 3]
    elif m == 6:
        taps = [6, 5]
    elif m == 7:
        taps = [7, 6]
    elif m == 8:
        taps = [8, 6, 5, 4]
    else:
        # Use x^m + x^(m//2) + 1 as a rough guess (may not be primitive)
        taps = [m, m // 2]
    
    # Initialize LFSR
    state = 1
    seq = []
    for i in range(seq_len):
        seq.append(state & 1)
        # Feedback
        fb = 0
        for t in taps:
            fb ^= (state >> (t - 1)) & 1
        state = (state >> 1) | (fb << (m - 1))
    
    seq = np.array(seq[:n_target], dtype=np.float64)
    if len(seq) < n_target:
        seq = np.pad(seq, (0, n_target - len(seq)))
    # Smooth: blend with a cosine window
    j = np.arange(n_target, dtype=np.float64)
    window = 0.5 + 0.5 * np.cos(2 * np.pi * j / n_target)
    f = seq * (0.5 + 0.5 * window)
    f = f + 0.1
    s = np.sum(f)
    return f / s

def _mirror_descent_smoothed(f0, n_iter, mu_init, mu_final, eta0, momentum, time_limit):
    """
    Accelerated mirror descent with log-sum-exp smoothing.
    Minimizes 2n * max(conv(f,f)) s.t. f>=0, sum(f)=1.
    """
    n = len(f0)
    f = _normalize(f0.copy()).astype(np.float64)
    v = np.zeros(n, dtype=np.float64)  # momentum
    best_f = f.copy()
    best_score = _score(f)
    start = time.monotonic()
    
    for it in range(n_iter):
        if time.monotonic() - start > time_limit:
            break
        
        h = _fft_convolve(f)
        Mu = float(np.max(h))
        if Mu <= 1e-30:
            break
        
        # Smoothed max using log-sum-exp
        # Use a temperature that decays from mu_init to mu_final
        frac = it / max(1, n_iter - 1)
        mu = mu_init * (mu_final / mu_init) ** frac
        
        # Compute softmax weights
        h_scaled = h / mu
        h_max = np.max(h_scaled)
        exp_h = np.exp(h_scaled - h_max)
        w = exp_h / np.sum(exp_h)
        
        # Gradient of smoothed max: sum_k w[k] * 2*f[k-j] for each j
        # This is 2 * (conv(w, f_reversed))[j] approximately
        # Actually: grad[j] = sum_k w[k] * 2 * f[k mod n - j] ... 
        # Let's compute it directly: grad[j] = 2 * sum_k w[k] * f[(k-j) mod ...]
        # More precisely: grad[j] = 2 * sum_k w[k] * f[k-j] where f[x] = 0 if x not in [0,n)
        # This is 2 * (the cross-correlation of w and f, evaluated at lag j)
        # = 2 * conv(w, f_reversed)[j + n - 1] ... let me just use FFT
        
        # grad[j] = 2 * sum_k w[k] * f[k-j] for valid k-j
        # = 2 * (w * f_flipped cross-correlation)
        # Let's compute: for each j, sum over k of w[k] * f[k-j]
        # This is the convolution of w with flip(f) evaluated at appropriate index
        # Actually it's simpler: grad = 2 * conv(w, f[::-1])[n-1:n-1+n] ... 
        # Let me just compute it via FFT
        size = 1
        while size < 2 * n - 1:
            size <<= 1
        W = np.fft.rfft(w, size)
        F = np.fft.rfft(f, size)
        # sum_k w[k] * f[k-j] is the cross-correlation, which is conv(w, flip(f))
        # flip(f) has flip(f)[i] = f[n-1-i]
        # conv(w, flip(f))[j + n - 1] = sum_k w[k] * flip(f)[j + n - 1 - k] = sum_k w[k] * f[n-1-(j+n-1-k)] = sum_k w[k] * f[k-j]
        # Yes! So grad[j] = 2 * conv(w, flip(f))[j + n - 1]
        f_flip = f[::-1]
        grad = 2.0 * np.fft.irfft(W * F, size)[:2*n-1]
        # Wait, that's conv(w, f), not conv(w, f_flip).
        # Let me redo: I want sum_k w[k] * f[k-j]
        # conv(w, g)[m] = sum_k w[k] * g[m-k]
        # If g = f, then conv(w, f)[j] = sum_k w[k] * f[j-k]
        # I want sum_k w[k] * f[k-j] = sum_k w[k] * f[-(j-k)] = cross-correlation
        # cross_corr(w, f)[j] = sum_k w[k] * f[k+j] = conv(w, flip(f))[j + n - 1]
        # Hmm, let me just compute it directly for clarity
        # grad[j] = 2 * sum_k w[k] * f[k-j] (f out of range = 0)
        # = 2 * sum_k w[k] * f[k-j]
        # Let's define: for each j, the valid k are those with 0 <= k-j < n, i.e., j <= k <= j+n-1
        # Since k ranges over 0..2n-2 (length of conv), and f ranges over 0..n-1
        # k-j must be in [0, n-1], so k in [j, j+n-1]
        # grad[j] = 2 * sum_{k=j}^{min(j+n-1, 2n-2)} w[k] * f[k-j]
        
        # This is the same as: grad = 2 * ifft(FFT(w_padded) * FFT(f_padded_conj)) ... 
        # Actually, the simplest is to use the correlation formula
        # sum_k w[k] * f[k-j] = <w, shift_j(f)> = convolution-based
        
        # Let me just use the FFT approach properly:
        # Define A = w (length 2n-1), B = f (length n, zero-padded to 2n-1)
        # I want C[j] = sum_k A[k] * B[k-j] for j = 0, ..., n-1
        # This is the cross-correlation of A and B at lag j
        # cross_correlation(A, B)[j] = sum_k A[k] * conj(B[k+j]) for complex
        # For real: sum_k A[k] * B[k+j]
        # I want sum_k A[k] * B[k-j] = cross_correlation(A, B)[-j] = cross_correlation(B, A)[j]
        # = sum_k B[k] * A[k+j]
        
        # Using FFT: cross_corr(A, B) = ifft(conj(FFT(A)) * FFT(B)) (with proper padding)
        # So C[j] = cross_corr(f, w)[j] = ifft(conj(FFT(f)) * FFT(w))[j]
        
        W_full = np.fft.rfft(w, size)
        F_full = np.fft.rfft(f, size)
        corr = np.fft.irfft(np.conj(F_full) * W_full, size)
        # corr[j] = sum_k f[k] * w[k+j] ... hmm, that's not quite right either
        # Let me just verify: ifft(conj(FFT(a)) * FFT(b))[n] = sum_k conj(a[k]) * b[k+n mod N]
        # For real: sum_k a[k] * b[(k+n) mod N]
        # So corr[j] = sum_k f[k] * w[(k+j) mod size]
        # Since w has length 2n-1 and size >= 2n-1, and f has length n:
        # corr[j] = sum_{k=0}^{n-1} f[k] * w[(k+j) mod size]
        # For 0 <= j < n and w[k] = 0 for k >= 2n-1:
        # corr[j] = sum_{k=0}^{n-1} f[k] * w[k+j]
        # I want: grad[j] = 2 * sum_k w[k] * f[k-j]
        # = 2 * sum_{k=j}^{2n-2} w[k] * f[k-j]
        # = 2 * sum_{m=0}^{n-1} w[m+j] * f[m]  (substituting m=k-j)
        # = 2 * corr[j]  !!! (since corr[j] = sum_m f[m] * w[m+j])
        
        grad = 2.0 * corr[:n]
        
        # Mirror descent update with momentum
        eta = eta0 / (Mu + 1e-15)
        log_ratio = -eta * (grad - momentum * v)  # Hmm, this momentum formulation is odd for mirror descent
        # Actually, for mirror descent, the natural "momentum" is to average the subgradients
        # Let me use a simpler approach: exponential moving average of the gradient
        v = momentum * v + (1 - momentum) * grad
        log_ratio = -eta * v
        log_ratio = np.clip(log_ratio, -30.0, 30.0)
        
        f = f * np.exp(log_ratio)
        f_norm = _normalize(f)
        if f_norm is None:
            f = np.ones(n) / n
        else:
            f = f_norm
        
        score = _score(f)
        if score < best_score:
            best_score = score
            best_f = f.copy()
        
        if (it + 1) % 500 == 0:
            elapsed = time.monotonic() - start
            print(f"    MD it={it+1}/{n_iter}: score={best_score:.6f}, Mu={Mu:.8f}, mu={mu:.6f}, t={elapsed:.1f}s")
    
    return best_f, best_score

def _lp_frank_wolfe(f0, max_iters, time_budget):
    """LP-based Frank-Wolfe for normalized problem (sum f = 1, f >= 0)."""
    from scipy.optimize import linprog
    from scipy.sparse import coo_matrix
    
    f = _normalize(f0.copy()).astype(np.float64)
    n = len(f)
    best_f = f.copy()
    best_score = _score(f)
    start = time.monotonic()
    
    for it in range(max_iters):
        if time.monotonic() - start > time_budget:
            break
        
        h = np.convolve(f, f)  # Use direct for accuracy with LP
        M = float(np.max(h))
        
        # Build constraint: conv(g, f)[k] + ... <= M - ... 
        # Actually, we want: find g >= 0 such that conv(f+g, f+g)[k] <= M for all k
        # conv(f+g, f+g) = conv(f,f) + conv(f,g) + conv(g,f) + conv(g,g)
        # For small g (Frank-Wolfe step), we approximate: conv(f,f) + conv(f,g) + conv(g,f) <= M
        # conv(f,g)[k] = sum_j f[j] * g[k-j]
        # conv(g,f)[k] = sum_j g[j] * f[k-j]
        # So the linear approximation is: conv(f,f)[k] + conv(f,g)[k] + conv(g,f)[k] <= M
        # i.e., 2*conv(f,g)[k] <= M - conv(f,f)[k] (by symmetry of the convolution of f and g? No!)
        # Actually conv(f,g) != conv(g,f) in general.
        # conv(f,g)[k] = sum_j f[j]*g[k-j], conv(g,f)[k] = sum_j g[j]*f[k-j]
        # conv(g,f)[k] = sum_j g[j]*f[k-j] = sum_j f[k-j]*g[j] = conv(f,g)[k]... 
        # Wait: conv(a,b)[k] = sum_j a[j]*b[k-j]. So conv(g,f)[k] = sum_j g[j]*f[k-j].
        # And conv(f,g)[k] = sum_j f[j]*g[k-j].
        # These are different in general. But conv(g,f) = reverse of conv(f,g)? No.
        # Actually, for the full linear convolution: conv(f,g) = conv(g,f) because convolution is commutative!
        # Yes, convolution IS commutative: (f*g)[k] = sum_j f[j]g[k-j] = sum_j g[j]f[k-j] = (g*f)[k].
        # So conv(f,g) = conv(g,f).
        # Therefore: conv(f+g, f+g) ≈ conv(f,f) + 2*conv(f,g) (first-order in g)
        # Constraint: conv(f,f)[k] + 2*conv(f,g)[k] <= M for all k
        # i.e., 2*conv(f,g)[k] <= M - conv(f,f)[k] for all k
        
        rhs = M - h  # rhs[k] = M - conv(f,f)[k] >= 0
        
        # Build A_ub: A_ub[k, j] = 2*f[k-j] if 0 <= k-j < n, else 0
        # This gives: A_ub @ g = 2 * conv(f, g)
        
        row_idx = []
        col_idx = []
        vals = []
        for j in range(n):
            # For each j, g[j] appears in conv(f,g)[k] = sum_j f[j]*g[k-j]
            # g[j] appears when k-j is a valid index of f, i.e., 0 <= k-j < n
            # So k ranges from j to j+n-1
            k_start = j
            k_end = j + n - 1
            # f indices: k-j ranges from 0 to n-1
            for k in range(k_start, k_end + 1):
                row_idx.append(k)
                col_idx.append(j)
                vals.append(2.0 * f[k - j])
        
        # Since f is the same for all offsets, A is a Toeplitz-like matrix
        # A[k, j] = 2*f[k-j] if 0 <= k-j < n
        
        # Actually, this is just 2 * the convolution matrix of f
        # Let me build it more efficiently
        row_idx = []
        col_idx = []
        vals = []
        for j in range(n):
            ks = np.arange(j, j + n)
            row_idx.extend(ks.tolist())
            col_idx.extend([j] * n)
            vals.extend((2.0 * f).tolist())
        
        A_ub = coo_matrix((vals, (row_idx, col_idx)), shape=(2*n-1, n)).tocsr()
        
        # LP: maximize sum(g) subject to A_ub @ g <= rhs, g >= 0
        c_obj = -np.ones(n)
        res = linprog(c_obj, A_ub=A_ub, b_ub=rhs,
                      bounds=[(0, None)] * n, method='highs')
        
        if not res.success or res.x is None:
            print(f"    LP failed at iter {it}: {res.message}")
            break
        
        g = res.x
        Sg = float(np.sum(g))
        if Sg < 1e-10:
            break
        g = g / Sg
        
        # Frank-Wolfe step
        t = 2.0 / (it + 2)
        f_new = (1.0 - t) * f + t * g
        f_n = _normalize(f_new)
        if f_n is None:
            break
        f = f_n
        
        score = _score(f)
        if score < best_score:
            best_score = score
            best_f = f.copy()
        
        if (it + 1) % 5 == 0:
            print(f"    LP it={it+1}: score={best_score:.6f}, t={time.monotonic()-start:.1f}s")
    
    return best_f, best_score

"""
Optimizes non-negative step function heights to minimize 2n*max(conv(f,f))/(sum(f))^2.
Uses accelerated mirror descent with log-sum-exp smoothing, diverse initializations,

import numpy as np
import time

def _fft_convolve(f):
    """Compute full linear convolution f*f using FFT."""
    n = len(f)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    F = np.fft.rfft(f, size)
    h = np.fft.irfft(F * F, size)
    return h[:2 * n - 1]

def _score(f):
    """Compute score = 2n * max(conv(f,f)) / (sum(f))^2."""
    n = len(f)
    h = _fft_convolve(f)
    M = float(np.max(h))
    S = float(np.sum(f))
    if S < 0.01:
        return float('inf')
    return 2.0 * n * M / (S * S)

def _normalize(f):
    """Normalize f to sum to 1, clip to non-negative. Returns None if degenerate."""
    f = np.maximum(f, 0.0)
    s = float(np.sum(f))
    if s < 1e-30:
        return None
    return f / s

def _init_uniform(n):

    start = time.monotonic()
    best_x = x0.copy()
    best_val = objective(x0)
    
    res = minimize(objective, x0, jac=gradient, method='L-BFGS-B',
                   bounds=[(0, None)] * n,
                   options={'maxiter': 200, 'gtol': 1e-8, 'maxfun': 5000})
    
    if res.x is not None:
        val = objective(res.x)
        if val < best_val:
            best_x = res.x.copy()
            best_val = val
    
    # Evaluate actual (non-smoothed) score
    actual_score = _score(best_x)
    return best_x, actual_score

def _lp_frank_wolfe(f0, max_iters, time_budget):
    """LP-based Frank-Wolfe for the normalized problem."""
    from scipy.optimize import linprog
    from scipy.sparse import coo_matrix
    
    f = _normalize(f0.copy()).astype(np.float64)
    n = len(f)
    best_f = f.copy()
    best_score = _score(f)
    start = time.monotonic()
    
    for it in range(max_iters):
        if time.monotonic() - start > time_budget:
            break
        
        h = np.convolve(f, f)
        M = float(np.max(h))
        rhs = M - h
        
        # Build A_ub: A_ub[k,j] = 2*f[k-j] if 0<=k-j<n
        row_idx = []
        col_idx = []
        vals = []
        for j in range(n):
            ks = np.arange(j, j + n)
            row_idx.extend(ks.tolist())
            col_idx.extend([j] * n)
            vals.extend((2.0 * f).tolist())
        
        A_ub = coo_matrix((vals, (row_idx, col_idx)), shape=(2 * n - 1, n)).tocsr()
        
        c_obj = -np.ones(n)
        res = linprog(c_obj, A_ub=A_ub, b_ub=rhs,
                      bounds=[(0, None)] * n, method='highs')
        
        if not res.success or res.x is None:
            break
        
        g = res.x
        Sg = float(np.sum(g))
        if Sg < 1e-10:
            break
        g = g / Sg
        
        t = 2.0 / (it + 2)
        f_new = (1.0 - t) * f + t * g
        f_n = _normalize(f_new)
        if f_n is None:
            break
        f = f_n
        
        score = _score(f)
        if score < best_score:
            best_score = score
            best_f = f.copy()
        
        if (it + 1) % 5 == 0:
            print(f"      LP it={it+1}: score={best_score:.6f}, t={time.monotonic()-start:.1f}s")
    
    return best_f, best_score

def propose_candidate(seed=42, budget_s=1500, **kwargs):
    """
    Search for the best non-negative sequence minimizing 2n*max(conv(f,f))/(sum(f))^2.
    Multi-phase: diverse mirror descent -> L-BFGS refinement -> LP Frank-Wolfe.
    """
    deadline = time.monotonic() + float(budget_s) - 15.0
    rng = np.random.default_rng(seed)
    
    # Get parent
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64).copy()
        parent = np.clip(parent, 0.0, 1000.0)
    except Exception:
        parent = np.ones(512, dtype=np.float64)
    
    best_f = parent.copy()
    best_score = float(evaluate_sequence(best_f.tolist()))
    print(f"Parent: n={len(best_f)}, score={best_score:.6f}")
    
    # Generate diverse initializations
    candidates = []
    p = _normalize(parent)
    if p is not None:
        candidates.append(("parent", p))
    
    for n in [512, 1024, 2048]:
        candidates.append((f"uni_{n}", _init_uniform(n)))
        candidates.append((f"cos_{n}", _init_cosine(n)))
        candidates.append((f"gau30_{n}", _init_gaussian(n, 0.30)))
        candidates.append((f"gau15_{n}", _init_gaussian(n, 0.15)))
        candidates.append((f"rand_{n}", _init_rand(n, rng)))
        candidates.append((f"strip7_{n}", _init_stripe(n, 7, rng)))
        candidates.append((f"bipol_{n}", _init_bipolar_smooth(n, rng)))
    
    print(f"Total candidates: {len(candidates)}")
    
    # Phase 1: Mirror descent on all candidates
    results = []
    remaining = deadline - time.monotonic()
    time_per = max(4.0, min(25.0, remaining / len(candidates) * 0.7))
    
    for idx, (name, f0) in enumerate(candidates):
        if time.monotonic() > deadline - 90:
            break
        n = len(f0)
        n_iter = int(max(200, min(3000, time_per * 250)))
        
        f_md, s_md = _mirror_descent_smoothed(
            f0, n_iter,
            mu_init=0.02, mu_final=0.002,
            eta0=0.5, momentum=0.6,
            time_limit=time_per
        )
        
        print(f"  [{idx+1}/{len(candidates)}] {name} (n={n}): {s_md:.6f}")
        results.append((s_md, name, f_md))
        if s_md < best_score:
            best_score = s_md
            best_f = f_md.copy()
            print(f"  ** New best: {best_score:.6f} ({name}) **")
    
    # Phase 2: Refine top candidates
    results.sort(key=lambda x: x[0])
    top_k = min(4, len(results))
    
    for r in range(top_k):
        if time.monotonic() > deadline - 45:
            break
        s_cur, name, f_cur = results[r]
        n = len(f_cur)
        print(f"\n  Refining {name} (n={n}, score={s_cur:.6f})")
        
        # 2a: More mirror descent with finer mu
        rem = deadline - time.monotonic() - 30
        if rem > 10:
            t_alloc = min(50.0, rem * 0.4)
            f_ref, s_ref = _mirror_descent_smoothed(
                f_cur, int(min(5000, t_alloc * 200)),
                mu_init=0.008, mu_final=0.0005,
                eta0=0.3, momentum=0.7,
                time_limit=t_alloc
            )
            if s_ref < best_score:
                best_score = s_ref
                best_f = f_ref.copy()
                print(f"  ** MD refine: {best_score:.6f} **")
            f_cur = f_ref
            s_cur = s_ref
        
        # 2b: L-BFGS on smoothed objective
        if (deadline - time.monotonic()) > 25:
            t_alloc = min(40.0, deadline - time.monotonic() - 20)
            if t_alloc > 8:
                f_lb, s_lb = _lbfgs_refine(f_cur, t_alloc, mu=0.003)
                if s_lb < best_score:
                    best_score = s_lb
                    best_f = f_lb.copy()
                    print(f"  ** L-BFGS: {best_score:.6f} **")
                f_cur = f_lb
                s_cur = s_lb
        
        # 2c: LP Frank-Wolfe for small sequences
        if len(f_cur) <= 700 and (deadline - time.monotonic()) > 25:
            t_alloc = min(50.0, deadline - time.monotonic() - 15)
            if t_alloc > 12:
                f_lp, s_lp = _lp_frank_wolfe(f_cur, max_iters=20, time_budget=t_alloc)
                if s_lp < best_score:
                    best_score = s_lp
                    best_f = f_lp.copy()
                    print(f"  ** LP: {best_score:.6f} **")
                f_cur = f_lp
                s_cur = s_lp
        
        # 2d: Perturbation restarts
        if (deadline - time.monotonic()) > 15:
            for frac in [0.03, 0.08]:
                if time.monotonic() > deadline - 10:
                    break
                noise = rng.random(len(f_cur)) * frac
                f_p = _normalize(f_cur + noise)
                if f_p is None:
                    continue
                rem_p = deadline - time.monotonic() - 8
                if rem_p < 5:
                    break
                f_p2, s_p2 = _mirror_descent_smoothed(
                    f_p, int(min(2000, rem_p * 150)),
                    mu_init=0.005, mu_final=0.001,
                    eta0=0.3, momentum=0.6,
                    time_limit=rem_p
                )
                if s_p2 < best_score:
                    best_score = s_p2
                    best_f = f_p2.copy()
                    print(f"  ** Perturb({frac}): {best_score:.6f} **")
    
    # Phase 3: Try upsampled versions of best
    if (deadline - time.monotonic()) > 20:
        for new_n in [1024, 2048]:
            if time.monotonic() > deadline - 10:
                break
            if len(best_f) >= new_n:
                continue
            ratio = new_n / len(best_f)
            new_f = np.interp(
                np.linspace(0, len(best_f) - 1, new_n),
                np.arange(len(best_f)),
                best_f
            )
            new_f = _normalize(new_f)
            if new_f is None:
                continue
            rem3 = deadline - time.monotonic() - 8
            if rem3 < 5:
                break
            f_up, s_up = _mirror_descent_smoothed(
                new_f, int(min(3000, rem3 * 120)),
                mu_init=0.01, mu_final=0.001,
                eta0=0.3, momentum=0.6,
                time_limit=rem3
            )
            if s_up < best_score:
                best_score = s_up
                best_f = f_up.copy()
                print(f"  ** Upsampled to {new_n}: {best_score:.6f} **")
    
    # Final verification
    final_list = [float(x) for x in best_f]
    final_score = float(evaluate_sequence(final_list))
    print(f"\nFinal: score={final_score:.6f}, n={len(final_list)}")
    
    if not np.isfinite(final_score) or final_score > 2.0:
        print("  Falling back to parent.")
        return [float(x) for x in parent]
    
    return final_list
