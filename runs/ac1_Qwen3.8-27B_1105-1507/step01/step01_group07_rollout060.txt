```python
import numpy as np
from scipy.optimize import linprog
import time

def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    conv = np.convolve(a, a)
    max_b = float(np.max(conv))
    sum_a = float(np.sum(a))
    if sum_a < 0.01:
        return np.inf
    return 2.0 * n * max_b / (sum_a * sum_a)

def build_A(a):
    """Build constraint matrix for LP: A[k,j] gives (a*g)_k = sum_j A[k,j]*g_j."""
    n = len(a)
    A = np.zeros((2 * n - 1, n), dtype=np.float64)
    for k in range(2 * n - 1):
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        for j in range(j_start, j_end + 1):
            A[k, j] = a[k - j]
    return A

def solve_lp_for_g(a, M_target, upper_bound=1000.0):
    """Solve LP: max sum(g) s.t. (a*g)_k <= M_target for all k, 0 <= g_i <= ub."""
    n = len(a)
    A = build_A(a)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M_target)
    bounds = [(0.0, upper_bound)] * n
    res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
    if res.status == 0:
        return res.x, float(-res.fun)
    return None, None
```

```python
Combines alternating LP, smoothed subgradient descent, and fine-tuning.
"""

from scipy.optimize import linprog
import time

def compute_obj(a):
    n = len(a)
    if n == 0:
        return np.inf
    conv = np.convolve(a, a)
    max_b = float(np.max(conv))
    sum_a = float(np.sum(a))
    if sum_a < 0.01:
        return np.inf
    return 2.0 * n * max_b / (sum_a * sum_a)

def build_A(a):
    n = len(a)
    A = np.zeros((2 * n - 1, n), dtype=np.float64)
    for k in range(2 * n - 1):
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        for j in range(j_start, j_end + 1):
            A[k, j] = a[k - j]
    return A

def solve_lp_for_g(a, M_target, ub=5.0):
    n = len(a)
    A = build_A(a)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M_target)
    bounds = [(0.0, ub)] * n
    res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
    if res.status == 0:
        return res.x, float(-res.fun)
    return None, None

def smooth_subgradient_step(a, temperature=0.01, eta=0.1):
    """Smoothed subgradient step using softmax weighting over lags."""
    n = len(a)
    a = a / float(np.sum(a))
    conv = np.convolve(a, a)
    
    # Softmax weighting
    shifted = conv - np.max(conv)
    weights = np.exp(shifted / temperature)
    weights = weights / np.sum(weights)
    
    # Compute gradient direction
    g = np.zeros(n)
    for k in range(2 * n - 1):
        if weights[k] < 1e-10:
            continue
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        for j in range(j_start, j_end + 1):
            g[j] += weights[k] * a[k - j]
    
    # Update
    a_new = a - eta * g
    a_new = np.maximum(a_new, 0)
    s = np.sum(a_new)
    if s > 1e-10:
        a_new = a_new / s
    return a_new

    a = a_init.copy()
    a = a / float(np.sum(a))
    best_a = a.copy()
    best_obj = compute_obj(a)
    
    for it in range(n_iter):
        a = smooth_subgradient_step(a, temperature, eta)
        obj = compute_obj(a)
        if obj < best_obj:
            best_a = a.copy()
            best_obj = obj
    
    return best_a, best_obj

            a = a_new
        else:
```

```python
"""
Multi-strategy optimization for minimizing 2*n*max(conv(a,a))/(sum(a))^2.
Combines alternating LP, FFT-accelerated smoothed subgradient descent,
and neighborhood search with simulated-annealing-style acceptance.
Exploits parent sequence and explores nearby sequence lengths.
"""

import numpy as np
from scipy.optimize import linprog
import time


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    conv = np.convolve(a, a)
    max_b = float(np.max(conv))
    sum_a = float(np.sum(a))
    if sum_a < 0.01:
        return np.inf
    return 2.0 * n * max_b / (sum_a * sum_a)


def build_A(a):
    """Build constraint matrix for LP: row k has A[k,j] = a[k-j] if 0<=k-j<n."""
    n = len(a)
    A = np.zeros((2 * n - 1, n), dtype=np.float64)
    for k in range(2 * n - 1):
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        for j in range(j_start, j_end + 1):
            A[k, j] = a[k - j]
    return A


def solve_lp_for_g(a, M_target, ub=5.0):
    """LP: max sum(g) s.t. (a*g)_k <= M_target for all k, 0 <= g <= ub."""
    n = len(a)
    A = build_A(a)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M_target)
    bounds = [(0.0, ub)] * n
    res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
    if res.status == 0:
        return res.x, float(-res.fun)
    return None, None


def smooth_subgrad_step(a, temperature, eta):
    """One smoothed subgradient step using FFT-accelerated correlation."""
    n = len(a)
    a = a / float(np.sum(a))
    conv = np.convolve(a, a)
    max_c = float(np.max(conv))
    weights = np.exp((conv - max_c) / temperature)
    wsum = float(np.sum(weights))
    if wsum < 1e-30:
        return a
    weights = weights / wsum

    # g_j = sum_m a[m] * w[j+m] = cross-correlation, computed via FFT
    w_flipped = weights[::-1]
    conv_full = np.convolve(a, w_flipped, mode='full')
    g = conv_full[2 * n - 2::-1][:n]

    a_new = a - eta * g
    a_new = np.maximum(a_new, 0.0)
    s = float(np.sum(a_new))
    if s > 1e-10:
        a_new = a_new / s
    else:
        a_new = a
    return a_new


def run_smooth_subgrad(a_init, n_iter, eta, temperature):
    """Run smoothed subgradient iterations with best-tracking."""
    a = a_init.copy()
    a = a / float(np.sum(a))
    best_a = a.copy()
    best_obj = compute_obj(a)
    for _ in range(n_iter):
        a = smooth_subgrad_step(a, temperature, eta)
        obj = compute_obj(a)
        if obj < best_obj:
            best_obj = obj
            best_a = a.copy()
    return best_a, best_obj


def run_alt_lp(a_init, max_iter, t, M_scale, use_geo, ub):
    """Alternating LP with optional geometric blending."""
    a = a_init.copy()
    a = a / float(np.sum(a))
    best_a = a.copy()
    best_obj = compute_obj(a)
    stall = 0
    for _ in range(max_iter):
        conv = np.convolve(a, a)
        M = float(np.max(conv))
        M_target = M * M_scale
        g, g_sum = solve_lp_for_g(a, M_target, ub=ub)
        if g is None:
            break
        if g_sum <= 1.0 + 1e-3:
            stall += 1
            if stall > 15:
                break
            continue
        stall = 0
        g_sum_val = float(np.sum(g))
        if g_sum_val < 1e-10:
            break
        g_norm = g / g_sum_val
        if use_geo:
            a_safe = np.maximum(a, 1e-15)
            g_safe = np.maximum(g_norm, 1e-15)
            a_new = np.sqrt(a_safe * g_safe)
        else:
            a_new = (1.0 - t) * a + t * g_norm
            a_new = np.maximum(a_new, 0.0)
        s = float(np.sum(a_new))
        if s < 1e-10:
            break
        a_new = a_new / s
        a = a_new
        obj = compute_obj(a)
        if obj < best_obj:
            best_obj = obj
            best_a = a.copy()
    return best_a, best_obj


def neighborhood_search(a, n_iter, rng):
    """Neighborhood search with geometric perturbations and SA-style acceptance."""
    n = len(a)
    best_a = a.copy()
    best_obj = compute_obj(a)
    current_obj = best_obj
    base_noise = 0.04
    for it in range(n_iter):
        progress = it / max(1, n_iter)
        noise = base_noise * (1.0 - 0.5 * progress)
        width = max(1, min(n // 25, 15))
        start = int(rng.integers(0, max(1, n - width + 1)))
        candidate = best_a.copy()
        candidate[start:start + width] *= np.exp(rng.normal(0.0, noise, width))
        candidate = np.maximum(candidate, 0.0)
        s = float(np.sum(candidate))
        if s < 1e-10:
            continue
        candidate = candidate / s
        obj_new = compute_obj(candidate)
        delta = obj_new - current_obj
        accept = False
        if delta < 0:
            accept = True
        else:
            temp = 0.002 * (1.0 - progress) + 0.0001
            if rng.random() < np.exp(-delta / temp):
                accept = True
        if accept:
            current_obj = obj_new
        if obj_new < best_obj:
            best_obj = obj_new
            best_a = candidate.copy()
    return best_a, best_obj


def fine_tune(a, n_iter):
    """Very fine local search with small perturbations."""
    n = len(a)
    rng = np.random.default_rng(88888)
    best_a = a.copy()
    best_obj = compute_obj(a)
    for _ in range(n_iter):
        width = max(1, min(5, n // 40))

        candidate = best_a.copy()
        candidate[start:start + width] *= np.exp(rng.normal(0.0, 0.015, width))
        candidate = np.maximum(candidate, 0.0)
        s = float(np.sum(candidate))
        if s < 1e-10:
            continue
        obj_new = compute_obj(candidate)
        if obj_new < best_obj:
            best_obj = obj_new
            best_a = candidate
    return best_a, best_obj


def structured_init(n, rng):
    """Generate diverse structured initializations."""
    choice = int(rng.integers(0, 6))
    if choice == 0:
        freq = rng.uniform(1.0, 4.0)
        phase = rng.uniform(0.0, 2.0 * np.pi)
        a = np.maximum(0.0, 1.0 + np.cos(2.0 * np.pi * freq * np.arange(n) / n + phase))
    elif choice == 1:
        a = np.ones(n)
        a[1::2] = rng.uniform(0.3, 0.7)
    elif choice == 2:
        period = int(rng.integers(3, max(4, n // 8)))
        pattern = rng.random(period)
        a = np.tile(pattern, n // period + 1)[:n]
    elif choice == 3:
        x = np.arange(n)
        center = n / 2.0
        sigma = n * rng.uniform(0.2, 0.4)
        a = np.exp(-0.5 * ((x - center) / sigma) ** 2)
    elif choice == 4:
        f1 = rng.uniform(1.0, 5.0)
        f2 = rng.uniform(1.0, 5.0)
        p1 = rng.uniform(0.0, 2.0 * np.pi)
        p2 = rng.uniform(0.0, 2.0 * np.pi)
        a = np.maximum(0.0, 1.0 + 0.5 * np.cos(2.0 * np.pi * f1 * np.arange(n) / n + p1)
                        + 0.3 * np.cos(2.0 * np.pi * f2 * np.arange(n) / n + p2))
    else:
        a = rng.random(n)
    s = float(np.sum(a))
    if s < 1e-10:
        a = rng.random(n)
        s = float(np.sum(a))
    return a / s


def resample_sequence(source, target_n):
    """Resample a sequence to a different length using linear interpolation."""
    from scipy.interpolate import interp1d
    n_src = len(source)
    if n_src < 2 or target_n < 2:
        return source[:target_n] if target_n <= n_src else np.pad(source, (0, target_n - n_src))
    x_old = np.linspace(0.0, 1.0, n_src)
    x_new = np.linspace(0.0, 1.0, target_n)
    f = interp1d(x_old, source, kind='linear', fill_value=(source[0], source[-1]),
                 bounds_error=False)
    a_new = f(x_new)
    a_new = np.maximum(a_new, 0.0)
    s = float(np.sum(a_new))
    if s > 1e-10:
        a_new = a_new / s
    return a_new

        iters = int(min(100, max(10, (deadline - time.monotonic()) / 5.0)))
        if iters > 10:
            a_opt, _ = run_alt_lp(a_init, iters, 0.5, 1.0, False, ub=5.0)
            record(a_opt, "altLP n={}".format(n_try))

        # Subgradient
        if time.monotonic() < deadline - 20:
            iters2 = int(min(3000, max(200, (deadline - time.monotonic()) / 0.06)))
            a_sg, _ = run_smooth_subgrad(a_opt, iters2, 0.05, 0.01)
            record(a_sg, "subgrad n={}".format(n_try))

    # --- Phase 5: Combined refinement on best ---
    if best_seq is not None and time.monotonic() < deadline - 20:
        print("Phase 5: Combined refinement")
        a_work = best_seq.copy()
        # Interleave subgradient and NSTS
        for cycle in range(5):
            if time.monotonic() > deadline - 10:
                break
            # Subgradient burst
            iters = int(min(2000, max(100, (deadline - time.monotonic()) / 0.12)))
            a_work, _ = run_smooth_subgrad(a_work, iters, 0.03, 0.01)
            obj_w = compute_obj(a_work)
            if obj_w < best_score:
                best_score = obj_w
                best_seq = a_work.copy()
            # NSTS burst
            if time.monotonic() < deadline - 10:
                iters2 = int(min(3000, max(500, (deadline - time.monotonic()) / 0.04)))
                a_work, _ = neighborhood_search(a_work, iters2, rng)
                obj_w2 = compute_obj(a_work)
                if obj_w2 < best_score:
                    best_score = obj_w2
                    best_seq = a_work.copy()

    # --- Phase 6: Final fine-tuning ---
    if best_seq is not None and time.monotonic() < deadline - 5:
        print("Phase 6: Final fine-tune")
        iters = int(max(100, (deadline - time.monotonic()) / 0.015))
        a_ft, _ = fine_tune(best_seq, iters)
        record(a_ft, "final_fine")

        # Ultra-fine
        if time.monotonic() < deadline - 3:
            iters2 = int(max(50, (deadline - time.monotonic()) / 0.015))
            a_ft2, _ = fine_tune(a_ft, iters2)
            record(a_ft2, "ultra_fine")

    # Fallback
    if best_seq is None:
        best_seq = np.ones(100.0)
        best_score = compute_obj(best_seq)

    elapsed = time.monotonic() - start_time
    print("\nDONE: n={}| score={:.6f}| time={:.1f}s".format(len(best_seq), best_score, elapsed))
    return [float(x) for x in best_seq]
```
