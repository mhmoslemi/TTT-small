Minimizes J(a) = 2*n*max(convolve(a,a))/(sum(a))^2 using:
1. LP-based mirror descent with adaptive step size from parent sequence
2. Smooth log-sum-exp gradient refinement via L-BFGS-B

import numpy as np
from scipy.optimize import linprog, minimize
import time


def fast_conv(a):
    """Full self-convolution via FFT."""
    n = len(a)
    s = 1
    while s < 2 * n - 1:
        s <<= 1
    fa = np.fft.rfft(a, s)
    return np.fft.irfft(fa * fa, s)[:2 * n - 1]


def compute_J(a):
    """Compute J(a) = 2n * max(conv(a,a)) / sum(a)^2."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-8:
        return 1e10
    a_norm = a / s
    b = fast_conv(a_norm)
    return float(2 * len(a_norm) * b.max())


def build_constraint_matrix(a):
    """Build matrix A where A[k,j] = a[k-j] for valid indices."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for k in range(2 * n - 1):
        jl = max(0, k - n + 1)
        jh = min(n - 1, k)
        if jl <= jh:
            A[k, jl:jh+1] = a[jh-k+jl:jh+1][::-1]
    return A


def lp_solve(a, M):
    """Solve LP: max sum(g) s.t. (a*g)_k <= M, g >= 0."""
    n = len(a)
    c = -np.ones(n)
    A = build_constraint_matrix(a)
    r = linprog(c, A_ub=A, b_ub=np.full(2 * n - 1, M),
               bounds=[(0, None)] * n, method='highs')
    if r.success:
        g = r.x
        if g.sum() > 1e-12:
            return g
    return None


def smooth_obj(x, beta=20.0):
    """Smooth objective using log-sum-exp."""
    a = np.maximum(x, 0.0)
    s = a.sum()
    if s < 1e-8:
        return 1e10
    a = a / s
    b = fast_conv(a)
    # log-sum-exp for numerical stability
    b_max = b.max()
    lse = b_max + np.log(np.sum(np.exp(beta * (b - b_max))))
    n = len(a)
    return float(2 * n * lse / (beta * s * s))


def smooth_grad(x, beta=20.0):
    """Gradient of smooth objective (approximate via finite differences is too slow,
    so we compute analytically)."""
    a = np.maximum(x, 0.0)
    s = a.sum()
    if s < 1e-8:
        return np.zeros_like(x)
    a = a / s
    b = fast_conv(a)
    b_max = b.max()
    weights = np.exp(beta * (b - b_max))
    weights /= weights.sum()
    n = len(a)
    
    # Gradient of (a*a)_k with respect to a_m is 2*a_{k-m}
    # So gradient of sum_k w_k * (a*a)_k is sum_k w_k * 2*a_{k-m}
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        if weights[k] > 1e-15:
            for m in range(n):
                idx = k - m
                if 0 <= idx < n:
                    grad[m] += 2.0 * weights[k] * a[idx]
    
    # Chain rule for normalization: a = x/s
    # This gets complicated. Let's just use the unnormalized version.
    return 2.0 * grad


def lp_mirror_descent(a, max_iters=200, deadline=None):

    n = len(a)
    a = a.copy()
    a = np.maximum(a, 1e-6)
    a /= a.sum()
    
    best_a = a.copy()
    best_j = compute_J(a)
    
    for it in range(max_iters):
        if deadline is not None and time.monotonic() > deadline:

        b = fast_conv(a)
        M = float(b.max())
        
        g = lp_solve(a, M)
        if g is None:

        g = np.maximum(g, 0)
        g_sum = g.sum()
        if g_sum < 1e-10:
            break
        g /= g_sum
        
        # Adaptive mixing: start large, decrease
        t = max(0.1, 0.8 * (1.0 - it / max_iters))
        
        a_new = (1.0 - t) * a + t * g
        a_new = np.maximum(a_new, 0)
        s = a_new.sum()
        if s < 1e-10:

        a_new /= s
        
        j_new = compute_J(a_new)
        if j_new < best_j:
            best_j = j_new
            best_a = a_new.copy()
            a = a_new
        else:

            a_new = (1.0 - t) * a + t * g
            a_new = np.maximum(a_new, 0)
            s = a_new.sum()
            if s < 1e-10:
                break
            a_new /= s
            a = a_new

def gradient_refine(a, deadline=None, beta=20.0):
    """L-BFGS-B refinement with smooth log-sum-exp objective."""
    n = len(a)
    a0 = np.maximum(a, 0.0).copy()
    
    def obj_and_grad(x):
        ax = np.maximum(x, 0.0)
        s = ax.sum()
        if s < 1e-8:
            return 1e10, np.zeros_like(x)
        ax = ax / s
        b = fast_conv(ax)
        b_max = b.max()
        weights = np.exp(beta * (b - b_max))
        wsum = weights.sum()
        lse = b_max + np.log(wsum)
        val = 2 * n * lse / (beta)  # normalized by s^2=1
        
        # Gradient with respect to x (through ax = ax/s)
        # d(ax)/dx_i = (δ_i - ax_i/s) / s ... complicated
        # Just use subgradient of max(b) as approximation
        k_star = int(np.argmax(b))
        grad = np.zeros(n)
        for m in range(n):
            idx = k_star - m
            if 0 <= idx < n:
                grad[m] = 2.0 * ax[idx]
        # Adjust for normalization
        grad = (grad - ax * (grad @ ax / s)) / s
        
        return float(val), grad
    
    res = minimize(obj_and_grad, a0, method='L-BFGS-B',
                   bounds=[(0, None)] * n,
                   options={'maxiter': 500, 'max_fun': 20000,
                            'ftol': 1e-12, 'gtol': 1e-8, 'disp': False})
    
    if res.fun < 1e10:
        a_result = np.maximum(res.x, 0)
        s = a_result.sum()
        if s > 1e-8:
            a_result /= s
            return a_result, compute_J(a_result)
    
    return a, compute_J(a)

def smooth_objective_and_grad(x, beta):
    """Smooth log-sum-exp objective and gradient."""
    n = len(x)
    a = np.maximum(x, 0.0)
    s = a.sum()
    if s < 1e-8:
        return 1e10, np.zeros(n)
    a = a / s
    b = fast_conv(a)
    bmax = b.max()
    eb = np.exp(beta * (b - bmax))
    w = eb / eb.sum()
    lse = bmax + np.log(eb.sum()) / beta
    val = 2.0 * n * lse
    
    # Gradient: d/dx_m of [2n * lse]
    # = 2n * sum_k w_k * d(b_k)/dx_m
    # d(b_k)/dx_m = d(a*a)_k / dx_m where a = x/s
    # = (delta_{m,i} * a_{k-i} + a_i * delta_{k-i,m}) / s  [summed over i]
    # = (a_{k-m} + a_{k-m}) / s = 2*a_{k-m}/s  (when valid)
    # But we also need to account for the 1/s normalization of a = x/s
    # Actually since we normalized a = x/s and s=1 after normalization,
    # the gradient with respect to x is:
    # d(val)/dx_m = 2n * sum_k w_k * d(b_k)/dx_m
    # where b_k = sum_i (x_i/s)*(x_{k-i}/s) = (1/s^2) * sum_i x_i * x_{k-i}
    # d(b_k)/dx_m = (1/s^2)*(2*x_{k-m}) - (2/s^3)*(b_k)*x_m  [when valid indices]
    # Hmm this is getting complex. Let me just use the simpler form:
    # Since a = x/s and we want d/dx_m of f(a(x)):
    # f = 2n * lse(beta * (a*a))
    # df/da_m = 2n * sum_k w_k * 2*a_{k-m}  (subgradient-like)
    # da_m/dx_m = 1/s - a_m/s^2
    # da_i/dx_m = -a_i/s^2 for i != m
    # So df/dx_m = sum_j (df/da_j)(da_j/dx_m)
    #            = (df/da_m)(1/s - a_m/s^2) + sum_{j≠m} (df/da_j)(-a_j/s^2)
    #            = (df/da_m)/s - (df/da_m)*a_m/s^2 - sum_j (df/da_j)*a_j/s^2 + (df/da_m)*a_m/s^2
    #            = (df/da_m)/s - (sum_j df/da_j * a_j)/s^2
    
    # df/da_j = 4n * sum_k w_k * a_{k-j}
    grad_a = np.zeros(n)
    for k in range(2 * n - 1):
        if w[k] > 1e-15:
            for m in range(n):
                idx = k - m
                if 0 <= idx < n:
                    grad_a[m] += 2.0 * w[k] * a[idx]
    grad_a *= 2.0 * n  # = 4n * sum_k w_k * a_{k-j}
    
    # Now df/dx_m = grad_a[m]/s - (grad_a @ a)/s^2
    grad_x = grad_a / s - np.dot(grad_a, a) / (s * s)
    
    return float(val), grad_x


def gradient_refine(a, deadline=None, beta=30.0, maxiter=300):
    """L-BFGS-B with smooth objective."""
    n = len(a)
    
    def objfun(x):
        val, _ = smooth_objective_and_grad(x, beta)
        return val
    
    def objgrad(x):
        val, g = smooth_objective_and_grad(x, beta)
        return val, g
    
    a0 = np.maximum(a, 0.0).copy()
    res = minimize(objfun, a0, jac=objgrad, method='L-BFGS-B',
                   bounds=[(0, None)] * n,
                   options={'maxiter': maxiter, 'max_fun': 50000,
                            'ftol': 1e-14, 'gtol': 1e-10, 'disp': False})
    if res.fun < 1e10:
        a_r = np.maximum(res.x, 0)
        s = a_r.sum()
        if s > 1e-8:
            a_r /= s
            return a_r, compute_J(a_r)
    return a, compute_J(a)

"""
Minimizes J(a) = 2*n*max(convolve(a,a))/(sum(a))^2 using:
1. Active-set LP mirror descent with line-search adaptive mixing
2. Log-sum-exp smooth gradient refinement via L-BFGS-B
3. Multi-length exploration with parent interpolation
4. Randomized active sets for escaping local optima
5. Final intensive polishing phase

import numpy as np
from scipy.optimize import linprog, minimize
import time


def fast_conv(a):
    """Full linear self-convolution via FFT, O(n log n)."""
    n = len(a)
    s = 1
    while s < 2 * n - 1:
        s <<= 1
    fa = np.fft.rfft(a, s)
    return np.fft.irfft(fa * fa, s)[:2 * n - 1]


def compute_J(a):
    """J(a) = 2n * max(conv(a,a)) / sum(a)^2"""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-8:
        return 1e10
    a_n = a / s
    b = fast_conv(a_n)
    return float(2 * len(a_n) * b.max())


def build_A_rows(a, rows):
    """Build rows of the convolution constraint matrix A[k,j] = a[k-j]."""
    n = len(a)
    A = np.zeros((len(rows), n))
    for i, k in enumerate(rows):
        jl = max(0, k - n + 1)
        jh = min(n - 1, k)
        if jl <= jh:
            A[i, jl:jh + 1] = a[k - jh:k - jl + 1][::-1]
    return A


def solve_lp_active(a, M, n_active, rng=None):
    """LP: max 1'g s.t. (a*g)_k <= M for active k, g >= 0.
    Uses top n_active convolution values plus optionally random ones."""
    n = len(a)
    b = fast_conv(a)
    top_idx = np.argsort(b)[-n_active:]
    kmax = int(np.argmax(b))
    if kmax not in top_idx:
        top_idx = np.concatenate([top_idx, [kmax]])
    # Add some random constraints for diversity
    if rng is not None and n_active < (2 * n - 1):
        n_random = max(5, n_active // 10)
        remaining = np.setdiff1d(np.arange(2 * n - 1), top_idx)
        if len(remaining) > 0:
            rand_idx = rng.choice(remaining, size=min(n_random, len(remaining)), replace=False)
            top_idx = np.concatenate([top_idx, rand_idx])
    A = build_A_rows(a, top_idx)
    c = -np.ones(n)
    res = linprog(c, A_ub=A, b_ub=np.full(len(top_idx), M),
                  bounds=[(0, None)] * n, method='highs')
    if res.success and res.x.sum() > 1e-12:
        return np.maximum(res.x, 0)
    return None


def lp_mirrordesc(a, max_iters, deadline, n_active, rng=None):
    """LP mirror descent with line-search mixing."""
    n = len(a)
    a = np.maximum(a.astype(np.float64), 1e-6)
    a /= a.sum()
    best_a = a.copy()
    best_j = compute_J(a)
    stall_count = 0

    for it in range(max_iters):
        if deadline is not None and time.monotonic() > deadline:
            break
        b = fast_conv(a)
        M = float(b.max())
        g = solve_lp_active(a, M, n_active, rng)
        if g is None:
            break
        gs = g.sum()
        if gs < 1e-10:
            break
        g /= gs

        # Line search for optimal mixing parameter
        best_t = 0.05
        best_jt = best_j
        t_candidates = [0.8, 0.5, 0.3, 0.2, 0.1, 0.05, 0.02]
        for t in t_candidates:
            a_try = (1.0 - t) * a + t * g
            a_try = np.maximum(a_try, 0.0)
            ts = a_try.sum()
            if ts < 1e-10:
                continue
            a_try /= ts
            jt = compute_J(a_try)
            if jt < best_jt:
                best_jt = jt
                best_t = t

        a_new = (1.0 - best_t) * a + best_t * g
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s < 1e-10:
            break
        a_new /= s
        j_new = compute_J(a_new)
        if j_new < best_j:
            best_j = j_new
            best_a = a_new.copy()
            stall_count = 0
        else:
            stall_count += 1
            if stall_count > 50:
                break
        a = a_new

    return best_a, best_j


def smooth_obj_grad(x, beta):
    """Smooth log-sum-exp objective and gradient for optimization."""
    n = len(x)
    a = np.maximum(x, 0.0)
    s = a.sum()
    if s < 1e-8:
        return 1e10, np.zeros(n)
    a = a / s
    b = fast_conv(a)
    bmax = b.max()
    eb = np.exp(beta * (b - bmax))
    w = eb / eb.sum()
    lse = bmax + np.log(eb.sum()) / beta
    val = 2.0 * n * lse

    # grad_a[m] = 4n * sum_k w[k] * a[k-m]
    # Using cross-correlation: np.correlate(w, a, 'valid')[m] = sum_j w[m+j]*a[j]
    grad_base = np.correlate(w, a, mode='valid')
    grad_a = 4.0 * n * grad_base

    # Convert to gradient w.r.t. x (before normalization)
    # d(a)/dx: a = x/s, so da_m/dx_m = 1/s - a_m/s^2, da_i/dx_m = -a_i/s^2
    # df/dx_m = grad_a[m]/s - (grad_a . a)/s^2
    grad_x = grad_a / s - np.dot(grad_a, a) / (s * s)

    return float(val), grad_x


def gradient_refine(a, deadline, beta, maxiter):
    """L-BFGS-B refinement with smooth log-sum-exp objective."""
    n = len(a)

    def objfun(x):
        val, _ = smooth_obj_grad(x, beta)
        return val

    def objgrad(x):
        val, g = smooth_obj_grad(x, beta)
        return val, g

    a0 = np.maximum(a, 0.0).copy()
    res = minimize(objfun, a0, jac=objgrad, method='L-BFGS-B',
                   bounds=[(0, None)] * n,
                   options={'maxiter': maxiter, 'max_fun': 50000,
                            'ftol': 1e-14, 'gtol': 1e-10, 'disp': False})
    if res.fun < 1e10:
        a_r = np.maximum(res.x, 0.0)
        s = a_r.sum()
        if s > 1e-8:
            a_r /= s
            return a_r, compute_J(a_r)
    return a, compute_J(a)


def propose_candidate(seed=42, budget_s=1500, **kwargs):
    """
    Main search function combining LP mirror descent, gradient refinement,
    multi-length exploration, and final polishing.
    """
    rng = np.random.default_rng(seed)
    t_start = time.monotonic()
    deadline = t_start + float(budget_s) - 15.0
    best = None
    best_score = float('inf')

    def _try(a):
        nonlocal best, best_score
        a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
        if a.size == 0 or a.sum() < 0.01:
            return
        sc = float(evaluate_sequence(a.tolist()))
        if np.isfinite(sc) and sc < best_score:
            best = a.copy()
            best_score = sc
            elapsed = time.monotonic() - t_start
            print(f"  NEW BEST: J={sc:.6f} n={a.size} t={elapsed:.0f}s", flush=True)

    # Load parent sequence
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        if p.ndim == 1 and p.size > 0 and np.all(np.isfinite(p)):
            parent = np.maximum(p, 0.0)
            ps = parent.sum()
            if ps > 1e-8:
                parent /= ps
            _try(parent)
    except (NameError, AttributeError):
        pass

    if parent is None:
        parent = np.ones(512, dtype=np.float64)
        _try(parent)

    n_p = len(parent)

    # === Phase 1: LP Mirror Descent from parent (budget: 500s) ===
    print("\n=== Phase 1: LP Mirror Descent ===", flush=True)
    p1_deadline = min(deadline, t_start + 500.0)
    t0 = time.monotonic()

    # Primary run from parent with full active set
    a1, j1 = lp_mirrordesc(parent, max_iters=500, deadline=p1_deadline,
                           n_active=100, rng=rng)
    _try(a1)
    print(f"  Parent LP: J={j1:.6f} ({time.monotonic()-t0:.1f}s)", flush=True)

    # Perturbed starts
    for ps in range(6):
        if time.monotonic() > p1_deadline:
            break
        t0 = time.monotonic()
        noise_scale = 0.15 + 0.1 * ps
        a_pert = parent * (1.0 + noise_scale * rng.standard_normal(n_p))
        a_pert = np.maximum(a_pert, 1e-4)
        a_pert /= a_pert.sum()
        a_r, j_r = lp_mirrordesc(a_pert, max_iters=250, deadline=p1_deadline,
                                 n_active=80, rng=rng)
        _try(a_r)
        print(f"  Perturb {ps}: J={j_r:.6f} ({time.monotonic()-t0:.1f}s)", flush=True)

    print(f"  Phase 1 done: {time.monotonic()-t0:.1f}s, best={best_score:.6f}", flush=True)

    # === Phase 2: Gradient Refinement (budget: 400s) ===
    print("\n=== Phase 2: Gradient Refinement ===", flush=True)
    p2_deadline = min(deadline, t_start + 900.0)
    t0 = time.monotonic()

    base = best if best is not None else parent
    for beta in [20.0, 40.0, 80.0]:
        if time.monotonic() > p2_deadline:
            break
        t0b = time.monotonic()
        a_r, j_r = gradient_refine(base, deadline=p2_deadline, beta=beta, maxiter=400)
        _try(a_r)
        print(f"  Grad beta={beta:.0f}: J={j_r:.6f} ({time.monotonic()-t0b:.1f}s)", flush=True)
        if best is not None:
            base = best.copy()

    # Gradient from perturbed best
    for ps in range(4):
        if time.monotonic() > p2_deadline:
            break
        a_pert = base * (1.0 + 0.1 * rng.standard_normal(len(base)))
        a_pert = np.maximum(a_pert, 1e-4)
        a_pert /= a_pert.sum()
        a_r, j_r = gradient_refine(a_pert, deadline=p2_deadline, beta=40.0, maxiter=300)
        _try(a_r)

    print(f"  Phase 2 done: {time.monotonic()-t0:.1f}s, best={best_score:.6f}", flush=True)

    # === Phase 3: Multi-length exploration (budget: 300s) ===
    print("\n=== Phase 3: Multi-length ===", flush=True)
    p3_deadline = min(deadline, t_start + 1200.0)
    t0 = time.monotonic()

    base = best if best is not None else parent
    for n_try in [384, 640, 768, 1024]:
        if time.monotonic() > p3_deadline:
            break
        t0b = time.monotonic()
        # Interpolate best to new length
        t_old = np.linspace(0, 1, len(base))
        t_new = np.linspace(0, 1, n_try)
        a_interp = np.interp(t_new, t_old, base)
        a_interp = np.maximum(a_interp, 0.0)
        if a_interp.sum() < 1e-8:
            continue
        a_interp /= a_interp.sum()

        a_r, j_r = lp_mirrordesc(a_interp, max_iters=120, deadline=p3_deadline,
                                 n_active=60, rng=rng)
        _try(a_r)

        if time.monotonic() < p3_deadline:
            a_r2, j_r2 = gradient_refine(a_r, deadline=p3_deadline, beta=30.0, maxiter=200)
            _try(a_r2)
        print(f"  n={n_try}: best J so far={best_score:.6f} ({time.monotonic()-t0b:.1f}s)", flush=True)

    print(f"  Phase 3 done: {time.monotonic()-t0:.1f}s, best={best_score:.6f}", flush=True)

    # === Phase 4: Final intensive polish (remaining budget) ===
    print("\n=== Phase 4: Final Polish ===", flush=True)
    t0 = time.monotonic()

    if best is not None and time.monotonic() < deadline:
        n = len(best)
        if n <= 1024:
            # Intensive LP with larger active set
            a, j = lp_mirrordesc(best, max_iters=300, deadline=deadline,
                                 n_active=150, rng=rng)
            _try(a)

            # Gradient polish on result
            if time.monotonic() < deadline:
                a2, j2 = gradient_refine(a, deadline=deadline, beta=60.0, maxiter=300)
                _try(a2)

            # One more LP pass
            if time.monotonic() < deadline:
                base_for_polish = best if best is not None else a
                a3, j3 = lp_mirrordesc(base_for_polish, max_iters=200, deadline=deadline,
                                       n_active=120, rng=rng)
                _try(a3)

    print(f"  Phase 4 done: {time.monotonic()-t0:.1f}s, best={best_score:.6f}", flush=True)

    if best is None:
        best = np.ones(512)

    print(f"\nFINAL: J={best_score:.6f}, n={best.size}", flush=True)
    return [float(x) for x in best]
