def simplex_project(a):
    """Project onto the simplex: a >= 0, sum(a) = 1."""
    n = len(a)
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-10:
        return np.ones(n) / n
    return a / s


def gradient_descent(a, max_iters=10000, lr_init=0.01, beta=5.0, time_budget=60):
    """Projected gradient descent on the simplex."""
    n = len(a)
    a = a.copy()
    a = simplex_project(a)
    
    best_a = a.copy()
    best_score = compute_score_fast(a)
    lr = lr_init
    start = time.monotonic()
    
    for it in range(max_iters):
        if time.monotonic() - start > time_budget:
            break
        
        score, grad, conv, M = score_and_grad(a, beta=beta)
        
        if not np.isfinite(score):
            break
        
        # Gradient step
        a_new = a - lr * grad
        a_new = simplex_project(a_new)
        
        new_score = compute_score_fast(a_new)
        
        if new_score < best_score:
            best_a = a_new.copy()
            best_score = new_score
            a = a_new
            # Adaptive step size: increase if improving
            lr = min(lr * 1.05, 0.1)
        else:
            lr *= 0.8
            if lr < 1e-6:
                # Reset with a small perturbation
                a_pert = a + np.random.default_rng(it).normal(0, 0.01, n)
                a_pert = simplex_project(a_pert)
                if compute_score_fast(a_pert) < best_score:
                    a = a_pert
                    best_a = a.copy()
                    best_score = compute_score_fast(a)
                    lr = lr_init * 0.5
                else:
                    if it > 100:
                        break
    
    return best_a, best_score


def lp_optimize_step(a, M):
    """One LP step: find direction g that maximizes sum(g) subject to convolve(a,g) <= M, g >= 0.
    Returns the rescaled g (sum=1) or None if infeasible."""
    n = len(a)
    
    # Build the constraint matrix A such that A @ g = convolve(a, g)
    # A[k, j] = a[k-j] if 0 <= k-j < n, else 0
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        j_start = max(0, k - n + 1)
        j_end = min(n, k + 1)
        for j in range(j_start, j_end):
            rows.append(k)
            cols.append(j)
            vals.append(a[k - j])
    
    A = sp.csr_matrix((vals, (rows, cols)), shape=(2*n-1, n))
    
    c_obj = -np.ones(n)
    b_ub = np.full(2*n-1, M)
    bounds = [(0, None)] * n
    
    try:
        res = linprog(c_obj, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
    except Exception:
        return None
    
    if not res.success:
        return None
    
    g = np.clip(res.x, 0, None)
    gs = g.sum()
    if gs < 1e-15:
        return None
    
    return g / gs


def lp_optimize(a, time_budget=100, t0=0.3, rng=None):

"""
Multi-stage optimization combining projected gradient descent and LP-based
direction search to minimize 2*n*max(convolve(a,a))/sum(a)^2.
Uses FFT-based convolution for speed, simplex projection for non-negativity,
and multiple sequence lengths with adaptive time allocation.
"""

import numpy as np
from scipy.optimize import linprog
from scipy.fft import fft, ifft
import time
import scipy.sparse as sp


def convolve_fft(a, b=None):
    """Full convolution using FFT."""
    if b is None:
        b = a
    n_out = len(a) + len(b) - 1
    size = 1 << (n_out - 1).bit_length()
    return np.real(ifft(fft(a, size) * fft(b, size)))[:n_out]


def compute_score(a):
    """Score = 2*n*max(conv)/sum^2. Assumes a is non-negative."""
    n = len(a)
    S = a.sum()
    if S < 1e-10:
        return np.inf
    conv = np.convolve(a, a)
    M = conv.max()
    return 2.0 * n * M / (S * S)


def normalize(a):
    """Normalize to sum=1, non-negative."""
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(len(a)) / len(a)
    return a / s


def grad_smoothed_max_conv(a, beta=5.0):
    """
    Compute gradient of log-sum-exp(beta * conv) w.r.t. a.
    This approximates the gradient of max(conv) for large beta.
    Returns (score, grad) where score uses actual max.
    """
    n = len(a)
    conv = np.convolve(a, a)
    M = conv.max()
    
    # Softmax weights
    shifted = conv - M
    exp_vals = np.exp(beta * shifted)
    weights = exp_vals / exp_vals.sum()
    
    # grad_j = sum_k weights[k] * a[k-j] (for valid k-j in [0, n-1])
    # = cross-correlation of 'weights' and 'a' at shift j
    # R[j] = sum_i a[i] * weights[j+i]
    
    # Use FFT for cross-correlation
    L = 1 << (3 * n - 2).bit_length()  # enough for linear cross-correlation
    fa = fft(a, L)
    fw = fft(weights, L)
    R = np.real(ifft(np.conj(fa) * fw))
    grad = R[:n]
    
    score = 2.0 * n * M  # normalized (sum=1)
    return score, grad


def gd_optimize(a, time_budget=30, beta=5.0, lr_init=0.05, rng=None):

    best_score = compute_score(a)

    start = time.monotonic()
    it = 0
    stall = 0

        score, grad = grad_smoothed_max_conv(a, beta)

        a_new = a - lr * grad
        a_new = normalize(a_new)

            best_score = new_score
            a = a_new
            stall = 0
            lr = min(lr * 1.01, 0.2)

            stall += 1

def build_conv_matrix(a):
    """Build sparse matrix A s.t. A @ g = convolve(a, g)."""
    n = len(a)
    k = np.arange(2 * n - 1)[:, None]
    j = np.arange(n)[None, :]
    idx = k - j
    mask = (idx >= 0) & (idx < n)
    rows_idx, cols_idx = np.where(mask)
    vals = a[idx[rows_idx, cols_idx]]
    return sp.csr_matrix((vals, (rows_idx, cols_idx)), shape=(2*n-1, n))


def lp_step(a, M):
    """One LP step. Returns rescaled direction g (sum=1) or None."""
    n = len(a)
    A = build_conv_matrix(a)
    c_obj = -np.ones(n)
    b_ub = np.full(2*n-1, M)
    bounds = [(0, None)] * n
    try:
        res = linprog(c_obj, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
    except Exception:
        return None
    if not res.success:
        return None
    g = np.clip(res.x, 0, None)
    gs = g.sum()
    if gs < 1e-15:
        return None
    return g / gs

        g = lp_step(a, M)

        new_score = compute_score(a_new)

            a = a_new

                    best_a = a.copy()
                    best_score = compute_score(a)
