def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:

    conv = np.convolve(a, a)
    max_b = float(np.max(conv))
    sum_a = float(np.sum(a))

    return 2.0 * n * max_b / (sum_a * sum_a)


def build_A(a):
    """Build constraint matrix A where (a*g)_k = sum_j A[k,j] * g_j."""
    n = len(a)
    A = np.zeros((2 * n - 1, n), dtype=np.float64)
    for k in range(2 * n - 1):
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        for j in range(j_start, j_end + 1):
            A[k, j] = a[k - j]
    return A


def solve_lp_joint(a, upper_bound=1000.0):
    """
    Solve LP: min M s.t. (a*g)_k <= M for all k, sum(g)=1, 0<=g<=ub.
    Returns (g, M) or (None, None) if infeasible.
    """
    n = len(a)
    A = build_A(a)
    num_constraints = 2 * n - 1
    
    # Variables: g_0,...,g_{n-1}, M (total n+1)
    # Objective: minimize M
    c = np.zeros(n + 1)
    c[-1] = 1.0  # coefficient of M
    
    # Inequality constraints: (a*g)_k - M <= 0 for each k
    # [A, -I_num_constraints] @ [g; M] <= 0
    A_ub = np.zeros((num_constraints, n + 1))
    A_ub[:, :n] = A
    for i in range(num_constraints):
        A_ub[i, i] = -1.0
    b_ub = np.zeros(num_constraints)

    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])
    
    # Bounds: 0 <= g_i <= upper_bound, M >= 0
    bounds = [(0.0, upper_bound)] * n + [(0.0, None)]

                  bounds=bounds, method='highs')
    
    if res.status == 0:
        g = res.x[:n]
        M = res.x[n]
        return g, M
    return None, None

"""
Multi-strategy optimization for minimizing 2*n*max(conv(a,a))/(sum(a))^2.
Combines joint LP (min M given a), subgradient descent with momentum,
adaptive local search, and diverse restarts across multiple sequence lengths.
"""

import numpy as np
from scipy.optimize import linprog
import time


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    conv = np.convolve(a, a)
    max_b = float(np.max(conv))
    sum_a = float(np.sum(a))
    if sum_a < 0.01:
        return np.inf
    return 2.0 * n * max_b / (sum_a * sum_a)


def build_A(a):
    """Build constraint matrix A where (a*g)_k = sum_j A[k,j] * g_j."""
    n = len(a)
    A = np.zeros((2 * n - 1, n), dtype=np.float64)
    for k in range(2 * n - 1):
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        for j in range(j_start, j_end + 1):
            A[k, j] = a[k - j]
    return A


def solve_lp_joint(a):
    """
    Solve LP: min M s.t. (a*g)_k <= M for all k, sum(g)=1, g>=0.
    Returns (g, M) or (None, None) if infeasible.
    """
    n = len(a)
    A = build_A(a)
    num_con = 2 * n - 1

    c = np.zeros(n + 1)
    c[-1] = 1.0

    A_ub = np.zeros((num_con, n + 1))
    A_ub[:, :n] = A
    for i in range(num_con):
        A_ub[i, i] = -1.0
    b_ub = np.zeros(num_con)

    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])

    bounds = [(0.0, None)] * n + [(0.0, None)]

    res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                  bounds=bounds, method='highs')

    if res.status == 0:
        g = res.x[:n]
        M = res.x[n]
        g_sum = np.sum(g)
        if g_sum > 1e-10:
            g = g / g_sum
        return g, M
    return None, None


def solve_lp_max_sum(a, M):
    """Solve LP: max sum(g) s.t. (a*g)_k <= M, g>=0."""
    n = len(a)
    A = build_A(a)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
    if res.status == 0:
        g = res.x
        g_sum = float(-res.fun)
        if g_sum > 1e-10:
            return g / g_sum, g_sum
    return None, None

                    a = a / s
                stall_count = 0

        prev_obj = compute_obj(a)

    return best_a, best_obj


def subgradient_optimize(a_init, max_iter=5000, verbose=False):
    """
    Subgradient descent with momentum for the non-smooth max objective.
    Tracks the top-k active constraints for smoothing.
    """
    n = len(a_init)
    a = a_init.copy()
    s = np.sum(a)
    if s < 1e-10:
        a = np.ones(n) / n
    else:
        a = a / s

    best_a = a.copy()
    best_obj = compute_obj(best_a)

    momentum = np.zeros(n)
    beta = 0.9
    lr = 0.01
    top_k = min(5, 2 * n - 1)

    for it in range(max_iter):
        conv = np.convolve(a, a)
        sum_a = np.sum(a)
        max_b = np.max(conv)
        obj = 2.0 * n * max_b / (sum_a * sum_a)

        if obj < best_obj:
            best_obj = obj
            best_a = a.copy()
            if verbose and (it % 500 == 0 or it < 5):
                print(f"    sg it={it}: obj={best_obj:.8f}")

        # Find top-k indices of convolution
        idx = np.argsort(conv)[-top_k:]
        weights = np.zeros(2 * n - 1)
        top_vals = conv[idx]
        max_top = np.max(top_vals)
        exp_vals = np.exp(50.0 * (top_vals - max_top))
        w_sum = np.sum(exp_vals)
        if w_sum > 0:
            for i, k in enumerate(idx):
                weights[k] = exp_vals[i] / w_sum

        # Compute gradient of weighted max convolution
        # d(sum_k w_k * (a*a)_k) / da_i = 2 * sum_k w_k * a_{k-i} (if valid)
        grad_conv = np.zeros(n)
        for k in idx:
            wk = weights[k]
            for i in range(n):
                ki = k - i
                if 0 <= ki < n:
                    grad_conv[i] += 2.0 * wk * a[ki]

        # Full gradient of f(a) = 2n * S(a) / (sum(a))^2
        # df/da_i = 2n * (grad_S_i * sum_a - 2*S*a_i) / sum_a^3
        S = np.sum(weights * conv)  # weighted max approx
        grad = 2.0 * n * (grad_conv * sum_a - 2.0 * S * a) / (sum_a ** 3)

        # Nesterov momentum
        momentum = beta * momentum - lr * grad
        a = a + momentum
        a = np.clip(a, 0, None)
        s = np.sum(a)
        if s < 1e-10:
            a = best_a.copy()
            continue
        # Normalize to keep sum = 1
        a = a / s

        # Decay learning rate slowly
        if it > max_iter * 0.5:
            lr = max(0.001, lr * 0.999)
        beta = min(0.99, 0.9 + 0.09 * it / max_iter)

    return best_a, best_obj


def local_search(a_init, max_iter=20000, verbose=False):
    """
    Local search with diverse perturbation strategies.
    """
    best_a = a_init.copy()
    best_obj = compute_obj(best_a)
    n = len(best_a)
    rng = np.random.default_rng(12345 + n)
    cur_obj = best_obj

    for it in range(max_iter):
        strategy = rng.integers(0, 5)

        if strategy == 0:
            # Small block
            width = max(1, n // 100)
            start = int(rng.integers(0, n - width + 1))
            candidate = best_a.copy()
            candidate[start:start + width] *= np.exp(rng.normal(0, 0.08, width))
        elif strategy == 1:
            # Single element
            idx = int(rng.integers(0, n))
            candidate = best_a.copy()
            candidate[idx] = max(0, candidate[idx] * np.exp(rng.normal(0, 0.15)))
        elif strategy == 2:
            # Two blocks
            w1 = max(1, n // 150)
            w2 = max(1, n // 150)
            s1 = int(rng.integers(0, n - w1 + 1))
            s2 = int(rng.integers(0, n - w2 + 1))
            candidate = best_a.copy()
            candidate[s1:s1 + w1] *= np.exp(rng.normal(0, 0.06, w1))
            candidate[s2:s2 + w2] *= np.exp(rng.normal(0, 0.06, w2))
        elif strategy == 3:
            # Very small global
            candidate = best_a * np.exp(rng.normal(0, 0.02, n))
        else:
            # Swap-like: redistribute mass between two positions
            i1 = int(rng.integers(0, n))
            i2 = int(rng.integers(0, n))
            candidate = best_a.copy()
            total = candidate[i1] + candidate[i2]
            if total > 0:
                frac = rng.uniform(0.1, 0.9)
                candidate[i1] = frac * total
                candidate[i2] = (1 - frac) * total

        candidate = np.clip(candidate, 0, None)
        s = np.sum(candidate)
        if s < 1e-10:
            continue
        # Don't renormalize - objective is scale-invariant
        obj = compute_obj(candidate)
        if obj < best_obj:
            best_obj = obj
            best_a = candidate.copy()
            cur_obj = obj
        elif cur_obj > best_obj * 1.01 and it % 1000 == 0:
            # Occasionally reset to best if stuck
            pass

    return best_a, best_obj


def structured_init(n, rng):
    """Generate diverse structured initializations."""
    choice = rng.integers(0, 8)
    x = np.arange(n, dtype=float)

    if choice == 0:
        freq = rng.uniform(0.02, 0.3)
        phase = rng.uniform(0, 2 * np.pi)
        a = np.abs(np.cos(2 * np.pi * freq * x + phase))
    elif choice == 1:
        # Two peaks
        c1 = rng.uniform(0.2, 0.8) * n
        c2 = rng.uniform(0.2, 0.8) * n
        width = rng.uniform(n * 0.05, n * 0.2)
        a = np.exp(-0.5 * ((x - c1) / width) ** 2) + np.exp(-0.5 * ((x - c2) / width) ** 2)
    elif choice == 2:
        # Periodic
        period = int(rng.integers(max(2, n // 40), max(3, n // 8)))
        pattern = rng.random(period)
        a = np.tile(pattern, n // period + 1)[:n]
    elif choice == 3:
        # Linear ramp + noise
        a = np.linspace(0.2, 1.0, n) * np.exp(rng.normal(0, 0.2, n))
    elif choice == 4:
        # Power law
        alpha = rng.uniform(0.1, 1.0)
        a = (x + 1) ** (-alpha)
    elif choice == 5:
        # Random with smoothing
        a = rng.random(n)
        k = max(1, n // 20)
        kernel = np.ones(k) / k
        a = np.convolve(a, kernel, mode='same')
    elif choice == 6:
        # Checkerboard-like
        a = np.where(np.arange(n) % 2 == 0, 1.0, rng.uniform(0.3, 0.7))
    else:
        a = rng.random(n)

    a = np.abs(a)
    s = np.sum(a)
    if s < 1e-10:
        a = rng.random(n)
        s = np.sum(a)
    return a / s


def equalize_top_bumps(a, max_iter=2000):
    """
    Try to equalize the top convolution values by adjusting a.
    If multiple k values achieve the max, try to reduce them simultaneously.
    """
    n = len(a)
    best_a = a.copy()
    best_obj = compute_obj(best_a)

    for it in range(max_iter):
        conv = np.convolve(best_a, best_a)
        max_b = np.max(conv)
        # Find all indices within 1% of max
        threshold = max_b * 0.99
        active_idx = np.where(conv >= threshold)[0]

        if len(active_idx) <= 1:
            break

        # For each active k, the gradient direction is 2*a_{k-i}
        # We want to decrease all active values
        # Take negative gradient averaged over active set
        grad = np.zeros(n)
        for k in active_idx:
            for i in range(n):
                ki = k - i
                if 0 <= ki < n:
                    grad[i] -= 2.0 * best_a[ki]

        # Normalize sum to 1
        sum_a = np.sum(best_a)
        S = np.mean(conv[active_idx])

        full_grad = 2.0 * n * (grad * sum_a - 2.0 * S * best_a) / (sum_a ** 3)

        # Small step
        lr = 0.005
        candidate = best_a - lr * full_grad
        candidate = np.clip(candidate, 0, None)
        s = np.sum(candidate)
        if s < 1e-10:
            break
        candidate = candidate / s

        obj = compute_obj(candidate)
        if obj < best_obj:
            best_obj = obj
            best_a = candidate

    return best_a, best_obj


def propose_candidate(seed=42, budget_s=1500, **kwargs):
    """
    Multi-strategy optimization: LP iterations, subgradient descent,
    local search, and equalization across multiple sequence lengths.
    """
    start_time = time.monotonic()
    deadline = start_time + float(budget_s) - 5.0
    rng = np.random.default_rng(seed)

    best_seq = None
    best_score = np.inf

    # Load parent
    parent = None
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64)
        if parent_raw.size > 0 and np.all(np.isfinite(parent_raw)):
            parent = np.clip(parent_raw, 0, 1000.0)
            obj = compute_obj(parent)
            if obj < best_score:
                best_score = obj
                best_seq = parent.copy()
            print(f"Parent (n={parent.size}): obj={obj:.8f}")
    except Exception:
        pass

    # Define which (n, method) combinations to try
    tasks = []

    # Priority 1: Refine parent
    if parent is not None:
        n_p = len(parent)
        tasks.append((n_p, 'lp', 0, parent.copy()))
        tasks.append((n_p, 'subgrad', 0, parent.copy()))
        tasks.append((n_p, 'local', 0, parent.copy()))
        tasks.append((n_p, 'equalize', 0, parent.copy()))

    # Priority 2: Various n with random init
    n_candidates = []
    if parent is not None:
        # Near parent length
        for n in [len(parent) - 10, len(parent) - 5, len(parent) + 5, len(parent) + 10]:
            if 20 <= n <= 500:
                n_candidates.append(n)
    for n in [100, 120, 150, 180, 200, 250, 300, 80, 400]:
        if n not in n_candidates:
            n_candidates.append(n)

    for n in n_candidates[:8]:
        tasks.append((n, 'lp', 1, None))
        tasks.append((n, 'lp', 2, None))

    for n in n_candidates[:4]:
        tasks.append((n, 'subgrad', 3, None))

    print(f"Total tasks: {len(tasks)}, deadline in {deadline - time.monotonic():.0f}s")

    for task_idx, (n, method, variant, init_a) in enumerate(tasks):
        if time.monotonic() > deadline - 10:
            break
        remaining = deadline - time.monotonic()
        if remaining < 15:
            break

        if init_a is None:
            init_a = structured_init(n, rng)

        tag = f"task{task_idx}: n={n} {method} v{variant}"
        print(f"[{tag}] starting, {remaining:.0f}s left, best={best_score:.8f}")

        if method == 'lp':
            max_iter = int(min(600, max(30, remaining * 0.15 / 0.5)))
            t0 = time.monotonic()
            a, obj = run_lp_iterations(init_a, max_iter=max_iter, verbose=False)
            elapsed = time.monotonic() - t0
            if obj < best_score:
                best_score = obj
                best_seq = a.copy()
                print(f"[{tag}] LP: obj={obj:.8f} ({elapsed:.1f}s) NEW BEST")
            else:
                print(f"[{tag}] LP: obj={obj:.8f} ({elapsed:.1f}s)")

            # Follow up with equalization
            if time.monotonic() < deadline - 20 and obj < best_score * 1.01:
                t0 = time.monotonic()
                a2, obj2 = equalize_top_bumps(a, max_iter=2000)
                elapsed2 = time.monotonic() - t0
                if obj2 < best_score:
                    best_score = obj2
                    best_seq = a2.copy()
                    print(f"[{tag}] EQ: obj={obj2:.8f} ({elapsed2:.1f}s) NEW BEST")

        elif method == 'subgrad':
            max_iter = int(min(10000, max(100, remaining * 0.2 / 0.05)))
            t0 = time.monotonic()
            a, obj = subgradient_optimize(init_a, max_iter=max_iter, verbose=False)
            elapsed = time.monotonic() - t0
            if obj < best_score:
                best_score = obj
                best_seq = a.copy()
                print(f"[{tag}] SG: obj={obj:.8f} ({elapsed:.1f}s) NEW BEST")
            else:
                print(f"[{tag}] SG: obj={obj:.8f} ({elapsed:.1f}s)")

        elif method == 'local':
            max_iter = int(min(30000, max(500, remaining * 0.2 / 0.005)))
            t0 = time.monotonic()
            a, obj = local_search(init_a, max_iter=max_iter)
            elapsed = time.monotonic() - t0
            if obj < best_score:
                best_score = obj
                best_seq = a.copy()
                print(f"[{tag}] LS: obj={obj:.8f} ({elapsed:.1f}s) NEW BEST")
            else:
                print(f"[{tag}] LS: obj={obj:.8f} ({elapsed:.1f}s)")

        elif method == 'equalize':
            t0 = time.monotonic()
            a, obj = equalize_top_bumps(init_a, max_iter=5000)
            elapsed = time.monotonic() - t0
            if obj < best_score:
                best_score = obj
                best_seq = a.copy()
                print(f"[{tag}] EQ: obj={obj:.8f} ({elapsed:.1f}s) NEW BEST")
            else:
                print(f"[{tag}] EQ: obj={obj:.8f} ({elapsed:.1f}s)")

    # Final intensive refinement on the best solution found
    if best_seq is not None and time.monotonic() < deadline - 5:
        remaining = deadline - time.monotonic()
        n_best = len(best_seq)
        print(f"\n=== FINAL REFINEMENT: n={n_best}, obj={best_score:.8f}, {remaining:.0f}s ===")

        # LP
        if n_best <= 600 and remaining > 30:
            max_iter = int(remaining * 0.3 / 0.5)
            a, obj = run_lp_iterations(best_seq, max_iter=max(10, max_iter))
            if obj < best_score:
                best_score = obj
                best_seq = a.copy()
                print(f"  Final LP: {best_score:.8f}")

        # Equalize
        if remaining > 10:
            a, obj = equalize_top_bumps(best_seq, max_iter=int(remaining * 0.3 / 0.01))
            if obj < best_score:
                best_score = obj
                best_seq = a.copy()
                print(f"  Final EQ: {best_score:.8f}")

        # Local search
        if time.monotonic() < deadline - 3:
            max_iter = int((deadline - time.monotonic() - 2) / 0.003)
            a, obj = local_search(best_seq, max_iter=max(100, max_iter))
            if obj < best_score:
                best_score = obj
                best_seq = a.copy()
                print(f"  Final LS: {best_score:.8f}")

        # Subgradient
        if time.monotonic() < deadline - 3:
            max_iter = int((deadline - time.monotonic() - 2) / 0.02)
            a, obj = subgradient_optimize(best_seq, max_iter=max(50, max_iter))
            if obj < best_score:
                best_score = obj
                best_seq = a.copy()
                print(f"  Final SG: {best_score:.8f}")

    # Fallback
    if best_seq is None:
        n_fb = 150
        best_seq = np.ones(n_fb) / n_fb
        best_score = compute_obj(best_seq)

    # Validate
    elapsed = time.monotonic() - start_time
    assert len(best_seq) > 0
    assert all(x >= 0 for x in best_seq)
    assert all(np.isfinite(x) for x in best_seq)
    final_obj = compute_obj(best_seq)
    print(f"\nDONE: n={len(best_seq)}, score={final_obj:.8f}, time={elapsed:.1f}s")

    return [float(x) for x in best_seq]
