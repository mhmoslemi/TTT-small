import numpy as np

def fast_conv(a):
    """Full self-convolution via FFT."""
    n = len(a)
    s = 1
    while s < 2 * n - 1:
        s <<= 1
    fa = np.fft.rfft(a, s)
    return np.fft.irfft(fa * fa, s)[:2 * n - 1]

def conv_max(a):
    """Return max of full self-convolution of a."""
    return float(fast_conv(a).max())

def build_constraint_matrix(a):
    """Build (2n-1) x n matrix A where (Ax)_k = (a*x)_k (convolution)."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for k in range(2 * n - 1):
        jl = max(0, k - n + 1)
        jh = min(n - 1, k)
        if jl <= jh:
            A[k, jl:jh + 1] = a[k - jh:k - jl + 1][::-1]
    return A

def solve_lp(a, M):
    """Solve LP: max sum(g) s.t. (a*g)_k <= M, g >= 0."""
    n = len(a)

    A = build_constraint_matrix(a)
    r = linprog(c, A_ub=A, b_ub=np.full(2 * n - 1, M),
               bounds=[(0, None)] * n, method='highs')
    if r.success and r.x is not None:
        g = r.x
        if g.sum() > 1e-10:
            return g
    return None

"""
Minimizes J(a) = 2*n*max(convolve(a,a))/(sum(a))^2 using:
1. LP best-response iterations with Polyak averaging and adaptive step size
2. L-BFGS-B gradient refinement with FFT-accelerated convolution
3. Nesterov accelerated subgradient descent
4. Multi-length exploration (512, 768, 1024, 2048) with diverse initializations
5. Local perturbation search around best candidate
"""

import numpy as np
from scipy.optimize import linprog, minimize
import time


def fast_conv(a):
    """Full self-convolution via FFT, O(n log n)."""
    n = len(a)
    s = 1
    while s < 2 * n - 1:
        s <<= 1
    fa = np.fft.rfft(a, s)
    return np.fft.irfft(fa * fa, s)[:2 * n - 1]


def conv_max(a):
    """Return max of full self-convolution of a."""
    return float(fast_conv(a).max())


def build_constraint_matrix(a):
    """Build (2n-1) x n matrix A where (Ax)_k = (a*x)_k."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for k in range(2 * n - 1):
        jl = max(0, k - n + 1)
        jh = min(n - 1, k)
        if jl <= jh:
            A[k, jl:jh + 1] = a[k - jh:k - jl + 1][::-1]
    return A


def solve_lp(a, M):
    """Solve LP: max sum(g) s.t. (a*g)_k <= M, g >= 0."""
    n = len(a)
    c = -np.ones(n)
    A = build_constraint_matrix(a)
    r = linprog(c, A_ub=A, b_ub=np.full(2 * n - 1, M),
               bounds=[(0, None)] * n, method='highs')
    if r.success and r.x is not None:
        g = r.x
        if g.sum() > 1e-10:
            return g
    return None


def update_state(state, a):

    s = float(evaluate_sequence(a_arr.tolist()))
    if np.isfinite(s) and s < state['score']:
        state['best'] = a_arr.copy()
        state['score'] = s
        print(f"  new best: J={s:.6f} n={a_arr.size}", flush=True)
    return state


def make_initializations(n, rng, parent):

    inits = []
    if parent is not None:
        t_old = np.linspace(0, 1, parent.size)
        t_new = np.linspace(0, 1, n)

        inits.append(p / p.sum())
    inits.append(np.ones(n) / n)
    inits.append(rng.uniform(0.3, 1.0, n).astype(np.float64))
    inits[-1] = inits[-1] / inits[-1].sum()
    t = np.arange(n) / n

    a = 1.0 + 0.5 * np.cos(4 * np.pi * t ** 2)
    a = np.maximum(a, 0.01)
    inits.append(a / a.sum())
    return inits


def lp_best_response_loop(a, n_iter, deadline, verbose=True):
    """
    LP best-response loop with Polyak averaging and adaptive step.
    Returns the best sequence found during the loop.
    """
    n = len(a)
    a = np.clip(a.astype(np.float64).copy(), 0, None)
    s = a.sum()
    if s < 1e-10:
        return a
    a = a / s

    step = 0.5
    best_a = a.copy()
    best_val = conv_max(a)
    avg_a = a.copy()
    prev_val = best_val

    for it in range(n_iter):
        if time.monotonic() > deadline:
            break
        M = conv_max(a)
        g = solve_lp(a, M)
        if g is None:
            if verbose:
                print(f"    LP failed at iter {it}", flush=True)
            break
        g = g / g.sum()

        # Adaptive step: Polyak-like
        avg_a = 0.95 * avg_a + 0.05 * g
        avg_a = np.maximum(avg_a, 0)
        avg_a = avg_a / avg_a.sum()

        # Mix current with LP solution, using averaged direction
        direction = 0.7 * g + 0.3 * avg_a
        direction = direction / direction.sum()

        a_new = (1.0 - step) * a + step * direction
        a_new = np.clip(a_new, 0, None)
        s_new = a_new.sum()
        if s_new < 1e-10:
            break
        a_new = a_new / s_new

        new_val = conv_max(a_new)
        if new_val < prev_val:
            step = min(step * 1.05, 0.8)
            prev_val = new_val
        else:
            step = max(step * 0.7, 0.05)

        a = a_new
        if new_val < best_val:
            best_val = new_val
            best_a = a.copy()
            if verbose and it % 20 == 0:
                print(f"    iter {it}: max_conv={new_val:.8f} J={2*n*new_val:.6f}", flush=True)

    return best_a


def lbfgs_refine(a0, n, deadline, max_iter=300):
    """L-BFGS-B refinement of sequence a0 of length n."""
    a0 = np.clip(a0.astype(np.float64).copy(), 0, None)
    s = a0.sum()
    if s < 1e-8:
        return a0
    a0 = a0 / s

    def obj_func(x):
        ax = np.maximum(x, 0.0)
        s_local = ax.sum()
        if s_local < 1e-8:
            return 1e10
        ax = ax / s_local
        return float(fast_conv(ax).max())

    res = minimize(obj_func, a0, method='L-BFGS-B',
                   bounds=[(0, None)] * n,
                   options={'maxiter': max_iter, 'max_fun': 8000,
                            'ftol': 1e-15, 'disp': False})
    if res.fun < 1e10:
        return res.x
    return a0


def nesterov_subgrad(a, n_iter, deadline):
    """Nesterov accelerated subgradient descent on max(conv) with sum=1."""
    n = len(a)
    a = np.clip(a.astype(np.float64).copy(), 0, None)
    s = a.sum()
    if s < 1e-10:
        return a
    a = a / s
    y = a.copy()
    best_a = a.copy()
    best_val = conv_max(a)
    t_k = 1.0
    lr = 1.0 / np.sqrt(n)

    for it in range(n_iter):
        if time.monotonic() > deadline:
            break
        b = fast_conv(y)
        M = float(b.max())
        tol = max(1e-10, M * 0.01)
        active = np.where(b >= M - tol)[0]
        sg = np.zeros(n)
        for k in active:
            ki = int(k)
            jl = max(0, ki - n + 1)
            jh = min(n - 1, ki)
            sg[jl:jh + 1] += a[ki - jh:ki - jl + 1][::-1]

        sg_norm = np.linalg.norm(sg)
        if sg_norm < 1e-15:
            break
        step_size = lr / (it + 1)
        y_new = y - step_size * sg / sg_norm
        y_new = np.clip(y_new, 0, None)
        s_new = y_new.sum()
        if s_new < 1e-10:
            break
        y_new = y_new / s_new

        t_new = (1 + np.sqrt(1 + 4 * t_k ** 2)) / 2
        beta = (t_k - 1) / t_new
        y = (1 + beta) * y_new - beta * a
        y = np.clip(y, 0, None)
        s_y = y.sum()
        if s_y < 1e-10:
            y = y_new
        else:
            y = y / s_y
        t_k = t_new

        a = y_new
        val = conv_max(a)
        if val < best_val:
            best_val = val
            best_a = a.copy()

    return best_a


def local_perturbation_search(a, rng, n_trials, deadline):
    """Try random perturbations around a to find improvements."""
    n = len(a)
    best = a.copy()
    best_val = conv_max(a)

    for trial in range(n_trials):
        if time.monotonic() > deadline:
            break
        # Random structured perturbation
        scale = rng.choice([0.01, 0.02, 0.05, 0.1])
        noise = rng.normal(0, scale, n)
        a_try = a + noise
        a_try = np.clip(a_try, 0, None)
        s = a_try.sum()
        if s < 1e-8:
            continue
        a_try = a_try / s
        val = conv_max(a_try)
        if val < best_val:
            best_val = val
            best = a_try.copy()

    return best
