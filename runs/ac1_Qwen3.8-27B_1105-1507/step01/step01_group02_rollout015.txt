"""
Optimizes nonnegative sequence to minimize 2*n*max(convolve(a,a))/sum(a)^2.
Uses smoothed mirror descent (multiplicative weights) with log-sum-exp
approximation of the max, FFT-based gradient computation, and adaptive
temperature scheduling. Tries multiple sequence lengths and initializations.

import numpy as np
import time


def fast_convolve(a):
    """Compute full linear convolution of a with itself using FFT."""
    n = len(a)
    size = 2 * n - 1
    N = 1 << (size - 1).bit_length()
    fa = np.fft.rfft(a, N)
    fc = fa * fa
    result = np.fft.irfft(fc, N)[:size]
    result[result < 0] = 0.0
    return result


def compute_gradient_fft(a, p):
    """
    Compute gradient of (1/tau)*log(sum_k exp(tau*c[k])) w.r.t. a,
    where c = convolve(a,a) and p[k] = exp(tau*c[k])/sum(exp(tau*c[m])).
    
    g[i] = sum_k p[k] * (a[k-i] + a[i-k]) with bounds.
    Uses FFT for O(n log n) computation.
    """
    n = len(a)
    N = 1 << (3 * n).bit_length()
    
    A = np.fft.rfft(a, N)
    P = np.fft.rfft(p, N)
    P_rev = np.fft.rfft(p[::-1], N)
    
    # term2[i] = convolve(p, a)[i] for i = 0..n-1
    term2 = np.fft.irfft(P * A, N)[:n]
    
    # term1[i] = convolve(a, p[::-1])[2n-2-i] for i = 0..n-1
    conv_ap = np.fft.irfft(A * P_rev, N)
    # Need positions 2n-2, 2n-3, ..., n-1 (n elements)
    term1 = conv_ap[n-1:2*n-1][::-1]
    
    return term1 + term2


def project_simplex(v):
    """Project vector v onto the probability simplex."""
    n = len(v)
    u = np.sort(v)[::-1]
    cv = np.cumsum(u) - 1.0
    rho_candidates = np.where(u * np.arange(1, n + 1) > cv)[0]
    if len(rho_candidates) == 0:
        return np.ones(n) / n
    rho = rho_candidates[-1]
    theta = cv[rho] / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def alt_projection_init(n, rng, n_iter=80):
    """Alternating projection for broadband initialization."""
    N = 1 << (2 * n - 2).bit_length()
    phases = rng.uniform(0.0, 2.0 * np.pi, N // 2 + 1)
    F = np.exp(1j * phases)
    F[0] = 1.0 + 0j
    a = np.ones(n) / n
    
    for it in range(n_iter):
        a_full = np.fft.irfft(F, N)
        a = np.maximum(a_full[:n], 0.0)
        s = a.sum()
        if s < 1e-10:
            a = np.ones(n) / n
        else:
            a = a / s
        F_full = np.fft.rfft(a, N)
        weights = np.linspace(1.0, 0.7, len(F_full))
        mag_target = weights ** 0.1
        F = mag_target * np.exp(1j * np.angle(F_full))
        F[0] = F[0].real + 0j
    
    return a
