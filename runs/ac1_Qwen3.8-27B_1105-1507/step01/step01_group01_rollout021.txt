```python
    u = np.sort(v)[::-1]

    rho = np.nonzero(u * np.arange(1, n+1) > (cssv - target_sum))[0][-1]
    theta = (cssv[rho] - target_sum) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

    return np.convolve(a, a)

    c is the convolution of a with itself.

    beta_c = beta * c
    max_bc = np.max(beta_c)
    exp_bc = np.exp(beta_c - max_bc)
    p = exp_bc / np.sum(exp_bc)
```

```python
def smooth_max_grad_decent(a, beta_start, beta_end, max_iters, eta, momentum, deadline):
    """Run smooth-max gradient descent."""
    a = np.asarray(a, dtype=np.float64).copy()
    a = normalize_to_sum1(a)
    n = a.size
    
    best = a.copy()
    best_score = local_score(best)
    
    v = np.zeros(n, dtype=np.float64)  # momentum
    beta = beta_start
    it = 0
    stuck = 0
    
    while it < max_iters and time.monotonic() < deadline:
        c, M = convolution_and_max(a)
        c_max = np.max(beta * c)
        exp_bc = np.exp(beta * c - c_max)
        p = exp_bc / np.sum(exp_bc)
        
        # Gradient
        q_rev = p[::-1]
        conv_result = np.convolve(a, q_rev)
        cross = conv_result[2*n-2 : n-2 : -1]
        grad = 2.0 * cross
        
        # Momentum update
        v = momentum * v - eta * grad
        a_new = a + v
        a_new = project_simplex(a_new, 1.0)
        v = momentum * v + (a - a_new)  # projection reflection
        
        a = a_new
        it += 1
        
        # Check improvement
        c, M = convolution_and_max(a)
        score = 2.0 * n * M
        if score < best_score:
            best = a.copy()
            best_score = score
            stuck = 0
        else:
            stuck += 1
        
        # Adaptive beta
        if it % 50 == 0:
            progress = it / max_iters
            beta = beta_start * (beta_end / beta_start) ** progress
        
        # Line search / step size adjustment
        if stuck > 20:
            eta *= 0.5
            stuck = 0
        elif stuck == 0 and eta < 0.1:
            eta *= 1.5
    
    return best, best_score
```
