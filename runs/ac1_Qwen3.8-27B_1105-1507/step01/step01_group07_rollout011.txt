Hybrid optimization for minimizing 2*n*max(conv(a,a))/(sum(a))^2.
Combines smooth mirror descent, LP-based refinement, and coordinate descent.
"""

import numpy as np
from scipy.optimize import linprog
import time

def compute_obj(a):
    n = len(a)
    if n == 0:
        return np.inf
    conv = np.convolve(a, a)
    max_b = float(np.max(conv))
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    return 2.0 * n * max_b / (s * s)

def smooth_objective_and_grad(a, tau):
    """Compute smooth approximation of max(conv(a,a)) and its gradient."""
    n = len(a)
    conv = np.convolve(a, a)
    scaled = conv / tau
    max_s = np.max(scaled)
    exp_s = np.exp(scaled - max_s)
    log_sum_exp = max_s + np.log(np.sum(exp_s))
    w = exp_s / np.sum(exp_s)  # softmax weights
    grad = 2.0 * np.correlate(w, a, mode='valid')
    obj = tau * log_sum_exp  # smooth approximation of max
    return obj, grad, conv

def lp_step(a, upper_bound=10.0):

            if g is not None and g_sum > 1.002:
                g = g / g_sum
                result = blend_and_check(best_a, g, best_a, best_obj, [0.2, 0.4, 0.6, 0.8])
                if result is not None:
                    best_a, best_obj = result
                    a = best_a.copy()

    if verbose:
        print(f"      Phase2 (exact MD): obj={best_obj:.6f}")

    # --- Phase 3: Dedicated LP refinement ---
    for it in range(100):
        if time.monotonic() > deadline - 15:
            break
        g, g_sum, M = lp_best_response(best_a)
        if g is None:
            break
        if g_sum <= 1.002:
            break
        g = g / g_sum
        result = blend_and_check(best_a, g, best_a, best_obj, [0.1, 0.2, 0.3, 0.5, 0.7, 0.9])
        if result is None:
            break
        best_a, best_obj = result

    if verbose:
        print(f"      Phase3 (LP): obj={best_obj:.6f}")

    # --- Phase 4: Coordinate descent polish ---
    remaining = deadline - time.monotonic()
    if remaining > 5:
        n_coord = int(remaining * 500)
        for it in range(n_coord):
            if time.monotonic() > deadline - 2:
                break
            m = int(rng.integers(0, n))
            factor = np.exp(rng.normal(0, 0.08))
            cand = best_a.copy()
            cand[m] *= factor
            cand = np.clip(cand, 0.0, None)
            s = np.sum(cand)
            if s < 1e-10:
                continue
            obj_c = compute_obj(cand)
            if obj_c < best_obj:
                best_a = cand
                best_obj = obj_c

    if verbose:
        print(f"      Phase4 (coord): obj={best_obj:.6f}")

    return best_a, best_obj


def structured_init(n, rng):
    """Generate diverse structured initializations."""
    choice = rng.integers(0, 8)
    if choice == 0:
        a = rng.random(n)
    elif choice == 1:
        phase = rng.uniform(0, 2 * np.pi)
        freq = rng.uniform(1, 5)
        a = np.maximum(0.01, np.cos(2 * np.pi * freq * np.arange(n) / n + phase))
    elif choice == 2:
        a = np.ones(n)
        a[n // 2:] = rng.uniform(0.2, 0.7)
    elif choice == 3:
        r = rng.uniform(0.8, 0.99)
        a = r ** np.arange(n)
    elif choice == 4:
        half = (n + 1) // 2
        h = rng.random(half)
        a = np.concatenate([h, h[-2::-1]])[:n]
    elif choice == 5:
        center = n // 2
        width = max(1, n // 5)
        a = np.ones(n) * 0.3
        a[max(0, center - width):min(n, center + width)] = 1.0
    elif choice == 6:
        period = int(rng.integers(max(2, n // 30), max(3, n // 5)))
        pattern = rng.random(period)
        a = np.tile(pattern, n // period + 1)[:n]
    else:
        # Two-scale: large blocks with internal structure
        block_size = max(2, n // 10)
        n_blocks = n // block_size
        block_weights = rng.random(n_blocks)
        a = np.repeat(block_weights, block_size)[:n]
    s = np.sum(a)
    if s < 1e-10:
        a = rng.random(n)
        s = np.sum(a)
    return a / s


def refine_best(best_seq, best_score, rng, deadline):
    """Final intensive refinement of the best solution found."""
    n = len(best_seq)
    a = best_seq.copy()
    obj = best_score

    # LP refinement
    for it in range(200):
        if time.monotonic() > deadline - 5:
            break
        g, g_sum, M = lp_best_response(a)
        if g is None or g_sum <= 1.001:
            break
        g = g / g_sum
        result = blend_and_check(a, g, a, obj, [0.05, 0.1, 0.2, 0.3, 0.5, 0.7, 0.9])
        if result is None:
            break
        a, obj = result

    # Exact mirror descent polish
    for it in range(3000):
        if time.monotonic() > deadline - 3:
            break
        eta = 0.1 / (1.0 + 0.0005 * it)
        grad, _ = exact_subgrad(a)
        a_new = mirror_step(a, grad, eta)
        obj_c = compute_obj(a_new)
        if obj_c < obj:
            a = a_new
            obj = obj_c

    # Fine coordinate descent
    if time.monotonic() < deadline - 1:
        for it in range(int((deadline - time.monotonic() - 1) * 1000)):
            if time.monotonic() > deadline - 1:
                break
            m = int(rng.integers(0, n))
            factor = np.exp(rng.normal(0, 0.03))
            cand = a.copy()
            cand[m] *= factor
            cand = np.clip(cand, 0.0, None)
            s = np.sum(cand)
            if s < 1e-10:
                continue
            obj_c = compute_obj(cand)
            if obj_c < obj:
                a = cand
                obj = obj_c

    return a, obj


def propose_candidate(seed=42, budget_s=1500, **kwargs):
    """
    Hybrid optimization: smooth/exact mirror descent + LP best-response + coord descent.
    Explores sequence lengths 10-400 with diverse initializations.
    Targets objective below 1.5027.
    """
    rng = np.random.default_rng(seed)
    start_time = time.monotonic()
    deadline = start_time + float(budget_s) - 15.0

    best_seq = None
    best_score = np.inf

    # Evaluate parent
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64)
        parent = np.clip(parent, 0, 1000.0)
        if parent.size > 0 and np.isfinite(parent).all():
            obj = compute_obj(parent)
            if obj < best_score:
                best_score = obj
                best_seq = parent.copy()
            print(f"Parent (n={parent.size}): obj={obj:.6f}")
    except Exception:
        pass

    # Also try to extend/modify parent
    if best_seq is not None:
        n_parent = len(best_seq)
        # Try with small perturbations
        for _ in range(5):
            pert = best_seq * np.exp(rng.normal(0, 0.05, n_parent))
            pert = np.clip(pert, 0, None)
            s = np.sum(pert)
            if s > 0.01:
                obj_p = compute_obj(pert)
                if obj_p < best_score:
                    best_score = obj_p
                    best_seq = pert.copy()

    # Main search over sequence lengths
    n_values = [10, 20, 30, 40, 60, 80, 100, 120, 150, 200, 250, 300, 400]

    for n in n_values:
        if time.monotonic() > deadline - 20:
            break
        remaining = deadline - time.monotonic()
        if remaining < 15:
            break

        # Allocate budget per n
        if n <= 50:
            t_per = min(remaining * 0.1, 50)
            n_restarts = 3
        elif n <= 150:
            t_per = min(remaining * 0.08, 60)
            n_restarts = 2
        else:
            t_per = min(remaining * 0.06, 50)
            n_restarts = 1
        t_per = max(t_per, 12)

        print(f"N={n}: {remaining:.0f}s left, t_per={t_per:.0f}s, restarts={n_restarts}")

        for restart in range(n_restarts):
            if time.monotonic() > deadline - 15:
                break
            a_init = structured_init(n, rng)
            t_alloc = min(t_per, deadline - time.monotonic() - 15)
            if t_alloc < 10:
                break
            a, obj = run_optimize(n, seed=seed + restart * 104729 + n * 31,
                                  time_budget=t_alloc, verbose=(restart == 0))
            if obj < best_score:
                best_score = obj
                best_seq = a.copy()
                print(f"  IMPROVED: n={n} r={restart}: obj={obj:.6f}")

    # Final intensive refinement
    if best_seq is not None and time.monotonic() < deadline - 20:
        print(f"Final refinement: n={len(best_seq)}, current best={best_score:.6f}")
        refined, refined_obj = refine_best(best_seq, best_score, rng, deadline)
        if refined_obj < best_score:
            best_score = refined_obj
            best_seq = refined
            print(f"  Refined: obj={best_score:.6f}")

    # Fallback
    if best_seq is None:
        best_seq = np.ones(100)
        best_score = compute_obj(best_seq)

    elapsed = time.monotonic() - start_time
    print(f"DONE: n={len(best_seq)}, score={best_score:.6f}, time={elapsed:.1f}s")

    best_seq = np.clip(np.asarray(best_seq, dtype=np.float64), 0.0, 1000.0)
    result = [float(x) for x in best_seq]
    # Verify
    final_score = compute_obj(best_seq)
    print(f"Final verification: score={final_score:.6f}")
    return result
