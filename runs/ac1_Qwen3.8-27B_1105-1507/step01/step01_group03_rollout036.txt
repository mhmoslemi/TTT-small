Let me try a different angle: what if I use iterative FFT-based optimization?

Algorithm (similar to Gerchberg-Saxe / Fienup):
1. Start with initial f (non-negative, sum = 1)
2. Compute F = DFT(f)
3. Compute |F|² (power spectrum)
4. Modify |F|² to be flatter (e.g., clip outliers, or apply a nonlinearity)
5. Set F_new = sqrt(modified_power) * exp(i*phase(F))
6. f_new = IDFT(F_new)
7. Project f_new onto non-negative, sum = 1
8. Repeat

This is a classic approach for flat spectrum design. Let me try this!

The modification in step 4 is the key. Options:
- Clip: |F|²_new = min(|F|², τ) where τ is chosen so that sum stays correct
- Soft clip: |F|²_new = |F|² / (1 + |F|²/τ)
- Power compression: |F|²_new = (|F|²)^α for α < 1

"""
Multi-strategy optimization for minimizing 2*n*max(conv(f,f))/(sum(f))^2
over non-negative sequences f.

Strategies used:
1. FISTA with log-sum-exp smoothing (fast, many iterations, FFT-accelerated)
2. LP-based linearized optimization (precise, for final refinement)
3. Flat-spectrum FFT iteration (exploits frequency-domain flatness)
4. Coordinate descent (local refinement of individual elements)

The search is run from multiple starting points and multiple sequence lengths.
"""

import time
import numpy as np
from scipy.fft import fft, ifft
from scipy.optimize import linprog
from scipy.sparse import csr_matrix


def normalize_sequence(arr):
    """Return nonnegative float array normalized to sum 1, or None."""
    try:
        x = np.asarray(arr, dtype=np.float64).ravel()
    except Exception:
        return None
    if x.size == 0:
        return None
    x = np.where(np.isfinite(x), x, 0.0)
    x = np.maximum(x, 0.0)
    s = float(x.sum())
    if s < 1e-300:
        return None
    return x / s


def compute_score(f):
    """Compute 2*n*max(conv(f,f)) assuming sum(f)=1."""
    n = len(f)
    conv = np.convolve(f, f)
    return float(2.0 * n * np.max(conv))


def compute_score_full(f):
    """Compute the full score including sum normalization."""
    n = len(f)
    s = float(np.sum(f))
    if s < 1e-300:
        return float('inf')
    conv = np.convolve(f, f)
    return float(2.0 * n * np.max(conv) / (s * s))


def gradient_smoothed(f, beta):
    """
    Compute gradient of (2n/beta)*log(sum(exp(beta*conv(f,f)))) w.r.t. f.
    f should have sum=1 and be non-negative.
    Returns gradient vector of same length as f.
    """
    n = len(f)
    conv = np.convolve(f, f)
    # Softmax for numerical stability
    logit = beta * (conv - conv.max())
    p = np.exp(logit)
    p_sum = p.sum()
    if p_sum < 1e-300:
        p = np.ones(len(conv)) / len(conv)
    else:
        p /= p_sum
    # grad[i] = 4*n * sum_k p[k] * f[k-i]
    # Compute cross-correlation via convolution
    temp = np.convolve(f, p[::-1], mode='full')
    cross = temp[n - 1:2 * n - 1][::-1]
    grad = 4.0 * n * cross
    return grad


def fista_optimize(f_init, beta_schedule, iters_per_beta, time_limit=None):

    FISTA optimization with log-sum-exp smoothing and annealing.
    Returns best (f, score) found.

        return None, float('inf')

    best_f = f.copy()
    best_score = compute_score(f)

    start_time = time.monotonic()

    for beta in beta_schedule:
        if time_limit is not None and (time.monotonic() - start_time) > time_limit:
            break

        z = f.copy()
        z_prev = f.copy()
        t = 1.0

        # Step size: L ~ 4*n*beta*max(f) is a rough Lipschitz bound
        fmax = max(float(np.max(f)), 1e-8)
        eta = 1.0 / (4.0 * n * beta * fmax)
        eta = min(eta, 50.0 / n)  # cap for stability

        for k in range(iters_per_beta):
            if time_limit is not None and (time.monotonic() - start_time) > time_limit:
                break

            t_new = (1.0 + np.sqrt(1.0 + 4.0 * t * t)) / 2.0
            y = z + ((t - 1.0) / t_new) * (z - z_prev)
            y = np.maximum(y, 0.0)
            sy = float(y.sum())
            if sy < 1e-300:
                y = f.copy()
            else:
                y = y / sy

            g = gradient_smoothed(y, beta)
            g = g - np.mean(g)  # tangent projection to sum=1

            z_new = y - eta * g
            z_new = np.maximum(z_new, 0.0)
            sz = float(z_new.sum())

            z = z_new
            t = t_new

        f = z.copy()
        sc = compute_score(f)
        if sc < best_score:
            best_score = sc
            best_f = f.copy()

    return best_f, best_score


def flat_spectrum_iterate(f_init, n_iter, alpha=0.5, time_limit=None):
    """
    Flat-spectrum iterative method: flatten the power spectrum to reduce
    autocorrelation peak.
    """
    f = normalize_sequence(f_init)
    if f is None:
        return None, float('inf')
    n = f.size
    best_f = f.copy()
    best_score = compute_score(f)

    # FFT size for linear convolution
    N = 1
    while N < 2 * n - 1:
        N *= 2
    N = 2 * N  # extra padding for safety

    start_time = time.monotonic()

    for k in range(n_iter):
        if time_limit is not None and (time.monotonic() - start_time) > time_limit:
            break

        F = fft(f, N)
        P = np.abs(F) ** 2

        # Power compression
        P_mean = P.mean()
        if P_mean < 1e-300:
            break
        P_norm = P / P_mean
        P_mod = P_norm ** alpha

        # Reconstruct with modified amplitude
        phase = np.angle(F)
        F_new = np.sqrt(P_mod) * np.exp(1j * phase)
        f_new = np.real(ifft(F_new))[:n]

        # Project to non-negative simplex
        f_new = np.maximum(f_new, 0.0)
        s = float(f_new.sum())
        if s < 1e-300:
            break
        f_new = f_new / s

        # Check score
        if k % 100 == 0 or k == n_iter - 1:
            sc = compute_score(f_new)
            if sc < best_score:
                best_score = sc
                best_f = f_new.copy()
                f = f_new.copy()
            else:
                # Keep trying but use f_new as next iterate
                # (might escape local structure)
                f = f_new.copy()
        else:
            f = f_new.copy()

    return best_f, best_score


def lp_direction_solve(f, n, M, A_template, edges, c_obj, bounds, m):
    """Solve LP: max sum(g) s.t. A@f <= M*scale, g >= 0."""
    A = A_template.copy()
    A.data[:] = f[edges]
    A.data *= float(n)
    b = np.full(m, M * n * n, dtype=np.float64)
    try:
        res = linprog(c_obj, A_ub=A, b_ub=b, bounds=bounds, method='highs')
        return res
    except Exception:
        return None


def make_lp_template(n):
    """Build sparse template for LP."""
    m = 2 * n - 1
    k64 = np.arange(m, dtype=np.int64)
    js = np.maximum(0, k64 - n + 1).astype(np.int32)
    je = np.minimum(n - 1, k64).astype(np.int32)
    counts = (je.astype(np.int64) - js.astype(np.int64) + 1).astype(np.int64)
    total = int(counts.sum())

    cum = np.cumsum(counts)
    starts = np.repeat(cum - counts, counts)
    local = np.arange(total, dtype=np.int64) - starts

    row = np.repeat(k64, counts).astype(np.int32)
    col_base = js[row.astype(np.int64)]
    col = (col_base + local).astype(np.int32)
    offsets = (row.astype(np.int64) - col.astype(np.int64)).astype(np.int32)

    A_off = csr_matrix((offsets, (row, col)), shape=(m, n), dtype=np.int32)
    edges = A_off.data.astype(np.int32, copy=True)
    A_template = A_off.astype(np.float64).copy()
    A_template.data.fill(0.0)

    c_obj = -np.ones(n, dtype=np.float64)
    bds = [(0.0, None)] * n
    return A_template, edges, c_obj, bds, m

def coordinate_descent(f_init, n_iter, time_limit=None, rng=None):
    """
    Coordinate descent: optimize one element at a time.
    For each coordinate i, find the value of f_i that minimizes the score
    given all other coordinates fixed.
    """
    f = normalize_sequence(f_init)
    if f is None:
        return None, float('inf')
    n = f.size
    best_f = f.copy()
    best_score = compute_score(f)

    if rng is None:
        rng = np.random.default_rng(0)

    start_time = time.monotonic()

    # Precompute which convolution indices are affected by each f[i]
    # Changing f[i] affects conv[k] for k in [i, i+n-1] (and symmetrically)
    # Specifically: conv[k] += delta * (f[k-i] + f[k-i]) = 2*delta*f[k-i]
    # (plus the diagonal term if 2i = k: additional delta^2)

    for it in range(n_iter):
        if time_limit is not None and (time.monotonic() - start_time) > time_limit:
            break

        # Random order of coordinates
        order = rng.permutation(n)

        for i in order:
            if time_limit is not None and (time.monotonic() - start_time) > time_limit:
                break

            old_val = f[i]
            if old_val < 1e-15:
                continue

            # conv[k] = sum_j f[j]*f[k-j]
            # d conv[k] / d f[i] = 2*f[k-i] (as derived)
            # If we change f[i] from old_val to new_val, delta = new_val - old_val
            # conv_new[k] = conv_old[k] + 2*delta*f[k-i] + delta^2*(1 if k=2i else 0)

            # To minimize max(conv), we want to reduce the peak.
            # This is a 1D optimization: minimize max_k (conv[k] + 2*d*f[k-i] + d^2*I(k=2i))
            # subject to f[i] >= 0 and sum(f) = 1 (so we'd need to redistribute)

            # For simplicity, let's just try reducing f[i] if it's at a peak location
            # and increasing it if it's at a low location.

            # Quick heuristic: try a few values
            # The optimal 1D problem is:
            # min_t max_k (c_k + 2*t*a_k)  where a_k = f[k-i], c_k = conv[k] - old_val*(2*f[k-i] + I(k=2i)*old_val)
            # subject to t >= 0 (and we'll renormalize)

            # This is a convex piecewise-linear problem in t.
            # The minimum is where the lines cross.

            # For efficiency, use a simple bisection on the subgradient condition.
            # Actually, let's just try a few candidates:
            # - t = 0 (set f[i] = 0)
            # - t that equalizes the top conv values

            # Simple approach: try reducing f[i] by a factor
            for factor in [0.5, 0.7, 0.9]:
                f[i] = old_val * factor
                s = f.sum()
                f = f / s
                sc = compute_score(f)
                if sc < best_score:
                    best_score = sc
                    best_f = f.copy()
                    # Accept and try to go lower
                else:
                    f[i] = old_val / s * s  # restore (approximately)
                    break

            # Restore if we broke out
            f[i] = old_val
            s = f.sum()
            if s > 1e-300:
                f[i] = f[i] * best_f.sum() / s  # this is getting messy

        # Actually, let me redo coordinate descent more carefully

    # Let me rewrite this more carefully
    f = normalize_sequence(f_init)
    n = f.size
    best_f = f.copy()
    best_score = compute_score(f)
    start_time = time.monotonic()

    for it in range(n_iter):
        if time_limit is not None and (time.monotonic() - start_time) > time_limit:
            break
        improved = False
        order = rng.permutation(n)
        for i in order:
            if time_limit is not None and (time.monotonic() - start_time) > time_limit:
                break
            conv = np.convolve(f, f)
            current_max = conv.max()
            old_val = f[i]
            if old_val < 1e-15:
                continue

            # Try to reduce f[i] and redistribute to the minimum element
            # Find the coordinate with minimum value (to put the mass)
            j = int(np.argmin(f))
            if j == i:
                continue

            # Binary search for optimal reduction
            lo, hi = 0.0, old_val
            for _ in range(15):
                mid = (lo + hi) / 2.0
                f[i] = old_val - mid
                f[j] = f[j] + mid / (f.sum() - old_val + old_val - mid)  # approximate
                # Actually this is getting messy with the normalization
                # Let me just do: set f[i] = old_val - mid, f[j] += mid, then normalize
                f[i] = old_val - mid
                f[j] = f[j] + mid
                s = f.sum()
                f = f / s
                conv = np.convolve(f, f)
                new_max = conv.max()
                if new_max < current_max:
                    lo = mid
                else:
                    hi = mid

            # Apply the best found
            f[i] = old_val - lo
            f[j] = f[j] + lo
            s = f.sum()
            f = f / s
            sc = compute_score(f)
            if sc < best_score:
                best_score = sc
                best_f = f.copy()
                improved = True

        if not improved and it > 5:
            break

    return best_f, best_score


def interpolate_to_length(f, n):
    """Resample sequence to length n."""
    f = normalize_sequence(f)
    if f is None or n <= 0:
        return None
    m = f.size
    if m == n:
        return f
    old = np.linspace(0.0, 1.0, m)
    new = np.linspace(0.0, 1.0, n)
    out = np.interp(new, old, f)
    return normalize_sequence(out)


def make_raised_cosine(n, beta=0.5):
    """Create a raised cosine (Hann-like) window."""
    t = np.linspace(0, 1, n)
    x = 0.5 * (1 + np.cos(2 * np.pi * t))
    return normalize_sequence(x)


def make_gaussian_window(n, sigma_frac=0.4):
    """Create a Gaussian window."""
    t = np.linspace(0, 1, n)
    center = 0.5
    x = np.exp(-0.5 * ((t - center) / (sigma_frac * 0.5)) ** 2)
    return normalize_sequence(x)


def propose_candidate(seed=42, budget_s=1500, **kwargs):
    """
    Search for a low-scoring nonnegative sequence minimizing
    2*n*max(convolve(f,f))/(sum(f))^2.

    rng = np.random.default_rng(rng_seed)

    deadline = start_time + total_budget - 20.0  # reserve 20s for final steps

    given_f = None
    try:
        raw = np.asarray(height_sequence_1, dtype=np.float64)

            best_score = compute_score(given_f)
            print(f"Given sequence: n={given_f.size}, score={best_score:.6f}", flush=True)
    except Exception:
        pass

    if best_f is None:
        best_f = np.ones(64) / 64.0
        best_score = 2.0

    print("\n=== Phase 1: FISTA + Flat Spectrum ===", flush=True)

        # Multiple starting points
        starts = []
        if given_f is not None:
            s1 = interpolate_to_length(given_f, n)

            f_res, s_res = fista_optimize(start, beta_schedule,
                                          iters_per_beta=iters // len(beta_schedule),
                                          time_limit=tb * 0.6)

                best_score = s_res

                print(f"    FISTA start {si}: score={s_res:.6f} ({dt:.1f}s)", flush=True)
