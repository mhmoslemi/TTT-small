"""
Search for low-score non-negative step-function coefficients.

Uses a "linearized LP" (successive linear programming) approach:
  - At each iteration, linearize the quadratic convolution constraints at the current point.
  - Solve the resulting LP to minimize the peak convolution value M.
  - Perform a line search between the current point and the LP solution.
  - Polish with multiplicative subgradient updates targeting active peaks.

This differs from the previous SOTA's LP (which maximizes sum(g) subject to
convolve(f,g) <= M) by directly minimizing the peak via convexification.

def normalize(a):
    a = np.asarray(a, dtype=np.float64)
    a = np.clip(a, 0.0, 1e6)
    s = a.sum()
    if s <= 1e-30:
        return None
    return a / s


def local_score(a):
    a = np.asarray(a, dtype=np.float64)
    if a.size == 0:
        return float('inf')
    a = np.clip(a, 0.0, 1000.0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    c = np.convolve(a, a)
    m = c.max()
    return float(2.0 * len(a) * m / (s * s))


def build_lp_matrix(a, n, active_threshold=None):
    """
    Build the sparse constraint matrix for the linearized LP.
    Constraint k: 2 * sum_{j in S_k} a[k-j] * a[j] - M <= c_t[k]
    """
    c = np.convolve(a, a)
    M_curr = c.max()

    if active_threshold is not None:
        threshold = M_curr * active_threshold
        active_mask = c >= threshold
    else:
        active_mask = np.ones(2 * n - 1, dtype=bool)

    rows = []
    cols = []
    data = []
    b_vec = []

    nconstraints = 2 * n - 1
    # We'll only include active constraints
    for k in range(nconstraints):
        if not active_mask[k]:
            continue
        row_idx = len(b_vec)
        b_vec.append(c[k])
        # For j in S_k: max(0, k-n+1) <= j <= min(k, n-1)
        j_start = max(0, k - n + 1)
        j_end = min(k, n - 1)
        for j in range(j_start, j_end + 1):
            val = 2.0 * a[k - j]
            if val > 1e-15:
                rows.append(row_idx)
                cols.append(j)
                data.append(val)
        # M column (column n)
        rows.append(row_idx)
        cols.append(n)
        data.append(-1.0)

    A = csr_matrix((data, (rows, cols)), shape=(len(b_vec), n + 1))
    b = np.array(b_vec, dtype=np.float64)
    return A, b, active_mask


def solve_lp(a, n, active_threshold=1.0):
    """Solve the linearized LP and return the solution."""
    A, b, active_mask = build_lp_matrix(a, n, active_threshold)
    m_row = A.shape[0]
    if m_row == 0:
        return None

    c_obj = np.zeros(n + 1)
    c_obj[n] = 1.0  # minimize M

    # Equality constraint: sum(a) = 1
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])

    bounds = [(0, None)] * n + [(0, None)]

    try:
        res = linprog(
            c_obj,
            A_ub=A,
            b_ub=b,
            A_eq=A_eq,
            b_eq=b_eq,
            bounds=bounds,
            method='highs',
            options={'presolve': True, 'time_limit': 30}
        )
    except Exception:
        return None

    if res is None or not res.success:
        return None

    x = np.asarray(res.x, dtype=np.float64)
    a_sol = x[:n]
    a_sol = np.clip(a_sol, 0.0, None)
    s = a_sol.sum()
    if s < 1e-10:
        return None
    a_sol = a_sol / s
    return a_sol

def multiplicative_polish(a, time_budget, seed):
    """Polish using multiplicative subgradient updates on active peaks."""
    a = normalize(a)
    if a is None:
        return None, float('inf')
    n = a.size
    best = a.copy()
    best_M = np.convolve(best, best).max()
    best_score = local_score(best)

    t0 = time.monotonic()
    deadline = t0 + time_budget
    rng = np.random.default_rng(seed)
    eta = 0.3
    stall = 0
    it = 0

    while time.monotonic() < deadline:
        c = np.convolve(best, best)
        M = c.max()
        if M <= 0:
            break

        # Find active peaks (top fraction)
        frac = 0.3
        threshold = M * (1.0 - frac)
        active = np.where(c >= threshold)[0]

        # Build subgradient
        grad = np.zeros(n)
        for k in active:
            w = c[k] / M
            j_start = max(0, k - n + 1)
            j_end = min(k, n - 1)
            j = np.arange(j_start, j_end + 1)
            grad[j] += 2.0 * w * a[k - j]

        # Normalize gradient
        gmax = grad.max()
        if gmax <= 0:
            break
        grad = grad / gmax

        # Multiplicative update
        fac = eta * (1.0 - grad)
        fac = np.clip(fac, -3.0, 3.0)
        a_new = best * np.exp(fac)
        a_new = np.clip(a_new, 0.0, None)
        s = a_new.sum()
        if s < 1e-10:
            a_new = best.copy()
        else:
            a_new = a_new / s

        M_new = np.convolve(a_new, a_new).max()
        if M_new < best_M - 1e-12:
            best = a_new.copy()
            best_M = float(M_new)
            best_score = local_score(best)
            stall = 0
            if eta < 0.5:
                eta *= 1.05
        else:
            stall += 1
            if stall > 30:
                if eta > 0.05:
                    eta *= 0.8
                else:
                    noise = np.exp(rng.normal(0, 0.15, n))
                    a_new = best * noise
                    a_new = normalize(a_new)
                    if a_new is not None:
                        best = a_new
                        best_M = np.convolve(best, best).max()
                        best_score = local_score(best)
                    eta = 0.3
                    stall = 0

        it += 1

    return best, best_score


def make_random_start(n, seed, kind=0, parent=None):
    rng = np.random.default_rng(seed)
    if kind == 0:
        a = rng.uniform(0.1, 1.0, n)
    elif kind == 1:
        a = rng.beta(2.0, 2.0, n)
    elif kind == 2:
        # Bimodal
        center = n // 2
        a = np.ones(n) * 0.3
        w = n // 4
        a[center - w:center + w] = 1.0
        a += rng.normal(0, 0.1, n)
    elif kind == 3:
        # Interpolate from parent
        if parent is not None and parent.size > 1:
            old = np.linspace(0, 1, parent.size)
            new = np.linspace(0, 1, n)
            a = np.interp(new, old, parent)
        else:
            a = np.ones(n)
        a *= np.exp(rng.normal(0, 0.3, n))
    elif kind == 4:
        # Cosine window variant
        x = np.linspace(0, np.pi, n)
        alpha = rng.uniform(0.5, 2.0)
        a = np.abs(np.cos(alpha * x))
    else:
        a = rng.exponential(1.0, n)

    a = np.clip(a, 0.01, None)
    a = normalize(a)
    return a

            (1024, 0.20),
            (2048, 0.25),
            (4096, 0.18),
            (8192, 0.10),
        ]
    elif budget >= 600:
        plan = [
            (512, 0.15),
            (1024, 0.30),
            (2048, 0.30),
            (4096, 0.20),
        ]
    else:
        plan = [
            (256, 0.15),
            (512, 0.35),
            (1024, 0.30),
            (2048, 0.20),
        ]

    total_frac = sum(f for _, f in plan)
    seed_cur = int(seed)

    for n, frac in plan:
        rem = deadline - time.monotonic()
        if rem < 30:
            break
        alloc = min(frac / total_frac * budget * 0.85, rem - 20)
        if alloc < 20:
            alloc = max(15, 0.3 * rem)

        print(f"\n--- Optimizing n={n}, budget={alloc:.1f}s ---", flush=True)
        cand, sc = optimize_length(n, alloc, seed_cur, parent_arr)
        if cand is not None and np.isfinite(sc) and sc < best_score:
            best_seq = cand.copy()
            best_score = sc
            print(f"NEW BEST: n={n} score={sc:.6f}", flush=True)
        else:
            print(f"  n={n} best={sc:.6f} (not improved)", flush=True)
        seed_cur += 7919

    # Final polishing on the global best
    rem = deadline - time.monotonic()
    if best_seq is not None and rem > 20:
        print(f"\n--- Final polish (rem={rem:.1f}s) ---", flush=True)
        n_best = best_seq.size

        # LP polish
        b_lp = min(rem * 0.4, 120.0)
        if b_lp > 5:
            a, s = lp_optimize(best_seq, b_lp, seed_cur + 1, active_threshold=0.9)
            if a is not None and s < best_score:
                best_seq = a
                best_score = s
                print(f"  Final LP: {best_score:.6f}", flush=True)

        rem = deadline - time.monotonic()
        if rem > 10:
            b_mu = min(rem * 0.5, 80.0)
            a, s = multiplicative_polish(best_seq, b_mu, seed_cur + 2)
            if a is not None and s < best_score:
                best_seq = a
                best_score = s
                print(f"  Final Mu: {best_score:.6f}", flush=True)

        # One more LP round with tighter active set
        rem = deadline - time.monotonic()
        if rem > 5:
            a, s = lp_optimize(best_seq, min(rem * 0.6, 60.0), seed_cur + 3, active_threshold=0.92)
            if a is not None and s < best_score:
                best_seq = a
                best_score = s
                print(f"  Final LP2: {best_score:.6f}", flush=True)

    # Fallback
    if best_seq is None or not np.isfinite(local_score(best_seq)):
        if parent_arr is not None:
            p = normalize(parent_arr)
            if p is not None:
                best_seq = p
        if best_seq is None:
            best_seq = np.ones(128, dtype=np.float64) / 128.0

    # Final validation
    final = normalize(best_seq)
    if final is None:
        final = np.ones(64, dtype=np.float64) / 64.0

    try:
        official = float(evaluate_sequence([float(x) for x in final]))
        if np.isfinite(official):
            best_score = official
    except Exception:
        pass

    print(f"\nFINAL: score={best_score:.6f} length={len(final)}", flush=True)
    return [float(x) for x in np.clip(np.asarray(final, dtype=np.float64), 0.0, 1000.0)]
