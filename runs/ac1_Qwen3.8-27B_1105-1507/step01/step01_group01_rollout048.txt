def normalize_to_sum1(a):
    a = np.asarray(a, dtype=np.float64)
    if a.size == 0:
        return None
    a = np.clip(a, 0.0, 1000.0)
    if not np.isfinite(a).all():
        return None
    s = float(np.sum(a))
    if not np.isfinite(s) or s <= 1e-300:
        return None
    return a / s


def force_sym(a):
    a = normalize_to_sum1(a)
    if a is None:
        return None
    a = 0.5 * (a + a[::-1])
    return normalize_to_sum1(a)


def local_score(a):
    a = np.asarray(a, dtype=np.float64)
    if a.size == 0:
        return float('inf')
    a = np.clip(a, 0.0, 1000.0)
    if not np.isfinite(a).all():
        return float('inf')
    s = float(np.sum(a))
    if not np.isfinite(s) or s <= 1e-12:
        return float('inf')
    c = np.convolve(a, a)
    m = float(np.max(c))
    if not np.isfinite(m):
        return float('inf')
    return float(2.0 * len(a) * m / (s * s))


def compute_subgradient(a, K=None):
    """Compute subgradient of 2n*max(convolve(a,a)) w.r.t. a."""
    n = a.size
    c = np.convolve(a, a)
    M = float(np.max(c))
    if not np.isfinite(M) or M <= 0:
        return np.zeros(n), c, M
    if K is None or K <= 0:
        K = min(n, len(c))
    active = np.argpartition(c, -K)[-K:]
    # Weight by how close each active peak is to the max
    weights = np.maximum(0, (c[active] - 0.5 * M) / M)
    if weights.sum() == 0:
        weights = np.ones(K) / K
    else:
        weights /= weights.sum()
    
    g = np.zeros(n, dtype=np.float64)
    for idx in range(len(active)):
        k = int(active[idx])
        w = float(weights[idx])
        # Subgradient of C[k] w.r.t. a[i] is 2*a[k-i] for 0<=k-i<=n-1
        s = max(0, k - (n - 1))
        e = min(k, n - 1)
        if s <= e:
            j = np.arange(s, e + 1)
            g[j] += 2.0 * w * a[k - j]
    return g, c, M


def project_simplex(v, target_sum=1.0):
    """Project v onto {x >= 0, sum(x) = target_sum}."""
    v = np.asarray(v, dtype=np.float64)
    v = np.clip(v, 0.0, None)
    s = float(np.sum(v))
    if s <= 0:
        return np.full(v.size, target_sum / v.size)
    return v * (target_sum / s)


def subgradient_step(a, g, step_size):
    """Take a subgradient step and project."""
    a_new = a - step_size * g
    a_new = project_simplex(a_new, 1.0)
    return a_new


def coordinate_descent_pass(a, n_coord=50, rng=None):
    """Perform coordinate descent: for selected coordinates, try to reduce the peak."""
    n = a.size
    c = np.convolve(a, a)
    M = float(np.max(c))
    if not np.isfinite(M) or M <= 0:
        return a.copy()
    
    if rng is None:
        rng = np.random.default_rng(42)
    
    # Select coordinates proportional to their influence on the peak
    active = np.argpartition(c, -min(n, 32))[-min(n, 32):]
    influence = np.zeros(n, dtype=np.float64)
    for k in active:
        k = int(k)
        s = max(0, k - (n - 1))
        e = min(k, n - 1)
        if s <= e:
            j = np.arange(s, e + 1)
            influence[j] += a[k - j]
    
    # Pick top n_coord coordinates by influence
    if n_coord >= n:
        coords = np.arange(n)
    else:
        # Mix of high-influence and random
        top_k = int(n_coord * 0.7)
        rand_k = n_coord - top_k
        top_idx = np.argsort(influence)[-top_k:]
        rand_idx = rng.integers(0, n, rand_k)
        coords = np.concatenate([top_idx, rand_idx])
    
    a_new = a.copy()
    M_new = M
    improved = False
    
    for i in coords:
        i = int(i)
        # Try several values for a[i]
        base_val = a_new[i]
        candidates = [0.0, base_val * 0.25, base_val * 0.5, base_val * 0.75, base_val * 1.5, base_val * 2.0]
        # Also try moving mass from i to the least influential coordinate
        best_trial = None
        best_peak = M_new
        
        for frac in [0.0, 0.3, 0.5, 0.7, 1.0]:
            a_trial = a_new.copy()
            a_trial[i] = base_val * frac
            a_trial = project_simplex(a_trial, 1.0)
            ct = np.convolve(a_trial, a_trial)
            pt = float(np.max(ct))
            if pt < best_peak:
                best_peak = pt
                best_trial = a_trial.copy()
        
        if best_trial is not None:
            a_new = best_trial
            M_new = best_peak
            improved = True
    
    return a_new


def lp_direction(a, n, seed, use_sym=False):
    """Solve LP to find a direction that reduces the peak."""
    if not HAVE_SCIPY:
        return None
    
    a = normalize_to_sum1(a)
    if a is None:
        return None
    
    c = np.convolve(a, a)
    M = float(np.max(c))
    if not np.isfinite(M) or M <= 0:
        return None
    
    # Use only top-K constraints for speed
    K = min(2 * n - 1, max(64, n))
    active = np.argpartition(c, -K)[-K:]
    active = np.sort(active)
    
    # Build constraint matrix A[i,j] such that (a*b)[i] = sum_j A[i,j] * b[j]
    # (a*b)[k] = sum_j a[k-j] * b[j] for valid j
    A = np.zeros((K, n), dtype=np.float64)
    for row, k in enumerate(active):
        k = int(k)
        for j in range(n):
            ki = k - j
            if 0 <= ki < n:
                A[row, j] = a[ki]
    
    rng = np.random.default_rng(seed + 987654321)
    cvec = -np.ones(n, dtype=np.float64)
    cvec += 1e-10 * rng.uniform(-1, 1, n)
    bvec = np.full(K, M * 1.0001, dtype=np.float64)
    bounds = [(0.0, None)] * n
    
    try:
        res = linprog(cvec, A_ub=A, b_ub=bvec, bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': 30})
    except Exception:
        return None
    
    if res is None or not res.success:
        return None
    
    x = np.asarray(res.x, dtype=np.float64)
    x = np.clip(x, 0.0, None)
    s = float(np.sum(x))
    if s <= 0:
        return None
    x = x / s
    return x


def simulated_annealing_step(a, score, temperature, rng):
    """Accept a worse solution with probability exp(-(new-old)/T)."""
    if temperature <= 0:
        return a, score
    # Generate a perturbed candidate
    noise = np.exp(rng.normal(0.0, 0.1, a.size))
    a_new = a * noise
    a_new = project_simplex(a_new, 1.0)
    sc_new = local_score(a_new)
    if np.isfinite(sc_new):
        if sc_new < score or rng.random() < np.exp(-(sc_new - score) / max(temperature, 1e-10)):
            return a_new, sc_new
    return a, score

def make_initial(n, seed, kind=None):
    """Create diverse initial sequences."""
    rng = np.random.default_rng(seed)
    if kind is None:
        kind = int(rng.integers(0, 8))
    kind = kind % 8
    
    if kind == 0:
        # Random
        a = rng.uniform(0.1, 1.0, n)
    elif kind == 1:
        # Bimodal (two bumps)
        center1 = n // 4
        center2 = 3 * n // 4
        width = n // 8
        x = np.arange(n)
        a = np.exp(-0.5 * ((x - center1) / width) ** 2) + np.exp(-0.5 * ((x - center2) / width) ** 2)
        a += 0.1 * rng.uniform(0, 1, n)
    elif kind == 2:
        # Asymmetric tent
        x = np.arange(n)
        peak = int(n * rng.uniform(0.3, 0.7))
        a = np.exp(-np.abs(x - peak) / (n * 0.3))
        a += 0.05 * rng.uniform(0, 1, n)
    elif kind == 3:
        # Cosine-tapered (Hanning-like but asymmetric)
        x = np.arange(n) / (n - 1)
        shift = rng.uniform(-0.2, 0.2)
        a = 0.5 * (1 + np.cos(2 * np.pi * (x + shift)))
        a = np.clip(a, 0.01, None)
    elif kind == 4:
        # Power law decay from center
        center = n // 2
        x = np.arange(n)
        a = (1.0 + np.abs(x - center)) ** (-rng.uniform(0.5, 2.0))
        a += 0.02 * rng.uniform(0, 1, n)
    elif kind == 5:
        # Random + smoothed
        raw = rng.uniform(0, 1, n)
        w = int(n * rng.uniform(0.02, 0.1))
        if w > 1:
            kern = np.ones(w) / w
            a = np.convolve(raw, kern, mode='same')
        else:
            a = raw
    elif kind == 6:
        # Alternating high/low with noise
        a = np.where(np.arange(n) % 2 == 0, 1.0, 0.3)
        a += rng.uniform(-0.1, 0.1, n)
        a = np.clip(a, 0.01, None)
    else:
        # Beta-distributed
        a = rng.beta(2, 2, n) + 0.01
    
    a = np.clip(a, 0.0, None)
    a = normalize_to_sum1(a)
    return a


def optimize_length(n, budget, seed, use_sym=False, parent=None):
    """Optimize at a given length with multiple starts."""
    t0 = time.monotonic()
    deadline = t0 + budget
    best = None
    best_sc = float('inf')
    
    # Get parent if available
    if parent is not None and parent.size == n:
        p = normalize_to_sum1(parent)
        if p is not None:
            best = p.copy()
            best_sc = local_score(p)
    
    # Multiple starts
    num_starts = min(12, max(4, int(budget / 30)))
    
    for s in range(num_starts):
        rem = deadline - time.monotonic()
        if rem < 15:
            break
        
        if s == 0 and parent is not None and parent.size == n:
            a = best
        else:
            a = make_initial(n, seed + s * 1009, kind=s)
        
        alloc = min(rem * 0.7, budget / max(1, num_starts - s) * 1.5)
        alloc = max(10.0, alloc)
        
        cand, sc = optimize_single(a, alloc, seed + s * 7919, use_sym)
        
        if cand is not None and np.isfinite(sc) and sc < best_sc:
            best = cand.copy()
            best_sc = float(sc)
            print(f"  length={n} start={s} sym={use_sym}: score={best_sc:.6f}", flush=True)
    
    return best, best_sc


def propose_candidate(seed=42, budget_s=1500, **kwargs):
    """Main entry point: search for the best sequence within the time budget."""
    start_time = time.monotonic()
    try:
        budget = float(budget_s)
    except Exception:
        budget = 1500.0
    if not np.isfinite(budget) or budget <= 0:
        budget = 1500.0
    
    deadline = start_time + max(10.0, budget) - 5.0
    
    # Try to get parent
    parent = globals().get('height_sequence_1', None)
    parent_arr = None
    if parent is not None:
        try:
            parent_arr = np.asarray(parent, dtype=np.float64)
            parent_arr = np.clip(parent_arr, 0.0, 1000.0)
            if parent_arr.size == 0 or not np.isfinite(parent_arr).all():
                parent_arr = None
        except Exception:
            parent_arr = None
    
    best_seq = None
    best_score = float('inf')
    
    # Initialize best from parent
    if parent_arr is not None:
        p = normalize_to_sum1(parent_arr)
        if p is not None:
            best_seq = p.copy()
            best_score = local_score(p)
            print(f"Parent: length={p.size}, score={best_score:.6f}", flush=True)
    
    # Strategy: focus on lengths that give the best score/effort ratio
    # For length n, theoretical lower bound ~ 1 + c/sqrt(n)
    # n=512: ~1.05, n=1024: ~1.035, n=2048: ~1.025, n=4096: ~1.018
    # In practice, we're at ~1.51 for n=512, so there's room to improve
    
    if budget >= 1200:
        lengths = [
            (512, 0.12, False),
            (1024, 0.18, False),
            (2048, 0.22, False),
            (4096, 0.25, False),
            (8192, 0.23, False),
        ]
    elif budget >= 600:
        lengths = [
            (512, 0.15, False),
            (1024, 0.25, False),
            (2048, 0.35, False),
            (4096, 0.25, False),
        ]
    else:
        lengths = [
            (512, 0.2, False),
            (1024, 0.3, False),
            (2048, 0.5, False),
        ]
    
    seedcur = int(seed)
    
    for n, frac, sym in lengths:
        rem = deadline - time.monotonic()
        if rem < 20:
            break
        alloc = min(frac * budget, rem * 0.6)
        alloc = max(15.0, alloc)
        
        try:
            cand, sc = optimize_length(n, alloc, seedcur, sym, parent_arr)
            if cand is not None and np.isfinite(sc) and sc < best_score:
                best_seq = cand.copy()
                best_score = float(sc)
                print(f"NEW BEST: length={n} sym={sym} score={best_score:.6f}", flush=True)
            seedcur += 997
        except Exception as e:
            print(f"  length={n} failed: {e}", flush=True)
    
    # If we have time, do additional optimization on the best found
    rem = deadline - time.monotonic()
    if best_seq is not None and rem > 20:
        print(f"Polishing best (length={best_seq.size}, score={best_score:.6f}) with {rem:.1f}s", flush=True)
        cand, sc = optimize_single(best_seq, min(rem * 0.8, 120.0), seedcur + 54321, False)
        if cand is not None and np.isfinite(sc) and sc < best_score:
            best_seq = cand.copy()
            best_score = float(sc)
            print(f"After polish: score={best_score:.6f}", flush=True)
    
    # Final coordinate descent on best
    rem = deadline - time.monotonic()
    if best_seq is not None and rem > 10:
        rng = np.random.default_rng(seedcur + 777)
        a_cd = coordinate_descent_pass(best_seq, n_coord=min(2000, best_seq.size // 2), rng=rng)
        sc_cd = local_score(a_cd)
        if np.isfinite(sc_cd) and sc_cd < best_score:
            best_seq = a_cd.copy()
            best_score = float(sc_cd)
            print(f"After CD: score={best_score:.6f}", flush=True)
    
    # Fallback
    if best_seq is None:
        best_seq = np.ones(64, dtype=np.float64) / 64.0
        best_score = 2.0
    
    # Validate with official evaluator
    final = normalize_to_sum1(best_seq)
    if final is None:
        final = np.ones(64, dtype=np.float64) / 64.0
    
    try:
        official = float(evaluate_sequence([float(x) for x in final]))
        if np.isfinite(official):
            best_score = official
        else:
            if parent_arr is not None:
                fp = normalize_to_sum1(parent_arr)
                if fp is not None:
                    try:
                        off2 = float(evaluate_sequence([float(x) for x in fp]))
                        if np.isfinite(off2):
                            final = fp
                            best_score = off2
                    except Exception:
                        pass
    except Exception:
        pass
    
    elapsed = time.monotonic() - start_time
    print(f"FINAL: score={best_score:.6f}, length={len(final)}, time={elapsed:.1f}s", flush=True)
    
    return [float(x) for x in np.clip(np.asarray(final, dtype=np.float64), 0.0, 1000.0)]
