    #   = max_k (base[k] + 2*x*a[k-j])

    #   The max of lines is a convex piecewise-linear function.
    #   We can compute it for multiple x values simultaneously.

    X = np.linspace(0, max(aj * 3, S * 1.5), num_trials)

    valid_ks = [k for k in range(k_start, k_end+1) if k != 2*j]
    if valid_ks:

        coeffs = np.array([2 * a[k-j] for k in valid_ks])
        bases = np.array([base[k] for k in valid_ks])
        # new_conv[k, i] = bases[k] + coeffs[k] * X[i]
        # max over k: max_linear[i] = max_k(bases[k] + coeffs[k]*X[i])
        linear_vals = bases[:, None] + coeffs[:, None] * X[None, :]  # shape (len(valid_ks), num_trials)
        max_linear = np.max(linear_vals, axis=0)  # shape (num_trials,)
    else:
        max_linear = np.zeros(num_trials)
    
    # Quadratic part (k = 2j, if valid)
    if k_start <= 2*j <= k_end:
        max_quad = base[2*j] + X**2
    else:
        max_quad = np.full(num_trials, -np.inf)
    
    # Constant part (invalid k)
    invalid_ks = [k for k in range(2*n-1) if k < k_start or k > k_end]
    if invalid_ks:
        max_const = np.max([base[k] for k in invalid_ks])
    else:
        max_const = 0.0  # or -inf
    
    # Overall max
    max_conv = np.maximum(np.maximum(max_linear, max_quad), max_const)
    
    # Objective: 2n * max_conv / (S_other + X)^2
    new_S = S_other + X
    obj = 2 * n * max_conv / (new_S * new_S)
    
    # Find the best x
    best_idx = np.argmin(obj)
    best_x = X[best_idx]
    best_obj = obj[best_idx]
    
    return best_x, best_obj

    return np.convolve(a, a)

def evaluate_fast(a):
    """Fast evaluation without the security checks (for internal use)."""
    n = len(a)
    conv = np.convolve(a, a)
    max_c = np.max(conv)
    S = np.sum(a)
    if S < 0.01:
        return np.inf
    return 2.0 * n * max_c / (S * S)

def coord_sweep(a, conv=None, verbose=False):
    """One full sweep of coordinate descent. Returns updated a and conv."""
    n = len(a)
    if conv is None:
        conv = np.convolve(a, a)
    
    current_best = 2.0 * n * np.max(conv) / (np.sum(a) ** 2)
    
    for j in range(n):
        aj = a[j]
        S = np.sum(a)
        S_other = S - aj
        
        # Compute base (conv without a[j]'s contribution)
        base = conv.copy()
        k_start = max(0, j)
        k_end = min(2*n-2, j+n-1)
        
        # Subtract a[j]'s contribution
        # For k in [k_start, k_end]:
        #   if k == 2j: base[k] -= aj^2
        #   else: base[k] -= 2*aj*a[k-j]
        if k_start <= k_end:
            for k in range(k_start, k_end + 1):
                kj = k - j
                if k == 2 * j:
                    base[k] -= aj * aj
                else:
                    base[k] -= 2.0 * aj * a[kj]
        
        # 1D optimization: find x that minimizes 2n * max_k conv_k(x) / (S_other + x)^2
        # where conv_k(x) = base[k] + 2*x*a[k-j] (for valid k, k≠2j) or base[2j] + x^2 (for k=2j)
        
        # Determine search bounds
        x_hi = max(4.0 * aj + S / (2.0 * n), 0.01)
        
        # Grid search
        X = np.linspace(0.0, x_hi, 60)
        
        # Vectorized computation
        # For valid k (k_start <= k <= k_end, k != 2j): conv_k(x) = base[k] + 2*x*a[k-j]
        # For k = 2j (if in range): conv_k(x) = base[2j] + x^2
        # For other k: conv_k(x) = base[k] (constant)
        
        # Linear part
        valid_ks = [k for k in range(k_start, k_end + 1) if k != 2 * j]
        if valid_ks:
            coeffs = np.array([2.0 * a[k - j] for k in valid_ks])
            bases = np.array([base[k] for k in valid_ks])
            # Shape: (len(valid_ks), len(X))
            linear_part = bases[:, None] + coeffs[:, None] * X[None, :]
            max_linear = np.max(linear_part, axis=0)
        else:
            max_linear = np.zeros(len(X))
        
        # Quadratic part (k = 2j)
        if k_start <= 2 * j <= k_end:
            max_quad = base[2 * j] + X ** 2
        else:
            max_quad = np.full(len(X), -1e30)
        
        # Constant part (k outside [k_start, k_end])
        if k_start > 0 or k_end < 2 * n - 2:
            const_vals = np.concatenate([base[:k_start], base[k_end+1:2*n-1]]) if (k_start > 0 or k_end < 2*n-2) else np.array([0.0])
            max_const = np.max(const_vals) if len(const_vals) > 0 else 0.0
        else:
            max_const = 0.0
        
        max_conv = np.maximum(np.maximum(max_linear, max_quad), max_const)
        
        new_S = S_other + X
        # Avoid division by zero
        new_S = np.maximum(new_S, 0.01)
        obj = 2.0 * n * max_conv / (new_S * new_S)
        
        best_idx = np.argmin(obj)
        best_x = X[best_idx]
        best_obj = obj[best_idx]
        
        # Refinement around best
        if best_idx > 0 and best_idx < len(X) - 1:
            x_lo = X[best_idx - 1]
            x_hi_ref = X[best_idx + 1]
            X2 = np.linspace(x_lo, x_hi_ref, 40)
            
            if valid_ks:
                linear_part2 = bases[:, None] + coeffs[:, None] * X2[None, :]
                max_linear2 = np.max(linear_part2, axis=0)
            else:
                max_linear2 = np.zeros(len(X2))
            
            if k_start <= 2 * j <= k_end:
                max_quad2 = base[2 * j] + X2 ** 2
            else:
                max_quad2 = np.full(len(X2), -1e30)
            
            max_conv2 = np.maximum(np.maximum(max_linear2, max_quad2), max_const)
            new_S2 = S_other + X2
            new_S2 = np.maximum(new_S2, 0.01)
            obj2 = 2.0 * n * max_conv2 / (new_S2 * new_S2)
            
            best_idx2 = np.argmin(obj2)
            best_x = X2[best_idx2]
            best_obj = obj2[best_idx2]
        
        if best_obj < current_best:
            a[j] = best_x
            current_best = best_obj
    
    # Final recompute
    conv = np.convolve(a, a)
    return a, conv
