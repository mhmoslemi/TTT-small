def fast_conv(a):
    """Full self-convolution via FFT, O(n log n)."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]

def compute_smoothed_grad(a, T):
    """Compute gradient of (1/T)*log(sum exp(b_k/T)) w.r.t. a.
    
    Uses the identity: grad_j = (2/T) * sum_k p_k * a_{k-j}
    which is (2/T) * cross_correlation(p, a)_j
    """
    n = len(a)
    b = fast_conv(a)
    # Compute softmax of b/T
    b_scaled = b / T
    b_max = b_scaled.max()
    exp_b = np.exp(b_scaled - b_max)
    p = exp_b / exp_b.sum()
    
    # Cross-correlation: grad_j = (2/T) * sum_i a_i * p_{i+j}
    # Using FFT: Re[IFFT(FFT(a, N) * conj(FFT(p, N)))]
    size = 1
    while size < 3 * n - 2:
        size <<= 1
    A = np.fft.rfft(a, size)
    P = np.fft.rfft(p, size)
    cc = np.fft.irfft(A * np.conj(P), size)[:n]
    grad = (2.0 / T) * cc
    return grad, b, p

def simplex_project(v):
    """Project v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    mu = np.sort(v)[::-1]
    cv = np.cumsum(mu) - 1.0
    rho = np.where(mu * np.arange(1, n+1) - cv > 0)[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho = rho[-1]
    theta = (cv[rho] / (rho + 1.0))
    return np.maximum(v - theta, 0.0)

def fista_step(a, T, lr, max_iters, deadline):
    """FISTA for minimizing smoothed max on the simplex."""
    n = len(a)
    a = a.astype(np.float64).copy()
    a = a / a.sum()
    y = a.copy()
    t_k = 1.0
    
    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        grad, b, p = compute_smoothed_grad(y, T)
        # Gradient step + projection
        z = y - lr * grad
        a_new = simplex_project(z)
        # FISTA update
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t_k * t_k)) / 2.0
        y = a_new + ((t_k - 1.0) / t_new) * (a_new - a)
        y = np.maximum(y, 0.0)
        if y.sum() > 0:
            y = y / y.sum()
        else:
            y = a_new
        a = a_new
        t_k = t_new
    
    return a

    n = len(a)

def build_lp_matrix(a):

        jh = min(n - 1, k)
        if jl <= jh:
            A[k, jl:jh + 1] = a[k - jh:k - jl + 1][::-1]

"""
Minimizes J(a) = 2*n*max(convolve(a,a))/(sum(a))^2 using:
1. Adaptive multiplicative soft-deflation (softmax-weighted peak targeting)
2. FISTA with smoothed max (log-sum-exp) for gradient refinement
3. LP best-response mirror descent for final polish
4. Multiple sequence lengths (512-4096) and diverse initializations with random restarts

import numpy as np
from scipy.optimize import linprog
import time


def fast_conv(a):
    """Full self-convolution via FFT, O(n log n)."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]


def compute_softmax_weights(b, T):
    """Compute softmax(b/T) safely."""
    b_scaled = b / T
    bm = b_scaled.max()
    exp_b = np.exp(b_scaled - bm)
    s = exp_b.sum()
    if s < 1e-300:
        return np.full_like(b, 1.0 / len(b))
    return exp_b / s


def compute_deflation_direction(a, p):
    """Compute sum_k p_k * a_{k-j} for all j using FFT.
    This is the cross-correlation of a with p, evaluated at lags 0..n-1.
    """
    n = len(a)
    size = 1
    while size < 3 * n - 2:
        size <<= 1
    A = np.fft.rfft(a, size)
    P = np.fft.rfft(p, size)
    cc = np.fft.irfft(A * np.conj(P), size)[:n]
    return cc


def optimize_soft_deflation(a, T, max_iters, deadline):
    """Adaptive multiplicative soft-deflation with softmax weights.
    a_j <- a_j * exp(-eta * (2/T) * sum_k p_k * a_{k-j})
    where p_k = softmax(b_k/T). Adaptive step size.
    """
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return a
    a = a / s

    b = fast_conv(a)
    M = float(b.max())
    best_M = M
    eta = T * 0.3
    stagnation = 0

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        p = compute_softmax_weights(b, T)
        cc = compute_deflation_direction(a, p)
        grad = (2.0 / T) * cc

        log_update = -eta * grad
        log_update = np.clip(log_update, -8.0, 8.0)
        a_new = a * np.exp(log_update)
        a_new = np.maximum(a_new, 1e-25)
        sn = a_new.sum()
        if sn < 1e-15:

        a_new = a_new / sn

        M_new = float(b_new.max())

            eta = min(eta * 1.15, T)
        else:
            stagnation += 1
            eta = eta * 0.7
            if stagnation > 20:

def optimize_fista_smoothed(a, T, max_iters, deadline):
    """FISTA for minimizing (1/T)*log(sum exp(b_k/T)) on simplex."""
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return a
    a = a / s
    y = a.copy()
    t_k = 1.0
    # Estimate gradient Lipschitz constant
    b0 = fast_conv(y)
    M0 = b0.max()
    L = (4.0 / (T * T)) * M0 + 1e-8
    lr = 1.0 / L

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        p = compute_softmax_weights(fast_conv(y), T)
        cc = compute_deflation_direction(y, p)
        grad = (2.0 / T) * cc

        z = y - lr * grad
        # Project onto simplex
        u = np.sort(z)[::-1]
        cssv = np.cumsum(u) - 1.0
        valid = (u * np.arange(1, n + 1) - cssv) > 0
        rho = int(valid.sum()) - 1
        if rho < 0:
            a_new = np.full(n, 1.0 / n)
        else:
            theta = cssv[rho] / (rho + 1.0)
            a_new = np.maximum(z - theta, 0.0)

        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t_k * t_k)) / 2.0
        y = a_new + ((t_k - 1.0) / t_new) * (a_new - a)
        y = np.maximum(y, 0.0)
        ys = y.sum()
        if ys > 1e-15:
            y = y / ys
        else:
            y = a_new.copy()
        a = a_new
        t_k = t_new

    return a


def build_lp_matrix(a):
    """Build constraint matrix for LP: row k has a[k-j] at column j."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for k in range(2 * n - 1):
        jl = max(0, k - n + 1)
        jh = min(n - 1, k)
        if jl <= jh:
            A[k, jl:jh + 1] = a[k - jh:k - jl + 1][::-1]
    return A


def lp_best_response(a, M):
    """LP: max sum(g) s.t. (a*g)_k <= M for all k, g >= 0."""
    n = len(a)
    c = -np.ones(n)
    A = build_lp_matrix(a)
    try:
        r = linprog(c, A_ub=A, b_ub=np.full(2 * n - 1, M),
                    bounds=[(0, None)] * n, method='highs',
                    options={'presolve': True, 'time_limit': 60})
        if r.success and r.x is not None and r.x.sum() > 1e-10:
            return r.x
    except Exception:
        pass
    return None


def optimize_lp_descent(a, max_iters, deadline):
    """LP-based mirror descent (best response iterations)."""
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return a
    a = a / s

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        b = fast_conv(a)
        M = float(b.max())
        g = lp_best_response(a, M)
        if g is None:
            break
        g = g / g.sum()
        progress = it / max(max_iters - 1, 1)
        t_mix = max(0.03, 0.6 * (1.0 - progress) ** 0.7)
        a_new = (1.0 - t_mix) * a + t_mix * g
        a_new = np.maximum(a_new, 0.0)
        s2 = a_new.sum()
        if s2 < 1e-15:
            break
        a_new = a_new / s2
        a = a_new

    return a


def get_score(a):
    """Compute J(a) = 2*n*max_b/(sum_a)^2."""
    a = np.asarray(a, dtype=np.float64)
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    a_norm = a / s
    n = len(a_norm)
    b = fast_conv(a_norm)
    return float(2.0 * n * b.max())


def resample_seq(a, n_new):
    """Resample a 1D sequence to a new length using linear interpolation."""
    n_old = len(a)
    if n_old == n_new:
        return a.copy()
    x_old = np.linspace(0, 1, n_old)
    x_new = np.linspace(0, 1, n_new)
    return np.maximum(np.interp(x_new, x_old, a), 0.0)


def hard_deflation(a, frac, eta, max_iters, deadline):
    """Hard deflation: target only top `frac` fraction of peak coefficients."""
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return a
    a = a / s

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        b = fast_conv(a)
        M = b.max()
        if M < 1e-15:
            break
        threshold = M * (1.0 - frac)
        mask = (b >= threshold).astype(np.float64)
        if mask.sum() == 0:
            mask[np.argmax(b)] = 1.0
        p = mask / mask.sum()
        cc = compute_deflation_direction(a, p)
        a = a * np.exp(-eta * cc)
        a = np.maximum(a, 1e-20)
        a = a / a.sum()

    return a


def propose_candidate(seed=42, budget_s=1500, **kwargs):
    """
    Main search function. Combines:
    - Soft deflation with adaptive T (softmax-weighted peak reduction)
    - FISTA with smoothed max (accelerated gradient on smoothed objective)
    - LP best-response (principled monotone improvement)
    - Hard deflation (targeting top peaks)
    Across multiple lengths and diverse initializations.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + float(budget_s) - 30.0
    best = None
    best_score = float('inf')

    # Try parent sequence
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        if p.ndim == 1 and p.size > 0 and np.all(np.isfinite(p)):
            p = np.maximum(p, 0.0)
            if p.sum() > 0.01:
                sc = get_score(p)
                if sc < best_score:
                    best_score = sc
                    best = p.copy()
                    print(f"Parent: J={sc:.6f}, n={p.size}", flush=True)
    except (NameError, AttributeError):
        pass

    # Lengths to try
    lengths = [512, 1024, 2048]

    # === Phase 1: Soft deflation with T sweep (main search) ===
    for n in lengths:
        if time.monotonic() > deadline:
            break
        print(f"\n=== Phase 1: n={n} ===", flush=True)
        t_phase = time.monotonic()

        # Generate diverse starting points
        starts = []
        if best is not None:
            starts.append(resample_seq(best, n))
            # Perturbed version
            pstart = resample_seq(best, n)
            pstart = pstart + 0.05 * rng.standard_normal(n)
            starts.append(np.maximum(pstart, 0.01))
        t = np.linspace(0, 1, n, endpoint=False)
        starts.append(np.ones(n))
        starts.append(rng.uniform(0.1, 1.0, n))
        starts.append(1.0 + 0.4 * np.cos(2 * np.pi * t))
        starts.append(1.0 + 0.3 * np.cos(4 * np.pi * t + 0.5))
        starts.append(np.exp(-2.0 * (t - 0.5) ** 2) + 0.1)
        starts.append(1.0 / (1.0 + 20.0 * (t - 0.5) ** 2))

        for si, start in enumerate(starts):
            if time.monotonic() > deadline:
                break
            a0 = np.maximum(start, 0.01)
            s0 = a0.sum()
            if s0 < 0.01:
                continue
            a0 = a0 / s0

            # T sweep: start with large T, decrease
            b0 = fast_conv(a0)
            M0 = b0.max()
            T_values = [M0 * max(1.0, 5.0 / (it + 1)) for it in range(6)]
            # Ensure T is reasonable
            T_values = [max(tv, 1e-8) for tv in T_values]

            for T in T_values:
                if time.monotonic() > deadline:
                    break
                max_it = 150
                a_opt = optimize_soft_deflation(a0, T, max_it, deadline)
                sc = get_score(a_opt)
                if sc < best_score:
                    best_score = sc
                    best = a_opt.copy()
                    print(f"  Def n={n} T={T:.5f} start{si}: J={sc:.6f}", flush=True)
                a0 = a_opt

            # Try FISTA after deflation
            if time.monotonic() > deadline:
                break
            b_cur = fast_conv(a0)
            M_cur = b_cur.max()
            T_fista = M_cur * 0.8
            a_fista = optimize_fista_smoothed(a0, T_fista, 100, deadline)
            sc = get_score(a_fista)
            if sc < best_score:
                best_score = sc
                best = a_fista.copy()
                print(f"  FIST n={n} T={T_fista:.5f} start{si}: J={sc:.6f}", flush=True)
            a0 = a_fista

        elapsed = time.monotonic() - t_phase
        print(f"  Phase 1 n={n}: {elapsed:.1f}s, best={best_score:.6f}", flush=True)

    # === Phase 2: LP mirror descent polish ===
    if best is not None and time.monotonic() < deadline:
        print(f"\n=== Phase 2: LP Polish ===", flush=True)
        t_phase = time.monotonic()

        # Collect candidates to polish
        candidates = [best.copy()]
        # Add parent if different
        try:
            p2 = np.asarray(height_sequence_1, dtype=np.float64)
            if p2.ndim == 1 and p2.size in lengths and np.all(np.isfinite(p2)):
                p2 = np.maximum(p2, 0.0)
                if p2.sum() > 0.01:
                    candidates.append(p2 / p2.sum())
        except (NameError, AttributeError):
            pass

        polished = set()
        for cand in candidates:
            if time.monotonic() > deadline:
                break
            nc = cand.size
            key = (nc, round(get_score(cand), 4))
            if key in polished:
                continue
            polished.add(key)

            if nc > 1500:
                # Too large for LP, use deflation instead
                a_pol = optimize_soft_deflation(cand, cand.sum() ** 2 / (2.0 * nc) * 0.5, 200, deadline)
            else:
                max_lp_iters = 80 if nc <= 512 else 40
                a_pol = optimize_lp_descent(cand, max_lp_iters, deadline)

            sc = get_score(a_pol)
            if sc < best_score:
                best_score = sc
                best = a_pol.copy()
                print(f"  LP n={nc}: J={sc:.6f}", flush=True)

        print(f"  Phase 2: {time.monotonic()-t_phase:.1f}s, best={best_score:.6f}", flush=True)

    # === Phase 3: Final refinement with hard deflation + adaptive steps ===
    if best is not None and time.monotonic() < deadline:
        print(f"\n=== Phase 3: Final Refinement ===", flush=True)
        t_phase = time.monotonic()
        a = best.copy()

        # Multi-scale deflation
        for round_idx in range(8):
            if time.monotonic() > deadline:
                break
            b = fast_conv(a)
            M = b.max()
            eta = 0.2 * (0.8 ** round_idx)
            frac = 0.1 + 0.05 * round_idx  # Target more peaks over time
            a = hard_deflation(a, frac, eta, 40, deadline)
            sc = get_score(a)
            if sc < best_score:
                best_score = sc
                best = a.copy()
                print(f"  Hard def r{round_idx} frac={frac:.2f}: J={sc:.6f}", flush=True)
            else:
                a = best.copy()

        # Final FISTA polish
        if time.monotonic() > deadline:
            pass
        else:
            b = fast_conv(a / a.sum())
            M = b.max()
            for T_mult in [0.3, 0.1, 0.05]:
                if time.monotonic() > deadline:
                    break
                T = M * T_mult
                a_fin = optimize_fista_smoothed(a, T, 200, deadline)
                sc = get_score(a_fin)
                if sc < best_score:
                    best_score = sc
                    best = a_fin.copy()
                    print(f"  Final FISTA T={T:.5f}: J={sc:.6f}", flush=True)
                a = a_fin

        # Final LP if length allows
        if time.monotonic() < deadline and best.size <= 1024:
            a_final = optimize_lp_descent(best, 30, deadline)
            sc = get_score(a_final)
            if sc < best_score:
                best_score = sc
                best = a_final.copy()
                print(f"  Final LP: J={sc:.6f}", flush=True)

        print(f"  Phase 3: {time.monotonic()-t_phase:.1f}s, best={best_score:.6f}", flush=True)

    if best is None:
        best = np.ones(256)

    # Validate
    final_list = [float(max(0.0, min(1000.0, x))) for x in best]
    final_eval = evaluate_sequence(final_list)
    print(f"\nFINAL: J={final_eval:.6f}, n={best.size}", flush=True)
    return final_list
