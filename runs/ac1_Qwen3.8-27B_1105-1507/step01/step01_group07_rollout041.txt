"""
Advanced optimization for minimizing 2*n*max(conv(a,a))/(sum(a))^2.
Combines LP-based optimization, anti-symmetric binary local search,
and gradient-based polishing to beat the 1.5027 target.
"""

import numpy as np
from scipy.optimize import linprog, minimize
from scipy.sparse import csr_matrix
import time

def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    conv = np.convolve(a, a)
    max_b = float(np.max(conv))
    sum_a = float(np.sum(a))
    if sum_a < 0.01:
        return np.inf
    return 2.0 * n * max_b / (sum_a * sum_a)

def build_A_matrix(a):
    """Build the constraint matrix for the LP. Dense for small n."""
    n = len(a)
    A = np.zeros((2 * n - 1, n), dtype=np.float64)
    for k in range(2 * n - 1):
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        for j in range(j_start, j_end + 1):
            A[k, j] = a[k - j]
    return A

def build_A_sparse(a):
    """Build sparse constraint matrix."""
    n = len(a)
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        for j in range(j_start, j_end + 1):
            rows.append(k)
            cols.append(j)
            vals.append(a[k - j])
    return csr_matrix((vals, (rows, cols)), shape=(2*n-1, n))

def solve_lp(a, M):
    """Solve LP: max sum(g) s.t. (a*g)[k] <= M, 0 <= g."""
    n = len(a)
    A = build_A_sparse(a)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
    if res.status == 0:
        return res.x
    return None

def lp_optimize(a_init, max_iter=200, t=0.5, verbose=False):
    """LP-based iterative optimization."""
    n = len(a_init)
    a = a_init.copy()
    s = np.sum(a)
    if s < 1e-10:
        a = np.ones(n) / n
        s = 1.0
    a = a / s  # normalize to sum=1
    best_a = a.copy()
    best_obj = compute_obj(best_a)
    
    stall = 0
    for it in range(max_iter):
        conv = np.convolve(a, a)
        M = float(np.max(conv))
        
        g = solve_lp(a, M)
        if g is None:
            break
        
        g_sum = float(np.sum(g))
        if g_sum <= 1.0 + 1e-6:
            stall += 1
            if stall > 10:
                # Perturb
                a = best_a * np.exp(np.random.default_rng().normal(0, 0.5, n))
                a = np.clip(a, 0, None)
                s = np.sum(a)
                if s < 1e-10:
                    a = np.ones(n) / n
                    s = 1.0
                a = a / s
                stall = 0
            continue
        stall = 0
        
        g = g / g_sum
        a_new = (1 - t) * a + t * g
        a_new = np.clip(a_new, 0, None)
        s_new = np.sum(a_new)
        if s_new < 1e-10:
            break
        a_new = a_new / s_new
        
        obj_new = compute_obj(a_new)
        if obj_new < best_obj:
            best_a = a_new.copy()
            best_obj = obj_new
            if verbose and (it % 20 == 0):
                print(f"    LP it={it}: obj={best_obj:.6f}")
        a = a_new
    
    return best_a, best_obj

def antisymmetric_binary_search(n, max_iter=50000, seed=42):
    """Local search on anti-symmetric binary sequences."""
    rng = np.random.default_rng(seed)
    half = n // 2
    
    # Initialize
    a = np.zeros(n)
    for j in range(half):
        if rng.random() < 0.5:
            a[j] = 1.0
            a[n-1-j] = 0.0
        else:
            a[j] = 0.0
            a[n-1-j] = 1.0
    if n % 2 == 1:
        a[half] = 1.0  # middle element
    
    best_obj = compute_obj(a)
    best_a = a.copy()
    
    # Local search with simulated annealing
    T = 0.01
    T_min = 0.0001
    cooling = (T_min / T) ** (1.0 / max_iter)
    
    for it in range(max_iter):
        j = int(rng.integers(0, half))
        # Swap a[j] and a[n-1-j]
        a[j], a[n-1-j] = a[n-1-j], a[j]
        
        new_obj = compute_obj(a)
        delta = new_obj - best_obj
        
        if delta < 0 or rng.random() < np.exp(-delta / T):
            if new_obj < best_obj:
                best_obj = new_obj
                best_a = a.copy()
        else:
            a[j], a[n-1-j] = a[n-1-j], a[j]
        
        T *= cooling
    
    return best_a, best_obj

    n = len(a_init)
    a = a_init.copy().astype(np.float64)
    s = np.sum(a)
    a = a / s

        obj_grad -= np.mean(obj_grad)  # keep sum approximately constant
        obj_grad[np.where(a < 0.01)] += 1.0  # penalty for going negative

        a_new = np.clip(a_new, 0, None)
        s_new = np.sum(a_new)

            break

        if obj_new < best_obj:
            best_obj = obj_new
            best_a = a_new.copy()
        
        a = a_new

        lr *= 0.999
