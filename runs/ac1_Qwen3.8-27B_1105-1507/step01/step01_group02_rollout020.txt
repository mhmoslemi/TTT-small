def project_simplex(v):
    """Project vector v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:

    u = np.sort(v)[::-1]
    cv = np.cumsum(u) - 1.0
    rho_candidates = np.where(u * np.arange(1, n + 1) > cv)[0]
    if len(rho_candidates) == 0:

    rho = rho_candidates[-1]

    result[result < 0] = 0.0

def project_simplex(v):
    """Project vector v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v
    v = np.asarray(v, dtype=np.float64)
    u = np.sort(v)[::-1]
    cv = np.cumsum(u) - 1.0
    rho_candidates = np.where(u * np.arange(1, n + 1) > cv)[0]
    if len(rho_candidates) == 0:
        return np.ones(n) / n
    rho = rho_candidates[-1]
    theta = cv[rho] / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def fast_convolve(a):
    """Compute full linear convolution of a with itself using FFT."""
    n = len(a)
    size = 2 * n - 1
    N = 1 << (size - 1).bit_length()
    fa = np.fft.rfft(a, N)
    fc = fa * fa
    result = np.fft.irfft(fc, N)[:size]
    result[result < 0] = 0.0
    return result

def compute_smooth_grad(a, c, tau):
    """
    Compute gradient of (1/tau)*log(sum(exp(tau*c))) w.r.t. a.
    Uses FFT-based cross-correlation.
    """
    n = len(a)
    # Softmax weights over c
    c_shifted = c - np.max(c)
    log_w = tau * c_shifted
    log_w -= np.max(log_w)
    w = np.exp(log_w)
    w /= np.sum(w)
    
    # grad[i] = 2 * sum_k w[k] * a[k-i] = 2 * cross_corr(w, a)[i]
    # Use FFT: DFT(cross_corr(w, a))[k] = DFT(w)[k] * conj(DFT(a)[k])
    N2 = 1 << (3 * n - 2).bit_length()
    fw = np.fft.rfft(w, N2)
    fa = np.fft.rfft(a, N2)
    corr = np.fft.irfft(fw * np.conj(fa), N2)
    grad = 2.0 * corr[:n]
    return grad


def solve_lp(a, time_limit=5.0):
    """
    Solve: min t s.t. (a*b)[k] <= t for all k, sum(b)=1, b>=0.
    Returns optimal b and t.
    """
    n = len(a)
    if n > 3000:
        return None, None
    
    # Build constraint matrix A of shape (2n-1, n)
    # A[k, j] = a[k-j] if 0 <= k-j < n

    for k in range(2 * n - 1):

        hi = min(n - 1, k)
        A[k, lo:hi + 1] = a[k - hi:k - lo + 1][::-1]

    #      b >= 0, t >= 0

    A_ub[:, n] = -1.0  # -t
    b_ub = np.zeros(2 * n - 1)
    
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])

    theta = cv[rho] / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def fast_convolve(a):
    """Compute full linear convolution of a with itself using FFT."""
    n = len(a)
    size = 2 * n - 1
    N = 1 << (size - 1).bit_length()
    fa = np.fft.rfft(a, N)
    fc = fa * fa
    result = np.fft.irfft(fc, N)[:size]
    result[result < 0] = 0.0
    return result


def alt_projection_init(n, rng, n_iter=250, target_taper=0.0, warm_start=None):
    """
    Alternating projection between non-negative simplex (time) and
    target-magnitude spectrum (frequency). Returns normalized non-negative sequence.
    """
    N = 1 << (2 * n - 2).bit_length()

    if warm_start is not None:
        a = warm_start.copy()
        F = np.fft.rfft(a, N)
    else:
        phases = rng.uniform(0.0, 2.0 * np.pi, N // 2 + 1)
        F = np.exp(1j * phases)
        F[0] = 1.0 + 0j
        a = np.ones(n) / n

    for it in range(n_iter):
        # Time domain projection: IFFT, clip non-negative, normalize
        a_full = np.fft.irfft(F, N)
        a = a_full[:n].copy()
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s < 1e-10:
            a = np.ones(n) / n
        else:
            a /= s

        # Frequency domain projection: set target magnitude
        F = np.fft.rfft(a, N)
        mag = np.abs(F)
        if target_taper > 0:
            idx = np.arange(len(mag), dtype=np.float64)
            frac = idx / max(1, len(mag) - 1)
            target_mag = np.ones(len(mag))
            target_mag = 1.0 - target_taper * frac
            target_mag = np.maximum(target_mag, 0.1)
        else:
            target_mag = np.ones(len(mag))

        # Blend for smoother convergence (more aggressive early on)
        alpha = 0.8 if it < n_iter // 3 else 0.95
        new_mag = alpha * target_mag + (1.0 - alpha) * mag
        F = new_mag * np.exp(1j * np.angle(F))
        F[0] = np.abs(F[0]) + 0j

    return a


def compute_smooth_grad(a, c, tau):
    """
    Compute gradient of (1/tau)*log(sum(exp(tau*c))) w.r.t. a.
    Uses FFT cross-correlation for efficiency.
    c = convolve(a, a), length 2n-1.
    """
    n = len(a)
    # Softmax weights
    c_shifted = c - np.max(c)
    exponent = tau * c_shifted
    exponent -= np.max(exponent)
    w = np.exp(exponent)
    w_sum = w.sum()
    if w_sum < 1e-300:
        w = np.ones(len(c)) / len(c)
    else:
        w /= w_sum

    # grad[i] = 2 * cross_corr(w, a)[i] = 2 * sum_k w[k] * a[k-i]
    # DFT(cross_corr(w, a))[k] = DFT(w)[k] * conj(DFT(a)[k])
    N2 = 1 << (3 * n - 2).bit_length()
    fw = np.fft.rfft(w, N2)
    fa = np.fft.rfft(a, N2)
    corr = np.fft.irfft(fw * np.conj(fa), N2)
    grad = 2.0 * corr[:n]
    return grad


def compute_hard_subgrad(a, c, top_k=10):
    """
    Compute subgradient of max(c) w.r.t. a, averaged over top-k peak positions.
    """
    n = len(a)
    size = 2 * n - 1
    N = 1 << (size - 1).bit_length()
    # Get top-k indices
    if top_k >= size:
        peak_idx = np.arange(size)
    else:
        peak_idx = np.argpartition(c, -top_k)[-top_k:]

    # grad[i] = 2 * sum_{k in peak_idx} a[k-i] / |peak_idx|
    # = 2 * cross_corr(indicator, a)[i] / |peak_idx|
    indicator = np.zeros(size)
    indicator[peak_idx] = 1.0 / len(peak_idx)

    N2 = 1 << (3 * n - 2).bit_length()
    fi = np.fft.rfft(indicator, N2)
    fa = np.fft.rfft(a, N2)
    corr = np.fft.irfft(fi * np.conj(fa), N2)
    grad = 2.0 * corr[:n]
    return grad


def solve_lp(a, time_limit=5.0):
    """
    Solve: min t s.t. (a*b)[k] <= t for all k, sum(b)=1, b>=0.
    Returns (b, t) or (None, None).
    """
    n = len(a)
    if n > 2000:
        return None, None

    # Build A[k,j] = a[k-j] if 0 <= k-j < n
    A = np.zeros((2 * n - 1, n), dtype=np.float64)
    for k in range(2 * n - 1):
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        A[k, lo:hi + 1] = a[k - hi:k - lo + 1][::-1]

    # min t, s.t. A*b - t <= 0, 1^T b = 1, b >= 0, t >= 0
    c_obj = np.zeros(n + 1)
    c_obj[n] = 1.0

    A_ub = np.zeros((2 * n - 1, n + 1))
    A_ub[:, :n] = A
    A_ub[:, n] = -1.0
    b_ub = np.zeros(2 * n - 1)

    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])

    bounds = [(0.0, None)] * n + [(0.0, None)]

    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs',
                      options={'time_limit': time_limit})
        if res.success:
            b = np.maximum(res.x[:n], 0.0)
            s = b.sum()
            if s > 1e-10:
                b /= s
            return b, res.x[n]
    except Exception:
        pass
    return None, None


def smooth_gradient_descent(a, deadline, rng, tau_init=10.0, max_iters=500000):
    """
    Nesterov smooth gradient descent to minimize max(convolve(a,a))
    subject to a >= 0, sum(a) = 1.
    """
    n = len(a)
    a = project_simplex(a.copy().astype(np.float64))

    c = fast_convolve(a)
    best_peak = np.max(c)
    best_a = a.copy()

    v = np.zeros(n)
    momentum = 0.85
    lr = 1.0 / np.sqrt(n)
    tau = tau_init
    stall = 0
    max_stall = 400
    it = 0
    last_improve_it = 0

    while time.time() < deadline and it < max_iters:
        it += 1

        # Nesterov lookahead
        y = a + momentum * v
        y = project_simplex(y)

        c_y = fast_convolve(y)
        peak_y = np.max(c_y)

        # Adaptive tau
        if it % 200 == 0:
            if stall > max_stall // 3:
                tau = min(tau * 1.5, 200.0)
            elif stall < max_stall // 10:
                tau = max(tau * 0.85, tau_init * 0.3)

        # Choose gradient type based on tau
        if tau < 5.0:
            grad = compute_hard_subgrad(y, c_y, top_k=min(20, 2 * n - 1))
        else:
            grad = compute_smooth_grad(y, c_y, tau)

        # Momentum update
        v = momentum * v - lr * grad
        a_new = a + v
        a = project_simplex(a_new)

        # Evaluate
        c_a = fast_convolve(a)
        peak_a = np.max(c_a)

        if peak_a < best_peak * (1.0 - 1e-10):
            best_peak = peak_a
            best_a = a.copy()
            stall = 0
            last_improve_it = it
        else:
            stall += 1

        # Adaptive LR
        if stall > max_stall:
            lr *= 0.6
            v *= 0.3
            stall = 0
            if lr < 1e-9 and time.time() < deadline - 30:
                # Random perturbation to escape
                scale = 0.05 + 0.1 * rng.random()
                a = project_simplex(a + rng.normal(0, scale / np.sqrt(n), n))
                v = np.zeros(n)
                lr = max(lr, 0.5 / np.sqrt(n))

        # Periodic momentum adjustment
        if it % 1000 == 0:
            momentum = 0.85 + 0.1 * (it % 3) / 3.0

    return best_a


def basin_hopping(a_best, deadline, rng, max_hops=50):
    """
    Basin hopping: perturb the best solution, re-optimize, keep the better.
    """
    n = len(a_best)
    current_best = a_best.copy()
    best_peak = np.max(fast_convolve(a_best))
    hop = 0

    while time.time() < deadline and hop < max_hops:
        hop += 1
        # Perturbation: random direction with adaptive scale
        scale = 0.1 + 0.2 * rng.random()
        perturbation = rng.normal(0, scale / np.sqrt(n), n)
        a_perturbed = project_simplex(current_best + perturbation)

        # Short re-optimization
        sub_deadline = min(deadline - 5, time.time() + max(10, (deadline - time.time()) / (max_hops - hop + 1) * 1.5))
        a_trial = smooth_gradient_descent(a_perturbed, sub_deadline, rng, tau_init=5.0, max_iters=50000)

        c_trial = fast_convolve(a_trial)
        peak_trial = np.max(c_trial)

        if peak_trial < best_peak * (1.0 - 1e-9):
            best_peak = peak_trial
            current_best = a_trial.copy()
            print(f"    Bassin hop {hop}: improved to peak={best_peak:.8f}")

    return current_best


def try_length(n, rng, deadline, init_type='alt_proj'):
    """Initialize and optimize a sequence of given length."""
    if init_type == 'alt_proj':
        a = alt_projection_init(n, rng, n_iter=250)
    elif init_type == 'alt_proj_taper':
        a = alt_projection_init(n, rng, n_iter=250, target_taper=0.3)
    elif init_type == 'random':
        a = rng.random(n)
        a /= a.sum()
    elif init_type == 'cosine':
        x = np.linspace(0.0, 1.0, n)
        a = 0.5 * (1.0 + np.cos(2.0 * np.pi * x)) + 0.005
        a /= a.sum()
    elif init_type == 'triangle':
        x = np.linspace(0.0, 1.0, n)
        a = 1.0 - np.abs(2.0 * x - 1.0)
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s < 1e-10:
            a = np.ones(n) / n
        else:
            a /= s
    elif init_type == 'parent':
        try:
            parent = np.array(height_sequence_1, dtype=np.float64)
            parent = np.clip(parent, 0.0, 1000.0)
            if len(parent) >= n:
                a = parent[:n].copy()
            else:
                a = np.zeros(n)
                a[:len(parent)] = parent
            a = np.maximum(a, 0.0)
            s = a.sum()
            a = a / s if s > 1e-10 else np.ones(n) / n
        except Exception:
            a = alt_projection_init(n, rng, n_iter=100)
    else:
        a = alt_projection_init(n, rng, n_iter=200)

    score_init = evaluate_sequence(a.tolist())

    # Main optimization
    a_opt = smooth_gradient_descent(a, deadline, rng)
    score = evaluate_sequence(a_opt.tolist())

    return a_opt, score


def lp_refine(a, deadline, rng, num_solves=5):
    """
    Use LP to find directions that reduce the cross-convolution peak,
    then try mixing with the current solution.
    """
    n = len(a)
    if n > 1500:
        return a

    best_a = a.copy()
    best_peak = np.max(fast_convolve(a))

    for _ in range(num_solves):
        if time.time() > deadline - 30:
            break
        b_lp, t_lp = solve_lp(best_a, time_limit=4.0)
        if b_lp is None:
            break

        # Try the LP solution directly
        c_lp = fast_convolve(b_lp)
        peak_lp = np.max(c_lp)
        if peak_lp < best_peak:
            best_peak = peak_lp
            best_a = b_lp.copy()
            continue

        # Try mixtures
        for t_mix in [0.2, 0.4, 0.6, 0.8]:
            a_mix = project_simplex((1.0 - t_mix) * best_a + t_mix * b_lp)
            c_mix = fast_convolve(a_mix)
            peak_mix = np.max(c_mix)
            if peak_mix < best_peak:
                best_peak = peak_mix
                best_a = a_mix.copy()
                break

    return best_a


def propose_candidate(budget_s=1500, **kwargs):
    """
    Multi-phase search:
    Phase 1: Multi-config smooth gradient descent from various initializations.
    Phase 2: LP refinement on best solutions.
    Phase 3: Basin hopping for further improvement.
    Phase 4: Final polish.
    """
    overall_deadline = time.time() + budget_s - 45.0
    rng = np.random.default_rng(7)

    best_score = float('inf')
    best_seq = None

    configs = [
        (512, 'alt_proj'),
        (512, 'alt_proj_taper'),
        (512, 'parent'),
        (768, 'alt_proj'),
        (1024, 'alt_proj'),
        (512, 'random'),
        (1024, 'alt_proj_taper'),
        (768, 'alt_proj'),
        (640, 'alt_proj'),
        (1280, 'alt_proj'),
    ]

    num_configs = len(configs)
    remaining = overall_deadline - time.time()

    # Phase 1: Multi-config optimization (60% of time)
    phase1_deadline = time.time() + remaining * 0.60
    first_budget = remaining * 0.60 / 3.0
    rest_budget = remaining * 0.30 / max(1, num_configs - 3)

    for i, (n, init_type) in enumerate(configs):
        if time.time() > phase1_deadline - 30:
            break

        if i < 3:
            config_deadline = min(phase1_deadline - 10, time.time() + first_budget)
        else:
            config_deadline = min(phase1_deadline - 10, time.time() + rest_budget)

        if config_deadline < time.time() + 15:
            continue

        try:
            a_opt, score = try_length(n, rng, config_deadline, init_type)

            # Quick LP refinement if time permits
            if n <= 1000 and time.time() < config_deadline - 20:
                a_lp = lp_refine(a_opt, config_deadline - 10, rng, num_solves=3)
                score_lp = evaluate_sequence(a_lp.tolist())
                if np.isfinite(score_lp) and score_lp < score:
                    a_opt = a_lp
                    score = score_lp

        except Exception as e:
            print(f"  Error in config ({n}, {init_type}): {e}")
            continue

        print(f"n={n}, init={init_type}, score={score:.6f}, time_left={overall_deadline - time.time():.0f}s")

        if np.isfinite(score) and score < best_score:
            best_score = score
            best_seq = a_opt.copy()
            print(f"  >>> New best: {best_score:.6f} (n={n})")

    # Phase 2: Basin hopping on best (20% of time)
    if best_seq is not None:
        remaining2 = overall_deadline - time.time()
        phase2_deadline = time.time() + remaining2 * 0.30
        if phase2_deadline > time.time() + 30:
            print("Phase 2: Basin hopping")
            n_b = len(best_seq)
            a_hopped = basin_hopping(best_seq, phase2_deadline, rng, max_hops=30)
            score_h = evaluate_sequence(a_hopped.tolist())
            print(f"  After basin hopping: score={score_h:.6f}")
            if np.isfinite(score_h) and score_h < best_score:
                best_score = score_h
                best_seq = a_hopped.copy()

    # Phase 3: LP refinement on best (10% of time)
    if best_seq is not None and len(best_seq) <= 1200:
        remaining3 = overall_deadline - time.time()
        phase3_deadline = time.time() + remaining3 * 0.30
        if phase3_deadline > time.time() + 20:
            print("Phase 3: LP refinement")
            a_lp = lp_refine(best_seq, phase3_deadline, rng, num_solves=10)
            score_lp = evaluate_sequence(a_lp.tolist())
            print(f"  After LP refinement: score={score_lp:.6f}")
            if np.isfinite(score_lp) and score_lp < best_score:
                best_score = score_lp
                best_seq = a_lp.copy()

    # Phase 4: Final polish (remaining time)
    if best_seq is not None and time.time() < overall_deadline - 10:
        print("Phase 4: Final polish")
        a_polished = smooth_gradient_descent(best_seq, overall_deadline, rng,
                                             tau_init=50.0, max_iters=200000)
        score = evaluate_sequence(a_polished.tolist())
        print(f"  After final polish: score={score:.6f}")
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_seq = a_polished

    # Fallback
    if best_seq is None:
        print("Falling back to uniform sequence")
        best_seq = np.ones(256)

    result = [float(x) for x in best_seq]
    final_score = evaluate_sequence(result)
    print(f"FINAL: score={final_score:.6f}, length={len(result)}")
    return result
