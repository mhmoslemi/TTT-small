import numpy as np
import time
from scipy.optimize import linprog

def fast_convolve(a):
    n = len(a)
    size = 2*n - 1
    N = 1 << (size-1).bit_length()
    fa = np.fft.rfft(a, N)
    result = np.fft.irfft(fa * fa, N)[:size]
    result[result < 0] = 0
    return result

def project_simplex(v):
    # Project onto {x >= 0, sum(x) = 1}
    n = len(v)
    u = np.sort(v)[::-1]
    cv = np.cumsum(u) - 1.0
    rho = np.where(u * np.arange(1, n+1) > cv)[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho = rho[-1]
    theta = cv[rho] / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def score_from_a(a):
    n = len(a)
    c = fast_convolve(a)
    s = a.sum()
    if s < 0.01:
        return np.inf
    return 2.0 * n * c.max() / s**2

def lp_refine(a, P, n_samples_period=50):
    """
    Solve LP: max sum(b) s.t. conv(a,b) <= P, b >= 0
    Returns b or None.
    """
    n = len(a)
    if n > 4096:
        return None
    
    # Build convolution matrix
    # A[k, j] = contribution of b[j] to c[k]
    # c[k] = sum_j a[k-j] * b[j] (when k-j is in [0, n-1])
    # So A[k, j] = a[k-j] if 0 <= k-j < n, else 0
    A = np.zeros((2*n-1, n), dtype=np.float64)
    for j in range(n):
        # b[j] contributes a[k-j]*b[j] to c[k] for k where 0 <= k-j < n
        # i.e., j <= k <= j+n-1
        A[j:j+n, j] = a
    
    # LP: min -sum(b) s.t. A*b <= P, b >= 0
    c_obj = -np.ones(n)
    b_ub = np.full(2*n-1, P)
    
    try:
        res = linprog(c_obj, A_ub=A, b_ub=b_ub, bounds=[(0, None)]*n, 
                     method='highs', options={'time_limit': 10})
        if res.success:
            return res.x
    except:
        pass
    return None
