Minimizes J(a) = 2n*max(conv(a,a))/(sum(a))² using:
1. FFT-accelerated multiplicative (mirror) descent with soft active-set weighting
2. Multi-scale search across sequence lengths
3. L-BFGS-B polishing on the best candidate
Key innovation: replace expensive LP with O(n log n) gradient steps,
enabling >10^6 iterations within the time budget.
"""

import numpy as np
from scipy.optimize import minimize
import time

def fast_conv(a):
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]

def project_simplex(v):
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1, dtype=np.float64)
    cond = (u - cssv / ind) > 0
    if not np.any(cond):
        return np.full(n, 1.0 / n)
    rho = np.where(cond)[0][-1] + 1
    theta = (u[:rho].sum() - 1.0) / rho
    return np.maximum(v - theta, 0.0)

    """
    Multiplicative (mirror) descent to minimize max_k (p*p)_k.
    Uses soft active-set weighting for the subgradient.
    """
    best_p = p.copy()
    best_val = float(fast_conv(p).max())
    t0 = time.monotonic()
    
    for t in range(max_iters):
        if time.monotonic() > deadline:
            break
        
        b = fast_conv(p)
        b_max = b.max()
        
        if b_max < best_val:
            best_val = b_max
            best_p = p.copy()

"""
Minimizes J(a) = 2n*max(conv(a,a))/(sum(a))² using:
1. FFT-accelerated multiplicative (mirror) descent with soft active-set weighting
2. Multi-scale search across sequence lengths (256-4096)
3. LP best-response polishing for exact refinement
4. L-BFGS-B gradient polishing on the final candidate
Key innovation: O(n log n) per iteration via FFT, enabling >10^6 iterations.

import numpy as np
from scipy.optimize import linprog, minimize
import time


def fast_conv(a):
    """Full self-convolution via FFT, O(n log n)."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]


def project_simplex(v):
    """Project v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1, dtype=np.float64)
    cond = (u - cssv / ind) > 0
    if not np.any(cond):
        return np.full(n, 1.0 / n)
    rho = np.where(cond)[0][-1] + 1
    theta = (u[:rho].sum() - 1.0) / rho
    return np.maximum(v - theta, 0.0)


def mirror_descent(p, n, max_iters, deadline, eta_init=0.5, eps_temp=0.01):
    """
    Multiplicative mirror descent to minimize max_k (p*p)_k.
    Returns (best_p, best_max_conv).
    """
    best_p = p.copy()
    b = fast_conv(p)
    best_val = float(b.max())
    stagnant = 0
    stall_threshold = best_val
    
    for t in range(max_iters):
        if time.monotonic() > deadline:
            break
        
        b = fast_conv(p)
        b_max = float(b.max())
        
        if b_max < best_val - 1e-12:
            best_val = b_max
            best_p = p.copy()
            stagnant = 0
            stall_threshold = best_val
        else:
            stagnant += 1
            if stagnant > 50000 and eta_init > 0.005:
                eta_init *= 0.5
                stagnant = 0
        
        # Soft active set weighting
        weights = np.exp((b - b_max) / eps_temp)
        w_sum = weights.sum()
        if w_sum < 1e-30:
            break
        weights /= w_sum
        
        # Compute weighted subgradient (vectorized over significant lags)
        g = np.zeros(n)
        significant = np.where(weights > 1e-8)[0]
        for k in significant:
            lo = max(0, k - n + 1)
            hi = min(n - 1, k)
            g[lo:hi + 1] += weights[k] * 2.0 * p[k - hi:k - lo + 1][::-1]
        
        # Adaptive step size
        eta = eta_init / (1.0 + 0.0005 * t)
        
        # Multiplicative update with numerical safety
        log_shift = -eta * g
        log_shift -= log_shift.max()
        p = p * np.exp(log_shift)
        # Floor to prevent underflow
        p = np.maximum(p, 1e-30)
        s = p.sum()
        if s < 1e-25:
            break
        p /= s
    
    return best_p, best_val


def build_A_matrix(a):
    """Build constraint matrix A where A[k,j] = a[k-j] if valid."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for k in range(2 * n - 1):
        jl = max(0, k - n + 1)
        jh = min(n - 1, k)
        if jl <= jh:
            A[k, jl:jh + 1] = a[k - jh:k - jl + 1][::-1]
    return A


def lp_best_response(a, M):
    """LP: max sum(g) s.t. (a*g)_k <= M, g >= 0. Returns g or None."""
    n = len(a)
    c = -np.ones(n)
    A = build_A_matrix(a)
    r = linprog(c, A_ub=A, b_ub=np.full(2 * n - 1, M),
                bounds=[(0, None)] * n, method='highs')
    if r.success:
        g = r.x
        if g.sum() > 1e-10:
            return g
    return None

        # Multiple runs with different hyperparameters
        for eta_init in [0.3, 0.1, 0.05, 0.02]:
            for eps_t in [0.005, 0.002, 0.001]:
                if time.monotonic() > phase2_end:
                    break
                p_in = np.clip(best_seq, 1e-15, None)
                p_in /= p_in.sum()
                # Small random perturbation
                p_in = p_in + rng.normal(0, 0.01, n)
                p_in = np.maximum(p_in, 1e-15)
                p_in /= p_in.sum()
                
                p_best, val_best = mirror_descent(
                    p_in, n, max_iters=2000000,
                    deadline=phase2_end,
                    eta_init=eta_init, eps_temp=eps_t
                )
                sc = score_sequence(p_best)
                if sc < best_score:
                    best_score = sc
                    best_seq = p_best.copy()
                    print(f"  refine (eta={eta_init}, eps={eps_t}): J={best_score:.6f}", flush=True)
        
        print(f"  Refinement: {time.monotonic()-t0:.1f}s, best={best_score:.6f}", flush=True)
    
    # === Phase 3: LP best-response polishing (15% of budget) ===
    phase3_end = min(time.monotonic() + 0.15 * total_budget, deadline)
    if best_seq is not None and best_seq.size <= 1024 and time.monotonic() < phase3_end:
        n = best_seq.size
        print(f"\n=== LP Polish: n={n} ===", flush=True)
        t0 = time.monotonic()
        
        a = np.clip(best_seq, 1e-10, None)
        a /= a.sum()
        
        for it in range(200):
            if time.monotonic() > phase3_end:
                break
            b = fast_conv(a)
            M = float(b.max())
            g = lp_best_response(a, M)
            if g is None:
                break
            g = np.maximum(g, 0.0)
            if g.sum() < 1e-10:
                break
            g /= g.sum()
            t_mix = max(0.05, 0.4 * (1.0 - it / 200.0))
            a_new = (1.0 - t_mix) * a + t_mix * g
            a_new = np.clip(a_new, 1e-10, None)
            a_new /= a_new.sum()
            a = a_new
        
        sc = score_sequence(a)
        if sc < best_score:
            best_score = sc
            best_seq = a.copy()
            print(f"  LP polish: J={best_score:.6f}", flush=True)
        
        print(f"  LP Polish: {time.monotonic()-t0:.1f}s, best={best_score:.6f}", flush=True)
    
    # === Phase 4: L-BFGS-B polishing (5% of budget) ===
    phase4_end = min(time.monotonic() + 0.05 * total_budget, deadline)
    if best_seq is not None and best_seq.size <= 2048 and time.monotonic() < phase4_end:
        n = best_seq.size
        print(f"\n=== L-BFGS: n={n} ===", flush=True)
        t0 = time.monotonic()
        
        p0 = np.clip(best_seq, 0.0, None)
        p0 /= p0.sum()
        
        def lbfgs_obj(x):
            ax = np.maximum(x, 0.0)
            s = ax.sum()
            if s < 1e-8:
                return 1e10
            ax = ax / s
            b = fast_conv(ax)
            return float(b.max())
        
        res = minimize(lbfgs_obj, p0, method='L-BFGS-B',
                       bounds=[(0, None)] * n,
                       options={'maxiter': 2000, 'max_fun': 20000,
                                'ftol': 1e-15, 'disp': False})
        if res.fun < 1e10:
            sc = score_sequence(res.x)
            if sc < best_score:
                best_score = sc
                best_seq = np.clip(res.x, 0.0, None)
                best_seq /= best_seq.sum()
                print(f"  L-BFGS: J={best_score:.6f}", flush=True)
        
        print(f"  L-BFGS: {time.monotonic()-t0:.1f}s, best={best_score:.6f}", flush=True)
    
    # === Phase 5: Try adjacent lengths (5% of budget) ===
    phase5_end = deadline
    if best_seq is not None and time.monotonic() < phase5_end:
        n = best_seq.size
        print(f"\n=== Length Search ===", flush=True)
        
        for n_try in [n // 2, n * 2, n + 128, n - 128, 384, 768, 1536, 3072]:
            if n_try < 64 or n_try > 4096 or time.monotonic() > phase5_end:
                continue
            if n_try == n:
                continue
            t_old = np.linspace(0, 1, n)
            t_new = np.linspace(0, 1, n_try)
            p = np.interp(t_new, t_old, best_seq)
            p = np.maximum(p, 0.0)
            if p.sum() < 0.01:
                continue
            p /= p.sum()
            
            sc = score_sequence(p)
            if sc < best_score:
                best_score = sc
                best_seq = p.copy()
                print(f"  n={n_try}: J={best_score:.6f}", flush=True)
            
            # Quick mirror descent
            p_best, _ = mirror_descent(p, n_try, max_iters=200000,
                                       deadline=phase5_end, eta_init=0.3, eps_temp=0.01)
            sc = score_sequence(p_best)
            if sc < best_score:
                best_score = sc
                best_seq = p_best.copy()
                print(f"  n={n_try} (MD): J={best_score:.6f}", flush=True)
    
    # Fallback
    if best_seq is None:
        best_seq = np.ones(256) / 256
        best_score = score_sequence(best_seq)
    
    print(f"\nFINAL: J={best_score:.6f}, n={best_seq.size}", flush=True)
    return [float(x) for x in best_seq]
