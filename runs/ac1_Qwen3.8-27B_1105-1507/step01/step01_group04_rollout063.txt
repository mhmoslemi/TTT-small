Optimizes non-negative step function heights to minimize 2n*max(conv(f,f))/(sum(f))^2.
Uses efficient coordinate descent with O(n) convolution updates, combined with
soft-max smoothed gradient descent and periodic perturbations.

import numpy as np
import time


def _fft_convolve(f):
    """Compute full linear convolution using FFT."""
    n = len(f)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    F = np.fft.rfft(f, size)
    h = np.fft.irfft(F * F, size)
    return h[:2 * n - 1]


def _score_from_conv(conv, f):
    """Compute score from precomputed convolution."""
    n = len(f)
    S = float(np.sum(f))
    if S < 0.01:
        return float('inf')
    M = float(np.max(conv))
    return 2.0 * n * M / (S * S)


def _score(f):
    """Compute score directly."""
    n = len(f)
    if n == 0:
        return float('inf')
    S = float(np.sum(f))
    if S < 0.01:
        return float('inf')
    conv = _fft_convolve(f)
    M = float(np.max(conv))
    return 2.0 * n * M / (S * S)


def _coord_descent(f_init, time_budget):

                        max_lin = 0.0

def _softmax_grad_descent(f_init, time_budget, tau_init=0.01, tau_min=0.001):
    """
    Gradient descent using soft-max smoothing of the max.
    Minimize 2n * M_smooth(f) / (sum f)^2 where M_smooth = tau * log(sum exp(conv_k/tau)).
    """
    f = f_init.astype(np.float64).copy()
    n = len(f)
    start = time.monotonic()
    
    # Normalize
    S = float(np.sum(f))
    if S < 1e-30:
        return f_init, float('inf')
    f = f / S
    best_score = _score(f)
    best_f = f.copy()
    
    tau = tau_init
    
    for outer in range(200):
        if time.monotonic() - start > time_budget:
            break
        
        tau = max(tau * 0.9, tau_min)
        
        # Inner iterations
        for inner in range(500):
            if time.monotonic() - start > time_budget:
                break
            
            conv = _fft_convolve(f)
            S = float(np.sum(f))
            
            # Soft-max weights
            shifted = conv / tau - np.max(conv) / tau  # numerical stability
            weights = np.exp(shifted)
            weights /= weights.sum()
            
            # Gradient of M_smooth with respect to f[i]
            # d M_smooth / d f[i] = sum_k weights[k] * d(conv_k)/d(f[i])
            # d(conv_k)/d(f[i]) = 2*f[k-i] (if 0<=k-i<n and k!=2i) or 2*f[i] (if k=2i)
            # Actually: d(conv_k)/d(f[i]) = f[k-i] + f[k-i] = 2*f[k-i] for valid k-i
            # But if k-i == i (i.e., k=2i), it's just f[i] (one term, derivative of f[i]^2 is 2*f[i])
            # Wait: conv_k = sum_j f[j]*f[k-j]. d/d(f[i]) = f[k-i] (from j=i term) + f[k-i] (from j=k-i term)
            # = 2*f[k-i] if k-i != i, or f[i] + f[i] = 2*f[i] if k-i = i.
            # So in all valid cases: d(conv_k)/d(f[i]) = 2*f[k-i] where 0 <= k-i < n.
            
            # Gradient computation:
            # grad[i] = sum_k weights[k] * 2*f[k-i] for 0 <= k-i < n
            #         = 2 * sum_{j=0}^{n-1} weights[i+j] * f[j]  (where i+j < 2n-1)
            
            grad = np.zeros(n)
            for i in range(n):
                k_start = max(0, i)
                k_end = min(2 * n - 2, i + n - 1)
                j_start = k_start - i
                j_end = k_end - i
                grad[i] = 2.0 * float(np.dot(weights[k_start:k_end+1], f[j_start:j_end+1]))
            
            # Full gradient of score = 2n * M_smooth / S^2
            # d(score)/d(f[i]) = 2n / S^2 * grad_M[i] - 2 * 2n * M_smooth / S^3
            M_smooth = tau * (np.log(np.sum(np.exp(conv / tau - np.max(conv)/tau))) + np.max(conv)/tau)
            score_grad = (2.0 * n / (S * S)) * grad - (4.0 * n * M_smooth / (S * S * S))
            
            # Gradient step with line search
            # Projected onto f >= 0
            step_size = 0.1
            best_f_trial = f.copy()
            best_s_trial = best_score
            
            for lr in [0.5, 0.3, 0.1, 0.05, 0.01]:
                f_trial = f - lr * score_grad
                f_trial = np.maximum(f_trial, 0.0)
                S_trial = float(np.sum(f_trial))
                if S_trial < 1e-30:
                    continue
                f_trial = f_trial / S_trial  # renormalize
                s_trial = _score(f_trial)
                if s_trial < best_s_trial:
                    best_s_trial = s_trial
                    best_f_trial = f_trial.copy()
                    break
            
            if best_s_trial < best_score:
                f = best_f_trial
                best_score = best_s_trial
                best_f = f.copy()
        
        if (outer + 1) % 20 == 0:
            print(f"    SGD outer={outer+1}: tau={tau:.6f}, best={best_score:.6f}")
    
    return best_f, best_score


def _pressure_descent(f_init, time_budget, eta_scale=0.3, threshold=0.7):
    """
    Mirror descent (multiplicative weight updates) with pressure vector.
    f_new = f * exp(-eta * p) / normalization
    where p[i] = average over active lags of 2*f[k-i].
    """
    f = f_init.astype(np.float64).copy()
    n = len(f)
    S = float(np.sum(f))
    if S < 1e-30:
        return f_init, float('inf')
    f = f / S
    best_score = _score(f)
    best_f = f.copy()
    start = time.monotonic()
    
    for it in range(200000):
        if time.monotonic() - start > time_budget:
            break
        
        conv = _fft_convolve(f)
        M = float(np.max(conv))
        S = float(np.sum(f))
        
        if M <= 0:
            break
        
        # Active set
        threshold_val = threshold * M
        active = np.where(conv >= threshold_val)[0]
        if len(active) == 0:
            active = np.array([np.argmax(conv)])
        
        # Pressure: p[i] = (1/|K|) * sum_{k in active} 2*f[k-i] for valid k-i
        p = np.zeros(n)
        for k in active:
            j_start = max(0, k - n + 1)
            j_end = min(n - 1, k)
            if j_end >= j_start:
                idx = k - np.arange(j_start, j_end + 1)
                p[j_start:j_end+1] += 2.0 * f[idx]
        p /= len(active)
        
        # Update
        eta = eta_scale / max(M, 1e-10)
        log_ratio = -eta * p
        log_ratio = np.clip(log_ratio, -20.0, 20.0)
        f = f * np.exp(log_ratio)
        f = np.maximum(f, 0.0)
        S = float(np.sum(f))
        if S < 1e-30:
            f = np.ones(n) / n
            S = 1.0
        f = f / S
        
        score = _score(f)
        if score < best_score:
            best_score = score
            best_f = f.copy()
    
    return best_f, best_score


def _perturb(f, rng, frac=0.05):
    """Additive perturbation with re-normalization."""
    n = len(f)
    noise = rng.random(n) * frac
    f_p = f + noise
    f_p = np.maximum(f_p, 0.0)
    S = float(np.sum(f_p))
    if S < 1e-30:
        return f.copy()
    return f_p / S


def _multiplicative_perturb(f, rng, frac=0.1):
    """Multiplicative perturbation."""
    n = len(f)
    noise = rng.random(n)
    f_p = f * (1.0 + frac * (noise - 0.5))
    f_p = np.maximum(f_p, 0.0)
    S = float(np.sum(f_p))
    if S < 1e-30:
        return f.copy()
    return f_p / S


def _cosine_init(n):
    j = np.arange(n, dtype=np.float64)
    f = np.cos(np.pi * (j + 0.5) / n) ** 2
    f = np.maximum(f, 0.0)
    s = np.sum(f)
    if s > 0:
        f /= s
    return f


def _tapered_init(n, power=2.0):
    j = np.arange(n, dtype=np.float64)
    x = np.sin(np.pi * (j + 0.5) / n)
    f = x ** power
    f = np.maximum(f, 0.0)
    s = np.sum(f)
    if s > 0:
        f /= s
    return f


def _gaussian_init(n, sigma_frac=0.3):
    j = np.arange(n, dtype=np.float64) - (n - 1) / 2.0
    sigma = sigma_frac * n
    f = np.exp(-j * j / (2.0 * sigma * sigma))
    s = np.sum(f)
    if s > 0:
        f /= s
    return f

        f_res, s_res = _pressure_descent(f0, tb, eta_scale=0.3, threshold=0.7)
        _try_update(f_res, f"pressure_{name}")
        
        # Follow up with coordinate descent if improved
        if s_res < best_score + 0.01 and (deadline - time.monotonic()) > 15:
            cd_budget = min(30.0, (deadline - time.monotonic()) - 10)
            if cd_budget > 5:
                f_cd2, s_cd2 = _coord_descent(f_res, cd_budget)
                _try_update(f_cd2, f"cd_after_pressure_{name}")
    
    # Phase 3: Try different lengths
    print("\n=== Phase 3: Different lengths ===")
    
    for n_try in [256, 384, 640, 768]:
        if time.monotonic() > deadline - 60:
            break
        
        for name, init in [
            ("cos", _cosine_init(n_try)),
            ("tap2", _tapered_init(n_try, 2.0)),
            ("tap1", _tapered_init(n_try, 1.0)),
            ("tap3", _tapered_init(n_try, 3.0)),
        ]:
            if time.monotonic() > deadline - 30:
                break
            tb = min(20.0, (deadline - time.monotonic()) - 10)
            if tb < 5:
                break
            
            f_res, s_res = _pressure_descent(init, tb, eta_scale=0.4, threshold=0.75)
            _try_update(f_res, f"n{n_try}_{name}_press")
            
            if (deadline - time.monotonic()) > 15:
                cd_tb = min(15.0, (deadline - time.monotonic()) - 10)
                if cd_tb > 3:
                    f_cd3, s_cd3 = _coord_descent(f_res, cd_tb)
                    _try_update(f_cd3, f"n{n_try}_{name}_cd")
    
    # Phase 4: Intensive refinement of best
    print("\n=== Phase 4: Intensive refinement ===")
    
    f_best_norm = best_f / float(np.sum(best_f))
    
    # Multiple coordinate descent passes with perturbations
    for round_num in range(20):
        if time.monotonic() > deadline - 15:
            break
        
        tb = min(30.0, (deadline - time.monotonic()) - 10)
        if tb < 5:
            break
        
        pert_frac = 0.02 * (1 + round_num * 0.5)
        f_pert = _perturb(f_best_norm, rng, min(pert_frac, 0.2))
        
        f_res, s_res = _coord_descent(f_pert, tb)
        _try_update(f_res, f"refine_cd_r{round_num}")
        
        # Also try pressure descent from the perturbed point
        if (deadline - time.monotonic()) > 10:
            tb2 = min(15.0, (deadline - time.monotonic()) - 10)
            if tb2 > 3:
                f_res2, s_res2 = _pressure_descent(f_pert, tb2, eta_scale=0.2, threshold=0.85)
                _try_update(f_res2, f"refine_press_r{round_num}")
    
    # Phase 5: Soft-max gradient descent on best
    if (deadline - time.monotonic()) > 30:
        print("\n=== Phase 5: Soft-max gradient descent ===")
        tb5 = min(120.0, (deadline - time.monotonic()) - 15)
        f_sg, s_sg = _softmax_grad_descent(f_best_norm, tb5)
        _try_update(f_sg, "softmax_gd")
        
        # Follow up with coordinate descent
        if (deadline - time.monotonic()) > 15:
            f_cd5, s_cd5 = _coord_descent(f_sg, (deadline - time.monotonic()) - 10)
            _try_update(f_cd5, "cd_after_softmax")
    
    # Final verification
    final_list = [float(max(0.0, min(1000.0, x))) for x in best_f]
    final_score = float(evaluate_sequence(final_list))
    print(f"\nFinal verified score: {final_score:.6f} (n={len(final_list)})")
    
    if not np.isfinite(final_score) or final_score > 2.0:
        print("  Falling back to parent.")
        return [float(max(0.0, min(1000.0, x))) for x in parent]
    
    return final_list
