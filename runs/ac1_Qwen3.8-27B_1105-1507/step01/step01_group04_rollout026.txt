def _softmax_mirror_descent(f_init, n_iter, tau, eta0, beta, label=""):
    """
    Nesterov-accelerated mirror descent with softmax smoothing.
    """
    f = np.asarray(f_init, dtype=np.float64).copy()
    f = _normalize(f)
    n = len(f)
    
    # Nesterov momentum
    m = f.copy()
    best_f = f.copy()
    best_score = _score_raw(f)
    
    for it in range(n_iter):
        # Nesterov: evaluate gradient at extrapolated point
        m = (1 + beta) * m - beta * f  # or m = f + beta * (f - f_prev)
        
        # Compute convolution
        h = _fft_convolve_full(m)  # or use linear convolution
        c = h  # h[k] = (m*m)[k]
        
        # Softmax weights
        c_max = np.max(c)
        exp_c = np.exp(tau * (c - c_max))
        w = exp_c / np.sum(exp_c)
        
        # Gradient: g[j] = sum_k w[k] * 2*m[k-j] (for valid k-j)
        # This is essentially a weighted sum of shifted versions of m
        g = np.zeros(n)
        for k in range(2*n-1):
            if w[k] > 1e-15:
                # (m*m)[k] derivative wrt m[j] is 2*m[k-j] for 0<=k-j<n
                j_start = max(0, k - n + 1)
                j_end = min(n - 1, k)
                if j_end >= j_start:
                    idx = np.arange(j_start, j_end+1)
                    m_idx = k - idx  # indices into m
                    g[idx] += w[k] * 2.0 * m[m_idx]
        
        # Mirror descent step
        eta = eta0 / (it + 1) ** 0.5  # or some schedule
        log_ratio = -eta * g
        log_ratio = np.clip(log_ratio, -20, 20)
        
        f_new = f * np.exp(log_ratio)
        f_new = _normalize(f_new)
        
        # Update momentum
        m = f_new
        
        score = _score_raw(f)
        if score < best_score:
            best_score = score
            best_f = f.copy()
    
    return best_f, best_score

"""
Optimizes non-negative step function to minimize 2n*max(conv(f,f))/(sum(f))^2.
Usessoftmax-smoothed mirror descent with Nesterov acceleration,
adaptive temperature cooling, and local perturbation restarts.
"""

import numpy as np
import time

def _fft_convolve(f):
    """Full linear convolution of f with itself via FFT."""
    n = len(f)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    F = np.fft.rfft(f, size)
    h = np.fft.irfft(F * F, size)
    return h[:2 * n - 1]

def _score_raw(f):
    """Compute score = 2n*max(conv(f,f))/sum(f)^2."""
    n = len(f)
    h = _fft_convolve(f)
    M = float(np.max(h))
    S = float(np.sum(f))
    if S < 0.01:
        return float('inf')
    return 2.0 * n * M / (S * S)

def _normalize(f):
    """Normalize to sum=1, clip to non-negative."""
    f = np.maximum(f, 0.0)
    s = float(np.sum(f))
    if s < 1e-30:
        return None
    return f / s

def _compute_gradient(m, w, n):
    """
    Compute gradient g[j] = 2 * sum_{d=0}^{n-1} m[d] * w[j+d] for j=0..n-1.
    Uses FFT cross-correlation.
    """
    size = 1
    while size < 3 * n - 1:
        size <<= 1
    
    m_ext = np.zeros(size)
    m_ext[:n] = m
    w_ext = np.zeros(size)
    w_ext[:2*n-1] = w
    
    M = np.fft.fft(m_ext)
    W = np.fft.fft(w_ext)
    
    # r[k] = sum_j m[j] * w[j+k] (linear cross-correlation)
    r = np.fft.ifft(M * np.conj(W)).real
    
    return 2.0 * r[:n]

def _md_step(f, m, eta, tau, label=""):
    """
    One step of Nesterov-accelerated mirror descent.
    Returns updated f and m.
    """
    n = len(f)
    
    # Nesterov extrapolation
    m = (1.0 + 0.9) * m - 0.9 * f  # beta = 0.9
    m = np.maximum(m, 1e-30)
    
    # Compute convolution of m with itself
    h = _fft_convolve(m)
    
    # Softmax weights
    c_max = np.max(h)
    if c_max < 1e-30:
        w = np.ones(2*n-1) / (2*n-1)
    else:
        exp_c = np.exp(tau * (h - c_max))
        w = exp_c / np.sum(exp_c)
    
    # Compute gradient
    g = _compute_gradient(m, w, n)
    
    # Mirror descent multiplicative update
    log_ratio = -eta * g
    log_ratio = np.clip(log_ratio, -20.0, 20.0)
    
    f_new = f * np.exp(log_ratio)
    f_new = _normalize(f_new)
    if f_new is None:
        f_new = np.ones(n) / n
    
    return f_new, m

def _mirror_descent_optimize(f0, max_iters, tau_start, tau_end, eta0, beta=0.9, label=""):
    """
    Main mirror descent optimization with adaptive temperature.
    """
    f = np.asarray(f0, dtype=np.float64).copy()
    f = _normalize(f)
    if f is None:
        return f0, float('inf')
    
    n = len(f)
    m = f.copy()
    best_f = f.copy()
    best_score = _score_raw(f)
    
    for it in range(max_iters):
        # Adaptive temperature: cool from tau_start to tau_end
        frac = (it + 1) / max_iters
        tau = tau_start + (tau_end - tau_start) * frac
        
        # Adaptive step size
        eta = eta0 / np.sqrt(it + 1)
        
        f, m = _md_step(f, m, eta, tau)
        
        score = _score_raw(f)
        if score < best_score:
            best_score = score
            best_f = f.copy()
        
        if (it + 1) % 500 == 0:
            print(f"      {label} it={it+1}: score={best_score:.6f}, tau={tau:.1f}")
    
    return best_f, best_score

def _local_perturb(f, rng, n_perturb, strength):
    """Perturb a few random coordinates."""
    n = len(f)
    f_p = f.copy()
    idx = rng.choice(n, size=min(n_perturb, n), replace=False)
    for i in idx:
        f_p[i] *= (1.0 + strength * (rng.random() - 0.5))
    f_p = np.maximum(f_p, 0.0)
    return _normalize(f_p)

def _init_exponential(n, alpha):
    j = np.arange(n, dtype=np.float64)
    f = np.exp(-alpha * j / n)
    return _normalize(f)

def _init_ramp(n):
    j = np.arange(n, dtype=np.float64)
    f = (j + 1.0)
    return _normalize(f)

def _init_bimodal(n):
    """Two bumps at 1/4 and 3/4."""
    j = np.arange(n, dtype=np.float64)
    f = np.exp(-4.0 * (j - n/4)**2 / n) + np.exp(-4.0 * (j - 3*n/4)**2 / n)
    return _normalize(f)

def _init_cosine(n):
    j = np.arange(n, dtype=np.float64)
    f = np.cos(np.pi * (j + 0.5) / n) ** 2
    f = np.maximum(f, 0.0)
    return _normalize(f)

def _init_asym_gaussian(n, center_frac, sigma_frac):
    j = np.arange(n, dtype=np.float64)
    center = center_frac * n
    sigma = sigma_frac * n
    f = np.exp(-0.5 * ((j - center) / sigma) ** 2)
    return _normalize(f)
