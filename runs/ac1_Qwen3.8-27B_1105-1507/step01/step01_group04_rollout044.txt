def _mirror_descent(f_init, n_iter, eta0, thresh_frac, time_budget=None, verbose=False):
    """Mirror descent with geometrically decaying step size and adaptive threshold."""
    f = np.asarray(f_init, dtype=np.float64).copy()
    f = _normalize(f)
    if f is None:
        return f_init.copy(), float('inf')
    n = len(f)
    best_f = f.copy()
    best_score = _score_fast(f)
    t0 = time.monotonic()
    decay = max(0.995, 1.0 - 1.0 / max(n_iter, 10))

    for it in range(n_iter):
        if time_budget is not None and (time.monotonic() - t0) > time_budget:
            break

        h = _fft_conv_full(f)
        M = float(np.max(h))
        if M <= 0:
            break

        thresh = thresh_frac * M
        active = np.where(h >= thresh)[0]
        if len(active) == 0:
            active = np.array([int(np.argmax(h))], dtype=np.int64)

        p = _pressure_vec(f, active)
        eta = eta0 * (decay ** it) / (M + 1e-30)
        log_factor = -np.clip(eta * p, -30.0, 30.0)
        f = f * np.exp(log_factor)
        f = _normalize(f)
        if f is None:
            f = np.ones(n) / n
            break

        s = _score_fast(f)
        if s < best_score:
            best_score = s
            best_f = f.copy()

        if verbose and (it + 1) % max(1, n_iter // 10) == 0:
            print(f"    MD[{it+1}/{n_iter}] score={best_score:.6f} M={M:.6f} |K|={len(active)}")

    return best_f, best_score


def _lp_frank_wolfe(f_init, max_iters, time_budget):
    """LP-based Frank-Wolfe polishing for n <= ~1200."""
    from scipy.optimize import linprog
    from scipy.sparse import coo_matrix

    f = np.asarray(f_init, dtype=np.float64).copy()
    f = _normalize(f)
    if f is None:
        return f_init.copy(), float('inf')
    n = len(f)
    best_f = f.copy()
    best_score = _score_fast(f)
    t0 = time.monotonic()

    # Precompute constraint matrix structure
    row_idx = []
    col_idx = []
    for j in range(n):
        rows_j = np.arange(j, j + n)
        row_idx.append(rows_j)
        col_idx.append(np.full(n, j, dtype=np.int32))
    row_idx_all = np.concatenate(row_idx)
    col_idx_all = np.concatenate(col_idx)

    for it in range(max_iters):
        if time.monotonic() - t0 > time_budget:
            break

        h = _fft_conv_full(f)
        M = float(np.max(h))

        vals = np.ones(2 * n - 1)
        for j in range(n):
            rows_j = np.arange(j, j + n)
            vals = np.where((row_idx_all >= j) & (row_idx_all < j + n), 1.0, vals)

        # Build sparse matrix: C[k, j] = f[k-j] if 0 <= k-j < n else 0
        # Rethink: C has shape (2n-1, n), C[k,j] = f[k-j] if 0<=k-j<n
        C_data = np.zeros(2 * n - 1)
        for j in range(n):
            k_start = j
            k_end = j + n
            C_data[k_start:k_end] = f  # f[k-j] = f[0..n-1] as k goes from j to j+n-1

        # Actually need to build it properly
        Cvals = []
        Crows = []
        Ccols = []
        for j in range(n):
            for dk in range(n):
                k = j + dk
                if 0 <= k < 2 * n - 1:
                    Cvals.append(f[dk])
                    Crows.append(k)
                    Ccols.append(j)

        C = coo_matrix((Cvals, (Crows, Ccols)), shape=(2 * n - 1, n))
        c_obj = -np.ones(n)
        b_ub = np.full(2 * n - 1, M)
        res = linprog(c_obj, A_ub=C, b_ub=b_ub,
                      bounds=[(0, None)] * n, method='highs',
                      options={'presolve': True, 'dual_feasibility_tolerance': 1e-8})

        if not res.success or res.x is None:
            break

        g = np.maximum(res.x, 0)
        Sg = float(np.sum(g))
        if Sg < 1e-10:
            break
        g = g / Sg

        t = 2.0 / (it + 2)
        f_new = (1.0 - t) * f + t * g
        f = _normalize(f_new)
        if f is None:
            break

        score = _score_fast(f)
        if score < best_score:
            best_score = score
            best_f = f.copy()

        if (it + 1) % 5 == 0:
            print(f"    LP[{it+1}] score={best_score:.6f}")

    return best_f, best_score


def _init_cosine(n):
    j = np.arange(n, dtype=np.float64)
    f = np.cos(np.pi * (j + 0.5) / n) ** 2
    return _normalize(f)


def _init_multi_gauss(n, rng):
    """Sum of 2-3 Gaussians at random positions."""
    f = np.zeros(n)
    n_components = rng.integers(2, 4)
    for _ in range(n_components):
        center = rng.uniform(0.1 * n, 0.9 * n)
        sigma = rng.uniform(0.1 * n, 0.4 * n)
        f += np.exp(-0.5 * ((np.arange(n) - center) / sigma) ** 2) * rng.uniform(0.5, 2.0)
    return _normalize(f)


def _init_taper(n, power):
    j = np.arange(n, dtype=np.float64)
    x = np.sin(np.pi * (j + 0.5) / n)
    f = x ** power
    return _normalize(f)


def _init_rippled(n, n_ridges, rng):
    """Multi-ridge initialization."""
    f = np.zeros(n)
    for _ in range(n_ridges):
        center = rng.uniform(0, n)
        width = rng.uniform(n * 0.05, n * 0.2)
        amp = rng.uniform(0.5, 1.5)
        f += amp * np.exp(-0.5 * ((np.arange(n) - center) / width) ** 2)
    f += 0.1  # small baseline
    return _normalize(f)


def _upsample_dft(f, factor):
    """Upsample f by 'factor' using DFT zero-padding, clip negatives."""
    n = len(f)
    new_n = n * factor
    F = np.fft.rfft(f, 2 * n - 1)
    # Zero-pad the spectrum
    new_size = 2 * new_n - 1
    F_new = np.zeros(len(F) * factor, dtype=np.complex128)
    F_new[:len(F)] = F
    # Inverse
    f_new = np.fft.irfft(F_new, new_size * 2)[:new_n]
    f_new = np.maximum(f_new, 0)
    return _normalize(f_new)


def _perturb(f, rng, frac):
    noise = rng.random(len(f)) * frac
    fp = f + noise
    return _normalize(fp)


def _coordinate_descent(f_init, n_sweeps, time_budget=None, verbose=False):
    """Cycle through coordinates, adjusting each to reduce the score."""
    f = np.asarray(f_init, dtype=np.float64).copy()
    f = _normalize(f)
    if f is None:
        return f_init.copy(), float('inf')
    n = len(f)
    best_f = f.copy()
    best_score = _score_fast(f)
    t0 = time.monotonic()

    for sweep in range(n_sweeps):
        if time_budget is not None and (time.monotonic() - t0) > time_budget:
            break
        for j in range(n):
            # Try a few multipliers for f[j]
            f[j] = max(0, f[j])
            for mult in [0.5, 0.75, 1.0, 1.3, 1.5]:
                f_try = f.copy()
                f_try[j] = f[j] * mult
                f_try = _normalize(f_try)
                if f_try is not None:
                    s = _score_fast(f_try)
                    if s < _score_fast(f):
                        f = f_try
            score = _score_fast(f)
            if score < best_score:
                best_score = score
                best_f = f.copy()
        if verbose and (sweep + 1) % 5 == 0:
            print(f"    CD[{sweep+1}/{n_sweeps}] score={best_score:.6f}")

    return best_f, best_score


def propose_candidate(seed=42, budget_s=1500):
    """
    Search for non-negative step function heights minimizing the evaluation score.
    Multi-scale progressive optimization with diverse initializations,
    mirror descent, and LP polishing.
    """
    deadline = time.monotonic() + float(budget_s) - 5.0
    rng = np.random.default_rng(seed)

    # Get parent sequence
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64).copy()
        parent = np.clip(parent, 0.0, 1000.0)
    except Exception:
        parent = np.ones(512, dtype=np.float64)

    parent_norm = _normalize(parent)
    if parent_norm is None:
        parent_norm = np.ones(512) / 512

    best_f = parent_norm.copy()
    best_score = _score_fast(best_f)
    print(f"Parent: n={len(best_f)}, score={best_score:.6f}")

    all_results = []  # list of (score, f, name)

    # === Phase 1: Multi-scale mirror descent with diverse inits ===
    sizes = [512, 1024, 2048]
    inits_per_size = {}

    for n in sizes:
        inits = []
        f_cos = _init_cosine(n)
        if f_cos is not None:
            inits.append(("cos", f_cos))
        f_tap1 = _init_taper(n, 1.0)
        if f_tap1 is not None:
            inits.append(("tap1", f_tap1))
        f_tap2 = _init_taper(n, 2.0)
        if f_tap2 is not None:
            inits.append(("tap2", f_tap2))
        f_tap4 = _init_taper(n, 4.0)
        if f_tap4 is not None:
            inits.append(("tap4", f_tap4))
        f_rip = _init_rippled(n, 3, rng)
        if f_rip is not None:
            inits.append(("rip3", f_rip))
        f_rip5 = _init_rippled(n, 5, rng)
        if f_rip5 is not None:
            inits.append(("rip5", f_rip5))
        f_mg = _init_multi_gauss(n, rng)
        if f_mg is not None:
            inits.append(("mg", f_mg))

        inits_per_size[n] = inits

    # Also add parent (resized if needed)
    for n in sizes:
        if n == len(parent_norm):
            inits_per_size[n].append(("parent", parent_norm.copy()))
        else:
            # Upsample or downsample parent
            if n > len(parent_norm):
                factor = n // len(parent_norm)
                if factor >= 2:
                    p_up = _upsample_dft(parent_norm, factor)
                    if p_up is not None:
                        inits_per_size[n].append(("parent_up", p_up))
            else:
                # Downsample by averaging
                n_parent = len(parent_norm)
                block = n_parent // n
                if block >= 1:
                    p_down = np.array([parent_norm[i * block:(i + 1) * block].mean()
                                       for i in range(n)])
                    p_dn = _normalize(p_down)
                    if p_dn is not None:
                        inits_per_size[n].append(("parent_dn", p_dn))

    # Allocate time for Phase 1
    phase1_time = min(400.0, (deadline - time.monotonic()) * 0.45)
    n_total_inits = sum(len(v) for v in inits_per_size.values())
    time_per_init = max(3.0, phase1_time / max(n_total_inits, 1))

    print(f"\nPhase 1: {n_total_inits} initializations, ~{time_per_init:.1f}s each")

    for n in sizes:
        for name, f0 in inits_per_size[n]:
            if time.monotonic() > deadline - 60:
                break
            t0 = time.monotonic()
            n_iter = int(max(200, min(3000, time_per_init * 300)))

            # Try two threshold fractions
            for tf in [0.75, 0.9]:
                f_md, s_md = _mirror_descent(f0, n_iter, 1.0, tf,
                                             time_budget=time_per_init * 0.8,
                                             verbose=False)
                tag = f"n{n}_{name}_tf{tf}"
                all_results.append((s_md, f_md, tag))
                if s_md < best_score:
                    best_score = s_md
                    best_f = f_md.copy()
                    print(f"  ** New best: {best_score:.6f} from {tag} **")

            elapsed = time.monotonic() - t0
            if elapsed > time_per_init:
                break

    # Sort results
    all_results.sort(key=lambda x: x[0])
    print(f"\nPhase 1 complete. Top 5:")
    for s, f, name in all_results[:5]:
        print(f"  {name}: {s:.6f} (n={len(f)})")

    # === Phase 2: Refine top candidates ===
    phase2_time = (deadline - time.monotonic()) * 0.35
    top_k = min(6, len(all_results))
    refined = []

    for idx in range(top_k):
        if time.monotonic() > deadline - 60:
            break
        s_cur, f_cur, name = all_results[idx]
        n = len(f_cur)
        remaining = deadline - time.monotonic() - 60
        if remaining < 10:
            break

        print(f"\nPhase 2: Refining {name} (n={n}, score={s_cur:.6f})")
        budget_here = min(phase2_time / max(top_k - idx, 1), remaining * 0.8)

        # More mirror descent with smaller eta and higher threshold
        for eta0 in [0.3, 0.5]:
            for tf in [0.85, 0.92, 0.95]:
                if budget_here < 3:
                    break
                n_iter = int(max(500, budget_here * 150))
                f_r, s_r = _mirror_descent(f_cur, n_iter, eta0, tf,
                                           time_budget=budget_here * 0.4,
                                           verbose=False)
                if s_r < s_cur:
                    s_cur = s_r
                    f_cur = f_r
                    if s_r < best_score:
                        best_score = s_r
                        best_f = f_r.copy()
                        print(f"  ** New best: {best_score:.6f} **")

        # Perturbation restart
        for frac in [0.02, 0.05, 0.1]:
            if budget_here < 3:
                break
            f_p = _perturb(f_cur, rng, frac)
            if f_p is not None:
                f_rp, s_rp = _mirror_descent(f_p, int(budget_here * 200), 0.8, 0.85,
                                             time_budget=budget_here * 0.3,
                                             verbose=False)
                if s_rp < s_cur:
                    s_cur = s_rp
                    f_cur = f_rp
                    if s_rp < best_score:
                        best_score = s_rp
                        best_f = f_rp.copy()
                        print(f"  ** New best (perturb): {best_score:.6f} **")

        refined.append((s_cur, f_cur, name + "_ref"))

    # === Phase 3: LP polishing for smaller sequences ===
    print("\nPhase 3: LP Polishing")
    for s_cur, f_cur, name in refined:
        if time.monotonic() > deadline - 30:
            break
        n = len(f_cur)
        if n > 1200:
            continue
        remaining = deadline - time.monotonic() - 20
        if remaining < 15:
            break
        lp_budget = min(remaining * 0.4, 60.0)
        print(f"  LP on {name} (n={n}), budget={lp_budget:.0f}s")
        f_lp, s_lp = _lp_frank_wolfe(f_cur, max_iters=int(lp_budget // 2),
                                     time_budget=lp_budget)
        if s_lp < s_cur:
            s_cur = s_lp
            f_cur = f_lp
            if s_lp < best_score:
                best_score = s_lp
                best_f = f_lp.copy()
                print(f"  ** New best (LP): {best_score:.6f} **")

        # Mirror descent after LP
        if (deadline - time.monotonic()) > 15:
            f_post, s_post = _mirror_descent(f_cur, 2000, 0.2, 0.92,
                                             time_budget=min(30.0, (deadline - time.monotonic()) - 15),
                                             verbose=False)
            if s_post < best_score:
                best_score = s_post
                best_f = f_post.copy()
                print(f"  ** New best (post-LP MD): {best_score:.6f} **")

    # === Phase 4: Final polishing on the best ===
    if (deadline - time.monotonic()) > 20:
        print(f"\nPhase 4: Final polishing (score={best_score:.6f})")
        remaining = (deadline - time.monotonic()) - 5
        for eta0 in [0.1, 0.15, 0.2, 0.3]:
            if (deadline - time.monotonic()) < 5:
                break
            f_fin, s_fin = _mirror_descent(best_f, int(remaining * 100), eta0, 0.93,
                                           time_budget=(deadline - time.monotonic()) - 8,
                                           verbose=False)
            if s_fin < best_score:
                best_score = s_fin
                best_f = f_fin.copy()
                print(f"  ** Final best: {best_score:.6f} (eta={eta0}) **")

    # === Verification ===
    final_list = [float(x) for x in best_f]
    final_score = float(evaluate_sequence(final_list))
    print(f"\nFinal verified score: {final_score:.6f} (n={len(final_list)})")

    if not np.isfinite(final_score) or final_score > best_score + 0.01:
        print("  Fallback to parent.")
        return [float(x) for x in parent]

    return final_list
