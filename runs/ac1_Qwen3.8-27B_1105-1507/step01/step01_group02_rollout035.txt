"""
Optimizes a nonnegative sequence to minimize 2*n*max(convolve(a,a))/sum(a)^2.
Uses log-sum-exp smoothed objective with Adam optimizer, FFT-based gradients,
alternating projection initialization, and LP-based local refinement.
"""

import numpy as np
import time
from scipy.optimize import linprog


def fast_convolve(a):
    """Compute full linear convolution of a with itself using FFT."""
    n = len(a)
    size = 2 * n - 1
    N = 1 << (size - 1).bit_length()
    fa = np.fft.rfft(a, N)
    fc = fa * fa
    result = np.fft.irfft(fc, N)[:size]
    result[result < 0] = 0.0
    return result


def project_simplex(v):
    """Project vector v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cv = np.cumsum(u) - 1.0
    rho_candidates = np.where(u * np.arange(1, n + 1) > cv)[0]
    if len(rho_candidates) == 0:
        return np.ones(n) / n
    rho = rho_candidates[-1]
    theta = cv[rho] / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def cross_correlate_valid(v, a):
    """Compute out[j] = sum_{i=0}^{n-1} v[j+i] * a[i] for j = 0..len(v)-len(a).
    Uses FFT for efficiency."""
    n = len(a)
    m = len(v)
    if m < n:
        return np.array([])
    size = m + n - 1
    N = 1 << (size - 1).bit_length()
    V = np.fft.rfft(v, N)
    A_rev = np.fft.rfft(a[::-1].astype(np.float64), N)
    result = np.fft.irfft(V * A_rev, N)[:size]
    return result[n - 1:m]


def compute_score_from_conv(c, n):
    """Compute score from convolution (assuming sum(a)=1)."""
    return 2.0 * n * c.max()


def alt_projection_init(n, rng, n_iter=100):
    """
    Alternating projection (Gerchberg-Saxton) to get a nonnegative sequence
    with near-flat magnitude spectrum.
    """
    N = 1 << (2 * n - 2).bit_length()
    phases = rng.uniform(0.0, 2.0 * np.pi, N // 2 + 1)
    F = np.exp(1j * phases)
    F[0] = 1.0 + 0j
    
    a = np.ones(n) / n
    best_a = a.copy()
    best_peak = fast_convolve(a).max()
    
    for it in range(n_iter):
        a_full = np.fft.irfft(F, N)
        a = a_full[:n].copy()
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s < 1e-10:
            a = np.ones(n) / n
        else:
            a = a / s
        
        c = fast_convolve(a)
        if c.max() < best_peak:
            best_peak = c.max()
            best_a = a.copy()
        
        F_full = np.fft.rfft(a, N)
        F = np.ones(len(F_full), dtype=complex)
        F *= np.exp(1j * np.angle(F_full))
        F[0] = abs(F_full[0]) + 0j
    
    return best_a


def optimize_lse(a, deadline, rng, max_iters=500000):
    """
    Optimize using log-sum-exp smoothed objective with Adam optimizer.
    Minimizes max(conv(a,a)) subject to a >= 0, sum(a) = 1.
    """
    n = len(a)
    a = project_simplex(a.copy().astype(np.float64))
    
    # Adam state
    m_adam = np.zeros(n)
    v_adam = np.zeros(n)
    t_step = 0
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    
    # Log-sum-exp parameters
    beta_lse = 5.0
    beta_lse_max = 200.0
    beta_growth = 1.003
    
    learning_rate = 0.1 / np.sqrt(n)
    
    best_peak = fast_convolve(a).max()
    best_a = a.copy()
    
    stall_count = 0
    max_stall = 2000
    
    while time.time() < deadline and t_step < max_iters:
        t_step += 1
        
        c = fast_convolve(a)
        c_max = c.max()
        
        if c_max < best_peak * (1.0 - 1e-10):
            best_peak = c_max
            best_a = a.copy()
            stall_count = 0
        else:
            stall_count += 1
        
        # Compute LSE gradient
        scaled = beta_lse * (c - c_max)
        scaled = np.clip(scaled, -60, 0)
        exp_scaled = np.exp(scaled)
        w = exp_scaled / exp_scaled.sum()
        
        # grad[j] = beta_lse * sum_m a[m] * w[j+m]
        grad = beta_lse * cross_correlate_valid(w, a)
        
        # Adam update
        m_adam = beta1 * m_adam + (1 - beta1) * grad
        v_adam = beta2 * v_adam + (1 - beta2) * grad ** 2
        m_hat = m_adam / (1.0 - beta1 ** t_step)
        v_hat = v_adam / (1.0 - beta2 ** t_step)
        
        a_new = a - learning_rate * m_hat / (np.sqrt(v_hat) + eps)
        a = project_simplex(a_new)
        
        # Grow beta
        if beta_lse < beta_lse_max:
            beta_lse = min(beta_lse * beta_growth, beta_lse_max)
        
        # Decay learning rate slightly
        if t_step % 5000 == 0:
            learning_rate *= 0.95
        
        # Stall detection: add perturbation
        if stall_count > max_stall:
            noise_scale = 0.02 / np.sqrt(n)
            a = project_simplex(a + rng.normal(0, noise_scale, n))
            stall_count = 0
            # Reset some Adam state
            v_adam *= 0.5
    
    return best_a


def lp_refine(a, deadline, rng, max_lp_calls=200):
    """
    LP-based refinement: repeatedly solve LP to find direction that reduces peak.
    """
    n = len(a)
    if n > 3000:
        return a
    
    best_a = a.copy()
    best_peak = fast_convolve(a).max()
    lp_count = 0
    
    A = np.zeros((2 * n - 1, n), dtype=np.float64)
    for k in range(2 * n - 1):
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        A[k, lo:hi + 1] = a[k - hi:k - lo + 1][::-1]
    
    c_obj = -np.ones(n)
    
    while time.time() < deadline and lp_count < max_lp_calls:
        lp_count += 1
        c = fast_convolve(best_a)
        P = c.max() * 0.999  # Slightly below current peak
        
        b_ub = np.full(2 * n - 1, P)
        
        try:
            res = linprog(c_obj, A_ub=A, b_ub=b_ub,
                          bounds=[(0.0, None)] * n, method='highs',
                          options={'time_limit': 2.0})
            if res.success:
                b = res.x
                b_sum = b.sum()
                if b_sum > 1e-8:
                    b = b / b_sum  # Normalize to sum=1
                    # Blend
                    t_mix = 0.3 + 0.4 * rng.random()
                    a_new = (1.0 - t_mix) * best_a + t_mix * b
                    a_new = np.maximum(a_new, 0.0)
                    s = a_new.sum()
                    if s > 1e-10:
                        a_new = a_new / s
                        c_new = fast_convolve(a_new)
                        if c_new.max() < best_peak:
                            best_peak = c_new.max()
                            best_a = a_new.copy()
                    # Rebuild A matrix with new a
                    A_new = np.zeros((2 * n - 1, n), dtype=np.float64)
                    for k in range(2 * n - 1):
                        lo = max(0, k - n + 1)
                        hi = min(n - 1, k)
                        A_new[k, lo:hi + 1] = best_a[k - hi:k - lo + 1][::-1]
                    A = A_new
        except Exception:
            break
    
    return best_a


def coordinate_descent_refine(a, deadline):
    """
    Try small perturbations of individual coordinates to reduce the peak.
    """
    n = len(a)
    best_a = a.copy()
    best_peak = fast_convolve(a).max()
    
    perturbations = [0.01, 0.005, 0.001, -0.01, -0.005, -0.001]
    
    it = 0
    while time.time() < deadline:
        it += 1
        improved = False
        for j in range(n):
            if time.time() > deadline:
                break
            for delta in perturbations:
                a_new = best_a.copy()
                a_new[j] += delta
                if a_new[j] < 0:
                    continue
                s = a_new.sum()
                if s < 0.01:
                    continue
                a_norm = a_new / s
                c = fast_convolve(a_norm)
                if c.max() < best_peak:
                    best_peak = c.max()
                    best_a = a_norm.copy()
                    improved = True
                    break
        
        if not improved:
            # Reduce perturbation size
            perturbations = [p * 0.5 for p in perturbations]
            if max(abs(p) for p in perturbations) < 1e-5:
                break
    
    return best_a

"""
Optimizes a nonnegative sequence to minimize 2*n*max(convolve(a,a))/sum(a)^2.
Uses log-sum-exp smoothed objective with Adam optimizer, FFT-based gradients,
alternating projection (Gerchberg-Saxton) initialization, LP-based local
refinement, and coordinate descent polishing. Tries multiple sequence lengths
and initializations with adaptive time allocation.
"""

import numpy as np
import time
from scipy.optimize import linprog


def fast_convolve(a):
    """Compute full linear convolution of a with itself using FFT."""
    n = len(a)
    size = 2 * n - 1
    N = 1 << (size - 1).bit_length()
    fa = np.fft.rfft(a, N)
    fc = fa * fa
    result = np.fft.irfft(fc, N)[:size]
    result[result < 0] = 0.0
    return result


def project_simplex(v):
    """Project vector v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cv = np.cumsum(u) - 1.0
    rho_candidates = np.where(u * np.arange(1, n + 1) > cv)[0]
    if len(rho_candidates) == 0:
        return np.ones(n) / n
    rho = rho_candidates[-1]
    theta = cv[rho] / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def cross_correlate_valid(v, a):
    """Compute out[j] = sum_{i=0}^{n-1} v[j+i] * a[i] for j = 0..len(v)-len(a)."""
    n = len(a)
    m = len(v)
    if m < n:
        return np.zeros(0)
    size = m + n - 1
    N = 1 << (size - 1).bit_length()
    V = np.fft.rfft(v.astype(np.float64), N)
    A_rev = np.fft.rfft(a[::-1].astype(np.float64), N)
    result = np.fft.irfft(V * A_rev, N)[:size]
    return result[n - 1:m]


def alt_projection_init(n, rng, n_iter=100):
    """Gerchberg-Saxton alternating projection for flat-spectrum nonneg sequence."""
    N = 1 << (2 * n - 2).bit_length()
    phases = rng.uniform(0.0, 2.0 * np.pi, N // 2 + 1)
    F = np.exp(1j * phases)
    F[0] = 1.0 + 0j

    a = np.ones(n) / n
    best_a = a.copy()
    best_peak = fast_convolve(a).max()

    for it in range(n_iter):
        a_full = np.fft.irfft(F, N)
        a = a_full[:n].copy()
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s < 1e-10:
            a = np.ones(n) / n
        else:
            a = a / s

        c = fast_convolve(a)
        if c.max() < best_peak:
            best_peak = c.max()
            best_a = a.copy()

        F_full = np.fft.rfft(a, N)
        F = np.exp(1j * np.angle(F_full))
        F[0] = abs(F_full[0]) + 0j

    return best_a


def optimize_lse(a, deadline, rng):

    Minimizes max(conv(a,a)) subject to a >= 0, sum(a) = 1.

    beta2 = 0.999
    eps = 1e-8

    beta_lse = 5.0
    beta_lse_max = 300.0
    beta_growth = 1.002

    learning_rate = 0.15 / np.sqrt(n)
    lr_decay = 0.9995

    best_a = a.copy()

            print(f"  Parent n=512: score={score:.6f}")
            if np.isfinite(score) and score < best_score:
                best_score = score
                best_seq = a_opt.copy()
                print(f"  >>> New best: {best_score:.6f}")
        except Exception as e:
            print(f"  Error: {e}")

    # Phase 2: Alternative initializations for n=512
    print("=== Phase 2: Alternative inits n=512 ===")
    p2_deadline = time.time() + (budget_s * 0.20)
    remaining = p2_deadline - time.time()
    n_configs = 3
    per_config = remaining / n_configs

    for i, init_type in enumerate(['alt_proj', 'alt_proj_long', 'parent_noise']):
        if time.time() > p2_deadline - 10:
            break
        if per_config < 15:
            break
        try:
            a_opt = run_config(512, init_type, per_config, rng)
            score = evaluate_sequence(a_opt.tolist())
            print(f"  n=512, {init_type}: score={score:.6f}")
            if np.isfinite(score) and score < best_score:
                best_score = score
                best_seq = a_opt.copy()
                print(f"  >>> New best: {best_score:.6f}")
        except Exception as e:
            print(f"  Error ({init_type}): {e}")

    # Phase 3: Other n values
    print("=== Phase 3: Other lengths ===")
    p3_deadline = time.time() + (budget_s * 0.20)
    other_configs = [
        (1024, 'alt_proj'),
        (256, 'alt_proj'),
        (768, 'alt_proj'),
        (1024, 'cosine'),
    ]
    n_other = len(other_configs)
    per_config = (p3_deadline - time.time()) / n_other

    for n, init_type in other_configs:
        if time.time() > p3_deadline - 10:
            break
        if per_config < 15:
            break
        try:
            a_opt = run_config(n, init_type, per_config, rng)
            score = evaluate_sequence(a_opt.tolist())
            print(f"  n={n}, {init_type}: score={score:.6f}")
            if np.isfinite(score) and score < best_score:
                best_score = score
                best_seq = a_opt.copy()
                print(f"  >>> New best: {best_score:.6f}")
        except Exception as e:
            print(f"  Error (n={n}, {init_type}): {e}")

    # Phase 4: Final polish on best sequence
    print("=== Phase 4: Final polish ===")
    if best_seq is not None and time.time() < overall_end - 5:
        n_best = len(best_seq)
        polish_time = overall_end - time.time() - 5
        print(f"  Polishing n={n_best}, time={polish_time:.0f}s")

        # LSE polish
        if polish_time > 30:
            lse_polish_time = min(polish_time * 0.4, 120)
            best_seq = optimize_lse(best_seq, time.time() + lse_polish_time, rng)

        # LP polish
        if n_best <= 2048 and time.time() < overall_end - 20:
            lp_polish_time = min((overall_end - time.time() - 10) * 0.4, 120)
            if lp_polish_time > 10:
                best_seq = lp_refine(best_seq, time.time() + lp_polish_time, rng)
            # Targeted LP
            if time.time() < overall_end - 15:
                tl_time = min((overall_end - time.time() - 5) * 0.5, 60)
                if tl_time > 5:
                    best_seq = targeted_lp_refine(best_seq, time.time() + tl_time, rng)

        # Coordinate descent
        if time.time() < overall_end - 3:
            cd_time = overall_end - time.time() - 3
            if cd_time > 2:
                best_seq = coordinate_descent_refine(best_seq, time.time() + cd_time)

        final_polish_score = evaluate_sequence(best_seq.tolist())
        print(f"  After polish: score={final_polish_score:.6f}")
        if np.isfinite(final_polish_score) and final_polish_score < best_score:
            best_score = final_polish_score

    # Fallback
    if best_seq is None:
        print("Fallback: trying quick n=512 alt_proj")
        try:
            a = alt_projection_init(512, rng, n_iter=50)
            a = optimize_lse(a, time.time() + 60, rng)
            best_seq = a
        except Exception:
            best_seq = np.ones(256)

    result = [float(x) for x in best_seq]
    final_score = evaluate_sequence(result)
    print(f"FINAL: score={final_score:.6f}, length={len(result)}")
    return result
