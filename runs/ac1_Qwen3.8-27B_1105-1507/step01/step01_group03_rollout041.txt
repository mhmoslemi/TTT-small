def normalize_sequence(arr):
    x = np.asarray(arr, dtype=np.float64).ravel()
    if x.size == 0:
        return None
    x = np.where(np.isfinite(x), x, 0.0)
    x = np.maximum(x, 0.0)
    s = float(x.sum())
    if s <= 1e-300:
        return None
    return x / s


def score_normalized(x, n):
    conv = np.convolve(x, x)
    m = float(np.max(conv))
    return float(2.0 * n * m)


def project_simplex(v):
    n = len(v)
    if n == 0:
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.arange(1, n + 1)
    cond = u > (cssv - 1.0) / rho
    if not np.any(cond):
        return np.ones(n) / n
    idx = int(np.max(np.where(cond)[0]))
    rho_val = rho[idx]
    theta = (cssv[rho_val - 1] - 1.0) / rho_val
    return np.maximum(v - theta, 0.0)


def l2_gradient(f, n):
    conv = np.convolve(f, f)
    grad = 4.0 * np.convolve(conv, f, mode='valid')
    return grad, conv


def smooth_max_gradient(f, n, tau):
    conv = np.convolve(f, f)
    conv_scaled = conv / tau
    cmax = float(np.max(conv_scaled))
    exp_vals = np.exp(conv_scaled - cmax)
    wsum = float(np.sum(exp_vals))
    w = exp_vals / wsum
    LSE = cmax + np.log(wsum)
    cv = np.convolve(w, f, mode='valid')
    grad = (4.0 * n / tau) * cv - 4.0 * n * LSE
    return grad, conv, LSE

def make_conv_template(n):
    m = 2 * n - 1
    k64 = np.arange(m, dtype=np.int64)
    js = np.maximum(0, k64 - n + 1).astype(np.int32)
    je = np.minimum(n - 1, k64).astype(np.int32)
    counts = (je.astype(np.int64) - js.astype(np.int64) + 1).astype(np.int64)
    total = int(counts.sum())
    cum = np.cumsum(counts)
    starts = np.repeat(cum - counts, counts)
    local = np.arange(total, dtype=np.int64) - starts
    row = np.repeat(k64, counts).astype(np.int32)
    col_base = js[row.astype(np.int64)]
    col = (col_base + local).astype(np.int32)
    offsets = (row.astype(np.int64) - col.astype(np.int64)).astype(np.int32)
    A_off = csr_matrix((offsets, (row, col)), shape=(m, n), dtype=np.int32)
    edge_offsets = A_off.data.astype(np.int32, copy=True)
    A_template = A_off.astype(np.float64).copy()
    A_template.data.fill(0.0)
    c_obj = -np.ones(n, dtype=np.float64)
    bounds = [(0.0, None)] * n
    return A_template, edge_offsets, c_obj, bounds, m


def solve_direction_lp(template, f, n, M):
    A_template, edge_offsets, c_obj, bounds, m = template
    lp_scale = float(n)
    A = A_template.copy()
    A.data[:] = f[edge_offsets]
    A.data *= lp_scale
    b = np.full(m, M * lp_scale * lp_scale, dtype=np.float64)
    return linprog(c_obj, A_ub=A, b_ub=b, bounds=bounds, method="highs")

def central_binomial_init(n):
    k = np.arange(n, dtype=np.float64)
    from math import lgamma
    log_c = lgamma(2 * k + 1) - lgamma(k + 1) - k * np.log(4.0)
    f = np.exp(log_c)
    f_right = f[::-1]
    f = f + f_right
    s = f.sum()
    if s <= 0:
        return np.ones(n) / n
    return f / s

            # Phase 3: LP refinement
            rem = deadline - time.monotonic()
            if rem < 15.0:
                break
            tb_lp_actual = min(tb_lp, rem - 5)
            if tb_lp_actual > 10.0:
                try:
                    template = make_conv_template(n)
                    f3, s3 = optimize_lp(f2, n, tb_lp_actual, rng, template)
                    if s3 < best_score:
                        best_score = s3
                        best_f = f3.copy()
                        print(f"    LP result: {s3:.6f}", flush=True)
                    del template
                except Exception as e:
                    print(f"    LP failed: {e}", flush=True)

        # After all inits for this n, do one more LP pass on the best
        rem = deadline - time.monotonic()
        if rem > 30.0 and best_f is not None and best_f.size == n:
            print(f"  Final LP pass for n={n}, rem={rem:.0f}s", flush=True)
            try:
                template = make_conv_template(n)
                ff, sf = optimize_lp(best_f, n, min(rem - 10, 120.0), rng, template)
                if sf < best_score:
                    best_score = sf
                    best_f = ff.copy()
                    print(f"    Final LP: {sf:.6f}", flush=True)
                del template
            except Exception:
                pass

    # Final: try one more LP polish on the global best
    rem = deadline - time.monotonic()
    if rem > 20.0 and best_f is not None:
        n_best = best_f.size
        print(f"\nFinal polish: n={n_best}, rem={rem:.0f}s, score={best_score:.6f}", flush=True)
        try:
            template = make_conv_template(n_best)
            ff, sf = optimize_lp(best_f, n_best, min(rem - 5, 180.0), rng, template)
            if sf < best_score:
                best_score = sf
                best_f = ff.copy()
            print(f"  After final polish: {best_score:.6f}", flush=True)
            del template
        except Exception as e:
            print(f"  Final polish failed: {e}", flush=True)

    # Convert to list
    if best_f is None:
        seq = [1.0 / 64.0] * 64
    else:
        seq = [float(max(0.0, x)) for x in best_f]

    # Verify
    try:
        final_score = float(evaluate_sequence(seq))
    except Exception:
        final_score = float("inf")

    if not np.isfinite(final_score):
        if given_f is not None:
            seq = [float(max(0.0, x)) for x in given_f]
            try:
                final_score = float(evaluate_sequence(seq))
            except Exception:
                final_score = float("inf")
        if not np.isfinite(final_score):
            seq = [1.0 / 64.0] * 64

    print(f"\nFINAL: n={len(seq)}, score={final_score:.6f}", flush=True)
    return seq
