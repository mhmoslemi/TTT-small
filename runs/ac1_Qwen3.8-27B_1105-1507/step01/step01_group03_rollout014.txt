def normalize_sequence(arr):
    x = np.asarray(arr, dtype=np.float64).ravel()
    x = np.where(np.isfinite(x), x, 0.0)
    x = np.maximum(x, 0.0)
    s = x.sum()
    if s <= 1e-300:
        return None
    return x / s

def score_normalized(x):
    n = x.size
    conv = np.convolve(x, x)
    M = float(np.max(conv))
    return 2.0 * n * M

def project_onto_simplex(v):
    """Project v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = v.size
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n+1) > (cssv - 1))[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

    bounds = [(0.0, None)] * n

    try:
        res = linprog(c_obj, A_ub=A, b_ub=b, bounds=bounds, method='highs')
    except Exception:
        return None

    if res is None or not res.success or res.x is None:
        return None
    x = np.asarray(res.x, dtype=np.float64)
    x = np.maximum(x, 0.0)
    return x


def lse_gradient(f, tau):
    """Compute gradient of (1/tau)*log(sum exp(tau*(f*f)[k])) with respect to f."""
    n = f.size
    c = np.convolve(f, f)
    c_max = c.max()
    exp_vals = np.exp(tau * (c - c_max))
    total = exp_vals.sum()
    if total <= 0:
        return np.zeros(n), c
    w = exp_vals / total

    # Gradient: 2 * sum_k w[k] * f[k-i] = 2 * cross_correlate(w, f)[:n]
    fft_size = 1
    while fft_size < 3 * n:
        fft_size <<= 1
    fft_f = np.fft.rfft(f, fft_size)
    fft_w = np.fft.rfft(w, fft_size)
    corr = np.fft.irfft(fft_w * np.conj(fft_f), fft_size)[:n]
    grad = 2.0 * corr
    return grad, c


def smooth_optimize(f_init, time_budget, tau_start=20.0, tau_end=5000.0):
    """Smooth gradient descent with increasing tau."""
    f = normalize_sequence(f_init)
    if f is None:
        return f_init, float('inf')
    n = f.size
    deadline = time.monotonic() + time_budget
    best_f = f.copy()
    best_score = score_normalized(f)

    fft_size = 1
    while fft_size < 3 * n:
        fft_size <<= 1

    alpha = 0.5 / tau_start
    t0 = time.monotonic()
    stall = 0

    while time.monotonic() < deadline:
        elapsed = time.monotonic() - t0
        frac = min(1.0, elapsed / max(0.1, time_budget))
        tau = tau_start * (tau_end / tau_start) ** (frac ** 0.7)

        grad, c = lse_gradient(f, tau)
        c_max = c.max()

        # Backtracking line search
        best_trial = None
        best_trial_score = score_normalized(f)
        alpha_try = alpha
        for _ in range(15):
            f_new = project_onto_simplex(f - alpha_try * grad)
            sn = score_normalized(f_new)
            if sn < best_trial_score - 1e-12:
                best_trial = f_new
                best_trial_score = sn
                alpha = min(alpha_try * 1.1, 1.0 / (2.0 * tau * 1e-4))
                break
            alpha_try *= 0.5
            if alpha_try < 1e-15:
                break

        if best_trial is not None:
            f = best_trial
            if best_trial_score < best_score:
                best_score = best_trial_score
                best_f = f.copy()
                stall = 0
            else:
                stall += 1
        else:
            stall += 1
            alpha = max(alpha * 0.1, 1e-10)

        if stall > 200:
            # Perturb and reset
            f = project_onto_simplex(f + 0.01 * np.random.randn(n))
            alpha = 0.5 / tau
            stall = 0

    return best_f, best_score


def lp_optimize(f_init, time_budget, template, use_smooth_between=True):
    """LP-based optimization with optional smooth steps between LP calls."""
    f = normalize_sequence(f_init)
    if f is None:
        return f_init, float('inf')
    n = f.size
    deadline = time.monotonic() + time_budget
    best_f = f.copy()
    best_score = score_normalized(f)

    fft_size = 1
    while fft_size < 3 * n:
        fft_size <<= 1

    stall = 0
    it = 0

    while time.monotonic() < deadline:
        it += 1
        rem = deadline - time.monotonic()
        if rem < 2.0:
            break

        c_ff = np.convolve(f, f)
        M = float(c_ff.max())
        if M <= 0 or not np.isfinite(M):
            break

        # Solve LP
        x_star = solve_lp_direction(f, M, template, active_frac=1.0)

        if x_star is not None:
            s_star = x_star.sum()
            if s_star > 1.0 + 1e-8:
                g = x_star / s_star
                # Compute safe step
                r = np.convolve(f, g)
                r_max = float(r.max())
                G = float(np.max(np.convolve(g, g)))

                if r_max < M * (1 - 1e-10):
                    a_coef = 2.0 * (r_max - M)
                    b_coef = G + M - 2.0 * r_max
                    if b_coef > 1e-14:
                        t_max = -a_coef / b_coef
                    else:
                        t_max = 1.0
                    t_max = min(max(t_max, 1e-10), 1.0)

                    # Line search
                    best_t = None
                    best_score_local = best_score
                    for tv in [0.8 * t_max, 0.5 * t_max, 0.25 * t_max,
                               0.1 * t_max, 0.05 * t_max, 0.01 * t_max, 1e-4]:
                        if tv >= 1.0 or tv <= 0:
                            continue
                        f_new = project_onto_simplex((1 - tv) * f + tv * g)
                        sc = score_normalized(f_new)
                        if sc < best_score_local:
                            best_score_local = sc
                            best_t = tv

                    if best_t is not None:
                        f = project_onto_simplex((1 - best_t) * f + best_t * g)
                        if best_score_local < best_score:
                            best_score = best_score_local
                            best_f = f.copy()
                            stall = 0
                        else:
                            stall += 1
                    else:
                        stall += 1
                else:
                    stall += 1
            else:
                stall += 1
        else:
            stall += 1

        # Smooth steps between LP calls
        if use_smooth_between and rem > 5.0 and stall < 10:
            tau = max(50.0, 100.0 * (1.0 + 0.1 * it))
            smooth_budget = min(2.0, rem * 0.1)
            f, _ = smooth_optimize(f, smooth_budget, tau_start=tau, tau_end=tau * 2)

        if stall > 15:
            # Restart with noise
            f = project_onto_simplex(0.8 * best_f + 0.2 * np.random.dirichlet(np.ones(n)))
            stall = 0

        if it % 50 == 0:
            print(f"    LP it={it}, score={best_score:.6f}, M={M:.6g}, stall={stall}", flush=True)

    return best_f, best_score


def upscale_sequence(x, n):
    """Upsample/downsample sequence to length n via linear interpolation."""
    x = normalize_sequence(x)
    if x is None or n <= 0:
        return None
    m = x.size
    if m == n:
        return x
    if m == 1:
        return np.ones(n) / n
    old = np.linspace(0.0, 1.0, m)
    new = np.linspace(0.0, 1.0, n)
    out = np.interp(new, old, x)
    return normalize_sequence(out)


def mix_noisy(base, rng, eps):
    f = normalize_sequence(base)
    if f is None:
        return None
    n = f.size
    if eps <= 1e-12:
        return f
    if eps >= 1.0:
        return normalize_sequence(rng.dirichlet(np.ones(n)))
    mix = rng.dirichlet(np.ones(n))
    return normalize_sequence((1 - eps) * f + eps * mix)


def propose_candidate(seed=42, budget_s=1500, **kwargs):
    """
    Search for a low-scoring nonnegative sequence minimizing
    2*n*max(convolve(f,f))/(sum(f))^2.
    
    Strategy:
    1. Load previous best as primary start.
    2. Run LP-based optimization at n=1024 (main workhorse).
    3. Intersperse smooth gradient steps for faster local progress.
    4. Try n=2048 if time allows.
    5. Multi-start with diverse initializations to escape local minima.
    """
    rng = np.random.default_rng(int(seed) if seed is not None else 42)
    total_budget = max(60.0, float(budget_s))
    reserve = 15.0
    start_time = time.monotonic()
    deadline = start_time + total_budget - reserve

    best_f = None
    best_score = float('inf')
    given_f = None

    # Load given sequence
    if total_budget >= 30.0:
        try:
            raw = np.asarray(height_sequence_1, dtype=np.float64)
            given_f = normalize_sequence(raw)
            if given_f is not None:
                best_f = given_f.copy()
                best_score = score_normalized(given_f)
                print(f"Given: n={given_f.size}, score={best_score:.6f}", flush=True)
        except Exception:
            given_f = None

    if best_f is None:
        best_f = np.ones(1024) / 1024.0
        best_score = 2.0

    # Primary: n=1024 LP optimization
    n1 = 1024
    alloc1 = min(900.0, (deadline - time.monotonic()) * 0.7)
    if alloc1 > 30.0:
        print(f"\n=== Phase 1: n={n1}, budget={alloc1:.0f}s ===", flush=True)
        try:
            template1 = make_conv_matrix(n1)
        except Exception as e:
            print(f"  Template failed: {e}", flush=True)
            template1 = None

        if template1 is not None:
            # Start from given (upsampled to n1)
            f_start = upscale_sequence(best_f, n1)
            if f_start is None:
                f_start = np.ones(n1) / n1

            f_res, s_res = lp_optimize(f_start, alloc1 * 0.85, template1, use_smooth_between=True)
            if s_res < best_score:
                best_score = s_res
                best_f = f_res.copy()
                print(f"  Phase 1 result: score={best_score:.6f}", flush=True)

            # Second start: noisier
            rem = deadline - time.monotonic()
            if rem > 30.0:
                f_start2 = mix_noisy(best_f, rng, 0.15)
                f_res2, s_res2 = lp_optimize(f_start2, min(alloc1 * 0.15, rem - 5.0),
                                             template1, use_smooth_between=True)
                if s_res2 < best_score:
                    best_score = s_res2
                    best_f = f_res2.copy()
                    print(f"  Phase 1b result: score={best_score:.6f}", flush=True)

    # Phase 2: Smooth optimization refinement
    rem = deadline - time.monotonic()
    if rem > 30.0 and best_f is not None:
        print(f"\n=== Phase 2: Smooth refinement, budget={rem - 5:.0f}s ===", flush=True)
        n2 = best_f.size
        alloc2 = min(200.0, rem - 5.0)
        f_res, s_res = smooth_optimize(best_f, alloc2, tau_start=100.0, tau_end=10000.0)
        if s_res < best_score:
            best_score = s_res
            best_f = f_res.copy()
            print(f"  Phase 2 result: score={best_score:.6f}", flush=True)

    # Phase 3: Try n=2048 if time allows
    rem = deadline - time.monotonic()
    if rem > 120.0 and best_f is not None:
        print(f"\n=== Phase 3: n=2048 attempt, budget={rem - 5:.0f}s ===", flush=True)
        n3 = 2048
        try:
            template3 = make_conv_matrix(n3)
            f_start3 = upscale_sequence(best_f, n3)
            f_res3, s_res3 = lp_optimize(f_start3, min(300.0, rem - 10.0),
                                         template3, use_smooth_between=True)
            if s_res3 < best_score:
                best_score = s_res3
                best_f = f_res3.copy()
                print(f"  Phase 3 result: score={best_score:.6f}", flush=True)
        except Exception as e:
            print(f"  Phase 3 failed: {e}", flush=True)

    # Phase 4: Final LP polish at best n
    rem = deadline - time.monotonic()
    if rem > 20.0 and best_f is not None:
        print(f"\n=== Phase 4: Final polish, budget={rem - 3:.0f}s ===", flush=True)
        n4 = best_f.size
        try:
            template4 = make_conv_matrix(n4)
            f_res4, s_res4 = lp_optimize(best_f, rem - 5.0, template4, use_smooth_between=True)
            if s_res4 < best_score:
                best_score = s_res4
                best_f = f_res4.copy()
                print(f"  Phase 4 result: score={best_score:.6f}", flush=True)
        except Exception as e:
            print(f"  Phase 4 failed: {e}", flush=True)

    # Final evaluation
    if best_f is None:
        best_f = np.ones(64) / 64.0

    seq = [float(x) for x in best_f]
    try:
        final_score = float(evaluate_sequence(seq))
    except Exception:
        final_score = float('inf')

    if not np.isfinite(final_score):
        seq = [1.0 / 1024.0] * 1024
        final_score = float(evaluate_sequence(seq))

    print(f"\nFINAL: n={len(seq)}, local_best={best_score:.6f}, evaluate={final_score:.6f}", flush=True)
    return seq
