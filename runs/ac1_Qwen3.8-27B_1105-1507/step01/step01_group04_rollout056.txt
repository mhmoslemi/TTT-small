def simplex_project(v):
    """Project v onto the probability simplex {x: sum(x)=1, x>=0}."""
    n = len(v)
    # Sort in descending order
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.where(u * np.arange(1, n+1) > (cssv - 1))[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho = rho[-1]
    theta = (cssv[rho] - 1) / (rho + 1)
    return np.maximum(v - theta, 0)

def fw_step(f, M, active_lags, method='lp'):
    """
    Frank-Wolfe step: find g that minimizes <L, g> on the simplex.
    L is the subgradient of max_k (f*f)_k at f.
    """
    n = len(f)
    
    if method == 'lp':
        # The previous approach: maximize sum(g) s.t. (f*g)_k <= M, g >= 0
        # Build sparse constraint matrix
        rows = []
        cols = []
        vals = []
        for j in range(n):
            # (f*g)_k = sum_j f_{k-j} * g_j
            # For fixed j, f_{k-j} is nonzero when 0 <= k-j <= n-1, i.e., j <= k <= j+n-1
            for k in range(j, min(j+n, 2*n-1)):
                rows.append(k)
                cols.append(j)
                vals.append(f[k-j])
        
        C = coo_matrix((vals, (rows, cols)), shape=(2*n-1, n))
        
        c_obj = -np.ones(n)  # minimize -sum(g) = maximize sum(g)
        res = linprog(c_obj, A_ub=C, b_ub=np.full(2*n-1, M),
                      bounds=[(0, None)]*n, method='highs')
        if not res.success:
            return None
        g = res.x
        Sg = np.sum(g)
        if Sg < 1e-10:
            return None
        return g / Sg
    else:
        # Subgradient projection
        L = np.zeros(n)
        for k in active_lags:
            for j in range(n):
                k_minus_j = k - j
                if 0 <= k_minus_j < n:
                    L[j] += f[k_minus_j]
                if 2*j == k:
                    L[j] += f[j]
        L /= len(active_lags)
        return simplex_project(-L)

At iteration t, we have f_t. We solve the LP to get g_t. Then we want to find:
alpha* = argmin_{alpha in [0,1]} F((1-alpha)*f_t + alpha*g_t)

Since F is convex in its argument (and the argument is linear in alpha), F((1-alpha)*f_t + alpha*g_t) is convex in alpha. So we can use ternary search or golden section search to find alpha*.

"""
Optimizes non-negative step function heights to minimize 2n*max(conv(f,f))/(sum(f))^2.
Uses a combination of:
1. Diverse initializations (cosine, Gaussian, prolate-like)
2. Mirror descent with adaptive step size for coarse optimization
3. Frank-Wolfe with exact line search for refinement
4. Final polishing with projected gradient
"""

import numpy as np
import time
from scipy.optimize import linprog, minimize_scalar
from scipy.sparse import coo_matrix

def _fft_conv(f):
    n = len(f)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    F = np.fft.rfft(f, size)
    h = np.fft.irfft(F * F, size)
    return h[:2 * n - 1]

def _score(f):
    n = len(f)
    h = _fft_conv(f)
    M = np.max(h)
    S = np.sum(f)
    if S < 0.01:
        return np.inf
    return 2.0 * n * M / (S * S)

def _normalize(f):
    f = np.maximum(f, 0.0)
    s = np.sum(f)
    if s < 1e-30:
        return None
    return f / s

def _simplex_project(v):
    """Euclidean projection onto {x: sum(x)=1, x>=0}."""
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    rho = np.where(cond)[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho = rho[-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def _mirror_descent(f_init, n_iter, eta_scale, threshold_frac):
    """Mirror descent with multiplicative updates."""
    f = _normalize(np.asarray(f_init, dtype=np.float64))
    if f is None:
        return np.asarray(f_init, dtype=np.float64).copy(), np.inf
    n = len(f)
    best_f = f.copy()
    best_score = _score(f)
    
    for it in range(n_iter):
        h = _fft_conv(f)
        M = float(np.max(h))
        if M <= 0:
            break
        threshold = threshold_frac * M + 1e-15
        active = np.where(h >= threshold)[0]
        if len(active) == 0:
            active = np.array([int(np.argmax(h))])
        
        # Compute pressure vector
        p = np.zeros(n)
        for k in active:
            j_start = max(0, k - n + 1)
            j_end = min(n - 1, k)
            lo = k - j_end
            hi = k - j_start
            p[j_start:j_end + 1] += 2.0 * f[lo:hi + 1][::-1]
        p /= len(active)
        
        eta = eta_scale / M
        f = f * np.exp(np.clip(-eta * p, -20, 20))
        f_n = _normalize(f)
        if f_n is None:
            f = np.ones(n) / n
        else:
            f = f_n
        
        s = _score(f)
        if s < best_score:
            best_score = s
            best_f = f.copy()
    
    return best_f, best_score

def _fw_line_search(f, g, M):
    """Exact line search for FW: minimize F((1-t)f + t*g) over t in [0,1]."""
    n = len(f)
    
    def obj(t):
        ft = (1.0 - t) * f + t * g
        h = _fft_conv(ft)
        return float(np.max(h))
    
    result = minimize_scalar(obj, bounds=(0.0, 1.0), method='bounded',
                             options={'xatol': 1e-8})
    return result.x

def _fw_step_lp(f, M):
    """Solve the FW LP subproblem."""
    n = len(f)
    rows = []
    cols = []
    vals = []
    for j in range(n):
        for k in range(j, min(j + n, 2 * n - 1)):
            rows.append(k)
            cols.append(j)
            vals.append(f[k - j])
    
    C = coo_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))
    c_obj = -np.ones(n)
    res = linprog(c_obj, A_ub=C, b_ub=np.full(2 * n - 1, M),
                  bounds=[(0, None)] * n, method='highs')
    if not res.success or res.x is None:
        return None
    g = np.asarray(res.x, dtype=np.float64)
    Sg = np.sum(g)
    if Sg < 1e-10:
        return None
    return g / Sg

def _fw_refine(f_init, max_iters, time_budget, use_line_search=True):
    """Frank-Wolfe refinement."""
    f = _normalize(np.asarray(f_init, dtype=np.float64))
    if f is None:
        return np.asarray(f_init, dtype=np.float64).copy(), np.inf
    n = len(f)
    best_f = f.copy()
    best_score = _score(f)
    start = time.monotonic()
    
    for it in range(max_iters):
        if time.monotonic() - start > time_budget:
            break
        
        h = _fft_conv(f)
        M = float(np.max(h))
        
        g = _fw_step_lp(f, M)
        if g is None:
            break
        
        if use_line_search:
            t = _fw_line_search(f, g, M)
        else:
            t = 2.0 / (it + 2)
        
        f_new = (1.0 - t) * f + t * g
        f_n = _normalize(f_new)
        if f_n is None:
            break
        f = f_n
        
        s = _score(f)
        if s < best_score:
            best_score = s
            best_f = f.copy()
        
        if (it + 1) % 5 == 0:
            print(f"    FW it={it+1}: score={best_score:.6f}, t={t:.4f}")
    
    return best_f, best_score

def _cosine_init(n):

    f = np.cos(np.pi * (j + 0.5) / n) ** 2

    sigma = sigma_frac * n

def _pswf_init(n, bandwidth):

    # PSWF is approximately a sinc-like function
    j = np.arange(n, dtype=np.float64) - (n - 1) / 2.0
    x = 2.0 * bandwidth * j / n
    # sinc approximation
    f = np.ones(n)
    mask = np.abs(x) > 1e-10
    f[mask] = np.sin(x[mask]) / x[mask]
    f = np.maximum(f, 0.0)  # Take only positive part for non-negativity
    return _normalize(f)

def _bipolar_init(n):

    f = np.zeros(n)

"""
Optimizes non-negative step function heights to minimize 2n*max(conv(f,f))/(sum(f))^2.
Uses a multi-phase approach: mirror descent for coarse optimization, standard Frank-Wolfe
with exact line search for fast iterations, and LP-based FW for final refinement.
Exploits convexity of the problem (max of PSD quadratic forms over a simplex).
"""

import numpy as np
import time
from scipy.optimize import linprog, minimize_scalar
from scipy.sparse import coo_matrix


def _fft_conv(f):
    """Full linear convolution of f with itself via FFT."""
    n = len(f)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    F = np.fft.rfft(f, size)
    h = np.fft.irfft(F * F, size)
    return h[:2 * n - 1]


def _score(f):
    """Compute score: 2n * max(conv(f,f)) / sum(f)^2"""
    n = len(f)
    h = _fft_conv(f)
    M = float(np.max(h))
    S = float(np.sum(f))
    if S < 0.01:
        return float('inf')
    return 2.0 * n * M / (S * S)


def _normalize(f):
    """Normalize to sum=1, clip negatives."""
    f = np.maximum(f, 0.0)
    s = float(np.sum(f))
    if s < 1e-30:
        return None
    return f / s


def _subgradient(f, active_lags):
    """Compute subgradient of max_k (f*f)_k using active set."""
    n = len(f)
    L = np.zeros(n, dtype=np.float64)
    for k in active_lags:
        start = max(0, k - n + 1)
        end = min(n - 1, k)
        lo = k - end
        hi = k - start
        L[start:end + 1] += f[lo:hi + 1][::-1]
        if k % 2 == 0:
            j = k // 2
            if j < n:
                L[j] += f[j]
    L /= float(len(active_lags))
    return L


def _mirror_descent(f_init, n_iter, eta_scale, threshold_frac):
    """Mirror descent with multiplicative updates and adaptive step."""
    f = _normalize(np.asarray(f_init, dtype=np.float64))
    if f is None:
        return np.asarray(f_init, dtype=np.float64).copy(), float('inf')
    n = len(f)
    best_f = f.copy()
    best_score = _score(f)

    for it in range(n_iter):
        h = _fft_conv(f)
        M = float(np.max(h))
        if M <= 0:
            break
        threshold = threshold_frac * M + 1e-15
        active = np.where(h >= threshold)[0]
        if len(active) == 0:
            active = np.array([int(np.argmax(h))])

        L = _subgradient(f, active)
        eta = eta_scale / (M + 1e-15)
        f = f * np.exp(np.clip(-eta * L, -20.0, 20.0))
        f_n = _normalize(f)
        if f_n is None:
            f = np.ones(n) / n
        else:
            f = f_n

        s = _score(f)
        if s < best_score:
            best_score = s
            best_f = f.copy()

    return best_f, best_score


def _fw_standard_line_search(f, n_iter, time_budget, threshold_frac=0.7):
    """Standard Frank-Wolfe with dense vertex and exact line search."""
    f = _normalize(np.asarray(f, dtype=np.float64))
    if f is None:
        return np.asarray(f, dtype=np.float64).copy(), float('inf')
    n = len(f)
    best_f = f.copy()
    best_score = _score(f)
    start = time.monotonic()

    for it in range(n_iter):
        if time.monotonic() - start > time_budget:
            break

        h = _fft_conv(f)
        M = float(np.max(h))
        threshold = threshold_frac * M + 1e-15
        active = np.where(h >= threshold)[0]
        if len(active) == 0:
            active = np.array([int(np.argmax(h))])

        L = _subgradient(f, active)
        j_star = int(np.argmin(L))

        # Line search: minimize F((1-t)f + t*e_{j*})

            ft = (1.0 - t) * f + t * (np.arange(n) == j_star).astype(np.float64)
            hc = _fft_conv(ft)
            return float(np.max(hc))

        res = minimize_scalar(obj, bounds=(0.0, 1.0), method='bounded',
                              options={'xatol': 1e-10})
        t = float(res.x)

        f_new = (1.0 - t) * f + t * (np.arange(n) == j_star).astype(np.float64)
        f = _normalize(f_new)
        if f is None:
            break

        s = _score(f)
        if s < best_score:
            best_score = s
            best_f = f.copy()

        if (it + 1) % 200 == 0:
            print(f"    FW-std it={it+1}: score={best_score:.6f}, t={t:.6f}")

    return best_f, best_score


def _fw_lp_step(f, M):
    """Solve LP for better FW direction: max sum(g) s.t. (f*g)_k <= M, g>=0."""
    n = len(f)
    rows = []
    cols = []
    vals = []
    for j in range(n):
        k_lo = j
        k_hi = min(j + n, 2 * n - 1)
        for k in range(k_lo, k_hi):
            rows.append(k)
            cols.append(j)
            vals.append(f[k - j])
    C = coo_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))
    c_obj = -np.ones(n)
    res = linprog(c_obj, A_ub=C, b_ub=np.full(2 * n - 1, M),
                  bounds=[(0, None)] * n, method='highs',
                  options={'dual_feasibility_tolerance': 1e-9})
    if not res.success or res.x is None:
        return None
    g = np.asarray(res.x, dtype=np.float64)
    Sg = float(np.sum(g))
    if Sg < 1e-10:
        return None
    return g / Sg

        h = _fft_conv(f)
        M = float(np.max(h))

        t0 = time.monotonic()
        g = _fw_lp_step(f, M)
        lp_time = time.monotonic() - t0
        if g is None:
            break

        # Exact line search along [f, g]
        def obj(t):
            ft = (1.0 - t) * f + t * g
            hc = _fft_conv(ft)
            return float(np.max(hc))

        res = minimize_scalar(obj, bounds=(0.0, 1.0), method='bounded',

        t = float(res.x)

        f_new = (1.0 - t) * f + t * g
        f = _normalize(f_new)
        if f is None:

            best_f = f.copy()
