    """Return a nonnegative floating vector normalized to sum 1, or None."""
    try:
        x = np.asarray(arr, dtype=np.float64).ravel()
    except Exception:
        return None
    if x.size == 0:
        return None
    x = np.where(np.isfinite(x), x, 0.0)
    x = np.maximum(x, 0.0)
    s = float(x.sum())
    if s <= 1e-300:
        return None
    x /= s
    return x


def score_normalized(x, n):
    """Score assuming x is nonnegative and sum(x) = 1."""
    conv = np.convolve(x, x)
    m = float(np.max(conv))
    if not np.isfinite(m):
        return float("inf")
    return float(2.0 * n * m)


def mix_noisy(base, rng, eps):
    """Mix a base vector with a random Dirichlet vector."""
    f = normalize_sequence(base)
    if f is None:
        return None
    n = f.size
    if eps <= 1e-12:
        return f
    try:
        mix = rng.dirichlet(np.ones(n, dtype=np.float64))
    except Exception:
        mix = np.ones(n, dtype=np.float64) / float(n)
    if eps >= 1.0:
        f = mix
    else:
        f = (1.0 - eps) * f + eps * mix
    return normalize_sequence(f)


def interpolate_sequence(x, n):
    """Resample a nonnegative sequence to length n and normalize it."""
    x = normalize_sequence(x)
    if x is None or n <= 0:
        return None
    m = x.size
    if m == n:
        return x
    if m == 1:
        return np.ones(n, dtype=np.float64) / float(n)
    old = np.linspace(0.0, 1.0, m, dtype=np.float64)
    new = np.linspace(0.0, 1.0, n, dtype=np.float64)
    out = np.interp(new, old, x)
    return normalize_sequence(out)


def make_conv_template(n):
    """Precompute the CSR matrix structure for convolve(f, g)."""
    m = 2 * n - 1
    k64 = np.arange(m, dtype=np.int64)
    js = np.maximum(0, k64 - n + 1).astype(np.int64)
    je = np.minimum(n - 1, k64).astype(np.int64)
    counts = (je - js + 1).astype(np.int64)
    total = int(counts.sum())

    cum = np.cumsum(counts)
    starts = np.repeat(cum - counts, counts)
    local = np.arange(total, dtype=np.int64) - starts

    row = np.repeat(k64, counts).astype(np.int64)
    col_base = js[row]
    col = (col_base + local).astype(np.int64)
    offsets = (row - col).astype(np.int64)

    A_off = csr_matrix((offsets, (row, col)), shape=(m, n), dtype=np.int64)
    edge_offsets = A_off.data.astype(np.int64, copy=True)
    A_template = A_off.astype(np.float64).copy()
    A_template.data.fill(0.0)

    c_obj = -np.ones(n, dtype=np.float64)
    bounds = [(0.0, None)] * n
    return A_template, edge_offsets, c_obj, bounds, m


def solve_direction_lp(template, f, n, M):
    """Solve the LP direction problem."""
    A_template, edge_offsets, c_obj, bounds, m = template
    A = A_template.copy()
    A.data[:] = f[edge_offsets] * float(n)
    b = np.full(m, M * float(n) * float(n), dtype=np.float64)
    return linprog(c_obj, A_ub=A, b_ub=b, bounds=bounds, method="highs")


def coordinate_descent_sweep(f, conv_ff, n, max_sweeps=5, deadline=None):
    """
    Perform coordinate descent: for each coordinate j, find the optimal value
    x >= 0 that minimizes max_k (new_convolution)_k, using golden section search.
    Updates f and conv_ff in place. Returns the number of improvements.
    """
    total_improvements = 0
    for sweep in range(max_sweeps):
        if deadline is not None and time.monotonic() > deadline:
            break
        improvements = 0
        for j in range(n):
            if deadline is not None and time.monotonic() > deadline:
                break
            fj = f[j]
            if fj < 1e-15:
                # Try setting to a small value
                x_lo, x_hi = 0.0, 2.0 / n
            else:
                x_lo, x_hi = 0.0, fj * 4.0 + 1.0 / n

            # Precompute the k range: k from j to j+n-1
            k_start = j
            k_end = min(j + n, 2 * n - 1)  # exclusive

            # The quadratic term is at k = 2*j
            k_quad = 2 * j

            # Precompute: for the linear part, we need f_{k-j} for k in [k_start, k_end)
            # f_{k-j} where k-j ranges from 0 to n-1
            # So f_{k-j} = f[0], f[1], ..., f[n-1] (shifted)
            # For k in [j, j+n-1], k-j ranges from 0 to n-1
            # B_k = 2 * f[k-j] for k != 2j

            # Compute g(x) = max( max_{k in range, k != 2j} ((conv)_k + 2*f[k-j]*(x - fj)),
            #                     (conv)_{2j} + x^2 - fj^2 )

            # Precompute the "base" values:
            # For k in [k_start, k_end), k != 2j: base_k = conv_ff[k] - 2*f[k-j]*fj
            # quad_base = conv_ff[2j] - fj^2

            quad_base = conv_ff[k_quad] - fj * fj

            # For the linear part, precompute conv_ff[k] - 2*f[k-j]*fj for k != 2j
            # This is the value when x = 0 (well, when f_j = 0)
            # Actually: g(x) at k (linear) = conv_ff[k] + 2*f[k-j]*(x - fj)
            # At x=0: conv_ff[k] - 2*f[k-j]*fj
            # Derivative w.r.t. x: 2*f[k-j]

            k_indices = np.arange(k_start, k_end)
            f_indices = k_indices - j  # 0 to n-1

            # Mask out the quadratic position
            mask = k_indices != k_quad
            k_lin = k_indices[mask]
            f_lin = f_indices[mask]

            if len(k_lin) > 0:
                # base[k] = conv_ff[k] - 2*f[k-j]*fj  (value of linear part at x=0)
                # slope[k] = 2*f[k-j]
                base_lin = conv_ff[k_lin] - 2.0 * f[f_lin] * fj
                slope_lin = 2.0 * f[f_lin]
            else:
                base_lin = np.array([])
                slope_lin = np.array([])

            # Golden section search
            if len(k_lin) == 0 and quad_base <= 0:
                # Only quadratic term && it's <= 0 at x=0, so x=0 is optimal
                best_x = 0.0
            else:
                gr = (np.sqrt(5.0) - 1.0) / 2.0  # golden ratio ~0.618

                def eval_g(x):
                    val_q = quad_base + x * x
                    if len(k_lin) > 0:
                        val_lin = np.max(base_lin + slope_lin * x)
                        return max(val_q, val_lin)
                    return val_q

                # Golden section search
                a, b = x_lo, x_hi
                c = b - gr * (b - a)
                d_ = a + gr * (b - a)
                fc = eval_g(c)
                fd = eval_g(d_)
                for _ in range(50):
                    if fc < fd:
                        b = d_
                        d_ = c
                        fd = fc
                        c = b - gr * (b - a)
                        fc = eval_g(c)
                    else:
                        a = c
                        c = d_
                        fc = fd
                        d_ = a + gr * (b - a)
                        fd = eval_g(d_)
                    if b - a < 1e-14:
                        break
                best_x = (a + b) / 2.0

            # Check if there's improvement
            if best_x >= fj * (1.0 - 1e-10):
                # No improvement or worse
                if best_x > fj * (1.0 + 1e-10):
                    # Actually got worse, try x=0
                    g_at_0 = eval_g(0.0)
                    g_at_fj = float(np.max(conv_ff))
                    if g_at_0 < g_at_fj - 1e-12:
                        best_x = 0.0
                    else:
                        continue
                else:
                    continue

            # Apply the update
            delta = best_x - fj
            if abs(delta) < 1e-15:
                continue

            # Update conv_ff
            # conv_ff[k] += 2*f[k-j]*(best_x - fj) for k in [k_start, k_end), k != 2j
            # conv_ff[2j] += best_x^2 - fj^2
            if len(k_lin) > 0:
                conv_ff[k_lin] += 2.0 * f[f_lin] * delta
            conv_ff[k_quad] += best_x * best_x - fj * fj

            # Make sure no negative conv values (numerical)
            np.maximum(conv_ff, 0.0, out=conv_ff)

            f[j] = best_x
            improvements += 1
            total_improvements += 1

        if improvements == 0:
            break

    return total_improvements


def smooth_gradient_step(f, conv_ff, n, beta, lr, iters=10):
    """
    Perform smoothed gradient descent using log-sum-exp approximation of max.
    Returns updated f and improvement count.
    """
    improvements = 0
    for _ in range(iters):
        # Compute softmax weights
        conv_clipped = np.clip(conv_ff, -100.0 * beta, 100.0 * beta)
        shifted = beta * conv_clipped - np.max(beta * conv_clipped)
        exp_vals = np.exp(shifted)
        sum_exp = np.sum(exp_vals)
        if sum_exp < 1e-300:
            break
        weights = exp_vals / sum_exp

        # Gradient: grad_j = sum_k weights[k] * 2 * f[k-j] for valid k
        # This is like a weighted convolution
        # grad = 2 * convolve(f, weights reversed) but only for valid indices
        # Actually: grad_j = 2 * sum_{k: 0<=k-j<=n-1} weights[k] * f[k-j]
        # Let m = k-j, then k = j+m, m from 0 to n-1
        # grad_j = 2 * sum_{m=0}^{n-1} weights[j+m] * f[m]  (if j+m < 2n-1)
        # This is: 2 * (f * weights_flipped)[j] but with appropriate indexing
        # Actually it's: 2 * sum_m f[m] * weights[j+m]
        # This is a cross-correlation-like operation

        # Efficient computation using FFT or direct
        # For n=1024, direct is O(n^2) which is fine
        grad = np.zeros(n, dtype=np.float64)
        for j in range(n):
            k_start = j
            k_end = min(j + n, 2 * n - 1)
            m_indices = np.arange(0, k_end - k_start)
            k_indices = k_start + m_indices
            grad[j] = 2.0 * np.dot(f[m_indices], weights[k_indices])

        # Projected gradient step
        f_new = f - lr * grad
        f_new = np.maximum(f_new, 0.0)
        s = f_new.sum()
        if s < 1e-300:
            break
        f_new /= s

        # Compute new conv and check improvement
        conv_new = np.convolve(f_new, f_new)
        new_score = np.max(conv_new)
        old_score = np.max(conv_ff)
        if new_score < old_score - 1e-12:
            f = f_new
            conv_ff = conv_new
            improvements += 1
        else:
            break

    return f, conv_ff, improvements


def optimize_hybrid(n, f_init, time_budget, rng, template, deadline):
    """
    Run the hybrid optimization: LP + coordinate descent + smooth gradient.
    Returns the best normalized vector and its score.
    """
    f = normalize_sequence(f_init)
    if f is None:
        f = np.ones(n, dtype=np.float64) / float(n)

    conv_ff = np.convolve(f, f)
    best_f = f.copy()
    best_score = float(np.max(conv_ff))
    start_score = best_score

    cycle_count = 0
    stall = 0

    while time.monotonic() < deadline:
        cycle_count += 1
        t0 = time.monotonic()
        M = float(np.max(conv_ff))

        if not np.isfinite(M) or M <= 1e-300:
            break

        # Phase 1: LP direction finding
        lp_improved = False
        try:
            res = solve_direction_lp(template, f, n, M)
            x = getattr(res, "x", None)
            if x is not None:
                x = np.asarray(x, dtype=np.float64).ravel()
                if x.size == n:
                    x = np.where(np.isfinite(x), x, 0.0)
                    x = np.maximum(x, 0.0)
                    s_star = float(x.sum())
                    if s_star > 1e-300:
                        s_ratio = s_star / float(n)
                        if s_ratio > 1.0 + 1e-6:
                            g = x / s_star
                            r = np.convolve(f, g)
                            r_max = float(np.max(r))
                            if r_max < M * (1.0 - 1e-10) and r_max > 0:
                                G = float(np.max(np.convolve(g, g)))
                                a_coef = 2.0 * (r_max - M)
                                b_coef = G + M - 2.0 * r_max
                                if b_coef > 1e-14:
                                    t_max = -a_coef / b_coef
                                else:
                                    t_max = 1.0
                                if not np.isfinite(t_max) or t_max <= 0:
                                    t_max = 1e-8
                                t_max = min(1.0, t_max)

                                # Try multiple step sizes
                                best_t = None
                                best_sc = float(np.max(conv_ff))
                                for frac in [0.9, 0.7, 0.5, 0.3, 0.15, 0.05, 0.01]:
                                    tv = t_max * frac
                                    if tv <= 0:
                                        continue
                                    f_try = (1.0 - tv) * f + tv * g
                                    f_try = np.maximum(f_try, 0.0)
                                    ss = f_try.sum()
                                    if ss < 1e-300:
                                        continue
                                    f_try /= ss
                                    sc = float(np.max(np.convolve(f_try, f_try)))
                                    if sc < best_sc - 1e-10:
                                        best_sc = sc
                                        best_t = tv

                                if best_t is not None and best_sc < M - 1e-10:
                                    f = (1.0 - best_t) * f + best_t * g
                                    f = np.maximum(f, 0.0)
                                    f /= f.sum()
                                    conv_ff = np.convolve(f, f)
                                    M = float(np.max(conv_ff))
                                    if M < best_score - 1e-10:
                                        best_score = M
                                        best_f = f.copy()
                                    lp_improved = True
        except Exception:
            pass

        if not lp_improved:
            # Phase 2: Smooth gradient step (to escape LP local minima)
            beta = 50.0 + 10.0 * cycle_count
            lr = 1.0 / (beta * n)
            if cycle_count % 3 == 0:
                f, conv_ff, sm_imp = smooth_gradient_step(f, conv_ff, n, beta, lr, iters=5)
                M = float(np.max(conv_ff))
                if M < best_score - 1e-10:
                    best_score = M
                    best_f = f.copy()
                    stall = 0
                    continue

        # Phase 3: Coordinate descent fine-tuning
        cd_imps = coordinate_descent_sweep(f, conv_ff, n, max_sweeps=3, deadline=deadline)
        M = float(np.max(conv_ff))
        f = f / f.sum()  # Renormalize after CD
        conv_ff = np.convolve(f, f)
        M = float(np.max(conv_ff))

        if M < best_score - 1e-10:
            best_score = M
            best_f = f.copy()
            stall = 0
            if cycle_count <= 3 or cycle_count % 10 == 0:
                print(f"      n={n} cycle={cycle_count} score={best_score:.6f} "
                      f"cd_imps={cd_imps} dt={time.monotonic()-t0:.2f}s", flush=True)
        else:
            stall += 1
            if cd_imps == 0 and stall > 8:
                # Escalate: try more aggressive perturbation
                eps = min(0.5, 0.1 * stall)
                f = mix_noisy(best_f, rng, eps)
                conv_ff = np.convolve(f, f)
                stall = 0
            elif cd_imps == 0 and stall > 20:
                break

    return best_f, float(best_score)


def try_single_length(n, time_budget, rng, best_f_global, given_f, deadline):
    """Try to optimize at a single length n, return best (f, score)."""
    best_f = None
    best_score = float("inf")

    template = None
    try:
        template = make_conv_template(n)
    except Exception as e:
        print(f"  Template build failed for n={n}: {e}", flush=True)
        return best_f, best_score

    try:
        # Build diverse starts
        starts = []

        if given_f is not None:
            up = interpolate_sequence(given_f, n)
            if up is not None:
                starts.append(mix_noisy(up, rng, 0.03))
                starts.append(mix_noisy(up, rng, 0.1))

        if best_f_global is not None and best_f_global.size != n:
            up2 = interpolate_sequence(best_f_global, n)
            if up2 is not None:
                starts.append(mix_noisy(up2, rng, 0.05))

        # Uniform
        starts.append(mix_noisy(np.ones(n, dtype=np.float64), rng, 0.5))
        # Random Dirichlet
        starts.append(rng.dirichlet(np.ones(n, dtype=np.float64)))
        # Concentrated Dirichlet
        starts.append(rng.dirichlet(np.full(n, 0.5, dtype=np.float64)))

        # Filter and deduplicate
        unique_starts = []
        for s in starts:
            s = normalize_sequence(s)
            if s is None or s.size != n:
                continue
            is_dup = False
            for u in unique_starts:
                if np.linalg.norm(s - u) < 1e-4:
                    is_dup = True
                    break
            if not is_dup:
                unique_starts.append(s)

        m = len(unique_starts)
        per_start_budget = (time_budget - 2.0) / m

        for idx, init in enumerate(unique_starts):
            rem = deadline - time.monotonic()
            if rem < 5.0:
                break
            tb = min(per_start_budget, rem - 1.0)
            if tb < 3.0:
                continue

            print(f"    n={n} start {idx+1}/{m} budget={tb:.1f}s", flush=True)
            f_res, s_res = optimize_hybrid(n, init, tb, rng, template, deadline)

            if np.isfinite(s_res) and s_res < best_score:
                best_score = s_res
                best_f = f_res.copy()
                print(f"      n={n} new best: {best_score:.6f}", flush=True)

    finally:
        try:
            del template
        except Exception:
            pass

    return best_f, best_score


def propose_candidate(seed=42, budget_s=1500, **kwargs):
    """
    Search for a low-scoring nonnegative sequence of length ~1024.
    Uses hybrid LP + coordinate descent + smooth gradient optimization.
    """
    rng = np.random.default_rng(int(seed) if seed is not None else 42)
    total_budget = max(1.0, float(budget_s))
    start_time = time.monotonic()
    deadline = start_time + total_budget - 10.0  # Reserve 10s for final evaluation

    best_f_global = None
    best_score_global = float("inf")
    given_f = None

    # Load given sequence if available
    try:
        raw = np.asarray(height_sequence_1, dtype=np.float64)
        given_f = normalize_sequence(raw)
        if given_f is not None:
            best_f_global = given_f.copy()
            best_score_global = score_normalized(given_f, given_f.size)
            print(f"Given sequence: n={given_f.size}, initial score={best_score_global:.6f}",
                  flush=True)
    except Exception:
        given_f = None

    if best_f_global is None:
        best_f_global = np.ones(64, dtype=np.float64) / 64.0
        best_score_global = 2.0

    # Phase 1: Quick exploration at smaller lengths
    # Phase 2: Main optimization at n=1024
    # Phase 3: Try slightly different lengths
    # Phase 4: Final polish

    # Time allocation
    t1 = min(30.0, total_budget * 0.05)  # Quick small lengths
    t2 = total_budget * 0.55  # Main at 1024
    t3 = total_budget * 0.25  # Other lengths
    t4 = total_budget * 0.10  # Final polish

    print(f"\nBudget: total={total_budget:.0f}s, phase1={t1:.0f}s, "
          f"phase2={t2:.0f}s, phase3={t3:.0f}s, phase4={t4:.0f}s", flush=True)

    # Phase 1: Quick checks at smaller lengths (to find good basins)
    for n in [64, 128, 256, 512]:
        rem = deadline - time.monotonic()
        if rem < 5.0:
            break
        tb = min(t1 / 4.0, rem - 2.0)
        if tb < 3.0:
            continue
        print(f"\n== Phase 1: n={n}", flush=True)
        f_res, s_res = try_single_length(n, tb, rng, best_f_global, given_f, deadline)
        if np.isfinite(s_res) and s_res < best_score_global:
            best_score_global = s_res
            best_f_global = f_res
            print(f"  Updated global best: n={n}, score={best_score_global:.6f}", flush=True)

    # Phase 2: Main optimization at n=1024
    n_main = 1024
    rem = deadline - time.monotonic()
    tb2 = min(t2, rem - 5.0)
    if tb2 > 10.0:
        print(f"\n== Phase 2: n={n_main}, budget={tb2:.0f}s", flush=True)
        f_res, s_res = try_single_length(n_main, tb2, rng, best_f_global, given_f, deadline)
        if np.isfinite(s_res) and s_res < best_score_global:
            best_score_global = s_res
            best_f_global = f_res
            print(f"  Updated global best: n={n_main}, score={best_score_global:.6f}", flush=True)

    # Phase 3: Try other lengths
    for n in [768, 1280, 1536, 2048]:
        rem = deadline - time.monotonic()
        if rem < 10.0:
            break
        tb = min(t3 / 4.0, rem - 5.0)
        if tb < 5.0:
            continue
        print(f"\n== Phase 3: n={n}, budget={tb:.0f}s", flush=True)
        f_res, s_res = try_single_length(n, tb, rng, best_f_global, given_f, deadline)
        if np.isfinite(s_res) and s_res < best_score_global:
            best_score_global = s_res
            best_f_global = f_res
            print(f"  Updated global best: n={n}, score={best_score_global:.6f}", flush=True)

    # Phase 4: Final polish at the best length found
    rem = deadline - time.monotonic()
    if rem > 5.0 and best_f_global is not None:
        n_polish = best_f_global.size
        print(f"\n== Phase 4: Polish at n={n_polish}, budget={rem:.0f}s", flush=True)
        f_res, s_res = try_single_length(n_polish, rem - 5.0, rng,
                                         best_f_global, given_f, deadline)
        if np.isfinite(s_res) and s_res < best_score_global:
            best_score_global = s_res
            best_f_global = f_res
            print(f"  Updated global best: n={n_polish}, score={best_score_global:.6f}", flush=True)

    # Final: coordinate descent polish on the best sequence
    if best_f_global is not None:
        f_final = best_f_global.copy()
        conv_final = np.convolve(f_final, f_final)
        n_final = f_final.size
        # A few more CD sweeps
        coordinate_descent_sweep(f_final, conv_final, n_final, max_sweeps=10,
                                 deadline=deadline)
        f_final = f_final / f_final.sum()
        conv_final = np.convolve(f_final, f_final)
        final_sc = float(np.max(conv_final))
        if final_sc < best_score_global:
            best_score_global = final_sc
            best_f_global = f_final
            print(f"  CD polish: score={best_score_global:.6f}", flush=True)

    # Convert to list for evaluation
    if best_f_global is None:
        best_f_global = np.ones(64, dtype=np.float64) / 64.0

    seq = [float(x) for x in best_f_global]
    try:
        final_eval = float(evaluate_sequence(seq))
    except Exception:
        final_eval = float("inf")

    if not np.isfinite(final_eval):
        # Fallback to given
        try:
            if given_f is not None:
                seq = [float(x) for x in given_f]
                final_eval = float(evaluate_sequence(seq))
        except Exception:
            pass

    if not np.isfinite(final_eval):
        seq = [1.0 / 64.0] * 64
        final_eval = 2.0

    print(f"\nFINAL: n={len(seq)}, local_best={best_score_global:.6f}, "
          f"evaluate_sequence={final_eval:.6f}", flush=True)
    return seq
