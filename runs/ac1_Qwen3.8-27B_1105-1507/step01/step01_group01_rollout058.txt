def multiplicative_smooth_step(a, eta, beta):
    arr=normalize_to_sum1(a)
    if arr is None: return None
    c=np.convolve(arr,arr)
    L=c.size
    m=max(float(np.max(c)),1e-300)
    # weights: softmax beta*(c-m)
    x=beta*(c-m)
    # clip x for stability
    x=np.clip(x,-50,50)
    ex=np.exp(x)
    w=ex/ex.sum()
    r=w[::-1]
    conv=np.convolve(arr,r,mode='full')
    n=arr.size
    B=conv[L-1 : L-1-n : -1]
    # B may length n
    maxh=float(np.max(np.abs(B))) if B.size else 1.0
    if maxh <= 1e-300:
       return arr.copy()
    meanh=float(np.mean(B))
    fac = -float(eta) * (B - meanh) / maxh
    fac=np.clip(fac, -50.0, 50.0)
    # use log of arr but zeros -> -inf
    loga = np.log(np.maximum(arr, 1e-300))
    new = np.exp(loga + fac)
    new = np.clip(new, 0.0, 1000.0)
    return normalize_to_sum1(new)

"""
Searches for low-scoring nonnegative step-function heights.

The main practical idea is that many simple one-sided polynomial profiles,
especially the linear ramp profile, often have very low score for this
particular convolution objective.  The program samples such profiles over a
few lengths and exponents, validates the most promising ones with the
provided evaluate_sequence, and keeps a smooth multiplicative local search as
a fallback in case the simple profiles are not competitive.
"""

import time
import numpy as np


TARGET_SCORE = 1.5027


def normalize_to_sum1(a):
    """Clip to valid nonnegative range and normalize to sum 1."""
    arr = np.asarray(a, dtype=np.float64).ravel()
    if arr.size == 0:
        return None
    arr = np.where(np.isfinite(arr), arr, 0.0)
    arr = np.clip(arr, 0.0, 1000.0)
    s = float(np.sum(arr))
    if not np.isfinite(s) or s <= 1e-300:
        return None
    arr = arr / s
    arr = np.clip(arr, 0.0, 1000.0)
    return arr


def make_power_sequence(n, exponent):
    """
    Create a one-sided power-law profile.

    exponent = 1.0 gives the simple ramp profile i/(n-1).
    A tiny positive floor is added to keep every entry strictly positive while
    changing the score only at floating-point noise level.
    """
    n = int(max(2, n))
    i = np.arange(n, dtype=np.float64)
    x = i / float(n - 1)
    x = np.maximum(x, 0.0)
    a = np.power(x, float(exponent))
    a = np.where(np.isfinite(a), a, 0.0)
    a = a + 1e-300
    a = np.clip(a, 0.0, 1000.0)
    return normalize_to_sum1(a)


def score_fft(a):
    """
    Fast local score using FFT for self-convolution.

    The evaluator uses np.convolve(a, a), i.e. polynomial square, so the FFT
    must use F * F, not F * conjugate(F).
    """
    arr = np.asarray(a, dtype=np.float64).ravel()
    if arr.size == 0:
        return float("inf")
    arr = np.where(np.isfinite(arr), arr, 0.0)
    arr = np.clip(arr, 0.0, 1000.0)
    s = float(np.sum(arr))
    if not np.isfinite(s) or s <= 1e-300:
        return float("inf")

    n = arr.size
    L = 2 * n - 1
    N = 1
    while N < L:
        N <<= 1

    F = np.fft.fft(arr, N)
    c = np.fft.ifft(F * F).real[:L]
    if c.size == 0:
        return float("inf")
    m = float(np.max(c))
    if not np.isfinite(m) or m < 0.0:
        m = 0.0
    return float(2.0 * n * m / (s * s))


def score_exact(a):
    """Exact local score compatible with evaluate_sequence assumptions."""
    arr = np.asarray(a, dtype=np.float64).ravel()
    if arr.size == 0:
        return float("inf")
    arr = np.where(np.isfinite(arr), arr, 0.0)
    arr = np.clip(arr, 0.0, 1000.0)
    s = float(np.sum(arr))
    if not np.isfinite(s) or s <= 1e-300:
        return float("inf")
    c = np.convolve(arr, arr)
    if c.size == 0:
        return float("inf")
    m = float(np.max(c))
    if not np.isfinite(m) or m < 0.0:
        m = 0.0
    return float(2.0 * arr.size * m / (s * s))


def official_score(a):
    """Call the authoritative evaluator if available."""
    try:
        val = evaluate_sequence([float(x) for x in a])
        val = float(val)
        if np.isfinite(val):
            return val
    except Exception:
        pass
    return float("inf")


def update_best_official(best_seq, best_score, seq, score, tag):
    """Return updated best official sequence/score."""
    if seq is None:
        return best_seq, best_score
    if not np.isfinite(score):
        return best_seq, best_score
    try:
        if seq.size > 100000:
            return best_seq, best_score
    except Exception:
        return best_seq, best_score
    if score < best_score:
        print(
            "official update: tag=%s score=%.9f n=%d"
            % (tag, score, int(seq.size)),
            flush=True,
        )
        return seq.copy(), float(score)
    return best_seq, best_score


def multiplicative_smooth_step(a, eta, beta):
    """
    One mirror-descent-style multiplicative step using softmax weights on the
    convolution peaks.  Direction is -grad of a smooth log-sum-exp surrogate
    for max(convolve(a, a)), applied multiplicatively to keep nonnegativity.
    """
    arr = normalize_to_sum1(a)
    if arr is None:
        return None

    n = arr.size
    if n <= 1:
        return arr.copy()

    c = np.convolve(arr, arr)
    L = c.size
    if c.size == 0:
        return arr.copy()

    cmax = float(np.max(c))
    if not np.isfinite(cmax) or cmax <= 0.0:
        return arr.copy()

    x = beta * (c - cmax)
    x = np.clip(x, -50.0, 50.0)
    e = np.exp(x)
    sw = float(np.sum(e))
    if not np.isfinite(sw) or sw <= 1e-300:
        w = np.zeros(L, dtype=np.float64)
        w[int(np.argmax(c))] = 1.0
    else:
        w = e / sw

    r = w[::-1]
    conv = np.convolve(arr, r, mode="full")
    # B_i = sum_k w_k a_{k-i}.
    B = conv[L - 1 : L - 1 - n : -1]
    if B.size != n:
        return arr.copy()

    B = np.where(np.isfinite(B), B, 0.0)
    maxh = float(np.max(np.abs(B)))
    if not np.isfinite(maxh) or maxh <= 1e-300:
        return arr.copy()

    meanh = float(np.mean(B))
    fac = -float(eta) * (B - meanh) / maxh
    fac = np.clip(fac, -50.0, 50.0)

    loga = np.log(np.maximum(arr, 1e-300))
    new = np.exp(loga + fac)
    new = np.where(np.isfinite(new), new, 0.0)
    new = np.clip(new, 0.0, 1000.0)
    return normalize_to_sum1(new)


def multiplicative_smooth_optimize(a, max_time, seed):
    """
    Local fallback optimizer.

    Uses smooth-weighted multiplicative updates and accepts only exact
    improvements under the local objective.  This is intentionally simple and
    time-bounded; the main candidates are the polynomial/ramp profiles.
    """
    arr = normalize_to_sum1(a)
    if arr is None:
        return None, float("inf")

    best = arr.copy()
    best_sc = score_fft(best)
    if not np.isfinite(best_sc):
        best_sc = score_exact(best)
    if not np.isfinite(best_sc):
        return best, best_sc

    if max_time <= 0.0:
        return best, best_sc

    deadline = time.monotonic() + float(max_time)
    rng = np.random.default_rng(seed)
    betas = [5.0, 20.0, 100.0, 500.0, 2000.0]

    for beta in betas:
        eta = 0.05
        stall = 0

        while time.monotonic() < deadline:
            etas = [
                eta,
                eta * 2.0,
                eta * 0.5,
                0.005,
                0.01,
                0.02,
                0.05,
                0.1,
                0.2,
                0.4,
            ]

            cand = None
            cand_sc = best_sc

            for e in etas:
                trial = multiplicative_smooth_step(best, e, beta)
                if trial is None:
                    continue
                sc = score_fft(trial)
                if not np.isfinite(sc):
                    sc = score_exact(trial)
                if np.isfinite(sc) and sc < cand_sc:
                    cand = trial
                    cand_sc = float(sc)

            if cand is not None:
                best = cand
                best_sc = cand_sc
                stall = 0
                eta = min(1.5, eta * 1.1)
            else:
                stall += 1
                eta = max(0.005, eta * 0.7)

                if stall > 35:
                    pert = best * np.exp(rng.normal(0.0, 0.08, best.size))
                    pert = normalize_to_sum1(pert)
                    if pert is not None:
                        sc = score_fft(pert)
                        if not np.isfinite(sc):
                            sc = score_exact(pert)
                        if np.isfinite(sc) and sc < best_sc:
                            best = pert
                            best_sc = float(sc)
                            stall = 0
                            eta = 0.05
                            continue
                    break

    return best.copy(), float(best_sc)


def propose_candidate(seed=42, budget_s=1500, **kwargs):
    """
    Return the best nonnegative sequence found.

    The evaluator is scale invariant, so all candidates are normalized to sum 1.
    The program first samples one-sided polynomial profiles, evaluates the most
    promising with the official evaluator, and only then uses a local search
    fallback if necessary.
    """
    start_time = time.monotonic()
    try:
        budget = float(budget_s)
    except Exception:
        budget = 1500.0
    if not np.isfinite(budget) or budget <= 0.0:
        budget = 1500.0
    deadline = start_time + max(10.0, budget) - 10.0

    entries = []
    idx = 0

    # Use the previously found construction, if present, as a seed/candidate.
    try:
        parent = globals().get("height_sequence_1", None)
    except Exception:
        parent = None

    if parent is not None:
        p = normalize_to_sum1(parent)
        if p is not None:
            sc = score_fft(p)
            if np.isfinite(sc):
                entries.append((float(sc), idx, p, "parent"))
                idx += 1
                print(
                    "parent candidate: n=%d local=%.9f"
                    % (int(p.size), sc),
                    flush=True,
                )

    # Sample simple one-sided power-law profiles.  exponent=1.0 is the ramp.
    lengths = [1024, 2048, 4096]
    exponents = [0.5, 0.75, 1.0, 1.25, 1.5, 2.0, 3.0]

    for n in lengths:
        if time.monotonic() > deadline - 5.0:
            break
        for exp in exponents:
            a = make_power_sequence(n, exp)
            if a is None:
                continue
            sc = score_fft(a)
            if np.isfinite(sc):
                entries.append((float(sc), idx, a, "power n=%d exp=%.2f" % (n, exp)))
                idx += 1

    if not entries:
        a = make_power_sequence(2048, 1.0)
        if a is None:
            a = np.ones(512, dtype=np.float64) / 512.0
        entries.append((score_fft(a), 0, a, "fallback-ramp"))
        idx += 1

    # Sort by local score.  The tuple contains an integer index before the
    # array, so NumPy arrays are never compared during sorting.
    entries.sort()

    best_official_seq = None
    best_official_score = float("inf")

    # Officially validate the best few safe-length candidates.
    tested = 0
    for sc, _, a, tag in entries:
        if tested >= 6:
            break
        if a.size > 8192:
            continue
        off = official_score(a)
        if np.isfinite(off):
            best_official_seq, best_official_score = update_best_official(
                best_official_seq,
                best_official_score,
                a,
                off,
                tag,
            )
        tested += 1
        print(
            "official check: tag=%s score=%.9f"
            % (tag, off if np.isfinite(off) else float("inf")),
            flush=True,
        )

    # A safe fallback is always available from the sampled profiles.
    safe_seq = None
    safe_score = float("inf")
    for sc, _, a, _ in entries:
        try:
            if a.size <= 4096 and sc < safe_score:
                safe_seq = a.copy()
                safe_score = float(sc)
                break
        except Exception:
            pass

    # If the official best is already excellent, do not spend the remaining
    # time on local search.
    if best_official_seq is None or (
        np.isfinite(best_official_score) and best_official_score > TARGET_SCORE - 1e-9
    ):
        rem = deadline - time.monotonic()
        init_seq = best_official_seq if best_official_seq is not None else safe_seq
        if init_seq is None:
            init_seq = make_power_sequence(2048, 1.0)

        if (
            init_seq is not None
            and rem > 30.0
            and init_seq.size <= 2048
        ):
            try:
                opt_seq, opt_sc = multiplicative_smooth_optimize(
                    init_seq,
                    max(5.0, rem - 5.0),
                    int(seed) + 101237,
                )
                if opt_seq is not None and opt_seq.size <= 8192:
                    opt_off = official_score(opt_seq)
                    if np.isfinite(opt_off):
                        best_official_seq, best_official_score = update_best_official(
                            best_official_seq,
                            best_official_score,
                            opt_seq,
                            opt_off,
                            "local-fallback",
                        )
            except Exception as exc:
                print("local fallback failed: %s" % exc, flush=True)

    # Final selection.
    final_seq = best_official_seq if best_official_seq is not None else safe_seq
    if final_seq is None:
        final_seq = make_power_sequence(2048, 1.0)
    if final_seq is None:
        final_seq = np.ones(512, dtype=np.float64) / 512.0

    final_seq = normalize_to_sum1(final_seq)
    if final_seq is None:
        final_seq = make_power_sequence(2048, 1.0)
    if final_seq is None:
        final_seq = np.ones(512, dtype=np.float64) / 512.0

    # One last validation of the object we actually return.
    off = official_score(final_seq)
    if not np.isfinite(off) and safe_seq is not None:
        final_seq = normalize_to_sum1(safe_seq)
        off2 = official_score(final_seq)
        if np.isfinite(off2):
            off = off2

    if final_seq is None:
        final_seq = np.ones(512, dtype=np.float64) / 512.0

    final_seq = np.clip(np.asarray(final_seq, dtype=np.float64), 0.0, 1000.0)
    print(
        "FINAL returned: n=%d official=%.9f"
        % (int(final_seq.size), off if np.isfinite(off) else float("inf")),
        flush=True,
    )
    return [float(x) for x in final_seq]
