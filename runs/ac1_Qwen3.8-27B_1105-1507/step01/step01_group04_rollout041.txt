import numpy as np
import time
from scipy.optimize import linprog
from scipy.sparse import coo_matrix

def _fft_convolve_full(f):
    n = len(f)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    F = np.fft.rfft(f, size)
    h = np.fft.irfft(F * F, size)
    return h[:2 * n - 1]

def _score(f):
    n = len(f)
    h = _fft_convolve_full(f)
    M = float(np.max(h))
    S = float(np.sum(f))
    if S < 0.01:
        return float('inf')
    return 2.0 * n * M / (S * S)

def _normalize(f):
    f = np.maximum(f, 0.0)
    s = float(np.sum(f))
    if s < 1e-30:
        return None
    return f / s

def _linprog_step(f, M):
    """Solve LP: max sum(g) s.t. conv(f,g)[k] <= M, g >= 0."""
    n = len(f)
    # Build constraint matrix: A[k, j] = f[k-j] if 0 <= k-j < n, else 0
    # Shape: (2n-1) x n
    row_idx = []
    col_idx = []
    vals = []
    for j in range(n):
        # For variable g[j], it appears in constraints k where 0 <= k-j < n
        # i.e., j <= k <= j+n-1
        # The coefficient is f[k-j]
        k_start = j
        k_end = j + n - 1
        for k in range(k_start, k_end + 1):
            row_idx.append(k)
            col_idx.append(j)
            vals.append(f[k - j])
    
    A = coo_matrix((vals, (row_idx, col_idx)), shape=(2 * n - 1, n))
    c_obj = -np.ones(n)  # minimize -sum(g) = maximize sum(g)
    b = np.full(2 * n - 1, M)
    res = linprog(c_obj, A_ub=A, b_ub=b, bounds=[(0, None)] * n, method='highs')
    if not res.success or res.x is None:
        return None
    return res.x
