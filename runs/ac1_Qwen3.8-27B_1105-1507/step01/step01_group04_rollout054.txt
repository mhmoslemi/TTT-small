```python
"""
Optimizes non-negative step function heights to minimize 2n*max(conv(f,f))/(sum(f))^2.
Uses improved mirror descent with soft-max pressure, adaptive step size,
active-set LP Frank-Wolfe polish, and multi-start strategy.
"""

import numpy as np
import time

def _fft_convolve(f):
    n = len(f)
    size = 1
    while size < 2*n - 1:
        size <<= 1
    F = np.fft.rfft(f, size)
    return np.fft.irfft(F*F, size)[:2*n-1]

def _score(f):
    n = len(f)
    h = _fft_convolve(f)
    M = np.max(h)
    S = np.sum(f)
    if S < 0.01:
        return float('inf')
    return 2.0 * n * M / (S*S)

def _normalize(f):
    f = np.maximum(f, 0.0)
    s = np.sum(f)
    if s < 1e-30:
        return None
    return f / s

def _pressure_soft(f, h, temperature):
    """Soft-max weighted pressure."""
    n = len(f)
    h_shifted = h - np.max(h)
    w = np.exp(h_shifted / temperature)
    w_sum = np.sum(w)
    if w_sum < 1e-300:
        # Degenerate: use hard max
        idx = np.argmax(h)
        w = np.zeros_like(h)
        w[idx] = 1.0
    else:
        w /= w_sum
    # Compute p[j] = 2 * sum_i w[j+i] * f[i] via FFT
    size = 1
    while size < len(w) + n:
        size <<= 1
    W = np.fft.rfft(w, size)
    G = np.fft.rfft(f[::-1], size)
    conv = np.fft.irfft(W * G, size)
    return 2.0 * conv[n-1:2*n-1]

def _pressure_hard(f, h, threshold_frac):
    """Hard active set pressure (as in previous code)."""
    n = len(f)
    M = np.max(h)
    threshold = threshold_frac * M
    active = np.where(h >= threshold)[0]
    if len(active) == 0:
        active = np.array([np.argmax(h)])
    p = np.zeros(n)
    for k in active:
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        if j_end >= j_start:
            lo = k - j_end
            hi = k - j_start
            p[j_start:j_end+1] += 2.0 * f[lo:hi+1][::-1]
    p /= len(active)
    return p

def _mirror_descent_v2(f0, max_iters, time_budget, T_schedule, eta0, eta_decay):
    """
    Improved mirror descent with soft-max pressure and adaptive step size.
    T_schedule: list of (start_iter, end_iter, temperature) or a function
    """
    f = _normalize(np.asarray(f0, dtype=np.float64))
    if f is None:
        return f0.copy(), float('inf')
    n = len(f)
    best_f = f.copy()
    best_score = _score(f)
    start = time.monotonic()
    
    for it in range(max_iters):
        if time.monotonic() - start > time_budget:
            break
        
        h = _fft_convolve(f)
        M = np.max(h)
        if M <= 0:
            break
        
        # Determine temperature based on schedule
        T = _get_temperature(it, max_iters, T_schedule)
        
        # Compute pressure
        p = _pressure_soft(f, h, T)
        
        # Adaptive step size
        eta = eta0 / (1.0 + eta_decay * it)
        
        # Multiplicative update
        log_ratio = -eta * p
        log_ratio = np.clip(log_ratio, -20.0, 20.0)
        f = f * np.exp(log_ratio)
        f_new = _normalize(f)
        if f_new is None:
            f = f.copy()
        else:
            f = f_new
        
        score = _score(f)
        if score < best_score:
            best_score = score
            best_f = f.copy()
    
    return best_f, best_score

def _get_temperature(it, max_iters, schedule):
    """Get temperature from schedule (list of (frac_start, frac_end, T))."""
    frac = it / max(max_iters, 1)
    for start_f, end_f, T in schedule:
        if start_f <= frac < end_f:
            return T
    return schedule[-1][2] if schedule else 1.0

def _lp_polish(f, max_iters, time_budget, top_k=50):
    """Active-set LP Frank-Wolfe polish."""
    from scipy.optimize import linprog
    from scipy.sparse import coo_matrix
    
    f = _normalize(np.asarray(f, dtype=np.float64))
    if f is None:
        return f, float('inf')
    n = len(f)
    best_f = f.copy()
    best_score = _score(f)
    start = time.monotonic()
    
    for it in range(max_iters):
        if time.monotonic() - start > time_budget:
            break
        
        h = np.convolve(f, f)  # Use direct conv for moderate n
        M = np.max(h)
        
        # Select top-k active lags
        k = min(top_k, len(h))
        active_indices = np.argsort(h)[-k:][::-1].astype(int)
        # Only include lags that are close to the max
        active_indices = active_indices[h[active_indices] >= 0.9 * M]
        if len(active_indices) == 0:
            active_indices = np.array([np.argmax(h)])
        
        # Build constraint matrix for active lags only
        row_idx = []
        col_idx = []
        vals = []
        for j in range(n):
            for k_idx in active_indices:
                i = k_idx - j
                if 0 <= i < n:
                    row_idx.append(k_idx)
                    col_idx.append(j)
                    vals.append(f[i])
        
        if not row_idx:
            break
            
        # Map active lags to 0..len(active_indices)-1
        active_map = {k_idx: idx for idx, k_idx in enumerate(active_indices)}
        row_idx = [active_map[r] for r in row_idx]
        
        C = coo_matrix((vals, (row_idx, col_idx)), shape=(len(active_indices), n))
        
        c_obj = -np.ones(n)
        res = linprog(c_obj, A_ub=C, b_ub=np.full(len(active_indices), M),
                      bounds=[(0, None)] * n, method='highs')
        
        if not res.success or res.x is None:
            break
        
        g = np.maximum(res.x, 0)
        Sg = np.sum(g)
        if Sg < 1e-10:
            break
        g /= Sg
        
        t = 2.0 / (it + 2)
        f_new = (1 - t) * f + t * g
        f = _normalize(f_new)
        if f is None:
            break
        
        score = _score(f)
        if score < best_score:
            best_score = score
            best_f = f.copy()
        
        if (it+1) % 5 == 0:
            print(f"    LP it={it+1}: score={best_score:.6f}, M={M:.8f}")
    
    return best_f, best_score
```

```python
"""
Optimizes non-negative step function heights to minimize 2n*max(conv(f,f))/(sum(f))^2.
Uses improved mirror descent with soft-max pressure, adaptive step size, temperature
annealing, active-set LP Frank-Wolfe polish, and coordinate descent refinement.
"""

import numpy as np
import time


def _fft_convolve(f):
    """Compute full linear convolution f*f using FFT."""
    n = len(f)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    F = np.fft.rfft(f, size)
    return np.fft.irfft(F * F, size)[:2 * n - 1]


def _score_fast(f):
    """Fast score: 2n * max(conv(f,f)) / (sum f)^2. Assumes f >= 0, sum > 0."""
    n = len(f)
    h = _fft_convolve(f)
    M = float(np.max(h))
    S = float(np.sum(f))
    if S < 0.01:
        return float('inf')
    return 2.0 * n * M / (S * S)


def _normalize(f):
    """Normalize f to sum=1, clip negatives. Returns None if degenerate."""
    f = np.maximum(f, 0.0)
    s = float(np.sum(f))
    if s < 1e-30:
        return None
    return f / s


def _pressure_soft(f, h, temperature):
    """
    Compute pressure vector p[j] = 2 * sum_{i>=0} w[j+i] * f[i]
    where w is the soft-max over h (all lags).
    Computed via FFT for O(n log n) complexity.
    """
    n = len(f)
    h_shifted = h - np.max(h)
    w = np.exp(h_shifted / temperature)
    w_sum = float(np.sum(w))
    if w_sum < 1e-300:
        idx = int(np.argmax(h))
        w = np.zeros_like(h)
        w[idx] = 1.0
    else:
        w /= w_sum
    # p[j] = 2 * sum_i w[j+i] * f[i] = 2 * cross_corr(w, f)[j]
    # Using FFT: cross_corr(w, f)[j] = conv(w, f_rev)[j + n - 1]
    size = 1
    while size < len(w) + n:
        size <<= 1
    W = np.fft.rfft(w, size)
    G = np.fft.rfft(f[::-1], size)
    conv = np.fft.irfft(W * G, size)
    return 2.0 * conv[n - 1:2 * n - 1]


def _pressure_active(f, h, threshold_frac):
    """Hard active-set pressure using top lags above threshold."""
    n = len(f)
    M = float(np.max(h))
    threshold = threshold_frac * M + 1e-15
    active = np.where(h >= threshold)[0]
    if len(active) == 0:
        active = np.array([int(np.argmax(h))])
    p = np.zeros(n, dtype=np.float64)
    for k in active:
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        if j_end >= j_start:
            lo = k - j_end
            hi = k - j_start
            p[j_start:j_end + 1] += 2.0 * f[lo:hi + 1][::-1]
    p /= float(len(active))
    return p


def _mirror_descent(f0, max_iters, time_budget, T_low, T_high, eta0, eta_decay,
                    use_soft=True, reheat_interval=2000, reheat_frac=0.03):
    """
    Mirror descent with soft-max pressure and temperature annealing.
    T anneals from T_low to T_high over the course of optimization.
    eta decays as eta0 / (1 + eta_decay * t).
    Reheats with perturbation when stuck.
    """
    f = _normalize(np.asarray(f0, dtype=np.float64))
    if f is None:
        return np.asarray(f0, dtype=np.float64).copy(), float('inf')
    n = len(f)
    best_f = f.copy()
    best_score = _score_fast(f)
    start = time.monotonic()
    it_no_improve = 0
    rng = np.random.default_rng(12345)

    for it in range(max_iters):
        if time.monotonic() - start > time_budget:
            break

        h = _fft_convolve(f)
        M = float(np.max(h))
        if M <= 0:
            break

        # Temperature schedule: anneal from T_low to T_high
        frac = float(it) / max(max_iters, 1)
        T = T_low + (T_high - T_low) * frac

        if use_soft:
            p = _pressure_soft(f, h, T)
        else:
            thresh = 0.6 + 0.35 * frac
            p = _pressure_active(f, h, thresh)

        # Adaptive step size
        eta = eta0 / (1.0 + eta_decay * it)

        # Multiplicative update
        log_ratio = -eta * p
        log_ratio = np.clip(log_ratio, -20.0, 20.0)
        f = f * np.exp(log_ratio)
        f_new = _normalize(f)
        if f_new is None:
            f = np.ones(n) / n
        else:
            f = f_new

        score = _score_fast(f)
        if score < best_score:
            best_score = score
            best_f = f.copy()
            it_no_improve = 0
        else:
            it_no_improve += 1

        # Reheat: perturb and continue
        if it_no_improve >= reheat_interval:
            f = f + rng.random(n) * reheat_frac * np.max(f)
            f = _normalize(f)
            if f is not None:
                f = f
            it_no_improve = 0

        if (it + 1) % 5000 == 0:
            print(f"      MD it={it+1}: best={best_score:.6f}, M={M:.8f}, T={T:.4f}")

    return best_f, best_score


def _lp_frank_wolfe(f0, max_iters, time_budget, top_k=80):
    """
    LP-based Frank-Wolfe with active set (top_k lags only).
    Solves: max sum(g) s.t. (f*g)[k] <= M for top_k lags, g >= 0.
    Step size: t = 2/(iter+2).
    """
    from scipy.optimize import linprog
    from scipy.sparse import coo_matrix

    f = _normalize(np.asarray(f0, dtype=np.float64))
    if f is None:
        return np.asarray(f0, dtype=np.float64).copy(), float('inf')
    n = len(f)
    best_f = f.copy()
    best_score = _score_fast(f)
    start = time.monotonic()

    for it in range(max_iters):
        if time.monotonic() - start > time_budget:
            break

        if n <= 3000:
            h = np.convolve(f, f)
        else:
            h = _fft_convolve(f)
        M = float(np.max(h))

        # Select top-k active lags
        k_sel = min(top_k, len(h))
        top_indices = np.argsort(h)[-k_sel:][::-1].astype(int)
        # Keep only lags within 90% of max
        active_indices = top_indices[h[top_indices] >= 0.90 * M]
        if len(active_indices) == 0:
            active_indices = np.array([int(np.argmax(h))])
        n_active = len(active_indices)

        # Build sparse constraint matrix: C[i, j] = f[k_i - j] if 0 <= k_i - j < n
        row_idx = []
        col_idx = []
        vals = []
        for j in range(n):
            for ai in range(n_active):
                k_val = int(active_indices[ai])
                i_val = k_val - j
                if 0 <= i_val < n:
                    row_idx.append(ai)
                    col_idx.append(j)
                    vals.append(float(f[i_val]))

        if not row_idx:
            break

        C = coo_matrix((vals, (row_idx, col_idx)), shape=(n_active, n))
        c_obj = -np.ones(n)
        try:
            res = linprog(c_obj, A_ub=C, b_ub=np.full(n_active, M),
                          bounds=[(0, None)] * n, method='highs',
                          options={'presolve': True})
        except Exception:
            break

        if not res.success or res.x is None:
            break

        g = np.maximum(res.x, 0.0)
        Sg = float(np.sum(g))
        if Sg < 1e-10:
            break
        g = g / Sg

        step = 2.0 / (it + 2)
        f_new = (1.0 - step) * f + step * g
        f = _normalize(f_new)
        if f is None:
            break

        score = _score_fast(f)
        if score < best_score:
            best_score = score
            best_f = f.copy()

        if (it + 1) % 3 == 0:
            print(f"      LP it={it+1}: score={best_score:.6f}, M={M:.10f}, |act|={n_active}")

    return best_f, best_score


def _coordinate_descent(f0, max_sweeps=10, time_budget=30.0):
    """
    Coordinate descent: for each coordinate j, try several candidate values
    and keep the one that minimizes the score.
    """
    f = _normalize(np.asarray(f0, dtype=np.float64))
    if f is None:
        return np.asarray(f0, dtype=np.float64).copy(), float('inf')
    n = len(f)
    best_f = f.copy()
    best_score = _score_fast(f)
    start = time.monotonic()
    rng = np.random.default_rng(999)

    for sweep in range(max_sweeps):
        if time.monotonic() - start > time_budget:
            break
        improved = False
        # Randomize coordinate order
        order = rng.permutation(n)
        for j in order:
            if time.monotonic() - start > time_budget:
                break
            old_val = f[j]
            # Try several candidate values
            candidates = [0.0, old_val * 0.5, old_val * 0.75, old_val * 1.25,
                          old_val * 1.5, old_val + 0.01 * np.max(f),
                          np.max(f), 2.0 * old_val]
            # Also try a small random perturbation
            candidates.append(max(0.0, old_val + (rng.random() - 0.5) * 0.05 * np.max(f)))

            for cand in candidates:
                f[j] = cand
                f[j] = max(0.0, float(f[j]))
                s = _score_fast(f)
                if s < best_score - 1e-12:
                    best_score = s
                    best_f = f.copy()
                    improved = True
                    break
            if not improved or f[j] != old_val:
                # If we found improvement, keep it; otherwise revert
                if f[j] == old_val and not improved:
                    pass  # Already restored if no improvement
            # Ensure we keep the best
            if best_f[j] != f[j]:
                f[j] = best_f[j]
                    # Actually, let's just track best globally
                    # and revert f[j] if it got worse
                    pass

            # Simpler: just track best and revert
            f[j] = best_f[j]  # restore best

        if not improved:
            break

    return best_f, best_score


def _coord_descent_simple(f0, time_budget=20.0, n_sweeps=5):
    """Simpler coordinate descent: for each j, do a 1D search around current value."""
    f = _normalize(np.asarray(f0, dtype=np.float64))
    if f is None:
        return np.asarray(f0, dtype=np.float64).copy(), float('inf')
    n = len(f)
    best_f = f.copy()
    best_score = _score_fast(f)
    start = time.monotonic()
    rng = np.random.default_rng(42)

    for sweep in range(n_sweeps):
        if time.monotonic() - start > time_budget:
            break
        improved_any = False
        order = rng.permutation(n)
        for j in order:
            if time.monotonic() - start > time_budget:
                break
            old = f[j]
            # Try geometric multipliers
            for mult in [0.5, 0.7, 0.85, 1.15, 1.3, 1.5]:
                f[j] = old * mult
                s = _score_fast(f)
                if s < best_score:

                    best_f = f.copy()
                    improved_any = True
            # Revert if not improved
            if f[j] != best_f[j]:
                f = best_f.copy()
        if not improved_any:
            break

    return best_f, best_score


def _init_uniform(n):
    return np.ones(n, dtype=np.float64) / n


def _init_cosine(n):
    j = np.arange(n, dtype=np.float64)
    f = np.cos(np.pi * (j + 0.5) / n) ** 2
    f = np.maximum(f, 0.0)
    s = np.sum(f)
    if s > 0:
        f /= s
    return f


def _init_gaussian(n, sigma_frac=0.3):
    j = np.arange(n, dtype=np.float64) - (n - 1) / 2.0
    sigma = sigma_frac * n
    f = np.exp(-j * j / (2.0 * sigma * sigma))
    s = np.sum(f)
    if s > 0:
        f /= s
    return f


def _init_tapered(n, power=2.0):
    j = np.arange(n, dtype=np.float64)
    x = np.sin(np.pi * (j + 0.5) / n)
    f = x ** power
    f = np.maximum(f, 0.0)
    s = np.sum(f)
    if s > 0:
        f /= s
    return f


def _init_two_peak(n, pos_frac=0.35):
    """Two Gaussian peaks."""
    j = np.arange(n, dtype=np.float64)
    center1 = pos_frac * n
    center2 = (1.0 - pos_frac) * n
    sigma = 0.15 * n
    f = np.exp(-(j - center1) ** 2 / (2 * sigma ** 2)) + \
        np.exp(-(j - center2) ** 2 / (2 * sigma ** 2))
    s = np.sum(f)
    if s > 0:
        f /= s
    return f


def _init_edge_concentrated(n, alpha=2.0):
    """Mass concentrated at edges: f[j] ~ (1 - 4*(j/n - 0.5)^2)^alpha."""
    j = np.arange(n, dtype=np.float64) / n
    x = 1.0 - 4.0 * (j - 0.5) ** 2
    f = np.maximum(x, 0.0) ** alpha
    s = np.sum(f)
    if s > 0:
        f /= s
    return f


def _init_barker_like(n):
    """Use Barker-inspired pattern: blocks of +1 encoded as high/low amplitude."""
    # Barker sequence of length 7: [1,1,1,-1,-1,1,-1]
    # Map to amplitudes: 1 -> high, -1 -> low
    barker = np.array([1, 1, 1, -1, -1, 1, -1], dtype=np.float64)
    f = np.tile(barker, n // len(barker) + 1)[:n]
    # Map to [0.3, 1.0] range
    f = 0.3 + 0.35 * (f + 1)
    f = np.maximum(f, 0.0)
    s = np.sum(f)
    if s > 0:
        f /= s
    return f


def _init_periodic(n, period, duty=0.5):
    """Periodic rectangular pattern."""
    j = np.arange(n, dtype=np.float64)
    f = ((j % period) < int(period * duty)).astype(np.float64)
    f += 0.1  # Add small baseline to avoid zeros
    s = np.sum(f)
    if s > 0:
        f /= s
    return f

                                       (0.1, 0.8, 0.3, 0.0003),
                                       (0.05, 0.2, 1.0, 0.002)]:
            if time.monotonic() > deadline - 15:
                break
            tb = min(60.0, deadline - time.monotonic() - 5)
            if tb < 5:
                break
            f_r, s_r = _mirror_descent(
                best_f, int(min(50000, tb * 6000 / max(n_best, 256))), tb,
                T_low=T_lo, T_high=T_hi, eta0=eta0, eta_decay=etad,
                use_soft=True, reheat_interval=5000, reheat_frac=0.01
            )
            if s_r < best_score:
                best_score = s_r
                best_f = f_r.copy()
                print(f"  ** New best (T={T_lo}-{T_hi}): {best_score:.6f} **")
            else:
                print(f"  MD (T={T_lo}-{T_hi}): {s_r:.6f} (no improvement)")

    # Phase 4: Coordinate descent polish
    print("\n--- Phase 4: Coordinate Descent ---")
    if time.monotonic() < deadline - 10:
        n_best = len(best_f)
        if n_best <= 2000:
            tb4 = min(60.0, deadline - time.monotonic() - 5)
            if tb4 > 5:
                f_cd, s_cd = _coord_descent_simple(best_f, time_budget=tb4, n_sweeps=3)
                if s_cd < best_score:
                    best_score = s_cd
                    best_f = f_cd.copy()
                    print(f"  ** New best from CD: {best_score:.6f} **")
                else:
                    print(f"  CD: {s_cd:.6f} (no improvement)")

    # Phase 5: Final LP polish if time remains
    if time.monotonic() < deadline - 10:
        n_best = len(best_f)
        if n_best <= 3000:
            tb5 = min(50.0, deadline - time.monotonic() - 3)
            if tb5 > 8:
                print(f"\n--- Phase 5: Final LP (n={n_best}) ---")
                f_lp5, s_lp5 = _lp_frank_wolfe(best_f, max_iters=20, time_budget=tb5, top_k=60)
                if s_lp5 < best_score:
                    best_score = s_lp5
                    best_f = f_lp5.copy()
                    print(f"  ** Final LP best: {best_score:.6f} **")

    # Final verification
    final_list = [float(x) for x in best_f]
    # Ensure all values are valid (non-negative, finite, not too large)
    final_list = [min(max(0.0, x), 1000.0) for x in final_list]
    final_score = float(evaluate_sequence(final_list))
    print(f"\nFinal verified score: {final_score:.6f} (n={len(final_list)})")

    if not np.isfinite(final_score) or final_score > 2.0:
        print("  Falling back to parent.")
        return [float(x) for x in parent]

    return final_list
```
