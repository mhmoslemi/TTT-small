import numpy as np
import time
from scipy.optimize import minimize
import math

def score_of(a):
    """Compute the objective score for sequence a."""
    a = np.clip(a, 0.0, 1000.0)
    n = len(a)
    S = float(np.sum(a))
    if S < 0.01:
        return float('inf')
    conv = np.convolve(a, a)
    max_b = float(np.max(conv))
    return 2.0 * n * max_b / (S * S)

def project_simplex(v):
    """Project v onto probability simplex."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho_candidates = np.nonzero(u * np.arange(1, n + 1) > (cssv - 1))[0]
    if len(rho_candidates) == 0:
        return np.ones(n) / n
    rho = rho_candidates[-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def softmax_val_and_grad(p, beta):
    """Compute soft_max(conv(p,p), beta) and its gradient."""
    n = len(p)
    conv = np.convolve(p, p)
    max_c = np.max(conv)
    scaled = beta * (conv - max_c)
    exp_c = np.exp(scaled)
    sum_exp = np.sum(exp_c)
    sm_val = (max_c + np.log(sum_exp)) / beta
    w = exp_c / sum_exp
    
    grad = np.zeros(n)
    for j in range(n):
        k_lo = j
        k_hi = min(j + n - 1, 2 * n - 2)
        if k_lo <= k_hi:
            grad[j] = 2.0 * np.dot(w[k_lo:k_hi + 1], p[0:k_hi - k_lo + 1])
    grad /= beta
    
    return sm_val, grad

def projected_gradient_descent(p, deadline, lr_init=0.05, momentum=0.9, beta_init=10.0, beta_end=100.0, verbose_prefix=""):
    """
    Projected gradient descent on the probability simplex.
    Minimizes soft_max(conv(p,p), beta) with increasing beta.
    """
    p = project_simplex(p.copy())
    n = len(p)
    best_p = p.copy()
    best_score = 2.0 * n * float(np.max(np.convolve(p, p)))
    v = np.zeros_like(p)
    lr = lr_init
    beta = beta_init
    no_improve = 0
    it = 0
    
    while time.monotonic() < deadline:
        sm_val, grad = softmax_val_and_grad(p, beta)
        
        # Momentum step
        v = momentum * v - lr * grad
        new_p = p + v
        new_p = project_simplex(new_p)
        
        # Evaluate actual (non-soft) score
        new_score = 2.0 * n * float(np.max(np.convolve(new_p, new_p)))
        it += 1
        
        if new_score < best_score:
            p = new_p
            best_p = p.copy()
            best_score = new_score
            no_improve = 0
            if it % 50 == 0:
                print(f"  {verbose_prefix}iter {it}: score={best_score:.6f} (beta={beta:.1f}, lr={lr:.4f})")
            # Adapt beta
            if sm_val > 0 and it > 20:
                beta = min(beta * 1.01, beta_end)
        else:
            no_improve += 1
            if no_improve % 30 == 0:
                lr *= 0.7
                momentum = min(momentum + 0.01, 0.99)
            if lr < 1e-6 or no_improve > 300:
                # Try restarting with higher beta
                beta = min(beta * 2, beta_end * 2)
                lr = lr_init * 0.5
                momentum = 0.9
                if beta > beta_end * 4:
                    break
    
    return best_p, best_score

def coordinate_descent(p, deadline, max_sweeps=50, verbose_prefix=""):
    """
    Coordinate descent: for each coordinate, try several values and pick the best.
    """
    p = project_simplex(p.copy())
    n = len(p)
    best_p = p.copy()
    best_score = 2.0 * n * float(np.max(np.convolve(p, p)))
    rng = np.random.default_rng(123)
    
    for sweep in range(max_sweeps):
        if time.monotonic() > deadline:
            break
        improved = False
        # Random order
        idx_order = rng.permutation(n)
        for j in idx_order:
            old_val = p[j]
            # Try several multipliers
            candidates = [0.0, old_val * 0.3, old_val * 0.7, old_val, old_val * 1.3, old_val * 1.7, old_val * 2.5, 1.0 / n]
            best_candidate = old_val
            best_local_score = best_score
            
            for cv in candidates:
                if cv < 0:
                    continue
                p_test = p.copy()
                p_test[j] = cv
                p_test = project_simplex(p_test)
                s = 2.0 * n * float(np.max(np.convolve(p_test, p_test)))
                if s < best_local_score:
                    best_local_score = s
                    best_candidate = cv
            
            if best_local_score < best_score:
                p[j] = best_candidate
                p = project_simplex(p)
                best_score = best_local_score
                best_p = p.copy()
                improved = True
        
        if not improved:
            break
    
    return best_p, best_score

def simulated_annealing(p, deadline, T_init=0.001, T_final=1e-6, verbose_prefix=""):
    """
    Simulated annealing: random perturbations with acceptance criterion.
    """
    p = project_simplex(p.copy())
    n = len(p)
    best_p = p.copy()
    best_score = 2.0 * n * float(np.max(np.convolve(p, p)))
    rng = np.random.default_rng(456)
    T = T_init
    decay = (T_final / T_init) ** (1.0 / max(1, int((deadline - time.monotonic()) / 0.5)))
    
    it = 0

        p_new = p.copy()

"""
Projected gradient descent on the probability simplex with soft-max smoothing,
coordinate descent fine-tuning, and simulated annealing for escaping local minima.
Minimizes max(conv(p,p)) over probability vectors p (equivalently minimizing
2n*max(conv(a,a))/sum(a)^2).
"""
import numpy as np
import time
import math
from scipy.optimize import minimize


def score_of(a):
    """Compute the objective score for sequence a."""
    a = np.clip(a, 0.0, 1000.0)
    n = len(a)
    if n == 0:
        return float('inf')
    S = float(np.sum(a))
    if S < 0.01:
        return float('inf')
    conv = np.convolve(a, a)
    max_b = float(np.max(conv))
    return 2.0 * n * max_b / (S * S)


def project_simplex(v):
    """Project v onto probability simplex {x: x>=0, sum(x)=1}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not np.any(cond):
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def softmax_val_and_grad(p, beta):
    """
    Compute soft_max(conv(p,p), beta) and its gradient w.r.t. p.
    p should be on the probability simplex.
    Returns (sm_value, gradient).
    """
    n = len(p)
    conv = np.convolve(p, p)
    max_c = np.max(conv)
    exp_c = np.exp(beta * (conv - max_c))
    sum_exp = np.sum(exp_c)
    sm_val = (max_c + np.log(sum_exp)) / beta
    w = exp_c / sum_exp

    # grad[j] = (2/beta) * sum_{k=j}^{min(j+n-1, 2n-2)} w[k] * p[k-j]
    grad = np.zeros(n)
    for j in range(n):
        k_lo = j
        k_hi = min(j + n - 1, 2 * n - 2)
        if k_lo <= k_hi:
            grad[j] = np.dot(w[k_lo:k_hi + 1], p[: k_hi - k_lo + 1])
    grad *= 2.0 / beta

    return sm_val, grad


def true_score_p(p):
    """Score for probability vector p: 2n * max(conv(p,p))."""
    n = len(p)
    return 2.0 * n * float(np.max(np.convolve(p, p)))


def projected_gradient_descent(p, deadline, lr_init=0.05, momentum=0.9,
                                beta_init=5.0, beta_end=500.0,
                                lr_min=1e-6, verbose_prefix=""):
    """
    Projected gradient descent on the probability simplex with beta annealing.
    """
    p = project_simplex(p.astype(np.float64).copy())
    n = len(p)
    best_p = p.copy()
    best_score = true_score_p(p)
    v = np.zeros_like(p)
    lr = lr_init
    beta = beta_init
    no_improve = 0
    it = 0
    total_iter_limit = max(1000, int((deadline - time.monotonic()) / 0.02))

    while time.monotonic() < deadline and it < total_iter_limit:
        sm_val, grad = softmax_val_and_grad(p, beta)

        # Momentum step
        v = momentum * v - lr * grad
        new_p = p + v
        new_p = project_simplex(new_p)

        new_score = true_score_p(new_p)
        it += 1

        if new_score < best_score:
            p = new_p
            best_p = p.copy()
            best_score = new_score
            no_improve = 0
            # Anneal beta (increase sharpness of soft-max)
            beta = min(beta * 1.02, beta_end)
            if it % 100 == 0:
                print(f"  {verbose_prefix}iter {it}: score={best_score:.6f} "
                      f"(beta={beta:.1f}, lr={lr:.5f})")
        else:
            no_improve += 1
            if no_improve % 40 == 0:
                lr *= 0.75
            if lr < lr_min or no_improve > 400:
                # Restart with adjusted parameters
                beta = min(beta * 2.0, beta_end * 2)
                lr = max(lr_init * 0.3, lr_min * 10)
                momentum = min(momentum + 0.02, 0.99)
                no_improve = 0
                if beta > beta_end * 5:
                    break

    return best_p, best_score


def coordinate_descent(p, deadline, max_sweeps=50, verbose_prefix=""):
    """
    Coordinate descent: for each coordinate, try a few candidate values.
    Uses the fact that changing p[j] only affects O(n) convolution entries.
    """
    p = project_simplex(p.astype(np.float64).copy())
    n = len(p)
    best_p = p.copy()
    best_score = true_score_p(p)
    rng = np.random.default_rng(123)

    for sweep in range(max_sweeps):
        if time.monotonic() > deadline:
            break
        improved = False
        idx_order = rng.permutation(n)
        for j in idx_order:
            if time.monotonic() > deadline:
                return best_p, best_score
            old_val = p[j]
            # Candidates: reduction, increase, zero, uniform
            candidates = [0.0, old_val * 0.5, old_val * 1.5, 1.0 / n, old_val * 0.7, old_val * 1.3]
            best_candidate = old_val
            best_local_score = best_score

            for cv in candidates:
                if cv < 0 or cv > 1.0:
                    continue
                p_test = p.copy()
                p_test[j] = cv
                p_test = project_simplex(p_test)
                s = true_score_p(p_test)
                if s < best_local_score:
                    best_local_score = s
                    best_candidate = cv

            if best_local_score < best_score * (1 - 1e-10):
                p[j] = best_candidate
                p = project_simplex(p)
                best_score = best_local_score
                best_p = p.copy()
                improved = True

        if not improved:
            break
        if (sweep + 1) % 10 == 0:
            print(f"  {verbose_prefix}CD sweep {sweep+1}: score={best_score:.6f}")

    return best_p, best_score


def simulated_annealing(p, deadline, T_init=0.0003, T_final=1e-7,
                        perturbation_scale=0.3, verbose_prefix=""):
    """
    Simulated annealing with multiplicative perturbations.
    """
    p = project_simplex(p.astype(np.float64).copy())
    n = len(p)
    best_p = p.copy()
    best_score = true_score_p(p)
    current_score = best_score
    rng = np.random.default_rng(456)
    T = T_init

    # Estimate total iterations possible
    start_time = time.monotonic()
    # Do a quick timing
    p_test = p.copy()
    p_test[0] *= 2
    p_test = project_simplex(p_test)
    _ = true_score_p(p_test)
    per_iter_time = max(1e-5, time.monotonic() - start_time)
    n_iters = max(100, int((deadline - time.monotonic()) / per_iter_time))
    decay = (T_final / T_init) ** (1.0 / n_iters)

    it = 0
    while time.monotonic() < deadline and it < n_iters:
        # Random multiplicative perturbation on a subset
        n_change = max(1, int(n * rng.uniform(0.02, 0.15)))
        idxs = rng.choice(n, size=n_change, replace=False)
        p_new = p.copy()
        for idx in idxs:
            p_new[idx] *= rng.uniform(1.0 - perturbation_scale, 1.0 + perturbation_scale)
        p_new = project_simplex(p_new)

        new_score = true_score_p(p_new)
        it += 1

        delta = new_score - current_score
        if new_score < best_score:
            best_p = p_new.copy()
            best_score = new_score
        if new_score < current_score or (T > T_final and rng.random() < math.exp(-delta / T)):
            p = p_new
            current_score = new_score

        T *= decay
        if it % 500 == 0:
            print(f"  {verbose_prefix}SA iter {it}: best={best_score:.6f} "
                  f"(T={T:.2e}, current={current_score:.6f})")

    return best_p, best_score


def multi_lag_gradient_descent(p, deadline, top_k=5, lr_init=0.03,
                               momentum=0.9, verbose_prefix=""):
    """
    Gradient descent targeting the top-k largest convolution lags (weighted).
    This is a deterministic alternative to soft-max that can be more stable.
    """
    p = project_simplex(p.astype(np.float64).copy())
    n = len(p)
    best_p = p.copy()
    best_score = true_score_p(p)
    v = np.zeros_like(p)
    lr = lr_init
    no_improve = 0
    it = 0

    while time.monotonic() < deadline:
        conv = np.convolve(p, p)
        # Get top-k lags
        top_indices = np.argsort(conv)[::-1][:top_k]
        top_vals = conv[top_indices]
        max_val = top_vals[0]

        # Normalize weights (exponential of lags / max_val)
        weights = np.exp((top_vals - max_val) * 5.0)
        weights /= np.sum(weights)

        # Combined gradient
        grad = np.zeros(n)
        for (k, w) in zip(top_indices, weights):
            for j in range(n):
                if 0 <= k - j < n:
                    grad[j] += w * 2.0 * p[k - j]

        # Momentum step
        v = momentum * v - lr * grad
        new_p = p + v
        new_p = project_simplex(new_p)

        new_score = true_score_p(new_p)
        it += 1

        if new_score < best_score:
            p = new_p
            best_p = p.copy()
            best_score = new_score
            no_improve = 0
            if it % 200 == 0:
                print(f"  {verbose_prefix}ML iter {it}: score={best_score:.6f} (lr={lr:.5f})")
        else:
            no_improve += 1
            if no_improve % 30 == 0:
                lr *= 0.8
            if lr < 1e-8 or no_improve > 300:
                break

    return best_p, best_score

        p = best_seq / S0
        print(f"\n--- Phase 5: Final Polish (n={n}) ---")

        # PGD with high beta
        sub_deadline = min(deadline - 5, time.monotonic() + 40)
        p_final, s_final = projected_gradient_descent(p, sub_deadline, lr_init=0.01,
                                                       momentum=0.95, beta_init=100.0,
                                                       beta_end=1000.0, verbose_prefix="Fin")
        if s_final < best_score:
            best_score = s_final
            best_seq = p_final * S0
            print(f"  Final PGD: {s_final:.6f}")

        # Quick CD
        if time.monotonic() < deadline - 3:
            p_cd, s_cd = coordinate_descent(p_final, deadline - 3, max_sweeps=10,
                                            verbose_prefix="FinCD")
            if s_cd < best_score:
                best_score = s_cd
                best_seq = p_cd * S0
                print(f"  Final CD: {s_cd:.6f}")

    # Ensure valid output
    if best_seq is None:
        best_seq = np.ones(128)

    best_seq = np.clip(best_seq, 0.0, 1000.0)
    # Ensure sum > 0.01
    if np.sum(best_seq) < 0.01:
        best_seq = np.ones(128)

    # Final verification
    final_score = score_of(best_seq)
    elapsed = time.monotonic() - t0
    print(f"\nFINAL: n={len(best_seq)}, score={final_score:.6f}, time={elapsed:.1f}s")

    return [float(x) for x in best_seq]
