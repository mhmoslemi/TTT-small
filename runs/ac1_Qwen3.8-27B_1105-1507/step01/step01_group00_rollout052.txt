```python
"""
Hybrid optimization for minimizing 2n*max(conv(a,a))/sum(a)^2.
Uses vectorized subgradient descent with momentum, LP-based direction finding,
adaptive step sizes, and multi-stage refinement. Combines fast local methods
with slower global methods for robust convergence.

import numpy as np
import time
from scipy.optimize import linprog
from scipy.sparse import csr_matrix


def build_conv_matrix_sparse(f):
    """Build sparse matrix A where (A@g)[k] = convolve(f,g)[k]."""
    n = len(f)
    # For each k (0 to 2n-2), the row k has non-zeros at positions j
    # where 0 <= j < n and 0 <= k-j < n
    # i.e., max(0, k-n+1) <= j <= min(k, n-1)
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        j_min = max(0, k - (n - 1))
        j_max = min(k, n - 1)
        for j in range(j_min, j_max + 1):
            rows.append(k)
            cols.append(j)
            vals.append(f[k - j])
    return csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))


def compute_gradient(max_indices, f, n):
    """
    Compute subgradient of max(conv(a,a)) w.r.t. a.
    For each maximizing index k, the gradient at position j is 2*f[k-j] if 0 <= k-j < n.
    We sum over all maximizing indices.
    """
    grad = np.zeros(n)
    for k in max_indices:
        j_min = max(0, k - (n - 1))
        j_max = min(k, n - 1)
        for j in range(j_min, j_max + 1):
            grad[j] += 2.0 * f[k - j]
    return grad / len(max_indices)  # Average over maximizing indices


def compute_gradient_vectorized(top_k_indices, f, n):
    """
    Vectorized version: compute weighted subgradient using top-k convolution indices.
    """
    grad = np.zeros(n)
    for k in top_k_indices:
        j_min = max(0, k - (n - 1))
        j_max = min(k, n - 1)
        if j_max >= j_min:
            # grad[j] += 2*f[k-j] for j in [j_min, j_max]
            # This is: grad[j_min:j_max+1] += 2*f[k-j_max : k-j_min+1 : -1]... 
            # Let's just do it with a slice
            indices = np.arange(j_min, j_max + 1)
            grad[indices] += 2.0 * f[k - indices]
    return grad


def lp_solve_direction(f, M, n):
    """Solve: max sum(g) s.t. g >= 0, convolve(f,g) <= M."""
    A = build_conv_matrix_sparse(f)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    try:
        result = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
        if result.success and result.x is not None:
            return result.x
    except Exception as e:
        pass
    return None

def random_perturb(f, sigma=0.05):

    S = float(np.sum(f))
    new_S = float(np.sum(new_f))

def upsample_sequence(f, new_n):
    """Upsample a sequence to a new length using linear interpolation."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    new_f = np.interp(new_x, old_x, f)
    return new_f
```

```python
def spectral_equalize(f, n_iter=1000, n_harmonics=None):
    """
    Iteratively equalize the spectrum of f while maintaining non-negativity.
    Uses Griffin-Lim style alternating projections.
    """
    f = f.copy().astype(np.float64)
    n = len(f)
    S = float(np.sum(f))
    
    # Target magnitude: flat, equal to S/sqrt(n) (so that sum of |A|^2 = S^2 by Parseval)
    # Actually, by Parseval: sum |A[k]|^2 = n * sum |a[j]|^2 (for unnormalized DFT)
    # We want sum |A[k]|^2 to be as small as possible given sum a = S.
    # By Cauchy-Schwarz: (sum a)^2 <= n * sum a^2, so sum a^2 >= S^2/n.
    # And sum |A[k]|^2 = n * sum a^2 >= n * S^2/n = S^2.
    # Equality when a is constant. So the minimum total "energy" in the spectrum is S^2,
    # achieved by the uniform sequence. But uniform has a peak autocorrelation!
    # 
    # What we actually want is to minimize the MAX of the (circular) autocorrelation,
    # which is equivalent to minimizing the max of the inverse DFT of |A|^2.
    # 
    # If |A|^2 is flat, then the circular autocorrelation is (S/n) * delta,
    # i.e., only the DC term is non-zero. This gives the ideal (zero peak sidelobe).
    # 
    # So target: |A[k]|^2 = S^2/n for all k, i.e., |A[k]| = S/sqrt(n).
    
    target_mag = S / np.sqrt(n)
    
    for it in range(n_iter):
        A = np.fft.fft(f)
        # Set magnitude to target, keep phase
        phase = A / (np.abs(A) + 1e-15)
        A = target_mag * phase
        # Inverse DFT
        f_new = np.fft.ifft(A).real
        # Project to non-negative
        f_new = np.maximum(f_new, 0)
        # Renormalize
        S_new = float(np.sum(f_new))
        if S_new > 1e-10:
            f_new = f_new * (S / S_new)
        else:
            break
        f = f_new
    
    return f
```

```python
"""
Hybrid spectral-equalization + gradient + LP optimization for minimizing
2n*max(conv(a,a))/sum(a)^2. Uses Griffin-Lim style spectral projection as a
novel initialization, fast vectorized subgradient descent with momentum for
local polishing, and LP direction finding for global improvements.
"""
import numpy as np
import time
from scipy.optimize import linprog
from scipy.sparse import csr_matrix


def build_conv_matrix_sparse(f):
    """Build sparse matrix A where (A@g)[k] = convolve(f,g)[k]."""
    n = len(f)
    rows = np.empty(n * n, dtype=np.int32)
    cols = np.empty(n * n, dtype=np.int32)
    vals = np.empty(n * n, dtype=np.float64)
    idx = 0
    for k in range(2 * n - 1):
        j_min = max(0, k - (n - 1))
        j_max = min(k, n - 1)
        for j in range(j_min, j_max + 1):
            rows[idx] = k
            cols[idx] = j
            vals[idx] = f[k - j]
            idx += 1
    rows = rows[:idx]
    cols = cols[:idx]
    vals = vals[:idx]
    return csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))


def lp_solve_direction(f, M):
    """Solve: max sum(g) s.t. g >= 0, convolve(f,g) <= M."""
    n = len(f)
    A = build_conv_matrix_sparse(f)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    try:
        result = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                         options={'time_limit': 120})
        if result.success and result.x is not None:
            return result.x
    except Exception:
        pass
    return None


def compute_subgradient(f, top_k_indices):
    """Vectorized subgradient of max conv w.r.t. f, averaged over top-k lags."""
    n = len(f)
    grad = np.zeros(n)
    for k in top_k_indices:
        i_min = max(0, k - (n - 1))
        i_max = min(k, n - 1)
        if i_max >= i_min:
            indices = np.arange(i_min, i_max + 1)
            grad[indices] += 2.0 * f[k - indices]
    return grad / max(len(top_k_indices), 1)


def spectral_equalize(f, n_iter=2000, tolerance=1e-8):
    """
    Griffin-Lim style spectral equalization for linear autocorrelation.
    Projects between flat-spectrum constraint and non-negativity + support constraint.
    """
    f = f.copy().astype(np.float64)
    n = len(f)
    S = float(np.sum(f))
    if S < 1e-10:
        return f

    # Zero-pad to length >= 2n-1 (use next power of 2)
    L = 1
    while L < 2 * n - 1:
        L *= 2

    prev_score = np.inf
    prev_change = np.inf

    for it in range(n_iter):
        # Zero-pad
        x = np.zeros(L)
        x[:n] = f

        # DFT
        X = np.fft.fft(x)

        # Target: flat magnitude = sqrt(sum(f^2))
        target = np.sqrt(np.sum(f ** 2))
        if target < 1e-15:
            break

        # Set magnitude, keep phase
        mag = np.abs(X)
        phase = X / (mag + 1e-15)
        X_new = target * phase

        # Inverse DFT
        x_new = np.fft.ifft(X_new).real

        # Project: first n elements non-negative, rest zero
        f_new = np.maximum(x_new[:n], 0.0)

        # Renormalize to keep sum constant
        S_new = float(np.sum(f_new))
        if S_new < 1e-10:
            break
        f_new = f_new * (S / S_new)

        # Check convergence
        change = np.linalg.norm(f_new - f) / max(np.linalg.norm(f), 1e-15)
        f = f_new

        if change < tolerance:
            break

    return f


def gradient_descent_momentum(f, deadline, lr_init=0.01, momentum=0.9,
                              top_k=3, verbose_prefix=""):
    """
    Momentum-based subgradient descent on max(conv(a,a)) with S fixed.
    """
    f = f.copy().astype(np.float64)
    n = len(f)
    best_f = f.copy()
    best_score = evaluate_sequence(f.tolist())
    S = float(np.sum(f))
    v = np.zeros(n)
    lr = lr_init
    no_improve = 0
    it = 0

    while time.monotonic() < deadline:
        # Batch of iterations (check time less frequently)
        for _ in range(50):
            conv_f = np.convolve(f, f)
            M = float(np.max(conv_f))
            if M <= 0:
                break

            # Get top-k indices
            sorted_idx = np.argsort(conv_f)[::-1][:top_k]

            # Compute subgradient
            grad = compute_subgradient(f, sorted_idx.tolist())

            # Normalize gradient
            gnorm = np.linalg.norm(grad)
            if gnorm < 1e-15:
                break
            grad = grad / gnorm

            # Momentum update
            v = momentum * v - lr * grad
            new_f = f + v
            new_f = np.clip(new_f, 0.0, 1000.0)

            # Renormalize to keep sum constant
            new_S = float(np.sum(new_f))
            if new_S < 1e-10:
                v = np.zeros(n)
                lr *= 0.5
                break
            new_f = new_f * (S / new_S)

            new_score = evaluate_sequence(new_f.tolist())
            it += 1

            if np.isfinite(new_score) and new_score < best_score:
                f = new_f
                best_f = f.copy()
                best_score = new_score
                no_improve = 0
            else:

                v *= 0.5
                lr *= 0.9
                if lr < 1e-10:

            if no_improve > 300:
                break

        if time.monotonic() > deadline:
            break
        if no_improve > 300:
            break
        if it % 2000 == 0 and it > 0:

def lp_iterate(f, deadline, step_size_init=0.1, max_lp_calls=50, verbose_prefix=""):
    """LP-based iterative improvement with backtracking step size."""
    f = f.copy().astype(np.float64)
    n = len(f)
    best_f = f.copy()
    best_score = evaluate_sequence(f.tolist())
    step_size = step_size_init
    no_improve = 0
    it = 0

    while time.monotonic() < deadline and it < max_lp_calls:
        conv_f = np.convolve(f, f)
        M = float(np.max(conv_f))
        if M <= 0:
            break

        start_t = time.monotonic()
        g = lp_solve_direction(f, M)
        lp_time = time.monotonic() - start_t

        if g is None:
            no_improve += 1
            if no_improve > 10:
                break
            continue

        sum_g = float(np.sum(g))
        if sum_g <= 1e-15:
            no_improve += 1
            continue

        # Rescale g to match sum of f
        g = g * (float(np.sum(f)) / sum_g)

        # Try multiple step sizes (backtracking)
        improved = False
        for t in [step_size, step_size * 0.5, step_size * 0.25, step_size * 0.1]:
            new_f = (1.0 - t) * f + t * g
            new_f = np.clip(new_f, 0.0, 1000.0)
            new_score = evaluate_sequence(new_f.tolist())
            if np.isfinite(new_score) and new_score < best_score:
                f = new_f
                best_f = f.copy()
                best_score = new_score
                it += 1
                improved = True
                break

        if improved:
            no_improve = 0
            step_size = min(step_size_init, step_size * 1.15)
        else:
            no_improve += 1
            step_size *= 0.6
            if step_size < 0.003:
                break

        if it % 5 == 0:
            print(f"    {verbose_prefix}LP it={it}: score={best_score:.6f} ({lp_time:.1f}s/call)")

    return best_f, best_score

def upsample_sequence(f, new_n):
    """Upsample via linear interpolation on [0,1] domain."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    return np.interp(new_x, old_x, f).astype(np.float64)
```
