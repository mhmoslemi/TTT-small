def evaluate_sequence(sequence: list[float]) -> float:
    """Evaluates a sequence of coefficients with enhanced security checks.
    Returns np.inf if the input is invalid."""
    #---Security Checks---
    if not isinstance(sequence, list):
        return np.inf
    if len(sequence) == 0:
        return np.inf
    valid_sequence = []
    for x in sequence:
        # Reject boolean types (they are a subclass of int)
        if isinstance(x, bool) or not isinstance(x, (int, float)):
            return np.inf
        # Reject Not-a-Number (NaN) and infinity values
        if np.isnan(x) or np.isinf(x):
            return np.inf
        # Replace other invalid types (like complex) with -1
        if isinstance(x, complex):
            continue
        valid_sequence.append(x)
    # Convert all elements to float and ensure non-negative
    sequence = np.array(valid_sequence, dtype=np.float64)
    sequence = np.maximum(0, sequence)
    coefficient_sum = np.sum(sequence)
    # Reject very small sum values (less than 0.01)
    if coefficient_sum < 0.01 or not np.isfinite(coefficient_sum):
        return np.inf
    # Reject values that are too large (greater than 1000)
    sequence = np.minimum(1000.0, sequence).clip(0, 1000)
    # Compute full convolution (mode='full')
    conv_seq = np.convolve(sequence, sequence, mode='full')
    conv_max = np.max(conv_seq)
    # Return score based on convolution
    return float(2.0 * len(sequence) * conv_max / (coefficient_sum ** 2))

def normalize_sequence(sequence):
    seq = np.array(sequence, dtype=np.float64)
    total = np.sum(seq)
    if not np.isfinite(total) or total <= 0:
        return None
    return seq / total
