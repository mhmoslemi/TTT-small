"""
Minimizes J(a) = 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Smooth-max gradient descent with FFT-accelerated convolution and tau annealing
2. LP best-response refinement for structured updates
3. Diverse structured initializations (anti-symmetric patterns, ramps, bimodal)
4. Coordinate descent polish on best candidates
Multi-length search (256 to 2048) with time-budgeted phases.

import numpy as np
from scipy.optimize import linprog
import time


def fast_conv(a):
    """Full self-convolution via FFT."""
    n = len(a)
    s = 1
    while s < 2 * n - 1:
        s <<= 1
    fa = np.fft.rfft(a, s)
    return np.fft.irfft(fa * fa, s)[:2 * n - 1]


def project_simplex(v):
    """Project v onto probability simplex {x: sum=1, x>=0}."""
    n = len(v)
    if v.sum() < 1e-12:
        return np.ones(n) / n
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - 1))[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho = rho[-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0)


def smooth_max_and_weights(c, tau):
    """Compute smooth max value and softmax weights."""
    c_shifted = c - c.max()
    exp_tau = np.exp(tau * c_shifted)
    s = exp_tau.sum()
    p = exp_tau / s
    val = c.max() + np.log(s) / tau
    return val, p


def compute_gradient(a, p):
    """Compute gradient of smooth max: 2 * sum_k p_k * a_{k-m} for each m."""
    n = len(a)
    # g[m] = sum_k p[k] * a[k-m] for valid (k-m) in [0,n-1]
    # g[m] = (a * p_rev)[2n-2-m] where p_rev = p[::-1]
    s = 1
    while s < 4 * n:
        s <<= 1
    fa = np.fft.rfft(a, s)
    fp = np.fft.rfft(p[::-1], s)
    conv_result = np.fft.irfft(fa * fp, s)
    # g[m] = conv_result[2n-2-m], for m=0..n-1
    g = conv_result[n - 1:2 * n - 1][::-1]
    return 2.0 * g


def smooth_max_grad_descent(a, tau_list, steps_per_tau, seed=0):
    """Run smooth-max gradient descent with tau annealing."""
    rng = np.random.default_rng(seed)
    a = np.asarray(a, dtype=np.float64).copy()
    a = np.clip(a, 0, None)
    if a.sum() < 1e-10:
        a = np.ones(len(a)) / len(a)
    a = a / a.sum()
    n = len(a)
    best_a = a.copy()
    best_val = float('inf')

    for tau in tau_list:
        for step in range(steps_per_tau):
            c = fast_conv(a)
            val, p = smooth_max_and_weights(c, tau)
            if val < best_val:
                best_val = val
                best_a = a.copy()

            grad = compute_gradient(a, p)
            # Adaptive step size
            a_max = a.max()
            eta = 1.0 / (4.0 * tau * a_max + 1.0)
            # Limit step to avoid going too far
            eta = min(eta, 0.5 / max(1, n))

            a_new = a - eta * grad
            a_new = project_simplex(a_new)

            # Check if this step improved things (using the same tau)
            c_new = fast_conv(a_new)
            val_new, _ = smooth_max_and_weights(c_new, tau)
            if val_new < val:
                a = a_new
            else:
                # Try smaller step
                eta = eta * 0.5
                a_new = a - eta * grad
                a_new = project_simplex(a_new)
                a = a_new

    return a, best_val, best_a


def build_A_matrix(a):
    """Build constraint matrix for LP: A[k,j] = a[k-j] if valid."""
    n = len(a)
    num_constraints = 2 * n - 1
    A = np.zeros((num_constraints, n))
    for k in range(num_constraints):
        jl = max(0, k - n + 1)
        jh = min(n - 1, k)
        if jl <= jh:
            A[k, jl:jh + 1] = a[k - jh:k - jl + 1][::-1]
    return A


def lp_best_response(a, M):
    """LP: max sum(g) s.t. (a*g)_k <= M, g >= 0."""
    n = len(a)
    c_obj = -np.ones(n)
    A = build_A_matrix(a)
    b_ub = np.full(2 * n - 1, M)
    r = linprog(c_obj, A_ub=A, b_ub=b_ub,
                bounds=[(0, None)] * n, method='highs')
    if r.success and r.x is not None:
        g = r.x
        if g.sum() > 1e-10:
            return g
    return None


def lp_refine(a, max_iters=80, deadline=None):
    """LP-based best response refinement."""
    n = len(a)
    a = np.asarray(a, dtype=np.float64).copy()
    a = np.clip(a, 1e-6, None)
    a = a / a.sum()
    best_a = a.copy()
    best_val = float('inf')

    for it in range(max_iters):
        if deadline is not None and time.monotonic() > deadline:
            break
        c = fast_conv(a)
        M = float(c.max())
        val = M
        if val < best_val:
            best_val = val
            best_a = a.copy()

        g = lp_best_response(a, M)
        if g is None:
            break
        g = g / g.sum()
        t_mix = max(0.03, 0.5 * (1.0 - it / max_iters))
        a_new = (1.0 - t_mix) * a + t_mix * g
        a_new = np.clip(a_new, 0, None)
        s = a_new.sum()
        if s < 1e-10:
            break
        a_new = a_new / s
        a = a_new

    return a, best_a, best_val


def coordinate_refine(a, max_rounds=5, deadline=None):
    """Try small perturbations of individual coordinates."""
    n = len(a)
    a = np.asarray(a, dtype=np.float64).copy()
    a = a / a.sum()
    best_val = fast_conv(a).max()
    best_a = a.copy()

    for rnd in range(max_rounds):
        if deadline is not None and time.monotonic() > deadline:
            break
        improved = False
        for j in range(n):
            if deadline is not None and time.monotonic() > deadline:
                break
            old_val = a[j]
            if old_val < 1e-8:
                continue
            for frac in [0.8, 0.9, 1.1, 1.2]:
                a[j] = old_val * frac
                s = a.sum()
                if s < 1e-8:
                    a[j] = old_val
                    continue
                a_norm = a / s
                c = fast_conv(a_norm)
                if c.max() < best_val - 1e-10:
                    best_val = c.max()
                    best_a = a_norm.copy()
                    improved = True
                    a = a_norm.copy()
                else:
                    a[j] = old_val
                    s = a.sum()
                    a = a / s
        if not improved:
            break

    return best_a


def generate_initializations(n, rng, parent=None):
    """Generate diverse initializations of length n."""
    inits = []
    t = np.arange(n, dtype=np.float64)
    t_norm = t / max(1, n - 1)

    # Uniform
    inits.append(np.ones(n) / n)

    # Linear ramp (anti-symmetric about center)
    a = 1.0 + 0.8 * (t_norm - 0.5)
    a = np.clip(a, 0.01, None)
    inits.append(a / a.sum())

    # Anti-symmetric cosine (low freq)
    for freq in [1, 2, 3]:
        a = 1.0 + 0.3 * np.cos(np.pi * freq * t_norm)
        # Make anti-symmetric: subtract mean and add anti-symmetric part
        a_asym = a - a[::-1]
        a = 1.0 + 0.3 * a_asym / (np.abs(a_asym).max() + 1e-10)
        a = np.clip(a, 0.01, None)
        inits.append(a / a.sum())

    # Bimodal (high at edges)
    for alpha in [1.0, 2.0, 4.0]:
        center_dist = np.minimum(t_norm, 1.0 - t_norm)
        a = np.exp(-alpha * center_dist ** 2 * 4.0) + 0.1
        inits.append(a / a.sum())

    # Inverted U (high in middle)
    a = 1.0 + np.sin(np.pi * t_norm)
    inits.append(a / a.sum())

    # Quadratic U-shape
    a = (t_norm - 0.5) ** 2 + 0.1
    inits.append(a / a.sum())

    # Random smooth
    for _ in range(2):
        a = rng.uniform(0.3, 1.0, n)
        # Smooth with a moving average
        kernel_size = max(3, n // 32)
        kernel = np.ones(kernel_size) / kernel_size
        a = np.convolve(a, kernel, mode='same')
        a = np.clip(a, 0.01, None)
        inits.append(a / a.sum())

    # Alternating with envelope
    alt = np.ones(n) * (1 + 0.3 * (-1) ** np.arange(n))
    env = 1.0 + 0.5 * np.sin(np.pi * t_norm)
    a = alt * env
    a = np.clip(a, 0.01, None)
    inits.append(a / a.sum())

    # Parent resampled
    if parent is not None and len(parent) > 0:
        t_old = np.linspace(0, 1, len(parent))
        t_new = np.linspace(0, 1, n)
        a = np.interp(t_new, t_old, parent)
        a = np.clip(a, 0.01, None)
        inits.append(a / a.sum())

    return inits

            try_update(a_final)

    if best is None:
        best = np.ones(256) / 256

    print(f"\nFINAL: J={best_score:.6f}, n={best.size}", flush=True)
    return [float(x) for x in best]
