Hybrid LP + Subgradient + Simulated Annealing optimization for minimizing
2*n*max(convolve(a,a))/(sum(a))^2.
Uses alternating LP, subgradient descent with subgradient estimation,

import numpy as np
from scipy.optimize import linprog
import time


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    conv = np.convolve(a, a)
    max_b = float(np.max(conv))
    sum_a = float(np.sum(a))
    if sum_a < 0.01:
        return np.inf
    return 2.0 * n * max_b / (sum_a * sum_a)


def build_A(a):
    """Build constraint matrix for LP. A[k,j] = a[k-j] if 0<=k-j<n else 0."""
    n = len(a)
    A = np.zeros((2 * n - 1, n), dtype=np.float64)
    for k in range(2 * n - 1):
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        A[k, j_start:j_end + 1] = a[k - j_end:k - j_start + 1][::-1]
    return A


def solve_lp(a, M):
    """Solve LP: max sum(g) s.t. (a*g)_k <= M for all k, g >= 0."""
    n = len(a)
    A = build_A(a)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
    if res.status == 0:
        return res.x
    return None


def subgradient_step(a, eta):
    """Compute subgradient of max(conv(a,a)) and take a step."""
    n = len(a)
    conv = np.convolve(a, a)
    k_star = int(np.argmax(conv))
    
    # Subgradient: for the active constraint k_star
    # d/d a[i] of conv[k_star] = a[k_star - i] (if 0 <= k_star-i < n) + a[i - k_star] (if 0 <= i-k_star < n)
    grad = np.zeros(n)
    for i in range(n):
        val = 0.0
        j1 = k_star - i
        if 0 <= j1 < n:
            val += a[j1]
        j2 = i - k_star
        if 0 <= j2 < n:
            val += a[j2]
        grad[i] = val
    
    # Projected gradient descent onto simplex {a >= 0, sum(a) = 1}
    a_new = a - eta * grad
    # Project onto simplex
    a_new = project_simplex(a_new)
    return a_new


def project_simplex(v):
    """Project v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - 1))[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0)


def lp_optimize(a, max_iter=200, t=0.5, verbose=False):

    n = len(a)
    a = a / np.sum(a)
    best_a = a.copy()
    best_obj = compute_obj(best_a)
    
    stall = 0
    for it in range(max_iter):
        conv = np.convolve(a, a)
        M = float(np.max(conv))
        
        g = solve_lp(a, M)
        if g is None:
            break
        g_sum = float(np.sum(g))

            stall += 1
            if stall > 10:
                break
            continue
        stall = 0

        best_blend_a = best_a

            a_try = np.clip(a_try, 0, None)
            s = np.sum(a_try)
            if s < 1e-10:

            a_try = a_try / s
            obj_try = compute_obj(a_try)

            if obj_try < best_obj:

def subgrad_optimize(a, max_iter=5000, eta0=0.01, verbose=False):

def random_perturb(a, sigma=0.1):
    """Perturb a random block of a."""
    n = len(a)
    width = max(1, n // 30)
    start = int(np.random.randint(0, n - width + 1))
    a_new = a.copy()
    a_new[start:start + width] *= np.exp(np.random.normal(0, sigma, width))
    a_new = np.clip(a_new, 0, None)
    s = np.sum(a_new)
    if s < 1e-10:
        return a
    return a_new / s

    curr_a = a.copy()
    curr_obj = compute_obj(curr_a)
    best_a = curr_a.copy()
    best_obj = curr_obj
    
    for it in range(max_iters):
        T = T0 * (T_min / T0) ** (it / max_iters)
        
        # Random perturbation
        width = max(1, n // 20)
        start = int(rng.integers(0, n - width + 1))
        candidate = curr_a.copy()
        candidate[start:start + width] *= np.exp(rng.normal(0, 0.2, width))
        candidate = np.clip(candidate, 0, None)

        if s < 1e-10:
            continue
        candidate = candidate / s
        
        obj_candidate = compute_obj(candidate)
        
        if obj_candidate < curr_obj or rng.uniform() < np.exp((curr_obj - obj_candidate) / max(T, 1e-10)):

            curr_obj = obj_candidate
            if obj_candidate < best_obj:
                best_a = candidate.copy()
                best_obj = obj_candidate
    
    return best_a, best_obj

"""
Hybrid LP + Momentum Subgradient + Simulated Annealing for minimizing
2*n*max(conv(a,a))/(sum(a))^2. Uses alternating LP with dual pass,
momentum subgradient with Polyak step, and SA-based escape.
"""

import numpy as np
from scipy.optimize import linprog
import time


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    conv = np.convolve(a, a)
    max_b = float(np.max(conv))
    sum_a = float(np.sum(a))
    if sum_a < 0.01:
        return np.inf
    return 2.0 * n * max_b / (sum_a * sum_a)


def build_A(a):
    """Build constraint matrix: A[k,j] = a[k-j] if 0<=k-j<n else 0."""
    n = len(a)
    A = np.zeros((2 * n - 1, n), dtype=np.float64)
    for k in range(2 * n - 1):
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        A[k, j_start:j_end + 1] = a[k - j_end:k - j_start + 1][::-1]
    return A


def solve_lp_for_g(a, M):
    """Solve LP: max sum(g) s.t. (a*g)[k] <= M, g >= 0. Returns g or None."""
    n = len(a)
    A = build_A(a)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
    if res.status == 0:
        return res.x
    return None


def project_simplex(v):
    """Project v onto probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    idx = np.arange(1, n + 1)
    rho = int(np.nonzero(u * idx > (cssv - 1.0))[0][-1])
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def subgradient_max_conv(a):
    """Compute subgradient of max_k conv(a,a)[k] at a. Returns (grad, k_star, val)."""
    n = len(a)
    conv = np.convolve(a, a)
    k_star = int(np.argmax(conv))
    val = float(conv[k_star])
    grad = np.zeros(n)
    for i in range(n):
        j = k_star - i
        if 0 <= j < n:
            grad[i] = 2.0 * a[j]
    return grad, k_star, val


def lp_optimize(a, max_iter=200, verbose=False):
    """LP alternating optimization with line search and dual pass."""
    n = len(a)
    a = a.copy()
    s = np.sum(a)
    if s < 1e-10:
        return a, np.inf
    a = a / s
    best_a = a.copy()
    best_obj = compute_obj(a)
    if verbose:
        print(f"    LP init: obj={best_obj:.6f}")

    stall = 0
    for it in range(max_iter):
        conv = np.convolve(a, a)
        M = float(np.max(conv))

        # Forward LP: fix a, optimize g
        g = solve_lp_for_g(a, M)
        if g is None:
            break
        g_sum = float(np.sum(g))

        if g_sum <= 1.0 + 1e-6:
            stall += 1
            if stall > 15:
                break
            continue
        stall = 0

        g = g / g_sum

        # Line search: try multiple blend ratios
        best_blend_obj = best_obj
        best_blend_a = None
        for t_try in [1.0, 0.7, 0.5, 0.3, 0.2, 0.1]:
            a_try = (1.0 - t_try) * a + t_try * g
            a_try = np.clip(a_try, 0.0, None)
            s_try = np.sum(a_try)
            if s_try < 1e-10:
                continue
            a_try = a_try / s_try
            obj_try = compute_obj(a_try)
            if obj_try < best_blend_obj:
                best_blend_obj = obj_try
                best_blend_a = a_try

        if best_blend_a is not None:
            best_a = best_blend_a.copy()
            best_obj = best_blend_obj
            a = best_a
        else:
            # Small step forward
            a = (1.0 - 0.05) * a + 0.05 * g
            a = np.clip(a, 0.0, None)
            s = np.sum(a)
            if s > 1e-10:
                a = a / s

        # Dual LP: fix the updated a, optimize the other direction
        if it % 3 == 2 and time.monotonic() % 1.0 < 0.5:
            conv2 = np.convolve(a, a)
            M2 = float(np.max(conv2))
            h = solve_lp_for_g(a, M2)
            if h is not None:
                h_sum = float(np.sum(h))
                if h_sum > 1.0 + 1e-6:
                    h = h / h_sum
                    a_dual = 0.7 * a + 0.3 * h
                    a_dual = np.clip(a_dual, 0.0, None)
                    s_dual = np.sum(a_dual)
                    if s_dual > 1e-10:
                        a_dual = a_dual / s_dual
                        obj_dual = compute_obj(a_dual)
                        if obj_dual < best_obj:
                            best_a = a_dual.copy()
                            best_obj = obj_dual
                            a = best_a

        if verbose and (it % 50 == 0):
            print(f"    LP it={it}: obj={best_obj:.6f}, M={M:.6f}, gsum={g_sum:.4f}")

    return best_a, best_obj


def momentum_subgrad(a, max_iter=5000, verbose=False):
    """Momentum subgradient with Polyak step size."""
    n = len(a)
    a = a.copy()
    s = np.sum(a)
    if s < 1e-10:
        return a, np.inf
    a = a / s
    best_a = a.copy()
    best_obj = compute_obj(a)

    # Lower bound: for sum(a)=1, max(conv) >= 1/n (from ||a||_2^2 >= 1/n)
    # So objective >= 2n * (1/n) = 2... that's not tight enough.
    # Better: max(conv) >= sum(conv)/(2n-1) = 1/(2n-1)
    # So objective >= 2n/(2n-1)
    lb = 2.0 * n / (2.0 * n - 1.0)

    momentum = 0.9
    v = np.zeros(n)
    gamma = 0.01  # base step size

    for it in range(max_iter):
        grad, k_star, val = subgradient_max_conv(a)
        g_norm = np.linalg.norm(grad)
        if g_norm < 1e-12:
            break

        # Polyak step: eta = (f(a) - lb) / ||g||^2, clipped
        obj_current = 2.0 * n * val  # since sum(a) = 1
        eta_polyak = (obj_current - lb) / (g_norm * g_norm + 1e-12)
        eta = min(eta_polyak, gamma / (1.0 + 0.01 * it))
        eta = max(eta, 1e-8)

        # Momentum update
        v = momentum * v + (1.0 - momentum) * grad
        a_new = a - eta * v
        a_new = project_simplex(a_new)

        obj_new = compute_obj(a_new)
        if obj_new < best_obj:
            best_a = a_new.copy()
            best_obj = obj_new
        a = a_new

        if verbose and (it % 500 == 0):
            print(f"    SG it={it}: obj={best_obj:.6f}, eta={eta:.2e}")

    return best_a, best_obj


def simulated_annealing(a, rng, max_iters=20000, T0=0.05, T_end=0.001, verbose=False):
    """Simulated annealing with block multiplicative perturbations."""
    n = len(a)
    curr_a = a.copy()
    curr_obj = compute_obj(curr_a)
    best_a = curr_a.copy()
    best_obj = curr_obj

    for it in range(max_iters):
        T = T0 * (T_end / T0) ** (float(it) / max_iters)

        # Random block perturbation
        width = max(1, int(n * (0.05 + 0.1 * rng.random())))
        start = int(rng.integers(0, n - width + 1))
        candidate = curr_a.copy()
        scale = np.exp(rng.normal(0, 0.3, width))
        candidate[start:start + width] *= scale
        candidate = np.clip(candidate, 0.0, None)
        s = np.sum(candidate)
        if s < 1e-10:
            continue
        candidate = candidate / s

        obj_cand = compute_obj(candidate)
        delta = obj_cand - curr_obj

        if delta < 0 or rng.uniform() < np.exp(-delta / max(T, 1e-12)):
            curr_a = candidate
            curr_obj = obj_cand
            if obj_cand < best_obj:
                best_a = candidate.copy()
                best_obj = obj_cand
                if verbose and (it % 2000 == 0):
                    print(f"    SA it={it}: best={best_obj:.6f}, T={T:.4f}")

    return best_a, best_obj


def fourier_init(n, rng):
    """Initialize using Fourier domain: try to spread out the periodogram."""
    # Start with random, then flatten the power spectrum
    a = rng.random(n)
    a = a / np.sum(a)

    # Iteratively flatten in Fourier domain
    for _ in range(5):
        # Zero-pad to 2n-1 for linear convolution
        a_pad = np.zeros(2 * n - 1)
        a_pad[:n] = a
        F = np.fft.fft(a_pad)
        # Modify to reduce peak in |F|^2 (which corresponds to peak in conv)
        # Actually, we want to reduce the peak of IFFT(|F|^2), which is the conv
        # Just do a few iterations of accelerating convergence to flat autocorrelation
        a_new = a * np.exp(-0.5 * np.abs(np.fft.ifft(np.abs(np.fft.fft(a_pad[:n]))**2).real[:n])**2)
        s = np.sum(a_new)
        if s > 1e-10:
            a = a_new / s

    return a

        sigma = max(2.0, n / 15.0)
        a = np.exp(-(np.arange(n) - p1) ** 2 / (2 * sigma ** 2)) + \
            0.7 * np.exp(-(np.arange(n) - p2) ** 2 / (2 * sigma ** 2))
    elif choice == 3:
        # Periodic pattern
        period = int(rng.integers(max(2, n // 12), max(3, n // 4)))
        pattern = np.abs(rng.normal(0, 1, period)) + 0.3
        a = np.tile(pattern, n // period + 1)[:n]
    elif choice == 4:
        # Edge-peaked (inverse cosine)
        x = np.linspace(-1, 1, n)
        a = np.sqrt(np.maximum(0, 1 - x ** 2)) + 0.1
    else:
        # Pure random
        a = rng.random(n)

    a = np.clip(a, 0.0, None)
    s = np.sum(a)
    if s < 1e-10:
        a = rng.random(n)
        s = np.sum(a)
    return a / s


def full_pipeline(a_init, n, rng, time_budget, verbose=False):
    """Run the full optimization pipeline on an initial sequence."""
    start = time.monotonic()
    a = a_init.copy()
    a = np.clip(a, 0, None)
    s = np.sum(a)
    if s < 1e-10:
        return a, np.inf
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    if verbose:
        print(f"  Init: obj={best_obj:.6f}")

    # Phase 1: LP optimization
    if time.monotonic() - start < time_budget * 0.4:
        max_lp = int((time_budget * 0.4 - (time.monotonic() - start)) / 0.05)
        max_lp = max(20, min(500, max_lp))
        a, obj = lp_optimize(a, max_iter=max_lp, verbose=verbose)
        if obj < best_obj:
            best_a = a.copy()
            best_obj = obj
            if verbose:
                print(f"  After LP: obj={best_obj:.6f}")

    # Phase 2: Momentum subgradient
    elapsed = time.monotonic() - start
    if elapsed < time_budget * 0.7:
        remaining_sg = time_budget * 0.7 - elapsed
        max_sg = int(remaining_sg / 0.005)
        max_sg = max(100, min(10000, max_sg))
        a, obj = momentum_subgrad(a, max_iter=max_sg, verbose=verbose)
        if obj < best_obj:
            best_a = a.copy()
            best_obj = obj
            if verbose:
                print(f"  After SG: obj={best_obj:.6f}")

    # Phase 3: Simulated annealing
    elapsed = time.monotonic() - start
    if elapsed < time_budget * 0.95:
        remaining_sa = time_budget * 0.95 - elapsed
        max_sa = int(remaining_sa / 0.003)
        max_sa = max(500, min(30000, max_sa))
        a, obj = simulated_annealing(a, rng, max_iters=max_sa, T0=0.03, T_end=0.0005)
        if obj < best_obj:
            best_a = a.copy()
            best_obj = obj
            if verbose:
                print(f"  After SA: obj={best_obj:.6f}")

    # Phase 4: Final LP polish
    elapsed = time.monotonic() - start
    if elapsed < time_budget:
        remaining = time_budget - elapsed
        if remaining > 1.0:
            max_lp2 = int(remaining / 0.05)
            max_lp2 = max(10, min(200, max_lp2))
            a, obj = lp_optimize(a, max_iter=max_lp2)
            if obj < best_obj:
                best_a = a.copy()
                best_obj = obj

    return best_a, best_obj


def propose_candidate(seed=42, budget_s=1500, **kwargs):
    """
    Hybrid optimization: LP alternating + momentum subgradient + SA.
    Multiple initializations, sequence lengths, and restarts.
    """
    rng = np.random.default_rng(seed)
    start_time = time.monotonic()
    deadline = start_time + float(budget_s) - 5.0

    best_seq = None
    best_score = np.inf

    # Try parent sequence
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64)
        parent = np.clip(parent, 0, 1000.0)
        if parent.size > 0 and np.isfinite(parent).all():
            obj = compute_obj(parent)
            if obj < best_score:
                best_score = obj
                best_seq = parent.copy()
            print(f"Parent (n={parent.size}): obj={obj:.6f}")
    except Exception:
        pass

    # Define search plan: (n, time_budget, n_restarts)
    # Focus on n=150 (parent length) and nearby values
    n_values = [150, 130, 170, 120, 180, 200, 100, 250]

    # Allocate time: 40% to n=150, rest distributed
    total_time = deadline - time.monotonic()
    allocations = []
    for i, n in enumerate(n_values):
        if i == 0:
            frac = 0.35
        elif i == 1:
            frac = 0.15
        elif i == 2:
            frac = 0.12
        elif i == 3:
            frac = 0.10
        else:
            frac = max(0.04, 0.08 / (i - 2 + 1))
        allocations.append(frac)
    # Normalize
    total_frac = sum(allocations)
    allocations = [f / total_frac for f in allocations]

    for idx, n in enumerate(n_values):
        if time.monotonic() > deadline:
            break
        remaining = deadline - time.monotonic()
        if remaining < 15:
            break

        time_alloc = remaining * allocations[idx]
        time_alloc = min(time_alloc, remaining)
        if time_alloc < 10:
            time_alloc = remaining

        n_restarts = max(1, int(time_alloc / 30))
        per_restart = time_alloc / n_restarts

        print(f"\nN={n}: {remaining:.0f}s left, {n_restarts} restarts, {time_alloc:.0f}s alloc")

        for r in range(n_restarts):
            if time.monotonic() > deadline:
                break

            # Special case: for n=150, first restart uses parent
            if n == 150 and r == 0 and 'best_seq' in dir() and best_seq is not None and len(best_seq) == 150:
                a_init = best_seq.copy()
            else:
                a_init = structured_init(n, rng)

            a_out, obj_out = full_pipeline(a_init, n, rng, per_restart, verbose=(r == 0))

            if obj_out < best_score:
                best_score = obj_out
                best_seq = a_out.copy()
                print(f"  n={n} r={r}: NEW BEST obj={best_score:.6f}")
            else:
                print(f"  n={n} r={r}: obj={obj_out:.6f} (best={best_score:.6f})")

    # Final intensive refinement on the best solution
    if best_seq is not None and time.monotonic() < deadline - 10:
        remaining = deadline - time.monotonic()
        n_best = len(best_seq)
        print(f"\nFinal refinement: n={n_best}, remaining={remaining:.0f}s, score={best_score:.6f}")

        # Multiple short SA passes with decreasing temperature
        for sa_pass in range(3):
            if time.monotonic() > deadline - 5:
                break
            rem = (deadline - 5) - time.monotonic()
            if rem < 5:
                break
            T0 = 0.02 / (sa_pass + 1)
            max_sa = int(rem / 0.003) / 3
            a_out, obj_out = simulated_annealing(best_seq, rng, max_iters=int(max_sa),
                                                 T0=T0, T_end=T0 * 0.01)
            if obj_out < best_score:
                best_score = obj_out
                best_seq = a_out.copy()
                print(f"  SA pass {sa_pass}: obj={best_score:.6f}")

        # Final LP
        if time.monotonic() < deadline - 3:
            rem = (deadline - 3) - time.monotonic()
            if rem > 2:
                max_lp = int(rem / 0.05)
                a_out, obj_out = lp_optimize(best_seq, max_iter=max_lp)
                if obj_out < best_score:
                    best_score = obj_out
                    best_seq = a_out.copy()
                    print(f"  Final LP: obj={best_score:.6f}")

    # Fallback
    if best_seq is None:
        best_seq = np.ones(150)
        best_score = compute_obj(best_seq)

    # Verify with evaluate_sequence
    final_score = evaluate_sequence([float(x) for x in best_seq])
    if np.isinf(final_score):
        print("WARNING: evaluate_sequence returned inf, using fallback")
        best_seq = np.ones(150)
        final_score = evaluate_sequence([float(x) for x in best_seq])

    elapsed = time.monotonic() - start_time
    print(f"\nDONE: n={len(best_seq)}, score={final_score:.6f}, time={elapsed:.1f}s")
    return [float(x) for x in best_seq]
