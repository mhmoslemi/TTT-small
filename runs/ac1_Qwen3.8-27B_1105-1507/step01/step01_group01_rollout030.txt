```python
import time
import numpy as np

def fft_convolve(a, b):
    """Fast linear convolution using FFT."""
    n = len(a) + len(b) - 1
    N = 1 << (n - 1).bit_length()  # next power of 2
    A = np.fft.rfft(a, N)
    B = np.fft.rfft(b, N)
    result = np.fft.irfft(A * B, N)
    return result[:n]

def local_score_fast(a):
    """Compute score using FFT convolution."""
    n = len(a)
    s = float(np.sum(a))
    if s <= 1e-12:
        return float('inf')
    c = fft_convolve(a, a)
    m = float(np.max(c))
    return float(2.0 * n * m / (s * s))

def normalize_sum1(a):
    a = np.asarray(a, dtype=np.float64)
    a = np.clip(a, 0.0, 1000.0)
    s = float(np.sum(a))
    if s <= 1e-300:
        return None
    return a / s
```

```python
def project_to_simplex(v):
    """Project v onto the probability simplex {x >= 0, sum x = 1}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - 1))[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    w = np.maximum(v - theta, 0)
    return w
```

```python
"""
Optimize non-negative step-function coefficients to minimize:
    score = 2*n*max(convolve(a,a)) / sum(a)^2

Strategy:
- FFT-based fast convolution and gradient computation
- Smoothed max (log-sum-exp) objective with beta annealing
- Adam optimizer with simplex projection
- Multi-start with diverse initializations and lengths
- Leverage parent sequence as strong starting point
- Periodic restarts with perturbations to escape local minima

import time
import numpy as np


def fft_convolve(a, b):
    """Fast linear convolution using real FFT."""
    n = len(a) + len(b) - 1
    N = 1 << (n - 1).bit_length()
    A = np.fft.rfft(a, N)
    B = np.fft.rfft(b, N)
    result = np.fft.irfft(A * B, N)
    return result[:n]


def compute_score(a):
    """Compute the objective score for sequence a."""
    a = np.asarray(a, dtype=np.float64)
    n = len(a)
    if n < 1:
        return float('inf')
    s = float(np.sum(a))
    if s <= 1e-300:
        return float('inf')
    c = fft_convolve(a, a)
    m = float(np.max(c))
    if not np.isfinite(m):
        return float('inf')
    return float(2.0 * n * m / (s * s))


def compute_raw_max_conv(a):
    """Compute max of convolution (assuming sum(a)=1)."""
    c = fft_convolve(a, a)
    return float(np.max(c))


def project_simplex(v):
    """Project vector v onto the probability simplex {x >= 0, sum(x) = 1}."""
    v = np.asarray(v, dtype=np.float64)
    n = len(v)
    if n == 0:
        return v.copy()
    if n == 1:
        return np.array([max(0.0, v[0])], dtype=np.float64)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - 1.0))[0]
    if len(rho) == 0:
        return np.ones(n, dtype=np.float64) / n
    rho = rho[-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    sw = float(np.sum(w))
    if sw <= 0:
        return np.ones(n, dtype=np.float64) / n
    return w / sw


def cross_correlation_fft(w, a):
    """
    Compute linear cross-correlation: result[j] = sum_k w[k] * a[k-j]
    for j = 0, ..., len(a)-1, where a[m] = 0 for m outside [0, len(a)-1].
    Uses real FFT for efficiency.
    """
    na = len(a)
    nw = len(w)
    # Need N >= nw + na - 1 for linear cross-correlation
    N = 1 << (nw + na - 2).bit_length()
    A = np.fft.rfft(a, N)
    W = np.fft.rfft(w, N)
    corr = np.fft.irfft(np.conj(A) * W, N)
    return corr[:na]


def smooth_max_and_grad(a, beta):
    """
    Compute the log-sum-exp smoothed max of conv(a,a) and its gradient.
    Returns (smoothed_value, gradient).
    """
    n = len(a)
    c = fft_convolve(a, a)
    K = len(c)  # 2n-1
    max_c = float(np.max(c))
    if max_c <= 0:
        return 0.0, np.zeros(n)

    # Numerically stable log-sum-exp
    scaled = beta * c / max_c
    scaled_max = float(np.max(scaled))
    shifted = scaled - scaled_max
    exp_shifted = np.exp(shifted)
    log_sum = scaled_max + float(np.log(np.sum(exp_shifted)))
    smoothed_val = max_c / beta * log_sum  # = max_c/beta * (max(beta*c/max_c) + log(sum(exp(...))))

    # Soft weights: p_k = exp(beta * c_k) / sum_k exp(beta * c_k)
    p = exp_shifted / float(np.sum(exp_shifted))

    # Gradient: grad_j = (2*beta/K) * sum_k p_k * a_{k-j}
    corr = cross_correlation_fft(p, a)
    grad = (2.0 * beta / K) * corr

    return smoothed_val, grad


def optimize_adam(a0, max_time, seed, beta_start=10.0, beta_end=200.0,
                  lr_init=0.05, use_sym=False):
    """
    Optimize sequence using Adam with smoothed max objective and simplex projection.
    """
    a = np.asarray(a0, dtype=np.float64).copy()
    a = np.clip(a, 0.0, 1000.0)
    s = float(np.sum(a))
    if s <= 1e-300:
        a = np.full(len(a), 1.0 / len(a))
        s = 1.0
    a = a / s  # normalize to sum 1
    n = len(a)
    if n < 2:
        return a, compute_score(a)

    best = a.copy()
    best_score = compute_score(best)

    t0 = time.monotonic()
    deadline = t0 + max_time
    rng = np.random.default_rng(seed)

    # Adam state
    m_adam = np.zeros(n, dtype=np.float64)
    v_adam = np.zeros(n, dtype=np.float64)
    t_count = 0
    lr = lr_init
    b1 = 0.85
    b2 = 0.999
    eps = 1e-8

    stuck = 0
    it = 0

    while time.monotonic() < deadline:
        # Anneal beta based on progress
        progress = min(1.0, (time.monotonic() - t0) / max(1.0, max_time))
        if progress < 0.3:
            beta = beta_start
        else:
            frac = (progress - 0.3) / 0.7
            beta = beta_start + (beta_end - beta_start) * (frac ** 0.6)

        # Compute smoothed objective and gradient
        smooth_val, grad = smooth_max_and_grad(a, beta)

        # Normalize gradient for stable Adam updates
        grad_norm = float(np.linalg.norm(grad))
        if grad_norm > 0:
            grad_use = grad / max(grad_norm, 1e-10)
        else:
            break

        # Adam update
        t_count += 1
        m_adam = b1 * m_adam + (1.0 - b1) * grad_use
        v_adam = b2 * v_adam + (1.0 - b2) * grad_use ** 2
        m_hat = m_adam / (1.0 - b1 ** t_count)
        v_hat = v_adam / (1.0 - b2 ** t_count)

        step = lr * m_hat / (np.sqrt(v_hat) + eps)

        # Try the step
        a_new = a - step
        a_new = project_simplex(a_new)

        # Evaluate
        score = compute_score(a_new)

        if np.isfinite(score) and score < best_score - 1e-12:
            best = a_new.copy()
            best_score = float(score)
            a = a_new
            stuck = 0
        else:
            stuck += 1
            # Keep a at best for next iteration if stuck
            if stuck % 20 == 0:
                a = best.copy()

        # Learning rate schedule
        if t_count % 100 == 0:
            lr *= 0.97
        if lr < 1e-5:
            lr = 1e-5

        # Restart mechanism
        if stuck > 100:
            noise_level = 0.03 + 0.04 * float(rng.random())
            noise = np.exp(rng.normal(0.0, noise_level, n))
            a = best * noise
            a = project_simplex(a)
            m_adam = np.zeros(n, dtype=np.float64)
            v_adam = np.zeros(n, dtype=np.float64)
            t_count = 0
            lr = max(lr_init * 0.3, 0.01)
            stuck = 0

        it += 1

        # Periodically check time
        if it % 50 == 0 and time.monotonic() >= deadline:
            break

    return best, best_score


def optimize_with_restarts(a0, max_time, seed, beta_start=10.0, beta_end=200.0,
                           lr_init=0.05, n_restarts=5, restart_interval=None):
    """
    Run Adam optimization with periodic restarts for robustness.
    """
    best = np.asarray(a0, dtype=np.float64).copy()
    best = np.clip(best, 0.0, 1000.0)
    s = float(np.sum(best))
    if s > 1e-300:
        best = best / s
    else:
        best = np.full(len(a0), 1.0 / len(a0))
    best_score = compute_score(best)
    n = len(best)

    t0 = time.monotonic()
    deadline = t0 + max_time

    if restart_interval is None:
        restart_interval = max(5.0, max_time / (n_restarts + 1))

    current_seed = seed
    rng = np.random.default_rng(seed)

    while time.monotonic() < deadline:
        rem = deadline - time.monotonic()
        if rem < 3.0:
            break

        seg_time = min(restart_interval, rem - 1.0)
        if seg_time < 2.0:
            break

        # Optimize from current best
        cand, sc = optimize_adam(best, seg_time, current_seed,
                                  beta_start, beta_end, lr_init)
        if cand is not None and np.isfinite(sc) and sc < best_score:
            best = cand
            best_score = float(sc)

        # Perturb for next restart
        if time.monotonic() < deadline - 5.0:
            noise_level = 0.02 + 0.03 * float(rng.random())
            noise = np.exp(rng.normal(0.0, noise_level, n))
            perturbed = best * noise
            perturbed = project_simplex(perturbed)
            sc_p = compute_score(perturbed)
            if sc_p < best_score * 1.05:  # accept if not too much worse
                best = perturbed
                best_score = float(sc_p)

        current_seed += 7919

    return best, best_score


def make_random_start(n, seed):
    """Create a random non-negative sequence normalized to sum 1."""
    rng = np.random.default_rng(seed)
    kind = int(rng.integers(0, 5))

    if kind == 0:
        a = rng.uniform(0.1, 1.0, n)
    elif kind == 1:
        # Smooth random
        a = rng.uniform(0.0, 1.0, max(2, n // 8))
        old = np.linspace(0, 1, len(a))
        new = np.linspace(0, 1, n)
        a = np.interp(new, old, a)
    elif kind == 2:
        # Cosine-weighted (emphasize edges)
        x = np.linspace(0, np.pi, n)
        a = 0.5 + 0.5 * np.cos(x) + 0.1 * rng.random(n)
    elif kind == 3:
        # Beta distribution
        a = rng.beta(0.5, 0.5, n)
    else:
        # Exponential decay with noise
        decay = np.exp(-np.linspace(0, 3, n))
        a = decay + 0.3 * rng.random(n)

    a = np.clip(a, 0.0, None)
    s = float(np.sum(a))
    if s <= 0:
        a = np.ones(n) / n
    else:
        a = a / s
    return a


def interpolate_sequence(parent, target_n, seed):
    """Interpolate parent sequence to target length."""
    rng = np.random.default_rng(seed)
    p = np.asarray(parent, dtype=np.float64)
    pp = np.clip(p, 0.0, None)
    sp = float(np.sum(pp))
    if sp <= 0:
        return np.ones(target_n) / target_n
    pp = pp / sp
    old = np.linspace(0.0, 1.0, len(pp))
    new = np.linspace(0.0, 1.0, target_n)
    a = np.interp(new, old, pp)
    # Add small noise
    a = a * np.exp(rng.normal(0.0, 0.02, target_n))
    a = np.clip(a, 0.0, None)
    s = float(np.sum(a))
    if s <= 0:
        a = np.ones(target_n) / target_n
    else:
        a = a / s
    return a


def lp_jump(a, n_constraints=80, seed=0):
    """
    LP-based jump: find g that maximizes sum(g) subject to conv(a,g) <= M.
    This provides a "direction" to move in.
    """
    try:
        from scipy.optimize import linprog
    except ImportError:
        return None

    a = np.asarray(a, dtype=np.float64)
    n = len(a)

    c = fft_convolve(a, a)
    M = float(np.max(c))
    if M <= 0:
        return None

    # Select top K convolution constraints
    K = min(n_constraints, 2 * n - 1)
    top_idx = np.argpartition(c, -K)[-K:]

    # Build constraint matrix: A[i,j] = a[top_idx[i] - j] if valid, else 0
    A = np.zeros((K, n), dtype=np.float64)
    for i, k in enumerate(top_idx):
        for j in range(n):
            m = int(k) - j
            if 0 <= m < n:
                A[i, j] = a[m]

    c_vec = -np.ones(n, dtype=np.float64)
    b_vec = np.full(K, M * (1.0 + 1e-6), dtype=np.float64)
    bounds = [(0.0, 1.0)] * n

    try:
        res = linprog(c_vec, A_ub=A, b_ub=b_vec, bounds=bounds,
                      method='highs', options={'presolve': True})
    except Exception:
        return None

    if res is None or not res.success:
        return None

    x = np.asarray(res.x, dtype=np.float64)
    x = np.clip(x, 0.0, 1.0)
    sx = float(np.sum(x))
    if sx <= 0:
        return None
    x = x / sx
    return x

    if best_seq is None:
        best_seq = make_random_start(512, int(seed))
        best_score = compute_score(best_seq)

    print(f"Phase 3: Fine polishing, budget={phase3_time:.0f}s", flush=True)

    # Very fine optimization with high beta
    if time.monotonic() < phase3_deadline:
        rem = phase3_deadline - time.monotonic()
        cur_seed = int(seed) + 90000
        cand, sc = optimize_with_restarts(
            best_seq, rem * 0.6, cur_seed,
            beta_start=50.0, beta_end=500.0,
            lr_init=0.01, n_restarts=4
        )
        if cand is not None and np.isfinite(sc) and sc < best_score:
            best_seq = cand
            best_score = float(sc)
            print(f"  P3 fine: {sc:.6f}", flush=True)

    # === Phase 4: Final LP-based refinement ===
    rem = deadline - time.monotonic()
    if rem > 10.0 and best_seq is not None:
        n = len(best_seq)
        g = lp_jump(best_seq, n_constraints=min(200, 2 * n - 1),
                    seed=int(seed) + 777777)
        if g is not None:
            for alpha in [0.05, 0.1, 0.15, 0.2, 0.3]:
                blend = (1.0 - alpha) * best_seq + alpha * g
                blend = project_simplex(blend)
                scb = compute_score(blend)
                if np.isfinite(scb) and scb < best_score:
                    best_seq = blend
                    best_score = float(scb)
                    print(f"  Final LP alpha={alpha}: {scb:.6f}", flush=True)

    # Final micro-polish
    rem = deadline - time.monotonic()
    if rem > 5.0 and best_seq is not None:
        cand, sc = optimize_adam(
            best_seq, min(rem * 0.7, 30.0), int(seed) + 111111,
            beta_start=100.0, beta_end=500.0, lr_init=0.005
        )
        if cand is not None and np.isfinite(sc) and sc < best_score:
            best_seq = cand
            best_score = float(sc)

    # Ensure valid output
    if best_seq is None:
        best_seq = np.ones(64, dtype=np.float64) / 64.0

    final = np.asarray(best_seq, dtype=np.float64)
    final = np.clip(final, 0.0, 1000.0)

    # Validate with official evaluator
    try:
        official = float(evaluate_sequence([float(x) for x in final]))
        if np.isfinite(official):
            best_score = official
        else:
            # Fallback
            fp = None
            if parent_arr is not None:
                fp = np.asarray(parent_arr, dtype=np.float64)
                fp = np.clip(fp, 0.0, 1000.0)
            if fp is not None and len(fp) > 0:
                sfp = float(np.sum(fp))
                if sfp > 0:
                    fp = fp / sfp
                official2 = float(evaluate_sequence([float(x) for x in fp]))
                if np.isfinite(official2):
                    final = fp
                    best_score = official2
    except Exception:
        pass

    print(f"FINAL: score={best_score:.6f}, n={len(final)}", flush=True)
    return [float(x) for x in final]
```
