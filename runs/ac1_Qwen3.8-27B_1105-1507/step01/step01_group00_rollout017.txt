def build_conv_matrix_sparse(f):
    """Build sparse matrix A where (A@g)[k] = convolve(f,g)[k]."""
    n = len(f)
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        j_min = max(0, k - (n - 1))
        j_max = min(k, n - 1)
        for j in range(j_min, j_max + 1):
            rows.append(k)
            cols.append(j)
            vals.append(f[k - j])
    return csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))


def solve_lp_direction(f, M):
    """Solve: max sum(g) s.t. g >= 0, convolve(f,g)_k <= M for all k."""
    n = len(f)
    A = build_conv_matrix_sparse(f)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    try:
        result = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
        if result.success:
            return result.x
    except Exception:
        pass
    return None

"""
Smooth-max gradient descent with temperature annealing and LP refinement
for minimizing 2n*max(conv(a,a))/sum(a)^2. Exploits scale invariance by
working on the probability simplex (sum(a)=1). Uses tilted initializations
and adaptive learning rates to escape local minima.
"""
import numpy as np
import time
from scipy.optimize import linprog
from scipy.sparse import csr_matrix
from scipy.interpolate import interp1d


def compute_smooth_max_and_grad(a, t):
    """Compute smooth max of conv(a,a) and its gradient w.r.t. a.
    Returns (smooth_max_value, gradient_vector)."""
    n = len(a)
    conv = np.convolve(a, a)

    # Numerically stable softmax weights
    conv_shifted = conv - np.max(conv)
    exp_t = np.exp(t * conv_shifted)
    w = exp_t / np.sum(exp_t)

    # Smooth max value
    sm = np.max(conv) + (1.0 / t) * np.log(np.sum(exp_t))

    # Gradient: d(sm)/da[j] = 2 * sum_{m=0}^{n-1} w[j+m] * a[m]
    # Computed as: 2 * convolve(a[::-1], w)[n-1 : 2*n-1]
    a_rev = a[::-1].copy()
    corr = np.convolve(a_rev, w)
    grad = 2.0 * corr[n - 1:2 * n - 1]

    return sm, grad


def smooth_max_optimize(a_init, deadline, t_start=200.0, t_end=1.0,
                        lr_start=0.02, lr_min=1e-5, momentum=0.85,
                        name="", verbose_every=50):
    """Optimize using smooth-max gradient with temperature annealing and momentum."""
    a = a_init.copy().astype(np.float64)
    a = np.clip(a, 0.0, None)
    S = np.sum(a)
    if S <= 0:
        return a_init.copy(), float('inf')
    a = a / S

    n = len(a)
    best_a = a.copy()
    best_score = 2.0 * n * np.max(np.convolve(a, a))

    v = np.zeros(n)
    lr = lr_start
    no_improve = 0
    it = 0

    while time.monotonic() < deadline:
        # Geometric temperature schedule
        progress = min(1.0, it / 800.0)
        t = t_start * (t_end / t_start) ** progress

        # Normalize gradient for stable step size
        grad_norm = np.linalg.norm(grad)
        if grad_norm > 1e-15:
            grad = grad / grad_norm

        # Momentum update
        v = momentum * v + (1.0 - momentum) * grad
        v_norm = np.linalg.norm(v)
        if v_norm > 1e-15:
            v = v / v_norm

        a_new = a - lr * v

        # Project to simplex
        a_new = np.clip(a_new, 0.0, None)
        S_new = np.sum(a_new)
        if S_new < 1e-12:
            a_new = np.ones(n) / n
        else:
            a_new = a_new / S_new

        # Evaluate (use exact max for score, smooth for direction)
        conv_new = np.convolve(a_new, a_new)
        score = 2.0 * n * np.max(conv_new)

        if score < best_score - 1e-12:
            best_score = score
            best_a = a_new.copy()
            a = a_new
            no_improve = 0
            # Occasionally increase lr back if we've been stuck then improved
        else:
            no_improve += 1
            lr *= 0.95
            v *= 0.5

        if lr < lr_min and no_improve > 100:
            break

        # Periodically reset momentum to escape local minima
        if no_improve > 80:
            v *= 0.1
            no_improve = 40

        it += 1
        if (it + 1) % verbose_every == 0 and name:
            actual = evaluate_sequence(a.tolist())
            if actual < best_score:
                best_score = actual
                best_a = a.copy()
            print(f"  {name} it={it} t={t:.1f} lr={lr:.5f} s={score:.6f} best={best_score:.6f}")

    return best_a, best_score


def build_conv_matrix_sparse(f):
    """Build sparse matrix A where (A@g)[k] = convolve(f,g)[k]."""
    n = len(f)
    n_nonzeros = 0
    for k in range(2 * n - 1):
        j_min = max(0, k - (n - 1))
        j_max = min(k, n - 1)
        n_nonzeros += j_max - j_min + 1

    rows = np.empty(n_nonzeros, dtype=np.int32)
    cols = np.empty(n_nonzeros, dtype=np.int32)
    vals = np.empty(n_nonzeros, dtype=np.float64)
    idx = 0
    for k in range(2 * n - 1):
        j_min = max(0, k - (n - 1))
        j_max = min(k, n - 1)
        for j in range(j_min, j_max + 1):
            rows[idx] = k
            cols[idx] = j
            vals[idx] = f[k - j]
            idx += 1

    return csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))


def solve_lp_direction(f, M):
    """Solve: max sum(g) s.t. g >= 0, convolve(f,g)_k <= M for all k."""
    n = len(f)
    A = build_conv_matrix_sparse(f)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    try:
        result = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                         options={'time_limit': 30})
        if result.success:
            return result.x
    except Exception:
        pass
    return None


def lp_iterate(f, deadline, step_size_init=0.1, max_iters=100):
    """Run LP-based iterations on f (normalized to sum=1)."""
    f = f.copy().astype(np.float64)
    f = np.clip(f, 0.0, None)
    S = np.sum(f)
    if S <= 0:
        return f, float('inf')
    f = f / S

    best_f = f.copy()
    best_score = 2.0 * len(f) * np.max(np.convolve(f, f))
    step_size = step_size_init
    no_improve = 0
    it = 0

    while time.monotonic() < deadline and it < max_iters:
        conv_f = np.convolve(f, f)
        M = float(np.max(conv_f))
        if M <= 0:
            break

        g = solve_lp_direction(f, M)

        if g is None:
            no_improve += 1
            if no_improve > 15:
                break
            continue

        sum_g = float(np.sum(g))
        if sum_g <= 1e-15:
            no_improve += 1
            continue

        g = g / sum_g  # Normalize g to sum=1

        new_f = (1.0 - step_size) * f + step_size * g
        new_f = np.clip(new_f, 0.0, None)
        S_new = np.sum(new_f)
        if S_new > 0:
            new_f = new_f / S_new

        score = 2.0 * len(new_f) * np.max(np.convolve(new_f, new_f))
        it += 1

        if score < best_score - 1e-12:
            f = new_f
            best_f = f.copy()
            best_score = score
            no_improve = 0
        else:
            no_improve += 1
            step_size *= 0.75
            if step_size < 0.003 or no_improve > 25:
                break

    return best_f, best_score


def softmax_param_optimize(a_init, deadline, name="", t_max=100.0, steps=300):
    """Optimize using softmax parameterization: a[i] = exp(x[i])/sum(exp(x))."""
    a = a_init.copy().astype(np.float64)
    a = np.clip(a, 1e-10, None)
    a = a / np.sum(a)
    n = len(a)

    # Initialize x from a
    x = np.log(a)
    x -= np.max(x)  # For numerical stability

    best_a = a.copy()
    best_score = 2.0 * n * np.max(np.convolve(a, a))
    lr = 0.05
    no_improve = 0
    v = np.zeros(n)

    for it in range(steps):
        if time.monotonic() > deadline:
            break

        a = np.exp(x - np.max(x))
        a = a / np.sum(a)

        conv = np.convolve(a, a)
        M = np.max(conv)
        score = 2.0 * n * M

        # Subgradient of max(conv) w.r.t. a: use one-hot at argmax
        max_lag = int(np.argmax(conv))
        grad_a = np.zeros(n)
        for j in range(n):
            k_minus_j = max_lag - j
            if 0 <= k_minus_j < n:
                grad_a[j] = 2.0 * a[k_minus_j]

        # Transform gradient to x-space
        # dx/dx = I, da/dx = diag(a) - a*a^T
        # d(f)/dx = (diag(a) - a*a^T)^T * df/da = a*(df/da) - a*sum(a*df/da)
        grad_x = a * grad_a - np.sum(a * grad_a) * a

        # Gradient descent on x
        v = 0.8 * v + 0.2 * grad_x
        x_new = x - lr * v

        # Check improvement
        a_new = np.exp(x_new - np.max(x_new))
        a_new = a_new / np.sum(a_new)
        score_new = 2.0 * n * np.max(np.convolve(a_new, a_new))

        if score_new < best_score - 1e-12:
            best_score = score_new
            best_a = a_new.copy()
            x = x_new
            a = a_new
            no_improve = 0
        else:
            no_improve += 1
            lr *= 0.93
            if lr < 1e-5:
                break

        if no_improve > 60:
            break

    return best_a, best_score


def propose_candidate(budget_s=1500, **kwargs):
    t0 = time.monotonic()
    deadline = t0 + max(0.0, float(budget_s) - 10.0)

    best_sequence = None
    best_score = float('inf')

    # Load parent
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64).copy()
        parent = np.clip(parent, 0.0, 1000.0)
        if parent.ndim == 1 and parent.size > 0 and np.all(np.isfinite(parent)):
            score = evaluate_sequence(parent.tolist())
            if np.isfinite(score):
                best_score = score
                best_sequence = parent.copy()
                parent_norm = parent / np.sum(parent)
                print(f"Parent n={parent.size}: score={score:.6f}")
    except (NameError, Exception) as e:
        print(f"Could not load parent: {e}")
        parent_norm = None

    rng = np.random.default_rng(777)

    # Lengths to try - focus on sizes near the parent
    lengths = [256, 512, 768, 1024]

    for n in lengths:
        if time.monotonic() > deadline - 40:
            print(f"Time running out, skipping n={n}")
            break

        print(f"\n{'='*50}")
        print(f"--- n={n} ---")
        print(f"{'='*50}")

        # Build initializations
        inits = []
        idx = np.arange(n, dtype=np.float64)

        # 1. Uniform
        inits.append(('uniform', np.ones(n)))

        # 2. Linear tilt (increasing)
        for alpha in [0.5, 1.0, 2.0, 4.0]:
            tilt = 1.0 + alpha * idx / (n - 1)
            inits.append((f'tilt_a{alpha}', tilt))

        # 3. Exponential tilt
        for alpha in [1.0, 2.0]:
            etilt = np.exp(alpha * idx / (n - 1))
            inits.append((f'etilt_a{alpha}', etilt))

        # 4. Cosine
        cos_t = (1.0 + np.cos(np.pi * idx / (n - 1))) / 2.0
        inits.append(('cosine', cos_t))

        # 5. Triangle
        tri = 1.0 - 2.0 * np.abs(idx - (n - 1) / 2.0) / (n - 1)
        inits.append(('triangle', tri))

        # 6. Random
        r = rng.uniform(0.1, 1.0, size=n)
        inits.append(('random', r))

        # 7. Parent interpolation
        if parent_norm is not None and len(parent_norm) != n:
            x_old = np.linspace(0, 1, len(parent_norm))
            x_new = np.linspace(0, 1, n)
            f_interp = interp1d(x_old, parent_norm, kind='cubic', fill_value='extrapolate')
            p_interp = np.clip(f_interp(x_new), 0.0, None)
            if np.sum(p_interp) > 0:
                inits.append(('parent_interp', p_interp))

        # 8. Bimodal
        two_peak = (np.exp(-0.5 * ((idx - n * 0.2) / (n / 10.0)) ** 2) +
                    np.exp(-0.5 * ((idx - n * 0.8) / (n / 10.0)) ** 2))
        inits.append(('bimodal', two_peak))

        # Time budget per initialization
        remaining = deadline - time.monotonic()
        time_per_init = min(80.0, remaining / max(1, len(inits)))

        results = []  # (score, sequence) for this n

        for label, f_init in inits:
            if time.monotonic() > deadline - 15:
                break

            f_init = np.clip(f_init.astype(np.float64), 0.0, None)
            if np.sum(f_init) <= 0:
                continue

            # Score the init
            f_norm = f_init / np.sum(f_init)
            score = 2.0 * n * np.max(np.convolve(f_norm, f_norm))

            if score < best_score:
                best_score = score
                best_sequence = f_norm.copy()
                print(f"  NEW BEST from init: n={n} {label}: {score:.6f}")

            print(f"  n={n} {label}: init={score:.6f}", end="")

            # Phase 1: Smooth-max optimization
            sub_dl = min(deadline - 5, time.monotonic() + time_per_init * 0.6)
            best_f, best_s = smooth_max_optimize(
                f_norm, sub_dl,
                t_start=200.0, t_end=1.0,
                lr_start=0.015, momentum=0.85,
                name=f"[{label}]"
            )

            if best_s < best_score:
                best_score = best_s
                best_sequence = best_f.copy()
                print(f" -> sm: {best_s:.6f} ***", end="")
            else:
                print(f" -> sm: {best_s:.6f}", end="")

            # Phase 2: LP refinement
            if time.monotonic() < deadline - 15:
                sub_dl2 = min(deadline - 5, time.monotonic() + time_per_init * 0.3)
                lp_f, lp_s = lp_iterate(best_f, sub_dl2, step_size_init=0.08)
                if lp_s < best_score:
                    best_score = lp_s
                    best_sequence = lp_f.copy()
                    print(f" -> LP: {lp_s:.6f} ***", end="")
                else:
                    print(f" -> LP: {lp_s:.6f}", end="")

            # Phase 3: Softmax parameterization optimization
            if time.monotonic() < deadline - 10:
                sub_dl3 = min(deadline - 5, time.monotonic() + time_per_init * 0.15)
                sm_f, sm_s = softmax_param_optimize(best_f, sub_dl3, name=f"[{label} softmax]")
                if sm_s < best_score:
                    best_score = sm_s
                    best_sequence = sm_f.copy()
                    print(f" -> soft: {sm_s:.6f} ***", end="")

            print()

    # Final intensive refinement on the best found
    if best_sequence is not None and time.monotonic() < deadline - 20:
        print(f"\n{'='*50}")
        print(f"--- FINAL REFINEMENT (n={len(best_sequence)}, score={best_score:.6f}) ---")
        print(f"{'='*50}")

        # Multiple rounds of smooth-max with different temperatures
        n_final = len(best_sequence)
        for round_num in range(3):
            if time.monotonic() > deadline - 10:
                break
            sub_dl = min(deadline - 5, time.monotonic() + 50)
            t_s = [100.0, 50.0, 20.0][round_num]
            best_f, best_s = smooth_max_optimize(
                best_sequence, sub_dl,
                t_start=t_s, t_end=0.5,
                lr_start=0.008, momentum=0.9,
                name=f"[final_r{round_num}]"
            )
            if best_s < best_score:
                best_score = best_s
                best_sequence = best_f.copy()
                print(f"  Round {round_num}: {best_s:.6f}")

        # LP refinement
        if time.monotonic() < deadline - 8:
            best_f, best_s = lp_iterate(best_sequence, deadline - 3, step_size_init=0.05)
            if best_s < best_score:
                best_score = best_s
                best_sequence = best_f.copy()
                print(f"  Final LP: {best_s:.6f}")

        # One more smooth-max round after LP
        if time.monotonic() < deadline - 5:
            best_f, best_s = smooth_max_optimize(
                best_sequence, deadline - 2,
                t_start=30.0, t_end=0.5,
                lr_start=0.005, momentum=0.9,
                name="[final_poly]"
            )
            if best_s < best_score:
                best_score = best_s
                best_sequence = best_f.copy()
                print(f"  Final polish: {best_s:.6f}")

    if best_sequence is None:
        return [1.0] * 128

    # Verify with evaluate_sequence
    final_score = evaluate_sequence([float(x) for x in best_sequence])
    if np.isfinite(final_score):
        print(f"\nFINAL VERIFIED: n={len(best_sequence)}, score={final_score:.6f}")
    else:
        print("WARNING: Final sequence failed evaluation, using fallback")
        return [1.0] * 256

    elapsed = time.monotonic() - t0
    print(f"Total time: {elapsed:.1f}s")
    return [float(x) for x in best_sequence]
