There are 2n-1 values of conv, and if they're all = c, then (2n-1)*c = sum^2.

objective = 2n*c/sum^2 = 2n/(2n-1) ≈ 1 for large n!

So if we could achieve a perfectly flat convolution, the objective would approach 1! The theoretical minimum is 1 (as n → ∞ with flat convolution).

The constraint is that the sequence must be non-negative. This is the key difficulty - we need a non-negative sequence whose autocorrelation (in the convolution sense) is as flat as possible.

This is related to the concept of "ideal" or "perfect" autocorrelation, which is a well-studied problem in signal processing. For non-negative sequences, this is related to " amplitude techniques" and "zero-autocorrelation" sequences (which don't exist for non-negative sequences of length > 1).

```python
"""
Hybrid optimization for minimizing 2*n*max(conv(a,a))/(sum(a))^2.
Uses soft-max gradient descent with Adam, periodic LP kicks, and coordinate search.
"""

import numpy as np
from scipy.optimize import linprog
import time

def compute_obj(a):
    n = len(a)
    if n == 0:
        return np.inf
    conv = np.convolve(a, a)
    max_b = float(np.max(conv))
    sum_a = float(np.sum(a))
    if sum_a < 1e-10:
        return np.inf
    return 2.0 * n * max_b / (sum_a * sum_a)

def build_conv_matrix(a):
    """Build the convolution matrix A where (A @ g)[k] = conv(a, g)[k]."""
    n = len(a)
    A = np.zeros((2*n-1, n))
    for k in range(2*n-1):
        for j in range(max(0, k-n+1), min(n, k+1)):
            A[k, j] = a[k-j]
    return A

def soft_max_weights(conv, T):
    """Compute softmax weights for the convolution values."""
    shifted = conv - np.max(conv)
    w = np.exp(T * shifted)
    w /= np.sum(w)
    return w

def compute_soft_grad(a, weights):
    """
    Compute gradient of soft_max_T(conv(a,a)) w.r.t. a.
    grad[i] = 2 * sum_j weights[i+j] * a[j] for j=0..n-1
    """
    n = len(a)
    grad = np.zeros(n)
    for i in range(n):
        grad[i] = 2.0 * np.dot(weights[i:i+n], a)
    return grad

def adam_step(a, grad, m, v, t, lr=0.01, beta1=0.9, beta2=0.999, eps=1e-8):
    """One step of Adam optimizer."""
    m = beta1 * m + (1 - beta1) * grad
    v = beta2 * v + (1 - beta2) * grad**2
    m_hat = m / (1 - beta1**t)
    v_hat = v / (1 - beta2**t)
    step = lr * m_hat / (np.sqrt(v_hat) + eps)
    return a - step, m, v

def lp_kick(a, M, t_step=1.0):
    """
    LP kick: find direction d that reduces the max convolution.
    max sum(d) s.t. conv(a,d)_k <= (M - conv(a,a)_k) / (2*t_step) for all k
                    -a/t_step <= d_i
    """
    n = len(a)
    conv = np.convolve(a, a)
    rhs = (M - conv) / (2.0 * t_step)
    A = build_conv_matrix(a)
    c = -np.ones(n)  # minimize -sum(d) = maximize sum(d)
    bounds = [(-a[i] / t_step, a[i] / t_step)] for i in range(n)
    res = linprog(c, A_ub=A, b_ub=rhs, bounds=bounds, method='highs')
    if res.status == 0:
        return res.x
    return None

def coordinate_search(a, rng, max_iter=10000, step_scale=0.1):
    """Random coordinate perturbation search."""
    best_a = a.copy()
    best_obj = compute_obj(best_a)
    n = len(best_a)
    
    for it in range(max_iter):
        # Pick 1-3 random coordinates to perturb
        n_perturb = int(rng.integers(1, 4))
        indices = rng.choice(n, size=n_perturb, replace=False)
        candidate = best_a.copy()
        for idx in indices:
            candidate[idx] *= np.exp(rng.normal(0, step_scale))
        candidate = np.clip(candidate, 0, None)
        s = np.sum(candidate)
        if s < 1e-10:
            continue
        obj = compute_obj(candidate)
        if obj < best_obj:
            best_a = candidate
            best_obj = obj
    
    return best_a, best_obj

def optimize_sequence(a_init, n, seed, max_time=100):
    """Full optimization pipeline for a sequence of length n."""
    rng = np.random.default_rng(seed)
    start = time.monotonic()
    deadline = start + max_time
    
    a = a_init.copy()
    a = np.clip(a, 0, None)
    s = np.sum(a)
    if s < 1e-10:
        a = rng.random(n)
        s = np.sum(a)
    a = a / s  # normalize to sum=1
    
    best_a = a.copy()
    best_obj = compute_obj(best_a)
    
    # Adam optimizer state
    m = np.zeros(n)
    v = np.zeros(n)
    t_adam = 0
    
    T = 5.0  # soft-max temperature
    lr = 0.05
    lp_counter = 0
    
    while time.monotonic() < deadline - 5:
        # Compute objective and convolution
        conv = np.convolve(a, a)
        M = float(np.max(conv))
        sum_a = float(np.sum(a))
        obj = 2.0 * n * M / (sum_a * sum_a)
        
        if obj < best_obj:
            best_a = a.copy()
            best_obj = obj
        
        # Soft-max gradient
        weights = soft_max_weights(conv, T)
        grad = compute_soft_grad(a, weights)
        
        # Normalize gradient (since we're optimizing over sum=1 constraint)
        # The gradient of f = 2n*soft_max/sum^2 is more complex, but for now
        # we just minimize the max while keeping sum roughly constant.
        # Project gradient to zero-sum subspace:
        grad -= np.mean(grad)
        
        # Skip zero gradients
        if np.max(np.abs(grad)) < 1e-10:
            # Do LP kick
            d = lp_kick(a, M, t_step=1.0)
            if d is not None and np.any(d != 0):
                a_new = a + 0.5 * d
                a_new = np.clip(a_new, 0, None)
                s_new = np.sum(a_new)
                if s_new > 1e-10:
                    a = a_new / s_new
            continue
        
        # Adam step
        t_adam += 1
        a, m, v = adam_step(a, grad, m, v, t_adam, lr=lr, beta1=0.9, beta2=0.999)
        
        # Enforce non-negativity and normalize
        a = np.clip(a, 0, None)
        s = np.sum(a)
        if s < 1e-10:
            a = best_a.copy()
            continue
        a = a / s
        
        # Periodic LP kick (every 100 iterations or when stuck)
        lp_counter += 1
        if lp_counter % 200 == 0 or (lp_counter > 50 and obj >= best_obj * 0.999):
            d = lp_kick(a, M, t_step=1.0)
            if d is not None and np.any(np.abs(d) > 1e-8):
                a_new = a + 0.3 * d
                a_new = np.clip(a_new, 0, None)
                s_new = np.sum(a_new)
                if s_new > 1e-10:
                    a_new = a_new / s_new
                    obj_new = compute_obj(a_new)
                    if obj_new < best_obj:
                        a = a_new
        
        # Adapt learning rate
        if lp_counter % 500 == 0:
            lr *= 0.8
            T *= 1.5  # sharpen the soft-max over time
        
        # Adapt temperature
        if time.monotonic() > start + max_time * 0.5:
            T = max(T, 10.0)
    
    # Final coordinate search
    if time.monotonic() < deadline - 2:
        a, obj = coordinate_search(best_a, rng, max_iter=5000, step_scale=0.05)
        if obj < best_obj:
            best_a = a
            best_obj = obj
    
    return best_a, best_obj
```
