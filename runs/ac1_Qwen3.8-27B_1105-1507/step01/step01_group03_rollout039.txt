```python
def lse_gradient_descent(f, n, time_budget, rng, initial_T=0.01):
    """
    Refine f using log-sum-exp projected gradient descent.
    f is normalized (sum=1, f>=0).
    """
    f = f.copy()
    best_f = f.copy()
    best_score = score_normalized(f, n)
    deadline = time.monotonic() + time_budget
    
    T = initial_T
    lr = 1.0 / n  # initial learning rate
    
    while time.monotonic() < deadline:
        c = np.convolve(f, f)
        c_max = np.max(c)
        c_min = np.min(c)
        
        # Compute softmax weights (with temperature T)
        c_scaled = (c - c_max) / T  # shift for numerical stability
        exp_c = np.exp(c_scaled)
        Z = np.sum(exp_c)
        w = exp_c / Z
        
        # Compute gradient: grad_j = (2/T) * sum_k w_k * f[k-j]
        # Using FFT cross-correlation
        N = 1
        while N < 3*n:
            N *= 2
        fft_w = np.fft.fft(w, N)
        fft_f = np.fft.fft(f, N)
        g_full = np.fft.ifft(fft_w * np.conj(fft_f)).real
        grad = (2.0 / T) * g_full[:n]
        
        # Projected gradient step
        f_new = f - lr * grad
        f_new = np.maximum(f_new, 0.0)
        s = f_new.sum()
        if s < 1e-300:
            f = best_f.copy()
            T *= 0.9
            continue
        f_new /= s
        
        sc = score_normalized(f_new, n)
        if sc < best_score:
            best_score = sc
            best_f = f_new.copy()
        
        f = f_new
        
        # Decay temperature and learning rate
        T *= 0.995
        lr *= 0.999
        
        if T < 1e-8:
            break
    
    return best_f, best_score
```

```python
def project_simplex(v):
    """Project v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n+1) > (cssv - 1))[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0)
```

```python
"""
Multi-phase search: LP + subgradient/log-sum-exp refinement + LP polish.
"""

import time
import numpy as np
from scipy.optimize import linprog
from scipy.sparse import csr_matrix

def normalize_sequence(arr):
    """Normalize to sum 1, non-negative."""
    try:
        x = np.asarray(arr, dtype=np.float64).ravel()
    except Exception:
        return None
    if x.size == 0:
        return None
    x = np.where(np.isfinite(x), x, 0.0)
    x = np.maximum(x, 0.0)
    s = float(x.sum())
    if s <= 1e-300:
        return None
    x /= s
    return x

def score_normalized(x, n):
    """Score = 2*n*max(conv(x,x)) assuming sum(x)=1."""
    conv = np.convolve(x, x)
    m = float(np.max(conv))
    if not np.isfinite(m):
        return float('inf')
    return float(2.0 * n * m)

def project_simplex(v):
    """Euclidean projection onto {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 1:
        return np.array([1.0])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    idx = np.arange(1, n + 1)
    cond = u * idx > (cssv - 1)
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

    Returns (new_f, new_momentum_v, active_k).

    grad = np.zeros(n)

    if j_min <= j_max:
        indices = np.arange(j_min, j_max + 1)
        grad[indices] = 2.0 * f[k_star - indices]
    
    if momentum_v is None:
        momentum_v = np.zeros(n)
    
    momentum_v = mu * momentum_v - lr * grad
    f_new = f + momentum_v
    f_new = project_simplex(f_new)
    return f_new, momentum_v, k_star

def lse_gradient(f, n, T):
    """
    Compute gradient of log-sum-exp of conv(f,f) with temperature T.
    Returns (grad, c, T_used).
    """
    c = np.convolve(f, f)
    c_max = float(np.max(c))
    c_shifted = (c - c_max) / T
    exp_c = np.exp(c_shifted)
    Z = float(np.sum(exp_c))
    if Z < 1e-300:
        # Fallback to subgradient
        k_star = int(np.argmax(c))
        grad = np.zeros(n)
        j_min = max(0, k_star - n + 1)
        j_max = min(n - 1, k_star)
        if j_min <= j_max:
            indices = np.arange(j_min, j_max + 1)
            grad[indices] = 2.0 * f[k_star - indices]
        return grad, c, T
    
    w = exp_c / Z
    
    # grad[j] = (2/T) * sum_k w[k] * f[k-j]
    # Compute cross-correlation via FFT
    N = 1
    while N < 3 * n:
        N *= 2
    fft_w = np.fft.fft(w, N)
    fft_f = np.fft.fft(f, N)
    xcorr = np.fft.ifft(fft_w * np.conj(fft_f)).real
    grad = (2.0 / T) * xcorr[:n]
    return grad, c, T

def gradient_refine(f, n, time_budget, use_lse=True, initial_T=None):
    """
    Refine f using projected (sub)gradient descent.
    f is normalized (sum=1, f>=0).
    Returns (best_f, best_score).
    """
    f = f.copy()
    best_f = f.copy()
    best_score = score_normalized(f, n)
    c0 = np.max(np.convolve(f, f))
    
    if initial_T is None:
        initial_T = c0 * 5.0
    
    T = initial_T
    lr = 1.0 / (n * c0 + 1e-300)
    mu = 0.7  # momentum
    momentum_v = np.zeros(n)
    
    deadline = time.monotonic() + max(0.0, time_budget)
    it = 0
    stall = 0
    stall_limit = 200
    
    while time.monotonic() < deadline:
        it += 1
        
        if use_lse:
            grad, c, T = lse_gradient(f, n, T)
        else:
            c = np.convolve(f, f)
            k_star = int(np.argmax(c))
            grad = np.zeros(n)
            j_min = max(0, k_star - n + 1)
            j_max = min(n - 1, k_star)
            if j_min <= j_max:
                indices = np.arange(j_min, j_max + 1)
                grad[indices] = 2.0 * f[k_star - indices]
        
        # Momentum update
        momentum_v = mu * momentum_v - lr * grad
        f_new = f + momentum_v
        f_new = project_simplex(f_new)
        
        sc = score_normalized(f_new, n)
        if sc < best_score:
            best_score = sc
            best_f = f_new.copy()
            stall = 0
        else:
            stall += 1
        
        f = f_new
        
        # Annealing schedule
        T *= 0.99
        lr *= 0.9995
        mu = 0.7 + 0.2 * (1.0 - T / initial_T * 0.5)  # increase momentum over time
        
        if stall > stall_limit:
            # Try restarting from best with fresh momentum
            f = best_f.copy()
            momentum_v = np.zeros(n)
            lr = lr * 1.5  # boost learning rate
            stall = 0
        
        if T < c0 * 0.001:
            break
    
    # Final evaluation
    final_sc = score_normalized(best_f, n)
    return best_f, final_sc

def make_conv_template(n):
    """Precompute CSR structure for conv(f, g) LP constraints."""
    m = 2 * n - 1
    k64 = np.arange(m, dtype=np.int64)
    js = np.maximum(0, k64 - n + 1).astype(np.int32)
    je = np.minimum(n - 1, k64).astype(np.int32)
    counts = (je.astype(np.int64) - js.astype(np.int64) + 1).astype(np.int64)
    total = int(counts.sum())
    
    cum = np.cumsum(counts)
    starts = np.repeat(cum - counts, counts)
    local = np.arange(total, dtype=np.int64) - starts
    
    row = np.repeat(k64, counts).astype(np.int32)
    col_base = js[row.astype(np.int64)]
    col = (col_base + local).astype(np.int32)
    offsets = (row.astype(np.int64) - col.astype(np.int64)).astype(np.int32)
    
    A_off = csr_matrix((offsets, (row, col)), shape=(m, n), dtype=np.int32)
    edge_offsets = A_off.data.astype(np.int32, copy=True)
    A_template = A_off.astype(np.float64).copy()
    A_template.data.fill(0.0)
    
    c_obj = -np.ones(n, dtype=np.float64)
    bounds = [(0.0, None)] * n
    return A_template, edge_offsets, c_obj, bounds, m

def solve_direction_lp(template, f, n, M):
    """Solve LP: max sum(g) s.t. conv(f,g)_k <= M, g >= 0."""
    A_template, edge_offsets, c_obj, bounds, m = template
    lp_scale = float(n)
    
    A = A_template.copy()
    A.data[:] = f[edge_offsets]
    A.data *= lp_scale
    
    b = np.full(m, M * lp_scale * lp_scale, dtype=np.float64)
    
    res = linprog(c_obj, A_ub=A, b_ub=b, bounds=bounds, method='highs',
                  options={'time_limit': 60, 'presolve': True})
    return res

def structured_starts(n, rng):
    """Generate diverse structured starting points."""
    starts = []
    # Uniform
    starts.append(np.ones(n, dtype=np.float64) / n)
    # Cosine window (Hann-like)
    t = np.linspace(0, np.pi, n, dtype=np.float64)
    c = 0.5 + 0.5 * np.cos(t)
    c = np.maximum(c, 1e-10)
    starts.append(c / c.sum())
    # Gaussian
    x = np.arange(n, dtype=np.float64)
    sigma = n * 0.3
    g = np.exp(-0.5 * ((x - n/2) / sigma)**2)
    starts.append(g / g.sum())
    # Triangular
    tri = np.minimum(x, n - 1 - x) + 1
    starts.append(tri / tri.sum())
    # Raised cosine (Blackman-like)
    bm = 0.42 + 0.5 * np.cos(t) + 0.08 * np.cos(2*t)
    bm = np.maximum(bm, 1e-10)
    starts.append(bm / bm.sum())
    return starts

def mix_noisy(base, rng, eps):
    """Mix base with random Dirichlet."""
    f = normalize_sequence(base)
    if f is None:
        return None
    n = f.size
    if eps <= 1e-12:
        return f
    try:
        mix = rng.dirichlet(np.ones(n, dtype=np.float64))
    except Exception:
        mix = np.ones(n) / n
    if eps >= 1.0:
        f = mix
    else:
        f = (1.0 - eps) * f + eps * mix
    return normalize_sequence(f)

def interpolate_sequence(x, n):
    """Resample normalized sequence to length n."""
    x = normalize_sequence(x)
    if x is None or n <= 0:
        return None
    m = x.size
    if m == n:
        return x
    if m == 1:
        return np.ones(n) / n
    old = np.linspace(0.0, 1.0, m)
    new = np.linspace(0.0, 1.0, n)
    out = np.interp(new, old, x)
    return normalize_sequence(out)
```

"""
Multi-phase search combining LP-based direction optimization,
subgradient/log-sum-exp projected gradient descent with momentum,
and LP polishing for minimizing 2*n*max(conv(f,f))/(sum(f))^2
over nonnegative sequences.

Improvements over previous single-phase LP approach:

- Momentum-accelerated gradient steps

- Adaptive learning rate and temperature schedules
- Random restart mechanisms to escape local minima

from scipy.sparse import csr_matrix


def normalize_sequence(arr):
    """Return a nonnegative float vector normalized to sum 1, or None."""
    try:
        x = np.asarray(arr, dtype=np.float64).ravel()
    except Exception:
        return None
    if x.size == 0:
        return None
    x = np.where(np.isfinite(x), x, 0.0)
    x = np.maximum(x, 0.0)
    s = float(x.sum())
    if s <= 1e-300:
        return None
    x /= s
    return x


def score_normalized(x, n):
    """Score = 2*n*max(conv(x,x)) assuming sum(x)=1, x>=0."""
    conv = np.convolve(x, x)
    m = float(np.max(conv))
    if not np.isfinite(m):
        return float("inf")
    return float(2.0 * n * m)


def project_simplex(v):
    """Euclidean projection onto {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 1:
        return np.array([1.0])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    idx = np.arange(1, n + 1)
    cond = u * idx > (cssv - 1)
    feasible = np.nonzero(cond)[0]
    if len(feasible) == 0:
        return np.ones(n) / n
    rho = feasible[-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def compute_subgradient(f, n):
    """
    Compute the subgradient of max_k conv(f,f)_k at f.
    Returns (grad, c, k_star) where grad is the subgradient vector.
    """
    c = np.convolve(f, f)
    k_star = int(np.argmax(c))
    grad = np.zeros(n)
    j_min = max(0, k_star - n + 1)
    j_max = min(n - 1, k_star)
    if j_min <= j_max:
        indices = np.arange(j_min, j_max + 1)
        grad[indices] = 2.0 * f[k_star - indices]
    return grad, c, k_star


def compute_lse_gradient(f, n, T):
    """
    Compute gradient of log-sum-exp(conv(f,f)/T) via FFT cross-correlation.
    Returns (grad, c, c_max).
    """
    c = np.convolve(f, f)
    c_max = float(np.max(c))
    c_shifted = (c - c_max) / T
    exp_c = np.exp(c_shifted)
    Z = float(np.sum(exp_c))
    if Z < 1e-300 or not np.isfinite(Z):
        grad, c, _ = compute_subgradient(f, n)
        return grad, c, c_max
    w = exp_c / Z
    # grad[j] = (2/T) * sum_k w[k] * f[k-j]
    # This is cross-correlation of w and f
    N = 1
    while N < 3 * n:
        N *= 2
    fft_w = np.fft.fft(w, N)
    fft_f = np.fft.fft(f, N)
    xcorr = np.fft.ifft(fft_w * np.conj(fft_f)).real
    grad = (2.0 / T) * xcorr[:n]
    return grad, c, c_max


def gradient_refine(f, n, time_budget, use_lse=True, rng=None, initial_T=None):
    """
    Refine f using projected (sub)gradient descent with momentum.
    f must be normalized (sum=1, f>=0).
    Returns (best_f, best_score).
    """
    if rng is None:
        rng = np.random.default_rng(0)
    f = f.copy()
    best_f = f.copy()
    best_score = score_normalized(f, n)
    c0 = float(np.max(np.convolve(f, f)))

    if initial_T is None:
        initial_T = max(c0 * 10.0, 1e-6)
    T = initial_T
    grad_norm_init = c0 * 2.0
    lr = 5.0 / (n * 0.01 + 1.0)
    mu = 0.5
    momentum_v = np.zeros(n)

    deadline = time.monotonic() + max(0.0, time_budget)
    it = 0
    stall = 0
    stall_limit = 300
    improve_interval = 0

    while time.monotonic() < deadline:
        it += 1

        if use_lse:
            grad, c, c_max = compute_lse_gradient(f, n, T)
        else:
            grad, c, _ = compute_subgradient(f, n)

        grad_norm = float(np.linalg.norm(grad))
        if grad_norm < 1e-300:
            break

        # Adaptive learning rate
        lr_eff = lr / (1.0 + 0.01 * it)

        # Momentum update
        momentum_v = mu * momentum_v - lr_eff * grad
        f_new = f + momentum_v
        f_new = project_simplex(f_new)

        sc = score_normalized(f_new, n)
        if sc < best_score:
            best_score = sc
            best_f = f_new.copy()
            stall = 0
            improve_interval = it - improve_interval
        else:
            stall += 1

        f = f_new

        # Annealing
        T *= 0.995
        mu = 0.5 + 0.35 * min(1.0, it / 200.0)
        lr *= 0.999

        # Restart mechanism
        if stall > stall_limit:
            f = best_f.copy()
            momentum_v = np.zeros(n)
            lr = lr * 2.0
            mu = 0.5
            stall = 0
            # Add small random perturbation
            f = f + rng.standard_normal(n) * (0.01 / n)
            f = project_simplex(f)

        if T < c0 * 0.0005 and use_lse:
            # Switch to pure subgradient for fine-tuning
            use_lse = False

    final_sc = score_normalized(best_f, n)
    return best_f, float(final_sc)


def make_conv_template(n):
    """Precompute CSR structure for conv(f, g) LP constraints."""
    m = 2 * n - 1
    k64 = np.arange(m, dtype=np.int64)
    js = np.maximum(0, k64 - n + 1).astype(np.int32)
    je = np.minimum(n - 1, k64).astype(np.int32)
    counts = (je.astype(np.int64) - js.astype(np.int64) + 1).astype(np.int64)
    total = int(counts.sum())

    cum = np.cumsum(counts)
    starts = np.repeat(cum - counts, counts)
    local = np.arange(total, dtype=np.int64) - starts

    row = np.repeat(k64, counts).astype(np.int32)
    col_base = js[row.astype(np.int64)]
    col = (col_base + local).astype(np.int32)
    offsets = (row.astype(np.int64) - col.astype(np.int64)).astype(np.int32)

    A_off = csr_matrix((offsets, (row, col)), shape=(m, n), dtype=np.int32)
    edge_offsets = A_off.data.astype(np.int32, copy=True)
    A_template = A_off.astype(np.float64).copy()
    A_template.data.fill(0.0)

    c_obj = -np.ones(n, dtype=np.float64)
    bounds = [(0.0, None)] * n
    return A_template, edge_offsets, c_obj, bounds, m


def solve_direction_lp(template, f, n, M):
    """Solve LP: max sum(g) s.t. scaled conv(f,g)_k <= M, g >= 0."""
    A_template, edge_offsets, c_obj, bounds, m = template
    lp_scale = float(n)

    A = A_template.copy()
    A.data[:] = f[edge_offsets]
    A.data *= lp_scale

    b = np.full(m, M * lp_scale * lp_scale, dtype=np.float64)

    res = linprog(c_obj, A_ub=A, b_ub=b, bounds=bounds, method="highs",
                  options={"time_limit": 90, "presolve": True})
    return res

def structured_starts(n, rng):
    """Generate diverse structured starting points for length n."""
    starts = []
    x = np.arange(n, dtype=np.float64)

    # Uniform
    starts.append(np.ones(n, dtype=np.float64) / n)

    # Cosine (Hann) window
    t = np.linspace(0.0, np.pi, n, dtype=np.float64)
    c = 0.5 + 0.5 * np.cos(t)
    c = np.maximum(c, 1e-10)
    starts.append(c / c.sum())

    # Gaussian
    sigma = n * 0.25
    g = np.exp(-0.5 * ((x - n / 2.0) / sigma) ** 2)
    g = np.maximum(g, 1e-15)
    starts.append(g / g.sum())

    # Triangular
    tri = np.minimum(x, n - 1.0 - x) + 1.0
    starts.append(tri / tri.sum())

    # Blackman window
    bm = 0.42 + 0.5 * np.cos(t) + 0.08 * np.cos(2.0 * t)
    bm = np.maximum(bm, 1e-10)
    starts.append(bm / bm.sum())

    # Flat-top approximation
    ft = 0.21557895 + 0.41663158 * np.cos(t) + 0.277263158 * np.cos(2 * t) \
         + 0.083578947 * np.cos(3 * t) + 0.006947368 * np.cos(4 * t)
    ft = np.maximum(ft, 1e-10)
    starts.append(ft / ft.sum())

    return starts


def mix_noisy(base, rng, eps):
    """Mix base with random Dirichlet for diversity."""
    f = normalize_sequence(base)
    if f is None:
        return None
    n = f.size
    if eps <= 1e-12:
        return f
    try:
        mix = rng.dirichlet(np.full(n, max(0.5, 1.0 - eps * 2), dtype=np.float64))
    except Exception:
        mix = np.ones(n, dtype=np.float64) / n
    if eps >= 1.0:
        f = mix
    else:
        f = (1.0 - eps) * f + eps * mix
    return normalize_sequence(f)


def interpolate_sequence(x, n):
    """Resample a normalized sequence to length n."""
    x = normalize_sequence(x)
    if x is None or n <= 0:
        return None
    m = x.size
    if m == n:
        return x
    if m == 1:
        return np.ones(n, dtype=np.float64) / n
    old = np.linspace(0.0, 1.0, m, dtype=np.float64)
    new = np.linspace(0.0, 1.0, n, dtype=np.float64)
    out = np.interp(new, old, x)
    return normalize_sequence(out)
