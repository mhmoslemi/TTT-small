def fast_conv(a):
    """Full self-convolution via FFT."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]

def build_sparse_A(a):
    """Build sparse constraint matrix for (a*g)_k <= M."""
    n = len(a)
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        jl = max(0, k - n + 1)
        jh = min(n - 1, k)
        for j in range(jl, jh + 1):
            rows.append(k)
            cols.append(j)
            vals.append(a[k - j])
    return coo_matrix((vals, (rows, cols)), shape=(2 * n - 1, n)).tocsr()

def lp_best_response(a, M):

    res = linprog(c, A_ub=A, b_ub=b, bounds=[(0, None)] * n, method='highs')
    if res.success and res.x is not None:

    return None

def subgradient_descent(a, max_iters, deadline):
    """Subgradient method for min max_k (a*a)_k s.t. a >= 0, sum(a) = 1."""
    n = len(a)
    a = np.maximum(a, 0)
    a = a / a.sum()
    
    # Reset the running average
    a_avg = a.copy()
    iter_count = 0
    
    for t in range(1, max_iters + 1):
        if time.monotonic() > deadline:
            break
        b = fast_conv(a)
        k_star = int(np.argmax(b))
        
        # Subgradient: g_j = 2 * a[k_star - j] (valid indices)
        # This is equivalent to: g = 2 * np.roll(a, k_star) but with zero-padding
        g = np.zeros(n)
        jl = max(0, k_star - n + 1)
        jh = min(n - 1, k_star)
        for j in range(jl, jh + 1):
            g[j] = 2 * a[k_star - j]
        
        # Step size
        eta = 1.0 / np.sqrt(t)
        
        a_new = a - eta * g
        # Project onto simplex
        a_new = project_simplex(a_new)
        
        # Running average
        a_avg = (iter_count * a_avg + a_new) / (iter_count + 1)
        iter_count += 1
        
        a = a_new
    
    # Return both the last iterate and the average
    return a, a_avg

def project_simplex(v):
    """Project v onto {x >= 0, sum(x) = 1}."""
    n = len(v)
    # Use the water-filling algorithm
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.argmax(1 + np.arange(n) * u > cssv)
    theta = (cssv[rho] - 1) / (rho + 1)
    return np.maximum(v - theta, 0)

"""
Minimizes J(a) = 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Fast subgradient descent on the simplex (O(n) per iteration)
2. Nesterov-accelerated projection for faster convergence
3. LP-based mirror descent for precise refinement
4. L-BFGS-B gradient polishing
5. Central binomial coefficient initialization + multi-start
6. Adaptive budget allocation across sequence lengths
"""

import numpy as np
from scipy.optimize import linprog, minimize
from scipy.sparse import coo_matrix
from scipy.special import gammaln
import time


def fast_conv(a):
    """Full self-convolution via FFT, O(n log n)."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]


def project_simplex(v):
    """Project v onto {x >= 0, sum(x) = 1} using water-filling."""
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.where(np.arange(1, n + 1) * u > cssv - 1.0)[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho = rho[-1]
    theta = (cssv[rho] - 1.0) / (rho + 1)
    return np.maximum(v - theta, 0.0)


def central_binomial_init(n):
    """Truncated central binomial coefficients, normalized."""
    ks = np.arange(n, dtype=np.float64)
    log_c = gammaln(2.0 * ks + 1.0) - 2.0 * gammaln(ks + 1.0) - ks * np.log(4.0)
    c = np.exp(log_c)
    c = np.maximum(c, 0.0)
    s = c.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return c / s


def subgradient_step(a, k_star):
    """Compute the 2*a shifted subgradient for the max-convolution objective."""
    n = len(a)
    g = np.zeros(n)
    jl = max(0, k_star - n + 1)
    jh = min(n - 1, k_star)
    if jl <= jh:
        g[jl:jh + 1] = 2.0 * a[k_star - jh:k_star - jl + 1][::-1]
    return g


def subgradient_descent(a0, max_iters, deadline, target_max_callable=None):
    """
    Subgradient method with decreasing step size on the simplex.
    Minimize max_k (a*a)_k subject to a >= 0, sum(a) = 1.
    Returns best iterate found.
    """
    n = len(a0)
    a = np.maximum(a0, 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    best_a = a.copy()
    best_val = fast_conv(a).max()
    avg_a = a.copy()
    count = 0

    # Polyak-style: use lower bound for max(conv) = 1/(2n-1)
    f_star_lower = 1.0 / (2.0 * n - 1.0)

    for t in range(1, max_iters + 1):
        if time.monotonic() > deadline:
            break

        b = fast_conv(a)
        f_val = float(b.max())
        k_star = int(np.argmax(b))

        if f_val < best_val:
            best_val = f_val
            best_a = a.copy()

        # Compute subgradient
        g = subgradient_step(a, k_star)
        g_norm = np.linalg.norm(g)
        if g_norm < 1e-15:
            break

        # Polyak step size with relaxed lower bound
        gap = f_val - f_star_lower
        if gap > 1e-12:
            eta = gap / (g_norm * g_norm)
            eta = min(eta, 1.0)  # cap to avoid overshooting
        else:
            break

        a_new = a - eta * g
        a_new = project_simplex(a_new)

        # Running average (last 10000 iterates weighted)
        count += 1
        if count <= 10000:
            avg_a = (avg_a * (count - 1) + a_new) / count

        a = a_new

    # Also try the average
    b_avg = fast_conv(avg_a).max()
    if b_avg < best_val:
        best_val = b_avg
        best_a = avg_a.copy()

    return best_a, best_val


def accelerated_subgradient(a0, max_iters, deadline):
    """
    Nesterov-accelerated projection for min max(conv) on simplex.
    Uses doubling technique for step size.
    """
    n = len(a0)
    a = np.maximum(a0, 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    y = a.copy()  # extrapolated point
    best_a = a.copy()
    best_val = fast_conv(a).max()
    x_old = a.copy()

    for t in range(1, max_iters + 1):
        if time.monotonic() > deadline:
            break

        # Evaluate at extrapolated point
        b_y = fast_conv(y)
        f_y = float(b_y.max())
        k_star = int(np.argmax(b_y))

        # Subgradient at y
        g = subgradient_step(y, k_star)
        g_norm = np.linalg.norm(g)
        if g_norm < 1e-15:
            break

        # Step size (Nesterov: eta ~ 2/(t+1) scaled)
        eta = 2.0 / (t + 1.0)
        # Scale by Lipschitz-like constant
        eta = eta * (1.0 / (g_norm + 1e-10))
        eta = min(eta, 1.0)

        # Projected gradient step
        x_new = y - eta * g
        x_new = project_simplex(x_new)

        # Update extrapolation
        x_old, x_new_copy = x_old, x_new.copy()
        momentum = 2.0 / (t + 3.0)
        y = x_new_copy + momentum * (x_new_copy - x_old)
        y = project_simplex(y)

        a = x_new

        # Track best
        b_a = fast_conv(a)
        f_a = float(b_a.max())
        if f_a < best_val:
            best_val = f_a
            best_a = a.copy()

    return best_a, best_val


def build_sparse_A(a):
    """Build sparse matrix for convolution constraints."""
    n = len(a)
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        jl = max(0, k - n + 1)
        jh = min(n - 1, k)
        for j in range(jl, jh + 1):
            rows.append(k)
            cols.append(j)
            vals.append(a[k - j])
    return coo_matrix((vals, (rows, cols)), shape=(2 * n - 1, n)).tocsr()


def lp_best_response(a, M):
    """Solve LP: max sum(g) s.t. (a*g)_k <= M, g >= 0."""
    n = len(a)
    c = -np.ones(n)
    A = build_sparse_A(a)
    b = np.full(2 * n - 1, M)
    res = linprog(c, A_ub=A, b_ub=b, bounds=[(0, None)] * n,
                  method='highs',
                  options={'presolve': True, 'time_limit': 5.0})
    if res.success and res.x is not None:
        return res.x
    return None


def lp_mirror_descent(a0, max_iters, deadline, init_step=0.5, decay=0.97):
    """LP-based mirror descent (best response) iterations."""
    n = len(a0)
    a = np.maximum(a0, 0.0)
    s = a.sum()
    if s < 1e-15:
        return a0
    a = a / s

    best_a = a.copy()
    best_val = fast_conv(a).max()
    step = init_step

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        b = fast_conv(a)
        M = float(b.max())
        g = lp_best_response(a, M)
        if g is None:
            break
        g_sum = g.sum()
        if g_sum < 1e-12:
            break
        g = g / g_sum

        a_new = (1.0 - step) * a + step * g
        a_new = np.maximum(a_new, 0.0)
        s_new = a_new.sum()
        if s_new < 1e-12:
            step *= 0.5
            continue
        a_new = a_new / s_new

        b_new = fast_conv(a_new)
        M_new = float(b_new.max())

        if M_new < M * (1.0 + 1e-10):
            a = a_new
            if M_new < best_val:
                best_val = M_new
                best_a = a.copy()
            step = min(0.8, step * 1.02)
        else:
            step *= 0.5
            if step < 0.005:
                break

        step *= decay

    return best_a


def polish_lbfgs(a0, deadline, max_iter=500):
    """L-BFGS-B polishing of the max-convolution objective."""
    n = len(a0)
    a = np.maximum(a0, 0.0)
    s = a.sum()
    if s < 1e-15:
        return a0
    a = a / s

    def obj(x):
        ax = np.maximum(x, 0.0)
        sx = ax.sum()
        if sx < 1e-10:
            return 1e10
        ax = ax / sx
        return float(fast_conv(ax).max())

    res = minimize(obj, a, method='L-BFGS-B',
                   bounds=[(0, None)] * n,
                   options={'maxiter': max_iter, 'max_fun': 5000,
                            'ftol': 1e-16, 'gtol': 1e-14, 'disp': False})
    if res.fun < 1e10:
        result = np.maximum(res.x, 0.0)
        rs = result.sum()
        if rs > 1e-10:
            return result / rs
    return a
