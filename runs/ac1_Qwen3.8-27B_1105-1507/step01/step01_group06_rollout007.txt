"""
Minimizes J(a) = 2*n*max(convolve(a,a))/(sum(a))^2 using:
1. L-BFGS-B with log-sum-exp smoothing of the max (FFT-accelerated gradients)
2. Beta-annealing for progressive peak focusing
3. LP best-response polish (from previous SOTA)
4. Multi-scale search over n = 256..8192 with diverse initializations
"""

import numpy as np
from scipy.optimize import minimize, linprog
import time

def fast_conv(a):
    n = len(a)
    s = 1
    while s < 2 * n - 1:
        s <<= 1
    fa = np.fft.rfft(a, s)
    return np.fft.irfft(fa * fa, s)[:2*n-1]

def logsumexp_obj(a, beta):
    """Compute (1/beta)*log(sum exp(beta*b)) / S^2 * 2n, where b=conv(a,a), S=sum(a)."""
    S = a.sum()
    if S < 1e-10:
        return 1e10
    b = fast_conv(a)
    beta_b = beta * b
    beta_b_max = beta_b.max()
    L = (beta_b_max + np.log(np.sum(np.exp(beta_b - beta_b_max)))) / beta
