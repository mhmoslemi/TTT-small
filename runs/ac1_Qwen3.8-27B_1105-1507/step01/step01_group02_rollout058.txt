import numpy as np
import time
from scipy.optimize import linprog

def project_simplex(v):
    """Project vector v onto {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v
    # Sort in descending order
    u = np.sort(v)[::-1]
    cv = np.cumsum(u) - 1.0
    # Find rho
    rho_candidates = np.where(u * np.arange(1, n + 1) > cv)[0]
    if len(rho_candidates) == 0:
        return np.ones(n) / n
    rho = rho_candidates[-1]
    theta = cv[rho] / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def fast_convolve(a):
    """Compute full linear convolution of a with itself using FFT."""
    n = len(a)
    size = 2 * n - 1
    N = 1 << (size - 1).bit_length()
    fa = np.fft.rfft(a, N)
    fc = fa * fa
    result = np.fft.irfft(fc, N)[:size]
    result[result < 0] = 0.0
    return result

def compute_subgradient_avg(a, peak_indices, c):
    """Compute weighted subgradient with weights based on peak values."""
    n = len(a)
    grad = np.zeros(n)
    c_max = c[peak_indices].max()
    # Use exponential weighting based on how close each peak is to the max
    tau = 50.0  # sharpness parameter
    weights = np.exp(tau * (c[peak_indices] - c_max))
    weights /= weights.sum()
    for w, k in zip(weights, peak_indices):
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        if lo <= hi:
            grad[lo:hi+1] += w * 2.0 * a[k-hi:k-lo+1][::-1]
    return grad

"""
Optimizes non-negative sequence to minimize 2*n*max(conv(a,a))/sum(a)^2.
Uses Nesterov-accelerated subgradient descent with soft-max peak weighting,
FFT-based convolution, aggressive LP-based improvement steps with line search,
diverse multi-start initializations, and annealed sharpness for peak focusing.
Targets improvement over previous best of 1.514229 toward 1.5027.

import numpy as np
import time
from scipy.optimize import linprog


def project_simplex(v):
    """Project vector v onto {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cv = np.cumsum(u) - 1.0
    rho_candidates = np.where(u * np.arange(1, n + 1) > cv)[0]
    if len(rho_candidates) == 0:
        return np.ones(n) / n
    rho = rho_candidates[-1]
    theta = cv[rho] / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def fast_convolve(a):
    """Compute full linear convolution of a with itself using FFT."""
    n = len(a)
    size = 2 * n - 1
    N = 1 << (size - 1).bit_length()
    fa = np.fft.rfft(a, N)
    result = np.fft.irfft(fa * fa, N)[:size]
    result[result < 0] = 0.0
    return result


def subgradient_weighted(a, c, tau_sm):
    """
    Compute subgradient of max(conv(a,a)) with soft-max exponential weighting.
    Higher tau_sm focuses more on the top peaks.
    """
    n = len(a)
    peak = c.max()
    # Exponential weights: ~1 near peak, ~0 in valleys
    weights = np.exp(tau_sm * (c - peak))
    w_sum = weights.sum()
    if w_sum < 1e-30:
        peak_idx = np.argmax(c)
        weights = np.zeros(len(c))
        weights[peak_idx] = 1.0
        w_sum = 1.0
    weights /= w_sum

    grad = np.zeros(n)
    active = np.where(weights > 1e-8)[0]
    for k in active:
        w = weights[k]
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        if lo <= hi:
            grad[lo:hi + 1] += w * 2.0 * a[k - hi:k - lo + 1][::-1]
    return grad


def lp_improve(a, P, time_limit=4.0):
    """
    Solve LP: max sum(b) s.t. conv(a, b) <= P, b >= 0.
    Returns b normalized to sum 1, or None.
    """
    n = len(a)
    if n > 4000:
        return None
    # Build convolution matrix A: (A @ b)[k] = conv(a, b)[k]
    A = np.zeros((2 * n - 1, n), dtype=np.float64)
    for k in range(2 * n - 1):
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        A[k, lo:hi + 1] = a[k - hi:k - lo + 1][::-1]

    c_obj = -np.ones(n)
    b_ub = np.full(2 * n - 1, P)

    try:
        res = linprog(c_obj, A_ub=A, b_ub=b_ub,
                      bounds=[(0.0, None)] * n, method='highs',
                      options={'time_limit': time_limit, 'presolve': True})
        if res.success and res.x is not None:
            b = res.x
            s = b.sum()
            if s > 1e-10:
                return b / s
    except Exception:
        pass
    return None


def peak_deflation(a, n_peaks=5, n_steps=3, step_size=0.01):
    """
    Local repair: take small subgradient steps focused on the top peaks.
    Reduces the worst peaks quickly after an LP or perturbation step.
    """
    for _ in range(n_steps):
        c = fast_convolve(a)
        peak = c.max()
        # Focus on top n_peaks
        top_thresh = np.sort(c)[::-1][min(n_peaks, len(c) - 1)]
        active = np.where(c >= top_thresh * 0.999)[0]
        if len(active) > 20:
            idx = np.argsort(c[active])[::-1][:10]
            active = active[idx]
        # Equal weights on active peaks
        grad = np.zeros(len(a))

            lo = max(0, k - n_a + 1)
            hi = min(n_a - 1, k)
            if lo <= hi:
                grad[lo:hi + 1] += (1.0 / len(active)) * 2.0 * a[k - hi:k - lo + 1][::-1]
        a = project_simplex(a - step_size * grad)
    return a

    inits.append(a)
    names.append('two_bump')

    # Triangular
    a = np.minimum(np.arange(n) + 1, np.arange(n, 0, -1))
    a = a / a.sum()
    inits.append(a)
    names.append('triangular')

    # Random
    a = rng.random(n)
    a = a / a.sum()
    inits.append(a)
    names.append('random')

    return list(zip(names, inits))


def propose_candidate(budget_s=1500):
    """
    Multi-phase search: diverse initializations optimized with Nesterov
    subgradient + LP improvements + peak deflation. Multi-start across
    different lengths with parent-guided initialization.
    """
    overall_deadline = time.time() + budget_s - 25.0
    rng = np.random.default_rng(2024)

    best_score = float('inf')
    best_seq = None

    # Load parent solution
    parent = None
    try:
        parent = np.array(height_sequence_1, dtype=np.float64)
        parent = np.clip(parent, 0, 1000)
        if parent.sum() > 0.01:
            parent = parent / parent.sum()
    except Exception:
        pass

    # Build config list: (priority, n, name, init_array)
    configs = []

    if parent is not None:
        n_p = len(parent)
        configs.append((0, n_p, 'parent_native', parent.copy()))
        for n in [512, 768, 1024]:
            if n != n_p:
                x_orig = np.linspace(0, 1, n_p)
                x_new = np.linspace(0, 1, n)
                a = np.interp(x_new, x_orig, parent)
                a = np.maximum(a, 0)
                s = a.sum()
                if s > 0:
                    a = a / s
                    configs.append((1, n, f'parent_{n}', a))

    # Additional structured starts at key lengths
    for n in [512, 1024]:
        x = np.linspace(0, 1, n)
        a = 0.5 + 0.5 * np.cos(2 * np.pi * x) + 0.005
        a = a / a.sum()
        configs.append((2, n, f'cos_{n}', a))

    if parent is not None and len(parent) == 512:
        # Skip duplicate 512 configs, add 256
        a = np.interp(np.linspace(0, 1, 256), np.linspace(0, 1, 512), parent)
        a = np.maximum(a, 0)
        a = a / a.sum()
        configs.append((2, 256, 'parent_256', a))

    # Sort by priority then length
    configs.sort(key=lambda x: (x[0], x[1]))

    num_configs = len(configs)
    remaining = overall_deadline - time.time()
    print(f"={budget_s}s budget, {remaining:.0f}s remaining for optimization")
    print(f"Configs: {num_configs}, target=1.5027, prev_best=1.514229")

    # Time allocation: top 4 get 55% total, rest split 45%
    top_n = min(4, num_configs)
    top_alloc = remaining * 0.55 / top_n if top_n > 0 else 0
    rest_alloc = remaining * 0.45 / max(1, num_configs - top_n) if num_configs > top_n else 0

    for i, (priority, n, name, init_a) in enumerate(configs):
        if time.time() > overall_deadline - 30:
            break

        if i < top_n:
            alloc = min(top_alloc, overall_deadline - time.time() - 20)
        else:
            alloc = min(rest_alloc, overall_deadline - time.time() - 20)

        if alloc < 12:
            continue

        dl = time.time() + alloc
        t0 = time.time()
        print(f"[{i+1}/{num_configs}] {name} (n={n}): {alloc:.0f}s allocated")

        try:
            a_opt = optimize(init_a, dl, rng, use_lp=(n <= 3000))
            score = evaluate_sequence(a_opt.tolist())
        except Exception as e:
            print(f"  Error: {e}")
            continue

        elapsed = time.time() - t0
        print(f"  score={score:.6f} ({elapsed:.0f}s used) | remaining={overall_deadline - time.time():.0f}s")

        if np.isfinite(score) and score < best_score:
            best_score = score
            best_seq = a_opt.copy()
            print(f"  >>> NEW BEST: {best_score:.6f} (n={n}, {name})")

    # Final intensive polish
    if best_seq is not None and time.time() < overall_deadline - 5:
        pol_time = overall_deadline - time.time()
        print(f"\nFinal polish: {pol_time:.0f}s remaining, current best={best_score:.6f}")
        a_pol = optimize(best_seq, overall_deadline, rng, use_lp=(len(best_seq) <= 3000))
        score = evaluate_sequence(a_pol.tolist())
        print(f"After polish: {score:.6f}")
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_seq = a_pol

    if best_seq is None:
        print("FALLBACK to uniform 512")
        best_seq = np.ones(512)

    result = [float(max(0.0, x)) for x in best_seq]
    fs = evaluate_sequence(result)
    print(f"\nFINAL: score={fs:.6f}, length={len(result)}")
    return result
