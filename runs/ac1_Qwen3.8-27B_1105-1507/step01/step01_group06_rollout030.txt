```python
def fast_conv(a):
    """Full self-convolution via FFT."""
    n = len(a)
    if n == 0:
        return np.array([])
    s = 1
    while s < 2 * n - 1:
        s <<= 1
    fa = np.fft.rfft(a, s)
    return np.fft.irfft(fa * fa, s)[:2 * n - 1]


def compute_objective(a):
    """Compute J(a) = 2*n*max(a*a)/(sum(a))^2 directly."""
    n = len(a)
    s = a.sum()
    if s < 1e-10:
        return np.inf
    b = fast_conv(a)
    return float(2.0 * n * b.max() / (s * s))


def solve_direction_lp(a, c, K):
    """
    Solve LP to find direction d that reduces the linearized peak.
    min t s.t. c_k + 2*(a*d)_k <= t for k in K, sum(d)=0, d_i >= -a_i
    Returns d (length n) or None.
    """
    n = len(a)
    m = len(K)
    
    # Build A_ub: m x (n+1)
    A_ub = np.zeros((m, n + 1))
    b_ub = np.zeros(m)
    
    for idx, k in enumerate(K):
        # (a*d)_k = sum_j a_{k-j} * d_j for valid j
        # Coefficient of d_j is 2*a_{k-j} if 0 <= k-j <= n-1
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        for j in range(j_start, j_end + 1):
            A_ub[idx, j] = 2.0 * a[k - j]
        A_ub[idx, n] = -1.0  # coefficient of t
        b_ub[idx] = -c[k]
    
    # Equality: sum(d) = 0
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([0.0])
    
    # Objective: minimize t
    c_obj = np.zeros(n + 1)
    c_obj[n] = 1.0
    
    # Bounds
    bounds = []
    for i in range(n):
        bounds.append((-a[i], 1.0))  # d_i in [-a_i, 1.0]
    bounds.append((0.0, c.max() * 3 + 1.0))  # t in [0, 3*max(c)+1]
    
    res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                  bounds=bounds, method='highs')
    
    if res.success:
        d = res.x[:n]
        if np.abs(d).max() > 1e-12:
            return d
    return None


def bundle_optimize(a, deadline, max_iters=200, m_max=80):
    """
    Bundle method: iteratively solve LP for descent direction.
    Modifies a in-place and returns final score.
    """
    n = len(a)
    a = a.copy()
    a = np.clip(a, 0, None)
    s = a.sum()
    if s < 1e-10:
        return a, np.inf
    a = a / s
    score = compute_objective(a)
    
    best_a = a.copy()
    best_score = score
    
    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        
        c = fast_conv(a)
        t = c.max()
        
        # Select working set: top m indices
        m = min(m_max, 2 * n - 1)
        # Use indices where c_k > t - 0.5*t (i.e., above 50% of peak)
        threshold = t * 0.5
        K = np.where(c >= threshold)[0]
        if len(K) < 5:
            K = np.argsort(c)[-m:]
        else:
            K = K[:m]  # cap at m
        
        d = solve_direction_lp(a, c, K)
        if d is None:
            break
        
        # Line search
        best_alpha = 0.0
        best_try_score = score
        for alpha in [1.0, 0.7, 0.5, 0.3, 0.2, 0.1, 0.05]:
            a_try = a + alpha * d
            a_try = np.clip(a_try, 0.0, None)
            s_try = a_try.sum()
            if s_try < 1e-10:
                continue
            a_try = a_try / s_try
            s_try_val = compute_objective(a_try)
            if s_try_val < best_try_score:
                best_try_score = s_try_val
                best_alpha = alpha
        
        if best_alpha == 0.0:
            # Try smaller alphas
            for alpha in [0.02, 0.01, 0.005]:
                a_try = a + alpha * d
                a_try = np.clip(a_try, 0.0, None)
                s_try = a_try.sum()
                if s_try < 1e-10:
                    continue
                a_try = a_try / s_try
                s_try_val = compute_objective(a_try)
                if s_try_val < best_try_score:
                    best_try_score = s_try_val
                    best_alpha = alpha
        
        if best_alpha == 0.0:
            break
        
        a = a + best_alpha * d
        a = np.clip(a, 0.0, None)
        s = a.sum()
        if s < 1e-10:
            break
        a = a / s
        score = best_try_score
        
        if score < best_score:
            best_score = score
            best_a = a.copy()
        
        # Check convergence
        if it > 0 and score > best_score * 1.0001:
            # Allow some oscillation
            pass
    
    return best_a, best_score


def refine_lbfgsb(a, deadline, max_iter=300):
    """L-BFGS-B refinement."""
    n = len(a)
    a = np.clip(a, 0, None)
    s = a.sum()
    if s < 1e-10:
        return a, np.inf
    a = a / s
    
    def obj(x):
        ax = np.maximum(x, 0.0)
        s = ax.sum()
        if s < 1e-8:
            return 1e10
        ax = ax / s
        b = fast_conv(ax)
        return float(2.0 * n * b.max())
    
    res = minimize(obj, a, method='L-BFGS-B',
                   bounds=[(0, None)] * n,
                   options={'maxiter': max_iter, 'max_fun': 5000,
                            'ftol': 1e-16, 'gtol': 1e-12, 'disp': False})
    
    if res.fun < 1e10 and time.monotonic() < deadline:
        a_out = np.clip(res.x, 0, None)
        s_out = a_out.sum()
        if s_out > 1e-10:
            a_out = a_out / s_out
            return a_out, float(2.0 * n * fast_conv(a_out).max())
    
    return a, compute_objective(a)


def resize_sequence(a, new_n):
    """Resize sequence using interpolation."""
    old_n = len(a)
    if old_n == new_n:
        return a.copy()
    t_old = np.linspace(0, 1, old_n)
    t_new = np.linspace(0, 1, new_n)
    return np.maximum(np.interp(t_new, t_old, a), 0.0)
```

```python
"""
Minimizes J(a) = 2*n*max(convolve(a,a))/(sum(a))^2 using:
1. Bundle method with LP linearization of the peak constraint (primary)
2. L-BFGS-B gradient refinement for local polish
3. Multi-start across diverse initializations and sequence lengths (256-4096)
The LP at each bundle iteration finds the direction d that maximally reduces
the linearized peak of the autocorrelation, subject to non-negativity and
normalization constraints. This is a first-order (Gauss-Newton-like) step
for the underlying QCQP.
"""

import numpy as np
from scipy.optimize import linprog, minimize
from functools import partial
import time


def fast_conv(a):
    """Full self-convolution via FFT, O(n log n)."""
    n = len(a)
    if n == 0:
        return np.array([])
    s = 1
    while s < 2 * n - 1:
        s <<= 1
    fa = np.fft.rfft(a, s)
    return np.fft.irfft(fa * fa, s)[:2 * n - 1]


def compute_obj(a):
    """Compute J(a) = 2*n*max(a*a)/(sum(a))^2 directly."""
    n = len(a)
    s = a.sum()
    if s < 1e-10:
        return np.inf
    b = fast_conv(a)
    return float(2.0 * n * b.max() / (s * s))


def solve_lp_direction(a, c, K):
    """
    Solve LP: min t s.t. c_k + 2*(a*d)_k <= t for k in K, sum(d)=0, d_i >= -a_i.
    Returns (d, t_value) or (None, None).
    """
    n = len(a)
    m = len(K)
    if m == 0:
        return None, None

    A_ub = np.zeros((m, n + 1))
    b_ub = np.zeros(m)

    for idx, k in enumerate(K):
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        js = np.arange(j_start, j_end + 1)
        A_ub[idx, js] = 2.0 * a[k - js]
        A_ub[idx, n] = -1.0
        b_ub[idx] = -c[k]

    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([0.0])

    c_obj = np.zeros(n + 1)
    c_obj[n] = 1.0

    bounds = [(-a[i], 0.5) for i in range(n)]
    bounds.append((0.0, c.max() * 3.0 + 1.0))

    res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                  bounds=bounds, method='highs')

    if res.success:
        d = res.x[:n]
        if np.abs(d).max() > 1e-12:
            return d, float(res.x[n])
    return None, None


def bundle_step(a, m_max=60):
    """One bundle method iteration. Returns (new_a, improved)."""
    n = len(a)
    c = fast_conv(a)
    t = c.max()

    # Select working set: indices where c is above 50% of peak
    threshold = t * 0.5
    K = np.where(c >= threshold)[0]
    if len(K) < 10:
        idx = np.argsort(c)
        K = idx[-min(m_max, len(idx)):]
    elif len(K) > m_max:
        # Keep the top m_max
        K_idx = np.argsort(c[K])
        K = K[K_idx[-m_max:]]

    d, t_lp = solve_lp_direction(a, c, K)
    if d is None:
        return a, False

    # Line search with multiple step sizes
    best_alpha = 0.0
    current_score = compute_obj(a)
    for alpha in [1.0, 0.6, 0.4, 0.25, 0.15, 0.08, 0.04]:
        a_try = a + alpha * d
        a_try = np.clip(a_try, 0.0, None)
        s_try = a_try.sum()
        if s_try < 1e-10:
            continue
        a_try = a_try / s_try
        score = compute_obj(a_try)
        if score < best_alpha * 0 + current_score:
            best_score = score
            best_alpha = alpha
            best_a_try = a_try.copy()

    if best_alpha == 0.0:
        # Try even smaller steps
        for alpha in [0.02, 0.01]:
            a_try = a + alpha * d
            a_try = np.clip(a_try, 0.0, None)
            s_try = a_try.sum()
            if s_try < 1e-10:
                continue
            a_try = a_try / s_try
            score = compute_obj(a_try)
            if score < current_score:
                best_score = score
                best_alpha = alpha
                best_a_try = a_try.copy()
                break

    if best_alpha == 0.0:
        return a, False

    return best_a_try, True


def bundle_optimize(a, deadline, max_iters=200, m_max=60):
    """Run bundle method until convergence or time limit."""
    n = len(a)
    a = np.clip(a.astype(np.float64), 0.01, None)
    a = a / a.sum()
    best_a = a.copy()
    best_score = compute_obj(a)
    stagnant = 0

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        a, improved = bundle_step(a, m_max)

        if not improved:
            # Add small perturbation to escape
            if stagnant > 10:
                break
            stagnant += 1
            # Perturb and retry
            noise = np.random.default_rng(it * 7919).uniform(-0.01, 0.01, n)
            a = np.clip(a + noise, 0.01, None)
            a = a / a.sum()
            continue

        stagnant = 0
        score = compute_obj(a)
        if score < best_score:
            best_score = score
            best_a = a.copy()

    return best_a, best_score


def lbfgsb_objective(x, n):
    """Objective for L-BFGS-B: 2*n*max(conv(normalized(x), normalized(x)))."""
    ax = np.maximum(x, 0.0)
    s = ax.sum()
    if s < 1e-8:
        return 1e10
    ax = ax / s
    b = fast_conv(ax)
    return float(2.0 * n * b.max())


def refine_lbfgsb(a, deadline, max_iter=400):
    """L-BFGS-B local refinement."""
    n = len(a)
    a = np.clip(a.astype(np.float64), 0, None)
    s = a.sum()
    if s < 1e-10:
        return a, np.inf
    a = a / s

    obj_func = partial(lbfgsb_objective, n=n)
    res = minimize(obj_func, a, method='L-BFGS-B',
                   bounds=[(0, None)] * n,
                   options={'maxiter': max_iter, 'max_fun': 10000,
                            'ftol': 1e-16, 'gtol': 1e-12, 'disp': False})

    if res.fun < 1e10:
        a_out = np.clip(res.x, 0, None)
        s_out = a_out.sum()
        if s_out > 1e-10:
            a_out = a_out / s_out
            sc = float(2.0 * n * fast_conv(a_out).max())
            return a_out, sc

    return a, compute_obj(a)


def resize_sequence(a, new_n):
    """Resize sequence using linear interpolation."""
    old_n = len(a)
    if old_n == new_n:
        return a.copy()
    t_old = np.linspace(0, 1, old_n)
    t_new = np.linspace(0, 1, new_n)
    return np.maximum(np.interp(t_new, t_old, a), 0.0)

    # Chirp-like
    if n >= 256:
        starts.append(np.clip(1.0 + 0.4 * np.cos(2 * np.pi * (n * 0.1) * t_arr**2), 0.01, None))
    # Two-bump
    bump = np.exp(-((t_arr - 0.3) / 0.15) ** 2) + np.exp(-((t_arr - 0.7) / 0.15) ** 2)
    starts.append(bump + 0.1)

    # Normalize all
    out = []
    for s in starts:
        s = np.clip(s, 0.01, None)
        if s.sum() > 1e-10:
            out.append(s / s.sum())
    return out


def propose_candidate(seed=42, budget_s=1500, **kwargs):
    """
    Main search: bundle LP + L-BFGS-B, multi-start, multi-length.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + float(budget_s) - 15.0
    best = None
    best_score = float('inf')

    # Try parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        if p.ndim == 1 and p.size > 0 and np.all(np.isfinite(p)):
            parent = p.copy()
            score = float(evaluate_sequence(p.tolist()))
            if np.isfinite(score):
                best = parent.copy()
                best_score = score
                print(f"Parent: J={score:.6f}, n={p.size}", flush=True)
    except (NameError, AttributeError):
        pass

    # Determine lengths
    if parent is not None:
        pn = parent.size
        lengths = sorted(set([max(128, pn // 2), pn, min(8192, pn * 2)]))
    else:
        lengths = [256, 512, 1024]

    # === Phase 1: Bundle method ===
    print("\n=== Phase 1: Bundle LP ===", flush=True)
    for n in lengths:
        if time.monotonic() > deadline:
            break
        print(f"\n  Bundle n={n}", flush=True)
        t0 = time.monotonic()
        starts = make_starts(n, parent, best, rng)
        for a0 in starts:
            if time.monotonic() > deadline:
                break
            a_res, score = bundle_optimize(a0, deadline, max_iters=200, m_max=80)
            s = float(evaluate_sequence(a_res.tolist()))
            if np.isfinite(s) and s < best_score:
                best_score = s
                best = a_res.copy()
                print(f"    NEW BEST: J={s:.6f} n={n}", flush=True)
        print(f"    Done: {time.monotonic()-t0:.1f}s, best={best_score:.6f}", flush=True)

    # === Phase 2: L-BFGS-B refinement ===
    print("\n=== Phase 2: L-BFGS-B ===", flush=True)
    if best is not None:
        bn = best.size
        rlengths = sorted(set([max(128, bn // 2), bn, min(4096, bn * 2)]))
    else:
        rlengths = [512, 1024]

    for n in rlengths:
        if time.monotonic() > deadline:
            break
        print(f"\n  L-BFGS-B n={n}", flush=True)
        t0 = time.monotonic()
        starts = make_starts(n, parent, best, rng)
        for a0 in starts:
            if time.monotonic() > deadline:
                break
            a_res, score = refine_lbfgsb(a0, deadline, max_iter=600)
            s = float(evaluate_sequence(a_res.tolist()))
            if np.isfinite(s) and s < best_score:
                best_score = s
                best = a_res.copy()
                print(f"    NEW BEST: J={s:.6f} n={n}", flush=True)
        print(f"    Done: {time.monotonic()-t0:.1f}s, best={best_score:.6f}", flush=True)

    # === Phase 3: Polish the best ===
    if best is not None and time.monotonic() < deadline:
        print("\n=== Phase 3: Polish ===", flush=True)
        a_res, score = bundle_optimize(best, deadline, max_iters=150, m_max=120)
        s = float(evaluate_sequence(a_res.tolist()))
        if np.isfinite(s) and s < best_score:
            best_score = s
            best = a_res.copy()
            print(f"    Bundle polish: J={s:.6f}", flush=True)

        if time.monotonic() < deadline:
            a_res2, score2 = refine_lbfgsb(best, deadline, max_iter=400)
            s2 = float(evaluate_sequence(a_res2.tolist()))
            if np.isfinite(s2) and s2 < best_score:
                best_score = s2
                best = a_res2.copy()
                print(f"    L-BFGS polish: J={s2:.6f}", flush=True)

    # === Phase 4: Exploration with random restarts ===
    if time.monotonic() < deadline:
        print("\n=== Phase 4: Exploration ===", flush=True)
        n_choices = [256, 512, 1024, 2048]
        if best is not None:
            n_choices.insert(0, best.size)
        for trial in range(40):
            if time.monotonic() > deadline:
                break
            n = int(rng.choice(n_choices))
            # Structured random start
            a0 = rng.uniform(0.0, 0.5, n)
            freq = rng.uniform(1, 12)
            a0 += 0.5 * np.abs(np.sin(2 * np.pi * freq * np.arange(n) / n))
            a0 += 0.2 * np.random.default_rng(trial * 31).uniform(size=n)
            a0 = np.clip(a0, 0.01, None)
            a0 = a0 / a0.sum()

            a_res, score = bundle_optimize(a0, deadline, max_iters=80, m_max=50)
            s = float(evaluate_sequence(a_res.tolist()))
            if np.isfinite(s) and s < best_score:
                best_score = s
                best = a_res.copy()
                print(f"    Trial {trial}: J={s:.6f} n={n}", flush=True)

    # === Phase 5: Final L-BFGS on overall best ===
    if best is not None and time.monotonic() < deadline - 20:
        print("\n=== Phase 5: Final L-BFGS ===", flush=True)
        a_res, score = refine_lbfgsb(best, deadline, max_iter=500)
        s = float(evaluate_sequence(a_res.tolist()))
        if np.isfinite(s) and s < best_score:
            best_score = s
            best = a_res.copy()
            print(f"    Final: J={s:.6f}", flush=True)

    if best is None:
        best = np.ones(256)
        best_score = compute_obj(best)

    print(f"\nFINAL: J={best_score:.6f}, n={best.size}", flush=True)
    return [float(x) for x in best]
```
