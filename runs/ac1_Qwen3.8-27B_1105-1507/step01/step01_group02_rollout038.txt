def project_simplex(v):
    """Project vector v onto {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cv = np.cumsum(u) - 1.0
    css = u * np.arange(1, n + 1) - cv
    rho_candidates = np.where(css > 0)[0]
    if len(rho_candidates) == 0:

    rho = rho_candidates[-1]
    theta = cv[rho] / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def fast_convolve(a):
    """Compute full linear convolution of a with itself using FFT."""
    n = len(a)
    size = 2 * n - 1
    N = 1 << (size).bit_length()
    fa = np.fft.rfft(a, N)
    fc = fa * fa
    result = np.fft.irfft(fc, N)[:size]
    result[result < 0] = 0.0
    return result

    """

    Objective: (1/t) * log(sum(exp(t * c_k))) where c = conv(a,a)

using log-sum-exp smoothed gradient descent with adaptive temperature annealing,
combined with LP-based local optimization and multiple initialization strategies.

    cv = np.cumsum(u) - 1.0

    rho_candidates = np.where(css > 0)[0]

"""
Optimizes a non-negative sequence to minimize 2*n*max(convolve(a,a))/sum(a)^2
using log-sum-exp smoothed gradient descent with adaptive temperature annealing,
Nesterov momentum, LP-based local refinement, and multiple initialization strategies.
"""

import numpy as np
import time
from scipy.optimize import linprog


def project_simplex(v):
    """Project vector v onto {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cv = np.cumsum(u) - 1.0
    css = u * np.arange(1, n + 1) - cv
    rho_candidates = np.where(css > 0)[0]
    if len(rho_candidates) == 0:
        return np.ones(n) / n
    rho = rho_candidates[-1]
    theta = cv[rho] / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def fast_convolve(a):
    """Compute full linear convolution of a with itself using FFT."""
    n = len(a)
    size = 2 * n - 1
    N = 1 << (size).bit_length()
    fa = np.fft.rfft(a, N)
    fc = fa * fa
    result = np.fft.irfft(fc, N)[:size]
    result[result < 0] = 0.0
    return result


def lse_grad(a, t):
    """
    Compute gradient of (1/t)*log(sum(exp(t*c))) where c = conv(a,a).
    Returns (objective_value, gradient, softmax_weights, convolution).
    """
    n = len(a)
    c = fast_convolve(a)
    max_c = c.max()
    exp_tc = np.exp(t * (c - max_c))
    sum_exp = np.sum(exp_tc)
    obj = (max_c + np.log(sum_exp)) / t
    w = exp_tc / sum_exp

    # Gradient: 2 * (w cross_correlate a)[i] for i in 0..n-1
    N = 1 << (3 * n - 2).bit_length()
    W = np.fft.rfft(w, N)
    A = np.fft.rfft(a, N)
    cross = np.fft.irfft(np.conj(W) * A, N)
    grad = 2.0 * cross[:n]

    return obj, grad, w, c

def peak_target_refine(a, rng, n_steps=200):

    then specifically modify contributing elements to reduce it.

        peak_val = c[peak_idx]

        # c[peak_idx] = sum_j a[j]*a[peak_idx - j]
        # The gradient of c[peak_idx] w.r.t. a[i] is 2*a[peak_idx-i]
        lo = max(0, peak_idx - n + 1)
        hi = min(n - 1, peak_idx)
        if lo > hi:
            break

        # Gradient contribution from this peak
        contrib_idx = peak_idx - np.arange(hi + 1 - lo)
        grad_part = 2.0 * a[np.clip(contrib_idx, 0, n - 1)]

        # Only update the elements that contribute
        a[lo:hi + 1] -= lr * grad_part

        # Project to simplex
        a = project_simplex(a)

    return a


def propose_candidate(budget_s=1500, **kwargs):
    """
    Multi-phase search using log-sum-exp smoothed optimization with adaptive
    temperature annealing, momentum, LP-based local refinement, and multiple
    initialization strategies for minimizing 2*n*max(conv(a,a))/sum(a)^2.
    """
    overall_deadline = time.time() + budget_s - 35.0
    rng = np.random.default_rng(42)

    best_score = float('inf')
    best_seq = None

    print(f"Total budget: {budget_s:.0f}s, deadline in {overall_deadline - time.time():.0f}s")

    # ==================== PHASE 1: Parent sequence optimization ====================
    print("\n=== Phase 1: Parent sequence + LSE ===")
    t0 = time.time()
    parent_norm = None
    try:
        parent = np.array(height_sequence_1, dtype=np.float64)
        parent = np.clip(parent, 0.0, 1000.0)
        s = parent.sum()
        if s > 0.01:
            parent_norm = parent / s
            n_p = len(parent_norm)
            print(f"  Parent length: {n_p}, initial score: {evaluate_sequence(parent_norm.tolist()):.6f}")

            # LSE with momentum
            dl1 = min(overall_deadline - 400, time.time() + 200)
            a_opt, mc = lse_optimize(parent_norm, dl1, rng, use_momentum=True)
            sc = evaluate_sequence(a_opt.tolist())
            if np.isfinite(sc) and sc < best_score:
                best_score = sc
                best_seq = a_opt.copy()
                print(f"  After LSE (momentum): {sc:.6f}")

            # LP refinement
            if n_p <= 1200 and time.time() < overall_deadline - 300:
                a_lp = lp_improve(a_opt, rng, time_limit=8.0)
                sc = evaluate_sequence(a_lp.tolist())
                if np.isfinite(sc) and sc < best_score:
                    best_score = sc
                    best_seq = a_lp.copy()
                    print(f"  After LP: {sc:.6f}")

                # LSE again after LP
                if time.time() < overall_deadline - 200:
                    s2 = a_lp.sum()
                    if s2 > 0.01:
                        a_norm2 = a_lp / s2
                        dl2 = min(overall_deadline - 150, time.time() + 100)
                        a_opt2, mc2 = lse_optimize(a_norm2, dl2, rng, use_momentum=True)
                        sc2 = evaluate_sequence(a_opt2.tolist())
                        if np.isfinite(sc2) and sc2 < best_score:
                            best_score = sc2
                            best_seq = a_opt2.copy()
                            print(f"  After LSE round 2: {sc2:.6f}")
    except Exception as e:
        print(f"  Phase 1 error: {e}")
    print(f"  Phase 1 time: {time.time()-t0:.0f}s, best so far: {best_score:.6f}")

    # ==================== PHASE 2: Multi-length LSE exploration ====================
    print("\n=== Phase 2: Multi-length exploration ===")
    t0 = time.time()

    configs = [
        (512, 'alt_proj'),
        (1024, 'alt_proj'),
        (512, 'smooth'),
        (1024, 'smooth'),
        (256, 'alt_proj'),
        (768, 'alt_proj'),
        (2048, 'alt_proj'),
        (1536, 'smooth'),
        (512, 'random'),
        (1024, 'random'),
    ]

    # If we have a parent, also try sibling initializations
    if parent_norm is not None:
        configs.append((len(parent_norm), 'sibling'))

    remaining = overall_deadline - time.time() - 150  # reserve for phase 3
    time_per = max(20.0, remaining / len(configs))

    for n, init_type in configs:
        if time.time() > overall_deadline - 150:
            break

        dl = min(overall_deadline - 150, time.time() + time_per)
        if dl < time.time() + 15:
            continue

        try:
            if init_type == 'alt_proj':
                a = alt_projection_init(n, rng, n_iter=min(120, max(40, n // 8)))
            elif init_type == 'smooth':
                a = smooth_window_init(n, rng)
            elif init_type == 'random':
                a = rng.random(n)
                a = a / a.sum()
            elif init_type == 'sibling':
                a = sibling_init(n, rng, parent_norm)
            else:
                a = np.ones(n) / n

            a_opt, mc = lse_optimize(a, dl, rng, use_momentum=True)
            sc = evaluate_sequence(a_opt.tolist())
            if np.isfinite(sc) and sc < best_score:
                best_score = sc
                best_seq = a_opt.copy()
                print(f"  n={n}, init={init_type}: {sc:.6f} ***")
            else:
                print(f"  n={n}, init={init_type}: {sc:.6f}")

            # Quick LP for smaller sequences
            if n <= 1024 and time.time() < overall_deadline - 160:
                a_lp = lp_improve(a_opt, rng, time_limit=4.0)
                sc = evaluate_sequence(a_lp.tolist())
                if np.isfinite(sc) and sc < best_score:
                    best_score = sc
                    best_seq = a_lp.copy()
                    print(f"    +LP: {sc:.6f} ***")
        except Exception as e:
            print(f"  Error in config ({n}, {init_type}): {e}")

    print(f"  Phase 2 time: {time.time()-t0:.0f}s, best so far: {best_score:.6f}")

    # ==================== PHASE 3: LSE without momentum (different landscape) ====================
    if best_seq is not None and time.time() < overall_deadline - 60:
        print("\n=== Phase 3: LSE without momentum on best ===")
        t0 = time.time()
        a = best_seq.copy()
        s = a.sum()
        if s > 0.01:
            a = a / s
            dl3 = min(overall_deadline - 40, time.time() + 100)
            a_opt, mc = lse_optimize(a, dl3, rng, use_momentum=False)
            sc = evaluate_sequence(a_opt.tolist())
            if np.isfinite(sc) and sc < best_score:
                best_score = sc
                best_seq = a_opt.copy()
                print(f"  After LSE (no momentum): {sc:.6f} ***")
            else:
                print(f"  After LSE (no momentum): {sc:.6f}")
        print(f"  Phase 3 time: {time.time()-t0:.0f}s")

    # ==================== PHASE 4: Peak targeting + final polish ====================
    if best_seq is not None and time.time() < overall_deadline - 15:
        print("\n=== Phase 4: Peak targeting + final polish ===")
        t0 = time.time()
        a = best_seq.copy()
        s = a.sum()
        if s > 0.01:
            a = a / s

            # Peak targeting
            a_pt = peak_target_refine(a, rng, n_steps=300)
            sc = evaluate_sequence(a_pt.tolist())
            if np.isfinite(sc) and sc < best_score:
                best_score = sc
                best_seq = a_pt.copy()
                print(f"  After peak targeting: {sc:.6f} ***")
            else:
                print(f"  After peak targeting: {sc:.6f}")

            # Final LSE polish with remaining time
            if time.time() < overall_deadline - 10:
                s2 = best_seq.sum()
                if s2 > 0.01:
                    a_final = best_seq / s2
                    a_pol, mc = lse_optimize(a_final, overall_deadline - 5, rng, use_momentum=True)
                    sc = evaluate_sequence(a_pol.tolist())
                    if np.isfinite(sc) and sc < best_score:
                        best_score = sc
                        best_seq = a_pol.copy()
                        print(f"  After final polish: {sc:.6f} ***")
                    else:
                        print(f"  After final polish: {sc:.6f}")

        print(f"  Phase 4 time: {time.time()-t0:.0f}s")

    # Fallback
    if best_seq is None:
        print("\nFalling back to uniform sequence")
        best_seq = np.ones(256) / 256.0

    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    final_score = evaluate_sequence(result)
    print(f"\nFINAL: score={final_score:.6f}, length={len(result)}")
    return result
