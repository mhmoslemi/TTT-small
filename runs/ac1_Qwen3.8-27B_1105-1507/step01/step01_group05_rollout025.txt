```python
    """Build matrix A such that A @ b = np.convolve(f, b, mode='full')."""
    n = len(f)
    A = np.zeros((2 * n - 1, n))
    for k in range(2 * n - 1):
        j_start = max(0, k - n + 1)
        j_end = min(n, k + 1)
        for j in range(j_start, j_end):
            A[k, j] = f[k - j]
    return A

    n = len(f)
    conv = np.convolve(f, f)
    M = conv.max()
    return 2.0 * n * M

def multiplicative_update(f, eta, n_iters):
    """Apply multiplicative (exponentiation) updates to reduce peak autocorrelation."""
    n = len(f)
    f = f.copy()
    for _ in range(n_iters):
        conv = np.convolve(f, f)
        k_max = np.argmax(conv)
        # Subgradient at j: 2 * f[k_max - j] (if valid)
        grad = np.zeros(n)
        j_start = max(0, k_max - n + 1)
        j_end = min(n, k_max + 1)

            grad[j] = 2.0 * f[k_max - j]
        f = f * (1.0 - eta * grad)
        f = np.clip(f, 0, None)
        S = f.sum()
        if S < 1e-15:
            break
        f = f / S
    return f
```

```python
"""
LP-based iterative optimization with active-set acceleration, multiplicative updates,
and multi-scale search for minimizing 2*n*max(convolve(a,a))/sum(a)^2.
Uses FFT-accelerated convolutions, sparse LP formulations, and adaptive step sizes.
"""

import numpy as np
from scipy.optimize import linprog
from scipy.sparse import csr_matrix
import time


def fast_convolve(f, g):
    """Compute full convolution using FFT for speed with large sequences."""
    n = len(f)
    m = len(g)
    size = n + m - 1
    fft_size = 1 << (size - 1).bit_length()
    F = np.fft.rfft(f, fft_size)
    G = np.fft.rfft(g, fft_size)
    return np.fft.irfft(F * G, fft_size)[:size]


def compute_score_fast(f):
    """Compute score = 2*n*max(conv)/sum^2 for normalized sequence (sum=1)."""
    n = len(f)
    conv = fast_convolve(f, f)
    return 2.0 * n * conv.max()


def build_conv_matrix_sparse(f):
    """Build sparse matrix A such that A @ b = np.convolve(f, b, mode='full')."""
    n = len(f)
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        j_start = max(0, k - n + 1)
        j_end = min(n, k + 1)
        for j in range(j_start, j_end):
            rows.append(k)
            cols.append(j)
            vals.append(f[k - j])
    A = csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))
    return A


def multiplicative_update(f, eta, n_iters):
    """Apply multiplicative (exponentiation/mirror descent) updates to reduce peak."""
    n = len(f)
    f = f.copy().astype(np.float64)
    for _ in range(n_iters):
        conv = fast_convolve(f, f)
        k_max = int(np.argmax(conv))
        grad = np.zeros(n)
        j_start = max(0, k_max - n + 1)
        j_end = min(n, k_max + 1)
        for j in range(j_start, j_end):
            idx = k_max - j
            if 0 <= idx < n:
                grad[j] = 2.0 * f[idx]
        f = f * np.maximum(0.0, 1.0 - eta * grad)
        S = f.sum()
        if S < 1e-15:
            break
        f /= S
    return f


def multiplicative_update_multi(f, eta, n_iters):
    """Multiplicative update using top-k peaks for better gradients."""
    n = len(f)
    f = f.copy().astype(np.float64)
    k_top = min(5, 2 * n - 1)
    for _ in range(n_iters):
        conv = fast_convolve(f, f)
        top_idx = np.argsort(conv)[-k_top:]
        grad = np.zeros(n)
        for k_max in top_idx:
            k_max = int(k_max)
            j_start = max(0, k_max - n + 1)
            j_end = min(n, k_max + 1)
            for j in range(j_start, j_end):
                idx = k_max - j
                if 0 <= idx < n:
                    grad[j] += 2.0 * f[idx]
        grad /= k_top
        f = f * np.maximum(0.0, 1.0 - eta * grad)
        S = f.sum()
        if S < 1e-15:
            break
        f /= S
    return f


def block_local_search(f, rng, n_iter):
    """Block perturbation local search."""
    n = len(f)
    f = f.copy().astype(np.float64)
    best = f.copy()
    best_score = compute_score_fast(f)
    for _ in range(n_iter):
        fc = f.copy()
        width = max(1, int(rng.integers(1, max(2, n // 16))))
        start = int(rng.integers(0, n - width))
        scale = np.exp(rng.normal(0, 0.4))
        fc[start:start + width] *= scale
        fc = np.clip(fc, 0, None)
        S = fc.sum()
        if S < 1e-10:
            continue
        fc /= S
        score = compute_score_fast(fc)
        if score < best_score:
            best = fc.copy()
            best_score = score
            f = fc
    return best, best_score


def optimize_lp(f, time_budget, t0=0.4, verbose=False):
    """LP-based optimization with backtracking step size and sparse formulation."""
    n = len(f)
    f = f.astype(np.float64).copy()
    f /= f.sum()
    best_f = f.copy()
    best_score = compute_score_fast(f)
    t = t0
    start = time.monotonic()
    stall = 0
    it = 0

    while time.monotonic() - start < time_budget:
        conv = fast_convolve(f, f)
        M = conv.max()

        A = build_conv_matrix_sparse(f)
        c_obj = -np.ones(n)
        b_ub = np.full(2 * n - 1, M)
        bounds = [(0, None)] * n

        try:
            res = linprog(c_obj, A_ub=A, b_ub=b_ub, bounds=bounds,
                          method='highs', options={'presolve': True})
        except Exception:
            break

        if not res.success or res.x is None:
            break

        b = np.clip(res.x, 0, None)
        b_sum = b.sum()
        if b_sum < 1e-15:
            break

        g = b / b_sum

        # Backtracking: try largest step first
        improved = False
        for t_try in [min(t, 1.0), t * 0.8, t * 0.5, t * 0.3, t * 0.1, 0.05, 0.01]:
            if t_try <= 0:
                continue
            f_new = (1.0 - t_try) * f + t_try * g
            f_new = np.clip(f_new, 0, None)
            S = f_new.sum()
            if S < 1e-10:
                continue
            f_new /= S
            new_score = compute_score_fast(f_new)
            if new_score < best_score - 1e-14:
                best_f = f_new.copy()
                best_score = new_score
                f = f_new
                t = t_try
                stall = 0
                improved = True
                break

        if not improved:
            t *= 0.6
            stall += 1
            if stall > 80:
                break

        it += 1
        if verbose and it % 20 == 0:
            elapsed = time.monotonic() - start
            print(f"    it={it}, score={best_score:.8f}, t={t:.4f}, elapsed={elapsed:.1f}s")

    return best_f, best_score


def optimize_lp_aggressive(f, time_budget, verbose=False):
    """More aggressive LP optimization: alternate full and partial steps."""
    n = len(f)
    f = f.astype(np.float64).copy()
    f /= f.sum()
    best_f = f.copy()
    best_score = compute_score_fast(f)
    start = time.monotonic()
    stall = 0
    it = 0
    t = 0.5

    while time.monotonic() - start < time_budget:
        conv = fast_convolve(f, f)
        M = conv.max()

        A = build_conv_matrix_sparse(f)
        c_obj = -np.ones(n)
        b_ub = np.full(2 * n - 1, M)
        bounds = [(0, None)] * n

        try:
            res = linprog(c_obj, A_ub=A, b_ub=b_ub, bounds=bounds,
                          method='highs', options={'presolve': True})
        except Exception:
            break

        if not res.success or res.x is None:
            break

        b = np.clip(res.x, 0, None)
        b_sum = b.sum()
        if b_sum < 1e-15:
            break

        g = b / b_sum

        # Try multiple step sizes adaptively
        best_step_score = best_score
        best_step_f = f
        best_step_t = 0

        candidate_ts = [t, 1.0, 0.8, 0.6, 0.4, 0.2, 0.1]
        candidate_ts = sorted(set(np.clip(candidate_ts, 0.01, 1.0)), reverse=True)

        for t_try in candidate_ts:
            f_new = (1.0 - t_try) * f + t_try * g
            f_new = np.clip(f_new, 0, None)
            S = f_new.sum()
            if S < 1e-10:

            f_new /= S
            new_score = compute_score_fast(f_new)
            if new_score < best_step_score:
                best_step_score = new_score
                best_step_f = f_new
                best_step_t = t_try

        if best_step_t > 0:
            f = best_step_f
            best_f = f.copy()
            best_score = best_step_score
            t = best_step_t
            stall = 0
        else:
            t *= 0.7
            stall += 1
            if stall > 60:
                break

        it += 1
        if verbose and it % 20 == 0:
            elapsed = time.monotonic() - start
            print(f"    it={it}, score={best_score:.8f}, t={t:.4f}, elapsed={elapsed:.1f}s")

    return best_f, best_score
```
