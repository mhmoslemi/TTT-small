    conv = np.convolve(f_norm, f_norm)

    IO (Gerchberg-Saxton) algorithm for spectral flatness.
    Uses power compression (gamma < 1) to flatten the spectrum.
    """

        N *= 2

def gradient_step(a, lr=0.01, top_k=5):
    """
    One subgradient step to reduce the peak of the convolution.
    Uses top_k peaks with weights proportional to their values.
    """
    n = len(a)
    conv = np.convolve(a, a)
    
    # Find top_k peaks
    peak_indices = np.argsort(conv)[::-1][:top_k]
    peak_values = conv[peak_indices]
    weights = peak_values / peak_values.sum()
    
    # Compute combined subgradient
    grad = np.zeros(n)
    for w, k in zip(weights, peak_indices):
        i_start = max(0, k - n + 1)
        i_end = min(n, k + 1)
        j_vals = k - np.arange(i_start, i_end)
        grad[i_start:i_end] += 2.0 * w * a[j_vals]
    
    # Update
    a_new = a - lr * grad
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s < 1e-15:
        return a
    return a_new / s

Hybrid optimization for minimizing 2*n*max(conv(a,a))/sum(a)^2.
Combines subgradient descent with momentum, LSE smoothing, LP refinement,
and local search across multiple sequence lengths.
"""

import numpy as np
from scipy.optimize import linprog
import time


def build_conv_matrix(f):
    """Build matrix A such that A @ b = np.convolve(f, b, mode='full')."""
    n = len(f)
    A = np.zeros((2 * n - 1, n))
    for k in range(2 * n - 1):
        j_start = max(0, k - n + 1)
        j_end = min(n, k + 1)
        for j in range(j_start, j_end):
            A[k, j] = f[k - j]
    return A


def compute_score_fast(f):
    """Compute score = 2*n*max(conv)/sum^2."""
    n = len(f)
    S = f.sum()
    if S < 1e-15:
        return np.inf
    f_norm = f / S
    conv = np.convolve(f_norm, f_norm)
    M = conv.max()
    return 2.0 * n * M
