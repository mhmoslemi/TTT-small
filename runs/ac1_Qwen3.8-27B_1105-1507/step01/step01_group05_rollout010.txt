def solve_direction_lp(f):
    """
    Solve: min t s.t. conv(f,g)_k <= t for all k, sum(g)=1, g>=0.
    Returns (g, t) where g is the optimal direction.
    """
    n = len(f)
    A = build_conv_matrix_for_f(f)
    
    # Variables: [g_0, ..., g_{n-1}, t]
    c = np.zeros(n + 1)
    c[-1] = 1.0  # minimize t
    
    # Inequality: A@g - t <= 0
    A_ub = np.hstack([A, -np.ones((2 * n - 1, 1))])
    b_ub = np.zeros(2 * n - 1)
    
    # Equality: sum(g) = 1
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])
    
    # Bounds
    bounds = [(0, None)] * n + [(0, None)]
    
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                     bounds=bounds, method='highs', options={'presolve': True, 'time_limit': 30})
    except Exception as e:
        return None, None
    
    if not res.success:
        return None, None
    
    g = np.clip(res.x[:n], 0, None)
    t_val = res.x[n]
    g_sum = g.sum()
    if g_sum < 1e-10:
        return None, None
    g = g / g_sum  # ensure sum=1
    return g, t_val

def subgradient_polish(f, n_iter=5000, verbose=False):
    """Polish with projected subgradient descent."""
    n = len(f)
    f = f.copy()
    f = f / f.sum()
    
    best_f = f.copy()
    best_score = compute_score_normalized(f)
    
    v = np.zeros(n)
    beta = 0.9
    alpha0 = 0.1
    
    for it in range(n_iter):
        conv = compute_conv_fft(f)
        M = conv.max()
        
        # Find active set
        threshold = 0.85 * M
        active = np.where(conv >= threshold)[0]
        
        # Compute subgradient
        grad = np.zeros(n)
        for k in active:
            j_start = max(0, k - (n - 1))
            j_end = min(n - 1, k)
            j_range = np.arange(j_start, j_end + 1)
            grad[j_range] += 2.0 * f[k - j_range]
        grad /= max(1, len(active))
        
        # Adaptive step size
        alpha = alpha0 / (1.0 + 0.005 * it)
        
        # Update
        v = beta * v - alpha * grad
        f_new = project_simplex(f + v)
        
        score = compute_score_normalized(f_new)
        if score < best_score:
            best_score = score
            best_f = f_new.copy()
        
        f = f_new
    
    return best_f, best_score


def coordinate_descent(f, n_sweeps=10, verbose=False):
    """Coordinate descent: optimize one element at a time."""
    n = len(f)
    f = f.copy()
    f = f / f.sum()
    best_score = compute_score_normalized(f)
    
    for sweep in range(n_sweeps):
        improved = False
        for j in range(n):
            best_val = f[j]
            best_sc = best_score
            # Try a few values
            for scale in [0.5, 0.7, 0.85, 1.15, 1.3, 1.5]:
                f[j] = best_val * scale
                s = f.sum()
                if s < 1e-10:
                    f[j] = best_val
                    continue
                f_norm = f / s
                sc = compute_score_normalized(f_norm)
                if sc < best_sc - 1e-12:
                    best_sc = sc
                    best_val = best_val * scale
                    improved = True
            f[j] = best_val
        
        if not improved and sweep > 2:
            break
        if verbose:
            print(f"    CD sweep {sweep}: score={best_score:.6f}")
    
    return f, best_score

def build_conv_matrix_for_f(f):
    """Build matrix A (2n-1 x n) such that A @ g = convolve(f, g)."""
    n = len(f)
    A = np.zeros((2 * n - 1, n))

def compute_score_normalized(a):
    """Score = 2*n*max(conv(a,a)) assuming sum(a)=1."""
    n = len(a)
    conv = np.fft.irfft(np.fft.rfft(a, 2 * n - 1)**2, 2 * n - 1)
    return 2.0 * n * conv.max()


def compute_conv_fft(a):
    """Compute full autocorrelation conv(a,a) using FFT."""
    n = len(a)
    size = 2 * n - 1
    F = np.fft.rfft(a, size)
    return np.fft.irfft(F * F, size)


def project_simplex(v):
    """Project v onto the probability simplex {x: sum(x)=1, x>=0}."""
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = (u * np.arange(1, n + 1)) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = int(np.max(np.where(cond)[0]))
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0)

        if best_alpha is not None and best_new_score < best_score - 1e-12:
            f_new = (1.0 - best_alpha) * f + best_alpha * g
            f_new = np.clip(f_new, 0, None)
            f_new = f_new / f_new.sum()
            f = f_new
            best_f = f_new.copy()
            best_score = best_new_score
            stall = 0
            if verbose and it % 5 == 0:
                elapsed = time.monotonic() - start
                print(f"    LP it={it:3d}, score={best_score:.6f}, alpha={best_alpha:.2f}, "
                      f"t_lp={t_lp if t_lp else 'N/A'}, elapsed={elapsed:.0f}s")
        else:
            stall += 1
            if stall % 20 == 0:
                # Random perturbation to escape
                f_pert = f + 0.03 * rng.standard_normal(n)
                f_pert = np.clip(f_pert, 0, None)
                if f_pert.sum() > 1e-10:
                    f_pert = f_pert / f_pert.sum()
                    sc_pert = compute_score_normalized(f_pert)
                    if sc_pert < best_score:
                        f = f_pert
                        best_f = f.copy()
                        best_score = sc_pert
                        stall = 0
                        if verbose:
                            print(f"    Perturbed at it={it}, new best={best_score:.6f}")
            if stall > 60 and time.monotonic() - start > time_budget * 0.8:
                break

        it += 1

    return best_f, best_score


def subgradient_polish(f, n_iter=5000, verbose=False):
    """Projected subgradient descent with momentum."""
    n = len(f)
    f = f.copy().astype(np.float64)
    f = f / f.sum()

    best_f = f.copy()
    best_score = compute_score_normalized(f)

    v = np.zeros(n)
    beta = 0.9
    alpha0 = 0.15
    threshold_ratio = 0.80

    for it in range(n_iter):
        conv = compute_conv_fft(f)
        M = conv.max()

        if M < 1e-15:
            break

        # Active set
        active = np.where(conv >= threshold_ratio * M)[0]

        # Subgradient (average over active indices)
        grad = np.zeros(n)
        for k in active:
            j_start = max(0, k - (n - 1))
            j_end = min(n - 1, k)
            j_idx = np.arange(j_start, j_end + 1)
            grad[j_idx] += 2.0 * f[k - j_idx]
        grad /= max(1, len(active))

        # Adaptive step size
        alpha = alpha0 / (1.0 + 0.003 * it)

        # Momentum update
        v = beta * v - alpha * grad
        f_new = project_simplex(f + v)

        sc = compute_score_normalized(f_new)
        if sc < best_score:
            best_score = sc
            best_f = f_new.copy()

        f = f_new

        if verbose and it % 1000 == 0:
            print(f"      SG it={it:5d}, score={best_score:.6f}")

    return best_f, best_score


def coordinate_descent(f, n_sweeps=8, verbose=False):
    """Coordinate descent: try scaling one element at a time."""
    n = len(f)
    f = f.copy().astype(np.float64)
    f = f / f.sum()
    best_score = compute_score_normalized(f)
    scales = [0.5, 0.7, 0.85, 1.15, 1.3, 1.5]

    for sweep in range(n_sweeps):
        improved_any = False
        for j in range(n):
            orig = f[j]
            best_val = orig
            best_sc = best_score
            for s in scales:
                f[j] = orig * s
                tot = f.sum()
                if tot < 1e-10:
                    f[j] = orig
                    continue
                sc = compute_score_normalized(f / tot)
                if sc < best_sc - 1e-12:
                    best_sc = sc
                    best_val = orig * s
                    improved_any = True
            f[j] = best_val
            best_score = min(best_score, best_sc)

        if verbose:
            print(f"    CD sweep {sweep}: score={best_score:.6f}")
        if not improved_any and sweep >= 2:
            break

    f = f / f.sum()
    return f, best_score


def anneal_polish(f, n_iter=3000, rng=None, verbose=False):
    """Simulated annealing style random perturbation."""
    if rng is None:
        rng = np.random.default_rng(42)
    n = len(f)
    f = f.copy().astype(np.float64)
    f = f / f.sum()
    best_f = f.copy()
    best_score = compute_score_normalized(f)
    temp = 0.01

    for it in range(n_iter):
        # Random block perturbation
        width = max(1, int(rng.integers(1, max(2, n // 32))))
        start = int(rng.integers(0, n - width))
        scale = np.exp(rng.normal(0, 0.3))
        f_new = f.copy()
        f_new[start:start + width] *= scale
        f_new = np.clip(f_new, 0, None)
        s = f_new.sum()
        if s < 1e-10:
            continue
        f_new = f_new / s
        sc = compute_score_normalized(f_new)

        if sc < best_score:
            best_score = sc
            best_f = f_new.copy()
            f = f_new
        else:
            # Accept with probability exp(-(sc - best_score)/temp)
            if rng.random() < np.exp(-(sc - best_score) / max(temp, 1e-10)):
                f = f_new

        temp *= 0.9995

    return best_f, best_score


def propose_candidate(seed=42, budget_s=1500, **kwargs):
    """
    Main search: multi-start LP optimization + subgradient + CD + annealing.
    Tries multiple n values and starting structures.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + budget_s - 20.0

    best_overall = None
    best_score_overall = np.inf

    # Get parent
    parent = None
    try:
        parent = np.array(height_sequence_1, dtype=np.float64)
        if parent.ndim != 1 or len(parent) == 0 or not np.isfinite(parent).all():
            parent = None
    except Exception:
        parent = None

    if parent is not None:
        parent = np.clip(parent, 0, None)
        if parent.sum() > 1e-10:
            parent = parent / parent.sum()
            p_score = compute_score_normalized(parent)
            print(f"Parent: n={len(parent)}, score={p_score:.6f}")
            best_overall = parent.copy()
            best_score_overall = p_score
        else:
            parent = None

    # Build configs: (name, sequence)
    configs = []

    # Parent as primary
    if parent is not None:
        configs.append(('parent', parent.copy()))
        # Perturbed parent
        pp = parent * (1.0 + 0.15 * rng.standard_normal(len(parent)))
        pp = np.clip(pp, 0, None)
        if pp.sum() > 1e-10:
            configs.append(('parent_pert', pp / pp.sum()))

    # Other starting points
    for n in [128, 256, 384, 512]:
        # Uniform with noise
        a = np.ones(n) * (1.0 + 0.08 * rng.standard_normal(n))
        a = np.clip(a, 0.01, None)
        a = a / a.sum()
        configs.append((f'uniform_{n}', a))

        # Cosine window
        a = np.sin(np.pi * (np.arange(n) + 0.5) / (2.0 * n)) ** 2
        a = a / a.sum()
        configs.append((f'cosine_{n}', a))

    print(f"\nConfigs ({len(configs)}):")
    for name, a in configs:
        print(f"  {name}: n={len(a)}, initial score={compute_score_normalized(a):.4f}")
    print(f"Target: 1.5027 | Current best: {best_score_overall:.6f}")

    # Time allocation: give more time to parent-based configs
    total_time = budget_s - 40
    # Weight parent configs higher
    weights = []
    for name, a in configs:
        if 'parent' in name:
            weights.append(2.0)
        elif len(a) == 256:
            weights.append(1.5)
        elif len(a) <= 384:
            weights.append(1.0)
        else:
            weights.append(0.7)
    total_w = sum(weights)

    idx = 0
    for w_idx, (name, f0) in enumerate(configs):
        if time.monotonic() > deadline:
            break

        remaining = deadline - time.monotonic()
        if remaining < 20:
            break

        # Time for this config
        t_alloc = total_time * weights[w_idx] / total_w
        t_alloc = min(t_alloc, remaining * 0.8)
        t_alloc = max(t_alloc, 15.0)

        # Split: LP 60%, SG 20%, CD+anneal 20%
        t_lp = t_alloc * 0.6
        t_sg = t_alloc * 0.2
        t_cd = t_alloc * 0.2

        print(f"\n[{idx+1}/{len(configs)}] {name} (n={len(f0)}): "
              f"LP={t_lp:.0f}s, SG={t_sg:.0f}s, CD={t_cd:.0f}s")

        # Phase 1: LP optimization
        f_opt, score = lp_optimize(f0, t_lp, verbose=True, rng=rng)
        print(f"  After LP: {score:.6f}")

        # Phase 2: Subgradient polish
        if t_sg > 5 and np.isfinite(score):
            n_iter_sg = min(30000, max(1000, int(t_sg * 1500)))
            f_sg, score_sg = subgradient_polish(f_opt, n_iter=n_iter_sg, verbose=False)
            if score_sg < score:
                f_opt = f_sg
                score = score_sg
                print(f"  After SG: {score:.6f}")

        # Phase 3: Coordinate descent + annealing
        if t_cd > 3:
            f_cd, score_cd = coordinate_descent(f_opt, n_sweeps=4, verbose=False)
            if score_cd < score:
                f_opt = f_cd
                score = score_cd
                print(f"  After CD: {score:.6f}")

            # Annealing
            n_iter_an = min(5000, int(t_cd * 500))
            f_an, score_an = anneal_polish(f_opt, n_iter=n_iter_an, rng=rng)
            if score_an < score:
                f_opt = f_an
                score = score_an
                print(f"  After Anneal: {score:.6f}")

        # Update best
        if np.isfinite(score) and score < best_score_overall:
            best_score_overall = score
            best_overall = f_opt.copy()
            print(f"  ** NEW BEST: {score:.6f} **")

        idx += 1

    # Final polish on the best sequence found
    if best_overall is not None:
        remaining = deadline - time.monotonic()
        if remaining > 20:
            print(f"\n=== Final polish ({remaining:.0f}s remaining) ===")
            n_best = len(best_overall)

            # One more LP round
            t_lp_final = min(remaining * 0.4, 120.0)
            if t_lp_final > 10:
                f_lp, sc_lp = lp_optimize(best_overall, t_lp_final, verbose=True, rng=rng)
                if sc_lp < best_score_overall:
                    best_overall = f_lp
                    best_score_overall = sc_lp
                    print(f"  After final LP: {sc_lp:.6f}")

            # Subgradient
            remaining2 = deadline - time.monotonic()
            if remaining2 > 10:
                n_iter_final = min(50000, int(remaining2 * 0.5 * 1000))
                f_sg, sc_sg = subgradient_polish(best_overall, n_iter=n_iter_final, verbose=True)
                if sc_sg < best_score_overall:
                    best_overall = f_sg
                    best_score_overall = sc_sg
                    print(f"  After final SG: {sc_sg:.6f}")

            # CD
            remaining3 = deadline - time.monotonic()
            if remaining3 > 5:
                f_cd, sc_cd = coordinate_descent(best_overall, n_sweeps=6, verbose=True)
                if sc_cd < best_score_overall:
                    best_overall = f_cd
                    best_score_overall = sc_cd
                    print(f"  After final CD: {sc_cd:.6f}")

    # Verify and return
    if best_overall is not None:
        best_overall = np.clip(best_overall, 0, 1000.0)
        final_list = best_overall.tolist()
        final_score = evaluate_sequence(final_list)
        print(f"\n{'='*50}")
        print(f"FINAL: score={final_score:.6f}, n={len(final_list)}")
        print(f"Target was 1.5027, previous was 1.515592")
        print(f"{'='*50}")
        return final_list

    # Fallback
    print("Fallback to uniform 256")
    return [1.0] * 256
