```
    try:

       c = np.zeros(n+1, dtype=np.float64)

       b_ub = np.zeros(2*n-1)
       A_eq = np.zeros((1,n+1)); A_eq[0,:n]=1
       b_eq = np.array([1.0])

       res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq, bounds=bounds, method='highs', options={'presolve': True})

           g = np.asarray(res.x[:n], dtype=np.float64)
           g = normalize_sequence(g, 1e-16)
           if g is not None:
              return g
    except Exception:
       pass
    return None
```

```
def upsample_sequence(parent, n):
    if parent is None: return None
    p = normalize_sequence(parent,0.0)
    if p is None: return None
    m = int(p.size)
    if m == n: return p.copy()
    if m == 1:

    x_old = np.linspace(0.0,1.0,m)
    x_new = np.linspace(0.0,1.0,n)
    v = np.interp(x_new, x_old, p)
```

```python
"""
Search for low-score non-negative step-function coefficients.

Objective:
    score(a) = 2 * n * max_k conv(a,a)[k] / (sum(a)**2)

The scale is invariant, so all internal candidates are normalized to sum 1.
For non-negative a, conv(a,a) is real and non-negative.

This solver combines:
  - FFT-based exact convolution/correlation,
  - a smooth multiplicative active-peak optimizer (weights proportional to
    (c_k / max c)^q, plus a top-active fallback),
  - a linear-programming step that finds g minimizing max_k (f*g)[k]
    for a fixed f, then blends f with g,
  - multiple lengths and multiple randomized/structured starts.
"""

import time

import numpy as np

try:
    from scipy.optimize import linprog
    HAVE_SCIPY = True
except Exception:
    HAVE_SCIPY = False


def normalize_sequence(a, floor=0.0):
    """Return a non-negative finite sequence normalized to sum 1, or None."""
    try:
        b = np.asarray(a, dtype=np.float64)
    except Exception:
        return None
    if b.size == 0:
        return None
    if not np.all(np.isfinite(b)):
        return None
    b = np.clip(b, 0.0, 1000.0)
    if floor > 0.0:
        b = np.maximum(b, floor)
    s = float(np.sum(b))
    if not np.isfinite(s) or s <= 1e-300:
        return None
    return b / s


def fft_len(n):
    """Power-of-two FFT length large enough for full linear convolution."""
    n = int(n)
    m = 2 * n - 1
    if m <= 1:
        return 1
    return 1 << (m - 1).bit_length()


def convolve_fft(a, L=None):
    """FFT full linear convolution a*a. Returns (c, fa)."""
    n = int(a.size)
    if L is None:
        L = fft_len(n)
    fa = np.fft.rfft(a, L)
    c = np.fft.irfft(fa * fa, L, n=L)
    c = c[:2 * n - 1]
    if c.size:
        c = np.maximum(c, 0.0)
    return c, fa


def correlate_p_a(fa, p, n, L):
    """
    h_i = sum_j a_j p_{i+j}, where a was used for fa.
    This is the convolution-correlation gradient direction for active weights p.
    """
    fp = np.fft.rfft(p, L)
    h = np.fft.irfft(fa * np.conj(fp), L, n=L)
    h = h[:int(n)]
    h = np.maximum(h, 0.0)
    return h


def score_normalized(a):
    """Normalize a and compute score using FFT."""
    b = normalize_sequence(a, 0.0)
    if b is None:
        return None, float("inf")
    c, _ = convolve_fft(b)
    M = float(np.max(c)) if c.size else float("inf")
    return b, 2.0 * float(b.size) * M


def try_exp_step(a, inc, step, L, floor=1e-18):
    """Multiplicative step a <- a * exp(step*inc), normalized and scored."""
    e = np.clip(step * inc, -30.0, 30.0)
    b = a * np.exp(e)
    b = normalize_sequence(b, floor)
    if b is None:
        return None, float("inf"), None, None
    c, fa = convolve_fft(b, L)
    M = float(np.max(c)) if c.size else float("inf")
    return b, M, c, fa


def smooth_optimize(a, max_time, seed, q_start=1.0, q_end=32.0, eta_start=1.0, max_iters=300000):
    """
    Local optimizer.

    Direction:
        c = conv(a,a), M = max(c)
        w = (c/M)^q          (or uniform over top-active when stuck)
        h = correlation(w,a)
        inc = dot(a,h) - h  (weighted mean zero)
        a <- a * exp(eta * inc), normalized.

    The step is accepted only if exact max(c) decreases.
    """
    a = normalize_sequence(a, 1e-18)
    if a is None:
        return None, float("inf")

    n = int(a.size)
    L = fft_len(n)
    c_cur, fa_cur = convolve_fft(a, L)
    M_cur = float(np.max(c_cur)) if c_cur.size else float("inf")

    best = a.copy()
    bestM = M_cur
    bestScore = 2.0 * float(n) * bestM

    eta = float(eta_start)
    q = float(q_start)
    stuck = 0
    it = 0

    t0 = time.monotonic()
    deadline = t0 + max(0.0, float(max_time))
    rng = np.random.default_rng(seed)

    while time.monotonic() < deadline and it < int(max_iters):
        prog = min(1.0, float(it) / 3000.0)
        target_q = float(q_start) + (float(q_end) - float(q_start)) * prog
        if stuck > 100:
            target_q = max(0.8, 0.5 * target_q)
        q = max(0.8, target_q)

        cmax = M_cur if M_cur > 1e-300 else 1.0

        # Construct active weights.
        if stuck > 50:
            # More focused top-active weights when the smoother q-update stalls.
            frac = max(0.02, 0.25 / (1.0 + stuck / 100.0))
            m = max(1, int(frac * c_cur.size))
            if m >= c_cur.size:
                w = c_cur / cmax
            else:
                w = np.zeros_like(c_cur)
                top = np.argpartition(c_cur, -m)[-m:]
                w[top] = 1.0
        else:
            ratio = c_cur / cmax
            ratio = np.clip(ratio, 0.0, 1.0)
            logs = q * np.log(ratio + 1e-300)
            logs = np.clip(logs, -700.0, 0.0)
            w = np.exp(logs)

        wsum = float(np.sum(w))
        if not np.all(np.isfinite(w)) or wsum <= 0.0:
            w = np.ones_like(c_cur)

        h = correlate_p_a(fa_cur, w, n, L)
        r = float(np.dot(w, c_cur))

        if not np.isfinite(r) or r <= 1e-300:
            w = np.ones_like(c_cur)
            h = correlate_p_a(fa_cur, w, n, L)
            r = float(np.dot(w, c_cur))
            if r <= 1e-300:
                break

        inc = r - h
        scale = float(np.max(np.abs(inc)))
        if scale <= 1e-300:
            break
        inc = np.clip(inc / scale, -1.0, 1.0)

        accepted = False
        for mult in (1.0, 0.5, 0.25):
            step = eta * float(mult)
            if step <= 1e-8:
                continue
            b, M_try, c_try, fa_try = try_exp_step(a, inc, step, L, 1e-18)
            if b is None:
                continue
            if M_try < M_cur - 1e-15:
                a = b
                c_cur = c_try
                fa_cur = fa_try
                M_cur = M_try

                if M_cur < bestM - 1e-15:
                    best = a.copy()
                    bestM = M_cur
                    bestScore = 2.0 * float(n) * bestM
                    stuck = 0
                    eta = min(3.0, eta * 1.05)
                else:
                    stuck = max(0, stuck - 2)
                    eta = min(3.0, eta * 1.01)

                accepted = True
                break

        if not accepted:
            stuck += 1
            eta = max(0.005, eta * 0.7)

            if stuck == 60:
                # Small noise to leave a flat plateau.
                noise = rng.normal(0.0, 0.04, n)
                b = a * np.exp(noise)
                b = normalize_sequence(b, 1e-18)
                if b is not None:
                    c_n, fa_n = convolve_fft(b, L)
                    M_n = float(np.max(c_n))
                    if M_n < M_cur * 1.05:
                        a = b
                        c_cur = c_n
                        fa_cur = fa_n
                        M_cur = M_n

            if stuck >= 180:
                # Restart from current best with larger noise.
                sigma = 0.05 + 0.002 * float(min(500, stuck - 180))
                if sigma > 0.6:
                    sigma = 0.6
                b = best * np.exp(rng.normal(0.0, sigma, n))
                b = normalize_sequence(b, 1e-18)
                if b is not None:
                    c_n, fa_n = convolve_fft(b, L)
                    M_n = float(np.max(c_n))
                    if M_n < bestM * 1.25:
                        a = b
                        c_cur = c_n
                        fa_cur = fa_n
                        M_cur = M_n
                        stuck = 0
                        eta = 1.0
                        q = max(1.0, target_q * 0.5)
                    else:
                        a = best.copy()
                        c_cur, fa_cur = convolve_fft(a, L)
                        M_cur = bestM
                        stuck = 0
                else:
                    a = best.copy()
                    c_cur, fa_cur = convolve_fft(a, L)
                    M_cur = bestM
                    stuck = 0

        it += 1

    fin = normalize_sequence(best, 0.0)
    if fin is None:
        fin = best
    cfin, _ = convolve_fft(fin, L)
    Mfin = float(np.max(cfin)) if cfin.size else bestM
    return fin, 2.0 * float(n) * Mfin


def upsample_sequence(parent, n):
    """Upsample a parent sequence to length n by linear interpolation."""
    if parent is None:
        return None
    p = normalize_sequence(parent, 0.0)
    if p is None:
        return None
    m = int(p.size)
    if m == int(n):
        return p.copy()
    if m == 1:
        return np.full(int(n), 1.0 / float(n), dtype=np.float64)

    x_old = np.linspace(0.0, 1.0, m)
    x_new = np.linspace(0.0, 1.0, int(n))
    v = np.interp(x_new, x_old, p)
    return normalize_sequence(v, 1e-12)


def random_candidate(n, seed, parent, kind=None):
    """Generate a diverse non-negative candidate sequence of length n."""
    n = int(n)
    rng = np.random.default_rng(seed)
    if kind is None:
        k = int(rng.integers(0, 10))
    else:
        k = int(kind) % 10

    t = np.linspace(0.0, 1.0, n)

    if k == 0:
        val = np.ones(n, dtype=np.float64)

    elif k == 1:
        x = rng.normal(0.0, 1.0, n)
        w = int(rng.integers(1, max(2, n // 64 + 1)))
        if w > 1:
            val = np.convolve(x, np.ones(w) / float(w), mode="same")
        else:
            val = x
        val = np.exp(np.clip(val, -4.0, 4.0))

    elif k == 2:
        freq = int(rng.integers(1, max(2, n // 8 + 1)))
        phase = rng.uniform(0.0, 2.0 * np.pi)
        rho = rng.uniform(0.1, 0.8)
        val = 1.0 + rho * np.cos(2.0 * np.pi * float(freq) * t + phase)

    elif k == 3:
        gamma = [0.4, 0.7, 1.0, 2.0, 4.0][int(rng.integers(0, 5))]
        edge = 0.5 + 0.5 * np.cos(np.pi * t)  # 1 at ends, 0 in center
        val = 0.05 + 0.95 * (edge ** gamma)
        if rng.random() < 0.4:
            val = 1.0 / np.maximum(0.2, val)  # occasionally invert to peak center

    elif k == 4:
        width = float(rng.uniform(0.08, 0.35))
        eps = float(rng.uniform(0.01, 0.25))
        val = np.where((t < width) | (t > 1.0 - width), 1.0, eps)
        if n > 16:
            w = max(1, n // 64)
            val = np.convolve(val, np.ones(w) / float(w), mode="same")

    elif k == 5:
        p = upsample_sequence(parent, n)
        if p is not None:
            val = p * np.exp(rng.normal(0.0, float(rng.uniform(0.05, 0.35)), n))
        else:
            val = rng.beta(0.5, 0.5, n)

    elif k == 6:
        # Smooth low-pass random shape.
        L = fft_len(n)
        X = rng.normal(0.0, 1.0, L // 2 + 1) + 1j * rng.normal(0.0, 1.0, L // 2 + 1)
        freqs = np.fft.rfftfreq(L) * float(L)
        bw = max(1.0, float(n) / 12.0)
        mask = np.exp(-(freqs / bw) ** 2)
        X = X * mask
        X[0] = 0.0
        x = np.fft.irfft(X, L, n=L)[:n]
        val = np.exp(np.clip(0.9 * x, -4.0, 4.0))

    elif k == 7:
        ab = float(rng.uniform(0.2, 3.0))
        bb = float(rng.uniform(0.2, 3.0))
        val = rng.beta(ab, bb, n)

    elif k == 8:
        alpha = float(rng.uniform(0.3, 3.0))
        if rng.random() < 0.5:
            x = t
        else:
            x = 1.0 - t
        val = 0.3 * np.exp(-alpha * x) + 0.7 * np.exp(-alpha * (1.0 - x))

    else:
        # Mild endpoint singularities (kept finite by offset).
        p = float(rng.uniform(0.0, 0.45))
        q = float(rng.uniform(0.0, 0.45))
        val = (0.002 + t) ** (-p) * (0.002 + (1.0 - t)) ** (-q)

    val = np.asarray(val, dtype=np.float64)
    val = np.maximum(val, 1e-8)
    val = normalize_sequence(val, 1e-12)
    if val is None:
        val = np.ones(n, dtype=np.float64) / float(n)
    return val


def lp_direction(f, max_n=1200, seed=0):
    """
    LP best-response direction.

    For fixed f, solve:
        min t
        s.t. f*g - t <= 0 for all convolution lags,
             sum(g) = 1,
             g >= 0.

    The returned g has low max cross-convolution with f and is often a good
    blend direction for reducing max(g*g).
    """
    if not HAVE_SCIPY:
        return None

    f = normalize_sequence(f, 0.0)
    if f is None:
        return None

    n = int(f.size)
    if n <= 1 or n > int(max_n):
        return None

    try:
        rows = 2 * n - 1
        A_ub = np.zeros((rows, n + 1), dtype=np.float64)
        A_ub[:, n] = -1.0

        # A_ub[k, i] = f[k - i] when 0 <= k-i < n.
        for i in range(n):
            A_ub[i:i + n, i] = f

        cvec = np.zeros(n + 1, dtype=np.float64)
        cvec[n] = 1.0

        rng = np.random.default_rng(seed + 987654321)
        cvec[:n] = 1e-9 * rng.normal(0.0, 1.0, n)

        b_ub = np.zeros(rows, dtype=np.float64)

        A_eq = np.zeros((1, n + 1), dtype=np.float64)
        A_eq[0, :n] = 1.0
        b_eq = np.array([1.0], dtype=np.float64)

        bounds = [(0.0, None)] * (n + 1)

        res = linprog(
            cvec,
            A_ub=A_ub,
            b_ub=b_ub,
            A_eq=A_eq,
            b_eq=b_eq,
            bounds=bounds,
            method="highs",
            options={"presolve": True}
        )

        if res is not None and res.success:
            g = np.asarray(res.x[:n], dtype=np.float64)
            g = normalize_sequence(g, 1e-16)
            if g is not None:
                return g
    except Exception:
        pass

    return None


def lp_optimize(a, max_time, seed, max_n=1200):
    """
    Iterate LP cross-convolution directions and blend with current sequence.
    """
    a = normalize_sequence(a, 1e-18)
    if a is None:
        return None, float("inf")

    n = int(a.size)
    if n <= 1:
        return a, float("inf")

    if n > int(max_n):
        c, _ = convolve_fft(a)
        M = float(np.max(c)) if c.size else float("inf")
        return a, 2.0 * float(n) * M

    L = fft_len(n)
    c_best, _ = convolve_fft(a, L)
    bestM = float(np.max(c_best)) if c_best.size else float("inf")
    best = a.copy()
    bestScore = 2.0 * float(n) * bestM

    t0 = time.monotonic()
    deadline = t0 + max(0.0, float(max_time))
    it = 0
    stall = 0

    while time.monotonic() < deadline - 10.0:
        g = lp_direction(best, max_n, seed + 1234567 + it)
        if g is None:
            break

        local_best = best.copy()
        local_M = bestM

        for tv in (1.0, 0.8, 0.6, 0.4, 0.25, 0.12, 0.05):
            cand = (1.0 - tv) * best + tv * g
            cand = normalize_sequence(cand, 1e-18)
            if cand is None:
                continue
            c_try, _ = convolve_fft(cand, L)
            M_try = float(np.max(c_try)) if c_try.size else float("inf")
            if M_try < local_M - 1e-15:
                local_best = cand
                local_M = M_try

        if local_M < bestM - 1e-15:
            best = local_best
            bestM = local_M
            bestScore = 2.0 * float(n) * bestM
            stall = 0
        else:
            stall += 1
            if stall >= 2:
                break

        it += 1

    return best, bestScore


def explore_length(n, budget, seed, parent, deadline):
    """
    Explore a fixed length n with several starts and local optimizations.
    """
    n = int(n)
    if n <= 1:
        return None, float("inf")

    budget = float(budget)
    if budget <= 5.0:
        return None, float("inf")

    cands = []

    if parent is not None:
        u = upsample_sequence(parent, n)
        if u is not None:
            b, s = score_normalized(u)
            if b is not None and np.isfinite(s):
                cands.append((s, b))

    max_cand = 3 if n <= 1024 else 2
    kinds = [1, 2, 3, 5, 6, 9, 4, 7, 8]
    kidx = 0
    while len(cands) < max_cand and kidx < len(kinds) and time.monotonic() < deadline - 5.0:
        c = random_candidate(n, seed + 101 * kidx, parent, kinds[kidx])
        if c is not None:
            b, s = score_normalized(c)
            if b is not None and np.isfinite(s):
                cands.append((s, b))
        kidx += 1

    if not cands:
        b = np.ones(n, dtype=np.float64) / float(n)
        cands.append((2.0, b))

    # Manually order by score without lambda.
    ordered = list(cands)
    selected = []
    while ordered:
        best_i = 0
        best_s = ordered[0][0]
        for i in range(1, len(ordered)):
            if ordered[i][0] < best_s:
                best_s = ordered[i][0]
                best_i = i
        selected.append(ordered.pop(best_i))

    best_seq = selected[0][1]
    best_score = selected[0][0]
    best_n = int(best_seq.size)

    if len(selected) == 1:
        fracs = (0.85, 0.15)
    elif len(selected) == 2:
        fracs = (0.65, 0.35)
    else:
        fracs = (0.5, 0.3, 0.2)

    for idx, item in enumerate(selected):
        if idx >= len(fracs):
            break

        rem = deadline - time.monotonic()
        if rem < 20.0:
            break

        alloc = min(budget * float(fracs[idx]), rem * 0.9)
        if item[0] > best_score * 1.05 and idx > 0:
            alloc = min(alloc, budget * 0.15)
        alloc = max(5.0, alloc)

        if time.monotonic() + alloc > deadline:
            alloc = max(1.0, deadline - time.monotonic() - 5.0)
        if alloc < 3.0:
            break

        c1, s1 = smooth_optimize(item[1], alloc, seed + 1000 + idx)
        if c1 is not None and np.isfinite(s1) and s1 < best_score:
            best_score = float(s1)
            best_seq = c1.copy()
            best_n = int(c1.size)

        rem = deadline - time.monotonic()
        if idx == 0 and best_n <= 1200 and HAVE_SCIPY and rem > 50.0:
            lp_alloc = min(budget * 0.1, 40.0, rem * 0.7)
            if lp_alloc > 15.0:
                c2, s2 = lp_optimize(best_seq, lp_alloc, seed + 2000 + idx, max_n=1200)
                if c2 is not None and np.isfinite(s2) and s2 < best_score:
                    best_score = float(s2)
                    best_seq = c2.copy()
                    best_n = int(c2.size)

    return best_seq, best_score


def propose_candidate(seed=42, budget_s=1500.0, **kwargs):
    """
    Entry point. Searches for a low-score sequence and returns a list[float].
    """
    start_time = time.monotonic()
    try:
        budget = float(budget_s)
    except Exception:
        budget = 1500.0
    if not np.isfinite(budget) or budget <= 0.0:
        budget = 1500.0

    deadline = start_time + max(10.0, budget) - 20.0

    parent = globals().get("height_sequence_1", None)
    parent_arr = normalize_sequence(parent, 0.0) if parent is not None else None

    best_seq = None
    best_score = float("inf")

    if parent_arr is not None:
        b, s = score_normalized(parent_arr)
        if b is not None and np.isfinite(s):
            best_seq = b
            best_score = float(s)
            print(f"parent length={b.size} score={best_score:.6f}", flush=True)

    if best_seq is None:
        b = np.ones(512, dtype=np.float64) / 512.0
        best_seq = b
        best_score = 2.0
        print(f"initial fallback uniform length={b.size} score={best_score:.6f}", flush=True)

    seed_cur = int(seed)
    polished_len = -1

    # Polish the parent sequence first, if present and time allows.
    rem = deadline - time.monotonic()
    if parent_arr is not None and rem > 200.0 and int(best_seq.size) <= 2048:
        alloc = min(260.0, rem - 150.0)
        if alloc > 20.0:
            c, s = smooth_optimize(best_seq, alloc, seed_cur)
            if c is not None and np.isfinite(s) and s < best_score:
                best_seq = c.copy()
                best_score = float(s)
                print(f"parent polished score={best_score:.6f}", flush=True)
            polished_len = int(best_seq.size)

        rem = deadline - time.monotonic()
        if HAVE_SCIPY and int(best_seq.size) <= 1200 and rem > 140.0:
            lp_alloc = min(50.0, rem - 100.0)
            if lp_alloc > 20.0:
                c, s = lp_optimize(best_seq, lp_alloc, seed_cur + 777, max_n=1200)
                if c is not None and np.isfinite(s) and s < best_score:
                    best_seq = c.copy()
                    best_score = float(s)
                    print(f"parent LP polished score={best_score:.6f}", flush=True)

    # Build length plan.
    length_plan = []
    if parent_arr is None:
        length_plan = [(512, 400.0), (1024, 350.0), (2048, 250.0), (4096, 200.0)]
    else:
        plen = int(best_seq.size)
        if polished_len == -1 and plen <= 1024:
            length_plan.append((plen, 200.0))
        if plen != 512 and plen != 1024:
            length_plan.append((512, 160.0))
        length_plan.append((1024, 320.0))
        length_plan.append((2048, 300.0))
        length_plan.append((4096, 250.0))

    # De-duplicate while preserving order.
    seen = []
    plan = []
    for n, alloc in length_plan:
        ni = int(n)
        if ni in seen or ni < 2:
            continue
        seen.append(ni)
        plan.append((ni, float(alloc)))

    # Skip the parent length if it was already polished.
    if polished_len > 0:
        plan2 = []
        for n, alloc in plan:
            if n != polished_len:
                plan2.append((n, alloc))
        plan = plan2

    for n, alloc in plan:
        rem = deadline - time.monotonic()
        if rem < 80.0:
            break

        a_alloc = min(float(alloc), rem - 60.0)
        if a_alloc < 40.0:
            break

        c, s = explore_length(n, a_alloc, seed_cur, parent_arr, deadline)
        if c is not None and np.isfinite(s) and s < best_score:
            best_score = float(s)
            best_seq = c.copy()
            print(f"improved length={c.size} score={best_score:.6f}", flush=True)
        else:
            if np.isfinite(s):
                print(f"length={n} explored best_here={s:.6f}", flush=True)
            else:
                print(f"length={n} explored failed", flush=True)

        seed_cur += 101237

    # Final global polish.
    rem = deadline - time.monotonic()
    if best_seq is not None and rem > 60.0:
        n = int(best_seq.size)
        if n <= 1024:
            alloc = min(rem - 30.0, 250.0)
        elif n <= 2048:
            alloc = min(rem - 30.0, 200.0)
        else:
            alloc = min(rem - 30.0, 160.0)

        if alloc > 20.0:
            c, s = smooth_optimize(best_seq, alloc, seed_cur + 991)
            if c is not None and np.isfinite(s) and s < best_score:
                best_seq = c.copy()
                best_score = float(s)
                print(f"final smooth score={best_score:.6f}", flush=True)

        rem = deadline - time.monotonic()
        if HAVE_SCIPY and int(best_seq.size) <= 1200 and rem > 80.0:
            lpal = min(rem - 40.0, 60.0)
            if lpal > 20.0:
                c, s = lp_optimize(best_seq, lpal, seed_cur + 997, max_n=1200)
                if c is not None and np.isfinite(s) and s < best_score:
                    best_seq = c.copy()
                    best_score = float(s)
                    print(f"final LP score={best_score:.6f}", flush=True)

    # Finalize.
    final = normalize_sequence(best_seq, 0.0)
    if final is None:
        final = np.ones(512, dtype=np.float64) / 512.0

    if "evaluate_sequence" in globals():
        try:
            official = float(evaluate_sequence([float(x) for x in final]))
            if not np.isfinite(official) and parent_arr is not None:
                fp = normalize_sequence(parent_arr, 0.0)
                if fp is not None:
                    official2 = float(evaluate_sequence([float(x) for x in fp]))
                    if np.isfinite(official2):
                        final = fp
                        official = official2
            if np.isfinite(official):
                best_score = float(official)
        except Exception:
            pass

    final = np.clip(np.asarray(final, dtype=np.float64), 0.0, 1000.0)
    out = [float(float(x)) for x in final]
    print(f"FINAL score={best_score:.6f} length={len(out)}", flush=True)
    return out
```
