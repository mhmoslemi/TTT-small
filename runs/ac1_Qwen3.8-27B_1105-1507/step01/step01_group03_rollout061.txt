Hybrid optimization for minimum peak linear autocorrelation of non-negative
sequences. Combines LP-based direction finding with subgradient/exponential-
weights refinement, FFT-accelerated convolution, and structured restarts.
"""

import time
import numpy as np
from scipy.optimize import linprog
from scipy.sparse import csr_matrix


def normalize_seq(arr):
    x = np.asarray(arr, dtype=np.float64).ravel()
    if x.size == 0:
        return None
    x = np.where(np.isfinite(x), x, 0.0)
    x = np.maximum(x, 0.0)
    s = x.sum()
    if s < 1e-300:
        return None
    return x / s


def score_norm(x, n):
    c = np.convolve(x, x)
    M = float(c.max())
    if not np.isfinite(M):
        return float('inf')
    return 2.0 * n * M


def fft_linear_conv(a, b):
    n = len(a) + len(b) - 1
    size = 1 << (n - 1).bit_length()
    return np.fft.irfft(np.fft.rfft(a, size) * np.fft.rfft(b, size), size)[:n]


def make_template(n):
    m = 2 * n - 1
    k64 = np.arange(m, dtype=np.int64)
    # For row k, column m: A[k,m] = f[k-m] if 0 <= k-m <= n-1
    # Valid m range: max(0, k-(n-1)) <= m <= min(n-1, k)
    js = np.maximum(0, k64 - (n - 1))
    je = np.minimum(n - 1, k64)
    counts = je - js + 1
    total = int(counts.sum())
    
    cum = np.cumsum(counts)
    starts = np.repeat(cum - counts, counts)
    local = np.arange(total, dtype=np.int64) - starts
    
    row = np.repeat(k64, counts).astype(np.int32)
    col = (js[row.astype(np.int64)] + local).astype(np.int32)
    # offset = k - m = k64[col_idx] - col... 
    # Actually, for entry at row k, col m: offset = k - m
    # k = row[i], m = col[i], offset = row[i] - col[i]
    offsets = (row.astype(np.int64) - col.astype(np.int64)).astype(np.int32)
    
    data = np.zeros(total, dtype=np.float64)
    A = csr_matrix((data, (row, col)), shape=(m, n))
    
    c_obj = -np.ones(n, dtype=np.float64)
    bounds = [(0.0, None)] * n
    return A, offsets, c_obj, bounds, m


def solve_lp(template, f, M, n):
    A, offsets, c_obj, bounds, m = template
    # Fill A with f[offsets]
    A.data[:] = f[offsets]
    # Scale: (f*g)_k <= M  =>  A @ g <= M
    # A has f values, so A @ g = (f*g) directly. No scaling needed.
    b = np.full(m, M, dtype=np.float64)
    res = linprog(c_obj, A_ub=A, b_ub=b, bounds=bounds, method='highs')
    return res


def subgradient_step(f, eta, n, top_k=5):
    """Exponential weights subgradient step."""
    c = np.convolve(f, f)
    M = c.max()
    if M <= 0:
        return f, M
    
    # Find top-k active constraints
    sorted_idx = np.argsort(c)[::-1]
    active = sorted_idx[:top_k]
    
    # Compute subgradient as weighted average of active rows
    # Row k of the conv matrix: s[m] = f[k-m] if 0 <= k-m <= n-1
    s = np.zeros(n, dtype=np.float64)
    weights = c[active]
    w_sum = weights.sum()
    if w_sum <= 0:
        return f, M
    weights = weights / w_sum
    
    for idx, k in enumerate(active):
        k = int(k)
        m_start = max(0, k - (n - 1))
        m_end = min(n - 1, k)
        # s[m] += f[k-m] for m in [m_start, m_end]
        offsets = k - np.arange(m_start, m_end + 1)
        s[m_start:m_end+1] += weights[idx] * f[offsets]

    f_new = f * np.exp(-eta * s)
    s_new = f_new.sum()
    if s_new < 1e-300:
        return f, M
    f_new /= s_new
    return f_new, M

"""
Hybrid LP + subgradient optimization for minimum peak linear autocorrelation
of non-negative sequences. Combines LP direction finding with exponential-
weights subgradient refinement, FFT-accelerated evaluation, structured
restarts, and multi-scale search to escape local minima.

from scipy.optimize import linprog
from scipy.sparse import csr_matrix


def normalize_seq(arr):
    """Normalize to sum 1, non-negative, finite."""
    x = np.asarray(arr, dtype=np.float64).ravel()
    if x.size == 0:
        return None
    x = np.where(np.isfinite(x), x, 0.0)
    x = np.maximum(x, 0.0)
    s = float(x.sum())
    if s < 1e-300:
        return None
    return x / s


def score_norm(x, n):
    """Score for normalized (sum=1) non-negative sequence."""
    c = np.convolve(x, x)
    M = float(c.max())
    if not np.isfinite(M):
        return float('inf')
    return 2.0 * n * M


def fft_conv(a, b):
    """Linear convolution via FFT."""
    n_out = len(a) + len(b) - 1
    size = 1 << (n_out - 1).bit_length()
    return np.fft.irfft(np.fft.rfft(a, size) * np.fft.rfft(b, size), size)[:n_out]


def make_template(n):
    """Build sparse LP template for convolution constraints."""
    m = 2 * n - 1
    k64 = np.arange(m, dtype=np.int64)
    js = np.maximum(0, k64 - (n - 1)).astype(np.int64)
    je = np.minimum(n - 1, k64).astype(np.int64)
    counts = je - js + 1
    total = int(counts.sum())

    cum = np.cumsum(counts)
    starts = np.repeat(cum - counts, counts)
    local = np.arange(total, dtype=np.int64) - starts

    row = np.repeat(k64, counts).astype(np.int32)
    col_base = js[row.astype(np.int64)]
    col = (col_base + local).astype(np.int32)
    offsets = (row.astype(np.int64) - col.astype(np.int64)).astype(np.int32)

    data = np.zeros(total, dtype=np.float64)
    A = csr_matrix((data, (row, col)), shape=(m, n))

    c_obj = -np.ones(n, dtype=np.float64)
    bounds = [(0.0, None)] * n
    return A, offsets, c_obj, bounds, m


def solve_lp(template, f, M, n, warm_x=None):
    """Solve the LP direction problem."""
    A, offsets, c_obj, bounds, m = template
    A.data[:] = f[offsets]
    b = np.full(m, M, dtype=np.float64)
    options = {'presolve': True, 'dual_feasibility_tolerance': 1e-9}
    if warm_x is not None:
        options['presolve'] = False
    res = linprog(c_obj, A_ub=A, b_ub=b, bounds=bounds, method='highs',
                  options=options)
    return res


def subgradient_step(f, eta, n, top_k=5, rng=None):
    """Exponential-weights subgradient step using top-k active constraints."""
    c = np.convolve(f, f)
    M = float(c.max())
    if M <= 0 or not np.isfinite(M):
        return f, M

    sorted_idx = np.argsort(c)[::-1]
    active = sorted_idx[:top_k]
    weights = c[active].copy()
    w_sum = weights.sum()
    if w_sum <= 0:
        return f, M
    weights = weights / w_sum

    s = np.zeros(n, dtype=np.float64)
    for idx, k in enumerate(active):
        k = int(k)
        m_start = max(0, k - (n - 1))
        m_end = min(n - 1, k)
        if m_end < m_start:
            continue
        offsets = k - np.arange(m_start, m_end + 1)
        s[m_start:m_end + 1] += weights[idx] * f[offsets]

    f_new = f * np.exp(-eta * s)
    s_new = f_new.sum()
    if s_new < 1e-300:
        return f, M
    f_new /= s_new
    return f_new, M

def interpolate_seq(x, n):
    """Resample normalized sequence to length n."""
    x = normalize_seq(x)
    if x is None or n <= 0:
        return None
    m = x.size
    if m == n:
        return x
    if m == 1:
        return np.ones(n, dtype=np.float64) / n
    old = np.linspace(0.0, 1.0, m)
    new = np.linspace(0.0, 1.0, n)
    out = np.interp(new, old, x)
    out = np.maximum(out, 0.0)
    return normalize_seq(out)

            if s is not None and s.size == n:
                if not any(np.linalg.norm(s - c) < 1e-4 for c in clean_starts):
                    clean_starts.append(s)
        starts = clean_starts[:6]
        if not starts:
            starts = [np.ones(n, dtype=np.float64) / n]

        # Allocate time across starts (more time for first starters)
        m = len(starts)
        if m <= 1:
            weights = [1.0]
        elif m == 2:
            weights = [0.6, 0.4]
        elif m == 3:
            weights = [0.45, 0.35, 0.20]
        elif m == 4:
            weights = [0.35, 0.30, 0.20, 0.15]
        else:
            weights = [0.30, 0.25, 0.20, 0.15, 0.07, 0.03][:m]
            # Normalize weights
            wsum = sum(weights)
            weights = [w / wsum for w in weights]

        for idx, init in enumerate(starts):
            rem = deadline - time.monotonic()
            if rem < 5.0:
                break
            tb = alloc * weights[idx]
            tb = min(tb, rem - 2.0)
            if tb < 3.0:
                continue

            print(f"  Start {idx+1}/{m}, budget={tb:.1f}s", flush=True)
            f_res, s_res = optimize_n(n, init, tb, rng, template)

            if np.isfinite(s_res) and s_res < best_score:
                best_score = float(s_res)
                best_f = np.asarray(f_res, dtype=np.float64).copy()
                print(f"  *** New best: n={best_f.size}, score={best_score:.7f} ***", flush=True)

        try:
            del template
        except Exception:
            pass

    # Final validation
    if best_f is None:
        best_f = np.ones(64, dtype=np.float64) / 64.0

    seq = [float(x) for x in best_f]
    try:
        final_score = float(evaluate_sequence(seq))
    except Exception:
        final_score = float('inf')

    if not np.isfinite(final_score):
        # Fallback
        seq = [1.0 / 64.0] * 64
        try:
            final_score = float(evaluate_sequence(seq))
        except Exception:
            final_score = float('inf')

    print(f"\nFINAL: n={len(seq)}, local_best={best_score:.7f}, evaluate={final_score:.7f}", flush=True)
    return seq
