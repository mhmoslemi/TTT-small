import numpy as np

def fast_convolve(a):
    """Full linear convolution of a with itself via FFT."""
    n = len(a)
    size = 2 * n - 1
    N = 1 << (size - 1).bit_length()
    fa = np.fft.rfft(a, N)
    fc = fa * fa
    result = np.fft.irfft(fc, N)[:size]
    result[result < 0] = 0.0
    return result


def project_simplex(v):
    """Project vector v onto {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v.copy()
    u = np.sort(v)[::-1]
    cv = np.cumsum(u) - 1.0
    rho_candidates = np.where(u * np.arange(1, n + 1) > cv)[0]
    if len(rho_candidates) == 0:
        return np.ones(n) / n
    rho = rho_candidates[-1]
    theta = cv[rho] / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def compute_smooth_grad(a, p):
    """
    Compute gradient of smoothed max: grad[i] = 2 * sum_j a[j] * p[i+j].
    p is a probability vector of length 2n-1 (softmax of convolution).
    Uses FFT for O(n log n) complexity.
    """
    n = len(a)
    r = p[::-1]
    size = 3 * n - 2
    N_fft = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, N_fft)
    R = np.fft.rfft(r, N_fft)
    full_conv = np.fft.irfft(A * R, N_fft)[:size]
    # grad[i] = 2 * full_conv[2n-2-i] for i=0..n-1
    # That's full_conv indices 2n-2 down to n-1, reversed
    grad = 2.0 * full_conv[n - 1:2 * n - 1][::-1]
    return grad


def compute_subgrad_topk(a, c, k=10):
    """Compute subgradient averaged over top-k peaks of c."""
    n = len(a)
    size = 2 * n - 1
    # Get top k indices
    if k >= size:
        top_idx = np.arange(size)
    else:
        top_idx = np.argpartition(c, -k)[-k:]
    weights = np.zeros(size)
    top_vals = c[top_idx]
    # Softmax over top-k
    top_vals_shifted = top_vals - top_vals.max()
    exp_vals = np.exp(top_vals_shifted)
    weights[top_idx] = exp_vals / exp_vals.sum()
    return compute_smooth_grad(a, weights)


def lp_improve(a, P, timeout=5.0):
    """
    LP step: maximize sum(b) s.t. conv(a,b) <= P, b >= 0.
    Returns rescaled b or None.
    """
    n = len(a)
    if n > 3000:
        return None
    # Build convolution matrix
    A = np.zeros((2 * n - 1, n), dtype=np.float64)
    for k in range(2 * n - 1):
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        A[k, lo:hi + 1] = a[k - hi:k - lo + 1][::-1]
    c_obj = -np.ones(n)
    b_ub = np.full(2 * n - 1, P)
    try:
        res = linprog(c_obj, A_ub=A, b_ub=b_ub,
                      bounds=[(0.0, None)] * n, method='highs',
                      options={'time_limit': timeout})
        if res.success and res.x is not None:
            b = np.maximum(res.x, 0.0)
            b_sum = b.sum()
            if b_sum > 1e-8:
                a_sum = a.sum()
                b = b * (a_sum / b_sum)
                return b
    except Exception:
        pass
    return None

    """
    Smoothed gradient descent with Nesterov momentum.
    Minimizes max(conv(a,a)) subject to sum(a)=1, a>=0.
    Uses log-sum-exp smoothing with annealed temperature.
    """
    n = len(a)
    a = project_simplex(a.copy().astype(np.float64))

    c = fast_convolve(a)
    best_peak = c.max()
    best_a = a.copy()

    # Nesterov momentum state
    y = a.copy()
    momentum = 0.0
    momentum_target = 0.85
    momentum_decay = 0.9995

    # Step size
    step = 1.0 / np.sqrt(n)
    step_decay = 0.999
    step_min = 1e-5 / np.sqrt(n)

    # Temperature for smoothing (start large, anneal)
    tau = 0.05 * best_peak
    tau_min = 1e-6 * best_peak
    tau_decay = 0.999
    tau_anneal_start = 5000

    it = 0
    stall = 0
    fval = best_peak
    fval_prev = best_peak

    while time.time() < deadline and it < max_iters:
        it += 1

        # Update momentum (Nesterov)
        if it % 500 == 0:
            momentum = momentum * momentum_decay + momentum_target * (1 - momentum_decay)
        beta = momentum
        t_k = (1.0 + np.sqrt(1.0 + 4.0 * (t_k_prev if it > 0 else 0.0) ** 2)) / 2.0
        # Simplified: use constant beta for stability
        t_k_prev = t_k if it > 1 else 1.0

        # Compute smoothing weights
        c = fast_convolve(y)
        c_shifted = c - c.max()
        exp_c = np.exp(c_shifted / tau)
        p = exp_c / exp_c.sum()

        # Gradient at y
        grad_y = compute_smooth_grad(y, p)

        # Nesterov: compute gradient at y + beta*(y - a)
        y_new = y + beta * (y - a)
        y_new = project_simplex(y_new) if beta > 0.5 else y_new
        if beta > 0.3:
            c_y = fast_convolve(y_new)
            c_y_shifted = c_y - c_y.max()
            exp_y = np.exp(c_y_shifted / tau)
            p_y = exp_y / exp_y.sum()
            grad = compute_smooth_grad(y_new, p_y)
        else:
            grad = grad_y

        # Gradient step
        step = max(step * step_decay, step_min)
        a_new = y - step * grad
        a = project_simplex(a_new)

        # Update y for next iteration
        y = a + beta * (a - a + (a - a))  # placeholder
        # Actually for Nesterov: y_{k+1} = a_{k+1} + (t_{k+1}-1)/(t_{k+2})*(a_{k+1} - a_k)
        # Simplified heavy-ball: y = a - beta * (a - a_prev) -> y = (1+beta)*a - beta*a_prev
        if it == 1:
            a_prev = a.copy()
        else:
            a_prev_copy = a_prev.copy()
            a_prev = a.copy()
        y = (1.0 + beta) * a - beta * a_prev_copy

        # Evaluate
        c = fast_convolve(a)
        fval = c.max()

        if fval < best_peak * (1.0 - 1e-10):
            best_peak = fval
            best_a = a.copy()
            stall = 0
        else:
            stall += 1

        # Anneal tau
        if it > tau_anneal_start:
            tau = max(tau * tau_decay, tau_min)

        # Adaptive step: if stalling, reduce step and add noise
        if stall > 500:
            step = max(step * 0.5, step_min)
            stall = 0
            if time.time() < deadline - 20:
                noise = rng.normal(0, 0.03 / np.sqrt(n), n)
                a = project_simplex(a + noise)
                y = a.copy()

        # LP step
        if use_lp and n <= 2048 and it % 500 == 0 and time.time() < deadline - 30:
            b = lp_improve(a, best_peak, timeout=4.0)
            if b is not None:
                t_mix = 0.3 + 0.3 * rng.random()
                a = project_simplex((1.0 - t_mix) * a + t_mix * b)
                y = a.copy()
                c_check = fast_convolve(a)
                if c_check.max() < best_peak:
                    best_peak = c_check.max()
                    best_a = a.copy()

        # Periodic perturbation
        if it % 5000 == 0 and time.time() < deadline - 60:
            temp = 0.01 / np.sqrt(it / 1000.0 + 1)
            a = project_simplex(a + rng.normal(0, temp / np.sqrt(n), n))
            y = a.copy()

    return best_a


def optimize_subgrad_hybrid(a, deadline, rng, use_lp=True, max_iters=300000):
    """
    Hybrid optimizer: switches between smoothed gradient and top-k subgradient.
    Includes LP steps and adaptive restarts.
    """
    n = len(a)
    a = project_simplex(a.copy().astype(np.float64))

    c = fast_convolve(a)
    best_peak = c.max()
    best_a = a.copy()

    # State
    step = 2.0 / np.sqrt(n)
    step_decay = 0.9995
    step_min = 1e-6 / np.sqrt(n)
    momentum = 0.0
    v = np.zeros(n)  # momentum velocity

    it = 0
    stall = 0
    smooth_iters = 0
    subgrad_iters = 0

    while time.time() < deadline and it < max_iters:
        it += 1

        # Choose mode: first 20000 iters smoothed, then alternate
        if smooth_iters < 20000:
            mode = 'smooth'
        else:
            mode = 'smooth' if (it % 10 % 7 == 0) else 'subgrad'

        if mode == 'smooth':
            smooth_iters += 1
            c = fast_convolve(a)
            # Anneal temperature
            frac = min(1.0, it / 30000.0)
            tau = c.max() * (0.1 * (1.0 - frac) + 1e-5 * frac)
            c_shifted = c - c.max()
            exp_c = np.exp(c_shifted / tau)
            p = exp_c / exp_c.sum()
            grad = compute_smooth_grad(a, p)
        else:
            subgrad_iters += 1
            c = fast_convolve(a)
            grad = compute_subgrad_topk(a, c, k=min(20, 2 * n - 1))

        # Momentum update (polyak heavy-ball)
        if it % 2000 == 0:
            momentum = min(momentum + 0.001, 0.9)
        elif it < 1000:
            momentum = min(momentum + 0.002, 0.5)

        v = momentum * v - step * grad
        a_new = a + v
        a = project_simplex(a_new)

        # Step decay
        step = max(step * step_decay, step_min)

        # Evaluate
        c = fast_convolve(a)
        fval = c.max()

        if fval < best_peak * (1.0 - 1e-10):
            best_peak = fval
            best_a = a.copy()
            stall = 0
        else:
            stall += 1

        # Reset momentum on stall
        if stall > 1000:
            v = np.zeros(n)
            stall = 0
            step = max(step * 0.8, step_min)
            if time.time() < deadline - 30:
                # Perturb to escape
                noise_scale = 0.05 * (1.0 + 0.1 * rng.random())
                a = project_simplex(a + rng.normal(0, noise_scale / np.sqrt(n), n))

        # LP step
        if use_lp and n <= 2048 and it % 300 == 0 and time.time() < deadline - 40:
            b = lp_improve(a, best_peak, timeout=3.0)
            if b is not None:
                t_mix = 0.2 + 0.4 * rng.random()
                a = project_simplex((1.0 - t_mix) * a + t_mix * b)
                v = np.zeros(n)
                c_check = fast_convolve(a)
                if c_check.max() < best_peak:
                    best_peak = c_check.max()
                    best_a = a.copy()

    return best_a


def try_config(n, init_type, rng, deadline, use_lp=True):
    """Initialize and optimize a sequence of given length."""
    if init_type == 'chirp':
        a = init_chirp(n, rng)
    elif init_type == 'cosine':
        a = init_cosine_window(n)
    elif init_type == 'flat_spec':
        a = init_flat_spectrum(n, rng, n_iter=80)
    elif init_type == 'rand_spec':
        a = init_random_spectral(n, rng)
    elif init_type == 'parent':
        a = init_parent(n, rng)
    elif init_type == 'uniform':
        a = np.ones(n) / n
    else:
        a = rng.random(n) + 0.5
        a = a / a.sum()

    # First pass: smoothed optimizer
    a_opt = optimize_smoothed(a, deadline, rng, use_lp=use_lp, max_iters=100000)

    # Second pass: hybrid on the result
    if time.time() < deadline - 30:
        a_opt2 = optimize_subgrad_hybrid(a_opt, deadline, rng, use_lp=use_lp, max_iters=100000)
        c1 = fast_convolve(a_opt).max()
        c2 = fast_convolve(a_opt2).max()
        if c2 < c1:
            a_opt = a_opt2

    score = evaluate_sequence(a_opt.tolist())
    return a_opt, score


def propose_candidate(budget_s=1500):
    """
    Multi-phase search: tries multiple lengths and initializations using
    smoothed gradient descent with momentum and LP-based improvements.
    """
    overall_deadline = time.time() + budget_s - 40.0
    rng = np.random.default_rng(12345)

    best_score = float('inf')
    best_seq = None

    # Priority configs: (length, init_type)
    configs = [
        (512, 'parent'),
        (512, 'chirp'),
        (1024, 'flat_spec'),
        (512, 'flat_spec'),
        (1024, 'chirp'),
        (768, 'flat_spec'),
        (512, 'rand_spec'),
        (1024, 'rand_spec'),
        (256, 'chirp'),
        (1536, 'flat_spec'),
        (768, 'chirp'),
        (1024, 'parent'),
    ]

    num_configs = len(configs)
    # Time allocation: top 3 get 35%, rest split remaining
    first3_time_frac = 0.35
    remaining_frac = 1.0 - first3_time_frac
    per_first = first3_time_frac / 3.0
    per_rest = remaining_frac / max(1, num_configs - 3)

    for i, (n, init_type) in enumerate(configs):
        if time.time() > overall_deadline - 60:
            break

        if i < 3:
            time_alloc = per_first * (budget_s - 40)
        else:
            time_alloc = per_rest * (budget_s - 40)

        config_deadline = min(overall_deadline - 30, time.time() + time_alloc)
        if config_deadline < time.time() + 20:
            continue

        use_lp = (n <= 2048)
        try:
            a_opt, score = try_config(n, init_type, rng, config_deadline, use_lp=use_lp)
        except Exception as e:
            print(f"  Error in config ({n}, {init_type}): {e}")
            continue

        rem = overall_deadline - time.time()
        print(f"n={n}, init={init_type}, score={score:.6f}, time_remaining={rem:.0f}s")

        if np.isfinite(score) and score < best_score:
            best_score = score
            best_seq = a_opt.copy()
            print(f"  >>> New best: {best_score:.6f} (n={n}, init={init_type})")

    # Final polish on best sequence
    if best_seq is not None and time.time() < overall_deadline:
        polish_deadline = overall_deadline
        n_best = len(best_seq)
        # Use the hybrid optimizer for final polish
        a_polished = optimize_subgrad_hybrid(
            best_seq, polish_deadline, rng,
            use_lp=(n_best <= 2048), max_iters=200000
        )
        score = evaluate_sequence(a_polished.tolist())
        print(f"After final polish: score={score:.6f}")
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_seq = a_polished

        # One more smoothed pass
        if time.time() < overall_deadline - 10:
            a_p2 = optimize_smoothed(
                best_seq, overall_deadline, rng,
                use_lp=(n_best <= 2048), max_iters=100000
            )
            score2 = evaluate_sequence(a_p2.tolist())
            print(f"After second polish: score={score2:.6f}")
            if np.isfinite(score2) and score2 < best_score:
                best_score = score2
                best_seq = a_p2

    # Fallback
    if best_seq is None:
        print("Falling back to uniform sequence")
        best_seq = np.ones(256) / 256

    result = [float(x) for x in best_seq]
    # Ensure all values are valid
    result = [max(0.0, min(1000.0, x)) for x in result]
    final_score = evaluate_sequence(result)
    print(f"FINAL: score={final_score:.6f}, length={len(result)}")
    return result
