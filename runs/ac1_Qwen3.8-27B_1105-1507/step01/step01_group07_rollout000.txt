    n = len(a)
    conv = np.convolve(a, a)

def lp_optimize(n, a_init, max_iter=200, verbose=False):
    """LP-based alternating optimization with direct update."""
    a = a_init.copy()
    s = np.sum(a)
    if s < 1e-10:
        a = np.ones(n) / n
    else:
        a = a / s
    
    best_a = a.copy()
    best_obj = compute_obj(best_a)
    
    for it in range(max_iter):
        g, M = solve_lp_for_g(a)
        if g is None:

        g_sum = np.sum(g)
        if g_sum < 1e-10:

            best_obj = obj_g

        s = np.sum(a_new)

"""
Multi-phase optimization minimizing 2*n*max(conv(a,a))/(sum(a))^2.
Combines LP-based alternating optimization, subgradient polishing,
and coordinate refinement with diverse initializations and multiple sequence lengths.
"""

import numpy as np
from scipy.optimize import linprog
import time


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    conv = np.convolve(a, a)
    max_b = float(np.max(conv))
    sum_a = float(np.sum(a))
    if sum_a < 0.01:
        return np.inf
    return 2.0 * n * max_b / (sum_a * sum_a)


def build_A(a):
    """Build constraint matrix: A[k,j] = a[k-j] if 0<=k-j<n else 0."""
    n = len(a)
    A = np.zeros((2 * n - 1, n), dtype=np.float64)
    for k in range(2 * n - 1):
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        A[k, j_start:j_end + 1] = a[k - j_end:k - j_start + 1][::-1]
    return A


def solve_lp_g(a, M_relax=1.0):
    """Solve LP: max sum(g) s.t. (a*g)[k] <= M*M_relax, g >= 0."""
    n = len(a)
    conv = np.convolve(a, a)
    M = float(np.max(conv)) * M_relax
    A = build_A(a)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
    if res.status == 0:
        return res.x
    return None


def lp_alternate(n, a_init, max_iter=300, verbose=False, relax=1.0):
    """LP alternating optimization with EMA smoothing."""
    a = a_init.copy().astype(np.float64)
    s = np.sum(a)
    if s < 1e-10:
        a = np.ones(n) / n
    else:
        a = a / s

    best_a = a.copy()
    best_obj = compute_obj(best_a)
    ema = a.copy()

    for it in range(max_iter):
        g = solve_lp_g(a, M_relax=relax)
        if g is None:
            break
        g_sum = np.sum(g)
        if g_sum < 1e-10:
            break
        g = g / g_sum

        # Update EMA
        ema = 0.7 * ema + 0.3 * g

        # Check objective of g and ema
        obj_g = compute_obj(g)
        obj_ema = compute_obj(ema)

        if obj_g < best_obj:
            best_a = g.copy()
            best_obj = obj_g
        if obj_ema < best_obj:
            best_a = ema.copy()
            best_obj = obj_ema

        # Next iterate: blend of current and g
        t = 0.6
        a_new = (1.0 - t) * a + t * g
        a_new = np.maximum(a_new, 0.0)
        s = np.sum(a_new)
        if s < 1e-10:
            a = g
        else:
            a = a_new / s

        if verbose and (it % 50 == 0):
            print(f"    LP it={it}: best_obj={best_obj:.6f}")

    return best_a, best_obj


def subgradient_polish(a, n_steps=2000, verbose=False):
    """Subgradient method for polishing. Minimize max(conv(a,a)) s.t. sum(a)=1, a>=0."""
    n = len(a)
    a = a.copy().astype(np.float64)
    s = np.sum(a)
    a = a / s
    best_a = a.copy()
    best_obj = compute_obj(best_a)

    for step in range(n_steps):
        h = np.convolve(a, a)
        max_h = np.max(h)
        if max_h < 1e-15:
            break

        # Active set with relative tolerance
        tol = max_h * (0.001 + 0.01 * (step / n_steps))
        active = h >= max_h - tol
        n_active = int(np.sum(active))
        if n_active == 0:
            active = h >= max_h - 1e-10
            n_active = int(np.sum(active))

        # Compute subgradient (average of active gradients)
        grad = np.zeros(n)
        active_idx = np.where(active)[0]
        for k in active_idx:
            j_start = max(0, k - n + 1)
            j_end = min(n - 1, k)
            vals = a[k - j_end:k - j_start + 1][::-1]
            grad[j_start:j_end + 1] += 2.0 * vals
            if k % 2 == 0 and 0 <= k // 2 < n:
                grad[k // 2] -= a[k // 2]
        grad /= n_active

        # Adaptive step size
        step_size = 0.5 / (1.0 + 0.01 * step)

        a_new = a - step_size * grad
        a_new = np.maximum(a_new, 0.0)
        s = np.sum(a_new)
        if s < 1e-10:
            break
        a_new = a_new / s

        obj_new = compute_obj(a_new)
        if obj_new < best_obj:
            best_obj = obj_new
            best_a = a_new.copy()

        a = a_new

    return best_a, best_obj


def coord_refine(a, n_rounds=3, verbose=False):
    """Coordinate descent: for each element, try several values."""
    n = len(a)
    a = a.copy().astype(np.float64)
    best_a = a.copy()
    best_obj = compute_obj(best_a)

    for rnd in range(n_rounds):
        for j in range(n):
            old_val = a[j]
            # Try several multipliers
            multipliers = [0.0, 0.2, 0.5, 0.8, 1.0, 1.2, 1.5, 2.0]
            best_x = old_val
            best_local = best_obj

            for mult in multipliers:
                x = old_val * mult
                if x < 0:
                    x = 0.0
                a[j] = x
                obj = compute_obj(a)
                if obj < best_local:
                    best_local = obj
                    best_x = x

            a[j] = best_x

        if compute_obj(a) < best_obj:
            best_obj = compute_obj(a)
            best_a = a.copy()

    return best_a, best_obj


def structured_init(n, rng):
    """Generate diverse structured initializations."""
    choice = int(rng.integers(0, 6))
    if choice == 0:
        # Cosine with random frequency
        freq = rng.uniform(0.5, 3.0)
        phase = rng.uniform(0, 2 * np.pi)
        a = np.maximum(0, np.cos(2 * np.pi * freq * np.arange(n) / n + phase))
    elif choice == 1:
        # Two-lobe
        mid = n // 2
        a = np.ones(n)
        a[:mid] = rng.uniform(0.5, 1.5)
        a[mid:] = rng.uniform(0.5, 1.5)
    elif choice == 2:
        # Periodic
        period = int(rng.integers(max(2, n // 30), max(3, n // 5)))
        pattern = rng.random(period)
        a = np.tile(pattern, n // period + 1)[:n]
    elif choice == 3:
        # Quadratic phase
        a = np.ones(n) * np.exp(-rng.uniform(0, 2) * (np.arange(n) / n - 0.5) ** 2)
    elif choice == 4:
        # Sawtooth
        a = np.abs(np.sin(np.arange(n) * rng.uniform(0.1, 1.0)))
    else:
        # Random
        a = rng.random(n)

    s = np.sum(a)
    if s < 1e-10:
        a = rng.random(n)
        s = np.sum(a)
    return a / s
