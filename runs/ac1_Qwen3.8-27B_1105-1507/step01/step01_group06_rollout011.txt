"""
Minimizes J(a) = 2*n*max(convolve(a,a))/(sum(a))^2 using:
1. LP-based mirror descent (best-response) with Nesterov-like momentum and careful step sizes
2. Symmetrization exploiting reversal invariance
3. Targeted subgradient polish on peak lags
4. L-BFGS-B final refinement
5. Multi-start across n=512, 1024 with diverse initializations
"""

import numpy as np
from scipy.optimize import linprog, minimize
import time


def fast_conv(a):
    """Full self-convolution via FFT, O(n log n)."""
    n = len(a)
    s = 1
    while s < 2 * n - 1:
        s <<= 1
    fa = np.fft.rfft(a, s)
    return np.fft.irfft(fa * fa, s)[:2 * n - 1]


def build_conv_matrix(a):
    """Build matrix A (2n-1 x n) where A @ g = full_convolve(a, g)."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for k in range(2 * n - 1):
        jl = max(0, k - n + 1)
        jh = min(n - 1, k)
        if jl <= jh:
            A[k, jl:jh + 1] = a[k - jh:k - jl + 1][::-1]
    return A


def lp_best_response(a, M):
    """Solve LP: max 1^T g s.t. (a*g)_k <= M for all k, g >= 0."""
    n = len(a)
    c_vec = -np.ones(n)
    A = build_conv_matrix(a)
    res = linprog(c_vec, A_ub=A, b_ub=np.full(2 * n - 1, M),
                  bounds=[(0, None)] * n, method='highs')
    if res.success:
        g = res.x
        if g.sum() > 1e-12:
            return g
    return None


def normalize(a):
    """Clip to non-negative and normalize to sum=1."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return a
    return a / s


def symmetrize(a):
    """Exploit reversal symmetry: (a + flip(a)) / 2."""
    return 0.5 * (a + a[::-1].copy())


def objective_normalized(a):
    """Compute J for a normalized sequence (sum=1)."""
    n = len(a)
    b = fast_conv(a)
    return 2.0 * n * float(b.max())


def lp_mirror_descent(a_init, max_iters=300, t_start=0.25, t_end=0.005,
                      sym_period=15, momentum=0.0, deadline=None):
    """
    LP mirror descent with optional momentum and symmetrization.
    Returns (best_a, best_obj).
    """
    a = normalize(a_init.copy())
    best_a = a.copy()
    best_obj = objective_normalized(a)
    v = np.zeros_like(a)  # momentum buffer
    n = len(a)
    stagnant = 0

    for it in range(max_iters):
        if deadline is not None and time.monotonic() > deadline:
            break

        # Geometric step size schedule
        if max_iters > 1:
            t = t_start * (t_end / t_start) ** (it / (max_iters - 1))
        else:
            t = t_start

        b = fast_conv(a)
        M = float(b.max())

        g = lp_best_response(a, M)
        if g is None:
            break
        g = normalize(g)

        # Update with optional momentum
        if momentum > 0.0:
            direction = g - a
            v = momentum * v + direction
            a_new = a + t * v
        else:
            a_new = (1.0 - t) * a + t * g

        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s < 1e-15:
            break
        a_new = a_new / s

        # Periodic symmetrization
        if sym_period > 0 and it % sym_period == sym_period - 1:
            a_new = symmetrize(a_new)
            a_new = normalize(a_new)

        a = a_new
        obj = objective_normalized(a)

        if obj < best_obj - 1e-12:
            best_obj = obj
            best_a = a.copy()
            stagnant = 0
        else:
            stagnant += 1

        if stagnant > 50 and it > 30:
            # Try a small perturbation to escape
            if momentum > 0:
                momentum *= 0.5
            stagnant = 0
            pert = 0.01 * np.random.default_rng(it).uniform(-1, 1, n)
            a = normalize(a + pert)

    return best_a, best_obj


def subgradient_polish(a_init, max_iters=300, lr_start=0.02, deadline=None):
    """
    Projected subgradient descent on max_k (a*a)_k.
    Focuses on reducing the peak lags specifically.
    """
    a = normalize(a_init.copy())
    best_a = a.copy()
    best_obj = objective_normalized(a)
    n = len(a)

    for it in range(max_iters):
        if deadline is not None and time.monotonic() > deadline:
            break

        b = fast_conv(a)
        M = float(b.max())

        # Active set: lags within tolerance of max
        tol = max(1e-8, 0.05 * (M - b.min()))
        active = np.where(b >= M - tol)[0]

        # Compute subgradient: grad[i] = sum_{k in active} a[k-i] for valid k-i
        grad = np.zeros(n)
        for k in active:
            jl = max(0, k - n + 1)
            jh = min(n - 1, k)
            for i in range(jl, jh + 1):
                grad[i] += a[k - i]

        grad_norm = np.linalg.norm(grad)
        if grad_norm < 1e-15:
            break

        # Learning rate with decay
        lr = lr_start / (1.0 + 0.05 * it)

        a_new = a - lr * (grad / grad_norm)
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s < 1e-15:
            break
        a_new = a_new / s
        a = a_new

        obj = objective_normalized(a)
        if obj < best_obj:
            best_obj = obj
            best_a = a.copy()

    return best_a, best_obj
