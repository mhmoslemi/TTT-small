```python
def coordinate_descent(a, deadline, ...):
    n = len(a)
    S = sum(a)  # keep this fixed
    
    while time < deadline:
        conv = np.convolve(a, a)
        M = conv.max()
        max_idx = argmax(conv)
        
        # Find active lags (within epsilon of max)
        active = np.where(conv >= M - 1e-6)[0]
        
        # For each coordinate j, compute the gradient of the max
        # (using all active lags)
        grad = np.zeros(n)
        for k in active:
            for j in range(n):
                if 0 <= k - j < n:
                    grad[j] += a[k-j]
        # grad[j] = sum over active k of a[k-j] (times 2, but constant factor doesn't matter)
        
        # We want to decrease a[j] where grad[j] is large
        # But we need to maintain sum(a) = S and a[j] >= 0
        # Simple approach: find j with max grad, decrease it, and distribute the mass
        # to coordinates with min grad.
        
        j_decrease = argmax(grad)
        j_increase = argmin(grad)
        
        # How much can we decrease a[j_decrease]?
        # We need: for all active k, conv[k] - 2*a[j_decrease] * delta <= M (approximately)
        # And a[j_decrease] - delta >= 0
        
        max_delta = a[j_decrease]  # can't go below 0
        # Also need to not create new peaks elsewhere...
        
        # For simplicity, use a small step
        delta = min(max_delta, lr * a[j_decrease])
        
        a[j_decrease] -= delta
        a[j_increase] += delta
        
        # Renormalize
        a *= S / sum(a)
```
