Optimizes non-negative step function heights to minimize 2n*max(conv(f,f))/(sum(f))^2.
Uses smoothed mirror descent in the log-domain with temperature annealing,
frequency-domain initialization, and adaptive step size.
"""

import numpy as np

def _fft_convolve(a, b):
    """Compute linear convolution using FFT."""
    n = len(a) + len(b) - 1
    size = 1
    while size < n:
        size <<= 1
    A = np.fft.rfft(a, size)
    B = np.fft.rfft(b, size)
    result = np.fft.irfft(A * B, size)
    return result[:n]

def _score_normalized(f):
    """Compute score for normalized f (sum=1, f>=0). Returns 2n*max(conv)."""
    n = len(f)
    c = _fft_convolve(f, f)
    M = float(np.max(c))
    return 2.0 * n * M

def _cross_correlation(f, s):
    """Compute (f cross-corr s)[i] = sum_j f[j] * s[j+i] for i=0..len(f)-1."""
    nf = len(f)
    ns = len(s)
    # We need R[i] = sum_j f[j] * s[j+i] for i = 0, ..., nf-1

        f]

    f_rev = f[::-1]

    while size < total_len:

    F = np.fft.rfft(f_rev, size)
    S = np.fft.rfft(s, size)
    corr = np.fft.irfft(F * S, size)
    # R[i] = corr[nf-1+i] for i = 0, ..., nf-1
    return corr[nf-1:nf-1+n

    # Number of T values depends on available time

    t0 = time.monotonic()

    grad, s = _smoothed_gradient(f, 1.0, c)
    iter_time = (time.monotonic() - t0) / 2.0
    
    # Budget iterations
    max_iter = int(remaining / max(iter_time * 1.5, 0.001))
    max_iter = max(100, min(max_iter, 1000000))
    
    # Create temperature schedule
    n_phases = min(50, max(10, max_iter // 200))
    T_schedule = np.geomspace(0.5, max(10.0, n * 0.1), n_phases)
    
    eta = 1.0
    total_steps = 0
    
    for T in T_schedule:
        if time.monotonic() > deadline - 2:
            break
        
        steps = max(5, max_iter // n_phases)
        
        for step in range(steps):
            if time.monotonic() > deadline - 1:
                break
            
            c = _fft_convolve(f, f)
            M = float(np.max(c))
            
            if M < 1e-20:
                break
            
            # Softmax
            c_scaled = T * (c - M) / M  # Normalize by M for numerical stability
            exp_val = np.exp(c_scaled)
            s = exp_val / float(np.sum(exp_val))
            
            # Cross-correlation for gradient
            R, _ = _cross_correlation(f, s)
            grad = 2.0 * R[:n]
            
            # Adaptive step size: eta proportional to 1/M (since we're minimizing 2n*M)
            # The "natural" scale is 1/M for the gradient
            eta_local = eta / max(M, 1e-10) * M  # = eta
            # Actually, let me think about this differently.
            # We want: f_i * exp(-eta * grad_i / scale)
            # where scale makes the update reasonable
            # grad_i is on the order of 2*max(f)*sum(s relevant) ~ 2/n
            # We want the exponent to be O(1), so eta * grad / (2/n) ~ 1
            # => eta ~ n/2
            
            scale = float(np.max(np.abs(grad)))
            if scale < 1e-20:
                break
            
            f = f * np.exp(-eta * grad / scale)
            f = np.maximum(f, 1e-30)
            f = f / np.sum(f)
            
            total_steps += 1
            
            # Check score every 50 steps
            if total_steps % 50 == 0:
                score = _score_normalized(f)
                if score < best_score:
                    best_score = score
                    best_f = f.copy()
        
        # End of T phase
        score = _score_normalized(f)
        if score < best_score:
            best_score = score
            best_f = f.copy()
        
        if total_steps % 2000 == 0 or T == T_schedule[-1]:
            print(f"  {label}: T={T:.2f}, steps={total_steps}, score={best_score:.6f}")
    
    return best_f, best_score
