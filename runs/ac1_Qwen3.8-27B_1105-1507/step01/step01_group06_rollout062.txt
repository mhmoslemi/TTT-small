import math

def fast_conv(a):

    while s < 2 * n - 1:
        s *= 2
    fa = np.fft.rfft(a, s)
    return np.fft.irfft(fa * fa, s)[:2*n-1]

def build_A(a):
    n = len(a)
    A = np.zeros((2*n-1, n))

"""
Minimizes J(a) = 2*n*max(convolve(a,a))/(sum(a))^2 using:
1. LP-based best-response iterations with line search for structured updates
2. Coordinate descent refinement exploiting per-coordinate convexity
3. Asymmetric initializations (power-law, exponential, shifted Gaussian)
4. L-BFGS-B gradient polish with subgradient
5. Multiple sequence lengths (128-2048) with diverse starts
Key insight: optimal sequences must be asymmetric (symmetric gives J>=2).
"""

import numpy as np
from scipy.optimize import linprog, minimize
import time


def fast_conv(a):
    """Full self-convolution via FFT, O(n log n)."""
    n = len(a)
    s = 1
    while s < 2 * n - 1:
        s *= 2
    fa = np.fft.rfft(a, s)
    return np.fft.irfft(fa * fa, s)[:2 * n - 1]


def build_A(a):
    """Build constraint matrix A where A[k,j] gives (a*g)[k] = A[k,:] @ g."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for k in range(2 * n - 1):
        jl = max(0, k - n + 1)
        jh = min(n - 1, k)
        if jl <= jh:
            A[k, jl:jh + 1] = a[k - jh:k - jl + 1][::-1]
    return A


def lp_best_response(a, M, alpha=1.0):
    """LP: max sum(g) s.t. (a*g)[k] <= alpha*M, g >= 0."""
    n = len(a)
    A = build_A(a)
    c = -np.ones(n)
    r = linprog(c, A_ub=A, b_ub=np.full(2 * n - 1, alpha * M),
               bounds=[(0, None)] * n, method='highs')
    if r.success and r.x is not None and r.x.sum() > 1e-10:
        return r.x
    return None


def compute_J(a):
    """Compute objective J(a) = 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    s = a.sum()
    if s < 1e-10:
        return float('inf')
    b = fast_conv(a)
    return float(2 * n * b.max() / (s * s))


def line_search(a, g, n_evals=25):
    """Find best mixing parameter t in [0,1] for a_t = (1-t)*a + t*g."""
    n = len(a)
    best_t = 0.0
    best_J = compute_J(a)
    for t in np.linspace(0.05, 1.0, n_evals):
        a_t = (1.0 - t) * a + t * g
        a_t = np.maximum(a_t, 0.0)
        s = a_t.sum()
        if s < 1e-10:
            continue
        a_t = a_t / s
        J = compute_J(a_t)
        if J < best_J:
            best_J = J
            best_t = t
    return best_t, best_J


def lp_iterate(a, n_iters=150, deadline=None, verbose=False):
    """LP-based iterative optimization with line search."""
    a = np.asarray(a, dtype=np.float64).copy()
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-10:
        return a, float('inf')
    a = a / s
    n = len(a)
    best_a = a.copy()
    best_J = compute_J(a)
    stall = 0

    for it in range(n_iters):
        if deadline is not None and time.monotonic() > deadline:
            break
        b = fast_conv(a)
        M = float(b.max())
        J = 2.0 * n * M / (a.sum() ** 2)
        if J < best_J:
            best_J = J
            best_a = a.copy()
            stall = 0
        else:
            stall += 1

        # Try different alpha values when stalled
        alphas = [1.0, 0.8, 1.2, 0.6] if stall > 5 else [1.0]
        improved = False
        for alpha in alphas:
            g = lp_best_response(a, M, alpha=alpha)
            if g is None:
                continue
            g = g / g.sum()
            t_best, J_new = line_search(a, g, n_evals=20)
            if J_new < best_J * (1 - 1e-6):
                a = (1.0 - t_best) * a + t_best * g
                a = np.maximum(a, 0.0)
                s = a.sum()
                if s < 1e-10:
                    break
                a = a / s
                best_J = min(best_J, J_new)
                best_a = a.copy()
                improved = True
                stall = 0
                break

        if not improved and stall > 30:
            # Reheat with small perturbation
            a = a + 0.01 * np.random.default_rng(it).uniform(0, 1, n)
            a = np.maximum(a, 0)
            a = a / a.sum()
            stall = 0

        if verbose and it % 20 == 0:
            print(f"    LP iter {it}: J={best_J:.6f}", flush=True)

    return best_a, best_J


def coordinate_descent(a, n_sweeps=20, deadline=None, verbose=False):
    """Coordinate descent: optimize one coordinate at a time."""
    a = np.asarray(a, dtype=np.float64).copy()
    n = len(a)
    best_J = compute_J(a)
    best_a = a.copy()

    for sweep in range(n_sweeps):
        if deadline is not None and time.monotonic() > deadline:
            break
        for j in range(n):
            if deadline is not None and time.monotonic() > deadline:
                break
            a_old = a[j]
            if a_old < 1e-10:
                # Try small positive values
                candidates = [0.01, 0.05, 0.1] * (max(a) if max(a) > 0 else 1.0)
            else:
                candidates = [a_old * f for f in [0.5, 0.7, 0.9, 1.1, 1.3, 1.6]] + [0.0, a_old * 0.1]

            best_val = best_J
            best_aj = a_old
            for aj in candidates:
                aj = max(0.0, aj)
                a[j] = aj
                s = a.sum()
                if s < 1e-10:
                    continue
                b = fast_conv(a)
                J = 2.0 * n * b.max() / (s * s)
                if J < best_val:
                    best_val = J
                    best_aj = aj
            a[j] = best_aj

        J = compute_J(a)
        if J < best_J:
            best_J = J
            best_a = a.copy()
        elif verbose:
            print(f"    CD sweep {sweep}: J={best_J:.6f}", flush=True)

    return best_a, best_J


def make_initializations(n, rng):
    """Generate diverse asymmetric initializations."""
    inits = []
    t = np.arange(n, dtype=np.float64)
    tn = t / max(1, n - 1)

    # Power-law decays (asymmetric)
    for alpha in [0.5, 1.0, 1.5, 2.0, 3.0]:
        inits.append(np.power(1.0 - tn, alpha))
        inits.append(np.power(tn, alpha))

    # Exponential decays
    for beta in [1.0, 2.0, 3.0, 5.0]:
        inits.append(np.exp(-beta * tn))
        inits.append(np.exp(-beta * (1.0 - tn)))

    # Shifted Gaussians (asymmetric)
    for center in [0.25, 0.35, 0.6, 0.75]:
        for sigma in [0.1, 0.15, 0.2, 0.3]:
            inits.append(np.exp(-((tn - center) ** 2) / (2 * sigma ** 2)))

    # Beta-like distributions (asymmetric)
    for p, q in [(2, 5), (5, 2), (1.5, 4), (4, 1.5), (3, 7), (7, 3)]:
        # Approximate Beta pdf shape
        inits.append(np.power(tn, p - 1) * np.power(1.0 - tn, q - 1))

    # Cosine ramps (asymmetric)
    inits.append(0.5 + 0.5 * np.cos(np.pi * tn))
    inits.append(0.5 + 0.5 * np.cos(2 * np.pi * tn))
    inits.append(np.abs(np.sin(np.pi * tn * 1.5)))

    # Random
    inits.append(rng.uniform(0.1, 1.0, n))

    # Triangle (asymmetric)
    peak = int(0.3 * n)
    tri = np.minimum(t / max(1, peak), (n - 1 - t) / max(1, n - 1 - peak))
    inits.append(np.maximum(tri, 0.01))

    # Two-bump
    inits.append(np.exp(-((tn - 0.25) ** 2) / 0.01) + 0.5 * np.exp(-((tn - 0.7) ** 2) / 0.02))

    # Clean up
    clean = []
    for x in inits:
        x = np.maximum(x, 0.0)
        if x.sum() > 1e-10:
            x = x / x.sum()
            clean.append(x)
    return clean


def gradient_refine(a, maxiter=300, deadline=None):
    """L-BFGS-B refinement using smoothed max."""
    n = len(a)

    def smooth_max(vals, s=50.0):
        mx = vals.max()
        if mx < 1e-15:
            return 0.0, np.zeros_like(vals)
        shifted = vals / mx
        exps = np.exp(s * (shifted - 1.0))
        logsum = mx + np.log(np.sum(exps)) / s
        # Gradient w.r.t. vals
        grad = exps / np.sum(exps)
        return logsum, grad * mx

    def obj_and_grad(x):
        ax = np.maximum(x, 1e-15)
        s = ax.sum()
        if s < 1e-10:
            return 1e10, np.zeros_like(x)
        ax = ax / s
        b = fast_conv(ax)
        vals = b
        sm, grad_sm = smooth_max(vals, s=50.0)
        # Chain rule: db/dax = A @ 1 (for the active component)
        # Approximate gradient of max(conv) w.r.t. ax
        # d(conv(ax,ax))[k]/d(ax[j]) = ax[k-j] + ax[k-j] (symmetric) = 2*ax[k-j]
        # So grad of sm w.r.t. ax = sum_k grad_sm[k] * 2*ax[k-j] for each j
        # This is 2 * (grad_sm * ax) convolved... actually it's A.T @ grad_sm where A is the conv matrix
        # For self-conv: d(ax*ax)[k]/d(ax[j]) = ax[k-j] (if valid) + ax[k-j] (if valid)
        # Actually: (ax*ax)[k] = sum_i ax[i]*ax[k-i], d/d(ax[j]) = ax[k-j] + ax[k-j] = 2*ax[k-j] (if j != k-j)
        # or ax[k-j] (if j == k-j, i.e., k=2j)
        # So the gradient vector is: for each j, sum_k grad_sm[k] * (ax[k-j] if k-j valid) * (1 + 1[j!=k-j])
        # Simplified: grad_j = sum_k grad_sm[k] * ax[k-j] * (2 if k != 2j else 1)
        # This is essentially 2*(grad_sm * ax) correlated... let's just use the matrix form

        # Build gradient via the convolution structure
        # d(ax*ax)[k]/d(ax[j]) contributes to grad[j]
        # grad[j] = sum_k (d sm/d b[k]) * (d b[k]/d ax[j])
        # d b[k]/d ax[j] = ax[k-j] + ax[k-j] = 2*ax[k-j] when 0<=k-j<n and k-j != j
        #                              = ax[k-j] = ax[j] when k-j = j (i.e., k=2j)
        # So grad[j] = sum_k grad_sm[k] * ax[k-j] * (1 + I[k != 2j])
        #            = sum_k grad_sm[k] * ax[k-j] + sum_{k != 2j} grad_sm[k] * ax[k-j]

        # First term: (grad_sm * ax) full convolution reversed = correlate(grad_sm, ax)
        # Actually let's just do it directly for correctness (n is at most ~2048)
        grad = np.zeros(n)
        for k in range(2 * n - 1):
            gk = grad_sm[k]
            if abs(gk) < 1e-15:
                continue
            jl = max(0, k - n + 1)
            jh = min(n - 1, k)
            for j in range(jl, jh + 1):
                coeff = 1.0 if k == 2 * j else 2.0
                grad[j] += gk * ax[k - j] * coeff

        # Chain rule through normalization (ax = x/s)
        # d(ax)/d(x[j]) = (1/s) * (I - ax/s * 1) ... this gets complex
        # For simplicity, just return grad (ignoring normalization since we normalize at end)
        obj_val = float(sm)
        # Scale gradient appropriately
        grad = grad / s
        return obj_val, grad

    x0 = a.copy()
    res = minimize(lambda x: obj_and_grad(x)[0], x0, jac=lambda x: obj_and_grad(x)[1],
                   method='L-BFGS-B', bounds=[(0, None)] * n,
                   options={'maxiter': maxiter, 'ftol': 1e-14, 'gtol': 1e-8, 'disp': False})
    if res.fun < 1e10:
        return res.x
    return a

    and gradient refinement across multiple sequence lengths and initializations.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + float(budget_s) - 15.0
    best = None
    best_score = float('inf')

    def try_update(a):
        nonlocal best, best_score
        a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
        if a.sum() < 0.01:
            return
        score = float(evaluate_sequence(a.tolist()))
        if np.isfinite(score) and score < best_score:
            best = a.copy()
            best_score = score
            print(f"  NEW BEST: J={score:.6f} (n={a.size})", flush=True)

    # Try parent
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        if p.ndim == 1 and p.size > 0 and np.all(np.isfinite(p)):
            try_update(p)
    except (NameError, AttributeError):
        pass

    # Phase 1: LP iterations at multiple lengths with diverse inits
    for n in [256, 512, 1024]:
        if time.monotonic() > deadline:
            break
        print(f"\n=== LP Phase: n={n} ===", flush=True)
        t0 = time.monotonic()

        inits = make_initializations(n, rng)

        # Also add parent resampled if available
        try:
            p = np.asarray(height_sequence_1, dtype=np.float64)
            if p.ndim == 1 and p.size > 0:
                t_old = np.linspace(0, 1, p.size)
                t_new = np.linspace(0, 1, n)
                inits.append(np.maximum(np.interp(t_new, t_old, p), 0.0))
        except (NameError, AttributeError):
            pass

        # Shuffle and take a subset for time management
        np.random.default_rng(seed + n).shuffle(inits)
        max_inits = min(len(inits), 15 if n <= 512 else 8)

        for i, a0 in enumerate(inits[:max_inits]):
            if time.monotonic() > deadline:
                break
            a0 = np.maximum(a0, 0.0)
            if a0.sum() < 1e-10:
                continue
            a0 = a0 / a0.sum()

            # LP iteration
            n_iters = 100 if n <= 512 else 60
            a_res, J_res = lp_iterate(a0, n_iters=n_iters, deadline=deadline)
            try_update(a_res)

            # Quick coordinate descent polish
            if time.monotonic() < deadline:
                a_cd, J_cd = coordinate_descent(a_res, n_sweeps=5, deadline=deadline)
                try_update(a_cd)

        elapsed = time.monotonic() - t0
        print(f"  LP n={n} done in {elapsed:.1f}s, best={best_score:.6f}", flush=True)

    # Phase 2: More aggressive LP at n=512 with more iterations
    if time.monotonic() < deadline:
        print(f"\n=== LP Extended: n=512 ===", flush=True)
        n = 512
        # Use best found so far (resampled)
        if best is not None:
            t_old = np.linspace(0, 1, best.size)
            t_new = np.linspace(0, 1, n)
            a_start = np.maximum(np.interp(t_new, t_old, best), 0.0)
            a_start = a_start / a_start.sum()
            a_res, J_res = lp_iterate(a_start, n_iters=200, deadline=deadline)
            try_update(a_res)
            a_cd, J_cd = coordinate_descent(a_res, n_sweeps=10, deadline=deadline)
            try_update(a_cd)
        else:
            inits = make_initializations(n, rng)
            for a0 in inits[:6]:
                if time.monotonic() > deadline:
                    break
                a_res, J_res = lp_iterate(a0, n_iters=100, deadline=deadline)
                try_update(a_res)

    # Phase 3: Try n=2048 with LP
    if time.monotonic() < deadline:
        print(f"\n=== LP: n=2048 ===", flush=True)
        t0 = time.monotonic()
        n = 2048
        # Use best resampled
        if best is not None:
            t_old = np.linspace(0, 1, best.size)
            t_new = np.linspace(0, 1, n)
            a_start = np.maximum(np.interp(t_new, t_old, best), 0.01)
            a_start = a_start / a_start.sum()
            a_res, J_res = lp_iterate(a_start, n_iters=50, deadline=deadline)
            try_update(a_res)

        # Also try some direct inits at n=2048
        inits = [
            np.power(1.0 - np.arange(n) / n, 1.5),
            np.exp(-2.0 * np.arange(n) / n),
            np.power(np.arange(n) / n, 2.0) * (1.0 - np.arange(n) / n),
        ]
        for a0 in inits:
            if time.monotonic() > deadline:
                break
            a0 = np.maximum(a0, 0.0)
            if a0.sum() < 1e-10:
                continue
            a0 = a0 / a0.sum()
            a_res, J_res = lp_iterate(a0, n_iters=40, deadline=deadline)
            try_update(a_res)
        print(f"  n=2048 done in {time.monotonic()-t0:.1f}s, best={best_score:.6f}", flush=True)

    # Phase 4: Gradient refinement on best found
    if best is not None and time.monotonic() < deadline:
        print(f"\n=== Gradient Polish ===", flush=True)
        n = best.size
        if n <= 1024:
            a_res = gradient_refine(best, maxiter=200, deadline=deadline)
            if a_res is not None and np.all(np.isfinite(a_res)):
                a_res = np.maximum(a_res, 0.0)
                if a_res.sum() > 0.01:
                    try_update(a_res)

        # Try gradient at different lengths
        for target_n in [512, 1024]:
            if time.monotonic() > deadline:
                break
            t_old = np.linspace(0, 1, best.size)
            t_new = np.linspace(0, 1, target_n)
            a_start = np.maximum(np.interp(t_new, t_old, best), 0.01)
            a_start = a_start / a_start.sum()
            a_res = gradient_refine(a_start, maxiter=100, deadline=deadline)
            if a_res is not None and np.all(np.isfinite(a_res)):
                a_res = np.maximum(a_res, 0.0)
                if a_res.sum() > 0.01:
                    try_update(a_res)

    # Phase 5: Final coordinate descent on best
    if best is not None and time.monotonic() < deadline:
        print(f"\n=== Final CD Polish ===", flush=True)
        a_cd, J_cd = coordinate_descent(best, n_sweeps=15, deadline=deadline)
        try_update(a_cd)

    # Fallback
    if best is None:
        best = np.ones(256)
        best_score = evaluate_sequence(best.tolist())

    print(f"\nFINAL: J={best_score:.6f}, n={best.size}", flush=True)
    return [float(x) for x in best]
