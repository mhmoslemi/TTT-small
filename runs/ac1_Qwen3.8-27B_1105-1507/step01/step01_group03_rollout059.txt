def normalize_sequence(arr):
    """Return a nonnegative floating vector normalized to sum 1, or None."""
    try:
        x = np.asarray(arr, dtype=np.float64).ravel()
    except Exception:
        return None
    if x.size == 0:
        return None
    x = np.where(np.isfinite(x), x, 0.0)
    x = np.maximum(x, 0.0)
    s = float(x.sum())
    if s <= 1e-300:

    x /= s
    return x


def score_normalized(x, n):
    """Score assuming x is nonnegative and sum(x) = 1."""
    conv = np.convolve(x, x)
    m = float(np.max(conv))
    if not np.isfinite(m):
        return float("inf")
    return float(2.0 * n * m)


def make_conv_template(n):
    """Precompute the CSR matrix structure for convolve(f, g)."""
    m = 2 * n - 1
    k64 = np.arange(m, dtype=np.int64)
    js = np.maximum(0, k64 - n + 1).astype(np.int64)
    je = np.minimum(n - 1, k64).astype(np.int64)
    counts = (je - js + 1).astype(np.int64)
    total = int(counts.sum())
    
    cum = np.cumsum(counts)
    starts = np.repeat(cum - counts, counts)
    local = np.arange(total, dtype=np.int64) - starts
    
    row = np.repeat(k64, counts).astype(np.int32)
    col_base = js[row.astype(np.int64)]
    col = (col_base + local).astype(np.int32)
    offsets = (row.astype(np.int64) - col.astype(np.int64)).astype(np.int32)
    
    A_template = csr_matrix((offsets, (row, col)), shape=(m, n), dtype=np.float64)

    c_obj = -np.ones(n, dtype=np.float64)
    bounds = [(0.0, None)] * n
    return A_template, edge_offsets, c_obj, bounds, m

    A.data[:] = f[edge_offsets]
    A.data *= lp_scale
    
    b = np.full(m, M * lp_scale * lp_scale, dtype=np.float64)
    res = linprog(c_obj, A_ub=A, b_ub=b, bounds=bounds, method="highs")
    return res

    d = a - phi * (a - b)

            c = d
            fc = fd
            d = a - phi * (a - b)
            fd = eval_step(f, g, d, n)

            stall += 2
            f = mix_noisy(best_f, rng, 0.3)
            if f is None:
                f = best_f.copy()
            continue
        
        s_ratio = s_star / float(n)
        if s_ratio <= 1.0 + 1e-6:
            stall += 2
            if stall > 20:
                break
            f = mix_noisy(best_f, rng, 0.3)
            if f is None:
                f = best_f.copy()
            continue
        
        g = x / s_star
        
        # Momentum
        if use_momentum and prev_g is not None:
            g = 0.7 * g + 0.3 * prev_g
            g = np.maximum(g, 0.0)
            s_g = float(g.sum())
            if s_g > 1e-300:

        prev_g = g.copy()

        r = np.convolve(f, g)

            f = mix_noisy(best_f, rng, 0.3)

                f = best_f.copy()
            continue

                f = best_f.copy()

        else:

        except Exception:
            pass

        best_cand_score = best_score
        best_t = None

                best_cand_f = f_new

            f = best_cand_f
            best_f = f.copy()

            stall = 0

            else:
                f = best_f.copy()
            if f is None:
                f = best_f.copy()
            if stall > 20:
                break
    
    return best_f, float(best_score)


def _golden_in_range(f, g, n, lo, hi):
    """Golden section search in [lo, hi]."""
    if hi <= lo:
        return 0.0
    gr = 0.6180339887498949
    c = hi - gr * (hi - lo)
    d = lo + gr * (hi - lo)
    fc = eval_step(f, g, c, n)
    fd = eval_step(f, g, d, n)
    
    for _ in range(20):
        if hi - lo < 1e-8:
            break
        if fc < fd:
            hi = d
            d = c
            fd = fc
            c = hi - gr * (hi - lo)
            fc = eval_step(f, g, c, n)
        else:
            lo = c
            c = d
            fc = fd
            d = lo + gr * (hi - lo)
            fd = eval_step(f, g, d, n)
    
    return (lo + hi) / 2.0


def mix_noisy(base, rng, eps):
    """Mix a base vector with a random Dirichlet vector."""
    f = normalize_sequence(base)
    if f is None:
        return None
    n = f.size
    if eps <= 1e-12:
        return f
    try:
        mix = rng.dirichlet(np.ones(n, dtype=np.float64))
    except Exception:
        mix = np.ones(n, dtype=np.float64) / float(n)
    if eps >= 1.0:
        f = mix
    else:
        f = (1.0 - eps) * f + eps * mix
    return normalize_sequence(f)


def interpolate_sequence(x, n):
    """Resample a nonnegative sequence to length n and normalize it."""
    x = normalize_sequence(x)
    if x is None or n <= 0:
        return None
    m = x.size
    if m == n:
        return x
    if m == 1:
        return np.ones(n, dtype=np.float64) / float(n)
    old = np.linspace(0.0, 1.0, m, dtype=np.float64)
    new = np.linspace(0.0, 1.0, n, dtype=np.float64)
    out = np.interp(new, old, x)
    return normalize_sequence(out)


def legendre_start(n, rng):
    """Create a Legendre-sequence-inspired starting vector."""
    # Find a prime near n
    p = n
    while True:
        if _is_prime(p):
            break
        p += 1
    # Legendre symbol sequence
    seq = np.zeros(p)
    for a in range(1, p):
        ls = (a ** ((p - 1) // 2)) % p
        if ls == p - 1:
            seq[a] = 1.0
    # Interpolate to length n
    return interpolate_sequence(seq, n)


def _is_prime(n):
    if n < 2:
        return False
    if n < 4:
        return True
    if n % 2 == 0 or n % 3 == 0:
        return False
    i = 5
    while i * i <= n:
        if n % i == 0 or n % (i + 2) == 0:
            return False
        i += 6
    return True


def periodic_start(n, rng):
    """Create a periodic pattern as starting vector."""
    # Random period
    period = rng.integers(3, max(4, n // 10))
    x = np.zeros(n)
    for i in range(n):
        x[i] = 0.5 + 0.5 * np.sin(2 * np.pi * i / period + rng.uniform(0, 2 * np.pi))
    return normalize_sequence(x)


def gaussian_start(n, rng):
    """Create a Gaussian-shaped starting vector."""
    center = n * rng.uniform(0.3, 0.7)
    sigma = n * rng.uniform(0.1, 0.4)
    x = np.exp(-0.5 * ((np.arange(n) - center) / sigma) ** 2)
    return normalize_sequence(x)

"""
Improved LP-based iterative search for a nonnegative step function minimizing
2*n*max(convolve(f,f))/(sum(f))^2.

Key improvements over parent:
1. Golden-section line search for step-size selection
2. M-relaxation to generate diverse LP directions
3. Structured initializations (Legendre sequences, periodic patterns, Gaussian bumps)
4. Momentum in direction updates
5. Adaptive stall handling with increasing perturbation strength
6. Better time allocation across lengths and starting points
"""

import time
import numpy as np
from scipy.optimize import linprog
from scipy.sparse import csr_matrix


def normalize_sequence(arr):
    """Return a nonnegative floating vector normalized to sum 1, or None."""
    try:
        x = np.asarray(arr, dtype=np.float64).ravel()
    except Exception:
        return None
    if x.size == 0:
        return None
    x = np.where(np.isfinite(x), x, 0.0)
    x = np.maximum(x, 0.0)
    s = float(x.sum())
    if s <= 1e-300:
        return None
    x /= s
    return x


def score_normalized(x, n):
    """Score assuming x is nonnegative and sum(x) = 1."""
    conv = np.convolve(x, x)
    m = float(np.max(conv))
    if not np.isfinite(m):
        return float("inf")
    return float(2.0 * n * m)


def make_conv_template(n):
    """Precompute the CSR matrix structure for convolve(f, g)."""
    m = 2 * n - 1
    k64 = np.arange(m, dtype=np.int64)
    js = np.maximum(0, k64 - n + 1).astype(np.int64)
    je = np.minimum(n - 1, k64).astype(np.int64)
    counts = (je - js + 1).astype(np.int64)
    total = int(counts.sum())

    cum = np.cumsum(counts)
    starts = np.repeat(cum - counts, counts)
    local = np.arange(total, dtype=np.int64) - starts

    row = np.repeat(k64, counts).astype(np.int32)
    col_base = js[row.astype(np.int64)]
    col = (col_base + local).astype(np.int32)
    offsets = (row.astype(np.int64) - col.astype(np.int64)).astype(np.int32)

    A_template = csr_matrix((offsets, (row, col)), shape=(m, n), dtype=np.float64)
    A_template.data.fill(0.0)
    edge_offsets = A_template.data.astype(np.int32, copy=True)

    c_obj = -np.ones(n, dtype=np.float64)
    bounds = [(0.0, None)] * n
    return A_template, edge_offsets, c_obj, bounds, m


def solve_direction_lp(template, f, n, M):
    """Solve the LP direction problem for the given f and M."""
    A_template, edge_offsets, c_obj, bounds, m = template
    lp_scale = float(n)

    A = A_template.copy()
    A.data[:] = f[edge_offsets]
    A.data *= lp_scale

    b = np.full(m, M * lp_scale * lp_scale, dtype=np.float64)
    res = linprog(c_obj, A_ub=A, b_ub=b, bounds=bounds, method="highs")
    return res


def eval_step(f, g, t, n):
    """Evaluate the normalized score at interpolation parameter t."""
    f_new = (1.0 - t) * f + t * g
    f_new = np.maximum(f_new, 0.0)
    s = float(f_new.sum())
    if s <= 1e-300:
        return float('inf')
    f_new /= s
    return score_normalized(f_new, n)


def golden_section(f, g, n, lo, hi, max_iter=25):
    """Golden section search for minimum of eval_step in [lo, hi]."""
    if hi <= lo + 1e-12:
        return (lo + hi) / 2.0
    gr = 0.6180339887498949
    c = hi - gr * (hi - lo)
    d = lo + gr * (hi - lo)
    fc = eval_step(f, g, c, n)
    fd = eval_step(f, g, d, n)

    for _ in range(max_iter):
        if hi - lo < 1e-10:
            break
        if fc < fd:
            hi = d
            d = c
            fd = fc
            c = hi - gr * (hi - lo)
            fc = eval_step(f, g, c, n)
        else:
            lo = c
            c = d
            fc = fd
            d = lo + gr * (hi - lo)
            fd = eval_step(f, g, d, n)

    return (lo + hi) / 2.0


def mix_noisy(base, rng, eps):

def mix_concentrated(base, rng, eps):
    """Mix with a concentrated Dirichlet (peaked) vector for more aggressive perturbation."""
    f = normalize_sequence(base)
    if f is None:
        return None
    n = f.size
    if eps <= 1e-12:
        return f.copy()
    # Concentrated Dirichlet: pick a random center and give it high alpha
    center = rng.integers(0, n)
    alpha = np.ones(n) * 0.3
    alpha[center] = 5.0
    # Add a few more peaks
    for _ in range(3):
        center2 = rng.integers(0, n)
        alpha[center2] += 2.0
    try:
        mix = rng.dirichlet(alpha)
    except Exception:
        mix = np.ones(n) / n
    if eps >= 1.0:
        f = mix
    else:
        f = (1.0 - eps) * f + eps * mix
    return normalize_sequence(f)


def interpolate_sequence(x, n):
    """Resample a nonnegative sequence to length n and normalize it."""
    x = normalize_sequence(x)
    if x is None or n <= 0:
        return None
    m = x.size
    if m == n:
        return x
    if m == 1:
        return np.ones(n, dtype=np.float64) / float(n)
    old = np.linspace(0.0, 1.0, m, dtype=np.float64)
    new = np.linspace(0.0, 1.0, n, dtype=np.float64)
    out = np.interp(new, old, x)
    return normalize_sequence(out)


def _is_prime(num):
    """Check if a number is prime."""
    if num < 2:
        return False
    if num < 4:
        return True
    if num % 2 == 0 or num % 3 == 0:
        return False
    i = 5
    while i * i <= num:
        if num % i == 0 or num % (i + 2) == 0:
            return False
        i += 6
    return True


def legendre_start(n, rng):
    """Create a Legendre-sequence (quadratic residues) inspired starting vector."""
    # Find a prime >= n
    p = n if _is_prime(n) else n + 1
    while not _is_prime(p):
        p += 1
    if p > n * 3:
        # Fallback
        return None
    seq = np.zeros(p)
    for a in range(1, p):
        ls = pow(a, (p - 1) // 2, p)
        if ls == p - 1:
            seq[a] = 1.0
    # Also add the symmetric part for better linear autocorrelation
    seq[p // 2] = 0.5
    return interpolate_sequence(seq, n)


def periodic_start(n, rng):
    """Create a periodic pattern as starting vector."""
    period = int(rng.integers(3, max(4, n // 8)))
    phase = rng.uniform(0.0, 2.0 * np.pi)
    x = 0.5 + 0.5 * np.sin(2.0 * np.pi * np.arange(n) / period + phase)
    x = np.maximum(x, 0.0)
    return normalize_sequence(x)


def gaussian_start(n, rng):
    """Create a Gaussian-shaped starting vector."""
    center = n * rng.uniform(0.3, 0.7)
    sigma = n * rng.uniform(0.1, 0.35)
    x = np.exp(-0.5 * ((np.arange(n) - center) / sigma) ** 2)
    return normalize_sequence(x)


def banded_start(n, rng):
    """Create a banded (on/off blocks) starting vector."""
    x = np.zeros(n)
    num_bands = int(rng.integers(3, max(4, n // 50)))
    for _ in range(num_bands):
        start = int(rng.integers(0, n))
        width = int(rng.integers(max(5, n // 20), n // 4))
        end = min(n, start + width)
        x[start:end] = 1.0
    # Add some noise
    x += rng.uniform(0.0, 0.1, n)
    return normalize_sequence(x)


def optimize_lp_improved(n, f_init, time_budget, rng, template, use_momentum=True):
    """
    Run the improved LP-based iteration from one initial vector.
    Uses golden-section step sizes, M-relaxation for direction diversity,
    and momentum in the update direction.
    """
    f = normalize_sequence(f_init)
    if f is None:
        f = np.ones(n, dtype=np.float64) / float(n)

    if template is None:
        try:
            template = make_conv_template(n)
        except Exception:
            return f, score_normalized(f, n)

    best_f = f.copy()
    best_score = score_normalized(f, n)
    start_score = best_score
    start_time = time.monotonic()
    deadline = start_time + max(0.0, float(time_budget))

    stall = 0
    iteration = 0
    prev_g = None
    total_improvements = 0

    while time.monotonic() < deadline:
        if stall > 25:
            break
        iteration += 1
        t0 = time.monotonic()

        conv_ff = np.convolve(f, f)
        M = float(np.max(conv_ff))
        if not np.isfinite(M) or M <= 0.0:
            break

        # Solve LP with primary M
        best_x = None
        best_s_star = 0.0

        try:
            res = solve_direction_lp(template, f, n, M)
            x = getattr(res, "x", None)
            if x is not None:
                x = np.asarray(x, dtype=np.float64).ravel()
                if x.size == n:
                    x = np.where(np.isfinite(x), x, 0.0)
                    x = np.maximum(x, 0.0)
                    s_star = float(x.sum())
                    if s_star > best_s_star:
                        best_s_star = s_star
                        best_x = x.copy()
        except Exception:
            pass

        # If primary LP gave a good direction, also try relaxed M for diversity
        if best_x is not None and best_s_star / n > 1.01:
            # Try slightly relaxed M for a different direction
            relaxations = [1.005, 1.015, 1.03]
            for rel in relaxations:
                if time.monotonic() > deadline - 5:
                    break
                try:
                    res2 = solve_direction_lp(template, f, n, M * rel)
                    x2 = getattr(res2, "x", None)
                    if x2 is not None:
                        x2 = np.asarray(x2, dtype=np.float64).ravel()
                        if x2.size == n:
                            x2 = np.where(np.isfinite(x2), x2, 0.0)
                            x2 = np.maximum(x2, 0.0)
                            s2 = float(x2.sum())
                            # Use relaxed direction if it has comparable sum
                            # but different structure (avoids redundancy)
                            if s2 > 0 and best_x.sum() > 0:
                                # Cosine similarity check
                                cos_sim = float(np.dot(x2, best_x) / (np.linalg.norm(x2) * np.linalg.norm(best_x) + 1e-300))
                                if cos_sim < 0.95:
                                    # Different direction - use it
                                    best_x = x2.copy()
                                    best_s_star = s2
                                    break
                except Exception:
                    continue

        if best_x is None:
            stall += 3
            eps = min(0.8, 0.2 + 0.05 * stall)
            f = mix_noisy(best_f, rng, eps)
            if f is None:
                f = best_f.copy()
            prev_g = None
            continue

        x = best_x
        s_star = best_s_star
        s_ratio = s_star / float(n)

        if s_ratio <= 1.0 + 1e-6:
            stall += 2
            eps = min(0.7, 0.15 + 0.04 * stall)
            f = mix_noisy(best_f, rng, eps)
            if f is None:
                f = best_f.copy()
            prev_g = None
            continue

        g = x / s_star

        # Apply momentum
        if use_momentum and prev_g is not None:
            g = 0.65 * g + 0.35 * prev_g
            g = np.maximum(g, 0.0)
            sg = float(g.sum())
            if sg > 1e-300:
                g /= sg
            else:
                g = x / s_star
        prev_g = g.copy()

        # Compute safe step bound
        r = np.convolve(f, g)
        r_max = float(np.max(r))
        if not np.isfinite(r_max) or r_max <= 0.0:
            stall += 2
            f = mix_noisy(best_f, rng, 0.25)
            if f is None:
                f = best_f.copy()
            continue

        G = float(np.max(np.convolve(g, g)))
        if not np.isfinite(G):
            stall += 2
            continue

        # Safe step: H(t) = (1-t)^2*M + 2*(1-t)*t*r_max + t^2*G <= M
        # H(t) - M = t*2*(r_max - M) + t^2*(G + M - 2*r_max) <= 0
        a_coef = 2.0 * (r_max - M)
        b_coef = G + M - 2.0 * r_max
        if b_coef > 1e-14:
            t_max = -a_coef / b_coef
        else:
            t_max = 1.0
        if not np.isfinite(t_max) or t_max <= 0.0:
            t_max = 1e-8
        t_max = min(1.0, t_max * 0.95)

        # Golden section search for optimal step
        t_best = 0.0
        sc_best = best_score

        # First try golden section in [0, t_max]
        try:
            t_gs = golden_section(f, g, n, 0.0, t_max, max_iter=20)
            sc_gs = eval_step(f, g, t_gs, n)
            if np.isfinite(sc_gs) and sc_gs < sc_best:
                sc_best = sc_gs
                t_best = t_gs
        except Exception:
            pass

        # Also try a few fixed points as backup
        fixed_ts = [0.1 * t_max, 0.3 * t_max, 0.5 * t_max, 0.7 * t_max, 0.9 * t_max]
        for tv in fixed_ts:
            if tv <= 0 or tv > 1.0:
                continue
            sc = eval_step(f, g, tv, n)
            if np.isfinite(sc) and sc < sc_best:
                sc_best = sc
                t_best = tv

        if t_best > 0 and sc_best < best_score:
            f_new = (1.0 - t_best) * f + t_best * g
            f_new = np.maximum(f_new, 0.0)
            ss = float(f_new.sum())
            if ss > 1e-300:
                f_new /= ss
                f = f_new
                best_f = f_new.copy()
                best_score = sc_best
                stall = 0
                total_improvements += 1
                if iteration <= 5 or iteration % 15 == 0:
                    print(
                        f"      n={n} iter={iteration} score={best_score:.6f} "
                        f"t={t_best:.4f} s_ratio={s_ratio:.4f} "
                        f"dt={time.monotonic()-t0:.2f}s",
                        flush=True,
                    )
        else:
            stall += 1
            if stall <= 5:
                f = best_f.copy()
            elif stall <= 10:
                f = mix_noisy(best_f, rng, 0.15)
            else:
                eps = min(0.8, 0.2 + 0.05 * (stall - 10))
                f = mix_concentrated(best_f, rng, eps)
            if f is None:
                f = best_f.copy()
            if stall % 5 == 0:
                print(
                    f"      n={n} iter={iteration} stalled at {stall}, "
                    f"best={best_score:.6f}",
                    flush=True,
                )

    return best_f, float(best_score)


def propose_candidate(seed=42, budget_s=1500, **kwargs):
    """
    Search for a low-scoring nonnegative sequence using improved LP-based
    refinement with golden-section step sizes, M-relaxation for diverse
    directions, momentum, and structured initializations.
    """
    rng = np.random.default_rng(int(seed) if seed is not None else 42)
    total_budget = max(1.0, float(budget_s))
    start_time = time.monotonic()
    deadline = start_time + max(1.0, total_budget - 25.0)

    best_f = None
    best_score = float("inf")
    given_f = None

    # Load given sequence
    try:
        raw = np.asarray(height_sequence_1, dtype=np.float64)
        given_f = normalize_sequence(raw)
        if given_f is not None:
            best_f = given_f.copy()
            best_score = score_normalized(given_f, given_f.size)
            print(f"Given: n={given_f.size}, score={best_score:.6f}", flush=True)
    except Exception as e:
        print(f"Could not load given sequence: {e}", flush=True)

    if best_f is None:
        n_def = 1024
        best_f = np.ones(n_def) / n_def
        best_score = 2.0
        print(f"No given sequence. Starting uniform n={n_def}, score=2.0", flush=True)

    # === Phase 1: Quick refinement of given sequence (if available) ===
    if given_f is not None and given_f.size >= 128:
        n0 = given_f.size
        rem = deadline - time.monotonic()
        t1 = min(80.0, rem - 20.0)
        if t1 > 15:
            print(f"\n=== Phase 1: Quick LP on given (n={n0}), t={t1:.0f}s ===", flush=True)
            try:
                tmpl = make_conv_template(n0)
                f_res, s_res = optimize_lp_improved(n0, given_f, t1, rng, tmpl)
                if s_res < best_score:
                    best_score = s_res
                    best_f = f_res.copy()
                    print(f"  Phase 1: score={best_score:.6f}", flush=True)
                del tmpl
            except Exception as e:
                print(f"  Phase 1 error: {e}", flush=True)

    # === Main search phases ===
    # Focus on n=1024 (known best) and try 1536, 2048 for more DOF
    n_candidates = [1024, 1024, 1536, 2048]
    time_per_phase = [350.0, 250.0, 250.0, 250.0]

    for phase_idx, (n, alloc) in enumerate(zip(n_candidates, time_per_phase)):
        rem = deadline - time.monotonic()
        if rem < 40.0:
            break
        alloc = min(alloc, rem - 15.0)
        if alloc < 30.0:
            continue

        print(f"\n=== Phase {phase_idx + 2}: n={n}, budget={alloc:.0f}s, best={best_score:.6f} ===", flush=True)

        try:
            template = make_conv_template(n)
        except MemoryError:
            print(f"  MemoryError for n={n}, skipping", flush=True)
            continue
        except Exception as e:
            print(f"  Template error n={n}: {e}", flush=True)
            continue

        # Build diverse starts
        starts = []
        start_names = []

        # S1: Interpolated best (primary)
        if best_f is not None:
            s1 = interpolate_sequence(best_f, n)
            if s1 is not None and s1.size == n:
                starts.append(s1)
                start_names.append("interp_best")

        # S2: Slightly perturbed best
        if best_f is not None:
            s2 = mix_noisy(best_f, rng, 0.08)
            if s2 is not None and s2.size == n:
                s2n = interpolate_sequence(s2, n)
                if s2n is not None:
                    starts.append(s2n)
                    start_names.append("perturbed_best")

        # S3: Legendre
        try:
            s3 = legendre_start(n, rng)
            if s3 is not None:
                starts.append(s3)
                start_names.append("legendre")
        except Exception:
            pass

        # S4: Periodic
        try:
            s4 = periodic_start(n, rng)
            if s4 is not None:
                starts.append(s4)
                start_names.append("periodic")
        except Exception:
            pass

        # S5: Banded
        try:
            s5 = banded_start(n, rng)
            if s5 is not None:
                starts.append(s5)
                start_names.append("banded")
        except Exception:
            pass

        # S6: Gaussian
        try:
            s6 = gaussian_start(n, rng)
            if s6 is not None:
                starts.append(s6)
                start_names.append("gaussian")
        except Exception:
            pass

        # Deduplicate (remove starts too similar to each other)
        unique_starts = []
        unique_names = []
        for s, name in zip(starts, start_names):
            if s is None or s.size != n:
                continue
            is_dup = False
            for us in unique_starts:
                if float(np.linalg.norm(s - us)) < 0.01:
                    is_dup = True
                    break
            if not is_dup:
                unique_starts.append(s)
                unique_names.append(name)

        if not unique_starts:
            unique_starts = [np.ones(n) / n]
            unique_names = ["uniform"]

        # Time allocation: first start gets most time
        m = len(unique_starts)
        if m == 1:
            weights = [1.0]
        elif m == 2:
            weights = [0.55, 0.45]
        elif m == 3:
            weights = [0.45, 0.30, 0.25]
        elif m == 4:
            weights = [0.40, 0.25, 0.20, 0.15]
        else:
            w0 = 0.40
            rest = (1.0 - w0) / (m - 1)
            weights = [w0] + [rest] * (m - 1)

        for idx, (init, sname) in enumerate(zip(unique_starts, unique_names)):
            rem = deadline - time.monotonic()
            if rem < 15.0:
                break
            tb = alloc * weights[idx]
            tb = min(tb, rem - 5.0)
            if tb < 15.0:
                continue

            print(f"  [{sname}] budget={tb:.0f}s", flush=True)
            t_start = time.monotonic()
            f_res, s_res = optimize_lp_improved(n, init, tb, rng, template)
            elapsed = time.monotonic() - t_start

            if np.isfinite(s_res) and s_res < best_score:
                best_score = float(s_res)
                best_f = np.asarray(f_res, dtype=np.float64).copy()
                print(f"  ** NEW BEST: n={best_f.size}, score={best_score:.6f} ** [{sname}, {elapsed:.0f}s]", flush=True)
            else:
                print(f"  [{sname}] best={s_res:.6f} (no improvement, {elapsed:.0f}s)", flush=True)

        try:
            del template
        except Exception:
            pass

    # === Final polish ===
    if best_f is not None:
        rem = deadline - time.monotonic()
        if rem > 20:
            n_best = best_f.size
            print(f"\n=== Final polish: n={n_best}, t={rem - 5:.0f}s ===", flush=True)
            try:
                tmpl = make_conv_template(n_best)
                f_res, s_res = optimize_lp_improved(n_best, best_f, rem - 10, rng, tmpl)
                if s_res < best_score:
                    best_score = s_res
                    best_f = f_res.copy()
                    print(f"  Polish result: score={best_score:.6f}", flush=True)
                del tmpl
            except Exception as e:
                print(f"  Polish error: {e}", flush=True)

    # Build final output
    if best_f is None:
        best_f = np.ones(1024) / 1024.0

    seq = [float(x) for x in best_f]

    # Safety check with evaluate_sequence
    try:
        final_score = float(evaluate_sequence(seq))
    except Exception:
        final_score = float("inf")

    if not np.isfinite(final_score):
        # Fallback
        seq = [1.0 / 1024.0] * 1024
        try:
            final_score = float(evaluate_sequence(seq))
        except Exception:
            final_score = 2.0

    print(f"\nFINAL: n={len(seq)}, internal_score={best_score:.6f}, evaluate={final_score:.6f}", flush=True)
    return seq
