Optimize non-negative step function to minimize 2*n*max(conv(a,a))/sum(a)^2.
Uses log-sum-exp mirror descent (multiplicative weights) with adaptive parameters,
multiple restarts, and LP polishing.

import time
import numpy as np
from scipy.optimize import linprog

def local_score(a):
    a = np.asarray(a, dtype=np.float64)
    n = len(a)
    if n == 0:
        return float('inf')
    a = np.clip(a, 0.0, 1000.0)
    s = np.sum(a)
    if s < 1e-300:
        return float('inf')
    c = np.convolve(a, a)
    m = np.max(c)
    return float(2.0 * n * m / (s * s))

def normalize(a):
    a = np.clip(a, 0.0, 1000.0)
    s = np.sum(a)
    if s < 1e-300:
        return None
    return a / s

def lse_mirror_descent(a_init, max_time, seed, tau=100.0, lr_scale=0.5, momentum=0.8):
    """
    Log-sum-exp mirror descent with multiplicative weight updates.
    """
    a = normalize(a_init.copy())
    if a is None:
        return a_init, local_score(a_init)
    
    n = len(a)
    best = a.copy()
    best_score = local_score(a)
    
    v = np.zeros(n, dtype=np.float64)  # momentum
    lr = lr_scale * n
    rng = np.random.default_rng(seed)
    
    t0 = time.monotonic()
    deadline = t0 + max_time
    iteration = 0
    stall = 0
    current_score = best_score
    
    while time.monotonic() < deadline:
        # Compute convolution
        C = np.convolve(a, a)
        
        # Compute softmax weights
        tauC = tau * C
        tauC_max = np.max(tauC)
        exp_vals = np.exp(tauC - tauC_max)
        p = exp_vals / np.sum(exp_vals)
        
        # Compute gradient
        grad = 2.0 * np.correlate(p, a, mode='valid')
        
        # Update momentum
        v = momentum * v + (1.0 - momentum) * grad
        
        # Multiplicative update
        a_new = a * np.exp(-lr * v)
        a_new = np.clip(a_new, 0.0, 1000.0)
        a_new = normalize(a_new)
        if a_new is None:
            a_new = a.copy()
        
        # Evaluate
        sc = local_score(a_new)
        
        if sc < best_score:
            best = a_new.copy()
            best_score = sc
            stall = 0
            # Slightly increase lr
            lr = min(lr * 1.02, 5.0 * n)
        else:
            stall += 1
            # Decrease lr if stuck

                lr *= 0.9
            if stall > 100:

                a = best * np.exp(rng.normal(0.0, 0.3, n))
                a = normalize(a)
                if a is None:
                    a = best.copy()
                v = np.zeros(n, dtype=np.float64)
                lr = lr_scale * n
                tau *= 0.7  # Spread out attention
                stall = 0
        
        a = a_new
        iteration += 1
        
        # Adapt tau
        if iteration % 500 == 0:
            tau = max(10.0, tau * 0.95)
    
    return best, best_score
