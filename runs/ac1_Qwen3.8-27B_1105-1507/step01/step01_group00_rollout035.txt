def _softmax(x):
    x = x - np.max(x)
    e = np.exp(x)
    s = np.sum(e)
    if s < 1e-300:
        return np.ones(len(x)) / len(x)
    return e / s

    """
    n = len(b)

def propose_candidate(budget_s=1500, **kwargs):
    """Main search function. Returns best sequence found within time budget."""
    t0 = time.monotonic()
    deadline = t0 + max(10.0, float(budget_s) - 20.0)

    rng = np.random.default_rng(42)

    best_score = float('inf')
    best_seq = None

    # Load parent
    parent = None
    n_parent = 512
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64).copy()
        parent = np.clip(parent, 0.0, 1000.0)
        if parent.ndim != 1 or parent.size == 0 or not np.all(np.isfinite(parent)):
            parent = None
        else:
            n_parent = len(parent)
            score = evaluate_sequence(parent.tolist())
            if np.isfinite(score):
                best_score = score
                best_seq = parent.copy()
                print(f"Parent: n={n_parent}, score={score:.6f}")
    except (NameError, Exception):
        pass

    if parent is None:
        print("No valid parent, using defaults")
        parent = np.ones(n_parent)
        best_score = evaluate_sequence(parent.tolist())
        best_seq = parent.copy()
        print(f"Default uniform n={n_parent}: score={best_score:.6f}")

    # Lengths to try, prioritizing the parent length
    all_lengths = sorted(set([n_parent, 256, 512, 768, 1024, 1536, 2048]))
    # Reorder: parent first, then nearby lengths
    def dist_to_parent(l):
        return abs(l - n_parent)
    all_lengths.sort(key=dist_to_parent)

    total_time = deadline - t0
    # Allocate time: 40% to parent length, 30% to next best, 30% to others
    n_lengths = len(all_lengths)
    time_alloc = [0.0] * n_lengths
    time_alloc[0] = total_time * 0.40
    if n_lengths > 1:
        time_alloc[1] = total_time * 0.25
    if n_lengths > 2:
        remaining = total_time * 0.35 / (n_lengths - 2)
        for i in range(2, n_lengths):
            time_alloc[i] = remaining

    for li, n in enumerate(all_lengths):
        if time.monotonic() > deadline - 5:
            break

        alloc = time_alloc[li]
        sub_deadline = min(deadline, time.monotonic() + alloc)

        print(f"\n--- n={n} (budget {alloc:.0f}s) ---")

        inits = _generate_initializations(n, rng, parent)

        # Split time among initializations (parent gets more)
        n_inits = len(inits)
        init_times = alloc / n_inits

        for ii, (label, f_init) in enumerate(inits):
            if time.monotonic() > sub_deadline - 2:
                break

            f_init = np.maximum(f_init.astype(np.float64), 0.0)

            # Evaluate initial
            s_init = evaluate_sequence(f_init.tolist())
            if not np.isfinite(s_init):
                continue

            if s_init < best_score:
                best_score = s_init
                best_seq = f_init.copy()
                print(f"  {label}: init={best_score:.6f} **")

            # Time for this initialization
            init_deadline = min(sub_deadline, time.monotonic() + init_times)
            if time.monotonic() > init_deadline - 1:
                continue

            # Optimize
            optimized, opt_score = _optimize_subgrad(
                f_init, init_deadline, verbose=True, label=f"[n={n} {label}]"
            )

            if opt_score < best_score:
                best_score = opt_score
                best_seq = optimized.copy()
                print(f"  {label}: -> {best_score:.6f} **")

    # Final polish on best sequence with remaining time
    if best_seq is not None and time.monotonic() < deadline - 3:
        remaining_time = deadline - time.monotonic()
        if remaining_time > 10:
            print(f"\n--- Final polish (n={len(best_seq)}, {remaining_time:.0f}s) ---")
            polished, pol_score = _optimize_subgrad(
                best_seq, deadline, verbose=True, label="[polish]"
            )
            if pol_score < best_score:
                best_score = pol_score
                best_seq = polished.copy()
                print(f"  Polish result: {best_score:.6f}")

    elapsed = time.monotonic() - t0
    if best_seq is not None:
        print(f"\nFINAL: n={len(best_seq)}, score={best_score:.6f}, time={elapsed:.1f}s")
    else:
        print(f"\nNo valid result found, time={elapsed:.1f}s")

    if best_seq is None or not np.isfinite(best_score):
        return [1.0] * 128

    result = np.clip(best_seq, 0.0, 1000.0)
    result = result[np.isfinite(result)]
    if len(result) == 0:
        return [1.0] * 128
    if np.sum(result) < 0.01:
        result = result * (1.0 / np.sum(result))

    return [float(x) for x in result]
