```python
def build_conv_matrix(f):
    """Build matrix A such that A @ b = np.convolve(f, b, mode='full')."""
    n = len(f)
    A = np.zeros((2 * n - 1, n))
    for k in range(2 * n - 1):
        j_start = max(0, k - n + 1)
        j_end = min(n, k + 1)
        for j in range(j_start, j_end):
            A[k, j] = f[k - j]
    return A


def lp_refine(a, time_budget=100, K=10, verbose=False):
    """
    LP-based local refinement.
    a: sequence with sum=1.
    K: number of active peaks to include.
    """
    n = len(a)
    a = a.astype(np.float64).copy()
    a = project(a)
    
    best_a = a.copy()
    best_score = compute_score_normalized(a)
    
    start_time = time.monotonic()
    it = 0
    t_lr = 0.5  # learning rate for the LP direction
    
    while time.monotonic() - start_time < time_budget:
        r = np.convolve(a, a)
        M = r.max()
        
        # Select active set: top K peaks
        top_k = np.argsort(r)[-K:]
        
        # Build LP: maximize sum(b) subject to (a*b)[k] <= M for k in top_k, b >= 0
        A_active = np.zeros((K, n))
        for i, k in enumerate(top_k):
            j_start = max(0, k - n + 1)
            j_end = min(n, k + 1)
            for j in range(j_start, j_end):
                A_active[i, j] = a[k - j]
        
        c_obj = -np.ones(n)
        b_ub = np.full(K, M)
        bounds = [(0, None)] * n
        
        try:
            res = linprog(c_obj, A_ub=A_active, b_ub=b_ub, bounds=bounds,
                         method='highs', options={'presolve': True})
        except Exception:
            break
        
        if not res.success or res.x is None:
            break
        
        b = np.clip(res.x, 0, None)
        b_sum = b.sum()
        if b_sum < 1e-15:
            break
        
        g = b / b_sum
        
        # Line search for best step size
        best_t = 0
        best_score_ls = compute_score_normalized(a)
        for t_candidate in [t_lr, t_lr*0.5, t_lr*0.25, t_lr*0.1, t_lr*2, t_lr*0.75]:
            a_try = (1 - t_candidate) * a + t_candidate * g
            a_try = project(a_try)
            s_try = compute_score_normalized(a_try)
            if s_try < best_score_ls:
                best_score_ls = s_try
                best_t = t_candidate
        
        if best_t > 0:
            a = (1 - best_t) * a + best_t * g
            a = project(a)
            new_score = compute_score_normalized(a)
            if new_score < best_score:
                best_a = a.copy()
                best_score = new_score
                t_lr = min(best_t * 1.2, 1.0)
            else:
                t_lr *= 0.7
        else:
            t_lr *= 0.5
            if t_lr < 1e-4:
                break
        
        it += 1
        if verbose and it % 20 == 0:
            elapsed = time.monotonic() - start_time
            print(f"    LP it={it}, score={best_score:.6f}, t={t_lr:.4f}, elapsed={elapsed:.1f}s")
    
    return best_a, best_score
```

```python
"""
Combined optimization for minimizing 2*n*max(convolve(a,a))/sum(a)^2.
Uses projected gradient with differentiable soft-peak, LP-based active set
refinement with line search, and coordinate descent. Multiple diverse
starting points with different window shapes and lengths are explored.
"""

import numpy as np
from scipy.optimize import linprog
import time


def compute_score_normalized(a):
    """Compute 2*n*max(conv) for sum(a)=1."""
    n = len(a)
    conv = np.convolve(a, a)
    return float(2.0 * n * conv.max())


def project_simplex(a):
    """Project onto {a >= 0, sum(a) = 1}."""
    a = np.maximum(a, 0.0)
    S = a.sum()
    if S < 1e-15:
        return np.ones(len(a)) / len(a)
    return a / S


def softmax_weights(r, T):
    """Stable softmax weights."""
    r_shifted = r - r.max()
    w = np.exp(T * r_shifted)
    s = w.sum()
    if s < 1e-300:
        return np.ones(len(r)) / len(r)
    return w / s


def gradient_soft_peak(a, T):
    """Gradient of 2n*(1/T)*log(sum exp(T*r_k)) w.r.t. a."""
    n = len(a)
    r = np.convolve(a, a)
    w = softmax_weights(r, T)
    # grad[m] = 4n * sum_k w[k] * a[k-m] (valid indices)
    # = 4n * np.convolve(w, a_flipped, 'full')[n-1 : 2n-1]
    a_flip = a[::-1]
    conv_full = np.convolve(w, a_flip, 'full')
    grad = 4.0 * n * conv_full[n - 1: 2 * n - 1]
    return grad


def projected_gradient(a, T_init=5.0, T_end=200.0, time_budget=60.0,
                       step_init=0.05, verbose=False):
    """Projected gradient descent with animated temperature."""
    a = project_simplex(a.astype(np.float64).copy())
    n = len(a)
    best_a = a.copy()
    best_score = compute_score_normalized(a)
    step = step_init
    start = time.monotonic()
    n_steps = 5000
    stall = 0

    for it in range(n_steps):
        if time.monotonic() - start > time_budget:
            break
        progress = min(1.0, it / n_steps)
        T = T_init * (T_end / T_init) ** progress

        grad = gradient_soft_peak(a, T)

        # Backtracking line search
        a_try = a
        s_try = step
        score_try = compute_score_normalized(a_try)
        accepted = False
        for _ in range(12):
            a_try = project_simplex(a - s_try * grad)
            score_try = compute_score_normalized(a_try)
            if score_try < best_score:
                accepted = True
                break
            s_try *= 0.5

        if accepted:
            a = a_try
            if score_try < best_score:
                best_a = a.copy()
                best_score = score_try
                stall = 0
                step = min(s_try * 1.05, 0.3)
            else:
                step = s_try
        else:
            stall += 1
            step *= 0.7
            if stall > 200:
                # Random kick
                a = project_simplex(a + 0.05 * np.random.randn(n))
                best_score = compute_score_normalized(a)
                if best_score < compute_score_normalized(best_a):
                    best_a = a.copy()
                step = step_init * 0.5
                stall = 0

        if verbose and it % 500 == 0:
            print(f"    PGD it={it}, T={T:.1f}, step={step:.6f}, score={best_score:.6f}, elapsed={time.monotonic()-start:.1f}s")

    return best_a, best_score


def build_conv_matrix_rows(f, row_indices):
    """Build rows of convolution matrix for given lags."""
    n = len(f)
    rows = []
    for k in row_indices:
        row = np.zeros(n)
        j_start = max(0, k - n + 1)
        j_end = min(n, k + 1)
        for j in range(j_start, j_end):
            row[j] = f[k - j]
        rows.append(row)
    return np.array(rows)


def lp_refine(a, time_budget=120.0, K=10, verbose=False):
    """LP-based local refinement with adaptive active set and line search."""
    n = len(a)
    a = project_simplex(a.astype(np.float64).copy())
    best_a = a.copy()
    best_score = compute_score_normalized(a)
    start = time.monotonic()
    t_lr = 0.4
    it = 0
    stall = 0

    while time.monotonic() - start < time_budget:
        r = np.convolve(a, a)
        M = r.max()

        # Active set: top K peaks
        top_k = np.argsort(r)[-K:]

        A_active = build_conv_matrix_rows(a, top_k)
        c_obj = -np.ones(n)
        b_ub = np.full(K, M)
        bounds = [(0, None)] * n

        try:
            res = linprog(c_obj, A_ub=A_active, b_ub=b_ub, bounds=bounds,
                         method='highs', options={'presolve': True, 'time_limit': max(1, time_budget - (time.monotonic() - start))})
        except Exception:
            break

        if not res.success or res.x is None:
            break

        b = np.maximum(res.x, 0.0)
        b_sum = b.sum()
        if b_sum < 1e-15:
            break
        g = b / b_sum

        # Line search
        best_t = 0.0
        best_s_ls = best_score
        for t_cand in [t_lr, t_lr * 0.5, t_lr * 0.25, t_lr * 2.0, t_lr * 0.75, t_lr * 1.5]:
            t_cand = min(max(t_cand, 0.01), 0.99)
            a_try = project_simplex((1 - t_cand) * a + t_cand * g)
            s_try = compute_score_normalized(a_try)
            if s_try < best_s_ls:
                best_s_ls = s_try
                best_t = t_cand

        if best_t > 0:
            a = project_simplex((1 - best_t) * a + best_t * g)
            new_score = compute_score_normalized(a)
            if new_score < best_score:
                best_a = a.copy()
                best_score = new_score
                stall = 0
                t_lr = min(best_t * 1.1, 0.8)
            else:
                stall += 1
                t_lr *= 0.8
        else:
            stall += 1
            t_lr *= 0.6

        if stall > 40:
            # Perturb to escape
            a = project_simplex(a + 0.02 * np.random.randn(n))
            if compute_score_normalized(a) < best_score:
                best_a = a.copy()
                best_score = compute_score_normalized(a)
            t_lr = 0.3
            stall = 0

        it += 1
        if verbose and it % 30 == 0:
            print(f"    LP it={it}, score={best_score:.6f}, t={t_lr:.4f}, elapsed={time.monotonic()-start:.1f}s")

    return best_a, best_score


def coordinate_descent(a, time_budget=60.0, verbose=False):
    """Coordinate descent for final polish."""
    n = len(a)
    a = project_simplex(a.astype(np.float64).copy())
    best_a = a.copy()
    best_score = compute_score_normalized(a)
    start = time.monotonic()
    scales = [0.3, 0.5, 0.7, 0.85, 0.9, 1.1, 1.3, 1.5, 2.0, 3.0]
    sweep = 0

    while time.monotonic() - start < time_budget:
        improved = False
        sweep += 1
        for j in range(n):
            if time.monotonic() - start > time_budget:
                break
            saved = a[j]
            best_local = saved
            best_local_s = best_score
            for s in scales:
                a[j] = max(0.0, saved * s)
                S = a.sum()
                if S < 1e-15:
                    a[j] = saved
                    break
                a_try = a / S
                sc = compute_score_normalized(a_try)
                if sc < best_local_s:
                    best_local_s = sc
                    best_local = a_try.copy()
            if best_local is not saved:
                a = best_local
                if best_local_s < best_score:
                    best_score = best_local_s
                    best_a = a.copy()
                    improved = True
            else:
                a[j] = saved
        if not improved:
            break
        if verbose and sweep % 5 == 0:
            print(f"    CD sweep={sweep}, score={best_score:.6f}, elapsed={time.monotonic()-start:.1f}s")

    return best_a, best_score


def make_starting_points(n, rng):
    """Generate diverse starting sequences of length n."""
    points = []

    # Uniform
    points.append(np.ones(n))

    # Raised cosine
    j = np.arange(n)
    rc = 0.5 * (1 + np.cos(2 * np.pi * j / (n - 1)))
    points.append(rc)

    # Hamming window
    points.append(0.54 - 0.46 * np.cos(2 * np.pi * j / (n - 1)))

    # Blackman window
    points.append(0.42 - 0.5 * np.cos(2 * np.pi * j / (n - 1)) + 0.08 * np.cos(4 * np.pi * j / (n - 1)))

    # Triangle (tent)
    mid = n // 2
    tri = 1.0 - np.abs(j - mid) / (mid + 1)
    points.append(tri)

    # Sine bump
    points.append(np.sin(np.pi * j / (n - 1)) ** 2 + 0.01)

    # Gaussian center
    sigma = n / 4.0
    gauss = np.exp(-0.5 * ((j - mid) / sigma) ** 2)
    points.append(gauss)

    # Two bumps
    bump1 = np.exp(-0.5 * ((j - n * 0.25) / (n * 0.1)) ** 2)
    bump2 = np.exp(-0.5 * ((j - n * 0.75) / (n * 0.1)) ** 2)
    points.append(bump1 + bump2 + 0.01)

    # Cosine taper (different)
    points.append(0.3 + 0.7 * np.cos(np.pi * j / (n - 1)) ** 2)

    # Random with structure
    for _ in range(3):
        freq = rng.uniform(0.5, 5.0)
        phase = rng.uniform(0, 2 * np.pi)
        structured = 0.5 + 0.5 * np.cos(freq * 2 * np.pi * j / n + phase)
        noisy = structured * (1 + 0.1 * rng.standard_normal(n))
        points.append(np.maximum(noisy, 0.01))

    # Pure random
    for _ in range(2):
        points.append(np.abs(rng.standard_normal(n)) + 0.01)

    return [project_simplex(p.astype(np.float64)) for p in points]


def propose_candidate(seed=42, budget_s=1500, **kwargs):
    """Main search: multi-start with PGD + LP + coordinate descent."""
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + budget_s - 30.0

    # Get parent if available
    parent = None
    try:
        parent = np.array(height_sequence_1, dtype=np.float64)
        if parent.ndim != 1 or len(parent) == 0 or not np.isfinite(parent).all():
            parent = None
    except Exception:
        parent = None
    if parent is not None:
        parent = project_simplex(parent)

    best_overall = None
    best_score_overall = np.inf

    # Choose lengths to try
    lengths_to_try = [256, 512, 1024]
    if parent is not None and 128 <= len(parent) <= 2048:
        parent_n = len(parent)
        if parent_n not in lengths_to_try:
            lengths_to_try.insert(0, parent_n)

    print(f"Lengths to try: {lengths_to_try}")
    print(f"Parent available: {parent is not None}, len={len(parent) if parent is not None else 0}")
    print(f"Target: 1.5027, Previous best: 1.515592")
    print(f"Total budget: {budget_s}s")

    # Allocate time per length
    time_per_length = (budget_s - 90) / len(lengths_to_try)

    for n in lengths_to_try:
        if time.monotonic() > deadline:
            break
        remaining = deadline - time.monotonic()
        t_alloc = min(time_per_length, remaining - 30)
        if t_alloc < 30:
            break

        print(f"\n{'='*60}")
        print(f"Length n={n}, time allocation={t_alloc:.0f}s")
        print(f"{'='*60}")

        # Generate starting points
        starts = make_starting_points(n, rng)
        if parent is not None and len(parent) == n:
            starts.append(parent.copy())
            # Also add perturbed parent
            starts.append(project_simplex(parent * (1 + 0.1 * rng.standard_normal(n))))

        n_starts = len(starts)
        print(f"  {n_starts} starting points")

        # Phase 1: PGD on each start
        pgd_time = t_alloc * 0.35
        pgd_per_start = max(5.0, pgd_time / n_starts)
        pgd_results = []

        for i, s in enumerate(starts):
            t_left = deadline - time.monotonic() - 20
            if t_left < 5:
                break
            t_this = min(pgd_per_start, t_left)
            if t_this < 3:
                break
            a_opt, sc = projected_gradient(s, T_init=5.0, T_end=150.0,
                                           time_budget=t_this, step_init=0.05, verbose=False)
            pgd_results.append((sc, a_opt))
            if sc < best_score_overall:
                best_score_overall = sc
                best_overall = a_opt.copy()
                print(f"    Start {i}: PGD score={sc:.6f} ** NEW BEST **")
            else:
                print(f"    Start {i}: PGD score={sc:.6f}")

        # Sort by score, keep top candidates
        pgd_results.sort(key=lambda x: x[0])
        n_lp = min(len(pgd_results), max(3, int(t_alloc * 0.4 / 60)))
        lp_time_each = max(30.0, t_alloc * 0.4 / max(1, n_lp))

        # Phase 2: LP refinement on top candidates
        lp_results = []
        for i in range(n_lp):
            t_left = deadline - time.monotonic() - 20
            if t_left < 10:
                break
            t_this = min(lp_time_each, t_left)
            if t_this < 10:
                break
            sc_init, a_init = pgd_results[i]
            a_opt, sc = lp_refine(a_init, time_budget=t_this, K=10, verbose=(i == 0))
            lp_results.append((sc, a_opt))
            if sc < best_score_overall:
                best_score_overall = sc
                best_overall = a_opt.copy()
                print(f"    LP [{i}]: score={sc:.6f} ** NEW BEST **")
            else:
                print(f"    LP [{i}]: score={sc:.6f}")

        # Phase 3: Coordinate descent on best LP result
        if lp_results:
            t_left = deadline - time.monotonic() - 10
            if t_left > 30:
                sc_init, a_init = lp_results[0]
                cd_time = min(120.0, t_left - 10)
                a_opt, sc = coordinate_descent(a_init, time_budget=cd_time, verbose=True)
                if sc < best_score_overall:
                    best_score_overall = sc
                    best_overall = a_opt.copy()
                    print(f"    CD: score={sc:.6f} ** NEW BEST **")
                else:
                    print(f"    CD: score={sc:.6f}")

                # Phase 4: Another LP pass after CD
                t_left = deadline - time.monotonic() - 10
                if t_left > 30:
                    lp_time2 = min(90.0, t_left - 10)
                    a_opt2, sc2 = lp_refine(a_opt, time_budget=lp_time2, K=15, verbose=True)
                    if sc2 < best_score_overall:
                        best_score_overall = sc2
                        best_overall = a_opt2.copy()
                        print(f"    LP2: score={sc2:.6f} ** NEW BEST **")

        print(f"  Best for n={n}: {best_score_overall:.6f}")

    # Final: try the overall best with extra time
    if best_overall is not None:
        t_left = deadline - time.monotonic()
        if t_left > 20:
            print(f"\nFinal polish on best (n={len(best_overall)}), time={t_left:.0f}s")
            a, sc = lp_refine(best_overall, time_budget=min(60, t_left * 0.4), K=12, verbose=True)
            if sc < best_score_overall:
                best_score_overall = sc
                best_overall = a
            t_left2 = deadline - time.monotonic()
            if t_left2 > 20:
                a, sc = coordinate_descent(best_overall, time_budget=min(60, t_left2 * 0.5), verbose=True)
                if sc < best_score_overall:
                    best_score_overall = sc
                    best_overall = a
            t_left3 = deadline - time.monotonic()
            if t_left3 > 15:
                a, sc = lp_refine(best_overall, time_budget=min(45, t_left3), K=20, verbose=False)
                if sc < best_score_overall:
                    best_score_overall = sc
                    best_overall = a

    # Verify and return
    if best_overall is not None:
        best_overall = np.clip(best_overall, 0, 1000)
        final_list = best_overall.tolist()
        final_score = evaluate_sequence(final_list)
        print(f"\n{'='*60}")
        print(f"FINAL SCORE: {final_score:.6f}")
        print(f"Sequence length: {len(final_list)}")
        print(f"{'='*60}")
        return final_list

    # Fallback
    print("Fallback")
    return [1.0] * 256
```
