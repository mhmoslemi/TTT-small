"""
Hybrid optimization for minimizing 2*n*max(conv(a,a))/(sum(a))^2.
Uses mirror descent with soft-max gradient, LP-based polishing,
and local search. Tries multiple sequence lengths and initializations.
"""

import numpy as np
from scipy.optimize import linprog
import time

def compute_obj(p):
    """Compute 2n * max(conv(p,p)) assuming sum(p)=1, p>=0."""
    n = len(p)
    if n == 0:
        return np.inf
    conv = np.convolve(p, p)
    return 2.0 * n * float(np.max(conv))

def normalize(p):
    """Normalize to sum=1, clip to non-negative."""
    p = np.clip(p, 0.0, None)
    s = p.sum()
    if s < 1e-15:
        p = np.ones(len(p))
        s = p.sum()
    return p / s

def init_cosine(n, rng):
    x = np.arange(n, dtype=np.float64)
    a = np.cos(np.pi * x / n) ** 2
    a = np.clip(a, 0, None)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n)
        s = n
    return a / s

def init_unif_noise(n, rng, sigma=0.5):
    a = 1.0 + sigma * rng.standard_normal(n)
    a = np.clip(a, 0, None)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n)
        s = n
    return a / s

def init_two_tone(n, rng):
    """Alternating high/low pattern with noise."""
    freq = rng.uniform(0.5, 5.0)
    x = np.arange(n)
    a = 1.0 + 0.7 * np.cos(2 * np.pi * freq * x / n)
    a = np.clip(a, 0, None)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n)
        s = n
    return a / s

def init_parent(n, parent):
    """Resample parent to length n."""
    if len(parent) == n:
        return normalize(parent.copy())
    # Linear interpolation
    x_old = np.linspace(0, 1, len(parent))
    x_new = np.linspace(0, 1, n)
    a = np.interp(x_new, x_old, parent)
    return normalize(a)

def soft_gradient(p, beta=100):
    """Compute soft-max gradient of max(conv(p,p))."""
    n = len(p)
    conv = np.convolve(p, p)
    max_c = conv.max()
    w = np.exp(beta * (conv - max_c))
    w /= w.sum()
    
    # g[j] = sum_k w[k] * (p[k-j] + p[j]) for valid k,j
    # Part 1: sum_k w[k] * p[k-j]
    # Part 2: p[j] * sum_k w[k] (for valid k)
    
    g = np.zeros(n)
    
    # Part 1: cross-correlation
    # g1[j] = sum_{m: 0<=m<n, 0<=m+j<2n-1} p[m] * w[m+j]
    for j in range(n):
        m_max = min(n - 1, 2 * n - 2 - j)
        if m_max >= 0:
            g[j] += np.dot(p[:m_max+1], w[j:j+m_max+1])
    
    # Part 2: p[j] * (sum of w[k] for k in [j, j+n-1] ∩ [0, 2n-2])
    w_cs = np.zeros(2 * n)
    w_cs[1:] = np.cumsum(w)
    for j in range(n):
        k_lo = j
        k_hi = min(2 * n - 1, j + n)
        g[j] += p[j] * (w_cs[k_hi] - w_cs[k_lo])
    
    return g, conv

def mirror_descent(p, n_iter=5000, beta_start=20, beta_end=200, 
                   eta_start=0.3, eta_end=0.01, verbose=False):
    """Run mirror descent to minimize max(conv(p,p))."""
    n = len(p)
    p = normalize(p.copy())
    best_p = p.copy()
    best_obj = compute_obj(p)
    
    for it in range(n_iter):
        frac = it / max(1, n_iter - 1)
        beta = beta_start + (beta_end - beta_start) * frac
        eta = eta_start + (eta_end - eta_start) * frac
        
        g, conv = soft_gradient(p, beta)
        p_new = p * np.exp(-eta * g / (g.max() + 1e-15))
        p_new = normalize(p_new)
        
        obj = compute_obj(p_new)
        if obj < best_obj:
            best_p = p_new.copy()
            best_obj = obj
        
        p = p_new
    
    return best_p, best_obj

def build_A(p):
    """Build convolution matrix A where A[k,j] = p[k-j] if valid."""
    n = len(p)
    A = np.zeros((2 * n - 1, n), dtype=np.float64)
    for k in range(2 * n - 1):
        j_lo = max(0, k - n + 1)
        j_hi = min(n - 1, k)
        A[k, j_lo:j_hi+1] = p[k - j_hi : k - j_lo + 1][::-1]
    return A

def lp_alternate(p, max_iter=50, t=0.5, verbose=False):
    """LP-based alternating optimization."""
    n = len(p)
    p = normalize(p.copy())
    best_p = p.copy()
    best_obj = compute_obj(p)
    
    for it in range(max_iter):
        conv = np.convolve(p, p)
        M = float(conv.max())
        A = build_A(p)
        
        # LP: max sum(q) s.t. A @ q <= M, q >= 0
        c = -np.ones(n)
        res = linprog(c, A_ub=A, b_ub=np.full(2*n-1, M), 
                     bounds=[(0, None)]*n, method='highs')
        if res.status != 0:
            break
        
        q = res.x
        q_sum = -res.fun
        if q_sum < 1e-10:
            break
        
        q = q / q_sum
        obj_q = compute_obj(q)
        
        # Blend
        p_new = (1 - t) * p + t * q
        p_new = normalize(p_new)
        obj_new = compute_obj(p_new)
        
        if obj_new < best_obj:
            best_p = p_new.copy()
            best_obj = obj_new
        
        p = p_new
        
        if verbose and it % 10 == 0:
            print(f"    LP it={it}: obj={best_obj:.6f} q_sum={q_sum:.4f}")
    
    return best_p, best_obj

def local_search(p, rng, n_iter=10000, base_scale=0.1):
    """Random perturbation local search."""
    n = len(p)
    p = normalize(p.copy())
    best_p = p.copy()
    best_obj = compute_obj(p)
    
    for it in range(n_iter):
        scale = base_scale * (1.0 + 0.5 * np.log(1 + it / 100))
        # Perturb a random block
        width = max(1, n // 30)
        start = int(rng.integers(0, n - width + 1))
        p_try = p.copy()
        p_try[start:start+width] *= np.exp(rng.standard_normal(width) * scale)
        p_try = normalize(p_try)
        
        # Also try single-element perturbations

            idx = int(rng.integers(0, n))
            p_try2 = p.copy()
            p_try2[idx] *= np.exp(rng.standard_normal() * scale * 0.5)
            p_try2 = normalize(p_try2)
            if compute_obj(p_try2) < best_obj:
                best_obj = compute_obj(p_try2)
                best_p = p_try2
                p = p_try2
                continue
        
        obj_try = compute_obj(p_try)
        if obj_try < best_obj:
            best_obj = obj_try
            best_p = p_try.copy()
            p = p_try
    
    return best_p, best_obj

"""
Hybrid optimization for minimizing 2*n*max(conv(a,a))/(sum(a))^2.
Uses mirror descent with soft-max gradient smoothing, LP-based alternating
polishing, and local random search. Tries multiple sequence lengths and
diverse initializations to escape local optima.
"""

import numpy as np
from scipy.optimize import linprog
import time

def compute_obj(p):
    """Compute 2n * max(conv(p,p)) assuming p is normalized (sum=1)."""
    n = len(p)
    if n == 0:
        return np.inf
    conv = np.convolve(p, p)
    return 2.0 * n * float(np.max(conv))

def normalize(p):
    """Clip to non-negative and normalize to sum=1."""
    p = np.clip(p, 0.0, None)
    s = p.sum()
    if s < 1e-15:
        return np.ones(len(p)) / len(p)
    return p / s

def init_cosine(n, rng):
    x = np.arange(n, dtype=np.float64)
    a = np.cos(np.pi * x / n) ** 2
    return normalize(a)

def init_unif_noise(n, rng, sigma=0.5):
    a = 1.0 + sigma * rng.standard_normal(n)
    return normalize(a)

def init_two_tone(n, rng):
    freq = rng.uniform(0.3, 8.0)
    x = np.arange(n)
    a = 1.0 + 0.7 * np.cos(2 * np.pi * freq * x / n)
    return normalize(a)

def init_triangular(n, rng):
    x = np.arange(n, dtype=np.float64)
    center = n / 2.0
    a = np.maximum(0, 1.0 - np.abs(x - center) / center)
    a += 0.1 * rng.standard_normal(n)
    return normalize(a)

def init_split(n, rng):
    a = np.ones(n)
    a[:n//2] = rng.uniform(0.2, 1.0)
    a[n//2:] = rng.uniform(0.2, 1.0)
    return normalize(a)

def init_parent_resampled(n, parent):
    if len(parent) == n:
        return normalize(parent.astype(np.float64).copy())
    x_old = np.linspace(0, 1, len(parent))
    x_new = np.linspace(0, 1, n)
    a = np.interp(x_new, x_old, parent)
    return normalize(a)

def soft_gradient(p, beta=50.0):
    """Compute soft-max weighted gradient of max(conv(p,p)).
    Returns (gradient_vector, convolution)."""
    n = len(p)
    conv = np.convolve(p, p)
    max_c = conv.max()
    w = np.exp(beta * (conv - max_c))
    w_sum = w.sum()
    if w_sum < 1e-30:
        w = np.zeros(len(conv))
        w.argmax()
        w[np.argmax(conv)] = 1.0
    else:
        w /= w_sum

    g = np.zeros(n)
    # Part 1: g1[j] = sum_m p[m] * w[m+j] (cross-correlation)
    for j in range(n):
        m_max = min(n - 1, 2 * n - 2 - j)
        if m_max >= 0:
            g[j] += np.dot(p[:m_max + 1], w[j:j + m_max + 1])

    # Part 2: g2[j] = p[j] * sum_{k in valid range} w[k]
    w_cs = np.zeros(2 * n)
    w_cs[1:] = np.cumsum(w)
    for j in range(n):
        k_hi = min(2 * n - 1, j + n)
        g[j] += p[j] * (w_cs[k_hi] - w_cs[j])

    return g, conv

def mirror_descent(p, n_iter=10000, beta_start=20.0, beta_end=500.0,
                   eta=0.2, verbose=False, time_budget=None, start_time=None):
    """Mirror descent with soft-max smoothing and multiplicative updates."""
    n = len(p)
    p = normalize(p.copy())
    best_p = p.copy()
    best_obj = compute_obj(p)
    patience = 0

    for it in range(n_iter):
        if time_budget is not None and time.monotonic() - start_time > time_budget:
            break

        frac = it / max(1, n_iter - 1)
        beta = beta_start + (beta_end - beta_start) * frac

        g, conv = soft_gradient(p, beta)
        g_max = g.max()
        if g_max < 1e-15:
            break

        p_new = p * np.exp(-eta * g / g_max)
        p_new = normalize(p_new)

        obj = compute_obj(p_new)
        if obj < best_obj:
            best_obj = obj
            best_p = p_new.copy()
            patience = 0
        else:
            patience += 1

        p = p_new

        if verbose and (it % 500 == 0 or it < 5):
            print(f"      MD it={it}: obj={best_obj:.6f} beta={beta:.1f}")

    return best_p, best_obj

def build_A_matrix(p):
    """Build the convolution matrix A[k,j] = p[k-j] for valid indices."""
    n = len(p)
    A = np.zeros((2 * n - 1, n), dtype=np.float64)
    for k in range(2 * n - 1):
        j_lo = max(0, k - n + 1)
        j_hi = min(n - 1, k)
        A[k, j_lo:j_hi + 1] = p[k - j_hi:k - j_lo + 1][::-1]
    return A

def lp_polish(p, max_iter=30, t=0.4, verbose=False):
    """LP-based alternating optimization for final polishing."""
    n = len(p)
    p = normalize(p.copy())
    best_p = p.copy()
    best_obj = compute_obj(p)
    A = build_A_matrix(p)

    for it in range(max_iter):
        conv = np.convolve(p, p)
        M = float(conv.max())

        c = -np.ones(n)
        b_ub = np.full(2 * n - 1, M)
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=[(0.0, None)] * n, method='highs')
        if res.status != 0:
            break

        q = res.x
        q_sum = float(-res.fun)
        if q_sum < 1e-10:
            break

        q = q / q_sum
        p_new = (1.0 - t) * p + t * q
        p_new = normalize(p_new)
        obj_new = compute_obj(p_new)

        if obj_new < best_obj:
            best_p = p_new.copy()
            best_obj = obj_new

        p = p_new

        # Update A matrix for next iteration (since p changed)
        if it < max_iter - 1:
            A = build_A_matrix(p)

        if verbose and it % 5 == 0:
            print(f"      LP it={it}: obj={best_obj:.6f}")

    return best_p, best_obj

def local_search(p, rng, n_iter=20000, verbose=False):
    """Random block perturbation local search."""
    n = len(p)
    p = normalize(p.copy())
    best_p = p.copy()
    best_obj = compute_obj(p)
    accept_count = 0

    for it in range(n_iter):
        # Adaptive scale
        scale = 0.05 * (1.0 + 0.3 * np.log1p(it / 200.0))

        # Choose perturbation type
        r = rng.random()
        if r < 0.4:
            # Block perturbation
            width = max(1, n // 50)
            start = int(rng.integers(0, n - width + 1))
            p_try = p.copy()
            p_try[start:start + width] *= np.exp(rng.standard_normal(width) * scale)
        elif r < 0.7:
            # Single element
            idx = int(rng.integers(0, n))
            p_try = p.copy()
            p_try[idx] *= np.exp(rng.standard_normal() * scale * 2)
        elif r < 0.85:
            # Two-point exchange
            i = int(rng.integers(0, n))
            j = int(rng.integers(0, n))
            if i == j:
                continue
            p_try = p.copy()
            delta = scale * p.try_sum() if False else scale * 0.05
            p_try[i] += delta
            p_try[j] -= delta
        else:
            # Smooth perturbation (low-pass)
            mask = np.ones(n)
            mask[:n//4] = rng.uniform(0.5, 1.5)
            mask[-n//4:] = rng.uniform(0.5, 1.5)
            p_try = p * mask
            p_try = normalize(p_try)
            continue

        p_try = normalize(p_try)
        obj_try = compute_obj(p_try)

        if obj_try < best_obj:
            best_obj = obj_try
            best_p = p_try.copy()
            p = p_try
            accept_count += 1

    return best_p, best_obj

def peak_shaving(p, rng, n_iter=5000):
    """Targeted peak reduction: identify convolution peaks and reduce them."""
    n = len(p)
    p = normalize(p.copy())
    best_p = p.copy()
    best_obj = compute_obj(p)

    for it in range(n_iter):
        conv = np.convolve(p, p)
        max_c = conv.max()
        # Find near-peak locations (top 5%)
        threshold = max_c * 0.95
        peak_indices = np.where(conv >= threshold)[0]

        if len(peak_indices) == 0:
            break

        # Compute subgradient from peak locations
        g = np.zeros(n)
        for k in peak_indices:
            j_lo = max(0, k - n + 1)
            j_hi = min(n - 1, k)
            for j in range(j_lo, j_hi + 1):
                kj = k - j
                if 0 <= kj < n:
                    g[j] += p[kj] + p[j]

        g /= len(peak_indices)

        # Multiplicative reduction
        eta = 0.3 / (g.max() + 1e-15)
        p_try = p * np.exp(-eta * g)
        p_try = normalize(p_try)

        obj_try = compute_obj(p_try)
        if obj_try < best_obj:
            best_obj = obj_try
            best_p = p_try.copy()
            p = p_try

    return best_p, best_obj
