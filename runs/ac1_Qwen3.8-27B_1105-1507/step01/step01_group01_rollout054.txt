"""
Search for low-score non-negative step-function coefficients.

Uses log-sum-exp smoothed gradient descent with FFT-based gradient,
simplex projection, tau annealing, and multiplicative updates.
Explores multiple lengths and diverse initializations.
"""

import time
import numpy as np
from scipy.fft import rfft, irfft

try:
    from scipy.optimize import linprog
    HAVE_SCIPY = True
except Exception:
    HAVE_SCIPY = False


def normalize_to_sum1(a):
    a = np.asarray(a, dtype=np.float64)
    if a.size == 0:
        return None
    a = np.clip(a, 0.0, 1000.0)
    if not np.isfinite(a).all():
        return None
    s = float(np.sum(a))
    if not np.isfinite(s) or s <= 1e-300:
        return None
    return a / s


def local_score(a):
    a = np.asarray(a, dtype=np.float64)
    if a.size == 0:
        return float('inf')
    a = np.clip(a, 0.0, 1000.0)
    s = float(np.sum(a))
    if s <= 1e-12:
        return float('inf')
    c = np.convolve(a, a)
    m = float(np.max(c))
    if not np.isfinite(m):
        return float('inf')
    return float(2.0 * len(a) * m / (s * s))


def project_simplex(v, s=1.0):
    n = len(v)
    if n == 0:
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - s)
    if not np.any(cond):
        return np.full(n, s / n, dtype=np.float64)
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - s) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def smoothed_obj_and_grad(a, tau):
    """Log-sum-exp smoothed max(autoconv(a)) and gradient via FFT."""
    n = a.size
    b = np.convolve(a, a)
    b_max = np.max(b)
    shifted = tau * (b - b_max)
    shifted -= np.max(shifted)
    exp_b = np.exp(shifted)
    sum_exp = np.sum(exp_b)
    p = exp_b / sum_exp

    F = (1.0 / tau) * (np.log(sum_exp) + tau * b_max + np.max(shifted))

    # Gradient via FFT cross-correlation
    L = 1
    while L < 3 * n - 2:
        L *= 2
    P = rfft(p, L)
    A = rfft(a, L)
    c = irfft(P * np.conj(A), L)
    grad = 2.0 * c[:n].copy()

    return F, grad


def smooth_grad_descent(a, budget, seed, tau_start=300.0, tau_end=5.0, lr_start=0.05):
    """Smoothed gradient descent with simplex projection and annealing."""
    a = normalize_to_sum1(a).copy()
    if a is None:
        return None, float('inf')
    n = a.size
    best = a.copy()
    best_sc = local_score(best)

    t0 = time.monotonic()
    deadline = t0 + float(budget)
    rng = np.random.default_rng(seed)
    lr = lr_start
    stuck = 0
    fail_count = 0

    while time.monotonic() < deadline:
        progress = min(1.0, (time.monotonic() - t0) / float(budget))
        tau = tau_start * (tau_end / tau_start) ** progress

        F, grad = smoothed_obj_and_grad(a, tau)
        if not np.isfinite(F):
            a = best.copy()
            fail_count += 1
            if fail_count > 20:
                break
            continue
        fail_count = 0

        # Backtracking line search
        accepted = False
        for attempt in range(10):
            a_new = a - lr * grad
            a_new = project_simplex(a_new, 1.0)
            sc_new = local_score(a_new)
            if np.isfinite(sc_new) and sc_new < best_sc:
                best = a_new.copy()
                best_sc = float(sc_new)
                a = a_new
                stuck = 0
                accepted = True
                if lr < 0.5:
                    lr = min(lr * 1.1, 0.5)
                break
            lr *= 0.5

        if not accepted:
            stuck += 1
            if stuck > 80:
                a = best * np.exp(rng.normal(0.0, 0.15, n))
                a = normalize_to_sum1(a)
                if a is None:
                    a = best.copy()
                lr = lr_start
                stuck = 0
            else:
                a = best.copy()

    return best, best_sc


def multiplicative_update(a, eta, K, use_all_peaks=False):
    a = np.asarray(a, dtype=np.float64)
    n = a.size
    if n == 0:
        return a
    c = np.convolve(a, a)
    M = float(np.max(c))
    if not np.isfinite(M) or M <= 0.0:
        return a.copy()

    if use_all_peaks:
        threshold = 0.7 * M
        active = np.nonzero(c >= threshold)[0]
    else:
        if K is None or K <= 0:
            K = max(1, n // 4)
        K = int(K)
        if K >= c.size:
            active = np.nonzero(c > 1e-15)[0]
            if active.size == 0:
                active = np.array([int(np.argmax(c))], dtype=np.int64)
        else:
            active = np.argpartition(c, -K)[-K:]

    w = c[active] / M
    h = np.zeros(n, dtype=np.float64)

    for idx in range(active.size):
        k = int(active[idx])
        wk = float(w[idx])
        s = max(0, k - (n - 1))
        e = min(k, n - 1)
        if s > e:
            continue
        i = np.arange(s, e + 1, dtype=np.int64)
        j = k - i
        h[i] += wk * a[j]
        h[j] += wk * a[i]

    maxh = float(np.max(h))
    if not np.isfinite(maxh) or maxh <= 0.0:
        return a.copy()

    meanh = float(np.mean(h))
    fac = eta * (meanh - h) / maxh
    fac = np.clip(fac, -4.0, 4.0)
    a2 = a * np.exp(fac)
    a2 = np.clip(a2, 0.0, 1000.0)
    a2 = normalize_to_sum1(a2)
    if a2 is None:
        return a.copy()
    return a2

            bestsc = float(sc2)

    # Phase 3: Multiplicative updates (sharp max)
    rem = deadline - time.monotonic()
    if rem > 10.0:
        b = min(0.3 * rem, 120.0)
        c3, sc3 = multiplicative_optimize(best, b, seed + 66666)
        if c3 is not None and sc3 < bestsc:
            best = c3.copy()
            bestsc = float(sc3)

    # Phase 4: Alternate smoothed + multiplicative
    rem = deadline - time.monotonic()
    stall = 0
    while rem > 8.0:
        half = min(rem / 2.0 - 1.0, 60.0)
        if half < 2.0:
            break
        c4, sc4 = smooth_grad_descent(best, half, seed + 77777 + stall,
                                       tau_start=100.0, tau_end=3.0)
        if c4 is not None and sc4 < bestsc:
            best = c4.copy()
            bestsc = float(sc4)
            stall = 0
        else:
            stall += 1
        rem = deadline - time.monotonic()
        if rem > 5.0:
            half2 = min(rem / 2.0 - 1.0, 40.0)
            if half2 > 2.0:
                c5, sc5 = multiplicative_optimize(best, half2, seed + 88888 + stall,
                                                  K_ratio=0.15, use_all=True)
                if c5 is not None and sc5 < bestsc:
                    best = c5.copy()
                    bestsc = float(sc5)
                    stall = 0
        if stall > 8:
            break

    # Phase 5: Final fine-tuning with very small steps
    rem = deadline - time.monotonic()
    if rem > 5.0:
        c6, sc6 = smooth_grad_descent(best, min(rem - 2.0, 30.0), seed + 99999,
                                       tau_start=50.0, tau_end=2.0, lr_start=0.01)
        if c6 is not None and sc6 < bestsc:
            best = c6.copy()
            bestsc = float(sc6)

    return best, bestsc
