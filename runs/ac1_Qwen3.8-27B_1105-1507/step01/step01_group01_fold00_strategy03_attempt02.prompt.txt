```python
import numpy as np

def evaluate_sequence(sequence: list[float]) -> float:
    """
    Evaluates a sequence of coefficients with enhanced security checks.
    Returns np.inf if the input is invalid.
    """
    # --- Security Checks ---

    # Verify that the input is a list
    if not isinstance(sequence, list):
        return np.inf

    # Reject empty lists
    if not sequence:
        return np.inf

    # Check each element in the list for validity
    for x in sequence:
        # Reject boolean types (as they are a subclass of int) and
        # any other non-integer/non-float types (like strings or complex numbers).
        if isinstance(x, bool) or not isinstance(x, (int, float)):
            return np.inf

        # Reject Not-a-Number (NaN) and infinity values.
        if np.isnan(x) or np.isinf(x):
            return np.inf

    # Convert all elements to float for consistency
    sequence = [float(x) for x in sequence]

    # Protect against negative numbers
    sequence = [max(0, x) for x in sequence]

    # Protect against numbers that are too large
    sequence = [min(1000.0, x) for x in sequence]

    n = len(sequence)
    b_sequence = np.convolve(sequence, sequence)
    max_b = max(b_sequence)
    sum_a = np.sum(sequence)

    # Protect against the case where the sum is too close to zero
    if sum_a < 0.01:
        return np.inf

    return float(2 * n * max_b / (sum_a**2))
```

```python
def normalize_to_sum1(a):
    a = np.asarray(a, dtype=np.float64)
    if a.size == 0:
        return None
    a = np.clip(a, 0.0, 1000.0)
    if not np.isfinite(a).all():
        return None
    s = float(np.sum(a))
    if not np.isfinite(s) or s <= 1e-300:

    return a / s

def local_score(a):
    """Exact same objective as evaluate_sequence, assuming finite nonnegative values."""
    a = np.asarray(a, dtype=np.float64)
    if a.size == 0:
        return float('inf')
    a = np.clip(a, 0.0, 1000.0)
    if not np.isfinite(a).all():
        return float('inf')
    s = float(np.sum(a))
    if not np.isfinite(s) or s <= 1e-12:
        return float('inf')
    c = np.convolve(a, a)
    m = float(np.max(c))
    if not np.isfinite(m):
        return float('inf')
    return float(2.0 * len(a) * m / (s * s))


def sym_expand(h, n):
    """Expand symmetric half h to full length n."""
    h = np.asarray(h, dtype=np.float64)
    m = (n + 1) // 2
    if h.size < m:
        pad = np.zeros(m - h.size, dtype=np.float64)
        h = np.concatenate((h, pad))
    else:
        h = h[:m]
    if n == 1:
        return h.copy()
    if n % 2 == 0:
        return np.concatenate((h, h[::-1]))
    else:
        if m == 1:
            return np.array([h[0]], dtype=np.float64)
        return np.concatenate((h, h[-2::-1]))

def multiplicative_update(a, eta, K, force_sym=False):
    a = np.asarray(a, dtype=np.float64)
    n = a.size
    if n == 0:
        return a
    c = np.convolve(a, a)
    M = float(np.max(c))
    if not np.isfinite(M) or M <= 0.0:
        return a.copy()

    if K is None or K <= 0:
        K = max(1, n // 4)
    K = int(K)
    if K >= c.size:
        active = np.nonzero(c > 1e-15)[0]
        if active.size == 0:
            active = np.array([int(np.argmax(c))], dtype=np.int64)
    else:
        active = np.argpartition(c, -K)[-K:]

    w = c[active] / M
    h = np.zeros(n, dtype=np.float64)

    for idx in range(active.size):
        k = int(active[idx])
        wk = float(w[idx])
        s = k - (n - 1)
        if s < 0:
            s = 0
        e = k
        if e >= n:
            e = n - 1
        if s > e:
            continue
        i = np.arange(s, e + 1, dtype=np.int64)
        j = k - i
        h[i] += wk * a[j]
        h[j] += wk * a[i]

    maxh = float(np.max(h))
    if not np.isfinite(maxh) or maxh <= 0.0:
        return a.copy()

    meanh = float(np.mean(h))
    fac = eta * (meanh - h) / maxh
    fac = np.clip(fac, -5.0, 5.0)
    a2 = a * np.exp(fac)
    a2 = np.clip(a2, 0.0, 1000.0)

    if force_sym:
        a2 = 0.5 * (a2 + a2[::-1])
        a2 = normalize_to_sum1(a2)
    else:
        a2 = normalize_to_sum1(a2)

    if a2 is None:
        return a.copy()
    return a2

def build_A_full(f, n):
    """Dense constraint matrix for LP step with full variables."""
    A = np.zeros((2 * n - 1, n), dtype=np.float64)
    for k in range(2 * n - 1):
        s = k - (n - 1)
        if s < 0:
            s = 0
        e = k
        if e >= n:
            e = n - 1
        if s <= e:
            j = np.arange(s, e + 1, dtype=np.int64)
            A[k, j] = f[k - j]
    return A


def build_A_sym(f, n):
    """Dense constraint matrix for LP step with symmetry variables."""
    m = (n + 1) // 2
    A = np.zeros((n, m), dtype=np.float64)
    for v in range(m):
        A[v:n, v] += f[:n - v]
        j2 = n - 1 - v
        if j2 != v:
            A[j2:n, v] += f[:v + 1]
    return A


def lp_solve_step(f, n, force_sym, seed):
    if not HAVE_SCIPY:
        return None

    f = np.asarray(f, dtype=np.float64)
    if force_sym:
        f = force_sym_norm(f)
    else:
        f = normalize_to_sum1(f)
    if f is None:
        return None

    c = np.convolve(f, f)
    M = float(np.max(c))
    if not np.isfinite(M) or M <= 0.0:
        return None

    rng = np.random.default_rng(seed + 987654321)

    try:
        if force_sym:
            m = (n + 1) // 2
            A = build_A_sym(f, n)
            rows = n
        else:
            m = n
            A = build_A_full(f, n)
            rows = 2 * n - 1

        cvec = -np.ones(m, dtype=np.float64)
        # Tiny objective noise helps avoid pathological vertices when the LP optimum
        # set is large.
        cvec += 1e-8 * rng.uniform(-1.0, 1.0, m)
        bvec = np.full(rows, M * (1.0 + 1e-7), dtype=np.float64)
        bounds = [(0.0, 1.0)] * m

        res = linprog(
            cvec,
            A_ub=A,
            b_ub=bvec,
            bounds=bounds,
            method='highs',
            options={'presolve': True}
        )
    except MemoryError:
        return None
    except Exception:
        return None

    if res is None or not res.success:
        return None

    x = np.asarray(res.x, dtype=np.float64)
    x = np.clip(x, 0.0, 1.0)
    if x.size == 0 or float(np.sum(x)) <= 0.0:
        return None

    if force_sym:
        g = sym_expand(x, n)
        g = force_sym_norm(g)
    else:
        g = normalize_to_sum1(x)

    return g


def lp_optimize(a, max_time, seed, force_sym=False):
    f = normalize_to_sum1(a)
    if force_sym:
        f = force_sym_norm(f)
    if f is None:
        return None, float('inf')

    best = f.copy()
    bestsc = local_score(best)
    if max_time <= 0.0:
        return best, bestsc

    t0 = time.monotonic()
    deadline = t0 + float(max_time)
    ts = (0.05, 0.08, 0.12, 0.18, 0.25, 0.35, 0.5, 0.7, 1.0)
    stall = 0

    while time.monotonic() < deadline:
        if not HAVE_SCIPY:
            break

        rem = deadline - time.monotonic()
        if rem < 3.0:
            break

        g = lp_solve_step(best, len(best), force_sym, seed)
        if g is None:
            break

        new_best = best
        new_sc = bestsc

        for t in ts:
            new = (1.0 - t) * best + t * g
            if force_sym:
                new = force_sym_norm(new)
            else:
                new = normalize_to_sum1(new)
            if new is None:
                continue
            sc = local_score(new)
            if np.isfinite(sc) and sc < new_sc:
                new_sc = float(sc)
                new_best = new.copy()

        if new_sc < bestsc:
            best = new_best
            bestsc = new_sc
            stall = 0
        else:
            stall += 1
            if stall > 10:
                break

            # Small multiplicative polish between LP calls.
            K = max(20, min(80, len(best) // 4))
            for _ in range(8):
                if time.monotonic() >= deadline:
                    break
                cand = multiplicative_update(best, 0.22, K, force_sym)
                sc = local_score(cand)
                if np.isfinite(sc) and sc < bestsc:
                    best = cand
                    bestsc = float(sc)
                    break

    return best, bestsc


def block_perturb(a, rng):
    a = np.asarray(a, dtype=np.float64)
    n = a.size
    if n == 0:
        return a
    out = a.copy()
    wmin = max(1, n // 40)
    wmax = max(wmin + 1, n // 8)
    w = int(rng.integers(wmin, min(wmax, n) + 1))
    start = int(rng.integers(0, max(1, n - w + 1)))
    out[start:start + w] *= float(np.exp(rng.normal(0.0, 0.18)))
    if rng.random() < 0.35:
        out = 0.5 * (out + out[::-1])
    return out


def block_local_search(a, max_time, seed):
    a = normalize_to_sum1(a)
    if a is None:
        return None, float('inf')
    best = a.copy()
    bestsc = local_score(best)
    if max_time <= 0.0:
        return best, bestsc

    t0 = time.monotonic()
    deadline = t0 + float(max_time)
    rng = np.random.default_rng(seed)
    n = best.size
    stuck = 0

    while time.monotonic() < deadline:
        cand = block_perturb(best, rng)
        cand = normalize_to_sum1(cand)
        if cand is None:
            cand = best.copy()

        sc = local_score(cand)
        if np.isfinite(sc) and sc < bestsc:
            best = cand.copy()
            bestsc = float(sc)
            stuck = 0
        else:
            stuck += 1

        if stuck > 250:
            noise = np.exp(rng.normal(0.0, 0.08, n))
            cand = best * noise
            cand = normalize_to_sum1(cand)
            if cand is None:
                cand = best.copy()
            sc = local_score(cand)
            if np.isfinite(sc) and sc < bestsc:
                best = cand.copy()
                bestsc = float(sc)
            stuck = 0

    return best, bestsc

    while time.monotonic() < deadline:
```
