"""
Combined optimization for minimizing 2*n*max(conv(a,a))/(sum(a))^2.
Uses LP-based alternating method with gradient polish, multiple initializations,
and adaptive n selection.
"""

import numpy as np
from scipy.optimize import linprog
import time


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    conv = np.convolve(a, a)
    max_b = float(np.max(conv))
    return 2.0 * n * max_b / (s * s)


def build_conv_matrix(a):
    """Build convolution matrix A where (A@g)_k = conv(a,g)_k."""
    n = len(a)
    A = np.zeros((2 * n - 1, n), dtype=np.float64)
    for k in range(2 * n - 1):
        for j in range(max(0, k - n + 1), min(n, k + 1)):
            A[k, j] = a[k - j]
    return A


def lp_best_response(a, upper_bound=100.0):
    """Solve LP: max sum(g) s.t. conv(a,g) <= M, 0 <= g <= ub."""
    n = len(a)
    conv_aa = np.convolve(a, a)
    M = float(np.max(conv_aa))
    A = build_conv_matrix(a)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, upper_bound)] * n
    res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
    if res.status == 0:
        return res.x, float(-res.fun), M
    return None, None, M


def gradient_polish(a, max_iter=5000, eta_init=0.1, eta_decay=0.999):
    """Subgradient-based local search to polish the solution."""
    n = len(a)
    best_a = a.copy()
    best_obj = compute_obj(best_a)
    eta = eta_init
    
    for it in range(max_iter):
        conv = np.convolve(best_a, best_a)
        max_b = float(np.max(conv))
        # Find critical lags (within 1% of max)
        threshold = max_b * 0.99
        critical = np.where(conv >= threshold)[0]
        
        # Compute subgradient: for each j, sum over critical lags k of a[k-j] (where valid)
        grad = np.zeros(n)
        for k in critical:
            j_start = max(0, k - n + 1)
            j_end = min(n - 1, k)
            for j in range(j_start, j_end + 1):
                grad[j] += best_a[k - j]
        
        # Normalize gradient
        grad_norm = np.max(grad)
        if grad_norm < 1e-15:
            break
        grad /= grad_norm
        
        # Update: decrease where gradient is high
        candidate = best_a * (1.0 - eta * grad + eta * np.mean(grad))
        candidate = np.clip(candidate, 0, None)
        s = np.sum(candidate)
        if s < 1e-10:
            break
        candidate /= s
        
        obj_new = compute_obj(candidate)
        if obj_new < best_obj:
            best_a = candidate
            best_obj = obj_new
            eta *= 1.01  # slightly increase step
        else:
            eta *= eta_decay
        
        if eta < 1e-10:
            break
    
    return best_a, best_obj


def random_block_polish(a, rng, max_iter=10000, block_frac=(1/100, 1/10)):
    """Random block perturbation local search."""
    best_a = a.copy()
    best_obj = compute_obj(best_a)
    n = len(best_a)
    lo, hi = block_frac
    min_w = max(1, int(n * lo))
    max_w = max(min_w + 1, int(n * hi))
    
    for it in range(max_iter):
        width = int(rng.integers(min_w, max_w))
        start = int(rng.integers(0, n - width + 1))
        candidate = best_a.copy()
        # Multiplicative perturbation

        candidate = np.clip(candidate, 0, None)
        s = np.sum(candidate)
        if s < 1e-10:

        obj_new = compute_obj(candidate)
        if obj_new < best_obj:

import numpy as np

def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    if n > 64:
        conv = fftconvolve(a, a, mode='full')
    else:
        conv = np.convolve(a, a, mode='full')
    max_b = float(np.max(conv))
    return 2.0 * n * max_b / (s * s)


def build_conv_matrix(a):
    """Build convolution matrix A where (A@g)_k = conv(a,g)_k."""
    n = len(a)
    A = np.zeros((2 * n - 1, n), dtype=np.float64)
    for k in range(2 * n - 1):
        j_lo = max(0, k - n + 1)
        j_hi = min(n - 1, k)
        A[k, j_lo:j_hi + 1] = a[k - j_hi:k - j_lo + 1][::-1]
    return A


def lp_best_response(a, upper_bound=50.0):
    """Solve LP: max sum(g) s.t. conv(a,g)_k <= M for all k, 0 <= g <= ub."""
    n = len(a)
    if n > 64:
        conv_aa = fftconvolve(a, a, mode='full')
    else:
        conv_aa = np.convolve(a, a, mode='full')
    M = float(np.max(conv_aa))
    A = build_conv_matrix(a)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, upper_bound)] * n
    res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
    if res.status == 0:
        return res.x, float(-res.fun), M
    return None, None, M

def coordinate_descent(a_init, max_epochs=50, seed=0, verbose=False):
    """Greedy coordinate descent: optimize one coordinate at a time."""
    rng = np.random.default_rng(seed)
    a = a_init.copy()
    s = float(np.sum(a))
    if s < 1e-10:
        return a_init, np.inf
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(best_a)
    n = len(a)
    factors = np.array([0.2, 0.4, 0.6, 0.8, 0.9, 1.1, 1.25, 1.5, 2.0, 3.0])

    for epoch in range(max_epochs):
        order = rng.permutation(n)
        improved = False
        for j in order:
            if a[j] < 1e-15:
                # Try setting from zero
                trial_vals = [0.01 / n, 0.05 / n, 0.1 / n, 0.5 / n]
                for tv in trial_vals:
                    candidate = a.copy()
                    candidate[j] = tv
                    obj = compute_obj(candidate)
                    if obj < best_obj:
                        a = candidate
                        best_obj = obj
                        best_a = a.copy()
                        improved = True
                continue

            best_val = a[j]
            best_local_obj = best_obj
            for f in factors:
                candidate = a.copy()
                candidate[j] = best_val * f
                candidate = np.clip(candidate, 0, None)
                obj = compute_obj(candidate)
                if obj < best_local_obj:
                    best_local_obj = obj
                    best_val = candidate[j]
            if best_val != a[j]:
                a[j] = best_val
                best_obj = best_local_obj
                best_a = a.copy()
                improved = True

        if verbose and epoch % 10 == 0:
            print(f"    CD epoch={epoch}: obj={best_obj:.6f}")
        if not improved and epoch > 5:
            break

    return best_a, best_obj


def gradient_subgrad_polish(a_init, max_iter=3000, seed=0):
    """Subgradient method using critical lag information."""
    rng = np.random.default_rng(seed)
    a = a_init.copy()
    s = float(np.sum(a))
    if s < 1e-10:
        return a_init, np.inf
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(best_a)
    n = len(a)
    eta = 0.05

    for it in range(max_iter):
        if n > 64:
            conv = fftconvolve(a, a, mode='full')
        else:
            conv = np.convolve(a, a, mode='full')
        max_b = float(np.max(conv))
        threshold = max_b * 0.98
        critical = np.where(conv >= threshold)[0]

        # Compute subgradient
        grad = np.zeros(n)
        for k in critical:
            j_lo = max(0, k - n + 1)
            j_hi = min(n - 1, k)
            if j_lo <= j_hi:
                grad[j_lo:j_hi + 1] += a[k - j_hi:k - j_lo + 1][::-1]

        grad_max = np.max(grad)
        if grad_max < 1e-15:
            break
        grad /= grad_max

        # Center the gradient
        grad_centered = grad - np.mean(grad)

        candidate = a * np.exp(-eta * grad_centered)
        candidate = np.clip(candidate, 0, None)
        s = float(np.sum(candidate))
        if s < 1e-10:
            break
        candidate = candidate / s

        obj_new = compute_obj(candidate)
        if obj_new < best_obj:
            best_a = candidate.copy()
            best_obj = obj_new
            a = candidate
            eta = min(0.5, eta * 1.05)
        else:
            eta *= 0.95
            if eta < 1e-8:
                break

    return best_a, best_obj


def random_block_polish(a_init, rng, max_iter=5000):

    a = a_init.copy()
    best_a = a.copy()
    best_obj = compute_obj(best_a)
    n = len(a)

    max_w = max(min_w + 1, n // 20)

        candidate[start:start + width] *= np.exp(rng.normal(0, 0.08, width))
        candidate = np.clip(candidate, 0, None)

        if s < 1e-10:
            continue
        obj_new = compute_obj(candidate)
        if obj_new < best_obj:
            best_a = candidate.copy()
            best_obj = obj_new

    return best_a, best_obj


def simulated_annealing(a_init, rng, max_iter=3000, T_init=0.01, T_end=1e-6):
    """Simulated annealing with block perturbations."""
    a = a_init.copy()
    best_a = a.copy()
    best_obj = compute_obj(best_a)
    n = len(a)
    width = max(1, n // 30)
    T = T_init

    for it in range(max_iter):
        T = T_init * (T_end / T_init) ** (it / max(max_iter, 1))
        start = int(rng.integers(0, n - width + 1))
        candidate = a.copy()
        candidate[start:start + width] *= np.exp(rng.normal(0, 0.15, width))
        candidate = np.clip(candidate, 0, None)
        s = float(np.sum(candidate))
        if s < 1e-10:
            continue
        obj_new = compute_obj(candidate)
        delta = obj_new - compute_obj(a)
        if delta < 0 or rng.random() < np.exp(-delta / max(T, 1e-10)):
            a = candidate
        if obj_new < best_obj:
            best_a = candidate.copy()
            best_obj = obj_new

    return best_a, best_obj

    elif choice == 2:

        a = rng.random(n)
        mask = rng.random(n) > 0.3
        a[~mask] *= 0.1
        a = np.maximum(a, 0.001)

    a = np.clip(a, 0.001, None)
    s = float(np.sum(a))
    return a / s


def propose_candidate(seed=42, budget_s=1500, **kwargs):
    """
    Main search: LP alternating + coordinate descent + gradient polish + SA.
    Tries multiple n values and initializations with time management.
    """
    rng = np.random.default_rng(seed)
    start_time = time.monotonic()
    deadline = start_time + float(budget_s) - 3.0

    best_seq = None
    best_score = np.inf

    # Try parent sequence
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64)
        parent = np.clip(parent, 0, 1000.0)
        if parent.size > 0 and np.isfinite(parent).all():
            obj = compute_obj(parent)
            if obj < best_score:
                best_score = obj
                best_seq = parent.copy()
            print(f"Parent (n={parent.size}): obj={obj:.6f}")
    except Exception as e:
        print(f"Parent load failed: {e}")

    # Define n values to try (balanced coverage)
    n_values = [50, 80, 100, 120, 150, 200, 250, 300, 400]

    # Phase 1: LP optimization for each n
    for n in n_values:
        if time.monotonic() > deadline:
            break
        remaining = deadline - time.monotonic()
        if remaining < 30:
            break
        print(f"\nN={n}: {remaining:.0f}s remaining, current best={best_score:.6f}")

        n_restarts = 3 if remaining > 200 else 2
        for restart in range(n_restarts):
            if time.monotonic() > deadline - 10:
                break

            a_init = structured_init(n, rng)

            # LP phase
            max_lp_iter = int(min(400, max(50, (deadline - time.monotonic()) / 2.0)))
            a_lp, obj_lp = lp_alternate(a_init, max_iter=max_lp_iter,
                                         t=0.6, seed=seed + restart * 1009 + n,
                                         verbose=(restart == 0))
            if obj_lp < best_score:
                best_score = obj_lp
                best_seq = a_lp.copy()
                print(f"  LP n={n} r={restart}: obj={best_score:.6f}")

            # Coordinate descent polish
            if time.monotonic() < deadline - 15:
                max_cd_epochs = int(min(40, (deadline - time.monotonic()) / 1.0))
                a_cd, obj_cd = coordinate_descent(a_lp, max_epochs=max_cd_epochs,
                                                  seed=seed + restart * 7 + n)
                if obj_cd < best_score:
                    best_score = obj_cd
                    best_seq = a_cd.copy()
                    print(f"  CD n={n} r={restart}: obj={best_score:.6f}")

            # Gradient polish
            if time.monotonic() < deadline - 10:
                a_gp, obj_gp = gradient_subgrad_polish(a_cd, max_iter=2000,
                                                       seed=seed + restart * 3 + n)
                if obj_gp < best_score:
                    best_score = obj_gp
                    best_seq = a_gp.copy()
                    print(f"  GP n={n} r={restart}: obj={best_score:.6f}")

            # Random block polish
            if time.monotonic() < deadline - 10:
                rb_iters = int(min(3000, (deadline - time.monotonic()) / 0.03))
                a_rb, obj_rb = random_block_polish(a_gp, rng, max_iter=max(100, rb_iters))
                if obj_rb < best_score:
                    best_score = obj_rb
                    best_seq = a_rb.copy()
                    print(f"  RB n={n} r={restart}: obj={best_score:.6f}")

    # Phase 2: Focus on the best n found, with more refinement
    if best_seq is not None and time.monotonic() < deadline - 20:
        remaining = deadline - time.monotonic()
        n_best = len(best_seq)
        print(f"\nPhase 2: refining n={n_best}, {remaining:.0f}s remaining")

        # Simulated annealing from best
        sa_iters = int(min(5000, remaining / 0.05))
        a_sa, obj_sa = simulated_annealing(best_seq, rng, max_iter=sa_iters)
        if obj_sa < best_score:
            best_score = obj_sa
            best_seq = a_sa.copy()
            print(f"  SA: obj={best_score:.6f}")

        # More coordinate descent
        if time.monotonic() < deadline - 10:
            cd_epochs = int(min(80, (deadline - time.monotonic()) / 0.5))
            a_cd2, obj_cd2 = coordinate_descent(best_seq, max_epochs=cd_epochs,
                                                seed=seed + 9999)
            if obj_cd2 < best_score:
                best_score = obj_cd2
                best_seq = a_cd2.copy()
                print(f"  CD2: obj={best_score:.6f}")

        # Final random block polish
        if time.monotonic() < deadline - 5:
            rb_iters2 = int((deadline - time.monotonic()) / 0.02)
            a_rb2, obj_rb2 = random_block_polish(best_seq, rng, max_iter=max(200, rb_iters2))
            if obj_rb2 < best_score:
                best_score = obj_rb2
                best_seq = a_rb2.copy()
                print(f"  RB2: obj={best_score:.6f}")

    # Phase 3: Try a few more n values if time permits
    if time.monotonic() < deadline - 60:
        remaining = deadline - time.monotonic()
        # Try n values not yet explored well
        extra_ns = [60, 70, 90, 130, 180, 220, 350, 500]
        for n in extra_ns:
            if time.monotonic() > deadline - 20:
                break
            a_init = structured_init(n, rng)
            max_lp = int(min(200, (deadline - time.monotonic()) / 2.0))
            a_loc, obj_loc = lp_alternate(a_init, max_iter=max_lp,
                                           t=0.5, seed=seed + 50000 + n)
            if obj_loc < best_score:
                best_score = obj_loc
                best_seq = a_loc.copy()
                print(f"  Extra LP n={n}: obj={best_score:.6f}")
            if time.monotonic() < deadline - 10:
                a_cd3, obj_cd3 = coordinate_descent(a_loc, max_epochs=30,
                                                    seed=seed + 60000 + n)
                if obj_cd3 < best_score:
                    best_score = obj_cd3
                    best_seq = a_cd3.copy()
                    print(f"  Extra CD n={n}: obj={best_score:.6f}")

    # Fallback
    if best_seq is None:
        best_seq = np.ones(100)
        best_score = compute_obj(best_seq)

    elapsed = time.monotonic() - start_time
    print(f"\nDONE: n={len(best_seq)}, score={best_score:.8f}, time={elapsed:.1f}s")
    return [float(x) for x in best_seq]
