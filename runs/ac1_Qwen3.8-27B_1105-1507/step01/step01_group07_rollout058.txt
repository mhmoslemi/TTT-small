Mixed optimization for minimizing 2*n*max(conv(a,a))/(sum(a))^2.
Combines subgradient descent, multiplicative weights, simulated annealing,
and LP-based alternating optimization with multiple restarts and sequence lengths.
"""

import numpy as np
from scipy.optimize import linprog
import time

def compute_obj(a):
    n = len(a)
    if n == 0:
        return np.inf
    conv = np.convolve(a, a)
    max_b = float(np.max(conv))
    sum_a = float(np.sum(a))
    if sum_a < 0.01:
        return np.inf
    return 2.0 * n * max_b / (sum_a * sum_a)

def compute_subgradient_fast(a):
    """Efficiently compute subgradient of max(conv(a,a))."""
    n = len(a)
    conv = np.convolve(a, a)
    max_b = float(np.max(conv))
    max_mask = (conv >= max_b - 1e-12).astype(np.float64)
    
    # sg[i] = 2 * sum_{m=0}^{n-1} a[m] * max_mask[i+m]
    # This is a cross-correlation, computable via FFT
    # But for n <= 1000, direct computation is fine
    sg = np.zeros(n)
    for i in range(n):
        sg[i] = 2.0 * np.dot(a, max_mask[i:i+n])
    return sg, max_b

def subgradient_descent(a, max_iter=20000, lr_init=None, verbose=False):
    """Subgradient descent with diminishing step size and momentum."""
    n = len(a)
    a = np.asarray(a, dtype=np.float64).copy()
    a = np.clip(a, 0, None)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
        s = 1.0
    a = a / s
    
    best_a = a.copy()
    best_obj = compute_obj(best_a)
    
    if lr_init is None:
        lr_init = 1.0 / np.sqrt(n)
    
    velocity = np.zeros(n)
    beta = 0.9
    
    for it in range(max_iter):
        sg, max_b = compute_subgradient_fast(a)
        sg_norm = np.linalg.norm(sg)
        if sg_norm < 1e-15:
            break
        sg_unit = sg / sg_norm
        
        # Nesterov momentum
        velocity = beta * velocity + (1 - beta) * sg_unit
        lr = lr_init / np.sqrt(it + 1)
        a_new = a - lr * velocity
        
        # Project onto simplex
        a_new = np.clip(a_new, 0, None)
        s = a_new.sum()
        if s < 1e-15:
            a_new = np.ones(n) / n
        else:
            a_new = a_new / s
        
        obj = compute_obj(a_new)
        if obj < best_obj:
            best_a = a_new.copy()
            best_obj = obj
        
        a = a_new
        
        if verbose and (it % 5000 == 0 or it < 5):
            print(f"    SGD it={it}: obj={best_obj:.6f}")
    
    return best_a, best_obj

def multiplicative_weights_opt(a, max_iter=20000, lr_init=0.5, verbose=False):

def build_A_matrix(a):
    """Build constraint matrix for LP: A[k,j] = a[k-j] if 0<=k-j<n else 0."""
    n = len(a)
    A = np.zeros((2 * n - 1, n), dtype=np.float64)
    for k in range(2 * n - 1):
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        A[k, j_start:j_end + 1] = a[k - j_end:k - j_start + 1][::-1]
    return A

def lp_alternate(a, max_iter=200, t=0.5, verbose=False):

    if s < 1e-15:

    best_obj = compute_obj(best_a)
    
    for it in range(max_iter):
        conv = np.convolve(a, a)
        M = float(np.max(conv))
        A = build_A_matrix(a)
        c = -np.ones(n)
        b_ub = np.full(2 * n - 1, M)
        bounds = [(0.0, 10.0)] * n
        
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')

            break

        g_sum = float(np.sum(g))
        if g_sum < 1e-10:
            break

        a_new = (1.0 - t) * a + t * g
        a_new = np.clip(a_new, 0, None)

        if s < 1e-15:

            best_obj = obj

                print(f"    LP it={it}: obj={best_obj:.6f}, M={M:.6f}, gsum={g_sum:.4f}")

Mixed optimization for minimizing 2*n*max(conv(a,a))/(sum(a))^2.
Combines subgradient descent, multiplicative weights, LP-based alternating

Key improvements over previous approach: adds subgradient/multiplicative weights

final refinement.

import numpy as np
from scipy.optimize import linprog
import time


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    a = np.asarray(a, dtype=np.float64)
    n = len(a)
    if n == 0:
        return np.inf
    conv = np.convolve(a, a)
    max_b = float(np.max(conv))
    sum_a = float(np.sum(a))
    if sum_a < 1e-10:
        return np.inf
    return 2.0 * n * max_b / (sum_a * sum_a)


def compute_subgradient_vec(a):
    """Efficiently compute subgradient of max(conv(a,a)) using vectorized indexing."""
    n = len(a)
    conv = np.convolve(a, a)
    max_b = float(np.max(conv))
    max_mask = (conv >= max_b - 1e-12).astype(np.float64)
    # sg[i] = 2 * sum_{m=0}^{n-1} a[m] * max_mask[i+m]
    idx = np.arange(n)
    M = max_mask[idx[:, None] + idx[None, :]]
    sg = 2.0 * (M @ a)
    return sg, max_b


def subgradient_descent(a, max_iter=100000, verbose=False):
    """Subgradient descent with diminishing step size and momentum."""

        s = 1.0

    best_obj = compute_obj(best_a)
    velocity = np.zeros(n)
    beta = 0.9
    lr_init = 1.0 / np.sqrt(n)
    for it in range(max_iter):
        sg, max_b = compute_subgradient_vec(a)
        sg_norm = np.linalg.norm(sg)
        if sg_norm < 1e-15:
            break
        sg_unit = sg / sg_norm
        velocity = beta * velocity + (1.0 - beta) * sg_unit
        lr = lr_init / np.sqrt(it + 1)
        a_new = a - lr * velocity
        a_new = np.clip(a_new, 0, None)
        s = a_new.sum()
        if s < 1e-15:
            a_new = np.ones(n) / n

            a_new = a_new / s

            best_a = a_new.copy()
            best_obj = obj
        a = a_new
        if verbose and (it % 20000 == 0 or it < 3):
            print(f"    SGD it={it}: obj={best_obj:.6f}")
    return best_a, best_obj


def multiplicative_weights_opt(a, max_iter=100000, verbose=False):
    """Exponentiated gradient / multiplicative weights for non-negative optimization."""
    n = len(a)
    a = np.asarray(a, dtype=np.float64).copy()
    a = np.clip(a, 0, None)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
        s = 1.0
    a = a / s
    best_a = a.copy()
    best_obj = compute_obj(best_a)
    log_a = np.log(np.clip(a, 1e-300, None))
    lr_init = 1.0
    for it in range(max_iter):
        sg, max_b = compute_subgradient_vec(a)
        sg_max = np.max(sg)
        if sg_max < 1e-15:
            break
        sg_norm = sg / sg_max
        lr = lr_init / np.sqrt(it + 1)
        log_a = log_a - lr * sg_norm
        log_a = log_a - np.max(log_a)
        a_new = np.exp(log_a)
        s = a_new.sum()
        if s < 1e-15:
            a_new = np.ones(n) / n
            log_a = np.log(a_new)
        else:
            a_new = a_new / s
        obj = compute_obj(a_new)
        if obj < best_obj:
            best_a = a_new.copy()
            best_obj = obj
        a = a_new
        if verbose and (it % 20000 == 0 or it < 3):
            print(f"    MW it={it}: obj={best_obj:.6f}")
    return best_a, best_obj


def simulated_annealing_opt(a, max_iter=50000, T_init=0.05, T_final=0.0001, seed=42, verbose=False):
    """Simulated annealing with adaptive perturbation."""
    n = len(a)
    a = np.asarray(a, dtype=np.float64).copy()
    a = np.clip(a, 0, None)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
        s = 1.0
    a = a / s
    best_a = a.copy()
    best_obj = compute_obj(best_a)
    cur_obj = best_obj
    rng = np.random.default_rng(seed)
    log_ratio = np.log(T_final / T_init) / max(1, max_iter)
    T = T_init
    for it in range(max_iter):
        frac = T / T_init
        if frac > 0.5:
            width = max(1, n // 8)
            scale = 0.2 + 0.3 * frac
        elif frac > 0.1:
            width = max(1, n // 20)
            scale = 0.1 + 0.2 * frac
        else:
            width = max(1, n // 50)
            scale = 0.05 + 0.1 * frac
        start = int(rng.integers(0, max(1, n - width + 1)))
        a_new = a.copy()
        a_new[start:start + width] *= np.exp(rng.normal(0, scale, width))
        a_new = np.clip(a_new, 0, None)
        s = a_new.sum()
        if s < 1e-15:
            a_new = a.copy()
        else:
            a_new = a_new / s
        new_obj = compute_obj(a_new)
        delta = new_obj - cur_obj
        if delta < 0:
            a = a_new
            cur_obj = new_obj
            if cur_obj < best_obj:
                best_a = a_new.copy()
                best_obj = cur_obj
                if verbose:
                    print(f"    SA it={it}: new best={best_obj:.6f}")
        else:
            if T > 1e-10 and delta < 1000:
                if rng.random() < np.exp(-delta / T):
                    a = a_new
                    cur_obj = new_obj
        T *= float(np.exp(log_ratio))
    return best_a, best_obj


def build_A_matrix(a):
    """Build constraint matrix for LP: A[k,j] = a[k-j] if 0<=k-j<n else 0."""
    n = len(a)
    A = np.zeros((2 * n - 1, n), dtype=np.float64)
    for k in range(2 * n - 1):
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        A[k, j_start:j_end + 1] = a[k - j_end:k - j_start + 1][::-1]
    return A


def lp_alternate(a, max_iter=200, t=0.5, verbose=False):
    """LP-based alternating optimization: fix a, solve LP for g, blend."""
    n = len(a)
    a = np.asarray(a, dtype=np.float64).copy()
    a = np.clip(a, 0, None)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
        s = 1.0
    a = a / s
    best_a = a.copy()
    best_obj = compute_obj(best_a)
    for it in range(max_iter):
        conv = np.convolve(a, a)
        M = float(np.max(conv))
        A = build_A_matrix(a)
        c = -np.ones(n)
        b_ub = np.full(2 * n - 1, M)
        bounds = [(0.0, 10.0)] * n
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
        if res.status != 0:
            break
        g = res.x
        g_sum = float(np.sum(g))
        if g_sum < 1e-10:
            break
        g = g / g_sum
        # Adaptive step size
        a_new = (1.0 - t) * a + t * g
        a_new = np.clip(a_new, 0, None)
        s = a_new.sum()
        if s < 1e-15:
            break
        a_new = a_new / s
        obj = compute_obj(a_new)
        if obj < best_obj:
            best_a = a_new.copy()
            best_obj = obj
            if verbose and (it % 20 == 0 or it < 5):
                print(f"    LP it={it}: obj={best_obj:.6f}, M={M:.6f}, gsum={g_sum:.4f}")
        a = a_new
        # Decrease step size over time
        if it > 50:
            t = max(0.1, t * 0.995)
    return best_a, best_obj


def random_refine(a, max_iter=50000, seed=123, verbose=False):
    """Random block perturbation local search with adaptive width."""
    n = len(a)
    a = np.asarray(a, dtype=np.float64).copy()
    a = np.clip(a, 0, None)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
        s = 1.0
    a = a / s
    best_a = a.copy()
    best_obj = compute_obj(best_a)
    rng = np.random.default_rng(seed)
    no_improve = 0
    for it in range(max_iter):
        # Adaptive width: start wider, narrow down
        if no_improve < max_iter * 0.3:
            width = max(1, n // 15)
            scale = 0.15
        elif no_improve < max_iter * 0.7:
            width = max(1, n // 40)
            scale = 0.08
        else:
            width = max(1, n // 80)
            scale = 0.04
        start = int(rng.integers(0, n - width + 1))
        candidate = best_a.copy()
        candidate[start:start + width] *= np.exp(rng.normal(0, scale, width))
        candidate = np.clip(candidate, 0, None)
        s = candidate.sum()
        if s < 1e-15:
            continue
        candidate = candidate / s
        obj = compute_obj(candidate)
        if obj < best_obj:
            best_a = candidate
            best_obj = obj
            no_improve = 0
            if verbose and (it % 10000 == 0):
                print(f"    RR it={it}: obj={best_obj:.6f}")
        else:
            no_improve += 1
    return best_a, best_obj


def propose_candidate(seed=42, budget_s=1500, **kwargs):
    """
    Mixed optimization combining subgradient descent, multiplicative weights,
    LP-based alternating optimization, and simulated annealing.
    Uses multiple sequence lengths, restarts, and intensive final refinement.
    """
    start_time = time.monotonic()
    deadline = start_time + float(budget_s) - 5.0

    best_seq = None
    best_score = np.inf

    # Try parent sequence
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64)
        parent = np.clip(parent, 0, 1000.0)
        if parent.size > 0 and np.isfinite(parent).all():
            obj = compute_obj(parent)
            if obj < best_score:
                best_score = obj
                best_seq = parent.copy()
            print(f"Parent (n={parent.size}): obj={obj:.6f}")
    except Exception:
        pass

    # Sequence lengths to try (prioritize parent length and nearby values)
    n_values = [150, 100, 200, 250, 300, 400]
    if best_seq is not None:
        pn = len(best_seq)
        if pn not in n_values:
            n_values.insert(0, pn)
    n_values = sorted(set(n_values))

    for n in n_values:
        if time.monotonic() > deadline:
            break
        remaining = deadline - time.monotonic()
        if remaining < 20:
            break
        print(f"\n=== N={n}: {remaining:.0f}s remaining ===")

        # Get initial sequence
        if best_seq is not None and len(best_seq) == n:
            a_init = best_seq.copy()
        else:
            rng_init = np.random.default_rng(seed + n * 1000)
            # Try different initializations
            choice = int(rng_init.integers(0, 3))
            if choice == 0:
                a_init = rng_init.random(n)
            elif choice == 1:
                # Cosine-shaped
                x = np.linspace(0, 1, n)
                a_init = np.maximum(0.01, np.cos(np.pi * (x - 0.5)))
            else:
                # Bimodal
                a_init = np.ones(n)
                a_init[n // 4:3 * n // 4] = 0.3
            a_init = np.clip(a_init, 0, None)
            s = a_init.sum()
            a_init = a_init / s

        a_work = a_init

        # Phase 1: Subgradient descent
        if time.monotonic() < deadline - 30:
            t_avail = deadline - time.monotonic() - 30
            max_iter = int(min(100000, t_avail / 0.001))
            a1, obj1 = subgradient_descent(a_work, max_iter=max_iter, verbose=False)
            if obj1 < best_score:
                best_score = obj1
                best_seq = a1.copy()
                print(f"  SGD: obj={best_score:.6f}")
            a_work = a1

        # Phase 2: Multiplicative weights
        if time.monotonic() < deadline - 30:
            t_avail = deadline - time.monotonic() - 30
            max_iter = int(min(100000, t_avail / 0.001))
            a2, obj2 = multiplicative_weights_opt(a_work, max_iter=max_iter, verbose=False)
            if obj2 < best_score:
                best_score = obj2
                best_seq = a2.copy()
                print(f"  MW: obj={best_score:.6f}")
            a_work = a2

        # Phase 3: LP-based optimization (only for manageable n)
        if n <= 250 and time.monotonic() < deadline - 60:
            t_avail = deadline - time.monotonic() - 60
            max_iter = int(min(200, t_avail / 0.3))
            a3, obj3 = lp_alternate(a_work, max_iter=max_iter, t=0.4, verbose=False)
            if obj3 < best_score:
                best_score = obj3
                best_seq = a3.copy()
                print(f"  LP: obj={best_score:.6f}")
            a_work = a3

        # Phase 4: Simulated annealing
        if time.monotonic() < deadline - 30:
            t_avail = deadline - time.monotonic() - 30
            max_iter = int(min(80000, t_avail / 0.001))
            a4, obj4 = simulated_annealing_opt(a_work, max_iter=max_iter, seed=seed + n, verbose=False)
            if obj4 < best_score:
                best_score = obj4
                best_seq = a4.copy()
                print(f"  SA: obj={best_score:.6f}")
            a_work = a4

        # Phase 5: Random refinement
        if time.monotonic() < deadline - 15:
            t_avail = deadline - time.monotonic() - 15
            max_iter = int(min(80000, t_avail / 0.001))
            a5, obj5 = random_refine(a_work, max_iter=max_iter, seed=seed + n * 7, verbose=False)
            if obj5 < best_score:
                best_score = obj5
                best_seq = a5.copy()
                print(f"  RR: obj={best_score:.6f}")
            a_work = a5

        print(f"  Best after n={n}: {best_score:.6f}")

    # Intensive final refinement on the best sequence
    if best_seq is not None:
        remaining = deadline - time.monotonic()
        n_best = len(best_seq)
        print(f"\n=== FINAL REFINEMENT (n={n_best}, {remaining:.0f}s) ===")

        if remaining > 30:
            # LP pass with small step
            if n_best <= 250:
                t_avail = min(remaining * 0.3, 120)
                max_iter = int(max(50, t_avail / 0.3))
                a_fin, obj_fin = lp_alternate(best_seq, max_iter=max_iter, t=0.2, verbose=True)
                if obj_fin < best_score:
                    best_score = obj_fin
                    best_seq = a_fin.copy()

            # Subgradient with slow decay
            if time.monotonic() < deadline - 20:
                t_avail = deadline - time.monotonic() - 20
                max_iter = int(min(200000, t_avail / 0.001))
                a_sg, obj_sg = subgradient_descent(best_seq, max_iter=max_iter, verbose=True)
                if obj_sg < best_score:
                    best_score = obj_sg
                    best_seq = a_sg.copy()

            # Multiplicative weights
            if time.monotonic() < deadline - 15:
                t_avail = deadline - time.monotonic() - 15
                max_iter = int(min(200000, t_avail / 0.001))
                a_mw, obj_mw = multiplicative_weights_opt(best_seq, max_iter=max_iter, verbose=True)
                if obj_mw < best_score:
                    best_score = obj_mw
                    best_seq = a_mw.copy()

            # Final random refinement
            if time.monotonic() < deadline - 5:
                t_avail = deadline - time.monotonic() - 3
                max_iter = int(min(200000, t_avail / 0.001))
                a_rf, obj_rf = random_refine(best_seq, max_iter=max_iter, seed=999, verbose=True)
                if obj_rf < best_score:
                    best_score = obj_rf
                    best_seq = a_rf.copy()

    # Fallback
    if best_seq is None:
        best_seq = np.ones(100)
        best_score = compute_obj(best_seq)

    elapsed = time.monotonic() - start_time
    print(f"\nDONE: n={len(best_seq)}, score={best_score:.6f}, time={elapsed:.1f}s")
    return [float(x) for x in best_seq]
