def normalize_sequence(arr):
    """Return a nonnegative floating vector normalized to sum 1, or None."""
    try:
        x = np.asarray(arr, dtype=np.float64).ravel()
    except Exception:
        return None
    if x.size == 0:
        return None
    x = np.where(np.isfinite(x), x, 0.0)
    x = np.maximum(x, 0.0)
    s = float(x.sum())
    if s <= 1e-300:
        return None
    x /= s
    return x


def score_normalized(x, n):
    """Score assuming x is nonnegative and sum(x) = 1."""
    conv = np.convolve(x, x)
    m = float(np.max(conv))
    if not np.isfinite(m):
        return float("inf")
    return float(2.0 * n * m)


def convolve_fft(a, b):
    """Compute linear convolution using FFT for efficiency."""
    n = len(a) + len(b) - 1
    size = 1
    while size < n:
        size *= 2
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    result = np.fft.irfft(fa * fb, size)
    return result[:n]


def mix_noisy(base, rng, eps):
    """Mix a base vector with random noise for perturbation."""
    f = normalize_sequence(base)
    if f is None:
        return None

def structured_perturb(base, rng, strength=0.3):

def interpolate_sequence(x, n):
    """Resample a nonnegative sequence to length n and normalize it."""
    x = normalize_sequence(x)
    if x is None or n <= 0:
        return None
    m = x.size
    if m == n:
        return x
    if m == 1:
        return np.ones(n, dtype=np.float64) / float(n)
    old = np.linspace(0.0, 1.0, m, dtype=np.float64)
    new = np.linspace(0.0, 1.0, n, dtype=np.float64)
    out = np.interp(new, old, x)
    out = np.maximum(out, 0.0)
    s = out.sum()
    if s > 1e-300:
        out /= s
    return normalize_sequence(out)


def make_conv_template(n):
    """
    Precompute the CSR matrix structure for convolve(f, g).
    The matrix A is m x n where A[k, j] = f[j] if j is valid for lag k, else 0.
    Wait, actually we need A @ g = f * g.
    
    (f * g)[k] = sum_j f[j] * g[k-j]
    
    So A[k, i] = f[i] if 0 <= k-i <= n-1, i.e., k-(n-1) <= i <= k.
    """
    m = 2 * n - 1
    k64 = np.arange(m, dtype=np.int64)
    # For each k, valid i range is [max(0, k-n+1), min(n-1, k)]
    js = np.maximum(0, k64 - n + 1).astype(np.int32)
    je = np.minimum(n - 1, k64).astype(np.int32)
    counts = (je.astype(np.int64) - js.astype(np.int64) + 1).astype(np.int64)
    total = int(counts.sum())

    cum = np.cumsum(counts)
    starts = np.repeat(cum - counts, counts)
    local = np.arange(total, dtype=np.int64) - starts

    row = np.repeat(k64, counts).astype(np.int32)
    col = (js[row.astype(np.int64)] + local).astype(np.int32)
    # The value A[k, j] = f[j] * 1 (so we just need to store f[j])
    # offset = k - j gives us the index into f
    offsets = (row.astype(np.int64) - col.astype(np.int64)).astype(np.int32)

    A_off = csr_matrix((offsets, (row, col)), shape=(m, n), dtype=np.int32)
    edge_offsets = A_off.data.astype(np.int32, copy=True)
    A_template = A_off.astype(np.float64).copy()
    A_template.data.fill(0.0)

    c_obj = -np.ones(n, dtype=np.float64)
    bounds = [(0.0, None)] * n
    return A_template, edge_offsets, c_obj, bounds, m

    b = np.full(m, M, dtype=np.float64)

Key improvements over previous version:

def normalize_sequence(arr):
    """Return a nonnegative floating vector normalized to sum 1, or None."""
    try:
        x = np.asarray(arr, dtype=np.float64).ravel()
    except Exception:
        return None
    if x.size == 0:
        return None
    x = np.where(np.isfinite(x), x, 0.0)
    x = np.maximum(x, 0.0)
    s = float(x.sum())
    if s <= 1e-300:
        return None
    x /= s
    return x


def score_normalized(x, n):
    """Score assuming x is nonnegative and sum(x) = 1. Returns 2*n*max(f*f)."""
    conv = np.convolve(x, x)
    m = float(np.max(conv))
    if not np.isfinite(m):
        return float("inf")
    return float(2.0 * n * m)

        return None

def make_conv_template(n):
    """
    Precompute the CSR matrix structure for the LP constraint A@g <= M,
    where (f*g)[k] = sum_i f[k-i] * g[i].

    """
    m = 2 * n - 1
    k64 = np.arange(m, dtype=np.int64)
    js = np.maximum(0, k64 - n + 1).astype(np.int32)
    je = np.minimum(n - 1, k64).astype(np.int32)
    counts = (je.astype(np.int64) - js.astype(np.int64) + 1).astype(np.int64)
    total = int(counts.sum())

    cum = np.cumsum(counts)
    starts = np.repeat(cum - counts, counts)
    local = np.arange(total, dtype=np.int64) - starts

    row = np.repeat(k64, counts).astype(np.int32)
    col = (js[row.astype(np.int64)] + local).astype(np.int32)
    offsets = (row.astype(np.int64) - col.astype(np.int64)).astype(np.int32)

    A_off = csr_matrix((offsets, (row, col)), shape=(m, n), dtype=np.int32)
    edge_offsets = A_off.data.astype(np.int32, copy=True)
    A_template = A_off.astype(np.float64).copy()
    A_template.data.fill(0.0)

    c_obj = -np.ones(n, dtype=np.float64)
    bounds = [(0.0, None)] * n
    return A_template, edge_offsets, c_obj, bounds, m

    A.data[:] = f[edge_offsets]
    # Scale for numerical conditioning
    scale = float(n)
    A.data *= scale
    b = np.full(m, M * scale * scale, dtype=np.float64)
    result = linprog(c_obj, A_ub=A, b_ub=b, bounds=bounds, method="highs")
    return result

            ts.append(tv)
            if len(ts) >= 8:
                break

        best_cand_f = None
        best_cand_score = best_score

        for tv in ts:
            f_new = (1.0 - tv) * f + tv * g
            f_new = np.maximum(f_new, 0.0)
            ssum = float(f_new.sum())
            if ssum <= 1e-300:
                continue
            f_new /= ssum
            sc = score_normalized(f_new, n)
            if np.isfinite(sc) and sc < best_cand_score:
                best_cand_score = sc
                best_cand_f = f_new.copy()

        if best_cand_f is not None and best_cand_score < best_score - 1e-12:
            f = best_cand_f
            best_f = f.copy()
            best_score = best_cand_score
            stall = 0
            iteration += 1
            consecutive_no_improve = 0
            if iteration <= 5 or iteration % 25 == 0:
                elapsed = time.monotonic() - t0
                print(f"      n={n} iter={iteration} score={best_score:.8f} M={M:.6g} dt={elapsed:.2f}s", flush=True)
        else:
            consecutive_no_improve += 1
            stall += 1
            if stall <= 3:
                f = mix_noisy(f, rng, 0.08)
            elif stall <= 10:
                f = structured_perturb(best_f, rng, 0.25)
            elif stall <= 25:
                f = block_perturb(best_f, rng, 0.2)
                if f is None:
                    f = structured_perturb(best_f, rng, 0.4)
            else:
                f = structured_perturb(best_f, rng, 0.5)
                if f is None:
                    f = mix_noisy(best_f, rng, 0.5)
            if f is None:
                f = best_f.copy()
            if stall % 15 == 0:
                print(f"      n={n} stall={stall} best={best_score:.8f} rem={remaining:.0f}s", flush=True)
            if stall > 60:
                break

    return best_f, float(best_score)


def propose_candidate(seed=42, budget_s=1500, **kwargs):
    """
    Search for a low-scoring nonnegative sequence using LP-based optimization
    with structured perturbations, multi-start diversity, and adaptive time allocation.
    """
    rng = np.random.default_rng(int(seed) if seed is not None else 42)
    total_budget = max(1.0, float(budget_s))
    reserve = 20.0
    start_time = time.monotonic()
    deadline = start_time + max(1.0, total_budget - reserve)

    best_f = None
    best_score = float("inf")
    given_f = None

    # Load given sequence
    if total_budget >= 30.0:
        try:
            raw = np.asarray(height_sequence_1, dtype=np.float64)
            given_f = normalize_sequence(raw)
            if given_f is not None:
                best_f = given_f.copy()
                best_score = score_normalized(given_f, given_f.size)
                print(f"Given: n={given_f.size}, score={best_score:.8f}", flush=True)
        except Exception as e:
            print(f"Could not load height_sequence_1: {e}", flush=True)
            given_f = None

    if best_f is None:
        best_f = np.ones(1024, dtype=np.float64) / 1024.0
        best_score = 2.0

    # Time allocation: focus on n=1024 (previous best) and try n=2048
    if best_score < 1.52:
        plan = [
            (512, 50.0),
            (1024, 950.0),
            (2048, 380.0),
        ]
    elif best_score < 1.55:
        plan = [
            (256, 40.0),
            (512, 100.0),
            (1024, 850.0),
            (2048, 350.0),
        ]
    else:
        plan = [
            (128, 30.0),
            (256, 60.0),
            (512, 150.0),
            (1024, 800.0),
            (2048, 350.0),
        ]

    for n, alloc in plan:
        rem = deadline - time.monotonic()
        if rem < 10.0:
            break
        alloc = min(alloc, rem - 5.0)
        if alloc <= 5.0:
            continue

        print(f"\n=== n={n}, alloc={alloc:.0f}s, current_best={best_score:.8f} ===", flush=True)

        try:
            template = make_conv_template(n)
        except MemoryError:
            print(f"  Memory error for n={n}, skipping", flush=True)
            continue
        except Exception as e:
            print(f"  Template error for n={n}: {e}", flush=True)
            continue

        # Generate diverse starting points
        starts = []

        # 1. Interpolated best (with light noise)
        if best_f is not None:
            up = interpolate_sequence(best_f, n)
            if up is not None:
                starts.append(up)
                noisy_up = mix_noisy(up, rng, 0.05)
                if noisy_up is not None:
                    starts.append(noisy_up)

        # 2. Given sequence interpolated
        if given_f is not None and given_f.size != n:
            g_up = interpolate_sequence(given_f, n)
            if g_up is not None:
                starts.append(mix_noisy(g_up, rng, 0.1))

        # 3. Structured perturbation of best
        base_at_n = best_f if best_f.size == n else (np.ones(n, dtype=np.float64) / n)
        sp = structured_perturb(base_at_n, rng, 0.3)
        if sp is not None:
            starts.append(sp)

        # 4. Block perturbation
        bp = block_perturb(base_at_n, rng, 0.3)
        if bp is not None:
            starts.append(bp)

        # 5. Pure random Dirichlet
        starts.append(rng.dirichlet(np.ones(n, dtype=np.float64)))

        # 6. Concentrated Dirichlet (peaked)
        starts.append(rng.dirichlet(np.full(n, 0.2, dtype=np.float64)))

        # Deduplicate and limit
        unique_starts = []
        for s in starts:
            s = normalize_sequence(s)
            if s is None or s.size != n:
                continue
            is_dup = False
            for u in unique_starts:
                if float(np.linalg.norm(s - u)) < 1e-5:
                    is_dup = True
                    break
            if not is_dup:
                unique_starts.append(s)

        starts = unique_starts
        max_starts = 5
        if len(starts) > max_starts:
            starts = starts[:max_starts]

        m = len(starts)
        if m == 0:
            starts = [np.ones(n, dtype=np.float64) / n]
            m = 1

        # Time weights: favor earlier (better-informed) starts
        if m == 1:
            weights = [1.0]
        elif m == 2:
            weights = [0.65, 0.35]
        elif m == 3:
            weights = [0.50, 0.30, 0.20]
        elif m == 4:
            weights = [0.40, 0.28, 0.20, 0.12]
        else:
            weights = [0.35, 0.25, 0.20, 0.12, 0.08]

        for idx, init in enumerate(starts):
            rem = deadline - time.monotonic()
            if rem < 5.0:
                break
            tb = alloc * weights[idx]
            tb = min(tb, rem - 2.0)
            if tb < 3.0:
                continue

            init_score = score_normalized(init, n)
            print(f"  Start {idx + 1}/{m}, budget={tb:.0f}s, init_score={init_score:.8f}", flush=True)
            f_res, s_res = optimize_lp(n, init, tb, rng, template)

            if np.isfinite(s_res) and s_res < best_score:
                best_score = float(s_res)
                best_f = np.asarray(f_res, dtype=np.float64).copy()
                print(f"  *** NEW BEST: n={best_f.size}, score={best_score:.8f} ***", flush=True)

        try:
            del template
        except Exception:
            pass

    # Final polish phase at the best length
    if best_f is not None:
        n_best = best_f.size
        rem = deadline - time.monotonic()
        if rem > 25:
            print(f"\n=== Final polish: n={n_best}, rem={rem:.0f}s ===", flush=True)
            try:
                template = make_conv_template(n_best)
                n_polish = int(rem / 8)
                n_polish = min(n_polish, 4)
                for trial in range(n_polish):
                    rem2 = deadline - time.monotonic()
                    if rem2 < 8:
                        break
                    tb = rem2 / (n_polish - trial) * 0.7
                    if tb < 5:
                        tb = 5
                    strength = 0.15 + 0.15 * trial
                    init = structured_perturb(best_f, rng, strength)
                    if init is None:
                        init = mix_noisy(best_f, rng, strength)
                    f_res, s_res = optimize_lp(n_best, init, tb, rng, template)
                    if np.isfinite(s_res) and s_res < best_score:
                        best_score = float(s_res)
                        best_f = np.asarray(f_res, dtype=np.float64).copy()
                        print(f"  Polish {trial + 1}: NEW BEST score={best_score:.8f}", flush=True)
            except Exception as e:
                print(f"  Polish error: {e}", flush=True)

    # Final consolidation
    if best_f is None:
        best_f = np.ones(64, dtype=np.float64) / 64.0

    # Ensure all values are valid for the evaluator
    seq = [float(max(0.0, min(1000.0, x))) for x in best_f]

    # Verify with actual evaluator
    try:
        final_score = float(evaluate_sequence(seq))
    except Exception:
        final_score = float("inf")

    if not np.isfinite(final_score):
        # Fallback
        if given_f is not None:
            seq = [float(max(0.0, min(1000.0, x))) for x in given_f]
            try:
                final_score = float(evaluate_sequence(seq))
            except Exception:
                final_score = float("inf")

    if not np.isfinite(final_score):
        seq = [1.0 / 1024.0] * 1024

    print(f"\nFINAL: n={len(seq)}, score={final_score:.8f}", flush=True)
    return seq
