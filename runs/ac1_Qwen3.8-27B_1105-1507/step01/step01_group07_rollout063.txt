import numpy as np
from scipy.optimize import linprog
import time

def compute_obj(a):
    n = len(a)
    conv = np.convolve(a, a)
    max_b = float(np.max(conv))
    sum_a = float(np.sum(a))
    if sum_a < 0.01:
        return np.inf
    return 2.0 * n * max_b / (sum_a * sum_a)

def build_conv_matrix(a):
    """Build A where A[k,j] = a[k-j] if 0<=k-j<n else 0."""
    n = len(a)
    A = np.zeros((2*n-1, n), dtype=np.float64)
    for k in range(2*n-1):
        j_min = max(0, k-n+1)
        j_max = min(n-1, k)
        for j in range(j_min, j_max+1):
            A[k, j] = a[k-j]
    return A

def solve_lp(a):
    """Solve: min M s.t. conv(a,g)_k <= M, sum(g)=1, g>=0."""
    n = len(a)
    A_conv = build_conv_matrix(a)
    
    # LP variables: [M, g0, g1, ..., g_{n-1}]
    c = np.zeros(n+1)
    c[0] = 1.0
    
    # A_ub: 2n-1 rows
    A_ub = np.zeros((2*n-1, n+1))
    A_ub[:, 0] = -1.0  # coefficient of M
    A_ub[:, 1:] = A_conv
    
    b_ub = np.zeros(2*n-1)
    
    # A_eq: sum(g) = 1
    A_eq = np.zeros((1, n+1))
    A_eq[0, 1:] = 1.0
    b_eq = np.array([1.0])
    
    bounds = [(0, None)] * (n+1)
    
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq, 
                  bounds=bounds, method='highs')
    if res.status == 0:
        return res.x[1:], float(res.x[0])
    return None, None

def run_optimization(n, seed, max_iter=200, t=0.5):
    rng = np.random.default_rng(seed)
    a = np.ones(n) / n  # uniform initialization
    
    best_a = a.copy()
    best_obj = compute_obj(best_a)
    
    stall = 0
    for it in range(max_iter):
        g, M_opt = solve_lp(a)
        if g is None:
            break
        
        current_max = np.max(np.convolve(a, a))
        
        if M_opt >= current_max * (1 - 1e-6):

                a = best_a * noise
                a = np.clip(a, 0, None)
                s = np.sum(a)
                if s < 1e-10:
                    a = np.ones(n) / n
                    s = 1
                a = a / s
                stall = 0
            continue

        a_new = np.clip(a_new, 0, None)
        s = np.sum(a_new)
        if s < 1e-10:

        a_new = a_new / s
        
        obj_new = compute_obj(a_new)
        if obj_new < best_obj:
            best_a = a_new.copy()
            best_obj = obj_new
        a = a_new
    
    return best_a, best_obj

"""

Minimizes 2*n*max(conv(a,a))/(sum(a))^2 for non-negative sequences.
Uses multiple sequence lengths, structured initializations, and random restarts.

import numpy as np
from scipy.optimize import linprog, minimize
import time

def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    conv = np.convolve(a, a)
    max_b = float(np.max(conv))
    sum_a = float(np.sum(a))
    if sum_a < 0.01:
        return np.inf
    return 2.0 * n * max_b / (sum_a * sum_a)

def build_A(a):
    """Build convolution matrix. A[k,j] = a[k-j] if 0<=k-j<n."""
    n = len(a)
    A = np.zeros((2*n-1, n), dtype=np.float64)
    for k in range(2*n-1):
        j_lo = max(0, k - n + 1)
        j_hi = min(n - 1, k)
        if j_lo <= j_hi:
            A[k, j_lo:j_hi+1] = a[k - j_hi:k - j_lo + 1][::-1]
    return A

def solve_lp_for_g(a, M=None):
    """
    Solve LP: min M s.t. conv(a,g)_k <= M for all k, sum(g)=1, g>=0.
    Returns (g, M_opt) or (None, None).
    """
    n = len(a)
    A_conv = build_A(a)
    
    c = np.zeros(n+1)
    c[0] = 1.0
    
    A_ub = np.zeros((2*n-1, n+1))
    A_ub[:, 0] = -1.0
    A_ub[:, 1:] = A_conv
    
    b_ub = np.zeros(2*n-1)
    
    A_eq = np.zeros((1, n+1))
    A_eq[0, 1:] = 1.0
    b_eq = np.array([1.0])
    
    bounds = [(0, None)] * (n+1)
    
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                  bounds=bounds, method='highs', options={'presolve': True})
    if res.status == 0:
        g = res.x[1:]
        M_opt = float(res.x[0])
        return g, M_opt
    return None, None

def lp_alternate(n, a_init, max_iter=300, t=0.5, rng=None, verbose=False):
    """Run LP-based alternating optimization."""
    if rng is None:
        rng = np.random.default_rng()

    a = np.clip(a, 0, None)
    s = np.sum(a)
    if s < 1e-10:
        a = np.ones(n) / n
        s = 1
    a = a / s
    
    best_a = a.copy()
    best_obj = compute_obj(best_a)
    
    stall = 0
    for it in range(max_iter):
        g, M_opt = solve_lp_for_g(a)
        if g is None:
            break
        
        current_max = float(np.max(np.convolve(a, a)))
        
        if M_opt >= current_max * (1 - 1e-7):
            stall += 1
            if stall > 15:
                # Perturb to escape
                perturb = rng.random(n) * 0.5
                a = (best_a + perturb)
                a = np.clip(a, 0, None)
                s = np.sum(a)
                if s < 1e-10:
                    a = np.ones(n) / n
                else:
                    a = a / s
                stall = 0
            continue
        
        stall = 0
        g_s = np.sum(g)
        if g_s < 1e-10:
            continue
        g = g / g_s
        
        # Blend
        a_new = (1 - t) * a + t * g
        a_new = np.clip(a_new, 0, None)
        s = np.sum(a_new)
        if s < 1e-10:
            break
        a_new = a_new / s

        a = a_new
        if obj_new < best_obj:
            best_a = a_new.copy()
            best_obj = obj_new
            if verbose and (it % 20 == 0):
                print(f"    iter={it}: obj={best_obj:.6f}")
    
    return best_a, best_obj

def random_refine(a, rng, max_iter=10000, block_frac=0.05):
    """Random block perturbation local search."""
    best_a = a.copy()
    best_obj = compute_obj(best_a)
    n = len(best_a)
    min_block = max(1, int(n * block_frac))

        start = int(rng.integers(0, n - block_size))
        candidate = best_a.copy()
        candidate[start:start+block_size] *= np.exp(rng.normal(0, 0.2, block_size))
        candidate = np.clip(candidate, 0, None)
        s = np.sum(candidate)
        if s < 1e-10:
            continue
        obj_new = compute_obj(candidate)
        if obj_new < best_obj - 1e-10:
            best_a = candidate
            best_obj = obj_new
    
    return best_a, best_obj

    bounds = [(0, None)] * (n + 1)
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                  bounds=bounds, method='highs', options={'presolve': True})
    if res.status == 0:
        return res.x[1:], float(res.x[0])
    return None, None


def lp_alternate(n, a_init, max_iter=300, deadline=None, rng=None, verbose=False):
    if rng is None:
        rng = np.random.default_rng()
    a = np.clip(a_init.copy(), 0, None)
    s = np.sum(a)
    if s < 1e-10:
        a = np.ones(n) / n
    else:
        a = a / s
    best_a = a.copy()
    best_obj = compute_obj(best_a)
    t = 0.5
    stall = 0
    for it in range(max_iter):
        if deadline is not None and time.monotonic() > deadline:
            break
        g, M_opt = solve_lp_variable_M(a)
        if g is None:
            break
        current_max = float(np.max(np.convolve(a, a)))
        if M_opt >= current_max * (1 - 1e-7):
            stall += 1
            if stall > 20:
                a = best_a * (1.0 + 0.15 * rng.normal(0, 1, n))
                a = np.clip(a, 0.001, None)
                a = a / np.sum(a)
                stall = 0
            continue
        stall = 0
        g_s = np.sum(g)
        if g_s < 1e-10:
            continue
        g = g / g_s
        a_new = (1.0 - t) * a + t * g
        a_new = np.clip(a_new, 0, None)
        s_new = np.sum(a_new)
        if s_new < 1e-10:
            t *= 0.5
            continue
        a_new = a_new / s_new
        obj_new = compute_obj(a_new)
        if obj_new < best_obj:
            best_a = a_new.copy()
            best_obj = obj_new
            a = a_new
            if t < 0.8:
                t = min(0.8, t * 1.05)
        else:
            t *= 0.7
            a_try = (1.0 - t) * a + t * g
            a_try = np.clip(a_try, 0, None)
            s_try = np.sum(a_try)
            if s_try > 1e-10:
                a_try = a_try / s_try
                obj_try = compute_obj(a_try)
                if obj_try < best_obj:
                    best_a = a_try.copy()
                    best_obj = obj_try
                a = a_try
            else:
                a = a_new
        if verbose and it % 30 == 0:
            print(f"      iter={it}: best={best_obj:.6f} t={t:.3f} M_opt={M_opt:.6f} cur_max={current_max:.6f}")
    return best_a, best_obj


def random_refine(a, rng, max_iter=5000):
    best_a = a.copy()
    best_obj = compute_obj(best_a)
    n = len(best_a)
    for _ in range(max_iter):
        block_size = int(rng.integers(max(1, n // 50), max(2, n // 4)))
        start = int(rng.integers(0, n - block_size))
        candidate = best_a.copy()
        candidate[start:start + block_size] *= np.exp(rng.normal(0, 0.15, block_size))
        candidate = np.clip(candidate, 0, None)
        s = np.sum(candidate)
        if s < 1e-10:
            continue
        obj_new = compute_obj(candidate)
        if obj_new < best_obj - 1e-12:
            best_a = candidate
            best_obj = obj_new
    return best_a, best_obj


def structured_init(n, rng):
    choice = int(rng.integers(0, 6))
    if choice == 0:
        a = np.ones(n)
    elif choice == 1:
        freq = int(rng.integers(1, max(2, n // 8)))
        phase = rng.uniform(0, 2 * np.pi)
        a = 1.0 + 0.3 * np.cos(2 * np.pi * freq * np.arange(n) / n + phase)
    elif choice == 2:
        a = np.ones(n)
        a[n // 3:] = 0.6
    elif choice == 3:
        period = int(rng.integers(3, max(4, n // 40)))
        pattern = np.clip(1.0 + 0.3 * rng.normal(0, 1, period), 0.1, None)
        a = np.tile(pattern, n // period + 1)[:n]
    elif choice == 4:
        a = 1.0 + 0.2 * rng.normal(0, 1, n)
    else:
        a = np.maximum(0.1, np.sin(np.pi * np.arange(n) / n) + 0.3 * rng.random(n))
    a = np.clip(a, 0.001, None)
    a = a / np.sum(a)
    return a


def propose_candidate(seed=42, budget_s=1500):
    rng = np.random.default_rng(seed)
    start_time = time.monotonic()
    deadline = start_time + float(budget_s) - 15.0

    best_seq = None
    best_score = np.inf

    # Try parent sequence
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64)
        parent = np.clip(parent, 0, 1000.0)
        if parent.size > 0 and np.isfinite(parent).all():
            obj = compute_obj(parent)
            if np.isfinite(obj) and obj < best_score:
                best_score = obj
                best_seq = parent.copy()
            print(f"Parent (n={parent.size}): obj={obj:.6f}")
    except Exception:
        pass

    # Refine parent with LP if available
    if best_seq is not None and time.monotonic() < deadline - 100:
        n_par = len(best_seq)
        if n_par <= 600:
            print("Refining parent with LP...")
            a_par, obj_par = lp_alternate(n_par, best_seq, max_iter=200,
                                          deadline=deadline, rng=rng, verbose=True)
            if obj_par < best_score:
                best_score = obj_par
                best_seq = a_par.copy()
                print(f"  Refined parent: obj={best_score:.6f}")

    # Try various n values
    n_values = [150, 180, 200, 250, 300, 400, 500, 700]

    for n in n_values:
        if time.monotonic() > deadline - 60:
            break
        remaining = deadline - time.monotonic()
        if remaining < 30:
            break
        num_restarts = min(3, max(1, int(remaining / 50)))
        print(f"\nN={n}: {remaining:.0f}s remaining, {num_restarts} restarts")

        for restart in range(num_restarts):
            if time.monotonic() > deadline - 30:
                break
            a_init = structured_init(n, rng)
            max_iter = min(250, max(40, int((deadline - time.monotonic()) / 0.4)))
            a, obj = lp_alternate(n, a_init, max_iter=max_iter,
                                  deadline=deadline, rng=rng, verbose=(restart == 0))
            if obj < best_score:
                best_score = obj
                best_seq = a.copy()
                print(f"  LP n={n} r={restart}: obj={obj:.6f}")

            # Quick random refinement
            if time.monotonic() < deadline - 20:
                a2, obj2 = random_refine(a, rng, max_iter=1500)
                if obj2 < best_score:
                    best_score = obj2
                    best_seq = a2.copy()
                    print(f"  RL n={n} r={restart}: obj={obj2:.6f}")

    # Final refinement on best
    if best_seq is not None and time.monotonic() < deadline - 5:
        remaining = deadline - time.monotonic()
        n_best = len(best_seq)
        print(f"\nFinal refinement (n={n_best}, {remaining:.0f}s left)")

        # More LP on best
        if n_best <= 1000 and remaining > 20:
            max_iter = int(remaining * 0.5 / 0.4)
            a_fin, obj_fin = lp_alternate(n_best, best_seq, max_iter=max_iter,
                                          deadline=start_time + time.monotonic() - start_time + remaining * 0.5,
                                          rng=rng, verbose=True)
            if obj_fin < best_score:
                best_score = obj_fin
                best_seq = a_fin.copy()
                print(f"  Final LP: obj={best_score:.6f}")

        # Random refinement
        if time.monotonic() < deadline - 2:
            rls_iters = int((deadline - time.monotonic()) / 0.015)
            a_r, obj_r = random_refine(best_seq, rng, max_iter=rls_iters)
            if obj_r < best_score:
                best_score = obj_r
                best_seq = a_r.copy()
                print(f"  Final RL: obj={best_score:.6f}")

    if best_seq is None:
        best_seq = np.ones(100)
        best_score = compute_obj(best_seq)

    elapsed = time.monotonic() - start_time
    print(f"\nDONE: n={len(best_seq)}, score={best_score:.6f}, time={elapsed:.1f}s")
    return [float(x) for x in best_seq]
