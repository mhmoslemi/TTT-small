def project_simplex(v):
    """Project vector v onto the probability simplex {x >= 0, sum(x) = 1}."""

    rho_candidates = np.where(u * np.arange(1, n + 1) > cv)[0]

    rho = rho_candidates[-1]
    theta = cv[rho] / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

"""
Optimizes non-negative sequences to minimize 2*n*max(convolve(a,a))/sum(a)^2.
Uses softmax-approximated gradient descent with adaptive temperature,
momentum, backtracking line search, and alternating projection initialization.
"""

import numpy as np
import time


def project_simplex(v):
    """Project vector v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cv = np.cumsum(u) - 1.0
    rho_candidates = np.where(u * np.arange(1, n + 1) > cv)[0]
    if len(rho_candidates) == 0:
        return np.ones(n) / n
    rho = rho_candidates[-1]
    theta = cv[rho] / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def fast_convolve(a):
    """Compute full linear convolution of a with itself using FFT."""
    n = len(a)
    size = 2 * n - 1
    N = 1 << (size - 1).bit_length()
    fa = np.fft.rfft(a, N)
    fc = fa * fa
    result = np.fft.irfft(fc, N)[:size]
    result[result < 0] = 0.0
    return result


def softmax_gradient(a, conv, T):
    """
    Compute gradient of log(sum_k exp(conv_k/T)) w.r.t. a.
    Approximates the gradient of max_k conv_k for small T.
    """
    n = len(a)
    size = 2 * n - 1
    
    # Softmax weights
    max_conv = np.max(conv)
    exp_vals = np.exp((conv - max_conv) / T)
    w = exp_vals / np.sum(exp_vals)
    
    # First term: first[i] = sum_{j=0}^{n-1} w[i+j] * a[j]
    first = np.correlate(w, a, mode='valid')
    
    # Second term: second[i] = a[i] * (sum_{k=i}^{min(size-1, i+n-1)} w[k] - w[2i])
    w_prefix = np.cumsum(w)
    # w_prefix[k-1] = sum_{j=0}^{k-1} w[j]
    # sum_{k=i}^{i+n-1} w[k] = w_prefix[i+n-1] - w_prefix[i-1] (with appropriate clamping)
    
    k_hi = np.minimum(size - 1, np.arange(n) + n - 1)  # = min(2n-2, i+n-1) = 2n-2 for all i (since i+n-1 <= n-1+n-1 = 2n-2)
    # Actually i+n-1 for i in [0,n-1] gives [n-1, 2n-2], and size-1 = 2n-2, so k_hi = i+n-1 always
    k_hi = np.arange(n) + n - 1  # i+n-1 for i=0..n-1, which is n-1 to 2n-2
    
    # sum_{k=i}^{k_hi} w[k] = w_prefix[k_hi] - w_prefix[i-1]
    sum_w = np.zeros(n)
    sum_w[0] = w_prefix[k_hi[0]]  # w_prefix[n-1] = sum_{k=0}^{n-1} w[k]
    sum_w[1:] = w_prefix[k_hi[1:]] - w_prefix[:n-1]
    # For i=0: sum = w_prefix[n-1] - 0 = sum_{k=0}^{n-1} w[k] ✓
    # For i>0: sum = w_prefix[i+n-1] - w_prefix[i-1] = sum_{k=i}^{i+n-1} w[k] ✓
    
    # Subtract w[2i]
    sum_w -= w[2 * np.arange(n)]
    
    second = a * sum_w
    
    grad = (first + second) / T
    return grad


def alt_projection_init(n, rng, n_iter=80):
    """
    Alternating projection to get a nonnegative sequence with near-flat
    magnitude spectrum.
    """
    N = 1 << (2 * n - 2).bit_length()
    phases = rng.uniform(0.0, 2.0 * np.pi, N // 2 + 1)
    F = np.exp(1j * phases)
    F[0] = 1.0 + 0j

    a = np.ones(n) / n
    for it in range(n_iter):
        a_full = np.fft.irfft(F, N)
        a = a_full[:n].copy()
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s < 1e-10:
            a = np.ones(n) / n
        else:
            a = a / s
        F_full = np.fft.rfft(a, N)
        # Flat magnitude target
        F = np.ones(len(F_full)) * np.exp(1j * np.angle(F_full))
        F[0] = F[0].real + 0j

    return a


def objective_val(a):
    """Compute the objective: max of auto-convolution (with sum(a)=1)."""
    conv = fast_convolve(a)
    return np.max(conv), conv


def optimize_softmax(a, deadline, rng, verbose=True):
    """
    Optimize using softmax-approximated gradient descent with momentum
    and adaptive temperature.
    """
    n = len(a)
    a = project_simplex(a.copy().astype(np.float64))
    
    best_conv, _ = objective_val(a)
    best_a = a.copy()
    
    # Temperature schedule: start high, decrease over time
    T_start = 1.0 / n  # fraction of typical conv value
    T_min = 1e-5 / n
    
    # Momentum
    momentum = 0.0
    v = np.zeros(n)
    
    # Learning rate
    lr = 1.0
    lr_min = 1e-6
    
    it = 0
    stall = 0
    max_stall = 500
    
    while time.time() < deadline:
        it += 1
        
        # Adaptive temperature
        progress = min(1.0, it / 5000.0)
        T = T_start * (T_min / T_start) ** progress
        
        conv, _ = objective_val(a)
        current_max = np.max(conv)
        
        if current_max < best_conv * (1.0 - 1e-10):
            best_conv = current_max
            best_a = a.copy()
            stall = 0
        else:
            stall += 1
        
        # Stall handling
        if stall > max_stall:
            if lr > lr_min:
                lr *= 0.7
            else:
                # Perturb to escape
                if time.time() < deadline - 10:
                    pert = rng.normal(0, 0.03 / np.sqrt(n), n)
                    a = project_simplex(a + pert)
                    momentum = 0.0
                    v = np.zeros(n)
            stall = 0
            # Reset temperature if we've been stuck
            if lr < lr_min * 2:
                T = T_start
        
        # Compute gradient
        grad = softmax_gradient(a, conv, T)
        
        # Normalize gradient
        grad_norm = np.linalg.norm(grad)
        if grad_norm < 1e-15:
            break
        
        # Backtracking line search
        alpha = lr
        f0 = current_max
        improved = False
        for _ in range(20):
            a_trial = project_simplex(a - alpha * grad / grad_norm)
            f_trial = np.max(fast_convolve(a_trial))
            if f_trial < f0 - 1e-12:
                improved = True
                break
            alpha *= 0.5
        
        if not improved:
            # Try a very small step
            a_trial = project_simplex(a - 1e-8 * grad / grad_norm)
            f_trial = np.max(fast_convolve(a_trial))
            if f_trial < f0:
                a = a_trial
            else:
                # Stuck, apply perturbation
                if time.time() < deadline - 10:
                    pert = rng.normal(0, 0.01 / np.sqrt(n), n)
                    a = project_simplex(a + pert)
            continue
        
        # Update momentum
        momentum = 0.9 if it % 1000 == 0 else momentum
        v = momentum * v + (1 - momentum) * grad / grad_norm
        a = project_simplex(a - alpha * v * grad_norm)
        
        # Occasionally re-evaluate and reset
        if it % 500 == 0:
            conv_check = fast_convolve(a)
            if np.max(conv_check) < best_conv:
                best_conv = np.max(conv_check)
                best_a = a.copy()
        
        # Print progress
        if verbose and it % 500 == 0:
            print(f"    it={it}, T={T:.6f}, max_conv={current_max:.8f}, best={best_conv:.8f}, lr={alpha:.6f}")
    
    return best_a

"""
Optimizes non-negative sequences to minimize 2*n*max(convolve(a,a))/sum(a)^2.
Uses softmax-approximated gradient descent with adaptive temperature,
backtracking line search, momentum, and alternating projection initialization.
Employs multiple sequence lengths and restart strategies.
"""

import numpy as np
import time


def project_simplex(v):
    """Project vector v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cv = np.cumsum(u) - 1.0
    rho_candidates = np.where(u * np.arange(1, n + 1) > cv)[0]
    if len(rho_candidates) == 0:
        return np.ones(n) / n
    rho = rho_candidates[-1]
    theta = cv[rho] / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def fast_convolve(a):
    """Compute full linear convolution of a with itself using FFT."""
    n = len(a)
    size = 2 * n - 1
    N = 1 << (size - 1).bit_length()
    fa = np.fft.rfft(a, N)
    fc = fa * fa
    result = np.fft.irfft(fc, N)[:size]
    result[result < 0] = 0.0
    return result


def softmax_gradient(a, conv, T):
    """
    Compute gradient of log(sum_k exp(conv_k/T)) w.r.t. a.
    This approximates the gradient of max_k conv_k for small T.
    """
    n = len(a)
    size = 2 * n - 1

    # Softmax weights (numerically stable)
    max_conv = np.max(conv)
    exp_vals = np.exp((conv - max_conv) / T)
    w = exp_vals / np.sum(exp_vals)

    # First term: first[i] = sum_{j=0}^{n-1} w[i+j] * a[j]
    first = np.correlate(w, a, mode='valid')

    # Second term: a[i] * (sum_{k=i}^{i+n-1} w[k] - w[2i])
    # Compute running sums of w
    w_prefix = np.zeros(size + 1)
    w_prefix[1:] = np.cumsum(w)
    # sum_{k=i}^{i+n-1} w[k] = w_prefix[i+n] - w_prefix[i]
    idx = np.arange(n)
    sum_w = w_prefix[idx + n] - w_prefix[idx]
    # Subtract w[2i]
    sum_w -= w[2 * idx]

    second = a * sum_w

    grad = (first + second) / T
    return grad


def exact_subgradient(a, conv, top_k=10):
    """Compute exact subgradient averaged over top-k active peaks."""
    n = len(a)
    size = 2 * n - 1
    peak_idx = np.argsort(conv)[::-1][:top_k]
    # Weight by how close to max
    weights = np.zeros(size)
    for k in peak_idx:
        weights[k] = max(0, conv[k] - 0.999 * conv[peak_idx[-1]])
    wsum = weights.sum()
    if wsum < 1e-15:
        weights[peak_idx[0]] = 1.0
        wsum = 1.0
    weights /= wsum

    # Same structure as softmax_gradient but with hard weights
    first = np.correlate(weights, a, mode='valid')
    w_prefix = np.zeros(size + 1)
    w_prefix[1:] = np.cumsum(weights)
    idx = np.arange(n)
    sum_w = w_prefix[idx + n] - w_prefix[idx]
    sum_w -= weights[2 * idx]
    second = a * sum_w

    return first + second


def alt_projection_init(n, rng, n_iter=100):
    """Alternating projection for near-flat-spectrum nonnegative sequence."""
    N = 1 << (2 * n - 2).bit_length()
    phases = rng.uniform(0.0, 2.0 * np.pi, N // 2 + 1)
    F = np.exp(1j * phases)
    F[0] = 1.0 + 0j

    a = np.ones(n) / n
    for it in range(n_iter):
        a_full = np.fft.irfft(F, N)
        a = a_full[:n].copy()
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s < 1e-10:
            a = np.ones(n) / n
        else:
            a = a / s
        F_full = np.fft.rfft(a, N)
        F = np.ones(len(F_full)) * np.exp(1j * np.angle(F_full))
        F[0] = F[0].real + 0j

    return a


def cosine_init(n, rng):
    """Cosine window initialization."""
    x = np.linspace(0, 1, n)
    a = 0.5 * (1.0 + np.cos(2.0 * np.pi * x)) + 0.01
    a = project_simplex(a)
    return a


def triangular_init(n, rng):
    """Triangular window initialization."""
    x = np.linspace(0, 1, n)
    a = 1.0 - np.abs(2.0 * x - 1.0)
    a = np.maximum(a, 0.01)
    a = project_simplex(a)
    return a


def optimize_main(a, deadline, rng, verbose=False):

    Main optimizer: softmax gradient descent with momentum, adaptive temperature,
    and backtracking line search.
    """
    n = len(a)
    a = project_simplex(a.copy().astype(np.float64))

    conv = fast_convolve(a)
    best_conv = np.max(conv)
    best_a = a.copy()

    # Temperature: scale relative to typical conv value (~1/n)
    T_start = 5.0 / n
    T_min = 0.01 / n

    # Momentum
    beta_mom = 0.85
    v = np.zeros(n)

    # Line search initial step
    alpha_init = 1.0
    alpha_min = 1e-10

    it = 0
    stall = 0
    max_stall = 300
    phase = 0  # 0: explore, 1: exploit
    phase_switch = 3000

    while time.time() < deadline:
        it += 1

        # Adaptive temperature
        T = T_start * max(T_min / T_start, (0.05) ** (min(1.0, it / 8000.0)))

        conv = fast_convolve(a)
        current_max = np.max(conv)

        if current_max < best_conv * (1.0 - 1e-12):
            best_conv = current_max
            best_a = a.copy()
            stall = 0
        else:
            stall += 1

        # Stall handling
        if stall > max_stall:
            if phase == 0:
                phase = 1
                beta_mom = 0.5
            # Perturb
            if time.time() < deadline - 5:
                scale = 0.02 / np.sqrt(n)
                a = project_simplex(a + rng.normal(0, scale, n))
                v = np.zeros(n)
                T = T_start  # Reset temperature for exploration
            stall = 0
            max_stall = max(200, max_stall // 2)
            alpha_init = max(0.1 * alpha_init, alpha_min)

        # Switch to exploitation phase
        if it > phase_switch:

            T = max(T, T_min)

        # Compute gradient (use exact subgradient when T is very small)

        if grad_norm < 1e-15:

        # Backtracking line search

        f0 = current_max
        improved = False

            alpha_init = max(0.5 * alpha_init, alpha_min)
            if time.time() < deadline - 5:
                a = project_simplex(a + rng.normal(0, 0.005 / np.sqrt(n), n))
            continue

        v = beta_mom * v + (1.0 - beta_mom) * grad
        a = project_simplex(a - alpha * grad)
        alpha_init = min(alpha * 1.5, 5.0)

        # Periodic check

                best_conv = np.max(conv_check)
                best_a = a.copy()
            if verbose:
