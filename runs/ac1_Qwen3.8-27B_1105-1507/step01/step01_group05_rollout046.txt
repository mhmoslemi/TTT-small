"""
Hybrid LP + smooth gradient optimization for minimizing autocorrelation peak.
Combines LP-based direction finding with log-sum-exp smoothed gradients,
multiple diverse initializations, and adaptive step sizes.
"""

import numpy as np
from scipy.optimize import linprog
import time


def build_conv_matrix(f):
    """Build matrix A such that A @ g = np.convolve(f, g, mode='full')."""
    n = len(f)
    A = np.zeros((2 * n - 1, n))
    for k in range(2 * n - 1):
        j_start = max(0, k - n + 1)
        j_end = min(n, k + 1)
        for j in range(j_start, j_end):
            A[k, j] = f[k - j]
    return A


def compute_score_normalized(f):
    """Compute score for sequence normalized to sum=1: 2*n*max(conv(f,f))."""
    n = len(f)
    conv = np.convolve(f, f)
    return 2.0 * n * conv.max()


def smooth_max_gradient(f, T):
    """
    Compute the gradient of the smooth max approximation of max(conv(f,f)).
    Uses log-sum-exp with temperature T.
    Returns (smooth_max_value, gradient_vector).
    """
    n = len(f)
    c = np.convolve(f, f)
    
    # Numerically stable log-sum-exp
    c_max = c.max()
    exp_c = np.exp(T * (c - c_max))
    weights = exp_c / exp_c.sum()
    smooth_val = c_max + np.log(exp_c.sum()) / T
    
    # Gradient: sum_k weights[k] * d(c[k])/df
    # d(c[k])/df[i] = 2*f[k-i] if 0 <= k-i < n else 0
    # total_grad[i] = 2 * sum_k weights[k] * f[k-i] * I(0<=k-i<n)
    # = 2 * sum_j f[j] * weights[i+j] * I(0<=i+j<2n-1)
    # = 2 * np.correlate(weights, f, mode='valid')
    
    # Verify: np.correlate(a, v, mode='valid')[i] = sum_j a[i+j] * v[j]
    # We want: sum_j f[j] * weights[i+j] = sum_j weights[i+j] * f[j]
    # = np.correlate(weights, f, mode='valid')[i]
    
    total_grad = 2.0 * np.correlate(weights, f, mode='valid')
    
    return smooth_val, total_grad


def project_simplex(f):
    """Project f onto the simplex {x >= 0, sum(x) = 1}."""
    f = np.asarray(f, dtype=np.float64)
    n = len(f)
    
    # Simple approach: clip and renormalize
    # (This is an approximation of the true projection, but works in practice
    # when we're close to the simplex)
    f = np.maximum(f, 0)
    s = f.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return f / s


def project_simplex_exact(v):
    """Exact projection onto simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - 1))[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0)


def lp_direction(f, M):
    """
    Find LP direction: maximize sum(g) s.t. (f*g)[k] <= M for all k, g >= 0.
    Returns the direction g (not yet rescaled).
    """
    n = len(f)
    A = build_conv_matrix(f)
    c_obj = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    
    try:
        res = linprog(c_obj, A_ub=A, b_ub=b_ub, bounds=bounds,
                     method='highs', options={'presolve': True, 'time_limit': 30})
    except Exception:
        return None
    
    if not res.success or res.x is None:
        return None
    
    g = np.clip(res.x, 0, None)
    if g.sum() < 1e-15:
        return None
    return g

    f0 should be normalized to sum=1.

    f = f / f.sum()

    best_score = compute_score_normalized(f)

    t_grad = 1.0 / n  # step size for gradient
    T_smooth = 20.0  # temperature for smooth max
    stall_count = 0
