import numpy as np
from scipy.optimize import linprog
import time

def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    conv = np.convolve(a, a)
    max_b = float(np.max(conv))
    sum_a = float(np.sum(a))
    if sum_a < 0.01:
        return np.inf
    return 2.0 * n * max_b / (sum_a * sum_a)

def build_A(a):
    """Build constraint matrix for LP. A[k,j] = a[k-j] if 0<=k-j<n else 0."""
    n = len(a)
    A = np.zeros((2 * n - 1, n), dtype=np.float64)
    for k in range(2 * n - 1):
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        A[k, j_start:j_end + 1] = a[k - j_end:k - j_start + 1][::-1]
    return A

def solve_lp_step(a):
    """Solve LP: max sum(g) s.t. (a*g)_k <= M for all k, 0 <= g_i."""
    n = len(a)
    conv = np.convolve(a, a)
    M = float(np.max(conv))
    if M <= 0:
        return np.ones(n), n, M
    A = build_A(a)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
    if res.status == 0:
        return res.x, float(-res.fun), M
    return None, None, M

def legendre_init(n, rng):
    """Create initialization based on Legendre symbols."""
    # Find a prime > n
    def is_prime(m):
        if m < 2:
            return False
        for i in range(2, int(np.sqrt(m)) + 1):
            if m % i == 0:
                return False
        return True
    
    p = n
    while not is_prime(p):
        p += 1
    
    # Legendre symbol sequence of length p-1, then truncate to n
    legendre = np.zeros(p - 1)
    for j in range(1, p):
        # Legendre symbol (j/p) = j^((p-1)/2) mod p
        ls = pow(j, (p - 1) // 2, p)
        if ls == p - 1:
            legendre[j - 1] = -1
        else:
            legendre[j - 1] = 1
    
    legendre = legendre[:n]
    # Map to non-negative: add 1 so values are 0 or 2
    a = (legendre + 1) / 2.0  # values are 0 or 1

    a = a + 0.3 * rng.random(n)
    a = np.clip(a, 0, None)
    s = np.sum(a)
    if s < 1e-10:
        a = rng.random(n)
        s = np.sum(a)
    return a / s

"""
Improved optimization for minimizing 2*n*max(conv(a,a))/(sum(a))^2.
Combines LP-based alternating optimization with subgradient methods,
Legendre-sequence initializations, multi-resolution search, and
adaptive line search to beat the 1.5027 target.
"""

import numpy as np
from scipy.optimize import linprog
import time


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    conv = np.convolve(a, a)
    max_b = float(np.max(conv))
    sum_a = float(np.sum(a))
    if sum_a < 0.01:
        return np.inf
    return 2.0 * n * max_b / (sum_a * sum_a)


def build_A(a):
    """Build constraint matrix for LP. A[k,j] = a[k-j] if 0<=k-j<n else 0."""
    n = len(a)
    A = np.zeros((2 * n - 1, n), dtype=np.float64)
    for k in range(2 * n - 1):
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        if j_start <= j_end:
            A[k, j_start:j_end + 1] = a[k - j_end:k - j_start + 1][::-1]
    return A


def solve_lp(a):
    """Solve LP: max sum(g) s.t. (a*g)_k <= M for all k, 0 <= g_i."""
    n = len(a)
    conv = np.convolve(a, a)
    M = float(np.max(conv))
    if M <= 1e-15:
        return np.ones(n) / n, 1.0, M
    A = build_A(a)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
        if res.status == 0:
            return res.x, float(-res.fun), M
    except Exception:
        pass
    return None, None, M


def lp_line_search(a, g, num_points=15):
    """Line search along direction from a to g."""
    n = len(a)
    best_t = 0.5
    best_obj = np.inf
    
    # Coarse search
    for t in np.linspace(0, 1, num_points):
        a_try = (1 - t) * a + t * g
        a_try = np.clip(a_try, 0, None)
        s_try = np.sum(a_try)
        if s_try < 1e-10:
            continue
        a_try = a_try / s_try
        obj_try = compute_obj(a_try)
        if obj_try < best_obj:
            best_obj = obj_try
            best_t = t
    
    # Fine search around best
    if 0 < best_t < 1:
        delta = 1.0 / num_points
        for t in np.linspace(max(0, best_t - delta), min(1, best_t + delta), 7):
            a_try = (1 - t) * a + t * g
            a_try = np.clip(a_try, 0, None)
            s_try = np.sum(a_try)
            if s_try < 1e-10:
                continue
            a_try = a_try / s_try
            obj_try = compute_obj(a_try)
            if obj_try < best_obj:
                best_obj = obj_try
                best_t = t
    
    a_new = (1 - best_t) * a + best_t * g
    a_new = np.clip(a_new, 0, None)
    s_new = np.sum(a_new)
    if s_new < 1e-10:
        return a, best_obj
    return a_new / s_new, best_obj


def run_lp_optimize(n, a_init, max_iter=100, verbose=False):
    """Run LP-based alternating optimization with adaptive line search."""
    a = a_init.copy()

        g = g / g_sum

            best_obj = obj_new

        else:

                a_try = a_try / s_try
                obj_try = compute_obj(a_try)
                if obj_try < best_obj:
                    best_a = a_try.copy()
                    best_obj = obj_try
                    a = a_try
                else:
                    a = a_new
            else:
                a = a_new
    
    return best_a, best_obj


def subgradient_optimize(n, a_init, max_iter=3000, verbose=False):
    """Multiplicative weights (mirror descent) for the minimax problem."""
    a = a_init.copy()
    a = np.clip(a, 0, None)
    s = np.sum(a)
    if s < 1e-10:
        a = np.random.random(n)
        s = np.sum(a)
    a = a / s
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    
    if verbose:
        print(f"      subgrad init obj={best_obj:.6f}")
    
    lr = 0.05
    for it in range(max_iter):
        conv = np.convolve(a, a)
        max_val = np.max(conv)
        
        # Find all argmax positions
        threshold = max_val - 1e-10 * max_val
        active = np.where(conv >= threshold)[0]
        
        # Compute subgradient (average over active components)
        grad = np.zeros(n)
        for k in active:
            for i in range(n):
                j = k - i
                if 0 <= j < n:
                    grad[i] += 2.0 * a[j]
        grad /= len(active)
        
        # Multiplicative update (mirror descent on simplex)
        a_new = a * np.exp(-lr * grad)
        a_new = np.clip(a_new, 0, None)
        s_new = np.sum(a_new)
        if s_new < 1e-10:
            lr *= 0.5
            continue
        a_new = a_new / s_new
        
        obj_new = compute_obj(a_new)
        if obj_new < best_obj:
            best_obj = obj_new
            best_a = a_new.copy()
            if verbose and (it % 500 == 0 or it < 5):
                print(f"        it={it}: obj={best_obj:.6f}")
        else:
            lr *= 0.95
            a = a_new  # still advance
        
        # Adaptive learning rate
        if it > 0 and it % 200 == 0:
            lr = max(0.001, min(0.1, lr * 1.1))
    
    return best_a, best_obj


def is_prime(m):
    """Check if m is prime."""
    if m < 2:
        return False
    if m < 4:
        return True
    if m % 2 == 0 or m % 3 == 0:
        return False
    i = 5
    while i * i <= m:
        if m % i == 0 or m % (i + 2) == 0:
            return False
        i += 6
    return True


def legendre_sequence(n, rng):
    """Create initialization from Legendre symbols (quadratic residues mod p)."""
    p = max(n + 1, 7)
    while not is_prime(p):
        p += 1
    # Compute Legendre symbols
    leg = np.zeros(min(n, p - 1))
    for j in range(1, min(n + 1, p)):
        ls = pow(j, (p - 1) // 2, p)
        leg[j - 1] = 1.0 if ls == 1 else 0.0
    if n > p - 1:
        leg = np.concatenate([leg, rng.random(n - (p - 1)) * 0.5])
    # Add noise for robustness
    a = leg + 0.3 * rng.random(n)
    a = np.clip(a, 0, None)
    s = np.sum(a)
    if s < 1e-10:
        a = rng.random(n)
        s = np.sum(a)
    return a / s


def get_initialization(n, init_type, rng, parent=None):
    """Generate an initialization of the given type."""
    if init_type == 0:
        # Pure random
        a = rng.random(n)
    elif init_type == 1:
        # Legendre-based
        a = legendre_sequence(n, rng)
    elif init_type == 2:
        # Cosine wave
        f = rng.uniform(0.5, 3.0)
        phase = rng.uniform(0, 2 * np.pi)
        a = np.clip(1.0 + 0.9 * np.cos(2 * np.pi * f * np.arange(n) / n + phase), 0, None)
    elif init_type == 3:
        # Bimodal (sparse)
        density = rng.uniform(0.3, 0.7)
        a = (rng.random(n) < density).astype(float)
        a[rng.choice(n, max(1, n // 5), replace=False)] += 0.5
    elif init_type == 4:
        # Smooth random (low-pass filtered)
        raw = rng.random(n)
        ksize = max(3, n // 15)
        kernel = np.ones(ksize) / ksize
        a = np.convolve(raw, kernel, mode='same')
    elif init_type == 5:
        # Parent (pad/truncate)
        if parent is not None and len(parent) > 0:
            if len(parent) >= n:
                a = parent[:n].copy().astype(float)
            else:
                a = np.zeros(n)
                a[:len(parent)] = parent
        else:
            a = rng.random(n)
    elif init_type == 6:
        # M-sequence style (based on LFSR)
        # Use a simple LFSR to generate a pseudorandom binary sequence
        lfsr = 1
        a = np.zeros(n)
        for i in range(n):
            a[i] = lfsr & 1
            new_bit = ((lfsr >> 0) ^ (lfsr >> 2) ^ (lfsr >> 3) ^ (lfsr >> 4)) & 1
            lfsr = (lfsr >> 1) | (new_bit << 15)
        a = a + 0.4 * rng.random(n)
    elif init_type == 7:
        # Concentrated center
        center = n // 2
        dist = np.abs(np.arange(n) - center)
        width = rng.uniform(n * 0.1, n * 0.4)
        a = np.exp(-0.5 * (dist / width) ** 2)
        a += 0.1 * rng.random(n)
    else:
        a = rng.random(n)
    
    a = np.clip(a, 0, None)
    s = np.sum(a)
    if s < 1e-10:
        a = rng.random(n)
        s = np.sum(a)
    return a / s
