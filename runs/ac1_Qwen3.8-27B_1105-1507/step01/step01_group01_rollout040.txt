```python
"""
Search for low-score non-negative step-function coefficients.

Objective:
    score = 2*n*max(np.convolve(a,a)) / sum(a)^2

Strategy:
  - scale-normalize to sum 1,
  - keep a pool of candidates: parent, perturbed parent, resampled parent,
    and deterministic/random smooth/asymmetric sequences,
  - polish with:
      * scaled log-sum-exp L-BFGS-B on log-ratio parameters,
      * multiplicative subgradient / mirror-descent updates with active
        and near-active convolution coefficients,
      * linear-program best-response directions for the cross-convolution,
        followed by line search,
  - return the best sequence found under the local score, validated by
    evaluate_sequence() when available.
"""

import time
import math
import numpy as np

try:
    from scipy.optimize import linprog, minimize
    HAVE_SCIPY = True
except Exception:
    HAVE_SCIPY = False


def normalize_to_sum1(a):
    try:
        a = np.asarray(a, dtype=np.float64).ravel()
    except Exception:
        return None
    if a.size == 0:
        return None
    a = np.clip(a, 0.0, 1000.0)
    if not np.isfinite(a).all():
        return None
    s = float(np.sum(a))
    if not np.isfinite(s) or s <= 1e-200:
        return None
    a = a / s
    return a


def score_normalized(a):
    a = np.asarray(a, dtype=np.float64)
    if a.size == 0:
        return float("inf")
    b = conv_full(a)
    if b.size == 0:
        return float("inf")
    m = float(np.max(b))
    if not np.isfinite(m):
        return float("inf")
    return float(2.0 * a.size * m)


def local_score(a):
    a = normalize_to_sum1(a)
    if a is None:
        return float("inf")
    return score_normalized(a)


def safe_score(a):
    a = normalize_to_sum1(a)
    if a is None:
        return float("inf")
    return score_normalized(a)


def conv_full(a):
    a = np.asarray(a, dtype=np.float64)
    n = a.size
    if n == 0:
        return np.array([0.0], dtype=np.float64)
    if n <= 1024:
        return np.convolve(a, a)
    need = 2 * n - 1
    N = 1
    while N < need:
        N <<= 1
    fa = np.fft.rfft(a, N)
    out = np.fft.irfft(fa * fa, N)
    return out[:need]


def make_lam_softmax(b, beta):
    b = np.asarray(b, dtype=np.float64)
    if b.size == 0:
        return np.zeros(0, dtype=np.float64)
    if not np.isfinite(beta) or beta <= 0.0:
        beta = 1000.0
    beta = float(np.clip(beta, 10.0, 1e6))
    m = float(np.max(b))
    z = beta * (b - m)
    z = np.clip(z, -50.0, 50.0)
    w = np.exp(z)
    s = float(np.sum(w))
    if not np.isfinite(s) or s <= 0.0:
        u = np.zeros(b.size, dtype=np.float64)
        u[np.argmax(b)] = 1.0
        return u
    return w / s


def make_lam_topk(b, K):
    b = np.asarray(b, dtype=np.float64)
    n = b.size
    if n == 0:
        return np.zeros(0, dtype=np.float64)
    K = int(K)
    if K <= 0:
        K = 1
    if K >= n:
        w = b.copy()
    else:
        idx = np.argsort(b)[-K:]
        w = np.zeros(n, dtype=np.float64)
        w[idx] = b[idx]
    s = float(np.sum(w))
    if not np.isfinite(s) or s <= 0.0:
        w = np.zeros(n, dtype=np.float64)
        w[np.argmax(b)] = 1.0
        return w
    return w / s


def make_lam_exceed(b, factor):
    b = np.asarray(b, dtype=np.float64)
    n = b.size
    if n == 0:
        return np.zeros(0, dtype=np.float64)
    m = float(np.max(b))
    if not np.isfinite(m):
        return make_lam_softmax(b, 1000.0)
    tgt = float(factor) * m
    w = np.maximum(b - tgt, 0.0)
    s = float(np.sum(w))
    if not np.isfinite(s) or s <= 1e-300:
        return make_lam_topk(b, max(8, min(64, n // 8 + 1)))
    return w / s


def gradient_from_lam(a, lam):
    """
    Gradient of sum_k lam[k]*b[k], where b = convolve(a,a).

    Since b_k = sum_{i+j=k} a_i a_j, the derivative is
        2 * sum_j lam[i+j] a_j.
    Computed by FFT cross-correlation.
    """
    a = np.asarray(a, dtype=np.float64)
    lam = np.asarray(lam, dtype=np.float64)
    n = a.size
    L = lam.size
    if n == 0:
        return np.zeros(0, dtype=np.float64)
    target = n + L - 1
    N = 1
    while N < target:
        N <<= 1
    A = np.fft.rfft(a, N)
    R = np.fft.rfft(lam[::-1], N)
    conv = np.fft.irfft(A * R, N)
    idx = (L - 1) - np.arange(n)
    c = conv[idx]
    g = 2.0 * c
    g = np.asarray(g, dtype=np.float64)
    g[~np.isfinite(g)] = 0.0
    return g


def mul_update(a, g, step):
    a = np.asarray(a, dtype=np.float64)
    g = np.asarray(g, dtype=np.float64)
    if a.size == 0:
        return a.copy()
    if g.size != a.size:
        return a.copy()
    gm = float(np.max(np.abs(g)))
    if not np.isfinite(gm) or gm <= 1e-300:
        return a.copy()
    g = g / gm
    c = float(np.dot(a, g))
    arg = -float(step) * (g - c)
    arg = np.clip(arg, -6.0, 6.0)
    a2 = a * np.exp(arg)
    s = float(np.sum(a2))
    if not np.isfinite(s) or s <= 1e-300:
        return a.copy()
    a2 = a2 / s
    return a2


def line_search_mul(a, g, base_sc, step):
    a = np.asarray(a, dtype=np.float64)
    best = a.copy()
    best_sc = base_sc
    if not np.isfinite(base_sc):
        best_sc = score_normalized(a)
    s = float(step)
    if not np.isfinite(s) or s <= 0.0:
        s = 0.6
    fail = 0
    for _ in range(8):

        sc = safe_score(cand)
        if np.isfinite(sc) and sc < best_sc:

def a_from_y(y, r, idx):
    n = len(idx) + 1
    x = np.empty(n, dtype=np.float64)
    x[r] = 0.0
    x[idx] = np.clip(np.asarray(y, dtype=np.float64), -50.0, 50.0)
    mx = float(np.max(x))
    ex = np.exp(x - mx)
    s = float(np.sum(ex))
    if not np.isfinite(s) or s <= 1e-300:
        return np.ones(n, dtype=np.float64) / float(n)
    return ex / s


def lse_objective(y, r, idx, beta):
    """
    Smooth objective: n * logsumexp(beta * conv(a,a)) / beta,
    where a = softmax(log-rates fixed to a[r]=0).
    Returns (objective, gradient w.r.t. y).
    """
    n = len(idx) + 1
    x = np.empty(n, dtype=np.float64)
    x[r] = 0.0
    x[idx] = np.clip(np.asarray(y, dtype=np.float64), -50.0, 50.0)

    mx = float(np.max(x))
    ex = np.exp(x - mx)
    s = float(np.sum(ex))
    if not np.isfinite(s) or s <= 1e-300:
        return 1e100, np.zeros_like(np.asarray(y, dtype=np.float64))

    a = ex / s
    b = conv_full(a)
    beta = float(beta)
    if not np.isfinite(beta) or beta <= 0.0:
        beta = 1000.0
    beta = float(np.clip(beta, 10.0, 1e6))

    z = beta * b
    zmax = float(np.max(z))
    ez = np.exp(np.clip(z - zmax, -50.0, 50.0))
    W = float(np.sum(ez))
    if not np.isfinite(W) or W <= 1e-300:
        return 1e100, np.zeros_like(np.asarray(y, dtype=np.float64))

    obj = float(n) * (zmax + math.log(W)) / beta
    lam = ez / W
    ga = gradient_from_lam(a, lam)
    dot = float(np.dot(a, ga))
    gx = a * (ga - dot)
    gx = np.asarray(gx, dtype=np.float64) * float(n)
    gx[~np.isfinite(gx)] = 0.0
    return obj, gx[idx]

            except Exception:
                y = ybest.copy()
    return best, best_sc


def run_multiplicative(a0, budget, seed):
    a = normalize_to_sum1(a0)
    if a is None:
        return None, float("inf")
    best = a.copy()
    best_sc = score_normalized(best)
    if budget <= 0.0 or best.size == 0:
        return best, best_sc

    deadline = time.monotonic() + float(budget)
    rng = np.random.default_rng(seed)
    n = best.size

    M0 = best_sc / max(1.0, 2.0 * n)
    if not np.isfinite(M0) or M0 <= 1e-30:
        M0 = 1.0 / n
    beta = float(np.clip(10.0 / max(M0, 1e-12), 50.0, 1e6))
    step = 0.65
    K = max(8, min(160, n // 6))
    stuck = 0
    it = 0

    while time.monotonic() < deadline:
        b = conv_full(a)
        if b.size == 0:
            break
        m = float(np.max(b))
        a_sc = float(2.0 * n * m)
        if not np.isfinite(a_sc):
            a_sc = best_sc

        if np.isfinite(a_sc) and a_sc < best_sc:
            best = a.copy()
            best_sc = float(a_sc)

        mode = it % 4
        if mode == 0:
            lam = make_lam_softmax(b, beta)
        elif mode == 1:
            lam = make_lam_topk(b, K)
        elif mode == 2:
            lam = make_lam_exceed(b, 0.999)
        else:
            lam = make_lam_exceed(b, 0.995)

        g = gradient_from_lam(a, lam)
        cand, cand_sc, step = line_search_mul(a, g, a_sc, step)

        if np.isfinite(cand_sc) and cand_sc < a_sc:
            a = cand
        else:
            step = max(step * 0.72, 0.05)

        if np.isfinite(cand_sc) and cand_sc < best_sc:
            best = cand
            best_sc = float(cand_sc)
            stuck = 0
            beta = float(np.clip(beta * 1.03, 50.0, 1e6))
            step = min(step * 1.05, 2.5)
        else:
            stuck += 1
            if stuck % 25 == 0:
                if rng.random() < 0.5:
                    beta = float(np.clip(beta * 1.2, 50.0, 1e6))
                else:
                    beta = float(np.clip(beta * 0.75, 50.0, 1e6))

            if stuck > 70:
                # Escape the current local patch.  Add a little uniform mass so
                # exact zeros can become positive again.
                sigma = float(rng.uniform(0.05, 0.18))
                noise = np.exp(rng.normal(0.0, sigma, n))
                eps = 1e-5 * rng.random(n)
                cand = best * noise + eps
                cand = normalize_to_sum1(cand)
                if cand is not None:
                    sc = score_normalized(cand)
                    if np.isfinite(sc) and sc < best_sc:
                        best = cand
                        best_sc = float(sc)
                a = best.copy()
                step = 0.65
                stuck = 0
        it += 1

    return best, best_sc


def lp_best_response(a, time_limit):
    if not HAVE_SCIPY:
        return None, None
    a = normalize_to_sum1(a)
    if a is None:
        return None, None
    n = a.size
    if n < 2 or n > 1200:
        return None, None

    rows = 2 * n - 1
    A_ub = np.zeros((rows, n + 1), dtype=np.float64)
    for k in range(rows):
        s = max(0, k - n + 1)
        e = min(k, n - 1)
        if s <= e:
            A_ub[k, s:e + 1] = a[k - np.arange(s, e + 1, dtype=np.int64)]
    A_ub[:, n] = -1.0

    c = np.zeros(n + 1, dtype=np.float64)
    c[n] = 1.0

    A_eq = np.zeros((1, n + 1), dtype=np.float64)
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0], dtype=np.float64)
    b_ub = np.zeros(rows, dtype=np.float64)

    bounds = [(0.0, None)] * n + [(0.0, None)]

    tl = float(time_limit)
    if not np.isfinite(tl) or tl <= 0.0:
        tl = 5.0
    tl = max(0.5, min(tl, 25.0))

    try:
        res = linprog(
            c,
            A_ub=A_ub,
            b_ub=b_ub,
            A_eq=A_eq,
            b_eq=b_eq,
            bounds=bounds,
            method="highs",
            options={"presolve": True, "time_limit": tl},
        )
    except Exception:
        try:
            res = linprog(
                c,
                A_ub=A_ub,
                b_ub=b_ub,
                A_eq=A_eq,
                b_eq=b_eq,
                bounds=bounds,
                method="highs",
                options={"presolve": True},
            )
        except Exception:
            return None, None

    if res is None or not res.success:
        return None, None

    gx = np.asarray(res.x, dtype=np.float64)
    if gx.size < n + 1:
        return None, None
    g = gx[:n]
    t = float(gx[n])
    g = np.clip(g, 0.0, None)
    s = float(np.sum(g))
    if not np.isfinite(s) or s <= 1e-30:
        return None, None
    g = g / s
    return g, t


def run_lp(a0, budget, seed):
    a = normalize_to_sum1(a0)
    if a is None:
        return None, float("inf")
    best = a.copy()
    best_sc = score_normalized(best)
    if not HAVE_SCIPY or budget <= 5.0:
        return best, best_sc

    deadline = time.monotonic() + float(budget)
    rng = np.random.default_rng(seed)
    n = best.size
    stalls = 0
    alphas = (1.0, 0.8, 0.65, 0.5, 0.35, 0.22, 0.12, 0.05)

    while time.monotonic() < deadline:
        rem = deadline - time.monotonic()
        if rem < 2.0:
            break

        g, t = lp_best_response(best, max(1.0, rem - 1.0))
        if g is None:
            break

        improved = False
        for alpha in alphas:
            cand = (1.0 - alpha) * best + alpha * g
            cand = normalize_to_sum1(cand)
            if cand is None:
                continue
            sc = score_normalized(cand)
            if np.isfinite(sc) and sc < best_sc:
                best = cand
                best_sc = float(sc)
                improved = True
                break

        if improved:
            stalls = 0
        else:
            stalls += 1
            if stalls > 10:
                break
            if stalls % 3 == 0:
                noise = best * np.exp(rng.normal(0.0, 0.03, n))
                noise += 1e-5 * rng.random(n)
                cand = normalize_to_sum1(noise)
                if cand is not None:
                    sc = score_normalized(cand)
                    if np.isfinite(sc) and sc < best_sc:
                        best = cand
                        best_sc = float(sc)

    return best, best_sc


def gen_candidate(n, kind, seed):
    n = int(n)
    if n <= 0:
        return None
    rng = np.random.default_rng(seed)
    x = (np.arange(n, dtype=np.float64) + 0.5) / float(n)
    val = None
    kind = int(kind)

    if kind == 0:
        theta = float(rng.uniform(-4.0, 4.0))
        val = np.exp(theta * (x - 0.5))

    elif kind == 1:
        alpha = float(rng.choice(np.array([0.2, 0.4, 0.7, 1.0, 1.5, 2.0, 3.0, 5.0], dtype=np.float64)))
        beta = float(rng.choice(np.array([0.2, 0.4, 0.7, 1.0, 1.5, 2.0, 3.0, 5.0], dtype=np.float64)))
        val = np.power(x, alpha - 1.0) * np.power(1.0 - x, beta - 1.0)

    elif kind == 2:
        c1 = float(rng.uniform(0.0, 1.0))
        c2 = float(rng.uniform(0.0, 1.0))
        w1 = float(rng.uniform(0.02, 0.30))
        w2 = float(rng.uniform(0.02, 0.30))
        g1 = float(rng.uniform(0.1, 1.0))
        g2 = float(rng.uniform(0.1, 1.0))
        val = g1 * np.exp(-np.power((x - c1) / w1, 2))
        val += g2 * np.exp(-np.power((x - c2) / w2, 2))

    elif kind == 3:
        val = rng.normal(size=n)
        w = int(rng.integers(3, min(16, max(4, n)) + 1))
        kern = np.ones(w, dtype=np.float64) / float(w)
        val = np.convolve(val, kern, mode="same")
        val = np.clip(val, 0.0, None)
        if float(np.sum(val)) <= 0.0:
            val = np.ones(n, dtype=np.float64)

    elif kind == 4:
        minb = 4
        maxb = max(minb + 1, min(36, n // 4 + 1))
        nb = int(rng.integers(minb, maxb + 1))
        heights = rng.uniform(0.1, 1.0, nb)
        val = np.zeros(n, dtype=np.float64)
        block = max(1, n // nb)
        for bi in range(nb):
            st = bi * block
            en = min(n, st + block)
            val[st:en] = heights[bi]
        kern = np.array([0.25, 0.5, 0.25], dtype=np.float64)
        val = 0.5 * val + 0.5 * np.convolve(val, kern, mode="same")
        val = np.clip(val, 0.0, None)

    elif kind == 5:
        theta = float(rng.uniform(-5.0, 5.0))
        c = float(rng.uniform(0.0, 1.0))
        w = float(rng.uniform(0.05, 0.5))
        amp = float(rng.uniform(0.1, 1.0))
        val = np.exp(theta * x) + amp * np.exp(-np.power((x - c) / w, 2))

    elif kind == 6:
        # Very asymmetric left-heavy / right-heavy shapes.
        p = float(rng.choice(np.array([0.1, 0.2, 0.3, 0.5, 0.7, 0.9], dtype=np.float64)))
        val = np.power(x, 1.0 / p - 1.0)

    elif kind == 7:
        # Two-block with variable gap, used only as a rough initializer.
        m1 = int(rng.integers(max(4, n // 12), max(5, n // 3)))
        m2 = int(rng.integers(max(4, n // 12), max(5, n // 3)))
        val = np.zeros(n, dtype=np.float64)
        val[:m1] = rng.uniform(0.5, 1.0)
        val[n - m2:] = rng.uniform(0.5, 1.0)
        kern = np.array([0.2, 0.6, 0.2], dtype=np.float64)
        val = 0.6 * val + 0.4 * np.convolve(val, kern, mode="same")

    else:
        val = rng.uniform(0.05, 1.0, n)

    val = np.asarray(val, dtype=np.float64)
    val = np.clip(val, 0.0, None)
    if not np.isfinite(val).all() or float(np.sum(val)) <= 0.0:
        val = np.ones(n, dtype=np.float64)
    return normalize_to_sum1(val)


def resample_to_length(a, m):
    a = normalize_to_sum1(a)
    if a is None or int(m) < 16:
        return None
    m = int(m)
    old = np.linspace(0.0, 1.0, a.size, dtype=np.float64)
    new = np.linspace(0.0, 1.0, m, dtype=np.float64)
    val = np.interp(new, old, a)
    return normalize_to_sum1(val)


def pool_key(item):
    return item[0]


def load_parent():
    p = globals().get("height_sequence_1", None)
    if p is None:
        return None
    return normalize_to_sum1(p)


def build_pool(seed):
    pool = []
    parent = load_parent()

    if parent is not None:
        sc = local_score(parent)
        if np.isfinite(sc):
            pool.append((sc, parent.copy()))

        n = parent.size
        if n <= 2048:
            rng = np.random.default_rng(seed + 911)
            for sigma in (0.04, 0.08, 0.14):
                cand = parent * np.exp(rng.normal(0.0, sigma, n))
                cand += 1e-5 * rng.random(n)
                cand = normalize_to_sum1(cand)
                if cand is not None:
                    sc = local_score(cand)
                    if np.isfinite(sc):
                        pool.append((sc, cand))

            for frac in (0.125, 0.25, 0.375):
                shift = int(frac * n)
                if shift != 0:
                    cand = normalize_to_sum1(np.roll(parent, shift))
                    if cand is not None:
                        sc = local_score(cand)
                        if np.isfinite(sc):
                            pool.append((sc, cand))

            for m in (n - 1, n + 1, 2 * n):
                if 16 <= m <= 4096:
                    cand = resample_to_length(parent, m)
                    if cand is not None:
                        sc = local_score(cand)
                        if np.isfinite(sc):
                            pool.append((sc, cand))

    lengths = []
    if parent is not None:
        lengths.append(parent.size)
    for m in (256, 512, 1024, 2048):
        if m not in lengths and m <= 4096:
            lengths.append(m)
    if not lengths:
        lengths = [512]
    lengths = lengths[:4]

    for n in lengths:
        for kind in range(10):
            cand = gen_candidate(n, kind, seed + 17 * (kind + 1) + 53 * n)
            if cand is None:
                continue
            sc = local_score(cand)
            if np.isfinite(sc):
                pool.append((sc, cand))

    if pool:
        pool.sort(key=pool_key)
    return pool, parent


def optimize_seq(a0, budget, seed):
    a = normalize_to_sum1(a0)
    if a is None or budget <= 0.0:
        return a, float("inf")

    best = a.copy()
    best_sc = local_score(best)
    n = best.size
    t0 = time.monotonic()

    # L-BFGS on log-sum-exp objective.
    rem = budget - (time.monotonic() - t0)
    if HAVE_SCIPY and n <= 2048 and rem > 15.0:
        allowed = min(0.38 * budget, 260.0, rem)
        try:
            c, sc = run_lbfgs(a, allowed, seed + 1)
            if c is not None and np.isfinite(sc) and sc < best_sc:
                best = c
                best_sc = float(sc)
        except Exception:
            pass

    # Multiplicative active-subgradient search.
    rem = budget - (time.monotonic() - t0)
    if rem > 8.0:
        allowed = min(0.34 * budget, 260.0, rem)
        try:
            c, sc = run_multiplicative(best, allowed, seed + 2)
            if c is not None and np.isfinite(sc) and sc < best_sc:
                best = c
                best_sc = float(sc)
        except Exception:
            pass

    # LP best-response polish.
    rem = budget - (time.monotonic() - t0)
    if HAVE_SCIPY and n <= 1200 and rem > 10.0:
        allowed = min(0.22 * budget, 150.0, rem)
        try:
            c, sc = run_lp(best, allowed, seed + 3)
            if c is not None and np.isfinite(sc) and sc < best_sc:
                best = c
                best_sc = float(sc)
        except Exception:
            pass

    # Final short multiplicative polish.
    rem = budget - (time.monotonic() - t0)
    if rem > 5.0:
        allowed = min(rem * 0.8, 100.0)
        try:
            c, sc = run_multiplicative(best, allowed, seed + 4)
            if c is not None and np.isfinite(sc) and sc < best_sc:
                best = c
                best_sc = float(sc)
        except Exception:
            pass

    return best, best_sc


def propose_candidate(seed=42, budget_s=1500, **kwargs):
    start = time.monotonic()
    try:
        budget = float(budget_s)
    except Exception:
        budget = 1500.0
    if not np.isfinite(budget) or budget <= 0.0:
        budget = 1500.0
    deadline = start + max(20.0, budget) - 25.0

    seed = int(seed)
    pool, parent = build_pool(seed + 101)

    best_seq = None
    best_score = float("inf")

    if parent is not None:
        sc = local_score(parent)
        if np.isfinite(sc):
            best_seq = parent.copy()
            best_score = float(sc)

    if pool:
        for item in pool:
            sc = item[0]
            if np.isfinite(sc) and sc < best_score:
                best_seq = item[1].copy()
                best_score = float(sc)

    print(f"initial best score={best_score:.8f}", flush=True)

    # Candidate list for deeper optimization: best initial candidates and
    # the parent (if present), avoiding too many duplicates.
    entries = []
    if parent is not None:
        entries.append((local_score(parent), parent.copy()))
    for item in pool[:6]:
        entries.append((item[0], item[1]))
    entries.sort(key=pool_key)

    fracs = (0.42, 0.20, 0.14, 0.10, 0.08, 0.04)
    for i, item in enumerate(entries[:6]):
        rem = deadline - time.monotonic()
        if rem < 25.0:
            break
        sc0 = item[0]
        a0 = item[1]
        base = budget * fracs[i] if i < len(fracs) else 60.0
        allowed = min(base, rem * 0.78)
        if allowed < 6.0:
            continue
        try:
            cand, sc = optimize_seq(a0, allowed, seed + 997 + 13 * i)
            if cand is not None and np.isfinite(sc) and sc < best_score:
                best_seq = cand
                best_score = float(sc)
                print(f"optimized candidate {i}: score={best_score:.8f} len={cand.size}", flush=True)
        except Exception as exc:
            print(f"candidate {i} failed: {exc}", flush=True)

    # Final polish on the global best.
    if best_seq is not None:
        rem = deadline - time.monotonic()
        if rem > 8.0:
            allowed = min(rem * 0.8, 70.0)
            try:
                cand, sc = run_multiplicative(best_seq, allowed, seed + 7777)
                if cand is not None and np.isfinite(sc) and sc < best_score:
                    best_seq = cand
                    best_score = float(sc)
                    print(f"final multiplicative score={best_score:.8f}", flush=True)
            except Exception:
                pass

    if best_seq is None:
        if pool:
            best_seq = pool[0][1].copy()
        if best_seq is None and parent is not None:
            best_seq = parent.copy()
        if best_seq is None:
            best_seq = np.ones(64, dtype=np.float64)

    final = normalize_to_sum1(best_seq)

    # Validate against the official evaluator and fall back if needed.
    official = float("inf")
    try:
        if final is not None:
            official = float(evaluate_sequence([float(x) for x in final]))
    except Exception:
        official = float("inf")

    if not np.isfinite(official) or official >= best_score:
        fallbacks = []
        if parent is not None:
            fallbacks.append(parent)
        for item in pool[:5]:
            fallbacks.append(item[1])
        fallbacks.append(np.ones(64, dtype=np.float64))

        for fb in fallbacks:
            ff = normalize_to_sum1(fb)
            if ff is None:
                continue
            try:
                o2 = float(evaluate_sequence([float(x) for x in ff]))
                if np.isfinite(o2) and o2 < official:
                    final = ff
                    official = float(o2)
                    break
            except Exception:
                continue

    if final is None:
        final = np.ones(64, dtype=np.float64)

    final = np.asarray(final, dtype=np.float64)
    final = np.clip(final, 0.0, 1000.0)

    try:
        o3 = float(evaluate_sequence([float(x) for x in final]))
        if np.isfinite(o3):
            best_score = float(o3)
        else:
            best_score = float("inf")
    except Exception:
        pass

    print(f"FINAL local={best_score:.8f} length={final.size}", flush=True)
    return [float(x) for x in final]
```
