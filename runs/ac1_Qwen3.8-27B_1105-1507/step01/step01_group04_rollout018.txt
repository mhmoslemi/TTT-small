import numpy as np

def _fft_conv_full(f):
    """Full linear convolution of f with itself via FFT."""
    n = len(f)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    F = np.fft.rfft(f, size)
    return np.fft.irfft(F * F, size)[:2 * n - 1]


def _norm_sum(f):
    """Normalize to sum=1, clip negatives. Returns None if degenerate."""
    f = np.maximum(f, 0.0)
    s = f.sum()
    if s < 1e-30:
        return None
    return f / s


def _fast_score(f):
    """Score assuming f>=0, sum(f)=1."""
    n = len(f)
    h = _fft_conv_full(f)
    return 2.0 * n * float(h.max())


def _gradient_fft(f, active, n, fft_size):
    """
    Compute (2/|K|) * sum_{k in K} f[k-j] for each j, using FFT.
    active: array of active lag indices (0 to 2n-2).
    """
    I = np.zeros(2 * n - 1)
    I[active] = 1.0
    J = I[::-1]
    F = np.fft.rfft(f, fft_size)
    JF = np.fft.rfft(J, fft_size)
    C = np.fft.irfft(F * JF, fft_size)[:3 * n - 2]
    return (2.0 / len(active)) * C[n - 1:2 * n - 1][::-1]
