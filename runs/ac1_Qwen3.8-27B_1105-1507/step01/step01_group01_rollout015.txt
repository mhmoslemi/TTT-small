def _lp_step(f, n, max_constraints=300, seed=0):
    """LP with reduced constraints (only top max_constraints)."""
    if not HAVE_SCIPY:
        return None
    f = _normalize(f)
    if f is None:
        return None

    conv = np.convolve(f, f)
    M = float(np.max(conv))
    if M < 1e-30:
        return None

    # Select top constraints
    n_cons = len(conv)
    K = min(max_constraints, n_cons)
    top_idx = np.argpartition(conv, -K)[-K:]

    # Build constraint matrix (only selected rows)
    A = np.zeros((K, n), dtype=np.float64)
    b = np.zeros(K, dtype=np.float64)
    for row, k in enumerate(top_idx):
        s = max(0, int(k) - n + 1)
        e = min(n - 1, int(k))
        if s <= e:
            A[row, s:e+1] = f[int(k) - (s:e+1)]
        b[row] = M * (1.0 + 1e-9)

    cvec = -np.ones(n, dtype=np.float64)
    bounds = [(0.0, None)] * n

    try:
        res = linprog(cvec, A_ub=A, b_ub=b, bounds=bounds,
                      method='highs', options={'presolve': True})
    except Exception:
        return None

    if res is None or not res.success:
        return None

    x = np.clip(res.x, 0.0, None)
    if float(np.sum(x)) < 1e-30:
        return None
    g = _normalize(x)
    return g

        return best, best_sc

    if step_choices is None:
        step_choices = (0.1, 0.2, 0.3, 0.5, 0.7, 1.0)

    t0 = time.monotonic()
    deadline = t0 + budget
    lg = time.monotonic() + budget
    stall = 0
    mp = ln = 0
    seed_cur = seed

    while time.monotonic() < deadline:
        if not HAVE_SCIPY:
            break
        rem = deadline - time.monotonic()
        if rem < 2.0:
            break

        seed_cur += 17
        g = _lp_step(best, len(best), max_constraints, seed_cur)
        if g is None:
            break

        new_best = best
        new_sc = best_sc
        for t in step_choices:
            cand = (1.0 - t) * best + t * g
            cand = _normalize(cand)
            if cand is None:
                continue
            sc = _score(cand)
            if sc < new_sc:
                new_sc = sc
                new_best = cand.copy()

        if new_sc < best_sc:
            best = new_best
            best_sc = new_sc
            stall = 0
        else:
            stall += 1
            # Polish with flatten
            for _ in range(3):
                if time.monotonic() >= deadline:
                    break
                cand = _flatten_step(best, 0.2, 1.5)
                sc = _score(cand)
                if sc < best_sc:
                    best = cand.copy()
                    best_sc = sc
                    break
            if stall > 15:
                break

    return best, best_sc


def _resample(a, new_n):
    """Resample sequence to new length via interpolation."""
    a = np.asarray(a, dtype=np.float64)
    old = np.linspace(0.0, 1.0, a.size)
    new = np.linspace(0.0, 1.0, new_n)
    out = np.interp(new, old, a)
    return _normalize(out)


def _make_random(n, seed, kind=0):
    rng = np.random.default_rng(seed)
    if kind == 0:
        val = rng.uniform(0.05, 1.0, n)
    elif kind == 1:
        val = rng.beta(2.0, 2.0, n)
    elif kind == 2:
        # Block structure
        nb = max(2, n // 16)
        bv = rng.uniform(0.1, 1.0, nb)
        val = np.repeat(bv, n // nb + 1)[:n]
    elif kind == 3:
        # Windowed
        x = np.linspace(0, 1, n)
        val = rng.uniform(0.1, 1.0, n) * (0.5 + 0.5 * np.sin(2 * np.pi * rng.uniform(0.5, 3.0) * x))
    else:
        val = np.abs(rng.normal(0.5, 0.3, n)) + 0.01
    val = np.clip(val, 0.0, 1000.0)
    return _normalize(val)


def _optimize_length(n, budget, seed, parent=None):
    """Optimize at a given length using flatten + LP."""
    if budget < 5.0:
        return None, float('inf')

    t0 = time.monotonic()
    deadline = t0 + budget
    best = None
    best_sc = float('inf')

    # Generate several candidates
    candidates = []
    if parent is not None and parent.size > 1:
        p = _resample(parent, n)
        if p is not None:
            candidates.append(p)
    for k in range(4):
        c = _make_random(n, seed + 100 * k, k)
        if c is not None:
            candidates.append(c)

    # Warmup: quick flatten on each candidate
    warmup_time = min(30.0, 0.15 * budget)
    per_cand = warmup_time / len(candidates) if candidates else 0

    for c in candidates:
        if per_cand > 1.0:
            c2, sc2 = _flatten_optimize(c, per_cand, seed + 7, power_start=1.0, power_end=2.0)
            if c2 is not None:
                sc2 = _score(c2)
                if sc2 < best_sc:
                    best = c2.copy()
                    best_sc = sc2

    if best is None and candidates:
        best = candidates[0].copy()
        best_sc = _score(best)

    # Main optimization loop
    phase = 0
    while time.monotonic() < deadline:
        rem = deadline - time.monotonic()
        if rem < 5.0:
            break

        if phase == 0:
            # Flatten phase
            b = min(60.0, 0.4 * rem)
            c2, sc2 = _flatten_optimize(best, b, seed + 1000, power_start=1.0, power_end=4.0)
            if c2 is not None and sc2 < best_sc:
                best = c2.copy()
                best_sc = sc2
            phase = 1

        elif phase == 1:
            # LP phase
            lp_b = min(120.0, 0.35 * rem)
            c2, sc2 = _lp_optimize(best, lp_b, seed + 2000, max_constraints=250)
            if c2 is not None and sc2 < best_sc:
                best = c2.copy()
                best_sc = sc2
            phase = 2

        elif phase == 2:
            # More flatten
            b = min(60.0, 0.3 * rem)
            c2, sc2 = _flatten_optimize(best, b, seed + 3000, power_start=2.0, power_end=5.0)
            if c2 is not None and sc2 < best_sc:
                best = c2.copy()
                best_sc = sc2
            phase = 3

        elif phase == 3:
            # LP again with different constraints
            lp_b = min(60.0, 0.2 * rem)
            c2, sc2 = _lp_optimize(best, lp_b, seed + 4000, max_constraints=min(500, 2 * n))
            if c2 is not None and sc2 < best_sc:
                best = c2.copy()
                best_sc = sc2
            phase = 4

        else:
            # Final flatten
            b = min(40.0, 0.2 * rem)
            c2, sc2 = _flatten_optimize(best, b, seed + 5000, power_start=3.0, power_end=6.0)
            if c2 is not None and sc2 < best_sc:
                best = c2.copy()
                best_sc = sc2
            phase = 0

    return best, best_sc


def propose_candidate(seed=42, budget_s=1500, **kwargs):
    """
    Main search function. Returns a list of non-negative floats minimizing
    the evaluate_sequence score.
    """
    start_time = time.monotonic()
    try:
        budget = float(budget_s)
    except Exception:
        budget = 1500.0
    if not np.isfinite(budget) or budget <= 0:
        budget = 1500.0
    deadline = start_time + budget - 15.0

    # Load parent
    parent_arr = None
    try:
        p = globals().get('height_sequence_1', None)
        if p is not None:
            parent_arr = np.asarray(p, dtype=np.float64)
            parent_arr = np.clip(parent_arr, 0.0, 1000.0)
            if parent_arr.size == 0 or not np.isfinite(parent_arr).all():
                parent_arr = None
    except Exception:
        parent_arr = None

    best_seq = None
    best_score = float('inf')

    if parent_arr is not None:
        pn = _normalize(parent_arr)
        if pn is not None:
            best_seq = pn.copy()
            best_score = _score(pn)
            print(f"Parent: len={pn.size} score={best_score:.6f}", flush=True)

    # Optimization plan
    if budget >= 1200:
        plan = [
            (512, 350, "parent"),
            (512, 300, "random"),
            (1024, 300, "parent"),
            (2048, 300, "parent"),
        ]
    elif budget >= 600:
        plan = [
            (512, 200, "parent"),
            (512, 150, "random"),
            (1024, 200, "parent"),
        ]
    else:
        plan = [
            (512, budget * 0.4, "parent"),
            (512, budget * 0.3, "random"),
            (1024, budget * 0.2, "parent"),
        ]

    seed_cur = int(seed)

    for i, (n, alloc, src) in enumerate(plan):
        rem = deadline - time.monotonic()
        if rem < 30.0:
            break
        b = min(float(alloc), rem - 15.0)
        if b < 10.0:
            break

        # Choose parent for this length
        par = None
        if src == "parent" and parent_arr is not None:
            par = parent_arr
        elif best_seq is not None and n != parent_arr.size if parent_arr is not None else True:
            par = best_seq

        t0 = time.monotonic()
        try:
            cand, sc = _optimize_length(n, b, seed_cur, par)
            elapsed = time.monotonic() - t0
            if cand is not None and np.isfinite(sc) and sc < best_score:
                best_seq = cand.copy()
                best_score = sc
                print(f"  L{n} [{src}] score={best_score:.6f} (took {elapsed:.0f}s)", flush=True)
            else:
                print(f"  L{n} [{src}] best_here={sc:.6f} (took {elapsed:.0f}s)", flush=True)
        except Exception as e:
            print(f"  L{n} [{src}] error: {e}", flush=True)

        seed_cur += 9973

    # Final polish on the globally best
    rem = deadline - time.monotonic()
    if best_seq is not None:
        if rem > 30.0:
            try:
                c, sc = _flatten_optimize(best_seq, min(rem * 0.4, 120.0), seed_cur + 1,
                                          power_start=2.0, power_end=5.0)
                if c is not None and sc < best_score:
                    best_seq = c.copy()
                    best_score = sc
                    print(f"  Polish flatten: {best_score:.6f}", flush=True)
            except Exception as e:
                print(f"  Polish flatten error: {e}", flush=True)

        rem = deadline - time.monotonic()
        if rem > 20.0:
            try:
                c, sc = _lp_optimize(best_seq, min(rem * 0.5, 100.0), seed_cur + 2,
                                     max_constraints=400)
                if c is not None and sc < best_score:
                    best_seq = c.copy()
                    best_score = sc
                    print(f"  Polish LP: {best_score:.6f}", flush=True)
            except Exception as e:
                print(f"  Polish LP error: {e}", flush=True)

        rem = deadline - time.monotonic()
        if rem > 15.0:
            try:
                c, sc = _flatten_optimize(best_seq, min(rem * 0.6, 80.0), seed_cur + 3,
                                          power_start=3.0, power_end=8.0)
                if c is not None and sc < best_score:
                    best_seq = c.copy()
                    best_score = sc
                    print(f"  Final flatten: {best_score:.6f}", flush=True)
            except Exception as e:
                print(f"  Final flatten error: {e}", flush=True)

    # Fallback
    if best_seq is None:
        best_seq = np.ones(64, dtype=np.float64) / 64.0
        best_score = _score(best_seq)

    # Final normalization and validation
    final = _normalize(best_seq)
    if final is None:
        final = np.ones(64, dtype=np.float64) / 64.0

    try:
        official = float(evaluate_sequence([float(x) for x in final]))
        if np.isfinite(official):
            best_score = official
    except Exception:
        pass

    print(f"FINAL: score={best_score:.6f} len={len(final)}", flush=True)
    return [float(x) for x in np.clip(final, 0.0, 1000.0)]
