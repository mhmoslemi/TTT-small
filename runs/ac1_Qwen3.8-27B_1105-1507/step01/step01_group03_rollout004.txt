```python
"""
Smooth gradient-based optimization (multiplicative weights / mirror descent)
for minimizing 2*n*max(conv(f,f))/(sum(f))^2 over non-negative sequences.

Uses a log-sum-exp smooth approximation of the max, with adaptive beta schedule
(annealing from smooth to sharp). Combines multiple starting points (given
sequence, uniform, Gaussian, random) and periodic restarts to escape local
minima.
"""

import numpy as np
import time


def normalize(arr):
    """Normalize a non-negative array to sum 1, or return None."""
    x = np.asarray(arr, dtype=np.float64).ravel()
    x = np.where(np.isfinite(x), x, 0.0)
    x = np.maximum(x, 0.0)
    s = float(x.sum())
    if s < 1e-300:
        return None
    return x / s


def score_of(f):
    """Compute score = 2*n*max(conv(f,f)) assuming sum(f)=1, f>=0."""
    c = np.convolve(f, f)
    m = float(np.max(c))
    if not np.isfinite(m):
        return float("inf")
    return float(2.0 * len(f) * m)


def compute_grad(f, beta):
    """
    Compute gradient of LSE(beta * conv(f,f)) / beta with respect to f.
    Returns array of same length as f.
    """
    c = np.convolve(f, f)
    bc = beta * c
    bc -= bc.max()
    w = np.exp(bc)
    w /= w.sum()
    # grad_i = (2/beta) * sum_j w[i+j] * f[j] for i=0..n-1
    grad = (2.0 / beta) * np.correlate(w, f, mode='valid')
    return grad


def interpolate_seq(x, n):
    """Resample a non-negative sequence to length n and normalize."""
    x = normalize(x)
    if x is None or n <= 0:
        return None
    m = x.size
    if m == n:
        return x
    if m == 1:
        return np.ones(n) / float(n)
    old = np.linspace(0.0, 1.0, m)
    new = np.linspace(0.0, 1.0, n)
    out = np.interp(new, old, x)
    return normalize(out)


def mirror_descent_optimize(f_init, n, time_budget, beta_start=2.0, beta_end=2000.0,
                            eta_frac=0.3, patience_frac=0.3, restart_prob=0.02):
    """
    Optimize f using multiplicative weights (mirror descent) with annealed beta.
    
    Parameters:
        f_init: initial point (will be normalized)
        n: sequence length
        time_budget: seconds to spend
        beta_start: initial beta (smoothness parameter)
        beta_end: final beta
        eta_frac: step size fraction (eta = eta_frac * beta)
        patience_frac: fraction of time_budget without improvement before restart
        restart_prob: probability of adding noise each step
    """
    f = normalize(f_init)
    if f is None:
        f = np.ones(n) / n
    f = f.astype(np.float64)

    best_f = f.copy()
    best_score = score_of(f)

    deadline = time.monotonic() + time_budget
    start_time = time.monotonic()
    total_time_est = max(time_budget, 1.0)
    patience = max(3.0, patience_frac * total_time_est)
    last_improve = start_time
    rng = np.random.default_rng(12345)

    total_iters = 0

    while time.monotonic() < deadline:
        # Beta schedule: exponential growth
        elapsed = time.monotonic() - start_time
        frac = min(1.0, elapsed / total_time_est)
        beta = beta_start * (beta_end / beta_start) ** frac
        beta = max(beta, beta_start)

        eta = eta_frac * beta

        # Compute gradient
        grad = compute_grad(f, beta)

        # Multiplicative weights update
        log_f = np.log(f + 1e-300)
        log_f -= eta * grad
        log_f -= log_f.max()  # numerical stability
        f_new = np.exp(log_f)
        f_new = np.maximum(f_new, 0.0)
        s = f_new.sum()
        if s < 1e-300:
            # Fallback: uniform
            f = np.ones(n) / n
            last_improve = time.monotonic() - 1.0  # force patience check
            continue
        f_new = f_new / s

        # Evaluate
        sc = score_of(f_new)
        if sc < best_score:
            best_score = sc
            best_f = f_new.copy()
            last_improve = time.monotonic()

        f = f_new
        total_iters += 1

        # Periodic noise injection to escape local minima
        if rng.random() < restart_prob:
            noise = rng.dirichlet(np.ones(n) * 0.1)
            mix = 0.05 + 0.1 * rng.random()
            f = (1.0 - mix) * best_f + mix * noise
            f = np.maximum(f, 0.0)
            f = f / f.sum()

        # Check patience
        if time.monotonic() - last_improve > patience:
            # Restart from best with perturbation
            eps = 0.1 + 0.2 * rng.random()
            noise = rng.dirichlet(np.ones(n) * 0.5)
            f = (1.0 - eps) * best_f + eps * noise
            f = np.maximum(f, 0.0)
            f = f / f.sum()
            last_improve = time.monotonic()
            beta = beta_start  # reset beta for exploration
            # But keep overall schedule going by adjusting start_time
            start_time = time.monotonic() - (time.monotonic() - start_time - patience_frac * total_time_est)

        # Print progress periodically
        if total_iters % 5000 == 0:
            elapsed = time.monotonic() - start_time
            print(f"    [MD] iter={total_iters}, beta={beta:.1f}, score={best_score:.6f}, "
                  f"elapsed={elapsed:.1f}s", flush=True)

    return best_f, best_score


def adam_optimize(f_init, n, time_budget, beta_start=5.0, beta_end=1000.0,
                  lr=0.05, beta1=0.8, beta2=0.95):
    """
    Adam-style optimization on the simplex with smooth max objective.
    Uses projection onto simplex after each step.
    """
    f = normalize(f_init)
    if f is None:
        f = np.ones(n) / n
    f = f.astype(np.float64)

    m = np.zeros(n)
    v = np.zeros(n)
    t = 0
    best_f = f.copy()
    best_score = score_of(f)

    deadline = time.monotonic() + time_budget
    start_time = time.monotonic()
    total_time_est = max(time_budget, 1.0)

    while time.monotonic() < deadline:
        t += 1
        elapsed = time.monotonic() - start_time
        frac = min(1.0, elapsed / total_time_est)
        beta = beta_start * (beta_end / beta_start) ** frac

        grad = compute_grad(f, beta)

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)

        f_new = f - lr * m_hat / (np.sqrt(v_hat) + 1e-8)
        f_new = np.maximum(f_new, 0.0)
        s = f_new.sum()
        if s < 1e-300:
            f = best_f.copy()
            continue
        f_new = f_new / s

        sc = score_of(f_new)
        if sc < best_score:
            best_score = sc
            best_f = f_new.copy()
        f = f_new

        if t % 5000 == 0:
            elapsed = time.monotonic() - start_time
            print(f"    [Adam] iter={t}, beta={beta:.1f}, score={best_score:.6f}, "
                  f"elapsed={elapsed:.1f}s", flush=True)

    return best_f, best_score


def project_simplex(v):
    """Project vector v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho = np.where(u * np.arange(1, n + 1) > cssv)[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho = rho[-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def projected_grad_optimize(f_init, n, time_budget, beta_start=5.0, beta_end=500.0,
                            lr_start=0.1, lr_end=0.001):
    """
    Projected gradient descent with smooth max, backtracking line search.
    """
    f = normalize(f_init)
    if f is None:
        f = np.ones(n) / n
    f = f.astype(np.float64)

    best_f = f.copy()
    best_score = score_of(f)

    deadline = time.monotonic() + time_budget
    start_time = time.monotonic()
    total_time_est = max(time_budget, 1.0)
    t = 0

    while time.monotonic() < deadline:
        t += 1
        elapsed = time.monotonic() - start_time
        frac = min(1.0, elapsed / total_time_est)
        beta = beta_start * (beta_end / beta_start) ** frac
        lr = lr_start * (lr_end / lr_start) ** frac

        grad = compute_grad(f, beta)
        # Remove mean to project the direction onto the tangent space of the simplex
        grad = grad - grad.mean()

        # Backtracking line search
        step = lr
        for _ in range(20):
            f_trial = f - step * grad
            f_trial = np.maximum(f_trial, 0.0)
            s = f_trial.sum()
            if s < 1e-300:
                step *= 0.5
                continue
            f_trial = f_trial / s
            sc = score_of(f_trial)
            if sc < best_score or step < 1e-10:
                f = f_trial
                break
            step *= 0.5
        else:
            # All line searches failed, just do a small step
            f_trial = f - 1e-10 * grad
            f_trial = np.maximum(f_trial, 0.0)
            s = f_trial.sum()
            if s > 1e-300:
                f = f_trial / s

        sc = score_of(f)
        if sc < best_score:
            best_score = sc
            best_f = f.copy()

        if t % 5000 == 0:
            elapsed = time.monotonic() - start_time
            print(f"    [PGD] iter={t}, beta={beta:.1f}, score={best_score:.6f}, "
                  f"elapsed={elapsed:.1f}s", flush=True)

    return best_f, best_score

                f_res, s_res = mirror_descent_optimize(
                    init_f, n, t_budget,
                    beta_start=params.get('beta_start', 2.0),
                    beta_end=params.get('beta_end', 2000.0),
                    eta_frac=params.get('eta_frac', 0.3)
                )
            elif method == 'adam':
                f_res, s_res = adam_optimize(
                    init_f, n, t_budget,
                    beta_start=params.get('beta_start', 5.0),
                    beta_end=params.get('beta_end', 1000.0),
                    lr=params.get('lr', 0.05)
                )
            elif method == 'pgd':
                f_res, s_res = projected_grad_optimize(
                    init_f, n, t_budget,
                    beta_start=params.get('beta_start', 5.0),
                    beta_end=params.get('beta_end', 500.0)
                )
            else:
                continue
        except Exception as e:
            print(f"    Error in {method}: {e}", flush=True)
            continue

        if np.isfinite(s_res) and s_res < best_score:
            best_score = float(s_res)
            best_f = np.asarray(f_res, dtype=np.float64).copy()
            print(f"  *** New best: n={best_f.size}, score={best_score:.6f} ***", flush=True)

    # Polish phase: take the best and run with very high beta for fine-tuning
    rem = deadline - time.monotonic()
    if rem > 30.0 and best_f is not None:
        print(f"\n=== Polish phase: budget={rem:.0f}s ===", flush=True)
        t_polish = min(120.0, rem - 5.0)
        if t_polish > 10.0:
            # Polish with multiplicative weights at high beta
            f_pol, s_pol = mirror_descent_optimize(
                best_f, best_f.size, t_polish,
                beta_start=100.0, beta_end=10000.0,
                eta_frac=0.15, patience_frac=0.5, restart_prob=0.0
            )
            if s_pol < best_score:
                best_score = s_pol
                best_f = f_pol.copy()
                print(f"  *** Polish improved: score={best_score:.6f} ***", flush=True)

            # Also try Adam polish
            rem2 = deadline - time.monotonic()
            if rem2 > 20.0:
                t_polish2 = min(60.0, rem2 - 5.0)
                if t_polish2 > 10.0:
                    f_pol2, s_pol2 = adam_optimize(
                        best_f, best_f.size, t_polish2,
                        beta_start=200.0, beta_end=5000.0,
                        lr=0.01
                    )
                    if s_pol2 < best_score:
                        best_score = s_pol2
                        best_f = f_pol2.copy()
                        print(f"  *** Adam polish improved: score={best_score:.6f} ***", flush=True)

    # Final: verify and return
    if best_f is None:
        best_f = np.ones(1024) / 1024.0

    seq = [float(max(0.0, x)) for x in best_f]
    try:
        final_score = float(evaluate_sequence(seq))
    except Exception:
        final_score = float("inf")

    if not np.isfinite(final_score):
        # Fallback
        seq = [1.0 / 1024.0] * 1024
        final_score = evaluate_sequence(seq)

    elapsed = time.monotonic() - start_time
    print(f"\nFINAL: n={len(seq)}, local_score={best_score:.6f}, "
          f"evaluate={final_score:.6f}, time={elapsed:.1f}s", flush=True)
    return seq
```
