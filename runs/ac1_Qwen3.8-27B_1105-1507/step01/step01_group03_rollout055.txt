"""
Improved LP-based iterative search with coordinate descent polishing for
minimizing 2*n*max(convolve(f,f))/(sum(f))^2 over nonnegative sequences.

Key improvements over previous version:
1. Proper bisection line search for step size
2. Momentum-smoothed direction updates
3. Coordinate descent polishing phase after LP convergence
4. Structured starting points (triangular, raised-cosine, asymmetric)
5. Adaptive target M that tightens over iterations
6. Multiple diverse restarts with geometric perturbation
7. Try multiple sequence lengths including 2048
"""

import time
import numpy as np
from scipy.optimize import linprog
from scipy.sparse import csr_matrix


def normalize_sequence(arr):
    """Return a nonnegative floating vector normalized to sum 1, or None."""
    try:
        x = np.asarray(arr, dtype=np.float64).ravel()
    except Exception:
        return None
    if x.size == 0:
        return None
    x = np.where(np.isfinite(x), x, 0.0)
    x = np.maximum(x, 0.0)
    s = float(x.sum())
    if s <= 1e-300:
        return None
    x /= s
    return x


def score_normalized(x, n):
    """Score assuming x is nonnegative and sum(x) = 1."""
    conv = np.convolve(x, x)
    m = float(np.max(conv))
    if not np.isfinite(m):
        return float("inf")
    return float(2.0 * n * m)


def make_conv_template(n):
    """Precompute the CSR matrix structure for convolution f*g."""
    m = 2 * n - 1
    k64 = np.arange(m, dtype=np.int64)
    js = np.maximum(0, k64 - n + 1).astype(np.int64)
    je = np.minimum(n - 1, k64).astype(np.int64)
    counts = (je - js + 1).astype(np.int64)
    total = int(counts.sum())

    cum = np.cumsum(counts)
    starts = np.repeat(cum - counts, counts)
    local = np.arange(total, dtype=np.int64) - starts

    row = np.repeat(k64, counts).astype(np.int32)
    col_base = js[row.astype(np.int64)]
    col = (col_base + local).astype(np.int32)

    A_data = np.ones(total, dtype=np.float64)
    A = csr_matrix((A_data, row, col), shape=(m, n), dtype=np.float64)

    # Store the column indices for fast data filling
    col_idx = A.indices.copy()
    row_ptr = A.indptr.copy()

    c_obj = -np.ones(n, dtype=np.float64)
    bounds = [(0.0, None)] * n
    return A, col_idx, row_ptr, c_obj, bounds, m


def build_conv_matrix(col_idx, row_ptr, f, n, m):
    """Build the convolution matrix C_f where (C_f @ g)[k] = (f*g)[k]."""
    n_nonzeros = len(col_idx)
    C_f_data = np.empty(n_nonzeros, dtype=np.float64)
    # For row k, columns are col_idx[row_ptr[k]:row_ptr[k+1]]
    # C_f[k, j] = f[k-j]
    for k in range(m):
        start = row_ptr[k]
        end = row_ptr[k + 1]
        if end > start:
            cols = col_idx[start:end]
            indices = k - cols
            C_f_data[start:end] = f[indices]
    return C_f_data


def solve_direction_lp(col_idx, row_ptr, c_obj, bounds, f, n, M):
    """Solve the LP: max sum(g) s.t. (f*g)[k] <= M for all k, g >= 0."""
    m = 2 * n - 1
    C_f_data = build_conv_matrix(col_idx, row_ptr, f, n, m)
    C_f = csr_matrix((C_f_data, col_idx, row_ptr), shape=(m, n), dtype=np.float64)
    b = np.full(m, M, dtype=np.float64)
    return linprog(c_obj, A_ub=C_f, b_ub=b, bounds=bounds, method="highs")


def line_search_step(f, g, M, n, max_t=1.0):
    """Find best step size t in [0, max_t] using bisection."""
    # We want to minimize score of (1-t)*f + t*g (normalized)
    # The safe bound: H(t) = (1-t)^2*M + 2(1-t)*t*r_max + t^2*G
    # where r_max = max(f*g), G = max(g*g)
    r = np.convolve(f, g)
    r_max = float(np.max(r))
    G = float(np.max(np.convolve(g, g)))

    if r_max >= M * (1.0 - 1e-10):
        return 0.0, None

    # Compute safe upper bound on t
    a_coef = 2.0 * (r_max - M)
    b_coef = G + M - 2.0 * r_max
    if b_coef > 1e-14:
        t_max_safe = -a_coef / b_coef
    else:
        t_max_safe = 1.0

    t_upper = min(max_t, 0.9 * max(t_max_safe, 1e-8))
    if t_upper <= 0:
        return 0.0, None

    # Bisection: find largest t where score improves
    f_current = f
    s_current = M  # max(f*f) currently
    best_t = 0.0
    best_f_new = None
    best_s = s_current

    # Try a set of candidate t values (geometric + bisection)
    candidates = [t_upper, 0.5 * t_upper, 0.25 * t_upper, 0.1 * t_upper,
                  0.05 * t_upper, 0.01 * t_upper, 0.001 * t_upper]
    candidates = [c for c in candidates if c > 1e-10]
    candidates.sort(reverse=True)

    for t in candidates:
        f_new = (1.0 - t) * f + t * g
        f_new = np.maximum(f_new, 0.0)
        ssum = f_new.sum()
        if ssum <= 1e-300:
            continue
        f_new = f_new / ssum
        conv_new = np.convolve(f_new, f_new)
        s_new = float(np.max(conv_new))
        if s_new < best_s - 1e-12:
            best_s = s_new
            best_t = t
            best_f_new = f_new

    if best_f_new is None:
        return 0.0, None
    return best_t, best_f_new


def coordinate_descent_polish(f, n, num_rounds=50, rng=None):
    """Polish f using coordinate descent on the normalized objective."""
    if rng is None:
        rng = np.random.default_rng(0)
    f = f.copy()
    s = float(f.sum())
    if s <= 0:
        return f
    f = f / s  # normalize

    best_score = score_normalized(f, n)
    no_improve = 0

    for rnd in range(num_rounds):
        improved = False
        # Process coordinates in random order
        order = rng.permutation(n)
        for i in order:
            old_val = f[i]
            if old_val < 1e-15:
                continue
            # Try geometric perturbations
            factors = [0.3, 0.5, 0.7, 1.3, 1.5, 2.0, 0.0]
            for fac in factors:
                t = old_val * fac
                if t < 0 or t > 10:
                    continue
                f[i] = t
                new_s = f.sum()
                if new_s < 1e-300:
                    f[i] = old_val
                    continue
                f_norm = f / new_s
                sc = score_normalized(f_norm, n)
                if sc < best_score - 1e-10:
                    f = f_norm
                    best_score = sc
                    improved = True
                    break
            f[i] = f[i] / f.sum() * f.sum()  # no-op, keep as is
            # Re-normalize after each coordinate change
            f = f / f.sum()

        if not improved:
            no_improve += 1
            if no_improve >= 3:
                break
        else:
            no_improve = 0

    return f / f.sum()


def structured_start(n, kind, rng):
    """Generate structured starting sequences."""
    x = np.zeros(n)
    if kind == 'triangular':
        mid = n / 2.0
        x = 1.0 - np.abs(np.arange(n) - mid) / mid
        x = np.maximum(x, 0.01)
    elif kind == 'raised_cosine':
        x = 0.5 * (1 + np.cos(2 * np.pi * np.arange(n) / n))
        x = np.maximum(x, 0.01)
    elif kind == 'asymmetric':
        # Skewed towards one side
        frac = np.arange(n) / n
        alpha = rng.uniform(0.5, 2.0)
        x = frac ** alpha
        x = np.maximum(x, 0.01)
    elif kind == 'bell':
        center = n * rng.uniform(0.3, 0.7)
        width = n * rng.uniform(0.1, 0.4)
        x = np.exp(-0.5 * ((np.arange(n) - center) / width) ** 2)
        x = np.maximum(x, 0.01)
    elif kind == 'flat_edges':
        # Flat in middle, small at edges
        edge = max(1, n // 16)
        x = np.ones(n)
        for i in range(edge):
            decay = 0.1 + 0.9 * i / edge
            x[i] = decay
            x[n - 1 - i] = decay
    elif kind == 'two_peaks':
        c1 = n * rng.uniform(0.2, 0.4)
        c2 = n * rng.uniform(0.6, 0.8)
        w = n * 0.1
        x = np.exp(-0.5 * ((np.arange(n) - c1) / w) ** 2) + \
            np.exp(-0.5 * ((np.arange(n) - c2) / w) ** 2)
        x = np.maximum(x, 0.01)
    else:
        x = np.ones(n)

    # Add small noise
    x += rng.uniform(0, 0.02, n)
    x = np.maximum(x, 0.0)
    s = x.sum()
    if s > 0:
        x /= s
    return x


def interpolate_sequence(x, n):
    """Resample a nonnegative sequence to length n and normalize."""
    x = normalize_sequence(x)
    if x is None or n <= 0:
        return None
    m = x.size
    if m == n:
        return x
    if m == 1:
        return np.ones(n) / float(n)
    old = np.linspace(0.0, 1.0, m)
    new = np.linspace(0.0, 1.0, n)
    out = np.interp(new, old, x)
    return normalize_sequence(out)


def optimize_lp(n, f_init, time_budget, rng, template):
    """
    Run the LP-based iteration with line search and momentum from one start.
    Returns best normalized vector and its score.
    """
    f = normalize_sequence(f_init)
    if f is None:
        f = np.ones(n, dtype=np.float64) / n

    col_idx, row_ptr, c_obj, bounds, m = template[1], template[2], template[3], template[4], template[5]

    best_f = f.copy()
    best_score = score_normalized(f, n)
    start_score = best_score

    deadline = time.monotonic() + max(1.0, time_budget)
    stall = 0
    iteration = 0
    momentum_dir = None
    momentum_beta = 0.3
    target_M_factor = 1.0  # We target slightly below current M to push harder

    while time.monotonic() < deadline:
        iteration += 1
        t0 = time.monotonic()

        conv_ff = np.convolve(f, f)
        M = float(np.max(conv_ff))
        if not np.isfinite(M) or M <= 0:
            break

        # Slightly tighten M to push harder
        M_target = M * max(0.95, 1.0 - 0.01 * iteration)

        try:
            res = solve_direction_lp(col_idx, row_ptr, c_obj, bounds, f, n, M_target)
        except Exception:
            try:
                res = solve_direction_lp(col_idx, row_ptr, c_obj, bounds, f, n, M)
            except Exception:
                stall += 1
                if stall > 15:
                    break
                f = normalize_sequence(f + rng.normal(0, 0.01, n))
                continue

        x = getattr(res, 'x', None)
        if x is None:
            stall += 1
            if stall > 15:
                break
            f = normalize_sequence(f + rng.normal(0, 0.02, n))
            continue

        x = np.asarray(x, dtype=np.float64).ravel()
        x = np.where(np.isfinite(x), x, 0.0)
        x = np.maximum(x, 0.0)
        s_star = float(x.sum())
        if s_star <= 1e-300:
            stall += 1
            if stall > 15:
                break
            f = normalize_sequence(f + rng.normal(0, 0.02, n))
            continue

        # Rescale g so that sum(g) = 1
        G_dir = x / s_star

        # Apply momentum
        if momentum_dir is not None:
            G_dir = (1 - momentum_beta) * G_dir + momentum_beta * momentum_dir
            G_dir = np.maximum(G_dir, 0)
            G_dir = G_dir / G_dir.sum()
        momentum_dir = G_dir.copy()

        # Line search
        t_best, f_new = line_search_step(f, G_dir, M, n, max_t=1.0)

        if f_new is not None:
            sc_new = score_normalized(f_new, n)
            if sc_new < best_score:
                f = f_new
                best_f = f.copy()
                best_score = sc_new
                stall = 0
                if iteration <= 5 or iteration % 20 == 0:
                    print(f"      n={n} iter={iteration} score={best_score:.6f} "
                          f"M={M:.6g} t={t_best:.4f} dt={time.monotonic()-t0:.2f}s",
                          flush=True)
            else:
                # Try a smaller step by scaling the direction
                for scale in [0.5, 0.2, 0.1]:
                    G_dir_scaled = G_dir * scale
                    G_dir_scaled = G_dir_scaled / G_dir_scaled.sum()
                    t_try, f_try = line_search_step(f, G_dir_scaled, M, n, max_t=1.0)
                    if f_try is not None:
                        sc_try = score_normalized(f_try, n)
                        if sc_try < best_score:
                            f = f_try
                            best_f = f.copy()
                            best_score = sc_try
                            stall = 0
                            break
                else:
                    stall += 1
        else:
            stall += 1

        if stall > 0 and stall % 5 == 0:
            # Perturb to escape
            eps = 0.05 + 0.05 * (stall // 5)
            f = normalize_sequence(f * (1 + eps * rng.normal(0, 1, n)))
            if f is None:
                f = best_f.copy()

        if stall > 20:
            break

        # Check time
        if time.monotonic() > deadline - 5.0:
            break

    # Coordinate descent polishing
    if best_score < start_score:
        print(f"      n={n} Doing CD polish from score={best_score:.6f}", flush=True)
        cd_budget = min(60.0, (deadline - time.monotonic()) * 0.5)
        cd_start = time.monotonic()
        max_cd_rounds = min(200, int(cd_budget / 0.5))
        f_polished = coordinate_descent_polish(best_f, n, max_cd_rounds, rng)
        sc_polished = score_normalized(f_polished, n)
        if sc_polished < best_score:
            best_f = f_polished
            best_score = sc_polished
            print(f"      n={n} CD polish improved to {best_score:.6f}", flush=True)

    return best_f, float(best_score)

        for s, name in starts:
            sc = score_normalized(s, n)
            scored_starts.append((sc, s, name))
        scored_starts.sort(key=lambda x: x[0])

        # Allocate time: give more time to better starts
        # Use top starts, with diminishing time
        max_starts = min(len(scored_starts), max(3, int(alloc / 30)))
        selected = scored_starts[:max_starts]

        # Time allocation: first start gets 40%, second 30%, rest split
        m = len(selected)
        if m == 1:
            times = [alloc * 0.95]
        elif m == 2:
            times = [alloc * 0.55, alloc * 0.4]
        elif m == 3:
            times = [alloc * 0.45, alloc * 0.35, alloc * 0.2]
        else:
            times = [alloc * 0.35, alloc * 0.25, alloc * 0.2, alloc * 0.2]
            times = [t / sum(times) for t in times]
            times = [t * alloc for t in times]

        for idx, (sc_init, s_init, name) in enumerate(selected):
            rem = deadline - time.monotonic()
            if rem < 5.0:
                break
            tb = min(times[idx], rem - 2.0)
            if tb < 5.0:
                continue

            print(f"  Start {idx+1}/{m} ({name}), init_score={sc_init:.6f}, "
                  f"budget={tb:.0f}s", flush=True)

            f_res, s_res = optimize_lp(n, s_init, tb, rng, template)

            if np.isfinite(s_res) and s_res < best_score:
                best_score = float(s_res)
                best_f = np.asarray(f_res, dtype=np.float64).copy()
                print(f"  *** NEW BEST: n={best_f.size}, score={best_score:.6f}",
                      flush=True)

        # Clean up template if memory is tight
        if n > 1024:
            del templates[n]

    # Final consolidation
    if best_f is None:
        best_f = np.ones(128) / 128.0

    # Final coordinate descent polishing on the overall best
    if time.monotonic() < deadline - 10.0:
        n_best = best_f.size
        print(f"\nFinal CD polish on n={n_best}, score={best_score:.6f}", flush=True)
        f_polished = coordinate_descent_polish(best_f, n_best, 100, rng)
        sc_polished = score_normalized(f_polished, n_best)
        if sc_polished < best_score:
            best_f = f_polished
            best_score = sc_polished
            print(f"Final CD polish improved to {best_score:.6f}", flush=True)

    seq = [float(x) for x in best_f]
    try:
        final_score = float(evaluate_sequence(seq))
        print(f"\nFINAL: n={len(seq)}, local_best={best_score:.6f}, "
              f"evaluate={final_score:.6f}", flush=True)
    except Exception as e:
        print(f"evaluate_sequence error: {e}", flush=True)
        final_score = float('inf')

    if not np.isfinite(final_score):
        if given_f is not None:
            seq = [float(x) for x in given_f]
        else:
            seq = [1.0 / 128.0] * 128

    return seq
