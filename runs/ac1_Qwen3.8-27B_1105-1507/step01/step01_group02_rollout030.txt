Uses softmax parameterization with Adam optimizer on a log-sum-exp smoothed

Temperature annealing transitions from smooth to sharp objective.

    b_max = np.max(b)
    e = np.exp(b - b_max)
    return e / np.sum(e)

    n = len(a)

def softmax(b):
    b_max = np.max(b)
    e = np.exp(b - b_max)

def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    N = 1 << (size - 1).bit_length()
    fa = np.fft.rfft(a, N)
    fc = fa * fa
    result = np.fft.irfft(fc, N)[:size]
    result[result < 0] = 0.0
    return result

    n = len(v)
    if n == 0:
        return v
    v = np.maximum(v, 0)
    s = v.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return v / s

    c = fast_convolve(a)

    grad_a = 2.0 * cross

    grad_b = grad_a * a - a * dot
    
    return value, grad_b

    def __init__(self, params, lr=0.001, beta1=0.9, beta2=0.999, eps=1e-8):
        self.params = params
        self.lr = lr
        self.beta1 = beta1
        self.beta2 = beta2
        self.eps = eps
        self.m = np.zeros_like(params)
        self.v = np.zeros_like(params)
        self.t = 0
    
    def step(self, grad):
        self.t += 1
        self.m = self.beta1 * self.m + (1 - self.beta1) * grad
        self.v = self.beta2 * self.v + (1 - self.beta2) * grad ** 2
        m_hat = self.m / (1 - self.beta1 ** self.t)
        v_hat = self.v / (1 - self.beta2 ** self.t)
        self.params = self.params - self.lr * m_hat / (np.sqrt(v_hat) + self.eps)
        return self.params

    size = 2 * n - 1
    N = 1 << (size - 1).bit_length()
    fa = np.fft.rfft(a, N)
    fc = fa * fa
    result = np.fft.irfft(fc, N)[:size]

    return result


def project_simplex(v):

    v = np.maximum(v, 0.0)

    peak_indices = np.where(c >= threshold)[0]
    if len(peak_indices) > n_top:
        sorted_idx = np.argsort(c[peak_indices])[::-1][:n_top]
        peak_indices = peak_indices[sorted_idx]
    elif len(peak_indices) == 0:
        peak_indices = np.array([int(np.argmax(c))])

    def __init__(self, params, lr=0.01, beta1=0.9, beta2=0.999, eps=1e-8):
        self.params = params.copy()
        self.lr = lr
        self.beta1 = beta1
        self.beta2 = beta2
        self.eps = eps
        self.m = np.zeros_like(params)
        self.v = np.zeros_like(params)
        self.t = 0

    def step(self, grad):
        self.t += 1
        self.m = self.beta1 * self.m + (1.0 - self.beta1) * grad
        self.v = self.beta2 * self.v + (1.0 - self.beta2) * grad * grad
        m_hat = self.m / (1.0 - self.beta1 ** self.t)
        v_hat = self.v / (1.0 - self.beta2 ** self.t)
        self.params = self.params - self.lr * m_hat / (np.sqrt(v_hat) + self.eps)
        return self.params

    def set_lr(self, lr):
        self.lr = lr

    def reset(self, params):
        self.params = params.copy()
        self.m = np.zeros_like(params)
        self.v = np.zeros_like(params)
        self.t = 0

    # Lengths to try
    lengths_to_try = [512, 768, 1024, 640]

    # Time allocation
    total_time = overall_deadline - time.time() - 60  # reserve 60s for polish
    # Give proportional time, more to 512
    weights = [0.35, 0.25, 0.20, 0.20]
    times = [total_time * w for w in weights]

    for li, n in enumerate(lengths_to_try):
        if time.time() > overall_deadline - 60:
            break

        config_time = times[li]
        config_deadline = min(overall_deadline - 40, time.time() + config_time)
        if config_deadline < time.time() + 20:
            break

        print(f"\n=== n={n}, budget={config_time:.0f}s ===")

        # Multiple initializations per length
        n_inits = 2 if n <= 768 else 1
        for init_i in range(n_inits):
            if time.time() > config_deadline - 15:
                break

            if init_i == 0 and parent_a is not None:
                if len(parent_a) >= n:
                    a0 = parent_a[:n].copy()
                else:
                    a0 = np.interp(np.linspace(0, 1, n),
                                   np.linspace(0, 1, len(parent_a)), parent_a)
                a0 = project_simplex(a0)
            elif init_i == 0:
                a0 = alt_projection_init(n, rng, n_iter=50)
            else:
                x = np.linspace(0, 1, n)
                freq = rng.uniform(0.5, 3.0)
                a0 = 0.5 * (1.0 + np.cos(2.0 * np.pi * freq * x)) + 0.02
                a0 = a0 + 0.05 * rng.random(n)
                a0 = project_simplex(a0)

            b0 = np.log(a0 + 1e-15)
            # Adjust deadline for multiple inits
            init_deadline = config_deadline - (n_inits - 1 - init_i) * 10
            try:
                a_opt, score_opt = optimize_one_length(n, b0, init_deadline, rng)
            except Exception as e:
                print(f"  Error in init {init_i}: {e}")
                continue

            print(f"  n={n}, init={init_i}: score={score_opt:.6f}")

            if score_opt < best_score:
                best_score = score_opt
                best_a = a_opt.copy()
                print(f"  >>> New best: {best_score:.6f}")

    # Final polish
    if best_a is not None and time.time() < overall_deadline - 5:
        print(f"\n=== Final polish (current best={best_score:.6f}) ===")
        n = len(best_a)
        params = np.log(best_a + 1e-15)
        m = np.zeros(n)
        v = np.zeros(n)
        t = 0
        lr = 0.0005
        polish_deadline = overall_deadline
        p_stall = 0

        while time.time() < polish_deadline:
            peak, grad_b = get_subgrad_b(params, n_top=5)
            params, m, v, t = adam_step(params, grad_b, m, v, t, lr)
            params = np.clip(params, -30.0, 30.0)

            if int(time.time() * 4) % 4 == 0:
                a_check = softmax_supply(params)
                s_check = compute_true_score(a_check)
                if s_check < best_score * (1.0 - 1e-8):
                    best_score = s_check
                    best_a = a_check.copy()
                    p_stall = 0
                else:
                    p_stall += 1
                    if p_stall > 300:
                        break

        print(f"  After polish: score={best_score:.6f}")

    if best_a is None:
        best_a = np.ones(512) / 512
        print("Fallback to uniform")

    result = [float(x) for x in best_a]
    final_score = evaluate_sequence(result)
    print(f"\nFINAL: score={final_score:.6f}, length={len(result)}")
    return result
