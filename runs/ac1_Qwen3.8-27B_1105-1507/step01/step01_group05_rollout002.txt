LP-based iterative optimization with diverse restarts and adaptive step sizing
to minimize 2*n*max(convolve(a,a))/sum(a)^2 for non-negative sequences.

import numpy as np
from scipy.optimize import linprog
import time


def build_conv_matrix(f):
    """Build matrix A such that A @ b gives np.convolve(f, b, mode='full')."""
    n = len(f)
    A = np.zeros((2 * n - 1, n))
    for k in range(2 * n - 1):
        j_start = max(0, k - n + 1)
        j_end = min(n, k + 1)
        for j in range(j_start, j_end):
            A[k, j] = f[k - j]
    return A


def compute_score_fast(f):
    """Compute score = 2*n*max(conv)/sum^2."""
    n = len(f)
    S = f.sum()
    if S < 1e-15:
        return np.inf
    conv = np.convolve(f, f)
    M = conv.max()
    return 2.0 * n * M / (S * S)


def compute_score_normalized(f):
    """Compute score assuming sum(f) = 1."""
    n = len(f)
    conv = np.convolve(f, f)
    M = conv.max()
    return 2.0 * n * M

"""
LP-based iterative optimization with diverse structured restarts, adaptive
step sizing, and multi-length search to minimize 2*n*max(convolve(a,a))/sum(a)^2
for non-negative sequences. Combines the active-set LP direction method with
simulated-annealing-style restarts and structured initializations exploiting
the edge-effects of linear convolution.
"""

import numpy as np
from scipy.optimize import linprog
import time


def build_conv_matrix(f):
    """Build sparse-like matrix A such that A @ b = np.convolve(f, b, mode='full')."""
    n = len(f)
    A = np.zeros((2 * n - 1, n))
    for k in range(2 * n - 1):
        j_start = max(0, k - n + 1)
        j_end = min(n, k + 1)
        for j in range(j_start, j_end):
            A[k, j] = f[k - j]
    return A


def compute_score_normalized(f):
    """Compute score = 2*n*max(conv(f,f)) assuming sum(f) = 1."""
    n = len(f)
    conv = np.convolve(f, f)
    M = conv.max()
    return 2.0 * n * M


def compute_score_general(f):
    """Compute score = 2*n*max(conv(f,f))/sum(f)^2."""
    n = len(f)
    S = f.sum()
    if S < 1e-15:
        return np.inf
    conv = np.convolve(f, f)
    M = conv.max()
    return 2.0 * n * M / (S * S)


def optimize_lp(f, time_budget, rng, verbose=False, t_init=0.3):
    """
    LP-based iterative optimization with adaptive step sizing and restarts.
    f: non-negative array (will be normalized to sum=1 internally)
    Returns (best_f, best_score) where best_score uses normalized form.
    """
    n = len(f)
    f = f.astype(np.float64).copy()
    S = f.sum()
    if S < 1e-10:
        return f, np.inf
    f = f / S

    best_f = f.copy()
    best_score = compute_score_normalized(f)

    t = t_init
    t_min = 0.005
    t_max = 0.9
    stall = 0
    max_stall = 25

    start = time.monotonic()
    it = 0
    consecutive_no_improve = 0

    while time.monotonic() - start < time_budget:
        elapsed = time.monotonic() - start
        if elapsed > time_budget - 2.0:
            break

        conv = np.convolve(f, f)
        M = conv.max()

        # Build LP and solve
        A = build_conv_matrix(f)
        c_obj = -np.ones(n)
        b_ub = np.full(2 * n - 1, M)
        bounds = [(0, None)] * n

        try:
            res = linprog(c_obj, A_ub=A, b_ub=b_ub, bounds=bounds,
                         method='highs', options={'presolve': True, 'time_limit': 10})
        except Exception:
            break

        if not res.success or res.x is None:
            break

        b = np.clip(res.x, 0, None)
        b_sum = b.sum()
        if b_sum < 1e-15:
            break

        g = b / b_sum

        # Check if g == f (converged)
        if np.allclose(f, g, atol=1e-10):
            break

        # Try multiple step sizes
        candidates = [t, max(t * 0.5, t_min), min(t * 1.5, t_max), min(t * 2.0, t_max)]
        best_trial = None
        best_trial_score = best_score

        for try_t in set(candidates):
            if try_t < t_min or try_t > t_max:
                continue
            f_new = (1.0 - try_t) * f + try_t * g
            f_new = np.clip(f_new, 0, None)
            S_new = f_new.sum()
            if S_new < 1e-10:
                continue
            f_new = f_new / S_new
            s_new = compute_score_normalized(f_new)
            if s_new < best_trial_score:
                best_trial_score = s_new
                best_trial = f_new
                best_trial_t = try_t

        if best_trial is not None and best_trial_score < best_score - 1e-13:
            best_f = best_trial.copy()
            best_score = best_trial_score
            f = best_trial
            stall = 0
            consecutive_no_improve = 0
            # Nudgestep size up on success

        else:
            stall += 1
            consecutive_no_improve += 1

                if ptype == 0:

                    f_cand = f + rng.normal(0, 0.15, n)
                elif ptype == 1:
                    # Block randomization
                    bs = max(1, n // int(rng.integers(4, 12)))
                    f_cand = f.copy()
                    idx = rng.integers(0, n - bs)
                    f_cand[idx:idx + bs] = rng.random(bs) * f.max()
                elif ptype == 2:
                    # Exponential taper (random rate)

                    f_cand = np.exp(-np.arange(n) * rate / n)
                    f_cand = 0.5 * f_cand + 0.5 * f
                elif ptype == 3:
                    # Reverse and mix
                    f_cand = 0.5 * f + 0.5 * f[::-1]
                    f_cand += 0.05 * rng.random(n)
                elif ptype == 4:
                    # Power taper
                    alpha = rng.uniform(0.3, 2.0)

                else:

                        t = min(t_init * 1.5, t_max)
                        stall = 0
                        if verbose:
                            print(f"    it={it}, RESTART: score_cand={s_c:.6f}, best={best_score:.6f}")

                if time.monotonic() - start > time_budget * 0.9:
                    break

        it += 1

    return best_f, best_score


def make_starts(n, rng, count):
    """Generate diverse starting sequences for length n."""
    starts = []

    # 1. Monotonically decreasing (exponential)
    for rate in [0.5, 1.0, 2.0, 4.0]:
        f = np.exp(-np.arange(n) * rate / max(1, n // 4))
        f += 0.01 * rng.random(n)
        starts.append(f / f.sum())

    # 2. Monotonically decreasing (power law)
    for alpha in [0.5, 1.0, 1.5, 2.5]:
        f = (np.arange(n) + 1) ** (-alpha)
        f += 0.01 * rng.random(n)
        starts.append(f / f.sum())

    # 3. Uniform with noise
    f = np.ones(n) * (1 + 0.2 * rng.normal(0, 1, n))
    starts.append(np.clip(f, 0.01, None) / np.clip(f, 0.01, None).sum())

    # 4. Pure random
    starts.append(rng.random(n) / rng.random(n).sum())

    # 5. Spiky: few large values
    n_spikes = max(2, n // 10)
    f = np.zeros(n)
    spike_positions = rng.choice(n, size=n_spikes, replace=False)
    f[spike_positions] = rng.random(n_spikes) + 0.5
    f += 0.01 * rng.random(n)
    starts.append(f / f.sum())

    # 6. Periodic
    period = max(2, n // int(rng.integers(3, 12)))
    base = np.clip(rng.random(period), 0.01, None)
    f = np.array([base[i % period] for i in range(n)])
    starts.append(f / f.sum())

    # 7. Sine-modulated
    freq = rng.integers(1, max(2, n // 4))
    f = 0.5 + 0.4 * np.sin(2 * np.pi * freq * np.arange(n) / n)
    f = np.clip(f, 0.01, None)
    starts.append(f / f.sum())

    # 8. Two-sided taper (large at both ends)
    f = np.ones(n)
    taper = np.linspace(0.1, 1.0, n // 2)
    f[:n // 2] = taper
    f[n // 2:] = taper[::-1]
    f += 0.02 * rng.random(n)
    starts.append(f / f.sum())

    # Shuffle and take requested count
    rng.shuffle(starts)
    return starts[:count]
