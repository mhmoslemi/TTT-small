def compute_obj(a):

    n = len(a)

        return np.inf

    max_b = float(np.max(conv))

    """Fast convolution using FFT."""
    n = len(a)

    while fft_size < size:
        fft_size *= 2

    conv = np.fft.irfft(fa * fa, fft_size)[:size]
    # Fix negative values due to numerical errors
    conv = np.maximum(conv, 0)
    return conv

Minimize 2*n*max(conv(a,a))/(sum(a))^2 using smooth log-sum-exp L-BFGS
with FFT-accelerated convolution, multiple initializations, and progressive
temperature schedule. Includes coordinate descent polishing.
"""

from scipy.optimize import minimize
import time


def compute_obj(a):
    """Compute true objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    conv = np.convolve(a, a)
    max_b = float(np.max(conv))
    sum_a = float(np.sum(a))
    if sum_a < 0.01:
        return np.inf
    return 2.0 * n * max_b / (sum_a * sum_a)

    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    fa = np.fft.rfft(a, fft_size)
    conv = np.fft.irfft(fa * fa, fft_size)[:size]
    return np.maximum(conv, 0.0)


def smooth_obj_grad(z, n, T):
    """
    Compute smooth objective (log-sum-exp approximation) and gradient
    w.r.t. softmax parameters z.
    Returns (objective_value, gradient_vector).
    """
    # Softmax to get a
    z_max = np.max(z)
    exp_z = np.exp(z - z_max)
    a = exp_z / exp_z.sum()
    
    # Convolution
    conv = convolve_fft(a)
    
    # Log-sum-exp of T * conv
    u = T * conv
    u_max = np.max(u)
    exp_u = np.exp(u - u_max)
    W = exp_u.sum()
    p = exp_u / W  # softmax weights over convolution positions
    
    LSE = u_max + np.log(W)
    max_approx = LSE / T
    
    # Objective (sum(a) = 1 by construction)
    obj = 2.0 * n * max_approx
    
    # Gradient: dL/dz_j = 2n * a_j * (grad_j - E_p[conv])
    # where grad_j = sum_k p_k * a_{k-j} (cross-correlation)
    # Use FFT for cross-correlation
    corr_size = 3 * n  # enough for p (len 2n-1) * a (len n) correlation
    fft_size_c = 1 << (corr_size - 1).bit_length()
    Fp = np.fft.rfft(p, fft_size_c)
    Fa = np.fft.rfft(a, fft_size_c)
    corr = np.fft.irfft(Fp * np.conj(Fa), fft_size_c)[:n]
    
    E_conv = np.dot(p, conv)
    
    grad = 2.0 * n * a * (corr - E_conv)
    
    return obj, grad


def optimize_smooth(a_init, n, T_list, max_iter=150, verbose=False):
    """
    Run L-BFGS-B on smooth objective with progressive temperature.
    a_init: initial sequence (will be converted to z = log(a))
    Returns best (a, obj) found.
    """
    a_init = np.maximum(a_init, 1e-10)
    z = np.log(a_init)
    z = z - z.mean()  # center for numerical stability
    
    best_a = a_init.copy()
    best_obj = compute_obj(best_a)
    
    for idx, T in enumerate(T_list):
        result = minimize(
            smooth_obj_grad, z,
            args=(n, T),
            method='L-BFGS-B',
            jac=True,
            options={'maxiter': max_iter, 'gtol': 1e-10, 'ftol': 1e-12, 'maxls': 50}
        )
        
        z = result.x
        # Convert back to a
        z_max = np.max(z)
        a = np.exp(z - z_max)
        a = a / a.sum()
        a = np.maximum(a, 0.0)
        
        obj = compute_obj(a)
        if obj < best_obj:
            best_obj = obj
            best_a = a.copy()
            if verbose:
                print(f"    T={T}: obj={best_obj:.6f} (iter {idx+1}/{len(T_list)})")
        
        # Re-center z
        z = z - z.mean()
    
    return best_a, best_obj


def coord_descent(a, max_sweeps=50, verbose=False):
    """
    Coordinate descent with 1D exact minimization for each coordinate.
    """
    a = a.copy().astype(np.float64)
    n = len(a)
    best_obj = compute_obj(a)
    
    for sweep in range(max_sweeps):
        improved = False
        conv = np.convolve(a, a)
        S = a.sum()
        
        for j in range(n):
            s_other = S - a[j]
            if s_other < 1e-15:
                continue
            
            # Build d_k and c_k
            # d_k = a[k-j] if 0 <= k-j < n else 0
            # c_k = conv_k - a[j]*d_k
            d = np.zeros(2*n-1)
            lo = max(0, j)
            hi = min(2*n-1, j+n)
            for k in range(lo, hi):
                d[k] = a[k-j]
            
            c = conv.copy()
            c[lo:hi] -= a[j] * d[lo:hi]
            
            # Minimize max_k (x*d_k + c_k) / (x+s_other)^2 over x >= 0
            # Grid search
            x_max = 3.0 * s_other  # upper bound
            x_grid = np.concatenate([[0.0], np.geomspace(1e-4 * s_other, x_max, 80)])
            
            # Vectorized evaluation
            # h[i,k] = (x_grid[i] * d[k] + c[k]) / (x_grid[i] + s_other)^2
            denom = (x_grid[:, None] + s_other) ** 2
            h = (x_grid[:, None] * d[None, :] + c[None, :]) / denom
            g_vals = h.max(axis=1)
            
            j_min = np.argmin(g_vals)
            x_best = x_grid[j_min]
            
            # Golden section refinement
            if 0 < j_min < len(x_grid) - 1:
                lo_g = x_grid[j_min-1]
                hi_g = x_grid[j_min+1]
                gr = (np.sqrt(5) - 1) / 2
                for _ in range(40):
                    x1 = hi_g - gr * (hi_g - lo_g)
                    x2 = lo_g + gr * (hi_g - lo_g)
                    f1 = np.max((x1 * d + c) / (x1 + s_other)**2)
                    f2 = np.max((x2 * d + c) / (x2 + s_other)**2)
                    if f1 < f2:
                        hi_g = x2
                    else:
                        lo_g = x1
                    if hi_g - lo_g < 1e-10 * max(1.0, s_other):
                        break
                x_best = (lo_g + hi_g) / 2
            
            if x_best > 0:
                old_val = a[j]
                a[j] = x_best
                if abs(x_best - old_val) > 1e-10:
                    improved = True
                # Update conv and S
                # conv_k changes by: delta_k = x_best*d_k + c_k - (old_val*d_k + c_k) = (x_best - old_val) * d_k
                delta = (x_best - a[j])  # a[j] already updated, this is 0... 
                # Let me redo: delta_k = (x_best - old_val) * d_k
                # But I already set a[j] = x_best, so let me recompute
                pass  # Skip incremental update, recompute at end of sweep
        
        # Recompute objective at end of sweep
        new_obj = compute_obj(a)
        if new_obj < best_obj:
            best_obj = new_obj
        else:
            # Rollback? Actually, coordinate descent should always improve or keep same
            pass
        
        if not improved:
            break
    
    return a, best_obj


def make_init_uniform(n):
    return np.ones(n) / n


def make_init_two_peak(n, rng):
    """Mass concentrated at the two ends."""
    a = np.ones(n) * 0.3
    m = max(1, n // 5)
    a[:m] = 1.0
    a[-m:] = 1.0
    a = a + rng.normal(0, 0.05, n)
    a = np.maximum(a, 0)
    return a / a.sum()


def make_init_cosine(n, rng):
    """Cosine-shaped initialization."""
    x = np.arange(n)
    phase = rng.uniform(0, np.pi)
    freq = rng.uniform(0.5, 2.0)
    a = np.maximum(0.1, np.cos(freq * np.pi * x / n + phase))
    return a / a.sum()


def make_init_periodic(n, rng):
    """Periodic pattern."""
    period = int(rng.integers(max(2, n // 10), max(3, n // 3)))
    pattern = rng.uniform(0.2, 1.0, period)
    a = np.tile(pattern, n // period + 1)[:n]
    return a / a.sum()


def make_init_edge_emphasis(n, rng):
    """Higher values at edges, lower in middle (sedimentary edge pattern)."""
    x = np.linspace(0, 1, n)
    # Edge emphasis: U-shaped or W-shaped
    shape_type = rng.integers(0, 3)
    if shape_type == 0:
        # U-shaped (high at edges, low in middle)
        a = (x - 0.5)**2 + 0.1
    elif shape_type == 1:
        # W-shaped
        a = np.abs(np.sin(2 * np.pi * x))**rng.uniform(0.5, 2.0) + 0.1
    else:
        # Inverted U with notches
        a = 1 - 0.5 * np.abs(np.sin(np.pi * x))**2 + 0.1
    a = a + rng.normal(0, 0.02, n)
    a = np.maximum(a, 0.01)
    return a / a.sum()


def make_init_random(n, rng):
    """Log-normal random."""
    sigma = rng.uniform(0.3, 1.5)
    a = np.exp(rng.normal(0, sigma, n))
    return a / a.sum()
