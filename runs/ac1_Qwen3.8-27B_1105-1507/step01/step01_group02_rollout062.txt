def iterative_fft_init(n, rng, n_iters=200, freq_target='flat'):
    """
    Gerchberg-Saxton style iterative FFT to find a non-negative sequence
    with approximately flat auto-convolution spectrum.
    """
    N = 1 << (2*n - 2).bit_length()
    
    # Initialize
    a = rng.random(n)
    a = a / a.sum()
    
    for it in range(n_iters):
        # Compute FFT
        F = np.fft.rfft(a, N)
        
        # Modify magnitudes
        mag = np.abs(F)
        if freq_target == 'flat':
            target = np.ones(len(F))
        elif freq_target == 'lowpass_boost':
            # Boost low frequencies slightly to help non-negativity
            weights = 1.0 + 0.3 * np.exp(-np.linspace(0, len(F), len(F)) / (len(F)*0.1))
            target = weights
        else:
            target = np.ones(len(F))
        
        # Interpolate current to target
        alpha = 0.5  # mixing parameter
        new_mag = alpha * target + (1-alpha) * mag
        F = new_mag * np.exp(1j * np.angle(F))

        # IFFT
        a = np.fft.irfft(F, N)[:n]

        # Normalize
        s = a.sum()
        if s < 1e-10:
            a = rng.random(n)
        else:
            a = a / s
    
    return a

"""
Minimizes 2*n*max(convolve(a,a))/sum(a)^2 for non-negative sequences.
Uses iterative FFT (Gerchberg-Saxton) initialization, subgradient descent with
log-sum-exp smoothing, FFT-based exact gradients, momentum, and adaptive
step sizes. Explores multiple lengths and initializations.
"""

import numpy as np
import time


def fast_convolve(a):
    """Compute full linear convolution of a with itself using FFT."""
    n = len(a)
    size = 2 * n - 1
    N = 1 << (size - 1).bit_length()
    fa = np.fft.rfft(a, N)
    result = np.fft.irfft(fa * fa, N)[:size]
    result[result < 0] = 0.0
    return result


def logsumexp_value(c, T):
    """Compute log-sum-exp of c with temperature T."""
    c_max = np.max(c)
    return c_max + T * np.log(np.sum(np.exp((c - c_max) / T)))


def compute_grad_fft(a, w, N_fft):
    """
    Compute r[i] = sum_j w[j] * a[j-i] (linear cross-correlation, first n values).
    Gradient of logsumexp(conv(a,a)) w.r.t. a is 2*r.
    """
    W = np.fft.rfft(w, N_fft)
    A = np.fft.rfft(a, N_fft)
    corr = np.fft.irfft(W * np.conj(A), N_fft)
    return corr[:len(a)]


def iterative_fft_init(n, rng, n_iters=300, boost=0.0, seed_phase=True):
    """
    Gerchberg-Saxton style: iteratively enforce non-negativity in time
    and target spectrum shape in frequency.
    """
    N = 1 << (2 * n - 2).bit_length()
    a = np.abs(rng.normal(0, 1, n)) + 0.05
    a = a / a.sum()

    for it in range(n_iters):
        F = np.fft.rfft(a, N)
        mag = np.abs(F)

        if boost > 0:
            freqs = np.linspace(0, 1, len(F))
            target = 1.0 + boost * np.exp(-freqs / 0.08)
        else:
            target = np.ones(len(F))

        # Anneal mixing: start conservative, become more aggressive
        alpha = 0.2 + 0.5 * min(1.0, it / (n_iters * 0.5))
        new_mag = alpha * target + (1.0 - alpha) * mag
        F = new_mag * np.exp(1j * np.angle(F))
        F[0] = abs(F[0])

        a = np.fft.irfft(F, N)[:n]
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s < 1e-10:
            a = np.abs(rng.normal(0, 1, n)) + 0.05
        a = a / s

    return a


def two_bump_init(n, rng, ratio=0.6):
    """Initialize with mass concentrated in first 'ratio' fraction."""
    a = np.zeros(n)
    split = int(n * ratio)
    a[:split] = 1.0
    a[split:] = 0.1
    a += 0.05 * rng.random(n)
    a = np.maximum(a, 0)
    return a / a.sum()


def subgradient_optimize(a, deadline, rng, verbose=False):
    """
    Subgradient descent with log-sum-exp smoothing, momentum, and
    adaptive step size. Maintains sum(a)=1 and a>=0 via projection.
    """
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-10:
        a = np.ones(n) / n
    else:
        a = a / s

    N_fft = 1 << (4 * n).bit_length()

    best = a.copy()
    best_score = evaluate_sequence(a.tolist())

    # Parameters
    T = 0.05
    T_floor = 0.0003
    lr = 2.0 / np.sqrt(n)
    momentum = 0.85
    v = np.zeros(n)

    stall = 0
    max_stall = 4000
    it = 0
    eval_interval = 500
    no_improve_long = 0

    while time.time() < deadline:
        it += 1

        # Anneal temperature
        progress = min(1.0, it / 40000.0)
        T = max(T_floor, 0.05 * (0.006) ** progress)

        # Compute convolution and softmax weights
        c = fast_convolve(a)
        c_max = np.max(c)
        w = np.exp((c - c_max) / T)
        w_sum = np.sum(w)
        if w_sum > 0:
            w = w / w_sum
        else:
            w = np.zeros(2 * n - 1)
            w[n - 1] = 1.0

        # Gradient
        r = compute_grad_fft(a, w, N_fft)
        grad = 2.0 * r

        # Momentum update
        v = momentum * v + (1.0 - momentum) * grad
        eff_grad = v / (1.0 - momentum ** min(it, 100))  # bias correction (approx)

        # Adaptive lr: scale by T to handle changing gradient scale
        lr_eff = lr * max(1.0, 0.05 / T)
        lr_eff = min(lr_eff, 2.0)

        # Projected step
        a_new = a - lr_eff * eff_grad
        a_new = np.maximum(a_new, 0.0)
        s_new = a_new.sum()
        if s_new < 1e-10:
            a_new = a
        else:
            a = a_new / s_new

        # Periodic evaluation
        if it % eval_interval == 0:
            score = evaluate_sequence(a.tolist())
            if score < best_score * (1.0 - 1e-9):
                prev_best = best_score
                best_score = score
                best = a.copy()
                stall = 0
                no_improve_long = 0
                if verbose:
                    print(f"    it={it}, score={score:.6f} (was {prev_best:.6f}), T={T:.5f}")
            else:
                stall += 1
                if score > best_score * 1.01:
                    no_improve_long += 1

            # Adapt step size
            if stall >= max_stall:
                lr *= 0.6
                momentum = max(0.7, momentum - 0.02)
                stall = 0

            # Strong perturbation when stuck for a long time
            if no_improve_long > 20:
                no_improve_long = 0
                # Random restart blending
                t_blend = 0.3 + 0.4 * rng.random()
                if rng.random() < 0.5:
                    a_random = np.abs(rng.normal(0, 1, n)) + 0.01
                    a_random = a_random / a_random.sum()
                else:
                    a_random = iterative_fft_init(n, rng, n_iters=50, boost=rng.random() * 0.5)
                a = (1.0 - t_blend) * best + t_blend * a_random
                a = np.maximum(a, 0)
                a = a / a.sum()
                v = np.zeros(n)

        # Gentle periodic shake
        if it % 8000 == 0:
            shake_temp = 0.02 * max(0.0, 1.0 - it / 60000.0)
            if shake_temp > 0.002:
                a = np.maximum(a + rng.normal(0, shake_temp / np.sqrt(n), n), 0)
                a = a / a.sum()

        if verbose and it % 20000 == 0:
            cur_score = evaluate_sequence(a.tolist())
            print(f"    it={it}, T={T:.5f}, lr={lr:.4f}, cur={cur_score:.6f}, best={best_score:.6f}")

    # Final check
    score = evaluate_sequence(a.tolist())
    if score < best_score:
        best_score = score
        best = a.copy()

    return best, best_score


def targeted_refine(a, deadline, rng, target_lag=None):
    """
    Additional refinement focused on reducing specific peak lags.
    Tries to reduce the maximum by adjusting elements that contribute most.
    """
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0.0)
    a = a / a.sum()

    best = a.copy()
    best_score = evaluate_sequence(a.tolist())

    N_fft = 1 << (4 * n).bit_length()
    it = 0

    while time.time() < deadline:
        it += 1
        c = fast_convolve(a)
        peak_idx = np.argmax(c)
        peak_val = c[peak_idx]

        # Find which pairs (i, j) with i+j = peak_idx contribute most
        # c[peak_idx] = sum_i a[i]*a[peak_idx - i]
        lo = max(0, peak_idx - n + 1)
        hi = min(n - 1, peak_idx)
        # Elements a[lo:hi+1] pair with a[peak_idx-hi:peak_idx-lo+1][::-1]
        pairs = a[lo:hi + 1] * a[peak_idx - hi:peak_idx - lo + 1][::-1]

        # Reduce the largest contributors
        top_k = min(10, len(pairs))
        top_positions = np.argsort(pairs)[::-1][:top_k]

        for pos in top_positions:
            idx1 = lo + pos
            idx2 = peak_idx - idx1
            # Reduce the larger of the two, shift mass to a less contributing position
            if a[idx1] > a[idx2]:
                reduce_idx = idx1
            else:
                reduce_idx = idx2

            # Find a "quiet" position to shift mass to
            # Positions that don't contribute to the peak
            quiet_candidates = []
            for q in range(n):
                if q == reduce_idx:
                    continue
                if q + reduce_idx == peak_idx:
                    continue  # would still contribute
                if 2 * q == peak_idx and q == reduce_idx:
                    continue
                quiet_candidates.append(q)

            if quiet_candidates:
                # Pick a random quiet position
                q = quiet_candidates[rng.integers(len(quiet_candidates))]
                eps = 0.01 * a[reduce_idx]
                a[reduce_idx] -= eps
                a[q] += eps
                a = np.maximum(a, 0)
                a = a / a.sum()

        # Evaluate
        if it % 200 == 0:
            score = evaluate_sequence(a.tolist())
            if score < best_score:
                best_score = score
                best = a.copy()

    return best, best_score

def _get_init(n, init_type, boost, rng, parent):
    """Get initial sequence based on type."""
    if init_type == 'parent' and parent is not None:
        if len(parent) >= n:
            a = parent[:n].copy()
        else:
            a = np.zeros(n)
            a[:len(parent)] = parent
        a = np.maximum(a, 0)
        s = a.sum()
        return a / s if s > 1e-10 else np.ones(n) / n
    elif init_type == 'fft':
        return iterative_fft_init(n, rng, n_iters=200, boost=boost)
    elif init_type == 'twobump':
        return two_bump_init(n, rng, ratio=0.5 + 0.3 * rng.random())
    elif init_type == 'random':
        a = np.abs(rng.normal(0, 1, n)) + 0.01
        return a / a.sum()
    else:
        a = np.ones(n)
        a += 0.1 * rng.random(n)
        return a / a.sum()
