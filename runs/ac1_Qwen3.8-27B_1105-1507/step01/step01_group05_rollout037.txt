def compute_score_normalized(f):
    """Compute score for a sequence (will normalize internally)."""
    S = f.sum()
    if S < 1e-15:
        return np.inf
    fn = f / S
    n = len(fn)
    conv = np.convolve(fn, fn)

def smooth_conv_max_and_grad(f, tau):
    """
    Compute smooth approximation of max(conv(f,f)) and its gradient.
    f should have sum=1.
    Returns (smooth_value, gradient).
    """
    n = len(f)
    conv = np.convolve(f, f)  # length 2n-1
    
    # Smooth max: (1/tau) * log(sum exp(tau * conv_k))
    max_c = conv.max()
    exp_vals = np.exp(tau * (conv - max_c))
    exp_sum = exp_vals.sum()
    smooth_val = max_c + np.log(exp_sum) / tau
    weights = exp_vals / exp_sum

    # d(conv_k)/d(f_i) = 2*f[k-i] if 0 <= k-i < n, else 0

    # = 2 * sum_{j=0}^{n-1} weights[i+j] * f[j] (where i+j < 2n-1)

    # T[i,j] = weights[i+j] if i+j < 2n-1, else 0
    # grad = 2 * T @ f

    # But for simplicity and correctness, use direct approach for n <= 4096

        j_idx = np.arange(n)[None, :]
        k_idx = i_idx + j_idx
        T = np.zeros((n, n), dtype=np.float64)

        T[valid] = weights[k_idx[valid]]

            if end > 0:

                          tau_max=1000.0, lr=0.01, verbose=True, rng=None):

    f should be non-negative, sum=1.

        rng = np.random.default_rng(42)
    
    n = len(f)

    S = f.sum()
    f = f / S
    
    best_f = f.copy()
    best_score = compute_score_normalized(f)
    
    tau = initial_tau
    m = np.zeros(n)  # Adam first moment
    v = np.zeros(n)  # Adam second moment
    t = 0
    beta1, beta2, eps = 0.9, 0.999, 1e-8
    
    start_time = time.monotonic()
    
    while time.monotonic() - start_time < time_budget:
        # Compute gradient
        smooth_val, grad = smooth_conv_max_and_grad(f, tau)
        
        # Adam update
        t += 1
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * grad ** 2
        m_hat = m / (1 - beta1 ** t)
        v_hat = v / (1 - beta2 ** t)
        
        # Update in the negative gradient direction
        f_new = f - lr * m_hat / (np.sqrt(v_hat) + eps)
        
        # Project onto simplex {x >= 0, sum(x) = 1}

        if score < best_score:

            best_score = score
        
        f = f_new
        
        # Increase tau periodically

            print(f"      Adam it={t}, score={best_score:.6f}, tau={tau:.1f}, elapsed={elapsed:.1f}s")

"""
Hybrid optimization for minimizing 2*n*max(convolve(a,a))/sum(a)^2.
Uses FFT-accelerated smooth-max Adam optimization followed by LP polishing.
Explores multiple sequence lengths and starting points within time budget.
"""

import numpy as np
from scipy.optimize import linprog
import time


def compute_score_normalized(f):
    """Compute score = 2*n*max(conv) for normalized f (sum=1)."""
    S = f.sum()
    if S < 1e-15:
        return np.inf
    fn = f / S
    n = len(fn)
    conv = np.convolve(fn, fn)
    return 2.0 * n * conv.max()


def smooth_conv_max_and_grad(f, tau):
    """
    Compute smooth approximation of max(conv(f,f)) and its gradient.
    f should have sum=1 and be non-negative.
    Returns (smooth_value, gradient_array).
    """
    n = len(f)
    conv = np.convolve(f, f)  # length 2n-1
    max_c = conv.max()
    shifted = tau * (conv - max_c)
    exp_shifted = np.exp(shifted)
    exp_sum = exp_shifted.sum()
    smooth_val = max_c + np.log(exp_sum) / tau
    weights = exp_shifted / exp_sum

    # Gradient: grad[i] = 2 * sum_j weights[i+j] * f[j] (for valid i+j < 2n-1)
    # Use FFT: c = irfft(conj(rfft(f)) * rfft(weights))[:n]
    target = 3 * n - 2
    N = 1 << (target - 1).bit_length()
    F_fft = np.fft.rfft(f, N)
    W_fft = np.fft.rfft(weights, N)
    corr = np.fft.irfft(np.conj(F_fft) * W_fft, N)
    grad = 2.0 * corr[:n]

    return smooth_val, grad


def adam_smooth_optimize(f, time_budget, initial_tau=5.0, tau_growth=1.03,

    f = f / S

    best_score = compute_score_normalized(f)

    tau = initial_tau
    m = np.zeros(n)
    v = np.zeros(n)
    t = 0
    beta1, beta2, eps = 0.85, 0.999, 1e-8

    start_time = time.monotonic()
    patience = 0
    best_in_window = best_score

    while time.monotonic() - start_time < time_budget:
        smooth_val, grad = smooth_conv_max_and_grad(f, tau)

        t += 1
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)

def build_conv_matrix(f):
    """Build matrix A such that A @ b = np.convolve(f, b, 'full')."""
    n = len(f)
    A = np.zeros((2 * n - 1, n))
    for k in range(2 * n - 1):
        j_start = max(0, k - n + 1)
        j_end = min(n, k + 1)
        for j in range(j_start, j_end):
            A[k, j] = f[k - j]
    return A


def lp_polish(f, time_budget, t0=0.5, verbose=False, rng=None):

    LP-based polishing: find direction g that reduces max conv, blend with f.

        rng = np.random.default_rng(0)

    f = f.astype(np.float64).copy()
    S0 = f.sum()
    if S0 < 1e-10:
        return f, np.inf
    f = f / S0

    best_f = f.copy()
    best_score = compute_score_normalized(f)
    t = t0
    start = time.monotonic()
    it = 0
    stall_count = 0

    while time.monotonic() - start < time_budget:
        conv = np.convolve(f, f)
        M = conv.max()

        A = build_conv_matrix(f)
        c_obj = -np.ones(n)
        b_ub = np.full(2 * n - 1, M)
        bounds = [(0, None)] * n

        try:
            res = linprog(c_obj, A_ub=A, b_ub=b_ub, bounds=bounds,
                          method='highs', options={'presolve': True})
        except Exception:
            break

        if not res.success or res.x is None:
            break

        b = np.clip(res.x, 0, None)
        b_sum = b.sum()
        if b_sum < 1e-15:
            break

        g = b / b_sum

        f_new = (1 - t) * f + t * g
        f_new = np.clip(f_new, 0, None)
        S_new = f_new.sum()
        if S_new < 1e-10:
            t *= 0.5
            continue
        f_new = f_new / S_new

        new_score = compute_score_normalized(f_new)

        if new_score < best_score - 1e-10:
            best_f = f_new.copy()
            best_score = new_score
            f = f_new
            stall_count = 0
            if verbose and it % 3 == 0:
                elapsed = time.monotonic() - start
                print(f"    LP it={it}, score={best_score:.6f}, t={t:.4f}, elapsed={elapsed:.1f}s")
        else:
            t *= 0.7
            stall_count += 1
            if stall_count > 30:
                # Random perturbation to escape
                f_cand = f + rng.normal(0, 0.02 / n * n, n)
                f_cand = np.clip(f_cand, 0, None)
                S_c = f_cand.sum()
                if S_c > 1e-10:
                    f_cand /= S_c
                    if compute_score_normalized(f_cand) < best_score:
                        f = f_cand
                        best_f = f.copy()
                        best_score = compute_score_normalized(f)
                        t = t0 * 0.5
                        stall_count = 0
                if time.monotonic() - start > time_budget * 0.85:
                    break

        it += 1

    return best_f, best_score


def local_search_block(f, rng, n_iter=3000, verbose=False):
    """Block perturbation local search to polish a sequence."""
    n = len(f)
    f = f.copy()
    S = f.sum()
    f = f / S
    best = f.copy()
    best_score = compute_score_normalized(f)

    for i in range(n_iter):
        f_cand = f.copy()
        # Random block perturbation
        width = max(1, int(rng.integers(1, max(2, n // 16))))
        start = int(rng.integers(0, n - width))
        scale = np.exp(rng.normal(0, 0.2))
        f_cand[start:start + width] *= scale
        f_cand = np.clip(f_cand, 0, None)
        S_new = f_cand.sum()
        if S_new < 1e-10:
            continue
        f_cand = f_cand / S_new
        score = compute_score_normalized(f_cand)
        if score < best_score:
            best = f_cand.copy()
            best_score = score
            f = f_cand

    return best, best_score


def directed_reduction(f, n_iters=100, verbose=False):
    """
    Targeted reduction: identify hot spots in convolution and reduce them
    by decreasing contributing f values and redistributing mass.
    """
    n = len(f)
    f = f.copy()
    S = f.sum()
    f = f / S
    best_f = f.copy()
    best_score = compute_score_normalized(f)

    for it in range(n_iters):
        conv = np.convolve(f, f)
        M = conv.max()
        # Find positions near max
        threshold = M * 0.95
        hot_positions = np.where(conv >= threshold)[0]

        improvement = False
        for k in hot_positions:
            # Identify contributing pairs (j, k-j)
            j_start = max(0, k - n + 1)
            j_end = min(n, k + 1)
            js = range(j_start, j_end)

            # Reduce the largest contributors
            contributions = [(f[j] * f[k - j], j) for j in js]
            contributions.sort(reverse=True)

            # Reduce top few contributors by a small amount
            for _, j in contributions[:3]:
                j2 = k - j
                if j == j2:
                    # Same index
                    delta = 0.01 * f[j]
                    f[j] -= delta
                    # Redistribute to a random cold position
                    cold = np.argmin(conv)
                    f[cold] += delta
                else:
                    delta = 0.005 * f[j]
                    f[j] -= delta
                    f[j2] -= delta
                    # Redistribute
                    cold = np.argmin(conv)
                    f[cold] += 2 * delta

        f = np.clip(f, 0, None)
        S_new = f.sum()
        if S_new < 1e-10:
            break
        f = f / S_new
        score = compute_score_normalized(f)
        if score < best_score:
            best_f = f.copy()
            best_score = score
            improvement = True

        if not improvement and it > 10:
            break

    return best_f, best_score


def propose_candidate(seed=42, budget_s=1500):
    """
    Main search function: hybrid Adam + LP + local search.
    Tries multiple sequence lengths and starting points.
    Returns the best sequence found.
    """
    rng = np.random.default_rng(seed)
    t_start = time.monotonic()
    deadline = t_start + budget_s - 15.0

    best_overall = None
    best_score_overall = np.inf

    # Get parent sequence if available
    parent = None
    try:
        parent = np.array(height_sequence_1, dtype=np.float64)
        if parent.ndim != 1 or len(parent) == 0 or not np.isfinite(parent).all():
            parent = None
        if parent is not None:
            parent = np.clip(parent, 0, 1000)
    except Exception:
        parent = None

    # Define configurations: (name, n, start_mode)
    configs = []

    n_values = [128, 256, 512, 1024]
    for n in n_values:
        configs.append((f"adam_uniform_{n}", n, 'uniform'))
        configs.append((f"adam_ramp_{n}", n, 'ramp'))

    if parent is not None:
        n_p = len(parent)
        if 64 <= n_p <= 2048:
            configs.append((f"adam_parent_{n_p}", n_p, 'parent'))
            # Also try resampling parent to other sizes
            for target_n in [256, 512]:
                if target_n != n_p:
                    configs.append((f"adam_parent_resample_{target_n}", target_n, 'parent_resample'))

    # Also add structured starts
    configs.append(("adam_sin_512", 512, 'sin'))
    configs.append(("adam_cos_512", 512, 'cos'))

    print(f"Total configs: {len(configs)}, budget: {budget_s}s")
    print(f"Target: 1.5027, Previous best: 1.515592")

    # Strategy: Run quick Adam on all configs, then LP polish the best ones
    phase1_time = min(120.0, (budget_s - 60.0) * 0.4)  # Phase 1: Adam
    phase2_time = min(300.0, (budget_s - 60.0) * 0.4)  # Phase 2: LP polish
    phase3_time = (budget_s - 60.0) * 0.2  # Phase 3: Final polish

    # Phase 1: Quick Adam optimization on all configs
    time_per_config_p1 = max(10.0, phase1_time / len(configs))
    results = []  # (score, f, name, n)

    print("\n=== Phase 1: Adam Smooth Optimization ===")
    for name, n, start_mode in configs:
        remaining = deadline - time.monotonic()
        if remaining < 30:
            break

        # Create starting sequence
        if start_mode == 'uniform':
            f0 = np.ones(n)
            f0 *= (1 + 0.1 * rng.standard_normal(n))
            f0 = np.clip(f0, 0.01, None)
        elif start_mode == 'ramp':
            f0 = np.linspace(0.1, 1.0, n)
            f0 *= (1 + 0.05 * rng.standard_normal(n))
            f0 = np.clip(f0, 0.01, None)
        elif start_mode == 'parent':
            f0 = parent.copy()
        elif start_mode == 'parent_resample':
            # Interpolate parent to target length
            x_old = np.linspace(0, 1, len(parent))
            x_new = np.linspace(0, 1, n)
            f0 = np.interp(x_new, x_old, parent)
            f0 = np.clip(f0, 0.01, None)
        elif start_mode == 'sin':
            x = np.linspace(0, 2 * np.pi, n)
            f0 = 1.0 + 0.3 * np.sin(x)
            f0 = np.clip(f0, 0.01, None)
        elif start_mode == 'cos':
            x = np.linspace(0, 2 * np.pi, n)
            f0 = 1.0 + 0.3 * np.cos(2 * x)
            f0 = np.clip(f0, 0.01, None)
        else:
            f0 = np.ones(n)

        f0 = np.clip(f0, 0, 1000)
        if f0.sum() < 0.01:
            continue
        f0 = f0 / f0.sum()

        t_alloc = min(time_per_config_p1, max(10.0, remaining / 2))
        f_opt, score = adam_smooth_optimize(f0, t_alloc, initial_tau=5.0,
                                             tau_growth=1.05, tau_max=200.0,
                                             lr=0.003, verbose=False, rng=rng)

        results.append((score, f_opt.copy(), name, n))
        if score < best_score_overall:
            best_score_overall = score
            best_overall = f_opt.copy()
            print(f"  [{name}] n={n}, score={score:.6f} ** NEW BEST **")
        else:
            print(f"  [{name}] n={n}, score={score:.6f}")

    # Sort results by score
    results.sort(key=lambda x: x[0])

    # Phase 2: LP polish on top results
    print(f"\n=== Phase 2: LP Polishing (top {min(6, len(results))}) ===")
    top_results = results[:min(6, len(results))]
    time_per_lp = max(30.0, phase2_time / max(1, len(top_results)))

    for score, f, name, n in top_results:
        remaining = deadline - time.monotonic()
        if remaining < 20:
            break

        t_alloc = min(time_per_lp, max(15.0, remaining / 2))
        print(f"  Polishing [{name}] n={n}, current={score:.6f}, time={t_alloc:.0f}s")

        f_lp, score_lp = lp_polish(f, t_alloc, t0=0.4, verbose=False, rng=rng)

        if score_lp < score:
            score = score_lp
            f = f_lp
            print(f"    LP improved: {score:.6f}")

        # Quick directed reduction
        remaining2 = deadline - time.monotonic()
        if remaining2 > 10:
            f_dr, score_dr = directed_reduction(f, n_iters=50)
            if score_dr < score:
                score = score_dr
                f = f_dr
                print(f"    Directed reduction: {score:.6f}")

        if score < best_score_overall:
            best_score_overall = score
            best_overall = f.copy()
            print(f"    ** NEW BEST: {score:.6f} **")

    # Phase 3: Final intensive optimization on the best
    print(f"\n=== Phase 3: Final Intensive Optimization ===")
    if best_overall is not None:
        n_best = len(best_overall)
        remaining = deadline - time.monotonic()

        # Run longer Adam on the best
        if remaining > 60:
            t_alloc = min(remaining * 0.4, 120.0)
            print(f"  Final Adam on n={n_best}, time={t_alloc:.0f}s")
            f_final, score_final = adam_smooth_optimize(
                best_overall, t_alloc, initial_tau=20.0,
                tau_growth=1.02, tau_max=500.0, lr=0.001,
                verbose=True, rng=rng)
            if score_final < best_score_overall:
                best_score_overall = score_final
                best_overall = f_final.copy()
                print(f"    ** NEW BEST: {score_final:.6f} **")

        # LP polish
        remaining2 = deadline - time.monotonic()
        if remaining2 > 30:
            t_alloc = min(remaining2 * 0.4, 100.0)
            print(f"  Final LP on n={n_best}, time={t_alloc:.0f}s")
            f_lp, score_lp = lp_polish(best_overall, t_alloc, t0=0.3,
                                        verbose=True, rng=rng)
            if score_lp < best_score_overall:
                best_score_overall = score_lp
                best_overall = f_lp.copy()
                print(f"    ** NEW BEST: {score_lp:.6f} **")

        # Local search
        remaining3 = deadline - time.monotonic()
        if remaining3 > 15:
            n_iter = min(5000, int(remaining3 * 200))
            print(f"  Final local search, n_iter={n_iter}")
            f_ls, score_ls = local_search_block(best_overall, rng, n_iter=n_iter)
            if score_ls < best_score_overall:
                best_score_overall = score_ls
                best_overall = f_ls.copy()
                print(f"    ** NEW BEST: {score_ls:.6f} **")

    # Also try a few random restarts if time permits
    remaining = deadline - time.monotonic()
    if best_overall is not None and remaining > 40:
        n_best = len(best_overall)
        n_restarts = min(3, int(remaining / 40))
        print(f"\n=== Random Restarts: {n_restarts} tries ===")
        for r in range(n_restarts):
            remaining = deadline - time.monotonic()
            if remaining < 20:
                break
            f0 = best_overall * (1 + 0.3 * rng.standard_normal(n_best))
            f0 = np.clip(f0, 0, None)
            if f0.sum() < 1e-10:
                continue
            f0 = f0 / f0.sum()
            t_alloc = remaining / (n_restarts - r) - 5
            if t_alloc < 10:
                break
            f_r, score_r = adam_smooth_optimize(f0, t_alloc, initial_tau=10.0,
                                                 tau_growth=1.05, tau_max=300.0,
                                                 lr=0.002, verbose=False, rng=rng)
            if score_r < best_score_overall:
                best_score_overall = score_r
                best_overall = f_r.copy()
                print(f"  Restart {r}: ** NEW BEST: {score_r:.6f} **")

    # Final verification
    if best_overall is not None:
        best_overall = np.clip(best_overall, 0, 1000).astype(np.float64)
        final_list = best_overall.tolist()
        final_score = evaluate_sequence(final_list)
        total_time = time.monotonic() - t_start
        print(f"\n{'='*50}")
        print(f"FINAL RESULT: score={final_score:.6f}, n={len(final_list)}, time={total_time:.1f}s")
        print(f"Target: 1.5027")
        print(f"{'='*50}")
        return final_list

    # Fallback
    print("Fallback to uniform sequence")
    return [1.0] * 256
