def compute_smooth_obj_and_grad(a, T):
    """Compute smooth objective value and gradient.
    Assumes a is already normalized (sum=1) and non-negative.
    Returns (objective_value, gradient)."""
    n = len(a)
    s = 1
    while s < 2 * n - 1:
        s <<= 1
    A = np.fft.rfft(a, s)
    conv = np.fft.irfft(A * A, s)[:2*n-1]
    # Smooth objective
    m = conv.max()
    lse = m + np.log(np.sum(np.exp(conv - m))) / T
    obj = 2.0 * n * lse
    # Gradient
    p = np.exp(T * (conv - m))
    p /= p.sum()

    p_pad[:2*n-1] = p
    P = np.fft.fft(p_pad)
    a_pad = np.zeros(s)
    a_pad[:n] = a
    A_full = np.fft.fft(a_pad)
    xcorr = np.fft.ifft(P * np.conj(A_full)).real[:n]
    grad = 4.0 * n * xcorr
    return obj, grad

"""
Minimizes J(a) = 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Analytical gradient via FFT cross-correlation (O(n log n) per step)
2. Log-sum-exp smoothing with adaptive temperature annealing
3. Projected gradient descent with Nesterov momentum on the simplex
4. Spectral flattening kicks to escape local minima
5. Multi-length exploration with diverse initializations
6. Iterative best-response (LP-free) refinement via dual averaging
"""

import numpy as np
import time

def fast_conv(a):
    """Full self-convolution via FFT."""
    n = len(a)
    s = 1
    while s < 2 * n - 1:
        s <<= 1
    fa = np.fft.rfft(a, s)
    return np.fft.irfft(fa * fa, s)[:2 * n - 1]

def true_score(a):
    """Compute J(a) = 2n*max(conv)/sum^2."""
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 0.01:
        return np.inf
    conv = fast_conv(a)
    return 2.0 * len(a) * conv.max() / (s * s)

def get_fft_size(n):
    s = 1
    while s < 2 * n - 1:
        s <<= 1
    return s

def compute_smooth_obj_and_grad(a, T, s=None):
    """Compute smooth objective and gradient for normalized non-negative a."""
    n = len(a)
    if s is None:
        s = get_fft_size(n)
    A = np.fft.rfft(a, s)
    conv = np.fft.irfft(A * A, s)[:2 * n - 1]
    m = conv.max()
    lse = m + np.log(np.sum(np.exp(conv - m))) / T
    obj = 2.0 * n * lse
    p = np.exp(T * (conv - m))
    p /= p.sum()
    p_pad = np.zeros(s)
    p_pad[:2 * n - 1] = p
    P = np.fft.fft(p_pad)
    a_pad = np.zeros(s)
    a_pad[:n] = a
    A_full = np.fft.fft(a_pad)
    xcorr = np.fft.ifft(P * np.conj(A_full)).real[:n]
    grad = 4.0 * n * xcorr
    return obj, grad

def spectral_flatten(a, alpha, beta):
    """Spectral magnitude compression for flattening."""
    n = len(a)
    s = get_fft_size(n)
    A = np.fft.rfft(a, s)
    mag = np.abs(A)
    phase = np.angle(A)
    mag_new = np.abs(mag) ** alpha
    A_new = mag_new * np.exp(1j * phase)
    a_new = np.fft.irfft(A_new, s)[:n]
    a_new = np.maximum(a_new, 0.0)
    s_new = a_new.sum()
    if s_new < 1e-10:
        return a.copy()
    a_new = a_new / s_new
    a_curr = a / a.sum()
    return (1.0 - beta) * a_curr + beta * a_new

def best_response_step(a, M, iterations=50):
    """Iterative best-response: find g maximizing sum(g) s.t. (a*g)_k <= M, g >= 0.
    Uses multiplicative weight updates (Frank-Wolfe style)."""
    n = len(a)
    s = get_fft_size(n)
    A_fft = np.fft.rfft(a, s)
    g = np.ones(n) / n
    for _ in range(iterations):
        g_agg = np.fft.irfft(np.abs(A_fft) * np.fft.rfft(g, s), s)[:n]
        # Multiplicative update: g_j *= exp(-eta * sum_k p[k]*(a[k-j]+a[j-k]))
        # Simpler: g_j /= (a*g_full)_j ... use dual update
        conv_ag = fast_conv(np.concatenate([a, g]))[:2*n-1]
        # Actually let's just do: gradient of sum_k relu((a*g)[k] - M) wrt g
        # d(a*g)[k]/dg[j] = a[|k-j|] or similar
        # Simpler approach: g_new[j] proportional to exp(-lambda * (gradient_j))
        # Use subgradient: for each k where (a*g)[k] > M, push g down at positions that contribute
        # Efficient subgradient via FFT:
        # d/dg[j] of (a*g)[k] = a[k-j] (for valid k-j)
        # So subgrad[j] = sum_{k: (a*g)[k]>M} a[k-j]
        excess = np.maximum(conv_ag - M, 0)
        excess_pad = np.zeros(s)
        excess_pad[:2*n-1] = excess
        E = np.fft.fft(excess_pad)
        a_pad = np.zeros(s)
        a_pad[:n] = a
        A_f = np.fft.fft(a_pad)
        subgrad = np.fft.ifft(E * A_f).real[:n]
        eta = 0.1
        g = np.exp(-eta * subgrad)
        g = np.maximum(g, 1e-10)
        g = g / g.sum()
    return g

def optimize_gradient(a0, budget_s, T_start=5.0, T_end=500.0, max_iters=20000,
                      momentum_init=0.85, spectral_kick_interval=200):
    """Projected gradient with Nesterov momentum and temperature annealing."""
    n = len(a0)
    s = get_fft_size(n)
    a = np.maximum(a0.copy(), 0.0)
    sm = a.sum()
    if sm < 1e-10:
        return a, np.inf
    a = a / sm

    best_a = a.copy()
    best_score = 2.0 * n * fast_conv(a).max()
    deadline = time.monotonic() + budget_s

    lr = 0.01
    velocity = np.zeros(n)
    mom = momentum_init

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        frac = it / max(1, max_iters - 1)
        T = T_start + (T_end - T_start) * frac

        # Nesterov lookahead
        a_look = a - vel * lr * mom
        a_look = np.maximum(a_look, 0.0)
        sl = a_look.sum()
        if sl < 1e-10:
            a_look = a
            sl = a.sum()
        a_look = a_look / sl

        obj, grad = compute_smooth_obj_and_grad(a_look, T, s)

        # Momentum update
        velocity = mom * velocity + (1.0 - mom) * grad
        a_new = a - lr * velocity
        a_new = np.maximum(a_new, 0.0)
        s_new = a_new.sum()
        if s_new < 1e-10:
            lr *= 0.5
            continue
        a_new = a_new / s_new

        # Simple line search via smooth objective
        obj_new, _ = compute_smooth_obj_and_grad(a_new, T, s)
        if obj_new < obj:
            a = a_new
            lr = min(lr * 1.02, 0.1)
        else:
            # Try half step
            a_half = a - 0.5 * lr * velocity
            a_half = np.maximum(a_half, 0.0)
            sh = a_half.sum()
            if sh > 1e-10:
                a_half = a_half / sh
                obj_half, _ = compute_smooth_obj_and_grad(a_half, T, s)
                if obj_half < obj:
                    a = a_half
                    lr *= 0.75
                else:
                    lr *= 0.5
            else:
                lr *= 0.5

        # Track best
        if it % 20 == 0:
            sc = 2.0 * n * fast_conv(a).max()
            if sc < best_score:
                best_score = sc
                best_a = a.copy()

        # Spectral kick
        if it > 0 and it % spectral_kick_interval == spectral_kick_interval - 20:
            alpha = 0.5 + 0.3 * (1.0 - frac)
            a = spectral_flatten(a, alpha, 0.3)

    final_sc = 2.0 * n * fast_conv(a).max()
    if final_sc < best_score:
        best_score = final_sc
        best_a = a.copy()

    return best_a, best_score

def make_initializations(n, rng):
    """Generate diverse initializations of length n."""
    t = np.linspace(0, 1, n, endpoint=False)
    inits = [np.ones(n)]
    inits.append(rng.uniform(0.1, 1.0, n))
    # Wavelength-1 modulation
    inits.append(np.maximum(1.0 + 0.5 * np.cos(2 * np.pi * t), 0.01))
    # Wavelength-2
    inits.append(np.maximum(1.0 + 0.5 * np.cos(4 * np.pi * t), 0.01))
    # Chirp
    inits.append(np.maximum(1.0 + 0.4 * np.cos(2 * np.pi * t * t * 4), 0.01))
    # Ramp
    inits.append(np.linspace(0.1, 2.0, n))
    # Bell
    inits.append(np.exp(-4.0 * (t - 0.5) ** 2))
    # Two-peak
    inits.append(np.maximum(1.0 + 0.8 * np.cos(6 * np.pi * t), 0.01))
    # Fourier random
    phases = rng.uniform(0, 2 * np.pi, n)
    amps = rng.exponential(0.3, n)
    inits.append(np.maximum(1.0 + np.sum(amps[:min(n,50)] * np.cos(2*np.pi*np.arange(1, min(n,50)+1)[:,None] * t[None,:] + phases[:min(n,50)][:,None]), axis=0), 0.01))
    return inits
