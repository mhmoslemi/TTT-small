def _conv_full(f):
    """Full linear convolution via FFT."""
    n = len(f)
    size = 1
    while size < 2*n - 1:
        size <<= 1

"""
Optimizes non-negative step function heights to minimize 2n*max(conv(f,f))/(sum(f))^2
using smooth log-sum-exp approximation with L-BFGS-B in Dirichlet (softmax) parameterization.
Multi-scale annealing on beta, multiple initializations, and zero-insertion growth.
"""

import numpy as np
import time
from scipy.optimize import minimize

def _conv_full(f):
    """Full linear convolution via FFT."""
    n = len(f)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    F = np.fft.rfft(f, size)
    return np.fft.irfft(F * F, size)[:2 * n - 1]

def _score(f):
    """Compute the score 2n*max(conv(f,f))/(sum(f))^2."""
    n = len(f)
    h = _conv_full(f)
    S = float(np.sum(f))
    if S < 0.01:
        return float('inf')
    return float(2.0 * n * np.max(h) / (S * S))

def _softmax_forward(theta):
    """Compute f = softmax(theta) with numerical stability."""
    t = theta - np.max(theta)
    e = np.exp(t)
    s = e.sum()
    if s < 1e-300:
        return np.ones(len(theta)) / len(theta)
    return e / s

def _smooth_val_grad_theta(theta, beta):
    """
    Value and gradient of 2n*(1/beta)*log(sum_k exp(beta*h[k])) in theta-space.
    Uses softmax parameterization f = softmax(theta) so that sum(f)=1, f>0 automatically.
    """
    n = len(theta)
    f = _softmax_forward(theta)
    h = _conv_full(f)

    # Log-sum-exp with stability
    h_max = float(np.max(h))
    shifted = beta * (h - h_max)
    shifted = np.clip(shifted, -500.0, 500.0)
    exp_shifted = np.exp(shifted)
    log_sum_exp = h_max + np.log(float(exp_shifted.sum()))
    g_val = log_sum_exp / beta
    val = 2.0 * n * g_val

    # Softmax weights
    w = exp_shifted / float(exp_shifted.sum())

    # Compute cross[j] = sum_{m=0}^{n-1} f[m] * w[j+m] via FFT cross-correlation
    f_ext = np.zeros(2 * n - 1)
    f_ext[:n] = f
    N = 1
    while N < 4 * n:
        N <<= 1
    F_ext = np.fft.rfft(f_ext, N)
    W = np.fft.rfft(w, N)
    cross = np.fft.irfft(np.conj(F_ext) * W, N)
    cross = cross[:n]

    # Gradient in theta-space:
    # grad_theta[i] = 4n * f[i] * (cross[i] - cross . f)
    cross_dot_f = float(np.dot(cross, f))
    grad_theta = 4.0 * n * f * (cross - cross_dot_f)

    return val, grad_theta

def _optimize_smooth(f0, beta_schedule, maxiter_per_beta=300, deadline=None, label=""):
    """
    Run L-BFGS-B smooth optimization with increasing beta in theta-space.
    f0 is the initial sequence (will be converted to theta via log).
    """
    n = len(f0)
    f = np.maximum(f0, 1e-15)
    f = f / f.sum()
    theta0 = np.log(f)

    best_f = f.copy()
    best_score = _score(f)
    theta = theta0.copy()

    print(f"  [{label}] n={n}, init score={best_score:.6f}")

    for beta in beta_schedule:
        if deadline is not None and time.monotonic() > deadline - 30:
            break

        def obj(x):
            v, _ = _smooth_val_grad_theta(x, beta)
            return v

        def grad(x):
            _, g = _smooth_val_grad_theta(x, beta)
            return g

        t0 = time.monotonic()
        try:
            res = minimize(obj, theta, jac=grad, method='L-BFGS-B',
                          options={'maxiter': maxiter_per_beta, 'maxfun': maxiter_per_beta * 20,
                                   'gtol': 1e-10, 'ftol': 1e-12})
        except Exception as e:
            print(f"  [{label}] beta={beta}: exception {e}")
            break

        elapsed = time.monotonic() - t0
        theta_new = res.x
        f_new = _softmax_forward(theta_new)
        s = _score(f_new)

        if s < best_score:
            best_score = s
            best_f = f_new.copy()
            theta = theta_new.copy()
            print(f"  [{label}] beta={beta}: score={best_score:.6f} ({elapsed:.1f}s, {res.nfev} evals)")
        else:
            print(f"  [{label}] beta={beta}: no improvement ({s:.6f}) ({elapsed:.1f}s)")

    return best_f, best_score

def _cosine_init(n):
    j = np.arange(n, dtype=np.float64)
    f = np.cos(np.pi * (j + 0.5) / float(n)) ** 2
    f = np.maximum(f, 1e-10)
    return f / f.sum()

def _gaussian_init(n, sigma_frac=0.3):
    j = np.arange(n, dtype=np.float64) - (n - 1) / 2.0
    sigma = sigma_frac * n
    f = np.exp(-j * j / (2.0 * sigma * sigma))
    return f / f.sum()

def _tapered_init(n, power=2.0):
    j = np.arange(n, dtype=np.float64)
    x = np.sin(np.pi * (j + 0.5) / float(n))
    f = x ** power
    f = np.maximum(f, 1e-10)
    return f / f.sum()

def _asymmetric_init(n, skew=0.3):
    """Create an asymmetric initialization."""
    j = np.arange(n, dtype=np.float64)
    center = (1.0 + skew) * (n - 1) / 2.0
    f = np.exp(-0.5 * ((j - center) / (0.25 * n)) ** 2)
    return f / f.sum()

def _insert_zero(f, pos):
    """Insert a zero at position pos (0-indexed, 0 to len(f))."""
    n = len(f)
    if pos < 0 or pos > n:
        return f
    result = np.empty(n + 1)
    result[:pos] = f[:pos]
    result[pos] = 0.0
    result[pos + 1:] = f[pos:]
    return result

        return [float(max(0.0, min(1000.0, x))) for x in parent]

    return final_list
