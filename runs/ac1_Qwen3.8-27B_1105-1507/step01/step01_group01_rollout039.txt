def enforce_conj_sym(A):

       if N==2: A[0]=A[0].real; A[1]=A[1].real
       return A
    half=N//2
    idx=np.arange(1,half)
    mid=0.5*(A[idx] + np.conj(A[N-idx]))
    A[idx]=mid
    A[N-idx]=np.conj(mid)
    A[0]=A[0].real
    if N%2==0:

```
def eval_conv(a, b=None):
    x=np.asarray(a,dtype=np.float64).reshape(-1)
    if x.size==0:
        return float('inf'), None
    x=np.where(np.isfinite(x),x,0.0)
    x=np.clip(x,0.0,1000.0)
    s=float(np.sum(x))
    if not np.isfinite(s) or s <= 1e-12:
        return float('inf'), None
    n=x.size
    if b is None:
        bb=conv_square(x)
    else:
        bb=np.asarray(b,dtype=np.float64).reshape(-1)
        if bb.size != 2*n-1:
            bb=conv_square(x)
    bb=np.where(np.isfinite(bb),bb,0.0)
    bb=np.where(bb>0.0,bb,0.0)
    m=float(np.max(bb)) if bb.size else 0.0
    if not np.isfinite(m):
        return float('inf'), bb
    val=2.0*n*m/(s*s)
    if not np.isfinite(val) or val < 0.0:
        return float('inf'), bb
    return val, bb
```

```
def weights_from_b(b,q):
    b=np.asarray(b,dtype=np.float64).reshape(-1)
    b=np.where(np.isfinite(b)&(b>0.0), b, 0.0)
    if b.size==0: return np.array([],dtype=float)
    M=float(np.max(b))
    if not np.isfinite(M) or M<=0:
       return np.full(b.size,1.0/b.size)
    q=max(0.0,float(q))
    if q < 1e-12:
       w=(b>0).astype(float)
       s=w.sum()
       if s>0: return w/s
       return np.full(b.size,1.0/b.size)
    logM=np.log(M)
    w=np.zeros(b.size,dtype=float)
    pos=b>0
    if np.any(pos):
       lv = q*(np.log(b[pos]) - logM)
       lv = np.clip(lv, -50.0, 0.0)
       wpos=np.exp(lv)
       w[pos]=wpos
    # threshold
    maxw=float(np.max(w)) if w.size else 0.0
    if maxw > 0:
       w[w < 1e-12*maxw] = 0.0
    s=float(np.sum(w))
    if not np.isfinite(s) or s <= 1e-300:
       w=(b >= 0.5*M).astype(float)
       s=float(np.sum(w))
       if s<=0:
          w=np.ones(b.size,dtype=float)
          s=float(b.size)
    return w/s
```

```python
"""
Search for low-score non-negative step-function coefficients.

Objective:
    score = 2*n*max(convolve(a,a)) / sum(a)^2

The program uses:
  - normalized sequence representation (sum = 1)
  - fast FFT-based convolution when n is large
  - soft-max active-peak weights
  - multiplicative and additive projected gradient steps
  - spectral projection / flattening moves that move the current convolution
    toward a flatter target in the frequency domain
  - random smooth restarts, U-shapes, 1/sqrt(i) profiles, and parent resampling
  - sequential multi-length optimization followed by final polishing

The returned sequence is the best one found before the time budget expires.
"""

import time
import numpy as np


def normalize_to_sum1(a):
    try:
        x = np.asarray(a, dtype=np.float64).reshape(-1)
    except Exception:
        return None
    if x.size == 0:
        return None
    x = np.where(np.isfinite(x), x, 0.0)
    x = np.clip(x, 0.0, 1000.0)
    s = float(np.sum(x))
    if not np.isfinite(s) or s <= 1e-300:
        return None
    return x / s


def resample_to_length(arr, n):
    x = normalize_to_sum1(arr)
    if x is None:
        return None
    n = max(1, int(n))
    m = x.size
    if m == n:
        return x.copy()
    if m == 1:
        return normalize_to_sum1(np.full(n, float(x[0])))
    xold = np.linspace(0.0, 1.0, m)
    xnew = np.linspace(0.0, 1.0, n)
    y = np.interp(xnew, xold, x)
    return normalize_to_sum1(y)


def conv_square(a):
    x = np.asarray(a, dtype=np.float64).reshape(-1)
    n = x.size
    if n == 0:
        return np.array([], dtype=np.float64)
    L = 2 * n - 1
    if n < 768:
        c = np.convolve(x, x)
    else:
        N = 1 << max(0, (L - 1).bit_length())
        fa = np.fft.rfft(x, N)
        c = np.fft.irfft(fa * fa, N)[:L]
    return np.asarray(c, dtype=np.float64)


def eval_conv(a, b=None):
    x = np.asarray(a, dtype=np.float64).reshape(-1)
    if x.size == 0:
        return float("inf"), None
    x = np.where(np.isfinite(x), x, 0.0)
    x = np.clip(x, 0.0, 1000.0)
    s = float(np.sum(x))
    if not np.isfinite(s) or s <= 1e-12:
        return float("inf"), None
    n = x.size
    if b is None:
        bb = conv_square(x)
    else:
        bb = np.asarray(b, dtype=np.float64).reshape(-1)
        if bb.size != 2 * n - 1:
            bb = conv_square(x)
    bb = np.where(np.isfinite(bb), bb, 0.0)
    bb = np.where(bb > 0.0, bb, 0.0)
    m = float(np.max(bb)) if bb.size else 0.0
    if not np.isfinite(m):
        return float("inf"), bb
    val = 2.0 * float(n) * m / (s * s)
    if not np.isfinite(val) or val < 0.0:
        return float("inf"), bb
    return val, bb


def local_score(a):
    val, _ = eval_conv(a, None)
    return val


def partner(a, lam):
    x = np.asarray(a, dtype=np.float64).reshape(-1)
    l = np.asarray(lam, dtype=np.float64).reshape(-1)
    n = x.size
    if n == 0:
        return np.array([], dtype=np.float64)
    if l.size != 2 * n - 1:
        return np.zeros(n, dtype=np.float64)
    l = np.where(np.isfinite(l), l, 0.0)

    if n == 1:
        return np.array([x[0] * l[0]], dtype=np.float64)

    nz_mask = l > 1e-14
    nnz = int(np.count_nonzero(nz_mask))

    # Sparse path for strongly peak-focused weights.
    if nnz <= 80:
        h = np.zeros(n, dtype=np.float64)
        nz = np.nonzero(nz_mask)[0]
        for k in nz:
            wk = float(l[k])
            if wk == 0.0:
                continue
            if k < n:
                s = 0
                e = k
            else:
                s = k - (n - 1)
                e = n - 1
            if s > e:
                continue
            idx = np.arange(s, e + 1, dtype=np.int64)
            j = k - idx
            h[idx] += wk * x[j]
        return h

    L = n + l.size - 1
    if n < 768:
        c = np.convolve(x, l[::-1])
    else:
        N = 1 << max(0, (L - 1).bit_length())
        ca = np.fft.rfft(x, N)
        cl = np.fft.rfft(l[::-1], N)
        c = np.fft.irfft(ca * cl, N)[:L]

    out = c[2 * n - 2 : n - 2 : -1]
    return np.asarray(out, dtype=np.float64)


def weights_from_b(b, q):
    b = np.asarray(b, dtype=np.float64).reshape(-1)
    b = np.where(np.isfinite(b) & (b > 0.0), b, 0.0)
    if b.size == 0:
        return np.array([], dtype=np.float64)
    M = float(np.max(b))
    if not np.isfinite(M) or M <= 0.0:
        return np.full(b.size, 1.0 / float(b.size), dtype=np.float64)

    q = max(0.0, float(q))
    if q < 1e-12:
        w = (b > 0.0).astype(np.float64)
        s = float(np.sum(w))
        if s > 0.0:
            return w / s
        return np.full(b.size, 1.0 / float(b.size), dtype=np.float64)

    logM = np.log(M)
    w = np.zeros(b.size, dtype=np.float64)
    pos = b > 0.0
    if np.any(pos):
        lv = q * (np.log(b[pos]) - logM)
        lv = np.clip(lv, -50.0, 0.0)
        w[pos] = np.exp(lv)

    maxw = float(np.max(w)) if w.size else 0.0
    if maxw > 0.0:
        w[w < 1e-12 * maxw] = 0.0

    s = float(np.sum(w))
    if not np.isfinite(s) or s <= 1e-300:
        w = (b >= 0.5 * M).astype(np.float64)
        s = float(np.sum(w))
        if s <= 0.0:
            w = np.ones(b.size, dtype=np.float64)
            s = float(b.size)
    return w / s


def apply_natural(a, h, eta, mode):
    x = normalize_to_sum1(a)
    if x is None:
        return None
    h = np.asarray(h, dtype=np.float64).reshape(-1)
    if h.size != x.size:
        return None
    h = np.where(np.isfinite(h), h, 0.0)

    maxa = float(np.max(x)) if x.size else 0.0
    if mode == "natural":
        # A small floor avoids the standard log-gradient path assigning zero
        # derivative to zero coefficients.
        aeff = x + max(1e-12, maxa * 1e-3)
        g = aeff * h
    else:
        g = h

    mu = float(np.dot(x, g))
    v = g - mu
    sd2 = float(np.dot(x, v * v))
    if not np.isfinite(sd2) or sd2 <= 1e-24:
        return None
    sd = np.sqrt(sd2)
    z = -float(eta) * v / sd
    z = np.clip(z, -40.0, 40.0)
    fac = np.exp(z)
    c = x * fac
    c = np.where(np.isfinite(c), c, 0.0)
    c = np.clip(c, 0.0, 1000.0)
    return normalize_to_sum1(c)


def apply_euclid(a, h, eta):
    x = normalize_to_sum1(a)
    if x is None:
        return None
    h = np.asarray(h, dtype=np.float64).reshape(-1)
    if h.size != x.size:
        return None
    h = np.where(np.isfinite(h), h, 0.0)

    mu = float(np.dot(x, h))
    v = h - mu
    sd2 = float(np.dot(x, v * v))
    if not np.isfinite(sd2) or sd2 <= 1e-24:
        return None
    c = x - float(eta) * v / np.sqrt(sd2)
    c = np.clip(c, 0.0, 1000.0)
    return normalize_to_sum1(c)


def perturb_sequence(a, rng, strength=0.25):
    x = normalize_to_sum1(a)
    if x is None:
        return None
    n = x.size
    if n == 0:
        return x

    c = x * np.exp(rng.normal(0.0, max(0.01, strength * 0.5), n))

    wmin = max(1, n // 40)
    wmax = max(wmin + 1, min(n, n // 6))
    if wmax > wmin:
        w = int(rng.integers(wmin, wmax + 1))
    else:
        w = n
    start = int(rng.integers(0, max(1, n - w + 1)))
    c[start : start + w] *= np.exp(rng.normal(0.0, strength))

    if rng.random() < 0.5:
        modes = int(rng.integers(1, max(2, n // 64)))
        t = np.linspace(0.0, float(modes) * 2.0 * np.pi, n)
        amp = float(rng.uniform(0.05, max(0.1, strength * 0.5)))
        phase = float(rng.uniform(0.0, 2.0 * np.pi))
        c *= np.exp(amp * np.sin(t + phase))

    c = np.clip(c, 0.0, 1000.0)
    return normalize_to_sum1(c)


def enforce_conj_sym(A):
    A = np.asarray(A, dtype=np.complex128).copy()
    N = A.size
    if N <= 1:
        if N == 1:
            A[0] = A[0].real
        return A
    end = (N - 1) // 2
    if end >= 1:
        idx = np.arange(1, end + 1)
        mid = 0.5 * (A[idx] + np.conj(A[N - idx]))
        A[idx] = mid
        A[N - idx] = np.conj(mid)
    A[0] = A[0].real
    if N % 2 == 0:
        A[N // 2] = A[N // 2].real
    return A


def target_flat(n):
    m = 2 * int(n) - 1
    return np.full(m, 1.0 / float(m), dtype=np.float64)


def target_cap(b, frac=1.0):
    b = np.asarray(b, dtype=np.float64).reshape(-1)
    b = np.where(np.isfinite(b) & (b > 0.0), b, 0.0)
    if b.size == 0:
        return np.array([], dtype=np.float64)
    M = float(np.max(b))
    if not np.isfinite(M) or M <= 0.0:
        return np.full(b.size, 1.0 / float(b.size), dtype=np.float64)
    t = np.minimum(b, float(frac) * M)
    s = float(np.sum(t))
    if s <= 0.0 or not np.isfinite(s):
        return np.full(b.size, 1.0 / float(b.size), dtype=np.float64)
    return t / s


def target_smooth(n, edge_level=0.5, power=2.0):
    m = 2 * int(n) - 1
    center = int(n) - 1
    idx = np.arange(m, dtype=np.float64)
    d = np.abs(idx - float(center))
    denom = max(1.0, float(int(n) - 1))
    d = d / denom
    el = min(1.0, max(0.0, float(edge_level)))
    w = 1.0 - (1.0 - el) * np.power(d, float(power))
    w = np.where(w > 1e-4, w, 1e-4)
    s = float(np.sum(w))
    if s <= 0.0 or not np.isfinite(s):
        return target_flat(n)
    return w / s


def spectral_flat_start(n, seed=0):
    n = max(1, int(n))
    m = 2 * n - 1
    target = np.full(m, 1.0 / float(m), dtype=np.float64)
    N = 1 << max(0, (m - 1).bit_length())
    T = np.fft.fft(target, N)
    Tsafe = T.copy()
    small = np.abs(Tsafe) < 1e-300
    Tsafe[small] = 1e-300 + 0.0j
    A = np.exp(0.5 * np.log(Tsafe))
    A = enforce_conj_sym(A)
    y = np.fft.ifft(A).real[:n]

    pos = np.clip(y, 0.0, None)
    c = normalize_to_sum1(pos)
    if c is None:
        c = normalize_to_sum1(np.abs(y))
    u = np.ones(n, dtype=np.float64) / float(n)
    if c is None:
        return u
    return normalize_to_sum1(0.6 * c + 0.4 * u)


def spectral_flattening(a, target, alpha=0.5):
    x = normalize_to_sum1(a)
    if x is None:
        return None
    n = x.size
    m = 2 * n - 1

    t = np.asarray(target, dtype=np.float64).reshape(-1)
    if t.size != m:
        t = target_flat(n)
    t = np.where(np.isfinite(t) & (t > 0.0), t, 0.0)
    s = float(np.sum(t))
    if s <= 0.0 or not np.isfinite(s):
        t = target_flat(n)
        s = float(np.sum(t))
    t = t / s

    N = 1 << max(0, (m - 1).bit_length())
    A = np.fft.fft(x, N)
    B = A * A
    T = np.fft.fft(t, N)

    maxB = float(np.max(np.abs(B))) if B.size else 1.0
    eps = max(1e-300, maxB * 1e-10)

    A_new = A.copy()
    mask = np.abs(B) > eps
    if np.any(mask):
        ratio = T[mask] / B[mask]
        small = np.abs(ratio) < 1e-300
        if np.any(small):
            ratio = ratio.copy()
            ratio[small] = 1e-300 + 0.0j
        logratio = np.log(ratio)
        A_new[mask] = A[mask] * np.exp(0.5 * float(alpha) * logratio)

    if np.any(~mask):
        Tz = T[~mask]
        small = np.abs(Tz) < eps
        vals = np.zeros(Tz.shape, dtype=np.complex128)
        if np.any(~small):
            vals[~small] = np.exp(0.5 * float(alpha) * np.log(Tz[~small]))
        A_new[~mask] = vals

    A_new = enforce_conj_sym(A_new)
    c = np.fft.ifft(A_new).real[:n]
    c = np.clip(c, 0.0, None)
    return normalize_to_sum1(c)


def spectral_attempt(a, b, seed=0, max_evals=6):
    x = normalize_to_sum1(a)
    if x is None:
        return None, float("inf"), None
    sc0, bb0 = eval_conv(x, b)
    if not np.isfinite(sc0) or bb0 is None:
        return None, float("inf"), None

    best = x.copy()
    bestsc = float(sc0)
    bestb = bb0.copy()
    n = x.size

    targets = []
    targets.append(target_cap(bb0, 1.0))
    targets.append(target_flat(n))
    targets.append(target_smooth(n, 0.5, 2.0))
    targets.append(target_cap(bb0, 0.95))

    alphas = (0.12, 0.25, 0.45)
    evals = 0

    for target in targets:
        for alpha in alphas:
            if evals >= max_evals:
                break
            cand = spectral_flattening(x, target, alpha)
            if cand is None:
                continue
            for tau in (1.0, 0.5):
                if evals >= max_evals:
                    break
                mm = tau * cand + (1.0 - tau) * x
                mm = normalize_to_sum1(mm)
                if mm is None:
                    continue
                sc, bb = eval_conv(mm, None)
                evals += 1
                if np.isfinite(sc) and bb is not None and sc < bestsc - 1e-13:
                    best = mm.copy()
                    bestsc = float(sc)
                    bestb = bb.copy()
        if evals >= max_evals:
            break

    return best, bestsc, bestb


def generate_starts(n, seed, parent=None, max_starts=16):
    n = max(2, int(n))
    rng = np.random.default_rng(seed)
    starts = []

    if parent is not None:
        p = resample_to_length(parent, n)
        if p is not None:
            starts.append(p)
            starts.append(resample_to_length(parent[::-1], n))
            sym = 0.5 * (p + p[::-1])
            starts.append(normalize_to_sum1(sym))

    starts.append(np.ones(n, dtype=np.float64) / float(n))

    idx = np.arange(n, dtype=np.float64)
    denom = float(n - 1) if n > 1 else 1.0
    t = idx / denom

    # 1/sqrt profiles and related tapered variants.
    base_inv = 1.0 / np.sqrt(idx + 1.0)
    base_sqrt = np.sqrt(idx + 1.0)
    starts.append(normalize_to_sum1(base_inv))
    starts.append(normalize_to_sum1(base_sqrt))

    for beta in (-0.8, -0.4, -0.1, 0.1, 0.4, 0.8):
        v = base_inv * np.exp(float(beta) * t)
        v = np.clip(v, 0.0, None)
        starts.append(normalize_to_sum1(v))

    for beta in (0.2, 0.5, 0.9, 1.4):
        v = base_sqrt * np.exp(-float(beta) * t)
        v = np.clip(v, 0.0, None)
        starts.append(normalize_to_sum1(v))

    for alpha in (0.25, 0.4, 0.5):
        v = np.power(idx + 1.0, -float(alpha))
        starts.append(normalize_to_sum1(v))
        v = np.power(idx + 1.0, float(alpha))
        starts.append(normalize_to_sum1(v))

    for alpha in (0.2, 0.5, 1.0):
        v = 1.0 / np.power(idx + 10.0, float(alpha))
        starts.append(normalize_to_sum1(v))

    # Linear trends.
    for c in (-0.8, -0.5, -0.2, 0.0, 0.2, 0.5, 0.8):
        v = 1.0 + float(c) * t
        v = np.clip(v, 1e-8, None)
        starts.append(normalize_to_sum1(v))

    # Edge/center shapes.
    for alpha in (0.5, 1.0, 2.0, 4.0):
        v = np.exp(float(alpha) * (t - 0.5) * (t - 0.5))
        starts.append(normalize_to_sum1(v))
        v = 1.0 / (1.0 + float(alpha) * (t - 0.5) * (t - 0.5))
        starts.append(normalize_to_sum1(v))

    # Low-frequency cosine patterns clipped to non-negativity.
    for mode in (1, 2, 3, 5):
        for amp in (0.3, 0.6, 0.9):
            v = 1.0 + float(amp) * np.cos(2.0 * np.pi * float(mode) * t)
            if np.min(v) < 0.0:
                v = np.clip(v, 1e-8, None)
            starts.append(normalize_to_sum1(v))

    # Smooth random starts.
    for _ in range(5):
        x = rng.normal(size=n)
        w = int(rng.integers(3, max(4, min(17, n + 1))))
        k = np.ones(w, dtype=np.float64) / float(w)
        x = np.convolve(x, k, mode="same")
        x = np.clip(x, 0.0, None) + 1e-6
        starts.append(normalize_to_sum1(x))

    # Spectral flat-root start.
    starts.append(spectral_flat_start(n, seed))

    out = []
    for arr in starts:
        arr = normalize_to_sum1(arr)
        if arr is not None and arr.size == n:
            out.append(arr)
        if len(out) >= max_starts:
            break
    if not out:
        out.append(np.ones(n, dtype=np.float64) / float(n))
    return out


def local_optimize(a0, budget, seed=0, q0=1.0, qmax=64.0):
    base = normalize_to_sum1(a0)
    if base is None:
        return None, float("inf")

    best = base.copy()
    sc, bb = eval_conv(best, None)
    if not np.isfinite(sc) or bb is None:
        return best, float("inf")
    best_b = np.where(bb > 0.0, bb, 0.0)
    bestsc = float(sc)

    cur = best.copy()
    cur_b = best_b.copy()
    cur_sc = float(bestsc)

    t_start = time.monotonic()
    budget = float(budget)
    if not np.isfinite(budget) or budget <= 0.0:
        return best, bestsc
    deadline = t_start + budget
    if deadline <= t_start:
        return best, bestsc

    rng = np.random.default_rng(seed)
    eta = 0.5
    q = max(1.0, float(q0))
    no_improve = 0
    last_spectral = t_start
    it = 0

    while True:
        now = time.monotonic()
        if now >= deadline:
            break
        rem = deadline - now
        if rem < 0.2:
            break

        # If the working point wandered too far below the best, reset it.
        if bestsc > 0.0 and (cur_sc > bestsc + 0.02 or cur_sc > 1.02 * bestsc):
            cur = best.copy()
            cur_b = best_b.copy()
            cur_sc = float(bestsc)
            no_improve = 0
            eta = 0.5
            q = max(1.0, float(q0))

        if no_improve > 8:
            q = min(float(qmax), q * 1.3)
        elif no_improve < 3:
            q = max(1.0, q * 0.85)

        r = int(rng.integers(100))
        if r < 55:
            mode = "partner"
        elif r < 85:
            mode = "natural"
        else:
            mode = "euclid"

        w = weights_from_b(cur_b, q)
        if w is None or w.size != cur_b.size:
            w = np.full(cur_b.size, 1.0 / float(max(1, cur_b.size)), dtype=np.float64)
        h = partner(cur, w)

        etas = [eta, 0.5 * eta, 2.0 * eta]
        if rng.random() < 0.30:
            etas.append(0.25 * eta)
        if rng.random() < 0.20:
            etas.append(4.0 * eta)

        best_trial = None
        best_trial_sc = float(cur_sc)
        best_trial_b = None
        best_eta = eta

        for et in etas:
            if mode == "partner":
                cand = apply_natural(cur, h, et, "partner")
            elif mode == "natural":
                cand = apply_natural(cur, h, et, "natural")
            else:
                cand = apply_euclid(cur, h, et)
            if cand is None:
                continue
            sc_trial, bb_trial = eval_conv(cand, None)
            if np.isfinite(sc_trial) and bb_trial is not None and sc_trial < best_trial_sc - 1e-13:
                best_trial = cand
                best_trial_sc = float(sc_trial)
                best_trial_b = bb_trial
                best_eta = et

        if best_trial is not None:
            cur = best_trial
            cur_sc = float(best_trial_sc)
            cur_b = best_trial_b
            if cur_sc < bestsc - 1e-13:
                best = cur.copy()
                best_b = cur_b.copy()
                bestsc = cur_sc
                no_improve = 0
                eta = min(8.0, eta * 1.08)
                q = min(float(qmax), q * 1.02)
            else:
                no_improve += 1
                eta = best_eta
        else:
            no_improve += 1
            if eta > 0.05:
                eta *= 0.72
            else:
                eta = 0.5

        if no_improve > 25:
            if rem > 2.0 and (time.monotonic() - last_spectral) > 5.0:
                c, sc_sp, bb_sp = spectral_attempt(cur, cur_b, seed + it, 5)
                last_spectral = time.monotonic()
                if c is not None and np.isfinite(sc_sp) and bb_sp is not None:
                    if sc_sp < bestsc - 1e-13:
                        best = c.copy()
                        best_b = bb_sp.copy()
                        bestsc = float(sc_sp)
                    if sc_sp < cur_sc - 1e-13:
                        cur = c.copy()
                        cur_b = bb_sp.copy()
                        cur_sc = float(sc_sp)

            p = perturb_sequence(cur, rng, 0.25)
            if p is not None:
                sc_p, bb_p = eval_conv(p, None)
                if np.isfinite(sc_p) and bb_p is not None and sc_p < cur_sc - 1e-13:
                    cur = p.copy()
                    cur_sc = float(sc_p)
                    cur_b = bb_p.copy()
                else:
                    blend = 0.5 * cur + 0.5 * best
                    blend = normalize_to_sum1(blend)
                    if blend is not None:
                        sc_bl, bb_bl = eval_conv(blend, None)
                        if np.isfinite(sc_bl) and bb_bl is not None and sc_bl < cur_sc - 1e-13:
                            cur = blend.copy()
                            cur_sc = float(sc_bl)
                            cur_b = bb_bl.copy()

            no_improve = 0
            eta = 0.5
            q = max(1.0, float(q0))

        it += 1

    return best, bestsc


def optimize_length(n, budget, seed=0, parent=None):
    n = max(2, int(n))
    budget = float(budget)
    if not np.isfinite(budget) or budget <= 0.0:
        return None, float("inf")

    if n <= 256:
        max_starts = 20
    elif n <= 1024:
        max_starts = 14
    elif n <= 4096:
        max_starts = 10
    else:
        max_starts = 7

    starts = generate_starts(n, seed, parent, max_starts)

    scored = []
    counter = 0
    for s in starts:
        sc, _ = eval_conv(s, None)
        if np.isfinite(sc):
            scored.append((float(sc), counter, s))
            counter += 1
    if not scored:
        return None, float("inf")

    score_list = []
    for item in scored:
        score_list.append(item[0])
    order = np.argsort(score_list).tolist()

    scored_sorted = []
    for idx in order:
        scored_sorted.append(scored[idx])
    scored = scored_sorted

    t_start = time.monotonic()
    deadline = t_start + budget
    best_local = None
    best_local_sc = float("inf")

    max_rank = len(scored)
    if budget > 200.0:
        max_rank = min(max_rank, 6)
    elif budget > 50.0:
        max_rank = min(max_rank, 4)
    else:
        max_rank = min(max_rank, 2)

    for rank in range(max_rank):
        now = time.monotonic()
        if now >= deadline:
            break
        rem = deadline - now
        if rem < 3.0:
            break

        sc0, _, s0 = scored[rank]

        if rank == 0:
            tb = 0.55 * budget
        elif rank == 1:
            tb = 0.25 * budget
        elif rank == 2:
            tb = 0.12 * budget
        elif rank == 3:
            tb = 0.05 * budget
        else:
            tb = 0.03 * budget

        tb = min(tb, rem - 1.0)
        tb = max(1.0, tb)

        try:
            c, sc = local_optimize(s0, tb, seed + rank * 7919, q0=1.0, qmax=64.0)
        except Exception as exc:
            c = None
            sc = float("inf")

        if c is not None and np.isfinite(sc) and sc < best_local_sc:
            best_local = c
            best_local_sc = float(sc)

        # If the very first line is already excellent, give it some of the rest.
        if rank == 0 and best_local is not None and best_local_sc < 1.55:
            rem2 = deadline - time.monotonic()
            if rem2 > 30.0:
                try:
                    c2, sc2 = local_optimize(
                        best_local,
                        min(0.45 * rem2, 0.35 * budget),
                        seed + 424242,
                        q0=2.0,
                        qmax=128.0,
                    )
                except Exception:
                    c2 = None
                    sc2 = float("inf")
                if c2 is not None and np.isfinite(sc2) and sc2 < best_local_sc:
                    best_local = c2
                    best_local_sc = float(sc2)

    return best_local, best_local_sc


def load_parent():
    try:
        obj = globals().get("height_sequence_1", None)
    except Exception:
        obj = None
    if obj is None:
        return None
    try:
        arr = np.asarray(obj, dtype=np.float64)
    except Exception:
        return None
    if arr.size == 0:
        return None
    arr = np.where(np.isfinite(arr), arr, 0.0)
    arr = np.clip(arr, 0.0, 1000.0)
    return arr

            bestsc = local_score(final)

    # If still above the target and there is time, use the leftover budget.
    if (not np.isfinite(bestsc) or bestsc > 1.5027) and time.monotonic() < absolute_end - 35.0:
        rem = absolute_end - time.monotonic() - 20.0
        if rem > 30.0:
            try:
                c, sc = local_optimize(best, rem, seedcur + 98765, q0=4.0, qmax=128.0)
            except Exception:
                c = None
                sc = float("inf")
            if c is not None and np.isfinite(sc) and sc < bestsc:
                sc2, bb2 = eval_conv(c, None)
                if np.isfinite(sc2) and bb2 is not None:
                    best = c.copy()
                    best_b = np.where(bb2 > 0.0, bb2, 0.0)
                    best = best
                final = normalize_to_sum1(c)
                if final is not None:
                    try:
                        off = float(evaluate_sequence([float(x) for x in final]))
                    except Exception:
                        off = float("inf")
                    if np.isfinite(off):
                        bestsc = float(off)
                        print(f"extra polish official score={bestsc:.6f}", flush=True)

    final = normalize_to_sum1(best)
    if final is None:
        final = np.ones(64, dtype=np.float64) / 64.0

    print(f"FINAL score_local_or_official={bestsc:.6f} length={len(final)}", flush=True)
    return [float(x) for x in np.clip(np.asarray(final, dtype=np.float64), 0.0, 1000.0)]
```
