def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 1e-10:
        return np.inf
    conv = np.convolve(a, a)
    max_b = float(np.max(conv))
    return 2.0 * n * max_b / (s * s)


def compute_conv(a):
    """Compute convolution of a with itself."""
    return np.convolve(a, a)


def subgradient(a, k_star):
    """Compute subgradient of max_b at a, given argmax index k_star."""
    n = len(a)
    g = np.zeros(n)
    # g_m = 2 * a[k_star - m] if 0 <= k_star - m < n
    m_min = max(0, k_star - n + 1)
    m_max = min(n - 1, k_star)
    for m in range(m_min, m_max + 1):
        g[m] = 2.0 * a[k_star - m]
    return g


def project_simplex(v):
    """Project v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v
    # Use sorting-based algorithm
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.where(u * np.arange(1, n + 1) > (cssv - 1))[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho = rho[-1]
    theta = (cssv[rho] - 1) / (rho + 1)
    return np.maximum(v - theta, 0)


def subgradient_descent(a_init, max_iter=5000, verbose=False):
    """Projected subgradient descent for minimizing 2n*max_b subject to sum(a)=1, a>=0."""
    n = len(a_init)
    a = a_init.copy()
    a = a / np.sum(a)
    best_a = a.copy()
    best_obj = compute_obj(best_a)
    
    for it in range(max_iter):
        conv = compute_conv(a)
        k_star = int(np.argmax(conv))
        max_b = conv[k_star]
        
        g = subgradient(a, k_star)
        
        # Step size: eta = c / sqrt(it+1)
        eta = 1.0 / np.sqrt(it + 1)
        # Scale by n to account for the 2n factor
        # Actually, the gradient of 2n*max_b is 2n * g (where g is the subgradient of max_b)
        # So: grad = 2n * g
        # But since we're on the simplex and the function is homogeneous, let's just use g scaled appropriately
        
        # The actual subgradient of f(a) = 2n*max_b is 2n * (2*a[k*-m]) for valid m
        # So the step is: a - eta * 2n * g_sub where g_sub[m] = 2*a[k*-m]
        # Let's just use: a_new = a - eta * g_full where g_full = 2n * subgradient
        g_full = 2.0 * n * subgradient(a, k_star)
        
        a_new = a - eta * g_full / max(1.0, np.linalg.norm(g_full))  # normalized step
        a_new = project_simplex(a_new)
        
        obj_new = compute_obj(a_new)
        if obj_new < best_obj:
            best_obj = obj_new
            best_a = a_new.copy()
        
        # Adaptive step: if no improvement, reduce step
        a = a_new
        
        if verbose and (it % 100 == 0 or it < 5):
            print(f"  SD it={it}: obj={best_obj:.6f} k*={k_star} max_b={max_b:.6f}")
    
    return best_a, best_obj

def build_A_fast(a):
    """Build constraint matrix A for LP: A[k,j] = a[k-j] if valid."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for k in range(2 * n - 1):
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        A[k, lo:hi+1] = a[k-hi:k-lo+1][::-1]
    return A


def solve_lp(a, upper_bound=1000.0):
    n = len(a)
    conv = np.convolve(a, a)
    M = float(np.max(conv))
    A = build_A_fast(a)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, upper_bound)] * n
    res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
    if res.status == 0:
        return res.x, M
    return None, M

    """Generate a sedimentary profile: edge + core + edge."""
    if rng is None:

    core_len = n - 2 * m
    if core_len < 1:

    if core_profile == 'flat':
        core = np.ones(core_len)

        core = 1.0 / np.sqrt(idx + 1.0)
    elif core_profile == 'cosine':
        core = np.cos(np.pi * idx / (core_len + 1))

    taper = np.arange(1, m + 1) / (m + 1.0)

    a = np.concatenate([left_edge, core, right_edge])
    s = np.sum(a)
    if s < 1e-10:

    return a / s
