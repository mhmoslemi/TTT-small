    return None

"""
Minimizes J(a) = 2*n*max(convolve(a,a))/(sum(a))^2 using:
1. Coordinate descent with exact 1D line search per coordinate
2. LP-based mirror descent for structured updates
3. L-BFGS-B gradient-based refinement
4. Multiple sequence lengths and diverse initializations
Key innovation: fast coordinate descent exploiting the linear structure
of b_k as a function of a_single coordinate.
"""

import numpy as np
from scipy.optimize import linprog, minimize, minimize_scalar
import time

def fast_conv(a):
    """Full self-convolution via FFT."""
    n = len(a)
    s = 1
    while s < 2 * n - 1:
        s <<= 1
    fa = np.fft.rfft(a, s)
    return np.fft.irfft(fa * fa, s)[:2 * n - 1]

def evaluate_local(a):
    """Compute J(a) locally (same as evaluate_sequence but on numpy array)."""
    a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
    n = len(a)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = np.convolve(a, a, mode='full')
    return float(2.0 * n * b.max() / (s * s))

def coordinate_descent(a, max_sweeps=50, tol=1e-12, verbose=False):
    """Coordinate descent with exact 1D optimization per coordinate."""
    n = len(a)
    a = a.astype(np.float64).copy()
    target_sum = a.sum()
    
    for sweep in range(max_sweeps):
        max_change = 0.0
        for j in range(n):
            old = a[j]
            a[j] = 0.0
            b_base = np.convolve(a, a, mode='full')
            
            # Define f(x) = max_k b_k when a[j] = x
            # b_k(x) = b_base_k + 2*x*a[k-j] (for valid k, k != 2j)
            #         b_base_k + x^2 (for k = 2j)
            k_start = max(0, j)
            k_end = min(2 * n - 2, j + n - 1)
            
            # Precompute the linear part coefficients
            if k_start <= k_end:
                lin_coeffs = 2.0 * a[k_start - j:k_end - j + 1]
                base_slice = b_base[k_start:k_end + 1]
            else:
                lin_coeffs = np.array([])
                base_slice = np.array([])
            
            has_quad = (2 * j < 2 * n - 1)
            quad_base = b_base[2 * j] if has_quad else 0.0
            
            global_max_base = b_base.max()
            
            def f_of_x(x):
                # Start with global max of base
                m = global_max_base
                if len(lin_coeffs) > 0:
                    # max of (base_slice + x * lin_coeffs)
                    vals = base_slice + x * lin_coeffs
                    if has_quad and 2*j >= k_start and 2*j <= k_end:
                        vals[2*j - k_start] = quad_base + x * x
                    vm = vals.max()
                    if vm > m:
                        m = vm
                elif has_quad:
                    qv = quad_base + x * x
                    if qv > m:
                        m = qv
                return m
            
            hi = 2.0 * old + target_sum / n
            if hi < 1e-15:
                hi = target_sum / n
            
            res = minimize_scalar(f_of_x, bounds=(0.0, hi), method='bounded',
                                  options={'xatol': 1e-12, 'maxiter': 200})
            new_val = max(0.0, res.x)
            a[j] = new_val
            max_change = max(max_change, abs(new_val - old))
        
        s = a.sum()
        if s > 1e-10:
            a *= target_sum / s
        
        if verbose and sweep % 10 == 0:
            score = evaluate_local(a)
            print(f"    CD sweep {sweep}: J={score:.8f}, max_change={max_change:.2e}")
        
        if max_change < tol:
            if verbose:
                print(f"    CD converged at sweep {sweep}")
            break
    
    return a

def lp_best_response(a, M=None):
    """Solve LP: max sum(g) s.t. (a*g)_k <= M, g >= 0."""
    n = len(a)
    if M is None:
        b = fast_conv(a)
        M = float(b.max())
    
    c = -np.ones(n)
    # Build constraint matrix A where (A @ g)[k] = (a * g)[k]
    A = np.zeros((2 * n - 1, n))
    for k in range(2 * n - 1):
        jl = max(0, k - n + 1)
        jh = min(n - 1, k)
        if jl <= jh:
            A[k, jl:jh + 1] = a[k - jh:k - jl + 1][::-1]
    
    r = linprog(c, A_ub=A, b_ub=np.full(2 * n - 1, M),
               bounds=[(0, None)] * n, method='highs')
    if r.success and r.x is not None and r.x.sum() > 1e-10:
        return r.x
    return None

def subgradient_method(a, n_steps=200, verbose=False):
    """Subgradient method with averaging for minimizing max_k(b_k)."""
    n = len(a)
    a = a.astype(np.float64).copy()
    target_sum = a.sum()
    a_avg = a.copy()
    
    for t in range(1, n_steps + 1):
        b = fast_conv(a)
        k_star = int(np.argmax(b))
        
        # Subgradient of b_{k*} w.r.t. a
        grad = np.zeros(n)
        for j in range(n):
            idx = k_star - j
            if 0 <= idx < n:
                grad[j] = 2.0 * a[idx]
        
        # Step size: 1/sqrt(t)
        lr = target_sum / (np.linalg.norm(grad) + 1e-10) / np.sqrt(t)
        
        a_new = a - lr * grad
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s > 1e-10:
            a_new /= s * (1.0 / target_sum)  # Normalize to target_sum
            a_new = a_new * target_sum / a_new.sum()
        
        a = a_new
        a_avg += a
    
    a_avg /= n_steps
    a_avg = np.clip(a_avg, 0.0, None)
    s = a_avg.sum()
    if s > 1e-10:
        a_avg *= target_sum / s
    return a_avg

def lbb_refine(a, maxiter=2000):
    """L-BFGS-B refinement."""
    n = len(a)
    target_sum = a.sum()
    
    def obj_and_grad(x):
        ax = np.maximum(x, 0.0)
        s = ax.sum()
        if s < 1e-8:
            return 1e10, np.zeros(n)
        ax_n = ax / s
        b = fast_conv(ax_n)
        k_star = int(np.argmax(b))
        val = float(b.max())
        
        # Subgradient (not exact gradient of max, but useful for L-BFGS)
        grad = np.zeros(n)
        for j in range(n):
            idx = k_star - j
            if 0 <= idx < n:
                grad[j] = 2.0 * ax_n[idx]
        # Chain rule: x -> ax = max(x,0) -> ax_n = ax/s -> b
        # d(val)/dx_j = (2*ax_n[k_star-j])/s - val/s (for the max part)
        # This is approximate; L-BFGS will handle it
        grad = grad / s
        return val, grad
    
    res = minimize(obj_and_grad, a, method='L-BFGS-B',
                   jac=True,
                   bounds=[(0, None)] * n,
                   options={'maxiter': maxiter, 'max_fun': 10000,
                            'gtol': 1e-10, 'disp': False})
    if res.fun < 1e10:
        return np.maximum(res.x, 0.0)
    return a

"""
Minimizes J(a) = 2*n*max(convolve(a,a))/(sum(a))^2 using:
1. LP-based mirror descent (best-response) for structured updates
2. Fast coordinate descent with exact 1D line search per coordinate
3. Subgradient method with iterate averaging
4. L-BFGS-B gradient-based refinement
5. Multiple sequence lengths (256-4096) with diverse initializations
Key innovation: coordinate descent exploits the linear structure of b_k(a_j)
to achieve O(n) per coordinate evaluation, enabling many polish sweeps.
"""

import numpy as np
from scipy.optimize import linprog, minimize, minimize_scalar
import time


def fast_conv(a):
    """Full self-convolution via FFT, O(n log n)."""
    n = len(a)
    s = 1
    while s < 2 * n - 1:
        s <<= 1
    fa = np.fft.rfft(a, s)
    return np.fft.irfft(fa * fa, s)[:2 * n - 1]


def objective_local(a):
    """Compute J(a) = 2*n*max(conv)/(sum)^2 for a numpy array."""
    a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
    n = len(a)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = np.convolve(a, a, mode='full')
    return float(2.0 * n * b.max() / (s * s))


def coordinate_descent(a, max_sweeps=50, tol=1e-12, verbose=False):
    """
    Coordinate descent: for each coordinate j, find optimal a_j via 1D search.
    Exploits that b_k is linear (or quadratic at k=2j) in a_j.
    """
    n = len(a)
    a = a.astype(np.float64).copy()
    target_sum = a.sum()
    if target_sum < 1e-15:
        return a

    for sweep in range(max_sweeps):
        max_change = 0.0
        for j in range(n):
            old = a[j]
            a[j] = 0.0
            b_base = np.convolve(a, a, mode='full')

            k_start = max(0, j)
            k_end = min(2 * n - 2, j + n - 1)
            has_linear = (k_start <= k_end)
            has_quad = (2 * j < 2 * n - 1)

            if has_linear:
                lin_coeffs = 2.0 * a[k_start - j:k_end - j + 1]
                base_slice = b_base[k_start:k_end + 1].copy()
            else:
                lin_coeffs = np.array([])
                base_slice = np.array([])

            if has_quad:
                quad_base = b_base[2 * j]
                # At k=2j, the linear coeff is 2*a[j]=0, so we replace with x^2
                if has_linear and 2 * j >= k_start and 2 * j <= k_end:
                    quad_offset = 2 * j - k_start
                else:
                    quad_offset = -1
            else:
                quad_base = 0.0
                quad_offset = -1

            global_max_base = b_base.max()

            def f_of_x(x):
                m = global_max_base
                if has_linear:
                    vals = base_slice + x * lin_coeffs
                    if has_quad and quad_offset >= 0:
                        vals[quad_offset] = quad_base + x * x
                    vm = vals.max()
                    if vm > m:
                        m = vm
                elif has_quad:
                    qv = quad_base + x * x
                    if qv > m:
                        m = qv
                return m

            hi = max(2.0 * old, target_sum / n) + target_sum / n
            res = minimize_scalar(f_of_x, bounds=(0.0, hi), method='bounded',
                                  options={'xatol': 1e-13, 'maxiter': 300})
            new_val = max(0.0, res.x)
            a[j] = new_val
            max_change = max(max_change, abs(new_val - old))

        s = a.sum()
        if s > 1e-10:
            a *= target_sum / s

        if verbose and (sweep % 10 == 0 or sweep == max_sweeps - 1):
            score = objective_local(a)
            print(f"    CD sweep {sweep}/{max_sweeps}: J={score:.8f}, "
                  f"max_ch={max_change:.2e}", flush=True)

        if max_change < tol:
            if verbose:
                print(f"    CD converged at sweep {sweep}", flush=True)
            break

    return a


def build_constraint_matrix(a):
    """Build matrix A where (A @ g)[k] = (a * g)[k] (linear convolution)."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for k in range(2 * n - 1):
        jl = max(0, k - n + 1)
        jh = min(n - 1, k)
        if jl <= jh:
            A[k, jl:jh + 1] = a[k - jh:k - jl + 1][::-1]
    return A


def lp_best_response(a, M=None):
    """LP: max sum(g) s.t. (a*g)_k <= M, g >= 0. Returns g or None."""
    n = len(a)
    if M is None:
        b = fast_conv(a)
        M = float(b.max())
    if M <= 0:
        return None

    c = -np.ones(n)
    A = build_constraint_matrix(a)
    r = linprog(c, A_ub=A, b_ub=np.full(2 * n - 1, M),
                bounds=[(0, None)] * n, method='highs')
    if r.success and r.x is not None and r.x.sum() > 1e-10:
        return np.maximum(r.x, 0.0)
    return None


def subgradient_method(a, n_steps=200, verbose=False):
    """Subgradient method with averaging for minimizing max_k(b_k)."""
    n = len(a)
    a = a.astype(np.float64).copy()
    target_sum = a.sum()
    a_avg = np.zeros(n)

    for t in range(1, n_steps + 1):
        b = fast_conv(a)
        k_star = int(np.argmax(b))

        grad = np.zeros(n)
        for j in range(n):
            idx = k_star - j
            if 0 <= idx < n:
                grad[j] = 2.0 * a[idx]

        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-15:
            break
        lr = target_sum / (gnorm * np.sqrt(t))

        a_new = a - lr * grad
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s < 1e-10:
            a_new = a.copy()
            s = target_sum
        a_new *= target_sum / s
        a = a_new
        a_avg += a

        if verbose and t % 50 == 0:
            a_sofar = a_avg / t
            a_sofar = np.clip(a_sofar, 0, None)
            ss = a_sofar.sum()
            if ss > 1e-10:
                a_sofar *= target_sum / ss
                print(f"    Subgrad step {t}: J={objective_local(a_sofar):.8f}",
                      flush=True)

    a_avg /= n_steps
    a_avg = np.clip(a_avg, 0.0, None)
    s = a_avg.sum()
    if s > 1e-10:
        a_avg *= target_sum / s
    return a_avg


def lbb_refine(a, maxiter=3000):
    """L-BFGS-B refinement using subgradient as approximate gradient."""
    n = len(a)
    target_sum = a.sum()

    def obj_and_grad(x):
        ax = np.maximum(x, 0.0)
        s = ax.sum()
        if s < 1e-8:
            return 1e10, np.zeros(n)
        ax_n = ax / s
        b = fast_conv(ax_n)
        k_star = int(np.argmax(b))
        val = float(b.max())

        grad_b = np.zeros(n)
        for j in range(n):
            idx = k_star - j
            if 0 <= idx < n:
                grad_b[j] = 2.0 * ax_n[idx]
        # Chain rule for normalization: d(ax_n)/dx_j = (I_j - ax_n)/s
        grad = (grad_b - val * np.ones(n)) / s
        return val, grad

    res = minimize(obj_and_grad, a, method='L-BFGS-B', jac=True,
                   bounds=[(0, None)] * n,
                   options={'maxiter': maxiter, 'max_fun': maxiter * 3,
                            'gtol': 1e-12, 'disp': False})
    if res.fun < 1e10:
        return np.maximum(res.x, 0.0)
    return a


def try_update_wrapper(a, best_state, source=""):
    """Evaluate and update best if improved."""
    a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
    if a.size == 0 or a.sum() < 0.01:
        return
    score = float(evaluate_sequence(a.tolist()))
    if np.isfinite(score) and score < best_state[0]:
        best_state[0] = score
        best_state[1] = a.copy()
        print(f"  ** NEW BEST: J={score:.8f} (n={a.size}) [{source}]", flush=True)


def propose_candidate(seed=42, budget_s=1500, **kwargs):
    """
    Multi-phase optimization to minimize J(a) = 2*n*max(conv(a,a))/(sum a)^2.
    Phases: LP mirror descent -> Coordinate descent -> Subgradient -> L-BFGS-B.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + float(budget_s) - 15.0
    best_state = [float('inf'), None]  # [score, sequence]

    def try_update(a, source=""):
        try_update_wrapper(a, best_state, source)

    # Load parent sequence
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        if p.ndim == 1 and p.size > 0 and np.all(np.isfinite(p)):
            parent = p.copy()
            try_update(parent, "parent")
    except (NameError, AttributeError):
        pass

    # === Phase 1: LP Mirror Descent ===
    lp_lengths = [256, 512, 1024]
    for n in lp_lengths:
        if time.monotonic() > deadline:
            break
        print(f"\n=== Phase 1: LP Mirror Descent n={n} ===", flush=True)
        t0 = time.monotonic()

        inits = []
        if parent is not None:
            if parent.size == n:
                inits.append(parent.copy())
            else:
                t_old = np.linspace(0, 1, parent.size)
                t_new = np.linspace(0, 1, n)
                inits.append(np.maximum(np.interp(t_new, t_old, parent), 0.0))

        inits.append(np.ones(n) / n)
        t_arr = np.arange(n, dtype=np.float64)
        inits.append((1.0 + 0.4 * np.cos(4 * np.pi * t_arr / n)) / n)
        inits.append(np.exp(-3.0 * ((t_arr / n - 0.5)) ** 2))
        inits[inits[-1].sum()] = inits[-1]  # normalize
        inits.append(rng.uniform(0.3, 1.0, n) / n)

        for idx, a0 in enumerate(inits):
            if time.monotonic() > deadline:
                break
            a = np.clip(a0, 0.001, None)
            a /= a.sum()
            try_update(a, f"LP-init-{idx}")

            max_iters = 60
            for it in range(max_iters):
                if time.monotonic() > deadline:
                    break
                b = fast_conv(a)
                M = float(b.max())
                g = lp_best_response(a, M)
                if g is None:
                    break
                g /= g.sum()
                # Adaptive mixing: start large, decrease
                t_mix = 0.4 * (1.0 - it / max_iters) + 0.05
                a_new = (1.0 - t_mix) * a + t_mix * g
                a_new = np.clip(a_new, 0.0, None)
                s = a_new.sum()
                if s < 1e-10:
                    break
                a_new /= s
                a = a_new

            try_update(a, f"LP-n{n}-init{idx}")

        print(f"  LP n={n}: {time.monotonic()-t0:.1f}s elapsed, "
              f"best={best_state[0]:.8f}", flush=True)

    # === Phase 2: Coordinate Descent ===
    cd_targets = []
    if best_state[1] is not None:
        cd_targets.append((best_state[1].copy(), "best"))
    if parent is not None:
        cd_targets.append((parent.copy(), "parent"))
    cd_targets.append((np.ones(512) / 512, "uniform-512"))
    if best_state[1] is not None and best_state[1].size != 512:
        cd_targets.append((best_state[1].copy(), "best-alt"))

    seen_sizes = set()
    for a0, label in cd_targets:
        if time.monotonic() > deadline:
            break
        n = a0.size
        if n in seen_sizes and label != "best":
            continue
        seen_sizes.add(n)

        print(f"\n=== Phase 2: Coordinate Descent n={n} ({label}) ===",
              flush=True)
        t0 = time.monotonic()

        # CD with many sweeps
        a_cd = coordinate_descent(a0, max_sweeps=60, tol=1e-13, verbose=True)
        try_update(a_cd, f"CD-{label}")

        # Subgradient + CD refinement
        if time.monotonic() < deadline - 30:
            a_sub = subgradient_method(a_cd, n_steps=150)
            try_update(a_sub, f"Subgrad-{label}")
            a_cd2 = coordinate_descent(a_sub, max_sweeps=30, tol=1e-13)
            try_update(a_cd2, f"CD2-{label}")

        # Another CD round with tighter tol
        if time.monotonic() < deadline - 30:
            a_cd3 = coordinate_descent(a_cd, max_sweeps=30, tol=1e-14)
            try_update(a_cd3, f"CD3-{label}")

        print(f"  CD {label}: {time.monotonic()-t0:.1f}s elapsed, "
              f"best={best_state[0]:.8f}", flush=True)

    # === Phase 3: L-BFGS-B Polish ===
    if best_state[1] is not None and time.monotonic() < deadline - 60:
        print(f"\n=== Phase 3: L-BFGS-B Polish ===", flush=True)
        t0 = time.monotonic()
        a_best = best_state[1].copy()
        n = a_best.size
        if n <= 2048:
            a_lbb = lbb_refine(a_best, maxiter=5000)
            try_update(a_lbb, "L-BFGS-B")
            # CD after L-BFGS-B
            if time.monotonic() < deadline - 30:
                a_final = coordinate_descent(a_lbb, max_sweeps=30, tol=1e-13)
                try_update(a_final, "CD-after-LBFGS")

        # Try direct CD on a perturbed version
        if time.monotonic() < deadline - 30:
            a_pert = a_best * (1.0 + 0.01 * rng.standard_normal(n))
            a_pert = np.clip(a_pert, 0, None)
            a_pert /= a_pert.sum()
            a_p_cd = coordinate_descent(a_pert, max_sweeps=40, tol=1e-13)
            try_update(a_p_cd, "CD-perturbed")

        print(f"  L-BFGS-B: {time.monotonic()-t0:.1f}s elapsed, "
              f"best={best_state[0]:.8f}", flush=True)

    # === Phase 4: Longer Sequences ===
    for n in [1024, 2048, 4096]:
        if time.monotonic() > deadline:
            break
        print(f"\n=== Phase 4: Longer Sequence n={n} ===", flush=True)
        t0 = time.monotonic()

        if best_state[1] is not None:
            t_old = np.linspace(0, 1, best_state[1].size)
            t_new = np.linspace(0, 1, n)
            a0 = np.maximum(np.interp(t_new, t_old, best_state[1]), 0.0)
            a0 /= a0.sum()
        else:
            a0 = np.ones(n) / n

        # CD on longer sequence (fewer sweeps due to cost)
        max_cd_sweeps = max(5, 200 // (n // 512))
        a_cd_long = coordinate_descent(a0, max_sweeps=max_cd_sweeps, tol=1e-11)
        try_update(a_cd_long, f"CD-long-n{n}")

        # Subgradient on longer
        if time.monotonic() < deadline - 30:
            a_sub_long = subgradient_method(a_cd_long, n_steps=100)
            try_update(a_sub_long, f"Subgrad-long-n{n}")

        print(f"  Long n={n}: {time.monotonic()-t0:.1f}s elapsed, "
              f"best={best_state[0]:.8f}", flush=True)

    # === Phase 5: Final Polish ===
    if best_state[1] is not None and time.monotonic() < deadline - 20:
        print(f"\n=== Phase 5: Final Polish ===", flush=True)
        t0 = time.monotonic()
        a_best = best_state[1].copy()
        n = a_best.size

        # Multiple rounds of CD with decreasing tolerance
        for i, (sweeps, tol) in enumerate([(40, 1e-12), (30, 1e-13), (20, 1e-14)]):
            if time.monotonic() > deadline:
                break
            a_polish = coordinate_descent(a_best, max_sweeps=sweeps, tol=tol)
            try_update(a_polish, f"Polish-{i}")
            if best_state[1] is not None:
                a_best = best_state[1].copy()

        # Random restarts around best
        for trial in range(5):
            if time.monotonic() > deadline:
                break
            noise_scale = 0.005 * (trial + 1)
            a_r = a_best * (1.0 + noise_scale * rng.standard_normal(n))
            a_r = np.clip(a_r, 0, None)
            s = a_r.sum()
            if s < 1e-10:
                continue
            a_r /= s
            a_r_cd = coordinate_descent(a_r, max_sweeps=20, tol=1e-12)
            try_update(a_r_cd, f"Restart-{trial}")
            if best_state[1] is not None:
                a_best = best_state[1].copy()

        print(f"  Final Polish: {time.monotonic()-t0:.1f}s elapsed, "
              f"best={best_state[0]:.8f}", flush=True)

    # Fallback
    if best_state[1] is None:
        best_state[1] = np.ones(256)
        best_state[0] = float(evaluate_sequence(best_state[1].tolist()))

    print(f"\n{'='*60}", flush=True)
    print(f"FINAL RESULT: J={best_state[0]:.8f}, n={best_state[1].size}", flush=True)
    print(f"{'='*60}", flush=True)
    return [float(x) for x in best_state[1]]
