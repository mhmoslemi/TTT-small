def normalize(arr):
    """Normalize to non-negative, sum=1."""
    x = np.asarray(arr, dtype=np.float64).ravel()

    x = np.where(np.isfinite(x), x, 0)

    for k in range(m):
        j_start = max(0, k - n + 1)
        j_end = min(k, n - 1)
        rows.extend([k] * (j_end - j_start + 1))
        cols.extend(range(j_start, j_end + 1))
    rows = np.array(rows, dtype=np.int32)
    cols = np.array(cols, dtype=np.int32)
    # The data will be filled in with f[col] values at solve time
    # Actually, I need a different approach: the matrix A depends on f
    # A @ g where A[k, j] = f[k-j] if 0 <= k-j < n, else 0
    # So A = Toeplitz-like matrix built from f
    return rows, cols

def solve_lp(f, n, M):
    """Solve max sum(g) s.t. (f*g)_k <= M, g >= 0."""
    m = 2 * n - 1
    # Build A: A[k, j] = f[k-j] if 0 <= k-j < n
    # This is a banded matrix
    rows = []
    cols = []
    for k in range(m):
        j_start = max(0, k - n + 1)
        j_end = min(k, n - 1)
        for j in range(j_start, j_end + 1):
            rows.append(k)
            cols.append(j)
    rows = np.array(rows, dtype=np.int32)
    cols = np.array(cols, dtype=np.int32)
    data = f[rows - cols]  # f[k-j]
    A = csr_matrix((data, (rows, cols)), shape=(m, n))
    b = np.full(m, M, dtype=np.float64)
    c = -np.ones(n, dtype=np.float64)
    res = linprog(c, A_ub=A, b_ub=b, bounds=[(0, None)]*n, method='highs')
    return res

def exact_step(f, g, M):
    """Compute the exact maximum step size t such that (f+tg)*(f+tg) <= M."""
    ff = np.convolve(f, f)
    fg = np.convolve(f, g)
    gg = np.convolve(g, g)
    # For each k: gg[k]*t^2 + 2*fg[k]*t + (ff[k]-M) <= 0
    a = gg  # >= 0
    b = 2 * fg
    c = ff - M  # <= 0 (since M = max(ff))
    # For a_k > 0: t_upper = (-b_k + sqrt(b_k^2 - 4*a_k*c_k)) / (2*a_k)
    # For a_k == 0 and b_k > 0: t_upper = -c_k / b_k
    # For a_k == 0 and b_k <= 0: no constraint (always satisfied since c_k <= 0)
    
    # Use np.where and clipping for numerical stability
    disc = b**2 - 4 * a * c
    disc = np.maximum(disc, 0)  # Clip for numerical stability
    
    t_upper = np.full(len(a), np.inf, dtype=np.float64)
    
    # Case 1: a > threshold
    mask1 = a > 1e-15
    t_upper[mask1] = (-b[mask1] + np.sqrt(disc[mask1])) / (2 * a[mask1])
    
    # Case 2: a <= threshold, b > threshold
    mask2 = (~mask1) & (b > 1e-15)
    t_upper[mask2] = -c[mask2] / b[mask2]
    
    # Case 3: a <= threshold, b <= threshold: no constraint (c <= 0)
    # t_upper remains inf
    
    t_max = np.min(t_upper)
    if not np.isfinite(t_max) or t_max <= 0:
        return 0.0
    return min(1.0, t_max)

Actually, I realize I should also think about whether the LP formulation is the right one. The LP finds g that maximizes sum(g) subject to (f*g)_k <= M. This gives a direction that tries to increase the "volume" of g while keeping the cross-correlation with f bounded.

An alternative formulation: minimize max_k (g*g)_k subject to sum(g) = 1, (f*g)_k <= M, g >= 0. This is a second-order cone program (SOCP) or can be reformulated as an LP with an extra variable:
min z
s.t. (g*g)_k <= z for all k
     sum(g) = 1
     (f*g)_k <= M
     g >= 0

But (g*g)_k <= z is non-linear (quadratic) in g. However, if we fix g, it's linear in z. So we can use a bisection on z:
- For a given z, check feasibility: is there g >= 0, sum(g) = 1, (f*g)_k <= M, (g*g)_k <= z?
- The (g*g)_k <= z constraint is still non-linear.

Alternatively, we can linearize: (g*g)_k = sum_j g_j * g_{k-j} <= sum_j g_j * M_g where M_g = max(g). But this is very loose.

I think the original LP formulation is the best tractable option. Let me stick with it.

"""
Improved LP-based iterative search for minimizing 2*n*max(convolve(f,f))/(sum(f))^2.
Key improvements over previous version:
1. Exact step size computation (no bisection needed - closed form via quadratic roots)
2. Precomputed sparse matrix structure for fast LP setup
3. Adaptive kick strategy with multiple perturbation types
4. Subgradient descent as complementary method
5. Simulated annealing finishing phase
6. Multi-start with diverse initializations
"""

import time
import numpy as np
from scipy.optimize import linprog
from scipy.sparse import csr_matrix


def normalize_arr(arr):
    """Normalize to non-negative vector with sum 1."""
    try:
        x = np.asarray(arr, dtype=np.float64).ravel()
    except Exception:
        return None
    if x.size == 0:
        return None
    x = np.where(np.isfinite(x), x, 0.0)
    x = np.maximum(x, 0.0)
    s = float(x.sum())
    if s < 1e-300:
        return None
    return x / s


def compute_score(x):
    """Compute 2*n*max(conv(x,x)) assuming sum(x)=1, x>=0."""
    n = x.size
    if n == 0:
        return float('inf')
    conv = np.convolve(x, x)
    m = float(np.max(conv))
    if not np.isfinite(m):
        return float('inf')
    return 2.0 * n * m


def precompute_conv_indices(n):
    """Precompute row/col/index arrays for the convolution sparse matrix."""
    m = 2 * n - 1
    k_arr = np.arange(m, dtype=np.int64)
    j_start = np.maximum(0, k_arr - n + 1).astype(np.int64)
    j_end = np.minimum(n - 1, k_arr).astype(np.int64)
    counts = (j_end - j_start + 1).astype(np.int64)
    total = int(counts.sum())

    cum = np.cumsum(counts)
    starts = cum - counts
    local = np.arange(total, dtype=np.int64) - np.repeat(starts, counts)

    rows = np.repeat(k_arr, counts).astype(np.int32)
    cols = (np.repeat(j_start, counts) + local).astype(np.int32)
    data_idx = (rows.astype(np.int64) - cols.astype(np.int64)).astype(np.int32)

    return rows, cols, data_idx, m


def build_conv_matrix(f, rows, cols, data_idx, n, m):
    """Build the sparse convolution matrix A where A@g = conv(f,g)."""
    data = f[data_idx].astype(np.float64)
    A = csr_matrix((data, (rows, cols)), shape=(m, n))
    return A


def solve_direction_lp(f, n, M, rows, cols, data_idx, m):
    """Solve max sum(g) s.t. (f*g)_k <= M, g >= 0."""
    data = f[data_idx].astype(np.float64)
    A = csr_matrix((data, (rows, cols)), shape=(m, n))
    b = np.full(m, M, dtype=np.float64)
    c = -np.ones(n, dtype=np.float64)
    bounds = [(0.0, None)] * n
    res = linprog(c, A_ub=A, b_ub=b, bounds=bounds, method='highs',
                  options={'presolve': True, 'time_limit': 30.0})
    return res


def exact_max_step(f, g, M):
    """
    Compute exact max t in [0,1] such that (f+t*(g-f))*(f+t*(g-f)) <= M elementwise.
    Equivalent to: for all k, the quadratic in s: ff_k + 2s*fg_k + s^2*gg_k <= M
    where s = t (since we write f_new = f + s*(g-f), but actually we want f_new = (1-t)f + tg).
    
    Let f_new = (1-t)f + t*g. Then f_new*f_new = f*f + 2t(f*(g-f)) + t^2*((g-f)*(g-f))... 
    Actually let's just compute directly with h = g - f:
    f_new = f + t*h
    f_new*f_new = f*f + 2t*(f*h) + t^2*(h*h)
    We need: f*f + 2t*(f*h) + t^2*(h*h) <= M for all k.
    """
    h = g - f
    f_h = np.convolve(f, h)
    h_h = np.convolve(h, h)
    ff = np.convolve(f, f)

    # Constraint: h_h[k]*t^2 + 2*f_h[k]*t + (ff[k]-M) <= 0
    a = h_h
    b = 2.0 * f_h
    c = ff - M  # <= 0 since M = max(ff)

    # For each k, find the largest t >= 0 satisfying the quadratic <= 0
    # If a_k == 0: linear constraint b_k*t + c_k <= 0
    # If a_k > 0: quadratic, roots at (-b ± sqrt(b^2-4ac))/(2a), valid between roots
    # Since c_k <= 0, t=0 is valid, and we want the upper root.

    t_upper = np.full(len(a), 1.0, dtype=np.float64)

    # Case: a > 0
    mask_quad = a > 1e-18
    if np.any(mask_quad):
        aq = a[mask_quad]
        bq = b[mask_quad]
        cq = c[mask_quad]
        disc = bq * bq - 4.0 * aq * cq
        disc = np.maximum(disc, 0.0)
        # Upper root: (-b + sqrt(disc)) / (2a)
        root = (-bq + np.sqrt(disc)) / (2.0 * aq)
        t_upper[mask_quad] = root

    # Case: a == 0, b > 0
    mask_lin = (~mask_quad) & (b > 1e-18)
    if np.any(mask_lin):
        t_upper[mask_lin] = -c[mask_lin] / b[mask_lin]

    # Case: a == 0, b <= 0: always satisfied (c <= 0), t_upper stays 1.0

    t_max = float(np.min(t_upper))
    if not np.isfinite(t_max) or t_max < 0:
        return 0.0
    return min(1.0, t_max)


def subgradient_step(f, n, eta, rng):
    """Take a subgradient descent step on 2n*max(f*f) with simplex projection."""
    ff = np.convolve(f, f)
    M = ff.max()
    # Active set: where ff is within 10% of max
    threshold = M * 0.9
    active = ff >= threshold
    n_active = np.sum(active)
    if n_active == 0:
        active = ff >= M - 1e-10
        n_active = max(1, np.sum(active))

    # Subgradient: d_j = (1/|A|) * sum_{k in A} h_k[j] where h_k is the kth row of conv matrix
    # h_k[j] = 1 if 0 <= k-j < n (i.e., the "kernel" for computing (f*f)_k)
    # Actually d_j = (1/|A|) * sum_{k in A} sum_{i: valid} f[k-j] ... 
    # More precisely: d( (f*f)_k )/d(f_j) = f[k-j] if 0<=k-j<n
    # So subgrad_j = (1/|A|) * sum_{k in A} f[k-j] * (indicator 0<=k-j<n)
    
    # Compute via convolution: the subgradient is (1/|A|) * convolve(indicator_A, f)
    # where indicator_A is the vector of length 2n-1 with 1s at active positions
    ind_A = np.zeros(2 * n - 1, dtype=np.float64)
    ind_A[active] = 1.0
    d = np.convolve(ind_A[:n], f)[:n] / n_active  # This isn't quite right
    # Actually: sum_{k in A} f[k-j] for valid k means sum over k where k-j is in [0,n-1]
    # = sum_{k} ind_A[k] * f[k-j] * (0 <= k-j < n)
    # = (ind_A * rev_f) evaluated appropriately... 
    # Let's just do it directly: for each j, sum over k in A where 0<=k-j<n of f[k-j]
    # = sum_{i=0}^{n-1} f[i] * (sum_{k in A, k=j+i} 1) = sum_i f[i] * ind_A[j+i] (if j+i < 2n-1)
    
    # Use FFT for speed
    # d_j = sum_{i=0}^{min(n-1, 2n-2-j)} f[i] * ind_A[j+i]
    # This is a cross-correlation of f and ind_A
    N_fft = 4 * n
    F_f = np.fft.rfft(f, N_fft)
    F_ind = np.fft.rfft(ind_A, N_fft)
    # Cross-correlation: corr[j] = sum_i f[i] * ind_A[j+i] = ifft(conj(F_f) * F_ind)[j]... 
    # Actually cross-correlation of a and b: sum_i a[i]*b[i+j] = ifft(conj(A)*B)[j]
    d = np.fft.irfft(np.conj(F_f) * F_ind, N_fft)[:n].real / n_active

    # Project -d onto tangent space of simplex
    d = d - np.mean(d)
    f_new = f - eta * d
    f_new = np.maximum(0, f_new)
    s = f_new.sum()
    if s < 1e-10:
        return f.copy()
    f_new /= s
    return f_new


def simulated_annealing(f, n, budget, rng, best_score):
    """Run simulated annealing to refine the sequence."""
    f = f.copy()
    best_f = f.copy()
    best_sc = compute_score(f)
    start_time = time.monotonic()
    deadline = start_time + budget

    T_start = 0.01
    T_end = 0.0001
    step_size = 0.05
    n_perturb = max(1, n // 64)

    step = 0
    while time.monotonic() < deadline:
        step += 1
        progress = (time.monotonic() - start_time) / max(budget, 1e-6)
        T = T_start * (T_end / T_start) ** min(1.0, progress)

        # Perturb random subset
        idx = rng.choice(n, size=n_perturb, replace=False)
        delta = rng.standard_normal(n_perturb) * step_size
        f_new = f.copy()
        f_new[idx] += delta
        f_new = np.maximum(0, f_new)
        s = f_new.sum()
        if s < 1e-10:
            continue
        f_new /= s

        sc_new = compute_score(f_new)

        if sc_new < best_sc:
            f = f_new
            best_f = f_new.copy()
            best_sc = sc_new
        elif rng.random() < np.exp(-(sc_new - best_sc) / T):
            f = f_new

        if step % 500 == 0:
            print(f"    SA step={step} T={T:.6f} best={best_sc:.6f}", flush=True)

    return best_f, best_sc

    iter_count = 0
    stall = 0
    last_improve_iter = 0

        # Solve LP
        try:
            res = solve_direction_lp(f, n, M, rows, cols, data_idx, m)
        except Exception as e:
            print(f"    LP error at iter {iter_count}: {e}", flush=True)
            stall += 5
            if stall > 30:
                break
            continue

        if res.status != 0 or res.x is None:
            stall += 3
            if stall > 30:
                break
            # Kick
            eps = min(0.5, 0.05 + 0.02 * stall)
            noise = rng.dirichlet(np.ones(n, dtype=np.float64))
            f = (1 - eps) * f + eps * noise
            f = normalize_arr(f)
            continue

        g = np.asarray(res.x, dtype=np.float64).copy()
        g = np.maximum(0, g)
        s_g = float(g.sum())
        if s_g < 1e-10:
            stall += 3
            if stall > 30:
                break
            continue
        g /= s_g

        # Check if LP gives improvement direction
        # The LP maximizes sum(g) with (f*g)_k <= M. If optimal sum(g)*n > n (i.e., sum(g)>1 under our normalization where sum(f)=1), then g is "bigger" than f in some sense.
        # Actually with our normalization sum(f)=1, the LP constraint is (f*g)_k <= M.
        # The optimal g has sum(g) = s_g_star. If s_g_star > 1, then the rescaled g (sum=1) is a valid direction.
        # The key metric: does (f+tg)*(f+tg) have max < M for some t>0?
        
        t_max = exact_max_step(f, g, M)

        if t_max < 1e-12:
            # No improvement possible with this direction
            stall += 1
            if stall > 15:
                # Try subgradient step
                eta = 0.1 / (iter_count ** 0.3)
                f_try = subgradient_step(f, n, eta, rng)
                sc_try = compute_score(f_try)
                if sc_try < best_score:
                    f = f_try
                    best_f = f.copy()
                    best_score = sc_try
                    stall = 0
                    last_improve_iter = iter_count
                elif rng.random() < 0.3:
                    # Kick
                    kick_size = min(0.3, 0.05 + 0.01 * stall)
                    noise = rng.dirichlet(np.ones(n, dtype=np.float64))
                    f = (1 - kick_size) * f + kick_size * noise
                    f = normalize_arr(f)
                if stall > 60:
                    # Large restart
                    noise = rng.dirichlet(np.ones(n, dtype=np.float64) * 0.5)
                    f = 0.3 * best_f + 0.7 * noise
                    f = normalize_arr(f)
                    stall = 20
            continue

        # Take the step
        # Try the full step and a few fractions
        candidates_t = []
        t_try = min(1.0, t_max)
        for frac in [1.0, 0.8, 0.5, 0.3, 0.1]:
            tv = frac * t_try
            if tv > 1e-15 and tv <= 1.0:
                candidates_t.append(tv)

        best_cand_score = best_score
        best_cand_f = None
        best_t = 0

        for tv in candidates_t:
            f_new = (1.0 - tv) * f + tv * g
            f_new = np.maximum(0, f_new)
            s_new = float(f_new.sum())
            if s_new < 1e-10:
                continue
            f_new /= s_new
            sc = compute_score(f_new)
            if np.isfinite(sc) and sc < best_cand_score:
                best_cand_score = sc
                best_cand_f = f_new
                best_t = tv

        if best_cand_f is not None:
            f = best_cand_f
            best_f = f_new.copy()
            best_score = best_cand_score
            stall = 0
            last_improve_iter = iter_count
            if iter_count <= 5 or iter_count % 20 == 0:
                dt = time.monotonic() - t0
                print(f"    iter={iter_count} score={best_score:.6f} t={best_t:.4f} "
                      f"t_max={t_max:.4f} dt={dt:.2f}s", flush=True)
        else:
            stall += 1
            if stall % 10 == 0:
                # Small perturbation
                eps = min(0.2, 0.02 * stall)
                noise = rng.dirichlet(np.ones(n, dtype=np.float64))
                f = (1 - eps) * f + eps * noise
                f = normalize_arr(f)
            if stall > 60:
                # Big kick back toward best
                f = 0.5 * best_f + 0.5 * rng.dirichlet(np.ones(n, dtype=np.float64))
                f = normalize_arr(f)
                stall = 30

        # Periodic subgradient exploration
        if iter_count % 25 == 0 and stall > 3:
            eta = 0.05 / (1 + iter_count * 0.01)
            f_try = subgradient_step(f, n, eta, rng)
            sc_try = compute_score(f_try)
            if sc_try < best_score:
                f = f_try
                best_f = f.copy()
                best_score = sc_try
                stall = 0
                last_improve_iter = iter_count

    return best_f, best_score


def interpolate_to_length(x, n_target):
    """Interpolate a normalized sequence to a different length."""
    x = normalize_arr(x)
    if x is None:
        return None
    n_src = x.size
    if n_src == n_target:
        return x
    if n_src == 1:
        return np.ones(n_target, dtype=np.float64) / n_target
    old_pos = np.linspace(0.0, 1.0, n_src)
    new_pos = np.linspace(0.0, 1.0, n_target)
    out = np.interp(new_pos, old_pos, x)
    return normalize_arr(out)


def propose_candidate(seed=42, budget_s=1500, **kwargs):
    """
    Search for a low-scoring nonnegative sequence using improved LP iteration,
    subgradient descent, and simulated annealing.
    """
    total_budget = max(1.0, float(budget_s))
    start_time = time.monotonic()
    deadline = start_time + total_budget - 15.0  # Reserve 15s for finalization
    
    rng = np.random.default_rng(seed)

    # Initialize from given sequence
    best_f = None
    best_score = float('inf')
    given_f = None

    if total_budget >= 30.0:
        try:
            raw = np.asarray(height_sequence_1, dtype=np.float64)
            given_f = normalize_arr(raw)
            if given_f is not None:
                best_f = given_f.copy()
                best_score = compute_score(given_f)
                print(f"Given sequence: n={given_f.size}, score={best_score:.6f}", flush=True)
        except Exception as e:
            print(f"Could not load given sequence: {e}", flush=True)
            given_f = None

    if best_f is None:
        best_f = np.ones(1024, dtype=np.float64) / 1024.0
        best_score = 2.0
        print("Using uniform initialization, score=2.0", flush=True)

    n_main = best_f.size
    print(f"\nMain search at n={n_main}", flush=True)

    # Phase 1: LP optimization from the given/best sequence
    alloc1 = min(900.0, total_budget * 0.55)
    if alloc1 > 30.0:
        print(f"\n=== Phase 1: LP from best start, budget={alloc1:.0f}s ===", flush=True)
        f1, s1 = optimize_with_lp(best_f, n_main, alloc1, seed=seed)
        if s1 < best_score:
            best_score = s1
            best_f = f1
            print(f"  Phase 1 best: {best_score:.6f}", flush=True)

    remaining = deadline - time.monotonic()
    if remaining < 30.0:
        print("\nPhase 1 used most of the budget.", flush=True)
    else:
        # Phase 2: Try different starting points
        print(f"\n=== Phase 2: Alternative starts, remaining={remaining:.0f}s ===", flush=True)

        starts = []
        # Uniform
        starts.append(('uniform', np.ones(n_main, dtype=np.float64) / n_main))
        # Random Dirichlet
        starts.append(('dirichlet', rng.dirichlet(np.ones(n_main, dtype=np.float64))))
        # Concentrated middle
        mid = np.zeros(n_main, dtype=np.float64)
        w = n_main // 4
        mid[n_main//2 - w:n_main//2 + w] = 1.0
        starts.append(('middle', normalize_arr(mid)))
        # Two bumps
        bump = np.zeros(n_main, dtype=np.float64)
        bump[n_main//4 - w//2:n_main//4 + w//2] = 1.0
        bump[3*n_main//4 - w//2:3*n_main//4 + w//2] = 1.0
        starts.append(('two_bump', normalize_arr(bump)))
        # Upsampled best with noise
        if best_f is not None:
            noisy_best = best_f + 0.1 * rng.standard_normal(n_main)
            starts.append(('noisy_best', normalize_arr(noisy_best)))

        # Allocate time to each start
        n_starts = len(starts)
        alloc_each = remaining * 0.6 / n_starts

        for i, (name, init) in enumerate(starts):
            rem = deadline - time.monotonic()
            if rem < 10.0:
                break
            tb = min(alloc_each, rem - 5.0)
            if tb < 10.0:
                continue
            print(f"\n  Start '{name}', budget={tb:.0f}s", flush=True)
            f_res, s_res = optimize_with_lp(init, n_main, tb, seed=seed + 100 + i)
            if s_res < best_score:
                best_score = s_res
                best_f = f_res
                print(f"  New global best from '{name}': {best_score:.6f}", flush=True)

    remaining = deadline - time.monotonic()
    if remaining > 50.0 and best_f is not None:
        # Phase 3: Try a larger sequence
        n_large = min(2048, 2 * n_main)
        alloc3 = min(remaining * 0.5, 300.0)
        if alloc3 > 30.0:
            print(f"\n=== Phase 3: Larger n={n_large}, budget={alloc3:.0f}s ===", flush=True)
            init_large = interpolate_to_length(best_f, n_large)
            if init_large is not None:
                f_res, s_res = optimize_with_lp(init_large, n_large, alloc3, seed=seed + 999)
                if s_res < best_score:
                    best_score = s_res
                    best_f = f_res
                    print(f"  New global best at n={n_large}: {best_score:.6f}", flush=True)

    remaining = deadline - time.monotonic()
    if remaining > 20.0 and best_f is not None:
        # Phase 4: Simulated annealing refinement
        print(f"\n=== Phase 4: SA refinement, budget={remaining:.0f}s ===", flush=True)
        f_res, s_res = simulated_annealing(best_f, best_f.size, remaining * 0.9, rng, best_score)
        if s_res < best_score:
            best_score = s_res
            best_f = f_res
            print(f"  SA improved to: {best_score:.6f}", flush=True)

    # Finalize
    # Ensure the sequence is valid for evaluate_sequence
    seq = [float(max(0.0, min(1000.0, x))) for x in best_f]
    s_check = sum(seq)
    if s_check < 0.01:
        seq = [1.0 / 64.0] * 64

    # Verify
    try:
        final_score = float(evaluate_sequence(seq))
    except Exception:
        final_score = float('inf')
        if given_f is not None:
            seq = [float(max(0.0, min(1000.0, x))) for x in given_f]
            try:
                final_score = float(evaluate_sequence(seq))
            except Exception:
                seq = [1.0 / 64.0] * 64
                final_score = 2.0

    elapsed = time.monotonic() - start_time
    print(f"\nFINAL: n={len(seq)}, local_score={best_score:.6f}, "
          f"evaluate_score={final_score:.6f}, elapsed={elapsed:.1f}s", flush=True)

    return seq
