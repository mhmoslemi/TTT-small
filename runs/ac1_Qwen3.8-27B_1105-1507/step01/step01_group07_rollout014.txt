def frank_wolfe(a, max_iter=10000, verbose=False):
    """Frank-Wolfe (conditional gradient) for minimizing max_k conv(a,a)_k on simplex."""
    n = len(a)
    a = a / a.sum()
    best_a = a.copy()
    best_val = float(np.max(np.convolve(a, a)))
    
    for t in range(max_iter):
        conv = np.convolve(a, a)
        k_star = int(np.argmax(conv))
        # Subgradient: grad[m] = 2 * a[k_star - m] if 0 <= k_star - m < n else 0
        grad = np.zeros(n)
        m_min = max(0, k_star - n + 1)
        m_max = min(n - 1, k_star)
        grad[m_min:m_max+1] = 2.0 * a[k_star - m_max : k_star - m_min + 1][::-1]
        
        # Linear minimization over simplex: min grad[m] * a[m] s.t. sum a = 1, a >= 0
        # Optimal: put all mass on argmin grad
        m_star = int(np.argmin(grad))
        
        # Step size: minimize f((1-alpha)*a + alpha*e_{m_star})
        # Use backtracking or exact line search
        # Simple: alpha = 2/(t+2)
        alpha = 2.0 / (t + 2)
        
        # Alternatively, do a line search
        a_new = (1 - alpha) * a + alpha * np.zeros(n)
        a_new[m_star] = alpha
        
        val = float(np.max(np.convolve(a_new, a_new)))
        if val < best_val:
            best_val = val
            best_a = a_new.copy()
        
        a = a_new
        if verbose and (t % 1000 == 0):
            print(f"    FW t={t}: val={val:.6f} best={best_val:.6f}")
    
    return best_a, best_val

def alternating_lp(a, max_iter=200, t=0.5, verbose=False):
    """Improved alternating LP."""
    n = len(a)
    a = a.copy()
    a = np.clip(a, 0, None)
    s = a.sum()
    if s < 1e-10:
        a = np.ones(n) / n
    else:
        a = a / s
    
    best_a = a.copy()
    best_val = float(np.max(np.convolve(a, a)))
    
    for it in range(max_iter):
        conv = np.convolve(a, a)
        M = float(np.max(conv))
        
        # Build LP: max sum(g) s.t. conv(a, g)_k <= M, g >= 0
        # conv(a, g)_k = sum_j a[j] * g[k-j]
        # This is linear in g!
        # Variable: g (n variables)
        # Constraints: sum_j a[j] * g[k-j] <= M for k = 0, ..., 2n-2
        # Objective: -sum(g) (minimize negative sum)
        
        A_mat = np.zeros((2*n-1, n))
        for k in range(2*n-1):
            j_start = max(0, k - n + 1)
            j_end = min(n - 1, k)
            # conv(a,g)_k = sum_{j=j_start}^{j_end} a[j] * g[k-j]
            # g index: k-j, ranges from k-j_end to k-j_start
            g_start = k - j_end
            g_end = k - j_start
            A_mat[k, g_start:g_end+1] = a[j_start:j_end+1]
        
        c = -np.ones(n)
        b_ub = np.full(2*n-1, M)
        bounds = [(0, None)] * n
        
        try:
            res = linprog(c, A_ub=A_mat, b_ub=b_ub, bounds=bounds, method='highs')
            if res.status != 0:
                break
            g = res.x
        except Exception:
            break
        
        g_sum = float(np.sum(g))
        if g_sum <= 1.0 + 1e-6:
            break
        
        # Blend
        g = g / g_sum
        a_new = (1 - t) * a + t * g
        a_new = np.clip(a_new, 0, None)
        s2 = a_new.sum()
        if s2 < 1e-10:
            break
        a_new = a_new / s2
        
        val = float(np.max(np.convolve(a_new, a_new)))
        if val < best_val:
            best_val = val
            best_a = a_new.copy()
            if verbose and (it % 10 == 0):
                print(f"    ALP it={it}: val={val:.6f} gsum={g_sum:.4f}")
        
        a = a_new
    
    return best_a, best_val

Minimize 2*n*max(conv(a,a))/(sum(a))^2 using mirror descent (multiplicative weights),
Frank-Wolfe, and alternating LP. Exploits structure with zero-insertion patterns.

import numpy as np

def compute_obj(a):
    n = len(a)
    if n == 0:
        return np.inf
    conv = np.convolve(a, a)
    max_b = float(np.max(conv))
    sum_a = float(np.sum(a))
    if sum_a < 0.01:
        return np.inf
    return 2.0 * n * max_b / (sum_a * sum_a)

def get_grad_hard_max(a):
    """Compute subgradient of max_k conv(a,a)_k using the hard max."""
    n = len(a)
    conv = np.convolve(a, a)
    k_star = int(np.argmax(conv))
    grad = np.zeros(n)
    m_min = max(0, k_star - n + 1)
    m_max = min(n - 1, k_star)
    grad[m_min:m_max+1] = 2.0 * a[k_star - m_max : k_star - m_min + 1][::-1]
    return grad, float(conv[k_star])

def get_grad_smooth_max(a, smoothing):
    """Compute gradient of smoothed max using log-sum-exp."""
    n = len(a)
    conv = np.convolve(a, a)
    # Softmax for stability
    conv_shifted = conv - conv.max()
    p = np.exp(smoothing * conv_shifted)
    p = p / p.sum()
    # grad[m] = 2 * sum_j p[m+j] * a[j]
    # Using FFT: this is cross-correlation of p and a
    a_flip = a[::-1]
    # conv(p, a_flip)[n-1+m] = sum_j p[j] * a_flip[n-1+m-j]
    # = sum_j p[j] * a[j-m+n-1-n+1]... 
    # Let me just use the formula I derived: grad[m] = 2 * fftconv(p, a_flip)[n-1+m]
    # p has length 2n-1, a_flip has length n
    # conv has length (2n-1) + n - 1 = 3n-2
    # We want indices n-1, n, ..., 2n-2
    full = np.fft.irfft(np.fft.rfft(p, 3*n-2) * np.fft.rfft(a_flip, 3*n-2), 3*n-2)
    grad = 2.0 * full[n-1:2*n-1]
    return grad
