"""
Search for low-score non-negative step-function coefficients using
smoothed gradient descent (log-sum-exp), multiplicative updates,
and LP-based direction finding.

Key innovations over previous approach:
1. FFT-based convolution for O(n log n) score/gradient computation
2. Smoothed objective with adaptive temperature for stable gradients
3. Nesterov momentum for accelerated convergence
4. Strategic combination of gradient steps and multiplicative updates
5. Diverse initialization including combinatorial structures
6. Multi-scale exploration across lengths 256-1024
"""

import time
import numpy as np

try:
    from scipy.optimize import linprog
    HAVE_SCIPY = True
except Exception:
    HAVE_SCIPY = False

try:
    from scipy.fft import fft, ifft, next_fast_len
    HAVE_SCIPY_FFT = True
except Exception:
    HAVE_SCIPY_FFT = False


def normalize_to_sum1(a):
    a = np.asarray(a, dtype=np.float64)
    if a.size == 0:
        return None
    a = np.clip(a, 0.0, 1000.0)
    if not np.isfinite(a).all():
        return None
    s = float(np.sum(a))
    if not np.isfinite(s) or s <= 1e-300:
        return None
    return a / s


def local_score(a):
    a = np.asarray(a, dtype=np.float64)
    if a.size == 0:
        return float('inf')
    a = np.clip(a, 0.0, 1000.0)
    if not np.isfinite(a).all():
        return float('inf')
    s = float(np.sum(a))
    if not np.isfinite(s) or s <= 1e-12:
        return float('inf')
    c = np.convolve(a, a)
    m = float(np.max(c))
    if not np.isfinite(m):
        return float('inf')
    return float(2.0 * len(a) * m / (s * s))


def fast_conv_autocorr(a):
    """Compute autocorrelation conv(a,a) using FFT for speed."""
    n = len(a)
    if HAVE_SCIPY_FFT and n > 256:
        N = next_fast_len(2 * n - 1)
        fa = fft(a, N)
        c = ifft(fa * fa).real[:2 * n - 1]
        return c
    else:
        return np.convolve(a, a)


def compute_smooth_gradient(a, beta):
    """
    Compute gradient of g_beta(a) = (1/beta)*log(sum_k exp(beta*c_k))
    where c = conv(a,a).
    Returns (gradient, softmax_weights, conv).
    """
    n = len(a)
    c = fast_conv_autocorr(a)
    c_max = float(np.max(c))
    c_shifted = c - c_max
    exp_bc = np.exp(beta * c_shifted)
    p_sum = float(np.sum(exp_bc))
    if p_sum <= 1e-300:
        return None, None, c
    p = exp_bc / p_sum

    # Gradient: grad[j] = 2 * sum_{m=0}^{n-1} p[j+m] * a[m]
    # Using convolution: G[j] = (a * p_rev)[2n-2-j]
    p_rev = p[::-1]
    if HAVE_SCIPY_FFT and n > 128:
        N = next_fast_len(3 * n - 2)
        fa = fft(a, N)
        fp = fft(p_rev, N)
        conv_full = ifft(fa * fp).real[:3 * n - 2]
    else:
        conv_full = np.convolve(a, p_rev)

    # G[j] = conv_full[2n-2-j] for j=0,...,n-1
    G = conv_full[2 * n - 2 : n - 2 : -1]
    grad = 2.0 * G
    return grad, p, c


def smooth_grad_descent(a, total_time, seed, beta_start=None, use_momentum=True):
    """
    Run smoothed gradient descent to minimize max(conv(a,a)) subject to
    sum(a)=1, a>=0. Uses log-sum-exp smoothing with adaptive beta.
    """
    a = normalize_to_sum1(a)
    if a is None:
        return a, float('inf')

    n = len(a)
    if beta_start is None:
        beta_start = 8.0 / np.sqrt(n)

    beta_max = 80.0 / np.sqrt(n)
    beta_min = 2.0 / np.sqrt(n)

    t0 = time.monotonic()
    deadline = t0 + float(total_time)
    rng = np.random.default_rng(seed)

    best = a.copy()
    best_score = local_score(best)

    beta = beta_start
    lr = 1.0 / n
    mu = 0.85
    momentum = np.zeros(n)

    it = 0
    stall = 0
    consecutive_reject = 0

    while time.monotonic() < deadline:
        # Check remaining time
        rem = deadline - time.monotonic()
        if rem < 0.1:
            break

        # Compute gradient
        grad, p, c = compute_smooth_gradient(a, beta)
        if grad is None or not np.isfinite(grad).all():
            break

        # Momentum update
        if use_momentum:
            momentum = mu * momentum + (1.0 - mu) * grad
            step = lr * momentum
        else:
            step = lr * grad

        a_new = a - step
        a_new = np.clip(a_new, 0.0, None)
        a_new = normalize_to_sum1(a_new)

        if a_new is None:
            a_new = a.copy()
            lr *= 0.5
            it += 1
            continue

        sc = local_score(a_new)

        if np.isfinite(sc) and sc < best_score:
            best = a_new.copy()
            best_score = float(sc)
            stall = 0
            consecutive_reject = 0
            # Slightly increase lr if consistently improving
            if it % 100 == 0 and lr < 3.0 / n:
                lr *= 1.03
        else:
            consecutive_reject += 1
            if consecutive_reject > 20:
                # Rejected: reduce step
                lr = max(lr * 0.6, 0.01 / n)
                a_new = a  # revert
                momentum = momentum * 0.5
            elif consecutive_reject > 5:
                lr *= 0.85

        # Anneal beta periodically
        if it % 300 == 0:
            if stall < 30:
                beta = min(beta_max, beta * 1.08)
            else:
                beta = max(beta_min, beta * 0.7)

        # Periodic exact evaluation for best tracking
        if it % 200 == 0:
            exact_sc = local_score(best)
            if exact_sc < best_score:
                best_score = exact_sc

        # Stall detection and restart
        stall += 1
        if stall > 500:
            # Restart from best with noise
            noise = np.exp(rng.normal(0.0, 0.15, n))
            a = best * noise
            a = normalize_to_sum1(a)
            if a is None:
                a = best.copy()
            momentum = np.zeros(n)
            lr = 0.5 / n
            beta = beta_start
            stall = 0
            consecutive_reject = 0

        a = a_new
        it += 1

    return best, best_score


def multiplicative_update(a, eta, K, force_sym=False):
    """Multiplicative (exponentiated gradient) update on active convolution peaks."""
    a = np.asarray(a, dtype=np.float64)
    n = a.size
    if n == 0:
        return a

    c = fast_conv_autocorr(a)
    M = float(np.max(c))
    if not np.isfinite(M) or M <= 0.0:
        return a.copy()

    if K is None or K <= 0:
        K = max(8, n // 6)
    K = int(K)
    if K >= c.size:
        active = np.nonzero(c > 1e-15)[0]
        if active.size == 0:
            active = np.array([int(np.argmax(c))], dtype=np.int64)
    else:
        active = np.argpartition(c, -K)[-K:]

    # Compute gradient contribution from active peaks
    h = np.zeros(n, dtype=np.float64)
    for k in active:
        k = int(k)
        s = max(0, k - (n - 1))
        e = min(k, n - 1)
        if s > e:
            continue
        i = np.arange(s, e + 1, dtype=np.int64)
        j = k - i
        h[i] += a[j]
        if np.any(i != j):
            mask = (i != j)
            h[j[mask]] += a[i[mask]]

    maxh = float(np.max(h))
    if not np.isfinite(maxh) or maxh <= 0.0:
        return a.copy()

    meanh = float(np.mean(h))
    fac = eta * (meanh - h) / maxh
    fac = np.clip(fac, -4.0, 4.0)
    a2 = a * np.exp(fac)
    a2 = np.clip(a2, 0.0, 1000.0)
    a2 = normalize_to_sum1(a2)
    if a2 is None:
        return a.copy()
    return a2


def multiplicative_optimize(a, max_time, seed, force_sym=False):
    """Iterative multiplicative updates with adaptive parameters."""
    a = normalize_to_sum1(a)
    if a is None:
        return None, float('inf')
    if force_sym:
        a = 0.5 * (a + a[::-1])
        a = normalize_to_sum1(a)

    best = a.copy()
    bestsc = local_score(best)
    if max_time <= 0.0:
        return best, bestsc

    t0 = time.monotonic()
    deadline = t0 + float(max_time)
    rng = np.random.default_rng(seed)
    n = best.size

    K = int(np.clip(max(16, n // 5), 8, 200))
    eta = 0.30
    stuck = 0
    it = 0

    while time.monotonic() < deadline:
        if it % 200 == 0:
            mult = 0.7 + 0.6 * float(rng.random())
            K = int(np.clip(int(K * mult), 8, min(300, n)))

        a = multiplicative_update(best if stuck > 100 else a, eta, K, force_sym)
        sc = local_score(a)

        if np.isfinite(sc) and sc < bestsc:
            best = a.copy()
            bestsc = float(sc)
            stuck = 0
            if eta > 0.12:
                eta = min(eta * 1.015, 0.7)
        else:
            stuck += 1

        if stuck > 80:
            if eta > 0.05:
                eta *= 0.7
            else:
                noise = np.exp(rng.normal(0.0, 0.2, n))
                a = best * noise
                if force_sym:
                    a = 0.5 * (a + a[::-1])
                    a = normalize_to_sum1(a)
                else:
                    a = normalize_to_sum1(a)
                if a is None:
                    a = best.copy()
                eta = 0.3
                stuck = 0

        it += 1

    return best, bestsc


def build_lp_matrix(f, n):
    """Build constraint matrix for LP: A[b] = conv(f,b)[b_index] for relevant indices."""
    # For the LP, we want: sum_i f[k-i] * b[i] <= M for all k
    # The matrix A has rows for each k (convolution position), columns for b[i]
    # A[k, i] = f[k-i] if 0 <= k-i <= n-1, else 0
    # To keep the LP tractable, we select a subset of rows (the "important" ones)
    A = np.zeros((2 * n - 1, n), dtype=np.float64)
    for k in range(2 * n - 1):
        s = max(0, k - (n - 1))
        e = min(k, n - 1)
        if s <= e:
            j = np.arange(s, e + 1, dtype=np.int64)
            A[k, j] = f[k - j]
    return A


def lp_step(f, n, seed, select_top=None):
    """
    LP step: maximize sum(b) subject to conv(f,b)[k] <= M and 0 <= b <= 1.
    Returns the optimized direction g.
    """
    if not HAVE_SCIPY:
        return None

    f = normalize_to_sum1(f)
    if f is None:
        return None

    c = fast_conv_autocorr(f)
    M = float(np.max(c))
    if not np.isfinite(M) or M <= 0.0:
        return None

    rng = np.random.default_rng(seed + 77777)

    # Select top constraints to keep LP tractable
    if select_top is None:
        select_top = min(n, 2 * n - 1)

    # Select the most important constraint rows
    # (those where conv is relatively large, or randomly for diversity)
    if select_top < 2 * n - 1:
        # Mix of top convolution positions and random
        top_frac = max(1, select_top // 2)
        top_indices = np.argpartition(c, -top_frac)[-top_frac:]
        n_rand = select_top - top_frac
        rand_indices = rng.choice(2 * n - 1, size=n_rand, replace=False)
        row_indices = np.concatenate([top_indices, rand_indices])
    else:
        row_indices = np.arange(2 * n - 1)

    A = np.zeros((len(row_indices), n), dtype=np.float64)
    for idx, k in enumerate(row_indices):
        k = int(k)
        s = max(0, k - (n - 1))
        e = min(k, n - 1)
        if s <= e:
            j = np.arange(s, e + 1, dtype=np.int64)
            A[idx, j] = f[k - j]

    cvec = -np.ones(n, dtype=np.float64)
    cvec += 1e-9 * rng.uniform(-1.0, 1.0, n)
    bvec = np.full(len(row_indices), M * (1.0 + 1e-6), dtype=np.float64)
    bounds = [(0.0, 1.0)] * n

    try:
        res = linprog(cvec, A_ub=A, b_ub=bvec, bounds=bounds,
                      method='highs', options={'presolve': True,
                                               'time_limit': 15.0})
    except Exception:
        return None

    if res is None or not res.success:
        return None

    x = np.asarray(res.x, dtype=np.float64)
    x = np.clip(x, 0.0, 1.0)
    if x.size == 0 or float(np.sum(x)) <= 0.0:
        return None

    g = normalize_to_sum1(x)
    return g


def lp_optimize(a, max_time, seed, force_sym=False):
    """Iterative LP-based optimization."""
    f = normalize_to_sum1(a)
    if f is None:
        return None, float('inf')
    if force_sym:
        f = 0.5 * (f + f[::-1])
        f = normalize_to_sum1(f)

    best = f.copy()
    bestsc = local_score(best)
    if max_time <= 0.0 or not HAVE_SCIPY:
        return best, bestsc

    t0 = time.monotonic()
    deadline = t0 + float(max_time)
    n = len(best)

    # Limit LP size
    if n > 2048:
        return best, bestsc

    ts = (0.05, 0.1, 0.15, 0.25, 0.4, 0.6, 0.8, 1.0)
    stall = 0
    it = 0

    while time.monotonic() < deadline:
        rem = deadline - time.monotonic()
        if rem < 5.0:
            break

        # Select top constraints (adaptively)
        select_top = min(n, max(64, n // 2))
        g = lp_step(best, n, seed + it * 1000, select_top)
        if g is None:
            break

        new_best = best
        new_sc = bestsc

        for t in ts:
            new = (1.0 - t) * best + t * g
            if force_sym:
                new = 0.5 * (new + new[::-1])
                new = normalize_to_sum1(new)
            else:
                new = normalize_to_sum1(new)
            if new is None:
                continue
            sc = local_score(new)
            if np.isfinite(sc) and sc < new_sc:
                new_sc = float(sc)
                new_best = new.copy()

        if new_sc < bestsc:
            best = new_best
            bestsc = new_sc
            stall = 0
        else:
            stall += 1
            if stall > 5:
                break

        it += 1

    return best, bestsc


def make_initial(n, seed, kind, parent=None):
    """Generate diverse initial sequences."""
    rng = np.random.default_rng(seed)

    if kind == 0:
        # Uniform random
        val = rng.uniform(0.1, 1.0, n)
    elif kind == 1:
        # Exponential decay
        rate = rng.uniform(0.5, 3.0)
        x = np.arange(n)
        val = np.exp(-rate * x / n) * rng.uniform(0.5, 1.5)
        val = val + 0.1
    elif kind == 2:
        # Block structure
        nb = int(rng.integers(max(2, n // 16), n // 4 + 1))
        sizes = np.full(nb, n // nb)
        sizes[:n % nb] += 1
        blk_vals = rng.uniform(0.1, 1.0, nb)
        val = np.repeat(blk_vals, sizes)
        # Smooth slightly
        val = 0.7 * val + 0.3 * np.convolve(val, np.ones(3)/3, mode='same')
    elif kind == 3:
        # Sinusoidal modulation
        freq = rng.uniform(0.5, 3.0)
        phase = rng.uniform(0, 2 * np.pi)
        x = np.arange(n)
        val = 0.5 + 0.4 * np.sin(2 * np.pi * freq * x / n + phase)
        val = np.clip(val, 0.05, None)
    elif kind == 4:
        # Parent-based with perturbation
        if parent is not None and len(parent) > 0:
            old = np.linspace(0, 1, len(parent))
            new = np.linspace(0, 1, n)
            val = np.interp(new, old, parent)
            val = np.clip(val, 0.01, None)
            val = val * np.exp(rng.normal(0, 0.3, n))
        else:
            val = rng.uniform(0.1, 1.0, n)
    elif kind == 5:
        # Difference-set inspired: high values at structured positions
        # Use a simple "every k-th" pattern with noise
        period = max(2, int(rng.integers(2, max(3, n // 8))))
        val = np.full(n, 0.1)
        for i in range(0, n, period):
            val[i] = rng.uniform(0.5, 1.5)
        val += 0.05 * rng.normal(0, 1, n)
        val = np.clip(val, 0.01, None)
    else:
        val = rng.beta(2, 2, n)

    val = np.asarray(val, dtype=np.float64)
    val = np.clip(val, 0.0, 1000.0)
    return normalize_to_sum1(val)


def optimize_length(n, budget, seed, parent):
    """Full optimization pipeline for a given length."""
    t0 = time.monotonic()
    deadline = t0 + float(budget)
    best = None
    bestsc = float('inf')

    # Try multiple starting points
    n_starts = max(2, min(5, int(budget / 60)))
    starts_per = max(2.0, budget / (n_starts * 3))

    for s in range(n_starts):
        rem = deadline - time.monotonic()
        if rem < 5.0:
            break

        kind = s % 6
        a0 = make_initial(n, seed + s * 31, kind, parent)
        if a0 is None:
            continue

        sc0 = local_score(a0)
        if np.isfinite(sc0) and sc0 < bestsc:
            best = a0.copy()
            bestsc = float(sc0)

        # Phase A: Smoothed gradient descent
        t_rema = deadline - time.monotonic()
        if t_rema > 5.0:
            ga_b = min(starts_per * 0.55, 0.5 * t_rema)
            if ga_b > 2.0:
                c1, s1 = smooth_grad_descent(a0, ga_b, seed + s * 100 + 1)
                if c1 is not None and s1 < bestsc:
                    best = c1.copy()
                    bestsc = float(s1)

        # Phase B: Multiplicative updates
        t_remb = deadline - time.monotonic()
        if t_remb > 3.0 and best is not None:
            gb_b = min(starts_per * 0.3, 0.3 * t_remb)
            if gb_b > 1.0:
                c2, s2 = multiplicative_optimize(best, gb_b, seed + s * 100 + 2)
                if c2 is not None and s2 < bestsc:
                    best = c2.copy()
                    bestsc = float(s2)

        # Phase C: LP step (only for moderate n)
        t_remc = deadline - time.monotonic()
        if t_remc > 10.0 and best is not None and n <= 1024:
            gc_b = min(starts_per * 0.2, 0.2 * t_remc)
            if gc_b > 5.0:
                c3, s3 = lp_optimize(best, gc_b, seed + s * 100 + 3)
                if c3 is not None and s3 < bestsc:
                    best = c3.copy()
                    bestsc = float(s3)

    return best, bestsc


def coordinate_descent_polish(a, max_time, seed):
    """Polish using coordinate descent: try adjusting each element individually."""
    a = normalize_to_sum1(a)
    if a is None:
        return a, float('inf')

    n = len(a)
    best = a.copy()
    bestsc = local_score(best)

    t0 = time.monotonic()
    deadline = t0 + float(max_time)
    rng = np.random.default_rng(seed)

    it = 0
    while time.monotonic() < deadline:
        # Select a random subset of coordinates to try
        nsample = max(10, n // 4)
        indices = rng.choice(n, size=nsample, replace=False)

        improved = False
        for idx in indices:
            if time.monotonic() > deadline:
                break

            orig = best[idx]
            if orig <= 0:
                continue

            # Try a few multipliers
            for mult in (0.5, 0.7, 1.3, 1.5):
                candidate = best.copy()
                candidate[idx] = orig * mult
                candidate = np.clip(candidate, 0.0, None)
                candidate = normalize_to_sum1(candidate)
                if candidate is None:
                    continue
                sc = local_score(candidate)
                if np.isfinite(sc) and sc < bestsc:
                    best = candidate
                    bestsc = float(sc)
                    improved = True
                    break  # move to next coordinate

        it += 1
        if not improved and it % 50 == 0:
            # Try a random perturbation
            candidate = best * np.exp(rng.normal(0, 0.05, n))
            candidate = normalize_to_sum1(candidate)
            if candidate is not None:
                sc = local_score(candidate)
                if np.isfinite(sc) and sc < bestsc:
                    best = candidate
                    bestsc = float(sc)

    return best, bestsc

                print(f"  n={n}: no valid result", flush=True)
        except Exception as exc:
            print(f"  n={n} failed: {exc}", flush=True)

        seedcur += 9973

    # Final intensive polishing phase
    rem = deadline - time.monotonic()
    if best_seq is not None and rem > 10.0:
        print(f"\nFinal polishing phase: rem={rem:.0f}s, "
              f"current best={best_score:.6f}, len={len(best_seq)}", flush=True)

        # 1. More gradient descent on the best
        polish1_budget = min(0.3 * rem, 200.0)
        if polish1_budget > 5.0:
            try:
                c1, s1 = smooth_grad_descent(best_seq, polish1_budget, seedcur + 555,
                                             beta_start=15.0 / np.sqrt(len(best_seq)))
                if c1 is not None and s1 < best_score:
                    best_seq = c1.copy()
                    best_score = float(s1)
                    print(f"  Polish (grad): score={best_score:.6f}", flush=True)
            except Exception as e:
                print(f"  Polish (grad) failed: {e}", flush=True)

        # 2. Multiplicative fine-tuning
        rem2 = deadline - time.monotonic()
        polish2_budget = min(0.25 * rem2, 150.0)
        if polish2_budget > 3.0:
            try:
                c2, s2 = multiplicative_optimize(best_seq, polish2_budget, seedcur + 666)
                if c2 is not None and s2 < best_score:
                    best_seq = c2.copy()
                    best_score = float(s2)
                    print(f"  Polish (mult): score={best_score:.6f}", flush=True)
            except Exception as e:
                print(f"  Polish (mult) failed: {e}

", flush=True)

        # 3. LP-based improvement
        rem3 = deadline - time.monotonic()
        polish3_budget = min(0.2 * rem3, 120.0)
        if polish3_budget > 8.0 and len(best_seq) <= 1024:
            try:
                c3, s3 = lp_optimize(best_seq, polish3_budget, seedcur + 777)
                if c3 is not None and s3 < best_score:
                    best_seq = c3.copy()
                    best_score = float(s3)
                    print(f"  Polish (LP): score={best_score:.6f}", flush=True)
            except Exception as e:
                print(f"  Polish (LP) failed: {e}", flush=True)

        # 4. Coordinate descent
        rem4 = deadline - time.monotonic()
        polish4_budget = min(0.2 * rem4, 100.0)
        if polish4_budget > 3.0:
            try:
                c4, s4 = coordinate_descent_polish(best_seq, polish4_budget, seedcur + 888)
                if c4 is not None and s4 < best_score:
                    best_seq = c4.copy()
                    best_score = float(s4)
                    print(f"  Polish (coord): score={best_score:.6f}", flush=True)
            except Exception as e:
                print(f"  Polish (coord) failed: {e}", flush=True)

        # 5. One more round of multiplicative + gradient
        rem5 = deadline - time.monotonic()
        if rem5 > 5.0:
            b5 = min(rem5 * 0.5, 60.0)
            if b5 > 2.0:
                try:
                    c5, s5 = multiplicative_optimize(best_seq, b5, seedcur + 999)
                    if c5 is not None and s5 < best_score:
                        best_seq = c5.copy()
                        best_score = float(s5)
                        print(f"  Final mult: score={best_score:.6f}", flush=True)
                except Exception:
                    pass

    # Fallback
    if best_seq is None:
        if parent_arr is not None:
            best_seq = normalize_to_sum1(parent_arr)
        if best_seq is None:
            best_seq = np.ones(64, dtype=np.float64) / 64.0
        best_score = local_score(best_seq)

    # Final validation
    final = normalize_to_sum1(best_seq)
    if final is None:
        final = np.ones(64, dtype=np.float64) / 64.0

    try:
        official = float(evaluate_sequence([float(x) for x in final]))
        if np.isfinite(official):
            best_score = official
    except Exception:
        pass

    elapsed = time.monotonic() - start_time
    print(f"\nFINAL: score={best_score:.6f} length={len(final)} time={elapsed:.1f}s", flush=True)

    return [float(x) for x in np.clip(np.asarray(final, dtype=np.float64), 0.0, 1000.0)]
