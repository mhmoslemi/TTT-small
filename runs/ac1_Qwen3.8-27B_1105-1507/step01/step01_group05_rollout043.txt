"""
Minimize 2*n*max(convolve(a,a))/sum(a)^2 using linear ramp initialization
with LP-based proportional fitting and smooth-max gradient refinement.
Key insight: monotone (ramp) sequences give score ~4/3 vs 2 for uniform.

import numpy as np
from scipy.optimize import linprog
import time


def compute_score(a):
    """Compute score = 2*n*max(conv)/sum^2 for normalized sequence."""
    n = len(a)
    conv = np.convolve(a, a)
    return 2.0 * n * conv.max()


def build_conv_matrix(a):
    """Build A such that A @ b = np.convolve(a, b)."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for k in range(2 * n - 1):
        js = max(0, k - n + 1)
        je = min(n, k + 1)
        for j in range(js, je):
            A[k, j] = a[k - j]
    return A


def build_conv_matrix_fast(a):
    """Build A using vectorized indexing for speed."""
    n = len(a)
    m = 2 * n - 1
    rows = []
    cols = []
    vals = []
    for k in range(m):
        js = max(0, k - n + 1)
        je = min(n, k + 1)
        for j in range(js, je):
            rows.append(k)
            cols.append(j)
            vals.append(a[k - j])
    A = np.zeros((m, n))
    A[np.array(rows), np.array(cols)] = np.array(vals)
    return A


def optimize_lp(a, time_budget, t0=0.5, verbose=False):
    """LP-based proportional fitting to reduce convolution peak."""
    n = len(a)
    a = np.array(a, dtype=np.float64)
    a = np.clip(a, 0, None)
    s = a.sum()
    if s < 1e-15:
        return a, np.inf
    a = a / s

    best_a = a.copy()
    best_score = compute_score(a)
    t = t0
    start = time.monotonic()
    it = 0
    stall = 0

    while time.monotonic() - start < time_budget:
        conv = np.convolve(a, a)
        M = conv.max()
        if M < best_score / (2.0 * n) * (1 - 1e-10):
            break

        A = build_conv_matrix(a)
        c_obj = -np.ones(n)
        b_ub = np.full(2 * n - 1, M)
        bounds = [(0, None)] * n

        try:
            res = linprog(c_obj, A_ub=A, b_ub=b_ub, bounds=bounds,
                         method='highs', options={'presolve': True})
        except Exception:
            break

        if not res.success or res.x is None:
            break

        b = np.clip(res.x, 0, None)
        b_sum = b.sum()
        if b_sum < 1e-15:
            break
        g = b / b_sum

        a_new = (1 - t) * a + t * g
        a_new = np.clip(a_new, 0, None)
        sn = a_new.sum()

            stall += 1

        else:

            stall += 1
            if stall > 150:
                break
        it += 1

    return best_a, best_score


def smooth_max_gradient(a, beta):
    """Gradient of log-sum-exp(beta * convolution / sum^2)."""
    n = len(a)
    conv = np.convolve(a, a)
    S = a.sum()
    nc = conv / (S * S)
    bc = beta * nc
    bc -= bc.max()
    w = np.exp(bc)
    w /= w.sum()

    grad = np.zeros(n)
    for i in range(n):
        k_start = i
        k_end = min(2 * n - 1, i + n)
        if k_start >= k_end:
            continue
        grad[i] = 2.0 * np.dot(w[k_start:k_end], a[:k_end - k_start])
    return grad

    """Directly shave peaks by reducing contributing elements."""
    n = len(a)

        peak_val = conv[peak_idx]

        js = max(0, peak_idx - n + 1)
        je = min(n, peak_idx + 1)
        contributions = np.zeros(n)
