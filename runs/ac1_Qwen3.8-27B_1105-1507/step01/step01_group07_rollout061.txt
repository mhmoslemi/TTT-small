    """Solve: min t s.t. (a*g)_k <= t for all k, sum(g)=1, g>=0"""
    n = len(a)
    A = build_A(a)  # A[k,j] such that (a*g)_k = sum_j A[k,j]*g[j]
    m = 2*n - 1

    # Constraints: A * g - t <= 0 (m constraints)
    #              sum(g) = 1
    #              g >= 0, t >= 0 (t >= 0 is implied)
    
    # LP: minimize [0,...,0, 1] dot [g, t]

    c = np.zeros(n + 1)

    A_ub = np.hstack([A, -np.ones((m, 1))])
    b_ub = np.zeros(m)
    
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1
    b_eq = [1.0]
    
    bounds = [(0, None)] * n + [(0, None)]  # g >= 0, t >= 0
    
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq, bounds=bounds, method='highs')
    if res.status == 0:

    return None, None

Improved optimization for minimizing 2*n*max(conv(a,a))/(sum(a))^2.
Uses min-t LP formulation, subgradient descent, and adaptive line search.

import numpy as np
from scipy.optimize import linprog
import time

def compute_obj(a):
    n = len(a)
    if n == 0:
        return np.inf
    conv = np.convolve(a, a)
    max_b = float(np.max(conv))
    sum_a = float(np.sum(a))
    if sum_a < 0.01:
        return np.inf
    return 2.0 * n * max_b / (sum_a * sum_a)

def build_A(a):
    n = len(a)
    A = np.zeros((2 * n - 1, n), dtype=np.float64)
    for k in range(2 * n - 1):
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        if j_start <= j_end:
            A[k, j_start:j_end + 1] = a[k - j_end:k - j_start + 1][::-1]
    return A

def solve_lp_min_t(a):
    """Solve: min t s.t. (a*g)_k <= t for all k, sum(g)=1, g>=0"""
    n = len(a)
    m = 2 * n - 1
    A = build_A(a)
    
    c = np.zeros(n + 1)
    c[-1] = 1.0
    
    A_ub = np.hstack([A, -np.ones((m, 1))])
    b_ub = np.zeros(m)
    
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])
    
    bounds = [(0, None)] * n + [(0, None)]
    
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq, 
                  bounds=bounds, method='highs')
    if res.status == 0:
        return res.x[:n], float(res.x[-1])
    return None, None

def subgradient_step(a, step_size):
    """One step of projected subgradient descent on min max(conv(a,a)) s.t. sum(a)=1, a>=0"""
    n = len(a)
    conv = np.convolve(a, a)
    
    # Find all near-max positions (within tolerance)
    max_val = np.max(conv)
    tol = 1e-10
    active = np.where(conv >= max_val - tol)[0]
    
    # Subgradient: average of gradients at active positions
    # grad_i = 2 * a[k-i] for each active k
    grad = np.zeros(n)
    for k in active:
        for i in range(n):
            j = k - i
            if 0 <= j < n:
                grad[i] += 2.0 * a[j]
    grad /= len(active)
    
    # Update
    a_new = a - step_size * grad
    
    # Project onto simplex {x >= 0, sum(x) = 1}
    a_new = project_simplex(a_new)
    return a_new

def project_simplex(v):
    """Project v onto the probability simplex."""
    n = len(v)
    # Algorithm from Duchi et al. (2008)
    if np.sum(v) <= 0:
        return np.ones(n) / n
    mu = np.sort(v)[::-1]
    cumsum = np.cumsum(mu)
    rho = np.where(mu * np.arange(1, n+1) > (cumsum - 1))[0][-1]
    theta = (cumsum[rho] - 1) / (rho + 1)
    return np.maximum(v - theta, 0)

def run_subgradient(a, max_iter=5000, c=0.1, verbose=False):
    """Subgradient descent with diminishing step size."""
    n = len(a)
    best_a = a.copy()
    best_obj = compute_obj(best_a)
    a = a.copy()

        a = subgradient_step(a, step)
        obj = compute_obj(a)
        if obj < best_obj - 1e-10:
            best_a = a.copy()
            best_obj = obj
    
    return best_a, best_obj

"""
Optimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Improved min-t LP formulation (directly minimizes max convolution)
2. Projected subgradient descent
3. Parametric family initialization
4. Adaptive line search with simulated annealing for local optima escape
5. Random local search for final polish
"""

import numpy as np
from scipy.optimize import linprog
import time


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    conv = np.convolve(a, a)
    max_b = float(np.max(conv))
    sum_a = float(np.sum(a))
    if sum_a < 0.01:
        return np.inf
    return 2.0 * n * max_b / (sum_a * sum_a)


def build_A(a):
    """Build constraint matrix: (a*g)_k = sum_j A[k,j]*g[j]."""
    n = len(a)
    A = np.zeros((2 * n - 1, n), dtype=np.float64)
    for k in range(2 * n - 1):
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        if j_start <= j_end:
            A[k, j_start:j_end + 1] = a[k - j_end:k - j_start + 1][::-1]
    return A


def solve_lp_min_t(a):
    """Solve: min t s.t. (a*g)_k <= t for all k, sum(g)=1, g>=0."""
    n = len(a)
    m = 2 * n - 1
    A = build_A(a)
    c = np.zeros(n + 1)
    c[-1] = 1.0
    A_ub = np.hstack([A, -np.ones((m, 1))])
    b_ub = np.zeros(m)
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])
    bounds = [(0, None)] * n + [(0, None)]
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                  bounds=bounds, method='highs')
    if res.status == 0:
        return res.x[:n], float(res.x[-1])
    return None, None


def solve_lp_max_sum(a):
    """Solve: max sum(g) s.t. (a*g)_k <= M, g>=0 where M=max(conv(a,a))."""
    n = len(a)
    m = 2 * n - 1
    A = build_A(a)
    conv = np.convolve(a, a)
    M = float(np.max(conv))
    c = -np.ones(n)
    b_ub = np.full(m, M)
    bounds = [(0, None)] * n
    res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
    if res.status == 0:
        return res.x, float(-res.fun), M
    return None, None, M


def project_simplex(v):
    """Project v onto the probability simplex {x>=0, sum(x)=1}."""
    n = len(v)
    if n == 0:
        return v
    if np.all(v <= 0):
        return np.ones(n) / n
    mu = np.sort(v)[::-1]
    cumsum = np.cumsum(mu)
    cond = mu * np.arange(1, n + 1) > (cumsum - 1.0)
    rho = np.where(cond)[0]
    if len(rho) == 0:
        return np.ones(n) / n
    r = rho[-1]
    theta = (cumsum[r] - 1.0) / (r + 1.0)
    return np.maximum(v - theta, 0.0)


def subgradient_step(a, step_size):
    """One projected subgradient step on min max(conv(a,a)) s.t. sum(a)=1, a>=0."""
    n = len(a)
    conv = np.convolve(a, a)
    max_val = np.max(conv)
    tol = max(1e-10, 1e-8 * max_val)
    active = np.where(conv >= max_val - tol)[0]

    grad = np.zeros(n)
    for k in active:
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        for i in range(lo, hi + 1):
            j = k - i
            grad[i] += 2.0 * a[j]
    grad /= len(active)

    a_new = a - step_size * grad
    return project_simplex(a_new)


def run_subgradient(a, max_iter=5000, c=0.5):
    """Projected subgradient descent with diminishing step size."""
    n = len(a)
    best_a = a.copy()
    best_obj = compute_obj(best_a)
    a = a.copy()

    for it in range(max_iter):
        step = c / np.sqrt(it + 1)
        a = subgradient_step(a, step)
        obj = compute_obj(a)
        if obj < best_obj:
            best_a = a.copy()
            best_obj = obj

    return best_a, best_obj

def make_u_shaped(n, strength=1.0, noise=0.0, rng=None):

    a = np.maximum(a, 0.01)
    return a / np.sum(a)

        print(f"Phase 1: Optimizing parent n={n_p}", flush=True)
        a_opt, obj_opt = run_lp_optimize(best_seq.copy(), max_iter=400, verbose=True, seed=seed)
        if obj_opt < best_score:
            best_score = obj_opt
            best_seq = a_opt.copy()
            print(f"  Parent optimized: obj={best_score:.6f}", flush=True)

    # Phase 2: Search over multiple n values with diverse initializations
    n_values = [80, 100, 120, 150, 180, 200, 250, 300]

    for n in n_values:
        if time.monotonic() > deadline - 120:
            break
        remaining = deadline - time.monotonic() - 120
        if remaining <= 0:
            break
        print(f"N={n}: {remaining:.0f}s available", flush=True)

        n_restarts = max(1, int(min(3, remaining / 60)))
        for restart in range(n_restarts):
            if time.monotonic() > deadline - 120:
                break

            # Diverse initialization
            if restart == 0:
                a_init = make_u_shaped(n, rng=rng)
            elif restart == 1:
                a_init = parametric_init(n, rng)
            else:
                a_init = parametric_init(n, rng)

            # Run LP optimization
            max_iter = int(min(250, max(50, remaining / (3.0 * (n_restarts - restart)))))
            a_opt, obj_opt = run_lp_optimize(a_init, max_iter=max_iter,
                                             verbose=False, seed=seed + n * 100 + restart)
            if obj_opt < best_score:
                best_score = obj_opt
                best_seq = a_opt.copy()
                print(f"  n={n} r={restart}: obj={best_score:.6f}", flush=True)

    # Phase 3: Subgradient refinement on best
    if best_seq is not None and time.monotonic() < deadline - 60:
        remaining = deadline - time.monotonic() - 60
        n_best = len(best_seq)
        print(f"Phase 3: Subgradient on n={n_best}, budget={remaining:.0f}s", flush=True)
        max_iter = int(min(8000, remaining * 50))
        a_sg, obj_sg = run_subgradient(best_seq.copy(), max_iter=max_iter, c=0.3)
        if obj_sg < best_score:
            best_score = obj_sg
            best_seq = a_sg.copy()
            print(f"  Subgradient: obj={best_score:.6f}", flush=True)

        # One more LP pass on the subgradient result
        if time.monotonic() < deadline - 40:
            a_lp2, obj_lp2 = run_lp_optimize(best_seq.copy(), max_iter=150,
                                             verbose=False, seed=seed + 999)
            if obj_lp2 < best_score:
                best_score = obj_lp2
                best_seq = a_lp2.copy()
                print(f"  LP after SG: obj={best_score:.6f}", flush=True)

    # Phase 4: Try a few more n values if time permits
    if time.monotonic() < deadline - 80:
        remaining = deadline - time.monotonic() - 80
        extra_n = [int(best_seq.size * 0.7), int(best_seq.size * 1.3),
                   int(best_seq.size * 1.5)]
        for n_extra in extra_n:
            if n_extra < 30 or n_extra > 500 or time.monotonic() > deadline - 60:
                continue
            if n_extra == len(best_seq):
                continue
            a_init = parametric_init(n_extra, rng)
            max_iter = int(min(150, remaining / 3))
            a_e, obj_e = run_lp_optimize(a_init, max_iter=max_iter,
                                         verbose=False, seed=seed + n_extra)
            if obj_e < best_score:
                best_score = obj_e
                best_seq = a_e.copy()
                print(f"  Extra n={n_extra}: obj={best_score:.6f}", flush=True)

    # Phase 5: Local search polish
    if best_seq is not None and time.monotonic() < deadline - 5:
        remaining = deadline - time.monotonic() - 3
        max_iter = int(min(50000, remaining * 30))
        print(f"Phase 5: Local search, {max_iter} iters, {remaining:.0f}s", flush=True)
        a_ls, obj_ls = random_local_search(best_seq.copy(), rng, max_iter=max_iter)
        if obj_ls < best_score:
            best_score = obj_ls
            best_seq = a_ls.copy()
            print(f"  Final local search: obj={best_score:.6f}", flush=True)

    # Final LP polish if we have time
    if best_seq is not None and time.monotonic() < deadline - 10:
        remaining = deadline - time.monotonic() - 5
        if remaining > 30:
            max_iter = int(min(200, remaining))
            a_final, obj_final = run_lp_optimize(best_seq.copy(), max_iter=max_iter,
                                                 verbose=False, seed=seed + 7777)
            if obj_final < best_score:
                best_score = obj_final
                best_seq = a_final.copy()
                print(f"  Final LP: obj={best_score:.6f}", flush=True)

    # Fallback
    if best_seq is None:
        best_seq = np.ones(100)
        best_score = compute_obj(best_seq)

    elapsed = time.monotonic() - start_time
    print(f"DONE: n={len(best_seq)}, score={best_score:.6f}, time={elapsed:.1f}s", flush=True)
    return [float(x) for x in best_seq]
