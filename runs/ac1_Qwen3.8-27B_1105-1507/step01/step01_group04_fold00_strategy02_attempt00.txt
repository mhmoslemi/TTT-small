def _fft_convolve(seq):
    """Full linear convolution using FFT."""
    n = len(seq)
    size = 1 if n == 0 else 2**next_power_of_two(n)
    seq_arr = np.asarray(seq, dtype=float)
    padded = np.pad(seq_arr, (0, size - n), mode='constant', constant_values=0)
    F = np.fft.rfft(padded)
    H = np.fft.irfft(F * np.conj(F))
