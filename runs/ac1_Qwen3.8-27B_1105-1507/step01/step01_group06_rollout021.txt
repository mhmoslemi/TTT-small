```python
    """Full self-convolution via FFT."""
    n = len(a)

    return np.fft.irfft(fa * fa, size)[:2*n-1]
```

The key insight is:
- We want to minimize J(a) = 2n * max(a*a) / (sum(a))^2
- This is scale-invariant, so we can fix sum(a) = 1
- Then we just minimize max(a*a)
- The subgradient of max(a*a) at point a is (1/|K|) * sum_{k in K} 2 * P(a, k) where K = argmax and P(a,k)_i = a_{k-i} (0 if out of bounds)

For the optimization, I'll use:
1. **Projected subgradient with Polyak step size** (if we have an upper bound) or **decreasing step size**
2. **Nesterov acceleration** for the smooth parts
3. **L-BFGS-B** for final polishing (treating the max as a smooth function at the current argmax)

Actually, the most practical approach for getting good results in limited time is:

1. Use L-BFGS-B with an analytical gradient for the "active" max
2. At each iteration, identify the active max index and compute the gradient assuming only that index is active
3. This gives a smooth function that L-BFGS can handle
4. The result will be a stationary point of the objective (where the gradient w.r.t. the active constraint is zero, projected to the feasible set)

Let me implement this:

```python
def lbfgs_smooth_objective(x, k_active):
    """
    Objective assuming k_active is the argmax of a*a.
    J = 2n * (a*a)_{k_active} / (sum(a))^2
    """
    n = len(x)
    a = np.maximum(x, 0)
    S = a.sum()
    if S < 1e-10:
        return 1e10, None
    
    # (a*a)_{k} = sum_i a_i * a_{k-i} (0 if out of bounds)
    i_vals = np.arange(n)
    j_vals = k_active - i_vals
    valid = (j_vals >= 0) & (j_vals < n)
    conv_k = np.sum(a[valid] * a[j_vals[valid]])
    
    # J = 2n * conv_k / S^2
    J = 2.0 * n * conv_k / (S * S)
    
    # Gradient: dJ/da_i = 2n * [S * d(conv_k)/da_i - 2*conv_k*dS/da_i] / S^3
    # d(conv_k)/da_i = 2*a_{k-i} (0 if out of bounds), with factor 1 if k-i == i
    # Actually: d(conv_k)/da_i = 2*a_{k-i} for all i where 0 <= k-i < n
    # (the factor of 2 accounts for both a_i*a_{k-i} and a_{k-i}*a_i)
    # Wait, when k-i == i (i.e., k = 2i), we have a_i^2, so derivative is 2*a_i = 2*a_{k-i}. Same formula.
    
    dconv = np.zeros(n)
    dconv[valid] = 2.0 * a[j_vals[valid]]
    
    dS = np.ones(n)
    grad = 2.0 * n * (S * dconv - 2.0 * conv_k * dS) / (S**3)
    
    return J, grad
```

But the problem is that k_active might change during optimization. We need to handle this.

**Strategy: Outer loop over k_active, inner L-BFGS**

```
for each candidate k:
    Run L-BFGS minimizing J_k(a) = 2n * (a*a)_k / S^2
    Check if the result actually has argmax at k
    If yes, we have a local minimum of the full objective
```

But this requires knowing which k to try. In practice, for a good solution, the argmax is likely at k = n-1 (for symmetric sequences) or near the center.

Let me think about a hybrid approach:

1. Start with the parent sequence
2. Compute b = a*a, find k* = argmax(b)
3. Run L-BFGS minimizing J_{k*}(a) = 2n * (a*a)_{k*} / S^2 with analytic gradient
4. At the end, check if argmax is still k*. If not, go to the new argmax and repeat.
5. This "active set" approach should converge to a good local minimum.

The advantage is that each L-BFGS call optimizes a smooth function (for fixed k), so convergence is fast.

Let me also add the LP-based approach as an alternative:

**LP approach (improved)**:
Instead of solving the full LP (which is slow), use the dual:
The primal is: max 1^T g s.t. (a*g)_k <= M for all k, g >= 0
The dual is: min M * sum_k y_k s.t. sum_k y_k * a_{k-i} >= 1 for all i, y_k >= 0

The dual has n constraints and 2n-1 variables. For large n, this is still large, but we can use the fact that at the optimum, most y_k = 0 (only the "active" constraints matter).

Actually, a simpler approximation: use **multiplicative weights**:
- Initialize g = a (normalized)
- Repeat:
  - Find k* = argmax (a*g)_k
  - Update: g_i *= exp(-eta * a_{k*-i} / M(a*g))
  - Normalize g

This is the "follow the regularized leader" or "exponential weights" approach, which converges to the LP solution.

```python
"""
Minimizes J(a) = 2*n*max(convolve(a,a))/(sum(a))^2 using:
1. Active-set L-BFGS with analytic gradients (key innovation)
2. Nesterov-accelerated projected subgradient method
3. LP-based mirror descent (multiplicative weights approximation)
4. Multi-length, multi-initialization search with adaptive time allocation
"""

import numpy as np
import time
from scipy.optimize import minimize

# --- Top-level helper functions ---

def fast_conv(a):
    """Full self-convolution via FFT, O(n log n)."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2*n-1]

def conv_at_k(a, k):
    """Compute (a*a)_k efficiently."""
    n = len(a)
    i_vals = np.arange(n)
    j_vals = k - i_vals
    valid = (j_vals >= 0) & (j_vals < n)
    return np.sum(a[valid] * a[j_vals[valid]])

def grad_conv_at_k(a, k):
    """Compute d(a*a)_k / da. Shape (n,)."""
    n = len(a)
    grad = np.zeros(n)
    i_vals = np.arange(n)
    j_vals = k - i_vals
    valid = (j_vals >= 0) & (j_vals < n)
    grad[valid] = 2.0 * a[j_vals[valid]]
    return grad

def objective_k(x, k):
    """
    Objective J_k(a) = 2n * (a*a)_k / (sum(a))^2 and its gradient.
    x is the raw input (will be clipped to non-negative).
    """
    n = len(x)
    a = np.maximum(x, 0.0)
    S = a.sum()
    if S < 1e-10:
        return 1e10, np.zeros(n)
    
    ck = conv_at_k(a, k)
    J = 2.0 * n * ck / (S * S)
    
    dck = grad_conv_at_k(a, k)
    grad = 2.0 * n * (S * dck - 2.0 * ck) / (S ** 3)
    
    return J, grad

def full_score(a):
    """Compute the full J(a)."""
    n = len(a)
    S = a.sum()
    if S < 0.01:
        return float('inf')
    a = np.clip(a, 0, 1000)
    b = fast_conv(a)
    return float(2.0 * n * b.max() / (S * S))

def active_set_lbfgs(a0, max_rounds=20, deadline=None):
    """
    Active-set method: repeatedly identify the argmax k, then run L-BFGS
    on the smooth function J_k until the argmax stabilizes.
    """
    n = len(a0)
    a = np.clip(a0.astype(np.float64), 0.0, None)
    S = a.sum()
    if S < 1e-10:
        return a0
    a = a / S
    
    best_a = a.copy()
    best_score = full_score(a)
    
    prev_k = -1
    for rnd in range(max_rounds):
        if deadline is not None and time.monotonic() > deadline:
            break
        
        b = fast_conv(a)
        k = int(np.argmax(b))
        
        if k == prev_k and rnd > 2:
            # Active set stabilized, do final polish
            break
        
        prev_k = k
        
        # Run L-BFGS on J_k
        def obj_grad(x):
            val, g = objective_k(x, k)
            return val, g
        
        res = minimize(obj_grad, a, jac=True, method='L-BFGS-B',
                       bounds=[(0, None)] * n,
                       options={'maxiter': 200, 'max_fun': 5000,
                                'ftol': 1e-16, 'gtol': 1e-12})
        
        a_new = np.clip(res.x, 0, None)
        S_new = a_new.sum()
        if S_new < 1e-10:
            break
        a_new = a_new / S_new
        
        score = full_score(a_new)
        if score < best_score:
            best_score = score
            best_a = a_new.copy()
        
        a = a_new
    
    return best_a

def nesterov_subgradient(a0, max_iters=3000, deadline=None):
    """
    Nesterov-accelerated projected subgradient for minimizing max(a*a)
    subject to sum(a)=1, a>=0.
    """
    n = len(a0)
    a = np.clip(a0.astype(np.float64), 0.0, None)
    S = a.sum()
    if S < 1e-10:
        return a0
    a = a / S
    
    best_a = a.copy()
    best_M = fast_conv(a).max()
    
    # Nesterov parameters
    gamma = 2.0  # related to smoothness
    y = a.copy()
    v = np.zeros(n)
    
    for t in range(max_iters):
        if deadline is not None and time.monotonic() > deadline:
            break
        
        # Nesterov step size
        t1 = t + 1
        lr = 1.0 / (np.sqrt(gamma) * t1)
        
        # Momentum
        beta = (1.0 - 1.0 / (t1 + 1.0))
        
        # Compute subgradient at y
        y = np.clip(y, 0, None)
        Sy = y.sum()
        if Sy < 1e-10:
            break
        y = y / Sy
        
        b = fast_conv(y)
        M = b.max()
        k_indices = np.where(b >= M - 1e-10)[0]
        
        # Subgradient of max
        subgrad = np.zeros(n)
        for k in k_indices:
            subgrad += grad_conv_at_k(y, k)
        subgrad /= len(k_indices)
        
        # Project gradient (maintain sum=1)
        pg = subgrad - subgrad.mean()
        
        # Update
        y = a - lr * pg
        v = beta * v - lr * pg
        a_new = a + v
        a_new = np.clip(a_new, 0, None)
        Sn = a_new.sum()
        if Sn < 1e-10:
            break
        a_new = a_new / Sn
        
        M_new = fast_conv(a_new).max()
        if M_new < best_M:
            best_M = M_new
            best_a = a_new.copy()
        
        a = a_new
    
    return best_a

def multiplicative_weights(a0, max_iters=500, eta_init=0.1, deadline=None):

        # (a*g)_k = sum_i a_i * g_{k-i}
        
        M = ag.max()
        k_star = int(np.argmax(ag))

        update = np.zeros(n)

        if score < best_score:
            best_score = score
            best_g = g_new.copy()
        
        g = g_new
        eta = max(0.01, eta_init / (1 + t * 0.01))
    
    return best_g

def cross_conv(a, b):
    """Cross-convolution a * b via FFT."""
    n = max(len(a), len(b))
    size = 1
    while size < len(a) + len(b) - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    return np.fft.irfft(fa * fb, size)[:len(a)+len(b)-1]

    # Clamp to positive
```

For each coordinate i, given all other coordinates fixed, the optimal a_i can be found in closed form (or by 1D search).

Given a_{-i} fixed, let S' = sum_{j != i} a_j.
(a*a)_k as a function of a_i:
- (a*a)_k = b_k(0) + 2*a_i*a_{k-i} (for k where 0 <= k-i < n)
- where b_k(0) is the convolution value with a_i = 0

So M(a_i) = max_k [b_k(0) + 2*a_i*c_k] where c_k = a_{k-i} (0 if k-i out of range)

J(a_i) = 2n * M(a_i) / (S' + a_i)^2

To minimize this, for each "piece" (where a specific k is the argmax):
J_k(a_i) = 2n * (b_k(0) + 2*a_i*c_k) / (S' + a_i)^2

dJ_k/da_i = 2n * [2*c_k*(S'+a_i)^2 - (b_k(0)+2*a_i*c_k)*2*(S'+a_i)] / (S'+a_i)^4
          = 2n * [2*c_k*(S'+a_i) - 2*(b_k(0)+2*a_i*c_k)] / (S'+a_i)^3
          = 4n * [c_k*S' + c_k*a_i - b_k(0) - 2*a_i*c_k] / (S'+a_i)^3
          = 4n * [c_k*S' - b_k(0) - a_i*c_k] / (S'+a_i)^3

Setting to 0: c_k*S' - b_k(0) - a_i*c_k = 0
=> a_i = S' - b_k(0)/c_k (if c_k > 0)

So the optimal a_i for a given active constraint k is: a_i* = S' - b_k(0)/c_k

We need a_i* >= 0 and k to be the argmax at a_i*. This gives us a way to do coordinate descent.

```python
Minimizes J(a) = 2*n*max(convolve(a,a))/(sum(a))^2 using:
1. Active-set L-BFGS with analytic gradients (primary method)
2. Nesterov-accelerated projected subgradient for diversification

Key innovation: explicit O(n) subgradient enables fast active-set optimization
where each L-BFGS call handles a smooth sub-problem.

import numpy as np
import time
from scipy.optimize import minimize

# Global state for passing k to L-BFGS callback
_current_k = 0
_current_n = 0


def fast_conv(a):
    """Full self-convolution via FFT, O(n log n)."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]


def cross_conv(a, b):
    """Cross-convolution a * b via FFT."""
    size = 1
    while size < len(a) + len(b) - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    return np.fft.irfft(fa * fb, size)[:len(a) + len(b) - 1]


def conv_at_k(a, k):
    """Compute (a*a)_k in O(n)."""
    n = len(a)
    i_vals = np.arange(n)
    j_vals = k - i_vals
    valid = (j_vals >= 0) & (j_vals < n)
    return float(np.sum(a[valid] * a[j_vals[valid]]))


def grad_conv_at_k(a, k):
    """Compute d(a*a)_k / da vector in O(n)."""
    n = len(a)
    grad = np.zeros(n)
    i_vals = np.arange(n)
    j_vals = k - i_vals
    valid = (j_vals >= 0) & (j_vals < n)
    grad[valid] = 2.0 * a[j_vals[valid]]
    return grad


def objective_k(x):
    """
    Objective J_k(a) = 2n * (a*a)_k / (sum(a))^2 using global _current_k.
    Returns (value, gradient) for L-BFGS.
    """
    k = _current_k
    n = _current_n
    a = np.maximum(x, 0.0)
    S = a.sum()
    if S < 1e-10:
        return 1e10, np.zeros(n)

    ck = conv_at_k(a, k)
    J = 2.0 * n * ck / (S * S)

    dck = grad_conv_at_k(a, k)
    grad = 2.0 * n * (S * dck - 2.0 * ck) / (S ** 3)

    return J, grad


def full_score(a):
    """Compute J(a) = 2n * max(a*a) / (sum(a))^2."""
    n = len(a)
    a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
    S = a.sum()
    if S < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2.0 * n * b.max() / (S * S))


def active_set_lbfgs(a0, max_rounds=25, deadline=None):
    """
    Active-set L-BFGS: identify argmax k, optimize J_k, repeat until stable.
    """
    global _current_k, _current_n
    n = len(a0)
    _current_n = n
    a = np.clip(np.asarray(a0, dtype=np.float64), 0.0, None)
    S = a.sum()
    if S < 1e-10:
        return np.asarray(a0, dtype=np.float64)
    a = a / S

    best_a = a.copy()
    best_score = full_score(a)
    prev_k = -1
    consecutive_same = 0

    for rnd in range(max_rounds):
        if deadline is not None and time.monotonic() > deadline:
            break

        b = fast_conv(a)
        k = int(np.argmax(b))
        _current_k = k

        if k == prev_k:
            consecutive_same += 1
        else:
            consecutive_same = 0
        prev_k = k

        # Run L-BFGS on smooth sub-problem
        maxiter = 300 if rnd < 5 else 150
        res = minimize(objective_k, a, jac=True, method='L-BFGS-B',
                       bounds=[(0.0, None)] * n,
                       options={'maxiter': maxiter, 'max_fun': 8000,
                                'ftol': 1e-16, 'gtol': 1e-12})

        a_new = np.clip(res.x, 0.0, None)
        S_new = a_new.sum()
        if S_new < 1e-10:
            break
        a_new = a_new / S_new

        sc = full_score(a_new)
        if sc < best_score:
            best_score = sc
            best_a = a_new.copy()

        a = a_new

        if consecutive_same >= 3:
            break

    return best_a


def nesterov_subgradient(a0, max_iters=5000, deadline=None):
    """
    Nesterov-accelerated projected subgradient for min max(a*a) s.t. sum(a)=1, a>=0.
    """
    n = len(a0)
    a = np.clip(np.asarray(a0, dtype=np.float64), 0.0, None)
    S = a.sum()
    if S < 1e-10:
        return np.asarray(a0, dtype=np.float64)
    a = a / S

    best_a = a.copy()
    best_M = float(fast_conv(a).max())

    y = a.copy()
    v = np.zeros(n)
    mu = 1.0  # smoothness parameter (upper bound on ||grad||)

    for t in range(max_iters):
        if deadline is not None and time.monotonic() > deadline:
            break

        t1 = t + 1
        lr = 1.0 / (np.sqrt(mu) * t1)
        beta = t / (t1 + 1.0)

        # Evaluate at y (Nesterov lookahead point)
        y = np.clip(y, 0.0, None)
        Sy = y.sum()
        if Sy < 1e-10:
            break
        y = y / Sy

        b = fast_conv(y)
        M = float(b.max())
        tol = 1e-12 * max(1.0, M)
        k_indices = np.where(b >= M - tol)[0]

        # Subgradient of max
        subgrad = np.zeros(n)
        for k in k_indices:
            subgrad += grad_conv_at_k(y, k)
        subgrad /= len(k_indices)

        # Project (maintain simplex)
        pg = subgrad - subgrad.mean()

        # Nesterov update

        a_new = np.clip(a_new, 0.0, None)
        Sn = a_new.sum()
        if Sn < 1e-10:

        M_new = float(fast_conv(a_new).max())
        if M_new < best_M:
            best_M = M_new
            best_a = a_new.copy()

        a = a_new
        y = a + (t1 / (t1 + 1.0)) * (a_new - a)  # Nesterov extrapolation
        y = np.clip(y, 0.0, None)
        y = y / max(y.sum(), 1e-15)

def multiplicative_weights(a0, max_iters=200, eta_init=0.5, deadline=None):
    """
    Exponential weights / multiplicative weights for approximate LP best-response.
    Updates g to reduce the peak of a*g.
    """
    n = len(a0)
    a = np.clip(np.asarray(a0, dtype=np.float64), 1e-8, None)
    a = a / a.sum()
    g = a.copy()

    best_g = g.copy()
    best_score = full_score(g)

    for t in range(max_iters):
        if deadline is not None and time.monotonic() > deadline:
            break

        ag = cross_conv(a, g)
        M = float(ag.max())
        if M < 1e-15:
            break
        k_star = int(np.argmax(ag))

        eta = eta_init / (1.0 + 0.05 * t)

        # Multiplicative update
        i_vals = np.arange(n)
        j_vals = k_star - i_vals
        valid = (j_vals >= 0) & (j_vals < n)
        log_update = np.zeros(n)
        log_update[valid] = -eta * a[j_vals[valid]] / M
        g = g * np.exp(log_update)
        g = np.clip(g, 1e-15, None)
        g = g / g.sum()

        # Convex combination with a
        mix = 0.8 + 0.1 * (t / max_iters)
        g_new = mix * g + (1.0 - mix) * a
        g_new = np.clip(g_new, 0.0, None)
        g_new = g_new / max(g_new.sum(), 1e-15)

        sc = full_score(g_new)
        if sc < best_score:
            best_score = sc
            best_g = g_new.copy()

        g = g_new

    return best_g


def generate_initializations(n, rng, parent=None):
    """Generate diverse initializations of length n, all normalized."""
    inits = []
    t = np.linspace(0, 1, n, endpoint=False)

    # Uniform
    inits.append(np.ones(n))

    # Cosine patterns with various frequencies and amplitudes
    for freq in [1, 2, 3, 4, 6, 8, 12]:
        for amp in [0.2, 0.4, 0.6]:
            inits.append(1.0 + amp * np.cos(2.0 * np.pi * freq * t))

    # Chirp patterns
    for bandwidth in [2, 4, 8]:
        inits.append(1.0 + 0.3 * np.cos(np.pi * bandwidth * t ** 2))

    # Triangular
    inits.append(1.0 - 1.5 * np.abs(t - 0.5))

    # Center-weighted (Gaussian bumps)
    center = n // 2
    idx = np.arange(n)
    inits.append(np.exp(-((idx - center) ** 2) / (2 * (n * 0.2) ** 2)) + 0.1)

    # Edge-weighted
    inits.append(np.exp(-((np.arange(n) - n * 0.2) ** 2) / (2 * (n * 0.15) ** 2)) +
                 np.exp(-((np.arange(n) - n * 0.8) ** 2) / (2 * (n * 0.15) ** 2)) + 0.1)

    # Random
    inits.append(rng.uniform(0.1, 1.0, n))

    # Two-level random
    mask = rng.random(n) > 0.5
    inits.append(np.where(mask, 1.0, 0.3))

    # Power law
    inits.append(1.0 + 0.3 * (t + 0.01) ** 0.5)
    inits.append(1.0 + 0.3 * (1.0 - t + 0.01) ** 0.7)

    # Sine with phase
    inits.append(1.0 + 0.3 * np.sin(2.0 * np.pi * 3 * t + 0.5))

    # Polynomial
    inits.append(1.0 + 0.2 * (2 * t - 1) ** 2)
    inits.append(1.0 + 0.2 * (2 * t - 1) ** 4)

    # Parent resampled
    if parent is not None and len(parent) > 0:
        t_old = np.linspace(0, 1, len(parent))
        t_new = np.linspace(0, 1, n)
        inits.append(np.maximum(np.interp(t_new, t_old, parent), 0.01))

    # Clamp and normalize
    result = []
    for x in inits:
        x = np.clip(x, 0.01, None)
        x = x / x.sum()
        result.append(x)
    return result
```
