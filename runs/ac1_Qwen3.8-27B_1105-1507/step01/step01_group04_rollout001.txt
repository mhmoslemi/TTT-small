import numpy as np
import time
from scipy.optimize import linprog
from scipy.sparse import coo_matrix

def _fft_convolve_full(f):
    n = len(f)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    F = np.fft.rfft(f, size)
    h = np.fft.irfft(F * F, size)
    return h[:2 * n - 1]

def _score(f):
    n = len(f)
    S = np.sum(f)
    if S < 0.01:
        return float('inf')
    h = _fft_convolve_full(f)
    M = np.max(h)
    return 2.0 * n * M / (S * S)

def _normalize(f):
    f = np.maximum(f, 0.0)
    s = np.sum(f)
    if s < 1e-30:
        return None
    return f / s

def _lp_frank_wolfe(f_init, max_iters, time_budget, verbose=True):
    f = np.asarray(f_init, dtype=np.float64).copy()
    f = _normalize(f)
    n = len(f)
    best_f = f.copy()
    best_score = _score(f)
    start = time.monotonic()
    
    if verbose:
        print(f"  FW start: n={n}, score={best_score:.8f}")
    
    for it in range(max_iters):
        if time.monotonic() - start > time_budget:
            break
        
        h = _fft_convolve_full(f)
        M = float(np.max(h))
        
        # Build constraint matrix
        # A[k, j] = f[k-j] if 0 <= k-j < n
        row_idx = []
        col_idx = []
        vals = []
        for j in range(n):
            # k ranges from j to j+n-1, clipped to [0, 2n-2]
            k_start = j
            k_end = min(2*n-2, j+n-1)
            # For k in [k_start, k_end], A[k,j] = f[k-j]
            ks = np.arange(k_start, k_end+1)
            row_idx.extend(ks.tolist())
            col_idx.extend([j] * (k_end - k_start + 1))
            vals.extend(f[0:k_end-k_start+1].tolist())

"""
Optimizes non-negative step function to minimize 2n*max(conv(f,f))/(sum(f))^2.
Uses LP-based Frank-Wolfe with backtracking line search, mirror descent warmup,
and coordinate descent polishing.
"""

import numpy as np
import time
from scipy.optimize import linprog
from scipy.sparse import coo_matrix

def _fft_conv(f):
    """Full linear convolution via FFT."""
    n = len(f)
    size = 1
    while size < 2*n - 1:
        size <<= 1
    F = np.fft.rfft(f, size)
    return np.fft.irfft(F * F, size)[:2*n-1]

def _score(f):
    """Compute the evaluation score."""
    n = len(f)
    S = np.sum(f)
    if S < 0.01:
        return float('inf')
    h = _fft_conv(f)
    return 2.0 * n * np.max(h) / (S * S)

def _normalize(f):
    f = np.maximum(f, 0.0)
    s = f.sum()
    if s < 1e-30:
        return None
    return f / s

def _build_conv_matrix(f):
    """Build the convolution constraint matrix A where A[k,j] = f[k-j]."""
    n = len(f)
    row_idx = np.zeros(n*n, dtype=np.int32)
    col_idx = np.zeros(n*n, dtype=np.int32)
    vals = np.zeros(n*n, dtype=np.float64)
    idx = 0
    for j in range(n):
        k_start = j
        k_end = min(j + n - 1, 2*n - 2)
        count = k_end - k_start + 1
        row_idx[idx:idx+count] = np.arange(k_start, k_end + 1)
        col_idx[idx:idx+count] = j
        vals[idx:idx+count] = f[:count]
        idx += count
    A = coo_matrix((vals[:idx], (row_idx[:idx], col_idx[:idx])), shape=(2*n-1, n))
    return A.tocsc()

def _fw_lp(f, M):
    """Solve the Frank-Wolfe LP subproblem. Returns optimal g (normalized)."""
    n = len(f)
    A = _build_conv_matrix(f)
    c_obj = -np.ones(n)
    b_ub = np.full(2*n-1, M)
    res = linprog(c_obj, A_ub=A, b_ub=b_ub,
                  bounds=[(0, None)]*n, method='highs',
                  options={'presolve': True, 'dual_feasibility_tolerance': 1e-9})
    if not res.success or res.x is None:
        return None
    g = np.maximum(res.x, 0)
    s = g.sum()
    if s < 1e-10:
        return None
    return g / s

def _frank_wolfe(f_init, time_budget, max_iters=1000, verbose=True):
    """LP-based Frank-Wolfe with backtracking line search."""
    f = _normalize(np.asarray(f_init, dtype=np.float64))
    if f is None:
        return np.asarray(f_init, dtype=np.float64).copy(), float('inf')
    n = len(f)
    best_f = f.copy()
    best_score = _score(f)
    start = time.monotonic()
    
    if verbose:
        print(f"    FW: n={n}, initial score={best_score:.8f}")
    
    no_improve = 0
    for it in range(max_iters):
        if time.monotonic() - start > time_budget:
            break
        
        h = _fft_conv(f)
        M = float(np.max(h))
        
        g = _fw_lp(f, M)
        if g is None:
            break
        
        # Backtracking line search
        best_t = 0.1  # default
        best_s = _score((1-0.1)*f + 0.1*g)
        for t_try in [1.0, 0.5, 0.25, 0.1, 0.5, 0.75, 0.33]:
            f_try = (1 - t_try) * f + t_try * g
            f_try = _normalize(f_try)
            if f_try is None:
                continue
            s_try = _score(f_try)
            if s_try < best_s:
                best_s = s_try
                best_t = t_try
        
        f = (1 - best_t) * f + best_t * g
        f = _normalize(f)
        if f is None:
            break
        
        s = _score(f)
        if s < best_score - 1e-10:
            best_score = s
            best_f = f.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        if verbose and (it+1) % 20 == 0:
            print(f"    FW it={it+1}: score={best_score:.8f}, t={best_t:.3f}, no_improve={no_improve}")
        
        if no_improve > 50:
            # Try a restart with perturbation
            f_pert = f + 0.01 * np.random.random(n)
            f_pert = _normalize(f_pert)
            if f_pert is not None and _score(f_pert) < best_score + 0.01:
                f = f_pert
                no_improve = 0
    
    if verbose:
        print(f"    FW done: best={best_score:.8f}")
    return best_f, best_score

def _mirror_descent(f_init, n_iter, eta_scale, threshold_frac, time_budget):
    """Mirror descent with multiplicative updates."""
    f = _normalize(np.asarray(f_init, dtype=np.float64))
    if f is None:
        return np.asarray(f_init, dtype=np.float64).copy(), float('inf')
    n = len(f)
    best_f = f.copy()
    best_score = _score(f)
    start = time.monotonic()
    
    for it in range(n_iter):
        if time.monotonic() - start > time_budget:
            break
        
        h = _fft_conv(f)
        M = float(np.max(h))
        if M <= 0:
            break
        
        threshold = threshold_frac * M
        active = np.where(h >= threshold)[0]
        if len(active) == 0:
            active = np.array([np.argmax(h)])
        
        # Pressure vector
        p = np.zeros(n)
        for k in active:
            j_lo = max(0, k - n + 1)
            j_hi = min(n - 1, k)
            for j in range(j_lo, j_hi + 1):
                p[j] += f[k - j]
        p /= len(active)
        
        eta = eta_scale / M
        f = f * np.exp(-np.clip(eta * p, -20, 20))
        f_new = _normalize(f)
        f = f_new if f_new is not None else np.ones(n)/n
        
        s = _score(f)
        if s < best_score:
            best_score = s
            best_f = f.copy()
    
    return best_f, best_score

def _coord_descent(f_init, time_budget):
    """Coordinate descent polish."""
    f = _normalize(np.asarray(f_init, dtype=np.float64).copy())
    if f is None:

        for j in order:
            if time.monotonic() - start > time_budget:
                break
            orig = f[j]
            best_val = orig
            best_s = best_score  # Use current best as reference
            
            for mult in [0.0, 0.5, 1.5, 0.25, 2.0, 0.75, 1.25, 0.33, 3.0]:
                f[j] = orig * mult

                if f_try is None:

                s_try = _score(f_try)
                if s_try < best_s:
                    best_s = s_try
                    best_val = orig * mult
