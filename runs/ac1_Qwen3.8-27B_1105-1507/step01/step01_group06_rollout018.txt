"""
Minimizes J(a) = 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Smooth mirror descent (log-sum-exp gradient + multiplicative weight updates)
2. LP best-response for structured moves
3. Multiple initializations and sequence lengths

import numpy as np
from scipy.optimize import linprog
from scipy.special import expit
import time

def fast_conv(a):
    """Full self-convolution via FFT."""
    n = len(a)
    s = 1
    while s < 2 * n - 1:
        s <<= 1
    fa = np.fft.rfft(a, s)
    return np.fft.irfft(fa * fa, s)[:2 * n - 1]

def compute_gradient(a, w):
    """Compute grad_i = 2 * sum_k w_k * a[k-i] for i=0,...,n-1.
    
    w has length 2n-1, a has length n.
    """
    n = len(a)
    # h_i = sum_m w[m+i] * a[m] for i=0,...,n-1
    # This is the cross-correlation of w and a
    # Using np.correlate: correlate(w, a, 'full')[n-1+i] = sum_m w[m+i]*a[m]
    corr = np.correlate(w, a, mode='full')
    h = corr[n-1:n-1+n]
    return 2.0 * h

def softmax_shifted(x, tau):
    """Compute softmax(tau * x) with shift for numerical stability."""
    shifted = tau * (x - x.max())
    exp_shifted = np.exp(shifted)
    return exp_shifted / exp_shifted.sum()

def build_A(a):

def fast_conv(a):
    n = len(a)
    s = 1
    while s < 2 * n - 1:
        s <<= 1
    fa = np.fft.rfft(a, s)
    return np.fft.irfft(fa * fa, s)[:2 * n - 1]

    n = len(a)

    A = np.fft.rfft(a, N)
    C = np.fft.irfft(W * np.conj(A), N)
