"""
Optimizes non-negative step function to minimize 2n*max(conv(f,f))/(sum(f))^2.
Uses adaptive mirror descent with momentum, followed by analytic coordinate descent polish.
Explores multiple sequence lengths and initialization families.
"""

import numpy as np
import time

def _fft_convolve(f):
    """Full linear convolution of f with itself via FFT."""
    n = len(f)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    F = np.fft.rfft(f, size)
    out = np.fft.irfft(F * F, size)
    return out[:2 * n - 1]

def _score(f):
    """Compute 2n*max(conv(f,f))/(sum(f))^2. Assumes f >= 0, sum(f) > 0."""
    n = len(f)
    S = float(np.sum(f))
    if S < 0.01:
        return float('inf')
    R = _fft_convolve(f)
    M = float(np.max(R))
    return 2.0 * n * M / (S * S)

def _normalize(f):
    f = np.maximum(f, 0.0)
    s = float(np.sum(f))
    if s < 1e-30:
        return None
    return f / s

def _init_cosine(n, p=2.0):
    j = np.arange(n, dtype=np.float64)
    f = np.cos(np.pi * (j + 0.5) / n) ** p
    f = np.maximum(f, 0.0)
    return _normalize(f)

def _init_gaussian(n, sigma=0.3):
    j = np.arange(n, dtype=np.float64) - (n - 1) / 2.0
    s = sigma * n
    f = np.exp(-j * j / (2 * s * s))
    return _normalize(f)

def _init_taper(n, p=1.5):
    j = np.arange(n, dtype=np.float64)
    x = np.sin(np.pi * (j + 0.5) / n)
    f = x ** p
    return _normalize(f)

def _mirror_descent(f0, max_iters=3000, eta0=0.5, thresh_start=0.9, thresh_end=0.5,
                    momentum=0.5, time_budget=300.0, label=""):
    """
    Mirror descent with adaptive threshold and momentum.
    Minimizes max(conv(f,f)) subject to f >= 0, sum(f) = 1.
    """
    f = np.asarray(f0, dtype=np.float64).copy()
    f = _normalize(f)
    if f is None:
        return f0.copy(), float('inf')
    n = len(f)
    best_f = f.copy()
    best_score = _score(f)
    start = time.monotonic()
    
    if label:
        print(f"  MD [{label}]: n={n}, start score={best_score:.8f}")
    
    p_mom = np.zeros(n)
    
    for it in range(max_iters):
        if time.monotonic() - start > time_budget:
            break
        
        R = _fft_convolve(f)
        M = float(np.max(R))
        if M <= 1e-30:
            break
        
        # Adaptive threshold
        progress = it / max(1, max_iters - 1)
        thresh = thresh_start + (thresh_end - thresh_start) * progress
        active = np.where(R >= thresh * M)[0]
        if len(active) == 0:
            active = np.array([int(np.argmax(R))])
        
        # Pressure vector: p[j] = (1/|K|) sum_{k in K} 2*f[k-j] (valid range)
        p = np.zeros(n)
        for k in active:
            j_lo = max(0, k - n + 1)
            j_hi = min(n - 1, k)
            if j_hi >= j_lo:
                # f[k-j] for j in [j_lo, j_hi]
                idx = k - np.arange(j_lo, j_hi + 1)
                p[j_lo:j_hi+1] += 2.0 * f[idx]
        p /= len(active)
        
        # Momentum
        p_mom = momentum * p_mom + (1 - momentum) * p
        
        # Step size: eta = eta0 / M (scales with the current peak)
        eta = eta0 / M
        log_update = np.exp(-eta * p_mom)
        log_update = np.clip(log_update, np.exp(-20), np.exp(20))
        
        f = f * log_update
        f = _normalize(f)
        if f is None:
            f = np.ones(n) / n
            p_mom = np.zeros(n)
            continue
        
        s = _score(f)
        if s < best_score:
            best_score = s
            best_f = f.copy()
        
        if (it + 1) % 1000 == 0:
            print(f"    MD [{label}] it={it+1}: best={best_score:.8f}, M={M:.6f}, |K|={len(active)}")
    
    return best_f, best_score

"""
Optimizes non-negative step function heights to minimize 2n*max(conv(f,f))/(sum(f))^2.
Strategy: multi-start mirror descent with adaptive threshold + momentum,
followed by analytic coordinate descent polish. Explores multiple lengths.
"""

import numpy as np
import time


def _fft_convolve(f):
    n = len(f)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    F = np.fft.rfft(f, size)
    out = np.fft.irfft(F * F, size)
    return out[:2 * n - 1]


def _score(f):
    n = len(f)
    S = float(np.sum(f))
    if S < 0.01:
        return float('inf')
    R = _fft_convolve(f)
    M = float(np.max(R))
    return 2.0 * n * M / (S * S)


def _normalize(f):
    f = np.maximum(f, 0.0)
    s = float(np.sum(f))
    if s < 1e-30:
        return None
    return f / s


def _init_cosine(n, p=2.0):
    j = np.arange(n, dtype=np.float64)
    f = np.cos(np.pi * (j + 0.5) / n) ** p
    f = np.maximum(f, 0.0)
    return _normalize(f)


def _init_gaussian(n, sigma=0.3):
    j = np.arange(n, dtype=np.float64) - (n - 1) / 2.0
    s = sigma * n
    f = np.exp(-j * j / (2.0 * s * s))
    return _normalize(f)


def _init_taper(n, p=1.5):
    j = np.arange(n, dtype=np.float64)
    x = np.sin(np.pi * (j + 0.5) / n)
    f = x ** p
    return _normalize(f)


def _mirror_descent(f0, max_iters=3000, eta0=0.5, thresh_start=0.9, thresh_end=0.5,
                    momentum=0.3, time_budget=60.0, label=""):
    f = np.asarray(f0, dtype=np.float64).copy()
    f = _normalize(f)
    if f is None:
        return f0.copy(), float('inf')
    n = len(f)
    best_f = f.copy()
    best_score = _score(f)
    start = time.monotonic()
    p_mom = np.zeros(n)

    for it in range(max_iters):
        if time.monotonic() - start > time_budget:
            break

        R = _fft_convolve(f)
        M = float(np.max(R))
        if M <= 1e-30:
            break

        progress = min(1.0, it / max(1, max_iters * 0.8))
        thresh = thresh_start + (thresh_end - thresh_start) * progress
        active = np.where(R >= thresh * M)[0]
        if len(active) == 0:
            active = np.array([int(np.argmax(R))])

        p = np.zeros(n)
        for k in active:
            j_lo = max(0, k - n + 1)
            j_hi = min(n - 1, k)
            if j_hi >= j_lo:
                p[j_lo:j_hi + 1] += 2.0 * f[k - j_hi:k - j_lo + 1]
        p /= len(active)

        p_mom = momentum * p_mom + (1.0 - momentum) * p

        eta = eta0 / M
        f = f * np.exp(np.clip(-eta * p_mom, -20.0, 20.0))
        f = _normalize(f)
        if f is None:
            f = np.ones(n) / n
            p_mom[:] = 0.0
            continue

        s = _score(f)
        if s < best_score:
            best_score = s
            best_f = f.copy()

        if (it + 1) % 2000 == 0:
            print(f"    MD [{label}] it={it+1}: best={best_score:.8f}")

    return best_f, best_score


def _coord_descent(f_in, n_passes=3, time_budget=120.0, label=""):
    f = np.asarray(f_in, dtype=np.float64).copy()
    best_score = _score(f)
    best_f = f.copy()
    n = len(f)
    start = time.monotonic()

    for pass_num in range(n_passes):
        if time.monotonic() - start > time_budget:
            break

        order = np.random.permutation(n)
        improved = False

        for i in order:
            if time.monotonic() - start > time_budget:
                break

            x_old = f[i]
            if x_old < 1e-14:
                continue

            f[i] = 0.0
            R_base = _fft_convolve(f)
            S_0 = float(np.sum(f))
            if S_0 < 1e-10:
                f[i] = x_old
                continue

            # Compute c[k] = 2*f[k-i] for valid k
            c = np.zeros(2 * n - 1)
            k_lo = max(0, i)
            k_hi = min(2 * n - 2, i + n - 1)
            if k_hi >= k_lo:
                c[k_lo:k_hi + 1] = 2.0 * f[k_lo - i:k_hi - i + 1]

            q = np.zeros(2 * n - 1)
            if 0 <= 2 * i < 2 * n - 1:
                q[2 * i] = 1.0

            # Optimize x
            x = x_old
            for _ in range(30):
                R_curr = R_base + c * x + q * x * x
                M_curr = float(np.max(R_curr))
                S_curr = S_0 + x
                if S_curr < 1e-10:
                    break
                phi = 2.0 * n * M_curr / (S_curr * S_curr)

                if phi < best_score - 1e-12:
                    best_score = phi
                    best_f = best_f.copy()
                    best_f[i] = x
                    improved = True

                k_star = int(np.argmax(R_curr))
                if q[k_star] > 0:
                    M_prime = 2.0 * x
                elif c[k_star] > 0:
                    M_prime = c[k_star]
                else:
                    M_prime = 0.0

                dphi = 2.0 * n * (M_prime * S_curr - 2.0 * M_curr) / (S_curr ** 3)
                if abs(dphi) < 1e-12:
                    break

                direction = -1.0 if dphi > 0 else 1.0
                step = max(x * 0.5, 1e-6)
                improved_step = False
                for _bt in range(15):
                    x_new = x + direction * step
                    if x_new < 0:
                        x_new = 0.0
                    R_new = R_base + c * x_new + q * x_new * x_new
                    M_new = float(np.max(R_new))
                    S_new = S_0 + x_new
                    if S_new < 1e-10:
                        break
                    phi_new = 2.0 * n * M_new / (S_new * S_new)
                    if phi_new < phi - 1e-14:
                        x = x_new
                        improved_step = True
                        break
                    step *= 0.5

                if not improved_step:
                    break

            f[i] = max(x, 0.0)

        if not improved:
            break
        f = _normalize(f)
        if f is None:
            break

    return best_f, best_score
