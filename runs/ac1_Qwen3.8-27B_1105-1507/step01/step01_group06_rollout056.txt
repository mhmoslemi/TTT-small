"""
Minimizes J(a) = 2*n*max(convolve(a,a))/(sum(a))^2 using:
1. Log-sum-exp gradient descent with lambda annealing (main workhorse)
2. LP best-response for structured large-step improvements
3. Spectral phase-retrieval style flattening
4. Multi-length search (512 to 4096) with diverse initializations
"""

import numpy as np
from scipy.fft import rfft, irfft
from scipy.optimize import linprog
import time


def fast_conv(a):
    """Full self-convolution via FFT, O(n log n)."""
    n = len(a)
    s = 1
    while s < 2 * n - 1:
        s <<= 1
    fa = rfft(a, s)
    return irfft(fa * fa, s)[:2 * n - 1]


def compute_logsumexp_gradient(a, lam):
    """
    Compute gradient of f(a) = (1/lam)*log(sum_k exp(lam*b[k]))
    where b = convolve(a, a).
    Returns (gradient, b, f_value).
    """
    n = len(a)
    b = fast_conv(a)
    b_max = b.max()
    shifted = lam * (b - b_max)
    exp_shifted = np.exp(shifted)
    p = exp_shifted / exp_shifted.sum()
    f_val = (b_max + np.log(exp_shifted.sum()) / lam)
    
    # Gradient: grad[i] = 2 * sum_k p[k] * a[k-i]
    s = 1
    while s < 2 * n - 1:
        s <<= 1
    fp = rfft(p, s)
    fa = rfft(a, s)
    corr = irfft(fp * np.conj(fa), s)
    grad = 2.0 * corr[:n]
    
    return grad, b, f_val


def build_A(a):
    """Build constraint matrix for LP: A[k,j] = a[k-j] if valid."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for k in range(2 * n - 1):
        jl = max(0, k - n + 1)
        jh = min(n - 1, k)
        if jl <= jh:
            A[k, jl:jh + 1] = a[k - jh:k - jl + 1][::-1]
    return A


def lp_best_response(a, M):
    """LP: max sum(g) s.t. (a*g)_k <= M for all k, g >= 0."""
    n = len(a)
    c = -np.ones(n)
    A = build_A(a)
    r = linprog(c, A_ub=A, b_ub=np.full(2 * n - 1, M),
               bounds=[(0, None)] * n, method='highs')
    if r.success:
        g = r.x
        if g.sum() > 1e-10:
            return g
    return None


def try_update(a, best, best_score):
    """Try to update best if a is better."""
    a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
    if a.sum() < 0.01:
        return best, best_score
    score = float(evaluate_sequence(a.tolist()))
    if np.isfinite(score) and score < best_score:
        return a.copy(), score
    return best, best_score


def gradient_descent_logsumexp(a, deadline, max_iters, lam_start, lam_end, lr_start, best, best_score):
    """Run log-sum-exp gradient descent with annealing."""
    n = len(a)
    a = a.copy()
    a = np.clip(a, 0.001, None)
    a /= a.sum()
    
    best, best_score = try_update(a, best, best_score)
    
    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        
        # Anneal lambda
        frac = it / max(1, max_iters - 1)
        lam = lam_start + (lam_end - lam_start) * frac
        
        grad, b, f_val = compute_logsumexp_gradient(a, lam)
        
        # Adaptive learning rate
        grad_norm = np.linalg.norm(grad)
        if grad_norm < 1e-15:
            break
        lr = lr_start * (1.0 - 0.5 * frac)
        
        a_new = a - lr * grad
        a_new = np.maximum(a_new, 0.0)
        s_sum = a_new.sum()
        if s_sum < 1e-8:
            # Backtrack
            a_new = a - 0.5 * lr * grad
            a_new = np.maximum(a_new, 0.0)
            s_sum = a_new.sum()
            if s_sum < 1e-8:
                break
        a_new /= s_sum
        a = a_new
        
        if (it + 1) % 200 == 0:
            best, best_score = try_update(a, best, best_score)
    
    best, best_score = try_update(a, best, best_score)
    return a, best, best_score


def lp_polish(a, deadline, max_iters, best, best_score):
    """LP best-response polishing."""
    n = len(a)
    a = a.copy()
    a = np.clip(a, 0.001, None)
    a /= a.sum()
    best, best_score = try_update(a, best, best_score)
    
    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        b = fast_conv(a)
        M = float(b.max())
        g = lp_best_response(a, M)
        if g is None:
            break
        g = np.clip(g, 0.0, None)
        if g.sum() < 1e-10:

"""
Minimizes J(a) = 2*n*max(convolve(a,a))/(sum(a))^2 using:
1. Power-norm gradient descent (L2, L4) with momentum for fast convergence
2. Log-sum-exp gradient descent with lambda annealing for peak-focused refinement
3. LP best-response for structured large-step improvements
4. Multi-length search (512-4096) with diverse initializations
Key insight: optimal sequences are U-shaped (high at edges, low in middle)
to flatten self-convolution.
"""

import numpy as np
from scipy.fft import rfft, irfft
from scipy.optimize import linprog
import time


def fast_conv(a):
    """Full self-convolution via FFT."""
    n = len(a)
    s = 1
    while s < 2 * n - 1:
        s <<= 1
    fa = rfft(a, s)
    return irfft(fa * fa, s)[:2 * n - 1]


def cross_corr_weights(weights, a):
    """Compute corr[i] = sum_k weights[k] * a[k-i] for i in [0, n-1]."""
    n = len(a)
    s = 1
    while s < 2 * n - 1:
        s <<= 1
    fw = rfft(weights, s)
    fa = rfft(a, s)
    corr = irfft(fw * np.conj(fa), s)
    return corr[:n]


def power_gradient(a, p):
    """Gradient of sum_k b[k]^p where b = conv(a,a). Returns (grad, b)."""
    n = len(a)
    b = fast_conv(a)
    # Avoid issues with b[k]=0 and p-1 < 1
    if p == 2:
        weights = b.copy()
    else:
        weights = np.maximum(b, 1e-30) ** (p - 1)
    corr = cross_corr_weights(weights, a)
    grad = 2.0 * p * corr
    return grad, b


def logsumexp_gradient(a, lam):
    """Gradient of (1/lam)*log(sum exp(lam*b[k])). Returns (grad, b)."""
    n = len(a)
    b = fast_conv(a)
    b_max = b.max()
    shifted = lam * (b - b_max)
    exp_shifted = np.exp(shifted)
    p = exp_shifted / exp_shifted.sum()
    corr = cross_corr_weights(p, a)
    grad = 2.0 * corr
    return grad, b


def project_simplex(a):
    """Project a onto {x >= 0, sum(x) = 1}."""
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(len(a)) / len(a)
    return a / s


def run_power_gradient_descent(a, deadline, max_iters, p, lr, momentum):
    """Power-norm gradient descent with momentum."""
    n = len(a)
    a = project_simplex(a.copy())
    v = np.zeros(n)
    
    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        grad, b = power_gradient(a, p)
        # Project gradient (remove constraint component)
        grad = grad - grad.mean()
        # Momentum update
        v = momentum * v + (1.0 - momentum) * grad
        a_new = a - lr * v
        a_new = project_simplex(a_new)
        a = a_new
    
    return a


def run_logsumexp_gradient_descent(a, deadline, max_iters, lam_start, lam_end, lr, momentum):
    """Log-sum-exp gradient descent with lambda annealing and momentum."""
    n = len(a)
    a = project_simplex(a.copy())
    v = np.zeros(n)
    
    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        frac = it / max(1, max_iters - 1)
        lam = lam_start + (lam_end - lam_start) * frac
        
        grad, b = logsumexp_gradient(a, lam)
        grad = grad - grad.mean()
        v = momentum * v + (1.0 - momentum) * grad
        a_new = a - lr * v
        a_new = project_simplex(a_new)
        a = a_new
    
    return a
