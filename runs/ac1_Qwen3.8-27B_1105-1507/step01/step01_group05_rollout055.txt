"""
Hybrid optimization for minimizing 2*n*max(convolve(a,a))/sum(a)^2.
Combines LP-based direction finding, peak deflation, adaptive local search,
and diverse initializations including structured asymmetric patterns.
"""

import numpy as np
from scipy.optimize import linprog
import time


def compute_score_fast(f):
    """Compute score = 2*n*max(conv)/sum^2 for a sequence."""
    n = len(f)
    S = f.sum()
    if S < 1e-15:
        return np.inf
    conv = np.convolve(f, f)
    M = conv.max()
    return 2.0 * n * M / (S * S)


def build_conv_matrix(f):
    """Build matrix A such that A @ b = np.convolve(f, b, mode='full')."""
    n = len(f)
    A = np.zeros((2 * n - 1, n))
    for k in range(2 * n - 1):
        for j in range(max(0, k - n + 1), min(n, k + 1)):
            A[k, j] = f[k - j]
    return A


def create_initializations(n, rng, parent=None):
    """Create diverse initial sequences of length n, normalized to sum=1."""
    inits = []
    
    # Uniform with noise
    f = np.ones(n) * (1 + 0.1 * rng.standard_normal(n))
    f = np.clip(f, 0.01, None)
    inits.append(f / f.sum())
    
    # Pure random
    f = rng.random(n) + 0.01
    inits.append(f / f.sum())
    
    # Ramp (asymmetric - important for reducing center peak)
    f = np.arange(1, n + 1, dtype=np.float64)
    inits.append(f / f.sum())
    
    # Reverse ramp
    f = np.arange(n, 0, -1, dtype=np.float64)
    inits.append(f / f.sum())
    
    # V-shaped (high at ends, low in middle)
    mid = n // 2
    f = np.concatenate([np.linspace(1, 0.1, mid), np.linspace(0.1, 1, n - mid)])
    inits.append(f / f.sum())
    
    # Inverted V (low at ends, high in middle)
    f = np.concatenate([np.linspace(0.1, 1, mid), np.linspace(1, 0.1, n - mid)])
    inits.append(f / f.sum())
    
    # Bimodal (two bumps)
    centers = [n // 4, 3 * n // 4]
    f = np.zeros(n)
    for c in centers:
        f += np.exp(-0.5 * ((np.arange(n) - c) / (n / 8)) ** 2)
    f += 0.01
    inits.append(f / f.sum())
    
    # Exponential decay
    f = np.exp(-np.arange(n) / (n / 3))
    inits.append(f / f.sum())
    
    # Alternating pattern with noise
    f = np.ones(n)
    f[1::2] = 0.3
    f += 0.1 * rng.standard_normal(n)
    f = np.clip(f, 0.01, None)
    inits.append(f / f.sum())
    
    # Cosine bump
    f = np.cos(np.pi * np.arange(n) / n) ** 2 + 0.01
    inits.append(f / f.sum())

            p2 = np.clip(p2, 0, None)

def lp_direction(f, M, top_k=None):
    """Find LP direction: maximize sum(g) s.t. conv(f,g) <= M, g >= 0."""
    n = len(f)
    A = build_conv_matrix(f)
    c = -np.ones(n)
    
    if top_k is not None:
        # Use only top-k constraints for speed
        conv = np.convolve(f, f)
        top_indices = np.argsort(conv)[-top_k:][::-1]
        A_ub = A[top_indices, :]
        b_ub = np.full(len(top_indices), M)
    else:
        A_ub = A
        b_ub = np.full(2 * n - 1, M)
    
    bounds = [(0, None)] * n
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds,
                     method='highs', options={'presolve': True, 'time_limit': 5})
        if res.success and res.x is not None:
            return np.clip(res.x, 0, None)
    except Exception:
        pass
    return None


def peak_deflation(f, alpha=0.05, n_rounds=50):
    """Directly deflate the peak of conv(f,f)."""
    n = len(f)
    f = f.copy()
    
    for _ in range(n_rounds):
        conv = np.convolve(f, f)
        k_star = np.argmax(conv)
        M = conv[k_star]
        
        # Sensitivity: d(conv[k_star])/df[j] = f[k_star - j]
        sens = np.zeros(n)
        for j in range(n):
            k = k_star - j
            if 0 <= k < n:
                sens[j] = f[k]
        
        # Also compute sensitivity to second-highest peak for balance
        conv_sorted = np.sort(conv)[::-1]
        if len(conv_sorted) > 1:
            k2 = np.argsort(conv)[::-1][1]
            k2_val = conv[k2]
            if k2_val > 0.8 * M:
                sens2 = np.zeros(n)
                for j in range(n):
                    k = k2 - j
                    if 0 <= k < n:
                        sens2[j] = f[k]
                sens = sens + 0.5 * sens2
        
        # Deflate
        f = f - alpha * sens
        f = np.clip(f, 0, None)
        S = f.sum()
        if S < 1e-10:
            break
        f = f / S
    
    return f


def localized_search(f, rng, n_iter=3000, verbose=False):

        elif r < 0.8:

                for idx in top:
                    f_cand[idx] *= np.exp(rng.normal(0, base_scale * 0.5))
        else:
            # Gaussian noise
            f_cand += rng.normal(0, base_scale * 0.1, n)

            g = lp_direction(f, M, top_k=top_k)
            
            if g is not None and g.sum() > 1e-10:
                g = g / g.sum()  # Normalize to sum=1
                f_new = (1 - t) * f + t * g
                f_new = np.clip(f_new, 0, None)
                S_new = f_new.sum()
                if S_new < 1e-10:
                    t *= 0.5
                    continue
                f_new = f_new / S_new
                
                new_score = compute_score_fast(f_new)
                if new_score < best_score - 1e-12:
                    best_f = f_new.copy()
                    best_score = new_score
                    f = f_new
                    stall_count = 0
                    # Increase step size slightly on success
                    t = min(0.5, t * 1.05)
                    if verbose and it % 10 == 0:
                        print(f"    it={it}, score={best_score:.6f}, t={t:.3f}, peaks_at={k_star}")
                else:
                    t *= 0.7
                    stall_count += 1
            else:
                stall_count += 5
                t *= 0.5
            
            # Switch to deflation if stuck
            if stall_count > 30:
                phase = "deflate"
                stall_count = 0
                if verbose:
                    print(f"    Switching to deflation phase")
        
        elif phase == "deflate":
            # Peak deflation
            f_def = peak_deflation(f, alpha=0.03, n_rounds=20)
            score_def = compute_score_fast(f_def)
            if score_def < best_score:
                best_f = f_def.copy()
                best_score = score_def
                f = f_def
                phase = "lp"
                t = 0.2
                if verbose:
                    print(f"    Deflation improved: {best_score:.6f}")
            else:
                # Try local search
                phase = "local"
        
        elif phase == "local":
            # Local search for a while
            local_time = min(30, remaining * 0.2)
            n_iter = int(local_time * 200)
            f_ls, score_ls = localized_search(f, rng, n_iter=n_iter, verbose=False)
            if score_ls < best_score:
                best_f = f_ls.copy()
                best_score = score_ls
                f = f_ls
                if verbose:
                    print(f"    Local search improved: {best_score:.6f}")
            phase = "lp"
            t = 0.2
            stall_count = 0
        
        # Periodic random escape
        if it > 0 and it % 100 == 0 and stall_count > 20:
            f_esc = f * (1 + 0.5 * rng.standard_normal(n))
            f_esc = np.clip(f_esc, 0, None)
            S_esc = f_esc.sum()
            if S_esc > 1e-10:
                f_esc = f_esc / S_esc
                score_esc = compute_score_fast(f_esc)
                if score_esc < best_score * 1.1:  # Accept if not too much worse
                    f = f_esc
                    stall_count = 0
                    t = 0.3
        
        it += 1
    
    return best_f, best_score


def try_different_lengths(f_best, score_best, remaining_time, rng, verbose=True):
    """Try to improve by extending or modifying the best sequence."""
    n = len(f_best)
    
    # Try padding with zeros
    for pad in [1, 2, 4, 8]:
        if remaining_time < 30:
            break
        n_new = n + pad
        f_pad = np.zeros(n_new)
        f_pad[:n] = f_best
        f_pad = f_pad / f_pad.sum()
        
        time_here = min(60, remaining_time / 2)
        f_opt, score = optimize_sequence(f_pad, time_here, rng, verbose=False)
        if score < score_best:
            if verbose:
                print(f"  Padding by {pad} improved: {score:.6f}")
            f_best = f_opt
            score_best = score
            remaining_time = time.monotonic() - (time.monotonic() - remaining_time)
    
    return f_best, score_best


def propose_candidate(seed=42, budget_s=1500, **kwargs):
    """
    Main search function. Uses hybrid approach combining:
    - Diverse initializations (including asymmetric patterns)
    - LP-based direction finding with adaptive step size
    - Peak deflation for direct peak reduction
    - Adaptive local search
    - Multiple sequence lengths
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + budget_s - 15.0
    
    best_overall = None
    best_score_overall = np.inf
    
    # Get parent sequence
    parent = None
    try:
        parent = np.array(height_sequence_1, dtype=np.float64)
        if parent.ndim != 1 or len(parent) == 0 or not np.isfinite(parent).all():
            parent = None
    except Exception:
        parent = None
    
    if parent is not None:
        parent = np.clip(parent, 0, 1000)
        if parent.sum() < 0.01:
            parent = None
    
    print(f"Parent available: {parent is not None}")
    if parent is not None:
        print(f"Parent length: {len(parent)}, score: {compute_score_fast(parent):.6f}")
    
    # Define configurations: (name, n, init_type)
    configs = []
    
    # Core lengths
    lengths = [128, 192, 256, 384, 512]
    
    # If parent is available, prioritize its length
    if parent is not None:
        n_parent = len(parent)
        if n_parent not in lengths:
            lengths.insert(0, n_parent)
    
    for n in lengths:
        configs.append((f"diverse_{n}", n))
    
    # Allocate time
    total_time = budget_s - 30  # Leave 30s for final polish
    n_configs = len(configs)
    
    # Start with shorter sequences (faster), then longer
    # Allocate time proportional to potential benefit
    time_allocation = {}
    for i, (name, n) in enumerate(configs):
        if n <= 256:
            time_allocation[name] = total_time * 0.2
        elif n <= 512:
            time_allocation[name] = total_time * 0.15
        else:
            time_allocation[name] = total_time * 0.1
    
    print(f"Configs: {[(name, n) for name, n in configs]}")
    print(f"Target: 1.5027")
    
    for name, n in configs:
        if time.monotonic() > deadline:
            break
        
        time_here = min(time_allocation.get(name, 60), deadline - time.monotonic() - 20)
        if time_here < 20:
            break
        
        print(f"\n[{name}] n={n}, budget={time_here:.0f}s")
        
        # Create initializations
        inits = create_initializations(n, rng, parent if parent is not None and len(parent) == n else None)
        
        # If parent has different length, try to adapt
        if parent is not None and len(parent) != n:
            # Interpolate parent to new length
            from scipy.interpolate import interp1d
            x_old = np.linspace(0, 1, len(parent))
            x_new = np.linspace(0, 1, n)
            f_interp = interp1d(x_old, parent, kind='cubic')(x_new)
            f_interp = np.clip(f_interp, 0, None)
            if f_interp.sum() > 0.01:
                inits.append(f_interp / f_interp.sum())
        
        # Evaluate all inits and pick the best 2-3 to optimize
        scored_inits = []
        for i, f_init in enumerate(inits):
            s = compute_score_fast(f_init)
            scored_inits.append((s, i, f_init))
        
        scored_inits.sort(key=lambda x: x[0])
        
        if verbose:
            print(f"  Init scores: {[f'{s:.4f}' for s, _, _ in scored_inits[:5]]}")
        
        # Optimize top few initializations
        n_top = min(3, len(scored_inits))
        for rank in range(n_top):
            if time.monotonic() > deadline:
                break
            
            s_init, _, f_init = scored_inits[rank]
            time_each = (deadline - time.monotonic() - 20) / max(1, n_top - rank)
            time_each = min(time_each, 120)
            if time_each < 10:
                break
            
            verbose_this = (rank == 0)
            f_opt, score = optimize_sequence(f_init, time_each, rng, verbose=verbose_this)
            
            # Local search polish
            remaining = deadline - time.monotonic()
            if remaining > 15 and np.isfinite(score):
                n_ls_iter = int(min(5000, remaining * 50))
                f_pol, score_pol = localized_search(f_opt, rng, n_iter=n_ls_iter, verbose=False)
                if score_pol < score:
                    f_opt = f_pol
                    score = score_pol
            
            if np.isfinite(score) and score < best_score_overall:
                best_score_overall = score
                best_overall = f_opt.copy()
                print(f"  ** New best from {name} init {rank}: {score:.6f} **")
                
                # If we beat the target, spend remaining time polishing
                if score < 1.5027:
                    print(f"  *** BEAT THE TARGET! Polishing... ***")
                    remaining = deadline - time.monotonic()
                    if remaining > 30:
                        f_pol, score_pol = localized_search(f_opt, rng, n_iter=int(remaining * 100), verbose=True)
                        if score_pol < score:
                            best_overall = f_pol
                            best_score_overall = score_pol
                            print(f"  Polished: {score_pol:.6f}")
                    break
        
        if score < 1.5027:
            break
    
    # Final phase: intensive polish of the best solution
    if best_overall is not None:
        remaining = deadline - time.monotonic()
        if remaining > 20:
            print(f"\nFinal polish phase: {remaining:.0f}s remaining")
            n = len(best_overall)
            
            # Multiple rounds of local search with different parameters
            for round_num in range(3):
                time_left = deadline - time.monotonic()
                if time_left < 10:
                    break
                
                n_iter = int(min(10000, time_left * 50))
                f_pol, score_pol = localized_search(best_overall, rng, n_iter=n_iter, verbose=False)
                if score_pol < best_score_overall:
                    best_overall = f_pol
                    best_score_overall = score_pol
                    print(f"  Polish round {round_num}: {best_score_overall:.6f}")
                else:
                    # Try a different perturbation scale
                    f_pert = best_overall * (1 + 0.1 * rng.standard_normal(n))
                    f_pert = np.clip(f_pert, 0, None)
                    if f_pert.sum() > 1e-10:
                        f_pert = f_pert / f_pert.sum()
                        n_iter2 = int(min(5000, (deadline - time.monotonic()) * 30))
                        f_pol2, score_pol2 = localized_search(f_pert, rng, n_iter=n_iter2, verbose=False)
                        if score_pol2 < best_score_overall:
                            best_overall = f_pol2
                            best_score_overall = score_pol2
                            print(f"  Perturbed polish: {best_score_overall:.6f}")
            
            # Try peak deflation on the final solution
            time_left = deadline - time.monotonic()
            if time_left > 5:
                f_def = peak_deflation(best_overall, alpha=0.02, n_rounds=100)
                score_def = compute_score_fast(f_def)
                if score_def < best_score_overall:
                    best_overall = f_def
                    best_score_overall = score_def
                    print(f"  Deflation improved: {best_score_overall:.6f}")
    
    # Verify and return
    if best_overall is not None:
        best_overall = np.clip(best_overall, 0, 1000)
        final_list = best_overall.tolist()
        final_score = evaluate_sequence(final_list)
        print(f"\nFinal verified score: {final_score:.8f}")
        print(f"Sequence length: {len(final_list)}")
        print(f"Target: 1.5027")
        return final_list
    else:
        # Fallback
        print("Fallback to uniform sequence")
        return [1.0] * 256
